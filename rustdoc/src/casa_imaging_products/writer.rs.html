3//! The CASA image writer.
4//!
5//! Each member is written to a private staging directory beside its target.
6//! Publishing then replaces each target on its own: the staged image is
7//! renamed into place, or exchanged with an existing image in one call
8//! (`renamex_np` with `RENAME_SWAP` on macOS, `renameat2` with
9//! `RENAME_EXCHANGE` on Linux), and both directories are synced. An open
10//! after the replacement sees the whole new image, and one before it the
11//! whole old image. A member already replaced stays valid if a later one
12//! fails; the set is then incomplete and the run must be repeated.
13//!
14//! Limits:
15//! - **A held reader keeps only what it has opened.** casacore opens some
16//!   storage, such as a pixel mask, by path when it is first read. A reader
17//!   that holds the old image across a replacement and first reads such
18//!   storage afterwards reads the new image's, or fails if the new image has
19//!   none. Reopen an image after a run replaces it. CASA's own in-place
20//!   overwrite gives a held reader no coherence at all.
21//! - **Exchange needs local filesystem support.** A filesystem without an
22//!   atomic exchange (SMB, some NFS) refuses a run into an existing image,
23//!   with [`ProductsError::Io`].
24//! - **A killed run leaves its staging.** A failed generation discards what
25//!   it staged; a killed process leaves `.<name>.casa-rs-stage-*`
26//!   directories beside the targets, which are safe to delete.
27

93/// Keep the `logtable` of the image at `target` in the image about to replace
94/// it: a run on an existing image appends to its history, as CASA's does,
95/// which updates the image in place.
96fn carry_history(target: &Path, staging: &Path) -> std::io::Result<()> {
97    let earlier = target.join("logtable");
98    if !earlier.is_dir() {
99        return Ok(());
100    }
101    let staged = staging.join("logtable");
102    fs::remove_dir_all(&staged)?;
103    copy_directory(&earlier, &staged)
104}
105
106fn copy_directory(from: &Path, to: &Path) -> std::io::Result<()> {
107    fs::create_dir(to)?;
108    for entry in fs::read_dir(from)? {
109        let entry = entry?;
110        let target = to.join(entry.file_name());
111        if entry.file_type()?.is_dir() {
112            copy_directory(&entry.path(), &target)?;
113        } else {
114            fs::copy(entry.path(), target)?;
115        }
116    }
117    Ok(())
118}

120impl CasaImageWriter {
121    /// Write the main domain's members under `base` with `coordinates`.
122    #[must_use]
123    pub fn new(base: PathBuf, coordinates: CoordinateSystem) -> Self {
124        Self::for_domains([CasaImageDomainOutput::new(
125            ImageDomainRole::Main,
126            base,
127            coordinates,
128        )])
129    }
130
131    /// Write every compiled image domain's members under its own prefix.
132    ///
133    /// # Panics
134    ///
135    /// If two outputs share a role or a prefix, or none is the main domain.
136    /// Compiling a request refuses domains that name one output, and always
137    /// compiles the main domain.
138    #[must_use]
139    pub fn for_domains(domains: impl IntoIterator<Item = CasaImageDomainOutput>) -> Self {
140        let mut outputs = BTreeMap::new();
141        let mut roots = BTreeSet::new();
142        for output in domains {
143            let new_root = roots.insert(output.base.clone());
144            let new_role = outputs.insert(output.role.clone(), output).is_none();
145            assert!(
146                new_root && new_role,
147                "compiled image domains have distinct roles and output roots"
148            );
149        }
150        assert!(
151            outputs.contains_key(&ImageDomainRole::Main),
152            "compiled image domains include the main domain"
153        );
154        Self {
155            domains: Arc::new(outputs),
156            alternate_coordinates: None,
157            mem_alpha: None,
158            staged: Arc::new(Mutex::new(Vec::new())),
159        }
160    }
161
162    /// Bind alternate coordinates to the same output domains and staging.
163    ///
164    /// Product families with different coordinate systems may be generated
165    /// sequentially with the returned writer and this writer. Publishing from
166    /// either publishes their combined staging; dropping one leaves the
167    /// staging alive while the other remains. A failed generation discards
168    /// only the members it staged.
169    ///
170    /// # Panics
171    ///
172    /// If the coordinates do not name each bound domain exactly once.
173    #[must_use]
174    pub fn with_coordinates(&self, coordinates: Vec<(ImageDomainRole, CoordinateSystem)>) -> Self {
175        assert_eq!(
176            coordinates.len(),
177            self.domains.len(),
178            "coordinates name every domain"
179        );
180        for (index, (role, _)) in coordinates.iter().enumerate() {
181            assert!(self.domains.contains_key(role), "a bound domain");
182            assert!(
183                coordinates[..index].iter().all(|(prior, _)| prior != role),
184                "each domain is named once"
185            );
186        }
187        Self {
188            domains: Arc::clone(&self.domains),
189            alternate_coordinates: Some(coordinates),
190            mem_alpha: self.mem_alpha,
191            staged: Arc::clone(&self.staged),
192        }
193    }
194
195    /// The coordinate binding of this family, or the original binding.
196    fn coordinates_for(&self, role: &ImageDomainRole) -> &CoordinateSystem {
197        match &self.alternate_coordinates {
198            Some(coordinates) => {
199                &coordinates
200                    .iter()
201                    .find(|(candidate, _)| candidate == role)
202                    .expect("coordinates name every bound domain")
203                    .1
204            }
205            None => &self.output(role).coordinates,
206        }
207    }

212    /// # Errors

216        if !alpha.is_finite() {

229    /// Bytes the writer holds while it writes `planned` a window at a time:
230    /// the image's tile cache and serialization buffers, the pixel mask's
231    /// cache (opened with the image's budget, so up to four windows of
232    /// validity), its coordinates and beam metadata, and the record of each
233    /// staged member kept until [`Self::publish`].
234    pub(crate) fn residency_bytes(
235        &self,
236        planned: &PlannedContinuumGeneration,
237        window_payload_bytes: u64,
238        window_validity_bytes: u64,
239    ) -> Result<u64, ProductsError> {
240        // The open image's table, data-manager and record metadata. An 8×8
241        // PB-only member with empty coordinates measured 46 kB live against
242        // 18 kB charged with a 4 kB envelope (review of #701).
243        const IMAGE_ENVELOPE_BYTES: u64 = 64 * 1024;
244        const STAGED_MEMBER_RECORD_BYTES: u64 = 512;
245        let overflow = || ProductsError::ResourceDemandOverflow("CASA image writer");
246        let coordinate_bytes = self.domains.keys().try_fold(0_u64, |maximum, role| {
247            let bytes = self
248                .coordinates_for(role)
249                .to_record()
250                .retained_heap_bytes()
251                .and_then(|bytes| bytes.checked_mul(4))
252                .and_then(|bytes| u64::try_from(bytes).ok())
253                .ok_or_else(overflow)?;
254            Ok::<_, ProductsError>(maximum.max(bytes))
255        })?;
256        let beam_bytes = planned
257            .members()
258            .iter()
259            .try_fold(0_u64, |maximum, member| {
260                let count = if member.storage().attach_beam() {
261                    member
262                        .axes()
263                        .spectral()
264                        .output_channels()
265                        .checked_mul(member.axes().polarization().len())
266                        .ok_or_else(overflow)?
267                } else {
268                    0
269                };
270                Ok::<_, ProductsError>(maximum.max(beam_metadata_residency_bytes(count)?))
271            })?;
272        let registry_bytes = planned.members().iter().try_fold(0_u64, |total, member| {
273            let base = self.output(member.axes().domain()).base.as_os_str().len();
274            let path_bytes = base
275                .checked_add(member.name().len())
276                .and_then(|bytes| bytes.checked_mul(2))
277                .and_then(|bytes| u64::try_from(bytes).ok())
278                .ok_or_else(overflow)?;
279            total
280                .checked_add(path_bytes)
281                .and_then(|bytes| bytes.checked_add(STAGED_MEMBER_RECORD_BYTES))
282                .ok_or_else(overflow)
283        })?;
284        window_payload_bytes
285            .checked_mul(4)
286            // The validity pack and its copy, and the mask's cache.
287            .and_then(|bytes| bytes.checked_add(window_validity_bytes.checked_mul(6)?))
288            .and_then(|bytes| bytes.checked_add(IMAGE_ENVELOPE_BYTES))
289            .and_then(|bytes| bytes.checked_add(coordinate_bytes))
290            .and_then(|bytes| bytes.checked_add(beam_bytes))
291            .and_then(|bytes| bytes.checked_add(registry_bytes))
292            .ok_or_else(overflow)
293    }

295    /// The number of members staged and not yet published.
296    pub(crate) fn staged_count(&self) -> usize {
297        self.staged
298            .lock()
299            .unwrap_or_else(PoisonError::into_inner)
300            .len()
301    }
302
303    /// Discard the members staged after the first `kept`, deleting their
304    /// staging: a failed generation publishes none of its members.
305    pub(crate) fn discard_staged_after(&self, kept: usize) {
306        self.staged
307            .lock()
308            .unwrap_or_else(PoisonError::into_inner)
309            .truncate(kept);
310    }

312    /// Replace every staged member's target, in generation order, telling
313    /// `published` each target once it is replaced.
314    ///
315    /// # Errors
316    ///
317    /// The first replacement that fails. Members replaced before it stay
318    /// replaced, and `published` has heard of each; the run must be
319    /// repeated.
320    pub fn publish(&self, mut published: impl FnMut(&Path)) -> Result<(), ProductsError> {
321        let staged =
322            std::mem::take(&mut *self.staged.lock().unwrap_or_else(PoisonError::into_inner));
323        for image in staged {
324            let failed = |source| ProductsError::Io {
325                operation: "replacing",
326                path: image.target.clone(),
327                source,
328            };
329            carry_history(&image.target, &image.staging).map_err(failed)?;
330            publish_staged_image(&image.staging, &image.target).map_err(failed)?;
331            published(&image.target);
332        }
333        Ok(())
334    }

336    /// Create one member's image in private staging beside its target, with
337    /// its units, beams and metadata.
338    pub(crate) fn begin(
339        &self,
340        member: &PlannedMember,
341        layout: ProductWindowLayout,
342        beams: &[Option<RestoringBeam>],
343        uses_weight_image: bool,
344    ) -> Result<MemberImage<'_>, ProductsError> {
345        let started = Instant::now();
346        let output = self.output(member.axes().domain());
347        let mut target = output.base.clone().into_os_string();
348        target.push(member.name());
349        let target = PathBuf::from(target);
350        let io = |operation, path: &Path| {
351            let path = path.to_path_buf();
352            move |source| ProductsError::Io {
353                operation,
354                path,
355                source,
356            }
357        };
358        let parent = target.parent().unwrap_or_else(|| Path::new("."));
359        fs::create_dir_all(parent).map_err(io("creating", parent))?;
360        let mut prefix = OsString::from(".");
361        prefix.push(target.file_name().expect("a product target names a file"));
362        prefix.push(".casa-rs-stage-");
363        let directory = tempfile::Builder::new()
364            .prefix(&prefix)
365            .tempdir_in(parent)
366            .map_err(io("staging", parent))?;
367        let staging = directory.path().join("image");
368        let image_error = |source| ProductsError::Image {
369            path: target.clone(),
370            source,
371        };
372        let mut tile = layout.shape();
373        tile[layout.spectral_axis()] = 1;
374        let mut image = PagedImage::<f32>::create_with_tile_shape_and_cache(
375            layout.shape().to_vec(),
376            tile.to_vec(),
377            self.coordinates_for(member.axes().domain()).clone(),
378            &staging,
379            layout.maximum_values() * size_of::<f32>(),
380        )
381        .map_err(image_error)?;
382        let storage = member.storage();
383        image
384            .set_units(storage.unit().map_or("", unit_label))
385            .map_err(image_error)?;
386        let role = role_label(member.role());
387        image
388            .set_image_info(&ImageInfo {
389                beam_set: if storage.attach_beam() {
390                    persisted_beam_set(beams, layout.shape()[layout.spectral_axis()])?
391                } else {
392                    ImageBeamSet::default()
393                },
394                // CASA's tclean leaves every product the default type.
395                image_type: ImageType::Intensity,
396                object_name: output.object_name.clone(),
397            })
398            .map_err(image_error)?;
399        // `SynthesisImager::createIMStore`'s, but its `memreq` and `memavail`,
400        // which describe CASA's run: one process, one chunk.
401        let mut miscinfo = vec![
402            RecordField::new(
403                "casars_imager_role",
404                Value::Scalar(ScalarValue::String(role.to_string())),
405            ),
406            RecordField::new(
407                "INSTRUME",
408                Value::Scalar(ScalarValue::String(output.telescope.clone())),
409            ),
410            RecordField::new("distance", Value::Scalar(ScalarValue::Float64(0.0))),
411            RecordField::new("mpiprocs", Value::Scalar(ScalarValue::Int32(1))),
412            RecordField::new("chnchnks", Value::Scalar(ScalarValue::Int32(1))),
413        ];
414        if matches!(member.role(), ProductRole::Model(_))
415            && let Some(alpha) = self.mem_alpha
416        {
417            miscinfo.push(RecordField::new(
418                "ALPHA",
419                Value::Scalar(ScalarValue::Float32(alpha)),
420            ));
421            miscinfo.push(RecordField::new(
422                "BETA",
423                Value::Scalar(ScalarValue::Float32(0.0)),
424            ));
425        }
426        // CASA tools read `.sumwt`'s `useweightimage` to learn whether the
427        // residual was divided by a weight image (`SIImageStore`
428        // `getUseWeightImage`).
429        if matches!(member.role(), ProductRole::SumWeights(_)) {
430            miscinfo.push(RecordField::new(
431                "useweightimage",
432                Value::Scalar(ScalarValue::Bool(uses_weight_image)),
433            ));
434        }
435        image
436            .set_misc_info(RecordValue::new(miscinfo))
437            .map_err(image_error)?;
438        Ok(MemberImage {
439            image,
440            explicit_mask: matches!(storage.pixel_mask(), ProductPixelMask::Explicit(_)),
441            writer: self,
442            staged: StagedImage {
443                _directory: directory,
444                staging,
445                target,
446            },
447            begin_nanos: started.elapsed().as_nanos(),
448            window_nanos: 0,
449        })
450    }
451}

453/// One member's image in staging, written a window at a time.
454pub(crate) struct MemberImage<'a> {
455    image: PagedImage<f32>,
456    explicit_mask: bool,
457    writer: &'a CasaImageWriter,
458    staged: StagedImage,
459    begin_nanos: u128,
460    window_nanos: u128,
461}
462
463impl MemberImage<'_> {
464    /// Write one generated window's values and, where the member stores
465    /// one, its pixel mask.
466    pub(crate) fn write(&mut self, window: ProductWindow) -> Result<(), ProductsError> {
467        let started = Instant::now();
468        let (start, shape, payload, validity) = window.into_parts();
469        let image_error = |source| ProductsError::Image {
470            path: self.staged.target.clone(),
471            source,
472        };
473        let data = ArrayD::from_shape_vec(IxDyn(&shape), payload)
474            .expect("a product window's payload fills its shape");
475        self.image
476            .put_slice_view(data.view(), &start)
477            .map_err(image_error)?;
478        if self.explicit_mask {
479            let mask = ArrayD::from_shape_vec(IxDyn(&shape), validity)
480                .expect("a product window's validity fills its shape");
481            self.image
482                .put_mask_slice("mask0", &mask, &start)
483                .map_err(image_error)?;
484        }
485        self.window_nanos += started.elapsed().as_nanos();
486        Ok(())
487    }
488
489    /// Save the image in staging and register it for [`CasaImageWriter::publish`].
490    pub(crate) fn finish(mut self) -> Result<(), ProductsError> {
491        let started = Instant::now();
492        let target = self.staged.target.clone();
493        let image_error = |source| ProductsError::Image {
494            path: target.clone(),
495            source,
496        };
497        if self.explicit_mask {
498            self.image.set_default_mask("mask0").map_err(image_error)?;
499        }
500        self.image.save().map_err(image_error)?;
501        if let Some(stats) = self.image.tiled_io_stats() {
502            tracing::debug!(
503                target = %self.staged.target.display(),
504                pack_nanos = stats.direct_tile_pack_ns,
505                swap_nanos = stats.direct_tile_swap_ns,
506                write_nanos = stats.direct_tile_write_ns,
507                direct_bytes = stats.direct_tile_write_bytes,
508                direct_calls = stats.direct_tile_write_calls,
509                "product image pixels written"
510            );
511        }
512        let Self {
513            image,
514            writer,
515            staged,
516            begin_nanos,
517            window_nanos,
518            ..
519        } = self;
520        drop(image);
521        tracing::debug!(
522            target = %staged.target.display(),
523            begin_nanos,
524            window_nanos,
525            finish_nanos = started.elapsed().as_nanos(),
526            "product image staged"
527        );
528        writer
529            .staged
530            .lock()
531            .unwrap_or_else(PoisonError::into_inner)
532            .push(staged);
533        Ok(())
534    }
535}

537fn beam_metadata_residency_bytes(count: usize) -> Result<u64, ProductsError> {
538    if count == 0 {
539        return Ok(0);
540    }
541    let overflow = || ProductsError::ResourceDemandOverflow("beam metadata");
542    let beam_record = GaussianBeam::new(1.0, 1.0, 0.0)
543        .to_record()
544        .retained_heap_bytes()
545        .ok_or_else(overflow)?;
546    let key_bytes = count.ilog10() as usize + 2;
547    // The growable outer keyword record has at most twice its live field count.
548    // Save overlaps owned keywords, the storage snapshot, table.dat keywords,
549    // and the encoded control buffer; the record heap bounds its encoded form.
550    let record = count
551        .checked_add(2)
552        .and_then(|fields| fields.checked_mul(2 * size_of::<RecordField>()))
553        .and_then(|bytes| {
554            bytes.checked_add(count.checked_mul(beam_record.checked_add(key_bytes)?)?)
555        })
556        .and_then(|bytes| bytes.checked_mul(4))
557        .ok_or_else(overflow)?;
558    let arrays = count
559        .checked_mul(
560            size_of::<RestoringBeam>() + size_of::<Vec<GaussianBeam>>() + size_of::<GaussianBeam>(),
561        )
562        .ok_or_else(overflow)?;
563    u64::try_from(record.checked_add(arrays).ok_or_else(overflow)?).map_err(|_| overflow())
564}

566/// The beam set casacore keeps for a member's beams, which are listed channel
567/// by channel with polarization fastest. As `SIImageStore::makeImageBeamSet`
568/// and `restore` store them, one beam is the image's beam and more are a
569/// channel × Stokes grid, equal or not: the shape `ImageInfo::checkBeamSet`
570/// requires of an image with `channels` channels before it opens it. A
571/// plane without a beam takes the largest one, as CASA's do. Beams that do
572/// not fill whole channels are refused.
573fn persisted_beam_set(
574    beams: &[Option<RestoringBeam>],
575    channels: usize,
576) -> Result<ImageBeamSet, ProductsError> {
577    if beams.is_empty() {
578        return Ok(ImageBeamSet::default());
579    }
580    if let [Some(only)] = beams {
581        return Ok(ImageBeamSet::new(gaussian_beam(*only)));
582    }
583    if channels == 0 || !beams.len().is_multiple_of(channels) {
584        return Err(ProductsError::BeamGrid {
585            beams: beams.len(),
586            channels,
587        });
588    }
589    let valid = beams.iter().flatten().copied().collect::<Vec<_>>();
590    let filler = valid
591        .iter()
592        .copied()
593        .max_by(|left, right| {
594            beam_area(*left)
595                .partial_cmp(&beam_area(*right))
596                .expect("validated beam areas are finite")
597        })
598        .map(gaussian_beam)
599        .unwrap_or_else(|| {
600            let one_microarcsecond_rad = std::f64::consts::PI / (180.0 * 3_600_000_000.0);
601            GaussianBeam::new(one_microarcsecond_rad, one_microarcsecond_rad, 0.0)
602        });
603    let polarizations = beams.len() / channels;
604    Ok(ImageBeamSet::from_grid(
605        beams
606            .chunks(polarizations)
607            .map(|channel| {
608                channel
609                    .iter()
610                    .map(|beam| beam.map_or(filler, gaussian_beam))
611                    .collect()
612            })
613            .collect(),
614    ))
615}
616
617fn gaussian_beam(beam: RestoringBeam) -> GaussianBeam {
618    GaussianBeam::new(
619        beam.major_fwhm_rad(),
620        beam.minor_fwhm_rad(),
621        beam.position_angle_rad(),
622    )
623}
624
625fn beam_area(beam: RestoringBeam) -> f64 {
626    beam.major_fwhm_rad() * beam.minor_fwhm_rad()
627}

629const fn unit_label(unit: ProductUnit) -> &'static str {
630    match unit {
631        ProductUnit::NotApplicable | ProductUnit::Dimensionless | ProductUnit::VisibilityWeight => {
632            ""
633        }
634        ProductUnit::JyPerBeam => "Jy/beam",
635        ProductUnit::JyPerPixel => "Jy/pixel",

659/// Publish a complete image from a private staging directory inside the
660/// target's parent. An existing target is exchanged atomically; its old
661/// contents remain at `staging` for the caller's staging guard to remove.
662/// Both containing directories are synced before success is returned.
663///
664/// # Errors
665///
666/// Invalid staging placement, unsupported directory exchange or an I/O error.
667/// A failed exchange leaves the previous target intact; a sync error after
668/// publication can report failure with the complete new image already visible.
669pub fn publish_staged_image(staging: &Path, target: &Path) -> std::io::Result<()> {
670    let parent = target.parent().ok_or_else(|| {
671        std::io::Error::new(
672            std::io::ErrorKind::InvalidInput,
673            "product target has no parent directory",
674        )
675    })?;
676    if staging.parent().and_then(Path::parent) != Some(parent) {
677        return Err(std::io::Error::new(
678            std::io::ErrorKind::InvalidInput,
679            "product staging must be private to the output directory",
680        ));
681    }
682    if target.exists() {
683        exchange_directories(staging, target)?;
684    } else {
685        fs::rename(staging, target)?;
686    }
687    File::open(parent).and_then(|directory| directory.sync_all())?;
688    File::open(
689        staging
690            .parent()
691            .expect("validated private output directory"),
692    )
693    .and_then(|directory| directory.sync_all())?;
694    Ok(())
695}

697#[cfg(target_os = "macos")]
698fn exchange_directories(staging: &Path, target: &Path) -> std::io::Result<()> {
699    use std::os::unix::ffi::OsStrExt;
700    let staging = CString::new(staging.as_os_str().as_bytes())
701        .map_err(|_| std::io::Error::other("staging path contains NUL"))?;
702    let target = CString::new(target.as_os_str().as_bytes())
703        .map_err(|_| std::io::Error::other("target path contains NUL"))?;
704    // SAFETY: both C strings remain valid for the duration of this atomic call.
705    let status = unsafe { libc::renamex_np(staging.as_ptr(), target.as_ptr(), libc::RENAME_SWAP) };
706    (status == 0)
707        .then_some(())
708        .ok_or_else(std::io::Error::last_os_error)
709}
710
711#[cfg(target_os = "linux")]
712fn exchange_directories(staging: &Path, target: &Path) -> std::io::Result<()> {
713    use std::os::unix::ffi::OsStrExt;
714    let staging = CString::new(staging.as_os_str().as_bytes())
715        .map_err(|_| std::io::Error::other("staging path contains NUL"))?;
716    let target = CString::new(target.as_os_str().as_bytes())
717        .map_err(|_| std::io::Error::other("target path contains NUL"))?;
718    // SAFETY: both C strings remain valid for the duration of this atomic call.
719    let status = unsafe {
720        libc::syscall(
721            libc::SYS_renameat2,
722            libc::AT_FDCWD,
723            staging.as_ptr(),
724            libc::AT_FDCWD,
725            target.as_ptr(),
726            libc::RENAME_EXCHANGE,
727        )
728    };
729    (status == 0)
730        .then_some(())
731        .ok_or_else(std::io::Error::last_os_error)
732}
733
734#[cfg(not(any(target_os = "macos", target_os = "linux")))]
735fn exchange_directories(_: &Path, _: &Path) -> std::io::Result<()> {
736    Err(std::io::Error::new(
737        std::io::ErrorKind::Unsupported,
738        "atomic non-empty directory exchange is unavailable",
739    ))
740}
