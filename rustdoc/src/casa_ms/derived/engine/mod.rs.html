46/// Engine for computing derived quantities from MS metadata.
47///
48/// Caches antenna positions and field directions extracted from subtables
49/// and provides methods to compute hour angle, parallactic angle, AZEL,
50/// LAST, and UVW in J2000.
51///
52/// # Usage
53///
54/// ```rust,no_run
55/// use casa_ms::derived::engine::MsCalEngine;
56/// use casa_ms::MeasurementSet;
57/// use casa_ms::MeasurementSetBuilder;
58///
59/// let ms = MeasurementSet::create_memory(MeasurementSetBuilder::new()).unwrap();
60/// let engine = MsCalEngine::new(&ms).unwrap();
61/// ```
62///
63/// Cf. C++ `MSCalEngine`.
64pub struct MsCalEngine {
65    /// Antenna positions in ITRF.
66    antenna_positions: Box<[MPosition]>,
67    /// Whether each antenna uses an alt-az mount.
68    antenna_mount_alt_az: Box<[bool]>,
69    /// How each antenna's mount turns its feed ([`Self::feed_parallactic_angle`]).
70    antenna_feed_mounts: Box<[FeedMount]>,
71    /// Field phase directions (constant term, J2000).
72    field_directions: Box<[MDirection]>,
73    /// Raw FIELD phase values, used as true-angle offsets for attached ephemerides.
74    field_phase_offsets: Box<[[f64; 2]]>,
75    /// Original references for corrections that use the stored phase measure.
76    field_phase_references: Box<[DirectionRef]>,
77    /// What each field follows, once attached
78    /// ([`Self::with_field_ephemerides`]); empty until then.
79    field_motions: Box<[FieldMotion]>,
80    calibration_field_directions: bool,
81    /// The epoch each field's `PHASE_DIR` is stated at (MSv2 `FIELD.TIME`),
82    /// when the store states one.
83    field_reference_epochs: Box<[Option<MEpoch>]>,
84    /// Observatory position (antenna 0 if no OBSERVATION subtable).
85    observatory_position: MPosition,
86    /// Epoch reference used by MAIN.TIME.
87    time_reference: EpochRef,
88    measures: Option<Arc<dyn MeasuresProvider>>,
89    azel_cache: RwLock<HashMap<GeometryCacheKey, (f64, f64)>>,
90    hadec_cache: RwLock<HashMap<GeometryCacheKey, (f64, f64)>>,
91    parallactic_angle_cache: RwLock<HashMap<GeometryCacheKey, f64>>,
92}

111fn cached_geometry_value<T: Copy>(
112    cache: &RwLock<HashMap<GeometryCacheKey, T>>,
113    key: GeometryCacheKey,
114) -> Option<T> {
115    cache
116        .read()
117        .expect("MS calibration geometry cache poisoned")
118        .get(&key)
119        .copied()
120}
121
122fn store_geometry_value<T: Copy>(
123    cache: &RwLock<HashMap<GeometryCacheKey, T>>,
124    key: GeometryCacheKey,
125    value: T,
126) {
127    cache
128        .write()
129        .expect("MS calibration geometry cache poisoned")
130        .insert(key, value);
131}

134    /// Create a new engine by extracting metadata from the MS subtables.
135    ///
136    /// Reads ANTENNA positions, FIELD phase directions, and resolves the
137    /// observatory position from `OBSERVATION::TELESCOPE_NAME` when possible,
138    /// falling back to antenna 0 only when no catalog entry is available.
139    ///
140    /// The engine opens the production Measures provider
141    /// ([`crate::open_measures_runtime`]); [`Self::with_measures`] takes one
142    /// the caller already holds.
143    pub fn new(ms: &MeasurementSet) -> MsResult<Self> {
144        Self::with_measures(ms, crate::open_measures_runtime()?)
145    }
146
147    /// Create an engine over `ms`, as [`Self::new`] does, that evaluates
148    /// every frame conversion with `measures`.
149    ///
150    /// An operation that opens one provider at its boundary builds its
151    /// engine with this, so the engine and everything else the operation
152    /// hands the provider to share one set of reference data.
153    pub fn with_measures(
154        ms: &MeasurementSet,
155        measures: Arc<dyn MeasuresProvider>,
156    ) -> MsResult<Self> {
157        let ant = ms.antenna()?;
158        let n_ant = ant.row_count();
159        let mut antenna_positions = Vec::with_capacity(n_ant);
160        let mut antenna_mount_alt_az = Vec::with_capacity(n_ant);
161        let mut antenna_feed_mounts = Vec::with_capacity(n_ant);
162        for row in 0..n_ant {
163            let pos = ant.position(row)?;
164            antenna_positions.push(MPosition::new_itrf(pos[0], pos[1], pos[2]));
165            let mount = ant.mount(row)?;
166            antenna_mount_alt_az.push(casa_observation::alt_az_mount(&mount));
167            antenna_feed_mounts.push(FeedMount::parse(&mount));
168        }
169
170        let observatory_position =
171            resolve_observatory_position(ms, &antenna_positions, measures.as_ref());
172        let field = ms.field()?;
173        let n_field = field.row_count();
174        let mut field_directions = Vec::with_capacity(n_field);
175        let mut field_phase_offsets = Vec::with_capacity(n_field);
176        let mut field_phase_references = Vec::with_capacity(n_field);
177        let mut field_reference_epochs = Vec::with_capacity(n_field);
178        for row in 0..n_field {
179            let (longitude, latitude) = phase_dir_constant(field.phase_dir(row)?)?;
180            let reference = resolve_direction_reference(field.table(), "FIELD", "PHASE_DIR", row)?;
181            field_phase_offsets.push([longitude, latitude]);
182            field_phase_references.push(reference);
183            field_directions.push(resolve_field_phase_direction_j2000_with_observatory(
184                MDirection::from_angles(longitude, latitude, reference),
185                &field,
186                row,
187                &observatory_position,
188                Arc::clone(&measures),
189            )?);
190            field_reference_epochs.push(field_reference_epoch(ms, row).ok());
191        }
192        let time_reference = detect_time_reference(ms);
193
194        Ok(Self {
195            antenna_positions: antenna_positions.into_boxed_slice(),
196            antenna_mount_alt_az: antenna_mount_alt_az.into_boxed_slice(),
197            antenna_feed_mounts: antenna_feed_mounts.into_boxed_slice(),
198            field_directions: field_directions.into_boxed_slice(),
199            field_phase_offsets: field_phase_offsets.into_boxed_slice(),
200            field_phase_references: field_phase_references.into_boxed_slice(),
201            field_motions: Box::new([]),
202            calibration_field_directions: false,
203            field_reference_epochs: field_reference_epochs.into_boxed_slice(),
204            observatory_position,
205            time_reference,
206            measures: Some(measures),
207            azel_cache: RwLock::new(HashMap::new()),
208            hadec_cache: RwLock::new(HashMap::new()),
209            parallactic_angle_cache: RwLock::new(HashMap::new()),
210        })
211    }

213    /// This engine with the ephemeris each field of `ms` follows
214    /// (`FIELD/EPHEM<id>_*.tab`, named by `EPHEMERIS_ID`), each table read
215    /// once. A field that follows one is then observed, at each time,
216    /// towards the body offset by its `PHASE_DIR`, whatever the imaging
217    /// phase centre ([`Self::observation_direction_j2000`]); a field whose
218    /// `NUM_POLY` is above zero follows its polynomial, as casacore reads
219    /// it. A field whose table cannot be read is refused only where its
220    /// direction is asked for. Imaging attaches them; an engine without
221    /// them reads every field's `PHASE_DIR` as its direction, as it always
222    /// has.
223    ///
224    /// # Errors
225    ///
226    /// FIELD cannot be read.
227    pub fn with_field_ephemerides(mut self, ms: &MeasurementSet) -> MsResult<Self> {
228        self.field_motions = crate::ephemeris::field_motions(ms, false)?;
229        Ok(self)
230    }

318    /// Build a measures frame at the specified visibility epoch and antenna.
319    /// # Errors
320    /// The antenna is outside the MeasurementSet antenna table.
321    pub fn antenna_frame(&self, time_mjd_sec: f64, antenna_id: usize) -> MsResult<MeasFrame> {
322        let epoch = MEpoch::from_mjd_seconds(time_mjd_sec, self.time_reference);
323        let position = self
324            .antenna_positions
325            .get(antenna_id)
326            .ok_or_else(|| MsError::InvalidIndex {
327                index: antenna_id,
328                max: self.antenna_positions.len(),
329                context: "antenna_id".to_string(),
330            })?
331            .clone();
332        let frame = MeasFrame::new().with_epoch(epoch).with_position(position);
333        Ok(match &self.measures {
334            Some(measures) => frame.with_measures(Arc::clone(measures)),
335            None => frame,
336        })
337    }

380    /// Evaluate one named moving-source direction at an observation timestamp.
381    ///
382    /// The direction is resolved from the immutable Measures provider bound to
383    /// this engine and is never retained as a sample-sized array. Solar-system
384    /// bodies therefore follow the selected row epoch while fixed catalog
385    /// names remain constant, with both projected into the J2000 frame used by
386    /// MeasurementSet imaging geometry.
387    pub(crate) fn named_direction_j2000(
388        &self,
389        time_mjd_sec: f64,
390        source_name: &str,
391    ) -> MsResult<MDirection> {
392        let frame =
393            self.make_frame_with_position(time_mjd_sec, self.observatory_position.clone())?;
394        MDirection::from_source_name(source_name, &frame)?
395            .convert_to(DirectionRef::J2000, &frame)
396            .map_err(Into::into)
397    }

399    /// Evaluate one immutable named, external, or FIELD-attached moving source.
400    pub fn ephemeris_direction_j2000(
401        &self,
402        time_mjd_sec: f64,
403        field_id: usize,
404        source_name: &str,
405        ephemeris: &PhaseCentreEphemeris,
406    ) -> MsResult<MDirection> {
407        if let Some(target) = ephemeris.named_target() {
408            if !target.eq_ignore_ascii_case(source_name) {
409                return Err(MsError::InvalidInput(format!(
410                    "moving target {source_name:?} does not match ephemeris target {target:?}"
411                )));
412            }
413            return self.named_direction_j2000(time_mjd_sec, target);
414        }
415        let ephemeris_mjd_tdb = self.ephemeris_mjd_tdb(time_mjd_sec)?;
416        let sample = ephemeris
417            .sample(field_id, ephemeris_mjd_tdb)
418            .map_err(|error| MsError::InvalidInput(error.to_string()))?;
419        self.sampled_ephemeris_direction_j2000(time_mjd_sec, sample)
420    }

482    /// Evaluate a scalar from the attached ephemeris at the FIELD column's
483    /// interpolation epoch. A fixed field has no attached value.
484    pub fn field_ephemeris_scalar(
485        &self,
486        field_id: usize,
487        time_mjd_sec: f64,
488        evaluate: impl FnOnce(&AttachedFieldEphemeris, f64) -> MsResult<f64>,
489    ) -> MsResult<Option<f64>> {
490        let Some((ephemeris, _)) = self.field_series(field_id)? else {
491            return Ok(None);
492        };
493        let mjd = self.field_ephemeris_mjd_utc(field_id, time_mjd_sec)?;
494        ephemeris.check_observation_epoch(mjd)?;
495        evaluate(ephemeris, mjd).map(Some)
496    }

498    fn field_ephemeris_mjd_utc(&self, field_id: usize, time_mjd_sec: f64) -> MsResult<f64> {
499        let origin = self
500            .field_reference_epochs
501            .get(field_id)
502            .and_then(Option::as_ref);
503        let reference = origin.map_or(self.time_reference, MEpoch::refer);
504        let epoch = if time_mjd_sec == 0.0 {
505            origin
506                .cloned()
507                .unwrap_or_else(|| MEpoch::from_mjd_seconds(time_mjd_sec, reference))
508        } else {
509            MEpoch::from_mjd_seconds(time_mjd_sec, reference)
510        };
511        let frame =
512            self.make_frame_with_position(time_mjd_sec, self.observatory_position.clone())?;
513        Ok(epoch.convert_to(EpochRef::UTC, &frame)?.value().as_mjd())
514    }

550    /// Return the phase centre evaluated without a row epoch, retaining its
551    /// recorded reference. A moving field is read at FIELD.TIME or, when
552    /// unavailable, its first observation time.
553    ///
554    /// # Errors
555    /// An unknown field or a moving phase centre that cannot be evaluated.
556    pub fn reference_phase_direction(
557        &self,
558        field_id: usize,
559        first_time_mjd_sec: f64,
560    ) -> MsResult<MDirection> {
561        match self.moving_field_reference_direction(field_id, first_time_mjd_sec)? {
562            Some(direction) => Ok(direction),
563            None => {
564                self.field_dir(field_id)?;
565                let [longitude, latitude] = self.field_phase_offsets[field_id];
566                Ok(MDirection::from_angles(
567                    longitude,
568                    latitude,
569                    self.field_phase_references[field_id],
570                ))
571            }
572        }
573    }

576    fn direction_at(&self, time_mjd_sec: f64, field_id: usize) -> MsResult<Cow<'_, MDirection>> {
577        let Some((ephemeris, offset)) = self.field_series(field_id)? else {
578            return Ok(Cow::Borrowed(self.field_dir(field_id)?));
579        };
580        if self.calibration_field_directions {
581            let mjd = self.ephemeris_mjd_utc(time_mjd_sec)?;
582            let reference = ephemeris.reference_direction(mjd, offset, true)?;
583            let frame =
584                self.make_frame_with_position(time_mjd_sec, self.observatory_position.clone())?;
585            return Ok(Cow::Owned(
586                reference.convert_to(DirectionRef::J2000, &frame)?,
587            ));
588        }
589        let direction = self.field_body_direction(time_mjd_sec, ephemeris)?;
590        Ok(Cow::Owned(
591            casa_core::measures::ephemeris::offset_true_angle(&direction, offset)?,
592        ))
593    }

656    /// observing offset. Calibration table construction supplies

658    /// `PHASE_DIR` offset.
659    ///
660    /// # Errors
661    /// The field or its reference epoch is outside the ephemeris domain.
662    pub fn moving_field_reference_direction_with_offset(
663        &self,
664        field_id: usize,
665        first_time_mjd_sec: f64,
666        offset: Option<[f64; 2]>,
667    ) -> MsResult<Option<MDirection>> {
668        let Some((ephemeris, phase_offset)) = self.field_series(field_id)? else {
669            return Ok(None);
670        };
671        let frame = MeasFrame::new().with_position(self.observatory_position.clone());
672        let frame = match &self.measures {
673            Some(measures) => frame.with_measures(Arc::clone(measures)),
674            None => frame,

682        let mjd_utc = origin.convert_to(EpochRef::UTC, &frame)?.value().as_mjd();
683        ephemeris
684            .reference_direction(mjd_utc, offset.unwrap_or(phase_offset), false)
685            .map(Some)
686    }

891    pub fn last(&self, time_mjd_sec: f64, antenna_id: usize) -> MsResult<f64> {
892        let frame = self.antenna_frame(time_mjd_sec, antenna_id)?;
893        let epoch = MEpoch::from_mjd_seconds(time_mjd_sec, EpochRef::UTC);
894        let last_epoch = epoch.convert_to(EpochRef::LAST, &frame)?;

896        let last_days = last_epoch.value().as_mjd();
897        // Take fractional part of day and convert to radians
898        let frac = last_days - last_days.floor();
899        Ok(frac * 2.0 * std::f64::consts::PI)
900    }

938    /// Compute UVW in J2000 frame for a baseline (ant1, ant2) at a given time and field.
939    ///
940    /// UVW = R · (pos2 − pos1), where R is the rotation matrix from ITRF to
941    /// the J2000 frame aligned with the field direction.
942    ///
943    /// Cf. C++ `MSCalEngine::getNewUVW`.
944    pub fn uvw_j2000(
945        &self,
946        time_mjd_sec: f64,
947        field_id: usize,
948        ant1: usize,
949        ant2: usize,
950    ) -> MsResult<[f64; 3]> {
951        let frame = self.antenna_frame(time_mjd_sec, ant1)?;
952
953        let pos1 = self
954            .antenna_positions
955            .get(ant1)
956            .ok_or_else(|| MsError::InvalidIndex {
957                index: ant1,
958                max: self.antenna_positions.len(),
959                context: "ant1".to_string(),
960            })?;
961        let pos2 = self
962            .antenna_positions
963            .get(ant2)
964            .ok_or_else(|| MsError::InvalidIndex {
965                index: ant2,
966                max: self.antenna_positions.len(),
967                context: "ant2".to_string(),
968            })?;
969
970        // Baseline in ITRF (meters)
971        let dx = pos2.as_itrf()[0] - pos1.as_itrf()[0];
972        let dy = pos2.as_itrf()[1] - pos1.as_itrf()[1];
973        let dz = pos2.as_itrf()[2] - pos1.as_itrf()[2];
974
975        // Get the field direction in J2000
976        let dir = self.direction_at(time_mjd_sec, field_id)?;
977        if self.field_moves(field_id) {
978            // MSCalEngine::getNewUVW forms the baseline from antenna 1 to 2
979            // and projects it in an observatory frame at this row's epoch.
980            let epoch = MEpoch::from_mjd_seconds(time_mjd_sec, self.time_reference);
981            let measures = self
982                .measures
983                .clone()
984                .map_or_else(crate::open_measures_runtime, Ok)?;
985            let projection = UvwFrame::new(
986                dir.as_angles().into(),
987                epoch,
988                &self.observatory_position,
989                measures,
990            )?;
991            return Ok(projection.baseline_uvw([dx, dy, dz]));
992        }
993        let (ra, dec) = dir.as_angles();
994
995        let (sin_dec, cos_dec) = dec.sin_cos();
996
997        // Rotate baseline from ITRF to equatorial using GMST, then to UVW using RA/Dec.
998        let epoch = MEpoch::from_mjd_seconds(time_mjd_sec, EpochRef::UTC);
999        let gmst_epoch = epoch.convert_to(EpochRef::GMST1, &frame)?;
1000        let gmst_frac = gmst_epoch.value().as_mjd();
1001        let gmst_rad = (gmst_frac - gmst_frac.floor()) * 2.0 * std::f64::consts::PI;
1002
1003        // Rotate baseline from ITRF to equatorial using GMST
1004        let (sin_g, cos_g) = gmst_rad.sin_cos();
1005        let ex = cos_g * dx - sin_g * dy;
1006        let ey = sin_g * dx + cos_g * dy;
1007        let ez = dz;
1008
1009        // Now rotate from equatorial to UVW using RA and Dec
1010        let (sin_ra, cos_ra) = ra.sin_cos();
1011
1012        let u = sin_ra * ex - cos_ra * ey;
1013        let v = -sin_dec * cos_ra * ex - sin_dec * sin_ra * ey + cos_dec * ez;
1014        let w = cos_dec * cos_ra * ex + cos_dec * sin_ra * ey + sin_dec * ez;
1015
1016        Ok([u, v, w])
1017    }

1020    ///
1021    /// This mirrors CASA `KAntPosJones`: build the frame at antenna 0, convert
1022    /// the earth-frame antenna offset to the field direction's sky frame, then
1023    /// use its W projection as the per-antenna delay.
1024    pub fn project_itrf_offset_to_uvw(
1025        &self,
1026        time_mjd_sec: f64,
1027        field_id: usize,
1028        _antenna_id: usize,
1029        offset_m: [f64; 3],
1030    ) -> MsResult<[f64; 3]> {
1031        let antenna = self
1032            .antenna_positions
1033            .first()
1034            .ok_or_else(|| MsError::InvalidIndex {
1035                index: 0,
1036                max: 0,
1037                context: "antenna position correction reference".to_owned(),
1038            })?;
1039        if offset_m == [0.0; 3] {
1040            return Ok([0.0; 3]);
1041        }
1042        let direction = self.reference_phase_direction(field_id, time_mjd_sec)?;
1043        // KAntPosJones sets its correction frame to UTC, including for a

1045        let epoch = MEpoch::from_mjd_seconds(time_mjd_sec, EpochRef::UTC);
1046        let measures = self
1047            .measures
1048            .clone()
1049            .map_or_else(crate::open_measures_runtime, Ok)?;

1064    /// Reproject raw MS UVW coordinates from one field phase center to another.
1065    ///
1066    /// The input UVW is assumed to be stored in native MeasurementSet
1067    /// convention for `source_field_id`. This applies the same
1068    /// direct `fixvis` geometry transform CASA applies to `MAIN.UVW` for a
1069    /// phase-center change between fixed J2000 field directions and returns
1070    /// the corresponding geometric phase shift in meters.
1071    ///
1072    /// Important: CASA's `GridFT` imaging path first applies `negateUV`, which
1073    /// imaging's `GridFtProjection` (casa-imaging-model) represents. This method preserves
1074    /// the direct `fixvis` convention only.
1075    pub fn reproject_raw_uvw_between_fields(
1076        &self,
1077        raw_uvw_m: [f64; 3],
1078        source_field_id: usize,
1079        target_field_id: usize,
1080    ) -> MsResult<([f64; 3], f64)> {
1081        if source_field_id == target_field_id {
1082            return Ok((raw_uvw_m, 0.0));
1083        }
1084
1085        let source_dir = self.field_dir(source_field_id)?;
1086        let target_dir = self.field_dir(target_field_id)?;
1087        // CASA's direct fixvis transform drives UVWMachine on the stored
1088        // MAIN.UVW row vector. Matching that path therefore
1089        // requires the target/source order opposite to the intuitive
1090        // source->target helper naming used elsewhere.
1091        Ok(UvwRotation::new(target_dir, source_dir).apply(raw_uvw_m))
1092    }
