3//! Peak flux <-> integrated flux conversion (`ia.convertflux`).

5use casa_coordinates::{Coordinate, CoordinateSystem};
6use casa_core::quanta::unit_val::UnitVal;
7use casa_core::quanta::{Quantity, Unit, casacore};
8use casa_images::{BeamComponent, GaussianBeam, ImagePixelType};
9
10use super::beam_view::BeamView;
11use super::beams::beam_for_units;
12use super::error::{ImageToolError, ImageToolResult, ToolLogLine};
13use super::handle::{ImageHandle, with_pixels, with_storage};

15struct FluxAreas {
16    pixel: f64,
17    beam: Option<f64>,
18}
19
20impl FluxAreas {
21    fn unit(&self, name: &str) -> ImageToolResult<Unit> {
22        let area = UnitVal::ANGLE.pow(2).dim;
23        let pixel = ("pixel", UnitVal::new(self.pixel, area));
24        let unit = if name.contains("beam") {
25            let beam = self.beam.ok_or_else(|| {
26                ImageToolError::new(format!(
27                    "No beam defined even though the image brightness units are {name}"
28                ))
29            })?;
30            Unit::with_local_definitions(name, &[pixel, ("beam", UnitVal::new(beam, area))])
31        } else {
32            Unit::with_local_definitions(name, &[pixel])
33        }?;
34        Ok(unit)
35    }
36
37    fn brightness_unit(&self, name: &str, logs: &mut Vec<ToolLogLine>) -> ImageToolResult<Unit> {
38        let unit = self.unit(name)?;
39        let dimensions = unit.val().dim * UnitVal::ANGLE.pow(2).dim;
40        let jy = Unit::new("Jy").expect("built-in flux unit");
41        if matches!(name, "K" | "Jy/beam.km/s") || dimensions == jy.val().dim {
42            return Ok(unit);
43        }
44        if name.is_empty() {
45            logs.push(ToolLogLine::warning(
46                "There are no image brightness units, assuming Jy/pixel",
47            ));
48        } else {
49            logs.push(ToolLogLine::warning(format!(
50                "The image units '{name}' are not consistent with Jy when integrated over the sky.  Assuming Jy/pixel"
51            )));
52        }
53        self.unit("Jy/pixel")
54    }
55}
56
57fn checked_axis(axis: &Quantity, label: &str, radians: &Unit) -> ImageToolResult<Quantity> {
58    if !axis.unit().conformant(radians) {
59        return Err(ImageToolError::new(format!(
60            "Exception: {label} axis must have angular units ({} is not)..",
61            axis.unit().name()
62        )));
63    }
64    if axis.value() < 0.0 {
65        return Err(ImageToolError::new(format!(
66            "Exception: {label} axis cannot be less than zero.."
67        )));
68    }
69    Ok(casacore::convert(axis, Some(radians))?)
70}

72/// Converts between peak brightness and integrated flux using CASA [`Quantity`] rules (`ia.convertflux`).
73#[allow(clippy::too_many_arguments)]
74pub fn convert_flux_quantity(
75    image: &ImageHandle,
76    beam_view: &BeamView,
77    value: Quantity,
78    axes: [Quantity; 2],
79    to_peak: bool,
80    plane: (i64, i64),
81) -> ImageToolResult<(Quantity, Vec<ToolLogLine>)> {
82    if image.kind() != ImagePixelType::Float32 {
83        return Err(ImageToolError::new(
84            "This method only supports Float valued images",
85        ));
86    }
87    let (_, direction) = CoordinateSystem::first_direction_in_record(&image.coordinate_record())?
88        .ok_or_else(|| {
89        let name = image
90            .path()
91            .map_or_else(|| "Temporary_Image".into(), |path| path.to_string_lossy());
92        ImageToolError::new(format!(
93            "Exception: Image {name} does not have required coordinate Direction."
94        ))
95    })?;
96    let radians = Unit::new("rad").expect("built-in angle unit");
97    let major = checked_axis(&axes[0], "Major", &radians)?;
98    let minor = checked_axis(&axes[1], "Minor", &radians)?;
99    if major.value() < minor.value() {
100        return Err(ImageToolError::new(
101            "Exception: Major axis must be greater or equal to minor axis.",
102        ));
103    }
104    let brightness =
105        with_pixels!(image, storage => with_storage!(storage, image => image.units().to_owned()));
106    let beam = beam_for_units(beam_view, plane.0, plane.1)?.map(|beam| {
107        GaussianBeam::new(
108            beam.component(BeamComponent::Major).get_si_value(),
109            beam.component(BeamComponent::Minor).get_si_value(),
110            0.0,
111        )
112    });
113    let inc = direction.increment();
114    let pixel = (inc[0] * inc[1]).abs();
115    let mut beam = beam.filter(|beam| !beam.is_null()).map(|beam| beam.area());
116    if brightness.contains("/beam") && beam.is_none() {
117        // CASA's fallback beam uses the numerical coordinate increments in
118        // their saved units as the full-width axes in radians.
119        let (_, increments) = direction.reference_and_increment_in_world_units();
120        let (major, minor) = (increments[0].abs(), increments[1].abs());
121        if major < minor {
122            return Err(ImageToolError::new(
123                "Exception: Major axis must be greater or equal to minor axis.",
124            ));
125        }
126        let fallback = GaussianBeam::new(major, minor, 0.0);
127        beam = (!fallback.is_null()).then(|| fallback.area());
128    }
129    let areas = FluxAreas { pixel, beam };
130    let jy = Unit::new("Jy").expect("built-in flux unit");
131    let jykms = Unit::new("Jy.km/s").expect("built-in integrated flux unit");
132    let kelvin_area = Unit::new("K.rad2").expect("built-in temperature-area unit");
133    let gaussian = std::f64::consts::PI / 4.0 / 2.0_f64.ln();
134    let mut logs = Vec::new();
135    let result = if to_peak {
136        let target = if value.unit().conformant(&jy) {
137            &jy
138        } else if value.unit().conformant(&jykms) {
139            &jykms
140        } else if value.unit().conformant(&kelvin_area) {
141            value.unit()
142        } else {
143            return Err(ImageToolError::new(format!(
144                "Cannot convert units of Flux integral ({}) to Jy",
145                value.unit().name()
146            )));
147        };
148        let flux = casacore::convert(&value, Some(target))?;
149        let scaled = Quantity::with_unit(flux.value() / gaussian, flux.unit().clone());
150        let scaled = casacore::div(&scaled, &major)?;
151        let peak = casacore::div(&scaled, &minor)?;
152        let brightness = areas.brightness_unit(&brightness, &mut logs)?;
153        casacore::convert(&peak, Some(&brightness))?
154    } else {
155        let brightness = areas.brightness_unit(value.unit().name(), &mut logs)?;
156        let integrated = areas.unit(&format!("{}.rad.rad", brightness.name()))?;
157        let flux = Quantity::with_unit(
158            value.value() * gaussian * major.value() * minor.value(),
159            integrated,
160        );
161        if flux.unit().conformant(&jy) {
162            casacore::convert(&flux, Some(&jy))?
163        } else if flux.unit().conformant(&jykms) {
164            casacore::convert(&flux, Some(&jykms))?
165        } else if flux.unit().conformant(&kelvin_area) {
166            flux
167        } else {
168            logs.push(ToolLogLine::warning(
169                "Cannot convert units of Flux integral to Jy - will assume Jy",
170            ));
171            Quantity::with_unit(flux.value(), jy)
172        }
173    };
174    Ok((result, logs))
175}
