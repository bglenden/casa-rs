23const L_BAND_BEAM_SIDE: usize = 512;
24const L_BAND_BEAM_CELL_RAD: f64 = 0.024 / 256.0;

26/// The aperture grid and the cropped sky grid used by
27/// `AWConvFunc::makeAConvFunc` for a small single-field image.
28pub struct Awp2Sampling {
29    image_side: usize,
30    cell_rad: f64,
31    aperture_side: usize,
32    sky_side: usize,
33    oversampling: u16,
34}
35
36impl Awp2Sampling {
37    /// Derive sampling from a square, unpadded image with square cells.
38    /// Only the small-field Lanczos branch is currently supported.
39    pub fn new(geometry: &GridGeometry) -> Result<Self, OperatorError> {
40        let extent = geometry.image();
41        let [side, ny] = extent.shape;
42        let [dx, dy] = extent.increment_rad.map(f64::abs);
43        let field_of_view = side as f64 * dx;
44        let beam_field_of_view = L_BAND_BEAM_SIDE as f64 * L_BAND_BEAM_CELL_RAD;
45        let oversampling = if side < 100 {
46            100
47        } else {
48            10000_usize.div_ceil(side)
49        };
50        if side != ny
51            || side >= 200
52            || (dx - dy).abs() > dx * f64::EPSILON
53            || !oversampling.is_multiple_of(2)
54            || field_of_view > 0.125 * beam_field_of_view
55        {
56            return Err(invalid(
57                "AWP2 requires a small square image with square cells and even oversampling (#1071)",
58            ));
59        }
60        let aperture_side = ((L_BAND_BEAM_SIDE as f64 / dx * L_BAND_BEAM_CELL_RAD / 16.0).ceil()
61            as usize)
62            .checked_mul(8)
63            .filter(|side| {
64                side.checked_mul(*side)
65                    .is_some_and(|n| n <= isize::MAX as usize / 64)
66            })
67            .ok_or_else(|| invalid("AWP2 aperture grid is too large"))?;
68        let sky_side =
69            ((field_of_view / beam_field_of_view * aperture_side as f64 / 2.0).ceil() as usize) * 2;
70        if sky_side < 12 || sky_side > aperture_side {
71            return Err(invalid("AWP2 sky crop does not fit its aperture grid"));
72        }
73        Ok(Self {
74            image_side: side,
75            cell_rad: dx,
76            aperture_side,
77            sky_side,
78            oversampling: oversampling as u16,
79        })
80    }

82    /// Temporary Jones and sky FFT planes and their planning workspaces.
83    #[must_use]
84    pub fn screens_bytes(&self, count: usize) -> u64 {
85        let aperture_cells = self.aperture_side * self.aperture_side;
86        let sky_cells = self.sky_side * self.sky_side;
87        ((4 * aperture_cells + 2 * sky_cells) * size_of::<Complex32>()) as u64
88            + casa_fft::planning_bytes::<f32>(aperture_cells)
89            + casa_fft::planning_bytes::<f32>(sky_cells)
90            + (count * (size_of::<ApertureCell>() + 2 * sky_cells * size_of::<Complex32>())) as u64
91    }

93    /// Calculate the A terms and their one-plane AW normalization with the
94    /// supplied VLA surface. Scratch is charged before calling this method.
95    pub fn screens(
96        self,
97        frequencies_hz: &[f64],
98        surface: EvlaDishSurface,
99    ) -> Result<Awp2Screens, OperatorError> {
100        if frequencies_hz.is_empty()
101            || frequencies_hz.len() > usize::from(u16::MAX)
102            || frequencies_hz.windows(2).any(|f| f[0] >= f[1])
103            || frequencies_hz.iter().any(|f| !(0.9e9..=2e9).contains(f))
104        {
105            return Err(invalid(
106                "AWP2 requires ordered VLA L-band beam frequencies (#1071)",
107            ));
108        }
109        let model = EvlaApertureModel::new(surface);
110        let mut jones = vec![Complex32::default(); 4 * self.aperture_side * self.aperture_side];
111        let mut aperture_fft = PlaneFft::new([self.aperture_side; 2], false)?;
112        let mut sky_fft = PlaneFft::new([self.sky_side; 2], false)?;
113        let mut cells = Vec::with_capacity(frequencies_hz.len());
114        // CASA crops every A plane to the support of the highest frequency.
115        let mut a_support = None;
116        for &frequency in frequencies_hz.iter().rev() {
117            let beam = self.sky_beam(&model, frequency, &mut aperture_fft, &mut jones)?;
118            let cell = ApertureCell::new(beam, self.sky_side, &mut sky_fft, a_support)?;
119            a_support.get_or_insert(cell.a_side / 2);
120            cells.push(cell);
121        }
122        cells.reverse();
123        Ok(Awp2Screens {
124            sampling: self,
125            cells,
126        })
127    }

129    fn sky_beam(
130        &self,
131        model: &EvlaApertureModel,
132        frequency: f64,
133        aperture_fft: &mut PlaneFft<f32>,
134        jones: &mut [Complex32],
135    ) -> Result<Vec<Complex32>, OperatorError> {
136        let wavelength = SPEED_OF_LIGHT_M_PER_S / f64::from(frequency as f32);
137        let cell_m = wavelength / (self.aperture_side as f64 * self.cell_rad);
138        let grid = EvlaApertureGrid::new(self.aperture_side, cell_m, 3, frequency, 0.0)
139            .map_err(|_| invalid("invalid AWP2 aperture sampling"))?;
140        model
141            .fill_vla_l_aperture(grid, jones)
142            .map_err(|_| invalid("AWP2 could not evaluate its VLA aperture"))?;
143        let norm = sky_jones_normalization(self.aperture_side, aperture_fft, jones)
144            .map_err(|_| invalid("AWP2 could not normalize its VLA aperture"))?;
145        let plane = self.aperture_side * self.aperture_side;
146        let mut beam = Vec::with_capacity(self.sky_side * self.sky_side);
147        let first = (self.aperture_side - self.sky_side) / 2;
148        for y in first..first + self.sky_side {
149            for x in first..first + self.sky_side {
150                let index = y * self.aperture_side + x;
151                // applyAvgSkyJones uses real(RL) for all four correlations.
152                let rr = jones[index] / norm;
153                let ll = jones[3 * plane + index] / norm;
154                beam.push(Complex32::new((ll * rr.conj()).re, 0.0));
155            }
156        }
157        Ok(beam)
158    }
159}

161struct ApertureCell {
162    imaging: Vec<Complex32>,
163    weight: Vec<Complex32>,
164    a_side: usize,
165    aw_support: usize,
166}
167
168impl ApertureCell {
169    fn new(
170        mut imaging: Vec<Complex32>,
171        side: usize,
172        fft: &mut PlaneFft<f32>,
173        crop_support: Option<usize>,
174    ) -> Result<Self, OperatorError> {
175        let mut weight: Vec<_> = imaging.iter().map(|value| value * value.conj()).collect();
176        fft.transform(&mut imaging, false)?;
177        fft.transform(&mut weight, false)?;
178        let support = a_support(&weight, side)?;
179        for plane in [&mut imaging, &mut weight] {
180            let sum = central_sum(plane, side, support, true);
181            if !sum.is_finite() || sum <= 0.0 {
182                return Err(invalid("AWP2 aperture normalization is not positive"));
183            }
184            *plane = crop(plane, side, crop_support.unwrap_or(support), sum.recip());
185        }
186        let a_side = 2 * crop_support.unwrap_or(support);
187        // The transform of the W=0 screen is a central impulse. Its
188        // convolution leaves the A plane up to scale, which cancels in
189        // supportResizeAWConv's second normalization. The weight stays A².
190        let aw_support = diagonal_support(&imaging, a_side, 0.01)
191            .map_or(5, |radius| radius.max(5))
192            .max(support);
193        let sum = central_sum(&imaging, a_side, aw_support.min(a_side / 2), false);
194        if sum.is_finite() && sum > 0.0 {
195            imaging.iter_mut().for_each(|value| *value *= sum.recip());
196        }
197        Ok(Self {
198            imaging,
199            weight,
200            a_side,
201            aw_support,
202        })
203    }
204}
205
206fn a_support(weight: &[Complex32], side: usize) -> Result<usize, OperatorError> {
207    let (minimum, maximum) = weight
208        .iter()
209        .map(|v| v.norm())
210        .fold((f32::INFINITY, 0.0_f32), |(minimum, maximum), value| {
211            (minimum.min(value), maximum.max(value))
212        });
213    if maximum - minimum <= 0.005 * maximum {
214        return Err(invalid("AWP2 found no aperture support"));
215    }
216    let fallback = if side > 10 { 5 } else { side / 2 - 4 };
217    Ok((diagonal_support(weight, side, 0.005)
218        .filter(|radius| *radius >= 5)
219        .unwrap_or(fallback)
220        + 1)
221    .min(side / 2 - 1))
222}
223
224fn diagonal_support(plane: &[Complex32], side: usize, cutoff: f32) -> Option<usize> {
225    let maximum = plane.iter().map(|v| v.norm()).fold(0.0, f32::max);
226    let centre = side / 2;
227    (1..=centre - 2).rev().find_map(|offset| {
228        let index = (centre - offset) * (side + 1);
229        (plane[index].norm() > cutoff * maximum)
230            .then_some((std::f64::consts::SQRT_2 * offset as f64) as usize)
231    })
232}
233
234fn central_sum(plane: &[Complex32], side: usize, radius: usize, inclusive: bool) -> f32 {
235    let centre = side / 2;
236    let first = centre - radius;
237    let end = centre + radius + usize::from(inclusive);
238    (first..end)
239        .flat_map(|y| &plane[y * side + first..y * side + end])
240        .map(|v| v.re)
241        .sum()
242}
243
244fn crop(plane: &[Complex32], side: usize, radius: usize, factor: f32) -> Vec<Complex32> {
245    let first = side / 2 - radius;
246    let end = first + 2 * radius;
247    (first..end)
248        .flat_map(|y| &plane[y * side + first..y * side + end])
249        .map(|value| *value * factor)
250        .collect()
251}

253/// A and W=0 convolution planes before the oversampled resident cells are
254/// allocated. The cells use one Mueller plane and separate sensitivity taps.
255pub struct Awp2Screens {
256    sampling: Awp2Sampling,
257    cells: Vec<ApertureCell>,
258}
259
260impl Awp2Screens {
261    fn half_support(&self) -> u16 {
262        // appendConvFuncs uses the highest-frequency support for each row.
263        self.cells
264            .last()
265            .expect("nonempty beam frequencies")
266            .aw_support as u16
267    }
268
269    fn fine_side(&self) -> usize {
270        2 * (usize::from(self.half_support()) + 1) * usize::from(self.sampling.oversampling)
271    }
272
273    /// Upper bound including the resident cells and one oversampled plane
274    /// and interpolation workspace used while constructing them.
275    #[must_use]
276    pub fn kernel_bytes(&self, geometry: &GridGeometry) -> u64 {
277        let fine_side = self.fine_side();
278        (self.cells.len() * (size_of::<f64>() + size_of::<Awp2Cell>())) as u64
279            + 2 * self.cells.len() as u64
280                * dense_cell_bytes(1, self.sampling.oversampling, [self.half_support(); 2])
281            + (fine_side * fine_side * size_of::<Complex32>()
282                + fine_side * size_of::<Option<LanczosTaps>>()
283                + 8000 * size_of::<f32>()) as u64
284            + ImageCorrection::bytes(geometry.grid_shape())
285            + MuellerRouting::MAXIMUM_BYTES
286    }
287
288    /// Resample each A/W and sensitivity plane with AWConvFuncHolder's
289    /// Lanczos rule and pack them in the shared gridding layout.
290    pub fn finish(
291        self,
292        geometry: &GridGeometry,
293        polarization: &PolarizationRouting,
294        frequencies_hz: Box<[f64]>,
295    ) -> Awp2 {
296        assert_eq!(self.cells.len(), frequencies_hz.len());
297        let half_support = self.half_support();
298        let fine_side = self.fine_side();
299        let oversampling = self.sampling.oversampling;
300        let factor = self.sampling.image_side as f32 * f32::from(oversampling)
301            / self.sampling.sky_side as f32;
302        let mut plane = vec![Complex32::default(); fine_side * fine_side];
303        let sinc_table: Vec<_> = (-4000..4000)
304            .map(|i| sinc((i as f64 / 1000.0) as f32))
305            .collect();
306        let a_side = self.cells[0].a_side;
307        let resampled_side = resampled_side(a_side, factor);
308        let axis = lanczos_axis(a_side, resampled_side, fine_side, &sinc_table);
309        let scale = (a_side * a_side * usize::from(oversampling).pow(2)) as f32
310            / (resampled_side * resampled_side) as f32;
311        let mut pack = |source: &[Complex32]| {
312            resample(source, a_side, &axis, scale, &mut plane);
313            dense_cell(
314                &[OversampledPlane {
315                    values: &plane,
316                    side: fine_side,
317                    centre: [fine_side / 2; 2],
318                    sampling: oversampling,
319                }],
320                [half_support; 2],
321            )
322        };
323        let cells = self
324            .cells
325            .into_iter()
326            .map(|cell| Awp2Cell {
327                imaging: pack(&cell.imaging),
328                weight: pack(&cell.weight),
329            })
330            .collect();
331        Awp2 {
332            frequencies_hz,
333            cells,
334            half_support,
335            mueller: MuellerRouting::scalar(polarization.pol_map(), polarization.grid_pols()),
336            correction: ImageCorrection::mosaic_sinc(geometry.grid_shape(), oversampling),
337        }
338    }
339}

341// MathUtils::resample receives the holder's Float factor as a Double.
342fn resampled_side(input_side: usize, factor: f32) -> usize {
343    (input_side as f64 * f64::from(factor) / 8.0).ceil() as usize * 8
344}
345
346fn sinc(value: f32) -> f32 {
347    if value == 0.0 {
348        1.0
349    } else {
350        let x = std::f64::consts::PI * f64::from(value);
351        (x.sin() / x) as f32
352    }
353}

355struct LanczosTaps {
356    first: usize,
357    weights: [[f32; 2]; 6],
358}
359
360fn lanczos_axis(
361    side: usize,
362    resampled: usize,
363    fine: usize,
364    table: &[f32],
365) -> Vec<Option<LanczosTaps>> {
366    let sinc_lookup = |value: f64| table[(value as f32 * 1000.0 + 4000.0) as usize];
367    (0..fine)
368        .map(|i| {
369            let i = i as isize + (resampled as isize - fine as isize) / 2;
370            let position = i as f64 / resampled as f64 * side as f64;
371            let pixel = position.floor() as isize;
372            if pixel < 3 || pixel >= side as isize - 3 {
373                return None;
374            }
375            let first = pixel as usize - 2;
376            let weights = std::array::from_fn(|tap| {
377                let distance = position - (first + tap) as f64;
378                [sinc_lookup(distance), sinc_lookup(distance / 3.0)]
379            });
380            Some(LanczosTaps { first, weights })
381        })
382        .collect()
383}
384
385fn resample(
386    source: &[Complex32],
387    side: usize,
388    axis: &[Option<LanczosTaps>],
389    scale: f32,
390    output: &mut [Complex32],
391) {
392    for (row, y) in output.chunks_exact_mut(axis.len()).zip(axis) {
393        for (value, x) in row.iter_mut().zip(axis) {
394            *value = match (x, y) {
395                (Some(x), Some(y)) => interpolate(source, side, x, y) * scale,
396                _ => Complex32::default(),
397            };
398        }
399    }
400}
401
402fn interpolate(source: &[Complex32], side: usize, x: &LanczosTaps, y: &LanczosTaps) -> Complex32 {
403    let mut sum = Complex32::default();
404    for (i, [wx, lx]) in x.weights.into_iter().enumerate() {
405        for (j, [wy, ly]) in y.weights.into_iter().enumerate() {
406            let value = source[(y.first + j) * side + x.first + i];
407            let weight = f64::from(wx) * f64::from(lx) * f64::from(wy) * f64::from(ly);
408            sum.re += (f64::from(value.re) * weight) as f32;
409            sum.im += (f64::from(value.im) * weight) as f32;
410        }
411    }
412    sum
413}

420/// AWPLPG's resident cells for one field, one W plane, and no squint.

423    cells: Box<[Awp2Cell]>,
424    half_support: u16,
425    mueller: MuellerRouting,
426    correction: ImageCorrection,
427}

429impl Awp2 {
430    /// Bytes retained after the screen and interpolation workspaces are freed.
431    #[must_use]
432    pub fn resident_bytes(&self) -> u64 {
433        (std::mem::size_of_val(self.frequencies_hz.as_ref())
434            + std::mem::size_of_val(self.cells.as_ref())
435            + self
436                .cells
437                .iter()
438                .map(|cell| cell.imaging.bytes() + cell.weight.bytes())
439                .sum::<usize>()) as u64
440            + self.mueller.bytes()
441            + self.correction.resident_bytes()
442    }
443}

445impl ConvolutionFunctionSet for Awp2 {
446    fn key(&self, _row: &RowContext, freq_hz: f64, _w_lambda: f64) -> CfKey {
447        // AWConvFuncHolder compares the gridding frequency directly, using
448        // the lower frequency when distances tie.
449        let (cube, _) = super::nearest(&self.frequencies_hz, freq_hz);
450        CfKey {
451            group: 0,
452            cube: cube as u16,
453        }
454    }
455
456    fn taps<'s>(&'s self, key: CfKey, _hold: &'s mut CellHold) -> TapLayout<'s> {
457        self.cells[usize::from(key.cube)].imaging.layout()
458    }
459
460    fn max_half_support(&self) -> [u16; 2] {
461        [self.half_support; 2]
462    }
463
464    fn weight_taps<'s>(&'s self, key: CfKey, _hold: &'s mut CellHold) -> Option<TapLayout<'s>> {
465        Some(self.cells[usize::from(key.cube)].weight.layout())
466    }
467
468    fn mueller(&self) -> &MuellerRouting {
469        &self.mueller
470    }
471
472    fn image_correction(&self) -> &ImageCorrection {
473        &self.correction
474    }
475
476    fn normalisation(&self) -> KernelNormalisation {
477        KernelNormalisation::UnitSum
478    }
479
480    fn pointing_ramp(&self) -> bool {
481        true
482    }
483
484    fn grid_edge(&self) -> GridEdge {
485        GridEdge::Clipped
486    }
487
488    fn corrects_weight_image(&self) -> bool {
489        false
490    }
491
492    fn tap_footprint(&self) -> TapFootprint {
493        TapFootprint::of(
494            self.cells
495                .iter()
496                .flat_map(|cell| [cell.imaging.layout(), cell.weight.layout()]),
497        )
498    }
499}

540    #[test]
541    fn awp2_interpolation_uses_the_casa_float_sinc_table() {
542        let table: Vec<_> = (-4000..4000)
543            .map(|i| sinc((i as f64 / 1000.0) as f32))
544            .collect();
545        // A single nonzero input also isolates the sinc-table multiplication
546        // from roundoff in a general convolution.
547        let mut input = [Complex32::default(); 12 * 12];
548        input[6 * 12 + 6] = Complex32::new(0.9771961, 0.0);
549        let taps = lanczos_axis(12, 2312, 1400, &table);
550        let value = interpolate(
551            &input,
552            12,
553            taps[757].as_ref().unwrap(),
554            taps[679].as_ref().unwrap(),
555        );
556        // sinc(0.295) sinc(0.098) sinc(-0.109) sinc(-0.037),
557        // evaluated at the four entries selected by MathUtils::sinc.
558        let expected = 0.9771961_f64
559            * [0.295_f64, 0.098, -0.109, -0.037]
560                .into_iter()
561                .map(|v| {
562                    let a = std::f64::consts::PI * v;
563                    a.sin() / a
564                })
565                .product::<f64>();
566        assert!(
567            (f64::from(value.re) - expected).abs() < 2.0e-7,
568            "{value} vs {expected}"
569        );
570        assert_eq!(value.im, 0.0);
571    }
