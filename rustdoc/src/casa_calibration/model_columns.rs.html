27const DATA: VisibilityDataColumn = VisibilityDataColumn::Data;
28const CORRECTED: VisibilityDataColumn = VisibilityDataColumn::CorrectedData;
29const MODEL: VisibilityDataColumn = VisibilityDataColumn::ModelData;

31pub(crate) fn reset_scratch_columns(
32    ms: &mut MeasurementSet,
33    selection: &MsSelection,
34    add_model: bool,
35    mut lock: TableWriteLock,
36) -> MsResult<TableWriteLock> {
37    let has_corrected = has_column(ms, CORRECTED);
38    let has_model = has_column(ms, MODEL);
39    let mut new = Vec::new();
40    if !has_corrected {
41        new.push(CORRECTED);
42    }
43    if add_model && !has_model {
44        new.push(MODEL);
45    }
46    if !new.is_empty() {
47        // CASA adds and initializes missing columns across the entire MS
48        // before resolving the user's selectors.
49        let all =
50            resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
51        lock = reset(ms, all, &new, true, lock)?;
52    }
53    if has_corrected {
54        let selected = resolve_casa_calibration_selection(ms, selection).map_err(invalid)?;
55        if selected.selected_rows.is_empty() {
56            return Err(invalid("selection matches no rows"));
57        }
58        let columns = if has_model {
59            &[CORRECTED, MODEL][..]
60        } else {
61            &[CORRECTED]
62        };
63        lock = reset(ms, selected, columns, false, lock)?;
64    }
65    Ok(lock)
66}

68/// Add one absent visibility scratch column before data selection.
69pub(crate) fn initialize_scratch_column(
70    ms: &mut MeasurementSet,
71    column: VisibilityDataColumn,
72    lock: TableWriteLock,
73) -> MsResult<TableWriteLock> {
74    if has_column(ms, column) {
75        return Ok(lock);
76    }
77    let all = resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
78    reset(ms, all, &[column], true, lock)
79}

81pub(crate) fn delete_model(
82    ms: &mut MeasurementSet,
83    field: Option<&str>,
84    virtual_model: bool,
85    scratch: bool,
86    lock: &mut TableWriteLock,
87) -> MsResult<()> {
88    if virtual_model {
89        let field = field.filter(|value| !value.is_empty() && !value.eq_ignore_ascii_case("false"));
90        let fields = field.map_or_else(
91            || Ok(Vec::new()),
92            |text| ms.resolve_field_selector(if text.trim().is_empty() { "*" } else { text }),
93        )?;
94        casa_ms::virtual_model::clear_virtual_model_holding(ms, &fields, &[], lock)?;
95    }
96    if scratch {
97        scratch_columns::remove_scratch_model_holding(ms, lock)?;
98    }
99    Ok(())
100}

102pub(crate) fn has_column(ms: &MeasurementSet, column: VisibilityDataColumn) -> bool {
103    ms.main_table()
104        .schema()
105        .is_some_and(|schema| schema.contains_column(column.name()))
106}

108pub(crate) fn observed_column(ms: &MeasurementSet) -> &'static str {
109    let float = OptionalMainColumn::FloatData.name();
110    if ms
111        .main_table()
112        .schema()
113        .is_some_and(|schema| schema.contains_column(float))
114    {
115        float
116    } else {
117        DATA.name()
118    }
119}

121/// Legacy calibration iterators visit the inclusive span of all selected segments.
122pub(crate) fn calibration_channel_span(selection: &ChannelSelection) -> Option<Range<usize>> {
123    let first = selection
124        .segments
125        .iter()
126        .map(|segment| segment.start.min(segment.end))
127        .min()?;
128    let last = selection
129        .segments
130        .iter()
131        .map(|segment| segment.start.max(segment.end))
132        .max()?;
133    Some(first as usize..last as usize + 1)
134}

136fn reset(
137    ms: &mut MeasurementSet,
138    selected: ResolvedMsSelection,
139    columns: &[VisibilityDataColumn],
140    create: bool,
141    mut lock: TableWriteLock,
142) -> MsResult<TableWriteLock> {
143    let (plan, destinations) = scratch_write_plan(
144        ms,
145        selected.row_indices().collect(),
146        columns,
147        create,
148        &mut lock,
149    )?;
150    let batch_rows = plan.batch_rows().max(1);
151    let polarizations = if columns.contains(&MODEL) {
152        model_polarizations(ms)?
153    } else {
154        BTreeMap::new()
155    };
156    let ranges = selected
157        .channel_selections
158        .iter()
159        .filter_map(|(&spw, selection)| Some((spw, calibration_channel_span(selection)?)))
160        .collect::<BTreeMap<_, _>>();
161    let mut session =
162        MeasurementSetWriteSession::start_selected_row_mutation_holding(ms, plan, lock)
163            .map_err(invalid)?;
164    for rows in selected.selected_rows.chunks(batch_rows) {
165        let batch = reset_batch(ms, rows, &destinations, &ranges, &polarizations)?;
166        session.write_mutation_batch(ms, batch).map_err(invalid)?;
167    }
168    let (_, lock) = session.finish_mutation_holding().map_err(invalid)?;
169    Ok(lock)
170}

172pub(crate) fn scratch_write_plan(
173    ms: &mut MeasurementSet,
174    row_indices: Vec<usize>,
175    columns: &[VisibilityDataColumn],
176    create: bool,
177    lock: &mut TableWriteLock,
178) -> MsResult<(MeasurementSetWritePlan, Vec<ScratchDestination>)> {
179    let destinations = columns
180        .iter()
181        .map(|&column| ScratchDestination::new(ms, column))
182        .collect::<MsResult<Vec<_>>>()?;
183    for &column in columns {
184        scratch_columns::prepare_write_holding(ms, column, create, lock)?;
185    }
186    if create {
187        for &column in columns {
188            let keyword = ms
189                .main_table()
190                .column_keywords(observed_column(ms))
191                .cloned();
192            let schema =
193                casa_ms::column_def::build_table_schema(&[*column.optional_column().column_def()])
194                    .map_err(invalid)?;
195            let definition = schema
196                .column(column.name())
197                .expect("scratch column schema")
198                .clone();
199            ms.main_table_mut().add_column(definition, None)?;
200            if let Some(keyword) = keyword {
201                ms.main_table_mut()
202                    .set_column_keywords(column.name(), keyword);
203            }
204        }
205    }
206    let sizes = maximum_visibility_cell_elements(ms).map_err(invalid)?;
207    let write_columns = destinations
208        .iter()
209        .flat_map(|column| column.plans(sizes, create))
210        .collect();
211    let resources = MeasurementSetWriteResources::from_system_memory(2).map_err(invalid)?;
212    let plan =
213        MeasurementSetWritePlan::selected_row_mutation(row_indices, write_columns, resources)
214            .map_err(invalid)?;
215    Ok((plan, destinations))
216}

218fn model_polarizations(ms: &MeasurementSet) -> MsResult<BTreeMap<i32, Vec<Complex32>>> {
219    let polarization = ms.polarization()?;
220    (0..polarization.row_count())
221        .map(|id| {
222            let values = polarization
223                .corr_type(id)?
224                .into_iter()
225                .map(initial_model_visibility)
226                .collect();
227            Ok((id as i32, values))
228        })
229        .collect()
230}

232fn reset_batch(
233    ms: &MeasurementSet,
234    rows: &[ResolvedMsSelectionRow],
235    columns: &[ScratchDestination],
236    ranges: &BTreeMap<i32, Range<usize>>,
237    polarizations: &BTreeMap<i32, Vec<Complex32>>,
238) -> MsResult<MeasurementSetMutationBatch> {
239    let row_indices = rows.iter().map(|row| row.row_index).collect::<Vec<_>>();
240    let main = ms.main_table();
241    let data = main.column_accessor(observed_column(ms))?;
242    let shapes = data.array_shapes_uncached(&row_indices)?;
243    let mut data = columns
244        .iter()
245        .any(|column| column.visibility == CORRECTED)
246        .then(|| data.array_cells_owned_uncached(&row_indices))
247        .transpose()?;
248    let needs_preserved_channels = rows
249        .iter()
250        .any(|row| ranges.contains_key(&row.spectral_window_id));
251    let mut batch = Vec::with_capacity(columns.len());
252    for destination in columns {
253        let column = destination.visibility;
254        let mut previous = needs_preserved_channels
255            .then(|| {
256                main.column_accessor(column.name())?
257                    .array_cells_owned_uncached(&row_indices)
258            })
259            .transpose()?;
260        let mut values = Vec::with_capacity(rows.len());
261        for (index, row) in rows.iter().enumerate() {
262            let shape = shapes[index]
263                .as_deref()
264                .ok_or_else(|| invalid("DATA is undefined"))?;
265            let &[correlations, channels] = shape else {
266                return Err(invalid("visibility data must have two axes"));
267            };
268            let replacement = if column == MODEL {
269                let model = polarizations
270                    .get(&row.polarization_id)
271                    .ok_or_else(|| invalid("invalid POLARIZATION_ID"))?;
272                if model.len() != correlations {
273                    return Err(invalid("POLARIZATION does not agree with DATA"));
274                }
275                ArrayValue::Complex32(ArrayD::from_shape_fn(IxDyn(shape).f(), |position| {
276                    model[position[0]]
277                }))
278            } else {
279                let observed = data
280                    .as_mut()
281                    .and_then(|rows| rows[index].take())
282                    .ok_or_else(|| invalid("observed visibility is undefined"))?;
283                match observed {
284                    ArrayValue::Float32(array) => {
285                        ArrayValue::Complex32(array.mapv(|value| Complex32::new(value, 0.0)))
286                    }
287                    complex => complex,
288                }
289            };
290            if let Some(range) = ranges.get(&row.spectral_window_id) {
291                if range.end > channels {
292                    return Err(invalid("spectral-window selection exceeds DATA shape"));
293                }
294                let mut previous = previous
295                    .as_mut()
296                    .and_then(|rows| rows[index].take())
297                    .ok_or_else(|| invalid("scratch column is undefined"))?;
298                copy_channels(&mut previous, &replacement, range.clone())?;
299                values.push(previous);
300            } else {
301                values.push(replacement);
302            }
303        }
304        batch.extend(destination.batch(ms, &row_indices, rows, ranges, values)?);
305    }
306    Ok(MeasurementSetMutationBatch {
307        row_indices,
308        columns: batch,
309    })
310}

312fn copy_channels(
313    current: &mut ArrayValue,
314    replacement: &ArrayValue,
315    channels: Range<usize>,
316) -> MsResult<()> {
317    let (ArrayValue::Complex32(current), ArrayValue::Complex32(replacement)) =
318        (current, replacement)
319    else {
320        return Err(invalid("expected complex scratch visibilities"));
321    };
322    if current.shape() != replacement.shape() {
323        return Err(invalid("scratch and DATA shapes do not match"));
324    }
325    for channel in channels {
326        current
327            .index_axis_mut(ndarray::Axis(1), channel)
328            .assign(&replacement.index_axis(ndarray::Axis(1), channel));
329    }
330    Ok(())
331}
