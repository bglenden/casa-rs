17fn evaluate_typed<T: ImageExprValue + PartialOrd + ExprValueConvert>(
18    expression: &str,
19    sources: HashMap<String, &dyn ImageInterface<T>>,
20    shape: &[usize],
21) -> PyResult<ArrayD<bool>> {
22    let compiled = parse_mask_expr(expression, &HashMapResolver(sources)).map_err(|error| {
23        if error.message.contains("boolean") || error.message.contains("Boolean") {
24            raised("Exception: The expression type must be Boolean.")
25        } else {
26            raised(format!("Image Expression: {error}"))
27        }
28    })?;
29    if compiled.shape().is_empty() {
30        return Ok(ArrayD::from_elem(
31            IxDyn(shape).f(),
32            compiled.get_at(&[]).map_err(raised)?,
33        ));
34    }
35    if compiled.shape() != shape {
36        return Err(raised(format!(
37            "The shape of the expression does not conform with the shape of the output imageExpression shape = {:?}Image shape      = {shape:?}",
38            compiled.shape()
39        )));
40    }
41    compiled.get().map_err(raised)
42}

44fn evaluate(expression: &str, shape: &[usize]) -> PyResult<ArrayD<bool>> {
45    if expression.is_empty() {
46        return Err(raised("Exception: You must specify an expression."));
47    }
48    let images = referenced_images(expression).map_err(raised)?
49        .into_iter()
50        .map(|name| {
51            let image = AnyPagedImage::open(&name).map_err(|_| {
52                raised(format!("ImageExprParse: '{name}' is an unknown lattice or image or it is an unqualified region"))
53            })?;
54            Ok((name, image))
55        })
56        .collect::<PyResult<Vec<_>>>()?;
57    if images
58        .iter()
59        .all(|(_, image)| matches!(image, AnyPagedImage::Float32(_)))
60    {
61        let images = images
62            .iter()
63            .filter_map(|(name, image)| match image {
64                AnyPagedImage::Float32(image) => {
65                    Some((name.clone(), image as &dyn ImageInterface<f32>))
66                }
67                _ => None,
68            })
69            .collect();
70        evaluate_typed(expression, images, shape)
71    } else if images
72        .iter()
73        .all(|(_, image)| matches!(image, AnyPagedImage::Float64(_)))
74    {
75        let images = images
76            .iter()
77            .filter_map(|(name, image)| match image {
78                AnyPagedImage::Float64(image) => {
79                    Some((name.clone(), image as &dyn ImageInterface<f64>))
80                }
81                _ => None,
82            })
83            .collect();
84        evaluate_typed(expression, images, shape)
85    } else {
86        Err(stub("calcmask").unwrap_err())
87    }
88}

90#[pymethods]
91impl PyImageTool {
92    #[pyo3(signature = (mask="", name="", asdefault=true), text_signature = "($self, mask='', name='', asdefault=True)")]
93    fn calcmask(&self, py: Python<'_>, mask: &str, name: &str, asdefault: bool) -> PyResult<bool> {
94        py.detach(|| {
95            let state = self.state();
96            let Some(image) = state.image.as_ref() else { return Ok(false) };
97            let mut image = image.lock();
98            let values = evaluate(mask, image.shape())?;
99            let generated;
100            let mask_name = if name.is_empty() {
101                generated = (0..)
102                    .map(|index| format!("mask{index}"))
103                    .find(|name| !has_region(&image, name))
104                    .expect("a fresh mask name");
105                generated.as_str()
106            } else {
107                validate_name(name)?;
108                name
109            };
110            with_pixels!(&mut *image, storage => with_storage!(storage, image => image.put_mask(mask_name, &values))).map_err(raised)?;
111            if asdefault {
112                with_pixels!(&mut *image, storage => with_storage!(storage, image => image.set_default_mask(mask_name))).map_err(raised)?;
113            }
114            if state.history {
115                image.record_history("calcmask", format!(
116                    "ia.calcmask(mask={mask:?}, name={name:?}, asdefault={asdefault})"
117                ))?;
118            }
119            image.save().map_err(raised)?;
120            Ok(true)
121        })
122    }
