12pub(super) fn draw(
13    py: Python<'_>,
14    document: &casa_plot_data::plot_document::PlotDocument,
15    payload: &std::path::Path,
16    preserve_title: bool,
17) -> PyResult<()> {
18    py.import("casars._plot_document")?.call_method1(
19        "render_current",
20        (pythonize::pythonize(py, document)?, payload, preserve_title),
21    )?;
22    Ok(())
23}
24
25fn pyplot(py: Python<'_>) -> PyResult<Bound<'_, PyModule>> {
26    py.import("matplotlib.pyplot")
27}
28
29fn small_ticks(plot: &Bound<'_, PyModule>) -> PyResult<()> {
30    let axis = plot.call_method0("gca")?;
31    let kwargs = [("fontsize", "x-small")].into_py_dict(plot.py())?;
32    for method in ["get_xticklabels", "get_yticklabels"] {
33        plot.call_method("setp", (axis.call_method0(method)?,), Some(&kwargs))?;
34    }
35    Ok(())
36}

38#[pymethods]
39impl PySimutil {
40    /// Draw the source elevation, transit and optional observation range.
41    /// A known telescope supplies its location; `cofa` supplies an unknown
42    /// telescope's. A missing `direction` uses this object's direction.
43    #[pyo3(signature = (date, usehourangle = true, direction = None, telescope = None, ms = None, cofa = None), text_signature = "($self, date, usehourangle=True, direction=None, telescope=None, ms=None, cofa=None)")]
44    #[allow(clippy::too_many_arguments)]
45    fn ephemeris(
46        &self,
47        py: Python<'_>,
48        date: Variant,
49        usehourangle: bool,
50        direction: Option<&Bound<'_, PyAny>>,
51        telescope: Option<String>,
52        ms: Option<std::path::PathBuf>,
53        cofa: Option<Variant>,
54    ) -> PyResult<()> {
55        use casa_plot_data::simulation::{EphemerisPlotRequest, ephemeris_plot};
56        use casa_types::measures::{
57            MPosition,
58            direction::{DirectionRef, MDirection},
59        };
60        use pyo3::exceptions::{PyAttributeError, PyRuntimeError, PyValueError};
61        use std::str::FromStr;
62
63        let argument = direction.unwrap_or(self.direction.bind(py));
64        let shown = argument.str()?.to_string();
65        let split = super::ops::directions(argument)?
66            .split()
67            .map_err(super::ops::raised)?;
68        let reference = DirectionRef::from_str(split.epoch.trim())
69            .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
70        let source = MDirection::from_angles(
71            value_in(&split.longitude, "rad")?,
72            value_in(&split.latitude, "rad")?,
73            reference,
74        );
75        let telescope = match telescope {
76            Some(telescope) => telescope,
77            None => self
78                .telescopename
79                .as_ref()
80                .ok_or_else(|| {
81                    PyAttributeError::new_err("'simutil' object has no attribute 'telescopename'")
82                })?
83                .extract::<String>(py)?,
84        };
85        let date = value_in(&super::ops::quantity(&date)?, "d")?;
86        let measurement_set = ms.as_deref().filter(|path| !path.as_os_str().is_empty());
87        let duration_hours = if measurement_set.is_some() {
88            0.0
89        } else {
90            let duration = super::ops::quantity(&self.totaltime.extract::<Variant>(py)?)?;
91            if duration.value() == 0.0 {
92                0.0
93            } else {
94                value_in(&duration, "h")?
95            }
96        };
97        let file = tempfile::NamedTempFile::new()?;
98        let document = py.detach(|| {
99            let data = casa_ms::open_measures_runtime().map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
100            let position = MPosition::from_observatory_name(&telescope, &*data)
101                .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
102            let position = match position {
103                Some(position) => position,
104                None => cofa.as_ref().map(|value| measures::position(value, &data)).transpose()?.flatten()
105                    .ok_or_else(|| PyRuntimeError::new_err("simutil::ephemeris needs either a known observatory or an explicitly specified cofa measure"))?,
106            };
107            ephemeris_plot(&EphemerisPlotRequest { source, date_mjd: date, position, measures: data,
108                use_hour_angle: usehourangle, measurement_set, duration_hours }, file.path())
109                .map_err(|error| if error.to_string().contains("never visible") { PyValueError::new_err(format!("{shown} is not visible from {telescope}")) }
110                    else { PyRuntimeError::new_err(error.to_string()) })
111        })?;
112        draw(py, &document, file.path(), true)
113    }
114
115    /// Plot all antenna positions and dish sizes in their local tangent
116    /// plane on the current axes, retaining CASA's marker/label choices.
117    #[pyo3(signature = (x, y, z, d, name), text_signature = "($self, x, y, z, d, name)")]
118    #[allow(clippy::too_many_arguments)]
119    fn plotants(
120        &self,
121        py: Python<'_>,
122        x: DoubleList,
123        y: DoubleList,
124        z: DoubleList,
125        d: DoubleList,
126        name: StringList,
127    ) -> PyResult<()> {
128        use pyo3::exceptions::PyRuntimeError;
129        if x.0.len() != y.0.len() || x.0.len() != z.0.len() {
130            return Err(PyRuntimeError::new_err("x, y and z have different lengths"));
131        }
132        let file = tempfile::NamedTempFile::new()?;
133        let positions: Vec<_> =
134            x.0.iter()
135                .zip(y.0)
136                .zip(z.0)
137                .map(|((&x, y), z)| [x, y, z])
138                .collect();
139        let document = py
140            .detach(|| {
141                casa_plot_data::simulation::antenna_plot(&positions, &d.0, &name.0, file.path())
142            })
143            .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
144        draw(py, &document, file.path(), true)
145    }

147    /// Clear the current figure and optionally select a multi-panel layout.
148    /// The caller's `multi` list is retained and advanced by `nextfig`, as
149    /// in CASA. `show` selects Matplotlib's interactive state.
150    #[pyo3(signature = (multi = None, show = true), text_signature = "($self, multi=0, show=True)")]
151    fn newfig(
152        &mut self,
153        py: Python<'_>,
154        multi: Option<Bound<'_, PyAny>>,
155        show: bool,
156    ) -> PyResult<()> {
157        let plot = pyplot(py)?;
158        plot.call_method0(if show { "ion" } else { "ioff" })?;
159        plot.call_method0("clf")?;
160        if let Some(multi) = multi {
161            if multi.eq(0)? {
162                return Ok(());
163            }
164            if !multi.is_instance_of::<PyList>() || multi.len()? != 3 {
165                emit(
166                    py,
167                    LOGGER,
168                    &[LogLine::warning(format!(
169                        "internal error setting multi-panel figure with multi={}",
170                        multi.str()?
171                    ))],
172                )?;
173            }
174            self.pmulti = multi.clone().unbind();
175            plot.call_method1(
176                "subplot",
177                (multi.get_item(0)?, multi.get_item(1)?, multi.get_item(2)?),
178            )?;
179            let kwargs = [
180                ("left", 0.05),
181                ("right", 0.98),
182                ("bottom", 0.09),
183                ("top", 0.95),
184                ("hspace", 0.2),
185                ("wspace", 0.2),
186            ]
187            .into_py_dict(py)?;
188            plot.call_method("subplots_adjust", (), Some(&kwargs))?;
189        }
190        Ok(())
191    }

193    /// Apply CASA's small tick labels and advance to the next subplot.
194    /// Beyond the last panel, keep the current axes and continue advancing
195    /// the retained list, matching CASA's stateful behavior.
196    #[pyo3(signature = (), text_signature = "($self)")]
197    fn nextfig(&self, py: Python<'_>) -> PyResult<()> {
198        let plot = pyplot(py)?;
199        small_ticks(&plot)?;
200        let multi = self.pmulti.bind(py);
201        if !multi.eq(0)? {
202            let next = multi.get_item(2)?.extract::<i64>()? + 1;
203            multi.set_item(2, next)?;
204            let rows = multi.get_item(0)?.extract::<i64>()?;
205            let columns = multi.get_item(1)?.extract::<i64>()?;
206            if next <= rows * columns {
207                plot.call_method1("subplot", (rows, columns, next))?;
208            }
209        }
210        Ok(())
211    }
212
213    /// Apply small tick labels, draw when requested, save when a filename is
214    /// supplied and disable interactive mode, as CASA does.
215    #[pyo3(signature = (show = true, filename = ""), text_signature = "($self, show=True, filename='')")]
216    fn endfig(&self, py: Python<'_>, show: bool, filename: &str) -> PyResult<()> {
217        let plot = pyplot(py)?;
218        small_ticks(&plot)?;
219        if show {
220            plot.call_method0("draw")?;
221        }
222        if !filename.is_empty() {
223            plot.call_method1("savefig", (filename,))?;
224        }
225        plot.call_method0("ioff")?;
226        Ok(())
227    }
