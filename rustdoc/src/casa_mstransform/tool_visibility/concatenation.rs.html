13pub(crate) struct MemberRows<'a> {
14    table: &'a Table,
15    rows: Vec<usize>,
16    output_positions: Vec<usize>,
17}
18
19pub(crate) fn member_rows<'a>(
20    tables: &'a [Table],
21    total_rows: usize,
22    rows: &[usize],
23) -> Vec<MemberRows<'a>> {
24    if tables.is_empty()
25        || rows.is_empty()
26        || tables.iter().map(Table::row_count).sum::<usize>() != total_rows
27    {
28        return Vec::new();
29    }
30    let mut permutation = Vec::new();
31    let ordered_rows;
32    let mut tail = rows;
33    if !rows.is_sorted() {
34        permutation.extend(0..rows.len());
35        permutation.sort_unstable_by_key(|&slot| rows[slot]);
36        ordered_rows = permutation
37            .iter()
38            .map(|&slot| rows[slot])
39            .collect::<Vec<_>>();
40        tail = &ordered_rows;
41    }
42    if tail.last().is_some_and(|&row| row >= total_rows) {
43        return Vec::new();
44    }
45    let mut offset = 0;
46    let mut positions = permutation.as_slice();
47    let mut selected = Vec::new();
48    for table in tables {
49        if tail.is_empty() {
50            break;
51        }
52        let end = offset + table.row_count();
53        let count = tail.partition_point(|&row| row < end);
54        let (local, remaining) = tail.split_at(count);
55        if !local.is_empty() {
56            let output_positions = if positions.is_empty() {
57                Vec::new()
58            } else {
59                let (assigned, next) = positions.split_at(count);
60                positions = next;
61                assigned.to_vec()
62            };
63            selected.push(MemberRows {
64                table,
65                rows: local.iter().map(|row| row - offset).collect(),
66                output_positions,
67            });
68        }
69        tail = remaining;
70        offset = end;
71    }
72    selected
73}

75fn restore_order<T: Copy>(values: Vec<T>, members: &[MemberRows<'_>], elements: usize) -> Vec<T> {
76    if values.is_empty() || members[0].output_positions.is_empty() {
77        return values;
78    }
79    let mut restored = vec![values[0]; values.len()];
80    let positions = members
81        .iter()
82        .flat_map(|part| part.output_positions.iter().copied());
83    for (source, destination) in values.chunks_exact(elements).zip(positions) {
84        let start = destination * elements;
85        restored[start..start + elements].copy_from_slice(source);
86    }
87    restored
88}

90pub(crate) fn concatenate_scalars(
91    members: &[MemberRows<'_>],
92    columns: &[&str],
93) -> MsResult<HashMap<String, RequiredScalarColumnValues>> {
94    use RequiredScalarColumnValues::{Bool, Float32, Float64, Int32};
95
96    let mut values = HashMap::with_capacity(columns.len());
97    for part in members {
98        for (name, loaded) in part
99            .table
100            .required_scalar_columns_owned_for_rows(columns, &part.rows)?
101        {
102            match values.entry(name) {
103                Entry::Vacant(entry) => {
104                    entry.insert(loaded);
105                }
106                Entry::Occupied(mut entry) => match (entry.get_mut(), loaded) {
107                    (Bool(result), Bool(part)) => result.extend(part),
108                    (Int32(result), Int32(part)) => result.extend(part),
109                    (Float32(result), Float32(part)) => result.extend(part),
110                    (Float64(result), Float64(part)) => result.extend(part),
111                    _ => return Err(input("concatenated scalar columns have different types")),
112                },
113            }
114        }
115    }
116    if members
117        .first()
118        .is_some_and(|member| !member.output_positions.is_empty())
119    {
120        for column in values.values_mut() {
121            match column {
122                Bool(values) => *values = restore_order(std::mem::take(values), members, 1),
123                Int32(values) => *values = restore_order(std::mem::take(values), members, 1),
124                Float32(values) => *values = restore_order(std::mem::take(values), members, 1),
125                Float64(values) => *values = restore_order(std::mem::take(values), members, 1),
126            }
127        }
128    }
129    Ok(values)
130}

132pub(super) fn concatenate_arrays(
133    members: &[MemberRows<'_>],
134    read: impl Fn(&Table, &[usize]) -> MsResult<ArrayValue>,
135) -> MsResult<ArrayValue> {
136    let (initial_member, remaining) = members.split_first().expect("nonempty selected members");
137    let first = read(initial_member.table, &initial_member.rows)?;
138    if remaining.is_empty() && initial_member.output_positions.is_empty() {
139        return Ok(first);
140    }
141    let total_rows = remaining.iter().map(|part| part.rows.len()).sum::<usize>();
142    macro_rules! concatenate {
143        ($variant:ident, $first:ident) => {{
144            let mut shape = $first.shape().to_vec();
145            let axis = shape.len() - 1;
146            if shape[axis] != initial_member.rows.len() {
147                return Err(input(
148                    "concatenated array row count does not match its selection",
149                ));
150            }
151            let initial = mem_order(&$first)?;
152            let elements = shape[..axis].iter().product();
153            let expected = (shape[axis] + total_rows)
154                .checked_mul(elements)
155                .ok_or_else(|| input("concatenated visibility array is too large"))?;
156            let mut values = if initial_member.output_positions.is_empty() || initial.is_empty() {
157                Vec::with_capacity(expected)
158            } else {
159                vec![initial[0]; expected]
160            };
161            append_cells(&mut values, initial, initial_member, elements);
162            drop($first);
163            for part in remaining {
164                let ArrayValue::$variant(block) = read(part.table, &part.rows)? else {
165                    return Err(input("concatenated array columns have different types"));
166                };
167                if block.ndim() != shape.len()
168                    || block.shape()[..axis] != shape[..axis]
169                    || block.shape()[axis] != part.rows.len()
170                {
171                    return Err(input(
172                        "concatenated array columns have different cell shapes",
173                    ));
174                }
175                shape[axis] += block.shape()[axis];
176                append_cells(&mut values, mem_order(&block)?, part, elements);
177            }
178            Ok(ArrayValue::$variant(shaped(&shape, values)?))
179        }};
180    }
181    match first {
182        ArrayValue::Bool(values) => concatenate!(Bool, values),
183        ArrayValue::Float32(values) => concatenate!(Float32, values),
184        ArrayValue::Float64(values) => concatenate!(Float64, values),
185        ArrayValue::Complex32(values) => concatenate!(Complex32, values),
186        ArrayValue::Complex64(values) => concatenate!(Complex64, values),
187        _ => Err(input("unsupported concatenated array column type")),
188    }
189}

191fn append_cells<T: Copy>(
192    output: &mut Vec<T>,
193    cells: &[T],
194    member: &MemberRows<'_>,
195    elements: usize,
196) {
197    if member.output_positions.is_empty() {
198        output.extend_from_slice(cells);
199    } else if elements != 0 {
200        for (cell, &position) in cells.chunks_exact(elements).zip(&member.output_positions) {
201            let first = position * elements;
202            output[first..first + elements].copy_from_slice(cell);
203        }
204    }
205}

207fn mem_order<T>(block: &ArrayD<T>) -> MsResult<&[T]> {
208    block
209        .as_slice_memory_order()
210        .ok_or_else(|| input("visibility block is not contiguous"))
