3//! Typed source catalog for continuum products.
4//!
5//! Every source role keeps its own typed owner record. The catalog is minted
6//! only from a whole released Major-Cycle join, so one atomic reconciliation
7//! result carries its multiple distinct completions together and no role's
8//! evidence can be substituted for another.
9
10use casa_imaging_model::{CompiledProblem, LogicalIdentity, ProductGraphId};
11use casa_imaging_reconstruction::{
12    FinalNormalState, FinalNormalStateCompletionId, MajorCycleCompletion, MajorCycleCompletionId,
13    ModelGeneration, ModelGenerationId, NormalStateCatalog,
14};
15
16use crate::digest::{COMMITMENT_DOMAIN, COMMITMENT_VERSION, Encoder};
17use crate::error::ProductsError;
18
19/// Identity catalog of every source role behind one continuum generation.
20///
21/// Constructed only by [`ContinuumSourceCatalog::from_major_cycle`]; there is
22/// no per-field public constructor, so a commitment set always describes one
23/// inseparable reconciliation lineage.
24#[derive(Debug, Clone)]
25pub struct ContinuumSourceCatalog {
26    problem: CompiledProblem,
27    graph_id: ProductGraphId,
28    major_cycle_completion: MajorCycleCompletionId,
29    normal_state_completion: FinalNormalStateCompletionId,
30    normal_state_content: LogicalIdentity,
31    normal_state_catalog: NormalStateCatalog,
32    input_model_generation: ModelGenerationId,
33    final_model_generation: ModelGenerationId,
34    weighting_generation: casa_imaging_reconstruction::WeightingGenerationId,
35    replay: casa_imaging_reconstruction::WeightingReplayId,
36    coverage: casa_imaging_reconstruction::WeightingReplayCoverageId,
37    selected_generation: casa_imaging_model::SelectedObservationGenerationId,
38    sample_count: u64,
39    block_count: u64,
40}
41
42impl ContinuumSourceCatalog {
43    /// Mint the catalog from one whole released Major-Cycle join.
44    ///
45    /// The problem must be the exact compiled problem the join reconciled.
46    ///
47    /// # Errors
48    ///
49    /// Rejects joins from another compiled problem and unsupported catalogs.
50    pub fn from_major_cycle(
51        problem: &CompiledProblem,
52        join: &MajorCycleCompletion,
53    ) -> Result<Self, ProductsError> {
54        let normal_state = join.normal_state();
55        if normal_state.problem_id() != problem.problem_id() {
56            return Err(ProductsError::SourceLineageMismatch);
57        }
58        if join.model_completion().problem() != problem.problem_id() {
59            return Err(ProductsError::SourceLineageMismatch);
60        }
61        if matches!(
62            normal_state.catalog(),
63            NormalStateCatalog::UnnormalizedNterms1V1
64        ) {
65            // The only supported continuum catalog.
66        } else {
67            return Err(ProductsError::UnsupportedProblem);
68        }
69        Ok(Self {
70            graph_id: problem.product_graph().graph_id(),
71            major_cycle_completion: join.completion_id(),
72            normal_state_completion: normal_state.completion_id(),
73            normal_state_content: normal_state.content_identity(),
74            normal_state_catalog: normal_state.catalog(),
75            input_model_generation: normal_state.input_model_generation(),
76            final_model_generation: normal_state.final_model_generation(),
77            weighting_generation: normal_state.weighting_generation(),
78            replay: normal_state.replay_id(),
79            coverage: normal_state.coverage(),
80            selected_generation: normal_state.selected_generation(),
81            sample_count: normal_state.sample_count(),
82            block_count: normal_state.block_count(),
83            problem: problem.clone(),
84        })
85    }
86
87    /// Borrow the exact compiled problem this lineage reconciled.
88    #[must_use]
89    pub const fn problem(&self) -> &CompiledProblem {
90        &self.problem
91    }
92
93    /// Return the compiler-owned Product Graph identity of this lineage.
94    #[must_use]
95    pub const fn graph_id(&self) -> ProductGraphId {
96        self.graph_id
97    }
98
99    /// Return the Major-Cycle completion identity of this lineage.
100    #[must_use]
101    pub const fn major_cycle_completion(&self) -> MajorCycleCompletionId {
102        self.major_cycle_completion
103    }
104
105    /// Return the final model generation named by this lineage.
106    #[must_use]
107    pub const fn final_model_generation(&self) -> ModelGenerationId {
108        self.final_model_generation
109    }
110
111    /// Return the schema-versioned digest binding every role's evidence.
112    #[must_use]
113    pub fn commitment_id(&self) -> [u8; 32] {
114        let mut encoder = Encoder::new(COMMITMENT_DOMAIN, COMMITMENT_VERSION);
115        encoder.identity(self.problem.problem_id().as_bytes());
116        encoder.identity(self.graph_id.as_bytes());
117        encoder.identity(self.major_cycle_completion.as_bytes());
118        encoder.identity(self.normal_state_completion.as_bytes());
119        encoder.identity(self.normal_state_content.as_bytes());
120        encoder.u8(match self.normal_state_catalog {
121            NormalStateCatalog::UnnormalizedNterms1V1 => 0,
122        });
123        encoder.identity(self.input_model_generation.as_bytes());
124        encoder.identity(self.final_model_generation.as_bytes());
125        encoder.identity(self.weighting_generation.as_bytes());
126        encoder.identity(self.replay.as_bytes());
127        encoder.identity(self.coverage.as_bytes());
128        encoder.identity(self.selected_generation.as_bytes());
129        encoder.u64(self.sample_count);
130        encoder.u64(self.block_count);
131        encoder.finish()
132    }
133}

135/// Borrowed scientific payloads for producing planned members.
136///
137/// Like the catalog, this value is minted only from a whole join plus its
138/// exact compiled problem, so production always consumes the same atomic
139/// reconciliation result the commitments name.
140#[derive(Debug)]
141pub struct ContinuumProductInputs<'a> {
142    problem: &'a CompiledProblem,
143    normal_state: &'a FinalNormalState,
144    final_model: &'a ModelGeneration,
145}
146
147impl<'a> ContinuumProductInputs<'a> {
148    /// Bind the borrowed payloads of one released join.
149    ///
150    /// # Errors
151    ///
152    /// Rejects joins whose normal state belongs to another compiled problem.
153    pub fn from_major_cycle(
154        problem: &'a CompiledProblem,
155        join: &'a MajorCycleCompletion,
156    ) -> Result<Self, ProductsError> {
157        if join.normal_state().problem_id() != problem.problem_id() {
158            return Err(ProductsError::SourceLineageMismatch);
159        }
160        Ok(Self {
161            problem,
162            normal_state: join.normal_state(),
163            final_model: join.final_model(),
164        })
165    }
166
167    /// Borrow the exact compiled problem behind these payloads.
168    #[must_use]
169    pub const fn problem(&self) -> &CompiledProblem {
170        self.problem
171    }
172
173    /// Return radians-per-pixel on each direction axis of the main domain.
174    #[must_use]
175    pub fn cell_size_rad(&self) -> [f64; 2] {
176        let domain = &self.problem.geometry().domains()[0];
177        let increment = domain.direction().increment_rad();
178        [increment[0].abs(), increment[1].abs()]
179    }
180
181    /// Borrow the authoritative normal state (residual, PSF, sensitivity).
182    #[must_use]
183    pub const fn normal_state(&self) -> &FinalNormalState {
184        self.normal_state
185    }
186
187    /// Borrow the authoritative final model generation.
188    #[must_use]
189    pub const fn final_model(&self) -> &ModelGeneration {
190        self.final_model
191    }
