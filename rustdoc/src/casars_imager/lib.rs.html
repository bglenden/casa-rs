4//! The command-line and task surface of native CASA-RS imaging.
5//!
6//! Both routes resolve the imager's parameters through the provider-
7//! contracts catalog: command-line flags and the catalog-named JSON of
8//! `--json-run` become the same resolved values, and those values, and
9//! nothing else, make the [`ImagingRequest`] the application runs.

27fn imager_surface() -> Result<SurfaceContractBundle, String> {
28    casa_provider_contracts::builtin_surface_bundle("imager")
29}

31fn workspace() -> Result<PathBuf, String> {
32    std::env::current_dir().map_err(|error| format!("the working directory: {error}"))
33}

35/// The request `resolved`, every active imager parameter, makes.
36fn request_from(
37    resolved: Map<String, Value>,
38) -> Result<(ImagingParameters, ImagingRequest), String> {
39    let request = serde_json::from_value(Value::Object(resolved.clone()))
40        .map_err(|error| format!("imager parameters: {error}"))?;
41    Ok((ImagingParameters(resolved), request))
42}
43
44/// Resolve `parameters`, catalog-named and sparse, over the catalog's
45/// defaults: the resolved parameters and the request they make.
46pub fn resolve_request(
47    parameters: &ImagingParameters,
48) -> Result<(ImagingParameters, ImagingRequest), String> {
49    let resolved = resolve_plain_values(imager_surface()?, workspace()?, &parameters.0)
50        .map_err(|error| format!("resolve imager parameters: {error}"))?;
51    request_from(resolved)
52}
53
54/// Resolve command-line flags over the catalog's defaults: the resolved
55/// parameters and the request they make.
56pub fn resolve_cli_request(
57    args: &[OsString],
58) -> Result<(ImagingParameters, ImagingRequest), String> {
59    let bundle = imager_surface()?;
60    let overrides = parse_parameter_cli_overrides(&bundle, args)?;
61    let values = resolve_active_values(bundle, workspace()?, overrides)
62        .map_err(|error| format!("resolve imager parameters: {error}"))?;
63    request_from(
64        values
65            .into_iter()
66            .map(|(name, value)| (name, value.to_plain_json()))
67            .collect(),
68    )
69}

71/// Run `request`, resolved as `parameters`: the run summary, with the
72/// parameters as its echo, is written beside the products
73/// (`<imagename>.summary.json`); SIGINT cancels the run.
74pub fn run(
75    parameters: ImagingParameters,
76    request: &ImagingRequest,
77) -> Result<ImagerRunTaskResult, String> {
78    let started = Instant::now();
79    let mut summary = request.imagename.clone().into_os_string();
80    summary.push(".summary.json");
81    let context = RunContext {
82        host: HostResources::detect().map_err(|error| error.to_string())?,
83        policy: request.resource_policy(),
84        cancel: interrupt::token().clone(),
85        summary: Some(SummaryTarget {
86            path: summary.into(),
87            request: Value::Object(parameters.0.clone()),
88        }),
89    };
90    let outcome = execute(request, context).map_err(|error| error.to_string())?;
91    Ok(ImagerRunTaskResult::from_outcome(
92        parameters,
93        request,
94        &outcome,
95        started.elapsed(),
96    ))
97}

99/// Run the command line: a machine action (`--json-run`, `--json-schema`,
100/// `--protocol-info`), `--help`, or imaging parameter flags.
101/// `--managed-output` prints the run's result as JSON.
102pub fn run_with_cli_args(args: impl IntoIterator<Item = OsString>) -> Result<(), String> {
103    let (managed_output, args) = take_managed_output(args.into_iter().collect());
104    let host = casa_task_runtime::TaskCliHost::new(
105        imager_task_schema_bundle(),
106        |request: ImagerTaskRequest| request.execute(),
107    );
108    if let Some(output) = host.dispatch(&args).map_err(|error| error.to_string())? {
109        if managed_output && args.iter().any(|argument| argument == "--json-run") {
110            let ImagerTaskResult::Run(result) = serde_json::from_str(&output)
111                .map_err(|error| format!("decode the imager result: {error}"))?;
112            println!("{}", json(&result)?);
113        } else {
114            println!("{output}");
115        }
116        return Ok(());
117    }
118    if args
119        .iter()
120        .any(|arg| matches!(arg.to_str(), Some("-h" | "--help")))
121    {
122        println!("{}", schema::render_help("casars-imager"));
123        return Ok(());
124    }
125    let (parameters, request) = resolve_cli_request(&args)?;
126    let result = run(parameters, &request)?;
127    if managed_output {
128        println!("{}", json(&result)?);
129    } else {
130        println!(
131            "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, \
132             {} reported minor iterations, {} actual components, stop={:?})",
133            request.imagename.display(),
134            result.run.gridded_samples,
135            result.run.major_cycles,
136            result.run.minor_iterations,
137            result.run.actual_minor_iterations,
138            result.run.clean_stop_reason,
139        );
140    }
141    Ok(())
142}
143
144fn json(result: &ImagerRunTaskResult) -> Result<String, String> {
145    serde_json::to_string_pretty(result).map_err(|error| format!("serialize the result: {error}"))
146}

148/// Split `--managed-output [true|false]` from the other arguments.
149fn take_managed_output(raw: Vec<OsString>) -> (bool, Vec<OsString>) {
150    let mut managed = false;
151    let mut args = Vec::with_capacity(raw.len());
152    let mut raw = raw.into_iter().peekable();
153    while let Some(argument) = raw.next() {
154        if argument != "--managed-output" {
155            args.push(argument);
156            continue;
157        }
158        managed = match raw.peek().and_then(|value| value.to_str()) {
159            Some("true") => {
160                raw.next();
161                true
162            }
163            Some("false") => {
164                raw.next();
165                false
166            }
167            _ => true,
168        };
169    }
170    (managed, args)
171}
