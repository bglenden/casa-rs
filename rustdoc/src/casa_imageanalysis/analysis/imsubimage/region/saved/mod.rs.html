20pub(super) fn load_saved_region(
21    image_path: &Path,
22    region_name: &str,
23    target_coords: &CoordinateSystem,
24    full_shape: &[usize],
25    plane: [usize; 2],
26) -> Result<ParsedRegion, ImageError> {
27    let record = load_saved_region_record(image_path, region_name)?;
28    parse_saved_region(region_name, &record, target_coords, full_shape, plane)
29}

31pub(super) fn load_saved_region_record(path: &Path, name: &str) -> Result<RecordValue, ImageError> {
32    match AnyPagedImage::open(path)? {
33        AnyPagedImage::Float32(img) => img.get_region_record(name),
34        AnyPagedImage::Float64(img) => img.get_region_record(name),
35        AnyPagedImage::Complex32(img) => img.get_region_record(name),
36        AnyPagedImage::Complex64(img) => img.get_region_record(name),
37    }
38}

40pub(super) fn parse_saved_region(
41    region_name: &str,
42    rec: &RecordValue,
43    target_coords: &CoordinateSystem,
44    full_shape: &[usize],
45    plane: [usize; 2],
46) -> Result<ParsedRegion, ImageError> {
47    let class_name = match rec.get("name") {
48        Some(Value::Scalar(ScalarValue::String(name))) => name.as_str(),
49        _ => "",
50    };
51
52    match class_name {
53        "WCPolygon" if relative::mode(rec).is_some() => {
54            relative::decode_polygon(region_name, rec, target_coords, full_shape)
55        }
56        "WCPolygon" => polygon_region(
57            region_name,
58            PolygonRegion::from_record(region_name, rec, target_coords)?,
59            target_coords,
60            full_shape,
61        ),
62        "WCUnion" | "WCIntersection" | "WCDifference" | "WCComplement" | "WCExtension"
63        | "WCConcatenation" => compound::decode(region_name, rec, target_coords, full_shape, plane),
64        "LCSlicer" | "LCBox" => {
65            let (blc, trc, stride) = if class_name == "LCSlicer" {
66                let section = decode_lc_slicer(region_name, rec, target_coords, full_shape)?;
67                let stride = section
68                    .stride
69                    .iter()
70                    .any(|&step| step != 1)
71                    .then_some(section.stride);
72                (section.lower, section.upper, stride)
73            } else {
74                let (lower, upper) = decode_lc_box(region_name, rec, full_shape)?;
75                (lower, upper, None)
76            };
77            let shapes = if full_shape.len() >= 2 {
78                vec![
79                    SpatialShape::Box([blc[plane[0]], blc[plane[1]], trc[plane[0]], trc[plane[1]]])
80                        .into(),
81                ]
82            } else {
83                Vec::new()
84            };
85            let shape: Vec<usize> = blc
86                .iter()
87                .zip(trc.iter())
88                .map(|(lo, hi)| hi - lo + 1)
89                .collect();
90            Ok(ParsedRegion {
91                is_compound: false,
92                is_crtf: false,
93                shapes,
94                saved_mask: None,
95                saved_bounds: Some((blc, shape)),
96                stride,
97            })
98        }
99        "WCBox" => {
100            let (blc, trc) = decode_wc_box(region_name, rec, target_coords, full_shape)?;
101            let shapes = if full_shape.len() >= 2 {
102                vec![
103                    SpatialShape::Box([blc[plane[0]], blc[plane[1]], trc[plane[0]], trc[plane[1]]])
104                        .into(),
105                ]
106            } else {
107                Vec::new()
108            };
109            let shape: Vec<usize> = blc
110                .iter()
111                .zip(trc.iter())
112                .map(|(lo, hi)| hi - lo + 1)
113                .collect();
114            Ok(ParsedRegion {
115                is_compound: false,
116                is_crtf: false,
117                shapes,
118                saved_mask: None,
119                saved_bounds: Some((blc, shape)),
120                stride: None,
121            })
122        }
123        "WCEllipsoid" => {
124            let ellipse = decode_wc_ellipsoid(region_name, rec, target_coords, full_shape, plane)?;
125            let bounds = ellipsoid_bounds(&ellipse, full_shape, plane)?;
126            Ok(ParsedRegion {
127                is_compound: false,
128                is_crtf: false,
129                shapes: vec![ellipse.into()],
130                saved_mask: None,
131                saved_bounds: Some(bounds),
132                stride: None,
133            })
134        }
135        other => Err(ImageError::InvalidMetadata(format!(
136            "unsupported saved region '{region_name}' of class {other:?}"
137        ))),
138    }
139}

141fn polygon_region(
142    region_name: &str,
143    poly: PolygonRegion,
144    target_coords: &CoordinateSystem,
145    full_shape: &[usize],
146) -> Result<ParsedRegion, ImageError> {
147    let plane = poly
148        .lattice_plane(target_coords, full_shape)?
149        .ok_or_else(|| {
150            ImageError::CasaSelection(format!(
151                "saved region '{region_name}' selects no pixels in image"
152            ))
153        })?;
154    let axes = poly.pixel_axes;
155    let bounds = plane.section(full_shape, axes);
156    Ok(ParsedRegion {
157        is_compound: false,
158        is_crtf: false,
159        shapes: Vec::new(),
160        saved_mask: Some(SavedRegionMask::Polygon { axes, plane }),
161        saved_bounds: Some(bounds),
162        stride: None,
163    })
164}

166pub(in crate::analysis::imsubimage) fn lattice_pixel_section(
167    rec: &RecordValue,
168    coords: &CoordinateSystem,
169    shape: &[usize],
170) -> Result<Option<StridedSection>, ImageError> {
171    if matches!(rec.get("name"), Some(Value::Scalar(ScalarValue::String(name))) if name == "LCBox")
172    {
173        let stored_shape = read_numeric_vec(rec.get("shape"))
174            .ok_or_else(|| ImageError::CasaSelection("LCBox region has no lattice shape".into()))?;
175        if !stored_shape
176            .iter()
177            .copied()
178            .eq(shape.iter().map(|&size| size as f64))
179        {
180            return Err(ImageError::CasaSelection(format!(
181                "shape of lattice {shape:?} mismatches lattice shape in region {stored_shape:?}"
182            )));
183        }
184        let (start, end) = decode_lc_box("region", rec, shape)?;
185        let count = start.iter().zip(end).map(|(&lo, hi)| hi - lo + 1).collect();
186        return Ok(Some(StridedSection {
187            start,
188            count,
189            stride: vec![1; shape.len()],
190        }));
191    }
192    lc_slicer_section(rec, coords, shape)
193}

195pub(super) fn lc_slicer_section(
196    rec: &RecordValue,
197    coords: &CoordinateSystem,
198    shape: &[usize],
199) -> Result<Option<StridedSection>, ImageError> {
200    if !matches!(
201        rec.get("name"),
202        Some(Value::Scalar(ScalarValue::String(name))) if name == "LCSlicer"
203    ) {
204        return Ok(None);
205    }
206    let section = decode_lc_slicer("region", rec, coords, shape)?;
207    let count = section
208        .lower
209        .iter()
210        .zip(&section.upper)
211        .zip(&section.stride)
212        .map(|((&first, &last), &step)| (last - first) / step + 1)
213        .collect();
214    Ok(Some(StridedSection {
215        start: section.lower,
216        count,
217        stride: section.stride,
218    }))
219}

235pub(super) fn relative_polygon_section(
236    record: &RecordValue,
237    coords: &CoordinateSystem,
238    shape: &[usize],
239) -> Result<Option<StridedSection>, ImageError> {
240    if relative::mode(record).is_none() {
241        return Ok(None);
242    }

245    Ok(Some(StridedSection {
246        start,
247        count,
248        stride: vec![1; shape.len()],
249    }))

258fn decode_lc_slicer(
259    name: &str,
260    rec: &RecordValue,
261    coords: &CoordinateSystem,
262    full_shape: &[usize],
263) -> Result<SlicerSection, ImageError> {
264    let numeric = |field| {
265        read_numeric_vec(rec.get(field)).ok_or_else(|| {
266            ImageError::InvalidMetadata(format!("saved region '{name}' missing numeric {field}"))
267        })
268    };
269    let one_rel = match rec.get("oneRel") {
270        Some(Value::Scalar(ScalarValue::Bool(value))) => *value,
271        Some(Value::Scalar(ScalarValue::Int32(value))) => *value != 0,
272        None => return Err(unknown_record_field("oneRel")),
273        _ => {
274            return Err(ImageError::InvalidMetadata(
275                "RecordInterface::asBool - invalid data type".into(),
276            ));
277        }
278    };
279    let refpix = coords.reference_pixel();
280    let [lower, upper, step] = [numeric("blc")?, numeric("trc")?, numeric("inc")?];
281    let ndim = lower.len().max(upper.len()).max(step.len());
282    if ndim > full_shape.len() {
283        return Err(ImageError::ShapeMismatch {
284            expected: vec![full_shape.len()],
285            got: vec![ndim],
286        });
287    }
288    let resolve = |values: &[f64], fraction_key, mode_key, low| {
289        let fractions = slicer_flags(rec, fraction_key)?;
290        let modes = numeric(mode_key)?;
291        if fractions.len() != values.len() || modes.len() != values.len() {
292            let fields = if low {
293                "blc, fracBlc, and absRelBlc"
294            } else {
295                "trc, fracTrc, and absRelTrc"
296            };
297            return Err(ImageError::InvalidMetadata(format!(
298                "LCSlicer::LCSLicer - {fields} have unequal lengths"
299            )));
300        }
301        full_shape
302            .iter()
303            .enumerate()
304            .map(|(axis, &len)| {
305                let default = if low { 0 } else { len - 1 } as i64;
306                let Some(&stored) = values.get(axis) else {
307                    return Ok(default);
308                };
309                if is_mimic_source(stored) {
310                    return Ok(default);
311                }
312                let fractional = fractions[axis];
313                let mut pixel = stored as f32;
314                if fractional {
315                    pixel = pixel * len as f32 - if low { 0.0 } else { 1.0 };
316                }
317                pixel += match modes[axis] {
318                    1.0 if one_rel && !fractional => -1.0,
319                    1.0 => 0.0,
320                    2.0 => *refpix.get(axis).ok_or_else(|| {
321                        ImageError::InvalidMetadata(format!(
322                            "saved region '{name}' has no reference pixel on axis {axis}"
323                        ))
324                    })? as f32,
325                    3.0 => len as f32 / 2.0,
326                    _ => {
327                        return Err(ImageError::InvalidMetadata(format!(
328                            "saved region '{name}' has invalid {mode_key} on axis {axis}"
329                        )));
330                    }
331                };
332                if !pixel.is_finite() {
333                    return Err(ImageError::InvalidMetadata(format!(
334                        "saved region '{name}' has non-finite bounds on axis {axis}"
335                    )));
336                }
337                let pixel = (pixel + 0.5) as i64;
338                Ok(if low {
339                    pixel.max(0)
340                } else {
341                    pixel.min(default)
342                })
343            })
344            .collect::<Result<Vec<_>, _>>()
345    };
346    let blc = resolve(&lower, "fracblc", "arblc", true)?;
347    let trc = resolve(&upper, "fractrc", "artrc", false)?;
348    if blc.iter().zip(&trc).any(|(lo, hi)| lo > hi) {
349        return Err(ImageError::CasaSelection(format!(
350            "LCSlicer::toSlicer - blc {blc:?} must be <= trc {trc:?}"
351        )));
352    }
353    let fractional_step = slicer_flags(rec, "fracinc")?;
354    if fractional_step.len() != step.len() {
355        return Err(ImageError::InvalidMetadata(
356            "LCSlicer::LCSLicer - inc and fracInc have unequal lengths".into(),
357        ));
358    }
359    let mut strides = vec![1; full_shape.len()];
360    for (axis, (&stored, fractional)) in step.iter().zip(fractional_step).enumerate() {
361        let stride = if is_mimic_source(stored) {
362            1.0
363        } else if fractional {
364            stored as f32 * full_shape[axis] as f32
365        } else {
366            stored as f32
367        };
368        if !stride.is_finite() || stride < 0.5 {
369            return Err(ImageError::InvalidMetadata(format!(
370                "saved region '{name}' requires a positive stride on axis {axis}"
371            )));
372        }
373        strides[axis] = (stride + 0.5) as usize;
374    }
375    let as_indices = |values: Vec<i64>| values.into_iter().map(|value| value as usize).collect();
376    Ok(SlicerSection {
377        lower: as_indices(blc),
378        upper: as_indices(trc),
379        stride: strides,
380    })
381}

383fn unknown_record_field(field: &str) -> ImageError {
384    ImageError::InvalidMetadata(format!("RecordInterface: field {field} is unknown"))
385}
386
387fn slicer_flags(record: &RecordValue, field: &str) -> Result<Vec<bool>, ImageError> {
388    let flags = match record.get(field) {
389        Some(Value::Array(ArrayValue::Bool(values))) if values.ndim() == 1 => {
390            values.iter().copied().collect()
391        }
392        Some(Value::Array(ArrayValue::Int32(values))) if values.ndim() == 1 => {
393            values.iter().map(|&value| value != 0).collect()
394        }
395        Some(Value::Scalar(ScalarValue::Bool(value))) => vec![*value],
396        Some(Value::Scalar(ScalarValue::Int32(value))) => vec![*value != 0],
397        None => return Err(unknown_record_field(field)),
398        _ => {
399            return Err(ImageError::InvalidMetadata(format!(
400                "TableRecordRep::get_pointer - incorrect data type used for field {field}"
401            )));
402        }
403    };
404    Ok(flags)
405}

407fn is_mimic_source(value: f64) -> bool {
408    // The integer sentinel is imprecise in the record's Float32 representation.
409    value < -2_147_483_646.0 + 10.0
410}

412fn decode_lc_box(
413    name: &str,
414    rec: &RecordValue,
415    full_shape: &[usize],
416) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
417    let one_rel = matches!(
418        rec.get("oneRel"),
419        Some(Value::Scalar(ScalarValue::Bool(true)))
420    );
421    let blc_raw = read_numeric_vec(rec.get("blc")).ok_or_else(|| {
422        ImageError::InvalidMetadata(format!("saved region '{name}' missing numeric blc"))
423    })?;
424    let trc_raw = read_numeric_vec(rec.get("trc")).ok_or_else(|| {
425        ImageError::InvalidMetadata(format!("saved region '{name}' missing numeric trc"))
426    })?;
427    if let Some(inc) = read_numeric_vec(rec.get("inc"))
428        && inc.iter().any(|&v| (v - 1.0).abs() > 1e-6)
429    {
430        return Err(ImageError::InvalidMetadata(format!(
431            "saved region '{name}' has non-unit stride {inc:?}"
432        )));
433    }
434    let ndim = full_shape.len();
435    if blc_raw.len() != ndim || trc_raw.len() != ndim {
436        return Err(ImageError::ShapeMismatch {
437            expected: vec![ndim],
438            got: vec![blc_raw.len()],
439        });
440    }
441    let frac_blc = read_bool_vec(rec.get("fracblc"), ndim);
442    let frac_trc = read_bool_vec(rec.get("fractrc"), ndim);
443    let mut blc = Vec::with_capacity(ndim);
444    let mut trc = Vec::with_capacity(ndim);
445    for axis in 0..ndim {
446        let len = full_shape[axis];
447        let lo = if frac_blc[axis] {
448            (blc_raw[axis] * (len.saturating_sub(1)) as f64).round()
449        } else if one_rel {
450            (blc_raw[axis] - 1.0).round()
451        } else {
452            blc_raw[axis].round()
453        };
454        let hi = if frac_trc[axis] {
455            (trc_raw[axis] * (len.saturating_sub(1)) as f64).round()
456        } else if one_rel {
457            (trc_raw[axis] - 1.0).round()
458        } else {
459            trc_raw[axis].round()
460        };
461        if !lo.is_finite() || !hi.is_finite() || lo < 0.0 || hi < lo || (hi as usize) >= len {
462            return Err(ImageError::CasaSelection(format!(
463                "saved region '{name}' bounds [{lo}, {hi}] out of range for axis {axis} (len {len})"
464            )));
465        }
466        blc.push(lo as usize);
467        trc.push(hi as usize);
468    }
469    Ok((blc, trc))
470}

472fn decode_wc_box(
473    name: &str,
474    rec: &RecordValue,
475    coords: &CoordinateSystem,
476    full_shape: &[usize],
477) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
478    decode_wc_box_axes(name, rec, coords, full_shape, |_| true)
479}

481fn decode_wc_box_axes(
482    name: &str,
483    rec: &RecordValue,
484    coords: &CoordinateSystem,
485    full_shape: &[usize],
486    include: impl Fn(usize) -> bool,
487) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
488    let (axes, origin) = wc_pixel_axes(name, "WCBox", rec, coords)?;
489    let mut modes = read_numeric_vec(rec.get("absrel")).unwrap_or_default();
490    if modes.is_empty() {
491        modes.resize(axes.len(), 1.0);
492    }
493    let dimension_mismatch =
494        || ImageError::InvalidMetadata(format!("WCBox {name:?} dimension mismatch"));
495    if modes.len() != axes.len() {
496        return Err(dimension_mismatch());
497    }
498    let selected_axes = axes
499        .iter()
500        .copied()
501        .filter(|&axis| include(axis))
502        .collect::<Vec<_>>();
503    let modes = modes
504        .into_iter()
505        .zip(&axes)
506        .filter_map(|(mode, &axis)| include(axis).then_some(mode))
507        .collect::<Vec<_>>();
508    let mut bounds = Vec::new();
509    for (key, low) in [("blc", true), ("trc", false)] {
510        let quantities = read_quantum_list(rec.get(key))?;
511        if quantities.len() != axes.len() {
512            return Err(dimension_mismatch());
513        }
514        let quantities = quantities
515            .into_iter()
516            .zip(&axes)
517            .filter_map(|(quantity, &axis)| include(axis).then_some(quantity))
518            .collect::<Vec<_>>();
519        bounds.push(wc_box_corner(
520            coords,
521            full_shape,
522            &selected_axes,
523            &quantities,
524            &modes,
525            origin,
526            low,
527        )?);
528    }
529    let trc = bounds.pop().expect("upper corner");
530    let blc = bounds.pop().expect("lower corner");
531    if blc.iter().zip(&trc).any(|(lower, upper)| lower > upper) {
532        return Err(ImageError::CasaSelection(format!(
533            "WCBox {name:?} has reversed bounds"
534        )));
535    }
536    Ok((blc, trc))
537}

539fn wc_pixel_axes(
540    name: &str,
541    class: &str,
542    rec: &RecordValue,
543    coords: &CoordinateSystem,
544) -> Result<(Vec<usize>, usize), ImageError> {
545    let origin = usize::from(matches!(
546        rec.get("oneRel"),
547        Some(Value::Scalar(ScalarValue::Bool(true)))
548    ));
549    let Some(Value::Record(source_record)) = rec.get("coordinates") else {
550        return Err(ImageError::InvalidMetadata(format!(
551            "{class} has no coordinate system"
552        )));
553    };
554    let saved_coords = CoordinateSystem::from_record(source_record)?;
555    let original_axes = read_numeric_vec(rec.get("pixelAxes"))
556        .ok_or_else(|| ImageError::InvalidMetadata(format!("{class} has no pixelAxes")))?
557        .into_iter()
558        .map(|axis| {
559            (axis as usize).checked_sub(origin).ok_or_else(|| {
560                ImageError::InvalidMetadata(format!(
561                    "{class} {name:?} has invalid pixel axis {axis}"
562                ))
563            })
564        })
565        .collect::<Result<Vec<_>, _>>()?;
566    let axes = map_world_region_axes(name, &saved_coords, coords, original_axes)?;
567    Ok((axes, origin))
568}

570fn decode_wc_ellipsoid(
571    name: &str,
572    rec: &RecordValue,
573    coords: &CoordinateSystem,
574    shape: &[usize],
575    plane: [usize; 2],
576) -> Result<SpatialShape, ImageError> {
577    let invalid =
578        || ImageError::InvalidMetadata(format!("invalid direction-plane WCEllipsoid {name:?}"));
579    let (axes, origin) = wc_pixel_axes(name, "WCEllipsoid", rec, coords)?;
580    let mode = match rec.get("absrel") {
581        Some(Value::Scalar(ScalarValue::Int32(mode @ (1..=3)))) => f64::from(*mode),
582        _ => return Err(invalid()),
583    };
584    let [x, y] = plane;
585    let reversed = mode != 1.0 && axes == [y, x];
586    if axes != plane && !reversed {
587        return Err(invalid());
588    }
589    let center = read_quantum_list(rec.get("center"))?;
590    let [first, second] = center.as_slice() else {
591        return Err(invalid());
592    };
593    let rad = Unit::new("rad").expect("radians are a built-in unit");
594    let in_radians = |quantity: &Quantity| quantity.get_value_in(&rad).map_err(|_| invalid());
595    let [cx, cy] = if mode != 1.0 {
596        // Casacore indexes the world center and radii by lattice axis. It
597        // applies pixel overrides in the record's order before returning the plane.
598        let mapped = if reversed { [y, x] } else { plane };
599        let mut projected = relative::project_points(
600            coords,
601            shape,
602            world_axes_for_plane(coords, plane)?,
603            mapped,
604            [first.unit().name(), second.unit().name()],
605            [[first.value(), second.value()]],
606            &vec![mode; coords.n_world_axes()],
607        )?[0];
608        if reversed {
609            projected.reverse();
610        }
611        projected
612    } else if matches!(first.unit().name(), "pix" | "frac")
613        && matches!(second.unit().name(), "pix" | "frac")
614    {
615        let pixel = |q: &Quantity, axis: usize| match q.unit().name() {
616            "frac" => q.value() * shape[axis] as f64,
617            _ => q.value() - origin as f64,
618        };
619        [pixel(first, x), pixel(second, y)]
620    } else {
621        let mut world = coords.reference_value();
622        let [wx, wy] = world_axes_for_plane(coords, plane)?;
623        world[wx] = in_radians(first)?;
624        world[wy] = in_radians(second)?;
625        let pixel = coords.to_pixel(&world)?;
626        [pixel[x], pixel[y]]
627    };
628    let scales = super::direction_pixel_scales(coords).ok_or_else(invalid)?;
629    let radius = |q: &Quantity, axis: usize| {
630        let size = if q.unit().name() == "pix" {
631            q.value()
632        } else {
633            in_radians(q)? / scales[axis]
634        };
635        if !size.is_finite() || size <= 0.0 {
636            return Err(invalid());
637        }
638        Ok(size)
639    };
640    match rec.get("type") {
641        Some(Value::Scalar(ScalarValue::Int32(1))) => {
642            let Some(Value::Record(value)) = rec.get("radius") else {
643                return Err(invalid());
644            };
645            let r = radius(&casa_images::quantity_from_record(value)?, 0)?;
646            Ok(SpatialShape::Circle {
647                cx,
648                cy,
649                rx: r,
650                ry: r,
651            })
652        }
653        Some(Value::Scalar(ScalarValue::Int32(0 | 2))) => {
654            let values = read_quantum_list(rec.get("radii"))?;
655            let [major, minor] = values.as_slice() else {
656                return Err(invalid());
657            };
658            let (a, b) = (radius(major, 0)?, radius(minor, 1)?);
659            if matches!(rec.get("type"), Some(Value::Scalar(ScalarValue::Int32(0)))) {
660                return Ok(SpatialShape::Circle {
661                    cx,
662                    cy,
663                    rx: a,
664                    ry: b,
665                });
666            }
667            let Some(Value::Record(value)) = rec.get("theta") else {
668                return Err(invalid());
669            };
670            let pa = in_radians(&casa_images::quantity_from_record(value)?)?
671                - std::f64::consts::FRAC_PI_2;
672            Ok(SpatialShape::Ellipse {
673                cx,
674                cy,
675                semi_major: a,
676                semi_minor: b,
677                cos_pa: pa.cos(),
678                sin_pa: pa.sin(),
679            })
680        }
681        _ => Err(invalid()),
682    }
683}

685fn ellipsoid_bounds(
686    ellipse: &SpatialShape,
687    full_shape: &[usize],
688    plane: [usize; 2],
689) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
690    let (center, radii) = match *ellipse {
691        SpatialShape::Circle { cx, cy, rx, ry } => ([cx, cy], [rx, ry]),
692        SpatialShape::Ellipse {
693            cx,
694            cy,
695            semi_major,
696            semi_minor,
697            cos_pa,
698            sin_pa,
699        } => {
700            let radii = if sin_pa.abs() <= 1e-7 {
701                [semi_minor, semi_major]
702            } else if cos_pa.abs() <= 1e-7 {
703                [semi_major, semi_minor]
704            } else {
705                [semi_major.max(semi_minor); 2]
706            };
707            ([cx, cy], radii)
708        }
709        _ => unreachable!("WC ellipsoid decodes as a circle or ellipse"),
710    };
711    // LCEllipsoid sets its geometric box before rasterizing the mask, so the
712    // crop can include outer rows or columns with no selected pixel centers.
713    let mut start = vec![0; full_shape.len()];
714    let mut shape = full_shape.to_vec();
715    for ((axis, center), radius) in plane.into_iter().zip(center).zip(radii) {
716        let (center, radius) = (center as f32, radius as f32);
717        if !center.is_finite() || !radius.is_finite() || radius <= 0.0 {
718            return Err(ImageError::InvalidMetadata(
719                "invalid saved ellipsoid bounds".into(),
720            ));
721        }
722        let epsilon = 10.0f32.powi((2.0 * f64::from(radius)).log10() as i32 - 5);
723        let lo = ((center - radius + 1.0 - epsilon) as i64).max(0);
724        let hi = ((center + radius + epsilon) as i64).min((full_shape[axis] - 1) as i64);
725        if center + radius < 0.0 || center - radius > (full_shape[axis] - 1) as f32 || lo > hi {
726            return Err(ImageError::CasaSelection(
727                "saved ellipsoid does not contain pixels inside the image".into(),
728            ));
729        }
730        start[axis] = lo as usize;
731        shape[axis] = (hi - lo + 1) as usize;
732    }
733    Ok((start, shape))
734}

736fn wc_box_corner(
737    coords: &CoordinateSystem,
738    shape: &[usize],
739    axes: &[usize],
740    quantities: &[Quantity],
741    modes: &[f64],
742    origin: usize,
743    low: bool,
744) -> Result<Vec<usize>, ImageError> {
745    let reference = coords.reference_value();
746    let mut world = reference.clone();
747    let reference_pixels = coords.reference_pixel();
748    let units = coords.world_axis_units();
749    let mut has_world_bounds = false;
750    for (&axis, quantity) in axes.iter().zip(quantities) {
751        if !matches!(quantity.unit().name(), "pix" | "frac" | "default") {
752            has_world_bounds = true;
753            let world_axis = coords.pixel_axis_to_world_axis(axis).ok_or_else(|| {
754                ImageError::InvalidMetadata(format!("WCBox pixel axis {axis} lacks a world axis"))
755            })?;
756            let unit = Unit::new(&units[world_axis])
757                .map_err(|err| ImageError::InvalidMetadata(err.to_string()))?;
758            world[world_axis] = quantity
759                .get_value_in(&unit)
760                .map_err(|err| ImageError::InvalidMetadata(err.to_string()))?;
761        }
762    }
763    if has_world_bounds && modes.iter().any(|&mode| mode != 1.0) {
764        // WCBox passes its stored mode positions directly after mapping quantities.
765        make_region_world_absolute(coords, shape, modes, &reference, &mut world)?;
766    }
767    let converted = coords.to_pixel(&world)?;
768    let mut result = shape
769        .iter()
770        .map(|len| if low { 0 } else { len - 1 })
771        .collect::<Vec<_>>();
772    for (position, ((&axis, quantity), &mode)) in axes.iter().zip(quantities).zip(modes).enumerate()
773    {
774        let unit = quantity.unit().name();
775        if unit == "default" {
776            continue;
777        }
778        let limit = (shape[axis] - 1) as f64;
779        let relative = match mode {
780            1.0 => 0.0,
781            // WCBox looks up the reference pixel by the quantity's position.
782            2.0 => reference_pixels[position],
783            3.0 => shape[axis] as f64 * 0.5,
784            _ => return Err(ImageError::InvalidMetadata("invalid WCBox absrel".into())),
785        };
786        let pixel = match unit {
787            "pix" => quantity.value() + relative - if mode == 1.0 { origin as f64 } else { 0.0 },
788            "frac" => quantity.value() * shape[axis] as f64 + relative - f64::from(!low),
789            _ => converted[axis],
790        };
791        // LCBox rounds both corners, then clips only the lower side of BLC
792        // and the upper side of TRC. A disjoint box must stay disjoint.
793        let pixel = (pixel + 0.5).trunc();
794        if !pixel.is_finite() || (low && pixel > limit) || (!low && pixel < 0.0) {
795            return Err(ImageError::CasaSelection(format!(
796                "saved WCBox does not intersect the image on axis {axis}"
797            )));
798        }
799        result[axis] = pixel.clamp(0.0, limit) as usize;
800    }
801    Ok(result)
802}
803
804fn make_region_world_absolute(
805    coords: &CoordinateSystem,
806    shape: &[usize],
807    modes: &[f64],
808    reference: &[f64],
809    world: &mut [f64],
810) -> Result<(), ImageError> {
811    let offsets = world
812        .iter()
813        .enumerate()
814        .map(|(axis, &value)| match modes.get(axis) {
815            Some(2.0 | 3.0) => value,
816            _ => 0.0,
817        })
818        .collect::<Vec<_>>();
819    // WCRegion evaluates the image center even for reference-relative bounds.
820    let center_pixel = shape
821        .iter()
822        .map(|&length| (length - 1) as f64 * 0.5)
823        .collect::<Vec<_>>();
824    let center = coords.to_world(&center_pixel)?;
825    for (mode, anchor) in [(2.0, reference), (3.0, center.as_slice())] {
826        if !modes.contains(&mode) {
827            continue;
828        }
829        for i in 0..coords.n_coordinates() {
830            let coord = coords.coordinate(i);
831            let axes = coords
832                .coordinate_world_axes(i)
833                .expect("existing coordinate");
834            if !axes
835                .iter()
836                .flatten()
837                .any(|&axis| modes.get(axis) == Some(&mode))
838            {
839                continue;
840            }
841            let direction = if matches!(coord, CoordinateModel::Direction(_)) {
842                let [Some(lon), Some(lat)] = axes.as_slice() else {
843                    return Err(ImageError::InvalidMetadata(
844                        "relative WCBox requires both direction axes in the image".into(),
845                    ));
846                };
847                Some(offset_direction(
848                    [anchor[*lon], anchor[*lat]],
849                    [offsets[*lon], offsets[*lat]],
850                ))
851            } else {
852                None
853            };
854            for (local_axis, axis) in axes.into_iter().enumerate() {
855                if let Some(axis) = axis
856                    && modes.get(axis) == Some(&mode)
857                {
858                    world[axis] = match direction {
859                        Some(values) => values[local_axis],
860                        // makeWorldAbsolute is a no-op for Stokes at the reference.
861                        None if mode == 2.0 && matches!(coord, CoordinateModel::Stokes(_)) => {
862                            offsets[axis]
863                        }
864                        None => anchor[axis] + offsets[axis],
865                    };
866                }
867            }
868        }
869    }
870    Ok(())
871}
872
873fn offset_direction(anchor: [f64; 2], offsets: [f64; 2]) -> [f64; 2] {
874    // DirectionCoordinate::makeWorldAbsolute rotates the offset from the
875    // equator to the anchor latitude. Relative longitude is scaled by cos(lat).
876    let (sin_lat, cos_lat) = offsets[1].sin_cos();
877    let (sin_lon, cos_lon) = (offsets[0] / cos_lat).sin_cos();
878    let (sin_anchor, cos_anchor) = anchor[1].sin_cos();
879    let x = cos_lat * cos_lon;
880    let y = cos_lat * sin_lon;
881    let equatorial = cos_anchor * x - sin_anchor * sin_lat;
882    let polar = sin_anchor * x + cos_anchor * sin_lat;
883    let longitude = anchor[0] + y.atan2(equatorial);
884    [
885        longitude.sin().atan2(longitude.cos()),
886        polar.atan2(equatorial.hypot(y)),
887    ]
888}

890fn read_quantum_list(val: Option<&Value>) -> Result<Vec<Quantity>, ImageError> {
891    let Some(Value::Record(rec)) = val else {
892        return Err(ImageError::InvalidMetadata(
893            "saved region is missing a quantity list".into(),
894        ));
895    };
896    rec.fields()
897        .iter()
898        .map(|field| match &field.value {
899            Value::Record(value) => casa_images::quantity_from_record(value),
900            _ => Err(ImageError::InvalidMetadata(
901                "invalid saved region quantity".into(),
902            )),
903        })
904        .collect()
905}

907fn read_numeric_vec(val: Option<&Value>) -> Option<Vec<f64>> {
908    match val? {
909        Value::Array(ArrayValue::Float64(arr)) => Some(arr.iter().copied().collect()),
910        Value::Array(ArrayValue::Float32(arr)) => Some(arr.iter().map(|&v| f64::from(v)).collect()),
911        Value::Array(ArrayValue::Int32(arr)) => Some(arr.iter().map(|&v| f64::from(v)).collect()),
912        Value::Array(ArrayValue::Int64(arr)) => Some(arr.iter().map(|&v| v as f64).collect()),
913        _ => None,
914    }
915}

917fn read_bool_vec(val: Option<&Value>, ndim: usize) -> Vec<bool> {
918    if let Some(Value::Array(ArrayValue::Bool(arr))) = val
919        && arr.len() == ndim
920    {
921        return arr.iter().copied().collect();
922    }
923    vec![false; ndim]
