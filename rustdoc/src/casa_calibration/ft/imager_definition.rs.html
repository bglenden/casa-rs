22/// CASA's `Imager::makeEmptyImage` fits square direction tiles to the
23/// first axis, with at most four Stokes and 32 spectral planes per tile.
24pub(super) fn model_tile_shape(
25    [nx, ny, npol, nchan]: [usize; 4],
26    volume: i32,
27) -> MsResult<[usize; 4]> {
28    let pol = npol.min(4);
29    let chan = nchan.min(32);
30    let volume = volume.checked_abs().unwrap_or_default() as usize;
31    let mut side = 32;
32    if volume > 0 {
33        let area = volume / pol.max(1) / chan.max(1);
34        let floor = area.isqrt();
35        let preferred = floor + usize::from(floor * floor < area);
36        side = if preferred > nx.min(ny) {
37            nx.min(ny)
38        } else {
39            nx.checked_div(preferred).map_or(0, |tiles| nx / tiles)
40        };
41        side = side.max(10);
42    }
43    let tiles = [side.min(nx), side.min(ny), pol, chan];
44    let bytes = tiles
45        .iter()
46        .try_fold(size_of::<f32>(), |bytes, &axis| bytes.checked_mul(axis));
47    if bytes.is_none_or(|bytes| bytes > i32::MAX as usize) {
48        return Err(invalid(
49            "model image tiles exceed the representable data-manager bucket size",
50        ));
51    }
52    Ok(tiles)
53}

55/// Direction of the pixel at the centre of the model image.
56pub enum ImagerPhaseCentre {
57    /// Use a phase direction from FIELD.
58    Field(usize),
59    /// Use a literal CASA direction measure.
60    Direction(MDirection),
61}
62
63/// Spectral layout of the model image.
64pub enum ImagerSpectralAxis {
65    /// A single plane spanning the observed band.
66    Continuum(Option<FrequencyRef>),
67    /// Resample the selected windows on a configured spectral grid.
68    Cube {
69        /// The number of requested image channels.
70        channels: usize,
71        /// Reference written on the image axis.
72        frame: Option<FrequencyRef>,
73        /// Use the selected window's frequency reference for gridding.
74        native_output_frame: bool,
75        /// Channel spacing and the requested frequencies or velocities.
76        config: CubeAxisConfig,
77    },
78}

80/// Geometry requested by the legacy imager before creating a model image.
81pub struct ImagerDefinitionRequest {
82    /// Direction-axis lengths in pixels.
83    pub shape: [usize; 2],
84    /// Number of facets on each direction axis of the first model.
85    pub facets: usize,
86    /// Signed angular increments for longitude and latitude, in radians.
87    pub cell_rad: [f64; 2],
88    /// Polarization planes in image order.
89    pub stokes: Vec<StokesType>,
90    /// The direction to use at the image centre.
91    pub centre: ImagerPhaseCentre,
92    /// A source whose changing observed direction shifts the Fourier phase centre.
93    pub moving_source: Option<MDirection>,
94    /// The supported image projection.
95    pub projection: ProjectionType,
96    /// Spectral windows for the image grid. Empty uses the selected windows.
97    pub windows: Vec<usize>,
98    /// Explicit rest frequency or the MeasurementSet default.
99    pub rest_frequency_hz: Option<f64>,
100    /// The model image's spectral layout.
101    pub spectral: ImagerSpectralAxis,
102}

104impl ImagerDefinitionRequest {
105    pub(super) fn tangent_point(&self, ms: &MeasurementSet) -> MsResult<Option<MDirection>> {
106        if self.facets <= 1 {
107            return Ok(None);
108        }
109        let direction = match &self.centre {
110            ImagerPhaseCentre::Field(field) => raw_field_phase_direction(ms, *field)?,
111            ImagerPhaseCentre::Direction(direction) => direction.clone(),
112        };
113        Ok(Some(direction))
114    }
115
116    pub(super) fn build_for_selection(
117        &self,
118        ms: &MeasurementSet,
119        selected: &ResolvedMsSelection,
120        location: Option<MPosition>,
121    ) -> MsResult<ImagerImageDefinition> {
122        let row = selected
123            .selected_rows
124            .first()
125            .ok_or_else(|| invalid("an image needs a selected visibility"))?;
126        let representative_field = usize::try_from(row.field_id).map_err(invalid)?;
127        let (field_id, centre) = match &self.centre {
128            ImagerPhaseCentre::Field(field) => (*field, raw_field_phase_direction(ms, *field)?),
129            ImagerPhaseCentre::Direction(direction) => (representative_field, direction.clone()),
130        };
131        let engine = MsCalEngine::new(ms)?;
132        let mut frame =
133            engine.spectral_frame_observatory_direction(row.time_mjd_seconds, centre.clone())?;
134        if let Some(position) = location {
135            frame = frame.with_position(position);
136        }
137        let (lon, lat) = centre.as_angles();
138        let telescope = ms.observation()?.string(0, "TELESCOPE_NAME")?;
139        let projection = if self.projection == ProjectionType::SIN
140            && matches!(telescope.as_str(), "ATCASCP" | "WSRT" | "DRAO")
141            && lat.sin() != 0.
142        {
143            Projection::with_parameters(self.projection, vec![0., 1. / lat.tan()])
144        } else {
145            Projection::new(self.projection)
146        };
147        let direction = DirectionCoordinate::new(
148            centre.refer(),
149            projection,
150            [lon.rem_euclid(std::f64::consts::TAU), lat],
151            self.cell_rad,
152            self.shape.map(|length| (length / 2) as f64),
153        );
154        let mut obs = ObsInfo::new(telescope)
155            .with_observer(ms.observation()?.string(0, "OBSERVER")?)
156            .with_pointing_center(lon, lat);
157        if let Some(epoch) = frame.epoch() {
158            obs = obs.with_date(epoch.clone());
159        }
160        if let Some(position) = frame.position() {
161            obs = obs.with_telescope_position(position.clone());
162        }
163
164        let windows: Vec<_> = if self.windows.is_empty() {
165            selected
166                .selected_rows
167                .iter()
168                .map(|row| row.spectral_window_id as usize)
169                .collect::<BTreeSet<_>>()
170                .into_iter()
171                .collect()
172        } else {
173            self.windows.clone()
174        };
175        let window = *windows
176            .first()
177            .ok_or_else(|| invalid("an image needs a spectral window"))?;
178        let spectral_window = ms.spectral_window()?;
179        let native_frame = spectral_window.frequency_reference(window)?;
180        let rest = match self.rest_frequency_hz {
181            Some(rest) => Some(rest),
182            None => rest_frequency(ms, representative_field, window)?,
183        };
184        let (spectral, channels) = match &self.spectral {
185            ImagerSpectralAxis::Continuum(requested_frame) => {
186                let chosen = if native_frame == FrequencyRef::REST {
187                    requested_frame.unwrap_or(native_frame)
188                } else {
189                    FrequencyRef::LSRK
190                };
191                (continuum_axis(ms, selected, &engine, chosen, rest)?, 1)
192            }
193            ImagerSpectralAxis::Cube {
194                channels,
195                frame,
196                native_output_frame,
197                config,
198            } => {
199                let mut config = config.clone();
200                if *native_output_frame {
201                    config.outframe = native_frame;
202                }
203                let report_frame = if native_frame == FrequencyRef::REST {
204                    native_frame
205                } else {
206                    frame.unwrap_or(native_frame)
207                };
208                let rest = match (rest, &config.start) {
209                    (Some(rest), _) => rest,
210                    (None, Some(CubeAxisValue::FrequencyHz { hz, .. })) => *hz,
211                    (None, _) => spectral_window.ref_frequency(window)?,
212                };
213                // With a single frequency plane CASA records the exact
214                // frequency and spacing supplied in the definition.
215                let single_frequency = match (&config.start, &config.width, channels) {
216                    (
217                        Some(CubeAxisValue::FrequencyHz { hz, .. }),
218                        Some(CubeAxisValue::FrequencyHz { hz: delta, .. }),
219                        1,
220                    ) => Some((*hz, *delta)),
221                    _ => None,
222                };
223                config.rest_frequency_hz = Some(rest);
224                let data = windows
225                    .into_iter()
226                    .map(|id| {
227                        Ok((
228                            spectral_window.chan_freq(id)?,
229                            spectral_window.chan_width(id)?,
230                        ))
231                    })
232                    .collect::<MsResult<Vec<_>>>()?;
233                let bands = data
234                    .iter()
235                    .map(|(f, w)| (f.as_slice(), w.as_slice()))
236                    .collect::<Vec<_>>();
237                let (frequencies, widths) = combine_spectral_windows(&bands)?;
238                let spectral = CubeSpectralSetup::for_casa_cube_axis(
239                    native_frame,
240                    &frequencies,
241                    &widths,
242                    Some(*channels),
243                    &config,
244                    row.time_mjd_seconds,
245                    field_id,
246                    Some(centre),
247                    &engine,
248                )?;
249                let values = &spectral.output_channel_frequencies_hz;
250                let Some(&first) = values.first() else {
251                    return Err(invalid("the model image has no spectral channels"));
252                };
253                let width = values
254                    .get(1)
255                    .map_or(spectral.output_channel_widths_hz[0], |next| next - first);
256                let (first, increment) = single_frequency.unwrap_or((first, width));
257                let axis = if values.len() > 1
258                    && matches!(config.start, Some(CubeAxisValue::VelocityMs { .. }))
259                    && config.veltype != casa_core::measures::doppler::DopplerRef::RADIO
260                {
261                    SpectralCoordinate::from_tabular(
262                        report_frame,
263                        (0..values.len()).map(|i| i as f64).collect(),
264                        values.clone(),
265                        first,
266                        increment,
267                        0.,
268                        rest,
269                    )
270                    .map_err(invalid)?
271                } else {
272                    SpectralCoordinate::new(report_frame, first, increment, 0., rest)
273                };
274                (axis, values.len())
275            }
276        };
277        let spectral = if native_frame == FrequencyRef::REST {
278            spectral
279        } else {
280            spectral.with_conversion(FrequencyRef::LSRK, frame)
281        };
282        let shape = [self.shape[0], self.shape[1], self.stokes.len(), channels];
283        let mut coordinates = CoordinateSystem::new().with_obs_info(obs);
284        coordinates.add_coordinate(direction);
285        coordinates.add_coordinate(StokesCoordinate::new(self.stokes.clone()));
286        coordinates.add_coordinate(spectral);
287        Ok(ImagerImageDefinition {
288            shape,
289            coordinates,
290            field_id,
291        })
292    }
293}

295fn continuum_axis(
296    ms: &MeasurementSet,
297    selected: &ResolvedMsSelection,
298    engine: &MsCalEngine,
299    chosen: FrequencyRef,
300    rest: Option<f64>,
301) -> MsResult<SpectralCoordinate> {
302    let spectral_window = ms.spectral_window()?;
303    let mut epochs = BTreeMap::<_, [f64; 2]>::new();
304    for row in &selected.selected_rows {
305        let time = row.time_mjd_seconds;
306        let [first, last] = epochs
307            .entry((row.spectral_window_id, row.field_id))
308            .or_insert([time, time]);
309        *first = first.min(time);
310        *last = last.max(time);
311    }
312    let mut lower = f64::INFINITY;
313    let mut upper = f64::NEG_INFINITY;
314    for ((spw, field), times) in epochs {
315        let id = spw as usize;
316        let frequencies = spectral_window.chan_freq(id)?;
317        let widths = spectral_window.chan_width(id)?;
318        let selected_channels = match selected.channel_selections.get(&spw) {
319            Some(selection) => selection.indices(frequencies.len())?,
320            None => (0..frequencies.len()).collect(),
321        };
322        let frequency_frame = spectral_window.frequency_reference(id)?;
323        let mut low = f64::INFINITY;
324        let mut high = f64::NEG_INFINITY;
325        for channel in selected_channels {
326            let frequency = frequencies[channel];
327            let half_width = 0.5 * widths[channel].abs();
328            low = low.min(frequency - half_width);
329            high = high.max(frequency + half_width);
330        }
331        for time in times {
332            let data_frame = engine.spectral_frame_observatory(time, field as usize)?;
333            let ratio = convert_frequency_to_frame_with_frame(
334                frequency_frame,
335                chosen,
336                1.,
337                Some(&data_frame),
338            )?;
339            lower = lower.min(low * ratio);
340            upper = upper.max(high * ratio);
341        }
342    }
343    let center = (lower + upper) * 0.5;
344    Ok(SpectralCoordinate::new(
345        chosen,
346        center,
347        upper - lower,
348        0.,
349        rest.unwrap_or(center),
350    ))
351}

353fn rest_frequency(ms: &MeasurementSet, field: usize, window: usize) -> MsResult<Option<f64>> {
354    let Some(table) = ms.subtable(casa_ms::SubtableId::Source) else {
355        return Ok(None);
356    };
357    let source = casa_ms::MsSource::new(table);
358    let id = ms.field()?.source_id(field)?;
359    for row in 0..source.row_count() {
360        let definition_window = source.i32(row, "SPECTRAL_WINDOW_ID")?;
361        if source.i32(row, "SOURCE_ID")? == id
362            && (definition_window == -1 || definition_window == window as i32)
363            && let Some(ArrayValue::Float64(values)) =
364                source.optional_array(row, "REST_FREQUENCY")?
365            && let Some(value) = values.iter().next()
366        {
367            return Ok(Some(*value));
368        }
369    }
370    Ok(None)
371}
