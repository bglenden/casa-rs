15/// The front end of the `impv` binary.
16pub fn impv_front_end()
17-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
18    image_task_front_end("impv", &["impv"], run)
19}

21pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
22    let request = ImpvRequest {
23        imagename: task.input_path("imagename")?,
24        outfile: task.path("outfile")?,
25        mode: task.required_text("mode")?,
26        start: optional(task, "start")?,
27        end: optional(task, "end")?,
28        center: optional(task, "center")?,
29        length: amount(task, "length")?,
30        pa: amount(task, "pa")?,
31        width: amount(task, "width")?.unwrap_or(PvAmount::Whole(1)),
32        unit: task.required_text("unit")?,
33        region: parse_region_parameter(task.value("region"), true)?,
34        chans: task.text("chans")?,
35        stokes: task.text("stokes")?,
36        mask: task.text("mask")?,
37        stretch: task.flag("stretch")?,
38        overwrite: task.flag("overwrite")?,
39    };
40    image_task_success(impv(&request).map(ImageAnalysisTaskResult::Impv))
41}

43fn optional<T: DeserializeOwned>(
44    task: &TaskInvocation<'_>,
45    name: &str,
46) -> Result<Option<T>, TaskFailure> {
47    let Some(value) = task.value(name) else {
48        return Ok(None);
49    };
50    if matches!(value, ParameterValue::String(text) if text.trim().is_empty())
51        || matches!(value, ParameterValue::Array(values) if values.is_empty())
52    {
53        return Ok(None);
54    }
55    serde_json::from_value(value.to_plain_json())
56        .map(Some)
57        .map_err(|error| TaskFailure::validation(format!("invalid {name}: {error}")).at(name))
58}
59
60fn amount(task: &TaskInvocation<'_>, name: &str) -> Result<Option<PvAmount>, TaskFailure> {
61    if let Some(ParameterValue::Table(record)) = task.value(name) {
62        return Ok(Some(PvAmount::Record(table_fields_to_record(record)?)));
63    }
64    if let Some(ParameterValue::String(text)) = task.value(name)
65        && let Ok(pixels) = text.parse::<i64>()
66    {
67        return Ok(Some(PvAmount::Whole(pixels)));
68    }
69    let Some(parsed) = optional::<PvTaskAmount>(task, name)? else {
70        return Ok(None);
71    };
72    Ok(Some(match parsed {
73        PvTaskAmount::Whole(v) => PvAmount::Whole(v),
74        PvTaskAmount::Fractional(v) => PvAmount::Fractional(v),
75        PvTaskAmount::Angular(v) => PvAmount::Angular(v),
76        PvTaskAmount::Quantity(record) => PvAmount::Record(table_fields_to_record(&record)?),
77    }))
78}
