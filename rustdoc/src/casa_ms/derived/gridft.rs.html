2//! CASA `GridFT`'s uvw rotation, for callers outside the imaging read path.
3
4use casa_types::measures::direction::{DirectionRef, MDirection};
5
6use super::engine::CasaUvwMachine;
7use crate::error::MsResult;
8
9/// CASA `GridFT`'s rotation of rows from one phase centre to an image's
10/// tangent point, built once and applied to every row
11/// (`FTMachine::rotateUVW`, with a `UVWMachine` that neither projects nor
12/// treats the array as east–west).
13///
14/// `GridFT` rotates every row, even when the two directions are the same:
15/// `FTMachine::initMaps` resets the last field it saw, so the first row it
16/// meets turns the rotation on. The rotation of two equal directions is the
17/// identity up to rounding.
18///
19/// Cf. casacore `UVWMachine::convertUVW(Double&, Vector<Double>&)`.
20#[derive(Debug, Clone, Copy)]
21pub struct GridFtProjection {
22    machine: CasaUvwMachine,
23}
24
25impl GridFtProjection {
26    /// The rotation from rows phased on `phase_centre_rad` to an image whose
27    /// tangent point is `tangent_point_rad`, both J2000 `[ra, dec]` in
28    /// radians.
29    #[must_use]
30    pub fn between_j2000_directions(
31        phase_centre_rad: [f64; 2],
32        tangent_point_rad: [f64; 2],
33    ) -> Self {
34        let source = MDirection::from_angles(
35            phase_centre_rad[0],
36            phase_centre_rad[1],
37            DirectionRef::J2000,
38        );
39        let target = MDirection::from_angles(
40            tangent_point_rad[0],
41            tangent_point_rad[1],
42            DirectionRef::J2000,
43        );
44        Self {
45            machine: CasaUvwMachine::new(&target, &source),
46        }
47    }
48
49    /// A row's raw `UVW` in metres rotated to the tangent point, and the
50    /// phase shift in metres the gridder applies (`−dphase`, the
51    /// measurement operator's `phase_shift_m`).
52    ///
53    /// CASA's `UVWMachine` works in the opposite sign of u and v to the
54    /// MeasurementSet's; the rotation is applied in its convention and the
55    /// result returned in the MeasurementSet's.
56    #[must_use]
57    pub fn apply(&self, raw_uvw_m: [f64; 3]) -> ([f64; 3], f64) {
58        let casa_input = [-raw_uvw_m[0], -raw_uvw_m[1], raw_uvw_m[2]];
59        let (casa_output, casa_dphase_m) = self.machine.convert_uvw(casa_input);
60        (
61            [-casa_output[0], -casa_output[1], casa_output[2]],
62            -casa_dphase_m,
63        )
64    }
65}
66
67/// Rotate a row's raw `UVW` from its phase centre to an image's tangent
68/// point, as CASA's `GridFT` does before it degrids or grids the row
69/// (`FTMachine::rotateUVW`, with a `UVWMachine` that neither projects nor
70/// treats the array as east–west). Both directions are J2000 `[ra, dec]` in
71/// radians. A caller rotating many rows between the same two directions
72/// builds one [`GridFtProjection`] instead.
73///
74/// Returns the rotated uvw in metres and the phase shift in metres the
75/// gridder applies (`−dphase`, the measurement operator's
76/// `phase_shift_m`).
77///
78/// # Errors
79///
80/// A direction that does not convert.
81pub fn gridft_uvw_between_j2000_directions(
82    raw_uvw_m: [f64; 3],
83    phase_centre_rad: [f64; 2],
84    tangent_point_rad: [f64; 2],
85) -> MsResult<([f64; 3], f64)> {
86    Ok(
87        GridFtProjection::between_j2000_directions(phase_centre_rad, tangent_point_rad)
88            .apply(raw_uvw_m),
89    )
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95
96    /// The reusable projection is the per-row helper the imaging read path
97    /// uses, bit for bit, and the rotation between equal directions keeps
98    /// uvw to rounding with no phase shift.
99    #[test]
100    fn the_projection_is_the_per_row_rotation_bit_for_bit() {
101        let phase_centre = [2.6179938779914944, -0.5235987755982988];
102        let tangent = [2.6180011, -0.5236035];
103        let projection = GridFtProjection::between_j2000_directions(phase_centre, tangent);
104        for uvw in [
105            [120.5, -48.25, 3.5],
106            [-1_500.0, 732.125, -40.0],
107            [0.0, 0.0, 0.0],
108        ] {
109            let engine = super::super::engine::gridft_projection_between_j2000_directions(
110                uvw,
111                phase_centre,
112                tangent,
113                false,
114            )
115            .expect("rotation");
116            let projected = projection.apply(uvw);
117            assert_eq!(projected.0.map(f64::to_bits), engine.0.map(f64::to_bits));
118            assert_eq!(projected.1.to_bits(), engine.1.to_bits());
119            assert_eq!(
120                gridft_uvw_between_j2000_directions(uvw, phase_centre, tangent).expect("helper"),
121                projected
122            );
123        }
124        let same = GridFtProjection::between_j2000_directions(phase_centre, phase_centre);
125        let (uvw, shift) = same.apply([120.5, -48.25, 3.5]);
126        for (rotated, raw) in uvw.iter().zip([120.5, -48.25, 3.5]) {
127            assert!((rotated - raw).abs() < 1e-12, "{rotated} against {raw}");
128        }
129        assert!(shift.abs() < 1e-12, "{shift}");
130        assert_eq!(same.apply([0.0; 3]), ([0.0; 3], 0.0));
131    }
