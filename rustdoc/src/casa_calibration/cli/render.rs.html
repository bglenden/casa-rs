15/// `result` in `format`. JSON is the workflow's own report; the task reply
16/// carries the tagged [`CalibrationTaskResult`].
17pub(super) fn render(
18    format: OutputFormat,
19    result: &CalibrationTaskResult,
20) -> Result<String, String> {
21    match format {
22        OutputFormat::Text => Ok(render_text(result)),
23        OutputFormat::Json => render_json(result),
24    }
25}
26
27fn render_json(result: &CalibrationTaskResult) -> Result<String, String> {
28    let report = match result {
29        CalibrationTaskResult::Apply(report) => serde_json::to_string_pretty(report),
30        CalibrationTaskResult::ExportCorrectedData(report) => serde_json::to_string_pretty(report),
31        CalibrationTaskResult::Summary(report) => serde_json::to_string_pretty(report),
32        CalibrationTaskResult::PlanApply(report) => serde_json::to_string_pretty(report),
33        CalibrationTaskResult::Stats(report) => serde_json::to_string_pretty(report),
34        CalibrationTaskResult::SolveGain(report) => serde_json::to_string_pretty(report),
35        CalibrationTaskResult::SolveBandpass(report) => serde_json::to_string_pretty(report),
36        CalibrationTaskResult::FluxScale(report) => serde_json::to_string_pretty(report),
37        CalibrationTaskResult::Gencal(report) => serde_json::to_string_pretty(report),
38    };
39    report
40        .map(|json| json + "\n")
41        .map_err(|error| format!("serialize the calibration report: {error}"))
42}

44fn render_text(result: &CalibrationTaskResult) -> String {
45    match result {
46        CalibrationTaskResult::Apply(report) => render_apply_report_text(report),
47        CalibrationTaskResult::ExportCorrectedData(report) => {
48            render_export_corrected_data_report_text(report)
49        }
50        CalibrationTaskResult::Summary(report) => render_summary_text(report),
51        CalibrationTaskResult::PlanApply(report) => render_apply_plan_text(report),
52        CalibrationTaskResult::Stats(report) => render_stats_text(report),
53        CalibrationTaskResult::SolveGain(report) => render_gain_solve_report_text(report),
54        CalibrationTaskResult::SolveBandpass(report) => render_bandpass_solve_report_text(report),
55        CalibrationTaskResult::FluxScale(report) => render_fluxscale_report_text(report),
56        CalibrationTaskResult::Gencal(report) => render_gencal_report_text(report),
57    }
58}

60pub(super) fn render_summary_text(summaries: &[CalibrationTableSummary]) -> String {
61    let mut out = String::new();
62    for (index, summary) in summaries.iter().enumerate() {
63        if index > 0 {
64            out.push('\n');
65        }
66        let _ = writeln!(out, "Calibration Table: {}", summary.path.display());
67        let _ = writeln!(
68            out,
69            "  type={} subtype={}",
70            summary.table_type, summary.table_subtype
71        );
72        let _ = writeln!(out, "  rows={}", summary.row_count);
73        let _ = writeln!(out, "  par_type={:?}", summary.keywords.par_type);
74        let _ = writeln!(out, "  vis_cal={:?}", summary.keywords.vis_cal);
75        let _ = writeln!(out, "  parameter_family={:?}", summary.parameter_family);
76        let _ = writeln!(
77            out,
78            "  supported_for_v1_apply={}",
79            summary.supported_for_v1_apply()
80        );
81        let _ = writeln!(out, "  columns={}", summary.columns.join(", "));
82        let _ = writeln!(out, "  field_ids={:?}", summary.field_ids);
83        let _ = writeln!(
84            out,
85            "  spectral_window_ids={:?}",
86            summary.spectral_window_ids
87        );
88        let _ = writeln!(out, "  antenna1_ids={:?}", summary.antenna1_ids);
89        let _ = writeln!(out, "  antenna2_ids={:?}", summary.antenna2_ids);
90        let _ = writeln!(out, "  observation_ids={:?}", summary.observation_ids);
91        if let Some(time) = &summary.time_coverage {
92            let _ = writeln!(
93                out,
94                "  time_coverage=[{}, {}] interval={:?}..{:?}",
95                time.min_time, time.max_time, time.min_interval, time.max_interval
96            );
97        }
98        for subtable in &summary.subtables {
99            let _ = writeln!(
100                out,
101                "  subtable {} exists={} rows={:?} path={}",
102                subtable.name,
103                subtable.exists,
104                subtable.row_count,
105                subtable
106                    .resolved_path
107                    .as_deref()
108                    .unwrap_or_else(|| Path::new("<missing>"))
109                    .display()
110            );
111        }
112        if summary.issues.is_empty() {
113            let _ = writeln!(out, "  issues=none");
114        } else {
115            let _ = writeln!(out, "  issues:");
116            for issue in &summary.issues {
117                let _ = writeln!(
118                    out,
119                    "    - {:?} {}: {}",
120                    issue.severity, issue.code, issue.message
121                );
122            }
123        }
124    }
125    out
126}

128pub(super) fn render_apply_report_text(report: &ApplyExecutionReport) -> String {
129    let mut out = String::new();
130    let _ = writeln!(
131        out,
132        "Calibration Apply Report: {}",
133        report
134            .plan
135            .measurement_set_path
136            .as_deref()
137            .unwrap_or_else(|| Path::new("<in-memory>"))
138            .display()
139    );
140    let _ = writeln!(out, "  apply_mode={:?}", report.plan.apply_mode);
141    let _ = writeln!(out, "  parang={}", report.plan.parang);
142    let _ = writeln!(out, "  selected_rows={}", report.plan.selected_row_count);
143    let _ = writeln!(
144        out,
145        "  created_corrected_data_column={}",
146        report.created_corrected_data_column
147    );
148    let _ = writeln!(
149        out,
150        "  wrote_measurement_set={}",
151        report.wrote_measurement_set
152    );
153    let _ = writeln!(out, "  updated_rows={}", report.updated_row_count);
154    let _ = writeln!(out, "  flagged_rows={}", report.flagged_row_count);
155    let _ = writeln!(out, "  flagged_samples={}", report.flagged_sample_count);
156    if let Some(version) = &report.flag_backup_version {
157        let _ = writeln!(out, "  flag_backup={version}");
158    }
159    let _ = writeln!(out, "  timings:");
160    let _ = writeln!(
161        out,
162        "    total={} planning={} open_ms={}",
163        format_duration_ns(report.timings.total_ns),
164        format_duration_ns(report.timings.planning_ns),
165        format_duration_ns(report.timings.open_measurement_set_ns)
166    );
167    let _ = writeln!(
168        out,
169        "    planning.selection={} planning.selected_rows={} planning.ms_spws={} planning.table_plans={}",
170        format_duration_ns(report.timings.planning_selection_ns),
171        format_duration_ns(report.timings.planning_selected_rows_ns),
172        format_duration_ns(report.timings.planning_measurement_set_spectral_windows_ns),
173        format_duration_ns(report.timings.planning_calibration_table_plans_ns)
174    );
175    let _ = writeln!(
176        out,
177        "    ensure_corrected_data={} correlation_lookup={} calibration_load={}",
178        format_duration_ns(report.timings.ensure_corrected_data_ns),
179        format_duration_ns(report.timings.correlation_lookup_ns),
180        format_duration_ns(report.timings.calibration_load_ns)
181    );
182    let _ = writeln!(
183        out,
184        "    row_compute={} row_writeback={} save={}",
185        format_duration_ns(report.timings.row_compute_ns),
186        format_duration_ns(report.timings.row_writeback_ns),
187        format_duration_ns(report.timings.save_ns)
188    );
189    for table in &report.plan.calibration_tables {
190        let _ = writeln!(out, "  table {}", table.spec.path.display());
191        let _ = writeln!(
192            out,
193            "    applicable_selected_rows={}",
194            table.applicable_selected_row_count
195        );
196        let _ = writeln!(out, "    interp={:?}", table.interp);
197        let _ = writeln!(out, "    calwt={}", table.calwt);
198        if !table.spec.apply_to.is_empty() {
199            let _ = writeln!(
200                out,
201                "    apply_to={}",
202                format_apply_table_selection(&table.spec.apply_to)
203            );
204        }
205        if let Some(gainfield) = &table.resolved_gainfield {
206            let _ = writeln!(
207                out,
208                "    gainfield={} {:?}",
209                gainfield.field_id, gainfield.field_name
210            );
211        }
212        for nearest in &table.resolved_nearest_gainfields {
213            let _ = writeln!(
214                out,
215                "    gainfield[ms_field={}] -> {} {:?} sep={:.6}rad",
216                nearest.measurement_set_field_id,
217                nearest.calibration_field_id,
218                nearest.calibration_field_name,
219                nearest.angular_separation_rad
220            );
221        }
222    }
223    out
224}

226pub(super) fn format_duration_ns(duration_ns: u64) -> String {
227    if duration_ns >= 1_000_000_000 {
228        format!("{:.3}s", duration_ns as f64 / 1_000_000_000.0)
229    } else if duration_ns >= 1_000_000 {
230        format!("{:.3}ms", duration_ns as f64 / 1_000_000.0)
231    } else if duration_ns >= 1_000 {
232        format!("{:.3}us", duration_ns as f64 / 1_000.0)
233    } else {
234        format!("{duration_ns}ns")
235    }
236}

238pub(super) fn render_apply_plan_text(plan: &ApplyPlan) -> String {
239    let mut out = String::new();
240    let _ = writeln!(
241        out,
242        "Apply Plan: {}",
243        plan.measurement_set_path
244            .as_deref()
245            .unwrap_or_else(|| Path::new("<in-memory>"))
246            .display()
247    );
248    let _ = writeln!(out, "  apply_mode={:?}", plan.apply_mode);
249    let _ = writeln!(out, "  parang={}", plan.parang);
250    let _ = writeln!(
251        out,
252        "  requires_corrected_data_column={}",
253        plan.requires_corrected_data_column
254    );
255    let _ = writeln!(out, "  selected_rows={}", plan.selected_row_count);

257    let _ = writeln!(
258        out,

260        plan.selected_data_desc_ids
261    );
262    let _ = writeln!(out, "  selected_data_spws={:?}", plan.selected_data_spw_ids);
263    for table in &plan.calibration_tables {
264        let _ = writeln!(out, "  table {}", table.spec.path.display());
265        let _ = writeln!(out, "    vis_cal={:?}", table.summary.keywords.vis_cal);
266        let _ = writeln!(
267            out,
268            "    applicable_selected_rows={}",
269            table.applicable_selected_row_count
270        );
271        let _ = writeln!(out, "    interp={:?}", table.interp);
272        let _ = writeln!(out, "    calwt={}", table.calwt);
273        if !table.spec.apply_to.is_empty() {
274            let _ = writeln!(
275                out,
276                "    apply_to={}",
277                format_apply_table_selection(&table.spec.apply_to)
278            );
279        }
280        if let Some(gainfield) = &table.resolved_gainfield {
281            let _ = writeln!(
282                out,
283                "    gainfield={} {:?}",
284                gainfield.field_id, gainfield.field_name
285            );
286        }
287        for nearest in &table.resolved_nearest_gainfields {
288            let _ = writeln!(
289                out,
290                "    gainfield[ms_field={}] -> {} {:?} sep={:.6}rad",
291                nearest.measurement_set_field_id,
292                nearest.calibration_field_id,
293                nearest.calibration_field_name,
294                nearest.angular_separation_rad
295            );
296        }
297        for mapping in &table.spw_mapping {
298            let _ = writeln!(
299                out,
300                "    spw {} -> {}",
301                mapping.data_spw_id, mapping.calibration_spw_id
302            );
303        }
304    }

308fn format_apply_table_selection(selection: &crate::ApplyTableSelection) -> String {
309    let mut parts = Vec::new();
310    if !selection.field_ids.is_empty() {
311        parts.push(format!("field={:?}", selection.field_ids));
312    }
313    if !selection.spectral_window_ids.is_empty() {
314        parts.push(format!("spw={:?}", selection.spectral_window_ids));
315    }
316    if !selection.observation_ids.is_empty() {
317        parts.push(format!("obs={:?}", selection.observation_ids));
318    }
319    parts.join(" ")
320}

322pub(super) fn render_stats_text(report: &CalibrationStatsReport) -> String {
323    let mut out = String::new();
324    let _ = writeln!(out, "Calibration Stats: {}", report.path.display());
325    let _ = writeln!(out, "  axis={}", report.axis.display_name());
326    let _ = writeln!(out, "  datacolumn={:?}", report.datacolumn);
327    let _ = writeln!(out, "  rows={}", report.row_count);
328    let _ = writeln!(out, "  global:");
329    render_value_stats_block(&mut out, &report.global, 4);
330
331    for (label, groups) in [
332        ("field_id", &report.by_field_id),
333        ("spectral_window_id", &report.by_spectral_window_id),
334        ("antenna1_id", &report.by_antenna1_id),
335        ("observation_id", &report.by_observation_id),
336    ] {
337        let _ = writeln!(out, "  grouped_by_{label}:");
338        for group in groups {
339            let _ = writeln!(out, "    {}={}", label, group.key);
340            render_value_stats_block(&mut out, &group.stats, 6);
341        }
342    }
343
344    out
345}

398pub(super) fn render_fluxscale_report_text(report: &FluxScaleReport) -> String {
399    let mut out = String::new();
400    let _ = writeln!(out, "FluxScale Report: {}", report.output_table.display());
401    let _ = writeln!(out, "  spw_ids={:?}", report.spw_ids);
402    let _ = writeln!(out, "  frequencies_hz={:?}", report.frequencies_hz);
403    for (field_id, field) in &report.fields {
404        let _ = writeln!(out, "  field {field_id} ({})", field.field_name);
405        for (spw_id, spw) in &field.spw_results {
406            let _ = writeln!(
407                out,
408                "    spw {spw_id}: fluxd={:?} fluxd_err={:?} num_sol={:?}",
409                spw.fluxd, spw.fluxd_err, spw.num_sol
410            );
411        }
412    }
413    out
414}

431fn render_value_stats_block(out: &mut String, stats: &crate::CalibrationValueStats, indent: usize) {
432    let prefix = " ".repeat(indent);
433    let _ = writeln!(
434        out,
435        "{prefix}npts={} flagged_npts={} total_npts={}",
436        stats.npts, stats.flagged_npts, stats.total_npts
437    );
438    let _ = writeln!(
439        out,
440        "{prefix}min={:.6} max={:.6} mean={:.6} median={:.6}",
441        stats.min, stats.max, stats.mean, stats.median
442    );
443    let _ = writeln!(
444        out,
445        "{prefix}sum={:.6} sumsq={:.6} rms={:.6}",
446        stats.sum, stats.sumsq, stats.rms
447    );
448    let _ = writeln!(
449        out,
450        "{prefix}stddev={:.6} var={:.6} medabsdevmed={:.6} quartile={:.6}",
451        stats.stddev, stats.var, stats.medabsdevmed, stats.quartile
452    );
