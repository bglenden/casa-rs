2//! The row-work pass: each region-split domain's gridding work by anchor
3//! row, measured before the first pass so that even it cuts balanced strips
4//! (#736), and the work of the PSF and the weight image beside it, so that
5//! a pass that grids the weight image cuts them balanced too.

14/// Groups a chunk's weight-image count holds before it forgets them
15/// ([`GroupCount`]). A chunk's count only estimates what the weight owner's
16/// table grids. A mosaic's few groups are counted once per chunk however
17/// small this is, and AW's rarely recur once a few rows have passed, so a
18/// count far smaller than the table's suffices and keeps the pass light.
19const COUNTED_GROUPS: usize = 1 << 10;
20use super::{
21    BoundedSource, Cancel, Partition, PassDomain, PassError, SampleContent, WEIGHT_SUM_WORK,
22    WorkerTeam, chunk_ranges, slices, stream_blocks,
23};

25/// What a row-work pass measures ([`run_row_work_pass`]): each domain's
26/// data gridding work by anchor row, `None` for a domain of plane owners,
27/// and each domain's work in the other modes.
28#[derive(Clone, Debug, Default, PartialEq, Eq)]
29pub struct MeasuredRowWork {
30    /// Each domain's gridding work by anchor row.
31    pub rows: Vec<Option<Vec<u64>>>,
32    /// Each domain's work in every mode; zero for a domain of plane owners,
33    /// which the pass skips.
34    pub modes: Vec<ModeWork>,
35}
36
37impl MeasuredRowWork {
38    /// The bytes a run keeps for row work of `rows` histogram rows in all
39    /// over `domains` domains: the histograms and one [`ModeWork`] per
40    /// domain. Both the worker fit's estimate and the kept charge use it.
41    #[must_use]
42    pub const fn kept_bytes(rows: usize, domains: usize) -> u64 {
43        (rows * size_of::<u64>() + domains * size_of::<ModeWork>()) as u64
44    }
45
46    /// The bytes this row work holds when kept ([`Self::kept_bytes`]).
47    #[must_use]
48    pub fn bytes(&self) -> u64 {
49        let rows = self.rows.iter().flatten().map(Vec::len).sum();
50        Self::kept_bytes(rows, self.modes.len())
51    }
52}
53
54/// A region-split domain's work in each mode a pass may grid, summed over
55/// every placement the row work counts: the data mode's kernel taps
56/// (`Router::owner_and_work`) and the PSF's taps, at each sample's own
57/// position (the weight cells for an AW catalog), each with
58/// [`PLACEMENT_WORK`](super::PLACEMENT_WORK); and the weight image's. The
59/// weight owner sums each placement into its table ([`WEIGHT_SUM_WORK`]) and
60/// stamps a weight kernel once per group it grids (#787). The pass counts
61/// those groups as the table would (`GroupCount`), so a mosaic's few groups
62/// cost almost nothing while AW's, about one per two placements, keep
63/// half a stamp each.
64#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
65pub struct ModeWork {
66    /// The data mode's work, the row work's total.
67    pub data: u64,
68    /// The PSF's work.
69    pub psf: u64,
70    /// The weight image's work.
71    pub weight: u64,
72}
73
74impl ModeWork {
75    /// The work by anchor row of a pass of `modes`, from the data mode's
76    /// `row_work`, when the pass grids the weight image: the data and PSF
77    /// modes' work, which falls on each sample's own anchor row, scaled
78    /// from the data mode's; and on the centre row, whose owner grids the
79    /// weight image (`Router::weight_owner`), all of the weight image's.
80    /// `None` without the weight mode or its work, when `row_work` itself
81    /// cuts the pass's strips.
82    #[must_use]
83    pub fn weighed(&self, row_work: &[u64], modes: ModeSet) -> Option<Vec<u64>> {
84        if !modes.weight || self.weight == 0 || self.data == 0 || row_work.is_empty() {
85            return None;
86        }
87        let strips = u128::from(if modes.data { self.data } else { 0 })
88            + u128::from(if modes.psf { self.psf } else { 0 });
89        let mut weighed = row_work
90            .iter()
91            .map(|&work| {
92                u64::try_from(u128::from(work) * strips / u128::from(self.data)).unwrap_or(u64::MAX)
93            })
94            .collect::<Vec<_>>();
95        let centre = weighed.len() / 2;
96        weighed[centre] = weighed[centre].saturating_add(self.weight);
97        Some(weighed)
98    }
99}

101/// Row chunks per worker of a row-work pass.
102const CHUNKS_PER_WORKER: usize = 4;
103
104/// Rows a chunk of a row-work pass places at once. It keeps only a
105/// histogram, so this bounds the work between the workers' meetings, not
106/// memory.
107const CHUNK_ROWS: usize = 1 << 12;
108
109/// Whether some domain is split among several region owners
110/// ([`Partition::splits_rows`]), the only case a row-work pass serves.
111#[must_use]
112pub fn needs_row_work(domains: &[PassDomain<'_>]) -> bool {
113    domains.iter().any(|domain| domain.partition.splits_rows())
114}
115
116/// The grid rows a row-work pass measures for `domain`: every row of one
117/// split among region owners, none otherwise.
118fn measured_rows(domain: &PassDomain<'_>) -> usize {
119    if domain.partition.splits_rows() {
120        domain.operator.geometry().grid_shape()[1]
121    } else {
122        0
123    }
124}
125
126/// Bytes a row-work pass over `domains` on `workers` workers holds:
127/// - each chunk's histogram, allocated once at the widest measured domain's
128///   grid rows;
129/// - the totals it returns;
130/// - each domain's router: an owner per grid row for region owners, a
131///   start per owner for plane owners;
132/// - the containers of chunks, routers and totals;
133/// - when the first pass grids a weight image (`weight`, its kernel sets
134///   having weight taps), each chunk's count of its groups per domain
135///   (`GroupCount`), which is allocated only for weight placements.
136///
137/// The source's blocks are the source's charge ([`super::source_block_bytes`]).
138#[must_use]
139pub fn row_work_pass_bytes(domains: &[PassDomain<'_>], workers: usize, weight: bool) -> u64 {
140    let chunks = workers * CHUNKS_PER_WORKER;
141    let widest = domains.iter().map(measured_rows).max().unwrap_or(0);
142    let totals = domains.iter().map(measured_rows).sum::<usize>();
143    let routing = domains
144        .iter()
145        .map(|domain| match domain.partition {
146            Partition::Regions { .. } => {
147                domain.operator.geometry().grid_shape()[1] * size_of::<u16>()
148            }
149            Partition::Planes { owners } => (owners + 1) * size_of::<u32>(),
150        })
151        .sum::<usize>();
152    let containers = chunks * size_of::<(Range<usize>, Vec<u64>, ModeWork, Vec<GroupCount>)>()
153        + domains.len()
154            * (size_of::<Router>() + size_of::<Option<Vec<u64>>>() + size_of::<ModeWork>());
155    // Each chunk's weight-image group count per domain, allocated whole on
156    // its first weight placement (`GroupCount`).
157    let counts = if weight {
158        (chunks * domains.len()) as u64 * GroupCount::bytes(COUNTED_GROUPS)
159    } else {
160        0
161    } + (chunks * domains.len() * size_of::<GroupCount>()) as u64;
162    // The team's result per chunk, collected on each call
163    // ([`WorkerTeam::for_each_mut`]).
164    let results = chunks * size_of::<Result<(), PassError>>();
165    ((chunks * widest + totals) * size_of::<u64>() + routing + containers + results) as u64 + counts
166}

168/// Each domain's gridding work by anchor row over one traversal that reads
169/// flags and weights only ([`SampleContent::FlagsAndWeights`]): every row
170/// is placed as a pass places it, with `weighting`, and each placement adds
171/// its work, its kernel's taps and
172/// [`PLACEMENT_WORK`](super::PLACEMENT_WORK), to its anchor row
173/// (`Router::owner_and_work`). `None`
174/// for a domain of plane owners. It is what a first pass measures
175/// ([`super::PassSummary::row_work`]) without reading or gridding the
176/// visibilities, so the first pass's strips can be cut balanced
177/// ([`Partition::regions_balanced`]). Each counted placement also adds its
178/// PSF and weight work ([`ModeWork`]), from the kernel set's metadata. It
179/// holds [`row_work_pass_bytes`].
180pub fn run_row_work_pass(
181    domains: &[PassDomain<'_>],
182    weighting: &WeightingGeneration,
183    source: &mut dyn BoundedSource,
184    team: &WorkerTeam,
185    cancel: &Cancel,
186) -> Result<MeasuredRowWork, PassError> {
187    let Some(main) = domains.first() else {
188        return Err(PassError::NoDomains);
189    };
190    let planes = PlaneRange::new(0, main.operator.basis().planes());
191    source
192        .begin(planes, false, SampleContent::FlagsAndWeights)
193        .map_err(PassError::Source)?;
194    let routers = domains
195        .iter()
196        .map(|domain| Router::new(&domain.partition, planes))
197        .collect::<Vec<_>>();
198    let mut totals = domains
199        .iter()
200        .map(|domain| {
201            let rows = measured_rows(domain);
202            (rows > 0).then(|| vec![0_u64; rows])
203        })
204        .collect::<Vec<_>>();
205    let count = team.workers() * CHUNKS_PER_WORKER;
206    // Each chunk's histogram holds the widest domain's rows from the start,
207    // so a later, taller domain never grows it ([`row_work_pass_bytes`]).
208    let widest = domains.iter().map(measured_rows).max().unwrap_or(0);
209    let mut modes = vec![ModeWork::default(); domains.len()];
210    // Each chunk counts, per domain, the weight-image groups its rows start
211    // (`GroupCount`), kept across slices as the weight owner's table is.
212    let mut chunks = (0..count)
213        .map(|_| {
214            (
215                0..0,
216                vec![0_u64; widest],
217                ModeWork::default(),
218                (0..domains.len())
219                    .map(|_| GroupCount::new(COUNTED_GROUPS))
220                    .collect::<Vec<_>>(),
221            )
222        })
223        .collect::<Vec<_>>();
224    stream_blocks(source, cancel, false, |block| {
225        for (index, (domain, router)) in domains.iter().zip(&routers).enumerate() {
226            let Some(total) = &mut totals[index] else {
227                continue;
228            };
229            let cf = domain.operator.cf();
230            for slice in slices(block.len(), count, CHUNK_ROWS) {
231                let mut used = 0;
232                for ((range, _, _, _), rows) in chunks.iter_mut().zip(chunk_ranges(slice, count)) {
233                    *range = rows;
234                    used += 1;
235                }
236                let rows = total.len();
237                team.for_each_mut(&mut chunks[..used], |_, (range, work, other, counts)| {
238                    let work = &mut work[..rows];
239                    work.fill(0);
240                    *other = ModeWork::default();
241                    let groups = &mut counts[index];
242                    // In the block's locality order, as a pass places them:
243                    // a baseline's rows share kernel cells, which a set that
244                    // pages its cells (AW) would otherwise reload row by row.
245                    for position in range.clone() {
246                        domain.resampler.place_each(
247                            domain.operator,
248                            weighting,
249                            &block.row(index, block.ordered_row(position)),
250                            None,
251                            |placement, _, _| {
252                                if !planes.contains(placement.plane) {
253                                    return;
254                                }
255                                let (_, placed) =
256                                    router.owner_and_work(domain.operator, &placement);
257                                if let Some((anchor, taps)) = placed
258                                    && let Some(cell) = work.get_mut(anchor)
259                                {
260                                    *cell += taps;
261                                    other.data += taps;
262                                    other.psf += placement_work(cf.psf_half_support(placement.cf));
263                                    if let Some(support) = cf.weight_half_support(placement.cf) {
264                                        other.weight += WEIGHT_SUM_WORK;
265                                        if groups.starts(&placement) {
266                                            other.weight += placement_work(support);
267                                        }
268                                    }
269                                }
270                            },
271                        )?;
272                    }
273                    Ok::<_, PassError>(())
274                })?;
275                for (_, work, other, _) in &chunks[..used] {
276                    for (total, work) in total.iter_mut().zip(work) {
277                        *total += work;
278                    }
279                    let domain = &mut modes[index];
280                    domain.data += other.data;
281                    domain.psf += other.psf;
282                    domain.weight += other.weight;
283                }
284            }
285        }
286        Ok(())
287    })?;
288    Ok(MeasuredRowWork {
289        rows: totals,
290        modes,
291    })
