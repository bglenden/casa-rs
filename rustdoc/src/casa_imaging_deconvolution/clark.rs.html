13/// Clark's major cycles per step (`setMaxNumberMajorCycles(10)`).
14const MAJOR_CYCLES: usize = 10;
15
16/// Clark's point CLEAN on one plane.
17///
18/// Each Clark cycle builds the active list of supported pixels above
19/// `max(threshold, peak × exterior sidelobe × factor)`, cleans it with
20/// subtraction inside the PSF patch only, and then refreshes the whole
21/// residual exactly ([`LinearRefresh`]). A step ends after ten Clark cycles
22/// or when the peak reaches the threshold; its peak residual is the active
23/// list's last maximum, as CASA reports it.
24#[derive(Default)]
25pub struct Clark {
26    refresh: Cell<Option<LinearRefresh>>,
27}
28
29impl Clark {
30    /// Clark that reuses `refresh` when it serves this plane's PSF.
31    #[must_use]
32    pub fn new(refresh: Option<LinearRefresh>) -> Self {
33        Self {
34            refresh: Cell::new(refresh),
35        }
36    }
37
38    /// The refresh, to reuse in later cycles of the same PSF.
39    #[must_use]
40    pub fn into_refresh(self) -> Option<LinearRefresh> {
41        self.refresh.into_inner()
42    }
43}

45/// An active pixel: its `[x, y]`, kept rather than its storage index because
46/// every component's subtraction needs each pixel's offset from it.
47struct Active {
48    pixel: [u32; 2],
49    value: f64,
50}
51
52/// Active pixels per block of [`ClarkState`]'s block maxima.
53const BLOCK: usize = 64;
54
55/// The largest magnitude of each [`BLOCK`] of `active`, NaN ignored.
56fn block_maxima(active: &[Active], blocks: &mut Vec<f64>) {
57    blocks.clear();
58    blocks.extend(active.chunks(BLOCK).map(largest_magnitude));
59}
60
61/// Recompute the blocks of `blocks` that hold any of `active[range]`.
62fn refresh_blocks(active: &[Active], blocks: &mut [f64], range: std::ops::Range<usize>) {
63    if range.is_empty() {
64        return;
65    }
66    for block in range.start / BLOCK..=(range.end - 1) / BLOCK {
67        let end = ((block + 1) * BLOCK).min(active.len());
68        blocks[block] = largest_magnitude(&active[block * BLOCK..end]);
69    }
70}
71
72fn largest_magnitude(pixels: &[Active]) -> f64 {
73    pixels
74        .iter()
75        .fold(0.0_f64, |largest, pixel| largest.max(pixel.value.abs()))
76}
77
78/// ABSMAXF: the first pixel of largest magnitude in list order, as a scan
79/// that keeps the first pixel unless a later one is strictly larger finds
80/// it (so a NaN first pixel is the peak, and a later NaN never is). It runs
81/// once per component, so it reads the blocks' maxima (`blocks`, kept by
82/// [`block_maxima`] and [`refresh_blocks`]) and then only the first block
83/// that holds the largest.
84fn first_largest<'a>(active: &'a [Active], blocks: &[f64]) -> Option<&'a Active> {
85    let first = active.first()?;
86    if first.value.is_nan() {
87        return Some(first);
88    }
89    let largest = blocks
90        .iter()
91        .fold(0.0_f64, |largest, &block| largest.max(block));
92    let block = blocks
93        .iter()
94        .position(|&block| block == largest)
95        .unwrap_or(0);
96    active[block * BLOCK..]
97        .iter()
98        .take(BLOCK)
99        .find(|pixel| pixel.value.abs() == largest)
100        .or(Some(first))
101}

103/// Working state of one Clark step.
104pub struct ClarkState {
105    shape: PlaneShape,
106    psf_peak: [usize; 2],
107    patch: ClarkPatch,
108    threshold: f64,
109    factor: f64,
110    max_residual: f64,
111    flux_limit: f64,
112    iteration_flux_limit: f64,
113    fac: f64,
114    fmn: f64,
115    cycle_iterations: usize,
116    maximum_cycle_iterations: usize,
117    cycles: usize,
118    step_components: usize,
119    active: Vec<Active>,
120    /// The largest magnitude of each [`BLOCK`] of `active`.
121    blocks: Vec<f64>,
122    refresh: LinearRefresh,
123}
124
125impl Solver for Clark {
126    type State = ClarkState;
127
128    fn initialize(
129        &self,
130        view: &MinorCycleView<'_>,
131        residual: &[Vec<f64>],
132        threshold: f64,
133    ) -> Result<ClarkState, Error> {
134        let shape = view.shape;
135        let peak = view.summary.peak();
136        let refresh = match self.refresh.take() {
137            Some(refresh) if refresh.serves(shape, peak) => refresh,
138            _ => LinearRefresh::new(&view.psf[0], shape, peak)?,
139        };
140        let patch = view.summary.clark();
141        // A PSF whose sidelobes outside the patch are large gets short Clark
142        // cycles (ClarkCleanLatModel::solve).
143        let maximum_cycle_iterations = if patch.exterior_sidelobe > 0.5 {
144            5
145        } else if patch.exterior_sidelobe > 0.35 {
146            50
147        } else {
148            usize::MAX
149        };
150        let max_residual = residual[0]
151            .iter()
152            .zip(view.support.as_slice())
153            .filter(|(_, supported)| **supported)
154            .fold(0.0_f64, |peak, (value, _)| peak.max(value.abs()));
155        let mut state = ClarkState {
156            shape,
157            psf_peak: shape.pixel(peak),
158            patch,
159            threshold,
160            factor: 1.0 / 3.0,
161            max_residual,
162            flux_limit: 0.0,
163            iteration_flux_limit: 0.0,
164            fac: 0.0,
165            fmn: 0.0,
166            cycle_iterations: 0,
167            maximum_cycle_iterations,
168            cycles: 0,
169            step_components: 0,
170            active: Vec::new(),
171            blocks: Vec::new(),
172            refresh,
173        };
174        state.begin(&residual[0], view);
175        Ok(state)
176    }
177
178    fn next(
179        &self,
180        state: &mut ClarkState,
181        view: &MinorCycleView<'_>,
182        residual: &mut [Vec<f64>],
183    ) -> Result<Next, Error> {
184        loop {
185            let peak = first_largest(&state.active, &state.blocks);
186            debug_assert_eq!(
187                peak.map(|pixel| pixel.pixel),
188                {
189                    let mut fresh = Vec::new();
190                    block_maxima(&state.active, &mut fresh);
191                    first_largest(&state.active, &fresh).map(|pixel| pixel.pixel)
192                },
193                "the block maxima are current"
194            );
195            if state.cycle_iterations < state.maximum_cycle_iterations
196                && let Some(pixel) = peak
197                && pixel.value.abs() > state.iteration_flux_limit
198            {
199                if !pixel.value.is_finite() {
200                    return Err(Error::NonFinite);
201                }
202                return Ok(Next::Clean(Candidate::Pixel {
203                    index: state
204                        .shape
205                        .index(pixel.pixel[0] as usize, pixel.pixel[1] as usize),
206                    scale: 0,
207                    strength: pixel.value,
208                }));
209            }
210            if state.cycle_iterations == 0 {
211                return Ok(Next::Stop);
212            }
213            state.refresh_pending(&mut residual[0], view)?;
214            if state.max_residual <= state.threshold || state.cycles >= MAJOR_CYCLES {
215                return Ok(Next::Stop);
216            }
217            state.begin(&residual[0], view);
218        }
219    }
220
221    fn accept(
222        &self,
223        state: &mut ClarkState,
224        view: &MinorCycleView<'_>,
225        _: &mut [Vec<f64>],
226        candidate: Candidate,
227        gain: f64,
228        delta: &mut Delta,
229    ) -> Result<(), Error> {
230        let Candidate::Pixel {
231            index, strength, ..
232        } = candidate;
233        let flux = gain * strength;
234        delta.add(0, index, flux);
235        state.refresh.add(index, flux);
236        let shape = state.shape;
237        let [px, py] = shape.pixel(index);
238        let patch = state.patch;
239        let psf = &view.psf[0];
240        // The list runs x outer, so the pixels whose x lies in the patch
241        // are one run of it; only they can change.
242        debug_assert!(
243            state
244                .active
245                .windows(2)
246                .all(|pair| pair[0].pixel[0] <= pair[1].pixel[0]),
247            "the active list runs x outer (`begin`'s order)"
248        );
249        let first_x = px as isize - patch.radius[0] as isize;
250        let end_x = px as isize + (patch.size[0] - patch.radius[0]) as isize;
251        let start = state
252            .active
253            .partition_point(|pixel| (pixel.pixel[0] as isize) < first_x);
254        let end = state
255            .active
256            .partition_point(|pixel| (pixel.pixel[0] as isize) < end_x);
257        for pixel in &mut state.active[start..end] {
258            let [tx, ty] = pixel.pixel;
259            let relative = [tx as isize - px as isize, ty as isize - py as isize];
260            if relative[0] < -(patch.radius[0] as isize)
261                || relative[0] >= (patch.size[0] - patch.radius[0]) as isize
262                || relative[1] < -(patch.radius[1] as isize)
263                || relative[1] >= (patch.size[1] - patch.radius[1]) as isize
264            {
265                continue;
266            }
267            let source = [
268                state.psf_peak[0] as isize + relative[0],
269                state.psf_peak[1] as isize + relative[1],
270            ];
271            if source[0] < 0
272                || source[1] < 0
273                || source[0] >= shape.nx as isize
274                || source[1] >= shape.ny as isize
275            {
276                continue;
277            }
278            pixel.value -= flux * psf[shape.index(source[0] as usize, source[1] as usize)];
279        }
280        refresh_blocks(&state.active, &mut state.blocks, start..end);
281        state.cycle_iterations += 1;
282        state.step_components += 1;
283        state.fmn += state.fac / state.step_components as f64;
284        state.iteration_flux_limit = if state.flux_limit > 0.0 {
285            (state.flux_limit * state.fmn).max(state.threshold)
286        } else {
287            // Without an exterior sidelobe the flux limit is 0, CASA's
288            // `Fac = absRes / fluxLimit` (speedup −1) is infinite and its
289            // iteration limit `max(0 · ∞, threshold)` is NaN, so the Clark
290            // cycle ends after one component.
291            f64::INFINITY
292        };
293        Ok(())
294    }
295
296    fn finalize(
297        &self,
298        mut state: ClarkState,
299        view: &MinorCycleView<'_>,
300        residual: &mut [Vec<f64>],
301        _: &Delta,
302    ) -> Result<StepEnd, Error> {
303        if state.cycle_iterations > 0 {
304            state.refresh_pending(&mut residual[0], view)?;
305        }
306        let end = StepEnd {
307            peak: state.max_residual,
308            refreshes: state.cycles,
309        };
310        self.refresh.set(Some(state.refresh));
311        Ok(end)
312    }
313
314    /// The refresh ([`LinearRefresh::bytes`]), the active list, which
315    /// grows to at most every pixel (an upper bound, since the list holds
316    /// only the supported pixels above the cycle's flux limit), and its
317    /// block maxima.
318    fn working_bytes(&self, shape: PlaneShape, _terms: usize) -> u64 {
319        let pixels = shape.len().next_power_of_two();
320        let active = pixels * size_of::<Active>();
321        let blocks = pixels.div_ceil(BLOCK).next_power_of_two() * size_of::<f64>();
322        LinearRefresh::bytes(shape) + (active + blocks) as u64
323    }
324
325    /// A point component.
326    fn component_cells(&self, _shape: PlaneShape) -> usize {
327        1
328    }
329}

331impl ClarkState {
332    /// Start a Clark cycle: the flux limit from the exterior sidelobe and
333    /// the active list of supported pixels above it.
334    fn begin(&mut self, residual: &[f64], view: &MinorCycleView<'_>) {
335        self.flux_limit = self.max_residual * self.patch.exterior_sidelobe * self.factor;
336        if self.factor > 1.0 {
337            self.flux_limit = self.flux_limit.min(0.95 * self.max_residual);
338        }
339        let cutoff = self.flux_limit.max(self.threshold);
340        // GETBIMF: supported pixels at or above the cutoff, x outer and y
341        // inner, which is storage order. (CASA lists a tiled residual tile
342        // by tile; a plane in one tile is listed in this order.)
343        self.active.clear();
344        for (index, (&value, &supported)) in
345            residual.iter().zip(view.support.as_slice()).enumerate()
346        {
347            if supported && value.abs() >= cutoff {
348                let [x, y] = self.shape.pixel(index);
349                self.active.push(Active {
350                    pixel: [x as u32, y as u32],
351                    value,
352                });
353            }
354        }
355        block_maxima(&self.active, &mut self.blocks);
356        let peak = self
357            .blocks
358            .iter()
359            .fold(0.0_f64, |best, &block| best.max(block));
360        // `pow(fluxLimit / absRes, speedup)` with CASA's speedup of −1.
361        self.fac = if self.flux_limit > 0.0 {
362            peak / self.flux_limit
363        } else {
364            0.0
365        };
366        self.fmn = 0.0;
367        self.iteration_flux_limit = cutoff;
368        self.cycle_iterations = 0;
369    }
370
371    /// End a Clark cycle: refresh the residual exactly and widen the next
372    /// cycle's flux limit when the peak grew.
373    fn refresh_pending(
374        &mut self,
375        residual: &mut [f64],
376        view: &MinorCycleView<'_>,
377    ) -> Result<(), Error> {
378        let previous = self.max_residual;
379        self.max_residual = self
380            .active
381            .iter()
382            .fold(0.0_f64, |peak, pixel| peak.max(pixel.value.abs()));
383        self.refresh.refresh(residual, &view.psf[0])?;
384        self.cycles += 1;
385        self.cycle_iterations = 0;
386        if self.max_residual > previous {
387            self.factor *= 3.0;
388            self.maximum_cycle_iterations = 10;
389        }
390        Ok(())
391    }
392}

394#[cfg(test)]
395mod tests {
396    use super::{Active, BLOCK, block_maxima, first_largest, refresh_blocks};

398    /// The scan `first_largest` replaces: keep the first pixel unless a
399    /// later one is strictly larger.
400    fn scan(active: &[Active]) -> Option<&Active> {
401        active.iter().fold(None::<&Active>, |best, pixel| {
402            if best.is_none_or(|current| pixel.value.abs() > current.value.abs()) {
403                Some(pixel)
404            } else {
405                best
406            }
407        })
408    }
409
410    /// `first_largest` over block maxima finds the scan's pixel, on lists
411    /// spanning several blocks with ties, zeros, infinities and NaN, and
412    /// still does after a run of the list changes and only that run's blocks
413    /// are refreshed, as `accept` does.
414    #[test]
415    fn the_first_largest_pixel_is_the_scans() {
416        let mut state = 0x2545_f491_4f6c_dd1d_u64;
417        let mut next = move || {
418            state ^= state << 13;
419            state ^= state >> 7;
420            state ^= state << 17;
421            state
422        };
423        // Few distinct magnitudes, so ties are common, and the edge values.
424        let specials = [
425            0.0,
426            -0.0,
427            f64::INFINITY,
428            f64::NEG_INFINITY,
429            f64::NAN,
430            2.0,
431            -2.0,
432            1.5,
433        ];
434        let value = |draw: u64| specials[(draw % specials.len() as u64) as usize];
435        let mut blocks = Vec::new();
436        let mut multi_block_runs = 0;
437        for length in (0..40).chain([BLOCK - 1, BLOCK, BLOCK + 1, 3 * BLOCK + 7]) {
438            for _ in 0..200 {
439                let mut active = (0..length)
440                    .map(|index| Active {
441                        pixel: [index as u32, 0],
442                        value: value(next()),
443                    })
444                    .collect::<Vec<_>>();
445                block_maxima(&active, &mut blocks);
446                let found = first_largest(&active, &blocks).map(|pixel| pixel.pixel);
447                assert_eq!(found, scan(&active).map(|pixel| pixel.pixel));
448                if length == 0 {
449                    continue;
450                }
451                // A run of 1 to 2·BLOCK pixels from anywhere, so runs cross
452                // block edges.
453                let start = (next() % length as u64) as usize;
454                let end = (start + 1 + (next() % (2 * BLOCK) as u64) as usize).min(length);
455                if start / BLOCK != (end - 1) / BLOCK {
456                    multi_block_runs += 1;
457                }
458                for pixel in &mut active[start..end] {
459                    pixel.value = value(next());
460                }
461                refresh_blocks(&active, &mut blocks, start..end);
462                let found = first_largest(&active, &blocks).map(|pixel| pixel.pixel);
463                assert_eq!(found, scan(&active).map(|pixel| pixel.pixel));
464            }
465        }
466        assert!(
467            multi_block_runs > 100,
468            "only {multi_block_runs} refreshed runs crossed a block edge"
469        );
470    }
