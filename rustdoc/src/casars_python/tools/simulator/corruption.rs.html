2//! Thermal noise: `setseed`, `setnoise`, `corrupt` and `reset`, converted as
3//! casatools' `simulator_cmpt.cc` converts them and handed to the session.
4//!
5//! Before `open` each returns `False` without converting its arguments, as
6//! CASA does. The noise is applied when `close` writes the MeasurementSet;
7//! `corrupt` records which scans it covers.
8
9use casa_simulation::{NoiseSpec, TsysParameters};
10use pyo3::prelude::*;
11
12use super::{PySimulator, outcome};
13use crate::convert::{Variant, quantity, value_in};
14
15/// `sm.setseed(seed)`: the seed of noise terms set from now on.
16///
17/// CASA's binding takes a C++ `long` and passes it to an `Int`, which keeps
18/// the low 32 bits; so does casa-rs, so a seed beyond the 32-bit range works
19/// as it does in CASA.
20pub(super) fn setseed(tool: &PySimulator, py: Python<'_>, seed: i64) -> PyResult<bool> {
21    let set = tool.on_session(py, |session, _| {
22        session.simulator.set_seed(seed as i32);
23        Ok(true)
24    })?;
25    Ok(set.unwrap_or(false))
26}

28/// The arguments of `sm.setnoise`: its quantities as given, the others as
29/// the signature typed them.
30pub(super) struct NoiseRequest<'a> {
31    pub(super) mode: &'a str,
32    pub(super) table: &'a str,
33    pub(super) simple_noise: Variant,
34    pub(super) ground_pressure: Variant,
35    pub(super) altitude: Variant,
36    pub(super) water_height: Variant,
37    pub(super) water_vapour: Variant,
38    /// `tatmos`, `tau`, the efficiencies, the temperatures, `senscoeff` and
39    /// `rxtype`.
40    pub(super) tsys: TsysParameters,
41    pub(super) on_the_fly: bool,
42}
43
44/// `sm.setnoise(...)`: add a noise term.
45///
46/// As CASA's binding does, every quantity argument is read (an unreadable
47/// one raises); `simplenoise` is converted to Jy in `simplenoise` mode, and
48/// `pwv` to mm in the Tsys modes, which refuse one above 100 mm; `[]` is 0.
49/// A `simplenoise` whose unit is not a flux density, or a `pwv` whose unit
50/// is not a length, raises, where CASA scales it silently (see
51/// [`crate::convert::quantity_value`]); a `pwv` without units is metres, as
52/// in CASA. The atmosphere and receiver parameters are `tsys-manual`'s; the
53/// ground pressure, humidity, altitude and water scale height are
54/// `tsys-atm`'s, a later phase's. With `OTF=True` CASA writes no table for
55/// `simplenoise`; for `tsys-manual` it writes the `T` term's, which casa-rs
56/// refuses as not yet implemented.
57pub(super) fn setnoise(
58    tool: &PySimulator,
59    py: Python<'_>,
60    request: &NoiseRequest<'_>,
61) -> PyResult<bool> {
62    let result = tool.on_session(py, |session, lines| {
63        let simple_noise = quantity(&request.simple_noise)?;
64        for other in [
65            &request.ground_pressure,
66            &request.altitude,
67            &request.water_height,
68        ] {
69            quantity(other)?;
70        }
71        let water_vapour = quantity(&request.water_vapour)?;
72        let simple_noise_jy = if request.mode == "simplenoise" {
73            value_in(&simple_noise, "Jy")?
74        } else {
75            0.0
76        };
77        let water_vapour_mm = if request.mode.starts_with("tsys") {
78            value_in(&water_vapour, "mm")?
79        } else {
80            0.0
81        };
82        let spec = NoiseSpec {
83            mode: request.mode.to_string(),
84            simple_noise_jy,
85            table: request.table.to_string(),
86            water_vapour_mm,
87            tsys: request.tsys,
88            on_the_fly: request.on_the_fly,
89        };
90        outcome(session.simulator.set_noise(&spec), lines)
91    })?;
92    Ok(result.unwrap_or(false))
93}

95/// `sm.corrupt()`: apply every noise term to the rows observed so far.
96pub(super) fn corrupt(tool: &PySimulator, py: Python<'_>) -> PyResult<bool> {
97    let result = tool.on_session(py, |session, lines| {
98        outcome(session.simulator.corrupt(), lines)
99    })?;
100    Ok(result.unwrap_or(false))
101}
102
103/// `sm.reset()`: remove every noise term; what `corrupt` applied stays.
104pub(super) fn reset(tool: &PySimulator, py: Python<'_>) -> PyResult<bool> {
105    let reset = tool.on_session(py, |session, _| {
106        session.simulator.reset();
107        Ok(true)
108    })?;
109    Ok(reset.unwrap_or(false))
110}
