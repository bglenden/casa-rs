14pub(super) fn create(
15    header: &ImportHeader,
16    column: OptionalMainColumn,
17    fields: &[Field],
18    sources: &[Source],
19) -> MsResult<MeasurementSet> {
20    let builder = SubtableId::ALL_OPTIONAL.iter().fold(
21        MeasurementSetBuilder::new().with_main_column(column),
22        |builder, &id| builder.with_optional_subtable(id),
23    );
24    let mut ms = MeasurementSet::create_memory(builder)?;
25    let mut data_keywords = RecordValue::default();
26    data_keywords.upsert("UNIT", s(&header.data_unit));
27    ms.main_table_mut()
28        .set_column_keywords(column.name(), data_keywords);
29    add_fixed_tables(&mut ms, header)?;
30    add_windows(&mut ms, header)?;
31    add_sources(&mut ms, sources)?;
32    add_fields(&mut ms, fields)?;
33    for (id, columns) in [
34        (
35            SubtableId::Field,
36            &["DELAY_DIR", "PHASE_DIR", "REFERENCE_DIR"][..],
37        ),
38        (SubtableId::Pointing, &["DIRECTION", "TARGET"][..]),
39        (SubtableId::Source, &["DIRECTION"][..]),
40    ] {
41        for column in columns {
42            TableMeasDesc::new_fixed(column, MeasureType::Direction, header.direction_frame)
43                .write(subtable_mut(&mut ms, id)?)?;
44        }
45    }
46    add_optional(
47        subtable_mut(&mut ms, SubtableId::SysCal)?,
48        SubtableId::SysCal,
49        &["TCAL", "TCAL_SPECTRUM", "TSYS", "TSYS_SPECTRUM"],
50    )?;
51    add_optional(
52        subtable_mut(&mut ms, SubtableId::Weather)?,
53        SubtableId::Weather,
54        &[
55            "PRESSURE",
56            "REL_HUMIDITY",
57            "TEMPERATURE",
58            "WIND_DIRECTION",
59            "WIND_SPEED",
60        ],
61    )?;
62    Ok(ms)
63}

65fn add_fixed_tables(ms: &mut MeasurementSet, header: &ImportHeader) -> MsResult<()> {
66    for antenna in &header.antenna {
67        insert(
68            ms,
69            SubtableId::Antenna,
70            &[
71                ("DISH_DIAMETER", f(antenna.diameter)),
72                ("NAME", s(&antenna.name)),
73                ("STATION", s(&antenna.station)),
74                ("TYPE", s("GROUND-BASED")),
75                ("MOUNT", s("ALT-AZ")),
76                ("POSITION", f64_array(&antenna.position, vec![3])),
77            ],
78        )?;
79    }
80    let mut observation = row_from_defs(
81        schema::observation::REQUIRED_COLUMNS,
82        &[
83            ("TIME_RANGE", f64_array(&header.range, vec![2])),
84            ("OBSERVER", s(&header.observer)),
85            ("PROJECT", s(&header.project)),
86            ("TELESCOPE_NAME", s(&header.telescope)),
87        ],
88    );
89    observation = without(observation, &["LOG", "SCHEDULE"]);
90    subtable_mut(ms, SubtableId::Observation)?.add_row(observation)?;
91    insert(
92        ms,
93        SubtableId::Processor,
94        &[("MODE_ID", i(-1)), ("TYPE_ID", i(-1))],
95    )?;
96    Ok(())
97}

99fn add_windows(ms: &mut MeasurementSet, header: &ImportHeader) -> MsResult<()> {
100    let table = subtable_mut(ms, SubtableId::SpectralWindow)?;
101    for (&index, window) in &header.windows {
102        if index < 0 || window.channels == 0 {
103            return Err(MsError::InvalidInput(format!(
104                "Invalid imported spectral window {index}"
105            )));
106        }
107        while table.row_count() < index as usize {
108            table.add_row(row_from_defs(
109                schema::spectral_window::REQUIRED_COLUMNS,
110                &[("NUM_CHAN", i(1))],
111            ))?;
112        }
113        let center = window.reference_frequency - window.reference_channel * window.increment;
114        let mut frequencies = Vec::with_capacity(window.channels);
115        let mut frequency = center;
116        for _ in 0..window.channels {
117            frequencies.push(frequency);
118            frequency += window.increment;
119        }
120        let bandwidth = window.increment.abs();
121        let channels = vec![window.channels];
122        let spacing = f64_array(&vec![window.increment; window.channels], channels.clone());
123        let resolution = f64_array(&vec![bandwidth; window.channels], channels.clone());
124        let row = row_from_defs(
125            schema::spectral_window::REQUIRED_COLUMNS,
126            &[
127                ("MEAS_FREQ_REF", i(window.frame)),
128                ("NUM_CHAN", i(window.channels as i32)),
129                ("REF_FREQUENCY", f(center)),
130                ("CHAN_FREQ", f64_array(&frequencies, channels)),
131                ("CHAN_WIDTH", spacing),
132                ("EFFECTIVE_BW", resolution.clone()),
133                ("RESOLUTION", resolution),
134                ("TOTAL_BANDWIDTH", f(window.channels as f64 * bandwidth)),
135                ("NET_SIDEBAND", i(i32::from(window.increment < 0.0))),
136            ],
137        );
138        table.add_row(row)?;
139    }
140    Ok(())
141}

143fn add_fields(ms: &mut MeasurementSet, fields: &[Field]) -> MsResult<()> {
144    for field in fields {
145        debug_assert_eq!(ms.field()?.row_count(), field.id as usize);
146        let (degree, direction) = direction_cell(field.direction);
147        insert(
148            ms,
149            SubtableId::Field,
150            &[
151                ("TIME", f(field.time)),
152                ("NAME", s(&field.name)),
153                ("SOURCE_ID", i(field.source_id)),
154                ("NUM_POLY", i(degree)),
155                ("DELAY_DIR", direction.clone()),
156                ("PHASE_DIR", direction.clone()),
157                ("REFERENCE_DIR", direction),
158            ],
159        )?;
160    }
161    Ok(())
162}

164fn add_sources(ms: &mut MeasurementSet, sources: &[Source]) -> MsResult<()> {
165    let table = subtable_mut(ms, SubtableId::Source)?;
166    add_optional(
167        table,
168        SubtableId::Source,
169        &["REST_FREQUENCY", "SYSVEL", "TRANSITION"],
170    )?;
171    for source in sources {
172        let mut row = row_from_defs(
173            schema::source::REQUIRED_COLUMNS,
174            &[
175                ("TIME", f(source.time)),
176                ("INTERVAL", f(source.interval)),
177                ("NAME", s(&source.name)),
178                ("SOURCE_ID", i(source.id)),
179                ("SPECTRAL_WINDOW_ID", i(source.window)),
180                ("NUM_LINES", i(source.rest_frequencies.len() as i32)),
181                ("CALIBRATION_GROUP", i(-1)),
182                ("DIRECTION", f64_array(&source.direction, vec![2])),
183                ("PROPER_MOTION", f64_array(&source.proper_motion, vec![2])),
184            ],
185        );
186        if !source.rest_frequencies.is_empty() {
187            let lines = source.rest_frequencies.len();
188            row.upsert(
189                "REST_FREQUENCY",
190                f64_array(&source.rest_frequencies, vec![lines]),
191            );
192            row.upsert(
193                "SYSVEL",
194                f64_array(&vec![source.velocity; lines], vec![lines]),
195            );
196        }
197        if !source.transitions.is_empty() {
198            let names = source
199                .transitions
200                .iter()
201                .map(String::as_str)
202                .collect::<Vec<_>>();
203            row.upsert("TRANSITION", string_array(&names, vec![names.len()]));
204        }
205        table.add_row(row)?;
206    }
207    Ok(())
208}

210fn add_optional(table: &mut Table, id: SubtableId, names: &[&str]) -> MsResult<()> {
211    for &name in names {
212        let definition = schema::optional_columns(id)
213            .iter()
214            .find(|column| column.name == name)
215            .expect("a defined optional MeasurementSet column");
216        let default = match definition.column_kind {
217            ColumnKind::Scalar => Some(Value::Scalar(ScalarValue::Float32(0.))),
218            _ => None,
219        };
220        table.add_column(column_schema(definition), default)?;
221        if let Some(measure) = definition.measure_type {
222            let frame = if name == "REST_FREQUENCY" {
223                "LSRK"
224            } else {
225                definition.measure_ref
226            };
227            TableMeasDesc::new_fixed(name, measure, frame).write(table)?;
228        }
229        if !definition.unit.is_empty() {
230            TableQuantumDesc::with_unit(name, definition.unit).write(table)?;
231        }
232    }
233    Ok(())
234}

236pub(super) fn insert(
237    ms: &mut MeasurementSet,
238    id: SubtableId,
239    cells: &[(&str, Value)],
240) -> MsResult<i32> {
241    let table = subtable_mut(ms, id)?;
242    let row = table.row_count() as i32;
243    table.add_row(row_from_defs(schema::required_columns(id), cells))?;
244    Ok(row)
245}

247pub(super) fn without(row: RecordValue, names: &[&str]) -> RecordValue {
248    RecordValue::new(
249        row.into_fields()
250            .into_iter()
251            .filter(|field| !names.contains(&field.name.as_str()))
252            .collect(),
253    )
254}

256pub(super) fn direction_cell(direction: Direction) -> (i32, Value) {
257    let [lon, lat] = direction.position;
258    let [dlon, dlat] = direction.rate;
259    if direction.rate == [0.0; 2] {
260        (0, f64_array(&[lon, lat], vec![2, 1]))
261    } else {
262        (1, f64_array(&[lon, dlon, lat, dlat], vec![2, 2]))
263    }
264}

266pub(super) fn state_row(ms: &mut MeasurementSet, intent: &str, subscan: i32) -> MsResult<i32> {
267    // The filler matches the entire state string against its on-source label.
268    let signal = intent == "OBSERVE_TARGET#ON_SOURCE";
269    insert(
270        ms,
271        SubtableId::State,
272        &[
273            ("OBS_MODE", s(intent)),
274            ("SUB_SCAN", i(subscan)),
275            ("SIG", b(signal)),
276            ("REF", b(!signal)),
277        ],
278    )
