14impl ReplayPreparation<'_> {
15    pub(super) fn consume_numeric<W, F, E>(
16        &mut self,
17        problem: &CompiledProblem,
18        consumer: &mut SelectedObservationBlockConsumer<'_>,
19        weights: &mut W,
20        storage: &SelectedObservationBlock,
21        execution: crate::bounded_stream::BoundedExecution<'_>,
22        emit: &mut F,
23    ) -> Result<(), WeightingBlockKernelError<E>>
24    where
25        W: StreamingWeightPhase + Sync,
26        F: FnMut(
27                &ReconstructionWeightedBlock,
28                crate::bounded_stream::BoundedExecution<'_>,
29            ) -> Result<(), E>
30            + Send,
31        E: Error + Send + 'static,
32    {
33        let geometry = self.geometry.as_mut().expect("numeric preparation plan");
34        let started = Instant::now();
35        crate::weighting::bulk_source::project_geometry(problem, storage, geometry, execution)
36            .map_err(|error| {
37                WeightingBlockKernelError::Traversal(SelectedObservationTraversalError::Source(
38                    error,
39                ))
40            })?;
41        self.preparation_nanos += started.elapsed().as_nanos();
42        let channels = geometry.channels().len();
43        let runs = geometry.row_count() * channels;
44        let mut producer = NumericProducer {
45            problem,
46            storage,
47            geometry,
48            weights,
49            workers: &mut self.workers,
50            plan: self.plan,
51            active: &self.active,
52            peak_active: &self.peak_active,
53            next_run: 0,
54            next_worker: self.plan.workers,
55            runs,
56            preparation_nanos: 0,
57            commit_nanos: 0,
58        };
59        let mut consumer_nanos = 0;
60        consumer
61            .consume_numeric(storage, geometry, || {
62                let mut current = producer
63                    .next_block(execution)
64                    .map_err(ReplayCallbackError::Owner)?;
65                while let Some(block) = current {
66                    let next = if W::OVERLAP_PREPARATION && execution.is_parallel() {
67                        let (next, consumed) = execution.join_pipeline(
68                            || {
69                                producer
70                                    .next_block(execution)
71                                    .map_err(ReplayCallbackError::Owner)
72                            },
73                            || {
74                                let started = Instant::now();
75                                let result =
76                                    emit(&block, execution).map_err(ReplayCallbackError::Consumer);
77                                consumer_nanos += started.elapsed().as_nanos();
78                                result
79                            },
80                        );
81                        finish_pipeline(next, consumed)?
82                    } else {
83                        let started = Instant::now();
84                        emit(&block, execution).map_err(ReplayCallbackError::Consumer)?;
85                        consumer_nanos += started.elapsed().as_nanos();
86                        producer
87                            .weights
88                            .reuse_emitted_block(block)
89                            .map_err(ReplayCallbackError::Owner)?;
90                        current = producer
91                            .next_block(execution)
92                            .map_err(ReplayCallbackError::Owner)?;
93                        continue;
94                    };
95                    producer
96                        .weights
97                        .reuse_emitted_block(block)
98                        .map_err(ReplayCallbackError::Owner)?;
99                    current = next;
100                }
101                Ok::<(), ReplayCallbackError<E>>(())
102            })
103            .map_err(WeightingBlockKernelError::Traversal)?;
104        self.preparation_nanos += producer.preparation_nanos;
105        // These are nested job durations when pipelined, not disjoint wall time.
106        self.ordered_commit_nanos += producer.commit_nanos + consumer_nanos;
107        self.runs += runs as u64;
108        Ok(())
109    }
110}

112struct NumericProducer<'a, 'p, W> {
113    problem: &'a CompiledProblem,
114    storage: &'a SelectedObservationBlock,
115    geometry: &'a casa_ms::SelectedObservationNumericGeometry,
116    weights: &'a mut W,
117    workers: &'a mut [PreparationWorker<'p>],
118    plan: ReplayPreparationPlan,
119    active: &'a AtomicUsize,
120    peak_active: &'a AtomicUsize,
121    next_run: usize,
122    next_worker: usize,
123    runs: usize,
124    preparation_nanos: u128,
125    commit_nanos: u128,
126}
127
128impl<W: StreamingWeightPhase + Sync> NumericProducer<'_, '_, W> {
129    fn next_block(
130        &mut self,
131        execution: crate::bounded_stream::BoundedExecution<'_>,
132    ) -> Result<Option<ReconstructionWeightedBlock>, WeightingError> {
133        loop {
134            while self.next_worker < self.workers.len() {
135                let prepared = &mut self.workers[self.next_worker].prepared;
136                if prepared.is_empty() {
137                    self.next_worker += 1;
138                    continue;
139                }
140                let started = Instant::now();
141                let block = self.weights.commit_prepared(self.problem, prepared)?;
142                self.commit_nanos += started.elapsed().as_nanos();
143                if block.is_some() {
144                    return Ok(block);
145                }
146            }
147            if self.next_run == self.runs {
148                return Ok(None);
149            }
150            let first = self.next_run;
151            let last = (first + self.plan.runs_per_batch).min(self.runs);
152            let runs_per_worker = (last - first).div_ceil(self.plan.workers);
153            let channels = self.geometry.channels().len();
154            let (problem, storage, geometry, weights) =
155                (self.problem, self.storage, self.geometry, &*self.weights);
156            let (active, peak_active) = (self.active, self.peak_active);
157            let started = Instant::now();
158            execution.for_each_mut(self.workers, |ordinal, worker| {
159                worker.prepared.clear();
160                let lower = (first + ordinal * runs_per_worker).min(last);
161                let upper = (lower + runs_per_worker).min(last);
162                if lower == upper {
163                    return Ok::<(), WeightingError>(());
164                }
165                let live = active.fetch_add(1, Ordering::Relaxed) + 1;
166                peak_active.fetch_max(live, Ordering::Relaxed);
167                let _active = ActivePreparation(active);
168                for row_index in lower / channels..upper.div_ceil(channels) {
169                    let row = storage
170                        .numeric_row(geometry, row_index)
171                        .expect("source owner validated numeric rows");
172                    let frequencies = &geometry.frequencies_hz()
173                        [row_index * channels..(row_index + 1) * channels];
174                    let row_geometry = spectral_geometry(
175                        problem,
176                        row,
177                        frequencies,
178                        geometry.original_pairs_hz()[row_index],
179                    )?;
180                    let begin = lower.saturating_sub(row_index * channels);
181                    let end = (upper - row_index * channels).min(channels);
182                    for (channel, &frequency) in
183                        frequencies.iter().enumerate().take(end).skip(begin)
184                    {
185                        prepare_channel(
186                            problem,
187                            weights,
188                            row,
189                            channel,
190                            frequency,
191                            row_geometry,
192                            &mut worker.prepared,
193                        )?;
194                    }
195                }
196                worker.samples += worker.prepared.len() as u64;
197                Ok(())
198            })?;
199            self.preparation_nanos += started.elapsed().as_nanos();
200            self.next_run = last;
201            self.next_worker = 0;
202        }
203    }
204}

206fn finish_pipeline<T, E: Error>(
207    produced: std::thread::Result<Result<T, ReplayCallbackError<E>>>,
208    consumed: std::thread::Result<Result<(), ReplayCallbackError<E>>>,
209) -> Result<T, ReplayCallbackError<E>> {
210    // Preserve the original typed consumer/I/O error. Both jobs are joined;
211    // report a second failure rather than silently dropping it on early return.
212    let describe_panic = |payload: &Box<dyn std::any::Any + Send>| {
213        payload
214            .downcast_ref::<String>()
215            .map(String::as_str)
216            .or_else(|| payload.downcast_ref::<&str>().copied())
217            .unwrap_or("non-string panic payload")
218            .to_owned()
219    };
220    match consumed {
221        Ok(Err(error)) => {
222            match produced {
223                Ok(Err(sibling)) => eprintln!("weighting producer also failed: {sibling}"),
224                Err(panic) => eprintln!(
225                    "weighting producer also panicked: {}",
226                    describe_panic(&panic)
227                ),
228                Ok(Ok(_)) => {}
229            }
230            Err(error)
231        }
232        Err(panic) => match produced {
233            Ok(Err(error)) => {
234                eprintln!(
235                    "weighting consumer also panicked: {}",
236                    describe_panic(&panic)
237                );
238                Err(error)
239            }
240            Err(sibling) => panic!(
241                "weighting consumer panicked: {}; producer also panicked: {}",
242                describe_panic(&panic),
243                describe_panic(&sibling)
244            ),
245            Ok(Ok(_)) => std::panic::resume_unwind(panic),
246        },
247        Ok(Ok(())) => produced.unwrap_or_else(|panic| std::panic::resume_unwind(panic)),
248    }
249}

251fn correlation(
252    row: SelectedNumericRow<'_>,
253    channel: usize,
254    ordinal: usize,
255) -> SelectedObservationRunCorrelation {
256    let product = row.correlations[ordinal];
257    let index = (row.channels[channel].channel_index - row.first_stored_channel) as usize
258        * row.stored_correlations
259        + product.correlation_index() as usize;
260    SelectedObservationRunCorrelation {
261        correlation_index: product.correlation_index(),
262        correlation_type: product.correlation_type(),
263        visibility: match row.visibility {
264            SelectedNumericVisibility::Complex32(values) => {
265                SelectedVisibilitySample::Complex32([values[index].re, values[index].im])
266            }
267            SelectedNumericVisibility::Float32(values) => {
268                SelectedVisibilitySample::Float32(values[index])
269            }
270        },
271        channel_flag: row.flags[index],
272        parallel_hand_group_flag: false,
273        input_weight: match row.weights {
274            SelectedNumericWeights::PerRow(values) => values[product.correlation_index() as usize],
275            SelectedNumericWeights::PerChannel(values) => values[index],
276        },
277    }
278}

280fn spectral_geometry(
281    problem: &CompiledProblem,
282    row: SelectedNumericRow<'_>,
283    frequencies: &[f64],
284    original_pair: [f64; 2],
285) -> Result<SelectedRowSpectralGeometry, WeightingError> {
286    let first = correlation(row, 0, 0);
287    let sample = SelectedObservationSampleView::from_run(row.row, &row.channels[0], &first);
288    let geometry = SelectedRowSpectralGeometry::new(
289        sample,
290        problem.geometry().spectral().output_frame(),
291        row.channels.len(),
292        (row.channels[0].channel_index, frequencies[0]),
293        row.channels
294            .get(1)
295            .map(|channel| (channel.channel_index, frequencies[1])),
296    )
297    .ok_or(WeightingError::RowSpectralGeometryMismatch)?;
298    if row.channels.len() > 1 {
299        geometry
300            .with_lattice_first_pair_hz(original_pair)
301            .ok_or(WeightingError::RowSpectralGeometryMismatch)
302    } else {
303        Ok(geometry)
304    }
305}

307fn prepare_channel<W: StreamingWeightPhase>(
308    problem: &CompiledProblem,
309    weights: &W,
310    row: SelectedNumericRow<'_>,
311    channel: usize,
312    frequency: f64,
313    geometry: SelectedRowSpectralGeometry,
314    prepared: &mut Vec<ReconstructionWeightedSample>,
315) -> Result<(), WeightingError> {
316    if row.correlations.len() > prepared.capacity() - prepared.len() {
317        return Err(WeightingError::ResidencyOverflow);
318    }
319    let first = correlation(row, channel, 0);
320    let last = correlation(row, channel, row.correlations.len() - 1);
321    let mut group_flag = false;
322    let mut parallel_flag = false;
323    for ordinal in 0..row.correlations.len() {
324        let value = correlation(row, channel, ordinal);
325        group_flag |= value.channel_flag;
326        parallel_flag |= value.channel_flag && value.correlation_type.contributes_to_stokes_i();
327    }
328    let group = SelectedInputWeightGroup::correlation_run(
329        first.input_weight,
330        last.input_weight,
331        row.correlations.len(),
332    )
333    .with_imaging_flag(group_flag);
334    // The existing constant-basis stencil is one unit contribution evaluated
335    // at the source owner's converted channel frequency, regardless of flags.
336    let contributions =
337        SelectedSpectralContributions::new([SelectedSpectralContribution::new(0, 1.0, frequency)])
338            .ok_or(WeightingError::RowSpectralGeometryMismatch)?;
339    let first_prepared = prepared.len();
340    for ordinal in 0..row.correlations.len() {
341        let mut value = correlation(row, channel, ordinal);
342        let weighted = if ordinal == 0 {
343            value.parallel_hand_group_flag = parallel_flag;
344            let sample =
345                SelectedObservationSampleView::from_run(row.row, &row.channels[channel], &value)
346                    .with_input_weight_group(
347                        group.with_terminal_member(row.correlations.len() == 1),
348                    )
349                    .with_row_spectral_geometry(Some(geometry));
350            weights.prepare_sample(problem, sample, frequency, contributions.clone())?
351        } else {
352            prepared[first_prepared]
353                .prepare_group_member(value, ordinal + 1 == row.correlations.len())
354        };
355        prepared.push(weighted);
356    }
357    Ok(())
358}

360#[cfg(test)]
361mod tests {
362    use super::*;
363
364    #[test]
365    fn pipeline_preserves_consumer_io_error_with_a_failed_or_panicked_producer() {
366        for producer in [
367            Ok(Err(ReplayCallbackError::Owner(
368                WeightingError::ReturnedBlockMismatch,
369            ))),
370            Err(Box::new("producer panic") as Box<dyn std::any::Any + Send>),
371        ] {
372            let error = finish_pipeline::<(), std::io::Error>(
373                producer,
374                Ok(Err(ReplayCallbackError::Consumer(std::io::Error::new(
375                    std::io::ErrorKind::PermissionDenied,
376                    "writer failure",
377                )))),
378            )
379            .unwrap_err();
380            let ReplayCallbackError::Consumer(error) = error else {
381                panic!("lost I/O error")
382            };
383            assert_eq!(error.kind(), std::io::ErrorKind::PermissionDenied);
384            assert_eq!(error.to_string(), "writer failure");
385        }
386    }
387
388    #[test]
389    fn pipeline_cannot_succeed_after_a_producer_error_or_consumer_panic() {
390        let error = finish_pipeline::<(), std::io::Error>(
391            Ok(Err(ReplayCallbackError::Owner(
392                WeightingError::ReturnedBlockMismatch,
393            ))),
394            Err(Box::new("consumer panic")),
395        )
396        .unwrap_err();
397        assert!(matches!(
398            error,
399            ReplayCallbackError::Owner(WeightingError::ReturnedBlockMismatch)
400        ));
401        assert!(
402            std::panic::catch_unwind(|| finish_pipeline::<(), std::io::Error>(
403                Ok(Ok(())),
404                Err(Box::new("consumer panic")),
405            ))
406            .is_err()
407        );
408    }
