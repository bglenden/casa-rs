91/// Checked logical payload plan for one bounded selected-content block.
92///
93/// The projection charges every owner-visible heap capacity in the casa-ms selected-content
94/// buffer, the bounded POINTING scalar scan and candidate set, evaluated row geometry, shared
95/// compiler manifests, and retained metadata. Physical row indices, selected visibility
96/// precision, flags, exact input weights, raw UVW, time coordinates, and MAIN provenance are all
97/// included. Boolean vectors are conservatively charged as one byte per value. Platform allocator
98/// bookkeeping outside those Rust allocations remains the allocator's responsibility.
99#[derive(Debug, Clone, Copy, PartialEq, Eq)]
100pub struct SelectedObservationContentPlan {
101    retained_bytes: usize,
102    initialization_scratch_bytes: usize,
103    pointing_reference_scratch_bytes: usize,
104    resident_bytes_per_row: usize,
105    preparation_bytes_per_row: usize,
106    rows_per_block: usize,
107    resident_bytes_per_block: usize,
108    preparation_bytes_per_block: usize,
109    maximum_resident_bytes: usize,
110    maximum_live_blocks: usize,
111    maximum_pointing_polynomial_terms: usize,
112}

190/// Failure to derive an exact bounded selected-content plan.
191#[derive(Debug, Error)]
192pub enum SelectedObservationContentPlanError {
193    /// Storage metadata required by the logical payload projection was unreadable.
194    #[error(transparent)]
195    Storage(#[from] MsError),
196    /// The explicit resource budget is empty or otherwise inconsistent.
197    #[error("selected-observation content budget must have positive bytes and live blocks")]
198    InvalidBudget,
199    /// One selected row cannot fit in the per-block budget.
200    #[error(
201        "one selected-observation row requires {required_bytes} bytes but one content block has {available_bytes} bytes"
202    )]
203    InsufficientBudget {
204        /// Maximum logical bytes required by one selected row.
205        required_bytes: usize,
206        /// Bytes available to one simultaneously live block.
207        available_bytes: usize,
208    },
209    /// Retained metadata and bounded coordinate-construction scratch exceed the budget.
210    #[error(
211        "selected-observation retained metadata requires {required_bytes} bytes but the content budget has {available_bytes} bytes"
212    )]
213    InsufficientRetainedBudget {
214        /// Retained metadata plus its maximum initialization scratch.
215        required_bytes: usize,
216        /// Total source content budget.
217        available_bytes: usize,
218    },
219    /// Checked logical byte arithmetic overflowed.
220    #[error("selected-observation content byte projection overflowed")]
221    ByteOverflow,
222    /// A compiled selected spectral or polarization coordinate was empty or invalid.
223    #[error("compiled selected-observation coordinate shape is invalid")]
224    InvalidCoordinateShape,
225}

227pub(crate) fn selected_content_plan(
228    measurement_set: &MeasurementSet,
229    problem: &CompiledProblem,
230    source: &ObservationSource,
231    shared_bytes: SelectedObservationSharedBytes,
232    budget: SelectedObservationContentBudget,
233) -> Result<SelectedObservationContentPlan, SelectedObservationContentPlanError> {
234    if budget.available_bytes == 0
235        || budget.maximum_live_blocks == 0
236        || budget.maximum_pointing_polynomial_terms == 0
237    {
238        return Err(SelectedObservationContentPlanError::InvalidBudget);
239    }
240    let (retained_bytes, coordinate_construction_scratch_bytes, pointing_reference_scratch_bytes) =
241        retained_metadata_bytes(
242            measurement_set,
243            problem,
244            source,
245            shared_bytes.shared_measures_retained_bytes,
246            shared_bytes.shared_source_slots_retained_bytes,
247        )?;
248    let initialization_scratch_bytes = coordinate_construction_scratch_bytes
249        .checked_add(shared_bytes.shared_binding_graph_initialization_bytes)
250        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
251    let initialization_peak_bytes = retained_bytes
252        .checked_add(initialization_scratch_bytes)
253        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
254    if initialization_peak_bytes > budget.available_bytes {
255        return Err(
256            SelectedObservationContentPlanError::InsufficientRetainedBudget {
257                required_bytes: initialization_peak_bytes,
258                available_bytes: budget.available_bytes,
259            },
260        );
261    }
262    let inspection_bytes = problem
263        .selected_observation()
264        .inspection_scratch_bytes()
265        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
266    let domain_projection_payload_bytes =
267        SelectedImageDomainProjections::retained_heap_bytes_for_len(
268            problem
269                .geometry()
270                .domains()
271                .iter()
272                .try_fold(0_usize, |count, domain| {
273                    count.checked_add(domain.facets().len())
274                })
275                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
276        )
277        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
278    let run_scratch_bytes = maximum_selected_correlations(problem)
279        .checked_mul(
280            size_of::<SelectedObservationRunCorrelation>()
281                .checked_add(size_of::<SelectedSpectralEvaluation>())
282                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
283        )
284        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
285    let row_replay_fixed_bytes = BoundObservationSource::row_replay_fixed_bytes(
286        source.selection().data_descriptions().len(),
287    )
288    .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
289    // VecDeque retains storage for every ready block while Option retains the
290    // active block inline. Heap payloads are charged separately below.
291    let block_container_bytes = budget
292        .maximum_live_blocks
293        .checked_add(1)
294        .and_then(|blocks| blocks.checked_mul(size_of::<BufferedObservationBlock>()))
295        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
296    let traversal_base_bytes = retained_bytes
297        .checked_add(inspection_bytes)
298        // The generation encoder may retain the final row's shared projection
299        // payload while its source block is recycled.
300        .and_then(|bytes| bytes.checked_add(domain_projection_payload_bytes))
301        .and_then(|bytes| bytes.checked_add(run_scratch_bytes))
302        .and_then(|bytes| bytes.checked_add(block_container_bytes))
303        .and_then(|bytes| bytes.checked_add(row_replay_fixed_bytes))
304        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
305    if traversal_base_bytes >= budget.available_bytes {
306        return Err(
307            SelectedObservationContentPlanError::InsufficientRetainedBudget {
308                required_bytes: traversal_base_bytes,
309                available_bytes: budget.available_bytes,
310            },
311        );
312    }
313    let polarization = measurement_set.polarization()?;
314    let mut resident_bytes_per_row = 0_usize;
315    let mut fill_bytes_per_row = 0_usize;
316    let mut preparation_bytes_per_row = 0_usize;
317    let empty_fill = selected_observation_buffer_residency(0, 0, 0, 0)
318        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
319    let fill_fixed_bytes = empty_fill.fill_peak_bytes;
320    let pointing_direction_column = match problem.geometry().centres().pointing() {
321        PointingCentreLaw::Observation(law) => Some(match law.direction_column() {
322            casa_imaging_model::PointingDirectionColumn::Direction => {
323                crate::PointingDirectionColumn::Direction
324            }
325            casa_imaging_model::PointingDirectionColumn::Target => {
326                crate::PointingDirectionColumn::Target
327            }
328        }),
329        PointingCentreLaw::PhaseTrackingCentre | PointingCentreLaw::Fixed(_) => None,
330    };
331    for description in source.selection().data_descriptions() {
332        let channels = source
333            .selection()
334            .spectral_windows()
335            .iter()
336            .find(|selection| selection.spectral_window_id() == description.spectral_window_id())
337            .expect("compiled DATA_DESCRIPTION has one spectral-window selection")
338            .channel_indices();
339        let (Some(first_channel), Some(last_channel)) = (channels.first(), channels.last()) else {
340            return Err(SelectedObservationContentPlanError::InvalidCoordinateShape);
341        };
342        let covering_channels = usize::try_from(
343            last_channel
344                .checked_sub(*first_channel)
345                .and_then(|span| span.checked_add(1))
346                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
347        )
348        .map_err(|_| SelectedObservationContentPlanError::ByteOverflow)?;
349        let polarization_row = usize::try_from(description.polarization_id())
350            .map_err(|_| SelectedObservationContentPlanError::InvalidCoordinateShape)?;
351        let correlations =
352            selected_i32_array_len(polarization.table(), "CORR_TYPE", polarization_row)?
353                .filter(|count| *count > 0)
354                .ok_or(SelectedObservationContentPlanError::InvalidCoordinateShape)?;
355        let sample_count = covering_channels
356            .checked_mul(correlations)
357            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
358        let visibility_bytes = match source.generations().columns().visibility() {
359            VisibilityColumn::Data | VisibilityColumn::CorrectedData => 8,
360            VisibilityColumn::FloatData => 4,
361        };
362        let weight_values = match source.generations().columns().weights() {
363            WeightColumn::Weight => correlations,
364            WeightColumn::WeightSpectrum => sample_count,
365        };
366        let buffer =
367            selected_observation_buffer_residency(1, sample_count, weight_values, visibility_bytes)
368                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
369        let resident = buffer
370            .resident_bytes
371            .checked_add(size_of::<EvaluatedRowGeometry>())
372            .and_then(|bytes| bytes.checked_add(domain_projection_payload_bytes))
373            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
374        // A recycled block keeps its row-geometry allocation until the new
375        // storage buffer and POINTING output are ready. Charge that allocation
376        // during both preparation phases, not only after the block is complete.
377        let retained_geometry = size_of::<EvaluatedRowGeometry>()
378            .checked_add(domain_projection_payload_bytes)
379            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
380        let fill = buffer
381            .fill_peak_bytes
382            .checked_sub(fill_fixed_bytes)
383            .and_then(|bytes| bytes.checked_add(retained_geometry))
384            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
385        let geometry_build = buffer
386            .resident_bytes
387            .checked_add(size_of::<EvaluatedRowGeometry>())
388            // Vec-to-Arc construction can hold source and destination payloads
389            // simultaneously for the row currently being arranged.
390            .and_then(|bytes| {
391                domain_projection_payload_bytes
392                    .checked_mul(2)
393                    .and_then(|payload| bytes.checked_add(payload))
394            })
395            .and_then(|bytes| {
396                let pointing_output = if pointing_direction_column.is_some() {
397                    size_of::<SelectedPointingDirections>()
398                } else {
399                    0
400                };
401                pointing_output.checked_add(bytes)
402            })
403            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
404        let pointing = if let Some(direction_column) = pointing_direction_column {
405            let pointing_scratch = selected_pointing_preparation_peak_bytes(
406                1,
407                1,
408                budget.maximum_pointing_polynomial_terms,
409                direction_column,
410            )
411            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
412            buffer
413                .resident_bytes
414                .checked_add(retained_geometry)
415                .and_then(|bytes| bytes.checked_add(pointing_scratch))
416                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?
417        } else {
418            0
419        };
420        resident_bytes_per_row = resident_bytes_per_row.max(resident);
421        fill_bytes_per_row = fill_bytes_per_row.max(fill);
422        preparation_bytes_per_row = preparation_bytes_per_row
423            .max(fill)
424            .max(pointing)
425            .max(geometry_build);
426    }
427    if resident_bytes_per_row == 0 || preparation_bytes_per_row == 0 {
428        return Err(SelectedObservationContentPlanError::InvalidCoordinateShape);
429    }
430    let selected_rows = usize::try_from(source.selection().rows().selected_row_count())
431        .map_err(|_| SelectedObservationContentPlanError::ByteOverflow)?;
432    let prior_live_blocks = budget.maximum_live_blocks - 1;
433    let prior_resident_bytes_per_row = resident_bytes_per_row
434        .checked_mul(prior_live_blocks)
435        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
436    let traversal_bytes = budget.available_bytes - traversal_base_bytes;
437    let fill_denominator = prior_resident_bytes_per_row
438        .checked_add(fill_bytes_per_row)
439        .and_then(|bytes| bytes.checked_add(BoundObservationSource::row_replay_bytes_per_row()))
440        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
441    let preparation_denominator = prior_resident_bytes_per_row
442        .checked_add(preparation_bytes_per_row)
443        .and_then(|bytes| bytes.checked_add(BoundObservationSource::row_replay_bytes_per_row()))
444        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
445    let rows_by_fill = traversal_bytes
446        .checked_sub(fill_fixed_bytes)
447        .map_or(0, |bytes| bytes / fill_denominator);
448    let rows_by_preparation = traversal_bytes
449        .checked_sub(pointing_reference_scratch_bytes)
450        .map_or(0, |bytes| bytes / preparation_denominator);
451    let rows_per_block = rows_by_fill.min(rows_by_preparation).min(selected_rows);
452    if rows_per_block == 0 {
453        return Err(SelectedObservationContentPlanError::InsufficientBudget {
454            required_bytes: fill_fixed_bytes
455                .checked_add(fill_denominator)
456                .and_then(|fill| {
457                    pointing_reference_scratch_bytes
458                        .checked_add(preparation_denominator)
459                        .map(|preparation| fill.max(preparation))
460                })
461                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
462            available_bytes: traversal_bytes,
463        });
464    }
465    let resident_bytes_per_block = rows_per_block
466        .checked_mul(resident_bytes_per_row)
467        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
468    let preparation_bytes_per_block = rows_per_block
469        .checked_mul(preparation_bytes_per_row)
470        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
471    let prior_blocks = resident_bytes_per_block
472        .checked_mul(prior_live_blocks)
473        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
474    let fill_payload = rows_per_block
475        .checked_mul(fill_bytes_per_row)
476        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
477    let fill_peak = traversal_base_bytes
478        .checked_add(prior_blocks)
479        .and_then(|bytes| bytes.checked_add(fill_fixed_bytes))
480        .and_then(|bytes| bytes.checked_add(fill_payload))
481        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
482    let preparation_peak = traversal_base_bytes
483        .checked_add(prior_blocks)
484        .and_then(|bytes| bytes.checked_add(preparation_bytes_per_block))
485        .and_then(|bytes| bytes.checked_add(pointing_reference_scratch_bytes))
486        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
487    let traversal_peak_bytes = fill_peak.max(preparation_peak);
488    Ok(SelectedObservationContentPlan {
489        retained_bytes,
490        initialization_scratch_bytes,
491        pointing_reference_scratch_bytes,
492        resident_bytes_per_row,
493        preparation_bytes_per_row,
494        rows_per_block,
495        resident_bytes_per_block,
496        preparation_bytes_per_block,
497        maximum_resident_bytes: initialization_peak_bytes.max(traversal_peak_bytes),
498        maximum_live_blocks: budget.maximum_live_blocks,
499        maximum_pointing_polynomial_terms: budget.maximum_pointing_polynomial_terms,
500    })
501}

503fn retained_metadata_bytes(
504    measurement_set: &MeasurementSet,
505    problem: &CompiledProblem,
506    source: &ObservationSource,
507    shared_measures_retained_bytes: usize,
508    shared_source_slots_retained_bytes: usize,
509) -> Result<(usize, usize, usize), SelectedObservationContentPlanError> {
510    let storage_bytes = measurement_set
511        .retained_read_metadata_heap_bytes()
512        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
513    let geometry_bytes = MsCalEngine::selected_observation_retained_heap_bytes(measurement_set)?;
514    let manifest_bytes = source
515        .selection()
516        .retained_manifest_bytes()
517        .and_then(|bytes| {
518            source
519                .generations()
520                .retained_manifest_bytes()
521                .and_then(|generations| bytes.checked_add(generations))
522        })
523        .and_then(|bytes| bytes.checked_add(source.provenance().retained_locator_bytes()))
524        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
525    let predicate_bytes = CompiledRowPredicate::shared_retained_heap_bytes(source)
526        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
527    let spectral_windows = measurement_set.spectral_window()?;
528    let polarizations = measurement_set.polarization()?;
529    let mut coordinate_bytes = source
530        .selection()
531        .data_descriptions()
532        .len()
533        .checked_mul(size_of::<SelectedCoordinates>())
534        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
535    let pointing_reference_scratch_bytes =
536        selected_pointing_reference_scratch_bytes(measurement_set, problem)?;
537    let mut maximum_scratch_bytes = selected_geometry_construction_scratch_bytes(measurement_set)?
538        .max(pointing_reference_scratch_bytes);
539    for description in source.selection().data_descriptions() {
540        let spectral_window = source
541            .selection()
542            .spectral_windows()
543            .iter()
544            .find(|selection| selection.spectral_window_id() == description.spectral_window_id())
545            .expect("compiled DATA_DESCRIPTION has one spectral-window selection");
546        let polarization = source
547            .selection()
548            .correlations()
549            .iter()
550            .find(|selection| selection.polarization_id() == description.polarization_id())
551            .expect("compiled DATA_DESCRIPTION has one polarization selection");
552        coordinate_bytes = coordinate_bytes
553            .checked_add(
554                spectral_window
555                    .channel_indices()
556                    .len()
557                    .checked_mul(size_of::<SelectedChannel>())
558                    .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
559            )
560            .and_then(|bytes| {
561                polarization
562                    .products()
563                    .len()
564                    .checked_mul(size_of::<CorrelationProduct>())
565                    .and_then(|products| bytes.checked_add(products))
566            })
567            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
568
569        let spectral_window_row = usize::try_from(description.spectral_window_id())
570            .map_err(|_| SelectedObservationContentPlanError::InvalidCoordinateShape)?;
571        let frequency_count =
572            selected_f64_array_len(spectral_windows.table(), "CHAN_FREQ", spectral_window_row)?
573                .filter(|count| *count > 0)
574                .ok_or(SelectedObservationContentPlanError::InvalidCoordinateShape)?;
575        let width_count =
576            selected_f64_array_len(spectral_windows.table(), "CHAN_WIDTH", spectral_window_row)?
577                .filter(|count| *count > 0)
578                .ok_or(SelectedObservationContentPlanError::InvalidCoordinateShape)?;
579        let spectral_scratch_bytes = frequency_count
580            .checked_add(width_count)
581            .and_then(|values| values.checked_mul(size_of::<f64>()))
582            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
583        let polarization_row = usize::try_from(description.polarization_id())
584            .map_err(|_| SelectedObservationContentPlanError::InvalidCoordinateShape)?;
585        let full_correlation_count =
586            selected_i32_array_len(polarizations.table(), "CORR_TYPE", polarization_row)?
587                .filter(|count| *count > 0)
588                .ok_or(SelectedObservationContentPlanError::InvalidCoordinateShape)?;
589        let correlation_scratch_bytes = full_correlation_count
590            .checked_mul(size_of::<i32>())
591            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
592        // Frequency and width arrays are released before CORR_TYPE is loaded.
593        // Charge the larger payload plus the transient dynamic-array metadata.
594        let coordinate_scratch_bytes = spectral_scratch_bytes
595            .max(correlation_scratch_bytes)
596            // Three dynamic 1-D ndarrays retain one shape and one stride word
597            // each; CHAN_FREQ and CHAN_WIDTH are the simultaneous pair.
598            .checked_add(4 * size_of::<usize>())
599            // One selected-cell wrapper and its accessor name are transient
600            // while the already extracted array payloads remain live.
601            .and_then(|bytes| bytes.checked_add(size_of::<Option<casa_types::ArrayValue>>()))
602            .and_then(|bytes| {
603                ["CHAN_FREQ", "CHAN_WIDTH", "CORR_TYPE"]
604                    .iter()
605                    .map(|name| name.len())
606                    .max()
607                    .and_then(|name| bytes.checked_add(name))
608            })
609            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
610        maximum_scratch_bytes = maximum_scratch_bytes.max(coordinate_scratch_bytes);
611    }
612    // validate_data_descriptions retains one temporary DDID/wavelength vector
613    // while CompiledRowPredicate shares the compiler-owned predicate catalog.
614    let predicate_construction_scratch = source
615        .selection()
616        .data_descriptions()
617        .len()
618        .checked_mul(size_of::<(u32, f64)>())
619        .and_then(|bytes| bytes.checked_add(size_of::<Option<casa_types::ScalarValue>>()))
620        .and_then(|bytes| {
621            ["SPECTRAL_WINDOW_ID", "POLARIZATION_ID", "REF_FREQUENCY"]
622                .iter()
623                .map(|name| name.len())
624                .max()
625                .and_then(|name| bytes.checked_add(name))
626        })
627        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
628    maximum_scratch_bytes = maximum_scratch_bytes.max(predicate_construction_scratch);
629    let retained_bytes = shared_source_slots_retained_bytes
630        .checked_add(shared_measures_retained_bytes)
631        .and_then(|bytes| bytes.checked_add(storage_bytes))
632        .and_then(|bytes| bytes.checked_add(geometry_bytes))
633        .and_then(|bytes| bytes.checked_add(manifest_bytes))
634        .and_then(|bytes| bytes.checked_add(predicate_bytes))
635        .and_then(|bytes| bytes.checked_add(coordinate_bytes))
636        .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
637    Ok((
638        retained_bytes,
639        maximum_scratch_bytes,
640        pointing_reference_scratch_bytes,
641    ))
642}

644fn selected_f64_array_len(
645    table: &casa_tables::Table,
646    column: &str,
647    row: usize,
648) -> Result<Option<usize>, SelectedObservationContentPlanError> {
649    Ok(
650        match table
651            .column_accessor(column)
652            .map_err(MsError::from)?
653            .array_cells_owned_uncached(&[row])
654            .map_err(MsError::from)?
655            .pop()
656            .flatten()
657        {
658            Some(casa_types::ArrayValue::Float64(values)) => Some(values.len()),
659            _ => None,
660        },
661    )
662}

664fn selected_geometry_construction_scratch_bytes(
665    measurement_set: &MeasurementSet,
666) -> Result<usize, SelectedObservationContentPlanError> {
667    let antenna = measurement_set.antenna()?;
668    let mut maximum = 0_usize;
669    for row in 0..antenna.row_count() {
670        maximum = maximum
671            .max(selected_f64_array_scratch_bytes(
672                antenna.table(),
673                "POSITION",
674                row,
675            )?)
676            .max(selected_string_scratch_bytes(
677                antenna.table(),
678                "MOUNT",
679                row,
680            )?);
681    }
682    let field = measurement_set.field()?;
683    let direction_reference_column =
684        selected_direction_reference_column(field.table(), "PHASE_DIR");
685    for row in 0..field.row_count() {
686        maximum = maximum.max(selected_f64_array_scratch_bytes(
687            field.table(),
688            "PHASE_DIR",
689            row,
690        )?);
691        if let Some(reference_column) = direction_reference_column {
692            maximum = maximum.max(selected_scalar_scratch_bytes(
693                field.table(),
694                reference_column,
695                row,
696            )?);
697        }
698    }
699    if let Ok(observation) = measurement_set.observation() {
700        for row in 0..observation.row_count() {
701            if let Ok(bytes) =
702                selected_string_scratch_bytes(observation.table(), "TELESCOPE_NAME", row)
703            {
704                maximum = maximum.max(bytes);
705            }
706        }
707    }
708    Ok(maximum)
709}

711fn selected_pointing_reference_scratch_bytes(
712    measurement_set: &MeasurementSet,
713    problem: &CompiledProblem,
714) -> Result<usize, SelectedObservationContentPlanError> {
715    let PointingCentreLaw::Observation(law) = problem.geometry().centres().pointing() else {
716        return Ok(0);
717    };
718    let Ok(pointing) = measurement_set.pointing() else {
719        return Ok(0);
720    };
721    let column = match law.direction_column() {
722        casa_imaging_model::PointingDirectionColumn::Direction => "DIRECTION",
723        casa_imaging_model::PointingDirectionColumn::Target => "TARGET",
724    };
725    let Some(reference_column) = selected_direction_reference_column(pointing.table(), column)
726    else {
727        return Ok(0);
728    };
729    (0..pointing.row_count()).try_fold(0_usize, |maximum, row| {
730        Ok(maximum.max(selected_scalar_scratch_bytes(
731            pointing.table(),
732            reference_column,
733            row,
734        )?))
735    })
736}

738fn selected_f64_array_scratch_bytes(
739    table: &casa_tables::Table,
740    column: &str,
741    row: usize,
742) -> Result<usize, SelectedObservationContentPlanError> {
743    let value = table
744        .column_accessor(column)
745        .map_err(MsError::from)?
746        .array_cells_owned_uncached(&[row])
747        .map_err(MsError::from)?
748        .pop()
749        .flatten()
750        .ok_or(SelectedObservationContentPlanError::InvalidCoordinateShape)?;
751    let casa_types::ArrayValue::Float64(values) = value else {
752        return Err(SelectedObservationContentPlanError::InvalidCoordinateShape);
753    };
754    values
755        .len()
756        .checked_mul(size_of::<f64>())
757        .and_then(|bytes| {
758            values
759                .ndim()
760                .checked_mul(2 * size_of::<usize>())
761                .and_then(|dimensions| bytes.checked_add(dimensions))
762        })
763        .and_then(|bytes| bytes.checked_add(size_of::<Option<casa_types::ArrayValue>>()))
764        .and_then(|bytes| bytes.checked_add(column.len()))
765        .ok_or(SelectedObservationContentPlanError::ByteOverflow)
766}

768fn selected_string_scratch_bytes(
769    table: &casa_tables::Table,
770    column: &str,
771    row: usize,
772) -> Result<usize, SelectedObservationContentPlanError> {
773    let value = selected_scalar_scratch_value(table, column, row)?;
774    if !matches!(value, casa_types::ScalarValue::String(_)) {
775        return Err(SelectedObservationContentPlanError::InvalidCoordinateShape);
776    }
777    scalar_scratch_bytes(&value, column)
778}

780fn selected_scalar_scratch_bytes(
781    table: &casa_tables::Table,
782    column: &str,
783    row: usize,
784) -> Result<usize, SelectedObservationContentPlanError> {
785    let value = selected_scalar_scratch_value(table, column, row)?;
786    scalar_scratch_bytes(&value, column)
787}
788
789fn selected_scalar_scratch_value(
790    table: &casa_tables::Table,
791    column: &str,
792    row: usize,
793) -> Result<casa_types::ScalarValue, SelectedObservationContentPlanError> {
794    let value = table
795        .column_accessor(column)
796        .map_err(MsError::from)?
797        .scalar_cells_owned_for_rows(&[row])
798        .map_err(MsError::from)?
799        .pop()
800        .flatten()
801        .ok_or(SelectedObservationContentPlanError::InvalidCoordinateShape)?;
802    Ok(value)
803}
804
805fn scalar_scratch_bytes(
806    value: &casa_types::ScalarValue,
807    column: &str,
808) -> Result<usize, SelectedObservationContentPlanError> {
809    let dynamic_bytes = match value {
810        casa_types::ScalarValue::String(value) => value.capacity(),
811        _ => 0,
812    };
813    dynamic_bytes
814        .checked_add(size_of::<Option<casa_types::ScalarValue>>())
815        .and_then(|bytes| bytes.checked_add(column.len()))
816        .ok_or(SelectedObservationContentPlanError::ByteOverflow)
817}
