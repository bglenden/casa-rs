2//! Reads of the CASA-named calibration surfaces, `applycal`, `gaincal`,
3//! `bandpass`, `fluxscale` and `gencal`, into their requests (#931).
4//!
5//! These surfaces take CASA 6.7.6's parameters and values, read as CASA's
6//! task scripts read them:
7//!
8//! - `field`, `spw` and `intent` always select; the other selectors only
9//!   while `selectdata` is set, as CASA resets them otherwise.
10//! - The chain is `callib` when `docallib` is set, else `gaintable`. Its
11//!   per-table lists are positional: `gaintable[i]` takes `gainfield[i]`,
12//!   `interp[i]` and `spwmap[i]`, and a list shorter than `gaintable` leaves
13//!   the later tables CASA's defaults (every field, `'linear'`, no window
14//!   map). An empty `gaintable` entry is skipped. `calwt` with one value
15//!   holds for every table; a solve applies each table with `calwt=True`.
16//!
17//! A value CASA takes that casa-rs does not implement yet is refused naming
18//! #979; the session already refuses the parameters casa-rs does not
19//! implement at all unless they hold CASA's default.
20

44/// The request a CASA-named surface makes.
45pub(super) fn task_request(
46    task: &TaskInvocation<'_>,
47) -> Result<CalibrationTaskRequest, TaskFailure> {
48    Ok(match task.surface() {
49        "applycal" => CalibrationTaskRequest::ExecuteApply(applycal(task)?),
50        "gaincal" => CalibrationTaskRequest::SolveGain(gaincal(task)?),
51        "fringefit" => CalibrationTaskRequest::SolveGain(fringefit(task)?),
52        "accor" => CalibrationTaskRequest::SolveAccor(accor(task)?),
53        "blcal" => blcal(task)?,
54        "bandpass" => CalibrationTaskRequest::SolveBandpass(bandpass(task)?),
55        "fluxscale" => CalibrationTaskRequest::FluxScale(fluxscale(task)?),
56        "gencal" => CalibrationTaskRequest::Gencal(gencal(task)?),
57        other => {
58            return Err(TaskFailure::validation(format!(
59                "calibrate runs no surface {other:?}"
60            )));
61        }
62    })
63}

65fn applycal(task: &TaskInvocation<'_>) -> Result<ExecuteApplyTaskRequest, TaskFailure> {
66    let (calibration_tables, callibrary_error, argument_error) = if task.flag("docallib")? {
67        let source = task.text("callib")?.unwrap_or_default();
68        match load_apply_specs_from_callib_at(&task.io_path(&source), task.workspace()) {
69            Ok(tables) => (tables, None, None),
70            Err(error) => (Vec::new(), Some(error.to_string()), None),
71        }
72    } else {
73        let (tables, error) = traditional_chain(task, Chain::Apply)?;
74        (tables, None, error)
75    };
76    Ok(ExecuteApplyTaskRequest {
77        measurement_set: task.path("vis")?,
78        selection: selection(task)?,
79        calibration_tables,
80        callibrary_error,
81        argument_error,
82        apply_mode: applymode(task)?,
83        parang: task.flag("parang")?,
84        flagbackup: task.flag("flagbackup")?,
85    })
86}

88fn gaincal(task: &TaskInvocation<'_>) -> Result<SolveGainTaskRequest, TaskFailure> {
89    let gain_type = gaintype(task)?;
90    let spline = gain_type == GainType::GSpline;
91    let spline_controls = if spline {
92        Some(GainSplineControls {
93            splinetime: task.number("splinetime")?,
94            npointaver: task.integer("npointaver")?,
95            phasewrap: task.number("phasewrap")?,
96        })
97    } else {
98        None
99    };
100    let measurement_set = task.path("vis")?;
101    let min_snr = if spline { 0. } else { min_snr(task)? };
102    let (model_source, smodel) = model(task, &measurement_set, true)?;
103    let reference = task.required_text("refant")?;
104    let strict = match task.required_text("refantmode")?.as_str() {
105        _ if spline => false,
106        "flex" => false,
107        "strict" => true,
108        other => return Err(unknown("refantmode", other, "'flex' or 'strict'")),
109    };
110    Ok(SolveGainTaskRequest {
111        append: task.flag("append")?,
112        selection: selection(task)?,
113        output_table: task.path("caltable")?,
114        gain_type,
115        spline_controls,
116        fringe_controls: None,
117        solve_mode: calmode(task)?,
118        solve_interval: if spline {
119            GainSolveInterval::Infinite
120        } else {
121            solint(task)?
122        },
123        preaverage_seconds: task.number("preavg")?,
124        correlation_dependent_flags: task.flag("corrdepflags")?,
125        combine: if spline {
126            GainSolveCombine::default()
127        } else {
128            combine(task)?
129        },
130        refant: if reference.is_empty() {
131            RefAntSelector::Unspecified
132        } else {
133            RefAntSelector::Expression {
134                selection: reference,
135                strict,
136            }
137        },
138        prior_calibration_tables: chain(task, Chain::Solve)?,
139        parang: task.flag("parang")?,
140        model_source,
141        normalize_average_amplitude: task.flag("solnorm")?,
142        normalization_type: match task.required_text("normtype")?.as_str() {
143            "mean" => GainNormalization::Mean,
144            "median" => GainNormalization::Median,
145            other => return Err(unknown("normtype", other, "'mean' or 'median'")),
146        },
147        robustness: if spline {
148            Default::default()
149        } else {
150            robust_gain_solve(task)?
151        },
152        min_snr,
153        min_baselines_per_antenna: if spline {
154            0
155        } else {
156            count(task, "minblperant")?
157        },
158        smodel,
159        measurement_set,
160    })
161}

163fn fringefit(task: &TaskInvocation<'_>) -> Result<SolveGainTaskRequest, TaskFailure> {
164    let reference = task.required_text("refant")?;
165    let active_parameters = task.flags("paramactive")?;
166    let active_parameters = match active_parameters.as_slice() {
167        [] => [true, true, false],
168        [delay, rate, dispersion] => [*delay, *rate, *dispersion],
169        _ => {
170            return Err(TaskFailure::validation(
171                "paramactive vector must have exactly three entries",
172            ));
173        }
174    };
175    let search_window = |name| -> Result<[f64; 2], TaskFailure> {
176        Ok(task.numbers(name)?.try_into().unwrap_or([-1e6, 1e6]))
177    };
178    let iterations = usize::try_from(task.integer("niter")?).unwrap_or(usize::MAX);
179    let correlation_combination = match task.required_text("corrcomb")?.as_str() {
180        "none" => casa_calibration::solve::FringeCorrelation::Separate,
181        "stokes" => casa_calibration::solve::FringeCorrelation::Stokes,
182        "parallel" => casa_calibration::solve::FringeCorrelation::Parallel,
183        other => return Err(unknown("corrcomb", other, "'none', 'stokes' or 'parallel'")),
184    };
185    Ok(SolveGainTaskRequest {
186        measurement_set: task.path("vis")?,
187        selection: selection(task)?,
188        output_table: task.path("caltable")?,
189        append: task.flag("append")?,
190        gain_type: GainType::K,
191        spline_controls: None,
192        fringe_controls: Some(casa_calibration::solve::FringeControls {
193            delay_window: search_window("delaywindow")?,
194            rate_window: search_window("ratewindow")?,
195            global_solve: task.flag("globalsolve")?,
196            iterations,
197            active_parameters,
198            zero_rates: task.flag("zerorates")?,
199            correlation_combination,
200            concatenate_windows: task.flag("concatspws")?,
201        }),
202        solve_mode: GainSolveMode::Phase,
203        solve_interval: solint(task)?,
204        preaverage_seconds: 0.,
205        correlation_dependent_flags: task.flag("corrdepflags")?,
206        combine: combine(task)?,
207        refant: if reference.is_empty() {
208            RefAntSelector::Unspecified
209        } else {
210            RefAntSelector::Expression {
211                selection: reference,
212                strict: false,
213            }
214        },
215        prior_calibration_tables: chain(task, Chain::Solve)?,
216        parang: task.flag("parang")?,
217        model_source: GainSolveModelSource::FromMeasurementSet,
218        normalize_average_amplitude: false,
219        normalization_type: GainNormalization::Mean,
220        robustness: Default::default(),
221        min_snr: task.number("minsnr")? as f32,
222        min_baselines_per_antenna: 0,
223        smodel: [1., 0., 0., 0.],
224    })
225}

254fn blcal(task: &TaskInvocation<'_>) -> Result<CalibrationTaskRequest, TaskFailure> {
255    if task.flag("selectdata")?
256        && let Some(ParameterValue::Bool(observation)) = task.value("observation")
257    {
258        let name = if *observation { "True" } else { "False" };
259        let first = name.chars().next().expect("boolean text");
260        return Err(TaskFailure::execution(format!(
261            "Error in data selection specification: Scan Expression: Parse error at or near '{first}'\n(near char. 1 in string \"{name}\")"
262        )));
263    }
264    let frequency_dependent = task.flag("freqdep")?;
265    let output_table = task.required_text("caltable")?;
266    if output_table.is_empty() {
267        return Err(TaskFailure::execution("Error in Calibrater::setsolve."));
268    }
269    let supplied_interval = selector_text(task.value("solint")).unwrap_or_default();
270    let (time, frequency) = supplied_interval
271        .split_once(',')
272        .unwrap_or((&supplied_interval, ""));
273    let frequency_interval = if frequency_dependent {
274        parse_frequency_solint(frequency)?
275    } else {
276        None
277    };
278    Ok(CalibrationTaskRequest::SolveBaseline {
279        frequency_dependent,
280        frequency_interval,
281        gain: SolveGainTaskRequest {
282            measurement_set: task.path("vis")?,
283            selection: selection(task)?,
284            output_table: task.io_path(output_table),
285            append: false,
286            gain_type: GainType::G,
287            spline_controls: None,
288            fringe_controls: None,
289            solve_mode: calmode(task)?,
290            solve_interval: parse_time_solint(time)?,
291            preaverage_seconds: 0.,
292            correlation_dependent_flags: false,
293            combine: combine(task)?,
294            refant: RefAntSelector::Unspecified,
295            prior_calibration_tables: traditional_chain(task, Chain::Solve)?.0,
296            parang: task.flag("parang")?,
297            model_source: GainSolveModelSource::FromMeasurementSet,
298            normalize_average_amplitude: task.flag("solnorm")?,
299            normalization_type: GainNormalization::Mean,
300            robustness: Default::default(),
301            min_snr: 0.,
302            min_baselines_per_antenna: 0,
303            smodel: [1., 0., 0., 0.],
304        },
305    })
306}

308fn bandpass(task: &TaskInvocation<'_>) -> Result<SolveBandpassTaskRequest, TaskFailure> {
309    let measurement_set = task.path("vis")?;
310    let band_type = match task.required_text("bandtype")?.as_str() {
311        "B" => BandpassType::B,
312        "BPOLY" => BandpassType::BPoly,
313        other => return Err(unknown("bandtype", other, "'B' or 'BPOLY'")),
314    };
315    let (model_source, smodel) = model(task, &measurement_set, true)?;
316    let supplied_interval = selector_text(task.value("solint")).unwrap_or_default();
317    let (time_interval, frequency_interval) = supplied_interval
318        .split_once(',')
319        .unwrap_or((&supplied_interval, ""));
320    let combine = combine(task)?;
321    let degrees = [task.integer("degamp")?, task.integer("degphase")?];
322    // CASA reads maskedge only for BPOLY.
323    let mask_edge_percent = match band_type {
324        BandpassType::B => BandpassSolveCuts::CASA_DEFAULTS.mask_edge_percent,
325        BandpassType::BPoly => task.number("maskedge")? as f32,
326    };
327    Ok(SolveBandpassTaskRequest {
328        append: task.flag("append")?,
329        selection: selection(task)?,
330        output_table: task.path("caltable")?,
331        refant: bandpass_refant(&task.required_text("refant")?),
332        prior_calibration_tables: chain(task, Chain::Solve)?,
333        parang: task.flag("parang")?,
334        combine: BandpassSolveCombine {
335            scans: combine.scans,
336            fields: combine.fields,
337            observations: combine.observations,
338            spectral_windows: combine.spectral_windows,
339        },
340        band_type,
341        normalize_average_amplitude: task.flag("solnorm")?,
342        amplitude_degree: degrees[0].max(0) as usize,
343        phase_degree: degrees[1].max(0) as usize,
344        model_source,
345        solve_interval: parse_time_solint(time_interval)?,
346        frequency_interval: parse_frequency_solint(frequency_interval)?,
347        correlation_dependent_flags: task.flag("corrdepflags")?,
348        smodel,
349        cuts: Some(BandpassSolveCuts {
350            min_baselines_per_antenna: task.integer("minblperant")?.max(0) as usize,
351            min_snr: task.number("minsnr")? as f32,
352            fill_gaps: match band_type {
353                BandpassType::B => {
354                    usize::try_from(task.integer("fillgaps")?.max(0)).unwrap_or(usize::MAX)
355                }
356                BandpassType::BPoly => 0,
357            },
358            mask_edge_percent,
359            mask_center_channels: if band_type == BandpassType::BPoly {
360                usize::try_from(task.integer("maskcenter")?.max(0)).unwrap_or(usize::MAX)
361            } else {
362                0
363            },
364            normalize_visibilities: band_type == BandpassType::BPoly && task.flag("visnorm")?,
365            omit_polynomial: degrees.map(|degree| degree == -1),
366        }),
367        measurement_set,
368    })
369}

371pub(super) fn leakage_request(
372    task: &TaskInvocation<'_>,
373    vis: &Path,
374) -> Result<casa_calibration::solve::grouping::leakage::LeakageSolveRequest, TaskFailure> {
375    use casa_calibration::solve::grouping::leakage::{LeakageSolveKind, LeakageSolveRequest};
376
377    let gain = task.surface() == "gaincal";
378    let poltype = task.required_text(if gain { "gaintype" } else { "poltype" })?;
379    let kind = match poltype.as_str() {
380        "D" => LeakageSolveKind::D,
381        "Df" => LeakageSolveKind::Df,
382        "D+QU" => LeakageSolveKind::DQu,
383        "Df+QU" => LeakageSolveKind::DfQu,
384        "D+X" => LeakageSolveKind::DX,
385        "Df+X" => LeakageSolveKind::DfX,
386        "Dgen" => LeakageSolveKind::DGen,
387        "Dfgen" => LeakageSolveKind::DfGen,
388        "Dgen+QU" => LeakageSolveKind::DGenQu,
389        "Dfgen+QU" => LeakageSolveKind::DfGenQu,
390        "Dgen+X" => LeakageSolveKind::DGenX,
391        "Dfgen+X" => LeakageSolveKind::DfGenX,
392        "Xf" => LeakageSolveKind::Xf,
393        "Xj" => LeakageSolveKind::Xj,
394        "PosAng" => LeakageSolveKind::PosAng,
395        "Xparang+QU" => LeakageSolveKind::Xparang,
396        "Xfparang+QU" => LeakageSolveKind::Xfparang,
397        "XY+QU" if gain => LeakageSolveKind::XY,
398        "XYf+QU" if gain => LeakageSolveKind::XYf,
399        "X" => {
400            return Err(TaskFailure::execution(
401                "Can't do selfGatherAndSolve for X Mueller",
402            ));
403        }
404        "Dlls" => LeakageSolveKind::Dlls,
405        "Dflls" => LeakageSolveKind::Dflls,
406        _ => {
407            return Err(TaskFailure::validation(format!(
408                "poltype={poltype:?} is under implementation in #1522"
409            ))
410            .at("poltype"));
411        }
412    };
413    let (model_source, stokes) = model(task, vis, true)?;
414    let interval = selector_text(task.value("solint")).unwrap_or_default();
415    let (time, frequency) = interval.split_once(',').unwrap_or((&interval, ""));
416    Ok(LeakageSolveRequest {
417        output: task.path("caltable")?,
418        selection: selection(task)?,
419        kind,
420        model_source,
421        stokes,
422        priors: chain(task, Chain::Solve)?,
423        parang_terms: usize::from(!gain || task.flag("parang")?),
424        correlation_dependent_flags: gain && task.flag("corrdepflags")?,
425        solve_mode: if gain {
426            calmode(task)?
427        } else {
428            GainSolveMode::AmplitudePhase
429        },
430        normalization: None,
431        interval: parse_time_solint(time)?,
432        frequency_interval: parse_frequency_solint(frequency)?,
433        combine: combine(task)?,
434        preaverage_seconds: task.number("preavg")?,
435        min_baselines: task.integer("minblperant")?.max(0) as usize,
436        min_snr: task.number("minsnr")? as f32,
437        refant: bandpass_refant(&task.required_text("refant")?),
438        append: task.flag("append")?,
439    })
440}

442/// `minsnr`, which CASA takes as any number; casa-rs takes it finite and
443/// not negative.
444fn min_snr(task: &TaskInvocation<'_>) -> Result<f32, TaskFailure> {
445    let min_snr = task.number("minsnr")? as f32;
446    if !min_snr.is_finite() || min_snr < 0.0 {
447        return Err(
448            TaskFailure::validation("minsnr must be a finite non-negative number").at("minsnr"),
449        );
450    }
451    Ok(min_snr)
452}

454fn fluxscale(task: &TaskInvocation<'_>) -> Result<FluxScaleRequest, TaskFailure> {
455    // CASA opens `vis` for its field and window names; casa-rs reads them
456    // from the gain table, but a MeasurementSet that is not there is still
457    // an error, as it is to CASA.
458    let vis = task.path("vis")?;
459    if !vis.exists() {
460        return Err(TaskFailure::validation(format!(
461            "MeasurementSet {} does not exist",
462            vis.display()
463        ))
464        .at("vis"));
465    }
466    let reference_fields = fields(task, "reference")?;
467    if reference_fields.is_empty() {
468        return Err(
469            TaskFailure::validation("fluxscale requires at least one reference field")
470                .at("reference"),
471        );
472    }
473    let refspwmap = task
474        .integers("refspwmap")?
475        .into_iter()
476        .map(|id| i32::try_from(id).map_err(|_| out_of_range("refspwmap", id)))
477        .collect::<Result<Vec<_>, _>>()?;
478    let gainthreshold = task.number("gainthreshold")?;
479    Ok(FluxScaleRequest {
480        options: casa_calibration::fluxscale::FluxScaleOptions {
481            measurement_set: Some(vis),
482            antenna: task.text("antenna")?.unwrap_or_default(),
483            timerange: task.text("timerange")?.unwrap_or_default(),
484            scan: task.text("scan")?.unwrap_or_default(),
485            fitorder: i32::try_from(task.integer("fitorder")?).map_err(|_| {
486                TaskFailure::validation("fitorder exceeds CASA's integer range").at("fitorder")
487            })?,
488            listfile: task
489                .text("listfile")?
490                .filter(|text| !text.is_empty())
491                .map(|path| task.io_path(path)),
492            display: task.flag("display")?,
493        },
494        append: task.flag("append")?,
495        input_table: task.path("caltable")?,
496        output_table: task.path("fluxtable")?,
497        reference_fields,
498        transfer_fields: fields(task, "transfer")?,
499        // CASA's [-1] maps no window.
500        refspwmap: if refspwmap == [-1] {
501            Vec::new()
502        } else {
503            refspwmap
504        },
505        // CASA's -1 excludes no gain.
506        gainthreshold: (gainthreshold >= 0.0).then_some(gainthreshold),
507        incremental: task.flag("incremental")?,
508    })
509}

511fn gencal(task: &TaskInvocation<'_>) -> Result<GencalTaskRequest, TaskFailure> {
512    let caltype = match task.required_text("caltype")?.as_str() {
513        "" => {
514            return Err(TaskFailure::validation("gencal requires a caltype").at("caltype"));
515        }
516        "amp" => GencalType::Amp,
517        "ph" => GencalType::Ph,
518        "sbd" => GencalType::Sbd,
519        "mbd" => GencalType::Mbd,
520        "antpos" => GencalType::Antpos,
521        "antposvla" => GencalType::AntposVla,
522        "gc" => GencalType::Gc,
523        "gceff" => GencalType::Gceff,
524        "eff" => GencalType::Eff,
525        "tsys" => GencalType::Tsys,
526        "opac" => GencalType::Opac,
527        "swpow" => GencalType::Swpow,
528        "evlagain" => GencalType::Evlagain,
529        "swp/rq" => GencalType::SwpOverRq,
530        "swpwts" => GencalType::Swpwts,
531        "rq" => GencalType::Rq,
532        "tecim" => GencalType::Tecim,
533        "eop" => GencalType::Eop,
534        other => return Err(not_yet("caltype", other)),
535    };
536    let antenna = task.text("antenna")?.unwrap_or_default();
537    let infile = if caltype.is_gain_curve()
538        || matches!(
539            caltype,
540            GencalType::Antpos | GencalType::Tecim | GencalType::Eop
541        ) {
542        task.text("infile")?
543            .filter(|path| !path.is_empty() && !(caltype == GencalType::Antpos && path == "None"))
544            .map(|path| task.io_path(path))
545    } else {
546        None
547    };
548    if caltype == GencalType::Tecim && !infile.as_ref().is_some_and(|path| path.exists()) {
549        return Err(
550            TaskFailure::validation("An existing tec map must be specified in infile").at("infile"),
551        );
552    }
553    Ok(GencalTaskRequest {
554        prior: GencalRequest {
555            measurement_set: task.path("vis")?,
556            output_table: task.path("caltable")?,
557            caltype,
558            antenna,
559            spw: task.text("spw")?.unwrap_or_default(),
560            parameter: task.numbers("parameter")?,
561            gaincurve_table: task.text("gaincurve_table")?.map(|path| task.io_path(path)),
562        },
563        polarization: task.text("pol")?.unwrap_or_default(),
564        gaincurve_infile: infile.clone().filter(|_| caltype.is_gain_curve()),
565        antpos_infile: infile.clone().filter(|_| caltype == GencalType::Antpos),
566        tec_image: infile.clone().filter(|_| caltype == GencalType::Tecim),
567        eop_infile: infile.filter(|_| caltype == GencalType::Eop),
568        ant_pos_time_limit: (caltype == GencalType::Antpos)
569            .then(|| task.integer("ant_pos_time_limit"))
570            .transpose()?,
571        uniform: (caltype == GencalType::Tsys)
572            .then(|| task.flag("uniform"))
573            .transpose()?,
574    })
575}

577/// CASA's selection: `field`, `spw` and `intent` always; the others are inactive,
578/// and so unset, while `selectdata` is not.
579fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
580    let text = |name: &str| {
581        task.text(name)
582            .map(|value| value.filter(|text| !text.is_empty()))
583    };
584    let mut selection = MsSelection {
585        selectdata: true,
586        field: text("field")?,
587        spw: text("spw")?,
588        antenna: text("antenna")?,
589        scan: text("scan")?,
590        observation: selector_text(task.value("observation")),
591        array: text("array")?,
592        timerange: text("timerange")?,
593        msselect: text("msselect")?,
594        ..MsSelection::default()
595    };
596    selection.intent = task.text("intent")?.filter(|text| !text.is_empty());
597    selection.uvrange = uv_selector_text(task.value("uvrange"));
598    // calibrater.selectvis converts these variants with toCasaString, which
599    // removes one outer bracket pair even when the input was a string.
600    for selector in [
601        &mut selection.field,
602        &mut selection.spw,
603        &mut selection.intent,
604        &mut selection.timerange,
605        &mut selection.uvrange,
606        &mut selection.antenna,
607        &mut selection.scan,
608        &mut selection.observation,
609    ] {
610        if let Some(text) = selector
611            && let Some(inner) = text
612                .strip_prefix('[')
613                .and_then(|text| text.strip_suffix(']'))
614        {
615            *selector = (!inner.is_empty()).then(|| inner.to_string());
616        }
617    }
618    // CASA accepts these parameters even when disabled, retaining their input
619    // values for HISTORY while passing empty selectors to calibrater.selectvis.
620    if !task.flag("selectdata")? {
621        selection.timerange = None;
622        selection.uvrange = None;
623        selection.antenna = None;
624        selection.scan = None;
625        selection.observation = None;
626        selection.msselect = None;
627    }
628    if matches!(task.surface(), "gaincal" | "bandpass" | "polcal") {
629        // The public task always adds this predicate, including when the
630        // optional selectors are disabled.
631        selection.msselect = Some(match selection.msselect {
632            Some(predicate) => format!("({predicate}) && ANTENNA1!=ANTENNA2"),
633            None => "ANTENNA1!=ANTENNA2".into(),
634        });
635    }
636    Ok(selection)
637}

639pub(super) fn selector_text(value: Option<&ParameterValue>) -> Option<String> {
640    let text = match value? {
641        ParameterValue::String(text) => text.clone(),
642        ParameterValue::NumpyScalar { value, .. } | ParameterValue::ScalarArray(value) => {
643            return selector_text(Some(value));
644        }
645        other => python_repr(other),
646    };
647    (!text.is_empty()).then_some(text)
648}

650pub(super) fn uv_selector_text(value: Option<&ParameterValue>) -> Option<String> {
651    match value? {
652        ParameterValue::Array(values) => {
653            let text = values
654                .iter()
655                .filter_map(|value| selector_text(Some(value)))
656                .collect::<Vec<_>>()
657                .join(",");
658            (!text.is_empty()).then_some(text)
659        }
660        ParameterValue::EmptyArray { .. } => None,
661        value => selector_text(Some(value)),
662    }
663}

665/// Whether a chain is applied, each table with its own `calwt`, or
666/// pre-applied while solving, each with `calwt=True`.
667#[derive(Debug, Clone, Copy, PartialEq, Eq)]
668enum Chain {
669    Apply,
670    Solve,
671}
672
673/// The chain `callib` names when `docallib` is set, else `gaintable` with
674/// its positional lists.
675fn chain(
676    task: &TaskInvocation<'_>,
677    chain: Chain,
678) -> Result<Vec<ApplyCalibrationTableSpec>, TaskFailure> {
679    if task.flag("docallib")? {
680        let callib = task.text("callib")?.ok_or_else(|| {
681            TaskFailure::validation("docallib requires callib, a calibration library file")
682                .at("callib")
683        })?;
684        return load_apply_specs_from_callib_at(&task.io_path(&callib), task.workspace())
685            .map_err(|error| TaskFailure::validation(error.to_string()).at("callib"));
686    }
687    traditional_chain(task, chain).map(|(tables, _)| tables)
688}
689
690fn traditional_chain(
691    task: &TaskInvocation<'_>,
692    chain: Chain,
693) -> Result<(Vec<ApplyCalibrationTableSpec>, Option<ApplyArgumentError>), TaskFailure> {
694    let tables = task.texts("gaintable")?;
695    let gainfield = task.texts("gainfield")?;
696    let interp = task.texts("interp")?;
697    let spwmap = match apply_spw_maps(task.value("spwmap")) {
698        Ok(maps) => maps,
699        Err(error) if chain == Chain::Apply => return Ok((Vec::new(), Some(error))),
700        Err(error) => return Err(TaskFailure::validation(error.to_string()).at("spwmap")),
701    };
702    let calwt = match chain {
703        Chain::Apply => task.flags("calwt")?,
704        Chain::Solve => vec![true],
705    };
706    let mut specs = Vec::new();
707    for (index, table) in tables.iter().enumerate() {
708        if table.is_empty() {
709            continue;
710        }
711        let weight = match calwt.as_slice() {
712            [one] => *one,
713            values => match values.get(index) {
714                Some(weight) => *weight,
715                // Python indexes calwt immediately before setapply for an active
716                // table. All earlier table setup still takes place first.
717                None => return Ok((specs, Some(ApplyArgumentError::MissingCalwt))),
718            },
719        };
720        let mut spec = ApplyCalibrationTableSpec::new(task.io_path(table));
721        spec.gainfield = gainfield
722            .get(index)
723            .and_then(|field| parse_gainfield_value(field));
724        spec.interp = match interp.get(index) {
725            Some(text) => parse_interp_value(text)
726                .unwrap_or(casa_calibration::apply::ApplyInterpolationMode::Invalid),
727            None => casa_calibration::apply::ApplyInterpolationMode::Linear,
728        };
729        if let Some(map) = spwmap.get(index) {
730            match apply_spw_map(map) {
731                Ok(map) => spec.spwmap = map,
732                Err(error) if chain == Chain::Apply => return Ok((specs, Some(error))),
733                Err(error) => return Err(TaskFailure::validation(error.to_string()).at("spwmap")),
734            }
735        }
736        spec.calwt = weight;
737        specs.push(spec);
738    }
739    Ok((specs, None))
740}
741
742fn apply_spw_maps(
743    value: Option<&ParameterValue>,
744) -> Result<Vec<ParameterValue>, ApplyArgumentError> {
745    Ok(match value {
746        Some(ParameterValue::Array(items))
747            if !items.is_empty() && !matches!(items[0], ParameterValue::Array(_)) =>
748        {
749            vec![ParameterValue::Array(items.clone())]
750        }
751        Some(ParameterValue::Array(items)) => items.clone(),
752        Some(ParameterValue::String(text)) if text.is_empty() => Vec::new(),
753        Some(ParameterValue::Table(entries)) if entries.is_empty() => Vec::new(),
754        Some(ParameterValue::Table(_)) => return Err(ApplyArgumentError::SpwMap("0".into())),
755        Some(ParameterValue::EmptyArray { .. }) | None => Vec::new(),
756        Some(value) => {
757            let ty = match value {
758                ParameterValue::Integer(_) => "int",
759                ParameterValue::Float(_) => "float",
760                ParameterValue::Bool(_) => "bool",
761                _ => {
762                    return Err(ApplyArgumentError::SpwMap(
763                        "expected an integer window map".into(),
764                    ));
765                }
766            };
767            return Err(ApplyArgumentError::SpwMap(format!(
768                "object of type '{ty}' has no len()"
769            )));
770        }
771    })
772}
773
774fn apply_spw_map(value: &ParameterValue) -> Result<Vec<i32>, ApplyArgumentError> {
775    let ParameterValue::Array(items) = value else {
776        return Err(ApplyArgumentError::SpwMap(
777            "expected an integer window map".into(),
778        ));
779    };
780    let ids = items
781        .iter()
782        .map(|item| match item {
783            ParameterValue::Integer(id) => Ok(*id),
784            ParameterValue::Bool(id) => Ok(i64::from(*id)),
785            ParameterValue::NumpyScalar { value, .. }
786                if matches!(value.as_ref(), ParameterValue::Integer(_)) =>
787            {
788                let ParameterValue::Integer(id) = value.as_ref() else {
789                    unreachable!()
790                };
791                Ok(*id)
792            }
793            _ => Err(ApplyArgumentError::SpwMap(
794                "expected an integer window map".into(),
795            )),
796        })
797        .collect::<Result<Vec<_>, _>>()?;
798    window_map(&ids).map_err(|failure| ApplyArgumentError::SpwMap(failure.message))
799}
800
801fn window_map(map: &[i64]) -> Result<Vec<i32>, TaskFailure> {
802    if map == [-1] {
803        return Ok(Vec::new());
804    }
805    map.iter()
806        .map(|&id| i32::try_from(id).map_err(|_| out_of_range("spwmap", id)))
807        .collect()
808}

810/// Parse the point-source Stokes vector, with omitted Stokes set to zero.
811/// Gaincal and channelized bandpass resolve an empty vector from the MS.
812fn model(
813    task: &TaskInvocation<'_>,
814    measurement_set: &Path,
815    use_stored_models: bool,
816) -> Result<(GainSolveModelSource, [f32; 4]), TaskFailure> {
817    let smodel = task.numbers("smodel")?;
818    match smodel.as_slice() {
819        [] if use_stored_models => {
820            std::fs::metadata(measurement_set).map_err(|error| {
821                TaskFailure::validation(format!(
822                    "open MeasurementSet {}: {error}",
823                    measurement_set.display()
824                ))
825                .at("vis")
826            })?;
827            Ok((
828                GainSolveModelSource::FromMeasurementSet,
829                [1.0, 0.0, 0.0, 0.0],
830            ))
831        }
832        [] => {
833            let ms = MeasurementSet::open(measurement_set).map_err(|error| {
834                TaskFailure::validation(format!(
835                    "open MeasurementSet {}: {error}",
836                    measurement_set.display()
837                ))
838                .at("vis")
839            })?;
840            let virtual_fields = casa_ms::virtual_model_fields(&ms);
841            if !virtual_fields.is_empty() {
842                return Err(TaskFailure::validation(format!(
843                    "smodel=[] reads the virtual model of field(s) {virtual_fields:?} (setjy with \
844                     usescratch=False), which casa-rs does not implement yet (#979); give smodel, \
845                     or write MODEL_DATA"
846                ))
847                .at("smodel"));
848            }
849            let source = if has_model_column(&ms) {
850                GainSolveModelSource::ModelColumn
851            } else {
852                GainSolveModelSource::PointSource
853            };
854            Ok((source, [1.0, 0.0, 0.0, 0.0]))
855        }
856        &[_, ref polarized @ ..] if polarized.len() < 4 => {
857            if !use_stored_models && polarized.iter().any(|stokes| *stokes != 0.0) {
858                return Err(not_yet("smodel", &format!("{smodel:?}")));
859            }
860            let stokes = casa_calibration::solve::point_model_stokes(&smodel).ok_or_else(|| {
861                TaskFailure::validation("smodel needs finite Stokes values representable as floats")
862                    .at("smodel")
863            })?;
864            Ok((GainSolveModelSource::PointSource, stokes))
865        }
866        _ => Err(TaskFailure::validation(format!(
867            "smodel takes [I, Q, U, V], or fewer with the rest zero, or [] for the MODEL_DATA \
868             column; found {} values",
869            smodel.len()
870        ))
871        .at("smodel")),
872    }
873}

927/// CASA's `solint`, read as `SolvableVisCal::setSolve` reads it: empty
928/// text, or text that contains `inf`, in any case, is one solution per
929/// scan; text that contains `int`, one per integration; anything else is a
930/// time, in seconds when it has no unit. A time of zero is CASA's `inf`,
931/// and a negative one its `int`. G and T have a single gain per spectral
932/// window, so CASA ignores the frequency interval after a comma.
933fn solint(task: &TaskInvocation<'_>) -> Result<GainSolveInterval, TaskFailure> {
934    let supplied = selector_text(task.value("solint")).unwrap_or_default();
935    let time = supplied
936        .split_once(',')
937        .map_or(supplied.as_str(), |(time, _)| time);
938    parse_time_solint(time)
939}

944}
