22#[pymethods]
23impl PyImageTool {
24    /// Collapse masked image profiles and return the first requested moment image.
25    #[allow(clippy::too_many_arguments)]
26    #[pyo3(signature = (moments=None, axis=-10, region=None, mask=None, method=None, smoothaxes=None, smoothtypes=None, smoothwidths=None, includepix=None, excludepix=None, peaksnr=3., stddev=0., doppler="RADIO", outfile="", smoothout="", overwrite=false, drop=true, stretch=false), text_signature = "($self, moments=[0], axis=-10, region={}, mask='', method=[], smoothaxes=[-1], smoothtypes=[], smoothwidths=[0.0], includepix=[-1.0], excludepix=[-1.0], peaksnr=3.0, stddev=0.0, doppler='RADIO', outfile='', smoothout='', overwrite=False, drop=True, stretch=False)")]
27    fn moments(
28        &self,
29        py: Python<'_>,
30        moments: Option<IntList>,
31        axis: i64,
32        region: Option<&Bound<'_, PyAny>>,
33        mask: Option<&Bound<'_, PyAny>>,
34        method: Option<StringList>,
35        smoothaxes: Option<IntList>,
36        smoothtypes: Option<&Bound<'_, PyAny>>,
37        smoothwidths: Option<DoubleList>,
38        includepix: Option<DoubleList>,
39        excludepix: Option<DoubleList>,
40        peaksnr: f64,
41        stddev: f64,
42        doppler: &str,
43        outfile: &str,
44        smoothout: &str,
45        #[pyo3(from_py_with = require_bool)] overwrite: bool,
46        #[pyo3(from_py_with = require_bool)] drop: bool,
47        #[pyo3(from_py_with = require_bool)] stretch: bool,
48    ) -> PyResult<Py<Self>> {
49        let moments = moments
50            .map_or_else(|| vec![0], |values| values.0)
51            .into_iter()
52            .map(i32::try_from)
53            .collect::<Result<Vec<_>, _>>()
54            .map_err(raised)?;
55        let region: RegionArgument =
56            region.map_or_else(|| Ok(Default::default()), |value| value.extract())?;
57        let mask = mask_argument(mask)?;
58        let methods = method.map_or_else(Vec::new, |value| value.0);
59        let axes = smoothaxes.map_or_else(|| vec![-1], |value| value.0);
60        let kernels = match smoothtypes
61            .map(|value| value.extract::<Variant>())
62            .transpose()?
63        {
64            Some(Variant::Text(name)) => name
65                .split(|character: char| character == ',' || character.is_ascii_whitespace())
66                .filter(|name| !name.is_empty())
67                .map(str::to_owned)
68                .collect(),
69            Some(Variant::Texts(names)) => names,
70            _ => Vec::new(),
71        };
72        let widths = smoothwidths
73            .filter(|value| !value.0.is_empty())
74            .map_or_else(|| vec![0.], |value| value.0);
75        let include = includepix.map_or_else(|| vec![-1.], |value| value.0);
76        let exclude = excludepix.map_or_else(|| vec![-1.], |value| value.0);
77        let parsed = if doppler.eq_ignore_ascii_case("TRUE") {
78            Ok(DopplerRef::BETA)
79        } else {
80            doppler.parse::<DopplerRef>()
81        };
82        let logs = (!doppler.is_empty() && parsed.is_err())
83            .then(|| LogLine::warning(format!("Illegal velocity type {doppler}. Using RADIO")))
84            .into_iter()
85            .collect::<Vec<_>>();
86        let history = format!(
87            "ia.moments(moments={moments:?}, axis={axis}, region={}, mask={mask:?}, method={methods:?}, smoothaxes={axes:?}, smoothtypes={kernels:?}, smoothwidths={widths:?}, includepix={include:?}, excludepix={exclude:?}, peaksnr={peaksnr}, stddev={stddev}, doppler={doppler:?}, outfile={outfile:?}, smoothout={smoothout:?}, overwrite={}, drop={}, stretch={})",
88            region.history(),
89            python_bool(overwrite),
90            python_bool(drop),
91            python_bool(stretch),
92        );
93        let options = ImageMomentOptions {
94            moments: &moments,
95            axis,
96            region: region.selection(),
97            mask: Some(&mask),
98            stretch,
99            drop,
100            methods: &methods,
101            smoothing: (&axes, &kernels, &widths),
102            inclusion: &include,
103            exclusion: &exclude,
104            peak_snr: peaksnr as f32,
105            noise: stddev as f32,
106            doppler: parsed.unwrap_or(DopplerRef::RADIO),
107            destinations: &[],
108        };
109        let output = py.detach(|| {
110            let state = self.state();
111            let Some(input) = &state.image else {
112                return Ok(None);
113            };
114            build(
115                input,
116                options,
117                outfile.trim(),
118                smoothout.trim(),
119                overwrite,
120                state.history.then_some(history),
121            )
122            .map(Some)
123        })?;
124        if output.is_some() {
125            emit(py, "casars.tools.image", &logs)?;
126        }
127        Py::new(py, output.map_or_else(Self::new, Self::attached))
128    }
129}

131fn build(
132    input: &AttachedImage,
133    options: ImageMomentOptions<'_>,
134    outfile: &str,
135    smoothout: &str,
136    overwrite: bool,
137    history: Option<String>,
138) -> PyResult<AttachedImage> {
139    let (stages, output) = read_locked(input, |handle| {
140        let ImageHandle::Float(storage) = handle else {
141            return Err(raised("This method only supports float-valued images"));
142        };
143        let mut names = moment_output_names(
144            options.moments,
145            outfile,
146            handle.path().and_then(Path::to_str),
147        )
148        .map_err(raised)?;
149        let temporary = names[0].is_empty();
150        names.push(smoothout.to_owned());
151        let destinations = names
152            .iter()
153            .filter(|name| !name.is_empty())
154            .map(Path::new)
155            .collect::<Vec<_>>();
156        for destination in &destinations {
157            for source in handle
158                .path()
159                .into_iter()
160                .chain(input.operand_paths().iter().map(|path| path.as_path()))
161            {
162                ensure_outfile_does_not_alias_input(destination, source).map_err(raised)?;
163            }
164            if let Some(region) = options.region.and_then(SubimageRegion::external_path) {
165                ensure_artifacts_do_not_overlap(destination, region).map_err(raised)?;
166            }
167        }
168        let mut stages = NamedImageStages::new(names.iter().map(String::as_str), overwrite)?;
169        let mut products = storage.with_float_view(|image| {
170            let mut beams = image.image_info().map_err(raised)?.to_record();
171            input.beam_view().apply_to(&mut beams);
172            let options = ImageMomentOptions {
173                destinations: &destinations,
174                ..options
175            };
176            make_image_moments(image, handle.coordinate_record(), Some(&beams), options)
177                .map_err(raised)
178        })?;
179        let entries = match storage {
180            Storage::Paged(image) => {
181                image_history_rows(image.name().expect("persistent image")).map_err(raised)?
182            }
183            Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
184        };
185        let command = history.map(|details| {
186            [
187                concat!("version: casa-rs ", env!("CARGO_PKG_VERSION")).to_owned(),
188                details,
189            ]
190        });
191        let paths = stages.paths();
192        for (path, image) in paths.iter().zip(&mut products.moments) {
193            image.append_history_entries(&entries).map_err(raised)?;
194            if let Some(command) = &command {
195                image
196                    .append_history_at("image::moments", command)
197                    .map_err(raised)?;
198            }
199            if let Some(path) = path {
200                image.save_as(path).map_err(raised)?;
201            }
202        }
203        if let (Some(path), Some(image)) = (paths[products.moments.len()], &products.smoothed) {
204            image.save_as(path).map_err(raised)?;
205        }
206        if products.smoothed.is_none() {
207            stages.omit(products.moments.len());
208        }
209        let output = temporary.then(|| products.moments.remove(0));
210        Ok((stages, output))
211    })?;
212    let primary = stages.publish_and_open_primary()?;
213    match (primary, output) {
214        (None, Some(image)) => {
215            let record = image.coordinates().to_record();
216            Ok(AttachedImage::temporary(ImageHandle::Float(
217                Storage::Temp {
218                    image: Box::new(image),
219                    fits: None,
220                    record,
221                },
222            )))
223        }
224        (Some(image), None) => Ok(image),
225        _ => Err(raised("Expected exactly one primary moment image")),
226    }
227}
228
229fn python_bool(value: bool) -> &'static str {
230    if value { "True" } else { "False" }
