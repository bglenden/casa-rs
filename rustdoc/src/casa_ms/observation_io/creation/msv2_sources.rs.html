2//! The `SOURCE` rows of an MSv2 a plan creates.
3//!
4//! A catalogue that states its sources has each logical source's records
5//! written in stored order under one `SOURCE_ID` (its catalogue index),
6//! each in its window (`SPECTRAL_WINDOW_ID` -1 for every window), with
7//! `TRANSITION`, `REST_FREQUENCY`, `SYSVEL`, `POSITION` and `PULSAR_ID`
8//! when some record states them; a record's unstated line array is an
9//! undefined cell. A catalogue that states none (an MSv4's, until #976) has
10//! one row per source its fields name and per window, at the first such
11//! field's direction, as CASA's fillers write them and XRADIO's converter
12//! looks them up, which is noted as defaulted.
13

30/// Seconds per day, for MSv2 `TIME` (MJD seconds).
31const SECONDS_PER_DAY: f64 = 86_400.0;
32
33/// The sources a catalogue that states none has its fields name: each
34/// name once, at its first field's direction ([`source_direction`]).
35pub(super) struct NamedSources(Vec<(String, [f64; 2])>);
36
37/// The sources `catalogue`'s fields name, when it states no sources of
38/// its own, each at its first field's direction at that field's time
39/// ([`field_time`] from `times`).
40///
41/// # Errors
42///
43/// A field epoch in another scale than UTC, or a body's direction that
44/// cannot be evaluated.
45pub(super) fn named(
46    catalogue: &Catalogue,
47    times: (f64, &[Option<f64>]),
48) -> Result<NamedSources, ObservationError> {
49    let mut named: Vec<(String, [f64; 2])> = Vec::new();
50    if catalogue.sources.is_empty() {
51        for (index, field) in catalogue.fields.iter().enumerate() {
52            if let Some(name) = &field.source_name
53                && !named.iter().any(|known| known.0 == *name)
54            {
55                let (time, _) = field_time(field, index, times)?;
56                named.push((name.clone(), source_direction(field, time)?));
57            }
58        }
59    }
60    Ok(NamedSources(named))
61}

63/// A source's `DIRECTION`, from its first field (at `field_time`, MJD
64/// seconds): a fixed field's phase centre, or the body a moving field
65/// follows at that time (clamped to its samples), in J2000 as the column
66/// states it. The source is the body; its fields are offsets from it.
67fn source_direction(field: &FieldEntry, field_time: f64) -> Result<[f64; 2], ObservationError> {
68    let Some(ephemeris) = field.ephemeris.as_deref() else {
69        let (longitude, latitude) = field.phase_centre.as_angles();
70        return Ok([longitude, latitude]);
71    };
72    let times = ephemeris.times_mjd();
73    let mjd = (field_time / SECONDS_PER_DAY).clamp(times[0], times[times.len() - 1]);
74    let body = crate::ephemeris::reference_direction(ephemeris, mjd, [0.0, 0.0])
75        .and_then(|body| {
76            body.convert_to(DirectionRef::J2000, &MeasFrame::new())
77                .map_err(Into::into)
78        })
79        .map_err(storage)?;
80    let (longitude, latitude) = body.as_angles();
81    Ok([longitude, latitude])
82}

84/// The `SOURCE_ID` of `field`: its logical source's, or in a catalogue
85/// that states no sources the source its name names; -1 for none.
86pub(super) fn field_source(catalogue: &Catalogue, field: &FieldEntry, named: &NamedSources) -> i32 {
87    if let Some(source) = field.source {
88        return source.0 as i32;
89    }
90    if !catalogue.sources.is_empty() {
91        return -1;
92    }
93    field
94        .source_name
95        .as_ref()
96        .and_then(|name| named.0.iter().position(|known| known.0 == *name))
97        .map_or(-1, |id| id as i32)
98}

100/// The `SOURCE` rows of `catalogue`, each window's by its row in
101/// `window_rows` (`None` for a window not written, whose records are a
102/// loss), over data spanning `time_range` (MJD seconds).
103pub(super) fn write(
104    ms: &mut MeasurementSet,
105    catalogue: &Catalogue,
106    (named, window_rows): (&NamedSources, &[Option<i32>]),
107    time_range: [f64; 2],
108    notes: &mut Notes,
109) -> Result<(), ObservationError> {
110    if catalogue.sources.is_empty() {
111        return from_fields(ms, named, window_rows, time_range, notes);
112    }
113    let table = subtable_mut(ms, SubtableId::Source).map_err(storage)?;
114    let columns = Columns::add(table, catalogue)?;
115    let mut dropped = 0;
116    for (id, source) in catalogue.sources.iter().enumerate() {
117        for record in &source.records {
118            let window = match record.window {
119                SourceWindow::All => -1,
120                SourceWindow::Window(window) => {
121                    match window_rows.get(window.0 as usize).copied().flatten() {
122                        Some(row) => row,
123                        None => {
124                            dropped += 1;
125                            continue;
126                        }
127                    }
128                }
129            };
130            let mut row = row_from_defs(
131                schema::source::REQUIRED_COLUMNS,
132                &required((id as i32, &source.name, window), record, time_range, notes)?,
133            );
134            columns.optional(&mut row, record, notes)?;
135            table.add_row(row).map_err(storage)?;
136        }
137    }
138    if dropped > 0 {
139        notes.lost(Loss::Table(format!(
140            "SOURCE records of spectral windows not written ({dropped})"
141        )));
142    }
143    columns.describe(table)
144}

146/// A record's required columns: its source's `id` and `name`, the row of
147/// its `window`, and its own members or MSv2's defaults.
148fn required(
149    (id, name, window): (i32, &str, i32),
150    record: &SourceRecord,
151    time_range: [f64; 2],
152    notes: &mut Notes,
153) -> Result<Vec<(&'static str, casa_types::Value)>, ObservationError> {
154    let time = record.time.as_ref().map(utc_seconds).transpose()?;
155    let direction = record.direction.as_ref().map(|direction| {
156        let (longitude, latitude) = direction.as_angles();
157        [longitude, latitude]
158    });
159    Ok(vec![
160        (
161            "CALIBRATION_GROUP",
162            i(notes.or(record.calibration_group, "SOURCE CALIBRATION_GROUP", || -1)),
163        ),
164        (
165            "CODE",
166            s(&notes.or(record.code.clone(), "SOURCE CODE", String::new)),
167        ),
168        (
169            "DIRECTION",
170            f64_array(
171                &notes.or(direction, "SOURCE DIRECTION", || [0.0, 0.0]),
172                vec![2],
173            ),
174        ),
175        (
176            "INTERVAL",
177            f(notes.or(record.interval_s, "SOURCE INTERVAL", || {
178                time_range[1] - time_range[0]
179            })),
180        ),
181        ("NAME", s(name)),
182        (
183            "NUM_LINES",
184            i(i32::try_from(record.num_lines).map_err(|_| request_error("too many lines"))?),
185        ),
186        (
187            "PROPER_MOTION",
188            f64_array(
189                &notes.or(record.proper_motion_rad_s, "SOURCE PROPER_MOTION", || {
190                    [0.0, 0.0]
191                }),
192                vec![2],
193            ),
194        ),
195        ("SOURCE_ID", i(id)),
196        ("SPECTRAL_WINDOW_ID", i(window)),
197        (
198            "TIME",
199            f(notes.or(time, "SOURCE TIME", || {
200                0.5 * (time_range[0] + time_range[1])
201            })),
202        ),
203    ])
204}

206/// The optional `SOURCE` columns some record of the catalogue states, and
207/// the one frame each measure column's records share.
208struct Columns {
209    direction: DirectionRef,
210    transitions: bool,
211    rest: Option<String>,
212    /// `SYSVEL`, and its frame when stated.
213    velocities: Option<Option<String>>,
214    positions: bool,
215    pulsars: bool,
216}
217
218impl Columns {
219    /// Add to `table` the optional columns `catalogue`'s records state,
220    /// refusing a measure column whose records are in several frames.
221    fn add(table: &mut Table, catalogue: &Catalogue) -> Result<Self, ObservationError> {
222        let records: Vec<&SourceRecord> = catalogue
223            .sources
224            .iter()
225            .flat_map(|source| &source.records)
226            .collect();
227        let direction = one_frame(
228            records
229                .iter()
230                .filter_map(|record| record.direction.as_ref().map(|d| d.refer())),
231            "writing MSv2 source directions in several frames",
232        )?
233        .unwrap_or(DirectionRef::J2000);
234        let rest = one_frame(
235            records
236                .iter()
237                .filter_map(|record| record.rest_frequencies.as_ref())
238                .flatten()
239                .map(|frequency| frequency.refer().as_str().to_string()),
240            "writing MSv2 rest frequencies in several frames",
241        )?;
242        let any_rest = records
243            .iter()
244            .any(|record| record.rest_frequencies.is_some());
245        let velocity_frame = one_frame(
246            records
247                .iter()
248                .filter_map(|record| record.systemic_velocities.as_ref())
249                .flatten()
250                .map(|velocity| velocity.frame.map(|frame| frame.as_str().to_string())),
251            "writing MSv2 systemic velocities in several frames, or some unstated",
252        )?;
253        let any_velocity = records
254            .iter()
255            .any(|record| record.systemic_velocities.is_some());
256        let columns = Self {
257            direction,
258            transitions: records.iter().any(|record| record.transitions.is_some()),
259            rest: rest.or_else(|| any_rest.then(|| "LSRK".to_string())),
260            velocities: any_velocity.then(|| velocity_frame.flatten()),
261            positions: records.iter().any(|record| record.position.is_some()),
262            pulsars: records.iter().any(|record| record.pulsar_id.is_some()),
263        };
264        // In casacore's order (MSSourceEnums), as CASA's writers add them.
265        for (name, wanted) in [
266            ("POSITION", columns.positions),
267            ("PULSAR_ID", columns.pulsars),
268            ("REST_FREQUENCY", columns.rest.is_some()),
269            ("SYSVEL", columns.velocities.is_some()),
270            ("TRANSITION", columns.transitions),
271        ] {
272            if !wanted {
273                continue;
274            }
275            let definition = optional_columns(SubtableId::Source)
276                .iter()
277                .find(|definition| definition.name == name)
278                .expect("an optional SOURCE column");
279            // A scalar or fixed-shape column is added with the default every
280            // cell then has until its row is written: MSv2's.
281            let (schema, default) = match name {
282                "PULSAR_ID" => (
283                    ColumnSchema::scalar(name, definition.data_type),
284                    Some(i(-1)),
285                ),
286                "POSITION" => (
287                    ColumnSchema::array_fixed(name, definition.data_type, vec![3]),
288                    Some(f64_array(&[0.0; 3], vec![3])),
289                ),
290                _ => (
291                    ColumnSchema::array_variable(name, definition.data_type, Some(1)),
292                    None,
293                ),
294            };
295            table.add_column(schema, default).map_err(storage)?;
296        }
297        Ok(columns)
298    }
299
300    /// Add `record`'s optional cells to `row`: a stated line array or
301    /// position, and its pulsar, MSv2's default when the column is written
302    /// but the record states none.
303    fn optional(
304        &self,
305        row: &mut casa_types::RecordValue,
306        record: &SourceRecord,
307        notes: &mut Notes,
308    ) -> Result<(), ObservationError> {
309        let lines = record.num_lines as usize;
310        let length = |count: usize, column: &str| {
311            if count == lines {
312                Ok(())
313            } else {
314                Err(request_error(format!(
315                    "a SOURCE record of {lines} lines states {count} in {column}"
316                )))
317            }
318        };
319        if let Some(transitions) = &record.transitions {
320            length(transitions.len(), "TRANSITION")?;
321            let names: Vec<&str> = transitions.iter().map(String::as_str).collect();
322            row.push(RecordField::new(
323                "TRANSITION",
324                string_array(&names, vec![lines]),
325            ));
326        }
327        if let Some(frequencies) = &record.rest_frequencies {
328            length(frequencies.len(), "REST_FREQUENCY")?;
329            let hz: Vec<f64> = frequencies.iter().map(|frequency| frequency.hz()).collect();
330            row.push(RecordField::new(
331                "REST_FREQUENCY",
332                f64_array(&hz, vec![lines]),
333            ));
334        }
335        if let Some(velocities) = &record.systemic_velocities {
336            length(velocities.len(), "SYSVEL")?;
337            let values: Vec<f64> = velocities
338                .iter()
339                .map(|velocity| velocity.value_m_s)
340                .collect();
341            row.push(RecordField::new("SYSVEL", f64_array(&values, vec![lines])));
342        }
343        if self.positions {
344            let position = record.position.as_ref().map(|position| position.as_itrf());
345            let position = notes.or(position, "SOURCE POSITION", || [0.0; 3]);
346            row.push(RecordField::new("POSITION", f64_array(&position, vec![3])));
347        }
348        if self.pulsars {
349            let pulsar = notes.or(record.pulsar_id, "SOURCE PULSAR_ID", || -1);
350            row.push(RecordField::new("PULSAR_ID", i(pulsar)));
351        }
352        Ok(())
353    }
354
355    /// Write the measure and unit descriptions of the columns written.
356    fn describe(&self, table: &mut Table) -> Result<(), ObservationError> {
357        TableMeasDesc::new_fixed("DIRECTION", MeasureType::Direction, self.direction.as_str())
358            .write(table)
359            .map_err(storage)?;
360        if let Some(frame) = &self.rest {
361            TableMeasDesc::new_fixed("REST_FREQUENCY", MeasureType::Frequency, frame)
362                .write(table)
363                .map_err(storage)?;
364            TableQuantumDesc::with_unit("REST_FREQUENCY", "Hz")
365                .write(table)
366                .map_err(storage)?;
367        }
368        if let Some(frame) = &self.velocities {
369            if let Some(frame) = frame {
370                TableMeasDesc::new_fixed("SYSVEL", MeasureType::RadialVelocity, frame)
371                    .write(table)
372                    .map_err(storage)?;
373            }
374            TableQuantumDesc::with_unit("SYSVEL", "m/s")
375                .write(table)
376                .map_err(storage)?;
377        }
378        if self.positions {
379            TableMeasDesc::new_fixed("POSITION", MeasureType::Position, "ITRF")
380                .write(table)
381                .map_err(storage)?;
382            TableQuantumDesc::with_unit("POSITION", "m")
383                .write(table)
384                .map_err(storage)?;
385        }
386        Ok(())
387    }
388}

390/// The one value of `frames`, `None` when there is none; several are
391/// refused as `what`.
392fn one_frame<T: PartialEq>(
393    frames: impl Iterator<Item = T>,
394    what: &'static str,
395) -> Result<Option<T>, ObservationError> {
396    let mut one = None;
397    for frame in frames {
398        match &one {
399            None => one = Some(frame),
400            Some(known) if *known == frame => {}
401            Some(_) => return Err(ObservationError::Unsupported { what }),
402        }
403    }
404    Ok(one)
405}

407/// One `SOURCE` row per source the fields name and per window written, at
408/// the first such field's direction: what a catalogue without sources says
409/// of them.
410fn from_fields(
411    ms: &mut MeasurementSet,
412    named: &NamedSources,
413    window_rows: &[Option<i32>],
414    time_range: [f64; 2],
415    notes: &mut Notes,
416) -> Result<(), ObservationError> {
417    if named.0.is_empty() {
418        return Ok(());
419    }
420    notes.defaulted("SOURCE (rows written from the fields' source names)");
421    let table = subtable_mut(ms, SubtableId::Source).map_err(storage)?;
422    for (id, (name, direction)) in named.0.iter().enumerate() {
423        for window in window_rows.iter().flatten() {
424            let row = row_from_defs(
425                schema::source::REQUIRED_COLUMNS,
426                &[
427                    ("CALIBRATION_GROUP", i(-1)),
428                    ("CODE", s("")),
429                    ("DIRECTION", f64_array(direction, vec![2])),
430                    ("INTERVAL", f(time_range[1] - time_range[0])),
431                    ("NAME", s(name)),
432                    ("NUM_LINES", i(0)),
433                    ("PROPER_MOTION", f64_array(&[0.0, 0.0], vec![2])),
434                    ("SOURCE_ID", i(id as i32)),
435                    ("SPECTRAL_WINDOW_ID", i(*window)),
436                    ("TIME", f(0.5 * (time_range[0] + time_range[1]))),
437                ],
438            );
439            table.add_row(row).map_err(storage)?;
440        }
441    }
442    Ok(())
443}

449    use casa_observation::{FieldEntry, FieldEphemeris};

455    /// A moving field's source is the body it follows at the field's time
456    /// (clamped to the samples), not its offset; a fixed field's is its
457    /// phase centre.
458    #[test]
459    fn a_moving_fields_source_is_its_body_at_the_fields_time() {
460        let times = vec![60_000.0, 60_000.01, 60_000.02];
461        let directions = vec![[1.0, 0.2], [1.01, 0.21], [1.02, 0.22]];
462        let ephemeris = FieldEphemeris::new(
463            "comet",
464            times.clone(),
465            directions.clone(),
466            DirectionRef::J2000,
467            vec![3.8e8; 3],
468            (vec![10.0; 3], RadialVelocityRef::GEO),
469        )
470        .expect("an ephemeris");
471        let offset = MDirection::from_angles(1.0e-5, -5.0e-6, DirectionRef::J2000);
472        let field = FieldEntry {
473            name: "comet".to_string(),
474            phase_centre: offset.clone(),
475            reference_epoch: None,
476            source_name: Some("comet".to_string()),
477            source: None,
478            code: None,
479            delay_centre: None,
480            reference_centre: None,
481            flagged: None,
482            ephemeris: Some(Arc::new(ephemeris)),
483        };
484        let close = |found: [f64; 2], expected: [f64; 2]| {
485            assert!(
486                (found[0] - expected[0]).abs() < 1.0e-12
487                    && (found[1] - expected[1]).abs() < 1.0e-12,
488                "{found:?} is not {expected:?}"
489            );
490        };
491        close(
492            source_direction(&field, times[1] * SECONDS_PER_DAY).expect("a direction"),
493            directions[1],
494        );
495        close(
496            source_direction(&field, (times[2] + 1.0) * SECONDS_PER_DAY).expect("clamped"),
497            directions[2],
498        );
499        let fixed = FieldEntry {
500            ephemeris: None,
501            ..field
502        };
503        let (longitude, latitude) = offset.as_angles();
504        close(
505            source_direction(&fixed, 0.0).expect("a direction"),
506            [longitude, latitude],
507        );
508    }
