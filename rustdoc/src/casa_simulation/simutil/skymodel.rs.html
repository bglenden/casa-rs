2//! The sky model `simobserve` observes (simulation Phase 5b, #841): CASA
3//! 6.7.6's `simutil.modifymodel` and `simutil.flatimage`, `ia.modify`
4//! adding a component list to an image, and the component-only model
5//! `simobserve` makes when it has no image.
6//!
7//! | CASA | casa-rs |
8//! |---|---|
9//! | `simutil.modifymodel` | [`modify_model`] |
10//! | `simutil.flatimage` | [`flat_image`] |
11//! | `ia.modify(cl.torecord(), subtract=False)` (`ComponentListImage`) | [`add_components`] |
12//! | `simobserve`'s `.compskymodel` | [`ComponentModel`] |
13//!
14//! The images written are what CASA writes: their shape, brightness unit,
15//! coordinate values, Stokes and spectral axes, and pixels. Their
16//! coordinate systems carry CASA's `ia.fromshape` observation information
17//! (ALMA, "Karl Jansky", 2000-01-01), and the spectral axis is LSRK with no
18//! conversion layer. The tests compare a grid of CASA's outputs
19//! (`tests/data/skymodel_casa.json.gz`, from
20//! `tools/simulation/skymodel_oracle.py`): strings exactly, coordinate values
21//! to 1e-12, pixels to 1e-6 of the peak.
22//!
23//! **Where casa-rs differs**, each recorded in `docs/CASA (C++) bugs.md`:
24//! - **Axis order.** `modifymodel` reorders the axes by swapping each into
25//!   place, which leaves a cyclic order (spectral axis first) in the wrong
26//!   order, and CASA then raises "internal error: I'm confused about the
27//!   shape if your model data cube"; casa-rs permutes the axes.
28//! - **Stokes labels.** CASA's new image takes its Stokes axis from
29//!   `ia.fromshape`, which labels planes by their count (I, IQ, IQU, IQUV):
30//!   an I,V model comes out labelled I,Q, and three unlabelled planes come
31//!   out IQU though `modifymodel` reports and warns IQV. casa-rs keeps the
32//!   model's labels, and labels unlabelled planes as it reports them.
33//! - **A cell per axis.** `modifymodel` takes `incell` as a list of the two
34//!   axes' cells, but under Python 3 indexes the `map` it makes of them and
35//!   raises `TypeError`; casa-rs takes the two cells, as CASA's Python 2
36//!   code did.
37//! - **An unset `compwidth`.** The component-only model's spectral
38//!   increment is `compwidth` as given; with none, CASA's
39//!   `coordsys.setincrement` puts a meaningless value there (the model's
40//!   right ascension, in radians, as hertz). casa-rs uses the 2 GHz
41//!   `simobserve` observes with.
42//! - **Frequency frames without a usable frame.** A component whose
43//!   spectrum is in another frame than the image's is converted into the
44//!   image's frame with the image's conversion-layer frame, as CASA converts
45//!   it. When the image has none, or the placeholder `ia.fromshape` leaves
46//!   (an epoch of 0 at the geocentre), CASA converts through the
47//!   placeholder; casa-rs leaves the frequencies unconverted
48//!   ([`add_components`]).

73/// What the sky-model functions refuse.
74#[derive(Debug, Error)]
75pub enum SkyModelError {
76    /// A refusal CASA makes with its own message (`simutil.msg` at error
77    /// priority, which raises).
78    #[error("{0}")]
79    Model(String),
80    /// More than one `indirection`, for which CASA logs the message and
81    /// `modifymodel` returns `False`.
82    #[error("error parsing indirection {0} -- should be a single direction string")]
83    Indirections(String),
84    /// A quantity or direction the `simutil` helpers refuse.
85    #[error(transparent)]
86    Simutil(#[from] SimutilError),
87    /// An image that cannot be read or written.
88    #[error(transparent)]
89    Image(#[from] ImageError),
90    /// A coordinate system casa-coordinates refuses.
91    #[error(transparent)]
92    Coordinates(#[from] CoordinateError),
93    /// A component casa-components does not sample.
94    #[error(transparent)]
95    Component(#[from] ComponentError),
96    /// A component list that cannot be read.
97    #[error(transparent)]
98    ComponentList(#[from] ComponentListError),
99    /// A file-system operation that failed.
100    #[error("{path}: {source}")]
101    Io {
102        /// The path.
103        path: PathBuf,
104        /// The failure.
105        source: std::io::Error,
106    },
107}

109/// An image as `ia.newimagefromfile` opens it and `ia.getchunk` reads it:
110/// its pixels in double precision and its coordinate system.
111pub(crate) struct InputImage {
112    pub(crate) pixels: ArrayD<f64>,
113    pub(crate) coordinates: CoordinateSystem,
114}
115
116impl InputImage {
117    /// Read the CASA image (a directory) or FITS file (a file) at `path`.
118    pub(crate) fn read(path: &Path) -> Result<Self, SkyModelError> {
119        if path.is_file() {
120            let fits = casa_images::read_fits_image(path)?;
121            return Ok(Self {
122                pixels: fits.pixels.mapv(f64::from),
123                coordinates: fits.coordinates,
124            });
125        }
126        match AnyPagedImage::open(path)? {
127            AnyPagedImage::Float32(image) => Ok(Self {
128                pixels: image.get()?.mapv(f64::from),
129                coordinates: image.coordinates().clone(),
130            }),
131            AnyPagedImage::Float64(image) => Ok(Self {
132                pixels: image.get()?,
133                coordinates: image.coordinates().clone(),
134            }),
135            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => Err(SkyModelError::Model(
136                format!("{}: a complex image is not a sky model", path.display()),
137            )),
138        }
139    }
140}

142/// The coordinate and pixel axes of the first coordinate of `kind`.
143pub(crate) fn pixel_axes(
144    coordinates: &CoordinateSystem,
145    kind: CoordinateType,
146) -> Option<(usize, Vec<usize>)> {
147    let mut first = 0;
148    for index in 0..coordinates.n_coordinates() {
149        let coordinate = coordinates.coordinate(index);
150        let count = coordinate.n_pixel_axes();
151        if coordinate.coordinate_type() == kind {
152            return Some((index, (first..first + count).collect()));
153        }
154        first += count;
155    }
156    None
157}

159/// The observation information of an image CASA makes from a shape
160/// (`CoordinateUtil::makeCoordinateSystem`): ALMA, observed by "Karl
161/// Jansky" at 2000-01-01T00:00:00.0001 UTC.
162pub(crate) fn shape_obs_info() -> ObsInfo {
163    ObsInfo::new("ALMA")
164        .with_observer("Karl Jansky")
165        .with_date(MEpoch::from_mjd(
166            51_544.0 + 1.0e-4 / 86_400.0,
167            EpochRef::UTC,
168        ))

171/// `projection` as casacore holds it: a SIN projection given no parameters
172/// has its two, zero (`Projection(Projection::SIN)`).
173pub(crate) fn casacore_projection(projection: &Projection) -> Projection {
174    if projection.projection_type() == ProjectionType::SIN && projection.parameters().is_empty() {
175        Projection::with_parameters(ProjectionType::SIN, vec![0.0, 0.0])
176    } else {
177        projection.clone()
178    }
179}

181/// Remove whatever is at `path`, as `ia.fromshape(..., overwrite=True)`
182/// replaces it.
183pub(crate) fn remove_existing(path: &Path) -> Result<(), SkyModelError> {
184    let removed = if path.is_dir() {
185        std::fs::remove_dir_all(path)
186    } else if path.exists() {
187        std::fs::remove_file(path)
188    } else {
189        return Ok(());
190    };
191    removed.map_err(|source| SkyModelError::Io {
192        path: path.to_path_buf(),
193        source,
194    })
195}

197/// Write `pixels` as a new single-precision CASA image at `path`, replacing
198/// what is there.
199pub(crate) fn write_image(
200    path: &Path,
201    pixels: &ArrayD<f32>,
202    coordinates: CoordinateSystem,
203    units: &str,
204) -> Result<(), SkyModelError> {
205    remove_existing(path)?;
206    let shape = pixels.shape().to_vec();
207    let mut image = PagedImage::<f32>::create(shape.clone(), coordinates, path)?;
208    image.put_slice(pixels, &vec![0; shape.len()])?;
209    image.set_units(units)?;
210    image.save()?;
211    Ok(())
212}
