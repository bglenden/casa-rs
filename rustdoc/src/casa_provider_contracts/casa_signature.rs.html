2//! The rules for a surface that mimics a CASA task (#931).
3//!
4//! A surface whose [`casa_task`](super::TaskDefinition::casa_task) names a
5//! CASA task presents that task's signature: CASA's parameters, in CASA's
6//! order and with CASA's literal defaults, then casa-rs-only controls as
7//! keyword-only arguments
8//! ([`keyword_only`](super::PythonParameterProjection::keyword_only)). A CASA parameter that
9//! must be given (CASA's `imagename=''`) keeps CASA's default and is checked
10//! when the task runs, through its `required_when`.
11//!
12//! A CASA parameter casa-rs does not implement yet is an
13//! [`UnimplementedParameter`]: it is accepted at its default, so CASA's
14//! signature is exact, and any other value is refused before the task
15//! starts, naming the issue that tracks it.
16//!
17//! Two checks hold this, and they guarantee different things. Contract
18//! validation here checks the rules' shape for the bindings a surface has:
19//! literal defaults, Python names, keyword-only extras last, markers only
20//! on CASA parameters. It does not know CASA's signature, so a deleted
21//! parameter or a changed but well-typed default passes it. Equality with
22//! CASA 6.7.6.14 itself, every parameter, its order, kind and default, is
23//! checked against the pinned inventory by the Python package's
24//! `test_task_signatures.py`, which also fails on a CASA-named task it
25//! neither checks nor lists as pending, and reported by
26//! `scripts/check-casa-inventory.py`.
27

38/// A CASA parameter casa-rs accepts only at CASA's default until `issue`
39/// implements it (#931).
40///
41/// "CASA's default" is the default as CASA reads it, which is this law:
42///
43/// - Where [`applies_when`](Self::applies_when) is false over the resolved
44///   values, CASA ignores the parameter (immath's `sigma` outside its
45///   polarization modes, any task's `stretch` without a `mask`), and so
46///   does casa-rs: every value is accepted and none is used.
47/// - Elsewhere a value is accepted when it equals the binding's literal
48///   default or one of the [`equivalent`](Self::equivalent) values CASA
49///   reads as that default (immath's `varnames=[]` for `''`). Equality is
50///   the concept's semantic equality: both sides are first normalized by
51///   the concept's rule (text trimmed or lowercased, a quantity in its
52///   canonical unit, a scalar as a one-item list), then compared exactly,
53///   except that numbers agree within a relative 1e-12, so `-1` and `-1.0`
54///   are one value.
55///
56/// Any other value is refused before the task starts.
57#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
58pub struct UnimplementedParameter {
59    /// The casa-rs GitHub issue that tracks implementing the parameter.
60    pub issue: u32,
61    /// Other values CASA reads exactly as the default, compared as the
62    /// default is.
63    #[serde(default, skip_serializing_if = "Vec::is_empty")]
64    pub equivalent: Vec<ParameterValue>,
65    /// Where CASA reads the parameter at all; elsewhere it ignores the
66    /// value, and casa-rs accepts any value without using it.
67    #[serde(default = "always", skip_serializing_if = "is_always")]
68    pub applies_when: Predicate,
69}
70
71fn always() -> Predicate {
72    Predicate::Always
73}
74
75fn is_always(predicate: &Predicate) -> bool {
76    matches!(predicate, Predicate::Always)
77}
78
79impl UnimplementedParameter {
80    /// The marker for a parameter `issue` tracks, accepted at its default
81    /// alone wherever it is given.
82    pub fn new(issue: u32) -> Self {
83        Self {
84            issue,
85            equivalent: Vec::new(),
86            applies_when: Predicate::Always,
87        }
88    }
89
90    /// The refusal for `name` set to anything but its `default`.
91    pub fn refusal(&self, name: &str, default: &ParameterValue) -> String {
92        let equivalent = if self.equivalent.is_empty() {
93            String::new()
94        } else {
95            let values = self.equivalent.iter().map(python_repr).collect::<Vec<_>>();
96            format!(" (or {}, which CASA reads as it)", values.join(" or "))
97        };
98        format!(
99            "parameter {name:?} is not implemented by casa-rs yet (#{}); only CASA's default \
100             {}{equivalent} is accepted",
101            self.issue,
102            python_repr(default)
103        )
104    }
105}

107/// `value` as Python writes it, the spelling CASA's signatures show.
108pub fn python_repr(value: &ParameterValue) -> String {
109    match value {
110        ParameterValue::Bool(true) => "True".into(),
111        ParameterValue::Bool(false) => "False".into(),
112        ParameterValue::Integer(value) => value.to_string(),
113        ParameterValue::Float(value) => format!("{value:?}"),
114        ParameterValue::String(value) => {
115            format!("'{}'", value.replace('\\', "\\\\").replace('\'', "\\'"))
116        }
117        ParameterValue::Array(values) => format!(
118            "[{}]",
119            values
120                .iter()
121                .map(python_repr)
122                .collect::<Vec<_>>()
123                .join(", ")
124        ),
125        ParameterValue::Table(values) => format!(
126            "{{{}}}",
127            values
128                .iter()
129                .map(|(key, value)| format!("'{key}': {}", python_repr(value)))
130                .collect::<Vec<_>>()
131                .join(", ")
132        ),
133    }
134}

136/// Check the CASA-signature rules of `surface`.
137pub(super) fn validate_casa_signature(
138    surface: &SurfaceDefinition,
139    errors: &mut Vec<ContractValidationError>,
140) {
141    let mut bindings = surface.bindings().iter().collect::<Vec<_>>();
142    bindings.sort_by_key(|binding| binding.order);
143    let mimics_casa = surface.casa_task().is_some();
144    let mut first_keyword_only: Option<&str> = None;
145    for binding in bindings {
146        let error = |code: &str, message: String| {
147            ContractValidationError::new(code, message).at(surface.id(), Some(&binding.name))
148        };
149        let python = binding.projections.python.as_ref();
150        let keyword_only = python.is_some_and(|python| python.keyword_only);
151        if binding.unimplemented.is_some()
152            && !matches!(binding.default, DefaultSpec::Literal { .. })
153        {
154            errors.push(error(
155                "unimplemented_default",
156                "a parameter accepted only at its default needs a literal default".into(),
157            ));
158        }
159        if !mimics_casa {
160            if binding.unimplemented.is_some() || keyword_only {
161                errors.push(error(
162                    "casa_signature_only",
163                    "unimplemented and keyword-only parameters belong to a surface that \
164                     mimics a CASA task"
165                        .into(),
166                ));
167            }
168            continue;
169        }
170        if python.is_none() {
171            errors.push(error(
172                "casa_python_projection",
173                "every parameter of a CASA task's surface has a Python projection".into(),
174            ));
175        }
176        if keyword_only {
177            if binding.unimplemented.is_some() {
178                errors.push(error(
179                    "unimplemented_extra",
180                    "a keyword-only control is casa-rs's own; only a CASA parameter can be \
181                     unimplemented"
182                        .into(),
183                ));
184            }
185            first_keyword_only.get_or_insert(&binding.name);
186            continue;
187        }
188        if let Some(extra) = first_keyword_only {
189            errors.push(error(
190                "keyword_only_order",
191                format!(
192                    "CASA parameter {:?} follows the keyword-only {extra:?}",
193                    binding.name
194                ),
195            ));
196        }
197        if !matches!(binding.default, DefaultSpec::Literal { .. }) {
198            errors.push(error(
199                "casa_default",
200                "a CASA parameter takes CASA's literal default; one that must be given is \
201                 checked when the task runs"
202                    .into(),
203            ));
204        }
205    }
206}

208/// Check that every [`UnimplementedParameter`] of `bundle` gives equivalent
209/// defaults of its concept's type and an `applies_when` over the surface's
210/// parameters (`known`).
211pub(super) fn validate_unimplemented(
212    bundle: &SurfaceContractBundle,
213    known: &BTreeSet<String>,
214    errors: &mut Vec<ContractValidationError>,
215) {
216    let surface = bundle.surface.id();
217    for binding in bundle.surface.bindings() {
218        let Some(unimplemented) = &binding.unimplemented else {
219            continue;
220        };
221        super::validate_predicate_refs(
222            &unimplemented.applies_when,
223            known,
224            surface,
225            &binding.name,
226            errors,
227        );
228        let Some(concept) = bundle.catalog.concept(&binding.concept) else {
229            continue;
230        };
231        for value in &unimplemented.equivalent {
232            if let Err(reason) = super::validate_parameter_value(value, &concept.value_domain) {
233                errors.push(
234                    ContractValidationError::new(
235                        "unimplemented_equivalent",
236                        format!(
237                            "equivalent default {} is not a {} value: {reason}",
238                            python_repr(value),
239                            concept.id
240                        ),
241                    )
242                    .at(surface, Some(&binding.name)),
243                );
244            }
245        }
246    }
247}

255    fn binding(name: &str, order: usize, default: Value, extra: Value) -> Value {
256        let mut binding = json!({
257            "name": name,
258            "concept": {"id": "example.text", "semantic_revision": 1},
259            "order": order,
260            "default": default,
261            "required_when": {"kind": "never"},
262            "active_when": {"kind": "always"},
263            "projections": {
264                "python": {"name": name},
265                "presentation": {"label": name, "group": "General"}
266            }
267        });
268        for (key, value) in extra.as_object().unwrap() {
269            if key == "keyword_only" {
270                binding["projections"]["python"]["keyword_only"] = value.clone();
271            } else {
272                binding[key] = value.clone();
273            }
274        }
275        binding
276    }
