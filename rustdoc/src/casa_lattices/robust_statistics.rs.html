2//! Robust sample statistics: CASA's statistics-framework algorithms.
3//!
4//! C++ casacore implements `imstat`'s `hinges-fences`, `chauvenet`,
5//! `fit-half` and `biweight` algorithms in
6//! `casacore/scimath/StatsFramework`. This module follows those sources on
7//! a materialized sample of `(value, linear position)` pairs: every
8//! algorithm selects its own effective sample from the caller's, and the
9//! statistics reported are the classical ones over that sample.
10//!
11//! The quantile rule is the framework's: the quantile at fraction `q` of
12//! `n` sorted values is the value at index `ceil(q * n) - 1`, without
13//! interpolation; the median and the median absolute deviation average
14//! the two middle values of even samples. The biweight's initial scale
15//! is casacore's `C::probit_3_4` constant, `1.482602218505602` — despite
16//! its name, the reciprocal of the probit at 3/4, the factor that makes
17//! the median absolute deviation estimate a Gaussian sigma.
18

24/// The MAD-to-sigma factor casacore names `C::probit_3_4`.
25const PROBIT_3_4: f64 = 1.482602218505602;
26
27/// The biweight tuning constant CASA's statistics surface uses.
28const BIWEIGHT_C: f64 = 6.0;

30/// Which of CASA's statistics algorithms a computation ran.
31///
32/// It fixes the keys the CASA record carries: the biweight reports only
33/// its location and scale, and `fit-half` reports only the real
34/// extremum's position.
35#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
36pub enum StatisticsAlgorithmKind {
37    /// The classical moments and order statistics.
38    Classic,
39    /// Classical statistics over the hinges-and-fences range.
40    HingesFences,
41    /// Classical statistics over Chauvenet's converged range.
42    Chauvenet,
43    /// Classical statistics over the reflected half sample.
44    FitToHalf,
45    /// The biweight location and scale.
46    Biweight,
47}

49/// One of CASA's statistics algorithms, with its parameters.
50#[derive(Debug, Clone, Copy, PartialEq)]
51pub enum StatisticsAlgorithm {
52    /// The classical moments and order statistics.
53    Classic,
54    /// Data outside `[q1 - fence * d, q3 + fence * d]`, for the first
55    /// and third quartiles of the full sample and their difference `d`,
56    /// is excluded. A negative fence keeps every value, exactly as
57    /// `classic`.
58    HingesFences {
59        /// The fence multiplier; negative keeps the whole sample.
60        fence: f64,
61    },
62    /// Data is excluded iteratively beyond `zscore` standard deviations
63    /// of the current mean, or, when `zscore` is negative, beyond
64    /// Chauvenet criterion for the current sample size. `maxiter`
65    /// bounds the iterations; negative means until the excluded set
66    /// stops changing.
67    Chauvenet {
68        /// The fixed z-score, or a negative value for Chauvenet's
69        /// criterion.
70        zscore: f64,
71        /// The iteration bound; negative means until convergence.
72        maxiter: i64,
73    },
74    /// Only values at or below (or at or above) a center are kept, and
75    /// each is mirrored about the center so the doubled sample is
76    /// symmetric. The center is the full sample's mean or median, or
77    /// zero.
78    FitToHalf {
79        /// Which full-sample statistic the center is.
80        center: FitHalfCenter,
81        /// Whether the kept half is at or above the center, rather
82        /// than at or below it.
83        upper: bool,
84    },
85    /// The biweight location and scale, iterating at most `niter`
86    /// times; a negative count takes the single fast pass.
87    Biweight {
88        /// The iteration bound; negative takes the fast pass.
89        niter: i64,
90    },
91}
92
93impl StatisticsAlgorithm {
94    /// The kind of record this algorithm produces.
95    pub fn kind(&self) -> StatisticsAlgorithmKind {
96        match self {
97            Self::Classic => StatisticsAlgorithmKind::Classic,
98            Self::HingesFences { .. } => StatisticsAlgorithmKind::HingesFences,
99            Self::Chauvenet { .. } => StatisticsAlgorithmKind::Chauvenet,
100            Self::FitToHalf { .. } => StatisticsAlgorithmKind::FitToHalf,
101            Self::Biweight { .. } => StatisticsAlgorithmKind::Biweight,
102        }
103    }
104}

191/// Parse CASA's `algorithm` argument: minimum match, in any case, of
192/// `biweight`, `chauvenet`, `classic`, `fit-half` and `hinges-fences`.
193///
194/// An empty, ambiguous or unknown token fails with CASA's wording, as
195/// `c` does against `classic` and `chauvenet`.
196pub fn parse_algorithm(token: &str) -> Result<StatisticsAlgorithmKind, String> {
197    const NAMES: [(&str, StatisticsAlgorithmKind); 5] = [
198        ("biweight", StatisticsAlgorithmKind::Biweight),
199        ("chauvenet", StatisticsAlgorithmKind::Chauvenet),
200        ("classic", StatisticsAlgorithmKind::Classic),
201        ("fit-half", StatisticsAlgorithmKind::FitToHalf),
202        ("hinges-fences", StatisticsAlgorithmKind::HingesFences),
203    ];
204    let lowered = token.to_ascii_lowercase();
205    let matches: Vec<_> = NAMES
206        .iter()
207        .filter(|(name, _)| name.starts_with(&lowered))
208        .collect();
209    match matches.as_slice() {
210        [(_, kind)] => Ok(*kind),
211        _ => Err(format!("Unsupported algorithm {lowered}")),
212    }
213}

215/// Parse CASA's `center` argument for `fit-half`: minimum match, in any
216/// case, of `mean`, `median` and `zero`.
217///
218/// A minimum match must be at least three characters long for the mean
219/// and median, as CASA's parser requires; `zero` matches from its first
220/// letter.
221pub fn parse_center(token: &str) -> Result<FitHalfCenter, String> {
222    const NAMES: [(&str, FitHalfCenter, usize); 3] = [
223        ("mean", FitHalfCenter::Mean, 3),
224        ("median", FitHalfCenter::Median, 3),
225        ("zero", FitHalfCenter::Zero, 1),
226    ];
227    let lowered = token.to_ascii_lowercase();
228    let matches: Vec<_> = NAMES
229        .iter()
230        .filter(|(name, _, minimum)| lowered.len() >= *minimum && name.starts_with(&lowered))
231        .collect();
232    match matches.as_slice() {
233        [(_, center, _)] => Ok(*center),
234        _ => Err(format!("Unsupported center value {lowered}")),
235    }
236}

238/// Parse CASA's `clmethod` argument: minimum match, in any case, of
239/// `auto`, `tiled` and `framework`.
240///
241/// The choice is an execution strategy and the statistics it produces
242/// are the same, so the parsed value is accepted and not otherwise used.
243pub fn parse_clmethod(token: &str) -> Result<(), String> {
244    const NAMES: [&str; 3] = ["auto", "tiled", "framework"];
245    let lowered = token.to_ascii_lowercase();
246    if NAMES.iter().any(|name| name.starts_with(&lowered)) {
247        Ok(())
248    } else {
249        Err(format!("Unsupported classical method {lowered}"))
250    }
251}

253/// Compute one robust algorithm over the sample.
254///
255/// # Errors
256///
257/// The biweight with fewer than two points, or a `fit-half` whose center
258/// excludes every point, with CASA's wording.
259pub fn compute_robust_stats(
260    sample: &[SamplePoint],
261    algorithm: &StatisticsAlgorithm,
262) -> Result<RobustStats, RobustStatsError> {
263    match algorithm {
264        StatisticsAlgorithm::Classic => Ok(RobustStats::Sample(classical_stats(sample))),
265        StatisticsAlgorithm::HingesFences { fence } => {
266            if *fence < 0.0 {
267                return Ok(RobustStats::Sample(classical_stats(sample)));
268            }
269            let mut values: Vec<f64> = sample.iter().map(|point| point.value).collect();
270            let [q1, _, q3] = quartiles_and_median(&mut values);
271            let span = q3 - q1;
272            let (low, high) = (q1 - fence * span, q3 + fence * span);
273            let kept: Vec<SamplePoint> = sample
274                .iter()
275                .copied()
276                .filter(|point| low <= point.value && point.value <= high)
277                .collect();
278            Ok(RobustStats::Sample(classical_stats(&kept)))
279        }
280        StatisticsAlgorithm::Chauvenet { zscore, maxiter } => {
281            let max_iterations = if *maxiter < 0 { 1000 } else { *maxiter };
282            let mut kept: Vec<SamplePoint> = sample.to_vec();
283            let mut previous_npts: Option<f64> = None;
284            let mut reductions = 0;
285            for iteration in 0..=max_iterations {
286                let stats = classical_stats(&kept);
287                if iteration > 0 && previous_npts == Some(stats.npts) {
288                    break;
289                }
290                let z = if *zscore >= 0.0 {
291                    *zscore
292                } else {
293                    max_zscore(stats.npts as u64)
294                };
295                let center = if stats.npts == 0. { 0. } else { stats.mean };
296                let low = center - z * stats.sigma;
297                let high = center + z * stats.sigma;
298                // Casacore replaces the range on the original data at each
299                // iteration. An empty previous range has mean and sigma zero.
300                kept.clear();
301                kept.extend(
302                    sample
303                        .iter()
304                        .copied()
305                        .filter(|point| low <= point.value && point.value <= high),
306                );
307                if kept.len() < stats.npts as usize {
308                    reductions += 1;
309                }
310                previous_npts = Some(stats.npts);
311            }
312            let mut stats = classical_stats(&kept);
313            stats.chauvenet_iterations = Some(reductions.min(max_iterations as usize));
314            Ok(RobustStats::Sample(stats))
315        }
316        StatisticsAlgorithm::FitToHalf { center, upper } => {
317            let center_value = match center {
318                FitHalfCenter::Mean => sample_mean(sample),
319                FitHalfCenter::Median => sample_median(sample),
320                FitHalfCenter::Zero => 0.0,
321            };
322            let reals: Vec<&SamplePoint> = sample
323                .iter()
324                .filter(|point| {
325                    if *upper {
326                        point.value >= center_value
327                    } else {
328                        point.value <= center_value
329                    }
330                })
331                .collect();
332            if reals.is_empty() {
333                return Err(RobustStatsError::FitHalfEmpty);
334            }
335            // The effective sample is the kept values joined with their
336            // reflections about the center; only the real extremum has a
337            // position, and the median is the center itself.
338            let mut values: Vec<f64> = Vec::with_capacity(2 * reals.len());
339            for point in &reals {
340                values.push(point.value);
341                values.push(2.0 * center_value - point.value);
342            }
343            let mut stats = classical_stats_of_values(&values);
344            stats.median = center_value;
345            if *upper {
346                let real_max = reals
347                    .iter()
348                    .copied()
349                    .max_by(|left, right| {
350                        left.value
351                            .total_cmp(&right.value)
352                            .then(right.position.cmp(&left.position))
353                    })
354                    .expect("the kept sample is non-empty");
355                stats.max = real_max.value;
356                stats.maxpos = Some(real_max.position);
357                stats.min = 2.0 * center_value - real_max.value;
358            } else {
359                let real_min = reals
360                    .iter()
361                    .copied()
362                    .min_by(|left, right| {
363                        left.value
364                            .total_cmp(&right.value)
365                            .then(left.position.cmp(&right.position))
366                    })
367                    .expect("the kept sample is non-empty");
368                stats.min = real_min.value;
369                stats.minpos = Some(real_min.position);
370                stats.max = 2.0 * center_value - real_min.value;
371            }
372            Ok(RobustStats::Sample(stats))
373        }
374        StatisticsAlgorithm::Biweight { niter } => {
375            let values: Vec<f64> = sample.iter().map(|point| point.value).collect();
376            let (location, scale) = biweight_location_scale(&values, BIWEIGHT_C, *niter)?;
377            Ok(RobustStats::Biweight { location, scale })
378        }
379    }
380}

382/// The classical statistics of a sample: the framework's moments and
383/// order statistics, with the extremum positions of the first extremum
384/// in sample order.
385fn classical_stats(sample: &[SamplePoint]) -> RobustSampleStats {
386    let mut stats =
387        classical_stats_of_values(&sample.iter().map(|point| point.value).collect::<Vec<_>>());
388    let mut minpos = None;
389    let mut maxpos = None;
390    for point in sample {
391        if minpos.is_none() || point.value < stats.min {
392            stats.min = point.value;
393            minpos = Some(point.position);
394        }
395        if maxpos.is_none() || point.value > stats.max {
396            stats.max = point.value;
397            maxpos = Some(point.position);
398        }
399    }
400    stats.minpos = minpos;
401    stats.maxpos = maxpos;
402    stats
403}

405/// The classical statistics of a bare value sample, without positions.
406fn classical_stats_of_values(values: &[f64]) -> RobustSampleStats {
407    let npts = values.len();
408    let mut moments = RunningStats::default();
409    let first = values.first().copied().unwrap_or(f64::NAN);
410    let (mut min, mut max) = (first, first);
411    for &value in values {
412        moments.push(value);
413        if value.total_cmp(&min).is_lt() {
414            min = value;
415        }
416        if value.total_cmp(&max).is_gt() {
417            max = value;
418        }
419    }
420    let [sum, sumsq, mean, rms, sigma] = if npts == 0 {
421        [0.0, 0.0, f64::NAN, f64::NAN, 0.0]
422    } else {
423        [
424            Statistic::Sum,
425            Statistic::SumSq,
426            Statistic::Mean,
427            Statistic::Rms,
428            Statistic::Sigma,
429        ]
430        .map(|statistic| moments.finish(statistic))
431    };
432    let OrderStatsSummary {
433        median,
434        q1,
435        q3,
436        quartile,
437        mad: medabsdevmed,
438    } = framework_order_summary(&mut values.to_vec());
439    RobustSampleStats {
440        npts: npts as f64,
441        min,
442        max,
443        sum,
444        sumsq,
445        mean,
446        rms,
447        sigma,
448        median,
449        medabsdevmed,
450        q1,
451        q3,
452        quartile,
453        minpos: None,
454        maxpos: None,
455        chauvenet_iterations: None,
456    }
457}

471/// The biweight location and scale, as casacore's
472/// `BiweightStatistics` computes them.
473///
474/// The location starts at the sample median and the scale at
475/// `probit_3_4 * MAD`. With a non-negative iteration count the scale is
476/// refined once from the initial weights, and then location-and-scale
477/// pairs iterate until the scale's relative change falls below
478/// `0.03 / sqrt(2 * (npts - 1))` or the count runs out. With a negative
479/// count both are taken in one pass over the initial weights. Only
480/// values strictly inside `c * scale` of the location contribute.
481fn biweight_location_scale(
482    values: &[f64],
483    c: f64,
484    max_niter: i64,
485) -> Result<(f64, f64), RobustStatsError> {
486    let npts = values.len() as u64;
487    if npts <= 1 {
488        return Err(RobustStatsError::BiweightNeedsTwoPoints { npts });
489    }
490    let mut workspace = values.to_vec();
491    let mut location = selection_median(&mut workspace);
492    for (deviation, value) in workspace.iter_mut().zip(values) {
493        *deviation = (value - location).abs();
494    }
495    let mut scale = PROBIT_3_4 * selection_median(&mut workspace);
496    drop(workspace);
497
498    let mut spread = c * scale;
499    let (mut low, mut high) = (location - spread, location + spread);
500    let location_sums = |location: f64, scale: f64, low: f64, high: f64| -> f64 {
501        let (mut sxw2, mut sw2) = (0.0, 0.0);
502        for &value in values {
503            if low < value && value < high {
504                let u = (value - location) / (c * scale);
505                let w = 1.0 - u * u;
506                sxw2 += value * w * w;
507                sw2 += w * w;
508            }
509        }
510        sxw2 / sw2
511    };
512    let scale_sums = |location: f64, scale: f64, low: f64, high: f64| -> f64 {
513        let (mut sx_m2w4, mut ww_4u2) = (0.0, 0.0);
514        for &value in values {
515            if low < value && value < high {
516                let u = (value - location) / (c * scale);
517                let w = 1.0 - u * u;
518                sx_m2w4 += (value - location) * (value - location) * w * w * w * w;
519                ww_4u2 += w * (5.0 * w - 4.0);
520            }
521        }
522        let p = ww_4u2.abs();
523        (npts as f64 * sx_m2w4 / (p * (p - 1.0).max(1.0))).sqrt()
524    };
525
526    if max_niter >= 0 {
527        scale = scale_sums(location, scale, low, high);
528        let epsilon = 0.03 / std::f64::consts::SQRT_2 / (npts as f64 - 1.0).sqrt();
529        for iteration in 1..=max_niter {
530            let previous = scale;
531            location = location_sums(location, scale, low, high);
532            low = location - spread;
533            high = location + spread;
534            scale = scale_sums(location, scale, low, high);
535            if (1.0 - scale / previous).abs() < epsilon || iteration == max_niter {
536                break;
537            }
538            spread = c * scale;
539            low = location - spread;
540            high = location + spread;
541        }
542    } else {
543        let initial_location = location;
544        location = location_sums(initial_location, scale, low, high);
545        scale = scale_sums(initial_location, scale, low, high);
546    }
547    Ok((location, scale))
548}

550/// The maximum z-score Chauvenet's criterion allows for a sample of
551/// `npts` points, as casacore's `ZScoreCalculator` computes it: the
552/// tabulated breakpoints, then a bisection on
553/// `0.5 / erfc(z / sqrt(2))`.
554pub fn max_zscore(npts: u64) -> f64 {
555    const TABLE: [(u64, f64); 14] = [
556        (0, 0.5),
557        (1, 1.0),
558        (3, 1.5),
559        (10, 2.0),
560        (40, 2.5),
561        (185, 3.0),
562        (1074, 3.5),
563        (7893, 4.0),
564        (73579, 4.5),
565        (872138, 5.0),
566        (13165126, 5.5),
567        (253398672, 6.0),
568        (6225098696, 6.5),
569        (195341107722, 7.0),
570    ];
571    if let Some((_, z)) = TABLE.iter().find(|(n, _)| *n == npts) {
572        return *z;
573    }
574    let (mut low_z, mut high_z);
575    if npts > TABLE[TABLE.len() - 1].0 {
576        low_z = TABLE[TABLE.len() - 1].1;
577        high_z = low_z + 0.5;
578        while zscore_to_npts(high_z) < npts {
579            low_z = high_z;
580            high_z += 0.5;
581        }
582    } else {
583        let pair = TABLE
584            .windows(2)
585            .find(|pair| pair[0].0 < npts && npts < pair[1].0)
586            .expect("the table starts at zero");
587        low_z = pair[0].1;
588        high_z = pair[1].1;
589    }
590    let mut z = (low_z + high_z) / 2.0;
591    loop {
592        let npts_min = zscore_to_npts(z);
593        if npts_min == npts || (high_z - low_z).abs() <= 1.0e-13 * high_z.abs().max(1.0) {
594            return z;
595        }
596        if npts_min > npts {
597            high_z = z;
598        } else {
599            low_z = z;
600        }
601        z = (low_z + high_z) / 2.0;
602    }
603}

605/// The number of points whose maximum expected z-score is `z`:
606/// casacore's `zscoreToNpts`, `0.5 / erfc(z / sqrt(2))`.
607fn zscore_to_npts(z: f64) -> u64 {
608    (0.5 / erfc(z / std::f64::consts::SQRT_2)) as u64
609}

611/// The complementary error function.
612///
613/// `erfc(x) = Q(1/2, x * x)`, the regularized upper incomplete gamma
614/// function, evaluated with its alternating series for small arguments
615/// and the modified Lentz continued fraction for large ones, to
616/// double-precision accuracy.
617fn erfc(x: f64) -> f64 {
618    if x.is_nan() {
619        return f64::NAN;
620    }
621    if x < 0.0 {
622        return 2.0 - erfc(-x);
623    }
624    if x == 0.0 {
625        return 1.0;
626    }
627    if x < 1.3 {
628        // The alternating Maclaurin series of erf.
629        let x_squared = x * x;
630        let mut term = x;
631        let mut sum = x;
632        for n in 1..100 {
633            term *= -x_squared * (2 * n - 1) as f64 / (n * (2 * n + 1)) as f64;
634            sum += term;
635            if term.abs() < 1.0e-18 * sum.abs() {
636                break;
637            }
638        }
639        1.0 - 2.0 / std::f64::consts::PI.sqrt() * sum
640    } else {
641        // erfc(x) = Q(1/2, x^2): the continued fraction of the
642        // regularized upper incomplete gamma function.
643        let z = x * x;
644        let a = 0.5;
645        let tiny = 1.0e-300;
646        let mut b = z + 1.0 - a;
647        let mut c = 1.0 / tiny;
648        let mut d = 1.0 / b;
649        let mut h = d;
650        for iteration in 1..300 {
651            let an = -(iteration as f64) * (iteration as f64 - a);
652            b += 2.0;
653            d = an * d + b;
654            if d.abs() < tiny {
655                d = tiny;
656            }
657            c = b + an / c;
658            if c.abs() < tiny {
659                c = tiny;
660            }
661            d = 1.0 / d;
662            let delta = d * c;
663            h *= delta;
664            if (delta - 1.0).abs() < 1.0e-16 {
665                break;
666            }
667        }
668        (-z + a * z.ln() - std::f64::consts::PI.sqrt().ln()).exp() * h
669    }
670}
