2//! The simulator session: CASA's `sm` tool in Rust.
3//!
4//! A [`Simulator`] holds what the `sm` setters describe, as CASA's
5//! `Simulator` and `NewMSSimulator` would have written it to the subtables,
6//! and the scans `observe` plans. It writes nothing until
7//! [`Simulator::close`], which creates the MeasurementSet in one streamed
8//! pass (plan §4, deferred single-pass execution), so no reader can see an
9//! intermediate state.
10//!
11//! The setters keep CASA's semantics, including order-dependent ones:
12//! `setfield` records the reference time and first spectral window in force
13//! when it is called, `setconfig` and `setfeed` append, and `observe` reads
14//! the limits, autocorrelation weight and antennas defined at that moment.
15//! Where CASA crashes, writes garbage or corrupts data, the session refuses
16//! or diverges, as recorded in `docs/CASA (C++) bugs.md`.
17

42/// Controls CASA's `sm` lacks: the session's measures data, memory and the
43/// clock that stands for "now".
44#[derive(Clone)]
45pub struct SimulatorOptions {
46    /// Measures data (Earth orientation, observatories).
47    pub measures: Arc<dyn MeasuresProvider>,
48    /// The write session's memory and buffers.
49    pub write_resources: MeasurementSetWriteResources,
50    /// The time the session opens, UTC: CASA's reference time until
51    /// `settimes`.
52    pub now: MEpoch,
53}
54
55impl SimulatorOptions {
56    /// The production options: the discovered measures data, a write budget
57    /// from the machine's physical memory (two live batches), and the system
58    /// clock as "now".
59    ///
60    /// # Errors
61    ///
62    /// No measures data was found, or the memory budget could not be read.
63    pub fn from_system() -> SimulatorResult<Self> {
64        let measures = casa_ms::open_measures_runtime()
65            .map_err(|error| SimulatorError::raised(format!("measures data: {error}")))?;
66        let write_resources = MeasurementSetWriteResources::from_system_memory(2)
67            .map_err(|error| SimulatorError::raised(format!("memory budget: {error}")))?;
68        Ok(Self {
69            measures,
70            write_resources,
71            now: system_now(),
72        })
73    }
74}
75
76/// The system clock as a UTC epoch.
77fn system_now() -> MEpoch {
78    /// MJD of the Unix epoch, 1970-01-01.
79    const UNIX_EPOCH_MJD: f64 = 40_587.0;
80    let since_unix_s = std::time::SystemTime::now()
81        .duration_since(std::time::UNIX_EPOCH)
82        .map_or(0.0, |elapsed| elapsed.as_secs_f64());
83    MEpoch::from_mjd(UNIX_EPOCH_MJD + since_unix_s / 86_400.0, EpochRef::UTC)
84}

114    /// `sm.open(ms)`: a session that will create `path`, replacing any table
115    /// there, at [`close`](Self::close).
116    pub fn open(path: impl AsRef<Path>, options: SimulatorOptions) -> Self {
117        let times = Times::defaults(options.now.clone());
118        Self {
119            path: path.as_ref().to_path_buf(),
120            options,
121            // NewMSSimulator's telescope until setconfig.
122            telescope: "Unknown".to_string(),
123            antennas: Vec::new(),
124            feeds: Vec::new(),
125            feeds_set: false,
126            spectral_windows: Vec::new(),
127            fields: Vec::new(),
128            times,
129            elevation_limit_rad: 8.0_f64.to_radians(),
130            shadow_fraction_limit: casa_ms::shadowing::CASA_SHADOW_FRACTION_LIMIT,
131            autocorrelation_weight: 1.0,
132            observations: Vec::new(),
133            states: Vec::new(),
134            scans: Vec::new(),
135        }
136    }

149    /// `sm.setconfig(...)`: append the antennas.
150    ///
151    /// # Errors
152    ///
153    /// Coordinates or per-antenna lists of different lengths, or
154    /// `longlat` coordinates.
155    pub fn set_config(&mut self, config: &ArrayConfig) -> SimulatorResult<()> {
156        let antennas = config.antennas()?;
157        match config.coordinate_system {
158            array::CoordinateSystem::Global => {
159                tracing::info!("Using global coordinates for the antennas");
160            }
161            _ => tracing::info!("Using local coordinates for the antennas"),
162        }
163        self.telescope.clone_from(&config.telescope);
164        self.antennas.extend(antennas);
165        tracing::info!("Added rows to ANTENNA table");
166        Ok(())
167    }

169    /// `sm.setfeed(mode, x, y, pol)`: append one perfect feed per antenna.
170    ///
171    /// # Errors
172    ///
173    /// A mode other than `perfect R L`, `perfect X Y` or `list` (CASA
174    /// returns `False`), no antennas yet, or several feeds per antenna,
175    /// which casa-rs does not write yet.
176    pub fn set_feed(
177        &mut self,
178        mode: &str,
179        x: &[f64],
180        y: &[f64],
181        pol: &[String],
182    ) -> SimulatorResult<()> {
183        let feeds = feeds::feeds(mode, x, y, pol, self.antennas.len())?;
184        self.feeds.extend(feeds);
185        self.feeds_set = true;
186        tracing::info!("Added rows to FEED table");
187        Ok(())
188    }

190    /// `sm.setspwindow(...)`: add a spectral window and its polarization.
191    ///
192    /// # Errors
193    ///
194    /// No channels (CASA returns `False`).
195    pub fn set_spectral_window(&mut self, spec: &SpectralWindowSpec) -> SimulatorResult<()> {
196        let window = SpectralWindow::new(spec)?;
197        tracing::info!(
198            "Creating new spectral window {}, ID {}",
199            window.name,
200            self.spectral_windows.len() + 1
201        );
202        self.spectral_windows.push(window);
203        Ok(())
204    }

206    /// `sm.setfield(sourcename, sourcedirection, calcode, distance)`: add a
207    /// field and its source.
208    ///
209    /// # Errors
210    ///
211    /// An empty name (CASA returns `False`), or a direction that cannot be
212    /// converted to J2000 without an epoch and position.
213    pub fn set_field(
214        &mut self,
215        name: &str,
216        direction: &MDirection,
217        calcode: &str,
218        distance_m: f64,
219    ) -> SimulatorResult<()> {
220        if name.is_empty() {
221            return Err(SimulatorError::refused("must provide a source name"));
222        }
223        let j2000 = if direction.refer() == DirectionRef::J2000 {
224            direction.clone()
225        } else {
226            let frame = MeasFrame::new().with_measures(self.options.measures.clone());
227            direction
228                .convert_to(DirectionRef::J2000, &frame)
229                .map_err(|error| {
230                    SimulatorError::raised(format!(
231                        "Could not convert source direction to a J2000 Direction Measure: {error}"
232                    ))
233                })?
234        };
235        let (ra, dec) = j2000.as_angles();
236        // casacore's MVDirection longitude is in (−π, π].
237        let ra = ra.sin().atan2(ra.cos());
238        self.fields.push(Field {
239            name: name.to_string(),
240            code: calcode.to_string(),
241            direction_rad: [ra, dec],
242            distance_m,
243            source_time_s: self.times.reference_utc_s(),
244            rest_frequency_hz: self
245                .spectral_windows
246                .first()
247                .map(|window| window.start_frequency_hz),
248        });
249        Ok(())
250    }

252    /// `sm.settimes(integrationtime, usehourangle, referencetime)`.
253    ///
254    /// A reference epoch in another time scale is converted to UTC, which
255    /// the session works in.
256    ///
257    /// # Errors
258    ///
259    /// A negative integration time (CASA's assertion raises), or a
260    /// reference epoch that does not convert to UTC.
261    pub fn set_times(
262        &mut self,
263        integration_s: f64,
264        use_hour_angle: bool,
265        reference: MEpoch,
266    ) -> SimulatorResult<()> {
267        if integration_s < 0.0 {
268            return Err(SimulatorError::raised(
269                "Failed AlwaysAssert qIntTime.getValue(\"s\")>=0",
270            ));
271        }
272        let reference = if reference.refer() == EpochRef::UTC {
273            reference
274        } else {
275            let frame = MeasFrame::new().with_measures(self.options.measures.clone());
276            reference
277                .convert_to(EpochRef::UTC, &frame)
278                .map_err(|error| {
279                    SimulatorError::raised(format!(
280                        "Could not convert the {} reference time to UTC: {error}",
281                        reference.refer()
282                    ))
283                })?
284        };
285        tracing::info!("Times");
286        tracing::info!("     Integration time {integration_s}s");
287        if use_hour_angle {
288            tracing::info!("     Times will be interpreted as hour angles for first source");
289        }
290        self.times = Times {
291            integration_s,
292            use_hour_angle,
293            reference,
294            set: true,
295            hour_angle_offset_s: None,
296        };
297        Ok(())
298    }

331    /// The `observe` bookkeeping; see [`observe`](self::observe).
332    fn plan_scan(&mut self, spec: &ObserveSpec) -> SimulatorResult<()> {
333        if self.antennas.is_empty() {
334            return Err(SimulatorError::refused(
335                "Antenna information not yet defined",
336            ));
337        }
338        if !self.feeds_set && self.feeds.is_empty() {
339            tracing::warn!("Feeds have not been set - using default perfect R L");
340            self.feeds = feeds::perfect_feeds("perfect R L", self.antennas.len());
341        }
342        if !self.times.set {
343            tracing::warn!("Times have not been set - using defaults");
344            tracing::warn!("     Times will be interpreted as hour angles for first source");
345        }
346        if self.feeds.len() != self.antennas.len() {
347            return Err(SimulatorError::refused(
348                "Feed table format is incompatible with existing code of NewMSSimulator::observe",
349            ));
350        }
351        if self.spectral_windows.is_empty() {
352            return Err(SimulatorError::refused(
353                "Spectral window information not yet defined",
354            ));
355        }
356        let spectral_window_id = self
357            .spectral_windows
358            .iter()
359            .position(|window| window.name == spec.spectral_window)
360            .ok_or_else(|| {
361                SimulatorError::refused(format!(
362                    "Spectral window named {} not yet defined",
363                    spec.spectral_window
364                ))
365            })?;
366        if self.fields.is_empty() {
367            return Err(SimulatorError::refused("Field information not yet defined"));
368        }
369        let field_id = self
370            .fields
371            .iter()
372            .position(|field| field.name == spec.source)
373            .ok_or_else(|| {
374                SimulatorError::refused(format!("Field named {} not yet defined", spec.source))
375            })?;
376        let field = &self.fields[field_id];
377        tracing::info!("First source: {} @ {:?}", field.name, field.direction_rad);
378
379        let integration_s = self.times.integration_s;
380        if integration_s <= 0.0 {
381            return Err(SimulatorError::refused(
382                "observe: the integration time must be positive",
383            ));
384        }
385        if self.times.use_hour_angle && self.times.hour_angle_offset_s.is_none() {
386            self.times.hour_angle_offset_s = Some(observe::hour_angle_offset_s(
387                field.direction_rad,
388                &self.times.reference,
389                self.antennas[0].position_itrf_m,
390                self.options.measures.clone(),
391            )?);
392        }
393        let offset_s = if self.times.use_hour_angle {
394            self.times.hour_angle_offset_s.unwrap_or(0.0)
395        } else {
396            0.0
397        };
398        let reference_s = self.times.reference_utc_s();
399        let start_s = spec.start_s + reference_s + offset_s;
400        let stop_s = spec.stop_s + reference_s + offset_s;
401        tracing::info!("Full time range: {start_s} -- {stop_s} UTC with int = {integration_s}");
402        let integration_count = observe::integration_count(start_s, stop_s, integration_s);
403
404        let observation_id = if spec.add_observation || self.observations.is_empty() {
405            self.observations.push(Observation {
406                telescope: self.telescope.clone(),
407                time_range_s: [start_s, stop_s],
408                observer: spec.observer.clone(),
409                project: spec.project.clone(),
410            });
411            // One more than the largest OBSERVATION_ID in MAIN so far.
412            self.scans
413                .iter()
414                .map(|scan| scan.observation_id)
415                .max()
416                .map_or(0, |largest| largest + 1)
417        } else {
418            0
419        };
420        let state_id = match self.states.iter().rposition(|state| *state == spec.state) {
421            Some(existing) => existing,
422            None => {
423                self.states.push(spec.state.clone());
424                self.states.len() - 1
425            }
426        };
427        let scan_number = self.scans.last().map_or(0, |scan| scan.scan_number) + 1;
428        let scan = Scan {
429            scan_number,
430            field_id,
431            spectral_window_id,
432            start_s,
433            integration_s,
434            integration_count,
435            observation_id,
436            state_id,
437            antenna_count: self.antennas.len(),
438            phase_center_rad: field.direction_rad,
439            autocorrelation_weight: self.autocorrelation_weight,
440            elevation_limit_rad: self.elevation_limit_rad,
441            shadow_fraction_limit: self.shadow_fraction_limit,
442        };
443        tracing::info!(
444            "Calculating a total of {} integrations",
445            scan.integration_count
446        );
447        self.scans.push(scan);
448        Ok(())
449    }
