18/// Runtime-owned storage for canonical model samples.
19///
20/// The reconstruction owner retains this handle exclusively. Implementations
21/// must preserve exact values and support bits, check every requested range,
22/// and report I/O failures. Reads do not resize a cache beyond the physical
23/// plan's admitted capacity. Only owner-queued increments may mutate a logically
24/// immutable generation on first access. External handles must not mutate it.
25#[doc(hidden)]
26pub trait ModelSampleStorage: fmt::Debug + Send + Sync {
27    /// Return the immutable logical number of samples.
28    fn sample_count(&self) -> usize;
29
30    /// Copy one canonical range into the caller's already allocated window.
31    fn read(
32        &self,
33        start: usize,
34        destination: &mut [ModelSample],
35    ) -> Result<(), ModelLifecycleError>;
36
37    /// Replace one canonical range without changing the logical shape.
38    fn write(&mut self, start: usize, samples: &[ModelSample]) -> Result<(), ModelLifecycleError>;
39
40    /// Apply sorted sparse increments within one admitted window, returning the
41    /// largest updated magnitude. The owner serializes updates to overlapping
42    /// windows; disjoint windows may execute concurrently. Implementations must
43    /// batch backing access, preserve support, and propagate partial-write errors.
44    fn apply_updates(
45        &self,
46        updates: &[ModelSampleUpdate],
47        precision: NumericPrecision,
48        bound: f64,
49    ) -> Result<f64, ModelLifecycleError>;
50}
51
52/// A canonical sparse increment evaluated at its owning model window.
53#[doc(hidden)]
54#[derive(Debug, Clone)]
55pub struct ModelSampleUpdate {
56    pub(crate) index: usize,
57    pub(crate) increment: f64,
58}
59
60impl ModelSampleUpdate {
61    /// Absolute canonical sample index.
62    pub fn index(&self) -> usize {
63        self.index
64    }
65
66    /// Apply the compiled arithmetic and scientific checks to one changed cell.
67    pub fn apply(
68        &self,
69        sample: ModelSample,
70        precision: NumericPrecision,
71        bound: f64,
72    ) -> Result<ModelSample, ModelLifecycleError> {
73        if sample.support() != ModelSupport::Valid {
74            return Err(ModelLifecycleError::DeltaOutsideValidSupport);
75        }
76        let value = ModelValue::new(add_with_precision(
77            precision,
78            sample.value().value(),
79            self.increment,
80        ))?;
81        crate::validate_model_value(value, bound)?;
82        Ok(ModelSample::valid(value))
83    }
84}
85
86/// `left + right`, rounded to the lifecycle's arithmetic precision.
87fn add_with_precision(precision: NumericPrecision, left: f64, right: f64) -> f64 {
88    match precision {
89        NumericPrecision::F32 => f64::from((left as f32) + (right as f32)),
90        NumericPrecision::F64 => left + right,
91    }
92}

94#[derive(Debug)]
95struct PendingWindow {
96    range: Range<usize>,
97    updates: Box<[ModelSampleUpdate]>,
98    precision: NumericPrecision,
99    bound: f64,
100    applied: OnceLock<Result<(), ModelLifecycleError>>,
101}

103/// Physical allocation capability supplied by execution composition.
104#[doc(hidden)]
105pub trait ModelStorageFactory: fmt::Debug + Send + Sync {
106    /// Allocate an uninitialized logical model. The owner writes every sample
107    /// before it can mint a generation from the returned storage.
108    fn create(
109        &self,
110        sample_count: usize,
111    ) -> Result<Box<dyn ModelSampleStorage>, ModelLifecycleError>;
112}

114/// Admitted storage and maximum resident window for model lifecycle work.
115#[doc(hidden)]
116#[derive(Debug, Clone)]
117pub struct ModelStoragePlan {
118    factory: Arc<dyn ModelStorageFactory>,
119    window_samples: usize,
120}
121
122impl ModelStoragePlan {
123    /// Maximum queued sparse payload for a phase, excluding the input and
124    /// compiled delta which coexist during transfer. Each changed window has
125    /// one completion cell; arrays are allocated at their exact logical sizes.
126    pub fn pending_update_bytes(
127        terms: usize,
128        samples: usize,
129        window_samples: usize,
130    ) -> Option<usize> {
131        if window_samples == 0 {
132            return None;
133        }
134        let windows = terms.min(samples.div_ceil(window_samples));
135        terms
136            .checked_mul(size_of::<ModelSampleUpdate>())?
137            .checked_add(windows.checked_mul(size_of::<PendingWindow>())?)
138    }
139    /// Bind a physical storage capability to a positive resident sample limit.
140    pub fn new(
141        factory: Arc<dyn ModelStorageFactory>,
142        window_samples: usize,
143    ) -> Result<Self, ModelLifecycleError> {
144        if window_samples == 0 {
145            return Err(ModelLifecycleError::Storage(
146                "model storage window must contain at least one sample".into(),
147            ));
148        }
149        Ok(Self {
150            factory,
151            window_samples,
152        })
153    }
154
155    /// Select a fully resident allocation when the execution plan admits it.
156    /// This uses the same windowed model algorithms as paged allocations.
157    pub fn resident(window_samples: usize) -> Result<Self, ModelLifecycleError> {
158        Self::new(Arc::new(ResidentModelStorage), window_samples)
159    }
160
161    pub(crate) fn create(&self, count: usize) -> Result<ModelSamples, ModelLifecycleError> {
162        let storage = self.factory.create(count)?;
163        if storage.sample_count() != count {
164            return Err(ModelLifecycleError::SampleCountMismatch {
165                expected: count,
166                actual: storage.sample_count(),
167            });
168        }
169        Ok(ModelSamples {
170            storage,
171            window_samples: self.window_samples.min(count),
172            maximum_magnitude: AtomicU64::new(0),
173            pending: Vec::new(),
174        })
175    }
176}

178#[derive(Debug)]
179pub(crate) struct ModelSamples {
180    storage: Box<dyn ModelSampleStorage>,
181    window_samples: usize,
182    maximum_magnitude: AtomicU64,
183    pending: Vec<PendingWindow>,
184}
185
186impl ModelSamples {
187    pub(crate) fn maximum_magnitude(&self) -> f64 {
188        f64::from_bits(self.maximum_magnitude.load(Ordering::Relaxed))
189    }
190
191    pub(crate) fn queue_updates<I>(
192        &mut self,
193        updates: I,
194        precision: NumericPrecision,
195        bound: f64,
196    ) -> Result<(), ModelLifecycleError>
197    where
198        I: IntoIterator<Item = ModelSampleUpdate>,
199        I::IntoIter: Clone,
200    {
201        self.finish_updates()?;
202        let updates = updates.into_iter();
203        let mut previous = None;
204        let windows = updates
205            .clone()
206            .filter(|update| {
207                let window = update.index / self.window_samples;
208                let distinct = previous != Some(window);
209                previous = Some(window);
210                distinct
211            })
212            .count();
213        self.pending = Vec::with_capacity(windows);
214        let mut updates = updates.peekable();
215        while let Some(first) = updates.next() {
216            let start = first.index / self.window_samples * self.window_samples;
217            let end = start.saturating_add(self.window_samples).min(self.len());
218            let count = 1 + updates.clone().take_while(|next| next.index < end).count();
219            let mut window = Vec::with_capacity(count);
220            window.push(first);
221            while updates.peek().is_some_and(|next| next.index < end) {
222                window.push(updates.next().expect("peeked update"));
223            }
224            self.pending.push(PendingWindow {
225                range: start..end,
226                updates: window.into_boxed_slice(),
227                precision,
228                bound,
229                applied: OnceLock::new(),
230            });
231        }
232        Ok(())
233    }
234
235    fn apply_window(&self, window: &PendingWindow) -> Result<(), ModelLifecycleError> {
236        window
237            .applied
238            .get_or_init(|| {
239                let maximum =
240                    self.storage
241                        .apply_updates(&window.updates, window.precision, window.bound)?;
242                self.maximum_magnitude
243                    .fetch_max(maximum.to_bits(), Ordering::Relaxed);
244                Ok(())
245            })
246            .clone()
247    }
248
249    /// Complete only still-pending sparse writes before transferring the model.
250    /// Already-applied windows require no backing access or content verification.
251    pub(crate) fn finish_updates(&self) -> Result<(), ModelLifecycleError> {
252        for window in &self.pending {
253            self.apply_window(window)?;
254        }
255        Ok(())
256    }
257
258    pub(crate) fn complete_updates(&mut self) -> Result<(), ModelLifecycleError> {
259        self.finish_updates()?;
260        self.pending = Vec::new();
261        Ok(())
262    }
263
264    fn prepare_range(&self, range: Range<usize>) -> Result<(), ModelLifecycleError> {
265        if range.is_empty() {
266            return Ok(());
267        }
268        let first = self
269            .pending
270            .partition_point(|window| window.range.end <= range.start);
271        for window in self.pending[first..]
272            .iter()
273            .take_while(|window| window.range.start < range.end)
274        {
275            self.apply_window(window)?;
276        }
277        Ok(())
278    }
279    pub(crate) fn len(&self) -> usize {
280        self.storage.sample_count()
281    }
282
283    pub(crate) fn window_samples(&self) -> usize {
284        self.window_samples
285    }
286
287    pub(crate) fn read(
288        &self,
289        range: Range<usize>,
290    ) -> Result<Box<[ModelSample]>, ModelLifecycleError> {
291        if range.start > range.end || range.end > self.len() {
292            return Err(ModelLifecycleError::CellOutsideShape);
293        }
294        if range.len() > self.window_samples {
295            return Err(ModelLifecycleError::Storage(format!(
296                "model read requests {} samples beyond admitted window {}",
297                range.len(),
298                self.window_samples
299            )));
300        }
301        self.prepare_range(range.clone())?;
302        let mut result = vec![ModelSample::invalid(); range.len()].into_boxed_slice();
303        self.storage.read(range.start, &mut result)?;
304        Ok(result)
305    }
306
307    pub(crate) fn write(
308        &mut self,
309        start: usize,
310        samples: &[ModelSample],
311    ) -> Result<(), ModelLifecycleError> {
312        if start
313            .checked_add(samples.len())
314            .is_none_or(|end| end > self.len())
315        {
316            return Err(ModelLifecycleError::CellOutsideShape);
317        }
318        if samples.len() > self.window_samples {
319            return Err(ModelLifecycleError::Storage(format!(
320                "model write requests {} samples beyond admitted window {}",
321                samples.len(),
322                self.window_samples
323            )));
324        }
325        self.prepare_range(start..start + samples.len())?;
326        self.storage.write(start, samples)?;
327        let mut maximum = self.maximum_magnitude();
328        for sample in samples {
329            maximum = maximum.max(sample.value().value().abs());
330        }
331        self.maximum_magnitude
332            .store(maximum.to_bits(), Ordering::Relaxed);
333        Ok(())
334    }
335
336    pub(crate) fn for_each_window(
337        &self,
338        mut visit: impl FnMut(usize, &[ModelSample]) -> Result<(), ModelLifecycleError>,
339    ) -> Result<(), ModelLifecycleError> {
340        for start in (0..self.len()).step_by(self.window_samples) {
341            let samples =
342                self.read(start..start.saturating_add(self.window_samples).min(self.len()))?;
343            visit(start, &samples)?;
344        }
345        Ok(())
346    }
347}

355    #[derive(Debug)]
356    struct ProbeStorage {
357        samples: RwLock<Box<[ModelSample]>>,
358        calls: Arc<AtomicUsize>,
359        rendezvous: Option<(mpsc::Sender<usize>, Mutex<mpsc::Receiver<()>>)>,
360        fail_after_write: bool,
361    }

363    impl ModelSampleStorage for ProbeStorage {
364        fn sample_count(&self) -> usize {
365            8
366        }
367        fn read(
368            &self,
369            start: usize,
370            destination: &mut [ModelSample],
371        ) -> Result<(), ModelLifecycleError> {
372            ModelSampleStorage::read(&self.samples, start, destination)
373        }
374        fn write(
375            &mut self,
376            start: usize,
377            values: &[ModelSample],
378        ) -> Result<(), ModelLifecycleError> {
379            ModelSampleStorage::write(&mut self.samples, start, values)
380        }
381        fn apply_updates(
382            &self,
383            updates: &[ModelSampleUpdate],
384            precision: NumericPrecision,
385            bound: f64,
386        ) -> Result<f64, ModelLifecycleError> {
387            self.calls.fetch_add(1, Ordering::Relaxed);
388            if let Some((entered, resume)) = &self.rendezvous {
389                entered.send(updates[0].index()).unwrap();
390                resume
391                    .lock()
392                    .unwrap()
393                    .recv_timeout(Duration::from_secs(10))
394                    .map_err(|e| ModelLifecycleError::Storage(e.to_string()))?;
395            }
396            let maximum = self.samples.apply_updates(updates, precision, bound)?;
397            if self.fail_after_write {
398                return Err(ModelLifecycleError::Storage("partial write".into()));
399            }
400            Ok(maximum)
401        }
402    }

440    #[test]
441    fn sparse_updates_are_lazy_shared_once_and_disjoint_windows_can_start_together() {
442        let (entered, entries) = mpsc::channel();
443        let (resume, resumed) = mpsc::channel();
444        let (samples, calls) = probe(Some((entered, Mutex::new(resumed))), false);
445        assert_eq!(
446            calls.load(Ordering::Relaxed),
447            0,
448            "no serial cube preparation"
449        );
450        std::thread::scope(|scope| {
451            let first = scope.spawn(|| samples.read(0..4).unwrap());
452            let second = scope.spawn(|| samples.read(4..8).unwrap());
453            let shared = scope.spawn(|| samples.read(0..4).unwrap());
454            let mut indices = [
455                entries.recv_timeout(Duration::from_secs(10)).unwrap(),
456                entries.recv_timeout(Duration::from_secs(10)).unwrap(),
457            ];
458            indices.sort();
459            assert_eq!(indices, [1, 6], "both regions enter before either finishes");
460            resume.send(()).unwrap();
461            resume.send(()).unwrap();
462            let first = first.join().unwrap();
463            assert_eq!(first[1].value().value(), 2.0);
464            assert_eq!(second.join().unwrap()[2].value().value(), -3.0);
465            assert_eq!(
466                shared.join().unwrap(),
467                first,
468                "overlapping halo waits for one update"
469            );
470        });
471        samples.finish_updates().unwrap();
472        assert_eq!(
473            calls.load(Ordering::Relaxed),
474            2,
475            "exactly one sparse call per changed window"
476        );
477        assert_eq!(samples.maximum_magnitude(), 3.0);
478    }

559impl ModelStorageFactory for ResidentModelStorage {
560    fn create(
561        &self,
562        sample_count: usize,
563    ) -> Result<Box<dyn ModelSampleStorage>, ModelLifecycleError> {
564        Ok(Box::new(RwLock::new(
565            vec![ModelSample::invalid(); sample_count].into_boxed_slice(),
566        )))
567    }
568}
569
570impl ModelSampleStorage for RwLock<Box<[ModelSample]>> {
571    fn sample_count(&self) -> usize {
572        self.read().expect("model storage poisoned").len()
573    }
574
575    fn read(
576        &self,
577        start: usize,
578        destination: &mut [ModelSample],
579    ) -> Result<(), ModelLifecycleError> {
580        let end = start
581            .checked_add(destination.len())
582            .ok_or(ModelLifecycleError::CellOutsideShape)?;
583        let samples = self
584            .read()
585            .map_err(|e| ModelLifecycleError::Storage(e.to_string()))?;
586        let source = samples
587            .get(start..end)
588            .ok_or(ModelLifecycleError::CellOutsideShape)?;
589        destination.copy_from_slice(source);
590        Ok(())
591    }
592
593    fn write(&mut self, start: usize, samples: &[ModelSample]) -> Result<(), ModelLifecycleError> {
594        let end = start
595            .checked_add(samples.len())
596            .ok_or(ModelLifecycleError::CellOutsideShape)?;
597        let destination = self
598            .get_mut()
599            .map_err(|e| ModelLifecycleError::Storage(e.to_string()))?
600            .get_mut(start..end)
601            .ok_or(ModelLifecycleError::CellOutsideShape)?;
602        destination.copy_from_slice(samples);
603        Ok(())
604    }
605
606    fn apply_updates(
607        &self,
608        updates: &[ModelSampleUpdate],
609        precision: NumericPrecision,
610        bound: f64,
611    ) -> Result<f64, ModelLifecycleError> {
612        let mut samples = self
613            .write()
614            .map_err(|e| ModelLifecycleError::Storage(e.to_string()))?;
615        let mut maximum: f64 = 0.0;
616        for update in updates {
617            let sample = samples
618                .get_mut(update.index)
619                .ok_or(ModelLifecycleError::CellOutsideShape)?;
620            *sample = update.apply(*sample, precision, bound)?;
621            maximum = maximum.max(sample.value().value().abs());
622        }
623        Ok(maximum)
624    }
