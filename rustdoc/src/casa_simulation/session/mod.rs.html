2//! The simulator session: CASA's `sm` tool in Rust.
3//!
4//! A [`Simulator`] holds what the `sm` setters describe, as CASA's
5//! `Simulator` and `NewMSSimulator` would have written it to the subtables,
6//! and the scans `observe` plans. It writes nothing until
7//! [`Simulator::close`], which creates the MeasurementSet in one streamed
8//! pass (plan §4, deferred single-pass execution), so no reader can see an
9//! intermediate state. The steps that change DATA (`predict`, `corrupt`) are
10//! recorded in a plan of which scans each covers, and applied as `close`
11//! writes each row (`plan`, `prediction`, `corruption`); `predict` prepares
12//! its model's grids when it is called.
13//!
14//! A session [`Simulator::open_from_ms`] opens on an existing MeasurementSet
15//! (`sm.openfromms`) reads its state back, takes its rows as scans already
16//! observed, and patches their DATA in place at `close` (`from_ms`).
17//!
18//! The session runs within its [`SessionResources`]: `predict` admits its
19//! model's grids, and `close` what it holds, against the imaging runtime's
20//! resource policy, and `close` computes each write batch's geometry,
21//! prediction and noise on the runtime's worker team, so the MeasurementSet
22//! is bitwise the same for any budget or worker count. A model is predicted
23//! on the CPU or the host's Metal device ([`SessionResources`]), whose
24//! predictions agree with the CPU's to `1e-4` of the largest; noise is
25//! drawn on the CPU on either.
26//!
27//! The setters keep CASA's semantics, including order-dependent ones:
28//! `setfield` records the reference time and first spectral window in force
29//! when it is called, `setconfig` and `setfeed` append, and `observe` reads
30//! the limits, autocorrelation weight and antennas defined at that moment.
31//! Where CASA crashes, writes garbage or corrupts data, the session refuses
32//! or diverges, as recorded in `docs/CASA (C++) bugs.md`.

79/// Controls CASA's `sm` lacks: the session's measures data, its resources
80/// and the clock that stands for "now".
81#[derive(Clone)]
82pub struct SimulatorOptions {
83    /// Measures data (Earth orientation, observatories).
84    pub measures: Arc<dyn MeasuresProvider>,
85    /// The host, memory budget, workers and backend `close` runs with.
86    pub resources: SessionResources,
87    /// The time the session opens, UTC: CASA's reference time until
88    /// `settimes`.
89    pub now: MEpoch,
90}

93    /// The production options: the discovered measures data, `resources`,
94    /// and the system clock as "now".
95    ///
96    /// # Errors
97    ///
98    /// No measures data was found.
99    pub fn from_system(resources: SessionResources) -> SimulatorResult<Self> {
100        let measures = casa_ms::open_measures_runtime()
101            .map_err(|error| SimulatorError::raised(format!("measures data: {error}")))?;
102        Ok(Self {
103            measures,
104            resources,
105            now: system_now(),
106        })
107    }
108}

110/// The system clock as a UTC epoch.
111fn system_now() -> MEpoch {
112    /// MJD of the Unix epoch, 1970-01-01.
113    const UNIX_EPOCH_MJD: f64 = 40_587.0;
114    let since_unix_s = std::time::SystemTime::now()
115        .duration_since(std::time::UNIX_EPOCH)
116        .map_or(0.0, |elapsed| elapsed.as_secs_f64());
117    MEpoch::from_mjd(UNIX_EPOCH_MJD + since_unix_s / 86_400.0, EpochRef::UTC)
118}

257        Ok(())

299    /// `sm.setfield(sourcename, sourcedirection, calcode, distance)`: add a
300    /// field and its source.
301    ///
302    /// # Errors
303    ///
304    /// An empty name (CASA returns `False`), or a direction that cannot be
305    /// converted to J2000 without an epoch and position.
306    pub fn set_field(
307        &mut self,
308        name: &str,
309        direction: &MDirection,
310        calcode: &str,
311        distance_m: f64,
312    ) -> SimulatorResult<()> {
313        self.refuse_after_openfromms("setfield")?;
314        if name.is_empty() {
315            return Err(SimulatorError::refused("must provide a source name"));
316        }
317        let direction_rad = self.j2000_rad(direction, "source direction")?;
318        self.fields.push(Field {
319            name: name.to_string(),
320            code: calcode.to_string(),
321            direction_rad,
322            distance_m,
323            source_time_s: self.times.reference_utc_s(),
324            rest_frequency_hz: self
325                .spectral_windows
326                .first()
327                .map(|window| window.reference_frequency_hz),
328        });
329        Ok(())
330    }

332    /// `sm.setmosaicfield(sourcename, calcode, fieldcenter, xmosp, ymosp,
333    /// mosspacing, distance)`: add an `nx × ny` grid of fields, named
334    /// `{name}_{k}` with `k` counting down each column (y fastest) and then
335    /// across, `spacing_rad` apart in latitude and in longitude at the
336    /// centre's latitude, centred on `centre` and in its frame.
337    ///
338    /// **Divergence** (plan §4, `docs/CASA (C++) bugs.md`).
339    /// - CASA tests an odd count as `nx/2 != floor(nx/2)` in integer
340    ///   arithmetic, so every grid takes its even-count offsets. Those count
341    ///   from zero where one was meant, so CASA's grids sit off their centre.
342    /// - Over a pole, CASA sets the longitude to `|π − α|`, where crossing
343    ///   the pole moves it by π.
344    ///
345    /// casa-rs centres each grid on `centre`: offsets `(i − (n − 1)/2)` times
346    /// the spacing. It crosses a pole as the sphere does.
347    ///
348    /// # Errors
349    ///
350    /// As [`set_field`](Self::set_field), for any of the fields.
351    pub fn set_mosaic_field(
352        &mut self,
353        name: &str,
354        calcode: &str,
355        centre: &MDirection,
356        (nx, ny): (usize, usize),
357        spacing_rad: f64,
358        distance_m: f64,
359    ) -> SimulatorResult<()> {
360        use std::f64::consts::{FRAC_PI_2, PI, TAU};
361        self.refuse_after_openfromms("setmosaicfield")?;
362        let (ra0, dec0) = centre.as_angles();
363        let offset = |index: usize, count: usize| index as f64 - (count as f64 - 1.0) / 2.0;
364        let mut k = 0;
365        for i in 0..nx {
366            for j in 0..ny {
367                let mut ra = ra0 + offset(i, nx) * spacing_rad / dec0.cos();
368                let mut dec = dec0 + offset(j, ny) * spacing_rad;
369                if ra > TAU {
370                    ra -= TAU;
371                }
372                if dec.abs() > FRAC_PI_2 {
373                    dec = dec.signum() * (PI - dec.abs());
374                    ra = (ra + PI).rem_euclid(TAU);
375                }
376                let direction = MDirection::from_angles(ra, dec, centre.refer());
377                self.set_field(&format!("{name}_{k}"), &direction, calcode, distance_m)?;
378                k += 1;
379            }
380        }
381        Ok(())
382    }

384    /// `sm.settimes(integrationtime, usehourangle, referencetime)`.
385    ///
386    /// A reference epoch in another time scale is converted to UTC, which
387    /// the session works in.
388    ///
389    /// # Errors
390    ///
391    /// A negative integration time (CASA's assertion raises), or a
392    /// reference epoch that does not convert to UTC.
393    pub fn set_times(
394        &mut self,
395        integration_s: f64,
396        use_hour_angle: bool,
397        reference: MEpoch,
398    ) -> SimulatorResult<()> {
399        if integration_s < 0.0 {
400            return Err(SimulatorError::raised(
401                "Failed AlwaysAssert qIntTime.getValue(\"s\")>=0",
402            ));
403        }
404        let reference = if reference.refer() == EpochRef::UTC {
405            reference
406        } else {
407            let frame = MeasFrame::new().with_measures(self.options.measures.clone());
408            reference
409                .convert_to(EpochRef::UTC, &frame)
410                .map_err(|error| {
411                    SimulatorError::raised(format!(
412                        "Could not convert the {} reference time to UTC: {error}",
413                        reference.refer()
414                    ))
415                })?
416        };
417        tracing::info!("Times");
418        tracing::info!("     Integration time {integration_s}s");
419        if use_hour_angle {
420            tracing::info!("     Times will be interpreted as hour angles for first source");
421        }
422        self.times = Times {
423            integration_s,
424            use_hour_angle,
425            reference,
426            set: true,
427            hour_angle_offset_s: None,
428        };
429        Ok(())
430    }

470    /// `sm.setdata(spwid, fieldid, msselect)`: select the rows the next
471    /// `predict` and `corrupt` act on, until the next `observe`, which drops
472    /// CASA's selection.
473    ///
474    /// CASA's selection works on the rows observed so far:
475    /// - their FIELD_ID must be among `field_ids`, when there are any;
476    /// - their DATA_DESC_ID must be one of the windows `spectral_window_ids`
477    ///   names, when it names any (casa-rs's data description `w` is
478    ///   window `w`);
479    /// - then the TaQL `msselect` applies;
480    /// - a selection that keeps no row reverts to every row, with a warning.
481    ///
482    /// casa-rs takes a selection that keeps every row, as `simobserve`'s
483    /// `setdata(fieldid=range(nfld))` does. The following steps then act on
484    /// every row observed so far, as without `setdata`.
485    ///
486    /// # Errors
487    ///
488    /// Not yet implemented: a selection that keeps some but not all of the
489    /// rows, or a non-blank `msselect`.
490    pub fn set_data(
491        &mut self,
492        spectral_window_ids: &[i64],
493        field_ids: &[i64],
494        ms_select: &str,
495    ) -> SimulatorResult<()> {
496        tracing::info!("Selecting data");
497        if !ms_select.trim().is_empty() {
498            return Err(SimulatorError::not_yet_implemented(format!(
499                "setdata(msselect='{ms_select}') is not yet implemented in casa-rs (#841)"
500            )));
501        }
502        let windows: Vec<usize> = (0..self.spectral_windows.len())
503            .filter(|window| spectral_window_ids.contains(&(*window as i64)))
504            .collect();
505        if field_ids.is_empty() && windows.is_empty() {
506            return Ok(());
507        }
508        let selected = |scan: &Scan| {
509            (field_ids.is_empty() || field_ids.contains(&(scan.field_id as i64)))
510                && (windows.is_empty() || windows.contains(&scan.spectral_window_id))
511        };
512        let kept = self.scans.iter().filter(|scan| selected(scan)).count();
513        if kept == 0 {
514            tracing::warn!("Selection is empty: reverting to original MeasurementSet");
515            return Ok(());
516        }
517        if kept == self.scans.len() {
518            return Ok(());
519        }
520        Err(SimulatorError::not_yet_implemented(format!(
521            "setdata selecting {kept} of the {} scans observed so far is not yet implemented in \
522             casa-rs (simulation Phase 2); a selection of them all is",
523            self.scans.len()
524        )))
525    }

527    /// `sm.close()` / `sm.done()`: write the MeasurementSet, or, after
528    /// `openfromms`, patch the rows the DATA steps rewrite in place; then
529    /// write the calibration tables `setgain` and `setleakage` named.
530    ///
531    /// # Errors
532    ///
533    /// A failure to create, read or write the MeasurementSet or a
534    /// calibration table.
535    pub fn close(self) -> SimulatorResult<()> {
536        if self.opened.is_some() {
537            write::patch_measurement_set(&self)?;
538        } else {
539            write::write_measurement_set(&self)?;
540        }
541        self.write_jones_tables()
542    }

544    /// `direction` in J2000, as `[ra, dec]` radians with the longitude in
545    /// (−π, π] as casacore's `MVDirection` keeps it; `what` names it in the
546    /// error.
547    fn j2000_rad(&self, direction: &MDirection, what: &str) -> SimulatorResult<[f64; 2]> {
548        let j2000 = if direction.refer() == DirectionRef::J2000 {
549            direction.clone()
550        } else {
551            let frame = MeasFrame::new().with_measures(self.options.measures.clone());
552            direction
553                .convert_to(DirectionRef::J2000, &frame)
554                .map_err(|error| {
555                    SimulatorError::raised(format!(
556                        "Could not convert {what} to a J2000 Direction Measure: {error}"
557                    ))
558                })?
559        };
560        let (ra, dec) = j2000.as_angles();
561        Ok([ra.sin().atan2(ra.cos()), dec])
562    }
