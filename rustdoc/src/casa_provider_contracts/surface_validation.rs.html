4use super::*;

6/// A validation view avoids copying the catalog when many surfaces share it.
7pub(super) struct SurfaceContractRef<'a> {
8    pub(super) schema_version: u32,
9    pub(super) surface: &'a SurfaceDefinition,
10    pub(super) catalog: &'a ParameterCatalog,
11}
12
13impl<'a> From<&'a SurfaceContractBundle> for SurfaceContractRef<'a> {
14    fn from(bundle: &'a SurfaceContractBundle) -> Self {
15        Self {
16            schema_version: bundle.schema_version,
17            surface: &bundle.surface,
18            catalog: &bundle.catalog,
19        }
20    }
21}
22
23pub(super) fn validate_surface_views(
24    bundles: &[SurfaceContractRef<'_>],
25    validate_cross_surface_names: bool,
26) -> Result<(), Vec<ContractValidationError>> {
27    let mut errors = Vec::new();
28    let mut merged = BTreeMap::<(&ParameterConceptId, SemanticRevision), &ParameterConcept>::new();
29    let mut surface_ids = BTreeSet::new();
30    let mut last_stable_catalog: Option<&ParameterCatalog> = None;
31
32    for bundle in bundles {
33        if bundle.schema_version != PARAMETER_CATALOG_SCHEMA_VERSION
34            || bundle.catalog.schema_version != PARAMETER_CATALOG_SCHEMA_VERSION
35        {
36            errors.push(ContractValidationError::new(
37                "schema_version",
38                format!(
39                    "surface {} uses unsupported parameter schema {} / catalog {}",
40                    bundle.surface.id(),
41                    bundle.schema_version,
42                    bundle.catalog.schema_version
43                ),
44            ));
45        }
46        // A former id names the surface in saved profiles, so it may name no
47        // other surface, now or before.
48        for id in std::iter::once(bundle.surface.id())
49            .chain(bundle.surface.former_ids().iter().map(String::as_str))
50        {
51            if !surface_ids.insert(id.to_string()) {
52                errors.push(ContractValidationError::new(
53                    "duplicate_surface",
54                    format!("duplicate surface {id}"),
55                ));
56            }
57        }
58        // Consecutive surfaces often borrow the very same catalog. A second
59        // scan cannot add definitions or diagnostics if the first found no
60        // divergent definitions. If it did, scan again: those errors are
61        // intentionally reported for every surface.
62        if !last_stable_catalog.is_some_and(|last| std::ptr::eq(last, bundle.catalog)) {
63            let mut stable = true;
64            for concept in &bundle.catalog.concepts {
65                let key = (&concept.id, concept.semantic_revision);
66                match merged.get(&key) {
67                    Some(existing) if *existing != concept => {
68                        stable = false;
69                        errors.push(ContractValidationError::new(
70                            "divergent_concept",
71                            format!(
72                                "concept {} revision {} has divergent definitions",
73                                concept.id, concept.semantic_revision.0
74                            ),
75                        ));
76                    }
77                    Some(_) => {}
78                    None => {
79                        // A definition containing NaN is not equal to itself.
80                        // The next surface must still compare it and report
81                        // the same divergence as the full scan.
82                        if !concept.eq(concept) {
83                            stable = false;
84                        }
85                        if let NormalizationRule::CasaSelector { capabilities, .. } =
86                            &concept.normalization
87                        {
88                            for capability in capabilities.iter().filter(|capability| {
89                                !SELECTOR_CAPABILITY_NAMES.contains(&capability.as_str())
90                            }) {
91                                errors.push(ContractValidationError::new(
92                                    "unknown_selector_capability",
93                                    format!(
94                                        "concept {} declares unknown selector capability {capability:?}",
95                                        concept.id
96                                    ),
97                                ));
98                            }
99                        }
100                        merged.insert(key, concept);
101                    }
102                }
103            }
104            last_stable_catalog = stable.then_some(bundle.catalog);
105        }
106        validate_one_surface(bundle, &mut errors);
107    }
108
109    if validate_cross_surface_names {
110        validate_repeated_names(bundles, &mut errors);
111    }
112    if errors.is_empty() {
113        Ok(())
114    } else {
115        Err(errors)
116    }
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122
123    #[test]
124    fn reused_and_distinct_catalogs_preserve_error_order() {
125        let base = crate::builtin_surface_catalog().unwrap();
126        let first = &base.catalog;
127        let mut different = first.clone();
128        different.concepts[0].casa_name.push_str("_different");
129        let surfaces = [
130            (&base.surfaces[0], first, 999),
131            (&base.surfaces[1], first, base.schema_version),
132            (&base.surfaces[2], &different, base.schema_version),
133            (&base.surfaces[0], first, base.schema_version),
134        ];
135        let views: Vec<_> = surfaces
136            .iter()
137            .map(|(surface, catalog, schema_version)| SurfaceContractRef {
138                schema_version: *schema_version,
139                surface,
140                catalog,
141            })
142            .collect();
143        let owned: Vec<_> = surfaces
144            .iter()
145            .map(|(surface, catalog, schema_version)| SurfaceContractBundle {
146                schema_version: *schema_version,
147                surface: (*surface).clone(),
148                catalog: (*catalog).clone(),
149            })
150            .collect();
151        let expected = validate_surface_bundles(&owned);
152        let errors = expected.as_ref().unwrap_err();
153        for code in ["schema_version", "divergent_concept", "duplicate_surface"] {
154            assert!(errors.iter().any(|error| error.code == code), "{code}");
155        }
156        assert_eq!(validate_surface_views(&views, true), expected);
157    }
158}
