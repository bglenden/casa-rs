15#[pymethods]
16impl PyImageTool {
17    fn brightnessunit(&self, py: Python<'_>) -> String {
18        py.detach(|| {
19            self.state().image.as_ref().map_or_else(String::new, |image| {
20                with_pixels!(&*image.lock(), storage => with_storage!(storage, image => image.units().into()))
21            })
22        })
23    }
24
25    fn setbrightnessunit(&self, py: Python<'_>, unit: String) -> PyResult<bool> {
26        py.detach(|| {
27            let state = self.state();
28            let history = state.history;
29            let Some(image) = state.image.as_ref() else {
30                return Ok(false);
31            };
32            let mut image = image.lock();
33            image.check_single_precision("setbrightnessunit")?;
34            let normalized = normalize_casa_unit_name(unit.trim());
35            Unit::new(&normalized)
36                .map_err(|_| raised(format!("Unit::check Illegal unit string '{unit}'")))?;
37            with_pixels!(&mut *image, storage => with_storage!(storage, image => image.set_units(normalized)))
38                .map_err(raised)?;
39            if history {
40                image.record_history(
41                    "setbrightnessunit",
42                    format!("ia.setbrightnessunit(unit={unit:?})"),
43                )?;
44            }
45            image.save().map_err(raised)?;
46            Ok(true)
47        })
48    }

50    #[pyo3(signature = (axes=None), text_signature = "($self, axes=[-1])")]
51    fn coordsys(&self, py: Python<'_>, axes: Option<IntList>) -> PyResult<Py<PyAny>> {
52        let tool = py.detach(|| {
53            let state = self.state();
54            let Some(image) = state.image.as_ref() else {
55                return Ok(PyCoordsys::empty());
56            };
57            let image = image.lock();
58            let axes = axes.as_ref().map_or(&[][..], |arg| arg.0.as_slice());
59            let all_axes = axes.is_empty()
60                || axes == [-1]
61                || axes.iter().copied().eq(0..image.shape().len() as i64);
62            let record = image.coordinate_record();
63            if all_axes {
64                return PyCoordsys::from_record(record);
65            }
66            let coordinates = CoordinateSystem::from_record(&record).map_err(raised)?;
67            let mut result = CoordinateSystem::new().with_obs_info(coordinates.obs_info().clone());
68            let mut first = 0;
69            let mut selected = Vec::new();
70            for index in 0..coordinates.n_coordinates() {
71                let coordinate = coordinates.coordinate(index);
72                let end = first + coordinate.n_pixel_axes() as i64;
73                if (first..end).all(|axis| axes.contains(&axis)) {
74                    result.add_coordinate(coordinate.clone());
75                    selected.extend(first..end);
76                }
77                first = end;
78            }
79            if selected != axes {
80                return Err(stub("coordsys").unwrap_err());
81            }
82            PyCoordsys::from_record(result.to_record())
83        })?;
84        Ok(Py::new(py, tool)?.into_any())
85    }
86
87    fn setcoordsys(&self, py: Python<'_>, csys: &Bound<'_, PyDict>) -> PyResult<bool> {
88        let record = casa_record(py, Some(csys))?;
89        py.detach(|| {
90            let state = self.state();
91            let history = state.history;
92            let Some(image) = state.image.as_ref() else {
93                return Ok(false);
94            };
95            let mut image = image.lock();
96            if record.is_empty() {
97                return Err(raised("Exception: Record is empty."));
98            }
99            let coordinates = CoordinateSystem::from_record(&record).map_err(raised)?;
100            if coordinates.n_pixel_axes() != image.shape().len() {
101                return Err(raised(
102                    "Exception: Supplied CoordinateSystem and image shape are inconsistent",
103                ));
104            }
105            image.set_coordinate_record(record)?;
106            if history {
107                image.record_history("setcoordsys", "ia.setcoordsys(csys={...})".into())?;
108            }
109            image.save().map_err(raised)?;
110            Ok(true)
111        })
112    }
113
114    #[pyo3(signature = (list=true), text_signature = "($self, list=True)")]
115    fn history(&self, py: Python<'_>, list: bool) -> PyResult<Vec<String>> {
116        let entries = py.detach(|| {
117            let state = self.state();
118            let Some(image) = state.image.as_ref() else {
119                return Ok(Vec::new());
120            };
121            with_pixels!(&*image.lock(), storage => with_storage!(storage, image => image.history()))
122                .map_err(raised)
123        })?;
124        if list {
125            crate::tools::emit(
126                py,
127                "casars.tools.image",
128                &entries
129                    .iter()
130                    .cloned()
131                    .map(crate::tools::LogLine::info)
132                    .collect::<Vec<_>>(),
133            )?;
134        }
135        Ok(entries)
136    }
