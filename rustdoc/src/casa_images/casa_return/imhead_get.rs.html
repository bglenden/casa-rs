5use casa_provider_contracts::CasaValue;

8/// A failure selecting a CASA image-header value.
9#[derive(Debug, thiserror::Error)]
10pub enum ImheadGetError {
11    /// The requested keyword has not been implemented.
12    #[error("imhead get of hdkey {0:?} is not implemented by casa-rs yet (#977)")]
13    UnsupportedKey(String),
14    /// CASA rejects the requested metadata operation.
15    #[error("{0}")]
16    Execution(String),
17    /// The image or its metadata could not be read.
18    #[error("{0}")]
19    Image(#[source] ImageError),
20}
21
22/// Return a CASA `imhead(mode="get")` value from an image header.
23///
24/// `path` must name the image from which `summary` was computed. For extrema
25/// keys the summary must include the extrema computed by
26/// [`crate::imhead_extrema`]. Unknown and unsupported fields return an error.
27///
28/// # Errors
29///
30/// Returns an error for unavailable metadata, invalid axes, unsupported keys,
31/// or a failure reading a user-defined field from the image.
32pub fn imhead_get_value(
33    summary: &ImageHeaderSummary,
34    path: &Path,
35    key: &str,
36) -> Result<CasaValue, ImheadGetError> {
37    let lower = key.to_ascii_lowercase();
38    let value = match lower.as_str() {
39        "bunit" => CasaValue::string(&summary.units),
40        "imtype" => CasaValue::string(&summary.image_type),
41        "object" => CasaValue::string(&summary.object_name),
42        "date-obs" | "epoch" => CasaValue::string(&summary.observation_date),
43        "observer" => CasaValue::string(&summary.observer),
44        "telescope" => CasaValue::string(&summary.telescope),
45        "equinox" => CasaValue::string(summary.direction_reference.as_deref().unwrap_or("")),
46        "projection" => CasaValue::string(summary.direction_projection.as_deref().unwrap_or("")),
47        "reffreqtype" => {
48            CasaValue::string(summary.spectral.as_ref().map_or("", |s| &s.reference_frame))
49        }
50        "restfreq" => {
51            let spectral = summary.spectral.as_ref().ok_or_else(|| {
52                execution("Image has no spectral axis so there is no rest frequency")
53            })?;
54            CasaValue::dict([
55                ("unit", CasaValue::string(&spectral.unit)),
56                ("value", CasaValue::float(spectral.rest_frequency)),
57            ])
58        }
59        "masks" => CasaValue::str_vector(summary.masks.clone()),
60        "shape" => CasaValue::int64_vector(summary.shape.iter().map(|&n| n as i64)),
61        "datamin" | "datamax" | "minpixpos" | "maxpixpos" | "minpos" | "maxpos" => {
62            let extrema = summary
63                .extrema
64                .as_ref()
65                .ok_or_else(|| execution(format!("RecordInterface: field {lower} is unknown")))?;
66            match lower.as_str() {
67                "datamin" => CasaValue::float(extrema.min),
68                "datamax" => CasaValue::float(extrema.max),
69                "minpixpos" => {
70                    CasaValue::int64_vector(extrema.min_position.iter().map(|&n| n as i64))
71                }
72                "maxpixpos" => {
73                    CasaValue::int64_vector(extrema.max_position.iter().map(|&n| n as i64))
74                }
75                "minpos" => CasaValue::string(&extrema.min_world),
76                _ => CasaValue::string(&extrema.max_world),
77            }
78        }
79        "beammajor" | "beamminor" | "beampa" | "bmaj" | "bmin" | "bpa" => {
80            let beam = summary.restoring_beam.as_ref().ok_or_else(|| {
81                execution(if summary.has_multiple_beams {
82                    "This image has multiple beams."
83                } else {
84                    "This image has no beam(s)."
85                })
86            })?;
87            let value = match lower.as_str() {
88                "beammajor" | "bmaj" => &beam.major,
89                "beamminor" | "bmin" => &beam.minor,
90                _ => &beam.wrapped_position_angle,
91            };
92            CasaValue::dict([
93                ("unit", CasaValue::string(&value.unit)),
94                ("value", CasaValue::float(value.value)),
95            ])
96        }
97        _ if crate::analysis::imhead_is_custom_key(key) => {
98            return crate::analysis::imhead_custom_value(path, key).map_err(ImheadGetError::Image);
99        }
100        _ => return axis_header_value(summary, key, &lower),
101    };
102    Ok(value)
103}

105fn axis_header_value(
106    summary: &ImageHeaderSummary,
107    key: &str,
108    lower: &str,
109) -> Result<CasaValue, ImheadGetError> {
110    if let Some((prefix, suffix)) = lower
111        .get(..5)
112        .zip(lower.get(5..))
113        .filter(|(prefix, _)| ["cdelt", "crpix", "crval", "ctype", "cunit"].contains(prefix))
114    {
115        let index = crate::analysis::imhead_axis_index(suffix);
116        let axis = index
117            .and_then(|index| summary.axes.get(index))
118            .ok_or_else(|| execution(format!("imhead: invalid axis keyword {key:?}")))?;
119        let quantity = |value| {
120            CasaValue::dict([
121                ("unit", CasaValue::string(&axis.unit)),
122                ("value", CasaValue::float(value)),
123            ])
124        };
125        return match prefix {
126            "cdelt" => Ok(quantity(axis.increment)),
127            "crpix" => Ok(CasaValue::float(axis.reference_pixel)),
128            "crval" if axis.coordinate_type == "Stokes" => {
129                Ok(CasaValue::str_vector(axis.stokes.clone()))
130            }
131            "crval" => Ok(quantity(axis.reference_value)),
132            "ctype" => Ok(CasaValue::string(&axis.name)),
133            "cunit" => Ok(CasaValue::string(&axis.unit)),
134            _ => Err(unimplemented_header_key(key)),
135        };
136    }
137    Err(unimplemented_header_key(key))
138}
139
140fn unimplemented_header_key(key: &str) -> ImheadGetError {
141    ImheadGetError::UnsupportedKey(key.to_string())
142}
143
144fn execution(message: impl Into<String>) -> ImheadGetError {
145    ImheadGetError::Execution(message.into())
146}
