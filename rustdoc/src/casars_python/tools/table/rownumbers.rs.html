20impl State {
21    fn selectrows(
22        &mut self,
23        rownrs: Vec<i64>,
24        name: String,
25        log: &mut Vec<LogLine>,
26    ) -> Result<Open, Fail> {
27        if self.open.is_none() {
28            no_table(log);
29            no_table_specified(log);
30            return Err(Fail::Runtime("Unable to create table".to_string()));
31        }
32        self.synchronize()?;
33        let open = self.open.as_ref().expect("checked above");
34        let max_rows = open.nrows();
35        let mut selected = Vec::with_capacity(rownrs.len());
36        for row in rownrs {
37            let Ok(index) = usize::try_from(row) else {
38                return Err(Fail::Runtime(
39                    "rownumbers should be >= 1 and <= nrow".to_string(),
40                ));
41            };
42            if index >= max_rows {
43                return Err(Fail::Runtime(
44                    "rownumbers should be >= 1 and <= nrow".to_string(),
45                ));
46            }
47            selected.push(index);
48        }
49        let root_rows = open.translate_rows(&selected);
50        if name.is_empty() {
51            return Ok(open.view(scratch_name(), root_rows));
52        }
53        let target = std::path::absolute(Path::new(&name))
54            .map_err(|error| Fail::Runtime(error.to_string()))?;
55        for protected in [open.name.as_path(), open.root_name.as_path()]
56            .into_iter()
57            .chain(open.plain_paths.iter().map(PathBuf::as_path))
58        {
59            if same_directory(&target, protected) {
60                return Err(Fail::Runtime(format!(
61                    "Table {} already exists",
62                    target.display()
63                )));
64            }
65            refuse_nested(&target, protected)?;
66        }
67        check_target(&target)?;
68        self.refuse_open_target(&target)?;
69        if used_by_another_process(&target) {
70            return Err(Fail::Runtime(format!(
71                "Table {} is open in another process and was not replaced",
72                target.display()
73            )));
74        }
75        let root = open.root();
76        let ref_table = root.table.select_rows(&root_rows)?;
77        replace_with(&target, |staging| {
78            ref_table.save(TableOptions::new(staging))?;
79            Ok(())
80        })?;
81        drop(root);
82        Ok(open.view(target, root_rows))
83    }
84}

86#[pymethods]
87impl PyTableTool {
88    /// Return the root row numbers of the open table.
89    #[pyo3(signature = (tab = None, nbytes = None), text_signature = "($self, tab={}, nbytes=0)")]
90    fn rownumbers(
91        &self,
92        py: Python<'_>,
93        tab: Option<&Bound<'_, PyAny>>,
94        nbytes: Option<&Bound<'_, PyAny>>,
95    ) -> PyResult<Py<PyAny>> {
96        if let Some(tab) = tab {
97            tab.cast::<PyDict>()?;
98        }
99        integer(nbytes, "nbytes", 0)?;
100        self.read(py, |state, log| {
101            let rows = match state.open.as_ref() {
102                Some(open) => {
103                    let root_rows = open.root().table.root_row_numbers()?;
104                    match &open.row_map {
105                        Some(map) => map
106                            .iter()
107                            .map(|&row| {
108                                root_rows.get(row).copied().ok_or_else(|| {
109                                    Fail::Runtime(format!(
110                                        "Table row {row} is out of range ({} rows)",
111                                        root_rows.len()
112                                    ))
113                                })
114                            })
115                            .collect::<Result<Vec<_>, _>>()?,
116                        None => root_rows,
117                    }
118                }
119                None => {
120                    no_table(log);
121                    no_table_specified(log);
122                    Vec::new()
123                }
124            };
125            let count = rows.len();
126            let values = rows
127                .into_iter()
128                .map(|row| i32::try_from(row).map_or(-1, i64::from))
129                .collect();
130            Ok(CasaValue::Ints {
131                shape: vec![count],
132                values,
133            })
134        })
135    }
136
137    /// Return a new table tool referencing the requested rows of the open table.
138    #[pyo3(signature = (rownrs, name = None), text_signature = "($self, rownrs, name='')")]
139    fn selectrows(
140        &self,
141        py: Python<'_>,
142        rownrs: &Bound<'_, PyAny>,
143        name: Option<&Bound<'_, PyAny>>,
144    ) -> PyResult<Py<PyAny>> {
145        let rows = row_numbers(rownrs, "rownrs")?;
146        let name = name.map_or(Ok(String::new()), |value| text(value, "name"))?;
147        let mut lines = Vec::new();
148        let selected = py.detach(|| self.guard().selectrows(rows, name, &mut lines));
149        emit(py, LOGGER, &lines)?;
150        let tool = Self {
151            state: registry::track(State::new(Some(selected?))),
152        };
153        Ok(Py::new(py, tool)?.into_any())
154    }
155}
