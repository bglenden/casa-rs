3use casa_coordinates::{
4    Coordinate, CoordinateModel, CoordinateSystem, CoordinateType, DirectionCoordinate,
5};
6use casa_core::measures::{MeasFrame, direction::MDirection};
7
8use casa_images::ImageError;
9
10use super::SpatialShape;
11
12/// Transform a region defined on the attached image into the input's pixel plane.
13pub(super) struct Projection {
14    definition: DirectionCoordinate,
15    input: DirectionCoordinate,
16    frame: MeasFrame,
17}
18
19impl Projection {
20    pub(super) fn new(
21        origin: &CoordinateSystem,
22        input: &CoordinateSystem,
23    ) -> Result<Self, ImageError> {
24        let direction = |coordinates: &CoordinateSystem| {
25            if let Some(index) = coordinates.find_coordinate(CoordinateType::Direction)
26                && let CoordinateModel::Direction(direction) = coordinates.coordinate(index)
27                && direction.n_pixel_axes() == 2
28            {
29                return Ok(direction
30                    .clone()
31                    .with_conversion_direction_ref(direction.direction_ref()));
32            }
33            Err(ImageError::CasaSelection(
34                "Coordinate system has no direction coordinate".into(),
35            ))
36        };
37        let mut frame = MeasFrame::new();
38        if let Some(epoch) = origin.obs_info().date.as_ref() {
39            frame = frame.with_epoch(epoch.clone());
40        }
41        if let Some(position) = origin.obs_info().telescope_position.as_ref() {
42            frame = frame.with_position(position.clone());
43        }
44        Ok(Self {
45            definition: direction(origin)?,
46            input: direction(input)?,
47            frame,
48        })
49    }
50
51    fn point(&self, pixel: [f64; 2]) -> Result<[f64; 2], ImageError> {
52        let world = self.definition.pixel_to_world(pixel)?;
53        let world = if self.definition.direction_ref() == self.input.direction_ref() {
54            world
55        } else {
56            let direction =
57                MDirection::from_angles(world[0], world[1], self.definition.direction_ref())
58                    .convert_to(self.input.direction_ref(), &self.frame)
59                    .map_err(|error| ImageError::CasaSelection(error.to_string()))?;
60            [direction.longitude_rad(), direction.latitude_rad()]
61        };
62        Ok(self.input.world_to_pixel(world)?)
63    }
64
65    pub(super) fn project_shape(
66        &self,
67        shape: SpatialShape,
68        snap_roundoff: bool,
69    ) -> Result<SpatialShape, ImageError> {
70        let project_point = |pixel| {
71            self.point(pixel).map(|projected| {
72                if snap_roundoff {
73                    // A direction WCS round trip can move an integer boundary
74                    // by a few billionths of a pixel.
75                    projected.map(|value| {
76                        if (value - value.round()).abs() < 1e-8 {
77                            value.round()
78                        } else {
79                            value
80                        }
81                    })
82                } else {
83                    projected
84                }
85            })
86        };
87        let polygon = |vertices: &[[f64; 2]]| {
88            vertices
89                .iter()
90                .map(|&pixel| project_point(pixel))
91                .collect::<Result<Vec<_>, _>>()
92                .map(SpatialShape::Polygon)
93        };
94        let rectangle =
95            |center: [f64; 2], half_width: f64, half_height: f64, cos: f64, sin: f64| {
96                let vertices = [
97                    [-half_width, -half_height],
98                    [half_width, -half_height],
99                    [half_width, half_height],
100                    [-half_width, half_height],
101                ]
102                .map(|[x, y]| [center[0] + x * cos - y * sin, center[1] + x * sin + y * cos]);
103                polygon(&vertices)
104            };
105        let origin_increment = self.definition.increment();
106        let input_increment = self.input.increment();
107        let scale = |axis: usize| (origin_increment[axis] / input_increment[axis]).abs();
108        match shape {
109            SpatialShape::Box([x0, y0, x1, y1]) => {
110                let [x0, y0, x1, y1] = [x0, y0, x1, y1].map(|value| value as f64);
111                polygon(&[[x0, y0], [x1, y0], [x1, y1], [x0, y1]])
112            }
113            SpatialShape::CenterBox {
114                cx,
115                cy,
116                half_w,
117                half_h,
118            } => rectangle([cx, cy], half_w, half_h, 1.0, 0.0),
119            SpatialShape::RotBox {
120                cx,
121                cy,
122                half_w,
123                half_h,
124                cos_pa,
125                sin_pa,
126            } => rectangle([cx, cy], half_w, half_h, cos_pa, sin_pa),
127            SpatialShape::Polygon(vertices) => polygon(&vertices),
128            SpatialShape::Circle { cx, cy, rx, .. } => {
129                let [cx, cy] = project_point([cx, cy])?;
130                let radius = rx * scale(0);
131                Ok(SpatialShape::Circle {
132                    cx,
133                    cy,
134                    rx: radius,
135                    ry: radius,
136                })
137            }
138            SpatialShape::Ellipse {
139                cx,
140                cy,
141                semi_major,
142                semi_minor,
143                cos_pa,
144                sin_pa,
145            } => {
146                let [cx, cy] = project_point([cx, cy])?;
147                Ok(SpatialShape::Ellipse {
148                    cx,
149                    cy,
150                    semi_major: semi_major * scale(0),
151                    semi_minor: semi_minor * scale(1),
152                    cos_pa,
153                    sin_pa,
154                })
155            }
156        }
157    }
158}
