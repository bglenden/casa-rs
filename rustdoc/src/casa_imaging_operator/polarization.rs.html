11/// Feed basis of the selected correlations.
12#[derive(Clone, Copy, Debug, PartialEq, Eq)]
13pub enum FeedBasis {
14    /// Values are already Stokes parameters.
15    Stokes,
16    /// Orthogonal X/Y receptors.
17    Linear,
18    /// Orthogonal R/L receptors.
19    Circular,
20}
21
22/// One grid polarization plane in CASA's complex-image representation.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum GridPolarization {
25    /// One correlation (or one Stokes parameter when the data are Stokes).
26    Correlation(CorrelationType),
27    /// Stokes I gridded directly: both parallel hands feed one plane and
28    /// their weights both count in its `sumwt` (CASA `isIOnly`).
29    StokesI,
30}
31
32/// Correlation routing in both directions of the operator.
33///
34/// The adjoint grids each selected correlation into the grid plane
35/// [`PolarizationRouting::pol_map`] names and converts the complex image
36/// planes to the requested coordinates with
37/// [`PolarizationRouting::to_requested`] (CASA `StokesImageUtil::To`,
38/// real part taken). The forward transform expands requested model planes
39/// to grid planes with [`PolarizationRouting::from_requested`]
40/// (`StokesImageUtil::From`). `sumwt` of a requested Stokes plane is that of
41/// grid plane 0 (`StokesImageUtil::ToStokesSumWt`).
42#[derive(Clone, Debug, PartialEq)]
43pub struct PolarizationRouting {
44    correlations: Vec<CorrelationType>,
45    requested: Vec<PolarizationCoordinate>,
46    grid: Vec<GridPolarization>,
47    pol_map: Vec<Option<u8>>,
48    feed: FeedBasis,
49    to_requested: Vec<Complex64>,
50    to_requested_psf: Vec<Complex64>,
51    from_requested: Vec<Complex64>,
52    sumwt_source: Vec<usize>,
53}
54
55impl PolarizationRouting {
56    /// Route the selected `correlations` (block order) to the `requested`
57    /// image coordinates.
58    ///
59    /// Requested coordinates are all Stokes or all correlations, the
60    /// selected correlations share one feed basis, every grid plane CASA's
61    /// representation needs receives at least one selected correlation, and
62    /// requested correlations must be selected.
63    pub fn compile(
64        correlations: &[CorrelationType],
65        requested: &[PolarizationCoordinate],
66    ) -> Result<Self, OperatorError> {
67        let feed = feed_basis(correlations)?;
68        if requested.is_empty() {
69            return Err(OperatorError::Polarization {
70                reason: "no polarization coordinate requested",
71            });
72        }
73        if requested
74            .iter()
75            .enumerate()
76            .any(|(index, coordinate)| requested[..index].contains(coordinate))
77        {
78            return Err(OperatorError::Polarization {
79                reason: "requested polarization coordinates repeat",
80            });
81        }
82        let stokes_requested = requested.iter().all(|coordinate| is_stokes(*coordinate));
83        if !stokes_requested && requested.iter().any(|coordinate| is_stokes(*coordinate)) {
84            return Err(OperatorError::Polarization {
85                reason: "requested coordinates mix Stokes parameters and correlations",
86            });
87        }
88        let grid = if !stokes_requested {
89            requested
90                .iter()
91                .map(|coordinate| {
92                    let correlation = correlation_of(*coordinate);
93                    correlations
94                        .contains(&correlation)
95                        .then_some(GridPolarization::Correlation(correlation))
96                        .ok_or(OperatorError::Polarization {
97                            reason: "requested correlation is not selected",
98                        })
99                })
100                .collect::<Result<Vec<_>, _>>()?
101        } else if feed == FeedBasis::Stokes {
102            requested
103                .iter()
104                .map(|coordinate| {
105                    let correlation = correlation_of(*coordinate);
106                    correlations
107                        .contains(&correlation)
108                        .then_some(GridPolarization::Correlation(correlation))
109                        .ok_or(OperatorError::Polarization {
110                            reason: "requested Stokes parameter is not selected",
111                        })
112                })
113                .collect::<Result<Vec<_>, _>>()?
114        } else {
115            casa_correlation_representation(feed, requested)
116        };
117        let pol_map = correlations
118            .iter()
119            .map(|correlation| {
120                grid.iter()
121                    .position(|plane| match plane {
122                        GridPolarization::Correlation(target) => target == correlation,
123                        GridPolarization::StokesI => is_parallel_hand(*correlation),
124                    })
125                    .map(|index| index as u8)
126            })
127            .collect::<Vec<_>>();
128        if (0..grid.len()).any(|gpol| !pol_map.contains(&Some(gpol as u8))) {
129            return Err(OperatorError::Polarization {
130                reason: "selected correlations do not feed every grid polarization",
131            });
132        }
133        let to_requested = requested
134            .iter()
135            .flat_map(|coordinate| {
136                grid.iter()
137                    .map(|plane| to_coefficient(feed, *coordinate, *plane))
138            })
139            .collect();
140        let to_requested_psf = requested
141            .iter()
142            .flat_map(|coordinate| {
143                grid.iter()
144                    .map(|plane| psf_coefficient(feed, requested, *coordinate, *plane))
145            })
146            .collect();
147        let from_requested = grid
148            .iter()
149            .flat_map(|plane| {
150                requested
151                    .iter()
152                    .map(|coordinate| from_coefficient(feed, *plane, *coordinate))
153            })
154            .collect();
155        let sumwt_source = requested
156            .iter()
157            .map(|coordinate| {
158                if stokes_requested {
159                    0
160                } else {
161                    grid.iter()
162                        .position(|plane| {
163                            *plane == GridPolarization::Correlation(correlation_of(*coordinate))
164                        })
165                        .expect("requested correlations are grid planes")
166                }
167            })
168            .collect();
169        Ok(Self {
170            correlations: correlations.to_vec(),
171            requested: requested.to_vec(),
172            grid,
173            pol_map,
174            feed,
175            to_requested,
176            to_requested_psf,
177            from_requested,
178            sumwt_source,
179        })
180    }
181
182    /// Selected correlations in block order.
183    #[must_use]
184    pub fn correlations(&self) -> &[CorrelationType] {
185        &self.correlations
186    }
187
188    /// Requested image coordinates in output order.
189    #[must_use]
190    pub fn requested(&self) -> &[PolarizationCoordinate] {
191        &self.requested
192    }
193
194    /// Grid polarization planes.
195    #[must_use]
196    pub fn grid(&self) -> &[GridPolarization] {
197        &self.grid
198    }
199
200    /// Number of grid polarization planes.
201    #[must_use]
202    pub fn grid_pols(&self) -> usize {
203        self.grid.len()
204    }
205
206    /// Grid plane fed by each selected correlation (`None`: not gridded).
207    #[must_use]
208    pub fn pol_map(&self) -> &[Option<u8>] {
209        &self.pol_map
210    }
211
212    /// Feed basis of the selected correlations.
213    #[must_use]
214    pub const fn feed(&self) -> FeedBasis {
215        self.feed
216    }
217
218    /// Coefficient of grid plane `gpol` in requested plane `requested`; the
219    /// requested image is the real part of the sum.
220    #[must_use]
221    pub fn to_requested(&self, requested: usize, gpol: usize) -> Complex64 {
222        self.to_requested[requested * self.grid.len() + gpol]
223    }
224
225    /// Coefficient of grid plane `gpol` in the PSF image of requested plane
226    /// `requested` (CASA `StokesImageUtil::ToStokesPSF`): with one or two
227    /// requested Stokes parameters every plane takes the parallel-hand sum,
228    /// or the cross-hand sum for U and V on linear feeds and Q and U on
229    /// circular feeds; with three or four, every plane takes the first
230    /// parameter's conversion; a single-correlation grid, a Stokes grid or a
231    /// request with a non-Stokes coordinate takes the data conversion.
232    ///
233    /// The operator forms its weight image with these coefficients too,
234    /// whereas CASA applies `ToStokesPSF` to the PSF only; no kernel set
235    /// with a weight image is installed before the AW and mosaic operators.
236    #[must_use]
237    pub fn to_requested_psf(&self, requested: usize, gpol: usize) -> Complex64 {
238        self.to_requested_psf[requested * self.grid.len() + gpol]
239    }
240
241    /// Coefficient of requested model plane `requested` in grid plane `gpol`.
242    #[must_use]
243    pub fn from_requested(&self, gpol: usize, requested: usize) -> Complex64 {
244        self.from_requested[gpol * self.requested.len() + requested]
245    }
246
247    /// Grid plane whose `sumwt` normalises requested plane `requested`.
248    #[must_use]
249    pub fn sumwt_source(&self, requested: usize) -> usize {
250        self.sumwt_source[requested]
251    }
252}

254fn feed_basis(correlations: &[CorrelationType]) -> Result<FeedBasis, OperatorError> {
255    let Some(first) = correlations.first() else {
256        return Err(OperatorError::Polarization {
257            reason: "no correlation selected",
258        });
259    };
260    let feed = basis_of(*first)?;
261    for (index, correlation) in correlations.iter().enumerate() {
262        if basis_of(*correlation)? != feed {
263            return Err(OperatorError::Polarization {
264                reason: "selected correlations mix feed bases",
265            });
266        }
267        if correlations[..index].contains(correlation) {
268            return Err(OperatorError::Polarization {
269                reason: "selected correlations repeat",
270            });
271        }
272    }
273    Ok(feed)
274}
275
276fn basis_of(correlation: CorrelationType) -> Result<FeedBasis, OperatorError> {
277    match correlation {
278        CorrelationType::StokesI
279        | CorrelationType::StokesQ
280        | CorrelationType::StokesU
281        | CorrelationType::StokesV => Ok(FeedBasis::Stokes),
282        CorrelationType::LinearXx
283        | CorrelationType::LinearXy
284        | CorrelationType::LinearYx
285        | CorrelationType::LinearYy => Ok(FeedBasis::Linear),
286        CorrelationType::CircularRr
287        | CorrelationType::CircularRl
288        | CorrelationType::CircularLr
289        | CorrelationType::CircularLl => Ok(FeedBasis::Circular),
290        _ => Err(OperatorError::Polarization {
291            reason: "selected correlation is not a Stokes, linear or circular product",
292        }),
293    }
294}

296const fn is_stokes(coordinate: PolarizationCoordinate) -> bool {
297    matches!(
298        coordinate,
299        PolarizationCoordinate::StokesI
300            | PolarizationCoordinate::StokesQ
301            | PolarizationCoordinate::StokesU
302            | PolarizationCoordinate::StokesV
303    )
304}
305
306const fn is_parallel_hand(correlation: CorrelationType) -> bool {
307    matches!(
308        correlation,
309        CorrelationType::LinearXx
310            | CorrelationType::LinearYy
311            | CorrelationType::CircularRr
312            | CorrelationType::CircularLl
313    )
314}
315
316const fn correlation_of(coordinate: PolarizationCoordinate) -> CorrelationType {
317    match coordinate {
318        PolarizationCoordinate::StokesI => CorrelationType::StokesI,
319        PolarizationCoordinate::StokesQ => CorrelationType::StokesQ,
320        PolarizationCoordinate::StokesU => CorrelationType::StokesU,
321        PolarizationCoordinate::StokesV => CorrelationType::StokesV,
322        PolarizationCoordinate::LinearXx => CorrelationType::LinearXx,
323        PolarizationCoordinate::LinearXy => CorrelationType::LinearXy,
324        PolarizationCoordinate::LinearYx => CorrelationType::LinearYx,
325        PolarizationCoordinate::LinearYy => CorrelationType::LinearYy,
326        PolarizationCoordinate::CircularRr => CorrelationType::CircularRr,
327        PolarizationCoordinate::CircularRl => CorrelationType::CircularRl,
328        PolarizationCoordinate::CircularLr => CorrelationType::CircularLr,
329        PolarizationCoordinate::CircularLl => CorrelationType::CircularLl,
330    }
331}

333/// CASA `StokesImageUtil::changeLabelsStokesToCorrStokes`: the complex
334/// grid planes that serve a Stokes request on linear or circular feeds.
335fn casa_correlation_representation(
336    feed: FeedBasis,
337    requested: &[PolarizationCoordinate],
338) -> Vec<GridPolarization> {
339    use CorrelationType::{
340        CircularLl, CircularLr, CircularRl, CircularRr, LinearXx, LinearXy, LinearYx, LinearYy,
341    };
342    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
343    let planes = |list: &[CorrelationType]| {
344        list.iter()
345            .map(|correlation| GridPolarization::Correlation(*correlation))
346            .collect::<Vec<_>>()
347    };
348    let all = match feed {
349        FeedBasis::Linear => planes(&[LinearXx, LinearXy, LinearYx, LinearYy]),
350        FeedBasis::Circular | FeedBasis::Stokes => {
351            planes(&[CircularRr, CircularRl, CircularLr, CircularLl])
352        }
353    };
354    match (feed, requested) {
355        (_, [StokesI]) => vec![GridPolarization::StokesI],
356        (FeedBasis::Circular, [StokesQ] | [StokesU] | [StokesQ, StokesU]) => {
357            planes(&[CircularRl, CircularLr])
358        }
359        (FeedBasis::Circular, [StokesV] | [StokesI, StokesV]) => planes(&[CircularRr, CircularLl]),
360        (FeedBasis::Linear, [StokesQ] | [StokesI, StokesQ]) => planes(&[LinearXx, LinearYy]),
361        (FeedBasis::Linear, [StokesU] | [StokesV] | [StokesU, StokesV]) => {
362            planes(&[LinearXy, LinearYx])
363        }
364        _ => all,
365    }
366}

368/// `StokesImageUtil::To` (`CStokesVector::applySlinInv` / `applyScircInv`):
369/// the coefficient of one grid plane in one requested plane.
370fn to_coefficient(
371    feed: FeedBasis,
372    requested: PolarizationCoordinate,
373    plane: GridPolarization,
374) -> Complex64 {
375    use CorrelationType::{
376        CircularLl, CircularLr, CircularRl, CircularRr, LinearXx, LinearXy, LinearYx, LinearYy,
377    };
378    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
379    let one = Complex64::new(1.0, 0.0);
380    let half = Complex64::new(0.5, 0.0);
381    let half_i = Complex64::new(0.0, 0.5);
382    let correlation = match plane {
383        GridPolarization::StokesI => {
384            return if requested == StokesI {
385                one
386            } else {
387                Complex64::default()
388            };
389        }
390        GridPolarization::Correlation(correlation) => correlation,
391    };
392    if !is_stokes(requested) || feed == FeedBasis::Stokes {
393        return if correlation_of(requested) == correlation {
394            one
395        } else {
396            Complex64::default()
397        };
398    }
399    match (feed, requested, correlation) {
400        (FeedBasis::Linear, StokesI, LinearXx | LinearYy)
401        | (FeedBasis::Linear, StokesQ, LinearXx)
402        | (FeedBasis::Linear, StokesU, LinearXy | LinearYx)
403        | (FeedBasis::Circular, StokesI, CircularRr | CircularLl)
404        | (FeedBasis::Circular, StokesQ, CircularRl | CircularLr)
405        | (FeedBasis::Circular, StokesV, CircularRr) => half,
406        (FeedBasis::Linear, StokesQ, LinearYy) | (FeedBasis::Circular, StokesV, CircularLl) => {
407            -half
408        }
409        (FeedBasis::Linear, StokesV, LinearYx) | (FeedBasis::Circular, StokesU, CircularLr) => {
410            half_i
411        }
412        (FeedBasis::Linear, StokesV, LinearXy) | (FeedBasis::Circular, StokesU, CircularRl) => {
413            -half_i
414        }
415        _ => Complex64::default(),
416    }
417}

419/// `StokesImageUtil::ToStokesPSF` on physical feeds: with one or two
420/// requested Stokes parameters, every PSF plane comes from one Stokes row
421/// of the data conversion (CASA applies the cross-hand rule to both planes
422/// whenever either requested parameter is a cross-hand one); with more, every
423/// plane is the first requested parameter's PSF (`sv(outMap(0))`), since the
424/// grid holds more than one correlation.
425fn psf_coefficient(
426    feed: FeedBasis,
427    requested_all: &[PolarizationCoordinate],
428    requested: PolarizationCoordinate,
429    plane: GridPolarization,
430) -> Complex64 {
431    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
432    if feed == FeedBasis::Stokes
433        || plane == GridPolarization::StokesI
434        || !requested_all
435            .iter()
436            .all(|coordinate| is_stokes(*coordinate))
437    {
438        return to_coefficient(feed, requested, plane);
439    }
440    if requested_all.len() > 2 {
441        return to_coefficient(feed, requested_all[0], plane);
442    }
443    let cross_hand = |coordinate: &PolarizationCoordinate| match feed {
444        FeedBasis::Linear => matches!(coordinate, StokesU | StokesV),
445        FeedBasis::Circular | FeedBasis::Stokes => matches!(coordinate, StokesQ | StokesU),
446    };
447    let row = if requested_all.iter().any(cross_hand) {
448        match feed {
449            FeedBasis::Linear => StokesU,
450            FeedBasis::Circular | FeedBasis::Stokes => StokesQ,
451        }
452    } else {
453        StokesI
454    };
455    to_coefficient(feed, row, plane)
456}

458/// `StokesImageUtil::From` (`CStokesVector::applySlin` / `applyScirc`): the
459/// coefficient of one requested model plane in one grid plane.
460fn from_coefficient(
461    feed: FeedBasis,
462    plane: GridPolarization,
463    requested: PolarizationCoordinate,
464) -> Complex64 {
465    use CorrelationType::{
466        CircularLl, CircularLr, CircularRl, CircularRr, LinearXx, LinearXy, LinearYx, LinearYy,
467    };
468    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
469    let one = Complex64::new(1.0, 0.0);
470    let i = Complex64::new(0.0, 1.0);
471    let correlation = match plane {
472        GridPolarization::StokesI => {
473            return if requested == StokesI {
474                one
475            } else {
476                Complex64::default()
477            };
478        }
479        GridPolarization::Correlation(correlation) => correlation,
480    };
481    if !is_stokes(requested) || feed == FeedBasis::Stokes {
482        return if correlation_of(requested) == correlation {
483            one
484        } else {
485            Complex64::default()
486        };
487    }
488    match (feed, correlation, requested) {
489        (FeedBasis::Linear, LinearXx, StokesI | StokesQ)
490        | (FeedBasis::Linear, LinearYy, StokesI)
491        | (FeedBasis::Linear, LinearXy | LinearYx, StokesU)
492        | (FeedBasis::Circular, CircularRr, StokesI | StokesV)
493        | (FeedBasis::Circular, CircularLl, StokesI)
494        | (FeedBasis::Circular, CircularRl | CircularLr, StokesQ) => one,
495        (FeedBasis::Linear, LinearYy, StokesQ) | (FeedBasis::Circular, CircularLl, StokesV) => -one,
496        (FeedBasis::Linear, LinearXy, StokesV) | (FeedBasis::Circular, CircularRl, StokesU) => i,
497        (FeedBasis::Linear, LinearYx, StokesV) | (FeedBasis::Circular, CircularLr, StokesU) => -i,
498        _ => Complex64::default(),
499    }
