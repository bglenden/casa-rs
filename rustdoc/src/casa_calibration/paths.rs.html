2//! Resolve existing ancestors as well as paths of products not yet created.
3use std::collections::VecDeque;
4use std::io;
5use std::path::{Component, Path, PathBuf};
6
7/// Normalize a product path by resolving existing symlinks and `.`/`..` components.
8pub fn normalized(path: &Path) -> io::Result<PathBuf> {
9    let absolute = if path.is_absolute() {
10        path.to_path_buf()
11    } else {
12        std::env::current_dir()?.join(path)
13    };
14    let mut pending: VecDeque<_> = absolute
15        .components()
16        .map(|c| c.as_os_str().to_owned())
17        .collect();
18    let mut result = PathBuf::new();
19    let mut links = 0;
20    while let Some(part) = pending.pop_front() {
21        match Path::new(&part)
22            .components()
23            .next()
24            .expect("path component")
25        {
26            Component::ParentDir => {
27                result.pop();
28            }
29            Component::CurDir => {}
30            Component::RootDir | Component::Prefix(_) => result.push(&part),
31            Component::Normal(_) => {
32                result.push(&part);
33                match std::fs::symlink_metadata(&result) {
34                    Ok(meta) if meta.is_symlink() => {
35                        links += 1;
36                        if links > 40 {
37                            return Err(io::Error::other(
38                                "too many symbolic links in scientific product path",
39                            ));
40                        }
41                        let target = std::fs::read_link(&result)?;
42                        result.pop();
43                        for component in target.components().rev() {
44                            pending.push_front(component.as_os_str().to_owned());
45                        }
46                    }
47                    Ok(_) => result = result.canonicalize()?,
48                    Err(e) if e.kind() == io::ErrorKind::NotFound => {}
49                    Err(e) => return Err(e),
50                }
51            }
52        }
53    }
54    Ok(result)
55}
56
57/// Return whether either normalized path is a prefix of the other.
58pub fn overlap(left: &Path, right: &Path) -> io::Result<bool> {
59    let left = normalized(left)?;
60    let right = normalized(right)?;
61    Ok(left.starts_with(&right) || right.starts_with(&left))
62}

64/// The plain table whose record owns inherited subtable links.
65pub fn subtable_owner(
66    table: &casa_tables::Table,
67    path: &Path,
68) -> Result<PathBuf, casa_tables::TableError> {
69    if table.is_reference_table()?
70        && let Some(plain) = table.cache_plain_table_paths()?.into_iter().next()
71    {
72        return Ok(plain);
73    }
74    Ok(path.to_path_buf())
75}

77/// Copy a table tree, keeping its physical storage and relative links.
78pub(crate) fn copy_dir_recursive(source: &Path, destination: &Path) -> std::io::Result<()> {
79    copy_table_tree(source, destination, false)
80}

82/// Copy table files by value before editing, including regular file links.
83/// Directory links remain available for the caller to resolve as subtables.
84pub(crate) fn copy_editable_table_tree(source: &Path, destination: &Path) -> io::Result<()> {
85    copy_table_tree(source, destination, true)
86}
87
88fn copy_table_tree(source: &Path, destination: &Path, detach_file_links: bool) -> io::Result<()> {
89    std::fs::create_dir_all(destination)?;
90    for entry in std::fs::read_dir(source)? {
91        let entry = entry?;
92        let source_path = entry.path();
93        let destination_path = destination.join(entry.file_name());
94        let file_type = entry.file_type()?;
95        if file_type.is_dir() {
96            copy_table_tree(&source_path, &destination_path, detach_file_links)?;
97        } else if file_type.is_file()
98            || (detach_file_links && file_type.is_symlink() && source_path.is_file())
99        {
100            std::fs::copy(&source_path, &destination_path)?;
101        } else if file_type.is_symlink() {
102            let target = std::fs::read_link(&source_path)?;
103            std::os::unix::fs::symlink(target, destination_path)?;
104        }
105    }
106    Ok(())
107}

112
113    #[test]
114    fn dangling_report_link_resolves_to_the_future_product() {
115        let dir = tempfile::tempdir().unwrap();
116        let product = dir.path().join("future.cal");
117        let report = dir.path().join("report");
118        std::os::unix::fs::symlink(product.join("table.dat"), &report).unwrap();
119        assert!(overlap(&report, &product).unwrap());
120    }
121
122    #[test]
123    fn a_parent_component_follows_the_symlink_target() {
124        let dir = tempfile::tempdir().unwrap();
125        let product = dir.path().join("existing.cal");
126        std::fs::create_dir_all(product.join("ANTENNA")).unwrap();
127        std::fs::write(product.join("table.dat"), "keep").unwrap();
128        let alias = dir.path().join("alias");
129        std::os::unix::fs::symlink(product.join("ANTENNA"), &alias).unwrap();
130        assert!(overlap(&alias.join("../table.dat"), &product).unwrap());
131    }
