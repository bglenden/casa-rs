30/// The calibrator's field and source name.
31const CALIBRATOR: &str = "phase calibrator";

33/// The MAIN rows read at once to find the longest baseline.
34const UVW_ROWS: usize = 1 << 16;

36/// Observe the pointings and predict the model into `<stem>.ms` (or
37/// `<stem>.sd.ms`), then set its rest frequencies to the model's.
38pub(super) fn observe(
39    run: &mut Run<'_>,
40    sky: &Sky,
41    array: &Array,
42    pointings: &Pointings,
43    calibrator: Option<&Calibrator>,
44) -> Result<(), SimobserveError> {
45    let model = &sky.model;
46    let ms = run.measurement_set();
47    if run.parameters.verbose {
48        tracing::warn!("preparing empty measurement set");
49    } else {
50        tracing::info!("preparing empty measurement set");
51    }
52    let band = band_name(qa::value_in(&model.specrefval, "GHz")?);
53    let schedule = schedule(run, pointings, calibrator.is_some())?;
54    if entry_exists(&ms) && !run.parameters.overwrite {
55        return Err(SimobserveError::raised(format!(
56            "measurement set {} already exists and user does not wish to overwrite",
57            ms.display()
58        )));
59    }
60    // A MeasurementSet of the project's own stays until the new one
61    // replaces it at `close`.
62    make_way(&ms)?;
63    let mut sm = open_session(run, &ms);
64    set_up(run, &mut sm, sky, array, &band)?;
65    let integration_s = qa::value_in(&qa::read(&pointings.integration)?, "s")?;
66    sm.set_times(integration_s, true, schedule.reference.clone())?;
67    set_fields(run, &mut sm, pointings, calibrator)?;
68    observe_schedule(run, &mut sm, &schedule, &band)?;
69    // CASA selects the map's fields alone, which leaves the calibrator's
70    // scans unpredicted; casa-rs predicts every scan.
71    let fields = (0..pointings.pointings.len() + usize::from(calibrator.is_some()))
72        .map(|field| field as i64)
73        .collect::<Vec<_>>();
74    sm.set_data(&[], &fields, "")?;
75    set_prediction(run, &mut sm, sky, array, pointings.pointings.len())?;
76    tracing::info!("done setting up observations (blank visibilities)");
77    predict(run, &mut sm, sky, calibrator)?;
78    sm.close()?;
79    run.wrote(&ms);
80    tracing::info!("generation of measurement set {} complete", ms.display());
81    let hz = qa::value_in(&model.specrefval, "Hz")?;
82    rewrite_column(&ms.join("SPECTRAL_WINDOW"), "REF_FREQUENCY", hz)?;
83    rewrite_column(&ms.join("SOURCE"), "REST_FREQUENCY", hz)?;
84    check_baselines(&ms, run.uvmode)
85}

87/// `setconfig`, `setspwindow`, `setfeed`, `setlimits` (only when
88/// `verbose`) and `setauto`.
89fn set_up(
90    run: &Run<'_>,
91    sm: &mut Simulator,
92    sky: &Sky,
93    array: &Array,
94    band: &str,
95) -> Result<(), SimobserveError> {
96    let parameters = run.parameters;
97    let model = &sky.model;
98    let config = &array.config;
99    let telescope = config.telescope.as_str();
100    let mount = if matches!(telescope, "DRAO" | "WSRT") {
101        "EQUATORIAL"
102    } else {
103        "alt-az"
104    };
105    sm.set_config(&ArrayConfig {
106        telescope: telescope.to_string(),
107        x: config.x.clone(),
108        y: config.y.clone(),
109        z: config.z.clone(),
110        dish_diameter: config.dish_diameter.clone(),
111        offset: vec![0.0],
112        mount: vec![mount.to_string()],
113        antenna_names: config.antenna_names.clone(),
114        pad_names: config.pad_names.clone(),
115        coordinate_system: CoordinateSystem::Global,
116    })?;
117    let (correlations, feed) = if telescope.to_uppercase().contains("VLA") {
118        (["RR", "RL", "LR", "LL"], "perfect R L")
119    } else {
120        (["XX", "XY", "YX", "YY"], "perfect X Y")
121    };
122    let leakage = parameters.leakage;
123    let stokes = if leakage > 0. && run.uvmode {
124        tracing::info!(
125            "Parameter 'leakage' is {}. Simulating FULL-STOKES.",
126            python(leakage)
127        );
128        correlations.join(" ")
129    } else if model.stokes == "IQUV" && run.uvmode {
130        tracing::info!("Model Stokes is {}. Simulating FULL-STOKES.", model.stokes);
131        correlations.join(" ")
132    } else {
133        if leakage > 0. {
134            tracing::info!(
135                "Parameter 'leakage' is {} but we are not in interferometric mode. Will ignore.",
136                python(leakage)
137            );
138        }
139        format!("{} {}", correlations[0], correlations[3])
140    };
141    tracing::info!(" Simulated correlation products: {stokes}");
142    // The start is the centre of the first channel, and both it and the
143    // width reach `setspwindow` as `qa.tos` writes them.
144    let start = qa::sub(
145        &model.specrefval,
146        &qa::mul(&model.width, &qa::number(model.specrefpix))?,
147    )?;
148    let width_hz = written_hz(&model.width)?;
149    let frame: FrequencyRef = parameters.outframe.parse().map_err(|_| {
150        SimobserveError::raised(format!(
151            "Unknown frequency frame outframe='{}'",
152            parameters.outframe
153        ))
154    })?;
155    sm.set_spectral_window(&SpectralWindowSpec {
156        name: band.to_string(),
157        start_frequency_hz: written_hz(&start)?,
158        channel_width_hz: width_hz,
159        resolution_hz: width_hz,
160        frame,
161        channel_count: model.nchan,
162        stokes,
163    })?;
164    sm.set_feed(feed, &[0.0], &[0.0], &["R".to_string()])?;
165    if parameters.verbose {
166        tracing::info!(" spectral window set at {}", tos(&model.specrefval, 9));
167        sm.set_limits(0.01, 10_f64.to_radians());
168        tracing::info!(" Setting shadowlimit to 0.01 and elevationlimit to 10 deg");
169    } else {
170        tracing::info!(" shadowlimit and elevationlimit left at their default values");
171    }
172    // A single dish's autocorrelations are its data.
173    sm.set_auto(if run.uvmode { 0.0 } else { 1.0 });
174    Ok(())
175}

177/// A frequency as `setspwindow` reads `qa.tos` of it, Hz.
178fn written_hz(frequency: &Quantity) -> Result<f64, SimobserveError> {
179    Ok(qa::value_in(&qa::read(&tos(frequency, 9))?, "Hz")?)
180}

182/// A direction string as the session's measure.
183fn direction(text: &str) -> Result<MDirection, SimobserveError> {
184    let (refer, [longitude, latitude]) = direction_measure(text)?;
185    Ok(MDirection::from_angles(longitude, latitude, refer))
186}

188/// `setfield` of each pointing, `<stem>_<k>`, and of the calibrator.
189fn set_fields(
190    run: &Run<'_>,
191    sm: &mut Simulator,
192    pointings: &Pointings,
193    calibrator: Option<&Calibrator>,
194) -> Result<(), SimobserveError> {
195    for (field, pointing) in pointings.pointings.iter().enumerate() {
196        sm.set_field(&field_name(run, field), &direction(pointing)?, "OBJ", 0.0)?;
197    }
198    if let Some(calibrator) = calibrator {
199        sm.set_field(CALIBRATOR, &direction(&calibrator.direction)?, "C", 0.0)?;
200    }
201    Ok(())
202}

206    format!("{}_{field}", run.stem)

209/// The schedule's steps as `observemany` and `observe` calls.
210fn observe_schedule(
211    run: &Run<'_>,
212    sm: &mut Simulator,
213    schedule: &Schedule,
214    band: &str,
215) -> Result<(), SimobserveError> {
216    let state = |obs_mode: &str| StateSpec {
217        sig: true,
218        reference: false,
219        cal: 0.0,
220        load: 0.0,
221        sub_scan: 0,
222        obs_mode: obs_mode.to_string(),
223    };
224    for step in &schedule.steps {
225        match step {
226            Step::Targets(targets) => sm.observe_many(&ObserveManySpec {
227                pointings: targets
228                    .iter()
229                    .map(|target| Pointing {
230                        source: field_name(run, target.field),
231                        start_s: target.start_s,
232                        stop_s: target.stop_s,
233                        direction: None,
234                    })
235                    .collect(),
236                spectral_window: band.to_string(),
237                add_observation: false,
238                state: state("OBSERVE_TARGET#ON_SOURCE"),
239                observer: "CASA simulator".to_string(),
240                project: run.stem.clone(),
241            })?,
242            Step::Calibrator { start_s, stop_s } => sm.observe(&ObserveSpec {
243                source: CALIBRATOR.to_string(),
244                spectral_window: band.to_string(),
245                start_s: *start_s,
246                stop_s: *stop_s,
247                add_observation: false,
248                state: state("CALIBRATE_PHASE.ON_SOURCE"),
249                observer: "CASA simulator".to_string(),
250                project: run.stem.clone(),
251            })?,
252        }
253    }
254    Ok(())
255}

257/// `setvp` for an interferometer (or a component list), the mosaic
258/// machine for a heterogeneous array, and `SDGrid` for a single dish.
259fn set_prediction(
260    run: &Run<'_>,
261    sm: &mut Simulator,
262    sky: &Sky,
263    array: &Array,
264    fields: usize,
265) -> Result<(), SimobserveError> {
266    if run.uvmode || sky.components_only {
267        sm.set_vp(&VoltagePatternSpec {
268            dovp: true,
269            use_default_vp: false,
270            ..VoltagePatternSpec::default()
271        })?;
272        let diameters = &array.config.dish_diameter;
273        let telescope = array.config.telescope.as_str();
274        if diameters.windows(2).any(|pair| pair[0] != pair[1]) {
275            if fields <= 1 {
276                return Err(SimobserveError::raised(
277                    "Heterogeneous array only supported for mosaics (nfld>1), and make sure that \
278                     your image is larger than the primary beam or results may be unstable",
279                ));
280            }
281            sm.set_options(&PredictionOptions {
282                ftmachine: "mosaic".to_string(),
283                ..PredictionOptions::default()
284            })?;
285        } else {
286            // CASA predicts with the telescope's beam whatever the
287            // configuration's diameters, and says so.
288            let diameter = diameters.first().copied().unwrap_or_default();
289            match telescope {
290                "ALMA" if !(10.0..=13.0).contains(&diameter) => tracing::warn!(
291                    "Diameter = {diameter:.6} is inconsistent with telescope=ALMA in the \
292                     configuration file. *12m ALMA PB will be used*. Use observatory='ACA' in the \
293                     configuration file to use a 7m diameter."
294                ),
295                "ACA" if !(6.0..=7.5).contains(&diameter) => tracing::warn!(
296                    "Diameter = {diameter:.6} is inconsistent with telescope=ALMA in the \
297                     configuration file. *7m ACA PB will be used*"
298                ),
299                "ALMA" | "ACA" => {}
300                _ => tracing::info!(
301                    "Note: diameters in configuration file will not be used - PB for {telescope} \
302                     will be used"
303                ),
304            }
305        }
306    }
307    if !run.uvmode {
308        sm.set_options(&PredictionOptions {
309            ftmachine: "sd".to_string(),
310            grid_function: "pb".to_string(),
311            ..PredictionOptions::default()
312        })?;
313    }
314    Ok(())
315}

317/// `predict` of the model image (none for a component list alone) with
318/// `complist` and the calibrator's list.
319fn predict(
320    run: &Run<'_>,
321    sm: &mut Simulator,
322    sky: &Sky,
323    calibrator: Option<&Calibrator>,
324) -> Result<(), SimobserveError> {
325    let mut lists: Vec<PathBuf> = Vec::new();
326    if !run.parameters.complist.is_empty() {
327        lists.push(PathBuf::from(&run.parameters.complist));
328    }
329    lists.extend(calibrator.map(|calibrator| calibrator.list.clone()));
330    let named = lists
331        .iter()
332        .map(|list| list.display().to_string())
333        .collect::<Vec<_>>()
334        .join(",");
335    let message = match &sky.image {
336        Some(image) if named.is_empty() => format!("predicting from {}", image.display()),
337        Some(image) => format!("predicting from {} and {named}", image.display()),
338        None => format!("predicting from {named}"),
339    };
340    if run.parameters.verbose {
341        tracing::warn!("{message}");
342    } else {
343        tracing::info!("{message}");
344    }
345    sm.predict(&PredictSpec {
346        images: sky.image.iter().cloned().collect(),
347        component_lists: lists,
348        incremental: false,
349    })?;
350    Ok(())
351}

353/// Set every cell of the `Double` column `column` of `table` to `value`,
354/// as CASA's task sets the rest frequencies, which the simulator writes as
355/// the first channel's frequency, to the model's.
356fn rewrite_column(table: &Path, column: &str, value: f64) -> Result<(), SimobserveError> {
357    let failed = |reason: &dyn std::fmt::Display| SimobserveError::Table {
358        table: table.to_path_buf(),
359        reason: format!("{column}: {reason}"),
360    };
361    let mut handle = Table::open(TableOptions::new(table)).map_err(|error| failed(&error))?;
362    for row in 0..handle.row_count() {
363        let cell = handle
364            .cell_accessor(row, column)
365            .map_err(|error| failed(&error))?;
366        let new = match cell.value().map_err(|error| failed(&error))? {
367            Some(Value::Array(ArrayValue::Float64(values))) => {
368                Value::Array(ArrayValue::Float64(values.mapv(|_| value)))
369            }
370            _ => Value::Scalar(ScalarValue::Float64(value)),
371        };
372        handle
373            .cell_accessor_mut(row, column)
374            .and_then(|mut cell| cell.set(new))
375            .map_err(|error| failed(&error))?;
376    }
377    handle
378        .prepare_write()
379        .save_selected_columns(&[column])
380        .map_err(|error| failed(&error))
381}

383/// CASA's check of the observation against the mode: an interferometer's
384/// MeasurementSet must have a positive `u` or `v` (a baseline), a single
385/// dish's none.
386fn check_baselines(ms: &Path, uvmode: bool) -> Result<(), SimobserveError> {
387    let failed = |reason: &dyn std::fmt::Display| SimobserveError::Table {
388        table: ms.to_path_buf(),
389        reason: format!("UVW: {reason}"),
390    };
391    let set = MeasurementSet::open(ms).map_err(|error| failed(&error))?;
392    let main = set.main_table();
393    let rows = main.row_count();
394    if rows == 0 {
395        return Err(SimobserveError::raised(format!(
396            "no rows were observed in {}: is the source above the elevation limit?",
397            ms.display()
398        )));
399    }
400    let column = main
401        .column_accessor("UVW")
402        .map_err(|error| failed(&error))?;
403    let mut longest = f64::NEG_INFINITY;
404    let mut first = 0;
405    while first < rows {
406        let chunk: Vec<usize> = (first..rows.min(first + UVW_ROWS)).collect();
407        for cell in column
408            .array_cells_owned(&chunk)
409            .map_err(|error| failed(&error))?
410        {
411            if let Some(ArrayValue::Float64(uvw)) = cell {
412                longest = uvw.iter().take(2).copied().fold(longest, f64::max);
413            }
414        }
415        first += UVW_ROWS;
416    }
417    if longest > 0. && !uvmode {
418        return Err(SimobserveError::raised(format!(
419            "An single observation is requested but CROSS-correlation in {}",
420            ms.display()
421        )));
422    }
423    if longest <= 0. && uvmode {
424        return Err(SimobserveError::raised(format!(
425            "An interferometer observation is requested but only AUTO-correlation in {}",
426            ms.display()
427        )));
428    }
429    Ok(())
