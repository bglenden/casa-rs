2//! What CASA clears from a MeasurementSet before it writes a scratch
3//! column, so that the column casa-rs writes is the one CASA reads (CASA
4//! 6.7.5).
5//!
6//! - **Cached sort keywords.** CASA's iterators may cache a pre-sorted
7//!   reference table of MAIN and record it in MAIN's keywords as
8//!   `SORTED_TABLE`, with the columns it is sorted by as `SORT_COLUMNS`
9//!   (`MSIter2`). A sorted table made before a column was added does not
10//!   see it, so adding `MODEL_DATA` or `CORRECTED_DATA` removes both
11//!   keywords (`VisSetUtil::addScrCols`; the table itself is left, as CASA
12//!   leaves it). The MS definition's own `SORTED_TABLES` is untouched.
13//! - **The virtual model.** CASA can hold a field's model as a recipe
14//!   instead of a column: `setjy` (`usescratch=False`) and `tclean`
15//!   (`savemodel='virtual'`) write one, and CASA's iterator prefers it to
16//!   `MODEL_DATA` (`VisibilityIteratorImpl2::visibilityModel`). A task that
17//!   writes `MODEL_DATA` makes the column "the exclusive origin for model
18//!   visibilities", whether or not it adds it: it clears every field's
19//!   virtual model first. `addScrCols` does so when it adds the column
20//!   (`remOTFModel`, `VisModelData::clearModel` of
21//!   `synthesis/TransformMachines/VisModelData.cc`), and the imagers do so
22//!   whenever they write it (`SynthesisImagerVi2::selectData`,
23//!   `Imager::setjy` and `ft`, `refim::VisModelData::clearModel` of
24//!   `TransformMachines2`). The two `clearModel` bodies are the same.
25//!
26//! A field's virtual model is a keyword `definedmodel_field_<id>` naming a
27//! model key, in SOURCE's keywords when SOURCE has rows and MAIN's
28//! otherwise. Under SOURCE the model key is a keyword holding the SOURCE
29//! row whose `SOURCE_MODEL` cell holds the model record; under MAIN it is a
30//! keyword holding the record itself. Clearing a field removes both
31//! keywords wherever they are and empties the `SOURCE_MODEL` cell, which
32//! is what CASA leaves (measured with `setjy` then `delmod` on CASA 6.7.5).
33//! A pointer under SOURCE naming a key held in MAIN is removed with the
34//! key, where CASA keeps the pointer; either way the field has no model.
35//!
36//! `clearModel` also deletes the model images a gridder's record names
37//! (`ft_<k>.container.diskimage`), wherever they lie, and every `FT_MODEL*`
38//! table in the MeasurementSet and its SOURCE, where `tclean` writes them.
39//! casa-rs deletes only the `FT_MODEL*` tables, so that a path in a record
40//! never deletes anything else. A multi-MS is cleared as one table, not
41//! part by part as CASA clears it.
42

57/// Clear what CASA clears before it writes scratch column `column`
58/// (`MODEL_DATA` or `CORRECTED_DATA`) to `ms`, adding it when `adds`:
59/// for an added column the cached sort keywords, and for `MODEL_DATA`,
60/// added or not, every field's virtual model, as the module states. Every
61/// path that adds either column to an existing MeasurementSet, or writes
62/// `MODEL_DATA`, calls it before it changes the column.
63///
64/// What is cleared is found first, and a table with nothing to clear is
65/// not changed. A MeasurementSet on disk then has SOURCE and MAIN changed
66/// and written in turn, each under casacore's write lock on it: SOURCE in
67/// full as [`MeasurementSet::save`] writes it, MAIN's metadata alone. A
68/// table opened with locking is written under its own lock, held already
69/// or taken and released here; one opened without, under a
70/// [`TableWriteLock`] taken and released here, so a caller holds MAIN's
71/// write lock, if at all, through its handle, never through such a guard.
72/// A lock is waited for as casacore waits, except SOURCE's while the
73/// caller holds MAIN's write lock: one attempt is then made, as a save
74/// never waits for one table while it holds another, where CASA waits
75/// (casa-rs holders do not yet release a lock on request, #694). The
76/// `FT_MODEL*` model images are deleted last and, as CASA deletes them, at
77/// best: one that is held open or cannot be deleted is logged and left.
78///
79/// # Errors
80///
81/// [`MsError::InvalidInput`] for `DATA`, which is no scratch column. A
82/// [`TableError::LockFailed`] when SOURCE's lock is held elsewhere while
83/// the caller holds MAIN's write lock, and any error reading the fields
84/// or locking and writing SOURCE or MAIN. A table whose write fails is
85/// reloaded as it is on disk, holding whatever of the write reached it,
86/// as after an interrupted casacore write; when SOURCE's fails, MAIN is
87/// not changed.
88pub fn prepare_write(
89    ms: &mut MeasurementSet,
90    column: VisibilityDataColumn,
91    adds: bool,
92) -> MsResult<()> {
93    prepare(ms, column, adds, None)
94}

96/// [`prepare_write`] for a caller that holds MAIN's write lock through
97/// `main_lock`, a [`TableWriteLock`] on the MAIN of `ms` (opened without
98/// locking), as a writer of several mutation sessions in turn holds it:
99/// MAIN's metadata is written under it, the write recorded so that the
100/// caller's release publishes it, and SOURCE's lock is tried once, as while
101/// a caller holds MAIN's through its handle.
102///
103/// # Errors
104///
105/// As [`prepare_write`]; a MAIN whose write fails is reloaded as it is on
106/// disk, and the caller's lock publishes what reached it.
107pub fn prepare_write_holding(
108    ms: &mut MeasurementSet,
109    column: VisibilityDataColumn,
110    adds: bool,
111    main_lock: &mut TableWriteLock,
112) -> MsResult<()> {
113    prepare(ms, column, adds, Some(main_lock))
114}

116/// Storage managers whose selected-row save patches the written rows' tiles
117/// in place: the managers CASA and casa-rs store visibilities in.
118pub(crate) const BOUNDED_WRITE_MANAGERS: [&str; 2] = ["TiledShapeStMan", "TiledColumnStMan"];
119
120/// Why MAIN's `column` cannot be written in place in bounded memory, or
121/// `None` when it can: a column stored outside [`BOUNDED_WRITE_MANAGERS`]
122/// is saved by rewriting every row of its storage group, once a save. A
123/// column the writer creates (`created`) clones `DATA`'s manager, which must
124/// be one of them and hold `DATA` alone.
125pub(crate) fn unbounded_destination(main: &Table, column: &str, created: bool) -> Option<String> {
126    let stored = if created { "DATA" } else { column };
127    let Some(manager) = main
128        .data_manager_info()
129        .iter()
130        .find(|manager| manager.columns.iter().any(|name| name == stored))
131    else {
132        return Some(format!("{stored} has no storage manager on disk"));
133    };
134    let storage = if created {
135        format!(
136            "it would be created in a clone of DATA's {}",
137            manager.dm_type
138        )
139    } else {
140        format!("it is stored in {}", manager.dm_type)
141    };
142    if !BOUNDED_WRITE_MANAGERS.contains(&manager.dm_type.as_str()) {
143        return Some(format!(
144            "{storage}; only TiledShapeStMan and TiledColumnStMan columns are written in place"
145        ));
146    }
147    (created && manager.columns.len() != 1)
148        .then(|| format!("{storage}, which DATA shares with other columns"))
149}

151fn prepare(
152    ms: &mut MeasurementSet,
153    column: VisibilityDataColumn,
154    adds: bool,
155    main_lock: Option<&mut TableWriteLock>,
156) -> MsResult<()> {
157    if column == VisibilityDataColumn::Data {
158        return Err(MsError::InvalidInput(
159            "DATA is not a scratch column".to_string(),
160        ));
161    }
162    let mut clearing = Clearing::default();
163    if column == VisibilityDataColumn::ModelData {
164        clearing.plan_virtual_model(ms)?;
165    }
166    if adds {
167        let keywords = ms.main_table().keywords();
168        clearing.main.extend(
169            SORT_KEYWORDS
170                .iter()
171                .filter(|name| keywords.get(name).is_some())
172                .map(|name| name.to_string()),
173        );
174    }
175    let holds_main = main_lock.is_some() || holds_write_lock(ms.main_table());
176    if let Some(source) = ms.subtable_mut(SubtableId::Source)
177        && !(clearing.source.is_empty() && clearing.source_cells.is_empty())
178    {
179        let attempts = if holds_main { 1 } else { 0 };
180        change_in_place(source, attempts, Write::Full, None, |source| {
181            for row in &clearing.source_cells {
182                source
183                    .cell_accessor_mut(*row, "SOURCE_MODEL")
184                    .and_then(|mut cell| cell.set(Value::Record(RecordValue::default())))?;
185            }
186            remove(source, &clearing.source);
187            Ok(())
188        })?;
189    }
190    if !clearing.main.is_empty() {
191        change_in_place(ms.main_table_mut(), 0, Write::Metadata, main_lock, |main| {
192            remove(main, &clearing.main);
193            Ok(())
194        })?;
195    }
196    if column == VisibilityDataColumn::ModelData {
197        delete_model_images(ms);
198    }
199    Ok(())
200}

202/// The keywords to remove from MAIN and SOURCE and the SOURCE rows whose
203/// `SOURCE_MODEL` is emptied: each present, so that a table with nothing
204/// to clear is not changed.
205#[derive(Default)]
206struct Clearing {
207    main: Vec<String>,
208    source: Vec<String>,
209    source_cells: Vec<usize>,
210}
211
212impl Clearing {
213    /// `VisModelData::clearModel`: for every field MAIN names, its pointer
214    /// and model keys wherever they are, and the SOURCE row of a model
215    /// under SOURCE (`removeRecordByKey`).
216    fn plan_virtual_model(&mut self, ms: &MeasurementSet) -> MsResult<()> {
217        let fields = fields_with_models(ms)?;
218        let main = ms.main_table().keywords();
219        let source = ms.subtable(SubtableId::Source);
220        let source_rows = source.map_or(0, Table::row_count);
221        let source_keywords = source.map(Table::keywords);
222        let source_model = source
223            .and_then(Table::schema)
224            .is_some_and(|schema| schema.contains_column("SOURCE_MODEL"));
225        for field in fields {
226            let pointer = format!("definedmodel_field_{field}");
227            let in_source = source_rows > 0
228                && source_keywords.is_some_and(|keywords| keywords.get(&pointer).is_some());
229            let model = if in_source {
230                source_keywords.and_then(|keywords| string(keywords, &pointer))
231            } else {
232                string(main, &pointer)
233            };
234            if let Some(model) = model {
235                if let Some(Value::Scalar(ScalarValue::Int32(row))) =
236                    source_keywords.and_then(|keywords| keywords.get(&model))
237                    && let Ok(row) = usize::try_from(*row)
238                    && row < source_rows
239                    && source_model
240                {
241                    self.source_cells.push(row);
242                }
243                present(&mut self.source, source_keywords, &model);
244                present(&mut self.main, Some(main), &model);
245            }
246            present(&mut self.source, source_keywords, &pointer);
247            present(&mut self.main, Some(main), &pointer);
248        }
249        self.source_cells.sort_unstable();
250        self.source_cells.dedup();
251        Ok(())
252    }
253}

255/// Add `name` to `names` when `keywords` holds it and `names` does not.
256fn present(names: &mut Vec<String>, keywords: Option<&RecordValue>, name: &str) {
257    if keywords.is_some_and(|keywords| keywords.get(name).is_some())
258        && !names.iter().any(|held| held == name)
259    {
260        names.push(name.to_string());
261    }
262}
263
264fn remove(table: &mut Table, names: &[String]) {
265    let keywords = table.keywords_mut();
266    for name in names {
267        keywords.remove(name);
268    }
269}

271/// MAIN rows read at a time while looking for the fields that have models.
272const FIELD_SCAN_ROWS: usize = 65_536;
273
274/// The fields with a virtual model pointer, `definedmodel_field_<id>`, in
275/// MAIN's or SOURCE's keywords, as the module states: those CASA's iterator
276/// reads a virtual model for in place of `MODEL_DATA`. Only keywords are
277/// read.
278pub fn virtual_model_fields(ms: &MeasurementSet) -> BTreeSet<i32> {
279    let pointed = |keywords: &RecordValue| -> Vec<i32> {
280        keywords
281            .fields()
282            .iter()
283            .filter_map(|field| field.name.strip_prefix("definedmodel_field_")?.parse().ok())
284            .collect()
285    };
286    let mut fields: BTreeSet<i32> = pointed(ms.main_table().keywords()).into_iter().collect();
287    if let Some(source) = ms.subtable(SubtableId::Source) {
288        fields.extend(pointed(source.keywords()));
289    }
290    fields
291}

293/// The fields that have a model pointer, in MAIN's or SOURCE's keywords,
294/// and rows in MAIN: those `clearModel` reaches, since it visits the fields
295/// MAIN names and a field without a pointer has nothing to clear. MAIN's
296/// `FIELD_ID` is read only when a pointer exists, a bounded chunk at a
297/// time, and no further than to find every pointed field.
298fn fields_with_models(ms: &MeasurementSet) -> MsResult<BTreeSet<i32>> {
299    let wanted = virtual_model_fields(ms);
300    let mut found = BTreeSet::new();
301    if wanted.is_empty() {
302        return Ok(found);
303    }
304    let main = ms.main_table();
305    let column = main.column_accessor("FIELD_ID")?;
306    let mut start = 0;
307    while start < main.row_count() && found.len() < wanted.len() {
308        let rows: Vec<usize> = (start..main.row_count().min(start + FIELD_SCAN_ROWS)).collect();
309        for cell in column.scalar_cells_owned_for_rows(&rows)? {
310            if let Some(ScalarValue::Int32(field)) = cell
311                && wanted.contains(&field)
312            {
313                found.insert(field);
314            }
315        }
316        start += FIELD_SCAN_ROWS;
317    }
318    Ok(found)
319}

321fn string(keywords: &RecordValue, name: &str) -> Option<String> {
322    match keywords.get(name) {
323        Some(Value::Scalar(ScalarValue::String(value))) if !value.is_empty() => Some(value.clone()),
324        _ => None,
325    }

332    Full,
333    /// Its keywords and description alone, its storage left as it is.

337/// Whether `table` was opened with locking and holds its write lock.
338fn holds_write_lock(table: &Table) -> bool {
339    #[cfg(unix)]
340    {
341        table.has_lock(casa_tables::LockType::Write)
342    }
343    #[cfg(not(unix))]
344    {
345        let _ = table;
346        false
347    }
348}

350/// Apply `change` to `table` and write it to its directory under
351/// casacore's write lock on it, taken in `attempts` (0 waits) unless the
352/// handle holds it or the caller does through `held`, whose write is then
353/// recorded for the caller's release to publish. A table not on disk is
354/// changed in memory alone. When the change or the write fails, a lock
355/// taken here has the table reloaded as it is on disk and is released, and
356/// under `held` the table is reloaded and the lock left to the caller; a
357/// lock the handle holds is left to it.
358fn change_in_place(
359    table: &mut Table,
360    attempts: u32,
361    write: Write,
362    held: Option<&mut TableWriteLock>,
363    change: impl FnOnce(&mut Table) -> MsResult<()>,
364) -> MsResult<()> {
365    let Some(path) = table.path().map(Path::to_path_buf) else {
366        return change(table);
367    };
368    let written = |table: &mut Table| -> MsResult<()> {
369        change(table)?;
370        match write {
371            Write::Full => table.save(measurement_set_table_options(&path))?,
372            Write::Metadata => table.save_metadata_only(TableOptions::new(&path))?,
373        }
374        Ok(())
375    };
376    if let Some(lock) = held {
377        lock.record_write();
378        let result = written(table);
379        if result.is_err() {
380            let _ = table.resync();
381        }
382        return result;
383    }
384    #[cfg(unix)]
385    if table.lock_options().is_some() {
386        if holds_write_lock(table) {
387            // Published when the caller releases its lock.
388            return written(table);
389        }
390        if !table.lock(casa_tables::LockType::Write, attempts)? {
391            return Err(lock_refused(&path));
392        }
393        let result = written(table);
394        if result.is_err() {
395            let _ = table.resync();
396        }
397        let released = table.unlock_metadata_only_or_discard();
398        return result.and(released.map_err(MsError::from));
399    }
400    let mut lock = TableWriteLock::acquire(&path, attempts)?;
401    lock.record_write();
402    let result = written(table);
403    if result.is_err() {
404        let _ = table.resync();
405    }
406    let released = lock.release();
407    result.and(released.map_err(MsError::from))
408}

410fn lock_refused(path: &Path) -> MsError {
411    TableError::LockFailed {
412        path: path.display().to_string(),
413        message: "another process holds the table's lock".to_string(),
414    }
415    .into()
416}

418/// Where the table at `path` is held open, as `TableUtil::canDeleteTable`
419/// asks, read from its lock file without opening the table or waiting for
420/// a lock.
421fn held_open(path: &Path) -> std::io::Result<Option<casa_tables::TableHolder>> {
422    #[cfg(unix)]
423    {
424        casa_tables::table_holder(path)
425    }
426    #[cfg(not(unix))]
427    {
428        let _ = path;
429        Ok(None)
430    }
431}

433/// Delete every `FT_MODEL*` table in the MeasurementSet's directory and in
434/// its SOURCE's, as `clearModel` does, at best: a table held open, in this
435/// process or another, is left, as `TableUtil::canDeleteTable` leaves it,
436/// and so is one whose lock file cannot be read or that cannot be deleted,
437/// each logged.
438fn delete_model_images(ms: &MeasurementSet) {
439    let directories = [
440        ms.path(),
441        ms.subtable(SubtableId::Source).and_then(Table::path),
442    ];
443    for directory in directories.into_iter().flatten() {
444        let Ok(entries) = std::fs::read_dir(directory) else {
445            continue;
446        };
447        for entry in entries.flatten() {
448            let image = entry.path();
449            if !entry.file_name().to_string_lossy().starts_with("FT_MODEL")
450                || !image.join("table.dat").is_file()
451            {
452                continue;
453            }
454            let deleted = match held_open(&image) {
455                Ok(None) => std::fs::remove_dir_all(&image).map_err(|error| error.to_string()),
456                Ok(Some(holder)) => Err(format!("it is held open {holder}")),
457                Err(error) => Err(error.to_string()),
458            };
459            if let Err(error) = deleted {
460                tracing::warn!(
461                    image = %image.display(),
462                    %error,
463                    "a cleared virtual model's image is left"
464                );
465            }
466        }
467    }

919    /// Restores `table.dat`'s mode when dropped.
920    struct Writable(PathBuf);
921
922    impl Drop for Writable {
923        fn drop(&mut self) {
924            use std::os::unix::fs::PermissionsExt;
925            let _ = std::fs::set_permissions(&self.0, std::fs::Permissions::from_mode(0o644));
926        }
927    }

929    /// Make MAIN's `table.dat` read-only, or `None` where it stays
930    /// writable (a process running as root).
931    fn read_only_table_dat(path: &Path) -> Option<Writable> {
932        use std::os::unix::fs::PermissionsExt;
933        let table_dat = path.join("table.dat");
934        std::fs::set_permissions(&table_dat, std::fs::Permissions::from_mode(0o444))
935            .expect("read only");
936        let restore = Writable(table_dat);
937        if std::fs::OpenOptions::new()
938            .write(true)
939            .open(&restore.0)
940            .is_ok()
941        {
942            eprintln!("skipped: table.dat stays writable for this process");
943            return None;
944        }
945        Some(restore)
946    }
