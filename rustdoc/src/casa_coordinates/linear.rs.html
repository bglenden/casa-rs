2//! N-axis linear coordinate.
3//!
4//! [`LinearCoordinate`] implements a general linear transformation between
5//! pixel and world coordinates:
6//!
7//! ```text
8//! world = crval + cdelt * PC * (pixel - crpix)
9//! ```
10//!
11//! where `PC` is a rotation/coupling matrix (identity by default), `cdelt`
12//! is the axis increment vector, `crval` is the reference world value, and
13//! `crpix` is the reference pixel position.
14//!
15//! This corresponds to C++ `LinearCoordinate`.
16
17use casa_core::{ArrayValue, RecordValue, ScalarValue, Value};
18use ndarray::{Array2, ArrayView2, ArrayViewMut2, Axis, ShapeBuilder};
19
20use crate::coordinate::{Coordinate, CoordinateType};
21use crate::error::CoordinateError;
22use crate::record_utils::{get_optional_vec_f64, get_optional_vec_string, get_required_vec_f64};

24/// An N-axis linear coordinate with an affine pixel-to-world mapping.
25///
26/// The transformation follows the FITS WCS linear convention:
27///
28/// ```text
29/// world[i] = crval[i] + cdelt[i] * sum_j(PC[i][j] * (pixel[j] - crpix[j]))
30/// ```
31///
32/// The PC matrix defaults to the identity, making the transformation a simple
33/// scale-and-shift along each axis independently.
34///
35/// Corresponds to C++ `LinearCoordinate`.
36#[derive(Debug, Clone)]
37pub struct LinearCoordinate {
38    crval: Vec<f64>,
39    cdelt: Vec<f64>,
40    crpix: Vec<f64>,
41    pc: Array2<f64>,
42    names: Vec<String>,
43    units: Vec<String>,
44}
45
46impl LinearCoordinate {
47    pub(crate) fn to_pixel_columns(
48        &self,
49        world: ArrayView2<'_, f64>,
50        mut pixels: ArrayViewMut2<'_, f64>,
51    ) -> Result<(), CoordinateError> {
52        if world.ncols() == 0 {
53            return Ok(());
54        }
55        self.validate_increments()?;
56        let inverse = invert_matrix(&self.pc)?;
57        let mut scaled = vec![0.0; self.n_axes()];
58        for (input, mut output) in world.axis_iter(Axis(1)).zip(pixels.axis_iter_mut(Axis(1))) {
59            self.scale_world(input.iter().copied(), &mut scaled);
60            self.project_to_pixel(&inverse, &scaled, output.iter_mut());
61        }
62        Ok(())
63    }
64
65    fn validate_increments(&self) -> Result<(), CoordinateError> {
66        if let Some(axis) = self.cdelt.iter().position(|value| value.abs() < 1e-300) {
67            return Err(CoordinateError::ConversionFailed(format!(
68                "zero increment on axis {axis}"
69            )));
70        }
71        Ok(())
72    }
73
74    fn scale_world(&self, world: impl IntoIterator<Item = f64>, scaled: &mut [f64]) {
75        for (axis, (world, output)) in world.into_iter().zip(scaled).enumerate() {
76            *output = (world - self.crval[axis]) / self.cdelt[axis];
77        }
78    }
79
80    fn project_to_pixel<'a>(
81        &self,
82        inverse: &Array2<f64>,
83        scaled: &[f64],
84        output: impl IntoIterator<Item = &'a mut f64>,
85    ) {
86        for (axis, pixel) in output.into_iter().enumerate() {
87            let mut sum = 0.0;
88            for (index, scaled) in scaled.iter().enumerate() {
89                sum += inverse[[axis, index]] * scaled;
90            }
91            *pixel = sum + self.crpix[axis];
92        }
93    }
94
95    /// Creates a new linear coordinate with `naxes` axes and default values.
96    ///
97    /// The reference value, reference pixel, and all increments default to 0.0
98    /// (increments default to 1.0). Axis names default to `"Linear1"`,
99    /// `"Linear2"`, etc. Units default to empty strings. The PC matrix
100    /// defaults to the identity.
101    ///
102    /// # Panics
103    ///
104    /// Panics if `naxes` is 0.
105    pub fn new(naxes: usize, names: Vec<String>, units: Vec<String>) -> Self {
106        assert!(naxes > 0, "LinearCoordinate requires at least 1 axis");
107        let names = if names.len() == naxes {
108            names
109        } else {
110            (1..=naxes).map(|i| format!("Linear{i}")).collect()
111        };
112        let units = if units.len() == naxes {
113            units
114        } else {
115            vec![String::new(); naxes]
116        };
117        Self {
118            crval: vec![0.0; naxes],
119            cdelt: vec![1.0; naxes],
120            crpix: vec![0.0; naxes],
121            pc: Array2::eye(naxes),
122            names,
123            units,
124        }
125    }
126
127    /// Sets the reference world values. Returns `self` for chaining.
128    ///
129    /// # Panics
130    ///
131    /// Panics if `crval.len()` does not match the number of axes.
132    pub fn with_reference_value(mut self, crval: Vec<f64>) -> Self {
133        assert_eq!(crval.len(), self.n_axes());
134        self.crval = crval;
135        self
136    }

145        self.crpix = crpix;
146        self

151    /// # Panics

156        self.cdelt = cdelt;
157        self
158    }

160    /// Sets the PC (rotation/coupling) matrix. Returns `self` for chaining.
161    ///
162    /// # Panics
163    ///
164    /// Panics if the matrix shape is not `(naxes, naxes)`.
165    pub fn with_pc_matrix(mut self, pc: Array2<f64>) -> Self {
166        let n = self.n_axes();
167        assert_eq!(pc.shape(), [n, n]);
168        self.pc = pc;
169        self
170    }

172    /// Returns the number of axes.
173    fn n_axes(&self) -> usize {
174        self.crval.len()
175    }
176
177    /// Returns a reference to the PC rotation/coupling matrix.
178    pub fn pc_matrix(&self) -> &Array2<f64> {
179        &self.pc
180    }

182    /// This coordinate over only its axes `kept`, in that order.
183    ///
184    /// `None` if `kept` is empty, or if the PC matrix couples a kept axis
185    /// with a dropped one, so that the dropped axis's value would move the
186    /// kept axes' world values or theirs move it. Without coupling, the kept
187    /// axes convert exactly as before, whatever the dropped axes hold.
188    pub fn kept_axes(&self, kept: &[usize]) -> Option<Self> {
189        let dropped = (0..self.n_axes()).filter(|axis| !kept.contains(axis));
190        let coupled = dropped
191            .flat_map(|axis| kept.iter().map(move |&k| (axis, k)))
192            .any(|(axis, k)| self.pc[[k, axis]] != 0.0 || self.pc[[axis, k]] != 0.0);
193        if kept.is_empty() || coupled {
194            return None;
195        }
196        let pick = |values: &[f64]| kept.iter().map(|&k| values[k]).collect();
197        Some(Self {
198            crval: pick(&self.crval),
199            cdelt: pick(&self.cdelt),
200            crpix: pick(&self.crpix),
201            pc: Array2::from_shape_fn((kept.len(), kept.len()), |(i, j)| {
202                self.pc[[kept[i], kept[j]]]
203            }),
204            names: kept.iter().map(|&k| self.names[k].clone()).collect(),
205            units: kept.iter().map(|&k| self.units[k].clone()).collect(),
206        })
207    }

209    /// Reconstructs a linear coordinate from a serialized record.
210    pub fn from_record(rec: &RecordValue) -> Result<Self, CoordinateError> {
211        let crval = get_required_vec_f64(rec, "crval")?;
212        let cdelt = get_required_vec_f64(rec, "cdelt")?;
213        let crpix = get_required_vec_f64(rec, "crpix")?;
214        let naxes = crval.len();
215
216        if naxes == 0 || cdelt.len() != naxes || crpix.len() != naxes {
217            return Err(CoordinateError::InvalidRecord(
218                "linear coordinate axis vectors must have matching non-zero lengths".into(),
219            ));
220        }
221
222        let names = get_optional_vec_string(rec, "axes").unwrap_or_default();
223        let units = get_optional_vec_string(rec, "units").unwrap_or_default();
224
225        let mut coord = Self::new(naxes, names, units)
226            .with_reference_value(crval)
227            .with_increment(cdelt)
228            .with_reference_pixel(crpix);
229
230        if let Some(pc_flat) = get_optional_vec_f64(rec, "pc") {
231            if pc_flat.len() != naxes * naxes {
232                return Err(CoordinateError::InvalidRecord(format!(
233                    "linear pc matrix has {} elements, expected {}",
234                    pc_flat.len(),
235                    naxes * naxes
236                )));
237            }
238            // casacore's matrix is the transpose of the FITS PC matrix this
239            // coordinate holds (`LinearXform`: `pc[i*n+j] = pcIn(j,i)`), so
240            // its elements in row order are the PC's in column order.
241            let pc = Array2::from_shape_vec((naxes, naxes).f(), pc_flat).map_err(|err| {
242                CoordinateError::InvalidRecord(format!("invalid linear pc matrix: {err}"))
243            })?;
244            coord = coord.with_pc_matrix(pc.as_standard_layout().into_owned());
245        }
246
247        Ok(coord)
248    }
249}

251impl Coordinate for LinearCoordinate {
252    fn coordinate_type(&self) -> CoordinateType {
253        CoordinateType::Linear
254    }
255
256    fn n_pixel_axes(&self) -> usize {
257        self.n_axes()
258    }
259
260    fn n_world_axes(&self) -> usize {
261        self.n_axes()
262    }
263
264    fn to_world(&self, pixel: &[f64]) -> Result<Vec<f64>, CoordinateError> {
265        let n = self.n_axes();
266        if pixel.len() != n {
267            return Err(CoordinateError::DimensionMismatch {
268                expected: n,
269                got: pixel.len(),
270            });
271        }
272
273        // delta_pixel = pixel - crpix
274        let delta: Vec<f64> = pixel.iter().zip(&self.crpix).map(|(p, r)| p - r).collect();
275
276        // Apply PC matrix: rotated = PC * delta
277        let mut rotated = vec![0.0; n];
278        for (i, rotated_i) in rotated.iter_mut().enumerate() {
279            for (j, delta_j) in delta.iter().enumerate() {
280                *rotated_i += self.pc[[i, j]] * delta_j;
281            }
282        }
283
284        // world = crval + cdelt * rotated
285        let world: Vec<f64> = (0..n)
286            .map(|i| self.crval[i] + self.cdelt[i] * rotated[i])
287            .collect();
288
289        Ok(world)
290    }
291
292    fn to_pixel(&self, world: &[f64]) -> Result<Vec<f64>, CoordinateError> {
293        let n = self.n_axes();
294        if world.len() != n {
295            return Err(CoordinateError::DimensionMismatch {
296                expected: n,
297                got: world.len(),
298            });
299        }
300
301        // Compute (world - crval) / cdelt
302        self.validate_increments()?;
303        let mut scaled = vec![0.0; n];
304        self.scale_world(world.iter().copied(), &mut scaled);
305
306        // Invert PC matrix: pixel = crpix + PC^-1 * scaled
307        // For small N, use direct inversion via cofactor / Gauss elimination.
308        let inv_pc = invert_matrix(&self.pc)?;
309
310        let mut pixel = vec![0.0; n];
311        self.project_to_pixel(&inv_pc, &scaled, &mut pixel);
312
313        Ok(pixel)
314    }
315
316    fn reference_value(&self) -> Vec<f64> {
317        self.crval.clone()
318    }
319
320    fn reference_pixel(&self) -> Vec<f64> {
321        self.crpix.clone()
322    }
323
324    fn increment(&self) -> Vec<f64> {
325        self.cdelt.clone()
326    }
327
328    fn axis_names(&self) -> Vec<String> {
329        self.names.clone()
330    }
331
332    fn axis_units(&self) -> Vec<String> {
333        self.units.clone()
334    }
335}

337impl LinearCoordinate {
338    pub(crate) fn to_record(&self) -> RecordValue {
339        let mut rec = RecordValue::default();
340        let n = self.n_axes();
341
342        rec.upsert(
343            "coordinate_type",
344            Value::Scalar(ScalarValue::String("Linear".into())),
345        );
346        rec.upsert("naxes", Value::Scalar(ScalarValue::Int32(n as i32)));
347        rec.upsert(
348            "crval",
349            Value::Array(casa_core::ArrayValue::from_f64_vec(self.crval.clone())),
350        );
351        rec.upsert(
352            "cdelt",
353            Value::Array(casa_core::ArrayValue::from_f64_vec(self.cdelt.clone())),
354        );
355        rec.upsert(
356            "crpix",
357            Value::Array(casa_core::ArrayValue::from_f64_vec(self.crpix.clone())),
358        );
359        // casacore's matrix, the PC's transpose (see `from_record`).
360        rec.upsert(
361            "pc",
362            Value::Array(ArrayValue::Float64(
363                ndarray::ArrayD::from_shape_vec(
364                    ndarray::IxDyn(&[n, n]).f(),
365                    self.pc.iter().copied().collect(),
366                )
367                .expect("linear pc matrix shape should match axis count"),
368            )),
369        );
370        rec.upsert(
371            "axes",
372            Value::Array(casa_core::ArrayValue::from_string_vec(self.names.clone())),
373        );
374        rec.upsert(
375            "units",
376            Value::Array(casa_core::ArrayValue::from_string_vec(self.units.clone())),
377        );
378
379        rec
380    }
381}

383/// Inverts a square matrix using Gauss-Jordan elimination with partial pivoting.
384///
385/// Returns an error if the matrix is singular.
386pub(crate) fn invert_matrix(m: &Array2<f64>) -> Result<Array2<f64>, CoordinateError> {
387    let n = m.nrows();
388    assert_eq!(n, m.ncols());
389
390    // Augmented matrix [M | I]
391    let mut aug = Array2::zeros((n, 2 * n));
392    for i in 0..n {
393        for j in 0..n {
394            aug[[i, j]] = m[[i, j]];
395        }
396        aug[[i, n + i]] = 1.0;
397    }
398
399    // Forward elimination with partial pivoting
400    for col in 0..n {
401        // Find pivot
402        let mut max_val = aug[[col, col]].abs();
403        let mut max_row = col;
404        for row in (col + 1)..n {
405            let val = aug[[row, col]].abs();
406            if val > max_val {
407                max_val = val;
408                max_row = row;
409            }
410        }
411
412        if max_val == 0.0 || !max_val.is_finite() {
413            return Err(CoordinateError::SingularTransform("singular PC matrix"));
414        }
415
416        // Swap rows
417        if max_row != col {
418            for j in 0..2 * n {
419                let tmp = aug[[col, j]];
420                aug[[col, j]] = aug[[max_row, j]];
421                aug[[max_row, j]] = tmp;
422            }
423        }
424
425        // Scale pivot row
426        let pivot = aug[[col, col]];
427        for j in 0..2 * n {
428            aug[[col, j]] /= pivot;
429        }
430
431        // Eliminate column
432        for row in 0..n {
433            if row != col {
434                let factor = aug[[row, col]];
435                for j in 0..2 * n {
436                    aug[[row, j]] -= factor * aug[[col, j]];
437                }
438            }
439        }
440    }
441
442    // Extract inverse from right half
443    let mut inv = Array2::zeros((n, n));
444    for i in 0..n {
445        for j in 0..n {
446            inv[[i, j]] = aug[[i, n + j]];
447        }
448    }
449
450    Ok(inv)
451}
