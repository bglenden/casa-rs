2//! Convolving an image's direction planes with a Gaussian, as the image
3//! tool's `convolve2d` does (CASA's `Image2DConvolver`).
4//!
5//! `simutil.convimage` convolves a Jy/pixel sky model with the clean beam to
6//! compare it with the clean image. For an image in Jy/pixel, or any flux
7//! per pixel, CASA's rules (`Image2DConvolver::_dealWithRestoringBeam`) are:
8//! - the kernel is the Gaussian at unit peak, unscaled, so a point of 1
9//!   Jy/pixel becomes a peak of 1 Jy/beam;
10//! - the output is in Jy/beam, with the Gaussian as its restoring beam.
11//!
12//! The kernel is sampled on a square of an odd number of pixels: ±5σ of
13//! the wider axis, rounded as `Image2DConvolver::_sizeOfGaussian` rounds
14//! it. It is centred on the middle pixel, with the beam's position angle
15//! (north through east) turned into the pixel frame at the reference pixel.
16//! The convolution is linear (casacore's `LatticeConvolver` with
17//! `ConvEnums::LINEAR`): the plane is padded with zeros, so flux does not
18//! wrap around the edges, and the output has the input's shape. Masked
19//! pixels count as zero, and the output keeps the input's mask.
20//!
21//! As in CASA, the pixels must be square. Images in other units (Jy/beam,
22//! whose beam CASA convolves numerically, or none, whose kernel it scales to
23//! unit volume) are refused: the simulation tasks convolve Jy/pixel images
24//! only.

37/// An image convolved with a Gaussian.
38#[derive(Clone, Debug)]
39pub struct Convolved {
40    /// The pixels, in the input's shape.
41    pub data: ArrayD<f32>,

43    pub mask: Option<ArrayD<bool>>,
44    /// The output's brightness unit, `Jy/beam`.
45    pub units: String,
46    /// The output's restoring beam: the Gaussian, its position angle in
47    /// [−90°, 90°] as casacore's `GaussianBeam` keeps it.
48    pub beam: GaussianBeam,
49}

51/// The direction coordinate and its first pixel axis.
52fn direction_axes(
53    coordinates: &CoordinateSystem,
54) -> Result<(usize, &DirectionCoordinate), ImageError> {
55    let index = coordinates
56        .find_coordinate(CoordinateType::Direction)
57        .ok_or_else(|| {
58            ImageError::InvalidMetadata("convolve2d needs a direction coordinate".into())
59        })?;
60    let first = (0..index)
61        .map(|k| coordinates.coordinate(k).n_pixel_axes())
62        .sum();
63    let CoordinateModel::Direction(direction) = coordinates.coordinate(index) else {
64        unreachable!("find_coordinate found a direction coordinate");
65    };
66    Ok((first, direction))
67}

69/// The sky offsets (east, north), rad, of a pixel step along each axis at
70/// the reference pixel: `[[east per x, east per y], [north per x, north per
71/// y]]`, by central differences of the coordinate's own conversion. It
72/// carries the increments, the PC matrix and the projection, as casacore's
73/// `GaussianShape::toPixel` does when `worldWidthsToPixel` turns a beam into
74/// pixels.
75fn sky_per_pixel(direction: &DirectionCoordinate) -> Result<[[f64; 2]; 2], ImageError> {
76    const STEP: f64 = 0.5;
77    let crpix = direction.reference_pixel();
78    let world = |dx: f64, dy: f64| direction.to_world(&[crpix[0] + dx, crpix[1] + dy]);
79    let centre = world(0.0, 0.0)?;
80    let mut jacobian = [[0.0; 2]; 2];
81    for (axis, (dx, dy)) in [(STEP, 0.0), (0.0, STEP)].into_iter().enumerate() {
82        let (plus, minus) = (world(dx, dy)?, world(-dx, -dy)?);
83        let dlon = (plus[0] - minus[0] + PI).rem_euclid(2.0 * PI) - PI;
84        jacobian[0][axis] = dlon * centre[1].cos() / (2.0 * STEP);
85        jacobian[1][axis] = (plus[1] - minus[1]) / (2.0 * STEP);
86    }
87    Ok(jacobian)
88}

90/// `Image2DConvolver::_sizeOfGaussian`: twice (5σ rounded, plus one).
91fn size_of_gaussian(fwhm_pixels: f64) -> usize {
92    let sigma = fwhm_pixels / (8.0 * LN_2).sqrt();
93    ((5.0 * sigma + 0.5) as usize + 1) * 2
94}

96/// The Gaussian kernel for `beam` on pixels whose sky offsets are `sky`
97/// ([`sky_per_pixel`]): its side (odd) and values, first axis fastest, peak
98/// 1 at the centre. The major axis points north through east, (east,
99/// north) = (sin θ, cos θ). The side is sized from the beam's widths in
100/// pixels, the axes of the pixel-frame ellipse.
101fn kernel(beam: &GaussianBeam, sky: [[f64; 2]; 2]) -> (usize, Vec<f64>) {
102    let (sin_pa, cos_pa) = beam.position_angle.sin_cos();
103    // The exponent's quadratic form on pixel offsets, q(d) = dᵀ Q d, with the
104    // value exp(−4 ln 2 q): half maximum at q = 1/4.
105    let along = [
106        sky[0][0] * sin_pa + sky[1][0] * cos_pa,
107        sky[0][1] * sin_pa + sky[1][1] * cos_pa,
108    ];
109    let across = [
110        sky[0][0] * cos_pa - sky[1][0] * sin_pa,
111        sky[0][1] * cos_pa - sky[1][1] * sin_pa,
112    ];
113    let (a2, b2) = (beam.major.powi(2), beam.minor.powi(2));
114    let q = [
115        [
116            along[0] * along[0] / a2 + across[0] * across[0] / b2,
117            along[0] * along[1] / a2 + across[0] * across[1] / b2,
118        ],
119        [
120            along[1] * along[0] / a2 + across[1] * across[0] / b2,
121            along[1] * along[1] / a2 + across[1] * across[1] / b2,
122        ],
123    ];
124    let (trace, det) = (q[0][0] + q[1][1], q[0][0] * q[1][1] - q[0][1] * q[1][0]);
125    let root = (trace * trace / 4.0 - det).max(0.0).sqrt();
126    let (small, large) = (trace / 2.0 - root, trace / 2.0 + root);
127    let (major_pixels, minor_pixels) = (1.0 / small.sqrt(), 1.0 / large.sqrt());
128    let mut side = size_of_gaussian(major_pixels).max(size_of_gaussian(minor_pixels));
129    if side.is_multiple_of(2) {
130        side += 1;
131    }
132    let centre = (side - 1) as f64 / 2.0;
133    let mut values = Vec::with_capacity(side * side);
134    for j in 0..side {
135        for i in 0..side {
136            let (dx, dy) = (i as f64 - centre, j as f64 - centre);
137            let form = q[0][0] * dx * dx + 2.0 * q[0][1] * dx * dy + q[1][1] * dy * dy;
138            values.push((-4.0 * LN_2 * form).exp());
139        }
140    }
141    (side, values)
142}

144/// The smallest length at least `n` with no prime factor above 7, which
145/// FFTW transforms fastest.
146fn fast_length(n: usize) -> usize {
147    (n..)
148        .find(|&m| {
149            let mut rest = m;
150            for p in [2, 3, 5, 7] {
151                while rest % p == 0 {
152                    rest /= p;
153                }
154            }
155            rest == 1
156        })
157        .expect("a length with small prime factors exists")
158}

160/// The linear convolution of one plane (first axis fastest) with the
161/// kernel, cut to the plane's shape about the kernel's centre.
162struct PlaneConvolver {
163    plane: [usize; 2],
164    padded: [usize; 2],
165    kernel_side: usize,
166    kernel_transform: Vec<Complex64>,
167    fft: Fft2<f64>,
168}
169
170impl PlaneConvolver {
171    fn new(plane: [usize; 2], kernel_side: usize, kernel: &[f64]) -> Result<Self, ImageError> {
172        let padded = [
173            fast_length(plane[0] + kernel_side - 1),
174            fast_length(plane[1] + kernel_side - 1),
175        ];
176        // Row-major [rows = second axis][columns = first axis].
177        let mut fft = Fft2::<f64>::with_threads([padded[1], padded[0]], 1)
178            .map_err(|error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}")))?
179            .with_estimated_plan();
180        let mut kernel_transform = vec![Complex64::new(0.0, 0.0); padded[0] * padded[1]];
181        for j in 0..kernel_side {
182            for i in 0..kernel_side {
183                kernel_transform[i + j * padded[0]] =
184                    Complex64::new(kernel[i + j * kernel_side], 0.0);
185            }
186        }
187        fft.transform(&mut kernel_transform, false)
188            .map_err(|error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}")))?;
189        Ok(Self {
190            plane,
191            padded,
192            kernel_side,
193            kernel_transform,
194            fft,
195        })
196    }
197
198    fn convolve(&mut self, values: &[f64]) -> Result<Vec<f64>, ImageError> {
199        let [nx, ny] = self.plane;
200        let [px, py] = self.padded;
201        let mut grid = vec![Complex64::new(0.0, 0.0); px * py];
202        for j in 0..ny {
203            for i in 0..nx {
204                grid[i + j * px] = Complex64::new(values[i + j * nx], 0.0);
205            }
206        }
207        let fft_error = |error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}"));
208        self.fft.transform(&mut grid, false).map_err(fft_error)?;
209        for (value, kernel) in grid.iter_mut().zip(&self.kernel_transform) {
210            *value *= kernel;
211        }
212        self.fft.transform(&mut grid, true).map_err(fft_error)?;
213        let norm = (px * py) as f64;
214        let centre = (self.kernel_side - 1) / 2;
215        let mut out = vec![0.0; nx * ny];
216        for j in 0..ny {
217            for i in 0..nx {
218                out[i + j * nx] = grid[(i + centre) + (j + centre) * px].re / norm;
219            }
220        }
221        Ok(out)
222    }
223}

225/// `ia.convolve2d(major, minor, pa)` with a Gaussian on the direction plane
226/// of `input`, an image in a flux per pixel.
227///
228/// # Errors
229///
230/// An image without a direction coordinate, with pixels that are not square,
231/// or in a unit that is not a flux per pixel; a failure to read the image.
232pub fn convolve2d(
233    input: &impl ImageInterface<f32>,
234    beam: &GaussianBeam,
235) -> Result<Convolved, ImageError> {
236    let (axis, direction) = direction_axes(input.coordinates())?;
237    let increment = direction.increment();
238    let (x, y) = (increment[0].abs(), increment[1].abs());
239    if (x - y).abs() > 1e-13 * x.max(y) {
240        return Err(ImageError::InvalidMetadata(
241            "Pixels must be square, please regrid your image so that they are".into(),
242        ));
243    }
244    if !input.units().to_ascii_uppercase().contains("/PIXEL") {
245        return Err(ImageError::InvalidMetadata(format!(
246            "convolve2d: casa-rs convolves images in Jy/pixel; this one is in {:?}",
247            input.units()
248        )));
249    }
250    let shape = input.shape().to_vec();
251    let plane = [shape[axis], shape[axis + 1]];
252    let (side, values) = kernel(beam, sky_per_pixel(direction)?);
253    let mut convolver = PlaneConvolver::new(plane, side, &values)?;
254    let mut data = input.get()?;
255    let mask = input.default_mask()?;
256    let others: Vec<usize> = (0..shape.len())
257        .filter(|a| *a != axis && *a != axis + 1)
258        .collect();
259    let planes: usize = others.iter().map(|a| shape[*a]).product();
260    let mut index = vec![0usize; shape.len()];
261    let mut values = vec![0.0f64; plane[0] * plane[1]];
262    for plane_number in 0..planes {
263        let mut rest = plane_number;
264        for a in &others {
265            index[*a] = rest % shape[*a];
266            rest /= shape[*a];
267        }
268        for j in 0..plane[1] {
269            for i in 0..plane[0] {
270                index[axis] = i;
271                index[axis + 1] = j;
272                let good = mask
273                    .as_ref()
274                    .is_none_or(|mask| mask[ndarray::IxDyn(&index)]);
275                values[i + j * plane[0]] = if good {
276                    f64::from(data[ndarray::IxDyn(&index)])
277                } else {
278                    0.0
279                };
280            }
281        }
282        let convolved = convolver.convolve(&values)?;
283        for j in 0..plane[1] {
284            for i in 0..plane[0] {
285                index[axis] = i;
286                index[axis + 1] = j;
287                data[ndarray::IxDyn(&index)] = convolved[i + j * plane[0]] as f32;
288            }
289        }
290    }
291    Ok(Convolved {
292        data,
293        mask,
294        units: "Jy/beam".into(),
295        beam: GaussianBeam::new(
296            beam.major,
297            beam.minor,
298            unwrap_position_angle(beam.position_angle),
299        ),
300    })
301}

303/// A position angle in [−90°, 90°], as casacore's `GaussianBeam` keeps it.
304fn unwrap_position_angle(angle: f64) -> f64 {
305    use std::f64::consts::FRAC_PI_2;
306    let mut pa = angle % PI;
307    if pa > FRAC_PI_2 {
308        pa -= PI;
309    } else if pa < -FRAC_PI_2 {
310        pa += PI;
311    }
312    pa
313}

322    /// Sky offsets of RA increasing to the left and Dec up, 1e-6 rad pixels.
323    const UNROTATED: [[f64; 2]; 2] = [[-1.0e-6, 0.0], [0.0, 1.0e-6]];
324
325    #[test]
326    fn kernel_sizes_round_as_image2dconvolver_rounds_them() {
327        // σ = 5 / √(8 ln 2) ≈ 2.123: 5σ = 10.6 → (11 + 1) × 2 = 24, made odd.
328        assert_eq!(size_of_gaussian(5.0), 24);
329        let beam = GaussianBeam::new(5.0e-6, 2.0e-6, 0.0);
330        assert_eq!(kernel(&beam, UNROTATED).0, 25);
331    }
332
333    #[test]
334    fn the_major_axis_lies_north_through_east_on_the_sky() {
335        // RA increases to the left (negative increment): a position angle of
336        // 90° (east) puts the major axis along −x, the same line as +x.
337        let beam = GaussianBeam::new(6.0e-6, 2.0e-6, std::f64::consts::FRAC_PI_2);
338        let (side, values) = kernel(&beam, UNROTATED);
339        let c = (side - 1) / 2;
340        let along_x = values[(c + 3) + c * side];
341        let along_y = values[c + (c + 3) * side];
342        assert!(along_x > along_y, "{along_x} {along_y}");
343        // Peak 1 at the centre.
344        assert!((values[c + c * side] - 1.0).abs() < 1e-15);
345    }

347    #[test]
348    fn a_rotated_grid_turns_the_kernel_with_it() {
349        // The #941 review's case: PC [[0, −1], [1, 0]], 1″ pixels, a 6″ × 2″
350        // beam at PA 0°. Pixel x then runs north, so the major axis lies
351        // along x: half maximum 3 pixels out, as CASA 6.7.6 gives 0.5 there
352        // and 0.00195 three pixels along y.
353        let arcsec = PI / 648_000.0;
354        let rotated = DirectionCoordinate::new(
355            DirectionRef::J2000,
356            Projection::new(ProjectionType::SIN),
357            [1.0, 0.3],
358            [-arcsec, arcsec],
359            [10.0, 10.0],
360        )
361        .with_pc_matrix(ndarray::array![[0.0, -1.0], [1.0, 0.0]]);
362        let beam = GaussianBeam::new(6.0 * arcsec, 2.0 * arcsec, 0.0);
363        let (side, values) = kernel(&beam, sky_per_pixel(&rotated).unwrap());
364        let c = (side - 1) / 2;
365        assert!((values[(c + 3) + c * side] - 0.5).abs() < 1e-9);
366        assert!((values[c + (c + 3) * side] - 0.001_953_125).abs() < 1e-9);
367    }
