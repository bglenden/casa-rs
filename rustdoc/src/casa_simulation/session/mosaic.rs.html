2//! `setoptions(ftmachine='mosaic')`: what `predict` builds CASA's simulator
3//! mosaic machine from.
4//!
5//! `Simulator::createSkyEquation` builds `MosaicFTNew` with the
6//! heterogeneous-array beams of `TransformMachines/HetArrayConvFunc` when
7//! OBSERVATION row 0's telescope has PBMath's UNKNOWN, OVRO, ACA or ALMA
8//! beam. casa-rs builds it for ALMA and ACA, whose dishes
9//! `findAntennaSizes` replaces by the 10.7 m and 6.25 m Airy apertures
10//! ([`AiryDish::casa_alma_simulator`]). Other telescopes keep
11//! `MosaicFTNew`'s own `SimplePBConvFunc` of their beam, not yet here.
12//!
13//! - The dish classes are the distinct dish diameters in ANTENNA order, a
14//!   diameter compared as casacore's `String::toString` prints it, to six
15//!   significant digits.
16//! - Every channel of each window observed is selected.
17//! - A scan is pointed at its phase centre, which casa-rs's POINTING rows
18//!   hold; the kernels' phase ramp is its offset from the model's pixel
19//!   `(nx/2, ny/2)`. CASA predicts nothing for a pointing outside the
20//!   model image (`checkPBOfField`); casa-rs does not predict one yet.
21
22use casa_imaging_operator::{AiryDish, MosaicWindow};
23
24use super::Simulator;
25use super::error::{SimulatorError, SimulatorResult};
26use crate::predict::{MosaicBeams, PredictError, SkyModel};
27
28/// The telescopes whose mosaic casa-rs builds.
29const MOSAIC_TELESCOPES: [&str; 2] = ["ALMA", "ACA"];
30
31/// A dish diameter as `String::toString(Double)` prints it, which keys
32/// `HetArrayConvFunc`'s dish classes: six significant digits.
33fn diameter_key(diameter_m: f64) -> String {
34    format!("{diameter_m:.5e}")
35}
36
37/// The most a diameter key's text holds: `format!` grows its string from
38/// nothing, to 16 bytes for its nine to twelve characters.
39const DIAMETER_KEY_BYTES: usize = 32;
40
41/// One pointing the mosaic machine keeps: a phase centre's bits and its
42/// offset.
43pub(super) type MosaicPointing = ([u64; 2], [f64; 2]);
44
45impl Simulator {
46    /// Bytes the mosaic machine's state holds, known before it is formed:
47    /// what the prediction keeps (a pointing for each scan's phase centre at
48    /// most, and each antenna's dish class), and what builds the beams
49    /// until the kernels are built (a dish and a diameter key for each
50    /// antenna at most, and every window's channel frequencies twice).
51    pub(super) fn mosaic_state_bytes(&self) -> (u64, u64) {
52        let kept =
53            self.scans.len() * size_of::<MosaicPointing>() + self.antennas.len() * size_of::<u8>();
54        let antennas = self.antennas.len()
55            * (size_of::<AiryDish>() + size_of::<String>() + DIAMETER_KEY_BYTES);
56        let windows = self.spectral_windows.len() * size_of::<MosaicWindow>()
57            + self
58                .spectral_windows
59                .iter()
60                .map(|window| 2 * window.channel_count * size_of::<f64>())
61                .sum::<usize>();
62        (kept as u64, (antennas + windows) as u64)
63    }
64
65    /// The mosaic machine's beams for the scans observed so far, and each
66    /// antenna's dish class.
67    ///
68    /// # Errors
69    ///
70    /// Not yet implemented: a telescope other than ALMA and ACA, an antenna
71    /// without a dish, or more than 255 dish diameters.
72    pub(super) fn mosaic_beams(&self) -> SimulatorResult<(MosaicBeams, Vec<u8>)> {
73        let telescope = self.observing_telescope();
74        if !MOSAIC_TELESCOPES.contains(&telescope) {
75            return Err(SimulatorError::not_yet_implemented(format!(
76                "predict with setoptions(ftmachine='mosaic') for telescope {telescope} is not \
77                 yet implemented in casa-rs (#841); ALMA's and ACA's heterogeneous-array beams \
78                 are"
79            )));
80        }
81        // At most one class an antenna: the capacities are the bounds
82        // `mosaic_state_bytes` admits.
83        let mut keys: Vec<String> = Vec::with_capacity(self.antennas.len());
84        let mut dishes = Vec::with_capacity(self.antennas.len());
85        let mut classes = Vec::with_capacity(self.antennas.len());
86        for antenna in &self.antennas {
87            // `findAntennaSizes` gives a dish class only to `diameter > 0`.
88            if antenna.diameter_m.is_nan() || antenna.diameter_m <= 0.0 {
89                return Err(SimulatorError::not_yet_implemented(format!(
90                    "predict with setoptions(ftmachine='mosaic') for antenna {} of diameter {} \
91                     m is not yet implemented in casa-rs (#841)",
92                    antenna.name, antenna.diameter_m
93                )));
94            }
95            let key = diameter_key(antenna.diameter_m);
96            let class = keys
97                .iter()
98                .position(|known| *known == key)
99                .unwrap_or_else(|| {
100                    tracing::info!(
101                        "Overriding PB with Airy of diameter {} m starting with antenna {}",
102                        AiryDish::casa_alma_simulator(antenna.diameter_m).aperture_m,
103                        classes.len()
104                    );
105                    keys.push(key);
106                    dishes.push(AiryDish::casa_alma_simulator(antenna.diameter_m));
107                    keys.len() - 1
108                });
109            classes.push(u8::try_from(class).map_err(|_| {
110                SimulatorError::not_yet_implemented(
111                    "predict with setoptions(ftmachine='mosaic') for more than 255 dish \
112                     diameters is not yet implemented in casa-rs (#841)",
113                )
114            })?);
115        }
116        let mut windows: Vec<MosaicWindow> = Vec::with_capacity(self.spectral_windows.len());
117        for scan in &self.scans {
118            let id = scan.spectral_window_id;
119            if windows
120                .iter()
121                .any(|window| window.spectral_window as usize == id)
122            {
123                continue;
124            }
125            let window = &self.spectral_windows[id];
126            let frequencies = window.channel_frequencies_hz();
127            windows.push(MosaicWindow {
128                spectral_window: id as u32,
129                window_frequencies_hz: frequencies.clone(),
130                channel_width_hz: window.channel_width_hz,
131                selected_frequencies_hz: frequencies,
132            });
133        }
134        Ok((MosaicBeams { dishes, windows }, classes))
135    }
136
137    /// Each distinct pointing of the scans observed so far (its bits), where
138    /// their antennas point, and its offset in `model`
139    /// ([`SkyModel::pointing_offset_rad`]), sorted by the bits. The capacity
140    /// is a pointing for each scan, the bound `mosaic_state_bytes` admits.
141    ///
142    /// # Errors
143    ///
144    /// Not yet implemented: a pointing outside the model image. Raised: a
145    /// pointing the model's coordinates cannot convert.
146    pub(super) fn mosaic_pointings(
147        &self,
148        model: &SkyModel,
149    ) -> SimulatorResult<Vec<MosaicPointing>> {
150        let mut pointings: Vec<MosaicPointing> = Vec::with_capacity(self.scans.len());
151        for scan in &self.scans {
152            let key = scan.pointing_rad.map(f64::to_bits);
153            let offset = model
154                .pointing_offset_rad(scan.pointing_rad)
155                .map_err(|error: PredictError| {
156                    SimulatorError::raised(format!("Caught exception: {error}"))
157                })?
158                .ok_or_else(|| {
159                    SimulatorError::not_yet_implemented(format!(
160                        "predict with setoptions(ftmachine='mosaic') for field {} pointed \
161                         outside the model image is not yet implemented in casa-rs (#841); \
162                         CASA predicts nothing for it",
163                        self.fields[scan.field_id].name
164                    ))
165                })?;
166            pointings.push((key, offset));
167        }
168        pointings.sort_unstable_by_key(|(key, _)| *key);
169        pointings.dedup_by_key(|(key, _)| *key);
170        Ok(pointings)
171    }
172}

174#[cfg(test)]
175mod tests {
176    use super::diameter_key;
177
178    #[test]
179    fn diameters_compare_to_six_significant_digits() {
180        assert_eq!(diameter_key(12.0), diameter_key(12.000_000_1));
181        assert_ne!(diameter_key(12.0), diameter_key(12.000_1));
182        assert_ne!(diameter_key(12.0), diameter_key(7.0));
183    }
