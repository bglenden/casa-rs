24pub(super) fn needs_model_references(
25    ms: &MeasurementSet,
26    rows: &[ResolvedMsSelectionRow],
27    plot: &MsPlotSpec,
28    columns: &PlotMsColumnSelection<'_>,
29) -> bool {
30    let averaging = &plot.averaging;
31    let time_spans = (averaging.avgscan || averaging.avgfield)
32        && !averaging.avgbaseline
33        && !averaging.avgantenna
34        && !averaging.avgspw;
35    let vector_time = averaging.avgtime.is_some()
36        && !averaging.scalar
37        && (time_spans || uses_unspanned_antenna_or_baseline(plot))
38        && plot.iteration.iteraxis.is_none()
39        && supports_ordinary_time_axes(plot)
40        && plot.y_axes.iter().all(|axis| axis.is_visibility_math());
41    let spectral_windows = uses_spectral_window_average(plot)
42        && plot.transforms.freqframe.is_none()
43        && plot.iteration.iteraxis.is_none_or(|axis| {
44            !averaging.avgscan && !averaging.avgfield && axis.supports_averaged_ordinates()
45        });
46    columns.implicit_derived_model
47        && (vector_time || spectral_windows)
48        && visibility_columns(plot, columns).any(MsDataColumn::needs_model)
49        && {
50            let fields = virtual_model_fields(ms);
51            rows.iter().any(|row| fields.contains(&row.field_id))
52        }
53}

55pub(super) fn ordered_model_reference_slots(
56    rows: &[ResolvedMsSelectionRow],
57    row_order: &[usize],
58    starts_buffer: &[bool],
59    phase_reference_rows: &[usize],
60) -> (Vec<usize>, Vec<usize>) {
61    let slots_by_row: HashMap<_, _> = rows
62        .iter()
63        .enumerate()
64        .map(|(slot, row)| (row.row_index, slot))
65        .collect();
66    let mut phase_slots = vec![0; rows.len()];
67    let mut recipe_slots = vec![0; rows.len()];
68    let mut reference = 0;
69    for ((&row, &starts), phase_row) in row_order
70        .iter()
71        .zip(starts_buffer)
72        .zip(phase_reference_rows)
73    {
74        let slot = slots_by_row[&row];
75        if starts {
76            reference = slot;
77        }
78        recipe_slots[slot] = reference;
79        phase_slots[slot] = slots_by_row[phase_row];
80    }
81    (phase_slots, recipe_slots)
82}

84fn uses_unspanned_antenna_or_baseline(plot: &MsPlotSpec) -> bool {
85    !plot.averaging.avgscan
86        && !plot.averaging.avgfield
87        && (uses_antenna_visibility_average(plot)
88            || uses_vector_baseline_average(plot)
89            || (plot.averaging.avgbaseline && uses_scalar_visibility_average(plot)))
90}

92fn visibility_columns<'a>(
93    plot: &MsPlotSpec,
94    columns: &'a PlotMsColumnSelection<'_>,
95) -> impl Iterator<Item = MsDataColumn> + Clone + 'a {
96    let axes = [
97        Some(plot.x_axis),
98        Some(plot.y_axis()),
99        plot.secondary_y_axis(),
100    ];
101    let picks = [
102        columns.x.unwrap_or(plot.data_column),
103        plot.data_column,
104        columns.secondary_y.unwrap_or(plot.data_column),
105    ];
106    axes.into_iter()
107        .zip(picks)
108        .filter_map(|(axis, column)| {
109            axis.is_some_and(MsAxis::is_visibility_math)
110                .then_some(column)
111        })
112        .chain(
113            columns
114                .additional_y
115                .iter()
116                .filter_map(|&(axis, column)| axis.is_visibility_math().then_some(column)),
117        )
118}

120/// Supply read-only predicted cells before a supported averaging transform.
121pub(super) fn selected_model_cells(
122    ms: &MeasurementSet,
123    rows: &[ResolvedMsSelectionRow],
124    plot: Option<&MsPlotSpec>,
125    columns: &PlotMsColumnSelection<'_>,
126    model_references: Option<PlotMsModelReferences<'_>>,
127    load_models: Option<PlotMsModelLoader>,
128) -> Result<Option<HashMap<usize, ArrayValue>>, String> {
129    let Some(plot) = plot else {
130        return Ok(None);
131    };
132    let mut visibility_columns = visibility_columns(plot, columns);
133    if !visibility_columns.clone().any(MsDataColumn::needs_model) {
134        return Ok(None);
135    }
136    let extended = uses_extended_averaging(plot);
137    let averaging = extended || plot.averaging.avgchannel.is_some_and(|width| width > 1);
138    let ordinary_time = uses_ordinary_time_averaging(plot) && supports_ordinary_time_axes(plot);
139    let scalar_span = columns.implicit_derived_model
140        && plot.averaging.avgtime.is_some()
141        && (plot.averaging.avgscan || plot.averaging.avgfield)
142        && !plot.averaging.avgbaseline
143        && plot.iteration.iteraxis.is_none()
144        && supports_ordinary_time_axes(plot)
145        && uses_scalar_visibility_average(plot);
146    let antenna_or_baseline = columns.implicit_derived_model
147        && plot.iteration.iteraxis.is_none()
148        && supports_ordinary_time_axes(plot)
149        && uses_unspanned_antenna_or_baseline(plot);
150    let standard_columns = visibility_columns.all(|column| {
151        matches!(
152            column,
153            MsDataColumn::Data | MsDataColumn::Corrected | MsDataColumn::Model
154        )
155    });
156    let field_models_supported =
157        !plot.averaging.avgfield || plot.averaging.scalar || model_references.is_some();
158    let derived_average = columns.implicit_derived_model
159        && field_models_supported
160        && (columns.extended_derived || (!extended && uses_channel_derived(plot, columns.x)));
161    let can_predict = !averaging
162        || ((!extended
163            || ordinary_time
164            || scalar_span
165            || model_references.is_some()
166            || antenna_or_baseline)
167            && standard_columns)
168        || derived_average;
169    if let Some(load) = load_models.filter(|_| can_predict) {
170        return load(ms, rows, model_references);
171    }
172    if averaging || (plot.iteration.iteraxis.is_some() && plot.secondary_y_axis().is_some()) {
173        let message = if can_predict {
174            "use casars.tasks.plotms to predict virtual MeasurementSet models"
175        } else {
176            "PlotMS virtual MeasurementSet models are pending #1032"
177        };
178        refuse_selected_virtual_fields(ms, rows, message)?;
179    }
180    Ok(None)
181}

183/// A selected virtual-model field overrides stored cells, including its
184/// fallback model in windows without a recipe.
185fn refuse_selected_virtual_fields(
186    ms: &MeasurementSet,
187    rows: &[ResolvedMsSelectionRow],
188    message: &str,
189) -> Result<(), String> {
190    for field in virtual_model_fields(ms) {
191        if rows.iter().any(|row| row.field_id == field)
192            && virtual_model(ms, field)
193                .map_err(|error| error.to_string())?
194                .is_some()
195        {
196            return Err(message.into());
197        }
198    }
199    Ok(())
200}

202/// Share one unit point model cell per selected data description.
203pub(super) fn default_cells(
204    ms: &MeasurementSet,
205    rows: &[usize],
206) -> Result<Vec<Arc<ArrayValue>>, String> {
207    refuse_matching_virtual_model(ms, rows)?;
208    let main = ms.main_table();
209    let descriptions = ms.data_description().map_err(|error| error.to_string())?;
210    let spectral_window = ms.spectral_window().map_err(|error| error.to_string())?;
211    let polarization = ms.polarization().map_err(|error| error.to_string())?;
212    let index =
213        |value: i32| usize::try_from(value).map_err(|_| format!("invalid MS index {value}"));
214    let mut cells = HashMap::<i32, Arc<ArrayValue>>::new();
215    selected_i32_values(main, "DATA_DESC_ID", rows)?
216        .into_iter()
217        .map(|ddid| {
218            if let Some(cell) = cells.get(&ddid) {
219                return Ok(Arc::clone(cell));
220            }
221            let description = index(ddid)?;
222            let spw_id = descriptions
223                .spectral_window_id(description)
224                .map_err(|error| error.to_string())?;
225            let pol_id = descriptions
226                .polarization_id(description)
227                .map_err(|error| error.to_string())?;
228            let channels = spectral_window
229                .num_chan(index(spw_id)?)
230                .map_err(|error| error.to_string())?;
231            let correlations = polarization
232                .corr_type(index(pol_id)?)
233                .map_err(|error| error.to_string())?;
234            let values =
235                Array2::from_shape_fn((correlations.len(), index(channels)?), |(corr, _)| {
236                    initial_model_visibility(correlations[corr])
237                });
238            let cell = Arc::new(ArrayValue::Complex32(values.into_dyn()));
239            cells.insert(ddid, Arc::clone(&cell));
240            Ok(cell)
241        })
242        .collect()
243}

245fn refuse_matching_virtual_model(ms: &MeasurementSet, rows: &[usize]) -> Result<(), String> {
246    let fields = virtual_model_fields(ms);
247    if fields.is_empty() {
248        return Ok(());
249    }
250    let selected = selected_model_combinations(ms, rows).map_err(|error| error.to_string())?;
251    for field in fields {
252        let combinations = selected
253            .index_combination
254            .iter()
255            .copied()
256            .filter(|combination| combination[0] == field)
257            .collect::<BTreeSet<_>>();
258        if combinations.is_empty() {
259            continue;
260        }
261        let Some(model) = virtual_model(ms, field).map_err(|error| error.to_string())? else {
262            continue;
263        };
264        let matches_selection = if matches!(
265            model.record.get("version"),
266            Some(Value::Scalar(ScalarValue::Int32(2)))
267        ) {
268            VirtualModelEntry::entries(&model.record)
269                .map_err(|error| error.to_string())?
270                .into_iter()
271                .flat_map(|entry| entry.index_combination)
272                .any(|combination| combinations.contains(&combination))
273        } else {
274            matches_legacy_model(&model.record, &combinations)?
275        };
276        if matches_selection {
277            return Err("PlotMS virtual MeasurementSet models are pending #1032".into());
278        }
279    }
280    Ok(())
281}

283fn matches_legacy_model(
284    record: &RecordValue,
285    combinations: &BTreeSet<[i32; 4]>,
286) -> Result<bool, String> {
287    for (count_name, prefix) in [("numft", "ft"), ("numcl", "cl")] {
288        let Some(Value::Scalar(ScalarValue::Int32(count))) = record.get(count_name) else {
289            continue;
290        };
291        for index in 0..*count {
292            let name = format!("{prefix}_{index}");
293            let Some(Value::Record(entry)) = record.get(&name) else {
294                return Err(format!("legacy virtual model has no {name} record"));
295            };
296            let ids = |name| match entry.get(name) {
297                Some(Value::Array(ArrayValue::Int32(values))) if values.ndim() == 1 => Ok(values),
298                _ => Err(format!("legacy virtual model has no {name} vector")),
299            };
300            let fields = ids("fields")?;
301            let windows = ids("spws")?;
302            if combinations.iter().any(|[field, window, _, _]| {
303                fields.iter().any(|value| value == field)
304                    && windows.iter().any(|value| value == window)
305            }) {
306                return Ok(true);
307            }
308        }
309    }
310    Ok(false)
