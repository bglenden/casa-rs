13#[derive(Clone, Copy, Debug)]
14pub(crate) struct NativePreparationPlan {
15    pub(super) workers: usize,
16    pub(super) heap_bytes: u64,
17    block_rows: usize,
18    worker_rows: usize,
19    channels: usize,
20    correlations: usize,
21}
22
23impl NativePreparationPlan {
24    pub(crate) fn new(
25        problem: &CompiledProblem,
26        weighting: &WeightingPlan,
27        workers: usize,
28        block_rows: usize,
29    ) -> Result<Self, WeightingError> {
30        let [source] = problem.selected_observation().read_set().sources() else {
31            return Err(WeightingError::ProblemMismatch);
32        };
33        let ([spw], [pol]) = (
34            source.selection().spectral_windows(),
35            source.selection().correlations(),
36        ) else {
37            return Err(WeightingError::ProblemMismatch);
38        };
39        if workers == 0 || block_rows == 0 {
40            return Err(WeightingError::ResidencyOverflow);
41        }
42        let channels = spw.channel_indices().len();
43        let correlations = pol.products().len();
44        let worker_rows = block_rows.div_ceil(workers);
45        let per_worker = NativeWeightingPreparation::worker_required_bytes(
46            problem,
47            weighting,
48            worker_rows,
49            channels,
50            correlations,
51        )
52        .map_err(|_| WeightingError::ResidencyOverflow)?
53        .checked_add(weighting.planned_residency().spectral_cache_bytes())
54        .and_then(|n| n.checked_add(SelectedObservationProjector::required_bytes(problem)?))
55        .and_then(|n| n.checked_add(size_of::<Worker<'_>>() + size_of::<&NativeBlock>()))
56        .ok_or(WeightingError::ResidencyOverflow)?;
57        let owned = per_worker
58            .checked_mul(workers)
59            .and_then(|n| {
60                n.checked_add(
61                    NativeWeightingPreparation::coordinator_required_bytes(weighting).ok()?,
62                )
63            })
64            .and_then(|n| {
65                n.checked_add(size_of::<
66                    NativeKernel<'_, fn(&[&NativeBlock], &NativeLayout) -> io::Result<()>>,
67                >())
68            })
69            .ok_or(WeightingError::ResidencyOverflow)? as u64;
70        let stacks = if workers > 1 {
71            workers as u64 * crate::bounded_stream::BOUNDED_WORKER_STACK_BYTES as u64
72        } else {
73            0
74        };
75        let heap_bytes = crate::bounded_stream::BoundedKernelPlan::new::<(), ()>(workers, 1, 0)
76            .ok()
77            .and_then(|plan| plan.capacity_bytes().checked_sub(stacks))
78            .and_then(|bytes| bytes.checked_add(owned))
79            .ok_or(WeightingError::ResidencyOverflow)?;
80        Ok(Self {
81            workers,
82            heap_bytes,
83            block_rows,
84            worker_rows,
85            channels,
86            correlations,
87        })
88    }
89}

98struct NativeKernel<'a, F> {
99    problem: &'a CompiledProblem,
100    weighting_plan: &'a WeightingPlan,
101    consumer: SelectedObservationBlockConsumer<'a>,
102    weights: NativeWeightingPreparation,
103    workers: Vec<Worker<'a>>,
104    plan: NativePreparationPlan,
105    batch_rows: usize,
106    emit: F,
107    inspection_nanos: u128,
108    preparation_nanos: u128,
109    ordered_commit_nanos: u128,
110}
111
112impl<F> NativeKernel<'_, F>
113where
114    F: FnMut(&[&NativeBlock], &NativeLayout) -> io::Result<()>,
115{
116    fn flush(&mut self) -> Result<(), WeightingBlockKernelError<io::Error>> {
117        if self.batch_rows == 0 {
118            return Ok(());
119        }
120        let started = Instant::now();
121        for worker in self.workers.iter_mut().filter(|worker| worker.rows != 0) {
122            let native = worker.native.as_mut().expect("populated worker");
123            native
124                .finish_batch()
125                .map_err(WeightingBlockKernelError::Consumer)?;
126            self.weights
127                .commit(native)
128                .map_err(WeightingBlockKernelError::Consumer)?;
129        }
130        {
131            let parts: Vec<_> = self
132                .workers
133                .iter()
134                .filter(|worker| worker.rows != 0)
135                .map(|worker| worker.native.as_ref().expect("populated worker").block())
136                .collect();
137            let layout = self.workers[0]
138                .native
139                .as_ref()
140                .expect("first populated worker")
141                .layout();
142            (self.emit)(&parts, layout).map_err(WeightingBlockKernelError::Consumer)?;
143        }
144        for worker in &mut self.workers {
145            worker.rows = 0;
146        }
147        self.batch_rows = 0;
148        self.ordered_commit_nanos += started.elapsed().as_nanos();
149        Ok(())
150    }
151
152    fn consume(
153        &mut self,
154        storage: &SelectedObservationBlock,
155        execution: crate::bounded_stream::BoundedExecution<'_>,
156    ) -> Result<(), WeightingBlockKernelError<io::Error>> {
157        let runs = storage.selected_run_count().map_err(|e| {
158            WeightingBlockKernelError::Traversal(SelectedObservationTraversalError::Source(e))
159        })?;
160        let channels = storage.selected_channels_per_row().map_err(|e| {
161            WeightingBlockKernelError::Traversal(SelectedObservationTraversalError::Source(e))
162        })?;
163        if channels != self.plan.channels || runs % channels != 0 {
164            return Err(WeightingBlockKernelError::Consumer(io::Error::other(
165                "native source row shape mismatch",
166            )));
167        }
168        let started = Instant::now();
169        self.consumer
170            .inspect_block_range(storage, 0..runs)
171            .map_err(|e| WeightingBlockKernelError::Traversal(widen_terminal_traversal_error(e)))?;
172        self.inspection_nanos += started.elapsed().as_nanos();
173        let rows = runs / channels;
174        let mut first = 0;
175        while first < rows {
176            let take = (rows - first).min(self.plan.block_rows - self.batch_rows);
177            let batch_first = self.batch_rows;
178            let batch_end = batch_first + take;
179            let plan = self.plan;
180            let problem = self.problem;
181            let weights = &self.weights;
182            let weighting_plan = self.weighting_plan;
183            let started = Instant::now();
184            execution.for_each_mut(&mut self.workers, |ordinal, worker| {
185                let lower = (ordinal * plan.worker_rows).max(batch_first);
186                let upper = ((ordinal + 1) * plan.worker_rows).min(batch_end);
187                if lower >= upper {
188                    return Ok(());
189                }
190                let source_first = first + lower - batch_first;
191                let source_last = source_first + upper - lower;
192                if worker.rows == 0
193                    && let Some(native) = &mut worker.native
194                {
195                    native
196                        .begin_batch()
197                        .map_err(WeightingBlockKernelError::Consumer)?;
198                }
199                worker
200                    .projector
201                    .visit_block_range(
202                        problem,
203                        storage,
204                        source_first * channels..source_last * channels,
205                        |run| {
206                            if let Some(reported) = run.samples().next() {
207                                if worker.native.is_none() {
208                                    let selected =
209                                        problem.selected_observation().read_set().sources()[0]
210                                            .selection();
211                                    let layout = NativeLayout::new(
212                                        reported.selected().address(),
213                                        selected.spectral_windows()[0].channel_indices().to_vec(),
214                                        selected.correlations()[0]
215                                            .products()
216                                            .iter()
217                                            .map(|c| (c.correlation_index(), c.correlation_type()))
218                                            .collect(),
219                                    )
220                                    .map_err(ReplayCallbackError::Consumer)?;
221                                    let mut native = weights
222                                        .worker(
223                                            problem,
224                                            weighting_plan,
225                                            layout,
226                                            NativeBlock::new(
227                                                plan.worker_rows,
228                                                plan.channels,
229                                                plan.correlations,
230                                            )
231                                            .map_err(ReplayCallbackError::Consumer)?,
232                                            problem.numerics().finite_values(),
233                                        )
234                                        .map_err(ReplayCallbackError::Consumer)?;
235                                    native
236                                        .begin_batch()
237                                        .map_err(ReplayCallbackError::Consumer)?;
238                                    worker.native = Some(native);
239                                }
240                                let contributions = worker
241                                    .spectral
242                                    .compile(reported.selected(), reported.spectral_evaluation())
243                                    .map_err(ReplayCallbackError::Owner)?;
244                                worker
245                                    .native
246                                    .as_mut()
247                                    .expect("initialized above")
248                                    .consume_channel(
249                                        run.row(),
250                                        run.channel(),
251                                        run.correlations(),
252                                        reported.spectral_evaluation().row_geometry().ok_or(
253                                            ReplayCallbackError::Owner(
254                                                WeightingError::RowSpectralGeometryMismatch,
255                                            ),
256                                        )?,
257                                        reported.spectral_evaluation().output_frame().centre_hz(),
258                                        contributions,
259                                    )
260                                    .map_err(ReplayCallbackError::Consumer)?;
261                            }
262                            Ok::<_, ReplayCallbackError<io::Error>>(())
263                        },
264                    )
265                    .map_err(WeightingBlockKernelError::Traversal)?;
266                worker.rows += upper - lower;
267                Ok(())
268            })?;
269            self.preparation_nanos += started.elapsed().as_nanos();
270            first += take;
271            self.batch_rows += take;
272            if self.batch_rows == self.plan.block_rows {
273                self.flush()?;
274            }
275        }
276        Ok(())
277    }
278}

280impl<'a, F> PartitionedKernel<SelectedObservationBlock> for NativeKernel<'a, F>
281where
282    F: FnMut(&[&NativeBlock], &NativeLayout) -> io::Result<()> + Send + Sync,
283{
284    type Partition = ();
285    type Partial = ();
286    type Completion =
287        WeightingBlockKernelCompletion<'a, (WeightingAlgorithmState, WeightingReplaySummary)>;
288    type Error = WeightingBlockKernelError<io::Error>;
289
290    fn partition_count(
291        &self,
292        _: BlockIdentity,
293        _: &SelectedObservationBlock,
294    ) -> Result<usize, Self::Error> {
295        Ok(1)
296    }
297    fn partition(
298        &self,
299        _: BlockIdentity,
300        _: &SelectedObservationBlock,
301        _: usize,
302    ) -> Result<KernelPartition<()>, Self::Error> {
303        Ok(KernelPartition::exclusive(0, 0, ()))
304    }
305    fn execute(
306        &self,
307        _: WorkIdentity,
308        _: &SelectedObservationBlock,
309        _: &(),
310    ) -> Result<(), Self::Error> {
311        Ok(())
312    }
313    fn commit(
314        &mut self,
315        _: WorkIdentity,
316        storage: &SelectedObservationBlock,
317        _: (),
318        execution: crate::bounded_stream::BoundedExecution<'_>,
319    ) -> Result<(), Self::Error> {
320        self.consume(storage, execution)
321    }
322    fn complete(
323        mut self,
324        _: crate::bounded_stream::BoundedExecution<'_>,
325    ) -> Result<Self::Completion, Self::Error> {
326        self.flush()?;
327        let weights = self
328            .weights
329            .finish()
330            .map_err(WeightingBlockKernelError::Consumer)?;
331        let samples = weights.1.sample_count();
332        eprintln!(
333            "imaging_native_preparation_summary workers={} samples={samples} indexed_runs=0 index_payload_bytes=0 retained_weighted_samples=0 inspection_nanos={} preparation_nanos={} ordered_commit_nanos={} planned_heap_bytes={}",
334            self.plan.workers,
335            self.inspection_nanos,
336            self.preparation_nanos,
337            self.ordered_commit_nanos,
338            self.plan.heap_bytes
339        );
340        Ok(WeightingBlockKernelCompletion {
341            consumer: self.consumer,
342            weights,
343            continuum: None,
344            spectral_support_sample_count: 0,
345            prepared_samples: samples,
346        })
347    }
348}

350pub(super) fn execute<'a, F>(
351    problem: &'a CompiledProblem,
352    selected: BoundSelectedObservation,
353    plan: BoundedStreamPlan,
354    preparation: NativePreparationPlan,
355    weighting: &'a WeightingPlan,
356    emit: F,
357) -> Result<
358    CompletedWeightingBlockStream<'a, (WeightingAlgorithmState, WeightingReplaySummary)>,
359    WeightingBlockStreamFailure<io::Error>,
360>
361where
362    F: FnMut(&[&NativeBlock], &NativeLayout) -> io::Result<()> + Send + Sync,
363{
364    let (source, consumer) = selected.into_block_stream(problem).map_err(|e| {
365        WeightingReplayError::Traversal(SelectedObservationTraversalError::Binding(e))
366    })?;
367    let workers = (0..preparation.workers)
368        .map(|_| {
369            Ok(Worker {
370                projector: SelectedObservationProjector::new(problem),
371                spectral: WeightingSpectralCache::new(problem)?,
372                native: None,
373                rows: 0,
374            })
375        })
376        .collect::<Result<Vec<_>, WeightingError>>()
377        .map_err(WeightingReplayError::Owner)?;
378    let outcome = execute_bounded(
379        plan,
380        0,
381        SelectedBlockSource { source },
382        NativeKernel {
383            problem,
384            weighting_plan: weighting,
385            consumer,
386            weights: NativeWeightingPreparation::new(problem, weighting)
387                .map_err(WeightingReplayError::Consumer)?,
388            workers,
389            plan: preparation,
390            batch_rows: 0,
391            emit,
392            inspection_nanos: 0,
393            preparation_nanos: 0,
394            ordered_commit_nanos: 0,
395        },
396    )
397    .map_err(|failure| WeightingBlockStreamFailure {
398        error: Box::new(map_bounded_stream_error(*failure.cause)),
399        measurements: failure.measurements,
400    })?;
401    let mut terminal = outcome.source_completion;
402    terminal
403        .record_runtime_residency(
404            outcome.measurements.peak_live_source_blocks,
405            outcome.measurements.peak_live_source_current_bytes,
406            outcome.measurements.peak_live_source_capacity_bytes,
407        )
408        .map_err(|e| {
409            WeightingReplayError::Traversal(SelectedObservationTraversalError::Source(e))
410        })?;
411    let WeightingBlockKernelCompletion {
412        consumer,
413        weights,
414        prepared_samples,
415        ..
416    } = outcome.kernel_completion;
417    let (selected, owner_completion) = consumer
418        .complete(terminal)
419        .map_err(|e| WeightingReplayError::Traversal(widen_terminal_traversal_error(e)))?;
420    Ok(CompletedWeightingBlockStream {
421        selected,
422        owner_completion,
423        weights,
424        continuum: None,
425        spectral_support_sample_count: 0,
426        prepared_samples,
427        measurements: outcome.measurements,
428    })
