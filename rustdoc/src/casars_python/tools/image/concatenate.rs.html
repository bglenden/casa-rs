18fn filename_argument(value: Option<&Bound<'_, PyAny>>) -> PyResult<Variant> {
19    let Some(value) = value else {
20        return Ok(Variant::Empty);
21    };
22    if let Ok(array) = value.cast::<PyUntypedArray>() {
23        return match array.dtype().kind() {
24            b'b' | b'O' => Ok(Variant::Empty),
25            b'U' if array.shape().iter().filter(|&&length| length > 1).count() > 1 => {
26                Ok(Variant::Empty)
27            }
28            b'S' | b'U' => flatten_filenames(value),
29            _ => value.extract(),
30        };
31    }
32    let captured: Variant = value.extract()?;
33    if matches!(captured, Variant::Other(_))
34        && (value.is_instance_of::<PyList>() || value.is_instance_of::<PyTuple>())
35    {
36        let array = value
37            .py()
38            .import("numpy")?
39            .call_method1("asarray", (value, "object"))?;
40        return flatten_filenames(&array);
41    }
42    Ok(captured)
43}

45fn flatten_filenames(array: &Bound<'_, PyAny>) -> PyResult<Variant> {
46    let flattened = array
47        .call_method1("ravel", ("F",))?
48        .call_method0("tolist")?;
49    let mut names = Vec::new();
50    for item in flattened.try_iter()? {
51        match item?.extract()? {
52            Variant::Text(name) => names.push(name),
53            _ => return Ok(Variant::Empty),
54        }
55    }
56    Ok(Variant::Texts(names))
57}

59fn filenames(py: Python<'_>, files: &Variant) -> PyResult<Vec<PathBuf>> {
60    let names = match files {
61        Variant::Empty => Vec::new(),
62        Variant::Text(text) => {
63            let delimiter = if text.contains(',') { ',' } else { ' ' };
64            text.split(delimiter)
65                .filter(|word| !word.is_empty())
66                .map(str::to_owned)
67                .collect()
68        }
69        Variant::Texts(names) => names.clone(),
70        _ => return Err(raised("Exception: Unrecognized infiles datatype.")),
71    };
72    let glob = py.import("glob")?;
73    let path_module = py.import("os.path")?;
74    let mut result = Vec::new();
75    for name in names {
76        let name = path_module.call_method1("expandvars", (name,))?;
77        let pattern: String = path_module.call_method1("expanduser", (name,))?.extract()?;
78        if let Some(parent) = Path::new(&pattern)
79            .parent()
80            .filter(|path| !path.as_os_str().is_empty())
81            && !parent.to_string_lossy().contains(['*', '?', '['])
82            && !parent.is_dir()
83        {
84            return Err(raised(format!(
85                "DirectoryIterator: error on directory {}: No such file or directory",
86                parent.display()
87            )));
88        }
89        let matches: Vec<String> = glob.call_method1("glob", (pattern,))?.extract()?;
90        for entry in matches {
91            result.push(std::path::absolute(entry).map_err(raised)?);
92        }
93    }
94    if result.len() < 2 {
95        return Err(raised(
96            "Exception: You must provide at least two images to concatentate.",
97        ));
98    }
99    Ok(result)
100}

102#[pymethods]
103impl PyImageTool {
104    /// Combine images along one axis and return the resulting image tool.
105    #[allow(clippy::too_many_arguments)]
106    #[pyo3(signature = (outfile="", infiles=None, axis=-1, relax=false, tempclose=true, overwrite=false, reorder=false, mode="paged"), text_signature = "($self, outfile='', infiles=[], axis=-1, relax=False, tempclose=True, overwrite=False, reorder=False, mode='paged')")]
107    fn imageconcat(
108        &self,
109        py: Python<'_>,
110        outfile: &str,
111        infiles: Option<&Bound<'_, PyAny>>,
112        axis: i64,
113        relax: bool,
114        tempclose: bool,
115        overwrite: bool,
116        reorder: bool,
117        mode: &str,
118    ) -> PyResult<Py<PyImageTool>> {
119        let files = filename_argument(infiles)?;
120        let paths = filenames(py, &files)?;
121        let mode_letter = mode.as_bytes().first().map(u8::to_ascii_lowercase);
122        let expanded = if mode_letter != Some(b'p') {
123            resolve_image_path(py, outfile)?
124        } else {
125            std::borrow::Cow::Borrowed(outfile)
126        };
127        let files_history = match &files {
128            Variant::Text(text) => format!("'{text}'"),
129            files => files.describe(),
130        };
131        let (result, logs) = py.detach(|| {
132            let history = self.state().history;
133            concatenate_casa_images(
134                outfile,
135                expanded.as_ref(),
136                &paths,
137                &files_history,
138                axis,
139                relax,
140                tempclose,
141                overwrite,
142                reorder,
143                mode,
144                history,
145                external_open_check(),
146            )
147            .map_err(raised)
148        })?;
149        emit_tool_logs(py, &logs)?;
150        Py::new(py, Self::attached(result))
151    }
152}
