2//! CASA `GridFT`'s and `MosaicFT`'s uvw reprojections, for callers outside
3//! the imaging read path.
4
5use casa_types::measures::direction::{DirectionRef, MDirection};
6
7use super::engine::{CasaGirarUvwTransform, CasaUvwMachine};
8use crate::error::MsResult;
9
10/// CASA `GridFT`'s rotation of rows from one phase centre to an image's
11/// tangent point, built once and applied to every row
12/// (`FTMachine::rotateUVW`, with a `UVWMachine` that neither projects nor
13/// treats the array as east–west), or `MosaicFT`'s reprojection
14/// ([`Self::mosaic_between_j2000_directions`]).
15///
16/// `GridFT` rotates every row, even when the two directions are the same:
17/// `FTMachine::initMaps` resets the last field it saw, so the first row it
18/// meets turns the rotation on. The rotation of two equal directions is the
19/// identity up to rounding.
20///
21/// Cf. casacore `UVWMachine::convertUVW(Double&, Vector<Double>&)`.
22#[derive(Debug, Clone, Copy)]
23pub struct GridFtProjection {
24    machine: Reprojection,
25}
26
27#[derive(Debug, Clone, Copy)]
28enum Reprojection {
29    /// `FTMachine::rotateUVW`.
30    Rotate(CasaUvwMachine),
31    /// `FTMachine::girarUVW`.
32    Girar(CasaGirarUvwTransform),
33}
34
35impl GridFtProjection {
36    /// `MosaicFT`'s reprojection of rows phased on `phase_centre_rad` to an
37    /// image whose tangent point is `tangent_point_rad`, both J2000
38    /// `[ra, dec]` in radians (`FTMachine::girarUVW`): the image-centre
39    /// rotation forms the new u and v from the old u and v alone, so w does
40    /// not reach them, and the phase shift is taken from the new u and v.
41    #[must_use]
42    pub fn mosaic_between_j2000_directions(
43        phase_centre_rad: [f64; 2],
44        tangent_point_rad: [f64; 2],
45    ) -> Self {
46        let [source, target] = [phase_centre_rad, tangent_point_rad]
47            .map(|[ra, dec]| MDirection::from_angles(ra, dec, DirectionRef::J2000));
48        Self {
49            machine: Reprojection::Girar(CasaGirarUvwTransform::new(&source, &target)),
50        }
51    }
52
53    /// The rotation from rows phased on `phase_centre_rad` to an image whose
54    /// tangent point is `tangent_point_rad`, both J2000 `[ra, dec]` in
55    /// radians.
56    #[must_use]
57    pub fn between_j2000_directions(
58        phase_centre_rad: [f64; 2],
59        tangent_point_rad: [f64; 2],
60    ) -> Self {
61        let source = MDirection::from_angles(
62            phase_centre_rad[0],
63            phase_centre_rad[1],
64            DirectionRef::J2000,
65        );
66        let target = MDirection::from_angles(
67            tangent_point_rad[0],
68            tangent_point_rad[1],
69            DirectionRef::J2000,
70        );
71        Self {
72            machine: Reprojection::Rotate(CasaUvwMachine::new(&target, &source)),
73        }
74    }
75
76    /// A row's raw `UVW` in metres reprojected to the tangent point, and
77    /// the phase shift in metres the gridder applies (`−dphase`, the
78    /// measurement operator's `phase_shift_m`).
79    ///
80    /// CASA's `UVWMachine` works in the opposite sign of u and v to the
81    /// MeasurementSet's; the reprojection is applied in its convention and
82    /// the result returned in the MeasurementSet's.
83    #[must_use]
84    pub fn apply(&self, raw_uvw_m: [f64; 3]) -> ([f64; 3], f64) {
85        match self.machine {
86            Reprojection::Rotate(machine) => {
87                let casa_input = [-raw_uvw_m[0], -raw_uvw_m[1], raw_uvw_m[2]];
88                let (casa_output, casa_dphase_m) = machine.convert_uvw(casa_input);
89                (
90                    [-casa_output[0], -casa_output[1], casa_output[2]],
91                    -casa_dphase_m,
92                )
93            }
94            Reprojection::Girar(transform) => {
95                let (uvw_m, casa_dphase_m) = transform.apply(raw_uvw_m);
96                (uvw_m, -casa_dphase_m)
97            }
98        }
99    }
100}

102/// Rotate a row's raw `UVW` from its phase centre to an image's tangent
103/// point, as CASA's `GridFT` does before it degrids or grids the row
104/// (`FTMachine::rotateUVW`, with a `UVWMachine` that neither projects nor
105/// treats the array as east–west). Both directions are J2000 `[ra, dec]` in
106/// radians. A caller rotating many rows between the same two directions
107/// builds one [`GridFtProjection`] instead.
108///
109/// Returns the rotated uvw in metres and the phase shift in metres the
110/// gridder applies (`−dphase`, the measurement operator's
111/// `phase_shift_m`).
112///
113/// # Errors
114///
115/// A direction that does not convert.
116pub fn gridft_uvw_between_j2000_directions(
117    raw_uvw_m: [f64; 3],
118    phase_centre_rad: [f64; 2],
119    tangent_point_rad: [f64; 2],
120) -> MsResult<([f64; 3], f64)> {
121    Ok(
122        GridFtProjection::between_j2000_directions(phase_centre_rad, tangent_point_rad)
123            .apply(raw_uvw_m),
124    )
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130
131    /// The reusable projection is the per-row helper the imaging read path
132    /// uses, bit for bit, and the rotation between equal directions keeps
133    /// uvw to rounding with no phase shift.
134    #[test]
135    fn the_projection_is_the_per_row_rotation_bit_for_bit() {
136        let phase_centre = [2.6179938779914944, -0.5235987755982988];
137        let tangent = [2.6180011, -0.5236035];
138        let projection = GridFtProjection::between_j2000_directions(phase_centre, tangent);
139        for uvw in [
140            [120.5, -48.25, 3.5],
141            [-1_500.0, 732.125, -40.0],
142            [0.0, 0.0, 0.0],
143        ] {
144            let engine = super::super::engine::gridft_projection_between_j2000_directions(
145                uvw,
146                phase_centre,
147                tangent,
148                false,
149            )
150            .expect("rotation");
151            let projected = projection.apply(uvw);
152            assert_eq!(projected.0.map(f64::to_bits), engine.0.map(f64::to_bits));
153            assert_eq!(projected.1.to_bits(), engine.1.to_bits());
154            assert_eq!(
155                gridft_uvw_between_j2000_directions(uvw, phase_centre, tangent).expect("helper"),
156                projected
157            );
158        }
159        let same = GridFtProjection::between_j2000_directions(phase_centre, phase_centre);
160        let (uvw, shift) = same.apply([120.5, -48.25, 3.5]);
161        for (rotated, raw) in uvw.iter().zip([120.5, -48.25, 3.5]) {
162            assert!((rotated - raw).abs() < 1e-12, "{rotated} against {raw}");
163        }
164        assert!(shift.abs() < 1e-12, "{shift}");
165        assert_eq!(same.apply([0.0; 3]), ([0.0; 3], 0.0));
166    }
167
168    /// `MosaicFT`'s reprojection gives CASA's `girarUVW` reference row (the
169    /// engine's field-to-field law), its phase as the gridder's `−dphase`,
170    /// and departs from `GridFT`'s rotation by about `w · Δθ` in u.
171    #[test]
172    fn the_mosaic_projection_is_casas_girar_uvw() {
173        let phase_centre = [3.149_884_015_678_791, -0.329_299_608_408_283];
174        let tangent = [3.149_807_242_890_337, -0.329_299_608_769_085_2];
175        let raw = [
176            4.324_749_385_347_82,
177            -121.757_651_483_641_07,
178            -15.081_385_958_149_532,
179        ];
180        let mosaic = GridFtProjection::mosaic_between_j2000_directions(phase_centre, tangent);
181        let (uvw, shift) = mosaic.apply(raw);
182        for (got, want) in uvw.iter().zip([
183            4.321_726_518_390_6,
184            -121.757_758_815_853_26,
185            -15.081_699_947_780_477,
186        ]) {
187            assert!((got - want).abs() < 1e-12, "{got} against {want}");
188        }
189        assert!(
190            (shift - 0.000_314_029_427_521_723_44).abs() < 1e-12,
191            "{shift}"
192        );
193        let (rotated, _) =
194            GridFtProjection::between_j2000_directions(phase_centre, tangent).apply(raw);
195        let w_share = (raw[2] * (tangent[0] - phase_centre[0]) * tangent[1].cos()).abs();
196        assert!(
197            (rotated[0] - uvw[0]).abs() > 0.5 * w_share,
198            "{rotated:?} against {uvw:?}"
199        );
200    }
