2//! Array column sources: the data-manager groups a save writes from a
3//! [`GeneratedArrayColumn`], and the checks on its cells.
4//!
5//! A group holding a generated array column is written from its column
6//! sources, a cell at a time, by the StandardStMan or IncrementalStMan
7//! group writer, so its rows are never materialized as records. That needs
8//! every column of the group overridden; other storage managers, and
9//! IncrementalStMan's variable-shape arrays and fixed-shape arrays without
10//! the Direct option, refuse the override.
11
12use std::borrow::Cow;
13
14use casa_types::{PrimitiveType, Value};

16use super::StorageError;
17use super::incremental_stman::refuse_ism_indirect_fixed_arrays;
18use super::scalar_sources::{ColumnSource, ColumnSources, scalar_column_source};
19use super::table_control::ColumnDescContents;
20use crate::table::{ColumnOverride, ColumnOverrides, DataManagerKind, GeneratedArrayColumn};

22/// A generated array column with the type and shape its cells must have.
23pub(crate) struct GeneratedArraySource<'a> {
24    column: &'a GeneratedArrayColumn,
25    name: String,
26    data_type: Option<PrimitiveType>,
27    /// The fixed cell shape, or `None` for a variable-shape column.
28    shape: Option<Vec<usize>>,
29    /// The dimensionality a variable-shape column fixes, if any.
30    ndim: Option<usize>,
31}
32
33impl<'a> GeneratedArraySource<'a> {
34    fn new(column: &'a GeneratedArrayColumn, desc: &ColumnDescContents) -> Self {
35        let shape =
36            (!desc.shape.is_empty()).then(|| desc.shape.iter().map(|&dim| dim as usize).collect());
37        Self {
38            column,
39            name: desc.col_name.clone(),
40            data_type: desc.primitive_type,
41            shape,
42            ndim: usize::try_from(desc.nrdim).ok().filter(|&ndim| ndim > 0),
43        }
44    }
45
46    /// The cell in `row`, refused when its type or shape does not fit the
47    /// column.
48    pub(crate) fn cell(&self, row: usize) -> Result<Option<Cow<'a, Value>>, StorageError> {
49        let Some(cell) = self.column.cell(row) else {
50            return Ok(None);
51        };
52        let Value::Array(array) = cell.as_ref() else {
53            unreachable!("a generated array cell is an array");
54        };
55        let mismatch = |what: String| {
56            Err(StorageError::FormatMismatch(format!(
57                "generated array column {} row {row}: {what}",
58                self.name
59            )))
60        };
61        if self.data_type != Some(array.primitive_type()) {
62            return mismatch(format!(
63                "{:?} cell in a {:?} column",
64                array.primitive_type(),
65                self.data_type
66            ));
67        }
68        if let Some(shape) = &self.shape {
69            if array.shape() != shape.as_slice() {
70                return mismatch(format!(
71                    "shape {:?} in a column of shape {shape:?}",
72                    array.shape()
73                ));
74            }
75        } else if let Some(ndim) = self.ndim
76            && array.ndim() != ndim
77        {
78            return mismatch(format!("{} axes in a column of {ndim}", array.ndim()));
79        }
80        Ok(Some(cell))
81    }
82}

84/// The column sources of a data-manager group that holds a generated array
85/// column, or `None` when it holds none.
86///
87/// # Errors
88///
89/// The group's storage manager is not StandardStMan or IncrementalStMan; a
90/// column of the group has no override, is deferred or is a record column;
91/// an override does not fit its column; or an IncrementalStMan column has
92/// arrays casacore keeps in its `table.fNi` array file, which its writer
93/// does not store: variable-shape arrays, and fixed-shape arrays without the
94/// Direct option ([`StorageError::IsmIndirectArrayColumn`]).
95pub(super) fn generated_array_group_sources<'a>(
96    dm_kind: DataManagerKind,
97    group_col_descs: &[ColumnDescContents],
98    column_overrides: &'a ColumnOverrides,
99    row_count: usize,
100) -> Result<Option<ColumnSources<'a>>, StorageError> {
101    let Some(generated) = group_col_descs.iter().find(|desc| {
102        matches!(
103            column_overrides.get(&desc.col_name),
104            Some(ColumnOverride::GeneratedArray(_))
105        )
106    }) else {
107        return Ok(None);
108    };
109    let refuse = |message: String| Err(StorageError::FormatMismatch(message));
110    if !matches!(
111        dm_kind,
112        DataManagerKind::StandardStMan | DataManagerKind::IncrementalStMan
113    ) {
114        return refuse(format!(
115            "generated array column {} needs StandardStMan or IncrementalStMan, not {dm_kind:?}",
116            generated.col_name
117        ));
118    }
119    let mut columns = Vec::with_capacity(group_col_descs.len());
120    for desc in group_col_descs {
121        let Some(override_value) = column_overrides.get(&desc.col_name) else {
122            return refuse(format!(
123                "generated array column {} shares its {dm_kind:?} group with column {}, \
124                 which has no override; every column of the group must be overridden",
125                generated.col_name, desc.col_name
126            ));
127        };
128        if desc.is_record() {
129            return refuse(format!(
130                "record column {} cannot share a group with generated array column {}",
131                desc.col_name, generated.col_name
132            ));
133        }
134        if desc.is_array && desc.shape.is_empty() && dm_kind == DataManagerKind::IncrementalStMan {
135            return refuse(format!(
136                "IncrementalStMan does not write variable-shape array column {}",
137                desc.col_name
138            ));
139        }
140        if dm_kind == DataManagerKind::IncrementalStMan {
141            refuse_ism_indirect_fixed_arrays(std::slice::from_ref(desc))?;
142        }
143        let source = if desc.is_array {
144            match override_value {
145                ColumnOverride::Values(values) => ColumnSource::Cells(values),
146                ColumnOverride::GeneratedArray(column) => {
147                    ColumnSource::GeneratedArray(GeneratedArraySource::new(column, desc))
148                }
149                ColumnOverride::GeneratedScalar(_) | ColumnOverride::StreamedScalar(_) => {
150                    return refuse(format!(
151                        "array column {} cannot take a scalar override",
152                        desc.col_name
153                    ));
154                }
155                ColumnOverride::Deferred => return Err(deferred(desc, generated)),
156            }
157        } else {
158            scalar_column_source(desc, override_value)?.ok_or_else(|| deferred(desc, generated))?
159        };
160        columns.push(source);
161    }
162    Ok(Some(ColumnSources::new(row_count, columns)))
163}
164
165fn deferred(desc: &ColumnDescContents, generated: &ColumnDescContents) -> StorageError {
166    StorageError::FormatMismatch(format!(
167        "deferred column {} cannot share a group with generated array column {}",
168        desc.col_name, generated.col_name
169    ))
170}
