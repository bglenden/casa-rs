44struct TimeAverageStatistics {
45    partition: [i32; 4],
46    first: Integration,
47    last: Integration,
48    last_usable: Option<Integration>,
49    weights: Vec<ChannelWeightSum>,
50    channel_bin_widths: Vec<usize>,
51    visibility_shape: Option<(usize, usize)>,
52}
53
54impl TimeAverageStatistics {
55    fn integration(&self) -> Integration {
56        let last = self.last_usable.unwrap_or(self.last);
57        let (time, interval) = centered_time_and_interval(
58            (self.first.time, self.first.interval),
59            (last.time, last.interval),
60        );
61        Integration { time, interval }
62    }
63
64    fn baseline_weight(&self) -> f64 {
65        let weight = if self.channel_bin_widths.is_empty() {
66            mean_weight_preferring_unflagged(self.weights.iter().map(ChannelWeightSum::sample))
67        } else {
68            let mut channels = self.weights.as_slice();
69            mean_weight_preferring_unflagged(self.channel_bin_widths.iter().map(|&width| {
70                let (bin, rest) = channels.split_at(width);
71                channels = rest;
72                plotms_columns::sum_channel_weights(
73                    bin.iter().map(ChannelWeightSum::sample),
74                    plotms_columns::ChannelWeightMask::Positive,
75                )
76            }))
77        };
78        plotms_columns::baseline_weight(f64::from(weight))
79    }
80}

82impl TimeBaselineStatistics {
83    pub(super) fn observe(&mut self, row: TimeBaselineRow<'_>) {
84        let group = self
85            .groups
86            .entry(row.time_group)
87            .or_insert_with(|| TimeAverageStatistics {
88                partition: row.partition,
89                first: row.integration,
90                last: row.integration,
91                last_usable: None,
92                weights: row.first_correlation.map_or_else(Vec::new, |_| {
93                    row.bins
94                        .iter()
95                        .flat_map(ChannelBin::channels)
96                        .map(|_| ChannelWeightSum::default())
97                        .collect()
98                }),
99                channel_bin_widths: if row.first_correlation.is_some()
100                    && row.bins.iter().any(|bin| bin.averaged)
101                {
102                    row.bins.iter().map(|bin| bin.channels().count()).collect()
103                } else {
104                    Vec::new()
105                },
106                visibility_shape: row.visibility_shape,
107            });
108        if row.integration.time < group.first.time {
109            group.first = row.integration;
110        }
111        if row.integration.time > group.last.time {
112            group.last = row.integration;
113        }
114        if row.has_unflagged_visibility
115            && group
116                .last_usable
117                .is_none_or(|last| row.integration.time > last.time)
118        {
119            group.last_usable = Some(row.integration);
120        }
121        if let Some(correlation) = row.first_correlation {
122            for (channel, sample) in row
123                .bins
124                .iter()
125                .flat_map(ChannelBin::channels)
126                .zip(&mut group.weights)
127            {
128                let weight = row.weights.map_or(1.0, |grid| {
129                    grid.values[correlation * grid.chan_count + channel] as f32
130                });
131                sample.observe(row.flags[(correlation, channel)], weight);
132            }
133        }
134    }

136    pub(super) fn plan(
137        self,
138        time_plan: &PlotTimeGroups,
139        rows: &[usize],
140        averaging: &MsAverageSpec,
141    ) -> Result<TimeBaselinePlan, String> {
142        let interval = averaging.avgtime.unwrap_or(0.0);
143        let mut averages: Vec<BaselineAverage> = Vec::new();
144        let mut rows_to_average = HashMap::with_capacity(self.groups.len());
145        let mut shapes = VisibilityAverageShapes::default();
146        for buffer in time_plan.buffers() {
147            let Some(anchor) = self.groups.get(&buffer.start) else {
148                continue;
149            };
150            let bounds = anchor.integration();
151            let [observation, scan, field, window] = anchor.partition;
152            let partition = [
153                observation,
154                if averaging.avgscan { 0 } else { scan },
155                if averaging.avgfield { 0 } else { field },
156                if averaging.avgspw { 0 } else { window },
157            ];
158            let (start, end) = (
159                bounds.time - bounds.interval / 2.0,
160                bounds.time + bounds.interval / 2.0,
161            );
162            let starts_new_average = averages.last().is_none_or(|average| {
163                let elapsed = bounds.time - average.anchor_time;
164                elapsed < 0.0 || elapsed > interval || average.partition != partition
165            });
166            if starts_new_average {
167                averages.push(BaselineAverage {
168                    partition,
169                    scan,
170                    field,
171                    anchor_time: bounds.time,
172                    row: rows[time_plan.first_row_slot(buffer.start)],
173                    first_buffer_group: buffer.start,
174                    last_buffer_group: buffer.start,
175                    start,
176                    end,
177                });
178            } else if let Some(average) = averages.last_mut() {
179                average.start = average.start.min(start);
180                average.end = average.end.max(end);
181                average.last_buffer_group = buffer.start;
182                if average.scan != scan {
183                    average.scan = -1;
184                }
185            }
186            let output = averages.len() - 1;
187            if !averaging.avgspw
188                && let Some((correlations, channels)) = anchor.visibility_shape
189            {
190                shapes.check_shape(
191                    "vector antenna time averaging",
192                    output,
193                    correlations,
194                    channels,
195                )?;
196            }
197            for group in buffer.clone() {
198                if let Some(statistics) = self.groups.get(&group) {
199                    if averaging.avgspw
200                        && let Some((correlations, channels)) = statistics.visibility_shape
201                    {
202                        shapes.check_shape(
203                            "vector time and spectral window averaging",
204                            output,
205                            correlations,
206                            channels,
207                        )?;
208                    }
209                    rows_to_average.insert(group, (output, statistics.baseline_weight()));
210                }
211            }
212        }
213        Ok(TimeBaselinePlan {
214            averages,
215            rows_to_average,
216        })
217    }
218}

220struct BaselineAverage {
221    partition: [i32; 4],
222    scan: i32,
223    field: i32,
224    anchor_time: f64,
225    row: usize,

230}

237impl TimeBaselinePlan {
238    pub(super) fn combine_uvws(
239        &self,
240        time_uvws: HashMap<(usize, i32, i32), TimeUvwAverage>,
241    ) -> HashMap<(usize, i32, i32), TimeUvwAverage> {
242        let mut uvws = HashMap::<(usize, i32, i32), TimeUvwAverage>::new();
243        for ((group, antenna1, antenna2), uvw) in time_uvws {
244            if let Some(&(output, weight)) = self.rows_to_average.get(&group) {
245                // PlotMSVBAverager weights the already-normalized UVW by the
246                // transformed baseline weight, including flagged windows.
247                uvws.entry((output, antenna1, antenna2)).or_default().add(
248                    uvw.mean(),
249                    false,
250                    weight,
251                );
252            }
253        }
254        uvws
255    }
256
257    pub(super) fn output_group(&self, time_group: usize) -> Option<usize> {
258        self.rows_to_average
259            .get(&time_group)
260            .map(|&(group, _)| group)
261    }
262
263    pub(super) fn scan_and_field(&self, average: usize) -> Option<(i32, i32)> {
264        self.averages
265            .get(average)
266            .map(|average| (average.scan, average.field))
267    }
268
269    pub(super) fn center(&self, average: usize) -> Option<f64> {
270        self.averages
271            .get(average)
272            .map(|average| average.start + (average.end - average.start) / 2.0)
273    }
274
275    pub(super) fn spectral_window_buffers(
276        &self,
277    ) -> impl Iterator<Item = (usize, usize, (i32, i32))> + '_ {
278        self.averages.iter().map(|average| {
279            (
280                average.first_buffer_group,
281                average.last_buffer_group,
282                (average.scan, average.field),
283            )
284        })
285    }

287    /// Each point here is a normalized complex time average for one baseline.
288    /// Only project to the displayed amplitude or phase after this second mean.
289    pub(super) fn combine_series(
290        &self,
291        series: &mut AveragedSeriesAccumulator,
292        retain_baselines: bool,
293    ) {
294        enum Sample {
295            SpectralWindow(TransformedVectorSample),
296            Baseline(Complex64),
297        }
298        let mean = |samples: &AveragedVisibility| {
299            if retain_baselines {
300                samples
301                    .transformed_vector_sample()
302                    .map(Sample::SpectralWindow)
303            } else {
304                samples.vector_mean().map(Sample::Baseline)
305            }
306        };
307        for (mut key, mut point) in std::mem::take(&mut series.points) {
308            if !point.has_usable_weight() {
309                continue;
310            }
311            let Some(&(group, weight)) = key
312                .time_key
313                .group()
314                .and_then(|time_group| self.rows_to_average.get(&time_group))
315            else {
316                continue;
317            };
318            let Some(y) = mean(&point.y_visibility) else {
319                continue;
320            };
321            let x = mean(&point.x_visibility);
322            key.time_key = AveragingTimeKey::Group(group);
323            let output = series.points.entry(key).or_default();
324            if output.representative.is_none() {
325                if let Some(mut representative) = point.representative.take() {
326                    if !retain_baselines {
327                        representative.row = self.averages[group].row;
328                    }
329                    output.representative = Some(representative);
330                }
331                output.display_channel = point.display_channel;
332                output.numeric_axis_samples = point.numeric_axis_samples;
333            }
334            for (output, sample) in [
335                (&mut output.x_visibility, x),
336                (&mut output.y_visibility, Some(y)),
337            ] {
338                match sample {
339                    Some(Sample::SpectralWindow(sample)) => {
340                        output.add_normalized_vector_sample(sample, weight as f32, false);
341                    }
342                    Some(Sample::Baseline(value)) => output.add_weighted_value(value, weight),
343                    None => {}
344                }
345            }
346        }
347    }

349    /// Reduce the complex means from each time-averaged baseline with its
350    /// transformed spectrum. The second antenna receives the conjugate.
351    pub(super) fn combine_antenna_series(&self, series: &mut AveragedSeriesAccumulator) {
352        for (key, point) in std::mem::take(&mut series.points) {
353            let AveragingBaselineToken::Baseline(first, second) = key.baseline else {
354                continue;
355            };
356            let Some(group) = key
357                .time_key
358                .group()
359                .and_then(|group| self.output_group(group))
360            else {
361                continue;
362            };
363            let Some(y) = point.y_visibility.transformed_vector_sample() else {
364                continue;
365            };
366            let x = point.x_visibility.transformed_vector_sample();
367            let weight = y
368                .channel_weight
369                .map(f64::from)
370                .or(point.combined_primary_weight);
371            let weight = plotms_columns::baseline_weight(weight.unwrap_or(1.0)) as f32;
372            for (antenna, conjugate) in [(first, false), (second, true)] {
373                let key = AveragingPointKey {
374                    time_key: AveragingTimeKey::Group(group),
375                    baseline: AveragingBaselineToken::Antenna(antenna),
376                    ..key.clone()
377                };
378                let output = series.points.entry(key).or_default();
379                if output.representative.is_none() {
380                    output.representative = point.representative.clone();
381                    output.display_channel = point.display_channel;
382                    output.numeric_axis_samples = point.numeric_axis_samples.clone();
383                }
384                if let Some(x) = x {
385                    output
386                        .x_visibility
387                        .add_normalized_vector_sample(x, weight, conjugate);
388                }
389                output
390                    .y_visibility
391                    .add_normalized_vector_sample(y, weight, conjugate);
392            }
393        }
394    }
