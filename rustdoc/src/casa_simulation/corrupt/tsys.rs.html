2//! `sm.setnoise(mode='tsys-manual')`: thermal noise whose σ comes from a
3//! system temperature, CASA's `AtmosCorruptor` in its manual mode.
4//!
5//! # What CASA computes
6//!
7//! `Simulator::setnoise` adds two terms for `tsys-manual`: an `ANoise` of
8//! unit amplitude, whose draws are those of `simplenoise` with σ = 1 Jy, and
9//! a `T` Jones term flagged `NoiseScale`, which `VisEquation::collapseForSim`
10//! applies in correct mode to the noise alone, not to the model. Antenna
11//! `i`'s `T` is (`AtmosCorruptor::simPar`, `CalCorruptor.cc`)
12//!
13//! ```text
14//! T_i = D_i / f / sqrt(Tsys(A_i)),   f = sqrt(c · a / sqrt(Δν · t)),
15//! ```
16//!
17//! so a row of antennas `i` and `j` gets noise of σ = s_i s_j per real and
18//! imaginary part, s = 1/T:
19//!
20//! ```text
21//! σ_ij = c · a / sqrt(Δν · t) · sqrt(Tsys(A_i) · Tsys(A_j)) / (D_i · D_j).
22//! ```
23//!
24//! - `D` is the dish diameter (ANTENNA's DISH_DIAMETER), metres.
25//! - `a = 8 k · 10²³ · 10⁻⁴ / (η_a η_q π)` (`AtmosCorruptor::initialize`),
26//!   `k` = 1.38062·10⁻¹⁶ erg/K: `2k/(η_a η_q A)` in Jy per kelvin for a dish
27//!   of area `A = π D²/4`, with `antefficiency` η_a and `correfficiency` η_q.
28//! - `c` is `senscoeff`, or 1/√2 when it is not positive.
29//! - `Δν` is the row's spectral window's |TOTAL_BANDWIDTH| / NUM_CHAN: one
30//!   width for every channel.
31//! - `t` is the integration time, EXPOSURE.
32//! - `A_i = 1/sin(e_i)` is the airmass at antenna `i`'s elevation `e_i` of
33//!   the field's phase centre (`VisIter::azel`), or 1000 below the horizon.
34//! - `Tsys(A) = (hν/k) · R · (1 + g)` (`AtmosCorruptor::tsys`, the branch
35//!   without frequency dependence that `tsys-manual` takes), with
36//!   `τ = τ₀ A` and
37//!   `R = η_s R(T_cmb) + e^τ (η_s (1 − e^−τ) R(T_atm) + (1 − η_s) R(T_gnd)
38//!   + T_rx / (hν/k))`, `R(T) = 1/(e^(hν/kT) − 1)`: `tau` τ₀,
39//!   `spillefficiency` η_s, `tcmb`, `tatmos` (250 K when it is at most
40//!   10), `tground` and `trx`, `rxtype` g (0 when negative). ν is the
41//!   window's REF_FREQUENCY, in single precision as CASA holds it: the
42//!   focus channel of a `T` without frequency dependence is channel 0.
43//!   CASA writes `hν/k` with two constants, 4.80145·10⁻¹¹ K/Hz in `tsys` and
44//!   4.799274551·10⁻¹¹ in the radiances (`setFocusChan`); both are kept.
45//!
46//! Measured against CASA 6.7.6 (`tools/simulation`, #841 Phase 3), per-row
47//! variances agree with this formula to 4·10⁻⁴ over 2·10⁷ draws, for 12 m
48//! and 7 m dishes together, autocorrelations, non-default parameters
49//! (`rxtype=1`, `senscoeff`, `tatmos`, …) and two spectral windows.
50//!
51//! A `T` that is not finite and positive does not scale: CASA's
52//! `JonesScal::invert` marks a zero or NaN `T` bad and `setMatByOk` applies
53//! a bad `T` as 1. That happens when Tsys overflows single precision (the
54//! airmass 1000 below the horizon) or is not a number; that antenna's scale
55//! is then 1, and a `T` of +∞ (Tsys 0) scales by 0.
56//!
57//! The arithmetic follows CASA's precisions (single-precision parameters,
58//! airmass, Tsys and `T`; double-precision intermediates), so σ agrees with
59//! CASA's to about 10⁻⁷ relatively; the draws differ anyway.
60//!
61//! # Where casa-rs departs
62//!
63//! Recorded in `docs/CASA (C++) bugs.md`: CASA caches each antenna's airmass
64//! for 100 s of solution times, evaluates the `T` of rows observed after
65//! `setnoise` at the nearest earlier solution, uses the EXPOSURE of the
66//! MeasurementSet's first row for every row, applies an antenna without an
67//! unflagged row as 1, and multiplies every noise term by every Tsys term's
68//! scale. casa-rs computes each row's σ from its own integration's
69//! elevations and its own EXPOSURE, whatever the flags, and each term adds
70//! its own noise.
71
72/// CASA's Boltzmann constant in erg/K (`AtmosCorruptor::initialize`).
73const BOLTZMANN_ERG_PER_K: f64 = 1.380_62e-16;
74
75/// h/k, K/Hz, as `AtmosCorruptor::tsys` writes it.
76const PLANCK_OVER_BOLTZMANN_TSYS: f64 = 4.801_45e-11;
77
78/// h/k, K/Hz, as `AtmosCorruptor::setFocusChan` writes it for the radiances.
79const PLANCK_OVER_BOLTZMANN_RADIANCE: f64 = 0.047_992_745_51 * 1e-9;
80
81/// The airmass CASA gives an antenna whose elevation is not positive.
82const BELOW_HORIZON_AIRMASS: f32 = 1000.0;
83
84/// An atmosphere temperature at or below this, K, is replaced by
85/// [`DEFAULT_ATMOSPHERE_K`] (`Simulator::setnoise`).
86const UNSET_ATMOSPHERE_K: f32 = 10.0;
87
88/// The atmosphere temperature `setnoise` uses when `tatmos` is unset.
89const DEFAULT_ATMOSPHERE_K: f32 = 250.0;

91/// The arguments of `setnoise(mode='tsys-manual')`, as `Simulator::setnoise`
92/// passes them to the corruptor.
93#[derive(Debug, Clone, Copy, PartialEq)]
94pub struct TsysParameters {
95    /// `tau`: the zenith opacity.
96    pub tau: f64,
97    /// `tatmos`: the atmosphere's temperature, K; 250 K when at most 10.
98    pub atmosphere_k: f64,
99    /// `antefficiency`: the aperture efficiency.
100    pub antenna_efficiency: f64,
101    /// `spillefficiency`: the forward (spillover) efficiency.
102    pub spillover_efficiency: f64,
103    /// `correfficiency`: the correlator efficiency.
104    pub correlator_efficiency: f64,
105    /// `trx`: the receiver temperature, K.
106    pub receiver_k: f64,
107    /// `tground`: the ground's temperature, K.
108    pub ground_k: f64,
109    /// `tcmb`: the cosmic background's temperature, K.
110    pub cmb_k: f64,
111    /// `senscoeff`: the sensitivity constant; 1/√2 when not positive.
112    pub sensitivity: f64,
113    /// `rxtype`: 0 for a sideband-separating receiver, 1 for a double
114    /// sideband one, whose Tsys is doubled; negative is 0.
115    pub receiver_type: i64,
116}

136/// The `tsys-manual` model as CASA's corruptor holds it: single-precision
137/// parameters, with the defaults `setnoise` fills in.
138#[derive(Debug, Clone, Copy, PartialEq)]
139pub(crate) struct TsysManual {
140    /// τ₀, CASA's `tauscale`.
141    tau: f32,
142    atmosphere_k: f32,
143    ground_k: f32,
144    cmb_k: f32,
145    receiver_k: f32,
146    spillover: f32,
147    /// `senscoeff() · amp()`, both single precision, multiplied in single
148    /// precision as `simPar` does: c · a, Jy m² per kelvin.
149    jansky_per_kelvin: f32,
150    /// 1 + g.
151    sidebands: f64,
152}

154/// What Tsys depends on at one frequency: hν/k and the radiances
155/// `1/(e^(hν/kT) − 1)` of the background, the ground and the atmosphere.
156#[derive(Debug, Clone, Copy, PartialEq)]
157struct Band {
158    /// hν/k, K, as `tsys` computes it.
159    h_nu_over_k: f64,
160    cmb: f64,
161    ground: f64,
162    atmosphere: f64,
163}

165/// What a spectral window gives its rows' noise.
166#[derive(Debug, Clone, Copy, PartialEq)]
167pub(crate) struct WindowNoise {
168    /// REF_FREQUENCY, Hz.
169    pub(crate) reference_frequency_hz: f64,
170    /// |TOTAL_BANDWIDTH| / NUM_CHAN, Hz.
171    pub(crate) channel_bandwidth_hz: f64,
172}

174/// What one row gives its noise: its antennas' dishes and airmasses, and
175/// its integration time.
176#[derive(Debug, Clone, Copy, PartialEq)]
177pub(crate) struct RowNoise {
178    /// DISH_DIAMETER of ANTENNA1 and ANTENNA2, metres.
179    pub(crate) diameters_m: [f64; 2],
180    /// Their airmasses ([`airmass`]).
181    pub(crate) airmass: [f32; 2],
182    /// EXPOSURE, seconds.
183    pub(crate) exposure_s: f64,
184}
185
186/// The airmass CASA gives an antenna at elevation `elevation_rad`:
187/// `1/sin(e)` with `e` and its sine in single precision, held in single
188/// precision; 1000 when `e` is not positive.
189pub(crate) fn airmass(elevation_rad: f64) -> f32 {
190    let elevation = elevation_rad as f32;
191    if elevation > 0.0 {
192        (1.0 / f64::from(elevation.sin())) as f32
193    } else {
194        BELOW_HORIZON_AIRMASS
195    }
196}

198impl TsysManual {
199    /// The model of `parameters`, held as `setnoise` and the corruptor hold
200    /// them.
201    pub(crate) fn new(parameters: &TsysParameters) -> Self {
202        let float = |value: f64| value as f32;
203        let atmosphere_k = float(parameters.atmosphere_k);
204        let sensitivity = if parameters.sensitivity > 0.0 {
205            float(parameters.sensitivity)
206        } else {
207            float(std::f64::consts::FRAC_1_SQRT_2)
208        };
209        // `8 k 1e23 1e-4 / (η_a η_q π)`: the efficiencies multiply in
210        // single precision, then everything in double, held in single.
211        let efficiencies =
212            float(parameters.antenna_efficiency) * float(parameters.correlator_efficiency);
213        let amplitude = (8.0 * BOLTZMANN_ERG_PER_K * 1e23 * 1e-4
214            / (f64::from(efficiencies) * std::f64::consts::PI)) as f32;
215        let receiver_type = if parameters.receiver_type >= 0 {
216            parameters.receiver_type as i32
217        } else {
218            0
219        };
220        Self {
221            tau: float(parameters.tau),
222            atmosphere_k: if atmosphere_k > UNSET_ATMOSPHERE_K {
223                atmosphere_k
224            } else {
225                DEFAULT_ATMOSPHERE_K
226            },
227            ground_k: float(parameters.ground_k),
228            cmb_k: float(parameters.cmb_k),
229            receiver_k: float(parameters.receiver_k),
230            spillover: float(parameters.spillover_efficiency),
231            jansky_per_kelvin: sensitivity * amplitude,
232            sidebands: 1.0 + f64::from(receiver_type),
233        }
234    }
235
236    /// The zenith opacity τ₀ the model holds.
237    pub(crate) fn tau(&self) -> f32 {
238        self.tau
239    }
240
241    /// The band at frequency `frequency_hz`.
242    fn band(&self, frequency_hz: f64) -> Band {
243        let h_nu_over_k = PLANCK_OVER_BOLTZMANN_RADIANCE * frequency_hz;
244        let radiance =
245            |temperature_k: f32| 1.0 / ((h_nu_over_k / f64::from(temperature_k)).exp() - 1.0);
246        Band {
247            h_nu_over_k: PLANCK_OVER_BOLTZMANN_TSYS * frequency_hz,
248            cmb: radiance(self.cmb_k),
249            ground: radiance(self.ground_k),
250            atmosphere: radiance(self.atmosphere_k),
251        }
252    }
253
254    /// Tsys at airmass `airmass` in `band`, K, in single precision
255    /// (`AtmosCorruptor::tsys`).
256    fn tsys(&self, band: &Band, airmass: f32) -> f32 {
257        let tau = f64::from(self.tau * airmass);
258        let spillover = f64::from(self.spillover);
259        let radiance = spillover * band.cmb
260            + tau.exp()
261                * (spillover * (1.0 - (-tau).exp()) * band.atmosphere
262                    + (1.0 - spillover) * band.ground
263                    + f64::from(self.receiver_k) / band.h_nu_over_k);
264        (band.h_nu_over_k * radiance * self.sidebands) as f32
265    }
266
267    /// The zenith Tsys `setnoise` checks and logs, K: at airmass 1, at the
268    /// centre of a window of `channels` channels from `reference_hz` over
269    /// `total_bandwidth_hz` (channel ⌊n/2⌋ as `setFocusChan` places it,
270    /// from the single-precision reference and width CASA holds).
271    pub(crate) fn zenith_tsys(
272        &self,
273        reference_hz: f64,
274        total_bandwidth_hz: f64,
275        channels: usize,
276    ) -> f32 {
277        let resolution = f64::from(total_bandwidth_hz as f32) / channels.max(1) as f64;
278        let centre = f64::from(reference_hz as f32) + (channels / 2) as f64 * resolution;
279        self.tsys(&self.band(centre), 1.0)
280    }
281
282    /// The model for the rows of `window`.
283    pub(crate) fn for_window(&self, window: &WindowNoise) -> WindowTsys {
284        WindowTsys {
285            model: *self,
286            band: self.band(f64::from(window.reference_frequency_hz as f32)),
287            channel_bandwidth_hz: window.channel_bandwidth_hz,
288        }
289    }
290}

292/// The `tsys-manual` model for the rows of one spectral window.
293#[derive(Debug, Clone, Copy, PartialEq)]
294pub(crate) struct WindowTsys {
295    model: TsysManual,
296    band: Band,
297    channel_bandwidth_hz: f64,
298}
299
300impl WindowTsys {
301    /// σ of each real and imaginary draw of `row`, Jy: s_i s_j.
302    pub(crate) fn sigma(&self, row: &RowNoise) -> f32 {
303        let factor = (f64::from(self.model.jansky_per_kelvin)
304            / (self.channel_bandwidth_hz * row.exposure_s).sqrt())
305        .sqrt();
306        let scale = |diameter_m: f64, airmass: f32| {
307            let tsys = self.model.tsys(&self.band, airmass);
308            let jones = (diameter_m / factor / f64::from(tsys.sqrt())) as f32;
309            // A zero or NaN `T` is bad, and a bad `T` applies as 1.
310            if jones.is_nan() || jones == 0.0 {
311                1.0
312            } else {
313                1.0 / jones
314            }
315        };
316        scale(row.diameters_m[0], row.airmass[0]) * scale(row.diameters_m[1], row.airmass[1])
317    }
318}

337    /// CASA's formula in double precision throughout, from the module's
338    /// equations.
339    fn reference_sigma(parameters: &TsysParameters, window: &WindowNoise, row: &RowNoise) -> f64 {
340        let nu = window.reference_frequency_hz;
341        let h_nu_k = 4.80145e-11 * nu;
342        let radiance = |t: f64| 1.0 / ((0.04799274551e-9 * nu / t).exp() - 1.0);
343        let spill = parameters.spillover_efficiency;
344        let tsys = |a: f64| {
345            let tau = parameters.tau * a;
346            let r = spill * radiance(parameters.cmb_k)
347                + tau.exp()
348                    * (spill * (1.0 - (-tau).exp()) * radiance(parameters.atmosphere_k)
349                        + (1.0 - spill) * radiance(parameters.ground_k)
350                        + parameters.receiver_k / h_nu_k);
351            h_nu_k * r * (1.0 + parameters.receiver_type.max(0) as f64)
352        };
353        let c = if parameters.sensitivity > 0.0 {
354            parameters.sensitivity
355        } else {
356            std::f64::consts::FRAC_1_SQRT_2
357        };
358        let a = 8.0 * 1.38062e-16 * 1e23 * 1e-4
359            / (parameters.antenna_efficiency
360                * parameters.correlator_efficiency
361                * std::f64::consts::PI);
362        c * a / (window.channel_bandwidth_hz * row.exposure_s).sqrt()
363            * (tsys(f64::from(row.airmass[0])) * tsys(f64::from(row.airmass[1]))).sqrt()
364            / (row.diameters_m[0] * row.diameters_m[1])
365    }

367    #[test]
368    fn sigma_is_casas_formula_to_single_precision() {
369        let defaults = TsysParameters::default();
370        let other = TsysParameters {
371            tau: 0.8,
372            atmosphere_k: 260.0,
373            antenna_efficiency: 0.7,
374            spillover_efficiency: 0.9,
375            correlator_efficiency: 0.95,
376            receiver_k: 80.0,
377            ground_k: 280.0,
378            cmb_k: 2.725,
379            sensitivity: 0.5,
380            receiver_type: 1,
381        };
382        for parameters in [defaults, other] {
383            let model = TsysManual::new(&parameters).for_window(&BAND6);
384            for row in [
385                row([12.0, 12.0], [1.0, 1.0], 10.0),
386                row([12.0, 7.0], [1.3, 2.1], 2.0),
387                row([7.0, 7.0], [5.7, 5.7], 30.0),
388            ] {
389                let sigma = f64::from(model.sigma(&row));
390                let expected = reference_sigma(&parameters, &BAND6, &row);
391                assert!(
392                    (sigma / expected - 1.0).abs() < 2e-6,
393                    "{parameters:?} {row:?}: {sigma} against {expected}"
394                );
395            }
396        }

399    #[test]
400    fn sigma_scales_as_the_formula_says() {
401        let model = TsysManual::new(&TsysParameters::default());
402        let sigma =
403            |window: &WindowNoise, row: &RowNoise| f64::from(model.for_window(window).sigma(row));
404        let base = row([12.0, 12.0], [1.2, 1.2], 10.0);
405        let at = sigma(&BAND6, &base);
406        // σ ∝ 1/(D_i D_j), 1/√t and 1/√Δν.
407        let smaller = sigma(&BAND6, &row([12.0, 6.0], [1.2, 1.2], 10.0));
408        assert!((smaller / at - 2.0).abs() < 1e-6);
409        let longer = sigma(&BAND6, &row([12.0, 12.0], [1.2, 1.2], 40.0));
410        assert!((longer / at - 0.5).abs() < 1e-6);
411        let wider = WindowNoise {
412            channel_bandwidth_hz: 4.0 * BAND6.channel_bandwidth_hz,
413            ..BAND6
414        };
415        assert!((sigma(&wider, &base) / at - 0.5).abs() < 1e-6);
416        // More airmass, more noise; one antenna's airmass enters as √Tsys.
417        let lower = sigma(&BAND6, &row([12.0, 12.0], [2.0, 2.0], 10.0));
418        let half = sigma(&BAND6, &row([12.0, 12.0], [1.2, 2.0], 10.0));
419        assert!(lower > at);
420        assert!((half * half / (at * lower) - 1.0).abs() < 1e-6);
421    }

431    /// Below the horizon Tsys overflows single precision for τ₀ above about
432    /// 0.09, and CASA applies the bad `T` as 1: the antenna does not scale.

434    fn an_overflowing_tsys_applies_as_one() {
435        let model = TsysManual::new(&TsysParameters {
