21/// Admission bound for one reusable index of complete selected row/channel runs.
22#[derive(Clone, Copy, Debug)]
23pub struct SelectedObservationBlockIndexPlan {
24    rows: usize,
25    runs: usize,
26    correlations: usize,
27    capacity_bytes: usize,
28}
29
30impl SelectedObservationBlockIndexPlan {
31    /// Plan a bounded window independently of the physical source-block size.
32    pub fn new(
33        problem: &CompiledProblem,
34        maximum_runs: usize,
35    ) -> Result<Self, BoundObservationSourceError> {
36        let overflow = || BoundObservationSourceError::MeasurementOverflow;
37        let minimum_channels = problem
38            .selected_observation()
39            .read_set()
40            .sources()
41            .iter()
42            .flat_map(|source| source.selection().spectral_windows())
43            .map(|selection| selection.channel_indices().len())
44            .min()
45            .unwrap_or(1)
46            .max(1);
47        let runs = maximum_runs;
48        // A contiguous window may start partway through its first row.
49        let rows = runs.min(
50            runs.div_ceil(minimum_channels)
51                .checked_add(1)
52                .ok_or_else(overflow)?,
53        );
54        let correlations = runs
55            .checked_mul(maximum_selected_correlations(problem))
56            .ok_or_else(overflow)?;
57        let capacity_bytes = index_bytes(rows, runs, correlations).ok_or_else(overflow)?;
58        Ok(Self {
59            rows,
60            runs,
61            correlations,
62            capacity_bytes,
63        })
64    }
65
66    /// Maximum number of complete row/channel runs in one indexed window.
67    #[must_use]
68    pub const fn maximum_runs(self) -> usize {
69        self.runs
70    }
71
72    /// Bound a contiguous window by both run and row admission. A source spectral
73    /// window can contain fewer channels per row than the original selection.
74    pub fn next_range(
75        self,
76        block: &super::SelectedObservationBlock,
77        start: usize,
78    ) -> Result<Range<usize>, BoundObservationSourceError> {
79        let count = block.selected_run_count()?;
80        let channels = block.selected_channels_per_row()?;
81        if start > count || channels == 0 || self.runs == 0 || self.rows == 0 {
82            return Err(BoundObservationSourceError::StoredSampleShapeMismatch);
83        }
84        let row_bound = self.rows.saturating_mul(channels) - start % channels;
85        Ok(start..start.saturating_add(self.runs.min(row_bound)).min(count))
86    }
87
88    /// Resident index bytes; row-domain payloads remain shared with the source block.
89    #[must_use]
90    pub const fn capacity_bytes(self) -> usize {
91        self.capacity_bytes
92    }
93
94    /// Allocate reusable bounded storage after the caller admits its residency.
95    #[must_use]
96    pub fn create_index(self) -> SelectedObservationBlockIndex {
97        SelectedObservationBlockIndex {
98            plan: self,
99            rows: Vec::with_capacity(self.rows),
100            runs: Vec::with_capacity(self.runs),
101            correlations: Vec::with_capacity(self.correlations),
102            binding: None,
103        }
104    }
105}
106
107fn index_bytes(rows: usize, runs: usize, correlations: usize) -> Option<usize> {
108    size_of::<SelectedObservationBlockIndex>()
109        .checked_add(rows.checked_mul(size_of::<SelectedObservationRunRow>())?)?
110        .checked_add(runs.checked_mul(size_of::<IndexedRun>())?)?
111        .checked_add(correlations.checked_mul(size_of::<SelectedObservationRunCorrelation>())?)
112}

114struct IndexedRun {

120/// Reusable source-block index. It holds each row once and compact correlation records,
121/// never materialized selected samples or projected spectral evaluations.
122pub struct SelectedObservationBlockIndex {
123    plan: SelectedObservationBlockIndexPlan,
124    rows: Vec<SelectedObservationRunRow>,
125    runs: Vec<IndexedRun>,
126    correlations: Vec<SelectedObservationRunCorrelation>,
127    pub(super) binding: Option<(CompiledProblemId, (u64, u64, u64))>,
128}
129
130impl SelectedObservationBlockIndex {
131    /// Populated bytes, excluding the source-owned shared row-domain payloads.
132    pub fn current_bytes(&self) -> Result<usize, BoundObservationSourceError> {
133        index_bytes(self.rows.len(), self.runs.len(), self.correlations.len())
134            .ok_or(BoundObservationSourceError::MeasurementOverflow)
135    }
136
137    /// Allocated bytes, excluding the source-owned shared row-domain payloads.
138    pub fn capacity_bytes(&self) -> Result<usize, BoundObservationSourceError> {
139        index_bytes(
140            self.rows.capacity(),
141            self.runs.capacity(),
142            self.correlations.capacity(),
143        )
144        .ok_or(BoundObservationSourceError::MeasurementOverflow)
145    }
146
147    /// Release row-domain references before the source block is refilled, retaining
148    /// the index allocations. This keeps shared payloads within source residency.
149    pub fn clear(&mut self) {
150        self.binding = None;
151        self.rows.clear();
152        self.runs.clear();
153        self.correlations.clear();
154    }
155
156    /// Borrow the inspected window with its unchanged source block. The owner,
157    /// traversal and block ordinal bind the view without scanning its content.
158    pub fn view<'a>(
159        &'a self,
160        block: &'a super::SelectedObservationBlock,
161        problem: &'a CompiledProblem,
162    ) -> Result<SelectedObservationIndexedBlock<'a>, BoundObservationSourceError> {
163        if self.binding
164            != block
165                .index_binding
166                .map(|binding| (problem.problem_id(), binding))
167            || self.binding.is_none()
168        {
169            return Err(BoundObservationSourceError::StoredSampleShapeMismatch);
170        }
171        let (geometry_engine, selection) = block.projection_context()?;
172        Ok(SelectedObservationIndexedBlock {
173            problem,
174            index: self,
175            geometry_engine,
176            selection,
177        })
178    }
179
180    pub(super) fn push(
181        &mut self,
182        row: &SelectedObservationRunRow,
183        channel: SelectedObservationRunChannel,
184        correlations: &[SelectedObservationRunCorrelation],
185    ) -> Result<(), BoundObservationSourceError> {
186        let new_row = self
187            .rows
188            .last()
189            .is_none_or(|last| last.physical_row != row.physical_row);
190        if self.runs.len() == self.plan.runs
191            || (new_row && self.rows.len() == self.plan.rows)
192            || correlations.len()
193                > self
194                    .plan
195                    .correlations
196                    .saturating_sub(self.correlations.len())
197        {
198            return Err(BoundObservationSourceError::StoredSampleShapeMismatch);
199        }
200        if new_row {
201            self.rows.push(row.clone());
202        }
203        let start = self.correlations.len();
204        self.correlations.extend_from_slice(correlations);
205        self.runs.push(IndexedRun {
206            row: self.rows.len() - 1,
207            channel,
208            correlations: start..self.correlations.len(),
209        });
210        Ok(())
211    }
212}

214/// Immutable, inspected view borrowing both the index and its source-owned geometry engine.
215/// Its lifetime prevents source refill or index reuse until all projections finish.
216pub struct SelectedObservationIndexedBlock<'a> {
217    pub(super) problem: &'a CompiledProblem,
218    pub(super) index: &'a SelectedObservationBlockIndex,
219    pub(super) geometry_engine: &'a MsCalEngine,
220    pub(super) selection: SelectedRowSpectralSelection,
221}
222
223impl SelectedObservationIndexedBlock<'_> {
224    /// Number of complete runs available for disjoint range partitioning.
225    #[must_use]
226    pub fn run_count(&self) -> usize {
227        self.index.runs.len()
228    }
229
230    /// Project one contiguous range using caller-owned scratch. No source inspection,
231    /// generation proof, or source read is repeated here. Callback references are transient.
232    pub fn visit_range<E: Error + 'static>(
233        &self,
234        projector: &mut SelectedObservationProjector,
235        range: Range<usize>,
236        mut consume: impl FnMut(SelectedObservationTraversalRun<'_>) -> Result<(), E>,
237    ) -> Result<(), SelectedObservationTraversalError<E>> {
238        let runs = self
239            .index
240            .runs
241            .get(range)
242            .ok_or(SelectedObservationTraversalError::Source(
243                BoundObservationSourceError::StoredSampleShapeMismatch,
244            ))?;
245        for run in runs {
246            let row = &self.index.rows[run.row];
247            let correlations = &self.index.correlations[run.correlations.clone()];
248            projector.evaluations.clear();
249            if correlations.len() > projector.evaluations.capacity() {
250                return Err(SelectedObservationTraversalError::Source(
251                    BoundObservationSourceError::StoredSampleShapeMismatch,
252                ));
253            }
254            for correlation in correlations {
255                let sample =
256                    SelectedObservationSampleView::from_run(row, &run.channel, correlation);
257                projector.evaluations.push(
258                    projector
259                        .spectral_evaluator
260                        .project(self.problem, sample, self.geometry_engine, self.selection)
261                        .map_err(SelectedObservationTraversalError::Source)?
262                        .spectral_evaluation(),
263                );
264            }
265            consume(SelectedObservationTraversalRun::new(
266                row,
267                run.channel,
268                correlations,
269                &projector.evaluations,
270            ))
271            .map_err(SelectedObservationTraversalError::Consumer)?;
272        }
273        Ok(())
274    }
275}

277/// Worker-local spectral caches and bounded evaluation scratch; scheduling stays with the caller.
278pub struct SelectedObservationProjector {
279    spectral_evaluator: SpectralEvaluationProjector,
280    correlations: Vec<SelectedObservationRunCorrelation>,
281    evaluations: Vec<SelectedSpectralEvaluation>,
282}
283
284impl SelectedObservationProjector {
285    /// Allocate scratch for the compiled problem's maximum correlation group.
286    #[must_use]
287    pub fn new(problem: &CompiledProblem) -> Self {
288        Self {
289            spectral_evaluator: SpectralEvaluationProjector::new(),
290            correlations: Vec::with_capacity(maximum_selected_correlations(problem)),
291            evaluations: Vec::with_capacity(maximum_selected_correlations(problem)),
292        }
293    }
294
295    /// Project a borrowed source range directly, using only worker-local bounded
296    /// correlation and spectral scratch. No index, selected sample collection,
297    /// source read, or generation inspection is created by this traversal.
298    /// The caller retains the block until every disjoint worker range completes
299    /// and inspects its runs once in canonical order through the source consumer.
300    pub fn visit_block_range<E: Error + 'static>(
301        &mut self,
302        problem: &CompiledProblem,
303        block: &super::SelectedObservationBlock,
304        range: Range<usize>,
305        mut consume: impl FnMut(SelectedObservationTraversalRun<'_>) -> Result<(), E>,
306    ) -> Result<(), SelectedObservationTraversalError<E>> {
307        if block.index_binding.is_none() {
308            return Err(SelectedObservationTraversalError::Source(
309                BoundObservationSourceError::StoredSampleShapeMismatch,
310            ));
311        }
312        let spectral_evaluator = &mut self.spectral_evaluator;
313        let evaluations = &mut self.evaluations;
314        block
315            .visit_selected_sample_range(
316                problem,
317                &mut self.correlations,
318                range,
319                |row, channel, correlations, geometry_engine, selection| {
320                    evaluations.clear();
321                    if correlations.len() > evaluations.capacity() {
322                        return Err(SelectedObservationTraversalError::Source(
323                            BoundObservationSourceError::StoredSampleShapeMismatch,
324                        ));
325                    }
326                    for correlation in correlations {
327                        let sample =
328                            SelectedObservationSampleView::from_run(row, &channel, correlation);
329                        evaluations.push(
330                            spectral_evaluator
331                                .project(problem, sample, geometry_engine, selection)
332                                .map_err(SelectedObservationTraversalError::Source)?
333                                .spectral_evaluation(),
334                        );
335                    }
336                    consume(SelectedObservationTraversalRun::new(
337                        row,
338                        channel,
339                        correlations,
340                        evaluations,
341                    ))
342                    .map_err(SelectedObservationTraversalError::Consumer)
343                },
344            )
345            .map_err(|error| match error {
346                BlockVisitError::Source(error) => SelectedObservationTraversalError::Source(error),
347                BlockVisitError::Consumer(error) => error,
348            })
349    }
350
351    /// Correlation/evaluation scratch and inline projector bytes required before allocation.
352    pub fn required_bytes(problem: &CompiledProblem) -> Option<usize> {
353        maximum_selected_correlations(problem)
354            .checked_mul(
355                size_of::<SelectedObservationRunCorrelation>()
356                    .checked_add(size_of::<SelectedSpectralEvaluation>())?,
357            )?
358            .checked_add(size_of::<Self>())
359    }
360
361    /// Correlation/evaluation scratch and inline projector bytes currently allocated.
362    pub fn capacity_bytes(&self) -> Option<usize> {
363        self.evaluations
364            .capacity()
365            .checked_mul(size_of::<SelectedSpectralEvaluation>())?
366            .checked_add(
367                self.correlations
368                    .capacity()
369                    .checked_mul(size_of::<SelectedObservationRunCorrelation>())?,
370            )?
371            .checked_add(size_of::<Self>())
372    }
