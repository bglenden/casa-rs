26/// The columns `readTable` attaches unconditionally; a table without one is
27/// not a component list.
28const REQUIRED_COLUMNS: [&str; 10] = [
29    FLUX,
30    FLUX_UNIT,
31    FLUX_POLARISATION,
32    SHAPE,
33    REFERENCE_DIRECTION,
34    SHAPE_PARAMETERS,
35    SPECTRUM_SHAPE,
36    REFERENCE_FREQUENCY,
37    SPECTRAL_PARAMETERS,
38    LABEL,
39];
40
41/// Read the component list at `path`.
42pub(crate) fn read_component_list(path: &Path) -> Result<ComponentList, ComponentListError> {
43    let table = open(path)?;
44    let reader = ListReader::attach(&table, path)?;
45    (0..table.row_count())
46        .map(|row| reader.component(row))
47        .collect()
48}
49
50/// Open the table read-only, holding casacore's read lock while it is read
51/// (CASA opens it with the default `AutoLocking`).
52fn open(path: &Path) -> Result<Table, ComponentListError> {
53    let table_error = |source| ComponentListError::Table {
54        path: path.to_path_buf(),
55        source,
56    };
57    #[cfg(unix)]
58    {
59        use casa_tables::{LockMode, LockOptions};
60        Table::open_with_lock(
61            TableOptions::new(path),
62            LockOptions::new(LockMode::AutoLocking),
63        )
64        .map_err(table_error)
65    }
66    #[cfg(not(unix))]
67    {
68        Table::open(TableOptions::new(path)).map_err(table_error)
69    }
70}

72/// The optional columns: old component lists lack the error columns and
73/// `Spectral_Record`, and only lists with a limb-darkened disk have
74/// `Optional_Parameters`.
75struct OptionalColumns {
76    flux_error: bool,
77    direction_error: bool,
78    shape_error: bool,
79    frequency_error: bool,
80    spectral_error: bool,
81    spectral_record: bool,
82    optional_parameters: bool,
83}
84
85struct ListReader<'a> {
86    table: &'a Table,
87    path: &'a Path,
88    direction: MeasureColumn,
89    frequency: MeasureColumn,
90    optional: OptionalColumns,
91}
92
93impl<'a> ListReader<'a> {
94    fn attach(table: &'a Table, path: &'a Path) -> Result<Self, ComponentListError> {
95        let has = |column: &str| {
96            table
97                .schema()
98                .is_some_and(|schema| schema.contains_column(column))
99        };
100        if let Some(column) = REQUIRED_COLUMNS.into_iter().find(|column| !has(column)) {
101            return Err(ComponentListError::MissingColumn {
102                path: path.to_path_buf(),
103                column,
104            });
105        }
106        Ok(Self {
107            table,
108            path,
109            direction: MeasureColumn::attach(
110                table,
111                path,
112                REFERENCE_DIRECTION,
113                MeasureType::Direction,
114            )?,
115            frequency: MeasureColumn::attach(
116                table,
117                path,
118                REFERENCE_FREQUENCY,
119                MeasureType::Frequency,
120            )?,
121            optional: OptionalColumns {
122                flux_error: has(FLUX_ERROR),
123                direction_error: has(DIRECTION_ERROR),
124                shape_error: has(SHAPE_ERROR),
125                frequency_error: has(FREQUENCY_ERROR),
126                spectral_error: has(SPECTRAL_ERROR),
127                spectral_record: has(SPECTRAL_RECORD),
128                optional_parameters: has(OPTIONAL_PARAMETERS),
129            },
130        })
131    }
132
133    /// The component in `row`, built in `readTable`'s order.
134    fn component(&self, row: usize) -> Result<SkyComponent, ComponentListError> {
135        let flux = self.flux(row)?;
136        let shape = self.shape(row)?;
137        let spectrum = self.spectrum(row)?;
138        let mut component =
139            SkyComponent::new(flux, shape, spectrum).with_label(self.string(row, LABEL)?);
140        if self.optional.optional_parameters && self.defined(row, OPTIONAL_PARAMETERS)? {
141            component.set_optional_parameters(self.reals(row, OPTIONAL_PARAMETERS)?);
142        }
143        Ok(component)
144    }
145
146    fn flux(&self, row: usize) -> Result<Flux, ComponentListError> {
147        let values = self.complex4(row, FLUX)?;
148        let unit = self.string(row, FLUX_UNIT)?;
149        let polarisation = Polarisation::parse_casa(&self.string(row, FLUX_POLARISATION)?)
150            .map_err(|source| self.refused(row, source))?;
151        let mut flux =
152            Flux::new(values, &unit, polarisation).map_err(|source| self.refused(row, source))?;
153        if self.optional.flux_error {
154            flux.set_errors(self.complex4(row, FLUX_ERROR)?);
155        }
156        Ok(flux)
157    }
158
159    fn shape(&self, row: usize) -> Result<ComponentShape, ComponentListError> {
160        let shape_type = ShapeType::parse_casa(&self.string(row, SHAPE)?)
161            .map_err(|source| self.refused(row, source))?;
162        let direction = self.direction.direction(self.table, self.path, row)?;
163        let parameters = self.reals(row, SHAPE_PARAMETERS)?;
164        let errors = if self.optional.shape_error {
165            Some(self.reals(row, SHAPE_ERROR)?)
166        } else {
167            None
168        };
169        let mut shape =
170            ComponentShape::from_parameters(shape_type, direction, &parameters, errors.as_deref())
171                .map_err(|source| self.refused(row, source))?;
172        if self.optional.direction_error {
173            shape.set_direction_error(self.direction_error(row)?);
174        }
175        Ok(shape)
176    }
177
178    /// `Direction_Error` holds `[latitude, longitude]`, each in its own unit.
179    fn direction_error(&self, row: usize) -> Result<DirectionError, ComponentListError> {
180        let values = self.reals(row, DIRECTION_ERROR)?;
181        let [latitude, longitude] = <[f64; 2]>::try_from(values.as_slice()).map_err(|_| {
182            cell_error(
183                self.path,
184                row,
185                DIRECTION_ERROR,
186                format!("expected 2 values, got {}", values.len()),
187            )
188        })?;
189        let units = quantum_units(self.table, self.path, row, DIRECTION_ERROR, 2)?;
190        DirectionError::new(
191            quantity(self.path, row, DIRECTION_ERROR, latitude, &units[0])?,
192            quantity(self.path, row, DIRECTION_ERROR, longitude, &units[1])?,
193        )
194        .map_err(|source| self.refused(row, source))
195    }
196
197    /// The spectrum: the record first, then the reference frequency, its
198    /// error, the parameters and their errors from their columns.
199    fn spectrum(&self, row: usize) -> Result<SpectralModel, ComponentListError> {
200        let spectral_type = SpectralType::parse_casa(&self.string(row, SPECTRUM_SHAPE)?)
201            .map_err(|source| self.refused(row, source))?;
202        let mut kind = SpectralKind::casa_default(spectral_type);
203        if self.optional.spectral_record && self.defined(row, SPECTRAL_RECORD)? {
204            let record = self
205                .table
206                .record_cell(row, SPECTRAL_RECORD)
207                .map_err(|source| ComponentListError::Table {
208                    path: self.path.to_path_buf(),
209                    source,
210                })?;
211            apply_spectral_record(&mut kind, &record)
212                .map_err(|source| self.refused(row, source))?;
213        }
214        let frequency = self.frequency.frequency(self.table, self.path, row)?;
215        let mut spectrum = SpectralModel::new(frequency, kind);
216        if self.optional.frequency_error {
217            let value = self.real(row, FREQUENCY_ERROR)?;
218            let unit = quantum_units(self.table, self.path, row, FREQUENCY_ERROR, 1)?;
219            spectrum
220                .set_reference_frequency_error(quantity(
221                    self.path,
222                    row,
223                    FREQUENCY_ERROR,
224                    value,
225                    &unit[0],
226                )?)
227                .map_err(|source| self.refused(row, source))?;
228        }
229        let parameters = self.reals(row, SPECTRAL_PARAMETERS)?;
230        spectrum
231            .set_parameters(&parameters)
232            .map_err(|source| self.refused(row, source))?;
233        if self.optional.spectral_error {
234            let errors = self.reals(row, SPECTRAL_ERROR)?;
235            // `readTable` reads the errors into the parameters' vector, which
236            // `ArrayColumn::get` resizes only when it is empty.
237            if !parameters.is_empty() && errors.len() != parameters.len() {
238                return Err(cell_error(
239                    self.path,
240                    row,
241                    SPECTRAL_ERROR,
242                    format!(
243                        "{} errors for {} spectral parameters",
244                        errors.len(),
245                        parameters.len()
246                    ),
247                ));
248            }
249            spectrum
250                .set_errors(&errors)
251                .map_err(|source| self.refused(row, source))?;
252        }
253        Ok(spectrum)
254    }
255
256    fn refused(&self, row: usize, source: ComponentError) -> ComponentListError {
257        ComponentListError::Component {
258            path: self.path.to_path_buf(),
259            row,
260            source,
261        }
262    }
263
264    fn defined(&self, row: usize, column: &str) -> Result<bool, ComponentListError> {
265        self.table
266            .cell_accessor(row, column)
267            .and_then(|cell| cell.is_defined())
268            .map_err(|source| ComponentListError::Table {
269                path: self.path.to_path_buf(),
270                source,
271            })
272    }
273
274    fn string(&self, row: usize, column: &str) -> Result<String, ComponentListError> {
275        match cell(self.table, self.path, row, column)? {
276            Value::Scalar(ScalarValue::String(text)) => Ok(text.clone()),
277            _ => Err(cell_error(
278                self.path,
279                row,
280                column,
281                "expected a string".to_string(),
282            )),
283        }
284    }
285
286    fn real(&self, row: usize, column: &str) -> Result<f64, ComponentListError> {
287        match cell(self.table, self.path, row, column)? {
288            Value::Scalar(ScalarValue::Float64(value)) => Ok(*value),
289            Value::Scalar(ScalarValue::Float32(value)) => Ok(f64::from(*value)),
290            _ => Err(cell_error(
291                self.path,
292                row,
293                column,
294                "expected a real number".to_string(),
295            )),
296        }
297    }
298
299    fn reals(&self, row: usize, column: &str) -> Result<Vec<f64>, ComponentListError> {
300        match cell(self.table, self.path, row, column)? {
301            Value::Array(array) => array_f64(array),
302            _ => None,
303        }
304        .ok_or_else(|| cell_error(self.path, row, column, "expected a real array".to_string()))
305    }
306
307    fn complex4(&self, row: usize, column: &str) -> Result<[Complex64; 4], ComponentListError> {
308        let values: Vec<Complex64> = match cell(self.table, self.path, row, column)? {
309            Value::Array(ArrayValue::Complex64(values)) => values.iter().copied().collect(),
310            Value::Array(ArrayValue::Complex32(values)) => values
311                .iter()
312                .map(|value| Complex64::new(f64::from(value.re), f64::from(value.im)))
313                .collect(),
314            _ => {
315                return Err(cell_error(
316                    self.path,
317                    row,
318                    column,
319                    "expected a complex array".to_string(),
320                ));
321            }
322        };
323        <[Complex64; 4]>::try_from(values.as_slice()).map_err(|_| {
324            cell_error(
325                self.path,
326                row,
327                column,
328                format!("expected 4 values, got {}", values.len()),
329            )
330        })
331    }
