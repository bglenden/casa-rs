38/// Why a native run did not complete.
39#[derive(Debug, thiserror::Error)]
40pub(crate) enum NativeError {
41    /// A phase's memory did not fit the resource policy.
42    #[error(transparent)]
43    Admission(#[from] Admission),
44    /// The run was cancelled.
45    #[error(transparent)]
46    Cancelled(#[from] Cancelled),
47    /// Any other failure.
48    #[error("{0}")]
49    Other(ApplicationError),
50}

52impl From<NativeError> for ApplicationDispatchError {
53    fn from(error: NativeError) -> Self {
54        match error {
55            NativeError::Admission(admission) => Self::Admission(admission),
56            NativeError::Cancelled(_) => Self::Cancelled,
57            NativeError::Other(error) => Self::Native(error),
58        }
59    }
60}
61
62impl From<ApplicationError> for NativeError {
63    fn from(error: ApplicationError) -> Self {
64        Self::Other(error)
65    }
66}
67
68impl From<ImagingError> for NativeError {
69    fn from(error: ImagingError) -> Self {
70        match error {
71            ImagingError::Admission(admission)
72            | ImagingError::Minor(casa_imaging_runtime::MinorCycleRunError::Admission(admission)) => {
73                Self::Admission(admission)
74            }
75            ImagingError::Cancelled(cancelled) => Self::Cancelled(cancelled),
76            ImagingError::Pass(error) => error.into(),
77            error => Self::Other(Box::new(error)),
78        }
79    }
80}

88    }
89}

112/// A native run ready to start: its admitted source, its worker teams and
113/// what its passes, cycles and products need.
114pub(crate) struct NativeRun {
115    runtime: ApplicationRuntime,
116    publication: ApplicationPublication,
117    aw_catalog: Option<AwCatalogDeployment>,
118    visibility_write: Option<imaging::VisibilityWriteTarget>,
119    access: ResolvedSelectedObservationAccess,
120    /// The source's charge, held while it is open.
121    source: Reservation,
122    teams: WorkerTeams,
123    masks: ImageDomainReconstructionMaskPlans,
124    image_response: Option<MinorCycleImageResponse>,
125}

127/// The end of a run's preparation: check the products against `problem`,
128/// resolve any visibility write, admit the source and form the worker teams
129/// of the policy's core allocation ([`ResourcePolicy::cores`]).
130///
131/// [`ResourcePolicy::cores`]: casa_imaging_runtime::ResourcePolicy::cores
132pub(crate) fn prepare(
133    problem: &CompiledProblem,
134    input: NativeInput,
135    native: ApplicationNative,
136) -> Result<NativeRun, NativeError> {
137    let ApplicationNative {
138        runtime,
139        publication,
140        aw_catalog,
141    } = native;
142    publication.controls.validate_for_problem(problem)?;
143    let visibility_write = (input.write_model_column || input.write_corrected_data)
144        .then(|| {
145            Ok::<_, ApplicationError>(imaging::VisibilityWriteTarget {
146                path: PathBuf::from(input.observation.locator()),
147                selection: visibility_write_selection(problem, input.observation.selection())?,
148                model_data: input.write_model_column,
149                corrected_data: input.write_corrected_data,
150            })
151        })
152        .transpose()?;
153    let (access, source) = finalize_source_access(
154        problem,
155        input.initial_access,
156        imaging::row_samples(problem)?,
157        &runtime.host,
158        &runtime.resource_policy,
159    )?;
160    let cores = runtime.resource_policy.cores(&runtime.host);
161    tracing::info!(
162        "workers: {} computing, {} streaming beside the source's producer",
163        cores.workers,
164        cores.streaming_workers
165    );
166    let teams = WorkerTeams::new(cores)?;
167    Ok(NativeRun {
168        runtime,
169        publication,
170        aw_catalog,
171        visibility_write,
172        access,
173        source,
174        teams,
175        masks: input.masks,
176        image_response: input.minor_cycle_image_response,
177    })
178}

180/// Run the major-cycle passes, the minor cycles between them and any
181/// visibility write, then publish the products. `summary` collects the
182/// phases and totals as they end; a completed run's outcome takes it once
183/// the caller has finished it.
184pub(crate) fn run(
185    problem: &CompiledProblem,
186    run: NativeRun,
187    summary: &mut RunSummary,
188) -> Result<impl FnOnce(RunSummary) -> ImagingOutcome + use<>, NativeError> {
189    summary.workers = run.teams.compute().workers();
190    let NativeRun {
191        runtime,
192        publication,
193        aw_catalog,
194        visibility_write,
195        access,
196        source,
197        teams,
198        masks,
199        image_response,
200    } = run;
201    let outcome = imaging::run(
202        imaging::ImagingInputs {
203            problem,
204            access,
205            masks,
206            image_response,
207            visibility_write,
208            host: runtime.host,
209            policy: runtime.resource_policy,
210            teams: &teams,
211            cancel: runtime.cancel.clone(),
212            spill_directory: &runtime.spill_directory,
213            aw_catalog,
214            backend: runtime.backend,
215            grid_precision: runtime.grid_precision,
216            diagnostics: runtime.diagnostics,
217        },
218        summary,
219    )?;
220    drop(source);
221    let mut published = Vec::new();
222    let products = run_phase(
223        "products",
224        &runtime.cancel,
225        runtime.diagnostics,
226        summary,
227        || {
228            publish_products(
229                problem,
230                ProductSource {
231                    scientific: outcome.scientific,
232                    masks: outcome.masks,
233                    facet_sum_weights: outcome.facet_sum_weights,
234                },
235                &runtime,
236                publication,
237                teams.compute(),
238                &mut published,
239            )
240        },
241    );
242    summary.products = published;
243    let products = products?;
244    let workers = teams.compute().workers();
245    let problem = problem.clone();
246    Ok(move |summary| ImagingOutcome {
247        problem,
248        minor_cycles: outcome.minor_cycles,
249        stop: outcome.stop,
250        major_cycle_count: outcome.major_cycle_count,
251        total_minor_iterations: outcome.total_minor_iterations,
252        total_actual_minor_iterations: outcome.total_actual_minor_iterations,
253        workers,
254        planes_per_wave: outcome.planes_per_wave,
255        visibility_products: outcome.visibility_products,
256        summary,
257        scientific: products.scientific,
258        planned_products: products.planned,
259        products: products.published,
260        _retained: outcome.retained,
261    })
262}

264/// The products of a run and the state they were generated from.
265struct Products {
266    planned: PlannedContinuumGeneration,
267    scientific: MajorCycleCompletion,
268    published: PublishedContinuumGeneration,
269}

271/// What the products are formed from: the final completion, its
272/// reconstruction masks and the faceted domains' facets' own sums of
273/// weights.
274struct ProductSource {
275    scientific: MajorCycleCompletion,
276    masks: Option<ReconstructionMaskSet>,
277    facet_sum_weights: FacetSumWeights,
278}

280/// Generate every planned product member into the writer's private staging,
281/// one output channel per window on the run's team, and publish them,
282/// adding each image published to `published`; a cancelled run publishes
283/// nothing.
284fn publish_products(
285    problem: &CompiledProblem,
286    source: ProductSource,
287    runtime: &ApplicationRuntime,
288    publication: ApplicationPublication,
289    team: &WorkerTeam,
290    published: &mut Vec<String>,
291) -> Result<Products, NativeError> {
292    let ProductSource {
293        scientific,
294        masks: reconstruction_masks,
295        facet_sum_weights,
296    } = source;
297    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific);
298    if let Some(masks) = reconstruction_masks.as_ref() {
299        inputs = match masks {
300            ReconstructionMaskSet::Shared(mask) => inputs.with_reconstruction_mask(mask),
301            ReconstructionMaskSet::Domains(masks) => inputs.with_domain_reconstruction_masks(masks),
302        };
303    }
304    if let Some(centres) = publication.primary_beam_centres.as_ref() {
305        inputs = inputs.with_primary_beam_centres(centres);
306    }
307    if !facet_sum_weights.is_empty() {
308        inputs = inputs.with_facet_sum_weights(&facet_sum_weights);
309    }
310    let planned = PlannedContinuumGeneration::new(&inputs, &publication.controls)?;
311    let window = ProductStoragePlan::new(1, team.workers())?;
312    let demand = planned.demand(&inputs, window, &publication.writer)?;
313    let held = admit(
314        &runtime.host,
315        &runtime.resource_policy,
316        &Demand {
317            phase: "products",
318            memory: demand.peak_residency_bytes(),
319        },
320    )?;
321    let generation =
322        produce_continuum_members(&planned, &inputs, window, team, &publication.writer)?;
323    if runtime.cancel.is_cancelled() {
324        // The staged members are removed when the writer drops.
325        return Err(Cancelled.into());
326    }
327    publication
328        .writer
329        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
330    // The charge covers the staged records until they are published.
331    drop(held);
332    Ok(Products {
333        planned,
334        scientific,
335        published: generation,
336    })
