26/// A scalar or fixed-shape cell value.
27pub(crate) fn scalar_i32(value: i32) -> Value {
28    Value::Scalar(ScalarValue::Int32(value))
29}
30
31pub(crate) fn scalar_f64(value: f64) -> Value {
32    Value::Scalar(ScalarValue::Float64(value))
33}
34
35fn scalar_bool(value: bool) -> Value {
36    Value::Scalar(ScalarValue::Bool(value))
37}
38
39fn scalar_string(value: &str) -> Value {
40    Value::Scalar(ScalarValue::String(value.to_string()))
41}

43fn f64_array(shape: &[usize], values: Vec<f64>) -> Value {
44    Value::Array(ArrayValue::Float64(
45        ArrayD::from_shape_vec(shape.to_vec(), values).expect("shape matches values"),
46    ))
47}
48
49fn i32_array(shape: &[usize], values: Vec<i32>) -> Value {
50    Value::Array(ArrayValue::Int32(
51        ArrayD::from_shape_vec(shape.to_vec(), values).expect("shape matches values"),
52    ))
53}
54
55fn string_array(values: &[&str]) -> Value {
56    Value::Array(ArrayValue::String(
57        ArrayD::from_shape_vec(
58            vec![values.len()],
59            values.iter().map(|value| (*value).to_string()).collect(),
60        )
61        .expect("shape matches values"),
62    ))
63}

71/// The default of a column CASA leaves unwritten: zero, false or empty for
72/// scalars and fixed shapes; `None` (an undefined cell) for variable
73/// shapes.
74fn default_value(definition: &ColumnDef) -> Option<Value> {
75    let zero = |data_type: PrimitiveType| match data_type {
76        PrimitiveType::Bool => ScalarValue::Bool(false),
77        PrimitiveType::Int32 => ScalarValue::Int32(0),
78        PrimitiveType::Float32 => ScalarValue::Float32(0.0),
79        PrimitiveType::String => ScalarValue::String(String::new()),
80        PrimitiveType::Complex32 => ScalarValue::Complex32(Complex32::new(0.0, 0.0)),
81        _ => ScalarValue::Float64(0.0),
82    };
83    match definition.column_kind {
84        ColumnKind::Scalar => Some(Value::Scalar(zero(definition.data_type))),
85        ColumnKind::FixedArray { shape } => {
86            let count = shape.iter().product();
87            Some(match definition.data_type {
88                PrimitiveType::Int32 => i32_array(shape, vec![0; count]),
89                PrimitiveType::Bool => Value::Array(ArrayValue::Bool(
90                    ArrayD::from_shape_vec(shape.to_vec(), vec![false; count])
91                        .expect("shape matches values"),
92                )),
93                _ => f64_array(shape, vec![0.0; count]),
94            })
95        }
96        ColumnKind::VariableArray { .. } => None,
97    }
98}

100/// A row of a subtable: `values` for the columns given, and CASA's defaults
101/// for `table`'s other required columns. Optional columns appear only when
102/// given.
103fn row(table: SubtableId, values: Vec<(&str, Value)>) -> RecordValue {
104    let given = |name: &str| values.iter().find(|(given, _)| *given == name);
105    let mut fields = Vec::new();
106    for definition in schema::required_columns(table) {
107        let value = match given(definition.name) {
108            Some((_, value)) => Some(value.clone()),
109            None => default_value(definition),
110        };
111        if let Some(value) = value {
112            fields.push(RecordField::new(definition.name, value));
113        }
114    }
115    for definition in schema::optional_columns(table) {
116        if let Some((_, value)) = given(definition.name) {
117            fields.push(RecordField::new(definition.name, value.clone()));
118        }
119    }
120    RecordValue::new(fields)
121}

123fn add_rows(
124    ms: &mut MeasurementSet,
125    table: SubtableId,
126    rows: Vec<RecordValue>,
127) -> SimulatorResult<()> {
128    let subtable = ms
129        .subtable_mut(table)
130        .ok_or_else(|| SimulatorError::raised(format!("{} subtable missing", table.name())))?;
131    for record in rows {
132        subtable
133            .add_row(record)
134            .map_err(|error| SimulatorError::raised(format!("{} row: {error}", table.name())))?;
135    }
136    Ok(())
137}

139/// Add the optional columns of `table` that CASA's simulator creates. As
140/// casacore's `addColumnToDesc` with its default dimensionality makes them,
141/// an array column has no fixed shape or dimensionality, so a cell CASA
142/// does not write (SOURCE.POSITION) stays undefined. The units and measure
143/// keywords come from the schema at save.
144fn add_optional_columns(
145    ms: &mut MeasurementSet,
146    table: SubtableId,
147    names: &[&str],
148) -> SimulatorResult<()> {
149    let subtable = ms
150        .subtable_mut(table)
151        .ok_or_else(|| SimulatorError::raised(format!("{} subtable missing", table.name())))?;
152    for name in names {
153        let definition = schema::optional_columns(table)
154            .iter()
155            .find(|definition| definition.name == *name)
156            .ok_or_else(|| {
157                SimulatorError::raised(format!("{} has no column {name}", table.name()))
158            })?;
159        let (column, default) = match definition.column_kind {
160            ColumnKind::Scalar => (column_schema(definition), default_value(definition)),
161            ColumnKind::FixedArray { .. } | ColumnKind::VariableArray { .. } => (
162                ColumnSchema::array_variable(definition.name, definition.data_type, None),
163                None,
164            ),
165        };
166        subtable
167            .add_column(column, default)
168            .map_err(|error| SimulatorError::raised(format!("{}.{name}: {error}", table.name())))?;
169    }
170    Ok(())
171}

173/// Fill every subtable but POINTING from the session.
174pub(crate) fn write_subtables(ms: &mut MeasurementSet, session: &Simulator) -> SimulatorResult<()> {
175    write_antennas(ms, session)?;
176    write_feeds(ms, session)?;
177    write_spectral_windows(ms, session)?;
178    write_fields_and_sources(ms, session)?;
179    write_observations(ms, session)?;
180    write_states(ms, session)
181}

183fn write_antennas(ms: &mut MeasurementSet, session: &Simulator) -> SimulatorResult<()> {
184    let mut table = ms
185        .antenna_mut()
186        .map_err(|error| SimulatorError::raised(format!("ANTENNA: {error}")))?;
187    for antenna in session.antennas() {
188        table
189            .add_antenna(
190                &antenna.name,
191                &antenna.station,
192                "GROUND-BASED",
193                &antenna.mount,
194                antenna.position_itrf_m,
195                [0.0; 3],
196                antenna.diameter_m,
197            )
198            .map_err(|error| SimulatorError::raised(format!("ANTENNA row: {error}")))?;
199    }
200    Ok(())
201}

203fn write_feeds(ms: &mut MeasurementSet, session: &Simulator) -> SimulatorResult<()> {
204    let identity = Value::Array(ArrayValue::Complex32(
205        ArrayD::from_shape_vec(
206            vec![2, 2],
207            vec![
208                Complex32::new(1.0, 0.0),
209                Complex32::new(0.0, 0.0),
210                Complex32::new(0.0, 0.0),
211                Complex32::new(1.0, 0.0),
212            ],
213        )
214        .expect("2x2"),
215    ));
216    let rows = session
217        .feeds()
218        .iter()
219        .map(|feed| {
220            row(
221                SubtableId::Feed,
222                vec![
223                    ("ANTENNA_ID", scalar_i32(feed.antenna_id)),
224                    ("BEAM_ID", scalar_i32(-1)),
225                    ("BEAM_OFFSET", f64_array(&[2, 2], vec![0.0; 4])),
226                    ("FEED_ID", scalar_i32(feed.feed_id)),
227                    ("INTERVAL", scalar_f64(1e30)),
228                    ("NUM_RECEPTORS", scalar_i32(2)),
229                    ("POLARIZATION_TYPE", string_array(&feed.polarization)),
230                    ("POL_RESPONSE", identity.clone()),
231                    ("POSITION", f64_array(&[3], vec![0.0; 3])),
232                    ("RECEPTOR_ANGLE", f64_array(&[2], vec![0.0; 2])),
233                    ("SPECTRAL_WINDOW_ID", scalar_i32(-1)),
234                    ("TIME", scalar_f64(0.0)),
235                ],
236            )
237        })
238        .collect();
239    add_rows(ms, SubtableId::Feed, rows)
240}

242fn write_spectral_windows(ms: &mut MeasurementSet, session: &Simulator) -> SimulatorResult<()> {
243    let mut spws = Vec::new();
244    let mut polarizations = Vec::new();
245    let mut descriptions = Vec::new();
246    for (index, window) in session.spectral_windows().iter().enumerate() {
247        let channels = window.channel_count;
248        let widths = vec![window.channel_width_hz; channels];
249        spws.push(row(
250            SubtableId::SpectralWindow,
251            vec![
252                (
253                    "CHAN_FREQ",
254                    f64_array(&[channels], window.channel_frequencies_hz()),
255                ),
256                ("CHAN_WIDTH", f64_array(&[channels], widths.clone())),
257                ("EFFECTIVE_BW", f64_array(&[channels], widths.clone())),
258                ("FLAG_ROW", scalar_bool(false)),
259                ("FREQ_GROUP", scalar_i32(0)),
260                ("FREQ_GROUP_NAME", scalar_string("Group 1")),
261                ("IF_CONV_CHAIN", scalar_i32(0)),
262                ("MEAS_FREQ_REF", scalar_i32(window.frame.casacore_code())),
263                ("NAME", scalar_string(&window.name)),
264                ("NET_SIDEBAND", scalar_i32(1)),
265                ("NUM_CHAN", scalar_i32(channels as i32)),
266                ("REF_FREQUENCY", scalar_f64(window.start_frequency_hz)),
267                ("RESOLUTION", f64_array(&[channels], widths)),
268                (
269                    "TOTAL_BANDWIDTH",
270                    scalar_f64(channels as f64 * window.channel_width_hz),
271                ),
272            ],
273        ));
274        let correlations = window.correlations.len();
275        let products = window.correlation_products();
276        let mut product_cells = vec![0; 2 * correlations];
277        for (correlation, pair) in products.iter().enumerate() {
278            product_cells[correlation] = pair[0];
279            product_cells[correlations + correlation] = pair[1];
280        }
281        polarizations.push(row(
282            SubtableId::Polarization,
283            vec![
284                ("CORR_PRODUCT", i32_array(&[2, correlations], product_cells)),
285                (
286                    "CORR_TYPE",
287                    i32_array(&[correlations], window.correlations.clone()),
288                ),
289                ("FLAG_ROW", scalar_bool(false)),
290                ("NUM_CORR", scalar_i32(correlations as i32)),
291            ],
292        ));
293        descriptions.push(row(
294            SubtableId::DataDescription,
295            vec![
296                ("FLAG_ROW", scalar_bool(false)),
297                ("POLARIZATION_ID", scalar_i32(index as i32)),
298                ("SPECTRAL_WINDOW_ID", scalar_i32(index as i32)),
299            ],
300        ));
301    }
302    add_rows(ms, SubtableId::SpectralWindow, spws)?;
303    add_rows(ms, SubtableId::Polarization, polarizations)?;
304    add_rows(ms, SubtableId::DataDescription, descriptions)?;
305    // Each window's frequencies are in its MEAS_FREQ_REF frame. casacore's
306    // table of codes ends with Undefined (64).
307    let (mut types, mut codes): (Vec<String>, Vec<i32>) = FrequencyRef::ALL
308        .iter()
309        .map(|frame| (frame.as_str().to_string(), frame.casacore_code()))
310        .unzip();
311    types.push("Undefined".to_string());
312    codes.push(64);
313    let table = ms
314        .subtable_mut(SubtableId::SpectralWindow)
315        .ok_or_else(|| SimulatorError::raised("SPECTRAL_WINDOW subtable missing"))?;
316    for column in ["CHAN_FREQ", "REF_FREQUENCY"] {
317        TableMeasDesc::new_variable_int(
318            column,
319            MeasureType::Frequency,
320            "MEAS_FREQ_REF",
321            types.clone(),
322            codes.clone(),
323        )
324        .and_then(|description| description.write(table))
325        .map_err(|error| SimulatorError::raised(format!("SPECTRAL_WINDOW.{column}: {error}")))?;
326    }
327    Ok(())
328}

330/// The units and measure references of the SOURCE columns CASA's
331/// simulator adds, as it writes them.
332fn write_source_measure_keywords(ms: &mut MeasurementSet) -> SimulatorResult<()> {
333    let table = ms
334        .subtable_mut(SubtableId::Source)
335        .ok_or_else(|| SimulatorError::raised("SOURCE subtable missing"))?;
336    for (column, measure, refer) in [
337        ("POSITION", MeasureType::Position, "ITRF"),
338        ("REST_FREQUENCY", MeasureType::Frequency, "LSRK"),
339        ("SYSVEL", MeasureType::RadialVelocity, "LSRK"),
340    ] {
341        TableMeasDesc::new_fixed(column, measure, refer)
342            .write(table)
343            .map_err(|error| SimulatorError::raised(format!("SOURCE.{column}: {error}")))?;
344    }
345    Ok(())
346}

348fn write_fields_and_sources(ms: &mut MeasurementSet, session: &Simulator) -> SimulatorResult<()> {
349    add_optional_columns(
350        ms,
351        SubtableId::Source,
352        &[
353            "POSITION",
354            "PULSAR_ID",
355            "REST_FREQUENCY",
356            "SYSVEL",
357            "TRANSITION",
358        ],
359    )?;
360    write_source_measure_keywords(ms)?;
361    let mut fields = Vec::new();
362    let mut sources = Vec::new();
363    for (index, field) in session.fields().iter().enumerate() {
364        let direction = direction_cell(field.direction_rad);
365        fields.push(row(
366            SubtableId::Field,
367            vec![
368                ("CODE", scalar_string(&field.code)),
369                ("DELAY_DIR", direction.clone()),
370                ("FLAG_ROW", scalar_bool(false)),
371                ("NAME", scalar_string(&field.name)),
372                ("NUM_POLY", scalar_i32(0)),
373                ("PHASE_DIR", direction.clone()),
374                ("REFERENCE_DIR", direction),
375                ("SOURCE_ID", scalar_i32(index as i32)),
376                ("TIME", scalar_f64(0.0)),
377            ],
378        ));
379        // With a spectral window defined by then, the source has one line,
380        // at the first window's reference frequency.
381        let (line_count, line_window) = match field.rest_frequency_hz {
382            Some(_) => (1, 0),
383            None => (0, -1),
384        };
385        let mut source = vec![
386            ("CALIBRATION_GROUP", scalar_i32(0)),
387            ("CODE", scalar_string(&field.code)),
388            ("DIRECTION", f64_array(&[2], field.direction_rad.to_vec())),
389            ("INTERVAL", scalar_f64(1e30)),
390            ("NAME", scalar_string(&field.name)),
391            ("NUM_LINES", scalar_i32(line_count)),
392            ("PROPER_MOTION", f64_array(&[2], vec![0.0; 2])),
393            ("PULSAR_ID", scalar_i32(0)),
394            ("SOURCE_ID", scalar_i32(index as i32)),
395            ("SPECTRAL_WINDOW_ID", scalar_i32(line_window)),
396            ("TIME", scalar_f64(field.source_time_s)),
397        ];
398        if let Some(rest_frequency_hz) = field.rest_frequency_hz {
399            source.extend([
400                ("REST_FREQUENCY", f64_array(&[1], vec![rest_frequency_hz])),
401                ("TRANSITION", string_array(&["X"])),
402                ("SYSVEL", f64_array(&[1], vec![0.0])),
403            ]);
404        }
405        sources.push(row(SubtableId::Source, source));
406    }
407    add_rows(ms, SubtableId::Field, fields)?;
408    add_rows(ms, SubtableId::Source, sources)
409}

411fn write_observations(ms: &mut MeasurementSet, session: &Simulator) -> SimulatorResult<()> {
412    let rows = session
413        .observations()
414        .iter()
415        .map(|observation| {
416            row(
417                SubtableId::Observation,
418                vec![
419                    ("FLAG_ROW", scalar_bool(false)),
420                    ("OBSERVER", scalar_string(&observation.observer)),
421                    ("PROJECT", scalar_string(&observation.project)),
422                    ("RELEASE_DATE", scalar_f64(0.0)),
423                    ("SCHEDULE_TYPE", scalar_string("")),
424                    ("TELESCOPE_NAME", scalar_string(&observation.telescope)),
425                    (
426                        "TIME_RANGE",
427                        f64_array(&[2], observation.time_range_s.to_vec()),
428                    ),
429                ],
430            )
431        })
432        .collect();
433    add_rows(ms, SubtableId::Observation, rows)
434}

458/// The subtables the MeasurementSet's save writes from generated columns:
459/// POINTING, one row per antenna per integration, which is not held as rows
460/// (see [`pointing`]).
461pub(crate) fn subtable_overrides(session: &Simulator) -> SimulatorResult<SubtableColumnOverrides> {
462    let scans = session
463        .scans()
464        .iter()
465        .map(pointing::PointingScan::of)
466        .collect::<Vec<_>>();
467    let pointing = pointing::pointing_overrides(&scans)
468        .map_err(|error| SimulatorError::raised(format!("POINTING columns: {error}")))?;
469    let mut overrides = SubtableColumnOverrides::new();
470    overrides.insert(SubtableId::Pointing, pointing);
471    Ok(overrides)
472}

476pub(crate) fn generated_bytes(session: &Simulator) -> u64 {
477    pointing::overrides_bytes(session.scans())
