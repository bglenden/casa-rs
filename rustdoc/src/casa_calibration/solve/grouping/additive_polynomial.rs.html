18#[derive(Clone, Debug)]
19pub(in crate::solve) struct Spectrum {
20    pub(in crate::solve) channels: Vec<usize>,
21    sample_shape: [usize; 2],
22    pairs: BTreeMap<(i32, i32), Baseline>,
23}

25#[derive(Clone, Debug)]
26struct Baseline {
27    sums: Vec<Complex32>,
28    observed: Vec<bool>,
29    weights: Vec<f32>,
30}
31
32impl Gather {
33    pub(super) fn observe(
34        &mut self,
35        row: &SolveSampleRow,
36        samples: &SolveSamples<'_>,
37        correlations: &[i32],
38        channels: SelectedChannels<'_>,
39    ) -> Result<bool, GainSolveError> {
40        let count = correlations.len();
41        if !matches!(count, 1 | 2 | 4) {
42            return Err(GainSolveError::Baseline(
43                "unsupported additive fit correlation count".into(),
44            ));
45        }
46        let sample_shape = [samples.data.nrows(), samples.data.ncols()];
47        let window = self
48            .windows
49            .entry(row.data_desc_id)
50            .or_insert_with(|| Spectrum {
51                channels: channels.indices.to_vec(),
52                sample_shape,
53                pairs: BTreeMap::new(),
54            });
55        if sample_shape != window.sample_shape || channels.indices != window.channels {
56            return Err(GainSolveError::UnsupportedParameterShape {
57                path: format!("DATA_DESC_ID {} additive fit", row.data_desc_id),
58                shape: sample_shape.to_vec(),
59            });
60        }
61        let accepted = channels
62            .indices
63            .iter()
64            .copied()
65            .enumerate()
66            .filter(|&(_, channel)| {
67                !channels.excluded.get(channel).copied().unwrap_or(false)
68                    && !samples.flags.column(channel).iter().any(|flagged| *flagged)
69            })
70            .collect::<Vec<_>>();
71        let Some(&(_, first)) = accepted.first() else {
72            return Ok(false);
73        };
74        let (p, q) = (row.antenna1, row.antenna2);
75        let spectrum = window
76            .pairs
77            .entry((p.min(q), p.max(q)))
78            .or_insert_with(|| Baseline {
79                sums: vec![Complex32::ZERO; count * channels.indices.len()],
80                observed: vec![false; count * channels.indices.len()],
81                weights: vec![0.; count],
82            });
83        let mut used = false;
84        for hand in 0..count {
85            let source_hand = if p > q && count == 4 && matches!(hand, 1 | 2) {
86                3 - hand
87            } else {
88                hand
89            };
90            let weight = samples.weights[[source_hand, first]];
91            if !weight.is_finite() || weight <= 0. {
92                continue;
93            }
94            // VI1 divides every channel by the same total baseline weight.
95            spectrum.weights[hand] += weight;
96            for &(position, channel) in &accepted {
97                let data = samples.data[[source_hand, channel]];
98                let value = if p > q { data.conj() } else { data };
99                let index = position * count + hand;
100                spectrum.sums[index] += value * weight;
101                spectrum.observed[index] = true;
102            }
103            used = true;
104        }
105        if used {
106            let first_weight = samples.weights[[0, first]];
107            let last_weight = samples.weights[[count - 1, first]];
108            self.time.observe(
109                row.time_seconds,
110                f64::from(0.5 * (first_weight + last_weight)),
111            );
112        }
113        Ok(used)
114    }

116    pub(in crate::solve) fn solutions(
117        &self,
118        group: &SolveAccumulator,
119        antennas: &BTreeSet<i32>,
120        frequencies: &BTreeMap<i32, Vec<f64>>,
121        domain: [f64; 2],
122        terms: usize,
123        mode: GainSolveMode,
124    ) -> Vec<SolutionRow> {
125        let correlations = *group
126            .baseline_correlation_counts
127            .first()
128            .expect("fit shape");
129        let time = self.time.mean().unwrap_or_else(|| group.aggregate_time());
130        let cells = terms * correlations;
131        antennas
132            .iter()
133            .flat_map(|&p| antennas.range(p..).map(move |&q| (p, q)))
134            .map(|pair| {
135                let mut gains = vec![Complex32::ONE; cells];
136                let mut flags = vec![true; cells];
137                for hand in 0..correlations {
138                    let observations = self
139                        .windows
140                        .iter()
141                        .filter_map(|(dd, window)| {
142                            window
143                                .pairs
144                                .get(&pair)
145                                .map(|averages| (&frequencies[dd], window, averages))
146                        })
147                        .flat_map(|(frequencies, window, baseline)| {
148                            let weight = baseline.weights[hand];
149                            window.channels.iter().enumerate().filter_map(
150                                move |(position, &channel)| {
151                                    let cell = position * correlations + hand;
152                                    if !baseline.observed[cell] {
153                                        return None;
154                                    }
155                                    let location = coordinate(frequencies[channel], domain);
156                                    let value = baseline.sums[cell] / weight;
157                                    let value = match mode {
158                                        GainSolveMode::Phase if value.norm() > 0. => {
159                                            value / value.norm()
160                                        }
161                                        GainSolveMode::Amplitude => {
162                                            Complex32::new(value.norm(), 0.)
163                                        }
164                                        _ => value,
165                                    };
166                                    Some((
167                                        location,
168                                        value,
169                                        f64::from(weight) / window.channels.len() as f64,
170                                    ))
171                                },
172                            )
173                        })
174                        .collect::<Vec<_>>();
175                    let coefficients = fit(&observations, terms);
176                    if !coefficients.is_empty() {
177                        for order in 0..terms {
178                            let cell = order * correlations + hand;
179                            gains[cell] =
180                                coefficients.get(order).copied().unwrap_or(Complex32::ZERO);
181                            flags[cell] = order >= coefficients.len();
182                        }
183                    }
184                }
185                SolutionRow {
186                    time_seconds: time,
187                    field_id: group.field_id,
188                    spw_id: group.spw_id,
189                    antenna_id: pair.0,
190                    scan_number: group.scan_number(),
191                    observation_id: group.observation_id,
192                    refant_id: pair.1,
193                    gains,
194                    flags,
195                    param_errors: vec![0.; cells],
196                    snrs: vec![0.; cells],
197                    weights: Vec::new(),
198                }
199            })
200            .collect()
201    }
202}

204pub(in crate::solve) fn coordinate(frequency: f64, [low, high]: [f64; 2]) -> f64 {
205    if low == high {
206        0.
207    } else {
208        2. * (frequency - low) / (high - low) - 1.
209    }
210}
211
212fn fit(observations: &[(f64, Complex32, f64)], terms: usize) -> Vec<Complex32> {
213    let terms = terms.min(observations.len());
214    if terms == 0 {
215        return Vec::new();
216    }
217    let scale = observations
218        .iter()
219        .map(|sample| sample.2)
220        .fold(0., f64::max);
221    let mut rows = observations
222        .iter()
223        .map(|&(x, sample, weight)| {
224            let basis = std::iter::successors(Some(1.), move |term| Some(term * x))
225                .take(terms)
226                .collect();
227            (basis, f64::from(sample.re), weight / scale)
228        })
229        .collect::<Vec<_>>();
230    let Some(real) = solve_weighted_least_squares(&rows, terms) else {
231        return Vec::new();
232    };
233    for (row, &(_, sample, _)) in rows.iter_mut().zip(observations) {
234        row.1 = f64::from(sample.im);
235    }
236    let Some(imaginary) = solve_weighted_least_squares(&rows, terms) else {
237        return Vec::new();
238    };
239    real.into_iter()
240        .zip(imaginary)
241        .map(|(re, im)| Complex32::new(re as f32, im as f32))
242        .collect()
243}
