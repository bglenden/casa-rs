25/// The operator of one image domain and the resampler that places rows on
26/// its planes.
27pub(crate) struct DomainOperator {
28    pub(crate) operator: MeasurementOperator,
29    pub(crate) resampler: SpectralResampler,
30    /// Whether the kernel set grids a sensitivity (weight) image
31    /// (`ConvolutionFunctionSet::weight_taps`): mosaic and AW.
32    pub(crate) weight_image: bool,
33    /// The charge of the kernel set, held while the operator lives.
34    kernels: Kernels,
35}
36
37/// The charge of a kernel set: admitted before the set was built or, for
38/// an AW catalog whose holders are the passes' workers, admitted once the
39/// run has chosen its workers (#737).
40enum Kernels {
41    Admitted {
42        _charge: Reservation,
43    },
44    /// An AW catalog's charge at its cache's bound, for at most `ceiling`
45    /// holders. An `automatic` bound (`cf_resident_mb = auto`) rises, as the
46    /// charge is admitted, into the memory the chosen workers leave (#449).
47    Catalog {
48        charge: CacheCharge,
49        bound: CacheBound,
50        ceiling: usize,
51        automatic: bool,
52    },
53}
54
55impl DomainOperator {
56    /// The charge held with an already-admitted kernel set.
57    pub(crate) fn retained_kernel_bytes(&self) -> u64 {
58        match &self.kernels {
59            Kernels::Admitted { _charge } => _charge.memory(),
60            Kernels::Catalog { .. } => 0,
61        }
62    }
63
64    /// Bytes the kernel set still to be admitted holds for passes on
65    /// `workers` streaming workers, at most the ceiling it was opened for,
66    /// at its cache's bound now; zero for a set already admitted.
67    pub(crate) fn pending_kernel_bytes(&self, workers: usize) -> u64 {
68        match &self.kernels {
69            Kernels::Admitted { .. } => 0,
70            Kernels::Catalog {
71                charge,
72                bound,
73                ceiling,
74                ..
75            } => charge.bytes(workers.clamp(1, *ceiling), bound.bytes()),
76        }
77    }
78
79    /// Admit the kernel set's charge for passes on `workers` streaming
80    /// workers, if it is still to be admitted. An automatic cell cache first
81    /// takes what it can of `spare`, the bytes the run's chosen workers
82    /// leave free: every cell when they fit, otherwise as much more bound as
83    /// `spare` allows. What the raise costs comes off `spare`, and what it
84    /// saves goes back for the next catalog ([`CacheCharge::raise_within`]).
85    pub(crate) fn admit_kernels(
86        &mut self,
87        admission: KernelAdmission<'_>,
88        workers: usize,
89        spare: &mut u64,
90    ) -> Result<(), ImagingError> {
91        if let Kernels::Catalog {
92            charge,
93            bound,
94            ceiling,
95            automatic,
96        } = &self.kernels
97        {
98            let holders = workers.clamp(1, *ceiling);
99            if *automatic {
100                charge.raise_within(holders, bound, spare);
101                tracing::info!(
102                    "AW cell cache: {:.0} MB of {:.0} MB of cells resident at most (auto)",
103                    bound.bytes().min(charge.every_bytes() as usize) as f64 / 1e6,
104                    charge.every_bytes() as f64 / 1e6
105                );
106            }
107            let memory = charge.bytes(holders, bound.bytes());
108            self.kernels = Kernels::Admitted {
109                _charge: admission.admit("convolution functions", memory)?,
110            };
111        }
112        Ok(())
113    }
114}

116/// Correlation types every block delivers, in block order: the first
117/// source's selection, which every partition of every source must share.
118pub(crate) fn selected_correlations(
119    problem: &CompiledProblem,
120) -> Result<Vec<Correlation>, ImagingError> {
121    let mut layouts = problem
122        .observation_transaction()
123        .read_set()
124        .sources()
125        .iter()
126        .flat_map(|source| source.selection().partitions())
127        .map(|selection| {
128            selection
129                .products()
130                .iter()
131                .map(|product| product.correlation_type())
132                .collect::<Vec<_>>()
133        });
134    let first = layouts.next().ok_or(ImagingError::Unsupported {
135        reason: "the selection names no correlations",
136    })?;
137    if layouts.any(|layout| layout != first) {
138        return Err(ImagingError::Unsupported {
139            reason: "selected partitions differ in their correlations",
140        });
141    }
142    Ok(first)
143}

145/// The most samples a row of `problem`'s source places and predicts on any
146/// of its image domains, which share one resampler
147/// ([`SpectralResampler::samples_per_row`],
148/// [`SpectralResampler::prediction_samples_per_row`]).
149pub(crate) fn row_samples(problem: &CompiledProblem) -> Result<RowSamples, ImagingError> {
150    let resampler = resampler(problem, basis(problem)?)?;
151    let spectrum = row_layout(problem).spectrum;
152    Ok(RowSamples {
153        placed: resampler.samples_per_row(spectrum),
154        predicted: resampler.prediction_samples_per_row(spectrum.channels),
155    })
156}

158/// The grid basis of the compiled reconstruction basis. A Taylor basis
159/// expands about the image's reference frequency.
160pub(crate) fn basis(problem: &CompiledProblem) -> Result<Basis, ImagingError> {
161    Ok(match problem.reconstruction().basis() {
162        ReconstructionBasis::Constant => Basis::Constant,
163        ReconstructionBasis::Taylor { terms } => Basis::Taylor {
164            terms: terms as u32,
165            reference_hz: reference_frequency_hz(problem)?,
166        },
167        ReconstructionBasis::ChannelLocal { channels } => Basis::ChannelLocal {
168            planes: channels as u32,
169        },
170    })
171}

186/// The grid precision: the requested one, else (`gridprecision = auto`) f64 on the
187/// CPU for every basis, since CASA accumulates its grids in double
188/// (`useDoublePrec`, `SynthesisUtilMethods.cc:3722`), and f32 on Metal,
189/// which accumulates in `f32` only.
190///
191/// A cube is no exception: an f32 running sum drifts with the number of
192/// samples a cell takes, and a mosaic's weight image takes every sample of
193/// a channel at the uv origin (#659).
194pub(crate) const fn precision(
195    backend: BackendChoice,
196    requested: Option<GridPrecision>,
197) -> GridPrecision {
198    if let Some(precision) = requested {
199        return precision;
200    }
201    match backend {
202        BackendChoice::Cpu => GridPrecision::F64,
203        BackendChoice::Metal => GridPrecision::F32,
204    }
205}

207/// Whether the problem grids with the standard kernel set, the one facets
208/// take.
209pub(crate) fn standard_kernel_set(problem: &CompiledProblem) -> bool {
210    kernel_set_kind(problem) == KernelSetKind::Standard
211}

213/// Whether the Metal backend grids the problem's kernel set: the standard
214/// spheroidal set, and the mosaic primary beams (#787). The mosaic's PSF
215/// and prediction cells are its imaging cells, and its weight cells share
216/// their support, so Metal's one tap route per mode serves it. Its weight
217/// image keeps f32 precision because the weight owner sums it by
218/// contribution first (#792). Each Metal backend of a pass copies the
219/// cells it meets to the device, which the wave demand admits per backend
220/// (`casa_imaging_metal::kernel_bytes`, #793 review). W-planes and the AW
221/// catalog run on the CPU until #653's rows for them.
222pub(crate) fn metal_kernel_set(problem: &CompiledProblem) -> bool {
223    match kernel_set_kind(problem) {
224        KernelSetKind::Standard => true,
225        // Metal predicts with the cells it grids with, which a mosaic's
226        // conjugate beams part.
227        KernelSetKind::Mosaic => !mosaic_conjugate_beams(problem),
228        KernelSetKind::WPlanes | KernelSetKind::Aw | KernelSetKind::Awp2 => false,
229    }
230}

232/// Whether the problem's mosaic grids with its conjugate beams.
233fn mosaic_conjugate_beams(problem: &CompiledProblem) -> bool {
234    problem
235        .science()
236        .measurement_equation()
237        .mosaic_conjugate_beams()
238}

240/// Whether the problem's kernel set grids several MeasurementSets: the
241/// standard set and the W planes, which key no row on its MeasurementSet,
242/// and the mosaic set, which keys its beams on each row's window key and
243/// dish classes over every MeasurementSet ([`StoreKeys`]). The AW catalog
244/// keys its cells on the first MeasurementSet's frequencies and first
245/// cross-correlation sample.
246pub(crate) fn multi_store_kernel_set(problem: &CompiledProblem) -> bool {
247    matches!(
248        kernel_set_kind(problem),
249        KernelSetKind::Standard | KernelSetKind::WPlanes | KernelSetKind::Mosaic
250    )
251}

253/// Which kernel set the compiled problem's measurement equation names.
254#[derive(Clone, Copy, Debug, PartialEq, Eq)]
255enum KernelSetKind {
256    /// The standard spheroidal set (`GridFT`).
257    Standard,
258    /// W-projection planes (`WProjectFT`).
259    WPlanes,
260    /// The heterogeneous-array mosaic beams (`MosaicFT` with
261    /// `HetArrayConvFunc`).
262    Mosaic,
263    /// The AW catalog (`AWProjectFT`).
264    Aw,
265    /// The resident AWPLPG kernels.
266    Awp2,
267}
268
269fn kernel_set_kind(problem: &CompiledProblem) -> KernelSetKind {
270    let equation = problem.science().measurement_equation();
271    if equation.aw_projection().is_some() {
272        KernelSetKind::Aw
273    } else if equation.w_projection().is_some() {
274        KernelSetKind::WPlanes
275    } else if problem.science().instrument_model() == Some(InstrumentModel::CasaVlaAwplpg) {
276        KernelSetKind::Awp2
277    } else if matches!(
278        problem.science().instrument_model(),
279        Some(
280            InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1
281                | InstrumentModel::CasaEvlaCommonInterferometricResponseV1
282                | InstrumentModel::CasaAiryHeterogeneousInterferometricResponseV1
283        )
284    ) {
285        KernelSetKind::Mosaic
286    } else {
287        KernelSetKind::Standard
288    }
289}

291/// The operator of a chart of `extent` (an image domain, or one of its
292/// facets) with the kernel set the problem names: the
293/// standard spheroidal set, W-projection planes sized by the W contract,
294/// the mosaic primary beams of the selected windows, or the AW catalog of
295/// `aw_catalog`, in the requested precision or the one `backend` grids at.
296/// The mosaic set takes one dish per aperture class of `keys`
297/// (`HetArrayConvFunc::findAntennaSizes`), the order the rows' antenna
298/// types index, and the beams of each window under its key. Mosaic and AW
299/// grid without padding (CASA `MosaicFT`, `AWProjectFT`); the others on
300/// CASA's composite-padded grid.
301/// The kernel set is admitted as [`kernel_set`] says, and its charge lives
302/// with the operator.
303#[allow(clippy::too_many_arguments)]
304pub(crate) fn domain_operator(
305    problem: &CompiledProblem,
306    extent: ImageExtent,
307    correlations: &[Correlation],
308    backend: BackendChoice,
309    requested_precision: Option<GridPrecision>,
310    requested_padding: Option<PaddingFactor>,
311    deployment: (Option<&AwCatalogDeployment>, Option<&Awp2Deployment>),
312    keys: &StoreKeys,
313    admission: KernelAdmission<'_>,
314) -> Result<DomainOperator, ImagingError> {
315    let kind = kernel_set_kind(problem);
316    // Each CASA facet's GridFT pads its own facet grid.
317    let padding = match kind {
318        KernelSetKind::Standard => {
319            GridPadding::CasaComposite(requested_padding.unwrap_or(PaddingFactor::TCLEAN))
320        }
321        KernelSetKind::WPlanes => GridPadding::CasaComposite(PaddingFactor::TCLEAN),
322        KernelSetKind::Mosaic | KernelSetKind::Aw | KernelSetKind::Awp2 => GridPadding::None,
323    };
324    let geometry = GridGeometry::new(extent, padding)?;
325    let polarization = PolarizationRouting::compile(
326        correlations,
327        problem.reconstruction().polarization().coordinates(),
328    )?;
329    let basis = basis(problem)?;
330    let resampler = resampler(problem, basis)?;
331    let (cf, kernels) = kernel_set(
332        problem,
333        kind,
334        (&geometry, &polarization),
335        deployment,
336        keys,
337        admission,
338    )?;
339    Ok(DomainOperator {
340        operator: MeasurementOperator::new(
341            geometry,
342            basis,
343            polarization,
344            cf,
345            precision(backend, requested_precision),
346        ),
347        resampler,
348        weight_image: matches!(
349            kind,
350            KernelSetKind::Mosaic | KernelSetKind::Aw | KernelSetKind::Awp2
351        ),
352        kernels,
353    })
354}

356/// Where a run admits its kernel sets: its host and policy, and the most
357/// workers that fetch cells from a paged set, the policy's; the run admits
358/// a paged set's charge once it has chosen how many do
359/// ([`DomainOperator::admit_kernels`]).
360#[derive(Clone, Copy)]
361pub(crate) struct KernelAdmission<'a> {
362    pub(crate) host: &'a HostResources,
363    pub(crate) policy: &'a ResourcePolicy,
364    pub(crate) workers: usize,
365}
366
367impl KernelAdmission<'_> {
368    fn admit(self, phase: &'static str, memory: u64) -> Result<Reservation, ImagingError> {
369        Ok(admit(self.host, self.policy, &Demand { phase, memory })?)
370    }
371}

373/// Build one chart's W kernels and retain their exact resident charge.
374fn w_kernel_set(
375    geometry: &GridGeometry,
376    polarization: &PolarizationRouting,
377    count: WPlaneCount,
378    admission: KernelAdmission<'_>,
379) -> Result<(WPlanes, Kernels), ImagingError> {
380    let screening = admission.admit(
381        "W-projection screens",
382        WPlanes::screens_bytes(geometry, count)?,
383    )?;
384    let screens = WPlanes::screens(geometry, polarization, count)?;
385    let mut kernels = admission.admit("convolution functions", screens.kernel_bytes())?;
386    let set = screens.finish()?;
387    drop(screening);
388    kernels.retain(set.resident_bytes());
389    tracing::debug!(
390        target: "casa_imaging_application::w_kernel_charge",
391        resident_bytes = set.resident_bytes(),
392        reserved_bytes = kernels.memory(),
393        "W-projection kernels retained"
394    );
395    Ok((set, Kernels::Admitted { _charge: kernels }))
396}

398/// The kernel set of `kind` on `geometry` routing `polarization`, with the
399/// charge of what it holds, admitted before it is built. W-projection and
400/// mosaic sets are built in two stages, their screens and then the kernels
401/// cut from them, each admitted before it allocates; the screens' charge
402/// is released once the kernels are built. An AW catalog's charge, its
403/// cache bound and what its workers can hold beyond it
404/// ([`AwCatalog::charge_bytes`]), is known once its index is read, for
405/// each count of workers up to `admission`'s, and is admitted when the run
406/// has chosen its workers.
407fn kernel_set(
408    problem: &CompiledProblem,
409    kind: KernelSetKind,
410    (geometry, polarization): (&GridGeometry, &PolarizationRouting),
411    (aw_catalog, awp2): (Option<&AwCatalogDeployment>, Option<&Awp2Deployment>),
412    keys: &StoreKeys,
413    admission: KernelAdmission<'_>,
414) -> Result<(Box<dyn ConvolutionFunctionSet>, Kernels), ImagingError> {
415    const PHASE: &str = "convolution functions";
416    Ok(match kind {
417        KernelSetKind::Standard => {
418            let kernels = admission.admit(PHASE, Spheroidal::bytes(geometry))?;
419            (
420                Box::new(Spheroidal::new(geometry, polarization)),
421                Kernels::Admitted { _charge: kernels },
422            )
423        }
424        KernelSetKind::WPlanes => {
425            let contract = problem
426                .science()
427                .measurement_equation()
428                .w_projection()
429                .expect("the kind names a W contract");
430            let count = match (contract.planes(), contract.statistics()) {
431                (Some(planes), _) => WPlaneCount::Fixed(planes.get() as u32),
432                (None, Some(statistics)) => WPlaneCount::Auto {
433                    min_w: statistics.minimum_abs_w_lambda(),
434                    max_w: contract.maximum_abs_w_lambda(),
435                    rms_w: statistics.rms_w_lambda(),
436                },
437                (None, None) => {
438                    return Err(ImagingError::Unsupported {
439                        reason: "an automatic W-plane count needs the selection's w statistics",
440                    });
441                }
442            };
443            let (set, kernels) = w_kernel_set(geometry, polarization, count, admission)?;
444            (Box::new(set), kernels)
445        }
446        KernelSetKind::Mosaic => {
447            if keys.dish_classes.is_empty() {
448                return Err(ImagingError::Unsupported {
449                    reason: "the mosaic set needs the dishes of its instrument model",
450                });
451            }
452            let dishes = keys
453                .dish_classes
454                .iter()
455                .map(|class| match class {
456                    AntennaResponseClass::CasaAlma12m => AiryDish::casa_alma(12.0, geometry).into(),
457                    AntennaResponseClass::CasaAca7m => AiryDish::casa_alma(7.0, geometry).into(),
458                    AntennaResponseClass::CasaEvla => PrimaryBeam::Evla,
459                    AntennaResponseClass::CasaAiry {
460                        dish_diameter_m_bits,
461                    } => AiryDish::casa_airy(f64::from_bits(*dish_diameter_m_bits)).into(),
462                })
463                .collect::<Vec<_>>();
464            let frequency_hz = reference_frequency_hz(problem)?;
465            let windows = mosaic_windows(problem, keys)?;
466            let rules = if mosaic_conjugate_beams(problem) {
467                MosaicRules::TcleanConjugateBeams
468            } else {
469                MosaicRules::Tclean
470            };
471            let screening = admission.admit(
472                "mosaic screens",
473                MosaicPb::screens_bytes(rules, geometry, frequency_hz, &dishes, &windows)?,
474            )?;
475            let screens = MosaicPb::screens(
476                rules,
477                geometry,
478                polarization,
479                frequency_hz,
480                &dishes,
481                &windows,
482            )?;
483            let mut kernels = admission.admit(PHASE, screens.kernel_bytes())?;
484            let set = screens.finish()?;
485            drop(screening);
486            kernels.retain(set.resident_bytes());
487            (Box::new(set), Kernels::Admitted { _charge: kernels })
488        }
489        KernelSetKind::Awp2 => {
490            let deployment = awp2.ok_or(ImagingError::Unsupported {
491                reason: "AWP2 requires a VLA dish surface and beam frequencies",
492            })?;
493            let sampling = Awp2Sampling::new(geometry, deployment.single_field)?;
494            let screen_charge = admission.admit(
495                "AWP2 aperture screens",
496                sampling.screens_bytes(deployment.frequencies_hz.len(), deployment.w_planes.len()),
497            )?;
498            let screens = sampling.screens(
499                &deployment.frequencies_hz,
500                deployment.surface.clone(),
501                deployment.legacy_vla,
502                &deployment.w_planes,
503            )?;
504            let mut kernels = admission.admit(PHASE, screens.kernel_bytes(geometry))?;
505            let set = screens.finish(
506                geometry,
507                polarization,
508                deployment.frequencies_hz.clone(),
509                deployment.w_planes.clone(),
510                deployment.row_frequency_hz,
511            );
512            drop(screen_charge);
513            kernels.retain(set.resident_bytes());
514            (Box::new(set), Kernels::Admitted { _charge: kernels })
515        }
516        KernelSetKind::Aw => {
517            let deployment = aw_catalog.ok_or(ImagingError::Unsupported {
518                reason: "an A-projection run needs its convolution-function catalog",
519            })?;
520            let catalog = AwCatalog::open_casa(
521                &deployment.root,
522                deployment.indexing.clone(),
523                geometry,
524                polarization,
525                deployment.resident_bytes,
526            )?;
527            let kernels = Kernels::Catalog {
528                charge: catalog.cache_charge(),
529                bound: catalog.cache_bound(),
530                ceiling: admission.workers.max(1),
531                automatic: deployment.automatic,
532            };
533            (Box::new(catalog), kernels)
534        }
535    })
536}

538/// The selected spectral windows of every store under their keys
539/// (`keys`), as the mosaic set keys them (CASA's `msId_spw`): every
540/// channel of the window, its first channel width and the selected
541/// channels' frequencies in LSRK at the window's first visibility buffer
542/// in its store, where `HetArrayConvFunc` evaluates the window's beams
543/// (`interpVisFreq_p`, `FTMachine::matchChannel`).
544fn mosaic_windows(
545    problem: &CompiledProblem,
546    keys: &StoreKeys,
547) -> Result<Vec<MosaicWindow>, ImagingError> {
548    let mut windows = Vec::new();
549    let sources = problem.observation_transaction().read_set().sources();
550    for (store, source) in sources.iter().enumerate() {
551        for window in source.selection().spectral_windows() {
552            let catalog = window
553                .coordinate_catalog()
554                .ok_or(ImagingError::Unsupported {
555                    reason: "the mosaic set needs every window's channel frequencies",
556                })?;
557            let factor = window
558                .first_buffer_lsrk_factor()
559                .ok_or(ImagingError::Unsupported {
560                    reason: "the mosaic set needs every window's LSRK factor",
561                })?;
562            let selected = window
563                .channel_indices()
564                .iter()
565                .map(|index| {
566                    catalog
567                        .channel_frequency_hz(*index as usize)
568                        .map(|hz| hz * factor)
569                        .ok_or(ImagingError::Unsupported {
570                            reason: "a selected channel lies outside its window",
571                        })
572                })
573                .collect::<Result<Vec<_>, _>>()?;
574            windows.push(MosaicWindow {
575                spectral_window: keys.window(store, window.spectral_window_id()),
576                window_frequencies_hz: catalog.channel_frequencies_hz().to_vec(),
577                channel_width_hz: catalog.first_channel_width_hz(),
578                selected_frequencies_hz: selected,
579            });
580        }
581    }
582    Ok(windows)
583}

585pub(crate) fn image_extent(domain: &CompiledImageDomain) -> Result<ImageExtent, ImagingError> {
586    let direction = domain.direction();
587    let reference = direction.reference_pixel();
588    if reference
589        .iter()
590        .any(|pixel| pixel.fract() != 0.0 || *pixel < 0.0)
591    {
592        return Err(ImagingError::Unsupported {
593            reason: "an odd image size is not supported: the reference pixel, half the size, \
594                     must be a whole pixel",
595        });
596    }
597    Ok(ImageExtent {
598        shape: domain.shape().pixels(),
599        increment_rad: direction.increment_rad(),
600        reference_pixel: [reference[0] as usize, reference[1] as usize],
601    })
602}

604/// CASA `FTMachine` channel mapping for a channel-local basis; every native
605/// channel on plane 0 otherwise.
606fn resampler(problem: &CompiledProblem, basis: Basis) -> Result<SpectralResampler, ImagingError> {
607    let Basis::ChannelLocal { planes } = basis else {
608        return Ok(SpectralResampler::direct(basis)?);
609    };
610    let Some([reference_pixel, reference_frequency_hz, increment_hz]) =
611        problem.geometry().spectral().linear_grid()
612    else {
613        return Err(ImagingError::Unsupported {
614            reason: "a tabular spectral axis has no uniform channel width",
615        });
616    };
617    let kernel = match problem.science().spectral().sampling().kernel() {
618        SpectralKernel::Nearest | SpectralKernel::Identity => {
619            casa_imaging_operator::SpectralKernel::Nearest
620        }
621        SpectralKernel::Linear => casa_imaging_operator::SpectralKernel::Linear,
622        SpectralKernel::Cubic => casa_imaging_operator::SpectralKernel::Cubic,
623        SpectralKernel::ChannelIntegration { .. } => {
624            return Err(ImagingError::Unsupported {
625                reason: "cube interpolation is nearest, linear or cubic",
626            });
627        }
628    };
629    let first_hz = reference_frequency_hz - reference_pixel * increment_hz;
630    Ok(
631        SpectralResampler::channel_local(
632            SpectralAxis::new(first_hz, increment_hz, planes)?,
633            kernel,
634        )
635        .with_terminal_plane_outside(problem.geometry().spectral().terminal_plane_outside()),
636    )
637}

639/// The widest spacing between adjacent selected native channels of any
640/// selected spectral window, from the storage owner's `CHAN_FREQ` catalog,
641/// widened by [`FRAME_MARGIN`]. A window without a catalog makes it
642/// infinite, so every wave holds the whole model.
643pub(crate) fn native_spacing_hz(problem: &CompiledProblem) -> f64 {
644    let mut widest = 0.0_f64;
645    for window in problem
646        .observation_transaction()
647        .read_set()
648        .sources()
649        .iter()
650        .flat_map(|source| source.selection().spectral_windows())
651    {
652        let Some(catalog) = window.coordinate_catalog() else {
653            return f64::INFINITY;
654        };
655        for pair in window.channel_indices().windows(2) {
656            let (Some(first), Some(second)) = (
657                catalog.channel_frequency_hz(pair[0] as usize),
658                catalog.channel_frequency_hz(pair[1] as usize),
659            ) else {
660                return f64::INFINITY;
661            };
662            widest = widest.max((second - first).abs());
663        }
664    }
665    widest * (1.0 + FRAME_MARGIN)
666}

668/// Shape of the weight-density grid of `domain`: one global plane on CASA's
669/// `VisImagingWeight` cells, or one plane per output channel plus the
670/// compiled padding planes on each side on `BriggsCubeWeightor` cells;
671/// under `mosweight` those planes for each of its fields.
672pub(crate) fn density_shape(
673    problem: &CompiledProblem,
674    domain: &CompiledImageDomain,
675) -> Result<DensityGridShape, ImagingError> {
676    let [width, height] = domain.shape().pixels();
677    let scope = problem.weighting().density_scope();
678    let (planes, padding, rule) = if scope.is_per_channel() {
679        let (planes, padding) = padded_channel_planes(problem)?;
680        (planes, padding, DensityCellRule::Cube)
681    } else {
682        (1, 0, DensityCellRule::Standard)
683    };
684    let fields = scope.fields().cloned();
685    let increment = domain.direction().increment_rad();
686    let cells = problem.weighting().density_cells();
687    let ([width, height], increment_rad, box_half_width) =
688        density_cells(cells, rule, [width, height], increment);
689    // The grid is charged by its bytes ([`DensityGridShape::bytes`]): every
690    // field's planes, each numbered in `u32`.
691    fields
692        .as_ref()
693        .map_or(Some(planes), |fields| planes.checked_mul(fields.count()))
694        .filter(|planes| u32::try_from(*planes).is_ok())
695        .and_then(|planes| {
696            width
697                .checked_mul(height)
698                .and_then(|plane| plane.checked_mul(planes))
699                .and_then(|cells| cells.checked_add(planes))
700        })
701        .and_then(|values| values.checked_mul(size_of::<f64>()))
702        .ok_or(ImagingError::Unsupported {
703            reason: "the weight-density grid overflows",
704        })?;
705    Ok(DensityGridShape {
706        width,
707        height,
708        planes,
709        padding,
710        box_half_width,
711        increment_rad,
712        rule,
713        fields,
714    })
715}

717/// The planes of one per-channel density and the padding planes on each
718/// side among them: CASA always pads the cube density axis
719/// (`BriggsCubeWeightor::estimateSwingChanPad`, or a fixed swing pad under
720/// `multiField`), and a contract without the padding cannot reproduce it.
721fn padded_channel_planes(problem: &CompiledProblem) -> Result<(usize, u32), ImagingError> {
722    let overflow = || ImagingError::Unsupported {
723        reason: "the padded density axis overflows",
724    };
725    let padding =
726        problem
727            .weighting()
728            .casa_cube_density_padding()
729            .ok_or(ImagingError::Unsupported {
730                reason: "a per-channel weight density needs its compiled padding",
731            })?;
732    let planes = padding
733        .checked_mul(2)
734        .and_then(|padding| padding.checked_add(problem.geometry().spectral().output_channels()))
735        .ok_or_else(overflow)?;
736    Ok((planes, u32::try_from(padding).map_err(|_| overflow())?))
737}

739/// The cells, cell increments and box half-width of a density grid over an
740/// image of `pixels` and `increment` (`SynthesisImagerVi2::weight`). CASA's
741/// continuum uniform and Briggs density (`VisImagingWeight`) has `n` cells
742/// a side for `npixels = n`, else the image's, each `(n·Δ)/n` wide;
743/// superuniform's box and the cube weightor's keep the image's cells and
744/// increments.
745fn density_cells(
746    cells: DensityCells,
747    rule: DensityCellRule,
748    pixels: [usize; 2],
749    increment: [f64; 2],
750) -> ([usize; 2], [f64; 2], usize) {
751    let visimagingweight = |[nx, ny]: [usize; 2]| {
752        [
753            DensityGridShape::visimagingweight_increment(nx, increment[0]),
754            DensityGridShape::visimagingweight_increment(ny, increment[1]),
755        ]
756    };
757    match (cells, rule) {
758        (DensityCells::Box(half_width), _) => (pixels, increment, half_width.get()),
759        (DensityCells::Square(side), _) => {
760            let side = [side.get(); 2];
761            (side, visimagingweight(side), 0)
762        }
763        (DensityCells::Image, DensityCellRule::Standard) => (pixels, visimagingweight(pixels), 0),
764        (DensityCells::Image, DensityCellRule::Cube) => (pixels, increment, 0),
765    }
766}
