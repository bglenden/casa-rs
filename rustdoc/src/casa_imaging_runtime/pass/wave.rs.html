17/// Row chunks per worker in the placement stage; enough to balance rows
18/// whose channel counts differ after flagging and the support test.
19const CHUNKS_PER_WORKER: usize = 4;
20
21/// One owner's accumulator and backend scratch.
22struct Owner {
23    acc: Option<GridAccumulator>,
24    backend: CpuBackend,
25    images: Option<NormalImages>,
26}
27
28/// One image domain's share of a wave.
29struct Domain<'w> {
30    router: Router,
31    owners: Vec<Owner>,
32    model: Option<&'w PreparedModelGrids>,
33}
34
35/// One row chunk's placements, routed by owner, its prediction scratch and
36/// its rows' native-channel residuals.
37struct Chunk {
38    rows: Range<usize>,
39    scratch: SampleBuffer,
40    owned: Vec<SampleBuffer>,
41    placed: u64,
42    backend: CpuBackend,
43    prediction: PredictionScratch,
44    predicted: Vec<Complex32>,
45    residual: Vec<Complex32>,
46}

48pub(super) struct Wave<'w, 'p> {
49    pass: &'w MajorCyclePass<'p>,
50    planes: PlaneRange,
51    domains: Vec<Domain<'w>>,
52    native_residuals: bool,
53    check_spacing: bool,
54    chunks: Vec<Chunk>,
55    predictions: Vec<Complex32>,
56    samples: u64,
57}

59impl<'w, 'p> Wave<'w, 'p> {
60    /// A wave over `planes` with each domain's prepared `models`, forming
61    /// residuals at native channels when `native_residuals`.
62    pub(super) fn new(
63        pass: &'w MajorCyclePass<'p>,
64        planes: PlaneRange,
65        models: Option<&'w [PreparedModelGrids]>,
66        native_residuals: bool,
67    ) -> Self {
68        let domains = pass
69            .domains
70            .iter()
71            .enumerate()
72            .map(|(index, domain)| {
73                let router = Router::new(&domain.partition, planes);
74                let owners = (0..router.owners())
75                    .map(|owner| {
76                        let (range, tile) = router.target(&domain.partition, planes, owner);
77                        Owner {
78                            acc: Some(domain.operator.accumulator(range, tile, pass.modes)),
79                            backend: CpuBackend::new(),
80                            images: None,
81                        }
82                    })
83                    .collect();
84                Domain {
85                    router,
86                    owners,
87                    model: models.map(|models| &models[index]),
88                }
89            })
90            .collect();
91        Self {
92            pass,
93            planes,
94            domains,
95            native_residuals,
96            check_spacing: false,
97            chunks: Vec::new(),
98            predictions: Vec::new(),
99            samples: 0,
100        }
101    }

103    /// Check every row's native spacing against the pass's
104    /// `native_spacing_hz`, which sized this wave's model halo.
105    pub(super) fn checking_spacing(mut self, check: bool) -> Self {
106        self.check_spacing = check;
107        self
108    }

110    /// Samples placed on the first domain so far.
111    pub(super) const fn samples(&self) -> u64 {
112        self.samples
113    }

115    /// Place every row of `block` on every domain and accumulate the
116    /// placements; with a visibility sink, first hand it the block, with
117    /// every selected sample's prediction when it asks for them.
118    ///
119    /// When the wave forms residuals at native channels, each row's samples
120    /// are placed from `V − Σ_d A_d·m_d` at its native channels and gridded
121    /// as data; otherwise the domain's model is subtracted at each placed
122    /// sample.
123    pub(super) fn consume(
124        &mut self,
125        block: &NativeBlock,
126        team: &WorkerTeam,
127        visibilities: Option<&mut VisibilitySink<'_>>,
128    ) -> Result<(), PassError> {
129        let npol = block.correlations();
130        let rows = block.len();
131        if self.check_spacing {
132            let bound_hz = self.pass.native_spacing_hz;
133            for row in 0..rows {
134                let observed_hz = block
135                    .row(0, row)
136                    .frequencies_hz
137                    .windows(2)
138                    .fold(0.0_f64, |widest, pair| {
139                        widest.max((pair[1] - pair[0]).abs())
140                    });
141                if observed_hz > bound_hz {
142                    return Err(PassError::NativeSpacing {
143                        observed_hz,
144                        bound_hz,
145                    });
146                }
147            }
148        }
149        let count = (team.workers() * CHUNKS_PER_WORKER).clamp(1, rows.max(1));
150        if self.chunks.len() < count {
151            self.chunks.resize_with(count, || Chunk {
152                rows: 0..0,
153                scratch: SampleBuffer::new(npol),
154                owned: Vec::new(),
155                placed: 0,
156                backend: CpuBackend::new(),
157                prediction: PredictionScratch::default(),
158                predicted: Vec::new(),
159                residual: Vec::new(),
160            });
161        }
162        for (index, chunk) in self.chunks[..count].iter_mut().enumerate() {
163            chunk.rows = index * rows / count..(index + 1) * rows / count;
164        }
165        let sink_predictions = visibilities.as_ref().is_some_and(|sink| sink.predictions);
166        if self.native_residuals || sink_predictions {
167            self.predict(block, team, count)?;
168        } else {
169            self.predictions.clear();
170        }
171        if let Some(sink) = visibilities {
172            let predictions: &[Complex32] = if sink.predictions {
173                &self.predictions
174            } else {
175                &[]
176            };
177            (sink.write)(block, predictions).map_err(PassError::VisibilityWrite)?;
178        }
179        for index in 0..self.domains.len() {
180            let placed = self.place(block, team, count, index)?;
181            if index == 0 {
182                self.samples += placed;
183            }
184        }
185        Ok(())
186    }

188    /// Place the rows of `block` on domain `index` and accumulate them;
189    /// returns the samples placed.
190    fn place(
191        &mut self,
192        block: &NativeBlock,
193        team: &WorkerTeam,
194        count: usize,
195        index: usize,
196    ) -> Result<u64, PassError> {
197        let pass = self.pass;
198        let target = &pass.domains[index];
199        let domain = &mut self.domains[index];
200        let owners = domain.owners.len();
201        let router = &domain.router;
202        let planes = self.planes;
203        let predictions = &self.predictions;
204        let native_residuals = self.native_residuals;
205        let cells = block.channels() * block.correlations();
206        team.for_each_mut(&mut self.chunks[..count], |_, chunk| {
207            chunk.placed = 0;
208            chunk
209                .owned
210                .resize_with(owners, || SampleBuffer::new(block.correlations()));
211            chunk.owned[..owners]
212                .iter_mut()
213                .for_each(SampleBuffer::clear);
214            for row in chunk.rows.clone() {
215                chunk.scratch.clear();
216                let native = block.row(index, row);
217                let native = if native_residuals {
218                    let model = &predictions[row * cells..(row + 1) * cells];
219                    chunk.residual.clear();
220                    chunk.residual.extend(
221                        native
222                            .values
223                            .iter()
224                            .zip(model)
225                            .map(|(value, model)| value - model),
226                    );
227                    NativeRow {
228                        values: &chunk.residual,
229                        ..native
230                    }
231                } else {
232                    native
233                };
234                target.resampler.place(
235                    target.operator,
236                    pass.weighting,
237                    &native,
238                    &mut chunk.scratch,
239                )?;
240                let placed = chunk.scratch.block();
241                for (sample, placement) in placed.placements.iter().enumerate() {
242                    if !planes.contains(placement.plane) {
243                        continue;
244                    }
245                    let owner = router.owner(target.operator, placement);
246                    chunk.owned[owner].push(
247                        *placement,
248                        placed.values_of(sample),
249                        placed.weights_of(sample),
250                    );
251                    chunk.placed += 1;
252                }
253            }
254            Ok::<_, PassError>(())
255        })?;
256        let chunks = &self.chunks[..count];
257        let model = domain.model.filter(|_| !native_residuals);
258        team.for_each_mut(&mut domain.owners, |owner_index, owner| {
259            let acc = owner
260                .acc
261                .as_mut()
262                .expect("owners accumulate until finished");
263            for chunk in chunks {
264                let block = chunk.owned[owner_index].block();
265                if block.is_empty() {
266                    continue;
267                }
268                accumulate(
269                    pass,
270                    target.operator,
271                    &mut owner.backend,
272                    &block,
273                    model,
274                    acc,
275                )?;
276            }
277            Ok::<_, PassError>(())
278        })?;
279        Ok(chunks.iter().map(|chunk| chunk.placed).sum())
280    }

282    /// Model visibilities of every selected sample of `block`, summed over
283    /// every domain's model (`SIMapperCollection::degrid`), `[row][channel]
284    /// [correlation]`, into `self.predictions`; zero without a model.
285    fn predict(
286        &mut self,
287        block: &NativeBlock,
288        team: &WorkerTeam,
289        count: usize,
290    ) -> Result<(), PassError> {
291        let cells = block.channels() * block.correlations();
292        let rows = block.len();
293        self.predictions.clear();
294        self.predictions.resize(rows * cells, Complex32::default());
295        if self.domains.iter().all(|domain| domain.model.is_none()) {
296            return Ok(());
297        }
298        let pass = self.pass;
299        let domains = &self.domains;
300        let mut pieces = Vec::with_capacity(count);
301        let mut remaining = self.predictions.as_mut_slice();
302        for chunk in &mut self.chunks[..count] {
303            let (head, tail) = remaining.split_at_mut(chunk.rows.len() * cells);
304            remaining = tail;
305            pieces.push((chunk, head));
306        }
307        team.for_each_mut(&mut pieces, |_, (chunk, out)| {
308            chunk.predicted.resize(cells, Complex32::default());
309            for (local, row) in chunk.rows.clone().enumerate() {
310                let out = &mut out[local * cells..(local + 1) * cells];
311                for (index, (target, domain)) in pass.domains.iter().zip(domains).enumerate() {
312                    let Some(model) = domain.model else {
313                        continue;
314                    };
315                    target.resampler.predict_row(
316                        target.operator,
317                        &mut chunk.backend,
318                        model,
319                        &block.row(index, row),
320                        &mut chunk.prediction,
321                        &mut chunk.predicted,
322                    )?;
323                    for (sum, value) in out.iter_mut().zip(&chunk.predicted) {
324                        *sum += value;
325                    }
326                }
327            }
328            Ok::<_, PassError>(())
329        })
330    }

332    /// Transform the accumulated grids into this wave's normal images, one
333    /// per domain.
334    pub(super) fn finish(self, team: &WorkerTeam) -> Result<Vec<NormalImages>, PassError> {
335        let pass = self.pass;
336        let planes = self.planes;
337        self.domains
338            .into_iter()
339            .zip(pass.domains)
340            .map(|(mut domain, target)| {
341                let operator = target.operator;
342                match &target.partition {
343                    Partition::Planes { .. } => {
344                        team.for_each_mut(&mut domain.owners, |_, owner| {
345                            let acc = owner.acc.take().expect("each owner finishes once");
346                            owner.images = Some(operator.finish(acc)?);
347                            Ok::<_, PassError>(())
348                        })?;
349                        Ok(concatenate(
350                            domain
351                                .owners
352                                .into_iter()
353                                .map(|owner| owner.images.expect("every owner finished")),
354                        ))
355                    }
356                    Partition::Regions { .. } => {
357                        let mut merged = operator.accumulator(planes, None, pass.modes);
358                        for owner in &domain.owners {
359                            merged
360                                .merge_from(owner.acc.as_ref().expect("owners hold their tiles"))?;
361                        }
362                        drop(domain.owners);
363                        Ok(operator.finish(merged)?)
364                    }
365                }
366            })
367            .collect()
368    }
369}

371/// Grid one owner's share of a block in every mode the pass holds; with a
372/// model the data terms receive the residual `V − A·m`.
373fn accumulate(
374    pass: &MajorCyclePass<'_>,
375    operator: &casa_imaging_operator::MeasurementOperator,
376    backend: &mut CpuBackend,
377    block: &casa_imaging_operator::SampleBlock<'_>,
378    model: Option<&PreparedModelGrids>,
379    acc: &mut GridAccumulator,
380) -> Result<(), PassError> {
381    let cf = operator.cf();
382    if pass.modes.data {
383        let work = match model {
384            Some(model) => Work::ResidualGrid {
385                model,
386                acc,
387                residual_out: None,
388            },
389            None => Work::Grid {
390                mode: Mode::Data,
391                acc,
392            },
393        };
394        backend.apply(block, cf, work)?;
395    }
396    if pass.modes.psf {
397        backend.apply(
398            block,
399            cf,
400            Work::Grid {
401                mode: Mode::Psf,
402                acc,
403            },
404        )?;
405    }
406    if pass.modes.weight {
407        backend.apply(
408            block,
409            cf,
410            Work::Grid {
411                mode: Mode::Weight,
412                acc,
413            },
414        )?;
415    }
416    Ok(())
417}

419/// Join per-owner images of consecutive plane ranges.
420pub(super) fn concatenate(parts: impl IntoIterator<Item = NormalImages>) -> NormalImages {
421    let mut parts = parts.into_iter();
422    let mut joined = parts.next().expect("a wave has at least one owner");
423    for part in parts {
424        assert!(
425            part.pols == joined.pols
426                && part.data_terms == joined.data_terms
427                && part.psf_terms == joined.psf_terms
428                && part.first_plane == joined.first_plane + joined.planes.len() as u32,
429            "owner images must continue the plane sequence with the same layout"
430        );
431        joined.planes.extend(part.planes);
432    }
433    joined
