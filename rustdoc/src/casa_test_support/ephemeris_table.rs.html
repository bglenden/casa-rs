2//! CASA ephemeris tables for tests: the layout casacore's `MeasComet`
3//! reads (an `IERS` table of regularly spaced samples with its `MJD0`,
4//! `dMJD` and `VS_*` keywords) and XRADIO's converter reads (`GeoLong`,
5//! `GeoLat`, `GeoDist`, `obsloc`, `posrefsys`), both as a MeasurementSet
6//! field's attached ephemeris (`FIELD/EPHEM<id>_<name>.tab`) and as a
7//! `phasecenter` table.
8
9use std::path::{Path, PathBuf};
10
11use casa_tables::{ColumnSchema, Table, TableInfo, TableOptions, TableSchema};
12use casa_types::{PrimitiveType, RecordField, RecordValue, ScalarValue, Value};
13
14/// One sample: MJD (days), RA and DEC (degrees, ICRS), geocentric
15/// distance `Rho` (AU) and radial velocity `RadVel` (AU a day).
16pub type EphemerisRow = [f64; 5];
17
18/// Write a geocentric ICRS ephemeris of body `name` at `path`.
19///
20/// # Panics
21///
22/// If the table cannot be built or saved, or `rows` are fewer than two or
23/// not equally spaced in time, as `MeasComet` requires.
24pub fn write_ephemeris_table(path: &Path, name: &str, rows: &[EphemerisRow]) {
25    let step = rows[1][0] - rows[0][0];
26    assert!(
27        rows.windows(2)
28            .all(|pair| ((pair[1][0] - pair[0][0]) - step).abs() <= 1.0e-9),
29        "MeasComet reads equally spaced samples"
30    );
31    let schema = TableSchema::new(
32        ["MJD", "RA", "DEC", "Rho", "RadVel"]
33            .into_iter()
34            .map(|column| ColumnSchema::scalar(column, PrimitiveType::Float64))
35            .collect(),
36    )
37    .expect("an ephemeris schema");
38    let mut table = Table::with_schema(schema);
39    for row in rows {
40        table
41            .add_row(RecordValue::new(
42                ["MJD", "RA", "DEC", "Rho", "RadVel"]
43                    .into_iter()
44                    .zip(row)
45                    .map(|(column, value)| {
46                        RecordField::new(column, Value::Scalar(ScalarValue::Float64(*value)))
47                    })
48                    .collect(),
49            ))
50            .expect("an ephemeris row");
51    }
52    for (keyword, value) in [
53        ("VS_CREATE", "2026/10/29/00:00"),
54        ("VS_DATE", "2026/10/29/00:00"),
55        ("VS_VERSION", "0001.0001"),
56        ("VS_TYPE", "Table of comet/planetary positions"),
57        ("NAME", name),
58        ("posrefsys", "ICRF/ICRS"),
59        ("obsloc", "GEOCENTRIC"),
60    ] {
61        table.keywords_mut().push(RecordField::new(
62            keyword,
63            Value::Scalar(ScalarValue::String(value.to_string())),
64        ));
65    }
66    // MeasComet indexes rows from MJD0, one step before the first sample.
67    for (keyword, value) in [
68        ("MJD0", rows[0][0] - step),
69        ("dMJD", step),
70        ("GeoLong", 0.0),
71        ("GeoLat", 0.0),
72        ("GeoDist", 0.0),
73    ] {
74        table.keywords_mut().push(RecordField::new(
75            keyword,
76            Value::Scalar(ScalarValue::Float64(value)),
77        ));
78    }
79    table.set_info(TableInfo {
80        table_type: "IERS".to_string(),
81        sub_type: String::new(),
82        readme: Vec::new(),
83    });
84    table
85        .save(TableOptions::new(path))
86        .expect("save the ephemeris table");
87}

89/// Write body `name`'s ephemeris as the table the fields of the
90/// MeasurementSet at `measurement_set` whose `EPHEMERIS_ID` is
91/// `ephemeris_id` follow, and return its path. The caller sets the
92/// fields' `EPHEMERIS_ID`.
93///
94/// # Panics
95///
96/// If the table cannot be built or saved.
97pub fn write_field_ephemeris(
98    measurement_set: &Path,
99    ephemeris_id: i32,
100    name: &str,
101    rows: &[EphemerisRow],
102) -> PathBuf {
103    let path = measurement_set
104        .join("FIELD")
105        .join(format!("EPHEM{ephemeris_id}_{name}.tab"));
106    write_ephemeris_table(&path, name, rows);
107    path
108}
