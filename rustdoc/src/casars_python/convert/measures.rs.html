2//! Measures from CASA arguments: casatools' `casaMEpoch`, `casaMDirection`,
3//! `casaMPosition` and `casaMFrequency` ([`frequency`]), and the reference
4//! codes they accept.
5//!
6//! Each function returns `Ok(None)` where CASA's returns `false` (the
7//! calling method then logs its own SEVERE message and continues with a
8//! default measure) and an error where CASA raises.
9//!
10//! - **Strings** are split at commas, or else at blanks, into a frame and
11//!   values: `"J2000 12h00m00 34d00m00"`, `"UTC 2020/01/01/00:00:00"`,
12//!   `"ITRF 6378137m 10deg 20deg"` (a position string is a length, then a
13//!   longitude and a latitude). A frame code may be any unique prefix
14//!   (`j20`); an unknown one falls back to the default frame, as in CASA.
15//!   A single token is a frame (its pole, for a direction), a catalogue
16//!   source or an observatory.
17//! - **Records** are `MeasureHolder` records, read by
18//!   [`Measure::from_record`](super::holder::Measure::from_record).
19//!
20//! [`direction_measure`] keeps a direction in casacore's form, with any
21//! reference code (`me.source('SUN')`); [`direction`] converts it to
22//! casa-types' form, which `sm` needs.
23//!
24//! Divergence: where a string's value after the first does not parse, CASA
25//! silently reuses the previous value (`"J2000 12h00m00 garbage"` is a
26//! declination of 180 degrees); this raises instead.
27

46/// CASA's `casaMEpoch`: a record, `"frame value"`, `"value"` (UTC), or a
47/// number of MJD days (UTC, so `[]` is MJD 0). `""` is CASA's `false`.
48pub(crate) fn epoch(value: &Variant) -> ConvertResult<Option<MEpoch>> {
49    match value {
50        Variant::Record(record) => epoch_record(record).map(Some),
51        Variant::Text(_) | Variant::Texts(_) => {
52            let tokens = tokens(value);
53            match tokens.as_slice() {
54                [only] if only.is_empty() => Ok(None),
55                [only] => epoch_of(&read_value(only)?, EpochRef::UTC).map(Some),
56                [frame, text] => {
57                    let refer = EPOCH_CODES
58                        .resolve(frame)
59                        .map_or(Ok(EpochRef::UTC), epoch_reference)?;
60                    epoch_of(&read_value(text)?, refer).map(Some)
61                }
62                _ => Err(ConvertError::runtime(format!(
63                    "Could not understand Epoch parameter {}",
64                    value.describe()
65                ))),
66            }
67        }
68        Variant::Empty => Ok(Some(MEpoch::from_mjd(0.0, EpochRef::UTC))),
69        Variant::Number(days) => Ok(Some(MEpoch::from_mjd(*days, EpochRef::UTC))),
70        Variant::Bool(_)
71        | Variant::Numbers(_)
72        | Variant::Complex(_)
73        | Variant::Complexes(_)
74        | Variant::Array { .. }
75        | Variant::Other(_) => Err(ConvertError::Type(format!(
76            "cannot convert {} to an epoch",
77            value.describe()
78        ))),
79    }
80}

82/// CASA's `casaMDirection` in casa-types' form, failing for a reference code
83/// casa-types does not convert; see [`direction_measure`].
84pub(crate) fn direction(
85    value: &Variant,
86    measures: &Arc<dyn MeasuresProvider>,
87) -> ConvertResult<Option<MDirection>> {
88    direction_measure(value, measures)?
89        .map(
90            |measure| match measure.to_casa().map_err(ConvertError::runtime)? {
91                CasaMeasure::Direction(direction) => Ok(direction),
92                _ => unreachable!("a direction measure converts to a direction"),
93            },
94        )
95        .transpose()
96}

98/// CASA's `casaMDirection`: a record, `"frame lon lat"`, `"lon lat"`
99/// (J2000), or one name (a frame's pole, or a catalogue source). Other
100/// values are CASA's `false`.
101pub(crate) fn direction_measure(
102    value: &Variant,
103    measures: &Arc<dyn MeasuresProvider>,
104) -> ConvertResult<Option<Measure>> {
105    match value {
106        Variant::Record(record) => direction_record(record).map(Some),
107        Variant::Text(_) | Variant::Texts(_) => match tokens(value).as_slice() {
108            [frame, longitude, latitude] => {
109                let refer = DIRECTION_CODES.resolve_or_default(frame);
110                direction_of(longitude, latitude, refer).map(Some)
111            }
112            [longitude, latitude] => direction_of(longitude, latitude, "J2000").map(Some),
113            [name] => named_direction(name, measures),
114            _ => Ok(None),
115        },
116        _ => Ok(None),
117    }
118}

120/// CASA's `casaMPosition`: a record, `"frame length lon lat"`,
121/// `"length lon lat"` (ITRF), or an observatory name. Other values are CASA's
122/// `false`.
123pub(crate) fn position(
124    value: &Variant,
125    measures: &Arc<dyn MeasuresProvider>,
126) -> ConvertResult<Option<MPosition>> {
127    match value {
128        Variant::Record(record) => position_record(record).map(Some),
129        Variant::Text(_) | Variant::Texts(_) => match tokens(value).as_slice() {
130            [frame, length, longitude, latitude] => {
131                let wgs84 = POSITION_CODES.resolve(frame) == Some("WGS84");
132                position_of(length, longitude, latitude, wgs84).map(Some)
133            }
134            [length, longitude, latitude] => {
135                position_of(length, longitude, latitude, false).map(Some)
136            }
137            [name] => MPosition::from_observatory_name(&name.to_ascii_uppercase(), &**measures)
138                .map_err(|error| ConvertError::runtime(error.to_string()))?
139                .map(Some)
140                .ok_or_else(|| ConvertError::runtime("Could not understand Position parameter")),
141            _ => Ok(None),
142        },
143        _ => Ok(None),
144    }
145}

147/// `MFrequency::getType` on an upper-cased `refcode`, as `sm.setspwindow`
148/// reads it; an empty code is TOPO.
149pub(crate) fn frequency_reference(code: &str) -> ConvertResult<FrequencyRef> {
150    if code.is_empty() {
151        return Ok(FrequencyRef::TOPO);
152    }
153    let upper = code.to_ascii_uppercase();
154    let invalid = || ConvertError::runtime(format!("Invalid frequency reference '{upper}'"));
155    let name = FREQUENCY_CODES.resolve(&upper).ok_or_else(invalid)?;
156    FrequencyRef::from_str(name).map_err(|_| {
157        ConvertError::runtime(format!(
158            "the {name} frequency frame is not yet implemented in casa-rs"
159        ))
160    })
161}

163/// CASA's `casaMFrequency`: a frequency record, or a quantity record
164/// (LSRK); `"frame value"`; `"value"` (LSRK); or a number of Hz (LSRK). An
165/// unknown frame is LSRK, as CASA defaults it (CASA prints a warning to
166/// stderr; this one is returned), and the two tokens are read again as one,
167/// so that a spaced quantity (`"1.4 GHz"`) keeps its value; when they do
168/// not read as one, the second token's value stands (`"foo 1.4GHz"`). `""`
169/// is CASA's `false`.
170pub(crate) fn frequency(value: &Variant) -> ConvertResult<(Option<MFrequency>, Option<String>)> {
171    match value {
172        Variant::Record(record) => {
173            frequency_record(record).map(|frequency| (Some(frequency), None))
174        }
175        Variant::Text(_) | Variant::Texts(_) => match tokens(value).as_slice() {
176            [frame, text] => {
177                let quantity = read_value(text)?;
178                let (refer, quantity, warning) = match FREQUENCY_CODES.resolve(frame) {
179                    Some(code) => (frequency_frame(code)?, quantity, None),
180                    None => (
181                        FrequencyRef::LSRK,
182                        Quantity::read(&format!("{frame}{text}")).unwrap_or(quantity),
183                        Some(
184                            "Could not understand frequency frame..defaulting to LSRK".to_string(),
185                        ),
186                    ),
187                };
188                Ok((Some(frequency_of(&quantity, refer)?), warning))
189            }
190            [only] if only.is_empty() => Ok((None, None)),
191            [only] => Ok((
192                Some(frequency_of(&read_value(only)?, FrequencyRef::LSRK)?),
193                None,
194            )),
195            _ => Ok((None, None)),
196        },
197        Variant::Number(hz) => Ok((Some(MFrequency::new(*hz, FrequencyRef::LSRK)), None)),
198        Variant::Empty => Ok((Some(MFrequency::new(0.0, FrequencyRef::LSRK)), None)),
199        Variant::Bool(_)
200        | Variant::Numbers(_)
201        | Variant::Complex(_)
202        | Variant::Complexes(_)
203        | Variant::Array { .. }
204        | Variant::Other(_) => Err(ConvertError::Type(format!(
205            "cannot convert {} to a frequency",
206            value.describe()
207        ))),
208    }
209}

211/// A frequency record, or a quantity record taken as LSRK.
212fn frequency_record(record: &BTreeMap<String, Variant>) -> ConvertResult<MFrequency> {
213    let wrap = |error: String| {
214        ConvertError::runtime(format!("Error {error}In converting Frequency parameter"))
215    };
216    match measure_of_kind(record, MeasureKind::Frequency, wrap)? {
217        Some(measure) => match measure.to_casa().map_err(ConvertError::runtime)? {
218            CasaMeasure::Frequency(frequency) => Ok(frequency),
219            _ => unreachable!("a frequency measure converts to a frequency"),
220        },
221        None => {
222            let quantity = quantity_record(record).map_err(wrap)?;
223            frequency_of(&quantity, FrequencyRef::LSRK)
224        }
225    }
226}

228/// `MFrequency(quantity, refer)`: a frequency, period, wavelength, energy
229/// or momentum (`MVFrequency::makeF`).
230fn frequency_of(quantity: &Quantity, refer: FrequencyRef) -> ConvertResult<MFrequency> {
231    let hz = frequency_hz(quantity).ok_or_else(|| {
232        ConvertError::runtime(format!(
233            "Illegal unit type '{}' for a frequency",
234            quantity.unit().name()
235        ))
236    })?;
237    Ok(MFrequency::new(hz, refer))
238}

240/// A frequency frame casa-types converts.

242    FrequencyRef::from_str(code).map_err(|_| {
243        ConvertError::runtime(format!(
244            "the {code} frequency frame is not yet implemented in casa-rs"
245        ))
246    })

253/// casatools' `sepCommaEmptyToVectorStrings` for a `str`, or the strings of
254/// a string list.
255fn tokens(value: &Variant) -> Vec<String> {
256    match value {
257        Variant::Text(text) => {
258            let separator = if text.contains(',') { ',' } else { ' ' };
259            if !text.contains(separator) {
260                return vec![text.clone()];
261            }
262            text.split(separator)
263                .filter(|token| !token.is_empty() && *token != " ")
264                .map(str::to_string)
265                .collect()
266        }
267        Variant::Texts(texts) => texts.clone(),
268        _ => Vec::new(),
269    }
270}

272/// One value of a measure string, raising casacore's message when it does not
273/// read.
274fn read_value(text: &str) -> ConvertResult<Quantity> {
275    Quantity::read(text)
276        .map_err(|_| ConvertError::runtime(format!("{EMPTY_HOLDER} (could not read {text:?})")))
277}

279/// `MEpoch(quantity, refer)`, which requires a time.
280fn epoch_of(quantity: &Quantity, refer: EpochRef) -> ConvertResult<MEpoch> {
281    require(quantity, Dimension::Time, "assure non-conforming unit type")?;
282    Ok(MEpoch::from_mjd(value_in(quantity, "d")?, refer))
283}

285/// `MDirection(lon, lat, refer)`: casacore's `cos` and `sin` require angles
286/// (the latitude is checked first), so `"J2000 10h 20d"` raises.
287fn direction_of(longitude: &str, latitude: &str, refer: &'static str) -> ConvertResult<Measure> {
288    let longitude = read_value(longitude)?;
289    let latitude = read_value(latitude).map_err(|_| {
290        ConvertError::runtime(format!(
291            "Could not read the direction latitude {latitude:?}"
292        ))
293    })?;
294    let latitude = angle_rad(&latitude)?;
295    let longitude = angle_rad(&longitude)?;
296    Ok(Measure {
297        refer,
298        value: MeasureValue::Direction(direction_cosines(longitude, latitude)),
299    })
300}

302/// A name alone: a frame's pole, or a catalogue source; otherwise CASA's
303/// `false`.
304fn named_direction(
305    name: &str,
306    measures: &Arc<dyn MeasuresProvider>,
307) -> ConvertResult<Option<Measure>> {
308    if let Some(refer) = DIRECTION_CODES.resolve(name) {
309        return Ok(Some(Measure {
310            refer,
311            value: MeasureValue::Direction([0.0, 0.0, 1.0]),
312        }));
313    }
314    let source = measures
315        .source(name)
316        .map_err(|error| ConvertError::runtime(format!("source catalogue: {error}")))?;
317    Ok(source.map(|source| Measure {
318        refer: DIRECTION_CODES.resolve_or_default(&source.reference),
319        value: MeasureValue::Direction(direction_cosines(
320            source.longitude_rad,
321            source.latitude_rad,
322        )),
323    }))
324}

326/// `MPosition(length, lon, lat, refer)`; casacore checks the latitude, then
327/// the longitude, then the length.
328fn position_of(
329    length: &str,
330    longitude: &str,
331    latitude: &str,
332    wgs84: bool,
333) -> ConvertResult<MPosition> {
334    let length = read_value(length)?;
335    let longitude = read_value(longitude).map_err(|_| {
336        ConvertError::runtime(format!(
337            "Could not read the position longitude {longitude:?}"
338        ))
339    })?;
340    let latitude = read_value(latitude).map_err(|_| {
341        ConvertError::runtime(format!("Could not read the position latitude {latitude:?}"))
342    })?;
343    let latitude = angle_rad(&latitude)?;
344    let longitude = angle_rad(&longitude)?;
345    require(
346        &length,
347        Dimension::Length,
348        "assure non-conforming unit type",
349    )?;
350    Ok(spherical_position(
351        longitude,
352        latitude,
353        length.get_si_value(),
354        wgs84,
355    ))
356}

358/// A position from spherical coordinates: ITRF with a geocentric radius, or
359/// WGS84 with a height.
360fn spherical_position(longitude: f64, latitude: f64, length_m: f64, wgs84: bool) -> MPosition {
361    if wgs84 {
362        MPosition::new_wgs84(longitude, latitude, length_m)
363    } else {
364        let (sin_lon, cos_lon) = longitude.sin_cos();
365        let (sin_lat, cos_lat) = latitude.sin_cos();
366        MPosition::new_itrf(
367            length_m * cos_lat * cos_lon,
368            length_m * cos_lat * sin_lon,
369            length_m * sin_lat,
370        )
371    }
372}

374/// An angle in radians, raising casacore's `cos` message otherwise.
375fn angle_rad(quantity: &Quantity) -> ConvertResult<f64> {
376    require(quantity, Dimension::Angle, "cos illegal unit type")?;
377    Ok(quantity.get_si_value())
378}

380/// Raises `Quantum::<what> '<unit>'` unless `quantity` has dimension
381/// `wanted`.
382fn require(quantity: &Quantity, wanted: Dimension, what: &str) -> ConvertResult<()> {
383    if quantity.unit().val().dim == UnitDim::basis(wanted) {
384        Ok(())
385    } else {
386        Err(ConvertError::runtime(format!(
387            "Quantum::{what} '{}'",
388            quantity.unit().name()
389        )))
390    }
391}

393/// A measure record of `kind` (`MeasureHolder::fromRecord` and `as<Kind>`):
394/// `Ok(None)` for a record that is not a measure record; errors are wrapped
395/// by `wrap` as CASA wraps them.
396fn measure_of_kind(
397    record: &BTreeMap<String, Variant>,
398    kind: MeasureKind,
399    wrap: impl Fn(String) -> ConvertError,
400) -> ConvertResult<Option<Measure>> {
401    match Measure::from_record(record) {
402        Ok(read) if read.measure.kind() == kind => Ok(Some(read.measure)),
403        Ok(_) => Err(ConvertError::runtime(format!(
404            "Empty or wrong MeasureHolder for asM{}",
405            kind.tell_me()
406        ))),
407        Err(HolderError::NotAMeasure) => Ok(None),
408        Err(error) => Err(wrap(error.message())),
409    }
410}

412/// An epoch record, or a quantity record taken as UTC.
413fn epoch_record(record: &BTreeMap<String, Variant>) -> ConvertResult<MEpoch> {
414    let wrap = |error: String| {
415        ConvertError::runtime(format!("Error {error}In converting Epoch parameter"))
416    };
417    match measure_of_kind(record, MeasureKind::Epoch, wrap)? {
418        Some(measure) => match measure.to_casa().map_err(ConvertError::runtime)? {
419            CasaMeasure::Epoch(epoch) => Ok(epoch),
420            _ => unreachable!("an epoch measure converts to an epoch"),
421        },
422        None => {
423            let quantity = quantity_record(record).map_err(wrap)?;
424            epoch_of(&quantity, EpochRef::UTC)
425        }
426    }
427}

429/// A direction record: two angles, or three direction cosines.
430fn direction_record(record: &BTreeMap<String, Variant>) -> ConvertResult<Measure> {
431    let wrap = |error: String| {
432        ConvertError::runtime(format!("Error {error} in converting Direction parameter"))
433    };
434    measure_of_kind(record, MeasureKind::Direction, wrap)?
435        .ok_or_else(|| wrap(HolderError::NotAMeasure.message()))

438/// A position record (`MVPosition::putValue`).
439fn position_record(record: &BTreeMap<String, Variant>) -> ConvertResult<MPosition> {
440    let wrap = |error: String| {
441        ConvertError::runtime(format!("Error {error}In converting Position parameter"))
442    };
443    let measure = measure_of_kind(record, MeasureKind::Position, wrap)?
444        .ok_or_else(|| wrap(HolderError::NotAMeasure.message()))?;
445    match measure.to_casa().map_err(ConvertError::runtime)? {
446        CasaMeasure::Position(position) => Ok(position),
447        _ => unreachable!("a position measure converts to a position"),
448    }
449}
