3//! Thin PyO3 `ia.torecord` / `ia.fromrecord` wrapper delegating to [`casa_imageanalysis::tool::serialization`].
4
5use std::path::Path;
6
7use casa_imageanalysis::tool::serialization::{image_from_record_with_history, image_to_record};
8use pyo3::prelude::*;
9use pyo3::types::PyDict;
10
11use crate::tools::table::{casa_record, values};
12
13use super::PyImageTool;
14use super::handle::{external_open_check, raised};
15
16#[pymethods]
17impl PyImageTool {
18    fn torecord(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
19        py.detach(|| -> PyResult<crate::convert::value::CasaValue> {
20            let state = self.state();
21            let Some(attachment) = state.image.as_ref() else {
22                return Ok(crate::convert::value::CasaValue::empty_record());
23            };
24            let record = image_to_record(attachment).map_err(raised)?;
25            Ok(values::record(&record, Path::new(".")))
26        })?
27        .into_python(py)
28    }
29
30    #[pyo3(signature = (record, outfile=""), text_signature = "($self, record, outfile='')")]
31    fn fromrecord(
32        &self,
33        py: Python<'_>,
34        record: &Bound<'_, PyDict>,
35        outfile: &str,
36    ) -> PyResult<bool> {
37        let record = casa_record(py, Some(record))?;
38        py.detach(|| {
39            let mut state = self.state();
40            state.image = None;
41            let attached = image_from_record_with_history(
42                &record,
43                outfile,
44                true,
45                state.history,
46                external_open_check(),
47            )
48            .map_err(raised)?;
49            state.image = Some(attached);
50            Ok(true)
51        })
52    }
53}
