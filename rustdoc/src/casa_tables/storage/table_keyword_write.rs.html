14pub(crate) fn save_keywords(
15    table: &Path,
16    row_count: u64,
17    keywords: &RecordValue,
18    sync: fn(&File) -> io::Result<()>,
19) -> Result<(), StorageError> {
20    let path = table.join(TABLE_CONTROL_FILE);
21    let metadata = path.symlink_metadata()?;
22    if !metadata.is_file() {
23        return Err(StorageError::FormatMismatch(
24            "atomic keyword save needs a regular table.dat".into(),
25        ));
26    }
27    check_write_access(&path)?;
28    let TableDatResult::Plain(mut contents) = read_table_dat_dispatch(&path)? else {
29        return Err(StorageError::FormatMismatch(
30            "atomic keyword save is only supported for PlainTable storage".into(),
31        ));
32    };
33    contents.raise_row_count(row_count);
34    contents.table_desc.table_keywords = keywords.clone();
35    let bytes = serialize_table_dat(&stored_table_dat_keywords(&contents, table))?;
36    let mut staged = tempfile::NamedTempFile::new_in(table)?;
37    staged.write_all(&bytes)?;
38    staged.as_file().set_permissions(metadata.permissions())?;
39    sync(staged.as_file())?;
40    staged.persist(&path).map_err(|error| error.error)?;
41    Ok(())
42}

44pub(crate) fn sync_cache_keywords(file: &File) -> io::Result<()> {
45    #[cfg(target_vendor = "apple")]
46    {
47        use std::os::fd::AsRawFd;
48
49        // File::sync_all requests F_FULLFSYNC on Apple. The sorted cache is
50        // derived from published rows and uses casacore's ordinary fsync.
51        loop {
52            // SAFETY: the borrowed file keeps the descriptor alive.
53            if unsafe { libc::fsync(file.as_raw_fd()) } == 0 {
54                return Ok(());
55            }
56            let error = io::Error::last_os_error();
57            if error.kind() != io::ErrorKind::Interrupted {
58                return Err(error);
59            }
60        }
61    }
62    #[cfg(not(target_vendor = "apple"))]
63    file.sync_all()
64}
