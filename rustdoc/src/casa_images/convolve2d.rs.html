2//! Convolving an image's direction planes with a Gaussian, as the image
3//! tool's `convolve2d` does (CASA's `Image2DConvolver`).
4//!
5//! `simutil.convimage` convolves a Jy/pixel sky model with the clean beam to
6//! compare it with the clean image. For an image in Jy/pixel, or any flux
7//! per pixel, CASA's rules (`Image2DConvolver::_dealWithRestoringBeam`) are:
8//! - the kernel is the Gaussian at unit peak, unscaled, so a point of 1
9//!   Jy/pixel becomes a peak of 1 Jy/beam;
10//! - the output is in Jy/beam, with the Gaussian as its restoring beam.
11//!
12//! The kernel is sampled on a square of an odd number of pixels: ±5σ of
13//! the wider axis, rounded as `Image2DConvolver::_sizeOfGaussian` rounds
14//! it. It is centred on the middle pixel, with the beam's position angle
15//! (north through east) turned into the pixel frame at the reference pixel.
16//! The convolution is linear (casacore's `LatticeConvolver` with
17//! `ConvEnums::LINEAR`): the plane is padded with zeros, so flux does not
18//! wrap around the edges, and the output has the input's shape. Masked
19//! pixels count as zero, and the output keeps the input's mask.
20//!
21//! As in CASA, the pixels must be square. Images in other units (Jy/beam,
22//! whose beam CASA convolves numerically, or none, whose kernel it scales to
23//! unit volume) are refused: the simulation tasks convolve Jy/pixel images
24//! only.
25

37/// An image convolved with a Gaussian.
38#[derive(Clone, Debug)]
39pub struct Convolved {
40    /// The pixels, in the input's shape.
41    pub data: ArrayD<f32>,
42    /// The input's mask, when it has one.
43    pub mask: Option<ArrayD<bool>>,
44    /// The output's brightness unit, `Jy/beam`.
45    pub units: String,
46    /// The output's restoring beam: the Gaussian, its position angle in
47    /// [−90°, 90°] as casacore's `GaussianBeam` keeps it.
48    pub beam: GaussianBeam,
49}

51/// The direction coordinate and its first pixel axis.
52fn direction_axes(
53    coordinates: &CoordinateSystem,
54) -> Result<(usize, &DirectionCoordinate), ImageError> {
55    let index = coordinates
56        .find_coordinate(CoordinateType::Direction)
57        .ok_or_else(|| {
58            ImageError::InvalidMetadata("convolve2d needs a direction coordinate".into())
59        })?;
60    let first = (0..index)
61        .map(|k| coordinates.coordinate(k).n_pixel_axes())
62        .sum();
63    let CoordinateModel::Direction(direction) = coordinates.coordinate(index) else {
64        unreachable!("find_coordinate found a direction coordinate");
65    };
66    if direction.n_pixel_axes() != 2 {
67        return Err(ImageError::InvalidMetadata(
68            "convolve2d needs both direction axes".into(),
69        ));
70    }
71    Ok((first, direction))
72}

74fn require_square_pixels(direction: &DirectionCoordinate) -> Result<(), ImageError> {
75    let increment = direction.increment();
76    let (x, y) = (increment[0].abs(), increment[1].abs());
77    if (x - y).abs() > 1e-13 * x.max(y) {
78        return Err(ImageError::InvalidMetadata(
79            "Pixels must be square, please regrid your image so that they are".into(),
80        ));
81    }
82    Ok(())
83}

85/// The sky offsets (east, north), rad, of a pixel step along each axis at
86/// the reference pixel: `[[east per x, east per y], [north per x, north per
87/// y]]`, by central differences of the coordinate's own conversion. It
88/// carries the increments, the PC matrix and the projection, as casacore's
89/// `GaussianShape::toPixel` does when `worldWidthsToPixel` turns a beam into
90/// pixels.
91fn sky_per_pixel(direction: &DirectionCoordinate) -> Result<[[f64; 2]; 2], ImageError> {
92    const STEP: f64 = 0.5;
93    let crpix = direction.reference_pixel();
94    let world = |dx: f64, dy: f64| direction.to_world(&[crpix[0] + dx, crpix[1] + dy]);
95    let centre = world(0.0, 0.0)?;
96    let mut jacobian = [[0.0; 2]; 2];
97    for (axis, (dx, dy)) in [(STEP, 0.0), (0.0, STEP)].into_iter().enumerate() {
98        let (plus, minus) = (world(dx, dy)?, world(-dx, -dy)?);
99        let dlon = (plus[0] - minus[0] + PI).rem_euclid(2.0 * PI) - PI;
100        jacobian[0][axis] = dlon * centre[1].cos() / (2.0 * STEP);
101        jacobian[1][axis] = (plus[1] - minus[1]) / (2.0 * STEP);
102    }
103    Ok(jacobian)
104}

106/// `Image2DConvolver::_sizeOfGaussian`: twice (5σ rounded, plus one).
107fn size_of_gaussian(fwhm_pixels: f64) -> usize {
108    let sigma = fwhm_pixels / (8.0 * LN_2).sqrt();
109    ((5.0 * sigma + 0.5) as usize + 1) * 2
110}

112/// The Gaussian kernel for `beam` on pixels whose sky offsets are `sky`
113/// ([`sky_per_pixel`]): its side (odd) and values, first axis fastest, peak
114/// 1 at the centre. The major axis points north through east, (east,
115/// north) = (sin θ, cos θ). The side is sized from the beam's widths in
116/// pixels, the axes of the pixel-frame ellipse.
117fn kernel(beam: &GaussianBeam, sky: [[f64; 2]; 2]) -> (usize, Vec<f64>) {
118    let (sin_pa, cos_pa) = beam.position_angle.sin_cos();
119    // The exponent's quadratic form on pixel offsets, q(d) = dᵀ Q d, with the
120    // value exp(−4 ln 2 q): half maximum at q = 1/4.
121    let along = [
122        sky[0][0] * sin_pa + sky[1][0] * cos_pa,
123        sky[0][1] * sin_pa + sky[1][1] * cos_pa,
124    ];
125    let across = [
126        sky[0][0] * cos_pa - sky[1][0] * sin_pa,
127        sky[0][1] * cos_pa - sky[1][1] * sin_pa,
128    ];
129    let (a2, b2) = (beam.major.powi(2), beam.minor.powi(2));
130    let q = [
131        [
132            along[0] * along[0] / a2 + across[0] * across[0] / b2,
133            along[0] * along[1] / a2 + across[0] * across[1] / b2,
134        ],
135        [
136            along[1] * along[0] / a2 + across[1] * across[0] / b2,
137            along[1] * along[1] / a2 + across[1] * across[1] / b2,
138        ],
139    ];
140    let (trace, det) = (q[0][0] + q[1][1], q[0][0] * q[1][1] - q[0][1] * q[1][0]);
141    let root = (trace * trace / 4.0 - det).max(0.0).sqrt();
142    let (small, large) = (trace / 2.0 - root, trace / 2.0 + root);
143    let (major_pixels, minor_pixels) = (1.0 / small.sqrt(), 1.0 / large.sqrt());
144    let mut side = size_of_gaussian(major_pixels).max(size_of_gaussian(minor_pixels));
145    if side.is_multiple_of(2) {
146        side += 1;
147    }
148    let centre = (side - 1) as f64 / 2.0;
149    let mut values = Vec::with_capacity(side * side);
150    for j in 0..side {
151        for i in 0..side {
152            let (dx, dy) = (i as f64 - centre, j as f64 - centre);
153            let form = q[0][0] * dx * dx + 2.0 * q[0][1] * dx * dy + q[1][1] * dy * dy;
154            values.push((-4.0 * LN_2 * form).exp());
155        }
156    }
157    (side, values)
158}

160/// The smallest length at least `n` with no prime factor above 7, which
161/// FFTW transforms fastest.
162fn fast_length(n: usize) -> usize {
163    (n..)
164        .find(|&m| {
165            let mut rest = m;
166            for p in [2, 3, 5, 7] {
167                while rest % p == 0 {
168                    rest /= p;
169                }
170            }
171            rest == 1
172        })
173        .expect("a length with small prime factors exists")
174}

176/// The linear convolution of one plane (first axis fastest) with the
177/// kernel, cut to the plane's shape about the kernel's centre.
178struct PlaneConvolver {
179    plane: [usize; 2],
180    padded: [usize; 2],
181    kernel_side: usize,
182    kernel_transform: Vec<Complex64>,
183    fft: Fft2<f64>,
184}
185
186impl PlaneConvolver {
187    fn new(plane: [usize; 2], kernel_side: usize, kernel: &[f64]) -> Result<Self, ImageError> {
188        let padded = [
189            fast_length(plane[0] + kernel_side - 1),
190            fast_length(plane[1] + kernel_side - 1),
191        ];
192        // Row-major [rows = second axis][columns = first axis].
193        let mut fft = Fft2::<f64>::with_threads([padded[1], padded[0]], 1)
194            .map_err(|error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}")))?
195            .with_estimated_plan();
196        let mut kernel_transform = vec![Complex64::new(0.0, 0.0); padded[0] * padded[1]];
197        for j in 0..kernel_side {
198            for i in 0..kernel_side {
199                kernel_transform[i + j * padded[0]] =
200                    Complex64::new(kernel[i + j * kernel_side], 0.0);
201            }
202        }
203        fft.transform(&mut kernel_transform, false)
204            .map_err(|error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}")))?;
205        Ok(Self {
206            plane,
207            padded,
208            kernel_side,
209            kernel_transform,
210            fft,
211        })
212    }
213
214    fn convolve(&mut self, values: &[f64]) -> Result<Vec<f64>, ImageError> {
215        let [nx, ny] = self.plane;
216        let [px, py] = self.padded;
217        let mut grid = vec![Complex64::new(0.0, 0.0); px * py];
218        for j in 0..ny {
219            for i in 0..nx {
220                grid[i + j * px] = Complex64::new(values[i + j * nx], 0.0);
221            }
222        }
223        let fft_error = |error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}"));
224        self.fft.transform(&mut grid, false).map_err(fft_error)?;
225        for (value, kernel) in grid.iter_mut().zip(&self.kernel_transform) {
226            *value *= kernel;
227        }
228        self.fft.transform(&mut grid, true).map_err(fft_error)?;
229        let norm = (px * py) as f64;
230        let centre = (self.kernel_side - 1) / 2;
231        let mut out = vec![0.0; nx * ny];
232        for j in 0..ny {
233            for i in 0..nx {
234                out[i + j * nx] = grid[(i + centre) + (j + centre) * px].re / norm;
235            }
236        }
237        Ok(out)
238    }
239}

241/// `ia.convolve2d(major, minor, pa)` with a Gaussian on the direction plane
242/// of `input`, an image in a flux per pixel.
243///
244/// # Errors
245///
246/// An image without a direction coordinate, with pixels that are not square,
247/// or in a unit that is not a flux per pixel; a failure to read the image.
248pub fn convolve2d(
249    input: &impl ImageInterface<f32>,
250    beam: &GaussianBeam,
251) -> Result<Convolved, ImageError> {
252    let (axis, direction) = direction_axes(input.coordinates())?;
253    require_square_pixels(direction)?;
254    if !input.units().to_ascii_uppercase().contains("/PIXEL") {
255        return Err(ImageError::InvalidMetadata(format!(
256            "convolve2d: casa-rs convolves images in Jy/pixel; this one is in {:?}",
257            input.units()
258        )));
259    }
260    let shape = input.shape().to_vec();
261    let plane = [shape[axis], shape[axis + 1]];
262    let (side, values) = kernel(beam, sky_per_pixel(direction)?);
263    let mut convolver = PlaneConvolver::new(plane, side, &values)?;
264    let mut data = input.get()?;
265    let mask = input.default_mask()?;
266    let others: Vec<usize> = (0..shape.len())
267        .filter(|a| *a != axis && *a != axis + 1)
268        .collect();
269    let planes: usize = others.iter().map(|a| shape[*a]).product();
270    let mut index = vec![0usize; shape.len()];
271    let mut values = vec![0.0f64; plane[0] * plane[1]];
272    for plane_number in 0..planes {
273        let mut rest = plane_number;
274        for a in &others {
275            index[*a] = rest % shape[*a];
276            rest /= shape[*a];
277        }
278        for j in 0..plane[1] {
279            for i in 0..plane[0] {
280                index[axis] = i;
281                index[axis + 1] = j;
282                let good = mask
283                    .as_ref()
284                    .is_none_or(|mask| mask[ndarray::IxDyn(&index)]);
285                values[i + j * plane[0]] = if good {
286                    f64::from(data[ndarray::IxDyn(&index)])
287                } else {
288                    0.0
289                };
290            }
291        }
292        let convolved = convolver.convolve(&values)?;
293        for j in 0..plane[1] {
294            for i in 0..plane[0] {
295                index[axis] = i;
296                index[axis + 1] = j;
297                data[ndarray::IxDyn(&index)] = convolved[i + j * plane[0]] as f32;
298            }
299        }
300    }
301    Ok(Convolved {
302        data,
303        mask,
304        units: "Jy/beam".into(),
305        beam: GaussianBeam::new(
306            beam.major,
307            beam.minor,
308            unwrap_position_angle(beam.position_angle),
309        ),
310    })
311}

313/// Homogenize a spectral moment's selected planes to their common beam.
314/// The pixel mask remains on the same positions as the input.
315pub(crate) fn convolve_to_common_beam<T>(
316    data: &mut ArrayD<T>,
317    mask: Option<&ArrayD<bool>>,
318    coordinates: &CoordinateSystem,
319    units: &str,
320    beams: &crate::ImageBeamSet,
321) -> Result<(GaussianBeam, String), ImageError>
322where
323    T: Into<f64> + From<f32> + Copy,
324{
325    let target = beams.common_beam()?;
326    let (axis, direction) = direction_axes(coordinates)?;
327    require_square_pixels(direction)?;
328    let sky = sky_per_pixel(direction)?;
329    let shape = data.shape().to_vec();
330    let plane = [shape[axis], shape[axis + 1]];
331    let beam_axis = |kind| {
332        coordinates.find_coordinate(kind).map(|i| {
333            (0..i)
334                .map(|j| coordinates.coordinate(j).n_pixel_axes())
335                .sum::<usize>()
336        })
337    };
338    let spectral = beam_axis(CoordinateType::Spectral);
339    let stokes = beam_axis(CoordinateType::Stokes);
340    let others: Vec<_> = (0..shape.len())
341        .filter(|&i| i != axis && i != axis + 1)
342        .collect();
343    let count = others.iter().map(|&i| shape[i]).product();
344    let upper_units = units.to_ascii_uppercase();
345    let per_pixel = upper_units.contains("/PIXEL");
346    let per_beam = upper_units.contains("BEAM");
347    let mut units = units.to_owned();
348    let mut index = vec![0; shape.len()];
349    let mut values = vec![0.0; plane[0] * plane[1]];
350    for mut number in 0..count {
351        for &i in &others {
352            index[i] = number % shape[i];
353            number /= shape[i];
354        }
355        let channel = spectral.map_or(0, |i| index[i]);
356        let polarization = stokes.map_or(0, |i| index[i]);
357        let input = *beams.beam(channel, polarization);
358        let Some(smoothing) = target.deconvolving_beam(input)? else {
359            continue;
360        };
361        if smoothing.minor <= 0.0 {
362            return Err(ImageError::InvalidMetadata(
363                "the common-beam convolution requires a positive axial ratio".into(),
364            ));
365        }
366        let (side, mut weights) = kernel(&smoothing, sky);
367        let scale = if per_pixel {
368            // CASA loops over channels within each polarization. The last
369            // plane sets the output unit and keeps the input unit if copied.
370            if channel + 1 == beams.n_channels() && polarization + 1 == beams.n_stokes() {
371                units = "Jy/beam".into();
372            }
373            1.0
374        } else {
375            let (input_side, input_weights) = kernel(&input, sky);
376            let peak = central_gaussian_response(side, &weights, input_side, &input_weights);
377            if per_beam {
378                peak.recip()
379            } else {
380                input.area() / (target.area() * peak)
381            }
382        };
383        for weight in &mut weights {
384            *weight = f64::from((*weight * scale) as f32);
385        }
386        let mut convolver = PlaneConvolver::new(plane, side, &weights)?;
387        for j in 0..plane[1] {
388            index[axis + 1] = j;
389            for i in 0..plane[0] {
390                index[axis] = i;
391                let at = ndarray::IxDyn(&index);
392                values[i + j * plane[0]] = if mask.is_none_or(|mask| mask[at.clone()]) {
393                    data[at].into()
394                } else {
395                    0.0
396                };
397            }
398        }
399        let convolved = convolver.convolve(&values)?;
400        for j in 0..plane[1] {
401            index[axis + 1] = j;
402            for i in 0..plane[0] {
403                index[axis] = i;
404                data[ndarray::IxDyn(&index)] = T::from(convolved[i + j * plane[0]] as f32);
405            }
406        }
407    }
408    Ok((target, units))
409}

411fn central_gaussian_response(
412    left_side: usize,
413    left: &[f64],
414    right_side: usize,
415    right: &[f64],
416) -> f64 {
417    // Both odd-sized kernels are centred and even. The peak of their
418    // discrete convolution is their dot product over the central overlap.
419    let width = left_side.min(right_side);
420    let a = (left_side - width) / 2;
421    let b = (right_side - width) / 2;
422    let mut peak = 0.0;
423    for row in 0..width {
424        for col in 0..width {
425            peak += left[(row + a) * left_side + col + a] * right[(row + b) * right_side + col + b];
426        }
427    }
428    peak
429}

431/// A position angle in [−90°, 90°], as casacore's `GaussianBeam` keeps it.
432fn unwrap_position_angle(angle: f64) -> f64 {
433    use std::f64::consts::FRAC_PI_2;
434    let mut pa = angle % PI;
435    if pa > FRAC_PI_2 {
436        pa -= PI;
437    } else if pa < -FRAC_PI_2 {
438        pa += PI;
439    }
440    pa
441}

450    /// Sky offsets of RA increasing to the left and Dec up, 1e-6 rad pixels.

453    #[test]
454    fn kernel_sizes_round_as_image2dconvolver_rounds_them() {
455        // σ = 5 / √(8 ln 2) ≈ 2.123: 5σ = 10.6 → (11 + 1) × 2 = 24, made odd.
456        assert_eq!(size_of_gaussian(5.0), 24);
457        let beam = GaussianBeam::new(5.0e-6, 2.0e-6, 0.0);
458        assert_eq!(kernel(&beam, UNROTATED).0, 25);

461    #[test]
462    fn the_major_axis_lies_north_through_east_on_the_sky() {
463        // RA increases to the left (negative increment): a position angle of
464        // 90° (east) puts the major axis along −x, the same line as +x.
465        let beam = GaussianBeam::new(6.0e-6, 2.0e-6, std::f64::consts::FRAC_PI_2);
466        let (side, values) = kernel(&beam, UNROTATED);
467        let c = (side - 1) / 2;
468        let along_x = values[(c + 3) + c * side];
469        let along_y = values[c + (c + 3) * side];
470        assert!(along_x > along_y, "{along_x} {along_y}");
471        // Peak 1 at the centre.
472        assert!((values[c + c * side] - 1.0).abs() < 1e-15);
