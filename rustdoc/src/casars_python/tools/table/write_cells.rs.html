2//! The `table` tool's cell and column writes (`putcell`, `putcol`,
3//! `putvarcol`), as CASA 6.7.6.14's `tb` does them
4//! (`tools/casa-compat/table_tool_oracle.py`, `cells_write` and
5//! `ms_cells_write`).
6//!
7//! A value is coerced to the column's declared type ([`super::cell_value`]),
8//! checked against the column's shape, set in memory, and the cells written
9//! in place in the column's storage manager with
10//! `TableWritePlan::save_selected_rows`; no other cell and no other storage
11//! manager is touched. The lock handling is [`State::write_with`]'s.
12

25fn runtime<T>(message: String) -> Result<T, Fail> {
26    Err(Fail::Runtime(message))
27}

29pub(super) fn put_column_error(what: &str) -> Fail {
30    Fail::Runtime(format!("TableProxy::TableProxy::putColumn: {what}"))
31}

33/// What a column holds: its type (`None`: records) and whether its cells are
34/// arrays.
35fn describe(table: &Table, column: &str) -> Result<(Option<PrimitiveType>, bool), Fail> {
36    let schema: &ColumnSchema = table
37        .schema()
38        .and_then(|schema| schema.column(column))
39        .ok_or_else(|| put_column_error(&format!("column {column} does not exist")))?;
40    Ok((
41        schema.data_type(),
42        matches!(schema.column_type(), ColumnType::Array(_)),
43    ))
44}

46fn shape_text(shape: &[usize]) -> String {
47    let parts: Vec<String> = shape.iter().map(ToString::to_string).collect();
48    format!("[{}]", parts.join(", "))
49}

51/// The refusal of an array whose shape the column's cell cannot take.
52pub(super) fn conformance(row: usize, column: &str, shape: &[usize], expected: &[usize]) -> Fail {
53    Fail::Runtime(format!(
54        "ArrayColumn::put for row {row} in column {column}: Table array conformance error \
55         (shape={}, expected {})",
56        shape_text(shape),
57        shape_text(expected)
58    ))
59}

61/// Whether the column's storage manager holds one shape per cell once set
62/// (the tiled managers do; the indirect ones let a cell be reshaped).
63fn keeps_cell_shapes(table: &Table, column: &str) -> bool {
64    table.data_manager_info().iter().any(|manager| {
65        manager.dm_type.contains("Tiled") && manager.columns.iter().any(|c| c == column)
66    })
67}

69/// What a column asks of the shape of the arrays written to it, worked out
70/// once for all the cells of a call.
71struct ShapeRule {
72    /// The one shape every cell has.
73    fixed: Option<Vec<usize>>,
74    /// Whether a cell keeps the shape it has once set (a tiled manager).
75    kept: bool,
76}
77
78impl ShapeRule {
79    fn of(table: &Table, column: &str) -> Self {
80        let contract = table
81            .schema()
82            .and_then(|schema| schema.column(column))
83            .map(ColumnSchema::column_type);
84        Self {
85            fixed: match contract {
86                Some(ColumnType::Array(ArrayShapeContract::Fixed { shape })) => Some(shape.clone()),
87                _ => None,
88            },
89            kept: keeps_cell_shapes(table, column),
90        }
91    }
92
93    /// Checks an array cell's shape against its column's.
94    fn check(&self, table: &Table, column: &str, row: usize, cell: &Value) -> Result<(), Fail> {
95        let Value::Array(array) = cell else {
96            return Ok(());
97        };
98        let shape = array.shape();
99        if let Some(fixed) = &self.fixed
100            && shape != fixed.as_slice()
101        {
102            return Err(conformance(row, column, shape, fixed));
103        }
104        if self.kept
105            && let Some(existing) = table.array_shape(row, column)?
106            && existing != shape
107        {
108            return Err(conformance(row, column, shape, &existing));
109        }
110        Ok(())
111    }
112}

114/// Sets cells of one column in memory: each cell is checked, and none is
115/// set before every one has been (a refused call changes nothing). A cell
116/// of a scalar or array column is set where it is, without loading the
117/// rest of the table (a table is read lazily, and a MeasurementSet's visibilities
118/// would not fit in memory); a record cell goes through the row store.
119fn set_cells(table: &mut Table, column: &str, cells: Vec<(usize, Value)>) -> Result<(), Fail> {
120    let record_column = matches!(
121        table
122            .schema()
123            .and_then(|schema| schema.column(column))
124            .map(ColumnSchema::column_type),
125        Some(ColumnType::Record)
126    );
127    let rule = ShapeRule::of(table, column);
128    for (row, cell) in &cells {
129        rule.check(table, column, *row, cell)?;
130    }
131    if record_column
132        || cells
133            .iter()
134            .any(|(_, cell)| matches!(cell, Value::Record(_)))
135    {
136        let mut accessor = table.column_accessor_mut(column)?;
137        for (row, cell) in cells {
138            match cell {
139                Value::Record(record) => accessor.set_record(row, record)?,
140                other => accessor.set(row, other)?,
141            }
142        }
143        return Ok(());
144    }
145    let mut writer = table.row_accessor_mut().prepare(&[column])?;
146    for (row, cell) in cells {
147        writer.seek(row)?;
148        writer.set_value_at(0, cell)?;
149    }
150    writer.flush()?;
151    Ok(())
152}

154/// The distinct rows of `rows`, in order, as a save of selected rows wants
155/// them.
156fn distinct(rows: &[usize]) -> Vec<usize> {
157    let mut seen = std::collections::HashSet::with_capacity(rows.len());
158    rows.iter()
159        .copied()
160        .filter(|row| seen.insert(*row))
161        .collect()
162}
163
164/// The elements of `data` at `indices`.
165fn take(data: &Data, indices: &[usize]) -> Data {
166    match data {
167        Data::Bools(values) => Data::Bools(indices.iter().map(|i| values[*i]).collect()),
168        Data::Ints(values) => Data::Ints(indices.iter().map(|i| values[*i]).collect()),
169        Data::Floats(values) => Data::Floats(indices.iter().map(|i| values[*i]).collect()),
170        Data::Complexes(values) => Data::Complexes(indices.iter().map(|i| values[*i]).collect()),
171        Data::Texts(values) => Data::Texts(indices.iter().map(|i| values[*i].clone()).collect()),
172    }
173}

175/// A column's values split into one value per row: the rows are the last
176/// axis of an array column's values (NumPy's, in C order), the only axis of
177/// a scalar column's.
178pub(super) fn split_rows(
179    wide: Wide,
180    array: bool,
181    column: &str,
182    wanted: usize,
183) -> Result<Vec<Wide>, Fail> {
184    let shape = wide.shape.clone().unwrap_or_else(|| vec![1]);
185    let (cell_shape, nrows) = if array {
186        match shape.split_last() {
187            Some((rows, cell)) if !cell.is_empty() => (Some(cell.to_vec()), *rows),
188            _ => (None, usize::MAX),
189        }
190    } else {
191        (
192            None,
193            if shape.len() == 1 {
194                shape[0]
195            } else {
196                usize::MAX
197            },
198        )
199    };
200    if nrows != wanted {
201        return runtime(if array {
202            format!(
203                "ArrayColumn::putColumn - column {column} has {wanted}, array has {} rows: \
204                 Table array conformance error",
205                if nrows == usize::MAX { 0 } else { nrows }
206            )
207        } else {
208            "ScalarColumn::putColumn(Vector&): Table conformance error (#rows mismatch)".into()
209        });
210    }
211    let per_row: usize = cell_shape.as_ref().map_or(1, |cell| cell.iter().product());
212    Ok((0..nrows)
213        .map(|row| {
214            let indices: Vec<usize> = (0..per_row).map(|element| element * nrows + row).collect();
215            Wide {
216                shape: cell_shape.clone(),
217                data: take(&wide.data, &indices),
218            }
219        })
220        .collect())
221}

223impl State {
224    /// `putcell(columnname, rownr, thevalue)`: one row number, or several,
225    /// each given the value. A row that does not exist refuses the call
226    /// whole (CASA writes the rows before it, then fails).
227    pub(super) fn putcell(
228        &mut self,
229        column: String,
230        rows: Vec<i64>,
231        value: Incoming,
232        log: &mut Vec<LogLine>,
233    ) -> Reply {
234        self.write_with(Target::Cells, log, move |table| {
235            let (ty, array) = describe(table, &column)?;
236            let nrows = table.row_count();
237            let rows = rows
238                .into_iter()
239                .map(|row| {
240                    usize::try_from(row)
241                        .ok()
242                        .filter(|row| *row < nrows)
243                        .ok_or_else(|| put_column_error("no such row"))
244                })
245                .collect::<Result<Vec<_>, _>>()?;
246            if rows.is_empty() {
247                return Ok(Persist::Nothing);
248            }
249            let cell = value.into_cell(ty, array)?;
250            let rows = distinct(&rows);
251            set_cells(
252                table,
253                &column,
254                rows.iter().map(|row| (*row, cell.clone())).collect(),
255            )?;
256            Ok(Persist::Rows { column, rows })
257        })
258    }
259
260    /// `putcol(columnname, value, startrow, nrow, rowincr)`.
261    pub(super) fn putcol(
262        &mut self,
263        column: String,
264        value: Incoming,
265        selection: (i64, i64, i64),
266        log: &mut Vec<LogLine>,
267    ) -> Reply {
268        self.write_with(Target::Cells, log, move |table| {
269            let (ty, array) = describe(table, &column)?;
270            let selected = rows(
271                table.row_count(),
272                selection.0,
273                selection.1,
274                selection.2,
275                "TableProxy::TableProxy::putColumn: no such row",
276            )?;
277            let cells = match (value, ty) {
278                (Incoming::Wide(wide), Some(_)) => {
279                    split_rows(wide, array, &column, selected.len())?
280                }
281                (value, _) => {
282                    // A record column is written a cell at a time with
283                    // `putcell`; `putcol` of one is not possible.
284                    let _ = value;
285                    return runtime(
286                        "TableProxy::putColumn not possible for a column containing records".into(),
287                    );
288                }
289            };
290            let cells = selected
291                .iter()
292                .zip(cells)
293                .map(|(row, cell)| Ok((*row, cell.into_cell(ty.expect("typed column"), array)?)))
294                .collect::<Result<Vec<_>, Fail>>()?;
295            if cells.is_empty() {
296                return Ok(Persist::Nothing);
297            }
298            set_cells(table, &column, cells)?;
299            Ok(Persist::Rows {
300                column,
301                rows: selected,
302            })
303        })
304    }
305
306    /// `putvarcol(columnname, value, startrow, nrow, rowincr)`: a record
307    /// whose fields, in the order they have, are the values of the selected
308    /// rows, each with a trailing axis of one, as `getvarcol` returns them.
309    /// The field names are labels (`getvarcol` names them `r1`, `r2`, ...),
310    /// not row numbers.
311    pub(super) fn putvarcol(
312        &mut self,
313        column: String,
314        fields: Vec<(String, Incoming)>,
315        selection: (i64, i64, i64),
316        log: &mut Vec<LogLine>,
317    ) -> Reply {
318        self.write_with(Target::Cells, log, move |table| {
319            let (ty, array) = describe(table, &column)?;
320            let selected = rows_stepping(
321                table.row_count(),
322                selection.0,
323                selection.1,
324                selection.2,
325                "TableProxy::TableProxy::putVarColumn: no such row",
326                "TableProxy::putVarColumn: rowincr<=0",
327            )?;
328            if fields.len() != selected.len() {
329                return runtime("TableProxy::putVarColumn: #rows mismatches #elem in value".into());
330            }
331            let Some(ty) = ty else {
332                return runtime(
333                    "TableProxy::putVarColumn not possible for a column containing records".into(),
334                );
335            };
336            let mut cells = Vec::with_capacity(fields.len());
337            for ((key, value), row) in fields.into_iter().zip(&selected) {
338                let Incoming::Wide(wide) = value else {
339                    return runtime(format!("a value of {key} is not an array"));
340                };
341                let mut shape = wide.shape.clone().unwrap_or_else(|| vec![1]);
342                if array && shape.len() < 2 || shape.pop() != Some(1) {
343                    return runtime(format!(
344                        "Invalid Table operation: ArrayColumn::setShape: mismatch in #dim of array of column {column}"
345                    ));
346                }
347                let cell = Wide {
348                    shape: array.then_some(shape),
349                    data: wide.data,
350                };
351                cells.push((*row, cell.into_cell(ty, array)?));
352            }
353            if cells.is_empty() {
354                return Ok(Persist::Nothing);
355            }
356            set_cells(table, &column, cells)?;
357            Ok(Persist::Rows {
358                column,
359                rows: selected,
360            })
361        })
362    }
