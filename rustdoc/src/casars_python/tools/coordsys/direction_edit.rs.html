27fn raised(error: impl ToString) -> PyErr {
28    PyRuntimeError::new_err(error.to_string())
29}

31enum Transform {
32    Default,
33    Invalid,
34    Matrix(Array2<f64>),
35}
36
37fn capture_transform(value: Option<&Bound<'_, PyAny>>) -> PyResult<Transform> {
38    let Some(value) = value.filter(|value| !value.is_none()) else {
39        return Ok(Transform::Default);
40    };
41    if let Some(unset) = bool_vector(value)? {
42        return Ok(if unset {
43            Transform::Default
44        } else {
45            Transform::Invalid
46        });
47    }
48    let py = value.py();
49    let array = py.import("numpy")?.call_method1("asarray", (value,))?;
50    let array = array.cast::<PyUntypedArray>()?;
51    if !matches!(array.dtype().char(), b'f' | b'd') || array.ndim() == 0 {
52        return Ok(Transform::Invalid);
53    }
54    matrix_input(Some(array.as_any())).map(Transform::Matrix)
55}
56
57fn capture_poles(value: Option<&Bound<'_, PyAny>>) -> PyResult<[f64; 2]> {
58    let Some(value) = value.filter(|value| !value.is_none()) else {
59        return Ok([PI, 0.0]);
60    };
61    if bool_vector(value)? == Some(true) {
62        return Ok([PI, 0.0]);
63    }
64    let invalid = || raised("pole parameter is not understood");
65    let texts = match value.extract::<Variant>()? {
66        Variant::Text(value) => quantity_tokens(&value),
67        Variant::Texts(values) => values,
68        _ => return Err(invalid()),
69    };
70    let [longitude, latitude] = text_quantities(&texts)?.try_into().map_err(|_| invalid())?;
71    let rad = casacore::casacore_unit("rad").map_err(raised)?;
72    Ok([
73        casacore::get_value(&longitude, &rad),
74        casacore::get_value(&latitude, &rad),
75    ])
76}

78fn direction_value(
79    value: WorldArgument,
80    state: &CoordinateState,
81    defaults: Vec<f64>,
82) -> PyResult<[f64; 2]> {
83    let values = if matches!(value, WorldArgument::Unset) {
84        defaults
85    } else {
86        value.values_with_defaults(state, None, defaults)?
87    };
88    values
89        .try_into()
90        .map_err(|_| raised("Direction requires two world-axis values"))
91}

93fn capture_world_value(
94    py: Python<'_>,
95    value: Option<&Bound<'_, PyAny>>,
96) -> PyResult<WorldArgument> {
97    if let Some(value) = value
98        && let Ok(array) = value.cast::<PyUntypedArray>()
99        && array.ndim() == 0
100    {
101        // CASA treats a zero-dimensional array as a one-element coordinate
102        // vector, including when the array holds an integer.
103        let flattened = value.call_method1("reshape", (1,))?;
104        return WorldArgument::capture(py, Some(&flattened));
105    }
106    if let Some(value) = value
107        && value.is_instance(&py.import("numpy")?.getattr("generic")?)?
108    {
109        let dtype = value.getattr("dtype")?;
110        let kind: char = dtype.getattr("kind")?.extract()?;
111        let code: char = dtype.getattr("char")?.extract()?;
112        if kind == 'i' {
113            return Err(raised("Unrecognized format for world coordinate"));
114        }
115        if matches!(kind, 'u' | 'b') || matches!(code, 'g' | 'F' | 'G') {
116            return Ok(WorldArgument::Unset);
117        }
118    }
119    WorldArgument::capture(py, value)
120}

122struct DirectionEdit {
123    frame: String,
124    projection: String,
125    parameters: Vec<f64>,
126    pixels: Vec<f64>,
127    reference: WorldArgument,
128    increment: WorldArgument,
129    transform: Transform,
130    poles: [f64; 2],
131}
132
133fn set_direction(
134    state: &mut CoordinateState,
135    edit: DirectionEdit,
136    log: &mut Vec<LogLine>,
137) -> PyResult<bool> {
138    let (index, _, placement) = numbered_coordinates(state)?
139        .into_iter()
140        .find(|(_, model, _)| matches!(model, CoordinateModel::Direction(_)))
141        .ok_or_else(|| missing(CoordinateType::Direction))?;
142    let CoordinateModel::Direction(original) = placement.model(&state.record).map_err(raised)?
143    else {
144        return Err(missing(CoordinateType::Direction));
145    };
146    let pc = match edit.transform {
147        Transform::Default => Array2::eye(2),
148        Transform::Invalid => {
149            log.push(LogLine::warning("xform paramater invalid"));
150            return Ok(false);
151        }
152        Transform::Matrix(matrix) => matrix,
153    };
154    let frame = if edit.frame.is_empty() {
155        original.direction_ref()
156    } else {
157        DIRECTION_CODES
158            .resolve(&edit.frame)
159            .and_then(|code| code.parse().ok())
160            .ok_or_else(|| raised(format!("Invalid direction code '{}' given", edit.frame)))?
161    };
162    let name = if edit.projection.is_empty() {
163        original.projection().name()
164    } else {
165        &edit.projection
166    };
167    let parameters = if edit.parameters == [-1.0] {
168        original.projection().parameters().to_vec()
169    } else {
170        edit.parameters
171    };
172    let projection = direction_projection(name, parameters)?;
173    projection
174        .validate_conversion_parameters()
175        .map_err(raised)?;
176    let pixels = if edit.pixels == [-1.0] {
177        [placement.pixel_axes[0], placement.pixel_axes[1]]
178            .map(|axis| axis.map_or(-1.0, |axis| axis as f64))
179    } else if let [longitude, latitude, ..] = edit.pixels.as_slice() {
180        [*longitude, *latitude]
181    } else {
182        return Err(raised("Direction reference pixel requires two components"));
183    };
184    let mut local = CoordinateSystem::new().with_obs_info(state.coordinates.obs_info().clone());
185    let old_reference = original.reference_value();
186    let old_increment = original.increment();
187    local.add_coordinate(original);
188    let local = CoordinateState::new(local);
189    let reference = direction_value(edit.reference, &local, old_reference)?;
190    let increment = direction_value(edit.increment, &local, old_increment)?;
191    if reference
192        .iter()
193        .chain(&increment)
194        .chain(&pixels)
195        .any(|value| !value.is_finite())
196    {
197        return Err(raised(
198            "Direction reference values, increments and pixels must be finite",
199        ));
200    }
201    if pc.shape() != [2, 2]
202        || pc[[0, 0]] * pc[[1, 1]] == pc[[0, 1]] * pc[[1, 0]]
203        || increment.contains(&0.0)
204    {
205        return Err(raised(
206            "wcs wcsset_error: Linear transformation matrix is singular",
207        ));
208    }
209    let changed = CoordinateModel::from(
210        DirectionCoordinate::new(frame, projection, reference, increment, pixels)
211            .with_normalized_pc_matrix(pc)
212            .and_then(|model| model.with_wcs_poles(edit.poles[0], edit.poles[1]))
213            .map_err(raised)?,
214    );
215    let record = Value::Record(changed.to_record());
216    replace_at(state, index as i32, changed, record)?;
217    Ok(true)
218}

220#[pymethods]
221impl PyCoordsys {
222    #[allow(clippy::too_many_arguments)]
223    #[pyo3(signature = (refcode=FrameName::default(), proj=FrameName::default(), projpar=DoubleList::of(&[-1.0]), refpix=DoubleList::of(&[-1.0]), refval=None, incr=None, xform=None, poles=None), text_signature = "($self, refcode='', proj='', projpar=[-1.0], refpix=[-1.0], refval=[], incr=[], xform=[], poles=[])")]
224    fn setdirection(
225        &self,
226        py: Python<'_>,
227        refcode: FrameName,
228        proj: FrameName,
229        projpar: DoubleList,
230        refpix: DoubleList,
231        refval: Option<&Bound<'_, PyAny>>,
232        incr: Option<&Bound<'_, PyAny>>,
233        xform: Option<&Bound<'_, PyAny>>,
234        poles: Option<&Bound<'_, PyAny>>,
235    ) -> PyResult<bool> {
236        let edit = DirectionEdit {
237            frame: refcode.into_name("refcode")?,
238            projection: proj.into_name("proj")?,
239            parameters: projpar.0,
240            pixels: refpix.0,
241            reference: capture_world_value(py, refval)?,
242            increment: capture_world_value(py, incr)?,
243            transform: capture_transform(xform)?,
244            poles: capture_poles(poles)?,
245        };
246        let mut log = Vec::new();
247        let result = py.detach(|| set_direction(&mut self.state(), edit, &mut log));
248        emit(py, "casars.tools.coordsys", &log)?;
249        result
250    }
