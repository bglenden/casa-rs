2//! Legacy GSPLINE gains for simulated DATA, following GJonesSpline::calcPar.
3use super::*;
4use casa_calibration::{CalibrationTableLayout, ComplexSolution};
5use casa_types::{ArrayValue, ScalarValue};
6use std::collections::HashMap;
7use std::path::Path;
8
9pub(super) struct SplineTable {
10    rows: HashMap<(i32, usize), Row>,
11    pub(super) windows: usize,
12}
13
14struct Row {
15    antenna: i32,
16    field: i32,
17    scale: Complex32,
18    reference_hz: f64,
19    amplitude: Option<Curve>,
20    phase: Option<Curve>,
21}
22
23struct Curve {
24    knots: Vec<f64>,
25    coefficients: Vec<f64>,
26    stride: usize,
27}

29impl SplineTable {
30    pub(super) fn layout(path: &Path, windows: usize) -> SimulatorResult<CalibrationTableLayout> {
31        let table = Table::open(TableOptions::new(path)).map_err(raised)?;
32        let rows = table.row_count();
33        let mut values = 0usize;
34        for name in [
35            "SPLINE_KNOTS_AMP",
36            "SPLINE_KNOTS_PHASE",
37            "POLY_COEFF_AMP",
38            "POLY_COEFF_PHASE",
39        ] {
40            let column = table.column_accessor(name).map_err(raised)?;
41            for start in (0..rows).step_by(512) {
42                let selected: Vec<_> = (start..rows.min(start + 512)).collect();
43                for shape in column
44                    .array_shapes_uncached(&selected)
45                    .map_err(raised)?
46                    .into_iter()
47                    .flatten()
48                {
49                    let size = shape
50                        .into_iter()
51                        .try_fold(1usize, usize::checked_mul)
52                        .ok_or_else(|| raised("GSPLINE cell size overflows usize"))?;
53                    values = values
54                        .checked_add(size)
55                        .ok_or_else(|| raised("GSPLINE value count overflows usize"))?;
56                }
57            }
58        }
59        Ok(CalibrationTableLayout {
60            rows,
61            values,
62            max_cell_values: 2,
63            frequency_values: 0,
64            frequency_windows: windows,
65        })
66    }
67
68    pub(super) fn open(path: &Path, windows: usize) -> SimulatorResult<Self> {
69        let table = Table::open(TableOptions::new(path)).map_err(raised)?;
70        let mut rows = HashMap::with_capacity(table.row_count());
71        for index in 0..table.row_count() {
72            let mode = match table
73                .cell_accessor(index, "POLY_MODE")
74                .map_err(raised)?
75                .scalar()
76                .map_err(raised)?
77            {
78                ScalarValue::String(mode) => mode,
79                _ => return Err(raised("GSPLINE POLY_MODE is not a string")),
80            };
81            let scale = match table
82                .cell_accessor(index, "SCALE_FACTOR")
83                .map_err(raised)?
84                .scalar()
85                .map_err(raised)?
86            {
87                ScalarValue::Complex32(value) => *value,
88                _ => return Err(raised("GSPLINE SCALE_FACTOR is not complex")),
89            };
90            let reference = floats(&table, index, "REF_FREQUENCY")?;
91            let row = Row {
92                antenna: integer(&table, index, "ANTENNA1")?,
93                field: integer(&table, index, "FIELD_ID")?,
94                scale,
95                reference_hz: reference.first().copied().unwrap_or(0.),
96                amplitude: (mode.contains("AMP") || mode.contains("A&P"))
97                    .then(|| Curve::open(&table, index, "AMP"))
98                    .transpose()?,
99                phase: (mode.contains("PHAS") || mode.contains("A&P"))
100                    .then(|| Curve::open(&table, index, "PHASE"))
101                    .transpose()?,
102            };
103            rows.entry((row.antenna, row.field as usize)).or_insert(row);
104        }
105        Ok(Self { rows, windows })
106    }
107
108    pub(super) fn solution(
109        &self,
110        antenna: i32,
111        field: usize,
112        time: f64,
113        frequency: f64,
114    ) -> Option<ComplexSolution> {
115        // The legacy buffer matches antenna AND field and takes its first
116        // matching row. Interpolation arguments do not change these splines.
117        let row = self.rows.get(&(antenna, field))?;
118        let ratio = if row.reference_hz.abs() > 0. {
119            frequency / row.reference_hz
120        } else {
121            1.
122        };
123        let values = (0..2)
124            .map(|receptor| {
125                let amplitude = row
126                    .amplitude
127                    .as_ref()
128                    .map_or(1., |c| c.value(time, receptor).exp());
129                // GILDAS phases have the opposite sign; CASA also scales them
130                // by mean observing frequency / calibration reference frequency.
131                let phase = row
132                    .phase
133                    .as_ref()
134                    .map_or(0., |c| -c.value(time, receptor) * ratio);
135                row.scale * Complex32::from_polar(amplitude as f32, phase as f32)
136            })
137            .collect();
138        Some(ComplexSolution {
139            receptors: 2,
140            channels: 1,
141            values,
142            flags: vec![false; 2],
143        })
144    }
145}

147impl Curve {
148    fn open(table: &Table, row: usize, suffix: &str) -> SimulatorResult<Self> {
149        let stride = usize::try_from(integer(table, row, &format!("N_POLY_{suffix}"))?)
150            .map_err(|_| raised("GSPLINE coefficient count is negative"))?;
151        let count = usize::try_from(integer(table, row, &format!("N_KNOTS_{suffix}"))?)
152            .map_err(|_| raised("GSPLINE knot count is negative"))?;
153        let knots = floats(table, row, &format!("SPLINE_KNOTS_{suffix}"))?;
154        let coefficients = floats(table, row, &format!("POLY_COEFF_{suffix}"))?;
155        if count < 8
156            || knots.len() != count
157            || stride < count - 4
158            || stride.checked_mul(2) != Some(coefficients.len())
159            || knots.iter().chain(&coefficients).any(|v| !v.is_finite())
160            || knots.windows(2).any(|w| w[0] > w[1])
161            || knots[3] >= knots[count - 4]
162        {
163            return Err(raised("invalid GSPLINE cubic knot/coefficient layout"));
164        }
165        Ok(Self {
166            knots,
167            coefficients,
168            stride,
169        })
170    }
171
172    fn value(&self, time: f64, receptor: usize) -> f64 {
173        let knots = &self.knots;
174        let count = knots.len() - 4;
175        // GETBSPL returns zero outside the fitted interval. Consequently
176        // the gain retains SCALE_FACTOR, with unit amplitude and zero phase.
177        if time < knots[3] || time > knots[count] {
178            return 0.;
179        }
180        let span = knots
181            .partition_point(|&knot| knot <= time)
182            .saturating_sub(1)
183            .clamp(3, count - 1);
184        let coefficients = &self.coefficients[receptor * self.stride..];
185        let mut values: [f64; 4] = std::array::from_fn(|j| coefficients[span - 3 + j]);
186        // de Boor's cubic recurrence, retaining only four values on the stack.
187        for order in 1..=3 {
188            for j in (order..=3).rev() {
189                let i = span - 3 + j;
190                let width = knots[i + 4 - order] - knots[i];
191                let fraction = if width > 0. {
192                    (time - knots[i]) / width
193                } else {
194                    0.
195                };
196                values[j] = (1. - fraction) * values[j - 1] + fraction * values[j];
197            }
198        }
199        values[3]
200    }
201}

203fn integer(table: &Table, row: usize, name: &str) -> SimulatorResult<i32> {
204    match table
205        .cell_accessor(row, name)
206        .map_err(raised)?
207        .scalar()
208        .map_err(raised)?
209    {
210        ScalarValue::Int32(value) => Ok(*value),
211        _ => Err(raised(format!("GSPLINE {name} is not an integer"))),
212    }
213}
214
215fn floats(table: &Table, row: usize, name: &str) -> SimulatorResult<Vec<f64>> {
216    match table
217        .cell_accessor(row, name)
218        .map_err(raised)?
219        .array()
220        .map_err(raised)?
221    {
222        ArrayValue::Float64(values) => Ok(values.iter().copied().collect()),
223        _ => Err(raised(format!("GSPLINE {name} is not a double array"))),
224    }
