3//! CASA's automatic multi-threshold mask
4//! (`SDMaskHandler::autoMask` and `autoMaskByMultiThreshold`, CASA 6.7).
5//!
6//! At each major-cycle boundary the plane's thresholds come from its
7//! residual: the larger of the PSF sidelobe level times `sidelobethreshold`
8//! times the residual's absolute peak and the noise times
9//! `noisethreshold`, both above the residual's median. The mask the plane
10//! inherits then gains
11//!
12//! * the regions above that threshold, less those smaller than
13//!   `minbeamfrac` beams, smoothed by `smoothfactor` beams and cut at
14//!   `cutthreshold` of the smoothed peak;
15//! * after the first cycle, the inherited mask grown by binary dilation
16//!   through the pixels above the low-noise threshold, pruned when
17//!   `dogrowprune`, then smoothed and cut the same way;
18//! * with `negativethreshold`, the regions below the negative threshold,
19//!   smoothed and cut but never pruned.
20//!
21//! A plane whose threshold is the noise's stops evolving
22//! (`SDMaskHandler::skipChannels`) once its mask is empty, or once the
23//! cycle threshold has reached the global threshold and the mask changed
24//! by less than `minpercentchange` percent.
25

34/// The primary PSF's fitted main lobe (pixels) and sidelobe level, measured
35/// once per PSF by the minor cycle, as the automatic mask needs them.
36#[derive(Debug, Clone, Copy, PartialEq)]
37pub struct AutoMaskBeam {
38    major_fwhm_pixels: f64,
39    minor_fwhm_pixels: f64,
40    position_angle_rad: f64,
41    area_pixels: f64,
42    sidelobe_fraction: f64,
43}
44
45impl AutoMaskBeam {
46    /// A beam of the given full widths at half maximum (pixels), position
47    /// angle and fitted sidelobe level.
48    #[must_use]
49    pub const fn new(
50        major_fwhm_pixels: f64,
51        minor_fwhm_pixels: f64,
52        position_angle_rad: f64,
53        sidelobe_fraction: f64,
54    ) -> Self {
55        Self {
56            major_fwhm_pixels,
57            minor_fwhm_pixels,
58            position_angle_rad,
59            area_pixels: std::f64::consts::PI * major_fwhm_pixels * minor_fwhm_pixels
60                / (4.0 * std::f64::consts::LN_2),
61            sidelobe_fraction,
62        }
63    }
64}

66/// CASA auto-multithreshold controls: tclean's parameters of the same
67/// names.
68#[derive(Debug, Clone, Copy, PartialEq)]
69pub struct AutoMultithreshControls {
70    /// Sidelobe threshold multiplier (`sidelobethreshold`).
71    pub sidelobe_factor: f64,
72    /// Robust-noise threshold multiplier (`noisethreshold`).
73    pub noise_factor: f64,
74    /// Low-noise growth multiplier (`lownoisethreshold`).
75    pub low_noise_factor: f64,
76    /// Negative-feature threshold multiplier; zero disables negative masks
77    /// (`negativethreshold`).
78    pub negative_factor: f64,
79    /// Minimum connected-region area as a fraction of the beam area; zero
80    /// or less prunes nothing (`minbeamfrac`).
81    pub minimum_beam_fraction: f64,
82    /// Gaussian smoothing FWHM as a fraction of the restoring beam
83    /// (`smoothfactor`).
84    pub smooth_factor: f64,
85    /// Fraction of the smoothed-mask peak retained (`cutthreshold`).
86    pub cut_threshold: f64,
87    /// Bound on the binary dilations that grow the mask
88    /// (`growiterations`).
89    pub grow_iterations: usize,
90    /// Whether the grown mask is pruned as the threshold mask is
91    /// (`dogrowprune`).
92    pub grow_prune: bool,
93    /// Percent change below which a plane's mask stops evolving once the
94    /// global threshold is reached; negative never stops on change
95    /// (`minpercentchange`).
96    pub minimum_percent_change: f64,
97    /// How the plane's noise is estimated (`fastnoise`).
98    pub noise: NoiseEstimate,
99    /// Whether each update logs CASA's automask summary (`verbose`).
100    pub verbose: bool,
101}

103/// Which threshold a plane's mask used (CASA's `thresh_type`).
104#[derive(Debug, Clone, Copy, PartialEq, Eq)]
105pub enum AutoMaskThreshold {
106    /// The PSF sidelobe level times the residual's absolute peak.
107    Sidelobe,
108    /// The noise times `noisethreshold`.
109    Noise,
110}

112impl AutoMaskThreshold {

114    #[must_use]
115    pub const fn name(self) -> &'static str {
116        match self {
117            Self::Sidelobe => "sidelobe",
118            Self::Noise => "noise",
119        }

173/// Heap bytes per pixel [`auto_multithresh`] holds at most: the normalised
174/// residual and the PSF it is divided by, a copy for its statistics and a
175/// smoothed plane (`f64` each); region pruning's component list and
176/// queue, grown by pushes (`usize` each, twice); and its twelve `bool`
177/// planes.
178pub const AUTO_MULTITHRESH_BYTES_PER_CELL: u64 =
179    4 * size_of::<f64>() as u64 + 4 * size_of::<usize>() as u64 + 12;
180
181/// Heap bytes [`auto_multithresh`] holds at most for one plane of `shape`
182/// (`[width, height]`): `AUTO_MULTITHRESH_BYTES_PER_CELL` per pixel and
183/// one row of its smoothing kernel, which holds at most one value per
184/// offset along the plane's longer axis (`2 * extent - 1`, `f64` each).
185#[must_use]
186pub fn auto_multithresh_bytes(shape: [usize; 2]) -> u64 {
187    let cells = (shape[0] as u64).saturating_mul(shape[1] as u64);
188    let row = 2 * shape[0].max(shape[1]) as u64 * size_of::<f64>() as u64;
189    cells
190        .saturating_mul(AUTO_MULTITHRESH_BYTES_PER_CELL)
191        .saturating_add(row)
192}

194/// The thresholds of one plane (`autoMaskByMultiThreshold`).
195#[derive(Debug, Clone, Copy)]
196struct Thresholds {
197    positive: f64,
198    low: f64,
199    negative: f64,
200    kind: AutoMaskThreshold,
201}
202
203impl Thresholds {
204    /// Every threshold is relative to the median; the positive one is the
205    /// larger of the sidelobe and noise thresholds, and is the sidelobe's
206    /// when they tie.
207    fn new(
208        noise: RobustNoise,
209        peak: f64,
210        sidelobe_level: f64,
211        controls: &AutoMultithreshControls,
212    ) -> Self {
213        let sidelobe = sidelobe_level * controls.sidelobe_factor * peak.abs();
214        let noise_threshold = controls.noise_factor * noise.rms;
215        let low_noise = controls.low_noise_factor * noise.rms;
216        let negative = controls.negative_factor * noise.rms;
217        let positive = (sidelobe + noise.median).max(noise_threshold + noise.median);
218        Self {
219            positive,
220            low: (sidelobe + noise.median).max(low_noise + noise.median),
221            negative: noise.median - sidelobe.max(negative),
222            kind: if positive == sidelobe + noise.median {
223                AutoMaskThreshold::Sidelobe
224            } else {
225                AutoMaskThreshold::Noise
226            },
227        }
228    }
229}

231/// Generate the next auto-multithreshold mask of the primary plane of
232/// `normal` from the mask it inherits, `previous`; `valid_support` is the
233/// plane's valid pixels (the residual's pixel mask) in canonical x-major
234/// order, and `beam` the PSF's fitted main lobe and sidelobe level.
235///
236/// # Errors
237///
238/// The state is not one channel of one plane, shapes disagree, the
239/// residual is not finite, or the controls or beam are invalid.
240pub fn auto_multithresh(
241    coordinate: DirectionCoordinateSpec,
242    normal: &FinalNormalState,
243    previous: Option<&ReconstructionMask>,
244    valid_support: &[bool],
245    cycle: AutoMaskCycle,
246    controls: AutoMultithreshControls,
247    beam: AutoMaskBeam,
248) -> Result<(ReconstructionMask, AutoMultithreshEvidence), MaskError> {
249    validate_auto_controls(controls, beam.area_pixels)?;
250    let residual = primary_residual(normal)?;
251    let shape = normal.shape();
252    if valid_support.len() != shape[0] * shape[1]
253        || previous.is_some_and(|mask| mask.shape != shape || mask.coordinate != coordinate)
254    {
255        return Err(MaskError::ShapeMismatch);
256    }
257    let inherited =
258        previous.map_or_else(|| vec![false; residual.len()], |mask| mask.support.to_vec());
259    let (noise, peak) = statistics(&residual, valid_support, &inherited, cycle, controls);
260    let thresholds = Thresholds::new(noise, peak, beam.sidelobe_fraction, &controls);
261    let mut evidence = AutoMultithreshEvidence {
262        median: noise.median,
263        robust_rms: noise.rms,
264        peak,
265        positive_threshold: thresholds.positive,
266        low_noise_threshold: thresholds.low,
267        negative_threshold: (controls.negative_factor > 0.0).then_some(thresholds.negative),
268        threshold: thresholds.kind,
269        masked: !cycle.stopped,
270        regions: 0,
271        pruned_regions: 0,
272        grow_regions: 0,
273        grow_pruned_regions: 0,
274        negative_pixels: 0,
275        changed_pixels: 0,
276        channel_stopped: cycle.stopped,
277    };
278    if cycle.completed_major_cycles == 0 && controls.cut_threshold >= 0.2 {
279        tracing::warn!(
280            "Faint regions may not be included in the final mask. Consider decreasing \
281             cutthreshold."
282        );
283    }
284    let added = if cycle.stopped {
285        vec![false; residual.len()]
286    } else {
287        let plane = Plane {
288            residual: &residual,
289            valid: valid_support,
290            shape,
291            beam,
292            controls: &controls,
293        };
294        plane.additions(&inherited, &thresholds, cycle, &mut evidence)
295    };
296    let support = inherited
297        .iter()
298        .zip(&added)
299        .zip(valid_support)
300        .map(|((kept, added), valid)| *valid && (*kept || *added))
301        .collect::<Vec<_>>();
302    evidence.changed_pixels = support
303        .iter()
304        .zip(&inherited)
305        .filter(|(now, before)| now != before)
306        .count();
307    evidence.channel_stopped = skip_channel(&inherited, &support, &thresholds, cycle, &controls);
308    if controls.verbose {
309        log_summary(normal.slab().core_range().start, &evidence);
310    }
311    Ok((
312        ReconstructionMask::new(coordinate, shape, support),
313        evidence,
314    ))
315}

317/// The primary plane's residual, divided by its PSF peak as the minor
318/// cycle divides it: the first polarization of a one-channel state, or
319/// the zeroth Taylor term.
320fn primary_residual(normal: &FinalNormalState) -> Result<Vec<f64>, MaskError> {
321    if normal.channel_count() != 1 {
322        return Err(MaskError::ShapeMismatch);
323    }
324    let window = normal.read_window(normal.slab().core_range())?;
325    let cells = normal.shape()[0] * normal.shape()[1];
326    let (residual, psf) = if window.basis() == NormalBasis::Taylor {
327        (
328            window
329                .coefficient_term(0)
330                .ok_or(MaskError::ShapeMismatch)?
331                .residual()
332                .iter()
333                .map(|value| value.re)
334                .collect::<Vec<_>>(),
335            window
336                .normal_moment(0)
337                .ok_or(MaskError::ShapeMismatch)?
338                .normal_approximation()
339                .iter()
340                .map(|value| value.re)
341                .collect::<Vec<_>>(),
342        )
343    } else {
344        let plane = |values: crate::NormalValues<'_>| {
345            values
346                .slice(0..cells)
347                .map(|values| values.iter().map(|value| value.re).collect::<Vec<_>>())
348                .ok_or(MaskError::ShapeMismatch)
349        };
350        (
351            plane(window.residual())?,
352            plane(window.normal_approximation())?,
353        )
354    };
355    // `FinalNormalState` retains the unnormalised normal-equation planes.
356    // CASA derives automask thresholds from the displayed residual in model
357    // units, so remove the accumulated normalisation before applying
358    // Jy-valued controls.
359    let psf_peak = psf.iter().copied().fold(f64::NEG_INFINITY, f64::max);
360    if !psf_peak.is_finite() || psf_peak <= 0.0 {
361        return Err(MaskError::InvalidBeamArea);
362    }
363    let residual = residual
364        .into_iter()
365        .map(|value| value / psf_peak)
366        .collect::<Vec<_>>();
367    if residual.len() != cells || residual.iter().any(|value| !value.is_finite()) {
368        return Err(MaskError::NonfiniteResidual);
369    }
370    Ok(residual)
371}

373/// The plane's noise and signed absolute peak over its valid pixels
374/// (`SDMaskHandler::autoMask`). The fast estimate is the classical
375/// statistics of every valid pixel; the robust one excludes the inherited
376/// mask, and is zero for a stopped plane, as
377/// `calcRobustImageStatistics` records a flagged channel.
378fn statistics(
379    residual: &[f64],
380    valid: &[bool],
381    inherited: &[bool],
382    cycle: AutoMaskCycle,
383    controls: AutoMultithreshControls,
384) -> (RobustNoise, f64) {
385    let peak = residual
386        .iter()
387        .zip(valid)
388        .filter(|(_, valid)| **valid)
389        .map(|(value, _)| *value)
390        .fold(0.0_f64, |peak, value| {
391            if value.abs() > peak.abs() {
392                value
393            } else {
394                peak
395            }
396        });
397    let zero = RobustNoise {
398        median: 0.0,
399        rms: 0.0,
400    };
401    let noise = if controls.noise == NoiseEstimate::Robust && cycle.stopped {
402        zero
403    } else {
404        let full = inherited.iter().all(|set| *set);
405        plane_noise(residual, valid, controls.noise, Some(inherited), full).unwrap_or(zero)
406    };
407    (noise, peak)
408}

410/// One plane's residual and what its mask is made with.
411struct Plane<'a> {
412    residual: &'a [f64],
413    valid: &'a [bool],
414    shape: [usize; 2],
415    beam: AutoMaskBeam,
416    controls: &'a AutoMultithreshControls,
417}
418
419impl Plane<'_> {
420    /// What one boundary adds to the inherited mask: the threshold mask,
421    /// the grown mask and the negative mask, each smoothed and cut.
422    fn additions(
423        &self,
424        inherited: &[bool],
425        thresholds: &Thresholds,
426        cycle: AutoMaskCycle,
427        evidence: &mut AutoMultithreshEvidence,
428    ) -> Vec<bool> {
429        let controls = self.controls;
430        let prunes = controls.minimum_beam_fraction > 0.0;
431        let prune_size = controls.minimum_beam_fraction * self.beam.area_pixels;
432        let mut threshold = threshold_mask(self.residual, thresholds.positive);
433        if prunes {
434            // Only the pruning branch drops the residual's masked pixels.
435            for (set, valid) in threshold.iter_mut().zip(self.valid) {
436                *set &= *valid;
437            }
438            (evidence.regions, evidence.pruned_regions) =
439                prune_regions(&mut threshold, self.shape, prune_size);
440        }
441        let mut added = self.smooth(&threshold);
442        if cycle.completed_major_cycles > 0 && controls.grow_iterations > 0 {
443            let constraint = threshold_mask(self.residual, thresholds.low);
444            let mut grown = dilate(inherited, &constraint, self.shape, controls.grow_iterations);
445            for (set, allowed) in grown.iter_mut().zip(&constraint) {
446                *set &= *allowed;
447            }
448            if prunes && controls.grow_prune {
449                (evidence.grow_regions, evidence.grow_pruned_regions) =
450                    prune_regions(&mut grown, self.shape, prune_size);
451            }
452            for (set, grown) in added.iter_mut().zip(self.smooth(&grown)) {
453                *set |= grown;
454            }
455        }
456        if cycle.completed_major_cycles == 0 && !added.iter().any(|set| *set) {
457            tracing::warn!(
458                "No regions found by automasking: used the {} threshold to mask, but no \
459                 emission was found.",
460                thresholds.kind.name()
461            );
462        }
463        if controls.negative_factor > 0.0 {
464            let negative = self.smooth(&threshold_mask(self.residual, thresholds.negative));
465            evidence.negative_pixels = negative.iter().filter(|set| **set).count();
466            if evidence.negative_pixels == 0 {
467                tracing::info!("No negative region was found by automask.");
468            }
469            for (set, negative) in added.iter_mut().zip(negative) {
470                *set |= negative;
471            }
472        }
473        added
474    }
475
476    /// `convolveMask` by `smoothfactor` beams, then CASA's cut of the
477    /// smoothed mask at `cutthreshold` of its peak.
478    fn smooth(&self, mask: &[bool]) -> Vec<bool> {
479        smooth_and_cut(
480            mask,
481            self.shape,
482            self.controls.smooth_factor,
483            self.beam,
484            self.controls.cut_threshold,
485        )
486    }
487}

489/// `SDMaskHandler::makeMaskByPerChanThreshold`: the pixels above a
490/// non-negative threshold, or below a negative one.
491fn threshold_mask(residual: &[f64], threshold: f64) -> Vec<bool> {
492    if threshold < 0.0 {
493        residual.iter().map(|value| *value < threshold).collect()
494    } else {
495        residual.iter().map(|value| *value > threshold).collect()
496    }
497}

499/// `SDMaskHandler::skipChannels`: a noise-thresholded plane stops evolving
500/// once its mask is empty, or once the global threshold is reached and its
501/// mask changed by nothing or by less than `minpercentchange` percent.
502fn skip_channel(
503    inherited: &[bool],
504    current: &[bool],
505    thresholds: &Thresholds,
506    cycle: AutoMaskCycle,
507    controls: &AutoMultithreshControls,
508) -> bool {
509    if cycle.stopped || thresholds.kind != AutoMaskThreshold::Noise {
510        return cycle.stopped;
511    }
512    let count = |mask: &[bool]| mask.iter().filter(|set| **set).count() as f64;
513    let (before, now) = (count(inherited), count(current));
514    let fraction = controls.minimum_percent_change / 100.0;
515    let change = (now - before).abs();
516    now == 0.0
517        || (fraction >= 0.0
518            && cycle.threshold_reached
519            && (change == 0.0 || change < fraction * before))
520}

522/// CASA's `printAutomaskSummary` row of one plane, through `tracing`.
523fn log_summary(channel: usize, evidence: &AutoMultithreshEvidence) {
524    let masked = !evidence.channel_stopped;
525    let (median, rms, threshold) = if masked {
526        (
527            evidence.median.to_string(),
528            evidence.robust_rms.to_string(),
529            evidence.positive_threshold.to_string(),
530        )
531    } else {
532        ("--".to_string(), "--".to_string(), "--".to_string())
533    };
534    tracing::info!("========== automask summary ==========");
535    tracing::info!(
536        "chan masking? median   RMS         peak   thresh_type   thresh_value   N_reg N_pruned \
537         N_grow N_grow_pruned N_neg_pix"
538    );
539    tracing::info!(
540        "[C{channel}] {}        {median}  {rms}  {}  {}  {threshold}        {}  {}  {}  {}  {}",
541        if masked { "T" } else { "F" },
542        evidence.peak,
543        evidence.threshold.name(),
544        evidence.regions,
545        evidence.pruned_regions,
546        evidence.grow_regions,
547        evidence.grow_pruned_regions,
548        evidence.negative_pixels,
549    );
550    tracing::info!("========== END of automask summary ==========");
551}

553fn validate_auto_controls(
554    controls: AutoMultithreshControls,
555    beam_area_pixels: f64,
556) -> Result<(), MaskError> {
557    let nonnegative = [
558        controls.sidelobe_factor,
559        controls.noise_factor,
560        controls.low_noise_factor,
561        controls.negative_factor,
562        beam_area_pixels,
563    ];
564    if nonnegative
565        .iter()
566        .any(|value| !value.is_finite() || *value < 0.0)
567        || [
568            controls.minimum_beam_fraction,
569            controls.cut_threshold,
570            controls.minimum_percent_change,
571        ]
572        .iter()
573        .any(|value| !value.is_finite())
574        || !(controls.smooth_factor.is_finite() && controls.smooth_factor > 0.0)
575        || beam_area_pixels == 0.0
576    {
577        Err(MaskError::InvalidAutoControls)
578    } else {
579        Ok(())
580    }
581}

583/// `SDMaskHandler::YAPruneRegions`: remove the four-connected regions
584/// smaller than `size` pixels; the region and pruned-region counts.
585fn prune_regions(mask: &mut [bool], shape: [usize; 2], size: f64) -> (usize, usize) {
586    if size == 0.0 {
587        return (0, 0);
588    }
589    let mut seen = vec![false; mask.len()];
590    let (mut regions, mut pruned) = (0, 0);
591    for seed in 0..mask.len() {
592        if !mask[seed] || seen[seed] {
593            continue;
594        }
595        let mut queue = VecDeque::from([seed]);
596        let mut component = Vec::new();
597        seen[seed] = true;
598        while let Some(index) = queue.pop_front() {
599            component.push(index);
600            let pixel = [index / shape[1], index % shape[1]];
601            for offset in [[-1, 0], [1, 0], [0, -1], [0, 1]] {
602                if let Some(neighbor) = offset_mask_pixel(pixel, offset, shape) {
603                    let flat = neighbor[0] * shape[1] + neighbor[1];
604                    if mask[flat] && !seen[flat] {
605                        seen[flat] = true;
606                        queue.push_back(flat);
607                    }
608                }
609            }
610        }
611        regions += 1;
612        if (component.len() as f64) < size {
613            pruned += 1;
614            for index in component {
615                mask[index] = false;
616            }
617        }
618    }
619    (regions, pruned)
620}

622/// `SDMaskHandler::binaryDilation` with CASA's cross structuring element:
623/// up to `iterations` dilations of `mask`, each seeded only by the mask's
624/// pixels inside `constraint`, stopping early once one changes nothing. A
625/// constraint with no pixel leaves the mask as it is
626/// (`binaryDilationCore`).
627fn dilate(mask: &[bool], constraint: &[bool], shape: [usize; 2], iterations: usize) -> Vec<bool> {
628    let mut current = mask.to_vec();
629    if !mask.iter().any(|set| *set) || !constraint.iter().any(|set| *set) {
630        return current;
631    }
632    for _ in 0..iterations {
633        let mut next = current.clone();
634        for index in 0..current.len() {
635            if !(current[index] && constraint[index]) {
636                continue;
637            }
638            let pixel = [index / shape[1], index % shape[1]];
639            for offset in [[-1, 0], [1, 0], [0, -1], [0, 1]] {
640                if let Some(neighbor) = offset_mask_pixel(pixel, offset, shape) {
641                    next[neighbor[0] * shape[1] + neighbor[1]] = true;
642                }
643            }
644        }
645        let changed = next != current;
646        current = next;
647        if !changed {
648            break;
649        }
650    }
651    current
652}

654/// `SDMaskHandler::convolveMask` by the beam scaled by `smooth_factor`,
655/// then the cut at `cut` times the smoothed peak.
656fn smooth_and_cut(
657    mask: &[bool],
658    shape: [usize; 2],
659    smooth_factor: f64,
660    beam: AutoMaskBeam,
661    cut: f64,
662) -> Vec<bool> {
663    if smooth_factor == 0.0 || mask.iter().all(|value| !*value) {
664        return mask.to_vec();
665    }
666    // CASA cuts with `makeMaskByPerChanThreshold`, so a negative
667    // `cutthreshold` keeps the pixels below it: none.
668    let smoothed = smooth(mask, shape, smooth_factor, beam);
669    let peak = smoothed.iter().copied().fold(0.0_f64, f64::max);
670    threshold_mask(&smoothed, cut * peak)
671}

673/// The mask convolved with the unnormalized elliptical Gaussian of the
674/// beam scaled by `smooth_factor`.
675fn smooth(mask: &[bool], shape: [usize; 2], smooth_factor: f64, beam: AutoMaskBeam) -> Vec<f64> {
676    let sigma_major = smooth_factor * beam.major_fwhm_pixels / 2.354_820_045_030_949_3;
677    let sigma_minor = smooth_factor * beam.minor_fwhm_pixels / 2.354_820_045_030_949_3;
678    // Image2DConvolver::_shapeOfKernel uses a square +/-5-sigma Gaussian
679    // kernel (`_sizeOfGaussian(width, 5.0)`) and then forces an odd shape.
680    // An offset beyond the image's extent on an axis reaches none of its
681    // pixels, so each axis's offsets stop there: a beam wider than the
682    // image (a PSF fitted to few short baselines) would otherwise ask for
683    // a kernel of its own size to the same sums.
684    let reach = (5.0 * sigma_major + 0.5) as isize + 1;
685    let extent = |axis: usize| isize::try_from(shape[axis]).unwrap_or(isize::MAX) - 1;
686    let [reach_x, reach_y] = [reach.min(extent(0)), reach.min(extent(1))];
687    let cos_pa = beam.position_angle_rad.cos();
688    let sin_pa = beam.position_angle_rad.sin();
689    // Each output pixel sums the kernel at its offsets to the mask's set
690    // pixels; scattering from the set pixels visits only those. The kernel
691    // is tabulated one offset row at a time, so the smoothing holds one
692    // row ([`auto_multithresh_bytes`] charges it)
693    // beside its output, not the whole (2r+1)^2 table.
694    let mut smoothed = vec![0.0; mask.len()];
695    let mut row = vec![0.0; 2 * reach_y.unsigned_abs() + 1];
696    for dx in -reach_x..=reach_x {
697        for (value, dy) in row.iter_mut().zip(-reach_y..=reach_y) {
698            let (dx, dy) = (dx as f64, dy as f64);
699            let u = dx * cos_pa + dy * sin_pa;
700            let v = -dx * sin_pa + dy * cos_pa;
701            *value = (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2))).exp();
702        }
703        for (index, _) in mask.iter().enumerate().filter(|(_, set)| **set) {
704            let pixel = [index / shape[1], index % shape[1]];
705            for (value, dy) in row.iter().zip(-reach_y..=reach_y) {
706                if let Some(target) = offset_mask_pixel(pixel, [-dx, -dy], shape) {
707                    smoothed[target[0] * shape[1] + target[1]] += value;
708                }
709            }
710        }
711    }
712    smoothed
713}
