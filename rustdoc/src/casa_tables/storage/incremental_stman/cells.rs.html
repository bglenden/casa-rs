2//! The bytes of one IncrementalStMan cell, a scalar or a fixed-shape array,
3//! as casacore's `ISMColumn` writes and reads them.
4//!
5//! A cell of `nrelem` elements is its elements in canonical form, in storage
6//! (Fortran) order: Bools bit-packed (`Conversion::boolToBit`), other
7//! fixed-width types at their canonical size in the file's byte order, and
8//! strings as `ISMColumn::fromString` lays them out. A scalar is a cell of
9//! one element, so a one-element array is stored exactly as a scalar of its
10//! type.

19/// One cell as the encoder takes it.
20#[derive(Clone, Copy)]
21pub(super) enum Cell<'a> {
22    Scalar(&'a ScalarValue),
23    Array(&'a ArrayValue),
24    /// No value: the cell encodes as zeros and empty strings.
25    Undefined,
26}
27
28impl<'a> From<Option<&'a Value>> for Cell<'a> {
29    fn from(value: Option<&'a Value>) -> Self {
30        match value {
31            Some(Value::Scalar(value)) => Self::Scalar(value),
32            Some(Value::Array(value)) => Self::Array(value),
33            _ => Self::Undefined,
34        }
35    }
36}
37
38impl<'a> From<Option<&'a ScalarValue>> for Cell<'a> {
39    fn from(value: Option<&'a ScalarValue>) -> Self {
40        value.map_or(Self::Undefined, Self::Scalar)
41    }
42}

44/// An element type of ISM cells: where its elements are in a scalar or an
45/// array of the type.
46trait Element: Sized {
47    fn of_scalar(value: &ScalarValue) -> Option<&Self>;
48    fn of_array(value: &ArrayValue) -> Option<&ArrayD<Self>>;
49}
50
51/// A fixed-width element type: its canonical size and encoding.
52trait Canonical: Element + Copy {
53    const BYTES: usize;
54    fn put(self, dst: &mut [u8], big_endian: bool);
55    fn get(src: &[u8], big_endian: bool) -> Self;
56}
57
58macro_rules! element {
59    ($type:ty, $variant:ident) => {
60        impl Element for $type {
61            fn of_scalar(value: &ScalarValue) -> Option<&Self> {
62                match value {
63                    ScalarValue::$variant(value) => Some(value),
64                    _ => None,
65                }
66            }
67
68            fn of_array(value: &ArrayValue) -> Option<&ArrayD<Self>> {
69                match value {
70                    ArrayValue::$variant(value) => Some(value),
71                    _ => None,
72                }
73            }
74        }
75    };
76}

78/// A fixed-width element type encoded by the primitive's own byte
79/// conversions.
80macro_rules! primitive {
81    ($type:ty, $variant:ident) => {
82        element!($type, $variant);
83
84        impl Canonical for $type {
85            const BYTES: usize = size_of::<$type>();
86
87            fn put(self, dst: &mut [u8], big_endian: bool) {
88                let bytes = if big_endian {
89                    self.to_be_bytes()
90                } else {
91                    self.to_le_bytes()
92                };
93                dst[..Self::BYTES].copy_from_slice(&bytes);
94            }
95
96            fn get(src: &[u8], big_endian: bool) -> Self {
97                let bytes = src[..Self::BYTES].try_into().expect("element bytes");
98                if big_endian {
99                    Self::from_be_bytes(bytes)
100                } else {
101                    Self::from_le_bytes(bytes)
102                }
103            }
104        }
105    };
106}

108/// A complex element type: its real part, then its imaginary part.
109macro_rules! complex {
110    ($type:ty, $variant:ident, $part:ty) => {
111        element!($type, $variant);
112
113        impl Canonical for $type {
114            const BYTES: usize = 2 * size_of::<$part>();
115
116            fn put(self, dst: &mut [u8], big_endian: bool) {
117                self.re.put(dst, big_endian);
118                self.im.put(&mut dst[size_of::<$part>()..], big_endian);
119            }
120
121            fn get(src: &[u8], big_endian: bool) -> Self {
122                Self::new(
123                    <$part>::get(src, big_endian),
124                    <$part>::get(&src[size_of::<$part>()..], big_endian),
125                )
126            }
127        }
128    };
129}

144/// Call `put(index, element)` for the first `nrelem` elements of `cell` of
145/// type `T`, in storage order: a scalar is the element of a one-element
146/// cell, an array gives its elements in Fortran order, and an undefined or
147/// mistyped cell gives none.
148fn for_each_element<'a, T: Element + 'a>(
149    cell: Cell<'a>,
150    nrelem: usize,
151    mut put: impl FnMut(usize, &'a T),
152) {
153    match cell {
154        Cell::Scalar(value) if nrelem == 1 => {
155            if let Some(element) = T::of_scalar(value) {
156                put(0, element);
157            }
158        }
159        Cell::Array(value) => {
160            if let Some(array) = T::of_array(value) {
161                for (index, element) in array.t().into_iter().take(nrelem).enumerate() {
162                    put(index, element);
163                }
164            }
165        }
166        _ => {}
167    }
168}

170/// The canonical bytes of `cell` in a column of `nrelem` elements of type
171/// `data_type`; the elements it does not give are zeros or empty strings.
172pub(super) fn encode_cell(
173    cell: Cell<'_>,
174    data_type: CasacoreDataType,
175    nrelem: usize,
176    big_endian: bool,
177) -> Vec<u8> {
178    match data_type {
179        CasacoreDataType::TpBool => {
180            let mut bytes = vec![0u8; nrelem.div_ceil(8)];
181            for_each_element::<bool>(cell, nrelem, |index, &element| {
182                if element {
183                    bytes[index / 8] |= 1 << (index % 8);
184                }
185            });
186            bytes
187        }
188        CasacoreDataType::TpString => encode_strings(cell, nrelem, big_endian),
189        CasacoreDataType::TpUChar => encode_fixed::<u8>(cell, nrelem, big_endian),
190        CasacoreDataType::TpShort => encode_fixed::<i16>(cell, nrelem, big_endian),
191        CasacoreDataType::TpUShort => encode_fixed::<u16>(cell, nrelem, big_endian),
192        CasacoreDataType::TpInt => encode_fixed::<i32>(cell, nrelem, big_endian),
193        CasacoreDataType::TpUInt => encode_fixed::<u32>(cell, nrelem, big_endian),
194        CasacoreDataType::TpInt64 => encode_fixed::<i64>(cell, nrelem, big_endian),
195        CasacoreDataType::TpFloat => encode_fixed::<f32>(cell, nrelem, big_endian),
196        CasacoreDataType::TpDouble => encode_fixed::<f64>(cell, nrelem, big_endian),
197        CasacoreDataType::TpComplex => encode_fixed::<Complex32>(cell, nrelem, big_endian),
198        CasacoreDataType::TpDComplex => encode_fixed::<Complex64>(cell, nrelem, big_endian),
199        _ => vec![0u8; ism_element_size(data_type) * nrelem],
200    }
201}
202
203fn encode_fixed<T: Canonical>(cell: Cell<'_>, nrelem: usize, big_endian: bool) -> Vec<u8> {
204    let mut bytes = vec![0u8; T::BYTES * nrelem];
205    for_each_element::<T>(cell, nrelem, |index, &element| {
206        element.put(&mut bytes[index * T::BYTES..], big_endian);
207    });
208    bytes
209}

211/// Strings as `ISMColumn::fromString` writes them: a length that includes
212/// itself, then each string. One string is its bytes after the length
213/// (`[4 + len][bytes]`, the layout of a scalar); several are each preceded
214/// by their own length.
215fn encode_strings(cell: Cell<'_>, nrelem: usize, big_endian: bool) -> Vec<u8> {
216    let length = |length: usize| {
217        let length = length as u32;
218        if big_endian {
219            length.to_be_bytes()
220        } else {
221            length.to_le_bytes()
222        }
223    };
224    if nrelem == 1 {
225        let mut string = "";
226        for_each_element::<String>(cell, 1, |_, element| string = element.as_str());
227        let mut bytes = Vec::with_capacity(4 + string.len());
228        bytes.extend_from_slice(&length(4 + string.len()));
229        bytes.extend_from_slice(string.as_bytes());
230        return bytes;
231    }
232    let mut strings = vec![""; nrelem];
233    for_each_element::<String>(cell, nrelem, |index, element| {
234        strings[index] = element.as_str();
235    });
236    let total = 4 + strings.iter().map(|string| 4 + string.len()).sum::<usize>();
237    let mut bytes = Vec::with_capacity(total);
238    bytes.extend_from_slice(&length(total));
239    for string in strings {
240        bytes.extend_from_slice(&length(string.len()));
241        bytes.extend_from_slice(string.as_bytes());
242    }
243    bytes
244}

246/// The fixed-shape array cell of `nrelem` elements of type `data_type` and
247/// shape `shape` at the start of `data`.
248pub(super) fn decode_array_cell(
249    data: &[u8],
250    data_type: CasacoreDataType,
251    nrelem: usize,
252    shape: &[i32],
253    big_endian: bool,
254) -> Result<Value, StorageError> {
255    let shape = shape.iter().map(|&dim| dim as usize).collect::<Vec<_>>();
256    fn shaped<T>(shape: &[usize], elements: Vec<T>) -> Result<ArrayD<T>, StorageError> {
257        ArrayD::from_shape_vec(IxDyn(shape).f(), elements)
258            .map_err(|err| StorageError::FormatMismatch(format!("array shape: {err}")))
259    }
260    fn fixed<T: Canonical>(data: &[u8], nrelem: usize, big_endian: bool) -> Vec<T> {
261        data[..nrelem * T::BYTES]
262            .chunks_exact(T::BYTES)
263            .map(|bytes| T::get(bytes, big_endian))
264            .collect()
265    }
266    let array = match data_type {
267        CasacoreDataType::TpBool => ArrayValue::Bool(shaped(
268            &shape,
269            (0..nrelem)
270                .map(|index| (data[index / 8] >> (index % 8)) & 1 != 0)
271                .collect(),
272        )?),
273        CasacoreDataType::TpString => {
274            ArrayValue::String(shaped(&shape, read_ism_string(data, nrelem, big_endian)?)?)
275        }
276        CasacoreDataType::TpUChar => {
277            ArrayValue::UInt8(shaped(&shape, fixed(data, nrelem, big_endian))?)
278        }
279        CasacoreDataType::TpShort => {
280            ArrayValue::Int16(shaped(&shape, fixed(data, nrelem, big_endian))?)
281        }
282        CasacoreDataType::TpUShort => {
283            ArrayValue::UInt16(shaped(&shape, fixed(data, nrelem, big_endian))?)
284        }
285        CasacoreDataType::TpInt => {
286            ArrayValue::Int32(shaped(&shape, fixed(data, nrelem, big_endian))?)
287        }
288        CasacoreDataType::TpUInt => {
289            ArrayValue::UInt32(shaped(&shape, fixed(data, nrelem, big_endian))?)
290        }
291        CasacoreDataType::TpInt64 => {
292            ArrayValue::Int64(shaped(&shape, fixed(data, nrelem, big_endian))?)
293        }
294        CasacoreDataType::TpFloat => {
295            ArrayValue::Float32(shaped(&shape, fixed(data, nrelem, big_endian))?)
296        }
297        CasacoreDataType::TpDouble => {
298            ArrayValue::Float64(shaped(&shape, fixed(data, nrelem, big_endian))?)
299        }
300        CasacoreDataType::TpComplex => {
301            ArrayValue::Complex32(shaped(&shape, fixed(data, nrelem, big_endian))?)
302        }
303        CasacoreDataType::TpDComplex => {
304            ArrayValue::Complex64(shaped(&shape, fixed(data, nrelem, big_endian))?)
305        }
306        _ => {
307            return Err(StorageError::FormatMismatch(format!(
308                "unsupported ISM array type: {data_type:?}"
309            )));
310        }
311    };
312    Ok(Value::Array(array))
313}

315/// Array cells of `nrelem` elements of type `data_type`, flattened in
316/// Fortran order, a mistyped or non-array cell as `nrelem` zeros or empty
317/// strings.
318pub(super) fn flatten_array_cells(
319    values: &[Value],
320    data_type: CasacoreDataType,
321    nrelem: usize,
322) -> Result<ColumnRawData, StorageError> {
323    fn flatten<T: Element + Clone + Default>(values: &[Value], nrelem: usize) -> Vec<T> {
324        let mut flat = Vec::with_capacity(values.len() * nrelem);
325        for value in values {
326            if let Value::Array(array) = value
327                && let Some(array) = T::of_array(array)
328            {
329                flat.extend(array.t().iter().cloned());
330            } else {
331                flat.extend(repeat_n(T::default(), nrelem));
332            }
333        }
334        flat
335    }
336    Ok(match data_type {
337        CasacoreDataType::TpBool => ColumnRawData::Bool(flatten(values, nrelem)),
338        CasacoreDataType::TpUChar => ColumnRawData::UInt8(flatten(values, nrelem)),
339        CasacoreDataType::TpShort => ColumnRawData::Int16(flatten(values, nrelem)),
340        CasacoreDataType::TpUShort => ColumnRawData::UInt16(flatten(values, nrelem)),
341        CasacoreDataType::TpInt => ColumnRawData::Int32(flatten(values, nrelem)),
342        CasacoreDataType::TpUInt => ColumnRawData::UInt32(flatten(values, nrelem)),
343        CasacoreDataType::TpInt64 => ColumnRawData::Int64(flatten(values, nrelem)),
344        CasacoreDataType::TpFloat => ColumnRawData::Float32(flatten(values, nrelem)),
345        CasacoreDataType::TpDouble => ColumnRawData::Float64(flatten(values, nrelem)),
346        CasacoreDataType::TpComplex => ColumnRawData::Complex32(flatten(values, nrelem)),
347        CasacoreDataType::TpDComplex => ColumnRawData::Complex64(flatten(values, nrelem)),
348        CasacoreDataType::TpString => ColumnRawData::String(flatten(values, nrelem)),
349        _ => {
350            return Err(StorageError::FormatMismatch(format!(
351                "unsupported ISM array type: {data_type:?}"
352            )));
353        }
354    })
355}

391    /// The one-element array holding `scalar`.
392    fn singleton(scalar: &ScalarValue) -> ArrayValue {
393        fn one<T: Clone>(value: &T) -> ArrayD<T> {
394            ArrayD::from_elem(IxDyn(&[1]), value.clone())
395        }
396        match scalar {
397            ScalarValue::Bool(value) => ArrayValue::Bool(one(value)),
398            ScalarValue::UInt8(value) => ArrayValue::UInt8(one(value)),
399            ScalarValue::Int16(value) => ArrayValue::Int16(one(value)),
400            ScalarValue::UInt16(value) => ArrayValue::UInt16(one(value)),
401            ScalarValue::Int32(value) => ArrayValue::Int32(one(value)),
402            ScalarValue::UInt32(value) => ArrayValue::UInt32(one(value)),
403            ScalarValue::Int64(value) => ArrayValue::Int64(one(value)),
404            ScalarValue::Float32(value) => ArrayValue::Float32(one(value)),
405            ScalarValue::Float64(value) => ArrayValue::Float64(one(value)),
406            ScalarValue::Complex32(value) => ArrayValue::Complex32(one(value)),
407            ScalarValue::Complex64(value) => ArrayValue::Complex64(one(value)),
408            ScalarValue::String(value) => ArrayValue::String(one(value)),
409        }
410    }

412    #[test]
413    fn a_one_element_array_encodes_as_its_scalar() {
414        for big_endian in [false, true] {
415            for (scalar, data_type) in scalars() {
416                let array = singleton(&scalar);
417                let from_scalar = encode_cell(Cell::Scalar(&scalar), data_type, 1, big_endian);
418                let from_array = encode_cell(Cell::Array(&array), data_type, 1, big_endian);
419                assert_eq!(from_array, from_scalar, "{data_type:?}");
420                assert_ne!(
421                    from_array,
422                    encode_cell(Cell::Undefined, data_type, 1, big_endian),
423                    "{data_type:?} keeps its value"
424                );
425                let decoded =
426                    decode_array_cell(&from_array, data_type, 1, &[1], big_endian).unwrap();
427                assert_eq!(decoded, Value::Array(array), "{data_type:?}");
428            }
429        }
430    }

443    #[test]
444    fn arrays_keep_their_elements_in_storage_order() {
445        let values = (0..6).map(|i| i * 3 + 1).collect::<Vec<i32>>();
446        let ints = ArrayD::from_shape_vec(IxDyn(&[2, 3]), values).unwrap();
447        let strings = ints.mapv(|value| format!("s{value}"));
448        let flags = ints.mapv(|value| value % 2 == 0);
449        for big_endian in [false, true] {
450            for (array, data_type) in [
451                (ArrayValue::Int32(ints.clone()), CasacoreDataType::TpInt),
452                (
453                    ArrayValue::String(strings.clone()),
454                    CasacoreDataType::TpString,
455                ),
456                (ArrayValue::Bool(flags.clone()), CasacoreDataType::TpBool),
457                (
458                    ArrayValue::Int16(ints.mapv(|value| value as i16)),
459                    CasacoreDataType::TpShort,
460                ),
461                (
462                    ArrayValue::UInt16(ints.mapv(|value| value as u16)),
463                    CasacoreDataType::TpUShort,
464                ),
465                (
466                    ArrayValue::UInt32(ints.mapv(|value| value as u32)),
467                    CasacoreDataType::TpUInt,
468                ),
469                (
470                    ArrayValue::Int64(ints.mapv(i64::from)),
471                    CasacoreDataType::TpInt64,
472                ),
473                (
474                    ArrayValue::UInt8(ints.mapv(|value| value as u8)),
475                    CasacoreDataType::TpUChar,
476                ),
477            ] {
478                let encoded = encode_cell(Cell::Array(&array), data_type, 6, big_endian);
479                let decoded =
480                    decode_array_cell(&encoded, data_type, 6, &[2, 3], big_endian).unwrap();
481                assert_eq!(decoded, Value::Array(array), "{data_type:?}");
482            }
483        }
484        // Storage order is Fortran order: the first element, then the one
485        // below it.
486        let encoded = encode_cell(
487            Cell::Array(&ArrayValue::Int32(ints)),
488            CasacoreDataType::TpInt,
489            6,
490            false,
491        );
492        assert_eq!(i32::from_le_bytes(encoded[4..8].try_into().unwrap()), 10);
493    }
