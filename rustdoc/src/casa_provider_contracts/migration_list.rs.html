2//! The list transform of a profile migration,
3//! [`MigrationTransform::TextList`](super::MigrationTransform::TextList)
4//! (#931).
5//!
6//! casa-rs's old calibration surfaces took CASA's lists as text: the
7//! fields of `reference_fields="3C286,J1331"`, a model `smodel="1,0,0,0"`,
8//! and per-table chains whose items are `;`-separated, such as
9//! `spwmap="0,0;1,1"`. CASA takes lists. A migration reads the old text
10//! into the list it meant; the old explicit absence, `"none"` or empty, is
11//! the empty list.
12
13use schemars::JsonSchema;
14use serde::{Deserialize, Serialize};
15
16use super::ParameterValue;
17
18/// How each item of an old text list is read.
19#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
20#[serde(rename_all = "snake_case")]
21pub enum ListElement {
22    /// The item as text, trimmed; an empty item stays, since a per-table
23    /// list is positional (`gainfield="0;;2"`).
24    String,
25    /// The item as a path, trimmed; an empty item is dropped, as the old
26    /// surfaces split a path list given as text (`gaintables="a.cal,b.cal"`).
27    Path,
28    /// The item as an integer.
29    Integer,
30    /// The item as a number.
31    Float,
32    /// The item as a flag: `true`, `t`, `1`, `yes` or `y`, or their
33    /// opposites, as the old surfaces read them.
34    Bool,
35    /// The item as a `,`-separated list of integers, in brackets or not:
36    /// one table's old `spwmap` (`"[0,0,1]"`); an empty item is the empty
37    /// list.
38    IntegerList,
39}
40
41impl ListElement {
42    /// `text` as a list of this element: its `separator`-separated items,
43    /// each read as one. `"none"` or empty text is the empty list.
44    pub fn parse_list(self, text: &str, separator: &str) -> Result<ParameterValue, String> {
45        let text = text.trim();
46        if text.is_empty() || text.eq_ignore_ascii_case("none") {
47            return Ok(ParameterValue::Array(Vec::new()));
48        }
49        text.split(separator)
50            .map(str::trim)
51            .filter(|item| self != Self::Path || !item.is_empty())
52            .map(|item| self.parse_item(item))
53            .collect::<Result<Vec<_>, _>>()
54            .map(ParameterValue::Array)
55    }
56
57    fn parse_item(self, item: &str) -> Result<ParameterValue, String> {
58        let number = |item: &str| {
59            item.parse::<f64>()
60                .map_err(|error| format!("{item:?} is not a number: {error}"))
61        };
62        match self {
63            Self::String | Self::Path => Ok(ParameterValue::String(item.to_string())),
64            Self::Integer => item
65                .parse::<i64>()
66                .map(ParameterValue::Integer)
67                .map_err(|error| format!("{item:?} is not an integer: {error}")),
68            Self::Float => number(item).map(ParameterValue::Float),
69            Self::Bool => match item.to_ascii_lowercase().as_str() {
70                "true" | "t" | "1" | "yes" | "y" => Ok(ParameterValue::Bool(true)),
71                "false" | "f" | "0" | "no" | "n" => Ok(ParameterValue::Bool(false)),
72                _ => Err(format!("{item:?} is not true or false")),
73            },
74            Self::IntegerList => {
75                let inner = item
76                    .strip_prefix('[')
77                    .and_then(|item| item.strip_suffix(']'))
78                    .unwrap_or(item);
79                inner
80                    .split(',')
81                    .map(str::trim)
82                    .filter(|value| !value.is_empty())
83                    .map(|value| Self::Integer.parse_item(value))
84                    .collect::<Result<Vec<_>, _>>()
85                    .map(ParameterValue::Array)
86            }
87        }
88    }
89}

91#[cfg(test)]
92mod tests {
93    use super::*;
94
95    fn strings(values: &[&str]) -> ParameterValue {
96        ParameterValue::Array(
97            values
98                .iter()
99                .map(|value| ParameterValue::String((*value).into()))
100                .collect(),
101        )
102    }
103
104    fn integers(values: &[i64]) -> ParameterValue {
105        ParameterValue::Array(
106            values
107                .iter()
108                .copied()
109                .map(ParameterValue::Integer)
110                .collect(),
111        )
112    }
113
114    #[test]
115    fn old_text_reads_as_the_list_it_meant() {
116        let list = |element: ListElement, text, separator| element.parse_list(text, separator);
117        assert_eq!(
118            list(ListElement::String, "3C286, J1331", ",").unwrap(),
119            strings(&["3C286", "J1331"])
120        );
121        // A per-table list is positional: the empty item stays.
122        assert_eq!(
123            list(ListElement::String, "0;;nearest", ";").unwrap(),
124            strings(&["0", "", "nearest"])
125        );
126        // A path list drops it, as the old surfaces split one.
127        assert_eq!(
128            list(ListElement::Path, "a.cal, ,b.cal", ",").unwrap(),
129            strings(&["a.cal", "b.cal"])
130        );
131        assert_eq!(
132            list(ListElement::Float, "1,0,0,0", ",").unwrap(),
133            ParameterValue::Array(vec![
134                ParameterValue::Float(1.0),
135                ParameterValue::Float(0.0),
136                ParameterValue::Float(0.0),
137                ParameterValue::Float(0.0),
138            ])
139        );
140        assert_eq!(
141            list(ListElement::Bool, "true, f", ",").unwrap(),
142            ParameterValue::Array(vec![
143                ParameterValue::Bool(true),
144                ParameterValue::Bool(false)
145            ])
146        );
147        assert_eq!(
148            list(ListElement::IntegerList, "[0,0,1]; 1,1 ;", ";").unwrap(),
149            ParameterValue::Array(vec![integers(&[0, 0, 1]), integers(&[1, 1]), integers(&[])])
150        );
151        assert_eq!(
152            list(ListElement::Integer, "0, 2", ",").unwrap(),
153            integers(&[0, 2])
154        );
155        for absent in ["none", " ", "None"] {
156            assert_eq!(
157                list(ListElement::IntegerList, absent, ";").unwrap(),
158                ParameterValue::Array(Vec::new())
159            );
160        }
161        assert!(list(ListElement::Integer, "0,x", ",").is_err());
162        assert!(list(ListElement::Bool, "maybe", ",").is_err());
163    }
164}
