14/// Checked numeric-array demand for one planned continuum generation.
15///
16/// This is deliberately non-persistent. It describes the current algorithm
17/// catalog's owned payload, validity, and working arrays so the runtime can
18/// acquire one bounded lease before production starts. Metadata containers,
19/// allocator bookkeeping, and borrowed reconstruction inputs are not charged
20/// here.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub struct ContinuumGenerationDemand {
23    produced_residency_bytes: u64,
24    sealed_residency_bytes: u64,
25    algorithm_scratch_bytes: u64,
26    maximum_member_payload_bytes: u64,
27    maximum_member_validity_bytes: u64,
28    peak_residency_bytes: u64,
29}
30
31impl ContinuumGenerationDemand {
32    /// Numeric payload and validity bytes retained by produced members.
33    #[must_use]
34    pub const fn produced_residency_bytes(self) -> u64 {
35        self.produced_residency_bytes
36    }
37
38    /// Numeric payload and validity bytes retained by sealed members.
39    #[must_use]
40    pub const fn sealed_residency_bytes(self) -> u64 {
41        self.sealed_residency_bytes
42    }
43
44    /// Maximum product-algorithm working-array bytes retained with production.
45    #[must_use]
46    pub const fn algorithm_scratch_bytes(self) -> u64 {
47        self.algorithm_scratch_bytes
48    }
49
50    /// Largest single member's binary32 payload.
51    #[must_use]
52    pub const fn maximum_member_payload_bytes(self) -> u64 {
53        self.maximum_member_payload_bytes
54    }
55
56    /// Largest single member's byte-addressed validity array.
57    #[must_use]
58    pub const fn maximum_member_validity_bytes(self) -> u64 {
59        self.maximum_member_validity_bytes
60    }
61
62    /// Peak owned numeric-array residency across production and authorization.
63    ///
64    /// Production overlaps the produced generation with algorithm scratch;
65    /// authorization overlaps produced and sealed generations after scratch is
66    /// released.
67    #[must_use]
68    pub const fn peak_residency_bytes(self) -> u64 {
69        self.peak_residency_bytes
70    }
71}

73impl PlannedContinuumGeneration {
74    /// Derive the exact current-catalog numeric-array demand for these inputs.
75    ///
76    /// # Errors
77    ///
78    /// Fails closed when the inputs do not name this plan's problem and source
79    /// generations, or when any byte calculation overflows.
80    pub fn demand(
81        &self,
82        inputs: &ContinuumProductInputs<'_>,
83    ) -> Result<ContinuumGenerationDemand, ProductsError> {
84        if inputs.problem().problem_id() != self.problem_id()
85            || inputs.final_model().generation_id() != self.final_model_generation()
86            || inputs
87                .reconstruction_mask()
88                .map(casa_imaging_reconstruction::ReconstructionMask::generation_id)
89                != self.reconstruction_mask_generation()
90            || inputs
91                .coupled_reconstruction_masks()
92                .map(|masks| masks.line().generation_id())
93                != self.line_reconstruction_mask_generation()
94        {
95            return Err(ProductsError::CommitmentMismatch);
96        }
97
98        let mut produced_residency_bytes = 0_u64;
99        let mut maximum_member_payload_bytes = 0_u64;
100        let mut maximum_member_validity_bytes = 0_u64;
101        for member in self.members() {
102            let values = checked_shape_values(member.shape())?;
103            if values != member.payload_values() {
104                return Err(ProductsError::PayloadLengthMismatch {
105                    expected: values,
106                    actual: member.payload_values(),
107                });
108            }
109            let payload = bytes_for::<f32>(values, "member payload")?;
110            let validity = bytes_for::<bool>(values, "member validity")?;
111            produced_residency_bytes = checked_add(
112                produced_residency_bytes,
113                checked_add(payload, validity, "member payload plus validity")?,
114                "produced generation residency",
115            )?;
116            maximum_member_payload_bytes = maximum_member_payload_bytes.max(payload);
117            maximum_member_validity_bytes = maximum_member_validity_bytes.max(validity);
118        }
119
120        let algorithm_scratch_bytes = match inputs.normal_state().catalog() {
121            NormalStateCatalog::UnnormalizedTaylorBlockV1 => taylor_scratch_bytes(inputs)?,
122            NormalStateCatalog::UnnormalizedJointBlockV1 => generic_scratch_bytes(self, inputs)?,
123            NormalStateCatalog::UnnormalizedPlaneV1
124            | NormalStateCatalog::UnnormalizedChannelSlabV1 => generic_scratch_bytes(self, inputs)?,
125        };
126        let production_peak = checked_add(
127            produced_residency_bytes,
128            algorithm_scratch_bytes,
129            "product production peak residency",
130        )?;
131        let authorization_peak = checked_add(
132            produced_residency_bytes,
133            produced_residency_bytes,
134            "product authorization peak residency",
135        )?;
136        Ok(ContinuumGenerationDemand {
137            produced_residency_bytes,
138            sealed_residency_bytes: produced_residency_bytes,
139            algorithm_scratch_bytes,
140            maximum_member_payload_bytes,
141            maximum_member_validity_bytes,
142            peak_residency_bytes: production_peak.max(authorization_peak),
143        })
144    }
145}

147fn generic_scratch_bytes(
148    planned: &PlannedContinuumGeneration,
149    inputs: &ContinuumProductInputs<'_>,
150) -> Result<u64, ProductsError> {
151    let shape = inputs.normal_state().shape();
152    let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
153    let plane = bytes_for::<f32>(cells, "generic plane")?;
154    let validity = bytes_for::<bool>(cells, "generic validity plane")?;
155    let requires_validity = planned
156        .members()
157        .iter()
158        .any(|member| member.validity() != ProductValidityRule::All);
159    let mut scratch = if requires_validity { validity } else { 0 };
160
161    // Normalization overlaps one converted source plane with its result.
162    scratch = scratch.max(checked_mul(plane, 2, "generic converted plane pair")?);
163    if planned
164        .members()
165        .iter()
166        .any(|member| matches!(member.role(), ProductRole::RestoredImage(_)))
167    {
168        // Restoration retains its restored result and normalized residual while
169        // a Gaussian kernel and one exact rustfft convolution workspace live.
170        scratch = scratch.max(checked_add(
171            checked_mul(plane, 3, "generic restoration planes")?,
172            fft_convolution_workspace_bytes(shape)?,
173            "generic restoration scratch",
174        )?);
175    }
176    Ok(scratch)
177}

179fn taylor_scratch_bytes(inputs: &ContinuumProductInputs<'_>) -> Result<u64, ProductsError> {
180    let state = inputs.normal_state();
181    let shape = state.shape();
182    let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
183    let terms = u64::try_from(state.coefficient_term_count())
184        .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor term count"))?;
185    let moments = u64::try_from(state.normal_moment_count())
186        .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor moment count"))?;
187    if terms < 2 || moments != terms.saturating_mul(2).saturating_sub(1) {
188        return Err(ProductsError::SourceLineageMismatch);
189    }
190    let cells = u64::try_from(cells)
191        .map_err(|_| ProductsError::ResourceDemandOverflow("Taylor cell count"))?;
192
193    // TaylorProducts retains PSF and weight moments; residual, model,
194    // restored, PB and PB-corrected term families; alpha, alpha-error and mask;
195    // two validity planes; and scalar sum weights until every planned member
196    // has been copied into the produced generation.
197    let retained_f32_planes = checked_add(
198        checked_mul(moments, 2, "Taylor PSF and weight planes")?,
199        checked_add(
200            checked_mul(terms, 5, "Taylor term families")?,
201            3,
202            "Taylor scalar product planes",
203        )?,
204        "Taylor retained plane count",
205    )?;
206    let retained = checked_add(
207        checked_mul(
208            checked_mul(retained_f32_planes, cells, "Taylor retained values")?,
209            size_of::<f32>() as u64,
210            "Taylor retained payload bytes",
211        )?,
212        checked_add(
213            checked_mul(
214                checked_mul(2, cells, "Taylor validity values")?,
215                size_of::<bool>() as u64,
216                "Taylor validity bytes",
217            )?,
218            checked_mul(moments, size_of::<f32>() as u64, "Taylor sum weights")?,
219            "Taylor validity and sum-weight bytes",
220        )?,
221        "Taylor retained scratch",
222    )?;
223
224    // Construction additionally retains the principal residual family and PB0
225    // source plane. The normal solve owns three dense term matrices plus two
226    // term vectors at its peak. This intentionally follows the product owner's
227    // concrete current data structures, rather than a runtime approximation.
228    let transient_planes = checked_mul(
229        checked_add(terms, 1, "Taylor transient plane count")?,
230        checked_mul(
231            cells,
232            size_of::<f32>() as u64,
233            "Taylor transient plane bytes",
234        )?,
235        "Taylor transient planes",
236    )?;
237    let term_square = checked_mul(terms, terms, "Taylor normal matrix values")?;
238    let solve = checked_mul(
239        checked_add(
240            checked_mul(term_square, 3, "Taylor normal matrices")?,
241            checked_mul(terms, 2, "Taylor solve vectors")?,
242            "Taylor solve values",
243        )?,
244        size_of::<f64>() as u64,
245        "Taylor solve bytes",
246    )?;
247    let mut total = checked_add(
248        retained,
249        checked_add(transient_planes, solve, "Taylor transient scratch")?,
250        "Taylor algorithm scratch",
251    )?;
252    if inputs.problem().products().restoring_beam() != RestoringBeamPolicy::None {
253        total = checked_add(
254            total,
255            checked_add(
256                checked_mul(cells, size_of::<f32>() as u64, "Taylor restoring kernel")?,
257                fft_convolution_workspace_bytes(shape)?,
258                "Taylor convolution scratch",
259            )?,
260            "Taylor restoration scratch",
261        )?;
262    }
263    Ok(total)
264}

266fn fft_convolution_workspace_bytes(shape: [usize; 2]) -> Result<u64, ProductsError> {
267    let cells = checked_shape_values([shape[0], shape[1], 1, 1])?;
268    let complex_planes = bytes_for::<Complex64>(
269        cells
270            .checked_mul(2)
271            .ok_or(ProductsError::ResourceDemandOverflow("FFT complex planes"))?,
272        "FFT complex planes",
273    )?;
274    let output = bytes_for::<f32>(cells, "FFT output plane")?;
275    let mut planner = FftPlanner::<f64>::new();
276    let mut lane_values = 0usize;
277    for length in shape {
278        for plan in [
279            planner.plan_fft_forward(length),
280            planner.plan_fft_inverse(length),
281        ] {
282            lane_values = lane_values.max(
283                length
284                    .checked_add(plan.get_inplace_scratch_len())
285                    .ok_or(ProductsError::ResourceDemandOverflow("FFT lane scratch"))?,
286            );
287        }
288    }
289    checked_add(
290        checked_add(complex_planes, output, "FFT planes and output")?,
291        bytes_for::<Complex64>(lane_values, "FFT lane and scratch")?,
292        "FFT convolution workspace",
293    )
294}

296fn checked_shape_values(shape: [usize; 4]) -> Result<usize, ProductsError> {
297    shape.into_iter().try_fold(1usize, |values, extent| {
298        values
299            .checked_mul(extent)
300            .ok_or(ProductsError::ResourceDemandOverflow("member shape"))
301    })
302}
303
304fn bytes_for<T>(count: usize, what: &'static str) -> Result<u64, ProductsError> {
305    let count = u64::try_from(count).map_err(|_| ProductsError::ResourceDemandOverflow(what))?;
306    checked_mul(count, size_of::<T>() as u64, what)
307}
308
309fn checked_add(left: u64, right: u64, what: &'static str) -> Result<u64, ProductsError> {
310    left.checked_add(right)
311        .ok_or(ProductsError::ResourceDemandOverflow(what))
312}
313
314fn checked_mul(left: u64, right: u64, what: &'static str) -> Result<u64, ProductsError> {
315    left.checked_mul(right)
316        .ok_or(ProductsError::ResourceDemandOverflow(what))
317}
