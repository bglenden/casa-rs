3//! Use the already open stores to grid the separately centred mosaic PSF.
4
5use std::sync::Arc;
6
7use casa_core::measures::MeasuresProvider;
8use casa_imaging_model::{CompiledProblem, GeometryPlan};
9use casa_imaging_runtime::{Admitted, RunSummary, run_phase};
10
11use super::NativeError;
12use crate::ApplicationRuntime;
13use crate::compile::resolve_partitions;
14use crate::imaging::{self, BatchRoute, Batching, PartitionSelection, SeparatePsf, StoreBatches};
15
16/// Temporary geometry and admission for a second view of the same rows.
17/// The stores, resolved selection and optional update sessions are shared.
18pub(super) struct PsfRoute {
19    problem: CompiledProblem,
20    geometry: Admitted<(Vec<StoreView>, Batching)>,
21}
22
23struct StoreView {
24    geometry: GeometryPlan,
25    selection: Option<PartitionSelection>,
26}
27
28impl PsfRoute {
29    pub(super) fn new(
30        problem: CompiledProblem,
31        measures: Arc<dyn MeasuresProvider>,
32        (main, route): (&CompiledProblem, &BatchRoute),
33        runtime: &ApplicationRuntime,
34    ) -> Result<Self, NativeError> {
35        let shared_selection = std::ptr::eq(
36            problem.observation().sources(),
37            main.observation().sources(),
38        );
39        let plans = route
40            .stores
41            .iter()
42            .enumerate()
43            .map(|(source, store)| {
44                let observation = store.observation();
45                let geometry = GeometryPlan::compile(
46                    &problem,
47                    observation.catalogue(),
48                    Arc::clone(&measures),
49                    None,
50                )
51                .map_err(|error| NativeError::Other(Box::new(error)))?;
52                let selection = (!shared_selection)
53                    .then(|| resolve_partitions(&problem, source, observation))
54                    .transpose()
55                    .map_err(|error| NativeError::Other(Box::new(error)))?;
56                Ok(StoreView {
57                    geometry,
58                    selection,
59                })
60            })
61            .collect::<Result<Vec<_>, NativeError>>()?;
62        let stores = route
63            .stores
64            .iter()
65            .zip(&plans)
66            .map(|(store, plan)| StoreBatches {
67                observation: store.observation(),
68                selection: plan.selection.as_ref().unwrap_or(&store.selection),
69                plan: &plan.geometry,
70                pointing_budget: store.pointing_budget,
71            })
72            .collect::<Vec<_>>();
73        let batching = imaging::admit_batches(
74            &problem,
75            &stores,
76            imaging::row_samples(&problem)?,
77            (&runtime.host, &runtime.resource_policy),
78        )?;
79        Ok(Self {
80            problem,
81            geometry: Admitted {
82                value: (plans, batching.value),
83                reservation: batching.reservation,
84            },
85        })
86    }
87
88    pub(super) fn form(
89        self,
90        route: &mut BatchRoute,
91        runtime: &ApplicationRuntime,
92        summary: &mut RunSummary,
93    ) -> Result<SeparatePsf, NativeError> {
94        let (mut plans, mut batching) = self.geometry.value;
95        swap_view(route, &mut plans, &mut batching);
96        let result = run_phase(
97            "separate mosaic PSF",
98            &runtime.cancel,
99            runtime.diagnostics,
100            summary,
101            || imaging::form_separate_psf(&self.problem, route, runtime),
102        );
103        swap_view(route, &mut plans, &mut batching);
104        result.map_err(NativeError::from)
105    }
106}
107
108fn swap_view(route: &mut BatchRoute, plans: &mut [StoreView], batching: &mut Batching) {
109    for (store, plan) in route.stores.iter_mut().zip(plans) {
110        std::mem::swap(&mut store.plan, &mut plan.geometry);
111        if let Some(selection) = &mut plan.selection {
112            std::mem::swap(&mut store.selection, selection);
113        }
114    }
115    std::mem::swap(&mut route.batching, batching);
116}
