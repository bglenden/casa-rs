4//! Metal implementation of the imaging gridding backend.
5//!
6//! This crate is the only place a device API appears in the imaging stack.
7//! [`MetalBackend`] implements the `casa-imaging-operator` backend trait for
8//! both tap layouts and every [`Work`](casa_imaging_operator::Work) variant
9//! with support-generic kernels: one SIMD group per sample, its lanes over
10//! the taps, `f32` grids updated with atomic adds (D2). The host locates
11//! every sample with the operator's rounding rule and accumulates `sumwt`
12//! in `f64`, so tap selection and `sumwt` equal the CPU backend's; only the
13//! order of the `f32` grid additions differs.
14//!
15//! Accumulators live in memory the device addresses
16//! ([`MetalBackend::accumulator`]). Every dispatch completes before `apply`
17//! returns, except grid work a backend made with
18//! [`MetalBackend::with_deferred_grid_completion`] leaves running; the
19//! accumulator's cells wait for that work, so the host reads them like any
20//! other accumulator. Without a
21//! unified-memory Metal 3 device, constructors return
22//! [`DeviceFailure::Unavailable`](casa_imaging_operator::DeviceFailure).
23//!
24//! The design is `docs/imaging-architecture/imaging-foundation-plan-20261007.md`
25//! sections 5.3 and 5.9, ADR-0016, and the IF-4 deviations on #653.

41/// The `tracing` target of the event a backend made with stage timing logs
42/// when it drops: its dispatches, samples and sub-blocks, the kernel cells
43/// it uploaded, the host's preparation time and its waits for the device,
44/// and the device's time on its commands (summed, and the span from the
45/// first start to the last end). A pass also logs, once per wave, its
46/// source stream: the producer's fill time and the pass's waits for it.
47pub const METAL_STAGE_TIMING: &str = "casa_imaging::metal_stage_timing";

49/// Ring slots of one backend: the host prepares one while the device runs
50/// the others. Twice the sub-blocks of one apply ([`SPLIT`]), so that an
51/// apply whose grid work is left in flight
52/// (`MetalBackend::with_deferred_grid_completion`) seldom waits for the apply
53/// before it.
54const RING: usize = 2 * SPLIT;
55/// Sub-blocks one apply is split into, so the host prepares one while the
56/// device runs another.
57const SPLIT: usize = 3;
58/// The most samples in one ring sub-block.
59const MAX_SUB: usize = 32_768;
60/// Device bytes per sample of a ring slot: the record, its native endpoints
61/// (or a native term), and per visibility polarization a value, a weight,
62/// an inverse norm and an output.
63const SAMPLE_BYTES: usize = 40;
64const SOURCE_BYTES: usize = 24;
65const POLARIZATION_BYTES: usize = 8 + 4 + 8 + 8;
66
67/// The most device memory one backend's ring holds for blocks of `npol`
68/// visibility polarizations; kernel tables add one copy of the taps of the
69/// cells the backend has met, and a native prediction its batch
70/// ([`batch_bytes`]).
71#[must_use]
72pub const fn ring_bytes(npol: usize) -> u64 {
73    (RING * MAX_SUB * (SAMPLE_BYTES + SOURCE_BYTES + npol * POLARIZATION_BYTES)) as u64
74}
75
76/// The device memory a backend's predictions of a native-prediction batch
77/// of `samples` samples of `npol` polarizations hold.
78#[must_use]
79pub const fn batch_bytes(npol: usize, samples: usize) -> u64 {
80    (samples * npol * size_of::<num_complex::Complex32>()) as u64
81}
