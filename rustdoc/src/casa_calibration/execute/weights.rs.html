11/// Calibrater uses partialMedians without takeEvenMean: an even channel count
12/// takes the lower central value, not the average of the two central values.
13pub(super) fn lower_median(mut values: Vec<f32>) -> f32 {
14    assert!(
15        !values.is_empty(),
16        "weight median requires at least one channel"
17    );
18    let middle = (values.len() - 1) / 2;
19    *values.select_nth_unstable_by(middle, f32::total_cmp).1
20}

22pub(super) fn from_sigma(
23    mut value: ArrayValue,
24    column: &str,
25) -> Result<ArrayValue, ApplyExecutionError> {
26    let ArrayValue::Float32(values) = &mut value else {
27        return Err(ApplyExecutionError::UnsupportedParameterShape {
28            path: format!("<measurement-set {column}>"),
29            shape: value.shape().to_vec(),
30        });
31    };
32    values.mapv_inplace(|sigma| if sigma > 0. { 1. / (sigma * sigma) } else { 0. });
33    Ok(value)
34}

36pub(super) fn has_spectrum(
37    table: &Table,
38    row: Option<usize>,
39    column: &str,
40) -> Result<bool, ApplyExecutionError> {
41    if !table
42        .schema()
43        .is_some_and(|schema| schema.contains_column(column))
44    {
45        return Ok(false);
46    }
47    let Some(row) = row else { return Ok(false) };
48    table
49        .cell_accessor(row, column)
50        .and_then(|cell| cell.array_shape())
51        .map(|shape| shape.is_some_and(|shape| shape.iter().all(|length| *length > 0)))
52        .map_err(|source| ApplyExecutionError::MutateMeasurementSet {
53            path: format!("<measurement-set {column}>"),
54            source: MsError::from(source),
55        })
56}

58/// Single-channel diagonal gains have one weight factor per correlation.
59/// Apply it over the spectrum without sampling the antenna gains per bin.
60pub(super) fn apply_scalar_pair(
61    first: &GainGrid,
62    second: &GainGrid,
63    correlations: &[i32],
64    data_desc_id: i32,
65    mut weights: Option<&mut ArrayD<f32>>,
66    mut implicit_weights: Option<&mut ArrayD<f32>>,
67) -> Result<(), ApplyExecutionError> {
68    for (correlation, &code) in correlations.iter().enumerate() {
69        let (left, right) = correlation_receptors(code).ok_or_else(|| {
70            ApplyExecutionError::UnsupportedCorrelationLayout {
71                data_desc_id,
72                correlation_types: correlations
73                    .iter()
74                    .map(|&code| stokes_name(code).to_string())
75                    .collect(),
76            }
77        })?;
78        let left = left.min(first.receptor_count.saturating_sub(1));
79        let right = right.min(second.receptor_count.saturating_sub(1));
80        let (left, _) = correcting_gain(first.value_at(left, 0), first.flag_at(left, 0));
81        let (right, _) = correcting_gain(second.value_at(right, 0), second.flag_at(right, 0));
82        let factor = left.norm_sqr() * right.norm_sqr();
83        for spectrum in [weights.as_deref_mut(), implicit_weights.as_deref_mut()]
84            .into_iter()
85            .flatten()
86        {
87            spectrum
88                .index_axis_mut(Axis(0), correlation)
89                .mapv_inplace(|weight| weight * factor);
90        }
91    }
92    Ok(())
93}
