2//! Typed wrappers for the POINTING subtable.
3//!
4//! Cf. C++ `MSPointingColumns`.
5
6use casa_types::{ArrayValue, measures::direction::MDirection};
7
8use crate::derived::engine::resolve_direction_reference;
9use crate::schema::{self, SubtableId};
10use crate::{MsError, MsResult};
11
12crate::subtables::define_generic_subtable!(
13    read = MsPointing,
14    write = MsPointingMut,
15    id = SubtableId::Pointing,
16    required = schema::pointing::REQUIRED_COLUMNS,
17    optional = schema::pointing::OPTIONAL_COLUMNS,
18    table_name = "POINTING",
19    read_doc = "Read-only typed wrapper for the POINTING subtable.",
20    write_doc = "Mutable typed wrapper for the POINTING subtable."
21);
22
23impl MsPointing<'_> {
24    /// Read the pointing used by CASA's legacy visibility buffer for an
25    /// antenna and visibility time. The returned direction is the recorded
26    /// direction at the pointing row's epoch.
27    ///
28    /// # Errors
29    /// The POINTING metadata or selected direction cell is invalid.
30    pub fn legacy_direction(&self, antenna: i32, time: f64) -> MsResult<Option<MDirection>> {
31        let mut rows = Vec::new();
32        for row in 0..self.row_count() {
33            if self.i32(row, "ANTENNA_ID")? == antenna {
34                rows.push((row, self.f64(row, "TIME")?, self.f64(row, "INTERVAL")?));
35            }
36        }
37        let nearest_sample = rows
38            .iter()
39            .filter(|(_, _, interval)| *interval <= 0.)
40            .min_by(|a, b| (a.1 - time).abs().total_cmp(&(b.1 - time).abs()));
41        let selected = rows
42            .iter()
43            .enumerate()
44            .find(|&(index, &(_, epoch, interval))| {
45                if interval > 0. {
46                    return (epoch - time).abs() <= interval / 2.;
47                }
48                let neighbor = rows[index + 1..]
49                    .iter()
50                    .chain(rows[..index].iter().rev())
51                    .find(|(_, candidate, _)| *candidate != epoch);
52                neighbor
53                    .is_none_or(|(_, next, _)| (epoch - time).abs() <= (epoch - next).abs() / 2.)
54            })
55            .map(|(_, sample)| sample);
56        let Some(&(row, _, _)) = (match (selected, nearest_sample) {
57            (Some(covering), Some(closest))
58                if (closest.1 - time).abs() <= (covering.1 - time).abs() =>
59            {
60                Some(closest)
61            }
62            (covering, closest) => covering.or(closest),
63        }) else {
64            return Ok(None);
65        };
66        let ArrayValue::Float64(values) = self.array(row, "DIRECTION")? else {
67            return Err(MsError::InvalidInput(
68                "POINTING DIRECTION must be double".into(),
69            ));
70        };
71        let (Some(&lon), Some(&lat)) = (
72            values.get(ndarray::IxDyn(&[0, 0])),
73            values.get(ndarray::IxDyn(&[1, 0])),
74        ) else {
75            return Err(MsError::InvalidInput(
76                "invalid POINTING DIRECTION shape".into(),
77            ));
78        };
79        let reference = resolve_direction_reference(self.table(), "POINTING", "DIRECTION", row)?;
80        Ok(Some(MDirection::from_angles(lon, lat, reference)))
81    }
82}
