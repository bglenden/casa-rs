23pub(super) fn initialize_columns(
24    ms: &mut MeasurementSet,
25    selected: ResolvedMsSelection,
26    columns: &[VisibilityDataColumn],
27    lock: &mut TableWriteLock,
28) -> MsResult<()> {
29    let (layouts, big_endian) = packing_layout(ms, &selected)?;
30    let path = lock
31        .path()
32        .parent()
33        .ok_or_else(|| invalid("MAIN has no parent directory"))?;
34    let temporary = tempfile::tempdir_in(path).map_err(invalid)?;
35    let mut arrays = Vec::with_capacity(columns.len());
36    let mut created = Vec::with_capacity(4 * columns.len());
37    for &column in columns {
38        let (scalars, packed, comment) = match column {
39            MODEL => ("ModelScaleOffset", "ModelCompTiled", "model data"),
40            CORRECTED => ("CorrScaleOffset", "CorrectedCompTiled", "corrected data"),
41            _ => return Err(invalid("cannot compress observed data during attachment")),
42        };
43        let name = column.name();
44        let schema = ColumnSchema::array_variable(name, PrimitiveType::Complex32, Some(2))
45            .with_comment(comment);
46        let writer = StreamingTiledShapeWriter::create(
47            temporary.path().join(name),
48            StreamedTiledShapeValueType::Int32,
49            layouts.clone(),
50            MeasurementSetWriteResources::from_system_memory(1)
51                .map_err(invalid)?
52                .available_bytes,
53            big_endian,
54        )
55        .map_err(invalid)?;
56        let data = pack_initial_cells(ms, &selected, column, writer)?;
57        arrays.push(CompressedComplexArray {
58            schema,
59            scalar_manager: scalars,
60            tiled_manager: packed,
61            packed: data.packed,
62            scales: data.scales,
63            offsets: data.offsets,
64        });
65        created.extend(
66            ["", "_COMPRESSED", "_SCALE", "_OFFSET"].map(|suffix| format!("{name}{suffix}")),
67        );
68    }
69    ms.main_table_mut()
70        .add_compressed_complex_arrays_in_place(&arrays)?;
71    lock.record_write();
72    let column = if columns.contains(&MODEL) {
73        MODEL
74    } else {
75        CORRECTED
76    };
77    if let Err(error) = casa_ms::prepare_write_holding(ms, column, true, lock) {
78        let added = created.iter().map(String::as_str).collect::<Vec<_>>();
79        ms.main_table_mut()
80            .remove_columns_preserving_managers(&added)
81            .map_err(|rollback| {
82                invalid(format!(
83                    "{error}. Removing the new scratch columns also failed: {rollback}"
84                ))
85            })?;
86        lock.record_write();
87        return Err(error);
88    }
89    Ok(())
90}

92fn packing_layout(
93    ms: &MeasurementSet,
94    selected: &ResolvedMsSelection,
95) -> MsResult<(Vec<StreamedTiledShapeCubeLayout>, bool)> {
96    let spectral = ms.spectral_window()?;
97    let mut max_channels = 0usize;
98    for window in 0..spectral.row_count() {
99        max_channels =
100            max_channels.max(usize::try_from(spectral.num_chan(window)?).map_err(invalid)?);
101    }
102    let main = ms.main_table();
103    let observed = main.column_accessor(observed_column(ms))?;
104    let mut shapes = BTreeMap::<Vec<usize>, usize>::new();
105    let mut first_correlations = None;
106    for batch in selected.selected_rows.chunks(BATCH_ROWS) {
107        let indices = batch.iter().map(|row| row.row_index).collect::<Vec<_>>();
108        for shape in observed.array_shapes_uncached(&indices)? {
109            let shape = shape.ok_or_else(|| invalid("observed data has an undefined cell"))?;
110            let &[correlations, _] = shape.as_slice() else {
111                return Err(invalid("observed visibility must have two axes"));
112            };
113            if correlations == 0 {
114                return Err(invalid("observed visibility has no correlations"));
115            }
116            first_correlations.get_or_insert(correlations);
117            *shapes.entry(shape).or_default() += 1;
118        }
119    }
120    let correlations = first_correlations.ok_or_else(|| invalid("observed data has no rows"))?;
121    let channel_tile = max_channels / 10 + 1;
122    let tile = vec![
123        correlations,
124        channel_tile,
125        131_072 / correlations / channel_tile + 1,
126    ];
127    let layouts = shapes
128        .into_iter()
129        .map(|(shape, count)| StreamedTiledShapeCubeLayout::new(shape, count, tile.clone()))
130        .collect();
131    let big_endian = main
132        .stored_endian_format()?
133        .is_none_or(casa_tables::EndianFormat::is_big_endian);
134    Ok((layouts, big_endian))
135}

137struct PackedScratch {
138    packed: casa_tables::streamed_tiled::StreamedTiledShapeColumn,
139    scales: Vec<Option<ScalarValue>>,
140    offsets: Vec<Option<ScalarValue>>,
141}
142
143fn pack_initial_cells(
144    ms: &MeasurementSet,
145    selected: &ResolvedMsSelection,
146    column: VisibilityDataColumn,
147    mut writer: StreamingTiledShapeWriter,
148) -> MsResult<PackedScratch> {
149    let codec = ComplexCompression::automatic_standard(column.name());
150    let polarizations = if column == MODEL {
151        model_polarizations(ms)?
152    } else {
153        BTreeMap::new()
154    };
155    let destination = ScratchDestination::new(ms, column, column.name())?;
156    let row_count = selected.selected_rows.len();
157    let mut scales = Vec::with_capacity(row_count);
158    let mut offsets = Vec::with_capacity(row_count);
159    for batch in selected.selected_rows.chunks(BATCH_ROWS) {
160        let columns = reset_batch(
161            ms,
162            observed_column(ms),
163            batch,
164            std::slice::from_ref(&destination),
165            &BTreeMap::new(),
166            &polarizations,
167        )?
168        .columns;
169        for column in columns {
170            let MeasurementSetMutationColumnValues::Arrays(values) = column.values else {
171                return Err(invalid("initial scratch cells must be arrays"));
172            };
173            for value in values {
174                let ArrayValue::Complex32(value) = value else {
175                    return Err(invalid("initial compressed scratch cells must be complex"));
176                };
177                let (packed, (scale, offset)) =
178                    codec.encode_visibility_cell(&value, None, (0., 0.), None)?;
179                let cells = packed.view().reversed_axes();
180                let contiguous = cells.as_standard_layout();
181                writer
182                    .push_i32_row(
183                        packed.shape(),
184                        contiguous.as_slice().expect("contiguous row"),
185                    )
186                    .map_err(invalid)?;
187                scales.push(Some(ScalarValue::Float32(scale)));
188                offsets.push(Some(ScalarValue::Float32(offset)));
189            }
190        }
191    }
192    Ok(PackedScratch {
193        packed: writer.finish().map_err(invalid)?,
194        scales,
195        offsets,
196    })
197}

199pub(crate) struct ScratchDestination<'a> {
200    pub(super) visibility: VisibilityDataColumn,
201    destination: VisibilityWriteDestination<'a>,
202}
203
204impl<'a> ScratchDestination<'a> {
205    pub(super) fn new(
206        ms: &MeasurementSet,
207        visibility: VisibilityDataColumn,
208        name: &'a str,
209    ) -> MsResult<Self> {
210        Ok(Self {
211            visibility,
212            destination: VisibilityWriteDestination::new(ms.main_table(), name)?,
213        })
214    }
215
216    pub(super) fn name(&self) -> &str {
217        self.destination.name()
218    }
219
220    pub(super) fn plans(
221        &self,
222        elements: usize,
223        create: bool,
224    ) -> Vec<MeasurementSetWriteColumnPlan> {
225        self.destination.plans(elements, create)
226    }
227
228    pub(crate) fn batch(
229        &self,
230        ms: &MeasurementSet,
231        row_indices: &[usize],
232        rows: &[ResolvedMsSelectionRow],
233        ranges: &BTreeMap<i32, Range<usize>>,
234        values: Vec<ArrayValue>,
235    ) -> MsResult<Vec<MeasurementSetMutationColumnBatch>> {
236        let channels = (!ranges.is_empty()).then(|| {
237            rows.iter()
238                .map(|row| ranges.get(&row.spectral_window_id).cloned())
239                .collect::<Vec<_>>()
240        });
241        self.destination
242            .batch(ms.main_table(), row_indices, values, channels.as_deref())
243    }
244}
