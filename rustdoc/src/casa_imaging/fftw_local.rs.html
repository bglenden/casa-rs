166/// Run the uncentered real-to-complex transform used by
167/// `FFTServer<Float, Complex>::fft0`.
168///
169/// The input has logical casacore shape `(nx, ny)`. It is copied into
170/// first-axis-contiguous storage before FFTW is called with reversed
171/// dimensions `[ny, nx]`, so FFTW's final-axis truncation becomes casacore's
172/// half first logical axis. The returned spectrum therefore has logical shape
173/// `(nx / 2 + 1, ny)` and first-axis-contiguous physical storage.
174pub(crate) fn fft0_r2c_f32_casacore_layout(
175    input: &Array2<f32>,
176    threads: usize,
177) -> Result<FftwRealHalfSpectrumF32, &'static str> {
178    let (nx, ny) = input.dim();
179    let (dimensions, real_elements, half_elements) = fft0_real_shape(nx, ny)?;
180    let threads = fftw_thread_count(threads)?;
181    let half_nx = nx / 2 + 1;
182    let api = api_f32()?;
183
184    let mut input_storage = Vec::with_capacity(real_elements);
185    for y in 0..ny {
186        for x in 0..nx {
187            input_storage.push(input[(x, y)]);
188        }
189    }
190    let mut output_storage = vec![Complex32::new(0.0, 0.0); half_elements];
191
192    let plan_configuration = plan_configuration("CASA_RS_EXPERIMENTAL_FFTW_F32_WISDOM")?;
193    let plan = {
194        let _guard = FFTW_PLAN_LOCK
195            .lock()
196            .map_err(|_| "fftw_local_plan_lock_poisoned")?;
197        // SAFETY: the symbol signature is FFTW's public single-precision C
198        // ABI. Both buffers match the reversed dimensions and remain
199        // allocated until the bound plan has executed and been destroyed.
200        unsafe {
201            if (api.init_threads)() == 0 {
202                return Err("fftw_local_init_threads_failed");
203            }
204            (api.plan_with_threads)(threads);
205            if let Some(path) = plan_configuration.wisdom_path.as_ref() {
206                let path = CString::new(path.as_os_str().as_encoded_bytes())
207                    .map_err(|_| "fftw_local_wisdom_path_contains_nul")?;
208                if (api.import_wisdom_from_filename)(path.as_ptr()) == 0 {
209                    return Err("fftw_local_wisdom_import_failed");
210                }
211                verify_wisdom_unchanged(&plan_configuration)?;
212            }
213            (api.plan_dft_r2c)(
214                2,
215                dimensions.as_ptr(),
216                input_storage.as_mut_ptr(),
217                output_storage.as_mut_ptr(),
218                plan_configuration.flags,
219            )
220        }
221    };
222    record_provenance_once("f32", &api.provenance, threads, &plan_configuration);
223    if plan.is_null() {
224        return Err("fftw_local_r2c_plan_failed");
225    }
226
227    // SAFETY: `plan` is bound to the live input and output allocations above.
228    unsafe {
229        (api.execute)(plan);
230        (api.destroy_plan)(plan);
231    }
232
233    let values = Array2::from_shape_vec((half_nx, ny).f(), output_storage)
234        .map_err(|_| "fftw_local_r2c_output_shape_failed")?;
235    Ok(FftwRealHalfSpectrumF32 {
236        logical_shape: (nx, ny),
237        values,
238    })
239}

241/// Run the uncentered complex-to-real transform used by
242/// `FFTServer<Float, Complex>::fft0`.
243///
244/// The half spectrum's first logical axis is restored to its recorded real
245/// length. FFTW sees reversed dimensions `[ny, nx]`. Its unnormalized output
246/// is divided element-by-element in `f64` by the full real element count and
247/// then cast to `f32`, matching casacore's `Float /= 1.0 * size` evaluation.
248/// The returned real array has logical shape `(nx, ny)` and
249/// first-axis-contiguous physical storage.
250pub(crate) fn fft0_c2r_f32_casacore_layout(
251    input: &FftwRealHalfSpectrumF32,
252    threads: usize,
253) -> Result<Array2<f32>, &'static str> {
254    let (nx, ny) = input.logical_shape;
255    let (dimensions, real_elements, half_elements) = fft0_real_shape(nx, ny)?;
256    let threads = fftw_thread_count(threads)?;
257    let half_nx = nx / 2 + 1;
258    if input.values.dim() != (half_nx, ny) || input.values.strides() != [1, half_nx as isize] {
259        return Err("fftw_local_c2r_requires_casacore_half_spectrum_layout");
260    }
261    let api = api_f32()?;
262    let mut input_storage = input
263        .values
264        .as_slice_memory_order()
265        .ok_or("fftw_local_c2r_requires_contiguous_half_spectrum")?
266        .to_vec();
267    if input_storage.len() != half_elements {
268        return Err("fftw_local_c2r_half_spectrum_length_mismatch");
269    }
270    let mut output_storage = vec![0.0_f32; real_elements];
271
272    let plan_configuration = plan_configuration("CASA_RS_EXPERIMENTAL_FFTW_F32_WISDOM")?;
273    let plan = {
274        let _guard = FFTW_PLAN_LOCK
275            .lock()
276            .map_err(|_| "fftw_local_plan_lock_poisoned")?;
277        // SAFETY: the symbol signature is FFTW's public single-precision C
278        // ABI. Both buffers match the reversed dimensions and remain
279        // allocated until the bound plan has executed and been destroyed.
280        unsafe {
281            if (api.init_threads)() == 0 {
282                return Err("fftw_local_init_threads_failed");
283            }
284            (api.plan_with_threads)(threads);
285            if let Some(path) = plan_configuration.wisdom_path.as_ref() {
286                let path = CString::new(path.as_os_str().as_encoded_bytes())
287                    .map_err(|_| "fftw_local_wisdom_path_contains_nul")?;
288                if (api.import_wisdom_from_filename)(path.as_ptr()) == 0 {
289                    return Err("fftw_local_wisdom_import_failed");
290                }
291                verify_wisdom_unchanged(&plan_configuration)?;
292            }
293            (api.plan_dft_c2r)(
294                2,
295                dimensions.as_ptr(),
296                input_storage.as_mut_ptr(),
297                output_storage.as_mut_ptr(),
298                plan_configuration.flags,
299            )
300        }
301    };
302    record_provenance_once("f32", &api.provenance, threads, &plan_configuration);
303    if plan.is_null() {
304        return Err("fftw_local_c2r_plan_failed");
305    }
306
307    // SAFETY: `plan` is bound to the live input and output allocations above.
308    unsafe {
309        (api.execute)(plan);
310        (api.destroy_plan)(plan);
311    }
312    let divisor = real_elements as f64;
313    for value in &mut output_storage {
314        *value = (f64::from(*value) / divisor) as f32;
315    }
316
317    Array2::from_shape_vec((nx, ny).f(), output_storage)
318        .map_err(|_| "fftw_local_c2r_output_shape_failed")
319}

321fn fft0_real_shape(nx: usize, ny: usize) -> Result<([c_int; 2], usize, usize), &'static str> {
322    let nx_c_int = c_int::try_from(nx)
323        .ok()
324        .filter(|&value| value > 0)
325        .ok_or("fftw_local_real_fft_requires_non_empty_int_range_axes")?;
326    let ny_c_int = c_int::try_from(ny)
327        .ok()
328        .filter(|&value| value > 0)
329        .ok_or("fftw_local_real_fft_requires_non_empty_int_range_axes")?;
330    let real_elements = nx
331        .checked_mul(ny)
332        .ok_or("fftw_local_real_fft_element_count_overflow")?;
333    let half_elements = (nx / 2 + 1)
334        .checked_mul(ny)
335        .ok_or("fftw_local_real_fft_element_count_overflow")?;
336    Ok(([ny_c_int, nx_c_int], real_elements, half_elements))
337}

363/// Run FFTW with the physical x-contiguous storage used by casacore arrays.
364///
365/// `ndarray`'s standard `(x, y)` layout keeps `y` contiguous, whereas
366/// `casacore::Array` keeps its first (`x`) axis contiguous. CASA's FFTServer
367/// reverses the logical shape before passing that storage to FFTW. Transpose
368/// into the corresponding physical layout, run the same centered transform,
369/// and expose the logical result with the same first-axis-contiguous physical
370/// layout without a second image-sized copy.
371pub(crate) fn centered_transform_f32_casacore_layout_with_threads(
372    input: &Array2<Complex32>,
373    direction: FftDirection,
374    use_case: FftUseCase,
375    threads: usize,
376) -> Result<(Array2<Complex32>, FftTiming), &'static str> {
377    let rows = input.shape()[0];
378    let columns = input.shape()[1];
379    let total_started = Instant::now();
380
381    let pack_started = Instant::now();
382    let casacore_storage = input.view().reversed_axes().as_standard_layout().to_owned();
383    let input_pack = pack_started.elapsed();
384
385    let (casacore_output, mut timing) =
386        centered_transform_f32_owned_with_threads(casacore_storage, direction, use_case, threads)?;
387
388    let output = casacore_output.reversed_axes();
389    timing.pack += input_pack;
390    timing.total = total_started.elapsed();
391
392    let spec = Fft2Spec::centered_c2c(
393        rows,
394        columns,
395        FftPrecision::F32,
396        direction,
397        use_case,
398        FftBackendChoice::Fftw,
399    );
400    timing.spec = spec;
401    timing.selection = select_fft_backend(spec);
402    Ok((output, timing))
403}

439fn centered_transform_f32_owned_with_threads(
440    mut output: Array2<Complex32>,
441    direction: FftDirection,
442    use_case: FftUseCase,
443    threads: usize,
444) -> Result<(Array2<Complex32>, FftTiming), &'static str> {
445    let threads = c_int::try_from(threads)
446        .ok()
447        .filter(|&value| value > 0)
448        .ok_or("fftw_local_requires_positive_int_range_thread_count")?;
449    if output.shape().len() != 2
450        || output.shape()[0] == 0
451        || output.shape()[1] == 0
452        || output.shape()[0] & 1 != 0
453        || output.shape()[1] & 1 != 0
454        || output.shape()[0] > c_int::MAX as usize
455        || output.shape()[1] > c_int::MAX as usize
456    {
457        return Err("fftw_local_requires_non_empty_even_int_range_axes");
458    }
459    let api = api_f32()?;
460    let rows = output.shape()[0];
461    let columns = output.shape()[1];
462    let spec = Fft2Spec::centered_c2c(
463        rows,
464        columns,
465        FftPrecision::F32,
466        direction,
467        use_case,
468        FftBackendChoice::Fftw,
469    );
470    let selection = select_fft_backend(spec);
471    if !selection.requested_backend_supported {
472        return Err("fftw_local_backend_not_supported");
473    }
474    let mut timing = FftTiming::new(spec, selection);
475    timing.plan_cache_hit = false;
476    let total_started = Instant::now();
477
478    let pack_started = Instant::now();
479    if !shift_quadrants_in_place_f32(&mut output) {
480        return Err("fftw_local_requires_contiguous_even_axes");
481    }
482    timing.pack += pack_started.elapsed();
483
484    let dimensions = [rows as c_int, columns as c_int];
485    let sign = match direction {
486        FftDirection::Forward => FFTW_FORWARD,
487        FftDirection::Inverse => FFTW_BACKWARD,
488    };
489    let plan_started = Instant::now();
490    let output_pointer = output
491        .as_slice_memory_order_mut()
492        .ok_or("fftw_local_requires_contiguous_input")?
493        .as_mut_ptr();
494    let plan_configuration = plan_configuration("CASA_RS_EXPERIMENTAL_FFTW_F32_WISDOM")?;
495    let plan = {
496        let _guard = FFTW_PLAN_LOCK
497            .lock()
498            .map_err(|_| "fftw_local_plan_lock_poisoned")?;
499        // SAFETY: the dynamically loaded symbols use FFTW's documented C ABI.
500        // Planning is serialized because FFTW's planner and thread count are
501        // global. The input is contiguous and remains alive until destruction.
502        unsafe {
503            if (api.init_threads)() == 0 {
504                return Err("fftw_local_init_threads_failed");
505            }
506            (api.plan_with_threads)(threads);
507            if let Some(path) = plan_configuration.wisdom_path.as_ref() {
508                let path = CString::new(path.as_os_str().as_encoded_bytes())
509                    .map_err(|_| "fftw_local_wisdom_path_contains_nul")?;
510                if (api.import_wisdom_from_filename)(path.as_ptr()) == 0 {
511                    return Err("fftw_local_wisdom_import_failed");
512                }
513                verify_wisdom_unchanged(&plan_configuration)?;
514            }
515            (api.plan_dft)(
516                2,
517                dimensions.as_ptr(),
518                output_pointer,
519                output_pointer,
520                sign,
521                plan_configuration.flags,
522            )
523        }
524    };
525    record_provenance_once("f32", &api.provenance, threads, &plan_configuration);
526    timing.plan = plan_started.elapsed();
527    if plan.is_null() {
528        return Err("fftw_local_plan_failed");
529    }
530
531    let exec_started = Instant::now();
532    // SAFETY: `plan` was created for `output`, which remains allocated and
533    // unmoved for the duration of execution.
534    unsafe {
535        (api.execute)(plan);
536    }
537    if direction == FftDirection::Inverse {
538        let scale = 1.0 / (rows * columns) as f32;
539        output.mapv_inplace(|value| value * scale);
540    }
541    timing.exec = exec_started.elapsed();
542
543    let pack_started = Instant::now();
544    let shifted = shift_quadrants_in_place_f32(&mut output);
545    debug_assert!(shifted);
546    timing.pack += pack_started.elapsed();
547    timing.total = total_started.elapsed();
548
549    // SAFETY: `plan` is a live FFTW plan and is destroyed exactly once after
550    // its bound execution has completed.
551    unsafe {
552        (api.destroy_plan)(plan);
553    }
554    Ok((output, timing))
555}

590    );
591    let selection = select_fft_backend(spec);
592    if !selection.requested_backend_supported {
593        return Err("fftw_local_backend_not_supported");
594    }
595    let mut timing = FftTiming::new(spec, selection);
596    timing.plan_cache_hit = false;
597    let total_started = Instant::now();
598
599    let pack_started = Instant::now();
600    if !shift_quadrants_in_place(&mut output) {
601        return Err("fftw_local_requires_contiguous_even_axes");
602    }
603    timing.pack += pack_started.elapsed();
604
605    let dimensions = [rows as c_int, columns as c_int];
606    let sign = match direction {
607        FftDirection::Forward => FFTW_FORWARD,
608        FftDirection::Inverse => FFTW_BACKWARD,
609    };
610    let plan_started = Instant::now();
611    let output_pointer = output
612        .as_slice_memory_order_mut()
613        .ok_or("fftw_local_requires_contiguous_input")?
614        .as_mut_ptr();
615    let threads = thread_count();
616    let plan_configuration = plan_configuration("CASA_RS_EXPERIMENTAL_FFTW_F64_WISDOM")?;
617    let plan = {
618        let _guard = FFTW_PLAN_LOCK
619            .lock()
620            .map_err(|_| "fftw_local_plan_lock_poisoned")?;
621        // SAFETY: the dynamically loaded symbols use FFTW's documented C ABI.
622        // Planning is serialized because FFTW's planner and thread count are
623        // global. The input is contiguous and remains alive until destruction.
624        unsafe {
625            if (api.init_threads)() == 0 {
626                return Err("fftw_local_init_threads_failed");
627            }
628            (api.plan_with_threads)(threads);
629            if let Some(path) = plan_configuration.wisdom_path.as_ref() {
630                let path = CString::new(path.as_os_str().as_encoded_bytes())
631                    .map_err(|_| "fftw_local_wisdom_path_contains_nul")?;
632                if (api.import_wisdom_from_filename)(path.as_ptr()) == 0 {
633                    return Err("fftw_local_wisdom_import_failed");
634                }
635                verify_wisdom_unchanged(&plan_configuration)?;
636            }
637            (api.plan_dft)(
638                2,
639                dimensions.as_ptr(),
640                output_pointer,
641                output_pointer,
642                sign,
643                plan_configuration.flags,
644            )
645        }
646    };
647    record_provenance_once("f64", &api.provenance, threads, &plan_configuration);
648    timing.plan = plan_started.elapsed();
649    if plan.is_null() {
650        return Err("fftw_local_plan_failed");
651    }
652
653    let exec_started = Instant::now();
654    // SAFETY: `plan` was created for `output`, which remains allocated and
655    // unmoved for the duration of execution.
656    unsafe {
657        (api.execute)(plan);
658    }
659    if direction == FftDirection::Inverse {
660        let scale = 1.0 / (rows * columns) as f64;
661        output.mapv_inplace(|value| value * scale);
662    }
663    timing.exec = exec_started.elapsed();
664
665    let pack_started = Instant::now();
666    let shifted = shift_quadrants_in_place(&mut output);
667    debug_assert!(shifted);
668    timing.pack += pack_started.elapsed();
669    timing.total = total_started.elapsed();
670
671    // SAFETY: `plan` is a live FFTW plan and is destroyed exactly once after
672    // its bound execution has completed.
673    unsafe {
674        (api.destroy_plan)(plan);
675    }
676    Ok((output, timing))
677}

798    resolved: &ResolvedLibraryPair,

800    version_symbol: &[u8],

818fn library_string_symbol(library: &Library, symbol_name: &[u8]) -> Option<String> {
819    // SAFETY: FFTW exposes its version and compiler metadata as
820    // NUL-terminated global character arrays. The retained `Library` owns the
821    // symbol storage for the duration of this read.
822    unsafe {
823        let symbol = library.get::<*const c_char>(symbol_name).ok()?;
824        let pointer = *symbol;
825        if pointer.is_null() {
826            return None;
827        }
828        Some(CStr::from_ptr(pointer).to_string_lossy().into_owned())
829    }

924fn receipt_token(bytes: &[u8]) -> String {
925    const HEX: &[u8; 16] = b"0123456789abcdef";
926    if bytes.is_empty() {
927        return "empty".to_owned();
928    }
929    let mut token = String::with_capacity(bytes.len());
930    for &byte in bytes {
931        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.' | b'/' | b':') {
932            token.push(char::from(byte));
933        } else {
934            token.push('%');
935            token.push(char::from(HEX[(byte >> 4) as usize]));
936            token.push(char::from(HEX[(byte & 0x0f) as usize]));
937        }
938    }
939    token

942struct Sha256 {
943    state: [u32; 8],
944    block: [u8; 64],
945    block_len: usize,
946    byte_len: u64,
947}
948
949impl Sha256 {
950    const INITIAL_STATE: [u32; 8] = [
951        0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab,
952        0x5be0cd19,
953    ];
954    const ROUND_CONSTANTS: [u32; 64] = [
955        0x428a2f98, 0x71374491, 0xb5c0fbcf, 0xe9b5dba5, 0x3956c25b, 0x59f111f1, 0x923f82a4,
956        0xab1c5ed5, 0xd807aa98, 0x12835b01, 0x243185be, 0x550c7dc3, 0x72be5d74, 0x80deb1fe,
957        0x9bdc06a7, 0xc19bf174, 0xe49b69c1, 0xefbe4786, 0x0fc19dc6, 0x240ca1cc, 0x2de92c6f,
958        0x4a7484aa, 0x5cb0a9dc, 0x76f988da, 0x983e5152, 0xa831c66d, 0xb00327c8, 0xbf597fc7,
959        0xc6e00bf3, 0xd5a79147, 0x06ca6351, 0x14292967, 0x27b70a85, 0x2e1b2138, 0x4d2c6dfc,
960        0x53380d13, 0x650a7354, 0x766a0abb, 0x81c2c92e, 0x92722c85, 0xa2bfe8a1, 0xa81a664b,
961        0xc24b8b70, 0xc76c51a3, 0xd192e819, 0xd6990624, 0xf40e3585, 0x106aa070, 0x19a4c116,
962        0x1e376c08, 0x2748774c, 0x34b0bcb5, 0x391c0cb3, 0x4ed8aa4a, 0x5b9cca4f, 0x682e6ff3,
963        0x748f82ee, 0x78a5636f, 0x84c87814, 0x8cc70208, 0x90befffa, 0xa4506ceb, 0xbef9a3f7,
964        0xc67178f2,
965    ];
966
967    fn new() -> Self {
968        Self {
969            state: Self::INITIAL_STATE,
970            block: [0; 64],
971            block_len: 0,
972            byte_len: 0,
973        }
974    }
975
976    fn update(&mut self, mut input: &[u8]) {
977        self.byte_len = self.byte_len.wrapping_add(input.len() as u64);
978        if self.block_len != 0 {
979            let copied = (64 - self.block_len).min(input.len());
980            self.block[self.block_len..self.block_len + copied].copy_from_slice(&input[..copied]);
981            self.block_len += copied;
982            input = &input[copied..];
983            if self.block_len < 64 {
984                return;
985            }
986            let block = self.block;
987            self.compress(&block);
988            self.block_len = 0;
989        }
990        while input.len() >= 64 {
991            let block = <&[u8; 64]>::try_from(&input[..64])
992                .expect("64-byte SHA256 block slice must convert");
993            self.compress(block);
994            input = &input[64..];
995        }
996        self.block[..input.len()].copy_from_slice(input);
997        self.block_len = input.len();
998    }
999
1000    fn finish(mut self) -> String {
1001        let bit_len = self.byte_len.wrapping_mul(8);
1002        self.block[self.block_len] = 0x80;
1003        self.block_len += 1;
1004        if self.block_len > 56 {
1005            self.block[self.block_len..].fill(0);
1006            let block = self.block;
1007            self.compress(&block);
1008            self.block_len = 0;
1009        }
1010        self.block[self.block_len..56].fill(0);
1011        self.block[56..].copy_from_slice(&bit_len.to_be_bytes());
1012        let block = self.block;
1013        self.compress(&block);
1014
1015        const HEX: &[u8; 16] = b"0123456789abcdef";
1016        let mut digest = String::with_capacity(64);
1017        for byte in self.state.into_iter().flat_map(u32::to_be_bytes) {
1018            digest.push(char::from(HEX[(byte >> 4) as usize]));
1019            digest.push(char::from(HEX[(byte & 0x0f) as usize]));
1020        }
1021        digest
1022    }
1023
1024    fn compress(&mut self, block: &[u8; 64]) {
1025        let mut schedule = [0_u32; 64];
1026        for (word, bytes) in schedule[..16].iter_mut().zip(block.chunks_exact(4)) {
1027            *word = u32::from_be_bytes(
1028                <[u8; 4]>::try_from(bytes).expect("four-byte SHA256 word must convert"),
1029            );
1030        }
1031        for index in 16..64 {
1032            let s0 = schedule[index - 15].rotate_right(7)
1033                ^ schedule[index - 15].rotate_right(18)
1034                ^ (schedule[index - 15] >> 3);
1035            let s1 = schedule[index - 2].rotate_right(17)
1036                ^ schedule[index - 2].rotate_right(19)
1037                ^ (schedule[index - 2] >> 10);
1038            schedule[index] = schedule[index - 16]
1039                .wrapping_add(s0)
1040                .wrapping_add(schedule[index - 7])
1041                .wrapping_add(s1);
1042        }
1043
1044        let [mut a, mut b, mut c, mut d, mut e, mut f, mut g, mut h] = self.state;
1045        for (&round_constant, &schedule_word) in Self::ROUND_CONSTANTS.iter().zip(&schedule) {
1046            let sum1 = e.rotate_right(6) ^ e.rotate_right(11) ^ e.rotate_right(25);
1047            let choice = (e & f) ^ (!e & g);
1048            let temporary1 = h
1049                .wrapping_add(sum1)
1050                .wrapping_add(choice)
1051                .wrapping_add(round_constant)
1052                .wrapping_add(schedule_word);
1053            let sum0 = a.rotate_right(2) ^ a.rotate_right(13) ^ a.rotate_right(22);
1054            let majority = (a & b) ^ (a & c) ^ (b & c);
1055            let temporary2 = sum0.wrapping_add(majority);
1056            h = g;
1057            g = f;
1058            f = e;
1059            e = d.wrapping_add(temporary1);
1060            d = c;
1061            c = b;
1062            b = a;
1063            a = temporary1.wrapping_add(temporary2);
1064        }
1065        for (state, value) in self.state.iter_mut().zip([a, b, c, d, e, f, g, h]) {
1066            *state = state.wrapping_add(value);
1067        }
1068    }
1069}

1071fn sha256_file(path: &Path) -> std::io::Result<String> {
1072    let mut file = File::open(path)?;
1073    let mut sha256 = Sha256::new();
1074    let mut buffer = [0_u8; 64 * 1024];
1075    loop {
1076        let count = file.read(&mut buffer)?;
1077        if count == 0 {
1078            break;
1079        }
1080        sha256.update(&buffer[..count]);
1081    }
1082    Ok(sha256.finish())
1083}

1138fn resolve_fftw_library_pair_from(
1139    explicit: Option<&OsStr>,
1140    candidates: &[(&str, &str)],
1141    conventional_directories: impl IntoIterator<Item = PathBuf>,
1142) -> Result<ResolvedLibraryPair, &'static str> {
1143    if let Some(explicit) = explicit {
1144        if explicit.is_empty() {
1145            return Err("fftw_local_explicit_library_directory_empty");
1146        }
1147        return resolve_library_pair_in_directory(
1148            Path::new(&explicit),
1149            candidates,
1150            FftwResolutionSource::ExplicitDirectory,
1151        );
1152    }
1153
1154    for directory in conventional_directories {
1155        if let Ok(resolved) = resolve_library_pair_in_directory(
1156            &directory,
1157            candidates,
1158            FftwResolutionSource::ConventionalDirectory,
1159        ) {
1160            return Ok(resolved);
1161        }
1162    }
1163    Err("fftw_runtime_libraries_not_found")
1164}

1166fn resolve_library_pair_in_directory(
1167    directory: &Path,
1168    candidates: &[(&str, &str)],
1169    source: FftwResolutionSource,
1170) -> Result<ResolvedLibraryPair, &'static str> {
1171    if !directory.is_dir() {
1172        return Err("fftw_local_library_directory_not_found");
1173    }
1174    for &(core_name, threads_name) in candidates {
1175        let core_path = directory.join(core_name);
1176        let threads_path = directory.join(threads_name);
1177        if !core_path.is_file() || !threads_path.is_file() {
1178            continue;
1179        }
1180        return Ok(ResolvedLibraryPair {
1181            source,
1182            core_path: core_path
1183                .canonicalize()
1184                .map_err(|_| "fftw_local_core_library_canonicalization_failed")?,
1185            threads_path: threads_path
1186                .canonicalize()
1187                .map_err(|_| "fftw_local_threads_library_canonicalization_failed")?,
1188        });
1189    }
1190    Err("fftw_local_library_pair_incomplete")
1191}

1225fn shift_quadrants_in_place(input: &mut Array2<Complex64>) -> bool {
1226    let rows = input.shape()[0];
1227    let columns = input.shape()[1];
1228    if rows & 1 != 0 || columns & 1 != 0 {
1229        return false;
1230    }
1231    let Some(values) = input.as_slice_memory_order_mut() else {
1232        return false;
1233    };
1234    let half_rows = rows / 2;
1235    let half_columns = columns / 2;
1236    for row in 0..half_rows {
1237        for column in 0..half_columns {
1238            let q00 = row * columns + column;
1239            let q11 = (row + half_rows) * columns + column + half_columns;
1240            values.swap(q00, q11);
1241
1242            let q10 = (row + half_rows) * columns + column;
1243            let q01 = row * columns + column + half_columns;
1244            values.swap(q10, q01);
1245        }
1246    }
1247    true
1248}
