202/// One typed requirement not implemented by the installed imaging build.
203#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
204pub enum UnsupportedRequirement {
205    /// A compiler-derived capability has no installed implementation.
206    Capability(RequiredCapability),
207    /// A task-only requirement has no installed implementation.
208    Task(TaskRequirement),
209    /// The implementation requires exactly one observation source.
210    SingleObservationSource,
211    /// Facet execution currently requires the constant spectral basis.
212    ConstantBasisForFacets,
213    /// Non-Stokes-I or multi-polarization execution requires an independent-plane basis.
214    IndependentBasisForPolarizationSelection,
215    /// The implementation requires a fixed phase centre.
216    FixedPhaseCentre,
217    /// The implementation does not accept an initial model.
218    EmptyInitialModel,
219    /// The implementation does not write `MODEL_DATA`.
220    NoModelColumnWrite,
221    /// The implementation requires a scalar measurement equation.
222    ScalarInstrumentResponse,
223}
224
225impl UnsupportedRequirement {
226    /// Return the stable reason family used by typed transport projections.
227    #[must_use]
228    pub const fn catalog_kind(self) -> &'static str {
229        match self {
230            Self::Capability(_) => "capability",
231            Self::Task(_) => "task",
232            Self::SingleObservationSource
233            | Self::ConstantBasisForFacets
234            | Self::IndependentBasisForPolarizationSelection
235            | Self::FixedPhaseCentre
236            | Self::EmptyInitialModel
237            | Self::NoModelColumnWrite
238            | Self::ScalarInstrumentResponse => "constraint",
239        }
240    }
241
242    /// Return the exact stable reason identity exposed by provider projections.
243    #[must_use]
244    pub fn catalog_id(self) -> String {
245        match self {
246            Self::Capability(requirement) => {
247                format!("capability.{}", requirement.catalog_id())
248            }
249            Self::Task(requirement) => format!("task.{}", requirement.catalog_id()),
250            Self::SingleObservationSource => "constraint.single_observation_source".to_string(),
251            Self::ConstantBasisForFacets => "constraint.constant_basis_for_facets".to_string(),
252            Self::IndependentBasisForPolarizationSelection => {
253                "constraint.independent_basis_for_polarization_selection".to_string()
254            }
255            Self::FixedPhaseCentre => "constraint.fixed_phase_centre".to_string(),
256            Self::EmptyInitialModel => "constraint.empty_initial_model".to_string(),
257            Self::NoModelColumnWrite => "constraint.no_model_column_write".to_string(),
258            Self::ScalarInstrumentResponse => "constraint.scalar_instrument_response".to_string(),
259        }
260    }
261}

299    unsupported: Option<UnsupportedRequirement>,

302impl ImagingCapabilityCatalogEntry {
303    /// Return the typed requirement.
304    #[must_use]
305    pub const fn requirement(&self) -> ImagingCapabilityRequirement {
306        self.requirement
307    }
308
309    /// Return the exact typed unavailability reason, or `None` when supported.
310    #[must_use]
311    pub const fn unsupported(&self) -> Option<UnsupportedRequirement> {
312        self.unsupported
313    }

316/// Return every stable scientific, product, and task capability understood by
317/// the current request/application contract with its exact installed status.
318#[must_use]
319pub fn installed_imaging_capability_catalog() -> Vec<ImagingCapabilityCatalogEntry> {
320    let mut catalog = RequiredCapability::catalog()
321        .into_iter()
322        .map(|requirement| ImagingCapabilityCatalogEntry {
323            requirement: ImagingCapabilityRequirement::Scientific(requirement),
324            unsupported: (!supports_capability(requirement))
325                .then_some(UnsupportedRequirement::Capability(requirement)),
326        })
327        .collect::<Vec<_>>();
328    catalog.extend(TaskRequirement::ALL.into_iter().map(|requirement| {
329        ImagingCapabilityCatalogEntry {
330            requirement: ImagingCapabilityRequirement::Task(requirement),
331            unsupported: (!supports_task(requirement))
332                .then_some(UnsupportedRequirement::Task(requirement)),
333        }
334    }));
335    catalog.sort_by_key(|entry| entry.requirement.catalog_id());
336    catalog
337}

365/// Require the compiled problem and task-only constraints to be supported by
366/// the implementation installed in this build.
367pub fn validate_installed_implementation(
368    problem: &CompiledProblem,
369    task_requirements: impl IntoIterator<Item = TaskRequirement>,
370) -> Result<(), ImplementationUnavailable> {
371    let mut unsupported = problem
372        .required_capabilities()
373        .iter()
374        .copied()
375        .filter(|capability| !supports_capability(*capability))
376        .map(UnsupportedRequirement::Capability)
377        .collect::<Vec<_>>();
378
379    unsupported.extend(
380        task_requirements
381            .into_iter()
382            .filter(|requirement| !supports_task(*requirement))
383            .map(UnsupportedRequirement::Task),
384    );
385
386    if problem.inputs().observation_snapshot().sources().len() != 1 {
387        unsupported.push(UnsupportedRequirement::SingleObservationSource);
388    }
389    debug_assert_eq!(
390        problem.geometry().domains()[0].role(),
391        &ImageDomainRole::Main
392    );
393    let is_faceted = problem
394        .geometry()
395        .domains()
396        .iter()
397        .any(|domain| domain.facets().len() != 1);
398    if is_faceted && problem.reconstruction().basis() != ReconstructionBasis::Constant {
399        unsupported.push(UnsupportedRequirement::ConstantBasisForFacets);
400    }
401    if coupled_basis_requires_independent_polarization(
402        problem.reconstruction().basis(),
403        problem.reconstruction().polarization().coordinates(),
404    ) {
405        unsupported.push(UnsupportedRequirement::IndependentBasisForPolarizationSelection);
406    }
407    if matches!(
408        problem.geometry().centres().phase_tracking(),
409        PhaseCentreLaw::Observation
410    ) {
411        unsupported.push(UnsupportedRequirement::FixedPhaseCentre);
412    }
413    if !matches!(problem.inputs().model(), ModelStateIdentity::Empty) {
414        unsupported.push(UnsupportedRequirement::EmptyInitialModel);
415    }
416    let installed_response = instrument_response_is_installed(
417        problem
418            .science()
419            .measurement_equation()
420            .instrument_response(),
421        problem.science().instrument_model(),
422        problem.reconstruction().basis(),
423        matches!(
424            problem.geometry().centres().pointing(),
425            casa_imaging_model::PointingCentreLaw::Observation(_)
426        ),
427    );
428    if !installed_response {
429        unsupported.push(UnsupportedRequirement::ScalarInstrumentResponse);
430    }
431    unsupported.sort_unstable();
432    unsupported.dedup();
433    if unsupported.is_empty() {
434        Ok(())
435    } else {
436        Err(ImplementationUnavailable { unsupported })
437    }
438}

440fn instrument_response_is_installed(
441    response: InstrumentResponse,
442    model: Option<InstrumentModel>,
443    basis: ReconstructionBasis,
444    observation_pointing: bool,
445) -> bool {
446    matches!(
447        (response, model, basis, observation_pointing),
448        (InstrumentResponse::Scalar, None, _, _)
449            | (
450                InstrumentResponse::PrimaryBeam,
451                Some(InstrumentModel::CasaAca7mInterferometricDirectPbV1),
452                ReconstructionBasis::TaylorViaChannelMajor { .. },
453                false,
454            )
455            | (
456                InstrumentResponse::PrimaryBeam,
457                Some(InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1),
458                ReconstructionBasis::Constant | ReconstructionBasis::ChannelLocal { .. },
459                true,
460            )
461    )
462}
463
464fn coupled_basis_requires_independent_polarization(
465    basis: ReconstructionBasis,
466    coordinates: &[PolarizationCoordinate],
467) -> bool {
468    matches!(
469        basis,
470        ReconstructionBasis::Taylor { .. }
471            | ReconstructionBasis::TaylorViaChannelMajor { .. }
472            | ReconstructionBasis::JointContinuumLine { .. }
473    ) && coordinates != [PolarizationCoordinate::StokesI]
474}

478        requirement,
479        TaskRequirement::SpectralCube

483            | TaskRequirement::MosaicGridder

488            | TaskRequirement::PerChannelWeightDensity
489            | TaskRequirement::SerialCpu
490            | TaskRequirement::FixedTileCpu
491            | TaskRequirement::RustFft
492    )
493}
