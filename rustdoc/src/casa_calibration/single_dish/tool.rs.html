16/// Family-specific controls on `calibrater.setsolve` for total-power data.
17pub struct SingleDishSolveControls {
18    /// Fraction of a map used for the sky reference spectrum.
19    pub fraction: f32,
20    /// Explicit raster edge integrations, or a negative automatic count.
21    pub numedge: i64,
22    /// Angular radius of the double-circle gain reference region.
23    pub radius: String,
24    /// Smooth the gain reference timeline before taking the ratio.
25    pub smooth: bool,
26    /// The generic amplitude/phase selector could not be resolved.
27    pub invalid_apmode: bool,
28}
29
30pub(crate) struct Solver {
31    pub(crate) request: GainSolveRequest,
32    pub(crate) mode: CalibrationOutputMode,
33    kind: Kind,
34    invalid_apmode: bool,
35}
36
37enum Kind {
38    PositionSwitch,
39    Edges(edges::Selection),
40    Gain(gain::Controls),
41}
42
43impl Solver {
44    pub(crate) fn new(
45        family: &str,
46        request: GainSolveRequest,
47        mode: CalibrationOutputMode,
48        controls: SingleDishSolveControls,
49        legacy: bool,
50    ) -> Result<Self, String> {
51        let fraction = f64::from(controls.fraction);
52        let kind = match family {
53            "SDSKY_PS" => Kind::PositionSwitch,
54            "SDSKY_OTF" if legacy => Kind::Edges(edges::Selection::Otf { fraction }),
55            "SDSKY_OTF" => {
56                return Err("SDSKY_OTF not yet supported via SingleDishOtfCal(msmc)".into());
57            }
58            "SDSKY_RASTER" => Kind::Edges(edges::Selection::Raster {
59                fraction,
60                noff: controls.numedge,
61            }),
62            "SDGAIN_OTFD" => Kind::Gain(gain::Controls {
63                radius: parse_radius(&controls.radius)?,
64                smooth: controls.smooth,
65                solve_mode: request.solve_mode,
66                correlation_dependent_flags: false,
67            }),
68            _ => return Err(format!("Unknown single-dish solve type {family}")),
69        };
70        Ok(Self {
71            request,
72            mode,
73            kind,
74            invalid_apmode: controls.invalid_apmode,
75        })
76    }
77
78    pub(crate) fn subtype(&self) -> &'static str {
79        match self.kind {
80            Kind::PositionSwitch => position_switch::SUBTYPE,
81            Kind::Edges(edges::Selection::Otf { .. }) => "SDSKY_OTF",
82            Kind::Edges(edges::Selection::Raster { .. }) => "SDSKY_RASTER",
83            Kind::Gain(_) => "SDGAIN_OTFD",
84        }
85    }
86
87    pub(crate) fn gain_controls(&self) -> Option<(f64, bool)> {
88        match self.kind {
89            Kind::Gain(controls) => Some((controls.radius.unwrap_or(0.), controls.smooth)),
90            _ => None,
91        }
92    }
93
94    pub(crate) fn recorded_output(&self) -> Option<&PathBuf> {
95        self.gain_controls()
96            .is_none()
97            .then_some(&self.request.output_table)
98    }
99
100    pub(crate) fn solve(
101        &self,
102        ms_path: &Path,
103        apply: ApplyPlanRequest,
104        legacy: bool,
105        correlation_dependent_flags: bool,
106        activity: &mut ToolSolveActivity,
107    ) -> Result<Option<SolveStats>, String> {
108        if legacy == matches!(self.kind, Kind::Gain(_)) {
109            return Err(if legacy {
110                "Attempt to call un-implemented selfSolveOne()".into()
111            } else {
112                format!("Can't do selfGatherAndSolve for {}", self.subtype())
113            });
114        }
115        let ms = MeasurementSet::open(ms_path).map_err(|error| error.to_string())?;
116        let output = &self.request.output_table;
117        if overlaps_measurement_set(&ms, output)? {
118            return Err("The MeasurementSet and output calibration table overlap.".into());
119        }
120        let _output_locks =
121            lock_solve_output(output, self.mode).map_err(|error| error.to_string())?;
122        let selection = resolve_casa_calibration_selection(&ms, &apply.selection)
123            .map_err(|error| error.to_string())?;
124        if let Kind::Gain(mut controls) = self.kind {
125            if self.invalid_apmode {
126                return Err("enforceAPonSoln divide-by-zero error.".into());
127            }
128            controls.correlation_dependent_flags = correlation_dependent_flags;
129            if self.mode == CalibrationOutputMode::Append && !output.join("table.dat").is_file() {
130                return Err(format!(
131                    "Table {}/SPECTRAL_WINDOW does not exist",
132                    output.display()
133                ));
134            }
135            let priors = if apply.calibration_tables.is_empty() && !apply.parang {
136                None
137            } else {
138                let plan = plan_apply(&ms, &apply).map_err(|error| error.to_string())?;
139                Some(SolvePriorApply::single_dish(&ms, plan).map_err(|error| error.to_string())?)
140            };
141            let (_, stats) = gain::generate(
142                &ms,
143                &selection,
144                priors.as_ref(),
145                output,
146                self.mode,
147                controls,
148                Some(activity),
149            )
150            .map_err(|error| error.into_message())?;
151            return Ok(stats);
152        }
153        let rows = selection.row_indices().collect::<Vec<_>>();
154        match self.kind {
155            Kind::PositionSwitch => position_switch::generate(&ms, &rows, output, self.mode),
156            Kind::Edges(controls) => edges::generate(&ms, &rows, output, self.mode, controls),
157            Kind::Gain(_) => unreachable!("double-circle gain uses its own selection"),
158        }?;
159        Ok(None)
160    }
161}

163fn parse_radius(text: &str) -> Result<Option<f64>, String> {
164    if text.trim().is_empty() {
165        return Ok(None);
166    }
167    let radians = "rad".parse::<Unit>().expect("standard angle unit");
168    let quantity = text
169        .parse::<Quantity>()
170        .map_err(|error| error.to_string())?;
171    let radius = quantity
172        .get_value_in(&radians)
173        .map_err(|error| error.to_string())?;
174    if !radius.is_finite() || radius < 0. {
175        return Err("Invalid central disk radius".into());
176    }
177    Ok(Some(radius))
178}
