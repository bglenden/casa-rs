2//! `sm.settrop`: the phase a turbulent screen of water vapour adds above
3//! each antenna, channel by channel (CASA's `AtmosCorruptor` in its
4//! `individual` and `screen` modes, a `Tf Jones` term; #841 Phase 4d).
5//!
6//! # What CASA computes
7//!
8//! Measured against CASA 6.7.6 (`settrop` probes of 64-channel windows over
9//! the 183 GHz and 325 GHz water lines, #841 Phase 4d): `Simulator::settrop`
10//! adds a `Tf Jones` term whose value for antenna `a` in channel `c` at a
11//! time in slot `s` is `exp(i·δ_a(s)·φ(c))`, where
12//! - `φ(c)` is ATM's zenith wet phase delay for the mean water column
13//!   `pwv`, radians, at a site `settrop` fixes (5000 m, 560 hPa, 270 K,
14//!   20 %, a 2 km water scale height), whatever `setnoise` set;
15//! - `δ_a(s)` is the fractional excess of water vapour over `pwv` above the
16//!   antenna in slot `s`: fractional Brownian motion of spectral exponent
17//!   `beta`, scaled to a root-mean-square of `deltapwv` (at most 1) about
18//!   its mean, the mean kept. The slots are even, 10 s wide (or `simint`,
19//!   at least 0.1 s), spanning the integrations observed so far, and a
20//!   solution takes the slot nearest its time (`CalCorruptor::setEvenSlots`
21//!   and `setCurrTime`). In `individual` mode each antenna has a series of
22//!   its own; in `screen` mode one two-dimensional screen of pixels half the
23//!   smallest dish across is blown along the array's east axis at
24//!   `windspeed`, and an antenna reads the pixel it is over, the screen
25//!   moved on by `floor(s·Δt·v/pixel)` pixels in slot `s`
26//!   (`AtmosCorruptor::cphase`).
27//!
28//! The term is phase only, so an autocorrelation is unchanged, and it has a
29//! solution per integration, window and antenna, as CASA's table has.
30//! The screen sits at the zenith: CASA neither scales the phase by the
31//! airmass nor moves an antenna's pixel with the direction it observes
32//! (its own `RI TODO`s), and casa-rs follows it; nor does it interpolate
33//! between pixels.
34//!
35//! # Where casa-rs departs
36//!
37//! Recorded in `docs/CASA (C++) bugs.md`:
38//! - **The wet path.** CASA's `φ` is ATM's *dispersive* wet phase delay
39//!   alone, the part of the water vapour's refraction its lines add. The
40//!   path a column of water vapour adds is mostly its non-dispersive part
41//!   (`ℒ_V ≃ 6.3 w`, Thompson, Moran & Swenson 3rd ed. eq. 13.19, p. 702):
42//!   at `settrop`'s site the whole wet delay is 8.5 to 940 times the
43//!   dispersive part between 345 and 43 GHz. casa-rs takes both: `φ` is the
44//!   whole wet phase delay.
45//! - **The channels' frequencies.** As for `tsys-atm` noise
46//!   ([`super::tsys_atm`]), CASA evaluates ATM half a band below each
47//!   channel, and reads past the end of a later one-channel window's map;
48//!   casa-rs evaluates it at each channel's CHAN_FREQ.
49//! - **The screen.** CASA's two-dimensional spectrum has amplitudes
50//!   `|(i, j)|^−(β+½)/2` in index units, on unwrapped indices (so the
51//!   negative wavenumbers of the full axis get the smallest amplitudes) and
52//!   on a screen whose two axes differ, and it writes past the end of its
53//!   spectrum. casa-rs draws `|k|^−(β+1)/2` on the wavenumbers themselves
54//!   (cycles per pixel, wrapped), the spectrum of a two-dimensional
55//!   fractional Brownian motion whose one-dimensional cuts have the
56//!   spectral exponent `beta`, as `individual` mode's series do, with the
57//!   conjugate pairs the real transform needs on its boundary columns.
58//! - **The series.** As for `setgain(mode='fbm')` ([`super::jones`]), CASA
59//!   reads a series' last two slots from past the end of its transform and
60//!   normalises with them; casa-rs transforms every slot, and refuses a
61//!   track of fewer than [`FEWEST_SERIES_SLOTS`] slots, on which a series
62//!   holds only its mean.
63//! - As for `setgain`, every antenna is simulated where CASA's
64//!   autocorrelation rows reset theirs to 1, and with `simint` longer than
65//!   an integration each solution is stamped with its own integration's
66//!   time, where CASA stamps it halfway between its interval's first
67//!   integration and its own and so applies a slot from earlier in the
68//!   interval.
69//!
70//! The draws are casa-rs's (Philox streams of their own purpose byte), so
71//! realisations are compared with CASA's statistically.
72

79/// The purpose byte of tropospheric screens' streams.
80const PURPOSE_TROPOSPHERE_SCREEN: u32 = 0x05;
81
82/// The Earth's radius CASA projects antenna positions with, metres.
83const EARTH_RADIUS_M: f64 = 6_371_000.0;
84
85/// Pixels CASA leaves around the antennas on each screen axis.
86const SCREEN_BORDER_PIXELS: f64 = 2.0;
87
88/// The margin CASA gives the screen along the wind, beyond what the wind
89/// blows over the track.
90const BLOW_MARGIN: f64 = 1.05;
91
92/// The longest screen CASA makes for its width: a screen more than this
93/// many times as long as wide is widened.
94const LONGEST_ASPECT: usize = 5;
95
96/// The most spectrum elements a screen may have: 2⁴⁰, 8 TiB of them, past
97/// any budget, so that the bytes counted for admission cannot overflow.
98pub(crate) const LARGEST_SCREEN_ELEMENTS: usize = 1 << 40;

100/// The fewest slots on which an `individual` series varies: CASA's spectrum
101/// fills bins `i < N/2 − 1`, so on fewer than six it holds only the mean
102/// and the series, scaled about it, is flat (#899 review).
103pub(crate) const FEWEST_SERIES_SLOTS: usize = 6;

105/// A tropospheric term's phases: each solution's fractional excess of water
106/// vapour above each antenna, and each window's wet phase delay per channel
107/// for the mean water column.
108#[derive(Debug, Clone, PartialEq)]
109pub(crate) struct TroposphericPhase {
110    /// `[block · antennas + antenna]`: δ, the excess over `pwv` as a
111    /// fraction of it.
112    excess: Vec<f32>,
113    /// `[window]`: each channel's zenith wet phase delay for `pwv`, radians;
114    /// `None` for a window defined after the term was set.
115    delays: Vec<Option<Vec<f64>>>,
116}
117
118impl TroposphericPhase {
119    /// The phases of `excess` (`[block · antennas + antenna]`) through
120    /// `delays` (`[window]`, radians per channel).
121    pub(crate) fn new(excess: Vec<f32>, delays: Vec<Option<Vec<f64>>>) -> Self {
122        Self { excess, delays }
123    }
124
125    /// The heap the phases keep.
126    pub(crate) fn bytes(&self) -> u64 {
127        (self.excess.capacity() * size_of::<f32>()
128            + self.delays.capacity() * size_of::<Option<Vec<f64>>>()
129            + self
130                .delays
131                .iter()
132                .flatten()
133                .map(|delays| delays.capacity() * size_of::<f64>())
134                .sum::<usize>()) as u64
135    }
136
137    /// The heap phases of `blocks` blocks of `antennas` antennas keep for
138    /// windows of `channels` channels each.
139    pub(crate) fn bytes_for(blocks: usize, antennas: usize, channels: &[usize]) -> u64 {
140        (blocks * antennas * size_of::<f32>()
141            + channels.len() * size_of::<Option<Vec<f64>>>()
142            + channels.iter().sum::<usize>() * size_of::<f64>()) as u64
143    }
144
145    /// Spectral window `window`'s channels, when the term has its delays.
146    pub(crate) fn channels(&self, window: usize) -> Option<usize> {
147        self.delays.get(window)?.as_ref().map(Vec::len)
148    }
149
150    /// Antenna `antenna`'s value in each channel of `window` in block
151    /// `block` of `antennas` antennas, `exp(i·δ·φ(c))`, into `phasors`.
152    pub(crate) fn phasors(
153        &self,
154        (window, block): (usize, usize),
155        (antenna, antennas): (usize, usize),
156        phasors: &mut [Complex32],
157    ) {
158        let excess = f64::from(self.excess[block * antennas + antenna]);
159        let delays = self.delays[window]
160            .as_ref()
161            .expect("a tropospheric term has the delays of every window it has solutions in");
162        for (phasor, &delay) in phasors.iter_mut().zip(delays) {
163            let (sin, cos) = (excess * delay).sin_cos();
164            *phasor = Complex32::new(cos as f32, sin as f32);
165        }
166    }
167}

169/// Each channel's zenith wet phase delay, radians, of `profile`'s channels
170/// for a water column of `water_column_m`: the dispersive part (the lines)
171/// and the non-dispersive part (the continuum) together.
172pub(crate) fn wet_delays(
173    profile: &AbsorptionProfile<ChannelTotals>,
174    water_column_m: f64,
175) -> Vec<f64> {
176    (0..profile.channels().len())
177        .map(|channel| {
178            (profile.dispersive_h2o_phase_delay_deg(channel, water_column_m)
179                + profile.non_dispersive_h2o_phase_delay_deg(channel, water_column_m))
180            .to_radians()
181        })
182        .collect()
183}

185/// `individual` mode's excess: for each of `antennas` antennas a series of
186/// fractional Brownian motion of spectral exponent `beta` over `slots`,
187/// scaled to a root-mean-square of `scale` about its mean, the mean kept
188/// (`AtmosCorruptor::initialize(Seed, Beta, scale, rxtype)`), drawn from
189/// `seed`'s streams. `[slot · antennas + antenna]`.
190pub(crate) fn antenna_series(
191    seed: i32,
192    (beta, scale): (f64, f64),
193    slots: usize,
194    antennas: usize,
195) -> Vec<f32> {
196    let key = [seed as u32, PURPOSE_TROPOSPHERE_SCREEN << 24];
197    let mut fft = super::jones::ScreenFft::new(slots);
198    let mut excess = vec![0.0; slots * antennas];
199    for antenna in 0..antennas {
200        let series = fft.normalised_fbm(key, [antenna as u32, 0, 0, 0], (beta, scale));
201        for (slot, value) in series.into_iter().enumerate() {
202            excess[slot * antennas + antenna] = value as f32;
203        }
204    }
205    excess
206}

208/// Where a `screen` mode screen lies over the array, as CASA lays it
209/// (`AtmosCorruptor::initialize(…, antcols)`).
210#[derive(Debug, Clone, PartialEq)]
211pub(crate) struct ScreenLayout {
212    /// The pixel's side, metres: half the smallest dish.
213    pub(crate) pixel_m: f64,
214    /// `[rows, columns]`: north by east.
215    pub(crate) shape: [usize; 2],
216    /// Each antenna's pixel, (row, column), before the wind.
217    pub(crate) antennas: Vec<[usize; 2]>,
218}
219
220impl ScreenLayout {
221    /// The screen over antennas at `positions_itrf_m` with dishes of
222    /// `diameters_m`, long enough for a wind of `windspeed_m_s` over a
223    /// track of `track_s`: CASA projects each position's longitude and
224    /// latitude onto a sphere of the Earth's radius, east scaled by the
225    /// cosine of the mean latitude, from the westmost and southmost
226    /// antennas, with two pixels' border and the wind's run along the east
227    /// axis, 5 % more; a screen more than five times as long as wide is
228    /// widened to a fifth of its length.
229    ///
230    /// # Errors
231    ///
232    /// A pixel that is not a positive length (a dish of no size), or a
233    /// screen of more than [`LARGEST_SCREEN_ELEMENTS`] spectrum elements or
234    /// whose sides are not finite (a wind run past single precision), which
235    /// CASA would make or index beyond (#899 review).
236    pub(crate) fn new(
237        positions_itrf_m: &[[f64; 3]],
238        diameters_m: &[f64],
239        (windspeed_m_s, track_s): (f64, f64),
240    ) -> Result<Self, String> {
241        let pixel_m = 0.5 * diameters_m.iter().copied().fold(f64::INFINITY, f64::min);
242        if !(pixel_m.is_finite() && pixel_m > 0.0) {
243            return Err(format!(
244                "a screen's pixels are half the smallest dish, here {pixel_m} m"
245            ));
246        }
247        let north: Vec<f64> = positions_itrf_m
248            .iter()
249            .map(|[x, y, z]| z.atan2(x.hypot(*y)) * EARTH_RADIUS_M)
250            .collect();
251        let mean_latitude = north.iter().sum::<f64>() / north.len().max(1) as f64 / EARTH_RADIUS_M;
252        let east: Vec<f64> = positions_itrf_m
253            .iter()
254            .map(|[x, y, _]| y.atan2(*x) * EARTH_RADIUS_M * mean_latitude.cos())
255            .collect();
256        let from_least = |values: &[f64]| {
257            let least = values.iter().copied().fold(f64::INFINITY, f64::min);
258            values
259                .iter()
260                .map(|value| (value - least) / pixel_m)
261                .collect::<Vec<_>>()
262        };
263        let (east, north) = (from_least(&east), from_least(&north));
264        let most = |values: &[f64]| values.iter().copied().fold(0.0, f64::max);
265        // CASA holds the track and the run of the wind as `Float`s.
266        let run_m = f64::from((windspeed_m_s * f64::from(track_s as f32) * BLOW_MARGIN) as f32);
267        let sides = [
268            (most(&north) + SCREEN_BORDER_PIXELS).ceil(),
269            (most(&east) + SCREEN_BORDER_PIXELS + run_m / pixel_m).ceil(),
270        ];
271        // Both sides below 2³¹ (FFTW's), so their product is exact.
272        if !sides
273            .iter()
274            .all(|&side| side.is_finite() && side < f64::from(i32::MAX))
275        {
276            return Err(format!(
277                "a screen of {} by {} pixels cannot be made",
278                sides[0], sides[1]
279            ));
280        }
281        let columns = sides[1] as usize;
282        let mut rows = sides[0] as usize;
283        if columns > LONGEST_ASPECT * rows {
284            rows = columns / LONGEST_ASPECT;
285        }
286        let layout = Self {
287            pixel_m,
288            shape: [rows, columns],
289            antennas: north
290                .iter()
291                .zip(&east)
292                .map(|(&row, &column)| [row as usize, column as usize])
293                .collect(),
294        };
295        if layout.spectrum_len() > LARGEST_SCREEN_ELEMENTS {
296            return Err(format!(
297                "a screen of {rows} by {columns} pixels is too large to make"
298            ));
299        }
300        Ok(layout)
301    }
302
303    /// The real transform's elements: the screen's spectrum, which the
304    /// real screen overwrites in place.
305    pub(crate) fn spectrum_len(&self) -> usize {
306        self.shape[0] * (self.shape[1] / 2 + 1)
307    }
308
309    /// The columns the wind has moved the screen by in slot `slot` of
310    /// `slots` at `windspeed_m_s`.
311    fn blown(&self, slots: &Slots, slot: usize, windspeed_m_s: f64) -> usize {
312        ((slot as f64 * slots.width_s * windspeed_m_s) / self.pixel_m).floor() as usize
313    }
314}

316/// The spectrum of a screen of fractional Brownian motion: at wavenumber
317/// `k` (cycles per pixel, the full axis wrapped), a normal amplitude of
318/// `|k|^−(β+1)/2` and a uniform phase, drawn from `seed`'s streams by bin.
319/// The amplitudes are relative to the largest (at the smallest wavenumber,
320/// or the largest when the exponent is positive), so no power overflows
321/// (#899 review); the screen is scaled afterwards, so the factor is
322/// immaterial.
323struct ScreenSpectrum {
324    key: [u32; 2],
325    shape: [usize; 2],
326    exponent: f64,
327    reference: f64,
328}
329
330impl ScreenSpectrum {
331    fn new(seed: i32, beta: f64, shape: [usize; 2]) -> Self {
332        let exponent = -0.5 * (beta + 1.0);
333        let [rows, columns] = shape;
334        let smallest = 1.0 / rows.max(columns) as f64;
335        let largest =
336            ((rows / 2) as f64 / rows as f64).hypot((columns / 2) as f64 / columns as f64);
337        Self {
338            key: [seed as u32, (PURPOSE_TROPOSPHERE_SCREEN << 24) | 1],
339            shape,
340            exponent,
341            reference: if exponent <= 0.0 { smallest } else { largest },
342        }
343    }
344
345    /// Bin (`row`, `column`)'s amplitude: zero at the mean.
346    fn amplitude(&self, row: usize, column: usize) -> f64 {
347        let [rows, columns] = self.shape;
348        let north = row.min(rows - row) as f64 / rows as f64;
349        let east = column.min(columns - column) as f64 / columns as f64;
350        let wavenumber = north.hypot(east);
351        if wavenumber == 0.0 {
352            return 0.0;
353        }
354        (wavenumber / self.reference).powf(self.exponent)
355    }
356
357    /// Bin (`row`, `column`)'s value: its amplitude times a normal draw, at
358    /// a uniform phase.
359    fn draw(&self, row: usize, column: usize) -> Complex64 {
360        let phase =
361            2.0 * std::f64::consts::PI * uniform(self.key, [row as u32, column as u32, 0, 0]);
362        Complex64::from_polar(self.real(row, column), phase)
363    }
364
365    /// Bin (`row`, `column`)'s real value, for a bin its own conjugate: its
366    /// amplitude times a normal draw.
367    fn real(&self, row: usize, column: usize) -> f64 {
368        let [normal, _] = normal_pair(self.key, [row as u32, column as u32, 1, 0]);
369        self.amplitude(row, column) * normal
370    }
371}

373/// `screen` mode's excess: one screen of fractional Brownian motion of
374/// spectral exponent `beta` laid out by `layout`, scaled to a
375/// root-mean-square of `scale` about its mean, the mean kept, drawn from
376/// `seed`'s streams; each antenna reads its pixel, the screen blown on by
377/// the wind at `windspeed_m_s` in each of `slots`. The transform runs on
378/// `threads` threads. `[slot · antennas + antenna]`.
379///
380/// # Errors
381///
382/// The screen's transform cannot be planned.
383pub(crate) fn screen_excess(
384    seed: i32,
385    (beta, scale): (f64, f64),
386    layout: &ScreenLayout,
387    (slots, windspeed_m_s): (&Slots, f64),
388    threads: usize,
389) -> Result<Vec<f32>, casa_fft::FftError> {
390    let [rows, columns] = layout.shape;
391    let half = columns / 2 + 1;
392    let mut fft = RealFft2::<f32>::with_threads([rows, columns], threads.max(1))?
393        .with_estimated_plan()
394        .with_owned_plan();
395    let mut plane = vec![Complex32::new(0.0, 0.0); layout.spectrum_len()];
396    let spectrum = ScreenSpectrum::new(seed, beta, [rows, columns]);
397    // The half spectrum's first column, and its last when the width is
398    // even, hold both signs of the full axis's wavenumbers: row r is the
399    // conjugate of row N − r there, and a row that is its own mirror is
400    // real (#899 review: drawn apart, C2R halves their power).
401    let paired =
402        |column: usize| column == 0 || (columns.is_multiple_of(2) && column == columns / 2);
403    for (row, values) in plane.chunks_exact_mut(half).enumerate() {
404        let mirror = (rows - row) % rows;
405        for (column, value) in values.iter_mut().enumerate() {
406            let drawn = if !paired(column) || row < mirror {
407                spectrum.draw(row, column)
408            } else if row == mirror {
409                Complex64::new(spectrum.real(row, column), 0.0)
410            } else {
411                spectrum.draw(mirror, column).conj()
412            };
413            *value = Complex32::new(drawn.re as f32, drawn.im as f32);
414        }
415    }
416    fft.inverse(&mut plane)?;
417    // The real screen, `2·half` scalars a row: a column's pixel is its pair's
418    // real or imaginary part.
419    debug_assert_eq!(fft.real_row_stride(), 2 * half);
420    let pixel = |row: usize, column: usize| -> f64 {
421        let value = plane[row * half + column / 2];
422        f64::from(if column.is_multiple_of(2) {
423            value.re
424        } else {
425            value.im
426        })
427    };
428    let count = (rows * columns) as f64;
429    let mean = (0..rows)
430        .flat_map(|row| (0..columns).map(move |column| (row, column)))
431        .map(|(row, column)| pixel(row, column))
432        .sum::<f64>()
433        / count;
434    let variance = (0..rows)
435        .flat_map(|row| (0..columns).map(move |column| (row, column)))
436        .map(|(row, column)| (pixel(row, column) - mean).powi(2))
437        .sum::<f64>()
438        / count;
439    let factor = if variance > 0.0 {
440        scale / variance.sqrt()
441    } else {
442        0.0
443    };
444    let antennas = layout.antennas.len();
445    let mut excess = vec![0.0; slots.count * antennas];
446    for slot in 0..slots.count {
447        let blown = layout.blown(slots, slot, windspeed_m_s);
448        for (antenna, &[row, column]) in layout.antennas.iter().enumerate() {
449            excess[slot * antennas + antenna] = (pixel(row, column + blown) * factor) as f32;
450        }
451    }
452    Ok(excess)
453}
