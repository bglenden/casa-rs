2//! Index selected visibility rows into antenna, baseline and time panels.
3
4use std::collections::{BTreeMap, HashMap};
5
6use casa_core::quanta::MvTime;
7
8use casa_ms::MeasurementSet;
9use casa_ms::subtables::MsAntenna;
10
11use super::{MsIterationAxis, MsScatterPanelPayload, MsScatterSeries, PointBudget};
12
13#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
14enum PanelId {
15    Antenna(i32),
16    Baseline(i32, i32),
17    Time(usize),
18}
19
20impl PanelId {
21    fn label(self, antenna: &MsAntenna<'_>, times: &[f64]) -> Result<(String, String), String> {
22        let station_name = |id| {
23            if let Ok(row) = usize::try_from(id) {
24                let name = antenna.name(row).map_err(|error| error.to_string())?;
25                let station = antenna.station(row).map_err(|error| error.to_string())?;
26                Ok(format!("{name}@{station}"))
27            } else {
28                Ok::<_, String>("*".into())
29            }
30        };
31        Ok(match self {
32            Self::Antenna(id) => (format!("antenna-{id}"), station_name(id)?),
33            Self::Baseline(left, right) => {
34                let mut label = format!("{} & {}", station_name(left)?, station_name(right)?);
35                if let (Ok(left), Ok(right)) = (usize::try_from(left), usize::try_from(right)) {
36                    let a = antenna.position(left).map_err(|error| error.to_string())?;
37                    let b = antenna.position(right).map_err(|error| error.to_string())?;
38                    let distance = a
39                        .into_iter()
40                        .zip(b)
41                        .map(|(a, b)| (a - b).powi(2))
42                        .sum::<f64>()
43                        .sqrt();
44                    label.push_str(&format!("_{distance:.0}m"));
45                }
46                (format!("baseline-{left}-{right}"), label)
47            }
48            Self::Time(slot) => (
49                format!("time-{slot}"),
50                MvTime::from_mjd_seconds(times[slot]).format_time(4),
51            ),
52        })
53    }
54}
55
56pub(super) fn partition(
57    ms: &MeasurementSet,
58    axis: MsIterationAxis,
59    rows: &[usize],
60    [antenna1, antenna2]: [&[i32]; 2],
61    times: &[f64],
62    series: Vec<MsScatterSeries>,
63    point_budget: &mut PointBudget,
64) -> Result<Vec<MsScatterPanelPayload>, String> {
65    let mut first_time_slots = HashMap::new();
66    let membership: HashMap<_, _> = rows
67        .iter()
68        .copied()
69        .enumerate()
70        .map(|(slot, row)| {
71            let left = antenna1[slot];
72            let right = antenna2[slot];
73            let ids = match axis {
74                MsIterationAxis::Antenna => [
75                    (left >= 0).then_some(PanelId::Antenna(left)),
76                    (right >= 0 && right != left).then_some(PanelId::Antenna(right)),
77                ],
78                MsIterationAxis::Baseline => [Some(PanelId::Baseline(left, right)), None],
79                MsIterationAxis::Time => {
80                    let first = *first_time_slots
81                        .entry(times[slot].to_bits())
82                        .or_insert(slot);
83                    [Some(PanelId::Time(first)), None]
84                }
85                _ => unreachable!("only row-based iteration uses row membership"),
86            };
87            (row, ids)
88        })
89        .collect();
90    let mut panels = membership
91        .values()
92        .copied()
93        .flatten()
94        .flatten()
95        .map(|id| (id, Vec::new()))
96        .collect::<BTreeMap<_, _>>();
97    for source in series {
98        let mut split = BTreeMap::new();
99        for (index, point) in source.provenance.iter().enumerate() {
100            let ids = membership[&point.row];
101            if ids[0].is_some() && ids[1].is_some() {
102                point_budget.record_points(1, axis.display_name())?;
103            }
104            for id in ids.into_iter().flatten() {
105                let target = split.entry(id).or_insert_with(|| MsScatterSeries {
106                    label: source.label.clone(),
107                    color_group: source.color_group.clone(),
108                    y_axis: source.y_axis,
109                    y_index: source.y_index,
110                    points: Vec::new(),
111                    provenance: Vec::new(),
112                    plotms_metadata: Vec::new(),
113                    plotms_reporting_frequencies_hz: Vec::new(),
114                    plotms_averaged_ids: None,
115                });
116                target.points.push(source.points[index]);
117                target.provenance.push(point.clone());
118                target
119                    .plotms_metadata
120                    .extend(source.plotms_metadata.get(index).copied());
121                target
122                    .plotms_reporting_frequencies_hz
123                    .extend(source.plotms_reporting_frequencies_hz.get(index).copied());
124            }
125        }
126        for (id, series) in split {
127            panels
128                .get_mut(&id)
129                .expect("selected iteration panel")
130                .push(series);
131        }
132    }
133    let antenna = ms.antenna().map_err(|error| error.to_string())?;
134    panels
135        .into_iter()
136        .map(|(id, series)| {
137            let (key, label) = id.label(&antenna, times)?;
138            Ok(MsScatterPanelPayload {
139                summary: format!(
140                    "{label}. Points={}.",
141                    series.iter().map(|entry| entry.points.len()).sum::<usize>()
142                ),
143                key,
144                label,
145                series,
146            })
147        })
148        .collect()
149}
