11/// Time lookup options used when imposing calibration errors on simulated DATA.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub struct TimeInterpolationOptions {
14    /// Amplitude/phase curve within each selected time series.
15    pub mode: TimeInterpolation,
16    /// Scale phase by data/calibration reference frequency, tracking cycles.
17    pub phase_delay: bool,
18    /// Partition by OBSERVATION_ID when calibration and MS observations agree.
19    pub per_observation: bool,
20    /// Partition by SCAN_NUMBER; takes precedence over per-observation lookup.
21    pub per_scan: bool,
22    /// Number of OBSERVATION rows in the simulated MeasurementSet.
23    pub observation_count: usize,
24}
25
26impl TimeInterpolationOptions {
27    /// A curve across all fields, without phase-delay or time partitioning.
28    #[must_use]
29    pub const fn new(mode: TimeInterpolation) -> Self {
30        Self {
31            mode,
32            phase_delay: false,
33            per_observation: false,
34            per_scan: false,
35            observation_count: 1,
36        }
37    }
38    fn contextual(self) -> bool {
39        self.phase_delay || self.per_observation || self.per_scan
40    }
41}

43impl CalibrationTableLayout {
44    /// Extra retained heap for segmented series and phase-cycle arrays.
45    #[must_use]
46    pub fn context_kept_bytes(self, options: TimeInterpolationOptions) -> u64 {
47        if options.contextual() {
48            (self.rows as u64)
49                .saturating_mul(128)
50                .saturating_add(if options.phase_delay {
51                    (self.values as u64).saturating_mul(4)
52                } else {
53                    0
54                })
55        } else {
56            0
57        }
58    }
59    /// Extra loading heap for scalar row headers and their grouping positions.
60    #[must_use]
61    pub fn context_loading_bytes(self, options: TimeInterpolationOptions) -> u64 {
62        if options.contextual() {
63            (self.rows as u64).saturating_mul(96)
64        } else {
65            0
66        }
67    }
68}

86impl CalibrationTableLookup {
87    pub(in crate::lookup) fn fringe_phases(
88        &self,
89        observation: i32,
90        scan: i32,
91        window: i32,
92        antenna: i32,
93        time: f64,
94        frequency: f64,
95    ) -> Result<[f32; 2], ApplyExecutionError> {
96        let plan = self
97            .advanced_time
98            .as_ref()
99            .ok_or_else(|| invalid(&self.path, "fringe requires real time curves"))?;
100        let segment = match plan.scope {
101            TimeScope::All => -1,
102            TimeScope::Observation => observation,
103            TimeScope::Scan => scan,
104        };
105        let rows = plan
106            .series
107            .get(&(segment, window, antenna))
108            .or_else(|| plan.series.get(&(segment, window, -1)))
109            .filter(|rows| !rows.is_empty())
110            .ok_or_else(|| invalid(&self.path, "fringe time series is missing"))?;
111        let reference = (rows[0].time_seconds - 1.).floor();
112        let query = relative(time, reference);
113        let at = |index: usize| relative(rows[index].time_seconds, reference);
114        let upper = rows.partition_point(|row| relative(row.time_seconds, reference) < query);
115        let exact = upper == 0 || upper == rows.len() || at(upper) == query;
116        let mut output = [0.; 2];
117        for (receptor, phase) in output.iter_mut().enumerate() {
118            let advanced = |index| {
119                crate::lookup::fringe::advanced_phase(
120                    &rows[index],
121                    receptor,
122                    time,
123                    frequency,
124                    &self.path,
125                )
126            };
127            *phase = if exact || plan.mode == TimeInterpolation::Nearest {
128                let index = if exact {
129                    upper.min(rows.len() - 1)
130                } else if at(upper) - query < query - at(upper - 1) {
131                    upper
132                } else {
133                    upper - 1
134                };
135                advanced(index)? as f32
136            } else {
137                let weight = ((at(upper) - query) / (at(upper) - at(upper - 1))) as f32;
138                let phasor = |phase: f64| Complex32::new(phase.cos() as f32, phase.sin() as f32);
139                let lower = phasor(advanced(upper - 1)?);
140                let higher = phasor(advanced(upper)?);
141                (weight * lower + (1. - weight) * higher).arg()
142            };
143        }
144        Ok(output)
145    }

148    /// reference frequency axis even for frequency-independent parameters.

160    /// Simulator layout with the data frequencies used to sample legacy
161    /// BPOLY coefficients. Other table families retain their own grids.
162    ///
163    /// # Errors
164    /// Shape metadata is invalid, or BPOLY has no supplied data grid.
165    pub fn layout_for_simulation_on_grids(
166        path: impl AsRef<Path>,
167        options: TimeInterpolationOptions,
168        frequencies: &[Vec<f64>],
169    ) -> Result<CalibrationTableLayout, ApplyExecutionError> {
170        let path = path.as_ref();
171        let table =
172            Table::open(TableOptions::new(path)).map_err(|e| invalid(path, &e.to_string()))?;
173        let bpoly = table.info().sub_type == "BPOLY";
174        let real = real_parameters(&table);
175        drop(table);
176        if bpoly {
177            return Self::bpoly_layout(path, frequencies);
178        }
179        let mut layout = Self::parameter_layout(path, if real { "FPARAM" } else { "CPARAM" })?;
180        if (options.phase_delay || real)
181            && layout.frequency_values == 0
182            && let Some(spectral) = super::super::spectral_window_path(path)?
183        {
184            layout.frequency_values = super::super::shape_counts(&spectral, "CHAN_FREQ")
185                .map_err(|e| invalid(path, &e))?
186                .1;
187        }
188        Ok(layout)
189    }

203    /// Load supplied simulator tables, sampling legacy BPOLY on the data
204    /// frequency grids before the ordinary time interpolation is prepared.
205    /// Admit `layout_for_simulation_on_grids` and time/context costs first.
206    ///
207    /// # Errors
208    /// The table, frequency grids, or time curves are invalid.
209    pub fn open_for_simulation_on_grids(
210        path: impl AsRef<Path>,
211        options: TimeInterpolationOptions,
212        frequencies: &[Vec<f64>],
213    ) -> Result<Self, ApplyExecutionError> {
214        let basic = if options.mode == TimeInterpolation::Nearest {
215            ApplyInterpolationMode::Nearest
216        } else {
217            ApplyInterpolationMode::Linear
218        };
219        let path = path.as_ref();
220        let table =
221            Table::open(TableOptions::new(path)).map_err(|e| invalid(path, &e.to_string()))?;
222        let real = real_parameters(&table);
223        let bpoly = table.info().sub_type == "BPOLY";
224        drop(table);
225        let mut lookup = if bpoly {
226            Self::open_bpoly(path, basic, frequencies)?
227        } else {
228            Self::open_parameters(path, basic, FieldSelection::AnyField, real)?
229        };
230        // CASA's BPOLY-to-B conversion writes unpartitioned (-1) scan and
231        // observation headers. The legacy input has neither modern column.
232        let options = if bpoly {
233            TimeInterpolationOptions {
234                per_scan: false,
235                per_observation: false,
236                phase_delay: false,
237                ..options
238            }
239        } else {
240            options
241        };
242        if real
243            || options.contextual()
244            || matches!(
245                options.mode,
246                TimeInterpolation::Cubic | TimeInterpolation::Spline
247            )
248        {
249            let plan = TimePlan::new(&mut lookup, options, real)?;
250            lookup.advanced_time = Some(plan);
251        }
252        Ok(lookup)
253    }
254
255    /// Load a baseline-based complex table for simulation. Query indices
256    /// are `min(p,q) * antennas + max(p,q)`; numerical interpolation and
257    /// flags are the ordinary simulator lookup's. Admit the ordinary layout,
258    /// time and context costs before loading.
259    ///
260    /// # Errors
261    /// The table, baseline coordinates, or interpolation parameters are invalid.
262    pub fn open_for_baseline_simulation(
263        path: impl AsRef<Path>,
264        options: TimeInterpolationOptions,
265        antennas: usize,
266    ) -> Result<Self, ApplyExecutionError> {
267        let basic = if options.mode == TimeInterpolation::Nearest {
268            ApplyInterpolationMode::Nearest
269        } else {
270            ApplyInterpolationMode::Linear
271        };
272        let mut lookup =
273            Self::open_parameters(path.as_ref(), basic, FieldSelection::AnyField, false)?;
274        lookup.baseline_rows(antennas)?;
275        if options.contextual()
276            || matches!(
277                options.mode,
278                TimeInterpolation::Cubic | TimeInterpolation::Spline
279            )
280        {
281            lookup.advanced_time = Some(TimePlan::new(&mut lookup, options, false)?);
282        }
283        Ok(lookup)
284    }
285
286    /// One simulator antenna's solution with observation/scan context.
287    /// `frequency_hz` is the native data spectral-centroid frequency for scalar-channel
288    /// phase-delay application; frequency-dependent tables pass `None`, as
289    /// CASA's vector-frequency interpolation does not apply phase-delay scaling.
290    /// All calibration fields participate in the selected time segment.
291    ///
292    /// # Errors
293    /// Interpolation fails or a phase-delay reference frequency is invalid.
294    pub fn simulation_solution(
295        &self,
296        observation: i32,
297        scan: i32,
298        spw: i32,
299        antenna: i32,
300        time: f64,
301        frequency_hz: Option<f64>,
302    ) -> Result<Option<ComplexSolution>, ApplyExecutionError> {
303        if let Some(plan) = &self.advanced_time {
304            plan.solution(
305                &self.path,
306                TimeQuery {
307                    observation,
308                    scan,
309                    window: spw,
310                    antenna,
311                    time,
312                    frequency_hz,
313                },
314            )
315        } else {
316            self.solution(0, spw, antenna, time)
317        }
318    }
319}

321impl TimePlan {
322    fn new(
323        lookup: &mut CalibrationTableLookup,
324        options: TimeInterpolationOptions,
325        real: bool,
326    ) -> Result<Self, ApplyExecutionError> {
327        let mut scope = TimeScope::All;
328        let series = if options.per_observation || options.per_scan {
329            let mut rows = headers(&lookup.path)?;
330            lookup.baseline_headers(&mut rows)?;
331            if options.per_scan {
332                if rows
333                    .iter()
334                    .map(|r| r[4])
335                    .min()
336                    .is_some_and(|scan| scan != -1)
337                {
338                    scope = TimeScope::Scan;
339                }
340            } else if options.per_observation {
341                let observations = rows.iter().map(|r| r[3]).collect::<HashSet<_>>();
342                let table = Table::open(TableOptions::new(&lookup.path))
343                    .map_err(|e| invalid(&lookup.path, &e.to_string()))?;
344                let path = table
345                    .keywords()
346                    .get("OBSERVATION")
347                    .and_then(casa_types::Value::as_table_ref)
348                    .map(|name| casa_tables::add_directory(name, &lookup.path))
349                    .unwrap_or_else(|| lookup.path.join("OBSERVATION"));
350                let count = Table::rows_on_disk(&path)
351                    .map_err(|e| invalid(&lookup.path, &e.to_string()))?;
352                if observations.len() > 1
353                    && observations.len() == count
354                    && count == options.observation_count
355                {
356                    scope = TimeScope::Observation;
357                }
358            }
359            let result = segmented(lookup, &rows, scope)?;
360            lookup.across_fields = None;
361            result
362        } else {
363            lookup
364                .across_fields
365                .take()
366                .expect("all fields")
367                .into_iter()
368                .map(|((spw, antenna), rows)| ((-1, spw, antenna), rows))
369                .collect()
370        };
371        let mut plan = Self {
372            mode: options.mode,
373            real,
374            scope,
375            series,
376            splines: HashMap::new(),
377            cycles: HashMap::new(),
378            reference_hz: Vec::new(),
379        };
380        if options.phase_delay && !plan.real {
381            plan.reference_hz = CalibrationTableLookup::frequency_grids_on_disk(&lookup.path)?
382                .iter()
383                .map(|axis| axis.get(axis.len() / 2).copied().unwrap_or(0.))
384                .collect();
385        }
386        for (&key, rows) in &plan.series {
387            if options.mode == TimeInterpolation::Spline && !rows.is_empty() {
388                plan.splines
389                    .insert(key, Spline::new(&lookup.path, rows, plan.real)?);
390            }
391            if options.phase_delay && !plan.real && !rows.is_empty() {
392                plan.cycles.insert(key, phase_cycles(&lookup.path, rows)?);
393            }
394        }
395        Ok(plan)
396    }
397
398    pub(super) fn phase_scale(
399        &self,
400        path: &Path,
401        spw: i32,
402        frequency: f64,
403    ) -> Result<f32, ApplyExecutionError> {
404        if !frequency.is_finite() || frequency <= 0. {
405            return Err(invalid(path, "phase-delay data frequency must be positive"));
406        }
407        let reference = self.reference_hz.get(spw as usize).copied().unwrap_or(0.);
408        Ok(if reference > 0. {
409            (frequency / reference) as f32
410        } else {
411            1.
412        })
413    }
414}

416/// Only scalar columns are read; CPARAM arrays are never rematerialized.
417fn headers(path: &Path) -> Result<Vec<[i32; 5]>, ApplyExecutionError> {
418    let table = Table::open(TableOptions::new(path)).map_err(|e| invalid(path, &e.to_string()))?;
419    let mut rows = Vec::with_capacity(table.row_count());
420    let [
421        mut fields,
422        mut windows,
423        mut antennas,
424        mut observations,
425        mut scans,
426    ] = std::array::from_fn(|_| Vec::new());
427    for first in (0..table.row_count()).step_by(512) {
428        for values in [
429            &mut fields,
430            &mut windows,
431            &mut antennas,
432            &mut observations,
433            &mut scans,
434        ] {
435            values.clear();
436        }
437        let indices = (first..table.row_count().min(first + 512)).collect::<Vec<_>>();
438        table
439            .required_scalar_columns_for_rows_into(
440                &indices,
441                &mut [
442                    Destination::new("FIELD_ID", Values::Int32(&mut fields)),
443                    Destination::new("SPECTRAL_WINDOW_ID", Values::Int32(&mut windows)),
444                    Destination::new("ANTENNA1", Values::Int32(&mut antennas)),
445                    Destination::new("OBSERVATION_ID", Values::Int32(&mut observations)),
446                    Destination::new("SCAN_NUMBER", Values::Int32(&mut scans)),
447                ],
448            )
449            .map_err(|e| invalid(path, &e.to_string()))?;
450        for i in 0..indices.len() {
451            rows.push([
452                fields[i],
453                windows[i],
454                antennas[i],
455                observations[i],
456                scans[i],
457            ]);
458        }
459    }
460    Ok(rows)
461}

463fn segmented(
464    lookup: &CalibrationTableLookup,
465    headers: &[[i32; 5]],
466    scope: TimeScope,
467) -> Result<TimeSeries, ApplyExecutionError> {
468    let mut positions = HashMap::<(i32, i32, i32), usize>::new();
469    let mut grouped = HashMap::<_, Vec<(i32, &CalibrationSolution)>>::new();
470    for &[field, window, antenna, observation, scan] in headers {
471        let position = positions.entry((field, window, antenna)).or_default();
472        let row = lookup
473            .table
474            .solutions
475            .get(&(field, window, antenna))
476            .and_then(|rows| rows.get(*position))
477            .ok_or_else(|| {
478                invalid(
479                    &lookup.path,
480                    "calibration scalar context differs from loaded solutions",
481                )
482            })?;
483        *position += 1;
484        let segment = match scope {
485            TimeScope::All => -1,
486            TimeScope::Observation => observation,
487            TimeScope::Scan => scan,
488        };
489        grouped
490            .entry((segment, window, antenna))
491            .or_default()
492            .push((field, row));
493    }
494    Ok(grouped
495        .into_iter()
496        .map(|(key, mut rows)| {
497            rows.sort_by(|a, b| {
498                a.1.time_seconds
499                    .total_cmp(&b.1.time_seconds)
500                    .then(a.0.cmp(&b.0))
501            });
502            (
503                key,
504                rows.into_iter()
505                    .map(|(_, row)| CalibrationSolution {
506                        time_seconds: row.time_seconds,
507                        grid: Arc::clone(&row.grid),
508                    })
509                    .collect(),
510            )
511        })
512        .collect())
513}

515fn phase_cycles(
516    path: &Path,
517    rows: &[CalibrationSolution],
518) -> Result<Vec<f32>, ApplyExecutionError> {
519    let first = grid(path, &rows[0])?;
520    let count = first.receptor_count * first.channel_count;
521    let mut cycles = vec![0.; rows.len() * count];
522    for index in 0..count {
523        let (z, flag) = value(first, index);
524        let mut previous = if flag { 0. } else { z.arg() };
525        let mut shift = 0.;
526        for (time, row) in rows.iter().enumerate().skip(1) {
527            let current = grid(path, row)?;
528            if current.values.shape() != first.values.shape() {
529                return Err(invalid(path, "phase-delay parameter shapes must match"));
530            }
531            let (z, flag) = value(current, index);
532            if !flag {
533                let difference = z.arg() - previous;
534                if difference > std::f32::consts::PI {
535                    shift -= std::f32::consts::TAU;
536                }
537                if difference < -std::f32::consts::PI {
538                    shift += std::f32::consts::TAU;
539                }
540                previous = z.arg();
541            }
542            cycles[time * count + index] = shift;
543        }
544    }
545    Ok(cycles)
546}

548fn real_parameters(table: &Table) -> bool {
549    use casa_types::{ScalarValue, Value};
550    match table.keywords().get("ParType") {
551        Some(Value::Scalar(ScalarValue::String(kind))) if kind == "Float" => true,
552        Some(Value::Scalar(ScalarValue::String(kind))) if kind == "Complex" => false,
553        _ => table.schema().is_some_and(|schema| {
554            let contains = |name: &str| schema.columns().iter().any(|column| column.name() == name);
555            contains("FPARAM") && !contains("CPARAM")
556        }),
557    }
