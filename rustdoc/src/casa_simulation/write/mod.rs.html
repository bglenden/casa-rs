2//! Writing a session's MeasurementSet at `close`.
3//!
4//! One pass creates the MeasurementSet in a staging directory, fills the
5//! subtables, streams MAIN through the bounded write session a batch of
6//! column-shaped rows at a time, and publishes the result. The layout is
7//! casa-rs's conventional one: IncrementalStMan and StandardStMan scalars,
8//! TiledShapeStMan visibility columns with CASA's tile shapes, and
9//! TiledColumnStMan for UVW. Following plan decision 6, MAIN has DATA but
10//! no MODEL_DATA or CORRECTED_DATA. DATA is the sum of the predictions of
11//! every `predict` step covering the row's scan (zero without one) plus the
12//! noise of every `corrupt` step covering it.
13//!
14//! Before anything is written, what the pass holds is admitted against the
15//! session's resource policy ([`resources`]); each batch's geometry,
16//! prediction ([`prediction`]) and noise are then computed on the session's
17//! worker team ([`pipeline`]).
18//!
19//! A session `openfromms` opened is patched in place instead ([`patch`]).

65/// Create the session's MeasurementSet.
66pub(crate) fn write_measurement_set(session: &Simulator) -> SimulatorResult<()> {
67    let window = ReservationWindow::open();
68    let planned = PlannedScan::of_all(session.scans()).ok_or_else(|| {
69        SimulatorError::raised("only a session's planned scans make a new MeasurementSet")
70    })?;
71    let cell = cell_shape(session)?;
72    let shape = cell.map(|cell| main_shape(session, &planned, cell));
73    let resources = &session.options().resources;
74    let prediction = match cell {
75        Some(cell) => PredictionStage::new(session, cell, resources.prediction_backend())?,
76        None => None,
77    };
78    // Admitted before anything is written, so a refusal leaves nothing.
79    let admission = shape
80        .as_ref()
81        .map(|shape| resources::admit_main_write(resources, shape, prediction.as_ref()))
82        .transpose()?;
83    let target = MeasurementSetCreateTarget::prepare(session.path(), true)
84        .map_err(write_error("prepare the MeasurementSet"))?;
85    let staging = target.staging_path().to_path_buf();
86    let mut ms = MeasurementSet::create(
87        &staging,
88        MeasurementSetBuilder::new()
89            .with_main_column(OptionalMainColumn::Data)
90            .with_optional_subtable(SubtableId::Source),
91    )
92    .map_err(|error| SimulatorError::raised(format!("create the MeasurementSet: {error}")))?;
93    std::fs::create_dir_all(&staging).map_err(|error| {
94        SimulatorError::raised(format!("create {}: {error}", staging.display()))
95    })?;
96    let info = ms.main_table().info().clone();
97    ms.main_table_mut().set_info(TableInfo {
98        table_type: info.table_type,
99        sub_type: "simulator".to_string(),
100        readme: MAIN_README.iter().map(|line| (*line).to_string()).collect(),
101    });
102    subtables::write_subtables(&mut ms, session)?;
103    let generated = subtables::subtable_overrides(&planned)?;
104    match shape.zip(admission) {
105        Some((shape, admission)) => {
106            let main = MainTarget {
107                staging: &staging,
108                generated: &generated,
109                window: &window,
110            };
111            write_main(
112                &mut ms,
113                (session, &planned),
114                &main,
115                &shape,
116                (admission, prediction),
117            )?;
118        }
119        None => ms
120            .save()
121            .map_err(|error| SimulatorError::raised(format!("save the MeasurementSet: {error}")))?,
122    }
123    target
124        .commit()
125        .map_err(write_error("publish the MeasurementSet"))
126}

128/// The cell shape every scan shares: (correlations, channels).
129fn cell_shape(session: &Simulator) -> SimulatorResult<Option<(usize, usize)>> {
130    let mut shape = None;
131    for scan in session.scans() {
132        let window = &session.spectral_windows()[scan.spectral_window_id];
133        let this = (window.correlations.len(), window.channel_count);
134        match shape {
135            None => shape = Some(this),
136            Some(first) if first != this => {
137                return Err(SimulatorError::raised(
138                    "observing spectral windows of different shapes in one MeasurementSet is not \
139                     yet implemented in casa-rs",
140                ));
141            }
142            Some(_) => {}
143        }
144    }
145    Ok(shape)
146}

148/// `[window]`: the receptors of each correlation of the session's spectral
149/// windows (`Stokes::receptor1`, `receptor2`), which gains and leakage take.
150pub(super) fn window_receptors(session: &Simulator) -> Vec<Vec<[usize; 2]>> {
151    session
152        .spectral_windows()
153        .iter()
154        .map(|window| {
155            window
156                .correlation_products()
157                .iter()
158                .map(|pair| pair.map(|receptor| usize::try_from(receptor).unwrap_or(0)))
159                .collect()
160        })
161        .collect()
162}

164/// MAIN as the session's planned scans make it, with cells of
165/// `(correlations, channels)`.
166fn main_shape<'s>(
167    session: &'s Simulator,
168    planned: &[PlannedScan<'_>],
169    (correlations, channels): (usize, usize),
170) -> MainShape<'s> {
171    MainShape {
172        rows: planned.iter().map(|scan| scan.rows.row_count()).sum(),
173        correlations,
174        channels,
175        telescope: session.telescope(),
176        largest_integration: planned
177            .iter()
178            .map(|scan| scan.rows.baseline_count())
179            .max()
180            .unwrap_or(0),
181        antennas: planned
182            .iter()
183            .map(|scan| scan.rows.antenna_count)
184            .max()
185            .unwrap_or(0),
186        generated_bytes: subtables::generated_bytes(planned),
187        noise_scales: NoiseScales::of(
188            session.data_plan(),
189            channels,
190            Some((
191                planned
192                    .iter()
193                    .map(|scan| scan.rows.baseline_count())
194                    .filter(|&rows| rows > 0)
195                    .min()
196                    .unwrap_or(1),
197                planned
198                    .iter()
199                    .map(|scan| scan.rows.antenna_count)
200                    .max()
201                    .unwrap_or(0),
202            )),
203        ),
204    }
205}

207/// Where MAIN goes: the staging directory, the subtable columns the save
208/// generates (POINTING), and the window whose peak the pass reports.
209struct MainTarget<'a> {

211    generated: &'a casa_ms::SubtableColumnOverrides,
212    window: &'a ReservationWindow,
213}

215/// Write MAIN and save the MeasurementSet, holding `admission` until done,
216/// predicting with `prediction` when the session predicts.
217fn write_main<'s>(
218    ms: &mut MeasurementSet,
219    (session, planned): (&'s Simulator, &[PlannedScan<'_>]),
220    target: &MainTarget<'_>,
221    shape: &MainShape<'_>,
222    (admission, mut prediction): (MainAdmission, Option<PredictionStage<'s>>),
223) -> SimulatorResult<()> {
224    let started = Instant::now();
225    let batch_rows = admission.plan.batch_rows().max(1);
226    if let Some(stage) = &mut prediction {
227        stage.start(admission.block_rows, admission.cores.streaming_workers);
228    }
229    let backend = prediction.as_ref().map_or_else(
230        || session.options().resources.resolved_backend(false),
231        PredictionStage::backend,
232    );
233    let mut writer = MainWriter::start(target.staging, admission.plan.clone())?;
234    let geometry = ArrayGeometry::new(
235        session.telescope(),
236        session.antennas(),
237        session.options().measures.clone(),
238    )?
239    .with_airmass(noise_varies_by_row(session.data_plan()));
240    let team = WorkerTeam::new(admission.cores.streaming_workers)
241        .map_err(|error| SimulatorError::raised(format!("the worker team: {error}")))?;
242    let mut pipeline = Pipeline::new(
243        (session.scans(), planned),
244        &geometry,
245        &team,
246        (
247            prediction,
248            session.data_plan(),
249            NoiseContext::of(session),
250            shape.noise_scales,
251        ),
252        (
253            shape.correlations,
254            shape.channels,
255            window_receptors(session),
256        ),
257    );
258    let integrations = planned
259        .iter()
260        .enumerate()
261        .filter(|(_, scan)| scan.rows.baseline_count() > 0)
262        .flat_map(|(index, scan)| {
263            scan.rows
264                .integration_times_s()
265                .map(move |time| (index, time))
266        });
267    let setup = started.elapsed();
268    pipeline.write_rows(&mut writer, integrations, shape.rows, batch_rows)?;
269    let finishing = Instant::now();
270    let written = writer.finish(ms, shape.rows, target.generated)?;
271    let finish = finishing.elapsed();
272    let times = &pipeline.times;
273    tracing::info!(
274        rows = shape.rows,
275        batch_rows,
276        block_rows = admission.block_rows,
277        batches = times.batches,
278        workers = admission.cores.streaming_workers,
279        policy_workers = admission.policy_cores.streaming_workers,
280        backend = ?backend,
281        reserved_bytes = admission.reservation.memory(),
282        peak_reserved_bytes = target.window.close(),
283        geometry_s = times.geometry.as_secs_f64(),
284        zero_s = times.zero.as_secs_f64(),
285        predict_s = times.predict.as_secs_f64(),
286        block_fill_s = pipeline.block_fill_time().as_secs_f64(),
287        cells_s = times.cells.as_secs_f64(),
288        send_wait_s = times.send.as_secs_f64(),
289        setup_s = setup.as_secs_f64(),
290        finish_s = finish.as_secs_f64(),
291        writer_write_s = written.write_seconds,
292        writer_finalize_s = written.finalize_seconds,
293        bytes_written = written.bytes_written,
294        total_s = started.elapsed().as_secs_f64(),
295        "simulation pass"
296    );
297    Ok(())
298}

300/// Whether any noise of `plan` has a σ that varies by row (the Tsys
301/// modes), whose rows then need their antennas' airmasses and exposure.
302/// Every `corrupt` covers scan 0, so its noise is all the plan's.
303fn noise_varies_by_row(plan: &DataPlan) -> bool {
304    plan.noise_for_scan(0)
305        .iter()
306        .any(crate::corrupt::NoiseApplication::varies_by_row)
307}

309/// The resolved MAIN scalar columns.
310struct ScalarColumns {
311    antenna1: ScalarColumnId,
312    antenna2: ScalarColumnId,
313    flag_row: ScalarColumnId,
314    time: ScalarColumnId,
315    time_centroid: ScalarColumnId,
316    constants: [(ScalarColumnId, &'static str); 11],
317}

319/// MAIN's write session and the scalar runs of the integration being
320/// pushed.
321struct MainWriter {
322    session: MeasurementSetWriteSession,
323    columns: ScalarColumns,
324    antenna1: Vec<i32>,
325    antenna2: Vec<i32>,
326    flag_row: Vec<bool>,
327}
328
329impl MainWriter {
330    fn start(staging: &std::path::Path, plan: MeasurementSetWritePlan) -> SimulatorResult<Self> {
331        let session = MeasurementSetWriteSession::start(staging, plan)
332            .map_err(write_error("start the MAIN write"))?;
333        let id = |name: &str| {
334            session
335                .scalar_column(name)
336                .map_err(write_error("MAIN scalar column"))
337        };
338        let constant = |name: &'static str| id(name).map(|column| (column, name));
339        let columns = ScalarColumns {
340            antenna1: id("ANTENNA1")?,
341            antenna2: id("ANTENNA2")?,
342            flag_row: id("FLAG_ROW")?,
343            time: id("TIME")?,
344            time_centroid: id("TIME_CENTROID")?,
345            constants: [
346                constant("ARRAY_ID")?,
347                constant("DATA_DESC_ID")?,
348                constant("EXPOSURE")?,
349                constant("FEED1")?,
350                constant("FEED2")?,
351                constant("FIELD_ID")?,
352                constant("INTERVAL")?,
353                constant("OBSERVATION_ID")?,
354                constant("PROCESSOR_ID")?,
355                constant("SCAN_NUMBER")?,
356                constant("STATE_ID")?,
357            ],
358        };
359        Ok(Self {
360            session,
361            columns,
362            antenna1: Vec::new(),
363            antenna2: Vec::new(),
364            flag_row: Vec::new(),
365        })
366    }
367
368    /// Append the scalar columns of one integration of `scan` at stored
369    /// TIME `time_s`, whose rows are `rows`.
370    fn push_scalars(
371        &mut self,
372        PlannedScan {
373            scan,
374            rows: planned,
375        }: PlannedScan<'_>,
376        time_s: f64,
377        rows: &[RowGeometry],
378    ) -> SimulatorResult<()> {
379        self.antenna1.clear();
380        self.antenna2.clear();
381        self.flag_row.clear();
382        for row in rows {
383            self.antenna1.push(row.antenna1 as i32);
384            self.antenna2.push(row.antenna2 as i32);
385            self.flag_row.push(row.flagged);
386        }
387        let count = rows.len();
388        let error = write_error("MAIN scalars");
389        let columns = &self.columns;
390        let session = &mut self.session;
391        session
392            .push_scalar_column(
393                columns.antenna1,
394                StreamedScalarValues::Int32(&self.antenna1),
395            )
396            .map_err(&error)?;
397        session
398            .push_scalar_column(
399                columns.antenna2,
400                StreamedScalarValues::Int32(&self.antenna2),
401            )
402            .map_err(&error)?;
403        session
404            .push_scalar_column(columns.flag_row, StreamedScalarValues::Bool(&self.flag_row))
405            .map_err(&error)?;
406        for column in [columns.time, columns.time_centroid] {
407            session
408                .push_repeated_scalar(column, Some(ScalarValue::Float64(time_s)), count)
409                .map_err(&error)?;
410        }
411        for (column, name) in columns.constants {
412            let value = match name {
413                "DATA_DESC_ID" => ScalarValue::Int32(scan.spectral_window_id as i32),
414                "EXPOSURE" | "INTERVAL" => ScalarValue::Float64(planned.integration_s),
415                "FIELD_ID" => ScalarValue::Int32(scan.field_id as i32),
416                "OBSERVATION_ID" => ScalarValue::Int32(planned.observation_id),
417                "SCAN_NUMBER" => ScalarValue::Int32(scan.scan_number),
418                "STATE_ID" => ScalarValue::Int32(planned.state_id as i32),
419                // ARRAY_ID, FEED1, FEED2 and PROCESSOR_ID.
420                _ => ScalarValue::Int32(0),
421            };
422            session
423                .push_repeated_scalar(column, Some(value), count)
424                .map_err(&error)?;
425        }
426        Ok(())
427    }
428
429    /// An empty batch, with the buffers of one the writer has finished with
430    /// when there is one.
431    fn visibility_columns(&self) -> VisibilityColumns {
432        self.session.visibility_columns()
433    }
434
435    /// Queue a full batch for the writer's thread; blocks while its queue is
436    /// full.
437    fn send(&self, batch: VisibilityColumns) -> SimulatorResult<()> {
438        self.session
439            .send_visibility_columns(batch)
440            .map_err(write_error("MAIN visibility batch"))
441    }
442
443    fn finish(
444        self,
445        ms: &mut MeasurementSet,
446        row_count: usize,
447        generated: &casa_ms::SubtableColumnOverrides,
448    ) -> SimulatorResult<MeasurementSetWriteTelemetry> {
449        let mut overrides = ColumnOverrides::for_row_count(row_count);
450        for column in ["DATA", "FLAG", "FLAG_CATEGORY", "UVW", "WEIGHT", "SIGMA"] {
451            overrides.insert_deferred(column);
452        }
453        self.session
454            .save_and_finish_with_subtables(ms, &overrides, generated)
455            .map_err(write_error("finish the MAIN write"))
456    }
