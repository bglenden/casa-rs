2//! CASA's recorded single-type integer parameter refusals.
3
4use std::collections::BTreeMap;
5use std::sync::OnceLock;
6
7use casa_provider_contracts::ParameterValue;
8
9/// Message for a value CASA's `cInt` validator refuses, where the pinned
10/// task schema has that single type. Python booleans count as integers in
11/// CASA; casa-rs currently refuses them, so they retain its own diagnostic.
12pub(crate) fn integer_refusal(surface: &str, name: &str, value: &ParameterValue) -> Option<String> {
13    if matches!(value, ParameterValue::Integer(_) | ParameterValue::Bool(_)) {
14        return None;
15    }
16    integer_parameter(surface, name).then(|| format!("{{'{name}': ['must be of cInt type']}}"))
17}
18
19/// Whether the pinned CASA schema declares a parameter as a single `cInt`.
20pub(crate) fn integer_parameter(surface: &str, name: &str) -> bool {
21    static PARAMETERS: OnceLock<BTreeMap<String, Vec<String>>> = OnceLock::new();
22    let parameters = PARAMETERS.get_or_init(|| {
23        serde_json::from_str(include_str!("../resources/casa-integer-parameters.json"))
24            .expect("the checked CASA integer-parameter inventory")
25    });
26    parameters
27        .get(surface)
28        .is_some_and(|names| names.iter().any(|item| item == name))
29}
30
31/// A JSON integer literal that cannot fit the native signed representation.
32/// Decimal and exponent forms are floats in CASA even when mathematically whole.
33pub(crate) fn integer_literal_outside_i64(raw: &str) -> bool {
34    let digits = raw.strip_prefix('-').unwrap_or(raw);
35    !digits.is_empty()
36        && digits.bytes().all(|byte| byte.is_ascii_digit())
37        && raw.parse::<i64>().is_err()
38}

40#[cfg(test)]
41mod tests {
42    use super::integer_literal_outside_i64;
43
44    #[test]
45    fn json_integer_bounds_are_distinct_from_float_spellings() {
46        for valid in [
47            "9223372036854775807",
48            "-9223372036854775808",
49            "9.223372036854776e18",
50        ] {
51            assert!(!integer_literal_outside_i64(valid), "{valid}");
52        }
53        for overflow in ["9223372036854775808", "-9223372036854775809"] {
54            assert!(integer_literal_outside_i64(overflow), "{overflow}");
55        }
56    }
