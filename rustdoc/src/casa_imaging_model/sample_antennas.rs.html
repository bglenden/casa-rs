3//! Per-sample antenna quantities of the batch geometry: each antenna's
4//! pointing direction and aperture response class.
5
6use crate::geometry::SkyDirection;
7
8/// Per-antenna pointing directions evaluated for one baseline sample.
9///
10/// Retaining both directions preserves antenna-dependent boresight offsets. A downstream
11/// operator may derive a baseline-effective direction according to its own closed response law.
12#[derive(Debug, Clone, Copy, PartialEq)]
13pub struct SelectedPointingDirections {
14    /// Evaluated direction for MAIN `ANTENNA1`.
15    pub antenna1: SkyDirection,
16    /// Evaluated direction for MAIN `ANTENNA2`.
17    pub antenna2: SkyDirection,
18}
19
20/// CASA aperture class selected from owner-controlled ANTENNA metadata.
21///
22/// These are scientific response identities rather than physical antenna IDs.
23/// Multiple antennas with the same class therefore share one cached response.
24#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
25pub enum AntennaResponseClass {
26    /// Physical 12 m ALMA dish evaluated with CASA's 10.7 m effective aperture.
27    CasaAlma12m,
28    /// Physical 7 m ACA dish evaluated with CASA's 6.25 m effective aperture.
29    CasaAca7m,
30}
31
32impl AntennaResponseClass {
33    /// Whether CASA treats `telescope` as ALMA or ACA, whose dishes it
34    /// classes by [`Self::casa_alma_aca`].
35    #[must_use]
36    pub fn casa_alma_aca_telescope(telescope: &str) -> bool {
37        matches!(telescope.trim(), "ALMA" | "ACA")
38    }
39
40    /// CASA's class of an ALMA or ACA dish `diameter_m` across: 12 m
41    /// within 0.5 m, 7 m within 1 m; `None` for another dish.
42    #[must_use]
43    pub fn casa_alma_aca(diameter_m: f64) -> Option<Self> {
44        if (diameter_m - 12.0).abs() < 0.5 {
45            Some(Self::CasaAlma12m)
46        } else if (diameter_m - 7.0).abs() < 1.0 {
47            Some(Self::CasaAca7m)
48        } else {
49            None
50        }
51    }
52
53    /// The largest response of an antenna family whose dishes are
54    /// `classes`: the family envelope CASA's heterogeneous mosaic sizes its
55    /// kernels by.
56    #[must_use]
57    pub fn family_envelope(classes: &[Option<Self>]) -> Option<Self> {
58        [Self::CasaAlma12m, Self::CasaAca7m]
59            .into_iter()
60            .find(|class| classes.contains(&Some(*class)))
61    }
62}
63
64/// Ordered response classes for one selected interferometric baseline.
65#[derive(Debug, Clone, Copy, PartialEq, Eq)]
66pub struct SelectedAntennaResponses {
67    /// Response class of MAIN `ANTENNA1`.
68    pub antenna1: AntennaResponseClass,
69    /// Response class of MAIN `ANTENNA2`.
70    pub antenna2: AntennaResponseClass,
71    /// Largest response class present in this MeasurementSet response family.
72    ///
73    /// CASA crops every heterogeneous convolution plane to one family-wide
74    /// extent so resampling has the same boundary domain for every pair.
75    pub family_envelope: AntennaResponseClass,
76}
