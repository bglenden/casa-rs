21/// Controls for atmospheric absorption and emission curves over a bandpass panel.
22pub struct AtmosphericOptions<'a> {
23    /// Whether to compute atmospheric transmission.
24    pub transmission: bool,
25    /// Whether to compute atmospheric sky brightness temperature.
26    pub sky_temperature: bool,
27    /// Precipitable water-vapor setting in millimeters, or the automatic CASA token.
28    pub water: &'a str,
29    /// Field selector for determining the observing direction and airmass.
30    pub reference_field: &'a str,
31    /// Whether to include the reflected image sideband.
32    pub image_sideband: bool,
33    /// Optional local oscillator frequency in GHz.
34    pub lo_ghz: Option<f64>,
35    /// Parent MeasurementSet to use for observing geometry and weather data.
36    pub parent_ms: Option<&'a Path>,
37    /// Whether to draw overlay samples as individual points.
38    pub show_points: bool,
39    /// Whether the plotted curves combine more than one solution time.
40    pub time_overlay: bool,
41}

73impl<'a> AtmosphereContext<'a> {
74    pub(super) fn open(
75        source: &'a Solutions,
76        options: &'a AtmosphericOptions<'a>,
77    ) -> Result<Option<Self>, String> {
78        if !(options.transmission || options.sky_temperature) {
79            return Ok(None);
80        }
81        let Some(measurement_set) = source.measurement_set.as_deref() else {
82            return Ok(None);
83        };
84        let observations = open(measurement_set, "OBSERVATION")?;
85        let site_name = text_cell(&observations, 0, "TELESCOPE_NAME")
86            .filter(|name| !name.is_empty())
87            .unwrap_or("ALMA")
88            .to_string();
89        Ok(Some(Self {
90            options,
91            site_name,
92            measures: Arc::new(
93                MeasuresRuntime::open_discovered(RuntimePolicy::default())
94                    .map_err(|error| error.to_string())?,
95            ),
96            fields: open(measurement_set, "FIELD")?,
97            antennas: open(measurement_set, "ANTENNA")?,
98            weather: open_optional(measurement_set, "WEATHER"),
99            pointing: open_optional(measurement_set, "POINTING"),
100            scan_times: scan_times(measurement_set)?,
101            water_samples: if options.water.contains("auto") {
102                water_samples(measurement_set)
103            } else {
104                Vec::new()
105            },
106            oscillators: if options.image_sideband && options.lo_ghz.is_none() {
107                receiver_oscillators(measurement_set, options.parent_ms)
108            } else {
109                BTreeMap::new()
110            },
111        }))
112    }

114    pub(super) fn curves(
115        &self,
116        window: &SolutionWindow,
117        row: SolutionRow,
118        frequency_axis: bool,
119    ) -> Result<Vec<AtmosphericCurve>, String> {
120        let frequency = &window.sky_frequencies_hz;
121        if frequency.len() < 2 {
122            return Ok(Vec::new());
123        }
124        let position =
125            antenna_position(&self.antennas, row.antenna).ok_or("missing antenna position")?;
126        let altitude_m = position
127            .convert_to(PositionRef::WGS84)
128            .map_err(|error| error.to_string())?
129            .values()[2];
130        let field = self.reference_field(row.field);
131        let scan = self.scan_times.get(&field).and_then(|scans| {
132            if self.options.reference_field.is_empty() {
133                scans.iter().min_by(|a, b| {
134                    (a.mean - row.time)
135                        .abs()
136                        .total_cmp(&(b.mean - row.time).abs())
137                })
138            } else {
139                scans.iter().min_by(|a, b| a.mean.total_cmp(&b.mean))
140            }
141        });
142        let airmass = self.airmass(row, field, scan, &position)?;
143        let timestamp = if self.options.reference_field.is_empty() {
144            row.time
145        } else {
146            scan.map_or(row.time, |scan| scan.mean)
147        };
148        let water_mm = self.water(timestamp, row.interval);
149        let weather = self
150            .weather
151            .as_ref()
152            .map(|table| conditions(table, scan.map_or([row.time; 2], |scan| scan.bounds)))
153            .unwrap_or_default();
154        let parameters = ProfileParameters {
155            altitude_m,
156            ground_pressure_hpa: weather.pressure_hpa,
157            ground_temperature_k: weather.temperature_k,
158            relative_humidity_percent: weather.humidity_percent,
159            water_vapour_scale_height_km: 1.0,
160            pressure_step_hpa: 5.0,
161            top_altitude_km: 60.0,
162            atmosphere: AtmosphereType::MidlatitudeWinter,
163            ..ProfileParameters::default()
164        };
165        let conditions = LineOfSight {
166            profile: parameters,
167            water_mm,
168            airmass,
169            field,
170        };
171        let mut curves =
172            vec![self.evaluate(window, frequency, &conditions, frequency_axis, None)?];
173        if self.options.image_sideband
174            && let Some(lo) = self.local_oscillator(window.original_id)
175        {
176            let lo_hz = lo * 1e9;
177            let image = frequency
178                .iter()
179                .map(|frequency| 2.0 * lo_hz - frequency)
180                .collect::<Vec<_>>();
181            curves.push(self.evaluate(window, &image, &conditions, frequency_axis, Some(lo_hz))?);
182        }
183        Ok(curves)
184    }

186    fn evaluate(
187        &self,
188        window: &SolutionWindow,
189        frequency: &[f64],
190        conditions: &LineOfSight,
191        frequency_axis: bool,
192        image_lo_hz: Option<f64>,
193    ) -> Result<AtmosphericCurve, String> {
194        let mut channels = frequency.len();
195        while channels > 512 {
196            channels /= 2;
197        }
198        let center = (frequency[frequency.len() / 2 - 1] + frequency[frequency.len() / 2]) * 0.5;
199        let separation = (frequency[frequency.len() - 1] - frequency[0]) / (channels - 1) as f64;
200        let mut grid = SpectralGrid::new();
201        let reference_channel = channels / 2;
202        let offset = if channels.is_multiple_of(2) {
203            0.5 * separation.abs()
204        } else {
205            0.0
206        };
207        grid.add_regular(
208            channels,
209            reference_channel,
210            center + offset,
211            separation.abs(),
212        )
213        .map_err(|error| error.to_string())?;
214        let model_frequencies = grid.frequencies_hz().collect::<Vec<_>>();
215        let profile =
216            AtmosphereProfile::new(&conditions.profile).map_err(|error| error.to_string())?;
217        let absorption = AbsorptionProfile::new(profile, grid);
218        let LineOfSight {
219            water_mm,
220            airmass,
221            field,
222            ..
223        } = *conditions;
224        let water_m = water_mm * 1e-3;
225        let zenith = Sky::zenith(water_m);
226        let mut y = (0..channels)
227            .map(|channel| {
228                let tau =
229                    absorption.dry_opacity(channel) + absorption.wet_opacity(channel, water_m);
230                let transmission = (-airmass * tau).exp();
231                if self.options.sky_temperature {
232                    absorption
233                        .tebb_sky(channel, &zenith)
234                        .map(|tebb| tebb * (-airmass * tau).exp_m1() / (-tau).exp_m1())
235                        .map_err(|error| error.to_string())
236                } else {
237                    Ok(transmission)
238                }
239            })
240            .collect::<Result<Vec<_>, _>>()?;
241        let average = frequency.iter().sum::<f64>() / frequency.len() as f64;
242        let high_reference = window.reference_frequency_hz.is_some_and(|hz| hz > average);
243        let reverse = high_reference == (window.sideband % 2 == 0);
244        let image = image_lo_hz.is_some();
245        let mut x = if frequency_axis {
246            model_frequencies
247                .into_iter()
248                .map(|hz| image_lo_hz.map_or(hz, |lo| 2.0 * lo - hz) * 1e-9)
249                .collect()
250        } else {
251            if reverse {
252                y.reverse();
253            }
254            let step = (frequency.len() - 1) as f64 / (channels - 1) as f64;
255            (0..channels)
256                .map(|channel| channel as f64 * step)
257                .collect::<Vec<_>>()
258        };
259        if image {
260            x.reverse();
261            if frequency_axis {
262                y.reverse();
263            }
264        }
265        Ok(AtmosphericCurve {
266            x,
267            y,
268            image,
269            sky_temperature: self.options.sky_temperature,
270            show_points: self.options.show_points,
271            label: if self.options.time_overlay {
272                format!("PWV {water_mm:.2}mm, airmass {airmass:.2}, maxAlt 60km (field {field})")
273            } else {
274                format!("PWV {water_mm:.2}mm, airmass {airmass:.3}, maxAlt 60km")
275            },
276        })
277    }

279    fn reference_field(&self, default: i32) -> i32 {
280        if self.options.reference_field.is_empty() {
281            return default;
282        }
283        self.options
284            .reference_field
285            .parse::<i32>()
286            .ok()
287            .or_else(|| {
288                (0..self.fields.row_count())
289                    .find(|&id| {
290                        text_cell(&self.fields, id, "NAME") == Some(self.options.reference_field)
291                    })
292                    .and_then(|id| i32::try_from(id).ok())
293            })
294            .unwrap_or(default)
295    }

297    fn airmass(
298        &self,
299        row: SolutionRow,
300        field_id: i32,
301        scan: Option<&ScanTimes>,
302        antenna: &MPosition,
303    ) -> Result<f64, String> {
304        if !self.site_name.contains("VLA")
305            && !self.site_name.contains("NRO")
306            && let (Some(pointing), Some(scan)) = (&self.pointing, scan)
307            && let Some(elevation) = pointing_elevation(pointing, row.antenna, scan.bounds)
308        {
309            return Ok(airmass_for_elevation(elevation));
310        }
311        let field = usize::try_from(field_id).map_err(|error| error.to_string())?;
312        let reference = array(&self.fields, field, "DELAY_DIR")?;
313        let ArrayValue::Float64(reference) = reference else {
314            return Err("invalid FIELD reference direction".into());
315        };
316        let mut angles = reference.iter().copied();
317        let ra = angles.next().ok_or("missing field right ascension")?;
318        let dec = angles.next().ok_or("missing field declination")?;
319        let position = MPosition::casa_observatory(&self.site_name, self.measures.as_ref())
320            .unwrap_or_else(|| antenna.clone());
321        let frame = MeasFrame::new()
322            .with_measures(self.measures.clone())
323            .with_epoch(MEpoch::from_mjd_seconds(
324                scan.map_or(row.time, |scan| scan.mean),
325                EpochRef::UTC,
326            ))
327            .with_position(position);
328        let direction = MDirection::from_angles(ra, dec, DirectionRef::J2000)
329            .convert_to(DirectionRef::AZEL, &frame)
330            .map_err(|error| error.to_string())?;
331        Ok(airmass_for_elevation(direction.latitude_rad()))
332    }

334    fn water(&self, time: f64, interval: f64) -> f64 {
335        let default = if self.site_name.contains("ALMA") {
336            1.0
337        } else {
338            5.0
339        };
340        if self.options.water.is_empty() {
341            return 1.0;
342        }
343        if !self.options.water.contains("auto") {
344            return self.options.water.parse().unwrap_or(default);
345        }
346        let start = time - interval * 0.5;
347        let end = time + interval * 0.5;
348        let mut water = self
349            .water_samples
350            .iter()
351            .filter_map(|&(epoch, value)| (epoch > start && epoch < end).then_some(value))
352            .collect::<Vec<_>>();
353        if water.is_empty() {
354            let closest = self
355                .water_samples
356                .iter()
357                .map(|(epoch, _)| (epoch - start).abs())
358                .min_by(f64::total_cmp)
359                .unwrap_or(f64::INFINITY);
360            water.extend(self.water_samples.iter().filter_map(|&(epoch, value)| {
361                (((epoch - start).abs() - closest).abs() < 1.0).then_some(value)
362            }));
363        }
364        let measured = 1000.0 * median(&water);
365        if measured >= 1e-5 { measured } else { default }
366    }

368    fn local_oscillator(&self, window: i32) -> Option<f64> {
369        self.options
370            .lo_ghz
371            .or_else(|| self.oscillators.get(&window).copied())
372    }

375fn water_samples(ms: &Path) -> Vec<(f64, f64)> {
376    for name in ["ASDM_CALWVR", "ASDM_CALATMOSPHERE"] {
377        let Some(table) = open_optional(ms, name).filter(|table| table.row_count() > 0) else {
378            continue;
379        };
380        return (0..table.row_count())
381            .filter_map(|row| {
382                let epoch = scalar_number(&table, row, "startValidTime")?;
383                let water = match table.cell_accessor(row, "water").ok()?.value().ok()?? {
384                    Value::Scalar(ScalarValue::Float64(value)) => *value,
385                    Value::Array(ArrayValue::Float64(values)) => values.iter().copied().next()?,
386                    _ => return None,
387                };
388                Some((epoch, water))
389            })
390            .collect();
391    }
392    Vec::new()
393}

395fn receiver_oscillators(ms: &Path, parent: Option<&Path>) -> BTreeMap<i32, f64> {
396    let Some(receiver) = open_optional(ms, "ASDM_RECEIVER") else {
397        return BTreeMap::new();
398    };
399    let mut by_window = BTreeMap::new();
400    for row in 0..receiver.row_count() {
401        if let Some(id) = text_cell(&receiver, row, "spectralWindowId")
402            .and_then(|id| id.rsplit_once('_'))
403            .and_then(|(_, id)| id.parse::<i32>().ok())
404            && let Ok(ArrayValue::Float64(frequencies)) = array(&receiver, row, "freqLO")
405            && let Some(hz) = frequencies.iter().next()
406        {
407            by_window.entry(id).or_insert(hz * 1e-9);
408        }
409    }
410    let Some(windows) = open_optional(ms, "SPECTRAL_WINDOW") else {
411        return by_window;
412    };
413    if by_window
414        .last_key_value()
415        .is_none_or(|(&last, _)| last as usize + 1 == windows.row_count())
416    {
417        return by_window;
418    }
419    let Some(original) = parent.and_then(|path| open_optional(path, "SPECTRAL_WINDOW")) else {
420        return BTreeMap::new();
421    };
422    let names = (0..original.row_count())
423        .filter_map(|row| Some((text_cell(&original, row, "NAME")?, row as i32)))
424        .collect::<BTreeMap<_, _>>();
425    (0..windows.row_count())
426        .filter_map(|row| {
427            let name = text_cell(&windows, row, "NAME")?;
428            let unrenumbered = names.get(name)?;
429            Some((row as i32, *by_window.get(unrenumbered)?))
430        })
431        .collect()
432}

440impl Default for Conditions {
441    fn default() -> Self {

444            temperature_k: 273.15,
445            humidity_percent: 20.0,
446        }
447    }

450fn conditions(weather: &Table, [start, stop]: [f64; 2]) -> Conditions {
451    let default = Conditions::default();
452    let mut rows = (0..weather.row_count())
453        .filter(|&row| {
454            scalar_number(weather, row, "TIME").is_some_and(|time| start <= time && time <= stop)
455        })
456        .collect::<Vec<_>>();
457    if rows.is_empty() {
458        let Some(row) = nearest_time(weather, "TIME", start) else {
459            return default;
460        };
461        rows.push(row);
462    }
463    let reported = |column| {
464        median(
465            &rows
466                .iter()
467                .filter_map(|&row| scalar_number(weather, row, column))
468                .collect::<Vec<_>>(),
469        )
470    };
471    let pressure = reported("PRESSURE");
472    let unit = weather
473        .column_keywords("PRESSURE")
474        .and_then(|record| record.get("QuantumUnits"))
475        .and_then(|value| match value {
476            Value::Array(ArrayValue::String(units)) => units.iter().next().map(String::as_str),
477            _ => None,
478        });
479    let hpa = if unit == Some("Pa") && pressure / 100.0 >= 100.0 {
480        pressure / 100.0
481    } else {
482        pressure
483    };
484    let temperature = reported("TEMPERATURE");
485    let humidity = reported("REL_HUMIDITY");
486    Conditions {
487        pressure_hpa: if hpa > 0.0 { hpa } else { default.pressure_hpa },
488        temperature_k: if !temperature.is_finite() {
489            default.temperature_k
490        } else if temperature > 100.0 {
491            temperature
492        } else {
493            temperature + 273.15
494        },
495        humidity_percent: if humidity > 0.0 {
496            humidity
497        } else {
498            default.humidity_percent
499        },
500    }
501}

508fn scan_times(ms: &Path) -> Result<BTreeMap<i32, Vec<ScanTimes>>, String> {
509    let table = Table::open(TableOptions::new(ms)).map_err(|error| error.to_string())?;
510    let fields = integers(&table, "FIELD_ID")?;
511    let scans = integers(&table, "SCAN_NUMBER")?;
512    let times = numbers(&table, "TIME")?;
513    let mut keyed = BTreeMap::<(i32, i32), Vec<f64>>::new();
514    for ((field, scan), time) in fields.into_iter().zip(scans).zip(times) {
515        if time.is_finite() {
516            keyed.entry((field, scan)).or_default().push(time);
517        }
518    }
519    let mut result = BTreeMap::<i32, Vec<ScanTimes>>::new();
520    for ((field, _), mut times) in keyed {
521        times.sort_unstable_by(f64::total_cmp);
522        times.dedup();
523        result.entry(field).or_default().push(ScanTimes {
524            bounds: [times[0], times[times.len() - 1]],
525            mean: times.iter().sum::<f64>() / times.len() as f64,
526        });
527    }
528    Ok(result)
529}

531fn pointing_elevation(table: &Table, antenna: i32, [start, stop]: [f64; 2]) -> Option<f64> {
532    let rows = (0..table.row_count())
533        .filter_map(|row| {
534            let ScalarValue::Int32(id) =
535                table.cell_accessor(row, "ANTENNA_ID").ok()?.scalar().ok()?
536            else {
537                return None;
538            };
539            if *id != antenna {
540                return None;
541            }
542            let time = scalar_number(table, row, "TIME")?;
543            let ArrayValue::Float64(direction) = array(table, row, "DIRECTION").ok()? else {
544                return None;
545            };
546            let elevation = direction.iter().nth(1).copied()?;
547            Some((time, elevation))
548        })
549        .collect::<Vec<_>>();
550    let first = rows.iter().position(|&(time, _)| time > start);
551    let last = rows.iter().rposition(|&(time, _)| time < stop);
552    if let (Some(first), Some(last)) = (first, last)
553        && first <= last
554    {
555        return Some(median(
556            &rows[first..=last]
557                .iter()
558                .map(|&(_, elevation)| elevation)
559                .collect::<Vec<_>>(),
560        ));
561    }
562    if let Some(first) = first {
563        return Some(rows[first].1);
564    }
565    (!rows.is_empty()).then(|| {
566        median(
567            &rows
568                .iter()
569                .map(|&(_, elevation)| elevation)
570                .collect::<Vec<_>>(),
571        )
572    })
573}

575fn airmass_for_elevation(elevation: f64) -> f64 {
576    if elevation <= 3.0_f64.to_radians() {
577        std::f64::consts::SQRT_2
578    } else {
579        elevation.sin().recip()
580    }
581}

583fn antenna_position(table: &Table, antenna: i32) -> Option<MPosition> {
584    let ArrayValue::Float64(coordinates) =
585        array(table, usize::try_from(antenna).ok()?, "POSITION").ok()?
586    else {
587        return None;
588    };
589    let [x, y, z] = coordinates.as_slice()? else {
590        return None;
591    };
592    Some(MPosition::new_itrf(*x, *y, *z))
593}

595fn nearest_time(table: &Table, column: &str, requested: f64) -> Option<usize> {
596    (0..table.row_count())
597        .filter_map(|row| Some((row, (scalar_number(table, row, column)? - requested).abs())))
598        .min_by(|left, right| left.1.total_cmp(&right.1))
599        .map(|(row, _)| row)
600}

602fn scalar_number(table: &Table, row: usize, name: &str) -> Option<f64> {
603    match table.cell_accessor(row, name).ok()?.scalar().ok()? {
604        ScalarValue::Float32(value) => Some(f64::from(*value)),
605        ScalarValue::Float64(value) => Some(*value),
606        _ => None,
607    }
608}

610fn text_cell<'a>(table: &'a Table, row: usize, name: &str) -> Option<&'a str> {
611    match table.cell_accessor(row, name).ok()?.scalar().ok()? {
612        ScalarValue::String(value) => Some(value),
613        _ => None,
614    }
615}

617fn open_optional(root: &Path, subtable: &str) -> Option<Table> {
618    root.join(subtable)
619        .is_dir()
620        .then(|| open(root, subtable).ok())
621        .flatten()
622}
