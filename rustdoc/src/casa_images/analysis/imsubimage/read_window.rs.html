13pub(in crate::analysis) struct PixelSections<'a, T: ImagePixel> {
14    pub(in crate::analysis) image: &'a dyn ImageInterface<T>,
15    pub(in crate::analysis) selection: &'a SubimageSelection,
16    pub(in crate::analysis) mask: Option<PreparedMask<'a>>,
17}
18
19impl<T: ImagePixel> PixelSections<'_, T> {
20    pub(in crate::analysis) fn read(
21        &self,
22        start: &[usize],
23        shape: &[usize],
24        stride: &[usize],
25    ) -> Result<(ArrayD<T>, Option<ArrayD<bool>>), ImageError> {
26        let pixels = self.image.get_slice(start, shape, stride)?;
27        let mut validity = self.image.get_mask_slice(start, shape, stride)?;
28        let section = StridedSection {
29            start,
30            shape,
31            stride,
32        };
33        if self.selection.stride.is_none() {
34            let extent = section.dense_shape(shape);
35            let mut membership = None;
36            self.selection
37                .apply_window_masks(&mut membership, start, &extent);
38            if let Some(membership) = membership {
39                let membership = membership.slice_each_axis(|slice| {
40                    let axis = slice.axis.index();
41                    sample_slice(0, shape[axis], stride[axis])
42                });
43                and_opt_mask(&mut validity, membership.to_owned());
44            }
45        }
46        if let Some(mask) = &self.mask {
47            and_opt_mask(&mut validity, mask.strided(&section)?);
48        }
49        Ok((pixels, validity))
50    }
51}
52
53pub(super) fn sample_slice(start: usize, count: usize, step: usize) -> Slice {
54    Slice::new(
55        start as isize,
56        Some((start + (count - 1) * step + 1) as isize),
57        step as isize,
58    )
59}
