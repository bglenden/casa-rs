177    pub(in crate::analysis) fn new_with_beams<T: ImagePixel>(
178        image: &dyn ImageInterface<T>,
179        native_coordinates: Option<RecordValue>,
180        options: SubimageOptions<'_>,
181        beam_record: Option<&RecordValue>,
182    ) -> Result<Self, ImageError> {
183        if matches!(&options.region, Some(SubimageRegion::Text(_)))
184            && image
185                .coordinates()
186                .find_coordinate(casa_coordinates::CoordinateType::Direction)
187                .is_none()
188        {
189            return Err(ImageError::CasaSelection(
190                "Coordinate system has no direction coordinate".into(),
191            ));
192        }
193        let selection =
194            SubimageSelection::resolve_section(image, None, options.region, None, None)?;
195        Self::from_selection(
196            image,
197            native_coordinates,
198            selection,
199            options.dropdeg,
200            options.keepaxes,
201            beam_record,
202        )
203    }

205    /// Resolve a writable selection, including a positive-stride pixel slice.
206    /// Returns the packed pixel geometry and its stride in the parent image.
207    ///
208    /// # Errors
209    ///
210    /// A region, coordinate system or stride the image cannot represent.
211    pub fn for_edit<T: ImagePixel>(
212        image: &dyn ImageInterface<T>,
213        region: Option<SubimageRegion<'_>>,
214    ) -> Result<(Self, Vec<usize>), ImageError> {
215        let plan = Self::new(
216            image,
217            None,
218            SubimageOptions {
219                region,
220                dropdeg: false,
221                keepaxes: &[],
222            },
223        )?;
224        let stride = plan
225            .selection
226            .stride
227            .clone()
228            .unwrap_or_else(|| vec![1; image.ndim()]);
229        Ok((plan, stride))
230    }

232    pub(in crate::analysis) fn from_selection<T: ImagePixel>(
233        image: &dyn ImageInterface<T>,
234        native_coordinates: Option<RecordValue>,
235        selection: SubimageSelection,
236        dropdeg: bool,
237        keepaxes: &[i64],
238        beam_record: Option<&RecordValue>,
239    ) -> Result<Self, ImageError> {
240        let packed_shape = selection.packed_shape();
241        let kept_axes = resolve_kept_axes(&packed_shape, dropdeg, keepaxes)?;
242        let step = |axis| selection.stride.as_ref().map_or(1, |stride| stride[axis]);
243        let cropped_stokes = if selection.selected_stokes.is_none()
244            && let Some(axis) = image.find_axis(casa_coordinates::CoordinateType::Stokes)
245            && let stokes = stokes_for_axis(image.coordinates(), axis)?
246            && (packed_shape[axis] != image.shape()[axis] || stokes.len() > image.shape()[axis])
247        {
248            let start = selection.start[axis];
249            Some(
250                stokes
251                    .get(start..start + selection.shape[axis])
252                    .ok_or_else(|| {
253                        ImageError::CasaSelection("Stokes region exceeds coordinate bounds".into())
254                    })?
255                    .iter()
256                    .step_by(step(axis))
257                    .copied()
258                    .collect::<Vec<_>>(),
259            )
260        } else {
261            None
262        };
263        let mut source_info = image.image_info()?;
264        if let Some(record) = beam_record {
265            source_info.beam_set = ImageInfo::from_record(record)?.beam_set;
266        }
267        let (coordinates, coordinate_record, image_info) = reduce_coordinates_and_beams(
268            image.coordinates(),
269            native_coordinates,
270            CoordReductionSpec {
271                full_shape: image.shape(),
272                start: &selection.start,
273                shape: &packed_shape,
274                stride: selection.stride.as_deref(),
275                kept_axes: &kept_axes,
276                selected_stokes: selection
277                    .selected_stokes
278                    .as_deref()
279                    .or(cropped_stokes.as_deref()),
280            },
281            source_info,
282        )?;
283        let shape = kept_axes.iter().map(|&axis| packed_shape[axis]).collect();
284        Ok(Self {
285            selection,
286            kept_axes,
287            shape,
288            coordinates,
289            coordinate_record,
290            image_info,
291            stored_image_info: None,
292        })
293    }

310        let first = *self.image_info.beam_set.beam(0, 0);

321    /// The lower and upper pixel bounds in the source image, inclusive.
322    pub fn bounds(&self) -> (&[usize], Vec<usize>) {
323        let end = self
324            .selection
325            .start
326            .iter()
327            .zip(self.selection.packed_shape())
328            .enumerate()
329            .map(|(axis, (start, count))| {
330                let step = self
331                    .selection
332                    .stride
333                    .as_ref()
334                    .map_or(1, |stride| stride[axis]);
335                start + (count - 1) * step
336            })
337            .collect();
338        (&self.selection.start, end)
339    }

346    fn transformed_coordinate_record(
347        &self,
348        coordinates: &casa_coordinates::CoordinateSystem,
349        axes: &[usize],
350    ) -> Result<RecordValue, ImageError> {
351        let mut output = coordinates.to_record();
352        if let Some(native) = &self.coordinate_record {
353            let originals = self.coordinates.coordinates_in_record(native)?;
354            let transformed = coordinates.coordinates_in_record(&output)?;
355            for (source, destination) in originals.iter().zip(transformed) {
356                if !source
357                    .pixel_axes
358                    .iter()
359                    .flatten()
360                    .any(|axis| axes.contains(axis))
361                    && let Some(value) = native.get(&source.field)
362                {
363                    output.upsert(destination.field, value.clone());
364                }
365            }
366        }
367        Ok(output)
368    }

401    /// Copies selected pixels, image metadata and the combined mask into a
402    /// new image allocated with the plan's shape and coordinates. The input
403    /// must be the source used to create the plan. Output is populated in
404    /// place, so file publication belongs to the caller after success.
405    pub fn copy_to<T: ImagePixel + Copy>(
406        &self,
407        image: &dyn ImageInterface<T>,
408        output: &mut dyn MutableImageInterface<T>,
409        mask_expression: Option<&str>,
410        stretch: bool,
411    ) -> Result<(), ImageError> {
412        self.copy_metadata(image, output)?;
413        let selection = &self.selection;
414        if let Some(stride) = selection.stride.as_deref() {
415            return self.copy_strided_to(image, output, mask_expression, stretch, stride);
416        }
417        let mut mask =
418            image.get_mask_slice(&selection.start, &selection.shape, &vec![1; image.ndim()])?;
419        selection.apply_selection_masks(&mut mask);
420        if let Some(expression) = evaluate_mask(
421            mask_expression,
422            image.shape(),
423            image.coordinates(),
424            stretch,
425            (&selection.start, &selection.shape),
426        )? {
427            and_opt_mask(&mut mask, expression);
428        }
429        self.write_selection_mask(output, mask)?;
430        let spec = TraversalSpec::tiles().section(selection.start.clone(), selection.shape.clone());
431        for chunk in TraversalIter::new(image, spec) {
432            let chunk = chunk?;
433            let position = self
434                .kept_axes
435                .iter()
436                .map(|&axis| chunk.cursor.position[axis] - selection.start[axis])
437                .collect::<Vec<_>>();
438            output.put_slice(&self.squeeze(chunk.data), &position)?;
439        }
440        Ok(())
441    }

443    fn copy_strided_to<T: ImagePixel + Copy>(
444        &self,
445        image: &dyn ImageInterface<T>,
446        output: &mut dyn MutableImageInterface<T>,
447        mask_expression: Option<&str>,
448        stretch: bool,
449        stride: &[usize],
450    ) -> Result<(), ImageError> {
451        let packed_shape = self.selection.packed_shape();
452        let section = StridedSection {
453            start: &self.selection.start,
454            shape: &packed_shape,
455            stride,
456        };
457        let mut mask = evaluate_strided_mask(
458            mask_expression,
459            image.shape(),
460            image.coordinates(),
461            stretch,
462            &section,
463        )?;
464        for cursor in section.cursors(image.nice_cursor_shape()) {
465            let cursor = cursor?;
466            let source = section.source_start(&cursor.position);
467            if let Some(inherited) = image.get_mask_slice(&source, &cursor.shape, stride)? {
468                let target =
469                    mask.get_or_insert_with(|| ArrayD::from_elem(IxDyn(&packed_shape), true));
470                let mut tile = target.slice_each_axis_mut(|axis| {
471                    let index = axis.axis.index();
472                    let start = cursor.position[index];
473                    Slice::from(start..start + cursor.shape[index])
474                });
475                Zip::from(&mut tile)
476                    .and(&inherited)
477                    .for_each(|out, &valid| *out &= valid);
478            }
479            let values = image.get_slice(&source, &cursor.shape, stride)?;
480            let destination = self
481                .kept_axes
482                .iter()
483                .map(|&axis| cursor.position[axis])
484                .collect::<Vec<_>>();
485            output.put_slice(&self.squeeze(values), &destination)?;
486        }
487        self.write_selection_mask(output, mask)
488    }

490    /// Copy the selected real and imaginary sections into a complex image.
491    /// Coordinates and metadata come from the real image used to plan the section.
492    pub fn make_complex_to<T>(
493        &self,
494        real: &dyn ImageInterface<T>,
495        imaginary: &dyn ImageInterface<T>,
496        output: &mut dyn MutableImageInterface<Complex<T>>,
497        region: Option<SubimageRegion<'_>>,
498    ) -> Result<(), ImageError>
499    where
500        T: ImagePixel + Copy,
501        Complex<T>: ImagePixel + Copy,
502    {
503        let selection = &self.selection;
504        if selection.stride.is_some() {
505            return Err(ImageError::InvalidMetadata(
506                "makecomplex requires a unit-stride image section".into(),
507            ));
508        }
509        let imaginary_selection = SubimageSelection::resolve(imaginary, None, region, None, None)?;
510        if selection.shape != imaginary_selection.shape {
511            return Err(ImageError::CasaSelection(
512                "Real and imaginary parts have different shapes".into(),
513            ));
514        }
515        self.copy_metadata(real, output)?;
516        let stride = vec![1; real.ndim()];
517        let mut real_membership = None;
518        let mut imaginary_membership = None;
519        selection.apply_selection_masks(&mut real_membership);
520        imaginary_selection.apply_selection_masks(&mut imaginary_membership);
521        let check_mask = real.has_pixel_mask()
522            || imaginary.has_pixel_mask()
523            || real_membership.is_some()
524            || imaginary_membership.is_some();
525        let mut has_invalid = false;
526
527        let spec = TraversalSpec::tiles().section(selection.start.clone(), selection.shape.clone());
528        for chunk in TraversalIter::new(real, spec) {
529            let chunk = chunk?;
530            let relative: Vec<_> = chunk
531                .cursor
532                .position
533                .iter()
534                .zip(&selection.start)
535                .map(|(&position, &start)| position - start)
536                .collect();
537            let imaginary_origin: Vec<_> = relative
538                .iter()
539                .zip(&imaginary_selection.start)
540                .map(|(&offset, &start)| offset + start)
541                .collect();
542            let imag = imaginary.get_slice(&imaginary_origin, &chunk.cursor.shape, &stride)?;
543            let data = Zip::from(&chunk.data)
544                .and(&imag)
545                .map_collect(|&real, &imag| Complex::new(real, imag));
546            let position: Vec<_> = self.kept_axes.iter().map(|&axis| relative[axis]).collect();
547            if check_mask {
548                let mut mask =
549                    real.get_mask_slice(&chunk.cursor.position, &chunk.cursor.shape, &stride)?;
550                if let Some(imaginary_mask) =
551                    imaginary.get_mask_slice(&imaginary_origin, &chunk.cursor.shape, &stride)?
552                {
553                    and_opt_mask(&mut mask, imaginary_mask);
554                }
555                for membership in [&real_membership, &imaginary_membership]
556                    .into_iter()
557                    .flatten()
558                {
559                    let slices: Vec<_> = relative
560                        .iter()
561                        .zip(&chunk.cursor.shape)
562                        .map(|(&start, &size)| SliceInfoElem::Slice {
563                            start: start as isize,
564                            end: Some((start + size) as isize),
565                            step: 1,
566                        })
567                        .collect();
568                    and_opt_mask(&mut mask, membership.slice(slices.as_slice()).to_owned());
569                }
570                let mask =
571                    mask.unwrap_or_else(|| ArrayD::from_elem(IxDyn(&chunk.cursor.shape), true));
572                has_invalid |= mask.iter().any(|&valid| !valid);
573                output.put_mask_slice("mask0", &self.squeeze(mask), &position)?;
574            }
575            output.put_slice(&self.squeeze(data), &position)?;
576        }
577        if has_invalid {
578            output.set_default_mask("mask0")?;
579        } else if check_mask {
580            output.remove_mask("mask0")?;
581        }
582        Ok(())
583    }

585    /// Indices of the original beam planes retained on a selected source axis.
586    pub fn beam_planes(&self, axis: Option<usize>, planes: usize) -> Vec<usize> {
587        let Some(axis) = axis.filter(|_| planes > 1) else {
588            return vec![0];
589        };
590        let selected = &self.selection;
591        let start = selected.start[axis];
592        let step = selected.stride.as_ref().map_or(1, |stride| stride[axis]);
593        (0..selected.shape[axis].div_ceil(step))
594            .map(|index| start + index * step)
595            .collect()
596    }

598    fn copy_metadata<T: ImagePixel, U: ImagePixel>(
599        &self,
600        source: &dyn ImageInterface<T>,
601        output: &mut dyn MutableImageInterface<U>,
602    ) -> Result<(), ImageError> {
603        if output.shape() != self.shape {
604            return Err(ImageError::ShapeMismatch {
605                expected: self.shape.clone(),
606                got: output.shape().to_vec(),
607            });
608        }
609        output.set_units_string(source.units().to_owned())?;
610        if let Some(info) = &self.stored_image_info {
611            output.set_image_info_record(info.clone())?;
612        } else {
613            output.set_image_info(&self.image_info)?;
614        }
615        output.set_misc_info(source.misc_info())
616    }

618    fn write_selection_mask<T: ImagePixel>(
619        &self,
620        output: &mut dyn MutableImageInterface<T>,
621        mask: Option<ArrayD<bool>>,
622    ) -> Result<(), ImageError> {
623        if let Some(mask) = mask
624            && mask.iter().any(|valid| !valid)
625        {
626            output.put_mask("mask0", &self.squeeze(mask))?;
627            output.set_default_mask("mask0")?;
628        }
629        Ok(())
630    }

632    fn squeeze<T>(&self, mut values: ArrayD<T>) -> ArrayD<T> {
633        for axis in (0..values.ndim()).rev() {
634            if !self.kept_axes.contains(&axis) {
635                values = values.index_axis_move(Axis(axis), 0);
636            }
637        }
638        values
639    }

661    fn paged(&self) -> Option<&PagedImage<T>> {
662        match self {
663            Self::Paged { image, .. } => Some(image.as_ref()),
664            Self::Memory { .. } => None,
665        }
666    }
667
668    fn history(&self) -> &[ImageHistoryEntry] {
669        match self {
670            Self::Paged { history, .. } | Self::Memory { history, .. } => history,
671        }
672    }
673}

675pub(in crate::analysis) fn run(
676    request: &ImsubimageRequest,
677) -> Result<ImageSubimageSummary, ImageError> {
678    let out_str = request.outfile.to_string_lossy();
679    if out_str.trim().is_empty() {
680        return Err(ImageError::InvalidMetadata(
681            "outfile must be specified.".into(),
682        ));
683    }
684    ensure_outfile_does_not_alias_input(&request.outfile, &request.imagename)?;
685    if request.outfile.exists() && !request.overwrite {
686        return Err(ImageError::Io(format!(
687            "subimage output already exists: {}",
688            request.outfile.display()
689        )));
690    }
691
692    if request.imagename.is_file() {
693        return run_fits_input(request);
694    }
695
696    let history = image_history_entries(&request.imagename)?;
697    match AnyPagedImage::open(&request.imagename)? {
698        AnyPagedImage::Float32(img) => {
699            let src = SourceData::Paged {
700                image: Box::new(img),
701                history,
702            };
703            run_typed(&src, request)
704        }
705        AnyPagedImage::Float64(img) => {
706            let src = SourceData::Paged {
707                image: Box::new(img),
708                history,
709            };
710            run_typed(&src, request)
711        }
712        AnyPagedImage::Complex32(img) => {
713            let src = SourceData::Paged {
714                image: Box::new(img),
715                history,
716            };
717            run_typed(&src, request)
718        }
719        AnyPagedImage::Complex64(img) => {
720            let src = SourceData::Paged {
721                image: Box::new(img),
722                history,
723            };
724            run_typed(&src, request)
725        }
726    }
727}

729/// Refuses to replace the input image, its ancestors, or existing entries inside it.
730pub fn ensure_outfile_does_not_alias_input(
731    outfile: &Path,
732    imagename: &Path,
733) -> Result<(), ImageError> {
734    let input = fs::canonicalize(imagename).ok();
735    let output = fs::canonicalize(outfile).ok();
736    if outfile == imagename
737        || input
738            .as_ref()
739            .zip(output.as_ref())
740            .is_some_and(|(input, output)| input.starts_with(output))
741    {
742        return Err(ImageError::InvalidMetadata(format!(
743            "outfile '{}' matches or contains input image '{}'",
744            outfile.display(),
745            imagename.display()
746        )));
747    }
748    if outfile.symlink_metadata().is_ok()
749        && input.as_ref().is_some_and(|input| {
750            let output_parent = outfile.parent().and_then(|parent| {
751                // A bare relative name has an empty parent that refers to this directory.
752                fs::canonicalize(if parent.as_os_str().is_empty() {
753                    Path::new(".")
754                } else {
755                    parent
756                })
757                .ok()
758            });
759            output
760                .iter()
761                .chain(output_parent.iter())
762                .any(|path| path.starts_with(input))
763        })
764    {
765        return Err(ImageError::InvalidMetadata(format!(
766            "outfile '{}' is inside input image '{}'",
767            outfile.display(),
768            imagename.display()
769        )));
770    }
771    Ok(())
772}

774/// Reject overlapping table, image or text paths before an operation creates
775/// any of them. Existing symlinks are resolved along each path.
776///
777/// # Errors
778/// Paths overlap, or their existing components cannot be inspected.
779pub fn ensure_artifacts_do_not_overlap(first: &Path, second: &Path) -> Result<(), ImageError> {
780    let a = resolve_artifact_path(first)?;
781    let b = resolve_artifact_path(second)?;
782    if a.starts_with(&b) || b.starts_with(&a) {
783        return Err(ImageError::CasaSelection(format!(
784            "Output path '{}' overlaps '{}'",
785            first.display(),
786            second.display()
787        )));
788    }
789    Ok(())
790}
791
792fn resolve_artifact_path(path: &Path) -> Result<std::path::PathBuf, ImageError> {
793    use std::path::Component;
794
795    let inspect_error = |error| {
796        ImageError::CasaSelection(format!(
797            "Could not resolve path '{}': {error}",
798            path.display()
799        ))
800    };
801    let absolute = std::path::absolute(path).map_err(inspect_error)?;
802    let mut resolved = std::path::PathBuf::new();
803    for part in absolute.components() {
804        if part == Component::ParentDir {
805            resolved.pop();
806        } else {
807            resolved.push(part);
808        }
809        if matches!(part, Component::Normal(_)) {
810            match resolved.canonicalize() {
811                Ok(canonical) => resolved = canonical,
812                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
813                Err(error) => return Err(inspect_error(error)),
814            }
815        }
816    }
817    Ok(resolved)
818}

820fn run_fits_input(request: &ImsubimageRequest) -> Result<ImageSubimageSummary, ImageError> {
821    let history = image_history_entries(&request.imagename)?;
822    let fits = read_fits_image(&request.imagename)?;
823    let mask = fits
824        .pixels
825        .iter()
826        .any(|v| v.is_nan())
827        .then(|| fits.pixels.mapv(|v| !v.is_nan()));
828    let image = build_fits_temp(
829        fits.shape,
830        fits.coordinates,
831        &fits.units,
832        &fits.image_info,
833        fits.pixels,
834        mask,
835    )?;
836    run_typed(
837        &SourceData::Memory {
838            image: Box::new(image),
839            history,
840        },
841        request,
842    )
843}

845fn build_fits_temp<T: ImagePixel>(
846    shape: Vec<usize>,
847    coordinates: casa_coordinates::CoordinateSystem,
848    units: &str,
849    info: &ImageInfo,
850    pixels: ArrayD<T>,
851    mask: Option<ArrayD<bool>>,
852) -> Result<TempImage<T>, ImageError> {
853    let ndim = shape.len();
854    let mut temp = TempImage::new(shape, coordinates, TempStoragePolicy::Memory)?;
855    temp.set_units(units)?;
856    temp.set_image_info(info)?;
857    temp.put_slice(&pixels, &vec![0; ndim])?;
858    if let Some(m) = mask {
859        temp.put_mask("mask0", &m)?;
860        temp.set_default_mask("mask0")?;
861    }
862    Ok(temp)
863}

865fn run_typed<T: ImagePixel + Copy>(
866    source: &SourceData<T>,
867    request: &ImsubimageRequest,
868) -> Result<ImageSubimageSummary, ImageError> {
869    let img = source.as_interface();
870    let region = request
871        .region
872        .as_deref()
873        .map(str::trim)
874        .filter(|text| !text.is_empty())
875        .map(SubimageRegion::Text);
876    let selection = SubimageSelection::resolve_section(
877        img,
878        request.box_pixels.as_deref(),
879        region,
880        request.chans.as_deref(),
881        request.stokes.as_deref(),
882    )?;
883    let native_coordinates =
884        source
885            .paged()
886            .and_then(|image| match image.table().keywords().get("coords") {
887                Some(Value::Record(record)) => Some(record.clone()),
888                _ => None,
889            });
890    let plan = SubimagePlan::from_selection(
891        img,
892        native_coordinates,
893        selection,
894        request.dropdeg,
895        &request.keepaxes,
896        None,
897    )?;
898
899    let destination = std::path::absolute(&request.outfile).map_err(output_io)?;
900    let parent = destination.parent().ok_or_else(|| {
901        ImageError::InvalidMetadata("outfile must name an image directory".into())
902    })?;
903    let stage = tempfile::tempdir_in(parent).map_err(output_io)?;
904    let product = stage.path().join("image");
905    let mut output =
906        PagedImage::<T>::create(plan.shape.clone(), plan.coordinates.clone(), &product)?;
907    if let Some(record) = &plan.coordinate_record {
908        output.set_coords_record_keyword(record.clone());
909    }
910    plan.copy_to(img, &mut output, request.mask.as_deref(), request.stretch)?;
911    output.save()?;
912    drop(output);
913    write_output_history(&product, source.history(), request)?;
914
915    match rename_without_replacing(&product, &destination) {
916        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists && request.overwrite => {
917            exchange_directories(&product, &destination).map_err(output_io)?;
918        }
919        result => result.map_err(output_io)?,
920    }
921
922    let (blc, trc) = plan.bounds();
923    Ok(ImageSubimageSummary {
924        imagename: request.imagename.display().to_string(),
925        outfile: request.outfile.display().to_string(),
926        blc: blc.to_vec(),
927        trc,
928        shape: plan.shape,
929        units: img.units().to_string(),
930    })
931}

933fn output_io(error: std::io::Error) -> ImageError {
934    ImageError::Io(error.to_string())
935}

937fn and_opt_mask(dst: &mut Option<ArrayD<bool>>, rhs: ArrayD<bool>) {
938    *dst = Some(match dst.take() {
939        Some(mut lhs) => {
940            Zip::from(&mut lhs).and(&rhs).for_each(|a, &b| *a = *a && b);
941            lhs
942        }
943        None => rhs,
944    });
945}

947fn resolve_kept_axes(
948    selected_shape: &[usize],
949    dropdeg: bool,
950    keepaxes: &[i64],
951) -> Result<Vec<usize>, ImageError> {
952    let ndim = selected_shape.len();
953    if !dropdeg {
954        return Ok((0..ndim).collect());
955    }
956    let mut keep_set = BTreeSet::new();
957    for &ax in keepaxes {
958        if ax < 0 {
959            // Matches CASA 6.7.6.14: negative axis entries are ignored.
960            continue;
961        }
962        let u = ax as usize;
963        if u >= ndim {
964            return Err(ImageError::InvalidMetadata(format!(
965                "keepaxes axis {ax} is out of range for {ndim}-D image"
966            )));
967        }
968        keep_set.insert(u);
969    }
970    let kept: Vec<usize> = (0..ndim)
971        .filter(|&ax| selected_shape[ax] > 1 || keep_set.contains(&ax))
972        .collect();
973    if kept.is_empty() {
974        return Err(ImageError::InvalidMetadata(
975            "cannot drop all axes of image (result would have 0 dimensions)".into(),
976        ));
977    }
978    Ok(kept)
979}
