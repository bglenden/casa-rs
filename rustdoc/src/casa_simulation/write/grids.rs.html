2//! The per-field model grids `close` prepares and keeps.
3//!
4//! A step that predicts several fields under the primary beam holds its
5//! model's planes, and `close` prepares a field's grids, the beam centred
6//! on it, as its write reaches the field
7//! ([`crate::session::prediction::Prediction::prepare_field`]). The cache
8//! keeps as many sets as its admitted bytes hold, at least those of the
9//! segment being predicted. The write visits the scans in order and knows
10//! them all, so when a set must go the cache evicts the one used again
11//! latest, or never (Belady's rule). A field evicted and visited again is
12//! prepared again, as CASA's `GridFT` transforms the model again whenever
13//! the field changes. Its grids are the same however often they are
14//! prepared, so the MeasurementSet does not depend on the budget.
15
16use std::sync::Arc;
17use std::time::{Duration, Instant};
18
19use casa_imaging_operator::PreparedModelGrids;
20
21use crate::session::error::{SimulatorError, SimulatorResult};
22
23/// A set of per-field grids: the step's index among the plan's predict
24/// steps, and the field.
25pub(super) type GridKey = (usize, usize);
26
27/// The per-field grids `close` holds, within its admitted bytes.
28pub(super) struct FieldGridCache<G = PreparedModelGrids> {
29    /// Bytes one set of each step costs, by step: its grids, and on Metal
30    /// their device copy as much again; zero for a step whose grids
31    /// `predict` prepared.
32    costs: Vec<u64>,
33    /// Each key's visits, by the first scan of each, ascending; sorted by
34    /// key.
35    visits: Vec<(GridKey, Vec<usize>)>,
36    /// The sets held.
37    resident: Vec<(GridKey, Arc<G>)>,
38    /// Bytes the sets may hold, as admitted.
39    capacity: u64,
40    /// Bytes the sets hold.
41    held: u64,
42    /// Sets prepared, and the time spent preparing them.
43    preparations: usize,
44    preparing: Duration,
45}
46
47impl<G> FieldGridCache<G> {
48    /// A cache of sets costing `costs` bytes by step, whose keys are used
49    /// in the visits `visits` (each key's visits by their first scans, in
50    /// any order). It holds nothing until it is given its bytes
51    /// ([`Self::set_capacity`]). Visits gathered scan by scan carry spare
52    /// room, up to as much again; the cache gives it back.
53    pub(super) fn new(costs: Vec<u64>, mut visits: Vec<(GridKey, Vec<usize>)>) -> Self {
54        visits.sort_unstable_by_key(|(key, _)| *key);
55        visits.shrink_to_fit();
56        for (_, firsts) in &mut visits {
57            firsts.sort_unstable();
58            firsts.shrink_to_fit();
59        }
60        Self {
61            costs,
62            visits,
63            resident: Vec::new(),
64            capacity: 0,
65            held: 0,
66            preparations: 0,
67            preparing: Duration::ZERO,
68        }
69    }
70
71    /// Bytes one set of `key`'s step costs.
72    pub(super) fn cost(&self, (step, _): GridKey) -> u64 {
73        self.costs[step]
74    }
75
76    /// Bytes every set the visits use costs together: the most the cache
77    /// holds, preparing each set once.
78    pub(super) fn all_bytes(&self) -> u64 {
79        self.visits.iter().map(|(key, _)| self.cost(*key)).sum()
80    }
81
82    /// Bytes the cache's bookkeeping holds: its visits and costs, as
83    /// allocated, and room to list every set it may hold.
84    pub(super) fn state_bytes(&self) -> u64 {
85        let firsts: usize = self
86            .visits
87            .iter()
88            .map(|(_, firsts)| firsts.capacity() * size_of::<usize>())
89            .sum();
90        (firsts
91            + self.visits.capacity() * size_of::<(GridKey, Vec<usize>)>()
92            + self.costs.capacity() * size_of::<u64>()
93            + self.visits.len() * size_of::<(GridKey, Arc<G>)>()) as u64
94    }
95
96    /// Give the cache `capacity` bytes, as admitted, and room to list every
97    /// set it may hold.
98    pub(super) fn set_capacity(&mut self, capacity: u64) {
99        self.capacity = capacity;
100        self.resident = Vec::with_capacity(self.visits.len());
101    }
102
103    /// Sets prepared so far, and the time spent preparing them.
104    pub(super) fn preparations(&self) -> (usize, Duration) {
105        (self.preparations, self.preparing)
106    }
107
108    /// The grids of `key` for the visit starting at scan `scan`: held, or
109    /// prepared by `prepare` once the sets used again latest, other than
110    /// the visit's own `keep`, are evicted to make room.
111    ///
112    /// An evicted set is freed before the next is prepared: the cache must
113    /// hold its only reference then, as the stage drops its predictor
114    /// before it selects another.
115    ///
116    /// # Errors
117    ///
118    /// `prepare`'s error, or a raised one when the sets of `keep` and this
119    /// one do not fit the cache, which is admitted to hold them, or when a
120    /// set to evict is still in use, whose bytes would stay beside the
121    /// next.
122    pub(super) fn grids(
123        &mut self,
124        key: GridKey,
125        (scan, keep): (usize, &[GridKey]),
126        prepare: impl FnOnce() -> SimulatorResult<G>,
127    ) -> SimulatorResult<Arc<G>> {
128        if let Some((_, grids)) = self.resident.iter().find(|(held, _)| *held == key) {
129            return Ok(Arc::clone(grids));
130        }
131        let cost = self.cost(key);
132        while self.held + cost > self.capacity {
133            let victim = self
134                .resident
135                .iter()
136                .enumerate()
137                .filter(|(_, (held, _))| !keep.contains(held))
138                .max_by_key(|(_, (held, _))| self.next_visit(*held, scan).unwrap_or(usize::MAX))
139                .map(|(index, _)| index)
140                .ok_or_else(|| {
141                    SimulatorError::raised(format!(
142                        "the per-field model grids' {} bytes cannot hold a set of {cost} bytes \
143                         beside the {} the visit holds",
144                        self.capacity, self.held
145                    ))
146                })?;
147            let (evicted, grids) = self.resident.swap_remove(victim);
148            self.held -= self.cost(evicted);
149            if Arc::into_inner(grids).is_none() {
150                return Err(SimulatorError::raised(format!(
151                    "the per-field model grids of step {}, field {}, are still in use when evicted",
152                    evicted.0, evicted.1
153                )));
154            }
155        }
156        let started = Instant::now();
157        let grids = Arc::new(prepare()?);
158        self.preparing += started.elapsed();
159        self.preparations += 1;
160        self.held += cost;
161        self.resident.push((key, Arc::clone(&grids)));
162        Ok(grids)
163    }
164
165    /// The first scan of `key`'s next visit after the visit at scan
166    /// `scan`; `None` when it is not visited again.
167    fn next_visit(&self, key: GridKey, scan: usize) -> Option<usize> {
168        let index = self
169            .visits
170            .binary_search_by_key(&key, |(known, _)| *known)
171            .ok()?;
172        let firsts = &self.visits[index].1;
173        firsts
174            .get(firsts.partition_point(|&first| first <= scan))
175            .copied()
176    }
177}

183    /// The fields a cache of `slots` one-byte sets of one step prepares, in
184    /// order, for fields visited in `order`, one scan a visit.
185    fn prepared(order: &[usize], slots: u64) -> Vec<usize> {
186        let mut visits: Vec<(GridKey, Vec<usize>)> = Vec::new();
187        for (scan, &field) in order.iter().enumerate() {
188            match visits.iter_mut().find(|(key, _)| *key == (0, field)) {
189                Some((_, firsts)) => firsts.push(scan),
190                None => visits.push(((0, field), vec![scan])),
191            }
192        }
193        let mut cache = FieldGridCache::<usize>::new(vec![1], visits);
194        assert_eq!(
195            cache.all_bytes(),
196            order.iter().max().map_or(0, |n| *n as u64 + 1)
197        );
198        cache.set_capacity(slots);
199        let mut prepared = Vec::new();
200        for (scan, &field) in order.iter().enumerate() {
201            let key = (0, field);
202            let grids = cache
203                .grids(key, (scan, &[key]), || {
204                    prepared.push(field);
205                    Ok(field)
206                })
207                .expect("one set fits");
208            assert_eq!(*grids, field);
209            assert!(cache.held <= slots);
210        }
211        assert_eq!(cache.preparations().0, prepared.len());
212        prepared
213    }

215    #[test]
216    fn every_field_is_prepared_once_when_all_fit() {
217        assert_eq!(prepared(&[0, 1, 2, 0, 1, 2], 3), vec![0, 1, 2]);
218    }
219
220    #[test]
221    fn a_second_pass_evicts_the_sets_used_again_latest() {
222        // Two passes over four fields with three slots: the first pass
223        // evicts 2, used again last, for 3. The second finds 0 and 1,
224        // prepares 2 again in place of a set not used again, and finds 3:
225        // five preparations, where evicting the least recently used set
226        // would prepare all eight visits.
227        let order = [0, 1, 2, 3, 0, 1, 2, 3];
228        assert_eq!(prepared(&order, 3), vec![0, 1, 2, 3, 2]);
229        // One slot prepares every visit.
230        assert_eq!(prepared(&order, 1), order.to_vec());
231    }
232
233    #[test]
234    fn a_visits_own_sets_are_never_evicted() {
235        let mut cache =
236            FieldGridCache::<()>::new(vec![1, 1], vec![((0, 0), vec![0]), ((1, 0), vec![0])]);
237        cache.set_capacity(1);
238        let keep = [(0, 0), (1, 0)];
239        cache
240            .grids((0, 0), (0, &keep), || Ok(()))
241            .expect("one set fits");
242        let error = cache
243            .grids((1, 0), (0, &keep), || Ok(()))
244            .expect_err("two sets do not fit one byte");
245        assert!(error.to_string().contains("cannot hold"), "{error}");
246    }

248    #[test]
249    fn a_set_still_in_use_is_not_evicted_silently() {
250        // A predictor kept past its segment would keep its grids: evicting
251        // them would free nothing (#922 review).
252        let mut cache =
253            FieldGridCache::<usize>::new(vec![1], vec![((0, 0), vec![0]), ((0, 1), vec![1])]);
254        cache.set_capacity(1);
255        let in_use = cache
256            .grids((0, 0), (0, &[(0, 0)]), || Ok(0))
257            .expect("one set fits");
258        let mut prepared = false;
259        let error = cache
260            .grids((0, 1), (1, &[(0, 1)]), || {
261                prepared = true;
262                Ok(1)
263            })
264            .expect_err("the set to evict is in use");
265        assert!(error.to_string().contains("still in use"), "{error}");
266        assert!(!prepared, "nothing is prepared beside a set in use");
267        assert_eq!(*in_use, 0);
268    }

270    #[test]
271    fn the_bookkeeping_charges_what_its_visits_keep() {
272        // Two fields alternating over 2 × 65,537 scans: visits gathered by
273        // pushing keep room for 131,072 scans each, about twice their
274        // length (#922 review).
275        const VISITS: usize = 65_537;
276        let mut visits: Vec<(GridKey, Vec<usize>)> = Vec::new();
277        for field in 0..2 {
278            let mut firsts = Vec::new();
279            for visit in 0..VISITS {
280                firsts.push(2 * visit + field);
281            }
282            assert!(firsts.capacity() > VISITS * 3 / 2);
283            visits.push(((0, field), firsts));
284        }
285        let cache = FieldGridCache::<()>::new(vec![1], visits);
286        let kept = cache.visits.capacity() * size_of::<(GridKey, Vec<usize>)>()
287            + cache
288                .visits
289                .iter()
290                .map(|(_, firsts)| firsts.capacity() * size_of::<usize>())
291                .sum::<usize>()
292            + cache.costs.capacity() * size_of::<u64>();
293        assert!(
294            cache.state_bytes() >= kept as u64,
295            "charged {} bytes for {kept} kept",
296            cache.state_bytes()
297        );
298        // The cache gives the spare room back.
299        assert!(
300            kept < 2 * VISITS * size_of::<usize>() * 5 / 4,
301            "{kept} bytes kept for {} visits",
302            2 * VISITS
303        );
304    }
