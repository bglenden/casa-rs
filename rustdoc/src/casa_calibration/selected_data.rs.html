14/// The data a solve selected: the sorted, distinct values each axis of its
15/// cross-correlation rows takes, and the intents the MeasurementSet names.
16///
17/// CASA's `gaincal` and `bandpass` return this as `selectvis`. Their
18/// selection always leaves out autocorrelations (`ANTENNA1!=ANTENNA2`), so a
19/// row whose antennas are equal does not count here either. Flags do not
20/// narrow it: an antenna whose data are all flagged is still selected.
21/// The intents are not narrowed by the selection at all: CASA 6.7.6 lists
22/// every `OBS_MODE` of the `STATE` table, split at its commas, sorted and
23/// distinct, and none when `STATE` has no rows. An empty `OBS_MODE` names
24/// none, though an empty item between two commas is kept.
25#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
26pub struct SelectedData {
27    /// The antennas of the selected baselines, `ANTENNA1` and `ANTENNA2`.
28    pub antennas: Vec<i32>,
29    /// The selected `FIELD_ID`s.
30    pub fields: Vec<i32>,
31    /// The intents the `STATE` table names.
32    pub intents: Vec<String>,
33    /// The selected `OBSERVATION_ID`s.
34    pub observations: Vec<i32>,
35    /// The selected `SCAN_NUMBER`s.
36    pub scans: Vec<i32>,
37    /// The selected spectral windows, through `DATA_DESCRIPTION`.
38    pub spectral_windows: Vec<i32>,
39}
40
41impl SelectedData {
42    /// The data `rows` of `ms` select.
43    pub(crate) fn of(
44        ms: &MeasurementSet,
45        rows: &[SelectedSolveRow],
46    ) -> Result<Self, GainSolveError> {
47        Self::of_rows(ms, rows.iter().filter(|row| row.antenna1 != row.antenna2))
48    }
49
50    /// The autocorrelation rows selected for sampler calibration.
51    pub(crate) fn of_autocorrelations(
52        ms: &MeasurementSet,
53        rows: &[SelectedSolveRow],
54    ) -> Result<Self, GainSolveError> {
55        Self::of_rows(ms, rows.iter().filter(|row| row.antenna1 == row.antenna2))
56    }
57
58    pub(crate) fn of_rows<'a>(
59        ms: &MeasurementSet,
60        rows: impl Iterator<Item = &'a SelectedSolveRow>,
61    ) -> Result<Self, GainSolveError> {
62        let mut selected = SelectedDataBuilder::default();
63        for row in rows {
64            selected.add(
65                [row.antenna1, row.antenna2],
66                row.field_id,
67                row.observation_id,
68                row.scan_number,
69                row.data_spw_id,
70            );
71        }
72        selected.finish(ms)
73    }
74}
75
76/// The distinct identities of selected rows, accumulated without retaining rows.
77#[derive(Default)]
78pub(crate) struct SelectedDataBuilder {
79    antennas: BTreeSet<i32>,
80    fields: BTreeSet<i32>,
81    observations: BTreeSet<i32>,
82    scans: BTreeSet<i32>,
83    windows: BTreeSet<i32>,
84}
85
86impl SelectedDataBuilder {
87    pub(crate) fn add(
88        &mut self,
89        antennas: [i32; 2],
90        field: i32,
91        observation: i32,
92        scan: i32,
93        window: i32,
94    ) {
95        self.antennas.extend(antennas);
96        self.fields.insert(field);
97        self.observations.insert(observation);
98        self.scans.insert(scan);
99        self.windows.insert(window);
100    }
101
102    #[cfg(unix)]
103    pub(crate) fn is_empty(&self) -> bool {
104        self.antennas.is_empty()
105    }
106
107    pub(crate) fn finish(self, ms: &MeasurementSet) -> Result<SelectedData, GainSolveError> {
108        Ok(SelectedData {
109            antennas: self.antennas.into_iter().collect(),
110            fields: self.fields.into_iter().collect(),
111            intents: intents(ms)?,
112            observations: self.observations.into_iter().collect(),
113            scans: self.scans.into_iter().collect(),
114            spectral_windows: self.windows.into_iter().collect(),
115        })
116    }
117}

119/// Every intent `ms`'s `STATE` table names, sorted and distinct; none when
120/// the MeasurementSet has no `STATE` table.
121fn intents(ms: &MeasurementSet) -> Result<Vec<String>, GainSolveError> {
122    let Ok(state) = ms.state() else {
123        return Ok(Vec::new());
124    };
125    let failed = |source| GainSolveError::OpenMeasurementSet {
126        path: ms
127            .path()
128            .map(|path| path.display().to_string())
129            .unwrap_or_else(|| "<in-memory>".to_string()),
130        source,
131    };
132    let modes = (0..state.row_count())
133        .map(|row| state.string(row, "OBS_MODE").map_err(failed))
134        .collect::<Result<Vec<_>, _>>()?;
135    Ok(intents_of(modes.iter().map(String::as_str)))
136}
137
138/// The intents `modes` name, as CASA 6.7.6.14 lists them: each mode split
139/// at its commas, untrimmed, sorted and distinct. An empty mode names none,
140/// though an empty item between two commas is kept.
141fn intents_of<'a>(modes: impl IntoIterator<Item = &'a str>) -> Vec<String> {
142    modes
143        .into_iter()
144        .filter(|mode| !mode.is_empty())
145        .flat_map(|mode| mode.split(','))
146        .map(str::to_string)
147        .collect::<BTreeSet<_>>()
148        .into_iter()
149        .collect()
150}

152#[cfg(test)]
153mod tests {
154    use super::intents_of;
155
156    #[test]
157    fn intents_are_casas_split_sorted_and_distinct() {
158        // Each case as CASA 6.7.6.14's gaincal lists it in `selectvis`.
159        for (modes, expected) in [
160            (&["B,A", "C", "B"][..], &["A", "B", "C"][..]),
161            (&[""], &[]),
162            (&["", "A"], &["A"]),
163            (&["A, B"], &[" B", "A"]),
164            (&["A,,B"], &["", "A", "B"]),
165        ] {
166            assert_eq!(intents_of(modes.iter().copied()), expected, "{modes:?}");
167        }
168    }
