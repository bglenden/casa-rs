10#[derive(Debug, Clone)]
11pub(super) struct IgrfTable {
12    years: Vec<f64>,
13    mjd0: f64,
14    step_days: f64,
15    coeffs_by_year: Vec<Vec<f64>>,
16    rates_by_year: Vec<Vec<f64>>,
17    nmax: usize,
18}
19
20impl IgrfTable {
21    pub(super) fn retained_heap_bytes(&self) -> Option<usize> {
22        let mut bytes = self
23            .years
24            .capacity()
25            .checked_mul(size_of::<f64>())?
26            .checked_add(
27                self.coeffs_by_year
28                    .capacity()
29                    .checked_mul(size_of::<Vec<f64>>())?,
30            )?
31            .checked_add(
32                self.rates_by_year
33                    .capacity()
34                    .checked_mul(size_of::<Vec<f64>>())?,
35            )?;
36        for coefficients in self.coeffs_by_year.iter().chain(&self.rates_by_year) {
37            bytes = bytes.checked_add(coefficients.capacity().checked_mul(size_of::<f64>())?)?;
38        }
39        Some(bytes)
40    }
41
42    pub(super) fn coefficients_at_mjd(
43        &self,
44        mjd: f64,
45    ) -> Result<(Vec<f64>, usize), MeasuresDataError> {
46        if !mjd.is_finite() || self.coeffs_by_year.is_empty() {
47            return Err(MeasuresDataError::MissingData(
48                "invalid IGRF model epoch or empty coefficient table".into(),
49            ));
50        }
51        // casacore uses a uniform five-year grid. MJD0 is one interval
52        // before the first row. The end rows supply extrapolation rates.
53        let row = (((mjd - self.mjd0) / self.step_days) as usize)
54            .saturating_sub(1)
55            .min(self.coeffs_by_year.len() - 1);
56        let row_mjd = self.mjd0 + (row + 1) as f64 * self.step_days;
57        let years = 5.0 * (mjd - row_mjd) / self.step_days;
58        let values = self.coeffs_by_year[row]
59            .iter()
60            .zip(&self.rates_by_year[row])
61            .map(|(base, rate)| base + rate * years)
62            .collect();
63        Ok((values, self.nmax))
64    }
65
66    pub(super) fn coefficients_for_decimal_year(
67        &self,
68        decimal_year: f64,
69    ) -> Result<(Vec<f64>, usize), MeasuresDataError> {
70        let min_year = *self.years.first().ok_or_else(|| {
71            MeasuresDataError::MissingData("IGRF table has no year rows".to_string())
72        })?;
73        let max_year = self.years.last().copied().unwrap_or(min_year) + 5.0;
74        if decimal_year < min_year || decimal_year > max_year {
75            return Err(MeasuresDataError::MissingData(format!(
76                "IGRF date must be between {min_year:.0}-01-01 and {max_year:.0}-12-31"
77            )));
78        }
79
80        let coeffs = if decimal_year >= *self.years.last().expect("years non-empty") {
81            extrapolate_coefficients(
82                decimal_year,
83                *self.years.last().expect("years non-empty"),
84                self.coeffs_by_year.last().expect("coefficients non-empty"),
85                self.rates_by_year.last().expect("rates non-empty"),
86            )
87        } else {
88            let upper = self
89                .years
90                .iter()
91                .position(|year| *year > decimal_year)
92                .expect("decimal year below final row has upper interval");
93            interpolate_coefficients(
94                decimal_year,
95                self.years[upper - 1],
96                &self.coeffs_by_year[upper - 1],
97                self.years[upper],
98                &self.coeffs_by_year[upper],
99            )
100        };
101
102        Ok((coeffs, self.nmax))
103    }
104}

106impl IgrfTable {
107    pub(super) fn hash_state(
108        &self,
109        state: &mut ScientificStateHasher,
110    ) -> Result<(), MeasuresDataError> {
111        state.sequence_len(self.years.len())?;
112        for year in &self.years {
113            state.float(*year);
114        }
115        state.sequence_len(self.coeffs_by_year.len())?;
116        for coefficients in &self.coeffs_by_year {
117            state.sequence_len(coefficients.len())?;
118            for coefficient in coefficients {
119                state.float(*coefficient);
120            }
121        }
122        state.float(self.mjd0);
123        state.float(self.step_days);
124        state.sequence_len(self.rates_by_year.len())?;
125        for rates in &self.rates_by_year {
126            state.sequence_len(rates.len())?;
127            for rate in rates {
128                state.float(*rate);
129            }
130        }
131        state.sequence_len(self.nmax)?;
132
133        Ok(())
134    }
135}

137pub(super) fn load_igrf_from(root: &Path) -> Result<IgrfTable, MeasuresDataError> {
138    let table = RuntimeTable::open(&root.join("geodetic/IGRF"))?;
139    let mjd = table.scalar_f64("MJD")?;
140    let mut years = Vec::with_capacity(table.row_count());
141    let mut coeffs_by_year = Vec::with_capacity(table.row_count());
142    let mut rates_by_year = Vec::with_capacity(table.row_count());
143    let keyword = |name| match table.table.keywords().get(name) {
144        Some(Value::Scalar(ScalarValue::Float64(value))) if value.is_finite() && *value > 0.0 => {
145            Ok(*value)
146        }
147        _ => Err(MeasuresDataError::MissingData(format!(
148            "invalid IGRF table keyword {name}"
149        ))),
150    };
151    let mjd0 = keyword("MJD0")?;
152    let step_days = keyword("dMJD")?;
153    let mut nmax = 0usize;
154
155    for (row, mjd_value) in mjd.iter().copied().enumerate().take(table.row_count()) {
156        years.push(decimal_year_from_mjd(mjd_value));
157        let coeffs = table.array_f64_cell("COEF", row)?;
158        nmax = solve_igrf_nmax(coeffs.len())?;
159        coeffs_by_year.push(coeffs);
160        let rates = table.array_f64_cell("dCOEF", row)?;
161        if rates.len() != coeffs_by_year[row].len() {
162            return Err(MeasuresDataError::MissingData(format!(
163                "wrong IGRF rate count at row {row}"
164            )));
165        }
166        rates_by_year.push(rates);
167    }
168
169    Ok(IgrfTable {
170        years,
171        coeffs_by_year,
172        mjd0,
173        step_days,
174        rates_by_year,
175        nmax,
176    })
177}

179fn decimal_year_from_mjd(mjd: f64) -> f64 {
180    let jd = mjd + 2_400_000.5;
181    let z = (jd + 0.5).floor();
182    let f = (jd + 0.5) - z;
183    let mut a = z;
184    if z >= 2_299_161.0 {
185        let alpha = ((z - 1_867_216.25) / 36_524.25).floor();
186        a = z + 1.0 + alpha - (alpha / 4.0).floor();
187    }
188    let b = a + 1524.0;
189    let c = ((b - 122.1) / 365.25).floor();
190    let d = (365.25 * c).floor();
191    let e = ((b - d) / 30.6001).floor();
192    let day = b - d - (30.6001 * e).floor() + f;
193    let month = if e < 14.0 { e - 1.0 } else { e - 13.0 };
194    let year = if month > 2.0 { c - 4716.0 } else { c - 4715.0 };
195
196    let year_i = year as i32;
197    let month_i = month as i32;
198    let day_i = day.floor() as i32;
199    let leap = if (year_i % 4 == 0) && ((year_i % 100 != 0) || (year_i % 400 == 0)) {
200        1
201    } else {
202        0
203    };
204    const DAYS: [i32; 12] = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334];
205    let mut day_in_year = DAYS[(month_i - 1) as usize] + day_i;
206    if month_i > 2 {
207        day_in_year += leap;
208    }
209    f64::from(year_i) + (f64::from(day_in_year) / (365.0 + f64::from(leap)))
210}

212fn solve_igrf_nmax(coeff_len: usize) -> Result<usize, MeasuresDataError> {
213    for nmax in 1..32 {
214        if nmax * (nmax + 2) == coeff_len {
215            return Ok(nmax);
216        }
217    }
218    Err(MeasuresDataError::MissingData(format!(
219        "unable to derive IGRF nmax from coefficient length {coeff_len}"
220    )))
221}

223fn interpolate_coefficients(
224    date: f64,
225    year1: f64,
226    coeffs1: &[f64],
227    year2: f64,
228    coeffs2: &[f64],
229) -> Vec<f64> {
230    let factor = (date - year1) / (year2 - year1);
231    coeffs1
232        .iter()
233        .zip(coeffs2.iter())
234        .map(|(lhs, rhs)| lhs + factor * (rhs - lhs))
235        .collect()
236}
237
238fn extrapolate_coefficients(date: f64, base_year: f64, main: &[f64], sv: &[f64]) -> Vec<f64> {
239    let factor = date - base_year;
240    main.iter()
241        .zip(sv.iter())
242        .map(|(main_coeff, sv_coeff)| main_coeff + factor * sv_coeff)
243        .collect()
244}
