54pub(super) fn read_locked<T>(
55    attachment: &AttachedImage,
56    read: impl FnOnce(&ImageHandle) -> PyResult<T>,
57) -> PyResult<T> {
58    let mut handle = attachment.lock();
59    attachment.require_pixel_read_lock(&mut handle)?;
60    let staged = read(&handle);
61    let release = attachment.release_requested_lock(&mut handle);
62    let result = staged?;
63    release?;
64    Ok(result)
65}

163    fn open_with_read_lock(
164        path: &Path,
165        probe: &Path,
166        name: &Path,
167        keep_read_lock: bool,
168        component_cache: bool,
169    ) -> PyResult<Self> {
170        if probe.is_file() {
171            let source =
172                FitsToolImport::open_attachment_named(path, probe, name).map_err(raised)?;
173            return ImageHandle::open_fits(path, source)
174                .map(Self::temporary)
175                .map_err(raised);
176        }
177        let unrecognized = || {
178            raised(format!(
179                "Exception: Unable to open image {}.",
180                name.display()
181            ))
182        };
183        if path != probe && (!probe.is_dir() || !path.is_dir()) {
184            return Err(unrecognized());
185        }
186        if casa_images::concat_file::is_image_concat(probe) {
187            return super::concatenate::virtual_image::open(path);
188        }
189        if casa_images::expr_file::is_image_expr(probe) {
190            return super::saved_expression::open(path);
191        }
192        let key = key(path)?;
193        let has_table_control = key.join("table.dat").is_file();
194        let paged = has_table_control
195            && fs::read_to_string(key.join("table.info"))
196                .is_ok_and(|info| TableInfo::parse(&info).table_type == "Image");
197        if !paged && !casa_images::ComponentListImage::recognizes_native(&key) {
198            if ImageHandle::recognizes_miriad(probe, &key) {
199                return Self::open_miriad(path);
200            }
201            return Err(unrecognized());
202        }
203        let name = std::path::absolute(path).map_err(raised)?;
204        let shared = Self::shared(&registry(), &key, &name);
205        if let Some(shared) = shared {
206            if keep_read_lock {
207                shared.read_current_beams()?;
208            }
209            return Ok(shared);
210        }
211
212        // The data lock may wait for a different process. Keep the registry
213        // available to other image names while this open is in progress.
214        let mut opened = Self::open_native(
215            &name,
216            &key,
217            Some(0),
218            TableUseLock::acquire_waiting_readonly,
219            component_cache,
220        )?;
221        if !keep_read_lock {
222            opened.lock().unlock_storage()?;
223        }
224        let shared = {
225            let mut registry = registry();
226            if let Some(shared) = Self::shared(&registry, &key, &name) {
227                shared
228            } else {
229                opened.register(&mut registry);
230                return Ok(opened);
231            }
232        };
233        drop(opened);
234        if keep_read_lock {
235            shared.read_current_beams()?;
236        }
237        Ok(shared)
238    }

240    fn read_current_beams(&self) -> PyResult<()> {
241        let mut handle = self.lock();
242        self.require_lock(&mut handle, casa_tables::LockType::Read)?;
243        *self.beam_view() = BeamView::from_image(&handle);
244        Ok(())
245    }

257        let mut opened = Self::open_native(name, key, Some(1), TableUseLock::acquire, true)?;

281    fn shared(registry: &OpenImages, key: &Path, name: &Path) -> Option<Self> {
282        registry
283            .get(key)
284            .into_iter()
285            .flatten()
286            .filter_map(Weak::upgrade)
287            .find(|image| {
288                image.shareable
289                    && image
290                        .use_lock
291                        .as_ref()
292                        .is_none_or(|lease| lease.matches_directory(key))
293            })
294            .map(|image| Self {
295                image,
296                key: Some(key.to_path_buf()),
297                native_name: Some(name.to_path_buf()),
298                registered: true,
299                beam_view: Mutex::default(),
300                retained_inputs: Vec::new(),
301                operand_paths: Vec::new(),
302                live_coordinates: None,
303            })
304    }

306    /// V1 Data callers get a fresh view on every open. The compatibility

308    pub(super) fn open_data(
309        registry: &mut OpenImages,
310        path: &Path,
311        writable: bool,
312    ) -> PyResult<Self> {
313        let key = key(path)?;
314        let acquire = if writable {
315            TableUseLock::acquire
316        } else {
317            TableUseLock::acquire_readonly
318        };
319        let mut opened = Self::open_native(&key, &key, None, acquire, true)?;
320        opened.register(registry);
321        Ok(opened)
322    }

324    fn open_native(
325        path: &Path,
326        key: &Path,
327        lock_attempts: Option<u32>,
328        acquire: fn(&Path) -> Result<TableUseLock, TableError>,
329        component_cache: bool,
330    ) -> PyResult<Self> {
331        let use_lock = key
332            .join("table.dat")
333            .is_file()
334            .then(|| acquire(key).map(Arc::new))
335            .transpose()
336            .map_err(raised)?;
337        let handle = ImageHandle::open_native_with_cache(key, lock_attempts, component_cache)
338            .map_err(raised)?;
339        let shareable = lock_attempts.is_some() && handle.image_type_name() != "ComponentListImage";
340        let beam_view = BeamView::from_image(&handle);
341        let expression_operands = handle.concat_expression_operands();
342        let image = Arc::new(SharedImage {
343            handle: Mutex::new(handle),
344            use_lock,
345            miriad_directory: None,
346            saved_expression_directory: None,
347            expression_operands,
348            shareable,
349            coordinates_replaced: AtomicBool::new(false),
350        });
351        let opened = Self {
352            image,
353            key: Some(key.to_path_buf()),
354            native_name: Some(std::path::absolute(path).map_err(raised)?),
355            registered: false,
356            beam_view: Mutex::new(beam_view),
357            retained_inputs: Vec::new(),
358            operand_paths: Vec::new(),
359            live_coordinates: None,
360        };
361        opened.check_bound_name(&opened.lock())?;
362        Ok(opened)
363    }

366        if let Some(key) = &self.key {
367            registry

383    pub(super) fn capture_live_coordinates(
384        image: &ImageHandle,
385    ) -> PyResult<Option<(RecordValue, CoordinateSystem)>> {
386        let coordinates = image.coordinates();
387        let has_samples =
388            (0..coordinates.n_coordinates()).any(|index| match coordinates.coordinate(index) {
389                CoordinateModel::Spectral(spectral) => !spectral.has_linear_mapping(),
390                CoordinateModel::Tabular(_) => true,
391                _ => false,
392            });
393        if !has_samples {
394            return Ok(None);
395        }
396        let record = image.coordinate_record();
397        let system = CoordinateSystem::from_live_record(&record).map_err(raised)?;
398        Ok(Some((record, system)))
399    }

411    pub(super) fn coordinate_view<'a>(
412        &'a self,
413        image: &'a ImageHandle,
414    ) -> (&'a CoordinateSystem, RecordValue) {
415        let record = image.coordinate_record();
416        let system = self
417            .live_coordinates
418            .as_ref()
419            .filter(|live| {
420                !self.image.coordinates_replaced.load(Ordering::Relaxed) && live.record == record
421            })
422            .map_or_else(|| image.coordinates(), |live| &live.system);
423        (system, record)
424    }

480    pub(super) fn check_directory(&self) -> PyResult<()> {
481        if let (Some(key), Some(lease)) = (&self.key, &self.image.use_lock)
482            && !lease.matches_directory(key)
483        {
484            return Err(raised(format!(
485                "The image at {} has changed",
486                key.display()
487            )));
488        }
489        Ok(())
490    }

492    pub(super) fn check_bound_name(&self, handle: &ImageHandle) -> PyResult<()> {
493        self.check_directory()?;
494        with_pixels!(handle, storage => {
495            if let Storage::Temp { image, .. } = storage
496                && let Some(source) = image.concat_source()
497            {
498                source.check_directory().map_err(raised)?;
499            }
500        });
501        if let (Some(key), Some(path)) = (&self.key, handle.path())
502            && path != key
503            && !path.canonicalize().is_ok_and(|actual| &actual == key)
504        {
505            return Err(raised(format!(
506                "The image at {} has changed",
507                path.display()
508            )));
509        }
510        Ok(())
511    }

513    pub(super) fn rename(
514        self,
515        name: &str,
516        expanded: PyResult<Cow<'_, str>>,
517        overwrite: bool,
518        history: bool,
519    ) -> PyResult<Self> {
520        if name.is_empty() {
521            return Err(raised("Exception: new file name must be specified."));
522        }
523        let Some(source) = &self.key else {
524            return self.rename_non_native(name, expanded, overwrite, history);
525        };
526        let source_changed = || {
527            raised(format!(
528                "Cannot rename image {} because the image at that path has changed",
529                source.display()
530            ))
531        };
532        let (old_name, virtual_concat) = {
533            let image = self.lock();
534            self.check_bound_name(&image)
535                .map_err(|_| source_changed())?;
536            (
537                self.name_path(&image).unwrap_or(source).to_path_buf(),
538                image.image_type_name() == "ImageConcat",
539            )
540        };
541        let target = native_destination(expanded?.as_ref(), name)?;
542        let mut registry = registry();
543        let target_metadata = match target.symlink_metadata() {
544            Ok(metadata) if overwrite => Some(metadata),
545            Ok(_) => {
546                return Err(raised(format!(
547                    "Exception: File {name} exists, and the user does not want to remove it.."
548                )));
549            }
550            Err(error) if error.kind() == std::io::ErrorKind::NotFound => None,
551            Err(error) => return Err(raised(error)),
552        };
553        let log_rename = |image: &mut ImageHandle| {
554            if history {
555                image.append_history_at(
556                    "::image::rename",
557                    &[
558                        "Ran ia.rename".into(),
559                        format!("ia.rename(name={name:?}, overwrite={overwrite})"),
560                        format!("original name was {}", old_name.display()),
561                    ],
562                )?;
563            }
564            Ok::<(), PyErr>(())
565        };
566        if std::ffi::OsStr::new(name) == old_name.as_os_str() {
567            return Err(raised(
568                "Exception: Specified output name is the same as the current image name.",
569            ));
570        }
571        if source == &target {
572            if self
573                .image
574                .use_lock
575                .as_ref()
576                .is_some_and(|lease| !lease.matches_directory(source))
577            {
578                return Err(source_changed());
579            }
580            log_rename(&mut self.lock())?;
581            return Ok(Self {
582                image: Arc::clone(&self.image),
583                key: Some(source.clone()),
584                native_name: Some(target.clone()),
585                registered: self.registered,
586                beam_view: Mutex::new(self.beam_view().clone()),
587                retained_inputs: Vec::new(),
588                operand_paths: Vec::new(),
589                live_coordinates: self.live_coordinates.clone(),
590            });
591        }
592        check_path_relationship(source, &target, name)?;
593        if virtual_concat {
594            super::concatenate::virtual_image::check_relocation(source, &target)?;
595        }
596        let replaces_link = target_metadata
597            .as_ref()
598            .is_some_and(|entry| entry.is_symlink());
599        if replaces_link && check_write_access(&target).is_err() {
600            return Err(raised(format!(
601                "RegularFile::copy/move: target file {} already exists and is not writable",
602                target.display()
603            )));
604        }
605        for (path, expected) in [(source.as_path(), 1), (target.as_path(), 0)] {
606            let in_table_use = if replaces_link && path == target {
607                // The link itself is replaced. Its underlying table and
608                // any tools attached to that table retain their storage.
609                containing_tables(path).any(crate::tools::table::path_is_in_use)
610            } else {
611                crate::tools::table::path_is_in_use(path)
612            };
613            if conflicting_images(&registry, path, expected)
614                || in_table_use
615                || (path == target && !replaces_link && in_use_by_expression(&registry, path))
616            {
617                return Err(raised(format!(
618                    "Cannot rename image because {} is still open in this process",
619                    path.display()
620                )));
621            }
622        }
623        let mut locks = BTreeMap::new();
624        {
625            let mut image = self.lock();
626            self.check_bound_name(&image)
627                .map_err(|_| source_changed())?;
628            image.unlock_storage()?;
629        }
630        lock_table_tree(source, &mut locks)?;
631        if target_metadata.as_ref().is_some_and(|entry| entry.is_dir()) {
632            lock_table_tree(&target, &mut locks)?;
633        } else if replaces_link {
634            lock_table_ancestors(&target, &mut locks)?;
635        }
636        let source_metadata = source.symlink_metadata().map_err(raised)?;
637        let concat_matches = with_pixels!(&*self.lock(), storage => match storage {
638            Storage::Temp { image, .. } => image.concat_source().is_none_or(|source|
639                source.matches_directory_metadata(&source_metadata)),
640            Storage::Paged(_) => true,
641        });
642        if !concat_matches
643            || self
644                .image
645                .use_lock
646                .as_ref()
647                .is_some_and(|lease| !lease.matches_directory_metadata(&source_metadata))
648        {
649            return Err(source_changed());
650        }
651        let open = |path: &Path| {
652            ImageHandle::open_native(path, self.image.shareable.then_some(1)).map_err(raised)
653        };
654        let moved = move_attached_directory(
655            virtual_concat,
656            source,
657            &source_metadata,
658            &target,
659            target_metadata.as_ref(),
660            |copied| {
661                if history {
662                    // Write only to the private copy. A competing rename of
663                    // the public destination must not redirect this history.
664                    log_rename(&mut open(copied)?)?;
665                }
666                Ok(())
667            },
668        )?;
669        let copied_lock = moved.use_lock();
670        // Keep both root locks during reattachment, but release child
671        // locks so the log-table writer can append the rename history.
672        locks.retain(|path, _| path == source || path == &target);
673        let reopen = || {
674            moved.check_target(&target)?;
675            let mut image = open(&target)?;
676            moved.check_target(&target)?;
677            if copied_lock.is_none() {
678                log_rename(&mut image)?;
679            }
680            Ok::<_, PyErr>(image)
681        };
682        let handle = match reopen() {
683            Ok(image) => image,
684            Err(error) => return moved.restore(source, &target, error),
685        };
686        moved.finish(&target)?;
687        let beam_view = BeamView::from_image(&handle);
688        let expression_operands = handle.concat_expression_operands();
689        let image = Arc::new(SharedImage {
690            handle: Mutex::new(handle),
691            use_lock: copied_lock.or_else(|| self.image.use_lock.clone()),
692            miriad_directory: None,
693            saved_expression_directory: None,
694            expression_operands,
695            shareable: self.image.shareable,
696            coordinates_replaced: AtomicBool::new(
697                self.image.coordinates_replaced.load(Ordering::Relaxed),
698            ),
699        });
700        registry.remove(source);
701        registry.insert(target.clone(), vec![Arc::downgrade(&image)]);
702        Ok(Self {
703            image,
704            key: Some(target.clone()),
705            native_name: Some(target),
706            registered: true,
707            beam_view: Mutex::new(beam_view),
708            retained_inputs: Vec::new(),
709            operand_paths: Vec::new(),
710            live_coordinates: self.live_coordinates.clone(),
711        })
712    }
713
714    fn rename_non_native(
715        mut self,
716        name: &str,
717        expanded: PyResult<Cow<'_, str>>,
718        overwrite: bool,
719        history: bool,
720    ) -> PyResult<Self> {
721        if let Some(directory) = &self.image.saved_expression_directory {
722            let mut relocated =
723                directory.rename(&self, name, expanded?.as_ref(), overwrite, history)?;
724            relocated.retain_expression_inputs(
725                std::mem::take(&mut self.retained_inputs),
726                std::mem::take(&mut self.operand_paths),
727            );
728            return Ok(relocated);
729        }
730        if let Some(directory) = &self.image.miriad_directory {
731            let old_name =
732                self.lock().path().map(Path::to_path_buf).ok_or_else(|| {
733                    raised("Exception: The image to be renamed no longer exists.")
734                })?;
735            return directory.rename(&old_name, name, expanded?.as_ref(), overwrite, history);
736        }
737        let fits_path = {
738            let image = self.lock();
739            image
740                .fits()
741                .and_then(|_| image.path())
742                .map(Path::to_path_buf)
743        };
744        if let Some(old_name) = fits_path {
745            return super::fits_rename::rename(
746                &old_name,
747                name,
748                expanded?.as_ref(),
749                overwrite,
750                history,
751            );
752        }
753        Err(raised(
754            "Exception: This image tool is not associated with a persistent disk file. It cannot be renamed.",
755        ))
756    }
757
758    pub(super) fn remove(&self) -> PyResult<String> {
759        {
760            let image = self.lock();
761            if image.image_type_name() == "ImageConcat" {
762                let name = self
763                    .name_path(&image)
764                    .expect("persistent concatenation name");
765                return Err(raised(format!(
766                    "Cannot delete file {} because table is not writable",
767                    name.display()
768                )));
769            }
770        }
771        let Some(key) = &self.key else {
772            if let Some(path) = self.lock().path() {
773                return Err(raised(format!(
774                    "Cannot delete file {} because table is not writable",
775                    path.display()
776                )));
777            }
778            return Err(raised(
779                "Exception: This image tool is not associated with a persistent disk file. It cannot be deleted.",
780            ));
781        };
782        let name = self.name_path(&self.lock()).unwrap_or(key).to_path_buf();
783        let mut registry = registry();
784        if conflicting_images(&registry, key, 1) || in_use_by_expression(&registry, key) {
785            return Err(in_use(key, "this"));
786        }
787        {
788            let mut image = self.lock();
789            self.check_bound_name(&image).map_err(|_| {
790                raised(format!(
791                    "Cannot delete file {} because the image at that path has changed",
792                    key.display()
793                ))
794            })?;
795            image.unlock_storage()?;
796        }
797        let entry = if name.canonicalize().is_ok_and(|actual| actual == *key) {
798            &name
799        } else {
800            key
801        };
802        remove_native_table(entry, key, self.image.use_lock.as_deref())?;
803        registry.remove(key);
804        Ok(name.to_string_lossy().into_owned())
805    }
806}

808fn move_attached_directory(
809    virtual_concat: bool,
810    source: &Path,
811    identity: &fs::Metadata,
812    target: &Path,
813    previous: Option<&fs::Metadata>,
814    prepare_copy: impl FnOnce(&Path) -> PyResult<()>,
815) -> PyResult<NativeTableMove> {
816    if virtual_concat {
817        // The virtual root is a manifest directory. Its history is rebuilt
818        // on attachment, and it has no native table lock file.
819        move_read_only_directory(source, identity, target, previous, |_| Ok(()))
820    } else {
821        move_native_table(source, identity, target, previous, prepare_copy)
822    }
823}

825/// Deletes an unopened table by name. The image tool also accepts a
826/// non-image casacore table here, as CASA's `removefile` does.
827pub(super) fn remove_file(path: &Path) -> PyResult<bool> {
828    if !path.is_dir() {
829        return Ok(false);
830    }
831    let name = std::path::absolute(path).map_err(raised)?;
832    let key = key(&name)?;
833    let registry = registry();
834    if conflicting_images(&registry, &key, 0) || in_use_by_expression(&registry, &key) {
835        return Err(in_use(&key, "this"));
836    }
837    remove_native_table(&name, &key, None)?;
838    Ok(true)
839}
840
841pub(super) fn conflicting_images(
842    registry: &OpenImages,
843    path: &Path,
844    expected_at_path: usize,
845) -> bool {
846    registry.iter().any(|(open_path, images)| {
847        if open_path == path {
848            images.iter().map(Weak::strong_count).sum::<usize>() > expected_at_path
849        } else {
850            (open_path.starts_with(path) || path.starts_with(open_path))
851                && images.iter().any(|image| image.strong_count() > 0)
852        }
853    })
854}
855
856pub(super) fn in_use_by_expression(registry: &OpenImages, path: &Path) -> bool {
857    registry
858        .values()
859        .flatten()
860        .filter_map(Weak::upgrade)
861        .any(|image| {
862            image
863                .expression_operands
864                .iter()
865                .any(|operand| path.starts_with(operand) || operand.starts_with(path))
866        })
867}

869fn in_use(path: &Path, process: &str) -> PyErr {
870    raised(format!(
871        "Cannot delete file {} because table is still open in {process} process",
872        path.display()
873    ))
874}
875
876fn remove_native_table(name: &Path, path: &Path, expected: Option<&TableUseLock>) -> PyResult<()> {
877    Table::open_metadata_only(TableOptions::new(path)).map_err(raised)?;
878    if crate::tools::table::path_is_in_use(path) {
879        return Err(in_use(path, "this"));
880    }
881    let mut leases = BTreeMap::new();
882    lock_table_tree(path, &mut leases)?;
883    if expected.is_some_and(|guard| !guard.matches_directory(path)) {
884        return Err(raised(format!(
885            "Cannot delete file {} because the image at that path has changed",
886            path.display()
887        )));
888    }
889    let named_link = name
890        .symlink_metadata()
891        .is_ok_and(|entry| entry.is_symlink());
892    if named_link
893        && let Some(parent) = name.parent()
894        && fs::metadata(parent)
895            .map_err(raised)?
896            .permissions()
897            .readonly()
898    {
899        return Err(raised(format!(
900            "Cannot delete file {} because its directory is not writable",
901            name.display()
902        )));
903    }
904    fs::remove_dir_all(path).map_err(raised)?;
905    if named_link {
906        // The table has been removed even if the alias's permissions changed
907        // after the preflight check.
908        let _ = fs::remove_file(name);
909    }
910    Ok(())
911}

927pub(super) fn lock_table_ancestors(
928    path: &Path,
929    leases: &mut BTreeMap<PathBuf, TableWriteLock>,
930) -> PyResult<()> {
931    let parent = path.parent().ok_or_else(|| not_writable(path))?;
932    if fs::metadata(parent)
933        .map_err(raised)?
934        .permissions()
935        .readonly()
936    {
937        return Err(not_writable(path));
938    }
939    for ancestor in containing_tables(path) {
940        lock_table_for_deletion(ancestor, leases)?;
941    }
942    Ok(())
943}

945pub(super) fn lock_table_tree(
946    path: &Path,
947    leases: &mut BTreeMap<PathBuf, TableWriteLock>,
948) -> PyResult<()> {
949    lock_tables_in_tree(path, leases, true)

955) -> PyResult<()> {

958
959fn lock_tables_in_tree(
960    path: &Path,
961    leases: &mut BTreeMap<PathBuf, TableWriteLock>,
962    require_writable_directories: bool,
963) -> PyResult<()> {
964    lock_table_ancestors(path, leases)?;
965    let mut pending = vec![path.to_path_buf()];
966    while let Some(directory) = pending.pop() {
967        if require_writable_directories
968            && fs::metadata(&directory)
969                .map_err(raised)?
970                .permissions()
971                .readonly()
972        {
973            return Err(not_writable(&directory));
974        }
975        let control = directory.join("table.dat");
976        if ImageHandle::has_native_control(&directory) {
977            OpenOptions::new()
978                .write(true)
979                .open(control)
980                .map_err(|_| not_writable(&directory))?;
981            lock_table_for_deletion(&directory, leases)?;
982        }
983        for entry in fs::read_dir(&directory).map_err(raised)? {
984            let entry = entry.map_err(raised)?;
985            if entry.file_type().map_err(raised)?.is_dir() {
986                pending.push(entry.path());
987            }
988        }
989    }
990    Ok(())
991}
992
993fn lock_table_for_deletion(
994    directory: &Path,
995    leases: &mut BTreeMap<PathBuf, TableWriteLock>,
996) -> PyResult<()> {
997    if leases.contains_key(directory) {
998        return Ok(());
999    }
1000    let lease = TableWriteLock::acquire(directory, 1).map_err(|error| match error {
1001        TableError::LockFailed { .. } => in_use(directory, "another"),
1002        _ => raised(error),
1003    })?;
1004    if lease.is_multi_used() {
1005        return Err(in_use(directory, "another"));
1006    }
1007    leases.insert(directory.to_path_buf(), lease);
1008    Ok(())
1009}

1011impl Drop for AttachedImage {
1012    fn drop(&mut self) {
1013        // A failed open can drop the candidate while publication holds the
1014        // registry lock. The candidate has no registry entry to remove.
1015        if self.registered
1016            && let Some(key) = &self.key
1017        {
1018            let mut registry = registry();
1019            if Arc::strong_count(&self.image) == 1
1020                && let Some(images) = registry.get_mut(key)
1021            {
1022                let current = Arc::downgrade(&self.image);
1023                images.retain(|image| !image.ptr_eq(&current) && image.strong_count() > 0);
1024                if images.is_empty() {
1025                    registry.remove(key);
1026                }
1027            }
1028        }
1029    }
1030}
