14const REQUIRED_COLUMNS: [&str; 5] = ["MJD", "RA", "DEC", "Rho", "RadVel"];
15
16/// A read-only comet ephemeris. It reads positions on demand and reuses the
17/// two consecutive rows that bracket the last observation.
18#[derive(Debug)]
19pub struct CometTable {
20    table: RuntimeTable,
21    name: String,
22    origin_mjd: f64,
23    step_days: f64,
24    topocentre_m: Option<[f64; 3]>,
25    table_reference: DirectionRef,
26    input_reference: DirectionRef,
27    mean_radius_au: Option<f64>,
28    au_m: f64,
29    zero_range_au: f64,
30    bracket: Mutex<Option<Bracket>>,
31}
32
33#[derive(Debug)]
34struct Bracket {
35    index: usize,
36    start_mjd: f64,
37    start_au: [f64; 3],
38    end_au: [f64; 3],
39}
40
41/// A direction and distance interpolated on the native comet table grid.
42#[derive(Debug)]
43pub struct CometDirection {
44    /// The interpolation expressed in the table's conversion axes.
45    pub direction: MDirection,
46    /// The length of the interpolated position, in metres.
47    pub distance_m: f64,
48    /// Casacore first removes observer parallax unless the table is APP.
49    pub remove_parallax: bool,
50}
51
52impl CometTable {
53    /// Open the requested casacore table and validate its fixed-grid metadata.
54    pub fn open(path: &Path) -> Result<Self, MeasuresDataError> {
55        let table = RuntimeTable::open(path)?;
56        let invalid = |reason| {
57            MeasuresDataError::TableRead(format!(
58                "invalid comet table {}: {reason}",
59                path.display()
60            ))
61        };
62        let keywords = table.table.keywords();
63        let version_date_valid = match keywords.get("VS_DATE") {
64            Some(Value::Scalar(ScalarValue::String(value))) => {
65                Quantity::read(value).is_ok_and(|date| {
66                    let unit = date.unit().val();
67                    [UnitVal::NODIM, UnitVal::TIME, UnitVal::ANGLE]
68                        .iter()
69                        .any(|valid| unit.conformant(valid))
70                })
71            }
72            _ => false,
73        };
74        if table.table.info().table_type != "IERS"
75            || !version_date_valid
76            || !matches!(
77                keywords.get("VS_VERSION"),
78                Some(Value::Scalar(ScalarValue::String(_)))
79            )
80            || keywords.get("VS_CREATE").is_none()
81            || keywords.get("VS_TYPE").is_none()
82        {
83            return Err(invalid("invalid IERS table metadata"));
84        }
85        let column = |name| table.table.schema().and_then(|schema| schema.column(name));
86        let scalar_double = |column: &casa_tables::ColumnSchema| {
87            column.column_type() == &ColumnType::Scalar
88                && column.data_type() == Some(PrimitiveType::Float64)
89        };
90        if REQUIRED_COLUMNS
91            .iter()
92            .any(|name| !column(name).is_some_and(scalar_double))
93            || ["DiskLong", "DiskLat"]
94                .iter()
95                .any(|name| column(name).is_some_and(|col| !scalar_double(col)))
96        {
97            return Err(invalid("expected double comet columns"));
98        }
99        let number = |name| keyword_double(keywords.get(name)).ok_or_else(|| invalid(name));
100        let name = match keywords.get("NAME") {
101            Some(Value::Scalar(ScalarValue::String(name))) => name.clone(),
102            _ => return Err(invalid("NAME")),
103        };
104        let origin_mjd = number("MJD0")?;
105        let step_days = number("dMJD")?;
106        let count = table.row_count();
107        if origin_mjd < 10_000.0 || step_days <= 0.0 || count == 0 {
108            return Err(invalid("invalid MJD sampling grid"));
109        }
110        let last_mjd = scalar(&table, "MJD", count - 1)?;
111        if !last_mjd.is_finite()
112            || (last_mjd - (origin_mjd + count as f64 * step_days)).abs() > step_days * 0.1
113        {
114            return Err(invalid("invalid MJD sampling grid"));
115        }
116
117        let distance_km = number("GeoDist")?;
118        let longitude = number("GeoLong")?.to_radians();
119        let latitude = number("GeoLat")?.to_radians();
120        let explicit = match keywords.get("posrefsys") {
121            Some(Value::Scalar(ScalarValue::String(reference))) => Some(comet_reference(reference)),
122            None => None,
123            _ => return Err(invalid("posrefsys")),
124        };
125        let table_reference = explicit.unwrap_or(if distance_km != 0.0 {
126            DirectionRef::TOPO
127        } else {
128            DirectionRef::APP
129        });
130        let topocentre_m = (table_reference == DirectionRef::TOPO).then(|| {
131            cartesian(
132                if distance_km == 0.0 {
133                    1e-6
134                } else {
135                    distance_km * 1e3
136                },
137                longitude,
138                latitude,
139            )
140        });
141
142        let au = Unit::new("AU").map_err(|_| invalid("invalid AU unit"))?;
143        let mean_radius_au = match keywords.get("meanrad") {
144            Some(Value::Record(record)) => {
145                match (keyword_double(record.get("value")), record.get("unit")) {
146                    (Some(value), Some(Value::Scalar(ScalarValue::String(unit)))) => {
147                        Quantity::new(value, unit)
148                            .ok()
149                            .and_then(|radius| casacore::convert(&radius, Some(&au)).ok())
150                            .map(|radius| radius.value())
151                    }
152                    _ => None,
153                }
154            }
155            _ => None,
156        };
157        Ok(Self {
158            table,
159            name,
160            origin_mjd,
161            step_days,
162            topocentre_m,
163            table_reference,
164            input_reference: explicit.unwrap_or(DirectionRef::APP),
165            mean_radius_au,
166            au_m: au.val().factor,
167            zero_range_au: 1e-6 / au.val().factor,
168            bracket: Mutex::new(None),
169        })
170    }
171
172    /// The object's name from the table keyword.
173    pub fn name(&self) -> &str {
174        &self.name
175    }
176
177    /// The first and last sample dates on the table's native MJD grid.
178    pub fn span_mjd(&self) -> [f64; 2] {
179        [
180            self.origin_mjd + self.step_days,
181            self.origin_mjd + self.table.row_count() as f64 * self.step_days,
182        ]
183    }
184
185    /// Cartesian topocentre in metres, when the ephemeris uses TOPO.
186    pub fn topocentre_m(&self) -> Option<[f64; 3]> {
187        self.topocentre_m
188    }
189
190    /// Mean physical radius, when the table provides it.
191    pub fn mean_radius_au(&self) -> Option<f64> {
192        self.mean_radius_au
193    }
194
195    /// Interpolate the geocentric or topocentric position on the table's
196    /// native MJD grid and return its norm in AU. The last row is the upper
197    /// endpoint of the last bracket and is excluded as an observation date.
198    pub fn distance_au(&self, mjd: f64) -> Result<Option<f64>, MeasuresDataError> {
199        self.position_au(mjd).map(|position| position.map(norm))
200    }
201
202    /// Interpolate the comet at the epoch expressed as TDB MJD. Casacore
203    /// uses this value directly as the coordinate on the table's native grid.
204    pub fn direction_at(&self, mjd: f64) -> Result<Option<CometDirection>, MeasuresDataError> {
205        let Some(xyz) = self.position_au(mjd)? else {
206            return Ok(None);
207        };
208        Ok(Some(CometDirection {
209            direction: MDirection::from_cosines(xyz, self.input_reference),
210            distance_m: norm(xyz) * self.au_m,
211            remove_parallax: self.table_reference != DirectionRef::APP,
212        }))
213    }
214
215    /// Interpolate the radial velocity in metres per second on the native
216    /// comet table grid. Returns `None` outside the interpolation span.
217    ///
218    /// # Errors
219    ///
220    /// Returns a data error when the bracketing rows cannot be read.
221    pub fn radial_velocity_m_s(&self, mjd: f64) -> Result<Option<f64>, MeasuresDataError> {
222        self.within_bracket(mjd, |bracket, fraction| {
223            let first = scalar(&self.table, "RadVel", bracket.index)?;
224            let last = scalar(&self.table, "RadVel", bracket.index + 1)?;
225            Ok((first + fraction * (last - first)) * self.au_m / 86_400.)
226        })?
227        .transpose()
228    }
229
230    fn position_au(&self, mjd: f64) -> Result<Option<[f64; 3]>, MeasuresDataError> {
231        self.within_bracket(mjd, |bracket, fraction| {
232            std::array::from_fn(|i| {
233                bracket.start_au[i] + fraction * (bracket.end_au[i] - bracket.start_au[i])
234            })
235        })
236    }
237
238    fn within_bracket<T>(
239        &self,
240        mjd: f64,
241        sample: impl FnOnce(&Bracket, f64) -> T,
242    ) -> Result<Option<T>, MeasuresDataError> {
243        let index = ((mjd - self.origin_mjd) / self.step_days).floor() - 1.0;
244        if !(0.0..(self.table.row_count() - 1) as f64).contains(&index) {
245            return Ok(None);
246        }
247        let index = index as usize;
248        let mut cache = self.bracket.lock().unwrap_or_else(PoisonError::into_inner);
249        if cache.as_ref().is_none_or(|bracket| bracket.index != index) {
250            *cache = Some(Bracket {
251                index,
252                start_mjd: scalar(&self.table, "MJD", index)?,
253                start_au: position(&self.table, index, self.zero_range_au)?,
254                end_au: position(&self.table, index + 1, self.zero_range_au)?,
255            });
256        }
257        let bracket = cache.as_ref().expect("comet bracket is loaded");
258        let fraction = (mjd - bracket.start_mjd) / self.step_days;
259        Ok(Some(sample(bracket, fraction)))
260    }
261}
262
263fn comet_reference(reference: &str) -> DirectionRef {
264    let reference = reference.to_ascii_uppercase();
265    [
266        DirectionRef::J2000,
267        DirectionRef::B1950,
268        DirectionRef::APP,
269        DirectionRef::ICRS,
270        DirectionRef::TOPO,
271    ]
272    .into_iter()
273    .find(|code| reference.contains(code.as_str()))
274    .unwrap_or(DirectionRef::APP)
275}
276
277fn norm(xyz: [f64; 3]) -> f64 {
278    xyz.iter().map(|part| part * part).sum::<f64>().sqrt()
279}
280
281// RecordInterface::asDouble promotes integer and float scalar keywords.
282fn keyword_double(value: Option<&Value>) -> Option<f64> {
283    let number = match value? {
284        Value::Scalar(ScalarValue::UInt8(value)) => f64::from(*value),
285        Value::Scalar(ScalarValue::Int16(value)) => f64::from(*value),
286        Value::Scalar(ScalarValue::Int32(value)) => f64::from(*value),
287        Value::Scalar(ScalarValue::UInt32(value)) => f64::from(*value),
288        Value::Scalar(ScalarValue::Int64(value)) => *value as f64,
289        Value::Scalar(ScalarValue::Float32(value)) => f64::from(*value),
290        Value::Scalar(ScalarValue::Float64(value)) => *value,
291        _ => return None,
292    };
293    number.is_finite().then_some(number)
294}

296fn scalar(table: &RuntimeTable, name: &str, row: usize) -> Result<f64, MeasuresDataError> {
297    match table.table.column_accessor(name)?.scalar_cell(row)? {
298        ScalarValue::Float64(value) => Ok(*value),
299        _ => Err(MeasuresDataError::TableRead(format!(
300            "comet column {name} row {row} must contain a double"
301        ))),
302    }
303}
304
305fn position(
306    table: &RuntimeTable,
307    row: usize,
308    zero_range_au: f64,
309) -> Result<[f64; 3], MeasuresDataError> {
310    let range = scalar(table, "Rho", row)?;
311    // MVPosition substitutes a micrometre when readjusting a zero radius.
312    let distance = if range == 0.0 { zero_range_au } else { range };
313    let longitude = scalar(table, "RA", row)?.to_radians();
314    let latitude = scalar(table, "DEC", row)?.to_radians();
315    Ok(cartesian(distance, longitude, latitude))
316}
317
318fn cartesian(distance: f64, longitude: f64, latitude: f64) -> [f64; 3] {
319    let (sin_lon, cos_lon) = longitude.sin_cos();
320    let (sin_lat, cos_lat) = latitude.sin_cos();
321    [
322        distance * cos_lat * cos_lon,
323        distance * cos_lat * sin_lon,
324        distance * sin_lat,
325    ]
326}
