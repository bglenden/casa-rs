2//! Parallactic-angle rotation epochs for an imported aperture buffer.
3
4use std::sync::Arc;
5
6/// Feed angles at the selected observation times and the rotation threshold.
7#[derive(Clone, Debug, PartialEq)]
8pub struct AwPaRotation {
9    /// Minimum absolute change that triggers rotation, in radians.
10    pub step_rad: f64,
11    /// Taylor passes share one schedule only when the retained angle at
12    /// the end of the first pass yields the same effective angles on replay.
13    pub require_repeatable_schedule: bool,
14    /// The selected times in increasing order, seconds since MJD 0, with
15    /// the corresponding antenna-0 feed PA in Float radians.
16    pub samples: Arc<[(f64, f32)]>,
17}
18
19impl AwPaRotation {
20    /// The input is shared with the deployment and stays allocated.
21    pub(super) fn samples_bytes(&self) -> usize {
22        self.samples.len() * size_of::<(f64, f32)>() + 2 * size_of::<usize>()
23    }
24
25    /// In the single-buffer case, record each rotation epoch. The cached
26    /// angle is retained until the next sample differs strictly more than
27    /// the requested threshold (`AWConvFunc::prepareConvFunction`).
28    pub(super) fn schedule(&self, base_deg: f64) -> Option<Vec<(f64, f64)>> {
29        let base = f64::from(base_deg as f32).to_radians();
30        let mut last = base;
31        let mut rotations = Vec::new();
32        for &(time, pa) in self.samples.iter() {
33            let pa = f64::from(pa);
34            if (last - pa).abs() > self.step_rad {
35                rotations.push((time, base - pa));
36                last = pa;
37            }
38        }
39        if self.require_repeatable_schedule && !self.repeats_from(base, last) {
40            return None;
41        }
42        Some(rotations)
43    }
44
45    fn repeats_from(&self, mut fresh: f64, mut replay: f64) -> bool {
46        self.samples.iter().all(|&(_, pa)| {
47            let pa = f64::from(pa);
48            if (fresh - pa).abs() > self.step_rad {
49                fresh = pa;
50            }
51            if (replay - pa).abs() > self.step_rad {
52                replay = pa;
53            }
54            // The saved header has Float degrees, while feed PA is Float
55            // radians. Thresholds use their original radian values.
56            fresh as f32 == replay as f32
57        })
58    }
59}
60
61/// The inverse pixel transform for an axis-aligned linear CF plane.
62pub(super) fn pixel_rotation(angle_rad: f64) -> [[f64; 2]; 2] {
63    let (sin, cos) = angle_rad.sin_cos();
64    [[cos, -sin], [sin, cos]]
65}

67#[cfg(test)]
68mod tests {
69    use super::*;
70
71    #[test]
72    fn rotation_samples_the_inverse_linear_coordinate_transform() {
73        let [[xx, xy], [yx, yy]] = pixel_rotation(std::f64::consts::FRAC_PI_2);
74        assert!(xx.abs() < 1.0e-15 && yy.abs() < 1.0e-15);
75        assert_eq!([xy, yx], [-1.0, 1.0]);
76    }
77
78    #[test]
79    fn rotation_epochs_use_the_previous_angle_and_rotate_the_base() {
80        let rotation = AwPaRotation {
81            step_rad: 0.125,
82            require_repeatable_schedule: false,
83            samples: Arc::new([(1.0, 0.125), (2.0, 0.25), (3.0, 0.375), (4.0, 0.5)]),
84        };
85        assert_eq!(rotation.schedule(0.0).unwrap(), [(2.0, -0.25), (4.0, -0.5)]);
86    }
87
88    #[test]
89    fn shared_rotation_requires_the_same_angles_after_wraparound() {
90        let rotation = |first_deg: f32| AwPaRotation {
91            step_rad: 5.0_f64.to_radians(),
92            require_repeatable_schedule: true,
93            samples: Arc::new([
94                (1.0, first_deg.to_radians()),
95                (2.0, 5.5_f32.to_radians()),
96                (3.0, 12.0_f32.to_radians()),
97            ]),
98        };
99        // The second pass rotates from 12 degrees to 1, then stays at 1
100        // for the next sample. The first pass uses 0 and then 5.5.
101        assert!(rotation(1.0).schedule(0.0).is_none());
102        // With first PA equal to the saved PA, both passes use the same
103        // angles even though the first replay sample triggers a rotation.
104        assert_eq!(rotation(0.0).schedule(0.0).unwrap().len(), 2);
105    }
