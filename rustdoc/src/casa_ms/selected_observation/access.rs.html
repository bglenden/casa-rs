36/// A retained, read-locked MeasurementSet bound to one compiled source selection.
37///
38/// Construction opens the retained storage capability and metadata needed to plan the bounded
39/// content buffers. The sole MAIN traversal happens when samples are consumed: that pass both
40/// validates the compact compiler-owned manifest and produces the selected values.
41pub(crate) struct BoundObservationSource {
42    source_identity: casa_imaging_model::MeasurementSetIdentity,
43    measurement_set: MeasurementSet,
44    geometry_engine: MsCalEngine,
45    row_predicate: CompiledRowPredicate,
46    coordinates: Box<[SelectedCoordinates]>,
47    content_plan: SelectedObservationContentPlan,
48    source_row_count_matches: bool,
49}

64    /// Open the source locator under retained read locks without traversing MAIN rows.
65    #[cfg(unix)]
66    pub(crate) fn open_with_measures(
67        problem: &CompiledProblem,
68        source: &ObservationSource,
69        current_state: &ObservationSourceState,
70        measures: &SelectedObservationMeasures,
71        shared_bytes: SelectedObservationSharedBytes,
72        content_budget: SelectedObservationContentBudget,
73    ) -> Result<Self, BoundObservationSourceError> {
74        measures.validate_problem(problem)?;
75        let measurement_set = MeasurementSet::open_retained_read(source.provenance().locator())?;
76        validate_current_state(source, current_state)?;
77        let content_plan = selected_content_plan(
78            &measurement_set,
79            problem,
80            source,
81            shared_bytes,
82            content_budget,
83        )?;
84        let row_predicate = selected_row_predicate(&measurement_set, source)?;
85        let coordinates = selected_coordinates(&measurement_set, source.selection())?;
86        let data_description_count = measurement_set.data_description()?.row_count();
87        if row_predicate.requires_every_source_row(data_description_count)
88            && source.selection().rows().selected_row_count()
89                != u64::try_from(measurement_set.row_count())
90                    .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?
91        {
92            return Err(BoundObservationSourceError::IncompleteUnconditionalRowManifest);
93        }
94        let geometry_engine = MsCalEngine::new_selected_observation(
95            &measurement_set,
96            measures.provider(),
97            measures.provider_state(),
98        )?;
99        geometry_engine.verify_selected_observation_measures()?;
100        Ok(Self {
101            source_identity: source.identity(),
102            source_row_count_matches: usize::try_from(source.selection().rows().source_row_count())
103                .ok()
104                == Some(measurement_set.row_count()),
105            measurement_set,
106            geometry_engine,
107            row_predicate,
108            coordinates,
109            content_plan,
110        })
111    }

155    /// Stream the exact selected samples for this source under the retained read locks.
156    ///
157    /// Samples are emitted in canonical physical-row, channel, and correlation order. Physical
158    /// row blocking comes only from the admitted content plan and is absent from scientific identity.
159    pub(crate) fn selected_samples<'a>(
160        &'a self,
161        problem: &'a CompiledProblem,
162    ) -> Result<BoundObservationSamples<'a>, BoundObservationSourceError> {
163        self.geometry_engine
164            .verify_selected_observation_measures()?;
165        let expected = problem
166            .selected_observation()
167            .read_set()
168            .sources()
169            .iter()
170            .find(|candidate| candidate.measurement_set() == self.source_identity)
171            .ok_or(BoundObservationSourceError::ProblemSourceMismatch)?;
172        let rows = expected
173            .selection()
174            .rows()
175            .ordered_main_rows()
176            .iter()
177            .copied();
178        Ok(BoundObservationSamples {
179            source: self,
180            logical_source: expected,
181            problem,
182            rows,
183            pending_row: None,
184            active_block: None,
185            ready_blocks: VecDeque::with_capacity(self.content_plan.maximum_live_blocks()),
186            next_buffer_slot: 0,
187            source_exhausted: false,
188            terminal_manifest_checked: false,
189            pending_error: None,
190            live_block_high_water: 0,
191            row_offset: 0,
192            channel_ordinal: 0,
193            correlation_ordinal: 0,
194            finished: false,
195        })
196    }

199/// One fallible bounded stream of canonical samples from a retained source.
200pub(crate) struct BoundObservationSamples<'a> {
201    source: &'a BoundObservationSource,
202    logical_source: &'a MeasurementSetReadAccess,
203    problem: &'a CompiledProblem,
204    rows: std::iter::Copied<std::slice::Iter<'a, SelectedMainRow>>,
205    pending_row: Option<SelectedMainRow>,
206    active_block: Option<BufferedObservationBlock>,
207    ready_blocks: VecDeque<BufferedObservationBlock>,
208    next_buffer_slot: usize,
209    source_exhausted: bool,
210    terminal_manifest_checked: bool,
211    pending_error: Option<BoundObservationSourceError>,
212    live_block_high_water: usize,
213    row_offset: usize,
214    channel_ordinal: usize,
215    correlation_ordinal: usize,
216    finished: bool,
217}
218
219impl Iterator for BoundObservationSamples<'_> {
220    type Item = Result<SelectedObservationSample, BoundObservationSourceError>;
221
222    fn next(&mut self) -> Option<Self::Item> {
223        if self.finished {
224            return None;
225        }
226        loop {
227            if let Some(block) = &self.active_block {
228                let coordinate_index = block.coordinate_index;
229                let coordinates = &self.source.coordinates[coordinate_index];
230                if self.row_offset < block.buffer.row_count() {
231                    let channel_index = coordinates.channels[self.channel_ordinal].channel_index;
232                    let product = coordinates.products[self.correlation_ordinal];
233                    let channel_offset = usize::try_from(channel_index)
234                        .ok()
235                        .and_then(|channel| channel.checked_sub(coordinates.channel_start));
236                    let correlation_offset = usize::try_from(product.correlation_index()).ok();
237                    let stored = match (channel_offset, correlation_offset) {
238                        (Some(channel), Some(correlation)) => {
239                            block.buffer.sample(channel, self.row_offset, correlation)
240                        }
241                        _ => None,
242                    };
243                    let Some(stored) = stored else {
244                        self.finished = true;
245                        return Some(Err(BoundObservationSourceError::StoredSampleShapeMismatch));
246                    };
247                    let sample = self.project_sample(
248                        coordinates,
249                        coordinates.channels[self.channel_ordinal],
250                        product,
251                        stored,
252                        block.row_geometry[self.row_offset],
253                    );
254                    self.advance(coordinates.channels.len(), coordinates.products.len());
255                    if sample.is_err() {
256                        self.finished = true;
257                    }
258                    return Some(sample);
259                }
260            }
261            self.recycle_active_block();
262            self.prefetch_to_capacity();
263            if let Some(block) = self.ready_blocks.pop_front() {
264                self.active_block = Some(block);
265                self.row_offset = 0;
266                self.channel_ordinal = 0;
267                self.correlation_ordinal = 0;
268                self.record_live_block_high_water();
269                continue;
270            }
271            if let Some(error) = self.pending_error.take() {
272                self.finished = true;
273                return Some(Err(error));
274            }
275            if let Err(error) = self
276                .source
277                .geometry_engine
278                .verify_selected_observation_measures()
279            {
280                self.finished = true;
281                return Some(Err(error.into()));
282            }
283            self.finished = true;
284            return None;
285        }
286    }
287}

289impl BoundObservationSamples<'_> {
290    fn advance(&mut self, channel_count: usize, product_count: usize) {
291        self.correlation_ordinal += 1;
292        if self.correlation_ordinal == product_count {
293            self.correlation_ordinal = 0;
294            self.channel_ordinal += 1;
295            if self.channel_ordinal == channel_count {
296                self.channel_ordinal = 0;
297                self.row_offset += 1;
298            }
299        }
300    }
301
302    fn recycle_active_block(&mut self) {
303        let Some(mut block) = self.active_block.take() else {
304            return;
305        };
306        if self.source_exhausted || self.pending_error.is_some() {
307            return;
308        }
309        debug_assert!(block.slot < self.source.content_plan.maximum_live_blocks());
310        match self.fill_next_block(&mut block) {
311            Ok(true) => {
312                self.ready_blocks.push_back(block);
313                self.record_live_block_high_water();
314            }
315            Ok(false) => self.source_exhausted = true,
316            Err(error) => {
317                self.source_exhausted = true;
318                self.pending_error = Some(error);
319            }
320        }
321    }
322
323    fn prefetch_to_capacity(&mut self) {
324        let maximum_live_blocks = self.source.content_plan.maximum_live_blocks();
325        while !self.source_exhausted
326            && self.pending_error.is_none()
327            && self.active_block.is_some() as usize + self.ready_blocks.len() < maximum_live_blocks
328        {
329            debug_assert!(self.next_buffer_slot < maximum_live_blocks);
330            let mut block = BufferedObservationBlock::new(
331                self.next_buffer_slot,
332                self.source.content_plan.rows_per_block(),
333            );
334            self.next_buffer_slot += 1;
335            match self.fill_next_block(&mut block) {
336                Ok(true) => {
337                    self.ready_blocks.push_back(block);
338                    self.record_live_block_high_water();
339                }
340                Ok(false) => self.source_exhausted = true,
341                Err(error) => {
342                    self.source_exhausted = true;
343                    self.pending_error = Some(error);
344                }
345            }
346        }
347    }
348
349    fn record_live_block_high_water(&mut self) {
350        self.live_block_high_water = self
351            .live_block_high_water
352            .max(self.active_block.is_some() as usize + self.ready_blocks.len());
353    }
354
355    fn project_sample(
356        &self,
357        coordinates: &SelectedCoordinates,
358        channel: SelectedChannel,
359        product: CorrelationProduct,
360        stored: SelectedStoredSample,
361        geometry: EvaluatedRowGeometry,
362    ) -> Result<SelectedObservationSample, BoundObservationSourceError> {
363        let time_scale = time_scale(self.source.geometry_engine.time_reference().as_str())?;
364        let physical_row = u64::try_from(stored.physical_row())
365            .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?;
366        let visibility = match stored.visibility() {
367            SelectedStoredVisibility::Float32(value) => SelectedVisibilitySample::Float32(value),
368            SelectedStoredVisibility::Complex32(value) => {
369                SelectedVisibilitySample::Complex32(value)
370            }
371        };
372        let prediction_target = if self
373            .problem
374            .observation_transaction()
375            .write_set()
376            .model_columns()
377            .iter()
378            .any(|write| write.measurement_set() == self.logical_source.measurement_set())
379        {
380            SelectedPredictionTarget::ModelData
381        } else {
382            SelectedPredictionTarget::NotRequested
383        };
384        Ok(SelectedObservationSample {
385            address: SelectedSampleAddress {
386                measurement_set: self.logical_source.measurement_set(),
387                physical_row,
388                data_description_id: stored.data_description_id(),
389                spectral_window_id: coordinates.data_description.spectral_window_id(),
390                channel_index: channel.channel_index,
391                frequency_centre_hz: channel.centre_hz,
392                frequency_lower_hz: channel.lower_hz,
393                frequency_upper_hz: channel.upper_hz,
394                channel_width_hz: channel.width_hz,
395                frequency_frame: channel.frame,
396                polarization_id: coordinates.data_description.polarization_id(),
397                correlation_index: product.correlation_index(),
398                correlation_type: product.correlation_type(),
399            },
400            visibility,
401            prediction_target,
402            channel_flag: stored.channel_flag(),
403            row_flag: stored.row_flag(),
404            input_weight: stored.input_weight(),
405            coordinates: SelectedSampleCoordinates {
406                raw_uvw_m: stored.uvw_m(),
407                density_uvw_m: geometry.density_uvw_m,
408                transformed_uvw_m: geometry.transformed_uvw_m,
409                phase_shift_m: geometry.phase_shift_m,
410                uvw_law: self.problem.geometry().uvw(),
411                time: Epoch::new(stored.time_mjd_seconds() / 86_400.0, time_scale),
412                time_centroid: Epoch::new(
413                    stored.time_centroid_mjd_seconds() / 86_400.0,
414                    time_scale,
415                ),
416                interval_seconds: stored.interval_seconds(),
417                exposure_seconds: stored.exposure_seconds(),
418                phase_direction: geometry.phase_direction,
419                delay_direction: geometry.delay_direction,
420                pointing_directions: geometry.pointing_directions,
421            },
422            metadata: SelectedSampleMetadata {
423                field_id: stored.field_id(),
424                antenna1: stored.antenna1(),
425                antenna2: stored.antenna2(),
426                feed1: stored.feed1(),
427                feed2: stored.feed2(),
428                scan_number: stored.scan_number(),
429                state_id: stored.state_id(),
430                observation_id: stored.observation_id(),
431                array_id: stored.array_id(),
432            },
433        })
434    }

436    #[cfg(test)]
437    pub(super) fn scheduling_state(&self) -> (Option<usize>, Vec<(usize, usize)>, usize) {
438        (
439            self.active_block.as_ref().map(|block| block.slot),
440            self.ready_blocks
441                .iter()
442                .map(|block| {
443                    let physical_row = block
444                        .buffer
445                        .sample(0, 0, 0)
446                        .expect("ready selected-observation block contains one sample")
447                        .physical_row();
448                    (block.slot, physical_row)
449                })
450                .collect(),
451            self.live_block_high_water,
452        )
453    }
454
455    fn fill_next_block(
456        &mut self,
457        block: &mut BufferedObservationBlock,
458    ) -> Result<bool, BoundObservationSourceError> {
459        let first = match self.pending_row.take() {
460            Some(row) => row,
461            None => match self.rows.next() {
462                Some(row) => row,
463                None => {
464                    if !self.terminal_manifest_checked {
465                        self.terminal_manifest_checked = true;
466                        if !self.source.source_row_count_matches {
467                            return Err(BoundObservationSourceError::SourceRowCountMismatch);
468                        }
469                    }
470                    return Ok(false);
471                }
472            },
473        };
474        let coordinate_index = self
475            .source
476            .coordinates
477            .iter()
478            .position(|coordinates| {
479                coordinates.data_description.data_description_id() == first.data_description_id()
480            })
481            .ok_or(
482                BoundObservationSourceError::DataDescriptionCoordinateMismatch {
483                    data_description_id: first.data_description_id(),
484                },
485            )?;
486        let maximum_rows = self.source.content_plan.rows_per_block();
487        let mut physical_rows = Vec::with_capacity(maximum_rows);
488        physical_rows.push(
489            usize::try_from(first.physical_row())
490                .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
491        );
492        while physical_rows.len() < maximum_rows {
493            let Some(row) = self.rows.next() else {
494                break;
495            };
496            if row.data_description_id()
497                != self.source.coordinates[coordinate_index]
498                    .data_description
499                    .data_description_id()
500            {
501                self.pending_row = Some(row);
502                break;
503            }
504            physical_rows.push(
505                usize::try_from(row.physical_row())
506                    .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
507            );
508        }
509        let coordinates = &self.source.coordinates[coordinate_index];
510        self.source
511            .measurement_set
512            .fill_selected_observation_buffer(
513                &SelectedObservationBufferRequest::new(
514                    selected_visibility(self.logical_source.selected_columns().visibility()),
515                    selected_weight(self.logical_source.selected_columns().weights()),
516                    physical_rows,
517                    VisibilityChannelReadRange::new(
518                        coordinates.channel_start,
519                        coordinates.channel_count,
520                    ),
521                ),
522                &mut block.buffer,
523            )?;
524        for row in 0..block.buffer.row_count() {
525            let stored = block
526                .buffer
527                .sample(0, row, 0)
528                .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
529            if u32::try_from(stored.data_description_id()).ok()
530                != Some(coordinates.data_description.data_description_id())
531            {
532                return Err(
533                    BoundObservationSourceError::DataDescriptionCoordinateMismatch {
534                        data_description_id: coordinates.data_description.data_description_id(),
535                    },
536                );
537            }
538            if !self
539                .source
540                .row_predicate
541                .matches(StoredMainRow::from(stored))
542            {
543                return Err(BoundObservationSourceError::SelectedRowPredicateMismatch {
544                    physical_row: u64::try_from(stored.physical_row())
545                        .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?,
546                });
547            }
548        }
549        let observation_pointings = evaluate_observation_pointings(
550            self.source,
551            self.problem,
552            &block.buffer,
553            self.source.content_plan.rows_per_block(),
554            self.source.content_plan.maximum_pointing_polynomial_terms(),
555        )?;
556        block.row_geometry.clear();
557        for row in 0..block.buffer.row_count() {
558            let stored = block
559                .buffer
560                .sample(0, row, 0)
561                .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
562            block.row_geometry.push(evaluate_row_geometry(
563                self.source,
564                self.problem,
565                stored,
566                observation_pointings
567                    .as_ref()
568                    .map(|pointings| pointings[row]),
569            )?);
570        }
571        block.coordinate_index = coordinate_index;
572        Ok(true)
573    }
574}

594/// Failure to bind a retained MeasurementSet to one compiled observation source.
595#[derive(Debug, Error)]
596pub enum BoundObservationSourceError {
597    /// The injected Measures provider is missing, stale, or unaccounted.
598    #[error(transparent)]
599    Measures(#[from] SelectedObservationMeasuresError),
600    /// The MeasurementSet could not be opened or read under the admitted content budget.
601    #[error(transparent)]
602    Storage(#[from] MsError),
603    /// Stored DATA_DESCRIPTION metadata contradicted the compiler-owned coordinate catalog.
604    #[error(
605        "stored DATA_DESCRIPTION row {data_description_id} does not match the compiled coordinate catalog"
606    )]
607    DataDescriptionCoordinateMismatch {
608        /// DATA_DESCRIPTION row that differed.
609        data_description_id: u32,
610    },
611    /// A wavelength UV predicate lacked a positive finite reference wavelength.
612    #[error(
613        "selected DATA_DESC_ID {data_description_id} has no positive finite reference wavelength"
614    )]
615    MissingReferenceWavelength {
616        /// Selected DATA_DESCRIPTION row without usable spectral metadata.
617        data_description_id: u32,
618    },
619    /// The explicit selected-content memory budget could not realize this source.
620    #[error(transparent)]
621    ContentPlan(#[from] SelectedObservationContentPlanError),
622    /// The retained source is not one exact member of the supplied compiled problem.
623    #[error("retained observation source does not match the compiled selected observation")]
624    ProblemSourceMismatch,
625    /// The fresh source-state probe names a different logical MeasurementSet.
626    #[error("current source-state probe names a different MeasurementSet")]
627    CurrentSourceIdentityMismatch,
628    /// Re-evaluated selected rows differ from the compiled row/DDID manifest.
629    #[error("current selected rows differ from the compiled source manifest")]
630    StaleSelectedRows,
631    /// A selected column, metadata, model-column, or consistency generation changed.
632    #[error("current source generations differ from the compiled source snapshot")]
633    StaleSourceGenerations,
634    /// This first native slice does not yet implement the compiled centre laws.
635    #[error(
636        "compiled centre laws require a selected-observation geometry evaluator not yet migrated"
637    )]
638    UnsupportedCentreLaw,
639    /// A fixed centre uses a celestial frame not yet migrated into the native tracer.
640    #[error("fixed selected-observation centres currently require J2000, found {frame:?}")]
641    UnsupportedFixedCentreFrame {
642        /// Unsupported fixed celestial frame.
643        frame: DirectionFrame,
644    },
645    /// A stored frequency reference cannot be represented by the selected-sample schema.
646    #[error("MeasurementSet frequency reference code {code} is unsupported")]
647    UnsupportedFrequencyFrame {
648        /// Standard casacore `MFrequency` reference code.
649        code: i32,
650    },
651    /// Compiled sampling and selected coordinates could not form a bounded contribution set.
652    #[error("selected sample has no valid bounded spectral contribution mapping")]
653    SpectralContributionMismatch,
654    /// A stored epoch reference cannot be represented by the selected-sample schema.
655    #[error("MeasurementSet epoch reference {name} is unsupported")]
656    UnsupportedTimeScale {
657        /// Canonical casacore epoch-reference name.

659    },
660    /// Stored spectral-window coordinates contradicted the compiled channel selection.
661    #[error(

668    /// Stored correlation coordinates contradicted the compiled polarization selection.

673        /// Polarization row whose correlation products differed.

675    },
676    /// A physical MAIN row index did not fit the host storage index domain.
677    #[error("selected physical MAIN row index exceeds the host storage index domain")]
678    PhysicalRowIndexOverflow,
679    /// Retained MAIN cardinality differs from the compiler-owned source manifest.
680    #[error("retained MAIN row count differs from the compiled source manifest")]
681    SourceRowCountMismatch,
682    /// An unconditional resolved selector omitted one or more retained MAIN rows.
683    #[error("compiled unconditional row selection omits retained MAIN rows")]
684    IncompleteUnconditionalRowManifest,
685    /// The bounded storage block did not contain one compiled sample coordinate.
686    #[error("bounded selected-observation storage block has an inconsistent sample shape")]
687    StoredSampleShapeMismatch,
688    /// Stored row geometry could not be represented by the compiled sample schema.
689    #[error("stored selected-observation row geometry is invalid")]
690    InvalidRowGeometry,
691    /// One manifest-listed row no longer satisfies the compiled resolved selector.
692    #[error(
693        "selected physical MAIN row {physical_row} no longer satisfies the compiled row selection"
694    )]
695    SelectedRowPredicateMismatch {
696        /// Manifest-listed physical MAIN row.

698    },
699    /// Observation POINTING evaluation was required but no per-antenna result was supplied.
700    #[error("selected-observation row is missing evaluated POINTING directions")]
701    MissingEvaluatedPointingDirections,
702    /// No POINTING row exists for one required antenna and epoch.
703    #[error("POINTING has no direction for antenna {antenna_id} at MJD seconds {time_mjd_seconds}")]
704    MissingPointingDirection {
705        /// Required antenna.
706        antenna_id: i32,
707        /// Required MAIN epoch in MJD seconds.
708        time_mjd_seconds: f64,
709    },
710    /// The query epoch lies outside POINTING coverage and extrapolation is forbidden.
711    #[error("POINTING coverage excludes antenna {antenna_id} at MJD seconds {time_mjd_seconds}")]
712    PointingOutsideCoverage {
713        /// Required antenna.
714        antenna_id: i32,
715        /// Required MAIN epoch in MJD seconds.
716        time_mjd_seconds: f64,
717    },
718    /// Great-circle POINTING interpolation produced an invalid direction.
719    #[error("POINTING great-circle interpolation is undefined")]
720    InvalidPointingInterpolation,
721}

732pub(super) struct SelectedChannel {
733    channel_index: u32,
734    centre_hz: f64,
735    lower_hz: f64,
736    upper_hz: f64,
737    width_hz: f64,
738    frame: FrequencyFrame,

741fn selected_coordinates(
742    measurement_set: &MeasurementSet,
743    selection: &ObservationSelection,
744) -> Result<Box<[SelectedCoordinates]>, BoundObservationSourceError> {
745    let spectral_windows = measurement_set.spectral_window()?;
746    let polarizations = measurement_set.polarization()?;
747    let mut coordinates = Vec::with_capacity(selection.data_descriptions().len());
748    for data_description in selection.data_descriptions().iter().copied() {
749        let spectral_window = selection
750            .spectral_windows()
751            .iter()
752            .find(|candidate| {
753                candidate.spectral_window_id() == data_description.spectral_window_id()
754            })
755            .expect("compiled DATA_DESCRIPTION has one spectral-window selection");
756        let spw_row = usize::try_from(data_description.spectral_window_id()).map_err(|_| {
757            BoundObservationSourceError::SpectralCoordinateMismatch {
758                spectral_window_id: data_description.spectral_window_id(),
759            }
760        })?;
761        let Some(casa_types::ArrayValue::Float64(centres)) = spectral_windows
762            .table()
763            .column_accessor("CHAN_FREQ")
764            .map_err(MsError::from)?
765            .array_cells_owned_uncached(&[spw_row])
766            .map_err(MsError::from)?
767            .pop()
768            .flatten()
769        else {
770            return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
771                spectral_window_id: data_description.spectral_window_id(),
772            });
773        };
774        let Some(casa_types::ArrayValue::Float64(widths)) = spectral_windows
775            .table()
776            .column_accessor("CHAN_WIDTH")
777            .map_err(MsError::from)?
778            .array_cells_owned_uncached(&[spw_row])
779            .map_err(MsError::from)?
780            .pop()
781            .flatten()
782        else {
783            return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
784                spectral_window_id: data_description.spectral_window_id(),
785            });
786        };
787        let frame = frequency_frame(selected_i32_scalar(
788            spectral_windows.table(),
789            "MEAS_FREQ_REF",
790            spw_row,
791        )?)?;
792        let mut channels = Vec::with_capacity(spectral_window.channel_indices().len());
793        for &channel_index in spectral_window.channel_indices() {
794            let channel = usize::try_from(channel_index).map_err(|_| {
795                BoundObservationSourceError::SpectralCoordinateMismatch {
796                    spectral_window_id: data_description.spectral_window_id(),
797                }
798            })?;
799            let centre_hz = *centres.get(channel).ok_or(
800                BoundObservationSourceError::SpectralCoordinateMismatch {
801                    spectral_window_id: data_description.spectral_window_id(),
802                },
803            )?;
804            let width_hz = *widths.get(channel).ok_or(
805                BoundObservationSourceError::SpectralCoordinateMismatch {
806                    spectral_window_id: data_description.spectral_window_id(),
807                },
808            )?;
809            let first_edge = centre_hz - 0.5 * width_hz;
810            let second_edge = centre_hz + 0.5 * width_hz;
811            if ![centre_hz, width_hz, first_edge, second_edge]
812                .into_iter()
813                .all(f64::is_finite)
814            {
815                return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
816                    spectral_window_id: data_description.spectral_window_id(),
817                });
818            }
819            channels.push(SelectedChannel {
820                channel_index,
821                centre_hz,
822                lower_hz: first_edge.min(second_edge),
823                upper_hz: first_edge.max(second_edge),
824                width_hz,
825                frame,
826            });
827        }
828        let channel_start = channels
829            .first()
830            .and_then(|channel| usize::try_from(channel.channel_index).ok())
831            .expect("compiled spectral-window selection is nonempty");
832        let channel_end = channels
833            .last()
834            .and_then(|channel| usize::try_from(channel.channel_index).ok())
835            .and_then(|channel| channel.checked_add(1))
836            .ok_or(BoundObservationSourceError::SpectralCoordinateMismatch {
837                spectral_window_id: data_description.spectral_window_id(),
838            })?;
839        drop(centres);
840        drop(widths);
841        let polarization = selection
842            .correlations()
843            .iter()
844            .find(|candidate| candidate.polarization_id() == data_description.polarization_id())
845            .expect("compiled DATA_DESCRIPTION has one polarization selection");
846        let polarization_row =
847            usize::try_from(data_description.polarization_id()).map_err(|_| {
848                BoundObservationSourceError::CorrelationCoordinateMismatch {
849                    polarization_id: data_description.polarization_id(),
850                }
851            })?;
852        let Some(casa_types::ArrayValue::Int32(stored_products)) = polarizations
853            .table()
854            .column_accessor("CORR_TYPE")
855            .map_err(MsError::from)?
856            .array_cells_owned_uncached(&[polarization_row])
857            .map_err(MsError::from)?
858            .pop()
859            .flatten()
860        else {
861            return Err(BoundObservationSourceError::CorrelationCoordinateMismatch {
862                polarization_id: data_description.polarization_id(),
863            });
864        };
865        for product in polarization.products() {
866            let product_index = usize::try_from(product.correlation_index()).map_err(|_| {
867                BoundObservationSourceError::CorrelationCoordinateMismatch {
868                    polarization_id: data_description.polarization_id(),
869                }
870            })?;
871            if stored_products
872                .get(product_index)
873                .and_then(|code| correlation_type(*code))
874                != Some(product.correlation_type())
875            {
876                return Err(BoundObservationSourceError::CorrelationCoordinateMismatch {
877                    polarization_id: data_description.polarization_id(),
878                });
879            }
880        }
881        coordinates.push(SelectedCoordinates {
882            data_description,
883            channels: channels.into_boxed_slice(),
884            products: polarization.products().into(),
885            channel_start,
886            channel_count: channel_end - channel_start,
887        });
888    }
889    Ok(coordinates.into_boxed_slice())
890}

893pub(super) struct EvaluatedRowGeometry {
894    density_uvw_m: [f64; 3],
895    transformed_uvw_m: [f64; 3],
896    phase_shift_m: f64,
897    phase_direction: SkyDirection,
898    delay_direction: SkyDirection,
899    pointing_directions: SelectedPointingDirections,

902fn evaluate_row_geometry(
903    source: &BoundObservationSource,
904    problem: &CompiledProblem,
905    stored: SelectedStoredSample,
906    observation_pointing: Option<SelectedPointingDirections>,
907) -> Result<EvaluatedRowGeometry, BoundObservationSourceError> {
908    let field_id = usize::try_from(stored.field_id())
909        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
910    let (longitude_rad, latitude_rad) = source
911        .geometry_engine
912        .field_direction_j2000(field_id)?
913        .as_angles();
914    let observation_direction =
915        SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad);
916    let centres = problem.geometry().centres();
917    let phase_direction = match centres.phase_tracking() {
918        PhaseCentreLaw::Observation => observation_direction,
919        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction)?,
920        PhaseCentreLaw::Ephemeris(_) => {
921            return Err(BoundObservationSourceError::UnsupportedCentreLaw);
922        }
923    };
924    let delay_direction = match centres.delay() {
925        DelayCentreLaw::PhaseTrackingCentre => phase_direction,
926        DelayCentreLaw::Observation => observation_direction,
927        DelayCentreLaw::Fixed(direction) => require_fixed_j2000(*direction)?,
928    };
929    let pointing_directions = match centres.pointing() {
930        PointingCentreLaw::PhaseTrackingCentre => SelectedPointingDirections {
931            antenna1: phase_direction,
932            antenna2: phase_direction,
933        },
934        PointingCentreLaw::Fixed(direction) => {
935            let direction = require_fixed_j2000(*direction)?;
936            SelectedPointingDirections {
937                antenna1: direction,
938                antenna2: direction,
939            }
940        }
941        PointingCentreLaw::Observation(_) => observation_pointing
942            .ok_or(BoundObservationSourceError::MissingEvaluatedPointingDirections)?,
943    };
944    let (density_uvw_m, transformed_uvw_m, phase_shift_m) =
945        if matches!(centres.phase_tracking(), PhaseCentreLaw::Observation) {
946            (stored.uvw_m(), stored.uvw_m(), 0.0)
947        } else {
948            let target = [
949                phase_direction.longitude_rad(),
950                phase_direction.latitude_rad(),
951            ];
952            let density_uvw_m = source
953                .geometry_engine
954                .reproject_raw_uvw_for_density_to_j2000(stored.uvw_m(), field_id, target)?;
955            let (transformed_uvw_m, phase_shift_m) = source
956                .geometry_engine
957                .reproject_raw_uvw_to_j2000(stored.uvw_m(), field_id, target)?;
958            (density_uvw_m, transformed_uvw_m, phase_shift_m)
959        };
960    Ok(EvaluatedRowGeometry {
961        density_uvw_m,
962        transformed_uvw_m,
963        phase_shift_m,
964        phase_direction,
965        delay_direction,
966        pointing_directions,
967    })
968}

970fn evaluate_observation_pointings(
971    source: &BoundObservationSource,
972    problem: &CompiledProblem,
973    buffer: &SelectedObservationBuffer,
974    scan_rows_per_block: usize,
975    maximum_polynomial_terms: usize,
976) -> Result<Option<Vec<SelectedPointingDirections>>, BoundObservationSourceError> {
977    let PointingCentreLaw::Observation(law) = problem.geometry().centres().pointing() else {
978        return Ok(None);
979    };
980    let mut queries = Vec::with_capacity(buffer.row_count().saturating_mul(2));
981    let mut phase_directions = Vec::with_capacity(buffer.row_count());
982    for row in 0..buffer.row_count() {
983        let stored = buffer
984            .sample(0, row, 0)
985            .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
986        let time_mjd_seconds = match law.time_sampling() {
987            PointingTimeSampling::VisibilityTime => stored.time_mjd_seconds(),
988            PointingTimeSampling::VisibilityTimeCentroid => stored.time_centroid_mjd_seconds(),
989        };
990        queries.push(PointingDirectionQuery::new(
991            stored.antenna1(),
992            time_mjd_seconds,
993        )?);
994        queries.push(PointingDirectionQuery::new(
995            stored.antenna2(),
996            time_mjd_seconds,
997        )?);
998        phase_directions.push(evaluate_phase_direction(source, problem, stored)?);
999    }
1000    let column = match law.direction_column() {
1001        PointingDirectionColumn::Direction => StoredPointingDirectionColumn::Direction,
1002        PointingDirectionColumn::Target => StoredPointingDirectionColumn::Target,
1003    };
1004    let brackets = source.measurement_set.pointing_direction_brackets(
1005        &source.geometry_engine,
1006        column,
1007        &queries,
1008        PointingReadPlan::new(scan_rows_per_block, maximum_polynomial_terms)?,
1009    )?;
1010    let mut pointings = Vec::with_capacity(buffer.row_count());
1011    for ((row, antenna_brackets), fallback) in
1012        brackets.chunks_exact(2).enumerate().zip(phase_directions)
1013    {
1014        pointings.push(SelectedPointingDirections {
1015            antenna1: resolve_pointing_direction(
1016                antenna_brackets[0],
1017                queries[2 * row],
1018                *law,
1019                fallback,
1020            )?,
1021            antenna2: resolve_pointing_direction(
1022                antenna_brackets[1],
1023                queries[2 * row + 1],
1024                *law,
1025                fallback,
1026            )?,
1027        });
1028    }
1029    Ok(Some(pointings))
1030}

1032fn evaluate_phase_direction(
1033    source: &BoundObservationSource,
1034    problem: &CompiledProblem,
1035    stored: SelectedStoredSample,
1036) -> Result<SkyDirection, BoundObservationSourceError> {
1037    let field_id = usize::try_from(stored.field_id())
1038        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
1039    let (longitude_rad, latitude_rad) = source
1040        .geometry_engine
1041        .field_direction_j2000(field_id)?
1042        .as_angles();
1043    let observation = SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad);
1044    match problem.geometry().centres().phase_tracking() {
1045        PhaseCentreLaw::Observation => Ok(observation),
1046        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction),
1047        PhaseCentreLaw::Ephemeris(_) => Err(BoundObservationSourceError::UnsupportedCentreLaw),
1048    }
1049}
1050
1051fn resolve_pointing_direction(
1052    bracket: PointingDirectionBracket,
1053    query: PointingDirectionQuery,
1054    law: casa_imaging_model::ObservationPointingLaw,
1055    fallback: SkyDirection,
1056) -> Result<SkyDirection, BoundObservationSourceError> {
1057    if let Some(covering) = bracket.covering() {
1058        return Ok(j2000_direction(covering.direction_j2000_rad()));
1059    }
1060    let direction = match (bracket.before(), bracket.after()) {
1061        (Some(before), Some(after)) if before.row_index() == after.row_index() => {
1062            before.direction_j2000_rad()
1063        }
1064        (Some(before), Some(after)) => match law.interpolation() {
1065            PointingInterpolation::Nearest => {
1066                let before_distance = query.time_mjd_seconds() - before.row_time_mjd_seconds();
1067                let after_distance = after.row_time_mjd_seconds() - query.time_mjd_seconds();
1068                if before_distance <= after_distance {
1069                    before.direction_j2000_rad()
1070                } else {
1071                    after.direction_j2000_rad()
1072                }
1073            }
1074            PointingInterpolation::GreatCircleShortestArc => interpolate_direction(
1075                before.direction_j2000_rad(),
1076                after.direction_j2000_rad(),
1077                (query.time_mjd_seconds() - before.row_time_mjd_seconds())
1078                    / (after.row_time_mjd_seconds() - before.row_time_mjd_seconds()),
1079            )?,
1080        },
1081        (Some(endpoint), None) | (None, Some(endpoint))
1082            if matches!(law.extrapolation(), PointingExtrapolation::HoldNearest) =>
1083        {
1084            endpoint.direction_j2000_rad()
1085        }
1086        (Some(_), None) | (None, Some(_)) => {
1087            return Err(BoundObservationSourceError::PointingOutsideCoverage {
1088                antenna_id: query.antenna_id(),
1089                time_mjd_seconds: query.time_mjd_seconds(),
1090            });
1091        }
1092        (None, None) if matches!(law.missing(), MissingPointingPolicy::UsePhaseTrackingCentre) => {
1093            return Ok(fallback);
1094        }
1095        (None, None) => {
1096            return Err(BoundObservationSourceError::MissingPointingDirection {
1097                antenna_id: query.antenna_id(),
1098                time_mjd_seconds: query.time_mjd_seconds(),
1099            });
1100        }
1101    };
1102    Ok(j2000_direction(direction))
1103}

1106    SkyDirection::new(DirectionFrame::J2000, direction_rad[0], direction_rad[1])

1109fn interpolate_direction(
1110    before: [f64; 2],
1111    after: [f64; 2],
1112    fraction: f64,
1113) -> Result<[f64; 2], BoundObservationSourceError> {
1114    if !(fraction.is_finite() && (0.0..=1.0).contains(&fraction)) {
1115        return Err(BoundObservationSourceError::InvalidPointingInterpolation);
1116    }
1117    let left = unit_direction(before);
1118    let right = unit_direction(after);
1119    let cosine = left
1120        .iter()
1121        .zip(right)
1122        .map(|(left, right)| left * right)
1123        .sum::<f64>()
1124        .clamp(-1.0, 1.0);
1125    let angle = cosine.acos();
1126    let vector = if angle.abs() < 1.0e-15 {
1127        left
1128    } else {
1129        let denominator = angle.sin();
1130        let before_weight = ((1.0 - fraction) * angle).sin() / denominator;
1131        let after_weight = (fraction * angle).sin() / denominator;
1132        [
1133            before_weight * left[0] + after_weight * right[0],
1134            before_weight * left[1] + after_weight * right[1],
1135            before_weight * left[2] + after_weight * right[2],
1136        ]
1137    };
1138    let norm = vector.iter().map(|value| value * value).sum::<f64>().sqrt();
1139    if !(norm.is_finite() && norm > 0.0) {
1140        return Err(BoundObservationSourceError::InvalidPointingInterpolation);
1141    }
1142    let [x, y, z] = vector.map(|value| value / norm);
1143    Ok([y.atan2(x).rem_euclid(std::f64::consts::TAU), z.asin()])
1144}

1146fn unit_direction(direction: [f64; 2]) -> [f64; 3] {
1147    [
1148        direction[1].cos() * direction[0].cos(),
1149        direction[1].cos() * direction[0].sin(),
1150        direction[1].sin(),
1151    ]
1152}

1239fn validate_current_state(
1240    expected: &ObservationSource,
1241    current: &ObservationSourceState,
1242) -> Result<(), BoundObservationSourceError> {
1243    if current.identity() != expected.identity() {
1244        return Err(BoundObservationSourceError::CurrentSourceIdentityMismatch);
1245    }
1246    if current.selected_rows() != expected.selection().rows() {
1247        return Err(BoundObservationSourceError::StaleSelectedRows);
1248    }
1249    if current.generations() != expected.generations() {
1250        return Err(BoundObservationSourceError::StaleSourceGenerations);
1251    }
1252    Ok(())
1253}
1254
1255fn selected_row_predicate(
1256    measurement_set: &MeasurementSet,
1257    source: &ObservationSource,
1258) -> Result<CompiledRowPredicate, BoundObservationSourceError> {
1259    let selection = source.selection();
1260    let wavelengths = validate_data_descriptions(measurement_set, selection)?;
1261    CompiledRowPredicate::new_shared(source, |data_description_id| {
1262        wavelengths
1263            .iter()
1264            .find(|(candidate, _)| *candidate == data_description_id)
1265            .map(|(_, wavelength_m)| *wavelength_m)
1266    })
1267    .map_err(|error| match error {
1268        RowSelectionEvaluationError::MissingReferenceWavelength {
1269            data_description_id,
1270        } => BoundObservationSourceError::MissingReferenceWavelength {
1271            data_description_id,
1272        },
1273    })
1274}

1276fn validate_data_descriptions(
1277    measurement_set: &MeasurementSet,
1278    selection: &ObservationSelection,
1279) -> Result<Vec<(u32, f64)>, BoundObservationSourceError> {
1280    let data_descriptions = measurement_set.data_description()?;
1281    let spectral_windows = measurement_set.spectral_window()?;
1282    let mut wavelengths = Vec::with_capacity(selection.data_descriptions().len());
1283    for expected in selection.data_descriptions() {
1284        let data_description_id = expected.data_description_id();
1285        let row = usize::try_from(data_description_id).map_err(|_| {
1286            BoundObservationSourceError::DataDescriptionCoordinateMismatch {
1287                data_description_id,
1288            }
1289        })?;
1290        let spectral_window_id =
1291            selected_i32_scalar(data_descriptions.table(), "SPECTRAL_WINDOW_ID", row)?;
1292        let polarization_id =
1293            selected_i32_scalar(data_descriptions.table(), "POLARIZATION_ID", row)?;
1294        if u32::try_from(spectral_window_id).ok() != Some(expected.spectral_window_id())
1295            || u32::try_from(polarization_id).ok() != Some(expected.polarization_id())
1296        {
1297            return Err(
1298                BoundObservationSourceError::DataDescriptionCoordinateMismatch {
1299                    data_description_id,
1300                },
1301            );
1302        }
1303        let spectral_window_row = usize::try_from(spectral_window_id).map_err(|_| {
1304            BoundObservationSourceError::DataDescriptionCoordinateMismatch {
1305                data_description_id,
1306            }
1307        })?;
1308        let reference_frequency_hz = selected_f64_scalar(
1309            spectral_windows.table(),
1310            "REF_FREQUENCY",
1311            spectral_window_row,
1312        )?;
1313        wavelengths.push((
1314            data_description_id,
1315            SPEED_OF_LIGHT_M_PER_S / reference_frequency_hz,
1316        ));
1317    }
1318    Ok(wavelengths)
1319}

1321fn selected_i32_scalar(
1322    table: &casa_tables::Table,
1323    column: &str,
1324    row: usize,
1325) -> Result<i32, BoundObservationSourceError> {
1326    match table
1327        .column_accessor(column)
1328        .map_err(MsError::from)?
1329        .scalar_cells_owned_for_rows(&[row])
1330        .map_err(MsError::from)?
1331        .pop()
1332        .flatten()
1333    {
1334        Some(casa_types::ScalarValue::Int32(value)) => Ok(value),
1335        value => Err(MsError::InvalidInput(format!(
1336            "required selected metadata {column} row {row} is not Int32: {value:?}"
1337        ))
1338        .into()),
1339    }
1340}
