16/// Why a `--json-run` request could not be read.
17#[derive(Debug, thiserror::Error)]
18pub enum TaskCliError {
19    #[error("failed to read JSON request from stdin: {0}")]
20    ReadStdin(io::Error),
21    #[error("failed to read JSON request from {path}: {source}")]
22    ReadFile { path: String, source: io::Error },
23    #[error("failed to parse task request: {0}")]
24    ParseRequest(serde_json::Error),
25}
26
27/// The text of a `--json-run` request: from stdin for `-`, or from the
28/// file `source` names.
29pub fn read_task_request(source: &str) -> Result<String, TaskCliError> {
30    if source == "-" {
31        let mut payload = String::new();
32        io::stdin()
33            .read_to_string(&mut payload)
34            .map_err(TaskCliError::ReadStdin)?;
35        return Ok(payload);
36    }
37    fs::read_to_string(source).map_err(|error| TaskCliError::ReadFile {
38        path: Path::new(source).display().to_string(),
39        source: error,
40    })
41}

43/// Parse catalog-declared CLI flags and positional values into one typed
44/// highest-precedence parameter patch.
45///
46/// This parser accepts only spellings declared by the surface contract. Value
47/// parsing, unit conversion, normalization, and constraints are delegated to
48/// the canonical parameter edit contract rather than reimplemented by a
49/// frontend.
50pub fn parse_parameter_cli_overrides(
51    bundle: &SurfaceContractBundle,
52    args: &[OsString],
53) -> Result<ResolutionPatch, String> {
54    let mut patch = ResolutionPatch::default();
55    let mut positional = bundle
56        .surface
57        .bindings()
58        .iter()
59        .filter_map(|binding| {
60            binding
61                .projections
62                .cli
63                .as_ref()
64                .and_then(|projection| projection.positional)
65                .map(|position| (position, binding))
66        })
67        .collect::<Vec<_>>();
68    positional.sort_by_key(|(position, _)| *position);
69    let mut positional_index = 0usize;
70    let mut index = 0usize;
71    while index < args.len() {
72        let raw = args[index]
73            .to_str()
74            .ok_or_else(|| "parameter arguments must be valid UTF-8".to_string())?;
75        if raw == "--unset" {
76            let name = args
77                .get(index + 1)
78                .and_then(|value| value.to_str())
79                .ok_or_else(|| "missing --unset parameter".to_string())?;
80            ensure_parameter_binding(bundle, name)?;
81            patch.values.remove(name);
82            patch.unset.insert(name.to_string());
83            index += 2;
84            continue;
85        }
86        // A surface may declare a short spelling too, such as msexplore's `-o`.
87        let declared_short = !raw.starts_with("--")
88            && raw.starts_with('-')
89            && cli_flag_binding(bundle, raw).is_some();
90        if raw.starts_with("--") || declared_short {
91            let (flag, inline) = raw
92                .split_once('=')
93                .map_or((raw, None), |(flag, value)| (flag, Some(value)));
94            let (binding, false_flag) = cli_flag_binding(bundle, flag)
95                .ok_or_else(|| format!("unknown {} parameter flag {flag}", bundle.surface.id()))?;
96            let concept = bundle
97                .catalog
98                .concept(&binding.concept)
99                .ok_or_else(|| format!("missing concept for {}", binding.name))?;
100            let (value, consumed) = if is_bool_domain(&concept.value_domain) {
101                if inline.is_some() {
102                    return Err(format!("boolean {flag} does not take a value"));
103                }
104                (ParameterValue::Bool(!false_flag), 1)
105            } else {
106                if false_flag {
107                    return Err(format!(
108                        "non-boolean parameter {} declares boolean flag {flag}",
109                        binding.name
110                    ));
111                }
112                let text = match inline {
113                    Some(value) => value,
114                    None => args
115                        .get(index + 1)
116                        .and_then(|value| value.to_str())
117                        .ok_or_else(|| format!("{flag} requires a value"))?,
118                };
119                (
120                    validated_cli_value(bundle, binding, text)?,
121                    if inline.is_some() { 1 } else { 2 },
122                )
123            };
124            // A list without a fixed size gathers what each occurrence
125            // names, as `--archivefile a.exp --archivefile b.exp` names two
126            // archives; any other value takes its last occurrence.
127            let value = match (patch.values.remove(&binding.name), value) {
128                (Some(ParameterValue::Array(mut earlier)), ParameterValue::Array(more))
129                    if is_open_list_domain(&concept.value_domain) =>
130                {
131                    earlier.extend(more);
132                    checked_value(bundle, binding, &ParameterValue::Array(earlier))?
133                }
134                (_, value) => value,
135            };
136            patch.unset.remove(&binding.name);
137            patch.values.insert(binding.name.clone(), value);
138            index += consumed;
139            continue;
140        }
141
142        let Some((_, binding)) = positional.get(positional_index) else {
143            return Err(format!("unexpected positional parameter {raw:?}"));
144        };
145        patch.unset.remove(&binding.name);
146        patch.values.insert(
147            binding.name.clone(),
148            validated_cli_value(bundle, binding, raw)?,
149        );
150        positional_index += 1;
151        index += 1;
152    }
153    Ok(patch)
154}

156/// Whether `flag` is one of `bundle`'s own parameter spellings, which win
157/// over the front end's flags of the same spelling (imregrid's `--template`).
158pub(crate) fn surface_declares_flag(bundle: &SurfaceContractBundle, flag: &str) -> bool {
159    cli_flag_binding(bundle, flag).is_some()
160}

162/// The binding a flag names, and whether it is a false flag.
163///
164/// The surface's declared flags win. Every parameter is also reachable by its
165/// own name, `--vis` or `--gaintable` (#931), and a binding that declares no
166/// spelling by its dashed name.
167fn cli_flag_binding<'a>(
168    bundle: &'a SurfaceContractBundle,
169    flag: &str,
170) -> Option<(&'a SurfaceParameterBinding, bool)> {
171    let bindings = bundle.surface.bindings();
172    let declared = bindings.iter().find_map(|binding| {
173        let projection = binding.projections.cli.as_ref()?;
174        if projection.flags.iter().any(|candidate| candidate == flag) {
175            Some((binding, false))
176        } else if projection
177            .false_flags
178            .iter()
179            .any(|candidate| candidate == flag)
180        {
181            Some((binding, true))
182        } else {
183            None
184        }
185    });
186    declared.or_else(|| {
187        let name = flag.strip_prefix("--")?;
188        bindings
189            .iter()
190            .find(|binding| {
191                binding.name == name
192                    || (binding.projections.cli.as_ref().is_some_and(|projection| {
193                        projection.flags.is_empty()
194                            && projection.false_flags.is_empty()
195                            && projection.positional.is_none()
196                    }) && binding.name.replace('_', "-") == name)
197            })
198            .map(|binding| (binding, false))
199    })
200}

202fn ensure_parameter_binding<'a>(
203    bundle: &'a SurfaceContractBundle,
204    name: &str,
205) -> Result<&'a SurfaceParameterBinding, String> {
206    bundle
207        .surface
208        .bindings()
209        .iter()
210        .find(|binding| binding.name == name)
211        .ok_or_else(|| format!("unknown {} parameter {name:?}", bundle.surface.id()))
212}

214/// Parse, normalize and validate one command-line value through the
215/// binding's concept and refinements.
216fn validated_cli_value(
217    bundle: &SurfaceContractBundle,
218    binding: &SurfaceParameterBinding,
219    text: &str,
220) -> Result<ParameterValue, String> {
221    let concept = bundle
222        .catalog
223        .concept(&binding.concept)
224        .ok_or_else(|| format!("missing parameter concept for {:?}", binding.name))?;
225    if text.trim().is_empty() {
226        return Err(format!("parameter {:?} has no typed value", binding.name));
227    }
228    let parsed = parse_parameter_text(text, &concept.value_domain)
229        .map_err(|error| format!("parameter {:?}: {error}", binding.name))?;
230    checked_value(bundle, binding, &parsed)
231}

233/// `value` normalized and validated through `binding`'s concept and
234/// refinements; a refusal names the parameter.
235pub(crate) fn checked_value(
236    bundle: &SurfaceContractBundle,
237    binding: &SurfaceParameterBinding,
238    value: &ParameterValue,
239) -> Result<ParameterValue, String> {
240    let named = |error: crate::NormalizationError| format!("parameter {:?}: {error}", binding.name);
241    let concept = bundle
242        .catalog
243        .concept(&binding.concept)
244        .ok_or_else(|| format!("missing parameter concept for {:?}", binding.name))?;
245    let normalized = normalize_value(value, concept).map_err(named)?;
246    validate_value(&normalized, concept, &binding.refinements).map_err(named)?;
247    Ok(normalized)
248}

250fn is_bool_domain(domain: &ParameterType) -> bool {
251    match domain {
252        ParameterType::Bool => true,
253        ParameterType::Optional { value, .. } => is_bool_domain(value),
254        _ => false,
255    }
256}
257
258/// Whether `domain` is a list with no fixed size, which a repeated flag
259/// extends.
260fn is_open_list_domain(domain: &ParameterType) -> bool {
261    match domain {
262        ParameterType::Array { max_items, .. } => max_items.is_none(),
263        ParameterType::Optional { value, .. } => is_open_list_domain(value),
264        _ => false,
265    }
266}

287    fn catalog_cli_parser_accepts_only_declared_flags_and_normalizes_values() {
288        let bundle = casa_provider_contracts::builtin_surface_bundle("imager").unwrap();
289        let patch = parse_parameter_cli_overrides(
290            &bundle,
291            &[
292                "--ms".into(),
293                "input.ms".into(),
294                "--imsize".into(),
295                "1024".into(),
296                "--cell-arcsec=0.0002777777777777778deg".into(),
297                "--gridder".into(),
298                "awproject".into(),

301                "f64".into(),
302            ],
303        )
304        .unwrap();
305
306        assert_eq!(
307            patch.values["vis"],
308            ParameterValue::Array(vec![ParameterValue::String("input.ms".into())])
309        );
310        assert_eq!(
311            patch.values["imsize"],
312            ParameterValue::Array(vec![ParameterValue::Integer(1024); 2])
313        );
314        assert_eq!(
315            patch.values["cell"],

321            ParameterValue::String("f64".into())
322        );

324        // The parameter's own name reaches the same binding and value.
325        let named =

327        let declared =
328            parse_parameter_cli_overrides(&bundle, &["--threshold-jy".into(), "1Jy".into()])
329                .unwrap();
330        assert_eq!(named, declared);
331        let unknown =
332            parse_parameter_cli_overrides(&bundle, &["--thresh".into(), "1Jy".into()]).unwrap_err();
333        assert!(unknown.contains("unknown imager parameter flag --thresh"));
