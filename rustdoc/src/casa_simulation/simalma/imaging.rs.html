19pub(super) fn execute<'a, 'b>(
20    p: &Parameters,
21    plan: &Plan,
22    resources: SessionResources,
23    mut plot_sink: Option<&'a mut (dyn SimulationPlotSink + 'b)>,
24    outcome: &mut crate::simobserve::SimobserveOutcome,
25    cancel: &Cancel,
26) -> Result<(), SimobserveError> {
27    let mut products = Vec::new();
28    if !plan.tp_vis.is_empty() {
29        products.extend(total_power(
30            p,
31            plan,
32            resources,
33            plot_sink.as_deref_mut(),
34            cancel,
35        )?);
36    }
37    for array in &plan.arrays {
38        let result = synthesis(
39            p,
40            plan,
41            &plan.vis(array),
42            resources,
43            plot_sink.as_deref_mut(),
44            cancel,
45        )?;
46        products.extend(result.products);
47    }
48    let high = if plan.arrays.len() > 1 {
49        let path = plan.directory.join(format!("{}.concat.ms", plan.project));
50        concatenate(
51            plan.arrays.iter().map(|array| plan.vis(array)).collect(),
52            plan.arrays
53                .iter()
54                .map(|array| if array.aca { 0.34 } else { 1. })
55                .collect(),
56            &path,
57            resources,
58        )?;
59        products.push(path.clone());
60        let result = synthesis(p, plan, &path, resources, plot_sink, cancel)?;
61        products.extend(result.products);
62        plan.directory
63            .join(format!("{}.concat.image", plan.project))
64    } else {
65        plan.vis(&plan.arrays[0]).with_extension("image")
66    };
67    if !plan.tp_vis.is_empty() {
68        check_cancel(cancel)?;
69        products.extend(super::combine::execute(plan, &high, resources)?);
70    }
71    outcome.products.extend(products);
72    Ok(())
73}

75fn synthesis<'a, 'b>(
76    p: &Parameters,
77    plan: &Plan,
78    vis: &Path,
79    resources: SessionResources,
80    plot_sink: Option<&'a mut (dyn SimulationPlotSink + 'b)>,
81    cancel: &Cancel,
82) -> Result<crate::simanalyze::SimanalyzeOutcome, SimobserveError> {
83    let cell = if p.text("cell").is_empty() {
84        format!("{}arcsec", plan.model.cell[0])
85    } else {
86        p.text("cell").to_owned()
87    };
88    let imsize = if p.0["imsize"]
89        .as_array()
90        .and_then(|v| v.first())
91        .and_then(Value::as_i64)
92        .unwrap_or(0)
93        > 0
94    {
95        p.0["imsize"].clone()
96    } else {
97        json!((plan.model.size[0] / plan.model.cell[0]) as usize)
98    };
99    let request = SimanalyzeRequest {
100        workspace: Some(plan.directory.clone()),
101        resources,
102        parameters: json!({"project":plan.directory,"vis":vis,"image":true,"modelimage":"","cell":cell,"imsize":imsize,"imdirection":if p.text("imdirection").is_empty(){plan.model.direction.as_str()}else{p.text("imdirection")},"niter":p.0["niter"],"threshold":p.text("threshold"),"weighting":"briggs","mask":[],"outertaper":[],"stokes":"I","analyze":true,"graphics":if plot_sink.is_some(){super::graphics_mode(p)}else{"none"},"verbose":p.flag("verbose"),"overwrite":true})
103            .as_object()
104            .unwrap()
105            .clone(),
106    };
107    simanalyze_with_controls(&request, plot_sink, cancel)
108}

110fn total_power<'a, 'b>(
111    p: &Parameters,
112    plan: &Plan,
113    resources: SessionResources,
114    plot_sink: Option<&'a mut (dyn SimulationPlotSink + 'b)>,
115    cancel: &Cancel,
116) -> Result<Vec<PathBuf>, SimobserveError> {
117    let primary_beam =
118        casa_imaging_singledish::primary_beam_fwhm_arcsec(plan.model.frequency, 12., 0.75, 10.);
119    let cell = format!("{}arcsec", primary_beam / 9.);
120    let raw = plan.directory.join(format!("{}.sd.image0", plan.project));
121    // The single-MS gridder sees all TP dishes through a private native
122    // concatenation; retain their row weights, antenna and pointing metadata.
123    let staging = tempfile::tempdir_in(&plan.directory).map_err(failure)?;
124    let input = if plan.tp_vis.len() > 1 {
125        let path = staging.path().join("tp.ms");
126        concatenate(plan.tp_vis.clone(), Vec::new(), &path, resources)?;
127        path
128    } else {
129        plan.tp_vis[0].clone()
130    };
131    let sparse = json!({"infiles":[input],"outfile":raw,"imsize":p.0["imsize"],"cell":[cell.clone(),cell],"phasecenter":plan.model.direction,"mode":"channel","nchan":plan.model.channels,"gridfunction":"sf","convsupport":6});
132    let result = casa_imaging::single_dish::image_from_sparse_parameters(
133        sparse.as_object().unwrap(),
134        &crate::simanalyze::imaging::context(resources, cancel),
135    )
136    .map_err(failure)?;
137    let beam = result
138        .beam
139        .ok_or_else(|| failure("TP beam could not be measured"))?;
140    let ratio =
141        beam.major_in("arcsec").map_err(failure)?.powi(2) / (plan.primary_beam * 0.96).powi(2);
142    let ratio: f32 = format!("{ratio:.6}").parse().map_err(failure)?;
143    let image = PagedImage::<f32>::open(&result.image).map_err(failure)?;
144    let memory = image
145        .shape()
146        .iter()
147        .try_fold(32u64, |bytes, &n| bytes.checked_mul(n as u64))
148        .and_then(|bytes| bytes.checked_add(32 << 20))
149        .ok_or_else(|| failure("TP dimensions overflow"))?;
150    let _memory = admit(
151        resources.host(),
152        resources.policy(),
153        &Demand {
154            phase: "ALMA TP intensity scaling",
155            memory,
156        },
157    )
158    .map_err(failure)?;
159    let output = plan.directory.join(format!("{}.sd.image", plan.project));
160    analysis::write_product(
161        &output,
162        &image.get().map_err(failure)?.mapv(|v| v * ratio),
163        image.coordinates().clone(),
164        image.units(),
165        &image.image_info().map_err(failure)?,
166        image.default_mask().map_err(failure)?.as_ref(),
167    )
168    .map_err(failure)?;
169    drop(image);
170    std::fs::rename(&result.image, &raw).map_err(failure)?;
171    drop(_memory);
172    let skymodel = plan.directory.join(format!("{}.skymodel", plan.tp_stem));
173    let request = SimanalyzeRequest {
174        workspace: Some(plan.directory.clone()),
175        resources,
176        parameters: json!({"project":plan.directory,"image":false,"imagename":output,"skymodel":skymodel,"analyze":true,"showuv":false,"showpsf":false,"showconvolved":true,"graphics":if plot_sink.is_some(){super::graphics_mode(p)}else{"none"},"overwrite":true})
177            .as_object()
178            .unwrap()
179            .clone(),
180    };
181    let analyzed = match plot_sink {
182        Some(sink) => simanalyze_with_plot_sink_and_override(
183            &request,
184            sink,
185            Some(
186                plan.directory
187                    .join(format!("{}.analysis.png", plan.project)),
188            ),
189            cancel,
190        )?,
191        None => simanalyze_with_controls(&request, None, cancel)?,
192    };
193    let mut products = vec![raw, result.weight, output];
194    products.extend(analyzed.products);
195    Ok(products)
196}

198fn concatenate(
199    vis: Vec<PathBuf>,
200    weights: Vec<f64>,
201    target: &Path,
202    resources: SessionResources,
203) -> Result<(), SimobserveError> {
204    let memory = resources
205        .policy()
206        .memory(resources.host())
207        .saturating_sub(casa_imaging_runtime::reserved_memory());
208    let _memory = admit(
209        resources.host(),
210        resources.policy(),
211        &Demand {
212            phase: "ALMA concatenation",
213            memory,
214        },
215    )
216    .map_err(failure)?;
217    concat_with_memory(
218        &ConcatRequest {
219            vis,
220            concatvis: target.to_path_buf(),
221            freqtol: json!(""),
222            dirtol: json!(""),
223            respectname: false,
224            timesort: false,
225            copypointing: true,
226            visweightscale: weights,
227            forcesingleephemfield: json!(""),
228        },
229        usize::try_from(memory).map_err(failure)?,
230    )
231    .map(|_| ())
232    .map_err(failure)
233}
