2//! An [`MFrequency`] conversion prepared once for many frequencies.
3
4use super::MeasFrame;
5use super::error::MeasureError;
6use super::frequency::{FrequencyRef, MFrequency, find_freq_path};
7
8/// The most hops a route between two frequency frames takes.
9const MOST_HOPS: usize = FrequencyRef::ALL.len() - 1;
10
11/// A conversion of frequencies from one frame to another, prepared once:
12/// the factor of each hop of the route between the frames, evaluated in
13/// that hop's frame, so converting a frequency costs one multiplication a
14/// hop. A hop's factor is the hop applied to 1 Hz, which every hop's
15/// Doppler shift scales linearly.
16///
17/// The route is the one [`MFrequency::convert_to`] takes. A conversion
18/// between two frames can carry a different context for each (a source
19/// direction and epoch, and the output's own velocity or position): each
20/// hop is evaluated in the source frame, except `LSRK → REST`,
21/// `BARY → LSRK`, `BARY → LSRD`, `LSRD → GALACTO`, `BARY → LGROUP`,
22/// `BARY → CMB`, `BARY → GEO` and `GEO → TOPO`, which are evaluated in the
23/// target frame.
24#[derive(Clone, Copy, Debug, PartialEq)]
25pub struct PreparedFrequencyConversion {
26    factors: [f64; MOST_HOPS],
27    hops: usize,
28}
29
30impl PreparedFrequencyConversion {
31    /// The conversion from `source` to `target`, the hops evaluated in
32    /// `source_frame` or `target_frame` as the type states. Equal frames
33    /// convert unchanged and need neither.
34    ///
35    /// # Errors
36    ///
37    /// [`MeasureError::MissingFrameData`] when a frame the route needs is
38    /// absent; the hop's own error when a frame lacks what it needs.
39    pub fn new(
40        source: FrequencyRef,
41        target: FrequencyRef,
42        source_frame: Option<&MeasFrame>,
43        target_frame: Option<&MeasFrame>,
44    ) -> Result<Self, MeasureError> {
45        let mut factors = [1.0; MOST_HOPS];
46        if source == target {
47            return Ok(Self { factors, hops: 0 });
48        }
49        let source_frame = source_frame.ok_or(MeasureError::MissingFrameData {
50            what: "source frame of a cross-frame frequency conversion",
51        })?;
52        let target_frame = target_frame.ok_or(MeasureError::MissingFrameData {
53            what: "target frame of a cross-frame frequency conversion",
54        })?;
55        let path = find_freq_path(source, target)?;
56        let mut current = source;
57        for (hop, next) in path.iter().copied().enumerate() {
58            let frame = if uses_target_frame(current, next) {
59                target_frame
60            } else {
61                source_frame
62            };
63            factors[hop] = MFrequency::new(1.0, current).convert_to(next, frame)?.hz();
64            current = next;
65        }
66        Ok(Self {
67            factors,
68            hops: path.len(),
69        })
70    }
71
72    /// `frequency_hz` converted.
73    #[must_use]
74    pub fn convert_hz(&self, frequency_hz: f64) -> f64 {
75        self.factors[..self.hops]
76            .iter()
77            .fold(frequency_hz, |converted, factor| converted * factor)
78    }
79}
80
81/// Whether the hop from `from` to `to` is evaluated in the target frame.
82const fn uses_target_frame(from: FrequencyRef, to: FrequencyRef) -> bool {
83    matches!(
84        (from, to),
85        (FrequencyRef::LSRK, FrequencyRef::REST)
86            | (FrequencyRef::BARY, FrequencyRef::LSRK)
87            | (FrequencyRef::BARY, FrequencyRef::LSRD)
88            | (FrequencyRef::LSRD, FrequencyRef::GALACTO)
89            | (FrequencyRef::BARY, FrequencyRef::LGROUP)
90            | (FrequencyRef::BARY, FrequencyRef::CMB)
91            | (FrequencyRef::BARY, FrequencyRef::GEO)
92            | (FrequencyRef::GEO, FrequencyRef::TOPO)
93    )
94}

96#[cfg(test)]
97mod tests {
98    use super::PreparedFrequencyConversion;
99    use crate::measures::MeasFrame;
100    use crate::measures::error::MeasureError;
101    use crate::measures::frequency::FrequencyRef;
102
103    /// Equal frames need no frame; different ones need both.
104    #[test]
105    fn a_cross_frame_conversion_needs_both_frames() {
106        let same =
107            PreparedFrequencyConversion::new(FrequencyRef::LSRK, FrequencyRef::LSRK, None, None)
108                .expect("equal frames");
109        assert_eq!(same.convert_hz(6.667_582_296e9), 6.667_582_296e9);
110        for (source, target) in [
111            (None, Some(MeasFrame::new())),
112            (Some(MeasFrame::new()), None),
113        ] {
114            let missing = PreparedFrequencyConversion::new(
115                FrequencyRef::TOPO,
116                FrequencyRef::LSRK,
117                source.as_ref(),
118                target.as_ref(),
119            );
120            assert!(matches!(
121                missing,
122                Err(MeasureError::MissingFrameData { .. })
123            ));
124        }
125    }
