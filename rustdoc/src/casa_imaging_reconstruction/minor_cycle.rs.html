3//! Bounded Högbom Minor Cycle over authoritative Normal State views.
4//!
5//! The Minor Cycle is a pure scientific operation: it consumes one immutable
6//! view of an authoritative Final Normal State (the T20 residual paired with
7//! its normal approximation), one named authoritative model generation, an
8//! explicit valid-support window, and explicit gain, threshold, iteration,
9//! and staleness controls. It accumulates Högbom components against the PSF
10//! exactly like the casacore `LatticeCleaner` HOGBOM loop — peak residual
11//! normalized by the PSF peak, gain-scaled PSF subtraction — and mints the
12//! resulting sparse update only through the T28 model lifecycle owner as one
13//! base-bound Model Delta. Authoritative model and residual state are never
14//! mutated; the solver works on private buffers.
15//!
16//! Every non-converged stop (`IterationBound`, `StalenessBound`) records that
17//! the linear approximation envelope is exhausted and explicitly requests
18//! Major-Cycle reconciliation. Component-sequence recording is optional
19//! diagnostic evidence: first-divergence comparisons between two recorded
20//! sequences are informational and never gate acceptance.
21

83/// Explicit Högbom Minor-Cycle controls.
84///
85/// Every control is explicit and validated; there are no defaults that could
86/// silently change deconvolution semantics. `maximum_model_update` is the
87/// error/staleness bound of the linear view envelope: once the cumulative
88/// absolute component flux reaches it, the solve stops before further
89/// extrapolating the frozen normal-state approximation and requests
90/// Major-Cycle reconciliation.
91#[derive(Debug, Clone, Copy, PartialEq)]
92pub struct HogbomControls {
93    gain: f64,
94    threshold: f64,
95    max_iterations: usize,
96    maximum_model_update: f64,
97    component_sequence_limit: Option<usize>,
98}
99
100impl HogbomControls {
101    /// Construct validated controls.
102    ///
103    /// # Errors
104    ///
105    /// Rejects a gain outside `(0, 1]`, a negative or non-finite threshold, a
106    /// zero iteration bound, and a non-positive or non-finite maximum model
107    /// update.
108    pub fn new(
109        gain: f64,
110        threshold: f64,
111        max_iterations: usize,
112        maximum_model_update: f64,
113    ) -> Result<Self, MinorCycleError> {
114        if !(gain > 0.0 && gain <= 1.0) {
115            return Err(MinorCycleError::InvalidGain);
116        }
117        if !threshold.is_finite() || threshold < 0.0 {
118            return Err(MinorCycleError::InvalidThreshold);
119        }
120        if max_iterations == 0 {
121            return Err(MinorCycleError::InvalidIterationBound);
122        }
123        if !maximum_model_update.is_finite() || maximum_model_update <= 0.0 {
124            return Err(MinorCycleError::InvalidMaximumModelUpdate);
125        }
126        Ok(Self {
127            gain,
128            threshold,
129            max_iterations,
130            maximum_model_update,
131            component_sequence_limit: None,
132        })
133    }
134
135    /// Record the first accepted components as diagnostic evidence.
136    ///
137    /// Recording is capped so a long clean cannot retain unbounded component
138    /// lists; cleaning itself continues past the cap.
139    ///
140    /// # Errors
141    ///
142    /// Rejects a zero limit.
143    pub fn record_component_sequence(mut self, limit: usize) -> Result<Self, MinorCycleError> {
144        if limit == 0 {
145            return Err(MinorCycleError::InvalidRecordingLimit);
146        }
147        self.component_sequence_limit = Some(limit);
148        Ok(self)
149    }
150
151    /// Return the loop gain fraction applied to each normalized peak.
152    #[must_use]
153    pub const fn gain(&self) -> f64 {
154        self.gain
155    }
156
157    /// Return the absolute normalized-flux stopping threshold.
158    #[must_use]
159    pub const fn threshold(&self) -> f64 {
160        self.threshold
161    }
162
163    /// Return the hard iteration bound.
164    #[must_use]
165    pub const fn max_iterations(&self) -> usize {
166        self.max_iterations
167    }
168
169    /// Return the cumulative absolute component flux that exhausts the view
170    /// envelope and forces an explicit reconciliation request.
171    #[must_use]
172    pub const fn maximum_model_update(&self) -> f64 {
173        self.maximum_model_update
174    }
175
176    /// Return the optional recorded-component-sequence capacity.
177    #[must_use]
178    pub const fn component_sequence_limit(&self) -> Option<usize> {
179        self.component_sequence_limit
180    }
181}

183/// Explicit valid-support window constraining component placement.
184///
185/// Bounds are inclusive pixel coordinates `[x, y]`. Only window cells whose
186/// named model-generation sample has valid support may host a component,
187/// because only the model owner accepts delta terms on valid support.
188#[derive(Debug, Clone, Copy, PartialEq, Eq)]
189pub struct CleanWindow {
190    blc: [usize; 2],
191    trc: [usize; 2],
192}
193
194impl CleanWindow {
195    /// Construct an inclusive window with `blc <= trc` on both axes.
196    ///
197    /// # Errors
198    ///
199    /// Rejects an inverted window.
200    pub fn new(blc: [usize; 2], trc: [usize; 2]) -> Result<Self, MinorCycleError> {
201        if blc[0] > trc[0] || blc[1] > trc[1] {
202            return Err(MinorCycleError::InvertedWindow);
203        }
204        Ok(Self { blc, trc })
205    }
206
207    /// The CASA default inner-quarter window for one plane shape.
208    #[must_use]
209    pub fn inner_quarter(shape: [usize; 2]) -> Self {
210        let half = [shape[0] / 2, shape[1] / 2];
211        let quarter = [shape[0] / 4, shape[1] / 4];
212        let mut blc = quarter;
213        let mut trc = [
214            quarter[0].saturating_add(half[0]).saturating_sub(1),
215            quarter[1].saturating_add(half[1]).saturating_sub(1),
216        ];
217        for axis in 0..2 {
218            trc[axis] = trc[axis].min(shape[axis] - 1);
219            blc[axis] = blc[axis].min(trc[axis]);
220        }
221        Self { blc, trc }
222    }
223
224    /// The full plane for one shape.
225    #[must_use]
226    pub fn full_plane(shape: [usize; 2]) -> Self {
227        Self {
228            blc: [0, 0],
229            trc: [shape[0] - 1, shape[1] - 1],
230        }
231    }
232
233    /// Return the inclusive lower bound.
234    #[must_use]
235    pub const fn blc(&self) -> [usize; 2] {
236        self.blc
237    }
238
239    /// Return the inclusive upper bound.
240    #[must_use]
241    pub const fn trc(&self) -> [usize; 2] {
242        self.trc
243    }
244
245    fn contains(&self, pixel: [usize; 2]) -> bool {
246        pixel[0] >= self.blc[0]
247            && pixel[0] <= self.trc[0]
248            && pixel[1] >= self.blc[1]
249            && pixel[1] <= self.trc[1]
250    }
251}

413    /// reconciliation.
414    ///
415    /// Only a threshold stop resolves the solve inside its valid envelope;
416    /// iteration and staleness stops leave potential update outstanding and
417    /// must be followed by a fresh complete-data reconciliation.
418    #[must_use]

420        !matches!(self.stop_reason, MinorCycleStopReason::ThresholdReached)
421    }
422
423    /// Return the optionally recorded leading component sequence.
424    #[must_use]
425    pub fn recorded_component_sequence(&self) -> Option<&[HogbomComponent]> {
426        self.recorded.as_deref()
427    }
428
429    /// Compare this evidence's recorded sequence against another baseline.
430    ///
431    /// Returns the first index where the sequences differ in pixel or exact
432    /// flux bits, or where one sequence ends early. Comparison is purely
433    /// informational and never gates acceptance; `None` means the sequences
434    /// agree or cannot be compared (either side unrecorded).
435    #[must_use]
436    pub fn first_divergence(&self, baseline: &Self) -> Option<ComponentDivergence> {
437        let candidate = self.recorded.as_deref()?;
438        let baseline = baseline.recorded.as_deref()?;
439        let shared = candidate.len().min(baseline.len());
440        for index in 0..shared {
441            if !baseline[index].same_value(&candidate[index]) {
442                return Some(ComponentDivergence {
443                    index,
444                    baseline: Some(baseline[index]),
445                    candidate: Some(candidate[index]),
446                });
447            }
448        }
449        if baseline.len() != candidate.len() {
450            let index = shared;
451            return Some(ComponentDivergence {
452                index,
453                baseline: baseline.get(index).copied(),
454                candidate: candidate.get(index).copied(),
455            });
456        }
457        None
458    }

489/// Exact reason a bounded Högbom solve failed closed.
490#[derive(Debug, Clone, PartialEq, Eq, Error)]
491pub enum MinorCycleError {
492    /// The Högbom gain was outside `(0, 1]`.
493    #[error("Högbom gain must lie in (0, 1]")]
494    InvalidGain,
495    /// The stopping threshold was negative or non-finite.
496    #[error("Högbom threshold must be finite and non-negative")]
497    InvalidThreshold,
498    /// The iteration bound was zero.
499    #[error("Högbom iteration bound must be non-zero")]
500    InvalidIterationBound,
501    /// The maximum model update was non-positive or non-finite.
502    #[error("Högbom maximum model update must be finite and positive")]
503    InvalidMaximumModelUpdate,
504    /// Component-sequence recording requested a zero capacity.
505    #[error("component-sequence recording requires a non-zero limit")]
506    InvalidRecordingLimit,
507    /// The window bounds were inverted.
508    #[error("clean window requires blc <= trc on both axes")]
509    InvertedWindow,
510    /// The window lay partly or wholly outside the normal-state plane.
511    #[error("clean window lies outside the normal-state plane")]
512    WindowOutsidePlane,
513    /// The window contained no valid model support at all.
514    #[error("clean window contains no valid model support")]
515    EmptyValidSupport,
516    /// The named generation does not match the single-plane constant-basis
517    /// normal-state geometry.
518    #[error("model generation does not match the normal-state plane")]
519    ModelShapeMismatch,
520    /// The named generation is not the view's final model generation.
521    #[error("named generation is not the normal state's final model generation")]
522    ForeignNormalState,
523    /// The normal approximation lacked a positive finite peak.
524    #[error("normal-state approximation lacks a positive finite PSF peak")]
525    InvalidPsfPeak,
526    /// Solver arithmetic produced a non-finite value.
527    #[error("minor-cycle arithmetic generated a non-finite value")]
528    GeneratedNonfinite,
529    /// The model lifecycle owner rejected delta validation or minting.
530    #[error(transparent)]
531    Lifecycle(#[from] crate::ModelLifecycleError),
532}

536        Self::Lifecycle(ModelLifecycleError::Contract(error))

540/// Run one bounded Högbom Minor Cycle.
541///
542/// Consumes an immutable view of one authoritative Final Normal State and the
543/// exact named final model generation, and returns the sparse Model Delta
544/// minted through the model lifecycle owner plus solver evidence. Neither the
545/// authoritative residual planes nor the model generation are mutated.
546///
547/// The loop preserves the casacore HOGBOM semantics: find the maximum-
548/// absolute residual over the valid-support window, normalize by the PSF
549/// peak, stop when below threshold, otherwise subtract `gain * strength`
550/// scaled by the PSF centered on the peak and accumulate the same flux at the
551/// peak cell. Scalar Stokes-I convention uses the real plane, matching the
552/// Float-image reference behavior.
553///
554/// # Errors
555///
556/// Fails closed on mismatched lineage or geometry, invalid controls or
557/// windows, a degenerate PSF peak, non-finite arithmetic, and every model
558/// owner rejection (term bounds, value bounds, closed lifecycle).
559pub fn hogbom_minor_cycle(
560    lifecycle: &ModelLifecycle,
561    base: &ModelGeneration,
562    view: &FinalNormalState,
563    window: CleanWindow,
564    controls: HogbomControls,
565) -> Result<HogbomMinorCycle, MinorCycleError> {
566    let shape = view.shape();
567    let cells = shape[0] * shape[1];
568    if base.shape().domains().len() != 1
569        || base.shape().coefficients() != 1
570        || base.shape().polarizations() != 1
571        || base.shape().domains()[0].pixels() != shape
572        || base.samples().len() != cells
573    {
574        return Err(MinorCycleError::ModelShapeMismatch);
575    }
576    if base.generation_id() != view.final_model_generation() {
577        return Err(MinorCycleError::ForeignNormalState);
578    }
579    if window.trc()[0] >= shape[0] || window.trc()[1] >= shape[1] {
580        return Err(MinorCycleError::WindowOutsidePlane);
581    }
582
583    // The PSF peak normalization follows the reference cleaner: peaks are
584    // reported in model units regardless of the accumulated weight scale.
585    let psf_peak_index = find_peak_abs(view.normal_approximation(), shape, |v| v.re, |_| true)
586        .ok_or(MinorCycleError::InvalidPsfPeak)?;
587    let psf_peak = view.normal_approximation()[psf_peak_index].re;
588    if !psf_peak.is_finite() || psf_peak <= 0.0 {
589        return Err(MinorCycleError::InvalidPsfPeak);
590    }
591    let psf_peak_pixel = plane_pixel(psf_peak_index, shape);
592
593    // Private working copy: authoritative state is never mutated.
594    let mut residual = Vec::with_capacity(cells);
595    for value in view.residual() {
596        let real = value.re;
597        if !real.is_finite() {
598            return Err(MinorCycleError::GeneratedNonfinite);
599        }
600        residual.push(real);
601    }
602
603    let mut terms = BTreeMap::<usize, f64>::new();
604    let mut recorded = Vec::with_capacity(
605        controls
606            .component_sequence_limit()
607            .unwrap_or(0)
608            .min(controls.max_iterations()),
609    );
610    let mut iterations = 0_usize;
611    let mut total_flux = 0.0_f64;
612    let mut final_peak_flux = 0.0_f64;
613    let mut stop_reason = None;
614
615    for _ in 0..controls.max_iterations() {
616        let Some(peak_index) = find_peak_abs(
617            &residual,
618            shape,
619            |value| *value,
620            |pixel| window.contains(pixel) && valid_support(base, shape, pixel),
621        ) else {
622            return Err(MinorCycleError::EmptyValidSupport);
623        };
624        let peak_pixel = plane_pixel(peak_index, shape);
625        let strength = residual[peak_index] / psf_peak;
626        if !strength.is_finite() {
627            return Err(MinorCycleError::GeneratedNonfinite);
628        }
629        final_peak_flux = strength.abs();
630        if strength == 0.0 || strength.abs() < controls.threshold() {
631            stop_reason = Some(MinorCycleStopReason::ThresholdReached);
632            break;
633        }
634        let flux = controls.gain() * strength;
635        if !flux.is_finite() {
636            return Err(MinorCycleError::GeneratedNonfinite);
637        }
638        subtract_psf(
639            &mut residual,
640            view.normal_approximation(),
641            shape,
642            peak_pixel,
643            psf_peak_pixel,
644            flux,
645        )?;
646        iterations += 1;
647        total_flux += flux.abs();
648        let cell =
649            model_cell(shape, peak_pixel).expect("a scanned peak pixel lies inside the plane");
650        *terms.entry(canonical_flat(base, cell)).or_insert(0.0) += flux;
651        if let Some(limit) = controls.component_sequence_limit()
652            && recorded.len() < limit
653        {
654            recorded.push(HogbomComponent { cell, flux });
655        }
656        if total_flux >= controls.maximum_model_update() {
657            stop_reason = Some(MinorCycleStopReason::StalenessBound);
658            break;
659        }
660    }
661    let stop_reason = stop_reason.unwrap_or(MinorCycleStopReason::IterationBound);
662
663    terms.retain(|_, flux| *flux != 0.0);
664    let delta = if terms.is_empty() {
665        None
666    } else {
667        let deltas = terms
668            .iter()
669            .map(|(flat, flux)| {
670                let cell = base
671                    .shape()
672                    .cell_at(*flat)
673                    .expect("accumulated flat keys stay inside the base shape");
674                Ok(ModelDeltaTerm::new(cell, ModelValue::new(*flux)?))
675            })
676            .collect::<Result<Vec<_>, MinorCycleError>>()?;
677        Some(lifecycle.compile_delta(base, deltas)?)
678    };
679
680    let evidence_id = minor_cycle_evidence_id(
681        lifecycle.authority(),
682        lifecycle.attempt(),
683        lifecycle.epoch(),
684        base.generation_id(),
685        view.completion_id(),
686        view.content_identity(),
687        &window,
688        &controls,
689        iterations,
690        total_flux,
691        final_peak_flux,
692        stop_reason,
693    );
694    Ok(HogbomMinorCycle {
695        delta,
696        evidence: MinorCycleEvidence {
697            evidence_id,
698            problem: lifecycle.problem(),
699            attempt: lifecycle.attempt(),
700            epoch: lifecycle.epoch(),
701            input_generation: base.generation_id(),
702            normal_state_completion: view.completion_id(),
703            normal_state_content: view.content_identity(),
704            iterations,
705            total_flux,
706            final_peak_flux,
707            stop_reason,
708            recorded: (!recorded.is_empty()).then(|| recorded.into_boxed_slice()),
709        },
710    })
711}

713/// Canonical single-plane pixel of one plane-storage index.
714///
715/// Normal-state planes are stored x-major (`finish_bound` pushes x outer, y
716/// inner), so index `p` maps to pixel `[p / H, p % H]`.
717fn plane_pixel(index: usize, shape: [usize; 2]) -> [usize; 2] {
718    [index / shape[1], index % shape[1]]
719}
720
721fn model_cell(shape: [usize; 2], pixel: [usize; 2]) -> Option<ModelCell> {
722    Some(ModelCell::new(0, 0, 0, pixel)).filter(|_| pixel[0] < shape[0] && pixel[1] < shape[1])
723}
724
725fn canonical_flat(base: &ModelGeneration, cell: ModelCell) -> usize {
726    base.shape()
727        .flat_index(cell)
728        .expect("component pixels stay inside the base shape")
729}
730
731fn valid_support(base: &ModelGeneration, shape: [usize; 2], pixel: [usize; 2]) -> bool {
732    let Some(cell) = model_cell(shape, pixel) else {
733        return false;
734    };
735    base.samples()[canonical_flat(base, cell)].support() == ModelSupport::Valid
736}

738/// Find the maximum-abs real plane value passing `accept`, scanning in
739/// canonical storage order so ties deterministically keep the first peak.
740fn find_peak_abs<T>(
741    plane: &[T],
742    shape: [usize; 2],
743    magnitude: impl Fn(&T) -> f64,
744    accept: impl Fn([usize; 2]) -> bool,
745) -> Option<usize> {
746    let mut best: Option<(f64, usize)> = None;
747    for (index, value) in plane.iter().enumerate() {
748        let magnitude = magnitude(value).abs();
749        if best.is_some_and(|(best_magnitude, _)| magnitude <= best_magnitude) {
750            continue;
751        }
752        let pixel = plane_pixel(index, shape);
753        if accept(pixel) {
754            best = Some((magnitude, index));
755        }
756    }
757    best.map(|(_, index)| index)
758}

760/// Subtract `flux * psf` centered on `peak` from the working residual.
761fn subtract_psf(
762    residual: &mut [f64],
763    psf: &[num_complex::Complex64],
764    shape: [usize; 2],
765    peak: [usize; 2],
766    psf_peak: [usize; 2],
767    flux: f64,
768) -> Result<(), MinorCycleError> {
769    // psf_shifted(x, y) = psf(x - peak + psf_peak, y - peak + psf_peak),
770    // clipped to the plane exactly like the reference cleaner's subregion.
771    let x_range = overlap(peak[0], psf_peak[0], shape[0]);
772    let y_range = overlap(peak[1], psf_peak[1], shape[1]);
773    for y in y_range.clone() {
774        for x in x_range.clone() {
775            let source = [(x - peak[0] + psf_peak[0]), (y - peak[1] + psf_peak[1])];
776            let index = source[0] * shape[1] + source[1];
777            let target = x * shape[1] + y;
778            let updated = residual[target] - flux * psf[index].re;
779            if !updated.is_finite() {
780                return Err(MinorCycleError::GeneratedNonfinite);
781            }
782            residual[target] = updated;
783        }
784    }
785    Ok(())
786}
787
788/// Inclusive-clipped axis range where a PSF centered at `psf_peak` overlaps
789/// the plane when recentered on `peak`.
790fn overlap(peak: usize, psf_peak: usize, length: usize) -> std::ops::Range<usize> {
791    let shift = psf_peak as isize - peak as isize;
792    let low = (-shift).max(0) as usize;
793    let high_exclusive = (length as isize - shift)
794        .min(length as isize)
795        .max(low as isize) as usize;
796    low..high_exclusive
797}

799#[allow(clippy::too_many_arguments)]
800fn minor_cycle_evidence_id(
801    authority: LogicalIdentity,
802    attempt: ModelExecutionAttemptId,
803    epoch: u64,
804    input_generation: ModelGenerationId,
805    normal_state_completion: FinalNormalStateCompletionId,
806    normal_state_content: LogicalIdentity,
807    window: &CleanWindow,
808    controls: &HogbomControls,
809    iterations: usize,
810    total_flux: f64,
811    final_peak_flux: f64,
812    stop_reason: MinorCycleStopReason,
813) -> MinorCycleEvidenceId {
814    let mut encoder = Encoder::new(MINOR_CYCLE_EVIDENCE_DOMAIN, MINOR_CYCLE_EVIDENCE_VERSION);
815    encoder.identity(authority.as_bytes());
816    encoder.identity(attempt.identity().as_bytes());
817    encoder.u64(epoch);
818    encoder.identity(input_generation.as_bytes());
819    encoder.identity(normal_state_completion.as_bytes());
820    encoder.identity(normal_state_content.as_bytes());
821    encoder.usize(window.blc()[0]);
822    encoder.usize(window.blc()[1]);
823    encoder.usize(window.trc()[0]);
824    encoder.usize(window.trc()[1]);
825    encoder.u64(crate::canonical_f64_bits(controls.gain()));
826    encoder.u64(crate::canonical_f64_bits(controls.threshold()));
827    encoder.usize(controls.max_iterations());
828    encoder.u64(crate::canonical_f64_bits(controls.maximum_model_update()));
829    match controls.component_sequence_limit() {
830        None => encoder.u8(0),
831        Some(limit) => {
832            encoder.u8(1);
833            encoder.usize(limit);
834        }
835    }
836    encoder.usize(iterations);
837    encoder.u64(crate::canonical_f64_bits(total_flux));
838    encoder.u64(crate::canonical_f64_bits(final_peak_flux));
839    encoder.u8(match stop_reason {
840        MinorCycleStopReason::ThresholdReached => 0,
841        MinorCycleStopReason::IterationBound => 1,
842        MinorCycleStopReason::StalenessBound => 2,
843    });
844    MinorCycleEvidenceId(LogicalIdentity::from_sha256(encoder.finish()))
845}
