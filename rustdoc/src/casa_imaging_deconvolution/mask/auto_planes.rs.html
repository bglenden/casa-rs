3//! CASA's automatic multi-threshold mask over every plane of an image
4//! domain: each channel and polarization masked on its own, as tclean masks
5//! a cube's channels and an image's Stokes planes
6//! (`SDMaskHandler::autoMaskByMultiThreshold`'s loop over channels, then
7//! polarizations).
8//!
9//! tclean masks a cube block by block of channels, each block an image
10//! store of its own (`SynthesisDeconvolver::executeCubeMinorCycle` with
11//! `CubeMinorCycleAlgorithm`, blocks from `numblockchans` on one process),
12//! and a continuum image whole. A block shares
13//!
14//! * the beam its masks are smoothed and pruned by: the common beam of the
15//!   block's PSF beams (`CasaImageBeamSet::getCommonBeam`), each CASA's fit
16//!   of its plane's PSF. A plane with no data holds the largest beam of
17//!   the whole image domain (`SIImageStore::makeImageBeamSet`), even when that
18//!   beam belongs to a different channel block;
19//! * whether its robust noise ignores the mask: when the inherited mask
20//!   covers the block's whole first plane, every plane of the block takes
21//!   Chauvenet statistics (`calcRobustImageStatistics`).
22//!
23//! Every plane shares the image's PSF sidelobe level, the largest of its
24//! planes' (`SIImageStore::getPSFSidelobeLevel`, which the cube's
25//! controller hands each block), and the planes of a channel share its
26//! stop: `SDMaskHandler::skipChannels` reads the first polarization's
27//! threshold and counts the mask over every polarization of the channel,
28//! and a stopped channel skips all of them.
29//!
30//! A plane with no data has a zero residual in CASA: its statistics and
31//! thresholds are zero and its mask stays as it was.
32

47/// One plane of an image domain: the domain, the channel within the normal
48/// state (from its slab's first) and the polarization.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
50pub struct MaskPlaneKey {
51    /// The image domain's ordinal.
52    pub domain: usize,
53    /// The channel, counted from the normal state's first.
54    pub channel: usize,
55    /// The polarization plane.
56    pub polarization: usize,
57}

59/// One plane an automatic mask is made from, as the minor cycle reads it.
60#[derive(Debug, Clone, PartialEq)]
61pub struct AutoMaskPlane {
62    /// The residual in the units the minor cycle cleans, CASA's
63    /// `.residual` plane, x-major.
64    pub residual: Vec<f64>,
65    /// The valid pixels the residual's pixel mask keeps, x-major: its
66    /// statistics, its pruned regions and its mask read only these
67    /// (`SDMaskHandler::autoMask` attaches the residual's mask, the
68    /// `.pb`'s above `pblimit`, `SIImageStore::divideResidualByWeight`).
69    pub valid: Vec<bool>,
70    /// The pixels the plane's median and noise are measured on when they
71    /// are not `valid`, x-major. At a run's first boundary with `nsigma`,
72    /// tclean's automatic mask takes the median and noise
73    /// `SIImageStore::calcRobustRMS` measured where the `.pb` exceeds
74    /// `pbmask` (`SDMaskHandler::autoMask` reuses them while `iterdone` is
75    /// 0); its peak is still the pixel mask's.
76    pub statistics: Option<Vec<bool>>,
77}

79/// A plane's residual and valid pixels; `None` for a plane with no data.

82/// A channel-scoped reader for an automatic mask. It may share working
83/// data among the channel's polarizations and observe each completed mask.
84/// The reader is dropped when the channel finishes.
85pub trait AutoMaskChannel<E> {
86    /// Read one polarization's residual and valid pixels.
87    fn read(&mut self, key: MaskPlaneKey) -> Result<Option<AutoMaskPlane>, E>;
88
89    /// Observe the mask before the next polarization is read.
90    fn masked(
91        &mut self,
92        key: MaskPlaneKey,
93        plane: Option<&AutoMaskPlane>,
94        support: &[bool],
95    ) -> Result<(), E>;
96}

105/// What an automatic mask reads of one major cycle's planes, from the
106/// owner of the minor cycle: each plane as [`AutoMaskRead`] gives it, read
107/// once per boundary, and its PSF's measurements ([`AutoMaskPsf`]), which
108/// do not change; and how many channels it masks at once.
109pub struct AutoMaskSource<'a, E> {
110    /// Each plane's residual and valid pixels.
111    pub read: &'a AutoMaskRead<'a, E>,
112    /// Optional channel-scoped reader, used instead of `read` when present.
113    pub channel: Option<&'a AutoMaskOpenChannel<'a, E>>,
114    /// Each plane's PSF measurements.
115    pub psf: &'a AutoMaskPsf<'a>,
116    /// The channels masked at once: more than one masks a block's channels
117    /// side by side on the current rayon pool (the owner's worker team),
118    /// one masks them one after another.
119    pub workers: usize,
120}

122/// One image domain's automatic mask at one boundary: a row of CASA's
123/// automask summary for each plane, channel-major.
124#[derive(Debug, Clone, PartialEq)]
125pub struct AutoMaskEvidence {
126    polarizations: usize,
127    planes: Box<[AutoMultithreshEvidence]>,
128}
129
130impl AutoMaskEvidence {
131    /// The polarizations of each channel.
132    #[must_use]
133    pub const fn polarizations(&self) -> usize {
134        self.polarizations
135    }
136
137    /// Every plane's row, channel after channel.
138    #[must_use]
139    pub fn planes(&self) -> &[AutoMultithreshEvidence] {
140        &self.planes
141    }
142
143    /// The row of `channel` and `polarization`.
144    #[must_use]
145    pub fn plane(&self, channel: usize, polarization: usize) -> Option<&AutoMultithreshEvidence> {
146        (polarization < self.polarizations)
147            .then(|| self.planes.get(channel * self.polarizations + polarization))
148            .flatten()
149    }
150
151    /// The first plane's row.
152    #[must_use]
153    pub fn primary(&self) -> &AutoMultithreshEvidence {
154        &self.planes[0]
155    }
156
157    /// Whether each channel stops evolving at later boundaries.
158    pub fn stopped_channels(&self) -> impl ExactSizeIterator<Item = bool> + '_ {
159        self.planes
160            .chunks(self.polarizations)
161            .map(|channel| channel[0].channel_stopped)
162    }
163}

185/// A row before its plane is masked.
186const UNMASKED: AutoMultithreshEvidence = AutoMultithreshEvidence {
187    median: 0.0,
188    robust_rms: 0.0,
189    peak: 0.0,
190    positive_threshold: 0.0,
191    low_noise_threshold: 0.0,
192    negative_threshold: None,
193    threshold: AutoMaskThreshold::Sidelobe,
194    masked: false,
195    regions: 0,
196    pruned_regions: 0,
197    grow_regions: 0,
198    grow_pruned_regions: 0,
199    negative_pixels: 0,
200    changed_pixels: 0,
201    channel_stopped: false,
202};

204/// What every channel of one block is masked with.
205struct Block<'a, 'b, E> {
206    state: &'a AutoMaskState<'a>,
207    source: &'a AutoMaskSource<'b, E>,
208    domain: usize,
209    shape: [usize; 2],
210    polarizations: usize,
211    beam: Option<AutoMaskBeam>,
212    first_plane_full: bool,
213    /// An empty plane, the mask a plane inherits before the first boundary.
214    nothing: &'a [bool],
215}
216
217impl<E: From<MaskError>> Block<'_, '_, E> {
218    /// Mask `slots`' channel into its slots, and apply its stop
219    /// (`SDMaskHandler::skipChannels`), which every plane of the channel
220    /// shares: whether it added a positive (emission) region.
221    fn channel(&self, slots: ChannelSlots<'_>) -> Result<bool, E> {
222        let ChannelSlots {
223            channel,
224            support,
225            rows,
226        } = slots;
227        let state = self.state;
228        let inherited = |polarization| {
229            state
230                .previous
231                .and_then(|mask| mask.plane(channel, polarization))
232                .unwrap_or(self.nothing)
233        };
234        let cycle = AutoMaskCycle {
235            completed_major_cycles: state.completed_major_cycles,
236            threshold_reached: state.threshold_reached,
237            stopped: state
238                .stopped_channels
239                .get(channel)
240                .copied()
241                .unwrap_or(false),
242        };
243        let cells = self.shape[0] * self.shape[1];
244        let mut reader = self
245            .source
246            .channel
247            .map(|open| {
248                open(MaskPlaneKey {
249                    domain: self.domain,
250                    channel,
251                    polarization: 0,
252                })
253            })
254            .transpose()?;
255        let mut positive = false;
256        for (polarization, (out, row)) in support.chunks_mut(cells).zip(rows.iter_mut()).enumerate()
257        {
258            let key = MaskPlaneKey {
259                domain: self.domain,
260                channel,
261                polarization,
262            };
263            let plane = match &mut reader {
264                Some(reader) => reader.read(key)?,
265                None => (self.source.read)(key)?,
266            };
267            let made = plane_mask(&PlaneInputs {
268                plane: plane.as_ref(),
269                inherited: inherited(polarization),
270                shape: self.shape,
271                first_plane_full: self.first_plane_full,
272                cycle,
273                controls: state.controls,
274                beam: self.beam,
275            })?;
276            if let Some(reader) = &mut reader {
277                reader.masked(key, plane.as_ref(), &made.support)?;
278            }
279            out.copy_from_slice(&made.support);
280            *row = made.evidence;
281            positive |= made.positive;
282        }
283        let count = |support: &[bool]| support.iter().filter(|set| **set).count();
284        let before = (0..self.polarizations)
285            .map(|polarization| count(inherited(polarization)))
286            .sum();
287        let stopped = skip_channel(
288            before,
289            count(support),
290            rows[0].threshold,
291            cycle,
292            state.controls,
293        );
294        for row in rows.iter_mut() {
295            row.channel_stopped = stopped;
296        }
297        Ok(positive)
298    }
299}

301/// The automatic mask of every plane of `domain`, `shape` pixels by
302/// `planes` channels and polarizations, from the mask it inherits. Each
303/// channel writes its planes and rows straight into the domain's new mask
304/// and summary; a block's channels are masked side by side on the current
305/// rayon pool when `source` masks more than one at once.
306pub(super) fn auto_mask_domain<E>(
307    state: &AutoMaskState<'_>,
308    domain: usize,
309    shape: [usize; 2],
310    planes: MaskPlanes,
311    source: &AutoMaskSource<'_, E>,
312) -> Result<(ReconstructionMask, AutoMaskEvidence), E>
313where
314    E: From<MaskError> + Send,
315{
316    validate_auto_controls(state.controls)?;
317    let cells = shape[0] * shape[1];
318    if state.previous.is_some_and(|mask| {
319        mask.shape != shape || mask.coordinate != state.coordinate || mask.planes != planes
320    }) {
321        return Err(MaskError::ShapeMismatch.into());
322    }
323    let polarizations = planes.polarizations();
324    // The materialization charge covers exactly one entry per plane.
325    let mut psfs = Vec::with_capacity(planes.count());
326    for channel in 0..planes.channels() {
327        for polarization in 0..polarizations {
328            psfs.push((source.psf)(MaskPlaneKey {
329                domain,
330                channel,
331                polarization,
332            }));
333        }
334    }
335    let sidelobe = psfs
336        .iter()
337        .flatten()
338        .map(PsfSummary::sidelobe)
339        .fold(0.0_f64, f64::max);
340    let nothing = vec![false; cells];
341    // One allocation each, written in place: the support is shared as it
342    // is by the mask (a trusted-length iterator collects into the `Arc`
343    // without a staging copy).
344    let mut support = (0..planes.count() * cells)
345        .map(|_| false)
346        .collect::<Arc<[bool]>>();
347    let mut rows = vec![UNMASKED; planes.count()];
348    let slots = Arc::get_mut(&mut support).expect("a support no one else holds yet");
349    let per_channel = polarizations * cells;
350    for (block, beam) in block_beams(&psfs, polarizations, shape, sidelobe, state.coordinate) {
351        let masking = Block {
352            state,
353            source,
354            domain,
355            shape,
356            polarizations,
357            beam: beam?,
358            first_plane_full: state
359                .previous
360                .and_then(|mask| mask.plane(block.start, 0))
361                .is_some_and(|plane| plane.iter().all(|set| *set)),
362            nothing: &nothing,
363        };
364        let block_support = &mut slots[block.start * per_channel..block.end * per_channel];
365        let block_rows = &mut rows[block.start * polarizations..block.end * polarizations];
366        let positive = if source.workers > 1 {
367            block_support
368                .par_chunks_mut(per_channel)
369                .zip(block_rows.par_chunks_mut(polarizations))
370                .zip(block.clone().into_par_iter())
371                .map(|((support, rows), channel)| {
372                    masking.channel(ChannelSlots {
373                        channel,
374                        support,
375                        rows,
376                    })
377                })
378                .try_reduce(|| false, |left, right| Ok(left || right))?
379        } else {
380            let mut positive = false;
381            for ((support, rows), channel) in block_support
382                .chunks_mut(per_channel)
383                .zip(block_rows.chunks_mut(polarizations))
384                .zip(block.clone())
385            {
386                positive |= masking.channel(ChannelSlots {
387                    channel,
388                    support,
389                    rows,
390                })?;
391            }
392            positive
393        };
394        report_block(
395            state,
396            block.start,
397            &rows[block.start * polarizations..block.end * polarizations],
398            positive,
399            polarizations,
400        );
401    }
402    Ok((
403        ReconstructionMask::from_plane_support(state.coordinate, shape, planes, support)?,
404        AutoMaskEvidence {
405            polarizations,
406            planes: rows.into_boxed_slice(),
407        },
408    ))
409}

411/// The blocks of channels tclean masks a cube in on one process
412/// (`SynthesisDeconvolver::numblockchans`): blocks of `8e6` pixels divided
413/// by the plane's, at least ten channels, or the whole cube when it has no
414/// more channels than that; the last block holds what remains.
415pub(super) fn channel_blocks(channels: usize, shape: [usize; 2]) -> Vec<Range<usize>> {
416    let size = block_size(channels, shape);
417    (0..channels)
418        .step_by(size)
419        .map(|start| start..(start + size).min(channels))
420        .collect()
421}
422
423/// The largest block, in channels, CASA forms for an image of `shape`.
424fn block_size(channels: usize, shape: [usize; 2]) -> usize {
425    let optimum = ((8.0e6 / shape[0] as f64 / shape[1] as f64) as usize).max(10);
426    channels.min(optimum).max(1)
427}

429/// The peak three simultaneously held beam arrays for one channel block:
430/// fitted beams, their numerical forms and the common-beam working copy.
431pub(super) fn block_beam_bytes(channels: usize, polarizations: usize, shape: [usize; 2]) -> u64 {
432    let planes = channels.min(block_size(channels, shape)) * polarizations;
433    let blocks = channels.div_ceil(block_size(channels, shape));
434    (planes * (size_of::<RestoringBeam>() + 2 * size_of::<casa_numerics::EllipticalGaussian>())
435        + blocks * size_of::<Range<usize>>()) as u64
436}

438/// Plan each channel block's beam from the complete image domain's beam set.
439fn block_beams(
440    psfs: &[Option<PsfSummary>],
441    polarizations: usize,
442    shape: [usize; 2],
443    sidelobe: f64,
444    coordinate: DirectionCoordinateSpec,
445) -> impl Iterator<Item = (Range<usize>, Result<Option<AutoMaskBeam>, MaskError>)> + '_ {
446    let largest = largest_beam(psfs);
447    channel_blocks(psfs.len() / polarizations, shape)
448        .into_iter()
449        .map(move |block| {
450            let beam = block_beam(
451                &psfs[block.start * polarizations..block.end * polarizations],
452                polarizations,
453                largest,
454                sidelobe,
455                coordinate,
456            );
457            (block, beam)
458        })
459}

461/// The largest fitted beam of the image domain, which CASA puts into each
462/// blank plane before it forms channel-block image stores.
463fn largest_beam(psfs: &[Option<PsfSummary>]) -> Option<RestoringBeam> {
464    psfs.iter().flatten().map(PsfSummary::beam).fold(
465        None,
466        |largest: Option<RestoringBeam>, beam| match largest {
467            Some(current) if current.area_sr() >= beam.area_sr() => Some(current),
468            _ => Some(beam),
469        },
470    )
471}

473/// The beam a block's masks are smoothed and pruned by: the common beam of
474/// its PSF beams, with blank planes holding the domain's largest. `None`
475/// means the whole domain has no fitted beam. The common beam is found in
476/// world coordinates, then its area and smoothing use both pixel increments.
477fn block_beam(
478    psfs: &[Option<PsfSummary>],
479    polarizations: usize,
480    largest: Option<RestoringBeam>,
481    sidelobe: f64,
482    coordinate: DirectionCoordinateSpec,
483) -> Result<Option<AutoMaskBeam>, MaskError> {
484    let cell = coordinate.increment_rad()[0].abs();
485    let mut beams = Vec::with_capacity(psfs.len());
486    // CASA iterates its channel-by-polarization beam matrix with channel
487    // varying fastest, even though our plane list is channel-major.
488    for polarization in 0..polarizations {
489        for channel in 0..psfs.len() / polarizations {
490            let summary = psfs[channel * polarizations + polarization];
491            if let Some(beam) = summary.map(|summary| summary.beam()).or(largest) {
492                beams.push(
493                    RestoringBeam::new(
494                        beam.major_fwhm_rad() * cell,
495                        beam.minor_fwhm_rad() * cell,
496                        beam.position_angle_rad(),
497                    )
498                    .map_err(|_| MaskError::InvalidBeamArea)?,
499                );
500            }
501        }
502    }
503    if beams.is_empty() {
504        return Ok(None);
505    }
506    let common = RestoringBeam::common_enclosing(&beams).map_err(|_| MaskError::InvalidBeamArea)?;
507    let beam = AutoMaskBeam::new(
508        common.major_fwhm_rad() / cell,
509        common.minor_fwhm_rad() / cell,
510        common.position_angle_rad(),
511        sidelobe,
512    )
513    .with_cell_scale([1.0, (coordinate.increment_rad()[1] / cell).abs()]);
514    validate_beam(&beam)?;
515    Ok(Some(beam))
516}

518/// What tclean logs of one block's mask, its `rows` channel-major from
519/// channel `first`: at the first boundary its warnings
520/// (`autoMaskByMultiThreshold`, `noMaskCheck`) when it added no positive
521/// region, and with `verbose` CASA's automask summary, a row per channel
522/// (`printAutomaskSummary`).
523fn report_block(
524    state: &AutoMaskState<'_>,
525    first: usize,
526    rows: &[AutoMultithreshEvidence],
527    positive: bool,
528    polarizations: usize,
529) {
530    let channels = rows.len() / polarizations;
531    if state.completed_major_cycles == 0 {
532        if state.controls.cut_threshold >= 0.2 {
533            tracing::warn!(
534                "Faint regions may not be included in the final mask. Consider decreasing \
535                 cutthreshold."
536            );
537        }
538        if !positive {
539            warn_no_regions(rows);
540            let pruned = rows
541                .iter()
542                .filter(|row| row.regions > 0 && row.pruned_regions == row.regions)
543                .count();
544            if pruned > 0 {
545                tracing::warn!(
546                    "{pruned} of {channels} channels had all regions removed by pruning. Try \
547                     decreasing minbeamfrac to remove fewer regions"
548                );
549            }
550        }
551    }
552    if state.controls.verbose {
553        log_summary(first, rows, polarizations);
554    }
555}

557/// `SDMaskHandler::noMaskCheck` of a block whose positive mask is empty.
558fn warn_no_regions(rows: &[AutoMultithreshEvidence]) {
559    tracing::warn!("No regions found by automasking");
560    let [first, ..] = rows else {
561        return;
562    };
563    if rows.len() == 1 {
564        let kind = first.threshold.name();
565        tracing::warn!(
566            "Used {kind} threshold to mask, but no emission was found.Try decreasing your \
567             {kind}threshold parameter if you want to capture emission in these channels."
568        );
569        return;
570    }
571    for kind in [AutoMaskThreshold::Sidelobe, AutoMaskThreshold::Noise] {
572        let count = rows.iter().filter(|row| row.threshold == kind).count();
573        if count > 0 {
574            let name = kind.name();
575            tracing::warn!(
576                "{count} of {} channels used the {name} threshold to mask, but no emission was \
577                 found. Try decreasing your {name}threshold parameter if you want to capture \
578                 emission in these channels.",
579                rows.len()
580            );
581        }
582    }
583}

585/// CASA's `printAutomaskSummary` of one block's `rows` from channel
586/// `first_channel`, through `tracing`: a row per channel, each value a list
587/// over its polarizations when it has several.
588fn log_summary(first_channel: usize, rows: &[AutoMultithreshEvidence], polarizations: usize) {
589    let list = |values: Vec<String>| {
590        if polarizations == 1 {
591            values.into_iter().next().unwrap_or_default()
592        } else {
593            format!("[{}]", values.join(", "))
594        }
595    };
596    tracing::info!("========== automask summary ==========");
597    tracing::info!(
598        "chan masking? median   RMS         peak   thresh_type   thresh_value   N_reg N_pruned \
599         N_grow N_grow_pruned N_neg_pix"
600    );
601    for (offset, rows) in rows.chunks(polarizations).enumerate() {
602        let column =
603            |value: fn(&AutoMultithreshEvidence) -> String| list(rows.iter().map(value).collect());
604        // A stopped channel was skipped before this boundary: CASA prints
605        // its median, RMS and threshold as `--`.
606        let masked = rows[0].masked;
607        let skipped = |text: String| if masked { text } else { "--".to_string() };
608        tracing::info!(
609            "[C{}] {}        {}  {}  {}  {}  {}        {}  {}  {}  {}  {}",
610            first_channel + offset,
611            if masked { "T" } else { "F" },
612            skipped(column(|row| row.median.to_string())),
613            skipped(column(|row| row.robust_rms.to_string())),
614            column(|row| row.peak.to_string()),
615            column(|row| row.threshold.name().to_string()),
616            skipped(column(|row| row.positive_threshold.to_string())),
617            column(|row| row.regions.to_string()),
618            column(|row| row.pruned_regions.to_string()),
619            column(|row| row.grow_regions.to_string()),
620            column(|row| row.grow_pruned_regions.to_string()),
621            column(|row| row.negative_pixels.to_string()),
622        );
623    }
624    tracing::info!("========== END of automask summary ==========");
625}

673                let [x, y] = shape.pixel(index);

675                (-r2 / 8.0).exp()
676            })
677            .collect::<Vec<_>>();
678        let narrow = PsfSummary::new(&psf, shape).expect("narrow beam");
679        let broad = narrow.with_domain_beam(

706        let without_blank = block_beam(&psfs[2..10], 1, largest, 0.1, coordinate).unwrap();

898    use casa_imaging_model::{DirectionFrame, NoiseEstimate, Projection, SkyDirection};
899    use std::sync::atomic::{AtomicUsize, Ordering};

901    struct Reader<'a> {
902        channel: usize,
903        live: &'a AtomicUsize,
904        reads: &'a AtomicUsize,
905        masked: &'a AtomicUsize,
906    }
907
908    impl Drop for Reader<'_> {
909        fn drop(&mut self) {
910            self.live.fetch_sub(1, Ordering::SeqCst);
911        }
912    }
913
914    impl AutoMaskChannel<MaskError> for Reader<'_> {
915        fn read(&mut self, key: MaskPlaneKey) -> Result<Option<AutoMaskPlane>, MaskError> {
916            assert_eq!(key.channel, self.channel);
917            self.reads.fetch_add(1, Ordering::SeqCst);
918            Ok(Some(AutoMaskPlane {
919                residual: vec![0.0; 64],
920                valid: vec![true; 64],
921                statistics: None,
922            }))
923        }
924
925        fn masked(
926            &mut self,
927            key: MaskPlaneKey,
928            _: Option<&AutoMaskPlane>,
929            support: &[bool],
930        ) -> Result<(), MaskError> {
931            assert_eq!(key.channel, self.channel);
932            assert_eq!(support.len(), 64);
933            self.masked.fetch_add(1, Ordering::SeqCst);
934            Ok(())
935        }
936    }

938    /// Channel work, including all four polarizations, owns at most one
939    /// reader per worker, regardless of the cube's channel count.
940    #[test]
941    fn automatic_mask_scopes_readers_to_channels_in_flight() {
942        const CHANNELS: usize = 21;
943        const WORKERS: usize = 4;
944        let live = AtomicUsize::new(0);
945        let peak = AtomicUsize::new(0);
946        let opened = AtomicUsize::new(0);
947        let reads = AtomicUsize::new(0);
948        let masked = AtomicUsize::new(0);
949        let open =
950            |key: MaskPlaneKey| -> Result<Box<dyn AutoMaskChannel<MaskError> + '_>, MaskError> {
951                opened.fetch_add(1, Ordering::SeqCst);
952                let current = live.fetch_add(1, Ordering::SeqCst) + 1;
953                peak.fetch_max(current, Ordering::SeqCst);
954                Ok(Box::new(Reader {
955                    channel: key.channel,
956                    live: &live,
957                    reads: &reads,
958                    masked: &masked,
959                }))
960            };
961        let legacy = |_: MaskPlaneKey| -> Result<Option<AutoMaskPlane>, MaskError> {
962            panic!("channel reader is used")
963        };
964        let mut values = vec![0.0; 64];
965        values[36] = 1.0;
966        let summary = PsfSummary::with_saved_beam(
967            &values,
968            crate::PlaneShape::new(8, 8),
969            RestoringBeam::new(2.0, 2.0, 0.0).unwrap(),
970        )
971        .unwrap();
972        let psf = |_: MaskPlaneKey| Some(summary);
973        let source = AutoMaskSource {
974            read: &legacy,
975            channel: Some(&open),
976            psf: &psf,
977            workers: WORKERS,
978        };
979        let coordinate = DirectionCoordinateSpec::new(
980            Projection::Sin,
981            SkyDirection::new(DirectionFrame::J2000, 1.0, -0.5),
982            [4.0, 4.0],
983            [-1.0e-4, 1.0e-4],
984            [[1.0, 0.0], [0.0, 1.0]],
985            [180.0, 90.0],
986        );
987        let controls = AutoMultithreshControls {
988            sidelobe_factor: 2.0,
989            noise_factor: 4.25,
990            low_noise_factor: 1.5,
991            negative_factor: 0.0,
992            minimum_beam_fraction: 0.0,
993            smooth_factor: 1.0,
994            cut_threshold: 0.01,
995            grow_iterations: 0,
996            grow_prune: false,
997            minimum_percent_change: -1.0,

1001        let state = AutoMaskState {

1006            previous: None,
1007            stopped_channels: &[],

1009        rayon::ThreadPoolBuilder::new()
1010            .num_threads(WORKERS)

1012            .unwrap()

1020                )
1021                .unwrap();
1022            });
1023        assert_eq!(opened.load(Ordering::SeqCst), CHANNELS);
1024        assert_eq!(reads.load(Ordering::SeqCst), CHANNELS * 4);
1025        assert_eq!(masked.load(Ordering::SeqCst), CHANNELS * 4);
