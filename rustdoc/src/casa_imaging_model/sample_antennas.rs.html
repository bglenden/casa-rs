8/// Per-antenna pointing directions evaluated for one baseline sample.
9///
10/// Retaining both directions preserves antenna-dependent boresight offsets. A downstream
11/// operator may derive a baseline-effective direction according to its own closed response law.
12#[derive(Debug, Clone, Copy, PartialEq)]
13pub struct SelectedPointingDirections {
14    /// Evaluated direction for MAIN `ANTENNA1`.
15    pub antenna1: SkyDirection,
16    /// Evaluated direction for MAIN `ANTENNA2`.
17    pub antenna2: SkyDirection,
18}

20/// CASA aperture class selected from owner-controlled ANTENNA metadata.
21///
22/// These are scientific response identities rather than physical antenna IDs.
23/// Multiple antennas with the same class therefore share one cached response.
24#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
25pub enum AntennaResponseClass {
26    /// Physical 12 m ALMA dish evaluated with CASA's 10.7 m effective aperture.
27    CasaAlma12m,
28    /// Physical 7 m ACA dish evaluated with CASA's 6.25 m effective aperture.
29    CasaAca7m,
30    /// Every antenna of an EVLA MeasurementSet, whatever its dish: CASA's
31    /// common EVLA beam (`PBMath1DEVLA`), which
32    /// `HetArrayConvFunc::findAntennaSizes` gives the whole array when
33    /// `SynthesisImagerVi2::createMosFTMachine` builds it for
34    /// `PBMath::EVLA` (`PBMathInterface::COMMONPB`).
35    CasaEvla,
36    /// A dish of a telescope CASA's voltage-pattern manager has no beam
37    /// for ([`Self::casa_airy_telescope`]): the Airy disk of the dish itself
38    /// that `HetArrayConvFunc::findAntennaSizes` forms for
39    /// `PBMathInterface::AIRY`, one per dish diameter.
40    ///
41    /// The class is the exact diameter. CASA keys a dish by its diameter's
42    /// text, `String::toString(Double)` to six significant digits, and
43    /// gives the class the diameter of the first antenna with that text, so
44    /// casa-rs may split dishes CASA merges (diameters that differ beyond
45    /// the sixth digit, whose beams then differ by less than a part in
46    /// 10⁵), but never merges dishes CASA splits.
47    CasaAiry {
48        /// The bits of the ANTENNA table's `DISH_DIAMETER` in metres, a
49        /// positive finite `f64`, whose bit order is its numeric order.
50        dish_diameter_m_bits: u64,
51    },
52}
53
54impl AntennaResponseClass {
55    /// Whether CASA treats `telescope` as ALMA or ACA, whose dishes it
56    /// classes by [`Self::casa_alma_aca`].
57    #[must_use]
58    pub fn casa_alma_aca_telescope(telescope: &str) -> bool {
59        matches!(telescope.trim(), "ALMA" | "ACA")
60    }
61
62    /// Whether CASA's mosaic gives `telescope` the common EVLA beam
63    /// ([`Self::CasaEvla`]): `PBMath::enumerateCommonPB` names it
64    /// `EVLA`, exactly.
65    #[must_use]
66    pub fn casa_evla_telescope(telescope: &str) -> bool {
67        telescope.trim() == "EVLA"
68    }
69
70    /// Whether CASA's mosaic forms an Airy disk of each dish of
71    /// `telescope` ([`Self::CasaAiry`]): `SynthesisImager::getVPRecord`
72    /// finds no default voltage pattern under the name (`VPManager` lists
73    /// those of `PBMath::nameCommonPB` and the observatories with antenna
74    /// responses, and matches the name exactly), so it leaves
75    /// `PBMath::UNKNOWN`, for which `SynthesisImagerVi2::createMosFTMachine`
76    /// gives `HetArrayConvFunc` its `PBMathInterface::AIRY` class.
77    /// `NGVLA1`, an ngVLA simulation's name, is one; `NGVLA`, whose common
78    /// beam the mosaic applies instead, is not. Nor are `UNKNOWN` and
79    /// `OSF`, which have default records but no common beam the mosaic
80    /// can form.
81    ///
82    /// `BIMA` and `NRAO140FT` are Airy telescopes too, although
83    /// `PBMath::enumerateCommonPB` knows them (as `HATCREEK` and
84    /// `NRAO140FT`): `VPManager` has no default under either name, so
85    /// `getVPRecord` resets its `PBMath::UNKNOWN` for the empty record
86    /// (SynthesisImager.cc, `rec.empty()`). CASA 6.7.6's tclean of
87    /// refim_sim_hetarray_ngvla.ms renamed to either logs "vpmanager does
88    /// not have a beam for antenna" and "Overriding PB with Airy of
89    /// diam,blockage=18 m,1.125 m"; renamed to `HATCREEK`, neither.
90    ///
91    /// The name is matched as given. casa-rs's compiler gives it trimmed
92    /// of surrounding blanks, as every telescope name it reads; CASA
93    /// matches the stored name as it is, so a padded `NGVLA ` is an Airy
94    /// telescope to CASA and the common ngVLA beam's name, refused, to
95    /// casa-rs.
96    #[must_use]
97    pub fn casa_airy_telescope(telescope: &str) -> bool {
98        !CASA_DEFAULT_VOLTAGE_PATTERNS.contains(&telescope)
99    }
100
101    /// CASA's class of a dish `diameter_m` across of a telescope without a
102    /// default voltage pattern ([`Self::casa_airy_telescope`]); `None` for a
103    /// diameter that is not positive and finite (`findAntennaSizes` gives
104    /// no dish to one that is not positive).
105    #[must_use]
106    pub fn casa_airy(diameter_m: f64) -> Option<Self> {
107        (diameter_m.is_finite() && diameter_m > 0.0).then_some(Self::CasaAiry {
108            dish_diameter_m_bits: diameter_m.to_bits(),
109        })
110    }
111
112    /// The dish diameter of a [`Self::CasaAiry`] class, in metres; `None`
113    /// for the other classes, whose apertures CASA fixes.
114    #[must_use]
115    pub fn airy_dish_diameter_m(self) -> Option<f64> {
116        match self {
117            Self::CasaAiry {
118                dish_diameter_m_bits,
119            } => Some(f64::from_bits(dish_diameter_m_bits)),
120            Self::CasaAlma12m | Self::CasaAca7m | Self::CasaEvla => None,
121        }
122    }
123
124    /// CASA's class of an ALMA or ACA dish `diameter_m` across: 12 m
125    /// within 0.5 m, 7 m within 1 m; `None` for another dish.
126    #[must_use]
127    pub fn casa_alma_aca(diameter_m: f64) -> Option<Self> {
128        if (diameter_m - 12.0).abs() < 0.5 {
129            Some(Self::CasaAlma12m)
130        } else if (diameter_m - 7.0).abs() < 1.0 {
131            Some(Self::CasaAca7m)
132        } else {
133            None
134        }
135    }
136
137    /// The largest response of an antenna family whose dishes are
138    /// `classes`: the family envelope CASA's heterogeneous mosaic sizes its
139    /// kernels by; of Airy dishes, the widest.
140    #[must_use]
141    pub fn family_envelope(classes: &[Option<Self>]) -> Option<Self> {
142        [Self::CasaAlma12m, Self::CasaAca7m, Self::CasaEvla]
143            .into_iter()
144            .find(|class| classes.contains(&Some(*class)))
145            .or_else(|| {
146                classes
147                    .iter()
148                    .flatten()
149                    .copied()
150                    .filter(|class| class.airy_dish_diameter_m().is_some())
151                    .max()
152            })
153    }
154}

208/// Ordered response classes for one selected interferometric baseline.
209#[derive(Debug, Clone, Copy, PartialEq, Eq)]
210pub struct SelectedAntennaResponses {
211    /// Response class of MAIN `ANTENNA1`.
212    pub antenna1: AntennaResponseClass,
213    /// Response class of MAIN `ANTENNA2`.
214    pub antenna2: AntennaResponseClass,
215    /// Largest response class present in this MeasurementSet response family.
216    ///
217    /// CASA crops every heterogeneous convolution plane to one family-wide
218    /// extent so resampling has the same boundary domain for every pair.
219    pub family_envelope: AntennaResponseClass,
220}

222#[cfg(test)]
223mod tests {
224    use super::*;
225
226    /// `HetArrayConvFunc::findAntennaSizes` gives every positive dish
227    /// diameter its own Airy disk and none to a dish that is not positive;
228    /// the widest is the family's envelope.
229    #[test]
230    fn airy_dishes_are_classed_by_their_diameter() {
231        let [eighteen, six] = [18.0, 6.0]
232            .map(|diameter| AntennaResponseClass::casa_airy(diameter).expect("a positive dish"));
233        assert_eq!(
234            eighteen,
235            AntennaResponseClass::CasaAiry {
236                dish_diameter_m_bits: 18.0_f64.to_bits()
237            }
238        );
239        // Diameters CASA's six-digit text merges stay apart.
240        assert_ne!(
241            AntennaResponseClass::casa_airy(18.000_000_1),
242            Some(eighteen)
243        );
244        assert_eq!(eighteen.airy_dish_diameter_m(), Some(18.0));
245        assert_eq!(six.airy_dish_diameter_m(), Some(6.0));
246        assert_eq!(
247            AntennaResponseClass::CasaAlma12m.airy_dish_diameter_m(),
248            None
249        );
250        for absent in [0.0, -6.0, f64::NAN, f64::INFINITY] {
251            assert_eq!(AntennaResponseClass::casa_airy(absent), None, "{absent}");
252        }
253        assert_eq!(
254            AntennaResponseClass::family_envelope(&[Some(six), None, Some(eighteen), Some(six)]),
255            Some(eighteen)
256        );
257        for airy in ["NGVLA1", "BIMA", "NRAO140FT"] {
258            assert!(AntennaResponseClass::casa_airy_telescope(airy), "{airy}");
259        }
260        for common in ["NGVLA", "HATCREEK", "ALMA", "UNKNOWN", "OSF"] {
261            assert!(
262                !AntennaResponseClass::casa_airy_telescope(common),
263                "{common}"
264            );
265        }
266    }
