2//! One partition read as one batch per selected data group for each range
3//! of times: a traversal per group, each sized to the same whole times per
4//! batch, the first with `first` roles and every other with its

14/// A further group's roles: its visibilities, beside the first group's
15/// flags, weights, UVW and time auxiliaries.
16const VISIBILITIES: Roles = Roles {
17    values: true,
18    flags: false,
19    weights: false,
20    uvw: false,
21    time_centroid: false,
22    interval: false,
23    exposure: false,
24    scan: false,
25};

27/// The weights alone, whose non-finite values are padding's one mark.
28pub(crate) const WEIGHTS: Roles = Roles {
29    values: false,
30    flags: false,
31    weights: true,
32    uvw: false,
33    time_centroid: false,
34    interval: false,
35    exposure: false,
36    scan: false,
37};

39/// What a lockstep read delivered.
40#[derive(Clone, Copy, Debug, Default)]
41pub(crate) struct Read {
42    /// Batches of each group.
43    pub(crate) batches: usize,
44    /// Samples of each group.
45    pub(crate) samples: usize,
46}
47
48/// The reads of one partition in lockstep: a request per group, and what
49/// each holds per time.
50pub(crate) struct Sizing {
51    requests: Vec<AccessRequest>,
52    per_time: Vec<usize>,
53    /// The partition's times.
54    pub(crate) times: usize,
55    /// The source's natural time block.
56    natural: usize,
57    /// The samples (baselines) of one time.
58    pub(crate) samples_per_time: usize,
59}
60
61impl Sizing {
62    /// The reads of partition `id` of `observation`, every channel and
63    /// product of each of `groups`, the first with `first` roles and every
64    /// other with its visibilities only.
65    pub(crate) fn of(
66        observation: &dyn Observation,
67        id: PartitionId,
68        groups: &[String],
69        first: Roles,
70    ) -> Result<Self, ObservationError> {
71        let entry = observation.catalogue().partition(id);
72        let channels = entry.frequencies_hz.len();
73        let requests: Vec<AccessRequest> = groups
74            .iter()
75            .enumerate()
76            .map(|(index, group)| AccessRequest {
77                partitions: vec![id],
78                data_group: group.clone(),
79                roles: if index == 0 { first } else { VISIBILITIES },
80                channels: 0..channels,
81                times: TimeSelection::All,
82                fields: casa_observation::FieldSelection::All,
83                weights: WeightRequest::Stored,
84                correlations: CorrelationRequest::ALL,
85                grouping: Grouping::None,
86                batch_bytes: usize::MAX,
87            })
88            .collect();
89        let mut sizing = Self {
90            per_time: Vec::with_capacity(requests.len()),
91            requests: Vec::new(),
92            times: 0,
93            natural: 1,
94            samples_per_time: entry.baselines.len(),
95        };
96        for request in &requests {
97            let estimate = &observation.estimate(request)?.partitions[0];
98            sizing.per_time.push(estimate.held_bytes_per_time());
99            sizing.times = estimate.times;
100            sizing.natural = estimate.natural_blocks.times;
101        }
102        sizing.requests = requests;
103        Ok(sizing)
104    }
105
106    /// One integration: what one time of every group's batches holds.
107    pub(crate) fn integration(&self) -> usize {
108        self.per_time.iter().sum()
109    }
110
111    /// Whole times per batch within `budget`, a multiple of `chunk` times
112    /// where that fits (with `None` the source's natural block).
113    pub(crate) fn times_per_batch(&self, budget: usize, chunk: Option<usize>) -> usize {
114        let chunk = chunk.unwrap_or(self.natural).max(1);
115        times_per_batch(budget, self.integration(), chunk, self.times)
116    }
117}

119/// Read partition `id` of `observation`, every channel and product of
120/// each of `groups`, handing `each` one batch per group for each range of
121/// times, in time order. The batches hold whole times, `budget` bytes of
122/// them for all groups together, and a multiple of `chunk` times where
123/// that fits: the destination's chunk, or with `None` the source's natural
124/// time block, so that no unit is decoded twice.
125pub(crate) fn read(
126    observation: &dyn Observation,
127    id: PartitionId,
128    groups: &[String],
129    first: Roles,
130    (budget, chunk): (usize, Option<usize>),
131    mut each: impl FnMut(&[VisibilityBatch<'_>]) -> Result<(), ObservationError>,
132) -> Result<Read, ObservationError> {
133    let mut sizing = Sizing::of(observation, id, groups, first)?;
134    let times = sizing.times_per_batch(budget, chunk);
135    let mut requests = std::mem::take(&mut sizing.requests);
136    for (request, bytes) in requests.iter_mut().zip(&sizing.per_time) {
137        request.batch_bytes = bytes * times;
138    }
139    let mut traversals = requests
140        .iter()
141        .map(|request| observation.bind_read(request))
142        .collect::<Result<Vec<_>, _>>()?;
143    let mut buffers: Vec<BatchBuffers> = groups.iter().map(|_| BatchBuffers::default()).collect();
144    let mut read = Read::default();
145    loop {
146        let mut lockstep = Vec::with_capacity(groups.len());
147        let mut ended = 0;
148        for (traversal, buffer) in traversals.iter_mut().zip(buffers.iter_mut()) {
149            match traversal.next(buffer)? {
150                Some(ReadEvent::Batch(batch)) => lockstep.push(batch),
151                Some(ReadEvent::GroupEnd(_)) => {
152                    return Err(request_error(
153                        "a conversion traversal delivered a group end",
154                    ));
155                }
156                None => ended += 1,
157            }
158        }
159        if ended == groups.len() {
160            return Ok(read);
161        }
162        if ended > 0 {
163            return Err(request_error("the data groups' traversals ended apart"));
164        }
165        read.batches += 1;
166        read.samples += lockstep[0].samples();
167        each(&lockstep)?;
168    }
169}

171/// Whole times per batch for `budget` bytes of `bytes_per_time` each: a
172/// multiple of `chunk` times where that fits, so that no chunk is held,
173/// and never more than the partition's `total`. A budget below one time
174/// still reads one (a plan refuses it first: `CreationPlan::new`).
175fn times_per_batch(budget: usize, bytes_per_time: usize, chunk: usize, total: usize) -> usize {
176    let fit = (budget / bytes_per_time.max(1)).max(1);
177    let aligned = if fit >= chunk { fit - fit % chunk } else { fit };
178    aligned.min(total.max(1))
179}

181#[cfg(test)]
182mod tests {
183    use super::times_per_batch;
184
185    #[test]
186    fn batches_hold_whole_chunks_of_times_where_the_budget_allows() {
187        assert_eq!(times_per_batch(1000, 10, 5, 100), 100, "the partition fits");
188        assert_eq!(times_per_batch(470, 10, 5, 100), 45, "whole chunks");
189        assert_eq!(times_per_batch(30, 10, 5, 100), 3, "less than a chunk");
190        assert_eq!(times_per_batch(5, 10, 5, 100), 1, "at least one time");
191    }
