22pub(super) fn run_evalexpr(
23    request: &ImmathRequest,
24    outfile: &Path,
25    inputs: &ResolvedInputs,
26    varnames: &[String],
27    expr: &str,
28    clear_units: bool,
29    prec: ImagePixelType,
30) -> Result<ImageMathSummary, ImageError> {
31    match prec {
32        ImagePixelType::Float32 => {
33            run_evalexpr_typed::<f32>(request, outfile, inputs, varnames, expr, clear_units)
34        }
35        ImagePixelType::Float64 => {
36            run_evalexpr_typed::<f64>(request, outfile, inputs, varnames, expr, clear_units)
37        }
38        _ => Err(ImageError::InvalidMetadata(
39            "Unsupported value for type, it must be 'float' or 'double'.".into(),
40        )),
41    }
42}

44fn run_evalexpr_typed<T: RealMathPixel>(
45    request: &ImmathRequest,
46    outfile: &Path,
47    inputs: &ResolvedInputs,
48    varnames: &[String],
49    expr: &str,
50    clear_units: bool,
51) -> Result<ImageMathSummary, ImageError> {
52    let imagename_paths = &inputs.imagename_paths;
53    let extra_quoted = &inputs.extra_quoted;
54    let all_paths: Vec<PathBuf> = imagename_paths
55        .iter()
56        .chain(extra_quoted.iter())
57        .cloned()
58        .collect();
59    let rewritten_expr = quote_var_aliases(expr, varnames);
60    let mut names = referenced_images(&rewritten_expr).map_err(expression_error)?;
61    if let Some(mask) = request
62        .mask
63        .as_deref()
64        .filter(|value| !value.trim().is_empty())
65    {
66        names.extend(
67            referenced_images(&quote_var_aliases(mask, varnames)).map_err(expression_error)?,
68        );
69    }
70    let mut raws: Vec<(usize, RawImage<T>)> = Vec::with_capacity(all_paths.len());
71    for (index, path) in all_paths.iter().enumerate() {
72        if names
73            .iter()
74            .any(|name| Path::new(name) == path || varnames.get(index) == Some(name))
75        {
76            raws.push((index, open_raw_image::<T>(path)?));
77        }
78    }
79
80    let has_selection = has_any_selection(request);
81    let beam_warnings = beam_warnings(&raws)?;
82    let selections: Vec<Option<ImmathSelection>> = if has_selection {
83        raws.iter()
84            .map(|(_, raw)| ImmathSelection::from_raw(raw, request, false).map(Some))
85            .collect::<Result<Vec<_>, _>>()?
86    } else {
87        vec![None; raws.len()]
88    };
89
90    let has_mask = request
91        .mask
92        .as_deref()
93        .is_some_and(|m| !m.trim().is_empty());
94
95    let mask_overrides: Vec<Option<casa_types::ArrayD<bool>>> = if has_mask {
96        let mut raw_aliases: Vec<(String, &dyn ImageInterface<T>)> = Vec::new();
97        for &(idx, ref raw) in &raws {
98            if idx < imagename_paths.len()
99                && let Some(vn) = varnames.get(idx)
100            {
101                raw_aliases.push((vn.clone(), raw.as_interface()));
102            }
103            if let Some(path) = all_paths.get(idx) {
104                raw_aliases.push((path.display().to_string(), raw.as_interface()));
105            }
106        }
107        let mut masks = Vec::with_capacity(raws.len());
108        for ((_, raw), sel) in raws.iter().zip(selections.iter()) {
109            let window = sel
110                .as_ref()
111                .map(|s| (s.start.as_slice(), s.shape.as_slice()));
112            masks.push(evaluate_immath_mask(
113                request.mask.as_deref(),
114                raw.as_interface().shape(),
115                &raw_aliases,
116                request.stretch,
117                window,
118            )?);
119        }
120        masks
121    } else {
122        vec![None; raws.len()]
123    };
124
125    let mut prepared: Vec<(usize, PreparedInput<T>)> = Vec::with_capacity(raws.len());
126    for (((index, raw), mask_override), sel) in
127        raws.into_iter().zip(mask_overrides).zip(selections.iter())
128    {
129        prepared.push((
130            index,
131            apply_selection_to_raw(raw, mask_override, sel.as_ref())?,
132        ));
133    }
134
135    let mut resolver_map: HashMap<String, &dyn ImageInterface<T>> = HashMap::new();
136    for &(idx, ref prep) in &prepared {
137        if idx < imagename_paths.len()
138            && let Some(vn) = varnames.get(idx)
139        {
140            resolver_map.insert(vn.clone(), prep.as_interface());
141        }
142        if let Some(path) = all_paths.get(idx) {
143            resolver_map.insert(path.display().to_string(), prep.as_interface());
144        }
145    }
146    let resolver = HashMapResolver(resolver_map);
147    let compiled =
148        parse_image_expr::<T, _>(&rewritten_expr, &resolver).map_err(expression_error)?;
149
150    let out_shape = compiled.shape().to_vec();
151    let (coords, units, image_info, misc_info, legacy_coords) =
152        resolve_output_metadata(request, imagename_paths, &compiled, clear_units)?;
153
154    super::remove_existing_outfile_if_overwrite(outfile, request.overwrite)?;
155
156    let (_saved, valid_pixels) = compiled.save_as_with_full_metadata(
157        outfile,
158        &coords,
159        &units,
160        &image_info,
161        &misc_info,
162        legacy_coords,
163    )?;
164
165    Ok(ImageMathSummary {
166        imagename: all_paths.iter().map(|p| p.display().to_string()).collect(),
167        outfile: outfile.display().to_string(),
168        expr: request.expr.clone(),
169        shape: out_shape,
170        units,
171        valid_pixels,
172        beam_warnings,
173    })
174}

176fn beam_warnings<T: RealMathPixel>(
177    raws: &[(usize, RawImage<T>)],
178) -> Result<Vec<String>, ImageError> {
179    let mut messages = Vec::new();
180    let mut reference = None;
181    for (_, raw) in raws {
182        let image = raw.as_interface();
183        let beams = image.image_info()?.beam_set;
184        let first = reference.get_or_insert_with(|| beams.clone());
185        if *first != beams {
186            messages.push("immath: input restoring beams differ".to_string());
187            if let (Some(beam), Some(previous)) = (beams.single_beam(), first.single_beam()) {
188                let input = beam_values(beam)?;
189                let previous = beam_values(previous)?;
190                let decimals = (14..=20)
191                    .find(|&places| {
192                        input.iter().zip(previous).all(|(&value, other)| {
193                            value == other
194                                || format!("{value:.places$}") != format!("{other:.places$}")
195                        })
196                    })
197                    .unwrap_or(20);
198                messages.push(format_beam("input", input, decimals));
199                messages.push(format_beam("reference", previous, decimals));
200            }
201        }
202    }
203    Ok(messages)
204}

206fn beam_values(beam: GaussianBeam) -> Result<[f64; 3], ImageError> {
207    Ok([
208        beam.major_in("arcmin")?,
209        beam.minor_in("arcmin")?,
210        beam.position_angle_in("deg")?,
211    ])
212}
213
214fn format_beam(label: &str, [major, minor, pa]: [f64; 3], places: usize) -> String {
215    format!(
216        "immath: {label} beam major={major:.places$}arcmin minor={minor:.places$}arcmin pa={pa:.places$}deg"
217    )
218}

220fn expression_error(error: crate::expr_parser::ExprParseError) -> ImageError {
221    ImageError::InvalidMetadata(format!("Invalid image expression: {error}"))
222}

224fn has_any_selection(request: &ImmathRequest) -> bool {
225    [
226        request.box_pixels.as_deref(),
227        request.region.as_deref(),
228        request.chans.as_deref(),
229        request.stokes.as_deref(),
230    ]
231    .into_iter()
232    .any(|opt| opt.is_some_and(|s| !s.trim().is_empty()))
233}

235fn resolve_output_metadata<T: RealMathPixel>(
236    request: &ImmathRequest,
237    filenames: &[PathBuf],
238    compiled: &crate::ImageExpression<T>,
239    clear_units: bool,
240) -> Result<
241    (
242        CoordinateSystem,
243        String,
244        ImageInfo,
245        RecordValue,
246        Option<RecordValue>,
247    ),
248    ImageError,
249> {
250    let out_shape = compiled.shape();
251    if let Some(raw_md) = request.imagemd.as_deref().filter(|s| !s.trim().is_empty()) {
252        let md_path = translate_imagemd(filenames, raw_md);
253        let md_raw = open_raw_image::<T>(&md_path).map_err(|_| {
254            ImageError::InvalidMetadata(format!(
255                "Cannot access {raw_md} so cannot copy its metadata to output image."
256            ))
257        })?;
258        let md = md_raw.as_interface();
259        let coords = if md.coordinates().n_pixel_axes() == out_shape.len() {
260            md.coordinates().clone()
261        } else {
262            compiled.coordinates().clone()
263        };
264        let legacy_coords = if md.coordinates().n_pixel_axes() == out_shape.len() {
265            md_raw.legacy_coords().cloned()
266        } else {
267            None
268        };
269        let units = if clear_units {
270            String::new()
271        } else {
272            md.units().to_string()
273        };
274        return Ok((
275            coords,
276            units,
277            md.image_info()?,
278            md.misc_info(),
279            legacy_coords,
280        ));
281    }
282
283    let coords = compiled.coordinates().clone();
284    let units = if clear_units {
285        String::new()
286    } else {
287        compiled.units().to_string()
288    };
289    let image_info = compiled.image_info_ref().clone();
290    let misc_info = compiled.misc_info_ref().clone();
291    let legacy_coords = compiled.legacy_coords_ref().cloned();
292    Ok((coords, units, image_info, misc_info, legacy_coords))
293}

295fn translate_imagemd(filenames: &[PathBuf], raw_md: &str) -> PathBuf {
296    let p = Path::new(raw_md);
297    if p.exists() {
298        return p.to_path_buf();
299    }
300    if let Some(rest) = raw_md.strip_prefix("IM")
301        && !rest.is_empty()
302        && rest.bytes().all(|b| b.is_ascii_digit())
303        && let Ok(idx) = rest.parse::<usize>()
304        && let Some(target) = filenames.get(idx)
305    {
306        return target.clone();
307    }
308    p.to_path_buf()
