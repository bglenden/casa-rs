30/// An uncommitted Fourier image. The scratch path uses it until the write
31/// finishes. The virtual model can keep it once its on-disk record is filed.
32pub(crate) struct ImageModel {
33    pub(crate) entry: VirtualModelEntry,
34    backing: GridLifetime,
35}
36
37impl ImageModel {
38    pub(crate) fn copy_to(&self, target: &MeasurementSet) -> MsResult<Self> {
39        let backing = GridLifetime {
40            path: virtual_model_image_path(target)?,
41            retained: false,
42        };
43        crate::paths::copy_dir_recursive(&self.backing.path, &backing.path).map_err(invalid)?;
44        let copied_lock = backing.path.join("table.lock");
45        if copied_lock.is_file() {
46            std::fs::remove_file(copied_lock).map_err(invalid)?;
47        }
48        let mut entry = self.entry.clone();
49        let path = Value::Scalar(ScalarValue::String(
50            backing.path.to_string_lossy().into_owned(),
51        ));
52        if let Some(Value::Record(grid)) = entry.container.get("subftm") {
53            let mut grid = grid.clone();
54            grid.upsert("diskimage", path);
55            entry.container.upsert("subftm", Value::Record(grid));
56        } else {
57            entry.container.upsert("diskimage", path);
58        }
59        Ok(Self { entry, backing })
60    }
61
62    pub(crate) fn retain(&mut self) {
63        self.backing.retained = true;
64    }
65}
66
67struct GridLifetime {
68    path: PathBuf,
69    retained: bool,
70}
71
72impl Drop for GridLifetime {
73    fn drop(&mut self) {
74        if !self.retained && self.path.is_dir() {
75            let _ = std::fs::remove_dir_all(&self.path);
76        }
77    }
78}

87/// The image spectral coordinate determines the default Taylor frequency.
88pub(crate) fn reference_frequency(path: &Path) -> MsResult<f64> {
89    let image = PagedImage::<f32>::open(path).map_err(invalid)?;
90    let index = coordinate_index(image.coordinates(), CoordinateType::Spectral)?;
91    let CoordinateModel::Spectral(spectral) = image.coordinates().coordinate(index) else {
92        unreachable!("a spectral coordinate index names a spectral coordinate")
93    };
94    let pixel = spectral.reference_pixel()[0];
95    spectral.world_frequency_hz(pixel).map_err(invalid)
96}

98pub(crate) fn prepare_image(
99    ms: &MeasurementSet,
100    selected: &ResolvedMsSelection,
101    path: &Path,
102    taylor: Option<TaylorTerm>,
103    keep_zero: bool,
104    scratch: bool,
105) -> MsResult<Option<ImageModel>> {
106    prepare_configured_image(
107        ms,
108        selected,
109        path,
110        taylor,
111        keep_zero,
112        scratch,
113        PredictionGeometry::default(),
114    )
115}
116
117pub(super) fn prepare_configured_image(
118    ms: &MeasurementSet,
119    selected: &ResolvedMsSelection,
120    path: &Path,
121    taylor: Option<TaylorTerm>,
122    keep_zero: bool,
123    scratch: bool,
124    options: PredictionGeometry,
125) -> MsResult<Option<ImageModel>> {
126    let image = PagedImage::<f32>::open(path).map_err(invalid)?;
127    let layout = ImageTransform::new(ms, selected, &image, options)?;
128    let mut path = virtual_model_image_path(ms)?;
129    if scratch {
130        let name = path
131            .file_name()
132            .expect("the grid has a name")
133            .to_string_lossy();
134        path.set_file_name(format!("scratch-{name}"));
135    }
136    let backing = GridLifetime {
137        path,
138        retained: false,
139    };
140    if !layout.grid(&image, &backing.path)? && !keep_zero {
141        return Ok(None);
142    }
143    let record = layout.record(ms, selected, &backing.path, scratch)?;
144    let container = match taylor {
145        Some(term) => legacy_multi_term(record, term.count, term.index, term.reference_hz),
146        None => record,
147    };
148    let entry = model_entry(ms, selected, VirtualModelKind::FtMachine, container)?;
149    Ok(Some(ImageModel { entry, backing }))
150}

164struct WRecord {
165    planes: usize,
166    size: usize,
167    scale: f64,
168    supports: Vec<i32>,
169    statistics: [f64; 3],
170}
171
172impl WRecord {
173    fn new(kernels: &WPlanes, count: WPlaneCount) -> Self {
174        let statistics = match count {
175            WPlaneCount::Fixed(_) => [-1.; 3],
176            WPlaneCount::Restored { max_w, .. } => [-1., max_w, -1.],
177            WPlaneCount::Auto {
178                min_w,
179                max_w,
180                rms_w,
181            } => [min_w, max_w, rms_w],
182        };
183        Self {
184            statistics,
185            planes: kernels.planes(),
186            size: kernels.screen_size(),
187            scale: kernels.w_scale(),
188            supports: (0..kernels.planes())
189                .map(|plane| i32::from(kernels.half_support(plane)))
190                .collect(),
191        }
192    }
193
194    fn save(&self, record: &mut RecordValue, scratch: bool, geometry: &GridGeometry) {
195        record.upsert(
196            "name",
197            Value::Scalar(ScalarValue::String("WProjectFT".into())),
198        );
199        record.remove("convtype");
200        let count = self.planes as i32;
201        for (key, value) in [
202            ("nwplanes", count),
203            ("wconvsize", count),
204            ("convsize", self.size as i32),
205            ("convsampling", if count > 1 { 4 } else { 1 }),
206            ("lastindex", 0),
207        ] {
208            record.upsert(key, Value::Scalar(ScalarValue::Int32(value)));
209        }
210        for (key, value) in ["minw", "maxw", "rmsw"].into_iter().zip(self.statistics) {
211            record.upsert(key, Value::Scalar(ScalarValue::Float64(value)));
212        }
213        record.upsert(
214            "savedwscale",
215            Value::Scalar(ScalarValue::Float64(self.scale)),
216        );
217        record.upsert(
218            "convsizes",
219            Value::Array(ArrayValue::from_i32_vec(Vec::new())),
220        );
221        record.upsert(
222            "convsupport",
223            Value::Array(ArrayValue::from_i32_vec(self.supports.clone())),
224        );
225        if let Some(Value::Array(ArrayValue::Float64(scale))) = record.get("uvscale") {
226            let mut scale = scale.iter().copied().collect::<Vec<f64>>();
227            scale.push(self.scale);
228            record.upsert("uvscale", Value::Array(ArrayValue::from_f64_vec(scale)));
229        }
230        if let Some(Value::Array(ArrayValue::Float64(offset))) = record.get("uvoffset") {
231            let mut offset = offset.iter().copied().collect::<Vec<f64>>();
232            offset.push(0.0);
233            record.upsert("uvoffset", Value::Array(ArrayValue::from_f64_vec(offset)));
234        }
235        if scratch {
236            // The temporary record does not leave this call. Preserve the
237            // geometry used by the legacy iterator before its grid was padded.
238            let GridPadding::CasaComposite(padding) = geometry.padding() else {
239                unreachable!("W projection has a padded grid");
240            };
241            let [nx, ny] = geometry.image().shape;
242            record.upsert(
243                "casars_prediction_geometry",
244                Value::Array(ArrayValue::from_f64_vec(vec![
245                    nx as f64,
246                    ny as f64,
247                    f64::from(padding.get()),
248                ])),
249            );
250        }
251    }
252}

254fn effective_padding(options: &PredictionGeometry) -> MsResult<GridPadding> {
255    // The legacy imager forwards its configured padding to WProjectFT.
256    // Its GridFT factory uses unit padding even when setoptions supplies
257    // another value.
258    let factor = match options.settings.gridder {
259        ImagerGridder::WProject(_) => options.settings.padding,
260        ImagerGridder::Spheroidal | ImagerGridder::Boxcar => 1.0,
261    };
262    Ok(GridPadding::CasaComposite(
263        PaddingFactor::new(factor).map_err(invalid)?,
264    ))
265}

267impl ImageTransform {
268    fn new(
269        ms: &MeasurementSet,
270        selected: &ResolvedMsSelection,
271        image: &PagedImage<f32>,
272        options: PredictionGeometry,
273    ) -> MsResult<Self> {
274        let [nx, ny, npol, channels] = *image.shape() else {
275            return Err(invalid(
276                "ft requires four direction, Stokes and spectral image axes",
277            ));
278        };
279        let coords = image.coordinates();
280        let direction = coordinate_index(coords, CoordinateType::Direction)?;
281        let polarization = coordinate_index(coords, CoordinateType::Stokes)?;
282        let spectral = coordinate_index(coords, CoordinateType::Spectral)?;
283        let CoordinateModel::Direction(direction) = coords.coordinate(direction) else {
284            unreachable!()
285        };
286        let CoordinateModel::Stokes(stokes) = coords.coordinate(polarization) else {
287            unreachable!()
288        };
289        let CoordinateModel::Spectral(spectral) = coords.coordinate(spectral) else {
290            unreachable!()
291        };
292        if npol != stokes.n_stokes() {
293            return Err(invalid(
294                "the model image's Stokes axis does not match its planes",
295            ));
296        }
297        let correlations = first_correlations(ms, selected)?
298            .into_iter()
299            .map(|code| {
300                Correlation::from_casacore_code(code)
301                    .ok_or_else(|| invalid("invalid correlation code"))
302            })
303            .collect::<MsResult<Vec<_>>>()?;
304        let requested = stokes
305            .stokes()
306            .iter()
307            .map(|code| {
308                polarization_coordinate(*code)
309                    .ok_or_else(|| invalid("unsupported polarization plane in model image"))
310            })
311            .collect::<MsResult<Vec<_>>>()?;
312        let routing = PolarizationRouting::compile(&correlations, &requested).map_err(invalid)?;
313        let increment = [direction.increment()[0], direction.increment()[1]];
314        // GridFT phases on the world direction of the integer image
315        // midpoint, which need not be the WCS reference pixel.
316        let reference = [nx / 2, ny / 2];
317        let padding = effective_padding(&options)?;
318        let geometry = GridGeometry::new(
319            ImageExtent {
320                shape: [nx, ny],
321                increment_rad: increment,
322                reference_pixel: reference,
323            },
324            padding,
325        )
326        .map_err(invalid)?;
327        let [gx, gy] = geometry.grid_shape();
328        let phase_center = direction
329            .pixel_to_world([reference[0] as f64, reference[1] as f64])
330            .map_err(invalid)?;
331        let mut coordinates = CoordinateSystem::new();
332        coordinates.add_coordinate(
333            direction
334                .clone()
335                .with_reference_value(phase_center)
336                .with_reference_pixel([(gx / 2) as f64, (gy / 2) as f64]),
337        );
338        coordinates.add_coordinate(StokesCoordinate::new(
339            routing
340                .grid()
341                .iter()
342                .map(grid_stokes)
343                .collect::<MsResult<_>>()?,
344        ));
345        coordinates.add_coordinate(CoordinateModel::Spectral(spectral.clone()));
346        *coordinates.obs_info_mut() = ObsInfo::new("UNKNOWN").with_observer("UNKNOWN");
347        let frequencies = (0..channels)
348            .map(|index| spectral.world_frequency_hz(index as f64).map_err(invalid))
349            .collect::<MsResult<_>>()?;
350        let basis = Basis::ChannelLocal {
351            planes: channels as u32,
352        };
353        let mut w_record = None;
354        let kernels: Box<dyn ConvolutionFunctionSet> = match options.settings.gridder {
355            ImagerGridder::Spheroidal => Box::new(Spheroidal::new(&geometry, &routing)),
356            ImagerGridder::Boxcar => Box::new(Boxcar::new(&geometry, &routing)),
357            ImagerGridder::WProject(planes) => {
358                let kernels = WPlanes::new(&geometry, &routing, planes).map_err(invalid)?;
359                w_record = Some(WRecord::new(&kernels, planes));
360                Box::new(kernels)
361            }
362        };
363        let operator = MeasurementOperator::new(
364            geometry,
365            basis,
366            routing,
367            kernels,
368            options.settings.precision,
369        );
370        let distance_m = match image.misc_info().get("distance") {
371            None => None,
372            Some(Value::Scalar(ScalarValue::Float64(distance))) => Some(*distance),
373            Some(Value::Scalar(ScalarValue::Float32(distance))) => Some(f64::from(*distance)),
374            Some(_) => return Err(invalid("model image distance must be a number in metres")),
375        };
376        Ok(Self {
377            operator,
378            coordinates,
379            frequencies,
380            increment,
381            phase_center,
382            frame: direction.direction_ref().as_str().to_owned(),
383            options,
384            distance_m,
385            w_record,
386        })
387    }

389    fn grid(&self, image: &PagedImage<f32>, path: &Path) -> MsResult<bool> {
390        let [nx, ny, npol, channels] = *image.shape() else {
391            unreachable!()
392        };
393        let [gx, gy] = self.operator.geometry().grid_shape();
394        let beam = self
395            .options
396            .voltage_pattern
397            .as_ref()
398            .map(|beam| {
399                let coords = image.coordinates();
400                let index = coordinate_index(coords, CoordinateType::Direction)?;
401                let CoordinateModel::Direction(direction) = coords.coordinate(index) else {
402                    unreachable!("direction coordinate");
403                };
404                Ok::<_, casa_ms::MsError>((&beam.pattern, beam.separations(direction, [nx, ny])?))
405            })
406            .transpose()?;
407        let grid_pols = self.operator.polarization().grid_pols();
408        let settings = self.options.settings;
409        let plane_bytes = gx.saturating_mul(gy).saturating_mul(8);
410        let cache_bytes = usize::try_from(settings.cache_size)
411            .ok()
412            .filter(|pixels| *pixels > 0)
413            .map_or(plane_bytes, |pixels| {
414                pixels.saturating_mul(8).min(plane_bytes)
415            });
416        let tile = usize::try_from(settings.tile_size)
417            .ok()
418            .filter(|tile| *tile > 0 && *tile != 16);
419        let tile_shape = vec![
420            tile.map_or(gx, |tile| gx.min(tile)),
421            tile.map_or(gy, |tile| gy.min(tile)),
422            1,
423            1,
424        ];
425        let mut output = PagedImage::<Complex32>::create_with_tile_shape_and_cache(
426            vec![gx, gy, grid_pols, channels],
427            tile_shape,
428            self.coordinates.clone(),
429            path,
430            cache_bytes,
431        )
432        .map_err(invalid)?;
433        let mut nonzero = false;
434        for plane in 0..channels {
435            let values = image
436                .get_slice(&[0, 0, 0, plane], &[nx, ny, npol, 1])
437                .map_err(invalid)?;
438            nonzero |= values.iter().any(|value| *value != 0.0);
439            let frequency = self.frequencies[plane];
440            let beam = beam
441                .as_ref()
442                .map(|(pattern, separation)| (pattern.at_frequency(frequency), separation));
443            let images = (0..npol)
444                .map(|pol| {
445                    Array2::from_shape_fn((ny, nx), |(y, x)| {
446                        let scale = beam.as_ref().map_or(1., |(pattern, separation)| {
447                            let radius = separation[[y, x]].to_degrees() * 60.0 * frequency / 1e9;
448                            pattern.power(radius)
449                        });
450                        values[[x, y, pol, 0]] * scale
451                    })
452                })
453                .collect();
454            let model = ModelImages {
455                first_plane: plane as u32,
456                planes: vec![ModelPlane { images }],
457            };
458            let grid = self
459                .operator
460                .prepare_model(&model, ModelPrescale::Unit)
461                .map_err(invalid)?;
462            for pol in 0..grid_pols {
463                let shape = IxDyn(&[gx, gy, 1, 1]).f();
464                let values = match self.operator.precision() {
465                    GridPrecision::F32 => {
466                        let cells = grid.block::<f32>(0, pol, 0);
467                        ArrayD::from_shape_fn(shape, |index| cells[index[1] * gx + index[0]])
468                    }
469                    GridPrecision::F64 => {
470                        let cells = grid.block::<f64>(0, pol, 0);
471                        ArrayD::from_shape_fn(shape, |index| {
472                            let cell = cells[index[1] * gx + index[0]];
473                            Complex32::new(cell.re as f32, cell.im as f32)
474                        })
475                    }
476                };
477                output
478                    .put_slice(&values, &[0, 0, pol, plane])
479                    .map_err(invalid)?;
480            }
481        }
482        output.save().map_err(invalid)?;
483        Ok(nonzero)
484    }

486    fn record(
487        &self,
488        ms: &MeasurementSet,
489        selected: &ResolvedMsSelection,
490        path: &Path,
491        scratch: bool,
492    ) -> MsResult<RecordValue> {
493        let engine = MsCalEngine::new(ms).map_err(invalid)?;
494        let (lon, lat, radius) = engine.observatory_position().as_spherical();
495        let row = &selected.selected_rows[0];
496        let spectral = ms.spectral_window()?;
497        let channels = spectral.chan_freq(row.spectral_window_id as usize)?;
498        let routing = self.operator.polarization();
499        let axis = coordinate_index(&self.coordinates, CoordinateType::Spectral)?;
500        let CoordinateModel::Spectral(axis) = self.coordinates.coordinate(axis) else {
501            unreachable!();
502        };
503        let channel_map = match_channels(axis, self.frequencies.len(), &channels);
504        let buffer = FirstBuffer {
505            field: row.field_id,
506            frequencies_lsrk_hz: channels,
507            channel_map,
508            correlations: first_correlations(ms, selected)?,
509            polarization_map: routing
510                .pol_map()
511                .iter()
512                .map(|pol| pol.map_or(-1, i32::from))
513                .collect(),
514        };
515        let spectral_record = casa_spectral(&CoordinateModel::Spectral(axis.clone()).to_record());
516        let mut record = GridFt {
517            disk_image: path,
518            grid: self.operator.geometry().grid_shape(),
519            polarizations: routing.grid_pols(),
520            image_frequencies_hz: &self.frequencies,
521            increment_rad: self.increment,
522            phase_centre_rad: self.phase_center,
523            direction_frame: &self.frame,
524            observatory: self
525                .options
526                .settings
527                .observatory
528                .unwrap_or([lon, lat, radius]),
529            spectral: spectral_record,
530            buffer: &buffer,
531            interpolation: self.options.settings.frequency_interpolation,
532            frequency_frame_valid: false,
533        }
534        .record();
535        let settings = self.options.settings;
536        record.upsert(
537            "usedoublegrid",
538            Value::Scalar(ScalarValue::Bool(settings.precision == GridPrecision::F64)),
539        );
540        record.upsert(
541            "cachesize",
542            Value::Scalar(ScalarValue::Int64(settings.cache_size)),
543        );
544        record.upsert(
545            "tilesize",
546            Value::Scalar(ScalarValue::Int32(settings.tile_size)),
547        );
548        record.upsert(
549            "numthreads",
550            Value::Scalar(ScalarValue::Int32(settings.threads)),
551        );
552        if let Some(epoch) = self.options.phase_centre_time {
553            record.upsert(
554                "phasecentertime",
555                Value::Scalar(ScalarValue::Float64(epoch)),
556            );
557        }
558        if let Some(distance) = self.distance_m {
559            record.upsert("distance", Value::Scalar(ScalarValue::Float64(distance)));
560        }
561        if scratch && settings.frequency_interpolation != 0 {
562            // This temporary record represents the live iterator, which
563            // keeps its configured interpolation when sampling MODEL_DATA.
564            record.upsert("tovis", Value::Scalar(ScalarValue::Bool(false)));
565        }
566        if matches!(settings.gridder, ImagerGridder::Boxcar) {
567            record.upsert("convtype", Value::Scalar(ScalarValue::String("BOX".into())));
568        }
569        if let Some(w_record) = &self.w_record {
570            w_record.save(&mut record, scratch, self.operator.geometry());
571        }
572        record.upsert(
573            "spwchanselflag",
574            Value::Array(ArrayValue::Int32(selection_mask(ms, selected)?)),
575        );
576        record_selection(&mut record, ms, selected, axis, self.frequencies.len())?;
577        Ok(record)
578    }
579}

581fn match_channels(axis: &SpectralCoordinate, channels: usize, frequencies: &[f64]) -> Vec<i32> {
582    frequencies
583        .iter()
584        .map(|&frequency| {
585            axis.pixel_at_frequency_hz(frequency)
586                .ok()
587                .map(f64::round)
588                .filter(|&index| (0.0..channels as f64).contains(&index))
589                .map_or(-1, |index| index as i32)
590        })
591        .collect()
592}

594fn record_selection(
595    record: &mut RecordValue,
596    ms: &MeasurementSet,
597    selected: &ResolvedMsSelection,
598    axis: &SpectralCoordinate,
599    channels: usize,
600) -> MsResult<()> {
601    let windows = ms.spectral_window()?;
602    let selected_windows: BTreeSet<_> = selected
603        .selected_rows
604        .iter()
605        .map(|row| row.spectral_window_id)
606        .collect();
607    let count = selected_windows.last().map_or(0, |&last| last as usize + 1);
608    let mut maps = RecordValue::default();
609    let mut counts = Vec::new();
610    for window in 0..count {
611        let map = if selected_windows.contains(&(window as i32)) {
612            let frequencies = windows.chan_freq(window)?;
613            counts.push(frequencies.len() as i32);
614            match_channels(axis, channels, &frequencies)
615        } else {
616            Vec::new()
617        };
618        maps.upsert(
619            format!("*{}", window + 1),
620            Value::Array(ArrayValue::from_i32_vec(map)),
621        );
622    }
623    record.upsert(
624        "selectedspw",
625        Value::Array(ArrayValue::from_i32_vec(
626            selected_windows.into_iter().collect(),
627        )),
628    );
629    record.upsert(
630        "nvischanmulti",
631        Value::Array(ArrayValue::from_i32_vec(counts)),
632    );
633    record.upsert("multichanmaprec", Value::Record(maps));
634    record.upsert(
635        "doconversion",
636        Value::Array(ArrayValue::Bool(ArrayD::from_elem(IxDyn(&[count]), false))),
637    );
638    Ok(())
639}

641fn selection_mask(ms: &MeasurementSet, selected: &ResolvedMsSelection) -> MsResult<ArrayD<i32>> {
642    let windows = ms.spectral_window()?;
643    let counts = (0..windows.row_count())
644        .map(|spw| windows.num_chan(spw))
645        .collect::<MsResult<Vec<_>>>()?;
646    let max_channels = counts.iter().copied().max().unwrap_or(0) as usize;
647    let mut mask = ArrayD::zeros(IxDyn(&[1, counts.len(), max_channels]).f());
648    let selected_windows: BTreeSet<_> = selected
649        .selected_rows
650        .iter()
651        .map(|row| row.spectral_window_id)
652        .collect();
653    for id in selected_windows {
654        for channel in 0..counts[id as usize] as usize {
655            mask[[0, id as usize, channel]] = 1;
656        }
657    }
658    for (&window, channels) in &selected.channel_selections {
659        let count = counts[window as usize] as usize;
660        for channel in 0..count {
661            mask[[0, window as usize, channel]] = 0;
662        }
663        let (span, _) = channel_span(channels);
664        // Imager::setdata shifts its mask to the first selected channel.
665        // GridFT then indexes this mask with the compact buffer index.
666        for channel in channels.indices(count)? {
667            mask[[0, window as usize, channel - span.start]] = 1;
668        }
669    }
670    Ok(mask)
671}

675        .corr_type(selected.selected_rows[0].polarization_id as usize)

684fn grid_stokes(plane: &GridPolarization) -> MsResult<StokesType> {
685    match plane {
686        GridPolarization::StokesI => Ok(StokesType::I),
687        GridPolarization::Correlation(correlation) => {
688            StokesType::from_code(correlation.casacore_code())
689                .ok_or_else(|| invalid("invalid grid polarization"))
690        }
691    }

694pub(crate) fn polarization_coordinate(stokes: StokesType) -> Option<PolarizationCoordinate> {
695    match stokes {
696        StokesType::I => Some(PolarizationCoordinate::StokesI),
697        StokesType::Q => Some(PolarizationCoordinate::StokesQ),
698        StokesType::U => Some(PolarizationCoordinate::StokesU),
699        StokesType::V => Some(PolarizationCoordinate::StokesV),
700        StokesType::XX => Some(PolarizationCoordinate::LinearXx),
701        StokesType::XY => Some(PolarizationCoordinate::LinearXy),
702        StokesType::YX => Some(PolarizationCoordinate::LinearYx),
703        StokesType::YY => Some(PolarizationCoordinate::LinearYy),
704        StokesType::RR => Some(PolarizationCoordinate::CircularRr),
705        StokesType::RL => Some(PolarizationCoordinate::CircularRl),
706        StokesType::LR => Some(PolarizationCoordinate::CircularLr),
707        StokesType::LL => Some(PolarizationCoordinate::CircularLl),
708        _ => None,
709    }
