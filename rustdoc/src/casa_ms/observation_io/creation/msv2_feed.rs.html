2//! Preserve the validity of retained static MSv2 feeds. The catalogue
3//! matches receptors; their original validity must not become the MAIN span.
4use super::{combine::Combined, inputs::Input, msv2_tables::Layout};
5use crate::observation_io::update::{request_error, storage};
6use crate::{MeasurementSet, SubtableId};
7use casa_core::{RecordValue, ScalarValue, Value};
8use casa_observation::ObservationError;
9
10fn integer(row: &RecordValue, name: &str) -> Result<i32, ObservationError> {
11    match row.get(name) {
12        Some(Value::Scalar(ScalarValue::Int32(value))) => Ok(*value),
13        _ => Err(request_error(format!("FEED {name} is not int32"))),
14    }
15}
16
17fn real(row: &RecordValue, name: &str) -> Result<f64, ObservationError> {
18    match row.get(name) {
19        Some(Value::Scalar(ScalarValue::Float64(value))) => Ok(*value),
20        _ => Err(request_error(format!("FEED {name} is not float64"))),
21    }
22}

24/// MSFeedColumns::matchFeed extends a matching row when its old validity
25/// does not cover the incoming one. An old zero interval means unlimited.
26fn merge_validity(
27    output: &mut casa_tables::Table,
28    target: usize,
29    old: &RecordValue,
30    incoming: &RecordValue,
31) -> Result<(), ObservationError> {
32    let old_time = real(old, "TIME")?;
33    let new_time = real(incoming, "TIME")?;
34    let old_interval = real(old, "INTERVAL")?;
35    let old_half = if old_interval == 0.0 {
36        5.0e17
37    } else {
38        old_interval / 2.0
39    };
40    let new_half = real(incoming, "INTERVAL")? / 2.0;
41    let offset = new_time - old_time;
42    if -old_half <= offset - new_half && old_half >= offset + new_half {
43        return Ok(());
44    }
45    let centre = old_time + offset / 2.0;
46    let interval = 2.0 * (old_half.max(offset + new_half) - offset / 2.0);
47    for (column, value) in [("TIME", centre), ("INTERVAL", interval)] {
48        output
49            .row_accessor_mut()
50            .set_cell(target, column, Value::Scalar(ScalarValue::Float64(value)))
51            .map_err(storage)?;
52    }
53    Ok(())
54}

56pub(super) fn preserve(
57    ms: &mut MeasurementSet,
58    inputs: &[Input],
59    combined: &Combined,
60    layout: &Layout,
61) -> Result<(), ObservationError> {
62    let Some(output) = ms.subtable_mut(SubtableId::Feed) else {
63        return Ok(());
64    };
65    let mut written = vec![false; output.row_count()];
66    for (input, map) in inputs.iter().zip(&combined.rows) {
67        let Some(source) = input
68            .msv2()
69            .and_then(|input| input.measurement_set().subtable(SubtableId::Feed))
70        else {
71            continue;
72        };
73        let catalogue = input.observation().catalogue();
74        for row in source.rows().map_err(storage)? {
75            let antenna = integer(row, "ANTENNA_ID")?;
76            let antenna = *usize::try_from(antenna)
77                .ok()
78                .and_then(|id| map.antennas.get(id))
79                .ok_or_else(|| request_error("FEED names an unknown antenna"))?;
80            let feed = integer(row, "FEED_ID")?;
81            let window = integer(row, "SPECTRAL_WINDOW_ID")?;
82            let windows: Vec<i32> = catalogue
83                .partitions
84                .iter()
85                .zip(&map.partitions)
86                .filter(|(partition, _)| window < 0 || partition.window.0 as i32 == window)
87                .map(|(_, partition)| layout.spectral_window[partition.output])
88                .collect();
89            let mut matched = false;
90            for (target, assigned) in written.iter_mut().enumerate() {
91                let target_row = output.row_accessor().row(target).map_err(storage)?.clone();
92                let target_window = integer(&target_row, "SPECTRAL_WINDOW_ID")?;
93                if integer(&target_row, "ANTENNA_ID")? != antenna
94                    || integer(&target_row, "FEED_ID")? != feed
95                    || (target_window >= 0 && !windows.contains(&target_window))
96                {
97                    continue;
98                }
99                matched = true;
100                if *assigned {
101                    merge_validity(output, target, &target_row, row)?;
102                    continue;
103                }
104                for column in ["TIME", "INTERVAL"] {
105                    let value = row
106                        .get(column)
107                        .ok_or_else(|| request_error(format!("FEED lacks {column}")))?;
108                    output
109                        .row_accessor_mut()
110                        .set_cell(target, column, value.clone())
111                        .map_err(storage)?;
112                }
113                if window < 0 {
114                    output
115                        .row_accessor_mut()
116                        .set_cell(
117                            target,
118                            "SPECTRAL_WINDOW_ID",
119                            Value::Scalar(ScalarValue::Int32(-1)),
120                        )
121                        .map_err(storage)?;
122                }
123                *assigned = true;
124            }
125            if !matched {
126                // An unused antenna still owns its FEED rows. CASA keeps
127                // them even though the scientific batches never name it.
128                let mut row = row.clone();
129                *row.get_mut("ANTENNA_ID").expect("checked antenna") =
130                    Value::Scalar(ScalarValue::Int32(antenna));
131                let window = if window < 0 {
132                    -1
133                } else {
134                    *windows.first().ok_or_else(|| {
135                        request_error("FEED of an unused spectral window is outside #939")
136                    })?
137                };
138                *row.get_mut("SPECTRAL_WINDOW_ID").expect("checked window") =
139                    Value::Scalar(ScalarValue::Int32(window));
140                output.add_row(row).map_err(storage)?;
141                written.push(true);
142            }
143        }
144    }
145    let mut duplicates = Vec::new();
146    let rows = output.rows().map_err(storage)?;
147    for row in 0..rows.len() {
148        if rows[..row].contains(&rows[row]) {
149            duplicates.push(row);
150        }
151    }
152    output.remove_rows(&duplicates).map_err(storage)?;
153    Ok(())
154}
