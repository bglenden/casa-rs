2//! What each output partition's grids hold, read before anything is
3//! written: its contributors' grids (one interval, and one field and scan
4//! at each time, as the MSv2 reader requires of a partition), the times
5//! they span, and the rows it will hold: its grids' cells, or with padding
6//! omitted its acquisitions, counted from the weights. With several
7//! contributors an acquired cell is kept as one bit of the combined grid,
8//! and a cell two inputs both acquire is refused, naming them. Times are
9//! compared as `MainRows` writes them, MJD seconds from the grid's origin.
10

23/// Seconds per day, for times as MJD seconds.
24const SECONDS_PER_DAY: f64 = 86_400.0;
25
26/// `grid`'s time `t` as MJD seconds, as a batch of its partition states
27/// it (`EpochView::seconds_since` from MJD 0).
28fn seconds(grid: &TimeBaselineGrid, t: usize) -> f64 {
29    let origin = grid.time_origin;
30    let zero = MjdHighPrec::new(0.0, 0.0);
31    ((origin.day() - zero.day()) + (origin.frac() - zero.frac())) * SECONDS_PER_DAY
32        + grid.times_s[t]
33}

35/// The times grids span, gathered grid by grid.
36struct Spanning {
37    range: [f64; 2],
38    sessions: Vec<[f64; 2]>,
39    /// Each output field's first time.
40    fields: Vec<Option<f64>>,
41}
42
43impl Spanning {
44    const EMPTY: [f64; 2] = [f64::INFINITY, f64::NEG_INFINITY];
45
46    /// No span yet, of `sessions` observing sessions and `fields` fields.
47    fn new(sessions: usize, fields: usize) -> Self {
48        Self {
49            range: Self::EMPTY,
50            sessions: vec![Self::EMPTY; sessions],
51            fields: vec![None; fields],
52        }
53    }
54
55    /// Widen the spans by `grid`, of observing session `session`, whose
56    /// fields are the output's `fields` (by the grid's field).
57    fn add(&mut self, grid: &TimeBaselineGrid, session: ObservationId, fields: &[i32]) {
58        if grid.times_s.is_empty() {
59            return;
60        }
61        for (t, field) in grid.field_per_time.iter().enumerate() {
62            let output = fields
63                .get(field.0 as usize)
64                .and_then(|output| usize::try_from(*output).ok());
65            if let Some(first) = output.and_then(|output| self.fields.get_mut(output)) {
66                let time = seconds(grid, t);
67                *first = Some(first.map_or(time, |known: f64| known.min(time)));
68            }
69        }
70        let (first, last) = (seconds(grid, 0), seconds(grid, grid.times_s.len() - 1));
71        let span =
72            |known: [f64; 2], half: f64| [known[0].min(first - half), known[1].max(last + half)];
73        self.range = span(self.range, 0.0);
74        // A session's TIME_RANGE spans its integrations, edge to edge, as
75        // fillers write it.
76        let half = if grid.interval_s.is_finite() {
77            grid.interval_s / 2.0
78        } else {
79            0.0
80        };
81        if let Some(known) = self.sessions.get_mut(session.0 as usize) {
82            *known = span(*known, half);
83        }
84    }
85
86    fn finish(self) -> Spans {
87        let spanned = |range: [f64; 2]| {
88            if range[0] > range[1] {
89                [0.0, 0.0]
90            } else {
91                range
92            }
93        };
94        Spans {
95            time_range: spanned(self.range),
96            observation_ranges: self.sessions.into_iter().map(spanned).collect(),
97            field_first: self.fields,
98        }
99    }
100}

102/// One output partition's combined grid: its times, and the field and
103/// scan number of each.
104struct Combination {
105    times: Vec<f64>,
106    fields: Vec<Option<i32>>,
107    scans: Vec<Option<i32>>,
108}
109
110impl Combination {
111    /// The combined grid of `grids`, each with its contributor's field and
112    /// scan number at each time (`identities`).
113    fn new(
114        grids: &[TimeBaselineGrid],
115        identities: &[Vec<(i32, i32)>],
116    ) -> Result<Self, ObservationError> {
117        let mut times: Vec<f64> = grids
118            .iter()
119            .flat_map(|grid| (0..grid.times_s.len()).map(|t| seconds(grid, t)))
120            .collect();
121        times.sort_by(f64::total_cmp);
122        times.dedup_by(|a, b| a.to_bits() == b.to_bits());
123        let mut combination = Self {
124            fields: vec![None; times.len()],
125            scans: vec![None; times.len()],
126            times,
127        };
128        for (grid, identities) in grids.iter().zip(identities) {
129            for (t, (field, scan)) in identities.iter().enumerate() {
130                let index = combination.time(seconds(grid, t))?;
131                for (known, value, what) in [
132                    (&mut combination.fields[index], *field, "fields"),
133                    (&mut combination.scans[index], *scan, "scans"),
134                ] {
135                    match known {
136                        Some(known) if *known != value => {
137                            return Err(request_error(format!(
138                                "the inputs combined into one partition hold two {what} at time \
139                                 {} s; an MSv2 partition holds one",
140                                combination.times[index]
141                            )));
142                        }
143                        Some(_) => {}
144                        None => *known = Some(value),
145                    }
146                }
147            }
148        }
149        Ok(combination)
150    }
151
152    /// The position of time `seconds` on the combined grid.
153    fn time(&self, seconds: f64) -> Result<usize, ObservationError> {
154        self.times
155            .binary_search_by(|time| time.total_cmp(&seconds))
156            .map_err(|_| request_error("a batch's time is not on its partition's grid"))
157    }
158}

160/// What the output partitions' grids hold, read from the indexes, before
161/// any weight is.
162pub(super) struct Grids {
163    combinations: Vec<Combination>,
164    /// The bytes admitted for the combinations, released once the rows
165    /// are counted.
166    combined_bytes: usize,
167    intervals: Vec<f64>,
168    /// Each output partition's grid cells: its rows with padding flagged.
169    cells: Vec<usize>,
170    /// The times the grids span.
171    pub(super) spans: Spans,
172}
173
174/// The bytes a combination of grids of `times` times in all holds: the
175/// times gathered (deduplicated in place), and a field and scan for each.
176fn combination_bytes(times: usize) -> usize {
177    times * (size_of::<f64>() + 2 * size_of::<Option<i32>>())
178}
179
180/// The grids of `combined`'s output partitions, from `inputs`: each one's
181/// combined grid checked for one interval and one field and scan at a
182/// time, and the times they span. Each partition's grids are admitted to
183/// `ledger` while they are combined, and their combination until the rows
184/// are counted ([`Grids::occupancy`]).
185pub(super) fn grids(
186    inputs: &[Input],
187    combined: &Combined,
188    ledger: &mut Ledger,
189) -> Result<Grids, ObservationError> {
190    let output = &combined.catalogue;
191    let mut spanned = Spanning::new(output.observations.len(), output.fields.len());
192    let count = output.partitions.len();
193    let mut read = Grids {
194        combinations: Vec::with_capacity(count),
195        combined_bytes: 0,
196        intervals: Vec::with_capacity(count),
197        cells: Vec::with_capacity(count),
198        spans: Spanning::new(0, 0).finish(),
199    };
200    for (index, contributors) in combined.contributors.iter().enumerate() {
201        let mut grids = Vec::with_capacity(contributors.len());
202        let mut identities = Vec::with_capacity(contributors.len());
203        let mut held = 0;
204        for (input, id) in contributors {
205            let map = &combined.rows[*input];
206            let placement = &map.partitions[id.0 as usize];
207            let observation = inputs[*input].observation();
208            let group = &placement.groups[0];
209            let times = Sizing::of(observation, *id, std::slice::from_ref(group), WEIGHTS)?.times;
210            let bytes = grid_bytes(
211                times,
212                observation.catalogue().partition(*id).baselines.len(),
213            ) + times * size_of::<(i32, i32)>();
214            ledger.admit("a partition's grids", bytes)?;
215            held += bytes;
216            let grid = observation.grid(*id, group)?;
217            spanned.add(&grid, output.partitions[index].observation, &map.fields);
218            identities.push(
219                grid.field_per_time
220                    .iter()
221                    .zip(&grid.scan_per_time)
222                    .map(|(field, scan)| {
223                        (
224                            map.fields[field.0 as usize],
225                            map.scans[scan.0 as usize] + placement.scan_offset,
226                        )
227                    })
228                    .collect::<Vec<_>>(),
229            );
230            grids.push(grid);
231        }
232        read.intervals.push(one_interval(&grids)?);
233        read.cells.push(
234            grids
235                .iter()
236                .map(|grid| grid.times_s.len() * grid.baselines.len())
237                .sum(),
238        );
239        let bytes = combination_bytes(grids.iter().map(|grid| grid.times_s.len()).sum());
240        ledger.admit("a partition's combined times", bytes)?;
241        read.combined_bytes += bytes;
242        read.combinations
243            .push(Combination::new(&grids, &identities)?);
244        drop((grids, identities));
245        ledger.release(held);
246    }
247    read.spans = spanned.finish();
248    Ok(read)
249}
250
251impl Grids {
252    /// The occupancy of `combined`'s output partitions under `padding`:
253    /// their grid cells, or with padding omitted their acquisitions, read
254    /// from `inputs`' weights with `budget` bytes per batch, each batch and
255    /// the cells acquired admitted to `ledger` while they are read; the
256    /// combinations' bytes are released after.
257    pub(super) fn occupancy(
258        self,
259        inputs: &[Input],
260        combined: &Combined,
261        (padding, budget): (PaddingRows, usize),
262        ledger: &mut Ledger,
263    ) -> Result<Occupancy, ObservationError> {
264        let rows = match padding {
265            PaddingRows::Flagged => self.cells,
266            PaddingRows::Omitted => combined
267                .contributors
268                .iter()
269                .zip(&self.combinations)
270                .zip(&combined.catalogue.partitions)
271                .map(|((contributors, combination), entry)| {
272                    acquisitions(
273                        inputs,
274                        combined,
275                        (contributors, &entry.baselines),
276                        combination,
277                        (budget, &mut *ledger),
278                    )
279                })
280                .collect::<Result<_, _>>()?,
281        };
282        drop(self.combinations);
283        ledger.release(self.combined_bytes);
284        Ok(Occupancy {
285            rows,
286            intervals: self.intervals,
287            spans: self.spans,
288        })
289    }
290}

292/// The times one observation's grids of `group` span, for subtables
293/// written without rows.
294#[cfg(feature = "msv4")]
295pub(super) fn spans(
296    observation: &dyn casa_observation::Observation,
297    group: &str,
298) -> Result<Spans, ObservationError> {
299    let catalogue = observation.catalogue();
300    let mut spanned = Spanning::new(catalogue.observations.len(), catalogue.fields.len());
301    // The observation's own fields.
302    let fields: Vec<i32> = (0..catalogue.fields.len() as i32).collect();
303    for id in catalogue.partition_ids() {
304        spanned.add(
305            &observation.grid(id, group)?,
306            catalogue.partition(id).observation,
307            &fields,
308        );
309    }
310    Ok(spanned.finish())
311}

313/// The one integration interval of `grids`, which an MSv2 partition
314/// shares; a grid of no acquisition states none.
315fn one_interval(grids: &[TimeBaselineGrid]) -> Result<f64, ObservationError> {
316    let mut stated = grids
317        .iter()
318        .map(|grid| grid.interval_s)
319        .filter(|interval| interval.is_finite());
320    let Some(first) = stated.next() else {
321        return Ok(f64::NAN);
322    };
323    if stated.any(|interval| interval.to_bits() != first.to_bits()) {
324        return Err(ObservationError::Unsupported {
325            what: "combining partitions of different integration intervals",
326        });
327    }
328    Ok(first)
329}

331/// The acquisitions of an output partition's `contributors`, whose
332/// combined grid is `combination` over its `baselines`: each contributor's
333/// weights read in batches of `budget` bytes, and a cell acquired twice
334/// refused; the bit of each cell and each batch admitted to `ledger` while
335/// they are held.
336fn acquisitions(
337    inputs: &[Input],
338    combined: &Combined,
339    (contributors, baselines): (
340        &[(usize, casa_observation::PartitionId)],
341        &[OrderedBaseline],
342    ),
343    combination: &Combination,
344    (budget, ledger): (usize, &mut Ledger),
345) -> Result<usize, ObservationError> {
346    let cells = combination.times.len() * baselines.len();
347    let words = if contributors.len() > 1 {
348        cells.div_ceil(64)
349    } else {
350        0
351    };
352    ledger.during("a partition's acquired cells", words * 8, |ledger| {
353        let mut filled = vec![0_u64; words];
354        let mut count = 0;
355        for (position, (input, id)) in contributors.iter().enumerate() {
356            let map = &combined.rows[*input];
357            let placement = &map.partitions[id.0 as usize];
358            let groups = [placement.groups[0].clone()];
359            let observation = inputs[*input].observation();
360            let sizing = Sizing::of(observation, *id, &groups, WEIGHTS)?;
361            let batch = sizing.times_per_batch(budget, None) * sizing.integration();
362            ledger.during("a weights batch", batch, |_| {
363                count += contributor(
364                    (observation, *id, &groups, budget),
365                    (combination, baselines, &map.antennas),
366                    &mut filled,
367                    (&contributors[..position], *input),
368                )?;
369                Ok(())
370            })?;
371        }
372        Ok(count)
373    })
374}

376/// The acquisitions of one contributor (input `input`, partition `id` of
377/// `observation`, read in batches of `budget`), each marked in `filled`
378/// when it is not empty and refused where it is already marked, naming the
379/// `earlier` contributors; antennas are the output's through `antennas`.
380fn contributor(
381    (observation, id, groups, budget): (
382        &dyn casa_observation::Observation,
383        casa_observation::PartitionId,
384        &[String],
385        usize,
386    ),
387    (combination, baselines, antennas): (&Combination, &[OrderedBaseline], &[i32]),
388    filled: &mut [u64],
389    (earlier, input): (&[(usize, casa_observation::PartitionId)], usize),
390) -> Result<usize, ObservationError> {
391    let mut count = 0;
392    lockstep::read(
393        observation,
394        id,
395        groups,
396        WEIGHTS,
397        (budget, None),
398        |batches| {
399            let batch = &batches[0];
400            let weights = batch
401                .group
402                .weights
403                .ok_or_else(|| request_error("an occupancy batch carries its weights"))?;
404            let origin = MjdHighPrec::new(0.0, 0.0);
405            for (sample, weights) in weights.axis_iter(Axis(0)).enumerate() {
406                if !acquired(weights)? {
407                    continue;
408                }
409                count += 1;
410                if filled.is_empty() {
411                    continue;
412                }
413                let time = combination.time(
414                    batch
415                        .time
416                        .seconds_since(sample / batch.samples_per_time(), origin),
417                )?;
418                let baseline = batch.sample(sample).baseline;
419                let antenna = |antenna: casa_observation::AntennaId| {
420                    casa_observation::AntennaId(antennas[antenna.0 as usize] as u32)
421                };
422                let remapped = OrderedBaseline {
423                    antenna1: antenna(baseline.antenna1),
424                    antenna2: antenna(baseline.antenna2),
425                };
426                let b = baselines
427                    .binary_search(&remapped)
428                    .map_err(|_| request_error("a batch's baseline is not on its grid"))?;
429                let cell = time * baselines.len() + b;
430                let (word, bit) = (cell / 64, 1_u64 << (cell % 64));
431                if filled[word] & bit != 0 {
432                    let earlier: Vec<usize> = earlier.iter().map(|(input, _)| *input).collect();
433                    return Err(request_error(format!(
434                        "input {input} acquires baseline {}-{} at {} s, which input(s) {earlier:?} \
435                     also acquire; an MSv2 holds one row there",
436                        remapped.antenna1.0, remapped.antenna2.0, combination.times[time]
437                    )));
438                }
439                filled[word] |= bit;
440            }
441            Ok(())
442        },
443    )?;
444    Ok(count)
