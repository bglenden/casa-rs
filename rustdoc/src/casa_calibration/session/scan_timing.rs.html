2//! Map calibration epochs to the MeasurementSet scans used by CASA.
3
4use std::collections::BTreeMap;
5
6use casa_tables::{RequiredScalarColumnValues, Table};
7
8use crate::constants::{COL_FIELD_ID, COL_OBSERVATION_ID, COL_SCAN_NUMBER, COL_TIME};
9
10pub(crate) struct ScanTiming(Vec<Scan>);
11
12pub(crate) struct Scan {
13    pub(crate) number: i32,
14    pub(crate) field: i32,
15    pub(crate) observation: i32,
16    first_time: f64,
17    last_time: f64,
18}
19
20impl ScanTiming {
21    pub(crate) fn read(main: &Table) -> Result<Self, String> {
22        let columns = main
23            .required_scalar_columns_owned(&[
24                COL_TIME,
25                COL_SCAN_NUMBER,
26                COL_FIELD_ID,
27                COL_OBSERVATION_ID,
28            ])
29            .map_err(|error| error.to_string())?;
30        let RequiredScalarColumnValues::Float64(times) = &columns[COL_TIME] else {
31            return Err("MeasurementSet TIME is not Float64.".into());
32        };
33        let ids = |name: &str| match &columns[name] {
34            RequiredScalarColumnValues::Int32(values) => Ok(values.as_slice()),
35            _ => Err(format!("MeasurementSet {name} is not Int32.")),
36        };
37        let numbers = ids(COL_SCAN_NUMBER)?;
38        let fields = ids(COL_FIELD_ID)?;
39        let observations = ids(COL_OBSERVATION_ID)?;
40        let mut by_scan = BTreeMap::new();
41        for (row, &time) in times.iter().enumerate() {
42            let scan = by_scan.entry(numbers[row]).or_insert(Scan {
43                number: numbers[row],
44                field: fields[row],
45                observation: observations[row],
46                first_time: time,
47                last_time: time,
48            });
49            scan.first_time = scan.first_time.min(time);
50            scan.last_time = scan.last_time.max(time);
51        }
52        let mut scans = by_scan.into_values().collect::<Vec<_>>();
53        if scans.is_empty() {
54            return Err("MeasurementSet has no visibility scans.".into());
55        }
56        scans.sort_by(|left, right| left.last_time.total_cmp(&right.last_time));
57        Ok(Self(scans))
58    }
59
60    pub(crate) fn at(&self, time: f64) -> &Scan {
61        let scans = &self.0;
62        let first = &scans[0];
63        let last = &scans[scans.len() - 1];
64        // assignCTScanField uses the next scan after a gap and clips the ends.
65        if time < first.last_time + 1e-5 {
66            first
67        } else if time > last.first_time - 1e-5 {
68            last
69        } else {
70            &scans[scans.partition_point(|scan| time > scan.last_time + 1e-5)]
71        }
72    }
73}
