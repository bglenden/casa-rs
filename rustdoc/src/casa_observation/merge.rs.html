2//! Merging the catalogues of several observations into one, as CASA's
3//! `concat` (casacore `MSConcat::concatenate`) merges MeasurementSets'
4//! subtables ([`merge`]).
5//!
6//! The first catalogue's entities are taken as they are. Each later one is
7//! appended in CASA's order (states, sources, windows, antennas, fields,
8//! observing sessions, processors), each entity matched against the merged
9//! entities so far by casacore's predicate and in its search order, so the
10//! representative of a chain of tolerance matches is the one CASA picks;
11//! an entity without a match is appended. Observing sessions and
12//! processors are instead appended and then made distinct over the whole
13//! table, as CASA does, each mapped to the first equivalent; sources are
14//! merged as [`sources`] describes.
15//!
16//! A predicate reads only what both entities state: an entity lacking a
17//! member its predicate needs (an MSv4 window's IF chain, a flag no store
18//! states) is not matched, but appended, and the merge reports it as a
19//! notice ([`Merged::notices`]). Values the predicates compare exactly
20//! (names, codes, modes) must be equal; an unstated one is equal to
21//! nothing.

36/// One milliarcsecond in radians: CASA `concat`'s default `dirtol`.
37const MILLIARCSECOND: f64 = std::f64::consts::PI / (180.0 * 3600.0 * 1000.0);
38
39/// How close two entities must be to be one: casacore's antenna position
40/// tolerance and CASA `concat`'s `dirtol`, `freqtol` and `respectname`.
41#[derive(Clone, Copy, Debug, PartialEq)]
42pub struct MergeTolerance {
43    /// Each component of an antenna's position (casacore: 1 m).
44    pub position_m: f64,
45    /// The separation of each of a field's directions, strictly below
46    /// (`dirtol`, 1 mas by default).
47    pub direction_rad: f64,
48    /// A window's reference frequency and each channel's (`freqtol`, 1 Hz
49    /// by default).
50    pub frequency_hz: f64,
51    /// Whether fields must also share their names (`respectname`).
52    pub respect_names: bool,
53}
54
55impl Default for MergeTolerance {
56    fn default() -> Self {
57        Self {
58            position_m: 1.0,
59            direction_rad: MILLIARCSECOND,
60            frequency_hz: 1.0,
61            respect_names: false,
62        }
63    }
64}

66/// Where one input's entities are in the merged catalogue, each by the
67/// input's own identity.
68#[derive(Clone, Debug, Default, PartialEq, Eq)]
69pub struct InputMap {
70    /// Each antenna's.
71    pub antennas: Vec<AntennaId>,
72    /// Each window's.

74    /// Each field's.

123/// A window's channels: the frequencies and frame of the first partition
124/// that uses it, `None` for a window no partition uses.
125type Channels = Option<(Vec<f64>, FrequencyRef)>;
126
127/// The channels of each of `catalogue`'s windows.
128fn window_channels(catalogue: &Catalogue) -> Vec<Channels> {
129    (0..catalogue.windows.len())
130        .map(|index| {
131            catalogue
132                .partitions
133                .iter()
134                .find(|entry| entry.window.0 as usize == index)
135                .map(|entry| (entry.frequencies_hz.clone(), entry.frequency_frame))
136        })
137        .collect()
138}

140/// Merge `catalogues` (at least one) as CASA's `concat` merges their
141/// MeasurementSets, within `tolerance`.
142///
143/// # Errors
144///
145/// [`ObservationError::Unsupported`] for what CASA's predicates cannot
146/// match or this merge does not: antenna positions in different frames
147/// (casacore refuses them) and a window whose channels match another's
148/// only reversed. (A catalogue holds no field in a solar-system frame, which
149/// CASA matches by frame alone: no store's adapter represents one.)
150pub fn merge(
151    catalogues: &[&Catalogue],
152    tolerance: &MergeTolerance,
153) -> Result<Merged, ObservationError> {
154    let mut merged = Merged::default();
155    let mut channels: Vec<Channels> = Vec::new();
156    for (index, catalogue) in catalogues.iter().enumerate() {
157        let own_channels = window_channels(catalogue);
158        if index == 0 {
159            let map = first(&mut merged, catalogue);
160            merged.inputs.push(map);
161            channels = own_channels;
162            continue;
163        }
164        let mut map = InputMap {
165            states: states(&mut merged, catalogue),
166            ..InputMap::default()
167        };
168        map.windows = windows(
169            &mut merged,
170            (&mut channels, catalogue, &own_channels),
171            tolerance,
172        )?;
173        map.antennas = antennas(&mut merged, catalogue, tolerance)?;
174        let before = merged.fields.len();
175        let (fields, matched) = fields(&mut merged, catalogue, tolerance);
176        map.fields = fields;
177        merged.inputs.push(map);
178        sources::append(&mut merged, catalogue, (before, matched))?;
179        observations(&mut merged, catalogue);
180        processors(&mut merged, catalogue);
181    }
182    Ok(merged)
183}

185/// The first input's entities, taken as they are.
186fn first(merged: &mut Merged, catalogue: &Catalogue) -> InputMap {
187    merged.antennas.clone_from(&catalogue.antennas);
188    merged.windows.clone_from(&catalogue.windows);
189    merged.fields.clone_from(&catalogue.fields);
190    merged.sources.clone_from(&catalogue.sources);
191    merged.states.clone_from(&catalogue.states);
192    merged.processors.clone_from(&catalogue.processors);
193    merged.observations.clone_from(&catalogue.observations);
194    let ids = |count: usize| 0..count as u32;
195    InputMap {
196        antennas: ids(catalogue.antennas.len()).map(AntennaId).collect(),
197        windows: ids(catalogue.windows.len()).map(WindowId).collect(),
198        fields: ids(catalogue.fields.len()).map(FieldId).collect(),
199        sources: ids(catalogue.sources.len())
200            .map(|id| Some(SourceId(id)))
201            .collect(),
202        states: ids(catalogue.states.len()).map(StateId).collect(),
203        processors: ids(catalogue.processors.len()).map(ProcessorId).collect(),
204        observations: ids(catalogue.observations.len())
205            .map(ObservationId)
206            .collect(),
207        counterparts: vec![false; catalogue.observations.len()],
208    }
209}

211/// Whether `values` are all stated: the evidence a predicate needs.
212macro_rules! stated {
213    ($($value:expr),+ $(,)?) => {
214        true $(&& $value.is_some())+
215    };
216}

218/// Each incoming state matched (casacore `MSStateColumns::matchState`,
219/// from the last state back) or appended. An input without states has no
220/// map: its rows keep none, where CASA drops every state of the result
221/// (recorded in `docs/CASA (C++) bugs.md`).
222fn states(merged: &mut Merged, catalogue: &Catalogue) -> Vec<StateId> {
223    const KELVIN: f64 = 1.0;
224    let mut map = Vec::with_capacity(catalogue.states.len());
225    for state in &catalogue.states {
226        let evident = stated!(
227            state.cal_k,
228            state.load_k,
229            state.signal,
230            state.reference,
231            state.sub_scan
232        );
233        let matches = |known: &StateEntry| {
234            known.flagged == Some(false)
235                && stated!(known.cal_k, known.load_k)
236                && (known.cal_k.unwrap_or(f64::NAN) - state.cal_k.unwrap_or(f64::NAN)).abs()
237                    < KELVIN
238                && (known.load_k.unwrap_or(f64::NAN) - state.load_k.unwrap_or(f64::NAN)).abs()
239                    < KELVIN
240                && known.obs_mode == state.obs_mode
241                && stated!(known.reference, known.signal, known.sub_scan)
242                && known.reference == state.reference
243                && known.signal == state.signal
244                && known.sub_scan == state.sub_scan
245        };
246        let found = if evident {
247            merged.states.iter().rposition(matches)
248        } else {
249            merged.notice(format!(
250                "state {} is appended: it does not state what matching needs",
251                state.obs_mode
252            ));
253            None
254        };
255        map.push(StateId(found.unwrap_or_else(|| {
256            merged.states.push(state.clone());
257            merged.states.len() - 1
258        }) as u32));
259    }
260    map
261}

263/// Each incoming window matched (casacore `MSSpWindowColumns::matchSpw`) or
264/// appended: of the merged windows in order, those unflagged with the same
265/// channel count and IF chain, a total bandwidth within a quarter of the
266/// incoming one and a reference frequency in the same frame within the
267/// tolerance, the first whose every channel is within it (the first
268/// outright for one channel). `channels` are the merged windows', and
269/// `own` the incoming catalogue's.
270///
271/// The incoming windows are visited as `MSConcat::copySpwAndPol` visits
272/// them, which matters because matching within the tolerance is not
273/// transitive: those its partitions use, in the order they are first used
274/// (an MSv2's partitions are ordered by data description first, so this is
275/// CASA's DATA_DESCRIPTION order), then the others in row order. A window
276/// of a DATA_DESCRIPTION row no MAIN row uses is among the others, which
277/// CASA would visit in that row's place.
278fn windows(
279    merged: &mut Merged,
280    (channels, catalogue, own): (&mut Vec<Channels>, &Catalogue, &[Channels]),
281    tolerance: &MergeTolerance,
282) -> Result<Vec<WindowId>, ObservationError> {
283    let count = catalogue.windows.len();
284    let mut seen = vec![false; count];
285    let mut order = Vec::with_capacity(count);
286    let used = catalogue
287        .partitions
288        .iter()
289        .map(|entry| entry.window.0 as usize);
290    for window in used.chain(0..count) {
291        if window < count && !seen[window] {
292            seen[window] = true;
293            order.push(window);
294        }
295    }
296    let mut map = vec![WindowId(0); count];
297    for index in order {
298        let (window, incoming) = (&catalogue.windows[index], &own[index]);
299        let evident = incoming.is_some()
300            && stated!(
301                window.if_chain,
302                window.total_bandwidth_hz,
303                window.reference_frequency
304            );
305        let found = if evident {
306            window_match(merged, channels, (window, incoming.as_ref()), tolerance)?
307        } else {
308            merged.notice(format!(
309                "window {} is appended: it does not state what matching needs",
310                window.name
311            ));
312            None
313        };
314        map[index] = WindowId(found.unwrap_or_else(|| {
315            merged.windows.push(window.clone());
316            channels.push(incoming.clone());
317            merged.windows.len() - 1
318        }) as u32);
319    }
320    Ok(map)
321}

323/// The merged window `window` (with `incoming` channels) matches, if any.
324fn window_match(
325    merged: &Merged,
326    channels: &[Channels],
327    (window, incoming): (&WindowEntry, Option<&(Vec<f64>, FrequencyRef)>),
328    tolerance: &MergeTolerance,
329) -> Result<Option<usize>, ObservationError> {
330    let (Some((frequencies, _)), Some(bandwidth), Some(reference)) = (
331        incoming,
332        window.total_bandwidth_hz,
333        window.reference_frequency.as_ref(),
334    ) else {
335        return Ok(None);
336    };
337    let near = |a: f64, b: f64| (a - b).abs() <= tolerance.frequency_hz;
338    let candidates =
339        merged
340            .windows
341            .iter()
342            .zip(channels)
343            .enumerate()
344            .filter(|(_, (known, known_channels))| {
345                let Some((known_frequencies, _)) = known_channels else {
346                    return false;
347                };
348                known.flagged == Some(false)
349                    && known_frequencies.len() == frequencies.len()
350                    && known.if_chain.is_some()
351                    && known.if_chain == window.if_chain
352                    && known
353                        .total_bandwidth_hz
354                        .is_some_and(|known| (known - bandwidth).abs() <= bandwidth.abs() / 4.0)
355                    && known.reference_frequency.as_ref().is_some_and(|known| {
356                        known.refer() == reference.refer() && near(known.hz(), reference.hz())
357                    })
358            });
359    for (index, (_, known_channels)) in candidates {
360        let Some((known, _)) = known_channels else {
361            continue;
362        };
363        if frequencies.len() == 1 || known.iter().zip(frequencies).all(|(a, b)| near(*a, *b)) {
364            return Ok(Some(index));
365        }
366        if known
367            .iter()
368            .zip(frequencies.iter().rev())
369            .all(|(a, b)| near(*a, *b))
370        {
371            return Err(ObservationError::Unsupported {
372                what: "merging a window whose channels match another's only reversed",
373            });
374        }
375    }
376    Ok(None)
377}

379/// Each incoming antenna matched (casacore
380/// `MSAntennaColumns::matchAntennaAndStation`, from the last antenna back)
381/// or appended: unflagged, of the same name, of the same station unless the
382/// incoming one states none, each position component within the
383/// tolerance.
384fn antennas(
385    merged: &mut Merged,
386    catalogue: &Catalogue,
387    tolerance: &MergeTolerance,
388) -> Result<Vec<AntennaId>, ObservationError> {
389    let mut map = Vec::with_capacity(catalogue.antennas.len());
390    for antenna in &catalogue.antennas {
391        if merged
392            .antennas
393            .first()
394            .is_some_and(|known| known.position.refer() != antenna.position.refer())
395        {
396            return Err(ObservationError::Unsupported {
397                what: "merging antennas whose positions are in different frames",
398            });
399        }
400        let position = antenna.position.values();
401        let found = merged.antennas.iter().rposition(|known| {
402            let known_position = known.position.values();
403            known.flagged == Some(false)
404                && (antenna.station.is_empty() || known.station == antenna.station)
405                && known.name == antenna.name
406                && known_position
407                    .iter()
408                    .zip(&position)
409                    .all(|(a, b)| (a - b).abs() <= tolerance.position_m)
410        });
411        map.push(AntennaId(found.unwrap_or_else(|| {
412            merged.antennas.push(antenna.clone());
413            merged.antennas.len() - 1
414        }) as u32));
415    }
416    Ok(map)
417}

419/// The great-circle separation of two directions, as casacore computes it
420/// (`MVDirection::separation`).
421pub(crate) fn separation(a: &MDirection, b: &MDirection) -> f64 {
422    let (a, b) = (a.cosines(), b.cosines());
423    let chord = ((a[0] - b[0]).powi(2) + (a[1] - b[1]).powi(2) + (a[2] - b[2]).powi(2)).sqrt();
424    2.0 * (chord / 2.0).min(1.0).asin()
425}

427/// Each incoming field matched (casacore `MSFieldColumns::matchDirection`
428/// as `MSConcat::copyField` calls it) or appended, and whether any matched:
429/// the last merged field of its name is tried first, then every field from
430/// the last back, for one unflagged whose reference, delay and phase
431/// directions are each in the frame of its reference direction and
432/// strictly within the tolerance of the incoming ones; with names
433/// respected, a match of another name is not used. An appended field's
434/// source is the incoming one's until [`sources::append`] maps it.
435fn fields(
436    merged: &mut Merged,
437    catalogue: &Catalogue,
438    tolerance: &MergeTolerance,
439) -> (Vec<FieldId>, bool) {
440    let mut map = Vec::with_capacity(catalogue.fields.len());
441    let mut matched = false;
442    for field in &catalogue.fields {
443        let found = match (&field.delay_centre, &field.reference_centre) {
444            (Some(delay), Some(reference)) => {
445                field_match(merged, field, (delay, reference), tolerance)
446            }
447            _ => {
448                merged.notice(format!(
449                    "field {} is appended: it does not state what matching needs",
450                    field.name
451                ));
452                None
453            }
454        };
455        matched |= found.is_some();
456        map.push(FieldId(found.unwrap_or_else(|| {
457            merged.fields.push(field.clone());
458            merged.fields.len() - 1
459        }) as u32));
460    }
461    (map, matched)
462}

464/// The merged field `field` matches, if any.
465fn field_match(
466    merged: &Merged,
467    field: &FieldEntry,
468    (delay, reference): (&MDirection, &MDirection),
469    tolerance: &MergeTolerance,
470) -> Option<usize> {
471    let matches = |known: &FieldEntry| {
472        let (Some(known_delay), Some(known_reference)) =
473            (&known.delay_centre, &known.reference_centre)
474        else {
475            return false;
476        };
477        let frame: DirectionRef = known_reference.refer();
478        known.flagged == Some(false)
479            && [reference, delay, &field.phase_centre]
480                .iter()
481                .all(|direction| direction.refer() == frame)
482            && separation(known_reference, reference) < tolerance.direction_rad
483            && separation(known_delay, delay) < tolerance.direction_rad
484            && separation(&known.phase_centre, &field.phase_centre) < tolerance.direction_rad
485    };
486    let named = merged
487        .fields
488        .iter()
489        .rposition(|known| known.name == field.name);
490    let found = named
491        .filter(|index| matches(&merged.fields[*index]))
492        .or_else(|| merged.fields.iter().rposition(matches))?;
493    (!tolerance.respect_names || merged.fields[found].name == field.name).then_some(found)
494}

496/// Whether two observing sessions are one (`MSConcat::obsRowsEquivalent`):
497/// their flags, observers, projects, release dates, telescopes and extents
498/// equal, each stated.
499fn same_session(a: &ObservationEntry, b: &ObservationEntry) -> bool {
500    let epoch = |a: &casa_types::measures::MEpoch, b: &casa_types::measures::MEpoch| {
501        a.refer() == b.refer() && a.value().as_mjd() == b.value().as_mjd()
502    };
503    stated!(a.flagged, a.release_date, a.time_range)
504        && a.flagged == b.flagged
505        && a.observers == b.observers
506        && a.project == b.project
507        && a.telescope == b.telescope
508        && b.release_date
509            .as_ref()
510            .zip(a.release_date.as_ref())
511            .is_some_and(|(a, b)| epoch(a, b))
512        && b.time_range
513            .as_ref()
514            .zip(a.time_range.as_ref())
515            .is_some_and(|(a, b)| epoch(&a[0], &b[0]) && epoch(&a[1], &b[1]))
516}

518/// Whether two processors are one (`MSConcat::procRowsEquivalent`): their
519/// flags, modes, types, type IDs and subtypes equal and stated, and their
520/// pass IDs equal (an optional column).
521fn same_processor(a: &ProcessorEntry, b: &ProcessorEntry) -> bool {
522    stated!(a.flagged, a.mode_id, a.kind, a.type_id, a.sub_type)
523        && a.flagged == b.flagged
524        && a.mode_id == b.mode_id
525        && a.kind == b.kind
526        && a.type_id == b.type_id
527        && a.sub_type == b.sub_type
528        && a.pass_id == b.pass_id
529}

531/// `entries` with each one equal (`same`) to an earlier one removed, and
532/// where each old position went: the first equivalent's position among
533/// those kept. CASA maps a duplicate to the last earlier equivalent and
534/// does not close the gaps the removals leave in its observation IDs
535/// (recorded in `docs/CASA (C++) bugs.md`); with an exact predicate the
536/// first equivalent is the same entity.
537fn distinct<T>(entries: Vec<T>, same: impl Fn(&T, &T) -> bool) -> (Vec<T>, Vec<usize>) {
538    let mut kept: Vec<T> = Vec::with_capacity(entries.len());
539    let mut moved = Vec::with_capacity(entries.len());
540    for entry in entries {
541        match kept.iter().position(|known| same(known, &entry)) {
542            Some(index) => moved.push(index),
543            None => {
544                moved.push(kept.len());
545                kept.push(entry);
546            }
547        }
548    }
549    (kept, moved)
550}

552/// The incoming observing sessions appended, then every session made
553/// distinct (`MSConcat::copyObservation` with redundant IDs removed), the
554/// maps of the inputs before remapped, and the counterparts noted.
555fn observations(merged: &mut Merged, catalogue: &Catalogue) {
556    let existing = merged.observations.len();
557    let mut entries = std::mem::take(&mut merged.observations);
558    entries.extend(catalogue.observations.iter().cloned());
559    let (kept, moved) = distinct(entries, same_session);
560    merged.observations = kept;
561    let to = |index: usize| ObservationId(moved[index] as u32);
562    let (input, earlier) = merged.inputs.split_last_mut().expect("this input's map");
563    for map in earlier {
564        for id in &mut map.observations {
565            *id = to(id.0 as usize);
566        }
567    }
568    input.observations = (0..catalogue.observations.len())
569        .map(|index| to(existing + index))
570        .collect();
571    // The earlier sessions keep the first positions, so an incoming one
572    // with a counterpart went below the last of them.
573    let earlier = moved[..existing].iter().max().map_or(0, |last| last + 1);
574    input.counterparts = (0..catalogue.observations.len())
575        .map(|index| moved[existing + index] < earlier)
576        .collect();
577}

579/// The incoming processors appended, then every processor made distinct
580/// (`MSConcat::copyProcessor` with redundant IDs removed), the maps of
581/// the inputs before remapped.
582fn processors(merged: &mut Merged, catalogue: &Catalogue) {
583    let existing = merged.processors.len();
584    let mut entries = std::mem::take(&mut merged.processors);
585    entries.extend(catalogue.processors.iter().cloned());
586    let (kept, moved) = distinct(entries, same_processor);
587    merged.processors = kept;
588    let to = |index: usize| ProcessorId(moved[index] as u32);
589    let (input, earlier) = merged.inputs.split_last_mut().expect("this input's map");
590    for map in earlier {
591        for id in &mut map.processors {
592            *id = to(id.0 as usize);
593        }
594    }
595    input.processors = (0..catalogue.processors.len())
596        .map(|index| to(existing + index))
597        .collect();
598}
