2//! What `close` holds while it writes MAIN, admitted against the session's
3//! resource policy before anything is written (plan §4: the runtime's
4//! budget, not a second one).
5//!
6//! In order:
7//! 1. **The writer.** The MeasurementSet write plan is sized from half of
8//!    what the policy leaves free, and its resident bytes are admitted with
9//!    the subtable columns its save generates (POINTING's runs, a few
10//!    hundred bytes a scan). Its batches are tile bands of about
11//!    [`casa_ms::TARGET_BATCH_BYTES`], so the budget is a ceiling that only
12//!    a small budget reaches.
13//! 2. **The pipeline.** The batch being filled, the geometry of the
14//!    integrations it spans, each worker's scratch and, when the session
15//!    predicts, the predictor, its block of native rows and (on Metal) the
16//!    device copies of the model grids. A block is the whole batch when
17//!    that fits with the policy's streaming workers; otherwise blocks are
18//!    small, and as many workers as fit what is then free are admitted
19//!    (#737's rule: workers shrink first). With one worker that does not
20//!    fit, `close` is refused.
21//!
22//! The model grids themselves were admitted by `predict`, and are held.
23//! A refusal is a typed [`Admission`] carried by the session error.
24

39/// The writer's phase, as an admission refusal names it.
40pub(super) const WRITER_PHASE: &str = "the simulator's MeasurementSet writer";
41
42/// The pipeline's phase, as an admission refusal names it.
43pub(super) const PIPELINE_PHASE: &str = "the simulator's geometry, prediction and noise";
44
45/// Batches the writer holds at once: the one it writes and one queued.
46const WRITER_LIVE_BATCHES: usize = 2;
47
48/// A writer budget larger than any plan's needs, to tell a budget too small
49/// from a plan that cannot be made.
50const UNBOUNDED_WRITER_BYTES: usize = 1 << 50;
51
52/// Scratch of one worker's integration, per antenna: the antennas' uvw,
53/// their shadow flags and elevations, and for `tsys-manual` noise their
54/// airmasses (and elevations at a moved pointing's phase centre), with room
55/// for the measures frames.
56const ANTENNA_SCRATCH_BYTES: u64 = 64;

58/// The MAIN table `close` writes.
59#[derive(Debug, Clone, Copy)]
60pub(super) struct MainShape<'a> {
61    pub(super) rows: usize,
62    pub(super) correlations: usize,
63    pub(super) channels: usize,
64    pub(super) telescope: &'a str,
65    /// Rows of the session's largest integration.
66    pub(super) largest_integration: usize,
67    /// Antennas of the session's largest scan.
68    pub(super) antennas: usize,
69    /// Heap of the subtable columns the save generates (POINTING's), which
70    /// grows with the scans.
71    pub(super) generated_bytes: u64,
72    /// What a batch's `tsys-atm` noise holds beside the cells.
73    pub(super) noise_scales: NoiseScales,
74}

76/// What a batch's `tsys-atm` noise holds beside the cells: its antenna
77/// scales ([`BatchScales`]) and each worker's σ of a row
78/// ([`NoiseScratch`]).
79#[derive(Debug, Clone, Copy, Default)]
80pub(super) struct NoiseScales {
81    /// The plan's `tsys-atm` terms.
82    terms: usize,
83    /// Channels of the cells.
84    channels: usize,
85    /// Rows of the smallest integration, and antennas of the largest, when
86    /// the rows come in whole integrations ([`BatchScales::most_entries`]).
87    integrations: Option<(usize, usize)>,
88}
89
90impl NoiseScales {
91    /// What `plan`'s noise holds, for cells of `channels` channels and the
92    /// rows' `integrations`. Every `corrupt` covers scan 0, so its noise is
93    /// all the plan's.
94    pub(super) fn of(
95        plan: &DataPlan,
96        channels: usize,
97        integrations: Option<(usize, usize)>,
98    ) -> Self {
99        let mut models: Vec<*const TsysAtm> = plan
100            .noise_for_scan(0)
101            .iter()
102            .filter_map(|application| application.tsys_atm().map(std::ptr::from_ref))
103            .collect();
104        models.sort_unstable();
105        models.dedup();
106        Self {
107            terms: models.len(),
108            channels,
109            integrations,
110        }
111    }
112
113    /// The most antenna-scale entries a batch of `rows` rows needs: the room
114    /// its table reserves.
115    pub(super) fn entries(&self, rows: usize) -> usize {
116        BatchScales::most_entries(self.terms, rows, self.integrations)
117    }
118
119    /// The most a batch of `rows` rows holds with `workers` workers.
120    pub(super) fn bytes(&self, rows: usize, workers: usize) -> u64 {
121        if self.terms == 0 {
122            return 0;
123        }
124        BatchScales::bytes(self.entries(rows), self.channels)
125            + workers as u64 * NoiseScratch::bytes(self.channels)
126    }
127}

129/// What `close` was admitted: the write plan, the workers that fit, and the
130/// memory, held until the write ends.
131#[derive(Debug)]
132pub(super) struct MainAdmission {
133    pub(super) plan: MeasurementSetWritePlan,
134    pub(super) cores: CoreAllocation,
135    /// The policy's allocation, before admission shrank it.
136    pub(super) policy_cores: CoreAllocation,
137    /// Rows of a prediction block: the write batch's, or fewer when the
138    /// budget is small.
139    pub(super) block_rows: usize,
140    pub(super) reservation: Reservation,
141}
142
143/// Admit `close`'s writer and pipeline for `shape` within `resources`, the
144/// pipeline predicting with `prediction` when the session predicts.
145///
146/// # Errors
147///
148/// [`SimulatorError::admission`] when the writer, or the pipeline with one
149/// worker, does not fit what the policy leaves free; a raised error when the
150/// MAIN write cannot be planned at all.
151pub(super) fn admit_main_write(
152    resources: &SessionResources,
153    shape: &MainShape<'_>,
154    prediction: Option<&PredictionStage<'_>>,
155) -> SimulatorResult<MainAdmission> {
156    let (host, policy) = (resources.host(), resources.policy());
157    let free = free_memory(host, policy);
158    let plan = writer_plan(shape, to_usize(free / 2)).map_err(|error| {
159        let required =
160            minimum_writer_bytes(shape).map_or(u64::MAX, |bytes| (bytes as u64).saturating_mul(2));
161        refusal_or(error, WRITER_PHASE, required, free)
162    })?;
163    let mut reservation = admit(
164        host,
165        policy,
166        &Demand {
167            phase: WRITER_PHASE,
168            memory: plan.maximum_resident_bytes() as u64 + shape.generated_bytes,
169        },
170    )
171    .map_err(SimulatorError::admission)?;
172    let policy_cores = resources
173        .close_cores(prediction.is_some_and(|stage| stage.backend() == BackendChoice::Metal));
174    let free = free_memory(host, policy);
175    let batch_rows = plan.batch_rows();
176    let demand = |cores: CoreAllocation, block_rows: usize| {
177        pipeline_bytes(shape, batch_rows, cores)
178            + prediction.map_or(0, |stage| stage.bytes(block_rows, cores.streaming_workers))
179    };
180    // A batch predicted as one block when that fits with the policy's
181    // workers; else small blocks, and the workers shrink first (#737).
182    let (cores, block_rows) = if demand(policy_cores, batch_rows) <= free {
183        (policy_cores, batch_rows)
184    } else {
185        let small = prediction.map_or(batch_rows, |stage| stage.small_block_rows(batch_rows));
186        let cores = (1..=policy_cores.streaming_workers)
187            .rev()
188            .map(|workers| policy_cores.at_most(workers))
189            .find(|cores| demand(*cores, small) <= free)
190            .unwrap_or_else(|| policy_cores.at_most(1));
191        (cores, small)
192    };
193    let pipeline = admit(
194        host,
195        policy,
196        &Demand {
197            phase: PIPELINE_PHASE,
198            memory: demand(cores, block_rows),
199        },
200    )
201    .map_err(SimulatorError::admission)?;
202    reservation.join(pipeline);
203    Ok(MainAdmission {
204        plan,
205        cores,
206        policy_cores,
207        block_rows,
208        reservation,
209    })
210}

212/// The MAIN write plan with `available_bytes` for the writer.
213pub(super) fn writer_plan(
214    shape: &MainShape<'_>,
215    available_bytes: usize,
216) -> Result<MeasurementSetWritePlan, MeasurementSetWriteError> {
217    MeasurementSetWritePlan::visibility_creation(
218        shape.rows,
219        shape.correlations,
220        shape.channels,
221        shape.telescope,
222        MeasurementSetWriteResources {
223            available_bytes,
224            maximum_live_batches: WRITER_LIVE_BATCHES,
225            tiled_column_buffer_bytes: STREAMING_TILED_COLUMN_BUFFER_BYTES,
226        },
227    )
228}

230/// The fewest writer bytes that plan `shape`'s MAIN write; `None` when no
231/// budget does.
232fn minimum_writer_bytes(shape: &MainShape<'_>) -> Option<usize> {
233    writer_plan(shape, UNBOUNDED_WRITER_BYTES).ok()?;
234    // Whether a budget plans is monotone in the budget.
235    let (mut refused, mut planned) = (0, UNBOUNDED_WRITER_BYTES);
236    while planned - refused > 1 {
237        let middle = refused + (planned - refused) / 2;
238        if writer_plan(shape, middle).is_ok() {
239            planned = middle;
240        } else {
241            refused = middle;
242        }
243    }
244    Some(planned)
245}

247/// A plan refused for its budget as an [`Admission`] of `required` bytes
248/// against `free`, and any other planning failure as raised.
249fn refusal_or(
250    error: MeasurementSetWriteError,
251    phase: &'static str,
252    required: u64,
253    free: u64,
254) -> SimulatorError {
255    if required == u64::MAX {
256        return SimulatorError::raised(format!("plan the MAIN write: {error}"));
257    }
258    SimulatorError::admission(Admission {
259        phase,
260        required,
261        available: free,
262    })
263}

265/// Bytes the pipeline holds beside the writer, with `cores`' streaming
266/// workers: the batch being filled, the geometry of the integrations it
267/// spans (with the unwritten rows of the one before), the list of those
268/// integrations, one integration's scalar runs, each worker's scratch, and
269/// the batch's `tsys-atm` antenna scales.
270pub(super) fn pipeline_bytes(
271    shape: &MainShape<'_>,
272    batch_rows: usize,
273    cores: CoreAllocation,
274) -> u64 {
275    let samples = (shape.correlations * shape.channels) as u64;
276    let rows = batch_rows as u64;
277    let largest = shape.largest_integration as u64;
278    let size = |bytes: usize| bytes as u64;
279    // DATA, FLAG_ROW, UVW, WEIGHT and SIGMA.
280    let row_bytes = samples * size(size_of::<Complex32>()) + 1 + 24 + 4 + 4;
281    let batch = rows * row_bytes;
282    let geometry = (rows + 2 * largest) * size(size_of::<RowGeometry>());
283    let integrations = (rows + 2) * size(size_of::<Integration>());
284    let scalars = largest * (4 + 4 + 1);
285    let workers = cores.streaming_workers as u64 * shape.antennas as u64 * ANTENNA_SCRATCH_BYTES;
286    let noise = shape
287        .noise_scales
288        .bytes(batch_rows, cores.streaming_workers);
289    batch + geometry + integrations + scalars + workers + noise
290}
291
292fn to_usize(bytes: u64) -> usize {
293    usize::try_from(bytes).unwrap_or(usize::MAX)
294}

296#[cfg(test)]
297mod tests {
298    use casa_imaging_runtime::{HostResources, ResourcePolicy};
299
300    use super::*;
301    use crate::session::BackendRequest;
302
303    fn shape() -> MainShape<'static> {
304        MainShape {
305            rows: 100_000,
306            correlations: 2,
307            channels: 64,
308            telescope: "ALMA",
309            largest_integration: 1_000,
310            antennas: 44,
311            generated_bytes: 0,
312            noise_scales: NoiseScales::default(),
313        }
314    }
315
316    fn resources(memory: u64, workers: usize) -> SessionResources {
317        SessionResources::pinned(
318            HostResources {
319                threads: 8,
320                performance_cores: 8,
321                available_memory: 8 << 30,
322                metal: false,
323            },
324            ResourcePolicy::Explicit { workers, memory },
325            BackendRequest::Cpu,
326        )
327        .expect("resources")
328    }
329
330    #[test]
331    fn the_writer_takes_half_the_budget_and_the_pipeline_fits_the_rest() {
332        let admitted = admit_main_write(&resources(1 << 30, 5), &shape(), None).expect("fits");
333        assert_eq!(admitted.cores.streaming_workers, 4);
334        let writer = admitted.plan.maximum_resident_bytes() as u64;
335        assert!(writer <= 1 << 29);
336        assert_eq!(
337            admitted.reservation.memory(),
338            writer + pipeline_bytes(&shape(), admitted.plan.batch_rows(), admitted.cores)
339        );
340    }
341
342    #[test]
343    fn a_budget_too_small_for_the_writer_is_a_typed_refusal() {
344        let error =
345            admit_main_write(&resources(1 << 20, 4), &shape(), None).expect_err("too small");
346        let refusal = error.admission_refused().expect("an admission refusal");
347        assert_eq!(refusal.phase, WRITER_PHASE);
348        assert_eq!(refusal.available, 1 << 20);
349        let minimum = minimum_writer_bytes(&shape()).expect("plannable") as u64;
350        assert_eq!(refusal.required, 2 * minimum);
351        assert!(writer_plan(&shape(), to_usize(minimum)).is_ok());
352        assert!(writer_plan(&shape(), to_usize(minimum - 1)).is_err());
353    }
354
355    /// Admission state is the process's, so these laws assume no other
356    /// test holds memory at once (nextest runs each test in its own
357    /// process).
358    #[test]
359    fn workers_shrink_before_the_pipeline_is_refused() {
360        // A budget of twice the writer's minimum, and arrays wide enough
361        // that each worker's scratch is a third of what the writer leaves:
362        // two workers fit, four do not.
363        let writer = minimum_writer_bytes(&shape()).expect("plannable");
364        let plan = writer_plan(&shape(), writer).expect("plan");
365        let budget = 2 * writer as u64;
366        let left = budget - plan.maximum_resident_bytes() as u64;
367        let shape = MainShape {
368            antennas: to_usize(left / 3 / ANTENNA_SCRATCH_BYTES),
369            ..shape()
370        };
371        let four = CoreAllocation {
372            workers: 5,
373            streaming_workers: 4,
374        };
375        let two = pipeline_bytes(&shape, plan.batch_rows(), four.at_most(2));
376        assert!(two <= left, "two workers fit: {two} of {left}");
377        assert!(pipeline_bytes(&shape, plan.batch_rows(), four.at_most(3)) > left);
378        let admitted = admit_main_write(&resources(budget, 5), &shape, None).expect("two fit");
379        assert_eq!(admitted.policy_cores, four);
380        assert_eq!(admitted.cores, four.at_most(2));
381        drop(admitted);
382        let error = admit_main_write(
383            &resources(budget, 5),
384            &MainShape {
385                antennas: shape.antennas * 4,
386                ..shape
387            },
388            None,
389        )
390        .expect_err("one worker does not fit");
391        let refusal = error.admission_refused().expect("an admission refusal");
392        assert_eq!(refusal.phase, PIPELINE_PHASE);
393    }
