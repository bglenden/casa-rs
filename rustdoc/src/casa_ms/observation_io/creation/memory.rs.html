2//! The memory a plan holds, admitted against
3//! [`CreationOptions::memory_bytes`](super::CreationOptions::memory_bytes)
4//! before it is allocated.
5//!
6//! Making a plan holds the `HISTORY` rows it copies, each input's opening
7//! (its tables, the rows its readers load, its row index and catalogue),
8//! the merged and written catalogues, the scans that find losses and
9//! pointing, each output partition's grids while they are combined, their
10//! combined times until the rows are counted and, with padding omitted, a
11//! weights batch and the cells acquired. Writing then holds, beside what
12//! the plan keeps, the subtables, `HISTORY` and `POINTING` until the save
13//! (#950), the pointing reads, the MAIN writers and one batch of every
14//! selected group. Each is admitted before it is allocated, so a plan that
15//! would exceed the budget is refused
16//! ([`ObservationError::OverBudget`]) with nothing more held, and the plan
17//! states what it admitted ([`MemoryCharges`]).
18//!
19//! What only building something gives its size is admitted as a bound
20//! first and trimmed to what was built: an opening's tables and catalogue
21//! (bounded by their files and by the rows the readers load), the merge
22//! (by the inputs' catalogues together) and the written catalogue (by the
23//! merged one). Catalogues are measured at their strings' and arrays'
24//! capacities (`heap::catalogue_bytes`); the counting law
25//! (`creation_plan_memory.rs`) holds the plan to its charges.
26

39/// Bytes a `HISTORY` row's record holds in its table beside its strings:
40/// its nine fields' names and values, and the arrays' headers.
41const HISTORY_ROW_BYTES: usize = 1 << 10;
42
43/// Bytes a subtable row's record holds in memory beside the strings and
44/// arrays of its entity: its fields' names, values and array headers.
45const SUBTABLE_ROW_BYTES: usize = 2 << 10;

47/// The memory a plan admitted ([`super::CreationPlan::charges`]), in bytes.
48#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
49pub struct MemoryCharges {
50    /// What the plan keeps while it lives: the `HISTORY` rows to write,
51    /// each input's opening (its tables, index and catalogue) and the
52    /// merged and written catalogues.
53    pub kept: usize,
54    /// The most making the plan held, `kept` included.
55    pub planning: usize,
56    /// What writing holds beyond `kept`: the subtables, `HISTORY` and
57    /// `POINTING` until the save, then the largest of the pointing reads,
58    /// the MAIN writers with one batch, and an ephemeris table.
59    pub writing: usize,
60    /// Of `writing`, the MAIN writers': their tile bands and I/O buffers,
61    /// and the keys of one batch's rows.
62    pub writer: usize,
63    /// Of `writing`, one time range's batches of every selected group.
64    pub batch: usize,
65    /// The least [`CreationOptions::memory_bytes`](super::CreationOptions)
66    /// the plan is made with: its planning, and its writing with the MAIN
67    /// writers at their least.
68    pub least: usize,
69    /// The least [`CreationOptions::batch_bytes`](super::CreationOptions):
70    /// one integration (one time of every group read) of the widest input
71    /// partition.
72    pub integration: usize,
73}

75/// The bytes admitted so far against a budget, and the most at once.
76pub(super) struct Ledger {
77    budget: usize,
78    held: usize,
79    peak: usize,
80}
81
82impl Ledger {
83    pub(super) fn new(budget: usize) -> Self {
84        Self {
85            budget,
86            held: 0,
87            peak: 0,
88        }
89    }
90
91    /// Admit `bytes` more for `what`, refused beyond the budget.
92    pub(super) fn admit(
93        &mut self,
94        what: &'static str,
95        bytes: usize,
96    ) -> Result<(), ObservationError> {
97        let required = self.held.saturating_add(bytes);
98        if required > self.budget {
99            return Err(ObservationError::OverBudget {
100                what,
101                required,
102                budget: self.budget,
103            });
104        }
105        self.held = required;
106        self.peak = self.peak.max(required);
107        Ok(())
108    }
109
110    /// Release `bytes` admitted earlier.
111    pub(super) fn release(&mut self, bytes: usize) {
112        self.held = self.held.saturating_sub(bytes);
113    }
114
115    /// Trim `bound` bytes, admitted for `what` before what they bound was
116    /// allocated, to the `held` bytes it holds.
117    pub(super) fn trim(
118        &mut self,
119        what: &'static str,
120        bound: usize,
121        held: usize,
122    ) -> Result<(), ObservationError> {
123        if held <= bound {
124            self.release(bound - held);
125            Ok(())
126        } else {
127            self.admit(what, held - bound)
128        }
129    }
130
131    /// Admit `bytes` for `what` while `run` runs.
132    pub(super) fn during<T>(
133        &mut self,
134        what: &'static str,
135        bytes: usize,
136        run: impl FnOnce(&mut Self) -> Result<T, ObservationError>,
137    ) -> Result<T, ObservationError> {
138        self.admit(what, bytes)?;
139        let value = run(self);
140        self.release(bytes);
141        value
142    }
143
144    pub(super) fn held(&self) -> usize {
145        self.held
146    }
147
148    pub(super) fn peak(&self) -> usize {
149        self.peak
150    }
151
152    pub(super) fn budget(&self) -> usize {
153        self.budget
154    }
155}

157/// The memory a creation may hold by default: an eighth of the host's
158/// physical memory, the share casa-ms's readers and writers plan with
159/// ([`MsSelectionIoBudget::from_system_memory`]).
160///
161/// # Errors
162///
163/// When the host's physical memory is unknown.
164pub fn default_memory_bytes() -> Result<usize, ObservationError> {
165    MsSelectionIoBudget::from_system_memory(1, 1, None)
166        .map(|budget| budget.available_bytes)
167        .map_err(|error| request_error(error.to_string()))
168}

170/// The bytes an owned copy of a partition's grid of `times` times and
171/// `baselines` baselines holds while it is made: its times, fields and
172/// scans and its baselines, and the MSv2 adapter's chunk of row numbers
173/// and intervals ([`GRID_READ_ROWS`] at most).
174pub(super) fn grid_bytes(times: usize, baselines: usize) -> usize {
175    let per_time = size_of::<f64>() + size_of::<FieldId>() + size_of::<ScanId>();
176    let read = times.saturating_mul(baselines).min(GRID_READ_ROWS)
177        * (size_of::<usize>() + size_of::<f64>());
178    times * per_time + baselines * size_of::<OrderedBaseline>() + read
179}

181/// What the merge and the combination of catalogues holding
182/// `catalogues` bytes together may hold: the merged catalogue, the
183/// combination's partitions beside it until it finishes, and the maps and
184/// notices. Each holds no more than the inputs' catalogues, whose entities
185/// it matches, appends or combines as clones.
186pub(super) fn merge_bound(catalogues: usize) -> usize {
187    3 * catalogues
188}
189
190/// What the subtables written from `catalogue` hold in memory, until
191/// they are saved: a record per row ([`SUBTABLE_ROW_BYTES`]; `FEED`'s an
192/// antenna's in each window, and a data description, a polarization and a
193/// state of intents for each partition) and the entities' strings and
194/// arrays again.
195pub(super) fn subtables_bytes(catalogue: &Catalogue) -> usize {
196    let windows = catalogue.windows.len();
197    let records: usize = catalogue
198        .sources
199        .iter()
200        .map(|source| source.records.len())
201        .sum();
202    let rows = catalogue.antennas.len() * (2 + windows)
203        + catalogue.fields.len()
204        + windows
205        + 3 * catalogue.partitions.len()
206        + records
207        + catalogue.states.len()
208        + catalogue.processors.len()
209        + catalogue.observations.len();
210    rows * SUBTABLE_ROW_BYTES + catalogue_bytes(catalogue)
211}

213/// What the catalogue a plan states it writes may hold
214/// ([`super::CreationPlan::output`]): the merged catalogue's entities with
215/// a state for each partition's intents, twice for the capacities reading
216/// them back leaves.
217pub(super) fn written_bound(combined: &Combined) -> usize {
218    let states: usize = combined
219        .intents
220        .iter()
221        .zip(&combined.modes)
222        .filter(|(intents, _)| **intents)
223        .map(|(_, mode)| size_of::<StateEntry>() + mode.len())
224        .sum();
225    2 * (catalogue_bytes(&combined.catalogue) + states)
226}

228/// The heap `combined` holds: its catalogue ([`catalogue_bytes`]), each
229/// input's row map, and the output partitions' keys, contributors, units,
230/// losses and the merge's notices.
231pub(super) fn combined_bytes(combined: &Combined) -> usize {
232    let Combined {
233        catalogue,
234        rows,
235        intents,
236        modes,
237        contributors,
238        units,
239        losses,
240        notices,
241    } = combined;
242    catalogue_bytes(catalogue)
243        + entries(rows, |map| {
244            let RowMap {
245                antennas,
246                fields,
247                scans,
248                states,
249                partitions,
250                weight_scale: _,
251            } = map;
252            plain(antennas)
253                + plain(fields)
254                + plain(scans)
255                + plain(states)
256                + entries(partitions, placement_bytes)
257        })
258        + plain(intents)
259        + strings(modes)
260        + entries(contributors, plain)
261        + entries(units, units_bytes)
262        + entries(losses, |loss| match loss {
263            Loss::DataGroup(what)
264            | Loss::Table(what)
265            | Loss::Column(what)
266            | Loss::Defaulted(what) => what.capacity(),
267            Loss::Sigma
268            | Loss::SigmaDerived
269            | Loss::RowFlags
270            | Loss::FlagCategories
271            | Loss::UnusedAntennas(_) => 0,
272        })
273        + strings(notices)
274}

276fn placement_bytes(placement: &Placement) -> usize {
277    let Placement {
278        output: _,
279        state,
280        processor: _,
281        scan_offset: _,
282        groups,
283        columns,
284    } = placement;
285    let states = match state {
286        StateIds::PerRow(states) => plain(states),
287        StateIds::One(_) | StateIds::Intents => 0,
288    };
289    states + strings(groups) + plain(columns)
290}

292/// The heap a copy of `history` holds: its rows and their strings.
293pub(super) fn history_bytes(history: &[HistoryRow]) -> usize {
294    history
295        .iter()
296        .map(|row| {
297            let HistoryRow {
298                time_s: _,
299                observation: _,
300                message,
301                priority,
302                origin,
303                application,
304                parameters,
305                command,
306            } = row;
307            size_of::<HistoryRow>()
308                + message.len()
309                + priority.len()
310                + origin.len()
311                + application.len()
312                + parameters
313                    .iter()
314                    .chain(command)
315                    .map(|line| size_of::<String>() + line.len())
316                    .sum::<usize>()
317        })
318        .sum()
319}

321/// What the created `HISTORY` table holds of `history` until it is saved:
322/// each row's record ([`HISTORY_ROW_BYTES`]) and its strings, and the
323/// strings again as the save encodes them.
324pub(super) fn history_table_bytes(history: &[HistoryRow]) -> usize {
325    history.len() * HISTORY_ROW_BYTES + 2 * history_bytes(history)
326}

328#[cfg(test)]
329mod tests {
330    use casa_observation::ObservationError;
331
332    use super::Ledger;
333
334    #[test]
335    fn a_ledger_refuses_beyond_its_budget_with_nothing_more_held() {
336        let mut ledger = Ledger::new(100);
337        ledger.admit("first", 60).expect("within");
338        let refused = ledger.admit("second", 41);
339        assert!(matches!(
340            refused,
341            Err(ObservationError::OverBudget {
342                what: "second",
343                required: 101,
344                budget: 100
345            })
346        ));
347        assert_eq!(ledger.held(), 60);
348        ledger
349            .during("transient", 40, |ledger| {
350                assert_eq!(ledger.held(), 100);
351                Ok(())
352            })
353            .expect("within");
354        assert_eq!((ledger.held(), ledger.peak()), (60, 100));
355        ledger.admit("bounded", 30).expect("within");
356        ledger
357            .trim("bounded", 30, 10)
358            .expect("a bound trimmed down");
359        assert_eq!(ledger.held(), 70);
360        ledger.release(70);
361        assert_eq!(ledger.held(), 0);
362    }
