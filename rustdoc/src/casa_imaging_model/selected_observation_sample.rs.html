142/// One source-sample contribution to a compiled output spectral channel.
143///
144/// The factor is the reconstruction-evaluated interpolation or integration
145/// coefficient. Cubic coefficients may be signed. This reported value carries no traversal authority
146/// and is deliberately outside [`SelectedObservationSample`]'s persisted
147/// schema and content identity.
148#[derive(Debug, Clone, Copy, PartialEq)]
149pub struct SelectedSpectralContribution {
150    output_channel: u32,
151    factor: f64,
152    evaluation_frequency_hz: f64,
153}
154
155impl SelectedSpectralContribution {
156    /// Construct one finite, non-zero contribution at its
157    /// owner-evaluated frequency in the compiled output frame.
158    #[must_use]
159    pub fn new(output_channel: u32, factor: f64, evaluation_frequency_hz: f64) -> Option<Self> {
160        (factor.is_finite()
161            && factor != 0.0
162            && evaluation_frequency_hz.is_finite()
163            && evaluation_frequency_hz > 0.0)
164            .then_some(Self {
165                output_channel,
166                factor,
167                evaluation_frequency_hz,
168            })
169    }
170
171    /// Return the zero-based compiled output-channel index.
172    #[must_use]
173    pub const fn output_channel(self) -> u32 {
174        self.output_channel
175    }
176
177    /// Return the interpolation or averaging coefficient.
178    #[must_use]
179    pub const fn factor(self) -> f64 {
180        self.factor
181    }
182
183    /// Return the source sample frequency evaluated in the compiled output frame.
184    #[must_use]
185    pub const fn evaluation_frequency_hz(self) -> f64 {
186        self.evaluation_frequency_hz
187    }
188}

190/// Compact sparse output-channel stencil compiled for one selected sample.
191///
192/// Four terms remain inline for nearest, linear, and cubic sampling. A
193/// planner-bounded integration stencil may spill only when its declared bound
194/// exceeds that inline capacity. Empty coverage is represented explicitly.
195#[derive(Debug, Clone, PartialEq)]
196pub struct SelectedSpectralContributions {
197    entries: SmallVec<[SelectedSpectralContribution; 4]>,
198}
199
200impl SelectedSpectralContributions {
201    /// Construct a compact contribution set with distinct output channels in
202    /// reconstruction-reported order.
203    #[must_use]
204    pub fn new<I, T>(entries: I) -> Option<Self>
205    where
206        I: IntoIterator<Item = T>,
207        T: Into<Option<SelectedSpectralContribution>>,
208    {
209        let entries = entries
210            .into_iter()
211            .filter_map(Into::into)
212            .collect::<SmallVec<[_; 4]>>();
213        if entries.iter().enumerate().any(|(index, entry)| {
214            entries[..index]
215                .iter()
216                .any(|prior| prior.output_channel == entry.output_channel)
217        }) {
218            return None;
219        }
220        Some(Self { entries })
221    }
222
223    /// Return an empty contribution set.
224    #[must_use]
225    pub fn empty() -> Self {
226        Self {
227            entries: SmallVec::new(),
228        }
229    }
230
231    /// Iterate through contributions in owner-reported order.
232    pub fn iter(&self) -> impl Iterator<Item = SelectedSpectralContribution> + '_ {
233        self.entries.iter().copied()
234    }
235
236    /// Return the exact number of non-zero terms.
237    #[must_use]
238    pub fn len(&self) -> usize {
239        self.entries.len()
240    }
241
242    /// Return whether this stencil has no mapped output support.
243    #[must_use]
244    pub fn is_empty(&self) -> bool {
245        self.entries.is_empty()
246    }
247}

249/// One channel interval in a named spectral frame.
250#[derive(Debug, Clone, Copy, PartialEq)]
251pub struct SelectedSpectralInterval {
252    centre_hz: f64,
253    first_boundary_hz: f64,
254    second_boundary_hz: f64,
255}
256
257impl SelectedSpectralInterval {
258    /// Construct a finite positive centre and two distinct finite positive boundaries.
259    #[must_use]
260    pub fn new(centre_hz: f64, first_boundary_hz: f64, second_boundary_hz: f64) -> Option<Self> {
261        (centre_hz.is_finite()
262            && centre_hz > 0.0
263            && first_boundary_hz.is_finite()
264            && first_boundary_hz > 0.0
265            && second_boundary_hz.is_finite()
266            && second_boundary_hz > 0.0
267            && first_boundary_hz != second_boundary_hz)
268            .then_some(Self {
269                centre_hz,
270                first_boundary_hz,
271                second_boundary_hz,
272            })
273    }
274
275    /// Return the channel centre.
276    #[must_use]
277    pub const fn centre_hz(self) -> f64 {
278        self.centre_hz
279    }
280
281    /// Return boundaries in original channel order, preserving descending axes.
282    #[must_use]
283    pub const fn boundaries_hz(self) -> [f64; 2] {
284        [self.first_boundary_hz, self.second_boundary_hz]
285    }
286
287    /// Return the positive channel width.
288    #[must_use]
289    pub fn width_hz(self) -> f64 {
290        (self.second_boundary_hz - self.first_boundary_hz).abs()
291    }
292}

345/// One backend-independent selected-observation sample report.

347/// This is a closed value schema only. Constructing it does not mint content

381/// Content digest of one ordered selected-observation sample sequence.
382///
383/// This ID is minted while the selected-observation owner validates and
384/// exhausts the canonical sample stream. It remains distinct from retained
385/// source access, traversal-completion evidence, and prediction destination.
386/// Prediction and residual consumers of the same selected values therefore
387/// bind the same generation.
388///
389/// Physical block boundaries are absent, but iteration order is identity
390/// bearing. The traversal owner must therefore emit one canonical logical
391/// sample order for every legal physical schedule.
392///
393/// There is deliberately no public constructor from raw digest bytes or
394/// caller-supplied sample reports.
395///
396/// ```compile_fail
397/// use casa_imaging_model::SelectedObservationGenerationId;
398///
399/// let _ = SelectedObservationGenerationId::from_sha256([0; 32]);
400/// ```
401///
402/// ```compile_fail
403/// use casa_imaging_model::{
404///     SelectedObservationGenerationId, SelectedObservationSample,
405/// };
406///
407/// let samples = std::iter::empty::<&SelectedObservationSample>();
408/// let _ = SelectedObservationGenerationId::from_samples(samples);
409/// ```
410#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
411pub struct SelectedObservationGenerationId([u8; 32]);
412
413impl SelectedObservationGenerationId {
414    /// Content identity schema version.
415    pub const SCHEMA_VERSION: u32 = SELECTED_OBSERVATION_GENERATION_VERSION;
416
417    /// Return the exact SHA-256 digest.
418    #[must_use]
419    pub const fn as_bytes(self) -> [u8; 32] {
420        self.0
421    }
422}

424pub(crate) struct SelectedObservationGenerationEncoder {
425    encoder: CanonicalEncoder,
426    sample_count: u64,
427}
428
429impl SelectedObservationGenerationEncoder {
430    pub(crate) fn new() -> Self {
431        let mut encoder = CanonicalEncoder::new();
432        encoder.bytes(SELECTED_OBSERVATION_GENERATION_DOMAIN);
433        encoder.u32(SELECTED_OBSERVATION_GENERATION_VERSION);
434        Self {
435            encoder,
436            sample_count: 0,
437        }
438    }
439
440    pub(crate) fn push(&mut self, sample: &SelectedObservationSample) {
441        self.encoder.u8(0xa5);
442        encode_sample(&mut self.encoder, sample);
443        self.sample_count = self
444            .sample_count
445            .checked_add(1)
446            .expect("selected-observation sample count fits u64");
447    }
448
449    pub(crate) fn finish(mut self) -> (SelectedObservationGenerationId, u64) {
450        self.encoder.u8(0xff);
451        self.encoder.u64(self.sample_count);
452        (
453            SelectedObservationGenerationId(self.encoder.finish()),
454            self.sample_count,
455        )
456    }
457}

473fn encode_sample(encoder: &mut CanonicalEncoder, sample: &SelectedObservationSample) {
474    let address = sample.address;
475    encoder.i32(address.data_description_id);
476    encoder.u32(address.spectral_window_id);
477    encoder.u32(address.channel_index);
478    encoder.f64(address.frequency_centre_hz);
479    encoder.f64(address.frequency_lower_hz);
480    encoder.f64(address.frequency_upper_hz);
481    encoder.f64(address.channel_width_hz);
482    encoder.u8(frequency_frame_tag(address.frequency_frame));
483    encoder.u32(address.polarization_id);
484    encoder.u32(address.correlation_index);
485    encoder.u8(correlation_type_tag(address.correlation_type));
486
487    match sample.visibility {
488        SelectedVisibilitySample::Float32(value) => {
489            encoder.u8(0);
490            encoder.f32(value);
491        }
492        SelectedVisibilitySample::Complex32([real, imaginary]) => {
493            encoder.u8(1);
494            encoder.f32(real);
495            encoder.f32(imaginary);
496        }
497    }
498    encoder.u8(u8::from(sample.channel_flag));
499    encoder.u8(u8::from(sample.parallel_hand_group_flag));
500    encoder.u8(u8::from(sample.row_flag));
501    encoder.f32(sample.input_weight);
502
503    let coordinates = sample.coordinates;
504    for value in coordinates.raw_uvw_m {
505        encoder.f64(value);
506    }
507    for value in coordinates.density_uvw_m {
508        encoder.f64(value);
509    }
510    for value in coordinates.transformed_uvw_m {
511        encoder.f64(value);
512    }
513    encoder.f64(coordinates.phase_shift_m);
514    encoder.u8(match coordinates.uvw_law {
515        UvwCoordinateLaw::PhaseTrackingCentre => 0,
516    });
517    encode_epoch(encoder, coordinates.time);
518    encode_epoch(encoder, coordinates.time_centroid);
519    encoder.f64(coordinates.interval_seconds);
520    encoder.f64(coordinates.exposure_seconds);
521    encode_sky_direction(encoder, coordinates.phase_direction);
522    encode_sky_direction(encoder, coordinates.delay_direction);
523    encode_sky_direction(encoder, coordinates.pointing_directions.antenna1);
524    encode_sky_direction(encoder, coordinates.pointing_directions.antenna2);
525
526    let metadata = sample.metadata;
527    encoder.i32(metadata.field_id);
528    encoder.i32(metadata.antenna1);
529    encoder.i32(metadata.antenna2);
530    encoder.i32(metadata.feed1);
531    encoder.i32(metadata.feed2);
532    encoder.i32(metadata.scan_number);
533    encoder.i32(metadata.state_id);
534    encoder.i32(metadata.observation_id);
535    encoder.i32(metadata.array_id);
536}
537
538fn encode_epoch(encoder: &mut CanonicalEncoder, epoch: Epoch) {
539    encoder.f64(epoch.mjd_days());
540    encoder.u8(time_scale_tag(epoch.scale()));
541}

543fn write_hex(formatter: &mut fmt::Formatter<'_>, bytes: &[u8]) -> fmt::Result {
544    for byte in bytes {
545        write!(formatter, "{byte:02x}")?;
546    }
547    Ok(())
