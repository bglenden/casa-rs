14fn angular_value(value: &Variant, name: &str) -> Result<f64, String> {
15    let empty = || "Empty QuantumHolder argument for asQuantumDouble".to_owned();
16    let quantity = match value {
17        Variant::Text(text) => Quantity::read(text).map_err(|_| empty())?,
18        Variant::Texts(_) => return Err(empty()),
19        Variant::Record(record) => quantity_variant_record(record).map_err(|error| {
20            let message = error.to_string();
21            if message.starts_with("Illegal Quantum record") {
22                empty()
23            } else {
24                message
25            }
26        })?,
27        other => quantity(other).map_err(|error| error.to_string())?,
28    };
29    if !quantity.unit().val().conformant(&UnitVal::ANGLE) {
30        return Err(format!("Exception: {name} must be an angular quantity."));
31    }
32    Ok(quantity.get_si_value())
33}

35fn shifted_record(
36    v: &RecordArg,
37    offset: &Variant,
38    position_angle: &Variant,
39    lines: &mut Vec<LogLine>,
40) -> Result<CasaValue, String> {
41    let offset = angular_value(offset, "offset")?;
42    let pa = angular_value(position_angle, "pa")?;
43    let mut direction = Measures::read(v)
44        .map_err(|_| "Exception: v is not a valid measure.".to_string())?
45        .into_reference_origin();
46    if let Some(warning) = direction.warning.take() {
47        lines.push(LogLine::warning(warning));
48    }
49    let value = &mut direction.values[0].value;
50    let MeasureValue::Direction(original) = *value else {
51        return Err("Exception: v is not a direction measure.".into());
52    };
53    let (longitude, latitude) = direction_angles(original);
54
55    let (sin_lat, cos_lat) = latitude.sin_cos();
56    let (sin_offset, cos_offset) = offset.sin_cos();
57    let (sin_pa, cos_pa) = pa.sin_cos();
58    let new_latitude = (cos_offset * sin_lat + sin_offset * cos_lat * cos_pa)
59        .clamp(-1.0, 1.0)
60        .asin();
61    // The two components retain the longitude quadrant for large shifts
62    // and when the great circle crosses a pole.
63    let east = sin_offset * sin_pa;
64    let meridian = cos_offset * cos_lat - sin_offset * sin_lat * cos_pa;
65    let longitude_offset = if latitude.abs() == std::f64::consts::FRAC_PI_2 {
66        // A direction starting exactly at a pole has no unique north axis.
67        // Keep the casacore convention for that position angle.
68        (east / new_latitude.cos()).clamp(-1.0, 1.0).asin()
69    } else {
70        east.atan2(meridian)
71    };
72    let new_longitude = longitude + longitude_offset;
73    *value = MeasureValue::Direction(direction_cosines(new_longitude, new_latitude));
74    Ok(direction.to_value())
75}

77#[pymethods]
78impl PyMeasures {
79    /// Shift a direction by an angular offset at a given position angle.
80    #[pyo3(
81        signature = (v = None, offset = Variant::Empty, pa = Variant::Empty),
82        text_signature = "($self, v={}, offset=[], pa=[])"
83    )]
84    fn shift(
85        &self,
86        py: Python<'_>,
87        #[pyo3(from_py_with=arguments::supplied)] v: Option<Bound<'_, PyAny>>,
88        offset: Variant,
89        pa: Variant,
90    ) -> PyResult<Py<PyAny>> {
91        let v = arguments::record(v.as_ref(), "v")?;
92        self.call(py, |_, lines| {
93            shifted_record(&v, &offset, &pa, lines).map_err(|error| {
94                lines.push(LogLine::error(format!("Exception Reports: {error}")));
95                PyRuntimeError::new_err(error)
96            })
97        })
98    }
99}
