19fn hex(bytes: [u8; 32]) -> String {
20    use std::fmt::Write;
21
22    let mut encoded = String::with_capacity(64);
23    for byte in bytes {
24        write!(&mut encoded, "{byte:02x}").expect("writing to String cannot fail");
25    }
26    encoded
27}

29pub(super) fn execute(config: &CliConfig) -> Result<RunSummary, String> {
30    let started = Instant::now();
31    let result =
32        execute_continuum(application_request(config)?).map_err(|error| error.to_string())?;
33    let minor_cycles = result.minor_cycles.clone();
34    let native = result.outcome.output;
35    let visibility_products = native.visibility_products.map(|completion| {
36        crate::task_contract::ImagerVisibilityProductDiagnostic {
37            problem_id: hex(completion.problem_id().as_bytes()),
38            final_model_generation: hex(completion.final_model().as_bytes()),
39            selected_generation: hex(completion.selected_generation().as_bytes()),
40            weighting_generation: hex(completion.weighting_generation().as_bytes()),
41            model_product: hex(completion.model_product().as_bytes()),
42            residual_product: hex(completion.residual_product().as_bytes()),
43            sample_count: completion.sample_count(),
44        }
45    });
46    let clean_stop_reason = result.minor_stop_reason.map(|reason| match reason {
47        ContinuumStopReason::ThresholdReached => CleanStopReason::GlobalThresholdReached,
48        ContinuumStopReason::IterationBound => CleanStopReason::IterationLimitReached,
49        ContinuumStopReason::StalenessBound => CleanStopReason::MajorCycleLimitReached,
50        ContinuumStopReason::MultiscaleDivergence => CleanStopReason::DivergenceDetected,
51    });
52    Ok(RunSummary {
53        warnings: Vec::new(),
54        gridded_samples: usize::try_from(native.scientific.normal_state().sample_count())
55            .map_err(|_| "native selected-sample count exceeds usize".to_string())?,
56        major_cycles: native.major_cycle_count,
57        minor_iterations: result.minor_iterations,
58        clean_stop_reason,
59        minor_cycles,
60        visibility_products,
61        elapsed: started.elapsed(),
62        output_products: result.product_names,
63    })
64}

66fn application_request(config: &CliConfig) -> Result<ContinuumImagingRequest, String> {
67    let maximum_model_update_jy = if config.dirty_only || config.niter == 0 {
68        1.0
69    } else {
70        config
71            .maximum_model_update_jy
72            .ok_or_else(|| "native deconvolution requires --maximum-model-update-jy".to_string())?
73    };
74    let spectral_mode = match config.spectral_mode {
75        SpectralMode::Mfs => SpectralImagingMode::Continuum,
76        SpectralMode::Cube | SpectralMode::Cubedata => {
77            let mut axis = config.cube_axis.clone();
78            axis.specmode = config.spectral_mode.cube_specmode();
79            if axis.start.is_none()
80                && let Some(start) = config.channel_start
81            {
82                axis.start = Some(CubeAxisValue::Channel(
83                    i32::try_from(start)
84                        .map_err(|_| "cube channel start exceeds i32".to_string())?,
85                ));
86            }
87            SpectralImagingMode::Cube {
88                axis,
89                output_channels: config.channel_count,
90            }
91        }
92    };
93    let casa_inclusive = config.hogbom_iteration_mode == super::HogbomIterationMode::CasaInclusive;
94    let iterations = if casa_inclusive && config.niter > 0 {
95        config.niter.saturating_add(1)
96    } else {
97        config.niter
98    };
99    let cycle_iterations = if casa_inclusive {
100        config.minor_cycle_length.saturating_add(1)
101    } else {
102        config.minor_cycle_length
103    }
104    .min(iterations.max(1));
105    Ok(ContinuumImagingRequest {
106        measurement_set: config.ms.clone(),
107        image_name: config.imagename.clone(),
108        image_size: config.imsize,
109        cell_arcsec: config.cell_arcsec,
110        field_ids: config.field_ids.clone(),
111        uv_range: config.uvrange.clone(),
112        intent: config.intent.clone(),
113        data_description: config.ddid,
114        spectral_window: config
115            .spw_selector
116            .clone()
117            .or_else(|| config.spw.map(|spw| spw.to_string())),
118        channel_start: config.channel_start,
119        channel_count: config.channel_count,
120        spectral_mode,
121        continuum_subtraction: config.continuum_fit_spw.as_ref().map(|fit_spw| {
122            VisibilityContinuumSubtraction {
123                fit_spw: fit_spw.clone(),
124                fit_order: config.continuum_fit_order,
125            }
126        }),
127        data_column: config.datacolumn.clone(),
128        algorithm: if config.dirty_only || config.niter == 0 {
129            ContinuumAlgorithm::Dirty
130        } else {
131            match config.deconvolver {
132                Deconvolver::Hogbom => ContinuumAlgorithm::Hogbom,
133                Deconvolver::Clark => ContinuumAlgorithm::Clark,
134                Deconvolver::Multiscale => ContinuumAlgorithm::Multiscale {
135                    scales_px: config
136                        .multiscale_scales
137                        .iter()
138                        .copied()
139                        .map(f64::from)
140                        .collect(),
141                    small_scale_bias: f64::from(config.small_scale_bias),
142                },
143                Deconvolver::Mtmfs => ContinuumAlgorithm::Mtmfs {
144                    terms: config.nterms,
145                },
146            }
147        },
148        weighting: match config.weighting {
149            WeightingMode::Natural => ContinuumWeighting::Natural,
150            WeightingMode::Uniform => ContinuumWeighting::Uniform,
151            WeightingMode::Briggs { robust } => ContinuumWeighting::Briggs(f64::from(robust)),
152            WeightingMode::BriggsBwTaper { robust } => {
153                ContinuumWeighting::BriggsBandwidthTaper(f64::from(robust))
154            }
155        },
156        iterations,
157        cycle_iterations,
158        maximum_major_cycles: config.nmajor.unwrap_or(1),
159        maximum_model_update_jy: f64::from(maximum_model_update_jy),
160        noise_sigma: (config.nsigma > 0.0).then_some(f64::from(config.nsigma)),
161        cycle_factor: f64::from(config.cyclefactor),
162        minimum_psf_fraction: f64::from(config.min_psf_fraction),
163        maximum_psf_fraction: f64::from(config.max_psf_fraction),
164        gain: f64::from(config.gain),
165        threshold_jy: f64::from(config.threshold_jy),
166        psf_cutoff: config.psf_cutoff,
167        beam_policy: match config.restoring_beam_mode {
168            RestoringBeamMode::PerPlane => ContinuumBeamPolicy::PerPlane,
169            RestoringBeamMode::Common => ContinuumBeamPolicy::Common,
170        },
171        mask: match (&config.mask_image, config.use_mask) {
172            (Some(path), CleanMaskMode::User) => ContinuumMask::Image(path.clone()),
173            (_, CleanMaskMode::AutoMultiThreshold) => {
174                ContinuumMask::AutoMultithresh(ContinuumAutoMaskControls {
175                    sidelobe_factor: f64::from(config.auto_mask.sidelobe_threshold),
176                    noise_factor: f64::from(config.auto_mask.noise_threshold),
177                    low_noise_factor: f64::from(config.auto_mask.low_noise_threshold),
178                    negative_factor: f64::from(config.auto_mask.negative_threshold),
179                    minimum_beam_fraction: f64::from(config.auto_mask.min_beam_frac),
180                    smooth_factor: f64::from(config.auto_mask.smooth_factor),
181                    cut_threshold: f64::from(config.auto_mask.cut_threshold),
182                    grow_iterations: config.auto_mask.grow_iterations,
183                    minimum_percent_change: f64::from(config.auto_mask.min_percent_change),
184                })
185            }
186            (None, CleanMaskMode::User) if config.mask_boxes.is_empty() => ContinuumMask::FullPlane,
187            (None, CleanMaskMode::User) => ContinuumMask::Boxes(
188                config
189                    .mask_boxes
190                    .iter()
191                    .map(|region| ContinuumMaskBox {
192                        blc: [region[0], region[1]],
193                        trc: [region[2], region[3]],
194                    })
195                    .collect(),
196            ),
197        },
198        save_model_column: config.save_model == SaveModelMode::ModelColumn,
199        save_continuum_residual: config.save_continuum_residual,
200        task_requirements: task_requirements(config),
201    })
202}

204fn task_requirements(config: &CliConfig) -> Vec<TaskRequirement> {
205    let mut requirements = vec![match config.spectral_mode {
206        SpectralMode::Mfs => TaskRequirement::SerialCpu,
207        SpectralMode::Cube => TaskRequirement::SpectralCube,
208        SpectralMode::Cubedata => TaskRequirement::SpectralCubedata,
209    }];
210    if config.aw_project.is_some() {
211        requirements.push(TaskRequirement::AwProjection);
212    } else if config.w_term_mode == WTermMode::WProject {
213        requirements.push(TaskRequirement::WProjection);
214    } else if config.use_pointing && !config.force_standard_gridder {
215        requirements.push(TaskRequirement::MosaicGridder);
216    }
217    if config.outlier_file.is_some() {
218        requirements.push(TaskRequirement::FacetsOutliers);
219    }
220    if config.use_mask == CleanMaskMode::AutoMultiThreshold {
221        requirements.push(TaskRequirement::Automasking);
222    }
223    if config.use_mask == CleanMaskMode::AutoMultiThreshold
224        || !config.mask_boxes.is_empty()
225        || config.mask_image.is_some()
226    {
227        requirements.push(TaskRequirement::MaskProduct);
228    }
229    if config.start_model.is_some() {
230        requirements.push(TaskRequirement::StartModel);
231    }
232    if config.save_model != SaveModelMode::None {
233        requirements.push(TaskRequirement::ModelColumnWrite);
234    }
235    requirements.extend(backend_requirements(config));
236    if unsupported_native_controls(config) {
237        requirements.push(TaskRequirement::UnsupportedControls);
238    }
239    requirements
240}

242fn backend_requirements(config: &CliConfig) -> Vec<TaskRequirement> {
243    let mut requirements = Vec::new();
244    match config.standard_mfs_acceleration {
245        StandardMfsAccelerationPolicy::Auto => {
246            requirements.push(TaskRequirement::ExecutionAuto);
247        }
248        StandardMfsAccelerationPolicy::Cpu => {}
249        StandardMfsAccelerationPolicy::MultiCpu => {
250            requirements.push(TaskRequirement::FixedTileCpu);
251        }
252        StandardMfsAccelerationPolicy::Metal => {
253            requirements.push(TaskRequirement::MetalRowRunGroupedGridder);
254        }
255    }
256    if let Some(backend) = config.standard_mfs_backend.as_deref() {
257        requirements.push(match backend {
258            "cpu" | "serial" | "serial-cpu" => TaskRequirement::SerialCpu,
259            "fixed-tile" | "fixed-tile-cpu" => TaskRequirement::FixedTileCpu,
260            "metal" | "metal-gridder" => TaskRequirement::MetalGridder,
261            "metal-row-run" | "metal-row-run-gridder" => TaskRequirement::MetalRowRunGridder,
262            "metal-row-run-grouped" | "metal-row-run-grouped-gridder" => {
263                TaskRequirement::MetalRowRunGroupedGridder
264            }
265            _ => TaskRequirement::UnsupportedControls,
266        });
267    }
268    match config.imaging_fft_backend {
269        ImagingFftBackendPolicy::RustFft => {}
270        ImagingFftBackendPolicy::Auto => requirements.push(TaskRequirement::FftAuto),
271        ImagingFftBackendPolicy::Accelerate => requirements.push(TaskRequirement::Accelerate),
272        ImagingFftBackendPolicy::MetalMpsGraph => requirements.push(TaskRequirement::MetalMpsGraph),
273        ImagingFftBackendPolicy::Fftw => requirements.push(TaskRequirement::Fftw),
274    }
275    requirements
276}

278fn unsupported_native_controls(config: &CliConfig) -> bool {
279    config.phasecenter_field.is_some()
280        || config.phasecenter.is_some()
281        || config
282            .correlation
283            .as_deref()
284            .is_some_and(|plane| !plane.eq_ignore_ascii_case("I"))
285        || config.uv_taper.is_some()
286        || config.fullsummary
287        || config.pbcor
288        || config.write_pb
289        || config.chanchunks.is_some()
290        || config.per_channel_weight_density
291        || config.w_project_planes.is_some()
292        || config.standard_mfs_grid_threads.is_some()
293        || config.standard_mfs_tile_anchor.is_some()
294        || config.standard_mfs_residual_backend.is_some()
295        || config.standard_mfs_initial_dirty_backend.is_some()
296        || config.standard_mfs_metal_minor_cycle_chunk.is_some()
297        || config.standard_mfs_metal_grouped_input_cache.is_some()
298        || config.standard_mfs_memory_target_mb.is_some()
299        || config.standard_mfs_prepare_buffer_mb.is_some()
300        || config.imaging_memory_target_mb.is_some()
301        || config.imaging_memory_pressure_policy != ImagingMemoryPressurePolicy::Auto
302        || config.imaging_prepare_buffer_mb.is_some()
303        || config.imaging_row_block_rows.is_some()
304        || config.imaging_prepare_workers.is_some()
305        || config.imaging_read_ahead_blocks.is_some()
306        || config.imaging_fft_precision != ImagingFftPrecisionPolicy::Auto
307        || config.write_preview_pngs
308}
