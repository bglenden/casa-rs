224/// Requested polarization reconstruction coordinates.
225#[derive(Debug, Clone, PartialEq, Eq)]
226pub struct PolarizationContract {
227    coordinates: Vec<PolarizationCoordinate>,
228}
229
230impl PolarizationContract {
231    /// Construct requested coordinates. Compilation canonicalizes ordering.
232    #[must_use]
233    pub const fn new(coordinates: Vec<PolarizationCoordinate>) -> Self {
234        Self { coordinates }
235    }
236
237    /// Return canonical requested coordinates after compilation.
238    #[must_use]
239    pub fn coordinates(&self) -> &[PolarizationCoordinate] {
240        &self.coordinates
241    }
242
243    fn canonicalize(mut self) -> Result<Self, CompileProblemError> {
244        self.coordinates.sort_unstable();
245        self.coordinates.dedup();
246        if self.coordinates.is_empty() {
247            return Err(CompileProblemError::InvalidReconstructionContract {
248                reason: "at least one polarization coordinate must be requested",
249            });
250        }
251        let categories = self
252            .coordinates
253            .iter()
254            .fold([false; 3], |mut present, coordinate| {
255                match coordinate {
256                    PolarizationCoordinate::StokesI
257                    | PolarizationCoordinate::StokesQ
258                    | PolarizationCoordinate::StokesU
259                    | PolarizationCoordinate::StokesV => present[0] = true,
260                    PolarizationCoordinate::LinearXx
261                    | PolarizationCoordinate::LinearXy
262                    | PolarizationCoordinate::LinearYx
263                    | PolarizationCoordinate::LinearYy => present[1] = true,
264                    PolarizationCoordinate::CircularRr
265                    | PolarizationCoordinate::CircularRl
266                    | PolarizationCoordinate::CircularLr
267                    | PolarizationCoordinate::CircularLl => present[2] = true,
268                }
269                present
270            });
271        if categories.into_iter().filter(|present| *present).count() > 1 {
272            return Err(CompileProblemError::InvalidReconstructionContract {
273                reason: "one reconstruction cannot mix Stokes, linear, and circular coordinates",
274            });
275        }
276        Ok(self)
277    }
278}

481        self.polarization = self.polarization.canonicalize()?;
482        if let ReconstructionAlgorithm::Multiscale { scales_px } = &mut self.algorithm {
483            for scale in scales_px.iter_mut() {

489            scales_px.dedup();
490        }
491        Ok(self)
492    }

997    permitted_precisions: Vec<NumericPrecision>,

1003impl NumericsContract {
1004    /// Construct numerical requirements. Compilation canonicalizes ordering.
1005    #[must_use]
1006    pub fn new(
1007        permitted_precisions: Vec<NumericPrecision>,
1008        reduction: ReductionPolicy,
1009        finite_values: FiniteValuePolicy,
1010        stage_error_budgets: Vec<(NumericalStage, StageErrorBudget)>,
1011    ) -> Self {
1012        Self {
1013            permitted_precisions,
1014            reduction,
1015            finite_values,
1016            stage_error_budgets,
1017        }
1018    }
1019
1020    /// Return permitted arithmetic precisions in canonical order after compilation.
1021    #[must_use]
1022    pub fn permitted_precisions(&self) -> &[NumericPrecision] {
1023        &self.permitted_precisions
1024    }
1025
1026    /// Return permitted reduction semantics.
1027    #[must_use]
1028    pub const fn reduction(&self) -> ReductionPolicy {
1029        self.reduction
1030    }
1031
1032    /// Return finite-value behavior.
1033    #[must_use]
1034    pub const fn finite_values(&self) -> FiniteValuePolicy {
1035        self.finite_values
1036    }
1037
1038    /// Return complete stage budgets in canonical stage order after compilation.
1039    #[must_use]
1040    pub fn stage_error_budgets(&self) -> &[(NumericalStage, StageErrorBudget)] {
1041        &self.stage_error_budgets
1042    }
1043
1044    fn canonicalize(mut self) -> Result<Self, CompileProblemError> {
1045        self.permitted_precisions.sort_unstable();
1046        self.permitted_precisions.dedup();
1047        if self.permitted_precisions.is_empty() {
1048            return Err(CompileProblemError::InvalidNumerics {
1049                reason: "at least one arithmetic precision must be permitted",
1050            });
1051        }
1052        self.stage_error_budgets
1053            .sort_unstable_by_key(|(stage, _)| *stage);
1054        if self
1055            .stage_error_budgets
1056            .windows(2)
1057            .any(|pair| pair[0].0 == pair[1].0)
1058        {
1059            return Err(CompileProblemError::InvalidNumerics {
1060                reason: "a numerical stage has more than one error budget",
1061            });
1062        }
1063        if self.stage_error_budgets.len() != NumericalStage::ALL.len()
1064            || NumericalStage::ALL
1065                .iter()
1066                .zip(&self.stage_error_budgets)
1067                .any(|(required, (actual, _))| required != actual)
1068        {
1069            return Err(CompileProblemError::InvalidNumerics {
1070                reason: "every numerical stage must have exactly one error budget",
1071            });
1072        }
1073        if self.stage_error_budgets.iter().any(|(_, budget)| {
1074            !(budget.absolute.is_finite()
1075                && budget.absolute >= 0.0
1076                && budget.relative.is_finite()
1077                && budget.relative >= 0.0)
1078        }) {
1079            return Err(CompileProblemError::InvalidNumerics {
1080                reason: "stage error budgets must be finite and non-negative",
1081            });
1082        }
1083        Ok(self)
1084    }

1420/// Compile and validate one immutable backend-independent imaging request.
1421pub fn compile(request: ImagingRequest) -> Result<CompiledProblem, CompileProblemError> {
1422    let ImagingRequest {
1423        version: ImagingRequestVersion::V3,
1424        specification,
1425        geometry,
1426        inputs,
1427    } = request;
1428    let geometry = compile_geometry(geometry, &inputs)?;
1429    let science = specification.science;
1430    let products = specification.products.canonicalize();
1431    let observation_transaction = compile_observation_transaction(
1432        inputs.observation_snapshot(),
1433        specification.observation_transaction,
1434    );
1435    let numerics = specification.numerics.canonicalize()?;
1436    validate_science(&science, &inputs)?;
1437    validate_reconstruction(&specification.reconstruction, &geometry)?;
1438    let reconstruction = specification.reconstruction.canonicalize()?;
1439    validate_weighting(specification.weighting)?;
1440    validate_products(&science, &reconstruction, &products)?;
1441    let normal_equation = compile_normal_equation(
1442        &geometry,
1443        &inputs,
1444        &science,
1445        &reconstruction,
1446        specification.weighting,
1447    );
1448    let required_capabilities = derive_capabilities(
1449        &geometry,
1450        &science,
1451        &reconstruction,
1452        normal_equation.weighting(),
1453        &products,
1454    );
1455    let problem_id = canonical_problem_id(ProblemIdentityInput {
1456        inputs: &inputs,
1457        geometry: &geometry,
1458        science: &science,
1459        reconstruction: &reconstruction,
1460        normal_equation: &normal_equation,
1461        products: &products,
1462        observation_transaction: &observation_transaction,
1463        numerics: &numerics,
1464    });
1465    let numerics_id = canonical_numerics_id(&numerics);
1466    let product_graph = compile_product_graph(&geometry, &reconstruction, &products);
1467    Ok(CompiledProblem {
1468        problem_id,
1469        numerics_id,
1470        inputs,
1471        geometry,
1472        science,
1473        reconstruction,
1474        normal_equation,
1475        products,
1476        product_graph,
1477        observation_transaction,
1478        numerics,
1479        required_capabilities,
1480    })
1481}

1483fn validate_science(
1484    science: &ScientificContract,
1485    inputs: &ProblemInputIdentities,
1486) -> Result<(), CompileProblemError> {
1487    if let SpectralSampling::ChannelAverage {
1488        channels_per_bin: 0,
1489    } = science.spectral.sampling
1490    {
1491        return Err(CompileProblemError::InvalidScientificContract {
1492            reason: "spectral channel averaging requires a positive bin width",
1493        });
1494    }
1495    if science.measurement_equation.instrument_response != InstrumentResponse::Scalar
1496        && !inputs
1497            .reference_data()
1498            .iter()
1499            .any(|(kind, _)| *kind == ReferenceDataKind::Instrument)
1500    {
1501        return Err(CompileProblemError::InvalidScientificContract {
1502            reason: "direction-dependent response requires bound instrument reference data",
1503        });
1504    }
1505    Ok(())
1506}

1508fn validate_reconstruction(
1509    contract: &ReconstructionContract,
1510    geometry: &CompiledGeometry,
1511) -> Result<(), CompileProblemError> {
1512    match contract.basis {
1513        ReconstructionBasis::Taylor { terms: 0 | 1 } => {
1514            return Err(CompileProblemError::InvalidCapabilityCombination {
1515                reason: "a Taylor basis requires at least two terms; single-term MFS uses the constant basis",
1516            });
1517        }
1518        ReconstructionBasis::ChannelLocal { channels: 0 } => {
1519            return Err(CompileProblemError::InvalidCapabilityCombination {
1520                reason: "a channel-local basis requires at least one channel",
1521            });
1522        }
1523        ReconstructionBasis::Constant
1524        | ReconstructionBasis::Taylor { .. }
1525        | ReconstructionBasis::ChannelLocal { .. } => {}
1526    }
1527    if let ReconstructionBasis::ChannelLocal { channels } = contract.basis
1528        && channels != geometry.spectral().output_channels()
1529    {
1530        return Err(CompileProblemError::SpectralChannelCountMismatch {
1531            geometry_channels: geometry.spectral().output_channels(),
1532            reconstruction_channels: channels,
1533        });
1534    }
1535    if matches!(contract.algorithm, ReconstructionAlgorithm::Mtmfs)
1536        != matches!(contract.basis, ReconstructionBasis::Taylor { .. })
1537    {
1538        return Err(CompileProblemError::InvalidCapabilityCombination {
1539            reason: "MT-MFS and Taylor-basis reconstruction must be requested together",
1540        });
1541    }
1542    if matches!(contract.algorithm, ReconstructionAlgorithm::Dirty)
1543        && contract.controls.max_minor_iterations != 0
1544    {
1545        return Err(CompileProblemError::InvalidCapabilityCombination {
1546            reason: "dirty reconstruction cannot request minor-cycle iterations",
1547        });
1548    }
1549    if matches!(contract.algorithm, ReconstructionAlgorithm::Dirty)
1550        && (contract.controls.gain != 1.0 || contract.controls.threshold_jy_per_beam != 0.0)
1551    {
1552        return Err(CompileProblemError::InvalidCapabilityCombination {
1553            reason: "dirty reconstruction requires canonical inactive controls: gain 1 and threshold 0",
1554        });
1555    }
1556    if !matches!(contract.algorithm, ReconstructionAlgorithm::Dirty)
1557        && contract.controls.max_minor_iterations == 0
1558    {
1559        return Err(CompileProblemError::InvalidCapabilityCombination {
1560            reason: "a minor-cycle algorithm requires a positive iteration budget",
1561        });
1562    }
1563    if !(contract.controls.gain.is_finite()
1564        && contract.controls.gain > 0.0
1565        && contract.controls.gain <= 1.0
1566        && contract.controls.threshold_jy_per_beam.is_finite()
1567        && contract.controls.threshold_jy_per_beam >= 0.0)
1568    {
1569        return Err(CompileProblemError::InvalidCapabilityCombination {
1570            reason: "reconstruction gain and threshold must be finite and in their valid domains",
1571        });
1572    }
1573    if let ReconstructionAlgorithm::Multiscale { scales_px } = &contract.algorithm {
1574        if scales_px.is_empty()
1575            || scales_px
1576                .iter()
1577                .any(|scale| !(scale.is_finite() && *scale >= 0.0))
1578        {
1579            return Err(CompileProblemError::InvalidCapabilityCombination {
1580                reason: "multiscale reconstruction requires finite non-negative explicit scales",
1581            });
1582        }
1583    }
1584    Ok(())
1585}

1587fn validate_weighting(contract: WeightingContract) -> Result<(), CompileProblemError> {
1588    match contract.scheme {
1589        WeightingScheme::Natural if contract.density_scope != WeightDensityScope::NotApplicable => {
1590            return Err(CompileProblemError::InvalidWeighting {
1591                reason: "natural weighting has no density generation",
1592            });
1593        }
1594        WeightingScheme::Uniform
1595        | WeightingScheme::Briggs { .. }
1596        | WeightingScheme::BriggsBandwidthTaper { .. }
1597            if contract.density_scope == WeightDensityScope::NotApplicable =>
1598        {
1599            return Err(CompileProblemError::InvalidWeighting {
1600                reason: "density weighting requires an explicit global density scope",
1601            });
1602        }
1603        WeightingScheme::Natural
1604        | WeightingScheme::Uniform
1605        | WeightingScheme::Briggs { .. }
1606        | WeightingScheme::BriggsBandwidthTaper { .. } => {}
1607    }
1608    let robust = match contract.scheme {
1609        WeightingScheme::Briggs { robust } | WeightingScheme::BriggsBandwidthTaper { robust } => {
1610            Some(robust)
1611        }
1612        WeightingScheme::Natural | WeightingScheme::Uniform => None,
1613    };
1614    if robust.is_some_and(|value| !(value.is_finite() && (-2.0..=2.0).contains(&value))) {
1615        return Err(CompileProblemError::InvalidWeighting {
1616            reason: "Briggs robustness must be finite and in [-2, 2]",
1617        });
1618    }
1619    if contract.uv_taper.is_some_and(|taper| {
1620        !(taper.major_lambda.is_finite()
1621            && taper.major_lambda > 0.0
1622            && taper.minor_lambda.is_finite()
1623            && taper.minor_lambda > 0.0
1624            && taper.position_angle_rad.is_finite())
1625    }) {
1626        return Err(CompileProblemError::InvalidWeighting {
1627            reason: "UV taper axes must be finite and positive and its angle must be finite",
1628        });
1629    }
1630    Ok(())
1631}

1633fn validate_products(
1634    science: &ScientificContract,
1635    reconstruction: &ReconstructionContract,
1636    products: &ProductRequirements,
1637) -> Result<(), CompileProblemError> {
1638    if products.products.is_empty() {
1639        return Err(CompileProblemError::InvalidProductCombination {
1640            reason: "at least one product must be requested",
1641        });
1642    }
1643    if matches!(
1644        products.normalization,
1645        ProductNormalization::FlatNoise | ProductNormalization::FlatSky
1646    ) && !products.contains(ProductKind::Sensitivity)
1647    {
1648        return Err(CompileProblemError::InvalidNormalizationCombination {
1649            reason: "flat-noise and flat-sky normalization require sensitivity state",
1650        });
1651    }
1652    let restored_image_requested = products.contains(ProductKind::RestoredImage);
1653    let restoring_beam_requested = !matches!(products.restoring_beam, RestoringBeamPolicy::None);
1654    if restored_image_requested != restoring_beam_requested {
1655        return Err(CompileProblemError::InvalidProductCombination {
1656            reason: "restored-image and restoring-beam requirements must be requested together",
1657        });
1658    }
1659    if restored_image_requested
1660        && !(products.contains(ProductKind::Residual) && products.contains(ProductKind::Model))
1661    {
1662        return Err(CompileProblemError::InvalidProductCombination {
1663            reason: "a restored image requires residual and model products",
1664        });
1665    }
1666    let common_spectral_beam = science.spectral.coupling == SpectralCoupling::CommonRestoringBeam;
1667    let common_product_beam = products.restoring_beam == RestoringBeamPolicy::Common;
1668    if common_spectral_beam != common_product_beam {
1669        return Err(CompileProblemError::InvalidProductCombination {
1670            reason: "common spectral coupling and common restoring-beam publication must be requested together",
1671        });
1672    }
1673    if products.contains(ProductKind::PbCorrectedImage)
1674        && !(products.contains(ProductKind::RestoredImage)
1675            && products.contains(ProductKind::PrimaryBeam))
1676    {
1677        return Err(CompileProblemError::InvalidProductCombination {
1678            reason: "a PB-corrected image requires restored-image and primary-beam products",
1679        });
1680    }
1681    if products.contains(ProductKind::PrimaryBeam)
1682        && science.measurement_equation.instrument_response == InstrumentResponse::Scalar
1683    {
1684        return Err(CompileProblemError::InvalidProductCombination {
1685            reason: "a primary-beam product requires a primary-beam or full-Mueller response",
1686        });
1687    }
1688    let taylor_terms = match reconstruction.basis {
1689        ReconstructionBasis::Taylor { terms } => terms,
1690        ReconstructionBasis::Constant | ReconstructionBasis::ChannelLocal { .. } => 0,
1691    };
1692    if products.contains(ProductKind::TaylorTerms) && taylor_terms == 0 {
1693        return Err(CompileProblemError::InvalidProductCombination {
1694            reason: "Taylor products require a Taylor reconstruction basis",
1695        });
1696    }
1697    if products.contains(ProductKind::TaylorTerms)
1698        && ![
1699            ProductKind::Psf,
1700            ProductKind::Residual,
1701            ProductKind::Model,
1702            ProductKind::RestoredImage,
1703            ProductKind::SumWeights,
1704            ProductKind::Weight,
1705            ProductKind::PrimaryBeam,
1706            ProductKind::PbCorrectedImage,
1707        ]
1708        .into_iter()
1709        .any(|product| products.contains(product))
1710    {
1711        return Err(CompileProblemError::InvalidProductCombination {
1712            reason: "a Taylor coefficient set requires at least one Taylor image product",
1713        });
1714    }
1715    if products.contains(ProductKind::SpectralIndex)
1716        && !(taylor_terms >= 2
1717            && products.contains(ProductKind::TaylorTerms)
1718            && products.contains(ProductKind::Residual)
1719            && products.contains(ProductKind::RestoredImage))
1720    {
1721        return Err(CompileProblemError::InvalidProductCombination {
1722            reason: "spectral index requires at least two Taylor terms plus Taylor, residual, and restored-image products",
1723        });
1724    }
1725    if products.contains(ProductKind::SpectralIndexError)
1726        && !products.contains(ProductKind::SpectralIndex)
1727    {
1728        return Err(CompileProblemError::InvalidProductCombination {
1729            reason: "spectral-index uncertainty requires a spectral-index product",
1730        });
1731    }
1732    if products.contains(ProductKind::PbCorrectedSpectralIndex)
1733        && !(products.contains(ProductKind::SpectralIndex)
1734            && products.contains(ProductKind::PrimaryBeam))
1735    {
1736        return Err(CompileProblemError::InvalidProductCombination {
1737            reason: "PB-corrected spectral index requires spectral-index and primary-beam products",
1738        });
1739    }
1740    if products.contains(ProductKind::Beam)
1741        && ![
1742            ProductKind::Psf,
1743            ProductKind::Residual,
1744            ProductKind::RestoredImage,
1745        ]
1746        .into_iter()
1747        .any(|product| products.contains(product))
1748    {
1749        return Err(CompileProblemError::InvalidProductCombination {
1750            reason: "beam metadata requires a PSF, residual, or restored-image product",
1751        });
1752    }
1753    Ok(())
1754}

1756fn derive_capabilities(
1757    geometry: &CompiledGeometry,
1758    science: &ScientificContract,
1759    reconstruction: &ReconstructionContract,
1760    weighting: &WeightingOperatorContract,
1761    products: &ProductRequirements,
1762) -> BTreeSet<RequiredCapability> {
1763    let mut capabilities = BTreeSet::new();
1764    if geometry
1765        .domains()
1766        .iter()
1767        .any(|domain| domain.facets().len() > 1)
1768    {
1769        capabilities.insert(RequiredCapability::FacetedGeometry);
1770    }
1771    if geometry.spectral().source_frame() != geometry.spectral().output_frame() {
1772        capabilities.insert(RequiredCapability::SpectralFrameTransform);
1773    }
1774    if science.spectral.sampling != SpectralSampling::Identity {
1775        capabilities.insert(RequiredCapability::SpectralResampling);
1776    }
1777    if science.spectral.coupling == SpectralCoupling::CommonRestoringBeam {
1778        capabilities.insert(RequiredCapability::CommonBeamSpectralCoupling);
1779    }
1780    capabilities.extend(
1781        reconstruction
1782            .polarization
1783            .coordinates
1784            .iter()
1785            .copied()
1786            .map(RequiredCapability::Polarization),
1787    );
1788    match science.measurement_equation.instrument_response {
1789        InstrumentResponse::Scalar => {}
1790        InstrumentResponse::PrimaryBeam => {
1791            capabilities.insert(RequiredCapability::PrimaryBeamResponse);
1792        }
1793        InstrumentResponse::FullMueller => {
1794            capabilities.insert(RequiredCapability::FullMuellerResponse);
1795        }
1796    }
1797    if weighting.uv_taper().is_some() {
1798        capabilities.insert(RequiredCapability::UvTaper);
1799    }
1800    capabilities.insert(match reconstruction.basis {
1801        ReconstructionBasis::Constant => RequiredCapability::ConstantBasis,
1802        ReconstructionBasis::Taylor { .. } => RequiredCapability::TaylorBasis,
1803        ReconstructionBasis::ChannelLocal { .. } => RequiredCapability::ChannelLocalBasis,
1804    });
1805    capabilities.insert(match reconstruction.algorithm {
1806        ReconstructionAlgorithm::Dirty => RequiredCapability::DirtyReconstruction,
1807        ReconstructionAlgorithm::Hogbom => RequiredCapability::HogbomReconstruction,
1808        ReconstructionAlgorithm::Clark => RequiredCapability::ClarkReconstruction,
1809        ReconstructionAlgorithm::Multiscale { .. } => RequiredCapability::MultiscaleReconstruction,
1810        ReconstructionAlgorithm::Mtmfs => RequiredCapability::MtmfsReconstruction,
1811    });
1812    capabilities.insert(match weighting.scheme() {
1813        WeightingScheme::Natural => RequiredCapability::NaturalWeighting,
1814        WeightingScheme::Uniform => RequiredCapability::UniformWeighting,
1815        WeightingScheme::Briggs { .. } => RequiredCapability::BriggsWeighting,
1816        WeightingScheme::BriggsBandwidthTaper { .. } => {
1817            RequiredCapability::BriggsBandwidthTaperWeighting
1818        }
1819    });
1820    capabilities.insert(match products.normalization {
1821        ProductNormalization::UnitResponse => RequiredCapability::UnitResponseNormalization,
1822        ProductNormalization::FlatNoise => RequiredCapability::FlatNoiseNormalization,
1823        ProductNormalization::FlatSky => RequiredCapability::FlatSkyNormalization,
1824    });
1825    capabilities.extend(
1826        products
1827            .products
1828            .iter()
1829            .copied()
1830            .map(RequiredCapability::Product),
1831    );
1832    capabilities
1833}

1836    inputs: &'a ProblemInputIdentities,
1837    geometry: &'a CompiledGeometry,
1838    science: &'a ScientificContract,
1839    reconstruction: &'a ReconstructionContract,

1846fn canonical_problem_id(input: ProblemIdentityInput<'_>) -> CompiledProblemId {
1847    let ProblemIdentityInput {
1848        inputs,
1849        geometry,
1850        science,
1851        reconstruction,
1852        normal_equation,
1853        products,
1854        observation_transaction,
1855        numerics,
1856    } = input;
1857    let mut encoder = CanonicalEncoder::new();
1858    encoder.bytes(COMPILED_PROBLEM_IDENTITY_DOMAIN);
1859    encoder.u32(COMPILED_PROBLEM_IDENTITY_VERSION);
1860    encoder.identity(inputs.observation().identity());
1861    encoder.digest(observation_transaction.transaction_id().as_bytes());
1862    encoder.digest(geometry.geometry_id().as_bytes());
1863    encoder.usize(inputs.reference_data().len());
1864    for (kind, identity) in inputs.reference_data() {
1865        encoder.u8(reference_data_tag(*kind));
1866        encoder.identity(*identity);
1867    }
1868    match inputs.model() {
1869        ModelStateIdentity::Empty => encoder.u8(0),
1870        ModelStateIdentity::Seed(identity) => {
1871            encoder.u8(1);
1872            encoder.identity(identity);
1873        }
1874        ModelStateIdentity::Generation(identity) => {
1875            encoder.u8(2);
1876            encoder.identity(identity);
1877        }
1878    }
1879    match science.spectral.sampling {
1880        SpectralSampling::Identity => encoder.u8(0),
1881        SpectralSampling::Nearest => encoder.u8(1),
1882        SpectralSampling::Linear => encoder.u8(2),
1883        SpectralSampling::ChannelAverage { channels_per_bin } => {
1884            encoder.u8(3);
1885            encoder.usize(channels_per_bin);
1886        }
1887    }
1888    encoder.u8(match science.spectral.coupling {
1889        SpectralCoupling::Independent => 0,
1890        SpectralCoupling::CommonRestoringBeam => 1,
1891    });
1892    encoder.u8(match science.measurement_equation.instrument_response {
1893        InstrumentResponse::Scalar => 0,
1894        InstrumentResponse::PrimaryBeam => 1,
1895        InstrumentResponse::FullMueller => 2,
1896    });
1897    let inner_products = science.measurement_equation.inner_products;
1898    encoder.u8(match inner_products.model() {
1899        ModelInnerProduct::HermitianEuclidean => 0,
1900    });
1901    encoder.u8(match inner_products.visibility() {
1902        VisibilityInnerProduct::HermitianEuclidean => 0,
1903    });
1904    let operator = normal_equation.measurement_operator();
1905    encoder.digest(operator.domain().geometry().as_bytes());
1906    encoder.u8(match operator.domain().basis() {
1907        ReconstructionBasis::Constant => 0,
1908        ReconstructionBasis::Taylor { .. } => 1,
1909        ReconstructionBasis::ChannelLocal { .. } => 2,
1910    });
1911    encoder.usize(operator.domain().polarization().coordinates().len());
1912    for coordinate in operator.domain().polarization().coordinates() {
1913        encoder.u8(polarization_tag(*coordinate));
1914    }
1915    encoder.identity(operator.codomain().observation().identity());
1916    encoder.usize(operator.transforms().len());
1917    for transform in operator.transforms() {
1918        match transform {
1919            PairedMeasurementTransform::SpectralBasis { basis } => {
1920                encoder.u8(0);
1921                match basis {
1922                    ReconstructionBasis::Constant => encoder.u8(0),
1923                    ReconstructionBasis::Taylor { terms } => {
1924                        encoder.u8(1);
1925                        encoder.usize(*terms);
1926                    }
1927                    ReconstructionBasis::ChannelLocal { channels } => {
1928                        encoder.u8(2);
1929                        encoder.usize(*channels);
1930                    }
1931                }
1932            }
1933            PairedMeasurementTransform::PolarizationMapping => encoder.u8(1),
1934            PairedMeasurementTransform::DirectionDependentResponse { response } => {
1935                encoder.u8(2);
1936                encoder.u8(match response {
1937                    InstrumentResponse::Scalar => 0,
1938                    InstrumentResponse::PrimaryBeam => 1,
1939                    InstrumentResponse::FullMueller => 2,
1940                });
1941            }
1942            PairedMeasurementTransform::PhaseRotation { convention } => {
1943                encoder.u8(3);
1944                encoder.u8(match convention {
1945                    crate::geometry::VisibilityPhaseConvention::NegativeTwoPiFrequencyDelay => 0,
1946                });
1947            }
1948            PairedMeasurementTransform::SpectralResampling { sampling } => {
1949                encoder.u8(4);
1950                encoder.u8(match sampling {
1951                    SpectralSampling::Nearest => 0,
1952                    SpectralSampling::Linear => 1,
1953                    SpectralSampling::Identity | SpectralSampling::ChannelAverage { .. } => {
1954                        unreachable!("compiled spectral resampling is nearest or linear")
1955                    }
1956                });
1957            }
1958            PairedMeasurementTransform::ChannelIntegration { channels_per_bin } => {
1959                encoder.u8(5);
1960                encoder.usize(*channels_per_bin);
1961            }
1962        }
1963    }
1964    let compiled_weighting = normal_equation.weighting();
1965    encoder.digest(compiled_weighting.generation_id().as_bytes());
1966    encoder.identity(compiled_weighting.snapshot().identity());
1967    encoder.usize(compiled_weighting.sources().len());
1968    for source in compiled_weighting.sources() {
1969        encoder.identity(source.source().identity());
1970        encoder.u8(match source.flags() {
1971            FlagPolicy::FlagOrFlagRow => 0,
1972        });
1973        encoder.u8(match source.input_weights() {
1974            WeightColumn::Weight => 0,
1975            WeightColumn::WeightSpectrum => 1,
1976        });
1977        encoder.identity(source.flag_generation());
1978        encoder.identity(source.flag_row_generation());
1979        encoder.identity(source.input_weight_generation());
1980    }
1981    encoder.u8(match normal_equation.output().normalization() {
1982        NormalStateNormalization::Unnormalized => 0,
1983    });
1984    match reconstruction.basis {
1985        ReconstructionBasis::Constant => encoder.u8(0),
1986        ReconstructionBasis::Taylor { terms } => {
1987            encoder.u8(1);
1988            encoder.usize(terms);
1989        }
1990        ReconstructionBasis::ChannelLocal { channels } => {
1991            encoder.u8(2);
1992            encoder.usize(channels);
1993        }
1994    }
1995    match &reconstruction.algorithm {
1996        ReconstructionAlgorithm::Dirty => encoder.u8(0),
1997        ReconstructionAlgorithm::Hogbom => encoder.u8(1),
1998        ReconstructionAlgorithm::Clark => encoder.u8(2),
1999        ReconstructionAlgorithm::Multiscale { scales_px } => {
2000            encoder.u8(3);
2001            encoder.usize(scales_px.len());
2002            for scale in scales_px {
2003                encoder.f64(*scale);
2004            }
2005        }
2006        ReconstructionAlgorithm::Mtmfs => encoder.u8(4),
2007    }
2008    encoder.usize(reconstruction.controls.max_minor_iterations);
2009    encoder.f64(reconstruction.controls.gain);
2010    encoder.f64(reconstruction.controls.threshold_jy_per_beam);
2011    encoder.usize(reconstruction.polarization.coordinates.len());
2012    for coordinate in &reconstruction.polarization.coordinates {
2013        encoder.u8(polarization_tag(*coordinate));
2014    }
2015    match compiled_weighting.scheme() {
2016        WeightingScheme::Natural => encoder.u8(0),
2017        WeightingScheme::Uniform => encoder.u8(1),
2018        WeightingScheme::Briggs { robust } => {
2019            encoder.u8(2);
2020            encoder.f64(robust);
2021        }
2022        WeightingScheme::BriggsBandwidthTaper { robust } => {
2023            encoder.u8(3);
2024            encoder.f64(robust);
2025        }
2026    }
2027    encoder.u8(match compiled_weighting.density_scope() {
2028        WeightDensityScope::NotApplicable => 0,
2029        WeightDensityScope::GlobalSelection => 1,
2030        WeightDensityScope::PerOutputChannel => 2,
2031    });
2032    match compiled_weighting.uv_taper() {
2033        None => encoder.u8(0),
2034        Some(taper) => {
2035            encoder.u8(1);
2036            encoder.f64(taper.major_lambda());
2037            encoder.f64(taper.minor_lambda());
2038            encoder.f64(taper.position_angle_rad());
2039        }
2040    }
2041    encoder.usize(products.products.len());
2042    for product in &products.products {
2043        encoder.u8(product_tag(*product));
2044    }
2045    encoder.u8(match products.normalization {
2046        ProductNormalization::UnitResponse => 0,
2047        ProductNormalization::FlatNoise => 1,
2048        ProductNormalization::FlatSky => 2,
2049    });
2050    encoder.u8(match products.restoring_beam {
2051        RestoringBeamPolicy::None => 0,
2052        RestoringBeamPolicy::PerPlane => 1,
2053        RestoringBeamPolicy::Common => 2,
2054    });
2055    let primary_beam_validity = products.validity.primary_beam();
2056    encoder.u32(primary_beam_validity.cutoff().to_bits());
2057    encoder.u8(match primary_beam_validity.comparison() {
2058        ProductSupportComparison::StrictlyGreater => 0,
2059    });
2060    encoder.u8(match primary_beam_validity.blanking() {
2061        ProductBlankingPolicy::ZeroAndFalseMask => 0,
2062    });
2063    let taylor_validity = products.validity.taylor();
2064    encoder.u8(match taylor_validity.reference() {
2065        TaylorSupportReference::PrincipalResidualTaylor0PositiveMaximum => 0,
2066    });
2067    encoder.u32(taylor_validity.peak_fraction().to_bits());
2068    encoder.u8(match taylor_validity.comparison() {
2069        ProductSupportComparison::StrictlyGreater => 0,
2070    });
2071    encoder.u8(match taylor_validity.blanking() {
2072        ProductBlankingPolicy::ZeroAndFalseMask => 0,
2073    });
2074    encoder.usize(products.normalization_boundary.operations().len());
2075    for operation in products.normalization_boundary.operations() {
2076        match operation {
2077            ProductBoundaryOperation::Normalize(normalization) => {
2078                encoder.u8(0);
2079                encoder.u8(match normalization {
2080                    ProductNormalization::UnitResponse => 0,
2081                    ProductNormalization::FlatNoise => 1,
2082                    ProductNormalization::FlatSky => 2,
2083                });
2084            }
2085            ProductBoundaryOperation::ScaleResidual => encoder.u8(1),
2086            ProductBoundaryOperation::Restore(policy) => {
2087                encoder.u8(2);
2088                encoder.u8(match policy {
2089                    RestoringBeamPolicy::None => 0,
2090                    RestoringBeamPolicy::PerPlane => 1,
2091                    RestoringBeamPolicy::Common => 2,
2092                });
2093            }
2094            ProductBoundaryOperation::CorrectPrimaryBeam => encoder.u8(3),
2095            ProductBoundaryOperation::BlankInvalid => encoder.u8(4),
2096            ProductBoundaryOperation::ConvertUnits => encoder.u8(5),
2097        }
2098    }
2099    encode_numerics(&mut encoder, numerics);
2100    CompiledProblemId(LogicalIdentity::from_sha256(encoder.finish()))
2101}

2111fn encode_numerics(encoder: &mut CanonicalEncoder, numerics: &NumericsContract) {
2112    encoder.usize(numerics.permitted_precisions.len());
2113    for precision in &numerics.permitted_precisions {
2114        encoder.u8(match precision {
2115            NumericPrecision::F32 => 0,
2116            NumericPrecision::F64 => 1,
2117        });
2118    }
2119    encoder.u8(match numerics.reduction {
2120        ReductionPolicy::DeterministicPairwise => 0,
2121        ReductionPolicy::Compensated => 1,
