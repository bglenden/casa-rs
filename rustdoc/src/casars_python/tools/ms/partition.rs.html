25impl PyMsTool {
26    /// Materialize selected rows while retaining the input subtable IDs.
27    #[allow(clippy::too_many_arguments)]
28    #[pyo3(signature = (outputms = String::new(), field = None, spw = None, baseline = None, timebin = None, time = String::new(), scan = None, uvrange = None, taql = String::new(), whichcol = "DATA".to_owned(), tileshape = None, subarray = None, combine = String::new(), intent = String::new(), obs = String::new()), text_signature = "($self, outputms='', field=[], spw=[], baseline=[], timebin=[], time='', scan=[], uvrange=[], taql='', whichcol='DATA', tileshape=[], subarray=[], combine='', intent='', obs='')")]
29    fn partition(
30        &self,
31        py: Python<'_>,
32        #[pyo3(from_py_with = output_argument)] outputms: String,
33        field: Option<&Bound<'_, PyAny>>,
34        spw: Option<&Bound<'_, PyAny>>,
35        baseline: Option<&Bound<'_, PyAny>>,
36        timebin: Option<Variant>,
37        #[pyo3(from_py_with = time_argument)] time: String,
38        scan: Option<&Bound<'_, PyAny>>,
39        uvrange: Option<&Bound<'_, PyAny>>,
40        #[pyo3(from_py_with = taql_argument)] taql: String,
41        #[pyo3(from_py_with = whichcol_argument)] whichcol: String,
42        tileshape: Option<Variant>,
43        subarray: Option<&Bound<'_, PyAny>>,
44        #[pyo3(from_py_with = combine_argument)] combine: String,
45        #[pyo3(from_py_with = intent_argument)] intent: String,
46        #[pyo3(from_py_with = obs_argument)] obs: String,
47    ) -> PyResult<bool> {
48        let spw = all_if_minus_one(spw)?.unwrap_or_else(|| "*".into());
49        let selection = MsSelection {
50            field: all_if_minus_one(field)?,
51            antenna: optional_expression(baseline)?,
52            timerange: (!time.is_empty()).then_some(time),
53            scan: optional_expression(scan)?,
54            uvrange: optional_expression(uvrange)?,
55            msselect: (!taql.is_empty()).then_some(taql),
56            array: optional_expression(subarray)?,
57            intent: (!intent.is_empty()).then_some(intent),
58            observation: (!obs.is_empty()).then_some(obs),
59            ..MsSelection::default()
60        };
61        let columns = visibility_columns(&whichcol)?;
62        let options = MsTransformTaskOptions {
63            visibility_columns: columns,
64            preserve_ids: true,
65            legacy_tool: Some(LegacyMsTool::Partition(LegacyTimeOptions {
66                combine_observations: combine.to_ascii_lowercase().contains("obs"),
67            })),
68            time_averaging: time_average(timebin.as_ref(), &combine)?,
69            tiling: tile_layout(tileshape.as_ref())?,
70            ..MsTransformTaskOptions::default()
71        };
72        self.write_subms(py, outputms, spw, selection, options, None)
73    }
74}

76impl PyMsTool {
77    pub(super) fn write_subms(
78        &self,
79        py: Python<'_>,
80        outputms: String,
81        spw: String,
82        selection: MsSelection,
83        options: MsTransformTaskOptions,
84        history: Option<String>,
85    ) -> PyResult<bool> {
86        let operation = if history.is_some() {
87            "split"
88        } else {
89            "partition"
90        };
91        py.detach(|| {
92            let attached = self.attached.lock().map_err(runtime_error)?;
93            let Some(ms) = attached.as_ref() else {
94                if operation == "split" {
95                    return Err(runtime_error("Table column DATA_DESC_ID is unknown"));
96                }
97                return Ok(false);
98            };
99            if operation == "partition" && spw.contains(':') {
100                return Ok(false);
101            }
102            let link = ms.working_metadata_link();
103            let view = link.working_view().map_err(runtime_error)?;
104            let input = view.path();
105            let output = (!outputms.is_empty())
106                .then(|| named_path(Path::new(&outputms), &ms.path, operation))
107                .transpose()?;
108            let destination = output
109                .as_deref()
110                .map(|output| {
111                    check_named_output_path(input, output).map_err(runtime_error)?;
112                    SortedDestination::inspect(output, operation)
113                })
114                .transpose()?;
115            let locks = match output.as_deref().zip(destination.as_ref()) {
116                Some((output, destination)) if destination.is_table() => {
117                    check_write_access(output).map_err(runtime_error)?;
118                    lock_table_tree_for_replacement(output).map_err(runtime_error)?
119                }
120                _ => Vec::new(),
121            };
122            let scratch = output
123                .as_deref()
124                .and_then(Path::parent)
125                .map_or_else(tempfile::tempdir, tempfile::tempdir_in)
126                .map_err(runtime_error)?;
127            let staged = scratch.path().join(format!("{operation}.ms"));
128            let request = MsTransformRequest {
129                input_ms: input.to_path_buf(),
130                output_ms: staged.clone(),
131                spw,
132                width: 1,
133                data_column: TransformDataColumn::All,
134                selection,
135                keep_flags: true,
136            };
137            link.verify_directory().map_err(runtime_error)?;
138            match mstransform_task(&request, &options) {
139                Err(MsTransformError::EmptySelection { .. }) => {
140                    if let Some(field) = &request.selection.field {
141                        let source = link.snapshot(None).map_err(runtime_error)?;
142                        source
143                            .ms
144                            .resolve_field_selector(field)
145                            .map_err(runtime_error)?;
146                    }
147                    return Ok(false);
148                }
149                Err(MsTransformError::ChannelWidthCount) if operation == "split" => {
150                    return Ok(false);
151                }
152                result => result.map_err(runtime_error)?,
153            };
154            if let Some(parameters) = history.as_ref()
155                && output.is_some()
156            {
157                let table = Table::open(TableOptions::new(&staged)).map_err(runtime_error)?;
158                let message = format!("{outputms} split from {}", ms.path.display());
159                append_tool_history(
160                    &table,
161                    true,
162                    &[message],
163                    parameters,
164                    "ms::split()",
165                    "ms",
166                    &mut Default::default(),
167                )
168                .map_err(runtime_error)?;
169            }
170            link.verify_directory().map_err(runtime_error)?;
171            if !locks
172                .iter()
173                .all(|lock| lock.matches_table_path(lock.path()))
174            {
175                return Err(runtime_error(format!(
176                    "{operation} output changed during staging"
177                )));
178            }
179            if let Some((destination, output)) = destination.zip(output) {
180                destination.publish(&staged, &output, operation)?;
181            }
182            Ok(true)
183        })
184    }
185}

187fn named_path(output: &Path, input: &Path, operation: &str) -> PyResult<PathBuf> {
188    let parent = output
189        .parent()
190        .filter(|path| !path.as_os_str().is_empty())
191        .unwrap_or_else(|| Path::new("."))
192        .canonicalize()
193        .map_err(runtime_error)?;
194    let output =
195        parent.join(output.file_name().ok_or_else(|| {
196            runtime_error(format!("{operation} output must name a MeasurementSet"))
197        })?);
198    if input.starts_with(&output) {
199        return Err(runtime_error(format!(
200            "{operation} output would replace its input"
201        )));
202    }
203    Ok(output)
204}

243pub(super) fn visibility_columns(whichcol: &str) -> PyResult<Option<Vec<OptionalMainColumn>>> {
244    use OptionalMainColumn::{CorrectedData, Data, FloatData, LagData, ModelData};
245    let text = whichcol.to_ascii_uppercase();
246    if text.contains("ALL") || text.contains("ANY") {
247        return Ok(None);
248    }
249    let columns = text
250        .split([' ', ','])
251        .filter_map(|token| match token {
252            "DATA" | "OBSERVED" => Some(Data),
253            "MODEL" | "MODEL_DATA" => Some(ModelData),
254            "CORRECTED" | "CORRECTED_DATA" => Some(CorrectedData),
255            "FLOAT" | "FLOAT_DATA" => Some(FloatData),
256            "LAG" | "LAG_DATA" => Some(LagData),
257            _ => None,
258        })
259        .collect::<Vec<_>>();
260    if columns.is_empty() {
261        Err(runtime_error("did not find and select any data columns"))
262    } else {
263        Ok(Some(columns))
264    }
265}

267pub(super) fn time_average(
268    timebin: Option<&Variant>,
269    combine: &str,
270) -> PyResult<Option<TimeAverageOptions>> {
271    let seconds = timebin
272        .map(|value| quantity_value(value, "s"))
273        .transpose()
274        .map_err(runtime_error)?
275        .unwrap_or(0.0);
276    if !seconds.is_finite() {
277        return Err(runtime_error("partition timebin must be finite"));
278    }
279    let combine = combine.to_ascii_lowercase();
280    Ok((seconds > 0.0).then_some(TimeAverageOptions {
281        interval_seconds: seconds,
282        span_scans: combine.contains("scan"),
283        span_states: combine.contains("state"),
284        span_fields: false,
285        max_uvw_distance: 0.0,
286    }))
287}

289pub(super) fn tile_layout(tileshape: Option<&Variant>) -> PyResult<Option<MsTransformTiling>> {
290    let Some(Variant::Numbers(values) | Variant::Array { values, .. }) = tileshape else {
291        return Ok(None);
292    };
293    let [correlations, channels, rows] = values.as_slice() else {
294        return Ok(None);
295    };
296    let dimensions = [*correlations, *channels, *rows];
297    if !dimensions
298        .iter()
299        .all(|value| (0.0..=f64::from(i32::MAX)).contains(value))
300    {
301        return Err(runtime_error(
302            "partition tile dimensions must be nonnegative 32-bit integers",
303        ));
304    }
305    Ok(Some(MsTransformTiling::Fixed(
306        dimensions.map(|value| value as usize),
307    )))
