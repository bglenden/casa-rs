2//! `imsubimage` - CASA-style image section extraction, on the shared task
3//! front end.
4
5use std::path::PathBuf;
6
7use crate::{TaskFailure, TaskFrontEnd, TaskInvocation, TaskSuccess};
8use casa_imageanalysis::{ImsubimageRequest, imsubimage};
9
10use super::front_end::{image_task_front_end, image_task_success};
11use super::task_contract::ImageAnalysisTaskResult;
12
13/// The front end of the `imsubimage` binary.
14pub fn imsubimage_front_end()
15-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
16    image_task_front_end("imsubimage", &["imsubimage"], run)
17}
18
19pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
20    let outfile = task
21        .text("outfile")?
22        .map_or_else(PathBuf::new, |path| task.io_path(path));
23    let request = ImsubimageRequest {
24        imagename: task.input_path("imagename")?,
25        outfile,
26        box_pixels: task.text("box")?,
27        region: super::front_end::region_in_workspace(task)?,
28        chans: task.text("chans")?,
29        stokes: task.text("stokes")?,
30        mask: task.text("mask")?,
31        dropdeg: task.flag("dropdeg")?,
32        overwrite: task.flag("overwrite")?,
33        verbose: task.flag("verbose")?,
34        stretch: task.flag("stretch")?,
35        keepaxes: task.integers("keepaxes")?,
36    };
37    let summary =
38        imsubimage(&request).map_err(|error| TaskFailure::execution(error.to_string()))?;
39    let msg = format!(
40        "Created subimage '{}' with shape {:?}",
41        summary.outfile, summary.shape
42    );
43    let mut success = image_task_success(Ok(ImageAnalysisTaskResult::Imsubimage(summary)))?;
44    if request.verbose {
45        let seconds = std::time::SystemTime::now()
46            .duration_since(std::time::UNIX_EPOCH)
47            .map_or(0.0, |elapsed| elapsed.as_secs_f64());
48        let mjd_seconds = seconds + 40_587.0 * 86_400.0;
49        success = success.with_replayed_info(mjd_seconds, "imsubimage", &msg, false);
50    }
51    Ok(success)
52}
