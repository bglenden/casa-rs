3//! binding (57) with its parameters and effective defaults (CASA 6.7.6), in
4//! the shared [`crate::tools::surface`] form.

6//! `python/tests/test_tools_quanta.py` keeps this table, the pyo3

19const V: [Parameter; 1] = [param("v", K::Quantity, Empty)];
20const V_A: [Parameter; 2] = [
21    param("v", K::Quantity, Empty),
22    param("a", K::Quantity, Empty),
23];
24const QUANTITY: [Parameter; 3] = [
25    param("v", K::Any, Empty),

29const FORMAT: [Parameter; 4] = [
30    param("v", K::Quantity, Empty),
31    param("prec", K::Int, Int(0)),
32    param("form", K::Texts, Texts(&[])),
33    param("showform", K::Bool, Bool(false)),
34];
35
36/// A one-argument quantity method.
37const fn unary(name: &'static str, returns: Returns, status: Status) -> Method {
38    method(name, &V, returns, status)
39}
40
41/// A two-argument quantity method.
42const fn binary(name: &'static str, returns: Returns, status: Status) -> Method {
43    method(name, &V_A, returns, status)
44}
