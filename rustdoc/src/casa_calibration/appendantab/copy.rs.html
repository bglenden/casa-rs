14pub(super) struct Subtables {
15    pub(super) gain: PathBuf,
16    pub(super) tsys: PathBuf,
17}
18
19pub(super) fn stage(
20    ms: &MeasurementSet,
21    source: &Path,
22    destination: &Path,
23    import: bool,
24) -> Result<Subtables, String> {
25    #[cfg(unix)]
26    let _gain = source
27        .join("GAIN_CURVE")
28        .is_dir()
29        .then(|| retained_table(&source.join("GAIN_CURVE")))
30        .transpose()?;
31    copy_tree(source, destination, &mut HashSet::new())?;
32    let path = |name| {
33        if import {
34            stage_linked(ms, source, destination, name)
35        } else {
36            Ok(destination.join(name))
37        }
38    };
39    Ok(Subtables {
40        gain: path("GAIN_CURVE")?,
41        tsys: path("SYSCAL")?,
42    })
43}
44
45fn stage_linked(
46    ms: &MeasurementSet,
47    source: &Path,
48    destination: &Path,
49    name: &str,
50) -> Result<PathBuf, String> {
51    let conventional = destination.join(name);
52    let Some(reference) = ms
53        .main_table()
54        .keywords()
55        .get(name)
56        .and_then(casa_types::Value::as_table_ref)
57    else {
58        return Ok(conventional);
59    };
60    let referenced = lexical_absolute(&add_directory(reference, source));
61    if let Ok(relative) = referenced.strip_prefix(source)
62        && !relative.as_os_str().is_empty()
63        && destination.join(relative).is_dir()
64    {
65        return Ok(destination.join(relative));
66    }
67    let independent = if conventional.symlink_metadata().is_ok() {
68        // Preserve an unlinked conventional table when MAIN names another.
69        tempfile::Builder::new()
70            .prefix(".antab-subtable-")
71            .tempdir_in(destination)
72            .map_err(message)?
73            .keep()
74            .join(name)
75    } else {
76        conventional
77    };
78    copy_tree(&referenced, &independent, &mut HashSet::new())?;
79    Ok(independent)
80}

82fn copy_tree(
83    source: &Path,
84    destination: &Path,
85    ancestors: &mut HashSet<PathBuf>,
86) -> Result<(), String> {
87    let source = fs::canonicalize(source).map_err(message)?;
88    if ancestors.contains(&source) {
89        return Err(format!(
90            "cyclic MeasurementSet link at {}",
91            source.display()
92        ));
93    }
94    #[cfg(unix)]
95    let _retained = source
96        .join("table.dat")
97        .is_file()
98        .then(|| retained_table(&source))
99        .transpose()?;
100    copy_locked_table_tree(&source, destination).map_err(message)?;
101    own_storage(&source, destination, ancestors)
102}
103
104fn own_storage(
105    source: &Path,
106    destination: &Path,
107    ancestors: &mut HashSet<PathBuf>,
108) -> Result<(), String> {
109    ancestors.insert(source.to_owned());
110    for entry in fs::read_dir(destination).map_err(message)? {
111        let entry = entry.map_err(message)?;
112        let kind = entry.file_type().map_err(message)?;
113        let original = source.join(entry.file_name());
114        let copied = entry.path();
115        if kind.is_symlink() {
116            let resolved = fs::canonicalize(&original).map_err(message)?;
117            let metadata = resolved.metadata().map_err(message)?;
118            fs::remove_file(&copied).map_err(message)?;
119            if metadata.is_dir() {
120                copy_tree(&resolved, &copied, ancestors)?;
121            } else if metadata.is_file() {
122                fs::copy(&resolved, &copied).map_err(message)?;
123            } else {
124                return Err(format!(
125                    "cannot copy MeasurementSet link {}",
126                    original.display()
127                ));
128            }
129        } else if kind.is_dir() {
130            own_storage(&original, &copied, ancestors)?;
131        }
132    }
133    ancestors.remove(source);
134    Ok(())
135}

137#[cfg(unix)]
138fn retained_table(path: &Path) -> Result<casa_tables::Table, String> {
139    use casa_tables::{LockMode, LockOptions, Table, TableOptions};
140    Table::open_with_lock(
141        TableOptions::new(path),
142        LockOptions::new(LockMode::AutoLocking),
143    )
144    .map_err(message)
145}
