2//! The VLA's and the EVLA's primary beams by frequency, as CASA's `PBMath`
3//! gives them (`synthesis/TransformMachines`):
4//! - [`VlaBand`]: the band a frequency selects among the VLA's common beams
5//!   (`PBMath::whichCommonPBtoUse`), each an annular-aperture pattern
6//!   ([`crate::AnnularApertureVoltageTable`]);
7//! - [`EvlaBand`] and [`EvlaPowerPattern`]: the EVLA's polynomial power
8//!   patterns (`PBMath1DEVLA`, `PBMath1DPoly`), one fit per tabulated
9//!   frequency.
10

15/// A band of the VLA's common beams (`PBMath::whichCommonPBtoUse`).
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum VlaBand {
18    /// `VLA_4`, below 0.1 GHz.
19    FourMetre,
20    /// `VLA_P`, 0.2–0.4 GHz.
21    P,
22    /// `VLA_L`, 1–2 GHz.
23    L,
24    /// `VLA_C`, 4–7 GHz.
25    C,
26    /// `VLA_X`, 7–11 GHz.
27    X,
28    /// `VLA_U`, 11–19 GHz.
29    U,
30    /// `VLA_K`, 19–35 GHz.
31    K,
32    /// `VLA_Q`, 35–55 GHz.
33    Q,
34}
35
36impl VlaBand {
37    /// The band of `frequency_hz` in CASA's open intervals, tested from Q
38    /// down, or `None` outside them, where CASA takes its `VLA_NVSS`
39    /// pattern.
40    #[must_use]
41    pub fn of(frequency_hz: f64) -> Option<Self> {
42        let ghz = frequency_hz / 1.0e9;
43        let within = |low: f64, high: f64| ghz > low && ghz < high;
44        [
45            (35.0, 55.0, Self::Q),
46            (19.0, 35.0, Self::K),
47            (11.0, 19.0, Self::U),
48            (7.0, 11.0, Self::X),
49            (4.0, 7.0, Self::C),
50            (1.0, 2.0, Self::L),
51            (0.2, 0.4, Self::P),
52        ]
53        .into_iter()
54        .find_map(|(low, high, band)| within(low, high).then_some(band))
55        .or((ghz < 0.1).then_some(Self::FourMetre))
56    }
57}

59/// A band of the EVLA's polynomial beams (`PBMath1DEVLA::feed`).
60#[derive(Clone, Copy, Debug, PartialEq, Eq)]
61pub enum EvlaBand {
62    /// P, 224–480 MHz.
63    P,
64    /// L, 900–2003 MHz.
65    L,
66    /// S, 1990–4001 MHz.
67    S,
68    /// C, 3990–8001 MHz.
69    C,
70    /// X, 7990–12001 MHz.
71    X,
72    /// U (Ku), 12000–18000 MHz.
73    U,
74    /// K, 19000–26000 MHz.
75    K,
76    /// A (Ka), 28000–38000 MHz.
77    A,
78    /// Q, 41000–50000 MHz.
79    Q,
80}
81
82impl EvlaBand {
83    /// The band of `frequency_hz`: the first of CASA's open intervals, in
84    /// its order, that holds it (the intervals overlap at their edges), or
85    /// `None` between them.
86    #[must_use]
87    pub fn of(frequency_hz: f64) -> Option<Self> {
88        let mhz = frequency_hz / 1.0e6;
89        [
90            (224.0, 480.0, Self::P),
91            (900.0, 2003.0, Self::L),
92            (1990.0, 4001.0, Self::S),
93            (3990.0, 8001.0, Self::C),
94            (7990.0, 12001.0, Self::X),
95            (12000.0, 18000.0, Self::U),
96            (19000.0, 26000.0, Self::K),
97            (28000.0, 38000.0, Self::A),
98            (41000.0, 50000.0, Self::Q),
99        ]
100        .into_iter()
101        .find_map(|(low, high, band)| (mhz > low && mhz < high).then_some(band))
102    }
103
104    /// `frequency_hz` kept within the band's fits
105    /// (`PBMath1DEVLA::limitFreqForBand`).
106    fn limit(self, frequency_hz: f64) -> f64 {
107        // (whether the low edge itself is limited, the low edge, its limit,
108        // the high edge, its limit), in MHz: P and L limit at or below their
109        // low edge, the others below it; every band at or above its high
110        // edge.
111        let (inclusive_low, low, low_limit, high, high_limit) = match self {
112            Self::P => (true, 224.0, 232.0, 480.0, 470.0),
113            Self::L => (true, 900.0, 1040.0, 2000.0, 2000.0),
114            Self::S => (false, 2052.0, 2052.0, 3948.0, 3948.0),
115            Self::C => (false, 4052.0, 4052.0, 7948.0, 7948.0),
116            Self::X => (false, 8052.0, 8052.0, 11948.0, 11948.0),
117            Self::U => (false, 12052.0, 12052.0, 17948.0, 17948.0),
118            Self::K => (false, 19052.0, 19052.0, 25948.0, 25948.0),
119            Self::A => (false, 28052.0, 28052.0, 38048.0, 38048.0),
120            Self::Q => (false, 41052.0, 41052.0, 43948.0, 43948.0),
121        };
122        let mut frequency_hz = frequency_hz;
123        let below = if inclusive_low {
124            frequency_hz <= low * 1.0e6
125        } else {
126            frequency_hz < low * 1.0e6
127        };
128        if below {
129            frequency_hz = low_limit * 1.0e6;
130        }
131        if frequency_hz >= high * 1.0e6 {
132            frequency_hz = high_limit * 1.0e6;
133        }
134        frequency_hz
135    }
136}

138/// One of the EVLA's polynomial power patterns (`PBMath1DPoly`), as
139/// `PBMath1DEVLA::nearestVPArray` picks it for a frequency.
140#[derive(Clone, Copy, Debug, PartialEq)]
141pub struct EvlaPowerPattern {
142    frequency_mhz: f64,
143    coefficients: &'static [f64],
144}
145
146impl EvlaPowerPattern {
147    /// The radius the pattern is tabulated to, in arcmin at 1 GHz
148    /// (`PBMath::initByTelescope`: 58′).
149    pub const MAXIMUM_RADIUS_ARCMIN_GHZ: f64 = 58.0;
150
151    /// CASA's table samples (`PBMath1DPoly::fillPBArray`).
152    const SAMPLES: f64 = 10_000.0;
153
154    /// The pattern CASA applies at `frequency_hz`
155    /// (`PBMath1DEVLA::nearestVPArray`): the frequency is kept within its
156    /// band's fits ([`EvlaBand::of`], `limitFreqForBand`; a frequency
157    /// between bands is kept as it is), and the fit of the nearest tabulated
158    /// frequency, the higher on a tie, is taken from the whole table.
159    #[must_use]
160    pub fn at(frequency_hz: f64) -> Self {
161        let frequency_hz =
162            EvlaBand::of(frequency_hz).map_or(frequency_hz, |band| band.limit(frequency_hz));
163        let mhz = frequency_hz * 1.0e-6;
164        // `std::map::lower_bound`: the first fit at or above the frequency.
165        let upper = EVLA_COEFFICIENTS.partition_point(|(fit, _)| *fit < mhz);
166        let index = if upper == EVLA_COEFFICIENTS.len() {
167            upper - 1
168        } else if upper == 0 {
169            0
170        } else if (mhz - EVLA_COEFFICIENTS[upper - 1].0).abs()
171            < (EVLA_COEFFICIENTS[upper].0 - mhz).abs()
172        {
173            upper - 1
174        } else {
175            upper
176        };
177        let (frequency_mhz, coefficients) = EVLA_COEFFICIENTS[index];
178        Self {
179            frequency_mhz,
180            coefficients,
181        }
182    }
183
184    /// The tabulated frequency of the fit, in MHz.
185    #[must_use]
186    pub const fn frequency_mhz(&self) -> f64 {
187        self.frequency_mhz
188    }
189
190    /// The fit's even-power coefficients: the power is `Σ c_k x^k`, with
191    /// `x` the square of the radius in arcmin times the frequency in GHz.
192    #[must_use]
193    pub const fn coefficients(&self) -> &'static [f64] {
194        self.coefficients
195    }
196
197    /// The power at `radius_arcmin_ghz`, the radius in arcmin times the
198    /// frequency in GHz. As CASA tabulates the pattern
199    /// (`PBMath1DPoly::fillPBArray`, `PBMath1D::apply`): 10,000 samples to
200    /// [`Self::MAXIMUM_RADIUS_ARCMIN_GHZ`], the lower sample taken, the
201    /// voltage stored as `Float` (the square root of the power) and
202    /// squared. Zero beyond the table and where the polynomial is not
203    /// positive.
204    #[must_use]
205    pub fn power(&self, radius_arcmin_ghz: f64) -> f32 {
206        if !(radius_arcmin_ghz.is_finite()
207            && (0.0..=Self::MAXIMUM_RADIUS_ARCMIN_GHZ).contains(&radius_arcmin_ghz))
208        {
209            return 0.0;
210        }
211        let inverse_increment = (Self::SAMPLES - 1.0) / Self::MAXIMUM_RADIUS_ARCMIN_GHZ;
212        let sampled = (radius_arcmin_ghz * inverse_increment).floor() / inverse_increment;
213        let x2 = sampled * sampled;
214        let mut response = 0.0;
215        let mut power = 1.0;
216        for coefficient in self.coefficients {
217            response += coefficient * power;
218            power *= x2;
219        }
220        if response <= 0.0 {
221            0.0
222        } else {
223            let voltage = response.sqrt() as f32;
224            voltage * voltage
225        }
226    }
227}
