3//! Samples along a polyline in an image's pixel plane.
4
5use crate::resample::{Plane, RegridMethod, RegridPixel};
6use crate::{ImageError, ImageInterface};
7
8/// Samples, valid-mask flags and positions along a pixel-space polyline.
9#[derive(Debug)]
10pub struct PixelSlice<T> {
11    /// Interpolated samples. Invalid samples have the pixel type's zero value.
12    pub pixels: Vec<T>,
13    /// True for samples whose interpolation stencil is valid.
14    pub mask: Vec<bool>,
15    /// Sample positions as `[x, y]`, rounded to the reference tool's Float precision.
16    pub positions: Vec<[f32; 2]>,
17    /// Cumulative straight-line distances between consecutive output positions.
18    pub distance: Vec<f32>,
19}
20
21fn failed(message: impl Into<String>) -> ImageError {
22    ImageError::CasaSelection(message.into())
23}
24
25fn positions(x: &[f64], y: &[f64], npts: usize) -> Result<Vec<[f32; 2]>, ImageError> {
26    if x.len() != y.len() {
27        return Err(failed("Failed AlwaysAssert x.nelements() == y.nelements()"));
28    }
29    if x.len() < 2 {
30        return Err(failed("Failed AlwaysAssert x.nelements() >= 2"));
31    }
32
33    let lengths: Vec<_> = x
34        .windows(2)
35        .zip(y.windows(2))
36        .map(|(x, y)| (x[1] - x[0]).hypot(y[1] - y[0]))
37        .collect();
38    let length = lengths.iter().sum::<f64>();
39    if !length.is_finite() || x.iter().chain(y).any(|value| !(*value as f32).is_finite()) {
40        return Err(failed("Slice positions must be finite"));
41    }
42    let count = if npts == 0 {
43        ((length + 0.1) as usize).saturating_add(1)
44    } else {
45        npts
46    };
47    if length == 0.0 {
48        return if count == 1 {
49            Ok(vec![[x[0] as f32, y[0] as f32]])
50        } else {
51            Err(failed("Failed AlwaysAssert np == npoints"))
52        };
53    }
54
55    let mut output = Vec::new();
56    output
57        .try_reserve_exact(count)
58        .map_err(|_| failed("Slice has too many sample positions"))?;
59    let step = length / (count - 1) as f64;
60    let mut segment = 0;
61    let mut start = 0.0;
62    for sample in 0..count {
63        let at = if sample == 0 {
64            0.0
65        } else {
66            step * sample as f64
67        };
68        while segment < lengths.len() - 1 && at >= start + lengths[segment] {
69            start += lengths[segment];
70            segment += 1;
71        }
72        let span = lengths[segment];
73        let fraction = if span > 0.0 {
74            ((at - start) / span).clamp(0.0, 1.0)
75        } else {
76            0.0
77        };
78        let px = x[segment] + fraction * (x[segment + 1] - x[segment]);
79        let py = y[segment] + fraction * (y[segment + 1] - y[segment]);
80        output.push([px as f32, py as f32]);
81    }
82    Ok(output)
83}

85/// Read only the selected image plane and interpolate along its pixel polyline.
86///
87/// The interpolation stencils and mask semantics match `Interpolate2D`.
88/// Plane storage follows ascending image-axis order, as in casacore's
89/// degenerate-axis-dropping lattice slice. The endpoint domain checks use
90/// the original order requested by the caller.
91///
92/// # Errors
93///
94/// Returns an error for an invalid path, axis, or fixed coordinate, or if
95/// the source cannot provide the requested plane.
96#[allow(private_bounds)]
97pub fn get_slice<T: RegridPixel>(
98    image: &(impl ImageInterface<T> + ?Sized),
99    x: &[f64],
100    y: &[f64],
101    axes: [i64; 2],
102    coord: &[i64],
103    npts: usize,
104    method: RegridMethod,
105) -> Result<PixelSlice<T>, ImageError> {
106    let positions = positions(x, y, npts)?;
107    let shape = image.shape();
108    let axis_index = |requested, name| {
109        usize::try_from(requested)
110            .ok()
111            .filter(|&axis| axis < shape.len())
112            .ok_or_else(|| failed(format!("Failed AlwaysAssert {name}<itsLatticePtr->ndim()")))
113    };
114    let axes = [axis_index(axes[0], "axis0")?, axis_index(axes[1], "axis1")?];
115    if coord != [-1] && coord.len() != shape.len() {
116        return Err(failed("coord must be of length number of image dimensions"));
117    }
118    let [first_x, first_y] = positions[0];
119    if first_x < -0.5 || first_y < -0.5 {
120        return Err(failed("x or y start of curve falls outside of lattice"));
121    }
122    let [last_x, last_y] = *positions.last().expect("at least one position");
123    if f64::from(last_x) > shape[axes[0]] as f64 - 0.5
124        || f64::from(last_y) > shape[axes[1]] as f64 - 0.5
125    {
126        return Err(failed("x or y end of curve falls outside of lattice"));
127    }
128
129    let mut start = vec![0; shape.len()];
130    let mut count = vec![1; shape.len()];
131    for (axis, &length) in shape.iter().enumerate() {
132        if axes.contains(&axis) {
133            count[axis] = length;
134        } else if coord != [-1] {
135            let requested = coord[axis];
136            let requested = if requested < 0 {
137                requested + length as i64
138            } else {
139                requested
140            };
141            start[axis] = usize::try_from(requested)
142                .ok()
143                .filter(|&index| index < length)
144                .ok_or_else(|| {
145                    failed(format!(
146                        "slice coordinate on axis {axis} falls outside of lattice"
147                    ))
148                })?;
149        }
150    }
151    let step = vec![1; shape.len()];
152    let data = image.get_slice(&start, &count, &step)?;
153    let mask = image
154        .get_mask_slice(&start, &count, &step)?
155        .map(|mask| mask.reversed_axes().iter().copied().collect::<Vec<_>>());
156
157    let mut dimensions = count.into_iter().filter(|&length| length != 1);
158    let shape = [
159        dimensions.next().unwrap_or(1),
160        dimensions.next().unwrap_or(1),
161    ];
162    let data = data.reversed_axes();
163    let components: Vec<Vec<f64>> = (0..T::COMPONENTS)
164        .map(|component| {
165            data.iter()
166                .map(|pixel| pixel.component(component))
167                .collect()
168        })
169        .collect();
170    let planes: Vec<_> = components
171        .iter()
172        .map(|data| Plane {
173            data,
174            mask: mask.as_deref(),
175            shape,
176        })
177        .collect();
178
179    let mut samples = Vec::with_capacity(positions.len());
180    let mut valid = Vec::with_capacity(positions.len());
181    for &position in &positions {
182        let at = position.map(f64::from);
183        let mut pixel = T::default_value();
184        let mut good = true;
185        for (component, plane) in planes.iter().enumerate() {
186            let Some(value) = plane.sample(method, at) else {
187                good = false;
188                break;
189            };
190            pixel = pixel.with_component(component, value);
191        }
192        samples.push(pixel);
193        valid.push(good);
194    }
195    let mut distance = vec![0.0; positions.len()];
196    for (index, pair) in positions.windows(2).enumerate() {
197        let dx = pair[1][0] - pair[0][0];
198        let dy = pair[1][1] - pair[0][1];
199        distance[index + 1] = distance[index] + (dx * dx + dy * dy).sqrt();
200    }
201    Ok(PixelSlice {
202        pixels: samples,
203        mask: valid,
204        positions,
205        distance,
206    })
207}
