3use std::path::Path;
4use std::sync::PoisonError;
5
6use casa_singledish::{SingleDishLineFinder, SingleDishLineFit};
7use pyo3::exceptions::PyRuntimeError;
8use pyo3::prelude::*;
9
10use crate::convert::{IntList, Variant};
11use crate::tools::calibrater::selector;
12
13use super::PySingleDishMs;
14
15#[pymethods]
16impl PySingleDishMs {
17    #[pyo3(signature = (datacolumn = "data", spw = Variant::Empty, pol = Variant::Empty, timebin = "", timespan = "", polaverage = "", fitfunc = "gaussian", nfit = "0", linefinding = false, threshold = 5.0, avg_limit = 4, minwidth = 4, edge = None, tempfile = "", tempoutfile = ""), text_signature = "($self, datacolumn='data', spw=[], pol=[], timebin='', timespan='', polaverage='', fitfunc='gaussian', nfit='0', linefinding=False, threshold=5.0, avg_limit=4, minwidth=4, edge=[0, 0], tempfile='', tempoutfile='')")]
18    #[allow(clippy::too_many_arguments)]
19    fn fit_line(
20        &self,
21        py: Python<'_>,
22        datacolumn: &str,
23        spw: Variant,
24        pol: Variant,
25        timebin: &str,
26        timespan: &str,
27        polaverage: &str,
28        fitfunc: &str,
29        nfit: &str,
30        linefinding: bool,
31        threshold: f64,
32        avg_limit: i64,
33        minwidth: i64,
34        edge: Option<IntList>,
35        tempfile: &str,
36        tempoutfile: &str,
37    ) -> PyResult<bool> {
38        let spw = selector(&spw).unwrap_or_default();
39        // The CASA tool uses the correlation from set_selection.
40        drop(pol);
41        py.detach(|| {
42            self.session
43                .lock()
44                .unwrap_or_else(PoisonError::into_inner)
45                .fit_line(SingleDishLineFit {
46                    datacolumn,
47                    spw: &spw,
48                    timebin,
49                    timespan,
50                    polaverage,
51                    fitfunc,
52                    nfit,
53                    finder: linefinding.then(|| {
54                        SingleDishLineFinder::new(
55                            threshold,
56                            avg_limit,
57                            minwidth,
58                            edge.as_ref().map_or(&[], |edge| edge.0.as_slice()),
59                        )
60                    }),
61                    tempfile: Path::new(tempfile),
62                    tempoutfile: Path::new(tempoutfile),
63                })
64        })
65        .map_err(PyRuntimeError::new_err)
66    }
67}
