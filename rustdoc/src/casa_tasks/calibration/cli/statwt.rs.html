15pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
16    let path = task.input_path("vis")?;
17    let options = options(task)?;
18    let selectdata = task.flag("selectdata")?;
19    let selector = |name: &str| {
20        task.text(name)
21            .map(|value| value.filter(|text| selectdata && !text.is_empty()))
22    };
23    let selection = MsSelection {
24        selectdata,
25        field: selector("field")?,
26        spw: selector("spw")?,
27        intent: selector("intent")?,
28        array: selector("array")?,
29        observation: selector("observation")?,
30        scan: selector("scan")?,
31        ..MsSelection::default()
32    };
33    let result = if task.flag("preview")? {
34        let mut ms = StatWtMs::open(&path, true).map_err(runtime_error)?;
35        let selection = ms.select(&selection).map_err(runtime_error)?;
36        estimate_statwt(&mut ms, &selection, &options, None).map_err(runtime_error)?
37    } else {
38        let paths = visibility_model_source_paths(&path).map_err(runtime_error)?;
39        let mut locks = StatWtMs::write_locks(&paths).map_err(runtime_error)?;
40        let mut ms = StatWtMs::open(&path, false).map_err(runtime_error)?;
41        let selection = ms.select(&selection).map_err(runtime_error)?;
42        if task.flag("flagbackup")? {
43            if paths.len() == 1 && path.canonicalize().is_ok_and(|path| paths[0] == path) {
44                let ms = MeasurementSet::open(&paths[0]).map_err(runtime_error)?;
45                save_next_flag_backup(&ms, "statwt").map_err(runtime_error)?;
46            } else {
47                save_next_partition_flag_backup(&path, &paths, "statwt").map_err(runtime_error)?;
48            }
49        }
50        let result = estimate_statwt(&mut ms, &selection, &options, Some(&mut locks))
51            .map_err(runtime_error)?;
52        drop(ms);
53        for lock in locks.into_values() {
54            lock.release().map_err(runtime_error)?;
55        }
56        super::history::record_task(task, &path);
57        result
58    };
59    let value = CasaValue::dict([
60        ("mean", CasaValue::float(result.mean)),
61        ("variance", CasaValue::float(result.variance)),
62    ]);
63    TaskSuccess::new(value, &result)
64}

66fn options(task: &TaskInvocation<'_>) -> Result<StatWtOptions, TaskFailure> {
67    let algorithm = StatWtAlgorithm::parse(
68        &task.text("statalg")?.unwrap_or_default(),
69        task.number("fence")?,
70        &task.text("center")?.unwrap_or_default(),
71        task.flag("lside")?,
72        task.number("zscore")?,
73        task.integer("maxiter")?,
74    )
75    .map_err(runtime_error)?;
76    let combine = StatWtCombine::parse(&task.text("combine")?.unwrap_or_default());
77    let minimum_samples = usize::try_from(task.integer("minsamp")?).unwrap_or(0);
78    let weight_range = statwt_weight_range(&task.numbers("wtrange")?)
79        .map_err(|error| runtime_error(error).at("wtrange"))?;
80    let data_column = StatWtDataColumn::parse(&task.text("datacolumn")?.unwrap_or_default())
81        .map_err(|error| runtime_error(error).at("datacolumn"))?;
82    Ok(StatWtOptions {
83        data_column,
84        algorithm,
85        minimum_samples,
86        combine,
87        time_bin: time_bin(task)?,
88        sliding: task.flag("slidetimebin")?,
89        channel_bin: channel_bin(task)?,
90        fit_spw: task.text("fitspw")?.unwrap_or_default(),
91        exclude_channels: task.flag("excludechans")?,
92        weight_range,
93    })
94}

96fn time_bin(task: &TaskInvocation<'_>) -> Result<StatWtTimeBin, TaskFailure> {
97    let bad_value = || {
98        TaskFailure::execution("timebin requires a positive timestamp count or a duration")
99            .at("timebin")
100    };
101    match task.value("timebin") {
102        Some(ParameterValue::Integer(value)) if *value > 0 => usize::try_from(*value)
103            .map(StatWtTimeBin::Timestamps)
104            .map_err(|_| bad_value()),
105        Some(ParameterValue::String(value)) => {
106            StatWtTimeBin::duration(value).map_err(|error| runtime_error(error).at("timebin"))
107        }
108        _ => Err(bad_value()),
109    }
110}
111
112fn channel_bin(task: &TaskInvocation<'_>) -> Result<StatWtChannelBin, TaskFailure> {
113    let bad_value = || {
114        TaskFailure::execution("chanbin requires a positive channel count or a frequency width")
115            .at("chanbin")
116    };
117    match task.value("chanbin") {
118        Some(ParameterValue::Integer(value)) if *value > 0 => usize::try_from(*value)
119            .map(StatWtChannelBin::Channels)
120            .map_err(|_| bad_value()),
121        Some(ParameterValue::String(value)) => {
122            StatWtChannelBin::parse(value).map_err(|error| runtime_error(error).at("chanbin"))
123        }
124        _ => Err(bad_value()),
125    }
126}

128fn runtime_error(error: impl std::fmt::Display) -> TaskFailure {
129    TaskFailure::execution(error.to_string())
