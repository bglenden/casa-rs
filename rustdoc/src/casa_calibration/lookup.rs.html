2//! Calibration-table lookup: a table's solution for an antenna, spectral
3//! window, field and time, found and interpolated in time as the apply
4//! executor finds it, for callers that apply a table to visibilities of
5//! their own (`casa-simulation`'s `sm.setapply`).
6//!
7//! [`CalibrationTableLookup`] loads a complex (`CPARAM`) table as the
8//! executor does and answers [`CalibrationTableLookup::solution`]: the
9//! solutions of the row's field, or of the table's sole field, nearest in
10//! time or interpolated linearly in amplitude and phase. With
11//! [`FieldSelection::AnyField`] it takes every field's solutions as one
12//! series in time instead, as CASA's simulator applies a table. The time
13//! interpolation the executor shares lives here.
14//!
15//! Loading reads the whole table; [`CalibrationTableLookup::rows_on_disk`],
16//! [`CalibrationTableLookup::loading_bytes`] and
17//! [`CalibrationTableLookup::kept_bytes`] size it beforehand, for a caller
18//! that admits the memory first.
19

35/// Which fields' solutions a row takes.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37pub enum FieldSelection {
38    /// The row's field's, or the table's sole field's: the apply
39    /// executor's default.
40    RowField,
41    /// Every field's, as one series in time: CASA's `Simulator::setapply`
42    /// ignores fields and does not partition the table by them (#893
43    /// review).
44    AnyField,
45}
46
47/// Heap loading a complex table holds per row at most: the row as
48/// casa-tables reads the table, and the solution it becomes. Measured at
49/// about 3,300 bytes (the #893 review's 9,996-row probe, and the setapply
50/// law in `casa-simulation`'s `session_memory/apply.rs`), with a quarter
51/// more for the allocator.
52pub const LOOKUP_LOADING_ROW_BYTES: u64 = 4_096;
53
54/// Heap a loaded lookup keeps per row: the solution's grid, its place in
55/// the table's series, and in the series across fields. Measured at about
56/// 320 bytes with the loading figure, with a quarter more.
57pub const LOOKUP_KEPT_ROW_BYTES: u64 = 400;
58
59/// A complex calibration table, loaded to look its solutions up.
60pub struct CalibrationTableLookup {
61    path: PathBuf,
62    table: LoadedCalibrationTable,
63    subtype: String,
64    interp: ApplyInterpolationMode,
65    /// For [`FieldSelection::AnyField`]: each (spectral window, antenna)'s
66    /// solutions across the fields, in time order.
67    across_fields: Option<HashMap<(i32, i32), Vec<CalibrationSolution>>>,
68}

70/// One antenna's solution: CPARAM's values and FLAG, per receptor and
71/// channel.
72#[derive(Debug, Clone, PartialEq)]
73pub struct ComplexSolution {
74    /// Receptors (CPARAM's first axis).
75    pub receptors: usize,
76    /// Channels (its second axis): one for a term that does not depend on
77    /// frequency.
78    pub channels: usize,
79    /// `[receptor · channels + channel]`.
80    pub values: Vec<Complex32>,
81    /// `[receptor · channels + channel]`.
82    pub flags: Vec<bool>,
83}
84
85impl ComplexSolution {
86    fn of(grid: &GainGrid) -> Self {
87        let (receptors, channels) = (grid.receptor_count, grid.channel_count);
88        let mut values = Vec::with_capacity(receptors * channels);
89        let mut flags = Vec::with_capacity(receptors * channels);
90        for receptor in 0..receptors {
91            for channel in 0..channels {
92                let (value, flag) = if grid.values.ndim() == 1 {
93                    (grid.values[[receptor]], grid.flags[[receptor]])
94                } else {
95                    (
96                        grid.values[[receptor, channel]],
97                        grid.flags[[receptor, channel]],
98                    )
99                };
100                values.push(value);
101                flags.push(flag);
102            }
103        }
104        Self {
105            receptors,
106            channels,
107            values,
108            flags,
109        }
110    }
111
112    /// The value of `receptor` in `channel`.
113    #[must_use]
114    pub fn value(&self, receptor: usize, channel: usize) -> Complex32 {
115        self.values[receptor * self.channels + channel]
116    }
117}

119impl CalibrationTableLookup {
120    /// Load the complex calibration table at `path` to look its solutions
121    /// up with `interp`, each row taking its own field's
122    /// ([`FieldSelection::RowField`]).
123    ///
124    /// # Errors
125    ///
126    /// The table cannot be read, or holds other than complex (`CPARAM`)
127    /// solutions.
128    pub fn open(
129        path: impl AsRef<Path>,
130        interp: ApplyInterpolationMode,
131    ) -> Result<Self, ApplyExecutionError> {
132        Self::open_with(path, interp, FieldSelection::RowField)
133    }
134
135    /// The rows of the table at `path`, read without loading them.
136    ///
137    /// # Errors
138    ///
139    /// The table's control file cannot be read.
140    pub fn rows_on_disk(path: impl AsRef<Path>) -> Result<usize, ApplyExecutionError> {
141        let path = path.as_ref();
142        Table::rows_on_disk(path).map_err(|error| {
143            ApplyExecutionError::UnsupportedCalibrationTable {
144                path: path.display().to_string(),
145                reason: error.to_string(),
146            }
147        })
148    }
149
150    /// Heap loading a table of `rows` rows holds at most, beside what the
151    /// lookup then keeps.
152    #[must_use]
153    pub const fn loading_bytes(rows: usize) -> u64 {
154        rows as u64 * LOOKUP_LOADING_ROW_BYTES
155    }
156
157    /// Heap a lookup of a table of `rows` rows keeps.
158    #[must_use]
159    pub const fn kept_bytes(rows: usize) -> u64 {
160        rows as u64 * LOOKUP_KEPT_ROW_BYTES
161    }
162
163    /// [`Self::open`], each row taking the solutions `fields` selects.
164    ///
165    /// # Errors
166    ///
167    /// As [`Self::open`].
168    pub fn open_with(
169        path: impl AsRef<Path>,
170        interp: ApplyInterpolationMode,
171        fields: FieldSelection,
172    ) -> Result<Self, ApplyExecutionError> {
173        let path = path.as_ref();
174        let unsupported = |reason: String| ApplyExecutionError::UnsupportedCalibrationTable {
175            path: path.display().to_string(),
176            reason,
177        };
178        let summary = summarize_table(path).map_err(|error| unsupported(error.to_string()))?;
179        if summary.parameter_family != CalibrationParameterFamily::Complex
180            || summary.table_subtype == "BPOLY"
181        {
182            return Err(unsupported(format!(
183                "a {} table is not complex (CPARAM)",
184                summary.table_subtype
185            )));
186        }
187        let subtype = summary.table_subtype.clone();
188        let plan = ApplyCalibrationTablePlan {
189            spec: ApplyCalibrationTableSpec {
190                interp,
191                ..ApplyCalibrationTableSpec::new(path)
192            },
193            applicable_selected_row_count: 0,
194            summary,
195            resolved_gainfield: None,
196            resolved_nearest_gainfields: Vec::new(),
197            spw_mapping: Vec::new(),
198            calibration_spectral_windows: Vec::new(),
199            interp,
200            calwt: false,
201        };
202        let table = load_calibration_table(&plan, None)?;
203        let across_fields = (fields == FieldSelection::AnyField).then(|| across_fields(&table));
204        Ok(Self {
205            path: path.to_path_buf(),
206            table,
207            subtype,
208            interp,
209            across_fields,
210        })
211    }
212
213    /// The table's subtype (`G Jones`, `D Jones`, `B Jones`, …).
214    #[must_use]
215    pub fn subtype(&self) -> &str {
216        &self.subtype
217    }
218
219    /// The solution of `antenna` in spectral window `spw` for a row of
220    /// field `field` at `time_s`: from the field's solutions, or the
221    /// table's sole field's (or, with [`FieldSelection::AnyField`], every
222    /// field's), nearest in time or interpolated linearly in amplitude and
223    /// phase. `None` where the table has none.
224    ///
225    /// # Errors
226    ///
227    /// The solutions cannot be interpolated (shapes differ in time).
228    pub fn solution(
229        &self,
230        field: i32,
231        spw: i32,
232        antenna: i32,
233        time_s: f64,
234    ) -> Result<Option<ComplexSolution>, ApplyExecutionError> {
235        let grid = match &self.across_fields {
236            Some(series) => self.across_fields_grid(series, spw, antenna, time_s)?,
237            None => self
238                .table
239                .lookup(field, spw, antenna, time_s, self.interp)?,
240        };
241        let Some(grid) = grid else {
242            return Ok(None);
243        };
244        match grid.as_ref() {
245            CalibrationGrid::Complex(grid) => Ok(Some(ComplexSolution::of(grid))),
246            _ => Err(ApplyExecutionError::UnsupportedCalibrationTable {
247                path: String::new(),
248                reason: "a complex table gave other than complex solutions".to_string(),
249            }),
250        }
251    }
252}

254impl CalibrationTableLookup {
255    /// The grid of `antenna` in window `spw` at `time_s` from the series
256    /// across fields: the nearest solution (the earlier of two as near), or
257    /// the bracketing pair interpolated as the executor interpolates.
258    fn across_fields_grid(
259        &self,
260        series: &HashMap<(i32, i32), Vec<CalibrationSolution>>,
261        spw: i32,
262        antenna: i32,
263        time_s: f64,
264    ) -> Result<Option<Arc<CalibrationGrid>>, ApplyExecutionError> {
265        let Some(candidates) = series
266            .get(&(spw, antenna))
267            .or_else(|| series.get(&(spw, -1)))
268            .filter(|candidates| !candidates.is_empty())
269        else {
270            return Ok(None);
271        };
272        // The first solution at or after the time, and the last at or
273        // before it.
274        let after = candidates.partition_point(|solution| solution.time_seconds < time_s);
275        let through = candidates.partition_point(|solution| solution.time_seconds <= time_s);
276        match self.interp {
277            ApplyInterpolationMode::Nearest | ApplyInterpolationMode::NearestLinear => {
278                let nearest = match (after.checked_sub(1), candidates.get(after)) {
279                    (Some(before), Some(later))
280                        if (later.time_seconds - time_s).abs()
281                            < (time_s - candidates[before].time_seconds).abs() =>
282                    {
283                        after
284                    }
285                    (Some(before), _) => before,
286                    (None, _) => 0,
287                };
288                Ok(Some(Arc::clone(&candidates[nearest].grid)))
289            }
290            ApplyInterpolationMode::Linear => {
291                let last = candidates.len() - 1;
292                let lower = through.checked_sub(1).unwrap_or(after.min(last));
293                let upper = after.min(last).max(lower);
294                interpolate_time_linear(&self.path, &candidates[lower..=upper], time_s).map(Some)
295            }
296        }
297    }
298}

300/// Each (spectral window, antenna)'s solutions across the fields of
301/// `table`, in time order, fields in order at equal times.
302fn across_fields(table: &LoadedCalibrationTable) -> HashMap<(i32, i32), Vec<CalibrationSolution>> {
303    let mut by_window: HashMap<(i32, i32), Vec<(i32, &CalibrationSolution)>> = HashMap::new();
304    for (&(field, spw, antenna), solutions) in &table.solutions {
305        by_window
306            .entry((spw, antenna))
307            .or_default()
308            .extend(solutions.iter().map(|solution| (field, solution)));
309    }
310    by_window
311        .into_iter()
312        .map(|(key, mut solutions)| {
313            solutions.sort_by(|a, b| {
314                a.1.time_seconds
315                    .total_cmp(&b.1.time_seconds)
316                    .then(a.0.cmp(&b.0))
317            });
318            let series = solutions
319                .into_iter()
320                .map(|(_, solution)| CalibrationSolution {
321                    time_seconds: solution.time_seconds,
322                    grid: Arc::clone(&solution.grid),
323                })
324                .collect();
325            (key, series)
326        })
327        .collect()
328}

330pub(crate) fn interpolate_time_linear(
331    path: &Path,
332    candidates: &[CalibrationSolution],
333    time_seconds: f64,
334) -> Result<Arc<CalibrationGrid>, ApplyExecutionError> {
335    let mut sorted = candidates.iter().collect::<Vec<_>>();
336    sorted.sort_by(|a, b| a.time_seconds.total_cmp(&b.time_seconds));
337
338    let lower = sorted
339        .iter()
340        .rev()
341        .find(|solution| solution.time_seconds <= time_seconds)
342        .copied();
343    let upper = sorted
344        .iter()
345        .find(|solution| solution.time_seconds >= time_seconds)
346        .copied();
347
348    match (lower, upper) {
349        (Some(lower), Some(upper))
350            if (upper.time_seconds - lower.time_seconds).abs() > f64::EPSILON =>
351        {
352            let fraction = ((time_seconds - lower.time_seconds)
353                / (upper.time_seconds - lower.time_seconds)) as f32;
354            match (lower.grid.as_ref(), upper.grid.as_ref()) {
355                (CalibrationGrid::Complex(lower), CalibrationGrid::Complex(upper)) => {
356                    if lower.values.shape() != upper.values.shape() {
357                        return Err(ApplyExecutionError::UnsupportedInterpolation {
358                            path: path.display().to_string(),
359                            interp: ApplyInterpolationMode::Linear,
360                            reason: "time interpolation requires matching parameter shapes"
361                                .to_string(),
362                        });
363                    }
364                    let mut values = lower.values.clone();
365                    for (value, upper_value) in values.iter_mut().zip(upper.values.iter()) {
366                        *value = interpolate_gain_amplitude_phase(*value, *upper_value, fraction);
367                    }
368                    let mut flags = lower.flags.clone();
369                    for (flag, upper_flag) in flags.iter_mut().zip(upper.flags.iter()) {
370                        *flag = *flag || *upper_flag;
371                    }
372                    Ok(Arc::new(CalibrationGrid::Complex(GainGrid {
373                        receptor_count: lower.receptor_count,
374                        channel_count: lower.channel_count,
375                        values,
376                        flags,
377                    })))
378                }
379                (CalibrationGrid::Delay(lower), CalibrationGrid::Delay(upper)) => {
380                    if lower.values_ns.shape() != upper.values_ns.shape() {
381                        return Err(ApplyExecutionError::UnsupportedInterpolation {
382                            path: path.display().to_string(),
383                            interp: ApplyInterpolationMode::Linear,
384                            reason: "time interpolation requires matching parameter shapes"
385                                .to_string(),
386                        });
387                    }
388                    let mut values_ns = lower.values_ns.clone();
389                    for (value, upper_value) in values_ns.iter_mut().zip(upper.values_ns.iter()) {
390                        *value = *value + (*upper_value - *value) * fraction;
391                    }
392                    let mut flags = lower.flags.clone();
393                    for (flag, upper_flag) in flags.iter_mut().zip(upper.flags.iter()) {
394                        *flag = *flag || *upper_flag;
395                    }
396                    Ok(Arc::new(CalibrationGrid::Delay(DelayGrid {
397                        receptor_count: lower.receptor_count,
398                        channel_count: lower.channel_count,
399                        values_ns,
400                        flags,
401                    })))
402                }
403                (
404                    CalibrationGrid::Antpos(_)
405                    | CalibrationGrid::GainCurve(_)
406                    | CalibrationGrid::Opacity(_),
407                    _,
408                )
409                | (
410                    _,
411                    CalibrationGrid::Antpos(_)
412                    | CalibrationGrid::GainCurve(_)
413                    | CalibrationGrid::Opacity(_),
414                ) => Err(ApplyExecutionError::UnsupportedInterpolation {
415                    path: path.display().to_string(),
416                    interp: ApplyInterpolationMode::Linear,
417                    reason:
418                        "VLA prior tables are row-geometry dependent and use nearest table rows"
419                            .to_string(),
420                }),
421                _ => Err(ApplyExecutionError::UnsupportedInterpolation {
422                    path: path.display().to_string(),
423                    interp: ApplyInterpolationMode::Linear,
424                    reason: "time interpolation requires matching calibration parameter families"
425                        .to_string(),
426                }),
427            }
428        }
429        (Some(lower), _) => Ok(lower.grid.clone()),
430        (_, Some(upper)) => Ok(upper.grid.clone()),
431        _ => Err(ApplyExecutionError::UnsupportedInterpolation {
432            path: path.display().to_string(),
433            interp: ApplyInterpolationMode::Linear,
434            reason: "no calibration rows available for linear interpolation".to_string(),
435        }),
436    }
437}

439pub(crate) fn interpolate_gain_amplitude_phase(
440    lower: Complex32,
441    upper: Complex32,
442    fraction: f32,
443) -> Complex32 {
444    let lower_amp = lower.norm();
445    let upper_amp = upper.norm();
446    let amp = lower_amp + (upper_amp - lower_amp) * fraction;
447    let lower_phase = lower.arg();
448    let mut upper_phase = upper.arg();
449    while upper_phase > lower_phase + std::f32::consts::PI {
450        upper_phase -= 2.0 * std::f32::consts::PI;
451    }
452    while upper_phase < lower_phase - std::f32::consts::PI {
453        upper_phase += 2.0 * std::f32::consts::PI;
454    }
455    let phase = lower_phase + (upper_phase - lower_phase) * fraction;
456    Complex32::new(amp * phase.cos(), amp * phase.sin())
457}
