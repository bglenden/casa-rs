2//! The geodesy of CASA's `simutil`: its datum and ellipsoid tables and its
3//! conversions between UTM, geodetic and Earth-centred coordinates.
4//!
5//! | CASA (`simutil`) | casa-rs |
6//! |---|---|
7//! | `getdatum(datumcode)` | [`datum`], [`DATUMS`], [`ELLIPSOIDS`] |
8//! | `utm2long(east, north, zone, datum, nors)` | [`utm_to_geodetic`] |
9//! | `long2xyz(lon, lat, elevation, datum)` | [`geodetic_to_xyz`] |
10//! | `utm2xyz(easting, northing, elevation, zone, datum, nors)` | [`utm_to_xyz`] |
11//! | `xyz2long(x, y, z, datum)` | [`xyz_to_geodetic`] |
12//! | `xyz2long_old(x, y, z, datum)` | [`xyz_to_geodetic_old`] |
13//! | `tconpc(sf, orlim, er, esq, rf)` | [`zone_constants`] |
14//! | `tmgeod(n, e, eps, cm, fe, sf, so, r, v0, v2, v4, v6, fn, er, esq)` | [`grid_to_geodetic`] |
15//! | `locxyz2itrf(lat, longitude, alt, locx, locy, locz)` | [`local_to_itrf`] |
16//!
17//! The arithmetic follows CASA's Python operation by operation (including
18//! its `180/π` divisions and its `**` powers), so results agree with CASA's
19//! to rounding. As in CASA, the "ITRF" these functions produce is the
20//! nominal right-handed Earth-centred frame of the datum's ellipsoid shifted
21//! by the datum's offsets, not a realised ITRF.
22//!
23//! The UTM conversion is T. Vincenty's transverse Mercator inversion
24//! (subroutines `tconpc` and `tmgeod` of the NGS `UTMS` program, 1985), which
25//! R. Indebetouw translated from Fortran for CASA in 2009.
26
27/// A reference ellipsoid of CASA's `getdatum` table.
28#[derive(Debug, Clone, Copy, PartialEq)]
29pub struct Ellipsoid {
30    /// CASA's two-letter code, e.g. `"WE"` for WGS 84.
31    pub code: &'static str,
32    /// Equatorial radius (semi-major axis), metres.
33    pub equatorial_radius_m: f64,
34    /// Reciprocal of the flattening.
35    pub inverse_flattening: f64,
36    /// CASA's descriptive name.
37    pub name: &'static str,
38}

40/// A geodetic datum of CASA's `getdatum` table: an ellipsoid and the
41/// offsets that take the datum's Earth-centred coordinates to WGS 84's.
42///
43/// CASA's docstring: one takes local Earth-centred x, y, z coordinates and
44/// adds `dx`, `dy`, `dz` to get WGS 84 Earth-centred ones.
45#[derive(Debug, Clone, Copy, PartialEq)]
46pub struct Datum {
47    /// CASA's datum code, e.g. `"SAM56"` (the `datum=` header value).
48    pub code: &'static str,
49    /// Offset added to x, metres.
50    pub dx_m: f64,
51    /// Offset added to y, metres.
52    pub dy_m: f64,
53    /// Offset added to z, metres.
54    pub dz_m: f64,
55    /// The datum's ellipsoid.
56    pub ellipsoid: Ellipsoid,
57    /// CASA's descriptive name.
58    pub name: &'static str,
59}

205/// The zone constants of the transverse Mercator inversion that `tconpc`
206/// returns: `eps` (the second eccentricity squared), `r` (the radius of
207/// the rectifying sphere), `so` (the scaled meridional distance from the
208/// equator to the southernmost parallel `orlim`) and the foot-point series
209/// coefficients `v0`, `v2`, `v4`, `v6`.
210struct ZoneConstants {
211    eps: f64,
212    r: f64,
213    so: f64,
214    v: [f64; 4],
215}
216
217/// `tconpc`: the zone constants for scale factor `sf`, southernmost
218/// parallel `orlim`, equatorial radius `er`, eccentricity squared `esq` and
219/// reciprocal flattening `rf`.
220fn tconpc(sf: f64, orlim: f64, er: f64, esq: f64, rf: f64) -> ZoneConstants {
221    let f = 1.0 / rf;
222    let eps = esq / (1.0 - esq);
223    let pr = (1.0 - f) * er;
224    let en = (er - pr) / (er + pr);
225    let en2 = en * en;
226    let en3 = en * en * en;
227    let en4 = en2 * en2;
228
229    let c2 = -3.0 * en / 2.0 + 9.0 * en3 / 16.0;
230    let c4 = 15.0 * en2 / 16.0 - 15.0 * en4 / 32.0;
231    let c6 = -35.0 * en3 / 48.0;
232    let c8 = 315.0 * en4 / 512.0;
233    let u0 = 2.0 * (c2 - 2.0 * c4 + 3.0 * c6 - 4.0 * c8);
234    let u2 = 8.0 * (c4 - 4.0 * c6 + 10.0 * c8);
235    let u4 = 32.0 * (c6 - 6.0 * c8);
236    let u6 = 128.0 * c8;
237
238    let c2 = 3.0 * en / 2.0 - 27.0 * en3 / 32.0;
239    let c4 = 21.0 * en2 / 16.0 - 55.0 * en4 / 32.0;
240    let c6 = 151.0 * en3 / 96.0;
241    let c8 = 1097.0 * en4 / 512.0;
242    let v0 = 2.0 * (c2 - 2.0 * c4 + 3.0 * c6 - 4.0 * c8);
243    let v2 = 8.0 * (c4 - 4.0 * c6 + 10.0 * c8);
244    let v4 = 32.0 * (c6 - 6.0 * c8);
245    let v6 = 128.0 * c8;
246
247    let r = er * (1.0 - en) * (1.0 - en * en) * (1.0 + 2.25 * en * en + (225.0 / 64.0) * en4);
248    let cosor = orlim.cos();
249    let omo = orlim
250        + orlim.sin()
251            * cosor
252            * (u0 + u2 * cosor * cosor + u4 * cosor.powf(4.0) + u6 * cosor.powf(6.0));
253    let so = sf * r * omo;
254    ZoneConstants {
255        eps,
256        r,
257        so,
258        v: [v0, v2, v4, v6],
259    }
260}

262/// The projection of one UTM zone: what `tmgeod` takes besides the point.
263struct Projection {
264    cm: f64,
265    fe: f64,
266    fn_: f64,
267    sf: f64,
268    er: f64,
269    esq: f64,
270}
271
272/// `tmgeod`: grid coordinates to geodetic latitude and west-positive
273/// longitude, radians, with the convergence and the point scale factor.
274fn tmgeod(n: f64, e: f64, zone: &ZoneConstants, p: &Projection) -> [f64; 4] {
275    let [v0, v2, v4, v6] = zone.v;
276    let om = (n - p.fn_ + zone.so) / (zone.r * p.sf);
277    let cosom = om.cos();
278    let foot = om
279        + om.sin()
280            * cosom
281            * (v0 + v2 * cosom * cosom + v4 * cosom.powf(4.0) + v6 * cosom.powf(6.0));
282    let sinf = foot.sin();
283    let cosf = foot.cos();
284    let tn = sinf / cosf;
285    let ts = tn * tn;
286    let ets = zone.eps * cosf * cosf;
287    let rn = p.er * p.sf / (1.0 - p.esq * sinf * sinf).sqrt();
288    let q = (e - p.fe) / rn;
289    let qs = q * q;
290    let b2 = -tn * (1.0 + ets) / 2.0;
291    let b4 = -(5.0 + 3.0 * ts + ets * (1.0 - 9.0 * ts) - 4.0 * ets * ets) / 12.0;
292    let b6 = (61.0 + 45.0 * ts * (2.0 + ts) + ets * (46.0 - 252.0 * ts - 60.0 * ts * ts)) / 360.0;
293    let b1 = 1.0;
294    let b3 = -(1.0 + ts + ts + ets) / 6.0;
295    let b5 = (5.0 + ts * (28.0 + 24.0 * ts) + ets * (6.0 + 8.0 * ts)) / 120.0;
296    let b7 = -(61.0 + 662.0 * ts + 1320.0 * ts * ts + 720.0 * ts.powf(3.0)) / 5040.0;
297    let lat = foot + b2 * qs * (1.0 + qs * (b4 + b6 * qs));
298    let l = b1 * q * (1.0 + qs * (b3 + qs * (b5 + b7 * qs)));
299    let lon = -l / cosf + p.cm;
300    // The convergence and the point scale factor, at the point found.
301    let (sinfi, cosfi) = lat.sin_cos();
302    let l1 = (lon - p.cm) * cosfi;
303    let ls = l1 * l1;
304    let tn = sinfi / cosfi;
305    let ts = tn * tn;
306    let c3 = (1.0 + 3.0 * ets + 2.0 * ets.powf(2.0)) / 3.0;
307    let c5 = (2.0 - ts) / 15.0;
308    let conv = -tn * l1 * (1.0 + ls * (c3 + c5 * ls));
309    let f2 = (1.0 + ets) / 2.0;
310    let f4 = (5.0 - 4.0 * ts + ets * (9.0 - 24.0 * ts)) / 12.0;
311    let kp = p.sf * (1.0 + f2 * ls * (1.0 + f4 * ls));
312    [lat, lon, conv, kp]
313}

315/// CASA's `tconpc(sf, orlim, er, esq, rf)`: the transverse Mercator zone
316/// constants `[eps, r, so, v0, v2, v4, v6]` for scale factor `sf`,
317/// southernmost parallel `orlim` (radians), equatorial radius `er`,
318/// eccentricity squared `esq` and reciprocal flattening `rf`.
319#[must_use]
320pub fn zone_constants(sf: f64, orlim: f64, er: f64, esq: f64, rf: f64) -> [f64; 7] {
321    let zone = tconpc(sf, orlim, er, esq, rf);
322    let [v0, v2, v4, v6] = zone.v;
323    [zone.eps, zone.r, zone.so, v0, v2, v4, v6]
324}

326/// CASA's `tmgeod(n, e, eps, cm, fe, sf, so, r, v0, v2, v4, v6, fn, er,
327/// esq)`: northing `n` and easting `e` to `[lat, lon, conv, kp]`, the
328/// geodetic latitude and west-positive longitude (radians), the
329/// convergence and the point scale factor, with the zone constants of
330/// [`zone_constants`] and central meridian `cm`, false easting `fe`, false
331/// northing `fn`, scale factor `sf`, equatorial radius `er` and
332/// eccentricity squared `esq`.
333// CASA's signature: one value each.
334#[allow(clippy::too_many_arguments)]
335#[must_use]
336pub fn grid_to_geodetic(
337    n: f64,
338    e: f64,
339    eps: f64,
340    cm: f64,
341    fe: f64,
342    sf: f64,
343    so: f64,
344    r: f64,
345    v: [f64; 4],
346    fn_: f64,
347    er: f64,
348    esq: f64,
349) -> [f64; 4] {
350    let zone = ZoneConstants { eps, r, so, v };
351    let projection = Projection {
352        cm,
353        fe,
354        fn_,
355        sf,
356        er,
357        esq,
358    };
359    tmgeod(n, e, &zone, &projection)
360}

362/// Convert UTM coordinates to geodetic longitude and latitude (CASA's
363/// `utm2long`).
364///
365/// - `east`, `north`: easting and northing, metres (false easting 500 km;
366///   false northing 10 000 km when `south`).
367/// - `zone`: the UTM zone number. As in CASA it is not range-checked: zones
368///   below 30 take the central meridian `183 − 6·zone` degrees west, the
369///   others `543 − 6·zone` degrees west.
370/// - `datum`: only its ellipsoid is used here; [`geodetic_to_xyz`] applies
371///   the offsets.
372/// - `south`: the southern-hemisphere flag (CASA's `nors == 'S'`).
373///
374/// Returns `(longitude, latitude)` in radians, longitude east-positive. The
375/// longitude is not normalised: for zones 30 and above it is near −360°
376/// plus the east longitude, exactly as CASA returns it.
377///
378/// CASA scale factor 0.9996, southernmost parallel 0, and no adjacent-zone
379/// correction (CASA has that code commented out).
380#[must_use]
381pub fn utm_to_geodetic(east: f64, north: f64, zone: i64, datum: &Datum, south: bool) -> (f64, f64) {
382    let rad = 180.0 / std::f64::consts::PI;
383    let er = datum.ellipsoid.equatorial_radius_m;
384    let rf = datum.ellipsoid.inverse_flattening;
385    let f = 1.0 / rf;
386    let esq = 2.0 * f - f * f;
387    // CASA's longitudes in this code are west-positive.
388    let icm = if zone < 30 {
389        183_i128 - 6 * i128::from(zone)
390    } else {
391        543_i128 - 6 * i128::from(zone)
392    };
393    // Python's float(icm): exact for any real UTM zone.
394    let cm = icm as f64 / rad;
395    let fn_ = if south { 10_000_000.0 } else { 0.0 };
396    let fe = 500_000.0;
397    let sf = 0.9996;
398    let orlim = 0.0;
399    let constants = tconpc(sf, orlim, er, esq, rf);
400    let projection = Projection {
401        cm,
402        fe,
403        fn_,
404        sf,
405        er,
406        esq,
407    };
408    let [lat, lon, _, _] = tmgeod(north, east, &constants, &projection);
409    // CASA: "convert to more usual convention of negative lon = W".
410    (-lon, lat)
411}

413/// Convert geodetic longitude, latitude and elevation on a datum to
414/// Earth-centred x, y, z (CASA's `long2xyz`).
415///
416/// `longitude_rad` is east-positive, `elevation_m` is above the datum's
417/// ellipsoid. The datum's offsets are added to the result, so for a datum
418/// other than WGS 84 the coordinates are WGS 84-centred. CASA's comment
419/// quotes the NGS check point S23°12′00″ W67°48′00″, 0 m on GRS 80, at
420/// (2216194.4188, −5430618.6472, −2497092.5213) m.
421#[must_use]
422pub fn geodetic_to_xyz(
423    longitude_rad: f64,
424    latitude_rad: f64,
425    elevation_m: f64,
426    datum: &Datum,
427) -> [f64; 3] {
428    let er = datum.ellipsoid.equatorial_radius_m;
429    let f = 1.0 / datum.ellipsoid.inverse_flattening;
430    let esq = 2.0 * f - f.powf(2.0);
431    let nu = er / (1.0 - esq * latitude_rad.sin().powf(2.0)).sqrt();
432    let x = (nu + elevation_m) * latitude_rad.cos() * longitude_rad.cos() + datum.dx_m;
433    let y = (nu + elevation_m) * latitude_rad.cos() * longitude_rad.sin() + datum.dy_m;
434    let z = ((1.0 - esq) * nu + elevation_m) * latitude_rad.sin() + datum.dz_m;
435    [x, y, z]
436}

438/// Convert UTM easting, northing and elevation to Earth-centred x, y, z
439/// (CASA's `utm2xyz`): [`utm_to_geodetic`] then [`geodetic_to_xyz`].
440#[must_use]
441pub fn utm_to_xyz(
442    easting_m: f64,
443    northing_m: f64,
444    elevation_m: f64,
445    zone: i64,
446    datum: &Datum,
447    south: bool,
448) -> [f64; 3] {
449    let (lon, lat) = utm_to_geodetic(easting_m, northing_m, zone, datum, south);
450    geodetic_to_xyz(lon, lat, elevation_m, datum)
451}

453/// Convert Earth-centred x, y, z to geodetic longitude, latitude and height
454/// on a datum (CASA's `xyz2long`, a transcription of SOFA's `gc2gde`).
455///
456/// Returns `(longitude_rad, latitude_rad, height_m)`, or `None` where CASA
457/// returns `-1, -1, -1` for invalid ellipsoid parameters (never for the
458/// tabulated datums). The datum's offsets are subtracted first.
459///
460/// One correction to CASA: CASA computes the longitude as
461/// `atan2(y − dx, x − dx)`, subtracting the x offset from y. This function
462/// subtracts `dy` from y. The two agree for WGS 84, the only datum
463/// `readantenna` converts with.
464#[must_use]
465pub fn xyz_to_geodetic(x: f64, y: f64, z: f64, datum: &Datum) -> Option<(f64, f64, f64)> {
466    let er = datum.ellipsoid.equatorial_radius_m;
467    let f = 1.0 / datum.ellipsoid.inverse_flattening;
468    if !(0.0..1.0).contains(&f) || er <= 0.0 {
469        return None;
470    }
471    let e2 = (2.0 - f) * f;
472    let e4t = e2 * e2 * 1.5;
473    let ec2 = 1.0 - e2;
474    if ec2 <= 0.0 {
475        return None;
476    }
477    let ec = ec2.sqrt();
478    let (xr, yr, zr) = (x - datum.dx_m, y - datum.dy_m, z - datum.dz_m);
479    // Distance from the polar axis.
480    let r = (xr.powf(2.0) + yr.powf(2.0)).sqrt();
481    let lon = if r > 0.0 { yr.atan2(xr) } else { 0.0 };
482    // Newton correction factors.
483    let s0 = zr.abs() / er;
484    let c0 = ec * r / er;
485    let a0 = (c0.powf(2.0) + s0.powf(2.0)).sqrt();
486    let d0 = ec * s0 * a0.powf(3.0) + e2 * s0.powf(3.0);
487    let f0 = r / er * a0.powf(3.0) - e2 * c0.powf(3.0);
488    // Halley correction factor.
489    let b0 = e4t * s0.powf(2.0) * c0.powf(2.0) * r / er * (a0 - ec);
490    let s1 = d0 * f0 - b0 * s0;
491    let cc = ec * (f0 * f0 - b0 * c0);
492    let mut lat = s1.atan2(cc);
493    let height = (r * cc + zr.abs() * s1 - er * (ec2 * s1.powf(2.0) + cc.powf(2.0)).sqrt())
494        / (s1.powf(2.0) + cc.powf(2.0)).sqrt();
495    if zr < 0.0 {
496        lat = -lat;
497    }
498    Some((lon, lat, height))
499}

501/// Convert Earth-centred x, y, z to geodetic longitude and latitude on a
502/// datum by CASA's older iteration (`xyz2long_old`): three Newton steps on
503/// the eccentricity term, then `atan2` for each angle. Returns
504/// `(longitude_rad, latitude_rad)`.
505///
506/// One correction to CASA: CASA's second term subtracts the x offset from
507/// z (`(z − dx)²`) where the rest of the function subtracts `dz`; this
508/// subtracts `dz` (`docs/CASA (C++) bugs.md`). The two agree for WGS 84.
509#[must_use]
510pub fn xyz_to_geodetic_old(x: f64, y: f64, z: f64, datum: &Datum) -> (f64, f64) {
511    let er = datum.ellipsoid.equatorial_radius_m;
512    let f = 1.0 / datum.ellipsoid.inverse_flattening;
513    let (dx, dy, dz) = (datum.dx_m, datum.dy_m, datum.dz_m);
514    let b = ((x - dx).powf(2.0) + (y - dy).powf(2.0)) / er.powf(2.0);
515    let c = (z - dz).powf(2.0) / er.powf(2.0);
516    let esq = 2.0 * f - f.powf(2.0);
517    let a0 = c * (1.0 - esq);
518    let a1 = 2.0 * a0;
519    let efth = esq.powf(2.0);
520    let a2 = a0 + b - efth;
521    let a3 = -2.0 * efth;
522    let a4 = -efth;
523    let b0 = 4.0 * a0;
524    let b1 = 3.0 * a1;
525    let b2 = 2.0 * a2;
526    let b3 = a3;
527    let mut nlqk = esq;
528    for _ in 0..3 {
529        let nlqks = nlqk * nlqk;
530        let nlqkc = nlqk * nlqks;
531        let nlf = a0 * nlqks * nlqks + a1 * nlqkc + a2 * nlqks + a3 * nlqk + a4;
532        let nlfprm = b0 * nlqkc + b1 * nlqks + b2 * nlqk + b3;
533        nlqk -= nlf / nlfprm;
534    }
535    let y0 = (1.0 + nlqk) * (z - dz);
536    let x0 = ((x - dx).powf(2.0) + (y - dy).powf(2.0)).sqrt();
537    let lat = y0.atan2(x0);
538    let lon = (y - dy).atan2(x - dx);
539    (lon, lat)
540}

542/// Convert local tangent-plane offsets to Earth-centred x, y, z (CASA's
543/// `locxyz2itrf`, from Rob Reid).
544///
545/// - `latitude_deg`, `longitude_deg`: geodetic WGS 84 position of the
546///   reference point, degrees (CASA passes the observatory's position).
547/// - `altitude_m`: the reference point's height above the ellipsoid.
548/// - `local_m`: the offsets `[x, y, z]` from that point: x east, y north,
549///   z along the ellipsoid normal, metres.
550///
551/// CASA uses the WGS 84 semi-axes `a = 6378137.0` m and
552/// `b = 6356752.3142` m (the rounded polar radius), as here.
553#[must_use]
554pub fn local_to_itrf(
555    latitude_deg: f64,
556    longitude_deg: f64,
557    altitude_m: f64,
558    local_m: [f64; 3],
559) -> [f64; 3] {
560    let [locx, locy, locz] = local_m;
561    let phi = latitude_deg.to_radians();
562    let lmbda = longitude_deg.to_radians();
563    let sphi = phi.sin();
564    let a = 6_378_137.0_f64;
565    let b = 6_356_752.314_2_f64;
566    let ae = (b / a).acos();
567    let n = a / (1.0 - (ae.sin() * sphi).powf(2.0)).sqrt();
568    let nploczcphimlocysphi = (n + locz + altitude_m) * phi.cos() - locy * sphi;
569    let clmb = lmbda.cos();
570    let slmb = lmbda.sin();
571    let x = nploczcphimlocysphi * clmb - locx * slmb;
572    let y = nploczcphimlocysphi * slmb + locx * clmb;
573    let z = (n * (b / a).powf(2.0) + locz + altitude_m) * sphi + locy * phi.cos();
574    [x, y, z]
575}
