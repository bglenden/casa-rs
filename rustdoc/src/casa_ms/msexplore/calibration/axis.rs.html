39impl Axis {
40    pub(super) fn parse(value: &str) -> Result<Self, String> {
41        use Axis::*;
42        let normalized = value.to_ascii_lowercase().replace([' ', '-'], "");
43        let axis = match normalized.as_str() {
44            "chan" | "channel" => Channel,
45            "freq" | "frequency" => Frequency,
46            "time" => Time,
47            "interval" | "timeint" | "timeinterval" | "time_interval" => Interval,
48            "scan" => Scan,
49            "field" => Field,
50            "spw" => Window,
51            "row" => Row,
52            "antenna" | "antenna1" | "ant" | "ant1" => Antenna1,
53            "antenna2" | "ant2" => Antenna2,
54            "poln" | "polarization" | "corr" | "correlation" => Receptor,
55            "observation" => Observation,
56            "amp" | "amplitude" => Amplitude,
57            "phase" => Phase,
58            "real" => Real,
59            "imag" | "imaginary" => Imaginary,
60            "gainamp" | "gamp" => GainAmplitude,
61            "gainphase" | "gphase" => GainPhase,
62            "gainreal" | "greal" => GainReal,
63            "gainimag" | "gimag" => GainImaginary,
64            "snr" => SignalToNoise,
65            "delay" | "del" => Delay,
66            "rate" | "delayrate" => Rate,
67            "disp" | "dispdelay" => DispersiveDelay,
68            "tsys" => SystemTemperature,
69            "opacity" | "opac" => Opacity,
70            "swp" | "swpower" | "switchedpower" | "spgain" => SwitchedPower,
71            "tec" => Tec,
72            "antennaposition" | "antpos" => AntennaPosition,
73            _ => return Err(format!("unknown calibration plot axis {value:?}")),
74        };
75        Ok(axis)
76    }
77
78    pub(super) fn label(self) -> &'static str {
79        use Axis::*;
80        match self {
81            Channel => "Channel",

83            Time => "Time",
84            Interval => "Interval",
85            Scan => "Scan",
86            Field => "Field",

88            Row => "Row",

91            Receptor => "Poln",
92            Observation => "Observation",
93            Amplitude => "Amp",
94            Phase => "Phase",
95            Real => "Real",
96            Imaginary => "Imag",
97            GainAmplitude => "Gain Amp",
98            GainPhase => "Gain Phase",
99            GainReal => "Gain Real",
100            GainImaginary => "Gain Imag",
101            SignalToNoise => "SNR",
102            Delay => "Delay",
103            Rate => "Delay Rate",
104            DispersiveDelay => "Disp Delay",
105            SystemTemperature => "Tsys",

111    }
112
113    pub(super) fn unit(self) -> &'static str {
114        match self {
115            Self::Frequency => "GHz",
116            Self::Time => "MJD seconds",
117            Self::Interval => "s",
118            Self::Phase | Self::GainPhase => "deg",
119            Self::Delay => "ns",
120            Self::Rate => "ps/s",
121            Self::SystemTemperature => "K",
122            Self::AntennaPosition => "m",
123            _ => "",
124        }
125    }
126
127    pub(super) fn value(self, metadata: &SampleCoordinates, value: Complex64, snr: f64) -> f64 {
128        use Axis::*;
129        match self {
130            Channel => metadata.channel as f64,
131            Frequency => metadata.frequency_ghz,
132            Time => metadata.time,
133            Interval => metadata.interval,
134            Scan => f64::from(metadata.scan),
135            Field => f64::from(metadata.field),
136            Window => f64::from(metadata.window),
137            Row => metadata.row as f64,
138            Antenna1 => f64::from(metadata.antennas[0]),
139            Antenna2 => f64::from(metadata.antennas[1]),
140            Receptor => metadata.receptor as f64,
141            Observation => f64::from(metadata.observation),
142            Amplitude | GainAmplitude => value.norm(),
143            Phase | GainPhase => value.arg().to_degrees(),
144            Imaginary | GainImaginary => value.im,
145            SignalToNoise => snr,
146            Real | GainReal | Delay | Rate | DispersiveDelay | SystemTemperature | Opacity
147            | SwitchedPower | Tec | AntennaPosition => value.re,
148        }
149    }
150}
