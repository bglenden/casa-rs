2//! What `simutil` reads of an image's layout without its pixels: `is4d`,
3//! whether a sky model has CASA's four axes in order, and `spectral`, its
4//! channel count and width.
5
6use std::path::Path;
7
8use casa_coordinates::{CoordinateSystem, CoordinateType};
9use casa_images::AnyPagedImage;
10
11use super::{SkyModelError, pixel_axes};
12use crate::simutil::python_float;
13
14/// An image's shape and coordinate system: a CASA image (a directory) or a
15/// FITS file (a file), as `ia.open` opens either.
16fn layout(path: &Path) -> Result<(Vec<usize>, CoordinateSystem), SkyModelError> {
17    if path.is_file() {
18        // casa-images reads a FITS file whole; sky models are small.
19        let fits = casa_images::read_fits_image(path)?;
20        return Ok((fits.shape, fits.coordinates));
21    }
22    let image = AnyPagedImage::open(path)?;
23    let coordinates = match &image {
24        AnyPagedImage::Float32(image) => image.coordinates().clone(),
25        AnyPagedImage::Float64(image) => image.coordinates().clone(),
26        AnyPagedImage::Complex32(image) => image.coordinates().clone(),
27        AnyPagedImage::Complex64(image) => image.coordinates().clone(),
28    };
29    Ok((image.shape().to_vec(), coordinates))
30}
31
32/// `simutil.is4d(image)`: whether the image has four axes, direction on
33/// pixel axes 0 and 1, Stokes on 2 and spectral on 3.
34///
35/// # Errors
36///
37/// An image that cannot be opened.
38pub fn is_4d(path: &Path) -> Result<bool, SkyModelError> {
39    let (shape, coordinates) = layout(path)?;
40    if shape.len() != 4 {
41        return Ok(false);
42    }
43    let axes = |kind| pixel_axes(&coordinates, kind).map(|(_, axes)| axes);
44    Ok(
45        axes(CoordinateType::Direction).as_deref() == Some(&[0, 1][..])
46            && axes(CoordinateType::Stokes).as_deref() == Some(&[2][..])
47            && axes(CoordinateType::Spectral).as_deref() == Some(&[3][..]),
48    )
49}

51/// What `simutil.spectral(image)` gives: the spectral axis's length and its
52/// increment as CASA writes it, Python's `str` of the number then the unit
53/// (`2000000.0Hz`).
54#[derive(Debug, Clone, PartialEq)]
55pub struct SpectralSize {
56    /// The number of channels.
57    pub channels: usize,
58    /// The channel width, as CASA's string.
59    pub width: String,
60}
61
62/// `simutil.spectral(image)`: the image's spectral size, or `None` when it
63/// has no spectral coordinate (CASA's `0.0`).
64///
65/// # Errors
66///
67/// An image that cannot be opened.
68pub fn spectral_size(path: &Path) -> Result<Option<SpectralSize>, SkyModelError> {
69    let (shape, coordinates) = layout(path)?;
70    let Some((index, axes)) = pixel_axes(&coordinates, CoordinateType::Spectral) else {
71        return Ok(None);
72    };
73    let coordinate = coordinates.coordinate(index);
74    let increment = coordinate.increment()[0];
75    let unit = coordinate.axis_units()[0].clone();
76    Ok(Some(SpectralSize {
77        channels: shape[axes[0]],
78        width: format!("{}{unit}", python_float(increment)),
79    }))
