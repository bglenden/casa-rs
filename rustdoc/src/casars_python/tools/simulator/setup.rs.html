2//! The setters and `observe`: CASA's arguments converted as casatools'
3//! `simulator_cmpt.cc` converts them, then handed to the session.
4//!
5//! Each method converts its arguments only once a session is open, as CASA
6//! does (before `open`, even unreadable arguments give `False`), and in
7//! CASA's order, so the first error is CASA's. Python's values are read into
8//! Rust ones at the call; turning those into the session's types and the
9//! session's work run with the GIL released (`PySimulator::on_session`), and
10//! messages are logged once it is back.

25/// `sm.settimes(integrationtime, usehourangle, referencetime)`.
26///
27/// A reference time CASA cannot convert (`""`) is logged and taken as MJD 0,
28/// as in CASA; `[]` is MJD 0 too. The session converts a reference in
29/// another time scale to UTC.
30pub(super) fn settimes(
31    tool: &PySimulator,
32    py: Python<'_>,
33    integration: &Variant,
34    use_hour_angle: bool,
35    reference: &Variant,
36) -> PyResult<bool> {
37    let result = tool.on_session(py, |session, lines| {
38        let reference = match measures::epoch(reference)? {
39            Some(epoch) => epoch,
40            None => {
41                lines.push(LogLine::error(
42                    "Could not convert referencetime to an Epoch Measures",
43                ));
44                MEpoch::from_mjd(0.0, EpochRef::UTC)
45            }
46        };
47        let integration_s = quantity_value(integration, "s")?;
48        let result = session
49            .simulator
50            .set_times(integration_s, use_hour_angle, reference);
51        outcome(result, lines)
52    })?;
53    Ok(result.unwrap_or(false))
54}

78/// The session's [`StateSpec`] for `state`: SUB_SCAN must fit an `Int`.
79fn state_spec(state: &StateRequest<'_>) -> PyResult<StateSpec> {
80    let sub_scan = i32::try_from(state.sub_scan).map_err(|_| {
81        PyRuntimeError::new_err(format!(
82            "state_sub_scan {} does not fit a 32-bit integer",
83            state.sub_scan
84        ))
85    })?;
86    Ok(StateSpec {
87        sig: state.sig,
88        reference: state.reference,
89        cal: state.cal,
90        load: state.load,
91        sub_scan,
92        obs_mode: state.obs_mode.to_string(),
93    })
94}

96/// `sm.observe(...)`: plan a scan. The GIL is released while the session
97/// plans it.
98pub(super) fn observe(
99    tool: &PySimulator,
100    py: Python<'_>,
101    request: &ObserveRequest<'_>,
102) -> PyResult<bool> {
103    if !tool.is_open(py) {
104        return Ok(false);
105    }
106    let spec = ObserveSpec {
107        source: request.source.to_string(),
108        spectral_window: request.spectral_window.to_string(),
109        start_s: quantity_value(&request.start, "s")?,
110        stop_s: quantity_value(&request.stop, "s")?,
111        add_observation: request.add_observation,
112        state: state_spec(&request.state)?,
113        observer: request.observer.to_string(),
114        project: request.project.to_string(),
115    };
116    let planned = py.detach(|| {
117        tool.lock()
118            .as_mut()
119            .map(|session| session.simulator.observe(&spec))
120    });
121    let Some(result) = planned else {
122        return Ok(false);
123    };
124    let mut lines = Vec::new();
125    let result = outcome(result, &mut lines);
126    emit(py, LOGGER, &lines)?;
127    result
128}

143/// `sm.observemany(...)`, as casatools' binding converts it.
144/// - There is a pointing per start time, and the stop times must match them.
145/// - One source name, or one direction, is padded to every pointing as
146///   casacore's `Vector::resize(n, true)` pads it: with empty strings, not
147///   copies. So a later pointing's field is `""`, which the session refuses
148///   as CASA does ("Field 1:  not yet defined"), and its direction is none.
149///   CASA has added rows by then; casa-rs plans nothing
150///   (`docs/CASA (C++) bugs.md`).
151/// - Lists of any other length raise, as its `AlwaysAssert`s do.
152/// - An empty direction is none; an unconvertible one is logged and taken
153///   as the pole. `NewMSSimulator` reads the pole as none too.
154pub(super) fn observemany(
155    tool: &PySimulator,
156    py: Python<'_>,
157    request: &ObserveManyRequest<'_>,
158) -> PyResult<bool> {
159    if !tool.is_open(py) {
160        return Ok(false);
161    }
162    let count = request.starts.len();
163    let check = |values: &[String], what: &str| {
164        if values.len() == 1 || values.len() == count {
165            Ok(())
166        } else {
167            Err(PyRuntimeError::new_err(format!(
168                "Failed AlwaysAssert {what}.nelements() == nptg"
169            )))
170        }
171    };
172    if request.stops.len() != count {
173        return Err(PyRuntimeError::new_err(
174            "Failed AlwaysAssert sstoptimes.nelements() == nptg",
175        ));
176    }
177    check(&request.sources, "ssourcenames")?;
178    check(&request.directions, "sdirections")?;
179    // A single value is the first pointing's, and the rest are padded empty.
180    let padded = |values: &[String], index: usize| -> String {
181        match (values.len(), index) {
182            (1, 0) => values[0].clone(),
183            (1, _) => String::new(),
184            _ => values[index].clone(),
185        }
186    };
187    let state = state_spec(&request.state)?;
188    let result = tool.on_session(py, |session, lines| {
189        let mut pointings = Vec::with_capacity(count);
190        for index in 0..count {
191            let given = padded(&request.directions, index);
192            let direction = if given.is_empty() {
193                None
194            } else {
195                let given = Variant::Text(given);
196                match measures::direction(&given, &session.measures)? {
197                    Some(direction) => Some(direction).filter(|direction| !is_pole(direction)),
198                    None => {
199                        lines.push(LogLine::error(
200                            "Could not convert direction to a Direction Measure.",
201                        ));
202                        None
203                    }
204                }
205            };
206            pointings.push(Pointing {
207                source: padded(&request.sources, index),
208                start_s: quantity_value(&Variant::Text(request.starts[index].clone()), "s")?,
209                stop_s: quantity_value(&Variant::Text(request.stops[index].clone()), "s")?,
210                direction,
211            });
212        }
213        let spec = ObserveManySpec {
214            pointings,
215            spectral_window: request.spectral_window.to_string(),
216            add_observation: request.add_observation,
217            state: state.clone(),
218            observer: request.observer.to_string(),
219            project: request.project.to_string(),
220        };
221        outcome(session.simulator.observe_many(&spec), lines)
222    })?;
223    Ok(result.unwrap_or(false))
224}

226/// Whether `direction` is the pole casatools passes for an empty direction:
227/// `NewMSSimulator` compares the angles alone, whatever the frame.
228fn is_pole(direction: &MDirection) -> bool {
229    direction.as_angles() == (0.0, std::f64::consts::FRAC_PI_2)
230}

242/// `sm.setmosaicfield(...)`. A centre CASA cannot convert is logged and
243/// taken as the J2000 pole; a count below one adds no fields, as CASA's
244/// loops run none.
245pub(super) fn setmosaicfield(
246    tool: &PySimulator,
247    py: Python<'_>,
248    request: &MosaicRequest<'_>,
249) -> PyResult<bool> {
250    let result = tool.on_session(py, |session, lines| {
251        let centre = match measures::direction(&request.centre, &session.measures)? {
252            Some(direction) => direction,
253            None => {
254                lines.push(LogLine::error(
255                    "Could not convert field center to a Direction Measures",
256                ));
257                MDirection::from_angles(0.0, std::f64::consts::FRAC_PI_2, DirectionRef::J2000)
258            }
259        };
260        let distance_m = quantity_value(&request.distance, "m")?;
261        let spacing_rad = quantity_value(&request.spacing, "rad")?;
262        let count = |value: i64| usize::try_from(value).unwrap_or(0);
263        let counts = (count(request.counts.0), count(request.counts.1));
264        let result = session.simulator.set_mosaic_field(
265            request.name,
266            request.calcode,
267            &centre,
268            counts,
269            spacing_rad,
270            distance_m,
271        );
272        outcome(result, lines)
273    })?;
274    Ok(result.unwrap_or(false))
275}

316/// `sm.setconfig(...)`: add antennas.
317///
318/// Divergence: CASA writes zero positions and returns `True` for a
319/// `coordsystem` it does not know (or for `longlat`, which it does not
320/// implement); casa-rs logs CASA's message and returns `False` (an unknown
321/// one) or raises (`longlat`).
322pub(super) fn setconfig(
323    tool: &PySimulator,
324    py: Python<'_>,
325    request: ConfigRequest<'_>,
326) -> PyResult<bool> {
327    let result = tool.on_session(py, |session, lines| {
328        let reference = match measures::position(&request.reference_location, &session.measures)? {
329            Some(position) => position,
330            None => {
331                if request.reference_location != Variant::Empty {
332                    lines.push(LogLine::error(format!(
333                        "Could not convert referencelocation {} to a Position Measures",
334                        request.reference_location.describe()
335                    )));
336                }
337                MPosition::new_itrf(0.0, 0.0, 0.0)
338            }
339        };
340        let coordinate_system = match request.coordinate_system {
341            "global" => CoordinateSystem::Global,
342            "local" => CoordinateSystem::Local { reference },
343            "longlat" => CoordinateSystem::LongLat,
344            other => {
345                lines.push(LogLine::error(format!(
346                    "Unknown coordinate system type: {other}"
347                )));
348                return Ok(false);
349            }
350        };
351        let config = ArrayConfig {
352            telescope: request.telescope.to_string(),
353            x: request.x,
354            y: request.y,
355            z: request.z,
356            dish_diameter: request.dish_diameter,
357            offset: request.offset,
358            mount: request.mount,
359            antenna_names: request.antenna_names,
360            pad_names: request.pad_names,
361            coordinate_system,
362        };
363        outcome(session.simulator.set_config(&config), lines)
364    })?;
365    match result {
366        Some(set) => Ok(set),
367        None => {
368            emit(
369                py,
370                LOGGER,
371                &[LogLine::error("The sm tool must be opened first.")],
372            )?;
373            Ok(false)
374        }
375    }
376}

426    pub(super) name: &'a str,
427    pub(super) frequency: Variant,
428    pub(super) width: Variant,
429    pub(super) resolution: Variant,
430    pub(super) frame: &'a str,
431    pub(super) channels: i64,
432    pub(super) stokes: &'a str,
433}

435/// `sm.setspwindow(...)`. Divergence: a negative `nchannels`, which CASA
436/// passes on to array allocation, raises.
437pub(super) fn setspwindow(
438    tool: &PySimulator,
439    py: Python<'_>,
440    request: &WindowRequest<'_>,
441) -> PyResult<bool> {
442    let result = tool.on_session(py, |session, lines| {
443        let start_frequency_hz = quantity_value(&request.frequency, "Hz")?;
444        let channel_width_hz = quantity_value(&request.width, "Hz")?;
445        let resolution_hz = quantity_value(&request.resolution, "Hz")?;
446        let frame = measures::frequency_reference(request.frame)?;
447        let channel_count = usize::try_from(request.channels).map_err(|_| {
448            ConvertError::runtime(format!(
449                "nchannels must not be negative, got {}",
450                request.channels
451            ))
452        })?;
453        let spec = SpectralWindowSpec {
454            name: request.name.to_string(),
455            start_frequency_hz,
456            channel_width_hz,
457            resolution_hz,
458            frame,
459            channel_count,
460            stokes: request.stokes.to_string(),
461        };
462        outcome(session.simulator.set_spectral_window(&spec), lines)
463    })?;
464    Ok(result.unwrap_or(false))
