2//! A column description as `tb.addcols` takes it, a port of casacore's
3//! `TableProxy::makeTableDesc` and `addArrayColumnDesc`, checked against what
4//! CASA 6.7.6.14 did with each case of `ADDCOLS_CASES` in
5//! `tools/casa-compat/table_tool_oracle.py`.
6//!
7//! The description is a record of column name to a record of `valueType`
8//! (required), `comment`, `option`, `maxlen`, `dataManagerType`,
9//! `dataManagerGroup`, `ndim`, `shape`, `_c_order` and `keywords`. The fields
10//! are read in casacore's order and the first that fails ends the call:
11//!
12//! * a column has `ndim` when it is an array, whatever the value's type
13//!   (`asInt` then takes only an integer);
14//! * a `shape` makes the array of fixed shape (`FixedShape`, 4, is added to
15//!   `option`), in Fortran order unless `_c_order` reverses it; it must agree
16//!   with `ndim` and have only positive extents; an `ndim` of 0 or less
17//!   leaves the dimension unspecified, and a shape is then refused;
18//! * `Direct` (1) and `Undefined` (2) are `option`'s other flags that the
19//!   schema keeps;
20//! * `_keywords_`, `_private_keywords_` and `_define_dminfo_` are not columns
21//!   and are skipped; `_define_hypercolumn_` is not implemented.
22//!
23//! A field of the wrong type raises casacore's own message, whose type names
24//! are padded to eight characters.
25
26use casa_tables::{ColumnOptions, ColumnSchema};
27use casa_types::{PrimitiveType, RecordValue, Value};
28
29use super::keyword_value::Incoming;
30use super::state::Fail;
31
32const DIRECT: i64 = 1;
33const UNDEFINED: i64 = 2;
34
35/// A column to add.
36pub(super) struct NewColumn {
37    pub(super) schema: ColumnSchema,
38    pub(super) keywords: Option<RecordValue>,
39    /// Whether the description names a storage manager for the column.
40    pub(super) names_manager: bool,
41}
42
43/// casacore's data type for a `valueType`, lower-cased: `Some(None)` is a
44/// record column, which only a scalar column can be.
45fn value_type(name: &str) -> Option<Option<PrimitiveType>> {
46    Some(Some(match name {
47        "boolean" | "bool" => PrimitiveType::Bool,
48        "byte" | "uchar" => PrimitiveType::UInt8,
49        "short" => PrimitiveType::Int16,
50        "ushort" => PrimitiveType::UInt16,
51        "integer" | "int" => PrimitiveType::Int32,
52        "uint" => PrimitiveType::UInt32,
53        "int64" => PrimitiveType::Int64,
54        "float" => PrimitiveType::Float32,
55        "double" => PrimitiveType::Float64,
56        "complex" => PrimitiveType::Complex32,
57        "dcomplex" => PrimitiveType::Complex64,
58        "string" => PrimitiveType::String,
59        "record" => return Some(None),
60        _ => return None,
61    }))
62}
63
64fn failure(message: String) -> Fail {
65    Fail::Runtime(message)
66}
67
68/// What `RecordRep::get_pointer` says of a field of the wrong type.
69fn wrong_type(expected: &str, field: &str, found: &str) -> Fail {
70    failure(format!(
71        "RecordRep::get_pointer - incorrect data type {expected} used for field {field} with \
72         type {found}"
73    ))
74}

76/// The fields of one column's description, read as casacore reads them.
77struct Fields<'a>(&'a [(String, Incoming)]);
78
79impl Fields<'_> {
80    fn get(&self, key: &str) -> Option<&Incoming> {
81        self.0
82            .iter()
83            .find(|(field, _)| field == key)
84            .map(|(_, value)| value)
85    }
86
87    /// `Record::asInt`: an integer, and nothing else.
88    fn int(&self, key: &str) -> Result<Option<i64>, Fail> {
89        match self.get(key) {
90            None => Ok(None),
91            Some(value) => match value.scalar_integer() {
92                Some(integer) => Ok(Some(integer)),
93                None => Err(failure("RecordInterface::asInt - invalid data type".into())),
94            },
95        }
96    }
97
98    /// `Record::asString`.
99    fn string(&self, key: &str) -> Result<Option<String>, Fail> {
100        match self.get(key) {
101            None => Ok(None),
102            Some(value) => match value.text() {
103                Some(text) => Ok(Some(text.to_string())),
104                None => Err(wrong_type("String  ", key, value.type_label())),
105            },
106        }
107    }
108
109    /// `Record::asBool`.
110    fn flag(&self, key: &str) -> Result<bool, Fail> {
111        match self.get(key) {
112            None => Ok(false),
113            Some(value) => value
114                .scalar_flag()
115                .ok_or_else(|| wrong_type("Bool    ", key, value.type_label())),
116        }
117    }
118
119    /// `Record::toArrayInt64`: an integer or integers (a real is refused).
120    fn integers(&self, key: &str) -> Result<Vec<i64>, Fail> {
121        match self.get(key) {
122            None => Ok(Vec::new()),
123            Some(value) => value
124                .integer_elements()
125                .ok_or_else(|| wrong_type("unknown ", key, "unknown ")),
126        }
127    }
128}

130/// The column `name` that `description` describes; `None` for one of the
131/// fields of a description that is not a column.
132pub(super) fn parse(name: &str, description: &Incoming) -> Result<Option<NewColumn>, Fail> {
133    let Some(fields) = description.fields() else {
134        return Err(wrong_type("Record  ", name, description.type_label()));
135    };
136    let fields = Fields(fields);
137    match name {
138        "_keywords_" | "_private_keywords_" | "_define_dminfo_" => return Ok(None),
139        "_define_hypercolumn_" => {
140            return Err(Fail::Unsupported(
141                "table.addcols with a hypercolumn definition is not yet implemented in casa-rs \
142                 (#1028)"
143                    .to_string(),
144            ));
145        }
146        _ => {}
147    }
148    if fields.get("valueType").is_none() {
149        return Err(failure(format!(
150            "addColumns failed: No value type for column {name}"
151        )));
152    }
153    let kind = fields
154        .string("valueType")?
155        .unwrap_or_default()
156        .to_ascii_lowercase();
157    let option = fields.int("option")?.unwrap_or(0);
158    let max_length = fields.int("maxlen")?.unwrap_or(0);
159    let comment = fields.string("comment")?;
160    let manager_type = fields.string("dataManagerType")?;
161    let manager_group = fields.string("dataManagerGroup")?;
162    let ndim = fields.int("ndim")?;
163    let (shape, c_order) = match ndim {
164        Some(_) => (fields.integers("shape")?, fields.flag("_c_order")?),
165        None => (Vec::new(), false),
166    };
167
168    let data_type = value_type(&kind);
169    let schema = match (ndim, data_type) {
170        (Some(_), Some(None)) | (Some(_), None) => {
171            return Err(failure(format!(
172                "addColumns failed: Unknown data type {kind} for array column {name}"
173            )));
174        }
175        (Some(ndim), Some(Some(ty))) => {
176            if ndim <= 0 && !shape.is_empty() {
177                return Err(failure(
178                    "addColumns failed: arrayColumnDesc: shape should not be given when ndim <= 0"
179                        .to_string(),
180                ));
181            }
182            if ndim > 0 && !shape.is_empty() && ndim as usize != shape.len() {
183                return Err(failure(
184                    "addColumns failed: arrayColumnDesc: ndim and shape mismatch".to_string(),
185                ));
186            }
187            if shape.iter().any(|&extent| extent <= 0) {
188                return Err(failure(
189                    "addColumns failed: arrayColumnDesc: shape < 0".to_string(),
190                ));
191            }
192            if shape.is_empty() {
193                ColumnSchema::array_variable(
194                    name,
195                    ty,
196                    usize::try_from(ndim).ok().filter(|&n| n > 0),
197                )
198            } else {
199                let mut extents: Vec<usize> = shape.iter().map(|&extent| extent as usize).collect();
200                if c_order {
201                    extents.reverse();
202                }
203                ColumnSchema::array_fixed(name, ty, extents)
204            }
205        }
206        (None, Some(Some(ty))) => ColumnSchema::scalar(name, ty),
207        (None, Some(None)) => ColumnSchema::record(name),
208        (None, None) => {
209            return Err(failure(format!(
210                "addColumns failed: Unknown data type {kind} for scalar column {name}"
211            )));
212        }
213    };
214    let schema = schema
215        .with_options(ColumnOptions {
216            direct: option & DIRECT != 0,
217            undefined: option & UNDEFINED != 0,
218        })
219        .map_err(|error| failure(format!("addColumns failed: {error}")))?;
220    let schema = match comment {
221        Some(comment) => schema.with_comment(comment),
222        None => schema,
223    };
224    let schema = match u32::try_from(max_length).ok().filter(|&length| length > 0) {
225        Some(length) => schema.with_max_length(length),
226        None => schema,
227    };
228    let keywords = match fields.get("keywords") {
229        None => None,
230        Some(keywords) if keywords.fields().is_some() => match keywords.clone().natural()? {
231            Value::Record(record) => Some(record),
232            _ => None,
233        },
234        Some(other) => return Err(wrong_type("Record  ", "keywords", other.type_label())),
235    };
236    let names_manager = [manager_type, manager_group]
237        .iter()
238        .any(|text| text.as_deref().is_some_and(|text| !text.is_empty()));
239    Ok(Some(NewColumn {
240        schema,
241        keywords,
242        names_manager,
243    }))
