2//! `simutil`'s analysis helpers, which CASA's `simanalyze` and `simalma` use
3//! on a simulated image: `ismstp`, `cellsize`, `statim` and `convimage`,
4//! over [`casa_simulation::simutil::analysis`], which records where its
5//! results depart from CASA's.
6//!
7//! Each returns what CASA 6.7.6.14's returns:
8//! - `ismstp`: a `bool`;
9//! - `cellsize`: a list of two quantity records, in the image's direction
10//!   units (radians);
11//! - `statim`: `(min, max, rms, unit)`, the three values NumPy 0-d `float64`
12//!   arrays, as CASA's `np.squeeze` leaves them;
13//! - `convimage`: `None`, having written `<modelflat>.regrid` and
14//!   `<modelflat>.regrid.conv`.
15//!
16//! `statim`'s plot is not drawn yet (#1025): with `plot=True`, CASA's
17//! default, it says so and returns the statistics. `convimage` with a
18//! `complist` is refused (#1053).
19

35/// A refusal CASA raises through `simutil.msg(..., priority='error')` is a
36/// `RuntimeError` with its message; a file that cannot be read an `OSError`.
37fn analysis_raised(error: AnalysisError) -> PyErr {
38    match error {
39        AnalysisError::Io { .. } => PyOSError::new_err(error.to_string()),
40        other => PyRuntimeError::new_err(other.to_string()),
41    }
42}
43
44#[pymethods]
45impl PySimutil {
46    /// Whether `s` is a total-power MeasurementSet: its first antenna's name
47    /// holds `TP`, it has one antenna, or every UVW value is zero. With
48    /// `halt`, a MeasurementSet that is not, or cannot be read, raises with
49    /// CASA's message; otherwise either is `False`.
50    #[pyo3(signature = (s, halt = false), text_signature = "($self, s, halt=False)")]
51    fn ismstp(&self, py: Python<'_>, s: PathBuf, halt: bool) -> PyResult<bool> {
52        let shown = s.display().to_string();
53        match py.detach(|| is_total_power(&s)) {
54            Ok(true) => Ok(true),
55            Ok(false) if halt => Err(PyRuntimeError::new_err(format!(
56                "input file '{shown}' is not a totalpower ms"
57            ))),
58            Err(_) if halt => Err(PyRuntimeError::new_err(format!(
59                "can't understand the file '{shown}'"
60            ))),
61            _ => Ok(false),
62        }
63    }
64
65    /// The image's cell on the sky: `[cellx, celly]`, quantity records in
66    /// radians, positive. An image rotated with respect to latitude and
67    /// longitude raises CASA's message.
68    #[pyo3(signature = (image), text_signature = "($self, image)")]
69    fn cellsize(&self, py: Python<'_>, image: PathBuf) -> PyResult<Py<PyAny>> {
70        let [x, y] = py
71            .detach(|| {
72                let opened = PagedImage::<f32>::open(&image)?;
73                cell_size(opened.coordinates())
74            })
75            .map_err(analysis_raised)?;
76        CasaValue::List(vec![
77            CasaValue::quantity(x, "rad"),
78            CasaValue::quantity(y, "rad"),
79        ])
80        .into_python(py)
81    }
82
83    /// The image's minimum, maximum, the rms of its lower-left quarter and
84    /// its brightness unit, as CASA's `statim` gives them. `plot`, CASA's
85    /// default, draws nothing yet (#1025); `incell`, `disprange`, `bar` and
86    /// `showstats` only shape that plot.
87    #[pyo3(
88        signature = (image, plot = true, incell = None, disprange = None, bar = true, showstats = true),
89        text_signature = "($self, image, plot=True, incell=None, disprange=None, bar=True, showstats=True)"
90    )]
91    // CASA's signature: the plot's controls are parameters of their own.
92    #[allow(unused_variables, clippy::too_many_arguments)]
93    fn statim<'py>(
94        &self,
95        py: Python<'py>,
96        image: PathBuf,
97        plot: bool,
98        incell: Option<&Bound<'py, PyAny>>,
99        disprange: Option<&Bound<'py, PyAny>>,
100        bar: bool,
101        showstats: bool,
102    ) -> PyResult<Bound<'py, PyTuple>> {
103        if plot {
104            emit(
105                py,
106                LOGGER,
107                &[LogLine::warning(
108                    "statim's plot is not yet implemented in casa-rs (#1025): no figure is drawn",
109                )],
110            )?;
111        }
112        let stats = py.detach(|| statim(&image)).map_err(analysis_raised)?;
113        let zero_d = |value: f64| ndarray::arr0(value).into_pyarray(py).into_any();
114        PyTuple::new(
115            py,
116            [
117                zero_d(stats.min),
118                zero_d(stats.max),
119                zero_d(stats.rms),
120                stats.unit.into_pyobject(py)?.into_any(),
121            ],
122        )
123    }
124
125    /// Regrid the flat sky model `modelflat` onto the flat image `outflat`
126    /// and convolve it with the image's restoring beam: `<modelflat>.regrid`
127    /// and `<modelflat>.regrid.conv`, as CASA writes them. A `complist` is
128    /// refused (#1053).
129    #[pyo3(signature = (modelflat, outflat, complist = ""), text_signature = "($self, modelflat, outflat, complist='')")]
130    fn convimage(
131        &self,
132        py: Python<'_>,
133        modelflat: PathBuf,
134        outflat: PathBuf,
135        complist: &str,
136    ) -> PyResult<()> {
137        if !complist.is_empty() {
138            return Err(PyNotImplementedError::new_err(
139                "simutil.convimage with a complist is not yet implemented in casa-rs (#1053)",
140            ));
141        }
142        py.detach(|| convimage(&modelflat, &outflat))
143            .map(|_| ())
144            .map_err(analysis_raised)
145    }
146}
