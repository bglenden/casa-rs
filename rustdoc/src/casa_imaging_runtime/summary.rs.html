23/// A run's optional diagnostic output (plan section 8.1). Each switch adds
24/// `tracing` events, at `info`, under its own target; none changes what the
25/// run computes, and with every switch off a run emits none of them. The
26/// provider-contracts catalog owns the defaults.
27#[derive(Clone, Copy, Debug, Deserialize, PartialEq, Eq)]
28pub struct Diagnostics {
29    /// One event per phase ([`STAGE_TIMING`]): the [`Phase`] the summary
30    /// records.
31    pub stage_timing: bool,
32    /// One event per Metal backend and wave
33    /// (`casa_imaging::metal_stage_timing`), for each gridding owner and
34    /// each row chunk's prediction backend that dispatched: whether its work
35    /// completed, its dispatches, samples (and how many it predicted) and
36    /// sub-blocks, the host's preparation time and its waits for the device,
37    /// and the device's time on its commands; and one event per wave for the
38    /// pass's source stream: the producer's fill time and the pass's waits
39    /// for filled blocks. A CPU run emits none.
40    pub metal_stage_timing: bool,
41    /// The controller's science quantities ([`SCIENCE_TRACE`]): the
42    /// residual statistics each cleaning decision reads, each minor cycle's
43    /// controls and what it cleaned.
44    pub science_trace: bool,
45}

47/// What one run did, O(1) per phase, written with how it ended
48/// ([`RunSummary::write`]).
49///
50/// Every step that can fail runs inside a phase, so for each outcome
51/// ([`RunStatus`]) the summary holds:
52/// - **completed:** every phase, the totals, the stop reason and the
53///   products;
54/// - **failed:** the phases that ran, the last not completed, with the
55///   minor-cycle totals and stop reason as far as the run got, any products
56///   published before the failure, and the error; a run that fails in
57///   preparation (request validation, compile, availability, admitting its
58///   source) records only its preparation phase, and one that fails in
59///   setup (its operators, choosing and forming its team) its setup phase
60///   too;
61/// - **cancelled:** the phases that ran before the cancellation took
62///   effect, the totals so far, and no products, since nothing is
63///   published; a run cancelled before it starts records no phase.
64///
65/// Memory is reported as the process sees it: the resident high-water mark
66/// and the reservations of every run in the process.
67#[derive(Debug, Serialize)]
68pub struct RunSummary {
69    /// The request as its caller states it.
70    pub request: serde_json::Value,
71    /// Why cleaning stopped, with CASA's `stopcode`; `None` for a dirty run,
72    /// a run that stopped before deciding, or one whose last minor cycle
73    /// cleaned nothing without a stop.
74    #[serde(serialize_with = "clean_stop")]
75    pub stop: Option<CleanStop>,
76    /// Bytes the resource policy lets the process's reservations hold.
77    pub capacity_bytes: u64,
78    /// Workers in the run's team; zero before the team forms.
79    pub workers: usize,
80    /// Workers the resource policy allows: more than [`Self::workers`]
81    /// when the run's memory did not fit them (#737); zero before the team
82    /// forms.
83    pub workers_policy: usize,
84    /// Why the run has fewer workers than its policy allows, with what the
85    /// policy's needed and what was free; `None` when it has as many.
86    pub workers_reason: Option<String>,
87    /// Lanes the products ran on: the most, up to [`Self::workers`], whose
88    /// products' memory fits (#763 review); zero before the products.
89    pub product_lanes: usize,
90    /// Why the products have fewer lanes than [`Self::workers`], with what
91    /// the workers' lanes needed and what was free; `None` when they have
92    /// as many.
93    pub product_lanes_reason: Option<String>,
94    /// Where the passes grid.
95    pub backend: BackendChoice,
96    /// The grid arithmetic the run resolved (`gridprecision`); `None`
97    /// before the operators form.
98    #[serde(serialize_with = "grid_precision")]
99    pub precision: Option<GridPrecision>,
100    /// Every phase in the order it ran.
101    pub phases: Vec<Phase>,
102    /// Minor cycles completed.
103    pub minor_cycles: usize,
104    /// Minor-cycle iterations charged to the iteration budget.
105    pub minor_iterations: usize,
106    /// The product images published, by path, in the order they were.
107    pub products: Vec<String>,
108    /// This run's view of the process's reservations, closed at each phase
109    /// end.
110    #[serde(skip)]
111    reservations: ReservationWindow,
112}

114/// How a run ended, written beside its [`RunSummary`].
115#[derive(Clone, Debug, PartialEq, Eq, Serialize)]
116#[serde(tag = "status", rename_all = "lowercase")]
117pub enum RunStatus {
118    /// Every phase completed and the products are published.
119    Completed,
120    /// A phase failed. Publication replaces the product images one by one,
121    /// so a failure while publishing leaves the images published before it
122    /// in place, and the summary lists them; a failure before publishing
123    /// publishes nothing.
124    Failed {
125        /// Why.
126        error: String,
127    },
128    /// Cancellation took effect before publishing; nothing was published.
129    Cancelled,
130}
131
132/// One phase of a run.
133#[derive(Clone, Debug, Serialize)]
134pub struct Phase {
135    /// What the phase did.
136    pub name: String,
137    /// Whether it completed; a phase that failed is recorded too.
138    pub completed: bool,
139    /// Wall-clock seconds.
140    pub seconds: f64,
141    /// The process's resident high-water mark sampled when the phase
142    /// ended, in bytes: it covers every earlier phase, and anything else
143    /// the process ran, not this phase alone.
144    pub process_peak_rss_bytes: u64,
145    /// The most the process's memory reservations held at once from the end
146    /// of the run's previous phase to the end of this one, in bytes: what
147    /// admission had granted at its peak, this run's and any other's in the
148    /// process.
149    pub peak_reserved_bytes: u64,
150    /// The process's reservations still held when the phase ended, in
151    /// bytes.
152    pub reserved_bytes: u64,
153}

169impl RunSummary {
170    /// The summary of a run about to start: `request` echoed, nothing run
171    /// yet. Its first phase's reservation peak starts here.
172    #[must_use]
173    pub fn begin(request: serde_json::Value, capacity_bytes: u64, backend: BackendChoice) -> Self {
174        Self {
175            request,
176            stop: None,
177            capacity_bytes,
178            workers: 0,
179            workers_policy: 0,
180            workers_reason: None,
181            product_lanes: 0,
182            product_lanes_reason: None,
183            backend,
184            precision: None,
185            phases: Vec::new(),
186            minor_cycles: 0,
187            minor_iterations: 0,
188            products: Vec::new(),
189            reservations: ReservationWindow::open(),
190        }
191    }
192
193    /// Write the summary of a run that ended as `status` to `path`, as
194    /// pretty-printed JSON with the status's fields first.
195    ///
196    /// # Errors
197    ///
198    /// The file could not be written.
199    pub fn write(&self, status: &RunStatus, path: &Path) -> std::io::Result<()> {
200        let text = serde_json::to_string_pretty(&Written {
201            status,
202            summary: self,
203        })
204        .map_err(std::io::Error::other)?;
205        std::fs::write(path, text + "\n")
206    }
207}

209/// A summary as written: how the run ended, then what it did.
210#[derive(Serialize)]
211struct Written<'a> {
212    #[serde(flatten)]
213    status: &'a RunStatus,
214    #[serde(flatten)]
215    summary: &'a RunSummary,
216}

218/// Run one phase named `name`, unless `cancel` is set, and record in
219/// `summary` whether it completed, its wall time and the memory after it;
220/// with [`Diagnostics::stage_timing`], log that record too.
221///
222/// # Errors
223///
224/// [`Cancelled`] (through `E`) when cancellation was requested before the
225/// phase started, and nothing is recorded; otherwise the step's error.
226pub fn run_phase<T, E: From<Cancelled>>(
227    name: impl Into<String>,
228    cancel: &Cancel,
229    diagnostics: Diagnostics,
230    summary: &mut RunSummary,
231    step: impl FnOnce() -> Result<T, E>,
232) -> Result<T, E> {
233    if cancel.is_cancelled() {
234        return Err(Cancelled.into());
235    }
236    let started = Instant::now();
237    let result = step();
238    let phase = Phase {
239        name: name.into(),
240        completed: result.is_ok(),
241        seconds: started.elapsed().as_secs_f64(),
242        process_peak_rss_bytes: peak_rss(),
243        peak_reserved_bytes: summary.reservations.close(),
244        reserved_bytes: reserved_memory(),
245    };
246    if diagnostics.stage_timing {
247        tracing::info!(
248            target: STAGE_TIMING,
249            phase = ?phase.name,
250            completed = phase.completed,
251            seconds = phase.seconds,
252            process_peak_rss_bytes = phase.process_peak_rss_bytes,
253            peak_reserved_bytes = phase.peak_reserved_bytes,
254            reserved_bytes = phase.reserved_bytes,
255            "imaging phase"
256        );
257    }
258    summary.phases.push(phase);
259    result
260}

262/// CASA's stop reason: its `stopcode` and the reason's name.
263fn clean_stop<S: Serializer>(stop: &Option<CleanStop>, serializer: S) -> Result<S::Ok, S::Error> {
264    #[derive(Serialize)]
265    struct Stop {
266        reason: &'static str,
267        stopcode: i32,
268    }
269    stop.map(|stop| Stop {
270        reason: stop.name(),
271        stopcode: stop.code(),
272    })
273    .serialize(serializer)
274}

276/// The catalog's spelling of a grid precision.
277fn grid_precision<S: Serializer>(
278    precision: &Option<GridPrecision>,
279    serializer: S,
280) -> Result<S::Ok, S::Error> {
281    precision
282        .map(|precision| match precision {
283            GridPrecision::F32 => "f32",
284            GridPrecision::F64 => "f64",
285        })
286        .serialize(serializer)
287}

289/// The process's peak resident set, in bytes.
290fn peak_rss() -> u64 {
291    let mut usage = std::mem::MaybeUninit::<libc::rusage>::uninit();
292    // SAFETY: `getrusage` fills the whole struct when it returns 0.
293    let status = unsafe { libc::getrusage(libc::RUSAGE_SELF, usage.as_mut_ptr()) };
294    assert_eq!(
295        status, 0,
296        "getrusage(RUSAGE_SELF) fails only on a bad pointer"
297    );
298    // SAFETY: initialised by the successful call above.
299    let maximum = u64::try_from(unsafe { usage.assume_init() }.ru_maxrss).expect("a resident size");
300    // macOS reports bytes, Linux kibibytes.
301    if cfg!(target_os = "macos") {
302        maximum
303    } else {
304        maximum.saturating_mul(1024)
305    }
306}

312    const QUIET: Diagnostics = Diagnostics {
313        stage_timing: false,

318    #[test]
319    fn a_phase_is_recorded_whether_it_completes_or_fails_but_not_once_cancelled() {
320        let cancel = Cancel::new();
321        let mut summary = RunSummary::begin(serde_json::Value::Null, 1, BackendChoice::Cpu);
322        let value = run_phase("first", &cancel, QUIET, &mut summary, || {
323            Ok::<_, Cancelled>(7)
324        })
325        .expect("runs");
326        assert_eq!(value, 7);
327        let failed = run_phase("second", &cancel, QUIET, &mut summary, || {
328            Err::<(), _>(Cancelled)
329        });
330        assert_eq!(failed, Err(Cancelled));
331        let recorded = summary
332            .phases
333            .iter()
334            .map(|phase| (phase.name.as_str(), phase.completed))
335            .collect::<Vec<_>>();
336        assert_eq!(recorded, [("first", true), ("second", false)]);
337        assert!(
338            summary
339                .phases
340                .iter()
341                .all(|phase| phase.process_peak_rss_bytes > 0)
342        );
343        cancel.cancel();
344        let mut ran = false;
345        let cancelled = run_phase("third", &cancel, QUIET, &mut summary, || {
346            ran = true;
347            Ok::<_, Cancelled>(())
348        });
349        assert_eq!(cancelled, Err(Cancelled));
350        assert!(!ran);
351        assert_eq!(summary.phases.len(), 2);
352    }

354    #[test]
355    fn another_runs_phases_take_nothing_from_this_runs_phase() {
356        let host = crate::HostResources {
357            threads: 1,
358            performance_cores: 1,
359            available_memory: 1 << 30,
360            metal: false,
361        };
362        let policy = crate::ResourcePolicy::Explicit {
363            workers: 1,
364            memory: 1 << 30,
365        };
366        let demand = crate::Demand {
367            phase: "transient",
368            memory: 5 << 20,
369        };
370        let cancel = Cancel::new();
371        let mut first = RunSummary::begin(serde_json::Value::Null, 1, BackendChoice::Cpu);
372        run_phase("first", &cancel, QUIET, &mut first, || {
373            drop(crate::admit(&host, &policy, &demand).expect("fits"));
374            let mut second = RunSummary::begin(serde_json::Value::Null, 1, BackendChoice::Cpu);
375            run_phase("second", &cancel, QUIET, &mut second, || {
376                Ok::<_, Cancelled>(())
377            })
378        })
379        .expect("runs");
380        // Other tests of this process may hold reservations too.
381        assert!(first.phases[0].peak_reserved_bytes >= 5 << 20);
382    }

384    #[test]
385    fn the_summary_spells_its_status_stop_and_precision_as_the_catalog_and_casa_do() {
386        let mut summary = RunSummary::begin(serde_json::Value::Null, 1 << 30, BackendChoice::Cpu);
387        summary.stop = Some(CleanStop::Threshold);
388        summary.precision = Some(GridPrecision::F64);
389        let written = |status: &RunStatus| -> serde_json::Value {
390            serde_json::to_value(Written {
391                status,
392                summary: &summary,
393            })
394            .expect("serializes")
395        };
396        let json = written(&RunStatus::Failed {
397            error: "a pass failed".to_string(),
398        });
399        assert_eq!(json["status"], "failed");
400        assert_eq!(json["error"], "a pass failed");
401        assert_eq!(
402            json["stop"],
403            serde_json::json!({"reason": "threshold", "stopcode": 2})
404        );
405        assert_eq!(json["precision"], "f64");
406        assert_eq!(json["backend"], "cpu");
407        assert_eq!(json["capacity_bytes"], 1_u64 << 30);
408        assert!(json.get("reservations").is_none());
409        let json = written(&RunStatus::Cancelled);
410        assert_eq!(json["status"], "cancelled");
411        assert!(json.get("error").is_none());
412        assert_eq!(written(&RunStatus::Completed)["status"], "completed");
413    }
