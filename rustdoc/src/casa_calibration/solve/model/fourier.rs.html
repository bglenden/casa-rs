70impl FourierModel {
71    pub(super) fn new(record: RecordValue) -> Self {
72        let first_term = match record.get("subftm_0").or_else(|| record.get("subftm")) {
73            Some(Value::Record(term)) => term,
74            _ => &record,
75        };
76        let mosaic = mosaic::is_mosaic(first_term);
77        let refocus = record_number(first_term, "distance").is_some_and(|distance| distance != 0.0);
78        Self {
79            record,
80            mosaic,
81            refocus,
82            prepared: Mutex::default(),
83        }
84    }
85
86    pub(super) fn needs_refocus(&self) -> bool {
87        self.refocus
88    }
89
90    pub(super) fn predict(
91        &self,
92        row: FourierRow<'_>,
93        output: &mut ArrayD<Complex32>,
94        merge: FourierMerge,
95    ) -> Result<(), GainSolveError> {
96        let mut prepared = self
97            .prepared
98            .lock()
99            .expect("solve model is single consumer");
100        let FourierCache { grids, layouts } = &mut *prepared;
101        let layouts = layouts
102            .entry(self.mosaic.then_some(row.row.data_spw_id))
103            .or_default();
104        if !layouts.contains_key(row.correlations) {
105            let model = PreparedFourierModel::load(&self.record, &row, grids.clone())?;
106            *grids = Some(Arc::clone(&model.grids));
107            layouts.insert(row.correlations.to_vec(), model);
108        }
109        layouts
110            .get_mut(row.correlations)
111            .expect("correlation layout was prepared")
112            .predict(row, output, merge)
113    }
114}

144impl PreparedFourierModel {
145    fn load(
146        record: &RecordValue,
147        row: &FourierRow<'_>,
148        cached_grids: Option<Arc<PreparedModelGrids>>,
149    ) -> Result<Self, GainSolveError> {
150        let (terms, continuum_basis) = grid_terms(record)?;
151        let images = terms
152            .iter()
153            .map(|term| open_grid(term, row.ms))
154            .collect::<Result<Vec<_>, _>>()?;
155        let image = &images[0];
156        let [nx, ny, npol, channels] = *image.shape() else {
157            return Err(model_error(
158                "virtual GridFT requires a four-axis complex image",
159            ));
160        };
161        let coords = image.coordinates();
162        let direction = coords
163            .find_coordinate(CoordinateType::Direction)
164            .and_then(|index| match coords.coordinate(index) {
165                CoordinateModel::Direction(direction) => Some(direction),
166                _ => None,
167            })
168            .ok_or_else(|| model_error("virtual grid image has no direction coordinate"))?;
169        let increments = direction.increment();
170        let centre = direction
171            .to_world(&[(nx / 2) as f64, (ny / 2) as f64])
172            .map_err(model_error)?;
173        let phase_centre = MDirection::from_angles(centre[0], centre[1], direction.direction_ref());
174        let (image_shape, padding) = grid_input_geometry(terms[0], [nx, ny])?;
175        let geometry = GridGeometry::new(
176            ImageExtent {
177                shape: image_shape,
178                increment_rad: [increments[0], increments[1]],
179                reference_pixel: image_shape.map(|length| length / 2),
180            },
181            padding,
182        )
183        .map_err(model_error)?;
184        let polarization = grid_polarization(image, row.correlations)?;
185        if polarization.grid_pols() != npol {
186            return Err(model_error(
187                "virtual grid polarization axis does not match its planes",
188            ));
189        }
190        let axis = image_spectral_axis(image, channels)?;
191        let (basis, resampler) = spectral_layout(record, &axis, continuum_basis)?;
192        let tracking = tracking::Tracking::load(terms[0])?;
193        let (kernels, mosaic) = if mosaic::is_mosaic(terms[0]) {
194            let (kernels, layout) = MosaicLayout::load(
195                terms[0],
196                row,
197                direction,
198                &geometry,
199                &polarization,
200                axis.first_hz(),
201                matches!(
202                    record.get("freqframevalid"),
203                    Some(Value::Scalar(ScalarValue::Bool(true)))
204                ),
205            )?;
206            (kernels, Some(layout))
207        } else {
208            (grid_kernel(terms[0], &geometry, &polarization)?, None)
209        };
210        let operator =
211            MeasurementOperator::new(geometry, basis, polarization, kernels, GridPrecision::F32);
212        let grids = match cached_grids {
213            Some(grids) => grids,
214            None => Arc::new(load_grids(&operator, &images)?),
215        };
216        let scale = if record_text(terms[0], "name")? == "SetJyGridFT" {
217            setjy::FluxScale::from_record(terms[0])?
218        } else {
219            None
220        };
221        let project = uses_projected_tangent(terms[0], row.ms)?;
222        let legacy_flags = terms[0].get("doconversion").is_some();
223        let channel_mask = match terms[0].get("spwchanselflag") {
224            Some(Value::Array(ArrayValue::Int32(mask))) if !mask.is_empty() => Some(mask.clone()),
225            _ => None,
226        };
227        Ok(Self {
228            operator,
229            resampler,
230            grids,
231            phase_centre,
232            tracking,
233            phase_centre_time: record_number(terms[0], "phasecentertime")
234                .filter(|epoch| epoch.is_finite() && *epoch > 0.),
235            timed_fields: BTreeMap::new(),
236            project,
237            refocus: focus::Refocus::from_distance(
238                record_number(terms[0], "distance").unwrap_or(0.0),
239            )?,
240            mosaic,
241            scale,
242            legacy_taylor: legacy_taylor_term(record)?,
243            legacy_flags,
244            live_interpolation: record.get("casars_live_image_frequency").is_some(),
245            channel_mask,
246            continuum_axis: (channels == 1).then_some(axis),
247            convert_frequencies: matches!(
248                record.get("freqframevalid"),
249                Some(Value::Scalar(ScalarValue::Bool(true)))
250            ),
251            backend: CpuBackend::default(),
252            scratch: PredictionScratch::default(),
253            data: Vec::new(),
254            output: Vec::new(),
255            covered: Vec::new(),
256            weights: Vec::new(),
257            flags: Vec::new(),
258            frequencies: Vec::new(),
259        })
260    }

262    fn predict(
263        &mut self,
264        row: FourierRow<'_>,
265        output: &mut ArrayD<Complex32>,
266        merge: FourierMerge,
267    ) -> Result<(), GainSolveError> {
268        let shift = self
269            .tracking
270            .as_mut()
271            .map(|tracking| tracking.shift(&row))
272            .transpose()?
273            .unwrap_or([0.; 2]);
274        let tracked_centre = self
275            .tracking
276            .as_ref()
277            .map(|tracking| tracking.centre(shift));
278        let phase_centre = tracked_centre.as_ref().unwrap_or(&self.phase_centre);
279        // VisBuffer::phaseCenter(time) uses the current buffer's field
280        // and evaluates that direction at the supplied epoch.
281        let field: &MDirection = if let Some(epoch) = self.phase_centre_time {
282            match self.timed_fields.entry(row.row.field_id) {
283                std::collections::btree_map::Entry::Occupied(field) => field.into_mut(),
284                std::collections::btree_map::Entry::Vacant(field) => {
285                    let field_id = usize::try_from(*field.key()).map_err(model_error)?;
286                    field.insert(
287                        super::field_direction(row.ms, row.engine, field_id, epoch)
288                            .map_err(model_error)?,
289                    )
290                }
291            }
292        } else {
293            row.field
294        };
295        let (projection, pointing_offset_rad, antenna_types) = match &self.mosaic {
296            Some(layout) => (
297                GridFtProjection::tracked_mosaic_with_frame(
298                    field,
299                    &self.phase_centre,
300                    shift,
301                    row.frame,
302                ),
303                layout.pointing_offset(&row, shift)?,
304                layout.antenna_types(&row)?,
305            ),
306            None if self.project => (
307                GridFtProjection::faceted_between_directions_with_frame(
308                    field,
309                    phase_centre,
310                    row.frame,
311                ),
312                [0.0; 2],
313                [0; 2],
314            ),
315            None => (
316                GridFtProjection::between_directions_with_frame(field, phase_centre, row.frame),
317                [0.0; 2],
318                [0; 2],
319            ),
320        };
321        let projection = projection.map_err(model_error)?;
322        let (uvw_m, phase_shift_m) = match &mut self.refocus {
323            Some(focus) => focus.apply(&row, &projection)?,
324            None => projection.apply(row.uvw),
325        };
326        if row.raw_flags.shape() != [row.correlations.len(), row.frequencies.len()] {
327            return Err(model_error(
328                "virtual Fourier prediction flags do not match the visibility layout",
329            ));
330        }
331        let cells = row.frequencies.len() * row.correlations.len();
332        self.data.resize(cells, Complex32::new(0.0, 0.0));
333        self.output.resize(cells, Complex32::new(0.0, 0.0));
334        self.weights.resize(cells, 1.0);
335        self.flags.resize(cells, false);
336        let lsrk_ratio = if self.convert_frequencies || self.scale.is_some() {
337            lsrk_frequency_ratio(&row)?
338        } else {
339            1.0
340        };
341        let ratio = if self.convert_frequencies {
342            lsrk_ratio
343        } else {
344            1.0
345        };
346        self.frequencies.clear();
347        self.frequencies
348            .extend(row.frequencies.iter().map(|frequency| frequency * ratio));
349        let interpolation_channels = row
350            .scratch_channels
351            .map_or(row.frequencies.len(), |[_, _, count, _]| count as usize);
352        let use_spectral_selection_mask = !self.live_interpolation || interpolation_channels <= 1;
353        for (channel, ((flags, raw), &frequency)) in self
354            .flags
355            .chunks_exact_mut(row.correlations.len())
356            .zip(row.raw_flags.axis_iter(Axis(1)))
357            .zip(&self.frequencies)
358            .enumerate()
359        {
360            let outside = self
361                .continuum_axis
362                .as_ref()
363                .is_some_and(|axis| axis.nearest_channel(frequency).is_none());
364            if self.legacy_flags {
365                // Legacy GridFT restores VisBuffer's per-correlation flags.
366                // FTMachine::fromRecord discards the original channel mask.
367                let masked = self
368                    .channel_mask
369                    .as_ref()
370                    .zip(row.scratch_channels)
371                    .is_some_and(|(mask, [spw, start, count, stride])| {
372                        let Some(relative) = channel.checked_sub(start as usize) else {
373                            return true;
374                        };
375                        let compact = relative / stride as usize;
376                        compact >= count as usize
377                            || !relative.is_multiple_of(stride as usize)
378                            || (use_spectral_selection_mask
379                                && mask
380                                    .get([0, spw as usize, compact].as_slice())
381                                    .is_some_and(|&value| value != 1))
382                    });
383                for (value, &raw_flag) in flags.iter_mut().zip(raw) {
384                    *value = outside || masked || raw_flag;
385                }
386            } else {
387                flags.fill(outside || raw.iter().any(|flag| *flag));
388            }
389        }
390        let native = NativeRow {
391            uvw_m,
392            phase_shift_m,
393            pointing_offset_rad,
394            frequencies_hz: &self.frequencies,
395            values: &self.data,
396            weights: &self.weights,
397            flags: &self.flags,
398            row_flag: row.row.flag_row,
399            context: RowContext {
400                native_frequency_ratio: 1.0 / ratio,
401                original_uvw_m: None,
402                time_s: row.row.time_seconds,
403                antennas: [row.row.antenna1 as u32, row.row.antenna2 as u32],
404                antenna_types,
405                parallactic_angle_rad: [0.0, 0.0],
406                field: row.row.field_id as u32,
407                spectral_window: row.row.data_spw_id as u32,
408            },
409        };
410        let is_scratch = self.legacy_flags && !matches!(merge, FourierMerge::VirtualSum);
411        let coverage = is_scratch.then(|| {
412            self.covered.resize(cells, false);
413            &mut self.covered[..]
414        });
415        self.resampler
416            .set_legacy_prediction_selection(row.scratch_channels.map(
417                |[_, start, count, stride]| [start as usize, count as usize, stride as usize],
418            ));
419        self.resampler
420            .predict_row_with_coverage(
421                &self.operator,
422                &mut self.backend,
423                &self.grids,
424                &native,
425                &mut self.scratch,
426                (&mut self.output, coverage),
427            )
428            .map_err(model_error)?;
429        self.accumulate(row, output, lsrk_ratio, is_scratch.then_some(merge));
430        Ok(())
431    }
432
433    fn accumulate(
434        &self,
435        row: FourierRow<'_>,
436        output: &mut ArrayD<Complex32>,
437        lsrk_ratio: f64,
438        scratch: Option<FourierMerge>,
439    ) {
440        let correlations = row.correlations.len();
441        let destination = output
442            .as_slice_memory_order_mut()
443            .expect("contiguous model output");
444        if scratch.is_none() && self.scale.is_none() && self.legacy_taylor.is_none() {
445            for (value, &model) in destination.iter_mut().zip(&self.output) {
446                *value += model;
447            }
448            return;
449        }
450        for (channel, &frequency) in row.frequencies.iter().enumerate() {
451            let factor = self
452                .scale
453                .as_ref()
454                .map_or(1.0, |scale| scale.at(frequency * lsrk_ratio));
455            let taylor = self.legacy_taylor.map_or(1.0, |(term, reference)| {
456                // Legacy MultiTermFT rounds the raw frequency to Float.
457                let offset = ((f64::from(frequency as f32) - reference) / reference) as f32;
458                offset.powi(term)
459            });
460            for correlation in 0..correlations {
461                let index = channel * correlations + correlation;
462                let assigned = scratch.is_none() || self.covered[index];
463                if matches!(scratch, Some(FourierMerge::ScratchReplaceComponents)) {
464                    if assigned {
465                        destination[index] = self.output[index] * (factor * taylor);
466                    }
467                } else {
468                    // An additive SkyEquation pass adds the original model
469                    // to GridFT's output, including samples it did not visit.
470                    let sample = if assigned {
471                        self.output[index]
472                    } else {
473                        destination[index]
474                    };
475                    destination[index] += sample * (factor * taylor);
476                }
477            }
478        }
479    }
480}

482fn lsrk_frequency_ratio(row: &FourierRow<'_>) -> Result<f64, GainSolveError> {
483    if row.frequency_ref == FrequencyRef::LSRK {
484        return Ok(1.0);
485    }
486    let reference = row.frequencies[0];
487    let converted = convert_frequency_to_frame(
488        row.frequency_ref,
489        FrequencyRef::LSRK,
490        reference,
491        row.row.time_seconds,
492        row.row.field_id as usize,
493        row.engine,
494    )
495    .map_err(model_error)?;
496    Ok(converted / reference)
497}

499fn uses_projected_tangent(
500    record: &RecordValue,
501    ms: &MeasurementSet,
502) -> Result<bool, GainSolveError> {
503    if !matches!(
504        record.get("tangentspecified"),
505        Some(Value::Scalar(ScalarValue::Bool(true)))
506    ) {
507        return Ok(false);
508    }
509    let telescope = ms
510        .observation()
511        .map_err(model_error)?
512        .string(0, "TELESCOPE_NAME")
513        .map_err(model_error)?;
514    // FTMachine::rotateUVW uses unprojected rotation for these telescopes.
515    Ok(!telescope.contains("ATCA") && !telescope.contains("WSRT"))
516}

518fn spectral_layout(
519    record: &RecordValue,
520    axis: &SpectralAxis,
521    continuum_basis: Basis,
522) -> Result<(Basis, SpectralResampler), GainSolveError> {
523    let channels = axis.channels();
524    if channels == 1 {
525        let resampler = SpectralResampler::direct(continuum_basis).map_err(model_error)?;
526        return Ok((continuum_basis, resampler));
527    }
528    if !matches!(continuum_basis, Basis::Constant) || channels < 2 {
529        return Err(model_error(
530            "invalid channel axis for virtual Fourier model",
531        ));
532    }
533    // FTMachine::fromRecord sets nearest neighbour when reopening a
534    // prediction grid, even if the original tclean used another method.
535    let kernel = match (record.get("tovis"), record.get("freqinterpmethod")) {
536        (Some(Value::Scalar(ScalarValue::Bool(true))), _)
537        | (_, Some(Value::Scalar(ScalarValue::Int32(0)))) => SpectralKernel::Nearest,
538        (_, Some(Value::Scalar(ScalarValue::Int32(1)))) => SpectralKernel::Linear,
539        (_, Some(Value::Scalar(ScalarValue::Int32(2)))) => SpectralKernel::Cubic,
540        _ => {
541            return Err(model_error(
542                "unsupported virtual grid spectral interpolation",
543            ));
544        }
545    };
546    let basis = Basis::ChannelLocal { planes: channels };
547    let resampler = if let Some(Value::Array(ArrayValue::Float64(frequencies))) =
548        record.get("casars_live_image_frequency")
549    {
550        let frequencies = frequencies.iter().copied().collect::<Vec<_>>();
551        if frequencies.len() != channels as usize {
552            return Err(model_error(
553                "legacy imager interpolation axis length does not match its model",
554            ));
555        }
556        let increment = frequencies[1] - frequencies[0];
557        let grid = SpectralAxis::tabular(&frequencies, increment).map_err(model_error)?;
558        SpectralResampler::channel_local(grid, kernel).with_legacy_prediction_axis(axis.clone())
559    } else {
560        SpectralResampler::channel_local(axis.clone(), kernel)
561    };
562    Ok((basis, resampler))
563}

565fn image_spectral_axis(
566    image: &PagedImage<Complex32>,
567    channels: usize,
568) -> Result<SpectralAxis, GainSolveError> {
569    let coords = image.coordinates();
570    let spectral = coords
571        .find_coordinate(CoordinateType::Spectral)
572        .and_then(|index| match coords.coordinate(index) {
573            CoordinateModel::Spectral(spectral) => Some(spectral),
574            _ => None,
575        })
576        .ok_or_else(|| model_error("virtual grid image has no spectral coordinate"))?;
577    let spectral = crate::ft::spectral_with_runtime(spectral).map_err(model_error)?;
578    let frequency = |channel| {
579        spectral
580            .world_frequency_hz(channel as f64)
581            .map_err(model_error)
582    };
583    let start = frequency(0)?;
584    let width = frequency(1)? - start;
585    if channels == 1 {
586        return SpectralAxis::new(start, width, 1).map_err(model_error);
587    }
588    let frequencies = (0..channels)
589        .map(frequency)
590        .collect::<Result<Vec<_>, _>>()?;
591    SpectralAxis::tabular(&frequencies, width).map_err(model_error)
592}

594fn grid_terms(record: &RecordValue) -> Result<(Vec<&RecordValue>, Basis), GainSolveError> {
595    match record_text(record, "name")? {
596        "GridFT" | "WProjectFT" | "MosaicFT" | "MosaicFTNew" | "SetJyGridFT" => {
597            Ok((vec![record], Basis::Constant))
598        }
599        "MultiTermFT" => match record.get("subftm") {
600            Some(Value::Record(term)) => Ok((vec![term], Basis::Constant)),
601            _ => Err(model_error("virtual MultiTermFT has no subftm record")),
602        },
603        "MultiTermFTNew" => {
604            let Some(Value::Scalar(ScalarValue::UInt32(terms))) = record.get("nterms") else {
605                return Err(model_error(
606                    "virtual MultiTermFTNew has no Taylor term count",
607                ));
608            };
609            let Some(Value::Scalar(ScalarValue::Float64(reference_hz))) = record.get("reffreq")
610            else {
611                return Err(model_error(
612                    "virtual MultiTermFTNew has no reference frequency",
613                ));
614            };
615            if *terms == 0 || *terms as usize > record.fields().len() || *reference_hz <= 0.0 {
616                return Err(model_error("invalid virtual MultiTermFTNew Taylor basis"));
617            }
618            let subrecords = (0..*terms)
619                .map(|index| {
620                    let name = format!("subftm_{index}");
621                    match record.get(&name) {
622                        Some(Value::Record(term)) => Ok(term),
623                        _ => Err(model_error(format!("virtual model has no {name} record"))),
624                    }
625                })
626                .collect::<Result<Vec<_>, _>>()?;
627            Ok((
628                subrecords,
629                Basis::Taylor {
630                    terms: *terms,
631                    reference_hz: *reference_hz,
632                },
633            ))
634        }
635        name => Err(model_error(format!(
636            "unsupported virtual Fourier machine {name}"
637        ))),
638    }
639}

641/// Legacy ft stores each Taylor grid in a separate model entry. Its machine
642/// contains one term and scales that term after degridding.
643fn legacy_taylor_term(record: &RecordValue) -> Result<Option<(i32, f64)>, GainSolveError> {
644    if record_text(record, "name")? != "MultiTermFT" {
645        return Ok(None);
646    }
647    let Some(Value::Scalar(ScalarValue::Int32(term))) = record.get("thisterm") else {
648        return Err(model_error("virtual MultiTermFT has no Taylor term index"));
649    };
650    let Some(reference) = record_number(record, "reffreq") else {
651        return Err(model_error(
652            "virtual MultiTermFT has no reference frequency",
653        ));
654    };
655    if *term < 0 || !reference.is_finite() || reference <= 0.0 {
656        return Err(model_error("invalid virtual MultiTermFT Taylor term"));
657    }
658    Ok((*term > 0).then_some((*term, reference)))
659}

661fn open_grid(
662    record: &RecordValue,
663    ms: &MeasurementSet,
664) -> Result<PagedImage<Complex32>, GainSolveError> {
665    if !matches!(
666        record_text(record, "name")?,
667        "GridFT" | "WProjectFT" | "MosaicFT" | "MosaicFTNew" | "SetJyGridFT"
668    ) {
669        return Err(model_error(
670            "unsupported Fourier image type in virtual model",
671        ));
672    }
673    let recorded = Path::new(record_text(record, "diskimage")?);
674    let path = match (recorded.exists(), ms.path(), recorded.file_name()) {
675        (false, Some(ms), Some(name)) => [ms.join("SOURCE").join(name), ms.join(name)]
676            .into_iter()
677            .find(|path| path.exists())
678            .unwrap_or_else(|| recorded.to_path_buf()),
679        _ => PathBuf::from(recorded),
680    };
681    PagedImage::open(path).map_err(model_error)
682}

691fn grid_input_geometry(
692    record: &RecordValue,
693    stored: [usize; 2],
694) -> Result<([usize; 2], GridPadding), GainSolveError> {
695    let Some(Value::Array(ArrayValue::Float64(geometry))) =
696        record.get("casars_prediction_geometry")
697    else {
698        return Ok((stored, grid_padding(record)?));
699    };
700    let Some([nx, ny, padding]) = geometry.as_slice() else {
701        return Err(model_error("invalid imager prediction geometry"));
702    };
703    Ok((
704        [*nx as usize, *ny as usize],
705        GridPadding::CasaComposite(PaddingFactor::new(*padding as f32).map_err(model_error)?),
706    ))
707}

709fn grid_padding(record: &RecordValue) -> Result<GridPadding, GainSolveError> {
710    if record_text(record, "name")? != "WProjectFT" {
711        return Ok(GridPadding::None);
712    }
713    let Some(padding) = record_number(record, "padding") else {
714        return Err(model_error("virtual WProjectFT has no padding factor"));
715    };
716    Ok(GridPadding::CasaComposite(
717        PaddingFactor::new(padding as f32).map_err(model_error)?,
718    ))
719}

721fn grid_kernel(
722    record: &RecordValue,
723    geometry: &GridGeometry,
724    polarization: &PolarizationRouting,
725) -> Result<Box<dyn ConvolutionFunctionSet>, GainSolveError> {
726    if record_text(record, "name")? == "WProjectFT" {
727        let Some(Value::Scalar(ScalarValue::Int32(planes))) = record.get("nwplanes") else {
728            return Err(model_error("virtual WProjectFT has no plane count"));
729        };
730        let planes = u32::try_from(*planes).map_err(model_error)?;
731        let max_w = record_number(record, "maxw")
732            .filter(|max_w| max_w.is_finite())
733            .ok_or_else(|| model_error("virtual WProjectFT has no valid maximum W"))?;
734        let saved_scale = record_number(record, "savedwscale")
735            .filter(|scale| scale.is_finite() && *scale > 0.0 && planes > 1);
736        let count = if let Some(scale) = saved_scale {
737            WPlaneCount::Restored {
738                planes,
739                max_w: (f64::from(planes - 1).powi(2) / scale) / 1.05,
740            }
741        } else if max_w > 0.0 {
742            WPlaneCount::Restored { planes, max_w }
743        } else {
744            WPlaneCount::Fixed(planes)
745        };
746        return Ok(Box::new(
747            WPlanes::new(geometry, polarization, count).map_err(model_error)?,
748        ));
749    }
750    match record_text(record, "convtype")? {
751        "SF" => Ok(Box::new(Spheroidal::new(geometry, polarization))),
752        "BOX" => Ok(Box::new(Boxcar::new(geometry, polarization))),
753        name => Err(model_error(format!(
754            "unsupported virtual grid convolution {name}"
755        ))),
756    }
757}

759fn record_number(record: &RecordValue, field: &str) -> Option<f64> {
760    match record.get(field) {
761        Some(Value::Scalar(ScalarValue::Float64(value))) => Some(*value),
762        Some(Value::Scalar(ScalarValue::Float32(value))) => Some(f64::from(*value)),
763        _ => None,
764    }
765}

767fn grid_polarization(
768    image: &PagedImage<Complex32>,
769    correlations: &[i32],
770) -> Result<PolarizationRouting, GainSolveError> {
771    let coords = image.coordinates();
772    let stokes = coords
773        .find_coordinate(CoordinateType::Stokes)
774        .and_then(|index| match coords.coordinate(index) {
775            CoordinateModel::Stokes(stokes) => Some(stokes.stokes()),
776            _ => None,
777        })
778        .ok_or_else(|| model_error("virtual grid image has no Stokes coordinate"))?;
779    let coordinates = stokes
780        .iter()
781        .copied()
782        .map(|stokes_type| {
783            crate::ft::image::polarization_coordinate(stokes_type)
784                .ok_or_else(|| model_error("unsupported polarization plane in virtual grid"))
785        })
786        .collect::<Result<Vec<_>, _>>()?;
787    let correlations = correlations
788        .iter()
789        .map(|&code| {
790            Correlation::from_casacore_code(code)
791                .ok_or_else(|| model_error(format!("invalid virtual grid correlation {code}")))
792        })
793        .collect::<Result<Vec<_>, _>>()?;
794    PolarizationRouting::from_saved_planes(&correlations, &coordinates).map_err(model_error)
795}

797fn load_grids(
798    operator: &MeasurementOperator,
799    images: &[PagedImage<Complex32>],
800) -> Result<PreparedModelGrids, GainSolveError> {
801    let channels = operator.basis().planes();
802    let [nx, ny] = operator.geometry().grid_shape();
803    let shape = [
804        nx,
805        ny,
806        operator.polarization().grid_pols(),
807        channels as usize,
808    ];
809    if images.iter().any(|image| image.shape() != shape) {
810        return Err(model_error(
811            "virtual model image axes do not match the grid",
812        ));
813    }
814    let mut grids = operator
815        .empty_model_grids(PlaneRange::new(0, channels))
816        .map_err(model_error)?;
817    for mut plane in grids.planes_mut() {
818        for (term, image) in images.iter().enumerate() {
819            for polarization in 0..shape[2] {
820                let values = image
821                    .get_slice(
822                        &[0, 0, polarization, plane.plane() as usize],
823                        &[nx, ny, 1, 1],
824                    )
825                    .map_err(model_error)?;
826                let cells = values
827                    .as_slice_memory_order()
828                    .filter(|_| values.strides()[0] == 1)
829                    .ok_or_else(|| model_error("virtual grid cells are not contiguous along x"))?;
830                plane
831                    .load_f32_block(polarization, term, cells)
832                    .map_err(model_error)?;
833            }
834        }
835    }
836    Ok(grids)
837}
