21struct Request<'a> {
22    outline: &'a str,
23    outcont: &'a str,
24    region: RegionArgument,
25    channels: IntList,
26    pol: &'a str,
27    fitorder: i64,
28    overwrite: bool,
29}
30
31impl Request<'_> {
32    fn fit(
33        &self,
34        image: &dyn ImageInterface<f32>,
35        record: RecordValue,
36        beams: &RecordValue,
37        outline: Option<&Path>,
38        outcont: Option<&Path>,
39    ) -> PyResult<(ImageHandle, Option<ImageHandle>, ContinuumSummary)> {
40        let mut plan = ImageContinuum::new(
41            image,
42            record,
43            self.region.selection(),
44            &self.channels.0,
45            self.fitorder,
46            beams,
47        )
48        .map_err(raised)?;
49        let allocate = |path| {
50            Storage::create(
51                plan.section.shape.clone(),
52                plan.section.coordinates.clone(),
53                path,
54                None,
55            )
56        };
57        let mut line = allocate(outline).map_err(raised)?;
58        let mut model = outcont
59            .map(|path| allocate(Some(path)))
60            .transpose()
61            .map_err(raised)?;
62        let summary = match &mut model {
63            Some(model) => with_storage!(&mut line, line => {
64                with_storage!(model, model => plan.copy_to(image, line.as_mut(), Some(model.as_mut())))
65            }),
66            None => with_storage!(&mut line, line => plan.copy_to(image, line.as_mut(), None)),
67        }.map_err(raised)?;
68        let mut line = ImageHandle::Float(line);
69        let mut model = model.map(ImageHandle::Float);
70        if let Some(record) = plan.section.coordinate_record.take() {
71            if let Some(model) = &mut model {
72                model.set_coordinate_record(record.clone())?;
73            }
74            line.set_coordinate_record(record)?;
75        }
76        Ok((line, model, summary))
77    }
78
79    fn build(
80        &self,
81        image: &ImageHandle,
82        beams: &RecordValue,
83        outline: Option<&Path>,
84        outcont: Option<&Path>,
85        history: bool,
86    ) -> PyResult<(ImageHandle, Option<ImageHandle>)> {
87        image.check_single_precision("continuumsub")?;
88        let ImageHandle::Float(storage) = image else {
89            return Err(raised("This method only supports Float valued images"));
90        };
91        let record = image.coordinate_record();
92        let (mut residual, mut model, summary) = match storage {
93            Storage::Temp {
94                image,
95                fits: Some(fits),
96                ..
97            } => self.fit(&fits.image_view(image), record, beams, outline, outcont),
98            _ => {
99                with_storage!(storage, image => self.fit(image.as_ref(), record, beams, outline, outcont))
100            }
101        }?;
102        let prior = match storage {
103            Storage::Paged(image) => {
104                image_history_rows(image.name().expect("native source")).map_err(raised)?
105            }
106            Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
107        };
108        let history_arguments = history.then(|| {
109            let Self { outline, outcont, channels, region, pol, fitorder, overwrite } = self;
110            let channels = channels.0.iter().map(i64::to_string).collect::<Vec<_>>().join(",");
111            let region = region.history();
112            format!("ia.continuumsub(outline={outline:?}, outcont={outcont:?}, region={region}, channels=[{channels}], pol={pol:?}, fitorder={fitorder}, overwrite={overwrite})")
113        });
114        let profiles = summary.profiles;
115        let attempted = summary.attempted;
116        let valid = summary.valid;
117        let statistics = [
118            format!("Number of profiles       = {profiles}"),
119            format!("Number of fits attempted = {attempted}"),
120            format!("Number succeeded         = {attempted}"),
121            format!("Number converged         = {attempted}"),
122            format!("Number valid             = {valid}"),
123        ];
124        for output in std::iter::once(&mut residual).chain(model.as_mut()) {
125            let ImageHandle::Float(storage) = &mut *output else {
126                unreachable!("continuum products store Float")
127            };
128            with_storage!(storage, product => product.append_history_entries(&prior))
129                .map_err(raised)?;
130            if let Some(command) = &history_arguments {
131                output.record_history("continuumsub", command.clone())?;
132            }
133            output.append_history_at("::ImageProfileFitter::fit", &statistics)?;
134            output.save().map_err(raised)?;
135        }
136        Ok((residual, model))
137    }
138}

140#[pymethods]
141impl PyImageTool {
142    /// Fit a polynomial to the chosen channels and return the residual image.
143    #[allow(clippy::too_many_arguments)]
144    #[pyo3(signature = (outline="", outcont="continuumsub.im", region=RegionArgument::default(), channels=IntList::of(&[-1]), pol="", fitorder=0, overwrite=false), text_signature = "($self, outline='', outcont='continuumsub.im', region={}, channels=[-1], pol='', fitorder=0, overwrite=False)")]
145    fn continuumsub(
146        &self,
147        py: Python<'_>,
148        outline: &str,
149        outcont: &str,
150        region: RegionArgument,
151        channels: IntList,
152        pol: &str,
153        fitorder: i64,
154        #[pyo3(from_py_with = require_bool)] overwrite: bool,
155    ) -> PyResult<Py<Self>> {
156        let request = Request {
157            outline,
158            outcont,
159            region,
160            channels,
161            pol,
162            fitorder,
163            overwrite,
164        };
165        let product = py.detach(|| {
166            let state = self.state();
167            let Some(attached) = state.image.as_ref() else {
168                return Ok(None);
169            };
170            let mut coordinates = None;
171            let outline = outline.trim();
172            let outcont = outcont.trim();
173            let mut product = publish_image_pair(outline, outcont, overwrite, |line, model| {
174                let mut image = attached.lock();
175                for target in [outline, outcont]
176                    .into_iter()
177                    .filter(|target| !target.is_empty())
178                {
179                    for source in image
180                        .path()
181                        .into_iter()
182                        .chain(attached.operand_paths().iter().map(|path| path.as_path()))
183                    {
184                        ensure_outfile_does_not_alias_input(Path::new(target), source)
185                            .map_err(raised)?;
186                    }
187                }
188                attached.require_pixel_read_lock(&mut image)?;
189                let beams = attached.beam_view().clone();
190                let staged = request.build(&image, beams.record(), line, model, state.history);
191                let released = attached.release_requested_lock(&mut image);
192                let staged = staged?;
193                released?;
194                coordinates = AttachedImage::capture_live_coordinates(&staged.0)?;
195                Ok(staged)
196            })?;
197            if let Some((record, system)) = coordinates {
198                product.keep_live_coordinates(record, system);
199            }
200            Ok::<_, PyErr>(Some(product))
201        })?;
202        Py::new(py, product.map_or_else(Self::new, Self::attached))
203    }
