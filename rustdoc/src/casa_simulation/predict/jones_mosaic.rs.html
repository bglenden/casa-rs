2//! Sample a saved Jones image on SimplePBConvFunc's common-beam screen.
3use super::jones::{ImagePlane, JonesImage};
4use casa_coordinates::Coordinate;
5use num_complex::Complex32;
6
7impl JonesImage {
8    pub(crate) fn fill_mosaic_screen(
9        &self,
10        values: &mut [Complex32],
11        side: usize,
12        increments_deg: [f64; 2],
13        frequency_hz: f64,
14        angle: f64,
15        weight: bool,
16    ) {
17        let pa = (-angle) as f32;
18        let (sin, cos) = (f64::from(pa.sin()), f64::from(pa.cos()));
19        let increments = increments_deg.map(f64::to_radians);
20        // The screen and the synthetic Jones coordinate share their
21        // projection, pointing and poles. Their pixel mapping is exactly
22        // affine: undo the intermediate-world rotation, then increments.
23        let mapping = |image: &ImagePlane| {
24            let scale = image.frequency_hz / frequency_hz;
25            let step = image.direction.increment();
26            let pixel = image.direction.reference_pixel();
27            let (dx, dy) = (-step[0] * scale, step[1] * scale);
28            (
29                [pixel[0], pixel[1]],
30                [cos * increments[0] / dx, -sin * increments[0] / dy],
31                [sin * increments[1] / dx, cos * increments[1] / dy],
32            )
33        };
34        let real = mapping(&self.real.value);
35        let imaginary = self.imaginary.as_ref().map(|image| mapping(&image.value));
36        let centre = side as f64 / 2.;
37        for (index, value) in values.iter_mut().enumerate() {
38            let offset = [
39                (index % side) as f64 - centre,
40                (index / side) as f64 - centre,
41            ];
42            let sample = |image: &ImagePlane, (origin, x, y): ([f64; 2], [f64; 2], [f64; 2])| {
43                image.linear(std::array::from_fn(|axis| {
44                    origin[axis] + offset[0] * x[axis] + offset[1] * y[axis]
45                }))
46            };
47            let mut voltage = sample(&self.real.value, real);
48            if let (Some(image), Some(mapping)) = (&self.imaginary, imaginary) {
49                for (value, imag) in voltage.iter_mut().zip(sample(&image.value, mapping)) {
50                    value.im = imag.re;
51                }
52            }
53            // CASA deliberately constructs a one-correlation common-beam
54            // convolution. Its complex path keeps the reversed first
55            // correlation; its Float path transforms the noise covariance.
56            let power = if weight {
57                0.5 * voltage.iter().map(|v| v.norm_sqr().powi(2)).sum::<f32>()
58            } else {
59                voltage[3].norm_sqr()
60            };
61            *value = Complex32::new(power, 0.);
62        }
63    }
64}
