14/// Casacore's in-use marker for an open table that acquires access locks
15/// separately for its I/O.

28    /// access lock while this handle attaches.

60    /// Whether a read-only directory has no lock file to join.
61    pub fn readonly_without_lock_file(path: &Path) -> bool {
62        let missing = path
63            .join("table.lock")
64            .symlink_metadata()
65            .is_err_and(|error| error.kind() == std::io::ErrorKind::NotFound);
66        missing
67            && crate::publication::check_write_access(path).is_err_and(|error| {
68                matches!(
69                    error.kind(),
70                    std::io::ErrorKind::PermissionDenied | std::io::ErrorKind::ReadOnlyFilesystem
71                )
72            })
73    }

75    fn open(path: &Path, readonly: bool, attempts: u32) -> Result<Self, TableError> {
76        #[cfg(unix)]
77        {
78            let lock_io = |error: std::io::Error| TableError::LockIo {
79                path: path.display().to_string(),
80                message: error.to_string(),
81            };
82            let directory = std::fs::File::open(path).map_err(lock_io)?;
83            let mut lock_file = match LockFile::create_or_open(
84                path,
85                false,
86                LockOptions::default().inspection_interval,
87                false,
88            ) {
89                Ok(lock_file) => lock_file,
90                Err(error)
91                    if readonly
92                        && matches!(
93                            error.kind(),
94                            std::io::ErrorKind::PermissionDenied
95                                | std::io::ErrorKind::ReadOnlyFilesystem
96                        )
97                        && !path.join("table.lock").exists() =>
98                {
99                    return Ok(Self {
100                        _lock_file: None,
101                        directory,
102                    });
103                }
104                Err(error) => return Err(lock_io(error)),
105            };
106            let outcome = lock_file
107                .acquire(LockType::Read, attempts)
108                .map_err(lock_io)?;
109            if !outcome.is_acquired() {
110                return Err(TableError::LockFailed {
111                    path: path.display().to_string(),
112                    message: outcome.refusal(LockType::Read, attempts),
113                });
114            }
115            lock_file.release().map_err(lock_io)?;
116            Ok(Self {
117                _lock_file: Some(lock_file),
118                directory,
119            })
120        }
121        #[cfg(not(unix))]
122        {
123            let _ = (path, readonly, attempts);
124            Ok(Self {})
125        }
126    }

128    /// Check whether the table's original directory is still at `path`.
129    pub fn matches_directory(&self, path: &Path) -> bool {
130        std::fs::metadata(path).is_ok_and(|current| self.matches_directory_metadata(&current))
131    }

133    /// Check a directory entry captured without following symbolic links.
134    pub fn matches_directory_metadata(&self, current: &std::fs::Metadata) -> bool {
135        if !current.is_dir() {
136            return false;
137        }
138        #[cfg(unix)]
139        {
140            use std::os::unix::fs::MetadataExt;
141
142            self.directory
143                .metadata()
144                .is_ok_and(|old| old.dev() == current.dev() && old.ino() == current.ino())
145        }
146        #[cfg(not(unix))]
147        {
148            true
149        }
150    }

153/// casacore's table write lock, held for one in-place change of a table on
154/// disk.
155///
156/// An in-place writer takes this lock before it changes a table and holds it
157/// until the change is complete or abandoned, as casacore holds a table's
158/// write lock while it writes. The lock is casacore's own: the `fcntl` write
159/// lock on byte 0 of the table's `table.lock`. It therefore excludes casacore
160/// processes, other casa-rs processes, and every other casa-rs handle on the
161/// table in this process.
162///
163/// A lock another process holds, read or write, is waited for as casacore
164/// waits (see [`acquire`](Self::acquire)): the waiter adds its process id to
165/// the request list in `table.lock`, so a casacore process holding the table
166/// open with `AutoLocking` releases its lock at its next inspection.
167/// casacore throttles inspections (`LockFile::inspect` looks at the request
168/// list only after 25 table accesses and the inspection interval), so a
169/// lightly used holder may keep its lock until it closes the table. A
170/// casa-rs process does not yet release a lock on request, so a waiter waits
171/// for it to finish ([#694](https://github.com/bglenden/casa-rs/issues/694)).
172/// A write lock
173/// held by another handle in this process is never waited for, because that
174/// handle may belong to the waiting thread: it is refused at once.
175///
176/// Releasing the lock after [`record_write`](Self::record_write) records the
177/// change in the lock file's sync data (`TableSyncData`), as casacore does
178/// when it releases a write lock, so a process holding the table open
179/// re-reads it when it next takes a lock. The sync data describes the table
180/// as persisted when the lock is released. Dropping the guard releases it
181/// the same way: rows written before an interruption stay written, as after
182/// an interrupted casacore write, and are announced.
183///
184/// The guard works on the table directory, not on a [`Table`](crate::Table)
185/// handle, so it can be held across calls that each borrow the handle. It is
186/// meant for handles opened without locking
187/// ([`Table::open`](crate::Table::open)). A handle opened with
188/// [`Table::open_with_lock`](crate::Table::open_with_lock) in auto-locking
189/// mode cannot write while another
190/// handle holds this lock: its temporary write lock is refused rather than
191/// waited on, because the holder may be the same thread.
192///
193/// On a file system without lock support (`fcntl` refused with `ENOLCK`, or
194/// `ENOTSUP` as on macOS SMB mounts) the lock counts as acquired, with one
195/// warning per lock file, as casacore does for `ENOLCK`: other handles in this
196/// process are still refused, other processes are not excluded. On platforms
197/// without `fcntl` locking the guard holds nothing.
198///
199/// # C++ equivalent
200///
201/// `Table::lock(FileLocker::Write, nattempts)` and `Table::unlock()` on a
202/// table opened with `TableLock::UserLocking`.
203pub struct TableWriteLock {
204    path: PathBuf,
205    #[cfg(unix)]
206    directory_key: (u64, u64),
207    #[cfg(unix)]
208    lock_file: Option<LockFile>,
209    #[cfg(unix)]
210    reservations: Option<Arc<[DelegatedTableWrite]>>,
211    /// Whether a write was recorded; releasing then publishes it.
212    #[cfg_attr(not(unix), allow(dead_code))]
213    written: bool,
214    /// `table.dat` as it was when the lock was taken, `None` when it could
215    /// not be read; the publication compares the file with it.
216    #[cfg(unix)]
217    table_dat_at_acquire: Option<Vec<u8>>,
218}

268    /// Take the write lock on the table at `table_dir`.
269    ///
270    /// `nattempts` has the meaning of [`Table::lock`](crate::Table::lock): 1 tries once without
271    /// waiting, more retries once a second, and 0 waits indefinitely for
272    /// another process, as casacore's default `AutoLocking` does. While it
273    /// waits, this process's id is in the lock file's request list, and the
274    /// wait is logged when it starts, every ten seconds and when it ends. An
275    /// indefinite wait blocks in the kernel, which refuses it when it would
276    /// deadlock with another waiting process. A holder in this process is
277    /// never waited for.
278    ///
279    /// A writer that had to wait checks the lock file's sync data once it
280    /// holds the lock. When another process published a write to the table
281    /// while it waited, the rows and metadata the caller read beforehand are
282    /// stale; casacore would re-read them, which a handle opened without
283    /// locking cannot, so the lock is released and the request refused.
284    ///
285    /// # Errors
286    ///
287    /// [`TableError::LockFailed`] when another handle in this process holds
288    /// the write lock, when another process still holds a conflicting lock
289    /// after `nattempts` (never with 0), when waiting would deadlock, or when
290    /// another process wrote the table while this writer waited; and
291    /// [`TableError::LockIo`] when
292    /// `table.lock` cannot be opened or `fcntl` fails for a reason other than
293    /// a held lock or missing lock support.
294    pub fn acquire(table_dir: impl AsRef<Path>, nattempts: u32) -> Result<Self, TableError> {
295        Self::take(
296            table_dir.as_ref(),
297            nattempts,
298            true,
299            #[cfg(unix)]
300            None,
301            #[cfg(unix)]
302            None,
303        )
304    }

308    /// write while this caller waited.
309    ///
310    /// # Errors
311    /// As [`Self::acquire`], or a lock I/O error if cancelled. Cancellation
312    /// releases any acquired lock.
313    #[cfg(unix)]
314    pub fn acquire_interruptibly(

316        cancelled: &dyn Fn() -> bool,

410    fn take(
411        table_dir: &Path,
412        nattempts: u32,
413        refuse_stale: bool,
414        #[cfg(unix)] reservations: Option<Arc<[DelegatedTableWrite]>>,
415        #[cfg(unix)] cancelled: Option<&dyn Fn() -> bool>,
416    ) -> Result<Self, TableError> {
417        let path = table_dir.to_path_buf();
418        #[cfg(unix)]
419        {
420            let lock_io = |error: std::io::Error| TableError::LockIo {
421                path: path.display().to_string(),
422                message: error.to_string(),
423            };
424            let lock_failed = |message: String| TableError::LockFailed {
425                path: path.display().to_string(),
426                message,
427            };
428            let directory_key = path_key(&path)
429                .map_err(lock_io)?
430                .ok_or_else(|| lock_failed("table directory is missing".into()))?;
431            let mut lock_file = LockFile::create_or_open(
432                &path,
433                false,
434                LockOptions::default().inspection_interval,
435                false,
436            )
437            .map_err(lock_io)?;
438            let mut published = None;
439            let mut outcome = match cancelled {
440                Some(cancelled) => lock_file.acquire_with_reservation_interruptibly(
441                    LockType::Write,
442                    reservations.as_deref(),
443                    cancelled,
444                    |lock| published = refuse_stale.then(|| published_modify_counter(lock)),
445                ),
446                None => {
447                    lock_file.acquire_with_reservation(LockType::Write, 1, reservations.as_deref())
448                }
449            }
450            .map_err(lock_io)?;
451            if outcome == LockOutcome::HeldByAnotherProcess && nattempts != 1 {
452                published = refuse_stale.then(|| published_modify_counter(&lock_file));
453                outcome = lock_file
454                    .acquire_with_reservation(LockType::Write, nattempts, reservations.as_deref())
455                    .map_err(lock_io)?;
456            }
457            if !outcome.is_acquired() {
458                return Err(lock_failed(outcome.refusal(LockType::Write, nattempts)));
459            }
460            if published.is_some_and(|before| published_modify_counter(&lock_file) != before) {
461                return Err(lock_failed(
462                    "another process wrote the table while this writer waited for its \
463                     write lock; reopen the table and retry"
464                        .into(),
465                ));
466            }
467            if !matches_table_identity(&path, directory_key, &lock_file) {
468                return Err(lock_failed("table directory or lock was replaced".into()));
469            }
470            let table_dat_at_acquire =
471                std::fs::read(path.join(crate::storage::TABLE_CONTROL_FILE)).ok();
472            Ok(Self {
473                path,
474                directory_key,
475                lock_file: Some(lock_file),
476                reservations,
477                written: false,
478                table_dat_at_acquire,
479            })
480        }
481        #[cfg(not(unix))]
482        {
483            let _ = (nattempts, refuse_stale);
484            Ok(Self {
485                path,
486                written: false,
487            })
488        }
489    }

531    /// The modify counter the table's sync data publishes, read now; `None`
532    /// when the table has no sync data.
533    ///
534    /// While this lock is held no other process can publish a write, so a
535    /// writer that releases the lock and takes it again later compares two
536    /// readings to learn whether another process wrote the table in between.
537    ///
538    /// # Errors
539    ///
540    /// [`TableError::LockIo`] when the sync data cannot be read.
541    pub fn modify_counter(&self) -> Result<Option<u32>, TableError> {
542        #[cfg(unix)]
543        {
544            let Some(lock_file) = &self.lock_file else {
545                return Ok(None);
546            };
547            lock_file
548                .read_sync_data()
549                .map(|sync| sync.map(|sync| sync.modify_counter))
550                .map_err(|error| TableError::LockIo {
551                    path: self.path.display().to_string(),
552                    message: error.to_string(),
553                })
554        }
555        #[cfg(not(unix))]
556        {
557            Ok(None)
558        }
559    }

585    pub(crate) fn sync_recorded_write(&mut self) -> Result<(), TableError> {
586        if !self.written {
587            return Ok(());
588        }
589        #[cfg(unix)]
590        {
591            let lock = self.sync_file()?;
592            publish_persisted_write(lock, &self.path, self.table_dat_at_acquire.as_deref())
593                .map_err(|error| self.sync_error(error))?;
594            self.table_dat_at_acquire =
595                std::fs::read(self.path.join(crate::storage::TABLE_CONTROL_FILE)).ok();
596        }
597        self.written = false;
598        Ok(())
599    }

601    pub(crate) fn restore_sync(
602        &mut self,
603        previous: WriteSyncCheckpoint,
604        attempted_sync: bool,
605    ) -> Result<(), TableError> {
606        #[cfg(unix)]
607        {
608            if attempted_sync {
609                let lock = self.sync_file()?;
610                match previous.sync {
611                    Some(sync) => lock.write_sync_data(&sync),
612                    None => lock.clear_sync_data(),
613                }
614                .map_err(|error| self.sync_error(error))?;
615            }
616            self.table_dat_at_acquire = previous.descriptor_before_write;
617        }
618        #[cfg(not(unix))]
619        let _ = attempted_sync;
620        self.written = previous.written;
621        Ok(())
622    }

640    /// Release the lock, publishing any recorded write in the sync data.
641    ///
642    /// The published row, column and data-manager counts describe the table
643    /// as persisted, not an in-memory handle: casacore takes a reopened
644    /// table's row count from the sync data and asserts that its
645    /// per-data-manager counters match the table's data managers. They are
646    /// read from the table's `table.dat` when this writer changed its bytes;
647    /// when the file is as it was when the lock was taken, the row count
648    /// published before is kept, since the file can trail it (#880).
649    ///
650    /// # Errors
651    ///
652    /// [`TableError::LockIo`] when the persisted table cannot be read, the
653    /// sync data cannot be written or the lock cannot be released.
654    pub fn release(mut self) -> Result<(), TableError> {
655        self.release_now()
656    }

658    fn release_now(&mut self) -> Result<(), TableError> {
659        #[cfg(unix)]
660        {
661            let Some(mut lock_file) = self.lock_file.take() else {
662                return Ok(());
663            };
664            let lock_io = |error: std::io::Error| TableError::LockIo {
665                path: self.path.display().to_string(),
666                message: error.to_string(),
667            };
668            let published = if self.written {
669                if matches_table_identity(&self.path, self.directory_key, &lock_file) {
670                    publish_persisted_write(
671                        &lock_file,
672                        &self.path,
673                        self.table_dat_at_acquire.as_deref(),
674                    )
675                    .map(|_| ())
676                } else {
677                    Err(std::io::Error::other(
678                        "table directory or lock was replaced",
679                    ))
680                }
681            } else {
682                Ok(())
683            };
684            let released = lock_file.release().map(|_| ());
685            published.and(released).map_err(lock_io)
686        }
687        #[cfg(not(unix))]
688        {
689            Ok(())
690        }
691    }
692}
693
694#[cfg(unix)]
695fn matches_table_identity(path: &Path, directory_key: (u64, u64), lock: &LockFile) -> bool {
696    path_key(path).is_ok_and(|key| key == Some(directory_key)) && lock.matches_table_path(path)
697}

699/// The shape of a table as persisted in its `table.dat`, which the sync data
700/// published for it must describe.
701#[cfg(unix)]
702struct PersistedShape {
703    rows: u64,
704    columns: u32,
705    data_managers: usize,
706}
707
708#[cfg(unix)]
709impl PersistedShape {
710    fn read(table_dir: &Path) -> std::io::Result<Self> {
711        let contents = crate::storage::table_control::read_table_dat(&table_dir.join("table.dat"))
712            .map_err(|error| {
713                std::io::Error::other(format!(
714                    "read the persisted table {} to publish its write: {error}",
715                    table_dir.display()
716                ))
717            })?;
718        Ok(Self {
719            rows: contents.nrrow,
720            columns: contents.table_desc.columns.len() as u32,
721            data_managers: contents.column_set.data_managers.len(),
722        })
723    }
724}
725
726/// Whether `table_dir`'s `table.dat` still holds `before`, byte for byte.
727#[cfg(unix)]
728fn table_dat_unchanged(table_dir: &Path, before: &[u8]) -> bool {
729    std::fs::read(table_dir.join(crate::storage::TABLE_CONTROL_FILE)).is_ok_and(|now| now == before)
730}
731
732/// Publish a write to the table at `table_dir` in its lock file's sync data,
733/// describing the table as persisted: its row and column counts and one
734/// change counter per data manager, as casacore's `PlainTable::putFile`
735/// publishes them. The counters continue from the sync data the lock file
736/// holds, and every data manager counts as changed. The caller holds the
737/// table's write lock. Returns the published sync data.
738///
739/// `table_dat_before` is `table.dat` as it was before the write. A write
740/// that changed the file's bytes publishes the counts it gives, which may be
741/// fewer rows than before, as casacore's full `putFile` flushes `table.dat`
742/// before it publishes. A write that left the file's bytes as they were (one
743/// that wrote only data, stopped before it reached the file, wrote nothing,
744/// or wrote the file back unchanged) keeps the row count published before,
745/// as casacore's data-only `putFile` publishes the count it holds without
746/// rewriting the file. The file can trail that count (#880), and publishing
747/// the file's would then announce fewer rows than the storage managers hold,
748/// which casacore refuses.
749#[cfg(unix)]
750pub(crate) fn publish_persisted_write(
751    lock_file: &LockFile,
752    table_dir: &Path,
753    table_dat_before: Option<&[u8]>,
754) -> std::io::Result<SyncData> {
755    let shape = PersistedShape::read(table_dir)?;
756    let published = lock_file.read_sync_data()?;
757    let rows = match &published {
758        Some(published)
759            if table_dat_before.is_some_and(|before| table_dat_unchanged(table_dir, before)) =>
760        {
761            published.nrrow
762        }
763        _ => shape.rows,
764    };
765    let mut sync = published.unwrap_or_else(SyncData::new);
766    sync.record_write(rows, shape.columns, true, &vec![true; shape.data_managers]);
767    lock_file.write_sync_data(&sync)?;
768    Ok(sync)
769}
770
771/// Brings the row count the lock file's sync data publishes to the rows
772/// `table.dat` holds now, while the write lock is still held, and changes
773/// nothing else in it: no counter moves, so the release publishes the write
774/// once, as it does for every write. Nothing is done when no sync data has
775/// been published (the count then reads as 0, and the table's own is used).
776///
777/// A full save that removed rows needs this: until the release, a
778/// metadata-only save, the admission of a partial save and a reload read
779/// that count back and take the larger of it and `table.dat`'s, which would
780/// raise the file to the rows the save removed.
781#[cfg(unix)]
782pub(crate) fn publish_persisted_rows(
783    lock_file: &LockFile,
784    table_dir: &Path,
785) -> std::io::Result<()> {
786    let Some(mut sync) = lock_file.read_sync_data()? else {
787        return Ok(());
788    };
789    let rows = PersistedShape::read(table_dir)?.rows;
790    if sync.nrrow != rows {
791        sync.nrrow = rows;
792        lock_file.write_sync_data(&sync)?;
793    }
794    Ok(())
795}
796
797/// The modify counter the lock file's sync data publishes, `None` when it has
798/// none. A read that fails means a writer was publishing at that moment, and
799/// counts as `None`, so a change is still seen.
800#[cfg(unix)]
801fn published_modify_counter(lock_file: &LockFile) -> Option<u32> {
802    lock_file
803        .read_sync_data()
804        .ok()
805        .flatten()
806        .map(|sync| sync.modify_counter)
807}
