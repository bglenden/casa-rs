55/// Why a native run did not complete.
56#[derive(Debug, thiserror::Error)]
57pub(crate) enum NativeError {
58    /// A phase's memory did not fit the resource policy.
59    #[error(transparent)]
60    Admission(#[from] Admission),
61    /// The run was cancelled.
62    #[error(transparent)]
63    Cancelled(#[from] Cancelled),
64    /// Any other failure.
65    #[error("{0}")]
66    Other(ApplicationError),
67}
68
69impl From<NativeError> for ApplicationDispatchError {
70    fn from(error: NativeError) -> Self {
71        match error {
72            NativeError::Admission(admission) => Self::Admission(admission),
73            NativeError::Cancelled(_) => Self::Cancelled,
74            NativeError::Other(error) => Self::Native(error),
75        }
76    }
77}
78
79impl From<ApplicationError> for NativeError {
80    fn from(error: ApplicationError) -> Self {
81        Self::Other(error)
82    }
83}
84
85impl From<ImagingError> for NativeError {
86    fn from(error: ImagingError) -> Self {
87        match error {
88            ImagingError::Admission(admission)
89            | ImagingError::Minor(casa_imaging_runtime::MinorCycleRunError::Admission(admission)) => {
90                Self::Admission(admission)
91            }
92            ImagingError::Cancelled(cancelled) => Self::Cancelled(cancelled),
93            ImagingError::Pass(error) => error.into(),
94            error => Self::Other(Box::new(error)),
95        }
96    }
97}

104        }
105    }

120/// A native run ready to start: its admitted source and what its passes,
121/// cycles and products need. Its setup chooses and forms its worker teams.
122pub(crate) struct NativeRun {
123    runtime: ApplicationRuntime,
124    publication: ApplicationPublication,
125    aw_catalog: Option<AwCatalogDeployment>,
126    /// The observation's batches the passes read, and through whose update
127    /// session a model or corrected column is written.
128    route: Box<imaging::BatchRoute>,
129    /// The source's charge, held while it is open.
130    source: Reservation,
131    /// What the products hold on one lane, which the run's setup fits
132    /// beside it ([`products_floor`]).
133    products: u64,
134    masks: ImageDomainReconstructionMaskPlans,
135    start_model: Option<crate::compile::StartModel>,
136    image_response: Option<MinorCycleImageResponse>,
137}

139/// The batch route of `problem` on `observation`, the observation compile
140/// opened: the selection resolved to its partitions, the geometry compiled
141/// and its batches admitted ([`imaging::admit_batches`]). A run that writes
142/// the model (`model_data`) or the corrected data (`corrected_data`) reads
143/// the observation through the update session that writes them
144/// ([`imaging::WriteBack`]).
145pub(crate) fn batch_route(
146    problem: &CompiledProblem,
147    observation: OpenedObservation,
148    (model_data, corrected_data): (bool, bool),
149    runtime: &ApplicationRuntime,
150) -> Result<Admitted<imaging::BatchRoute>, NativeError> {
151    let OpenedObservation {
152        store: Admitted {
153            value: opened,
154            reservation: mut opening,
155        },
156        measures,
157        phase_table,
158    } = observation;
159    let selection = resolve_partitions(problem, opened.observation())
160        .map_err(|error| NativeError::Other(Box::new(error)))?;
161    let plan = GeometryPlan::compile(problem, opened.catalogue(), measures, phase_table)
162        .map_err(|error| NativeError::Other(Box::new(error)))?;
163    let pointing_budget = if plan.reads_pointing() {
164        opened
165            .observation()
166            .pointing_bytes()
167            .map_err(|error| NativeError::Other(Box::new(error)))?
168    } else {
169        0
170    };
171    let Admitted {
172        value: batching,
173        reservation,
174    } = imaging::admit_batches(
175        (problem, opened.observation(), &selection, &plan),
176        imaging::row_samples(problem)?,
177        pointing_budget,
178        (&runtime.host, &runtime.resource_policy),
179    )?;
180    opening.join(reservation);
181    let store = if model_data || corrected_data {
182        imaging::RouteStore::Write(Box::new(
183            imaging::WriteBack::open(opened, model_data, corrected_data)
184                .map_err(|error| NativeError::Other(Box::new(error)))?,
185        ))
186    } else {
187        imaging::RouteStore::Read(opened.into_observation())
188    };
189    Ok(Admitted {
190        value: imaging::BatchRoute {
191            store,
192            selection,
193            plan,
194            batching,
195            pointing_budget,
196        },
197        reservation: opening,
198    })
199}

201/// The end of a run's preparation: check the products against `problem`
202/// and admit the source, its pass storage sized for the policy's streaming
203/// workers ([`casa_imaging_runtime::ResourcePolicy::cores`]), the most the run's setup then
204/// chooses.
205pub(crate) fn prepare(
206    problem: &CompiledProblem,
207    input: NativeInput,
208    native: ApplicationNative,
209) -> Result<NativeRun, NativeError> {
210    let ApplicationNative {
211        runtime,
212        publication,
213        aw_catalog,
214    } = native;
215    publication.controls.validate_for_problem(problem)?;
216    let products = products_floor(problem, &publication)?;
217    let Admitted {
218        value: route,
219        reservation: source,
220    } = batch_route(
221        problem,
222        input.observation,
223        (input.write_model_column, input.write_corrected_data),
224        &runtime,
225    )?;
226    Ok(NativeRun {
227        runtime,
228        publication,
229        aw_catalog,
230        route: Box::new(route),
231        source,
232        products,
233        masks: input.masks,
234        start_model: input.start_model,
235        image_response: input.minor_cycle_image_response,
236    })
237}

239/// Run the major-cycle passes, the minor cycles between them and any
240/// visibility write, then publish the products. `summary` collects the
241/// phases and totals as they end; a completed run's outcome takes it once
242/// the caller has finished it.
243pub(crate) fn run(
244    problem: &CompiledProblem,
245    run: NativeRun,
246    summary: &mut RunSummary,
247) -> Result<impl FnOnce(RunSummary) -> ImagingOutcome + use<>, NativeError> {
248    let NativeRun {
249        runtime,
250        publication,
251        aw_catalog,
252        route,
253        source,
254        products,
255        masks,
256        start_model,
257        image_response,
258    } = run;
259    let teams = OnceCell::new();
260    // The run's source borrows the batch route, which is released with the
261    // source's charge once the run has finished with it.
262    let visibility_write = route
263        .write_back()
264        .map(|_| imaging::VisibilityWriteTarget(&route));
265    let outcome = imaging::run(
266        imaging::ImagingInputs {
267            problem,
268            source: &route,
269            masks,
270            start_model,
271            image_response,
272            visibility_write,
273            products: (products, source.memory()),
274            host: runtime.host,
275            policy: runtime.resource_policy,
276            teams: &teams,
277            cancel: runtime.cancel.clone(),
278            spill_directory: &runtime.spill_directory,
279            aw_catalog,
280            backend: runtime.backend,
281            grid_precision: runtime.grid_precision,
282            diagnostics: runtime.diagnostics,
283        },
284        summary,
285    )?;
286    drop(route);
287    drop(source);
288    let team = teams
289        .get()
290        .expect("a run that completed formed its teams")
291        .compute();
292    let mut published = Vec::new();
293    let mut lanes = (0, None);
294    let products = run_phase(
295        "products",
296        &runtime.cancel,
297        runtime.diagnostics,
298        summary,
299        || {
300            publish_products(
301                problem,
302                ProductSource {
303                    scientific: outcome.scientific,
304                    masks: outcome.masks,
305                    facet_sum_weights: outcome.facet_sum_weights,
306                },
307                &runtime,
308                publication,
309                team,
310                &mut published,
311                &mut lanes,
312            )
313        },
314    );
315    summary.products = published;
316    (summary.product_lanes, summary.product_lanes_reason) = lanes;
317    let products = products?;
318    let workers = team.workers();
319    let problem = problem.clone();
320    Ok(move |summary| ImagingOutcome {
321        problem,
322        minor_cycles: outcome.minor_cycles,
323        stop: outcome.stop,
324        major_cycle_count: outcome.major_cycle_count,
325        total_minor_iterations: outcome.total_minor_iterations,
326        total_actual_minor_iterations: outcome.total_actual_minor_iterations,
327        workers,
328        planes_per_wave: outcome.planes_per_wave,
329        visibility_products: outcome.visibility_products,
330        summary,
331        scientific: products.scientific,
332        planned_products: products.planned,
333        products: products.published,
334        _retained: outcome.retained,
335    })
336}

338/// The products of a run and the state they were generated from.
339struct Products {

341    scientific: MajorCycleCompletion,

345/// What the products are formed from: the final completion, its
346/// reconstruction masks and the faceted domains' facets' own sums of
347/// weights.
348struct ProductSource {
349    scientific: MajorCycleCompletion,
350    masks: Option<ReconstructionMaskSet>,
351    facet_sum_weights: FacetSumWeights,
352}

354/// What `problem`'s products hold on one lane, before the run
355/// (`PlannedContinuumGeneration::planned_demand`): the least they can run
356/// on, which the run's setup fits beside it.
357fn products_floor(
358    problem: &CompiledProblem,
359    publication: &ApplicationPublication,
360) -> Result<u64, NativeError> {
361    Ok(
362        PlannedContinuumGeneration::for_problem(problem, &publication.controls)?
363            .planned_demand(problem, ProductStoragePlan::new(1, 1)?, &publication.writer)?
364            .peak_residency_bytes(),
365    )
366}

368/// Generate every planned product member into the writer's private staging,
369/// one output channel per window on the run's team, and publish them,
370/// adding each image published to `published`; a cancelled run publishes
371/// nothing. `lanes` receives the lanes the products ran on and, when fewer
372/// than the team's workers, why.
373fn publish_products(
374    problem: &CompiledProblem,
375    source: ProductSource,
376    runtime: &ApplicationRuntime,
377    publication: ApplicationPublication,
378    team: &WorkerTeam,
379    published: &mut Vec<String>,
380    lanes: &mut (usize, Option<String>),
381) -> Result<Products, NativeError> {
382    let ProductSource {
383        scientific,
384        masks: reconstruction_masks,
385        facet_sum_weights,
386    } = source;
387    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific);
388    if let Some(masks) = reconstruction_masks.as_ref() {
389        inputs = match masks {
390            ReconstructionMaskSet::Shared(mask) => inputs.with_reconstruction_mask(mask),
391            ReconstructionMaskSet::Domains(masks) => inputs.with_domain_reconstruction_masks(masks),
392        };
393    }
394    if let Some(centres) = publication.primary_beam_centres.as_ref() {
395        inputs = inputs.with_primary_beam_centres(centres);
396    }
397    if !facet_sum_weights.is_empty() {
398        inputs = inputs.with_facet_sum_weights(&facet_sum_weights);
399    }
400    let planned = PlannedContinuumGeneration::new(&inputs, &publication.controls)?;
401    // The products' scratch grows with their lanes, and the gridding's
402    // workers were chosen without it: the products take the most lanes, up
403    // to the team's, whose demand fits what the policy leaves free (#763
404    // review), and one when none does, which admission then refuses.
405    let free = free_memory(&runtime.host, &runtime.resource_policy);
406    let demand_on = |count: usize| -> Result<_, NativeError> {
407        let window = ProductStoragePlan::new(1, count)?;
408        Ok((
409            window,
410            planned.demand(&inputs, window, &publication.writer)?,
411        ))
412    };
413    let workers = team.workers();
414    let (mut window, mut demand) = demand_on(workers)?;
415    debug_assert_eq!(
416        planned.planned_demand(problem, window, &publication.writer)?,
417        demand,
418        "the products' demand before the run is the run's"
419    );
420    let needed = demand.peak_residency_bytes();
421    if needed > free {
422        for count in (1..workers).rev() {
423            (window, demand) = demand_on(count)?;
424            if demand.peak_residency_bytes() <= free {
425                break;
426            }
427        }
428        let count = window.maximum_workers();
429        let gigabytes = |bytes: u64| format!("{:.2} GB", bytes as f64 / 1e9);
430        let reason = if demand.peak_residency_bytes() <= free {
431            format!(
432                "{workers} lanes need {}, {count} fit {} free",
433                gigabytes(needed),
434                gigabytes(free)
435            )
436        } else {
437            format!(
438                "{workers} lanes need {} and no fewer fit {} free",
439                gigabytes(needed),
440                gigabytes(free)
441            )
442        };
443        if count < workers {
444            tracing::warn!("products on {count} lanes, not the team's {workers}: {reason}");
445            lanes.1 = Some(reason);
446        }
447    }
448    lanes.0 = window.maximum_workers();
449    let held = admit(
450        &runtime.host,
451        &runtime.resource_policy,
452        &Demand {
453            phase: "products",
454            memory: demand.peak_residency_bytes(),
455        },
456    )?;
457    let generation =
458        produce_continuum_members(&planned, &inputs, window, team, &publication.writer)?;
459    if runtime.cancel.is_cancelled() {
460        // The staged members are removed when the writer drops.
461        return Err(Cancelled.into());
462    }
463    publication
464        .writer
465        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
466    // The charge covers the staged records until they are published.
467    drop(held);
468    Ok(Products {
469        planned,
470        scientific,
471        published: generation,
472    })
