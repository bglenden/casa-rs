36#[derive(Debug, Clone, Copy, PartialEq)]
37struct AntennaPointingQueryDomain {
38    antenna_id: i32,
39    time_bounds: [f64; 2],
40}
41
42impl SelectedPointingQueryDomain {
43    #[must_use]
44    pub(crate) fn builder() -> SelectedPointingQueryDomainBuilder {
45        SelectedPointingQueryDomainBuilder::default()
46    }
47
48    pub(crate) fn antenna_ids(&self) -> impl ExactSizeIterator<Item = i32> + '_ {
49        self.antennas.iter().map(|antenna| antenna.antenna_id)
50    }
51
52    #[must_use]
53    pub(crate) fn time_bounds_mjd_seconds(&self, antenna_id: i32) -> Option<[f64; 2]> {
54        let antenna = self.antennas.get(
55            self.antennas
56                .binary_search_by_key(&antenna_id, |antenna| antenna.antenna_id)
57                .ok()?,
58        )?;
59        Some(antenna.time_bounds)
60    }
61}

63/// One-pass selected MAIN metadata accumulator for POINTING preparation.
64#[derive(Debug, Default)]
65pub(crate) struct SelectedPointingQueryDomainBuilder {
66    antennas: Vec<AntennaPointingQueryDomain>,
67}
68
69impl SelectedPointingQueryDomainBuilder {
70    pub(crate) fn observe_row(
71        &mut self,
72        antenna1: i32,
73        antenna2: i32,
74        time_mjd_seconds: f64,
75    ) -> MsResult<()> {
76        if antenna1 < 0 || antenna2 < 0 || !time_mjd_seconds.is_finite() {
77            return Err(MsError::InvalidInput(
78                "selected POINTING domain requires non-negative antennas and finite epochs"
79                    .to_string(),
80            ));
81        }
82        self.observe_antenna(antenna1, time_mjd_seconds);
83        self.observe_antenna(antenna2, time_mjd_seconds);
84        Ok(())
85    }
86
87    fn observe_antenna(&mut self, antenna_id: i32, time_mjd_seconds: f64) {
88        if let Some(antenna) = self
89            .antennas
90            .iter_mut()
91            .find(|antenna| antenna.antenna_id == antenna_id)
92        {
93            extend_bounds(&mut antenna.time_bounds, time_mjd_seconds);
94        } else {
95            self.antennas.push(AntennaPointingQueryDomain {
96                antenna_id,
97                time_bounds: [time_mjd_seconds; 2],
98            });
99        }
100    }
101
102    pub(crate) fn finish(mut self) -> MsResult<Option<SelectedPointingQueryDomain>> {
103        if self.antennas.is_empty() {
104            return Ok(None);
105        }
106        self.antennas.sort_by_key(|antenna| antenna.antenna_id);
107        Ok(Some(SelectedPointingQueryDomain {
108            antennas: self.antennas.into_boxed_slice(),
109        }))
110    }
111}
112
113fn extend_bounds([first, last]: &mut [f64; 2], value: f64) {
114    *first = first.min(value);
115    *last = last.max(value);
116}

199#[derive(Debug)]
200pub(crate) struct PreparedSelectedPointingCatalog {
201    rows: Vec<CatalogRow>,
202    coefficients: Vec<f64>,
203    /// What the read held, which the counting laws check.
204    #[cfg_attr(not(test), allow(dead_code))]
205    measurements: SelectedPointingCatalogMeasurements,
206}
207
208impl PreparedSelectedPointingCatalog {
209    #[cfg(test)]
210    #[must_use]
211    pub(crate) const fn measurements(&self) -> SelectedPointingCatalogMeasurements {
212        self.measurements
213    }
214
215    /// The rows as the observation boundary's pointing series: per antenna
216    /// (an antenna's catalogue id is its `ANTENNA` row), in the catalog's
217    /// (time, row) order, with their intervals and polynomial origins, and
218    /// coefficients padded with zeros to the most terms any row has.
219    ///
220    /// # Errors
221    ///
222    /// [`casa_observation::ObservationError::Unsupported`] when the rows'
223    /// directions are in several frames.
224    pub(crate) fn series(
225        &self,
226    ) -> Result<casa_observation::PointingSeries, casa_observation::ObservationError> {
227        let frame = self
228            .rows
229            .first()
230            .map_or(DirectionRef::J2000, |row| row.source_ref);
231        if self.rows.iter().any(|row| row.source_ref != frame) {
232            return Err(casa_observation::ObservationError::Unsupported {
233                what: "MSv2 POINTING directions in several frames",
234            });
235        }
236        let terms = self
237            .rows
238            .iter()
239            .map(|row| row.metadata.polynomial_terms)
240            .max()
241            .unwrap_or(1);
242        let mut series = casa_observation::PointingSeries {
243            terms,
244            ..casa_observation::PointingSeries::empty(frame)
245        };
246        for rows in self
247            .rows
248            .chunk_by(|left, right| left.metadata.antenna_id == right.metadata.antenna_id)
249        {
250            let mut coefficients = Vec::with_capacity(rows.len() * 2 * terms);
251            for row in rows {
252                let own = row.metadata.polynomial_terms;
253                for axis in 0..2 {
254                    let start = row.coefficient_offset + axis * own;
255                    coefficients.extend_from_slice(&self.coefficients[start..start + own]);
256                    coefficients.extend(std::iter::repeat_n(0.0, terms - own));
257                }
258            }
259            let antenna = casa_observation::AntennaId(
260                u32::try_from(rows[0].metadata.antenna_id).expect("ANTENNA_ID is non-negative"),
261            );
262            series.antennas.insert(
263                antenna,
264                casa_observation::AntennaPointing::new(
265                    rows.iter()
266                        .map(|row| row.metadata.time_mjd_seconds)
267                        .collect(),
268                    Some(
269                        rows.iter()
270                            .map(|row| row.metadata.interval_seconds)
271                            .collect(),
272                    ),
273                    rows.iter()
274                        .map(|row| row.metadata.time_origin_mjd_seconds)
275                        .collect(),
276                    terms,
277                    coefficients,
278                )?,
279            );
280        }
281        Ok(series)
282    }
283}

285impl MeasurementSet {
286    /// Bound POINTING preparation from row-count metadata without reading its cells.
287    ///
288    /// All POINTING rows are conservatively treated as retained candidates at the
289    /// planned polynomial bound. The measurements contain byte/count ceilings,
290    /// with a zero build time; the construction ceiling excludes scalar scan row
291    /// storage. The second result charges that storage per scan-block row. A
292    /// missing POINTING table requires neither catalog nor scan storage.
293    pub(crate) fn selected_pointing_catalog_requirements(
294        &self,
295        domain: &SelectedPointingQueryDomain,
296        maximum_polynomial_terms: usize,
297    ) -> MsResult<(SelectedPointingCatalogMeasurements, usize)> {
298        if maximum_polynomial_terms == 0 {
299            return Err(MsError::InvalidInput(
300                "POINTING requirements need a positive polynomial bound".to_string(),
301            ));
302        }
303        let rows = match self.pointing() {
304            Ok(pointing) => pointing.row_count(),
305            Err(MsError::MissingSubtable(_)) => {
306                return Ok((SelectedPointingCatalogMeasurements::default(), 0));
307            }
308            Err(error) => return Err(error),
309        };
310        let endpoint_count = domain.antenna_ids().len();
311        let endpoint_bytes = endpoint_count
312            .checked_mul(size_of::<[Option<CandidateMetadata>; 2]>())
313            .ok_or_else(byte_overflow)?;
314        let prechecked_rows = endpoint_count
315            .checked_mul(2)
316            .and_then(|endpoints| rows.checked_add(endpoints))
317            .ok_or_else(byte_overflow)?;
318        let scan_fixed_bytes =
319            catalog_upper_bound_bytes(prechecked_rows, maximum_polynomial_terms)?
320                .checked_add(endpoint_bytes)
321                .ok_or_else(byte_overflow)?;
322        let retained_bytes = catalog_upper_bound_bytes(rows, maximum_polynomial_terms)?;
323        // The catalog bound already includes one coefficient copy and array
324        // dimensions. Direction cells coexist with the final coefficient copy.
325        let copy_bytes_per_row = maximum_polynomial_terms
326            .checked_mul(2)
327            .and_then(|terms| terms.checked_mul(size_of::<f64>()))
328            .and_then(|bytes| bytes.checked_add(size_of::<CandidateMetadata>()))
329            .and_then(|bytes| bytes.checked_add(size_of::<usize>()))
330            .and_then(|bytes| bytes.checked_add(size_of::<Option<ArrayValue>>()))
331            .ok_or_else(byte_overflow)?;
332        let copy_peak_bytes = rows
333            .checked_mul(copy_bytes_per_row)
334            .and_then(|bytes| bytes.checked_add(retained_bytes))
335            .ok_or_else(byte_overflow)?;
336        Ok((
337            SelectedPointingCatalogMeasurements {
338                retained_bytes,
339                construction_peak_bytes: scan_fixed_bytes.max(copy_peak_bytes),
340                source_rows_scanned: rows,
341                retained_rows: rows,
342                build_nanos: 0,
343            },
344            SCALAR_SCAN_BYTES_PER_ROW,
345        ))
346    }
347
348    /// Scan POINTING scalar metadata once, then retain and read only selected candidates.
349    #[expect(
350        clippy::too_many_lines,
351        reason = "predates the function-length limit; #705"
352    )]
353    pub(crate) fn prepare_selected_pointing_catalog(
354        &self,
355        column: PointingDirectionColumn,
356        domain: &SelectedPointingQueryDomain,
357        plan: PointingReadPlan,
358    ) -> MsResult<PreparedSelectedPointingCatalog> {
359        let started = Instant::now();
360        let pointing = match self.pointing() {
361            Ok(pointing) => pointing,
362            Err(MsError::MissingSubtable(_)) => {
363                return Ok(PreparedSelectedPointingCatalog {
364                    rows: Vec::new(),
365                    coefficients: Vec::new(),
366                    measurements: SelectedPointingCatalogMeasurements {
367                        build_nanos: elapsed_nanos(started),
368                        ..Default::default()
369                    },
370                });
371            }
372            Err(error) => return Err(error),
373        };
374        let table = pointing.table();
375        let mut retained = Vec::<CandidateMetadata>::new();
376        let endpoint_count = domain.antenna_ids().len();
377        let endpoint_bytes = endpoint_count
378            .checked_mul(size_of::<[Option<CandidateMetadata>; 2]>())
379            .ok_or_else(byte_overflow)?;
380        let maximum_scan_bytes = plan
381            .scan_rows_per_block
382            .checked_mul(SCALAR_SCAN_BYTES_PER_ROW)
383            .and_then(|bytes| bytes.checked_add(endpoint_bytes))
384            .ok_or_else(byte_overflow)?;
385        ensure_budget(maximum_scan_bytes, plan.maximum_catalog_bytes)?;
386        let mut endpoints = Vec::new();
387        endpoints.try_reserve_exact(endpoint_count).map_err(|_| {
388            MsError::InvalidInput("POINTING endpoint allocation failed".to_string())
389        })?;
390        endpoints.resize(endpoint_count, [None; 2]);
391        let mut construction_peak_bytes = 0usize;
392
393        for block_start in (0..pointing.row_count()).step_by(plan.scan_rows_per_block) {
394            let block_end = (block_start + plan.scan_rows_per_block).min(pointing.row_count());
395            let rows = (block_start..block_end).collect::<Vec<_>>();
396            let antennas = table
397                .column_accessor("ANTENNA_ID")?
398                .scalar_cells_owned_for_rows(&rows)?;
399            let times = table
400                .column_accessor("TIME")?
401                .scalar_cells_owned_for_rows(&rows)?;
402            let intervals = table
403                .column_accessor("INTERVAL")?
404                .scalar_cells_owned_for_rows(&rows)?;
405            let time_origins = table
406                .column_accessor("TIME_ORIGIN")?
407                .scalar_cells_owned_for_rows(&rows)?;
408            let orders = table
409                .column_accessor("NUM_POLY")?
410                .scalar_cells_owned_for_rows(&rows)?;
411            let scan_bytes = rows.capacity() * size_of::<usize>()
412                + (antennas.capacity()
413                    + times.capacity()
414                    + intervals.capacity()
415                    + time_origins.capacity()
416                    + orders.capacity())
417                    * size_of::<Option<ScalarValue>>();
418            construction_peak_bytes = construction_peak_bytes.max(
419                retained.capacity() * size_of::<CandidateMetadata>()
420                    + endpoints.capacity() * size_of::<[Option<CandidateMetadata>; 2]>()
421                    + scan_bytes,
422            );
423
424            for (slot, row_index) in rows.into_iter().enumerate() {
425                let antenna_id = required_i32(&antennas, slot, "ANTENNA_ID", row_index)?;
426                let Some([first_query_time, last_query_time]) =
427                    domain.time_bounds_mjd_seconds(antenna_id)
428                else {
429                    continue;
430                };
431                let time_mjd_seconds = required_f64(&times, slot, "TIME", row_index)?;
432                let interval_seconds = required_f64(&intervals, slot, "INTERVAL", row_index)?;
433                let time_origin_mjd_seconds =
434                    required_f64(&time_origins, slot, "TIME_ORIGIN", row_index)?;
435                let order = required_i32(&orders, slot, "NUM_POLY", row_index)?;
436                let polynomial_terms = usize::try_from(order).ok()
437                    .and_then(|order| order.checked_add(1))
438                    .filter(|terms| *terms <= plan.maximum_polynomial_terms)
439                    .ok_or_else(|| MsError::InvalidInput(format!(
440                        "POINTING row {row_index} polynomial order {order} exceeds the planned bound"
441                    )))?;
442                if !time_mjd_seconds.is_finite()
443                    || !interval_seconds.is_finite()
444                    || !time_origin_mjd_seconds.is_finite()
445                {
446                    return Err(MsError::InvalidInput(format!(
447                        "POINTING row {row_index} has non-finite time metadata"
448                    )));
449                }
450                let candidate = CandidateMetadata {
451                    row_index,
452                    antenna_id,
453                    time_mjd_seconds,
454                    interval_seconds,
455                    time_origin_mjd_seconds,
456                    polynomial_terms,
457                };
458                let half_interval = 0.5 * interval_seconds;
459                let intersects = interval_seconds > 0.0
460                    && time_mjd_seconds + half_interval >= first_query_time
461                    && time_mjd_seconds - half_interval <= last_query_time;
462                let centre_inside =
463                    time_mjd_seconds >= first_query_time && time_mjd_seconds <= last_query_time;
464                if intersects || centre_inside {
465                    let next_rows = retained
466                        .len()
467                        .checked_add(1 + 2 * endpoints.len())
468                        .ok_or_else(byte_overflow)?;
469                    let required =
470                        catalog_upper_bound_bytes(next_rows, plan.maximum_polynomial_terms)?
471                            .checked_add(scan_bytes)
472                            .and_then(|bytes| {
473                                bytes.checked_add(
474                                    endpoints.capacity()
475                                        * size_of::<[Option<CandidateMetadata>; 2]>(),
476                                )
477                            })
478                            .ok_or_else(byte_overflow)?;
479                    ensure_budget(required, plan.maximum_catalog_bytes)?;
480                    retained.try_reserve_exact(1).map_err(|_| {
481                        MsError::InvalidInput(
482                            "POINTING candidate metadata allocation failed".to_string(),
483                        )
484                    })?;
485                    retained.push(candidate);
486                } else {
487                    let endpoint_index = domain
488                        .antenna_ids()
489                        .position(|selected| selected == antenna_id)
490                        .expect("selected antenna was already found");
491                    let endpoint = &mut endpoints[endpoint_index];
492                    if time_mjd_seconds < first_query_time {
493                        retain_later(&mut endpoint[0], candidate);
494                    }
495                    if time_mjd_seconds > last_query_time {
496                        retain_earlier(&mut endpoint[1], candidate);
497                    }
498                }
499            }
500            ensure_budget(
501                catalog_upper_bound_bytes(
502                    retained.capacity() + 2 * endpoints.len(),
503                    plan.maximum_polynomial_terms,
504                )?,
505                plan.maximum_catalog_bytes,
506            )?;
507        }
508
509        let endpoint_rows = endpoints.iter().flatten().flatten().count();
510        let final_metadata_rows = retained
511            .len()
512            .checked_add(endpoint_rows)
513            .ok_or_else(byte_overflow)?;
514        ensure_budget(
515            catalog_upper_bound_bytes(final_metadata_rows, plan.maximum_polynomial_terms)?,
516            plan.maximum_catalog_bytes,
517        )?;
518        retained.try_reserve_exact(endpoint_rows).map_err(|_| {
519            MsError::InvalidInput("POINTING endpoint retention allocation failed".to_string())
520        })?;
521        retained.extend(endpoints.into_iter().flatten().flatten());
522        retained.sort_by(|left, right| {
523            left.antenna_id
524                .cmp(&right.antenna_id)
525                .then_with(|| left.time_mjd_seconds.total_cmp(&right.time_mjd_seconds))
526                .then_with(|| left.row_index.cmp(&right.row_index))
527        });
528        retained.dedup_by_key(|row| row.row_index);
529        let index_bytes = retained
530            .len()
531            .checked_mul(size_of::<usize>())
532            .ok_or_else(byte_overflow)?;
533        let index_peak = catalog_upper_bound_bytes(retained.len(), plan.maximum_polynomial_terms)?
534            .checked_add(retained.capacity() * size_of::<CandidateMetadata>())
535            .and_then(|bytes| bytes.checked_add(index_bytes))
536            .ok_or_else(byte_overflow)?;
537        ensure_budget(index_peak, plan.maximum_catalog_bytes)?;
538        let mut row_indices = Vec::new();
539        row_indices.try_reserve_exact(retained.len()).map_err(|_| {
540            MsError::InvalidInput("POINTING row-index allocation failed".to_string())
541        })?;
542        row_indices.extend(retained.iter().map(|row| row.row_index));
543        ensure_budget(
544            construction_upper_bound(&retained, &row_indices, plan.maximum_polynomial_terms)?,
545            plan.maximum_catalog_bytes,
546        )?;
547        let direction_cells = table
548            .column_accessor(column.name())?
549            .array_cells_owned_uncached(&row_indices)?;
550        let direction_heap_bytes = direction_cells
551            .iter()
552            .flatten()
553            .try_fold(0usize, |bytes, value| {
554                bytes.checked_add(array_heap_bytes(value)?)
555            })
556            .ok_or_else(byte_overflow)?;
557        construction_peak_bytes = construction_peak_bytes.max(
558            retained.capacity() * size_of::<CandidateMetadata>()
559                + row_indices.capacity() * size_of::<usize>()
560                + direction_cells.capacity() * size_of::<Option<ArrayValue>>()
561                + direction_heap_bytes,
562        );
563
564        let total_coefficients = retained
565            .iter()
566            .try_fold(0usize, |total, metadata| {
567                total.checked_add(2 * metadata.polynomial_terms)
568            })
569            .ok_or_else(byte_overflow)?;
570        let copy_peak = retained
571            .capacity()
572            .checked_mul(size_of::<CandidateMetadata>())
573            .and_then(|bytes| bytes.checked_add(row_indices.capacity() * size_of::<usize>()))
574            .and_then(|bytes| {
575                bytes.checked_add(direction_cells.capacity() * size_of::<Option<ArrayValue>>())
576            })
577            .and_then(|bytes| bytes.checked_add(direction_heap_bytes))
578            .and_then(|bytes| bytes.checked_add(retained.len() * size_of::<CatalogRow>()))
579            .and_then(|bytes| bytes.checked_add(total_coefficients * size_of::<f64>()))
580            .ok_or_else(byte_overflow)?;
581        ensure_budget(copy_peak, plan.maximum_catalog_bytes)?;
582        construction_peak_bytes = construction_peak_bytes.max(copy_peak);
583        let mut catalog_rows = Vec::new();
584        catalog_rows
585            .try_reserve_exact(retained.len())
586            .map_err(|_| {
587                MsError::InvalidInput("POINTING catalog row allocation failed".to_string())
588            })?;
589        let mut coefficients = Vec::new();
590        coefficients
591            .try_reserve_exact(total_coefficients)
592            .map_err(|_| {
593                MsError::InvalidInput("POINTING coefficient allocation failed".to_string())
594            })?;
595        for ((metadata, direction), row_index) in retained
596            .into_iter()
597            .zip(direction_cells)
598            .zip(row_indices.iter().copied())
599        {
600            let direction = direction.ok_or_else(|| {
601                MsError::InvalidInput(format!(
602                    "POINTING.{} row {row_index} is undefined",
603                    column.name()
604                ))
605            })?;
606            let ArrayValue::Float64(coefficient_array) = direction else {
607                return Err(MsError::ColumnTypeMismatch {
608                    column: column.name().to_string(),
609                    table: "POINTING".to_string(),
610                    expected: "Float64 array shaped [2, nPoly+1]".to_string(),
611                    found: format!("{:?}", direction.primitive_type()),
612                });
613            };
614            if coefficient_array.shape() != [2, metadata.polynomial_terms] {
615                return Err(MsError::ColumnTypeMismatch {
616                    column: column.name().to_string(),
617                    table: "POINTING".to_string(),
618                    expected: format!("Float64 array shaped [2, {}]", metadata.polynomial_terms),
619                    found: format!("Float64 array shaped {:?}", coefficient_array.shape()),
620                });
621            }
622            let source_ref = resolve_direction_reference_selected(
623                table,
624                "POINTING",
625                column.name(),
626                metadata.row_index,
627            )?;
628            let coefficient_offset = coefficients.len();
629            coefficients.extend(coefficient_array.iter().copied());
630            catalog_rows.push(CatalogRow {
631                metadata,
632                coefficient_offset,
633                source_ref,
634            });
635        }
636        let retained_bytes = catalog_retained_bytes(&catalog_rows, &coefficients)?;
637        ensure_budget(retained_bytes, plan.maximum_catalog_bytes)?;
638        construction_peak_bytes = construction_peak_bytes.max(
639            retained_bytes
640                .checked_add(row_indices.capacity() * size_of::<usize>())
641                .ok_or_else(byte_overflow)?,
642        );
643        let measurements = SelectedPointingCatalogMeasurements {
644            retained_bytes,
645            construction_peak_bytes,
646            source_rows_scanned: pointing.row_count(),
647            retained_rows: catalog_rows.len(),
648            build_nanos: elapsed_nanos(started),
649        };
650        Ok(PreparedSelectedPointingCatalog {
651            rows: catalog_rows,
652            coefficients,
653            measurements,
654        })
655    }
656}

698fn retain_later(current: &mut Option<CandidateMetadata>, candidate: CandidateMetadata) {
699    if current.is_none_or(|existing| {
700        candidate
701            .time_mjd_seconds
702            .total_cmp(&existing.time_mjd_seconds)
703            .then_with(|| existing.row_index.cmp(&candidate.row_index))
704            .is_gt()
705    }) {
706        *current = Some(candidate);
707    }
708}
709
710fn retain_earlier(current: &mut Option<CandidateMetadata>, candidate: CandidateMetadata) {
711    if current.is_none_or(|existing| {
712        candidate
713            .time_mjd_seconds
714            .total_cmp(&existing.time_mjd_seconds)
715            .then_with(|| candidate.row_index.cmp(&existing.row_index))
716            .is_lt()
717    }) {
718        *current = Some(candidate);
719    }
720}

722fn catalog_upper_bound_bytes(rows: usize, maximum_polynomial_terms: usize) -> MsResult<usize> {
723    let bytes_per_row = maximum_polynomial_terms
724        .checked_mul(2)
725        .and_then(|terms| terms.checked_mul(size_of::<f64>()))
726        .and_then(|bytes| bytes.checked_add(size_of::<CatalogRow>()))
727        .and_then(|bytes| bytes.checked_add(4 * size_of::<usize>()))
728        .ok_or_else(byte_overflow)?;
729    rows.checked_mul(bytes_per_row).ok_or_else(byte_overflow)
730}
731
732fn construction_upper_bound(
733    metadata: &Vec<CandidateMetadata>,
734    row_indices: &Vec<usize>,
735    maximum_polynomial_terms: usize,
736) -> MsResult<usize> {
737    catalog_upper_bound_bytes(metadata.len(), maximum_polynomial_terms)?
738        .checked_add(metadata.capacity() * size_of::<CandidateMetadata>())
739        .and_then(|bytes| bytes.checked_add(row_indices.capacity() * size_of::<usize>()))
740        .and_then(|bytes| bytes.checked_add(metadata.len() * size_of::<Option<ArrayValue>>()))
741        .ok_or_else(byte_overflow)
742}
743
744fn catalog_retained_bytes(rows: &Vec<CatalogRow>, coefficients: &Vec<f64>) -> MsResult<usize> {
745    rows.capacity()
746        .checked_mul(size_of::<CatalogRow>())
747        .and_then(|bytes| bytes.checked_add(coefficients.capacity() * size_of::<f64>()))
748        .ok_or_else(byte_overflow)
749}
750
751fn array_heap_bytes(value: &ArrayValue) -> Option<usize> {
752    let dimensions = value.ndim().checked_mul(2 * size_of::<usize>())?;
753    match value {
754        ArrayValue::Float64(values) => values
755            .len()
756            .checked_mul(size_of::<f64>())?
757            .checked_add(dimensions),
758        _ => Some(dimensions),
759    }
760}
761
762fn ensure_budget(required: usize, available: usize) -> MsResult<()> {
763    if required > available {
764        return Err(MsError::OverBudget {
765            what: "POINTING catalog",
766            required,
767            available,
768        });
769    }
770    Ok(())
771}
772
773fn byte_overflow() -> MsError {
774    MsError::InvalidInput("POINTING catalog byte count overflow".to_string())
775}

777fn elapsed_nanos(started: Instant) -> u64 {
778    u64::try_from(started.elapsed().as_nanos()).unwrap_or(u64::MAX)
779}

781/// Conservative catalog construction peak for content planning.
782/// An upper bound on the bytes preparing a catalog of `rows` retained
783/// `POINTING` rows for `antennas` antennas (two nearest rows each) holds at
784/// any one time, scanning `scan_rows` rows at a time with polynomials of at
785/// most `maximum_polynomial_terms` terms, and the series made from it: the
786/// sum of every budget check [`MeasurementSet::prepare_selected_pointing_catalog`]
787/// makes, so a budget of at least this is never refused. `None` on
788/// overflow.
789#[cfg(unix)]
790pub(crate) fn catalog_peak_bytes(
791    rows: usize,
792    antennas: usize,
793    scan_rows: usize,
794    maximum_polynomial_terms: usize,
795) -> Option<usize> {
796    let rows = rows.checked_add(antennas.checked_mul(2)?)?;
797    let coefficients = 2 * maximum_polynomial_terms * size_of::<f64>();
798    // Metadata, row index, direction cell and its heap, catalog row and its
799    // coefficients, and the series' time, interval, origin and coefficients.
800    let per_row = size_of::<CandidateMetadata>()
801        + size_of::<usize>()
802        + size_of::<Option<ArrayValue>>()
803        + coefficients
804        + 4 * size_of::<usize>()
805        + size_of::<CatalogRow>()
806        + coefficients
807        + 3 * size_of::<f64>()
808        + coefficients;
809    catalog_upper_bound_bytes(rows, maximum_polynomial_terms)
810        .ok()?
811        .checked_add(rows.checked_mul(per_row)?)?
812        .checked_add(scan_rows.checked_mul(SCALAR_SCAN_BYTES_PER_ROW)?)?
813        .checked_add(antennas.checked_mul(size_of::<[Option<CandidateMetadata>; 2]>())?)
814}
