25pub(super) fn solve(
26    ms: &MeasurementSet,
27    request: &GainSolveRequest,
28    mut activity: Option<&mut ToolSolveActivity>,
29    legacy_iterator: bool,
30) -> Result<SolvedGains, GainSolveError> {
31    if matches!(request.refant, RefAntSelector::Unspecified) {
32        return Err(delay_error("specify a reference antenna explicitly"));
33    }
34    validate_solve_interval(request.solve_interval)?;
35    if request.model_source == GainSolveModelSource::PointSource {
36        validate_smodel(request.smodel, true)?;
37    }
38    let cross_hand = request.gain_type == GainType::KCross;
39    let antenna_position = request.gain_type == GainType::KAntPos;
40    let legacy_multiband = legacy_iterator && request.gain_type == GainType::KMbd;
41    let input_phase = crate::timing::Phase::start("gain_select_read_average");
42    let context = prepare_calibration_solve(
43        ms,
44        &request.selection,
45        &request.refant,
46        &request.prior_calibration_tables,
47        request.parang,
48        true,
49        PriorCalibrationMode::Delay {
50            share_input_flags: !request.correlation_dependent_flags,
51            rotate_model: request.parang,
52        },
53    )?;
54    let windows = delay_windows(ms, &context.rows, cross_hand)?;
55    let combined_output_spw = (legacy_iterator && request.combine.spectral_windows)
56        .then(|| context.rows.iter().map(|row| row.data_spw_id).min())
57        .flatten();
58    let groups = build_solve_groups(
59        ms,
60        &context.rows,
61        context.preapplied_rows.as_ref(),
62        &BTreeMap::new(),
63        SolveGroupOptions {
64            input_rows: if legacy_multiband {
65                SolveInputRows::LegacyDelay
66            } else {
67                SolveInputRows::default()
68            },
69            gain_type: request.gain_type,
70            model_source: request.model_source,
71            point_stokes: if request.model_source == GainSolveModelSource::PointSource {
72                request.smodel
73            } else {
74                [1., 0., 0., 0.]
75            },
76            rotate_model: request.parang,
77            solve_interval: request.solve_interval,
78            combine: request.combine,
79            preaverage_max_span: averaging_span(request),
80            // Prior evaluation has already combined raw input flags when
81            // needed. New flags from priors stay per correlation.
82            correlation_dependent_flags: request.correlation_dependent_flags
83                || context.preapplied_rows.is_some(),
84            robust_seed: false,
85            legacy_g_receptors: false,
86            channelized: true,
87            polynomial_bandpass: false,
88        },
89    )?;
90    if groups.is_empty() {
91        return Err(GainSolveError::EmptySelection);
92    }
93    drop(input_phase);
94    let _kernel_phase = crate::timing::Phase::start("gain_solve");
95    let selections = context
96        .rows
97        .iter()
98        .filter_map(|row| {
99            row.channel_selection
100                .as_ref()
101                .map(|selection| (row.data_spw_id, selection.clone()))
102        })
103        .collect::<BTreeMap<_, _>>();
104    let mut transforms = BTreeMap::new();
105    let mut rows = Vec::new();
106    let mut attempted_spw_ids = BTreeSet::new();
107    let parameters = request.gain_type.polarizations();
108    let window_ids = spectral_window_ids(ms)?;
109    if combined_output_spw.is_some()
110        && let Some(activity) = activity.as_deref_mut()
111    {
112        activity.initialize(window_ids.len());
113    }
114    let mut stats = SolveStatsBuilder::new(parameters, window_ids, &context.available_antennas);
115    for mut group in groups.into_values() {
116        group.finalize_for_solve(request.solve_mode, false);
117        let mut stages = IntervalStages::of(&group, 0);
118        if cross_hand || antenna_position {
119            // CASA reports data availability for the full output parameter
120            // shape and bypasses the iterative solver's two thresholds.
121            stages.with_data.resize(
122                parameters,
123                stages.with_data.first().cloned().unwrap_or_default(),
124            );
125        }
126        stages.constrained.clear();
127        stats.interval(&stages);
128        if let Some(output_spw) = combined_output_spw
129            && let Some(activity) = activity.as_deref_mut()
130        {
131            activity.count_mapped_interval(
132                group.spw_id,
133                output_spw,
134                group.attempted,
135                group.attempted,
136            );
137        }
138        if !group.attempted {
139            continue;
140        }
141        let output_spw = combined_output_spw.unwrap_or(group.spw_id);
142        let solutions = group_delays(
143            &group,
144            request,
145            context.refant_id,
146            &windows,
147            &mut transforms,
148            legacy_multiband,
149        )?;
150        attempted_spw_ids.insert(output_spw);
151        stats.record_success(output_spw, true);
152        rows.extend(context.available_antennas.iter().map(|&antenna| {
153            let delays = if cross_hand {
154                solutions.get(&0).copied().unwrap_or_default()
155            } else {
156                solutions.get(&antenna).copied().unwrap_or_default()
157            };
158            let mut gains = Vec::with_capacity(parameters);
159            let mut flags = Vec::with_capacity(parameters);
160            for delay in delays {
161                gains.push(Complex32::new(delay.unwrap_or_default(), 0.));
162                flags.push(delay.is_none());
163            }
164            if antenna_position {
165                // The inherited K solver fits the two parallel hands.
166                // Its reference row defines the unused third term as zero.
167                gains.push(Complex32::ZERO);
168                flags.push(antenna != context.refant_id);
169            }
170            SolutionRow {
171                time_seconds: group.solution_time,
172                field_id: group.field_id,
173                spw_id: output_spw,
174                antenna_id: antenna,
175                scan_number: group.scan_number(),
176                observation_id: group.observation_id,
177                refant_id: -1,
178                gains,
179                flags,
180                param_errors: vec![0.; parameters],
181                snrs: vec![0.; parameters],
182                weights: Vec::new(),
183            }
184        }));
185    }
186    let tool_solve = activity.is_some();
187    if combined_output_spw.is_none() {
188        stats.track_activity(activity);
189    }
190    let mut solve_stats = stats.finish();
191    if !cross_hand && !tool_solve {
192        // task_gaincal.resultsDictGaintypeExcept copies the unflagged
193        // counts at each level, regardless of the actual solved delays.
194        for window in &mut solve_stats.windows {
195            for counts in std::iter::once(&mut window.counts).chain(
196                window
197                    .antennas
198                    .iter_mut()
199                    .map(|antenna| &mut antenna.counts),
200            ) {
201                counts.above_minblperant.clone_from(&counts.data_unflagged);
202                counts.above_minsnr.clone_from(&counts.data_unflagged);
203            }
204        }
205    }
206    Ok(SolvedGains {
207        rows,
208        refant_id: context.refant_id,
209        attempted_spw_ids,
210        channel_selections: selections,
211        selected_data: context.selected_data,
212        solve_stats,
213    })
214}

216#[derive(Clone, Copy)]
217struct DelayWindow {
218    start_ghz: f64,
219    spacing_ghz: f64,
220    lowest_ghz: f64,
221    highest_ghz: f64,
222    channels: usize,
223    correlations: usize,
224}
225
226fn delay_windows(
227    ms: &MeasurementSet,
228    rows: &[SelectedSolveRow],
229    cross_hand: bool,
230) -> Result<BTreeMap<i32, DelayWindow>, GainSolveError> {
231    let spectral_windows = ms.spectral_window().map_err(delay_error)?;
232    let mut windows = BTreeMap::new();
233    for row in rows {
234        let std::collections::btree_map::Entry::Vacant(entry) = windows.entry(row.data_desc_id)
235        else {
236            continue;
237        };
238        let correlations = correlation_types_for_ddid(ms, row.data_desc_id)?.len();
239        if cross_hand && correlations != 4 {
240            return Err(delay_error("KCROSS requires four correlations"));
241        }
242        let frequencies = spectral_windows
243            .chan_freq(row.data_spw_id as usize)
244            .map_err(delay_error)?;
245        let indices = selected_channels(row.channel_selection.as_ref(), frequencies.len())?;
246        let selected = indices
247            .into_iter()
248            .map(|channel| frequencies[channel])
249            .collect::<Vec<_>>();
250        let &[start, next, ..] = selected.as_slice() else {
251            return Err(delay_error("at least two selected channels are needed"));
252        };
253        let spacing_ghz = (next - start) * 1e-9;
254        if spacing_ghz == 0. || !spacing_ghz.is_finite() {
255            return Err(delay_error(
256                "delay channels need a nonzero frequency spacing",
257            ));
258        }
259        entry.insert(DelayWindow {
260            start_ghz: start * 1e-9,
261            spacing_ghz,
262            lowest_ghz: selected.iter().copied().fold(f64::INFINITY, f64::min) * 1e-9,
263            highest_ghz: selected.iter().copied().fold(f64::NEG_INFINITY, f64::max) * 1e-9,
264            channels: selected.len(),
265            correlations,
266        });
267    }
268    Ok(windows)
269}

271type DelaySolutions = BTreeMap<i32, [Option<f32>; 2]>;
272
273fn group_delays(
274    group: &SolveAccumulator,
275    request: &GainSolveRequest,
276    reference: i32,
277    windows: &BTreeMap<i32, DelayWindow>,
278    transforms: &mut BTreeMap<i32, DelayTransform>,
279    legacy: bool,
280) -> Result<DelaySolutions, GainSolveError> {
281    let mut buffers = BTreeMap::<_, Vec<DelaySpectrum<'_>>>::new();
282    for spectrum in group.delay_spectra() {
283        buffers.entry(spectrum.buffer).or_default().push(spectrum);
284    }
285    let cross_hand = request.gain_type == GainType::KCross;
286    let mode = request.solve_mode;
287    if buffers.len() > 1 {
288        if request.gain_type == GainType::KAntPos {
289            return Err(delay_error(
290                "the KAntPos fit cannot combine multiple buffers in a solution interval",
291            ));
292        }
293        if request.gain_type == GainType::KMbd
294            && buffers
295                .keys()
296                .any(|(dd, _, _)| windows[dd].correlations < 2)
297        {
298            return Err(delay_error(
299                "joint KMBD fits need two parallel hands in each visibility buffer",
300            ));
301        }
302        return multiband::solve(
303            buffers
304                .into_iter()
305                .map(|((dd, _, _), spectra)| (windows[&dd], spectra))
306                .collect(),
307            reference,
308            mode,
309            cross_hand,
310            legacy,
311        );
312    }
313    let Some((&(dd, _, _), spectra)) = buffers.first_key_value() else {
314        return Ok(BTreeMap::new());
315    };
316    let transform = match transforms.entry(dd) {
317        std::collections::btree_map::Entry::Occupied(entry) => entry.into_mut(),
318        std::collections::btree_map::Entry::Vacant(entry) => {
319            entry.insert(DelayTransform::new(windows[&dd])?)
320        }
321    };
322    if cross_hand {
323        group_cross_hand(spectra, mode, transform)
324    } else {
325        group_antenna(spectra, reference, mode, transform)
326    }
327}

329fn averaging_span(request: &GainSolveRequest) -> Option<f64> {
330    let preaverage = request.preaverage_seconds as f32;
331    let preaverage = (preaverage > f32::EPSILON && preaverage.is_finite()).then_some(preaverage);
332    let interval = match request.solve_interval {
333        GainSolveInterval::Infinite => preaverage,
334        GainSolveInterval::Integration => None,
335        GainSolveInterval::Seconds(seconds) => {
336            Some(preaverage.map_or(seconds as f32, |pre| pre.min(seconds as f32)))
337        }
338    };
339    interval.map(|seconds| f64::from(seconds) * 0.999)
340}

342fn group_antenna(
343    spectra: &[DelaySpectrum<'_>],
344    reference: i32,
345    mode: GainSolveMode,
346    transform: &mut DelayTransform,
347) -> Result<BTreeMap<i32, [Option<f32>; 2]>, GainSolveError> {
348    let mut results = BTreeMap::from([(reference, [Some(0.); 2])]);
349    for spectrum in spectra.iter().filter(|spectrum| spectrum.parallel_hand) {
350        let ((first, second), receptor) = (spectrum.antennas, spectrum.receptor);
351        let (antenna, sign) = match (first == reference, second == reference) {
352            (true, false) => (second, -1.),
353            (false, true) => (first, 1.),
354            _ => continue,
355        };
356        let values = spectrum
357            .channels()
358            .map(|(value, weight)| constrain(value, weight, mode).0);
359        if let Some(delay) = transform.peak(values, false)? {
360            results.entry(antenna).or_default()[receptor] = Some(sign * delay);
361        }
362    }
363    Ok(results)
364}

366fn group_cross_hand(
367    spectra: &[DelaySpectrum<'_>],
368    mode: GainSolveMode,
369    transform: &mut DelayTransform,
370) -> Result<BTreeMap<i32, [Option<f32>; 2]>, GainSolveError> {
371    let mut sum = vec![(Complex32::default(), 0.); transform.channels];
372    // CASA accumulates both hands of each averaged baseline consecutively.
373    // For a rotated Stokes I model, their opposite signs can cancel exactly.
374    let mut cross_hands = spectra
375        .iter()
376        .filter(|spectrum| !spectrum.parallel_hand)
377        .collect::<Vec<_>>();
378    cross_hands.sort_unstable_by_key(|spectrum| (spectrum.antennas, spectrum.correlation));
379    for spectrum in cross_hands {
380        for ((total, weights), (value, weight)) in sum.iter_mut().zip(spectrum.channels()) {
381            let (value, weight) = constrain(value, weight, mode);
382            let value = if spectrum.correlation == 1 {
383                value
384            } else {
385                value.conj()
386            };
387            *total += weight * value;
388            *weights += weight;
389        }
390    }
391    let values = sum.into_iter().map(|(total, weights)| {
392        if weights > 0. {
393            total / weights
394        } else {
395            Complex32::default()
396        }
397    });
398    let delay = transform.peak(values, true)?;
399    Ok(BTreeMap::from([(0, [delay, delay.map(|_| 0.)])]))
400}
401
402fn constrain(value: Complex32, weight: f32, mode: GainSolveMode) -> (Complex32, f32) {
403    if weight <= 0. {
404        return (Complex32::default(), 0.);
405    }
406    let mean = value / weight;
407    let amplitude = mean.norm();
408    match mode {
409        GainSolveMode::AmplitudePhase => (mean, weight),
410        GainSolveMode::Amplitude => (Complex32::new(amplitude, 0.), weight),
411        GainSolveMode::Phase if amplitude > 0. => {
412            (mean / amplitude, weight * amplitude * amplitude)
413        }
414        GainSolveMode::Phase => (Complex32::default(), 0.),
415    }
416}

418struct DelayTransform {
419    channels: usize,
420    spacing_ghz: f32,
421    fft: Fft2<f32>,
422    padded: Vec<Complex32>,
423}
424
425impl DelayTransform {
426    fn new(window: DelayWindow) -> Result<Self, GainSolveError> {
427        let size = window.channels * 8;
428        Ok(Self {
429            channels: window.channels,
430            spacing_ghz: window.spacing_ghz as f32,
431            fft: Fft2::with_threads([size, 1], 1)
432                .map_err(delay_error)?
433                .with_estimated_plan(),
434            padded: vec![Complex32::default(); size],
435        })
436    }
437
438    fn peak(
439        &mut self,
440        samples: impl Iterator<Item = Complex32>,
441        cross_hand: bool,
442    ) -> Result<Option<f32>, GainSolveError> {
443        let size = self.padded.len();
444        let center = size / 2;
445        let offset = (size - self.channels) / 2;
446        self.padded.fill(Complex32::default());
447        for (out, sample) in self.padded[offset..offset + self.channels]
448            .iter_mut()
449            .zip(samples)
450        {
451            *out = sample;
452        }
453        self.fft
454            .transform(&mut self.padded, false)
455            .map_err(delay_error)?;
456        let amp = |bin: usize| self.padded[(bin + center) % size].norm();
457        // K requires two neighbors and nonzero curvature. KCROSS can use
458        // either endpoint and retains bin zero even when the sum is empty.
459        let edge = usize::from(!cross_hand);
460        let mut peak = edge;
461        let mut middle = 0.;
462        for bin in edge..size - edge {
463            let amplitude = amp(bin);
464            if amplitude > middle {
465                peak = bin;
466                middle = amplitude;
467            }
468        }
469        if !cross_hand && middle == 0. {
470            return Ok(None);
471        }
472        let sub_bin = if peak > 0 && peak + 1 < size {
473            let left = amp(peak - 1);
474            let right = amp(peak + 1);
475            let curvature = left - 2. * middle + right;
476            if curvature == 0. && !cross_hand {
477                return Ok(None);
478            }
479            0.5 - (right - middle) / curvature
480        } else {
481            0.
482        };
483        Ok(Some(
484            (peak as f32 + sub_bin - center as f32) / size as f32 / self.spacing_ghz,
485        ))
486    }
487}
