13pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
14    let path = task.input_path("vis")?;
15    if let Some(parts) = super::multi_ms::partition_paths(&path)? {
16        let mut session = task.session().clone();
17        for part in parts {
18            session
19                .set(
20                    "vis",
21                    ParameterValue::String(part.to_string_lossy().into_owned()),
22                )
23                .map_err(|error| TaskFailure::execution(error.to_string()))?;
24            let partition = TaskInvocation::new(&session, task.workspace());
25            if let Err(error) = run_one(&partition, &part) {
26                tracing::warn!(task = task.surface(), partition = %part.display(), error = %error.message, "Multi-MS partition failed");
27            }
28        }
29    } else {
30        run_one(task, &path)?;
31    }
32    TaskSuccess::new(CasaValue::None, &())
33}
34
35fn run_one(task: &TaskInvocation<'_>, path: &Path) -> Result<(), TaskFailure> {
36    let mut lock = TableWriteLock::acquire(path, 0)
37        .map_err(|error| TaskFailure::execution(error.to_string()).at("vis"))?;
38    let mut ms = MeasurementSet::open(path)
39        .map_err(|error| TaskFailure::execution(error.to_string()).at("vis"))?;
40    let field = task.text("field")?;
41    if task.surface() == "clearcal" {
42        let selection = MsSelection {
43            selectdata: true,
44            field: selectvis_text(field),
45            spw: selectvis_text(task.text("spw")?),
46            intent: selectvis_text(task.text("intent")?),
47            ..MsSelection::default()
48        };
49        lock = reset_scratch_columns(&mut ms, &selection, task.flag("addmodel")?, lock)
50            .map_err(|error| TaskFailure::execution(error.to_string()))?;
51    } else {
52        delete_model(
53            &mut ms,
54            selectvis_text(field).as_deref(),
55            task.flag("otf")?,
56            task.flag("scr")?,
57            &mut lock,
58        )
59        .map_err(|error| TaskFailure::execution(error.to_string()))?;
60    }
61    drop(ms);
62    lock.release()
63        .map_err(|error| TaskFailure::execution(error.to_string()).at("vis"))?;
64    super::history::record_task(task, path);
65    Ok(())
66}

68fn selectvis_text(value: Option<String>) -> Option<String> {
69    let text = value?;
70    let selector = text
71        .strip_prefix('[')
72        .and_then(|inner| inner.strip_suffix(']'))
73        .unwrap_or(&text);
74    (!selector.is_empty()).then(|| selector.to_owned())
75}
