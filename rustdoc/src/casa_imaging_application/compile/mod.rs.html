93/// engine, the source budget, the selection survey and the spectral axis.

95    ms: &'a MeasurementSet,
96    engine: &'a MsCalEngine,
97    budget: SelectedObservationContentBudget,
98    survey: &'a Survey,
99    spectral: &'a PreparedSpectralAxis,

102/// Compile `request` against its MeasurementSet.
103///
104/// What compile reads is admitted, not compile itself: the survey's walk
105/// of the MeasurementSet's rows is admitted once it is open and its rows
106/// are known, before the walk; the tables' descriptions are admitted before
107/// the MeasurementSet is opened. The observation the run
108/// reads is opened here, once ([`open_observation`]).
109pub(crate) fn prepare(
110    request: &ImagingRequest,
111    context: &RunContext,
112) -> Result<Prepared, PrepareError> {
113    let opening = casa_imaging_runtime::admit(
114        &context.host,
115        &context.policy,
116        &casa_imaging_runtime::Demand {
117            phase: "observation opening",
118            memory: Msv2Observation::opening_table_bytes(&request.vis) as u64,
119        },
120    )?;
121    let ms = MeasurementSet::open_retained_read(&request.vis)?;
122    let budget = casa_imaging_runtime::bootstrap_source_budget();
123    // The survey walks the MAIN rows under `budget`.
124    let walks = casa_imaging_runtime::admit(
125        &context.host,
126        &context.policy,
127        &casa_imaging_runtime::Demand {
128            phase: "MeasurementSet walks",
129            memory: ms.row_walk_bytes(budget.row_io_budget())? as u64,
130        },
131    )?;
132    let opened = open_observation(ms, (&context.host, &context.policy), opening)?;
133    let ms = opened.value.measurement_set();
134    // The one Measures provider of the run: compile's geometry and the
135    // batches' both evaluate with it.
136    let measures = casa_ms::open_measures_runtime()?;
137    let engine =
138        MsCalEngine::with_measures(ms, Arc::clone(&measures))?.with_field_ephemerides(ms)?;
139    // An MFS run's spectral envelope is held beside the survey walk and
140    // admitted on its own, against the run's memory.
141    let envelope_bytes = if request.specmode == SpecMode::Mfs {
142        selection::envelope_bytes(ms, context.policy.memory(&context.host))?
143    } else {
144        0
145    };
146    let envelope = casa_imaging_runtime::admit(
147        &context.host,
148        &context.policy,
149        &casa_imaging_runtime::Demand {
150            phase: "spectral envelope",
151            memory: envelope_bytes as u64,
152        },
153    )?;
154    let survey = selection::survey(request, ms, budget, envelope_bytes, &engine)?;
155    drop(envelope);
156    let centre = direction::resolve_centre(request, ms, &survey, &engine, budget)?;
157    let frame = FrameContext {
158        anchor_time_mjd_seconds: survey.first_time_mjd_seconds,
159        time_bounds_mjd_seconds: survey.time_bounds_mjd_seconds,
160        field_id: centre.field_id,
161        phase: centre.phase.clone(),
162        direction: centre.direction,
163        engine: &engine,
164    };
165    let moving_rest_frame = moving_rest_frame(request, &centre, &frame)?;
166    let spectral =
167        spectral::prepare_axis(request, ms, &survey, &frame, moving_rest_frame.as_ref())?;
168    let surveyed = Surveyed {
169        ms,
170        engine: &engine,
171        budget,
172        survey: &survey,
173        spectral: &spectral,
174    };
175    let (continuum_transform, channels) = continuum_selection(request, &surveyed, &centre)?;
176    let observation_info = direction::observation_info(ms, &survey, &centre, &engine)?;
177    let domains = domains::prepare_domains(
178        request,
179        &centre,
180        ImageSpectralCoordinate {
181            frequency_reference: spectral.output_frequency_reference,
182            reference_frequency_hz: spectral.reference_frequency_hz,
183            increment_hz: spectral.increment_hz,
184            rest_frequency_hz: spectral.image_rest_frequency_hz,
185        },
186        &observation_info,
187    )?;
188    let primary_beam = primary_beam_model(request, ms)?;
189    let primary_beam_centres = match primary_beam {
190        None | Some(AnalyticPrimaryBeamModel::MosaicSensitivity) => None,
191        Some(_) => Some(direction::primary_beam_centres(
192            &domains, ms, &survey, &centre, &engine,
193        )?),
194    };
195    let deployment = deployment(
196        request,
197        context,
198        &surveyed,
199        &domains,
200        primary_beam,
201        primary_beam_centres,
202    )?;
203    let instrument = instrument::scientific_instrument_model(&request.gridder, ms)?;
204    let specification = specification::specification(
205        request,
206        &spectral,
207        specification_inputs(
208            request,
209            &surveyed,
210            instrument,
211            primary_beam.is_some(),
212            continuum_transform,
213        )?,
214    )?;
215    let reconstruction_planes = match request.deconvolver {
216        Deconvolver::Mtmfs => request.nterms,
217        _ => spectral.output_channels,
218    };
219    let model_samples = domains::model_samples(&domains, reconstruction_planes, &request.stokes)?;
220    let start_model = request
221        .startmodel
222        .as_deref()
223        .map(|path| {
224            StartModel::prepare(
225                path,
226                &domains[0],
227                request.stokes.len(),
228                reconstruction_planes,
229            )
230        })
231        .transpose()?;
232    let geometry = geometry(request, &domains, &centre, &spectral);
233    let snapshot = snapshot_input(request, &opened.value, survey, &channels)?;
234    // A moving phase centre names a body and a table alike; the table, when
235    // it is one, was read here.
236    let phase_table = centre
237        .ephemeris
238        .as_ref()
239        .and_then(SelectedObservationEphemeris::external_series)
240        .cloned();
241    drop(engine);
242    Ok(Prepared {
243        specification,
244        geometry,
245        model_lifecycle: ModelLifecycleRequirements::new(
246            ModelBounds::new(model_samples, model_samples, f64::MAX, f64::MAX)?,
247            NumericPrecision::F64,
248        ),
249        masks: domains::mask_plans(&domains)?,
250        start_model,
251        minor_cycle_image_response: match &request.gridder {
252            Gridder::Mosaic { .. } | Gridder::Awproject(_) => Some(MinorCycleImageResponse::new(
253                specification::normalization(request),
254                specification::primary_beam_validity(request)?,
255            )?),
256            Gridder::Standard { .. } | Gridder::Wproject { .. } => None,
257        },
258        snapshot,
259        observation: OpenedObservation {
260            store: opened,
261            measures,
262            phase_table,
263        },
264        write_model_column: request.savemodel,
265        write_corrected_data: request.save_continuum_residual,
266        deployment,
267        walks,
268    })
269}

271/// The rest frame of a moving source (`cubesource`): the observatory frame
272/// at the anchor time with the ephemeris's radial velocity.
273fn moving_rest_frame(
274    request: &ImagingRequest,
275    centre: &Centre,
276    frame: &FrameContext<'_>,
277) -> Result<Option<MeasFrame>, PrepareError> {
278    if request.specmode != SpecMode::Cubesource {
279        return Ok(None);
280    }
281    let ephemeris = centre
282        .ephemeris
283        .as_ref()
284        .ok_or(PrepareError::MovingPhaseCentre)?;
285    let velocity = frame.engine.ephemeris_radial_velocity(
286        frame.anchor_time_mjd_seconds,
287        frame.field_id,
288        ephemeris,
289    )?;
290    Ok(Some(
291        frame
292            .engine
293            .spectral_frame_observatory_direction(
294                frame.anchor_time_mjd_seconds,
295                centre.phase.clone(),
296            )?
297            .with_radial_velocity(velocity),
298    ))
299}

301/// The continuum transform of a `fitspw` request and the source channels
302/// the observation reads: the fit channels with the output channels, or
303/// the output channels alone.
304fn continuum_selection(
305    request: &ImagingRequest,
306    surveyed: &Surveyed<'_>,
307    centre: &Centre,
308) -> Result<(Option<SequentialContinuumTransform>, WindowChannels), PrepareError> {
309    let Some(fitspw) = &request.fitspw else {
310        return Ok((None, surveyed.spectral.selected_source_channels.clone()));
311    };
312    let window = selection::one_window(&surveyed.survey.spectral_windows, "continuum subtraction")?;
313    let (transform, selected) = specification::continuum_transform(
314        fitspw,
315        request.fitorder,
316        i32::try_from(centre.field_id).expect("the centre's field id is a stored i32 FIELD_ID"),
317        window,
318        surveyed.spectral.window_channels(window.spw_id),
319    )?;
320    Ok((Some(transform), BTreeMap::from([(window.spw_id, selected)])))
321}

323/// The specification's inputs beyond the request: the instrument, the
324/// W and AW projections over the observed W range, the cube's density
325/// padding and the continuum transform.
326fn specification_inputs(
327    request: &ImagingRequest,
328    surveyed: &Surveyed<'_>,
329    instrument: Option<casa_imaging_model::InstrumentModel>,
330    primary_beam: bool,
331    continuum_transform: Option<SequentialContinuumTransform>,
332) -> Result<SpecificationInputs, PrepareError> {
333    let windows = &surveyed.survey.spectral_windows;
334    Ok(SpecificationInputs {
335        instrument,
336        primary_beam,
337        w_projection: match &request.gridder {
338            Gridder::Wproject { wprojplanes } => Some(specification::w_projection(
339                *wprojplanes,
340                windows,
341                surveyed.survey.w,
342            )?),
343            _ => None,
344        },
345        aw_projection: match &request.gridder {
346            Gridder::Awproject(aw) => Some(specification::aw_projection(
347                aw,
348                windows,
349                surveyed.survey.w,
350            )?),
351            _ => None,
352        },
353        cube_density_padding: cube_density_padding(request, surveyed)?,
354        continuum_transform,
355    })
356}

358/// The selected observation's manifest: the survey's rows, the channels
359/// read with each selected window's whole frequency axis, the visibility
360/// column, and the weights the selected partitions hold (each partition's
361/// by CASA's rule, [`partitions::stored_weights`]).
362fn snapshot_input(
363    request: &ImagingRequest,
364    observation: &Msv2Observation,
365    survey: Survey,
366    channels: &WindowChannels,
367) -> Result<ObservationSnapshotInput, PrepareError> {
368    let ms = observation.measurement_set();
369    let visibility = selection::visibility_column(ms, request.datacolumn)?;
370    let weights = partitions::stored_weights(observation, &survey.row_selection, visibility);
371    let selection = selection::observation_selection(ms, survey, channels)?;
372    Ok(ObservationSnapshotInput::new(vec![
373        ObservationSourceInput::new(
374            ObservationSourceProvenance::new(request.vis.display().to_string()),
375            selection,
376            SelectedColumns::new(visibility, FlagPolicy::FlagOrFlagRow, weights),
377            ms.data_column(VisibilityDataColumn::CorrectedData).is_ok(),
378        ),
379    ]))
380}

382/// The image geometry: the domains, the centre laws and the spectral axis.
383fn geometry(
384    request: &ImagingRequest,
385    domains: &[PreparedImageDomain],
386    centre: &Centre,
387    spectral: &PreparedSpectralAxis,
388) -> GeometryInput {
389    let direction_dependent = matches!(
390        request.gridder,
391        Gridder::Mosaic { .. } | Gridder::Awproject(_)
392    );
393    GeometryInput::new(
394        domains.iter().map(PreparedImageDomain::spec).collect(),
395        CentreLaws::new(
396            centre.law.clone(),
397            instrument::pointing_centre_law(&request.gridder),
398        ),
399        if direction_dependent {
400            UvwCoordinateLaw::MosaicPhaseTrackingCentre
401        } else {
402            UvwCoordinateLaw::PhaseTrackingCentre
403        },
404        spectral.coordinate(),
405    )
406}

408/// The primary beam the products are written with: the mosaic
409/// sensitivity of a direction-dependent gridder, else the telescope's
410/// analytic beam when `.pb` or `.image.pbcor` is asked for.
411fn primary_beam_model(
412    request: &ImagingRequest,
413    ms: &MeasurementSet,
414) -> Result<Option<AnalyticPrimaryBeamModel>, PrepareError> {
415    Ok(match request.gridder {
416        Gridder::Mosaic { .. } | Gridder::Awproject(_) => {
417            Some(AnalyticPrimaryBeamModel::MosaicSensitivity)
418        }
419        Gridder::Standard { .. } | Gridder::Wproject { .. }
420            if request.write_pb || request.pbcor =>
421        {
422            Some(instrument::standard_primary_beam_model(ms)?)
423        }
424        Gridder::Standard { .. } | Gridder::Wproject { .. } => None,
425    })
426}

428/// CASA's padded per-channel density grid of a linearly interpolated
429/// cube with per-channel density.
430fn cube_density_padding(
431    request: &ImagingRequest,
432    surveyed: &Surveyed<'_>,
433) -> Result<Option<usize>, PrepareError> {
434    let (survey, spectral) = (surveyed.survey, surveyed.spectral);
435    if !specification::cube_density_padded(request, spectral) {
436        return Ok(None);
437    }
438    let window = selection::one_window(&survey.spectral_windows, "cube density")?;
439    Ok(Some(
440        surveyed.ms.selected_observation_cube_density_padding(
441            &survey.row_selection,
442            SelectedObservationSpectralWindow::borrow_selected(
443                u32::try_from(window.spw_id).expect("SPW ids are nonnegative stored i32 values"),
444                window.frequency_reference,
445                &window.frequencies_hz,
446                &window.channel_widths_hz,
447                spectral.window_channels(window.spw_id),
448            ),
449            survey.fields.iter().copied(),
450            spectral.output_frequency_reference,
451            [
452                spectral.reference_frequency_hz,
453                spectral.reference_frequency_hz
454                    + (spectral.output_channels - 1) as f64 * spectral.increment_hz,
455            ],
456            spectral.output_channels,
457            surveyed.engine,
458            surveyed.budget.row_io_budget(),
459        )?,
460    ))
461}

463/// The native run's runtime, product publication and AW catalog, still to
464/// be deployed: no directory exists and no AW cell is generated until
465/// [`Deployment::deploy`].
466pub(crate) struct Deployment {
467    runtime: PendingRuntime,
468    publication: ApplicationPublication,
469    /// Every domain's output directory.
470    output_directories: Vec<PathBuf>,
471    aw: Option<AwPlan>,
472}

474/// The run's runtime but for its device, which the compiled problem resolves
475/// ([`Deployment::deploy`]); its spill directory is made canonical on
476/// deployment.
477struct PendingRuntime {
478    host: crate::HostResources,
479    resource_policy: crate::ResourcePolicy,
480    grid_precision: Option<crate::GridPrecision>,
481    cancel: crate::Cancel,
482    spill_directory: PathBuf,
483    diagnostics: crate::Diagnostics,
484}

486/// The AW catalog a run opens and, for a native cache, the cells to bring
487/// to the request's policy.
488struct AwPlan {
489    root: PathBuf,
490    native: Option<(NativeAwCachePolicy, NativeAwRequestInput)>,
491    indexing: AwIndexing,
492    resident_bytes: usize,
493    /// Whether the run raises the bound into the memory its workers leave
494    /// (`cf_resident_mb = auto`).
495    automatic: bool,
496}
497
498/// The cell cache's bound for `cf_resident_mb = auto` before the run raises
499/// it into the memory its workers leave (#449).
500const AUTOMATIC_CELL_CACHE_FLOOR: usize = 256 << 20;
501
502impl Deployment {
503    /// Create every domain's output directory, settle the spill directory
504    /// beside the main image, bring a native AW cache to the request's
505    /// policy and grid on `backend`, the device the request resolved to on
506    /// the compiled problem ([`crate::availability::resolve_backend`]).
507    pub(crate) fn deploy(self, backend: BackendChoice) -> Result<ApplicationNative, PrepareError> {
508        for directory in &self.output_directories {
509            std::fs::create_dir_all(directory).map_err(|source| PrepareError::OutputDirectory {
510                path: directory.clone(),
511                source,
512            })?;
513        }
514        let PendingRuntime {
515            host,
516            resource_policy,
517            grid_precision,
518            cancel,
519            spill_directory,
520            diagnostics,
521        } = self.runtime;
522        let spill_directory =
523            spill_directory
524                .canonicalize()
525                .map_err(|source| PrepareError::OutputDirectory {
526                    path: spill_directory.clone(),
527                    source,
528                })?;
529        Ok(ApplicationNative {
530            runtime: ApplicationRuntime {
531                host,
532                resource_policy,
533                backend,
534                grid_precision,
535                cancel,
536                spill_directory,
537                diagnostics,
538            },
539            publication: self.publication,
540            aw_catalog: self
541                .aw
542                .map(|aw| aw.deploy((&host, &resource_policy)))
543                .transpose()?,
544        })
545    }
546}

548impl AwPlan {
549    /// Under every policy the directory is compared with the request: its
550    /// request manifest, the cells the request names, and any it does not
551    /// (#735: a cell's conjugate beam and its lookup depend on the whole
552    /// frequency list). Reuse needs a directory generated for this request,
553    /// complete and holding nothing else. Generation fills in the absent
554    /// cells of such a directory, and clears and regenerates any other.
555    /// Regeneration clears the cache first. The loader then checks each
556    /// cell's sky increment against the image (plan section 5.6).
557    ///
558    /// The run reads a native cache under its shared [`NativeCacheLock`],
559    /// held in the deployment for as long as the run lives, since cells load
560    /// lazily. It writes one only under the exclusive lock, which it cannot
561    /// take while another run reads the cache. After writing, it takes the
562    /// shared lock again and checks the directory once more before it keeps
563    /// it (#739 review). Generation is admitted on `admission`'s host and
564    /// policy before it starts and released when it ends (#738).
565    fn deploy(
566        self,
567        admission: (&crate::HostResources, &crate::ResourcePolicy),
568    ) -> Result<AwCatalogDeployment, PrepareError> {
569        let (lock, generation_workers) = match &self.native {
570            Some((policy, input)) => {
571                let (lock, workers) = self.bring_native(*policy, input, admission)?;
572                (Some(Arc::new(lock)), workers)
573            }
574            None => (None, None),
575        };
576        Ok(AwCatalogDeployment {
577            root: self.root,
578            indexing: self.indexing,
579            resident_bytes: self.resident_bytes,
580            automatic: self.automatic,
581            _lock: lock,
582            generation_workers,
583        })
584    }
585
586    /// Bring the native cache to `policy` for `input` and return the shared
587    /// lock the run keeps while it reads it ([`Self::deploy`]), and the
588    /// workers that generated cells, if any did.
589    fn bring_native(
590        &self,
591        policy: NativeAwCachePolicy,
592        input: &NativeAwRequestInput,
593        (host, resources): (&crate::HostResources, &crate::ResourcePolicy),
594    ) -> Result<(NativeCacheLock, Option<usize>), PrepareError> {
595        let root = &self.root;
596        let shared = NativeCacheLock::shared(root)?;
597        let state = AwCatalog::native_cache_state(root, input)?;
598        let fill = match policy {
599            NativeAwCachePolicy::ReuseOnly if state.reusable() => return Ok((shared, None)),
600            NativeAwCachePolicy::ReuseOnly => {
601                return Err(PrepareError::AwCacheNotReusable { state });
602            }
603            NativeAwCachePolicy::GenerateMissing if state.reusable() => return Ok((shared, None)),
604            NativeAwCachePolicy::GenerateMissing if state.fillable() => true,
605            NativeAwCachePolicy::GenerateMissing => {
606                tracing::warn!(
607                    "native AW cache {} does not belong to this request ({state}); \
608                     regenerating it",
609                    root.display()
610                );
611                false
612            }
613            NativeAwCachePolicy::Regenerate => false,
614        };
615        drop(shared);
616        let exclusive = NativeCacheLock::exclusive(root)?;
617        // The most workers, up to the policy's and one per cell to generate,
618        // whose workspaces fit what the policy leaves free (#708); one when
619        // none does, which admission then refuses. Generation gives each of
620        // them a unit (#747 review).
621        let cells = if fill {
622            state.expected - state.present
623        } else {
624            state.expected
625        };
626        let free = casa_imaging_runtime::free_memory(host, resources);
627        let workers = (1..=resources.workers(host).min(cells))
628            .rev()
629            .find(|&workers| AwCatalog::generation_bytes(input, workers) <= free)
630            .unwrap_or(1);
631        let generating = casa_imaging_runtime::admit(
632            host,
633            resources,
634            &casa_imaging_runtime::Demand {
635                phase: "native AW generation",
636                memory: AwCatalog::generation_bytes(input, workers),
637            },
638        )?;
639        tracing::info!(
640            "generating {cells} native AW cells into {} on {workers} workers",
641            root.display()
642        );
643        AwCatalog::generate_native(root, input, fill, workers)?;
644        drop(generating);
645        drop(exclusive);
646        let shared = NativeCacheLock::shared(root)?;
647        let state = AwCatalog::native_cache_state(root, input)?;
648        if !state.reusable() {
649            // Another run replaced the cache between our write and our read.
650            return Err(PrepareError::AwCacheNotReusable { state });
651        }
652        Ok((shared, Some(workers)))
653    }
654}

656/// The native run's deployment: the runtime with the paged cube state
657/// beside the main image, the product controls and image writer, and the AW
658/// plan.
659fn deployment(
660    request: &ImagingRequest,
661    context: &RunContext,
662    surveyed: &Surveyed<'_>,
663    domains: &[PreparedImageDomain],
664    primary_beam: Option<AnalyticPrimaryBeamModel>,
665    primary_beam_centres: Option<PrimaryBeamCentres>,
666) -> Result<Deployment, PrepareError> {
667    let parent = |path: &Path| {
668        path.parent()
669            .unwrap_or_else(|| Path::new("."))
670            .to_path_buf()
671    };
672    let mut controls = ContinuumProductControls::new(request.psfcutoff as f32).map_err(|_| {
673        PrepareError::PsfCutoff {
674            psfcutoff: request.psfcutoff,
675        }
676    })?;
677    if let Some(model) = primary_beam {
678        controls = controls.with_primary_beam_model(model);
679    }
680    Ok(Deployment {
681        runtime: PendingRuntime {
682            host: context.host,
683            resource_policy: context.policy,
684            grid_precision: request.gridprecision,
685            cancel: context.cancel.clone(),
686            spill_directory: parent(&request.imagename),
687            diagnostics: request.diagnostics,
688        },
689        publication: ApplicationPublication {
690            controls,
691            primary_beam_centres,
692            writer: CasaImageWriter::for_domains(domains.iter().map(PreparedImageDomain::output)),
693        },
694        output_directories: domains
695            .iter()
696            .map(|domain| parent(&domain.output))
697            .collect(),
698        aw: match &request.gridder {
699            Gridder::Awproject(aw) => Some(aw_plan(request, aw, surveyed)?),
700            _ => None,
701        },
702    })
703}

705/// The AW catalog to open: a CASA cache read as it is, or a native cache
706/// in CASA's format with the input its cells are generated from.
707fn aw_plan(
708    request: &ImagingRequest,
709    aw: &AwProjection,
710    surveyed: &Surveyed<'_>,
711) -> Result<AwPlan, PrepareError> {
712    let (root, native) = match &aw.cf_source {
713        AwCfSource::CasaImport { cfcache } => (cfcache.clone(), None),
714        AwCfSource::NativeEvla {
715            native_cf_cache,
716            native_cf_policy,
717            ..
718        } => {
719            let first_row = surveyed
720                .survey
721                .first_cross_row
722                .ok_or(PrepareError::NoCrossCorrelationRow)?;
723            let input = native_aw::resolve(request, aw, surveyed, first_row)?;
724            input.validate()?;
725            (native_cf_cache.clone(), Some((*native_cf_policy, input)))
726        }
727    };
728    Ok(AwPlan {
729        root,
730        native,
731        indexing: AwIndexing {
732            conjugate_beams: true,
733            image_reference_hz: surveyed.spectral.reference_frequency_hz,
734        },
735        resident_bytes: match aw.cf_resident_mb {
736            Some(megabytes) => megabytes
737                .checked_mul(1 << 20)
738                .ok_or(PrepareError::CfResidentSize { megabytes })?,
739            None => AUTOMATIC_CELL_CACHE_FLOOR,
740        },
741        automatic: aw.cf_resident_mb.is_none(),
742    })
743}
