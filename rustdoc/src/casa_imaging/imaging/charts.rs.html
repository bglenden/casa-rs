2//! Faceted imaging (#664): CASA's `facets=N` splits the main image into N×N
3//! facets, each gridded about its own centre by a mapper of its own
4//! (`SynthesisImager::createFacetImageStoreList`, `SIMapperCollection`).
5//! Here every facet is a *chart*: an operator of the facet's size, a pass
6//! domain of the one traversal, onto which casa-ms projects each row with
7//! CASA's `UVWMachine(project=true)`. The pass sees charts; the normal state,
8//! the minor cycle and the products see the compiled image domains, whose
9//! images [`ChartStitch`] assembles from their charts.
10

21/// One operator's share of an image domain: a facet of a faceted domain, or
22/// the whole domain.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub(crate) struct Chart {
25    /// The compiled image domain.
26    pub(crate) domain: usize,
27    /// The facet's pixels when the domain is faceted.
28    pub(crate) window: Option<FacetBox>,
29}
30
31/// A facet's pixels in its domain: `origin` inclusive, `end` exclusive.
32#[derive(Clone, Copy, Debug, PartialEq, Eq)]
33pub(crate) struct FacetBox {
34    pub(crate) origin: [usize; 2],
35    pub(crate) end: [usize; 2],
36}
37
38impl FacetBox {
39    /// The facet's `[width, height]`.
40    pub(crate) const fn shape(self) -> [usize; 2] {
41        [self.end[0] - self.origin[0], self.end[1] - self.origin[1]]
42    }
43}
44
45/// The charts of `problem` in the order its rows carry their projections:
46/// domain-major, a faceted domain's facets in CASA's order (facet
47/// `row·N + column`, facet 0 at the bottom left).
48pub(crate) fn charts(problem: &CompiledProblem) -> Vec<Chart> {
49    problem
50        .geometry()
51        .domains()
52        .iter()
53        .enumerate()
54        .flat_map(|(domain, compiled)| {
55            let faceted = compiled.facets().len() > 1;
56            compiled.facets().iter().map(move |window| Chart {
57                domain,
58                window: faceted.then_some(FacetBox {
59                    origin: window.origin(),
60                    end: window.end_exclusive(),
61                }),
62            })
63        })
64        .collect()
65}
66
67impl Chart {
68    /// The extent of the chart's operator: the domain's, or the facet's,
69    /// whose reference pixel is its centre (CASA grids a facet about the
70    /// world direction of its pixel `shape / 2`, `FTMachine::initMaps`).
71    pub(crate) fn extent(
72        &self,
73        compiled: &CompiledImageDomain,
74    ) -> Result<ImageExtent, ImagingError> {
75        let Some(window) = self.window else {
76            return image_extent(compiled);
77        };
78        let shape = window.shape();
79        Ok(ImageExtent {
80            shape,
81            increment_rad: compiled.direction().increment_rad(),
82            // `FacetWindow::local_centre_pixel`: the facet's phase centre.
83            reference_pixel: [shape[0] / 2, shape[1] / 2],
84        })
85    }
86}

88/// One facet's sums of weights per (term, polarization), from the initial
89/// pass.
90#[derive(Clone, Debug)]
91struct FacetSums {
92    /// The PSF gridding's, which normalise the PSF.
93    psf: Vec<f64>,
94    /// CASA's `.sumwt`, which normalises the facet's residual.
95    published: Vec<f64>,
96}
97
98impl FacetSums {
99    fn populated(&self, weights: &Range<usize>, polarizations: usize) -> bool {
100        // CASA normalizes every Taylor residual by sumwt.tt0. The higher
101        // moments are signed and may vanish even on a populated facet.
102        self.published[weights.start..weights.start + polarizations]
103            .iter()
104            .all(|sum| casa_divides_plane(*sum))
105    }
106}

108/// A faceted domain's current wave of joined images. Each channel keeps
109/// the lowest populated facet seen so far as its PSF reference.
110struct Assembly {
111    images: PassImages,
112    arrived: usize,
113    references: Vec<Option<usize>>,
114}

116/// Why a faceted domain's images could not be stitched; each is a broken
117/// invariant of the pass, not a property of the data.
118#[derive(Debug, thiserror::Error)]
119pub(crate) enum StitchError {
120    /// A facet's residual arrived before its initial pass had given its
121    /// sums of weights.
122    #[error("facet {chart}'s residual arrived before its initial pass's sums of weights")]
123    UnscaledFacet { chart: usize },
124    /// The domain's facets disagree on their polarizations or planes.
125    #[error("facet {chart} does not match its domain's other facets")]
126    MismatchedFacet { chart: usize },
127}

129/// The images a pass hands the normal state: an unfaceted domain's as its
130/// chart forms them, a faceted domain's stitched from its facets as CASA's
131/// normalizer reads them through each facet store, independently per channel
132/// (`SynthesisNormalizer`), with a populated reference facet `r`, the lowest
133/// whose zeroth `sumwt` exceeds 1e-7 in every polarization (facet 0 unless
134/// it is empty):
135/// - each facet's residual divided by that facet's own zeroth `sumwt`, here by
136///   scaling facet `f` by `s_r / s_f` before the normal state divides the
137///   whole by `s_r`; an empty facet is zero, as `divideImageByWeightVal`
138///   leaves it;
139/// - facet `r`'s PSF moments, placed at the image centre in images of zeros
140///   (`setPsfFromOneFacet`, which takes facet 0), which the reconstruction
141///   normalises by the zeroth PSF's peak;
142/// - facet `r`'s sums of weights for the domain.
143///
144/// Each facet's own sums are published in its pixel of `.sumwt`
145/// ([`ChartStitch::facet_sum_weights`]).
146pub(crate) struct ChartStitch {
147    charts: Vec<Chart>,
148    /// Each image domain's `[width, height]`.
149    shapes: Vec<[usize; 2]>,
150    /// Each chart's sums, from the initial pass.
151    sums: Vec<Option<FacetSums>>,
152    assemblies: BTreeMap<usize, Assembly>,
153    /// The requested polarizations, one image each per plane and term.
154    pols: usize,
155    coefficients: usize,
156    moments: usize,
157    channels: usize,
158}
159
160impl ChartStitch {
161    /// The stitch of `problem`'s `charts`.
162    pub(crate) fn new(problem: &CompiledProblem, charts: Vec<Chart>) -> Self {
163        let shapes = problem
164            .geometry()
165            .domains()
166            .iter()
167            .map(|domain| domain.shape().pixels())
168            .collect();
169        let sums = vec![None; charts.len()];
170        let (coefficients, moments, channels) = match problem.reconstruction().basis() {
171            ReconstructionBasis::Taylor { terms } => (terms, 2 * terms - 1, 1),
172            ReconstructionBasis::ChannelLocal { channels } => (1, 1, channels),
173            ReconstructionBasis::Constant => (1, 1, 1),
174        };
175        Self {
176            charts,
177            shapes,
178            sums,
179            assemblies: BTreeMap::new(),
180            pols: problem.reconstruction().polarization().coordinates().len(),
181            coefficients,
182            moments,
183            channels,
184        }
185    }
186
187    /// Bytes beside the pass's own images, per channel in its wave: each
188    /// faceted domain's full residual, initial PSF and reference index.
189    pub(crate) fn pass_bytes_per_channel(&self, psf: bool) -> u64 {
190        let images = self.coefficients + if psf { self.moments } else { 0 };
191        self.charts
192            .iter()
193            .enumerate()
194            .filter(|(chart, candidate)| {
195                candidate.window.is_some() && self.first_chart(candidate.domain) == *chart
196            })
197            .map(|(_, chart)| {
198                let [width, height] = self.shapes[chart.domain];
199                (width as u64 * height as u64 * self.pols as u64 * images as u64)
200                    * size_of::<f32>() as u64
201                    + size_of::<Option<usize>>() as u64
202            })
203            .sum()
204    }
205
206    /// Weight vectors kept for each facet across all waves, including the
207    /// copy of the published weights handed on to product generation.
208    pub(crate) fn weight_bytes(&self) -> u64 {
209        let facets = self
210            .charts
211            .iter()
212            .filter(|chart| chart.window.is_some())
213            .count();
214        (facets * self.channels * self.moments * self.pols * 3 * size_of::<f64>()) as u64
215    }
216
217    /// Hold the weight reservation through product publication.
218    pub(crate) fn reserve_weights(
219        &self,
220        host: &HostResources,
221        policy: &ResourcePolicy,
222    ) -> Result<Reservation, ImagingError> {
223        let memory = self.weight_bytes();
224        if memory == 0 {
225            return Ok(Reservation::none());
226        }
227        Ok(admit(
228            host,
229            policy,
230            &Demand {
231                phase: "facet weights",
232                memory,
233            },
234        )?)
235    }
236
237    fn channel_weights(&self, channel: usize) -> Range<usize> {
238        let stride = self.moments * self.pols;
239        channel * stride..(channel + 1) * stride
240    }
241
242    /// Each faceted domain's facets' published sums of weights, per facet
243    /// in CASA's order, then by channel, moment and polarization:
244    /// CASA's `.sumwt` of a faceted image holds each facet store's own.
245    pub(crate) fn facet_sum_weights(&self) -> BTreeMap<usize, Vec<Vec<f64>>> {
246        let mut domains = BTreeMap::<usize, Vec<Vec<f64>>>::new();
247        for (chart, candidate) in self.charts.iter().enumerate() {
248            if candidate.window.is_none() {
249                continue;
250            }
251            if let Some(sums) = &self.sums[chart] {
252                domains
253                    .entry(candidate.domain)
254                    .or_default()
255                    .push(sums.published.clone());
256            }
257        }
258        domains
259    }
260
261    fn first_chart(&self, domain: usize) -> usize {
262        self.charts
263            .iter()
264            .position(|candidate| candidate.domain == domain)
265            .expect("a chart of the domain")
266    }

268    /// Accept chart `chart`'s images; the domain's images once its last
269    /// chart has arrived, at once for an unfaceted domain.
270    pub(crate) fn accept(
271        &mut self,
272        chart: usize,
273        mut images: PassImages,
274    ) -> Result<Option<PassImages>, StitchError> {
275        let Chart { domain, window } = self.charts[chart];
276        images.domain = domain;
277        let Some(window) = window else {
278            return Ok(Some(images));
279        };
280        let wave_start = images.channels.start;
281        let wave_end = images.channels.end;
282        let channels = images.channels.len();
283        let moments = self.channel_weights(wave_start).start..self.channel_weights(wave_end).start;
284        if images.psf.is_some() {
285            let length = self.channel_weights(self.channels).start;
286            let sums = self.sums[chart].get_or_insert_with(|| FacetSums {
287                psf: vec![0.0; length],
288                published: vec![0.0; length],
289            });
290            sums.psf[moments.clone()].copy_from_slice(&images.sum_weights);
291            sums.published[moments].copy_from_slice(&images.published_sum_weights);
292        }
293        let sums = self.sums[chart]
294            .as_ref()
295            .ok_or(StitchError::UnscaledFacet { chart })?;
296        let full = self.shapes[domain];
297        let assembly = self.assemblies.entry(domain).or_insert_with(|| Assembly {
298            images: empty_domain_images(&images, full),
299            arrived: 0,
300            references: vec![None; channels],
301        });
302        if assembly.images.channels != images.channels
303            || assembly.images.residual.len() / (full[0] * full[1])
304                != images.residual.len() / (images.shape[0] * images.shape[1])
305        {
306            return Err(StitchError::MismatchedFacet { chart });
307        }
308        let facet = window.shape();
309        scatter(
310            &images.residual,
311            &mut assembly.images.residual,
312            facet,
313            full,
314            window.origin,
315        );
316        let psf_planes = self.pols * self.moments;
317        for (local, reference) in assembly.references.iter_mut().enumerate() {
318            let start = (wave_start + local) * psf_planes;
319            if reference.is_some_and(|first| first < chart)
320                || !sums.populated(&(start..start + psf_planes), self.pols)
321            {
322                continue;
323            }
324            *reference = Some(chart);
325            if let (Some(psf), Some(stitched)) = (&images.psf, &mut assembly.images.psf) {
326                let input = psf_planes * facet[0] * facet[1];
327                let output = psf_planes * full[0] * full[1];
328                let centred = [(full[0] - facet[0]) / 2, (full[1] - facet[1]) / 2];
329                scatter(
330                    &psf[local * input..(local + 1) * input],
331                    &mut stitched[local * output..(local + 1) * output],
332                    facet,
333                    full,
334                    centred,
335                );
336            }
337        }
338        assembly.arrived += 1;
339        let facets = self
340            .charts
341            .iter()
342            .filter(|candidate| candidate.domain == domain)
343            .count();
344        if assembly.arrived < facets {
345            return Ok(None);
346        }
347        let assembly = self
348            .assemblies
349            .remove(&domain)
350            .expect("the domain's assembly");
351        self.finish(domain, assembly).map(Some)
352    }
353
354    /// Scale each facet's window of a complete assembly by `s_r / s_f`
355    /// (zero for an empty facet) and give each channel its reference weights.
356    fn finish(&self, domain: usize, mut assembly: Assembly) -> Result<PassImages, StitchError> {
357        let full = self.shapes[domain];
358        let pols = self.pols;
359        let coefficients = self.coefficients;
360        let channels = assembly.images.channels.clone();
361        let first = self.first_chart(domain);
362        let reference = |channel: usize| {
363            let chart = assembly.references[channel].unwrap_or(first);
364            self.sums[chart]
365                .as_ref()
366                .expect("the reference facet arrived")
367        };
368        for (chart, candidate) in self.charts.iter().enumerate() {
369            if candidate.domain != domain {
370                continue;
371            }
372            let sums = self.sums[chart]
373                .as_ref()
374                .ok_or(StitchError::UnscaledFacet { chart })?;
375            let window = candidate
376                .window
377                .expect("a faceted domain's chart has a window");
378            rescale(&mut assembly.images.residual, window, full, |image| {
379                let channel = image / (pols * coefficients);
380                let index = self.channel_weights(channels.start + channel).start + image % pols;
381                let own = sums.published[index];
382                if casa_divides_plane(own) {
383                    reference(channel).published[index] / own
384                } else {
385                    0.0
386                }
387            });
388        }
389        if assembly.images.psf.is_some() {
390            for (local, channel) in channels.enumerate() {
391                let weights = self.channel_weights(channel);
392                let sums = reference(local);
393                assembly
394                    .images
395                    .sum_weights
396                    .extend_from_slice(&sums.psf[weights.clone()]);
397                assembly
398                    .images
399                    .published_sum_weights
400                    .extend_from_slice(&sums.published[weights]);
401            }
402        }
403        Ok(assembly.images)
404    }
405}

407/// Scale `window` of each x-major `full` image in `images` by `scale(i)`.
408fn rescale(images: &mut [f32], window: FacetBox, full: [usize; 2], scale: impl Fn(usize) -> f64) {
409    let full_cells = full[0] * full[1];
410    let height = window.shape()[1];
411    for (image, values) in images.chunks_exact_mut(full_cells).enumerate() {
412        let factor = scale(image);
413        for x in window.origin[0]..window.end[0] {
414            let start = x * full[1] + window.origin[1];
415            for cell in &mut values[start..start + height] {
416                *cell = (f64::from(*cell) * factor) as f32;
417            }
418        }
419    }
420}

422/// Zeroed images of a whole `full` domain shaped like the facet `images`.
423fn empty_domain_images(images: &PassImages, full: [usize; 2]) -> PassImages {
424    let facet_cells = images.shape[0] * images.shape[1];
425    let full_cells = full[0] * full[1];
426    let zeros = |values: &[f32]| vec![0.0; values.len() / facet_cells * full_cells];
427    PassImages {
428        domain: images.domain,
429        shape: full,
430        channels: images.channels.clone(),
431        polarizations: images.polarizations,
432        residual: zeros(&images.residual),
433        psf: images.psf.as_deref().map(zeros),
434        sum_weights: Vec::new(),
435        published_sum_weights: Vec::new(),
436        weight: None,
437    }
438}

440/// Copy each x-major `facet` image of `from` into the x-major `full` image
441/// of `into` at pixel `origin`.
442fn scatter(
443    from: &[f32],
444    into: &mut [f32],
445    facet: [usize; 2],
446    full: [usize; 2],
447    origin: [usize; 2],
448) {
449    let (facet_cells, full_cells) = (facet[0] * facet[1], full[0] * full[1]);
450    for (image, values) in from.chunks_exact(facet_cells).enumerate() {
451        let target = &mut into[image * full_cells..(image + 1) * full_cells];
452        for (x, column) in values.chunks_exact(facet[1]).enumerate() {
453            let start = (origin[0] + x) * full[1] + origin[1];
454            target[start..start + facet[1]].copy_from_slice(column);
455        }
456    }
457}

463    /// Facet images of a `[4, 2]` facet with one polarization: residual
464    /// value `10·facet + cell`, PSF `100 + cell` on facet 0.
465    fn facet(sum: f64, values: impl Fn(usize) -> f32, initial: bool) -> PassImages {
466        PassImages {
467            domain: 9,
468            shape: [4, 2],
469            channels: 0..1,
470            polarizations: 1,
471            residual: (0..8).map(&values).collect(),
472            psf: initial.then(|| (0..8).map(|cell| 100.0 + cell as f32).collect()),
473            sum_weights: if initial { vec![sum] } else { Vec::new() },
474            published_sum_weights: if initial { vec![sum] } else { Vec::new() },
475            weight: None,
476        }

479    const ORIGINS: [[usize; 2]; 4] = [[0, 0], [4, 0], [0, 2], [4, 2]];

481    /// Four `[4, 2]` facets of an `[8, 4]` domain in CASA's order.
482    fn stitch() -> ChartStitch {
483        let charts = ORIGINS
484            .map(|origin| Chart {
485                domain: 0,
486                window: Some(FacetBox {
487                    origin,
488                    end: [origin[0] + 4, origin[1] + 2],
489                }),
490            })
491            .to_vec();
492        ChartStitch {
493            charts,
494            shapes: vec![[8, 4]],
495            sums: vec![None; 4],
496            assemblies: BTreeMap::new(),
497            pols: 1,
498            coefficients: 1,
499            moments: 1,
500            channels: 1,
501        }
502    }

504    /// The initial pass of facets with `sums`, arriving in `order`, residual
505    /// `10·facet + cell + 1`: the stitched images.
506    fn initial(stitch: &mut ChartStitch, sums: [f64; 4], order: [usize; 4]) -> PassImages {
507        let mut stitched = None;
508        for (arrived, index) in order.into_iter().enumerate() {
509            let images = facet(sums[index], |cell| (10 * index + cell + 1) as f32, true);
510            stitched = stitch.accept(index, images).unwrap();
511            assert_eq!(stitched.is_some(), arrived == 3, "facet {index}");
512        }
513        stitched.unwrap()
514    }

516    /// The value of facet `index`'s cell `(x, y)` in a stitched `[8, 4]`
517    /// residual.
518    fn at(residual: &[f32], index: usize, x: usize, y: usize) -> f64 {
519        f64::from(residual[(ORIGINS[index][0] + x) * 4 + ORIGINS[index][1] + y])
520    }

522    /// Each facet lands at its window divided by its own sum, here scaled
523    /// by facet 0's over its own; facet 0's PSF lands centred; each facet's
524    /// own sum is published; a refresh reuses the initial pass's sums.
525    #[test]
526    fn facets_scatter_to_their_windows_each_divided_by_its_own_sum() {
527        let mut stitch = stitch();
528        // Facet sums 2, 4, 2, 1: facet f's residual scales by 2 / s_f.
529        let sums = [2.0, 4.0, 2.0, 1.0];
530        let stitched = initial(&mut stitch, sums, [0, 1, 2, 3]);
531        assert_eq!((stitched.domain, stitched.shape), (0, [8, 4]));
532        assert_eq!(stitched.sum_weights, vec![2.0]);
533        assert_eq!(stitched.published_sum_weights, vec![2.0]);
534        for (index, sum) in sums.into_iter().enumerate() {
535            for x in 0..4 {
536                for y in 0..2 {
537                    let cell = x * 2 + y;
538                    let expected = (10 * index + cell + 1) as f64 * 2.0 / sum;
539                    assert_eq!(
540                        at(&stitched.residual, index, x, y),
541                        expected,
542                        "facet {index} cell {cell}"
543                    );
544                }
545            }
546        }
547        // Facet 0's PSF, centred: its centre pixel (2, 1) lands on (4, 2).
548        let psf = stitched.psf.unwrap();
549        assert_eq!(psf[4 * 4 + 2], 100.0 + (2 * 2 + 1) as f32);
550        assert_eq!(psf.iter().filter(|value| **value != 0.0).count(), 8);
551        assert_eq!(
552            stitch.facet_sum_weights(),
553            BTreeMap::from([(0, sums.map(|sum| vec![sum]).to_vec())])
554        );
555        // A refresh pass carries no sums and reuses the initial pass's.
556        let mut stitched = None;
557        for index in 0..4 {
558            stitched = stitch.accept(index, facet(0.0, |_| 1.0, false)).unwrap();
559        }
560        let residual = stitched.unwrap().residual;
561        assert_eq!(at(&residual, 1, 0, 0), 2.0 / 4.0);
562        assert_eq!(at(&residual, 3, 0, 0), 2.0 / 1.0);
563    }

591    /// An empty facet 0 does not empty the image: the lowest populated
592    /// facet is the reference, its sums the domain's and its PSF the
593    /// image's (CASA's `setPsfFromOneFacet` would take facet 0's empty
594    /// PSF), whatever order the facets arrive in.
595    #[test]
596    fn an_empty_first_facet_hands_the_reference_to_the_next_populated_one() {
597        let mut stitch = stitch();
598        let sums = [0.0, 4.0, 2.0, 1.0];
599        let stitched = initial(&mut stitch, sums, [3, 2, 1, 0]);
600        assert_eq!(stitched.sum_weights, vec![4.0]);
601        assert_eq!(stitched.published_sum_weights, vec![4.0]);
602        assert_eq!(at(&stitched.residual, 0, 0, 0), 0.0);
603        // Facet f scales by 4 / s_f, so each is its own sum's after the
604        // normal state divides by 4.
605        assert_eq!(at(&stitched.residual, 1, 0, 0), 11.0);
606        assert_eq!(at(&stitched.residual, 2, 0, 0), 21.0 * 4.0 / 2.0);
607        assert_eq!(at(&stitched.residual, 3, 0, 0), 31.0 * 4.0);
608        // Facet 1's PSF (every facet's PSF is 100 + cell), centred.
609        let psf = stitched.psf.unwrap();
610        assert_eq!(psf[4 * 4 + 2], 100.0 + (2 * 2 + 1) as f32);
611        assert_eq!(psf.iter().filter(|value| **value != 0.0).count(), 8);
612    }

614    fn taylor_facet(index: usize, initial: bool) -> PassImages {
615        let zero = [2.0, 4.0, 1.0, 0.0][index];
616        let mut images = facet(zero, |_| 1.0, initial);
617        images.residual.extend([3.0; 8]);
618        if let Some(psf) = &mut images.psf {
619            psf.reserve_exact(16);
620            psf[5] += index as f32;
621            psf.extend([-(index as f32) - 6.0; 8]);
622            psf.extend([index as f32 + 8.0; 8]);
623            images.sum_weights.extend([-0.5, 0.75]);
624            images.published_sum_weights.extend([-0.75, 0.5]);
625        }
626        images
627    }

666        stitch.coefficients = 2;
667        stitch.moments = 3;

689        [2.0, 0.0, 3.0],

695    fn cube_facet(index: usize, channels: Range<usize>, initial: bool) -> PassImages {
696        let mut images = facet(0.0, |_| 0.0, initial);
697        images.residual = vec![1.0; 8 * channels.len()];
698        if initial {
699            images.psf = Some(
700                channels
701                    .clone()
702                    .flat_map(|channel| [1.0 + 10.0 * index as f32 + 100.0 * channel as f32; 8])
703                    .collect(),
704            );
705            images.sum_weights = CUBE_WEIGHTS[index][channels.clone()].to_vec();
706            images.published_sum_weights = images.sum_weights.clone();
707        }
708        images.channels = channels;
709        images
710    }

712    #[test]
713    fn cube_facet_references_and_divisors_follow_the_channel_across_waves() {
714        let mut stitch = stitch();
715        stitch.channels = 3;
716        let references = [2.0, 8.0, 3.0];
717        let centered_psfs = [1.0, 111.0, 201.0];
718        for initial in [true, false] {
719            // Change wave boundaries and facet arrival order on refresh.
720            let waves = if initial { [0..2, 2..3] } else { [0..1, 1..3] };
721            for channels in waves {
722                let mut output = None;
723                for index in [3, 2, 1, 0] {
724                    output = stitch
725                        .accept(index, cube_facet(index, channels.clone(), initial))
726                        .unwrap();
727                }
728                let images = output.unwrap();
729                assert_eq!(images.channels, channels);
730                if let Some(psf) = images.psf {
731                    assert_eq!(images.sum_weights, references[channels.clone()]);
732                    assert_eq!(images.published_sum_weights, references[channels.clone()]);
733                    for (local, absolute) in channels.clone().enumerate() {
734                        assert_eq!(psf[local * 32 + 4 * 4 + 2], centered_psfs[absolute]);
735                    }
736                }
737                for (local, absolute) in channels.enumerate() {
738                    for (index, weights) in CUBE_WEIGHTS.iter().enumerate() {
739                        let own = weights[absolute];
740                        let expected = if own > 0.0 {
741                            references[absolute] / own
742                        } else {
743                            0.0
744                        };
745                        assert_eq!(
746                            at(&images.residual[local * 32..], index, 0, 0),
747                            expected,
748                            "channel {absolute}, facet {index}",
749                        );
750                    }
751                }
752            }
753            assert_eq!(stitch.facet_sum_weights()[&0], CUBE_WEIGHTS.map(Vec::from));
754        }
755        let cached = stitch
756            .sums
757            .iter()
758            .flatten()
759            .map(|sums| sums.psf.capacity() + sums.published.capacity())
760            .sum::<usize>();
761        let published = stitch.facet_sum_weights();
762        let kept = published[&0].iter().map(Vec::capacity).sum::<usize>();
763        assert_eq!(
764            stitch.weight_bytes(),
765            ((cached + kept) * size_of::<f64>()) as u64
766        );
767    }
