2//! `setoptions(ftmachine='mosaic')`: what `predict` builds CASA's simulator
3//! mosaic machine from.
4//!
5//! `Simulator::createSkyEquation` builds `MosaicFTNew` with the
6//! heterogeneous-array beams of `TransformMachines/HetArrayConvFunc` when
7//! OBSERVATION row 0's telescope has PBMath's UNKNOWN, OVRO, ACA or ALMA
8//! beam. casa-rs builds it for ALMA, ACA and OVRO. ALMA and ACA dishes
9//! `findAntennaSizes` replaces by the 10.7 m and 6.25 m Airy apertures
10//! ([`AiryDish::casa_alma_simulator`]); OVRO uses each physical diameter
11//! and its scaled blockage. Other supported default telescope patterns use
12//! `MosaicFTNew`'s `SimplePBConvFunc`, one common beam for all antennas.
13//!
14//! - The dish classes are the distinct dish diameters in ANTENNA order, a
15//!   diameter compared as casacore's `String::toString` prints it, to six
16//!   significant digits.
17//! - Every channel of each window observed is selected.
18//! - A scan's beam uses POINTING independently of its FIELD phase centre;
19//!   the kernels' phase ramp is its offset from the model's pixel
20//!   `(nx/2, ny/2)`. Heterogeneous-array beams predict nothing for a
21//!   pointing outside the model image (`checkPBOfField`); common beams
22//!   keep applying the phase ramp outside the image.
23

30/// The telescopes whose mosaic casa-rs builds.
31const MOSAIC_TELESCOPES: [&str; 3] = ["ALMA", "ACA", "OVRO"];

33/// A dish diameter as `String::toString(Double)` prints it, which keys
34/// `HetArrayConvFunc`'s dish classes: six significant digits.
35fn diameter_key(diameter_m: f64) -> String {
36    format!("{diameter_m:.5e}")
37}
38
39/// The most a diameter key's text holds: `format!` grows its string from
40/// nothing, to 16 bytes for its nine to twelve characters.
41const DIAMETER_KEY_BYTES: usize = 32;

43/// One pointing the mosaic machine keeps: a phase centre's bits and its
44/// offset.
45pub(super) type MosaicPointing = ([u64; 2], Option<[f64; 2]>);

49type FrameKey = (i32, usize, usize);

51#[derive(Default)]
52struct BufferClock {
53    rows: usize,
54    epoch_s: f64,
55    antenna: usize,
56    last_row: Option<u64>,

59/// CASA's visibility-buffer clock, sampled before numerical workers run.
60pub(super) struct MosaicGeometry {
61    geometry: crate::predict::PointingGeometry,
62    buffers: std::sync::Mutex<Vec<(FrameKey, BufferClock)>>,
63    rows: usize,
64    observer: Option<[f64; 3]>,
65    time_order: Option<time_order::TimeOrder>,
66}
67
68impl MosaicGeometry {
69    fn bytes_for(
70        model: &SkyModel,
71        centres: Option<&super::from_ms::polynomial::PhaseCentres>,
72    ) -> u64 {
73        model.pointing_geometry_bytes()
74            + (size_of::<Self>() - size_of::<crate::predict::PointingGeometry>()
75                + centres.map_or(0, |p| p.value.len()) * size_of::<(FrameKey, BufferClock)>())
76                as u64
77    }
78
79    pub(super) fn new(
80        model: &SkyModel,
81        observer: Option<[f64; 3]>,
82        bounded: bool,
83        centres: Option<&super::from_ms::polynomial::PhaseCentres>,
84        antennas: usize,
85    ) -> Self {
86        let first = super::from_ms::polynomial::first_chunk_rows(centres).min(101);
87        // Simulator::setRowBlocking(100) includes both endpoints. SkyEquation
88        // increases that blocking when the first buffer holds too few rows.
89        let rows = if antennas > 1 && first < antennas * (antennas - 1) / 4 {
90            antennas * (antennas + 1) / 2 + 1
91        } else {
92            101
93        };
94        let mut buffers = Vec::with_capacity(centres.map_or(0, |p| p.value.len()));
95        if let Some(centres) = centres {
96            buffers.extend(
97                centres
98                    .value
99                    .iter()
100                    .map(|(key, _)| (*key, BufferClock::default())),
101            );
102        }
103        Self {
104            geometry: model.pointing_geometry(bounded),
105            observer,
106            time_order: None,
107            buffers: std::sync::Mutex::new(buffers),
108            rows,
109        }
110    }
111
112    pub(super) fn bytes(&self) -> u64 {
113        self.geometry.bytes()
114            + (size_of::<Self>() - size_of::<crate::predict::PointingGeometry>()
115                + self.buffers.lock().expect("buffer metadata").capacity()
116                    * size_of::<(FrameKey, BufferClock)>()) as u64
117    }
118
119    pub(super) fn sample(
120        &self,
121        scan: &super::observe::Scan,
122        row: u64,
123        time: f64,
124        antenna: usize,
125    ) -> SimulatorResult<(f64, usize)> {
126        let key = (scan.array_id, scan.field_id, scan.data_description_id);
127        if let Some(order) = &self.time_order {
128            return order.sample(key, row, time);
129        }
130        let mut buffers = self
131            .buffers
132            .lock()
133            .map_err(|e| SimulatorError::raised(e.to_string()))?;
134        let index = buffers
135            .binary_search_by_key(&key, |(key, _)| *key)
136            .map_err(|_| SimulatorError::raised("missing mosaic frame chunk"))?;
137        let clock = &mut buffers[index].1;
138        // A private selection rendering or a retried transaction starts
139        // another pass over the same captured physical rows.
140        if clock.last_row.is_some_and(|previous| row <= previous) {
141            clock.rows = 0;
142        }
143        if clock.rows == 0 {
144            clock.epoch_s = time;
145            clock.antenna = antenna;
146        }
147        clock.last_row = Some(row);
148        clock.rows += 1;
149        if clock.rows == self.rows {
150            clock.rows = 0;
151        }
152        Ok((clock.epoch_s, clock.antenna))
153    }
154
155    pub(super) fn offset(
156        &self,
157        pointing: [f64; 2],
158        frame: &casa_core::measures::MeasFrame,
159    ) -> Result<Option<[f64; 2]>, PredictError> {
160        let mut frame = frame.clone();
161        if let Some([x, y, z]) = self.observer {
162            frame = frame.with_position(casa_core::measures::MPosition::new_itrf(x, y, z));
163        }
164        self.geometry.offset(pointing, &frame)
165    }
166}

168impl Simulator {
169    /// Bytes the mosaic machine's state holds, known before it is formed:
170    /// what the prediction keeps (fixed offsets or numeric direction geometry
171    /// and a clock per selected chunk, plus each antenna's dish class), and what builds the beams
172    /// until the kernels are built (a dish and a diameter key for each
173    /// antenna at most, and every window's channel frequencies twice).
174    pub(super) fn mosaic_state_bytes(
175        &self,
176        model: &SkyModel,
177        centres: Option<&super::from_ms::polynomial::PhaseCentres>,
178    ) -> (u64, u64) {
179        let timed = !crate::predict::fixed_direction_frame(model.tangent_direction().refer());
180        let geometry = if timed {
181            MosaicGeometry::bytes_for(model, centres) as usize
182        } else {
183            self.scans.len() * size_of::<MosaicPointing>()
184        };
185        let kept = geometry + self.antennas.len() * size_of::<u8>();
186        let antennas = self.antennas.len()
187            * (size_of::<PrimaryBeam>() + size_of::<String>() + DIAMETER_KEY_BYTES);
188        let windows = self.spectral_windows.len()
189            * (size_of::<MosaicWindow>() + size_of::<(u32, f64)>())
190            + self
191                .spectral_windows
192                .iter()
193                .map(|window| 2 * window.channel_count * size_of::<f64>())
194                .sum::<usize>();
195        (
196            kept as u64,
197            (antennas + windows + if timed { 256 } else { 0 }) as u64,
198        )
199    }

201    /// SimplePBConvFunc keeps the catalogue position only for VLA images
202    /// on a VLA-labelled observation; other buffers use their first antenna.
203    pub(super) fn mosaic_observer(&self, model: &SkyModel) -> SimulatorResult<Option<[f64; 3]>> {
204        if !model.observation_telescope().contains("VLA")
205            || !self.observing_telescope().contains("VLA")
206        {
207            return Ok(None);
208        }
209        casa_core::measures::MPosition::from_observatory_name(
210            self.observing_telescope(),
211            self.options.measures.as_ref(),
212        )
213        .map(|position| position.map(|p| p.as_itrf()))
214        .map_err(|error| SimulatorError::raised(error.to_string()))
215    }
216
217    /// The mosaic machine's beams for the scans observed so far, and each
218    /// antenna's dish class.
219    ///
220    /// # Errors
221    ///
222    /// Not yet implemented: a telescope without a supported default beam,
223    /// a heterogeneous-array antenna without a dish, or more than 255 dish
224    /// diameters.
225    pub(super) fn mosaic_beams(
226        &self,
227        custom: Option<&super::voltage_patterns::BeamSource>,
228    ) -> SimulatorResult<(MosaicBeams, Vec<u8>)> {
229        let telescope = self.observing_telescope();
230        let common = custom.is_some() || !MOSAIC_TELESCOPES.contains(&telescope);
231        let default = match custom {
232            Some(beam) => beam.configured(),
233            None => PrimaryBeam::for_telescope(telescope, self.reference_frequency_hz()),
234        };
235        if common && default.is_none() && custom.is_none() {
236            return Err(SimulatorError::refused(format!(
237                "Failed to create SkyEquation: vpmanager does not have a beam for {telescope}"
238            )));
239        }
240        // At most one class an antenna: the capacities are the bounds
241        // `mosaic_state_bytes` admits.
242        let mut keys: Vec<String> = Vec::with_capacity(self.antennas.len());
243        let mut dishes = Vec::with_capacity(self.antennas.len());
244        let mut classes = Vec::with_capacity(self.antennas.len());
245        if common && let Some(default) = default {
246            dishes.push(default);
247        }
248        for antenna in &self.antennas {
249            if common {
250                classes.push(0);
251                continue;
252            }
253            // `findAntennaSizes` gives a dish class only to `diameter > 0`.
254            if antenna.diameter_m.is_nan() || antenna.diameter_m <= 0.0 {
255                return Err(SimulatorError::not_yet_implemented(format!(
256                    "predict with setoptions(ftmachine='mosaic') for antenna {} of diameter {} \
257                     m is not yet implemented in casa-rs (#1053)",
258                    antenna.name, antenna.diameter_m
259                )));
260            }
261            let key = diameter_key(antenna.diameter_m);
262            let class = keys
263                .iter()
264                .position(|known| *known == key)
265                .unwrap_or_else(|| {
266                    let dish = if matches!(telescope, "ALMA" | "ACA") {
267                        AiryDish::casa_alma_simulator(antenna.diameter_m)
268                    } else {
269                        AiryDish::casa_airy(antenna.diameter_m)
270                    };
271                    tracing::info!(
272                        "Overriding PB with Airy of diameter {} m starting with antenna {}",
273                        dish.aperture_m,
274                        classes.len()
275                    );
276                    keys.push(key);
277                    dishes.push(PrimaryBeam::from(dish));
278                    keys.len() - 1
279                });
280            classes.push(u8::try_from(class).map_err(|_| {
281                SimulatorError::not_yet_implemented(
282                    "predict with setoptions(ftmachine='mosaic') for more than 255 dish \
283                     diameters is not yet implemented in casa-rs (#1053)",
284                )
285            })?);
286        }
287        let mut windows: Vec<MosaicWindow> = Vec::with_capacity(self.spectral_windows.len());
288        for scan in &self.scans {
289            let id = scan.spectral_window_id;
290            if windows
291                .iter()
292                .any(|window| window.spectral_window as usize == id)
293            {
294                continue;
295            }
296            let window = &self.spectral_windows[id];
297            let frequencies = window.channel_frequencies_hz();
298            windows.push(MosaicWindow {
299                spectral_window: id as u32,
300                window_frequencies_hz: frequencies.clone(),
301                channel_width_hz: window.channel_width_hz,
302                selected_frequencies_hz: frequencies,
303                buffer_band: None,
304            });
305        }
306        Ok((
307            MosaicBeams {
308                common,
309                custom_pattern: custom.cloned(),
310                image_angles_rad: Vec::new(),
311                dishes,
312                windows,
313            },
314            classes,
315        ))
316    }

318    /// Each distinct pointing of the scans observed so far (its bits), where
319    /// their antennas point, and its offset in `model`
320    /// ([`SkyModel::pointing_offset_and_coverage`]), sorted by the bits. The capacity
321    /// is a pointing for each scan, the bound `mosaic_state_bytes` admits.
322    ///
323    /// # Errors
324    ///
325    /// Raised: a pointing the model's coordinates cannot convert. A `None`
326    /// offset records CASA's heterogeneous-array zero contribution outside
327    /// the model image; common beams retain the offset.
328    pub(super) fn mosaic_pointings(
329        &self,
330        model: &SkyModel,
331        bounded: bool,
332    ) -> SimulatorResult<Vec<MosaicPointing>> {
333        let mut pointings: Vec<MosaicPointing> = Vec::with_capacity(self.scans.len());
334        for scan in &self.scans {
335            let key = scan.pointing_rad.map(f64::to_bits);
336            let (offset, inside) = model
337                .pointing_offset_and_coverage(scan.pointing_rad)
338                .map_err(|error: PredictError| {
339                    SimulatorError::raised(format!("Caught exception: {error}"))
340                })?;
341            pointings.push((key, (!bounded || inside).then_some(offset)));
342        }
343        pointings.sort_unstable_by_key(|(key, _)| *key);
344        pointings.dedup_by_key(|(key, _)| *key);
345        Ok(pointings)
346    }
