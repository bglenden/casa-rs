4/// The shape of one image plane.
5///
6/// Pixels are stored x-major: pixel `(x, y)` is at `x * ny + y`. That is the
7/// layout of the normal state, masks and model the minor cycle is fed from,
8/// so no plane is transposed on the way in or out. casacore scans arrays
9/// x fastest; the peak searches below reproduce its tie order explicitly.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11pub struct PlaneShape {
12    /// Pixels along x (the first image axis).
13    pub nx: usize,
14    /// Pixels along y (the second image axis).
15    pub ny: usize,
16}
17
18impl PlaneShape {
19    /// A plane of `nx` by `ny` pixels.
20    #[must_use]
21    pub const fn new(nx: usize, ny: usize) -> Self {
22        Self { nx, ny }
23    }
24
25    /// Number of pixels.
26    #[must_use]
27    pub const fn len(self) -> usize {
28        self.nx * self.ny
29    }
30
31    /// Whether the plane has no pixels.
32    #[must_use]
33    pub const fn is_empty(self) -> bool {
34        self.len() == 0
35    }
36
37    /// Storage index of pixel `(x, y)`.
38    #[must_use]
39    pub const fn index(self, x: usize, y: usize) -> usize {
40        x * self.ny + y
41    }
42
43    /// Pixel `[x, y]` of a storage index.
44    #[must_use]
45    pub const fn pixel(self, index: usize) -> [usize; 2] {
46        [index / self.ny, index % self.ny]
47    }
48
49    /// The image centre `(nx/2, ny/2)`, where CASA centres PSFs and scale
50    /// functions.
51    #[must_use]
52    pub const fn centre(self) -> [usize; 2] {
53        [self.nx / 2, self.ny / 2]
54    }
55
56    /// Whether storage index `a` comes before `b` in casacore's x-fastest
57    /// scan order.
58    #[must_use]
59    pub const fn scans_before(self, a: usize, b: usize) -> bool {
60        let [ax, ay] = self.pixel(a);
61        let [bx, by] = self.pixel(b);
62        ay < by || (ay == by && ax < bx)
63    }
64}

66/// The pixels a component may be placed on: the clean mask and the valid
67/// model support (the primary-beam limit) together.
68#[derive(Clone, Debug, PartialEq, Eq)]
69pub struct Support {
70    shape: PlaneShape,
71    pixels: Vec<bool>,
72    count: usize,
73}
74
75impl Support {
76    /// Support from one flag per pixel, in storage order.
77    ///
78    /// # Panics
79    ///
80    /// When the flags do not cover the plane.
81    #[must_use]
82    pub fn new(shape: PlaneShape, pixels: Vec<bool>) -> Self {
83        assert_eq!(pixels.len(), shape.len(), "support covers its plane");
84        let count = pixels.iter().filter(|pixel| **pixel).count();
85        Self {
86            shape,
87            pixels,
88            count,
89        }
90    }
91
92    /// Every pixel of the plane.
93    #[must_use]
94    pub fn full(shape: PlaneShape) -> Self {
95        Self::new(shape, vec![true; shape.len()])
96    }
97
98    /// The plane this support covers.
99    #[must_use]
100    pub const fn shape(&self) -> PlaneShape {
101        self.shape
102    }
103
104    /// Whether storage index `index` is supported.
105    #[must_use]
106    pub fn contains(&self, index: usize) -> bool {
107        self.pixels[index]
108    }
109
110    /// The flags in storage order.
111    #[must_use]
112    pub fn as_slice(&self) -> &[bool] {
113        &self.pixels
114    }
115
116    /// Number of supported pixels (CASA's mask sum).
117    #[must_use]
118    pub const fn count(&self) -> usize {
119        self.count
120    }
121
122    /// Whether no pixel is supported.
123    #[must_use]
124    pub const fn is_empty(&self) -> bool {
125        self.count == 0
126    }
127}

129/// The supported pixel of largest absolute value, as the Fortran `hclean`
130/// peak search finds it: a pixel replaces the best only when strictly
131/// larger, so among equal magnitudes the first in x-fastest order wins.
132/// NaN values are skipped ([`crate::run_plane`] refuses a nonfinite plane).
133/// Returns the index and the signed value; `None` when nothing is supported.
134#[must_use]
135pub fn first_peak(values: &[f64], support: &Support) -> Option<(usize, f64)> {
136    let shape = support.shape();
137    let mut best: Option<(usize, f64)> = None;
138    for (index, (&value, &supported)) in values.iter().zip(support.as_slice()).enumerate() {
139        if !supported {
140            continue;
141        }
142        let magnitude = value.abs();
143        if best.is_none_or(|(at, peak)| {
144            magnitude > peak.abs() || (magnitude == peak.abs() && shape.scans_before(index, at))
145        }) {
146            best = Some((index, value));
147        }
148    }
149    best
150}

152/// casacore's `findMaxAbsMask`: the signed extreme of largest magnitude
153/// over the supported pixels, taken from `minMax`. The maximum and the
154/// minimum are each the first occurrence in x-fastest order, and the
155/// minimum wins only when strictly larger in magnitude.
156///
157/// Unsupported pixels take part as zeros (casacore multiplies by the mask),
158/// so with no supported pixel the result is `(0, 0.0)`.
159#[must_use]
160pub fn casacore_max_abs(values: &[f64], support: &Support) -> (usize, f64) {
161    let shape = support.shape();
162    let mut maximum = (0, f64::NEG_INFINITY);
163    let mut minimum = (0, f64::INFINITY);
164    for (index, (&value, &supported)) in values.iter().zip(support.as_slice()).enumerate() {
165        let value = if supported { value } else { 0.0 };
166        if value > maximum.1 || (value == maximum.1 && shape.scans_before(index, maximum.0)) {
167            maximum = (index, value);
168        }
169        if value < minimum.1 || (value == minimum.1 && shape.scans_before(index, minimum.0)) {
170            minimum = (index, value);
171        }
172    }
173    if minimum.1.abs() > maximum.1.abs() {
174        minimum
175    } else {
176        maximum
177    }
178}

180/// The largest absolute value over the supported pixels; zero when none is.
181#[must_use]
182pub fn peak_magnitude(values: &[f64], support: &Support) -> f64 {
183    values
184        .iter()
185        .zip(support.as_slice())
186        .filter(|(_, supported)| **supported)
187        .fold(0.0_f64, |peak, (value, _)| peak.max(value.abs()))
188}

190/// The PSF peak as CASA's `MatrixCleaner::findPSFMaxAbs` finds it: the
191/// largest magnitude inside the central `findBeamPatch(0, nx, ny, 4, 20)`
192/// box, scanned x fastest with strictly-greater replacement. The box's
193/// upper bounds are exclusive, as in casacore, and an axis no longer than
194/// the patch is searched from 0 to its last pixel exclusive.
195#[must_use]
196pub fn psf_peak(psf: &[f64], shape: PlaneShape) -> Option<usize> {
197    let support = beam_patch(0.0, shape);
198    let bounds = |extent: usize| {
199        if extent > support {
200            (extent / 2 - support / 2, extent / 2 + support / 2)
201        } else {
202            (0, extent.saturating_sub(1))
203        }
204    };
205    let (x0, x1) = bounds(shape.nx);
206    let (y0, y1) = bounds(shape.ny);
207    let mut best: Option<(usize, f64)> = None;
208    for y in y0..y1 {
209        for x in x0..x1 {
210            let index = shape.index(x, y);
211            let magnitude = psf[index].abs();
212            if best.is_none_or(|(_, peak)| magnitude > peak) {
213                best = Some((index, magnitude));
214            }
215        }
216    }
217    best.map(|(index, _)| index)
218}

220/// `MatrixCleaner::findBeamPatch(maxScaleSize, nx, ny, 4, 20)`: the even
221/// side of the PSF patch the multi-term cleaner updates within, at least
222/// 80 pixels and at most the smaller image axis.
223#[must_use]
224pub fn beam_patch(maximum_scale_px: f64, shape: PlaneShape) -> usize {
225    const PSF_BEAM: f64 = 4.0;
226    const BEAMS: f64 = 20.0;
227    let mut support =
228        ((PSF_BEAM * PSF_BEAM + maximum_scale_px * maximum_scale_px).sqrt() * BEAMS) as usize;
229    if (support as f64) < PSF_BEAM * BEAMS {
230        support = (PSF_BEAM * BEAMS) as usize;
231    }
232    if support > shape.nx || support > shape.ny {
233        support = shape.nx.min(shape.ny);
234    }
235    if !support.is_multiple_of(2) {
236        support -= 1;
237    }
238    support
239}

241/// CASA's robust noise of one plane (`SIImageStore::calcRobustRMS`): the
242/// median of the supported values and 1.4826 times their median absolute
243/// deviation. `None` when nothing is supported.
244#[must_use]
245pub fn robust_noise(values: &[f64], support: &Support) -> Option<RobustNoise> {
246    let mut samples = values
247        .iter()
248        .zip(support.as_slice())
249        .filter(|(_, supported)| **supported)
250        .map(|(value, _)| *value)
251        .collect::<Vec<_>>();
252    if samples.is_empty() {
253        return None;
254    }
255    let middle = samples.len() / 2;
256    let median = *samples.select_nth_unstable_by(middle, f64::total_cmp).1;
257    for sample in &mut samples {
258        *sample = (*sample - median).abs();
259    }
260    let deviation = *samples.select_nth_unstable_by(middle, f64::total_cmp).1;
261    Some(RobustNoise {
262        median,
263        rms: 1.482_602_218_505_602 * deviation,
264    })
265}

267/// The robust statistics of one residual plane.
268#[derive(Clone, Copy, Debug, PartialEq)]
269pub struct RobustNoise {
270    /// Median of the supported values.
271    pub median: f64,
272    /// 1.4826 times the median absolute deviation.
273    pub rms: f64,
274}

280    #[test]
281    fn hclean_ties_go_to_the_first_pixel_in_x_fastest_order() {
282        let shape = PlaneShape::new(3, 4);
283        let mut values = vec![0.0; shape.len()];
284        // (2, 1) precedes (0, 3) in x-fastest order although it is stored later.
285        values[shape.index(0, 3)] = -2.0;
286        values[shape.index(2, 1)] = 2.0;
287        let support = Support::full(shape);
288        assert_eq!(
289            first_peak(&values, &support),
290            Some((shape.index(2, 1), 2.0))
291        );
292        values[shape.index(1, 0)] = -2.0;
293        assert_eq!(
294            first_peak(&values, &support),
295            Some((shape.index(1, 0), -2.0))
296        );
297        let masked = Support::new(
298            shape,
299            (0..shape.len()).map(|i| i != shape.index(1, 0)).collect(),
300        );
301        assert_eq!(first_peak(&values, &masked), Some((shape.index(2, 1), 2.0)));
302        assert_eq!(
303            first_peak(&values, &Support::new(shape, vec![false; 12])),
304            None
305        );
306    }

308    #[test]
309    fn casacore_max_abs_prefers_the_maximum_on_equal_magnitude() {
310        let shape = PlaneShape::new(4, 3);
311        let mut values = vec![0.5; shape.len()];
312        values[shape.index(0, 0)] = -3.0;
313        values[shape.index(3, 2)] = 3.0;
314        let support = Support::full(shape);
315        assert_eq!(
316            casacore_max_abs(&values, &support),
317            (shape.index(3, 2), 3.0)
318        );
319        values[shape.index(1, 1)] = -3.5;
320        assert_eq!(
321            casacore_max_abs(&values, &support),
322            (shape.index(1, 1), -3.5)
323        );
324        // Masked pixels count as zeros.
325        let only_low = Support::new(shape, (0..12).map(|i| values[i] == 0.5).collect());
326        assert_eq!(casacore_max_abs(&values, &only_low).1, 0.5);
327        let none = Support::new(shape, vec![false; 12]);
328        assert_eq!(casacore_max_abs(&values, &none), (0, 0.0));
329    }

331    #[test]
332    fn the_psf_peak_is_searched_in_the_central_patch_with_exclusive_bounds() {
333        // 100 pixels: patch 80, box 10..90 on both axes.
334        let shape = PlaneShape::new(100, 100);
335        let mut psf = vec![0.0; shape.len()];
336        psf[shape.index(50, 50)] = 1.0;
337        psf[shape.index(5, 50)] = 2.0;
338        psf[shape.index(50, 90)] = 2.0;
339        assert_eq!(psf_peak(&psf, shape), Some(shape.index(50, 50)));
340        // An axis no longer than the patch ends one short of its last pixel.
341        let small = PlaneShape::new(9, 7);
342        let mut psf = vec![0.0; small.len()];
343        psf[small.index(8, 3)] = 5.0;
344        psf[small.index(4, 3)] = 1.0;
345        assert_eq!(psf_peak(&psf, small), Some(small.index(4, 3)));
346    }
