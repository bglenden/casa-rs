26impl PyMsTool {
27    /// Materialize selected rows while retaining the input subtable IDs.
28    #[allow(clippy::too_many_arguments)]
29    #[pyo3(signature = (outputms = String::new(), field = None, spw = None, baseline = None, timebin = None, time = String::new(), scan = None, uvrange = None, taql = String::new(), whichcol = "DATA".to_owned(), tileshape = None, subarray = None, combine = String::new(), intent = String::new(), obs = String::new()), text_signature = "($self, outputms='', field=[], spw=[], baseline=[], timebin=[], time='', scan=[], uvrange=[], taql='', whichcol='DATA', tileshape=[], subarray=[], combine='', intent='', obs='')")]
30    fn partition(
31        &self,
32        py: Python<'_>,
33        #[pyo3(from_py_with = output_argument)] outputms: String,
34        field: Option<&Bound<'_, PyAny>>,
35        spw: Option<&Bound<'_, PyAny>>,
36        baseline: Option<&Bound<'_, PyAny>>,
37        timebin: Option<Variant>,
38        #[pyo3(from_py_with = time_argument)] time: String,
39        scan: Option<&Bound<'_, PyAny>>,
40        uvrange: Option<&Bound<'_, PyAny>>,
41        #[pyo3(from_py_with = taql_argument)] taql: String,
42        #[pyo3(from_py_with = whichcol_argument)] whichcol: String,
43        tileshape: Option<Variant>,
44        subarray: Option<&Bound<'_, PyAny>>,
45        #[pyo3(from_py_with = combine_argument)] combine: String,
46        #[pyo3(from_py_with = intent_argument)] intent: String,
47        #[pyo3(from_py_with = obs_argument)] obs: String,
48    ) -> PyResult<bool> {
49        let spw = all_if_minus_one(spw)?.unwrap_or_else(|| "*".into());
50        let selection = MsSelection {
51            field: all_if_minus_one(field)?,
52            antenna: optional_expression(baseline)?,
53            timerange: (!time.is_empty()).then_some(time),
54            scan: optional_expression(scan)?,
55            uvrange: optional_expression(uvrange)?,
56            msselect: (!taql.is_empty()).then_some(taql),
57            array: optional_expression(subarray)?,
58            intent: (!intent.is_empty()).then_some(intent),
59            observation: (!obs.is_empty()).then_some(obs),
60            ..MsSelection::default()
61        };
62        let columns = visibility_columns(&whichcol)?;
63        let options = MsTransformTaskOptions {
64            visibility_columns: columns,
65            preserve_ids: true,
66            legacy_tool: Some(LegacyMsTool::Partition(LegacyTimeOptions {
67                combine_observations: combine.to_ascii_lowercase().contains("obs"),
68            })),
69            time_averaging: time_average(timebin.as_ref(), &combine)?,
70            tiling: tile_layout(tileshape.as_ref())?,
71            ..MsTransformTaskOptions::default()
72        };
73        self.write_subms(py, outputms, spw, selection, options, ("partition", None))
74    }
75}

77impl PyMsTool {
78    pub(super) fn write_subms(
79        &self,
80        py: Python<'_>,
81        outputms: String,
82        spw: String,
83        selection: MsSelection,
84        options: MsTransformTaskOptions,
85        (operation, history): (&'static str, Option<String>),
86    ) -> PyResult<bool> {
87        py.detach(|| {
88            let attached = self.attached.lock().map_err(runtime_error)?;
89            let Some(ms) = attached.as_ref() else {
90                if operation != "partition" {
91                    return Err(runtime_error("Table column DATA_DESC_ID is unknown"));
92                }
93                return Ok(false);
94            };
95            if operation == "partition" && spw.contains(':') {
96                return Ok(false);
97            }
98            let link = ms.working_metadata_link();
99            let view = link.working_view().map_err(runtime_error)?;
100            let input = view.path();
101            let output = (!outputms.is_empty())
102                .then(|| named_path(Path::new(&outputms), &ms.path, operation))
103                .transpose()?;
104            let destination = output
105                .as_deref()
106                .map(|output| {
107                    check_named_output_path(input, output).map_err(runtime_error)?;
108                    SortedDestination::inspect(output, operation)
109                })
110                .transpose()?;
111            let locks = match output.as_deref().zip(destination.as_ref()) {
112                Some((output, destination)) if destination.is_table() => {
113                    check_write_access(output).map_err(runtime_error)?;
114                    lock_table_tree_for_replacement(output).map_err(runtime_error)?
115                }
116                _ => Vec::new(),
117            };
118            let scratch = output
119                .as_deref()
120                .and_then(Path::parent)
121                .map_or_else(tempfile::tempdir, tempfile::tempdir_in)
122                .map_err(runtime_error)?;
123            let staged = scratch.path().join(format!("{operation}.ms"));
124            let request = MsTransformRequest {
125                input_ms: input.to_path_buf(),
126                output_ms: staged.clone(),
127                spw,
128                width: 1,
129                data_column: TransformDataColumn::All,
130                selection,
131                keep_flags: true,
132            };
133            link.verify_directory().map_err(runtime_error)?;
134            match mstransform_task(&request, &options) {
135                Err(MsTransformError::EmptySelection { .. }) => {
136                    if let Some(field) = &request.selection.field {
137                        let source = link.snapshot(None).map_err(runtime_error)?;
138                        source
139                            .ms
140                            .resolve_field_selector(field)
141                            .map_err(runtime_error)?;
142                    }
143                    return Ok(false);
144                }
145                Err(MsTransformError::ChannelWidthCount) if operation == "split" => {
146                    return Ok(false);
147                }
148                result => result.map_err(runtime_error)?,
149            };
150            if let Some(parameters) = history.as_ref()
151                && output.is_some()
152            {
153                let table = Table::open(TableOptions::new(&staged)).map_err(runtime_error)?;
154                let message = if operation == "contsub" {
155                    format!("Continuum subtracted from {}", ms.path.display())
156                } else {
157                    format!("{outputms} split from {}", ms.path.display())
158                };
159                let origin = format!("ms::{operation}()");
160                append_tool_history(
161                    &table,
162                    true,
163                    &[message],
164                    parameters,
165                    &origin,
166                    "ms",
167                    &mut Default::default(),
168                )
169                .map_err(runtime_error)?;
170            }
171            link.verify_directory().map_err(runtime_error)?;
172            if !locks
173                .iter()
174                .all(|lock| lock.matches_table_path(lock.path()))
175            {
176                return Err(runtime_error(format!(
177                    "{operation} output changed during staging"
178                )));
179            }
180            if let Some((destination, output)) = destination.zip(output) {
181                destination.publish(&staged, &output, operation)?;
182            }
183            Ok(true)
184        })
185    }
186}

188fn named_path(output: &Path, input: &Path, operation: &str) -> PyResult<PathBuf> {
189    let parent = output
190        .parent()
191        .filter(|path| !path.as_os_str().is_empty())
192        .unwrap_or_else(|| Path::new("."))
193        .canonicalize()
194        .map_err(runtime_error)?;
195    let output =
196        parent.join(output.file_name().ok_or_else(|| {
197            runtime_error(format!("{operation} output must name a MeasurementSet"))
198        })?);
199    if input.starts_with(&output) {
200        return Err(runtime_error(format!(
201            "{operation} output would replace its input"
202        )));
203    }
204    Ok(output)
205}

244pub(super) fn visibility_columns(whichcol: &str) -> PyResult<Option<Vec<OptionalMainColumn>>> {
245    use OptionalMainColumn::{CorrectedData, Data, FloatData, LagData, ModelData};
246    let text = whichcol.to_ascii_uppercase();
247    if text.contains("ALL") || text.contains("ANY") {
248        return Ok(None);
249    }
250    let columns = text
251        .split([' ', ','])
252        .filter_map(|token| match token {
253            "DATA" | "OBSERVED" => Some(Data),
254            "MODEL" | "MODEL_DATA" => Some(ModelData),
255            "CORRECTED" | "CORRECTED_DATA" => Some(CorrectedData),
256            "FLOAT" | "FLOAT_DATA" => Some(FloatData),
257            "LAG" | "LAG_DATA" => Some(LagData),
258            _ => None,
259        })
260        .collect::<Vec<_>>();
261    if columns.is_empty() {
262        Err(runtime_error("did not find and select any data columns"))
263    } else {
264        Ok(Some(columns))
265    }
266}

268pub(super) fn time_average(
269    timebin: Option<&Variant>,
270    combine: &str,
271) -> PyResult<Option<TimeAverageOptions>> {
272    let seconds = timebin
273        .map(|value| quantity_value(value, "s"))
274        .transpose()
275        .map_err(runtime_error)?
276        .unwrap_or(0.0);
277    if !seconds.is_finite() {
278        return Err(runtime_error("partition timebin must be finite"));
279    }
280    let combine = combine.to_ascii_lowercase();
281    Ok((seconds > 0.0).then_some(TimeAverageOptions {
282        interval_seconds: seconds,
283        span_scans: combine.contains("scan"),
284        span_states: combine.contains("state"),
285        span_fields: false,
286        max_uvw_distance: 0.0,
287    }))
288}

290pub(super) fn tile_layout(tileshape: Option<&Variant>) -> PyResult<Option<MsTransformTiling>> {
291    let Some(Variant::Numbers(values) | Variant::Array { values, .. }) = tileshape else {
292        return Ok(None);
293    };
294    let [correlations, channels, rows] = values.as_slice() else {
295        return Ok(None);
296    };
297    let dimensions = [*correlations, *channels, *rows];
298    if !dimensions
299        .iter()
300        .all(|value| (0.0..=f64::from(i32::MAX)).contains(value))
301    {
302        return Err(runtime_error(
303            "partition tile dimensions must be nonnegative 32-bit integers",
304        ));
305    }
306    Ok(Some(MsTransformTiling::Fixed(
307        dimensions.map(|value| value as usize),
308    )))
