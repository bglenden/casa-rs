3//! Typed selected-row reads served cell by cell, packed into the typed
4//! layouts.
5//!
6//! `TiledShapeStMan` streams selected rows and channels straight into the
7//! 2-D packed layouts, and the tiled and incremental managers stream 1-D
8//! rows. Every other manager casacore writes (`StManAipsIO`, `StandardStMan`
9//! and the other tiled managers for 2-D cells) is read cell by cell through
10//! the general selected-row reader and packed here, so a MeasurementSet reads
11//! the same whatever manager CASA chose for a column.
12//!
13//! A read served here holds every selected row's whole stored cell until it
14//! is packed ([`SelectedReadFootprint::WholeCells`]); it never reads a whole
15//! column, so that footprint is a true bound for a memory planner. The
16//! classifiers [`channel_read_footprint`] and [`cell_read_footprint`] name
17//! the managers the typed readers stream, and the readers' dispatch uses the
18//! same lists.
19

32/// What the typed selected 2-D channel-range readers hold for a column
33/// stored by `data_manager`: only `TiledShapeStMan` streams the selected
34/// channels.
35pub(crate) fn channel_read_footprint(data_manager: &str) -> SelectedReadFootprint {
36    match data_manager {
37        "TiledShapeStMan" => SelectedReadFootprint::Streamed,
38        _ => SelectedReadFootprint::WholeCells,
39    }
40}

42/// What the typed selected 1-D readers hold for a column stored by
43/// `data_manager`: the tiled column managers and `IncrementalStMan` stream
44/// the selected rows.
45pub(crate) fn cell_read_footprint(data_manager: &str) -> SelectedReadFootprint {
46    match data_manager {
47        "IncrementalStMan" | "TiledColumnStMan" | "TiledShapeStMan" => {
48            SelectedReadFootprint::Streamed
49        }
50        _ => SelectedReadFootprint::WholeCells,
51    }
52}

58    table_dat: &'a TableDatContents,
59    column: &'a str,
60    selected_rows: &'a [usize],

63impl CompositeStorage {

65    /// cell.
66    pub(super) fn whole_cell_read<'a>(
67        &'a self,
68        table_path: &'a Path,
69        table_dat: &'a TableDatContents,
70        column: &'a str,
71        selected_rows: &'a [usize],
72    ) -> WholeCellRead<'a> {

75            table_path,
76            table_dat,

78            selected_rows,
79        }
80    }

83impl WholeCellRead<'_> {
84    /// The whole stored cells of the selected rows, read row by row.
85    ///
86    /// A layout its data manager can only read as a whole column is refused:
87    /// a typed selected read holds at most the selected rows' cells.
88    fn cells(&self) -> Result<Vec<Option<ArrayValue>>, StorageError> {
89        if self.selected_rows.is_empty() {
90            return Ok(Vec::new());
91        }
92        self.storage
93            .load_selected_array_cells(
94                self.table_path,
95                self.table_dat,
96                self.column,
97                self.selected_rows,
98            )?
99            .ok_or_else(|| {
100                StorageError::FormatMismatch(format!(
101                    "typed selected reads of column '{}' need its cells read row by row, but \
102                     its data manager can read this layout only as a whole column",
103                    self.column
104                ))
105            })
106    }
107
108    /// Load a 2-D channel range in the stored element type, packed
109    /// `[channel][row][axis0]`; `None` when a selected cell is undefined.
110    pub(super) fn load_2d(
111        &self,
112        channel_start: usize,
113        channel_count: usize,
114    ) -> Result<Option<SelectedArray2DCells>, StorageError> {
115        load_2d(self.column, &self.cells()?, channel_start, channel_count)
116    }
117
118    /// Fill a 2-D channel-range destination; `None` when a selected cell is
119    /// undefined.
120    pub(super) fn fill_2d(
121        &self,
122        channel_start: usize,
123        channel_count: usize,
124        destination: SelectedArray2DCellsMut<'_>,
125    ) -> Result<Option<SelectedArray2DShape>, StorageError> {
126        fill_2d(
127            self.column,
128            &self.cells()?,
129            channel_start,
130            channel_count,
131            destination,
132        )
133    }
134
135    /// Load 1-D cells in the stored element type.
136    pub(super) fn load_1d(&self) -> Result<SelectedArray1DCells, StorageError> {
137        load_1d(self.column, &self.cells()?)
138    }

140    /// Fill a 1-D destination.

143        destination: SelectedArray1DCellsMut<'_>,

145        fill_1d(self.column, &self.cells()?, destination)

149/// An element type the packed layouts carry.
150trait Element: Copy {
151    fn cells(value: &ArrayValue) -> Option<&ArrayD<Self>>;
152}
153
154macro_rules! element {
155    ($type:ty, $variant:ident) => {
156        impl Element for $type {
157            fn cells(value: &ArrayValue) -> Option<&ArrayD<Self>> {
158                match value {
159                    ArrayValue::$variant(values) => Some(values),
160                    _ => None,
161                }
162            }
163        }
164    };
165}
166
167element!(bool, Bool);
168element!(f32, Float32);
169element!(f64, Float64);
170element!(Complex32, Complex32);
171element!(Complex64, Complex64);

173/// How a 2-D `[axis0, channel]` cell block is packed.
174#[derive(Clone, Copy)]
175enum Layout2D {
176    /// `[channel][row][axis0]`.
177    ChannelRow,
178    /// `[row][channel][axis0]`.
179    RowChannel,
180}
181
182fn typed<'a, T: Element>(
183    column: &str,
184    value: &'a ArrayValue,
185) -> Result<&'a ArrayD<T>, StorageError> {
186    T::cells(value).ok_or_else(|| {
187        StorageError::FormatMismatch(format!(
188            "selected column '{column}' does not have the requested element type"
189        ))
190    })
191}
192
193/// Pack 1-D cells as `[row][axis0]`; `None` when a selected cell is undefined.
194fn pack_1d<T: Element>(
195    column: &str,
196    cells: &[Option<ArrayValue>],
197    values: &mut Vec<T>,
198) -> Result<Option<SelectedArray1DShape>, StorageError> {
199    values.clear();
200    let mut axis0_count = None;
201    for cell in cells {
202        let Some(cell) = cell else {
203            return Ok(None);
204        };
205        let cell = typed::<T>(column, cell)?;
206        if cell.ndim() != 1 || *axis0_count.get_or_insert(cell.len()) != cell.len() {
207            return Err(StorageError::FormatMismatch(format!(
208                "selected 1-D column '{column}' has cells of differing shape"
209            )));
210        }
211        values.extend(cell.iter().copied());
212    }
213    Ok(Some(SelectedArray1DShape {
214        row_count: cells.len(),
215        axis0_count: axis0_count.unwrap_or(0),
216    }))
217}

219/// Pack channels `channel_start..channel_start + channel_count` of 2-D
220/// `[axis0, channel]` cells; `None` when a selected cell is undefined.
221fn pack_2d<T: Element>(
222    column: &str,
223    cells: &[Option<ArrayValue>],
224    channel_start: usize,
225    channel_count: usize,
226    layout: Layout2D,
227    values: &mut Vec<T>,
228) -> Result<Option<SelectedArray2DShape>, StorageError> {
229    values.clear();
230    let mut typed_cells = Vec::with_capacity(cells.len());
231    let mut axis0_count = None;
232    for cell in cells {
233        let Some(cell) = cell else {
234            return Ok(None);
235        };
236        let cell = typed::<T>(column, cell)?;
237        let shape = cell.shape();
238        if shape.len() != 2
239            || *axis0_count.get_or_insert(shape[0]) != shape[0]
240            || channel_start + channel_count > shape[1]
241        {
242            return Err(StorageError::FormatMismatch(format!(
243                "selected 2-D column '{column}' cells do not cover the requested channels"
244            )));
245        }
246        typed_cells.push(cell);
247    }
248    let axis0_count = axis0_count.unwrap_or(0);
249    let channels = channel_start..channel_start + channel_count;
250    values.reserve(cells.len() * channel_count * axis0_count);
251    match layout {
252        Layout2D::ChannelRow => {
253            for channel in channels {
254                for cell in &typed_cells {
255                    values.extend((0..axis0_count).map(|axis0| cell[[axis0, channel]]));
256                }
257            }
258        }
259        Layout2D::RowChannel => {
260            for cell in &typed_cells {
261                for channel in channels.clone() {
262                    values.extend((0..axis0_count).map(|axis0| cell[[axis0, channel]]));
263                }
264            }
265        }
266    }
267    Ok(Some(SelectedArray2DShape {
268        row_count: cells.len(),
269        axis0_count,
270        channel_count,
271    }))
272}

274/// Fill a 1-D destination from whole-cell reads.
275fn fill_1d(
276    column: &str,
277    cells: &[Option<ArrayValue>],
278    destination: SelectedArray1DCellsMut<'_>,
279) -> Result<SelectedArray1DShape, StorageError> {
280    match destination {
281        SelectedArray1DCellsMut::Bool(values) => pack_1d(column, cells, values),
282        SelectedArray1DCellsMut::Float32(values) => pack_1d(column, cells, values),
283        SelectedArray1DCellsMut::Float64(values) => pack_1d(column, cells, values),
284        SelectedArray1DCellsMut::Complex32(values) => pack_1d(column, cells, values),
285        SelectedArray1DCellsMut::Complex64(values) => pack_1d(column, cells, values),
286    }?
287    .ok_or_else(|| {
288        StorageError::FormatMismatch(format!(
289            "selected 1-D column '{column}' has undefined cells"
290        ))
291    })
292}

294/// Load 1-D cells in the element type they are stored with.
295fn load_1d(
296    column: &str,
297    cells: &[Option<ArrayValue>],
298) -> Result<SelectedArray1DCells, StorageError> {
299    macro_rules! load {
300        ($variant:ident) => {{
301            let mut values = Vec::new();
302            let shape = fill_1d(
303                column,
304                cells,
305                SelectedArray1DCellsMut::$variant(&mut values),
306            )?;
307            SelectedArray1DCells::$variant(SelectedArray1D::new(
308                shape.row_count,
309                shape.axis0_count,
310                values,
311            ))
312        }};
313    }
314    Ok(match cells.iter().flatten().next() {
315        Some(ArrayValue::Bool(_)) => load!(Bool),
316        Some(ArrayValue::Float32(_)) => load!(Float32),
317        Some(ArrayValue::Complex32(_)) => load!(Complex32),
318        Some(ArrayValue::Complex64(_)) => load!(Complex64),
319        _ => load!(Float64),
320    })
321}

323/// Fill a 2-D channel-range destination from whole-cell reads; `None` when
324/// a selected cell is undefined.
325fn fill_2d(
326    column: &str,
327    cells: &[Option<ArrayValue>],
328    channel_start: usize,
329    channel_count: usize,
330    destination: SelectedArray2DCellsMut<'_>,
331) -> Result<Option<SelectedArray2DShape>, StorageError> {
332    use Layout2D::{ChannelRow, RowChannel};
333    let (start, count) = (channel_start, channel_count);
334    match destination {
335        SelectedArray2DCellsMut::Bool(values) => {
336            pack_2d(column, cells, start, count, ChannelRow, values)
337        }
338        SelectedArray2DCellsMut::Float32(values) => {
339            pack_2d(column, cells, start, count, ChannelRow, values)
340        }
341        SelectedArray2DCellsMut::Float64(values) => {
342            pack_2d(column, cells, start, count, ChannelRow, values)
343        }
344        SelectedArray2DCellsMut::Complex32(values) => {
345            pack_2d(column, cells, start, count, ChannelRow, values)
346        }
347        SelectedArray2DCellsMut::Complex64(values) => {
348            pack_2d(column, cells, start, count, ChannelRow, values)
349        }
350        SelectedArray2DCellsMut::RowChannelBool(values) => {
351            pack_2d(column, cells, start, count, RowChannel, values)
352        }
353        SelectedArray2DCellsMut::RowChannelFloat32(values) => {
354            pack_2d(column, cells, start, count, RowChannel, values)
355        }
356        SelectedArray2DCellsMut::RowChannelComplex32(values) => {
357            pack_2d(column, cells, start, count, RowChannel, values)
358        }
359    }
360}

362/// Load a 2-D channel range in the element type it is stored with, packed
363/// `[channel][row][axis0]`; `None` when a selected cell is undefined.
364fn load_2d(
365    column: &str,
366    cells: &[Option<ArrayValue>],
367    channel_start: usize,
368    channel_count: usize,
369) -> Result<Option<SelectedArray2DCells>, StorageError> {
370    macro_rules! load {
371        ($variant:ident) => {{
372            let mut values = Vec::new();
373            fill_2d(
374                column,
375                cells,
376                channel_start,
377                channel_count,
378                SelectedArray2DCellsMut::$variant(&mut values),
379            )?
380            .map(|shape| {
381                SelectedArray2DCells::$variant(SelectedArray2D::new(
382                    shape.row_count,
383                    shape.axis0_count,
384                    shape.channel_count,
385                    values,
386                ))
387            })
388        }};
389    }
390    Ok(match cells.iter().flatten().next() {
391        Some(ArrayValue::Bool(_)) => load!(Bool),
392        Some(ArrayValue::Float32(_)) => load!(Float32),
393        Some(ArrayValue::Float64(_)) => load!(Float64),
394        Some(ArrayValue::Complex64(_)) => load!(Complex64),
395        _ => load!(Complex32),
396    })
397}

399#[cfg(test)]
400mod tests {
401    use super::*;
402    use ndarray::{IxDyn, ShapeBuilder};

404    /// A `[axis0, channel]` cell of 2 x 3 values `100 row + 10 channel + axis0`,
405    /// stored Fortran order as casacore stores it.
406    fn cell(row: usize) -> Option<ArrayValue> {
407        let mut cell = ArrayD::from_elem(IxDyn(&[2, 3]).f(), 0.0_f32);
408        for axis0 in 0..2 {
409            for channel in 0..3 {
410                cell[[axis0, channel]] = (100 * row + 10 * channel + axis0) as f32;
411            }
412        }
413        Some(ArrayValue::Float32(cell))
414    }
415
416    #[test]
417    fn two_dimensional_cells_pack_a_channel_range_in_both_layouts() {
418        let cells = [cell(0), cell(1)];
419        let mut row_channel = Vec::new();
420        let shape = fill_2d(
421            "DATA",
422            &cells,
423            1,
424            2,
425            SelectedArray2DCellsMut::RowChannelFloat32(&mut row_channel),
426        )
427        .expect("pack")
428        .expect("defined");
429        assert_eq!(
430            (shape.row_count, shape.axis0_count, shape.channel_count),
431            (2, 2, 2)
432        );
433        assert_eq!(
434            row_channel,
435            [10.0, 11.0, 20.0, 21.0, 110.0, 111.0, 120.0, 121.0]
436        );
437        let mut channel_row = Vec::new();
438        fill_2d(
439            "DATA",
440            &cells,
441            1,
442            2,
443            SelectedArray2DCellsMut::Float32(&mut channel_row),
444        )
445        .expect("pack")
446        .expect("defined");
447        assert_eq!(
448            channel_row,
449            [10.0, 11.0, 110.0, 111.0, 20.0, 21.0, 120.0, 121.0]
450        );
451    }

453    #[test]
454    fn undefined_and_mismatched_cells_are_reported() {
455        let mut values = Vec::new();
456        assert!(
457            fill_2d(
458                "FLAG",
459                &[None],
460                0,
461                1,
462                SelectedArray2DCellsMut::Bool(&mut values)
463            )
464            .expect("undefined is not an error")
465            .is_none()
466        );
467        let uvw = Some(ArrayValue::Float64(ArrayD::from_elem(IxDyn(&[3]), 1.5)));
468        let mut floats = Vec::new();
469        assert!(
470            fill_1d(
471                "UVW",

473                SelectedArray1DCellsMut::Float32(&mut floats)
474            )
475            .is_err()
476        );
477        let SelectedArray1DCells::Float64(loaded) =
478            load_1d("UVW", &[uvw.clone(), uvw]).expect("load")
479        else {
480            panic!("UVW loads as f64")
481        };
482        assert_eq!(loaded.values(), &[1.5; 6]);
483    }
