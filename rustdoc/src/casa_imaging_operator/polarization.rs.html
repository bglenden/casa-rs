12/// Feed basis of the selected correlations.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum FeedBasis {
15    /// Values are already Stokes parameters.
16    Stokes,
17    /// Orthogonal X/Y receptors.
18    Linear,
19    /// Orthogonal R/L receptors.
20    Circular,
21}
22
23/// One grid polarization plane in CASA's complex-image representation.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25pub enum GridPolarization {
26    /// One correlation (or one Stokes parameter when the data are Stokes).
27    Correlation(Correlation),
28    /// Stokes I gridded directly: both parallel hands feed one plane and
29    /// their weights both count in its `sumwt` (CASA `isIOnly`).
30    StokesI,
31}
32
33/// Correlation routing in both directions of the operator.
34///
35/// The adjoint grids each selected correlation into the grid plane
36/// [`PolarizationRouting::pol_map`] names and converts the complex image
37/// planes to the requested coordinates with
38/// [`PolarizationRouting::to_requested`] (CASA `StokesImageUtil::To`,
39/// real part taken). The forward transform expands requested model planes
40/// to grid planes with [`PolarizationRouting::from_requested`]
41/// (`StokesImageUtil::From`). `sumwt` of a requested Stokes plane is that of
42/// grid plane 0 (`StokesImageUtil::ToStokesSumWt`).
43#[derive(Clone, Debug, PartialEq)]
44pub struct PolarizationRouting {
45    correlations: Vec<Correlation>,
46    requested: Vec<PolarizationCoordinate>,
47    grid: Vec<GridPolarization>,
48    pol_map: Vec<Option<u8>>,
49    feed: FeedBasis,
50    to_requested: Vec<Complex64>,
51    to_requested_psf: Vec<Complex64>,
52    from_requested: Vec<Complex64>,
53    sumwt_source: Vec<usize>,
54}
55
56impl PolarizationRouting {
57    /// Route the selected `correlations` (block order) to the `requested`
58    /// image coordinates.
59    ///
60    /// Requested coordinates are all Stokes or all correlations, the
61    /// selected correlations share one feed basis, every grid plane CASA's
62    /// representation needs receives at least one selected correlation, and
63    /// requested correlations must be selected.
64    pub fn compile(
65        correlations: &[Correlation],
66        requested: &[PolarizationCoordinate],
67    ) -> Result<Self, OperatorError> {
68        let feed = feed_basis(correlations)?;
69        if requested.is_empty() {
70            return Err(OperatorError::Polarization {
71                reason: "no polarization coordinate requested",
72            });
73        }
74        if requested
75            .iter()
76            .enumerate()
77            .any(|(index, coordinate)| requested[..index].contains(coordinate))
78        {
79            return Err(OperatorError::Polarization {
80                reason: "requested polarization coordinates repeat",
81            });
82        }
83        let stokes_requested = requested.iter().all(|coordinate| is_stokes(*coordinate));
84        if !stokes_requested && requested.iter().any(|coordinate| is_stokes(*coordinate)) {
85            return Err(OperatorError::Polarization {
86                reason: "requested coordinates mix Stokes parameters and correlations",
87            });
88        }
89        let grid = if !stokes_requested {
90            requested
91                .iter()
92                .map(|coordinate| {
93                    let correlation = correlation_of(*coordinate);
94                    correlations
95                        .contains(&correlation)
96                        .then_some(GridPolarization::Correlation(correlation))
97                        .ok_or(OperatorError::Polarization {
98                            reason: "requested correlation is not selected",
99                        })
100                })
101                .collect::<Result<Vec<_>, _>>()?
102        } else if feed == FeedBasis::Stokes {
103            requested
104                .iter()
105                .map(|coordinate| {
106                    let correlation = correlation_of(*coordinate);
107                    correlations
108                        .contains(&correlation)
109                        .then_some(GridPolarization::Correlation(correlation))
110                        .ok_or(OperatorError::Polarization {
111                            reason: "requested Stokes parameter is not selected",
112                        })
113                })
114                .collect::<Result<Vec<_>, _>>()?
115        } else {
116            casa_correlation_representation(feed, requested)
117        };
118        let pol_map = correlations
119            .iter()
120            .map(|correlation| {
121                grid.iter()
122                    .position(|plane| match plane {
123                        GridPolarization::Correlation(target) => target == correlation,
124                        GridPolarization::StokesI => correlation.is_parallel_hand(),
125                    })
126                    .map(|index| index as u8)
127            })
128            .collect::<Vec<_>>();
129        if (0..grid.len()).any(|gpol| !pol_map.contains(&Some(gpol as u8))) {
130            return Err(OperatorError::Polarization {
131                reason: "selected correlations do not feed every grid polarization",
132            });
133        }
134        let to_requested = requested
135            .iter()
136            .flat_map(|coordinate| {
137                grid.iter()
138                    .map(|plane| to_coefficient(feed, *coordinate, *plane))
139            })
140            .collect();
141        let to_requested_psf = requested
142            .iter()
143            .flat_map(|coordinate| {
144                grid.iter()
145                    .map(|plane| psf_coefficient(feed, requested, *coordinate, *plane))
146            })
147            .collect();
148        let from_requested = grid
149            .iter()
150            .flat_map(|plane| {
151                requested
152                    .iter()
153                    .map(|coordinate| from_coefficient(feed, *plane, *coordinate))
154            })
155            .collect();
156        let sumwt_source = requested
157            .iter()
158            .map(|coordinate| {
159                if stokes_requested {
160                    0
161                } else {
162                    grid.iter()
163                        .position(|plane| {
164                            *plane == GridPolarization::Correlation(correlation_of(*coordinate))
165                        })
166                        .expect("requested correlations are grid planes")
167                }
168            })
169            .collect();
170        Ok(Self {
171            correlations: correlations.to_vec(),
172            requested: requested.to_vec(),
173            grid,
174            pol_map,
175            feed,
176            to_requested,
177            to_requested_psf,
178            from_requested,
179            sumwt_source,
180        })
181    }
182
183    /// Selected correlations in block order.
184    #[must_use]
185    pub fn correlations(&self) -> &[Correlation] {
186        &self.correlations
187    }
188
189    /// Requested image coordinates in output order.
190    #[must_use]
191    pub fn requested(&self) -> &[PolarizationCoordinate] {
192        &self.requested
193    }
194
195    /// Grid polarization planes.
196    #[must_use]
197    pub fn grid(&self) -> &[GridPolarization] {
198        &self.grid
199    }
200
201    /// Number of grid polarization planes.
202    #[must_use]
203    pub fn grid_pols(&self) -> usize {
204        self.grid.len()
205    }
206
207    /// Grid plane fed by each selected correlation (`None`: not gridded).
208    #[must_use]
209    pub fn pol_map(&self) -> &[Option<u8>] {
210        &self.pol_map
211    }
212
213    /// Feed basis of the selected correlations.
214    #[must_use]
215    pub const fn feed(&self) -> FeedBasis {
216        self.feed
217    }
218
219    /// Coefficient of grid plane `gpol` in requested plane `requested`; the
220    /// requested image is the real part of the sum.
221    #[must_use]
222    pub fn to_requested(&self, requested: usize, gpol: usize) -> Complex64 {
223        self.to_requested[requested * self.grid.len() + gpol]
224    }
225
226    /// Coefficient of grid plane `gpol` in the PSF image of requested plane
227    /// `requested` (CASA `StokesImageUtil::ToStokesPSF`): with one or two
228    /// requested Stokes parameters every plane takes the parallel-hand sum,
229    /// or the cross-hand sum for U and V on linear feeds and Q and U on
230    /// circular feeds; with three or four, every plane takes the first
231    /// parameter's conversion; a single-correlation grid, a Stokes grid or a
232    /// request with a non-Stokes coordinate takes the data conversion.
233    ///
234    /// The operator forms its weight image with these coefficients too,
235    /// whereas CASA applies `ToStokesPSF` to the PSF only; no kernel set
236    /// with a weight image is installed before the AW and mosaic operators.
237    #[must_use]
238    pub fn to_requested_psf(&self, requested: usize, gpol: usize) -> Complex64 {
239        self.to_requested_psf[requested * self.grid.len() + gpol]
240    }
241
242    /// Coefficient of requested model plane `requested` in grid plane `gpol`.
243    #[must_use]
244    pub fn from_requested(&self, gpol: usize, requested: usize) -> Complex64 {
245        self.from_requested[gpol * self.requested.len() + requested]
246    }
247
248    /// Grid plane whose `sumwt` normalises requested plane `requested`.
249    #[must_use]
250    pub fn sumwt_source(&self, requested: usize) -> usize {
251        self.sumwt_source[requested]
252    }
253}

255fn feed_basis(correlations: &[Correlation]) -> Result<FeedBasis, OperatorError> {
256    let Some(first) = correlations.first() else {
257        return Err(OperatorError::Polarization {
258            reason: "no correlation selected",
259        });
260    };
261    let feed = basis_of(*first)?;
262    for (index, correlation) in correlations.iter().enumerate() {
263        if basis_of(*correlation)? != feed {
264            return Err(OperatorError::Polarization {
265                reason: "selected correlations mix feed bases",
266            });
267        }
268        if correlations[..index].contains(correlation) {
269            return Err(OperatorError::Polarization {
270                reason: "selected correlations repeat",
271            });
272        }
273    }
274    Ok(feed)
275}
276
277fn basis_of(correlation: Correlation) -> Result<FeedBasis, OperatorError> {
278    match correlation {
279        Correlation::I | Correlation::Q | Correlation::U | Correlation::V => Ok(FeedBasis::Stokes),
280        Correlation::XX | Correlation::XY | Correlation::YX | Correlation::YY => {
281            Ok(FeedBasis::Linear)
282        }
283        Correlation::RR | Correlation::RL | Correlation::LR | Correlation::LL => {
284            Ok(FeedBasis::Circular)
285        }
286        _ => Err(OperatorError::Polarization {
287            reason: "selected correlation is not a Stokes, linear or circular product",
288        }),
289    }
290}

292const fn is_stokes(coordinate: PolarizationCoordinate) -> bool {
293    matches!(
294        coordinate,
295        PolarizationCoordinate::StokesI
296            | PolarizationCoordinate::StokesQ
297            | PolarizationCoordinate::StokesU
298            | PolarizationCoordinate::StokesV
299    )
300}
301
302const fn correlation_of(coordinate: PolarizationCoordinate) -> Correlation {
303    match coordinate {
304        PolarizationCoordinate::StokesI => Correlation::I,
305        PolarizationCoordinate::StokesQ => Correlation::Q,
306        PolarizationCoordinate::StokesU => Correlation::U,
307        PolarizationCoordinate::StokesV => Correlation::V,
308        PolarizationCoordinate::LinearXx => Correlation::XX,
309        PolarizationCoordinate::LinearXy => Correlation::XY,
310        PolarizationCoordinate::LinearYx => Correlation::YX,
311        PolarizationCoordinate::LinearYy => Correlation::YY,
312        PolarizationCoordinate::CircularRr => Correlation::RR,
313        PolarizationCoordinate::CircularRl => Correlation::RL,
314        PolarizationCoordinate::CircularLr => Correlation::LR,
315        PolarizationCoordinate::CircularLl => Correlation::LL,
316    }
317}

319/// CASA `StokesImageUtil::changeLabelsStokesToCorrStokes`: the complex
320/// grid planes that serve a Stokes request on linear or circular feeds.
321fn casa_correlation_representation(
322    feed: FeedBasis,
323    requested: &[PolarizationCoordinate],
324) -> Vec<GridPolarization> {
325    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
326    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
327    let planes = |list: &[Correlation]| {
328        list.iter()
329            .map(|correlation| GridPolarization::Correlation(*correlation))
330            .collect::<Vec<_>>()
331    };
332    let all = match feed {
333        FeedBasis::Linear => planes(&[XX, XY, YX, YY]),
334        FeedBasis::Circular | FeedBasis::Stokes => planes(&[RR, RL, LR, LL]),
335    };
336    match (feed, requested) {
337        (_, [StokesI]) => vec![GridPolarization::StokesI],
338        (FeedBasis::Circular, [StokesQ] | [StokesU] | [StokesQ, StokesU]) => planes(&[RL, LR]),
339        (FeedBasis::Circular, [StokesV] | [StokesI, StokesV]) => planes(&[RR, LL]),
340        (FeedBasis::Linear, [StokesQ] | [StokesI, StokesQ]) => planes(&[XX, YY]),
341        (FeedBasis::Linear, [StokesU] | [StokesV] | [StokesU, StokesV]) => planes(&[XY, YX]),
342        _ => all,
343    }
344}

346/// `StokesImageUtil::To` (`CStokesVector::applySlinInv` / `applyScircInv`):
347/// the coefficient of one grid plane in one requested plane.
348fn to_coefficient(
349    feed: FeedBasis,
350    requested: PolarizationCoordinate,
351    plane: GridPolarization,
352) -> Complex64 {
353    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
354    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
355    let one = Complex64::new(1.0, 0.0);
356    let half = Complex64::new(0.5, 0.0);
357    let half_i = Complex64::new(0.0, 0.5);
358    let correlation = match plane {
359        GridPolarization::StokesI => {
360            return if requested == StokesI {
361                one
362            } else {
363                Complex64::default()
364            };
365        }
366        GridPolarization::Correlation(correlation) => correlation,
367    };
368    if !is_stokes(requested) || feed == FeedBasis::Stokes {
369        return if correlation_of(requested) == correlation {
370            one
371        } else {
372            Complex64::default()
373        };
374    }
375    match (feed, requested, correlation) {
376        (FeedBasis::Linear, StokesI, XX | YY)
377        | (FeedBasis::Linear, StokesQ, XX)
378        | (FeedBasis::Linear, StokesU, XY | YX)
379        | (FeedBasis::Circular, StokesI, RR | LL)
380        | (FeedBasis::Circular, StokesQ, RL | LR)
381        | (FeedBasis::Circular, StokesV, RR) => half,
382        (FeedBasis::Linear, StokesQ, YY) | (FeedBasis::Circular, StokesV, LL) => -half,
383        (FeedBasis::Linear, StokesV, YX) | (FeedBasis::Circular, StokesU, LR) => half_i,
384        (FeedBasis::Linear, StokesV, XY) | (FeedBasis::Circular, StokesU, RL) => -half_i,
385        _ => Complex64::default(),
386    }
387}

389/// `StokesImageUtil::ToStokesPSF` on physical feeds: with one or two
390/// requested Stokes parameters, every PSF plane comes from one Stokes row
391/// of the data conversion (CASA applies the cross-hand rule to both planes
392/// whenever either requested parameter is a cross-hand one); with more, every
393/// plane is the first requested parameter's PSF (`sv(outMap(0))`), since the
394/// grid holds more than one correlation.
395fn psf_coefficient(
396    feed: FeedBasis,
397    requested_all: &[PolarizationCoordinate],
398    requested: PolarizationCoordinate,
399    plane: GridPolarization,
400) -> Complex64 {
401    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
402    if feed == FeedBasis::Stokes
403        || plane == GridPolarization::StokesI
404        || !requested_all
405            .iter()
406            .all(|coordinate| is_stokes(*coordinate))
407    {
408        return to_coefficient(feed, requested, plane);
409    }
410    if requested_all.len() > 2 {
411        return to_coefficient(feed, requested_all[0], plane);
412    }
413    let cross_hand = |coordinate: &PolarizationCoordinate| match feed {
414        FeedBasis::Linear => matches!(coordinate, StokesU | StokesV),
415        FeedBasis::Circular | FeedBasis::Stokes => matches!(coordinate, StokesQ | StokesU),
416    };
417    let row = if requested_all.iter().any(cross_hand) {
418        match feed {
419            FeedBasis::Linear => StokesU,
420            FeedBasis::Circular | FeedBasis::Stokes => StokesQ,
421        }
422    } else {
423        StokesI
424    };
425    to_coefficient(feed, row, plane)
426}

428/// `StokesImageUtil::From` (`CStokesVector::applySlin` / `applyScirc`): the
429/// coefficient of one requested model plane in one grid plane.
430fn from_coefficient(
431    feed: FeedBasis,
432    plane: GridPolarization,
433    requested: PolarizationCoordinate,
434) -> Complex64 {
435    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
436    use PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
437    let one = Complex64::new(1.0, 0.0);
438    let i = Complex64::new(0.0, 1.0);
439    let correlation = match plane {
440        GridPolarization::StokesI => {
441            return if requested == StokesI {
442                one
443            } else {
444                Complex64::default()
445            };
446        }
447        GridPolarization::Correlation(correlation) => correlation,
448    };
449    if !is_stokes(requested) || feed == FeedBasis::Stokes {
450        return if correlation_of(requested) == correlation {
451            one
452        } else {
453            Complex64::default()
454        };
455    }
456    match (feed, correlation, requested) {
457        (FeedBasis::Linear, XX, StokesI | StokesQ)
458        | (FeedBasis::Linear, YY, StokesI)
459        | (FeedBasis::Linear, XY | YX, StokesU)
460        | (FeedBasis::Circular, RR, StokesI | StokesV)
461        | (FeedBasis::Circular, LL, StokesI)
462        | (FeedBasis::Circular, RL | LR, StokesQ) => one,
463        (FeedBasis::Linear, YY, StokesQ) | (FeedBasis::Circular, LL, StokesV) => -one,
464        (FeedBasis::Linear, XY, StokesV) | (FeedBasis::Circular, RL, StokesU) => i,
465        (FeedBasis::Linear, YX, StokesV) | (FeedBasis::Circular, LR, StokesU) => -i,
466        _ => Complex64::default(),
467    }
