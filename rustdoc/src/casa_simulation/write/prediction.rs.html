2//! The prediction stage of `close`: the model DATA of each write batch's
3//! rows, degridded by the imaging runtime's block predictor on the worker
4//! team.
5//!
6//! A scan's rows are predicted from every `predict` step covering it, their
7//! predictions summed (one [`PredictionDomain`] per step, as CASA's mappers
8//! sum their degrids), each row reprojected to each step's tangent point by
9//! [`GridFtProjection`] as CASA's FTMachine does (`MosaicFTNew`'s
10//! `girarUVW`, the others' `rotateUVW`), and through the mosaic machine
11//! pointed by its kernels' phase ramp. Scans with the same steps, field and
12//! phase centre share a segment, and one predictor serves a segment; it
13//! is rebuilt when the batches reach a scan of another segment, after the
14//! segment's per-field grids are found or prepared ([`super::grids`]). A
15//! run of a segment's rows is predicted in blocks, straight into the
16//! batch's DATA, which a scan no step covers leaves zero: a block is the
17//! whole write batch when the budget allows, else about
18//! [`SMALL_BLOCK_BYTES`] of native rows.
19//!
20//! On Metal the stage also corrupts the rows it predicts: the device adds
21//! their scans' noise to the model in its memory, with the draws of
22//! `casa_numerics::cell_noise`, before the model is copied into DATA, so
23//! the host makes no pass of its own over those cells (#855). A patch after
24//! `openfromms` adds the DATA on disk after the prediction, and the host
25//! adds that noise after it ([`device_noise`]).
26//!
27//! The rows are as `observe` writes them: the row flag is FLAG_ROW, every
28//! sample is unflagged with unit weight, and an autocorrelation predicts the
29//! model's zero spacing (`GridFT`'s `usezero`). On the CPU a row's
30//! prediction depends only on the row and the steps, not on the workers,
31//! the blocks or the batches.

56/// The noise of scan `scan` in `plan` that a stage predicting the scan may
57/// add on the device ([`PredictionStage::predict`]): all of it, but none
58/// when `from_disk`, the DATA on disk being added after the prediction (a
59/// patch after `openfromms`), whose noise the host adds after it (#886
60/// review), or when gains or leakage corrupt the scan, whose noise follows
61/// them (`session::plan`).
62///
63/// Noise added before a value that is then cancelled keeps only the
64/// precision of the sum: beside visibilities of 2²³ σ and more it is
65/// rounded to steps of σ, or lost. The DATA on disk may hold such a value.
66/// The component lists' and total power's visibilities, which the host
67/// adds after the device's noise, are a sky model's, far below that:
68/// adding them after the noise rounds each sample once, independently, and
69/// the device's noise saves the host's pass (#886 follow-up: 0.34 s of a
70/// 2.1 s Metal `close`).
71pub(super) fn device_noise(plan: &DataPlan, scan: usize, from_disk: bool) -> &[NoiseApplication] {
72    if from_disk || plan.has_jones_for_scan(scan) {
73        &[]
74    } else {
75        plan.noise_for_scan(scan)
76    }
77}

79/// Bytes of the native rows of one prediction block, about, when the budget
80/// cannot hold a whole write batch's ([`super::resources`]). A block ends
81/// with the device's work drained and its predictions copied out, so on
82/// Metal a batch predicted as one block ran 1.5× faster than in blocks of
83/// this size (#855: 1,092-row batches of 4,096 channels, about 120 rows a
84/// small block).
85const SMALL_BLOCK_BYTES: u64 = 16 << 20;

92/// The scans with the same predict steps, field and phase centre: the
93/// steps whose domains a predictor sums (indices of the stage's
94/// predictions), the field, each row's rotation to each domain's tangent
95/// point, and the pointing offset each domain's kernels ramp by.
96struct Segment {
97    steps: Vec<usize>,
98    field: usize,
99    projections: Vec<GridFtProjection>,
100    pointing_offsets: Vec<[f64; 2]>,
101}
102
103/// The segment of `scan`, which the predict steps `steps` of `predictions`
104/// cover.
105///
106/// # Errors
107///
108/// A step without grids for the scan's field, which `predict` prepares, or
109/// holds the model to prepare, for every field it covers.
110fn segment_of(
111    predictions: &[&Prediction],
112    steps: std::ops::Range<usize>,
113    scan: &Scan,
114) -> SimulatorResult<Segment> {
115    let mut projections = Vec::with_capacity(steps.len());
116    let mut pointing_offsets = Vec::with_capacity(steps.len());
117    for step in steps.clone() {
118        let prediction = predictions[step];
119        if prediction.grids_for(scan.field_id).is_none() {
120            return Err(SimulatorError::raised(format!(
121                "predict prepared no grids for field {}",
122                scan.field_id
123            )));
124        }
125        // `MosaicFT` reprojects through `girarUVW`, the others through
126        // `rotateUVW`.
127        let project = match prediction.model.machine() {
128            FtMachine::Mosaic => GridFtProjection::mosaic_between_j2000_directions,
129            FtMachine::Grid(_) | FtMachine::WProject(_) => {
130                GridFtProjection::between_j2000_directions
131            }
132        };
133        projections.push(project(scan.phase_center_rad, prediction.tangent_rad));
134        pointing_offsets.push(prediction.pointing_offset_rad(scan.pointing_rad));
135    }
136    Ok(Segment {
137        steps: steps.collect(),
138        field: scan.field_id,
139        projections,
140        pointing_offsets,
141    })
142}

186impl<'s> PredictionStage<'s> {
187    /// The stage for `session`'s predict steps on cells of `(correlations,
188    /// channels)`, on `backend`, or on the CPU when a step projects w;
189    /// `None` when no step predicts.
190    ///
191    /// # Errors
192    ///
193    /// A covered scan's field without grids, which `predict` prepares, or
194    /// holds the model to prepare, for every field it covers.
195    pub(super) fn new(
196        session: &'s Simulator,
197        (correlations, channels): (usize, usize),
198        backend: BackendChoice,
199    ) -> SimulatorResult<Option<Self>> {
200        let plan = session.data_plan();
201        let predictions: Vec<&'s Prediction> = plan
202            .predictions()
203            .map(|(prediction, _)| prediction)
204            .collect();
205        if predictions.is_empty() {
206            return Ok(None);
207        }
208        // Metal predicts through `GridFT` only; a session that projects w
209        // predicts on the CPU (`predict` refused it on an explicit Metal
210        // backend).
211        let backend = if predictions
212            .iter()
213            .all(|prediction| prediction.model.machine().runs_on_metal())
214        {
215            backend
216        } else {
217            BackendChoice::Cpu
218        };
219        let mut keys = HashMap::new();
220        let mut segments = Vec::new();
221        let mut scan_segments = Vec::with_capacity(session.scans().len());
222        // Each per-field set's visits, by their first scans: a visit is a
223        // run of scans of one segment.
224        let mut visits: HashMap<GridKey, Vec<usize>> = HashMap::new();
225        for (index, scan) in session.scans().iter().enumerate() {
226            let steps = plan.predictions_for_scan(index);
227            if steps.is_empty() {
228                scan_segments.push(None);
229                continue;
230            }
231            // Rows are phased on their field (#884); the mosaic machine's
232            // kernels ramp to where the antennas point.
233            let key = (
234                steps,
235                scan.field_id,
236                [scan.phase_center_rad, scan.pointing_rad]
237                    .map(|direction| direction.map(f64::to_bits)),
238            );
239            let segment = match keys.get(&key) {
240                Some(segment) => *segment,
241                None => {
242                    segments.push(segment_of(&predictions, key.0.clone(), scan)?);
243                    keys.insert(key, segments.len() - 1);
244                    segments.len() - 1
245                }
246            };
247            if scan_segments.last() != Some(&Some(segment)) {
248                let Segment { steps, field, .. } = &segments[segment];
249                for &step in steps {
250                    if matches!(
251                        predictions[step].grids_for(*field),
252                        Some(FieldGrids::PerField)
253                    ) {
254                        visits.entry((step, *field)).or_default().push(index);
255                    }
256                }
257            }
258            scan_segments.push(Some(segment));
259        }
260        // An opened MeasurementSet may have a segment a row, gathered with
261        // room to spare.
262        segments.shrink_to_fit();
263        let frequencies = session
264            .spectral_windows()
265            .iter()
266            .map(|window| window.channel_frequencies_hz())
267            .collect();
268        let prepared_bytes = predictions
269            .iter()
270            .map(|prediction| prediction.prepared_bytes())
271            .sum();
272        // A device copies each per-field set it predicts from.
273        let copies = match backend {
274            BackendChoice::Cpu => 1,
275            BackendChoice::Metal => 2,
276        };
277        let costs = predictions
278            .iter()
279            .map(|prediction| {
280                if prediction.per_field() {
281                    copies * prediction.set_bytes()
282                } else {
283                    0
284                }
285            })
286            .collect();
287        let grids = FieldGridCache::new(costs, visits.into_iter().collect());
288        let largest_segment = segments
289            .iter()
290            .map(|segment| segment.steps.len())
291            .max()
292            .unwrap_or(0);
293        // A mosaic step classes the antennas of its time in first-seen
294        // order, so a later step's classes extend an earlier one's.
295        let antenna_types = predictions
296            .iter()
297            .map(|prediction| prediction.antenna_types())
298            .max_by_key(|types| types.len())
299            .unwrap_or_default()
300            .to_vec();
301        let most_terms = (0..session.scans().len())
302            .map(|scan| plan.noise_for_scan(scan).len())
303            .max()
304            .unwrap_or(0);
305        Ok(Some(Self {
306            backend,
307            correlations,
308            channels,
309            predictions,
310            segments,
311            scan_segments,
312            frequencies,
313            antenna_types,
314            prepared_bytes,
315            grids,
316            keep: Vec::with_capacity(largest_segment),
317            block_rows: 0,
318            workers: 0,
319            block: NativeBlock::default(),
320            fill_time: std::time::Duration::ZERO,
321            noise_terms: Vec::new(),
322            noise_rows: Vec::new(),
323            most_terms,
324            projections: Vec::new(),
325            predictor: None,
326            channel_indices: (0..channels as u32).collect(),
327            correlation_indices: (0..correlations as u32).collect(),
328        }))
329    }

409    /// Rows of a small block for write batches of `batch_rows` rows: as
410    /// many as [`SMALL_BLOCK_BYTES`] holds of native rows with all their
411    /// samples, at most a batch and at least one. The block itself shares
412    /// its rows' samples ([`NativeBlock::reset_shared`]), but what the
413    /// predictor holds per row (its model cells and their output) is of
414    /// that size.
415    pub(super) fn small_block_rows(&self, batch_rows: usize) -> usize {
416        let row = self.block_shape(1, self.largest_domains()).bytes().max(1);
417        usize::try_from(SMALL_BLOCK_BYTES / row)
418            .unwrap_or(usize::MAX)
419            .clamp(1, batch_rows.max(1))
420    }

422    /// Bytes the stage holds while `close` predicts blocks of `rows` rows
423    /// with `workers` workers: the predictor of the segment that needs most
424    /// ([`BlockPredictor::bytes_for`]), one block of native rows sharing
425    /// their samples ([`BlockShape::shared_bytes`]) and each row's
426    /// projections; the per-field grids of the segment that needs most,
427    /// and a set's preparation ([`Self::extra_grid_bytes`] are admitted
428    /// apart); on Metal the device copy of every model grid `predict`
429    /// prepared
430    /// ([`casa_imaging_operator::PreparedModelGrids::device_copy`]) and a
431    /// block's noise rows and the most noise terms of a scan, here and in
432    /// the predictor ([`BlockPredictor::noise_bytes`]); and what it keeps
433    /// across blocks ([`Self::state_bytes`]).
434    pub(super) fn bytes(&self, rows: usize, workers: usize) -> u64 {
435        let predictor = self
436            .segments
437            .iter()
438            .map(|segment| {
439                BlockPredictor::bytes_for(
440                    segment.steps.iter().map(|&step| {
441                        let model = &self.predictions[step].model;
442                        (&model.operator, &model.resampler)
443                    }),
444                    self.backend,
445                    self.block_shape(rows, segment.steps.len()),
446                    workers,
447                )
448            })
449            .max()
450            .unwrap_or(0);
451        let grids = self.minimum_grid_bytes() + self.preparation_bytes(workers);
452        let domains = self.largest_domains();
453        let shape = self.block_shape(rows, domains);
454        let projections = (domains * size_of::<DomainProjection>()) as u64;
455        let terms = self.most_terms;
456        let device = match self.backend {
457            BackendChoice::Cpu => 0,
458            BackendChoice::Metal => {
459                // The noise's rows and terms here, and the predictor's copy.
460                let noise = rows * size_of::<MetalNoiseRow>() + terms * size_of::<MetalNoiseTerm>();
461                self.prepared_bytes
462                    + noise as u64
463                    + BlockPredictor::noise_bytes(self.backend, shape, terms)
464            }
465        };
466        predictor + shape.shared_bytes() + projections + grids + device + self.state_bytes()
467    }

469    /// Bytes the stage keeps across blocks: the predict steps, each scan's
470    /// segment, of which an opened MeasurementSet may have one a row (#886
471    /// review), each segment's steps, the windows' channel frequencies, the
472    /// antennas' dish classes, the per-field grids' bookkeeping and a
473    /// selection's domains and keys, each as allocated.
474    fn state_bytes(&self) -> u64 {
475        let segments: usize = self
476            .segments
477            .iter()
478            .map(|segment| {
479                segment.steps.capacity() * size_of::<usize>()
480                    + segment.projections.capacity() * size_of::<GridFtProjection>()
481                    + segment.pointing_offsets.capacity() * size_of::<[f64; 2]>()
482            })
483            .sum();
484        let frequencies: usize = self
485            .frequencies
486            .iter()
487            .map(|window| window.capacity() * size_of::<f64>())
488            .sum();
489        let selection =
490            self.largest_domains() * (size_of::<PredictionDomain<'_>>() + size_of::<GridKey>());
491        (self.predictions.capacity() * size_of::<&Prediction>()
492            + self.scan_segments.capacity() * size_of::<Option<usize>>()
493            + self.segments.capacity() * size_of::<Segment>()
494            + segments
495            + self.frequencies.capacity() * size_of::<Vec<f64>>()
496            + frequencies
497            + self.antenna_types.capacity()
498            + selection) as u64
499            + self.grids.state_bytes()
500    }

502    /// Allocate the block of `rows` rows, predicted with `workers` workers,
503    /// as [`Self::bytes`] charged it, and give the per-field grids
504    /// `grid_bytes`, at least the [`Self::bytes`] charged them and at most
505    /// [`Self::extra_grid_bytes`] more.
506    pub(super) fn start(&mut self, rows: usize, workers: usize, grid_bytes: u64) {
507        self.grids.set_capacity(grid_bytes);
508        self.block_rows = rows.max(1);
509        self.workers = workers.max(1);
510        let domains = self.largest_domains();
511        self.block = NativeBlock::shared_with_capacity(self.block_shape(self.block_rows, domains));
512        self.projections = Vec::with_capacity(domains);
513        if self.backend == BackendChoice::Metal {
514            self.noise_rows = Vec::with_capacity(self.block_rows);
515            self.noise_terms = Vec::with_capacity(self.most_terms);
516        }
517    }

519    /// Whether the stage adds `noise`, a predicted run's, itself, on the
520    /// device, as it predicts the run ([`Self::predict`]). That happens on
521    /// Metal, where the model is formed in device memory, when every term
522    /// has one σ for every row (`simplenoise`) and draws there exactly as on
523    /// the host ([`casa_numerics::cell_noise::metal_draws_exactly`]);
524    /// otherwise the host adds it after. A Tsys term, whose σ varies by row
525    /// (and for `tsys-atm` by channel), is the host's: the device's terms
526    /// hold one σ each.
527    pub(super) fn noises_on_device(&self, noise: &[NoiseApplication]) -> bool {
528        self.backend == BackendChoice::Metal
529            && noise.iter().all(|application| {
530                application
531                    .uniform()
532                    .is_some_and(|noise| metal_draws_exactly(noise.sigma()))
533            })
534    }

536    /// Whether the stage added `noise`, the [`device_noise`] it was given
537    /// for a run of scan `scan`, as it predicted the run: the host then
538    /// adds none.
539    pub(super) fn noised(&self, scan: usize, noise: &[NoiseApplication]) -> bool {
540        !noise.is_empty() && self.predicts(scan) && self.noises_on_device(noise)
541    }

543    /// Predict each run of `runs`, (scan, first row, rows) of `rows`, into
544    /// `data` (`correlations × channels` samples a row, the rows of `rows`
545    /// from the first); the batch's first row is MAIN row `first_row`. Runs
546    /// of scans no step covers are left as they are. When the stage
547    /// [`Self::noises_on_device`], a predicted run also gets its scan's
548    /// [`device_noise`], `noise(scan)`, on the device before it is copied
549    /// out.
550    pub(super) fn predict<'n>(
551        &mut self,
552        team: &WorkerTeam,
553        (scans, rows): (&[Scan], &[RowGeometry]),
554        runs: &[(usize, usize, usize)],
555        first_row: u64,
556        data: &mut [Complex32],
557        noise: impl Fn(usize) -> &'n [NoiseApplication],
558    ) -> SimulatorResult<()> {
559        let samples = self.correlations * self.channels;
560        for &(scan, start, count) in runs {
561            let Some(segment) = self.scan_segments[scan] else {
562                continue;
563            };
564            self.select(segment, scan, team)?;
565            self.noise_terms.clear();
566            if self.noises_on_device(noise(scan)) {
567                // A term of σ = 0 adds nothing on either side.
568                self.noise_terms.extend(
569                    noise(scan)
570                        .iter()
571                        .filter_map(NoiseApplication::uniform)
572                        .filter(|noise| noise.sigma() != 0.0)
573                        .map(|noise| MetalNoiseTerm {
574                            key: noise.key(),
575                            sigma: noise.sigma(),
576                        }),
577                );
578            }
579            let scan = &scans[scan];
580            let end = start + count;
581            let mut block_start = start;
582            while block_start < end {
583                let block_end = (block_start + self.block_rows).min(end);
584                let started = std::time::Instant::now();
585                self.fill_block(
586                    segment,
587                    scan,
588                    &rows[block_start..block_end],
589                    first_row + block_start as u64,
590                );
591                self.fill_time += started.elapsed();
592                let Some((_, predictor)) = &mut self.predictor else {
593                    unreachable!("selected above");
594                };
595                let out = &mut data[block_start * samples..block_end * samples];
596                let predicted = if self.noise_terms.is_empty() {
597                    predictor.predict_into(&self.block, team, out)
598                } else {
599                    self.noise_rows.clear();
600                    self.noise_rows.extend(
601                        rows[block_start..block_end]
602                            .iter()
603                            .zip(first_row + block_start as u64..)
604                            .map(|(row, number)| MetalNoiseRow {
605                                row: [number as u32, (number >> 32) as u32],
606                                autocorrelation: u32::from(row.antenna1 == row.antenna2),
607                            }),
608                    );
609                    let noise = DeviceNoise {
610                        rows: &self.noise_rows,
611                        terms: &self.noise_terms,
612                    };
613                    predictor.predict_into_with_noise(&self.block, team, out, noise)
614                };
615                predicted.map_err(|error| SimulatorError::raised(format!("predict: {error}")))?;
616                block_start = block_end;
617            }
618        }
619        Ok(())
620    }

622    /// Make the predictor `segment`'s, unless it is, for the visit at scan
623    /// `scan`: its per-field grids found or prepared on `team`
624    /// ([`FieldGridCache::grids`]).
625    fn select(&mut self, segment: usize, scan: usize, team: &WorkerTeam) -> SimulatorResult<()> {
626        if self
627            .predictor
628            .as_ref()
629            .is_some_and(|(current, _)| *current == segment)
630        {
631            return Ok(());
632        }
633        // The previous predictor is dropped first, so at most one is held,
634        // and the grids it kept can be evicted.
635        self.predictor = None;
636        let Segment { steps, field, .. } = &self.segments[segment];
637        let field = *field;
638        self.keep.clear();
639        self.keep.extend(steps.iter().map(|&step| (step, field)));
640        let mut domains = Vec::with_capacity(steps.len());
641        for &step in steps {
642            let prediction = self.predictions[step];
643            let model = match prediction.grids_for(field) {
644                Some(FieldGrids::Prepared(grids)) => DomainModel::Borrowed(grids),
645                Some(FieldGrids::PerField) => DomainModel::Shared(self.grids.grids(
646                    (step, field),
647                    (scan, &self.keep),
648                    || prediction.prepare_field(field, team),
649                )?),
650                // `segment_of` found every step's.
651                None => unreachable!("a segment's steps have its field's grids"),
652            };
653            domains.push(PredictionDomain {
654                operator: &prediction.model.operator,
655                resampler: &prediction.model.resampler,
656                model,
657            });
658        }
659        let predictor = BlockPredictor::new(
660            &domains,
661            self.backend,
662            self.block_shape(self.block_rows, domains.len()),
663            self.workers,
664            tracing::enabled!(target: METAL_STAGE_TIMING, tracing::Level::INFO),
665        )
666        .map_err(|error| SimulatorError::raised(format!("the predictor: {error}")))?;
667        // The predictor keeps its own domains, and with them the grids.
668        drop(domains);
669        self.predictor = Some((segment, predictor));
670        Ok(())
671    }

673    /// Fill the block with `rows` of `scan`, the first MAIN row
674    /// `first_row`, projected on `segment`'s domains.
675    fn fill_block(&mut self, segment: usize, scan: &Scan, rows: &[RowGeometry], first_row: u64) {
676        let segment = &self.segments[segment];
677        // Every row of the scan has its window's frequencies, and unit
678        // weights and no sample flags, so the block keeps them once.
679        self.block.reset_shared(
680            segment.steps.len(),
681            &self.channel_indices,
682            &self.correlation_indices,
683            &self.frequencies[scan.spectral_window_id],
684        );
685        let antenna_type = |antenna: usize| {
686            // A mosaic step classed every antenna of the scans it covers.
687            debug_assert!(
688                self.antenna_types.is_empty() || antenna < self.antenna_types.len(),
689                "antenna {antenna} has no dish class"
690            );
691            self.antenna_types.get(antenna).copied().unwrap_or(0)
692        };
693        for (offset, row) in rows.iter().enumerate() {
694            self.projections.clear();
695            self.projections.extend(
696                segment
697                    .projections
698                    .iter()
699                    .zip(&segment.pointing_offsets)
700                    .map(|(projection, pointing_offset_rad)| {
701                        let (uvw_m, phase_shift_m) = projection.apply(row.uvw_m);
702                        DomainProjection {
703                            uvw_m,
704                            phase_shift_m,
705                            pointing_offset_rad: *pointing_offset_rad,
706                        }
707                    }),
708            );
709            let header = NativeRowHeader {
710                row_flag: row.flagged,
711                context: RowContext {
712                    original_uvw_m: Some(row.uvw_m),
713                    time_s: 0.0,
714                    antennas: [row.antenna1 as u32, row.antenna2 as u32],
715                    antenna_types: [antenna_type(row.antenna1), antenna_type(row.antenna2)],
716                    parallactic_angle_rad: [0.0, 0.0],
717                    field: scan.field_id as u32,
718                    spectral_window: scan.spectral_window_id as u32,
719                },
720                address: RowAddress {
721                    physical_row: first_row + offset as u64,
722                    data_description: scan.spectral_window_id as i32,
723                },
724            };
725            self.block.push_shared_row(header, &self.projections);
726        }
727    }
