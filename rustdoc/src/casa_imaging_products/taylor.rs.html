36impl TaylorProducts {
37    pub(crate) fn build(
38        inputs: &ContinuumProductInputs<'_>,
39        psf_cutoff: f32,
40        primary_beam_model: Option<AnalyticPrimaryBeamModel>,
41    ) -> Result<Self, ProductsError> {
42        let state = inputs.normal_state();
43        let shape = state.shape();
44        let cells = shape[0] * shape[1];
45        let terms = state.coefficient_term_count();
46        let moments = state.normal_moment_count();
47        if terms < 2
48            || moments != terms.saturating_mul(2).saturating_sub(1)
49            || inputs.final_model().shape().coefficients() != terms
50        {
51            return Err(ProductsError::SourceLineageMismatch);
52        }
53        let principal_sum_weight = state
54            .normal_moment(0)
55            .ok_or(ProductsError::SourceLineageMismatch)?
56            .sum_weight();
57        if !(principal_sum_weight.is_finite() && principal_sum_weight > 0.0) {
58            return Err(ProductsError::SourceLineageMismatch);
59        }
60        let normalization = inputs.problem().products().normalization();
61        let mut psf = Vec::with_capacity(moments);
62        let mut weight: Vec<Vec<f32>> = Vec::with_capacity(moments);
63        let mut sum_weights = Vec::with_capacity(moments);
64        for moment in 0..moments {
65            let source = state
66                .normal_moment(moment)
67                .ok_or(ProductsError::SourceLineageMismatch)?;
68            psf.push(normalize_plane(
69                &source
70                    .normal_approximation()
71                    .iter()
72                    .map(|value| value.re as f32)
73                    .collect::<Vec<_>>(),
74                ProductNormalization::UnitResponse,
75                principal_sum_weight,
76            )?);
77            weight.push(
78                source
79                    .sensitivity()
80                    .iter()
81                    .map(|value| *value as f32)
82                    .collect(),
83            );
84            sum_weights.push(source.sum_weight() as f32);
85        }
86        let residual = (0..terms)
87            .map(|term| {
88                let source = state
89                    .coefficient_term(term)
90                    .ok_or(ProductsError::SourceLineageMismatch)?;
91                normalize_plane(
92                    &source
93                        .residual()
94                        .iter()
95                        .map(|value| value.re as f32)
96                        .collect::<Vec<_>>(),
97                    normalization,
98                    principal_sum_weight,
99                )
100            })
101            .collect::<Result<Vec<_>, _>>()?;
102        let model = (0..terms)
103            .map(|term| model_term(inputs, term, shape))
104            .collect::<Result<Vec<_>, _>>()?;
105
106        let peak = psf[0]
107            .iter()
108            .enumerate()
109            .max_by(|(_, left), (_, right)| left.abs().total_cmp(&right.abs()))
110            .map(|(index, _)| index)
111            .ok_or(ProductsError::SourceLineageMismatch)?;
112        let mut normal = vec![0.0; terms * terms];
113        for row in 0..terms {
114            for column in 0..terms {
115                normal[row * terms + column] = f64::from(psf[row + column][peak]);
116            }
117        }
118        let mut inverse = vec![0.0; terms * terms];
119        for column in 0..terms {
120            let mut unit = vec![0.0; terms];
121            unit[column] = 1.0;
122            let solution = solve_symmetric_ldlt_casacore_dynamic(normal.clone(), &unit)
123                .ok_or_else(|| {
124                    ProductsError::BeamFitFailed("singular Taylor normal block".into())
125                })?;
126            for row in 0..terms {
127                inverse[row * terms + column] = solution[row];
128            }
129        }
130        let mut principal_residual = vec![vec![0.0; cells]; terms];
131        for cell in 0..cells {
132            for row in 0..terms {
133                principal_residual[row][cell] = (0..terms)
134                    .map(|column| inverse[row * terms + column] * f64::from(residual[column][cell]))
135                    .sum::<f64>() as f32;
136            }
137        }
138
139        let needs_beam = inputs
140            .problem()
141            .product_graph()
142            .publication()
143            .members()
144            .iter()
145            .any(|ordinal| {
146                inputs.problem().product_graph().nodes()[ordinal.ordinal()].beam()
147                    != casa_imaging_model::ProductBeamRule::None
148            });
149        let fitted_beam = needs_beam
150            .then(|| fit_restoring_beam(&psf[0], shape, inputs.cell_size_rad(), psf_cutoff))
151            .transpose()?;
152        let restoring_beam = match inputs.problem().products().restoring_beam() {
153            RestoringBeamPolicy::None => None,
154            RestoringBeamPolicy::PerPlane | RestoringBeamPolicy::Common => fitted_beam,
155        };
156        let restored = if let Some(beam) = restoring_beam {
157            let kernel = gaussian_beam_image(shape, &beam, inputs.cell_size_rad());
158            model
159                .iter()
160                .zip(&principal_residual)
161                .map(|(model, residual)| {
162                    let mut image =
163                        fft_convolve(model, kernel.as_slice().expect("contiguous"), shape);
164                    image
165                        .iter_mut()
166                        .zip(residual)
167                        .for_each(|(image, residual)| *image += residual);
168                    image
169                })
170                .collect()
171        } else {
172            vec![vec![0.0; cells]; terms]
173        };
174
175        let requests_primary_beam = inputs
176            .problem()
177            .product_graph()
178            .publication()
179            .members()
180            .iter()
181            .any(|ordinal| {
182                matches!(
183                    inputs.problem().product_graph().nodes()[ordinal.ordinal()].role(),
184                    ProductRole::PrimaryBeam(_) | ProductRole::PbCorrectedImage(_)
185                )
186            });
187        let pb0 = match primary_beam_model {
188            Some(AnalyticPrimaryBeamModel::CasaEvlaCommon) => {
189                analytic_evla_common_primary_beam(inputs, shape)?
190            }
191            None if requests_primary_beam => return Err(ProductsError::UnsupportedProblem),
192            None => primary_beam_from_weight(&weight[0])?,
193        };
194        let mut primary_beam = vec![vec![0.0; cells]; terms];
195        primary_beam[0] = pb0.clone();
196
197        let validity = inputs.problem().products().validity();
198        let pb_policy = validity.primary_beam();
199        if pb_policy.comparison() != ProductSupportComparison::StrictlyGreater
200            || pb_policy.blanking() != ProductBlankingPolicy::ZeroAndFalseMask
201        {
202            return Err(ProductsError::UnsupportedProblem);
203        }
204        let primary_beam_validity = pb0
205            .iter()
206            .map(|value| value.is_finite() && *value > pb_policy.cutoff())
207            .collect::<Vec<_>>();
208        let pb_corrected = restored
209            .iter()
210            .map(|image| {
211                image
212                    .iter()
213                    .zip(&pb0)
214                    .zip(&primary_beam_validity)
215                    .map(|((value, pb), valid)| if *valid { *value / *pb } else { 0.0 })
216                    .collect()
217            })
218            .collect::<Vec<_>>();
219
220        let taylor_policy = validity.taylor();
221        if taylor_policy.reference()
222            != TaylorSupportReference::PrincipalResidualTaylor0PositiveMaximum
223            || taylor_policy.comparison() != ProductSupportComparison::StrictlyGreater
224            || taylor_policy.blanking() != ProductBlankingPolicy::ZeroAndFalseMask
225        {
226            return Err(ProductsError::UnsupportedProblem);
227        }
228        let (alpha, alpha_error, alpha_validity) = taylor_alpha_products(
229            &principal_residual[0],
230            &principal_residual[1],
231            &restored[0],
232            &restored[1],
233            taylor_policy.peak_fraction(),
234        );
235        let clean_mask = weight[0]
236            .iter()
237            .enumerate()
238            .map(|(index, value)| {
239                let selected = inputs
240                    .reconstruction_mask()
241                    .is_none_or(|mask| mask.support()[index]);
242                (selected && value.is_finite() && *value > 0.0) as u8 as f32
243            })
244            .collect();
245
246        Ok(Self {
247            shape,
248            psf,
249            residual,
250            model,
251            restored,
252            weight,
253            sum_weights,
254            primary_beam,
255            pb_corrected,
256            alpha,
257            alpha_error,
258            alpha_validity,
259            primary_beam_validity,
260            clean_mask,
261            fitted_beam,
262            restoring_beam,
263        })
264    }

274    pub(crate) fn payload(&self, role: ProductRole) -> Result<Vec<f32>, ProductsError> {
275        let term = |term: ProductTerm| match term {
276            ProductTerm::Taylor(term) => Ok(term),
277            ProductTerm::Single
278            | ProductTerm::Continuum(_)
279            | ProductTerm::Line
280            | ProductTerm::Total
281            | ProductTerm::JointNormal { .. } => Err(ProductsError::UnsupportedProductRole {
282                role,
283                catalog: crate::CONTINUUM_ALGORITHM_CATALOG_VERSION,
284            }),
285        };
286        match role {
287            ProductRole::Psf(value) => self.psf.get(term(value)?),
288            ProductRole::Residual(value) => self.residual.get(term(value)?),
289            ProductRole::Model(value) => self.model.get(term(value)?),
290            ProductRole::RestoredImage(value) => self.restored.get(term(value)?),
291            ProductRole::Weight(value) => self.weight.get(term(value)?),
292            ProductRole::PrimaryBeam(value) => self.primary_beam.get(term(value)?),
293            ProductRole::PbCorrectedImage(value) => self.pb_corrected.get(term(value)?),
294            ProductRole::SpectralIndex => return Ok(self.alpha.clone()),
295            ProductRole::SpectralIndexError => return Ok(self.alpha_error.clone()),
296            ProductRole::CleanMask => return Ok(self.clean_mask.clone()),
297            ProductRole::Sensitivity => self.weight.first(),
298            ProductRole::SumWeights(value) => {
299                return self
300                    .sum_weights
301                    .get(term(value)?)
302                    .copied()
303                    .map(|value| vec![value])
304                    .ok_or(ProductsError::SourceLineageMismatch);
305            }
306            _ => None,
307        }
308        .cloned()
309        .ok_or(ProductsError::UnsupportedProductRole {
310            role,
311            catalog: crate::CONTINUUM_ALGORITHM_CATALOG_VERSION,
312        })
313    }

315    pub(crate) fn validity(&self, rule: ProductValidityRule) -> Result<Vec<bool>, ProductsError> {
316        match rule {
317            ProductValidityRule::All | ProductValidityRule::FinalNormalState => {
318                Ok(vec![true; self.shape[0] * self.shape[1]])
319            }
320            ProductValidityRule::PrimaryBeam(_) => Ok(self.primary_beam_validity.clone()),
321            ProductValidityRule::Taylor(_) => Ok(self.alpha_validity.clone()),
322            ProductValidityRule::TaylorAndPrimaryBeam { .. } => Ok(self
323                .alpha_validity
324                .iter()
325                .zip(&self.primary_beam_validity)
326                .map(|(taylor, pb)| *taylor && *pb)
327                .collect()),
328        }
329    }
330}

332fn taylor_alpha_products(
333    principal_residual_taylor0: &[f32],
334    principal_residual_taylor1: &[f32],
335    restored_taylor0: &[f32],
336    restored_taylor1: &[f32],
337    peak_fraction: f32,
338) -> (Vec<f32>, Vec<f32>, Vec<bool>) {
339    let cells = restored_taylor0.len();
340    let mut alpha = vec![0.0; cells];
341    let mut alpha_error = vec![0.0; cells];
342    let mut validity = vec![false; cells];
343    let Some(positive_maximum) = principal_residual_taylor0
344        .iter()
345        .copied()
346        .filter(|value| value.is_finite() && *value > 0.0)
347        .reduce(f32::max)
348    else {
349        return (alpha, alpha_error, validity);
350    };
351    let threshold = positive_maximum * peak_fraction;
352    for cell in 0..cells {
353        let image0 = restored_taylor0[cell];
354        if !(image0.is_finite() && image0 > threshold) {
355            continue;
356        }
357        let image1 = restored_taylor1[cell];
358        let residual0 = principal_residual_taylor0[cell];
359        let residual1 = principal_residual_taylor1[cell];
360        alpha[cell] = image1 / image0;
361        alpha_error[cell] =
362            ((image1 * residual0 / image0.powi(2)).powi(2) + (residual1 / image0).powi(2)).sqrt();
363        validity[cell] = true;
364    }
365    (alpha, alpha_error, validity)
366}

368fn primary_beam_from_weight(weight: &[f32]) -> Result<Vec<f32>, ProductsError> {
369    // Retained only for product graphs that do not publish PB: CASA
370    // SIImageStore::makePBFromWeight normalizes the principal weight image.
371    let scale = weight
372        .iter()
373        .copied()
374        .map(f32::abs)
375        .fold(0.0_f32, f32::max)
376        .sqrt();
377    if !(scale.is_finite() && scale > 0.0) {
378        return Err(ProductsError::GeneratedNonfinite);
379    }
380    Ok(weight
381        .iter()
382        .map(|value| value.abs().sqrt() / scale)
383        .collect())
384}

386fn analytic_evla_common_primary_beam(
387    inputs: &ContinuumProductInputs<'_>,
388    shape: [usize; 2],
389) -> Result<Vec<f32>, ProductsError> {
390    let domain = inputs
391        .problem()
392        .geometry()
393        .domains()
394        .first()
395        .ok_or(ProductsError::UnsupportedProblem)?;
396    let direction = domain.direction();
397    let reference_pixel = direction.reference_pixel();
398    let increment_rad = direction.increment_rad();
399    let frequency_hz = match inputs.problem().geometry().spectral().wcs() {
400        SpectralWcs::Linear {
401            reference_frequency_hz,
402            ..
403        } => *reference_frequency_hz,
404        SpectralWcs::Tabular { .. } => return Err(ProductsError::UnsupportedProblem),
405    };
406    let coefficients = nearest_evla_common_coefficients(frequency_hz * 1.0e-6)
407        .ok_or(ProductsError::UnsupportedProblem)?;
408    let mut values = vec![0.0; shape[0] * shape[1]];
409    for x in 0..shape[0] {
410        for y in 0..shape[1] {
411            let longitude = (x as f64 - reference_pixel[0]) * increment_rad[0];
412            let latitude = (y as f64 - reference_pixel[1]) * increment_rad[1];
413            values[x * shape[1] + y] =
414                evla_common_power_pattern(longitude.hypot(latitude), frequency_hz, coefficients);
415        }
416    }
417    Ok(values)
418}

420fn evla_common_power_pattern(radius_rad: f64, frequency_hz: f64, coefficients: [f64; 4]) -> f32 {
421    if !(radius_rad.is_finite()
422        && radius_rad >= 0.0
423        && frequency_hz.is_finite()
424        && frequency_hz > 0.0)
425    {
426        return 0.0;
427    }
428    // CASA PBMath1DEVLA::nearestVPArray(), PBMath1DPoly::fillPBArray(), and
429    // PBMath1D::apply() use 10,000 samples and integer-truncated radial lookup.
430    let radius_arcmin_ghz = radius_rad.to_degrees() * 60.0 * (frequency_hz / 1.0e9);
431    if radius_arcmin_ghz > 58.0 {
432        return 0.0;
433    }
434    let inverse_increment_radius = 9_999.0 / 58.0;
435    let sample_index = (radius_arcmin_ghz * inverse_increment_radius).floor();
436    let sampled_radius_arcmin_ghz = sample_index / inverse_increment_radius;
437    let x2 = sampled_radius_arcmin_ghz * sampled_radius_arcmin_ghz;
438    let mut response = 0.0;
439    let mut power = 1.0;
440    for coefficient in coefficients {
441        response += coefficient * power;
442        power *= x2;
443    }
444    if response <= 0.0 {
445        0.0
446    } else {
447        response as f32
448    }
449}

451fn nearest_evla_common_coefficients(frequency_mhz: f64) -> Option<[f64; 4]> {
452    // Preserve PBMath1DEVLA::feed() precedence in its narrow band overlaps and
453    // limitFreqForBand() clamping. Frequencies outside the represented CASA
454    // bands fail closed instead of silently borrowing another band's beam.
455    let (frequency_mhz, coefficients) = if frequency_mhz > 900.0 && frequency_mhz < 2003.0 {
456        (
457            frequency_mhz.clamp(1040.0, 2000.0),
458            EVLA_L_BAND_COEFFICIENTS,
459        )
460    } else if frequency_mhz > 1990.0 && frequency_mhz < 4001.0 {
461        (
462            frequency_mhz.clamp(2052.0, 3948.0),
463            EVLA_S_BAND_COEFFICIENTS,
464        )
465    } else if frequency_mhz > 3990.0 && frequency_mhz < 8001.0 {
466        (
467            frequency_mhz.clamp(4052.0, 7948.0),
468            EVLA_C_BAND_COEFFICIENTS,
469        )
470    } else {
471        return None;
472    };
473    let mut best = coefficients[0].1;
474    let mut best_delta_mhz = f64::INFINITY;
475    for &(candidate_frequency_mhz, candidate_coefficients) in coefficients {
476        let delta_mhz = (frequency_mhz - candidate_frequency_mhz).abs();
477        if delta_mhz < best_delta_mhz {
478            best_delta_mhz = delta_mhz;
479            best = candidate_coefficients;
480        }
481    }
482    Some(best)
483}

556fn model_term(
557    inputs: &ContinuumProductInputs<'_>,
558    coefficient: usize,
559    shape: [usize; 2],
560) -> Result<Vec<f32>, ProductsError> {
561    use casa_imaging_model::ModelCell;
562
563    let model = inputs.final_model();
564    if model.shape().domains().len() != 1
565        || model.shape().polarizations() != 1
566        || model.shape().domains()[0].pixels() != shape
567    {
568        return Err(ProductsError::SourceLineageMismatch);
569    }
570    let mut plane = vec![0.0; shape[0] * shape[1]];
571    for y in 0..shape[1] {
572        for x in 0..shape[0] {
573            let index = model
574                .shape()
575                .flat_index(ModelCell::new(0, coefficient, 0, [x, y]))
576                .ok_or(ProductsError::SourceLineageMismatch)?;
577            plane[x * shape[1] + y] = model.samples()[index].value().value() as f32;
578        }
579    }
580    Ok(plane)
581}
