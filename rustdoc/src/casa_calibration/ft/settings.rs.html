14/// Legacy `SDGrid` convolution selected by `imager.setoptions`.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum ImagerSingleDishKernel {
17    /// Nearest-cell box convolution.
18    Box,
19    /// Prolate spheroidal convolution.
20    Spheroidal,
21    /// Gaussian convolution.
22    Gaussian,
23    /// Gaussian-times-jinc convolution.
24    GaussianJinc,
25    /// Telescope primary beam.
26    PrimaryBeam,
27}
28
29impl ImagerSingleDishKernel {
30    /// Parse a CASA single-dish grid function, case independently.
31    pub fn from_name(name: &str) -> Option<Self> {
32        match name.to_ascii_lowercase().as_str() {
33            "box" => Some(Self::Box),
34            "sf" => Some(Self::Spheroidal),
35            "gauss" => Some(Self::Gaussian),
36            "gjinc" => Some(Self::GaussianJinc),
37            "pb" => Some(Self::PrimaryBeam),
38            _ => None,
39        }
40    }
41
42    /// Grid function name used by CASA records.
43    pub fn name(self) -> &'static str {
44        match self {
45            Self::Box => "BOX",
46            Self::Spheroidal => "SF",
47            Self::Gaussian => "GAUSS",
48            Self::GaussianJinc => "GJINC",
49            Self::PrimaryBeam => "PB",
50        }
51    }
52}

63/// Geometry and convolution settings supplied to `imager.setsdoptions`.
64#[derive(Clone, Debug)]
65pub struct ImagerSingleDishSetup {
66    /// Explicit kernel support in pixels, or a negative default marker.
67    pub support: i64,
68    /// Gaussian and Gaussian-jinc cutoff.
69    pub truncate: Quantity,
70    /// Gaussian half-width at half-maximum.
71    pub gaussian_width: Quantity,
72    /// Jinc scale parameter.
73    pub jinc_width: Quantity,
74    /// MeasurementSet POINTING direction column.
75    pub pointing_column: String,
76    /// Whether to cache the image coordinates of previous samples.
77    pub cache_pointings: bool,
78    /// Whether to convert the recorded directions before interpolation.
79    pub convert_first: String,
80}

84        Self {

86            truncate: Quantity::dimensionless(-1.),
87            gaussian_width: Quantity::dimensionless(-1.),
88            jinc_width: Quantity::dimensionless(-1.),

96impl ImagerSingleDishSetup {
97    pub(super) fn kernel_parameters(
98        &self,
99        kernel: ImagerSingleDishKernel,
100        cell_rad: f64,
101    ) -> casa_ms::MsResult<KernelParameters> {
102        let mut parameters = KernelParameters {
103            convsupport: self.support,
104            ..KernelParameters::DEFAULT
105        };
106        if matches!(
107            kernel,
108            ImagerSingleDishKernel::Gaussian | ImagerSingleDishKernel::GaussianJinc
109        ) {
110            let radians = Unit::new("rad").map_err(super::invalid)?;
111            let pixels = |width: &Quantity| {
112                let value = match width.unit().name() {
113                    "" | "pixel" => width.value(),
114                    _ => width.get_value_in(&radians).map_err(super::invalid)? / cell_rad.abs(),
115                };
116                let value = f64::from(value as f32);
117                if !value.is_finite() {
118                    return Err(super::invalid("single-dish kernel width must be finite"));
119                }
120                Ok::<_, casa_ms::MsError>(value)
121            };
122            parameters.truncate = pixels(&self.truncate)?;
123            parameters.gwidth = pixels(&self.gaussian_width)?;
124            if kernel == ImagerSingleDishKernel::GaussianJinc {
125                parameters.jwidth = pixels(&self.jinc_width)?;
126            }
127        }
128        Ok(parameters)
129    }
130}

132/// Convolution selected for the legacy imager's Fourier transform.
133#[derive(Clone, Copy, Debug, Default)]
134pub enum ImagerGridder {
135    /// The ordinary spheroidal GridFT convolution.
136    #[default]
137    Spheroidal,
138    /// Nearest-cell Boxcar convolution.
139    Boxcar,
140    /// The legacy heterogeneous-array MosaicFT convolution.
141    Mosaic,
142    /// W projection and its fixed, automatic or restored plane policy.
143    WProject(WPlaneCount),
144    /// Measure W over the active visibility selection before choosing planes.
145    AutoWProject,
146}
147
148impl ImagerGridder {
149    /// W projection with an explicit plane count or CASA's automatic choice.
150    pub fn w_project(planes: i32) -> Self {
151        if planes > 0 {
152            Self::WProject(WPlaneCount::Fixed(planes as u32))
153        } else {
154            Self::AutoWProject
155        }
156    }
157}

159/// Settings applied when preparing new model grids.
160#[derive(Clone, Copy, Debug)]
161pub struct ImagerTransformOptions {
162    /// Convolution kernel.
163    pub gridder: ImagerGridder,
164    /// The optional image-plane transform for total-power measurements.
165    pub single_dish: Option<ImagerSingleDish>,
166    /// CASA grid padding factor.
167    pub padding: f32,
168    /// FFT and gridding precision.
169    pub precision: GridPrecision,
170    /// CASA interpolation code (nearest 0, linear 1, cubic 2, spline 3).
171    pub frequency_interpolation: i32,
172    /// Requested cache size in complex pixels, or the CASA default.
173    pub cache_size: i64,
174    /// Scratch grid tile side.
175    pub tile_size: i32,
176    /// Worker limit for independent model-plane transforms, also stored for
177    /// CASA readers. Nonpositive values use the available worker count.
178    pub threads: i32,
179    /// Override of the observatory position in ITRF spherical coordinates.
180    pub observatory: Option<[f64; 3]>,
181}
182
183impl Default for ImagerTransformOptions {
184    fn default() -> Self {
185        Self {
186            gridder: ImagerGridder::Spheroidal,
187            single_dish: None,
188            padding: 1.0,
189            precision: GridPrecision::F64,
190            frequency_interpolation: 0,
191            cache_size: 1_000_000_000,
192            tile_size: 16,
193            threads: -1,
194            observatory: None,
195        }
196    }
197}

239impl ImagerTransformOptions {
240    pub(super) fn observatory_override(&self) -> Option<MPosition> {
241        self.observatory.map(|[longitude, latitude, radius]| {
242            let horizontal = radius * latitude.cos();
243            MPosition::new_itrf(
244                horizontal * longitude.cos(),
245                horizontal * longitude.sin(),
246                radius * latitude.sin(),
247            )
248        })
249    }
250
251    /// Use the same gridding precision CASA selects through setoptions.
252    pub fn use_single_precision(&mut self, single: bool) {
253        self.precision = if single {
254            GridPrecision::F32
255        } else {
256            GridPrecision::F64
257        };
258    }
259}
