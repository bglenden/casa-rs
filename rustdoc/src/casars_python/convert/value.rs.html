2//! Values CASA's tools return: casatools' `record` and `variant` results,
3//! built without the GIL and converted to Python once a call is done.
4//!
5//! casatools hands records to Python as `dict`s whose keys are sorted (its
6//! records are `std::map`s), so [`CasaValue::record`] sorts them; arrays
7//! become NumPy arrays, real ones of `float64`, complex ones of `complex128`
8//! and string ones of NumPy's unicode type, as casatools returns them.
9//! [`CasaValue::from_record`] converts a casacore record built in Rust.
10
11use casa_types::{ArrayValue, RecordValue, ScalarValue, Value};
12use ndarray::{ArrayD, IxDyn, ShapeBuilder};
13use num_complex::Complex64;
14use numpy::IntoPyArray;
15use pyo3::prelude::*;
16use pyo3::types::{PyComplex, PyDict, PyList, PyTuple};
17
18use super::Variant;

20/// A value a CASA tool returns.
21#[derive(Debug, Clone, PartialEq)]
22pub(crate) enum CasaValue {
23    /// A `bool`.
24    Bool(bool),
25    /// An `int`.
26    Int(i64),
27    /// A `float`.
28    Float(f64),
29    /// A `complex`.
30    Complex(Complex64),
31    /// A `str`.
32    Text(String),
33    /// A NumPy `float64` array of `shape`, its values in Fortran (column-major)
34    /// order, as casacore arrays hold them.
35    Floats { shape: Vec<usize>, values: Vec<f64> },
36    /// A NumPy `complex128` array of `shape`, in Fortran order.
37    Complexes {
38        shape: Vec<usize>,
39        values: Vec<Complex64>,
40    },
41    /// A NumPy unicode array.
42    Texts(Vec<String>),
43    /// A NumPy `int64` array of `shape`, its values in Fortran order.
44    Ints { shape: Vec<usize>, values: Vec<i64> },
45    /// A NumPy `bool` array of `shape`, in Fortran order.
46    Bools {
47        shape: Vec<usize>,
48        values: Vec<bool>,
49    },
50    /// A NumPy unicode array of `shape`, in Fortran order.
51    TextArray {
52        shape: Vec<usize>,
53        values: Vec<String>,
54    },
55    /// A Python `list`.
56    List(Vec<CasaValue>),
57    /// A Python `tuple`.
58    Tuple(Vec<CasaValue>),
59    /// A `dict` with sorted keys.
60    Record(Vec<(String, CasaValue)>),
61}
62
63impl CasaValue {
64    /// A record of `fields`, sorted by key as casatools sorts them.
65    pub(crate) fn record(fields: impl IntoIterator<Item = (impl Into<String>, CasaValue)>) -> Self {
66        let mut fields = fields
67            .into_iter()
68            .map(|(key, value)| (key.into(), value))
69            .collect::<Vec<_>>();
70        fields.sort_by(|left, right| left.0.cmp(&right.0));
71        Self::Record(fields)
72    }
73
74    /// The empty record `{}`: what CASA's tools return for a failure they
75    /// log.
76    pub(crate) fn empty_record() -> Self {
77        Self::Record(Vec::new())
78    }
79
80    /// A one-dimensional `float64` array.
81    pub(crate) fn floats(values: Vec<f64>) -> Self {
82        Self::Floats {
83            shape: vec![values.len()],
84            values,
85        }
86    }
87
88    /// An argument as CASA hands a record field back (`me.getvalue`).
89    pub(crate) fn from_variant(value: &Variant) -> Self {
90        match value {
91            Variant::Empty => Self::List(Vec::new()),
92            Variant::Bool(flag) => Self::Bool(*flag),
93            Variant::Number(number) => Self::Float(*number),
94            Variant::Text(text) | Variant::Other(text) => Self::Text(text.clone()),
95            Variant::Texts(texts) => Self::Texts(texts.clone()),
96            Variant::Numbers(numbers) => Self::floats(numbers.clone()),
97            Variant::Complex(value) => Self::Complex(*value),
98            Variant::Complexes(values) => Self::Complexes {
99                shape: vec![values.len()],
100                values: values.clone(),
101            },
102            Variant::Array { shape, values } => Self::Floats {
103                shape: shape.clone(),
104                values: values.clone(),
105            },
106            Variant::Record(record) => Self::record(
107                record
108                    .iter()
109                    .map(|(key, value)| (key.clone(), Self::from_variant(value))),
110            ),
111        }
112    }
113
114    /// A casacore record as casatools hands it to Python: its fields sorted
115    /// by key, its arrays as NumPy arrays (integer arrays as `float64`).
116    pub(crate) fn from_record(record: &RecordValue) -> Self {
117        Self::record(
118            record
119                .fields()
120                .iter()
121                .map(|field| (field.name.clone(), Self::from_value(&field.value))),
122        )
123    }
124
125    /// A casacore value; see [`Self::from_record`].
126    pub(crate) fn from_value(value: &Value) -> Self {
127        match value {
128            Value::Scalar(scalar) => Self::from_scalar(scalar),
129            Value::Array(array) => Self::from_array(array),
130            Value::Record(record) => Self::from_record(record),
131            // casatools hands a table keyword to Python as its path.
132            Value::TableRef(path) => Self::Text(path.clone()),
133        }
134    }
135
136    /// A casacore scalar: any integer type becomes an `int`, any real type a
137    /// `float`.
138    pub(crate) fn from_scalar(scalar: &ScalarValue) -> Self {
139        match scalar {
140            ScalarValue::Bool(flag) => Self::Bool(*flag),
141            ScalarValue::UInt8(value) => Self::Int(i64::from(*value)),
142            ScalarValue::UInt16(value) => Self::Int(i64::from(*value)),
143            ScalarValue::UInt32(value) => Self::Int(i64::from(*value)),
144            ScalarValue::Int16(value) => Self::Int(i64::from(*value)),
145            ScalarValue::Int32(value) => Self::Int(i64::from(*value)),
146            ScalarValue::Int64(value) => Self::Int(*value),
147            ScalarValue::Float32(value) => Self::Float(f64::from(*value)),
148            ScalarValue::Float64(value) => Self::Float(*value),
149            ScalarValue::Complex32(value) => {
150                Self::Complex(Complex64::new(f64::from(value.re), f64::from(value.im)))
151            }
152            ScalarValue::Complex64(value) => Self::Complex(*value),
153            ScalarValue::String(text) => Self::Text(text.clone()),
154        }
155    }
156
157    fn from_array(array: &ArrayValue) -> Self {
158        /// Fortran order: the transpose's logical order.
159        fn fortran<T: Copy, U>(
160            array: &ArrayD<T>,
161            convert: impl Fn(T) -> U,
162        ) -> (Vec<usize>, Vec<U>) {
163            (
164                array.shape().to_vec(),
165                array.t().iter().map(|value| convert(*value)).collect(),
166            )
167        }
168        let floats = |(shape, values)| Self::Floats { shape, values };
169        match array {
170            ArrayValue::Bool(values) => {
171                Self::List(values.t().iter().map(|flag| Self::Bool(*flag)).collect())
172            }
173            ArrayValue::UInt8(values) => floats(fortran(values, f64::from)),
174            ArrayValue::UInt16(values) => floats(fortran(values, f64::from)),
175            ArrayValue::UInt32(values) => floats(fortran(values, f64::from)),
176            ArrayValue::Int16(values) => floats(fortran(values, f64::from)),
177            ArrayValue::Int32(values) => floats(fortran(values, f64::from)),
178            ArrayValue::Int64(values) => floats(fortran(values, |value| value as f64)),
179            ArrayValue::Float32(values) => floats(fortran(values, f64::from)),
180            ArrayValue::Float64(values) => floats(fortran(values, |value| value)),
181            ArrayValue::Complex32(values) => {
182                let (shape, values) = fortran(values, |value| {
183                    Complex64::new(f64::from(value.re), f64::from(value.im))
184                });
185                Self::Complexes { shape, values }
186            }
187            ArrayValue::Complex64(values) => {
188                let (shape, values) = fortran(values, |value| value);
189                Self::Complexes { shape, values }
190            }
191            ArrayValue::String(texts) => Self::Texts(texts.t().iter().cloned().collect()),
192        }
193    }
194
195    /// A quantity record `{'unit': unit, 'value': value}`.
196    pub(crate) fn quantity(value: f64, unit: impl Into<String>) -> Self {
197        Self::record([
198            ("unit", Self::Text(unit.into())),
199            ("value", Self::Float(value)),
200        ])
201    }
202
203    /// The value as a Python object.
204    pub(crate) fn into_python(self, py: Python<'_>) -> PyResult<Py<PyAny>> {
205        Ok(match self {
206            Self::Bool(flag) => flag.into_pyobject(py)?.to_owned().into_any().unbind(),
207            Self::Int(value) => value.into_pyobject(py)?.into_any().unbind(),
208            Self::Float(value) => value.into_pyobject(py)?.into_any().unbind(),
209            Self::Complex(value) => PyComplex::from_doubles(py, value.re, value.im)
210                .into_any()
211                .unbind(),
212            Self::Text(text) => text.into_pyobject(py)?.into_any().unbind(),
213            Self::Floats { shape, values } => fortran_array(shape, values)?
214                .into_pyarray(py)
215                .into_any()
216                .unbind(),
217            Self::Complexes { shape, values } => fortran_array(shape, values)?
218                .into_pyarray(py)
219                .into_any()
220                .unbind(),
221            Self::Ints { shape, values } => fortran_array(shape, values)?
222                .into_pyarray(py)
223                .into_any()
224                .unbind(),
225            Self::Bools { shape, values } => fortran_array(shape, values)?
226                .into_pyarray(py)
227                .into_any()
228                .unbind(),
229            Self::Texts(texts) => {
230                let keywords = PyDict::new(py);
231                keywords.set_item("dtype", "str")?;
232                py.import("numpy")?
233                    .call_method("array", (PyList::new(py, texts)?,), Some(&keywords))?
234                    .unbind()
235            }
236            Self::TextArray { shape, values } => {
237                let numpy = py.import("numpy")?;
238                let keywords = PyDict::new(py);
239                keywords.set_item("dtype", "str")?;
240                let flat =
241                    numpy.call_method("array", (PyList::new(py, values)?,), Some(&keywords))?;
242                let order = PyDict::new(py);
243                order.set_item("order", "F")?;
244                flat.call_method("reshape", (shape,), Some(&order))?
245                    .unbind()
246            }
247            Self::List(items) => {
248                let items = items
249                    .into_iter()
250                    .map(|item| item.into_python(py))
251                    .collect::<PyResult<Vec<_>>>()?;
252                PyList::new(py, items)?.into_any().unbind()
253            }
254            Self::Tuple(items) => {
255                let items = items
256                    .into_iter()
257                    .map(|item| item.into_python(py))
258                    .collect::<PyResult<Vec<_>>>()?;
259                PyTuple::new(py, items)?.into_any().unbind()
260            }
261            Self::Record(fields) => {
262                let dict = PyDict::new(py);
263                for (key, value) in fields {
264                    dict.set_item(key, value.into_python(py)?)?;
265                }
266                dict.into_any().unbind()
267            }
268        })
269    }
270}

272/// `values`, in Fortran order, as an array of `shape`.
273pub(crate) fn fortran_array<T>(shape: Vec<usize>, values: Vec<T>) -> PyResult<ArrayD<T>> {
274    ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(|error| {
275        pyo3::exceptions::PyRuntimeError::new_err(format!("array of shape {shape:?}: {error}"))
276    })
277}

282#[cfg(test)]
283mod tests {
284    use super::CasaValue;
285
286    #[test]
287    fn records_sort_their_keys_as_casatools_does() {
288        let record = CasaValue::record([
289            ("type", CasaValue::Text("epoch".into())),
290            ("m0", CasaValue::quantity(1.0, "d")),
291            ("refer", CasaValue::Text("UTC".into())),
292        ]);
293        let CasaValue::Record(fields) = record else {
294            panic!("a record");
295        };
296        let keys = fields
297            .iter()
298            .map(|(key, _)| key.as_str())
299            .collect::<Vec<_>>();
300        assert_eq!(keys, ["m0", "refer", "type"]);
301        let CasaValue::Record(quantity) = &fields[0].1 else {
302            panic!("a quantity record");
303        };
304        assert_eq!(quantity[0].0, "unit");
305        assert_eq!(quantity[1].0, "value");
306    }
