2//! Least-squares fitting and residual clipping shared by the baseline families.
3
4use casa_numerics::LeastSquaresDesign;
5
6pub(super) struct Fit {
7    pub(super) coefficients: Vec<f64>,
8    pub(super) residuals: Vec<f32>,
9    pub(super) included: Vec<bool>,
10    pub(super) rms: f32,
11    pub(super) boundaries: Vec<usize>,
12}
13
14pub(super) struct LinearFit {
15    pub(super) basis: Vec<Vec<f64>>,
16    pub(super) terms: usize,
17    // First endpoint after each segment. The sample immediately before
18    // this index stays eligible in Sakura's clipping pass.
19    pub(super) segment_ends: Vec<usize>,
20    pub(super) failure: &'static str,
21}
22
23impl LinearFit {
24    fn prepare(&self, channels: usize, included: &[bool]) -> Option<LeastSquaresDesign> {
25        let basis = self
26            .basis
27            .iter()
28            .take(channels)
29            .zip(included)
30            .filter_map(|(terms, &use_channel)| use_channel.then_some(terms.as_slice()));
31        LeastSquaresDesign::unweighted(basis, self.terms)
32    }
33
34    pub(super) fn solve(
35        &self,
36        spectrum: &[f32],
37        included: &[bool],
38        clip_sigma: f64,
39        max_fits: usize,
40        mut first_design: Option<&mut Option<LeastSquaresDesign>>,
41    ) -> Result<Option<Fit>, String> {
42        let mut included = included.to_vec();
43        for iteration in 0..max_fits {
44            let observed = self
45                .basis
46                .iter()
47                .zip(spectrum)
48                .zip(&included)
49                .filter_map(|((_, &value), &use_channel)| use_channel.then_some(f64::from(value)))
50                .collect::<Vec<_>>();
51            if observed.len() < self.terms {
52                return if iteration == 0 {
53                    Ok(None)
54                } else {
55                    Err(self.failure.into())
56                };
57            }
58            let mut uncached = None;
59            let design = first_design.take().unwrap_or(&mut uncached);
60            if design.is_none() && observed.iter().all(|value| value.is_finite()) {
61                *design = self.prepare(spectrum.len(), &included);
62            }
63            let coefficients = design
64                .as_ref()
65                .and_then(|design| design.solve(&observed))
66                .ok_or(self.failure)?;
67            let residuals = self
68                .basis
69                .iter()
70                .zip(spectrum)
71                .map(|(basis, &value)| {
72                    let fit = basis
73                        .iter()
74                        .zip(&coefficients)
75                        .map(|(value, coefficient)| value * coefficient)
76                        .sum::<f64>() as f32;
77                    value - fit
78                })
79                .collect::<Vec<_>>();
80            let (sum, power) = residuals
81                .iter()
82                .zip(&included)
83                .filter(|&(_, &use_channel)| use_channel)
84                .fold((0., 0.), |(sum, power), (&value, _)| {
85                    let value = f64::from(value);
86                    (sum + value, power + value * value)
87                });
88            let count = observed.len() as f64;
89            let mean = sum / count;
90            let rms = (power / count - mean * mean).abs().sqrt();
91            if iteration + 1 < max_fits
92                && self.clip(&residuals, &mut included, mean, clip_sigma * rms)
93            {
94                continue;
95            }
96            return Ok(Some(Fit {
97                coefficients,
98                residuals,
99                included,
100                rms: rms as f32,
101                boundaries: Vec::new(),
102            }));
103        }
104        Err(self.failure.into())
105    }
106
107    fn clip(&self, residuals: &[f32], included: &mut [bool], mean: f64, radius: f64) -> bool {
108        let radius = f64::from(radius as f32);
109        let lower = (mean - radius) as f32;
110        let upper = (mean + radius) as f32;
111        let mut changed = false;
112        let mut endpoints = self.segment_ends.iter().copied();
113        let mut next = endpoints.next();
114        for (channel, (&value, use_channel)) in residuals.iter().zip(included).enumerate() {
115            if next == Some(channel + 1) {
116                next = endpoints.next();
117            } else if *use_channel && (value - lower) * (upper - value) < 0. {
118                *use_channel = false;
119                changed = true;
120            }
121        }
122        changed
123    }
124}
