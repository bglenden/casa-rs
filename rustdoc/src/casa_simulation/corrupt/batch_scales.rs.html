2//! What a write batch's corruptions take from each antenna, channel by
3//! channel: the antenna scales of `tsys-atm` noise and the phasors of a
4//! tropospheric (`settrop`) term.
5//!
6//! A `tsys-atm` row's σ in channel `c` is `s_i(c) · s_j(c)`, each antenna's
7//! scale from its Tsys at its airmass ([`TsysAtm::antenna_scales`]), and a
8//! tropospheric term multiplies a row by `g_i(c) · conj(g_j(c))`, each
9//! antenna's phasor from the water vapour above it
10//! ([`TroposphericPhase::phasors`]). Both belong to the antenna and the
11//! integration, as a Jones term does, not to the row. A batch's are worked
12//! out once, on the workers, before the jobs that read them: each costs a
13//! line of sight through the atmosphere's layers, or a sine and cosine, per
14//! channel, and a job of its own for every run of rows would repeat it as
15//! many times as an integration's rows are split into jobs (for phasors,
16//! once per baseline rather than per antenna).
17//!
18//! An entry is keyed by what its values depend on (the term, the window,
19//! and the antenna's dish, airmass and the row's exposure, or the antenna
20//! and the term's solution), never by where the batch starts, so an
21//! integration that two batches share gets the same values in both.
22//!
23//! A table reserves room for the most entries its batch can need before any
24//! is added ([`BatchScalesBuilder::with_capacity`]): a vector or index that
25//! grew as entries came would hold its old allocation and its new one
26//! together while it moved (#897 review).
27

38/// Bytes of an index beside its buckets: its control group, and the
39/// alignment of the control bytes after the buckets.
40const INDEX_GROUP_BYTES: usize = 32;

42/// An entry still to be worked out: where its values start in its table,
43/// how many there are (one a channel), and how to work them out.
44pub(crate) trait Entry: Sync {
45    type Value: Copy + Default + Send;
46
47    fn offset(&self) -> usize;
48
49    fn width(&self) -> usize;
50
51    fn work_out(&self, values: &mut [Self::Value]);
52}

54/// What an antenna's scales depend on. The term is its model's address: a
55/// batch's terms are alive, and distinct, while it runs, and two
56/// applications of one term share their model.
57#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
58struct ScaleKey {
59    model: usize,
60    window: usize,
61    /// The dish diameter, airmass and exposure, as bits.
62    antenna: [u64; 3],
63}
64
65impl ScaleKey {
66    /// The key of antenna `end` (0 or 1) of `row` for `model` in spectral
67    /// window `window`.
68    fn new(model: &TsysAtm, window: usize, row: &RowNoise, end: usize) -> Self {
69        Self {
70            model: std::ptr::from_ref(model) as usize,
71            window,
72            antenna: [
73                row.diameters_m[end].to_bits(),
74                u64::from(row.airmass[end].to_bits()),
75                row.exposure_s.to_bits(),
76            ],
77        }
78    }
79}

81/// An antenna's scales still to be worked out.
82#[derive(Debug)]
83pub(crate) struct ScaleEntry<'a> {
84    model: &'a TsysAtm,
85    window: &'a WindowAtmosphere,
86    /// The antenna's dish diameter and airmass.
87    antenna: (f64, f32),
88    /// The window's channel bandwidth and the row's exposure.
89    integration: (f64, f64),
90    offset: usize,
91}
92
93impl Entry for ScaleEntry<'_> {
94    type Value = f32;
95
96    fn offset(&self) -> usize {
97        self.offset
98    }
99
100    fn width(&self) -> usize {
101        self.window.channels()
102    }
103
104    fn work_out(&self, values: &mut [f32]) {
105        self.model
106            .antenna_scales(self.window, self.antenna, self.integration, values);
107    }
108}

160/// One kind's table: each entry's values, found by its key.
161#[derive(Debug)]
162struct Table<K, V> {
163    /// Each entry's first value in `values`.
164    index: HashMap<K, usize>,
165    values: Vec<V>,
166}
167
168impl<K, V> Default for Table<K, V> {
169    fn default() -> Self {
170        Self {
171            index: HashMap::new(),
172            values: Vec::new(),
173        }
174    }
175}
176
177impl<K: Eq + Hash, V> Table<K, V> {
178    /// The `width` values of the entry of `key`.
179    fn get(&self, key: &K, width: usize) -> &[V] {
180        let offset = self.index[key];
181        &self.values[offset..offset + width]
182    }
183}

185/// The scales and phasors a batch's corruptions read, by antenna and
186/// integration.
187#[derive(Debug, Default)]
188pub(crate) struct BatchScales {
189    scales: Table<ScaleKey, f32>,
190    phasors: Table<PhaseKey, Complex32>,
191}
192
193impl BatchScales {
194    /// The most a table with room for `entries` entries, (scales,
195    /// phasors), of at most `channels` channels holds while it is built and
196    /// read: their values, and the room reserved for them.
197    pub(crate) fn bytes((scales, phasors): (usize, usize), channels: usize) -> u64 {
198        ((scales * size_of::<f32>() + phasors * size_of::<Complex32>()) * channels) as u64
199            + TableBuilder::<ScaleKey, ScaleEntry<'static>>::reserved_bytes(scales)
200            + TableBuilder::<PhaseKey, PhaseEntry<'static>>::reserved_bytes(phasors)
201    }
202
203    /// The most entries a batch of `rows` rows needs for `terms` terms: two
204    /// a row, and no more than each integration's antennas, for
205    /// integrations of at least `smallest_integration` rows of at most
206    /// `antennas` antennas (`None` when the rows need not come in whole
207    /// integrations). Two integrations may be partly in the batch.
208    pub(crate) fn most_entries(
209        terms: usize,
210        rows: usize,
211        integrations: Option<(usize, usize)>,
212    ) -> usize {
213        let by_rows = 2 * rows;
214        let by_integrations = integrations.map_or(by_rows, |(smallest, antennas)| {
215            (rows / smallest.max(1) + 2) * antennas
216        });
217        terms * by_rows.min(by_integrations)
218    }
219
220    /// The scales of antenna `end` (0 or 1) of `row` for `model` in
221    /// `window`, spectral window `window_id`.
222    ///
223    /// # Panics
224    ///
225    /// When the batch's table was not built with the row
226    /// ([`BatchScalesBuilder::add_rows`]).
227    pub(crate) fn get(
228        &self,
229        (model, window_id, window): (&TsysAtm, usize, &WindowAtmosphere),
230        row: &RowNoise,
231        end: usize,
232    ) -> &[f32] {
233        self.scales.get(
234            &ScaleKey::new(model, window_id, row, end),
235            window.channels(),
236        )
237    }
238
239    /// Antenna `antenna`'s phasors of the tropospheric `term` in its block
240    /// `block` of spectral window `window`.
241    ///
242    /// # Panics
243    ///
244    /// When the batch's table was not built with them
245    /// ([`BatchScalesBuilder::add_phase_rows`]).
246    pub(crate) fn phasors(
247        &self,
248        term: &JonesTerm,
249        (window, block): (usize, usize),
250        antenna: usize,
251    ) -> &[Complex32] {
252        let width = term
253            .tropospheric_phase()
254            .and_then(|phase| phase.channels(window))
255            .unwrap_or(0);
256        self.phasors
257            .get(&PhaseKey::new(term, (window, block), antenna), width)
258    }
259}

261/// One kind's table as its entries are added, before they are worked out.
262#[derive(Debug)]
263struct TableBuilder<K, P> {
264    index: HashMap<K, usize>,
265    pending: Vec<P>,
266    values: usize,
267}
268
269impl<K: Eq + Hash, P: Entry> TableBuilder<K, P> {
270    /// Room for `entries` entries, reserved at once.
271    fn with_capacity(entries: usize) -> Self {
272        Self {
273            index: HashMap::with_capacity(entries),
274            pending: Vec::with_capacity(entries),
275            values: 0,
276        }
277    }
278
279    /// What room for `entries` entries holds beside their values: the
280    /// index's buckets, a power of two at least 8/7 of the entries (std's
281    /// `HashMap`), each with a control byte, and its control group; each
282    /// entry's pending computation; and the jobs, at most one an entry,
283    /// they are worked out in.
284    fn reserved_bytes(entries: usize) -> u64 {
285        if entries == 0 {
286            return 0;
287        }
288        let buckets = (entries * 8).div_ceil(7).next_power_of_two().max(4);
289        (buckets * (size_of::<(K, usize)>() + 1)
290            + INDEX_GROUP_BYTES
291            + entries * (size_of::<P>() + size_of::<BatchJob<'static, 'static>>())) as u64
292    }
293
294    /// Add the entry of `key`, `width` values worked out by what `entry`
295    /// makes of its offset, unless it is in the table already.
296    fn add(&mut self, key: K, width: usize, entry: impl FnOnce(usize) -> P) {
297        if self.index.contains_key(&key) {
298            return;
299        }
300        debug_assert!(
301            self.pending.len() < self.pending.capacity(),
302            "a batch needs no more entries than were reserved for it"
303        );
304        self.index.insert(key, self.values);
305        self.pending.push(entry(self.values));
306        self.values += width;
307    }
308
309    /// The values, and the pending entries in about `jobs` runs of
310    /// consecutive ones.
311    fn runs(&self, jobs: usize) -> (Vec<P::Value>, impl Iterator<Item = &[P]>) {
312        let per_job = self.pending.len().div_ceil(jobs.max(1)).max(1);
313        (
314            vec![P::Value::default(); self.values],
315            self.pending.chunks(per_job),
316        )
317    }
318}

320/// A batch's table as its rows are added, before the values are worked
321/// out.
322#[derive(Debug)]
323pub(crate) struct BatchScalesBuilder<'a> {
324    scales: TableBuilder<ScaleKey, ScaleEntry<'a>>,
325    phasors: TableBuilder<PhaseKey, PhaseEntry<'a>>,
326}
327
328impl<'a> BatchScalesBuilder<'a> {
329    /// A table with room for `entries` entries, (scales, phasors), reserved
330    /// at once ([`BatchScales::most_entries`] of its batch), so adding them
331    /// moves nothing.
332    pub(crate) fn with_capacity((scales, phasors): (usize, usize)) -> Self {
333        Self {
334            scales: TableBuilder::with_capacity(scales),
335            phasors: TableBuilder::with_capacity(phasors),
336        }
337    }
338
339    /// Add the scales both antennas of each of `rows` need for `model` in
340    /// `window`, spectral window `window_id`, whose channels are
341    /// `channel_bandwidth_hz` wide.
342    ///
343    /// The entries must fit the room reserved for them
344    /// ([`Self::with_capacity`]); a debug build asserts it.
345    pub(crate) fn add_rows(
346        &mut self,
347        (model, window_id, window): (&'a TsysAtm, usize, &'a WindowAtmosphere),
348        channel_bandwidth_hz: f64,
349        rows: impl Iterator<Item = RowNoise>,
350    ) {
351        for row in rows {
352            for end in 0..2 {
353                let key = ScaleKey::new(model, window_id, &row, end);
354                self.scales
355                    .add(key, window.channels(), |offset| ScaleEntry {
356                        model,
357                        window,
358                        antenna: (row.diameters_m[end], row.airmass[end]),
359                        integration: (channel_bandwidth_hz, row.exposure_s),
360                        offset,
361                    });
362            }
363        }
364    }
365
366    /// Add the phasors both antennas of each of `rows`, (TIME, ANTENNA1,
367    /// ANTENNA2), need for the tropospheric `term` in spectral window
368    /// `window`.
369    ///
370    /// The entries must fit the room reserved for them
371    /// ([`Self::with_capacity`]); a debug build asserts it.
372    pub(crate) fn add_phase_rows(
373        &mut self,
374        (term, window): (&'a JonesTerm, usize),
375        rows: impl Iterator<Item = (f64, usize, usize)>,
376    ) {
377        let Some(phase) = term.tropospheric_phase() else {
378            return;
379        };
380        let width = phase.channels(window).unwrap_or(0);
381        for (time_s, antenna1, antenna2) in rows {
382            let block = term.block_at(window, time_s);
383            for antenna in [antenna1, antenna2] {
384                let key = PhaseKey::new(term, (window, block), antenna);
385                self.phasors.add(key, width, |offset| PhaseEntry {
386                    phase,
387                    solution: (window, block),
388                    antenna: (antenna, term.antennas()),
389                    width,
390                    offset,
391                });
392            }
393        }
394    }
395
396    /// Whether no row needs scales or phasors.
397    pub(crate) fn is_empty(&self) -> bool {
398        self.scales.pending.is_empty() && self.phasors.pending.is_empty()
399    }
400
401    /// Work the values out, in about `jobs` jobs of each kind run by
402    /// `run_jobs`.
403    ///
404    /// # Errors
405    ///
406    /// What `run_jobs` returns.
407    pub(crate) fn build<E>(
408        self,
409        jobs: usize,
410        run_jobs: impl FnOnce(&mut [BatchJob<'_, 'a>]) -> Result<(), E>,
411    ) -> Result<BatchScales, E> {
412        let (mut scales, scale_runs) = self.scales.runs(jobs);
413        let (mut phasors, phase_runs) = self.phasors.runs(jobs);
414        let mut work: Vec<BatchJob<'_, 'a>> = Vec::with_capacity(2 * jobs.max(1));
415        split(&mut scales, scale_runs, |job| {
416            work.push(BatchJob::Scales(job))
417        });
418        split(&mut phasors, phase_runs, |job| {
419            work.push(BatchJob::Phasors(job));
420        });
421        run_jobs(&mut work)?;
422        drop(work);
423        Ok(BatchScales {
424            scales: Table {
425                index: self.scales.index,
426                values: scales,
427            },
428            phasors: Table {
429                index: self.phasors.index,
430                values: phasors,
431            },
432        })
433    }
434}

436/// `values` split along `runs` of consecutive entries, each run's share a
437/// job handed to `push`.
438fn split<'p, P: Entry + 'p>(
439    values: &'p mut [P::Value],
440    runs: impl Iterator<Item = &'p [P]>,
441    mut push: impl FnMut(EntryJob<'p, P>),
442) {
443    let mut rest = values;
444    for pending in runs {
445        let last = pending.last().expect("runs are not empty");
446        let length = last.offset() + last.width() - pending[0].offset();
447        let (values, after) = std::mem::take(&mut rest).split_at_mut(length);
448        push(EntryJob { pending, values });
449        rest = after;
450    }
451}

453/// Consecutive entries of one kind, worked out in one job.
454pub(crate) struct EntryJob<'p, P: Entry> {
455    pending: &'p [P],
456    values: &'p mut [P::Value],
457}
458
459impl<P: Entry> EntryJob<'_, P> {
460    fn run(&mut self) {
461        let first = self.pending.first().map_or(0, Entry::offset);
462        for pending in self.pending {
463            let start = pending.offset() - first;
464            pending.work_out(&mut self.values[start..start + pending.width()]);
465        }
466    }
467}
468
469/// A job of a batch's table: consecutive entries of one kind.
470pub(crate) enum BatchJob<'p, 'a> {
471    Scales(EntryJob<'p, ScaleEntry<'a>>),
472    Phasors(EntryJob<'p, PhaseEntry<'a>>),
473}
474
475impl BatchJob<'_, '_> {
476    /// Work the job's values out.
477    pub(crate) fn run(&mut self) {
478        match self {
479            Self::Scales(job) => job.run(),
480            Self::Phasors(job) => job.run(),
481        }
482    }
483}
