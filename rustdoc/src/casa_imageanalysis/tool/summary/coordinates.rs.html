24struct Placement {
25    index: usize,
26    field: String,
27    model: CoordinateModel,
28    pixels: Vec<Option<usize>>,
29    world: Vec<Option<usize>>,
30}
31
32fn placements(record: &RecordValue) -> ImageToolResult<Vec<Placement>> {
33    CoordinateSystem::record_coordinates(record)?
34        .into_iter()
35        .map(|coordinate| {
36            let index = coordinate
37                .field
38                .trim_start_matches(char::is_alphabetic)
39                .parse::<usize>()
40                .map_err(|err| ImageToolError::new(err.to_string()))?;
41            Ok(Placement {
42                index,
43                model: coordinate.model(record)?,
44                field: coordinate.field,
45                pixels: coordinate.pixel_axes,
46                world: coordinate.world_axes,
47            })
48        })
49        .collect()
50}

52/// Extracted coordinate axis descriptors for `ia.summary`.
53#[derive(Debug, Clone)]
54pub struct SummaryCoordinateFields {
55    /// 0-based reference pixel per pixel axis.
56    pub refpix: Vec<f64>,
57    /// Axis names in pixel or world order.
58    pub axisnames: Vec<String>,
59    /// Axis units in pixel or world order.
60    pub axisunits: Vec<String>,
61    /// Reference value per axis in declared units.
62    pub refval: Vec<f64>,
63    /// Increment per axis in declared units.
64    pub incr: Vec<f64>,
65}
66
67/// Extracts coordinate summary descriptor fields from `record`.
68pub fn summary_coordinate_fields(
69    record: &RecordValue,
70    pixel_order: bool,
71) -> ImageToolResult<SummaryCoordinateFields> {
72    let mut pixels = Vec::new();
73    let mut descriptors = Vec::new();
74    for coordinate in placements(record)? {
75        let model = &coordinate.model;
76        if matches!(model, CoordinateModel::System(_))
77            && model.n_pixel_axes() != model.n_world_axes()
78        {
79            return Err(ImageToolError::new(
80                "Cannot summarize a nested coordinate system with unequal pixel and world axes",
81            ));
82        }
83        let names = model.axis_names();
84        let units = model.declared_axis_units();
85        let (values, increments) = model.reference_and_increment_in_declared_units();
86        let references = model.reference_pixel();
87        for local in 0..model.n_world_axes() {
88            let pixel = coordinate.pixels[local];
89            if let Some(index) = pixel {
90                pixels.push((index, references[local]));
91            }
92            let axis = if pixel_order {
93                pixel
94            } else {
95                coordinate.world[local]
96            };
97            if let Some(index) = axis {
98                descriptors.push((
99                    index,
100                    names[local].clone(),
101                    units[local].clone(),
102                    values[local],
103                    increments[local],
104                ));
105            }
106        }
107    }
108    pixels.sort_by_key(|axis| axis.0);
109    descriptors.sort_by_key(|axis| axis.0);
110    Ok(SummaryCoordinateFields {
111        refpix: pixels.into_iter().map(|axis| axis.1).collect(),
112        axisnames: descriptors.iter().map(|axis| axis.1.clone()).collect(),
113        axisunits: descriptors.iter().map(|axis| axis.2.clone()).collect(),
114        refval: descriptors.iter().map(|axis| axis.3).collect(),
115        incr: descriptors.iter().map(|axis| axis.4).collect(),
116    })
117}

119/// Converts frequency into velocity in `km/s` using `doppler` and `rest` frequency.
120pub(super) fn velocity(frequency: f64, rest: f64, doppler: DopplerRef) -> f64 {
121    MDoppler::new(frequency / rest, DopplerRef::RATIO)
122        .convert_to(doppler, &MeasFrame::default())
123        .map_or(f64::NAN, |velocity| velocity.value() * 299_792.458)
124}

126const HEADERS: [&str; 11] = [
127    "Axis",
128    "Coord",
129    "Type",
130    "Name",
131    "Proj",
132    "Shape",
133    "Tile",
134    "Coord value",
135    "at pixel",
136    "Coord incr",
137    " Units",
138];
139
140struct ReportRow {
141    order: (usize, usize),
142    fields: [String; HEADERS.len()],
143    labels_only: bool,
144}
145
146fn rows(
147    record: &RecordValue,
148    shape: &[usize],
149    tile: &[usize],
150    doppler: DopplerRef,
151) -> ImageToolResult<Vec<ReportRow>> {
152    let mut rows = Vec::new();
153    for coordinate in placements(record)? {
154        let model = &coordinate.model;
155        let names = model.axis_names();
156        let units = model.axis_units();
157        let values = model.reference_value();
158        let increments = model.increment();
159        let refs = model.reference_pixel();
160        let pixel_replace = match record.get(&format!("pixelreplace{}", coordinate.index)) {
161            Some(Value::Array(ArrayValue::Float64(pixels))) => {
162                Some(pixels.iter().copied().collect::<Vec<_>>())
163            }
164            _ => None,
165        };
166        let last_pixel = coordinate.pixels.iter().flatten().copied().max();
167        for local in 0..model.n_world_axes() {
168            if coordinate.world[local].is_none() {
169                continue;
170            }
171            let pixel = coordinate.pixels[local];
172            let order = pixel.map_or((last_pixel.unwrap_or(usize::MAX), 2 + local), |axis| {
173                (axis, 0)
174            });
175            let display_pixel = pixel.map_or_else(
176                || {
177                    pixel_replace
178                        .as_ref()
179                        .map_or(refs[local], |pixels| pixels[local])
180                },
181                |_| refs[local],
182            );
183            let reference = if let (None, Some(replacement)) = (pixel, &pixel_replace) {
184                model.to_world(replacement)?[local]
185            } else {
186                values[local]
187            };
188            let field = |axis: Option<usize>, lengths: &[usize]| {
189                axis.and_then(|axis| lengths.get(axis))
190                    .map_or_else(|| " ".into(), usize::to_string)
191            };
192            let mut row = ReportRow {
193                order,
194                fields: [
195                    pixel.map_or_else(|| "..".into(), |axis| axis.to_string()),
196                    coordinate.index.to_string(),
197                    model.coordinate_type().to_string(),
198                    names[local].clone(),
199                    " ".into(),
200                    field(pixel, shape),
201                    field(pixel, tile),
202                    format_number(reference, increments[0], false),
203                    format!("{display_pixel:.2}"),
204                    if pixel.is_some() {
205                        format_number_with_precision(increments[local], increments[0], true, 6)
206                    } else {
207                        " ".into()
208                    },
209                    format!(" {}", units[local]),
210                ],
211                labels_only: false,
212            };
213            let mut velocity_row = None;
214            match model {
215                CoordinateModel::Direction(direction) => {
216                    row.fields[4] = if direction.is_ncp() {
217                        "NCP"
218                    } else {
219                        direction.projection().name()
220                    }
221                    .into();
222                    let frame = direction.direction_ref();
223                    let (text, unit) = format_direction(frame, local, reference, true, false, 3);
224                    row.fields[7] = text;
225                    if pixel.is_some() {
226                        let (text, unit) =
227                            format_direction(frame, local, increments[local], false, true, 6);
228                        row.fields[9] = text;
229                        row.fields[10] = format!(" {unit}");
230                    } else {
231                        row.fields[10] = format!(" {unit}");
232                    }
233                }
234                CoordinateModel::Stokes(stokes) => {
235                    row.fields[7] = if pixel.is_some() {
236                        stokes
237                            .stokes()
238                            .iter()
239                            .map(|kind| kind.name())
240                            .collect::<Vec<_>>()
241                            .join(" ")
242                    } else {
243                        stokes
244                            .stokes()
245                            .get(display_pixel.round() as usize)
246                            .map_or("?", |kind| kind.name())
247                            .to_owned()
248                    };
249                    row.labels_only = true;
250                }
251                CoordinateModel::Quality(quality) => {
252                    row.fields[7] = if pixel.is_some() {
253                        quality
254                            .values()
255                            .iter()
256                            .map(|label| label.name())
257                            .collect::<Vec<_>>()
258                            .join(" ")
259                    } else {
260                        QualityType::from_world(reference).name().into()
261                    };
262                    row.labels_only = true;
263                }
264                CoordinateModel::Spectral(spectral) => {
265                    velocity_row = spectral_row(
266                        &mut row,
267                        spectral,
268                        subrecord(record, &coordinate.field),
269                        reference,
270                        pixel.map(|_| display_pixel),
271                        doppler,
272                    )?;
273                }
274                CoordinateModel::System(_) => {
275                    if let Some((value, increment, units)) =
276                        nested_axis_display(model, local, reference, increments[local])?
277                    {
278                        row.fields[7] = value;
279                        if pixel.is_some() {
280                            row.fields[9] = increment;
281                        }
282                        row.fields[10] = format!(" {units}");
283                    }
284                }
285                _ => (),
286            }
287            rows.push(row);
288            rows.extend(velocity_row);
289        }
290    }
291    rows.sort_by_key(|row| row.order);
292    Ok(rows)
293}

295fn spectral_row(
296    row: &mut ReportRow,
297    spectral: &SpectralCoordinate,
298    record: Option<&RecordValue>,
299    reference: f64,
300    pixel: Option<f64>,
301    doppler: DopplerRef,
302) -> ImageToolResult<Option<ReportRow>> {
303    if !spectral.has_linear_mapping() {
304        row.fields[3].push_str(" (tab)");
305    }
306    let native_units = spectral.axis_units();
307    let native = &native_units[0];
308    let requested = spectral_format_unit(record, native);
309    let unit = Unit::new(requested)?;
310    let convert =
311        |value, absolute| spectral_display_value(spectral, record, value, native, &unit, absolute);
312    let increment = convert(spectral.increment()[0], false)?;
313    let frequency_units = unit.conformant(&Unit::new("Hz").expect("built-in frequency unit"));
314    row.fields[7] = match convert(reference, true)? {
315        Some(value) if frequency_units => {
316            format_number(value, increment.unwrap_or_default(), false)
317        }
318        Some(value) => format_stream_value(value),
319        None => "Fail".into(),
320    };
321    row.fields[10] = format!(" {requested}");
322    if pixel.is_some() {
323        row.fields[9] = match increment {
324            Some(value) if frequency_units => format_number_with_precision(value, value, true, 6),
325            Some(value) => scientific(value, 6),
326            None => "Fail".into(),
327        };
328    }
329
330    let rest = spectral.rest_frequency();
331    if rest <= 0.0 {
332        return Ok(None);
333    }
334    let mut fields: [String; HEADERS.len()] = std::array::from_fn(|_| String::new());
335    fields[3] = "Velocity".into();
336    fields[7] = format_stream_value(velocity(spectral.reference_value()[0], rest, doppler));
337    if let Some(pixel) = pixel {
338        fields[8] = row.fields[8].clone();
339        let higher = spectral.to_world(&[pixel + 0.5]);
340        let lower = spectral.to_world(&[pixel - 0.5]);
341        fields[9] = match (higher, lower) {
342            (Ok(hi), Ok(lo)) => scientific(
343                velocity(hi[0], rest, doppler) - velocity(lo[0], rest, doppler),
344                6,
345            ),
346            _ => "Fail".into(),
347        };
348        fields[10] = " km/s".into();
349    } else {
350        fields[10] = " ".into();
351    }
352    let (axis, offset) = row.order;
353    Ok(Some(ReportRow {
354        order: (axis, offset + 1),
355        fields,
356        labels_only: false,
357    }))
358}

360/// Formats CASA's coordinate summary report section (reference frames, observatory, axis table).
361pub fn format_coordinate_summary_report(
362    record: &RecordValue,
363    coords: &CoordinateSystem,
364    shape: &[usize],
365    tile: &[usize],
366    doppler: DopplerRef,
367) -> ImageToolResult<String> {
368    let mut report = String::from("\n");
369    let direction = CoordinateSystem::first_direction_in_record(record)?;
370    let spectral = CoordinateSystem::first_spectral_in_record(record)?;
371    let obs = coords.obs_info();
372    if let Some((_, direction)) = &direction {
373        let frame = direction.direction_ref();
374        write!(report, "Direction reference : {frame}").unwrap();
375        let conversion = direction.conversion_direction_ref();
376        if conversion != frame {
377            write!(report, " (-> {conversion})").unwrap();
378        }
379        report.push('\n');
380    }
381    if let Some((_, spectral)) = spectral {
382        let frame = spectral.frequency_ref();
383        write!(report, "Spectral  reference : {frame}").unwrap();
384        let conversion = spectral.world_frequency_ref();
385        if conversion != frame {
386            write!(report, " (-> {conversion})").unwrap();
387        }
388        let doppler_name = match doppler {
389            DopplerRef::Z => "OPTICAL",
390            DopplerRef::BETA => "TRUE",
391            other => other.as_str(),
392        };
393        writeln!(report, "\nVelocity  type      : {doppler_name}").unwrap();
394        let rest = spectral.rest_frequency();
395        if rest > 0.0 {
396            let unit = &spectral.axis_units()[0];
397            write!(
398                report,
399                "Rest frequency      : {}",
400                format_stream_value(rest)
401            )
402            .unwrap();
403            let alternatives = spectral.rest_frequencies();
404            if alternatives.len() > 1 {
405                let other = alternatives
406                    .iter()
407                    .copied()
408                    .filter(|value| (value - rest).abs() > 1e-13 * rest.abs().max(value.abs()))
409                    .map(format_stream_value)
410                    .collect::<Vec<_>>()
411                    .join(", ");
412                write!(report, " [{other}]").unwrap();
413            }
414            writeln!(report, " {unit}").unwrap();
415        }
416    }
417    if let Some((_, direction)) = &direction
418        && !obs.pointing_center_initial
419    {
420        let frame = direction.direction_ref();
421        let [longitude, latitude] = obs.pointing_center_rad;
422        let (longitude, _) = format_direction(frame, 0, longitude, true, false, 6);
423        let (latitude, _) = format_direction(frame, 1, latitude, true, false, 6);
424        writeln!(report, "Pointing center     :  {longitude}  {latitude}").unwrap();
425    }
426    writeln!(report, "Telescope           : {}", obs.telescope).unwrap();
427    writeln!(report, "Observer            : {}", obs.observer).unwrap();
428    let date = obs
429        .date
430        .as_ref()
431        .filter(|epoch| epoch.value().as_mjd() != 0.0)
432        .map_or_else(
433            || "UNKNOWN".into(),
434            |epoch| {
435                let date = MvTime::from_mjd_days(epoch.value().as_mjd());
436                format!(
437                    "{:04}/{:02}/{:02}/{}",
438                    date.year(),
439                    date.month(),
440                    date.month_day(),
441                    date.format_time(6)
442                )
443            },
444        );
445    writeln!(report, "Date observation    : {date}").unwrap();
446    if let Some(position) = &obs.telescope_position {
447        let position = position.convert_to(PositionRef::ITRF)?;
448        let [x, y, z] = position.values().map(format_stream_value);
449        writeln!(
450            report,
451            "Telescope position: [{x}m, {y}m, {z}m] ({})",
452            position.refer()
453        )
454        .unwrap();
455    }
456    report.push('\n');
457    let rows = rows(record, shape, tile, doppler)?;
458    let widths = std::array::from_fn(|column| {
459        let content = rows
460            .iter()
461            .filter(|row| !row.labels_only || column < 8)
462            .map(|row| row.fields[column].chars().count())
463            .max()
464            .unwrap_or(0);
465        HEADERS[column].len().max(content) + usize::from(column < 10)
466    });
467    write_table_row(&mut report, &HEADERS, &widths, false);
468    let width = widths.iter().sum::<usize>();
469    writeln!(report, "\n{} ", "-".repeat(width - 1)).unwrap();
470    for (index, row) in rows.iter().enumerate() {
471        if index != 0 {
472            report.push('\n');
473        }
474        write_table_row(&mut report, &row.fields, &widths, row.labels_only);
475    }
476    Ok(report)
477}
478
479fn write_table_row(
480    report: &mut String,
481    fields: &[impl AsRef<str>; HEADERS.len()],
482    widths: &[usize; HEADERS.len()],
483    labels_only: bool,
484) {
485    for (column, (field, &width)) in fields.iter().zip(widths).enumerate() {
486        let field = field.as_ref();
487        match column {
488            0..=3 => write!(report, "{field:<width$}").unwrap(),
489            4..=7 => write!(report, "{field:>width$}").unwrap(),
490            _ if labels_only => break,
491            10 => report.push_str(field),
492            _ => write!(report, "{field:>width$}").unwrap(),
493        }
494    }
