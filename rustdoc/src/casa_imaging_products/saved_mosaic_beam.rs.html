3use std::path::{Path, PathBuf};
4
5use casa_images::{ImageError, PagedImage};
6
7use crate::ProductsError;
8
9/// The primary beam and default mask left by an earlier mosaic run.
10/// Product workers open and read one channel and Stokes plane at a time.
11#[derive(Clone, Debug)]
12pub struct SavedMosaicBeam {
13    path: PathBuf,
14    shape: [usize; 4],
15    reading_bytes: u64,
16}
17
18impl SavedMosaicBeam {
19    /// Tile cache per reader, for the image and its default mask.
20    pub const CACHE_BYTES: usize = 64 * 1024;
21
22    /// Record the dimensions and per-plane read bound from the saved image.
23    ///
24    /// # Errors
25    /// The image does not have four axes or its mask cannot be opened.
26    pub fn new(path: &Path, image: &PagedImage<f32>) -> Result<Self, ImageError> {
27        let shape: [usize; 4] = image.shape().try_into().map_err(|_| {
28            ImageError::InvalidMetadata("the saved mosaic PB must have four axes".into())
29        })?;
30        let cells = (shape[0] * shape[1]) as u64;
31        let pixel_backing = if image.tiled_io_stats().is_some() {
32            let tile = 4 * image.tile_shape().iter().product::<usize>() as u64;
33            Self::CACHE_BYTES as u64 + 2 * tile
34        } else {
35            // A legacy untiled image holds a full table cell while slicing.
36            12 * shape.iter().product::<usize>() as u64
37        };
38        let mask_backing = image.mask_slice_working_bytes()?;
39        Ok(Self {
40            path: path.to_path_buf(),
41            shape,
42            reading_bytes: 8 * cells + pixel_backing + mask_backing,
43        })
44    }
45
46    /// CASA axis order, `[x, y, Stokes, channel]`.
47    #[must_use]
48    pub const fn shape(&self) -> [usize; 4] {
49        self.shape
50    }
51
52    /// Scratch per lane for the pixel or mask read, including the x-major
53    /// copy and both backing stores' bounds. The value copy is the larger.
54    #[must_use]
55    pub const fn reading_bytes(&self) -> u64 {
56        self.reading_bytes
57    }
58
59    /// Read a saved PB plane in x-major order.
60    ///
61    /// # Errors
62    /// The image cannot be opened, sliced, or contains invalid beam values.
63    pub fn read_values(
64        &self,
65        channel: usize,
66        polarization: usize,
67    ) -> Result<Vec<f32>, ProductsError> {
68        let [nx, ny, _, _] = self.shape;
69        let values =
70            self.read(|image| image.get_slice(&[0, 0, polarization, channel], &[nx, ny, 1, 1]))?;
71        if values
72            .iter()
73            .any(|value| !value.is_finite() || *value < 0.0)
74        {
75            return Err(ProductsError::InvalidSavedMosaicBeam);
76        }
77        Ok(values.iter().copied().collect())
78    }
79
80    /// Read the default pixel mask of the same saved plane in x-major order.
81    ///
82    /// # Errors
83    /// The image or mask cannot be opened or sliced.
84    pub fn read_mask(
85        &self,
86        channel: usize,
87        polarization: usize,
88    ) -> Result<Vec<bool>, ProductsError> {
89        let [nx, ny, _, _] = self.shape;
90        let mask = self.read(|image| {
91            image.get_mask_slice(&[0, 0, polarization, channel], &[nx, ny, 1, 1], &[1; 4])
92        })?;
93        Ok(mask.map_or_else(
94            || vec![true; nx * ny],
95            |mask| mask.iter().copied().collect(),
96        ))
97    }
98
99    fn read<T>(
100        &self,
101        slice: impl FnOnce(PagedImage<f32>) -> Result<T, ImageError>,
102    ) -> Result<T, ProductsError> {
103        PagedImage::open_with_cache(&self.path, Self::CACHE_BYTES)
104            .and_then(slice)
105            .map_err(|source| ProductsError::SavedMosaicBeam {
106                path: self.path.clone(),
107                source,
108            })
109    }
110}
