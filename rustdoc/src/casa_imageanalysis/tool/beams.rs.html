20/// Returns the raw casacore `imageinfo` record from `image`, if present on disk or temp storage.
21pub fn image_info_record(image: &ImageHandle) -> Option<&RecordValue> {
22    with_pixels!(image, storage => match storage {
23        Storage::Paged(paged) => match paged.table().keywords().get("imageinfo") {
24            Some(Value::Record(record)) => Some(record),
25            _ => None,
26        },
27        Storage::Temp { image, .. } => image.image_info_record(),
28    })
29}

31/// Reads the parsed [`ImageInfo`] from `image`.
32pub fn image_info(image: &ImageHandle) -> Result<ImageInfo, ImageError> {
33    with_pixels!(image, storage => match storage {
34        Storage::Paged(paged) => paged.image_info(),
35        Storage::Temp { image, .. } => image.image_info(),
36    })
37}

39/// Builds an `ImageInfo` value inheriting non-beam fields (such as `objectname`) from `image`.
40pub fn image_info_with_beams(
41    image: &ImageHandle,
42    beam_set: ImageBeamSet,
43) -> Result<ImageInfo, ImageError> {
44    let mut info = image_info(image)?;
45    info.beam_set = beam_set;
46    if info.image_type == ImageType::Undefined {
47        info.image_type = ImageInfo::default_image_type();
48    }
49    Ok(info)
50}

57/// Formats a [`GaussianBeam`] using CASA's summary precision.
58pub fn format_gaussian_beam(beam: &GaussianBeam) -> ImageToolResult<String> {
59    let major = beam.major_in("arcsec")?;
60    let minor = beam.minor_in("arcsec")?;
61    let pa = beam.position_angle_in("deg")?;
62    Ok(format!(
63        "major: {} arcsec, minor: {} arcsec, pa: {} deg",
64        format_number_with_precision(major, 0.0, false, 6),
65        format_number_with_precision(minor, 0.0, false, 6),
66        format_number_with_precision(pa, 0.0, false, 6)
67    ))
68}

70/// Validates and resolves optional `(channel, polarization)` selectors against `shape` and `coordinates`.
71///
72/// Negative values (`-1`) mean "no specific slice selected" (unless checking per-plane rules).
73pub fn validate_beam_indices(
74    shape: &[usize],
75    coordinates: &CoordinateSystem,
76    channel: isize,
77    polarization: isize,
78    has_multi_beams: bool,
79) -> ImageToolResult<(Option<usize>, Option<usize>)> {
80    let spectral_axis = coordinates.find_coordinate(CoordinateType::Spectral);
81    let stokes_axis = coordinates.find_coordinate(CoordinateType::Stokes);
82    let n_chan = spectral_axis.map_or(1, |axis| shape[axis]);
83    let n_stokes = stokes_axis.map_or(1, |axis| shape[axis]);
84
85    let resolved_chan = if channel < 0 {
86        None
87    } else if spectral_axis.is_none() {
88        return Err(ImageToolError::new(
89            "Channel specified, but image has no spectral coordinate.",
90        ));
91    } else {
92        let ch = usize::try_from(channel).map_err(|_| "Invalid negative channel index")?;
93        if ch >= n_chan {
94            return Err(ImageToolError::new(format!(
95                "Channel {channel} is out of range for spectral axis of length {n_chan}"
96            )));
97        }
98        Some(ch)
99    };
100
101    let resolved_stokes = if polarization < 0 {
102        None
103    } else if stokes_axis.is_none() {
104        return Err(ImageToolError::new(
105            "Polarization specified, but image has no polarization coordinate.",
106        ));
107    } else {
108        let pol =
109            usize::try_from(polarization).map_err(|_| "Invalid negative polarization index")?;
110        if pol >= n_stokes {
111            return Err(ImageToolError::new(format!(
112                "Polarization {polarization} is out of range for polarization axis of length {n_stokes}"
113            )));
114        }
115        Some(pol)
116    };
117
118    if has_multi_beams
119        && ((spectral_axis.is_some() && n_chan > 1 && resolved_chan.is_none())
120            || (stokes_axis.is_some() && n_stokes > 1 && resolved_stokes.is_none()))
121        && (resolved_chan.is_some() || resolved_stokes.is_some())
122    {
123        return Err(ImageToolError::new(
124            "Both channel and polarization must be specified when querying a per-plane beam set.",
125        ));
126    }
127
128    Ok((resolved_chan, resolved_stokes))
129}

131/// Parses a `RecordValue` (direct `major`/`minor`/`positionangle` or `restoringbeam` wrapper) into [`GaussianBeam`].
132pub fn parse_beam_record(record: &RecordValue) -> ImageToolResult<GaussianBeam> {
133    if let Some(Value::Record(inner)) = record.get("restoringbeam") {
134        return Ok(GaussianBeam::from_record(inner)?);
135    }
136    let _ = StoredGaussianBeam::from_record;
137    Ok(GaussianBeam::from_record(record)?)
138}

140/// Subsets a multi-plane `beam_set` for dropped/sliced spectral and Stokes axes.
141pub fn subset_beam_set(
142    beams: &ImageBeamSet,
143    channels: Option<&[usize]>,
144    stokes: Option<&[usize]>,
145) -> ImageToolResult<ImageBeamSet> {
146    if beams.is_empty() || beams.is_single() {
147        return Ok(beams.clone());
148    }
149    let chan_indices: Vec<usize> = match channels {
150        Some(slice) if beams.n_channels() > 1 => slice.to_vec(),
151        _ => (0..beams.n_channels()).collect(),
152    };
153    let stokes_indices: Vec<usize> = match stokes {
154        Some(slice) if beams.n_stokes() > 1 => slice.to_vec(),
155        _ => (0..beams.n_stokes()).collect(),
156    };
157    Ok(beams.subset(&chan_indices, &stokes_indices)?)
158}

162    match record.get(field) {
163        Some(Value::Record(value)) => Some(value),
164        _ => None,

168/// Looks up the stored casacore beam record for `(channel, polarization)` from `info` or `beams`.
169pub fn stored_beam_record(
170    info: Option<&RecordValue>,
171    beams: &ImageBeamSet,
172    channel: usize,
173    polarization: usize,
174) -> RecordValue {
175    if let Some(info) = info {
176        if let Some(grid) = subrecord(info, "perplanebeams") {
177            let index = polarization * beams.n_channels() + channel;
178            if let Some(beam) = subrecord(grid, &format!("*{index}"))
179                .or_else(|| subrecord(grid, &format!("*{channel}_{polarization}")))
180            {
181                return beam.clone();
182            }
183        } else if let Some(beam) = subrecord(info, "restoringbeam") {
184            return beam.clone();
185        }
186    }
187    beams.beam(channel, polarization).to_record()
188}

190/// Updates `output`'s `restoringbeam` / `perplanebeams` for the selected `(channels, polarizations)` subset.
191pub fn subset_beam_record(
192    output: &mut RecordValue,
193    stored: Option<&RecordValue>,
194    beams: &ImageBeamSet,
195    channels: &[usize],
196    polarizations: &[usize],
197) -> Result<(), ImageError> {
198    let subset = beams.subset(channels, polarizations)?;
199    let beam = |channel, polarization| {
200        Value::Record(stored_beam_record(stored, beams, channel, polarization))
201    };
202    output.remove("perplanebeams");
203    output.remove("restoringbeam");
204    if subset.is_single() {
205        output.upsert("restoringbeam", beam(channels[0], polarizations[0]));
206    } else {
207        let mut grid = subset.to_record();
208        for (plane, &polarization) in polarizations.iter().enumerate() {
209            for (slot, &channel) in channels.iter().enumerate() {
210                grid.upsert(
211                    format!("*{}", plane * channels.len() + slot),
212                    beam(channel, polarization),
213                );
214            }
215        }
216        output.upsert("perplanebeams", Value::Record(grid));
217    }
218    Ok(())
219}

221/// Resolves the pixel axis indices of `(spectral, stokes)` from a raw coordinate record.
222pub fn plane_axes(record: &RecordValue) -> ImageToolResult<(Option<usize>, Option<usize>)> {
223    let mut axes = CoordinateSystem::record_coordinates(record)?;
224    // CASA assigns beams by coordinate creation order, including after transposition.
225    axes.sort_by_key(|axis| {
226        axis.field
227            .trim_start_matches(|character: char| character.is_ascii_alphabetic())
228            .parse::<usize>()
229            .ok()
230    });
231    let mut spectral = None;
232    let mut stokes = None;
233    for axis in axes {
234        let kind = if axis.field.starts_with("spectral") {
235            CoordinateType::Spectral
236        } else if axis.field.starts_with("stokes") {
237            CoordinateType::Stokes
238        } else if axis.field.starts_with("coordinate") {
239            axis.model(record)?.coordinate_type()
240        } else {
241            continue;
242        };
243        let selected = match kind {
244            CoordinateType::Spectral => &mut spectral,
245            CoordinateType::Stokes => &mut stokes,
246            _ => continue,
247        };
248        selected.get_or_insert_with(|| axis.pixel_axes.first().copied().flatten());
249        if spectral.is_some() && stokes.is_some() {
250            break;
251        }
252    }
253    Ok((spectral.flatten(), stokes.flatten()))
254}

256/// Returns the length of `axis` in `image`, or `1` when `axis` is `None`.
257pub fn axis_length(image: &ImageHandle, axis: Option<usize>) -> ImageToolResult<usize> {
258    match axis {
259        Some(axis) => {
260            image.shape().get(axis).copied().ok_or_else(|| {
261                ImageToolError::new("Image coordinate axis is outside the image shape")
262            })
263        }
264        None => Ok(1),
265    }
266}

268fn beam_matrix_planes(
269    beams: &ImageBeamSet,
270    stokes_first: bool,
271) -> ([usize; 2], impl Iterator<Item = (usize, usize)>) {
272    let channels = beams.n_channels();
273    let stokes = beams.n_stokes();
274    let (rows, columns) = if stokes_first {
275        (stokes, channels)
276    } else {
277        (channels, stokes)
278    };
279    let planes = (0..columns).flat_map(move |column| {
280        (0..rows).map(move |row| {
281            if stokes_first {
282                (column, row)
283            } else {
284                (row, column)
285            }
286        })
287    });
288    ([rows, columns], planes)
289}

291fn beam_matrix_array(
292    beams: &ImageBeamSet,
293    stokes_first: bool,
294    value: impl Fn(&GaussianBeam) -> Result<f64, ImageError>,
295) -> ImageToolResult<Value> {
296    let (shape, planes) = beam_matrix_planes(beams, stokes_first);
297    let mut values = Vec::with_capacity(beams.nelements());
298    for (channel, polarization) in planes {
299        values.push(value(beams.beam(channel, polarization))?);
300    }
301    let array = ArrayD::from_shape_vec(IxDyn(&shape).f(), values)
302        .map_err(|err| ImageToolError::new(err.to_string()))?;
303    Ok(Value::Array(ArrayValue::Float64(array)))
304}

313fn beam_parameter_matrices(
314    info: &RecordValue,
315    beams: &ImageBeamSet,
316    stokes_first: bool,
317) -> ImageToolResult<Vec<(&'static str, RecordValue)>> {
318    let (shape, planes) = beam_matrix_planes(beams, stokes_first);
319    let grid = subrecord(info, "perplanebeams")
320        .ok_or_else(|| ImageToolError::new("Missing per-plane beams"))?;
321    let mut stored_grid = HashMap::with_capacity(beams.nelements());
322    for field in grid.fields() {
323        stored_grid
324            .entry(field.name.as_str())
325            .or_insert(&field.value);
326    }
327    let parameters = [
328        ("major", BeamComponent::Major, "arcsec"),
329        ("minor", BeamComponent::Minor, "arcsec"),
330        ("pa", BeamComponent::PositionAngle, "deg"),
331    ];
332    let units = parameters.map(|(_, _, unit)| Unit::new(unit).expect("built-in angular unit"));
333    let mut arrays: [Vec<f64>; 3] = std::array::from_fn(|_| Vec::with_capacity(beams.nelements()));
334    for (channel, polarization) in planes {
335        if beams.beam(channel, polarization).is_null() {
336            return Err(ImageToolError::new("Invalid per plane beam found"));
337        }
338        let linear_index = polarization * beams.n_channels() + channel;
339        let lookup = |name: &str| match stored_grid.get(name) {
340            Some(Value::Record(record)) => Some(record),
341            _ => None,
342        };
343        let stored = lookup(&format!("*{linear_index}"))
344            .or_else(|| lookup(&format!("*{channel}_{polarization}")))
345            .ok_or_else(|| ImageToolError::new("Missing per-plane beam"))?;
346        let beam = StoredGaussianBeam::from_record(stored)?;
347        for (index, (_, component, _)) in parameters.iter().enumerate() {
348            arrays[index].push(casacore::get_conformant_value(
349                beam.component(*component),
350                &units[index],
351            )?);
352        }
353    }
354    parameters
355        .into_iter()
356        .zip(arrays)
357        .map(|((name, _, unit), values)| {
358            let array = ArrayD::from_shape_vec(IxDyn(&shape).f(), values)
359                .map_err(|err| ImageToolError::new(err.to_string()))?;
360            let rec = RecordValue::new(vec![
361                RecordField::new("unit", Value::from(unit.to_owned())),
362                RecordField::new("value", Value::Array(ArrayValue::Float64(array))),
363            ]);
364            Ok((name, rec))
365        })
366        .collect()
367}

369/// Returns the restoring beam record matching CASA's `ia.restoringbeam(channel, polarization, mbret)`.
370pub fn restoring_beam_record(
371    view: &BeamView,
372    coordinate_record: impl FnOnce() -> RecordValue,
373    channel: i64,
374    polarization: i64,
375    mode: &str,
376) -> ImageToolResult<RecordValue> {
377    let beams = view
378        .beam_set()
379        .map_err(|err| ImageToolError::new(err.to_string()))?;
380    let first = mode.as_bytes().first().map(u8::to_ascii_lowercase);
381    let matrix = first == Some(b'm');
382    if !matrix && first != Some(b'l') {
383        return Err(ImageToolError::new(format!(
384            "Unsupported mbret value {mode}"
385        )));
386    }
387    let indexed = channel >= 0 || polarization >= 0;
388    if matrix {
389        if !beams.is_multi() {
390            return Err(ImageToolError::new(
391                "Exception: mbret='matrix' only makes sense if the image has per-plane beams. This image does not.",
392            ));
393        }
394        if indexed {
395            return Err(ImageToolError::new(
396                "Exception: mbret='matrix' only makes sense if both channel and polarization are negative.",
397            ));
398        }
399    }
400    if beams.is_empty() {
401        return Ok(RecordValue::default());
402    }
403    let info = Some(view.record());
404    if beams.is_single() {
405        return Ok(stored_beam_record(info, &beams, 0, 0));
406    }
407    let channels = beams.n_channels() as i64;
408    let stokes = beams.n_stokes() as i64;
409    if indexed {
410        if channels > 1 && stokes > 1 && (channel < 0 || polarization < 0) {
411            return Err(ImageToolError::new(
412                "Exception: In the case of an image with both a nondegenerate spectral axis and a nondegenerate polarization axis either both channel and polarization must be positive, or both must be negative.",
413            ));
414        }
415        if channels > 1 && channel >= channels {
416            return Err(ImageToolError::new(format!(
417                "Exception: channel value specified as {channel} must be less than the number of channels in the image, which is {channels}."
418            )));
419        }
420        if stokes > 1 && polarization >= stokes {
421            return Err(ImageToolError::new(format!(
422                "Exception: polarization value must be less than the number of polarization planes in the image, which is {stokes}."
423            )));
424        }
425        if (channels > 1 && channel < 0) || (stokes > 1 && polarization < 0) {
426            return Err(ImageToolError::new(
427                "Failed AlwaysAssert chan >=0 && chan < Int(nchan()) && stokes >= 0 && stokes < Int(nstokes())",
428            ));
429        }
430        let channel = if channels == 1 { 0 } else { channel as usize };
431        let polarization = if stokes == 1 {
432            0
433        } else {
434            polarization as usize
435        };
436        return Ok(stored_beam_record(info, &beams, channel, polarization));
437    }
438    let mut fields = vec![
439        RecordField::new("nChannels", Value::Scalar(ScalarValue::Int64(channels))),
440        RecordField::new("nStokes", Value::Scalar(ScalarValue::Int64(stokes))),
441    ];
442    if matrix {
443        let (spectral, stokes) = plane_axes(&coordinate_record())?;
444        let stokes_first = stokes.is_some_and(|stokes| {
445            spectral.is_none_or(|spectral| spectral == 0 || spectral >= stokes)
446        });
447        for (name, rec) in beam_parameter_matrices(view.record(), &beams, stokes_first)? {
448            fields.push(RecordField::new(name, Value::Record(rec)));
449        }
450    } else {
451        let mut by_channel = RecordValue::default();
452        for channel in 0..beams.n_channels() {
453            let mut by_stokes = RecordValue::default();
454            for polarization in 0..beams.n_stokes() {
455                by_stokes.upsert(
456                    format!("*{polarization}"),
457                    Value::Record(stored_beam_record(info, &beams, channel, polarization)),
458                );
459            }
460            by_channel.upsert(format!("*{channel}"), Value::Record(by_stokes));
461        }
462        fields.push(RecordField::new("beams", Value::Record(by_channel)));
463    }
464    Ok(RecordValue::new(fields))
465}

467/// Computes the CASA `ia.beamarea(channel, polarization)` record (`arcsec2` and `pixels`).
468pub fn beam_area_record(
469    image: &ImageHandle,
470    view: &BeamView,
471    channel: i64,
472    polarization: i64,
473) -> ImageToolResult<RecordValue> {
474    let beams = view
475        .beam_set()
476        .map_err(|err| ImageToolError::new(err.to_string()))?;
477    if beams.is_empty() {
478        return Err(ImageToolError::new(
479            "Exception: This image has no beam(s). The setrestoringbeam() method may be used to set its beam(s).",
480        ));
481    }
482    let coord_record = image.coordinate_record();
483    let coords = CoordinateSystem::from_record(&coord_record)?;
484    let direction = coords
485        .find_coordinate(CoordinateType::Direction)
486        .ok_or_else(|| {
487            ImageToolError::new(
488                "directionCoordinate: Coordinate system has no direction coordinate",
489            )
490        })?;
491    let increment = coords.coordinate(direction).increment();
492    let pixel_area = (increment[0] * increment[1]).abs();
493    let to_arcsec = 3600.0 * 180.0 / std::f64::consts::PI;
494    let beam_in_arcsec = |area: f64| area * to_arcsec * to_arcsec;
495    let single = |area| {
496        RecordValue::new(vec![
497            RecordField::new(
498                "arcsec2",
499                Value::Scalar(ScalarValue::Float64(beam_in_arcsec(area))),
500            ),
501            RecordField::new(
502                "pixels",
503                Value::Scalar(ScalarValue::Float64(area / pixel_area)),
504            ),
505        ])
506    };
507    if let Some(beam) = beams.single_beam() {
508        return Ok(single(beam.area()));
509    }
510    let (spectral, stokes) = plane_axes(&coord_record)?;
511    let channels = axis_length(image, spectral)? as i64;
512    let polarizations = axis_length(image, stokes)? as i64;
513    if channel >= channels {
514        return Err(ImageToolError::new(format!(
515            "Exception: This image only has {channels} channels, so a channel value less than {channels} must be specified."
516        )));
517    }
518    if polarization >= polarizations {
519        return Err(ImageToolError::new(format!(
520            "Exception: This image only has {polarizations} polarizations, so a polarization value less than {polarizations} must be specified."
521        )));
522    }
523    if channel >= 0 && polarization >= 0 {
524        if (beams.n_channels() > 1 && channel as usize >= beams.n_channels())
525            || (beams.n_stokes() > 1 && polarization as usize >= beams.n_stokes())
526        {
527            return Err(ImageToolError::new(
528                "Selected plane is outside the restoring beam grid",
529            ));
530        }
531        return Ok(single(
532            beams.beam(channel as usize, polarization as usize).area(),
533        ));
534    }
535    if channel >= 0 || polarization >= 0 {
536        return Err(ImageToolError::new(
537            "Exception: In the case of a multibeam image, either both channel and polarization must be non-negative, or both must be negative.",
538        ));
539    }
540    let stokes_first = matches!((stokes, spectral), (Some(s), Some(c)) if s < c);
541    Ok(RecordValue::new(vec![
542        RecordField::new(
543            "arcsec2",
544            beam_matrix_array(&beams, stokes_first, |beam| Ok(beam_in_arcsec(beam.area())))?,
545        ),
546        RecordField::new(
547            "pixels",
548            beam_matrix_array(&beams, stokes_first, |beam| Ok(beam.area() / pixel_area))?,
549        ),
550    ]))
551}

553/// Computes CASA's `ia.commonbeam()` record (`pa`, `major`, `minor` quantities).
554pub fn common_beam_record(image: &ImageHandle, view: &BeamView) -> ImageToolResult<RecordValue> {
555    image.check_single_precision("commonbeam")?;
556    let beams = view
557        .beam_set()
558        .map_err(|err| ImageToolError::new(err.to_string()))?;
559    if beams.is_empty() {
560        return Err(ImageToolError::new("Exception: This image has no beam(s)."));
561    }
562    let common = beams.common_beam()?;
563    let info = Some(view.record());
564    let matching_plane = (0..beams.n_stokes()).find_map(|polarization| {
565        (0..beams.n_channels())
566            .find(|&channel| *beams.beam(channel, polarization) == common)
567            .map(|channel| (channel, polarization))
568    });
569    let record = if let Some((channel, polarization)) = matching_plane {
570        stored_beam_record(info, &beams, channel, polarization)
571    } else {
572        common.to_record()
573    };
574    let stored = StoredGaussianBeam::from_record(&record)?;
575    let degrees = Unit::new("deg").expect("built-in degrees");
576    let angle = casacore::convert(&stored.wrapped_position_angle()?, Some(&degrees))?;
577    let mut fields = vec![RecordField::new(
578        "pa",
579        Value::Record(quantity_record(angle.value(), "deg")),
580    )];
581    for component in ["major", "minor"] {
582        fields.push(RecordField::new(
583            component,
584            Value::Record(
585                subrecord(&record, component)
586                    .expect("validated beam component")
587                    .clone(),
588            ),
589        ));
590    }
591    Ok(RecordValue::new(fields))
592}

594/// Returns the stored beam for unit conversions (`convertflux`) at `(channel, polarization)`.
595pub fn beam_for_units(
596    current: &BeamView,
597    channel: i64,
598    polarization: i64,
599) -> ImageToolResult<Option<StoredGaussianBeam>> {
600    let beams = current
601        .beam_set()
602        .map_err(|err| ImageToolError::new(err.to_string()))?;
603    if beams.is_empty() {
604        return Ok(None);
605    }
606    let index = |index: i64, length: usize| {
607        if length == 1 {
608            Ok(0)
609        } else if index >= 0 && (index as usize) < length {
610            Ok(index as usize)
611        } else {
612            Err(ImageToolError::new(
613                "Failed AlwaysAssert chan >=0 && chan < Int(nchan()) && stokes >= 0 && stokes < Int(nstokes())",
614            ))
615        }
616    };
617    let record = stored_beam_record(
618        Some(current.record()),
619        &beams,
620        index(channel, beams.n_channels())?,
621        index(polarization, beams.n_stokes())?,
622    );
623    Ok(Some(StoredGaussianBeam::from_record(&record)?))
624}
