25/// Seconds per day, for MSv2 times (MJD seconds).
26const SECONDS_PER_DAY: f64 = 86_400.0;
27
28/// `epoch` as an MSv2 subtable time: MJD seconds in UTC, the scale the
29/// columns this writer writes declare.
30pub(super) fn utc_seconds(epoch: &MEpoch) -> Result<f64, ObservationError> {
31    if epoch.refer() != EpochRef::UTC {
32        return Err(ObservationError::Unsupported {
33            what: "writing MSv2 subtable times in another scale than UTC",
34        });
35    }
36    let value = epoch.value();
37    Ok((value.day() + value.frac()) * SECONDS_PER_DAY)
38}

40/// `direction`'s angles, which an MSv2 field holds in J2000 only.
41fn j2000(direction: &MDirection) -> Result<[f64; 2], ObservationError> {
42    if direction.refer() != DirectionRef::J2000 {
43        return Err(ObservationError::Unsupported {
44            what: "writing MSv2 field directions in another frame than J2000",
45        });
46    }
47    let (longitude, latitude) = direction.as_angles();
48    Ok([longitude, latitude])
49}

51/// The `TIME` of `field`, catalogue field `index`, and whether the field
52/// states it: its epoch, or else `time`, the data's first, but for a field
53/// that follows an ephemeris its own first time in `first` (by field): an
54/// image without a phase centre is centred on its body at that time (#987).
55///
56/// # Errors
57///
58/// An epoch in another scale than UTC.
59pub(super) fn field_time(
60    field: &FieldEntry,
61    index: usize,
62    (time, first): (f64, &[Option<f64>]),
63) -> Result<(f64, bool), ObservationError> {
64    if let Some(epoch) = &field.reference_epoch {
65        return Ok((utc_seconds(epoch)?, true));
66    }
67    let own = field
68        .ephemeris
69        .as_ref()
70        .and_then(|_| first.get(index).copied().flatten());
71    Ok((own.unwrap_or(time), false))
72}

74/// One `FIELD` row per catalogue field, its source named by `SOURCE_ID`
75/// as [`msv2_sources`] writes them (`named`), its `TIME` as [`field_time`]
76/// gives it from `times` (the data's first, and each field's first).
77pub(super) fn fields(
78    ms: &mut MeasurementSet,
79    catalogue: &Catalogue,
80    (named, times): (&NamedSources, (f64, &[Option<f64>])),
81    notes: &mut Notes,
82) -> Result<(), ObservationError> {
83    let table = subtable_mut(ms, SubtableId::Field).map_err(storage)?;
84    for (index, field) in catalogue.fields.iter().enumerate() {
85        let phase = &field.phase_centre;
86        let delay = notes.or(field.delay_centre.as_ref(), "FIELD DELAY_DIR", || phase);
87        let reference = notes.or(
88            field.reference_centre.as_ref(),
89            "FIELD REFERENCE_DIR",
90            || phase,
91        );
92        let direction = |centre| j2000(centre).map(|angles| f64_array(&angles, vec![2, 1]));
93        let (seconds, stated) = field_time(field, index, times)?;
94        let row = row_from_defs(
95            schema::field::REQUIRED_COLUMNS,
96            &[
97                ("NAME", s(&field.name)),
98                (
99                    "CODE",
100                    s(&notes.or(field.code.clone(), "FIELD CODE", String::new)),
101                ),
102                ("NUM_POLY", i(0)),
103                ("DELAY_DIR", direction(delay)?),
104                ("PHASE_DIR", direction(phase)?),
105                ("REFERENCE_DIR", direction(reference)?),
106                (
107                    "SOURCE_ID",
108                    i(msv2_sources::field_source(catalogue, field, named)),
109                ),
110                (
111                    "TIME",
112                    f(notes.or(stated.then_some(seconds), "FIELD TIME", || seconds)),
113                ),
114                (
115                    "FLAG_ROW",
116                    b(notes.or(field.flagged, "FIELD FLAG_ROW", || false)),
117                ),
118            ],
119        );
120        table.add_row(row).map_err(storage)?;
121    }
122    Ok(())
123}

125/// One `OBSERVATION` row per session; a session that states no extent
126/// spans the times its partitions span (`ranges`, by session), edge to
127/// edge.
128pub(super) fn observations(
129    ms: &mut MeasurementSet,
130    catalogue: &Catalogue,
131    ranges: &[[f64; 2]],
132    notes: &mut Notes,
133) -> Result<(), ObservationError> {
134    let table = subtable_mut(ms, SubtableId::Observation).map_err(storage)?;
135    for (observation, span) in catalogue.observations.iter().zip(ranges) {
136        // The archive identifiers as importasdm writes them, which is how
137        // the MSv2 adapter and XRADIO's converter read them back.
138        let schedule = schedule_lines(
139            observation.scheduling_block.as_deref(),
140            observation.execution_block.as_deref(),
141        );
142        let schedule: Vec<&str> = schedule.iter().map(String::as_str).collect();
143        let time_range = observation
144            .time_range
145            .as_ref()
146            .map(|[start, end]| Ok::<_, ObservationError>([utc_seconds(start)?, utc_seconds(end)?]))
147            .transpose()?;
148        let time_range = notes.or(time_range, "OBSERVATION TIME_RANGE", || *span);
149        let release = observation
150            .release_date
151            .as_ref()
152            .map(utc_seconds)
153            .transpose()?;
154        let row = row_from_defs(
155            schema::observation::REQUIRED_COLUMNS,
156            &[
157                (
158                    "FLAG_ROW",
159                    b(notes.or(observation.flagged, "OBSERVATION FLAG_ROW", || false)),
160                ),
161                ("LOG", string_array(&[], vec![0])),
162                ("OBSERVER", s(&observation.observers.join(", "))),
163                ("PROJECT", s(&observation.project)),
164                (
165                    "RELEASE_DATE",
166                    f(notes.or(release, "OBSERVATION RELEASE_DATE", || 0.0)),
167                ),
168                ("SCHEDULE", string_array(&schedule, vec![schedule.len()])),
169                ("SCHEDULE_TYPE", s("")),
170                ("TELESCOPE_NAME", s(&observation.telescope)),
171                ("TIME_RANGE", f64_array(&time_range, vec![2])),
172            ],
173        );
174        table.add_row(row).map_err(storage)?;
175    }
176    Ok(())
177}

179/// Whether two partitions of one window hold its channels alike: the same
180/// frame, frequencies and widths, bit for bit.
181fn same_channels(a: &PartitionEntry, b: &PartitionEntry) -> bool {
182    let bits = |values: &[f64]| values.iter().map(|v| v.to_bits()).collect::<Vec<_>>();
183    a.frequency_frame == b.frequency_frame
184        && bits(&a.frequencies_hz) == bits(&b.frequencies_hz)
185        && bits(&a.channel_widths_hz) == bits(&b.channel_widths_hz)
186}

188/// One `SPECTRAL_WINDOW` row per catalogue window a partition uses, with
189/// the channels its partitions hold, and the row of each window: `None`
190/// for a window no partition uses, which has no channels to write and is a
191/// loss.
192pub(super) fn spectral_windows(
193    ms: &mut MeasurementSet,
194    catalogue: &Catalogue,
195    notes: &mut Notes,
196) -> Result<Vec<Option<i32>>, ObservationError> {
197    let table = subtable_mut(ms, SubtableId::SpectralWindow).map_err(storage)?;
198    let mut rows = Vec::with_capacity(catalogue.windows.len());
199    let mut unused = 0;
200    for (index, window) in catalogue.windows.iter().enumerate() {
201        let mut partitions = catalogue
202            .partitions
203            .iter()
204            .filter(|entry| entry.window.0 as usize == index);
205        let Some(first) = partitions.next() else {
206            unused += 1;
207            rows.push(None);
208            continue;
209        };
210        if partitions.any(|entry| !same_channels(entry, first)) {
211            return Err(ObservationError::Unsupported {
212                what: "writing one MSv2 spectral window for partitions with different channels",
213            });
214        }
215        let channels = first.frequencies_hz.len();
216        let widths = &first.channel_widths_hz;
217        let magnitudes: Vec<f64> = widths.iter().map(|width| width.abs()).collect();
218        let reference = match &window.reference_frequency {
219            Some(frequency) if frequency.refer() == first.frequency_frame => Some(frequency.hz()),
220            Some(_) => {
221                return Err(ObservationError::Unsupported {
222                    what: "writing an MSv2 reference frequency in another frame than its channels",
223                });
224            }
225            None => None,
226        };
227        let mut per_channel = |values: &Option<Vec<f64>>, column: &str| match values {
228            Some(values) if values.len() == channels => Ok(values.clone()),
229            Some(_) => Err(request_error(format!(
230                "{column} of window {} holds other than its {channels} channels",
231                window.name
232            ))),
233            None => Ok(notes.or(None, column, || magnitudes.clone())),
234        };
235        let effective = per_channel(
236            &window.effective_bandwidths_hz,
237            "SPECTRAL_WINDOW EFFECTIVE_BW",
238        )?;
239        let resolution = per_channel(&window.resolutions_hz, "SPECTRAL_WINDOW RESOLUTION")?;
240        let row = row_from_defs(
241            schema::spectral_window::REQUIRED_COLUMNS,
242            &[
243                ("NUM_CHAN", i(channels as i32)),
244                ("NAME", s(&window.name)),
245                (
246                    "REF_FREQUENCY",
247                    f(notes.or(reference, "SPECTRAL_WINDOW REF_FREQUENCY", || {
248                        first.frequencies_hz[0]
249                    })),
250                ),
251                (
252                    "TOTAL_BANDWIDTH",
253                    f(notes.or(
254                        window.total_bandwidth_hz,
255                        "SPECTRAL_WINDOW TOTAL_BANDWIDTH",
256                        || magnitudes.iter().sum(),
257                    )),
258                ),
259                (
260                    "CHAN_FREQ",
261                    f64_array(&first.frequencies_hz, vec![channels]),
262                ),
263                ("CHAN_WIDTH", f64_array(widths, vec![channels])),
264                ("EFFECTIVE_BW", f64_array(&effective, vec![channels])),
265                ("RESOLUTION", f64_array(&resolution, vec![channels])),
266                ("MEAS_FREQ_REF", i(first.frequency_frame.casacore_code())),
267                (
268                    "NET_SIDEBAND",
269                    i(
270                        notes.or(window.net_sideband, "SPECTRAL_WINDOW NET_SIDEBAND", || {
271                            if widths[0] >= 0.0 { 1 } else { -1 }
272                        }),
273                    ),
274                ),
275                (
276                    "FREQ_GROUP",
277                    i(notes.or(window.frequency_group, "SPECTRAL_WINDOW FREQ_GROUP", || 0)),
278                ),
279                (
280                    "FREQ_GROUP_NAME",
281                    s(&notes.or(
282                        window.frequency_group_name.clone(),
283                        "SPECTRAL_WINDOW FREQ_GROUP_NAME",
284                        String::new,
285                    )),
286                ),
287                (
288                    "IF_CONV_CHAIN",
289                    i(notes.or(window.if_chain, "SPECTRAL_WINDOW IF_CONV_CHAIN", || 0)),
290                ),
291                (
292                    "FLAG_ROW",
293                    b(notes.or(window.flagged, "SPECTRAL_WINDOW FLAG_ROW", || false)),
294                ),
295            ],
296        );
297        table.add_row(row).map_err(storage)?;
298        rows.push(Some(table.row_count() as i32 - 1));
299    }
300    if unused > 0 {
301        notes.lost(Loss::Table(format!(
302            "SPECTRAL_WINDOW rows no partition uses ({unused})"
303        )));
304    }
305    for column in ["CHAN_FREQ", "REF_FREQUENCY"] {
306        TableMeasDesc::new_variable_int(
307            column,
308            MeasureType::Frequency,
309            "MEAS_FREQ_REF",
310            FrequencyRef::ALL
311                .iter()
312                .map(|refer| refer.as_str().to_owned())
313                .collect(),
314            FrequencyRef::ALL
315                .iter()
316                .map(|refer| refer.casacore_code())
317                .collect(),
318        )
319        .and_then(|description| description.write(table))
320        .map_err(storage)?;
321    }
322    Ok(rows)
323}

325/// What the `STATE` rows written say of each partition, in catalogue
326/// order: the `OBS_MODE` the MSv2 reader keys its rows by, and the
327/// `STATE_ID` of its state of intents, if it needs one.
328pub(super) struct States {
329    pub(super) modes: Vec<String>,
330    pub(super) intent_states: Vec<Option<i32>>,
331}
332
333/// One `STATE` row per catalogue state, then one per `OBS_MODE` of the
334/// partitions some of whose rows state intents but no state (an MSv4's),
335/// as `intents` says of each, their other columns MSv2's defaults. The
336/// rows that do state a state are written with it ([`super::msv2_rows`]).
337pub(super) fn states(
338    ms: &mut MeasurementSet,
339    catalogue: &Catalogue,
340    intents: &[bool],
341    notes: &mut Notes,
342) -> Result<States, ObservationError> {
343    let table = subtable_mut(ms, SubtableId::State).map_err(storage)?;
344    for state in &catalogue.states {
345        let row = row_from_defs(
346            schema::state::REQUIRED_COLUMNS,
347            &[
348                ("OBS_MODE", s(&state.obs_mode)),
349                ("CAL", f(notes.or(state.cal_k, "STATE CAL", || 0.0))),
350                ("LOAD", f(notes.or(state.load_k, "STATE LOAD", || 0.0))),
351                ("SIG", b(notes.or(state.signal, "STATE SIG", || true))),
352                ("REF", b(notes.or(state.reference, "STATE REF", || false))),
353                (
354                    "SUB_SCAN",
355                    i(notes.or(state.sub_scan, "STATE SUB_SCAN", || 0)),
356                ),
357                (
358                    "FLAG_ROW",
359                    b(notes.or(state.flagged, "STATE FLAG_ROW", || false)),
360                ),
361            ],
362        );
363        table.add_row(row).map_err(storage)?;
364    }
365    let mut stated: Vec<String> = Vec::new();
366    let mut states = States {
367        modes: Vec::with_capacity(catalogue.partitions.len()),
368        intent_states: Vec::with_capacity(catalogue.partitions.len()),
369    };
370    for (entry, needs) in catalogue.partitions.iter().zip(intents) {
371        let mode = match entry.states.first() {
372            Some(first) => catalogue
373                .states
374                .get(first.0 as usize)
375                .ok_or_else(|| request_error("a partition names a state the catalogue lacks"))?
376                .obs_mode
377                .clone(),
378            None => entry.intents.join(","),
379        };
380        let intent_state = if *needs {
381            let index = match stated.iter().position(|known| *known == mode) {
382                Some(index) => index,
383                None => {
384                    let row = row_from_defs(
385                        schema::state::REQUIRED_COLUMNS,
386                        &[
387                            ("OBS_MODE", s(&mode)),
388                            ("SIG", b(true)),
389                            ("REF", b(false)),
390                            ("FLAG_ROW", b(false)),
391                        ],
392                    );
393                    table.add_row(row).map_err(storage)?;
394                    notes.defaulted("STATE CAL, LOAD, SIG, REF, SUB_SCAN and FLAG_ROW");
395                    stated.push(mode.clone());
396                    stated.len() - 1
397                }
398            };
399            Some((catalogue.states.len() + index) as i32)
400        } else {
401            None
402        };
403        states.modes.push(mode);
404        states.intent_states.push(intent_state);
405    }
406    Ok(states)
407}

409/// Which of `catalogue`'s partitions state intents but no state: of one
410/// input, the partitions that need a state of intents.
411#[cfg(feature = "msv4")]
412pub(super) fn intent_partitions(catalogue: &Catalogue) -> Vec<bool> {
413    catalogue
414        .partitions
415        .iter()
416        .map(|entry| entry.states.is_empty() && !entry.intents.is_empty())
417        .collect()
418}

420/// One `PROCESSOR` row per catalogue processor, `PASS_ID` among them when
421/// some processor states one. Rows name theirs, -1 for none
422/// ([`super::msv2_rows`]).
423pub(super) fn processors(
424    ms: &mut MeasurementSet,
425    catalogue: &Catalogue,
426    notes: &mut Notes,
427) -> Result<(), ObservationError> {
428    let table = subtable_mut(ms, SubtableId::Processor).map_err(storage)?;
429    let passes = catalogue
430        .processors
431        .iter()
432        .any(|processor| processor.pass_id.is_some());
433    if passes {
434        let definition = optional_columns(SubtableId::Processor)
435            .iter()
436            .find(|definition| definition.name == "PASS_ID")
437            .expect("PROCESSOR's PASS_ID");
438        table
439            .add_column(
440                ColumnSchema::scalar(definition.name, definition.data_type),
441                None,
442            )
443            .map_err(storage)?;
444    }
445    for processor in &catalogue.processors {
446        let row = [
447            (
448                "TYPE",
449                s(&notes.or(processor.kind.clone(), "PROCESSOR TYPE", String::new)),
450            ),
451            (
452                "SUB_TYPE",
453                s(&notes.or(
454                    processor.sub_type.clone(),
455                    "PROCESSOR SUB_TYPE",
456                    String::new,
457                )),
458            ),
459            (
460                "TYPE_ID",
461                i(notes.or(processor.type_id, "PROCESSOR TYPE_ID", || -1)),
462            ),
463            (
464                "MODE_ID",
465                i(notes.or(processor.mode_id, "PROCESSOR MODE_ID", || -1)),
466            ),
467            (
468                "FLAG_ROW",
469                b(notes.or(processor.flagged, "PROCESSOR FLAG_ROW", || false)),
470            ),
471        ];
472        let mut record = row_from_defs(schema::processor::REQUIRED_COLUMNS, &row);
473        if passes {
474            let pass = notes.or(processor.pass_id, "PROCESSOR PASS_ID", || -1);
475            record.push(casa_types::RecordField::new("PASS_ID", i(pass)));
476        }
477        table.add_row(record).map_err(storage)?;
478    }
479    Ok(())
480}

482/// The `HISTORY` rows of `history`, in order.
483pub(super) fn history(
484    ms: &mut MeasurementSet,
485    history: &[HistoryRow],
486) -> Result<(), ObservationError> {
487    let table = subtable_mut(ms, SubtableId::History).map_err(storage)?;
488    for entry in history {
489        let parameters: Vec<&str> = entry.parameters.iter().map(String::as_str).collect();
490        let command: Vec<&str> = entry.command.iter().map(String::as_str).collect();
491        let row = row_from_defs(
492            schema::history::REQUIRED_COLUMNS,
493            &[
494                ("APPLICATION", s(&entry.application)),
495                (
496                    "APP_PARAMS",
497                    string_array(&parameters, vec![parameters.len()]),
498                ),
499                ("CLI_COMMAND", string_array(&command, vec![command.len()])),
500                ("MESSAGE", s(&entry.message)),
501                ("OBJECT_ID", i(0)),
502                ("OBSERVATION_ID", i(entry.observation)),
503                ("ORIGIN", s(&entry.origin)),
504                ("PRIORITY", s(&entry.priority)),
505                ("TIME", f(entry.time_s)),
506            ],
507        );
508        table.add_row(row).map_err(storage)?;
509    }
510    Ok(())
