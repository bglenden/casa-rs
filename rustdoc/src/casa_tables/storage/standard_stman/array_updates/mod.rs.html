18/// Append replacement records and patch their bucket offsets. SSMBase uses
19/// version zero array files (no reference counts). Appending, rather than
20/// overwriting an old record, also preserves any other cells sharing it.
21/// Unsupported layouts return false before changing any bytes.
22pub(crate) fn save_ssm_array_rows_in_place(
23    path: &Path,
24    dm_blob: &[u8],
25    descs: &[ColumnDescContents],
26    columns: ArrayColumns,
27    big_endian: bool,
28) -> Result<bool, StorageError> {
29    if columns.len() != descs.len() {
30        return Err(StorageError::FormatMismatch(
31            "SSM array column count mismatch".into(),
32        ));
33    }
34    if columns.iter().all(Option::is_none) {
35        return Ok(false);
36    }
37    let mut has_indirect = false;
38    for (desc, values) in descs.iter().zip(&columns) {
39        if let Some(values) = values {
40            let indirect = is_ssm_array_file_indirect(desc) && !desc.is_record();
41            let direct = !is_ssm_indirect(desc)
42                && desc.is_array
43                && desc.data_type == CasacoreDataType::TpComplex;
44            if !indirect && !direct {
45                return Ok(false);
46            }
47            has_indirect |= indirect;
48            for (_, value) in values {
49                if let Some(value) = value
50                    && Some(value.primitive_type()) != desc.primitive_type
51                {
52                    return Err(StorageError::FormatMismatch(format!(
53                        "SSM array replacement type mismatch for {}",
54                        desc.col_name
55                    )));
56                }
57            }
58        }
59    }
60    let mut array_path = path.as_os_str().to_os_string();
61    array_path.push("i");
62    let array_path = std::path::PathBuf::from(array_path);
63    if has_indirect && array_path.exists() {
64        let mut version = [0; 4];
65        File::open(&array_path)?.read_exact(&mut version)?;
66        // Zero is the same in either byte order.
67        if version != [0; 4] {
68            return Ok(false);
69        }
70    }
71    let mut file = OpenOptions::new().read(true).write(true).open(path)?;
72    let header = parse_ssm_header(&mut file)?;
73    if header.big_endian != big_endian {
74        return Err(StorageError::FormatMismatch(
75            "SSM array save endian mismatch".into(),
76        ));
77    }
78    let dm = parse_ssm_dm_blob(dm_blob)?;
79    let indices = parse_ssm_indices(&mut file, &header)?;
80    // Complete layout validation before appending or modifying any bucket.
81    let mut buckets: BTreeMap<u32, Vec<ArrayMutation>> = BTreeMap::new();
82    for (column, values) in columns.into_iter().enumerate() {
83        let Some(values) = values else { continue };
84        let (Some(&offset), Some(&index)) =
85            (dm.column_offsets.get(column), dm.col_index_map.get(column))
86        else {
87            return Err(StorageError::FormatMismatch(
88                "SSM missing array column metadata".into(),
89            ));
90        };
91        let index = indices
92            .get(index as usize)
93            .ok_or_else(|| StorageError::FormatMismatch("SSM missing array column index".into()))?;
94        let desc = &descs[column];
95        let data_type =
96            CasacoreDataType::from_primitive_type(desc.require_primitive_type()?, false);
97        let direct_elements = (!is_ssm_indirect(desc)).then(|| ssm_cell_element_count(desc));
98        let width = direct_elements.map_or(8, |elements| {
99            elements.saturating_mul(canonical_element_size(data_type).0)
100        });
101        for (row, value) in values {
102            let (bucket, start, _) = index.find_bucket(row as u64).ok_or_else(|| {
103                StorageError::FormatMismatch(format!("SSM missing array row {row}"))
104            })?;
105            let at = (offset as usize)
106                .saturating_add(((row as u64 - start) as usize).saturating_mul(width));
107            if at
108                .checked_add(width)
109                .is_none_or(|end| end > header.bucket_size as usize)
110            {
111                return Err(StorageError::FormatMismatch(
112                    "SSM array offset outside bucket".into(),
113                ));
114            }
115            buckets.entry(bucket).or_default().push(ArrayMutation {
116                offset: at,
117                data_type,
118                direct_elements,
119                value: value.map(Value::Array),
120            });
121        }
122    }
123    let file_length = file.metadata()?.len();
124    for &number in buckets.keys() {
125        let end = SSM_HEADER_SIZE + (u64::from(number) + 1) * u64::from(header.bucket_size);
126        if end > file_length {
127            return Err(StorageError::FormatMismatch(format!(
128                "SSM array bucket {number} is outside the data file"
129            )));
130        }
131    }
132    let mut writer = has_indirect
133        .then(|| {
134            if array_path.exists() {
135                StManArrayFileWriter::open_append(&array_path, big_endian)
136            } else {
137                StManArrayFileWriter::create(&array_path, big_endian, 0)
138            }
139        })
140        .transpose()?;
141    write_buckets(&mut file, &header, buckets, writer.as_mut())?;
142    if let Some(mut writer) = writer {
143        writer.finish()?;
144    }
145    Ok(true)
146}

148fn write_buckets(
149    file: &mut File,
150    header: &SsmHeader,
151    buckets: BTreeMap<u32, Vec<ArrayMutation>>,
152    mut writer: Option<&mut StManArrayFileWriter>,
153) -> Result<(), StorageError> {
154    // Patch adjacent touched buckets together, without rewriting a bucket
155    // outside the selection. The buffer never grows beyond 64 KiB or one
156    // stored bucket, whichever is larger.
157    let bucket_size = header.bucket_size as usize;
158    let maximum = (64 * 1024 / bucket_size.max(1)).max(1);
159    let mut pending = buckets.into_iter().peekable();
160    let mut batch = Vec::new();
161    let mut buffer = Vec::new();
162    while let Some((first, mutations)) = pending.next() {
163        batch.push((first, mutations));
164        while batch.len() < maximum
165            && pending
166                .peek()
167                .is_some_and(|(number, _)| first.checked_add(batch.len() as u32) == Some(*number))
168        {
169            batch.push(pending.next().expect("peeked adjacent bucket"));
170        }
171        let bytes = batch
172            .len()
173            .checked_mul(bucket_size)
174            .ok_or_else(|| StorageError::FormatMismatch("SSM bucket batch size overflow".into()))?;
175        buffer
176            .try_reserve_exact(bytes.saturating_sub(buffer.len()))
177            .map_err(|error| {
178                StorageError::FormatMismatch(format!("cannot allocate SSM bucket batch: {error}"))
179            })?;
180        buffer.resize(bytes, 0);
181        let start = SSM_HEADER_SIZE + u64::from(first) * u64::from(header.bucket_size);
182        file.seek(SeekFrom::Start(start))?;
183        file.read_exact(&mut buffer)?;
184        for (slot, (_, mutations)) in batch.drain(..).enumerate() {
185            let bucket = &mut buffer[slot * bucket_size..(slot + 1) * bucket_size];
186            for mutation in mutations {
187                let destination = &mut bucket[mutation.offset..];
188                if let Some(elements) = mutation.direct_elements {
189                    encode_fixed_cell(
190                        destination,
191                        mutation.value.as_ref().into(),
192                        mutation.data_type,
193                        elements,
194                        header.big_endian,
195                    );
196                    continue;
197                }
198                let offset = match mutation.value {
199                    Some(value) => writer
200                        .as_mut()
201                        .expect("indirect arrays have a writer")
202                        .write_array(&value, mutation.data_type)?,
203                    None => 0,
204                };
205                if header.big_endian {
206                    write_i64_be(destination, offset);
207                } else {
208                    write_i64_le(destination, offset);
209                }
210            }
211        }
212        file.seek(SeekFrom::Start(start))?;
213        file.write_all(&buffer)?;
214    }
215    Ok(())
216}
