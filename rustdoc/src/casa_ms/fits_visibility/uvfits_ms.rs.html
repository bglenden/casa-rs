23pub(super) fn write_metadata(
24    ms: &mut MeasurementSet,
25    header: &Header,
26    layout: &GroupLayout,
27    extensions: &Extensions,
28    usage: &Usage,
29    options: UvFitsImportOptions,
30) -> MsResult<()> {
31    let time = fits_date(&header.date)?;
32    let frame = direction_frame(header.epoch, &header.telescope);
33    let epoch = match extensions.array.time_system.as_str() {
34        "TAI" | "IAT" => "TAI",
35        _ => "UTC",
36    };
37    configure_references(ms, frame, epoch, true)?;
38    write_observation(ms, header, time)?;
39    write_antennas(
40        ms,
41        &extensions.array,
42        &header.telescope,
43        options.new_antenna_names,
44    )?;
45    write_polarization(ms, &layout.corr)?;
46    write_feeds(ms, layout, &extensions.array, time)?;
47    write_spectral_windows(ms, header, layout, extensions, usage.windows)?;
48    let directions = SourceDirections::new(header, extensions, frame, epoch, time)?;
49    write_sources(ms, header, layout, extensions, usage, &directions)?;
50    Ok(())
51}

59#[doc(hidden)]
60pub fn fits_date(value: &str) -> MsResult<f64> {
61    let value = value.trim();
62    if value.is_empty() {
63        return Ok(0.0);
64    }
65    let formatted = if value.as_bytes().get(4) == Some(&b'-') {
66        value
67            .replace(['-', 'T'], "/")
68            .trim_end_matches('Z')
69            .to_owned()
70    } else if let [day, month, year] = value.split('/').collect::<Vec<_>>().as_slice()
71        && year.len() == 2
72        && let Ok(year) = year.parse::<i32>()
73    {
74        format!("{}/{month}/{day}", 1900 + year)
75    } else {
76        value.to_owned()
77    };
78    let days = Unit::new("d").map_err(invalid)?;
79    Ok(Quantity::read(&formatted)
80        .map_err(invalid)?
81        .get_value_in(&days)
82        .map_err(invalid)?
83        * 86_400.0)
84}

86fn direction_frame(epoch: f64, telescope: &str) -> DirectionRef {
87    if (epoch - 1950.0).abs() < 0.01 {
88        if telescope == "VLA" {
89            DirectionRef::B1950Vla
90        } else {
91            DirectionRef::B1950
92        }
93    } else {
94        DirectionRef::J2000
95    }
96}

98#[doc(hidden)]
99pub fn configure_references(
100    ms: &mut MeasurementSet,
101    frame: DirectionRef,
102    epoch: &str,
103    uvw_uses_phase_frame: bool,
104) -> MsResult<()> {
105    for column in ["TIME", "TIME_CENTROID"] {
106        TableMeasDesc::new_fixed(column, MeasureType::Epoch, epoch).write(ms.main_table_mut())?;
107    }
108    // The native schema installs MEASINFO for UVW. Keep its type and units.
109    if uvw_uses_phase_frame
110        && let Some(mut keywords) = ms.main_table().column_keywords("UVW").cloned()
111        && let Some(Value::Record(mut record)) = keywords.get("MEASINFO").cloned()
112    {
113        record.upsert("Ref", s(&frame.to_string()));
114        keywords.upsert("MEASINFO", Value::Record(record));
115        ms.main_table_mut().set_column_keywords("UVW", keywords);
116    }
117    for id in [
118        SubtableId::Field,
119        SubtableId::Pointing,
120        SubtableId::Source,
121        SubtableId::Observation,
122        SubtableId::History,
123        SubtableId::Feed,
124        SubtableId::FlagCmd,
125        SubtableId::SysCal,
126        SubtableId::Weather,
127    ] {
128        let Some(table) = ms.subtable_mut(id) else {
129            continue;
130        };
131        for definition in schema::required_columns(id) {
132            let refer = match definition.measure_type {
133                Some(MeasureType::Epoch) => epoch.to_owned(),
134                Some(MeasureType::Direction) => frame.to_string(),
135                _ => continue,
136            };
137            TableMeasDesc::new_fixed(definition.name, definition.measure_type.unwrap(), &refer)
138                .write(table)?;
139        }
140    }
141    Ok(())
142}

144fn write_observation(ms: &mut MeasurementSet, header: &Header, time: f64) -> MsResult<()> {
145    let telescope = if header.telescope == "HATCREEK" {
146        "BIMA"
147    } else {
148        &header.telescope
149    };
150    let mut observation = row_from_defs(
151        schema::observation::REQUIRED_COLUMNS,
152        &[
153            ("TIME_RANGE", f64_array(&[time, time], vec![2])),
154            ("RELEASE_DATE", f(fits_date(&header.release_date)?)),
155            ("OBSERVER", s(&header.observer)),
156            ("TELESCOPE_NAME", s(telescope)),
157        ],
158    );
159    observation.remove("LOG");
160    observation.remove("SCHEDULE");
161    subtable_mut(ms, SubtableId::Observation)?.add_row(observation)?;
162    for entry in header.history.iter().take_while(|entry| !entry.is_empty()) {
163        insert(
164            ms,
165            SubtableId::History,
166            &[
167                ("OBSERVATION_ID", i(0)),
168                ("TIME", f(time)),
169                ("PRIORITY", s("NORMAL")),
170                ("ORIGIN", s("MSFitsInput::fillObsTables")),
171                ("APPLICATION", s("ms")),
172                ("CLI_COMMAND", string_array(&[""], vec![1])),
173                ("APP_PARAMS", string_array(&[""], vec![1])),
174                ("MESSAGE", s(entry)),
175            ],
176        )?;
177    }
178    Ok(())
179}

181fn write_antennas(
182    ms: &mut MeasurementSet,
183    array: &Array,
184    telescope: &str,
185    new_names: bool,
186) -> MsResult<()> {
187    let (origin, rotate) = observatory_origin(array)?;
188    let (sin, cos) = origin[1].atan2(origin[0]).sin_cos();
189    let count = array.antennas.last_key_value().map_or(0, |(&id, _)| id + 1);
190    for id in 0..count {
191        let Some(antenna) = array.antennas.get(&id) else {
192            insert(ms, SubtableId::Antenna, &[("FLAG_ROW", b(true))])?;
193            continue;
194        };
195        let name = match (new_names, telescope) {
196            (true, "CARMA") => format!("CA{}", antenna.number),
197            (true, "EVLA") => format!("EA{:02}", antenna.number),
198            (true, "VLA") => format!("VA{:02}", antenna.number),
199            _ => antenna.number.to_string(),
200        };
201        let mounts = [
202            "ALT-AZ",
203            "EQUATORIAL",
204            "ORBITING",
205            "X-Y",
206            "ALT-AZ+NASMYTH-R",
207            "ALT-AZ+NASMYTH-L",
208            "ALT-AZ+BWG-R",
209            "ALT-AZ+BWG-L",
210            "BIZARRE",
211        ];
212        let mount = mounts
213            .get(antenna.mount as usize)
214            .copied()
215            .unwrap_or("SPACE_HALCA");
216        let [mut x, mut y, z] = antenna.position;
217        if rotate {
218            (x, y) = (cos * x - sin * y, sin * x + cos * y);
219        }
220        if array.left_handed {
221            y = -y;
222        }
223        let position = [origin[0] + x, origin[1] + y, origin[2] + z];
224        insert(
225            ms,
226            SubtableId::Antenna,
227            &[
228                ("NAME", s(&name)),
229                ("STATION", s(&antenna.station)),
230                ("MOUNT", s(mount)),
231                ("TYPE", s("GROUND-BASED")),
232                (
233                    "OFFSET",
234                    f64_array(&[antenna.axis_offset, 0.0, 0.0], vec![3]),
235                ),
236                ("POSITION", f64_array(&position, vec![3])),
237                ("DISH_DIAMETER", f(antenna.diameter)),
238            ],
239        )?;
240    }
241    let table = subtable_mut(ms, SubtableId::Antenna)?;
242    let keywords = table.keywords_mut();
243    keywords.upsert("RDATE", f(fits_date(&array.date)?));
244    keywords.upsert("GSTIA0", f(array.gst));
245    keywords.upsert("DEGPDY", f(array.degrees_per_day));
246    let time_system = if array.time_system == "IAT" {
247        "TAI"
248    } else {
249        &array.time_system
250    };
251    keywords.upsert("TIMSYS", s(time_system));
252    column_keyword(table, "NAME", "ARRAY_NAME", s(&array.name));
253    column_keyword(
254        table,
255        "POSITION",
256        "ARRAY_POSITION",
257        f64_array(&origin, vec![3]),
258    );
259    Ok(())
260}

262fn observatory_origin(array: &Array) -> MsResult<([f64; 3], bool)> {
263    let norm = |vector: [f64; 3]| {
264        vector
265            .into_iter()
266            .map(|value| value * value)
267            .sum::<f64>()
268            .sqrt()
269    };
270    if array.name != "VLA" || (array.frame_defined && norm(array.origin) <= 1.0e6) {
271        return Ok((array.origin, false));
272    }
273    let provider = crate::open_measures_runtime()?;
274    let position = MPosition::casa_observatory("VLA", provider.as_ref())
275        .ok_or_else(|| invalid("Measures catalog has no VLA observatory"))?;
276    let center = position.convert_to(PositionRef::ITRF)?.values();
277    let difference = std::array::from_fn(|axis| array.origin[axis] - center[axis]);
278    Ok(if norm(difference) <= 10_000.0 {
279        (center, true)
280    } else {
281        (array.origin, false)
282    })
283}

292#[doc(hidden)]
293pub fn write_polarization(ms: &mut MeasurementSet, corr: &[Correlation]) -> MsResult<()> {
294    let codes = corr
295        .iter()
296        .map(|&correlation| correlation.casacore_code())
297        .collect::<Vec<_>>();
298    let mut receptors = Vec::with_capacity(codes.len() * 2);
299    for second in [false, true] {
300        receptors.extend(corr.iter().map(|correlation| {
301            i32::from(match correlation {
302                Correlation::LL | Correlation::YY => true,
303                Correlation::RL | Correlation::XY => second,
304                Correlation::LR | Correlation::YX => !second,
305                _ => false,
306            })
307        }));
308    }
309    insert(
310        ms,
311        SubtableId::Polarization,
312        &[
313            ("NUM_CORR", i(codes.len() as i32)),
314            ("CORR_TYPE", i32_array(&codes, vec![codes.len()])),
315            ("CORR_PRODUCT", i32_array(&receptors, vec![2, codes.len()])),
316        ],
317    )
318}

320fn write_feeds(
321    ms: &mut MeasurementSet,
322    layout: &GroupLayout,
323    array: &Array,
324    time: f64,
325) -> MsResult<()> {
326    let endpoints =
327        [layout.corr[0], layout.corr[layout.corr.len() - 1]].map(Correlation::casacore_code);
328    let polarization = if endpoints[0] >= 5 && endpoints[1] <= 8 {
329        ["R", "L"]
330    } else if endpoints[0] >= 9 && endpoints[1] <= 12 {
331        ["X", "Y"]
332    } else {
333        ["?", "?"]
334    };
335    let count = array.antennas.last_key_value().map_or(0, |(&id, _)| id + 1);
336    let mut angles = vec![[0.0; 2]; count as usize];
337    for (row, antenna) in array.antennas.values().enumerate() {
338        angles[row] = antenna.angles;
339    }
340    let response = [
341        Complex32::new(1.0, 0.0),
342        Complex32::ZERO,
343        Complex32::ZERO,
344        Complex32::new(1.0, 0.0),
345    ];
346    for (id, angles) in angles.into_iter().enumerate() {
347        insert(
348            ms,
349            SubtableId::Feed,
350            &[
351                ("ANTENNA_ID", i(id as i32)),
352                ("BEAM_ID", i(-1)),
353                ("SPECTRAL_WINDOW_ID", i(-1)),
354                ("TIME", f(time)),
355                ("NUM_RECEPTORS", i(2)),
356                ("BEAM_OFFSET", f64_array(&[0.0; 4], vec![2, 2])),
357                ("POLARIZATION_TYPE", string_array(&polarization, vec![2])),
358                ("POL_RESPONSE", complex_array(&response, vec![2, 2])),
359                ("RECEPTOR_ANGLE", f64_array(&angles, vec![2])),
360            ],
361        )?;
362    }
363    Ok(())
364}

366fn spectral_reference(header: &Header) -> FrequencyRef {
367    if let Some(reference) = header.frame.as_deref() {
368        use FrequencyRef::*;
369        return match reference.trim() {
370            "BARYCENT" => BARY,
371            "GEOCENTR" => GEO,
372            "SOURCE" => REST,
373            "GALACTOC" => GALACTO,
374            "LOCALGRP" => LGROUP,
375            "CMBDIPOL" => CMB,
376            "TOPOCENT" => TOPO,
377            name => name.parse().unwrap_or(TOPO),
378        };
379    }
380    match header.velocity_reference.unwrap_or(3).rem_euclid(256) {
381        1 => FrequencyRef::LSRK,
382        2 => FrequencyRef::BARY,
383        4 => FrequencyRef::LSRD,
384        5 => FrequencyRef::GEO,
385        6 => FrequencyRef::REST,
386        7 => FrequencyRef::GALACTO,
387        _ => FrequencyRef::TOPO,
388    }
389}

391fn source_velocity_frame(extensions: &Extensions) -> Option<FrequencyRef> {
392    let name = extensions.velocity_type.as_deref()?;
393    [
394        ("LSRD", FrequencyRef::LSRD),
395        ("LSR", FrequencyRef::LSRK),
396        ("REST", FrequencyRef::REST),
397        ("BARY", FrequencyRef::BARY),
398        ("GEO", FrequencyRef::GEO),
399        ("TOPO", FrequencyRef::TOPO),
400        ("GALAC", FrequencyRef::GALACTO),
401        ("LOCAL", FrequencyRef::LGROUP),
402        ("LGROUP", FrequencyRef::LGROUP),
403        ("CMB", FrequencyRef::CMB),
404    ]
405    .into_iter()
406    .find_map(|(token, reference)| name.contains(token).then_some(reference))
407}

409fn write_spectral_windows(
410    ms: &mut MeasurementSet,
411    header: &Header,
412    layout: &GroupLayout,
413    extensions: &Extensions,
414    windows: usize,
415) -> MsResult<()> {
416    let reference = spectral_reference(header);
417    let stored_reference = if header.frame.is_some() {
418        reference
419    } else {
420        source_velocity_frame(extensions).unwrap_or(reference)
421    };
422    let axis = &layout.frequency;
423    for id in 0..windows {
424        let mut window = extensions.spectral.as_ref().map_or(
425            SpectralWindow {
426                frequency: axis.reference,
427                increment: axis.increment,
428                bandwidth: (axis.increment * axis.length as f64).abs(),
429            },
430            |windows| windows[id],
431        );
432        let mut pixel = axis.pixel;
433        let mut channel_width = window.increment;
434        if let Some(velocity) = header.alternate_value
435            && header.frame.is_none()
436            && reference != FrequencyRef::TOPO
437            && header.rest_frequency > 0.0
438        {
439            let beta = velocity / 299_792_458.0;
440            window.frequency = header.rest_frequency
441                * if header.velocity_reference.unwrap_or(0) > 256 {
442                    1.0 - beta
443                } else {
444                    (1.0 + beta).recip()
445                };
446            pixel = header.alternate_pixel.unwrap_or(pixel);
447            channel_width = axis.increment;
448        }
449        let frequency = (0..axis.length)
450            .map(|channel| window.frequency + (channel as f64 + 1.0 - pixel) * channel_width)
451            .collect::<Vec<_>>();
452        let width = if extensions.spectral.is_some() {
453            window.increment.abs()
454        } else {
455            window.increment
456        };
457        let width = f64_array(&vec![width; axis.length], vec![axis.length]);
458        let band = i32::from(window.increment > 0.0) * 2 - 1;
459        insert(
460            ms,
461            SubtableId::SpectralWindow,
462            &[
463                ("NAME", s("none")),
464                ("FREQ_GROUP_NAME", s("none")),
465                ("IF_CONV_CHAIN", i((id % layout.windows) as i32)),
466                ("NUM_CHAN", i(axis.length as i32)),
467                ("REF_FREQUENCY", f(window.frequency)),
468                ("CHAN_FREQ", f64_array(&frequency, vec![axis.length])),
469                ("CHAN_WIDTH", width.clone()),
470                ("EFFECTIVE_BW", width.clone()),
471                ("RESOLUTION", width),
472                ("TOTAL_BANDWIDTH", f(window.bandwidth)),
473                ("NET_SIDEBAND", i(band)),
474                ("FREQ_GROUP", i((id / layout.windows) as i32)),
475                ("MEAS_FREQ_REF", i(stored_reference.casacore_code())),
476            ],
477        )?;
478        insert(
479            ms,
480            SubtableId::DataDescription,
481            &[("SPECTRAL_WINDOW_ID", i(id as i32))],
482        )?;
483    }
484    Ok(())
485}

487fn write_sources(
488    ms: &mut MeasurementSet,
489    header: &Header,
490    layout: &GroupLayout,
491    extensions: &Extensions,
492    usage: &Usage,
493    directions: &SourceDirections<'_>,
494) -> MsResult<()> {
495    let defaults = Source {
496        name: header.object.clone(),
497        code: " ".into(),
498        epoch: header.epoch,
499        direction_deg: layout.direction_deg,
500        apparent_deg: layout.direction_deg,
501        motion_per_day: [0.0; 2],
502        rest_frequency: 0.0,
503        velocity: 0.0,
504    };
505    let fields = extensions.sources.as_ref().map_or(1, |sources| {
506        sources.last_key_value().map_or(0, |(&id, _)| id + 1)
507    });
508    let source = |id| {
509        extensions
510            .sources
511            .as_ref()
512            .map_or(Some(&defaults), |sources| sources.get(&id))
513    };
514    for id in 0..fields {
515        let (name, code, direction, count, flag) = if let Some(source) = source(id) {
516            let coordinates = directions.convert(source)?;
517            let (direction, order) = coordinates.phase_direction();
518            (
519                source.name.as_str(),
520                source.code.as_str(),
521                direction,
522                order,
523                false,
524            )
525        } else {
526            ("", "", f64_array(&[0.0; 2], vec![2, 1]), 0, true)
527        };
528        let present = usage.occurrences.iter().any(|&(used, _, _)| used == id);
529        insert(
530            ms,
531            SubtableId::Field,
532            &[
533                ("NAME", s(name)),
534                ("CODE", s(code)),
535                ("TIME", f(directions.time)),
536                ("NUM_POLY", i(count)),
537                ("SOURCE_ID", i(if present { id } else { -1 })),
538                ("FLAG_ROW", b(flag)),
539                ("PHASE_DIR", direction.clone()),
540                ("DELAY_DIR", direction.clone()),
541                ("REFERENCE_DIR", direction),
542            ],
543        )?;
544    }
545    let names = ["REST_FREQUENCY", "SYSVEL", "TRANSITION"];
546    let defs = schema::source::OPTIONAL_COLUMNS
547        .iter()
548        .filter(|definition| names.contains(&definition.name))
549        .copied()
550        .collect::<Vec<_>>();
551    let schema = crate::column_def::build_table_schema(&defs)?;
552    let table = subtable_mut(ms, SubtableId::Source)?;
553    for column in schema.columns() {
554        table.add_column(column.clone(), None)?;
555    }
556    table.add_column(
557        ColumnSchema::array_variable("POSITION", PrimitiveType::Float64, Some(1)),
558        None,
559    )?;
560    crate::ms::apply_column_metadata(table, schema::source::OPTIONAL_COLUMNS);
561    TableMeasDesc::new_fixed("REST_FREQUENCY", MeasureType::Frequency, "LSRK").write(table)?;
562    table.add_column(ColumnSchema::record("SOURCE_MODEL"), None)?;
563    for &(id, window, start) in &usage.occurrences {
564        let source = source(id).ok_or_else(|| invalid("missing observed UVFITS source"))?;
565        let coordinates = directions.convert(source)?;
566        let mut row = row_from_defs(
567            schema::source::REQUIRED_COLUMNS,
568            &[
569                ("SOURCE_ID", i(id)),
570                ("SPECTRAL_WINDOW_ID", i(window)),
571                ("TIME", f(start)),
572                ("INTERVAL", f(f64::MAX)),
573                ("NAME", s(&source.name)),
574                ("CALIBRATION_GROUP", i(-1)),
575                ("NUM_LINES", i(1)),
576                ("DIRECTION", f64_array(&coordinates.angle, vec![2])),
577                ("PROPER_MOTION", f64_array(&coordinates.motion, vec![2])),
578            ],
579        );
580        row.upsert(
581            "REST_FREQUENCY",
582            f64_array(&[source.rest_frequency], vec![1]),
583        );
584        row.upsert("SYSVEL", f64_array(&[source.velocity], vec![1]));
585        row.upsert("TRANSITION", string_array(&[""], vec![1]));
586        table.add_row(row)?;
587    }
588    Ok(())
589}

591struct SourceDirections<'a> {
592    telescope: &'a str,
593    target: DirectionRef,
594    apparent_frame: MeasFrame,
595    time: f64,
596}
597
598impl<'a> SourceDirections<'a> {
599    fn new(
600        header: &'a Header,
601        extensions: &Extensions,
602        target: DirectionRef,
603        epoch: &str,
604        time: f64,
605    ) -> MsResult<Self> {
606        let has_motion = extensions.sources.as_ref().is_some_and(|sources| {
607            sources
608                .values()
609                .any(|source| source.motion_per_day != [0.0; 2])
610        });
611        let mut apparent_frame = MeasFrame::new();
612        if has_motion {
613            let mjd = time / 86_400.0;
614            let reference = epoch.parse::<EpochRef>().map_err(invalid)?;
615            apparent_frame =
616                apparent_frame.with_epoch(MEpoch::new(MjdHighPrec::from_mjd(mjd), reference));
617            if let Some(provider) = super::export_metadata::fits_measures() {
618                apparent_frame = apparent_frame.with_measures(provider);
619            }
620            if apparent_frame.dut1_for_mjd(mjd)?.is_none() {
621                apparent_frame = apparent_frame.with_dut1(0.0);
622            }
623            let [x, y, z] = extensions.array.origin;
624            if [x, y, z] != [0.0; 3] {
625                apparent_frame = apparent_frame.with_position(MPosition::new_itrf(x, y, z));
626            }
627        }
628        Ok(Self {
629            telescope: &header.telescope,
630            target,
631            apparent_frame,
632            time,
633        })
634    }
635
636    fn convert(&self, source: &Source) -> MsResult<SourceCoordinates> {
637        let moving = source.motion_per_day != [0.0; 2];
638        let (input, direction) = if moving {
639            (DirectionRef::APP, source.apparent_deg)
640        } else if source.epoch == -1.0 {
641            (self.target, source.direction_deg)
642        } else if (source.epoch - 1950.0).abs() <= 0.01 || (source.epoch - 2000.0).abs() <= 20.0 {
643            (
644                direction_frame(source.epoch, self.telescope),
645                source.direction_deg,
646            )
647        } else {
648            return Err(invalid(format!(
649                "Cannot handle epoch in UVFITS source table: {}",
650                source.epoch
651            )));
652        };
653        let input_angles = direction.map(f64::to_radians);
654        let convert = |[ra, dec]: [f64; 2]| -> MsResult<[f64; 2]> {
655            let position = MDirection::from_angles(ra, dec, input)
656                .convert_to(self.target, &self.apparent_frame)
657                .map_err(invalid)?;
658            let (ra, dec) = position.as_angles();
659            Ok([ra.sin().atan2(ra.cos()), dec])
660        };
661        let angle = convert(input_angles)?;
662        let motion = if !moving {
663            [0.0; 2]
664        } else {
665            // Differentiate the direction transform at the source position.
666            // The epoch stays fixed while the input direction changes.
667            let delta = source.motion_per_day.map(|rate| rate.to_radians() / 16.0);
668            let ahead = convert(std::array::from_fn(|axis| input_angles[axis] + delta[axis]))?;
669            let behind = convert(std::array::from_fn(|axis| input_angles[axis] - delta[axis]))?;
670            let ra = ahead[0] - behind[0];
671            [
672                ra.sin().atan2(ra.cos()) / 10_800.0,
673                (ahead[1] - behind[1]) / 10_800.0,
674            ]
675        };
676        Ok(SourceCoordinates {
677            angle,
678            motion,
679            moving,
680        })
681    }
682}
683
684struct SourceCoordinates {
685    angle: [f64; 2],
686    motion: [f64; 2],
687    moving: bool,
688}
689
690impl SourceCoordinates {
691    fn phase_direction(&self) -> (Value, i32) {
692        if self.moving {
693            (
694                f64_array(
695                    &[self.angle[0], self.motion[0], self.angle[1], self.motion[1]],
696                    vec![2, 2],
697                ),
698                1,
699            )
700        } else {
701            (f64_array(&self.angle, vec![2, 1]), 0)
702        }
703    }
704}
