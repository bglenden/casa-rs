31/// Storage boundary used by the runtime publication owner.
32///
33/// `stage` writes only private, non-visible state. `promote` independently and
34/// atomically replaces one conventional product. A later member failure does
35/// not invalidate members already promoted from the same sealed generation.
36pub trait SerialProductPublicationSink {
37    /// Sink-specific failure.
38    type Error: Error + 'static;
39
40    /// Declare the peak sink-owned heap used while staging one member.
41    ///
42    /// This excludes the sealed member borrowed from the product owner and the
43    /// runtime-owned publication bookkeeping. It includes every payload,
44    /// validity, image-adapter, and registry allocation made by [`Self::stage`].
45    fn staging_residency_bytes(
46        &self,
47        planned: &PlannedContinuumGeneration,
48        demand: &ContinuumGenerationDemand,
49    ) -> Result<u64, Self::Error>;
50
51    /// Privately stage one exact sealed member.
52    fn stage(
53        &self,
54        planned: ArtifactIdentity,
55        observed: ArtifactIdentity,
56        member: &SealedMember,
57    ) -> Result<(), Self::Error>;
58
59    /// Atomically activate one exact runtime-authorized member.
60    ///
61    /// Repeating the same planned/observed identity is idempotent and succeeds
62    /// without changing visible content.
63    fn promote(
64        &self,
65        entry: AuthorizedProductPublicationEntry,
66    ) -> Result<(), MemberPromotionFailure<Self::Error>>;
67}

69/// Certainty of a failed per-member atomic replacement.
70#[derive(Debug, Clone, Copy, PartialEq, Eq)]
71pub enum MemberPromotionFailureKind {
72    /// The sink proved that the prior member remains visible.
73    Failed,
74    /// The sink could not prove which member is visible.
75    Uncertain,
76}
77
78/// Sink error retaining whether one member's visibility is known.
79#[derive(Debug)]
80pub struct MemberPromotionFailure<E> {
81    kind: MemberPromotionFailureKind,
82    source: E,
83}
84
85impl<E> MemberPromotionFailure<E> {
86    /// Report a failure known to leave the prior member visible.
87    #[must_use]
88    pub const fn failed(source: E) -> Self {
89        Self {
90            kind: MemberPromotionFailureKind::Failed,
91            source,
92        }
93    }
94
95    /// Report a failure whose visibility outcome could not be proved.
96    #[must_use]
97    pub const fn uncertain(source: E) -> Self {
98        Self {
99            kind: MemberPromotionFailureKind::Uncertain,
100            source,
101        }
102    }
103
104    /// Return the visibility certainty.
105    #[must_use]
106    pub const fn kind(&self) -> MemberPromotionFailureKind {
107        self.kind
108    }
109
110    /// Borrow the sink-specific failure.
111    #[must_use]
112    pub const fn source(&self) -> &E {
113        &self.source
114    }
115}

146/// One ordinary runtime physical plan for a preplanned continuum generation.
147pub struct SerialProductPublicationPlan {
148    physical: PhysicalWorkBinding,
149    publication: ProductPublicationPlan,
150}

152impl SerialProductPublicationPlan {
153    /// Build the exact product DAG before member production and sealing.
154    pub fn new<R: ImplementationRegistry>(
155        problem: &CompiledProblem,
156        planned: &PlannedContinuumGeneration,
157        generation_demand: &ContinuumGenerationDemand,
158        staging_residency_bytes: u64,
159        registry: &R,
160        policy: SerialProductPublicationPolicy,
161    ) -> Result<Self, SerialProductPublicationPlanError> {
162        let publication = ProductPublicationPlan::bind(problem, planned)?;
163        let physical = build_physical(
164            registry,
165            &policy,
166            &publication,
167            generation_demand,
168            staging_residency_bytes,
169        )?;
170        Ok(Self {
171            physical,
172            publication,
173        })
174    }
175
176    /// Return the ordinary physical candidate.
177    #[must_use]
178    pub const fn physical_work(&self) -> &PhysicalWorkBinding {
179        &self.physical
180    }
181
182    /// Return the immutable pre-seal publication authority.
183    #[must_use]
184    pub const fn publication(&self) -> &ProductPublicationPlan {
185        &self.publication
186    }
187
188    /// Consume into ordinary planning and execution construction parts.
189    #[must_use]
190    pub fn into_parts(self) -> (PhysicalWorkBinding, ProductPublicationPlan) {
191        (self.physical, self.publication)
192    }
193}

195fn build_physical<R: ImplementationRegistry>(
196    registry: &R,
197    policy: &SerialProductPublicationPolicy,
198    publication: &ProductPublicationPlan,
199    generation_demand: &ContinuumGenerationDemand,
200    staging_residency_bytes: u64,
201) -> Result<PhysicalWorkBinding, SerialProductPublicationPlanError> {
202    let check = WorkNodeId::new(CHECK);
203    let generate = WorkNodeId::new(GENERATE);
204    let seal = WorkNodeId::new(SEAL);
205    let stage = WorkNodeId::new(STAGE);
206    let commit = WorkNodeId::new(COMMIT);
207    let scratch_allocation = AllocationId::new("product-generation-scratch");
208    let produced_allocation = AllocationId::new("product-generation-produced");
209    let sealed_allocation = AllocationId::new("product-generation-sealed");
210    let writer_allocation = AllocationId::new("product-publication-writer-buffer");
211    let first_residency_slot = PhysicalSlotId::new("product-generation-residency-slot-a");
212    let second_residency_slot = PhysicalSlotId::new("product-generation-residency-slot-b");
213    let commit_allocation = AllocationId::new("product-publication-commit-buffer");
214    let commit_slot = PhysicalSlotId::new("product-publication-commit-slot");
215    let payload_bytes = publication
216        .entries()
217        .iter()
218        .try_fold(0_u64, |total, entry| {
219            total
220                .checked_add(entry.payload_bytes())
221                .ok_or(SerialProductPublicationPlanError::Overflow)
222        })?;
223    let scratch_bytes = generation_demand.algorithm_scratch_bytes();
224    let produced_bytes = generation_demand.produced_residency_bytes();
225    let sealed_bytes = generation_demand.sealed_residency_bytes();
226    let writer_bytes = staging_residency_bytes.max(payload_bytes).max(1);
227    let first_slot_bytes = scratch_bytes.max(sealed_bytes).max(1);
228    let second_slot_bytes = produced_bytes.max(writer_bytes).max(1);
229    let publication_lifetime =
230        ClaimLifetime::through_fences([FenceKind::Io, FenceKind::Publication]);
231    let output_rate_demand = "product-publication-output-write-rate".to_string();
232    let output_queue_demand = "product-publication-output-queue".to_string();
233    let storage_demand = "product-publication-output".to_string();
234    let check_claims = vec![claim(LeaseResource::Workers, 1, ClaimLifetime::Work)];
235    let commit_claims = vec![
236        claim(
237            LeaseResource::FileDescriptors,
238            policy.output_file_descriptor_bound,
239            publication_lifetime.clone(),
240        ),
241        claim(
242            LeaseResource::Rate {
243                demand_id: output_rate_demand.clone(),
244            },
245            1,
246            publication_lifetime.clone(),
247        ),
248        claim(
249            LeaseResource::Queue {
250                demand_id: output_queue_demand.clone(),
251            },
252            1,
253            publication_lifetime.clone(),
254        ),
255        claim(
256            LeaseResource::Storage {
257                demand_id: storage_demand.clone(),
258                use_kind: StorageUseKind::StagedOutput,
259            },
260            payload_bytes,
261            publication_lifetime.clone(),
262        ),
263        claim(
264            LeaseResource::Storage {
265                demand_id: storage_demand.clone(),
266                use_kind: StorageUseKind::FinalOutput,
267            },
268            payload_bytes,
269            publication_lifetime.clone(),
270        ),
271        claim(
272            LeaseResource::IoBuffer(IoBufferKind::Publication),
273            1,
274            publication_lifetime.clone(),
275        ),
276    ];
277    let nodes = vec![
278        WorkNode {
279            id: check.clone(),
280            kind: WorkKind::DataCensus,
281            domain: WorkDomain::Cpu,
282            implementation: policy.implementation.clone(),
283            dependencies: BTreeSet::new(),
284            claims: check_claims,
285            allocations: vec![],
286            fences: BTreeSet::new(),
287            quiescence_after: BTreeSet::new(),
288        },
289        WorkNode {
290            id: generate.clone(),
291            kind: WorkKind::Compute,
292            domain: WorkDomain::Cpu,
293            implementation: policy.implementation.clone(),
294            dependencies: BTreeSet::from([WorkDependency::Work(check.clone())]),
295            claims: vec![claim(LeaseResource::Workers, 1, ClaimLifetime::Work)],
296            allocations: vec![
297                AllocationUse {
298                    allocation: scratch_allocation.clone(),
299                    lifetime: ClaimLifetime::Work,
300                },
301                AllocationUse {
302                    allocation: produced_allocation.clone(),
303                    lifetime: ClaimLifetime::Work,
304                },
305            ],
306            fences: BTreeSet::new(),
307            quiescence_after: BTreeSet::new(),
308        },
309        WorkNode {
310            id: seal.clone(),
311            kind: WorkKind::Compute,
312            domain: WorkDomain::Cpu,
313            implementation: policy.implementation.clone(),
314            dependencies: BTreeSet::from([WorkDependency::Work(generate.clone())]),
315            claims: vec![claim(LeaseResource::Workers, 1, ClaimLifetime::Work)],
316            allocations: vec![
317                AllocationUse {
318                    allocation: produced_allocation.clone(),
319                    lifetime: ClaimLifetime::Work,
320                },
321                AllocationUse {
322                    allocation: sealed_allocation.clone(),
323                    lifetime: ClaimLifetime::Work,
324                },
325            ],
326            fences: BTreeSet::new(),
327            quiescence_after: BTreeSet::new(),
328        },
329        WorkNode {
330            id: stage.clone(),
331            kind: WorkKind::Serialization,
332            domain: WorkDomain::Cpu,
333            implementation: policy.implementation.clone(),
334            dependencies: BTreeSet::from([WorkDependency::Work(seal.clone())]),
335            claims: vec![
336                claim(LeaseResource::Workers, 1, ClaimLifetime::Work),
337                claim(
338                    LeaseResource::FileDescriptors,
339                    policy.output_file_descriptor_bound,
340                    ClaimLifetime::Work,
341                ),
342                claim(
343                    LeaseResource::Storage {
344                        demand_id: storage_demand.clone(),
345                        use_kind: StorageUseKind::StagedOutput,
346                    },
347                    payload_bytes,
348                    ClaimLifetime::Work,
349                ),
350                claim(
351                    LeaseResource::IoBuffer(IoBufferKind::Serialization),
352                    writer_bytes,
353                    ClaimLifetime::Work,
354                ),
355            ],
356            allocations: vec![
357                AllocationUse {
358                    allocation: writer_allocation.clone(),
359                    lifetime: ClaimLifetime::Work,
360                },
361                AllocationUse {
362                    allocation: sealed_allocation.clone(),
363                    lifetime: ClaimLifetime::Work,
364                },
365            ],
366            fences: BTreeSet::new(),
367            quiescence_after: BTreeSet::new(),
368        },
369        WorkNode {
370            id: commit.clone(),
371            kind: WorkKind::Publication,
372            domain: WorkDomain::Io,
373            implementation: policy.implementation.clone(),
374            dependencies: BTreeSet::from([WorkDependency::Work(stage.clone())]),
375            claims: commit_claims,
376            allocations: vec![AllocationUse {
377                allocation: commit_allocation.clone(),
378                lifetime: publication_lifetime.clone(),
379            }],
380            fences: BTreeSet::from([FenceKind::Io, FenceKind::Publication]),
381            quiescence_after: BTreeSet::new(),
382        },
383    ];
384    let compatibility = |layout: &str| SlotCompatibility {
385        memory_domain: CapacityDomainId::new("host-memory"),
386        views: BTreeSet::from([CapacityViewId::new("host-memory")]),
387        alignment_bytes: 64,
388        storage_mode: StorageMode::Host,
389        layout: AllocationLayout::new(layout),
390        initialization: InitializationPolicy::OverwriteBeforeRead,
391        access: AllocationAccess::ReadWrite,
392    };
393    let residency_compat = compatibility("product-owner-residency");
394    let commit_compat = compatibility("product-publication-commit");
395    let allocations = vec![
396        allocation(
397            scratch_allocation,
398            scratch_bytes.max(1),
399            AllocationPurpose::Data,
400            residency_compat.clone(),
401            first_residency_slot.clone(),
402            generate.clone(),
403            WorkDependency::Work(generate.clone()),
404        ),
405        allocation(
406            produced_allocation,
407            produced_bytes.max(1),
408            AllocationPurpose::Data,
409            residency_compat.clone(),
410            second_residency_slot.clone(),
411            generate.clone(),
412            WorkDependency::Work(seal.clone()),
413        ),
414        allocation(
415            sealed_allocation,
416            sealed_bytes.max(1),
417            AllocationPurpose::Data,
418            residency_compat.clone(),
419            first_residency_slot.clone(),
420            seal.clone(),
421            WorkDependency::Work(stage.clone()),
422        ),
423        allocation(
424            writer_allocation.clone(),
425            writer_bytes,
426            AllocationPurpose::IoBuffer(IoBufferKind::Serialization),
427            residency_compat.clone(),
428            second_residency_slot.clone(),
429            stage.clone(),
430            WorkDependency::Work(stage.clone()),
431        ),
432        LogicalAllocation {
433            id: commit_allocation.clone(),
434            bytes: 1,
435            purpose: AllocationPurpose::IoBuffer(IoBufferKind::Publication),
436            compatibility: commit_compat.clone(),
437            physical_slot: commit_slot.clone(),
438            lifetime: AllocationLifetime {
439                acquire_at: commit.clone(),
440                release_after: BTreeSet::from([
441                    WorkDependency::Fence(FenceId::new(commit.clone(), FenceKind::Io)),
442                    WorkDependency::Fence(FenceId::new(commit.clone(), FenceKind::Publication)),
443                ]),
444            },
445        },
446    ];
447    let slots = vec![
448        slot(
449            first_residency_slot,
450            LeaseResource::Memory {
451                allocation_id: "product-generation-residency-a".to_string(),
452            },
453            first_slot_bytes,
454            residency_compat.clone(),
455        ),
456        slot(
457            second_residency_slot,
458            LeaseResource::Memory {
459                allocation_id: "product-generation-residency-b".to_string(),
460            },
461            second_slot_bytes,
462            residency_compat,
463        ),
464        slot(
465            commit_slot,
466            LeaseResource::Memory {
467                allocation_id: "product-publication-commit".to_string(),
468            },
469            1,
470            commit_compat,
471        ),
472    ];
473    let alternative = DemandAlternative {
474        id: AlternativeId::new("serial-product-publication"),
475        capabilities: CapabilityPredicate::default(),
476        demand: DemandEnvelope {
477            host_memory_view: CapacityViewId::new("host-memory"),
478            memory: vec![
479                memory("product-generation-residency-a", first_slot_bytes),
480                memory("product-generation-residency-b", second_slot_bytes),
481                memory("product-publication-commit", 1),
482            ],
483            workers: CountDemand::new(1, 1),
484            overhead: RuntimeOverheadDemand::zero(),
485            storage: vec![StorageDemand {
486                demand_id: storage_demand,
487                domain: policy.storage_io.domain().clone(),
488                temporary_bytes: 0,
489                staged_output_bytes: payload_bytes,
490                final_output_bytes: payload_bytes,
491                persistent_cache_bytes: 0,
492                read_rate: CountDemand::zero(),
493                write_rate: CountDemand::zero(),
494                operations_rate: CountDemand::zero(),
495                queue_slots: CountDemand::new(1, 1),
496            }],
497            rates: vec![RateDemand {
498                demand_id: output_rate_demand,
499                resource: policy.storage_io.write_rate().clone(),
500                amount: CountDemand::new(1, 1),
501            }],
502            caches: CacheDemand::zero(),
503            locks: CountDemand::zero(),
504            // The serial sink stages or promotes one output member at a time.
505            // This is a declared capacity bound, not an exact OS-FD count.
506            file_descriptors: CountDemand::new(
507                policy.output_file_descriptor_bound,
508                policy.output_file_descriptor_bound,
509            ),
510            queues: vec![QueueDemand {
511                demand_id: output_queue_demand,
512                resource: policy.storage_io.queue().clone(),
513                slots: CountDemand::new(1, 1),
514            }],
515            transfers: vec![],
516            accelerators: vec![],
517            io_buffers: IoBufferDemand {
518                serialization_bytes: writer_bytes,
519                publication_bytes: 1,
520                ..IoBufferDemand::zero()
521            },
522        },
523        headroom: ResourceHeadroom::default(),
524        scaling: ScalingMetadata {
525            minimum_workers: 1,
526            maximum_workers: 1,
527            maximum_batch_size: 1,
528            maximum_tile_width: 1,
529            maximum_tile_height: 1,
530            maximum_slab_depth: 1,
531            memory_bytes_per_worker: BTreeMap::new(),
532        },
533        quiescence_points: BTreeSet::from([QuiescencePoint::RunBoundary]),
534    };
535    let dag = ExecutionDag::new(ExecutionDagSpecification {
536        required_resource_capabilities: BTreeSet::new(),
537        resource_alternative: alternative,
538        nodes,
539        logical_allocations: allocations,
540        physical_slots: slots,
541        initial_knobs: ExecutionKnobs::serial(),
542        adaptations: vec![],
543    })?;
544    let predictions = dag
545        .nodes()
546        .keys()
547        .map(|node| {
548            let prediction = StagePrediction::new(node.clone(), policy.stage_nanos);
549            if node == &stage {
550                prediction.with_io(vec![IoPrediction::new(
551                    IoBufferKind::Serialization,
552                    payload_bytes,
553                    u64::try_from(publication.entries().len()).unwrap_or(u64::MAX),
554                )])
555            } else if node == &commit {
556                prediction.with_io(vec![IoPrediction::new(IoBufferKind::Publication, 1, 1)])
557            } else {
558                prediction
559            }
560        })
561        .collect::<Vec<_>>();
562    let prediction = PlanPrediction::new(
563        policy
564            .stage_nanos
565            .checked_mul(predictions.len() as u64)
566            .ok_or(SerialProductPublicationPlanError::Overflow)?,
567        PredictionConfidence::new(policy.confidence_parts_per_million)?,
568        vec![],
569        predictions,
570    )?;
571    let artifacts = publication
572        .entries()
573        .iter()
574        .map(|entry| {
575            PlannedArtifact::new(entry.artifact(), commit.clone(), ArtifactRole::Output, None)
576        })
577        .collect();
578    let layouts = PublicationLayoutLedger::new(
579        publication
580            .entries()
581            .iter()
582            .map(|entry| {
583                PublicationPhysicalLayout::new(
584                    PublicationParticipant::Product {
585                        graph_id: publication.graph_id(),
586                        node_id: entry.node(),
587                    },
588                    entry.artifact(),
589                    layout_id(entry.artifact()),
590                    PublicationStaging::new(
591                        stage.clone(),
592                        WorkDependency::Work(stage.clone()),
593                        IoBufferKind::Serialization,
594                        writer_allocation.clone(),
595                    )
596                    .expect("serial staging contract"),
597                    PublicationResourceBounds::new(
598                        entry.payload_bytes(),
599                        entry.payload_bytes(),
600                        entry.payload_bytes(),
601                        0,
602                    )
603                    .expect("planned payload is nonzero"),
604                )
605            })
606            .collect(),
607    )?;
608    let catalog =
609        ImplementationContractCatalog::from_registry(registry, [policy.implementation.clone()])?;
610    Ok(PhysicalWorkBinding::new_with_product_publication(
611        catalog,
612        dag,
613        prediction,
614        artifacts,
615        ObservationTransactionWork::new_sealed_product_publication(check, commit),
616        layouts,
617        publication,
618    )?)
619}

677/// Planning failure for serial product publication.
678#[derive(Debug)]
679pub enum SerialProductPublicationPlanError {
680    /// A resource or payload calculation overflowed.
681    Overflow,
682    /// Planned product authority rejected the generation.
683    Publication(ProductPublicationError),
684    /// Execution DAG validation failed.
685    Execution(ExecutionError),
686    /// Physical binding or prediction validation failed.
687    Physical(PhysicalWorkBindingError),
688    /// Publication layout validation failed.
689    Layout(PublicationLayoutError),
690}
691
692impl fmt::Display for SerialProductPublicationPlanError {
693    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
694        write!(f, "serial product publication planning failed: {self:?}")
695    }
696}
697impl Error for SerialProductPublicationPlanError {}
698impl From<ProductPublicationError> for SerialProductPublicationPlanError {
699    fn from(v: ProductPublicationError) -> Self {
700        Self::Publication(v)
701    }
702}
703impl From<ExecutionError> for SerialProductPublicationPlanError {
704    fn from(v: ExecutionError) -> Self {
705        Self::Execution(v)
706    }
707}
708impl From<PhysicalWorkBindingError> for SerialProductPublicationPlanError {
709    fn from(v: PhysicalWorkBindingError) -> Self {
710        Self::Physical(v)
711    }
712}
713impl From<PublicationLayoutError> for SerialProductPublicationPlanError {
714    fn from(v: PublicationLayoutError) -> Self {
715        Self::Layout(v)
716    }
717}

752/// Stateful leased generation, serial staging, and atomic publication implementation.
753pub struct SerialProductPublicationExecutor<S> {
754    id: WorkImplementationId,
755    publication: ProductPublicationPlan,
756    state: Mutex<SerialProductPublicationState>,
757    sink: S,
758}

760impl<S: SerialProductPublicationSink> SerialProductPublicationExecutor<S> {
761    /// Bind pending scientific inputs to their immutable pre-seal plan and sink.
762    pub fn new(
763        id: WorkImplementationId,
764        problem: CompiledProblem,
765        publication: ProductPublicationPlan,
766        planned: PlannedContinuumGeneration,
767        scientific: MajorCycleCompletion,
768        reconstruction_masks: Option<ReconstructionMaskSet>,
769        sink: S,
770    ) -> Result<Self, SerialProductPublicationExecutionError<S::Error>> {
771        if publication.problem_id() != problem.problem_id()
772            || publication.graph_id() != problem.product_graph().graph_id()
773            || publication.generation_id() != planned.generation_id()
774            || scientific.normal_state().problem_id() != problem.problem_id()
775            || scientific.model_completion().problem() != problem.problem_id()
776        {
777            return Err(SerialProductPublicationExecutionError::State);
778        }
779        Ok(Self {
780            id,
781            publication,
782            state: Mutex::new(SerialProductPublicationState {
783                problem,
784                planned: Some(planned),
785                scientific: Some(scientific),
786                reconstruction_masks,
787                produced: None,
788                sealed: None,
789                projection: None,
790                published: None,
791                staged_measurements: None,
792            }),
793            sink,
794        })
795    }
796
797    /// Borrow the application/storage sink.
798    #[must_use]
799    pub const fn sink(&self) -> &S {
800        &self.sink
801    }
802
803    /// Consume the payload-free completion after a successful publication run.
804    pub fn take_completion(&self) -> Option<SerialProductPublicationCompletion> {
805        let mut state = self.state.lock().ok()?;
806        Some(SerialProductPublicationCompletion {
807            planned: state.planned.take()?,
808            scientific: state.scientific.take()?,
809            published: state.published.take()?,
810        })
811    }
812}

814impl<S: SerialProductPublicationSink> WorkImplementation for SerialProductPublicationExecutor<S> {
815    type Error = SerialProductPublicationExecutionError<S::Error>;
816    fn implementation_id(&self) -> &WorkImplementationId {
817        &self.id
818    }
819    fn execute(&self, context: WorkExecutionContext<'_>) -> Result<WorkMeasurements, Self::Error> {
820        let mut artifact_measurements = Vec::new();
821        if context.node().id.as_str() == GENERATE {
822            let mut state = self
823                .state
824                .lock()
825                .map_err(|_| SerialProductPublicationExecutionError::State)?;
826            let produced = {
827                let planned = state
828                    .planned
829                    .as_ref()
830                    .ok_or(SerialProductPublicationExecutionError::State)?;
831                let scientific = state
832                    .scientific
833                    .as_ref()
834                    .ok_or(SerialProductPublicationExecutionError::State)?;
835                let mut inputs =
836                    ContinuumProductInputs::from_major_cycle(&state.problem, scientific)
837                        .map_err(SerialProductPublicationExecutionError::Products)?;
838                if let Some(masks) = state.reconstruction_masks.as_ref() {
839                    inputs = match masks {
840                        ReconstructionMaskSet::Shared(mask) => inputs
841                            .with_reconstruction_mask(mask)
842                            .map_err(SerialProductPublicationExecutionError::Products)?,
843                        ReconstructionMaskSet::Domains(masks) => {
844                            ContinuumSourceCatalog::from_major_cycle_with_domain_masks(
845                                &state.problem,
846                                scientific,
847                                masks,
848                            )
849                            .map_err(SerialProductPublicationExecutionError::Products)?;
850                            inputs
851                                .with_domain_reconstruction_masks(masks)
852                                .map_err(SerialProductPublicationExecutionError::Products)?
853                        }
854                        ReconstructionMaskSet::Coupled(masks) => inputs
855                            .with_coupled_reconstruction_masks(masks)
856                            .map_err(SerialProductPublicationExecutionError::Products)?,
857                    };
858                }
859                produce_continuum_members(planned, &inputs)
860                    .map_err(SerialProductPublicationExecutionError::Products)?
861            };
862            state.reconstruction_masks = None;
863            state.produced = Some(produced);
864        } else if context.node().id.as_str() == SEAL {
865            let mut state = self
866                .state
867                .lock()
868                .map_err(|_| SerialProductPublicationExecutionError::State)?;
869            let produced = state
870                .produced
871                .take()
872                .ok_or(SerialProductPublicationExecutionError::State)?;
873            let authority = ProductGenerationAuthority::bind(&state.problem);
874            let planned = state
875                .planned
876                .as_ref()
877                .ok_or(SerialProductPublicationExecutionError::State)?;
878            let sealed = authority
879                .authorize(planned, &produced)
880                .map_err(SerialProductPublicationExecutionError::Products)?;
881            let projection = PublicationProjection::from_sealed(&sealed)
882                .map_err(SerialProductPublicationExecutionError::Products)?;
883            self.publication
884                .authorize(&projection)
885                .map_err(SerialProductPublicationExecutionError::Publication)?;
886            state.projection = Some(projection);
887            state.sealed = Some(sealed);
888        } else if context.node().id.as_str() == STAGE {
889            let mut state = self
890                .state
891                .lock()
892                .map_err(|_| SerialProductPublicationExecutionError::State)?;
893            let projection = state
894                .projection
895                .as_ref()
896                .ok_or(SerialProductPublicationExecutionError::State)?;
897            let authorization = self
898                .publication
899                .authorize(projection)
900                .map_err(SerialProductPublicationExecutionError::Publication)?;
901            let sealed = state
902                .sealed
903                .as_ref()
904                .ok_or(SerialProductPublicationExecutionError::State)?;
905            for entry in authorization.entries() {
906                let member = sealed
907                    .members()
908                    .iter()
909                    .find(|member| member.node() == entry.node())
910                    .ok_or(SerialProductPublicationExecutionError::State)?;
911                self.sink
912                    .stage(entry.planned_identity(), entry.observed_identity(), member)
913                    .map_err(SerialProductPublicationExecutionError::Sink)?;
914                artifact_measurements.push(
915                    ArtifactMeasurement::new(
916                        entry.planned_identity(),
917                        Some(entry.observed_identity()),
918                        ArtifactDisposition::PublicationPrepared,
919                        entry.payload_bytes(),
920                        None,
921                    )
922                    .expect("staged is adapter-owned"),
923                );
924            }
925            let sealed = state
926                .sealed
927                .take()
928                .ok_or(SerialProductPublicationExecutionError::State)?;
929            state.published = Some(sealed.into_published_summary());
930            state.staged_measurements = Some(std::mem::take(&mut artifact_measurements));
931        } else if context.node().id.as_str() == COMMIT {
932            artifact_measurements = self
933                .state
934                .lock()
935                .map_err(|_| SerialProductPublicationExecutionError::State)?
936                .staged_measurements
937                .take()
938                .ok_or(SerialProductPublicationExecutionError::State)?;
939        }
940        let resources = context
941            .node()
942            .claims
943            .iter()
944            .map(|claim| {
945                ResourceMeasurement::new(
946                    claim.resource.clone(),
947                    claim.lifetime.clone(),
948                    claim.amount,
949                )
950            })
951            .collect();
952        let io = context
953            .node()
954            .claims
955            .iter()
956            .filter_map(|claim| match claim.resource {
957                LeaseResource::IoBuffer(kind) => Some(IoMeasurement::unobserved(kind)),
958                _ => None,
959            })
960            .collect();
961        Ok(WorkMeasurements::new(resources, io, artifact_measurements))
962    }
963    fn failure_measurements<'a>(&'a self, _error: &'a Self::Error) -> Option<&'a WorkMeasurements> {
964        None
965    }
966    fn wait_for_fence(
967        &self,
968        _context: WorkExecutionContext<'_>,
969        _fence: FenceKind,
970    ) -> Result<WorkMeasurements, Self::Error> {
971        Ok(WorkMeasurements::default())
972    }
973    fn complete_observation_read(
974        &self,
975        _completion: ObservationReadCompletionContext,
976    ) -> Result<AttemptBoundObservationCompletion, Self::Error> {
977        Err(SerialProductPublicationExecutionError::State)
978    }
979    fn complete_product_generation(
980        &self,
981        context: WorkExecutionContext<'_>,
982    ) -> Result<Option<PublicationProjection>, Self::Error> {
983        if context.node().id.as_str() != COMMIT {
984            return Err(SerialProductPublicationExecutionError::State);
985        }
986        Ok(Some(
987            self.state
988                .lock()
989                .map_err(|_| SerialProductPublicationExecutionError::State)?
990                .projection
991                .clone()
992                .ok_or(SerialProductPublicationExecutionError::State)?,
993        ))
994    }
995    fn publish(&self, context: WorkExecutionContext<'_>) -> Result<(), Self::Error> {
996        let _ = context;
997        Err(SerialProductPublicationExecutionError::MissingAuthorization)
998    }
999
1000    fn publish_product_member(
1001        &self,
1002        context: WorkExecutionContext<'_>,
1003        entry: AuthorizedProductPublicationEntry,
1004    ) -> Option<Result<ArtifactMeasurement, ProductMemberPublicationFailure<Self::Error>>> {
1005        let authorized = context.product_publication().is_some_and(|authorization| {
1006            authorization.problem_id() == self.publication.problem_id()
1007                && authorization.graph_id() == self.publication.graph_id()
1008                && authorization.generation_id() == self.publication.generation_id()
1009                && authorization.entries().contains(&entry)
1010        });
1011        if !authorized {
1012            return Some(Err(ProductMemberPublicationFailure::new(
1013                SerialProductPublicationExecutionError::MissingAuthorization,
1014                publication_measurement(entry, ArtifactDisposition::PublicationFailed),
1015            )));
1016        }
1017        Some(match self.sink.promote(entry) {
1018            Ok(()) => Ok(publication_measurement(
1019                entry,
1020                ArtifactDisposition::Published,
1021            )),
1022            Err(error) => {
1023                let disposition = match error.kind() {
1024                    MemberPromotionFailureKind::Failed => ArtifactDisposition::PublicationFailed,
1025                    MemberPromotionFailureKind::Uncertain => {
1026                        ArtifactDisposition::PublicationUncertain
1027                    }
1028                };
1029                Err(ProductMemberPublicationFailure::new(
1030                    SerialProductPublicationExecutionError::Promotion(error),
1031                    publication_measurement(entry, disposition),
1032                ))
1033            }
1034        })
1035    }
1036}
1037
1038fn publication_measurement(
1039    entry: AuthorizedProductPublicationEntry,
1040    disposition: ArtifactDisposition,
1041) -> ArtifactMeasurement {
1042    ArtifactMeasurement::new(
1043        entry.planned_identity(),
1044        Some(entry.observed_identity()),
1045        disposition,
1046        entry.payload_bytes(),
1047        None,
1048    )
1049    .expect("publication outcome is adapter-owned")
1050}

1052/// Execution failure from the serial publication owner.
1053#[derive(Debug)]
1054pub enum SerialProductPublicationExecutionError<E> {
1055    /// Affine execution state was missing or had already been consumed.
1056    State,
1057    /// The terminal publication context omitted the runtime-minted authorization.
1058    MissingAuthorization,
1059    /// The planned generation and completed projection did not match.
1060    Publication(ProductPublicationError),
1061    /// Product projection failed after scientific production.
1062    Products(casa_imaging_products::ProductsError),
1063    /// The storage sink rejected private staging.
1064    Sink(E),
1065    /// One independently atomic member replacement failed or became uncertain.
1066    Promotion(MemberPromotionFailure<E>),
1067}
1068impl<E: Error> fmt::Display for SerialProductPublicationExecutionError<E> {
1069    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
1070        write!(f, "serial product publication failed: {self:?}")
1071    }
1072}
1073impl<E: Error + 'static> Error for SerialProductPublicationExecutionError<E> {}
