2//! Distance units of baseline-length antenna selection.
3//!
4//! `MSAntennaParse::getUnitFactor` (`ms/MSSel/MSAntennaParse.cc:336-347`)
5//! builds a casacore `Unit` from the `UNIT` token (`[a-zA-Z]+`), refuses one
6//! that does not conform to metres, and returns how many of that unit make
7//! one metre. casacore resolves an alphabetic unit string as one field
8//! (`casa/Quanta/UnitVal.cc:204-226`): the whole name in the user, customary
9//! and SI maps, else a one-character prefix and a unit, else the
10//! two-character prefix `da` and a unit. The tables below hold every unit an
11//! alphabetic token can name (`casa/Quanta/UnitMap{3,4,5,6,7}.cc` and the SI
12//! base names of `UnitDim.cc:115-130`), since a non-length name must also
13//! shadow a prefixed reading (`min` is minutes, not milli-inches).
14

17    ("Q", 1.0e30),
18    ("R", 1.0e27),
19    ("Y", 1.0e24),
20    ("Z", 1.0e21),
21    ("E", 1.0e18),
22    ("P", 1.0e15),
23    ("T", 1.0e12),
24    ("G", 1.0e9),
25    ("M", 1.0e6),
26    ("k", 1.0e3),
27    ("h", 1.0e2),
28    ("da", 1.0e1),
29    ("d", 1.0e-1),
30    ("c", 1.0e-2),
31    ("m", 1.0e-3),
32    ("u", 1.0e-6),
33    ("n", 1.0e-9),
34    ("p", 1.0e-12),

43/// IAU light time for one astronomical unit (`UnitMap4.cc:35`).
44const IAU_TAU_A: f64 = 499.004_783_7;
45/// Speed of light, m/s (`casa/BasicSL/Constants.h`, `C::c`).
46const SPEED_OF_LIGHT: f64 = 2.997_924_58e8;
47/// One arcsecond in radians (`Constants.h`, `C::arcsec`).
48const ARCSEC: f64 = 0.000_004_848_136_811_095_36;
49
50/// The length units an alphabetic token can name, in metres, written as
51/// casacore composes them (`UnitVal(factor, "m" | "cm" | "AU")`).
52fn length_unit(name: &str) -> Option<f64> {
53    let centimetre = 1.0e-2;
54    let astronomical_unit = SPEED_OF_LIGHT * IAU_TAU_A;
55    Some(match name {
56        "m" => 1.0,
57        "AU" | "UA" | "AE" => astronomical_unit,
58        "pc" => astronomical_unit * (1.0 / ARCSEC),
59        "Angstrom" => 1.0e-10,
60        "ft" => 12.0 * 2.54e-2,
61        "fur" => centimetre * (7920.0 * 2.54),
62        "in" => centimetre * 2.54,
63        "ly" => 9.460_730_47e15,
64        "mile" => 63360.0 * 2.54e-2,
65        "yd" => centimetre * (36.0 * 2.54),
66        _ => return None,
67    })
68}

86/// What a whole unit name is in casacore's maps.
87#[derive(Clone, Copy)]
88enum Named {
89    Length(f64),
90    Other,
91}
92
93fn named_unit(name: &str) -> Option<Named> {
94    length_unit(name)
95        .map(Named::Length)
96        .or_else(|| OTHER_UNITS.contains(&name).then_some(Named::Other))
97}
98
99fn prefix(name: &str) -> Option<f64> {
100    PREFIXES
101        .iter()
102        .find(|(prefix, _)| *prefix == name)
103        .map(|(_, factor)| *factor)
104}
105
106/// `UnitVal::field` for an alphabetic key (`UnitVal.cc:212-226`).
107fn resolve(key: &str) -> Option<Named> {
108    if let Some(named) = named_unit(key) {
109        return Some(named);
110    }
111    // Prefixes are ASCII, so these byte splits are character boundaries.
112    let one = key.get(..1).and_then(prefix).filter(|_| key.len() > 1)?;
113    if let Some(named) = named_unit(&key[1..]) {
114        return Some(scale(one, named));
115    }
116    let two = key.get(..2).and_then(prefix).filter(|_| key.len() > 2)?;
117    named_unit(&key[2..]).map(|named| scale(two, named))
118}
119
120fn scale(prefix: f64, named: Named) -> Named {
121    match named {
122        Named::Length(metres) => Named::Length(prefix * metres),
123        Named::Other => Named::Other,
124    }
125}

127/// Why a unit cannot scale a baseline length.
128#[derive(Clone, Copy, Debug, PartialEq, Eq)]
129pub(super) enum UnitError {
130    /// casacore knows no such unit: `Unit::check` throws an `AipsError`
131    /// (`casa/Quanta/Unit.cc:218-222`).
132    Unknown,
133    /// The unit is not a length: `MSSelectionAntennaParseError`
134    /// (`MSAntennaParse.cc:341-344`).
135    NotDistance,
136}
137
138impl UnitError {
139    /// casacore's message for `unit`.
140    pub(super) fn message(self, unit: &str) -> String {
141        match self {
142            Self::Unknown => format!("Unit::check Illegal unit string '{unit}'"),
143            Self::NotDistance => format!("Unit {unit} must be a distance unit (like m)"),
144        }
145    }
146}

148/// `MSAntennaParse::getUnitFactor`: how many `unit` make one metre.
149///
150/// A baseline length written `value unit` is `value / factor` metres
151/// (`MSAntennaGram.yy:513-514`).
152pub(super) fn unit_factor(unit: &str) -> Result<f64, UnitError> {
153    match resolve(unit) {
154        // Quantum<Double>(1, "m").getValue(unit) (Quantum.tcc:222-229).
155        Some(Named::Length(metres)) => Ok(1.0 / (metres / 1.0)),
156        Some(Named::Other) => Err(UnitError::NotDistance),
157        None => Err(UnitError::Unknown),
158    }
159}
