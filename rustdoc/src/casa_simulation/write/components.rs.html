2//! The component lists' visibilities in a write batch: after the model
3//! images are predicted into DATA, each `predict(complist=…)` covering a
4//! row's scan adds its components' visibilities to the row
5//! ([`crate::predict::ComponentPrediction::add_rows`]), on the CPU whatever
6//! the backend, a run of rows per job on the team. The sum is the same for
7//! any worker count: each row is formed by one job, in the plan's order.
8
9use casa_imaging_runtime::pass::WorkerTeam;
10use num_complex::Complex32;
11
12use super::geometry::RowGeometry;
13use crate::session::error::{SimulatorError, SimulatorResult};
14use crate::session::observe::Scan;
15use crate::session::plan::DataPlan;
16
17/// Rows of one job.
18const JOB_ROWS: usize = 256;
19
20struct Job<'a> {
21    /// The scan's index in the session.
22    index: usize,
23    scan: &'a Scan,
24    rows: &'a [RowGeometry],
25    cells: &'a mut [Complex32],
26}
27
28/// Add the component lists `plan` predicts to `data`, the batch's DATA of
29/// `rows` (`samples` cells a row), for each run of `runs` (scan, first row,
30/// rows).
31pub(super) fn add_components(
32    team: &WorkerTeam,
33    (scans, rows): (&[Scan], &[RowGeometry]),
34    plan: &DataPlan,
35    runs: &[(usize, usize, usize)],
36    (data, samples): (&mut [Complex32], usize),
37) -> SimulatorResult<()> {
38    let mut jobs = Vec::new();
39    let mut rest = data;
40    let mut position = 0;
41    for &(scan, start, count) in runs {
42        let (_, tail) = std::mem::take(&mut rest).split_at_mut((start - position) * samples);
43        rest = tail;
44        position = start;
45        let mut offset = 0;
46        while offset < count {
47            let take = JOB_ROWS.min(count - offset);
48            let (cells, tail) = std::mem::take(&mut rest).split_at_mut(take * samples);
49            rest = tail;
50            position += take;
51            if plan.components_for_scan(scan).next().is_some() {
52                jobs.push(Job {
53                    index: scan,
54                    scan: &scans[scan],
55                    rows: &rows[start + offset..start + offset + take],
56                    cells,
57                });
58            }
59            offset += take;
60        }
61    }
62    team.for_each_mut(&mut jobs, |_, job| {
63        for components in plan.components_for_scan(job.index) {
64            components.add_rows(
65                (job.index, job.scan.spectral_window_id),
66                job.rows.iter().map(|row| row.uvw_m),
67                job.cells,
68            );
69        }
70        Ok::<_, SimulatorError>(())
71    })
72}
