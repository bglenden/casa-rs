15pub(super) struct CoordReductionSpec<'a> {
16    pub(super) full_shape: &'a [usize],
17    pub(super) start: &'a [usize],
18    pub(super) shape: &'a [usize],
19    pub(super) stride: Option<&'a [usize]>,
20    pub(super) kept_axes: &'a [usize],
21    pub(super) selected_stokes: Option<&'a [StokesType]>,
22}

24pub(super) fn reduce_coordinates_and_beams(
25    input_coords: &CoordinateSystem,
26    native: Option<RecordValue>,
27    spec: CoordReductionSpec<'_>,
28    mut image_info: ImageInfo,
29) -> Result<(CoordinateSystem, Option<RecordValue>, ImageInfo), ImageError> {
30    let ndim = spec.full_shape.len();
31    let section_record = |mut record: RecordValue| -> Result<RecordValue, ImageError> {
32        section_record_reference_pixels(&mut record, spec.start, spec.shape, spec.stride)?;
33        if let Some(stokes) = spec.selected_stokes {
34            replace_stokes_in_record(&mut record, stokes)?;
35        }
36        Ok(record)
37    };
38    let shifted = if spec.stride.is_some() {
39        input_coords.restore_record_with_layout(&section_record(input_coords.to_record())?)?
40    } else {
41        subimage_coordinates_with_stokes(
42            input_coords,
43            spec.start,
44            spec.shape,
45            spec.selected_stokes,
46        )?
47    };
48    let shifted_legacy = native.and_then(|record| section_record(record).ok());
49
50    let (out_coords, out_legacy) = if spec.kept_axes.len() == ndim {
51        (shifted, shifted_legacy)
52    } else {
53        let record = shifted_legacy.unwrap_or_else(|| shifted.to_record());
54        let reduced = reduce_subimage_coordinate_record(&record, ndim, spec.kept_axes)?;
55        (CoordinateSystem::from_record(&reduced)?, Some(reduced))
56    };
57
58    image_info.beam_set = subset_beam_section(
59        &image_info.beam_set,
60        input_coords,
61        ndim,
62        spec.start,
63        spec.shape,
64        spec.stride,
65    )?;
66
67    Ok((out_coords, out_legacy, image_info))
68}

70/// Reduce a native subimage coordinate record to the retained pixel axes,
71/// preserving independent world order and any mapped local replacements.
72///
73/// # Errors
74/// The source coordinate record cannot be sliced on the selected axes.
75pub fn reduce_subimage_coordinate_record(
76    record: &RecordValue,
77    ndim: usize,
78    kept_axes: &[usize],
79) -> Result<RecordValue, ImageError> {
80    drop_axes_from_legacy_record(record, ndim, kept_axes).ok_or_else(|| {
81        ImageError::InvalidMetadata("could not reduce the subimage axis maps (#1002)".into())
82    })
83}

85fn find_system_axis(coords: &CoordinateSystem, kind: CoordinateType) -> Option<usize> {
86    coords
87        .coordinate_pixel_axes(coords.find_coordinate(kind)?)?
88        .into_iter()
89        .flatten()
90        .next()
91}

103pub(in crate::analysis) fn subset_beam_section(
104    beam_set: &ImageBeamSet,
105    input_coords: &CoordinateSystem,
106    ndim: usize,
107    start: &[usize],
108    shape: &[usize],
109    stride: Option<&[usize]>,
110) -> Result<ImageBeamSet, ImageError> {
111    if !beam_set.is_multi() {
112        return Ok(beam_set.clone());
113    }
114    let [channels, stokes] =
115        beam_section_indices(beam_set, input_coords, ndim, start, shape, stride);
116    beam_set.subset(&channels, &stokes)
117}

119pub(in crate::analysis) fn beam_section_indices(
120    beam_set: &ImageBeamSet,
121    input_coords: &CoordinateSystem,
122    ndim: usize,
123    start: &[usize],
124    shape: &[usize],
125    stride: Option<&[usize]>,
126) -> [Vec<usize>; 2] {
127    beam_section_indices_for_shape(
128        [beam_set.n_channels(), beam_set.n_stokes()],
129        input_coords,
130        ndim,
131        start,
132        shape,
133        stride,
134    )
135}

137pub(super) fn beam_section_indices_for_shape(
138    [channels, stokes]: [usize; 2],
139    input_coords: &CoordinateSystem,
140    ndim: usize,
141    start: &[usize],
142    shape: &[usize],
143    stride: Option<&[usize]>,
144) -> [Vec<usize>; 2] {
145    let indices = |kind, length: usize| match find_system_axis(input_coords, kind) {
146        Some(axis) if axis < ndim && length > 1 => {
147            let step = stride.map_or(1, |stride| stride[axis]);
148            (0..shape[axis])
149                .map(|index| (start[axis] + step * index).min(length - 1))
150                .collect::<Vec<_>>()
151        }
152        _ => vec![0],
153    };
154    [
155        indices(CoordinateType::Spectral, channels),
156        indices(CoordinateType::Stokes, stokes),
157    ]
158}

160fn strip_index_suffix(field: &str) -> (&str, Option<usize>) {
161    let digits_start = field
162        .rfind(|ch: char| !ch.is_ascii_digit())
163        .map_or(0, |pos| pos + 1);
164    let prefix = &field[..digits_start];
165    let suffix = field[digits_start..].parse::<usize>().ok();
166    (prefix, suffix)
167}

169pub(in crate::analysis) fn drop_axes_from_legacy_record(
170    record: &RecordValue,
171    ndim: usize,
172    kept_axes: &[usize],
173) -> Option<RecordValue> {
174    let placed = CoordinateSystem::record_coordinates(record).ok()?;
175    let axis_remap: Vec<Option<usize>> = (0..ndim)
176        .map(|old_ax| kept_axes.iter().position(|&k| k == old_ax))
177        .collect();
178
179    struct LegacyCoordEntry<'a> {
180        prefix: &'a str,
181        old_index: usize,
182        coord_val: &'a Value,
183        pixelmap: Vec<i32>,
184        worldmap: Vec<i32>,
185        pixelreplace: Option<&'a Value>,
186        worldreplace: Option<&'a Value>,
187    }
188
189    let mut kept_entries: Vec<LegacyCoordEntry<'_>> = Vec::new();
190    let mut surviving_worlds: BTreeSet<i32> = BTreeSet::new();
191
192    for entry in &placed {
193        let (prefix, Some(old_idx)) = strip_index_suffix(&entry.field) else {
194            return None;
195        };
196        let has_kept = entry
197            .pixel_axes
198            .iter()
199            .any(|ax| ax.is_some_and(|a| axis_remap.get(a).copied().flatten().is_some()));
200        if !has_kept {
201            continue;
202        }
203        let coord_val = record.get(&entry.field)?;
204        let pixelmap = read_i32_array(record.get(&format!("pixelmap{old_idx}"))?)?;
205        let worldmap = read_i32_array(record.get(&format!("worldmap{old_idx}"))?)?;
206        for (j, &pax) in pixelmap.iter().enumerate() {
207            let retained = pax < 0 || axis_remap.get(pax as usize).copied().flatten().is_some();
208            if retained
209                && let Some(&wax) = worldmap.get(j)
210                && wax >= 0
211            {
212                surviving_worlds.insert(wax);
213            }
214        }
215        kept_entries.push(LegacyCoordEntry {
216            prefix,
217            old_index: old_idx,
218            coord_val,
219            pixelmap,
220            worldmap,
221            pixelreplace: record.get(&format!("pixelreplace{old_idx}")),
222            worldreplace: record.get(&format!("worldreplace{old_idx}")),
223        });
224    }
225
226    // Preserve creation index order when renumbering 0..kept.
227    kept_entries.sort_by_key(|e| e.old_index);
228    let world_remap: BTreeMap<i32, i32> = surviving_worlds
229        .into_iter()
230        .enumerate()
231        .map(|(new_w, old_w)| (old_w, new_w as i32))
232        .collect();
233
234    let mut out_fields = Vec::new();
235    for field in record.fields() {
236        let name = &field.name;
237        let is_coord_key = name.starts_with("pixelmap")
238            || name.starts_with("worldmap")
239            || name.starts_with("pixelreplace")
240            || name.starts_with("worldreplace")
241            || name.starts_with("coordinate")
242            || [
243                "direction",
244                "spectral",
245                "stokes",
246                "linear",
247                "tabular",
248                "quality",
249            ]
250            .iter()
251            .any(|p| name.starts_with(p) && name[p.len()..].chars().all(|c| c.is_ascii_digit()));
252        if !is_coord_key {
253            out_fields.push(field.clone());
254        }
255    }
256
257    for (new_idx, entry) in kept_entries.into_iter().enumerate() {
258        let mut new_pmap = Vec::with_capacity(entry.pixelmap.len());
259        let mut new_wmap = Vec::with_capacity(entry.worldmap.len());
260        for j in 0..entry.pixelmap.len() {
261            let old_p = entry.pixelmap[j];
262            let old_w = entry.worldmap.get(j).copied().unwrap_or(old_p);
263            let kept_pixel = usize::try_from(old_p)
264                .ok()
265                .and_then(|pixel| axis_remap.get(pixel).copied().flatten());
266            new_pmap.push(match (old_p < 0, kept_pixel) {
267                (true, _) => old_p,
268                (_, Some(pixel)) => pixel as i32,
269                _ => -(old_p + 1),
270            });
271            new_wmap.push(if old_w < 0 {
272                old_w
273            } else if old_p < 0 || kept_pixel.is_some() {
274                *world_remap.get(&old_w)?
275            } else {
276                -(old_w + 1)
277            });
278        }
279        out_fields.push(RecordField::new(
280            format!("{}{new_idx}", entry.prefix),
281            entry.coord_val.clone(),
282        ));
283        out_fields.push(RecordField::new(
284            format!("worldmap{new_idx}"),
285            Value::Array(ArrayValue::from_i32_vec(new_wmap)),
286        ));
287        out_fields.push(RecordField::new(
288            format!("pixelmap{new_idx}"),
289            Value::Array(ArrayValue::from_i32_vec(new_pmap)),
290        ));
291        for (kind, field, replacement, map) in [
292            ("pixel", "crpix", entry.pixelreplace, &entry.pixelmap),
293            ("world", "crval", entry.worldreplace, &entry.worldmap),
294        ] {
295            let Some(replacement) = replacement else {
296                continue;
297            };
298            let mut value = replacement.clone();
299            if entry.prefix == "direction"
300                && let Value::Record(direction) = entry.coord_val
301                && let Value::Array(ArrayValue::Float64(values)) = &mut value
302                && let Some(Value::Array(ArrayValue::Float64(reference))) = direction.get(field)
303            {
304                // Rebuilding the surviving direction coordinate uses its
305                // reference values for newly removed axes (dropRemovedAxes).
306                for (axis, &old) in map.iter().enumerate() {
307                    let retained = if kind == "pixel" {
308                        usize::try_from(old)
309                            .ok()
310                            .is_some_and(|pixel| axis_remap.get(pixel).copied().flatten().is_some())
311                    } else {
312                        world_remap.contains_key(&old)
313                    };
314                    if old >= 0 && !retained {
315                        values[axis] = reference[axis];
316                    }
317                }
318            }
319            out_fields.push(RecordField::new(format!("{kind}replace{new_idx}"), value));
320        }
321    }
322
323    let out = RecordValue::new(out_fields);
324    Some(out)
325}

327fn read_i32_array(value: &Value) -> Option<Vec<i32>> {
328    match value {
329        Value::Array(ArrayValue::Int32(arr)) => Some(arr.iter().copied().collect()),
330        Value::Array(ArrayValue::Int64(arr)) => {
331            arr.iter().map(|&v| i32::try_from(v).ok()).collect()
332        }
333        _ => None,
334    }
