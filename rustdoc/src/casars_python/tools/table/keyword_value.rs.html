2//! Python values as table keywords, the way casatools stores them
3//! (`tools/casa-compat/table_tool_oracle.py`, CASA 6.7.6.14).
4//!
5//! A value arrives in its widest form ([`Incoming`]): a `bool`, `int`,
6//! `float`, `complex` or `str`, a NumPy array or scalar, a list (read as the
7//! NumPy array it makes), or a `dict` for a sub-record. A new keyword takes
8//! the natural type, casatools': an integer is a 32-bit `Int` (a 64-bit one
9//! only when a value does not fit, where CASA truncates it to 0), a float a
10//! `Double` (also a NumPy `float32`, scalar or array), a complex a `DComplex`
11//! (also a `complex64`), a string a `String`, or a table reference when it
12//! is `Table: ` and the path of a readable table.
13//!
14//! An existing keyword keeps its type, as casacore's `defineDataField` does:
15//! the new value must be of exactly the type the keyword has (casatools gives
16//! a value one of `Bool`, `Int`, `Double`, `DComplex`, `String` or a table,
17//! so a keyword of another width takes none), scalar or array. An integer
18//! that does not fit an `Int` is refused, where CASA truncates it. A
19//! one-element array is accepted for a scalar keyword.
20

33/// The elements of the NumPy array `array`, in C order, cast to `dtype` (the
34/// widest of their kind; a cast that could lose a value, such as `uint64` to
35/// `int64`, is refused) and copied in one pass.
36fn bulk<T: numpy::Element + Clone>(array: &Bound<'_, PyAny>, dtype: &str) -> PyResult<Vec<T>> {
37    let options = PyDict::new(array.py());
38    options.set_item("casting", "safe")?;
39    options.set_item("copy", false)?;
40    let cast = array.call_method("astype", (dtype,), Some(&options))?;
41    let typed = cast.extract::<PyReadonlyArrayDyn<'_, T>>()?;
42    Ok(typed.as_array().iter().cloned().collect())
43}

45/// The elements of a value, in the widest type of their kind.
46#[derive(Debug, Clone)]
47pub(super) enum Data {
48    Bools(Vec<bool>),
49    Ints(Vec<i64>),
50    /// Reals; casatools takes single precision ones as double.
51    Floats(Vec<f64>),
52    /// Complex numbers; casatools takes single precision ones as double.
53    Complexes(Vec<Complex64>),
54    Texts(Vec<String>),
55}

57/// A scalar (`shape` is `None`) or an array of `shape` in C order.
58#[derive(Debug, Clone)]
59pub(super) struct Wide {
60    pub(super) shape: Option<Vec<usize>>,
61    pub(super) data: Data,
62}

64/// A keyword value as Python gave it.
65#[derive(Debug, Clone)]
66pub(super) enum Incoming {
67    Wide(Wide),
68    Record(Vec<(String, Incoming)>),
69}

71fn mismatch(name: &str) -> Fail {
72    Fail::Runtime(format!(
73        "TableRecordRep::defineDataField - incorrect data type used for field {name}"
74    ))
75}

77fn unsupported(obj: &Bound<'_, PyAny>) -> PyErr {
78    let kind = obj
79        .get_type()
80        .name()
81        .map_or_else(|_| "object".into(), |name| name.to_string());
82    PyTypeError::new_err(format!("cannot store a {kind} as a table keyword"))
83}

85impl Incoming {
86    /// A Python value as a keyword value.
87    pub(super) fn from_python(obj: &Bound<'_, PyAny>) -> PyResult<Self> {
88        if let Ok(dict) = obj.cast::<PyDict>() {
89            let mut fields = Vec::with_capacity(dict.len());
90            for (key, value) in dict.iter() {
91                fields.push((key.extract::<String>()?, Self::from_python(&value)?));
92            }
93            return Ok(Self::Record(fields));
94        }
95        Wide::from_python(obj).map(Self::Wide)
96    }
97
98    /// The value as a new keyword.
99    pub(super) fn natural(self) -> Result<Value, Fail> {
100        match self {
101            Self::Wide(wide) => wide.natural(),
102            Self::Record(fields) => {
103                let mut record = Vec::with_capacity(fields.len());
104                for (name, value) in fields {
105                    record.push(RecordField::new(name, value.natural()?));
106                }
107                Ok(Value::Record(RecordValue::new(record)))
108            }
109        }
110    }
111
112    /// The value as the existing keyword `old` holds one.
113    pub(super) fn like(self, old: &Value, name: &str) -> Result<Value, Fail> {
114        match (&self, old) {
115            (Self::Wide(_), Value::Scalar(_) | Value::Array(_) | Value::TableRef(_)) => {
116                match self {
117                    Self::Wide(wide) => wide.like(old, name),
118                    Self::Record(_) => Err(mismatch(name)),
119                }
120            }
121            (Self::Record(_), Value::Record(_)) => self.natural(),
122            _ => Err(mismatch(name)),
123        }
124    }
125}

127impl Wide {
128    fn from_python(obj: &Bound<'_, PyAny>) -> PyResult<Self> {
129        let scalar = |data| Self { shape: None, data };
130        if obj.is_instance_of::<PyBool>() {
131            return Ok(scalar(Data::Bools(vec![obj.extract()?])));
132        }
133        if obj.is_instance_of::<PyInt>() {
134            return Ok(scalar(Data::Ints(vec![obj.extract()?])));
135        }
136        if obj.is_instance_of::<PyFloat>() {
137            return Ok(scalar(Data::Floats(vec![obj.extract()?])));
138        }
139        if let Ok(number) = obj.cast::<PyComplex>() {
140            let value = Complex64::new(number.real(), number.imag());
141            return Ok(scalar(Data::Complexes(vec![value])));
142        }
143        if obj.is_instance_of::<PyString>() {
144            return Ok(scalar(Data::Texts(vec![obj.extract()?])));
145        }
146        if obj.is_instance_of::<PyList>() || obj.is_instance_of::<PyTuple>() {
147            let array = obj.py().import("numpy")?.call_method1("asarray", (obj,))?;
148            return Self::from_numpy(&array);
149        }
150        if obj.hasattr("dtype")? && obj.hasattr("shape")? {
151            return Self::from_numpy(obj);
152        }
153        Err(unsupported(obj))
154    }
155
156    /// A NumPy array, or a NumPy scalar, by its dtype's kind and size.
157    ///
158    /// An array of numbers is copied once, in bulk, in the widest type of
159    /// its kind (`bulk`): a large complex `DATA` array is not expanded into
160    /// a Python object per element.
161    fn from_numpy(obj: &Bound<'_, PyAny>) -> PyResult<Self> {
162        let numpy = obj.py().import("numpy")?;
163        let is_array = obj.is_instance(&numpy.getattr("ndarray")?)?;
164        let dtype = obj.getattr("dtype")?;
165        let kind: String = dtype.getattr("kind")?.extract()?;
166        if is_array {
167            let shape: Vec<usize> = obj.getattr("shape")?.extract()?;
168            let data = match kind.as_str() {
169                "b" => Some(Data::Bools(bulk::<bool>(obj, "bool")?)),
170                "i" | "u" => Some(Data::Ints(bulk::<i64>(obj, "int64")?)),
171                "f" => Some(Data::Floats(bulk::<f64>(obj, "float64")?)),
172                "c" => Some(Data::Complexes(bulk::<Complex64>(obj, "complex128")?)),
173                _ => None,
174            };
175            if let Some(data) = data {
176                return Ok(Self {
177                    shape: Some(shape),
178                    data,
179                });
180            }
181        }
182        let (shape, flat) = if is_array {
183            let shape: Vec<usize> = obj.getattr("shape")?.extract()?;
184            let flat = obj.call_method1("ravel", ("C",))?.call_method0("tolist")?;
185            (Some(shape), flat)
186        } else {
187            (
188                None,
189                PyList::new(obj.py(), [obj.call_method0("item")?])?.into_any(),
190            )
191        };
192        let data = match kind.as_str() {
193            "b" => Data::Bools(flat.extract()?),
194            "i" | "u" => Data::Ints(flat.extract()?),
195            "f" => Data::Floats(flat.extract()?),
196            "c" => {
197                let mut values = Vec::new();
198                for item in flat.try_iter()? {
199                    let number = item?.cast_into::<PyComplex>()?;
200                    values.push(Complex64::new(number.real(), number.imag()));
201                }
202                Data::Complexes(values)
203            }
204            "U" | "S" => Data::Texts(flat.extract()?),
205            _ => return Err(unsupported(obj)),
206        };
207        Ok(Self { shape, data })
208    }
209
210    fn natural(self) -> Result<Value, Fail> {
211        let Self { shape, data } = self;
212        match data {
213            Data::Bools(values) => build(shape, values, ScalarValue::Bool, ArrayValue::Bool),
214            Data::Ints(values) if values.iter().all(|v| i32::try_from(*v).is_ok()) => {
215                let narrowed = values.iter().map(|v| *v as i32).collect();
216                build(shape, narrowed, ScalarValue::Int32, ArrayValue::Int32)
217            }
218            Data::Ints(values) => build(shape, values, ScalarValue::Int64, ArrayValue::Int64),
219            Data::Floats(values) => build(shape, values, ScalarValue::Float64, ArrayValue::Float64),
220            Data::Complexes(values) => {
221                build(shape, values, ScalarValue::Complex64, ArrayValue::Complex64)
222            }
223            // A string that names a readable table is a reference to it
224            // (`TableRecord::defineFromValueHolder`); any other is a string.
225            Data::Texts(mut values) if shape.is_none() => {
226                let text = values.swap_remove(0);
227                Ok(match table_reference(&text) {
228                    Some(path) => Value::TableRef(path.to_string()),
229                    None => Value::Scalar(ScalarValue::String(text)),
230                })
231            }
232            Data::Texts(values) => build(shape, values, ScalarValue::String, ArrayValue::String),
233        }
234    }
235
236    /// The value as the keyword `old` holds it, or the refusal casacore
237    /// gives a value of another type. `defineDataField` takes a value of
238    /// exactly the type the field has, and casatools gives every value one
239    /// of five (`Bool`, `Int`, `Double`, `DComplex`, `String`, or a table),
240    /// so a field of another width (`Float`, `Short`, ...) takes none. The
241    /// exception is an integer beyond 32 bits, which casa-rs keeps (casatools
242    /// truncates it to 0) in a field of 64 bits.
243    fn like(self, old: &Value, name: &str) -> Result<Value, Fail> {
244        let Self { shape, data } = self;
245        if let Value::TableRef(_) = old {
246            return match (&shape, &data) {
247                (None, Data::Texts(values)) => table_reference(&values[0])
248                    .map(|path| Value::TableRef(path.to_string()))
249                    .ok_or_else(|| mismatch(name)),
250                _ => Err(mismatch(name)),
251            };
252        }
253        let scalar_old = matches!(old, Value::Scalar(_));
254        let single_element = matches!(&shape, Some(shape) if shape.iter().product::<usize>() == 1);
255        let shape = match (scalar_old, shape) {
256            (true, None) => None,
257            (true, Some(_)) if single_element => None,
258            (false, Some(shape)) => Some(shape),
259            _ => return Err(mismatch(name)),
260        };
261        let overflow = |value: i64| {
262            Fail::Runtime(format!(
263                "{value} does not fit the integer keyword {name}, which holds fewer bits"
264            ))
265        };
266        macro_rules! ints {
267            ($values:expr, $ty:ty, $scalar:ident, $array:ident) => {{
268                let mut narrowed: Vec<$ty> = Vec::with_capacity($values.len());
269                for value in &$values {
270                    narrowed.push(<$ty>::try_from(*value).map_err(|_| overflow(*value))?);
271                }
272                build(shape, narrowed, ScalarValue::$scalar, ArrayValue::$array)
273            }};
274        }
275        match (element_kind(old), data) {
276            (Kind::Bool, Data::Bools(values)) => {
277                build(shape, values, ScalarValue::Bool, ArrayValue::Bool)
278            }
279            (Kind::Int32, Data::Ints(values)) => ints!(values, i32, Int32, Int32),
280            (Kind::Int64, Data::Ints(values))
281                if values.iter().any(|v| i32::try_from(*v).is_err()) =>
282            {
283                build(shape, values, ScalarValue::Int64, ArrayValue::Int64)
284            }
285            (Kind::Float64, Data::Floats(values)) => {
286                build(shape, values, ScalarValue::Float64, ArrayValue::Float64)
287            }
288            (Kind::Complex64, Data::Complexes(values)) => {
289                build(shape, values, ScalarValue::Complex64, ArrayValue::Complex64)
290            }
291            // A string that names a readable table is a table to casacore, and
292            // a table takes no string field's place.
293            (Kind::Text, Data::Texts(values))
294                if shape.is_some() || table_reference(&values[0]).is_none() =>
295            {
296                build(shape, values, ScalarValue::String, ArrayValue::String)
297            }
298            _ => Err(mismatch(name)),
299        }
300    }
301}

303/// The path of the table a string names as casacore reads it
304/// (`TableRecord::defineFromValueHolder`): `Table: ` and the path of a
305/// readable table.
306fn table_reference(text: &str) -> Option<&str> {
307    let path = text.strip_prefix("Table: ")?;
308    std::fs::File::open(Path::new(path).join("table.dat"))
309        .ok()
310        .map(|_| path)
311}

313/// The type of an existing keyword, as far as a Python value can take its
314/// place.
315#[derive(Debug, Clone, Copy)]
316enum Kind {
317    Bool,
318    Int32,
319    Int64,
320    Float64,
321    Complex64,
322    Text,
323    /// Any other width (`Float`, `Short`, ...): casatools gives no value
324    /// that type.
325    Other,
326}
327
328fn element_kind(old: &Value) -> Kind {
329    match old {
330        Value::Scalar(scalar) => match scalar {
331            ScalarValue::Bool(_) => Kind::Bool,
332            ScalarValue::Int32(_) => Kind::Int32,
333            ScalarValue::Int64(_) => Kind::Int64,
334            ScalarValue::Float64(_) => Kind::Float64,
335            ScalarValue::Complex64(_) => Kind::Complex64,
336            ScalarValue::String(_) => Kind::Text,
337            _ => Kind::Other,
338        },
339        Value::Array(array) => match array {
340            ArrayValue::Bool(_) => Kind::Bool,
341            ArrayValue::Int32(_) => Kind::Int32,
342            ArrayValue::Int64(_) => Kind::Int64,
343            ArrayValue::Float64(_) => Kind::Float64,
344            ArrayValue::Complex64(_) => Kind::Complex64,
345            ArrayValue::String(_) => Kind::Text,
346            _ => Kind::Other,
347        },
348        _ => Kind::Other,
349    }
350}

352/// A scalar from the one element of `values`, or an array of `shape` (in C
353/// order: element `[i, j]` of a NumPy array is element `(i, j)` of the
354/// casacore array).
355pub(super) fn build<T>(
356    shape: Option<Vec<usize>>,
357    mut values: Vec<T>,
358    scalar: fn(T) -> ScalarValue,
359    array: fn(ArrayD<T>) -> ArrayValue,
360) -> Result<Value, Fail> {
361    match shape {
362        None => Ok(Value::Scalar(scalar(values.swap_remove(0)))),
363        Some(shape) => ArrayD::from_shape_vec(IxDyn(&shape), values)
364            .map(|cells| Value::Array(array(cells)))
365            .map_err(|error| Fail::Runtime(format!("array of shape {shape:?}: {error}"))),
366    }
367}

401    #[test]
402    fn a_new_keyword_takes_casatools_type_not_the_width_of_the_numpy_value() {
403        // np.float32(1.25) arrives as a real, a float32 array as an array of
404        // reals: CASA 6.7.6.14 stores TpDouble and TpArrayDouble.
405        assert_eq!(
406            scalar(Data::Floats(vec![1.25])).natural().unwrap(),
407            Value::Scalar(ScalarValue::Float64(1.25))
408        );
409        let reals = array(&[2], Data::Floats(vec![1.0, 2.0])).natural().unwrap();
410        assert!(
411            matches!(reals, Value::Array(ArrayValue::Float64(_))),
412            "{reals:?}"
413        );
414        let complexes = array(&[1], Data::Complexes(vec![Complex64::new(1.0, 2.0)]))
415            .natural()
416            .unwrap();
417        assert!(
418            matches!(complexes, Value::Array(ArrayValue::Complex64(_))),
419            "{complexes:?}"
420        );
421        // An integer is an Int unless it does not fit.
422        assert_eq!(
423            scalar(Data::Ints(vec![7])).natural().unwrap(),
424            Value::Scalar(ScalarValue::Int32(7))
425        );
426        assert_eq!(
427            scalar(Data::Ints(vec![1 << 40])).natural().unwrap(),
428            Value::Scalar(ScalarValue::Int64(1 << 40))
429        );
430    }

432    #[test]
433    fn a_keyword_takes_a_value_of_exactly_its_type() {
434        let double = Value::Scalar(ScalarValue::Float64(1.0));
435        assert_eq!(
436            scalar(Data::Floats(vec![2.5])).like(&double, "k").unwrap(),
437            Value::Scalar(ScalarValue::Float64(2.5))
438        );
439        // casatools gives no value the type Float (or Short, ...), so a
440        // keyword of that type takes none, as defineDataField refuses.
441        for old in [
442            Value::Scalar(ScalarValue::Float32(1.0)),
443            Value::Scalar(ScalarValue::Int16(1)),
444            Value::Scalar(ScalarValue::UInt8(1)),
445            Value::Scalar(ScalarValue::Complex32(Complex32::new(1.0, 0.0))),
446        ] {
447            let value = match old {
448                Value::Scalar(ScalarValue::Float32(_)) => Data::Floats(vec![2.5]),
449                Value::Scalar(ScalarValue::Complex32(_)) => {
450                    Data::Complexes(vec![Complex64::new(1.0, 1.0)])
451                }
452                _ => Data::Ints(vec![2]),
453            };
454            let refused = scalar(value).like(&old, "k").unwrap_err();
455            assert!(
456                matches!(&refused, Fail::Runtime(m) if m.contains("incorrect data type")),
457                "{old:?}: {refused:?}"
458            );
459        }
460        // A 64-bit keyword takes only an integer beyond 32 bits, the
461        // exception to the rule; a small one is an Int to casacore.
462        let wide = Value::Scalar(ScalarValue::Int64(1));
463        assert!(scalar(Data::Ints(vec![7])).like(&wide, "k").is_err());
464        assert_eq!(
465            scalar(Data::Ints(vec![1 << 40])).like(&wide, "k").unwrap(),
466            Value::Scalar(ScalarValue::Int64(1 << 40))
467        );
468        // A 32-bit keyword refuses one that does not fit, where CASA wraps.
469        let int = Value::Scalar(ScalarValue::Int32(1));
470        assert!(scalar(Data::Ints(vec![1 << 40])).like(&int, "k").is_err());
471    }

473    #[test]
474    fn a_string_naming_a_readable_table_is_a_table_reference() {
475        let table = readable_table();
476        let named = format!("Table: {}", table.path().display());
477        assert_eq!(
478            text(&named).natural().unwrap(),
479            Value::TableRef(table.path().display().to_string())
480        );
481        // Not readable, or not so named: an ordinary string.
482        for plain in ["Table: /no/such/table", "plain", "table: x"] {
483            assert_eq!(
484                text(plain).natural().unwrap(),
485                Value::Scalar(ScalarValue::String(plain.to_string()))
486            );
487        }
488        // A string array is never a table.
489        assert!(matches!(
490            array(&[1], Data::Texts(vec![named.clone()]))
491                .natural()
492                .unwrap(),
493            Value::Array(ArrayValue::String(_))
494        ));
495
496        // An existing reference takes another readable one and nothing else.
497        let old = Value::TableRef("././ANTENNA".to_string());
498        assert_eq!(
499            text(&named).like(&old, "k").unwrap(),
500            Value::TableRef(table.path().display().to_string())
501        );
502        assert!(text("plain text").like(&old, "k").is_err());
503        assert!(scalar(Data::Ints(vec![5])).like(&old, "k").is_err());
504        // And an existing string takes a string, not a table.
505        let string = Value::Scalar(ScalarValue::String("x".into()));
506        assert!(text(&named).like(&string, "k").is_err());
507        assert!(text("other").like(&string, "k").is_ok());
508    }
