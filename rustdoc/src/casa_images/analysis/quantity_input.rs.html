10/// Parse a CASA quantity string or a scalar value-and-unit record.
11pub(super) fn parse(value: &ParameterValue) -> Result<(Quantity, String), ImageError> {
12    let invalid = |message: String| ImageError::InvalidMetadata(message);
13    match value {
14        ParameterValue::String(text) => {
15            if text.contains('\0') {
16                return Err(invalid(
17                    "Quantity strings containing NUL are unsupported (#977)".into(),
18                ));
19            }
20            let quantity = Quantity::read(text)
21                .map_err(|_| invalid(format!("Error converting {text} to Quantity")))?;
22            Ok((quantity, format!("\"{text}\"")))
23        }
24        ParameterValue::Table(fields) => {
25            let Some(ParameterValue::String(unit)) = fields.get("unit") else {
26                return Err(invalid("Illegal Quantity record".into()));
27            };
28            if unit.contains('\0') {
29                return Err(invalid(
30                    "Quantity units containing NUL are unsupported (#977)".into(),
31                ));
32            }
33            let (number, ty, rendered) = match fields.get("value") {
34                Some(ParameterValue::Float(value)) => {
35                    (*value, "Double", cpp_number_precision(*value, 6))
36                }
37                Some(ParameterValue::Integer(value)) => {
38                    let value = casa_provider_contracts::casa_variant_integer(*value);
39                    (f64::from(value), "Int", value.to_string())
40                }
41                _ => return Err(invalid("Illegal Quantity record".into())),
42            };
43            let unit_value = casa_types::quanta::casacore::casacore_unit(unit)
44                .map_err(|error| invalid(error.to_string()))?;
45            let quantity = Quantity::with_unit(number, unit_value);
46            let rendered = if fields.len() == 2 {
47                format!("  unit: String \"{unit}\"\n  value: {ty} {rendered}")
48            } else {
49                let record = casa_provider_contracts::parameter_value_to_casacore(value)
50                    .map_err(|error| invalid(error.into()))?;
51                let casa_types::Value::Record(record) = record else {
52                    unreachable!()
53                };
54                super::custom_values::format_record(&record)
55            };
56            Ok((quantity, rendered))
57        }
58        _ => Err(invalid("Quantity must be a record or string".into())),
59    }
60}

62/// Convert a quantity to an image axis unit, treating no unit as the axis unit.
63pub(super) fn in_unit(quantity: &Quantity, unit: &str) -> Result<f64, ImageError> {
64    if quantity.unit().name().is_empty() {
65        return Ok(quantity.value());
66    }
67    let unit = Unit::new(unit).map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
68    casa_types::quanta::casacore::get_conformant_value(quantity, &unit)
69        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
70}

72/// Convert as CASA's coordinate metadata setter does, including its handling
73/// of nonconformant units. An empty unit means the image axis unit.
74pub(super) fn in_axis_unit(quantity: &Quantity, unit: &str) -> Result<f64, ImageError> {
75    if quantity.unit().name().is_empty() {
76        return Ok(quantity.value());
77    }
78    let unit = Unit::new(unit).map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
79    Ok(casa_types::quanta::casacore::get_value(quantity, &unit))
80}
