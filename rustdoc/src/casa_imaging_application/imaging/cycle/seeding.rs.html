2//! The initial major cycle, from an empty model or a start model (#841
3//! Phase 6b, #970 review).
4//!
5//! A start model is the model's first generation, in the units a run
6//! publishes its `.model` in. Under a direction-independent response, or
7//! flat sky, those are the physical units the pass predicts from, and the
8//! initial pass grids the residual of the start model. Under flat noise (a
9//! mosaic's or AW kernel set's `normtype`), `.model` is the physical model
10//! times the PB, and CASA divides a model by the weights' PB just before it
11//! predicts (`SIImageStore::divideModelByWeight`), once `makePSF` has
12//! formed them. Such a run takes CASA's order: the initial pass forms the
13//! PSF, weights and sums with no model, and the start model, made physical
14//! with those weights, is the first generation of a refresh of that pass's
15//! residual. That costs one more pass over the data, for such runs only.
16//! Either way, a run started from another run's model with no minor cycle
17//! gives that run's products (the T1 laws; CASA 6.7.6 does the same on a
18//! 14-pointing mosaic).
19

30/// The response a start model is made physical with before the first
31/// major cycle predicts it: a flat-noise one; `None` without a start
32/// model, or when its units are physical already.
33pub(super) fn reseed_response(
34    start_model: Option<&StartModel>,
35    response: Option<MinorCycleImageResponse>,
36) -> Option<MinorCycleImageResponse> {
37    start_model
38        .and(response)
39        .filter(|response| response.normalization() == ProductNormalization::FlatNoise)
40}

42impl Run<'_> {
43    /// The initial major cycle: data and PSF, from the start model when there
44    /// is one (its residual) and from an empty model otherwise; it writes
45    /// visibilities when it is also final.
46    pub(super) fn initial(&mut self, last: bool) -> Result<Major, ImagingError> {
47        self.measure_row_work()?;
48        match self.reseed {
49            Some(response) => {
50                let unseeded = self.initial_pass(false, false)?;
51                self.reseed(unseeded, response, last)
52            }
53            None => self.initial_pass(self.start_model.is_some(), last),
54        }
55    }
56
57    /// The initial pass, from the start model when `seeded`.
58    fn initial_pass(&mut self, seeded: bool, last: bool) -> Result<Major, ImagingError> {
59        // The model, and a start model's seeding beside it, are admitted
60        // with the pass before either is allocated, so the pass plans its
61        // waves and chunks in what seeding leaves (#970 review).
62        let seeding = if seeded { self.seeding_bytes() } else { 0 };
63        let Admitted {
64            value: plan,
65            reservation: mut pass,
66        } = self.admit_pass(
67            self.initial_modes(),
68            seeded,
69            self.lifecycle.resident_bytes() + self.write_bytes(last) + seeding,
70            self.writes(last),
71        )?;
72        let generation = match &self.start_model {
73            Some(start) if seeded => {
74                // Released once the generation holds the samples.
75                let _seeding = pass.split(seeding);
76                let samples = start.main_samples().map_err(ImagingError::StartModel)?;
77                self.lifecycle.initial_from_start_model(&samples)?
78            }
79            _ => self.lifecycle.initial_empty()?,
80        };
81        let model = self.lifecycle.prepare_final_model(generation, [])?;
82        let cycle = MajorCycle::initial(self.problem, model, self.normal_storage(plan.residency)?)?;
83        let major = self.reconcile(cycle, true, seeded, last, plan)?;
84        self.retain(&major, pass)?;
85        Ok(major)
86    }

88    /// The residual of the start model, made physical with `unseeded`'s
89    /// weights under `response`: a refresh of the initial pass, whose PSF,
90    /// weights and sums carry over.
91    fn reseed(
92        &mut self,
93        unseeded: Major,
94        response: MinorCycleImageResponse,
95        last: bool,
96    ) -> Result<Major, ImagingError> {
97        let start = self
98            .start_model
99            .clone()
100            .expect("a run that reseeds has a start model");
101        // The retained charge covers the normal state, and the empty model,
102        // which the seeded generation replaces at the same size.
103        let (normal_state, empty) = unseeded.completion.into_parts();
104        drop(empty);
105        let seeding = self.seeding_bytes();
106        let Admitted {
107            value: plan,
108            reservation: mut pass,
109        } = self.admit_pass(
110            ModeSet::DATA,
111            true,
112            self.write_bytes(last) + seeding,
113            self.writes(last),
114        )?;
115        let generation = {
116            let _seeding = pass.split(seeding);
117            self.physical_generation(&start, &normal_state, response)?
118        };
119        let model = self.lifecycle.prepare_final_model(generation, [])?;
120        let cycle = MajorCycle::refresh(
121            self.problem,
122            normal_state,
123            model,
124            self.normal_storage(plan.residency)?,
125        )?;
126        let major = self.reconcile(cycle, false, true, last, plan)?;
127        self.retain(&major, pass)?;
128        Ok(major)
129    }

131    /// The first generation from `start`, its flat-noise samples divided by
132    /// the PB of `normal`'s weights where the PB supports them and zero
133    /// elsewhere (`divideModelByWeight`), as the minor cycle maps solved
134    /// values to the physical model.
135    fn physical_generation(
136        &self,
137        start: &StartModel,
138        normal: &FinalNormalState,
139        response: MinorCycleImageResponse,
140    ) -> Result<ModelGeneration, ImagingError> {
141        let [nx, ny] = start.image_size();
142        let polarizations = start.polarizations();
143        let apparent = start.main_samples().map_err(ImagingError::StartModel)?;
144        let units = ImagingError::StartModelUnits;
145        let mut physical = Vec::with_capacity(apparent.len());
146        // Canonical order: spectral plane, polarization, y, x (x fastest);
147        // the normal state's planes are x-major (x · ny + y).
148        for (plane, samples) in apparent.chunks(nx * ny).enumerate() {
149            let weights = normal.read_plane(0, plane / polarizations, plane % polarizations)?;
150            let sensitivity = weights.read_sensitivity()?;
151            let bound = MosaicSensitivity::new(&sensitivity)
152                .and_then(|bound| bound.with_normal_sum_weight(weights.sum_weight()))
153                .map_err(units)?;
154            for y in 0..ny {
155                for x in 0..nx {
156                    let value = bound
157                        .apparent_to_physical(
158                            f64::from(samples[y * nx + x]),
159                            x * ny + y,
160                            response.normalization(),
161                            response.policy(),
162                        )
163                        .map_err(units)?;
164                    physical.push(value as f32);
165                }
166            }
167        }
168        Ok(self.lifecycle.initial_from_start_model(&physical)?)
169    }
170
171    /// What seeding the model from the start model holds besides the model
172    /// ([`StartModel::seeding_bytes`], and its conversion when it is made
173    /// physical); 0 without one.
174    pub(super) fn seeding_bytes(&self) -> u64 {
175        self.start_model.as_ref().map_or(0, |start| {
176            start.seeding_bytes()
177                + if self.reseed.is_some() {
178                    start.conversion_bytes()
179                } else {
180                    0
181                }
182        })
183    }
