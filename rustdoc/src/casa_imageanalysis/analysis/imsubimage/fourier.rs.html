21/// A native real or complex pixel and the transform precision it uses.
22pub trait FourierPixel: ImagePixel {
23    /// Real precision of the transform.
24    type Scalar: ImagePixel + FftScalar;
25
26    /// Represent one pixel in the transform's complex precision.
27    fn complex(self) -> Complex<Self::Scalar>;
28
29    /// Return an amplitude in the same native precision.
30    fn amplitude(value: Complex<Self::Scalar>) -> Self::Scalar;
31
32    /// Return a phase in radians in the same native precision.
33    fn phase(value: Complex<Self::Scalar>) -> Self::Scalar;
34}
35
36macro_rules! fourier_pixels {
37    ($scalar:ty) => {
38        impl FourierPixel for $scalar {
39            type Scalar = $scalar;
40
41            fn complex(self) -> Complex<Self::Scalar> {
42                Complex::new(self, 0.)
43            }
44
45            fn amplitude(value: Complex<Self::Scalar>) -> Self::Scalar {
46                value.norm()
47            }
48
49            fn phase(value: Complex<Self::Scalar>) -> Self::Scalar {
50                value.arg()
51            }
52        }
53
54        impl FourierPixel for Complex<$scalar> {
55            type Scalar = $scalar;
56
57            fn complex(self) -> Complex<Self::Scalar> {
58                self
59            }
60
61            fn amplitude(value: Complex<Self::Scalar>) -> Self::Scalar {
62                value.norm()
63            }
64
65            fn phase(value: Complex<Self::Scalar>) -> Self::Scalar {
66                value.arg()
67            }
68        }
69    };
70}

73fourier_pixels!(f64);

75/// Pixel selection and transform axes for an image Fourier transform.
76pub struct FourierOptions<'a> {
77    /// Optional image region.
78    pub region: Option<SubimageRegion<'a>>,
79    /// An empty list or a single negative value selects both sky axes.
80    pub axes: &'a [i64],
81    /// Additional validity expression on the source.
82    pub mask: Option<&'a str>,
83    /// Broadcast a compatible lower-rank mask.
84    pub stretch: bool,
85    /// Published destinations, checked against each named mask operand.
86    pub destinations: &'a [&'a Path],
87}

102impl<'a> ImageFourier<'a> {
103    /// Resolve the selected image domain and Fourier coordinates.
104    ///
105    /// # Errors
106    /// Returns an error for an unsupported selection, invalid axes, or an
107    /// invalid preserved coordinate model.
108    pub fn new<T: FourierPixel>(
109        image: &dyn ImageInterface<T>,
110        coordinate_record: RecordValue,
111        beam_record: &RecordValue,
112        options: FourierOptions<'a>,
113    ) -> Result<Self, ImageError> {
114        let section = SubimagePlan::new_with_beams(
115            image,
116            Some(coordinate_record),
117            SubimageOptions {
118                region: options.region,
119                dropdeg: false,
120                keepaxes: &[],
121            },
122            Some(beam_record),
123        )?;
124        section.require_single_region()?;
125        let mut axes = if options.axes.is_empty() || matches!(options.axes, [axis] if *axis < 0) {
126            section
127                .coordinates
128                .find_coordinate(CoordinateType::Direction)
129                .and_then(|index| section.coordinates.coordinate_pixel_axes(index))
130                .filter(|axes| axes.len() == 2 && axes.iter().all(Option::is_some))
131                .ok_or_else(|| invalid("Coordinate system does not have a direction coordinate"))?
132                .into_iter()
133                .flatten()
134                .collect()
135        } else {
136            options
137                .axes
138                .iter()
139                .map(|&axis| {
140                    let axis = usize::try_from(axis).map_err(|_| {
141                        invalid("None of the elements of axes may be less than zero")
142                    })?;
143                    if axis >= section.shape.len() {
144                        return Err(invalid(format!(
145                            "Fourier axis {axis} is outside an image of rank {}",
146                            section.shape.len()
147                        )));
148                    }
149                    Ok(axis)
150                })
151                .collect::<Result<Vec<_>, _>>()?
152        };
153        axes.sort_unstable();
154        axes.dedup();
155        let coordinates = section.coordinates.fourier_axes(&section.shape, &axes)?;
156        let beams = &section.image_info().beam_set;
157        if beams.is_multi() {
158            let planes = |kind| {
159                coordinates
160                    .find_coordinate(kind)
161                    .and_then(|index| coordinates.coordinate_pixel_axes(index))
162                    .and_then(|axes| axes.into_iter().flatten().next())
163                    .map_or(1, |axis| section.shape[axis])
164            };
165            if beams.n_channels() != 1 && beams.n_channels() != planes(CoordinateType::Spectral) {
166                return Err(invalid("Number of channels is not consistent"));
167            }
168            if beams.n_stokes() != 1 && beams.n_stokes() != planes(CoordinateType::Stokes) {
169                return Err(invalid("Number of polarizations is not consistent"));
170            }
171            if (0..beams.n_channels()).any(|channel| {
172                (0..beams.n_stokes()).any(|stokes| beams.beam(channel, stokes).is_null())
173            }) {
174                return Err(invalid(
175                    "At least one of the beams in the beam set is null and thus invalid",
176                ));
177            }
178        }
179        let coordinate_record = section.transformed_coordinate_record(&coordinates, &axes)?;
180        Ok(Self {
181            section,
182            axes,
183            coordinates,
184            coordinate_record,
185            options,
186        })
187    }

209    /// Transform the selected region to complex scratch storage.
210    ///
211    /// # Errors
212    /// The mask is incompatible or overlaps an output, or FFT planning,
213    /// allocation, or image I/O fails.
214    pub fn transform<T: FourierPixel>(
215        &self,
216        image: &dyn ImageInterface<T>,
217    ) -> Result<TempImage<Complex<T::Scalar>>, ImageError>
218    where
219        Complex<T::Scalar>: ImagePixel,
220    {
221        let selection = &self.section.selection;
222        let reader = PixelSections {
223            image,
224            selection,
225            mask: PreparedMask::new_for_outputs(
226                self.options.mask,
227                image.shape(),
228                image.coordinates(),
229                self.options.stretch,
230                self.options.destinations.iter().copied(),
231            )?,
232        };
233        let unit = vec![1; self.shape().len()];
234        let selected = StridedSection {
235            start: &selection.start,
236            shape: self.shape(),
237            stride: selection.stride.as_deref().unwrap_or(&unit),
238        };
239        let mut scratch = TempImage::<Complex<T::Scalar>>::new(
240            self.shape().to_vec(),
241            self.coordinates.clone(),
242            TempStoragePolicy::Auto {
243                memory_budget_bytes: 16 * 1024 * 1024,
244                scratch: ScratchSpace::SystemTemp,
245            },
246        )?;
247        scratch.set_pixel_cache_budget(16 * 1024 * 1024)?;
248        for tile in selected.cursors(bounded_read_cursor(selected.dense_shape(selected.shape))?) {
249            let tile = tile?;
250            let (pixels, validity) = reader.read(
251                &selected.source_start(&tile.position),
252                &tile.shape,
253                selected.stride,
254            )?;
255            let mut pixels = pixels.mapv(T::complex);
256            if let Some(mask) = validity {
257                set_mask(&mut scratch, &mask, &tile.position)?;
258                Zip::from(&mut pixels).and(&mask).for_each(|value, &valid| {
259                    if !valid {
260                        *value = Complex::default();
261                    }
262                });
263            }
264            scratch.put_slice(&pixels, &tile.position)?;
265        }
266        for &axis in &self.axes {
267            self.transform_axis::<T::Scalar>(&mut scratch, axis)?;
268        }
269        Ok(scratch)
270    }
271
272    fn transform_axis<T: FftScalar>(
273        &self,
274        scratch: &mut TempImage<Complex<T>>,
275        axis: usize,
276    ) -> Result<(), ImageError>
277    where
278        Complex<T>: ImagePixel,
279    {
280        let length = self.shape()[axis];
281        if length == 1 {
282            return Ok(());
283        }
284        let fft_error = |error: casa_fft::FftError| invalid(error.to_string());
285        let mut fft = Fft2::<T>::with_threads([length, 1], 1)
286            .map_err(fft_error)?
287            .with_estimated_plan();
288        let mut lane_buffer = vec![Complex::default(); length];
289        let mut cursor = vec![1; self.shape().len()];
290        cursor[axis] = length;
291        let mut remaining = (128 * 1024 / length).max(1);
292        for (i, &extent) in self.shape().iter().enumerate() {
293            if i != axis {
294                cursor[i] = extent.min(remaining);
295                remaining /= cursor[i];
296            }
297        }
298        for tile in TraversalCursorIter::new(self.shape().to_vec(), cursor, TraversalSpec::tiles())
299        {
300            let tile = tile?;
301            let mut pixels = scratch.get_slice(&tile.position, &tile.shape)?;
302            for mut lane in pixels.lanes_mut(Axis(axis)) {
303                for (index, entry) in lane_buffer.iter_mut().enumerate() {
304                    *entry = lane[(index + length / 2) % length];
305                }
306                fft.transform(&mut lane_buffer, false).map_err(fft_error)?;
307                for (index, part) in lane.iter_mut().enumerate() {
308                    *part = lane_buffer[(index + length.div_ceil(2)) % length];
309                }
310            }
311            scratch.put_slice(&pixels, &tile.position)?;
312        }
313        Ok(())
314    }

316    /// Copy one product's values, mask, and metadata from the shared transform.
317    ///
318    /// # Errors
319    /// The destination shape is wrong or image I/O fails.
320    pub fn write_product<S: ImagePixel, C: ImagePixel, D: ImagePixel>(
321        &self,
322        source: &dyn ImageInterface<S>,
323        scratch: &dyn ImageInterface<C>,
324        output: &mut dyn MutableImageInterface<D>,
325        phase: bool,
326        component: impl Fn(C) -> D,
327    ) -> Result<(), ImageError> {
328        self.section.copy_metadata(source, output)?;
329        let unit = if phase {
330            "rad"
331        } else {
332            match source.units() {
333                "Jy/beam" | "Jy/pixel" => "Jy",
334                "Jy" if self.section.image_info().beam_set.is_empty() => "Jy/pixel",
335                "Jy" => "Jy/beam",
336                other => other,
337            }
338        };
339        output.set_units_string(unit.to_owned())?;
340        let cursor = bounded_read_cursor(self.shape().to_vec())?;
341        let unit_stride = vec![1; self.shape().len()];
342        for tile in TraversalCursorIter::new(self.shape().to_vec(), cursor, TraversalSpec::tiles())
343        {
344            let tile = tile?;
345            let values = scratch.get_slice(&tile.position, &tile.shape, &unit_stride)?;
346            output.put_slice(&values.mapv(&component), &tile.position)?;
347            if let Some(mask) = scratch.get_mask_slice(&tile.position, &tile.shape, &unit_stride)? {
348                set_mask(output, &mask, &tile.position)?;
349            }
350        }
351        Ok(())
352    }
353}

355fn set_mask<T: ImagePixel>(
356    image: &mut dyn MutableImageInterface<T>,
357    mask: &casa_core::ArrayD<bool>,
358    start: &[usize],
359) -> Result<(), ImageError> {
360    if image.default_mask_name().is_none() {
361        image.create_filled_mask("mask0", true)?;
362        image.set_default_mask("mask0")?;
363    }
364    image.put_mask_slice("mask0", mask, start)
