3//! Unit-consistent algebra for two-dimensional elliptical Gaussians.
4//!
5//! Axes may use any common angular unit. Position angles use radians in the
6//! CASA convention: east from north, modulo pi.
7
8use std::{error::Error, fmt};
9
10/// One elliptical Gaussian described by full-width-at-half-maximum axes.
11#[derive(Debug, Clone, Copy, PartialEq)]
12pub struct EllipticalGaussian {
13    /// Major-axis FWHM in a caller-chosen angular unit.
14    pub major: f64,
15    /// Minor-axis FWHM in the same unit as [`Self::major`].
16    pub minor: f64,
17    /// CASA position angle, east from north, in radians.
18    pub position_angle: f64,
19}
20
21impl EllipticalGaussian {
22    /// Construct one numerical Gaussian descriptor.
23    #[must_use]
24    pub const fn new(major: f64, minor: f64, position_angle: f64) -> Self {
25        Self {
26            major,
27            minor,
28            position_angle,
29        }
30    }
31
32    /// Return the Gaussian area in squared axis units.
33    #[must_use]
34    pub fn area(self) -> f64 {
35        std::f64::consts::PI / (4.0 * 2.0_f64.ln()) * self.major * self.minor
36    }
37}

39/// Failure of elliptical-Gaussian enclosure or deconvolution algebra.
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub enum EllipticalGaussianError {
42    /// No usable Gaussian was supplied.
43    EmptySet,
44    /// A Gaussian had non-finite, non-positive, or inverted axes.
45    InvalidGaussian,
46    /// The requested target cannot be produced by convolving the source.
47    TargetSmallerThanSource,
48}
49
50impl fmt::Display for EllipticalGaussianError {
51    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
52        formatter.write_str(match self {
53            Self::EmptySet => "elliptical Gaussian set is empty",
54            Self::InvalidGaussian => {
55                "elliptical Gaussian requires finite positive major >= minor axes"
56            }
57            Self::TargetSmallerThanSource => "target Gaussian is smaller than the source Gaussian",
58        })
59    }
60}
61
62impl Error for EllipticalGaussianError {}

64/// Wrap a CASA beam position angle into `(-pi/2, pi/2]`.
65#[must_use]
66pub fn normalize_position_angle(angle: f64) -> f64 {
67    let mut wrapped = normalize_angle_pi(angle);
68    if wrapped <= -std::f64::consts::FRAC_PI_2 {
69        wrapped += std::f64::consts::PI;
70    } else if wrapped > std::f64::consts::FRAC_PI_2 {
71        wrapped -= std::f64::consts::PI;
72    }
73    wrapped
74}

76/// Compute CASA's minimum-area common enclosing Gaussian.
77///
78/// This is the numerical core of `CasaImageBeamSet::getCommonBeam()`.
79///
80/// # Errors
81///
82/// Returns [`EllipticalGaussianError::EmptySet`] for no beams and
83/// [`EllipticalGaussianError::InvalidGaussian`] for invalid beam parameters.
84pub fn common_enclosing_gaussian(
85    gaussians: &[EllipticalGaussian],
86) -> Result<EllipticalGaussian, EllipticalGaussianError> {
87    if gaussians.is_empty() {
88        return Err(EllipticalGaussianError::EmptySet);
89    }
90    if gaussians.iter().any(|gaussian| !valid(*gaussian)) {
91        return Err(EllipticalGaussianError::InvalidGaussian);
92    }
93    if gaussians.iter().all(|gaussian| *gaussian == gaussians[0]) {
94        return Ok(gaussians[0]);
95    }
96    common_enclosing_iterative(gaussians)
97}

99/// Compute the Gaussian that convolves `source` into `target`.
100///
101/// Returns `Ok(None)` when target and source are effectively identical.
102///
103/// # Errors
104///
105/// Returns [`EllipticalGaussianError::TargetSmallerThanSource`] when the
106/// covariance difference is not positive semidefinite.
107pub fn deconvolving_gaussian(
108    target: EllipticalGaussian,
109    source: EllipticalGaussian,
110) -> Result<Option<EllipticalGaussian>, EllipticalGaussianError> {
111    if !valid(target) || !valid(source) {
112        return Err(EllipticalGaussianError::InvalidGaussian);
113    }
114    let lhs = gaussian_covariance(target);
115    let rhs = gaussian_covariance(source);
116    let delta = [
117        [lhs[0][0] - rhs[0][0], lhs[0][1] - rhs[0][1]],
118        [lhs[1][0] - rhs[1][0], lhs[1][1] - rhs[1][1]],
119    ];
120    let trace = delta[0][0] + delta[1][1];
121    let determinant = delta[0][0] * delta[1][1] - delta[0][1] * delta[1][0];
122    let discriminant = ((trace * trace) / 4.0 - determinant).max(0.0).sqrt();
123    let lambda_major = trace / 2.0 + discriminant;
124    let lambda_minor = trace / 2.0 - discriminant;
125
126    // These are the existing CASA-compatible absolute covariance tolerances
127    // for radian beam axes: one treats numerical zero as a point source and
128    // the other rejects a genuinely negative covariance eigenvalue.
129    if lambda_minor < -1.0e-12 || lambda_major < -1.0e-12 {
130        return Err(EllipticalGaussianError::TargetSmallerThanSource);
131    }
132    if lambda_major <= 1.0e-24 && lambda_minor <= 1.0e-24 {
133        return Ok(None);
134    }
135    let x_axis_angle =
136        if delta[0][1].abs() <= 1.0e-18 && (delta[0][0] - delta[1][1]).abs() <= 1.0e-18 {
137            0.0
138        } else {
139            0.5 * (2.0 * delta[0][1]).atan2(delta[0][0] - delta[1][1])
140        };
141    Ok(Some(from_x_axis(
142        lambda_major.max(0.0).sqrt(),
143        lambda_minor.max(0.0).sqrt(),
144        x_axis_angle,
145    )))
146}

148fn valid(gaussian: EllipticalGaussian) -> bool {
149    gaussian.major.is_finite()
150        && gaussian.minor.is_finite()
151        && gaussian.position_angle.is_finite()
152        && gaussian.major > 0.0
153        && gaussian.minor > 0.0
154        && gaussian.major >= gaussian.minor
155}

157fn common_enclosing_iterative(
158    gaussians: &[EllipticalGaussian],
159) -> Result<EllipticalGaussian, EllipticalGaussianError> {
160    let mut gaussians = gaussians.to_vec();
161    loop {
162        let (max_index, &max_gaussian) = gaussians
163            .iter()
164            .enumerate()
165            .max_by(|(_, lhs), (_, rhs)| lhs.area().total_cmp(&rhs.area()))
166            .ok_or(EllipticalGaussianError::EmptySet)?;
167
168        // CASA skips every beam equal to the largest (`*iBeam != maxBeam`),
169        // not only the largest itself: a duplicate (Stokes planes share a
170        // beam) can fail the enclosure test by rounding, and enclosing a
171        // beam with itself gives no ellipse.
172        let problem = gaussians
173            .iter()
174            .copied()
175            .enumerate()
176            .filter(|(index, gaussian)| *index != max_index && *gaussian != max_gaussian)
177            .map(|(_, gaussian)| gaussian)
178            .find(|gaussian| !encloses(max_gaussian, *gaussian));
179        let Some(problem) = problem else {
180            return Ok(max_gaussian);
181        };
182
183        let relative_angle = normalize_position_angle(problem.position_angle)
184            - normalize_position_angle(max_gaussian.position_angle);
185        if (normalize_angle_pi(relative_angle).abs() - std::f64::consts::FRAC_PI_2).abs() <= 1.0e-12
186        {
187            let max_has_major = max_gaussian.major >= problem.major;
188            return Ok(EllipticalGaussian::new(
189                if max_has_major {
190                    max_gaussian.major
191                } else {
192                    problem.major
193                },
194                if max_has_major {
195                    problem.major
196                } else {
197                    max_gaussian.major
198                },
199                normalize_position_angle(if max_has_major {
200                    max_gaussian.position_angle
201                } else {
202                    problem.position_angle
203                }),
204            ));
205        }
206
207        let equal_area_axis = (max_gaussian.major * max_gaussian.minor).sqrt();
208        let x_scale = equal_area_axis / max_gaussian.major;
209        let y_scale = equal_area_axis / max_gaussian.minor;
210        let (problem_major, _, problem_angle) = transform_ellipse_by_scaling(
211            problem.major,
212            problem.minor,
213            relative_angle,
214            x_scale,
215            y_scale,
216        );
217        let (mut major, mut minor, common_angle) = transform_ellipse_by_scaling(
218            problem_major,
219            equal_area_axis,
220            problem_angle,
221            1.0 / x_scale,
222            1.0 / y_scale,
223        );
224        let position_angle = common_angle + normalize_position_angle(max_gaussian.position_angle);
225        // A degenerate construction would never enclose the pair: refuse it
226        // rather than grow it for ever.
227        if !(major.is_finite() && minor.is_finite() && position_angle.is_finite())
228            || major <= 0.0
229            || minor <= 0.0
230        {
231            return Err(EllipticalGaussianError::InvalidGaussian);
232        }
233        let mut enclosing =
234            EllipticalGaussian::new(major, minor, normalize_position_angle(position_angle));
235        while !(encloses(enclosing, max_gaussian) && encloses(enclosing, problem)) {
236            major *= 1.001;
237            minor *= 1.001;
238            enclosing =
239                EllipticalGaussian::new(major, minor, normalize_position_angle(position_angle));
240        }
241
242        gaussians[max_index] = enclosing;
243    }
244}

246fn from_x_axis(major: f64, minor: f64, x_axis_angle: f64) -> EllipticalGaussian {
247    EllipticalGaussian::new(
248        major,
249        minor,
250        normalize_position_angle(x_axis_angle - std::f64::consts::FRAC_PI_2),
251    )
252}

254fn normalize_angle_pi(angle: f64) -> f64 {
255    let mut wrapped = angle.rem_euclid(2.0 * std::f64::consts::PI);
256    if wrapped > std::f64::consts::PI {
257        wrapped -= 2.0 * std::f64::consts::PI;
258    }
259    wrapped
260}

262fn encloses(enclosing: EllipticalGaussian, other: EllipticalGaussian) -> bool {
263    let theta_source = normalize_position_angle(enclosing.position_angle);
264    let theta_beam = normalize_position_angle(other.position_angle);
265    let alpha = (enclosing.major * theta_source.cos()).powi(2)
266        + (enclosing.minor * theta_source.sin()).powi(2)
267        - (other.major * theta_beam.cos()).powi(2)
268        - (other.minor * theta_beam.sin()).powi(2);
269    let beta = (enclosing.major * theta_source.sin()).powi(2)
270        + (enclosing.minor * theta_source.cos()).powi(2)
271        - (other.major * theta_beam.sin()).powi(2)
272        - (other.minor * theta_beam.cos()).powi(2);
273    let gamma = 2.0
274        * (((enclosing.minor * enclosing.minor) - (enclosing.major * enclosing.major))
275            * theta_source.sin()
276            * theta_source.cos()
277            - ((other.minor * other.minor) - (other.major * other.major))
278                * theta_beam.sin()
279                * theta_beam.cos());
280    let sum = alpha + beta;
281    let difference = ((alpha - beta).powi(2) + gamma.powi(2)).sqrt();
282    alpha >= 0.0 && beta >= 0.0 && sum >= difference
283}

285/// Return the symmetric FWHM-squared covariance form in image x/y axes.
286///
287/// The common Gaussian-to-standard-deviation factor is deliberately omitted:
288/// it cancels in enclosure and deconvolution and preserves CASA arithmetic.
289#[must_use]
290pub fn gaussian_covariance(gaussian: EllipticalGaussian) -> [[f64; 2]; 2] {
291    let angle = normalize_position_angle(gaussian.position_angle) + std::f64::consts::FRAC_PI_2;
292    let cos = angle.cos();
293    let sin = angle.sin();
294    let major2 = gaussian.major * gaussian.major;
295    let minor2 = gaussian.minor * gaussian.minor;
296    [
297        [
298            cos * cos * major2 + sin * sin * minor2,
299            cos * sin * (major2 - minor2),
300        ],
301        [
302            cos * sin * (major2 - minor2),
303            sin * sin * major2 + cos * cos * minor2,
304        ],
305    ]
306}

308fn transform_ellipse_by_scaling(
309    major: f64,
310    minor: f64,
311    position_angle: f64,
312    x_scale: f64,
313    y_scale: f64,
314) -> (f64, f64, f64) {
315    let cos = position_angle.cos();
316    let sin = position_angle.sin();
317    let cos2 = cos * cos;
318    let sin2 = sin * sin;
319    let major2 = major * major;
320    let minor2 = minor * minor;
321    let a = cos2 / major2 + sin2 / minor2;
322    let b = -2.0 * cos * sin * (1.0 / major2 - 1.0 / minor2);
323    let c = sin2 / major2 + cos2 / minor2;
324    let x_scale2 = x_scale * x_scale;
325    let y_scale2 = y_scale * y_scale;
326    let r = a / x_scale2;
327    let s = b * b / (4.0 * x_scale2 * y_scale2);
328    let t = c / y_scale2;
329    let difference = r - t;
330    let discriminant = difference * difference + 4.0 * s;
331    let signed = discriminant.sqrt() * difference.abs();
332    let j1 = (signed + discriminant) / discriminant / 2.0;
333    let j2 = (-signed + discriminant) / discriminant / 2.0;
334    let k1 = (j1 * r + j1 * t - t) / (2.0 * j1 - 1.0);
335    let k2 = (j2 * r + j2 * t - t) / (2.0 * j2 - 1.0);
336    let axis1 = (1.0 / k1).sqrt();
337    let axis2 = (1.0 / k2).sqrt();
338
339    // CASA's `_transformEllipseByScaling` takes the ellipse for a circle
340    // only when its axes are exactly equal, whatever their unit. (Its circle
341    // has the reciprocal of the axis, `sqrt(k1)`, against its evident intent;
342    // the circle has the axis.)
343    if axis1 == axis2 {
344        return (axis1, axis1, 0.0);
345    }
346    if axis1 > axis2 {
347        (
348            axis1,
349            axis2,
350            if position_angle >= 0.0 {
351                j1.sqrt().acos()
352            } else {
353                -j1.sqrt().acos()
354            },
355        )
356    } else {
357        (
358            axis2,
359            axis1,
360            if position_angle >= 0.0 {
361                j2.sqrt().acos()
362            } else {
363                -j2.sqrt().acos()
364            },
365        )
366    }
367}
