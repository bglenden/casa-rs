57pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
58    let fringe = task.required_text("ff_table")?;
59    let request = PccorRequest {
60        vis: task.input_path("vis")?,
61        output: task.path("pccor_caltable")?,
62        refant: task.required_text("refant")?,
63        timerange: task.required_text("timerange")?,
64        scan: task.required_text("scan")?,
65        spw: task.required_text("spw")?,
66        antenna: task.required_text("antenna")?,
67        cable: task.flag("cablecal_correction")?,
68        fringe_table: (fringe != "none").then(|| PathBuf::from(fringe)),
69    };
70    let (result, rows_written) = execute(&request).map_err(|error| match error {
71        PccorError::Selection(message) => TaskFailure::execution(message).at("selection"),
72        PccorError::FringeTable(message) => TaskFailure::execution(message).at("ff_table"),
73        PccorError::Index(message) => TaskFailure::execution(message).at("pccor_index"),
74        _ => TaskFailure::execution(error.to_string()),
75    })?;
76    TaskSuccess::new(
77        result,
78        &Summary {
79            output_table: request.output,
80            rows_written,
81        },
82    )
83}

85fn execute(request: &PccorRequest) -> Result<(CasaValue, usize), PccorError> {
86    let open = crate::timing::Phase::start("pccor_select_read");
87    let ms = MeasurementSet::open(&request.vis)?;
88    let selection = Selection::read(&ms, request)?;
89    let pulses = pulse::read(&request.vis, &selection)?;
90    drop(open);
91
92    let fit = crate::timing::Phase::start("pccor_reference_fringe");
93    let fringes = fringe::read_or_solve(&ms, request, &selection)?;
94    drop(fit);
95    if selection.windows.is_empty() {
96        return Err(PccorError::Index("list index out of range".into()));
97    }
98
99    let align = crate::timing::Phase::start("pccor_align");
100    let mut antennas = Vec::with_capacity(selection.antennas.len());
101    let mut seen_antennas = BTreeSet::new();
102    let mut table_rows = Vec::new();
103    for &antenna in &selection.antennas {
104        let first_antenna = seen_antennas.insert(antenna);
105        let mut windows = Vec::with_capacity(selection.windows.len());
106        let mut seen_windows = BTreeSet::new();
107        for &window in &selection.windows {
108            let pair = (antenna, window);
109            if request.cable && pulses[&pair].missing_single_tone {
110                return Err(PccorError::Selection(
111                    "non-broadcastable output operand with shape (2,1) doesn't match the broadcast shape (2,0)".into(),
112                ));
113            }
114            let options = Alignment {
115                pulse: &pulses[&pair],
116                reference: &pulses[&(selection.reference, window)],
117                fringe: &fringes[&pair],
118                cable: request.cable,
119            };
120            let aligned = options.align(&selection);
121            table_rows.extend(aligned.iter().filter_map(|row| {
122                row.scan.map(|scan| table::Row {
123                    antenna,
124                    window,
125                    time: row.time,
126                    field: scan.field,
127                    scan: scan.number,
128                    phase: row.phase,
129                    delay: row.delay,
130                })
131            }));
132            if first_antenna && seen_windows.insert(window) {
133                windows.push((format!("spw_{window}"), options.record(aligned)));
134            }
135        }
136        if first_antenna {
137            let name = &selection.antenna_names[antenna as usize];
138            antennas.push((format!("ant_{name}"), CasaValue::dict(windows)));
139        }
140    }
141    drop(align);
142    let publish = crate::timing::Phase::start("pccor_write");
143    let rows_written = table_rows.len();
144    table::write(&ms, &request.output, selection.reference, table_rows)?;
145    drop(publish);
146    Ok((CasaValue::dict(antennas), rows_written))
147}

149struct Alignment<'a> {
150    pulse: &'a PulseSeries,
151    reference: &'a PulseSeries,
152    fringe: &'a FringeSolution,
153    cable: bool,
154}
155
156struct Aligned<'a> {
157    time: f64,
158    scan: Option<&'a selection::Scan>,
159    phase: [f64; 2],
160    delay: [f64; 2],
161}
162
163impl Alignment<'_> {
164    fn align<'a>(&self, selection: &'a Selection) -> Vec<Aligned<'a>> {
165        let fringe = self.fringe;
166        let fallback = match self.pulse.samples.as_slice() {
167            [] => true,
168            [only] => only.delay[0] == 0.,
169            _ => false,
170        };
171        if fallback {
172            return vec![Aligned {
173                time: fringe.time,
174                scan: selection.scan_at(fringe.time),
175                phase: fringe.phase.map(wrap_phase),
176                delay: fringe.delay,
177            }];
178        }
179        self.pulse
180            .samples
181            .iter()
182            .map(|sample| {
183                let cable = if self.cable { sample.cable } else { 0. };
184                let phase = [0, 1].map(|hand| {
185                    wrap_phase(
186                        fringe.phase[hand] + sample.phase[hand] - self.pulse.mean_phase[hand][0],
187                    )
188                });
189                let delay = [0, 1].map(|hand| {
190                    sample.delay[hand] + fringe.delay[hand]
191                        - (self.pulse.mean_delay[hand] - self.reference.mean_delay[hand])
192                        + cable
193                });
194                Aligned {
195                    time: sample.time,
196                    scan: selection.scan_at(sample.time),
197                    phase,
198                    delay,
199                }
200            })
201            .collect()
202    }
203
204    fn record(&self, aligned: Vec<Aligned<'_>>) -> CasaValue {
205        let vector = CasaValue::float64_vector;
206        CasaValue::dict([
207            (
208                "time",
209                vector(aligned.iter().map(|row| row.time).collect::<Vec<_>>()),
210            ),
211            (
212                "scan",
213                vector(
214                    aligned
215                        .iter()
216                        .map(|row| row.scan.map_or(f64::NAN, |scan| f64::from(scan.number)))
217                        .collect(),
218                ),
219            ),
220            (
221                "field_id",
222                vector(
223                    aligned
224                        .iter()
225                        .map(|row| row.scan.map_or(f64::NAN, |scan| f64::from(scan.field)))
226                        .collect(),
227                ),
228            ),
229            (
230                "phase",
231                hand_array(aligned.len(), aligned.iter().flat_map(|row| row.phase)),
232            ),
233            (
234                "delay",
235                hand_array(aligned.len(), aligned.iter().flat_map(|row| row.delay)),
236            ),
237            (
238                "ff_phase",
239                vector(self.fringe.phase.map(wrap_phase).to_vec()),
240            ),
241            ("ff_delay", vector(self.fringe.delay.to_vec())),
242            ("pc_mean_phase", tone_array(&self.pulse.mean_phase)),
243            ("pc_mean_delay", vector(self.pulse.mean_delay.to_vec())),
244            ("pc_refant_phase", tone_array(&self.reference.mean_phase)),
245            (
246                "pc_refant_delay",
247                vector(self.reference.mean_delay.to_vec()),
248            ),
249        ])
250    }
251}
252
253fn wrap_phase(phase: f64) -> f64 {
254    (phase + PI).rem_euclid(TAU) - PI
255}
256
257fn hand_array(columns: usize, values: impl Iterator<Item = f64>) -> CasaValue {
258    CasaValue::FloatArray {
259        dtype: FloatDtype::Float64,
260        shape: vec![2, columns],
261        data: values.map(CasaFloat).collect(),
262    }
263}
264
265fn tone_array(hands: &[Vec<f64>; 2]) -> CasaValue {
266    hand_array(
267        hands[0].len(),
268        hands[0]
269            .iter()
270            .zip(&hands[1])
271            .flat_map(|(&first, &second)| [first, second]),
272    )
273}
