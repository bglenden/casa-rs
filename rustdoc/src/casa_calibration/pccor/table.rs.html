15pub(super) struct Row {
16    pub(super) antenna: i32,
17    pub(super) window: i32,
18    pub(super) time: f64,
19    pub(super) field: i32,
20    pub(super) scan: i32,
21    pub(super) phase: [f64; 2],
22    pub(super) delay: [f64; 2],
23}
24
25pub(super) fn write(
26    ms: &MeasurementSet,
27    destination: &Path,
28    reference: i32,
29    mut rows: Vec<Row>,
30) -> Result<(), PccorError> {
31    rows.sort_by(|first, second| {
32        first
33            .time
34            .total_cmp(&second.time)
35            .then_with(|| (first.window, first.antenna).cmp(&(second.window, second.antenna)))
36    });
37    let mut writer = prepare_empty_calibration_table(
38        ms,
39        destination,
40        "Float",
41        "Fringe Jones",
42        Some(SingleChannelFrequency::MiddleChannel),
43    )
44    .map_err(PccorError::Execution)?;
45    // PCCOR writes time, SPW, antenna order into the empty NewCalTable.
46    writer.preserve_row_order();
47    let scalar = |name, number| RecordField::new(name, Value::Scalar(ScalarValue::Int32(number)));
48    let double = |name, number| RecordField::new(name, Value::Scalar(ScalarValue::Float64(number)));
49    let array =
50        |name, values| RecordField::new(name, Value::Array(ArrayValue::Float32(shaped(values))));
51    for row in rows {
52        let mut parameters = [0_f32; 8];
53        for (hand, (phase, delay)) in row.phase.into_iter().zip(row.delay).enumerate() {
54            parameters[4 * hand] = phase as f32;
55            parameters[4 * hand + 1] = (delay * 1e9) as f32;
56        }
57        writer
58            .append(RecordValue::new(vec![
59                double("TIME", row.time),
60                scalar("FIELD_ID", row.field),
61                scalar("SPECTRAL_WINDOW_ID", row.window),
62                scalar("ANTENNA1", row.antenna),
63                scalar("ANTENNA2", reference),
64                double("INTERVAL", 0.),
65                scalar("SCAN_NUMBER", row.scan),
66                scalar("OBSERVATION_ID", 0),
67                array("FPARAM", parameters),
68                RecordField::new("FLAG", Value::Array(ArrayValue::Bool(shaped([false; 8])))),
69                array("SNR", [999.; 8]),
70                array("WEIGHT", [1.; 8]),
71            ]))
72            .map_err(|error| PccorError::Execution(error.to_string()))?;
73    }
74    let _lock = destination
75        .exists()
76        .then(|| TableWriteLock::acquire_to_reread(destination, 1))
77        .transpose()
78        .map_err(|error| PccorError::Execution(error.to_string()))?;
79    writer
80        .save_main()
81        .map_err(|error| PccorError::Execution(error.to_string()))
82}
83
84fn shaped<T>(values: [T; 8]) -> ArrayD<T> {
85    ArrayD::from_shape_vec(IxDyn(&[8, 1]).f(), Vec::from(values)).expect("eight fringe parameters")
