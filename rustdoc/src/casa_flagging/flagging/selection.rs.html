10pub(super) fn flagging_channels(
11    ms: &MeasurementSet,
12    request: &FlagDataRequest,
13    resolved_ranges: BTreeMap<i32, ChannelSelection>,
14) -> Result<Option<BTreeMap<i32, ChannelSelection>>, FlaggingError> {
15    let Some(text) = request.spw.as_deref() else {
16        return Ok(None);
17    };
18    let retain_overlaps = request.casa_row_selection && request.mode == FlagDataMode::Extend;
19    let ranges =
20        if request.casa_row_selection && !retain_overlaps && request.spw == request.selection.spw {
21            Ok(resolved_ranges)
22        } else if request.casa_row_selection {
23            ms.resolve_flagdata_spw_selector(text).and_then(|spw| {
24                casa_ms::selection::flagdata_channel_ranges(ms, &spw, retain_overlaps)
25            })
26        } else {
27            parse_channel_selections(text)
28        }
29        .map_err(|source| FlaggingError::MutateFlags {
30            path: display_ms_path(ms),
31            reason: source.to_string(),
32        })?;
33    check_channel_selections(ms, &ranges)?;
34    Ok(Some(ranges))
35}

44pub(super) fn time_on_globally_selected_rows(
45    text: &str,
46    rows: &[ResolvedMsSelectionRow],
47) -> casa_ms::MsResult<TimeSelection> {
48    let Some(first) = rows.first() else {
49        return Ok(TimeSelection::all());
50    };
51    // Exposure affects the global predicate and the width in getTimeList(),
52    // but only the endpoints are used by the list agent.
53    let defaults = |row: &ResolvedMsSelectionRow| TimeDefaults {
54        time_mjd_s: row.time_mjd_seconds,
55        exposure_s: 0.0,
56    };
57    let first_unflagged = rows.iter().find(|row| !row.flag_row).unwrap_or(first);
58    resolve_time_selector(
59        text,
60        TimeSelectorDefaults {
61            first_term: defaults(first_unflagged),
62            later_terms: defaults(first),
63        },
64    )
65}

67pub(super) fn selected_rows(
68    ms: &MeasurementSet,
69    request: &FlagDataRequest,
70    time_selection: FlagTimeSelection,
71) -> Result<SelectedFlagRows, FlaggingError> {
72    let selection_error = |source| FlaggingError::MutateFlags {
73        path: display_ms_path(ms),
74        reason: format!("select rows: {source}"),
75    };
76    let timerange = request
77        .selection
78        .timerange
79        .as_deref()
80        .filter(|_| request.selection.selectdata)
81        .map(flagdata_c_string)
82        .filter(|text| !text.is_empty());
83    let agent_default_row = match time_selection {
84        FlagTimeSelection::Agent {
85            initial_default_row,
86        } => Some(initial_default_row),
87        _ => None,
88    };
89    let (rows, channel_ranges, missing_rows) = if request.casa_row_selection {
90        let resolved = resolve_flagdata_selection_with_system_budget(
91            ms,
92            &request.selection,
93            agent_default_row,
94        )
95        .map_err(selection_error)?;
96        let missing_rows = resolved.selected_rows.is_empty();
97        // CASA parses a one-command list's agent on the globally selected
98        // table, so omitted dates can differ from the initial global parse.
99        let agent_time = timerange
100            .filter(|_| time_selection == FlagTimeSelection::GlobalAndAgent)
101            .map(|text| time_on_globally_selected_rows(text, &resolved.selected_rows))
102            .transpose()
103            .map_err(selection_error)?;
104        (
105            resolved
106                .selected_rows
107                .into_iter()
108                .filter(|row| {
109                    agent_time
110                        .as_ref()
111                        .is_none_or(|time| time.holds_agent_time(row.time_mjd_seconds))
112                })
113                .map(|row| row.row_index)
114                .collect(),
115            resolved.channel_selections,
116            missing_rows,
117        )
118    } else {
119        let rows = resolve_row_indices_with_system_budget(ms, &request.selection)
120            .map_err(selection_error)?;
121        let missing_rows = rows.is_empty();
122        (rows, BTreeMap::new(), missing_rows)
123    };
124    let field = request
125        .selection
126        .field
127        .as_deref()
128        .filter(|text| !text.trim().is_empty());
129    let explicit_spw_filter = missing_rows
130        && request.casa_row_selection
131        && request
132            .selection
133            .spw
134            .as_deref()
135            .map(|text| {
136                ms.resolve_flagdata_spw_selector(text)
137                    .map(|spw| spw.rows_restricted())
138            })
139            .transpose()
140            .map_err(selection_error)?
141            .unwrap_or(false);
142    let explicit_row_filter = field.is_some()
143        || explicit_spw_filter
144        || (request.casa_row_selection
145            && [
146                request.selection.scan.as_deref(),
147                request.selection.antenna.as_deref(),
148                request.selection.timerange.as_deref(),
149                request.selection.observation.as_deref(),
150                request.selection.array.as_deref(),
151                request.selection.feed.as_deref(),
152                request.selection.intent.as_deref(),
153                request.selection.uvrange.as_deref(),
154            ]
155            .into_iter()
156            .flatten()
157            .any(|text| !text.is_empty() && !text.starts_with('\0')));
158    if missing_rows && request.selection.selectdata && explicit_row_filter {
159        // An empty result can mean no matching FIELD metadata or simply no
160        // selected MAIN rows. Keep separate diagnostics for the two cases.
161        if let Some(field) = field {
162            ms.resolve_field_selector(field).map_err(selection_error)?;
163        }
164        // An agent can have valid local time intervals that match no
165        // records. Keep it in the list, including its empty summary.
166        if !(request.casa_row_selection && timerange.is_some() && agent_default_row.is_some()) {
167            return Err(FlaggingError::MutateFlags {
168                path: display_ms_path(ms),
169                reason: "MSSelectionNullSelection : The selected table has zero rows.".to_string(),
170            });
171        }
172    }
173    Ok(SelectedFlagRows {
174        rows,
175        channel_ranges,
176    })
177}

179pub(super) fn flag_read_plan(
180    ms: &MeasurementSet,
181    row_count: usize,
182    primitive_bytes: usize,
183    include_data_description: bool,
184) -> Result<MsReadPlan, FlaggingError> {
185    let path = display_ms_path(ms);
186    let maximum_elements = maximum_flag_elements(ms)?;
187    let requested_bytes_per_row = maximum_elements
188        .checked_mul(primitive_bytes.saturating_add(2))
189        .and_then(|bytes| bytes.checked_add(usize::from(include_data_description) * 4))
190        .ok_or_else(|| FlaggingError::MutateFlags {
191            path: path.clone(),
192            reason: "flag row byte accounting overflowed".to_string(),
193        })?;
194    let budget = MsSelectionIoBudget::from_system_memory(1, requested_bytes_per_row, None)
195        .map_err(|source| FlaggingError::MutateFlags {
196            path: path.clone(),
197            reason: source.to_string(),
198        })?;
199    MsReadPlan::new(row_count, budget).map_err(|source| FlaggingError::MutateFlags {
200        path,
201        reason: source.to_string(),
202    })
203}

205pub(super) fn maximum_flag_elements(ms: &MeasurementSet) -> Result<usize, FlaggingError> {
206    let path = display_ms_path(ms);
207    let data_description = ms
208        .data_description()
209        .map_err(|source| FlaggingError::MutateFlags {
210            path: path.clone(),
211            reason: format!("open DATA_DESCRIPTION: {source}"),
212        })?;
213    let spectral_window = ms
214        .spectral_window()
215        .map_err(|source| FlaggingError::MutateFlags {
216            path: path.clone(),
217            reason: format!("open SPECTRAL_WINDOW: {source}"),
218        })?;
219    let polarization = ms
220        .polarization()
221        .map_err(|source| FlaggingError::MutateFlags {
222            path: path.clone(),
223            reason: format!("open POLARIZATION: {source}"),
224        })?;
225    let mut maximum_elements = 0usize;
226    for row in 0..data_description.row_count() {
227        let spw_id = data_description.spectral_window_id(row).map_err(|source| {
228            FlaggingError::MutateFlags {
229                path: path.clone(),
230                reason: format!("read DATA_DESCRIPTION.SPECTRAL_WINDOW_ID row {row}: {source}"),
231            }
232        })?;
233        let polarization_id =
234            data_description
235                .polarization_id(row)
236                .map_err(|source| FlaggingError::MutateFlags {
237                    path: path.clone(),
238                    reason: format!("read DATA_DESCRIPTION.POLARIZATION_ID row {row}: {source}"),
239                })?;
240        let channel_count = usize::try_from(spectral_window.num_chan(spw_id as usize).map_err(
241            |source| FlaggingError::MutateFlags {
242                path: path.clone(),
243                reason: format!("read SPECTRAL_WINDOW.NUM_CHAN row {spw_id}: {source}"),
244            },
245        )?)
246        .map_err(|_| FlaggingError::MutateFlags {
247            path: path.clone(),
248            reason: format!("SPECTRAL_WINDOW.NUM_CHAN row {spw_id} is negative"),
249        })?;
250        let correlation_count = usize::try_from(
251            polarization
252                .num_corr(polarization_id as usize)
253                .map_err(|source| FlaggingError::MutateFlags {
254                    path: path.clone(),
255                    reason: format!("read POLARIZATION.NUM_CORR row {polarization_id}: {source}"),
256                })?,
257        )
258        .map_err(|_| FlaggingError::MutateFlags {
259            path: path.clone(),
260            reason: format!("POLARIZATION.NUM_CORR row {polarization_id} is negative"),
261        })?;
262        maximum_elements =
263            maximum_elements.max(channel_count.checked_mul(correlation_count).ok_or_else(
264                || FlaggingError::MutateFlags {
265                    path: path.clone(),
266                    reason: "flag row shape byte accounting overflowed".to_string(),
267                },
268            )?);
269    }
270    Ok(maximum_elements)
271}

280pub(super) fn parse_channel_selections(
281    value: &str,
282) -> Result<BTreeMap<i32, ChannelSelection>, MsError> {
283    let mut selections = BTreeMap::new();
284    for selector in parse_spw_selector(value)? {
285        if let Some(channels) = selector.channels {
286            selections.insert(selector.spw_id, channels);
287        }
288    }
289    Ok(selections)
290}

292pub(super) fn data_description_spw_map(
293    ms: &MeasurementSet,
294) -> Result<BTreeMap<i32, i32>, FlaggingError> {
295    let table = ms
296        .data_description()
297        .map_err(|source| FlaggingError::MutateFlags {
298            path: display_ms_path(ms),
299            reason: format!("open DATA_DESCRIPTION: {source}"),
300        })?;
301    let mut result = BTreeMap::new();
302    for row in 0..table.row_count() {
303        result.insert(
304            row as i32,
305            table
306                .spectral_window_id(row)
307                .map_err(|source| FlaggingError::MutateFlags {
308                    path: display_ms_path(ms),
309                    reason: format!("read DATA_DESCRIPTION row {row}: {source}"),
310                })?,
311        );
312    }
313    Ok(result)
