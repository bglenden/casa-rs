2//! The minor cycle of one major-cycle completion: its planes read from the
3//! normal state as deconvolution views, solved on the worker team, and
4//! their components returned as model terms.
5//!
6//! A minor cycle has two halves around the run's [`Controller`]:
7//! [`prepare_minor_cycle`] forms the masks and measures every plane (CASA's
8//! `initminorcycle`), and [`run_minor_cycle`] cleans every plane under the
9//! controls the controller derived from those measurements.
10//!
11//! Each half reads its planes from the normal state itself, so a plane is
12//! read twice per cycle. A paged cube is not held in memory between the
13//! halves; only the planes in flight on the team are.
14//!
15//! [`Controller`]: casa_imaging_deconvolution::Controller
16

41/// A failure of the minor cycle.
42#[derive(Debug, thiserror::Error)]
43pub enum MinorCycleRunError {
44    /// The reconstruction masks could not be formed.
45    #[error("reconstruction mask: {0}")]
46    Mask(#[from] MaskError),
47    /// A normal-state plane could not be read.
48    #[error("normal state: {0}")]
49    Normal(#[from] SpectralOperatorError),
50    /// The model support could not be read or a term is out of range.
51    #[error("model: {0}")]
52    Model(#[from] ModelLifecycleError),
53    /// The direction-dependent response could not convert a value.
54    #[error("image response: {0}")]
55    Response(#[from] casa_imaging_reconstruction::ImageResponseError),
56    /// The solver failed.
57    #[error(transparent)]
58    Solver(#[from] casa_imaging_deconvolution::Error),
59    /// The selected algorithm has no minor cycle.
60    #[error("{0:?} has no minor cycle")]
61    Algorithm(ReconstructionAlgorithm),
62}

178/// Form the masks of `completion` and measure every plane: its peak
179/// residual, noise and PSF sidelobe (CASA's `initminorcycle`).
180///
181/// # Errors
182///
183/// When a mask, a plane or a PSF cannot be read or measured.
184pub fn prepare_minor_cycle(
185    completion: &MajorCycleCompletion,
186    mask_plans: &ImageDomainReconstructionMaskPlans,
187    setup: &MinorCycleSetup,
188    cache: &mut PsfCache,
189    team: &WorkerTeam,
190) -> Result<PreparedMinorCycle, MinorCycleRunError> {
191    let normal = completion.normal_state();
192    let base = completion.final_model();
193    let taylor = is_taylor(normal);
194    let keys = plane_keys(normal, taylor);
195    // The automatic mask (one-channel continuum only) smooths by the
196    // primary PSF's beam; a cube's first channel may hold no data.
197    let beam = if setup.automask {
198        let primary = keys[0];
199        summarise(normal, taylor, cache, primary)?;
200        Some(automask_beam(&cache.summaries[&primary]))
201    } else {
202        None
203    };
204    let (masks, auto_masks) = mask_plans.materialize(base, normal, beam)?.into_parts();
205    let inputs = Inputs {
206        normal,
207        base,
208        masks: &masks,
209        setup,
210        taylor,
211    };
212    let mut slots = keys
213        .iter()
214        .map(|key| (*key, None::<Measured>))
215        .collect::<Vec<_>>();
216    let missing = keys
217        .iter()
218        .filter(|key| !cache.summaries.contains_key(key))
219        .count();
220    let shared = &*cache;
221    team.for_each_mut(&mut slots, |_, (key, out)| {
222        let Some(plane) = load(&inputs, *key)? else {
223            return Ok(());
224        };
225        let summary = match shared.summaries.get(key) {
226            Some(summary) => *summary,
227            None => PsfSummary::new(&plane.psf[0], plane.shape)?,
228        };
229        let statistics = PlaneStatistics::measure(
230            &plane.residual[0],
231            &plane.support,
232            &plane.valid,
233            setup.nsigma,
234            setup.automask,
235        );
236        *out = Some(Measured {
237            statistics,
238            summary,
239        });
240        Ok::<_, MinorCycleRunError>(())
241    })?;
242    let mut planes = Vec::with_capacity(keys.len());
243    let mut sidelobes = Vec::with_capacity(keys.len());
244    for (key, measured) in slots {
245        let Some(Measured {
246            statistics,
247            summary,
248        }) = measured
249        else {
250            continue;
251        };
252        sidelobes.push(summary.sidelobe());
253        cache.summaries.entry(key).or_insert(summary);
254        planes.push((key, statistics));
255    }
256    // Each image domain is one CASA image store.
257    let mut statistics = ResidualStatistics::empty();
258    let domains = planes
259        .iter()
260        .map(|(key, _)| key.domain)
261        .collect::<BTreeSet<_>>();
262    for domain in domains {
263        statistics.include_store(
264            planes
265                .iter()
266                .zip(&sidelobes)
267                .filter(|((key, _), _)| key.domain == domain)
268                .map(|((_, plane), sidelobe)| (plane, *sidelobe)),
269            setup.nsigma,
270        );
271    }
272    tracing::debug!(
273        planes = planes.len(),
274        psf_fits = missing,
275        "minor cycle prepared"
276    );
277    Ok(PreparedMinorCycle {
278        masks,
279        auto_masks,
280        planes,
281        statistics,
282    })
283}

292/// Clean every plane of `completion` under `controls`.
293///
294/// Planes are independent: several run concurrently on `team`, each on one
295/// thread, and their terms are merged in model order, so the result does
296/// not depend on the worker count. A single plane spreads Clark's sparse
297/// refresh across the team instead.
298///
299/// # Errors
300///
301/// When a plane cannot be read or its solver fails.
302pub fn run_minor_cycle(
303    prepared: PreparedMinorCycle,
304    completion: &MajorCycleCompletion,
305    setup: &MinorCycleSetup,
306    controls: &CycleControls,
307    cache: &mut PsfCache,
308    team: &WorkerTeam,
309) -> Result<MinorCycleOutcome, MinorCycleRunError> {
310    let PreparedMinorCycle {
311        masks,
312        auto_masks,
313        planes,
314        ..
315    } = prepared;
316    let normal = completion.normal_state();
317    let inputs = Inputs {
318        normal,
319        base: completion.final_model(),
320        masks: &masks,
321        setup,
322        taylor: is_taylor(normal),
323    };
324    let mut results = Vec::with_capacity(planes.len());
325    if let [(key, statistics)] = planes.as_slice() {
326        let summary = cache.summaries[key];
327        let clark = &mut cache.clark;
328        results.push(team.install(|| solve(&inputs, *key, &summary, statistics, controls, clark))?);
329    } else {
330        let mut slots = planes
331            .iter()
332            .map(|(key, statistics)| Slot {
333                key: *key,
334                statistics: *statistics,
335                solved: None,
336            })
337            .collect::<Vec<_>>();
338        let shared = &*cache;
339        team.for_each_mut(&mut slots, |_, slot| {
340            let summary = shared.summaries[&slot.key];
341            slot.solved = Some(solve(
342                &inputs,
343                slot.key,
344                &summary,
345                &slot.statistics,
346                controls,
347                &mut None,
348            )?);
349            Ok::<_, MinorCycleRunError>(())
350        })?;
351        results.extend(
352            slots
353                .into_iter()
354                .map(|slot| slot.solved.expect("every plane slot ran")),
355        );
356    }
357    let mut summary = MinorCycleSummary {
358        iterations: 0,
359        components: 0,
360        absolute_flux: 0.0,
361        start_peak: 0.0,
362        peak: 0.0,
363        noise_rms: None,
364        stops: Vec::with_capacity(results.len()),
365        refreshes: 0,
366        trace: Vec::new(),
367    };
368    let mut terms = Vec::new();
369    for ((_, statistics), solved) in planes.iter().zip(results) {
370        let outcome = &solved.outcome;
371        summary.iterations += outcome.iterations;
372        summary.components += outcome.components;
373        summary.absolute_flux += outcome.absolute_flux;
374        summary.start_peak = summary.start_peak.max(outcome.start_peak);
375        if outcome.stop != PlaneStop::ZeroMask {
376            summary.peak = summary.peak.max(outcome.peak);
377        }
378        if let Some(noise) = statistics.noise {
379            summary.noise_rms = Some(
380                summary
381                    .noise_rms
382                    .map_or(noise.rms, |rms| rms.max(noise.rms)),
383            );
384        }
385        summary.stops.push(outcome.stop);
386        summary.refreshes += outcome.refreshes;
387        let room = TRACE - summary.trace.len();
388        summary.trace.extend(solved.trace.into_iter().take(room));
389        terms.extend(solved.terms);
390    }
391    terms.sort_unstable_by_key(|(index, _)| *index);
392    Ok(MinorCycleOutcome {
393        terms: terms.into_iter().map(|(_, term)| term).collect(),
394        masks,
395        auto_masks,
396        summary,
397    })
398}

400/// Whether `normal` holds one Taylor family (decided once per cycle half).
401fn is_taylor(normal: &FinalNormalState) -> bool {
402    normal.catalog() == NormalStateCatalog::UnnormalizedTaylorBlockV1
403}

405/// The planes a minor cycle cleans, in CASA's order within each image
406/// domain: one Taylor family, every channel of every polarization of a
407/// cube or continuum plane, or each polarization of every image field.
408fn plane_keys(normal: &FinalNormalState, taylor: bool) -> Vec<PlaneKey> {
409    let first = normal.slab().core_range().start;
410    if taylor {
411        return vec![PlaneKey {
412            domain: 0,
413            channel: first,
414            polarization: 0,
415        }];
416    }
417    let mut keys = Vec::new();
418    for polarization in 0..normal.polarization_count() {
419        for domain in 0..normal.domain_count() {
420            for channel in first..first + normal.channel_count() {
421                keys.push(PlaneKey {
422                    domain,
423                    channel,
424                    polarization,
425                });
426            }
427        }
428    }
429    keys
430}

432/// Measure the PSF of `key` into the cache.
433fn summarise(
434    normal: &FinalNormalState,
435    taylor: bool,
436    cache: &mut PsfCache,
437    key: PlaneKey,
438) -> Result<(), MinorCycleRunError> {
439    if cache.summaries.contains_key(&key) {
440        return Ok(());
441    }
442    let shape = plane_shape(normal, key);
443    let psf = read_psf(normal, taylor, key)?;
444    let peak = psf_peak_value(&psf, shape)?;
445    cache
446        .summaries
447        .insert(key, PsfSummary::new(&normalise(psf, peak), shape)?);
448    Ok(())
449}

462fn plane_shape(normal: &FinalNormalState, key: PlaneKey) -> PlaneShape {
463    let [nx, ny] = normal
464        .domain_shape(key.domain)
465        .expect("plane keys name existing domains");
466    PlaneShape::new(nx, ny)

469/// Term-0 PSF of `key`, unnormalised.
470fn read_psf(
471    normal: &FinalNormalState,
472    taylor: bool,
473    key: PlaneKey,
474) -> Result<Vec<f64>, MinorCycleRunError> {
475    if taylor {
476        let window = normal.read_window(normal.slab().core_range())?;
477        let moment = window
478            .normal_moment(0)
479            .expect("a Taylor family has its zeroth moment");
480        return Ok(moment
481            .normal_approximation()
482            .iter()
483            .map(|value| value.re)
484            .collect());
485    }
486    let plane = normal.read_reconstruction_plane(key.domain, key.channel, key.polarization)?;
487    Ok(plane
488        .normal_approximation()
489        .iter()
490        .map(|value| value.re)
491        .collect())
492}

505/// A direction-dependent response bound to one plane's weights: what turns
506/// a solved value back into the physical model (`divideModelByWeight`).
507struct ResponseBinding<'a> {
508    sensitivity: Cow<'a, [f64]>,
509    response: MinorCycleImageResponse,
510    sum_weight: f64,
511}

513/// One plane's residual and PSF terms as the solver sees them.
514struct PlaneTerms<'a> {
515    residual: Vec<Vec<f64>>,
516    psf: Vec<Vec<f64>>,
517    response: Option<ResponseBinding<'a>>,
518}

520/// One plane read as a deconvolution view.
521struct PlaneData<'a> {
522    shape: PlaneShape,
523    residual: Vec<Vec<f64>>,
524    psf: Vec<Vec<f64>>,
525    support: Support,
526    valid: Support,
527    response: Option<ResponseBinding<'a>>,
528}
529
530impl PlaneData<'_> {
531    /// The plane's conversion of solved values to the physical model,
532    /// bound once (binding scans the sensitivity plane).
533    fn physical(&self) -> Result<Physical<'_>, MinorCycleRunError> {
534        Ok(match &self.response {
535            None => Physical::Identity,
536            Some(binding) => Physical::Mosaic(
537                MosaicSensitivity::new(&binding.sensitivity)?
538                    .with_normal_sum_weight(binding.sum_weight)?,
539                &binding.response,
540            ),
541        })
542    }
543}
544
545/// Solved values to the physical model (`divideModelByWeight`).
546enum Physical<'a> {
547    Identity,
548    Mosaic(MosaicSensitivity<'a>, &'a MinorCycleImageResponse),
549}
550
551impl Physical<'_> {
552    fn apply(&self, value: f64, index: usize) -> Result<f64, MinorCycleRunError> {
553        match self {
554            Self::Identity => Ok(value),
555            Self::Mosaic(sensitivity, response) => Ok(sensitivity.apparent_to_physical(
556                value,
557                index,
558                response.normalization(),
559                response.policy(),
560            )?),
561        }
562    }
563}

565/// Read `key` as a view: residual and PSF normalised by the term-0 PSF peak
566/// (or the residual in CASA's flat-noise or flat-sky units under a
567/// direction-dependent response), and the support. `None` for a plane with
568/// no valid data.
569fn load<'a>(
570    inputs: &Inputs<'a>,
571    key: PlaneKey,
572) -> Result<Option<PlaneData<'a>>, MinorCycleRunError> {
573    let Inputs {
574        normal,
575        base,
576        masks,
577        setup,
578        taylor,
579    } = *inputs;
580    let shape = plane_shape(normal, key);
581    let validity = if taylor {
582        normal.support_validity()
583    } else {
584        normal.domain_channel_validity(
585            key.domain,
586            key.channel - normal.slab().core_range().start,
587            key.polarization,
588        )
589    };
590    if validity != Some(SpectralChannelValidity::Valid) {
591        return Ok(None);
592    }
593    let coefficients = if taylor {
594        0..normal.coefficient_term_count()
595    } else {
596        key.channel..key.channel + 1
597    };
598    let mut valid = vec![true; shape.len()];
599    for coefficient in coefficients {
600        let samples = base.read_plane(key.domain, coefficient, key.polarization)?;
601        for (index, supported) in valid.iter_mut().enumerate() {
602            let [x, y] = shape.pixel(index);
603            *supported &= samples[y * shape.nx + x].support() == ModelSupport::Valid;
604        }
605    }
606    let mask = masks
607        .get(key.domain)
608        .expect("a mask for every image domain");
609    let support = mask
610        .support()
611        .iter()
612        .zip(&valid)
613        .map(|(masked, valid)| *masked && *valid)
614        .collect();
615    let terms = if taylor {
616        read_taylor(normal, setup)?
617    } else {
618        read_single(normal, setup, key)?
619    };
620    Ok(Some(PlaneData {
621        shape,
622        residual: terms.residual,
623        psf: terms.psf,
624        support: Support::new(shape, support),
625        valid: Support::new(shape, valid),
626        response: terms.response,
627    }))
628}

630/// One plane's residual and PSF. Without a response both are divided by the
631/// PSF peak; with one the residual takes CASA's normalisation
632/// (`SIImageStore::divideResidualByWeight`: by the published sum of weights,
633/// and the weight image by the PSF gridding's) and the components convert
634/// back with `divideModelByWeight`. CASA normalises every (polarization,
635/// channel) plane by that plane's own sums and weight image, so a cube
636/// channel reads its own; a paged cube reads only this plane.
637fn read_single<'a>(
638    normal: &'a FinalNormalState,
639    setup: &MinorCycleSetup,
640    key: PlaneKey,
641) -> Result<PlaneTerms<'a>, MinorCycleRunError> {
642    let plane = normal.read_reconstruction_plane(key.domain, key.channel, key.polarization)?;
643    let shape = plane_shape(normal, key);
644    let psf = plane
645        .normal_approximation()
646        .iter()
647        .map(|value| value.re)
648        .collect::<Vec<_>>();
649    let peak = psf_peak_value(&psf, shape)?;
650    let raw = plane.residual();
651    let Some(response) = setup.response else {
652        return Ok(PlaneTerms {
653            residual: vec![raw.iter().map(|value| value.re / peak).collect()],
654            psf: vec![normalise(psf, peak)],
655            response: None,
656        });
657    };
658    let weights = normal.read_plane(key.domain, key.channel, key.polarization)?;
659    let sensitivity = weights.read_sensitivity()?;
660    let normal_weight = weights.sum_weight();
661    let published_weight = weights.published_sum_weight();
662    let bound = MosaicSensitivity::new(&sensitivity)?.with_normal_sum_weight(normal_weight)?;
663    let residual = raw
664        .iter()
665        .enumerate()
666        .map(|(index, value)| {
667            bound.normalize_weighted_residual_sample(
668                value.re,
669                index,
670                response.normalization(),
671                published_weight,
672                response.policy(),
673            )
674        })
675        .collect::<Result<Vec<_>, _>>()?;
676    Ok(PlaneTerms {
677        residual: vec![residual],
678        psf: vec![normalise(psf, peak)],
679        response: Some(ResponseBinding {
680            sensitivity,
681            response,
682            sum_weight: normal_weight,
683        }),
684    })
685}

687/// The Taylor family's residual terms and `2·N_t − 1` PSF moments, scaled
688/// alike: by the zeroth moment's peak, or under a response with the residuals
689/// in CASA's units and the moments over the zeroth moment's sum weight.
690fn read_taylor(
691    normal: &FinalNormalState,
692    setup: &MinorCycleSetup,
693) -> Result<PlaneTerms<'static>, MinorCycleRunError> {
694    let window = normal.read_window(normal.slab().core_range())?;
695    let [nx, ny] = normal.shape();
696    let shape = PlaneShape::new(nx, ny);
697    let principal = window
698        .normal_moment(0)
699        .expect("a Taylor family has its zeroth moment");
700    let moments = (0..normal.normal_moment_count())
701        .map(|moment| {
702            window
703                .normal_moment(moment)
704                .expect("every normal moment is present")
705                .normal_approximation()
706                .iter()
707                .map(|value| value.re)
708                .collect::<Vec<_>>()
709        })
710        .collect::<Vec<_>>();
711    let raw = (0..normal.coefficient_term_count()).map(|term| {
712        window
713            .coefficient_term(term)
714            .expect("every coefficient term is present")
715            .residual()
716    });
717    let Some(response) = setup.response else {
718        let peak = psf_peak_value(&moments[0], shape)?;
719        return Ok(PlaneTerms {
720            residual: raw
721                .map(|term| term.iter().map(|value| value.re / peak).collect())
722                .collect(),
723            psf: moments
724                .into_iter()
725                .map(|moment| normalise(moment, peak))
726                .collect(),
727            response: None,
728        });
729    };
730    let normal_weight = principal.sum_weight();
731    let published_weight = window.published_sum_weights()[0];
732    let sensitivity = principal.sensitivity().to_vec();
733    let bound = MosaicSensitivity::new(&sensitivity)?.with_normal_sum_weight(normal_weight)?;
734    let residual = raw
735        .map(|term| {
736            term.iter()
737                .enumerate()
738                .map(|(index, value)| {
739                    bound.normalize_weighted_residual_sample(
740                        value.re,
741                        index,
742                        response.normalization(),
743                        published_weight,
744                        response.policy(),
745                    )
746                })
747                .collect::<Result<Vec<_>, _>>()
748        })
749        .collect::<Result<Vec<_>, _>>()?;
750    Ok(PlaneTerms {
751        residual,
752        psf: moments
753            .into_iter()
754            .map(|moment| normalise(moment, normal_weight))
755            .collect(),
756        response: Some(ResponseBinding {
757            sensitivity: Cow::Owned(sensitivity),
758            response,
759            sum_weight: normal_weight,
760        }),
761    })
762}
763
764fn psf_peak_value(psf: &[f64], shape: PlaneShape) -> Result<f64, MinorCycleRunError> {
765    let peak = casa_imaging_deconvolution::psf_peak(psf, shape)
766        .map(|index| psf[index])
767        .filter(|value| value.is_finite() && *value > 0.0)
768        .ok_or(casa_imaging_deconvolution::Error::PsfPeak)?;
769    Ok(peak)
770}
771
772fn normalise(mut plane: Vec<f64>, scale: f64) -> Vec<f64> {
773    for value in &mut plane {
774        *value /= scale;
775    }
776    plane
777}

779/// A plane's solve: its outcome, model terms keyed by model order, and the
780/// first component records.
781struct Solved {
782    outcome: PlaneOutcome,
783    terms: Vec<(usize, ModelDeltaTerm)>,
784    trace: Vec<TracedComponent>,
785}
786
787/// Read and clean one plane. Clark takes `clark`'s refresh, when it serves
788/// this PSF, and leaves its own there for the next cycle.
789fn solve(
790    inputs: &Inputs<'_>,
791    key: PlaneKey,
792    summary: &PsfSummary,
793    statistics: &PlaneStatistics,
794    controls: &CycleControls,
795    clark: &mut Option<LinearRefresh>,
796) -> Result<Solved, MinorCycleRunError> {
797    let setup = inputs.setup;
798    let plane = load(inputs, key)?.expect("prepared planes have valid data");
799    let view = MinorCycleView {
800        shape: plane.shape,
801        residual: &plane.residual,
802        psf: &plane.psf,
803        summary,
804        support: &plane.support,
805    };
806    let outcome = match &setup.algorithm {
807        ReconstructionAlgorithm::Hogbom => clean(
808            &Hogbom::new(setup.accounting == HogbomIterationAccounting::CasaInclusive),
809            &view,
810            controls,
811            statistics,
812        )?,
813        ReconstructionAlgorithm::Clark => {
814            let solver = Clark::new(clark.take());
815            let outcome = clean(&solver, &view, controls, statistics);
816            *clark = solver.into_refresh();
817            outcome?
818        }
819        ReconstructionAlgorithm::Multiscale {
820            scales_px,
821            small_scale_bias,
822        } => clean(
823            &Multiscale::new(scales_px.clone(), *small_scale_bias),
824            &view,
825            controls,
826            statistics,
827        )?,
828        ReconstructionAlgorithm::Mtmfs {
829            scales_px,
830            small_scale_bias,
831        } => clean(
832            &Taylor::new(plane.residual.len(), scales_px.clone(), *small_scale_bias),
833            &view,
834            controls,
835            statistics,
836        )?,
837        algorithm => return Err(MinorCycleRunError::Algorithm(algorithm.clone())),
838    };
839    let terms = model_terms(&outcome.delta, &plane, inputs, key)?;
840    let trace = trace(&outcome.trace, &plane, inputs, key);
841    Ok(Solved {
842        outcome,
843        terms,
844        trace,
845    })
846}
847
848/// [`run_plane`] with the cycle's trace length.
849fn clean<S: Solver>(
850    solver: &S,
851    view: &MinorCycleView<'_>,
852    controls: &CycleControls,
853    statistics: &PlaneStatistics,
854) -> Result<PlaneOutcome, MinorCycleRunError> {
855    Ok(run_plane(solver, view, controls, statistics, TRACE)?)
856}

858/// The model coefficient a plane's term `term` goes to: the Taylor term, or
859/// the channel of a cube plane.
860const fn coefficient(inputs: &Inputs<'_>, key: PlaneKey, term: usize) -> usize {
861    if inputs.taylor { term } else { key.channel }
862}
863
864/// The plane's components as model terms keyed by model order.
865fn model_terms(
866    delta: &Delta,
867    plane: &PlaneData,
868    inputs: &Inputs<'_>,
869    key: PlaneKey,
870) -> Result<Vec<(usize, ModelDeltaTerm)>, MinorCycleRunError> {
871    let mut terms = Vec::new();
872    let physical = plane.physical()?;
873    for term in 0..delta.term_count() {
874        for (index, flux) in delta.term(term) {
875            let value = physical.apply(flux, index)?;
876            if value == 0.0 {
877                continue;
878            }
879            let cell = ModelCell::new(
880                key.domain,
881                coefficient(inputs, key, term),
882                key.polarization,
883                plane.shape.pixel(index),
884            );
885            let flat = inputs
886                .base
887                .shape()
888                .flat_index(cell)
889                .ok_or(ModelLifecycleError::CellOutsideShape)?;
890            terms.push((
891                flat,
892                ModelDeltaTerm::new(
893                    cell,
894                    ModelValue::new(value).map_err(ModelLifecycleError::Contract)?,
895                ),
896            ));
897        }
898    }
899    Ok(terms)
900}
901
902fn trace(
903    components: &[Component],
904    plane: &PlaneData,
905    inputs: &Inputs<'_>,
906    key: PlaneKey,
907) -> Vec<TracedComponent> {
908    let scales = match &inputs.setup.algorithm {
909        ReconstructionAlgorithm::Multiscale { scales_px, .. }
910        | ReconstructionAlgorithm::Mtmfs { scales_px, .. } => scales_px.as_slice(),
911        _ => &[],
912    };
913    components
914        .iter()
915        .map(|component| TracedComponent {
916            cell: ModelCell::new(
917                key.domain,
918                coefficient(inputs, key, component.term),
919                key.polarization,
920                plane.shape.pixel(component.index),
921            ),
922            flux: component.flux,
923            scale_px: scales.get(component.scale).copied().unwrap_or(0.0),
924        })
925        .collect()
926}
