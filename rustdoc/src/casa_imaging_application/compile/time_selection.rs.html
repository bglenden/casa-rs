3//! CASA's `timerange` resolved against a MeasurementSet, as `MSSelection`
4//! resolves it for `tclean` (casacore `aaf72eb7`).
5//!
6//! `SynthesisImagerVi2::selectData` always sets a field and a spw
7//! expression (`*` when empty), and `MSSelection::toTableExprNode` parses
8//! the time expression last, with the other expressions as its
9//! `otherTens`. A partial time (`09:30:00`, `>10:00`) therefore takes its
10//! missing date and time from the first MAIN row, in physical order, that
11//! the other expressions select, and later comma terms from row 0
12//! (`MSTimeParse::getDefaults`; [`casa_ms::TimeSelectorDefaults`]). The
13//! grammar itself is the MeasurementSet owner's port
14//! ([`casa_ms::TimeSelector`]); this module finds those rows and keeps the
15//! resolved conditions, which the survey and the run test at each time.
16

29const SPEED_OF_LIGHT_M_S: f64 = 299_792_458.0;

31/// The times `selection`'s `timerange` selects in its `store`, with the
32/// other selections already resolved (`windows`, `rows`, `states`).
33///
34/// # Errors
35///
36/// A time expression CASA refuses; a `timerange` of an MSv4 processing set
37/// ([`PrepareError::TimeRangeWithoutMainRows`]); an empty MAIN table, which
38/// has no logical row zero; a MAIN column that cannot be read.
39pub(super) fn resolve_times(
40    selection: &MeasurementSetSelection,
41    store: &OpenedStore,
42    windows: &WindowChoice,
43    (rows, states): (&RowSelection, &BTreeSet<StateId>),
44) -> Result<TimeSelection, PrepareError> {
45    let Some(text) = selection.timerange.as_deref() else {
46        return Ok(TimeSelection::All);
47    };
48    let selector = TimeSelector::parse(text)?;
49    if selector.selects_all() {
50        return Ok(TimeSelection::All);
51    }
52    let observation = store.msv2().ok_or(PrepareError::TimeRangeWithoutMainRows)?;
53    if observation.measurement_set().row_count() == 0 {
54        return Err(casa_ms::MsError::InvalidInput(
55            "MSTimeParse: No logical \"row zero\" found for time selection".to_string(),
56        )
57        .into());
58    }
59    let others = OtherSelections::new(observation, store, windows, (rows, states))?;
60    let first = observation
61        .first_main_row(|keys| others.hold(keys))?
62        .unwrap_or(0);
63    let defaults = |row: usize| -> Result<TimeDefaults, PrepareError> {
64        let (time_mjd_s, exposure_s) = observation.main_row_time_exposure(row)?;
65        Ok(TimeDefaults {
66            time_mjd_s,
67            exposure_s,
68        })
69    };
70    let resolved = selector.resolve(TimeSelectorDefaults {
71        first_term: defaults(first)?,
72        later_terms: defaults(0)?,
73    })?;
74    Ok(TimeSelection::AnyOf(
75        resolved
76            .terms()
77            .iter()
78            .map(|term| condition(term.condition))
79            .collect(),
80    ))
81}
82
83/// The model's copy of one resolved condition.
84fn condition(condition: casa_ms::TimeCondition) -> TimeCondition {
85    match condition {
86        casa_ms::TimeCondition::Near {
87            time_s,
88            tolerance_s,
89        } => TimeCondition::Near {
90            time_s,
91            tolerance_s,
92        },
93        casa_ms::TimeCondition::AtOrAfter { time_s } => TimeCondition::AtOrAfter { time_s },
94        casa_ms::TimeCondition::AtOrBefore { time_s } => TimeCondition::AtOrBefore { time_s },
95        casa_ms::TimeCondition::Within { lower_s, upper_s } => {
96            TimeCondition::Within { lower_s, upper_s }
97        }
98        casa_ms::TimeCondition::EdgeInclusive {
99            lower_s,
100            upper_s,
101            edge_s,
102        } => TimeCondition::EdgeInclusive {
103            lower_s,
104            upper_s,
105            edge_s,
106        },
107    }
108}

110/// The selections other than time, as tests of one MAIN row: CASA's
111/// `otherTens`.
112pub(super) struct OtherSelections<'a> {
113    observation: &'a Msv2Observation,
114    rows: &'a RowSelection,
115    states: &'a BTreeSet<StateId>,
116    /// Each `DATA_DESCRIPTION` row's selection by spw (or `ddid`), and its
117    /// window's reference wavelength when a UV distance is in wavelengths.
118    descriptions: Vec<(bool, Option<f64>)>,
119}
120
121impl<'a> OtherSelections<'a> {
122    /// The tests of `rows`, `states` and `windows` on `observation`'s
123    /// MAIN rows.
124    ///
125    /// # Errors
126    ///
127    /// When DATA_DESCRIPTION cannot be read.
128    pub(super) fn new(
129        observation: &'a Msv2Observation,
130        store: &OpenedStore,
131        windows: &WindowChoice,
132        (rows, states): (&'a RowSelection, &'a BTreeSet<StateId>),
133    ) -> Result<Self, PrepareError> {
134        let table = observation.measurement_set().data_description()?;
135        let catalogue = store.catalogue();
136        let mut descriptions = Vec::with_capacity(table.row_count());
137        for row in 0..table.row_count() {
138            let window = table.spectral_window_id(row)?;
139            let selected = match windows {
140                WindowChoice::All => true,
141                WindowChoice::Windows(set) => u32::try_from(window).is_ok_and(|w| set.contains(&w)),
142                WindowChoice::DataDescriptions(ddids) => {
143                    u32::try_from(row).is_ok_and(|r| ddids.contains(&r))
144                }
145            };
146            let wavelength = usize::try_from(window)
147                .ok()
148                .and_then(|window| catalogue.windows.get(window))
149                .and_then(|window| window.reference_frequency.as_ref())
150                .map(casa_types::measures::frequency::MFrequency::hz)
151                .filter(|hz| hz.is_finite() && *hz > 0.0)
152                .map(|hz| SPEED_OF_LIGHT_M_S / hz);
153            descriptions.push((selected, wavelength));
154        }
155        Ok(Self {
156            observation,
157            rows,
158            states,
159            descriptions,
160        })
161    }
162
163    /// Whether the other selections together select MAIN row `keys`.
164    pub(super) fn hold(&self, keys: &MainRowKeys) -> Result<bool, ObservationError> {
165        let Some(&(window_selected, wavelength)) = usize::try_from(keys.data_description)
166            .ok()
167            .and_then(|row| self.descriptions.get(row))
168        else {
169            return Ok(false);
170        };
171        let rows = self.rows;
172        let antennas = (u32::try_from(keys.antenna1), u32::try_from(keys.antenna2));
173        let held = window_selected
174            && rows.fields().holds(i64::from(keys.field))
175            && rows.scans().holds(i64::from(keys.scan))
176            && rows.observations().holds(i64::from(keys.observation))
177            && match antennas {
178                (Ok(antenna1), Ok(antenna2)) => rows.baselines().holds(antenna1, antenna2),
179                _ => rows.baselines().is_all(),
180            }
181            && self.state_holds(keys.state);
182        if !held || matches!(rows.uv_distances(), casa_imaging_model::UvSelection::All) {
183            return Ok(held);
184        }
185        let uvw = self.observation.main_row_uvw(keys.row)?;
186        Ok(rows.uv_distances().holds(uvw, wavelength))
187    }
188
189    /// Whether the intent selection holds a row of `STATE_ID` `state`: a
190    /// selected state, or, when it selects by mode alone, a selected mode.
191    fn state_holds(&self, state: i32) -> bool {
192        match self.rows.intents() {
193            IntentSelection::All => true,
194            IntentSelection::Only(modes) if self.states.is_empty() => modes
195                .binary_search_by(|mode| mode.as_str().cmp(self.observation.state_mode(state)))
196                .is_ok(),
197            IntentSelection::Only(_) => {
198                u32::try_from(state).is_ok_and(|state| self.states.contains(&StateId(state)))
199            }
200        }
201    }
202}
