2//! Converting between MeasurementSet v2 and MSv4 through the observation
3//! boundary: the source is read as batches like any consumer reads it, and
4//! the destination is created from those batches and the source's
5//! catalogue ([`to_msv4()`], [`to_msv2()`]).
6//!
7//! A conversion first inventories what the destination cannot hold (a
8//! [`Loss`]) and refuses, before any byte is written, every loss the
9//! caller did not accept ([`ConversionOptions::accept`]), as a
10//! [`CreationPlan`] does; converting into an MSv2 is a plan over one
11//! input. What it keeps it keeps exactly: every acquisition, its flags (row
12//! flags folded in), weights, UVW and time auxiliaries, and every selected
13//! data group. Padding, a grid cell no acquisition filled, stays padding:
14//! an MSv4 holds it as a NaN weight, and an MSv2 as a flagged row of NaN
15//! data and weights, or, by the declared policy that reproduces an
16//! original sparse MSv2, as no row ([`PaddingRows`]).

34/// The data groups a conversion into an MSv4 of the observation
35/// `catalogue` describes selects from `requested`, or every group of it,
36/// `base` first, for `None`; refused as [`to_msv4()`] refuses them: no group,
37/// a group selected twice, or one a partition lacks. A caller can check a
38/// request with it before converting.
39///
40/// # Errors
41///
42/// [`ObservationError::InvalidRequest`] naming the group refused.
43pub fn selected_data_groups(
44    catalogue: &casa_observation::Catalogue,
45    requested: Option<&[String]>,
46) -> Result<Vec<String>, ObservationError> {
47    super::creation::losses::selected_groups(catalogue, requested)
48}

50/// How a conversion reads its source and stores its destination.
51#[derive(Clone, Debug)]
52pub struct ConversionOptions {
53    /// The data groups to convert, the first carrying every role; `None`
54    /// converts every group of the source, `base` first.
55    pub data_groups: Option<Vec<String>>,
56    /// The losses accepted; any other refuses the conversion.
57    pub accept: Vec<LossKind>,
58    /// The most bytes one time range of every selected group's batches
59    /// may hold together.
60    pub batch_bytes: usize,
61    /// MSv4 destinations: partitions converted at once, each by its own
62    /// thread. An MSv2 is written by one.
63    pub workers: usize,
64    /// MSv4 destinations: the chunk shape `[time, baseline, frequency,
65    /// polarization]`, 0 for a whole axis.
66    pub chunks: [u64; 4],
67    /// MSv4 destinations: the payload variables' codec.
68    pub codec: Msv4Codec,
69    /// MSv2 destinations: what a padded grid cell becomes.
70    pub padding: PaddingRows,
71    /// Software recorded as the creator.
72    pub creator: String,
73    /// Creation time recorded, ISO 8601.
74    pub creation_date: String,
75}

77/// What a conversion did.
78#[derive(Clone, Debug, Default, PartialEq)]
79pub struct ConversionReport {
80    /// The losses the conversion took, all accepted.
81    pub losses: Vec<Loss>,
82    /// Partitions converted.
83    pub partitions: usize,
84    /// Batches of the first group read.
85    pub batches: usize,
86    /// Samples read (the grids' cells, padding included).
87    pub samples: usize,
88    /// MSv2 destinations: rows written, one per sample, or per acquisition
89    /// with padding omitted ([`PaddingRows`]).
90    pub rows: usize,
91    /// MSv4 destinations: payload elements encoded, over every variable.
92    pub elements: usize,
93    /// MSv4 destinations: elements held toward a chunk a batch covered only
94    /// partly.
95    pub held_elements: usize,
96}

98/// Convert the store at `source` (MSv2 or MSv4) into a new MeasurementSet
99/// v2 at `destination`: a [`CreationPlan`] over the one input, written as
100/// [`Destination::New`].
101///
102/// The selected data groups ([`ConversionOptions::data_groups`]) become
103/// `DATA` (`base`), `CORRECTED_DATA` (`corrected`) and `MODEL_DATA`
104/// (`model`), sharing the first group's flags, weights and UVW. Partitions
105/// sharing a spectral window, polarization or intents share their
106/// subtable rows. Every grid cell becomes one row; padding becomes a row
107/// whose every `FLAG` is set, with NaN data, weights and `UVW`, the grid's
108/// interval, no exposure and `FLAG_ROW` false (its flags already exclude
109/// it). With [`PaddingRows::Omitted`], padding becomes no
110/// row, which reproduces an original sparse MSv2. A field that follows an
111/// ephemeris gets the table casacore reads (`FIELD/EPHEM<id>_<name>.tab`)
112/// and its `EPHEMERIS_ID`, its `PHASE_DIR` the offset from the body and its
113/// `TIME` the epoch its catalogue entry states, or else its first time,
114/// the table stamped [`ConversionOptions::creation_date`]. `SIGMA` is derived
115/// from the weights, a loss the caller must accept as [`LossKind::Sigma`]
116/// ([`Loss::SigmaDerived`]) unless the source's stored `SIGMA` is already
117/// `1/√WEIGHT`. Names are the source's: an MSv4 converted from an
118/// MSv2 holds XRADIO's suffixed names (`NAME_FIELDID`, `NAME_STATION`,
119/// ...), so a round trip renames.
120///
121/// # Errors
122///
123/// As [`CreationPlan::new`] and [`CreationPlan::write`].
124pub fn to_msv2(
125    source: &Path,
126    destination: &Path,
127    options: &ConversionOptions,
128) -> Result<ConversionReport, ObservationError> {
129    let plan = CreationPlan::new(
130        &[CreationInput::new(source)],
131        &CreationOptions {
132            data_groups: options.data_groups.clone(),
133            accept: options.accept.clone(),
134            batch_bytes: options.batch_bytes,
135            memory_bytes: default_memory_bytes()?,
136            padding: options.padding,
137            tolerance: casa_observation::MergeTolerance::default(),
138            history: Vec::new(),
139            creation_date: options.creation_date.clone(),
140        },
141        Destination::New(destination.to_path_buf()),
142    )?;
143    let report = plan.write()?;
144    Ok(ConversionReport {
145        losses: report.losses,
146        partitions: report.partitions,
147        batches: report.batches,
148        samples: report.samples,
149        rows: report.rows,
150        ..ConversionReport::default()
151    })
152}
