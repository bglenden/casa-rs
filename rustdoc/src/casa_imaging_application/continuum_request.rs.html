53/// Native continuum reconstruction accepted by the application boundary.
54#[derive(Clone, Debug, PartialEq)]
55pub enum ContinuumAlgorithm {
56    /// Produce dirty and derived products without a minor cycle.
57    Dirty,
58    /// Run the serial Högbom minor cycle and mandatory final major cycle.
59    Hogbom,
60    /// Request the Clark point-component minor cycle.
61    Clark,
62    /// Request a multiscale minor cycle with explicit pixel scales.
63    Multiscale {
64        /// Canonical scale sizes in image pixels.
65        scales_px: Vec<f64>,
66    },
67    /// Request multi-term multi-frequency synthesis.
68    Mtmfs {
69        /// Number of Taylor coefficients.
70        terms: usize,
71    },
72}
73
74/// Native continuum visibility-weighting law.
75#[derive(Clone, Copy, Debug, PartialEq)]
76pub enum ContinuumWeighting {
77    /// Use natural visibility weights.
78    Natural,
79    /// Use uniform density weighting.
80    Uniform,
81    /// Use Briggs robust weighting.
82    Briggs(f64),
83    /// Use Briggs bandwidth-taper weighting.
84    BriggsBandwidthTaper(f64),
85}

87/// Restoring-beam policy requested by the task surface.
88#[derive(Clone, Copy, Debug, PartialEq, Eq)]
89pub enum ContinuumBeamPolicy {
90    /// Fit each plane independently.
91    PerPlane,
92    /// Use one common beam.
93    Common,
94}

96/// Application projection of the native minor-cycle terminal reason.
97#[derive(Clone, Copy, Debug, PartialEq, Eq)]
98pub enum ContinuumStopReason {
99    /// The absolute stopping threshold was reached.
100    ThresholdReached,
101    /// The requested component-update bound was reached.
102    IterationBound,
103    /// The accepted model-update envelope required a fresh major cycle.
104    StalenessBound,
105}

107/// Canonical application input for one MeasurementSet-backed continuum run.
108///
109/// The frontend parses strings and maps task enums into this record. All
110/// MeasurementSet inspection, scientific contract construction, resource
111/// binding, and product publication remain owned by this crate.
112#[derive(Clone, Debug)]
113pub struct ContinuumImagingRequest {
114    /// Input MeasurementSet.
115    pub measurement_set: PathBuf,
116    /// Output product prefix.
117    pub image_name: PathBuf,
118    /// Square image dimension.
119    pub image_size: usize,
120    /// Direction pixel size in arcseconds.
121    pub cell_arcsec: f64,
122    /// Optional selected field identifiers.
123    pub field_ids: Option<Vec<i32>>,
124    /// Optional UV-distance predicate.
125    pub uv_range: Option<String>,
126    /// Optional observing-intent predicate.
127    pub intent: Option<String>,
128    /// Optional selected data-description identifier.
129    pub data_description: Option<i32>,
130    /// Optional CASA spectral-window selector.
131    pub spectral_window: Option<String>,
132    /// Optional first selected source channel.
133    pub channel_start: Option<usize>,
134    /// Optional selected source-channel count.
135    pub channel_count: Option<usize>,
136    /// Optional explicit visibility column.
137    pub data_column: Option<String>,
138    /// Reconstruction algorithm.
139    pub algorithm: ContinuumAlgorithm,
140    /// Visibility-weighting law.
141    pub weighting: ContinuumWeighting,
142    /// Minor-cycle iteration limit.
143    pub iterations: usize,
144    /// Minor-cycle gain.
145    pub gain: f64,
146    /// Absolute stopping threshold in Jy/beam.
147    pub threshold_jy: f64,
148    /// Restoring-beam fit cutoff.
149    pub psf_cutoff: f32,
150    /// Restoring-beam policy.
151    pub beam_policy: ContinuumBeamPolicy,
152    /// Capability constraints derived by the task surface. Unsupported
153    /// capabilities are rejected by the router before physical execution.
154    pub route_requirements: Vec<TaskRouteRequirement>,
155}

157/// Small presentation projection of one completed native continuum run.
158pub struct ContinuumImagingResult {
159    /// Authoritative routed application result.
160    pub outcome: ApplicationOutcome,
161    /// Number of accepted minor-cycle component updates.
162    pub minor_iterations: usize,
163    /// Scientific minor-cycle terminal reason, when a solve ran.

169/// Prepare, route, execute, and publish one native continuum request.
170pub fn execute_continuum(
171    request: ContinuumImagingRequest,
172) -> Result<ContinuumImagingResult, ApplicationDispatchError> {
173    let prepared = prepare(request).map_err(ApplicationDispatchError::Preparation)?;
174    let outcome = crate::execute(prepared)?;
175    let minor = outcome.output.minor_cycle;
176    let product_names = outcome
177        .output
178        .planned_products
179        .members()
180        .iter()
181        .map(|member| member.name().to_string())
182        .collect();
183    Ok(ContinuumImagingResult {
184        minor_iterations: minor.map_or(0, |value| value.iterations),
185        minor_stop_reason: minor.map(|value| match value.stop_reason {
186            MinorCycleStopReason::ThresholdReached => ContinuumStopReason::ThresholdReached,
187            MinorCycleStopReason::IterationBound => ContinuumStopReason::IterationBound,
188            MinorCycleStopReason::StalenessBound => ContinuumStopReason::StalenessBound,
189        }),
190        product_names,
191        outcome,
192    })
193}

195fn prepare(
196    mut request: ContinuumImagingRequest,
197) -> Result<ApplicationRequest<CasaImageProductSink>, crate::ApplicationError> {
198    validate_request(&request)?;
199    let ms = MeasurementSet::open(&request.measurement_set)?;
200    let data_description = ms.data_description()?;
201    let spectral_window = ms.spectral_window()?;
202    let polarization = ms.polarization()?;
203    let ddids = selected_data_descriptions(&request, &data_description)?;
204    let row_selection = ms.selected_observation_row_selection(
205        &ddids,
206        request.field_ids.as_deref(),
207        request.uv_range.as_deref(),
208        request.intent.as_deref(),
209    )?;
210    let content_budget = SelectedObservationContentBudget::new(64 << 20, 2, 4);
211    let mut compact_rows = Vec::new();
212    let mut selected_ddid = None;
213    let mut multiple_ddids = false;
214    let mut selected_field = None;
215    let mut multiple_fields = false;
216    ms.visit_selected_observation_rows(
217        &row_selection,
218        MsSelectionIoBudget {
219            available_bytes: content_budget.available_bytes(),
220            maximum_live_blocks: content_budget.maximum_live_blocks(),
221            requested_bytes_per_row: SelectedObservationRow::STORAGE_BYTES_PER_ROW,
222            storage_alignment_rows: None,
223        },
224        |row| {
225            multiple_ddids |= selected_ddid.is_some_and(|value| value != row.data_description_id());
226            selected_ddid.get_or_insert(row.data_description_id());
227            multiple_fields |= selected_field.is_some_and(|value| value != row.field_id());
228            selected_field.get_or_insert(row.field_id());
229            compact_rows.push(SelectedMainRow::new(
230                u64::try_from(row.physical_row()).expect("row bounded by MS row count"),
231                u32::try_from(row.data_description_id()).expect("validated nonnegative DDID"),
232            ));
233        },
234    )?;
235    if compact_rows.is_empty() {
236        return Err(boxed("selection resolved to no rows"));
237    }
238    if multiple_ddids || multiple_fields {
239        request
240            .route_requirements
241            .push(TaskRouteRequirement::NativeV1UnsupportedControls);
242    }
243    let ddid = usize::try_from(selected_ddid.expect("nonempty selection"))
244        .map_err(|_| boxed("selected DATA_DESC_ID is negative"))?;
245    let field_id = usize::try_from(selected_field.expect("nonempty selection"))
246        .map_err(|_| boxed("selected FIELD_ID is negative"))?;
247    let (spw_id, polarization_id) = data_description_binding(&data_description, ddid)?;
248    let frequencies = spectral_window.chan_freq(spw_id)?;
249    let channels = selected_channels(&request, spw_id, &frequencies)?;
250    let selected_frequencies = channels
251        .iter()
252        .map(|channel| frequencies[*channel])
253        .collect::<Vec<_>>();
254    let reference_frequency =
255        selected_frequencies.iter().sum::<f64>() / selected_frequencies.len() as f64;
256    let correlation_codes = polarization.corr_type(polarization_id)?;
257    let correlations = correlation_codes
258        .iter()
259        .enumerate()
260        .map(|(index, code)| {
261            Ok(CorrelationProduct::new(
262                u32::try_from(index).map_err(|_| boxed("correlation index exceeds u32"))?,
263                correlation_type(*code)?,
264            ))
265        })
266        .collect::<Result<Vec<_>, crate::ApplicationError>>()?;
267    let rows = SelectedRows::from_ordered_main_row_vec(
268        u64::try_from(ms.row_count()).map_err(|_| boxed("MS row count exceeds u64"))?,
269        compact_rows,
270    )?;
271    let observation_selection = ObservationSelection::new(
272        rows,
273        row_selection.rows().clone(),
274        row_selection.data_descriptions().to_vec(),
275        vec![SpectralWindowSelection::new(
276            u32::try_from(spw_id).map_err(|_| boxed("SPW id exceeds u32"))?,
277            channels
278                .iter()
279                .map(|channel| u32::try_from(*channel).map_err(|_| boxed("channel exceeds u32")))
280                .collect::<Result<Vec<_>, _>>()?,
281        )],
282        vec![CorrelationSelection::new(
283            u32::try_from(polarization_id).map_err(|_| boxed("polarization id exceeds u32"))?,
284            correlations,
285        )],
286    );
287    let phase = casa_ms::derived::engine::resolve_field_phase_direction_j2000(&ms, field_id)?;
288    let (right_ascension, declination) = phase.as_angles();
289    let direction = direction_spec(&request, right_ascension, declination);
290    let frequency_reference =
291        FrequencyRef::from_casacore_code(spectral_window.meas_freq_ref(spw_id)?)
292            .unwrap_or(FrequencyRef::TOPO);
293    let geometry = casa_imaging_model::GeometryInput::new(
294        vec![ImageDomainSpec::new(
295            ImageDomainRole::Main,
296            ImageShape::new(request.image_size, request.image_size),
297            direction,
298            FacetLayout::Single,
299            AxisOrder::new([
300                ImageAxis::DirectionLongitude,
301                ImageAxis::DirectionLatitude,
302                ImageAxis::Polarization,
303                ImageAxis::Spectral,
304            ]),
305        )],
306        CentreLaws::new(
307            PhaseCentreLaw::Fixed(direction.reference_direction()),
308            DelayCentreLaw::PhaseTrackingCentre,
309            PointingCentreLaw::PhaseTrackingCentre,
310        ),
311        UvwCoordinateLaw::PhaseTrackingCentre,
312        SpectralCoordinateSpec::new(
313            FrequencyFrame::Topocentric,
314            FrequencyFrame::Topocentric,
315            SpectralFrameAnchor::NotApplicable,
316            SpectralWcs::Linear {
317                channels: 1,
318                reference_pixel: 0.0,
319                reference_frequency_hz: reference_frequency,
320                increment_hz: 1.0,
321            },
322            RestFrequency::NotApplicable,
323            DopplerConvention::NotApplicable,
324        ),
325    );
326    let coordinates = image_coordinates(
327        &request,
328        [right_ascension, declination],
329        frequency_reference,
330        reference_frequency,
331    );
332    let native = production_storage_profile(&request, content_budget)
333        .and_then(|profile| {
334            profile.ok_or_else(|| {
335                boxed("native continuum requires input and output on one filesystem")
336            })
337        })
338        .and_then(|profile| runtime(&request, &profile))
339        .map(|runtime| ApplicationNative {
340            runtime,
341            publication: ApplicationPublication {
342                controls: casa_imaging_products::ContinuumProductControls::new(request.psf_cutoff)
343                    .expect("validated PSF cutoff"),
344                sink: CasaImageProductSink::new(request.image_name.clone(), coordinates),
345            },
346        });
347    let digest = request_digest(&request, b"selection");
348    Ok(ApplicationRequest {
349        specification: specification(&request)?,
350        geometry,
351        model_lifecycle: ModelLifecycleRequirements::new(
352            ModelBounds::new(
353                request.image_size.saturating_mul(request.image_size),
354                1,
355                1,
356                request.iterations.max(1),
357                1.0e30,
358                1.0e30,
359            )?,
360            NumericPrecision::F64,
361            ModelInputCommitment::Empty,
362        ),
363        observation: SelectedObservationResolutionRequest::new(
364            request.measurement_set.display().to_string(),
365            LogicalIdentity::from_sha256(digest),
366            observation_selection,
367            visibility_column(&ms, request.data_column.as_deref())?,
368            if ms.main_table().column_accessor("WEIGHT_SPECTRUM").is_ok() {
369                OwnerWeightColumn::WeightSpectrum
370            } else {
371                OwnerWeightColumn::Weight
372            },
373            Vec::new(),
374            ModelStateIdentity::Empty,
375            content_budget,
376            casa_ms::open_measures_runtime()?,
377        ),
378        task_route_requirements: request.route_requirements,
379        native,
380    })
381}

383fn validate_request(request: &ContinuumImagingRequest) -> Result<(), crate::ApplicationError> {
384    if request.image_size == 0
385        || !request.cell_arcsec.is_finite()
386        || request.cell_arcsec <= 0.0
387        || !request.gain.is_finite()
388        || !request.threshold_jy.is_finite()
389        || !request.psf_cutoff.is_finite()
390        || request.psf_cutoff <= 0.0
391    {
392        return Err(boxed(
393            "native continuum geometry and controls must be finite and positive",
394        ));
395    }
396    Ok(())
397}

399fn selected_data_descriptions(
400    request: &ContinuumImagingRequest,
401    table: &casa_ms::MsDataDescription<'_>,
402) -> Result<Vec<i32>, crate::ApplicationError> {
403    if let Some(ddid) = request.data_description {
404        return Ok(vec![ddid]);
405    }
406    let selected_spws = request
407        .spectral_window
408        .as_deref()
409        .map(parse_spw_selector)
410        .transpose()?
411        .unwrap_or_default()
412        .into_iter()
413        .map(|selector| selector.spw_id)
414        .collect::<BTreeSet<_>>();
415    let mut ddids = Vec::new();
416    for row in 0..table.row_count() {
417        let spw = table.spectral_window_id(row)?;
418        let polarization = table.polarization_id(row)?;
419        if spw >= 0
420            && polarization >= 0
421            && (selected_spws.is_empty() || selected_spws.contains(&spw))
422        {
423            ddids.push(i32::try_from(row).map_err(|_| boxed("DDID exceeds i32"))?);
424        }
425    }
426    if ddids.is_empty() {
427        return Err(boxed("selection resolved to no data descriptions"));
428    }
429    Ok(ddids)
430}

432fn data_description_binding(
433    table: &casa_ms::MsDataDescription<'_>,
434    ddid: usize,
435) -> Result<(usize, usize), crate::ApplicationError> {
436    let spw = table.spectral_window_id(ddid)?;
437    let polarization = table.polarization_id(ddid)?;
438    Ok((
439        usize::try_from(spw).map_err(|_| boxed("selected DDID has a negative SPW id"))?,
440        usize::try_from(polarization)
441            .map_err(|_| boxed("selected DDID has a negative polarization id"))?,
442    ))
443}

445fn selected_channels(
446    request: &ContinuumImagingRequest,
447    spw_id: usize,
448    frequencies: &[f64],
449) -> Result<Vec<usize>, crate::ApplicationError> {
450    if let Some(text) = request.spectral_window.as_deref() {
451        if let Some(selector) = parse_spw_selector(text)?
452            .into_iter()
453            .find(|selector| usize::try_from(selector.spw_id).ok() == Some(spw_id))
454            .and_then(|selector| selector.channels)
455        {
456            return Ok(resolve_channel_selector_selection(frequencies, &selector)?.indices);
457        }
458    }
459    let start = request.channel_start.unwrap_or(0);
460    let count = request
461        .channel_count
462        .unwrap_or_else(|| frequencies.len().saturating_sub(start));
463    let end = start
464        .checked_add(count)
465        .ok_or_else(|| boxed("channel range overflows usize"))?;
466    if count == 0 || end > frequencies.len() {
467        return Err(boxed("selected channel range is empty or out of bounds"));
468    }
469    Ok((start..end).collect())
470}

472fn direction_spec(
473    request: &ContinuumImagingRequest,
474    right_ascension: f64,
475    declination: f64,
476) -> DirectionCoordinateSpec {
477    let cell = request.cell_arcsec * std::f64::consts::PI / (180.0 * 3600.0);
478    let reference_pixel = image_reference_pixel(request.image_size);
479    DirectionCoordinateSpec::new(
480        Projection::Sin,
481        SkyDirection::new(DirectionFrame::J2000, right_ascension, declination),
482        [reference_pixel, reference_pixel],
483        [-cell, cell],
484        [[1.0, 0.0], [0.0, 1.0]],
485        [180.0, 0.0],
486    )
487}

489fn image_coordinates(
490    request: &ContinuumImagingRequest,
491    phase: [f64; 2],
492    frequency_reference: FrequencyRef,
493    reference_frequency: f64,
494) -> CoordinateSystem {
495    let cell = request.cell_arcsec * std::f64::consts::PI / (180.0 * 3600.0);
496    let reference_pixel = image_reference_pixel(request.image_size);
497    let mut coordinates = CoordinateSystem::new();
498    coordinates.add_coordinate(DirectionCoordinate::new(
499        DirectionRef::J2000,
500        CoordinateProjection::new(ProjectionType::SIN),
501        phase,
502        [-cell, cell],
503        [reference_pixel, reference_pixel],
504    ));
505    coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
506    coordinates.add_coordinate(SpectralCoordinate::new(
507        frequency_reference,
508        reference_frequency,
509        1.0,
510        0.0,
511        reference_frequency,
512    ));
513    coordinates
514}

520fn specification(
521    request: &ContinuumImagingRequest,
522) -> Result<ProblemSpecification, crate::ApplicationError> {
523    let (basis, algorithm) = match &request.algorithm {
524        ContinuumAlgorithm::Dirty => (
525            ReconstructionBasis::Constant,
526            ReconstructionAlgorithm::Dirty,
527        ),
528        ContinuumAlgorithm::Hogbom => (
529            ReconstructionBasis::Constant,
530            ReconstructionAlgorithm::Hogbom,
531        ),
532        ContinuumAlgorithm::Clark => (
533            ReconstructionBasis::Constant,
534            ReconstructionAlgorithm::Clark,
535        ),
536        ContinuumAlgorithm::Multiscale { scales_px } => (
537            ReconstructionBasis::Constant,
538            ReconstructionAlgorithm::Multiscale {
539                scales_px: scales_px.clone(),
540            },
541        ),
542        ContinuumAlgorithm::Mtmfs { terms } => (
543            ReconstructionBasis::Taylor { terms: *terms },
544            ReconstructionAlgorithm::Mtmfs,
545        ),
546    };
547    let (weighting, density) = match request.weighting {
548        ContinuumWeighting::Natural => {
549            (WeightingScheme::Natural, WeightDensityScope::NotApplicable)
550        }
551        ContinuumWeighting::Uniform => (
552            WeightingScheme::Uniform,
553            WeightDensityScope::GlobalSelection,
554        ),
555        ContinuumWeighting::Briggs(robust) => (
556            WeightingScheme::Briggs { robust },
557            WeightDensityScope::GlobalSelection,
558        ),
559        ContinuumWeighting::BriggsBandwidthTaper(robust) => (
560            WeightingScheme::BriggsBandwidthTaper { robust },
561            WeightDensityScope::GlobalSelection,
562        ),
563    };
564    Ok(ProblemSpecification::new(
565        ScientificContract::new(
566            SpectralContract::new(SpectralSampling::Identity, SpectralCoupling::Independent),
567            MeasurementEquationContract::new(
568                InstrumentResponse::Scalar,
569                DeclaredInnerProducts::new(
570                    ModelInnerProduct::HermitianEuclidean,
571                    VisibilityInnerProduct::HermitianEuclidean,
572                ),
573            ),
574        ),
575        ReconstructionContract::new(
576            basis,
577            algorithm.clone(),
578            if algorithm == ReconstructionAlgorithm::Dirty {
579                ReconstructionControls::new(0, 1.0, 0.0)
580            } else {
581                ReconstructionControls::new(request.iterations, request.gain, request.threshold_jy)
582                    .with_maximum_model_update(1.0e30)
583            },
584            PolarizationContract::new(vec![PolarizationCoordinate::StokesI]),
585        ),
586        WeightingContract::new(weighting, density),
587        ProductRequirements::new(
588            vec![
589                ProductKind::Psf,
590                ProductKind::Residual,
591                ProductKind::Model,
592                ProductKind::RestoredImage,
593                ProductKind::SumWeights,
594            ],
595            ProductNormalization::UnitResponse,
596            match request.beam_policy {
597                ContinuumBeamPolicy::PerPlane => RestoringBeamPolicy::PerPlane,
598                ContinuumBeamPolicy::Common => RestoringBeamPolicy::Common,
599            },
600            ProductValidityPolicies::new(
601                PrimaryBeamValidityPolicy::new(
602                    0.2,
603                    ProductSupportComparison::StrictlyGreater,
604                    ProductBlankingPolicy::ZeroAndFalseMask,
605                )?,
606                TaylorValidityPolicy::new(
607                    TaylorSupportReference::PrincipalResidualTaylor0PositiveMaximum,
608                    0.1,
609                    ProductSupportComparison::StrictlyGreater,
610                    ProductBlankingPolicy::ZeroAndFalseMask,
611                )?,
612            ),
613        ),
614        ObservationTransactionRequirements::new(ModelColumnWrite::Disabled),
615        NumericsContract::new(
616            vec![NumericPrecision::F64],
617            ReductionPolicy::Compensated,
618            FiniteValuePolicy::FlagInputRejectGenerated,
619            NumericalStage::ALL
620                .into_iter()
621                .map(|stage| (stage, StageErrorBudget::new(1.0e-7, 1.0e-3)))
622                .collect(),
623        ),
624    ))
625}

627fn visibility_column(
628    ms: &MeasurementSet,
629    explicit: Option<&str>,
630) -> Result<OwnerVisibilityColumn, crate::ApplicationError> {
631    let column = match explicit.map(str::to_ascii_uppercase).as_deref() {
632        Some("DATA") => VisibilityDataColumn::Data,
633        Some("CORRECTED_DATA" | "CORRECTED") => VisibilityDataColumn::CorrectedData,
634        Some("MODEL_DATA" | "MODEL") => {
635            return Err(boxed("MODEL_DATA is not an observation input"));
636        }
637        Some(other) => return Err(boxed(format!("unsupported visibility column {other:?}"))),
638        None if ms.data_column(VisibilityDataColumn::CorrectedData).is_ok() => {
639            VisibilityDataColumn::CorrectedData
640        }
641        None if ms.data_column(VisibilityDataColumn::Data).is_ok() => VisibilityDataColumn::Data,
642        None => return Err(boxed("MS has neither CORRECTED_DATA nor DATA")),
643    };
644    Ok(match column {
645        VisibilityDataColumn::Data => OwnerVisibilityColumn::Data,
646        VisibilityDataColumn::CorrectedData => OwnerVisibilityColumn::CorrectedData,
647        VisibilityDataColumn::ModelData => unreachable!("rejected above"),
648    })
649}

690fn production_storage_profile(
691    request: &ContinuumImagingRequest,
692    content_budget: SelectedObservationContentBudget,
693) -> Result<Option<ProductionStorageProfile>, crate::ApplicationError> {
694    let output_parent = request
695        .image_name
696        .parent()
697        .unwrap_or_else(|| Path::new("."));
698    std::fs::create_dir_all(output_parent)?;
699    let output_root = filesystem_root(&output_parent.canonicalize()?)?;
700    let input_root = filesystem_root(&request.measurement_set.canonicalize()?)?;
701    if output_root != input_root {
702        return Ok(None);
703    }
704    let (capacity, available) = filesystem_capacity(&output_root)?;
705    let read_rate = positive_environment("CASA_RS_IMAGING_SPILL_READ_BYTES_PER_SECOND")?;
706    let write_rate = positive_environment("CASA_RS_IMAGING_SPILL_WRITE_BYTES_PER_SECOND")?;
707    Ok(Some(ProductionStorageProfile::new(
708        output_root,
709        capacity,
710        available,
711        read_rate,
712        write_rate,
713        u64::try_from(content_budget.maximum_live_blocks()).unwrap_or(u64::MAX),
714        2,
715    )?))
716}

718fn runtime(
719    request: &ContinuumImagingRequest,
720    profile: &ProductionStorageProfile,
721) -> Result<ApplicationRuntime, crate::ApplicationError> {
722    let digest = request_digest(request, b"attempt");
723    let receipts = request
724        .image_name
725        .parent()
726        .unwrap_or_else(|| Path::new("."))
727        .join(".casa-rs-imaging-receipts");
728    Ok(ApplicationRuntime {
729        registry: ImplementationRegistryId::from_sha256(hash(b"serial-continuum-registry")),
730        implementation: WorkImplementationId::new("serial-continuum-cpu-v1"),
731        weighting_limits: WeightingExecutionLimits::new(4096, 1)?,
732        stage_nanos: 1_000_000,
733        minor_cycle_bytes: u64::try_from(
734            request
735                .image_size
736                .saturating_mul(request.image_size)
737                .saturating_mul(16),
738        )
739        .unwrap_or(u64::MAX),
740        storage_io: profile.io_resources(),
741        confidence_parts_per_million: 900_000,
742        resource_policy: ResourcePolicy::Balanced,
743        cost_model: PlannerCostModelProfileId::from_sha256(hash(b"serial-continuum-cost-v1"))
744            .bootstrap(),
745        authority: ResourceAuthority::production_with_storage_profile(profile)?.clone(),
746        receipts: ExecutionReceiptStore::new(receipts, ReceiptRetention::new(128, 64 << 20)?)?,
747        build: BuildIdentity::from_sha256(hash(env!("CARGO_PKG_VERSION").as_bytes())),
748        attempts: [
749            ExecutionAttemptId::from_sha256(scoped(digest, 0)),
750            ExecutionAttemptId::from_sha256(scoped(digest, 1)),
751            ExecutionAttemptId::from_sha256(scoped(digest, 2)),
752        ],
753    })
754}

756#[cfg(unix)]
757fn filesystem_root(path: &Path) -> Result<PathBuf, crate::ApplicationError> {
758    use std::os::unix::fs::MetadataExt;
759    let device = std::fs::metadata(path)?.dev();
760    let mut root = path.to_path_buf();
761    while let Some(parent) = root.parent() {
762        if std::fs::metadata(parent)?.dev() != device {
763            break;
764        }
765        root = parent.to_path_buf();
766    }
767    Ok(root)
768}

770#[cfg(not(unix))]
771fn filesystem_root(_path: &Path) -> Result<PathBuf, crate::ApplicationError> {
772    Err(boxed(
773        "filesystem-root detection is unavailable on this platform",
774    ))
775}

777#[cfg(unix)]
778fn filesystem_capacity(root: &Path) -> Result<(u64, u64), crate::ApplicationError> {
779    use std::os::unix::ffi::OsStrExt;
780    let root = CString::new(root.as_os_str().as_bytes())?;
781    let mut statistics = std::mem::MaybeUninit::<libc::statvfs>::uninit();
782    // SAFETY: the path is live and NUL-terminated, and the output allocation is valid.
783    if unsafe { libc::statvfs(root.as_ptr(), statistics.as_mut_ptr()) } != 0 {
784        return Err(Box::new(std::io::Error::last_os_error()));
785    }
786    // SAFETY: successful statvfs initialized the complete value.
787    let statistics = unsafe { statistics.assume_init() };
788    let block_size = statistics.f_frsize;
789    Ok((
790        u64::from(statistics.f_blocks).saturating_mul(block_size),
791        u64::from(statistics.f_bavail).saturating_mul(block_size),
792    ))
793}

795#[cfg(not(unix))]
796fn filesystem_capacity(_root: &Path) -> Result<(u64, u64), crate::ApplicationError> {
797    Err(boxed(
798        "filesystem capacity detection is unavailable on this platform",
799    ))
800}

802fn positive_environment(name: &str) -> Result<u64, crate::ApplicationError> {
803    std::env::var(name)
804        .map_err(|_| boxed(format!("native continuum requires {name}")))?
805        .parse::<u64>()
806        .ok()
807        .filter(|value| *value > 0)
808        .ok_or_else(|| boxed(format!("{name} must be a positive integer")))
809}

811fn request_digest(request: &ContinuumImagingRequest, domain: &[u8]) -> [u8; 32] {
812    let mut hasher = Sha256::new();
813    hasher.update(domain);
814    hasher.update(request.measurement_set.as_os_str().as_encoded_bytes());
815    hasher.update(request.image_name.as_os_str().as_encoded_bytes());
816    hasher.update(request.image_size.to_le_bytes());
817    hasher.update(request.cell_arcsec.to_bits().to_le_bytes());
818    hasher.finalize().into()
819}
820
821fn hash(value: &[u8]) -> [u8; 32] {
822    Sha256::digest(value).into()
823}
824
825fn scoped(mut digest: [u8; 32], scope: u8) -> [u8; 32] {
826    digest[0] ^= scope;
827    digest
828}

830fn boxed(message: impl Into<String>) -> crate::ApplicationError {
831    Box::new(std::io::Error::other(message.into()))
832}
