2//! `setoptions(ftmachine='sd')`: total power, as CASA's simulator predicts
3//! it (`SDGrid`, built by `Simulator::createSkyEquation`), by sampling the
4//! model in the image plane.
5//!
6//! For a row pointed at pixel `(x, y)` of the model, `SDGrid::get` calls
7//! `dgridsd` (`fgridsd.f`):
8//! - the pointing's nearest pixel is `loc = nint(x + 1)` (one-based) and its
9//!   offset from it, in hundredths of a pixel, `off = nint((loc − x − 1) ·
10//!   100)`; a pointing whose nearest pixel is off the image predicts zero;
11//! - each pixel within `support` pixels of `loc` along both axes adds the
12//!   model's value times the radial kernel at `floor(√(rx² + ry²))`
13//!   hundredths, with `r = 100 i + off`, summed in single precision;
14//! - the visibility is the sum's conjugate.
15//!
16//! The kernels (`SDGrid::init`, `findPBAsConvFunction`) are tabulated at a
17//! hundredth of a pixel:
18//! - `BOX`: 1 out to half a pixel, support 0;
19//! - `SF`: CASA's spheroid `(1 − ν²) grdsf(ν)` over 3 pixels;
20//! - `PB`: the telescope's power pattern along the x axis, at the frequency
21//!   of model plane 0 for every channel, tabulated over `max(128, support)`
22//!   pixels (`PBMath1D::support`), cut at its first value below 1e-3, of
23//!   which CASA keeps one fewer (`Slice(0, cfLen − 1)`); the support is the
24//!   kept length in whole pixels.
25//!
26//! The model is a Stokes I image: the parallel hands take its value and the
27//! cross hands none. A channel takes the model plane `matchChannel` gives
28//! it, and none off the model's spectral axis. Every row of a scan is
29//! pointed at the scan's field, so a field's samples serve all its rows.
30//!
31//! Only autocorrelations are predicted, and flagged rows are not. CASA
32//! skips a row with a positive uvw component, which leaves a cross
33//! correlation with no positive component predicted as if it were one dish;
34//! casa-rs predicts autocorrelations only (`docs/CASA (C++) bugs.md`).
35

42/// Samples per pixel of every `SDGrid` kernel (`convSampling`).
43const SAMPLING: usize = 100;
44
45/// `findPBAsConvFunction`: the PB kernel ends before its first sample
46/// below this.
47const PB_CUT: f32 = 1.0e-3;
48
49/// `findPBAsConvFunction`: the PB kernel is tabulated over at least this
50/// many pixels.
51const PB_MINIMUM_PIXELS: usize = 128;
52
53/// The `SDGrid` kernel `setoptions(gridfunction=…)` names (`SDGrid::init`).
54#[derive(Debug, Clone, Copy, PartialEq, Eq)]
55pub enum SingleDishKernel {
56    /// `PB`: the telescope's primary beam.
57    Pb,
58    /// `BOX`: the nearest pixel.
59    Box,
60    /// `SF`: CASA's prolate spheroid over 3 pixels.
61    Spheroidal,
62}
63
64impl SingleDishKernel {
65    /// The kernel `gridfunction` names, in any case; `None` for the others
66    /// (`GAUSS` and `GJINC`, which CASA also has, among them).
67    #[must_use]
68    pub fn from_grid_function(grid_function: &str) -> Option<Self> {
69        match grid_function.to_lowercase().as_str() {
70            "pb" => Some(Self::Pb),
71            "box" => Some(Self::Box),
72            "sf" => Some(Self::Spheroidal),
73            _ => None,
74        }
75    }
76}

78/// A radial kernel at a hundredth of a pixel (`convFunc`) and its support
79/// in pixels.
80#[derive(Debug)]
81struct Kernel {
82    table: Vec<f32>,
83    support: usize,
84}
85
86impl Kernel {
87    fn new(kind: SingleDishKernel, model: &SkyModel, beam: Option<&BeamPattern>) -> Self {
88        let tabulated = |support: usize, values: Vec<f32>| {
89            let mut table = vec![0.0; SAMPLING * (2 * support + 2)];
90            table[..values.len()].copy_from_slice(&values);
91            Self { table, support }
92        };
93        match kind {
94            SingleDishKernel::Box => tabulated(0, vec![1.0; SAMPLING]),
95            SingleDishKernel::Spheroidal => {
96                let support = 3;
97                let samples = SAMPLING * support;
98                tabulated(
99                    support,
100                    (0..samples)
101                        .map(|i| {
102                            let nu = i as f64 / samples as f64;
103                            ((1.0 - nu * nu) * grdsf(nu)) as f32
104                        })
105                        .collect(),
106                )
107            }
108            SingleDishKernel::Pb => {
109                let beam = beam.expect("a PB kernel has a beam");
110                let frequency_hz = model.plane_frequency_hz(0);
111                let increment = model.increment_rad[0] * (1.0 / SAMPLING as f64);
112                let pixels = PB_MINIMUM_PIXELS
113                    .max(beam.support_pixels(model.increment_rad[0], frequency_hz));
114                let mut values: Vec<f32> = (0..SAMPLING * pixels)
115                    .map(|i| beam.power_on_axis(i as f64 * increment, frequency_hz))
116                    .take_while(|&power| power >= PB_CUT)
117                    .collect();
118                // CASA keeps `Slice(0, cfLen − 1)` of the `cfLen` samples
119                // before the cut, at least one counted, but takes the
120                // support from all `cfLen` (`convSupport = cfLen/convSampling`).
121                let counted = values.len().max(1);
122                values.truncate(counted - 1);
123                tabulated(counted / SAMPLING, values)
124            }
125        }
126    }
127
128    fn bytes(&self) -> u64 {
129        (self.table.len() * size_of::<f32>()) as u64
130    }
131}

142/// Where a total-power prediction samples the model: each distinct pointing
143/// direction of the scans it covers, and each scan's pointing. A scan
144/// points at its phase centre, which `observemany` can move off its field
145/// (the POINTING table casa-rs writes follows it).
146#[derive(Debug)]
147pub(crate) struct SingleDishPointings {
148    /// Each distinct pointing's J2000 direction, rad.
149    pub(crate) directions_rad: Vec<[f64; 2]>,
150    /// `[scan]`: the scan's pointing, an index into `directions_rad`.
151    pub(crate) of_scan: Vec<usize>,
152}
153
154impl SingleDishPointings {
155    /// The pointings of scans pointed at `phase_centres_rad`, in scan order.
156    pub(crate) fn of_scans(phase_centres_rad: impl ExactSizeIterator<Item = [f64; 2]>) -> Self {
157        let mut directions_rad: Vec<[f64; 2]> = Vec::new();
158        let of_scan = phase_centres_rad
159            .map(|centre| {
160                let same = |known: &[f64; 2]| known.map(f64::to_bits) == centre.map(f64::to_bits);
161                directions_rad.iter().position(same).unwrap_or_else(|| {
162                    directions_rad.push(centre);
163                    directions_rad.len() - 1
164                })
165            })
166            .collect();
167        Self {
168            directions_rad,
169            of_scan,
170        }
171    }
172}

174/// A model prepared for total-power prediction: each pointing's samples in
175/// each window.
176#[derive(Debug)]
177pub(crate) struct SingleDishPrediction {
178    /// `[pointing][window]`: `[channel][correlation]`.
179    samples: Vec<Vec<Vec<Complex32>>>,
180    /// `[scan]`: the scan's pointing.
181    of_scan: Vec<usize>,
182    /// The bytes [`Self::bytes`] counts.
183    _reservation: Reservation,
184}
185
186impl SingleDishPrediction {
187    /// Bytes a prediction of `windows` at `pointings` holds, known before it
188    /// is formed: the samples and the scans' pointings.
189    pub(crate) fn bytes(windows: &[SingleDishWindow], pointings: &SingleDishPointings) -> u64 {
190        let per_pointing: usize = size_of::<Vec<Vec<Complex32>>>()
191            + windows
192                .iter()
193                .map(|window| {
194                    size_of::<Vec<Complex32>>()
195                        + window.frequencies_hz.len()
196                            * window.correlations.len()
197                            * size_of::<Complex32>()
198                })
199                .sum::<usize>();
200        (pointings.directions_rad.len() * per_pointing
201            + pointings.of_scan.len() * size_of::<usize>()) as u64
202    }
203
204    /// Bytes preparing the prediction holds beside it:
205    /// - the kernel, as tabulated and as kept (`beam` is the `PB` kernel's
206    ///   pattern, `None` for the others);
207    /// - a model plane read, with the beam's power plane and their product
208    ///   when `setvp(dovp=True)` applies it first, and the image's tile
209    ///   cache ([`SkyModel::cache_bytes`]);
210    /// - each window channel's plane, and each pointing's direction, pixel
211    ///   and sample of the plane read.
212    pub(crate) fn preparation_bytes(
213        model: &SkyModel,
214        beam: Option<&BeamPattern>,
215        (windows, pointings): (&[SingleDishWindow], &SingleDishPointings),
216    ) -> u64 {
217        let pixels = beam.map_or(3, |beam| {
218            PB_MINIMUM_PIXELS
219                .max(beam.support_pixels(model.increment_rad[0], model.plane_frequency_hz(0)))
220        });
221        let kernel = SAMPLING * pixels + SAMPLING * (2 * pixels + 2);
222        let channels: usize = windows.iter().map(|w| w.frequencies_hz.len()).sum();
223        let planes_of =
224            windows.len() * size_of::<Vec<Option<usize>>>() + channels * size_of::<Option<usize>>();
225        let per_pointing = 2 * size_of::<[f64; 2]>() + size_of::<f32>();
226        (kernel * size_of::<f32>() + planes_of + pointings.directions_rad.len() * per_pointing)
227            as u64
228            + 2 * model.plane_bytes()
229            + model.cache_bytes()
230    }
231
232    /// Prepare `model` with `kernel` for `windows` at `pointings`, holding
233    /// `reservation`, its admission of [`Self::bytes`]. `beams` are the
234    /// telescope's pattern, which the `PB` kernel needs, and the pattern
235    /// `setvp(dovp=True)` applies to the model first, centred on each
236    /// pointing, as CASA's `SkyEquation` does whatever the kernel (so a
237    /// `PB` kernel then weights the model by the power pattern twice).
238    ///
239    /// # Errors
240    ///
241    /// The model's pixels could not be read, or a pointing could not be
242    /// placed on them.
243    pub(crate) fn new(
244        model: &SkyModel,
245        kernel: SingleDishKernel,
246        (beam, voltage_pattern): (Option<&BeamPattern>, Option<&BeamPattern>),
247        (windows, pointings): (&[SingleDishWindow], SingleDishPointings),
248        reservation: Reservation,
249    ) -> Result<Self, PredictError> {
250        let SingleDishPointings {
251            directions_rad,
252            of_scan,
253        } = pointings;
254        let kernel = Kernel::new(kernel, model, beam);
255        let axis = SpectralAxis::new(
256            model.first_frequency_hz,
257            model.frequency_increment_hz,
258            u32::try_from(model.planes())
259                .map_err(|_| PredictError::UnsupportedModel("too many model planes".into()))?,
260        )?;
261        // Each window channel's model plane (`matchChannel`), or none.
262        let planes_of: Vec<Vec<Option<usize>>> = windows
263            .iter()
264            .map(|window| {
265                window
266                    .frequencies_hz
267                    .iter()
268                    .map(|&frequency| axis.nearest_channel(frequency).map(|plane| plane as usize))
269                    .collect()
270            })
271            .collect();
272        let pixels: Vec<[f64; 2]> = directions_rad
273            .iter()
274            .map(|direction| model.pixel_of(*direction))
275            .collect::<Result<_, _>>()?;
276        // The samples, zero until a plane fills them. Sized exactly: they
277        // are kept, and charged by their length.
278        let mut samples: Vec<Vec<Vec<Complex32>>> = directions_rad
279            .iter()
280            .map(|_| {
281                windows
282                    .iter()
283                    .map(|window| {
284                        vec![
285                            Complex32::new(0.0, 0.0);
286                            window.frequencies_hz.len() * window.correlations.len()
287                        ]
288                    })
289                    .collect()
290            })
291            .collect();
292        // One plane read at a time, each pointing's sample of it written
293        // into the channels it feeds, so no plane × pointing matrix is held
294        // (#874 review).
295        let mut by_pointing = vec![0.0_f32; directions_rad.len()];
296        for plane in 0..model.planes() {
297            if !planes_of
298                .iter()
299                .flatten()
300                .any(|&mapped| mapped == Some(plane))
301            {
302                continue;
303            }
304            let values = model.read_plane(plane)?;
305            for (value, pixel) in by_pointing.iter_mut().zip(&pixels) {
306                *value = match voltage_pattern {
307                    Some(pattern) => {
308                        let apparent = &values * &pattern.power_plane(model, plane, *pixel);
309                        sample(&apparent, *pixel, &kernel)
310                    }
311                    None => sample(&values, *pixel, &kernel),
312                };
313            }
314            for (pointing_windows, &value) in samples.iter_mut().zip(&by_pointing) {
315                for ((cells, window), planes) in
316                    pointing_windows.iter_mut().zip(windows).zip(&planes_of)
317                {
318                    let correlations = window.correlations.len();
319                    for (channel, _) in planes
320                        .iter()
321                        .enumerate()
322                        .filter(|(_, mapped)| **mapped == Some(plane))
323                    {
324                        let channel_cells = &mut cells[channel * correlations..][..correlations];
325                        for (cell, &code) in channel_cells.iter_mut().zip(&window.correlations) {
326                            // The parallel hands carry Stokes I.
327                            if matches!(code, 1 | 5 | 8 | 9 | 12) {
328                                *cell = Complex32::new(value, 0.0);
329                            }
330                        }
331                    }
332                }
333            }
334        }
335        tracing::debug!(
336            support = kernel.support,
337            bytes = kernel.bytes(),
338            "single-dish kernel"
339        );
340        Ok(Self {
341            samples,
342            of_scan,
343            _reservation: reservation,
344        })
345    }
346
347    /// The samples of scan `scan`'s pointing in window `window`,
348    /// `[channel][correlation]`; empty for a scan not prepared.
349    pub(crate) fn samples(&self, scan: usize, window: usize) -> &[Complex32] {
350        self.of_scan
351            .get(scan)
352            .map_or(&[], |&pointing| self.samples[pointing][window].as_slice())
353    }
354}

356/// `dgridsd` for one pointing at fractional pixel `pixel` of `plane`
357/// (`[y][x]`): the real part of its conjugated sum.
358fn sample(plane: &ndarray::Array2<f32>, pixel: [f64; 2], kernel: &Kernel) -> f32 {
359    let (ny, nx) = plane.dim();
360    // `sgridsd`, in single precision and one-based.
361    let position = pixel.map(|value| (value + 1.0) as f32);
362    let location = position.map(|value| value.round() as i64);
363    let offset = [0, 1]
364        .map(|axis| ((location[axis] as f32 - position[axis]) * SAMPLING as f32).round() as i64);
365    // `ogridsd` with no support: the nearest pixel is on the image.
366    let on_image = |value: i64, size: usize| value >= 1 && value <= size as i64;
367    if !on_image(location[0], nx) || !on_image(location[1], ny) {
368        return 0.0;
369    }
370    let support = kernel.support as i64;
371    let mut sum = 0.0_f32;
372    for iy in -support..=support {
373        let y = location[1] + iy;
374        if !on_image(y, ny) {
375            continue;
376        }
377        let ry = (SAMPLING as i64 * iy + offset[1]) as f32;
378        for ix in -support..=support {
379            let x = location[0] + ix;
380            if !on_image(x, nx) {
381                continue;
382            }
383            let rx = (SAMPLING as i64 * ix + offset[0]) as f32;
384            let radius = (rx * rx + ry * ry).sqrt() as usize;
385            let weight = kernel.table.get(radius).copied().unwrap_or(0.0);
386            sum += weight * plane[[(y - 1) as usize, (x - 1) as usize]];
387        }
388    }
389    sum
390}
