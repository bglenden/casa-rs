2//! `sm.predict(complist=…)`: a component list's visibilities, as CASA's
3//! simulator predicts them (`SkyEquation::get` with its
4//! `SimpleComponentFTMachine`), analytically and with no gridding.
5//!
6//! For each row and component (`SimpleComponentFTMachine::get`):
7//! - the row's uvw is rotated from the scan's phase centre to the
8//!   component's reference direction through `ComponentFTMachine`'s
9//!   projected [`UvwMachine`] at the integration epoch and antenna 0;
10//! - the component's visibility is its flux, sampled by its spectral model
11//!   at the channel's frequency and converted to the window's polarisation
12//!   basis, times its shape's visibility at the rotated baseline
13//!   (`SkyCompRep::visibility`, [`SkyComponent::visibility`]);
14//! - with `setvp(dovp=True)`, the flux is scaled first by the primary beam's
15//!   power at the component's separation from the field's pointing, at each
16//!   channel's frequency (`BeamSkyJones::apply` on a component, with its
17//!   full-spectral option, as `SkyEquation::applySkyJones` calls it);
18//! - the rotation's phase multiplies it: `e^{−2πi s ν/c}` with `s` the
19//!   phase shift of [`UvwMachine::convert_projected`], in CASA's sign.
20//!
21//! What does not depend on the row is prepared once: each component's flux
22//! in each window's correlations and the beam's power, per channel; each
23//! shape's transform ([`ShapeTransform`]); and, for each observed field,
24//! each component's rotation (a scan is phased on its field's direction).
25//! Native-frame transforms are rebuilt once per integration, as CASA's
26//! component machine does, using the first antenna's position.
27//! A row then takes one rotation and one phasor per component, and each
28//! channel advances the phasor by the channel spacing's rotation: a
29//! window's channels are evenly spaced (`setspwindow`).
30//!
31//! The visibilities add to what DATA holds. Unlike an image's prediction,
32//! a component's is formed for flagged rows too: CASA's component machine
33//! does not read the flags.
34//!
35//! Divergences, recorded in `docs/CASA (C++) bugs.md`:
36//! - **A negative flux keeps its sign under the beam.** CASA's full-spectral
37//!   beam scales each channel by the flux at that channel divided by the
38//!   flux's magnitude, and so turns a negative flux positive.
39//! - **The beam is taken at the data's frequencies.** CASA tabulates the
40//!   beam-scaled flux at each channel's LSRK frequency and then samples that
41//!   table at the data's (here TOPO) frequencies, about 1e-4 of the
42//!   frequency away.
43//!
44//! Native direction references retain their axes when evaluating shapes.
45

55/// The speed of light, m/s.
56const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

58/// A window's channels and correlations, as the components are predicted
59/// into them.
60#[derive(Debug, Clone)]
61pub(crate) struct ComponentWindow {
62    pub(crate) stored_frequencies: Option<Arc<Admitted<Vec<f64>>>>,
63    /// The first channel's frequency, Hz.
64    pub(crate) first_hz: f64,
65    /// The channel spacing, Hz, for the generated linear axis.
66    pub(crate) step_hz: f64,
67    pub(crate) channels: usize,
68    /// casacore `Stokes::StokesTypes` codes, one per correlation.
69    pub(crate) correlations: Vec<i32>,
70}
71
72impl ComponentWindow {
73    fn frequency_hz(&self, channel: usize) -> f64 {
74        self.stored_frequencies.as_ref().map_or_else(
75            || self.first_hz + channel as f64 * self.step_hz,
76            |frequencies| frequencies.value[channel],
77        )
78    }
79}

81/// A failure to prepare a component list.
82#[derive(Debug, Clone, PartialEq, Eq)]
83pub(crate) enum ComponentRefusal {
84    /// Something casa-rs does not evaluate yet.
85    NotYetImplemented(String),
86}

88/// What a component list is predicted into: the session's windows, the
89/// distinct pairs of visibility phase centre and antenna beam centre,
90/// each scan's pair, and the primary beam with `setvp(dovp=True)`.
91/// FIELD determines phase; stored POINTING independently determines beam
92/// attenuation in a resumed simulation.
93#[derive(Clone, Copy)]
94pub(crate) struct ComponentTargets<'a> {
95    pub(crate) windows: &'a [ComponentWindow],
96    /// Slot, visibility phase centre, and antenna beam centre, J2000 rad.
97    pub(crate) pointings: &'a [(usize, [f64; 2], [f64; 2])],
98    /// `[scan]`: the scan's pointing.
99    pub(crate) pointing_of_scan: &'a [usize],
100    pub(crate) beam: Option<&'a BeamPattern>,
101    pub(crate) frame: &'a MeasFrame,
102}
103
104impl ComponentTargets<'_> {
105    /// Bytes a prediction of `components` components into these targets
106    /// holds ([`ComponentPrediction::bytes`]), known before it is formed.
107    pub(crate) fn bytes(&self, components: usize) -> u64 {
108        let samples: usize = self
109            .windows
110            .iter()
111            .map(|window| window.channels * window.correlations.len())
112            .sum();
113        let channels: usize = self.windows.iter().map(|window| window.channels).sum();
114        let field_slots = self
115            .pointings
116            .iter()
117            .map(|(id, _, _)| id + 1)
118            .max()
119            .unwrap_or(0);
120        let per_field = components * size_of::<UvwMachine>()
121            + self.beam.map_or(0, |_| {
122                self.windows.len() * size_of::<Vec<f32>>()
123                    + components * channels * size_of::<f32>()
124            });
125        (components
126            * (size_of::<ShapeTransform>()
127                + size_of::<MDirection>()
128                + samples * size_of::<Complex64>())
129            + self.windows.len() * size_of::<PreparedWindow>()
130            + field_slots * size_of::<Option<FieldTerms>>()
131            + self.pointings.len() * per_field
132            + size_of_val(self.pointing_of_scan)) as u64
133    }
134
135    /// Bytes [`ComponentPrediction::new`] holds only while it prepares
136    /// `components` components: their J2000 directions.
137    pub(crate) fn preparation_bytes(components: usize) -> u64 {
138        (components * size_of::<[f64; 2]>()) as u64
139    }
140}

142/// A component list prepared for prediction into a session's windows.
143#[derive(Debug)]
144pub(crate) struct ComponentPrediction {
145    directions: Vec<MDirection>,
146    frame: MeasFrame,
147    /// Each component's shape transform.
148    transforms: Vec<ShapeTransform>,
149    windows: Vec<PreparedWindow>,
150    /// `[pointing]`: a pointing's terms, `None` for one not prepared.
151    fields: Vec<Option<FieldTerms>>,
152    /// `[scan]`: the scan's pointing.
153    of_scan: Vec<usize>,
154    /// The bytes [`Self::bytes`] counts, admitted before they were formed
155    /// and held with them.
156    _reservation: Reservation,
157}

159#[derive(Debug)]
160struct PreparedWindow {
161    stored_frequencies: Option<Arc<Admitted<Vec<f64>>>>,
162    first_hz: f64,
163    step_hz: f64,
164    channels: usize,
165    correlations: usize,
166    /// The components' fluxes in the window's correlations, Jy:
167    /// `[component][channel][correlation]`.
168    fluxes: Vec<Complex64>,
169}

171#[derive(Debug)]
172struct FieldTerms {
173    phase: MDirection,
174    /// Each component's rotation from the field's direction.
175    projections: Vec<UvwMachine>,
176    /// `[window]`: the beam's power `[component][channel]`, with
177    /// `setvp(dovp=True)`.
178    powers: Option<Vec<Vec<f32>>>,
179}
180
181impl ComponentPrediction {
182    /// Prepare `list` for prediction into `targets`, holding `reservation`,
183    /// its admission of [`ComponentTargets::bytes`] for the list's length.
184    ///
185    /// # Errors
186    ///
187    /// [`ComponentRefusal::NotYetImplemented`]: a time-dependent component
188    /// beam needing separate observer context, correlations not all circular or all
189    /// linear, or a flux, spectrum or shape casa-components does not
190    /// evaluate.
191    pub(crate) fn new(
192        list: &ComponentList,
193        targets: ComponentTargets<'_>,
194        reservation: Reservation,
195    ) -> Result<Self, ComponentRefusal> {
196        let refused = ComponentRefusal::NotYetImplemented;
197        let components = list.components();
198        let mut directions_rad = Vec::with_capacity(components.len());
199        let mut transforms = Vec::with_capacity(components.len());
200        for (index, component) in components.iter().enumerate() {
201            let direction = component.shape().reference_direction();
202            if targets.beam.is_some()
203                && !matches!(
204                    direction.refer(),
205                    DirectionRef::J2000
206                        | DirectionRef::B1950
207                        | DirectionRef::B1950Vla
208                        | DirectionRef::GALACTIC
209                        | DirectionRef::ICRS
210                        | DirectionRef::SUPERGAL
211                        | DirectionRef::ECLIPTIC
212                )
213            {
214                return Err(refused(format!(
215                    "component {index} in {:?} needs its beam's actual array observer context (#1297)",
216                    direction.refer()
217                )));
218            }
219            let (ra, dec) = direction
220                .convert_to(DirectionRef::J2000, targets.frame)
221                .map_err(|error| refused(error.to_string()))?
222                .as_angles();
223            directions_rad.push([ra, dec]);
224            transforms.push(
225                component
226                    .transform()
227                    .map_err(|error| refused(format!("component {index}: {error}")))?,
228            );
229        }
230        let windows = targets
231            .windows
232            .iter()
233            .map(|window| prepare_window(components, window))
234            .collect::<Result<Vec<_>, _>>()?;
235        let directions: Vec<MDirection> = components
236            .iter()
237            .map(|component| component.shape().reference_direction().clone())
238            .collect();
239        let slots = targets
240            .pointings
241            .iter()
242            .map(|(id, _, _)| id + 1)
243            .max()
244            .unwrap_or(0);
245        let mut fields: Vec<Option<FieldTerms>> = (0..slots).map(|_| None).collect();
246        for &(field, direction, beam_centre) in targets.pointings {
247            let phase = MDirection::from_angles(direction[0], direction[1], DirectionRef::J2000);
248            let projections = directions
249                .iter()
250                .map(|component| {
251                    UvwMachine::with_frame(component, &phase, targets.frame)
252                        .map_err(|error| refused(error.to_string()))
253                })
254                .collect::<Result<Vec<_>, _>>()?;
255            let powers = targets.beam.map(|pattern| {
256                targets
257                    .windows
258                    .iter()
259                    .map(|window| {
260                        let mut powers = Vec::with_capacity(directions_rad.len() * window.channels);
261                        for component in &directions_rad {
262                            let separation = separation_rad(*component, beam_centre);
263                            powers.extend((0..window.channels).map(|channel| {
264                                pattern.power_at(separation, window.frequency_hz(channel))
265                            }));
266                        }
267                        powers
268                    })
269                    .collect()
270            });
271            fields[field] = Some(FieldTerms {
272                phase,
273                projections,
274                powers,
275            });
276        }
277        Ok(Self {
278            directions,
279            frame: targets.frame.clone(),
280            transforms,
281            windows,
282            fields,
283            of_scan: targets.pointing_of_scan.to_vec(),
284            _reservation: reservation,
285        })
286    }

288    /// Bytes the prediction holds beside the list it was read from: its
289    /// fluxes, the beam's powers, the shapes' transforms, the pointings'
290    /// rotations and the scans' pointings.
291    pub(crate) fn bytes(&self) -> u64 {
292        let windows: usize = self
293            .windows
294            .iter()
295            .map(|window| {
296                size_of::<PreparedWindow>() + window.fluxes.len() * size_of::<Complex64>()
297            })
298            .sum();
299        let fields: usize = self
300            .fields
301            .iter()
302            .map(|field| {
303                size_of::<Option<FieldTerms>>()
304                    + field.as_ref().map_or(0, |terms| {
305                        let powers = terms.powers.as_ref().map_or(0, |windows| {
306                            windows
307                                .iter()
308                                .map(|powers| {
309                                    size_of::<Vec<f32>>() + powers.len() * size_of::<f32>()
310                                })
311                                .sum()
312                        });
313                        terms.projections.len() * size_of::<UvwMachine>() + powers
314                    })
315            })
316            .sum();
317        (self.transforms.len() * size_of::<ShapeTransform>()
318            + self.directions.len() * size_of::<MDirection>()
319            + windows
320            + fields
321            + self.of_scan.len() * size_of::<usize>()) as u64
322    }

324    pub(crate) fn needs_epoch(&self) -> bool {
325        self.directions
326            .iter()
327            .any(|direction| direction.refer() != DirectionRef::J2000)
328    }

330    /// Add to `out` the visibilities of `rows` (each row's raw uvw in
331    /// metres) of scan `scan` with DATA_DESCRIPTION `description`, at the
332    /// scan's pointing. A description supplies both frequencies and products.
333    /// `out` holds the rows' samples back to back,
334    /// `[row][channel][correlation]`.
335    pub(crate) fn add_rows(
336        &self,
337        (scan, description): (usize, usize),
338        rows: impl Iterator<Item = ([f64; 3], f64)> + Clone,
339        out: &mut [Complex32],
340    ) -> Result<(), casa_components::ComponentError> {
341        let Some(terms) = self
342            .of_scan
343            .get(scan)
344            .and_then(|&pointing| self.fields.get(pointing))
345            .and_then(Option::as_ref)
346        else {
347            // A prediction covers the scans observed before it, whose
348            // pointings it prepared.
349            debug_assert!(false, "scan {scan} was not observed before the prediction");
350            return Ok(());
351        };
352        let prepared = &self.windows[description];
353        let (channels, correlations) = (prepared.channels, prepared.correlations);
354        let powers = terms
355            .powers
356            .as_ref()
357            .map(|powers| powers[description].as_slice());
358        for (component, (transform, initial_projection)) in
359            self.transforms.iter().zip(&terms.projections).enumerate()
360        {
361            let direction = &self.directions[component];
362            let mut projection = *initial_projection;
363            let mut epoch = None;
364            for (row, (uvw_m, time_s)) in rows.clone().enumerate() {
365                if direction.refer() != DirectionRef::J2000 && epoch != Some(time_s) {
366                    let current = self
367                        .frame
368                        .clone()
369                        .with_epoch(MEpoch::from_mjd(time_s / 86400., EpochRef::UTC));
370                    projection = UvwMachine::with_frame(direction, &terms.phase, &current)
371                        .map_err(|error| casa_components::ComponentError::Visibility {
372                            shape: "component direction",
373                            reason: error.to_string(),
374                        })?;
375                    epoch = Some(time_s);
376                }
377                let samples = &mut out[row * channels * correlations..][..channels * correlations];
378                let (rotated, phase) =
379                    projection.convert_projected([-uvw_m[0], -uvw_m[1], uvw_m[2]]);
380                let rotated = [-rotated[0], -rotated[1], rotated[2]];
381                let phase_shift_m = -phase;
382                let baseline = transform.at_baseline(rotated);
383                // e^{−2πi s ν/c}, advanced a channel at a time.
384                let radians_per_hz =
385                    -std::f64::consts::TAU * phase_shift_m / SPEED_OF_LIGHT_M_PER_S;
386                let mut phasor = Complex64::from_polar(1.0, radians_per_hz * prepared.first_hz);
387                let step = Complex64::from_polar(1.0, radians_per_hz * prepared.step_hz);
388                let fluxes = &prepared.fluxes[component * channels * correlations..]
389                    [..channels * correlations];
390                let powers = powers.map(|powers| &powers[component * channels..][..channels]);
391                for channel in 0..channels {
392                    let frequency_hz = prepared.stored_frequencies.as_ref().map_or_else(
393                        || prepared.first_hz + channel as f64 * prepared.step_hz,
394                        |frequencies| frequencies.value[channel],
395                    );
396                    let channel_phasor = if prepared.stored_frequencies.is_some() {
397                        Complex64::from_polar(1.0, radians_per_hz * frequency_hz)
398                    } else {
399                        phasor
400                    };
401                    let mut scale = baseline.at_frequency(frequency_hz)?;
402                    if let Some(powers) = powers {
403                        scale *= f64::from(powers[channel]);
404                    }
405                    let value = channel_phasor * scale;
406                    let cells = channel * correlations..(channel + 1) * correlations;
407                    for (cell, flux) in samples[cells.clone()].iter_mut().zip(&fluxes[cells]) {
408                        let sample = flux * value;
409                        *cell += Complex32::new(sample.re as f32, sample.im as f32);
410                    }
411                    phasor *= step;
412                }
413            }
414        }
415        Ok(())
416    }
417}

419/// `window`'s components' fluxes in its correlations
420/// ([`PreparedWindow::fluxes`]).
421fn prepare_window(
422    components: &[SkyComponent],
423    window: &ComponentWindow,
424) -> Result<PreparedWindow, ComponentRefusal> {
425    let refused = ComponentRefusal::NotYetImplemented;
426    let basis = |code: i32| match code {
427        5..=8 => Some((Polarisation::Circular, (code - 5) as usize)),
428        9..=12 => Some((Polarisation::Linear, (code - 9) as usize)),
429        _ => None,
430    };
431    let mapped = window
432        .correlations
433        .iter()
434        .map(|&code| basis(code))
435        .collect::<Option<Vec<_>>>()
436        .filter(|mapped| mapped.windows(2).all(|pair| pair[0].0 == pair[1].0))
437        .ok_or_else(|| {
438            refused(format!(
439                "predicting components into the correlations {:?}: casa-rs predicts them into \
440                 circular or linear correlations",
441                window.correlations
442            ))
443        })?;
444    let polarisation = mapped
445        .first()
446        .map_or(Polarisation::Circular, |(basis, _)| *basis);
447    let mut fluxes = Vec::with_capacity(components.len() * window.channels * mapped.len());
448    for (index, component) in components.iter().enumerate() {
449        for channel in 0..window.channels {
450            let flux = component
451                .flux_in(window.frequency_hz(channel), polarisation)
452                .map_err(|error| refused(format!("component {index}: {error}")))?;
453            fluxes.extend(mapped.iter().map(|&(_, basis)| flux[basis]));
454        }
455    }
456    Ok(PreparedWindow {
457        stored_frequencies: window.stored_frequencies.clone(),
458        first_hz: window.first_hz,
459        step_hz: window.step_hz,
460        channels: window.channels,
461        correlations: mapped.len(),
462        fluxes,
463    })
464}

466/// The great-circle separation of two J2000 directions, radians
467/// (`MVDirection::separation`: twice the arcsine of half the chord).
468fn separation_rad(a: [f64; 2], b: [f64; 2]) -> f64 {
469    let unit = |[ra, dec]: [f64; 2]| [dec.cos() * ra.cos(), dec.cos() * ra.sin(), dec.sin()];
470    let (a, b) = (unit(a), unit(b));
471    let chord = ((a[0] - b[0]).powi(2) + (a[1] - b[1]).powi(2) + (a[2] - b[2]).powi(2)).sqrt();
472    2.0 * (chord / 2.0).asin()
473}
