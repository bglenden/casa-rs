2//! Faceted imaging (#664): CASA's `facets=N` splits the main image into N×N
3//! facets, each gridded about its own centre by a mapper of its own
4//! (`SynthesisImager::createFacetImageStoreList`, `SIMapperCollection`).
5//! Here every facet is a *chart*: an operator of the facet's size, a pass
6//! domain of the one traversal, onto which casa-ms projects each row with
7//! CASA's `UVWMachine(project=true)`. The pass sees charts; the normal state,
8//! the minor cycle and the products see the compiled image domains, whose
9//! images [`ChartStitch`] assembles from their charts.
10

20/// One operator's share of an image domain: a facet of a faceted domain, or
21/// the whole domain.
22#[derive(Clone, Copy, Debug, PartialEq, Eq)]
23pub(crate) struct Chart {
24    /// The compiled image domain.
25    pub(crate) domain: usize,
26    /// The facet's pixels when the domain is faceted.
27    pub(crate) window: Option<FacetBox>,
28}
29
30/// A facet's pixels in its domain: `origin` inclusive, `end` exclusive.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub(crate) struct FacetBox {
33    pub(crate) origin: [usize; 2],
34    pub(crate) end: [usize; 2],
35}
36
37impl FacetBox {
38    /// The facet's `[width, height]`.
39    pub(crate) const fn shape(self) -> [usize; 2] {
40        [self.end[0] - self.origin[0], self.end[1] - self.origin[1]]
41    }
42}
43
44/// The charts of `problem` in the order its rows carry their projections:
45/// domain-major, a faceted domain's facets in CASA's order (facet
46/// `row·N + column`, facet 0 at the bottom left).
47pub(crate) fn charts(problem: &CompiledProblem) -> Vec<Chart> {
48    problem
49        .geometry()
50        .domains()
51        .iter()
52        .enumerate()
53        .flat_map(|(domain, compiled)| {
54            let faceted = compiled.facets().len() > 1;
55            compiled.facets().iter().map(move |window| Chart {
56                domain,
57                window: faceted.then_some(FacetBox {
58                    origin: window.origin(),
59                    end: window.end_exclusive(),
60                }),
61            })
62        })
63        .collect()
64}

66impl Chart {
67    /// The extent of the chart's operator: the domain's, or the facet's,
68    /// whose reference pixel is its centre (CASA grids a facet about the
69    /// world direction of its pixel `shape / 2`, `FTMachine::initMaps`).
70    pub(crate) fn extent(
71        &self,
72        compiled: &CompiledImageDomain,
73    ) -> Result<ImageExtent, ImagingError> {
74        let Some(window) = self.window else {
75            return image_extent(compiled);
76        };
77        let shape = window.shape();
78        Ok(ImageExtent {
79            shape,
80            increment_rad: compiled.direction().increment_rad(),
81            // `FacetWindow::local_centre_pixel`: the facet's phase centre.
82            reference_pixel: [shape[0] / 2, shape[1] / 2],
83        })
84    }
85}

87/// CASA's empty-plane threshold: a facet store whose `sumwt` is at most this
88/// sets its image to zero rather than divide (`divideImageByWeightVal`).
89const EMPTY_SUM_WEIGHT: f64 = 1e-7;
90
91/// One facet's sums of weights per (term, polarization), from the initial
92/// pass.
93#[derive(Clone, Debug)]
94struct FacetSums {
95    /// The PSF gridding's, which normalise the PSF.
96    psf: Vec<f64>,
97    /// CASA's `.sumwt`, which normalises the facet's residual.
98    published: Vec<f64>,
99}
100
101impl FacetSums {
102    fn populated(&self) -> bool {
103        self.published.iter().all(|sum| *sum > EMPTY_SUM_WEIGHT)
104    }
105}

107/// A faceted domain's images as its charts arrive in one pass: each facet's
108/// residual unscaled at its window, and on the initial pass the PSF of the
109/// lowest populated facet so far.
110struct Assembly {
111    images: PassImages,
112    arrived: usize,
113    psf: Option<(usize, Vec<f32>)>,
114}
115
116/// Why a faceted domain's images could not be stitched; each is a broken
117/// invariant of the pass, not a property of the data.
118#[derive(Debug, thiserror::Error)]
119pub(crate) enum StitchError {
120    /// A facet's residual arrived before its initial pass had given its
121    /// sums of weights.
122    #[error("facet {chart}'s residual arrived before its initial pass's sums of weights")]
123    UnscaledFacet { chart: usize },
124    /// The domain's facets disagree on their polarizations or planes.
125    #[error("facet {chart} does not match its domain's other facets")]
126    MismatchedFacet { chart: usize },
127}
128
129/// The images a pass hands the normal state: an unfaceted domain's as its
130/// chart forms them, a faceted domain's stitched from its facets as CASA's
131/// normalizer reads them through each facet store
132/// (`SynthesisNormalizer`), with a populated reference facet `r`, the lowest
133/// whose `sumwt` exceeds 1e-7 in every polarization (facet 0 unless it is
134/// empty):
135/// - each facet's residual divided by that facet's own `sumwt`, here by
136///   scaling facet `f` by `s_r / s_f` before the normal state divides the
137///   whole by `s_r`; an empty facet is zero, as `divideImageByWeightVal`
138///   leaves it;
139/// - facet `r`'s PSF, placed at the image centre in an image of zeros
140///   (`setPsfFromOneFacet`, which takes facet 0), which the reconstruction
141///   normalises by its peak;
142/// - facet `r`'s sums of weights for the domain.
143///
144/// Each facet's own sums are published in its pixel of `.sumwt`
145/// ([`ChartStitch::facet_sum_weights`]).
146pub(crate) struct ChartStitch {
147    charts: Vec<Chart>,
148    /// Each image domain's `[width, height]`.
149    shapes: Vec<[usize; 2]>,
150    /// Each chart's sums, from the initial pass.
151    sums: Vec<Option<FacetSums>>,
152    assemblies: BTreeMap<usize, Assembly>,
153    /// The requested polarizations, one image each per plane and term.
154    pols: usize,
155}
156
157impl ChartStitch {
158    /// The stitch of `problem`'s `charts`.
159    pub(crate) fn new(problem: &CompiledProblem, charts: Vec<Chart>) -> Self {
160        let shapes = problem
161            .geometry()
162            .domains()
163            .iter()
164            .map(|domain| domain.shape().pixels())
165            .collect();
166        let sums = vec![None; charts.len()];
167        Self {
168            charts,
169            shapes,
170            sums,
171            assemblies: BTreeMap::new(),
172            pols: problem.reconstruction().polarization().coordinates().len(),
173        }
174    }
175
176    /// Bytes a pass's stitch holds beside the pass's own images: each
177    /// faceted domain's full-size residual, and on an initial pass (`psf`)
178    /// its full-size PSF and one facet's PSF. Faceted domains have a
179    /// constant basis: one plane, one term.
180    pub(crate) fn pass_bytes(&self, psf: bool) -> u64 {
181        let image = |pixels: [usize; 2]| (pixels[0] * pixels[1] * self.pols * 4) as u64;
182        self.charts
183            .iter()
184            .enumerate()
185            .filter(|(chart, candidate)| {
186                candidate.window.is_some() && self.first_chart(candidate.domain) == *chart
187            })
188            .map(|(_, chart)| {
189                let full = image(self.shapes[chart.domain]);
190                let facet = chart.window.map_or(0, |window| image(window.shape()));
191                if psf { 2 * full + facet } else { full }
192            })
193            .sum()
194    }
195
196    /// Each faceted domain's facets' published sums of weights, per facet
197    /// in CASA's order and per polarization, from the initial pass:
198    /// CASA's `.sumwt` of a faceted image holds each facet store's own.
199    pub(crate) fn facet_sum_weights(&self) -> BTreeMap<usize, Vec<Vec<f64>>> {
200        let mut domains = BTreeMap::<usize, Vec<Vec<f64>>>::new();
201        for (chart, candidate) in self.charts.iter().enumerate() {
202            if candidate.window.is_none() {
203                continue;
204            }
205            if let Some(sums) = &self.sums[chart] {
206                domains
207                    .entry(candidate.domain)
208                    .or_default()
209                    .push(sums.published.clone());
210            }
211        }
212        domains
213    }
214
215    fn first_chart(&self, domain: usize) -> usize {
216        self.charts
217            .iter()
218            .position(|candidate| candidate.domain == domain)
219            .expect("a chart of the domain")
220    }
221
222    /// Accept chart `chart`'s images; the domain's images once its last
223    /// chart has arrived, at once for an unfaceted domain.
224    pub(crate) fn accept(
225        &mut self,
226        chart: usize,
227        mut images: PassImages,
228    ) -> Result<Option<PassImages>, StitchError> {
229        let Chart { domain, window } = self.charts[chart];
230        images.domain = domain;
231        let Some(window) = window else {
232            return Ok(Some(images));
233        };
234        if images.psf.is_some() {
235            self.sums[chart] = Some(FacetSums {
236                psf: images.sum_weights.clone(),
237                published: images.published_sum_weights.clone(),
238            });
239        }
240        let populated = self.sums[chart]
241            .as_ref()
242            .ok_or(StitchError::UnscaledFacet { chart })?
243            .populated();
244        let full = self.shapes[domain];
245        let assembly = self.assemblies.entry(domain).or_insert_with(|| Assembly {
246            images: empty_domain_images(&images, full),
247            arrived: 0,
248            psf: None,
249        });
250        if assembly.images.residual.len() / (full[0] * full[1])
251            != images.residual.len() / (images.shape[0] * images.shape[1])
252        {
253            return Err(StitchError::MismatchedFacet { chart });
254        }
255        let facet = [images.shape[0], images.shape[1]];
256        scatter(
257            &images.residual,
258            &mut assembly.images.residual,
259            facet,
260            full,
261            window.origin,
262            |_| 1.0,
263        );
264        // Keep the PSF of the lowest populated facet, whatever the order the
265        // facets arrive in.
266        if let Some(psf) = images.psf.take()
267            && populated
268            && assembly.psf.as_ref().is_none_or(|(held, _)| chart < *held)
269        {
270            assembly.psf = Some((chart, psf));
271        }
272        assembly.arrived += 1;
273        let facets = self
274            .charts
275            .iter()
276            .filter(|candidate| candidate.domain == domain)
277            .count();
278        if assembly.arrived < facets {
279            return Ok(None);
280        }
281        let assembly = self
282            .assemblies
283            .remove(&domain)
284            .expect("the domain's assembly");
285        self.finish(domain, assembly).map(Some)
286    }
287
288    /// Scale each facet's window of a complete assembly by `s_r / s_f`
289    /// (zero for an empty facet), place the reference facet's PSF and give
290    /// the domain the reference's sums.
291    fn finish(&self, domain: usize, mut assembly: Assembly) -> Result<PassImages, StitchError> {
292        let charts = self
293            .charts
294            .iter()
295            .enumerate()
296            .filter(|(_, candidate)| candidate.domain == domain)
297            .map(|(chart, candidate)| {
298                let sums = self.sums[chart]
299                    .as_ref()
300                    .ok_or(StitchError::UnscaledFacet { chart })?;
301                Ok((chart, *candidate, sums))
302            })
303            .collect::<Result<Vec<_>, _>>()?;
304        // Every facet empty: facet 0's (empty) sums make the normal state
305        // zero the whole image.
306        let reference = charts
307            .iter()
308            .find(|(_, _, sums)| sums.populated())
309            .unwrap_or(&charts[0]);
310        let full = self.shapes[domain];
311        let pols = assembly.images.polarizations;
312        for (_, chart, sums) in &charts {
313            let window = chart.window.expect("a faceted domain's chart has a window");
314            let scales = (0..pols)
315                .map(|pol| {
316                    let own = sums.published[pol];
317                    if own > EMPTY_SUM_WEIGHT {
318                        reference.2.published[pol] / own
319                    } else {
320                        0.0
321                    }
322                })
323                .collect::<Vec<_>>();
324            rescale(&mut assembly.images.residual, window, full, |image| {
325                scales[image % pols]
326            });
327        }
328        if let Some(stitched) = assembly.images.psf.as_mut()
329            && let Some((held, psf)) = assembly.psf.take()
330        {
331            debug_assert_eq!(held, reference.0, "the held PSF is the reference facet's");
332            let facet = reference.1.window.expect("a facet").shape();
333            let centred = [(full[0] - facet[0]) / 2, (full[1] - facet[1]) / 2];
334            scatter(&psf, stitched, facet, full, centred, |_| 1.0);
335        }
336        assembly.images.sum_weights.clone_from(&reference.2.psf);
337        assembly
338            .images
339            .published_sum_weights
340            .clone_from(&reference.2.published);
341        Ok(assembly.images)
342    }
343}

345/// Scale `window` of each x-major `full` image in `images` by `scale(i)`.
346fn rescale(images: &mut [f32], window: FacetBox, full: [usize; 2], scale: impl Fn(usize) -> f64) {
347    let full_cells = full[0] * full[1];
348    let height = window.shape()[1];
349    for (image, values) in images.chunks_exact_mut(full_cells).enumerate() {
350        let factor = scale(image);
351        for x in window.origin[0]..window.end[0] {
352            let start = x * full[1] + window.origin[1];
353            for cell in &mut values[start..start + height] {
354                *cell = (f64::from(*cell) * factor) as f32;
355            }
356        }
357    }
358}

360/// Zeroed images of a whole `full` domain shaped like the facet `images`.
361fn empty_domain_images(images: &PassImages, full: [usize; 2]) -> PassImages {
362    let facet_cells = images.shape[0] * images.shape[1];
363    let full_cells = full[0] * full[1];
364    let zeros = |values: &[f32]| vec![0.0; values.len() / facet_cells * full_cells];
365    PassImages {
366        domain: images.domain,
367        shape: full,
368        channels: images.channels.clone(),
369        polarizations: images.polarizations,
370        residual: zeros(&images.residual),
371        psf: images.psf.as_deref().map(zeros),
372        sum_weights: Vec::new(),
373        published_sum_weights: Vec::new(),
374        weight: None,
375    }
376}

378/// Copy each x-major `facet` image of `from` into the x-major `full` image
379/// of `into` at pixel `origin`, image `i` scaled by `scale(i)`.
380fn scatter(
381    from: &[f32],
382    into: &mut [f32],
383    facet: [usize; 2],
384    full: [usize; 2],
385    origin: [usize; 2],
386    scale: impl Fn(usize) -> f64,
387) {
388    let (facet_cells, full_cells) = (facet[0] * facet[1], full[0] * full[1]);
389    for (image, values) in from.chunks_exact(facet_cells).enumerate() {
390        let factor = scale(image);
391        let target = &mut into[image * full_cells..(image + 1) * full_cells];
392        for (x, column) in values.chunks_exact(facet[1]).enumerate() {
393            let start = (origin[0] + x) * full[1] + origin[1];
394            for (cell, value) in target[start..start + facet[1]].iter_mut().zip(column) {
395                *cell = (f64::from(*value) * factor) as f32;
396            }
397        }
398    }

405    /// Facet images of a `[4, 2]` facet with one polarization: residual
406    /// value `10·facet + cell`, PSF `100 + cell` on facet 0.
407    fn facet(sum: f64, values: impl Fn(usize) -> f32, initial: bool) -> PassImages {
408        PassImages {
409            domain: 9,
410            shape: [4, 2],
411            channels: 0..1,
412            polarizations: 1,
413            residual: (0..8).map(&values).collect(),
414            psf: initial.then(|| (0..8).map(|cell| 100.0 + cell as f32).collect()),
415            sum_weights: if initial { vec![sum] } else { Vec::new() },
416            published_sum_weights: if initial { vec![sum] } else { Vec::new() },
417            weight: None,
418        }
419    }

421    const ORIGINS: [[usize; 2]; 4] = [[0, 0], [4, 0], [0, 2], [4, 2]];
422
423    /// Four `[4, 2]` facets of an `[8, 4]` domain in CASA's order.
424    fn stitch() -> ChartStitch {
425        let charts = ORIGINS
426            .map(|origin| Chart {
427                domain: 0,
428                window: Some(FacetBox {
429                    origin,
430                    end: [origin[0] + 4, origin[1] + 2],
431                }),
432            })
433            .to_vec();
434        ChartStitch {
435            charts,
436            shapes: vec![[8, 4]],
437            sums: vec![None; 4],
438            assemblies: BTreeMap::new(),
439            pols: 1,
440        }
441    }

443    /// The initial pass of facets with `sums`, arriving in `order`, residual
444    /// `10·facet + cell + 1`: the stitched images.
445    fn initial(stitch: &mut ChartStitch, sums: [f64; 4], order: [usize; 4]) -> PassImages {
446        let mut stitched = None;
447        for (arrived, index) in order.into_iter().enumerate() {
448            let images = facet(sums[index], |cell| (10 * index + cell + 1) as f32, true);
449            stitched = stitch.accept(index, images).unwrap();
450            assert_eq!(stitched.is_some(), arrived == 3, "facet {index}");
451        }
452        stitched.unwrap()
453    }

455    /// The value of facet `index`'s cell `(x, y)` in a stitched `[8, 4]`
456    /// residual.
457    fn at(residual: &[f32], index: usize, x: usize, y: usize) -> f64 {
458        f64::from(residual[(ORIGINS[index][0] + x) * 4 + ORIGINS[index][1] + y])
459    }

461    /// Each facet lands at its window divided by its own sum, here scaled
462    /// by facet 0's over its own; facet 0's PSF lands centred; each facet's
463    /// own sum is published; a refresh reuses the initial pass's sums.
464    #[test]
465    fn facets_scatter_to_their_windows_each_divided_by_its_own_sum() {
466        let mut stitch = stitch();
467        // Facet sums 2, 4, 2, 1: facet f's residual scales by 2 / s_f.
468        let sums = [2.0, 4.0, 2.0, 1.0];
469        let stitched = initial(&mut stitch, sums, [0, 1, 2, 3]);
470        assert_eq!((stitched.domain, stitched.shape), (0, [8, 4]));
471        assert_eq!(stitched.sum_weights, vec![2.0]);
472        assert_eq!(stitched.published_sum_weights, vec![2.0]);
473        for (index, sum) in sums.into_iter().enumerate() {
474            for x in 0..4 {
475                for y in 0..2 {
476                    let cell = x * 2 + y;
477                    let expected = (10 * index + cell + 1) as f64 * 2.0 / sum;
478                    assert_eq!(
479                        at(&stitched.residual, index, x, y),
480                        expected,
481                        "facet {index} cell {cell}"
482                    );
483                }
484            }
485        }
486        // Facet 0's PSF, centred: its centre pixel (2, 1) lands on (4, 2).
487        let psf = stitched.psf.unwrap();
488        assert_eq!(psf[4 * 4 + 2], 100.0 + (2 * 2 + 1) as f32);
489        assert_eq!(psf.iter().filter(|value| **value != 0.0).count(), 8);
490        assert_eq!(
491            stitch.facet_sum_weights(),
492            BTreeMap::from([(0, sums.map(|sum| vec![sum]).to_vec())])
493        );
494        // A refresh pass carries no sums and reuses the initial pass's.
495        let mut stitched = None;
496        for index in 0..4 {
497            stitched = stitch.accept(index, facet(0.0, |_| 1.0, false)).unwrap();
498        }
499        let residual = stitched.unwrap().residual;
500        assert_eq!(at(&residual, 1, 0, 0), 2.0 / 4.0);
501        assert_eq!(at(&residual, 3, 0, 0), 2.0 / 1.0);
502    }

504    /// A facet whose `sumwt` is at most 1e-7 (rows that land in other
505    /// facets' grids but off its own) is zero, as CASA's
506    /// `divideImageByWeightVal` leaves an empty facet store, on the initial
507    /// pass and every refresh, while the populated facets keep their own
508    /// normalisation; its own (empty) sum is published.
509    #[test]
510    fn an_empty_facet_is_zero_and_the_others_keep_their_own_sums() {
511        let mut stitch = stitch();
512        let sums = [2.0, 0.0, 2.0, 1.0];
513        let stitched = initial(&mut stitch, sums, [0, 1, 2, 3]);
514        assert!(stitched.residual.iter().all(|value| value.is_finite()));
515        assert_eq!(at(&stitched.residual, 1, 0, 0), 0.0);
516        assert_eq!(at(&stitched.residual, 3, 0, 0), 31.0 * 2.0 / 1.0);
517        assert_eq!(at(&stitched.residual, 0, 0, 0), 1.0);
518        assert_eq!(stitched.sum_weights, vec![2.0]);
519        assert_eq!(stitch.facet_sum_weights()[&0][1], vec![0.0]);
520        let mut refreshed = None;
521        for index in 0..4 {
522            refreshed = stitch.accept(index, facet(0.0, |_| 1.0, false)).unwrap();
523        }
524        let residual = refreshed.unwrap().residual;
525        assert!(residual.iter().all(|value| value.is_finite()));
526        assert_eq!(at(&residual, 1, 0, 0), 0.0);
527        assert_eq!(at(&residual, 2, 0, 0), 1.0);
528    }

530    /// An empty facet 0 does not empty the image: the lowest populated
531    /// facet is the reference, its sums the domain's and its PSF the
532    /// image's (CASA's `setPsfFromOneFacet` would take facet 0's empty

537        let sums = [0.0, 4.0, 2.0, 1.0];
538        let stitched = initial(&mut stitch, sums, [3, 2, 1, 0]);

541        assert_eq!(at(&stitched.residual, 0, 0, 0), 0.0);

546        assert_eq!(at(&stitched.residual, 3, 0, 0), 31.0 * 4.0);
547        // Facet 1's PSF (every facet's PSF is 100 + cell), centred.
548        let psf = stitched.psf.unwrap();
549        assert_eq!(psf[4 * 4 + 2], 100.0 + (2 * 2 + 1) as f32);
550        assert_eq!(psf.iter().filter(|value| **value != 0.0).count(), 8);
