2//! Merge only the recorded coordinates addressed by a visibility write.
3
4use ndarray::{ArrayD, Axis, IxDyn};
5
6use super::super::VisibilityChannels;
7
8pub(super) fn patch_cell<T: Copy>(
9    original: &mut ArrayD<T>,
10    selected: &ArrayD<T>,
11    correlations: Option<&[usize]>,
12    channels: Option<VisibilityChannels>,
13) {
14    if let Some(channels) = channels {
15        for (bin, span) in channels.ranges().enumerate() {
16            let selected = selected.index_axis(Axis(1), bin);
17            for channel in span {
18                let mut original = original.index_axis_mut(Axis(1), channel);
19                if let Some(indices) = correlations {
20                    for (input, &output) in indices.iter().enumerate() {
21                        original[IxDyn(&[output])] = selected[IxDyn(&[input])];
22                    }
23                } else {
24                    original.assign(&selected);
25                }
26            }
27        }
28    } else if let Some(indices) = correlations {
29        for (input, &output) in indices.iter().enumerate() {
30            original
31                .index_axis_mut(Axis(0), output)
32                .assign(&selected.index_axis(Axis(0), input));
33        }
34    }
35}

37pub(super) fn patch_converted_flags(
38    original: &mut ArrayD<bool>,
39    selected: &ArrayD<bool>,
40    dependencies: &[&[usize]],
41    channels: Option<VisibilityChannels>,
42) {
43    let mut patch_bin = |bin, span: std::ops::Range<usize>| {
44        for correlation in 0..original.shape()[0] {
45            let mut flag = None;
46            for (product, contributors) in dependencies.iter().enumerate() {
47                if contributors.contains(&correlation) {
48                    flag = Some(flag.unwrap_or(false) || selected[IxDyn(&[product, bin])]);
49                }
50            }
51            if let Some(flag) = flag {
52                for channel in span.clone() {
53                    original[IxDyn(&[correlation, channel])] = flag;
54                }
55            }
56        }
57    };
58    if let Some(channels) = channels {
59        for (bin, span) in channels.ranges().enumerate() {
60            patch_bin(bin, span);
61        }
62    } else {
63        for channel in 0..selected.shape()[1] {
64            patch_bin(channel, channel..channel + 1);
65        }
66    }
67}
