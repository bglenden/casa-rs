2//! Allocation bounds for shared container representations.
3
4/// Bound the node allocations of a nonempty `BTreeMap<K, V>` or
5/// `BTreeSet<K>` (with `V = ()`), excluding heap owned by keys and values.
6///
7/// Rust's degree-six tree has eleven key/value slots and twelve edges per
8/// internal node. The root may hold one entry; every other node holds at
9/// least five. Charging an internal node for each possible node therefore
10/// also covers partially filled trees and small maps with one allocated
11/// leaf. Slots are charged at the pair's padded size, with aligned headers
12/// and edges, so separate key/value arrays fit within the bound.
13///
14/// Zero entries returns zero for a freshly constructed empty map. A map
15/// emptied by removal may retain its root; charge one entry in that case.
16#[must_use]
17pub const fn btree_node_bytes<K, V>(entries: usize) -> usize {
18    if entries == 0 {
19        return 0;
20    }
21    const SLOTS: usize = 11;
22    const MIN_ENTRIES: usize = 5;
23    let alignment = if align_of::<(K, V)>() > align_of::<usize>() {
24        align_of::<(K, V)>()
25    } else {
26        align_of::<usize>()
27    };
28    let header = (2 * size_of::<usize>()).next_multiple_of(alignment);
29    let node = (header + SLOTS * size_of::<(K, V)>() + (SLOTS + 1) * size_of::<usize>())
30        .next_multiple_of(alignment);
31    entries.div_ceil(MIN_ENTRIES).saturating_mul(node)
32}
