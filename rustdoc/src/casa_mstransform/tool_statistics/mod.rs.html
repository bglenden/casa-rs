29/// Selection, projection and reporting options for a public MS tool reduction.
30pub struct ToolStatisticsOptions {
31    /// CASA column name or alias.
32    pub column: String,
33    /// Projection for complex values: real, imaginary, amplitude or phase.
34    pub complex_value: String,
35    /// Omit flagged values according to the column's mask.
36    pub use_flags: bool,
37    /// Use per-visibility weights when the selected column supports them.
38    pub use_weights: bool,
39    /// CASA selection expressions applied to the current logical attachment.
40    pub selection: MsSelection,
41    /// Comma-separated axes for the returned statistics records.
42    pub reporting_axes: String,
43    /// Average visibility vectors in time before projecting to real samples.
44    pub time_average: bool,
45    /// CASA duration quantity for the averaging bins.
46    pub time_bin: String,
47    /// Whether the bins are allowed to span scan or state boundaries.
48    pub time_span: String,
49    /// Maximum separation in metres between baselines in the same bin.
50    pub max_uvw_distance: f64,
51    /// Compute median, median absolute deviation and the requested quartiles.
52    pub quantiles: bool,
53    /// Use the deprecated statisticsold projection, masking and record format.
54    pub legacy: bool,
55}

57/// Compute statistics for the logical rows and spectral axes seen by the MS tool.
58///
59/// # Errors
60/// Returns an error for invalid selection, shapes or physical column metadata.
61pub fn tool_statistics(
62    ms: &MeasurementSet,
63    options: &ToolStatisticsOptions,
64    model_input: &dyn TransformInputView,
65) -> MsResult<RecordValue> {
66    let selection = if options.selection.has_text_selection() {
67        Some(resolve_casa_selection_with_system_budget(
68            ms,
69            &options.selection,
70        )?)
71    } else {
72        None
73    };
74    let (label, column) = Column::parse(&options.column, &options.complex_value, options.legacy)?;
75    let Some(column) = column else {
76        return Ok(RecordValue::default());
77    };
78    let column = column.source_for_view(ms.main_table(), options.legacy)?;
79    let time_average = options
80        .time_average
81        .then(|| TimeAverage::new(ms, options, column))
82        .transpose()?;
83    let selected = selection
84        .as_ref()
85        .map(|selection| selection.row_indices().collect::<Vec<_>>());
86    let mut rows = read_rows(ms.main_table(), selected.as_deref())?;
87    if rows.is_empty() {
88        return Err(MsError::InvalidInput(if options.legacy {
89            "No data could be found!".into()
90        } else {
91            "MSIter::construct - Input MeasurementSet.has zero selected rows".into()
92        }));
93    }
94    let model = prepare_model(ms, column, options, selection.as_ref(), model_input)?;
95    let packed = !options.legacy
96        && time_average.is_none()
97        && model.is_none()
98        && matches!(column, Column::Complex(name, _) if name != "FLOAT_DATA")
99        && ms
100            .path()
101            .map(|path| Table::immediate_part_names(path, path))
102            .transpose()?
103            .is_some_and(|parts| matches!(parts, TablePartNames::Plain(_)));
104    let mut reporting = Axis::parse_axes(&options.reporting_axes);
105    if time_average.is_some() {
106        TimeAverage::reporting_axes(&mut reporting, &options.time_span);
107    }
108    if options.legacy {
109        sort_rows(&mut rows, &CHUNK_AXES);
110    } else {
111        sort_rows(&mut rows, &sort_order(&reporting));
112    }
113    let mut reducer = Reduction {
114        ms,
115        selection: selection.as_ref(),
116        grid: BTreeMap::new(),
117        cube: CubeReduction {
118            use_flags: options.use_flags,
119            weighting: column.weighting(options.use_weights),
120            legacy: options.legacy,
121        },
122        column,
123        model,
124        packed,
125    };
126    if options.legacy {
127        return reducer.legacy(rows, &label);
128    }
129
130    let mut record = RecordValue::default();
131    let mut remaining = rows.as_slice();
132    while let Some(first) = remaining.first() {
133        let mut moments = Moments::new(options.quantiles);
134        let (count, averaged_start) = if let Some(average) = time_average.as_ref() {
135            let chunks = average.report_chunks(remaining, &reporting);
136            let range = chunks.iter().map(|chunk| chunk.len()).sum();
137            let averaged_start = reducer.reduce_averaged(average, &chunks, &mut moments)?;
138            (range, Some(averaged_start))
139        } else {
140            let range = remaining
141                .iter()
142                .take_while(|row| same_group(first, row, &reporting))
143                .count();
144            reducer.reduce(&remaining[..range], &mut moments)?;
145            (range, None)
146        };
147        let key = reporting
148            .iter()
149            .map(|axis| match averaged_start {
150                Some(start) if *axis == Axis::Integration => format!("TIME={start:.6}"),
151                _ => first.group_item(*axis),
152            })
153            .collect::<Vec<_>>()
154            .join(",");
155        // FLOAT keeps reporting the mask and weights after averaging;
156        // CASA's complex providers consume those switches in averaging.
157        let report_mask_and_weights =
158            !options.time_average || matches!(column, Column::FloatCube("FLOAT_DATA"));
159        let stats = modern_record(
160            moments,
161            options.use_flags && report_mask_and_weights,
162            reducer.cube.weighting != Weighting::Unweighted && report_mask_and_weights,
163            options.quantiles,
164            column.is_boolean(),
165        );
166        record.upsert(key, Value::Record(stats));
167        remaining = &remaining[count..];
168    }
169    record
170        .fields_mut()
171        .sort_unstable_by(|a, b| a.name.cmp(&b.name));
172    Ok(record)
173}

175fn prepare_model<'a>(
176    ms: &'a MeasurementSet,
177    column: Column,
178    options: &ToolStatisticsOptions,
179    selection: Option<&ResolvedMsSelection>,
180    input: &dyn TransformInputView,
181) -> MsResult<Option<Box<dyn PreparedTransformInputView + 'a>>> {
182    if options.legacy || !matches!(column, Column::Complex("MODEL_DATA", _)) {
183        return Ok(None);
184    }
185    let stored = ms
186        .main_table()
187        .schema()
188        .is_some_and(|schema| schema.column("MODEL_DATA").is_some());
189    let virtual_fields = casa_ms::virtual_model_fields(ms);
190    if stored && virtual_fields.is_empty() {
191        return Ok(None);
192    }
193    let all_rows;
194    let selection = if let Some(selection) = selection {
195        selection
196    } else {
197        all_rows = resolve_casa_selection_with_system_budget(ms, &options.selection)?;
198        &all_rows
199    };
200    if stored
201        && !selection
202            .selected_rows
203            .iter()
204            .any(|row| virtual_fields.contains(&row.field_id))
205    {
206        return Ok(None);
207    }
208    Ok(Some(input.prepare(
209        ms,
210        &selection.selected_rows,
211        &BTreeMap::new(),
212    )?))
213}

215const CHUNK_AXES: [Axis; 4] = [
216    Axis::Array,
217    Axis::Field,
218    Axis::DataDescription,
219    Axis::Integration,
220];
221
222fn sort_order(reporting: &[Axis]) -> Vec<Axis> {
223    let mut axes = reporting.to_vec();
224    // VI sorts by the implicit integration axis before the remaining array,
225    // field or data-description keys, even if TIME is hidden in the record.
226    if !reporting.contains(&Axis::Integration) {
227        axes.push(Axis::Integration);
228    }
229    axes.extend(
230        CHUNK_AXES[..3]
231            .iter()
232            .copied()
233            .filter(|axis| !reporting.contains(axis)),
234    );
235    axes
236}
237
238fn sort_rows(rows: &mut [Row], axes: &[Axis]) {
239    rows.sort_by(|left, right| {
240        axes.iter()
241            .map(|axis| left.compare(right, *axis))
242            .find(|order| order.is_ne())
243            .unwrap_or(std::cmp::Ordering::Equal)
244    });
245}
246
247fn same_group(left: &Row, right: &Row, axes: &[Axis]) -> bool {
248    axes.iter().all(|axis| left.compare(right, *axis).is_eq())
249}

251struct Reduction<'a> {
252    ms: &'a MeasurementSet,
253    selection: Option<&'a ResolvedMsSelection>,
254    grid: BTreeMap<i32, SampleAxes>,
255    cube: CubeReduction,
256    column: Column,
257    model: Option<Box<dyn PreparedTransformInputView + 'a>>,
258    packed: bool,
259}
260
261impl Reduction<'_> {
262    fn reduce(&mut self, rows: &[Row], moments: &mut Moments) -> MsResult<()> {
263        let table = self.ms.main_table();
264        let mut dataset = 0;
265        let input = CubeInput {
266            table,
267            model: self.model.as_deref(),
268        };
269        match self.column {
270            Column::Complex(_, _) | Column::FloatCube(_) | Column::Flag => {
271                let descriptions = &CHUNK_AXES[..CHUNK_AXES.len() - 1];
272                for window in rows.chunk_by(|left, right| same_group(left, right, descriptions)) {
273                    let id = window[0].data_description;
274                    let axes = cached_sample_axes(&mut self.grid, self.ms, self.selection, id)?;
275                    if self.packed
276                        && let Some(buffers) = packed::reduce(
277                            &self.cube,
278                            table,
279                            window,
280                            axes,
281                            self.column,
282                            dataset,
283                            moments,
284                        )?
285                    {
286                        dataset += buffers;
287                    } else {
288                        for chunk in window
289                            .chunk_by(|left, right| same_group(left, right, &[Axis::Integration]))
290                        {
291                            self.cube
292                                .reduce(&input, chunk, axes, self.column, dataset, moments)?;
293                            dataset += 1;
294                        }
295                    }
296                }
297            }
298            column => {
299                for (dataset, chunk) in rows
300                    .chunk_by(|left, right| same_group(left, right, &CHUNK_AXES))
301                    .enumerate()
302                {
303                    scalar::reduce(
304                        table,
305                        chunk,
306                        column,
307                        self.cube.use_flags,
308                        self.cube.legacy,
309                        dataset,
310                        moments,
311                    )?;
312                }
313            }
314        }
315        Ok(())
316    }
317
318    fn reduce_averaged(
319        &mut self,
320        average: &TimeAverage,
321        chunks: &[&[Row]],
322        moments: &mut Moments,
323    ) -> MsResult<f64> {
324        let input = CubeInput {
325            table: self.ms.main_table(),
326            model: self.model.as_deref(),
327        };
328        let mut start = None;
329        for (dataset, chunk) in chunks.iter().enumerate() {
330            let id = chunk[0].data_description;
331            let axes = cached_sample_axes(&mut self.grid, self.ms, self.selection, id)?;
332            let chunk_start = average.reduce(&input, chunk, axes, self.column, dataset, moments)?;
333            start.get_or_insert(chunk_start);
334        }
335        start.ok_or_else(|| {
336            MsError::InvalidInput("time average produced no visibility buffers".into())
337        })
338    }
339
340    fn legacy(mut self, rows: Vec<Row>, label: &str) -> MsResult<RecordValue> {
341        let mut result = RecordValue::default();
342        if let Column::RowVector(column) = self.column {
343            let moments = scalar::reduce_vector(self.ms, self.selection, &rows, column)?;
344            for (index, moments) in moments.into_iter().enumerate() {
345                result.push(RecordField::new(
346                    format!("{label}_{index}"),
347                    Value::Record(legacy_record(moments)?),
348                ));
349            }
350        } else {
351            let mut moments = Moments::new(true);
352            self.reduce(&rows, &mut moments)?;
353            result.push(RecordField::new(
354                label,
355                Value::Record(legacy_record(moments)?),
356            ));
357        }
358        Ok(result)
359    }
360}

362fn cached_sample_axes<'a>(
363    grid: &'a mut BTreeMap<i32, SampleAxes>,
364    ms: &MeasurementSet,
365    selection: Option<&ResolvedMsSelection>,
366    id: i32,
367) -> MsResult<&'a SampleAxes> {
368    use std::collections::btree_map::Entry;
369    Ok(match grid.entry(id) {
370        Entry::Occupied(entry) => entry.into_mut(),
371        Entry::Vacant(entry) => entry.insert(SampleAxes::for_description(ms, selection, id)?),
372    })
373}
