13#[derive(Default)]
14struct StatisticsText(String);

16impl<'a, 'py> FromPyObject<'a, 'py> for StatisticsText {
17    type Error = PyErr;
18
19    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
20        selection_text(&value, "statistics").map(Self)

24impl StatisticsText {
25    fn selector(self) -> Option<String> {
26        (!self.0.is_empty()).then_some(self.0)
27    }
28}

169impl PyMsTool {
170    fn reduce_statistics(
171        &self,
172        py: Python<'_>,
173        options: ToolStatisticsOptions,
174    ) -> PyResult<Py<PyAny>> {
175        let result = self.attachment_info(py, move |attachment| {
176            let Some(attachment) = attachment else {
177                return Ok(CasaValue::empty_record());
178            };
179            let link = attachment.working_metadata_link();
180            let snapshot = link.snapshot(None).map_err(runtime_error)?;
181            let model_input = casa_calibration::models::visibility_model_input();
182            let record =
183                tool_statistics(&snapshot.ms, &options, &model_input).map_err(runtime_error)?;
184            link.verify_directory().map_err(runtime_error)?;
185            Ok(visibility_record(record))
186        })?;
187        result.into_python(py)
188    }
