2//! CASA's lattice geometry checks for Boolean masks on public image reads.
3
4use casa_coordinates::CoordinateSystem;
5
6use crate::image_expr::ImageExprValue;
7use crate::image_expr_lel::{can_broadcast_to, mask_coordinate_axes};
8
9use super::{ExprNode, ExprParseError, ExprValueConvert, ImageResolver, Parser};
10
11#[derive(Clone, Copy)]
12pub(super) enum Operator {
13    Infix,
14    Logical,
15    Function,
16}
17
18impl<T: ImageExprValue + PartialOrd + ExprValueConvert> ExprNode<'_, T> {
19    fn lattice_geometry(&self) -> Option<(&[usize], Option<&CoordinateSystem>)> {
20        let (shape, coordinates) = match self {
21            Self::Numeric(image, _) => (image.shape(), image.expression_coordinates()),
22            Self::Complex(image, _) => (image.shape(), image.expression_coordinates()),
23            Self::Mask(mask) => (mask.shape(), mask.expression_coordinates()),
24            Self::Scalar(_) | Self::IntArray(_) => return None,
25        };
26        (!shape.is_empty()).then_some((shape, coordinates))
27    }
28}
29
30impl<'a, T, R> Parser<'a, '_, T, R>
31where
32    T: ImageExprValue + PartialOrd + ExprValueConvert,
33    R: ImageResolver<'a, T>,
34{
35    pub(super) fn validate_lattice_operands(
36        &self,
37        lhs: &ExprNode<'a, T>,
38        rhs: &ExprNode<'a, T>,
39        position: usize,
40        operator: Operator,
41    ) -> Result<(), ExprParseError> {
42        if !self.check_lattice_geometry {
43            return Ok(());
44        }
45        let (Some((left, left_coordinates)), Some((right, right_coordinates))) =
46            (lhs.lattice_geometry(), rhs.lattice_geometry())
47        else {
48            return Ok(());
49        };
50        let rejected = |message: &str| ExprParseError {
51            message: if matches!(operator, Operator::Logical) {
52                "LatticeExprNode::resultDataType - Bool and numeric operands cannot mixed".into()
53            } else {
54                message.into()
55            },
56            position,
57        };
58        if matches!(operator, Operator::Function) && left != right {
59            return Err(rejected("LELAttribute: shapes of operands mismatch"));
60        }
61        let right_is_smaller = can_broadcast_to(right, left);
62        let left_is_smaller = can_broadcast_to(left, right);
63        if !left_is_smaller && !right_is_smaller {
64            return Err(rejected("LatticeExprNode - shapes of operands mismatch"));
65        }
66        let (small, large) = if right_is_smaller {
67            (right, left)
68        } else {
69            (left, right)
70        };
71        if let (Some(left), Some(right)) = (left_coordinates, right_coordinates) {
72            let left = mask_coordinate_axes(left);
73            let right = mask_coordinate_axes(right);
74            if !(left.starts_with(&right) || right.starts_with(&left)) {
75                return Err(rejected(
76                    "LatticeExprNode - coordinates of operands mismatch",
77                ));
78            }
79        }
80        if small != large && small.iter().all(|&length| length == 1) {
81            return Err(rejected("ExtendSpecifier - no axes remaining"));
82        }
83        Ok(())
84    }
85}
