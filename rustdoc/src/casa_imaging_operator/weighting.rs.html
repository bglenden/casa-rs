12/// Which CASA cell arithmetic a density grid uses.
13#[derive(Clone, Copy, Debug, PartialEq, Eq)]
14pub enum DensityCellRule {
15    /// `VisImagingWeight`: single-precision coordinates truncated toward
16    /// zero, `v` not mirrored.
17    Standard,
18    /// `BriggsCubeWeightor`: rounded coordinates with `v` mirrored, built in
19    /// double and looked up in single precision.
20    Cube,
21}
22
23/// A sample's density-cell coordinates in wavelengths, rounded the way CASA
24/// forms them before it picks a density cell (`VisImagingWeight::weightUniform`
25/// and its accumulation loop, `BriggsCubeWeightor::getWeightUniform`):
26/// `Float f = ν/c`, then `Float u = uvw_u·f` and `Float v = uvw_v·f`. At
27/// large image sizes one single-precision ulp is a visible fraction of a
28/// cell, so the rounding order decides which cell a boundary sample uses.
29/// Gridding and the uv taper keep the double coordinates of the
30/// [`Placement`].
31#[derive(Clone, Copy, Debug, PartialEq)]
32pub struct DensityUv {
33    /// u in wavelengths.
34    pub u: f32,
35    /// v in wavelengths.
36    pub v: f32,
37}
38
39impl DensityUv {
40    /// CASA's single-precision coordinates of baseline `uvw_m` at
41    /// `frequency_hz`.
42    #[must_use]
43    pub fn casa(uvw_m: [f64; 3], frequency_hz: f64) -> Self {
44        let wavelengths_per_m = f64::from((frequency_hz / SPEED_OF_LIGHT_M_PER_S) as f32);
45        Self {
46            u: (uvw_m[0] * wavelengths_per_m) as f32,
47            v: (uvw_m[1] * wavelengths_per_m) as f32,
48        }
49    }
50}
51
52/// CASA `C::c`.
53pub(crate) const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

55/// Shape of a density grid: image-sized cells per plane.
56#[derive(Clone, Copy, Debug, PartialEq)]
57pub struct DensityGridShape {
58    /// Cells along u (the image width).
59    pub width: usize,
60    /// Cells along v (the image height).
61    pub height: usize,
62    /// Density planes: 1 for a global generation; for a per-channel
63    /// generation the output channel count plus `padding` on each side.
64    pub planes: usize,
65    /// Per-channel generation: density planes on each side of the output
66    /// axis (CASA `BriggsCubeWeightor::estimateSwingChanPad`), so native
67    /// channels just beyond the image still find their own plane. 0 for a
68    /// global generation.
69    pub padding: u32,
70    /// Signed image increments `[Δx, Δy]` in radians.
71    pub increment_rad: [f64; 2],
72    /// Cell arithmetic.
73    pub rule: DensityCellRule,
74}
75
76impl DensityGridShape {
77    /// Cells over every plane.
78    #[must_use]
79    pub const fn cells(&self) -> usize {
80        self.width * self.height * self.planes
81    }
82
83    /// Density plane of a placement: plane 0 for a global generation.
84    fn plane_of(&self, placement_plane: u32) -> Option<usize> {
85        if self.planes == 1 {
86            Some(0)
87        } else {
88            let plane = placement_plane as usize;
89            (plane < self.planes).then_some(plane)
90        }
91    }
92
93    /// CASA's integer cell origin `[nx/2, ny/2]` (`uorigin_p`, `vorigin_p`;
94    /// `GridFT`'s `offset_p` likewise): integer division, so an odd size
95    /// puts the origin on the lower of the two middle cells.
96    fn origin(&self) -> [usize; 2] {
97        [self.width / 2, self.height / 2]
98    }
99
100    /// CASA's `Float` cell scales `uscale_p = nx·Δx`, `vscale_p = ny·Δy`,
101    /// each rounded once from the double product
102    /// (`VisImagingWeight::VisImagingWeight`, `BriggsCubeWeightor::init`).
103    fn scale(&self) -> [f32; 2] {
104        [
105            (self.width as f64 * self.increment_rad[0]) as f32,
106            (self.height as f64 * self.increment_rad[1]) as f32,
107        ]
108    }
109
110    /// The cell a density placement adds to. Standard placements carry
111    /// [`DensityUv`] coordinates; cube placements carry the double
112    /// coordinates CASA grids its weight density with (`GridFT::put`'s
113    /// `locuvw`: `nint(scale·u + offset + 1)` in Fortran indexing).
114    fn build_cell(&self, plane: usize, u: f64, v: f64) -> Option<usize> {
115        match self.rule {
116            DensityCellRule::Standard => self.standard_cell(plane, u as f32, v as f32),
117            DensityCellRule::Cube => {
118                let [origin_x, origin_y] = self.origin().map(|origin| origin as f64);
119                let x = (u * self.width as f64 * self.increment_rad[0] + origin_x + 1.0).round()
120                    as isize
121                    - 1;
122                let y = (-v * self.height as f64 * self.increment_rad[1] + origin_y + 1.0).round()
123                    as isize
124                    - 1;
125                self.cell_index(plane, x, y)
126            }
127        }
128    }
129
130    /// The cell a lookup reads: `BriggsCubeWeightor::getWeightUniform`
131    /// rounds `uscale·u + uorigin` in `Float`, with `v` mirrored.
132    fn lookup_cell(&self, plane: usize, uv: DensityUv) -> Option<usize> {
133        match self.rule {
134            DensityCellRule::Standard => self.standard_cell(plane, uv.u, uv.v),
135            DensityCellRule::Cube => {
136                let [scale_x, scale_y] = self.scale();
137                let [origin_x, origin_y] = self.origin().map(|origin| origin as f32);
138                let x = (scale_x * uv.u + origin_x).round();
139                let y = (scale_y * -uv.v + origin_y).round();
140                self.cell_index(plane, x as isize, y as isize)
141            }
142        }
143    }
144
145    /// `VisImagingWeight`: `Int(uscale·u + uorigin)` in `Float`, truncated
146    /// toward zero.
147    fn standard_cell(&self, plane: usize, u: f32, v: f32) -> Option<usize> {
148        let [scale_x, scale_y] = self.scale();
149        let [origin_x, origin_y] = self.origin().map(|origin| origin as f32);
150        let x = (scale_x * u + origin_x) as isize;
151        let y = (scale_y * v + origin_y) as isize;
152        self.cell_index(plane, x, y)
153    }
154
155    /// CASA accepts `0 < cell < extent` only.
156    fn cell_index(&self, plane: usize, x: isize, y: isize) -> Option<usize> {
157        if x <= 0 || y <= 0 || x >= self.width as isize || y >= self.height as isize {
158            return None;
159        }
160        Some((plane * self.height + y as usize) * self.width + x as usize)
161    }
162}

164/// Gridded weight density per plane, the input to uniform and Briggs
165/// weights.
166#[derive(Clone, Debug, PartialEq)]
167pub struct DensityGrid {
168    shape: DensityGridShape,
169    cells: Vec<f64>,
170    sum_weights: Vec<f64>,
171}
172
173impl DensityGrid {
174    /// An empty grid of `shape`, ready to [`DensityGrid::accumulate`] blocks.
175    #[must_use]
176    pub fn new(shape: DensityGridShape) -> Self {
177        Self {
178            cells: vec![0.0; shape.cells()],
179            sum_weights: vec![0.0; shape.planes],
180            shape,
181        }
182    }
183
184    /// Add one block of density samples.
185    ///
186    /// Blocks carry one polarization whose weight is CASA's unpolarized input
187    /// weight; each sample adds its weight at its cell and at the conjugate
188    /// cell, each when that cell lies inside the grid (CASA tests the two
189    /// independently), and once to the plane's weight sum when its own cell
190    /// does.
191    pub fn accumulate(&mut self, block: &SampleBlock<'_>) {
192        assert_eq!(block.npol, 1, "density blocks carry one unpolarized weight");
193        for (placement, weight) in block.placements.iter().zip(block.weights) {
194            if *weight <= 0.0 {
195                continue;
196            }
197            let Some(plane) = self.shape.plane_of(placement.plane) else {
198                continue;
199            };
200            let weight = f64::from(*weight);
201            let primary = self.shape.build_cell(plane, placement.u, placement.v);
202            let conjugate = self.shape.build_cell(plane, -placement.u, -placement.v);
203            if let Some(cell) = primary {
204                self.cells[cell] += weight;
205                self.sum_weights[plane] += weight;
206            }
207            if let Some(cell) = conjugate {
208                self.cells[cell] += weight;
209            }
210        }
211    }
212
213    /// The shape.
214    #[must_use]
215    pub const fn shape(&self) -> &DensityGridShape {
216        &self.shape
217    }
218
219    /// Every cell, plane-major then `[y][x]`.
220    #[must_use]
221    pub fn cells(&self) -> &[f64] {
222        &self.cells
223    }
224
225    /// Cells of one plane.
226    #[must_use]
227    pub fn plane(&self, plane: usize) -> &[f64] {
228        let cells = self.shape.width * self.shape.height;
229        &self.cells[plane * cells..(plane + 1) * cells]
230    }
231
232    /// Input weight summed once per sample, per plane.
233    #[must_use]
234    pub fn sum_weights(&self) -> &[f64] {
235        &self.sum_weights
236    }
237
238    /// Density at the lookup cell of `uv` on `plane`; `None` outside the
239    /// grid.
240    #[must_use]
241    pub fn lookup(&self, plane: usize, uv: DensityUv) -> Option<f64> {
242        self.shape
243            .lookup_cell(plane, uv)
244            .map(|cell| self.cells[cell])
245    }
246
247    /// Briggs factors `f² = (5·10^{−robust})² / (Σd² / Σd)` per plane, with
248    /// `Σd` the density sum (standard) or twice the input weight sum (cube).
249    #[must_use]
250    pub fn robust_factors(&self, robust: f64) -> RobustFactors {
251        let cells = self.shape.width * self.shape.height;
252        let f2 = (0..self.shape.planes)
253            .map(|plane| {
254                let density = &self.cells[plane * cells..(plane + 1) * cells];
255                let square_sum = density.iter().map(|value| value * value).sum::<f64>();
256                let sum = match self.shape.rule {
257                    DensityCellRule::Standard => density.iter().sum::<f64>(),
258                    DensityCellRule::Cube => 2.0 * self.sum_weights[plane],
259                };
260                if sum > 0.0 && square_sum > 0.0 {
261                    (5.0 * 10_f64.powf(-robust)).powi(2) / (square_sum / sum)
262                } else {
263                    0.0
264                }
265            })
266            .collect();
267        RobustFactors { robust, f2 }
268    }
269}

271/// Briggs `f²` per density plane for one robustness.
272#[derive(Clone, Debug, PartialEq)]
273pub struct RobustFactors {

275    f2: Vec<f64>,
276}

278impl RobustFactors {
279    /// The robustness parameter.
280    #[must_use]
281    pub const fn robust(&self) -> f64 {
282        self.robust
283    }
284

286    #[must_use]

288        self.f2[plane]

292/// The density grid of every block from `source` (see
293/// [`DensityGrid::accumulate`]).
294pub fn build_density_grid<'a>(
295    source: impl Iterator<Item = SampleBlock<'a>>,
296    shape: DensityGridShape,
297) -> DensityGrid {
298    let mut grid = DensityGrid::new(shape);
299    for block in source {
300        grid.accumulate(&block);
301    }
302    grid
303}

305/// CASA `briggsbwtaper` (CAS-13021): the fractional bandwidth
306/// `2(ν_last − ν_first)/(ν_last + ν_first)` of the image's spectral axis,
307/// which turns each sample's uv distance in cells into a divisor of the
308/// Briggs density term so every cube channel is weighted as if its uv
309/// coverage had the continuum's radial smearing.
310#[derive(Clone, Copy, Debug, PartialEq)]
311pub struct BandwidthTaper {
312    fractional_bandwidth: f64,
313}
314
315impl BandwidthTaper {
316    /// From the first and last channel centres of the image's spectral axis
317    /// (`SynthesisImagerVi2::weight`); the two must be finite, positive and
318    /// distinct.
319    pub fn from_frequency_range(first_hz: f64, last_hz: f64) -> Result<Self, OperatorError> {
320        let low = first_hz.min(last_hz);
321        let high = first_hz.max(last_hz);
322        if !(low.is_finite() && high.is_finite()) || low <= 0.0 || high <= low {
323            return Err(OperatorError::Weighting {
324                reason: "the bandwidth taper needs two distinct positive frequencies",
325            });
326        }
327        Ok(Self {
328            fractional_bandwidth: 2.0 * (high - low) / (high + low),
329        })
330    }
331
332    /// `2(ν_last − ν_first)/(ν_last + ν_first)`.
333    #[must_use]
334    pub const fn fractional_bandwidth(self) -> f64 {
335        self.fractional_bandwidth
336    }
337
338    /// CASA `uvDistanceFactor`: `n + 0.5`, or `(4 − n)/(4 − 2n)` when that is
339    /// below 1.5, with `n` the fractional bandwidth times the sample's
340    /// distance from the uv origin in grid cells.
341    fn uv_distance_factor(self, shape: &DensityGridShape, u: f64, v: f64) -> f64 {
342        let u_cells = u * shape.width as f64 * shape.increment_rad[0];
343        let v_cells = v * shape.height as f64 * shape.increment_rad[1];
344        let cells = self.fractional_bandwidth * u_cells.hypot(v_cells);
345        let factor = cells + 0.5;
346        if factor < 1.5 {
347            (4.0 - cells) / (4.0 - 2.0 * cells)
348        } else {
349            factor
350        }
351    }
352}

354/// The imaging-weight rule of one run.
355#[derive(Clone, Debug, PartialEq)]
356pub enum WeightingGeneration {
357    /// Input weight, optionally tapered.
358    Natural {
359        /// Gaussian uv taper.
360        taper: Option<Taper>,
361    },
362    /// Uniform (`robust` absent), Briggs or Briggs bandwidth-taper density
363    /// weighting.
364    Density {
365        /// Gridded weight density.
366        grid: DensityGrid,
367        /// Briggs factors; `None` for uniform weighting.
368        robust: Option<RobustFactors>,
369        /// CASA `briggsbwtaper`; needs `robust`.
370        bandwidth: Option<BandwidthTaper>,
371        /// Gaussian uv taper.
372        taper: Option<Taper>,
373    },
374}
375
376impl WeightingGeneration {
377    /// Uniform, Briggs or Briggs bandwidth-taper weighting over `grid`.
378    pub fn density(
379        grid: DensityGrid,
380        robust: Option<f64>,
381        bandwidth: Option<BandwidthTaper>,
382        taper: Option<Taper>,
383    ) -> Result<Self, OperatorError> {
384        let robust = match robust {
385            Some(robust) if !robust.is_finite() => {
386                return Err(OperatorError::Weighting {
387                    reason: "robustness is not finite",
388                });
389            }
390            Some(robust) => Some(grid.robust_factors(robust)),
391            None => None,
392        };
393        if bandwidth.is_some() && robust.is_none() {
394            return Err(OperatorError::Weighting {
395                reason: "the bandwidth taper is a Briggs mode and needs a robustness",
396            });
397        }
398        Ok(Self::Density {
399            grid,
400            robust,
401            bandwidth,
402            taper,
403        })
404    }
405
406    /// The padding of a per-channel density generation (CASA
407    /// `BriggsCubeWeightor`), whose output samples take the nearest native
408    /// channel's weight; `None` when native weights are interpolated
409    /// linearly (`VisImagingWeight`).
410    pub(crate) fn cube_padding(&self) -> Option<u32> {
411        match self {
412            Self::Density { grid, .. } if grid.shape.rule == DensityCellRule::Cube => {
413                Some(grid.shape.padding)
414            }
415            Self::Natural { .. } | Self::Density { .. } => None,
416        }
417    }
418
419    /// Imaging weight of one placement from its unpolarized input weight;
420    /// `density_uv` picks the density cell on the density plane
421    /// `placement.plane` (ignored by a global generation). Pure; CASA's
422    /// cell rules live in [`DensityGrid::lookup`].
423    #[must_use]
424    pub fn imaging_weight(
425        &self,
426        placement: &Placement,
427        density_uv: DensityUv,
428        input_weight: f32,
429    ) -> f32 {
430        if input_weight <= 0.0 {
431            return 0.0;
432        }
433        let input = f64::from(input_weight);
434        let (weighted, taper) = match self {
435            Self::Natural { taper } => (input, *taper),
436            Self::Density {
437                grid,
438                robust,
439                bandwidth,
440                taper,
441            } => {
442                let Some(plane) = grid.shape.plane_of(placement.plane) else {
443                    return 0.0;
444                };
445                // CASA weighs a sample whose density cell is empty or off
446                // the grid at zero, under uniform and Briggs alike
447                // (`VisImagingWeight::weightUniform` requires `a_gwt > 0`;
448                // `BriggsCubeWeightor::getWeightUniform` likewise).
449                let Some(density) = grid
450                    .lookup(plane, density_uv)
451                    .filter(|density| *density > 0.0)
452                else {
453                    return 0.0;
454                };
455                let weighted = match robust {
456                    None => input / density,
457                    Some(factors) => {
458                        let distance = bandwidth.map_or(1.0, |bandwidth| {
459                            bandwidth.uv_distance_factor(&grid.shape, placement.u, placement.v)
460                        });
461                        input / (density * factors.factor(plane) / distance + 1.0)
462                    }
463                };
464                (weighted, *taper)
465            }
466        };
467        (weighted * gaussian_taper(taper, placement.u, placement.v)) as f32
468    }
469}

471/// CASA `VisImagingWeight::filter`: an elliptical Gaussian in the uv plane.
472fn gaussian_taper(taper: Option<Taper>, u: f64, v: f64) -> f64 {
473    let Some(taper) = taper else {
474        return 1.0;
475    };
476    let sine = taper.position_angle_rad().sin();
477    let cosine = taper.position_angle_rad().cos();
478    let rotated_u = sine * u + cosine * v;
479    let rotated_v = -cosine * u + sine * v;
480    let major = std::f64::consts::LN_2 / taper.major_lambda().powi(2);
481    let minor = std::f64::consts::LN_2 / taper.minor_lambda().powi(2);
482    (-major * rotated_u.powi(2) - minor * rotated_v.powi(2)).exp()
483}
