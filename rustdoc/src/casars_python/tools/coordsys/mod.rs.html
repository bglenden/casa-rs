51pub(crate) fn default_image_obs_info() -> ObsInfo {
52    // ITRF position of ALMA from the casacore Observatories table.
53    let (sin_lon, cos_lon) = (-1.182_546_595_504_989_2_f64).sin_cos();
54    let (sin_lat, cos_lat) = (-0.399_414_986_926_273_8_f64).sin_cos();
55    let radius = 6_379_946.013_264_43_f64;
56    defaults::obs_info().with_telescope_position(MPosition::new_itrf(
57        cos_lon * cos_lat * radius,
58        sin_lon * cos_lat * radius,
59        sin_lat * radius,
60    ))
61}

63pub(crate) fn default_spectral_conversion() -> RecordValue {
64    // LAST is casacore's placeholder for an unset conversion frame.
65    let epoch = epoch_to_record(&MEpoch::from_mjd(0.0, EpochRef::LAST));
66    RecordValue::new(vec![
67        RecordField::new("epoch", Value::Record(epoch)),
68        RecordField::new(
69            "direction",
70            Value::Record(direction_to_record(&MDirection::from_angles(
71                0.0,
72                std::f64::consts::FRAC_PI_2,
73                DirectionRef::J2000,
74            ))),
75        ),
76        RecordField::new(
77            "position",
78            Value::Record(position_to_record(&MPosition::new_itrf(0.0, 0.0, 0.0))),
79        ),
80        RecordField::new("system", Value::Scalar(ScalarValue::String("LSRK".into()))),
81    ])
82}

84fn current_epoch(mut info: ObsInfo) -> ObsInfo {
85    if let Ok(now) = std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH) {
86        info.date = Some(MEpoch::from_mjd(
87            40_587.0 + now.as_secs_f64() / 86_400.0,
88            EpochRef::UTC,
89        ));
90    }
91    info
92}

94impl CoordinateState {
95    fn from_record(record: RecordValue) -> PyResult<Self> {
96        let coordinates = CoordinateSystem::from_record(&record)
97            .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
98        Ok(Self {
99            coordinates,
100            record,
101            parent_name: "unknown".into(),
102        })
103    }
104
105    fn empty() -> Self {
106        Self::new(CoordinateSystem::new().with_obs_info(current_epoch(default_image_obs_info())))
107    }
108
109    fn new(coordinates: CoordinateSystem) -> Self {
110        let record = coordinates.to_record();
111        Self {
112            coordinates,
113            record,
114            parent_name: "unknown".into(),
115        }
116    }
117}
118
119impl PyCoordsys {
120    pub(crate) fn empty() -> Self {
121        Self {
122            state: Mutex::new(CoordinateState::empty()),
123        }
124    }
125
126    pub(crate) fn from_record(record: RecordValue) -> PyResult<Self> {
127        Ok(Self {
128            state: Mutex::new(CoordinateState::from_record(record)?),
129        })
130    }
131
132    fn state(&self) -> MutexGuard<'_, CoordinateState> {
133        let mut state = self.lock_state();
134        if state.parent_name.is_empty() {
135            state.parent_name = "unknown".into();
136        }
137        state
138    }
139
140    fn lock_state(&self) -> MutexGuard<'_, CoordinateState> {
141        self.state.lock().unwrap_or_else(PoisonError::into_inner)
142    }
143}
