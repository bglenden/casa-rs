2//! Fixed-width NOSTAR observation and scan record decoding.
3
4use crate::{MsError, MsResult};
5
6pub(super) const OBSERVATION_HEADER_BYTES: usize = 15_136;
7pub(super) const SCAN_HEADER_BYTES: usize = 424;
8pub(super) const ARRAYS: usize = 35;
9
10#[derive(Clone, Copy)]
11pub(super) enum ByteOrder {
12    Big,
13    Little,
14}
15
16#[derive(Clone, Copy)]
17pub(super) struct Record<'a> {
18    bytes: &'a [u8],
19    order: ByteOrder,
20}
21
22impl<'a> Record<'a> {
23    pub(super) fn observation(bytes: &'a [u8; OBSERVATION_HEADER_BYTES]) -> MsResult<Self> {
24        let mut record = Self::new(bytes, ByteOrder::Big);
25        if !(1..=ARRAYS as i32).contains(&record.i32(144)) {
26            record.order = ByteOrder::Little;
27            if !(1..=ARRAYS as i32).contains(&record.i32(144)) {
28                return Err(invalid("array count or byte order"));
29            }
30        }
31        Ok(record)
32    }
33
34    pub(super) fn new(bytes: &'a [u8], order: ByteOrder) -> Self {
35        Self { bytes, order }
36    }
37
38    pub(super) fn order(self) -> ByteOrder {
39        self.order
40    }
41
42    pub(super) fn i16(self, offset: usize) -> i16 {
43        let bytes = self.bytes[offset..offset + 2].try_into().unwrap();
44        match self.order {
45            ByteOrder::Big => i16::from_be_bytes(bytes),
46            ByteOrder::Little => i16::from_le_bytes(bytes),
47        }
48    }
49
50    pub(super) fn i32(self, offset: usize) -> i32 {
51        let bytes = self.bytes[offset..offset + 4].try_into().unwrap();
52        match self.order {
53            ByteOrder::Big => i32::from_be_bytes(bytes),
54            ByteOrder::Little => i32::from_le_bytes(bytes),
55        }
56    }
57
58    pub(super) fn f32(self, offset: usize) -> f32 {
59        f32::from_bits(self.i32(offset) as u32)
60    }
61
62    pub(super) fn f64(self, offset: usize) -> f64 {
63        let bytes = self.bytes[offset..offset + 8].try_into().unwrap();
64        match self.order {
65            ByteOrder::Big => f64::from_be_bytes(bytes),
66            ByteOrder::Little => f64::from_le_bytes(bytes),
67        }
68    }
69
70    pub(super) fn text(self, offset: usize, len: usize) -> MsResult<&'a str> {
71        let bytes = &self.bytes[offset..offset + len];
72        let end = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
73        std::str::from_utf8(&bytes[..end])
74            .map(str::trim_end)
75            .map_err(|_| invalid("text field"))
76    }
77
78    pub(super) fn spectrum(self, count: usize, array_scale: f64) -> Vec<f32> {
79        let scale = self.f64(408);
80        let offset = self.f64(416);
81        if scale == 0. && offset == 0. {
82            return vec![0.; count];
83        }
84        let values = &self.bytes[SCAN_HEADER_BYTES..];
85        let scaled = |code: u16| ((f64::from(code) * scale + offset) * array_scale) as f32;
86        let mut samples = Vec::with_capacity(count);
87        for bytes in values.as_chunks::<3>().0.iter().take(count / 2) {
88            samples.push(scaled(u16::from_be_bytes([bytes[0], bytes[1]]) >> 4));
89            samples.push(scaled(u16::from_be_bytes([bytes[1], bytes[2]]) & 0xfff));
90        }
91        if !count.is_multiple_of(2) {
92            let last = &values[(count / 2) * 3..];
93            samples.push(scaled(u16::from_be_bytes([last[0], last[1]]) >> 4));
94        }
95        samples
96    }
97}
98
99pub(super) fn invalid(field: &str) -> MsError {
100    MsError::InvalidInput(format!("Invalid NOSTAR {field}"))
101}
