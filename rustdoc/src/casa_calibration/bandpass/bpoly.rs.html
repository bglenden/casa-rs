37pub(super) fn solve_polynomial(
38    ms: &MeasurementSet,
39    request: &BandpassSolveRequest,
40    context: CalibrationSolveContext,
41    groups: impl Iterator<Item = SolveAccumulator>,
42    bin_widths: &BTreeMap<i32, usize>,
43    cuts: BandpassSolveCuts,
44    mode: crate::CalibrationOutputMode,
45) -> Result<BandpassSolveReport, BandpassSolveError> {
46    let window = ms
47        .spectral_window()
48        .map_err(|source| BandpassSolveError::OpenMeasurementSet {
49            path: ms
50                .path()
51                .map(|path| path.display().to_string())
52                .unwrap_or_default(),
53            source,
54        })?;
55    let descriptions = context
56        .rows
57        .iter()
58        .map(|row| (row.data_desc_id, row))
59        .collect::<BTreeMap<_, _>>();
60    let mut frequencies_by_description = BTreeMap::new();
61    for (id, row) in descriptions {
62        let spw = row.data_spw_id;
63        let frequencies = window.chan_freq(spw as usize).map_err(|source| {
64            BandpassSolveError::OpenMeasurementSet {
65                path: ms
66                    .path()
67                    .map(|path| path.display().to_string())
68                    .unwrap_or_default(),
69                source,
70            }
71        })?;
72        let (channels, _) = crate::solve::grouping::polynomial::channel_selection(
73            row.channel_selection.as_ref(),
74            frequencies.len(),
75        )?;
76        let spacing = frequencies
77            .windows(2)
78            .next()
79            .map_or(1.0, |pair| (pair[1] - pair[0]).abs());
80        frequencies_by_description.insert(
81            id,
82            (
83                super::metadata::average_frequencies(
84                    &frequencies,
85                    &channels,
86                    bin_widths.get(&spw).copied().unwrap_or(1),
87                ),
88                spacing,
89            ),
90        );
91    }
92    let combined_window = request.combine.spectral_windows.then(|| {
93        context
94            .rows
95            .iter()
96            .map(|row| row.data_spw_id)
97            .min()
98            .expect("selected windows")
99    });
100    if combined_window.is_some()
101        && frequencies_by_description
102            .values()
103            .map(|(channels, _)| channels.len())
104            .min()
105            != frequencies_by_description
106                .values()
107                .map(|(channels, _)| channels.len())
108                .max()
109    {
110        tracing::warn!(
111            "Spectral windows with different selected channel counts cannot be combined"
112        );
113        return Ok(empty_report(
114            request,
115            context.refant_id,
116            context.selected_data,
117            SolveStats::default(),
118        ));
119    }
120    let mut channel_count = 0;
121    let mut fitted = Vec::new();
122    for group in groups {
123        let Some(polynomial) = &group.polynomial else {
124            continue;
125        };
126        let frequencies = polynomial
127            .descriptions()
128            .iter()
129            .map(|id| {
130                let (frequencies, spacing) = &frequencies_by_description[id];
131                (*id, (frequencies.as_slice(), *spacing))
132            })
133            .collect::<BTreeMap<_, _>>();
134        if frequencies.is_empty() {
135            continue;
136        }
137        let mut rows = match fit::fit_interval(
138            &group,
139            &frequencies,
140            &context.available_antennas,
141            context.refant_id,
142            request,
143            cuts,
144        ) {
145            Ok(rows) => rows,
146            Err(error @ BandpassSolveError::FitPolynomial { .. }) => {
147                tracing::warn!("{error}");
148                break;
149            }
150            Err(error) => return Err(error),
151        };
152        if !rows.is_empty() {
153            if let Some(spw) = combined_window {
154                for row in &mut rows {
155                    row.spw_id = spw;
156                }
157            }
158            channel_count = frequencies
159                .values()
160                .map(|(channels, _)| channels.len())
161                .max()
162                .unwrap_or(0);
163            fitted.extend(rows);
164        }
165    }
166    if fitted.is_empty() {
167        return Ok(empty_report(
168            request,
169            context.refant_id,
170            context.selected_data,
171            SolveStats::default(),
172        ));
173    }
174    let report =
175        write_polynomial_rows(ms, request, context.refant_id, &fitted, channel_count, mode)?;
176    Ok(BandpassSolveReport {
177        selected_data: context.selected_data,
178        ..report
179    })
180}

182pub(super) fn write_bpoly_caltable(
183    ms: &MeasurementSet,
184    request: &BandpassSolveRequest,
185    refant_id: i32,
186    rows: &[BandpassSolutionRow],
187    mode: crate::CalibrationOutputMode,
188) -> Result<BandpassSolveReport, BandpassSolveError> {
189    let fitted_rows = fit_bpoly_rows(
190        ms,
191        rows,
192        request.amplitude_degree,
193        request.phase_degree,
194        request.cuts,
195        &request.selection,
196    )?;
197    let channel_count = rows.first().map(|row| row.channel_count).unwrap_or(0);
198    write_polynomial_rows(ms, request, refant_id, &fitted_rows, channel_count, mode)
199}

205fn write_polynomial_rows(
206    ms: &MeasurementSet,
207    request: &BandpassSolveRequest,
208    refant_id: i32,
209    fitted_rows: &[BPolySolutionRow],
210    channel_count: usize,
211    mode: crate::CalibrationOutputMode,
212) -> Result<BandpassSolveReport, BandpassSolveError> {
213    // Existing casa-rs BPOLY tables used rank-one placeholder arrays and
214    // two extra diagnostic columns. Keep that layout when appending to one.
215    let casa_layout = mode != crate::CalibrationOutputMode::Append
216        || !request.output_table.exists()
217        || !Table::open(TableOptions::new(&request.output_table))
218            .map_err(|source| BandpassSolveError::SaveCalibrationTable {
219                path: request.output_table.display().to_string(),
220                source: Box::new(source),
221            })?
222            .schema()
223            .is_some_and(|schema| schema.contains_column(COL_PARAMERR));
224    let gain_rank = Some(if casa_layout { 4 } else { 1 });
225    let reference_rank = Some(if casa_layout { 3 } else { 1 });
226    let direction_rank = Some(if casa_layout { 4 } else { 1 });
227    let mut columns = vec![
228        ColumnSchema::scalar(COL_TIME, casa_types::PrimitiveType::Float64),
229        ColumnSchema::scalar(COL_TIME_EXTRA_PREC, casa_types::PrimitiveType::Float64),
230        ColumnSchema::scalar(COL_FIELD_ID, casa_types::PrimitiveType::Int32),
231        ColumnSchema::scalar(COL_ANTENNA1, casa_types::PrimitiveType::Int32),
232        ColumnSchema::scalar("FEED1", casa_types::PrimitiveType::Int32),
233        ColumnSchema::scalar(COL_INTERVAL, casa_types::PrimitiveType::Float64),
234        ColumnSchema::scalar(COL_SCAN_NUMBER, casa_types::PrimitiveType::Int32),
235        ColumnSchema::scalar(COL_OBSERVATION_ID, casa_types::PrimitiveType::Int32),
236        ColumnSchema::scalar(COL_ARRAY_ID, casa_types::PrimitiveType::Int32),
237        ColumnSchema::scalar("PROCESSOR_ID", casa_types::PrimitiveType::Int32),
238        ColumnSchema::scalar("STATE_ID", casa_types::PrimitiveType::Int32),
239        ColumnSchema::scalar("PHASE_ID", casa_types::PrimitiveType::Int32),
240        ColumnSchema::scalar("PULSAR_BIN", casa_types::PrimitiveType::Int32),
241        ColumnSchema::scalar("PULSAR_GATE_ID", casa_types::PrimitiveType::Int32),
242        ColumnSchema::scalar("FREQ_GROUP", casa_types::PrimitiveType::Int32),
243        ColumnSchema::scalar("FREQ_GROUP_NAME", casa_types::PrimitiveType::String),
244        ColumnSchema::scalar("FIELD_NAME", casa_types::PrimitiveType::String),
245        ColumnSchema::scalar("FIELD_CODE", casa_types::PrimitiveType::String),
246        ColumnSchema::scalar("SOURCE_NAME", casa_types::PrimitiveType::String),
247        ColumnSchema::scalar("SOURCE_CODE", casa_types::PrimitiveType::String),
248        ColumnSchema::scalar("CALIBRATION_GROUP", casa_types::PrimitiveType::Int32),
249        ColumnSchema::scalar(COL_CAL_DESC_ID, casa_types::PrimitiveType::Int32),
250        ColumnSchema::scalar("CAL_HISTORY_ID", casa_types::PrimitiveType::Int32),
251        ColumnSchema::array_variable("GAIN", casa_types::PrimitiveType::Complex32, gain_rank),
252        ColumnSchema::scalar("SIDEBAND_REF", casa_types::PrimitiveType::Complex32),
253        ColumnSchema::array_variable("REF_ANT", casa_types::PrimitiveType::Int32, reference_rank),
254        ColumnSchema::array_variable("REF_FEED", casa_types::PrimitiveType::Int32, reference_rank),
255        ColumnSchema::array_variable(
256            "REF_RECEPTOR",
257            casa_types::PrimitiveType::Int32,
258            reference_rank,
259        ),
260        ColumnSchema::array_variable(
261            "REF_FREQUENCY",
262            casa_types::PrimitiveType::Float64,
263            reference_rank,
264        ),
265        ColumnSchema::scalar("MEAS_FREQ_REF", casa_types::PrimitiveType::Int32),
266        ColumnSchema::array_variable(
267            "REF_DIRECTION",
268            casa_types::PrimitiveType::Float64,
269            direction_rank,
270        ),
271        ColumnSchema::scalar("MEAS_DIR_REF", casa_types::PrimitiveType::Int32),
272        ColumnSchema::scalar("TOTAL_SOLUTION_OK", casa_types::PrimitiveType::Bool),
273        ColumnSchema::scalar("TOTAL_FIT", casa_types::PrimitiveType::Float32),
274        ColumnSchema::scalar("TOTAL_FIT_WEIGHT", casa_types::PrimitiveType::Float32),
275        ColumnSchema::array_variable("SOLUTION_OK", casa_types::PrimitiveType::Bool, gain_rank),
276        ColumnSchema::array_variable("FIT", casa_types::PrimitiveType::Float32, gain_rank),
277        ColumnSchema::array_variable("FIT_WEIGHT", casa_types::PrimitiveType::Float32, gain_rank),
278        ColumnSchema::array_variable(COL_FLAG, casa_types::PrimitiveType::Bool, gain_rank),
279        ColumnSchema::array_variable(COL_PARAMERR, casa_types::PrimitiveType::Float32, Some(1)),
280        ColumnSchema::array_variable(COL_SNR, casa_types::PrimitiveType::Float32, gain_rank),
281        ColumnSchema::array_variable(COL_WEIGHT, casa_types::PrimitiveType::Float32, Some(1)),
282        ColumnSchema::scalar("POLY_TYPE", casa_types::PrimitiveType::String),
283        ColumnSchema::scalar("POLY_MODE", casa_types::PrimitiveType::String),
284        ColumnSchema::scalar(COL_SCALE_FACTOR, casa_types::PrimitiveType::Complex32),
285        ColumnSchema::scalar(COL_N_POLY_AMP, casa_types::PrimitiveType::Int32),
286        ColumnSchema::scalar(COL_N_POLY_PHASE, casa_types::PrimitiveType::Int32),
287        ColumnSchema::scalar(COL_PHASE_UNITS, casa_types::PrimitiveType::String),
288        ColumnSchema::array_variable(
289            COL_VALID_DOMAIN,
290            casa_types::PrimitiveType::Float64,
291            Some(1),
292        ),
293        ColumnSchema::array_variable(
294            COL_POLY_COEFF_AMP,
295            casa_types::PrimitiveType::Float64,
296            Some(4),
297        ),
298        ColumnSchema::array_variable(
299            COL_POLY_COEFF_PHASE,
300            casa_types::PrimitiveType::Float64,
301            Some(4),
302        ),
303    ];
304    if casa_layout {
305        columns.retain(|column| !matches!(column.name(), COL_PARAMERR | COL_WEIGHT));
306    }
307    let schema = TableSchema::new(columns).expect("valid BPOLY schema");
308    let mut writer = CalibrationTableWriter::with_mode(
309        ms,
310        CalibrationTableDescriptor {
311            output: &request.output_table,
312            schema,
313            subtype: "BPOLY",
314            parameter_type: None,
315            measurement_set_name: ms
316                .path()
317                .map(|path| path.display().to_string())
318                .unwrap_or_else(|| "<in-memory>".to_string()),
319            include_polarization_basis: false,
320            time_extra_precision_column: Some(COL_TIME_EXTRA_PREC),
321        },
322        mode,
323    )?;
324    {
325        let table = writer.table_mut();
326        table
327            .keywords_mut()
328            .upsert(LEGACY_CAL_DESC_KEYWORD, Value::table_ref("././CAL_DESC"));
329        table.keywords_mut().upsert(
330            LEGACY_CAL_HISTORY_KEYWORD,
331            Value::table_ref("././CAL_HISTORY"),
332        );
333        set_fixed_unit_keyword(table, COL_VALID_DOMAIN, &["Hz"]);
334        set_fixed_unit_keyword(table, "REF_FREQUENCY", &["Hz"]);
335        set_measinfo_keyword(
336            table,
337            "REF_FREQUENCY",
338            "frequency",
339            Some(if casa_layout { "LSRK" } else { "TOPO" }),
340        );
341        set_fixed_unit_keyword(table, "REF_DIRECTION", &["rad", "rad"]);
342        set_measinfo_keyword(table, "REF_DIRECTION", "direction", Some("J2000"));
343    }
344
345    let cal_desc_ids = fitted_rows
346        .iter()
347        .map(|row| row.spw_id)
348        .collect::<BTreeSet<_>>()
349        .into_iter()
350        .enumerate()
351        .map(|(cal_desc_id, spw_id)| {
352            (
353                spw_id,
354                i32::try_from(cal_desc_id).expect("small BPOLY CAL_DESC id set"),
355            )
356        })
357        .collect::<BTreeMap<_, _>>();
358
359    let window_names = ms
360        .spectral_window()
361        .ok()
362        .map(|window| {
363            cal_desc_ids
364                .keys()
365                .map(|&id| (id, window.freq_group_name(id as usize).unwrap_or_default()))
366                .collect::<BTreeMap<_, _>>()
367        })
368        .unwrap_or_default();
369    let unit = Complex32::new(1.0, 0.0);
370    let (gain_shape, reference_shape, direction_shape, diagnostic_shape): (
371        &[usize],
372        &[usize],
373        &[usize],
374        &[usize],
375    ) = if casa_layout {
376        (&[1, 1, 1, 1], &[1, 1, 1], &[2, 1, 1, 1], &[1, 1, 1, 1])
377    } else {
378        (&[1], &[1], &[2], &[2])
379    };
380    for row in fitted_rows {
381        let from_casa_task = request.cuts.is_some();
382        let status_weight = if casa_layout { 0.0 } else { 1.0 };
383        let unset = if casa_layout { -1 } else { 0 };
384        let reference_frequency_hz = (row.valid_domain_hz[0] + row.valid_domain_hz[1]) / 2.0;
385        let flat_amp = row
386            .amp_coefficients
387            .iter()
388            .flat_map(|coefficients| coefficients.iter().copied())
389            .collect::<Vec<_>>();
390        let flat_phase = row
391            .phase_coefficients
392            .iter()
393            .flat_map(|coefficients| coefficients.iter().copied())
394            .collect::<Vec<_>>();
395        let amp_shape = IxDyn(&[1, 1, 1, flat_amp.len()]).f();
396        let phase_shape = IxDyn(&[1, 1, 1, flat_phase.len()]).f();
397        let mut fields = vec![
398            RecordField::new(
399                COL_TIME,
400                Value::Scalar(ScalarValue::Float64(row.time_seconds)),
401            ),
402            RecordField::new(
403                COL_TIME_EXTRA_PREC,
404                Value::Scalar(ScalarValue::Float64(0.0)),
405            ),
406            RecordField::new(
407                COL_FIELD_ID,
408                Value::Scalar(ScalarValue::Int32(row.field_id)),
409            ),
410            RecordField::new(
411                COL_ANTENNA1,
412                Value::Scalar(ScalarValue::Int32(row.antenna_id)),
413            ),
414            RecordField::new("FEED1", Value::Scalar(ScalarValue::Int32(unset))),
415            RecordField::new(
416                COL_INTERVAL,
417                Value::Scalar(ScalarValue::Float64(row.interval_seconds)),
418            ),
419            RecordField::new(
420                COL_SCAN_NUMBER,
421                Value::Scalar(ScalarValue::Int32(row.scan_number)),
422            ),
423            RecordField::new(
424                COL_OBSERVATION_ID,
425                Value::Scalar(ScalarValue::Int32(row.observation_id)),
426            ),
427            RecordField::new(COL_ARRAY_ID, Value::Scalar(ScalarValue::Int32(unset))),
428            RecordField::new("PROCESSOR_ID", Value::Scalar(ScalarValue::Int32(-1))),
429            RecordField::new("STATE_ID", Value::Scalar(ScalarValue::Int32(-1))),
430            RecordField::new("PHASE_ID", Value::Scalar(ScalarValue::Int32(-1))),
431            RecordField::new("PULSAR_BIN", Value::Scalar(ScalarValue::Int32(-1))),
432            RecordField::new("PULSAR_GATE_ID", Value::Scalar(ScalarValue::Int32(-1))),
433            RecordField::new("FREQ_GROUP", Value::Scalar(ScalarValue::Int32(unset))),
434            RecordField::new(
435                "FREQ_GROUP_NAME",
436                Value::Scalar(ScalarValue::String(
437                    window_names.get(&row.spw_id).cloned().unwrap_or_default(),
438                )),
439            ),
440            RecordField::new(
441                "FIELD_NAME",
442                Value::Scalar(ScalarValue::String(String::new())),
443            ),
444            RecordField::new(
445                "FIELD_CODE",
446                Value::Scalar(ScalarValue::String(String::new())),
447            ),
448            RecordField::new(
449                "SOURCE_NAME",
450                Value::Scalar(ScalarValue::String(String::new())),
451            ),
452            RecordField::new(
453                "SOURCE_CODE",
454                Value::Scalar(ScalarValue::String(String::new())),
455            ),
456            RecordField::new(
457                "CALIBRATION_GROUP",
458                Value::Scalar(ScalarValue::Int32(unset)),
459            ),
460            RecordField::new(
461                COL_CAL_DESC_ID,
462                Value::Scalar(ScalarValue::Int32(
463                    *cal_desc_ids
464                        .get(&row.spw_id)
465                        .expect("BPOLY CAL_DESC entry for solved SPW"),
466                )),
467            ),
468            RecordField::new("CAL_HISTORY_ID", Value::Scalar(ScalarValue::Int32(unset))),
469            RecordField::new(
470                "GAIN",
471                Value::Array(ArrayValue::Complex32(ArrayD::from_elem(
472                    IxDyn(gain_shape).f(),
473                    if casa_layout {

478                ))),
479            ),
480            RecordField::new(
481                "SIDEBAND_REF",
482                Value::Scalar(ScalarValue::Complex32(Complex32::new(1.0, 0.0))),
483            ),
484            RecordField::new(
485                "REF_ANT",
486                Value::Array(ArrayValue::Int32(ArrayD::from_elem(
487                    IxDyn(reference_shape).f(),
488                    if from_casa_task || casa_layout {
489                        refant_id
490                    } else {
491                        row.antenna_id
492                    },
493                ))),
494            ),
495            RecordField::new(
496                "REF_FEED",
497                Value::Array(ArrayValue::Int32(ArrayD::zeros(IxDyn(reference_shape).f()))),
498            ),
499            RecordField::new(
500                "REF_RECEPTOR",
501                Value::Array(ArrayValue::Int32(ArrayD::zeros(IxDyn(reference_shape).f()))),
502            ),
503            RecordField::new(
504                "REF_FREQUENCY",
505                Value::Array(ArrayValue::Float64(ArrayD::from_elem(

507                    reference_frequency_hz,

509            ),
510            RecordField::new(

512                Value::Scalar(ScalarValue::Int32(i32::from(casa_layout))),

515                "REF_DIRECTION",

517                    IxDyn(direction_shape).f(),
518                ))),
519            ),
520            RecordField::new("MEAS_DIR_REF", Value::Scalar(ScalarValue::Int32(0))),
521            RecordField::new(
522                "TOTAL_SOLUTION_OK",
523                Value::Scalar(ScalarValue::Bool(!casa_layout)),
524            ),
525            RecordField::new("TOTAL_FIT", Value::Scalar(ScalarValue::Float32(0.0))),
526            RecordField::new(

542            ),
543            RecordField::new(

609            RecordField::new(
610                COL_VALID_DOMAIN,
611                Value::Array(ArrayValue::Float64(
612                    ArrayD::from_shape_vec(
613                        IxDyn(&[2]).f(),
614                        vec![row.valid_domain_hz[0], row.valid_domain_hz[1]],
615                    )
616                    .expect("BPOLY valid domain should reshape"),
617                )),
618            ),
619            RecordField::new(
620                COL_POLY_COEFF_AMP,
621                Value::Array(ArrayValue::Float64(
622                    ArrayD::from_shape_vec(amp_shape, flat_amp)
623                        .expect("BPOLY amp coefficients should reshape"),
624                )),
625            ),
626            RecordField::new(
627                COL_POLY_COEFF_PHASE,
628                Value::Array(ArrayValue::Float64(
629                    ArrayD::from_shape_vec(phase_shape, flat_phase)
630                        .expect("BPOLY phase coefficients should reshape"),
631                )),
632            ),
633        ];
634        if casa_layout {
635            fields.retain(|field| !matches!(field.name.as_str(), COL_PARAMERR | COL_WEIGHT));
636        }
637        writer.append(RecordValue::new(fields))?;
638    }
639
640    writer.copy_subtables(&[
641        (SubtableId::Observation, "OBSERVATION"),
642        (SubtableId::Antenna, "ANTENNA"),
643        (SubtableId::Field, "FIELD"),
644        (SubtableId::History, "HISTORY"),
645        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
646    ])?;
647
648    write_bpoly_cal_desc_subtable(
649        writer.output_path().join("CAL_DESC"),
650        &cal_desc_ids,
651        fitted_rows,
652        casa_layout,
653        ms.path()
654            .and_then(|path| path.file_name())
655            .and_then(|name| name.to_str())
656            .unwrap_or_default(),
657    )?;
658    let history_columns = ["CAL_PARMS", "CAL_TABLES", "CAL_SELECT", "CAL_NOTES"]
659        .map(|column| ColumnSchema::scalar(column, casa_types::PrimitiveType::String));
660    Table::with_schema(
661        TableSchema::new(history_columns.to_vec()).expect("BPOLY CAL_HISTORY schema"),
662    )
663    .save(TableOptions::new(writer.output_path().join("CAL_HISTORY")))
664    .map_err(|source| BandpassSolveError::SaveCalibrationTable {
665        path: writer
666            .output_path()
667            .join("CAL_HISTORY")
668            .display()
669            .to_string(),
670        source: Box::new(source),
671    })?;
672    writer.save_main()?;
673
674    let field_ids = fitted_rows
675        .iter()
676        .map(|row| row.field_id)
677        .collect::<BTreeSet<_>>();
678    let spw_ids = fitted_rows
679        .iter()
680        .map(|row| row.spw_id)
681        .collect::<BTreeSet<_>>();
682    Ok(BandpassSolveReport {
683        output_table: request.output_table.clone(),
684        table_subtype: "BPOLY".to_string(),
685        refant_antenna_id: refant_id,
686        field_ids: field_ids.iter().copied().collect(),
687        spectral_window_ids: spw_ids.iter().copied().collect(),
688        solution_row_count: fitted_rows.len(),
689        channel_count,
690        selected_data: SelectedData::default(),
691        selection_failed: false,
692        solve_stats: SolveStats::default(),
693    })
694}

696fn fit_bpoly_rows(
697    ms: &MeasurementSet,
698    rows: &[BandpassSolutionRow],
699    amplitude_degree: usize,
700    phase_degree: usize,
701    cuts: Option<BandpassSolveCuts>,
702    selection: &casa_ms::MsSelection,
703) -> Result<Vec<BPolySolutionRow>, BandpassSolveError> {
704    let spectral_window =
705        ms.spectral_window()
706            .map_err(|source| BandpassSolveError::OpenMeasurementSet {
707                path: ms
708                    .path()
709                    .map(|path| path.display().to_string())
710                    .unwrap_or_else(|| "<in-memory>".to_string()),
711                source,
712            })?;
713    rows.iter()
714        .map(|row| {
715            let spw_row = usize::try_from(row.spw_id).map_err(|_| {
716                BandpassSolveError::UnsupportedParameterShape {
717                    path: "SPECTRAL_WINDOW row id".to_string(),
718                    shape: vec![row.spw_id as usize],
719                }
720            })?;
721            let channel_frequencies_hz = spectral_window.chan_freq(spw_row).map_err(|source| {
722                BandpassSolveError::OpenMeasurementSet {
723                    path: ms
724                        .path()
725                        .map(|path| path.display().to_string())
726                        .unwrap_or_else(|| "<in-memory>".to_string()),
727                    source,
728                }
729            })?;
730            let selected = selection.channel_selection_for_spw(row.spw_id);
731            let channel_frequencies_hz = crate::solve::grouping::selected_channels(
732                selected.as_ref(),
733                channel_frequencies_hz.len(),
734            )?
735            .into_iter()
736            .map(|channel| channel_frequencies_hz[channel])
737            .collect::<Vec<_>>();
738            fit_bpoly_row(
739                row,
740                &channel_frequencies_hz,
741                [amplitude_degree, phase_degree],
742                cuts,
743            )
744        })
745        .collect()
746}

748/// `row`'s channel solutions fitted with `BPOLY`'s polynomials over the
749/// window, of `[amplitude, phase]` degrees: each receptor's log amplitudes
750/// and unwrapped phases at its unflagged channels, leaving out the edges
751/// `cuts` masks. A receptor with no such channel keeps a unit gain.
752pub(super) fn fit_bpoly_row(
753    row: &BandpassSolutionRow,
754    channel_frequencies_hz: &[f64],
755    degrees: [usize; 2],
756    cuts: Option<BandpassSolveCuts>,
757) -> Result<BPolySolutionRow, BandpassSolveError> {
758    if channel_frequencies_hz.len() != row.channel_count || row.channel_count == 0 {
759        return Err(BandpassSolveError::UnsupportedParameterShape {
760            path: "BPOLY fit channel grid".to_string(),
761            shape: vec![channel_frequencies_hz.len(), row.channel_count],
762        });
763    }
764    let valid_domain_hz = [
765        channel_frequencies_hz.first().copied().unwrap_or(0.0),
766        channel_frequencies_hz.last().copied().unwrap_or(0.0),
767    ];
768    // Both receptors take as many coefficients, as the table holds one count.
769    let [amplitude_count, phase_count] =
770        degrees.map(|degree| degree.saturating_add(1).min(row.channel_count));
771    let receptor_count = 2;
772    let mut amp_coefficients = Vec::with_capacity(receptor_count);
773    let mut phase_coefficients = Vec::with_capacity(receptor_count);
774    for receptor in 0..receptor_count {
775        let mut frequencies_hz = Vec::with_capacity(row.channel_count);
776        let mut log_amplitudes = Vec::with_capacity(row.channel_count);
777        let mut phases = Vec::with_capacity(row.channel_count);
778        for (channel, frequency_hz) in channel_frequencies_hz.iter().copied().enumerate() {
779            let index = receptor + receptor_count * channel;
780            let masked =
781                cuts.is_some_and(|cuts| cuts.masks_edge_channel(channel, row.channel_count));
782            if row.flags[index] || masked {
783                continue;
784            }
785            let gain = row.gains[index];
786            frequencies_hz.push(frequency_hz);
787            log_amplitudes.push(gain.norm().max(f32::EPSILON).ln() as f64);
788            phases.push((gain.im as f64).atan2(gain.re as f64));
789        }
790        let unwrapped_phases = unwrap_phases(&phases);
791        amp_coefficients.push(fit_legacy_bpoly_coefficients(
792            &frequencies_hz,
793            &log_amplitudes,
794            valid_domain_hz,
795            amplitude_count,
796            row,
797        )?);
798        phase_coefficients.push(fit_legacy_bpoly_coefficients(
799            &frequencies_hz,
800            &unwrapped_phases,
801            valid_domain_hz,
802            phase_count,
803            row,
804        )?);
805    }
806    Ok(BPolySolutionRow {
807        time_seconds: row.time_seconds,
808        interval_seconds: row.interval_seconds,
809        field_id: row.field_id,
810        spw_id: row.spw_id,
811        antenna_id: row.antenna_id,
812        valid_domain_hz,
813        scan_number: row.scan_number,
814        observation_id: row.observation_id,
815        scale_factor: Complex32::new(1.0, 0.0),
816        amp_coefficients,
817        phase_coefficients,
818    })
819}

821/// `coefficient_count` coefficients of the polynomial through `values` at
822/// `frequencies_hz`. Fewer points than coefficients fit as many
823/// coefficients as there are points, and the rest are zero.
824fn fit_legacy_bpoly_coefficients(
825    frequencies_hz: &[f64],
826    values: &[f64],
827    valid_domain_hz: [f64; 2],
828    coefficient_count: usize,
829    row: &BandpassSolutionRow,
830) -> Result<Vec<f64>, BandpassSolveError> {
831    let mut coefficients = vec![0.0_f64; coefficient_count];
832    let fitted = coefficient_count.min(frequencies_hz.len());
833    if fitted == 0 {
834        return Ok(coefficients);
835    }
836
837    let mut basis = vec![0.0_f64; fitted];
838    let mut rows = Vec::with_capacity(frequencies_hz.len());
839    for (frequency_hz, value) in frequencies_hz.iter().copied().zip(values.iter().copied()) {
840        legacy_bpoly_basis(
841            valid_domain_hz[0],
842            valid_domain_hz[1],
843            frequency_hz,
844            &mut basis,
845        );
846        rows.push((basis.clone(), value, 1.0));
847    }
848    let solved = solve_weighted_least_squares(&rows, fitted).ok_or_else(|| {
849        BandpassSolveError::FitPolynomial {
850            antenna_id: row.antenna_id,
851            field_id: row.field_id,
852            spw_id: row.spw_id,
853            reason: "least-squares system was singular".to_string(),
854        }
855    })?;
856    coefficients[..fitted].copy_from_slice(&solved);
857    Ok(coefficients)
858}
859
860fn legacy_bpoly_basis(x_start: f64, x_end: f64, x: f64, basis: &mut [f64]) {
861    if basis.is_empty() {
862        return;
863    }
864    basis.fill(0.0);
865    basis[0] = 0.5_f64;
866    if basis.len() == 1 || (x_end - x_start).abs() <= f64::EPSILON {
867        return;
868    }
869    let xcap = ((x - x_start) - (x_end - x)) / (x_end - x_start);
870    let mut t_prev = 1.0_f64;
871    let mut t_curr = xcap;
872    basis[1] = t_curr;
873    for slot in basis.iter_mut().skip(2) {
874        let t_next = 2.0_f64 * xcap * t_curr - t_prev;
875        *slot = t_next;
876        t_prev = t_curr;
877        t_curr = t_next;
878    }
879}
880
881fn unwrap_phases(phases: &[f64]) -> Vec<f64> {
882    let mut unwrapped = Vec::with_capacity(phases.len());
883    let mut offset = 0.0_f64;
884    let mut previous = None;
885    for phase in phases.iter().copied() {
886        if let Some(previous_phase) = previous {
887            let delta = phase - previous_phase;
888            if delta > std::f64::consts::PI {
889                offset -= 2.0_f64 * std::f64::consts::PI;
890            } else if delta < -std::f64::consts::PI {
891                offset += 2.0_f64 * std::f64::consts::PI;
892            }
893        }
894        let adjusted = phase + offset;
895        unwrapped.push(adjusted);
896        previous = Some(phase);
897    }
898    unwrapped
899}

901fn write_bpoly_cal_desc_subtable(
902    path: PathBuf,
903    cal_desc_ids: &BTreeMap<i32, i32>,
904    rows: &[BPolySolutionRow],
905    casa_layout: bool,
906    ms_name: &str,
907) -> Result<(), BandpassSolveError> {
908    use casa_types::PrimitiveType::{Float64, Int32, String as Text};
909    let mut columns = vec![
910        ColumnSchema::scalar("NUM_RECEPTORS", casa_types::PrimitiveType::Int32),
911        ColumnSchema::array_variable(
912            "SPECTRAL_WINDOW_ID",
913            casa_types::PrimitiveType::Int32,
914            Some(1),
915        ),
916    ];
917    if casa_layout {
918        columns.extend(
919            ["NUM_SPW", "N_JONES", "MEAS_FREQ_REF"].map(|name| ColumnSchema::scalar(name, Int32)),
920        );
921        columns.extend(
922            ["NUM_CHAN", "CHAN_RANGE"].map(|name| ColumnSchema::array_variable(name, Int32, None)),
923        );
924        columns.extend(
925            ["CHAN_FREQ", "CHAN_WIDTH"]
926                .map(|name| ColumnSchema::array_variable(name, Float64, None)),
927        );
928        columns.extend(["JONES_TYPE", "MS_NAME"].map(|name| ColumnSchema::scalar(name, Text)));
929        columns.push(ColumnSchema::array_variable(
930            "POLARIZATION_TYPE",
931            Text,
932            None,
933        ));
934    }
935    let schema = TableSchema::new(columns).expect("BPOLY CAL_DESC schema");
936    let mut table = Table::with_schema(schema);
937    if casa_layout {
938        set_fixed_unit_keyword(&mut table, "CHAN_FREQ", &["Hz"]);
939        set_measinfo_keyword(&mut table, "CHAN_FREQ", "frequency", Some("REST"));
940        set_fixed_unit_keyword(&mut table, "CHAN_WIDTH", &["Hz"]);
941    }
942    for spw_id in cal_desc_ids.keys() {
943        let receptor_count = if casa_layout {
944            0
945        } else {
946            rows.iter()
947                .find(|row| row.spw_id == *spw_id)
948                .map_or(1, |row| row.amp_coefficients.len() as i32)
949        };
950        let mut fields = vec![
951            RecordField::new(
952                "NUM_RECEPTORS",
953                Value::Scalar(ScalarValue::Int32(receptor_count)),
954            ),
955            RecordField::new(
956                "SPECTRAL_WINDOW_ID",
957                Value::Array(ArrayValue::from_i32_vec(vec![*spw_id])),
958            ),
959        ];
960        if casa_layout {
961            fields.extend(
962                ["NUM_SPW", "N_JONES", "MEAS_FREQ_REF"]
963                    .map(|name| RecordField::new(name, Value::Scalar(ScalarValue::Int32(0)))),
964            );
965            fields.push(RecordField::new(
966                "JONES_TYPE",
967                Value::Scalar(ScalarValue::String(String::new())),
968            ));
969            fields.push(RecordField::new(
970                "MS_NAME",
971                Value::Scalar(ScalarValue::String(ms_name.to_string())),
972            ));
973        }
974        table
975            .add_row(RecordValue::new(fields))
976            .expect("insert BPOLY CAL_DESC row");
977    }
978    table.save(TableOptions::new(&path)).map_err(|source| {
979        BandpassSolveError::SaveCalibrationTable {
980            path: path.display().to_string(),
981            source: Box::new(source),
982        }
983    })
984}
