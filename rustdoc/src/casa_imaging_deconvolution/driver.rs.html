2//! The minor-cycle driver: one plane, step by step, under the plane
3//! controller (`SDAlgorithmBase::deconvolve`).
4
5use crate::Error;
6use crate::controller::{CycleControls, PlaneControl, PlaneStatistics, PlaneStop};
7use crate::solver::{Candidate, Delta, MinorCycleView, Next, Solver};

9/// One model term of one accepted component, for diagnostics: a multi-term
10/// component gives one record per Taylor term.
11#[derive(Clone, Copy, Debug, PartialEq)]
12pub struct Component {
13    /// Storage index of the component centre.
14    pub index: usize,
15    /// Scale ordinal (0 for a point).
16    pub scale: usize,
17    /// The model term (Taylor coefficient) the flux goes to.
18    pub term: usize,
19    /// The term's flux after the loop gain.
20    pub flux: f64,
21}
22
23/// What one plane's minor cycle did.
24#[derive(Clone, Debug, PartialEq)]
25pub struct PlaneOutcome {
26    /// The model update, per term.
27    pub delta: Delta,
28    /// Iterations charged to the cycle (CASA's `iterdone`).
29    pub iterations: usize,
30    /// Components actually cleaned.
31    pub components: usize,
32    /// Sum of the absolute component fluxes over every term.
33    pub absolute_flux: f64,
34    /// The peak residual on entry.
35    pub start_peak: f64,
36    /// The peak residual the last step reported.
37    pub peak: f64,
38    /// Why the plane stopped.
39    pub stop: PlaneStop,
40    /// The first component records, up to the requested count.
41    pub trace: Vec<Component>,
42    /// Exact whole-plane residual refreshes.
43    pub refreshes: usize,
44}
45
46/// Run one plane's minor cycle.
47///
48/// `statistics` is the plane's own measurement before the cycle (its entry
49/// peak and noise). The plane is skipped when it has no cleanable pixel or
50/// its peak is already at the cycle threshold; otherwise the solver runs in
51/// steps of [`CycleControls::step_iterations`] until the plane controller
52/// stops it. `trace` asks for the first components.
53///
54/// # Errors
55///
56/// [`Error::NonFinite`] when a plane to clean holds a NaN or an infinity in
57/// a residual or PSF term; otherwise when the solver fails.
58pub fn run_plane<S: Solver>(
59    solver: &S,
60    view: &MinorCycleView<'_>,
61    cycle: &CycleControls,
62    statistics: &PlaneStatistics,
63    trace: usize,
64) -> Result<PlaneOutcome, Error> {
65    let start_peak = statistics.peak();
66    let mut control = PlaneControl::new(cycle, statistics, start_peak);
67    let mut outcome = PlaneOutcome {
68        delta: Delta::new(view.residual.len()),
69        iterations: 0,
70        components: 0,
71        absolute_flux: 0.0,
72        start_peak,
73        peak: start_peak,
74        stop: PlaneStop::ZeroMask,
75        trace: Vec::new(),
76        refreshes: 0,
77    };
78    let mut stop = control.stop(start_peak);
79    if view.support.is_empty() || stop.is_some() {
80        outcome.stop = stop.unwrap_or(PlaneStop::ZeroMask);
81        return Ok(outcome);
82    }
83    if view
84        .residual
85        .iter()
86        .chain(view.psf)
87        .flatten()
88        .any(|value| !value.is_finite())
89    {
90        return Err(Error::NonFinite);
91    }
92    let mut residual = view.residual.to_vec();
93    let threshold = control.step_threshold();
94    let step_budget = cycle.step_iterations();
95    while stop.is_none() {
96        // SDAlgorithmBase::deconvolve: setPeakResidual with the previous
97        // step's peak, takeOneStep, then checkStop on the new peak.
98        control.observe(outcome.peak);
99        let step = run_step(
100            solver,
101            view,
102            &mut residual,
103            step_budget,
104            threshold,
105            cycle.gain,
106            trace,
107            &mut outcome,
108        )?;
109        control.charge(step);
110        stop = control.stop(outcome.peak);
111        if stop.is_none() && step != step_budget {
112            stop = Some(PlaneStop::Exited);
113        }
114    }
115    outcome.stop = stop.expect("the step loop ends on a stop");
116    Ok(outcome)
117}

119/// One solver step (`takeOneStep`): returns the iterations it charges.
120#[allow(clippy::too_many_arguments)]
121fn run_step<S: Solver>(
122    solver: &S,
123    view: &MinorCycleView<'_>,
124    residual: &mut [Vec<f64>],
125    budget: usize,
126    threshold: f64,
127    gain: f64,
128    trace: usize,
129    outcome: &mut PlaneOutcome,
130) -> Result<usize, Error> {
131    let mut state = solver.initialize(view, residual, threshold)?;
132    let mut delta = Delta::new(residual.len());
133    let limit = budget + solver.extra_iterations();
134    let mut components = 0;
135    let mut charged_stop = 0;
136    let mut fluxes = Vec::new();
137    while components < limit {
138        match solver.next(&mut state, view, residual)? {
139            Next::Stop => {
140                charged_stop = usize::from(solver.charges_stop());
141                break;
142            }
143            Next::Clean(candidate) => {
144                fluxes.clear();
145                solver.component_fluxes(&state, candidate, gain, &mut fluxes);
146                solver.accept(&mut state, view, residual, candidate, gain, &mut delta)?;
147                let Candidate::Pixel { index, scale, .. } = candidate;
148                outcome.absolute_flux += fluxes.iter().map(|flux| flux.abs()).sum::<f64>();
149                for (term, &flux) in fluxes.iter().enumerate() {
150                    if outcome.trace.len() < trace {
151                        outcome.trace.push(Component {
152                            index,
153                            scale,
154                            term,
155                            flux,
156                        });
157                    }
158                }
159                components += 1;
160            }
161        }
162    }
163    let end = solver.finalize(state, view, residual, &delta)?;
164    outcome.peak = end.peak;
165    outcome.refreshes += end.refreshes;
166    outcome.delta.merge(&delta);
167    outcome.components += components;
168    let charged = components.min(budget) + charged_stop;
169    outcome.iterations += charged;
170    Ok(charged)
171}
