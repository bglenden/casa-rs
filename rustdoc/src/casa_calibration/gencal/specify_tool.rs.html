13pub(super) struct PriorCalibrationTool<'a> {
14    pub(super) legacy_iterator: bool,
15    pub(super) selection: &'a MsSelection,
16}

18/// A parameter prescription passed to the attached calibrater tool.
19pub struct CalibrationSpecification<'a> {
20    /// Calibration table to create or update.
21    pub table: &'a Path,
22    /// Spectral-window selection.
23    pub spw: &'a str,
24    /// Antenna selection.
25    pub antenna: &'a str,
26    /// Receptor selection and order.
27    pub polarization: &'a str,
28    /// CASA tool factory name or parameter type.
29    pub caltype: &'a str,
30    /// Numeric prescription in selection order.
31    pub parameters: Vec<f64>,
32    /// Optional calibration input used by the selected family.
33    pub infile: Option<&'a Path>,
34    /// Use uniform rather than exposure weights where the family supports it.
35    pub uniform: bool,
36}
37
38impl CalibrationSpecification<'_> {
39    pub(crate) fn write(
40        self,
41        dataset: &Path,
42        legacy_iterator: bool,
43        selection: &MsSelection,
44    ) -> Result<(), GencalError> {
45        if self.table.as_os_str().is_empty() {
46            return Err(GencalError::InvalidRequest(
47                "No output calibration table was specified.".into(),
48            ));
49        }
50        let kind = self.caltype.to_ascii_uppercase();
51        let (caltype, sky_subtype) = if let Some(caltype) = prior_type(&kind) {
52            (caltype, None)
53        } else if let Some(subtype) = ["SDSKY_PS", "SDSKY_RASTER", "SDSKY_OTF"]
54            .into_iter()
55            .find(|subtype| kind.contains(subtype))
56        {
57            if subtype == "SDSKY_OTF" && !legacy_iterator {
58                return Err(GencalError::InvalidRequest(
59                    "SDSKY_OTF not yet supported via SingleDishOtfCal(msmc)".into(),
60                ));
61            }
62            // The generic sky and delay prescriptions use the same two real
63            // additive parameters. The writer receives the actual sky subtype.
64            (GencalType::Sbd, Some(subtype))
65        } else {
66            return Err(GencalError::InvalidRequest(format!(
67                "Unrecognized caltype {:?}.",
68                self.caltype
69            )));
70        };
71        let antpos = matches!(caltype, GencalType::Antpos | GencalType::AntposVla);
72        let antenna = if antpos && self.antenna.trim().is_empty() {
73            "*"
74        } else {
75            self.antenna
76        };
77        let request = GencalRequest {
78            measurement_set: dataset.to_owned(),
79            output_table: self.table.to_owned(),
80            caltype,
81            antenna: antenna.to_owned(),
82            spw: self.spw.to_owned(),
83            parameter: self.parameters,
84            gaincurve_table: None,
85        };
86        if let Some(subtype) = sky_subtype {
87            let ms = open_measurement_set(dataset)?;
88            // Pinned CASA's sky factories can create a specified table, but
89            // reject loading their explicit rows again for an update.
90            let mode = CalibrationOutputMode::Create;
91            let output = crate::output::CalibrationOutput::prepare(self.table, mode)?;
92            let rows = specified_rows(&ms, &request, self.polarization, false)?;
93            write_prior_caltable(&ms, &request, subtype, &rows, output, mode, true)?;
94        } else {
95            generate_prior(
96                &request,
97                self.polarization,
98                self.infile.filter(|_| !antpos),
99                None,
100                self.uniform,
101                CalibrationOutputMode::Update,
102                Some(PriorCalibrationTool {
103                    legacy_iterator,
104                    selection,
105                }),
106            )?;
107        }
108        Ok(())
109    }
110}
111
112fn prior_type(kind: &str) -> Option<GencalType> {
113    Some(match kind {
114        kind if kind.contains("PH") => GencalType::Ph,
115        kind if kind == "G" || kind.contains("AMP") => GencalType::Amp,
116        kind if kind == "K"
117            || ["SBD", "DELAY", "MBD"]
118                .iter()
119                .any(|name| kind.contains(name)) =>
120        {
121            GencalType::Sbd
122        }
123        kind if kind.contains("ANTPOS") && kind.contains("VLA") => GencalType::AntposVla,
124        kind if kind.contains("ANTPOS") => GencalType::Antpos,
125        kind if kind.contains("TSYS") => GencalType::Tsys,
126        kind if ["EVLAGAIN", "SWP", "RQ", "WTS"]
127            .iter()
128            .any(|token| kind.contains(token)) =>
129        {
130            [
131                ("SWP/RQ", GencalType::SwpOverRq),
132                ("SWPOW", GencalType::Swpow),
133                ("EVLAGAIN", GencalType::Evlagain),
134                ("RQ", GencalType::Rq),
135                ("SWPWTS", GencalType::Swpwts),
136            ]
137            .into_iter()
138            .find_map(|(token, family)| kind.contains(token).then_some(family))?
139        }
140        kind if kind.contains("OPAC") => GencalType::Opac,
141        kind if kind.contains("GC") && kind.contains("EFF") => GencalType::Gceff,
142        kind if kind.contains("GC") => GencalType::Gc,
143        kind if kind.contains("EFF") => GencalType::Eff,
144        "TECIM" => GencalType::Tecim,
145        _ => return None,
146    })
147}
