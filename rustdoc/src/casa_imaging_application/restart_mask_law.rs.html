2//! Counting-law access to the same existing CLEAN-mask reader compile uses.
3
4use std::path::Path;
5
6/// An opened mask header, before any pixels are read.
7pub struct Input(crate::restart::MaskInput);
8
9impl Input {
10    /// Open the header with the production reader's bounded tile cache.
11    ///
12    /// # Errors
13    /// The image cannot be opened as a real single-precision image.
14    pub fn open(path: &Path) -> Result<Self, crate::PrepareError> {
15        crate::restart::MaskInput::open(path).map(Self)
16    }
17
18    /// Bytes admitted before reading and converting the pixels.
19    pub fn reading_bytes(&self) -> u64 {
20        self.0.reading_bytes()
21    }
22
23    /// Bytes retained for the static mask plan and its run-time clone, of
24    /// `planes` planes of `cells` pixels each.
25    pub fn retained_bytes(planes: casa_imaging_deconvolution::MaskPlanes, cells: usize) -> u64 {
26        crate::restart::MaskInput::retained_bytes(planes, cells)
27    }
28
29    /// Read exactly the checked grid, in the reconstruction mask's canonical
30    /// order: one plane when every plane agrees, else every plane.
31    ///
32    /// # Errors
33    /// Pixel I/O fails or the mask is not binary.
34    pub fn read(
35        self,
36    ) -> Result<(casa_imaging_deconvolution::MaskPlanes, Box<[bool]>), crate::PrepareError> {
37        self.0.read()
38    }
39}
