2//! Calibration tables of simulated antenna-based terms: what CASA's
3//! simulator writes for `sm.setgain(table=...)` and
4//! `sm.setleakage(table=...)` (`SolvableVisCal::setSimulate` and
5//! `storeNCT`), written here for `casa-simulation`.
6//!
7//! The layout is CASA 6.7.6's for a simulated `G Jones` or `D Jones` term
8//! (measured): one row per solution and antenna, ANTENNA2 −1, INTERVAL 0,
9//! SCAN_NUMBER −1, OBSERVATION_ID 0, CPARAM, PARAMERR, FLAG and SNR of one
10//! channel per receptor (`[2, 1]`), PARAMERR and SNR zero, FLAG false, and
11//! WEIGHT left undefined; the OBSERVATION, ANTENNA, FIELD and HISTORY
12//! subtables are the MeasurementSet's, and its SPECTRAL_WINDOW with each
13//! window reduced to one channel. As CASA's `NewCalTable::writeToDisk`, a
14//! table already at the output is replaced; the new table is built beside
15//! it and the old one removed only once the new one is in place. An output
16//! that is the MeasurementSet, lies inside it, or contains it is refused
17//! before anything is touched (#891 review).
18//!
19//! The table is built in memory and saved: [`simulated_table_bytes`] is
20//! what that holds, for the caller to admit.
21

41/// Which simulated term a table holds.
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub enum SimulatedJones {
44    /// `sm.setgain`: antenna gains, `G Jones`.
45    Gain,
46    /// `sm.setleakage`: polarization leakage, `D Jones`.
47    Leakage,
48}
49
50impl SimulatedJones {
51    /// The table's `VisCal` keyword and TableInfo subtype.
52    #[must_use]
53    pub fn vis_cal(self) -> &'static str {
54        match self {
55            Self::Gain => "G Jones",
56            Self::Leakage => "D Jones",
57        }
58    }
59}

61/// One antenna's solution: a row of the table.
62#[derive(Debug, Clone, Copy, PartialEq)]
63pub struct SimulatedSolution {
64    /// TIME, seconds (MJD): the solution's time.
65    pub time_s: f64,
66    /// FIELD_ID of the solution.
67    pub field_id: i32,
68    /// SPECTRAL_WINDOW_ID of the solution.
69    pub spectral_window_id: i32,
70    /// ANTENNA1.
71    pub antenna: i32,
72    /// The receptors' values, CPARAM.
73    pub receptors: [Complex32; 2],
74}

76/// Heap a simulated table holds for each row while it is built in memory
77/// and saved: the row's record (twelve named cells, four of them arrays)
78/// and its share of the columns the save forms. Measured at about 2,050
79/// bytes (the #891 review's 36,000-row probe, and the table law in
80/// `casa-simulation`'s `session_memory/jones.rs`); a quarter more is kept
81/// for the allocator.
82pub const SIMULATED_TABLE_ROW_BYTES: u64 = 2_560;
83
84/// Heap writing a simulated table of `rows` rows holds beyond its fixed
85/// metadata.
86#[must_use]
87pub const fn simulated_table_bytes(rows: u64) -> u64 {
88    rows * SIMULATED_TABLE_ROW_BYTES
89}

91/// Refuse `output` as a simulated table of the MeasurementSet at
92/// `measurement_set` when it is that MeasurementSet, lies inside it (a
93/// subtable), or contains it: replacing it would remove the data the table
94/// is written from. Paths are compared after resolving symbolic links and
95/// `..` in the part of each that exists, so an alias is caught as well.
96///
97/// # Errors
98///
99/// [`CalibrationTableWriteError::Prepare`] naming the conflict.
100pub fn check_simulated_table_destination(
101    measurement_set: &Path,
102    output: &Path,
103) -> Result<(), CalibrationTableWriteError> {
104    let (table, ms) = (resolved(output), resolved(measurement_set));
105    let conflict = if table == ms {
106        "is the MeasurementSet it is written from"
107    } else if table.starts_with(&ms) {
108        "lies inside the MeasurementSet it is written from"
109    } else if ms.starts_with(&table) {
110        "contains the MeasurementSet it is written from"
111    } else {
112        return Ok(());
113    };
114    Err(CalibrationTableWriteError::Prepare {
115        path: output.display().to_string(),
116        reason: format!("{conflict} ({})", measurement_set.display()),
117    })
118}
119
120/// `path` made absolute, with the longest part of it that exists resolved
121/// (symbolic links and `..`) and the rest normalised lexically.
122fn resolved(path: &Path) -> PathBuf {
123    let absolute = std::path::absolute(path).unwrap_or_else(|_| path.to_path_buf());
124    let mut base = PathBuf::new();
125    let mut exists = true;
126    for component in absolute.components() {
127        match component {
128            Component::CurDir => continue,
129            Component::ParentDir if !exists => {
130                base.pop();
131                continue;
132            }
133            other => base.push(other),
134        }
135        if exists {
136            match base.canonicalize() {
137                Ok(canonical) => base = canonical,
138                Err(_) => exists = false,
139            }
140        }
141    }
142    base
143}

145/// Write `solutions` of `kind` as the calibration table `output` of the
146/// MeasurementSet `ms`, replacing a table already there once the new one is
147/// saved. The rows are formed one at a time from `solutions`.
148///
149/// # Errors
150///
151/// `output` is, lies inside or contains the MeasurementSet
152/// ([`check_simulated_table_destination`]); `output` exists and is not a
153/// table; or the table cannot be written.
154pub fn write_simulated_jones_table(
155    ms: &MeasurementSet,
156    output: &Path,
157    kind: SimulatedJones,
158    solutions: impl IntoIterator<Item = SimulatedSolution>,
159) -> Result<(), CalibrationTableWriteError> {
160    if let Some(path) = ms.path() {
161        check_simulated_table_destination(path, output)?;
162    }
163    let measurement_set_name = ms.path().and_then(Path::file_name).map_or_else(
164        || "<in-memory>".to_string(),
165        |name| name.to_string_lossy().into_owned(),
166    );
167    let mut writer = CalibrationTableWriter::create_replacing(
168        ms,
169        CalibrationTableDescriptor {
170            output,
171            schema: schema(),
172            subtype: kind.vis_cal(),
173            parameter_type: Some("Complex"),
174            measurement_set_name,
175            include_polarization_basis: true,
176            time_extra_precision_column: None,
177        },
178    )?;
179    for solution in solutions {
180        writer.append(row(&solution))?;
181    }
182    writer.copy_subtables(&[
183        (SubtableId::Observation, "OBSERVATION"),
184        (SubtableId::Antenna, "ANTENNA"),
185        (SubtableId::Field, "FIELD"),
186        (SubtableId::History, "HISTORY"),
187        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
188    ])?;
189    make_windows_single_channel(&writer.output_path().join("SPECTRAL_WINDOW"))?;
190    writer.save_main()
191}

193/// Reduce every spectral window of the SPECTRAL_WINDOW subtable at `path`
194/// to one channel, as CASA's `NewCalTable::makeSpwSingleChan` does for a
195/// term that does not depend on frequency: the middle channel's frequency
196/// (`CHAN_FREQ[n/2]`), and CHAN_WIDTH, EFFECTIVE_BW and RESOLUTION the
197/// window's TOTAL_BANDWIDTH. CASA's `applycal` refuses a table whose
198/// windows keep the MeasurementSet's channels.
199fn make_windows_single_channel(path: &Path) -> Result<(), CalibrationTableWriteError> {
200    let failed = |source: TableError| CalibrationTableWriteError::Save {
201        path: path.display().to_string(),
202        source: Box::new(source),
203    };
204    let mut table = Table::open(TableOptions::new(path)).map_err(failed)?;
205    for row in 0..table.row_count() {
206        let value = |column: &str| -> Result<Value, TableError> {
207            Ok(table
208                .cell_accessor(row, column)?
209                .value()?
210                .cloned()
211                .unwrap_or_else(|| Value::Scalar(ScalarValue::Int32(0))))
212        };
213        let frequencies = match value("CHAN_FREQ").map_err(failed)? {
214            Value::Array(ArrayValue::Float64(frequencies)) => frequencies,
215            _ => continue,
216        };
217        let channels = frequencies.len();
218        if channels <= 1 {
219            continue;
220        }
221        let bandwidth = match value("TOTAL_BANDWIDTH").map_err(failed)? {
222            Value::Scalar(ScalarValue::Float64(bandwidth)) => bandwidth,
223            _ => continue,
224        };
225        let one = |value: f64| {
226            Value::Array(ArrayValue::Float64(
227                ArrayD::from_shape_vec(IxDyn(&[1]), vec![value]).expect("one channel"),
228            ))
229        };
230        let middle = frequencies
231            .iter()
232            .nth(channels / 2)
233            .copied()
234            .unwrap_or_default();
235        for (column, cell) in [
236            ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(1))),
237            ("CHAN_FREQ", one(middle)),
238            ("CHAN_WIDTH", one(bandwidth)),
239            ("EFFECTIVE_BW", one(bandwidth)),
240            ("RESOLUTION", one(bandwidth)),
241        ] {
242            table
243                .cell_accessor_mut(row, column)
244                .and_then(|mut accessor| accessor.set(cell))
245                .map_err(failed)?;
246        }
247    }
248    table.flush().map_err(failed)
249}

251fn schema() -> TableSchema {
252    TableSchema::new(vec![
253        ColumnSchema::scalar(COL_TIME, PrimitiveType::Float64),
254        ColumnSchema::scalar(COL_FIELD_ID, PrimitiveType::Int32),
255        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, PrimitiveType::Int32),
256        ColumnSchema::scalar(COL_ANTENNA1, PrimitiveType::Int32),
257        ColumnSchema::scalar(COL_ANTENNA2, PrimitiveType::Int32),
258        ColumnSchema::scalar(COL_INTERVAL, PrimitiveType::Float64),
259        ColumnSchema::scalar(COL_SCAN_NUMBER, PrimitiveType::Int32),
260        ColumnSchema::scalar(COL_OBSERVATION_ID, PrimitiveType::Int32),
261        ColumnSchema::array_variable(COL_CPARAM, PrimitiveType::Complex32, None),
262        ColumnSchema::array_variable(COL_PARAMERR, PrimitiveType::Float32, None),
263        ColumnSchema::array_variable(COL_FLAG, PrimitiveType::Bool, None),
264        ColumnSchema::array_variable(COL_SNR, PrimitiveType::Float32, None),
265        ColumnSchema::array_variable(COL_WEIGHT, PrimitiveType::Float32, None),
266    ])
267    .expect("valid simulated calibration schema")
268}

270/// A `[receptor, channel]` cell of two receptors and one channel.
271fn cell<T>(values: [T; 2]) -> ArrayD<T> {
272    ArrayD::from_shape_vec(IxDyn(&[2, 1]).f(), Vec::from(values))
273        .expect("two receptors by one channel")
274}
275
276fn row(solution: &SimulatedSolution) -> RecordValue {
277    let int =
278        |name: &str, value: i32| RecordField::new(name, Value::Scalar(ScalarValue::Int32(value)));
279    let double =
280        |name: &str, value: f64| RecordField::new(name, Value::Scalar(ScalarValue::Float64(value)));
281    RecordValue::new(vec![
282        double(COL_TIME, solution.time_s),
283        int(COL_FIELD_ID, solution.field_id),
284        int(COL_SPECTRAL_WINDOW_ID, solution.spectral_window_id),
285        int(COL_ANTENNA1, solution.antenna),
286        int(COL_ANTENNA2, -1),
287        double(COL_INTERVAL, 0.0),
288        int(COL_SCAN_NUMBER, -1),
289        int(COL_OBSERVATION_ID, 0),
290        RecordField::new(
291            COL_CPARAM,
292            Value::Array(ArrayValue::Complex32(cell(solution.receptors))),
293        ),
294        RecordField::new(
295            COL_PARAMERR,
296            Value::Array(ArrayValue::Float32(cell([0.0; 2]))),
297        ),
298        RecordField::new(COL_FLAG, Value::Array(ArrayValue::Bool(cell([false; 2])))),
299        RecordField::new(COL_SNR, Value::Array(ArrayValue::Float32(cell([0.0; 2])))),
300    ])
301}
