15const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

17/// One generated, independently cropped plane, `size × size` with x
18/// fastest and the kernel origin at `size / 2`.
19#[derive(Debug)]
20pub struct NativeAwPlane<'a> {
21    /// Square pixel extent after support-buffer cropping.
22    pub size: usize,
23    /// Symmetric support in non-oversampled grid pixels.
24    pub support: usize,
25    /// Complex32 samples with the CASA sampled-area normalisation applied.
26    pub values: &'a [Complex32],
27}
28
29/// Imaging and weight cells produced from the same scientific request.
30#[derive(Debug)]
31pub struct NativeAwPair<'a> {
32    /// A times W times the optional anti-aliasing term.
33    pub imaging: NativeAwPlane<'a>,
34    /// Frequency/conjugate-frequency beam product, without the W phase.
35    pub weight: NativeAwPlane<'a>,
36}

38/// Reusable cell workspace: seven square `Complex32` planes, the support
39/// search's ring table, the W screen's quadrant tables and one FFT plan,
40/// independent of catalog size.
41///
42/// A cell is the sky Mueller element times the W screen, transformed; only
43/// the screen depends on w. [`Self::prepare`] forms the w-independent part
44/// once (the aperture plane and the whole weight cell) and
45/// [`Self::realize`] then forms the imaging cell of each w from it with one
46/// transform, the arithmetic [`Self::generate`] performs per cell.
47///
48/// The workspace serves one aperture model, bound when it is made, so a
49/// prepared aperture always belongs to the model its cells are asked of.
50pub struct EvlaAwWorkspace {
51    model: EvlaApertureModel,
52    size: usize,
53    fft: PlaneFft<f32>,
54    jones: Vec<Complex32>,
55    /// The sky Mueller element at the prepared cell's frequency.
56    aperture: Vec<Complex32>,
57    imaging: Vec<Complex32>,
58    /// The prepared cell's cropped, normalised weight cell.
59    weight: Vec<Complex32>,
60    /// The request [`Self::prepare`] formed the aperture and weight cell
61    /// for, at w = 0, and the weight cell's `(size, support)`.
62    prepared: Option<(EvlaAwCellRequest, (usize, usize))>,
63    /// The imaging crop of the last realized pair.
64    realized: Option<(usize, usize)>,
65    /// Per working-grid pixel, the largest ring [`crop_normalize`]'s
66    /// support search samples it on (zero when none does).
67    rings: Vec<u16>,
68    /// `√(1 − l² − m²) − 1` over one quadrant of pixel offsets
69    /// `(|ix|, |iy|)`, for the sky increments it was formed for; the W
70    /// screen's phase is `2π·w` times it.
71    depth: Vec<f64>,
72    depth_increment: Option<[f64; 2]>,
73    /// The W screen's phase factors over the same quadrant, for one w.
74    phases: Vec<Complex32>,
75}
76
77impl EvlaAwWorkspace {
78    /// Allocate the working grid of `request`'s size for `model`'s cells;
79    /// the same allocation serves every cell of that size.
80    ///
81    /// # Errors
82    ///
83    /// The request is invalid or the FFT cannot be planned.
84    pub fn new(
85        model: EvlaApertureModel,
86        request: EvlaAwCellRequest,
87    ) -> Result<Self, NativeAwGenerationError> {
88        request
89            .validate()
90            .map_err(|_| NativeAwGenerationError::InvalidGrid)?;
91        let n = request.size * request.size;
92        let quadrant = (request.size / 2 + 1) * (request.size / 2 + 1);
93        Ok(Self {
94            model,
95            size: request.size,
96            fft: PlaneFft::<f32>::new([request.size; 2], false)
97                .map_err(|_| NativeAwGenerationError::WorkspaceMismatch)?,
98            jones: vec![Complex32::default(); 4 * n],
99            aperture: vec![Complex32::default(); n],
100            imaging: vec![Complex32::default(); n],
101            weight: vec![Complex32::default(); n],
102            prepared: None,
103            realized: None,
104            rings: ring_table(request.size),
105            depth: vec![0.0; quadrant],
106            depth_increment: None,
107            phases: vec![Complex32::default(); quadrant],
108        })
109    }
110
111    /// Bytes of the working planes and tables.
112    #[must_use]
113    pub fn resident_bytes(&self) -> usize {
114        Self::bytes(self.size)
115    }
116
117    /// Bytes of the working planes and tables of a workspace of `size`
118    /// ([`Self::new`]): the four Jones planes, the aperture, imaging and
119    /// weight planes, the ring table, and the W screen's depth and phases
120    /// over one quadrant.
121    #[must_use]
122    pub const fn bytes(size: usize) -> usize {
123        let quadrant = (size / 2 + 1) * (size / 2 + 1);
124        7 * size * size * size_of::<Complex32>()
125            + size * size * size_of::<u16>()
126            + quadrant * (size_of::<f64>() + size_of::<Complex32>())
127    }
128
129    /// Borrow the last successfully realized pair, with no pixel copies.
130    #[must_use]
131    pub fn pair(&self) -> Option<NativeAwPair<'_>> {
132        let (imaging_size, imaging_support) = self.realized?;
133        let (_, (weight_size, weight_support)) = self.prepared?;
134        Some(NativeAwPair {
135            imaging: NativeAwPlane {
136                size: imaging_size,
137                support: imaging_support,
138                values: &self.imaging[..imaging_size * imaging_size],
139            },
140            weight: NativeAwPlane {
141                size: weight_size,
142                support: weight_support,
143                values: &self.weight[..weight_size * weight_size],
144            },
145        })
146    }

148    /// Generate a paired EVLA A/W cell without CASA or cache access
149    /// (`AWConvFunc::makeConvFunction2`): the sky Mueller element at the
150    /// cell frequency times the W screen and the optional prolate
151    /// spheroidal, and its weight partner at the conjugate frequency; both
152    /// transformed, the imaging plane peak-normalised, both cropped to
153    /// their support buffers and normalised by their sampled area.
154    ///
155    /// # Errors
156    ///
157    /// The request does not match the workspace, the receiver band is
158    /// unsupported, or a value or normalisation is not finite.
159    pub fn generate(
160        &mut self,
161        request: EvlaAwCellRequest,
162    ) -> Result<NativeAwPair<'_>, NativeAwGenerationError> {
163        self.prepare(request)?;
164        self.realize(request)
165    }
166
167    /// Form the w-independent part of `request`'s cell, unless it is the
168    /// one already prepared: the sky Mueller element at the cell frequency,
169    /// kept for [`Self::realize`], and the whole weight cell, which carries
170    /// no W phase. `request.w_wavelengths` is ignored.
171    ///
172    /// # Errors
173    ///
174    /// As [`Self::generate`].
175    pub fn prepare(&mut self, request: EvlaAwCellRequest) -> Result<(), NativeAwGenerationError> {
176        self.realized = None;
177        request
178            .validate()
179            .map_err(|_| NativeAwGenerationError::InvalidGrid)?;
180        if self.size != request.size {
181            return Err(NativeAwGenerationError::WorkspaceMismatch);
182        }
183        let key = EvlaAwCellRequest {
184            w_wavelengths: 0.0,
185            ..request
186        };
187        if self.prepared.is_some_and(|(held, _)| held == key) {
188            return Ok(());
189        }
190        self.prepared = None;
191        let Self {
192            model,
193            fft,
194            jones,
195            aperture,
196            weight,
197            rings,
198            ..
199        } = self;
200        fill_sky_mueller(model, request, request.frequency_hz, fft, jones, aperture)?;
201        fill_sky_mueller(
202            model,
203            request,
204            request.conjugate_frequency_hz,
205            fft,
206            jones,
207            weight,
208        )?;
209        let origin = (request.size / 2) as isize;
210        let inner_half = (request.size / request.oversampling / 2) as isize;
211        for y in 0..request.size {
212            let iy = y as isize - origin;
213            for x in 0..request.size {
214                let ix = x as isize - origin;
215                let p = prolate(request, inner_half, ix, iy);
216                let pixel = x + request.size * y;
217                weight[pixel] =
218                    Complex32::new(p * p, 0.0) * (aperture[pixel] * weight[pixel].conj());
219            }
220        }
221        // CASA's copy back from the FFT image excludes the final row and
222        // column; the pre-transform edges survive in the spent Jones planes.
223        let edges = &mut jones[..2 * request.size];
224        save_edges(weight, request.size, edges);
225        transform(fft, weight)?;
226        restore_edges(weight, request.size, edges);
227        let weight_crop = crop_normalize(weight, request, rings)?;
228        self.prepared = Some((key, weight_crop));
229        Ok(())
230    }
231
232    /// Form the imaging cell of `request` at its w from the aperture
233    /// [`Self::prepare`] formed for the same request: the aperture times the
234    /// W screen and the optional prolate spheroidal, transformed,
235    /// peak-normalised and cropped, beside the prepared weight cell.
236    ///
237    /// # Errors
238    ///
239    /// The workspace holds no aperture for `request` (other than its w), or
240    /// a value or normalisation is not finite.
241    pub fn realize(
242        &mut self,
243        request: EvlaAwCellRequest,
244    ) -> Result<NativeAwPair<'_>, NativeAwGenerationError> {
245        self.realized = None;
246        let key = EvlaAwCellRequest {
247            w_wavelengths: 0.0,
248            ..request
249        };
250        if self.prepared.is_none_or(|(held, _)| held != key) {
251            return Err(NativeAwGenerationError::WorkspaceMismatch);
252        }
253        let Self {
254            fft,
255            jones,
256            aperture,
257            imaging,
258            rings,
259            depth,
260            depth_increment,
261            phases,
262            ..
263        } = self;
264        // The screen's phase depends on (l², m²) alone, so one quadrant of
265        // |offsets| serves all four; the same expressions as per pixel.
266        let half = request.size / 2;
267        let stride = half + 1;
268        if *depth_increment != Some(request.sky_increment_rad) {
269            for ay in 0..=half {
270                let m = request.sky_increment_rad[1] * ay as f64;
271                for ax in 0..=half {
272                    let l = request.sky_increment_rad[0] * ax as f64;
273                    let rsq = l * l + m * m;
274                    depth[ax + stride * ay] = if rsq < 1.0 {
275                        (1.0 - rsq).sqrt() - 1.0
276                    } else {
277                        f64::NAN
278                    };
279                }
280            }
281            *depth_increment = Some(request.sky_increment_rad);
282        }
283        let screened = request.w_wavelengths > 0.0;
284        if screened {
285            for (phase, depth) in phases.iter_mut().zip(depth.iter()) {
286                if !depth.is_nan() {
287                    let (sin, cos) =
288                        (std::f64::consts::TAU * request.w_wavelengths * depth).sin_cos();
289                    *phase = Complex32::new(cos as f32, sin as f32);
290                }
291            }
292        }
293        let origin = half as isize;
294        let inner_half = (request.size / request.oversampling / 2) as isize;
295        for y in 0..request.size {
296            let iy = y as isize - origin;
297            for x in 0..request.size {
298                let ix = x as isize - origin;
299                let p = prolate(request, inner_half, ix, iy);
300                let pixel = x + request.size * y;
301                let mut screen = Complex32::new(p, 0.0);
302                let quadrant = ix.unsigned_abs() + stride * iy.unsigned_abs();
303                if screened && !depth[quadrant].is_nan() {
304                    screen *= phases[quadrant];
305                }
306                imaging[pixel] = aperture[pixel] * screen;
307            }
308        }
309        let edges = &mut jones[..2 * request.size];
310        save_edges(imaging, request.size, edges);
311        transform(fft, imaging)?;
312        restore_edges(imaging, request.size, edges);
313        let mut peak = imaging[0];
314        for value in &imaging[1..] {
315            if value.norm_sqr() > peak.norm_sqr() {
316                peak = *value;
317            }
318        }
319        if peak.norm() == 0.0 || !peak.norm().is_finite() {
320            return Err(NativeAwGenerationError::InvalidNumerics);
321        }
322        for value in imaging.iter_mut() {
323            *value /= peak;
324        }
325        let crop = crop_normalize(imaging, request, rings)?;
326        self.realized = Some(crop);
327        Ok(self.pair().expect("both crops succeeded"))
328    }
329}

331/// `AWConvFunc::resizeCF`'s support radius: the outermost ring with a
332/// sample above `1e-3` of the centre, read from [`ring_table`] instead of
333/// walking the rings.
334fn support_radius(values: &[Complex32], size: usize, rings: &[u16]) -> Option<usize> {
335    let origin = size / 2;
336    let threshold = (f64::from(values[origin * (size + 1)].norm()) * 1e-3) as f32;
337    let mut largest = 0_u16;
338    for (value, &ring) in values.iter().zip(rings) {
339        if ring > largest && value.norm() > threshold {
340            largest = ring;
341        }
342    }
343    (largest >= 2).then_some(usize::from(largest))
344}
345
346/// Per pixel of a `size × size` working grid, the largest ring
347/// [`crop_normalize`]'s support search samples it on, zero when none does:
348/// ring `r` samples the pixels at `origin + (r·sin θ, r·cos θ)` for
349/// `θ = 2π·k/r`, `k < 90·r` (`AWConvFunc::resizeCF`), so the search's radius
350/// is the largest entry over the pixels above its threshold.
351fn ring_table(size: usize) -> Vec<u16> {
352    let origin = size / 2;
353    let mut rings = vec![0_u16; size * size];
354    for r in 2..=origin - 2 {
355        for pixel in 0..90 * r {
356            let angle = std::f64::consts::TAU * pixel as f64 / r as f64;
357            let x = (origin as f64 + r as f64 * angle.sin()) as usize;
358            let y = (origin as f64 + r as f64 * angle.cos()) as usize;
359            rings[x + size * y] = u16::try_from(r).unwrap_or(u16::MAX);
360        }
361    }
362    rings
363}

365/// The prolate-spheroidal anti-aliasing factor at pixel offset `(ix, iy)`
366/// of the working grid, or one without it: `PSTerm::applySky` writes the
367/// inner `[−n/2, n/2)²` pixels with `ConvolveGridder::correctX1D`, whose
368/// factor is `grdsf(|offset| / (n/2))` on each axis
369/// (`ConvolveGridder::correctionFactor1D`), and leaves the rest zero.
370fn prolate(request: EvlaAwCellRequest, inner_half: isize, ix: isize, iy: isize) -> f32 {
371    if !request.prolate_spheroidal {
372        return 1.0;
373    }
374    if (-inner_half..inner_half).contains(&ix) && (-inner_half..inner_half).contains(&iy) {
375        let factor = |offset: isize| grdsf(offset.unsigned_abs() as f64 / inner_half as f64) as f32;
376        factor(ix) * factor(iy)
377    } else {
378        0.0
379    }
380}

382/// The sky Mueller element `request.mueller` at `frequency`: the four
383/// Jones aperture planes transformed, normalised by the centre power, the
384/// selected plane conjugated and squared
385/// (`VLACalcIlluminationConvFunc::applyPB`); unity without the aperture.
386fn fill_sky_mueller(
387    model: &EvlaApertureModel,
388    request: EvlaAwCellRequest,
389    frequency: f64,
390    fft: &mut PlaneFft<f32>,
391    jones: &mut [Complex32],
392    output: &mut [Complex32],
393) -> Result<(), NativeAwGenerationError> {
394    if !request.aperture {
395        output.fill(Complex32::new(1.0, 0.0));
396        return Ok(());
397    }
398    let wavelength = SPEED_OF_LIGHT_M_PER_S / f64::from(frequency as f32);
399    let cell_m = wavelength / (request.size as f64 * request.sky_increment_rad[0].abs());
400    model.fill_aperture(
401        EvlaApertureGrid::new(
402            request.size,
403            cell_m,
404            3,
405            frequency,
406            request.parallactic_angle_rad,
407        )?,
408        jones,
409    )?;
410    let n = request.size * request.size;
411    for plane in jones.chunks_exact_mut(n) {
412        transform(fft, plane)?;
413    }
414    let center = request.size / 2 * (request.size + 1);
415    let mut norm_squared = 0.0_f32;
416    for plane in jones.chunks_exact(n) {
417        norm_squared = (f64::from(norm_squared)
418            + f64::from((plane[center] * plane[center]).norm()) / 2.0)
419            as f32;
420    }
421    if !norm_squared.is_finite() || norm_squared <= 0.0 {
422        return Err(NativeAwGenerationError::InvalidNumerics);
423    }
424    let normalization = norm_squared.sqrt();
425    let selected = if request.mueller == 0 { 0 } else { 3 };
426    for (output, value) in output
427        .iter_mut()
428        .zip(&jones[selected * n..(selected + 1) * n])
429    {
430        let value = (*value / normalization).conj();
431        *output = value * value.conj();
432    }
433    Ok(())
434}

436/// A centred forward transform of a `[y][x]` plane in `f32`
437/// (`FFT2D::c2cFFT`).
438fn transform(
439    fft: &mut PlaneFft<f32>,
440    values: &mut [Complex32],
441) -> Result<(), NativeAwGenerationError> {
442    fft.transform(values, false)
443        .map_err(|_| NativeAwGenerationError::WorkspaceMismatch)
444}
445
446fn save_edges(values: &[Complex32], size: usize, edges: &mut [Complex32]) {
447    edges[..size].copy_from_slice(&values[(size - 1) * size..]);
448    for y in 0..size {
449        edges[size + y] = values[y * size + size - 1];
450    }
451}
452
453fn restore_edges(values: &mut [Complex32], size: usize, edges: &[Complex32]) {
454    values[(size - 1) * size..].copy_from_slice(&edges[..size]);
455    for y in 0..size {
456        values[y * size + size - 1] = edges[size + y];
457    }
458}

460/// `AWConvFunc::resizeCF` and `cfArea`: the support from the last ring
461/// above `1e-3` of the centre, the crop to the support buffer, and the
462/// normalisation by the sampled area over `[−support, support)`.
463fn crop_normalize(
464    values: &mut [Complex32],
465    request: EvlaAwCellRequest,
466    rings: &[u16],
467) -> Result<(usize, usize), NativeAwGenerationError> {
468    let origin = request.size / 2;
469    let radius = support_radius(values, request.size, rings)
470        .ok_or(NativeAwGenerationError::InvalidNumerics)?;
471    let mut support = (0.5_f32 + radius as f32 / request.oversampling as f32) as usize + 1;
472    if support * request.oversampling + (request.oversampling as f32 / 2.0 + 0.5) as usize > origin
473    {
474        support = origin / request.oversampling - 1;
475    }
476    if support == 0 {
477        return Err(NativeAwGenerationError::InvalidNumerics);
478    }
479    let extent = (request.oversampling * (support + 2)) as isize;
480    let bottom = (((origin as isize - extent) / 2) * 2).max(0) as usize;
481    let top = ((((origin as isize + extent) / 2) * 2 - 1) as usize).min(request.size - 1);
482    let size = top - bottom + 1;
483    let mut area = Complex32::default();
484    for ix in -(support as isize)..support as isize {
485        for iy in -(support as isize)..support as isize {
486            let x = (origin as isize + ix * request.oversampling as isize) as usize;
487            let y = (origin as isize + iy * request.oversampling as isize) as usize;
488            area += values[x + request.size * y];
489        }
490    }
491    if area.norm() == 0.0 || !area.norm().is_finite() {
492        return Err(NativeAwGenerationError::InvalidNumerics);
493    }
494    for y in 0..size {
495        for x in 0..size {
496            values[x + size * y] = values[x + bottom + request.size * (y + bottom)] / area;
497        }
498    }
499    if values[..size * size]
500        .iter()
501        .any(|v| !v.re.is_finite() || !v.im.is_finite())
502    {
503        return Err(NativeAwGenerationError::InvalidNumerics);
504    }
505    Ok((size, support))
506}

699    #[test]
700    fn the_quadrant_screen_is_the_per_pixel_screen() {
701        let model = EvlaApertureModel::new(surface());
702        // At 0.012 rad a pixel the grid's corners lie past l² + m² = 1,
703        // where the screen carries no W phase.
704        for (prolate_spheroidal, increment) in [(false, 0.0021), (true, 0.0021), (true, 0.012)] {
705            let request = EvlaAwCellRequest {
706                size: 128,
707                sky_increment_rad: [-increment, increment],
708                frequency_hz: 3e9,
709                conjugate_frequency_hz: 3.1e9,
710                w_wavelengths: 180.0,
711                parallactic_angle_rad: 0.31,
712                mueller: 0,
713                oversampling: 4,
714                prolate_spheroidal,
715                aperture: true,
716            };
717            let mut workspace = EvlaAwWorkspace::new(model.clone(), request).expect("workspace");
718            workspace.prepare(request).expect("prepare");
719            // The imaging plane before its transform, screened pixel by
720            // pixel as the generator once did.
721            let origin = (request.size / 2) as isize;
722            let inner_half = (request.size / request.oversampling / 2) as isize;
723            let mut expected = workspace.aperture.clone();
724            for y in 0..request.size {
725                let iy = y as isize - origin;
726                let m = request.sky_increment_rad[1] * iy as f64;
727                for x in 0..request.size {
728                    let ix = x as isize - origin;
729                    let l = request.sky_increment_rad[0] * ix as f64;
730                    let rsq = l * l + m * m;
731                    let mut screen = Complex32::new(prolate(request, inner_half, ix, iy), 0.0);
732                    if rsq < 1.0 {
733                        let phase = std::f64::consts::TAU
734                            * request.w_wavelengths
735                            * ((1.0 - rsq).sqrt() - 1.0);
736                        let (sin, cos) = phase.sin_cos();
737                        screen *= Complex32::new(cos as f32, sin as f32);
738                    }
739                    expected[x + request.size * y] *= screen;
740                }
741            }
742            let mut fft = PlaneFft::<f32>::new([request.size; 2], false).expect("plan");
743            let mut edges = vec![Complex32::default(); 2 * request.size];
744            save_edges(&expected, request.size, &mut edges);
745            transform(&mut fft, &mut expected).expect("transform");
746            restore_edges(&mut expected, request.size, &edges);
747            let mut peak = expected[0];
748            for value in &expected[1..] {
749                if value.norm_sqr() > peak.norm_sqr() {
750                    peak = *value;
751                }
752            }
753            for value in &mut expected {
754                *value /= peak;
755            }
756            let rings = ring_table(request.size);
757            let crop = crop_normalize(&mut expected, request, &rings).expect("crop");
758            let pair = workspace.realize(request).expect("realize");
759            assert_eq!(crop, (pair.imaging.size, pair.imaging.support));
760            assert_eq!(
761                &expected[..crop.0 * crop.0],
762                pair.imaging.values,
763                "prolate spheroidal {prolate_spheroidal}"
764            );
765        }
766    }

768    /// `AWConvFunc::resizeCF`'s ring walk, as `crop_normalize` once did it.
769    fn walked_radius(values: &[Complex32], size: usize) -> Option<usize> {
770        let origin = size / 2;
771        let threshold = (f64::from(values[origin * (size + 1)].norm()) * 1e-3) as f32;
772        for r in (2..=origin - 2).rev() {
773            for pixel in 0..90 * r {
774                let angle = std::f64::consts::TAU * pixel as f64 / r as f64;
775                let x = (origin as f64 + r as f64 * angle.sin()) as usize;
776                let y = (origin as f64 + r as f64 * angle.cos()) as usize;
777                if values[x + size * y].norm() > threshold {
778                    return Some(r);
779                }
780            }
781        }
782        None
783    }

785    #[test]
786    fn the_ring_table_finds_the_walks_support_radius() {
787        for size in [64, 128, 256] {
788            let rings = ring_table(size);
789            let origin = (size / 2) as f64;
790            for (width, spike) in [(2.0, None), (9.5, None), (31.0, Some(17)), (3.0, Some(5))] {
791                let mut values = (0..size * size)
792                    .map(|pixel| {
793                        let dx = (pixel % size) as f64 - origin;
794                        let dy = (pixel / size) as f64 - origin;
795                        let g = (-(dx * dx + dy * dy) / (2.0 * width * width)).exp() as f32;
796                        Complex32::new(g, 0.5 * g)
797                    })
798                    .collect::<Vec<_>>();
799                if let Some(offset) = spike {
800                    // An isolated sample beyond the core, off the axes.
801                    values[(size / 2 + offset) + size * (size / 2 + offset / 2)] =
802                        Complex32::new(0.01, 0.0);
803                }
804                assert_eq!(
805                    support_radius(&values, size, &rings),
806                    walked_radius(&values, size),
807                    "size {size}, width {width}, spike {spike:?}"
808                );
809            }
810        }
811    }
