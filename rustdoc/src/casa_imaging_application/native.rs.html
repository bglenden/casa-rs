39/// Why a native run did not complete.
40#[derive(Debug, thiserror::Error)]
41pub(crate) enum NativeError {
42    /// A phase's memory did not fit the resource policy.
43    #[error(transparent)]
44    Admission(#[from] Admission),
45    /// The run was cancelled.
46    #[error(transparent)]
47    Cancelled(#[from] Cancelled),
48    /// Any other failure.
49    #[error("{0}")]
50    Other(ApplicationError),
51}
52
53impl From<NativeError> for ApplicationDispatchError {
54    fn from(error: NativeError) -> Self {
55        match error {
56            NativeError::Admission(admission) => Self::Admission(admission),
57            NativeError::Cancelled(_) => Self::Cancelled,
58            NativeError::Other(error) => Self::Native(error),
59        }
60    }
61}
62
63impl From<ApplicationError> for NativeError {
64    fn from(error: ApplicationError) -> Self {
65        Self::Other(error)
66    }
67}

69impl From<ImagingError> for NativeError {
70    fn from(error: ImagingError) -> Self {
71        match error {
72            ImagingError::Admission(admission)
73            | ImagingError::Minor(casa_imaging_runtime::MinorCycleRunError::Admission(admission)) => {
74                Self::Admission(admission)
75            }
76            ImagingError::Cancelled(cancelled) => Self::Cancelled(cancelled),
77            ImagingError::Pass(error) => error.into(),
78            error => Self::Other(Box::new(error)),
79        }
80    }
81}

113/// A native run ready to start: its admitted source and what its passes,
114/// cycles and products need. Its setup chooses and forms its worker teams.
115pub(crate) struct NativeRun {
116    runtime: ApplicationRuntime,
117    publication: ApplicationPublication,
118    aw_catalog: Option<AwCatalogDeployment>,
119    visibility_write: Option<imaging::VisibilityWriteTarget>,
120    access: ResolvedSelectedObservationAccess,
121    /// The source's charge, held while it is open.
122    source: Reservation,
123    masks: ImageDomainReconstructionMaskPlans,
124    image_response: Option<MinorCycleImageResponse>,
125}
126
127/// The end of a run's preparation: check the products against `problem`,
128/// resolve any visibility write and admit the source, its pass storage
129/// sized for the policy's streaming workers ([`ResourcePolicy::cores`]), the
130/// most the run's setup then chooses.
131///
132/// [`ResourcePolicy::cores`]: casa_imaging_runtime::ResourcePolicy::cores
133pub(crate) fn prepare(
134    problem: &CompiledProblem,
135    input: NativeInput,
136    native: ApplicationNative,
137) -> Result<NativeRun, NativeError> {
138    let ApplicationNative {
139        runtime,
140        publication,
141        aw_catalog,
142    } = native;
143    publication.controls.validate_for_problem(problem)?;
144    let visibility_write = (input.write_model_column || input.write_corrected_data)
145        .then(|| {
146            Ok::<_, ApplicationError>(imaging::VisibilityWriteTarget {
147                path: PathBuf::from(input.observation.locator()),
148                selection: visibility_write_selection(problem, input.observation.selection())?,
149                model_data: input.write_model_column,
150                corrected_data: input.write_corrected_data,
151            })
152        })
153        .transpose()?;
154    let Admitted {
155        value: access,
156        reservation: source,
157    } = finalize_source_access(
158        problem,
159        input.initial_access,
160        imaging::row_samples(problem)?,
161        &runtime.host,
162        &runtime.resource_policy,
163    )?;
164    Ok(NativeRun {
165        runtime,
166        publication,
167        aw_catalog,
168        visibility_write,
169        access,
170        source,
171        masks: input.masks,
172        image_response: input.minor_cycle_image_response,
173    })
174}

176/// Run the major-cycle passes, the minor cycles between them and any
177/// visibility write, then publish the products. `summary` collects the
178/// phases and totals as they end; a completed run's outcome takes it once
179/// the caller has finished it.
180pub(crate) fn run(
181    problem: &CompiledProblem,
182    run: NativeRun,
183    summary: &mut RunSummary,
184) -> Result<impl FnOnce(RunSummary) -> ImagingOutcome + use<>, NativeError> {
185    let NativeRun {
186        runtime,
187        publication,
188        aw_catalog,
189        visibility_write,
190        access,
191        source,
192        masks,
193        image_response,
194    } = run;
195    let teams = OnceCell::new();
196    let outcome = imaging::run(
197        imaging::ImagingInputs {
198            problem,
199            access,
200            masks,
201            image_response,
202            visibility_write,
203            host: runtime.host,
204            policy: runtime.resource_policy,
205            teams: &teams,
206            cancel: runtime.cancel.clone(),
207            spill_directory: &runtime.spill_directory,
208            aw_catalog,
209            backend: runtime.backend,
210            grid_precision: runtime.grid_precision,
211            diagnostics: runtime.diagnostics,
212        },
213        summary,
214    )?;
215    drop(source);
216    let team = teams
217        .get()
218        .expect("a run that completed formed its teams")
219        .compute();
220    let mut published = Vec::new();
221    let mut lanes = (0, None);
222    let products = run_phase(
223        "products",
224        &runtime.cancel,
225        runtime.diagnostics,
226        summary,
227        || {
228            publish_products(
229                problem,
230                ProductSource {
231                    scientific: outcome.scientific,
232                    masks: outcome.masks,
233                    facet_sum_weights: outcome.facet_sum_weights,
234                },
235                &runtime,
236                publication,
237                team,
238                &mut published,
239                &mut lanes,
240            )
241        },
242    );
243    summary.products = published;
244    (summary.product_lanes, summary.product_lanes_reason) = lanes;
245    let products = products?;
246    let workers = team.workers();
247    let problem = problem.clone();
248    Ok(move |summary| ImagingOutcome {
249        problem,
250        minor_cycles: outcome.minor_cycles,
251        stop: outcome.stop,
252        major_cycle_count: outcome.major_cycle_count,
253        total_minor_iterations: outcome.total_minor_iterations,
254        total_actual_minor_iterations: outcome.total_actual_minor_iterations,
255        workers,
256        planes_per_wave: outcome.planes_per_wave,
257        visibility_products: outcome.visibility_products,
258        summary,
259        scientific: products.scientific,
260        planned_products: products.planned,
261        products: products.published,
262        _retained: outcome.retained,
263    })
264}

273/// What the products are formed from: the final completion, its
274/// reconstruction masks and the faceted domains' facets' own sums of
275/// weights.
276struct ProductSource {
277    scientific: MajorCycleCompletion,
278    masks: Option<ReconstructionMaskSet>,
279    facet_sum_weights: FacetSumWeights,
280}

282/// Generate every planned product member into the writer's private staging,
283/// one output channel per window on the run's team, and publish them,
284/// adding each image published to `published`; a cancelled run publishes
285/// nothing. `lanes` receives the lanes the products ran on and, when fewer
286/// than the team's workers, why.
287fn publish_products(
288    problem: &CompiledProblem,
289    source: ProductSource,
290    runtime: &ApplicationRuntime,
291    publication: ApplicationPublication,
292    team: &WorkerTeam,
293    published: &mut Vec<String>,
294    lanes: &mut (usize, Option<String>),
295) -> Result<Products, NativeError> {
296    let ProductSource {
297        scientific,
298        masks: reconstruction_masks,
299        facet_sum_weights,
300    } = source;
301    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific);
302    if let Some(masks) = reconstruction_masks.as_ref() {
303        inputs = match masks {
304            ReconstructionMaskSet::Shared(mask) => inputs.with_reconstruction_mask(mask),
305            ReconstructionMaskSet::Domains(masks) => inputs.with_domain_reconstruction_masks(masks),
306        };
307    }
308    if let Some(centres) = publication.primary_beam_centres.as_ref() {
309        inputs = inputs.with_primary_beam_centres(centres);
310    }
311    if !facet_sum_weights.is_empty() {
312        inputs = inputs.with_facet_sum_weights(&facet_sum_weights);
313    }
314    let planned = PlannedContinuumGeneration::new(&inputs, &publication.controls)?;
315    // The products' scratch grows with their lanes, and the gridding's
316    // workers were chosen without it: the products take the most lanes, up
317    // to the team's, whose demand fits what the policy leaves free (#763
318    // review), and one when none does, which admission then refuses.
319    let free = free_memory(&runtime.host, &runtime.resource_policy);
320    let demand_on = |count: usize| -> Result<_, NativeError> {
321        let window = ProductStoragePlan::new(1, count)?;
322        Ok((
323            window,
324            planned.demand(&inputs, window, &publication.writer)?,
325        ))
326    };
327    let workers = team.workers();
328    let (mut window, mut demand) = demand_on(workers)?;
329    let needed = demand.peak_residency_bytes();
330    if needed > free {
331        for count in (1..workers).rev() {
332            (window, demand) = demand_on(count)?;
333            if demand.peak_residency_bytes() <= free {
334                break;
335            }
336        }
337        let count = window.maximum_workers();
338        let gigabytes = |bytes: u64| format!("{:.2} GB", bytes as f64 / 1e9);
339        let reason = if demand.peak_residency_bytes() <= free {
340            format!(
341                "{workers} lanes need {}, {count} fit {} free",
342                gigabytes(needed),
343                gigabytes(free)
344            )
345        } else {
346            format!(
347                "{workers} lanes need {} and no fewer fit {} free",
348                gigabytes(needed),
349                gigabytes(free)
350            )
351        };
352        if count < workers {
353            tracing::warn!("products on {count} lanes, not the team's {workers}: {reason}");
354            lanes.1 = Some(reason);
355        }
356    }
357    lanes.0 = window.maximum_workers();
358    let held = admit(
359        &runtime.host,
360        &runtime.resource_policy,
361        &Demand {
362            phase: "products",
363            memory: demand.peak_residency_bytes(),
364        },
365    )?;
366    let generation =
367        produce_continuum_members(&planned, &inputs, window, team, &publication.writer)?;
368    if runtime.cancel.is_cancelled() {
369        // The staged members are removed when the writer drops.
370        return Err(Cancelled.into());
371    }
372    publication
373        .writer
374        .publish(|image: &Path| published.push(image.to_string_lossy().into_owned()))?;
375    // The charge covers the staged records until they are published.
376    drop(held);
377    Ok(Products {
378        planned,
379        scientific,
380        published: generation,
381    })
