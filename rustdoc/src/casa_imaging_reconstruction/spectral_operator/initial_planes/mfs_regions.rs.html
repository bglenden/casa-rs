2//! Bounded routing into disjoint strips of one initial MFS grid pair.
3
4use super::*;
5use ndarray::{ArrayViewMut2, Axis};

7pub(super) const STRIP_ROWS: usize = 64;
8const _: () = assert!(TAP_COUNT <= STRIP_ROWS);
9
10#[derive(Debug)]
11struct Record {
12    sample: InitialSample,
13    taps: Option<SampleTaps>,
14}
15
16#[derive(Debug)]
17struct Route {
18    record: usize,
19    next: usize,
20}
21
22#[derive(Debug)]
23pub(super) struct MfsRegions {
24    records: Vec<Record>,
25    routes: Vec<Route>,
26    buckets: Box<[Bucket]>,
27    capacity: usize,
28}
29
30impl MfsRegions {
31    pub(super) fn supports(
32        spec: &SpectralOperatorSpecification,
33        pass: SpectralOperatorPass,
34    ) -> bool {
35        spec.is_initial_certified_zero(pass)
36            && spec.supports_bulk_mfs()
37            && spec.polarization_coordinates.as_ref() == [PolarizationCoordinate::StokesI]
38    }
39
40    pub(super) fn workspace_bytes(
41        shape: [usize; 2],
42        capacity: usize,
43    ) -> Result<usize, SpectralOperatorError> {
44        if capacity == 0 || shape.contains(&0) {
45            return Err(SpectralOperatorError::ResidencyOverflow);
46        }
47        capacity
48            .checked_mul(size_of::<Record>() + 2 * size_of::<Route>())
49            .and_then(|bytes| {
50                shape[0]
51                    .div_ceil(STRIP_ROWS)
52                    .checked_mul(size_of::<Bucket>() + size_of::<InitialPlaneWork<'static>>())
53                    .and_then(|jobs| bytes.checked_add(jobs))
54            })
55            .and_then(|bytes| bytes.checked_add(size_of::<Self>()))
56            .ok_or(SpectralOperatorError::ResidencyOverflow)
57    }
58
59    pub(super) fn new(shape: [usize; 2], capacity: usize) -> Result<Self, SpectralOperatorError> {
60        Self::workspace_bytes(shape, capacity)?;
61        Ok(Self {
62            records: Vec::with_capacity(capacity),
63            routes: Vec::with_capacity(capacity * 2),
64            buckets: vec![Bucket::EMPTY; shape[0].div_ceil(STRIP_ROWS)].into_boxed_slice(),
65            capacity,
66        })
67    }
68
69    pub(super) fn ready(&self) -> bool {
70        self.records.len() == self.capacity
71    }
72
73    pub(super) fn push(
74        &mut self,
75        operator: &SpectralSlabOperator,
76        sample: SpectralOperatorSample,
77    ) -> Result<(), SpectralOperatorError> {
78        if self.ready() {
79            return Err(SpectralOperatorError::ResidencyOverflow);
80        }
81        let sample = InitialSample::new(sample);
82        let taps = if sample.active {
83            operator.gridder.taps(sample.uvw_lambda)
84        } else {
85            None
86        };
87        let record = self.records.len();
88        self.records.push(Record { sample, taps });
89        if let Some(taps) = taps {
90            operator.gridder.validate_taps(taps)?;
91            // A standard seven-row stencil intersects at most two strips.
92            for strip in taps.x.start / STRIP_ROWS..=(taps.x.start + TAP_COUNT - 1) / STRIP_ROWS {
93                let bucket = &mut self.buckets[strip];
94                let index = self.routes.len();
95                if bucket.last == END {
96                    bucket.first = index;
97                } else {
98                    self.routes[bucket.last].next = index;
99                }
100                bucket.last = index;
101                self.routes.push(Route { record, next: END });
102            }
103        }
104        Ok(())
105    }
106
107    pub(super) fn dispatch(
108        &mut self,
109        operator: &mut SpectralSlabOperator,
110        dispatch: &mut impl FnMut(&mut [InitialPlaneWork<'_>]) -> Result<(), SpectralOperatorError>,
111    ) -> Result<(), SpectralOperatorError> {
112        if self.records.is_empty() {
113            return Ok(());
114        }
115        let ConvolutionOperator::Standard(gridder) = &operator.gridder else {
116            return Err(SpectralOperatorError::UnsupportedProblem);
117        };
118        let dirty = operator
119            .dirty_grids
120            .as_mut()
121            .ok_or(SpectralOperatorError::ProblemMismatch)?;
122        let psf = operator
123            .psf_grids
124            .as_mut()
125            .ok_or(SpectralOperatorError::ProblemMismatch)?;
126        let ([dirty], [psf]) = (dirty.as_mut_slice(), psf.as_mut_slice()) else {
127            return Err(SpectralOperatorError::ProblemMismatch);
128        };
129        let mut jobs = Vec::with_capacity(self.buckets.len());
130        for (strip, (dirty, psf)) in dirty
131            .axis_chunks_iter_mut(Axis(0), STRIP_ROWS)
132            .zip(psf.axis_chunks_iter_mut(Axis(0), STRIP_ROWS))
133            .enumerate()
134        {
135            let first = self.buckets[strip].first;
136            if first != END {
137                jobs.push(InitialPlaneWork(InitialWork::Mfs(MfsWork {
138                    gridder,
139                    dirty,
140                    psf,
141                    first_row: strip * STRIP_ROWS,
142                    records: &self.records,
143                    routes: &self.routes,
144                    first,
145                    executed: false,
146                    completed: false,
147                })));
148            }
149        }
150        dispatch(&mut jobs)?;
151        if jobs
152            .iter()
153            .any(|job| !matches!(&job.0, InitialWork::Mfs(work) if work.completed))
154        {
155            return Err(SpectralOperatorError::BlockSequence);
156        }
157        drop(jobs);
158        for record in &self.records {
159            operator.mapped_samples[0] = operator.mapped_samples[0]
160                .checked_add(1)
161                .ok_or(SpectralOperatorError::CoverageOverflow)?;
162            if record.taps.is_some() {
163                for (value, sum, compensation) in [
164                    (
165                        record.sample.normal_weight,
166                        &mut operator.sum_weights[0],
167                        &mut operator.sum_weight_compensations[0],
168                    ),
169                    (
170                        record.sample.published_weight,
171                        &mut operator.published_sum_weights[0],
172                        &mut operator.published_sum_weight_compensations[0],
173                    ),
174                ] {
175                    let corrected = value - *compensation;
176                    let updated = *sum + corrected;
177                    *compensation = (updated - *sum) - corrected;
178                    *sum = updated;
179                }
180                #[cfg(test)]
181                {
182                    record_measurement(
183                        &mut operator.measurements.dirty_grid_tap_visits,
184                        TAP_VISITS_PER_SAMPLE,
185                    );
186                    record_measurement(
187                        &mut operator.measurements.psf_grid_tap_visits,
188                        TAP_VISITS_PER_SAMPLE,
189                    );
190                }
191            }
192        }
193        self.records.clear();
194        self.routes.clear();
195        self.buckets.fill(Bucket::EMPTY);
196        Ok(())
197    }
198}

200pub(super) struct MfsWork<'a> {
201    gridder: &'a StandardConvolution,
202    dirty: ArrayViewMut2<'a, Complex64>,
203    psf: ArrayViewMut2<'a, Complex64>,
204    first_row: usize,
205    records: &'a [Record],
206    routes: &'a [Route],
207    first: usize,
208    executed: bool,
209    completed: bool,
210}
211
212impl MfsWork<'_> {
213    pub(super) fn execute(&mut self) -> Result<(), SpectralOperatorError> {
214        if self.executed {
215            return Err(SpectralOperatorError::BlockSequence);
216        }
217        self.executed = true;
218        let mut next = self.first;
219        while next != END {
220            let route = &self.routes[next];
221            let record = &self.records[route.record];
222            let taps = record.taps.ok_or(SpectralOperatorError::InvalidSample)?;
223            self.gridder.grid_rows(
224                &mut self.dirty,
225                taps,
226                record.sample.visibility,
227                self.first_row,
228            );
229            self.gridder.grid_rows(
230                &mut self.psf,
231                taps,
232                Complex64::new(record.sample.normal_weight, 0.0),
233                self.first_row,
234            );
235            next = route.next;
236        }
237        self.completed = true;
238        Ok(())
239    }
240}

258    #[test]
259    fn mfs_regions_match_scalar_across_boundaries_and_worker_counts() {
260        for workers in [1, 4, 8] {
261            let mut scalar = operator();
262            let mut candidate = operator();
263            let mut batch = MfsRegions::new([512, 64], 19).unwrap();
264            let records = batch.records.as_ptr();
265            let routes = batch.routes.as_ptr();
266            let mut largest_dispatch = 0;
267            for first in (0..152).step_by(19) {
268                for index in first..first + 19 {
269                    // Cross strip boundaries, cover every strip, include flags
270                    // and samples outside the supported grid.
271                    let row = if index % 17 == 0 {
272                        -20.0
273                    } else {
274                        (index % 8 * 64) as f64 + 0.25
275                    };
276                    let uv = [
277                        (row - 256.0) * scalar.gridder.standard().du_lambda,
278                        0.25 * scalar.gridder.standard().dv_lambda,
279                        0.0,
280                    ];
281                    let sample = SpectralOperatorSample::new(
282                        0,
283                        uv,
284                        299_792_458.0,
285                        0.03,
286                        [1.0 + index as f64 * 0.01, -0.2],
287                        if index % 13 == 0 { 0.0 } else { 0.7 },
288                        1.0,
289                    )
290                    .unwrap()
291                    .with_published_weight(0.4)
292                    .unwrap();
293                    scalar.push_polarization(sample, 0).unwrap();
294                    batch.push(&candidate, sample).unwrap();
295                }
296                assert!(batch.routes.len() <= 2 * batch.capacity);
297                batch
298                    .dispatch(&mut candidate, &mut |jobs| {
299                        largest_dispatch = largest_dispatch.max(jobs.len());
300                        let chunk = jobs.len().div_ceil(workers).max(1);
301                        std::thread::scope(|scope| {
302                            let handles = jobs
303                                .chunks_mut(chunk)
304                                .map(|jobs| {
305                                    scope.spawn(move || {
306                                        jobs.iter_mut().try_for_each(InitialPlaneWork::execute)
307                                    })
308                                })
309                                .collect::<Vec<_>>();
310                            for handle in handles {
311                                handle.join().unwrap()?;
312                            }
313                            Ok(())
314                        })
315                    })
316                    .unwrap();
317                assert_eq!(batch.records.as_ptr(), records);
318                assert_eq!(batch.routes.as_ptr(), routes);
319            }
320            assert_eq!(largest_dispatch, 8);
321            for (left, right) in scalar.dirty_grids.as_ref().unwrap()[0]
322                .iter()
323                .zip(candidate.dirty_grids.as_ref().unwrap()[0].iter())
324                .chain(
325                    scalar.psf_grids.as_ref().unwrap()[0]
326                        .iter()
327                        .zip(candidate.psf_grids.as_ref().unwrap()[0].iter()),
328                )
329            {
330                assert!((*left - *right).norm() < 1e-12);
331            }
332            assert_eq!(scalar.mapped_samples, candidate.mapped_samples);
333            assert!((scalar.sum_weights[0] - candidate.sum_weights[0]).abs() < 1e-12);
334            assert!(
335                (scalar.published_sum_weights[0] - candidate.published_sum_weights[0]).abs()
336                    < 1e-12
337            );
338        }
339    }
