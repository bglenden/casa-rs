5pub(super) struct CompileCtx<T: ImageExprValue> {
6    pub(super) sources: Vec<CompiledImageSource<T>>,
7    pub(super) paged_source_ids: std::collections::HashMap<PathBuf, usize>,
8    pub(super) transient_source_ids: std::collections::HashMap<usize, usize>,
9}
10
11impl<T: ImageExprValue> Default for CompileCtx<T> {
12    fn default() -> Self {
13        Self {
14            sources: Vec::new(),
15            paged_source_ids: std::collections::HashMap::new(),
16            transient_source_ids: std::collections::HashMap::new(),
17        }
18    }
19}

21impl<T: ImageExprValue> CompileCtx<T> {
22    fn register_snapshot_source(
23        &mut self,
24        data: ArrayD<T>,
25        mask_snapshot: Option<Arc<ArrayD<bool>>>,
26    ) -> usize {
27        self.register_shared_snapshot_source(Arc::new(data), mask_snapshot)
28    }
29
30    fn register_shared_snapshot_source(
31        &mut self,
32        data: Arc<ArrayD<T>>,
33        mask_snapshot: Option<Arc<ArrayD<bool>>>,
34    ) -> usize {
35        let source_id = self.sources.len();
36        let shape = data.shape().to_vec();
37        let has_mask = mask_snapshot.is_some();
38        self.sources.push(CompiledImageSource {
39            shape,
40            has_mask,
41            mask_snapshot,
42            tile_shape: None,
43            kind: CompiledImageSourceKind::Snapshot(data),
44            reference_count: 1,
45        });
46        source_id
47    }
48
49    fn register_image_source(
50        &mut self,
51        image: &dyn ImageInterface<T>,
52    ) -> Result<usize, ImageError> {
53        let source_key = image as *const dyn ImageInterface<T> as *const () as usize;
54        if let Some(any) = image.as_any() {
55            if let Some(expression) = any.downcast_ref::<ImageExpression<T>>()
56                && let CompiledNumericExprNode::Source(source_id) = expression.node
57                && let CompiledImageSourceKind::Sections(sections) =
58                    &expression.arena.sources[source_id].kind
59            {
60                if let Some(&source_id) = self.transient_source_ids.get(&source_key) {
61                    self.sources[source_id].reference_count += 1;
62                    return Ok(source_id);
63                }
64                let source_id = self.register_section_source(sections.clone());
65                self.transient_source_ids.insert(source_key, source_id);
66                return Ok(source_id);
67            }
68            if let Some(paged) = any.downcast_ref::<PagedImage<T>>()
69                && let Some(path) = paged.name()
70            {
71                paged.flush_pixels_for_reopen()?;
72                #[cfg(unix)]
73                let lock_reads = paged.has_lock(casa_tables::LockType::Read);
74                #[cfg(not(unix))]
75                let lock_reads = false;
76                if let Some(&source_id) = self.paged_source_ids.get(path) {
77                    let source = &mut self.sources[source_id];
78                    source.reference_count += 1;
79                    if let CompiledImageSourceKind::Paged {
80                        lock_reads: existing,
81                        ..
82                    } = &mut source.kind
83                    {
84                        *existing |= lock_reads;
85                    }
86                    return Ok(source_id);
87                }
88                let cursor_shape = clamp_cursor_shape(&paged.nice_cursor_shape(), paged.shape());
89                let source_id = self.sources.len();
90                let has_mask = paged.has_pixel_mask();
91                self.sources.push(CompiledImageSource {
92                    shape: paged.shape().to_vec(),
93                    has_mask,
94                    mask_snapshot: None,
95                    kind: CompiledImageSourceKind::Paged {
96                        path: Arc::new(path.to_path_buf()),
97                        cache_bytes: compiled_source_cache_bytes::<T>(
98                            &cursor_shape,
99                            paged.cache_bytes(),
100                        )?,
101                        lock_reads,
102                    },
103                    tile_shape: Some(paged.tile_shape().to_vec()),
104                    reference_count: 1,
105                });
106                self.paged_source_ids.insert(path.to_path_buf(), source_id);
107                return Ok(source_id);
108            }
109
110            if let Some(mem) = any.downcast_ref::<MemorySnapshotImage<T>>() {
111                if let Some(&source_id) = self.transient_source_ids.get(&source_key) {
112                    self.sources[source_id].reference_count += 1;
113                    return Ok(source_id);
114                }
115                let source_id =
116                    self.register_shared_snapshot_source(mem.data.clone(), mem.mask.clone());
117                self.transient_source_ids.insert(source_key, source_id);
118                return Ok(source_id);
119            }
120
121            if let Some(temp) = any.downcast_ref::<TempImage<T>>() {
122                if let Some(&source_id) = self.transient_source_ids.get(&source_key) {
123                    self.sources[source_id].reference_count += 1;
124                    return Ok(source_id);
125                }
126                let data = temp.get()?;
127                let mask = temp.default_mask()?.map(Arc::new);
128                let source_id = self.register_snapshot_source(data, mask);
129                self.transient_source_ids.insert(source_key, source_id);
130                return Ok(source_id);
131            }
132        }
133
134        if let Some(&source_id) = self.transient_source_ids.get(&source_key) {
135            self.sources[source_id].reference_count += 1;
136            return Ok(source_id);
137        }
138        let data = image.get()?;
139        let mask = image.default_mask()?.map(Arc::new);
140        let source_id = self.register_snapshot_source(data, mask);
141        self.transient_source_ids.insert(source_key, source_id);
142        Ok(source_id)
143    }

145    fn register_section_source(&mut self, source: SectionSource<T>) -> usize {
146        let source_id = self.sources.len();
147        self.sources.push(CompiledImageSource {
148            shape: source.shape.clone(),
149            has_mask: source.mask.is_some(),
150            mask_snapshot: None,
151            tile_shape: None,
152            kind: CompiledImageSourceKind::Sections(source),
153            reference_count: 1,
154        });
155        source_id
156    }

158    fn compile_numeric(
159        &mut self,
160        node: &NumericExprNode<'_, T>,
161        node_shape: &[usize],
162    ) -> Result<CompiledNumericExprNode<T>, ImageError> {
163        Ok(match node {
164            NumericExprNode::Sections(source) => {
165                CompiledNumericExprNode::Source(self.register_section_source(source.clone()))
166            }
167            NumericExprNode::Vector(values) => CompiledNumericExprNode::Source(
168                self.register_shared_snapshot_source(values.clone(), None),
169            ),
170            NumericExprNode::Source(image) => {
171                CompiledNumericExprNode::Source(self.register_image_source(*image)?)
172            }
173            NumericExprNode::Scalar(value) => CompiledNumericExprNode::Scalar(*value),
174            NumericExprNode::UnaryOp { op, child } => CompiledNumericExprNode::UnaryOp {
175                op: *op,
176                child: Box::new(self.compile_numeric(
177                    child,
178                    &child.try_shape().unwrap_or_else(|| node_shape.to_vec()),
179                )?),
180            },
181            NumericExprNode::BinaryOp { op, lhs, rhs } => {
182                let lhs_shape = lhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
183                let rhs_shape = rhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
184                CompiledNumericExprNode::BinaryOp {
185                    op: *op,
186                    lhs: Box::new(self.compile_numeric(lhs, &lhs_shape)?),
187                    lhs_shape,
188                    rhs: Box::new(self.compile_numeric(rhs, &rhs_shape)?),
189                    rhs_shape,
190                }
191            }
192            NumericExprNode::CustomUnary { child, func, .. } => {
193                CompiledNumericExprNode::CustomUnary {
194                    child: Box::new(self.compile_numeric(
195                        child,
196                        &child.try_shape().unwrap_or_else(|| node_shape.to_vec()),
197                    )?),
198                    func: BinaryOrUnary::Unary(func.clone()),
199                }
200            }
201            NumericExprNode::CustomBinary { lhs, rhs, func, .. } => {
202                let lhs_shape = lhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
203                let rhs_shape = rhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
204                CompiledNumericExprNode::CustomBinary {
205                    lhs: Box::new(self.compile_numeric(lhs, &lhs_shape)?),
206                    lhs_shape,
207                    rhs: Box::new(self.compile_numeric(rhs, &rhs_shape)?),
208                    rhs_shape,
209                    func: BinaryOrUnary::Binary(func.clone()),
210                }
211            }
212            NumericExprNode::Reduction {
213                op,
214                child,
215                child_shape,
216                empty_value,
217            } => CompiledNumericExprNode::Reduction {
218                op: *op,
219                child: Box::new(self.compile_numeric(child, child_shape)?),
220                child_shape: child_shape.clone(),
221                empty_value: *empty_value,
222            },
223            NumericExprNode::Fractile {
224                child,
225                child_shape,
226                fraction,
227            } => CompiledNumericExprNode::Fractile {
228                child: Box::new(self.compile_numeric(child, child_shape)?),
229                child_shape: child_shape.clone(),
230                fraction: *fraction,
231            },
232            NumericExprNode::FractileRange {
233                child,
234                child_shape,
235                fraction1,
236                fraction2,
237            } => CompiledNumericExprNode::FractileRange {
238                child: Box::new(self.compile_numeric(child, child_shape)?),
239                child_shape: child_shape.clone(),
240                fraction1: *fraction1,
241                fraction2: *fraction2,
242            },
243            NumericExprNode::Conditional {
244                condition,
245                if_true,
246                if_false,
247            } => {
248                let cond_shape = condition.try_shape().unwrap_or_else(|| node_shape.to_vec());
249                let true_shape = if_true.try_shape().unwrap_or_else(|| node_shape.to_vec());
250                let false_shape = if_false.try_shape().unwrap_or_else(|| node_shape.to_vec());
251                CompiledNumericExprNode::Conditional {
252                    condition: Box::new(self.compile_mask(condition, &cond_shape)?),
253                    cond_shape,
254                    if_true: Box::new(self.compile_numeric(if_true, &true_shape)?),
255                    true_shape,
256                    if_false: Box::new(self.compile_numeric(if_false, &false_shape)?),
257                    false_shape,
258                }
259            }
260            NumericExprNode::MaskCount {
261                count_true,
262                mask,
263                mask_shape,
264            } => CompiledNumericExprNode::MaskCount {
265                count_true: *count_true,
266                mask: Box::new(self.compile_mask(mask, mask_shape)?),
267                mask_shape: mask_shape.clone(),
268            },
269            NumericExprNode::Replace {
270                primary,
271                replacement,
272                mask,
273            } => {
274                let primary_shape = primary.try_shape().unwrap_or_else(|| node_shape.to_vec());
275                let replacement_shape = replacement
276                    .try_shape()
277                    .unwrap_or_else(|| node_shape.to_vec());
278                CompiledNumericExprNode::Replace {
279                    primary: Box::new(self.compile_numeric(primary, &primary_shape)?),
280                    primary_shape,
281                    replacement: Box::new(self.compile_numeric(replacement, &replacement_shape)?),
282                    replacement_shape,
283                    mask: mask.clone().map(Arc::new),
284                }
285            }
286            NumericExprNode::TypeBridge { eval_fn } => {
287                let start = vec![0; node_shape.len()];
288                let stride = vec![1; node_shape.len()];
289                let data = eval_fn(&start, node_shape, &stride)?;
290                let source_id = self.register_snapshot_source(data, None);
291                CompiledNumericExprNode::Source(source_id)
292            }
293            NumericExprNode::StripMask { child } => CompiledNumericExprNode::StripMask {
294                child: Box::new(self.compile_numeric(
295                    child,
296                    &child.try_shape().unwrap_or_else(|| node_shape.to_vec()),
297                )?),
298            },
299            NumericExprNode::SpectralIndex {
300                lhs,
301                rhs,
302                spec_axis_a,
303                freqs_a,
304                spec_axis_b,
305                freqs_b,
306            } => {
307                let lhs_shape = lhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
308                let rhs_shape = rhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
309                CompiledNumericExprNode::SpectralIndex {
310                    lhs: Box::new(self.compile_numeric(lhs, &lhs_shape)?),
311                    lhs_shape,
312                    rhs: Box::new(self.compile_numeric(rhs, &rhs_shape)?),
313                    rhs_shape,
314                    spec_axis_a: *spec_axis_a,
315                    freqs_a: freqs_a.clone(),
316                    spec_axis_b: *spec_axis_b,
317                    freqs_b: freqs_b.clone(),
318                }
319            }
320            NumericExprNode::Pa { u, q } => {
321                let u_shape = u.try_shape().unwrap_or_else(|| node_shape.to_vec());
322                let q_shape = q.try_shape().unwrap_or_else(|| node_shape.to_vec());
323                CompiledNumericExprNode::Pa {
324                    u: Box::new(self.compile_numeric(u, &u_shape)?),
325                    u_shape,
326                    q: Box::new(self.compile_numeric(q, &q_shape)?),
327                    q_shape,
328                }
329            }
330            NumericExprNode::Rebin {
331                child,
332                in_shape,
333                factors,
334                ..
335            } => CompiledNumericExprNode::Rebin {
336                child: Box::new(self.compile_numeric(child, in_shape)?),
337                in_shape: in_shape.clone(),
338                factors: factors.clone(),
339            },
340        })
341    }

343    fn compile_mask(
344        &mut self,
345        node: &MaskExprNode<'_, T>,
346        node_shape: &[usize],
347    ) -> Result<CompiledMaskExprNode<T>, ImageError> {
348        Ok(match node {
349            MaskExprNode::CompareScalar { op, expr, scalar } => {
350                CompiledMaskExprNode::CompareScalar {
351                    op: *op,
352                    expr: Box::new(self.compile_numeric(
353                        expr,
354                        &expr.try_shape().unwrap_or_else(|| node_shape.to_vec()),
355                    )?),
356                    scalar: *scalar,
357                }
358            }
359            MaskExprNode::CompareExpr { op, lhs, rhs } => {
360                let lhs_shape = lhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
361                let rhs_shape = rhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
362                CompiledMaskExprNode::CompareExpr {
363                    op: *op,
364                    lhs: Box::new(self.compile_numeric(lhs, &lhs_shape)?),
365                    lhs_shape,
366                    rhs: Box::new(self.compile_numeric(rhs, &rhs_shape)?),
367                    rhs_shape,
368                }
369            }
370            MaskExprNode::Logical { op, lhs, rhs } => {
371                let lhs_shape = lhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
372                let rhs_shape = rhs.try_shape().unwrap_or_else(|| node_shape.to_vec());
373                CompiledMaskExprNode::Logical {
374                    op: *op,
375                    lhs: Box::new(self.compile_mask(lhs, &lhs_shape)?),
376                    lhs_shape,
377                    rhs: Box::new(self.compile_mask(rhs, &rhs_shape)?),
378                    rhs_shape,
379                }
380            }
381            MaskExprNode::Not { child } => CompiledMaskExprNode::Not {
382                child: Box::new(self.compile_mask(
383                    child,
384                    &child.try_shape().unwrap_or_else(|| node_shape.to_vec()),
385                )?),
386            },
387            MaskExprNode::IsNan { child } => CompiledMaskExprNode::IsNan {
388                child: Box::new(self.compile_numeric(
389                    child,
390                    &child.try_shape().unwrap_or_else(|| node_shape.to_vec()),
391                )?),
392            },
393            MaskExprNode::Validity { child } => CompiledMaskExprNode::Validity {
394                child: Box::new(self.compile_numeric(child, node_shape)?),
395            },
396            MaskExprNode::BooleanValidity { child } => CompiledMaskExprNode::BooleanValidity {
397                child: Box::new(self.compile_mask(child, node_shape)?),
398            },
399            MaskExprNode::ConstantMask { mask } => CompiledMaskExprNode::ConstantMask {
400                mask: Arc::new(mask.clone()),
401            },
402            MaskExprNode::AllReduce { child, child_shape } => CompiledMaskExprNode::AllReduce {
403                child: Box::new(self.compile_mask(child, child_shape)?),
404                child_shape: child_shape.clone(),
405            },
406            MaskExprNode::AnyReduce { child, child_shape } => CompiledMaskExprNode::AnyReduce {
407                child: Box::new(self.compile_mask(child, child_shape)?),
408                child_shape: child_shape.clone(),
409            },
410            MaskExprNode::IndexIn { axis, indices } => CompiledMaskExprNode::IndexIn {
411                axis: *axis,
412                indices: indices.clone(),
413            },
414        })
415    }
416}

418impl<'a, T: ImageExprValue> ImageExprBuilder<'a, T> {
419    /// Compiles a borrowed lazy expression into an owned execution form.
420    ///
421    /// Compilation is Rust-specific functionality beyond C++ casacore's
422    /// borrowed `ImageExprBuilder<T>` surface. Persistent paged sources remain lazy
423    /// and reopenable; non-persistent or non-downcastable sources are
424    /// snapshotted so the compiled expression can be evaluated on worker
425    /// threads safely.
426    ///
427    /// # Example
428    ///
429    /// ```rust
430    /// use casa_coordinates::CoordinateSystem;
431    /// use casa_images::{ImageExprBuilder, TempImage};
432    /// use casa_lattices::{ExecutionPolicy, LatticeMut};
433    ///
434    /// let mut image = TempImage::<f32>::new(vec![16, 16], CoordinateSystem::new(), casa_lattices::TempStoragePolicy::Memory).unwrap();
435    /// image.set(2.0).unwrap();
436    ///
437    /// let mut compiled = ImageExprBuilder::from_image(&image)
438    ///     .unwrap()
439    ///     .multiply_scalar(2.0)
440    ///     .compile()
441    ///     .unwrap();
442    /// compiled.set_execution_policy(ExecutionPolicy::Parallel {
443    ///     workers: 2,
444    ///     prefetch_depth: 4,
445    /// });
446    /// assert_eq!(compiled.get_at(&[0, 0]).unwrap(), 4.0);
447    /// ```
448    pub fn compile(&self) -> Result<ImageExpression<T>, ImageError> {
449        let mut ctx = CompileCtx::default();
450        let node = ctx.compile_numeric(&self.node, &self.meta.shape)?;
451        Ok(ImageExpression {
452            node,
453            arena: Arc::new(CompiledImageArena {
454                sources: ctx.sources,
455            }),
456            meta: self.meta.clone(),
457            execution_policy: ExecutionPolicy::Auto,
458            execution_resources: ExecutionResources::default(),
459            expr_string: self.expr_string.as_deref().map(Arc::from),
460        })
461    }
462}

464impl<'a, T: ImageExprValue> MaskExprBuilder<'a, T> {
465    /// Compiles a borrowed lazy boolean mask into an owned execution form.
466    pub fn compile(&self) -> Result<MaskExpression<T>, ImageError> {
467        let mut ctx = CompileCtx::default();
468        let node = ctx.compile_mask(&self.node, &self.shape)?;
469        Ok(MaskExpression {
470            node,
471            arena: Arc::new(CompiledImageArena {
472                sources: ctx.sources,
473            }),
474            shape: self.shape.clone(),
475            execution_policy: ExecutionPolicy::Auto,
476            execution_resources: ExecutionResources::default(),
477        })
478    }
