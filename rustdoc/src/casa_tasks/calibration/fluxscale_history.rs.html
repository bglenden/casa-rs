2//! `Calibrater::fluxscale` task `HISTORY` provenance adapter.
3
4use crate::TaskInvocation;
5use casa_calibration::fluxscale::history::append_messages;
6use casa_calibration::fluxscale::{FluxScaleReport, FluxScaleRequest};
7use casa_contracts::{ParameterValue, python_repr};
8
9/// History failure is a warning after publication, as in the CASA task.
10pub(crate) fn record(
11    task: &TaskInvocation<'_>,
12    request: &FluxScaleRequest,
13    report: &FluxScaleReport,
14) {
15    let _phase = casa_calibration::timing::Phase::start("fluxscale_history");
16    if let Err(error) = append_messages(request, messages(task, request, report), 2) {
17        tracing::warn!("updating fluxscale HISTORY: {error}");
18    }
19}
20
21fn messages(
22    task: &TaskInvocation<'_>,
23    request: &FluxScaleRequest,
24    report: &FluxScaleReport,
25) -> Vec<String> {
26    let mut messages = casa_calibration::fluxscale::history::tool_messages(request);
27    messages.extend([
28        "taskname=fluxscale".into(),
29        format!("version: casa-rs {}", env!("CARGO_PKG_VERSION")),
30    ]);
31    let mut bindings = task
32        .session()
33        .bundle()
34        .surface
35        .bindings()
36        .iter()
37        .filter(|binding| binding.order < 15)
38        .collect::<Vec<_>>();
39    bindings.sort_by_key(|binding| binding.order);
40    for binding in bindings {
41        // Record effective normalized values, as the task provider receives them.
42        let value = task
43            .session()
44            .states()
45            .get(&binding.name)
46            .and_then(|state| state.value.as_ref());
47        if let Some(value) = value {
48            let text = if let ParameterValue::String(value) = value {
49                format!("\"{value}\"")
50            } else {
51                python_repr(value)
52            };
53            messages.push(format!("{:<11} = {text}", binding.name));
54        }
55    }
56    messages.push("Fluxscale results *****".into());
57    for (id, field) in &report.fields {
58        messages.push(format!("{}(field id={id}) ", field.field_name));
59        let (mut last_base, mut last_unit) = (1., "Hz");
60        for (index, spw) in report.spw_ids.iter().enumerate() {
61            let frequency = report.frequencies_hz[index];
62            let (base, unit) = if frequency > 1e9 {
63                (1e9, "GHz")
64            } else if frequency > 1e6 {
65                (1e6, "MHz")
66            } else if frequency > 1e3 {
67                (1e3, "kHz")
68            } else {
69                (1., "Hz")
70            };
71            if frequency > 0. {
72                last_base = base;
73                last_unit = unit;
74            }
75            let result = &field.spw_results[spw];
76            messages.push(if frequency < 0. {
77                format!("  Spw {spw} insufficient data, flux density is not determined.")
78            } else {
79                format!(
80                    "  Spw {spw}(freq = {:7.3}{unit}) Flux density = {:10.6}+/-{:10.6} Jy",
81                    frequency / base,
82                    result.fluxd[0],
83                    result.fluxd_err[0]
84                )
85            });
86        }
87        if report.spw_ids.len() > 1 {
88            let coefficients = |values: &[f64]| {
89                format!(
90                    "[{}]",
91                    values
92                        .iter()
93                        .map(|value| format!("{value:5.3}"))
94                        .collect::<Vec<_>>()
95                        .join(",")
96                )
97            };
98            messages.push(format!(
99                "  Spectral index fitting coefficients [zero-point, alpha, beta] with errors :{}+/-{}",
100                coefficients(&field.spidx),
101                coefficients(&field.spidx_err)
102            ));
103            messages.push(format!(
104                "  Fitted flux density = {:10.6}+/-{:10.6} Jy (reference freq = {:7.3}{last_unit})",
105                field.fit_fluxd,
106                field.fit_fluxd_err,
107                field.fit_ref_frequency_hz / last_base
108            ));
109        }
110    }
111    messages
112}
