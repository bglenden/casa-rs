69enum EntrySelection {
70    Rows(Vec<[i32; 4]>),
71    FieldAndWindow { fields: Vec<i32>, windows: Vec<i32> },
72}
73
74impl EntrySelection {
75    fn matches(&self, combination: [i32; 4]) -> bool {
76        match self {
77            Self::Rows(rows) => rows.contains(&combination),
78            Self::FieldAndWindow { fields, windows } => {
79                fields.contains(&combination[0]) && windows.contains(&combination[1])
80            }
81        }
82    }
83}

85enum Prediction {
86    Components(ComponentList),
87    FourierGrid(Box<FourierModel>),
88    SingleDish(Box<SingleDishModel>),
89}

109    fn load(
110        ms: &MeasurementSet,
111        rows: &[SelectedSolveRow],
112        source: GainSolveModelSource,
113        defer_refusals: bool,
114    ) -> Result<Self, GainSolveError> {
115        let read_column = source == GainSolveModelSource::ModelColumn
116            || (source == GainSolveModelSource::FromMeasurementSet
117                && ms
118                    .main_table()
119                    .schema()
120                    .is_some_and(|schema| schema.contains_column("MODEL_DATA")));
121        let mut inputs = Self::empty(read_column);
122        if source != GainSolveModelSource::FromMeasurementSet {
123            return Ok(inputs);
124        }
125        let fields: BTreeSet<_> = rows.iter().map(|row| row.field_id).collect();
126        let mut records = BTreeMap::<_, Arc<[Entry]>>::new();
127        for field in fields {
128            if let Some(record) =
129                virtual_model(ms, field).map_err(|source| read_error(ms, source))?
130            {
131                let entries = match records.entry(record.key) {
132                    std::collections::btree_map::Entry::Occupied(entry) => entry.into_mut(),
133                    std::collections::btree_map::Entry::Vacant(entry) => {
134                        let decoded = match decode_entries(&record.record) {
135                            Ok(decoded) => decoded,
136                            Err(GainSolveError::VirtualModel(error)) if defer_refusals => {
137                                inputs.virtual_refusals.insert(field, error);
138                                continue;
139                            }
140                            Err(error) => return Err(error),
141                        };
142                        entry.insert(decoded.into())
143                    }
144                };
145                inputs.virtual_models.insert(field, Arc::clone(entries));
146            }
147        }
148        inputs.load_geometry(ms, rows)?;
149        Ok(inputs)
150    }

152    pub(super) fn from_entries(
153        ms: &MeasurementSet,
154        rows: &[SelectedSolveRow],
155        entries: &[VirtualModelEntry],
156    ) -> Result<Self, GainSolveError> {
157        let mut inputs = Self::empty(false);
158        let entries: Arc<[Entry]> = entries
159            .iter()
160            .map(|entry| {
161                decode_entry(
162                    entry.kind,
163                    EntrySelection::Rows(entry.index_combination.clone()),
164                    &entry.container,
165                )
166                .map(|mut model| {
167                    model.scratch_channels = entry.channel_selection.clone();
168                    model
169                })
170            })
171            .collect::<Result<Vec<_>, _>>()?
172            .into();
173        for row in rows {
174            inputs
175                .virtual_models
176                .entry(row.field_id)
177                .or_insert_with(|| Arc::clone(&entries));
178        }
179        inputs.load_geometry(ms, rows)?;
180        Ok(inputs)
181    }

217    fn load_geometry(
218        &mut self,
219        ms: &MeasurementSet,
220        rows: &[SelectedSolveRow],
221    ) -> Result<(), GainSolveError> {
222        self.finite_fields = self.virtual_models.iter().filter_map(|(&field, models)| {
223            models.iter().any(|entry| matches!(&entry.prediction, Prediction::FourierGrid(grid) if grid.needs_refocus())).then_some(field)
224        }).collect();
225        if self
226            .virtual_models
227            .values()
228            .any(|entries| !entries.is_empty())
229        {
230            let spectral_window = ms
231                .spectral_window()
232                .map_err(|source| read_error(ms, source))?;
233            let windows: BTreeSet<_> = rows
234                .iter()
235                .filter(|row| self.virtual_models.contains_key(&row.field_id))
236                .map(|row| row.data_spw_id)
237                .collect();
238            for window in windows {
239                let frequencies = spectral_window
240                    .chan_freq(window as usize)
241                    .map_err(|source| read_error(ms, source))?;
242                let reference = spectral_window
243                    .frequency_reference(window as usize)
244                    .map_err(|source| read_error(ms, source))?;
245                self.frequencies.insert(window, (frequencies, reference));
246            }
247            let engine = MsCalEngine::new(ms)
248                .and_then(|engine| engine.with_field_ephemerides(ms))
249                .map_err(|source| read_error(ms, source))?;
250            let field = ms.field().map_err(|source| read_error(ms, source))?;
251            for &id in self.virtual_models.keys() {
252                let index = id as usize;
253                if engine.field_moves(index)
254                    || field
255                        .num_poly(index)
256                        .map_err(|source| read_error(ms, source))?
257                        > 0
258                {
259                    self.dynamic_fields.insert(id);
260                }
261            }
262            self.geometry = Some(engine);
263        }
264        Ok(())
265    }

328    fn bind_model_buffers(
329        &mut self,
330        ms: &MeasurementSet,
331        rows: &[BucketedRow<'_>],
332        combine: GainSolveCombine,
333        states: Option<&BTreeMap<usize, i32>>,
334    ) -> Result<(), GainSolveError> {
335        if !self.needs_buffers() {
336            return Ok(());
337        }
338        let mut buffers =
339            BTreeMap::<(VisibilityBuffer, Option<i32>), Vec<&SelectedSolveRow>>::new();
340        for row in rows
341            .iter()
342            .filter(|(row, _, _)| self.virtual_models.contains_key(&row.field_id))
343        {
344            buffers
345                .entry((
346                    VisibilityBuffer::new(row, combine),
347                    states.map(|states| states[&row.0.row_index]),
348                ))
349                .or_default()
350                .push(row.0);
351        }
352        let mosaic_fields: BTreeSet<_> = self
353            .virtual_models
354            .iter()
355            .filter_map(|(&field, entries)| {
356                entries
357                    .iter()
358                    .any(|entry| {
359                        matches!(&entry.prediction,
360                            Prediction::FourierGrid(grid) if grid.uses_legacy_mosaic_pointing()
361                        )
362                    })
363                    .then_some(field)
364            })
365            .collect();
366        let mut pointings =
367            if !mosaic_fields.is_empty() && ms.subtable(SubtableId::Pointing).is_some() {
368                let queries: Vec<_> = buffers
369                    .values()
370                    .map(|rows| rows[0])
371                    .filter(|row| mosaic_fields.contains(&row.field_id))
372                    .map(|row| (row.antenna1, row.time_seconds, row.interval_seconds))
373                    .collect();
374                ms.pointing()
375                    .and_then(|pointing| pointing.legacy_buffer_directions(&queries))
376                    .map_err(|source| read_error(ms, source))?
377            } else {
378                Vec::new()
379            }
380            .into_iter();
381        let feeds = if mosaic_fields.is_empty() {
382            FeedOffsets::default()
383        } else {
384            FeedOffsets::load(ms).map_err(|source| read_error(ms, source))?
385        };
386        for ((_, state), rows) in buffers {
387            let first = rows[0];
388            let state =
389                state.map_or_else(|| get_i32(ms.main_table(), first.row_index, "STATE_ID"), Ok)?;
390            let selection = [first.field_id, first.data_spw_id, first.scan_number, state];
391            let focus_rows = if self.finite_fields.contains(&first.field_id) {
392                rows.iter()
393                    .map(|row| FocusSample {
394                        index: row.row_index,
395                        antennas: [row.antenna1, row.antenna2],
396                    })
397                    .collect()
398            } else {
399                Vec::new()
400            };
401            let buffer = Arc::new(ModelBuffer {
402                selection,
403                focus_rows,
404                legacy_mosaic_pointing: if mosaic_fields.contains(&first.field_id) {
405                    Some(feeds.direction(
406                        ms,
407                        self.geometry.as_ref().expect("mosaic has geometry"),
408                        first,
409                        pointings.next().flatten(),
410                    )?)
411                } else {
412                    None
413                },
414            });
415            for row in rows {
416                self.model_buffers
417                    .insert(row.row_index, Arc::clone(&buffer));
418            }
419        }
420        self.load_single_dish_pointings(ms, rows)
421    }

423    fn load_single_dish_pointings(
424        &mut self,
425        ms: &MeasurementSet,
426        rows: &[BucketedRow<'_>],
427    ) -> Result<(), GainSolveError> {
428        let queries: HashSet<_> = self
429            .virtual_models
430            .values()
431            .flat_map(|entries| entries.iter())
432            .filter_map(|entry| match &entry.prediction {
433                Prediction::SingleDish(dish) => Some(dish.pointing()),
434                _ => None,
435            })
436            .collect();
437        if queries.is_empty() {
438            return Ok(());
439        }
440        let pointing = ms
441            .subtable(SubtableId::Pointing)
442            .is_some()
443            .then(|| ms.pointing())
444            .transpose()
445            .map_err(|error| read_error(ms, error))?;
446        let Some(pointing) = pointing.filter(|pointing| pointing.row_count() > 0) else {
447            self.single_dish_feeds =
448                FeedOffsets::load(ms).map_err(|error| read_error(ms, error))?;
449            return Ok(());
450        };
451        let engine = self.geometry.as_ref().expect("single-dish geometry");
452        for query in queries {
453            let selected: Vec<_> = rows.iter().map(|(row, _, _)| *row).filter(|row| {
454                if row.antenna1 != row.antenna2 || row.flag_row {
455                    return false;
456                }
457                let Some(buffer) = self.model_buffers.get(&row.row_index) else { return false };
458                self.virtual_models[&row.field_id].iter().any(|entry| {
459                    entry.selection.matches(buffer.selection)
460                        && matches!(&entry.prediction, Prediction::SingleDish(dish) if dish.pointing() == query)
461                })
462            }).collect();
463            let requests: Vec<_> = selected
464                .iter()
465                .map(|row| (row.antenna1, row.time_seconds, row.interval_seconds))
466                .collect();
467            let target = query.convert_first.map(|reference| (engine, reference));
468            let directions = pointing
469                .single_dish_directions(&requests, &query.column, target)
470                .map_err(|error| read_error(ms, error))?;
471            self.single_dish_pointings.insert(
472                query.clone(),
473                selected
474                    .into_iter()
475                    .zip(directions)
476                    .filter_map(|(row, direction)| {
477                        direction.map(|direction| (row.row_index, direction))
478                    })
479                    .collect(),
480            );
481        }
482        Ok(())
483    }

498    pub(super) fn predict_model(
499        &self,
500        ms: &MeasurementSet,
501        row: &SelectedSolveRow,
502        correlations: &[i32],
503        (raw_flags, previous): (&ArrayD<bool>, Option<ArrayD<Complex32>>),
504        scratch_model: bool,
505        uvw_cell: Option<&ArrayValue>,
506    ) -> Result<Option<ArrayD<Complex32>>, GainSolveError> {
507        if let Some(error) = self.virtual_refusal(row.field_id) {
508            return Err(error);
509        }
510        let Some(entries) = self.virtual_models.get(&row.field_id) else {
511            return Ok(None);
512        };
513        if entries.is_empty() {
514            return Ok(None);
515        }
516        let buffer = &self.model_buffers[&row.row_index];
517        let mut matching = entries
518            .iter()
519            .filter(|entry| entry.selection.matches(buffer.selection))
520            .peekable();
521        if matching.peek().is_none() {
522            return Ok(None);
523        }
524        // CASA restores the last component recipe registered for a visibility
525        // buffer. A MeasurementSet can retain older overlapping recipes, for
526        // example after replacing a setjy model with imager.ft.
527        let virtual_components = (!scratch_model)
528            .then(|| {
529                entries.iter().rev().find(|entry| {
530                    matches!(&entry.prediction, Prediction::Components(_))
531                        && entry.selection.matches(buffer.selection)
532                })
533            })
534            .flatten();
535        let engine = self
536            .geometry
537            .as_ref()
538            .expect("virtual models have geometry");
539        // A vector PlotMS time iterator can span fields within an input chunk.
540        // Keep the row's model recipe and UVW while using the chunk's direction.
541        let (phase_field, phase_time) = self
542            .plot_phase_references
543            .get(&row.row_index)
544            .copied()
545            .unwrap_or_else(|| {
546                let time = self
547                    .phase_times
548                    .get(&row.row_index)
549                    .copied()
550                    .unwrap_or(row.time_seconds);
551                (row.field_id, time)
552            });
553        let field = field_direction(ms, engine, phase_field as usize, phase_time)
554            .map_err(|source| read_error(ms, source))?;
555        let frame = if self.component_voltage.is_some() {
556            engine
557                .spectral_frame_observatory_direction(row.time_seconds, field.clone())
558                .map_err(|source| read_error(ms, source))?
559        } else {
560            MeasFrame::new()
561                .with_epoch(MEpoch::from_mjd(
562                    row.time_seconds / 86400.0,
563                    engine.time_reference(),
564                ))
565                .with_position(engine.observatory_position().clone())
566                .with_direction(field.clone())
567        };
568        let uvw = match uvw_cell {
569            Some(cell) => uvw_coordinates(cell)?,
570            None => uvw(ms, row.row_index, self.uncached_main_reads)?,
571        };
572        let (frequencies, frequency_ref) = &self.frequencies[&row.data_spw_id];
573        let shape = [correlations.len(), frequencies.len()];
574        let mut model = match previous {
575            Some(model) if model.shape() != shape => {
576                return Err(model_error("MODEL_DATA does not match its layout"));
577            }
578            Some(model) if model.t().is_standard_layout() => model,
579            Some(model) => ArrayD::from_shape_fn(IxDyn(&shape).f(), |index| model[index]),
580            None => ArrayD::<Complex32>::zeros(IxDyn(&shape).f()),
581        };
582        // The cube sky equation adds each result from a working VisBuffer.
583        // Unassigned cells retain their last value in that buffer.
584        let multiple_grids = scratch_model
585            && matching
586                .clone()
587                .filter(|entry| matches!(entry.prediction, Prediction::FourierGrid(_)))
588                .nth(1)
589                .is_some();
590        let mut scratch_buffer = None;
591        let mut components_written = false;
592        for entry in matching {
593            let grid_row = || FourierRow {
594                ms,
595                row,
596                correlations,
597                frequencies,
598                raw_flags,
599                frequency_ref: *frequency_ref,
600                engine,
601                field: &field,
602                phase_time,
603                frame: &frame,
604                uvw,
605                uncached_main_reads: self.uncached_main_reads,
606                focus_rows: &buffer.focus_rows,
607                legacy_mosaic_pointing: buffer.legacy_mosaic_pointing.as_ref(),
608                scratch_channels: entry
609                    .scratch_channels
610                    .iter()
611                    .find(|channels| channels[0] == row.data_spw_id)
612                    .copied(),
613            };
614            let components = match &entry.prediction {
615                Prediction::Components(components) => {
616                    if virtual_components.is_some_and(|active| !std::ptr::eq(active, entry)) {
617                        continue;
618                    }
619                    components_written = true;
620                    components
621                }
622                Prediction::SingleDish(dish) => {
623                    if row.antenna1 != row.antenna2 || row.flag_row {
624                        continue;
625                    }
626                    let corrected;
627                    let pointing = match self.single_dish_pointings.get(dish.pointing()) {
628                        Some(pointings) => pointings.get(&row.row_index),
629                        None if !self.single_dish_feeds.is_empty() => {
630                            corrected = self
631                                .single_dish_feeds
632                                .direction(ms, engine, row, Some(field.clone()))?
633                                .direction;
634                            Some(&corrected)
635                        }
636                        None => Some(&field),
637                    };
638                    if let Some(direction) = pointing {
639                        dish.predict(&grid_row(), direction, &mut model)?;
640                    }
641                    continue;
642                }
643                Prediction::FourierGrid(grid) => {
644                    let merge = if !scratch_model {
645                        FourierMerge::VirtualSum
646                    } else if multiple_grids {
647                        FourierMerge::ScratchAccumulate(
648                            scratch_buffer
649                                .get_or_insert_with(|| model.clone())
650                                .as_slice_memory_order_mut()
651                                .expect("contiguous scratch buffer"),
652                        )
653                    } else if components_written {
654                        FourierMerge::ScratchReplaceComponents
655                    } else {
656                        FourierMerge::ScratchSum
657                    };
658                    grid.predict(grid_row(), &mut model, merge)?;
659                    components_written = false;
660                    continue;
661                }
662            };
663            self.add_component_prediction(components, grid_row(), &mut model)?;
664        }
665        Ok(Some(model))
666    }
667    fn add_component_prediction(
668        &self,
669        components: &ComponentList,
670        sample: FourierRow<'_>,
671        model: &mut ArrayD<Complex32>,
672    ) -> Result<(), GainSolveError> {
673        let FourierRow {
674            ms,
675            row,
676            correlations,
677            frequencies,
678            engine,
679            field,
680            frame,
681            uvw,
682            scratch_channels,
683            ..
684        } = sample;
685        let (basis, offset) = component_basis(correlations)?;
686        let antenna_frame = if self.component_voltage.is_some() {
687            // ComponentFTMachine rotates baselines in the antenna-zero
688            // frame. BeamSkyJones uses the observatory frame separately.
689            Some(
690                engine
691                    .antenna_frame(row.time_seconds, 0)
692                    .map_err(|source| read_error(ms, source))?,
693            )
694        } else {
695            None
696        };
697        let projection_frame = antenna_frame.as_ref().unwrap_or(frame);
698        let spectra = if let Some(voltage) = &self.component_voltage {
699            let start = scratch_channels.map_or(0, |channels| channels[1] as usize);
700            Some(
701                voltage
702                    .spectra(
703                        components,
704                        frequencies,
705                        basis,
706                        (row.data_spw_id, row.time_seconds, start),
707                        frame,
708                    )
709                    .map_err(model_error)?,
710            )
711        } else {
712            None
713        };
714        for (index, component) in components.iter().enumerate() {
715            let projection = UvwMachine::with_frame(
716                component.shape().reference_direction(),
717                field,
718                projection_frame,
719            )
720            .map_err(model_error)?;
721            let (rotated, phase) = projection.convert_projected([-uvw[0], -uvw[1], uvw[2]]);
722            let baseline = [-rotated[0], -rotated[1], rotated[2]];
723            let apparent = spectra
724                .as_ref()
725                .map(|spectra| {
726                    let shape = component.transform()?.at_baseline(baseline);
727                    Ok::<_, casa_components::ComponentError>((
728                        spectra.component(index, frequencies.len()),
729                        shape,
730                    ))
731                })
732                .transpose()
733                .map_err(model_error)?;
734            for (channel, &frequency) in frequencies.iter().enumerate() {
735                let visibility = if let Some((flux, shape)) = &apparent {
736                    let scale = shape.at_frequency(frequency).map_err(model_error)?;
737                    flux[channel].map(|value| value * scale)
738                } else {
739                    component
740                        .visibility(baseline, frequency, basis)
741                        .map_err(model_error)?
742                };
743                let rotation = Complex64::from_polar(
744                    1.0,
745                    std::f64::consts::TAU * phase * frequency / 299_792_458.0,
746                );
747                for (correlation, &code) in correlations.iter().enumerate() {
748                    let value = visibility[(code - offset) as usize] * rotation;
749                    model[[correlation, channel]] +=
750                        Complex32::new(value.re as f32, value.im as f32);
751                }
752            }
753        }
754        Ok(())
755    }
756}
757
758fn component_basis(correlations: &[i32]) -> Result<(Polarisation, i32), GainSolveError> {
759    let (basis, offset) = match correlations.first().copied() {
760        Some(1..=4) => (Polarisation::Stokes, 1),
761        Some(5..=8) => (Polarisation::Circular, 5),
762        Some(9..=12) => (Polarisation::Linear, 9),
763        _ => {
764            return Err(model_error(
765                "virtual component prediction requires Stokes, linear, or circular correlations",
766            ));
767        }
768    };
769    if correlations
770        .iter()
771        .any(|&code| !(offset..offset + 4).contains(&code))
772    {
773        return Err(model_error(
774            "virtual component prediction requires one correlation basis",
775        ));
776    }
777    Ok((basis, offset))
778}

780fn decode_entries(record: &RecordValue) -> Result<Vec<Entry>, GainSolveError> {
781    if record.get("version").is_some() {
782        return VirtualModelEntry::entries(record)
783            .map_err(model_error)?
784            .into_iter()
785            .map(|entry| {
786                let mut model = decode_entry(
787                    entry.kind,
788                    EntrySelection::Rows(entry.index_combination),
789                    &entry.container,
790                )?;
791                // These native recipes replay the writer's channel span.
792                // The SDGrid span also controls its single-channel bypass.
793                let saved_span = match entry.container.get("name") {
794                    Some(Value::Scalar(ScalarValue::String(name))) => {
795                        matches!(
796                            name.as_str(),
797                            "nPBWProjectFT" | "PBMosaicFT" | crate::ft::legacy_aw::MACHINE
798                        ) || (name == "SDGrid"
799                            && entry.container.get("casars_live_image_frequency").is_some())
800                    }
801                    _ => false,
802                };
803                if saved_span {
804                    if entry
805                        .channel_selection
806                        .iter()
807                        .any(|&[spw, start, count, step]| {
808                            spw < 0 || start < 0 || count <= 0 || step <= 0
809                        })
810                    {
811                        return Err(model_error("invalid saved Fourier channel selection"));
812                    }
813                    model.scratch_channels = entry.channel_selection;
814                }
815                Ok(model)
816            })
817            .collect();
818    }
819    let mut entries = Vec::new();
820    for (kind, count, prefix) in [
821        (VirtualModelKind::FtMachine, "numft", "ft"),
822        (VirtualModelKind::ComponentList, "numcl", "cl"),
823    ] {
824        let Some(Value::Scalar(ScalarValue::Int32(count))) = record.get(count) else {
825            continue;
826        };
827        for index in 0..*count {
828            let name = format!("{prefix}_{index}");
829            let Some(Value::Record(entry)) = record.get(&name) else {
830                return Err(model_error(format!(
831                    "legacy virtual model has no {name} record"
832                )));
833            };
834            let Some(Value::Record(container)) = entry.get("container") else {
835                return Err(model_error(format!("legacy model {name} has no container")));
836            };
837            let selection = EntrySelection::FieldAndWindow {
838                fields: legacy_ids(entry, "fields")?,
839                windows: legacy_ids(entry, "spws")?,
840            };
841            entries.push(decode_entry(kind, selection, container)?);
842        }
843    }
844    Ok(entries)
845}
846
847fn decode_entry(
848    kind: VirtualModelKind,
849    selection: EntrySelection,
850    container: &RecordValue,
851) -> Result<Entry, GainSolveError> {
852    let prediction = match kind {
853        VirtualModelKind::ComponentList => {
854            Prediction::Components(ComponentList::from_record(container).map_err(model_error)?)
855        }
856        VirtualModelKind::FtMachine if matches!(container.get("name"), Some(Value::Scalar(ScalarValue::String(name))) if name == "SDGrid") => {
857            Prediction::SingleDish(Box::new(SingleDishModel::new(container.clone())?))
858        }
859        VirtualModelKind::FtMachine => {
860            Prediction::FourierGrid(Box::new(FourierModel::new(container.clone())))
861        }
862    };
863    Ok(Entry {
864        selection,
865        prediction,
866        scratch_channels: Vec::new(),
867    })
868}

870fn legacy_ids(record: &RecordValue, field: &str) -> Result<Vec<i32>, GainSolveError> {
871    match record.get(field) {
872        Some(Value::Array(ArrayValue::Int32(values))) if values.ndim() == 1 => {
873            Ok(values.iter().copied().collect())
874        }
875        _ => Err(model_error(format!(
876            "legacy virtual model has no {field} vector"
877        ))),
878    }
879}

881fn field_direction(
882    ms: &MeasurementSet,
883    engine: &MsCalEngine,
884    field_id: usize,
885    time: f64,
886) -> Result<MDirection, MsError> {
887    if engine.field_moves(field_id) {
888        return engine.observation_direction_j2000(time, field_id);
889    }
890    let field = ms.field()?;
891    let direction = raw_field_phase_direction(ms, field_id)?;
892    if field.num_poly(field_id)? == 0 {
893        return Ok(direction);
894    }
895    let dt = time - field.time(field_id)?;
896    let polynomial = DirectionColumn::phase_dir(field.table()).get_direction_at(field_id, dt)?;
897    let (longitude, latitude) = polynomial.as_angles();
898    Ok(MDirection::from_angles(
899        longitude,
900        latitude,
901        direction.refer(),
902    ))
903}

905fn uvw(ms: &MeasurementSet, row: usize, uncached: bool) -> Result<[f64; 3], GainSolveError> {
906    let error = |source: casa_tables::TableError| read_error(ms, source.into());
907    let table = ms.main_table();
908    let values = if uncached {
909        let value = table
910            .column_accessor("UVW")
911            .and_then(|column| column.array_cells_owned_uncached(&[row]))
912            .map_err(error)?
913            .pop()
914            .flatten()
915            .ok_or_else(|| model_error("UVW must contain three real coordinates"))?;
916        Cow::Owned(value)
917    } else {
918        Cow::Borrowed(
919            table
920                .cell_accessor(row, "UVW")
921                .and_then(|cell| cell.array())
922                .map_err(error)?,
923        )
924    };
925    uvw_coordinates(values.as_ref())
926}
927
928fn uvw_coordinates(cell: &ArrayValue) -> Result<[f64; 3], GainSolveError> {
929    if let ArrayValue::Float64(values) = cell
930        && let Some(values) = values.as_slice()
931        && let Ok(values) = values.try_into()
932    {
933        return Ok(values);
934    }
935    Err(model_error("UVW must contain three real coordinates"))
936}

938fn read_error(ms: &MeasurementSet, source: MsError) -> GainSolveError {
939    GainSolveError::OpenMeasurementSet {
940        path: ms
941            .path()
942            .unwrap_or(Path::new("<in-memory>"))
943            .display()
944            .to_string(),
945        source,
946    }
947}
