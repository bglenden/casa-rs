14/// One antenna pointing ordinate with a direction frame, angular component, and interpolation mode.
15pub struct DirectionAxis {
16    frame: DirectionRef,
17    spline: bool,
18    longitude: bool,
19}

21/// Pointing projections and ordinate-side choices in PlotMS series order.
22pub struct PointingAxes {
23    pairs: Vec<[Option<DirectionAxis>; 2]>,
24    right_ordinates: Vec<bool>,
25}
26
27impl PointingAxes {
28    /// Builds the pointing specifications for pairs of horizontal and vertical axes. Side choices
29    /// follow the same series order.
30    pub fn new(pairs: Vec<[Option<DirectionAxis>; 2]>, right_ordinates: Vec<bool>) -> Self {
31        Self {
32            pairs,
33            right_ordinates,
34        }
35    }
36
37    /// Returns the number of configured axis pairs.
38    pub fn len(&self) -> usize {
39        self.pairs.len()
40    }
41
42    /// Reports whether the configuration has no axis pairs.
43    pub fn is_empty(&self) -> bool {
44        self.pairs.is_empty()
45    }
46
47    /// Borrows the flags for drawing each series on the right ordinate.
48    pub fn right_ordinates(&self) -> &[bool] {
49        &self.right_ordinates
50    }
51
52    /// Combines distinct horizontal pointing labels, including frames and units.
53    pub fn x_label(&self) -> Option<String> {
54        let mut labels = Vec::new();
55        for index in 0..self.len() {
56            if let Some(label) = self.label(index, 0)
57                && !labels.contains(&label)
58            {
59                labels.push(label);
60            }
61        }
62        (!labels.is_empty()).then(|| labels.join(", "))
63    }
64
65    /// Returns the pointing label in degrees for the indexed series and component. Indices must
66    /// name an existing pair and its x or y component.
67    pub fn label(&self, index: usize, component: usize) -> Option<String> {
68        let axis = self.pairs[index][component]?;
69        let component = match (axis.frame, axis.longitude) {
70            (DirectionRef::GALACTIC, true) => "Longitude",
71            (DirectionRef::GALACTIC, false) => "Latitude",
72            (DirectionRef::AZELGEO, true) => "Azimuth",
73            (DirectionRef::AZELGEO, false) => "Elevation",
74            (_, true) => "Right Ascension",
75            (_, false) => "Declination",
76        };
77        Some(format!("{} {component} (degrees)", axis.frame))
78    }
79}

81/// Parses the frame and interpolation for a pointing axis. Other axis kinds return None, and
82/// unsupported pointing settings are errors.
83pub fn parse_axis(
84    axis: Option<MsAxis>,
85    frame: &str,
86    interp: &str,
87) -> Result<Option<DirectionAxis>, String> {
88    if !axis.is_some_and(MsAxis::is_pointing) {
89        return Ok(None);
90    }
91    let frame = match frame {
92        "" | "icrs" => DirectionRef::ICRS,
93        "j2000" => DirectionRef::J2000,
94        "b1950" => DirectionRef::B1950,
95        "galactic" => DirectionRef::GALACTIC,
96        "azelgeo" => DirectionRef::AZELGEO,
97        other => return Err(format!("invalid PlotMS pointing frame {other:?}")),
98    };
99    let spline = match interp {
100        "" | "spline" | "cubic spline" => true,
101        "nearest" => false,
102        other => return Err(format!("invalid PlotMS pointing interpolation {other:?}")),
103    };
104    Ok(Some(DirectionAxis {
105        frame,
106        spline,
107        longitude: axis == Some(MsAxis::AntRa),
108    }))
109}

111pub(super) struct PointingCoordinates<'a> {
112    axes: &'a PointingAxes,
113    source: DirectionRef,
114    tracks: BTreeMap<i32, PointingTrack>,
115    geometry: Option<MsCalEngine>,
116}
117
118struct PointingTrack {
119    times: Vec<f64>,
120    angles: Vec<[f64; 2]>,
121    second_derivatives: Vec<[f64; 2]>,
122}
123
124impl<'a> PointingCoordinates<'a> {
125    pub(super) fn read(
126        ms: &MeasurementSet,
127        axes: &'a PointingAxes,
128        antennas: &[i32],
129    ) -> Result<Self, String> {
130        let pointing = ms.pointing().map_err(|error| error.to_string())?;
131        if pointing.row_count() == 0 {
132            return Err("PlotMS antenna directions require a nonempty POINTING table".into());
133        }
134        let observation = ms.observation().map_err(|error| error.to_string())?;
135        if observation.row_count() == 0 {
136            return Err("PlotMS antenna directions require an observatory".into());
137        }
138        for row in 0..observation.row_count() {
139            let name = observation
140                .string(row, "TELESCOPE_NAME")
141                .map_err(|error| error.to_string())?;
142            if !matches!(name.as_str(), "ALMA" | "ASTE" | "NRO") {
143                return Err(format!(
144                    "PlotMS antenna directions are unsupported for {name}"
145                ));
146            }
147        }
148        let mut antennas = antennas.to_vec();
149        antennas.sort_unstable();
150        antennas.dedup();
151        let samples = pointing
152            .plotms_samples(&antennas)
153            .map_err(|error| error.to_string())?;
154        let frame =
155            resolve_direction_reference_selected(pointing.table(), "POINTING", "DIRECTION", 0)
156                .map_err(|error| error.to_string())?;
157        if samples.iter().any(|sample| sample.2.refer() != frame) {
158            return Err("PlotMS POINTING directions in multiple frames are pending #1032".into());
159        }
160        let mut tracks = BTreeMap::new();
161        let direction_axes = || axes.pairs.iter().flatten().flatten();
162        let spline = direction_axes().any(|axis| axis.spline);
163        for samples in samples.chunk_by(|left, right| left.0 == right.0) {
164            let times = samples.iter().map(|sample| sample.1).collect::<Vec<_>>();
165            let angles = samples
166                .iter()
167                .map(|sample| {
168                    let direction = &sample.2;
169                    let [x, y, _] = direction.cosines();
170                    [y.atan2(x), direction.latitude_rad()]
171                })
172                .collect::<Vec<_>>();
173            let second_derivatives = if spline && times.len() >= 4 {
174                spline_second_derivatives(&times, &angles)
175            } else {
176                Vec::new()
177            };
178            tracks.insert(
179                samples[0].0,
180                PointingTrack {
181                    times,
182                    angles,
183                    second_derivatives,
184                },
185            );
186        }
187        let geometry = direction_axes()
188            .any(|axis| axis.frame != frame)
189            .then(|| MsCalEngine::for_calibration(ms))
190            .transpose()
191            .map_err(|error| error.to_string())?;
192        Ok(Self {
193            axes,
194            source: frame,
195            tracks,
196            geometry,
197        })
198    }
199
200    pub(super) fn values(
201        &self,
202        antenna: i32,
203        time: f64,
204        values: &mut Vec<[Option<f64>; 2]>,
205    ) -> Result<(), String> {
206        let sample = |axis: DirectionAxis| -> Result<f64, String> {
207            let track = self
208                .tracks
209                .get(&antenna)
210                .filter(|track| track.times.len() >= 4);
211            if track.is_none()
212                && !axis.spline
213                && self.source != axis.frame
214                && (self.source == DirectionRef::AZELGEO || axis.frame == DirectionRef::AZELGEO)
215            {
216                return Err("PlotMS nearest pointing frame conversion with fewer than four samples is pending #1032".into());
217            }
218            let (angles, epoch) = track.map_or(([0.0, 0.0], time), |track| {
219                track.interpolate(time, axis.spline)
220            });
221            let direction = MDirection::from_angles(angles[0], angles[1], self.source);
222            let direction = if self.source == axis.frame {
223                direction
224            } else {
225                let frame = self
226                    .geometry
227                    .as_ref()
228                    .expect("pointing conversion prepared geometry")
229                    .antenna_frame(epoch, antenna as usize)
230                    .map_err(|error| error.to_string())?
231                    .with_epoch(MEpoch::from_mjd_seconds(epoch, EpochRef::UTC));
232                direction
233                    .convert_to(axis.frame, &frame)
234                    .map_err(|error| error.to_string())?
235            };
236            Ok(if axis.longitude {
237                direction.longitude_rad().to_degrees().rem_euclid(360.0)
238            } else {
239                direction.latitude_rad().to_degrees()
240            })
241        };
242        values.clear();
243        for [x, y] in &self.axes.pairs {
244            values.push([x.map(sample).transpose()?, y.map(sample).transpose()?]);
245        }
246        Ok(())
247    }
248}

250impl PointingTrack {
251    fn interpolate(&self, time: f64, spline: bool) -> ([f64; 2], f64) {
252        // CASA uses the segment start in nearest mode, including the final segment.
253        let left = self
254            .times
255            .partition_point(|&epoch| epoch <= time)
256            .saturating_sub(1)
257            .min(self.times.len() - 2);
258        if !spline {
259            return (self.angles[left], self.times[left]);
260        }
261        let right = left + 1;
262        let width = self.times[right] - self.times[left];
263        let b = (time - self.times[left]) / width;
264        let a = 1.0 - b;
265        let angles = std::array::from_fn(|axis| {
266            a * self.angles[left][axis]
267                + b * self.angles[right][axis]
268                + ((a.powi(3) - a) * self.second_derivatives[left][axis]
269                    + (b.powi(3) - b) * self.second_derivatives[right][axis])
270                    * width.powi(2)
271                    / 6.0
272        });
273        (angles, time)
274    }
275}
276
277fn spline_second_derivatives(times: &[f64], angles: &[[f64; 2]]) -> Vec<[f64; 2]> {
278    let n = times.len();
279    let mut upper = vec![0.0; n];
280    let mut second = vec![[0.0; 2]; n];
281    // Solve the tridiagonal system with zero second derivatives at the endpoints.
282    for i in 1..n - 1 {
283        let before = times[i] - times[i - 1];
284        let after = times[i + 1] - times[i];
285        let diagonal = 2.0 * (before + after) - before * upper[i - 1];
286        upper[i] = after / diagonal;
287        second[i] = std::array::from_fn(|axis| {
288            let slope_before = (angles[i][axis] - angles[i - 1][axis]) / before;
289            let slope_after = (angles[i + 1][axis] - angles[i][axis]) / after;
290            (6.0 * (slope_after - slope_before) - before * second[i - 1][axis]) / diagonal
291        });
292    }
293    for i in (1..n - 1).rev() {
294        second[i] = std::array::from_fn(|axis| second[i][axis] - upper[i] * second[i + 1][axis]);
295    }
296    second
