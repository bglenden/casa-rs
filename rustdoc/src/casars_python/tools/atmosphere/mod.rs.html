22const LOGGER: &str = "casars.tools.atmosphere";
23const TYPE_NAMES: [&str; 5] = [
24    "TROPICAL",
25    "MIDLATSUMMER",
26    "MIDLATWINTER",
27    "SUBARTSUMMER",
28    "SUBARTWINTER",
29];
30
31#[pyclass(name = "atmosphere", module = "casars.tools", frozen)]
32#[derive(Default)]
33pub(crate) struct PyAtmosphere {
34    session: Arc<Mutex<AtmosphereSession>>,
35}
36
37#[pymethods]
38impl PyAtmosphere {
39    #[new]
40    #[pyo3(signature = (*_args, **kwargs), text_signature = "(*args, **kwargs)")]
41    fn new(_args: &Bound<'_, PyTuple>, kwargs: Option<&Bound<'_, PyDict>>) -> PyResult<Self> {
42        // CASA consumes only the existing object hook in its variadic wrapper.
43        if let Some(kwargs) = kwargs
44            && let Some(existing) = kwargs.get_item("swig_object")?
45            && !existing.is_none()
46        {
47            let tool = existing.extract::<PyRef<'_, Self>>()?;
48            return Ok(Self {
49                session: Arc::clone(&tool.session),
50            });
51        }
52        Ok(Self::default())
53    }
54
55    fn close(&self, py: Python<'_>) -> bool {
56        self.run(py, AtmosphereSession::close);
57        true
58    }
59
60    fn done(&self, py: Python<'_>) -> bool {
61        self.close(py)
62    }
63
64    #[pyo3(name = "getAtmVersion")]
65    fn get_atm_version(&self) -> &'static str {
66        "ATM-1_0_0"
67    }
68
69    #[pyo3(name = "listAtmosphereTypes")]
70    fn list_atmosphere_types(&self) -> Vec<String> {
71        TYPE_NAMES
72            .iter()
73            .enumerate()
74            .map(|(index, name)| format!("{} - {name}", index + 1))
75            .collect()
76    }
77}
78
79impl PyAtmosphere {
80    fn run<T: Send>(
81        &self,
82        py: Python<'_>,
83        operation: impl FnOnce(&mut AtmosphereSession) -> T + Send,
84    ) -> T {
85        py.detach(|| {
86            let mut session = self.session.lock().unwrap_or_else(PoisonError::into_inner);
87            operation(&mut session)
88        })
89    }
90}
91
92fn tuple(py: Python<'_>, items: impl IntoIterator<Item = CasaValue>) -> PyResult<Py<PyAny>> {
93    let values = items
94        .into_iter()
95        .map(|item| reply_value(py, &item))
96        .collect::<PyResult<Vec<_>>>()?;
97    Ok(PyTuple::new(py, values)?.into_any().unbind())
98}
99
100fn warn(py: Python<'_>, message: &str) -> PyResult<()> {
101    emit(py, LOGGER, &[LogLine::warning(message)])
102}
103
104fn error(message: impl std::fmt::Display) -> PyErr {
105    PyRuntimeError::new_err(message.to_string())
106}
