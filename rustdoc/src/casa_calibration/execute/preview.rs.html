5pub(super) fn evaluate_rows(
6    ms: &MeasurementSet,
7    plan: &ApplyPlan,
8    share_input_flags: bool,
9    weight_source: PreviewWeightSource,
10) -> Result<HashMap<usize, EvaluatedApplyRow>, ApplyExecutionError> {
11    let mut rows = HashMap::new();
12    for_each_apply_row(
13        ms,
14        plan,
15        apply_observed_column(ms, false),
16        false,
17        share_input_flags,
18        weight_source,
19        None,
20        |index, row| {
21            rows.insert(index, row);
22        },
23    )?;
24    Ok(rows)
25}

27/// Evaluate one row at a time, allowing TRIAL validation without accumulating
28/// corrected data or publishing any MS changes.
29#[expect(
30    clippy::too_many_arguments,
31    reason = "preview and tool callers use the same row executor"
32)]
33pub(super) fn for_each_apply_row(
34    ms: &MeasurementSet,
35    plan: &ApplyPlan,
36    observed_column: &str,
37    forward_model: bool,
38    share_input_flags: bool,
39    weight_source: PreviewWeightSource,
40    mut tool: Option<&mut ToolApplyExecution<'_>>,
41    mut consume: impl FnMut(usize, EvaluatedApplyRow),
42) -> Result<(), ApplyExecutionError> {
43    let ms_path = display_ms_path(ms);
44    let correlation_types_by_ddid = load_correlation_types_by_ddid(ms).map_err(|source| {
45        ApplyExecutionError::MutateMeasurementSet {
46            path: ms_path.clone(),
47            source,
48        }
49    })?;
50    let mut chain = ApplyChain::load(ms, plan, forward_model)?;
51    if let Some(run) = &tool {
52        chain.arrange_for_tool(plan, run.correction.parang_terms, run.additive);
53    }
54
55    let any_calwt = applies_preview_weights(plan, &chain);
56    let mut columns = vec!["FLAG"];
57    if !forward_model {
58        columns.push(observed_column);
59    }
60    let plot_weight_columns = match weight_source {
61        PreviewWeightSource::UnitFactors => None,
62        PreviewWeightSource::FromSigma { channel_weighting } => Some(weights::plot_weight_columns(
63            ms.main_table(),
64            plan.selected_rows.first().map(|row| row.row_index),
65            &mut columns,
66            channel_weighting,
67        )?),
68    };
69    let read_error = |source| ApplyExecutionError::MutateMeasurementSet {
70        path: ms_path.clone(),
71        source: MsError::from(source),
72    };
73    let mut readers = PreviewReaders::new(ms.main_table(), &columns).map_err(read_error)?;
74    let data_index = (!forward_model).then_some(1);
75    let batch_rows = preview_batch_rows(plan, &correlation_types_by_ddid);
76    let mut arrays = Vec::new();
77    let mut geometry_cache = RowGeometryCache::default();
78    for (index, row) in plan.selected_rows.iter().enumerate() {
79        let offset = index % batch_rows;
80        if offset == 0 {
81            arrays.clear();
82            let end = (index + batch_rows).min(plan.selected_rows.len());
83            let indices: Vec<_> = plan.selected_rows[index..end]
84                .iter()
85                .map(|row| row.row_index)
86                .collect();
87            arrays = readers
88                .selected
89                .iter()
90                .map(|column| {
91                    column
92                        .as_ref()
93                        .map(|column| column.array_cells_owned_uncached(&indices))
94                        .transpose()
95                })
96                .collect::<Result<Vec<_>, _>>()
97                .map_err(read_error)?;
98        }
99        let correlation_types = correlation_types_by_ddid
100            .get(&row.data_desc_id)
101            .ok_or_else(|| ApplyExecutionError::UnsupportedCorrelationLayout {
102                data_desc_id: row.data_desc_id,
103                correlation_types: Vec::new(),
104            })?;
105        if let Some(prepared) = readers.prepared.as_mut() {
106            prepared.load(row.row_index).map_err(read_error)?;
107        }
108        let read = |slot: usize| {
109            if let (Some(prepared), Some(prepared_slot)) =
110                (&readers.prepared, readers.prepared_slots[slot])
111            {
112                return prepared.array_at(prepared_slot).map_err(read_error);
113            }
114            arrays[slot]
115                .as_ref()
116                .and_then(|cells| cells[offset].as_ref())
117                .ok_or_else(|| {
118                    read_error(casa_tables::TableError::ColumnNotFound {
119                        row_index: row.row_index,
120                        column: columns[slot].to_owned(),
121                    })
122                })
123        };
124        let input_flags = read(0)?;
125        let shared;
126        let original_flags =
127            if share_input_flags && let Some(flags) = shared_channel_flags(input_flags) {
128                shared = flags;
129                &shared
130            } else {
131                input_flags
132            };
133        let units;
134        let data = if let Some(index) = data_index {
135            read(index)?
136        } else {
137            units = ArrayValue::Complex32(ArrayD::from_elem(
138                IxDyn(original_flags.shape()).f(),
139                Complex32::new(1., 0.),
140            ));
141            &units
142        };
143        let complex_data;
144        let data = if let ArrayValue::Float32(values) = data {
145            complex_data = complex_visibility_from_float(values);
146            &complex_data
147        } else {
148            data
149        };
150        let plot_weights = plot_weight_columns
151            .map(|columns| columns.read(read, data.shape()))
152            .transpose()?;
153        if !availability::available(row, plan) {
154            if plan.apply_mode == ApplyMode::Trial {
155                continue;
156            }
157            // VisEquation::collapse2 excludes data that cannot be pre-calibrated
158            // from solving, independently of the table's calwt setting.
159            let shape = IxDyn(data.shape());
160            consume(
161                row.row_index,
162                EvaluatedApplyRow {
163                    corrected_data: data.clone(),
164                    flags: ArrayValue::Bool(ArrayD::from_elem(shape.clone(), true)),
165                    model_gains: None,
166                    weight_factors: Some(ArrayD::zeros(shape)),
167                    plot_weights,
168                },
169            );
170            continue;
171        }
172        // Unit weights through the calwt update leave its factors.
173        let unit_weight = (any_calwt && plot_weights.is_none()).then(|| {
174            ArrayValue::Float32(ArrayD::from_elem(IxDyn(&[correlation_types.len()]), 1.0))
175        });
176
177        let result = apply_row(
178            row,
179            ExecutionRowInputs {
180                correlation_types,
181                data,
182                original_flags: Some(original_flags),
183                original_weight: plot_weights
184                    .as_ref()
185                    .map(|weights| &weights.scalar)
186                    .or(unit_weight.as_ref()),
187                original_weight_spectrum: plot_weights
188                    .as_ref()
189                    .and_then(|weights| weights.spectrum.as_ref()),
190                has_weight_spectrum: plot_weight_columns
191                    .map_or(any_calwt, weights::PlotWeightColumns::has_spectrum),
192                preserved_corrected_data: None,
193                preserved_weight_spectrum: None,
194                channel_grid: None,
195                forward_model,
196                skip_unavailable_terms: false,
197                trial: false,
198                parang_calwt: false,
199            },
200            plan,
201            &chain,
202            Some(&mut geometry_cache),
203            None,
204            tool.as_deref_mut().map(|run| &mut *run.activity),
205        )?;
206        let (plot_weights, weight_factors) = match plot_weights {
207            Some(weights) => (
208                Some(PlotWeights {
209                    scalar: result.updated_weight.unwrap_or(weights.scalar),
210                    spectrum: result.updated_weight_spectrum.or(weights.spectrum),
211                }),
212                None,
213            ),
214            None => (
215                None,
216                match result.updated_weight_spectrum {
217                    Some(ArrayValue::Float32(factors)) => Some(factors),
218                    _ => None,
219                },
220            ),
221        };
222        consume(
223            row.row_index,
224            EvaluatedApplyRow {
225                corrected_data: result.corrected_data,
226                model_gains: None,
227                flags: result
228                    .updated_flags
229                    .unwrap_or_else(|| original_flags.clone()),
230                weight_factors,
231                plot_weights,
232            },
233        );
234    }
235
236    Ok(())
237}

239pub(super) struct PreviewReaders<'a> {
240    pub(super) selected: Vec<Option<casa_tables::TableColumn<'a>>>,
241    pub(super) prepared: Option<casa_tables::PreparedTableRow<'a>>,
242    pub(super) prepared_slots: Vec<Option<usize>>,
243}
244
245impl<'a> PreviewReaders<'a> {
246    fn new(
247        main: &'a casa_tables::Table,
248        columns: &[&str],
249    ) -> Result<Self, casa_tables::TableError> {
250        let selected = columns
251            .iter()
252            .map(|&name| {
253                let supports_selection = main.data_manager_info().iter().any(|manager| {
254                    matches!(
255                        manager.dm_type.as_str(),
256                        "StandardStMan"
257                            | "TiledColumnStMan"
258                            | "TiledShapeStMan"
259                            | "TiledCellStMan"
260                            | "TiledDataStMan"
261                    ) && manager.columns.iter().any(|column| column == name)
262                });
263                supports_selection
264                    .then(|| main.column_accessor(name))
265                    .transpose()
266            })
267            .collect::<Result<Vec<_>, _>>()?;
268        // Linked tables and other whole-column readers reuse one projection.
269        let mut prepared_columns = Vec::new();
270        let prepared_slots = selected
271            .iter()
272            .zip(columns)
273            .map(|(reader, &name)| {
274                reader.is_none().then(|| {
275                    let slot = prepared_columns.len();
276                    prepared_columns.push(name);
277                    slot
278                })
279            })
280            .collect();
281        let prepared = (!prepared_columns.is_empty())
282            .then(|| main.row_accessor().prepare(&prepared_columns))
283            .transpose()?;
284        Ok(Self {
285            selected,
286            prepared,
287            prepared_slots,
288        })
289    }
290}

292pub(super) fn preview_batch_rows(plan: &ApplyPlan, correlations: &HashMap<i32, Vec<i32>>) -> usize {
293    const BATCH_BYTES: usize = 8 * 1024 * 1024;
294    const MAX_ROWS: usize = 256;
295    let channels = plan
296        .measurement_set_spectral_windows
297        .iter()
298        .filter_map(|window| usize::try_from(window.num_chan).ok())
299        .max()
300        .unwrap_or(0);
301    let correlations = correlations.values().map(Vec::len).max().unwrap_or(0);
302    // Bound storage for the data, flags and an optional sigma spectrum.
303    let row_bytes = channels.saturating_mul(correlations).saturating_mul(16);
304    (BATCH_BYTES / row_bytes.max(1)).clamp(1, MAX_ROWS)
305}

307pub(super) fn applies_preview_weights(plan: &ApplyPlan, chain: &ApplyChain) -> bool {
308    plan.apply_mode.writes_corrected_data()
309        && plan
310            .calibration_tables
311            .iter()
312            .zip(&chain.tables)
313            .any(|(table, loaded)| table.calwt && loaded.supports_calwt)
314}

316pub(super) fn shared_channel_flags(flags: &ArrayValue) -> Option<ArrayValue> {
317    let ArrayValue::Bool(flags) = flags else {
318        return None;
319    };
320    if flags.ndim() != 2 {
321        return None;
322    }
323    let mut flags = flags.clone();
324    for mut channel in flags.axis_iter_mut(ndarray::Axis(1)) {
325        if channel.iter().any(|&flag| flag) {
326            channel.fill(true);
327        }
328    }
329    Some(ArrayValue::Bool(flags))
330}
