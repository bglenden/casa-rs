2//! The observation date in a FITS header: `DATE-OBS` and `TIMESYS`.
3//!
4//! casacore's `ObsInfo::toFITS` writes the date with `FITSDateUtil::toFITS`:
5//! `MVTime::string(MVTime::FITS, 12)` of the epoch in its own time scale,
6//! `yyyy-mm-ddThh:mm:ss.ssssss`, with that scale in `TIMESYS`.
7//! `FITSCoordinateUtil::getObsInfo` reads it back: `MJD-OBS` when the header
8//! has it, else `DATE-OBS` (`FITSDateUtil::fromFITS`, which also reads
9//! FITS's old `dd/mm/yy`), in the scale `TIMESYS` names, UTC without one.
10
11use casa_types::measures::epoch::{EpochRef, MEpoch};
12use casa_types::quanta::Quantity;
13use casa_types::quanta::format::{FITS, format_time};
14
15/// `DATE-OBS` and `TIMESYS` for `epoch`.
16///
17/// casacore names Terrestrial Time `TDT` and UT1 `UT`; this writes the FITS
18/// standard's `TT` and `UT1`, which casacore reads as the same scales.
19pub(super) fn fits_date(epoch: &MEpoch) -> (String, String) {
20    (
21        format_time(epoch.value().as_mjd(), FITS, 12),
22        epoch.refer().as_str().to_string(),
23    )
24}
25
26/// The epoch of a header's `MJD-OBS`, or else its `DATE-OBS`, in the scale
27/// of its `TIMESYS`; `None` when it has neither date, or the date or scale
28/// does not decode. casa-rs then sets no date rather than guess a scale:
29/// `FITSDateUtil::fromFITS` refuses an unknown one, though CASA 6.7.6's
30/// `importfits`, which loses every `TIMESYS`, labels the date UTC.
31pub(super) fn obs_epoch(
32    mjd_obs: Option<f64>,
33    date_obs: Option<&str>,
34    timesys: Option<&str>,
35) -> Option<MEpoch> {
36    let reference = match timesys.map(str::trim) {
37        None | Some("") => EpochRef::UTC,
38        Some(name) => name.parse().ok()?,
39    };
40    let mjd = match (mjd_obs, date_obs) {
41        (Some(mjd), _) => mjd,
42        (None, Some(date)) => date_mjd(date.trim())?,
43        (None, None) => return None,
44    };
45    Some(MEpoch::from_mjd(mjd, reference))
46}
47
48/// `date` in MJD days: `yyyy-mm-dd[Thh:mm:ss[.s…]]`, or the old `dd/mm/yy`
49/// of the twentieth century.
50fn date_mjd(date: &str) -> Option<f64> {
51    if date.contains('/') {
52        let parts = date.split('/').collect::<Vec<_>>();
53        let [day, month, year] = parts.as_slice() else {
54            return None;
55        };
56        let two_digits = |part: &&str| part.len() == 2 && part.bytes().all(|b| b.is_ascii_digit());
57        if ![day, month, year].into_iter().all(two_digits) {
58            return None;
59        }
60        return days(&format!("19{year}/{month}/{day}"));
61    }
62    days(date)
63}
64
65/// `text` read as `MVTime::read` reads a date, in days.
66fn days(text: &str) -> Option<f64> {
67    let quantity = Quantity::read(text).ok()?;
68    (quantity.unit().name() == "d").then(|| quantity.value())
69}

71#[cfg(test)]
72mod tests {
73    use super::*;
74
75    fn mjd(epoch: &MEpoch) -> f64 {
76        epoch.value().as_mjd()
77    }
78
79    #[test]
80    fn a_date_is_written_as_casa_writes_it() {
81        // CASA 6.7.6's exportfits of an image observed at MJD
82        // 51544.123456789 UTC, and of one at 59000.5 TAI.
83        assert_eq!(
84            fits_date(&MEpoch::from_mjd(51_544.123_456_789, EpochRef::UTC)),
85            ("2000-01-01T02:57:46.666569".to_string(), "UTC".to_string())
86        );
87        assert_eq!(
88            fits_date(&MEpoch::from_mjd(59_000.5, EpochRef::TAI)),
89            ("2020-05-31T12:00:00.000000".to_string(), "TAI".to_string())
90        );
91    }
92
93    #[test]
94    fn dates_at_day_and_leap_day_boundaries_are_written_as_casa_writes_them() {
95        // CASA 6.7.6's exportfits of each epoch (#1020 review): the last
96        // microsecond of a day, the last of a leap day and the next day's
97        // first, and an epoch whose rounding would carry over midnight.
98        for (value, reference, date) in [
99            (
100                59_000.999_999_999_99,
101                EpochRef::UTC,
102                "2020-05-31T23:59:59.999999",
103            ),
104            (
105                59_001.0 - 5.0e-12,
106                EpochRef::UTC,
107                "2020-05-31T23:59:59.999999",
108            ),
109            (58_908.5, EpochRef::UTC, "2020-02-29T12:00:00.000000"),
110            (
111                58_908.999_999_999_99,
112                EpochRef::UTC,
113                "2020-02-29T23:59:59.999999",
114            ),
115            (58_909.0, EpochRef::UTC, "2020-03-01T00:00:00.000000"),
116            // CASA names these scales UT and TDT; casa-rs writes the FITS
117            // standard's UT1 and TT, which casacore reads alike.
118            (59_000.25, EpochRef::UT1, "2020-05-31T06:00:00.000000"),
119            (51_544.5, EpochRef::TT, "2000-01-01T12:00:00.000000"),
120        ] {
121            let (written, timesys) = fits_date(&MEpoch::from_mjd(value, reference));
122            assert_eq!(written, date, "{value} {reference:?}");
123            assert_eq!(timesys, reference.as_str());
124        }
125    }
126
127    #[test]
128    fn a_written_date_reads_back_in_its_scale() {
129        for (value, reference) in [
130            (51_544.123_456_789, EpochRef::UTC),
131            (59_000.5, EpochRef::TAI),
132            (60_000.25, EpochRef::TT),
133            (59_000.25, EpochRef::UT1),
134            (58_908.999_999_999_99, EpochRef::UTC),
135        ] {
136            let (date, timesys) = fits_date(&MEpoch::from_mjd(value, reference));
137            let epoch = obs_epoch(None, Some(&date), Some(&timesys)).unwrap();
138            assert_eq!(epoch.refer(), reference);
139            // To the microsecond the date is written to.
140            assert!((mjd(&epoch) - value).abs() * 86_400.0 < 1e-6, "{date}");
141        }
142    }
143
144    #[test]
145    fn headers_read_as_casacore_reads_them() {
146        // No TIMESYS is UTC; casacore's aliases name their scales.
147        let epoch = obs_epoch(None, Some("2020-05-31T12:00:00"), None).unwrap();
148        assert_eq!((mjd(&epoch), epoch.refer()), (59_000.5, EpochRef::UTC));
149        let epoch = obs_epoch(None, Some("2020-05-31"), Some("TDT ")).unwrap();
150        assert_eq!((mjd(&epoch), epoch.refer()), (59_000.0, EpochRef::TT));
151        // MJD-OBS before DATE-OBS.
152        let epoch = obs_epoch(Some(51_544.5), Some("2020-05-31"), Some("IAT")).unwrap();
153        assert_eq!((mjd(&epoch), epoch.refer()), (51_544.5, EpochRef::TAI));
154        // FITS's old form, of the twentieth century.
155        let epoch = obs_epoch(None, Some("31/12/99"), None).unwrap();
156        assert_eq!(mjd(&epoch), 51_543.0);
157    }
158
159    #[test]
160    fn a_date_or_scale_that_does_not_decode_gives_no_date() {
161        // casa-rs's old placeholder, an unknown scale, and no date at all.
162        assert!(obs_epoch(None, Some("MJD 51544.0000000000"), None).is_none());
163        assert!(obs_epoch(None, Some("2020-05-31"), Some("GPS-ish")).is_none());
164        assert!(obs_epoch(None, Some("1/2/3"), None).is_none());
165        assert!(obs_epoch(None, None, Some("UTC")).is_none());
166    }
