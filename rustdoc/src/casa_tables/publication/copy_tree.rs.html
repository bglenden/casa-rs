11/// Copy a locked casacore table and its physical subtables into a private,
12/// absent destination tree. Hold the source locks for the duration of the copy.
13///
14/// A destination table gets a fresh request list and the source's persisted
15/// synchronization data. Symbolic links are copied as directory entries.
16///
17/// # Errors
18/// Returns an error for unreadable, unsupported or uncopyable entries.
19/// After failure the caller owns any partial destination.
20pub fn copy_locked_table_tree(source: &Path, target: &Path) -> io::Result<()> {
21    copy_tree(source, target, None)
22}
23
24/// Copy a non-table directory, retaining the table copy rules for physical
25/// subtables. Copy the referents of regular-file links selected at the root.
26///
27/// # Errors
28/// Returns an error for unreadable, unsupported or uncopyable entries.
29/// After failure the caller owns any partial destination.
30pub fn copy_locked_external_tree(
31    source: &Path,
32    target: &Path,
33    follow_file_link: impl Fn(&OsStr) -> bool,
34) -> io::Result<()> {
35    copy_tree(source, target, Some(&follow_file_link))
36}
37
38type FileLinkSelection<'a> = Option<&'a dyn Fn(&OsStr) -> bool>;
39
40fn copy_tree(source: &Path, target: &Path, external_root: FileLinkSelection<'_>) -> io::Result<()> {
41    let metadata = source.symlink_metadata()?;
42    if !metadata.is_dir() {
43        return Err(io::Error::new(
44            ErrorKind::InvalidInput,
45            "Expected a table directory",
46        ));
47    }
48    copy_directory(source, target, &metadata, external_root)
49}
50
51fn copy_directory(
52    source: &Path,
53    target: &Path,
54    metadata: &Metadata,
55    external_root: FileLinkSelection<'_>,
56) -> io::Result<()> {
57    fs::create_dir(target)?;
58    #[cfg(unix)]
59    let is_table =
60        external_root.is_none() && source.join(crate::storage::TABLE_CONTROL_FILE).is_file();
61    for entry in fs::read_dir(source)? {
62        let entry = entry?;
63        #[cfg(unix)]
64        if is_table && entry.file_name() == LOCK_FILE_NAME {
65            copy_table_lock(source, target)?;
66            continue;
67        }
68        let path = entry.path();
69        let output = target.join(entry.file_name());
70        if external_root.is_some_and(|follow| follow(&entry.file_name()))
71            && entry.file_type()?.is_symlink()
72            && path.is_file()
73        {
74            copy_entry(&path.canonicalize()?, &output)?;
75        } else {
76            copy_entry(&path, &output)?;
77        }
78    }
79    fs::set_permissions(target, metadata.permissions())
80}

82#[cfg(unix)]
83fn copy_table_lock(source: &Path, target: &Path) -> io::Result<()> {
84    // Opening and closing a second descriptor on a POSIX lock inode releases
85    // this process's locks. The reader uses the registered descriptor instead.
86    let sync = read_sync_data_from_table_dir(source)?;
87    let lock = LockFile::create_or_open(
88        target,
89        true,
90        LockOptions::default().inspection_interval,
91        false,
92    )?;
93    if let Some(sync) = sync {
94        lock.write_sync_data(&sync)?;
95    }
96    fs::set_permissions(
97        target.join(LOCK_FILE_NAME),
98        source.join(LOCK_FILE_NAME).metadata()?.permissions(),
99    )
100}

102fn copy_entry(source: &Path, target: &Path) -> io::Result<()> {
103    let metadata = source.symlink_metadata()?;
104    if metadata.is_symlink() {
105        #[cfg(unix)]
106        return std::os::unix::fs::symlink(source.read_link()?, target);
107        #[cfg(not(unix))]
108        return Err(io::Error::new(
109            ErrorKind::Unsupported,
110            "Copying symbolic links is unsupported",
111        ));
112    }
113    if metadata.is_dir() {
114        return copy_directory(source, target, &metadata, None);
115    }
116    if metadata.is_file() {
117        #[cfg(any(target_os = "linux", target_os = "macos"))]
118        {
119            use std::os::unix::fs::MetadataExt;
120            if metadata.blocks() < metadata.len() / 512 && copy_sparse(source, target, &metadata)? {
121                return Ok(());
122            }
123        }
124        return fs::copy(source, target).map(|_| ());
125    }
126    Err(io::Error::new(
127        ErrorKind::Unsupported,
128        format!("Cannot copy table entry {}", source.display()),
129    ))
130}

132#[cfg(any(target_os = "linux", target_os = "macos"))]
133fn copy_sparse(source: &Path, target: &Path, metadata: &Metadata) -> io::Result<bool> {
134    use std::fs::File;
135    use std::io::{Read, Seek, SeekFrom};
136
137    let mut input = File::open(source)?;
138    let mut offset = 0;
139    let mut data = next_extent(&input, offset, libc::SEEK_DATA);
140    if matches!(&data, Err(error) if matches!(error.raw_os_error(), Some(libc::EINVAL | libc::ENOTSUP)))
141    {
142        return Ok(false);
143    }
144    let mut output = File::create_new(target)?;
145    #[cfg(target_os = "macos")]
146    let block_size = destination_block_size(&output)?;
147    while let Some(start) = data? {
148        let end = next_extent(&input, start, libc::SEEK_HOLE)?.unwrap_or(metadata.len());
149        input.seek(SeekFrom::Start(start))?;
150        output.seek(SeekFrom::Start(start))?;
151        if io::copy(&mut (&mut input).take(end - start), &mut output)? != end - start {
152            return Err(io::Error::new(
153                ErrorKind::UnexpectedEof,
154                "Table extent changed during copy",
155            ));
156        }
157        #[cfg(target_os = "macos")]
158        preserve_sparse_gap(&output, offset, start, block_size)?;
159        offset = end;
160        data = next_extent(&input, offset, libc::SEEK_DATA);
161    }
162    output.set_len(metadata.len())?;
163    output.set_permissions(metadata.permissions())?;
164    Ok(true)
165}

167#[cfg(target_os = "macos")]
168fn destination_block_size(file: &std::fs::File) -> io::Result<u64> {
169    use std::mem::MaybeUninit;
170    use std::os::fd::AsRawFd;
171
172    let mut stats = MaybeUninit::<libc::statfs>::uninit();
173    // SAFETY: fstatfs writes the complete struct on success and the fd is live.
174    let block_size = unsafe {
175        if libc::fstatfs(file.as_raw_fd(), stats.as_mut_ptr()) != 0 {
176            return Err(io::Error::last_os_error());
177        }
178        u64::from(stats.assume_init().f_bsize)
179    };
180    if block_size == 0 {
181        return Err(io::Error::other("Destination filesystem has no block size"));
182    }
183    Ok(block_size)
184}

186#[cfg(target_os = "macos")]
187fn preserve_sparse_gap(
188    file: &std::fs::File,
189    start: u64,
190    end: u64,
191    block_size: u64,
192) -> io::Result<()> {
193    use std::os::fd::AsRawFd;
194
195    let start = start.next_multiple_of(block_size);
196    let end = end / block_size * block_size;
197    if start >= end {
198        return Ok(());
199    }
200    // APFS can allocate skipped ranges when the copied file is closed.
201    // Punch only ranges that the source filesystem reported as holes.
202    let gap = libc::fpunchhole_t {
203        fp_flags: 0,
204        reserved: 0,
205        fp_offset: start as libc::off_t,
206        fp_length: (end - start) as libc::off_t,
207    };
208    // SAFETY: the descriptor is live and the argument uses Darwin's ABI.
209    if unsafe { libc::fcntl(file.as_raw_fd(), libc::F_PUNCHHOLE, &gap) } == 0 {
210        return Ok(());
211    }
212    let error = io::Error::last_os_error();
213    if matches!(
214        error.raw_os_error(),
215        Some(libc::EINVAL | libc::ENOTSUP | libc::EOPNOTSUPP | libc::ENOTTY)
216    ) {
217        return Ok(());
218    }
219    Err(error)
220}

222#[cfg(any(target_os = "linux", target_os = "macos"))]
223fn next_extent(file: &std::fs::File, offset: u64, whence: libc::c_int) -> io::Result<Option<u64>> {
224    use std::os::fd::AsRawFd;
225
226    // SAFETY: this is a live file descriptor and does not outlive the call.
227    let result = unsafe { libc::lseek(file.as_raw_fd(), offset as libc::off_t, whence) };
228    if result >= 0 {
229        return Ok(Some(result as u64));
230    }
231    let error = io::Error::last_os_error();
232    if error.raw_os_error() == Some(libc::ENXIO) {
233        return Ok(None);
234    }
235    Err(error)
236}

249    #[cfg(target_os = "macos")]
250    #[test]
251    fn sparse_gap_rounds_inward_without_punching_adjacent_data() {
252        let root = tempfile::tempdir().unwrap();
253        let path = root.path().join("destination");
254        let file = fs::OpenOptions::new()
255            .read(true)
256            .write(true)
257            .create_new(true)
258            .open(&path)
259            .unwrap();
260        let block = destination_block_size(&file).unwrap() * 4;
261        let mut expected = vec![0; (block * 5) as usize];
262        expected[block as usize] = 0x9b;
263        expected[(4 * block - 1) as usize] = 0xd2;
264        fs::write(&path, &expected).unwrap();
265
266        preserve_sparse_gap(&file, block + 1, 4 * block - 1, block).unwrap();
267        assert_eq!(fs::read(&path).unwrap(), expected);
268        assert_eq!(
269            next_extent(&file, 2 * block, libc::SEEK_HOLE).unwrap(),
270            Some(2 * block)
271        );
272    }

274    #[test]
275    fn copy_preserves_source_locks_and_subtable_sync_data() {
276        let root = tempfile::tempdir().unwrap();
277        let source = root.path().join("image");
278        let target = root.path().join("copy");
279        fs::create_dir_all(source.join("logtable")).unwrap();
280        let mut guards = Vec::new();
281        for (subdir, rows) in [("", 1), ("logtable", 7)] {
282            let table = source.join(subdir);
283            fs::write(table.join("table.dat"), [0u8]).unwrap();
284            let mut guard = open_lock(&table);
285            assert!(guard.acquire(LockType::Write, 1).unwrap().is_acquired());
286            let mut sync = SyncData::new();
287            sync.record_write(rows, 1, true, &[true]);
288            guard.write_sync_data(&sync).unwrap();
289            guards.push(guard);
290        }
291
292        copy_locked_table_tree(&source, &target).unwrap();
293        for (subdir, rows) in [("", 1), ("logtable", 7)] {
294            let sync = read_sync_data_from_table_dir(&target.join(subdir))
295                .unwrap()
296                .unwrap();
297            assert_eq!(sync.nrrow, rows);
298        }
299        let child = Command::new(std::env::current_exe().unwrap())
300            .args([
301                "--exact",
302                "publication::copy_tree::tests::source_lock_probe",
303            ])
304            .env(SOURCE_PROBE, &source)
305            .output()
306            .unwrap();
307        let output = String::from_utf8_lossy(&child.stdout);
308        assert!(
309            child.status.success() && output.contains("1 passed"),
310            "{output}"
311        );
312        drop(guards);
313    }

315    #[test]
316    fn source_lock_probe() {
317        let Some(path) = std::env::var_os(SOURCE_PROBE) else {
318            return;
319        };
320        for subdir in ["", "logtable"] {
321            let mut lock = open_lock(&Path::new(&path).join(subdir));
322            assert_eq!(
323                lock.acquire(LockType::Write, 1).unwrap(),
324                LockOutcome::HeldByAnotherProcess,

326        }
327    }
