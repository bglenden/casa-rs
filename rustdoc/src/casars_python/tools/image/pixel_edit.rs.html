3//! Thin PyO3 `ia.set` wrapper delegating to [`casa_imageanalysis::tool::ops_edit::set_pixels_and_mask`].
4
5use casa_imageanalysis::tool::ops_edit::set_pixels_and_mask;
6use pyo3::prelude::*;
7
8use crate::convert::Variant;
9
10use super::PyImageTool;
11use super::handle::{emit_tool_logs, raised};
12use super::subimage::Region;
13
14fn pixel_expression(pixels: &Variant, formatted_pixels: &str) -> PyResult<Option<String>> {
15    if pixels.is_empty_sequence() {
16        return Ok(None);
17    }
18    match pixels {
19        Variant::Number(value) if value.is_finite() => Ok(Some(formatted_pixels.to_owned())),
20        Variant::Text(text) if text.is_empty() || text == "[]" => Ok(None),
21        Variant::Texts(_)
22        | Variant::Numbers(_)
23        | Variant::Complexes(_)
24        | Variant::Array { .. }
25        | Variant::ComplexArray { .. } => Err(raised("The pixels expression must be scalar")),
26        _ => Ok(Some(pixels.describe())),
27    }
28}
29
30#[pymethods]
31impl PyImageTool {
32    #[pyo3(signature = (pixels=None, pixelmask=-1, region=None, list=false), text_signature = "($self, pixels=[], pixelmask=-1, region={}, list=False)")]
33    fn set(
34        &self,
35        py: Python<'_>,
36        pixels: Option<&Bound<'_, PyAny>>,
37        pixelmask: i64,
38        region: Option<&Bound<'_, PyAny>>,
39        list: bool,
40    ) -> PyResult<bool> {
41        if self.state().image.is_none() {
42            return Ok(false);
43        }
44        let variant = pixels
45            .map(|value| value.extract::<Variant>())
46            .transpose()?
47            .unwrap_or(Variant::Empty);
48        let pixel_history = match &variant {
49            Variant::Number(value) if pixels.is_some_and(|v| v.extract::<i64>().is_err()) => {
50                format!("{value:.6}")
51            }
52            Variant::Text(text) => format!("\"{text}\""),
53            other => other.describe(),
54        };
55        let expr = pixel_expression(&variant, &pixel_history)?;
56        let region = Region::argument(py, region, "set")?;
57        let (outcome, logs) = py.detach(|| -> PyResult<_> {
58            let state = self.state();
59            let Some(attachment) = &state.image else {
60                return Ok((false, Vec::new()));
61            };
62            let history_command = state.history.then(|| {
63                let region_hist = region.as_ref().map_or_else(|| "{}".into(), Region::history);
64                format!("ia.set(pixels={pixel_history}, pixelmask={pixelmask}, region={region_hist}, list={list})")
65            });
66            set_pixels_and_mask(
67                attachment,
68                expr.as_deref(),
69                pixelmask,
70                region.as_ref().map(Region::as_region),
71                list,
72                history_command,
73            )
74            .map_err(raised)
75        })?;
76        emit_tool_logs(py, &logs)?;
77        Ok(outcome)
78    }
79}
