2//! The minor cycle of one major-cycle completion: its planes read from the
3//! normal state as deconvolution views, solved on the worker team, and
4//! their components returned as model terms.
5//!
6//! A minor cycle has two halves around the run's [`Controller`]:
7//! [`prepare_minor_cycle`] forms the masks and measures every plane (CASA's
8//! `initminorcycle`), and [`run_minor_cycle`] cleans every plane under the
9//! controls the controller derived from those measurements.
10//!
11//! Each half reads its planes from the normal state itself, so a plane is
12//! read twice per cycle. A paged cube is not held in memory between the
13//! halves; only the planes in flight on the team are.
14//!
15//! [`Controller`]: casa_imaging_deconvolution::Controller
16

44/// A failure of the minor cycle.
45#[derive(Debug, thiserror::Error)]
46pub enum MinorCycleRunError {
47    /// The reconstruction masks could not be formed.
48    #[error("reconstruction mask: {0}")]
49    Mask(#[from] MaskError),
50    /// A normal-state plane could not be read.
51    #[error("normal state: {0}")]
52    Normal(#[from] NormalStateError),
53    /// The model support could not be read or a term is out of range.
54    #[error("model: {0}")]
55    Model(#[from] ModelLifecycleError),
56    /// The direction-dependent normalization could not convert a value.
57    #[error("normalization: {0}")]
58    Response(#[from] ProductsError),
59    /// The solver failed.
60    #[error(transparent)]
61    Solver(#[from] casa_imaging_deconvolution::Error),
62    /// The selected algorithm has no minor cycle.
63    #[error("{0:?} has no minor cycle")]
64    Algorithm(ReconstructionAlgorithm),
65    /// A plane's terms do not fit what the policy leaves free.
66    #[error(transparent)]
67    Admission(#[from] Admission),
68}

105impl PsfCache {
106    /// Heap bytes the cached Clark refresh holds, which outlives each minor
107    /// cycle ([`LinearRefresh::bytes`]); zero when none is cached.
108    #[must_use]
109    pub fn refresh_bytes(&self, completion: &MajorCycleCompletion) -> u64 {
110        self.clark.as_ref().map_or(0, |_| {
111            let [nx, ny] = completion.normal_state().shape();
112            LinearRefresh::bytes(PlaneShape::new(nx, ny))
113        })
114    }
115}

117/// Heap bytes one loaded plane holds ([`load`]): its residual and PSF terms
118/// (`f64`), its support and validity, its sensitivity under a response, and
119/// what it is read through (the normal state's windows as `Complex64`, one
120/// model plane).
121fn load_bytes(cells: u64, residual_terms: u64, psf_terms: u64, response: bool) -> u64 {
122    let terms = residual_terms + psf_terms;
123    cells
124        * (terms * size_of::<f64>() as u64
125            + 2 * size_of::<bool>() as u64
126            + if response { size_of::<f64>() as u64 } else { 0 }
127            + terms * size_of::<num_complex::Complex64>() as u64
128            + size_of::<casa_imaging_model::ModelSample>() as u64)
129}

131/// The planes a minor cycle of `normal` loads, their largest cell count and
132/// their residual and PSF term counts.
133fn plane_layout(normal: &FinalNormalState) -> (Vec<PlaneKey>, u64, u64, u64) {
134    let taylor = is_taylor(normal);
135    let keys = plane_keys(normal, taylor);
136    let cells = keys
137        .iter()
138        .map(|key| plane_shape(normal, *key).len() as u64)
139        .max()
140        .unwrap_or(0);
141    let (residual, psf) = if taylor {
142        (
143            normal.coefficient_term_count() as u64,
144            normal.normal_moment_count() as u64,
145        )
146    } else {
147        (1, 1)
148    };
149    (keys, cells, residual, psf)
150}

152/// Heap bytes [`prepare_minor_cycle`] holds at most: the masks it
153/// materializes, and on each of the planes loaded at once on `workers`, the
154/// load and its measurement (the robust noise's two copies under `nsigma`,
155/// the PSF fit's single-precision copies for a plane not yet in `cache`).
156#[must_use]
157pub fn prepare_bytes(
158    completion: &MajorCycleCompletion,
159    mask_plans: &ImageDomainReconstructionMaskPlans,
160    setup: &MinorCycleSetup,
161    cache: &PsfCache,
162    workers: usize,
163) -> u64 {
164    let normal = completion.normal_state();
165    let (keys, cells, residual, psf) = plane_layout(normal);
166    let fits = keys.iter().any(|key| !cache.summaries.contains_key(key));
167    let measure = cells
168        * if setup.nsigma > 0.0 {
169            2 * size_of::<f64>() as u64
170        } else {
171            0
172        }
173        .max(if fits { 2 * size_of::<f32>() as u64 } else { 0 });
174    let concurrent = workers.min(keys.len()).max(1) as u64;
175    mask_plans.materialize_bytes(normal)
176        + concurrent * (load_bytes(cells, residual, psf, setup.response.is_some()) + measure)
177}

179/// Heap bytes the masks a minor cycle forms keep while the cycle runs and
180/// after it: one support per image domain.
181#[must_use]
182pub fn mask_bytes(completion: &MajorCycleCompletion) -> u64 {
183    let normal = completion.normal_state();
184    (0..normal.domain_count())
185        .filter_map(|domain| normal.domain_shape(domain))
186        .map(|[width, height]| (width * height * size_of::<bool>()) as u64)
187        .sum()
188}

190/// Bytes the model terms of one solved plane hold, charged as the plane
191/// finishes ([`run_minor_cycle`]): for each of `terms` terms its record and
192/// its copy while the cycle merges every plane's terms in model order,
193/// which also covers what the model holds per term while it queues them
194/// ([`ModelLifecycle::QUEUED_TERM_BYTES`]); and the plane's component
195/// records.
196#[must_use]
197pub const fn result_bytes(terms: usize) -> u64 {
198    let entry = size_of::<(usize, ModelDeltaTerm)>();
199    let merged = 2 * entry;
200    let queued = size_of::<ModelDeltaTerm>() + ModelLifecycle::QUEUED_TERM_BYTES;
201    let per_term = if merged > queued { merged } else { queued };
202    (terms * per_term + TRACE * (size_of::<Component>() + size_of::<TracedComponent>())) as u64
203}
204
205/// Heap bytes [`run_minor_cycle`] holds at most under `controls`, besides
206/// each plane's terms ([`result_bytes`]): on each of the planes solved at
207/// once on `workers` (one when a single plane spreads over the team), the
208/// load and the solve ([`casa_imaging_deconvolution::solve_bytes`]), and a
209/// record of every plane. A Clark refresh already in `cache` is charged
210/// with the cache, not here.
211#[must_use]
212pub fn run_bytes(
213    completion: &MajorCycleCompletion,
214    setup: &MinorCycleSetup,
215    controls: &CycleControls,
216    cache: &PsfCache,
217    workers: usize,
218) -> u64 {
219    let normal = completion.normal_state();
220    let (keys, cells, residual, psf) = plane_layout(normal);
221    let shape = keys
222        .iter()
223        .map(|key| plane_shape(normal, *key))
224        .max_by_key(|shape| shape.len())
225        .unwrap_or(PlaneShape::new(0, 0));
226    let terms = residual as usize;
227    let components = controls.iterations + 1;
228    let solve = match &setup.algorithm {
229        ReconstructionAlgorithm::Hogbom => casa_imaging_deconvolution::solve_bytes(
230            &Hogbom::new(false),
231            shape,
232            terms,
233            components,
234            TRACE,
235        ),
236        ReconstructionAlgorithm::Clark => {
237            let solve = casa_imaging_deconvolution::solve_bytes(
238                &Clark::new(None),
239                shape,
240                terms,
241                components,
242                TRACE,
243            );
244            if keys.len() == 1 {
245                solve.saturating_sub(cache.refresh_bytes(completion))
246            } else {
247                solve
248            }
249        }
250        ReconstructionAlgorithm::Multiscale {
251            scales_px,
252            small_scale_bias,
253        } => casa_imaging_deconvolution::solve_bytes(
254            &Multiscale::new(scales_px.clone(), *small_scale_bias),
255            shape,
256            terms,
257            components,
258            TRACE,
259        ),
260        ReconstructionAlgorithm::Mtmfs {
261            scales_px,
262            small_scale_bias,
263        } => casa_imaging_deconvolution::solve_bytes(
264            &Taylor::new(terms, scales_px.clone(), *small_scale_bias),
265            shape,
266            terms,
267            components,
268            TRACE,
269        ),
270        _ => 0,
271    };
272    let concurrent = if keys.len() == 1 {
273        1
274    } else {
275        workers.min(keys.len()) as u64
276    };
277    // Every plane's slot, solved record and stop, which the cycle keeps
278    // until it merges them.
279    let records = keys.len() * (size_of::<Slot>() + size_of::<Solved>() + size_of::<PlaneStop>());
280    concurrent * (load_bytes(cells, residual, psf, setup.response.is_some()) + solve)
281        + records as u64
282}

295/// The load of a cycle's only plane, without a direction-dependent
296/// response: preparation measures it and the solve cleans it, so it is read
297/// once a cycle.
298struct RetainedPlane {
299    key: PlaneKey,
300    residual: Vec<Vec<f64>>,
301    psf: Vec<Vec<f64>>,
302    support: Support,
303    valid: Support,
304}
305
306impl PreparedMinorCycle {
307    /// The masks the cycle cleans within.
308    #[must_use]
309    pub const fn masks(&self) -> &ImageDomainReconstructionMasks {
310        &self.masks
311    }
312
313    /// Automatic-mask diagnostics per image domain.
314    #[must_use]
315    pub fn auto_masks(&self) -> &[Option<AutoMultithreshEvidence>] {
316        &self.auto_masks
317    }
318
319    /// Bytes the plane kept for the solve holds: its residual and PSF terms
320    /// and its two supports, within what [`prepare_bytes`] charged for its
321    /// load. The cycle's [`run_bytes`] charges a load the solve then skips,
322    /// so a caller moves this much of the preparation's charge to the solve
323    /// and charges the solve this much less.
324    #[must_use]
325    pub fn retained_bytes(&self) -> u64 {
326        self.retained.as_ref().map_or(0, |plane| {
327            let values = plane
328                .residual
329                .iter()
330                .chain(&plane.psf)
331                .map(Vec::len)
332                .sum::<usize>();
333            (values * size_of::<f64>() + 2 * plane.support.shape().len() * size_of::<bool>()) as u64
334        })
335    }
336}

397/// Form the masks of `completion` and measure every plane: its peak
398/// residual, noise and PSF sidelobe (CASA's `initminorcycle`).
399///
400/// # Errors
401///
402/// When a mask, a plane or a PSF cannot be read or measured.
403pub fn prepare_minor_cycle(
404    completion: &MajorCycleCompletion,
405    mask_plans: &ImageDomainReconstructionMaskPlans,
406    setup: &MinorCycleSetup,
407    cache: &mut PsfCache,
408    team: &WorkerTeam,
409) -> Result<PreparedMinorCycle, MinorCycleRunError> {
410    let normal = completion.normal_state();
411    let base = completion.final_model();
412    let taylor = is_taylor(normal);
413    let keys = plane_keys(normal, taylor);
414    // The automatic mask (one-channel continuum only) smooths by the
415    // primary PSF's beam; a cube's first channel may hold no data.
416    let beam = if setup.automask {
417        let primary = keys[0];
418        summarise(normal, taylor, cache, primary)?;
419        Some(automask_beam(&cache.summaries[&primary]))
420    } else {
421        None
422    };
423    let (masks, auto_masks) = mask_plans.materialize(base, normal, beam)?.into_parts();
424    let inputs = Inputs {
425        normal,
426        base,
427        masks: &masks,
428        setup,
429        taylor,
430    };
431    let mut slots = keys
432        .iter()
433        .map(|key| (*key, None::<Measured>, None::<RetainedPlane>))
434        .collect::<Vec<_>>();
435    let missing = keys
436        .iter()
437        .filter(|key| !cache.summaries.contains_key(key))
438        .count();
439    // A cycle's only plane is solved straight after it is measured, so its
440    // load is kept for the solve rather than read twice.
441    let retain = keys.len() == 1 && setup.response.is_none();
442    let shared = &*cache;
443    team.for_each_mut(&mut slots, |_, (key, out, kept)| {
444        let Some(plane) = load(&inputs, *key)? else {
445            return Ok(());
446        };
447        let summary = match shared.summaries.get(key) {
448            Some(summary) => *summary,
449            None => PsfSummary::new(&plane.psf[0], plane.shape)?,
450        };
451        let statistics = PlaneStatistics::measure(
452            &plane.residual[0],
453            &plane.support,
454            &plane.valid,
455            setup.nsigma,
456            setup.automask,
457        );
458        *out = Some(Measured {
459            statistics,
460            summary,
461        });
462        if retain && plane.response.is_none() {
463            *kept = Some(RetainedPlane {
464                key: *key,
465                residual: plane.residual,
466                psf: plane.psf,
467                support: plane.support,
468                valid: plane.valid,
469            });
470        }
471        Ok::<_, MinorCycleRunError>(())
472    })?;
473    let mut planes = Vec::with_capacity(keys.len());
474    let mut sidelobes = Vec::with_capacity(keys.len());
475    let mut retained = None;
476    for (key, measured, kept) in slots {
477        let Some(Measured {
478            statistics,
479            summary,
480        }) = measured
481        else {
482            continue;
483        };
484        sidelobes.push(summary.sidelobe());
485        cache.summaries.entry(key).or_insert(summary);
486        planes.push((key, statistics));
487        retained = retained.or(kept);
488    }
489    // Each image domain is one CASA image store.
490    let mut statistics = ResidualStatistics::empty();
491    let domains = planes
492        .iter()
493        .map(|(key, _)| key.domain)
494        .collect::<BTreeSet<_>>();
495    for domain in domains {
496        statistics.include_store(
497            planes
498                .iter()
499                .zip(&sidelobes)
500                .filter(|((key, _), _)| key.domain == domain)
501                .map(|((_, plane), sidelobe)| (plane, *sidelobe)),
502            setup.nsigma,
503        );
504    }
505    tracing::debug!(
506        planes = planes.len(),
507        psf_fits = missing,
508        "minor cycle prepared"
509    );
510    Ok(PreparedMinorCycle {
511        masks,
512        auto_masks,
513        planes,
514        retained,
515        statistics,
516    })
517}

526/// Clean every plane of `completion` under `controls`.
527///
528/// Planes are independent: several run concurrently on `team`, each on one
529/// thread, and their terms are merged in model order, so the result does
530/// not depend on the worker count. A single plane spreads Clark's sparse
531/// refresh across the team instead. As each plane finishes, its terms are
532/// admitted under `policy` on `host` ([`result_bytes`]) before they are
533/// formed; the outcome hands their charge on.
534///
535/// # Errors
536///
537/// When a plane cannot be read, its solver fails or its terms are refused.
538pub fn run_minor_cycle(
539    prepared: PreparedMinorCycle,
540    completion: &MajorCycleCompletion,
541    setup: &MinorCycleSetup,
542    controls: &CycleControls,
543    cache: &mut PsfCache,
544    team: &WorkerTeam,
545    (host, policy): (&HostResources, &ResourcePolicy),
546) -> Result<MinorCycleOutcome, MinorCycleRunError> {
547    let PreparedMinorCycle {
548        masks,
549        auto_masks,
550        planes,
551        retained,
552        ..
553    } = prepared;
554    let normal = completion.normal_state();
555    let inputs = Inputs {
556        normal,
557        base: completion.final_model(),
558        masks: &masks,
559        setup,
560        taylor: is_taylor(normal),
561    };
562    let admission = (host, policy);
563    let mut results = Vec::with_capacity(planes.len());
564    if let [(key, statistics)] = planes.as_slice() {
565        let summary = cache.summaries[key];
566        let clark = &mut cache.clark;
567        results.push(team.install(|| {
568            solve(
569                &inputs,
570                admission,
571                (*key, retained),
572                &summary,
573                statistics,
574                controls,
575                clark,
576            )
577        })?);
578    } else {
579        let mut slots = planes
580            .iter()
581            .map(|(key, statistics)| Slot {
582                key: *key,
583                statistics: *statistics,
584                solved: None,
585            })
586            .collect::<Vec<_>>();
587        let shared = &*cache;
588        team.for_each_mut(&mut slots, |_, slot| {
589            let summary = shared.summaries[&slot.key];
590            slot.solved = Some(solve(
591                &inputs,
592                admission,
593                (slot.key, None),
594                &summary,
595                &slot.statistics,
596                controls,
597                &mut None,
598            )?);
599            Ok::<_, MinorCycleRunError>(())
600        })?;
601        results.extend(
602            slots
603                .into_iter()
604                .map(|slot| slot.solved.expect("every plane slot ran")),
605        );
606    }
607    let mut summary = MinorCycleSummary {
608        iterations: 0,
609        components: 0,
610        absolute_flux: 0.0,
611        start_peak: 0.0,
612        peak: 0.0,
613        noise_rms: None,
614        stops: Vec::with_capacity(results.len()),
615        refreshes: 0,
616        trace: Vec::new(),
617    };
618    // Each plane's charge covers its terms' copy here.
619    let mut terms = Vec::with_capacity(results.iter().map(|solved| solved.terms.len()).sum());
620    let mut charge = Reservation::none();
621    for ((_, statistics), solved) in planes.iter().zip(results) {
622        charge.join(solved.charge);
623        let outcome = &solved.outcome;
624        summary.iterations += outcome.iterations;
625        summary.components += outcome.components;
626        summary.absolute_flux += outcome.absolute_flux;
627        summary.start_peak = summary.start_peak.max(outcome.start_peak);
628        if outcome.stop != PlaneStop::ZeroMask {
629            summary.peak = summary.peak.max(outcome.peak);
630        }
631        if let Some(noise) = statistics.noise {
632            summary.noise_rms = Some(
633                summary
634                    .noise_rms
635                    .map_or(noise.rms, |rms| rms.max(noise.rms)),
636            );
637        }
638        summary.stops.push(outcome.stop);
639        summary.refreshes += outcome.refreshes;
640        let room = TRACE - summary.trace.len();
641        summary.trace.extend(solved.trace.into_iter().take(room));
642        terms.extend(solved.terms);
643    }
644    terms.sort_unstable_by_key(|(index, _)| *index);
645    let mut ordered = Vec::with_capacity(terms.len());
646    ordered.extend(terms.into_iter().map(|(_, term)| term));
647    Ok(MinorCycleOutcome {
648        terms: ordered,
649        results: charge,
650        masks,
651        auto_masks,
652        summary,
653    })
654}

656/// Whether `normal` holds one Taylor family (decided once per cycle half).
657fn is_taylor(normal: &FinalNormalState) -> bool {
658    normal.basis() == NormalBasis::Taylor
659}

661/// The planes a minor cycle cleans, in CASA's order within each image
662/// domain: one Taylor family, every channel of every polarization of a
663/// cube or continuum plane, or each polarization of every image field.
664fn plane_keys(normal: &FinalNormalState, taylor: bool) -> Vec<PlaneKey> {
665    let first = normal.slab().core_range().start;
666    if taylor {
667        return vec![PlaneKey {
668            domain: 0,
669            channel: first,
670            polarization: 0,
671        }];
672    }
673    let mut keys = Vec::new();
674    for polarization in 0..normal.polarization_count() {
675        for domain in 0..normal.domain_count() {
676            for channel in first..first + normal.channel_count() {
677                keys.push(PlaneKey {
678                    domain,
679                    channel,
680                    polarization,
681                });
682            }
683        }
684    }
685    keys
686}

688/// Measure the PSF of `key` into the cache.
689fn summarise(
690    normal: &FinalNormalState,
691    taylor: bool,
692    cache: &mut PsfCache,
693    key: PlaneKey,
694) -> Result<(), MinorCycleRunError> {
695    if cache.summaries.contains_key(&key) {
696        return Ok(());
697    }
698    let shape = plane_shape(normal, key);
699    let psf = read_psf(normal, taylor, key)?;
700    let peak = psf_peak_value(&psf, shape)?;
701    cache
702        .summaries
703        .insert(key, PsfSummary::new(&normalise(psf, peak), shape)?);
704    Ok(())
705}

718fn plane_shape(normal: &FinalNormalState, key: PlaneKey) -> PlaneShape {
719    let [nx, ny] = normal

725/// Term-0 PSF of `key`, unnormalised.
726fn read_psf(
727    normal: &FinalNormalState,
728    taylor: bool,
729    key: PlaneKey,
730) -> Result<Vec<f64>, MinorCycleRunError> {
731    if taylor {
732        let window = normal.read_window(normal.slab().core_range())?;
733        let moment = window
734            .normal_moment(0)
735            .expect("a Taylor family has its zeroth moment");
736        return Ok(moment
737            .normal_approximation()
738            .iter()
739            .map(|value| value.re)
740            .collect());
741    }
742    let plane = normal.read_reconstruction_plane(key.domain, key.channel, key.polarization)?;
743    Ok(plane
744        .normal_approximation()
745        .iter()
746        .map(|value| value.re)
747        .collect())
748}

750/// What every plane of one minor cycle is read from.
751#[derive(Clone, Copy)]
752struct Inputs<'a> {
753    normal: &'a FinalNormalState,
754    base: &'a ModelGeneration,
755    masks: &'a ImageDomainReconstructionMasks,
756    setup: &'a MinorCycleSetup,
757    /// Whether the normal state is one Taylor family.
758    taylor: bool,
759}

761/// A direction-dependent response bound to one plane's weights: what turns
762/// a solved value back into the physical model (`divideModelByWeight`).
763struct ResponseBinding<'a> {
764    sensitivity: Cow<'a, [f64]>,
765    response: MinorCycleImageResponse,
766    sum_weight: f64,
767}
768
769/// One plane's residual and PSF terms as the solver sees them.
770struct PlaneTerms<'a> {
771    residual: Vec<Vec<f64>>,
772    psf: Vec<Vec<f64>>,
773    response: Option<ResponseBinding<'a>>,
774}
775
776/// One plane read as a deconvolution view.
777struct PlaneData<'a> {
778    shape: PlaneShape,
779    residual: Vec<Vec<f64>>,
780    psf: Vec<Vec<f64>>,
781    support: Support,
782    valid: Support,
783    response: Option<ResponseBinding<'a>>,
784}
785
786impl PlaneData<'_> {
787    /// The plane's conversion of solved values to the physical model,
788    /// bound once (binding scans the sensitivity plane).
789    fn physical(&self) -> Result<Physical<'_>, MinorCycleRunError> {
790        Ok(match &self.response {
791            None => Physical::Identity,
792            Some(binding) => Physical::Mosaic(
793                MosaicSensitivity::new(&binding.sensitivity)?
794                    .with_normal_sum_weight(binding.sum_weight)?,
795                &binding.response,
796            ),
797        })
798    }
799}
800
801/// Solved values to the physical model (`divideModelByWeight`).
802enum Physical<'a> {
803    Identity,
804    Mosaic(MosaicSensitivity<'a>, &'a MinorCycleImageResponse),
805}
806
807impl Physical<'_> {
808    fn apply(&self, value: f64, index: usize) -> Result<f64, MinorCycleRunError> {
809        match self {
810            Self::Identity => Ok(value),
811            Self::Mosaic(sensitivity, response) => Ok(sensitivity.apparent_to_physical(
812                value,
813                index,
814                response.normalization(),
815                response.policy(),
816            )?),
817        }
818    }
819}

821/// Read `key` as a view: residual and PSF normalised by the term-0 PSF peak
822/// (or the residual in CASA's flat-noise or flat-sky units under a
823/// direction-dependent response), and the support. `None` for a plane with
824/// no valid data.
825fn load<'a>(
826    inputs: &Inputs<'a>,
827    key: PlaneKey,
828) -> Result<Option<PlaneData<'a>>, MinorCycleRunError> {
829    let Inputs {
830        normal,
831        base,
832        masks,
833        setup,
834        taylor,
835    } = *inputs;
836    let shape = plane_shape(normal, key);
837    let validity = if taylor {
838        normal.support_validity()
839    } else {
840        normal.domain_channel_validity(
841            key.domain,
842            key.channel - normal.slab().core_range().start,
843            key.polarization,
844        )
845    };
846    if validity != Some(SpectralChannelValidity::Valid) {
847        return Ok(None);
848    }
849    let coefficients = if taylor {
850        0..normal.coefficient_term_count()
851    } else {
852        key.channel..key.channel + 1
853    };
854    let mut valid = vec![true; shape.len()];
855    for coefficient in coefficients {
856        let samples = base.read_plane(key.domain, coefficient, key.polarization)?;
857        for (index, supported) in valid.iter_mut().enumerate() {
858            let [x, y] = shape.pixel(index);
859            *supported &= samples[y * shape.nx + x].support() == ModelSupport::Valid;
860        }
861    }
862    let mask = masks
863        .get(key.domain)
864        .expect("a mask for every image domain");
865    let support = mask
866        .support()
867        .iter()
868        .zip(&valid)
869        .map(|(masked, valid)| *masked && *valid)
870        .collect();
871    let terms = if taylor {
872        read_taylor(normal, setup)?
873    } else {
874        read_single(normal, setup, key)?
875    };
876    Ok(Some(PlaneData {
877        shape,
878        residual: terms.residual,
879        psf: terms.psf,
880        support: Support::new(shape, support),
881        valid: Support::new(shape, valid),
882        response: terms.response,
883    }))
884}

886/// One plane's residual and PSF. Without a response both are divided by the
887/// PSF peak; with one the residual takes CASA's normalisation
888/// (`SIImageStore::divideResidualByWeight`) and the components convert back
889/// with `divideModelByWeight`. The residual divides by the published sum of
890/// weights and the weight image by the normal one; for one term both are
891/// the PSF gridding's `.sumwt` (`FTMachine::finalizeToSkyNew` writes it
892/// once), so they cancel as CASA's `divideImageByWeightVal` and
893/// `dividePSFByWeight` do. CASA normalises every (polarization,
894/// channel) plane by that plane's own sums and weight image, so a cube
895/// channel reads its own; a paged cube reads only this plane.
896fn read_single<'a>(
897    normal: &'a FinalNormalState,
898    setup: &MinorCycleSetup,
899    key: PlaneKey,
900) -> Result<PlaneTerms<'a>, MinorCycleRunError> {
901    let plane = normal.read_reconstruction_plane(key.domain, key.channel, key.polarization)?;
902    let shape = plane_shape(normal, key);
903    let psf = plane
904        .normal_approximation()
905        .iter()
906        .map(|value| value.re)
907        .collect::<Vec<_>>();
908    let peak = psf_peak_value(&psf, shape)?;
909    let raw = plane.residual();
910    let Some(response) = setup.response else {
911        return Ok(PlaneTerms {
912            residual: vec![raw.iter().map(|value| value.re / peak).collect()],
913            psf: vec![normalise(psf, peak)],
914            response: None,
915        });
916    };
917    let weights = normal.read_plane(key.domain, key.channel, key.polarization)?;
918    let sensitivity = weights.read_sensitivity()?;
919    let normal_weight = weights.sum_weight();
920    let published_weight = weights.published_sum_weight();
921    let bound = MosaicSensitivity::new(&sensitivity)?.with_normal_sum_weight(normal_weight)?;
922    let residual = raw
923        .iter()
924        .enumerate()
925        .map(|(index, value)| {
926            bound.normalize_weighted_residual_sample(
927                value.re,
928                index,
929                response.normalization(),
930                ResidualFamily::Single,
931                published_weight,
932                response.policy(),
933            )
934        })
935        .collect::<Result<Vec<_>, _>>()?;
936    Ok(PlaneTerms {
937        residual: vec![residual],
938        psf: vec![normalise(psf, peak)],
939        response: Some(ResponseBinding {
940            sensitivity,
941            response,
942            sum_weight: normal_weight,
943        }),
944    })
945}

947/// The Taylor family's residual terms and `2·N_t − 1` PSF moments. Every
948/// moment divides by the zeroth moment's peak, with or without a response
949/// (`SIImageStore::normPSF`, which `SDAlgorithmMSMFS::initializeDeconvolver`
950/// reads, #704). Without a response the residuals divide by the same peak;
951/// with one they take CASA's units and the components convert back over
952/// the zeroth moment's sum weight.
953fn read_taylor(
954    normal: &FinalNormalState,
955    setup: &MinorCycleSetup,
956) -> Result<PlaneTerms<'static>, MinorCycleRunError> {
957    let window = normal.read_window(normal.slab().core_range())?;
958    let [nx, ny] = normal.shape();
959    let shape = PlaneShape::new(nx, ny);
960    let principal = window
961        .normal_moment(0)
962        .expect("a Taylor family has its zeroth moment");
963    let moments = (0..normal.normal_moment_count())
964        .map(|moment| {
965            window
966                .normal_moment(moment)
967                .expect("every normal moment is present")
968                .normal_approximation()
969                .iter()
970                .map(|value| value.re)
971                .collect::<Vec<_>>()
972        })
973        .collect::<Vec<_>>();
974    let raw = (0..normal.coefficient_term_count()).map(|term| {
975        window
976            .coefficient_term(term)
977            .expect("every coefficient term is present")
978            .residual()
979    });
980    let peak = psf_peak_value(&moments[0], shape)?;
981    let psf = moments
982        .into_iter()
983        .map(|moment| normalise(moment, peak))
984        .collect();
985    let Some(response) = setup.response else {
986        return Ok(PlaneTerms {
987            residual: raw
988                .map(|term| term.iter().map(|value| value.re / peak).collect())
989                .collect(),
990            psf,
991            response: None,
992        });
993    };
994    let normal_weight = principal.sum_weight();
995    let published_weight = window.published_sum_weights()[0];
996    let sensitivity = principal.sensitivity().to_vec();
997    let bound = MosaicSensitivity::new(&sensitivity)?.with_normal_sum_weight(normal_weight)?;
998    let residual = raw
999        .map(|term| {
1000            term.iter()
1001                .enumerate()
1002                .map(|(index, value)| {
1003                    bound.normalize_weighted_residual_sample(
1004                        value.re,
1005                        index,
1006                        response.normalization(),
1007                        ResidualFamily::Taylor,
1008                        published_weight,
1009                        response.policy(),
1010                    )
1011                })
1012                .collect::<Result<Vec<_>, _>>()
1013        })
1014        .collect::<Result<Vec<_>, _>>()?;
1015    Ok(PlaneTerms {
1016        residual,
1017        psf,
1018        response: Some(ResponseBinding {
1019            sensitivity: Cow::Owned(sensitivity),
1020            response,
1021            sum_weight: normal_weight,
1022        }),
1023    })
1024}

1026fn psf_peak_value(psf: &[f64], shape: PlaneShape) -> Result<f64, MinorCycleRunError> {
1027    let peak = casa_imaging_deconvolution::psf_peak(psf, shape)
1028        .map(|index| psf[index])
1029        .filter(|value| value.is_finite() && *value > 0.0)
1030        .ok_or(casa_imaging_deconvolution::Error::PsfPeak)?;
1031    Ok(peak)
1032}

1052/// Read and clean one plane, or clean the load preparation kept for it, and
1053/// admit its terms on `host` under `policy` before forming them. Clark
1054/// takes `clark`'s refresh, when it serves this PSF, and leaves its own
1055/// there for the next cycle.
1056#[allow(clippy::too_many_arguments)]
1057fn solve(
1058    inputs: &Inputs<'_>,
1059    (host, policy): (&HostResources, &ResourcePolicy),
1060    (key, retained): (PlaneKey, Option<RetainedPlane>),
1061    summary: &PsfSummary,
1062    statistics: &PlaneStatistics,
1063    controls: &CycleControls,
1064    clark: &mut Option<LinearRefresh>,
1065) -> Result<Solved, MinorCycleRunError> {
1066    let setup = inputs.setup;
1067    let plane = match retained {
1068        Some(kept) if kept.key == key => PlaneData {
1069            shape: kept.support.shape(),
1070            residual: kept.residual,
1071            psf: kept.psf,
1072            support: kept.support,
1073            valid: kept.valid,
1074            response: None,
1075        },
1076        _ => load(inputs, key)?.expect("prepared planes have valid data"),
1077    };
1078    let view = MinorCycleView {
1079        shape: plane.shape,
1080        residual: &plane.residual,
1081        psf: &plane.psf,
1082        summary,
1083        support: &plane.support,
1084    };
1085    let outcome = match &setup.algorithm {
1086        ReconstructionAlgorithm::Hogbom => clean(
1087            &Hogbom::new(setup.accounting == HogbomIterationAccounting::CasaInclusive),
1088            &view,
1089            controls,
1090            statistics,
1091        )?,
1092        ReconstructionAlgorithm::Clark => {
1093            let solver = Clark::new(clark.take());
1094            let outcome = clean(&solver, &view, controls, statistics);
1095            *clark = solver.into_refresh();
1096            outcome?
1097        }
1098        ReconstructionAlgorithm::Multiscale {
1099            scales_px,
1100            small_scale_bias,
1101        } => clean(
1102            &Multiscale::new(scales_px.clone(), *small_scale_bias),
1103            &view,
1104            controls,
1105            statistics,
1106        )?,
1107        ReconstructionAlgorithm::Mtmfs {
1108            scales_px,
1109            small_scale_bias,
1110        } => clean(
1111            &Taylor::new(plane.residual.len(), scales_px.clone(), *small_scale_bias),
1112            &view,
1113            controls,
1114            statistics,
1115        )?,
1116        algorithm => return Err(MinorCycleRunError::Algorithm(algorithm.clone())),
1117    };
1118    let mut outcome = outcome;
1119    let delta = std::mem::take(&mut outcome.delta);
1120    let entries = (0..delta.term_count())
1121        .map(|term| delta.term(term).count())
1122        .sum();
1123    let charge = admit(
1124        host,
1125        policy,
1126        &Demand {
1127            phase: "minor-cycle results",
1128            memory: result_bytes(entries),
1129        },
1130    )?;
1131    let terms = model_terms(&delta, entries, &plane, inputs, key)?;
1132    let trace = trace(&outcome.trace, &plane, inputs, key);
1133    Ok(Solved {
1134        outcome,
1135        terms,
1136        trace,
1137        charge,
1138    })
1139}

1152/// the channel of a cube plane.

1157/// The plane's components as model terms keyed by model order, in a vector
1158/// with room for `delta`'s `entries`.
1159fn model_terms(
1160    delta: &Delta,
1161    entries: usize,
1162    plane: &PlaneData,
1163    inputs: &Inputs<'_>,
1164    key: PlaneKey,
1165) -> Result<Vec<(usize, ModelDeltaTerm)>, MinorCycleRunError> {
1166    let mut terms = Vec::with_capacity(entries);
1167    let physical = plane.physical()?;
1168    for term in 0..delta.term_count() {
1169        for (index, flux) in delta.term(term) {
1170            let value = physical.apply(flux, index)?;
1171            if value == 0.0 {
1172                continue;
1173            }
1174            let cell = ModelCell::new(
1175                key.domain,
1176                coefficient(inputs, key, term),
1177                key.polarization,
1178                plane.shape.pixel(index),
1179            );
1180            let flat = inputs
1181                .base
1182                .shape()
1183                .flat_index(cell)
1184                .ok_or(ModelLifecycleError::CellOutsideShape)?;
1185            terms.push((
1186                flat,
1187                ModelDeltaTerm::new(
1188                    cell,
1189                    ModelValue::new(value).map_err(ModelLifecycleError::Contract)?,
1190                ),
1191            ));
1192        }
1193    }
1194    Ok(terms)
1195}

1197fn trace(
1198    components: &[Component],
1199    plane: &PlaneData,
1200    inputs: &Inputs<'_>,
1201    key: PlaneKey,
1202) -> Vec<TracedComponent> {
1203    let scales = match &inputs.setup.algorithm {
1204        ReconstructionAlgorithm::Multiscale { scales_px, .. }
1205        | ReconstructionAlgorithm::Mtmfs { scales_px, .. } => scales_px.as_slice(),
1206        _ => &[],
1207    };
1208    components
1209        .iter()
1210        .map(|component| TracedComponent {
1211            cell: ModelCell::new(
1212                key.domain,
1213                coefficient(inputs, key, component.term),
1214                key.polarization,
1215                plane.shape.pixel(component.index),
1216            ),
1217            flux: component.flux,
1218            scale_px: scales.get(component.scale).copied().unwrap_or(0.0),
1219        })
1220        .collect()
1221}
