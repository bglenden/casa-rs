2//! casacore `Regex` as antenna and spectral-window selection use it.
3//!
4//! Both match names with `String::matches(Regex)`, a full
5//! match (`casa/BasicSL/String.cc:367-372`, `std::regex_match`). The regex
6//! text first passes through `Regex::fromPattern` when it came from a
7//! shell-style pattern (`casa/Utilities/Regex.cc:228-367`), and always through
8//! `Regex::toEcma` (`Regex.cc:146-226`, applied by the `Regex(const String&)`
9//! constructor, `Regex.cc:41-58`) before `std::regex` compiles it with the
10//! ECMAScript grammar. Both conversions are ported byte for byte.
11//!
12//! The ECMAScript result is then translated to the `regex` crate's syntax
13//! and matched over bytes with Unicode disabled, as `std::regex` over
14//! `char` matches bytes. ECMAScript features the `regex` crate lacks
15//! (backreferences, lookahead) are refused rather than approximated.
16
17use regex::bytes::{Regex, RegexBuilder};
18
19/// A compiled casacore regex that answers `String::matches`: whether the
20/// whole of a string matches.
21#[derive(Debug)]
22pub(in crate::selection) struct FullMatch(Regex);
23
24impl FullMatch {
25    /// Compile casacore regex text: what `Regex(const String&)` receives,
26    /// before its `toEcma` conversion.
27    ///
28    /// The `Err` text is the reason the expression cannot be compiled, as
29    /// casacore's `AipsError("Error in regex ...")` (`Regex.cc:55-57`) or an
30    /// ECMAScript feature this port does not support.
31    pub(super) fn compile(casacore_regex: &str) -> Result<Self, String> {
32        let ecma = to_ecma(casacore_regex.as_bytes());
33        let body = ecma_to_rust(&ecma)
34            .map_err(|reason| format!("Error in regex {casacore_regex}: {reason}"))?;
35        RegexBuilder::new(&format!(r"\A(?:{body})\z"))
36            .unicode(false)
37            .build()
38            .map(Self)
39            .map_err(|error| format!("Error in regex {casacore_regex}: {error}"))
40    }
41
42    /// Compile a shell-style pattern as `Regex::fromPattern` does, then as
43    /// [`compile`](Self::compile).
44    pub(in crate::selection) fn compile_pattern(pattern: &str) -> Result<Self, String> {
45        let regex = from_pattern(pattern.as_bytes());
46        Self::compile(&String::from_utf8_lossy(&regex))
47    }
48
49    /// Whether all of `text` matches (`String::matches(const Regex&, 0)`).
50    pub(in crate::selection) fn matches(&self, text: &str) -> bool {
51        self.0.is_match(text.as_bytes())
52    }
53}

55/// `Regex::fromPattern` (`casa/Utilities/Regex.cc:228-367`): a shell-style
56/// pattern (`*`, `?`, `[...]`, `{a,b}`) as casacore regex text.
57pub(super) fn from_pattern(pattern: &[u8]) -> Vec<u8> {
58    #[derive(PartialEq)]
59    enum State {
60        Stream,
61        BracketOpen,
62        EscapeChar,
63    }
64    let mut brace_count = 0usize;
65    let mut in_bracket_count: i32 = -1;
66    let mut skip_char = false;
67    let mut char_class = false;
68    let mut empty_substrings: Vec<u32> = Vec::new();
69    let mut result = Vec::with_capacity(3 * pattern.len());
70    let mut state = State::Stream;
71    let mut i = 0;
72    while i < pattern.len() {
73        let mut c = pattern[i];
74        match state {
75            State::Stream => match c {
76                b'^' | b'$' | b'(' | b')' | b'.' | b'+' | b'|' => result.push(b'\\'),
77                b'{' => {
78                    c = b'(';
79                    brace_count += 1;
80                    empty_substrings.push(0);
81                }
82                b',' if brace_count > 0 => {
83                    c = b'|';
84                    if matches!(pattern[i - 1], b'{' | b',') {
85                        empty_substrings[brace_count - 1] += 1;
86                        skip_char = true;
87                    }
88                }
89                b'}' => {
90                    if brace_count > 0 {
91                        brace_count -= 1;
92                        c = b')';
93                        if empty_substrings.pop().is_some_and(|empty| empty > 0) {
94                            result.push(c);
95                            c = b'?';
96                        }
97                    } else {
98                        result.push(b'\\');
99                    }
100                }
101                b'[' => {
102                    state = State::BracketOpen;
103                    char_class = false;
104                    in_bracket_count = -1;
105                }
106                b'*' => result.push(b'.'),
107                b'?' => c = b'.',
108                b'\\' => state = State::EscapeChar,
109                _ => {}
110            },
111            State::BracketOpen => {
112                if c == b']' && in_bracket_count > 0 {
113                    state = State::Stream;
114                } else if matches!(c, b'!' | b'^') && in_bracket_count < 0 {
115                    c = b'^';
116                } else {
117                    in_bracket_count = in_bracket_count.max(0);
118                    if c == b'[' && pattern.get(i + 1) == Some(&b':') {
119                        result.push(c);
120                        i += 1;
121                        c = pattern[i];
122                        char_class = true;
123                    } else if char_class && c == b':' && pattern.get(i + 1) == Some(&b']') {
124                        result.push(c);
125                        i += 1;
126                        c = pattern[i];
127                        char_class = false;
128                    }
129                }
130                in_bracket_count += 1;
131            }
132            State::EscapeChar => {
133                // An escaped comma needs no backslash.
134                if c != b',' {
135                    result.push(b'\\');
136                }
137                state = State::Stream;
138            }
139        }
140        if !skip_char && state != State::EscapeChar {
141            result.push(c);
142        }
143        skip_char = false;
144        i += 1;
145    }
146    if state == State::EscapeChar {
147        result.push(b'\\');
148    }
149    result
150}

152/// `Regex::toEcma` (`casa/Utilities/Regex.cc:146-226`): casacore (POSIX
153/// flavoured) regex text as ECMAScript, escaping brackets that ECMAScript
154/// would read differently.
155pub(super) fn to_ecma(rx: &[u8]) -> Vec<u8> {
156    let mut in_bracket_count: i32 = -1;
157    let mut in_bracket = false;
158    let mut char_class = false;
159    let mut escaped = false;
160    let mut result = Vec::with_capacity(rx.len());
161    let mut i = 0;
162    while i < rx.len() {
163        let mut c = rx[i];
164        if escaped {
165            escaped = false;
166            // A two-digit backreference keeps its second digit literal.
167            if (b'1'..=b'9').contains(&c) && rx.get(i + 1).is_some_and(u8::is_ascii_digit) {
168                result.extend_from_slice(&[c, b'[', rx[i + 1]]);
169                c = b']';
170                i += 1;
171            }
172        } else if c == b'\\' {
173            escaped = true;
174            in_bracket_count = 1;
175        } else if !in_bracket {
176            if c == b'[' {
177                in_bracket = true;
178                char_class = false;
179                in_bracket_count = -1;
180            } else if c == b']' {
181                result.push(b'\\');
182            }
183        } else if c == b']' {
184            if in_bracket_count > 0 {
185                in_bracket = false;
186            } else {
187                result.push(b'\\');
188                in_bracket_count = 0;
189            }
190        } else if c != b'^' {
191            in_bracket_count = 0;
192            if c == b'[' {
193                if rx.get(i + 1) == Some(&b':') {
194                    result.push(c);
195                    i += 1;
196                    c = rx[i];
197                    char_class = true;
198                } else {
199                    result.push(b'\\');
200                }
201            } else if char_class && c == b':' && rx.get(i + 1) == Some(&b']') {
202                result.push(c);
203                i += 1;
204                c = rx[i];
205                char_class = false;
206            }
207        }
208        in_bracket_count += 1;
209        result.push(c);
210        i += 1;
211    }
212    result
213}

215/// Translate ECMAScript regex text (bytes) to the `regex` crate's syntax
216/// with the same meaning over bytes.
217///
218/// Every literal byte is written as `\xHH`, so no character is read as a
219/// `regex`-crate operator ECMAScript does not have (`&&`, `--`, `~~` in a
220/// class, `\<` and `\>` assertions). ECMAScript `.` excludes `\n` and `\r`.
221fn ecma_to_rust(ecma: &[u8]) -> Result<String, &'static str> {
222    let mut out = String::with_capacity(4 * ecma.len());
223    let mut i = 0;
224    while i < ecma.len() {
225        let c = ecma[i];
226        i += 1;
227        match c {
228            b'\\' => i = translate_escape(ecma, i, false, &mut out)?,
229            b'[' => i = translate_class(ecma, i, &mut out)?,
230            b'.' => out.push_str(r"[^\n\r]"),
231            b'(' => {
232                if ecma[i..].starts_with(b"?:") {
233                    out.push_str("(?:");
234                    i += 2;
235                } else if ecma.get(i) == Some(&b'?') {
236                    return Err("lookahead groups are not supported");
237                } else {
238                    out.push('(');
239                }
240            }
241            b')' | b'|' | b'*' | b'+' | b'?' | b'{' | b'}' | b'^' | b'$' => out.push(c as char),
242            b',' | b'0'..=b'9' if is_inside_repetition(&out) => out.push(c as char),
243            _ => push_byte(&mut out, c),
244        }
245    }
246    Ok(out)
247}

249/// Whether the translation so far ends inside an open `{n,m}` repetition,
250/// whose digits and comma must stay bare.
251fn is_inside_repetition(out: &str) -> bool {
252    let bytes = out.as_bytes();
253    let mut index = bytes.len();
254    while index > 0 && matches!(bytes[index - 1], b'0'..=b'9' | b',') {
255        index -= 1;
256    }
257    // `\{` was escaped as `\x7B`, so a bare `{` opens a repetition.
258    index > 0 && bytes[index - 1] == b'{'
259}

261/// Translate the ECMAScript escape whose character is at `ecma[i]` (the
262/// backslash already consumed); returns the index after it.
263fn translate_escape(
264    ecma: &[u8],
265    i: usize,
266    in_class: bool,
267    out: &mut String,
268) -> Result<usize, &'static str> {
269    let Some(&c) = ecma.get(i) else {
270        return Err("trailing backslash");
271    };
272    match c {
273        b'd' | b'D' | b'w' | b'W' | b's' | b'S' => {
274            out.push('\\');
275            out.push(c as char);
276        }
277        b'b' | b'B' if !in_class => {
278            out.push('\\');
279            out.push(c as char);
280        }
281        b'b' => push_byte(out, 0x08),
282        b'n' => push_byte(out, b'\n'),
283        b't' => push_byte(out, b'\t'),
284        b'r' => push_byte(out, b'\r'),
285        b'f' => push_byte(out, 0x0c),
286        b'v' => push_byte(out, 0x0b),
287        b'0' if !ecma.get(i + 1).is_some_and(u8::is_ascii_digit) => push_byte(out, 0),
288        b'1'..=b'9' | b'0' => return Err("backreferences are not supported"),
289        b'x' => {
290            let value = hex_value(ecma.get(i + 1..i + 3)).ok_or("invalid \\x escape")?;
291            push_byte(out, value);
292            return Ok(i + 3);
293        }
294        b'c' => {
295            let letter = ecma
296                .get(i + 1)
297                .filter(|letter| letter.is_ascii_alphabetic())
298                .ok_or("invalid \\c escape")?;
299            push_byte(out, letter % 32);
300            return Ok(i + 2);
301        }
302        b'u' => return Err("\\u escapes are not supported over bytes"),
303        _ => push_byte(out, c),
304    }
305    Ok(i + 1)
306}

308fn hex_value(digits: Option<&[u8]>) -> Option<u8> {
309    let digits = std::str::from_utf8(digits?).ok()?;
310    if digits.len() != 2 {
311        return None;
312    }
313    u8::from_str_radix(digits, 16).ok()
314}

316/// Translate an ECMAScript class whose `[` is just before `ecma[i]`;
317/// returns the index after its `]`.
318fn translate_class(ecma: &[u8], mut i: usize, out: &mut String) -> Result<usize, &'static str> {
319    let negated = ecma.get(i) == Some(&b'^');
320    if negated {
321        i += 1;
322    }
323    // ECMAScript `[]` matches nothing and `[^]` matches anything.
324    if ecma.get(i) == Some(&b']') {
325        out.push_str(if negated {
326            r"[\x00-\xFF]"
327        } else {
328            r"[^\x00-\xFF]"
329        });
330        return Ok(i + 1);
331    }
332    out.push_str(if negated { "[^" } else { "[" });
333    loop {
334        let Some(&c) = ecma.get(i) else {
335            return Err("unterminated bracket expression");
336        };
337        i += 1;
338        match c {
339            b']' => {
340                out.push(']');
341                return Ok(i);
342            }
343            b'\\' => i = translate_escape(ecma, i, true, out)?,
344            b'-' => out.push('-'),
345            b'[' if ecma.get(i) == Some(&b':') => {
346                let start = i + 1;
347                let length = ecma[start..]
348                    .windows(2)
349                    .position(|pair| pair == b":]")
350                    .ok_or("unterminated character class name")?;
351                let name = std::str::from_utf8(&ecma[start..start + length])
352                    .ok()
353                    .filter(|name| POSIX_CLASSES.contains(name))
354                    .ok_or("unknown character class name")?;
355                out.push_str("[:");
356                out.push_str(name);
357                out.push_str(":]");
358                i = start + length + 2;
359            }
360            _ => push_byte(out, c),
361        }
362    }
363}

365/// The POSIX class names both `std::regex` and the `regex` crate know.
366const POSIX_CLASSES: &[&str] = &[
367    "alnum", "alpha", "blank", "cntrl", "digit", "graph", "lower", "print", "punct", "space",
368    "upper", "xdigit",
369];
370
371fn push_byte(out: &mut String, byte: u8) {
372    use std::fmt::Write as _;
373    let _ = write!(out, "\\x{byte:02X}");
374}
