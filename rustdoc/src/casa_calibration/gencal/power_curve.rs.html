15pub(super) fn source_path(owner: &Path, table: &Table) -> Result<Option<PathBuf>, GencalError> {
16    match table.keywords().get("GAIN_CURVE") {
17        None => Ok(None),
18        Some(Value::TableRef(reference)) => Ok(Some(casa_tables::add_directory(reference, owner))),
19        Some(_) => Err(GencalError::InvalidRequest(
20            "Error in Calibrater::specifycal. GAIN_CURVE keyword does not name a table".into(),
21        )),
22    }
23}

25pub(super) fn rows(ms: &MeasurementSet, path: &Path) -> Result<Vec<PriorCalRow>, GencalError> {
26    let invalid = |message: String| {
27        GencalError::InvalidRequest(format!("Error in Calibrater::specifycal. {message}"))
28    };
29    let table =
30        Table::open(TableOptions::new(path)).map_err(|source| GencalError::OpenGainCurves {
31            path: path.display().to_string(),
32            source: Box::new(source),
33        })?;
34    let antenna_count = ms
35        .antenna()
36        .map_err(|error| invalid(error.to_string()))?
37        .row_count();
38    let window_count = ms
39        .spectral_window()
40        .map_err(|error| invalid(error.to_string()))?
41        .row_count();
42    let mut curves = Vec::new();
43    for row in 0..table.row_count() {
44        let integer = |column: &str| {
45            let value = table
46                .cell_accessor(row, column)
47                .and_then(|cell| cell.scalar())
48                .map_err(|error| invalid(error.to_string()))?;
49            match value {
50                ScalarValue::Int32(value) => Ok(*value),
51                _ => Err(invalid(format!(
52                    "GAIN_CURVE {column} row {row} is not an integer"
53                ))),
54            }
55        };
56        let window = integer("SPECTRAL_WINDOW_ID")?;
57        if usize::try_from(window).map_or(true, |id| id >= window_count) {
58            continue;
59        }
60        let antenna = integer("ANTENNA_ID")?;
61        if usize::try_from(antenna).map_or(true, |id| id >= antenna_count) {
62            return Err(invalid(format!(
63                "GAIN_CURVE row {row} has invalid antenna {antenna}"
64            )));
65        }
66        let terms = (integer("NUM_POLY")?.max(0) as usize).min(POWER_CURVE_TERMS);
67        let coordinate = get_string(&table, row, "TYPE")?;
68        let (elevation, voltage) = match coordinate.as_str() {
69            "POWER(ZA)" => (false, false),
70            "POWER(EL)" => (true, false),
71            "VOLTAGE(ZA)" => (false, true),
72            "VOLTAGE(EL)" => (true, true),
73            _ => {
74                return Err(invalid(format!(
75                    "unsupported gain-curve type {coordinate:?}"
76                )));
77            }
78        };
79        let array = table
80            .cell_accessor(row, COL_GAIN)
81            .and_then(|cell| cell.array())
82            .map_err(|error| invalid(error.to_string()))?;
83        let ArrayValue::Float32(raw) = array else {
84            return Err(invalid(format!("GAIN_CURVE row {row} has non-float GAIN")));
85        };
86        if raw.ndim() != 2 || raw.shape()[0] != 2 || raw.shape()[1] < terms {
87            return Err(invalid(format!(
88                "GAIN_CURVE row {row} has an incompatible GAIN shape"
89            )));
90        }
91        let sensitivity = get_f32_array_fortran(&table, row, "SENSITIVITY")?;
92        let [sensitivity_r, sensitivity_l] = sensitivity.as_slice() else {
93            return Err(invalid(format!(
94                "GAIN_CURVE row {row} needs two sensitivity values"
95            )));
96        };
97        let mut gain = [0.0; 2 * POWER_CURVE_TERMS];
98        for receptor in 0..2 {
99            let start = receptor * POWER_CURVE_TERMS;
100            let polynomial = &mut gain[start..start + POWER_CURVE_TERMS];
101            for (degree, term) in polynomial.iter_mut().enumerate().take(terms) {
102                *term = raw[[receptor, degree]];
103            }
104            if elevation {
105                let mut zenith = [0.0; POWER_CURVE_TERMS];
106                for (degree, value) in zenith.iter_mut().enumerate() {
107                    for (input_degree, coefficient) in polynomial.iter().enumerate().skip(degree) {
108                        let scale = choose(input_degree, degree) as f64
109                            * (-1.0_f64).powi(input_degree as i32)
110                            * (-90.0_f64).powi((input_degree - degree) as i32);
111                        *value += *coefficient * scale as f32;
112                    }
113                }
114                polynomial.copy_from_slice(&zenith);
115            }
116        }
117        if voltage {
118            // CASA's EPowerCurve squares the first receptor into a cleared
119            // vector. The second receptor remains zero in that legacy path.
120            let mut power = [0.0; 2 * POWER_CURVE_TERMS];
121            for (left, coefficient) in gain[..POWER_CURVE_TERMS].iter().enumerate() {
122                for (right, other) in gain[..POWER_CURVE_TERMS - left].iter().enumerate() {
123                    power[left + right] += coefficient * other;
124                }
125            }
126            gain = power;
127        }
128        for (coefficients, scale) in gain
129            .as_chunks_mut::<POWER_CURVE_TERMS>()
130            .0
131            .iter_mut()
132            .zip([*sensitivity_r, *sensitivity_l])
133        {
134            for coefficient in coefficients {
135                *coefficient *= scale;
136            }
137        }
138        let time = get_f64(&table, row, COL_TIME)?;
139        let half = get_f64(&table, row, "INTERVAL")? / 2.0;
140        let begin = time - half;
141        let end = time + half;
142        curves.push(PriorCalRow {
143            time_seconds: (begin + end) / 2.0,
144            interval_seconds: (end - begin) / 2.0,
145            field_id: -1,
146            spw_id: window,
147            antenna_id: antenna,
148            observation_id: -1,
149            scan_number: -1,
150            parameters: f32_column(&gain),
151            snr_value: 0.0,
152            preserved: None,
153        });
154    }
155    Ok(curves)
156}

158fn choose(n: usize, k: usize) -> usize {
159    (0..k).fold(1, |value, step| value * (n - step) / (step + 1))
160}
