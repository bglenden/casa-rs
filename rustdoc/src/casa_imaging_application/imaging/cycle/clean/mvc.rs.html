2//! The two state owners of an MVC cleaning cycle (#1069).
3
4use super::*;
5
6/// The cube completion and the Taylor PB and admission retained while the
7/// ordinary cleaning controller operates on the Taylor completion.
8pub(in crate::imaging::cycle) struct MvcState<'a> {
9    spec: MvcCycleSpec<'a>,
10    cube: Option<MajorCycleCompletion>,
11    mean_pb: Vec<f32>,
12    taylor_reservation: Reservation,
13}
14
15impl<'a> MvcState<'a> {
16    /// Reduce the first cube pass to the Taylor state used by the minor cycle.
17    pub(in crate::imaging::cycle) fn begin(
18        spec: MvcCycleSpec<'a>,
19        cube: MajorCycleCompletion,
20        run: &Run<'_>,
21    ) -> Result<(Self, MajorCycleCompletion), ImagingError> {
22        let taylor = crate::mvc::assemble_taylor_completion(
23            crate::mvc::TaylorAssemblyInputs {
24                cube_problem: run.problem,
25                cube: &cube,
26                taylor_problem: spec.taylor_problem,
27                expansion_reference_hz: spec.expansion_reference_hz,
28                controls: spec.controls,
29                centres: spec.centres,
30            },
31            (&run.host, &run.policy, &run.cancel),
32        )?;
33        Ok((
34            Self {
35                spec,
36                cube: Some(cube),
37                mean_pb: taylor.value.primary_beam,
38                taylor_reservation: taylor.reservation,
39            },
40            taylor.value.scientific,
41        ))
42    }
43
44    /// Apply a Taylor minor update, predict it through the cube, then reduce
45    /// the refreshed cube residual back around the same Taylor model.
46    pub(super) fn refresh(
47        &mut self,
48        run: &mut Run<'_>,
49        taylor: MajorCycleCompletion,
50        (terms, results): (Vec<ModelDeltaTerm>, Reservation),
51        last: bool,
52    ) -> Result<MajorCycleCompletion, ImagingError> {
53        let shape = taylor.normal_state().shape();
54        let cells = shape[0]
55            .checked_mul(shape[1])
56            .ok_or(ImagingError::Unsupported {
57                reason: "MVC Taylor model shape exceeds addressable size",
58            })?;
59        let lifecycle =
60            ModelLifecycle::new(self.spec.taylor_problem, ModelStoragePlan::resident(cells)?);
61        let (old_normal, model) = taylor.into_parts();
62        drop(old_normal);
63        let prepared = lifecycle.prepare_final_model(model, terms)?;
64        let cube = self.cube.as_ref().ok_or(ImagingError::Unsupported {
65            reason: "MVC cube state is absent",
66        })?;
67        let projected = crate::mvc::project_mvc_model(
68            crate::mvc::MvcModelProjectionInputs {
69                cube_problem: run.problem,
70                cube,
71                cube_lifecycle: &run.lifecycle,
72                taylor_problem: self.spec.taylor_problem,
73                taylor_model: prepared.generation(),
74                mean_pb: &self.mean_pb,
75                expansion_reference_hz: self.spec.expansion_reference_hz,
76                controls: self.spec.controls,
77                centres: self.spec.centres,
78            },
79            (&run.host, &run.policy, &run.cancel),
80        )?;
81        let cube = self.cube.take().expect("the cube was borrowed above");
82        let major = run.refresh_with_projected_model(cube, projected, last)?;
83        let taylor = crate::mvc::assemble_taylor_with_model(
84            crate::mvc::TaylorAssemblyInputs {
85                cube_problem: run.problem,
86                cube: &major.completion,
87                taylor_problem: self.spec.taylor_problem,
88                expansion_reference_hz: self.spec.expansion_reference_hz,
89                controls: self.spec.controls,
90                centres: self.spec.centres,
91            },
92            Some(prepared),
93            (&run.host, &run.policy, &run.cancel),
94        )?;
95        // The minor result's charge covers its queued Taylor update until
96        // assembly has completed and released the prepared model.
97        drop(results);
98        self.cube = Some(major.completion);
99        self.mean_pb = taylor.value.primary_beam;
100        self.taylor_reservation = taylor.reservation;
101        Ok(taylor.value.scientific)
102    }
103
104    /// Return the cube as the major result and the Taylor family with its
105    /// own admission for staged publication.
106    pub(super) fn finish(
107        mut self,
108        taylor: MajorCycleCompletion,
109    ) -> (MajorCycleCompletion, Admitted<crate::mvc::TaylorCompletion>) {
110        let cube = self.cube.take().expect("an MVC cycle retains its cube");
111        (
112            cube,
113            Admitted {
114                value: crate::mvc::TaylorCompletion {
115                    scientific: taylor,
116                    primary_beam: self.mean_pb,
117                },
118                reservation: self.taylor_reservation,
119            },
120        )
121    }
122}
123
124impl Run<'_> {
125    /// Refresh a cube using a separately admitted model projected from its
126    /// Taylor minor cycle, rather than applying cube-space sparse terms.
127    fn refresh_with_projected_model(
128        &mut self,
129        completion: MajorCycleCompletion,
130        projected: Admitted<ModelGeneration>,
131        last: bool,
132    ) -> Result<Major, ImagingError> {
133        let (normal, old_model) = completion.into_parts();
134        drop(old_model);
135        let model = self.lifecycle.prepare_final_model(projected.value, [])?;
136        let Admitted {
137            value: plan,
138            reservation: mut pass,
139        } = self.admit_pass(
140            ModeSet::DATA,
141            true,
142            self.write_bytes(last),
143            self.writes(last),
144        )?;
145        pass.join(projected.reservation);
146        let cycle = MajorCycle::refresh(
147            self.problem,
148            normal,
149            model,
150            self.normal_storage(plan.residency)?,
151        )?;
152        let major = self.reconcile(cycle, ModeSet::DATA, true, last, plan)?;
153        self.retain(&major, pass)?;
154        Ok(major)
155    }
