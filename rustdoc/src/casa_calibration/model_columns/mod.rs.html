32const DATA: VisibilityDataColumn = VisibilityDataColumn::Data;
33const CORRECTED: VisibilityDataColumn = VisibilityDataColumn::CorrectedData;
34const MODEL: VisibilityDataColumn = VisibilityDataColumn::ModelData;

36#[derive(Clone, Copy, PartialEq, Eq)]
37pub(crate) enum ScratchWriteMode<'a> {
38    Replace,
39    Create,
40    Projected { corrected: &'a str, model: &'a str },
41}
42
43impl<'a> ScratchWriteMode<'a> {
44    fn physical_name(self, column: VisibilityDataColumn) -> &'a str {
45        match (self, column) {
46            (Self::Projected { corrected, .. }, CORRECTED) => corrected,
47            (Self::Projected { model, .. }, MODEL) => model,
48            _ => column.name(),
49        }
50    }
51}

53pub fn reset_scratch_columns(
54    ms: &mut MeasurementSet,
55    selection: &MsSelection,
56    add_model: bool,
57    mut lock: TableWriteLock,
58) -> MsResult<TableWriteLock> {
59    let has_corrected = has_column(ms, CORRECTED);
60    let has_model = has_column(ms, MODEL);
61    let mut new = Vec::new();
62    if !has_corrected {
63        new.push(CORRECTED);
64    }
65    if add_model && !has_model {
66        new.push(MODEL);
67    }
68    if !new.is_empty() {
69        // CASA adds and initializes missing columns across the entire MS
70        // before resolving the user's selectors.
71        let all =
72            resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
73        lock = reset(ms, all, &new, true, false, lock)?;
74    }
75    if has_corrected {
76        lock = reset_existing_scratch(ms, selection, has_model, true, lock)?;
77    }
78    Ok(lock)
79}

85    add_model: bool,

92pub(crate) fn initialize_calibrater_columns_holding(
93    ms: &mut MeasurementSet,
94    compress: bool,
95    add_corrected: bool,
96    add_model: bool,
97    lock: &mut TableWriteLock,
98) -> MsResult<()> {
99    let missing = [(MODEL, add_model), (CORRECTED, add_corrected)]
100        .into_iter()
101        .filter(|&(column, requested)| requested && !has_column(ms, column))
102        .map(|(column, _)| column)
103        .collect::<Vec<_>>();
104    if missing.is_empty() {
105        // Calibrater preserves both existing scratch cells and virtual
106        // recipes on open. Adding a missing MODEL_DATA clears the recipes.
107        return Ok(());
108    }
109    let all = resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
110    if compress {
111        compression::initialize_columns(ms, all, &missing, lock)?;
112        return Ok(());
113    }
114    let observed = observed_column(ms);
115    reset_holding(
116        ms,
117        all,
118        &missing,
119        observed,
120        ScratchWriteMode::Create,
121        false,
122        lock,
123    )
124}

126pub(crate) fn initialize_projected_scratch_holding(
127    ms: &mut MeasurementSet,
128    rows: &[usize],
129    observed: &str,
130    [corrected, model]: [&str; 2],
131    lock: &mut TableWriteLock,
132) -> MsResult<()> {
133    let missing = [(CORRECTED, corrected), (MODEL, model)]
134        .into_iter()
135        .filter(|&(_, physical)| !has_column_named(ms, physical))
136        .map(|(column, physical)| {
137            if column.name() == physical {
138                Ok(column)
139            } else {
140                Err(invalid(format!(
141                    "missing projected scratch column {physical}"
142                )))
143            }
144        })
145        .collect::<MsResult<Vec<_>>>()?;
146    if missing.is_empty() {
147        return Ok(());
148    }
149    let whole_table = rows.iter().copied().eq(0..ms.row_count());
150    if whole_table && missing.contains(&CORRECTED) && can_clone_tiled_observed(ms, observed) {
151        return initialize_hanning_scratch_holding(ms, rows, observed, [corrected, model], lock);
152    }
153    let mut selection =
154        resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
155    if !whole_table {
156        let rows: HashSet<_> = rows.iter().copied().collect();
157        selection
158            .selected_rows
159            .retain(|row| rows.contains(&row.row_index));
160    }
161    reset_holding(
162        ms,
163        selection,
164        &missing,
165        observed,
166        ScratchWriteMode::Create,
167        false,
168        lock,
169    )
170}

172/// Reset existing scratch columns for `calibrater.initcalset`.
173///
174/// `calset=1` removes virtual models from the entire MeasurementSet before
175/// initializing the selected scratch cells. Other values keep virtual models.
176/// Like CASA, the reset requires the legacy visibility iterator mode.
177///
178/// # Errors
179///
180/// Returns an error for the modern iterator mode, a missing corrected column,
181/// an empty or invalid selection, or a failed read or write.
182pub fn reset_calibration_set(
183    ms: &mut MeasurementSet,
184    selection: &MsSelection,
185    calset: i64,
186    old_iterator: bool,
187    mut lock: TableWriteLock,
188) -> MsResult<TableWriteLock> {
189    check_main_lock(ms, &lock)?;
190    let clear_virtual = calset == 1;
191    if !old_iterator || !has_column(ms, CORRECTED) {
192        if clear_virtual {
193            casa_ms::clear_virtual_model_holding(ms, &[], &[], &mut lock)?;
194        }
195        return Err(invalid(if old_iterator {
196            "the MeasurementSet has no CORRECTED_DATA column"
197        } else {
198            "Calibrater::initCalSet not implemented"
199        }));
200    }
201    reset_existing_scratch(ms, selection, has_column(ms, MODEL), clear_virtual, lock)
202}
203
204fn reset_existing_scratch(
205    ms: &mut MeasurementSet,
206    selection: &MsSelection,
207    has_model: bool,
208    clear_virtual: bool,
209    lock: TableWriteLock,
210) -> MsResult<TableWriteLock> {
211    let selected = resolve_casa_calibration_selection(ms, selection).map_err(invalid)?;
212    if selected.selected_rows.is_empty() {
213        return Err(invalid("selection matches no rows"));
214    }
215    let columns = if has_model {
216        &[CORRECTED, MODEL][..]
217    } else {
218        &[CORRECTED]
219    };
220    reset(ms, selected, columns, false, clear_virtual, lock)
221}

223/// Prepare a visibility scratch column at attachment, before data selection.
224pub(crate) fn initialize_scratch_column(
225    ms: &mut MeasurementSet,
226    column: VisibilityDataColumn,
227    mut lock: TableWriteLock,
228) -> MsResult<TableWriteLock> {
229    if has_column(ms, column) {
230        if column == MODEL {
231            // CASA invalidates virtual recipes when it attaches a scratch
232            // model iterator, even if the scratch samples already exist.
233            casa_ms::prepare_write_holding(ms, column, false, &mut lock)?;
234        }
235        return Ok(lock);
236    }
237    let all = resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
238    reset(ms, all, &[column], true, false, lock)
239}

241/// Delete selected virtual models and/or the entire scratch model column.
242///
243/// CASA resolves `field` and `spw` when `virtual_model` is true. An empty
244/// field selection clears every virtual model. The scratch deletion includes
245/// any physical backing columns used for compression.
246///
247/// # Errors
248///
249/// Returns an error for an invalid selector or a failed table update.
250pub fn delete_calibration_model(
251    ms: &mut MeasurementSet,
252    field: Option<&str>,
253    spw: Option<&str>,
254    virtual_model: bool,
255    scratch: bool,
256    lock: &mut TableWriteLock,
257) -> MsResult<()> {
258    check_main_lock(ms, lock)?;
259    if virtual_model {
260        let field = field.filter(|value| !value.is_empty() && !value.eq_ignore_ascii_case("false"));
261        let spw = spw.filter(|value| !value.is_empty() && !value.eq_ignore_ascii_case("false"));
262        let fields = field.map_or_else(
263            || Ok(Vec::new()),
264            |text| ms.resolve_field_selector(if text.trim().is_empty() { "*" } else { text }),
265        )?;
266        let mut windows = Vec::new();
267        if !fields.is_empty()
268            && let Some(spw) = spw
269        {
270            windows.extend(
271                ms.resolve_spw_selector(spw)?
272                    .spw_ids()
273                    .iter()
274                    .map(|&id| id as i32),
275            );
276            windows.sort_unstable();
277            windows.dedup();
278        }
279        casa_ms::clear_virtual_model_holding(ms, &fields, &windows, lock)?;
280    }
281    if scratch {
282        casa_ms::remove_scratch_model_holding(ms, lock)?;
283    }
284    Ok(())
285}

287fn check_main_lock(ms: &MeasurementSet, lock: &TableWriteLock) -> MsResult<()> {
288    let path = ms
289        .path()
290        .ok_or_else(|| invalid("the MeasurementSet has no on-disk MAIN table"))?;
291    if path.canonicalize().map_err(invalid)? != lock.path().canonicalize().map_err(invalid)? {
292        return Err(invalid(format!(
293            "the write lock held on {} is not on MAIN of {}",
294            lock.path().display(),
295            path.display()
296        )));
297    }
298    Ok(())
299}

324/// Legacy calibration iterators visit the inclusive span of all selected segments.
325pub(crate) fn calibration_channel_span(selection: &ChannelSelection) -> Option<Range<usize>> {
326    let first = selection
327        .segments
328        .iter()
329        .map(|segment| segment.start.min(segment.end))
330        .min()?;
331    let last = selection
332        .segments
333        .iter()
334        .map(|segment| segment.start.max(segment.end))
335        .max()?;
336    Some(first as usize..last as usize + 1)
337}

339fn reset(
340    ms: &mut MeasurementSet,
341    selected: ResolvedMsSelection,
342    columns: &[VisibilityDataColumn],
343    create: bool,
344    clear_virtual: bool,
345    mut lock: TableWriteLock,
346) -> MsResult<TableWriteLock> {
347    let observed = observed_column(ms);
348    reset_holding(
349        ms,
350        selected,
351        columns,
352        observed,
353        if create {
354            ScratchWriteMode::Create
355        } else {
356            ScratchWriteMode::Replace
357        },
358        clear_virtual,
359        &mut lock,
360    )?;
361    Ok(lock)
362}

364/// Initialize Hanning scratch columns for the physical rows in the current
365/// attachment. Existing model cells and virtual model recipes are preserved.
366///
367/// # Errors
368/// The caller's MAIN lock is stale, or a required column cannot be filled.
369pub fn initialize_hanning_scratch_holding(
370    ms: &mut MeasurementSet,
371    rows: &[usize],
372    observed: &str,
373    [corrected, model]: [&str; 2],
374    lock: &mut TableWriteLock,
375) -> MsResult<()> {
376    if !ms.path().is_some_and(|path| lock.matches_table_path(path)) {
377        return Err(invalid(
378            "the Hanning write lock does not protect physical MAIN",
379        ));
380    }
381    let has_corrected = has_column_named(ms, corrected);
382    let has_model = has_column_named(ms, model);
383    if (!has_corrected && corrected != CORRECTED.name()) || (!has_model && model != MODEL.name()) {
384        return Err(invalid(
385            "a physical column named by the scratch projection is missing",
386        ));
387    }
388    let mode = ScratchWriteMode::Projected { corrected, model };
389    casa_ms::prepare_write_holding(ms, CORRECTED, true, lock)?;
390    let whole_table = !rows.is_empty() && rows.iter().copied().eq(0..ms.row_count());
391    if whole_table && !has_corrected && can_clone_tiled_observed(ms, observed) {
392        add_scratch_schema(ms, CORRECTED, observed)?;
393        lock.record_write();
394        ms.main_table_mut().prepare_write().add_tiled_column_clone(
395            observed,
396            CORRECTED.name(),
397            "TiledCORRECTED_DATA",
398        )?;
399        if !has_model {
400            let staging = tempfile::tempdir_in(lock.path()).map_err(invalid)?;
401            let model = stream_hanning_model(ms, rows, observed, &staging.path().join("MODEL"))?;
402            add_scratch_schema(ms, MODEL, observed)?;
403            lock.record_write();
404            ms.main_table_mut()
405                .prepare_write()
406                .add_streamed_tiled_shape_column(MODEL.name(), model)?;
407        }
408        return Ok(());
409    }
410    let mut columns = Vec::with_capacity(2);
411    if !has_corrected || (!rows.is_empty() && observed != corrected) {
412        columns.push(CORRECTED);
413    }
414    if !has_model {
415        columns.push(MODEL);
416    }
417    if columns.is_empty() {
418        return Ok(());
419    }
420    if rows.is_empty() {
421        let (plan, _) = scratch_write_plan(ms, Vec::new(), &columns, observed, mode, false, lock)?;
422        MeasurementSetWriteSession::mutate_selected_rows(ms, plan, Some(lock), invalid, |_, _| {
423            Ok(())
424        })?;
425        return Ok(());
426    }
427    let mut selection =
428        resolve_casa_calibration_selection(ms, &MsSelection::default()).map_err(invalid)?;
429    if !whole_table {
430        let selected: HashSet<_> = rows.iter().copied().collect();
431        selection
432            .selected_rows
433            .retain(|row| selected.contains(&row.row_index));
434    }
435    reset_holding(ms, selection, &columns, observed, mode, false, lock)
436}

438fn stream_hanning_model(
439    ms: &MeasurementSet,
440    rows: &[usize],
441    observed: &str,
442    temporary: &Path,
443) -> MsResult<StreamedTiledShapeColumn> {
444    let table = ms.main_table();
445    let shapes = table
446        .column_accessor(observed)?
447        .array_shapes_uncached(rows)?
448        .into_iter()
449        .map(|shape| {
450            let Some([correlations, channels]) = shape.as_deref() else {
451                return Err(invalid("observed data must have a defined rank-2 shape"));
452            };
453            Ok([*correlations, *channels])
454        })
455        .collect::<MsResult<Vec<_>>>()?;
456    let mut rows_per_shape = BTreeMap::<_, usize>::new();
457    for &shape in &shapes {
458        *rows_per_shape.entry(shape).or_default() += 1;
459    }
460    let layouts = rows_per_shape
461        .into_iter()
462        .map(|(shape, count)| {
463            let tiles = TileLayoutPlanner::repository_default()
464                .plan_column(&shape, count, std::mem::size_of::<Complex32>())
465                .map_err(invalid)?;
466            Ok(StreamedTiledShapeCubeLayout::new(
467                shape.to_vec(),
468                count,
469                tiles.tile_shape().to_vec(),
470            ))
471        })
472        .collect::<MsResult<Vec<_>>>()?;
473    let mut scalar = table.required_scalar_columns_owned_for_rows(&["DATA_DESC_ID"], rows)?;
474    let Some(RequiredScalarColumnValues::Int32(descriptions)) = scalar.remove("DATA_DESC_ID")
475    else {
476        return Err(invalid("DATA_DESC_ID must be Int32"));
477    };
478    let dd = ms.data_description()?;
479    let pol_ids = (0..dd.row_count())
480        .map(|row| dd.polarization_id(row))
481        .collect::<MsResult<Vec<_>>>()?;
482    let polarizations = model_polarizations(ms)?;
483    let big_endian = table
484        .stored_endian_format()?
485        .is_none_or(casa_tables::EndianFormat::is_big_endian);
486    let resources = MeasurementSetWriteResources::from_system_memory(1).map_err(invalid)?;
487    let mut writer = StreamingTiledShapeWriter::create(
488        temporary,
489        StreamedTiledShapeValueType::Complex32,
490        layouts,
491        resources.available_bytes,
492        big_endian,
493    )
494    .map_err(invalid)?;
495    let mut models = HashMap::new();
496    for (shape, description) in shapes.into_iter().zip(descriptions) {
497        let pol = *pol_ids
498            .get(description as usize)
499            .ok_or_else(|| invalid("invalid DATA_DESC_ID"))?;
500        let values = polarizations
501            .get(&pol)
502            .ok_or_else(|| invalid("invalid POLARIZATION_ID"))?;
503        if values.len() != shape[0] {
504            return Err(invalid("POLARIZATION does not agree with DATA"));
505        }
506        let cell = models.entry((shape, pol)).or_insert_with(|| {
507            values
508                .iter()
509                .copied()
510                .cycle()
511                .take(shape[0] * shape[1])
512                .collect::<Vec<_>>()
513        });
514        writer.push_complex32_row(&shape, cell).map_err(invalid)?;
515    }
516    writer.finish().map_err(invalid)
517}

519fn can_clone_tiled_observed(ms: &MeasurementSet, observed: &str) -> bool {
520    let table = ms.main_table();
521    table.schema().is_some_and(|schema| {
522        schema.column(observed).and_then(|col| col.data_type()) == Some(PrimitiveType::Complex32)
523            && !schema
524                .columns()
525                .iter()
526                .any(|col| table.is_virtual_column(col.name()))
527    }) && table
528        .data_manager_info()
529        .iter()
530        .any(|manager| manager.dm_type == "TiledShapeStMan" && manager.columns == [observed])
531}

533fn reset_holding(
534    ms: &mut MeasurementSet,
535    selected: ResolvedMsSelection,
536    columns: &[VisibilityDataColumn],
537    observed: &str,
538    mode: ScratchWriteMode<'_>,
539    clear_virtual: bool,
540    lock: &mut TableWriteLock,
541) -> MsResult<()> {
542    let (plan, destinations) = scratch_write_plan(
543        ms,
544        selected.row_indices().collect(),
545        columns,
546        observed,
547        mode,
548        clear_virtual,
549        lock,
550    )?;
551    let batch_rows = plan.batch_rows().max(1);
552    let polarizations = if columns.contains(&MODEL) {
553        model_polarizations(ms)?
554    } else {
555        BTreeMap::new()
556    };
557    let ranges = selected
558        .channel_selections
559        .iter()
560        .filter_map(|(&spw, selection)| Some((spw, calibration_channel_span(selection)?)))
561        .collect::<BTreeMap<_, _>>();
562    MeasurementSetWriteSession::mutate_selected_rows(
563        ms,
564        plan,
565        Some(lock),
566        invalid,
567        |session, ms| {
568            for rows in selected.selected_rows.chunks(batch_rows) {
569                let batch =
570                    reset_batch(ms, observed, rows, &destinations, &ranges, &polarizations)?;
571                session.write_mutation_batch(ms, batch).map_err(invalid)?;
572            }
573            Ok(())
574        },
575    )?;
576    Ok(())
577}

579pub(crate) fn scratch_write_plan<'a>(
580    ms: &mut MeasurementSet,
581    row_indices: Vec<usize>,
582    columns: &[VisibilityDataColumn],
583    observed: &str,
584    mode: ScratchWriteMode<'a>,
585    clear_virtual: bool,
586    lock: &mut TableWriteLock,
587) -> MsResult<(MeasurementSetWritePlan, Vec<ScratchDestination<'a>>)> {
588    let destinations = columns
589        .iter()
590        .map(|&column| ScratchDestination::new(ms, column, mode.physical_name(column)))
591        .collect::<MsResult<Vec<_>>>()?;
592    if clear_virtual {
593        casa_ms::clear_virtual_model_holding(ms, &[], &[], lock)?;
594    }
595    let creates = columns
596        .iter()
597        .map(|&column| match mode {
598            ScratchWriteMode::Replace => false,
599            ScratchWriteMode::Create => true,
600            ScratchWriteMode::Projected { .. } => !has_column_named(ms, mode.physical_name(column)),
601        })
602        .collect::<Vec<_>>();
603    if mode == ScratchWriteMode::Create {
604        for &column in columns {
605            casa_ms::prepare_write_holding(ms, column, true, lock)?;
606        }
607    }
608    for (&column, &create) in columns.iter().zip(&creates) {
609        if create {
610            add_scratch_schema(ms, column, observed)?;
611        }
612    }
613    let sizes = maximum_visibility_cell_elements(ms).map_err(invalid)?;
614    let write_columns = destinations
615        .iter()
616        .zip(creates)
617        .flat_map(|(column, create)| column.plans(sizes, create))
618        .collect();
619    let resources = MeasurementSetWriteResources::from_system_memory(2).map_err(invalid)?;
620    let plan =
621        MeasurementSetWritePlan::selected_row_mutation(row_indices, write_columns, resources)
622            .map_err(invalid)?;
623    Ok((plan, destinations))
624}

626fn add_scratch_schema(
627    ms: &mut MeasurementSet,
628    column: VisibilityDataColumn,
629    observed: &str,
630) -> MsResult<()> {
631    let keyword = (column == CORRECTED)
632        .then(|| ms.main_table().column_keywords(observed))
633        .flatten()
634        .cloned();
635    let schema =
636        casa_ms::build_table_schema(&[*column.optional_column().column_def()]).map_err(invalid)?;
637    let definition = schema
638        .column(column.name())
639        .expect("scratch column schema")
640        .clone();
641    ms.main_table_mut().add_column(definition, None)?;
642    if let Some(keyword) = keyword {
643        ms.main_table_mut()
644            .set_column_keywords(column.name(), keyword);
645    }
646    Ok(())
647}

649fn model_polarizations(ms: &MeasurementSet) -> MsResult<BTreeMap<i32, Vec<Complex32>>> {
650    let polarization = ms.polarization()?;
651    (0..polarization.row_count())
652        .map(|id| {
653            let values = polarization
654                .corr_type(id)?
655                .into_iter()
656                .map(initial_model_visibility)
657                .collect();
658            Ok((id as i32, values))
659        })
660        .collect()
661}

663fn reset_batch(
664    ms: &MeasurementSet,
665    observed: &str,
666    rows: &[ResolvedMsSelectionRow],
667    columns: &[ScratchDestination<'_>],
668    ranges: &BTreeMap<i32, Range<usize>>,
669    polarizations: &BTreeMap<i32, Vec<Complex32>>,
670) -> MsResult<MeasurementSetMutationBatch> {
671    let row_indices = rows.iter().map(|row| row.row_index).collect::<Vec<_>>();
672    let main = ms.main_table();
673    let data = main.column_accessor(observed)?;
674    let shapes = data.array_shapes_uncached(&row_indices)?;
675    let mut data = columns
676        .iter()
677        .any(|column| column.visibility == CORRECTED)
678        .then(|| data.array_cells_owned_uncached(&row_indices))
679        .transpose()?;
680    let needs_preserved_channels = rows
681        .iter()
682        .any(|row| ranges.contains_key(&row.spectral_window_id));
683    let mut batch = Vec::with_capacity(columns.len());
684    for destination in columns {
685        let column = destination.visibility;
686        let mut previous = needs_preserved_channels
687            .then(|| {
688                main.column_accessor(destination.name())?
689                    .array_cells_owned_uncached(&row_indices)
690            })
691            .transpose()?;
692        let mut values = Vec::with_capacity(rows.len());
693        for (index, row) in rows.iter().enumerate() {
694            let shape = shapes[index]
695                .as_deref()
696                .ok_or_else(|| invalid("DATA is undefined"))?;
697            let &[correlations, channels] = shape else {
698                return Err(invalid("visibility data must have two axes"));
699            };
700            let replacement = if column == MODEL {
701                let model = polarizations
702                    .get(&row.polarization_id)
703                    .ok_or_else(|| invalid("invalid POLARIZATION_ID"))?;
704                if model.len() != correlations {
705                    return Err(invalid("POLARIZATION does not agree with DATA"));
706                }
707                ArrayValue::Complex32(ArrayD::from_shape_fn(IxDyn(shape).f(), |position| {
708                    model[position[0]]
709                }))
710            } else {
711                let observed = data
712                    .as_mut()
713                    .and_then(|rows| rows[index].take())
714                    .ok_or_else(|| invalid("observed visibility is undefined"))?;
715                match observed {
716                    ArrayValue::Float32(array) => {
717                        ArrayValue::Complex32(array.mapv(|value| Complex32::new(value, 0.0)))
718                    }
719                    complex => complex,
720                }
721            };
722            if let Some(range) = ranges.get(&row.spectral_window_id) {
723                if range.end > channels {
724                    return Err(invalid("spectral-window selection exceeds DATA shape"));
725                }
726                let mut previous = previous
727                    .as_mut()
728                    .and_then(|rows| rows[index].take())
729                    .ok_or_else(|| invalid("scratch column is undefined"))?;
730                copy_channels(&mut previous, &replacement, range.clone())?;
731                values.push(previous);
732            } else {
733                values.push(replacement);
734            }
735        }
736        batch.extend(destination.batch(ms, &row_indices, rows, ranges, values)?);
737    }
738    Ok(MeasurementSetMutationBatch {
739        row_indices,
740        columns: batch,
741    })
742}
743
744fn copy_channels(
745    current: &mut ArrayValue,
746    replacement: &ArrayValue,
747    channels: Range<usize>,
748) -> MsResult<()> {
749    let (ArrayValue::Complex32(current), ArrayValue::Complex32(replacement)) =
750        (current, replacement)
751    else {
752        return Err(invalid("expected complex scratch visibilities"));
753    };
754    if current.shape() != replacement.shape() {
755        return Err(invalid("scratch and DATA shapes do not match"));
756    }
757    for channel in channels {
758        current
759            .index_axis_mut(ndarray::Axis(1), channel)
760            .assign(&replacement.index_axis(ndarray::Axis(1), channel));
761    }
762    Ok(())
763}

766    MsError::InvalidInput(error.to_string())
