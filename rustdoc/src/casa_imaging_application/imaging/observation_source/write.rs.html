2//! The batch route's write-back (#932, step 5): the final model into the
3//! `model` group (`MODEL_DATA`) and the continuum-subtracted data into the
4//! `corrected` group (`CORRECTED_DATA`), through the update session the
5//! route reads its batches from.
6//!
7//! The passes hand each block to their sink on the consumer thread, after
8//! the source has refilled the batch's buffers, so the sink writes against
9//! each batch's address ([`BatchAddress`]). While the writing traversal
10//! runs, the source queues, in delivery order, each partition it binds
11//! (its request, which the sink binds an updater to) and, for each block
12//! it fills, its batch's address and the batch sample of each of its rows.
13//! The runtime hands a writing pass's sink every block once, in source
14//! order ([`casa_imaging_runtime::pass::VisibilitySink`]), so the sink
15//! takes the queue's head and checks it against the block. It binds a
16//! partition's updater before it finishes the last one's, so an MSv2
17//! session holds MAIN's write lock from the first partition's updater (when
18//! the first block reaches the sink, after creating and seeding a
19//! `MODEL_DATA` it lacks) to the end of the pass. Today's route takes the
20//! lock and creates the column when its writer begins, before the first
21//! block is read.
22//!
23//! Each write leaves what it does not image: padding, samples the UV
24//! selection excludes, the channels of a batch's span the selection leaves
25//! out, the fit-only channels of a continuum transform, and the products
26//! not routed. Flagged samples are written, as CASA writes them.
27

44/// What a writing traversal queues for its sink, in delivery order.
45#[derive(Debug)]
46pub(super) enum Delivered {
47    /// A partition bound: the request the sink binds its updater to.
48    Partition(AccessRequest),
49    /// A block filled.
50    Block(Provenance),
51}

53/// Where a block's rows come from.
54#[derive(Debug)]
55pub(super) struct Provenance {
56    /// Its batch's address.
57    pub(super) address: BatchAddress,
58    /// The batch sample of each of its rows.
59    pub(super) samples: Vec<u32>,
60    /// The spectral window its rows carry.
61    pub(super) spectral_window: u32,
62}

64/// The provenance queue, and whether a sink waits for it.
65#[derive(Debug, Default)]
66struct Queue {
67    armed: bool,

71/// The update session a writing run reads and writes, the columns it
72/// writes, and the queue between its source and its sink.
73pub(crate) struct WriteBack {
74    session: UpdateSession,
75    /// Whether the session's targets are the model's (first) and the
76    /// corrected data's.
77    model: bool,
78    corrected: bool,
79    queue: Mutex<Queue>,
80}
81
82impl WriteBack {
83    /// An update session over `observation`, writing the model when
84    /// `model` and the corrected data when `corrected`. A `MODEL_DATA`
85    /// MAIN lacks is created with CASA's initial model; `CORRECTED_DATA` is
86    /// written only where it exists.
87    ///
88    /// # Errors
89    ///
90    /// When the session cannot open over the observation or a target
91    /// cannot be resolved ([`UpdateSession::over_msv2`]).
92    pub(crate) fn open(
93        observation: Msv2Observation,
94        model: bool,
95        corrected: bool,
96    ) -> Result<Self, ObservationError> {
97        let mut targets = Vec::with_capacity(2);
98        if model {
99            targets.push(UpdateTarget {
100                create: true,
101                ..UpdateTarget::new("model", WritableRole::Visibilities)
102            });
103        }
104        if corrected {
105            targets.push(UpdateTarget::new("corrected", WritableRole::Visibilities));
106        }
107        Ok(Self {
108            session: UpdateSession::over_msv2(observation, &targets)?,
109            model,
110            corrected,
111            queue: Mutex::default(),
112        })
113    }
114
115    /// The observation the session reads.
116    pub(crate) fn observation(&self) -> &dyn Observation {
117        self.session.observation()
118    }
119
120    /// The session's targets, every one of which each write gives values
121    /// for.
122    fn targets(&self) -> Vec<usize> {
123        (0..usize::from(self.model) + usize::from(self.corrected)).collect()
124    }
125
126    fn queue(&self) -> std::sync::MutexGuard<'_, Queue> {
127        self.queue.lock().unwrap_or_else(PoisonError::into_inner)
128    }
129
130    /// Whether the visibility traversal beginning now writes back: a sink
131    /// has begun and waits for it.
132    ///
133    /// # Errors
134    ///
135    /// When an earlier traversal's provenance was left unwritten.
136    pub(super) fn writing(&self) -> Result<bool, SourceError> {
137        let queue = self.queue();
138        if !queue.entries.is_empty() {
139            return Err(
140                "a writing traversal began before the last one's blocks were written".into(),
141            );
142        }
143        Ok(queue.armed)
144    }
145
146    /// Queue `entry` for the sink.
147    pub(super) fn deliver(&self, entry: Delivered) {
148        self.queue().entries.push_back(entry);
149    }
150
151    /// What one partition's write holds beyond the passes', for its
152    /// traversal `request` ([`WriteCharge`]).
153    pub(super) fn write_bytes(&self, request: &AccessRequest) -> Result<WriteCharge, SourceError> {
154        let targets = self.targets();
155        let updater = self.session.write_bytes(request, &targets)?;
156        let estimate = self.observation().estimate(request)?;
157        let partition = estimate
158            .partitions
159            .first()
160            .ok_or("an estimate without its partition")?;
161        let samples = partition.samples_per_batch();
162        let entry = self
163            .observation()
164            .catalogue()
165            .partition(partition.partition);
166        let products = match &request.correlations.deliver {
167            Deliver::These(products) => products.len(),
168            _ => entry.correlations.len(),
169        };
170        let elements = samples * request.channels.len() * products;
171        let patch = elements * (targets.len() * size_of::<Complex32>() + size_of::<bool>());
172        // The blocks in flight and the one being written, each with its
173        // batch's times and products.
174        let provenance = (STREAM_SLOTS + 1)
175            * (samples * size_of::<u32>()
176                + size_of::<Delivered>()
177                + partition.times_per_batch * size_of::<f64>()
178                + products * size_of::<Correlation>());
179        Ok(WriteCharge {
180            updater: updater as u64,
181            patch: patch as u64,
182            provenance: provenance as u64,
183        })
184    }
185}

187/// What one partition's write holds beyond the passes'.
188#[derive(Clone, Copy, Debug, Default)]
189pub(super) struct WriteCharge {
190    /// Its updater, writing its largest batch
191    /// ([`UpdateSession::write_bytes`]).
192    pub(super) updater: u64,
193    /// The patch the sink assembles for its largest batch: each column's
194    /// values and the mask over the batch's samples, channels and
195    /// products. The sink's buffers keep the largest partition's
196    /// capacity.
197    pub(super) patch: u64,
198    /// The provenance of the blocks in flight and the one being written.
199    pub(super) provenance: u64,
200}
201
202impl WriteCharge {
203    /// Each term's most over `self` and `other`: the sink's buffers keep
204    /// one partition's capacity while another's updater writes.
205    pub(super) fn most(self, other: Self) -> Self {
206        Self {
207            updater: self.updater.max(other.updater),
208            patch: self.patch.max(other.patch),
209            provenance: self.provenance.max(other.provenance),
210        }
211    }
212
213    /// The bytes charged.
214    pub(super) fn bytes(self) -> u64 {
215        self.updater + self.patch + self.provenance
216    }
217}

219/// The sink of a writing pass on the batch route: each block's values
220/// written to its batch, at the samples, channels and products the block
221/// holds.
222pub(crate) struct BatchWriter<'a> {
223    write_back: &'a WriteBack,
224    transform: Option<&'a SequentialContinuumTransform>,
225    updater: Option<Box<dyn Updater + 'a>>,
226    /// The patch's mask and each column's values, reused from batch to
227    /// batch.
228    written: Vec<bool>,
229    model: Vec<Complex32>,
230    corrected: Vec<Complex32>,
231    samples: u64,
232}
233
234impl<'a> BatchWriter<'a> {
235    /// A sink for the next visibility traversal, which the source then
236    /// records the provenance of. `CORRECTED_DATA` holds the output of
237    /// `transform`.
238    pub(crate) fn begin(
239        write_back: &'a WriteBack,
240        transform: Option<&'a SequentialContinuumTransform>,
241    ) -> Result<Self, SourceError> {
242        if write_back.corrected && transform.is_none() {
243            return Err("CORRECTED_DATA is written only by a continuum transform".into());
244        }
245        let mut queue = write_back.queue();
246        queue.entries.clear();
247        queue.armed = true;
248        drop(queue);
249        Ok(Self {
250            write_back,
251            transform,
252            updater: None,
253            written: Vec::new(),
254            model: Vec::new(),
255            corrected: Vec::new(),
256            samples: 0,
257        })
258    }
259
260    /// Whether the write needs model predictions.
261    pub(crate) const fn needs_predictions(&self) -> bool {
262        self.write_back.model
263    }
264
265    /// Write `block`'s predictions, `[row][channel][correlation]`, to the
266    /// model and its values to the corrected data, on the channels that
267    /// reach the line output.
268    pub(crate) fn write(
269        &mut self,
270        block: &NativeBlock,
271        predictions: &[Complex32],
272    ) -> Result<(), SourceError> {
273        let provenance = self.next_block()?;
274        check(block, &provenance)?;
275        let Provenance {
276            address, samples, ..
277        } = provenance;
278        let [batch_samples, channels, products] = address.shape();
279        let first = address.channels().start;
280        let entry = address.catalogue().partition(address.partition());
281        let channel_of = block
282            .channel_indices()
283            .iter()
284            .map(|channel| {
285                (*channel as usize)
286                    .checked_sub(first)
287                    .filter(|channel| *channel < address.channels().len())
288            })
289            .collect::<Option<Vec<_>>>()
290            .ok_or("a block holds a channel its batch does not deliver")?;
291        let product_of = block
292            .correlation_indices()
293            .iter()
294            .map(|index| {
295                address
296                    .correlations()
297                    .iter()
298                    .position(|product| *product == entry.correlations[*index as usize])
299            })
300            .collect::<Option<Vec<_>>>()
301            .ok_or("a block holds a product its batch does not deliver")?;
302        let len = batch_samples * channels * products;
303        let (model, corrected) = (self.write_back.model, self.write_back.corrected);
304        reset(&mut self.written, len, true, false);
305        reset(&mut self.model, len, model, Complex32::default());
306        reset(&mut self.corrected, len, corrected, Complex32::default());
307        let cells = block.channels() * block.correlations();
308        let columns = u64::from(model) + u64::from(corrected);
309        for (row, sample) in samples.iter().enumerate() {
310            let context = block.header(row).context;
311            let rule = self.transform.and_then(|transform| {
312                transform.rule(context.field as i32, context.spectral_window)
313            });
314            let values = block.row(0, row).values;
315            for (k, channel) in channel_of.iter().enumerate() {
316                if !writes_channel(rule, block.channel_indices()[k]) {
317                    continue;
318                }
319                for (j, product) in product_of.iter().enumerate() {
320                    let at = (*sample as usize * channels + channel) * products + product;
321                    let cell = k * block.correlations() + j;
322                    self.written[at] = true;
323                    if model {
324                        self.model[at] = predictions[row * cells + cell];
325                    }
326                    if corrected {
327                        self.corrected[at] = values[cell];
328                    }
329                    self.samples += columns;
330                }
331            }
332        }
333        let shape = (batch_samples, channels, products);
334        let written = ArrayView3::from_shape(shape, &self.written)?;
335        let mut patch = Vec::with_capacity(2);
336        for (writes, values) in [(model, &self.model), (corrected, &self.corrected)] {
337            if writes {
338                patch.push(PatchValues::Visibilities {
339                    values: ArrayView3::from_shape(shape, values)?,
340                    written: Some(written),
341                });
342            }
343        }
344        self.updater
345            .as_mut()
346            .ok_or("a block before its partition")?
347            .write_at(&address, &patch)?;
348        Ok(())
349    }
350
351    /// The next block's provenance, binding the updater of each partition
352    /// queued before it.
353    fn next_block(&mut self) -> Result<Provenance, SourceError> {
354        loop {
355            let next = self.write_back.queue().entries.pop_front();
356            match next.ok_or("a block without its batch's provenance")? {
357                Delivered::Partition(request) => self.bind(&request)?,
358                Delivered::Block(provenance) => return Ok(provenance),
359            }
360        }
361    }
362
363    /// Bind the updater of the partition `request` reads, then finish the
364    /// last partition's: the two overlap, so an MSv2 session keeps MAIN's
365    /// lock.
366    fn bind(&mut self, request: &AccessRequest) -> Result<(), SourceError> {
367        let updater = self
368            .write_back
369            .session
370            .updater(request, &self.write_back.targets())?;
371        if let Some(last) = self.updater.replace(updater) {
372            last.finish()?;
373        }
374        Ok(())
375    }
376
377    /// Finish the last partition's update; returns the cells written.
378    pub(crate) fn complete(mut self) -> Result<u64, SourceError> {
379        // Partitions bound after the last block hold nothing to write, but
380        // their updaters prepare the columns as the others did.
381        let rest: Vec<Delivered> = self.write_back.queue().entries.drain(..).collect();
382        for entry in rest {
383            let Delivered::Partition(request) = entry else {
384                return Err("a block's provenance was left unwritten".into());
385            };
386            self.bind(&request)?;
387        }
388        if let Some(last) = self.updater.take() {
389            last.finish()?;
390        }
391        Ok(self.samples)
392    }
393}

395impl Drop for BatchWriter<'_> {
396    fn drop(&mut self) {
397        let mut queue = self.write_back.queue();
398        queue.armed = false;
399        queue.entries.clear();
400    }
401}

403/// `buffer` as `len` elements of `fill`, or empty when not `used`; grown to
404/// exactly `len` when it holds less, so its capacity is the largest batch's.
405fn reset<T: Copy>(buffer: &mut Vec<T>, len: usize, used: bool, fill: T) {
406    buffer.clear();
407    if used {
408        buffer.reserve_exact(len);
409        buffer.resize(len, fill);
410    }
411}

413/// Refuse a block whose provenance is not its own: as many rows as the
414/// samples recorded, each in the recorded spectral window, at its sample's
415/// time and on its baseline. The times are the batch's own (the source
416/// takes a row's from the same epochs), so a block of another batch of the
417/// partition, which shares its baselines, is refused too, as is a block of
418/// another window's partition at the same times.
419fn check(block: &NativeBlock, provenance: &Provenance) -> Result<(), SourceError> {
420    let Provenance {
421        address,
422        samples,
423        spectral_window,
424    } = provenance;
425    let baselines = &address.catalogue().partition(address.partition()).baselines;
426    let time = address.time();
427    let mjd_zero = MjdHighPrec::new(0.0, 0.0);
428    let matches = samples.len() == block.len()
429        && samples.iter().enumerate().all(|(row, sample)| {
430            let sample = *sample as usize;
431            let context = &block.header(row).context;
432            sample < address.samples() && {
433                let baseline = baselines[sample % baselines.len()];
434                context.spectral_window == *spectral_window
435                    && context.antennas == [baseline.antenna1.0, baseline.antenna2.0]
436                    && context.time_s.to_bits()
437                        == time
438                            .seconds_since(sample / baselines.len(), mjd_zero)
439                            .to_bits()
440            }
441        });
442    if matches {
443        Ok(())
444    } else {
445        Err("a block's rows are not the samples its source recorded for it".into())
446    }
