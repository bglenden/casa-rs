3//! Bounded Högbom Minor Cycle over authoritative Normal State views.
4//!
5//! The Minor Cycle is a pure scientific operation: it consumes one immutable
6//! view of an authoritative Final Normal State (the T20 residual paired with
7//! its normal approximation), one named authoritative model generation, an
8//! explicit valid-support window, and explicit gain, threshold, iteration,
9//! and staleness controls. It accumulates Högbom components against the PSF
10//! exactly like the casacore `LatticeCleaner` HOGBOM loop — peak residual
11//! normalized by the PSF peak, gain-scaled PSF subtraction — and mints the
12//! resulting sparse update only through the T28 model lifecycle owner as one
13//! base-bound Model Delta. Authoritative model and residual state are never
14//! mutated; the solver works on private buffers.
15//!
16//! Every non-converged stop (`IterationBound`, `StalenessBound`) records that
17//! the linear approximation envelope is exhausted and explicitly requests
18//! Major-Cycle reconciliation. Component-sequence recording is optional
19//! diagnostic evidence: first-divergence comparisons between two recorded
20//! sequences are informational and never gate acceptance.
21

83/// Explicit Högbom Minor-Cycle controls.
84///
85/// Every control is explicit and validated; there are no defaults that could
86/// silently change deconvolution semantics. `maximum_model_update` is the
87/// error/staleness bound of the linear view envelope: before any component
88/// is applied, the candidate cumulative absolute flux is checked against it,
89/// and a candidate that would exceed the bound is rejected without touching
90/// the residual, the delta, recorded diagnostics, or evidence counters. The
91/// solve then stops and requests Major-Cycle reconciliation.
92#[derive(Debug, Clone, Copy, PartialEq)]
93pub struct HogbomControls {
94    gain: f64,
95    threshold: f64,
96    max_iterations: usize,
97    maximum_model_update: f64,
98    component_sequence_limit: Option<usize>,
99}
100
101impl HogbomControls {
102    /// Derive the executable minor-cycle controls from the compiled contract.
103    ///
104    /// Unlike the general model constructor, this production seam requires an
105    /// explicit staleness envelope and never invents an unbounded default.
106    pub fn from_compiled(controls: ReconstructionControls) -> Result<Self, MinorCycleError> {
107        let maximum_model_update = controls
108            .maximum_model_update()
109            .ok_or(MinorCycleError::MissingMaximumModelUpdate)?;
110        Self::new(
111            controls.gain(),
112            controls.threshold_jy_per_beam(),
113            controls.max_minor_iterations(),
114            maximum_model_update,
115        )
116    }
117
118    /// Construct validated controls.
119    ///
120    /// # Errors
121    ///
122    /// Rejects a gain outside `(0, 1]`, a negative or non-finite threshold, a
123    /// zero iteration bound, and a non-positive or non-finite maximum model
124    /// update.
125    pub fn new(
126        gain: f64,
127        threshold: f64,
128        max_iterations: usize,
129        maximum_model_update: f64,
130    ) -> Result<Self, MinorCycleError> {
131        if !(gain > 0.0 && gain <= 1.0) {
132            return Err(MinorCycleError::InvalidGain);
133        }
134        if !threshold.is_finite() || threshold < 0.0 {
135            return Err(MinorCycleError::InvalidThreshold);
136        }
137        if max_iterations == 0 {
138            return Err(MinorCycleError::InvalidIterationBound);
139        }
140        if !maximum_model_update.is_finite() || maximum_model_update <= 0.0 {
141            return Err(MinorCycleError::InvalidMaximumModelUpdate);
142        }
143        Ok(Self {
144            gain,
145            threshold,
146            max_iterations,
147            maximum_model_update,
148            component_sequence_limit: None,
149        })
150    }
151
152    /// Record the first accepted components as diagnostic evidence.
153    ///
154    /// Recording is capped so a long clean cannot retain unbounded component
155    /// lists; cleaning itself continues past the cap.
156    ///
157    /// # Errors
158    ///
159    /// Rejects a zero limit.
160    pub fn record_component_sequence(mut self, limit: usize) -> Result<Self, MinorCycleError> {
161        if limit == 0 {
162            return Err(MinorCycleError::InvalidRecordingLimit);
163        }
164        self.component_sequence_limit = Some(limit);
165        Ok(self)
166    }
167
168    /// Return the loop gain fraction applied to each normalized peak.
169    #[must_use]
170    pub const fn gain(&self) -> f64 {
171        self.gain
172    }
173
174    /// Return the absolute normalized-flux stopping threshold.
175    #[must_use]
176    pub const fn threshold(&self) -> f64 {
177        self.threshold
178    }
179
180    /// Return the hard iteration bound.
181    #[must_use]
182    pub const fn max_iterations(&self) -> usize {
183        self.max_iterations
184    }
185
186    /// Return the cumulative absolute component flux that exhausts the view
187    /// envelope and forces an explicit reconciliation request.
188    #[must_use]
189    pub const fn maximum_model_update(&self) -> f64 {
190        self.maximum_model_update
191    }
192
193    /// Return the optional recorded-component-sequence capacity.
194    #[must_use]
195    pub const fn component_sequence_limit(&self) -> Option<usize> {
196        self.component_sequence_limit
197    }
198}

200/// Explicit valid-support window constraining component placement.
201///
202/// Bounds are inclusive pixel coordinates `[x, y]`. Only window cells whose
203/// named model-generation sample has valid support may host a component,
204/// because only the model owner accepts delta terms on valid support.
205#[derive(Debug, Clone, Copy, PartialEq, Eq)]
206pub struct CleanWindow {
207    blc: [usize; 2],
208    trc: [usize; 2],
209}
210
211impl CleanWindow {
212    /// Construct an inclusive window with `blc <= trc` on both axes.
213    ///
214    /// # Errors
215    ///
216    /// Rejects an inverted window.
217    pub fn new(blc: [usize; 2], trc: [usize; 2]) -> Result<Self, MinorCycleError> {
218        if blc[0] > trc[0] || blc[1] > trc[1] {
219            return Err(MinorCycleError::InvertedWindow);
220        }
221        Ok(Self { blc, trc })
222    }
223
224    /// The CASA default inner-quarter window for one plane shape.
225    #[must_use]
226    pub fn inner_quarter(shape: [usize; 2]) -> Self {
227        let half = [shape[0] / 2, shape[1] / 2];
228        let quarter = [shape[0] / 4, shape[1] / 4];
229        let mut blc = quarter;
230        let mut trc = [
231            quarter[0].saturating_add(half[0]).saturating_sub(1),
232            quarter[1].saturating_add(half[1]).saturating_sub(1),
233        ];
234        for axis in 0..2 {
235            trc[axis] = trc[axis].min(shape[axis] - 1);
236            blc[axis] = blc[axis].min(trc[axis]);
237        }
238        Self { blc, trc }
239    }
240
241    /// The full plane for one shape.
242    #[must_use]
243    pub fn full_plane(shape: [usize; 2]) -> Self {
244        Self {
245            blc: [0, 0],
246            trc: [shape[0] - 1, shape[1] - 1],
247        }
248    }
249
250    /// Return the inclusive lower bound.
251    #[must_use]
252    pub const fn blc(&self) -> [usize; 2] {
253        self.blc
254    }
255
256    /// Return the inclusive upper bound.
257    #[must_use]
258    pub const fn trc(&self) -> [usize; 2] {
259        self.trc
260    }
261
262    fn contains(&self, pixel: [usize; 2]) -> bool {
263        pixel[0] >= self.blc[0]
264            && pixel[0] <= self.trc[0]
265            && pixel[1] >= self.blc[1]
266            && pixel[1] <= self.trc[1]
267    }
268}

326/// Why one bounded Högbom solve stopped.
327#[derive(Debug, Clone, Copy, PartialEq, Eq)]
328pub enum MinorCycleStopReason {
329    /// The normalized residual peak fell strictly below the explicit
330    /// threshold (the casacore HOGBOM convention; a peak exactly at the
331    /// threshold still cleans one component).
332    ThresholdReached,
333    /// The hard iteration bound was reached with work potentially remaining.
334    IterationBound,
335    /// Accepting the next candidate would push the cumulative component
336    /// flux past the maximum model update, so continuing would extrapolate
337    /// the frozen approximation beyond its validity envelope. The rejected
338    /// candidate left no trace in the delta or evidence.
339    StalenessBound,
340}

434    /// A threshold stop needs no reconciliation only when it accepted no

437    /// even when the working residual subsequently falls below threshold.
438    /// Iteration and staleness stops request reconciliation unconditionally.

447        self.recorded.as_deref()

450    /// Compare this evidence's recorded sequence against another baseline.
451    ///
452    /// Returns the first index where the sequences differ in pixel or exact
453    /// flux bits, or where one sequence ends early. Comparison is purely
454    /// informational and never gates acceptance; `None` means the sequences
455    /// agree or cannot be compared (either side unrecorded).
456    #[must_use]
457    pub fn first_divergence(&self, baseline: &Self) -> Option<ComponentDivergence> {
458        let candidate = self.recorded.as_deref()?;
459        let baseline = baseline.recorded.as_deref()?;
460        let shared = candidate.len().min(baseline.len());
461        for index in 0..shared {
462            if !baseline[index].same_value(&candidate[index]) {
463                return Some(ComponentDivergence {
464                    index,
465                    baseline: Some(baseline[index]),
466                    candidate: Some(candidate[index]),
467                });
468            }
469        }
470        if baseline.len() != candidate.len() {
471            let index = shared;
472            return Some(ComponentDivergence {
473                index,
474                baseline: baseline.get(index).copied(),
475                candidate: candidate.get(index).copied(),
476            });
477        }
478        None
479    }

510/// Exact reason a bounded Högbom solve failed closed.
511#[derive(Debug, Clone, PartialEq, Eq, Error)]
512pub enum MinorCycleError {
513    /// The compiled problem omitted the mandatory linear-view envelope.
514    #[error("compiled Högbom controls require an explicit maximum model update")]
515    MissingMaximumModelUpdate,
516    /// The Högbom gain was outside `(0, 1]`.
517    #[error("Högbom gain must lie in (0, 1]")]
518    InvalidGain,
519    /// The stopping threshold was negative or non-finite.
520    #[error("Högbom threshold must be finite and non-negative")]
521    InvalidThreshold,
522    /// The iteration bound was zero.
523    #[error("Högbom iteration bound must be non-zero")]
524    InvalidIterationBound,
525    /// The maximum model update was non-positive or non-finite.
526    #[error("Högbom maximum model update must be finite and positive")]
527    InvalidMaximumModelUpdate,
528    /// Component-sequence recording requested a zero capacity.
529    #[error("component-sequence recording requires a non-zero limit")]
530    InvalidRecordingLimit,
531    /// The window bounds were inverted.
532    #[error("clean window requires blc <= trc on both axes")]
533    InvertedWindow,
534    /// The window lay partly or wholly outside the normal-state plane.
535    #[error("clean window lies outside the normal-state plane")]
536    WindowOutsidePlane,
537    /// The window contained no valid model support at all.
538    #[error("clean window contains no valid model support")]
539    EmptyValidSupport,
540    /// The named generation does not match the single-plane constant-basis
541    /// normal-state geometry.
542    #[error("model generation does not match the normal-state plane")]
543    ModelShapeMismatch,
544    /// The named generation is not the view's final model generation.
545    #[error("named generation is not the normal state's final model generation")]
546    ForeignNormalState,
547    /// The normal approximation lacked a positive finite peak.
548    #[error("normal-state approximation lacks a positive finite PSF peak")]
549    InvalidPsfPeak,
550    /// Solver arithmetic produced a non-finite value.
551    #[error("minor-cycle arithmetic generated a non-finite value")]
552    GeneratedNonfinite,
553    /// The model lifecycle owner rejected delta validation or minting.
554    #[error(transparent)]
555    Lifecycle(#[from] crate::ModelLifecycleError),
556}

564/// Run one bounded Högbom Minor Cycle.
565///
566/// Consumes an immutable view of one authoritative Final Normal State and the
567/// exact named final model generation, and returns the sparse Model Delta
568/// minted through the model lifecycle owner plus solver evidence. Neither the
569/// authoritative residual planes nor the model generation are mutated.
570///
571/// The loop preserves the casacore HOGBOM semantics: find the maximum-
572/// absolute residual over the valid-support window, normalize by the PSF
573/// peak, stop when below threshold, otherwise subtract `gain * strength`
574/// scaled by the PSF centered on the peak and accumulate the same flux at the
575/// peak cell. Scalar Stokes-I convention uses the real plane, matching the
576/// Float-image reference behavior.
577///
578/// # Errors
579///
580/// Fails closed on mismatched lineage or geometry, invalid controls or
581/// windows, a degenerate PSF peak, non-finite arithmetic, and every model
582/// owner rejection (term bounds, value bounds, or foreign lineage).
583pub fn hogbom_minor_cycle(
584    lifecycle: &ModelLifecycle,
585    base: &ModelGeneration,
586    view: &FinalNormalState,
587    window: CleanWindow,
588    controls: HogbomControls,
589) -> Result<HogbomMinorCycle, MinorCycleError> {
590    let shape = view.shape();
591    let cells = shape[0] * shape[1];
592    if base.shape().domains().len() != 1
593        || base.shape().coefficients() != 1
594        || base.shape().polarizations() != 1
595        || base.shape().domains()[0].pixels() != shape
596        || base.samples().len() != cells
597    {
598        return Err(MinorCycleError::ModelShapeMismatch);
599    }
600    if base.generation_id() != view.final_model_generation() {
601        return Err(MinorCycleError::ForeignNormalState);
602    }
603    if window.trc()[0] >= shape[0] || window.trc()[1] >= shape[1] {
604        return Err(MinorCycleError::WindowOutsidePlane);
605    }
606
607    // The PSF peak normalization follows the reference cleaner: peaks are
608    // reported in model units regardless of the accumulated weight scale.
609    let psf_peak_index = find_peak_abs(view.normal_approximation(), shape, |v| v.re, |_| true)
610        .ok_or(MinorCycleError::InvalidPsfPeak)?;
611    let psf_peak = view.normal_approximation()[psf_peak_index].re;
612    if !psf_peak.is_finite() || psf_peak <= 0.0 {
613        return Err(MinorCycleError::InvalidPsfPeak);
614    }
615    let psf_peak_pixel = plane_pixel(psf_peak_index, shape);
616
617    // Private working copy: authoritative state is never mutated.
618    let mut residual = Vec::with_capacity(cells);
619    for value in view.residual() {
620        let real = value.re;
621        if !real.is_finite() {
622            return Err(MinorCycleError::GeneratedNonfinite);
623        }
624        residual.push(real);
625    }
626
627    let mut terms = BTreeMap::<usize, f64>::new();
628    let mut recorded = Vec::with_capacity(
629        controls
630            .component_sequence_limit()
631            .unwrap_or(0)
632            .min(controls.max_iterations()),
633    );
634    let mut iterations = 0_usize;
635    let mut total_flux = 0.0_f64;
636    let mut final_peak_flux = 0.0_f64;
637    let mut stop_reason = None;
638
639    for _ in 0..controls.max_iterations() {
640        let Some(peak_index) = find_peak_abs(
641            &residual,
642            shape,
643            |value| *value,
644            |pixel| window.contains(pixel) && valid_support(base, shape, pixel),
645        ) else {
646            return Err(MinorCycleError::EmptyValidSupport);
647        };
648        let peak_pixel = plane_pixel(peak_index, shape);
649        let strength = residual[peak_index] / psf_peak;
650        if !strength.is_finite() {
651            return Err(MinorCycleError::GeneratedNonfinite);
652        }
653        final_peak_flux = strength.abs();
654        // The casacore HOGBOM cleaner stops only when the normalized peak is
655        // strictly below the threshold (`lattices/LatticeMath/
656        // LatticeCleaner.tcc`, stopping rule 1: "stop if below threshold",
657        // tested as `abs(itsStrengthOptimum) < threshold()`), so a peak
658        // exactly at the threshold still cleans one component. A zero peak
659        // has no flux to clean and converges trivially.
660        if strength == 0.0 || strength.abs() < controls.threshold() {
661            stop_reason = Some(MinorCycleStopReason::ThresholdReached);
662            break;
663        }
664        let flux = controls.gain() * strength;
665        if !flux.is_finite() {
666            return Err(MinorCycleError::GeneratedNonfinite);
667        }
668        // Envelope guard: a candidate whose acceptance would push the
669        // cumulative absolute update past the maximum model update is
670        // rejected before it can touch the working residual, the accumulated
671        // delta terms, the recorded component sequence, or any evidence
672        // counter. Accepted components therefore never exceed the linear
673        // view envelope.
674        if total_flux + flux.abs() > controls.maximum_model_update() {
675            stop_reason = Some(MinorCycleStopReason::StalenessBound);
676            break;
677        }
678        subtract_psf(
679            &mut residual,
680            view.normal_approximation(),
681            shape,
682            peak_pixel,
683            psf_peak_pixel,
684            flux,
685        )?;
686        iterations += 1;
687        total_flux += flux.abs();
688        let cell =
689            model_cell(shape, peak_pixel).expect("a scanned peak pixel lies inside the plane");
690        *terms.entry(canonical_flat(base, cell)).or_insert(0.0) += flux;
691        if let Some(limit) = controls.component_sequence_limit()
692            && recorded.len() < limit
693        {
694            recorded.push(HogbomComponent { cell, flux });
695        }
696    }
697    let stop_reason = stop_reason.unwrap_or(MinorCycleStopReason::IterationBound);
698
699    terms.retain(|_, flux| *flux != 0.0);
700    let delta = if terms.is_empty() {
701        None
702    } else {
703        let deltas = terms
704            .iter()
705            .map(|(flat, flux)| {
706                let cell = base
707                    .shape()
708                    .cell_at(*flat)
709                    .expect("accumulated flat keys stay inside the base shape");
710                Ok(ModelDeltaTerm::new(cell, ModelValue::new(*flux)?))
711            })
712            .collect::<Result<Vec<_>, MinorCycleError>>()?;
713        Some(lifecycle.compile_delta(base, deltas)?)
714    };
715
716    let evidence_id = minor_cycle_evidence_id(
717        lifecycle.authority(),
718        lifecycle.attempt(),
719        lifecycle.epoch(),
720        base.generation_id(),
721        view.completion_id(),
722        view.content_identity(),
723        &window,
724        &controls,
725        iterations,
726        total_flux,
727        final_peak_flux,
728        stop_reason,
729    );
730    Ok(HogbomMinorCycle {
731        delta,
732        evidence: MinorCycleEvidence {
733            evidence_id,
734            problem: lifecycle.problem(),
735            attempt: lifecycle.attempt(),
736            epoch: lifecycle.epoch(),
737            input_generation: base.generation_id(),
738            normal_state_completion: view.completion_id(),
739            normal_state_content: view.content_identity(),
740            iterations,
741            total_flux,
742            final_peak_flux,
743            stop_reason,
744            recorded: (!recorded.is_empty()).then(|| recorded.into_boxed_slice()),
745        },
746    })
747}

749/// Canonical single-plane pixel of one plane-storage index.
750///
751/// Normal-state planes are stored x-major (`finish_bound` pushes x outer, y
752/// inner), so index `p` maps to pixel `[p / H, p % H]`.
753fn plane_pixel(index: usize, shape: [usize; 2]) -> [usize; 2] {
754    [index / shape[1], index % shape[1]]
755}
756
757fn model_cell(shape: [usize; 2], pixel: [usize; 2]) -> Option<ModelCell> {
758    if pixel[0] < shape[0] && pixel[1] < shape[1] {
759        Some(ModelCell::new(0, 0, 0, pixel))
760    } else {
761        None
762    }
763}
764
765fn canonical_flat(base: &ModelGeneration, cell: ModelCell) -> usize {
766    base.shape()
767        .flat_index(cell)
768        .expect("component pixels stay inside the base shape")
769}
770
771fn valid_support(base: &ModelGeneration, shape: [usize; 2], pixel: [usize; 2]) -> bool {
772    let Some(cell) = model_cell(shape, pixel) else {
773        return false;
774    };
775    base.samples()[canonical_flat(base, cell)].support() == ModelSupport::Valid
776}

778/// Find the maximum-abs real plane value passing `accept`, scanning in
779/// canonical storage order so ties deterministically keep the first peak.
780fn find_peak_abs<T>(
781    plane: &[T],
782    shape: [usize; 2],
783    magnitude: impl Fn(&T) -> f64,
784    accept: impl Fn([usize; 2]) -> bool,
785) -> Option<usize> {
786    let mut best: Option<(f64, usize)> = None;
787    for (index, value) in plane.iter().enumerate() {
788        let magnitude = magnitude(value).abs();
789        if best.is_some_and(|(best_magnitude, _)| magnitude <= best_magnitude) {
790            continue;
791        }
792        let pixel = plane_pixel(index, shape);
793        if accept(pixel) {
794            best = Some((magnitude, index));
795        }
796    }
797    best.map(|(_, index)| index)
798}

800/// Subtract `flux * psf` centered on `peak` from the working residual.
801fn subtract_psf(
802    residual: &mut [f64],
803    psf: &[num_complex::Complex64],
804    shape: [usize; 2],
805    peak: [usize; 2],
806    psf_peak: [usize; 2],
807    flux: f64,
808) -> Result<(), MinorCycleError> {
809    // psf_shifted(x, y) = psf(x - peak + psf_peak, y - peak + psf_peak),
810    // clipped to the plane exactly like the reference cleaner's subregion.
811    let x_range = overlap(peak[0], psf_peak[0], shape[0]);
812    let y_range = overlap(peak[1], psf_peak[1], shape[1]);
813    for y in y_range.clone() {
814        for x in x_range.clone() {
815            let source = [x + psf_peak[0] - peak[0], y + psf_peak[1] - peak[1]];
816            let index = source[0] * shape[1] + source[1];
817            let target = x * shape[1] + y;
818            let updated = residual[target] - flux * psf[index].re;
819            if !updated.is_finite() {
820                return Err(MinorCycleError::GeneratedNonfinite);
821            }
822            residual[target] = updated;
823        }
824    }
825    Ok(())
826}
827
828/// Inclusive-clipped axis range where a PSF centered at `psf_peak` overlaps
829/// the plane when recentered on `peak`.
830fn overlap(peak: usize, psf_peak: usize, length: usize) -> std::ops::Range<usize> {
831    let shift = psf_peak as isize - peak as isize;
832    let low = (-shift).max(0) as usize;
833    let high_exclusive = (length as isize - shift)
834        .min(length as isize)
835        .max(low as isize) as usize;
836    low..high_exclusive
837}

839#[allow(clippy::too_many_arguments)]
840fn minor_cycle_evidence_id(
841    authority: LogicalIdentity,
842    attempt: ModelExecutionAttemptId,
843    epoch: u64,
844    input_generation: ModelGenerationId,
845    normal_state_completion: FinalNormalStateCompletionId,
846    normal_state_content: LogicalIdentity,
847    window: &CleanWindow,
848    controls: &HogbomControls,
849    iterations: usize,
850    total_flux: f64,
851    final_peak_flux: f64,
852    stop_reason: MinorCycleStopReason,
853) -> MinorCycleEvidenceId {
854    let mut encoder = Encoder::new(MINOR_CYCLE_EVIDENCE_DOMAIN, MINOR_CYCLE_EVIDENCE_VERSION);
855    encoder.identity(authority.as_bytes());
856    encoder.identity(attempt.identity().as_bytes());
857    encoder.u64(epoch);
858    encoder.identity(input_generation.as_bytes());
859    encoder.identity(normal_state_completion.as_bytes());
860    encoder.identity(normal_state_content.as_bytes());
861    encoder.usize(window.blc()[0]);
862    encoder.usize(window.blc()[1]);
863    encoder.usize(window.trc()[0]);
864    encoder.usize(window.trc()[1]);
865    encoder.u64(crate::canonical_f64_bits(controls.gain()));
866    encoder.u64(crate::canonical_f64_bits(controls.threshold()));
867    encoder.usize(controls.max_iterations());
868    encoder.u64(crate::canonical_f64_bits(controls.maximum_model_update()));
869    match controls.component_sequence_limit() {
870        None => encoder.u8(0),
871        Some(limit) => {
872            encoder.u8(1);
873            encoder.usize(limit);
874        }
875    }
876    encoder.usize(iterations);
877    encoder.u64(crate::canonical_f64_bits(total_flux));
878    encoder.u64(crate::canonical_f64_bits(final_peak_flux));
879    encoder.u8(match stop_reason {
880        MinorCycleStopReason::ThresholdReached => 0,
881        MinorCycleStopReason::IterationBound => 1,
882        MinorCycleStopReason::StalenessBound => 2,
883    });
884    MinorCycleEvidenceId(LogicalIdentity::from_sha256(encoder.finish()))
885}
