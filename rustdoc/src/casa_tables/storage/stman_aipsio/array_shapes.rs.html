5pub(crate) fn read_stman_array_column_shapes(
6    path: &Path,
7    columns: &[ColumnDescContents],
8    column: usize,
9    rows: &[usize],
10    byte_order: ByteOrder,
11) -> Result<Option<Vec<Option<Vec<usize>>>>, StorageError> {
12    if rows.is_empty() {
13        return Ok(Some(Vec::new()));
14    }
15    let descriptor = columns.get(column).ok_or_else(|| {
16        StorageError::FormatMismatch("StManAipsIO shape column index is out of range".to_string())
17    })?;
18    if descriptor.is_array && descriptor.option & 1 != 0 && !descriptor.shape.is_empty() {
19        let shape = descriptor
20            .shape
21            .iter()
22            .map(|&size| size as usize)
23            .collect::<Vec<_>>();
24        return Ok(Some(vec![Some(shape); rows.len()]));
25    }
26    let layouts = parse_sparse_column_layouts(path, columns, byte_order)?;
27    let Some(Some(SparseColumnLayout::IndirectArrayOffsets { extents, .. })) = layouts.get(column)
28    else {
29        return Ok(None);
30    };
31    let mut file = std::fs::File::open(path)?;
32    let mut offsets = Vec::with_capacity(rows.len());
33    for &row in rows {
34        file.seek(SeekFrom::Start(locate_extent_slot(extents, row)?))?;
35        offsets.push(read_indirect_offset(&mut file)?);
36    }
37    let array_path = indirect_array_file_path(path);
38    if offsets.iter().all(|&offset| offset == 0) {
39        return Ok(Some(vec![None; rows.len()]));
40    }
41    let mut reader = StManArrayFileReader::open(&array_path, byte_order == ByteOrder::BigEndian)?;
42    offsets
43        .into_iter()
44        .map(|offset| reader.read_shape_at(offset))
45        .collect::<Result<Vec<_>, _>>()
46        .map(Some)
47}

49/// Large AipsIO offsets are marker+i64, interleaved with ordinary u32s.
50/// Keep contiguous runs of equal-width entries so requested rows can seek
51/// directly without reading any array payload.
52pub(super) fn parse_indirect_offsets_layout(
53    file: &mut std::fs::File,
54    nrrow: usize,
55    data_type: CasacoreDataType,
56    large_offsets: bool,
57) -> Result<Option<SparseColumnLayout>, StorageError> {
58    let mut extents: Vec<ExtentLayout> = Vec::new();
59    let mut row_start = 0usize;
60    while row_start < nrrow {
61        let raw_count = read_be_u32(file)? as usize;
62        let row_count = if raw_count == 0 {
63            nrrow - row_start
64        } else {
65            raw_count
66        };
67        if row_count > nrrow - row_start {
68            return Err(StorageError::FormatMismatch(
69                "AipsIO offset extent exceeds row count".into(),
70            ));
71        }
72        if !large_offsets {
73            let values_start = file.stream_position()?;
74            file.seek(SeekFrom::Current((row_count * size_of::<u32>()) as i64))?;
75            extents.push(ExtentLayout {
76                row_start,
77                row_count,
78                values_start,
79                row_width_bytes: size_of::<u32>(),
80            });
81            row_start += row_count;
82            continue;
83        }
84        for row in row_start..row_start + row_count {
85            let values_start = file.stream_position()?;
86            let _offset = read_indirect_offset(file)?;
87            let width = (file.stream_position()? - values_start) as usize;
88            if let Some(previous) = extents.last_mut().filter(|previous| {
89                previous.row_width_bytes == width
90                    && previous.values_start + (previous.row_count * width) as u64 == values_start
91            }) {
92                previous.row_count += 1;
93            } else {
94                extents.push(ExtentLayout {
95                    row_start: row,
96                    row_count: 1,
97                    values_start,
98                    row_width_bytes: width,
99                });
100            }
101        }
102        row_start += row_count;
103    }
104    Ok(Some(SparseColumnLayout::IndirectArrayOffsets {
105        data_type,
106        extents,
107    }))
108}

110fn read_indirect_offset(file: &mut std::fs::File) -> Result<i64, StorageError> {
111    let offset = read_be_u32(file)?;
112    if offset != LARGE_OFFSET_MARKER {
113        return Ok(i64::from(offset));
114    }
115    let mut bytes = [0u8; 8];
116    file.read_exact(&mut bytes)?;
117    Ok(i64::from_be_bytes(bytes))
118}
