2//! The spectral pass of a regrid: casacore's `ImageRegrid::regrid1D` with
3//! `Interpolate2D::LINEAR`, as `SIImageStore::regridToModelImage` regrids a
4//! start model's spectral axis.
5//!
6//! Each output channel is converted to world frequency on the template and
7//! back to a pixel on the input (`make1DCoordinateGrid`), and every line of
8//! the image along the spectral axis is interpolated there by casacore's
9//! `InterpolateArray1D::linear` with flags, good being true and without
10//! extrapolation, in single precision as casacore runs it for a `Float`
11//! image:
12//! - between two input channels, their linear interpolation, good where
13//!   both are;
14//! - within single-precision rounding of an input channel, that channel;
15//! - beyond either end, the end channel's value, masked.
16//!
17//! The pixel values beyond the ends are therefore the end channels' rather
18//! than zero, as casacore writes them into an image without a mask (a
19//! tclean model): only the mask says they were not sampled.
20

26/// How the spectral axis is regridded: its pixel axis and the input pixel
27/// at each output channel, `None` where the conversion fails.
28pub(super) struct SpectralPlan {
29    axis: usize,
30    positions: Vec<Option<f32>>,
31}
32
33impl SpectralPlan {
34    /// The plan for spectral coordinates `input` of `input_channels` and
35    /// `output` of `output_channels` on pixel axis `axis`.
36    ///
37    /// # Errors
38    ///
39    /// The reason when the two are not both spectral, are in different
40    /// frames (casa-rs makes no frequency conversion machine), the input
41    /// holds one channel, which casacore's linear interpolation reads past,
42    /// or the output does and the input more: `ImageRegrid::_checkAxes`
43    /// removes an axis of unit length from those it regrids and then refuses
44    /// the shapes.
45    pub(super) fn new(
46        (input, input_channels): (&CoordinateModel, usize),
47        (output, output_channels): (&CoordinateModel, usize),
48        axis: usize,
49    ) -> Result<Self, String> {
50        let (CoordinateModel::Spectral(ours), CoordinateModel::Spectral(theirs)) = (input, output)
51        else {
52            return Err(format!(
53                "the {} axes differ between the input and the template; casa-rs regrids the \
54                 direction and spectral axes only",
55                input.coordinate_type()
56            ));
57        };
58        if ours.world_frequency_ref() != theirs.world_frequency_ref() {
59            return Err(format!(
60                "the input's frequencies are {:?} and the template's {:?}; casa-rs does not \
61                 convert between spectral frames",
62                ours.world_frequency_ref(),
63                theirs.world_frequency_ref()
64            ));
65        }
66        if input_channels < 2 {
67            return Err(format!(
68                "the input has {input_channels} channel and the template {output_channels}; a \
69                 spectral regrid interpolates between two channels"
70            ));
71        }
72        if output_channels == 1 {
73            return Err(format!(
74                "any axis not being regridded must have the same input and output shapes: the \
75                 template's spectral axis has one channel and the input's {input_channels}"
76            ));
77        }
78        debug_assert_eq!(input.coordinate_type(), CoordinateType::Spectral);
79        let positions = (0..output_channels)
80            .map(|channel| {
81                output
82                    .to_world(&[channel as f64])
83                    .and_then(|world| input.to_pixel(&world))
84                    .ok()
85                    .and_then(|pixel| pixel.first().copied())
86                    .filter(|pixel| pixel.is_finite())
87                    .map(|pixel| pixel as f32)
88            })
89            .collect();
90        Ok(Self { axis, positions })
91    }

93    /// `data` and its `mask` (true where good) regridded along the spectral
94    /// axis onto `shape`, which differs from theirs on that axis alone.
95    pub(super) fn apply<T>(
96        &self,
97        data: &ArrayD<T>,
98        mask: &ArrayD<bool>,
99        shape: &[usize],
100    ) -> (ArrayD<T>, ArrayD<bool>)
101    where
102        T: ImagePixel + Into<f64> + From<f32> + Copy,
103    {
104        let mut out_data = ArrayD::<T>::from_elem(IxDyn(shape).f(), T::from(0.0));
105        let mut out_mask = ArrayD::<bool>::from_elem(IxDyn(shape).f(), false);
106        let axis = Axis(self.axis);
107        Zip::from(out_data.lanes_mut(axis))
108            .and(out_mask.lanes_mut(axis))
109            .and(data.lanes(axis))
110            .and(mask.lanes(axis))
111            .for_each(|mut out, mut good, line, line_mask| {
112                for (channel, position) in self.positions.iter().enumerate() {
113                    let Some(position) = position else {
114                        continue;
115                    };
116                    let (value, valid) = linear(
117                        *position,
118                        |k| (Into::<f64>::into(line[k]) as f32, line_mask[k]),
119                        line.len(),
120                    );
121                    out[channel] = T::from(value);
122                    good[channel] = valid;
123                }
124            });
125        (out_data, out_mask)
126    }
127}

129/// casacore `InterpolateArray1D::interpolatePtr`'s `linear` case with
130/// flags, `goodIsTrue` and no extrapolation, over input pixels `0..count`
131/// (`count` at least two): the value at `x` and whether it is good.
132fn linear(x: f32, at: impl Fn(usize) -> (f32, bool), count: usize) -> (f32, bool) {
133    // `binarySearchBrackets` over 0, 1, …: the input at `x`, else the
134    // first beyond it.
135    let bracket = if x <= 0.0 {
136        0
137    } else {
138        (x.ceil() as usize).min(count)
139    };
140    let (right, discard) = match bracket {
141        0 => (1, x != 0.0),
142        n if n == count => (count - 1, true),
143        n => (n, false),
144    };
145    let left = right - 1;
146    let ((y1, good1), (y2, good2)) = (at(left), at(right));
147    let frac = (x - left as f32) / ((right - left) as f32);
148    let limit = f32::EPSILON;
149    if frac > limit && frac < 1.0 - limit {
150        (y1 + frac * (y2 - y1), !discard && good1 && good2)
151    } else if frac <= limit {
152        (y1, !discard && good1)
153    } else {
154        (y2, !discard && good2)
155    }
156}

163    /// A 4 × 4 image of one Stokes plane and `channels` channels from
164    /// `first_hz`, `step_hz` apart, each plane holding `value(channel)`.
165    fn cube(
166        first_hz: f64,
167        step_hz: f64,
168        channels: usize,
169        frame: casa_types::measures::frequency::FrequencyRef,
170        value: impl Fn(usize) -> f32,
171    ) -> crate::TempImage<f32> {
172        use casa_coordinates::{
173            CoordinateSystem, DirectionCoordinate, Projection, ProjectionType, SpectralCoordinate,
174            StokesCoordinate, StokesType,
175        };
176        use casa_types::measures::direction::DirectionRef;
177        let mut coordinates = CoordinateSystem::new();
178        coordinates.add_coordinate(DirectionCoordinate::new(
179            DirectionRef::J2000,
180            Projection::new(ProjectionType::SIN),
181            [1.0, 0.3],
182            [-1e-5, 1e-5],
183            [2.0, 2.0],
184        ));
185        coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
186        coordinates.add_coordinate(SpectralCoordinate::new(
187            frame, first_hz, step_hz, 0.0, 1.0e9,
188        ));
189        let shape = vec![4, 4, 1, channels];
190        let mut image = crate::TempImage::<f32>::new(
191            shape.clone(),
192            coordinates,
193            casa_lattices::TempStoragePolicy::Memory,
194        )
195        .unwrap();
196        image.set_units("Jy/pixel").unwrap();
197        let data = ArrayD::from_shape_fn(IxDyn(&shape).f(), |index| value(index[3]));
198        image.put_slice(&data, &[0, 0, 0, 0]).unwrap();
199        image
200    }

202    /// `SIImageStore::regridToModelImage`'s spectral regrid: an output
203    /// channel between two input channels takes their line, and one beyond
204    /// the input's ends takes the end channel's value, masked.
205    #[test]
206    fn a_cube_is_regridded_along_frequency_as_casacore_regrids_it() {
207        use casa_types::measures::frequency::FrequencyRef;
208        let input = cube(1.0e9, 1.0e8, 4, FrequencyRef::LSRK, |k| {
209            10.0 * (k as f32 + 1.0)
210        });
211        let template = RegridTemplate::of(&cube(0.95e9, 1.0e8, 6, FrequencyRef::LSRK, |_| 0.0));
212        let regridded = regrid_along_frequency(&input, &template, RegridMethod::Linear).unwrap();
213        assert_eq!(regridded.data.shape(), &[4, 4, 1, 6]);
214        let planes = (0..6)
215            .map(|k| (regridded.data[[1, 2, 0, k]], regridded.mask[[1, 2, 0, k]]))
216            .collect::<Vec<_>>();
217        assert_eq!(
218            planes,
219            [
220                (10.0, false),
221                (15.0, true),
222                (25.0, true),
223                (35.0, true),
224                (40.0, false),
225                (40.0, false)
226            ]
227        );
228        // Another frame, or one input channel, is refused.
229        let topo = RegridTemplate::of(&cube(0.95e9, 1.0e8, 6, FrequencyRef::TOPO, |_| 0.0));
230        let error = regrid_along_frequency(&input, &topo, RegridMethod::Linear).unwrap_err();
231        assert!(error.to_string().contains("spectral frames"), "{error}");
232        let one = cube(1.0e9, 1.0e8, 1, FrequencyRef::LSRK, |_| 1.0);
233        let error = regrid_along_frequency(&one, &template, RegridMethod::Linear).unwrap_err();
234        assert!(
235            error.to_string().contains("one channel") || error.to_string().contains("1 channel"),
236            "{error}"
237        );
238        // A cube onto one channel: `_checkAxes` removes the axis of unit
239        // length from those it regrids and refuses the shapes.
240        let plane = RegridTemplate::of(&cube(1.0e9, 1.0e8, 1, FrequencyRef::LSRK, |_| 0.0));
241        let error = regrid_along_frequency(&input, &plane, RegridMethod::Linear).unwrap_err();
242        assert!(
243            error.to_string().contains("same input and output shapes"),
244            "{error}"
245        );
246    }

259    /// Between channels the line, at a channel its value, beyond the ends
260    /// the end value masked: casacore's flagged linear interpolation.
261    #[test]
262    fn the_line_follows_casacores_flagged_linear_interpolation() {
263        let values = [1.0_f32, 3.0, 4.0, 8.0];
264        let goods = [true, true, false, true];
265        let at = |k: usize| (values[k], goods[k]);
266        assert_eq!(linear(0.5, at, 4), (2.0, true));
267        assert_eq!(linear(1.0, at, 4), (3.0, true));
268        assert_eq!(linear(0.0, at, 4), (1.0, true));
269        assert_eq!(linear(1.5, at, 4), (3.5, false));
270        assert_eq!(linear(3.0, at, 4), (8.0, true));
271        assert_eq!(linear(-0.5, at, 4), (1.0, false));
272        assert_eq!(linear(4.5, at, 4), (8.0, false));
273    }
