13/// How one pass divides an image domain's grid accumulation among owners
14/// (one per worker).
15#[derive(Clone, Debug, PartialEq)]
16pub enum Partition {
17    /// Each owner accumulates a disjoint contiguous range of each wave's
18    /// planes. Nothing is merged, so the result does not depend on the owner
19    /// count.
20    Planes {
21        /// Number of owners.
22        owners: usize,
23    },
24    /// Each owner accumulates the samples whose anchor row lies in its
25    /// region, on a tile of the region plus `halo` rows; tiles are added in
26    /// region order before the transform.
27    Regions {
28        /// The owners' regions, top to bottom.
29        regions: Vec<Region>,
30        /// Rows each tile extends past its region: the kernel set's largest
31        /// half support along `y`.
32        halo: usize,
33    },
34}
35
36/// One owner of a [`Partition::Regions`] pass.
37#[derive(Clone, Debug, PartialEq, Eq)]
38pub struct Region {
39    /// Anchor rows (grid `y`) owned.
40    pub rows: Range<usize>,
41    /// The accumulated window: the owned rows plus the halo, clipped to the grid.
42    pub tile: Tile,
43}
44
45impl Partition {
46    /// `owners` horizontal strips of `operator`'s grid of near-equal height,
47    /// each with the kernel set's largest half support
48    /// ([`casa_imaging_operator::ConvolutionFunctionSet::max_half_support`])
49    /// as a halo on both sides, so every routed sample's support lies in its
50    /// tile.
51    #[must_use]
52    pub fn regions(operator: &MeasurementOperator, owners: usize) -> Self {
53        let [_, ny] = operator.geometry().grid_shape();
54        let owners = owners.clamp(1, ny);
55        let bounds = (0..=owners)
56            .map(|owner| owner * ny / owners)
57            .collect::<Vec<_>>();
58        Self::from_bounds(operator, &bounds)
59    }
60
61    /// `owners` horizontal strips of `operator`'s grid holding near-equal
62    /// shares of `row_work`, the gridding work of each anchor row that an
63    /// earlier pass measured ([`super::PassSummary::row_work`]), each with
64    /// [`Self::regions`]' halo. A uv grid's samples gather towards its
65    /// centre, so strips of equal height leave most of the work to the
66    /// middle owners (#736). Each strip keeps at least one row. Work that
67    /// does not cover the grid, or holds none, gives [`Self::regions`].
68    #[must_use]
69    pub fn regions_balanced(
70        operator: &MeasurementOperator,
71        owners: usize,
72        row_work: &[u64],
73    ) -> Self {
74        let [_, ny] = operator.geometry().grid_shape();
75        let total = row_work.iter().map(|&work| u128::from(work)).sum::<u128>();
76        if row_work.len() != ny || total == 0 {
77            return Self::regions(operator, owners);
78        }
79        let owners = owners.clamp(1, ny);
80        // `before[r]`: the work of the rows above row `r`.
81        let before = std::iter::once(0)
82            .chain(row_work.iter().scan(0_u128, |sum, &work| {
83                *sum += u128::from(work);
84                Some(*sum)
85            }))
86            .collect::<Vec<_>>();
87        let mut bounds = Vec::with_capacity(owners + 1);
88        bounds.push(0);
89        for owner in 1..owners {
90            let share = total * owner as u128 / owners as u128;
91            let row = before.partition_point(|&work| work < share);
92            let previous = *bounds.last().expect("the first bound");
93            bounds.push(row.clamp(previous + 1, ny - (owners - owner)));
94        }
95        bounds.push(ny);
96        Self::from_bounds(operator, &bounds)
97    }
98
99    /// Strips between consecutive `bounds` (rows `bounds[k]..bounds[k+1]`
100    /// for owner `k`), with the kernel set's largest half support as halo.
101    fn from_bounds(operator: &MeasurementOperator, bounds: &[usize]) -> Self {
102        let [nx, ny] = operator.geometry().grid_shape();
103        let halo = usize::from(operator.cf().max_half_support()[1]);
104        Self::Regions {
105            regions: bounds
106                .windows(2)
107                .map(|pair| {
108                    let rows = pair[0]..pair[1];
109                    let first = rows.start.saturating_sub(halo);
110                    let last = (rows.end + halo).min(ny);
111                    Region {
112                        tile: Tile {
113                            origin: [0, first],
114                            shape: [nx, last - first],
115                        },
116                        rows,
117                    }
118                })
119                .collect(),
120            halo,
121        }
122    }
123
124    /// Number of owners.
125    #[must_use]
126    pub fn owners(&self) -> usize {
127        match self {
128            Self::Planes { owners } => *owners,
129            Self::Regions { regions, .. } => regions.len(),
130        }
131    }
132
133    /// Whether the grid's rows are split among several region owners: the
134    /// one case that measures its gridding work by anchor row
135    /// ([`super::PassSummary::row_work`], [`super::run_row_work_pass`]) and
136    /// takes larger chunks ([`WaveDemand::plan_chunk_placements`]). A lone
137    /// region owner and plane owners do neither (#736).
138    #[must_use]
139    pub fn splits_rows(&self) -> bool {
140        matches!(self, Self::Regions { regions, .. } if regions.len() > 1)
141    }
142}

157/// What one wave of a pass holds in memory, for [`Residency::plan`].
158#[derive(Clone, Copy)]
159pub struct WaveDemand<'a> {
160    /// The pass's image domains; they share one plane axis.
161    pub domains: &'a [PassDomain<'a>],
162    /// Modes accumulated.
163    pub modes: ModeSet,
164    /// Whether the pass subtracts a model.
165    pub with_model: bool,
166    /// The widest spacing between adjacent selected native channels, which
167    /// sizes a wave's model halo
168    /// ([`casa_imaging_operator::SpectralResampler::model_halo`]).
169    pub native_spacing_hz: f64,
170    /// Workers transforming planes at once.
171    pub workers: usize,
172    /// Where the owners grid.
173    pub backend: BackendChoice,
174    /// The layout of the source's rows ([`super::BoundedSource::maximum_block`]).
175    pub block: BlockShape,
176    /// Placements a row chunk takes at once ([`super::CHUNK_PLACEMENTS`],
177    /// or [`Self::plan_chunk_placements`]).
178    pub chunk_placements: usize,
179}
180
181impl WaveDemand<'_> {
182    /// Planes on each image domain's axis.
183    #[must_use]
184    pub fn planes(&self) -> u32 {
185        self.domains
186            .first()
187            .map_or(0, |domain| domain.operator.basis().planes())
188    }
189
190    /// Bytes a wave of `planes` planes holds: the largest of its stages
191    /// (#738), beside what it holds throughout.
192    ///
193    /// Throughout the wave, each domain's prepared model grids, with a
194    /// model, for the wave's planes and its model halo on each side; on
195    /// Metal, their device copy, which the device degrids, and each owner's
196    /// ring ([`casa_imaging_metal::ring_bytes`]); and a region-partitioned
197    /// domain's gridding work by anchor row (#736). The stages, summed over
198    /// image domains, since a domain finishes and appends while the others
199    /// hold their tiles or images:
200    /// - gridding: the owners' tiles of the wave's planes, beside the row
201    ///   chunks the wave places [`Self::block`]'s rows with, of
202    ///   [`Self::chunk_placements`] placements each;
203    /// - finishing, beside the chunks: a region pass's tiles and the merged
204    ///   accumulator they are added into, which one owner's whole-grid tile
205    ///   needs none of; then the accumulator beside its `f32` images; and
206    ///   for the transforms, one transform plane and one image per grid
207    ///   polarization for each transform that runs at once, and planning
208    ///   one grid transform ([`grid_planning_bytes`]), which the process
209    ///   does one at a time. Domains finish one after another; a plane
210    ///   pass's owners finish together on up to the workers, a region
211    ///   pass's merged accumulator alone (#745 review);
212    /// - appending: the images, the copy the normal state takes of them and,
213    ///   for a basis whose normal state is resident, that state at its
214    ///   stored size ([`casa_imaging_deconvolution::resident_bytes_per_cell`]);
215    ///   a channel-local state is paged, under its cube cache's charge.
216    ///
217    /// On Metal the accumulators are device memory and cost what a host
218    /// one does. What grows with the source's block is the source's
219    /// ([`super::source_block_bytes`]).
220    #[must_use]
221    pub fn bytes(&self, planes: u32) -> u64 {
222        let mut stages = DomainStages::default();
223        let mut transforming = 0_u64;
224        let mut planning = 0_u64;
225        for domain in self.domains {
226            stages.add(self.domain_stages(domain, planes));
227            let operator = domain.operator;
228            let [width, height] = operator.geometry().image().shape;
229            let image_cells = (width * height) as u64;
230            let grid_cells = operator.geometry().cells() as u64;
231            let gpols = operator.polarization().grid_pols() as u64;
232            let at_once = match &domain.partition {
233                Partition::Planes { owners } => (*owners).min(self.workers),
234                Partition::Regions { .. } => 1,
235            } as u64;
236            transforming = transforming.max(at_once * (grid_cells * 16 + gpols * image_cells * 16));
237            planning = planning.max(grid_planning_bytes(operator.geometry().cells()));
238        }
239        let chunks = super::wave::chunk_bytes(
240            self.domains,
241            self.backend,
242            self.block,
243            self.workers,
244            self.modes.weight,
245            self.with_model,
246            self.chunk_placements,
247        );
248        let transforms = planning + transforming;
249        stages.held
250            + (stages.gridding + chunks)
251                .max(stages.finishing + chunks + transforms)
252                .max(stages.appending)
253    }
254
255    /// Placements a row chunk of a wave of `planes` planes takes: when a
256    /// domain is split among several region owners, the largest of
257    /// [`super::REGION_CHUNK_PLACEMENTS`] whose [`Self::bytes`] fit `budget`,
258    /// since the owners share larger slices more evenly (#736); otherwise,
259    /// or when none fits, [`super::CHUNK_PLACEMENTS`]. A pass that fits with
260    /// the default chunks therefore never fails for want of larger ones.
261    #[must_use]
262    pub fn plan_chunk_placements(&self, planes: u32, budget: u64) -> usize {
263        if !self
264            .domains
265            .iter()
266            .any(|domain| domain.partition.splits_rows())
267        {
268            return super::CHUNK_PLACEMENTS;
269        }
270        super::REGION_CHUNK_PLACEMENTS
271            .into_iter()
272            .find(|&chunk_placements| {
273                WaveDemand {
274                    chunk_placements,
275                    ..*self
276                }
277                .bytes(planes)
278                    <= budget
279            })
280            .unwrap_or(super::CHUNK_PLACEMENTS)
281    }
282
283    /// What `domain` holds at each stage of a wave of `planes` planes
284    /// ([`Self::bytes`]).
285    fn domain_stages(&self, domain: &PassDomain<'_>, planes: u32) -> DomainStages {
286        let operator = domain.operator;
287        let precision = operator.precision();
288        let one = PlaneRange::single(0);
289        let planes_u64 = u64::from(planes);
290        let layout = |tile: Option<Tile>, modes: ModeSet| {
291            operator
292                .accumulator_layout(one, tile, modes)
293                .bytes(precision) as u64
294        };
295        let accumulator = layout(None, self.modes) * planes_u64;
296        let (tiles, merging) = match &domain.partition {
297            Partition::Planes { .. } => (accumulator, accumulator),
298            Partition::Regions { regions, .. } => {
299                let tiles = regions
300                    .iter()
301                    .map(|region| layout(Some(region.tile), self.modes))
302                    .sum::<u64>()
303                    * planes_u64;
304                // One owner's tile is the whole grid, which the wave
305                // finishes as it is: both key on the owner count.
306                let merged = if regions.len() > 1 { accumulator } else { 0 };
307                (tiles, tiles + merged)
308            }
309        };
310        let [width, height] = operator.geometry().image().shape;
311        let image_cells = (width * height) as u64;
312        let basis = operator.basis();
313        let pols = operator.polarization().requested().len() as u64;
314        // The weight image has the PSF's terms (`AccumulatorLayout`).
315        let terms = (basis.data_terms()
316            + if self.modes.psf { basis.psf_terms() } else { 0 }
317            + if self.modes.weight {
318                basis.psf_terms()
319            } else {
320                0
321            }) as u64;
322        let images = terms * pols * image_cells * size_of::<f32>() as u64 * planes_u64;
323        let normal = if matches!(basis, Basis::ChannelLocal { .. }) {
324            0
325        } else {
326            pols * image_cells
327                * resident_bytes_per_cell(basis.data_terms(), basis.psf_terms(), self.modes.psf)
328                * planes_u64
329        };
330        let mut held = 0;
331        if self.with_model {
332            let model_planes = planes
333                .saturating_add(2 * domain.resampler.model_halo(self.native_spacing_hz))
334                .min(self.planes());
335            // Metal degrids the model on the device, fused into the
336            // residual or in the chunks' prediction batches.
337            let copies = if self.backend == BackendChoice::Metal {
338                2
339            } else {
340                1
341            };
342            held += copies * layout(None, ModeSet::DATA) * u64::from(model_planes);
343        }
344        if self.backend == BackendChoice::Metal {
345            let npol = operator.polarization().correlations().len();
346            held += domain.partition.owners() as u64 * casa_imaging_metal::ring_bytes(npol);
347        }
348        // A domain split among region owners measures its gridding work by
349        // anchor row, which the pass hands to the next one's partition
350        // (#736).
351        if domain.partition.splits_rows() {
352            held += (operator.geometry().grid_shape()[1] * size_of::<u64>()) as u64;
353        }
354        DomainStages {
355            held,
356            gridding: tiles,
357            finishing: merging.max(accumulator + images),
358            appending: 2 * images + normal,
359        }
360    }
361}

363/// What a wave holds at each of its stages ([`WaveDemand::bytes`]), summed
364/// over image domains: a domain finishes and appends while the others
365/// hold their tiles or images, so the sums bound every moment.
366#[derive(Clone, Copy, Default)]
367struct DomainStages {
368    /// Held through every stage.
369    held: u64,
370    /// The owners' tiles.
371    gridding: u64,
372    /// The merge, then the accumulator beside its images.
373    finishing: u64,
374    /// The images, their copy and the normal state.
375    appending: u64,
376}
377
378impl DomainStages {
379    fn add(&mut self, other: Self) {
380        self.held += other.held;
381        self.gridding += other.gridding;
382        self.finishing += other.finishing;
383        self.appending += other.appending;
384    }
385}

387impl Residency {
388    /// The fewest waves whose [`WaveDemand::bytes`] fit `budget`: every
389    /// plane when they fit, otherwise the largest wave that does and at
390    /// least one plane. Whether a wave fits at all is the admission of its
391    /// bytes ([`crate::admit`]).
392    #[must_use]
393    pub fn plan(demand: &WaveDemand<'_>, budget: u64) -> Self {
394        let planes = demand.planes();
395        if planes <= 1 || demand.bytes(planes) <= budget {
396            return Self::All;
397        }
398        if demand.bytes(1) > budget {
399            return Self::Waves { planes_per_wave: 1 };
400        }
401        // `bytes` grows with the wave: the largest fitting wave lies in
402        // `[fits, too_many)`.
403        let (mut fits, mut too_many) = (1, planes);
404        while too_many - fits > 1 {
405            let middle = fits + (too_many - fits) / 2;
406            if demand.bytes(middle) <= budget {
407                fits = middle;
408            } else {
409                too_many = middle;
410            }
411        }
412        Self::Waves {
413            planes_per_wave: fits,
414        }
415    }
416
417    /// The plane ranges of the waves over `planes` planes.
418    #[must_use]
419    pub fn waves(self, planes: u32) -> Vec<PlaneRange> {
420        let step = match self {
421            Self::All => planes.max(1),
422            Self::Waves { planes_per_wave } => {
423                assert!(planes_per_wave > 0, "a wave holds at least one plane");
424                planes_per_wave
425            }
426        };
427        (0..planes)
428            .step_by(step as usize)
429            .map(|start| PlaneRange::new(start, (start + step).min(planes)))
430            .collect()
431    }
432}

434/// The owner of each placement within one wave.
435pub(super) enum Router {
436    /// `starts[k]` is owner `k`'s first plane; owner ranges are contiguous.
437    Planes { starts: Vec<u32> },
438    /// `owner_of_row[y]` owns anchor row `y`; tiles extend `halo` rows.
439    Regions { owner_of_row: Vec<u16>, halo: usize },
440}
441
442impl Router {
443    pub(super) fn new(partition: &Partition, wave: PlaneRange) -> Self {
444        match partition {
445            Partition::Planes { owners } => {
446                let planes = wave.len();
447                let owners = (*owners).clamp(1, planes.max(1));
448                Self::Planes {
449                    starts: (0..owners)
450                        .map(|owner| wave.start + (owner * planes / owners) as u32)
451                        .collect(),
452                }
453            }
454            Partition::Regions { regions, halo } => {
455                let rows = regions.last().map_or(0, |region| region.rows.end);
456                let mut owner_of_row = vec![0; rows];
457                for (owner, region) in regions.iter().enumerate() {
458                    owner_of_row[region.rows.clone()].fill(owner as u16);
459                }
460                Self::Regions {
461                    owner_of_row,
462                    halo: *halo,
463                }
464            }
465        }
466    }
467
468    /// Number of owners in this wave.
469    pub(super) fn owners(&self) -> usize {
470        match self {
471            Self::Planes { starts } => starts.len(),
472            Self::Regions { owner_of_row, .. } => {
473                owner_of_row.last().map_or(1, |owner| *owner as usize + 1)
474            }
475        }
476    }
477
478    /// The one owner that grids the weight image when the grid is split
479    /// into regions: the owner of the centre row, where `Mode::Weight`
480    /// places every sample's `FT[PB²]` taps. `None` when every owner holds
481    /// whole planes and grids its own samples' weights.
482    pub(super) fn weight_owner(&self) -> Option<usize> {
483        match self {
484            Self::Planes { .. } => None,
485            Self::Regions { owner_of_row, .. } => {
486                Some(usize::from(owner_of_row[owner_of_row.len() / 2]))
487            }
488        }
489    }
490
491    /// Plane range and tile owner `owner` accumulates in `wave`.
492    pub(super) fn target(
493        &self,
494        partition: &Partition,
495        wave: PlaneRange,
496        owner: usize,
497    ) -> (PlaneRange, Option<Tile>) {
498        match (self, partition) {
499            (Self::Planes { starts }, _) => {
500                let end = starts.get(owner + 1).copied().unwrap_or(wave.end);
501                (PlaneRange::new(starts[owner], end), None)
502            }
503            (Self::Regions { .. }, Partition::Regions { regions, .. }) => {
504                (wave, Some(regions[owner].tile))
505            }
506            (Self::Regions { .. }, Partition::Planes { .. }) => {
507                unreachable!("router built from this partition")
508            }
509        }
510    }
511
512    /// Owner of one placement and, with region owners, its anchor row and
513    /// gridding work, the kernel's taps `(2·h_x + 1)(2·h_y + 1)`, which
514    /// [`Partition::regions_balanced`] balances; the placement lies on the
515    /// wave and fits the grid.
516    pub(super) fn owner_and_work(
517        &self,
518        operator: &MeasurementOperator,
519        placement: &Placement,
520    ) -> (usize, Option<(usize, u64)>) {
521        match self {
522            Self::Planes { starts } => (
523                starts.partition_point(|start| *start <= placement.plane) - 1,
524                None,
525            ),
526            Self::Regions { owner_of_row, halo } => {
527                let mut hold = CellHold::new();
528                let taps = operator.cf().taps(placement.cf, &mut hold);
529                let [hx, hy] = taps.half_support();
530                debug_assert!(
531                    usize::from(hy) <= *halo,
532                    "a kernel cell's support exceeds the regions' halo"
533                );
534                let anchor =
535                    operator
536                        .geometry()
537                        .locate(placement.u, placement.v, taps.oversampling());
538                let row = anchor.y as usize;
539                let work = (2 * u64::from(hx) + 1) * (2 * u64::from(hy) + 1);
540                (usize::from(owner_of_row[row]), Some((row, work)))
541            }
542        }
543    }
544}

587    /// Work gathered at the grid's centre, as uv samples are: equal-height
588    /// strips give the middle owner nearly all of it, balanced strips give
589    /// each owner its share to within one row's work. The strips still
590    /// cover the grid once, each with at least one row and the kernel halo.
591    #[test]
592    fn balanced_regions_share_centre_heavy_work() {
593        let operator = spheroidal_operator();
594        let ny = operator.geometry().grid_shape()[1];
595        let centre = ny as f64 / 2.0;
596        let row_work = (0..ny)
597            .map(|row| {
598                let offset = (row as f64 - centre) / (ny as f64 / 16.0);
599                (1.0e6 * (-offset * offset).exp()) as u64
600            })
601            .collect::<Vec<_>>();
602        let total = row_work.iter().sum::<u64>();
603        let largest = *row_work.iter().max().expect("rows");
604        let share = |rows: &Range<usize>| row_work[rows.clone()].iter().sum::<u64>();
605        let equal = rows_of(&Partition::regions(&operator, 3));
606        assert!(
607            share(&equal[1]) > total * 9 / 10,
608            "the middle strip holds the work"
609        );
610        let balanced = Partition::regions_balanced(&operator, 3, &row_work);
611        let rows = rows_of(&balanced);
612        assert_eq!(rows.first().expect("a strip").start, 0);
613        assert_eq!(rows.last().expect("a strip").end, ny);
614        for pair in rows.windows(2) {
615            assert_eq!(pair[0].end, pair[1].start);
616        }
617        for strip in &rows {
618            assert!(!strip.is_empty());
619            assert!(
620                share(strip).abs_diff(total / 3) <= largest,
621                "strip {strip:?} holds {} of {total}",
622                share(strip)
623            );
624        }
625        let Partition::Regions { regions, halo } = balanced else {
626            panic!("regions");
627        };
628        assert_eq!(halo, 3);
629        for region in &regions {
630            assert_eq!(
631                region.tile.origin[1],
632                region.rows.start.saturating_sub(halo)
633            );
634            assert_eq!(
635                region.tile.origin[1] + region.tile.shape[1],
636                (region.rows.end + halo).min(ny)
637            );
638        }
639    }

641    /// Without usable work, or with work on one row, the strips stay valid:
642    /// equal strips for no work or the wrong length, and one row at least
643    /// per strip when all the work is on one row.
644    #[test]
645    fn balanced_regions_fall_back_and_keep_every_strip() {
646        let operator = spheroidal_operator();
647        let ny = operator.geometry().grid_shape()[1];
648        let equal = rows_of(&Partition::regions(&operator, 4));
649        assert_eq!(
650            rows_of(&Partition::regions_balanced(&operator, 4, &[])),
651            equal
652        );
653        assert_eq!(
654            rows_of(&Partition::regions_balanced(&operator, 4, &vec![0; ny])),

656        );
657        let mut one_row = vec![0; ny];

659        let rows = rows_of(&Partition::regions_balanced(&operator, 4, &one_row));
660        assert_eq!(rows.len(), 4);

665        let rows = rows_of(&Partition::regions_balanced(&operator, 4, &last_row));
666        assert!(rows.iter().all(|strip| !strip.is_empty()));
