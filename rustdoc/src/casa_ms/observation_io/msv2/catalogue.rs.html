25/// A column's `MEASINFO` record, if it has one.
26fn measinfo<'t>(table: &'t Table, column: &str) -> Option<&'t casa_types::RecordValue> {
27    match table.column_keywords(column)?.get("MEASINFO")? {
28        Value::Record(record) => Some(record),
29        _ => None,
30    }
31}
32
33/// A column's fixed `MEASINFO.Ref`, if it names one. A variable reference
34/// (`VarRefCol`) is refused: the trial's views carry one reference per
35/// batch.
36pub(crate) fn measure_reference(
37    table: &Table,
38    column: &str,
39) -> Result<Option<String>, ObservationError> {
40    let Some(record) = measinfo(table, column) else {
41        return Ok(None);
42    };
43    if record.get("VarRefCol").is_some() {
44        return Err(ObservationError::Unsupported {
45            what: "an MSv2 measure column with a variable reference (VarRefCol)",
46        });
47    }
48    Ok(match record.get("Ref") {
49        Some(Value::Scalar(ScalarValue::String(reference))) => Some(reference.clone()),
50        _ => None,
51    })
52}

54/// Require a column's `QuantumUnits`, when present, to be `expected` in
55/// every position. An absent keyword is the MeasurementSet definition's
56/// unit, which is `expected`.
57pub(crate) fn require_units(
58    table: &Table,
59    column: &str,
60    expected: &str,
61) -> Result<(), ObservationError> {
62    let Some(units) = quantum_units(table, column) else {
63        return Ok(());
64    };
65    if units.iter().all(|unit| unit == expected) {
66        Ok(())
67    } else {
68        Err(invalid(format!(
69            "{column} is in {units:?}; the trial adapter reads only {expected} and does not convert"
70        )))
71    }
72}
73
74fn quantum_units(table: &Table, column: &str) -> Option<Vec<String>> {
75    match table.column_keywords(column)?.get("QuantumUnits")? {
76        Value::Array(ArrayValue::String(units)) => Some(units.iter().cloned().collect()),
77        Value::Scalar(ScalarValue::String(unit)) => Some(vec![unit.clone()]),
78        _ => None,
79    }
80}

82/// What visibility column `column`'s `QuantumUnits` declares: janskys
83/// only when it says `Jy`.
84pub(super) fn data_units(table: &Table, column: &str) -> VisibilityUnits {
85    match quantum_units(table, column).as_deref() {
86        None | Some([]) => VisibilityUnits::Undeclared,
87        Some([unit, ..]) if unit == "Jy" => VisibilityUnits::Jansky,
88        Some([unit, ..]) => VisibilityUnits::Declared(unit.clone()),
89    }
90}

92/// The time scale of MAIN `TIME` and `TIME_CENTROID`, in seconds. The
93/// MeasurementSet definition's default reference is UTC; another fixed
94/// reference is kept, a variable one refused, and the two columns must
95/// agree.
96pub(super) fn time_scale(main: &Table) -> Result<EpochRef, ObservationError> {
97    let mut scales = Vec::new();
98    for column in ["TIME", "TIME_CENTROID"] {
99        require_units(main, column, "s")?;
100        let scale = match measure_reference(main, column)? {
101            Some(name) => name
102                .parse::<EpochRef>()
103                .map_err(|_| invalid(format!("unknown {column} reference {name}")))?,
104            None => EpochRef::UTC,
105        };
106        scales.push(scale);
107    }
108    for column in ["INTERVAL", "EXPOSURE"] {
109        require_units(main, column, "s")?;
110    }
111    if scales[0] == scales[1] {
112        Ok(scales[0])
113    } else {
114        Err(ObservationError::Unsupported {
115            what: "MSv2 TIME and TIME_CENTROID in different time scales",
116        })
117    }
118}

129/// Each `FIELD` row's `PHASE_DIR` frame: fixed for the column, or
130/// variable per row (`VarRefCol`, integer or string codes, as ALMA's
131/// `PhaseDir_Ref`).
132pub(crate) fn phase_frames(table: &Table) -> Result<Vec<DirectionRef>, ObservationError> {
133    require_units(table, "PHASE_DIR", "rad")?;
134    (0..table.row_count())
135        .map(|row| {
136            resolve_direction_reference_selected(table, "FIELD", "PHASE_DIR", row).map_err(storage)
137        })
138        .collect()
139}

141/// The fields, their ephemerides, and the UVW convention.
142pub(super) fn fields_and_uvw(ms: &MeasurementSet) -> Result<Fields, ObservationError> {
143    let field = ms.field().map_err(storage)?;
144    let field_table = ms
145        .subtable(SubtableId::Field)
146        .ok_or_else(|| invalid("no FIELD subtable"))?;
147    // The fields share one frame: a data group has one UVW frame, and MSv4
148    // one phase-centre frame. Without fields it is J2000, the definition's
149    // default.
150    let frames = phase_frames(field_table)?;
151    let phase_frame = frames.first().copied().unwrap_or(DirectionRef::J2000);
152    if frames.iter().any(|frame| *frame != phase_frame) {
153        return Err(ObservationError::Unsupported {
154            what: "MSv2 fields whose phase centres are in different frames",
155        });
156    }
157    require_units(field_table, "TIME", "s")?;
158    let epoch_scale = match measure_reference(field_table, "TIME")? {
159        Some(name) => name
160            .parse::<EpochRef>()
161            .map_err(|_| invalid(format!("unknown FIELD TIME reference {name}")))?,
162        None => EpochRef::UTC,
163    };
164    let phase_dir = DirectionColumn::phase_dir(field_table);
165    let fields = (0..field.row_count())
166        .map(|row| {
167            let (longitude, latitude) = phase_dir.get_direction(row).map_err(storage)?.as_angles();
168            let source = field.source_id(row).map_err(storage)?;
169            let time_s = field.time(row).map_err(storage)?;
170            Ok(FieldEntry {
171                name: field.name(row).map_err(storage)?,
172                phase_centre: MDirection::from_angles(longitude, latitude, frames[row]),
173                reference_epoch: Some(MEpoch::from_mjd(time_s / 86_400.0, epoch_scale)),
174                source_name: source_name(ms, source)?,
175            })
176        })
177        .collect::<Result<_, ObservationError>>()?;
178    let ephemerides = (0..field.row_count())
179        .map(|row| field.ephemeris_id(row).map_err(storage))
180        .collect::<Result<_, ObservationError>>()?;
181    let main = ms.main_table();
182    require_units(main, "UVW", "m")?;
183    let stored_label = measure_reference(main, "UVW")?
184        .map(|name| {
185            name.parse::<DirectionRef>()
186                .map_err(|_| invalid(format!("unknown UVW reference {name}")))
187        })
188        .transpose()?;
189    // CASA's convention: UVW are in the phase centre's frame. An ITRF label
190    // (casacore's default, which simulators and fillers write) is the known
191    // mislabelling; any other label must name the phase centres' frame.
192    match stored_label {
193        None | Some(DirectionRef::ITRF) => {}
194        Some(label) if label == phase_frame => {}
195        Some(_) => {
196            return Err(ObservationError::Unsupported {
197                what: "MSv2 UVW labelled with a celestial frame other than the phase centres'",
198            });
199        }
200    }
201    Ok(Fields {
202        entries: fields,
203        ephemerides,
204        uvw_frame: UvwFrame {
205            effective: phase_frame,
206            stored_label,
207        },
208    })
209}

211/// Every `OBSERVATION` row. A MeasurementSet without one gets a single
212/// empty entry, which every row's `OBSERVATION_ID` 0 then names.
213pub(super) fn read_observations(
214    ms: &MeasurementSet,
215) -> Result<Vec<ObservationEntry>, ObservationError> {
216    let observation = ms.observation().map_err(storage)?;
217    if observation.row_count() == 0 {
218        return Ok(vec![ObservationEntry::default()]);
219    }
220    (0..observation.row_count())
221        .map(|row| {
222            let schedule: Vec<String> = match observation
223                .optional_array(row, "SCHEDULE")
224                .map_err(storage)?
225            {
226                Some(ArrayValue::String(lines)) => lines.iter().cloned().collect(),
227                _ => Vec::new(),
228            };
229            let uid = |line: usize, label: &str| {
230                schedule
231                    .get(line)
232                    .filter(|_| schedule.len() == 2)
233                    .and_then(|line| scheduled_uid(line, label))
234            };
235            Ok(ObservationEntry {
236                telescope: observation.string(row, "TELESCOPE_NAME").map_err(storage)?,
237                observers: vec![observation.string(row, "OBSERVER").map_err(storage)?],
238                project: observation.string(row, "PROJECT").map_err(storage)?,
239                execution_block: uid(1, "ExecBlock"),
240                session_reference: None,
241                scheduling_block: uid(0, "SchedulingBlock"),
242            })
243        })
244        .collect()
245}

247/// The UID after `label` in a `SCHEDULE` line, as XRADIO's converter reads
248/// one `importasdm` writes (`[SchedulingBlock uid://A001/X3571/X122,
249/// ExecBlock uid://A002/X1003af4/X75a3]`, one entry per line): the word
250/// characters, `/` and `:` that follow `label` and a space.
251fn scheduled_uid(line: &str, label: &str) -> Option<String> {
252    let start = line.find(&format!("{label} "))? + label.len() + 1;
253    let uid: String = line[start..]
254        .chars()
255        .take_while(|c| c.is_alphanumeric() || matches!(c, '_' | '/' | ':'))
256        .collect();
257    (!uid.is_empty()).then_some(uid)
258}

270/// The intents an `OBS_MODE` string lists, sorted and deduplicated.
271pub(super) fn intents(mode: &str) -> Vec<String> {
272    let set: BTreeSet<String> = mode
273        .split(',')
274        .map(str::trim)
275        .filter(|intent| !intent.is_empty())
276        .map(str::to_string)
277        .collect();
278    set.into_iter().collect()
279}

281/// Every `ANTENNA` row.
282pub(super) fn read_antennas(ms: &MeasurementSet) -> Result<Vec<AntennaEntry>, ObservationError> {
283    let antenna = ms.antenna().map_err(storage)?;
284    let antenna_table = ms
285        .subtable(SubtableId::Antenna)
286        .ok_or_else(|| invalid("no ANTENNA subtable"))?;
287    require_units(antenna_table, "POSITION", "m")?;
288    require_units(antenna_table, "DISH_DIAMETER", "m")?;
289    match measure_reference(antenna_table, "POSITION")?.as_deref() {
290        None | Some("ITRF") => {}
291        Some(_) => {
292            return Err(ObservationError::Unsupported {
293                what: "MSv2 antenna positions in other than ITRF",
294            });
295        }
296    }
297    (0..antenna.row_count())
298        .map(|row| {
299            let [x, y, z] = antenna.position(row).map_err(storage)?;
300            Ok(AntennaEntry {
301                name: antenna.name(row).map_err(storage)?,
302                station: antenna.station(row).map_err(storage)?,
303                position: MPosition::new_itrf(x, y, z),
304                mount: antenna.mount(row).map_err(storage)?,
305                dish_diameter: metres(antenna.dish_diameter(row).map_err(storage)?),
306            })
307        })
308        .collect()
309}

311/// Every `FEED` row's receptors, by (antenna, feed, spectral window), the
312/// window -1 standing for every window, in table order.
313pub(crate) struct Feeds {
314    rows: BTreeMap<(i32, i32, i32), Vec<Receptors>>,
315    /// Some row names a window: casacore's `MSIter` then reads only each
316    /// window's own rows, and XRADIO's converter likewise.
317    window_dependent: bool,
318}
319
320/// The `FEED` rows.
321pub(crate) fn read_feeds(ms: &MeasurementSet) -> Result<Feeds, ObservationError> {
322    let feed = ms.feed().map_err(storage)?;
323    if let Some(feed_table) = ms.subtable(SubtableId::Feed) {
324        require_units(feed_table, "RECEPTOR_ANGLE", "rad")?;
325    }
326    let mut rows = BTreeMap::<(i32, i32, i32), Vec<Receptors>>::new();
327    for row in 0..feed.row_count() {
328        let key = (
329            feed.i32(row, "ANTENNA_ID").map_err(storage)?,
330            feed.i32(row, "FEED_ID").map_err(storage)?,
331            feed.i32(row, "SPECTRAL_WINDOW_ID").map_err(storage)?,
332        );
333        let types = match feed.array(row, "POLARIZATION_TYPE").map_err(storage)? {
334            ArrayValue::String(types) => types.iter().cloned().collect(),
335            _ => return Err(invalid("FEED POLARIZATION_TYPE is not a string array")),
336        };
337        let angles = match feed.array(row, "RECEPTOR_ANGLE").map_err(storage)? {
338            ArrayValue::Float64(angles) => angles.iter().map(|angle| radians(*angle)).collect(),
339            _ => return Err(invalid("FEED RECEPTOR_ANGLE is not a Float64 array")),
340        };
341        rows.entry(key)
342            .or_default()
343            .push(Receptors { types, angles });
344    }
345    let window_dependent = rows.keys().any(|(_, _, window)| *window != -1);
346    Ok(Feeds {
347        rows,
348        window_dependent,
349    })
350}
351
352impl Feeds {
353    /// Antenna `antenna`'s receptors through feed `feed` in spectral window
354    /// `window`, as casacore's `MSIter::setFeedInfo` and XRADIO's converter
355    /// select them: the window's own rows when any row names a window, else
356    /// the rows for every window (-1). A partition's receptors are one set
357    /// (MSv4 holds no time dependence), so where rows of one antenna, feed
358    /// and window differ, `FEED` stating receptors per time, the last row
359    /// in table order is taken, as casacore's `MSIter` takes it (with a
360    /// warning); XRADIO's converter refuses such rows, and a conversion
361    /// reports them as a loss ([`Self::varying`]). `None` without a row.
362    pub(super) fn receptors(&self, antenna: i32, feed: i32, window: i32) -> Option<&Receptors> {
363        let window = if self.window_dependent { window } else { -1 };
364        self.rows
365            .get(&(antenna, feed, window))
366            .and_then(|rows| rows.last())
367    }
368
369    /// The first (antenna, feed, window) whose rows differ, among the rows
370    /// [`Self::receptors`] reads: `FEED` states receptors that vary in
371    /// time, of which it keeps the last. Rows for every window (-1) beside
372    /// window rows are never read, and do not count.
373    pub(crate) fn varying(&self) -> Option<(i32, i32, i32)> {
374        self.rows
375            .iter()
376            .filter(|((_, _, window), _)| !self.window_dependent || *window != -1)
377            .find(|(_, rows)| rows.iter().any(|row| row != &rows[0]))
378            .map(|(key, _)| *key)
379    }
380}

390/// The `SOURCE` name of source `id`, if the MeasurementSet has one.
391fn source_name(ms: &MeasurementSet, id: i32) -> Result<Option<String>, ObservationError> {
392    if id < 0 {
393        return Ok(None);
394    }
395    let Ok(source) = ms.source() else {
396        return Ok(None);
397    };
398    for row in 0..source.row_count() {
399        if source.i32(row, "SOURCE_ID").map_err(storage)? == id {
400            return source.string(row, "NAME").map(Some).map_err(storage);
401        }
402    }
403    Ok(None)
404}

406/// The scans: MAIN's distinct `SCAN_NUMBER`s in ascending order, named by
407/// their decimal number as XRADIO names them, with the `STATE::OBS_MODE`
408/// intents of their rows' states (`states_by_number`, each scan's
409/// `STATE_ID`s; `modes`, each `STATE` row's `OBS_MODE`); and the scan of
410/// each `SCAN_NUMBER`.
411pub(super) fn scans(
412    states_by_number: BTreeMap<i32, BTreeSet<i32>>,
413    modes: &[String],
414) -> (Vec<ScanEntry>, BTreeMap<i32, ScanId>) {
415    let index: BTreeMap<i32, ScanId> = states_by_number
416        .keys()
417        .enumerate()
418        .map(|(index, number)| (*number, ScanId(index as u32)))
419        .collect();
420    let scans = states_by_number
421        .into_iter()
422        .map(|(number, states)| {
423            let intents: BTreeSet<String> = states
424                .into_iter()
425                .filter_map(|state| {
426                    usize::try_from(state)
427                        .ok()
428                        .and_then(|state| modes.get(state))
429                })
430                .flat_map(|mode| intents(mode))
431                .collect();
432            ScanEntry {
433                name: number.to_string(),
434                number: Some(number),
435                intents: intents.into_iter().collect(),
436            }
437        })
438        .collect();
439    (scans, index)
440}

442/// The spectral window and correlation products of one data description.
443pub(super) struct SpectralSetup {
444    /// The `SPECTRAL_WINDOW` row.
445    pub(super) spectral_window_id: u32,
446    pub(super) spectral_window: String,
447    pub(super) frequencies_hz: Vec<f64>,
448    pub(super) frequency_frame: FrequencyRef,
449    pub(super) channel_widths_hz: Vec<f64>,
450    pub(super) correlations: Vec<Correlation>,
451}
452
453/// Every `DATA_DESCRIPTION` row's spectral setup.
454pub(super) fn spectral_setups(ms: &MeasurementSet) -> Result<Vec<SpectralSetup>, ObservationError> {
455    let descriptions = ms.data_description().map_err(storage)?;
456    let windows = ms.spectral_window().map_err(storage)?;
457    if let Some(window_table) = ms.subtable(SubtableId::SpectralWindow) {
458        require_units(window_table, "CHAN_FREQ", "Hz")?;
459        require_units(window_table, "CHAN_WIDTH", "Hz")?;
460    }
461    let polarizations = ms.polarization().map_err(storage)?;
462    (0..descriptions.row_count())
463        .map(|row| {
464            let window = usize::try_from(descriptions.spectral_window_id(row).map_err(storage)?)
465                .map_err(|_| invalid("negative SPECTRAL_WINDOW_ID"))?;
466            let polarization = usize::try_from(descriptions.polarization_id(row).map_err(storage)?)
467                .map_err(|_| invalid("negative POLARIZATION_ID"))?;
468            let frame_code = windows.meas_freq_ref(window).map_err(storage)?;
469            let frequency_frame = FrequencyRef::from_casacore_code(frame_code)
470                .ok_or_else(|| invalid(format!("unknown MEAS_FREQ_REF {frame_code}")))?;
471            let correlations = polarizations
472                .corr_type(polarization)
473                .map_err(storage)?
474                .into_iter()
475                .map(|code| {
476                    Correlation::from_casacore_code(code)
477                        .ok_or_else(|| invalid(format!("unknown CORR_TYPE {code}")))
478                })
479                .collect::<Result<_, _>>()?;
480            Ok(SpectralSetup {
481                spectral_window_id: window as u32,
482                spectral_window: windows.name(window).map_err(storage)?,
483                frequencies_hz: windows.chan_freq(window).map_err(storage)?,
484                frequency_frame,
485                channel_widths_hz: windows.chan_width(window).map_err(storage)?,
486                correlations,
487            })
488        })
489        .collect()
490}
