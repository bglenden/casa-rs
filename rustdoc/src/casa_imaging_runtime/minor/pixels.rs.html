2//! The pixels a plane's measurements read, as tclean's image store reads
3//! them.
4//!
5//! tclean's residual carries the `.pb`'s support (`pb > |pblimit|`) as its
6//! pixel mask wherever the run has a `.pb` and `pblimit` is not negative
7//! (`SIImageStore::divideResidualByWeight` copies it). The residual's
8//! peaks (`getPeakResidual`, `getPeakResidualWithinMask`) read only the
9//! pixels it keeps. Every beam CASA publishes is zero where it is not
10//! above `|pblimit|`, so `pb > pbmask` keeps no pixel outside that support,
11//! and at tclean's default `pbmask` of 0 it is the support.
12//!
13//! * The `nsigma` noise reads the pixels where the `.pb` exceeds `pbmask`
14//!   (`SIImageStore::calcRobustRMS`), or every pixel when no plane of the
15//!   image keeps one (CAS-13318, decided over the whole cube: a plane the
16//!   beam leaves empty in an image it does not measures nothing).
17//! * An automatic mask's statistics, pruning and support read the
18//!   residual's pixel mask (`SDMaskHandler::autoMask`), which a `pbmask`
19//!   above zero replaces with `pb > pbmask`, whatever `pblimit`'s sign
20//!   (`autoMaskWithinPB`). Under an automatic mask the noise is the mask's
21//!   own statistics, except at the run's first boundary, where the mask
22//!   takes `calcRobustRMS`'s (`SynthesisDeconvolver::initMinorCycle`).
23//! * A mask cut to the primary beam (`usemask='pb'`, or a named mask under
24//!   a `pbmask` above zero) keeps `pb > pbmask` in every plane
25//!   (`SDMaskHandler::makePBMask`), read here for the mask's plans
26//!   ([`plane_beam`]).
27//!
28//! The beam is the weight image's under a response (`makePBFromWeight`)
29//! and the telescope's analytic beam otherwise (`makePBImage`), the one
30//! the products publish.

53/// Whether a minor cycle under `setup` reads its planes' beams: for the
54/// `nsigma` noise, for the residual's pixel mask, or for an automatic
55/// mask's pixel mask at a `pbmask` above zero.
56pub(super) const fn reads_beam(setup: &MinorCycleSetup<'_>) -> bool {
57    setup.nsigma > 0.0 || setup.residual_pixel_mask || (setup.automask && setup.pbmask > 0.0)
58}

60/// tclean's `pbmask` as its Float `.pb` is compared with it.
61pub(super) fn pbmask(setup: &MinorCycleSetup<'_>) -> f32 {
62    setup.pbmask as f32
63}

65/// One plane's `.pb` inside its valid pixels, x-major: where it is
66/// positive, above the cutoff (`pb > |pblimit|`, the residual's pixel
67/// mask), and where it exceeds a `pbmask` above zero (`pb > pbmask`).
68#[derive(Clone)]
69pub(super) struct BeamPixels {
70    support: Vec<bool>,
71    /// `None` at a `pbmask` of 0, where it is `support`.
72    within: Option<Vec<bool>>,
73}
74
75impl BeamPixels {
76    /// The pixels of `valid` where the published `.pb`, `pb` (zero at or
77    /// below the cutoff), is positive and where it exceeds `pbmask`.
78    fn new(
79        pb: impl Iterator<Item = Result<f32, ProductsError>>,
80        valid: &[bool],
81        pbmask: f32,
82    ) -> Result<Self, ProductsError> {
83        let mut support = Vec::with_capacity(valid.len());
84        let mut within = (pbmask > 0.0).then(|| Vec::with_capacity(valid.len()));
85        for (pb, valid) in pb.zip(valid) {
86            let pb = pb?;
87            support.push(*valid && pb > 0.0);
88            if let Some(within) = &mut within {
89                within.push(*valid && pb > pbmask);
90            }
91        }
92        Ok(Self { support, within })
93    }
94
95    /// A channel's analytic beam, independent of its polarizations' validity.
96    pub(super) fn analytic(
97        setup: &MinorCycleSetup<'_>,
98        (roles, key, taylor): (&[ImageDomainRole], PlaneKey, bool),
99    ) -> Result<Self, MinorCycleRunError> {
100        #[cfg(test)]
101        CHANNEL_BEAM_FORMS.with(|count| count.set(count.get() + 1));
102        let beam = setup.primary_beam.as_ref().expect("an analytic beam");
103        let channel = if taylor { 0 } else { key.channel };
104        let pb = beam.pb(&roles[key.domain], channel)?;
105        let level = pbmask(setup);
106        let mut support = Vec::with_capacity(pb.len());
107        let mut within = (level > 0.0).then(|| Vec::with_capacity(pb.len()));
108        for value in pb {
109            support.push(value > 0.0);
110            if let Some(within) = &mut within {
111                within.push(value > level);
112            }
113        }
114        Ok(Self { support, within })
115    }
116
117    /// Intersect this channel's support with one polarization's valid pixels.
118    pub(super) fn for_valid(&self, valid: &[bool]) -> Self {
119        let intersect = |support: &[bool]| {
120            support
121                .iter()
122                .zip(valid)
123                .map(|(beam, valid)| *beam && *valid)
124                .collect()
125        };
126        Self {
127            support: intersect(&self.support),
128            within: self.within.as_deref().map(intersect),
129        }
130    }
131
132    /// The pixels above the cutoff: the residual's pixel mask.
133    pub(super) fn support(self) -> Vec<bool> {
134        self.support
135    }
136
137    /// The pixels above `pbmask`.
138    pub(super) fn within(self) -> Vec<bool> {
139        self.within.unwrap_or(self.support)
140    }
141
142    /// Both, as supports of `shape`: the pixels above the cutoff, and those
143    /// above a `pbmask` above zero.
144    pub(super) fn into_supports(
145        self,
146        shape: casa_imaging_deconvolution::PlaneShape,
147    ) -> (Support, Option<Support>) {
148        (
149            Support::new(shape, self.support),
150            self.within.map(|within| Support::new(shape, within)),
151        )
152    }
153}

155/// The valid pixels of plane `key` inside its `.pb` ([`BeamPixels`]): the
156/// weight image's Float beam under `response`, else the run's analytic
157/// beam at the plane's channel (channel zero for a Taylor family). `None`
158/// for a run without a `.pb`.
159pub(super) fn beam_pixels(
160    setup: &MinorCycleSetup<'_>,
161    (roles, key, taylor): (&[ImageDomainRole], PlaneKey, bool),
162    valid: &[bool],
163    response: Option<&ResponseBinding<'_>>,
164) -> Result<Option<BeamPixels>, MinorCycleRunError> {
165    if let Some(binding) = response {
166        let beam = MosaicSensitivity::new(&binding.sensitivity)?
167            .with_normal_sum_weight(binding.sum_weight)?;
168        let policy = binding.response.policy();
169        let pb = (0..valid.len()).map(|index| beam.weighted_primary_beam_at(index, policy));
170        return Ok(Some(BeamPixels::new(pb, valid, pbmask(setup))?));
171    }
172    let Some(beam) = &setup.primary_beam else {
173        return Ok(None);
174    };
175    #[cfg(test)]
176    CHANNEL_BEAM_FORMS.with(|count| count.set(count.get() + 1));
177    let channel = if taylor { 0 } else { key.channel };
178    let pb = beam.pb(&roles[key.domain], channel)?;
179    Ok(Some(BeamPixels::new(
180        pb.into_iter().map(Ok),
181        valid,
182        pbmask(setup),
183    )?))
184}

186/// One plane's pixels as its preparation measures them: its valid pixels
187/// and, when the run reads the `.pb`, their part inside it and their part
188/// above a `pbmask` above zero.
189pub(super) struct PlanePixels<'p> {
190    pub(super) valid: &'p Support,
191    pub(super) beam: Option<Support>,
192    pub(super) within: Option<Support>,
193}
194
195impl PlanePixels<'_> {
196    /// The pixels the residual's pixel mask keeps: the beam's under
197    /// `masks_residual`, else every valid pixel.
198    pub(super) fn residual_mask(&self, masks_residual: bool) -> &Support {
199        match &self.beam {
200            Some(beam) if masks_residual => beam,
201            _ => self.valid,
202        }
203    }
204
205    /// The pixels where the `.pb` exceeds `pbmask`, when the run reads it.
206    fn within_beam(&self) -> Option<&Support> {
207        self.within.as_ref().or(self.beam.as_ref())
208    }
209
210    /// The pixels `calcRobustRMS` measures: those where the `.pb` exceeds
211    /// `pbmask`, even none, while the plane's image domain has such a pixel
212    /// in some plane (`domain_beam`); every valid pixel otherwise, or when
213    /// the run has no beam.
214    pub(super) fn robust_rms(&self, domain_beam: bool) -> &Support {
215        match self.within_beam() {
216            Some(beam) if domain_beam => beam,
217            _ => self.valid,
218        }
219    }
220
221    /// The pixels the plane's `nsigma` noise reads under `setup`, at a
222    /// boundary after `completed` major cycles, in an image domain whose
223    /// beam keeps a pixel above `pbmask` in some plane when `domain_beam`:
224    /// after the first boundary of an automatic mask, the mask's pixel
225    /// mask, `pb > pbmask` at a `pbmask` above zero.
226    pub(super) fn noise(
227        &self,
228        setup: &MinorCycleSetup<'_>,
229        completed: usize,
230        domain_beam: bool,
231    ) -> &Support {
232        if !(setup.automask && completed > 0) {
233            return self.robust_rms(domain_beam);
234        }
235        match self.within_beam() {
236            Some(within) if setup.pbmask > 0.0 => within,
237            _ => self.residual_mask(setup.residual_pixel_mask),
238        }
239    }
240}

242/// The plane an automatic mask reads, of `residual` with its `valid`
243/// pixels and their part inside the `.pb`, `beam`, under `setup`: its
244/// pixel mask, and at the `first` boundary of a run with `nsigma`, the
245/// pixels `calcRobustRMS` measured when they differ, those above `pbmask`
246/// while its image domain keeps such a pixel in some plane
247/// (`domain_beam`). At a `pbmask` above zero the pixel mask is
248/// `pb > pbmask` (`SDMaskHandler::autoMaskWithinPB`), which also cuts the
249/// mask the plane makes.
250pub(super) fn automask_plane(
251    setup: &MinorCycleSetup<'_>,
252    (first, domain_beam): (bool, bool),
253    residual: Vec<f64>,
254    valid: Vec<bool>,
255    beam: Option<BeamPixels>,
256) -> AutoMaskPlane {
257    let Some(beam) = beam else {
258        return AutoMaskPlane {
259            residual,
260            valid,
261            statistics: None,
262        };
263    };
264    // `calcRobustRMS` measures the pixels above `pbmask`, or every valid
265    // pixel when the image's beam keeps none.
266    let reuses = first && setup.nsigma > 0.0;
267    if setup.pbmask > 0.0 {
268        return AutoMaskPlane {
269            residual,
270            statistics: (reuses && !domain_beam).then_some(valid),
271            valid: beam.within(),
272        };
273    }
274    let beam = beam.support();
275    let (valid, statistics) = match (setup.residual_pixel_mask, reuses, domain_beam) {
276        (true, true, false) => (beam, Some(valid)),
277        (true, _, _) => (beam, None),
278        (false, true, true) => (valid, Some(beam)),
279        (false, _, _) => (valid, None),
280    };
281    AutoMaskPlane {
282        residual,
283        valid,
284        statistics,
285    }
286}

288/// tclean's robust noise of one plane under `fastnoise=False`
289/// (`SIImageStore::calcRobustRMS`, `SDMaskHandler::autoMask`): Chauvenet
290/// statistics of the measured pixels without a mask, the classical
291/// statistics outside the mask otherwise, and zero for a plane whose
292/// automatic mask stopped evolving.
293pub(super) fn robust_plane_noise(
294    residual: &[f64],
295    measured: &Support,
296    exclusion: NoiseMaskExclusion<'_>,
297) -> Option<RobustNoise> {
298    let measured = measured.as_slice();
299    match exclusion {
300        NoiseMaskExclusion::Stopped => Some(RobustNoise {
301            median: 0.0,
302            rms: 0.0,
303        }),
304        NoiseMaskExclusion::Unmasked => {
305            plane_noise(residual, measured, NoiseEstimate::Robust, None, false)
306        }
307        NoiseMaskExclusion::Mask { support, full } => plane_noise(
308            residual,
309            measured,
310            NoiseEstimate::Robust,
311            Some(support),
312            full,
313        ),
314    }
315}

317/// Whether each image domain's `.pb` exceeds `pbmask` at a pixel in some
318/// plane of `keys` that holds data: CAS-13318's fallback to every pixel is
319/// `calcRobustRMS`'s for an image whose beam keeps none, decided over the
320/// whole cube. Planes are read one at a time until one keeps a pixel.
321pub(super) fn domain_beams(
322    normal: &FinalNormalState,
323    base: &ModelGeneration,
324    setup: &MinorCycleSetup<'_>,
325    taylor: bool,
326    keys: &[PlaneKey],
327) -> Result<Box<[bool]>, MinorCycleRunError> {
328    let roles = base.shape().domain_roles();
329    let mut beams = vec![false; normal.domain_count()];
330    for key in keys {
331        if beams[key.domain] || !super::holds_data(normal, taylor, *key) {
332            continue;
333        }
334        // Analytic beams do not depend on polarization. If an earlier
335        // polarization had data, this channel's beam was already tested.
336        if setup.response.is_none()
337            && setup.primary_beam.is_some()
338            && (0..key.polarization).any(|polarization| {
339                super::holds_data(
340                    normal,
341                    taylor,
342                    PlaneKey {
343                        polarization,
344                        ..*key
345                    },
346                )
347            })
348        {
349            continue;
350        }
351        // A run without a beam never falls back.
352        #[cfg(test)]
353        DOMAIN_BEAM_PROBES.with(|count| count.set(count.get() + 1));
354        beams[key.domain] = plane_beam(normal, setup, (roles, *key, taylor), pbmask(setup))?
355            .is_none_or(|kept| kept.contains(&true));
356    }
357    Ok(beams.into_boxed_slice())
358}

360/// The pixels of plane `key`, x-major, whose `.pb` exceeds `level`, zero
361/// or more (`pb > level`): the weight image's Float beam under a response
362/// (its polarization's Taylor family's principal one), else the analytic beam at the
363/// plane's channel (channel zero for a Taylor family). A plane no weighted
364/// sample reached has a zero `.pb`; `None` for a run without a `.pb`.
365pub(super) fn plane_beam(
366    normal: &FinalNormalState,
367    setup: &MinorCycleSetup<'_>,
368    (roles, key, taylor): (&[ImageDomainRole], PlaneKey, bool),
369    level: f32,
370) -> Result<Option<Vec<bool>>, MinorCycleRunError> {
371    let Some(response) = setup.response else {
372        let Some(beam) = &setup.primary_beam else {
373            return Ok(None);
374        };
375        let channel = if taylor { 0 } else { key.channel };
376        return Ok(Some(
377            beam.pb(&roles[key.domain], channel)?
378                .into_iter()
379                .map(|pb| pb > level)
380                .collect(),
381        ));
382    };
383    let (sensitivity, sum_weight) = if taylor {
384        let window = normal.read_window(normal.slab().core_range())?;
385        let principal = window
386            .normal_moment(0, key.polarization)
387            .expect("a Taylor family has its zeroth moment");
388        (
389            Cow::Owned(principal.sensitivity().to_vec()),
390            principal.sum_weight(),
391        )
392    } else {
393        let plane = normal.read_plane(key.domain, key.channel, key.polarization)?;
394        (plane.read_sensitivity()?, plane.sum_weight())
395    };
396    // A plane whose rows all miss the grid holds an AW weight image (they
397    // weigh at the uv origin) and no sum: `SIImageStore::divideImageByWeightVal`
398    // zeroes it, so it has no `.pb` to cut by.
399    let Some(beam) = MosaicSensitivity::try_new(&sensitivity)?.filter(|_| sum_weight > 0.0) else {
400        return Ok(Some(vec![false; sensitivity.len()]));
401    };
402    let beam = beam.with_normal_sum_weight(sum_weight)?;
403    let policy = response.policy();
404    let kept = (0..sensitivity.len())
405        .map(|index| Ok(beam.weighted_primary_beam_at(index, policy)? > level))
406        .collect::<Result<Vec<_>, ProductsError>>()?;
407    Ok(Some(kept))
