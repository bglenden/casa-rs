2//! Table-specific averaging and iterator controls for automatic flagging.
3
4use crate::{TaskFailure, TaskInvocation};
5use casa_contracts::{EmptyArrayType, NumpyScalarType, ParameterValue};
6use casa_ms::FlagDataMode;
7use casa_types::quanta::Quantity;
8
9use super::is_calibration_table;
10
11pub(super) fn controls(
12    task: &TaskInvocation<'_>,
13    mode: FlagDataMode,
14    calibration: bool,
15) -> Result<(bool, Option<&'static str>), TaskFailure> {
16    if !calibration || !matches!(mode, FlagDataMode::Tfcrop | FlagDataMode::Rflag) {
17        return Ok((true, None));
18    }
19    let channel_average = average_switch(task, "channelavg")?;
20    let time_average = average_switch(task, "timeavg")?;
21    let vector = if channel_average.is_none() {
22        Some("channelavg")
23    } else {
24        time_average.is_none().then_some("timeavg")
25    };
26    // The calibration iterator uses stored samples. The agent still
27    // parses its time-bin string when time averaging was requested.
28    if time_average == Some(true) {
29        let timebin = task.required_text("timebin")?;
30        let c_string = timebin
31            .split_once('\0')
32            .map_or(timebin.as_str(), |(head, _)| head);
33        return Ok((Quantity::read(c_string).is_ok(), vector));
34    }
35    Ok((true, vector))
36}

38pub(super) fn average_switch(
39    task: &TaskInvocation<'_>,
40    name: &'static str,
41) -> Result<Option<bool>, TaskFailure> {
42    let boolean = |value: &ParameterValue| {
43        matches!(
44            value,
45            ParameterValue::Bool(_)
46                | ParameterValue::NumpyScalar {
47                    dtype: NumpyScalarType::Bool,
48                    ..
49                }
50        )
51    };
52    match task.value(name) {
53        None => Ok(Some(false)),
54        Some(ParameterValue::Bool(value)) => Ok(Some(*value)),
55        Some(ParameterValue::NumpyScalar {
56            dtype: NumpyScalarType::Bool,
57            ..
58        }) => Ok(Some(false)),
59        Some(ParameterValue::Array(items)) if items.iter().all(boolean) => Ok(None),
60        Some(ParameterValue::ScalarArray(value)) if boolean(value) => Ok(None),
61        Some(ParameterValue::EmptyArray {
62            dtype: EmptyArrayType::Bool,
63            shape,
64        }) if shape.len() == 1 => Ok(None),
65        _ => Err(
66            TaskFailure::validation(format!("{name} must be a Boolean or Boolean vector")).at(name),
67        ),
68    }
69}

71pub(super) fn table_kind(
72    task: &TaskInvocation<'_>,
73    mode: &str,
74) -> Result<Option<bool>, TaskFailure> {
75    if !matches!(mode, "tfcrop" | "rflag") {
76        return Ok(None);
77    }
78    let calibration = is_calibration_table(&task.path("vis")?);
79    if calibration {
80        for name in ["channelavg", "timeavg"] {
81            average_switch(task, name)?;
82        }
83    }
84    Ok(Some(calibration))
85}
