5use std::collections::BTreeMap;

7use casa_imaging_deconvolution::{

9};
10
11use super::{
12    MinorCycleRunError, PsfCache, RetainedPlane, normalise, plane_shape, psf_peak_value, read_psf,
13};

15/// Give a plane whose PSF fit was refused its domain's largest beam by area
16/// (`SIImageStore::makeImageBeamSet`), as the products do. CASA's cube takes the
17/// largest of each sub-cube its memory splits it into, so its answer depends on
18/// the machine; this is its answer for one. A domain with no fitted plane takes
19/// CASA's tiny circular beam. A plane is read again only when its beam changes,
20/// which is rarely.
21pub(super) fn settle_refused_fits(
22    normal: &FinalNormalState,
23    base: &ModelGeneration,
24    taylor: bool,
25    cache: &mut PsfCache,
26    preserve_main: bool,
27    retained: Option<&RetainedPlane>,
28) -> Result<(), MinorCycleRunError> {
29    let mut largest = BTreeMap::<usize, RestoringBeam>::new();
30    for (key, summary) in cache.summaries.iter().filter(|(_, s)| !s.fit_refused()) {
31        let beam = summary.beam();
32        let area = |beam: RestoringBeam| beam.major_fwhm_rad() * beam.minor_fwhm_rad();
33        largest
34            .entry(key.domain)
35            .and_modify(|most| {
36                if area(beam) > area(*most) {
37                    *most = beam;
38                }
39            })
40            .or_insert(beam);
41    }
42    let reused = cache.reuses_single_psf;
43    for (key, summary) in &mut cache.summaries {
44        if preserve_main && key.domain == 0 {
45            continue;
46        }
47        if !summary.fit_refused() {
48            continue;
49        }
50        let increment = base
51            .shape()
52            .direction(key.domain)
53            .expect("domain direction")
54            .increment_rad();
55        let beam = match largest.get(&key.domain).copied() {
56            Some(beam) => beam,
57            None => {
58                let width = CASA_TINY_PSF_BEAM_RAD / increment[0].abs();
59                RestoringBeam::new(width, width, 0.0)
60                    .map_err(casa_imaging_deconvolution::Error::from)?
61            }
62        };
63        if summary.needs_domain_beam() || summary.beam() != beam {
64            let shape = plane_shape(normal, *key);
65            let (psf, peak) = match retained.filter(|plane| plane.key == *key) {
66                Some(plane) => (Cow::Borrowed(plane.psf[0].as_slice()), plane.psf_peak),
67                None => {
68                    let psf = read_psf(normal, taylor, *key)?;
69                    let peak = psf_peak_value(&psf, shape)?;
70                    (Cow::Owned(normalise(psf, peak)), peak)
71                }
72            };
73            let cell = [1.0, (increment[1] / increment[0]).abs()];
74            *summary =
75                summary.settle_domain_beam(&psf, shape, beam, cell, reused.then_some(peak))?;
76        }
77    }
78    Ok(())
79}
