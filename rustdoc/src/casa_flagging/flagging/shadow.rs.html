19/// A dish used for flagging shadows. Additional dishes do not become MS rows.
20#[derive(Clone, Copy, Debug)]
21pub struct ShadowAntenna {
22    /// ITRF position in meters.
23    pub position_m: [f64; 3],
24    /// Dish diameter in meters.
25    pub diameter_m: f64,
26}
27
28/// Controls CASA-style flagging of baselines that include a shadowed dish.
29#[derive(Clone, Debug)]
30pub struct ShadowOptions {
31    /// Allowed overlap of the projected dish diameters, in meters.
32    pub tolerance_m: f64,
33    /// Dishes outside the MS antenna table that can cast shadows.
34    pub additional: Vec<ShadowAntenna>,
35    /// Whether the parsed task configuration admits a shadow agent.
36    pub enabled: bool,
37}
38
39impl Default for ShadowOptions {
40    fn default() -> Self {
41        Self {
42            tolerance_m: 0.0,
43            additional: Vec::new(),
44            enabled: true,
45        }
46    }
47}

49pub(super) fn apply(
50    ms: &mut MeasurementSet,
51    selected: &[usize],
52    options: &ShadowOptions,
53    channels: Option<&BTreeMap<i32, ChannelSelection>>,
54    correlations: Option<&CorrelationSelection>,
55    path: &str,
56    cancellation: Option<&Cancellation>,
57) -> Result<ChangeSet, FlaggingError> {
58    if !options.enabled {
59        return Ok(ChangeSet {
60            agent_skipped: true,
61            ..ChangeSet::default()
62        });
63    }
64    if selected.is_empty() {
65        return Ok(ChangeSet::default());
66    }
67    let affected = shadowed_rows(ms, selected, options, path, cancellation)?;
68    apply_geometric_flags(ms, &affected, channels, correlations, cancellation)
69}

71struct Baseline {
72    centroid: f64,
73    first: usize,
74    second: usize,
75    uvw_m: [f64; 3],

78fn shadowed_rows(
79    ms: &MeasurementSet,
80    selected: &[usize],
81    options: &ShadowOptions,
82    path: &str,
83    cancellation: Option<&Cancellation>,
84) -> Result<Vec<usize>, FlaggingError> {
85    let fail = |reason: String| FlaggingError::MutateFlags {
86        path: path.to_owned(),
87        reason,
88    };
89    let antenna_table = ms.antenna().map_err(|error| fail(error.to_string()))?;
90    let antenna_count = antenna_table.row_count();
91    let antennas = (0..antenna_count)
92        .map(|row| {
93            Ok(ShadowAntenna {
94                position_m: antenna_table.position(row)?,
95                diameter_m: antenna_table.dish_diameter(row)?,
96            })
97        })
98        .collect::<MsResult<Vec<_>>>()
99        .map_err(|error| fail(error.to_string()))?;
100    let antennas: Vec<_> = antennas
101        .into_iter()
102        .chain(options.additional.iter().copied())
103        .collect();
104    let mut ordered = read_clip_times(ms, selected, path, cancellation)?;
105    sort_iterator_rows(&mut ordered);
106    let bytes_per_row = size_of::<Baseline>()
107        + size_of::<usize>()
108        + size_of::<Option<ScalarValue>>()
109        + size_of::<Option<ArrayValue>>()
110        + size_of::<[f64; 3]>();
111    let budget = MsSelectionIoBudget::from_system_memory(2, bytes_per_row, None)
112        .map_err(|error| fail(error.to_string()))?;
113    let plan = MsReadPlan::new(ordered.len(), budget).map_err(|error| fail(error.to_string()))?;
114    let batch_rows = plan.rows_per_block.max(1);
115    let mut affected = Vec::new();
116    let mut engine = None;
117    let mut measures = None;
118    let mut field = None;
119    let mut field_start = 0.0;
120    let mut center = None;
121    let mut time = None;
122    let mut shadowed = vec![false; antennas.len()];
123    for block in integration_blocks(&ordered, batch_rows) {
124        // Physical I/O is bounded even when one logical integration is large.
125        let mut baselines = Vec::with_capacity(block.len());
126        for rows in block.chunks(batch_rows) {
127            check_cancelled(cancellation)?;
128            baselines.extend(read_baselines(ms, rows, antenna_count, fail)?);
129        }
130        let mut unread = baselines.as_slice();
131        for integration in block.chunk_by(ClipTimedRow::shares_integration) {
132            check_cancelled(cancellation)?;
133            let next_field = usize::try_from(integration[0].field())
134                .map_err(|_| fail(format!("negative FIELD_ID at row {}", integration[0].row)))?;
135            if field != Some(next_field) {
136                field = Some(next_field);
137                field_start = integration[0].time();
138                center = None;
139            }
140            let (baselines, remaining) = unread.split_at(integration.len());
141            unread = remaining;
142            for (index, baseline) in baselines.iter().enumerate() {
143                if time != Some(baseline.centroid) {
144                    time = Some(baseline.centroid);
145                    let rows = baselines[index..]
146                        .iter()
147                        .take_while(|candidate| candidate.centroid <= baseline.centroid);
148                    shadowed = shadowed_antennas(&antennas, rows, options.tolerance_m, || {
149                        if engine.is_none() {
150                            // CASA's visibility-buffer phase center also
151                            // accepts legacy geocentric apparent ephemerides.
152                            engine = Some(MsCalEngine::for_calibration(ms)?);
153                            measures = Some(open_measures_runtime()?);
154                        }
155                        if center.is_none() {
156                            center = Some(phase_center(
157                                ms,
158                                engine.as_ref().expect("shadow measures are initialized"),
159                                next_field,
160                                field_start,
161                            )?);
162                        }
163                        let direction = center.as_ref().expect("the field has a phase center");
164                        project_positions(
165                            &antennas,
166                            direction,
167                            baseline.centroid,
168                            Arc::clone(measures.as_ref().expect("shadow measures are initialized")),
169                        )
170                    })
171                    .map_err(|error| fail(error.to_string()))?;
172                }
173                if shadowed[baseline.first] || shadowed[baseline.second] {
174                    affected.push(integration[index].row);
175                }
176            }
177        }
178    }
179    affected.sort_unstable();
180    Ok(affected)
181}

183fn read_baselines(
184    ms: &MeasurementSet,
185    rows: &[ClipTimedRow],
186    antenna_count: usize,
187    fail: impl Fn(String) -> FlaggingError,
188) -> Result<Vec<Baseline>, FlaggingError> {
189    let indices: Vec<_> = rows.iter().map(|row| row.row).collect();
190    let table = ms.main_table();
191    let times = table
192        .column_accessor("TIME_CENTROID")
193        .and_then(|column| column.scalar_cells_owned_for_rows(&indices))
194        .map_err(|error| fail(error.to_string()))?;
195    let uvws = table
196        .column_accessor("UVW")
197        .and_then(|column| column.array_cells_owned_uncached(&indices))
198        .map_err(|error| fail(error.to_string()))?;
199    rows.iter()
200        .zip(times)
201        .zip(uvws)
202        .map(|((row, time), uvw)| {
203            let antenna = |id: i32| {
204                usize::try_from(id)
205                    .ok()
206                    .filter(|&index| index < antenna_count)
207                    .ok_or_else(|| fail(format!("invalid antenna {id} at MAIN row {}", row.row)))
208            };
209            let [first, second] = row.antennas();
210            let centroid = match time {
211                Some(ScalarValue::Float64(seconds)) if seconds.is_finite() => seconds,
212                _ => {
213                    return Err(fail(format!(
214                        "TIME_CENTROID row {} must be a finite Double",
215                        row.row
216                    )));
217                }
218            };
219            let uvw_m = match uvw {
220                Some(ArrayValue::Float64(array)) if array.shape() == [3] => {
221                    [array[[0]], array[[1]], array[[2]]]
222                }
223                _ => {
224                    return Err(fail(format!(
225                        "UVW row {} must have three Double values",
226                        row.row
227                    )));
228                }
229            };
230            Ok(Baseline {
231                centroid,
232                first: antenna(first)?,
233                second: antenna(second)?,
234                uvw_m,
235            })
236        })
237        .collect()
238}

240fn shadowed_antennas<'a>(
241    antennas: &[ShadowAntenna],
242    observed: impl IntoIterator<Item = &'a Baseline>,
243    tolerance: f64,
244    project: impl FnOnce() -> MsResult<Vec<[f64; 3]>>,
245) -> MsResult<Vec<bool>> {
246    let mut shadowed = vec![false; antennas.len()];
247    let mut present = BTreeSet::new();
248    let mut check = |first: usize, second: usize, [u, v, w]: [f64; 3]| {
249        let separation = (u * u + v * v).sqrt();
250        let limit = (antennas[first].diameter_m + antennas[second].diameter_m) / 2.0 - tolerance;
251        if separation < limit {
252            shadowed[if w < 0.0 { first } else { second }] = true;
253        }
254    };
255    for row in observed {
256        if row.first != row.second {
257            present.insert((row.first.min(row.second), row.first.max(row.second)));
258            check(row.first, row.second, row.uvw_m);
259        }
260    }
261    let count = antennas.len();
262    if present.len() != count * count.saturating_sub(1) / 2 {
263        let positions = project()?;
264        for (first, position) in positions.iter().enumerate() {
265            for (second, other) in positions.iter().enumerate().skip(first + 1) {
266                if !present.contains(&(first, second)) {
267                    let baseline = std::array::from_fn(|axis| other[axis] - position[axis]);
268                    check(first, second, baseline);
269                }
270            }
271        }
272    }
273    Ok(shadowed)
274}

276fn project_positions(
277    antennas: &[ShadowAntenna],
278    center: &MDirection,
279    centroid: f64,
280    measures: Arc<dyn MeasuresProvider>,
281) -> MsResult<Vec<[f64; 3]>> {
282    let origin = antennas[0].position_m;
283    let position = MPosition::new_itrf(origin[0], origin[1], origin[2]);
284    let epoch = MEpoch::from_mjd_seconds(centroid, EpochRef::UT1);
285    let direction = if center.refer() == DirectionRef::J2000 {
286        center.clone()
287    } else {
288        let frame = MeasFrame::new()
289            .with_position(position.clone())
290            .with_epoch(epoch.clone())
291            .with_measures(measures.clone());
292        center.convert_to(DirectionRef::J2000, &frame)?
293    };
294    let projection = UvwFrame::new(direction.as_angles().into(), epoch, &position, measures)?;
295    Ok(antennas
296        .iter()
297        .map(|antenna| {
298            let offset = std::array::from_fn(|axis| origin[axis] - antenna.position_m[axis]);
299            projection.baseline_uvw(offset)
300        })
301        .collect())
302}
