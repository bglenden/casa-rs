19/// The front end of the `importvla` binary.
20pub fn task_front_end()
21-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
22    TaskFrontEnd::new(
23        "importvla",
24        importvla_task_schema_bundle(),
25        &["importvla"],
26        run_task,
27    )
28    .expect("the importvla contract carries its surface")
29}

31/// Build the machine-readable command schema for `importvla`.
32pub fn command_schema(program_name: &str) -> UiCommandSchema {
33    let bundle = casa_provider_contracts::builtin_surface_bundle("importvla")
34        .expect("built-in importvla parameter surface must remain valid");
35    let mut schema: UiCommandSchema =
36        serde_json::from_value(casa_provider_contracts::project_ui_form(&bundle))
37            .expect("canonical importvla UI projection must match UiCommandSchema");
38    schema.invocation_name = program_name.to_string();
39    schema.usage = format!("{program_name} [parameters]");
40    schema
41}

43/// Import one resolved request; CASA's `importvla` returns None.
44fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
45    let options = options_from(task)?;
46    let report = run(&options).map_err(failure)?;
47    let text = import_text(&report);
48    Ok(TaskSuccess::new(
49        casa_task_runtime::CasaValue::None,
50        &ImportVlaTaskResult::Import(report),
51    )?
52    .with_text(text))
53}

55/// The resolved run as [`ImportVlaOptions`]. `archivefiles` arrives as the
56/// list of archive paths: a comma-separated string is split before its paths
57/// are normalized, on every route.
58fn options_from(task: &TaskInvocation<'_>) -> Result<ImportVlaOptions, TaskFailure> {
59    let archivefiles = task.paths("archivefiles")?;
60    let mut vis = task.paths("vis")?;
61    if vis.len() > 1 {
62        return Err(TaskFailure::validation("importvla writes one vis").at("vis"));
63    }
64    let options = ImportVlaOptions {
65        archivefiles,
66        vis: vis.pop(),
67        bandname: task
68            .text("bandname")?
69            .map(|text| text.parse())
70            .transpose()
71            .map_err(failure)?,
72        frequencytol_hz: ImportVlaOptions::parse_frequencytol(&task.required_text("frequencytol")?)
73            .map_err(failure)?,
74        project: task.text("project")?,
75        starttime: task.text("starttime")?,
76        stoptime: task.text("stoptime")?,
77        applytsys: task.flag("applytsys")?,
78        autocorr: task.flag("autocorr")?,
79        antnamescheme: task
80            .required_text("antnamescheme")?
81            .parse()
82            .map_err(failure)?,
83        keepblanks: task.flag("keepblanks")?,
84        evlabands: task.flag("evlabands")?,
85    };
86    options.require_archivefiles().map_err(failure)?;
87    Ok(options)
88}

90/// A refused argument is a validation failure at that parameter; anything
91/// else failed the import.
92fn failure(error: VlaError) -> TaskFailure {
93    match &error {
94        VlaError::InvalidArgument { argument, .. } => {
95            let argument = (*argument).to_string();
96            TaskFailure::validation(error.to_string()).at(argument)
97        }
98        VlaError::NoArchiveFiles => TaskFailure::validation(error.to_string()).at("archivefiles"),
99        _ => TaskFailure::execution(error.to_string()),
100    }
101}

103fn run(options: &ImportVlaOptions) -> Result<ImportReport, VlaError> {
104    tracing::info!(
105        archive_count = options.archivefiles.len(),
106        vis = ?options.vis,
107        "importvla run started"
108    );
109    // A failed import removes the default output it started, never one that
110    // existed before the run: that one is a product it refused to replace.
111    let cleanup_vis = if options.vis.is_none() {
112        Some(options.effective_vis_for_import()?).filter(|path| !path.exists())
113    } else {
114        None
115    };
116    let report = match import_archive_files_to_measurement_set_from_options(options) {
117        Ok(report) => report,
118        Err(error) => {
119            tracing::error!(
120                casa.priority = "SEVERE",
121                error = %error,
122                cleanup_vis = ?cleanup_vis,
123                "importvla import failed"
124            );
125            if let Some(path) = cleanup_vis.as_deref().filter(|path| path.exists()) {
126                cleanup_failed_import_output(path).map_err(|cleanup_error| {
127                    VlaError::import(format!(
128                        "{error}; cleanup failed for {}: {cleanup_error}",
129                        path.display()
130                    ))
131                })?;
132            }
133            return Err(error);
134        }
135    };
136    tracing::info!(
137        archive_count = options.archivefiles.len(),
138        vis = ?report.vis,
139        rows = report.main_rows_written,
140        "importvla run completed"
141    );
142    Ok(report)
143}
144
145fn cleanup_failed_import_output(path: &Path) -> Result<(), std::io::Error> {
146    if path.is_dir() {
147        fs::remove_dir_all(path)
148    } else {
149        fs::remove_file(path)
150    }
151}
152
153fn import_text(report: &ImportReport) -> String {
154    format!(
155        "importvla disk import\nvis: {}\nlogical records seen: {}\nlogical records imported: {}\nlogical records skipped: {}\nmain rows written: {}\n",
156        report.vis.display(),
157        report.logical_records_seen,
158        report.logical_records_imported,
159        report.logical_records_skipped,
160        report.main_rows_written
161    )
162}
