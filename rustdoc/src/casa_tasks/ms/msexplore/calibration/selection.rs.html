2//! Resolve CALTABLE selections with the native MS metadata grammars.
3
4use std::collections::{BTreeMap, BTreeSet};

6use casa_ms::selection::antenna_expr::{AntennaRow, BaselineSelection, resolve_antenna_selector};
7use casa_ms::selection::parser::{FieldSelectorRow, resolve_field_selector_rows};
8use casa_ms::selection::spw_expr::{DataDescriptionRow, SpectralWindowRow, resolve_spw_selector};
9use casa_ms::selection::syntax::parse_numeric_id_selector;
10use casa_ms::selection::time_expr::{
11    TimeDefaults, TimeSelection, TimeSelector, TimeSelectorDefaults,
12};

14use super::axis::SampleCoordinates;
15use super::table_data::CalibrationTable;

17#[derive(Default)]
18pub(super) struct Expressions<'a> {
19    pub(super) antenna: &'a str,
20    pub(super) field: &'a str,
21    pub(super) spw: &'a str,
22    pub(super) scan: &'a str,
23    pub(super) timerange: &'a str,
24    pub(super) observation: &'a str,
25    pub(super) taql: &'a str,
26}
27
28pub(super) struct Selection {
29    antennas: BaselineSelection,
30    use_baseline: bool,
31    fields: Vec<i32>,
32    scans: Vec<i32>,
33    observations: Vec<i32>,
34    time: Option<TimeSelection>,
35    taql_rows: Option<BTreeSet<usize>>,
36    all_windows: bool,
37    channel_lists: BTreeMap<i32, Vec<usize>>,
38}
39
40impl Selection {
41    pub(super) fn resolve(
42        table: &CalibrationTable,
43        expressions: &Expressions<'_>,
44    ) -> Result<Self, String> {
45        let antenna_rows = table
46            .antennas
47            .iter()
48            .map(|row| AntennaRow {
49                name: &row.name,
50                station: &row.station,
51                position_m: row.position,
52            })
53            .collect::<Vec<_>>();
54        let antennas =
55            resolve_antenna_selector(&antenna_rows, expressions.antenna).map_err(error)?;
56        let field_rows = table
57            .fields
58            .iter()
59            .map(|row| FieldSelectorRow {
60                name: &row.name,
61                code: Some(&row.code),
62                flagged: row.flagged,
63            })
64            .collect::<Vec<_>>();
65        let fields = resolve_field_selector_rows(&field_rows, expressions.field).map_err(error)?;
66        let scans = parse_numeric_id_selector(expressions.scan, "scan").map_err(error)?;
67        let observations =
68            parse_numeric_id_selector(expressions.observation, "observation").map_err(error)?;
69        let windows = table
70            .windows
71            .iter()
72            .map(|row| SpectralWindowRow {
73                name: &row.name,
74                chan_freq_hz: &row.frequencies,
75                chan_width_hz: &row.widths,
76                ref_frequency_hz: row.reference,
77                total_bandwidth_hz: row.bandwidth,
78                flag_row: row.flagged,
79            })
80            .collect::<Vec<_>>();
81        let descriptions = windows
82            .iter()
83            .enumerate()
84            .map(|(id, row)| DataDescriptionRow {
85                spectral_window_id: id as i32,
86                flag_row: row.flag_row,
87            })
88            .collect::<Vec<_>>();
89        let spw = resolve_spw_selector(&windows, &descriptions, expressions.spw).map_err(error)?;
90        let mut channel_lists = BTreeMap::new();
91        for &window in spw.spw_ids() {
92            let channels = spw.channels_of(window).map_err(error)?.unwrap_or_default();
93            channel_lists.insert(window as i32, channels);
94        }
95        let mut selection = Self {
96            antennas,
97            use_baseline: expressions.antenna.contains('&'),
98            fields,
99            scans,
100            observations,
101            time: None,
102            taql_rows: table.taql_rows(expressions.taql)?,
103            all_windows: spw.selects_all(),
104            channel_lists,
105        };
106        if !expressions.timerange.is_empty() && table.rows() > 0 {
107            let defaults = |row: SampleCoordinates| TimeDefaults {
108                time_mjd_s: row.time,
109                // MSTimeParse uses 0.1 seconds for calibration tables.
110                exposure_s: 0.1,
111            };
112            let row_zero = table.coordinates(0)?;
113            let mut first = row_zero;
114            for index in 0..table.rows() {
115                let row = table.coordinates(index)?;
116                if selection.includes(&row) {
117                    first = row;
118                    break;
119                }
120            }
121            selection.time = Some(
122                TimeSelector::parse(expressions.timerange)
123                    .map_err(error)?
124                    .resolve(TimeSelectorDefaults {
125                        first_term: defaults(first),
126                        later_terms: defaults(row_zero),
127                    })
128                    .map_err(error)?,
129            );
130        }
131        Ok(selection)
132    }
133
134    pub(super) fn includes(&self, point: &SampleCoordinates) -> bool {
135        let includes_id = |values: &[i32], id| values.is_empty() || values.contains(&id);
136        let [primary, secondary] = point
137            .antennas
138            .map(|id| usize::try_from(id).unwrap_or(usize::MAX));
139        let secondary = if self.use_baseline {
140            secondary
141        } else {
142            usize::MAX
143        };
144        self.taql_rows
145            .as_ref()
146            .is_none_or(|rows| rows.contains(&point.row))
147            && includes_id(&self.fields, point.field)
148            && includes_id(&self.scans, point.scan)
149            && includes_id(&self.observations, point.observation)
150            && self.time.as_ref().is_none_or(|time| time.holds(point.time))
151            && self.antennas.holds(primary, secondary)
152            && (self.all_windows || self.channel_lists.contains_key(&point.window))
153    }
154
155    pub(super) fn channels(&self, window: i32, count: usize) -> Vec<usize> {
156        self.channel_lists
157            .get(&window)
158            .filter(|channels| !channels.is_empty())
159            .map_or_else(
160                || (0..count).collect(),
161                |channels| {
162                    channels
163                        .iter()
164                        .copied()
165                        .take_while(|&channel| channel < count)
166                        .collect()
167                },
168            )
169    }
170}
171
172fn error(error: impl std::fmt::Display) -> String {
173    error.to_string()
174}
