3use std::fs::{File, Metadata};
4use std::io::{self, Write};
5use std::path::{Path, PathBuf};
6
7use pyo3::prelude::*;
8
9use super::handle::raised;
10
11/// A text report written beside its destination and installed atomically.
12pub(super) struct StagedTextOutput {
13    path: PathBuf,
14    previous: Option<Metadata>,
15    file: tempfile::NamedTempFile,
16}
17
18impl StagedTextOutput {
19    pub(super) fn new(path: &Path, text: &str, append: bool) -> PyResult<Self> {
20        let (path, previous) = match path.symlink_metadata() {
21            Ok(_) => {
22                let target = path.canonicalize().map_err(raised)?;
23                let metadata = target.metadata().map_err(raised)?;
24                if !metadata.is_file() {
25                    return Err(raised(format!(
26                        "{} is not a regular log file",
27                        path.display()
28                    )));
29                }
30                (target, Some(metadata))
31            }
32            Err(error) if error.kind() == io::ErrorKind::NotFound => (path.to_path_buf(), None),
33            Err(error) => return Err(raised(error)),
34        };
35        let parent = path
36            .parent()
37            .filter(|parent| !parent.as_os_str().is_empty())
38            .unwrap_or_else(|| Path::new("."));
39        let mut file = tempfile::NamedTempFile::new_in(parent).map_err(raised)?;
40        if let Some(metadata) = &previous {
41            file.as_file()
42                .set_permissions(metadata.permissions())
43                .map_err(raised)?;
44            if append {
45                io::copy(&mut File::open(&path).map_err(raised)?, file.as_file_mut())
46                    .map_err(raised)?;
47            }
48        }
49        file.write_all(text.as_bytes()).map_err(raised)?;
50        file.flush().map_err(raised)?;
51        Ok(Self {
52            path,
53            previous,
54            file,
55        })
56    }
57
58    pub(super) fn install(self) -> PyResult<()> {
59        if let Some(previous) = self.previous {
60            let current = self.path.symlink_metadata().map_err(raised)?;
61            if !current.is_file()
62                || previous.len() != current.len()
63                || previous.modified().ok() != current.modified().ok()
64            {
65                return Err(raised(format!(
66                    "Log {} changed during fitting",
67                    self.path.display()
68                )));
69            }
70            self.file.persist(self.path).map_err(raised)?;
71        } else {
72            self.file.persist_noclobber(self.path).map_err(raised)?;
73        }
74        Ok(())
75    }
76}
