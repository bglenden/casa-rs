3//! The frequency conversions compile evaluates at the epochs of the
4//! selected samples ([`Epoch`]): a continuum run's spectral envelope and a
5//! cube's density padding.
6//!
7//! A cube's samples are converted towards their field's direction at
8//! their time, as CASA's visibility iterator converts them; a field that
9//! follows an ephemeris is converted towards its direction at the time, not
10//! its `PHASE_DIR`, which is then an offset. A continuum envelope converts
11//! towards each field's reference direction instead, as CASA's
12//! `MSUtil::getFreqRangeInSpw` does ([`continuum_direction`]).
13

27/// The selected channels' edges in the output frame, over every epoch.
28#[derive(Clone, Copy, Debug, PartialEq)]
29pub(super) struct SpectralEnvelope {
30    edges_hz: [f64; 2],
31}
32
33impl SpectralEnvelope {
34    /// The lower and upper selected-channel edges in the output frame.
35    pub(super) const fn edges_hz(self) -> [f64; 2] {
36        self.edges_hz
37    }
38
39    /// CASA's MFS and Taylor reference frequency: the midpoint of the
40    /// edges.
41    pub(super) fn midpoint_hz(self) -> f64 {
42        (self.edges_hz[0] + self.edges_hz[1]) / 2.0
43    }
44
45    /// The envelope of both: the least lower and the greatest upper edge, as
46    /// `buildCoordinateSystem` spans every MeasurementSet's band.
47    pub(super) fn union(self, other: Self) -> Self {
48        Self {
49            edges_hz: [
50                self.edges_hz[0].min(other.edges_hz[0]),
51                self.edges_hz[1].max(other.edges_hz[1]),
52            ],
53        }
54    }
55}

57/// The observatory frame towards `epoch`'s field at its time: a moving
58/// field's direction then, else a fixed field's stored phase centre, in its
59/// stored frame, as the deleted MAIN walk read `PHASE_DIR`.
60pub(super) fn epoch_frame(
61    (catalogue, engine): (&Catalogue, &MsCalEngine),
62    epoch: Epoch,
63) -> Result<MeasFrame, PrepareError> {
64    let field = usize::try_from(epoch.field_id).expect("a surveyed field is a FIELD row");
65    let direction = if engine.field_moves(field) {
66        engine.observation_direction_j2000(epoch.time_mjd_seconds, field)?
67    } else {
68        catalogue.fields[field].phase_centre.clone()
69    };
70    Ok(engine.spectral_frame_observatory_direction(epoch.time_mjd_seconds, direction)?)
71}

73/// The direction a continuum envelope converts field `field_id`'s samples
74/// towards: `MSUtil::getFreqRangeInSpw` asks `phaseDirMeas(field)` with no
75/// time, which for a field following an ephemeris is its geocentric
76/// direction at its reference epoch (MSv2 `FIELD.TIME`; the field's first
77/// selected time `first_time_mjd_seconds` on a store without one), in the
78/// ephemeris frame and including `PHASE_DIR`'s offset
79/// ([`MsCalEngine::moving_field_reference_direction`]); a fixed field's is
80/// its stored phase centre. Only the conversion epoch follows the samples.
81fn continuum_direction(
82    (catalogue, engine): (&Catalogue, &MsCalEngine),
83    field_id: i32,
84    first_time_mjd_seconds: f64,
85) -> Result<MDirection, PrepareError> {
86    let field = usize::try_from(field_id).expect("a surveyed field is a FIELD row");
87    Ok(
88        match engine.moving_field_reference_direction(field, first_time_mjd_seconds)? {
89            Some(direction) => direction,
90            None => catalogue.fields[field].phase_centre.clone(),
91        },
92    )
93}

95/// The lower and upper edges of `channels` of `window`: each channel's
96/// centre less and plus half its width.
97fn selected_edges(
98    window: &SourceSpectralWindow,
99    channels: &[usize],
100) -> Result<[f64; 2], PrepareError> {
101    let invalid = || PrepareError::SpectralWindowCoordinates {
102        spw_id: window.spw_id,
103    };
104    let mut edges = [f64::INFINITY, f64::NEG_INFINITY];
105    for &channel in channels {
106        let (Some(&centre), Some(&width)) = (
107            window.frequencies_hz.get(channel),
108            window.channel_widths_hz.get(channel),
109        ) else {
110            return Err(invalid());
111        };
112        if !(centre.is_finite() && width.is_finite() && width != 0.0) {
113            return Err(invalid());
114        }
115        edges[0] = edges[0].min(centre - width.abs() / 2.0);
116        edges[1] = edges[1].max(centre + width.abs() / 2.0);
117    }
118    if edges[0].is_finite() && edges[1] > edges[0] {
119        Ok(edges)
120    } else {
121        Err(invalid())
122    }
123}

125/// The envelope in frame `output` of the `selected` channels of `windows`
126/// over each window's epochs in `survey`: LSRK, where CASA grids an MFS
127/// image's samples, or the image's `outframe`, which its header records.
128///
129/// It follows CASA's `MSUtil::getFreqRangeInSpw` (`fromEdge`), which
130/// `SynthesisUtilMethods::buildCoordinateSystem` uses for the MFS spectral
131/// axis: each window converts its selected channel edges, low edge first,
132/// at each epoch of its samples in time order, through **one**
133/// `MFrequency::Convert`, whose aberration and dUT1 are anchored at the
134/// first epoch and extrapolated across 0.04 days
135/// ([`MFrequencyConverter`]); that moves the far end of a long track by
136/// about 1e-12. CASA keys its epochs by `TIME·(DATA_DESC_ID+1)`, each
137/// standing for the earliest row with the key; a window's samples at one
138/// time towards one field are one epoch here. CASA takes `REST` edges, and
139/// edges already in a frame other than `TOPO` that is the output's, as
140/// they are.
141pub(super) fn continuum_envelope(
142    (catalogue, engine): (&Catalogue, &MsCalEngine),
143    survey: &Survey,
144    selected: &WindowChannels,
145    output: FrequencyRef,
146    field: Option<i32>,
147) -> Result<SpectralEnvelope, PrepareError> {
148    let mut edges_hz = [f64::INFINITY, f64::NEG_INFINITY];
149    // Each field's direction, once: only the conversion epoch varies.
150    let mut directions = BTreeMap::<i32, MDirection>::new();
151    for window in &survey.spectral_windows {
152        let channels = selected
153            .get(&window.spw_id)
154            .expect("the continuum selects channels of every selected window");
155        let edges = selected_edges(window, channels)?;
156        fold_over_epochs(
157            (catalogue, engine),
158            (survey, window),
159            &edges,
160            (output, field),
161            &mut directions,
162            &mut |hz| {
163                edges_hz[0] = edges_hz[0].min(hz);
164                edges_hz[1] = edges_hz[1].max(hz);
165            },
166        )?;
167    }
168    if !(edges_hz[0].is_finite() && edges_hz[1].is_finite() && edges_hz[1] > edges_hz[0]) {
169        return Err(PrepareError::SpectralEnvelope { edges_hz });
170    }
171    Ok(SpectralEnvelope { edges_hz })
172}

174/// CASA's band of channels `[first, last]` of `window` in frame `output`
175/// (`SynthesisImagerVi2::selectData`, by `MSUtil::getFreqRangeInSpw`
176/// without `fromEdge`): the lowest and highest of the channels' centres
177/// converted at each epoch of the window's samples, as
178/// [`continuum_envelope`] converts edges. `None` when the window has no
179/// epoch to convert at, where CASA selects nothing of it by frequency.
180pub(super) fn selection_band(
181    (catalogue, engine): (&Catalogue, &MsCalEngine),
182    (survey, window): (&Survey, &SourceSpectralWindow),
183    [first, last]: [usize; 2],
184    output: FrequencyRef,
185) -> Result<Option<[f64; 2]>, PrepareError> {
186    let centres =
187        window
188            .frequencies_hz
189            .get(first..=last)
190            .ok_or(PrepareError::SpectralWindowCoordinates {
191                spw_id: window.spw_id,
192            })?;
193    let lowest = centres.iter().copied().fold(f64::INFINITY, f64::min);
194    let highest = centres.iter().copied().fold(f64::NEG_INFINITY, f64::max);
195    let mut band = [f64::INFINITY, f64::NEG_INFINITY];
196    let converted = fold_over_epochs(
197        (catalogue, engine),
198        (survey, window),
199        &[lowest, highest],
200        (output, None),
201        &mut BTreeMap::new(),
202        &mut |hz| {
203            band[0] = band[0].min(hz);
204            band[1] = band[1].max(hz);
205        },
206    )?;
207    Ok(converted.then_some(band))
208}

210/// Fold each of `frequencies_hz`, in `window`'s frame, into `fold` in frame
211/// `output`, as `MSUtil::getFreqRangeInSpw` converts a window's
212/// frequencies: at each epoch of the window's samples in time order,
213/// through **one** `MFrequency::Convert`, whose aberration and dUT1 are
214/// anchored at the first epoch and extrapolated across 0.04 days
215/// ([`MFrequencyConverter`]); that moves the far end of a long track by
216/// about 1e-12. CASA keys its epochs by `TIME·(DATA_DESC_ID+1)`, each
217/// standing for the earliest row with the key; a window's samples at one
218/// time towards one field are one epoch here. CASA takes `REST`
219/// frequencies, and those already in a frame other than `TOPO` that is the
220/// output's, as they are. Whether any frequency was folded.
221fn fold_over_epochs(
222    (catalogue, engine): (&Catalogue, &MsCalEngine),
223    (survey, window): (&Survey, &SourceSpectralWindow),
224    frequencies_hz: &[f64],
225    (output, selected_field): (FrequencyRef, Option<i32>),
226    directions: &mut BTreeMap<i32, MDirection>,
227    fold: &mut impl FnMut(f64),
228) -> Result<bool, PrepareError> {
229    let source = window.frequency_reference;
230    if source == FrequencyRef::REST || (source == output && source != FrequencyRef::TOPO) {
231        frequencies_hz.iter().copied().for_each(&mut *fold);
232        return Ok(true);
233    }
234    let mut converter = MFrequencyConverter::new(source, output);
235    let mut converted = false;
236    for epoch in survey
237        .window_epochs(window.spw_id)
238        .iter()
239        .filter(|epoch| selected_field.is_none_or(|field| epoch.field_id == field))
240    {
241        let direction = match directions.get(&epoch.field_id) {
242            Some(direction) => direction.clone(),
243            None => {
244                // Only a moving field without its own reference epoch
245                // (an MSv4's) reads its first selected time; finding it
246                // scans every window's epochs, so fixed fields skip it.
247                let field =
248                    usize::try_from(epoch.field_id).expect("a surveyed field is a FIELD row");
249                let first = if engine.field_moves(field) {
250                    survey
251                        .first_time_of(epoch.field_id)
252                        .unwrap_or(epoch.time_mjd_seconds)
253                } else {
254                    epoch.time_mjd_seconds
255                };
256                let direction = continuum_direction((catalogue, engine), epoch.field_id, first)?;
257                directions.insert(epoch.field_id, direction.clone());
258                direction
259            }
260        };
261        let frame =
262            engine.spectral_frame_observatory_direction(epoch.time_mjd_seconds, direction)?;
263        for &hz in frequencies_hz {
264            fold(converter.convert_hz(hz, &frame)?.hz());
265        }
266        converted = true;
267    }
268    Ok(converted)
269}

271/// Each selected window's factor from its frame to LSRK at the first
272/// visibility buffer CASA's tclean iterates of it, by window number.
273///
274/// `SynthesisImagerVi2::createVisSet` iterates by `ARRAY_ID`,
275/// `DATA_DESC_ID`, `FIELD_ID` and `TIME`, so a window's first buffer is
276/// its least selected field at that field's earliest selected time
277/// ([`Epoch`]; one array). `FTMachine::matchChannel` converts the buffer's
278/// frequencies to LSRK there, towards the field, whatever frame the image
279/// records, and `HetArrayConvFunc::findConvFunction` evaluates a mosaic's
280/// beams at them once for the window. A window in `REST`, or already in
281/// LSRK, keeps its frequencies (factor 1), as `matchChannel` keeps a rest
282/// frame's.
283pub(super) fn first_buffer_lsrk_factors(
284    (catalogue, engine): (&Catalogue, &MsCalEngine),
285    survey: &Survey,
286) -> Result<BTreeMap<usize, f64>, PrepareError> {
287    let mut factors = BTreeMap::new();
288    for window in &survey.spectral_windows {
289        let source = window.frequency_reference;
290        let first = survey
291            .window_epochs(window.spw_id)
292            .iter()
293            .copied()
294            .min_by(|a, b| {
295                a.field_id
296                    .cmp(&b.field_id)
297                    .then(a.time_mjd_seconds.total_cmp(&b.time_mjd_seconds))
298            });
299        let factor = match first {
300            Some(epoch) if !matches!(source, FrequencyRef::REST | FrequencyRef::LSRK) => {
301                let frame = epoch_frame((catalogue, engine), epoch)?;
302                PreparedFrequencyConversion::new(
303                    source,
304                    FrequencyRef::LSRK,
305                    Some(&frame),
306                    Some(&frame),
307                )?
308                .convert_hz(1.0)
309            }
310            _ => 1.0,
311        };
312        factors.insert(window.spw_id, factor);
313    }
314    Ok(factors)
315}

326/// CASA's nominal Briggs-cube density padding, per side
327/// (`BriggsCubeWeightor::estimateSwingChanPad`, half its `swingpad`).
328///
329/// CASA walks its visibility buffers in `DATA_DESC_ID`, then `FIELD_ID`
330/// order (one array; windows here in window order) and takes a window's
331/// range whenever the field changes from the buffer before, so with one
332/// field only the first window counts. For each such (window, field), the
333/// field's output interval is converted back to the window's frame at the
334/// epochs of its samples, then matched against the whole window with
335/// CASA's strict channel-edge overlap; a window it misses adds nothing.
336/// The output interval is widened by half its mean channel step,
337/// `|last − first| / max(nchan − 1, 1)`, which is zero for one channel.
338/// The padding is the largest swing of the matched ranges' ends plus the
339/// largest offset between the windows' lowest channels, in the image's
340/// channel increment rounded up, plus `max(min(4, nchan/10), 1)`, so it is
341/// at least one plane even when the frames agree. The selected channels'
342/// interpolation support does not redefine this nominal density domain.
343pub(super) fn cube_density_padding(
344    (catalogue, engine): (&Catalogue, &MsCalEngine),
345    survey: &Survey,
346    output: &OutputAxis,
347) -> Result<usize, PrepareError> {
348    for window in &survey.spectral_windows {
349        if window.frequencies_hz.is_empty()
350            || window.frequencies_hz.len() != window.channel_widths_hz.len()
351            || window
352                .frequencies_hz
353                .iter()
354                .any(|frequency| !frequency.is_finite() || *frequency <= 0.0)
355            || window
356                .channel_widths_hz
357                .iter()
358                .any(|width| !width.is_finite() || *width == 0.0)
359        {
360            return Err(PrepareError::SpectralWindowCoordinates {
361                spw_id: window.spw_id,
362            });
363        }
364    }
365    let extra_padding = (output.channels / 10).clamp(1, 4);
366    if survey.source_frequency_reference == output.frequency_reference {
367        return Ok(extra_padding);
368    }
369    let [first, last] = output.centres_hz;
370    let step = (last - first).abs() / output.channels.saturating_sub(1).max(1) as f64;
371    let increment = output.increment_hz.abs();
372    if !(step.is_finite() && increment.is_finite() && increment > 0.0) {
373        return Err(PrepareError::CubeIncrement {
374            increment_hz: output.increment_hz,
375        });
376    }
377    let interval = [first.min(last), first.max(last)];
378    let mut lower_centres = [f64::INFINITY, f64::NEG_INFINITY];
379    let mut upper_centres = [f64::INFINITY, f64::NEG_INFINITY];
380    let mut lowest_channels = [f64::INFINITY, f64::NEG_INFINITY];
381    let mut previous_field = None;
382    for window in &survey.spectral_windows {
383        let mut fields = survey
384            .window_epochs(window.spw_id)
385            .iter()
386            .map(|epoch| epoch.field_id)
387            .collect::<Vec<_>>();
388        fields.sort_unstable();
389        fields.dedup();
390        for field in fields {
391            if previous_field.replace(field) == Some(field) {
392                continue;
393            }
394            let Some([a, b]) = matched_range(
395                (catalogue, engine),
396                survey,
397                (window, field),
398                output.frequency_reference,
399                interval,
400                step,
401            )?
402            else {
403                continue;
404            };
405            lower_centres[0] = lower_centres[0].min(a.min(b));
406            lower_centres[1] = lower_centres[1].max(a.min(b));
407            upper_centres[0] = upper_centres[0].min(a.max(b));
408            upper_centres[1] = upper_centres[1].max(a.max(b));
409            let lowest = window
410                .frequencies_hz
411                .iter()
412                .copied()
413                .fold(f64::INFINITY, f64::min);
414            lowest_channels[0] = lowest_channels[0].min(lowest);
415            lowest_channels[1] = lowest_channels[1].max(lowest);
416        }
417    }
418    if !lower_centres[0].is_finite() {
419        return Ok(extra_padding);
420    }
421    let swing = (lower_centres[1] - lower_centres[0]).max(upper_centres[1] - upper_centres[0]);
422    let first_channel_shift = lowest_channels[1] - lowest_channels[0];
423    let swing_channels = ((swing + first_channel_shift) / increment).ceil();
424    if !(swing_channels.is_finite() && swing_channels < usize::MAX as f64) {
425        return Err(PrepareError::CubeIncrement {
426            increment_hz: increment,
427        });
428    }
429    Ok(swing_channels as usize + extra_padding)
430}

432/// The centres of the first and last channels of `window` that the output
433/// `interval` in frame `output_reference`, converted back to the window's
434/// frame at field `field`'s epochs in it and widened by half a `step`,
435/// overlaps (`MSUtil::getSpwInFreqRange`), or `None` when it overlaps
436/// none.
437fn matched_range(
438    (catalogue, engine): (&Catalogue, &MsCalEngine),
439    survey: &Survey,
440    (window, field): (&SourceSpectralWindow, i32),
441    output_reference: FrequencyRef,
442    interval: [f64; 2],
443    step: f64,
444) -> Result<Option<[f64; 2]>, PrepareError> {
445    let native_min = window
446        .frequencies_hz
447        .iter()
448        .copied()
449        .fold(f64::INFINITY, f64::min);
450    let native_max = window
451        .frequencies_hz
452        .iter()
453        .copied()
454        .fold(f64::NEG_INFINITY, f64::max);
455    let mut native = [f64::INFINITY, f64::NEG_INFINITY];
456    for epoch in survey
457        .window_epochs(window.spw_id)
458        .iter()
459        .filter(|epoch| epoch.field_id == field)
460    {
461        let frame = epoch_frame((catalogue, engine), *epoch)?;
462        let conversion = PreparedFrequencyConversion::new(
463            output_reference,
464            window.frequency_reference,
465            Some(&frame),
466            Some(&frame),
467        )?;
468        let a = conversion.convert_hz(interval[0]);
469        let b = conversion.convert_hz(interval[1]);
470        native[0] = native[0].min(a.min(b));
471        native[1] = native[1].max(a.max(b));
472    }
473    let lower = native[0] - 0.5 * step;
474    let upper = native[1] + 0.5 * step;
475    let partial =
476        (lower > native_min && lower < native_max) || (upper > native_min && upper < native_max);
477    let full = lower < native_min && upper > native_max;
478    if !partial && !full {
479        return Ok(None);
480    }
481    let mut matched = window
482        .frequencies_hz
483        .iter()
484        .zip(&window.channel_widths_hz)
485        .enumerate()
486        .filter(|(_, (centre, width))| {
487            full || (**centre + 0.5 * width.abs() > lower && **centre - 0.5 * width.abs() < upper)
488        })
489        .map(|(channel, _)| channel);
490    let Some(first) = matched.next() else {
491        return Ok(None);
492    };
493    let last = matched.next_back().unwrap_or(first);
494    Ok(Some([
495        window.frequencies_hz[first],
496        window.frequencies_hz[last],
497    ]))
498}

501mod tests {
502    use casa_types::measures::frequency::FrequencyRef;
503
504    use super::selected_edges;
505    use crate::compile::selection::SourceSpectralWindow;
506
507    #[test]
508    fn the_envelope_takes_the_exact_edges_of_nonuniform_selected_channels() {
509        let window = SourceSpectralWindow {
510            spw_id: 7,
511            frequency_reference: FrequencyRef::TOPO,
512            frequencies_hz: vec![100.0, 113.0, 151.0, 220.0],
513            channel_widths_hz: vec![8.0, -10.0, 14.0, 20.0],
514        };
515        let [lower, upper] = selected_edges(&window, &[1, 3]).expect("nonuniform selected edges");
516        assert_eq!(lower, 108.0);
517        assert_eq!(upper, 230.0);
518        assert_eq!(
519            (lower + upper) / 2.0,
520            169.0,
521            "the Taylor reference is the selected edge-envelope midpoint, not the mean channel centre"
522        );
523    }
