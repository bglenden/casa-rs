19pub(super) fn is_mosaic(record: &RecordValue) -> bool {
20    matches!(record_text(record, "name"), Ok("MosaicFT" | "MosaicFTNew"))

23pub(super) struct MosaicLayout {
24    direction: DirectionCoordinate,
25    centre: [f64; 2],
26    increment: [f64; 2],
27    antenna_classes: Vec<u8>,
28}

30impl MosaicLayout {
31    pub(super) fn load(
32        record: &RecordValue,
33        row: &FourierRow<'_>,
34        direction: &DirectionCoordinate,
35        geometry: &GridGeometry,
36        polarization: &PolarizationRouting,
37        reference_hz: f64,
38        convert_frequencies: bool,
39    ) -> Result<(Box<dyn ConvolutionFunctionSet>, Self), GainSolveError> {
40        let Some(Value::Record(beam)) = record.get("pbconvfunc") else {
41            return Err(model_error("virtual mosaic has no beam convolution record"));
42        };
43        if record_text(beam, "name")? != "HetArrayConvFunc" {
44            return Err(model_error(
45                "virtual mosaic requires an array beam convolution",
46            ));
47        }
48        let telescope = row
49            .ms
50            .observation()
51            .map_err(model_error)?
52            .string(row.row.observation_id as usize, "TELESCOPE_NAME")
53            .map_err(model_error)?;
54        let (window, complete_window) = beam_window(row, convert_frequencies)?;
55        let mut loaded = saved::planes(record, beam, &window)?;
56        if loaded.is_none()
57            && let Some(complete) = complete_window
58            && window
59                .shares_beam_frequencies(&complete)
60                .map_err(model_error)?
61        {
62            loaded = saved::planes(record, beam, &complete)?;
63        }
64        let class = saved::integer(beam, "pbclass")?;
65        let (kernels, antenna_classes) = if class == 8 {
66            let path = std::path::Path::new(record_text(beam, "vptable")?);
67            let beams = image_beam::ImageBeams::load(path, row)?;
68            let kernels = match restore(geometry, polarization, &window, loaded, beams.len())? {
69                Some(kernels) => kernels,
70                None => image_kernels(
71                    geometry,
72                    direction,
73                    polarization,
74                    reference_hz,
75                    window,
76                    &beams,
77                )?,
78            };
79            (kernels, beams.classes)
80        } else {
81            let (dishes, classes) = antenna_beams(row, geometry, &telescope, class, beam)?;
82            let kernels = match restore(geometry, polarization, &window, loaded, dishes.len())? {
83                Some(kernels) => kernels,
84                None => MosaicPb::new(
85                    MosaicRules::Tclean,
86                    geometry,
87                    polarization,
88                    reference_hz,
89                    &dishes,
90                    &[window],
91                )
92                .map_err(model_error)?,
93            };
94            (kernels, classes)
95        };
96        let grid = geometry.image();
97        Ok((
98            Box::new(kernels),
99            Self {
100                direction: direction.clone(),
101                centre: grid.reference_pixel.map(|pixel| pixel as f64),
102                increment: grid.increment_rad,
103                antenna_classes,
104            },
105        ))
106    }

108    pub(super) fn pointing_offset(
109        &self,
110        row: &FourierRow<'_>,
111        shift: [f64; 2],
112    ) -> Result<[f64; 2], GainSolveError> {
113        // HetArrayConvFunc does not save its inherited POINTING selector.
114        // Reopened MosaicFT calls get without initializing a new predictor,
115        // so its convolution phases use the field direction.
116        let pointing = row
117            .field
118            .convert_to(self.direction.conversion_direction_ref(), row.frame)
119            .map_err(model_error)?;
120        let (longitude, latitude) = pointing.as_angles();
121        let pixel = self
122            .direction
123            .world_to_pixel([longitude - shift[0], latitude - shift[1]])
124            .map_err(model_error)?;
125        Ok(std::array::from_fn(|axis| {
126            (pixel[axis] - self.centre[axis]) * self.increment[axis]
127        }))
128    }
129
130    pub(super) fn antenna_types(&self, row: &FourierRow<'_>) -> Result<[u8; 2], GainSolveError> {
131        let [Some(first), Some(second)] = [row.row.antenna1, row.row.antenna2]
132            .map(|antenna| self.antenna_classes.get(antenna as usize).copied())
133        else {
134            return Err(model_error(
135                "virtual mosaic refers to an unavailable antenna",
136            ));
137        };
138        Ok([first, second])
139    }
140}

142fn restore(
143    geometry: &GridGeometry,
144    polarization: &PolarizationRouting,
145    window: &MosaicWindow,
146    loaded: Option<SavedMosaicPlanes<'_>>,
147    dishes: usize,
148) -> Result<Option<MosaicPb>, GainSolveError> {
149    let Some(planes) = loaded else {
150        return Ok(None);
151    };
152    if planes.dishes != dishes {
153        return Err(model_error(
154            "virtual mosaic beam classes do not match the saved convolution",
155        ));
156    }
157    MosaicPb::from_saved_planes(geometry, polarization, window, planes).map_err(model_error)
158}

160fn image_kernels(
161    geometry: &GridGeometry,
162    direction: &DirectionCoordinate,
163    polarization: &PolarizationRouting,
164    reference_hz: f64,
165    window: MosaicWindow,
166    patterns: &image_beam::ImageBeams,
167) -> Result<MosaicPb, GainSolveError> {
168    let patterns = patterns.screens()?;
169    let support = patterns
170        .iter()
171        .map(|beam| beam.support_pixels(geometry.image().increment_rad[0], reference_hz))
172        .max()
173        .unwrap_or(0);
174    let pc = direction.pc_matrix();
175    MosaicPb::from_image_screens(
176        geometry,
177        polarization,
178        reference_hz,
179        patterns.len(),
180        support,
181        &[window],
182        |[first, second], side, increment, hz, voltage, power| {
183            let centre = (side / 2) as f64;
184            for (i, (voltage, power)) in voltage.iter_mut().zip(power).enumerate() {
185                let x = (i % side) as f64 - centre;
186                let y = (i / side) as f64 - centre;
187                let offset = std::array::from_fn(|axis| {
188                    increment[axis] * (pc[[axis, 0]] * x + pc[[axis, 1]] * y)
189                });
190                let a = patterns[first].value(offset, hz);
191                let b = patterns[second].value(offset, hz);
192                *voltage = a * b;
193                *power = Complex32::new(a.norm_sqr() * b.norm_sqr(), 0.);
194            }
195        },
196    )
197    .map_err(model_error)
198}

200fn beam_window(
201    row: &FourierRow<'_>,
202    convert_frequencies: bool,
203) -> Result<(MosaicWindow, Option<MosaicWindow>), GainSolveError> {
204    let spw = row.row.data_spw_id as usize;
205    let windows = row.ms.spectral_window().map_err(model_error)?;
206    let channel_width_hz = windows
207        .chan_width(spw)
208        .map_err(model_error)?
209        .into_iter()
210        .next()
211        .ok_or_else(|| model_error("virtual mosaic window has no channel width"))?;
212    let ratio = if convert_frequencies && row.frequency_ref != FrequencyRef::LSRK {
213        let frequency = row.frequencies[0];
214        convert_frequency_to_frame(
215            row.frequency_ref,
216            FrequencyRef::LSRK,
217            frequency,
218            row.row.time_seconds,
219            row.row.field_id as usize,
220            row.engine,
221        )
222        .map_err(model_error)?
223            / frequency
224    } else {
225        1.0
226    };
227    let full = MosaicWindow {
228        spectral_window: spw as u32,
229        window_frequencies_hz: row.frequencies.to_vec(),
230        channel_width_hz,
231        selected_frequencies_hz: row
232            .frequencies
233            .iter()
234            .map(|frequency| frequency * ratio)
235            .collect(),
236        buffer_band: None,
237    };
238    if let Some(selection) = &row.row.channel_selection {
239        let selected = selection
240            .indices(row.frequencies.len())
241            .map_err(model_error)?;
242        let mut window = full.clone();
243        window.selected_frequencies_hz = selected
244            .into_iter()
245            .map(|index| full.selected_frequencies_hz[index])
246            .collect();
247        Ok((window, Some(full)))
248    } else {
249        Ok((full, None))
250    }
251}

253fn antenna_beams(
254    row: &FourierRow<'_>,
255    geometry: &GridGeometry,
256    telescope: &str,
257    class: i32,
258    beam: &RecordValue,
259) -> Result<(Vec<PrimaryBeam>, Vec<u8>), GainSolveError> {
260    let antennas = row.ms.antenna().map_err(model_error)?;
261    if !matches!(class, 1 | 2) {
262        let path = std::path::Path::new(record_text(beam, "vptable")?);
263        let table = voltage_pattern::VoltagePatternTable::open(path).map_err(model_error)?;
264        let record = table
265            .default_record(telescope)
266            .map_err(model_error)?
267            .ok_or_else(|| model_error("no custom mosaic voltage pattern for the telescope"))?;
268        let pattern = voltage_pattern::radial_pattern(&record).map_err(model_error)?;
269        return Ok((vec![pattern], vec![0; antennas.row_count()]));
270    }
271    if class == 1 {
272        let reference = row
273            .ms
274            .spectral_window()
275            .map_err(model_error)?
276            .ref_frequency(0)
277            .map_err(model_error)?;
278        if let Some(beam) = PrimaryBeam::for_telescope(telescope, reference) {
279            return Ok((vec![beam], vec![0; antennas.row_count()]));
280        }
281    }
282    let alma = matches!(telescope, "ALMA" | "ACA");
283    let mut dishes = Vec::new();
284    let mut antenna_classes = Vec::with_capacity(antennas.row_count());
285    for antenna in 0..antennas.row_count() {
286        let diameter = antennas.dish_diameter(antenna).map_err(model_error)?;
287        let dish = if alma {
288            AiryDish::casa_alma(diameter, geometry)
289        } else {
290            AiryDish::casa_airy(diameter)
291        };
292        let beam = PrimaryBeam::from(dish);
293        let class = dishes
294            .iter()
295            .position(|known| *known == beam)
296            .unwrap_or_else(|| {
297                dishes.push(beam);
298                dishes.len() - 1
299            });
300        antenna_classes.push(u8::try_from(class).map_err(model_error)?);
301    }
302    Ok((dishes, antenna_classes))
303}
