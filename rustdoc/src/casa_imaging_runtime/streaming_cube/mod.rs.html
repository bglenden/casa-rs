14/// Charge enclosing process data in addition to the phase's explicit allocations.
15/// Subtract only live normal payload backed by authority-owned retention permits.
16fn enclosing_memory_bytes(retained_resident_bytes: u64) -> std::io::Result<u64> {
17    let observed = process_data_bytes()?;
18    let enclosing = unreserved_data_bytes(observed, retained_resident_bytes)?;
19    eprintln!(
20        "streaming_cube_enclosing_process_data_bytes={observed} retained_normal_payload_bytes={retained_resident_bytes} unreserved_bytes={enclosing}"
21    );
22    Ok(enclosing)
23}
24
25fn unreserved_data_bytes(observed: u64, retained: u64) -> std::io::Result<u64> {
26    if observed == 0 {
27        return Err(std::io::Error::other("process memory census unavailable"));
28    }
29    observed
30        .checked_sub(retained)
31        .ok_or_else(|| std::io::Error::other("retained normal payload exceeds process data census"))
32}

34#[cfg(target_os = "macos")]
35fn process_data_bytes() -> std::io::Result<u64> {
36    #[repr(C)]
37    #[derive(Default)]
38    struct Statistics {
39        blocks: u32,
40        in_use: usize,
41        maximum: usize,
42        allocated: usize,
43    }
44    unsafe extern "C" {
45        fn malloc_zone_statistics(zone: *mut std::ffi::c_void, stats: *mut Statistics);
46    }
47    let mut stats = Statistics::default();
48    // SAFETY: malloc_statistics_t has this C layout. A null zone requests the
49    // aggregate of all malloc zones; stats is a writable live output.
50    unsafe {
51        malloc_zone_statistics(std::ptr::null_mut(), &mut stats);
52    }
53    Ok(stats.in_use as u64)
54}

56#[cfg(target_os = "linux")]
57fn process_data_bytes() -> std::io::Result<u64> {
58    linux_process_data_bytes(&std::fs::read_to_string("/proc/self/status")?)
59}

61// VmData includes reserved data mappings, so this is a conservative charge,
62// not a live-malloc or RSS claim. It does not depend on the process allocator.
63// https://man7.org/linux/man-pages/man5/proc_pid_status.5.html
64#[cfg(any(target_os = "linux", test))]
65fn linux_process_data_bytes(status: &str) -> std::io::Result<u64> {
66    let mut fields = status
67        .lines()
68        .find_map(|line| line.strip_prefix("VmData:"))
69        .ok_or_else(|| std::io::Error::other("process status lacks VmData"))?
70        .split_whitespace();
71    let value = fields.next().and_then(|value| value.parse::<u64>().ok());
72    if fields.next() != Some("kB") || fields.next().is_some() {
73        return Err(std::io::Error::other("invalid process data census unit"));
74    }
75    value
76        .and_then(|value| value.checked_mul(1024))
77        .filter(|value| *value > 0)
78        .ok_or_else(|| std::io::Error::other("invalid process data census size"))
79}

81#[cfg(not(any(target_os = "macos", target_os = "linux")))]

85    ))

88#[cfg(test)]
89mod memory_tests {
90    use super::*;
91
92    #[test]
93    fn enclosing_data_subtracts_only_retained_payload() {
94        assert_eq!(unreserved_data_bytes(4096, 1024).unwrap(), 3072);
95        assert_eq!(unreserved_data_bytes(4096, 4096).unwrap(), 0);
96        assert!(unreserved_data_bytes(0, 0).is_err());
97        assert!(unreserved_data_bytes(4096, 4097).is_err());
98        assert!(enclosing_memory_bytes(0).unwrap() > 0);
99    }
100
101    #[test]
102    fn linux_data_census_has_explicit_units_and_fail_closed_bounds() {
103        assert_eq!(
104            linux_process_data_bytes("Name: test\nVmData:\t1234 kB\n").unwrap(),
105            1234 * 1024
106        );
107        for input in [
108            "",
109            "VmData: 0 kB",
110            "VmData: -1 kB",
111            "VmData: 1 MB",
112            "VmData: 1 kB extra",
113            "VmData: 18446744073709551615 kB",
114        ] {
115            assert!(linux_process_data_bytes(input).is_err(), "{input}");
116        }
117    }
