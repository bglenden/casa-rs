2//! The image task binaries on the shared task front end (#931).
3//!
4//! Each binary's `main` builds its front end here, passing the surfaces it
5//! hosts and a callback that reads the resolved run into its typed request.
6//! The front end owns the command line, `--json-run`, Last and logging.
7
8use casa_task_runtime::{TaskFailure, TaskFrontEnd, TaskInvocation, TaskSuccess};
9
10use crate::{ImageAnalysisTaskResult, ImageError, image_analysis_task_schema_bundle};
11
12/// The front end of the image binary `program`, which hosts `surfaces`.
13pub fn image_task_front_end<F>(program: &str, surfaces: &[&str], execute: F) -> TaskFrontEnd<F>
14where
15    F: Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>,
16{
17    TaskFrontEnd::new(
18        program,
19        image_analysis_task_schema_bundle(),
20        surfaces,
21        execute,
22    )
23    .expect("the image-analysis contract carries every image surface")
24}
25
26/// A product-writing image task's reply: its summary as the result, which a
27/// command line prints as JSON.
28///
29/// The return value is `None` until the CASA compatibility work gives each
30/// task CASA's own (#931 PR 4b).
31pub fn image_task_success(
32    result: Result<ImageAnalysisTaskResult, ImageError>,
33) -> Result<TaskSuccess, TaskFailure> {
34    let result = result.map_err(|error| TaskFailure::execution(error.to_string()))?;
35    let text = serde_json::to_string_pretty(&result)
36        .map_err(|error| TaskFailure::execution(format!("serialize the result: {error}")))?;
37    Ok(TaskSuccess::new(&(), &result)?.with_text(text + "\n"))
38}
39
40/// Read `name`, a CASA boolean held as text (`"true"`, `"false"`, `"T"`,
41/// `"F"`, any case).
42pub fn image_task_bool_text(task: &TaskInvocation<'_>, name: &str) -> Result<bool, TaskFailure> {
43    let text = task.required_text(name)?;
44    match text.to_ascii_lowercase().as_str() {
45        "true" | "t" => Ok(true),
46        "false" | "f" => Ok(false),
47        _ => Err(TaskFailure::validation(format!(
48            "parameter {name:?} must be true or false, found {text:?}"
49        ))
50        .at(name)),
51    }
52}
