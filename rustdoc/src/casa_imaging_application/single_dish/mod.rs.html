21/// Explicit geometry and native-channel gridding requested by `tsdimaging`.
22pub struct TotalPowerRequest {
23    /// Direction coordinate, including phase centre, signed cells and reference pixels.
24    pub direction: DirectionCoordinate,
25    /// Image width and height.
26    pub size: [usize; 2],
27    /// Native input channels, without spectral regridding.
28    pub channels: Range<usize>,
29    /// Exact observing intent; an empty string selects all intents.
30    pub intent: String,
31    /// CASA's gridding function and resolved pixel parameters.
32    pub kernel: GridFunction,
33    /// Rest frequency override, Hz; otherwise SOURCE, then mean SPW frequency.
34    pub rest_frequency_hz: Option<f64>,
35    /// Selected spectral windows; empty selects all compatible windows.
36    pub windows: Vec<casa_observation::WindowId>,
37    /// Selected autocorrelation antennas; empty selects all antennas.
38    pub antennas: Vec<casa_observation::AntennaId>,
39}

41/// The admitted scientific arrays and their coordinates, before publication.

52/// Why total-power imaging could not be prepared or run.
53#[derive(Debug, thiserror::Error)]
54pub enum TotalPowerError {
55    /// Invalid setup, failed admission or observation/image I/O.
56    #[error("tsdimaging: {0}")]
57    Execution(String),
58    /// A data-dependent mode outside the task's current core.
59    #[error("tsdimaging: {behavior} is not implemented; see #{issue}")]
60    Unsupported {
61        /// Requested observable behavior.
62        behavior: &'static str,
63        /// Work item owning the remaining implementation.
64        issue: u32,
65    },
66}
67
68fn failure(error: impl std::fmt::Display) -> TotalPowerError {
69    TotalPowerError::Execution(error.to_string())
70}

72/// Open the input with bounded admission and execute resolved CASA parameters.
73///
74/// # Errors
75/// Invalid parameters, observation opening, admission, gridding or publication.
76pub fn image_from_parameters(
77    values: &serde_json::Map<String, serde_json::Value>,
78    context: &RunContext,
79) -> Result<PublishedTotalPower, TotalPowerError> {
80    let parameters = TotalPowerParameters(values);
81    charge_process_caches(&context.host, &context.policy).map_err(failure)?;
82    let _metadata = admit(
83        &context.host,
84        &context.policy,
85        &Demand {
86            phase: "total-power task parameters",
87            memory: 16 << 10,
88        },
89    )
90    .map_err(failure)?;
91    let opened = crate::compile::opening::open_observation(
92        &parameters.input()?,
93        Default::default(),
94        (&context.host, &context.policy),
95    )
96    .map_err(failure)?;
97    let observation = opened.value.observation();
98    let _selection = admit(
99        &context.host,
100        &context.policy,
101        &Demand {
102            phase: "total-power parameter selection",
103            memory: (8 * observation.catalogue().partitions.len()) as u64,
104        },
105    )
106    .map_err(failure)?;
107    let (request, products) = parameters.request(observation.catalogue())?;
108    image_observation(observation, &request, &products, context)
109}

111/// Merge sparse `tsdimaging` overrides over [`default_parameters`] and execute
112/// total-power imaging.
113///
114/// # Errors
115///
116/// Invalid parameters, observation opening, admission, gridding or publication.
117pub fn image_from_sparse_parameters(
118    overrides: &serde_json::Map<String, serde_json::Value>,
119    context: &RunContext,
120) -> Result<PublishedTotalPower, TotalPowerError> {
121    charge_process_caches(&context.host, &context.policy).map_err(failure)?;
122    let _sparse = admit(
123        &context.host,
124        &context.policy,
125        &Demand {
126            phase: "total-power sparse parameter defaults",
127            memory: 16 << 10,
128        },
129    )
130    .map_err(failure)?;
131    let mut merged = default_parameters();
132    for (key, value) in overrides {
133        merged.insert(key.clone(), value.clone());
134    }
135    image_from_parameters(&merged, context)
136}

138/// Grid an observation's selected autocorrelations in the image plane.
139///
140/// The numerical core reads MSv2 and MSv4 through `casa-observation`.
141/// Inputs stay read-only; publication is a separate final operation.
142///
143/// # Errors
144/// Missing intent/data, incompatible spectral axes, unsupported correlation or
145/// pointing cases, invalid geometry, resource admission or observation I/O.
146pub fn grid_observation(
147    observation: &dyn Observation,
148    request: &TotalPowerRequest,
149    context: &RunContext,
150) -> Result<TotalPowerOutcome, TotalPowerError> {
151    if context.cancel.is_cancelled() {
152        return Err(failure("cancelled before gridding"));
153    }
154    charge_process_caches(&context.host, &context.policy).map_err(failure)?;
155    let mut metadata = admit(
156        &context.host,
157        &context.policy,
158        &Demand {
159            phase: "total-power planning and coordinates",
160            memory: (16 << 10) + (8 * observation.catalogue().partitions.len()) as u64,
161        },
162    )
163    .map_err(failure)?;
164    let plan = plan::Plan::new(observation, request)?;
165    let _measures_memory = admit(
166        &context.host,
167        &context.policy,
168        &Demand {
169            phase: "total-power measures",
170            memory: casa_ms::MEASURES_RUNTIME_BYTES as u64,
171        },
172    )
173    .map_err(failure)?;
174    let measures = casa_ms::open_measures_runtime().map_err(failure)?;
175    let (mut grid, _kernel_memory) = plan.grid(observation, request, context)?;
176    let (sampling, sampling_memory, date) = stream::grid(
177        observation,
178        request,
179        context,
180        &plan.partitions,
181        &measures,
182        &mut grid,
183    )?;
184    let mut arrays = grid.finish();
185    arrays.reservation.join(sampling_memory);
186    metadata.retain(16 << 10);
187    arrays.reservation.join(metadata);
188    let mut coordinates = plan.coordinates;
189    let catalogue = observation.catalogue();
190    let source = catalogue.observation(catalogue.partition(plan.partitions[0]).observation);
191    let information = coordinates.obs_info_mut();
192    information.telescope = source.telescope.clone();
193    information.observer = source.observers.first().cloned().unwrap_or_default();
194    information.date = date;
195    Ok(TotalPowerOutcome {
196        arrays,
197        coordinates,
198        sampling,
199        source_partition: plan.partitions[0],
200    })
