16/// Model grids ready for degridding: the forward FFT of the corrected,
17/// polarization-expanded model, one full-grid block per
18/// `(plane, grid pol, Taylor term)` in the operator's precision.
19///
20/// The grids are immutable. A device backend keeps its copy of them in the
21/// grids themselves ([`PreparedModelGrids::device_copy`]), made once and
22/// dropped with them.
23pub struct PreparedModelGrids {
24    layout: AccumulatorLayout,
25    storage: GridStorage,
26    device: OnceLock<Box<dyn Any + Send + Sync>>,
27}
28
29impl PreparedModelGrids {
30    pub(crate) fn new(layout: AccumulatorLayout, storage: GridStorage) -> Self {
31        debug_assert_eq!(storage.len(), layout.cells());
32        debug_assert!(layout.is_full_grid());
33        Self {
34            layout,
35            storage,
36            device: OnceLock::new(),
37        }
38    }
39
40    /// A device backend's copy of the grids, made by `copy` from the cells
41    /// the first time any caller asks; later calls, from any thread, return
42    /// the same copy. A failed copy is kept like a successful one, so a
43    /// backend stores a `Result` when copying can fail.
44    ///
45    /// Panics when an earlier caller stored a copy of another type: one
46    /// backend type serves a model.
47    pub fn device_copy<C: Any + Send + Sync>(&self, copy: impl FnOnce(&GridStorage) -> C) -> &C {
48        self.device
49            .get_or_init(|| Box::new(copy(&self.storage)))
50            .downcast_ref()
51            .expect("one device backend type serves a model")
52    }
53
54    /// Layout of the grids (data terms only, full grid).
55    #[must_use]
56    pub const fn layout(&self) -> &AccumulatorLayout {
57        &self.layout
58    }
59
60    /// Cell precision.
61    #[must_use]
62    pub const fn precision(&self) -> GridPrecision {
63        self.storage.precision()
64    }
65
66    /// The cells.
67    #[must_use]
68    pub const fn storage(&self) -> &GridStorage {
69        &self.storage
70    }
71
72    /// Cells of block `(plane_local, pol, term)` at precision `T`.
73    #[must_use]
74    pub fn block<T: GridScalar>(
75        &self,
76        plane_local: usize,
77        pol: usize,
78        term: usize,
79    ) -> &[Complex<T>] {
80        let offset = self.layout.block_offset(plane_local, pol, term);
81        &T::cells(&self.storage)[offset..offset + self.layout.block_cells()]
82    }
83}
84
85impl Clone for PreparedModelGrids {
86    /// The clone holds no device copy.
87    fn clone(&self) -> Self {
88        Self::new(self.layout.clone(), self.storage.clone())
89    }
90}
91
92impl PartialEq for PreparedModelGrids {
93    fn eq(&self, other: &Self) -> bool {
94        self.layout == other.layout && self.storage == other.storage
95    }
96}
97
98impl std::fmt::Debug for PreparedModelGrids {
99    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
100        formatter
101            .debug_struct("PreparedModelGrids")
102            .field("layout", &self.layout)
103            .field("storage", &self.storage)
104            .finish_non_exhaustive()
105    }
106}

108/// What one dispatch does with a sample block.
109pub enum Work<'a> {
110    /// Accumulate the block into `acc` in `mode`.
111    Grid {
112        /// Data, PSF or weight terms.
113        mode: Mode,
114        /// Target accumulator; it must hold `mode`.
115        acc: &'a mut GridAccumulator,
116    },
117    /// Predict raw visibilities (`npol` per placement, sample-major,
118    /// phase-centre phasor applied) from `model`.
119    Predict {
120        /// Prepared model grids covering every plane in the block.
121        model: &'a PreparedModelGrids,
122        /// `npol × placements` predicted visibilities.
123        out: &'a mut [Complex32],
124    },
125    /// Predict, subtract from the block and grid the residual data terms in
126    /// one dispatch; optionally hand back the raw residual samples for a
127    /// final pass that writes the corrected or model column.
128    ResidualGrid {
129        /// Prepared model grids covering every plane in the block.
130        model: &'a PreparedModelGrids,
131        /// Target accumulator; it must hold the data terms.
132        acc: &'a mut GridAccumulator,
133        /// `npol × placements` raw residual visibilities, when wanted.
134        residual_out: Option<&'a mut [Complex32]>,
135    },
136}

138/// A gridding backend: CPU or device. One implementation of the kernel
139/// contract in plan section 5.3, selected by the runtime.
140pub trait GridBackend: Send {
141    /// Apply `work` to one block with kernel set `cf`; the work is complete
142    /// when this returns.
143    ///
144    /// Every placement must lie on a plane the target holds and have its
145    /// whole support inside the target's tile; the block's polarization
146    /// count must equal the kernel set's visibility polarizations.
147    fn apply(
148        &mut self,
149        block: &SampleBlock<'_>,
150        cf: &dyn ConvolutionFunctionSet,
151        work: Work<'_>,
152    ) -> Result<(), OperatorError>;
153}
