4    BufferedObservationBlock, SelectedObservationBufferCapacity, SelectedObservationContentBudget,

7
8/// Payload-free memory requirements for one selected source.
9///
10/// The storage owner derives this curve under a short-lived read lock. It includes
11/// retained metadata, source-specific plans, POINTING construction, and overlapping
12/// block preparation. It retains neither table handles nor a prepared catalog.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub struct SelectedObservationContentRequirements {
15    pub(super) retained_bytes: usize,
16    pub(super) initialization_scratch_bytes: usize,
17    pub(super) initialization_scan_bytes_per_row: usize,
18    pub(super) traversal_base_bytes: usize,
19    /// One row of a block's buffer, which every phase holds whole.
20    pub(super) buffer_row: SelectedObservationBufferCapacity,
21    /// What [`Self::buffer_row`] retains, within each per-row charge below.
22    pub(super) buffer_bytes_per_row: usize,
23    pub(super) resident_bytes_per_row: usize,
24    pub(super) fill_bytes_per_row: usize,
25    pub(super) preparation_bytes_per_row: usize,
26    pub(super) fill_fixed_bytes: usize,
27    pub(super) selected_rows: usize,
28    pub(super) maximum_pointing_polynomial_terms: usize,
29}
30
31impl SelectedObservationContentRequirements {
32    /// Return the smallest complete envelope admitting one selected row.
33    pub fn minimum_bytes(self) -> Result<usize, SelectedObservationContentPlanError> {
34        self.bytes_for_rows(1)
35    }
36
37    /// Return the full initialization/traversal envelope for a block of
38    /// `rows` rows. The read path holds one block at a time.
39    ///
40    /// Rows beyond the selected source's row count do not enlarge the block.
41    pub fn bytes_for_rows(self, rows: usize) -> Result<usize, SelectedObservationContentPlanError> {
42        if rows == 0 || self.selected_rows == 0 {
43            return Err(SelectedObservationContentPlanError::InvalidBudget);
44        }
45        let rows = rows.min(self.selected_rows);
46        let replay = super::BoundObservationSource::row_replay_bytes_per_row();
47        let fill_per_row = self
48            .fill_bytes_per_row
49            .checked_add(replay)
50            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
51        let prepare_per_row = self
52            .preparation_bytes_per_row
53            .checked_add(replay)
54            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
55        let base = self.traversal_base()?;
56        let fill = rows
57            .checked_mul(fill_per_row)
58            .and_then(|bytes| bytes.checked_add(self.fill_fixed_bytes))
59            .and_then(|bytes| bytes.checked_add(base))
60            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
61        let prepare = rows
62            .checked_mul(prepare_per_row)
63            .and_then(|bytes| bytes.checked_add(base))
64            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
65        let initialization = rows
66            .checked_mul(self.initialization_scan_bytes_per_row)
67            .and_then(|bytes| bytes.checked_add(self.initialization_scratch_bytes))
68            .and_then(|bytes| bytes.checked_add(self.retained_bytes))
69            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
70        Ok(initialization.max(fill).max(prepare))
71    }
72
73    fn traversal_base(self) -> Result<usize, SelectedObservationContentPlanError> {
74        size_of::<BufferedObservationBlock>()
75            .checked_add(self.traversal_base_bytes)
76            .ok_or(SelectedObservationContentPlanError::ByteOverflow)
77    }
78
79    /// Return the POINTING polynomial bound used to derive this curve.
80    #[must_use]
81    pub const fn maximum_pointing_polynomial_terms(self) -> usize {
82        self.maximum_pointing_polynomial_terms
83    }
84
85    /// Select the largest bounded block that fits an explicit physical budget.
86    pub fn plan(
87        self,
88        budget: SelectedObservationContentBudget,
89    ) -> Result<SelectedObservationContentPlan, SelectedObservationContentPlanError> {
90        if budget.available_bytes() == 0
91            || budget.maximum_live_blocks() == 0
92            || budget.maximum_pointing_polynomial_terms() != self.maximum_pointing_polynomial_terms
93            || self.maximum_pointing_polynomial_terms == 0
94        {
95            return Err(SelectedObservationContentPlanError::InvalidBudget);
96        }
97        let fixed = self
98            .retained_bytes
99            .checked_add(self.initialization_scratch_bytes)
100            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?
101            .max(self.traversal_base()?);
102        if fixed > budget.available_bytes() {
103            return Err(
104                SelectedObservationContentPlanError::InsufficientRetainedBudget {
105                    required_bytes: fixed,
106                    available_bytes: budget.available_bytes(),
107                },
108            );
109        }
110        let minimum = self.minimum_bytes()?;
111        if minimum > budget.available_bytes() {
112            return Err(SelectedObservationContentPlanError::InsufficientBudget {
113                required_bytes: minimum,
114                available_bytes: budget.available_bytes(),
115            });
116        }
117        let mut low = 1;
118        let mut high = self.selected_rows;
119        while low < high {
120            let middle = low + (high - low).div_ceil(2);
121            if self.bytes_for_rows(middle)? <= budget.available_bytes() {
122                low = middle;
123            } else {
124                high = middle - 1;
125            }
126        }
127        let rows = low;
128        Ok(SelectedObservationContentPlan {
129            retained_bytes: self.retained_bytes,
130            initialization_scratch_bytes: self
131                .initialization_scan_bytes_per_row
132                .checked_mul(rows)
133                .and_then(|bytes| bytes.checked_add(self.initialization_scratch_bytes))
134                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
135            resident_bytes_per_row: self.resident_bytes_per_row,
136            preparation_bytes_per_row: self.preparation_bytes_per_row,
137            rows_per_block: rows,
138            buffer_capacity: self.buffer_row.with_rows(rows),
139            buffer_bytes_per_block: rows
140                .checked_mul(self.buffer_bytes_per_row)
141                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
142            resident_bytes_per_block: rows
143                .checked_mul(self.resident_bytes_per_row)
144                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
145            preparation_bytes_per_block: rows
146                .checked_mul(self.preparation_bytes_per_row)
147                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
148            maximum_resident_bytes: self.bytes_for_rows(rows)?,
149            maximum_pointing_polynomial_terms: self.maximum_pointing_polynomial_terms,
150        })
151    }
152}
