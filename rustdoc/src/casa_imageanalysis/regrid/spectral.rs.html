2//! The spectral pass of a regrid: casacore's `ImageRegrid::regrid1D` with
3//! `Interpolate2D::LINEAR`, as `SIImageStore::regridToModelImage` regrids a
4//! start model's spectral axis.
5//!
6//! Each output channel is converted to world frequency on the template and
7//! back to a pixel on the input (`make1DCoordinateGrid`), and every line of
8//! the image along the spectral axis is interpolated there by casacore's
9//! `InterpolateArray1D::linear` with flags, good being true and without
10//! extrapolation, in single precision as casacore runs it for a `Float`
11//! image:
12//! - between two input channels, their linear interpolation, good where
13//!   both are;
14//! - within single-precision rounding of an input channel, that channel;
15//! - beyond either end, the end channel's value, masked.
16//!
17//! The pixel values beyond the ends are therefore the end channels' rather
18//! than zero, as casacore writes them into an image without a mask (a
19//! tclean model): only the mask says they were not sampled.
20

27/// How the spectral axis is regridded: its pixel axis and the input pixel
28/// at each output channel, `None` where the conversion fails.
29pub(super) struct SpectralPlan {
30    axis: usize,
31    positions: Vec<Option<f64>>,
32    method: RegridMethod,
33}
34
35impl SpectralPlan {
36    /// The plan for spectral coordinates `input` of `input_channels` and
37    /// `output` of `output_channels` on pixel axis `axis`.
38    ///
39    /// # Errors
40    ///
41    /// The reason when the two are not both spectral, are in different
42    /// frames (casa-rs makes no frequency conversion machine), the input
43    /// holds one channel, which casacore's linear interpolation reads past,
44    /// or the output does and the input more: `ImageRegrid::_checkAxes`
45    /// removes an axis of unit length from those it regrids and then refuses
46    /// the shapes.
47    pub(super) fn new(
48        (input, input_channels): (&CoordinateModel, usize),
49        (output, output_channels): (&CoordinateModel, usize),
50        axis: usize,
51    ) -> Result<Self, String> {
52        let (CoordinateModel::Spectral(ours), CoordinateModel::Spectral(theirs)) = (input, output)
53        else {
54            return Err(format!(
55                "the {} axes differ between the input and the template; casa-rs regrids the \
56                 direction and spectral axes only",
57                input.coordinate_type()
58            ));
59        };
60        if ours.world_frequency_ref() != theirs.world_frequency_ref() {
61            return Err(format!(
62                "the input's frequencies are {:?} and the template's {:?}; casa-rs does not \
63                 convert between spectral frames",
64                ours.world_frequency_ref(),
65                theirs.world_frequency_ref()
66            ));
67        }
68        if input_channels < 2 {
69            return Err(format!(
70                "the input has {input_channels} channel and the template {output_channels}; a \
71                 spectral regrid interpolates between two channels"
72            ));
73        }
74        if output_channels == 1 {
75            return Err(format!(
76                "any axis not being regridded must have the same input and output shapes: the \
77                 template's spectral axis has one channel and the input's {input_channels}"
78            ));
79        }
80        debug_assert_eq!(input.coordinate_type(), CoordinateType::Spectral);
81        let positions = (0..output_channels)
82            .map(|channel| {
83                theirs
84                    .world_frequency_hz(channel as f64)
85                    .and_then(|frequency| ours.pixel_at_frequency_hz(frequency))
86                    .ok()
87                    .filter(|pixel| pixel.is_finite())
88            })
89            .collect();
90        Ok(Self::from_positions(axis, positions, RegridMethod::Linear))
91    }
92
93    pub(super) fn from_positions(
94        axis: usize,
95        positions: Vec<Option<f64>>,
96        method: RegridMethod,
97    ) -> Self {
98        Self {
99            axis,
100            positions,
101            method,
102        }
103    }

105    /// `data` and its `mask` (true where good) regridded along the spectral
106    /// axis onto `shape`, which differs from theirs on that axis alone.
107    pub(super) fn apply<T>(
108        &self,
109        data: &ArrayD<T>,
110        mask: &ArrayD<bool>,
111        shape: &[usize],
112    ) -> (ArrayD<T>, ArrayD<bool>)
113    where
114        T: ImagePixel + Into<f64> + From<f32> + Copy,
115    {
116        self.apply_with(
117            data,
118            mask,
119            shape,
120            |value| Into::<f64>::into(value) as f32,
121            T::from,
122        )
123    }
124
125    pub(super) fn apply_precise<T: RegridPixel>(
126        &self,
127        data: &ArrayD<T>,
128        mask: &ArrayD<bool>,
129        shape: &[usize],
130    ) -> (ArrayD<T>, ArrayD<bool>) {
131        self.apply_with(data, mask, shape, |value| value, |value| value)
132    }
133
134    fn apply_with<T: ImagePixel + Copy, V: RegridPixel>(
135        &self,
136        data: &ArrayD<T>,
137        mask: &ArrayD<bool>,
138        shape: &[usize],
139        read: impl Fn(T) -> V,
140        write: impl Fn(V) -> T,
141    ) -> (ArrayD<T>, ArrayD<bool>) {
142        let mut out_data = ArrayD::<T>::from_elem(IxDyn(shape).f(), T::default_value());
143        let mut out_mask = ArrayD::<bool>::from_elem(IxDyn(shape).f(), false);
144        let axis = Axis(self.axis);
145        Zip::from(out_data.lanes_mut(axis))
146            .and(out_mask.lanes_mut(axis))
147            .and(data.lanes(axis))
148            .and(mask.lanes(axis))
149            .for_each(|mut out, mut good, line, line_mask| {
150                let at = |k| (read(line[k]), line_mask[k]);
151                let derivatives = (self.method == RegridMethod::Cubic && line.len() > 1)
152                    .then(|| second_derivatives(|k| at(k).0, line.len()));
153                for (channel, position) in self.positions.iter().enumerate() {
154                    let Some(position) = position.and_then(cast::<V::Real>) else {
155                        continue;
156                    };
157                    let (value, valid) = match (self.method, &derivatives) {
158                        _ if line.len() == 1 => at(0),
159                        (RegridMethod::Nearest, _) => nearest(position, at, line.len()),
160                        (RegridMethod::Cubic, Some(derivatives)) => {
161                            spline(position, at, derivatives)
162                        }
163                        _ => linear(position, at, line.len()),
164                    };
165                    out[channel] = write(value);
166                    good[channel] = valid;
167                }
168            });
169        (out_data, out_mask)
170    }
171}

173fn nearest<F: NdFloat, V>(x: F, at: impl Fn(usize) -> (V, bool), count: usize) -> (V, bool) {
174    let last = number::<F>((count - 1) as f64);
175    let channel = index((x - number::<F>(0.5)).ceil(), count - 1);
176    let (value, valid) = at(channel);
177    (value, valid && (F::zero()..=last).contains(&x))
178}
179
180fn second_derivatives<F: NdFloat, V: RegridPixel<Real = F>>(
181    at: impl Fn(usize) -> V,
182    count: usize,
183) -> Vec<V> {
184    let mut coefficients = vec![F::zero(); count];
185    let mut result = vec![V::default_value(); count];
186    for i in 1..count - 1 {
187        let diagonal = number::<F>(4.0) - coefficients[i - 1];
188        let difference = (at(i + 1) - at(i)) - (at(i) - at(i - 1));
189        coefficients[i] = F::one() / diagonal;
190        result[i] = (difference * number::<F>(6.0) - result[i - 1]) / diagonal;
191    }
192    // InterpolateArray1D back-substitutes down through index 2.
193    for i in (2..count - 1).rev() {
194        result[i] = result[i] - result[i + 1] * coefficients[i];
195    }
196    result
197}

199fn spline<F: NdFloat, V: RegridPixel<Real = F>>(
200    x: F,
201    at: impl Fn(usize) -> (V, bool),
202    derivatives: &[V],
203) -> (V, bool) {
204    let last = derivatives.len() - 1;
205    let right = index(x.ceil(), last).max(1);
206    let left = right - 1;
207    let a = number::<F>((right) as f64) - x;
208    let b = F::one() - a;
209    let (y1, good1) = at(left);
210    let (y2, good2) = at(right);
211    let value = y1 * a
212        + y2 * b
213        + derivatives[left] * (a * a * a - a) / number::<F>(6.0)
214        + derivatives[right] * (b * b * b - b) / number::<F>(6.0);
215    (
216        value,
217        good1 && good2 && (F::zero()..=number::<F>((last) as f64)).contains(&x),
218    )
219}

221/// casacore `InterpolateArray1D::interpolatePtr`'s `linear` case with flags,
222/// goodIsTrue and no extrapolation, over `0..count` with at least two pixels.
223fn linear<F: NdFloat, V: RegridPixel<Real = F>>(
224    x: F,
225    at: impl Fn(usize) -> (V, bool),
226    count: usize,
227) -> (V, bool) {
228    let bracket = index(x.ceil(), count);
229    let (right, discard) = match bracket {
230        0 => (1, x != F::zero()),
231        n if n == count => (count - 1, true),
232        n => (n, false),
233    };
234    let left = right - 1;
235    let ((y1, good1), (y2, good2)) = (at(left), at(right));
236    let frac = x - number::<F>((left) as f64);
237    let limit = F::epsilon();
238    if frac > limit && frac < F::one() - limit {
239        (y1 + (y2 - y1) * frac, !discard && good1 && good2)
240    } else if frac <= limit {
241        (y1, !discard && good1)
242    } else {
243        (y2, !discard && good2)
244    }
245}

247fn number<F: NdFloat>(value: f64) -> F {
248    cast(value).expect("finite index or constant")
249}
250
251fn cast<F: NdFloat>(value: f64) -> Option<F> {
252    F::from(value)
253}
254
255fn index<F: NdFloat>(x: F, max: usize) -> usize {
256    x.max(F::zero())
257        .min(number::<F>((max) as f64))
258        .to_usize()
259        .unwrap_or(max)
260}

289    /// A 4 × 4 image of one Stokes plane and `channels` channels from
290    /// `first_hz`, `step_hz` apart, each plane holding `value(channel)`.
291    fn cube(
292        first_hz: f64,
293        step_hz: f64,
294        channels: usize,
295        frame: casa_core::measures::frequency::FrequencyRef,
296        value: impl Fn(usize) -> f32,
297    ) -> casa_images::TempImage<f32> {
298        use casa_coordinates::{
299            CoordinateSystem, DirectionCoordinate, Projection, ProjectionType, SpectralCoordinate,
300            StokesCoordinate, StokesType,
301        };
302        use casa_core::measures::direction::DirectionRef;
303        let mut coordinates = CoordinateSystem::new();
304        coordinates.add_coordinate(DirectionCoordinate::new(
305            DirectionRef::J2000,
306            Projection::new(ProjectionType::SIN),
307            [1.0, 0.3],
308            [-1e-5, 1e-5],
309            [2.0, 2.0],
310        ));
311        coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
312        coordinates.add_coordinate(SpectralCoordinate::new(
313            frame, first_hz, step_hz, 0.0, 1.0e9,
314        ));
315        let shape = vec![4, 4, 1, channels];
316        let mut image = casa_images::TempImage::<f32>::new(
317            shape.clone(),
318            coordinates,
319            casa_lattices::TempStoragePolicy::Memory,
320        )
321        .unwrap();
322        image.set_units("Jy/pixel").unwrap();
323        let data = ArrayD::from_shape_fn(IxDyn(&shape).f(), |index| value(index[3]));
324        image.put_slice(&data, &[0, 0, 0, 0]).unwrap();
325        image
326    }

328    /// `SIImageStore::regridToModelImage`'s spectral regrid: an output
329    /// channel between two input channels takes their line, and one beyond
330    /// the input's ends takes the end channel's value, masked.
331    #[test]
332    fn a_cube_is_regridded_along_frequency_as_casacore_regrids_it() {
333        use casa_core::measures::frequency::FrequencyRef;
334        let input = cube(1.0e9, 1.0e8, 4, FrequencyRef::LSRK, |k| {
335            10.0 * (k as f32 + 1.0)
336        });
337        let template = RegridTemplate::of(&cube(0.95e9, 1.0e8, 6, FrequencyRef::LSRK, |_| 0.0));
338        let regridded = regrid_along_frequency(&input, &template, RegridMethod::Linear).unwrap();
339        assert_eq!(regridded.data.shape(), &[4, 4, 1, 6]);
340        let planes = (0..6)
341            .map(|k| (regridded.data[[1, 2, 0, k]], regridded.mask[[1, 2, 0, k]]))
342            .collect::<Vec<_>>();
343        assert_eq!(
344            planes,
345            [
346                (10.0, false),
347                (15.0, true),
348                (25.0, true),
349                (35.0, true),
350                (40.0, false),
351                (40.0, false)
352            ]
353        );
354        // Another frame, or one input channel, is refused.
355        let topo = RegridTemplate::of(&cube(0.95e9, 1.0e8, 6, FrequencyRef::TOPO, |_| 0.0));
356        let error = regrid_along_frequency(&input, &topo, RegridMethod::Linear).unwrap_err();
357        assert!(error.to_string().contains("spectral frames"), "{error}");
358        let one = cube(1.0e9, 1.0e8, 1, FrequencyRef::LSRK, |_| 1.0);
359        let error = regrid_along_frequency(&one, &template, RegridMethod::Linear).unwrap_err();
360        assert!(
361            error.to_string().contains("one channel") || error.to_string().contains("1 channel"),
362            "{error}"
363        );
364        // A cube onto one channel: `_checkAxes` removes the axis of unit
365        // length from those it regrids and refuses the shapes.
366        let plane = RegridTemplate::of(&cube(1.0e9, 1.0e8, 1, FrequencyRef::LSRK, |_| 0.0));
367        let error = regrid_along_frequency(&input, &plane, RegridMethod::Linear).unwrap_err();
368        assert!(
369            error.to_string().contains("same input and output shapes"),
370            "{error}"
371        );
372    }

385    /// Between channels the line, at a channel its value, beyond the ends
386    /// the end value masked: casacore's flagged linear interpolation.
387    #[test]
388    fn the_line_follows_casacores_flagged_linear_interpolation() {
389        let values = [1.0_f32, 3.0, 4.0, 8.0];
390        let goods = [true, true, false, true];
391        let at = |k: usize| (values[k], goods[k]);
392        assert_eq!(linear(0.5, at, 4), (2.0, true));
393        assert_eq!(linear(1.0, at, 4), (3.0, true));
394        assert_eq!(linear(0.0, at, 4), (1.0, true));
395        assert_eq!(linear(1.5, at, 4), (3.5, false));
396        assert_eq!(linear(3.0, at, 4), (8.0, true));
397        assert_eq!(linear(-0.5, at, 4), (1.0, false));
398        assert_eq!(linear(4.5, at, 4), (8.0, false));
399    }
