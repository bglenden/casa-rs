5/// Edges in the direction conversion routing graph.
6pub(super) const ROUTING_EDGES: &[(DirectionRef, DirectionRef)] = &[
7    (DirectionRef::J2000, DirectionRef::GALACTIC),
8    (DirectionRef::J2000, DirectionRef::ICRS),
9    (DirectionRef::GALACTIC, DirectionRef::SUPERGAL),
10    (DirectionRef::J2000, DirectionRef::ECLIPTIC),
11    (DirectionRef::J2000, DirectionRef::JMEAN),
12    (DirectionRef::JMEAN, DirectionRef::JTRUE),
13    (DirectionRef::JMEAN, DirectionRef::MECLIPTIC),
14    (DirectionRef::JTRUE, DirectionRef::TECLIPTIC),
15    (DirectionRef::JNAT, DirectionRef::J2000),
16    (DirectionRef::JNAT, DirectionRef::APP),
17    (DirectionRef::HADEC, DirectionRef::AZEL),
18    (DirectionRef::HADEC, DirectionRef::AZELGEO),
19    (DirectionRef::AZEL, DirectionRef::AZELSW),
20    (DirectionRef::AZELGEO, DirectionRef::AZELSWGEO),
21    // casacore reaches HADEC from APP through TOPO.
22    (DirectionRef::APP, DirectionRef::TOPO),
23    (DirectionRef::TOPO, DirectionRef::HADEC),
24    (DirectionRef::J2000, DirectionRef::B1950),
25    // casacore converts between GALACTIC and B1950 directly, not via J2000.
26    (DirectionRef::GALACTIC, DirectionRef::B1950),
27    (DirectionRef::HADEC, DirectionRef::ITRF),
28    (DirectionRef::J2000, DirectionRef::B1950Vla),
29    (DirectionRef::B1950, DirectionRef::B1950Vla),
30    (DirectionRef::B1950, DirectionRef::BMEAN),
31    (DirectionRef::BMEAN, DirectionRef::BTRUE),
32];

34/// Maps a DirectionRef to a unique index for BFS.
35pub(super) fn dir_ref_index(r: DirectionRef) -> usize {
36    match r {
37        DirectionRef::J2000 => 0,
38        DirectionRef::JMEAN => 1,
39        DirectionRef::JTRUE => 2,
40        DirectionRef::APP => 3,
41        DirectionRef::JNAT => 4,
42        DirectionRef::B1950 => 5,
43        DirectionRef::GALACTIC => 6,
44        DirectionRef::SUPERGAL => 7,
45        DirectionRef::ICRS => 8,
46        DirectionRef::ECLIPTIC => 9,
47        DirectionRef::MECLIPTIC => 10,
48        DirectionRef::TECLIPTIC => 11,
49        DirectionRef::HADEC => 12,
50        DirectionRef::AZEL => 13,
51        DirectionRef::AZELSW => 14,
52        DirectionRef::AZELGEO => 15,
53        DirectionRef::AZELSWGEO => 16,
54        DirectionRef::ITRF => 17,
55        DirectionRef::TOPO => 18,
56        DirectionRef::B1950Vla => 19,
57        DirectionRef::BMEAN => 20,
58        DirectionRef::BTRUE => 21,
59        _ => unreachable!("dynamical references resolve before rotation routing"),
60    }
61}

63/// Pre-computed BFS paths for all reachable (source, target) pairs.
64pub(super) type DirPathMap = HashMap<(DirectionRef, DirectionRef), Option<Vec<DirectionRef>>>;
65
66pub(super) static DIR_PATH_CACHE: LazyLock<DirPathMap> = LazyLock::new(|| {
67    let mut cache = HashMap::new();
68    for &src in &DirectionRef::ALL {
69        for &tgt in &DirectionRef::ALL {
70            if src != tgt {
71                cache.insert((src, tgt), bfs_find_dir_path(src, tgt));
72            }
73        }
74    }
75    cache
76});

78/// Finds the shortest path from `source` to `target` in the routing graph.
79/// Results are cached in `DIR_PATH_CACHE`.
80pub(super) fn find_path(
81    source: DirectionRef,
82    target: DirectionRef,
83) -> Result<Vec<DirectionRef>, MeasureError> {
84    DIR_PATH_CACHE
85        .get(&(source, target))
86        .expect("all pairs pre-computed")
87        .clone()
88        .ok_or_else(|| MeasureError::NotYetImplemented {
89            route: format!("{source} → {target} (no route found)"),
90        })
91}

93/// BFS implementation used to populate the path cache.
94pub(super) fn bfs_find_dir_path(
95    source: DirectionRef,
96    target: DirectionRef,
97) -> Option<Vec<DirectionRef>> {
98    let mut visited = [false; NUM_DIR_REFS];
99    let mut parent: [Option<DirectionRef>; NUM_DIR_REFS] = [None; NUM_DIR_REFS];
100    let mut queue = VecDeque::new();
101
102    let src_idx = dir_ref_index(source);
103    visited[src_idx] = true;
104    queue.push_back(source);
105
106    let mut found = false;
107
108    while let Some(current) = queue.pop_front() {
109        if current == target {
110            found = true;
111            break;
112        }
113
114        for &(a, b) in ROUTING_EDGES {
115            let neighbor = if a == current {
116                b
117            } else if b == current {
118                a
119            } else {
120                continue;
121            };
122            let idx = dir_ref_index(neighbor);
123            if !visited[idx] {
124                visited[idx] = true;
125                parent[idx] = Some(current);
126                queue.push_back(neighbor);
127            }
128        }
129    }
130
131    if found {
132        let mut path = Vec::new();
133        let mut cur = target;
134        while cur != source {
135            path.push(cur);
136            cur = parent[dir_ref_index(cur)].expect("BFS parent must be set");
137        }
138        path.reverse();
139        Some(path)
140    } else {
141        None
142    }
143}

149/// Caches expensive epoch-derived values (TT JD pair, UT1 JD pair, GAST)
150/// so they are computed at most once per `convert_to` call.
151pub(super) struct ConvCtx {
152    pub(super) distance_m: f64,
153    pub(super) tt_jd: RefCell<Option<(f64, f64)>>,
154    pub(super) ut1_jd: RefCell<Option<(f64, f64)>>,
155    pub(super) gast: RefCell<Option<f64>>,
156    pub(super) precession: RefCell<Option<[[f64; 3]; 3]>>,
157    pub(super) nutation: RefCell<Option<[[f64; 3]; 3]>>,
158    pub(super) earth_vel: RefCell<Option<([f64; 3], f64)>>,
159}
160
161impl ConvCtx {
162    pub(super) fn new() -> Self {
163        Self {
164            distance_m: 0.0,
165            tt_jd: RefCell::new(None),
166            ut1_jd: RefCell::new(None),
167            gast: RefCell::new(None),
168            precession: RefCell::new(None),
169            nutation: RefCell::new(None),
170            earth_vel: RefCell::new(None),
171        }
172    }
173
174    pub(super) fn get_tt_jd(&self, frame: &MeasFrame) -> Result<(f64, f64), MeasureError> {
175        if let Some(v) = *self.tt_jd.borrow() {
176            return Ok(v);
177        }
178        let v = frame.cached_tt_jd(|| compute_tt_jd(frame))?;
179        *self.tt_jd.borrow_mut() = Some(v);
180        Ok(v)
181    }
182
183    pub(super) fn get_ut1_jd(&self, frame: &MeasFrame) -> Result<(f64, f64), MeasureError> {
184        if let Some(v) = *self.ut1_jd.borrow() {
185            return Ok(v);
186        }
187        let v = frame.cached_ut1_jd(|| compute_ut1_jd(frame))?;
188        *self.ut1_jd.borrow_mut() = Some(v);
189        Ok(v)
190    }
191
192    pub(super) fn get_gast(&self, frame: &MeasFrame) -> Result<f64, MeasureError> {
193        if let Some(v) = *self.gast.borrow() {
194            return Ok(v);
195        }
196        let v = frame.cached_gast(|| compute_gast(frame, self))?;
197        *self.gast.borrow_mut() = Some(v);
198        Ok(v)
199    }
200
201    pub(super) fn get_precession(&self, frame: &MeasFrame) -> Result<[[f64; 3]; 3], MeasureError> {
202        if let Some(v) = *self.precession.borrow() {
203            return Ok(v);
204        }
205        let v = frame.cached_precession(|| {
206            let (tt1, tt2) = self.get_tt_jd(frame)?;
207            Ok(match frame.iau_model() {
208                IauModel::Iau1976_1980 => sofars::pnp::pmat76(tt1, tt2),
209                IauModel::Iau2006_2000A => sofars::pnp::pmat00(tt1, tt2),
210            })
211        })?;
212        *self.precession.borrow_mut() = Some(v);
213        Ok(v)
214    }
215
216    pub(super) fn get_nutation(&self, frame: &MeasFrame) -> Result<[[f64; 3]; 3], MeasureError> {
217        if let Some(v) = *self.nutation.borrow() {
218            return Ok(v);
219        }
220        let v = frame.cached_nutation(|| {
221            let (tt1, tt2) = self.get_tt_jd(frame)?;
222            Ok(match frame.iau_model() {
223                IauModel::Iau1976_1980 => sofars::pnp::nutm80(tt1, tt2),
224                IauModel::Iau2006_2000A => sofars::pnp::num00a(tt1, tt2),
225            })
226        })?;
227        *self.nutation.borrow_mut() = Some(v);
228        Ok(v)
229    }
230
231    pub(super) fn get_earth_vel(&self, frame: &MeasFrame) -> Result<([f64; 3], f64), MeasureError> {
232        if let Some(v) = *self.earth_vel.borrow() {
233            return Ok(v);
234        }
235        let v = frame.cached_earth_velocity_au_per_day(|| {
236            let (tt1, tt2) = self.get_tt_jd(frame)?;
237            let (pvh, pvb) =
238                sofars::eph::epv00(tt1, tt2).ok_or(MeasureError::SofarsError { code: -1 })?;
239            let sun_dist = sofars::vm::pm(pvh[0]);
240            Ok((pvb[1], sun_dist))
241        })?;
242        *self.earth_vel.borrow_mut() = Some(v);
243        Ok(v)
244    }
245}

251/// Applies a single conversion hop.
252pub(super) fn apply_hop(
253    cosines: [f64; 3],
254    from: DirectionRef,
255    to: DirectionRef,
256    frame: &MeasFrame,
257    ctx: &ConvCtx,
258) -> Result<[f64; 3], MeasureError> {
259    use DirectionRef::*;
260
261    match (from, to) {
262        // ----- Constant matrices -----
263
264        // J2000, B1950 and SUPERGAL ↔ GALACTIC (casacore's matrices)
265        (J2000, GALACTIC) => Ok(casacore_frames::j2000_to_galactic(cosines)),
266        (GALACTIC, J2000) => Ok(casacore_frames::galactic_to_j2000(cosines)),
267        (B1950, GALACTIC) => Ok(casacore_frames::b1950_to_galactic(cosines)),
268        (GALACTIC, B1950) => Ok(casacore_frames::galactic_to_b1950(cosines)),
269        (SUPERGAL, GALACTIC) => Ok(casacore_frames::supergalactic_to_galactic(cosines)),
270        (GALACTIC, SUPERGAL) => Ok(casacore_frames::galactic_to_supergalactic(cosines)),
271
272        // J2000 ↔ ICRS (frame tie, ~17 mas)
273        (ICRS, J2000) => {
274            let m = icrs_to_j2000_matrix();
275            Ok(rotate(&m, &cosines))
276        }
277        (J2000, ICRS) => {
278            let m = icrs_to_j2000_matrix();
279            Ok(rotate_t(&m, &cosines))
280        }
281
282        // AZEL ↔ AZELSW (sign flip: Az_SW = Az + π, El_SW = El)
283        // In direction cosines: negate x and y, keep z.
284        (AZEL, AZELSW) | (AZELSW, AZEL) => Ok([-cosines[0], -cosines[1], cosines[2]]),
285
286        // AZELGEO ↔ AZELSWGEO (same sign flip)
287        (AZELGEO, AZELSWGEO) | (AZELSWGEO, AZELGEO) => Ok([-cosines[0], -cosines[1], cosines[2]]),
288
289        // J2000 ↔ JNAT (solar gravitational deflection)
290        (J2000, JNAT) => apply_solar_deflection(cosines, frame),
291        (JNAT, J2000) => remove_solar_deflection(cosines, frame),
292
293        // JNAT ↔ APP. C++ casacore combines annual aberration with the
294        // J2000-to-true precession/nutation rotation in this hop.
295        (JNAT, APP) => {
296            let beta = casacore_aberration_beta(frame)?;
297            let aberrated_j2000 = apply_relativistic_aberration(cosines, beta);
298            let precession = precession_matrix(frame, ctx)?;
299            let nutation = nutation_matrix(frame, ctx)?;
300            Ok(rotate(&nutation, &rotate(&precession, &aberrated_j2000)))
301        }
302        (APP, JNAT) => {
303            let precession = precession_matrix(frame, ctx)?;
304            let nutation = nutation_matrix(frame, ctx)?;
305            let aberrated_j2000 = rotate_t(&precession, &rotate_t(&nutation, &cosines));
306            let beta = casacore_aberration_beta(frame)?.map(|component| -component);
307            Ok(apply_relativistic_aberration(aberrated_j2000, beta))
308        }
309
310        // ----- Epoch-dependent matrices -----
311
312        // J2000 → JMEAN (precession with frame bias)
313        (J2000, JMEAN) => {
314            let rp = precession_matrix(frame, ctx)?;
315            Ok(rotate(&rp, &cosines))
316        }
317        (JMEAN, J2000) => {
318            let rp = precession_matrix(frame, ctx)?;
319            Ok(rotate_t(&rp, &cosines))
320        }
321
322        // JMEAN → JTRUE (nutation)
323        (JMEAN, JTRUE) => {
324            let rn = nutation_matrix(frame, ctx)?;
325            Ok(rotate(&rn, &cosines))
326        }
327        (JTRUE, JMEAN) => {
328            let rn = nutation_matrix(frame, ctx)?;
329            Ok(rotate_t(&rn, &cosines))
330        }
331
332        // J2000 ↔ ECLIPTIC (casacore's mean obliquity at J2000.0, no bias)
333        (J2000, ECLIPTIC) => Ok(rotate(&j2000_ecliptic_rotation(), &cosines)),
334        (ECLIPTIC, J2000) => Ok(rotate_t(&j2000_ecliptic_rotation(), &cosines)),
335
336        // JMEAN ↔ MECLIPTIC (casacore's mean obliquity at the TDB epoch)
337        (JMEAN, MECLIPTIC) | (MECLIPTIC, JMEAN) => {
338            let eps = casacore_frames::mean_obliquity_rad(tdb_mjd(frame)?);
339            Ok(rotate_obliquity(eps, &cosines, from == JMEAN))
340        }
341
342        // JTRUE ↔ TECLIPTIC (true obliquity of casacore's IAU 1980
343        // nutation, `Nutation::STANDARD`, under either IAU model)
344        (JTRUE, TECLIPTIC) | (TECLIPTIC, JTRUE) => {
345            let eps = standard_true_obliquity_rad(tdb_mjd(frame)?);
346            Ok(rotate_obliquity(eps, &cosines, from == JTRUE))
347        }
348
349        // JTRUE → APP (aberration)
350        (JTRUE, APP) => apply_annual_aberration(cosines, frame, ctx),
351        // APP → JTRUE (inverse aberration — iterative)
352        (APP, JTRUE) => remove_annual_aberration(cosines, frame, ctx),
353
354        // APP ↔ TOPO: casacore's `applyAPPtoTOPO` shifts only a direction
355        // with a distance (a planet's), so a sky direction is unchanged.
356        (APP, TOPO) => apparent_parallax(cosines, -ctx.distance_m, frame, ctx),
357        (TOPO, APP) => apparent_parallax(cosines, ctx.distance_m, frame, ctx),
358
359        // TOPO → HADEC (diurnal aberration + polar motion + sidereal time)
360        (TOPO, HADEC) => topocentric_to_hadec(cosines, frame, ctx),
361        // HADEC → TOPO (inverse of above)
362        (HADEC, TOPO) => hadec_to_topocentric(cosines, frame, ctx),
363
364        // HADEC ↔ AZEL (horizon coordinates using geocentric latitude)
365        // C++ casacore uses geocentric latitude for AZEL (MeasMath::applyHADECtoAZEL
366        // calls getInfo(LAT), which is asin(z/r) from ITRF XYZ).
367        (HADEC, AZEL) => {
368            let lat = get_geocentric_latitude(frame)?;
369            let (ha, dec) = sofars::vm::c2s(&cosines);
370            let [az, el] = sofars::coords::hd2ae(ha, dec, lat);
371            Ok(sofars::vm::s2c(az, el))
372        }
373        (AZEL, HADEC) => {
374            let lat = get_geocentric_latitude(frame)?;
375            let (az, el) = sofars::vm::c2s(&cosines);
376            let [ha, dec] = sofars::coords::ae2hd(az, el, lat);
377            Ok(sofars::vm::s2c(ha, dec))
378        }
379
380        // HADEC ↔ AZELGEO (horizon coordinates using geodetic latitude)
381        // C++ casacore uses geodetic latitude for AZELGEO (MeasMath::applyHADECtoAZELGEO
382        // calls getInfo(LATGEO), which is from WGS84 conversion).
383        (HADEC, AZELGEO) => {
384            let lat = get_latitude(frame)?;
385            let (ha, dec) = sofars::vm::c2s(&cosines);
386            let [az, el] = sofars::coords::hd2ae(ha, dec, lat);
387            Ok(sofars::vm::s2c(az, el))
388        }
389        (AZELGEO, HADEC) => {
390            let lat = get_latitude(frame)?;
391            let (az, el) = sofars::vm::c2s(&cosines);
392            let [ha, dec] = sofars::coords::ae2hd(az, el, lat);
393            Ok(sofars::vm::s2c(ha, dec))
394        }
395
396        (J2000, B1950Vla) => Ok(casacore_frames::j2000_to_b1950(cosines, 19.799 - 20.0)),
397        (B1950Vla, J2000) => Ok(casacore_frames::b1950_to_j2000(cosines, 19.799 - 20.0)),
398        (B1950, B1950Vla) => Ok(casacore_frames::j2000_to_b1950(
399            casacore_frames::b1950_to_j2000(cosines, b1950_epoch_centuries(frame, ctx)?),
400            19.799 - 20.0,
401        )),
402        (B1950Vla, B1950) => Ok(casacore_frames::j2000_to_b1950(
403            casacore_frames::b1950_to_j2000(cosines, 19.799 - 20.0),
404            b1950_epoch_centuries(frame, ctx)?,
405        )),
406        (B1950, APP) | (APP, B1950) => vector_rotation::fk4_apparent_direction(cosines, to, frame),
407        (B1950, BMEAN) | (BMEAN, B1950) => Ok(casacore_frames::besselian_precession(
408            cosines,
409            tdb_mjd(frame)?,
410            from == BMEAN,
411        )),
412        (BMEAN, BTRUE) | (BTRUE, BMEAN) => Ok(casacore_frames::besselian_nutation(
413            cosines,
414            tdb_mjd(frame)?,
415            from == BTRUE,
416        )),
417
418        // ----- B1950 (FK4, casacore's rotation and E-terms) -----
419        (J2000, B1950) => Ok(casacore_frames::j2000_to_b1950(
420            cosines,
421            b1950_epoch_centuries(frame, ctx)?,
422        )),
423        (B1950, J2000) => Ok(casacore_frames::b1950_to_j2000(
424            cosines,
425            b1950_epoch_centuries(frame, ctx)?,
426        )),
427
428        // ----- ITRF (via HADEC, matching C++ MeasMath::applyHADECtoITRF) -----
429
430        // HADEC → ITRF: rotate by -longitude around z, then negate y.
431        //
432        // C++ MeasMath::applyHADECtoITRF does `in *= Rz(lon)` (which means
433        // `in = Rz(lon)^T * in = Rz(-lon) * in`) then negates y.
434        (HADEC, ITRF) => {
435            let lon = get_longitude(frame)?;
436            let (s, c) = lon.sin_cos();
437            // Rz(-lon) * in
438            let x = c * cosines[0] + s * cosines[1];
439            let y = -s * cosines[0] + c * cosines[1];
440            let z = cosines[2];
441            // Negate y
442            Ok([x, -y, z])
443        }
444        // ITRF → HADEC: negate y, then rotate by +longitude around z.
445        //
446        // C++ MeasMath::deapplyHADECtoITRF negates y then does
447        // `in = Rz(lon) * in`.
448        (ITRF, HADEC) => {
449            let lon = get_longitude(frame)?;
450            let (s, c) = lon.sin_cos();
451            // Negate y first
452            let ny = -cosines[1];
453            // Rz(lon) * in
454            let x = c * cosines[0] - s * ny;
455            let y = s * cosines[0] + c * ny;
456            let z = cosines[2];
457            Ok([x, y, z])
458        }
459
460        _ => Err(MeasureError::NotYetImplemented {
461            route: format!("{from} → {to}"),
462        }),
463    }
464}

466/// JTRUE → APP hop: apply annual aberration from the Earth's barycentric
467/// velocity, rotated into the true equatorial frame.
468pub(super) fn apply_annual_aberration(
469    cosines: [f64; 3],
470    frame: &MeasFrame,
471    ctx: &ConvCtx,
472) -> Result<[f64; 3], MeasureError> {
473    let (v_au_day, sun_dist) = earth_velocity_au_per_day(frame, ctx)?;
474    // Convert velocity to units of c
475    let v = [
476        v_au_day[0] / C_AU_PER_DAY,
477        v_au_day[1] / C_AU_PER_DAY,
478        v_au_day[2] / C_AU_PER_DAY,
479    ];
480    // Rotate velocity to true equatorial frame
481    let (tt1, tt2) = get_tt_jd(frame, ctx)?;
482    let rbpn = bias_precession_nutation(frame, tt1, tt2);
483    let v_true = rotate(&rbpn, &v);
484    // Reciprocal of Lorentz factor: sqrt(1 - |v|²)
485    let v2 = v[0] * v[0] + v[1] * v[1] + v[2] * v[2];
486    let bm1 = (1.0 - v2).sqrt();
487    Ok(sofars::astro::ab(&cosines, &v_true, sun_dist, bm1))
488}
489
490/// APP → JTRUE hop: remove annual aberration by fixed-point iteration on
491/// the forward transform of [`apply_annual_aberration`].
492pub(super) fn remove_annual_aberration(
493    cosines: [f64; 3],
494    frame: &MeasFrame,
495    ctx: &ConvCtx,
496) -> Result<[f64; 3], MeasureError> {
497    let (v_au_day, sun_dist) = earth_velocity_au_per_day(frame, ctx)?;
498    let v = [
499        v_au_day[0] / C_AU_PER_DAY,
500        v_au_day[1] / C_AU_PER_DAY,
501        v_au_day[2] / C_AU_PER_DAY,
502    ];
503    let (tt1, tt2) = get_tt_jd(frame, ctx)?;
504    let rbpn = bias_precession_nutation(frame, tt1, tt2);
505    let v_true = rotate(&rbpn, &v);
506    let v2 = v[0] * v[0] + v[1] * v[1] + v[2] * v[2];
507    let bm1 = (1.0 - v2).sqrt();
508
509    // Iterative removal: start with apparent as guess for natural
510    let mut pnat = cosines;
511    for _ in 0..5 {
512        let papp = sofars::astro::ab(&pnat, &v_true, sun_dist, bm1);
513        // Correct: pnat += (cosines - papp)
514        for k in 0..3 {
515            pnat[k] += cosines[k] - papp[k];
516        }
517        // Re-normalize
518        let (_, unit) = sofars::vm::pn(&pnat);
519        pnat = unit;
520    }
521    Ok(pnat)
522}

524/// Observer displacement along the local geocentric zenith, in APP axes.
525pub(super) fn apparent_parallax(
526    cosines: [f64; 3],
527    signed_distance_m: f64,
528    frame: &MeasFrame,
529    ctx: &ConvCtx,
530) -> Result<[f64; 3], MeasureError> {
531    if signed_distance_m == 0.0 {
532        return Ok(cosines);
533    }
534    let last = get_gast(frame, ctx)? + get_longitude(frame)?;
535    let zenith = sofars::vm::s2c(last, get_geocentric_latitude(frame)?);
536    let scale = get_geocentric_radius(frame)? / signed_distance_m;
537    let shifted = std::array::from_fn(|i| cosines[i] + scale * zenith[i]);
538    Ok(sofars::vm::pn(&shifted).1)
539}
540
541/// TOPO → HADEC hop (diurnal aberration + polar motion + sidereal time),
542/// casacore's `MeasMath::applyTOPOtoHADEC`. The steps are:
543///   1. Diurnal aberration: add v/c in zenith direction, renormalize
544///   2. Polar motion + LAST rotation via Euler R = Ry(xp)×Rx(yp)×Rz(LAST)
545///   3. Apply R^T and negate y to get (HA, Dec)
546pub(super) fn topocentric_to_hadec(
547    cosines: [f64; 3],
548    frame: &MeasFrame,
549    ctx: &ConvCtx,
550) -> Result<[f64; 3], MeasureError> {
551    let gast = get_gast(frame, ctx)?;
552    let lon = get_longitude(frame)?;
553    let last = gast + lon;
554
555    // Diurnal aberration: shift in direction of (LAST, geocentric_lat)
556    let geo_lat = get_geocentric_latitude(frame)?;
557    let radius = get_geocentric_radius(frame)?;
558    let v_c = diurnal_aberration_factor(radius);
559    let aberr_dir = sofars::vm::s2c(last, geo_lat);
560    let shifted = [
561        cosines[0] + v_c * aberr_dir[0],
562        cosines[1] + v_c * aberr_dir[1],
563        cosines[2] + v_c * aberr_dir[2],
564    ];
565    let (_, unit) = sofars::vm::pn(&shifted);
566
567    // Polar motion + LAST rotation + y-negate via Euler matrix
568    // C++ uses: in *= R(xp, Ry, yp, Rx, LAST, Rz); in.y = -in.y
569    // which is: in = R^T * in, then negate y
570    let (_, ut2) = get_ut1_jd(frame, ctx)?;
571    let (xp, yp) = get_polar_motion_rad(frame, ut2)?;
572    let r = polar_motion_euler(-xp, -yp, last);
573    let result = rotate_t(&r, &unit);
574    Ok([result[0], -result[1], result[2]])
575}
576
577/// HADEC → TOPO hop: the inverse of [`topocentric_to_hadec`].
578pub(super) fn hadec_to_topocentric(
579    cosines: [f64; 3],
580    frame: &MeasFrame,
581    ctx: &ConvCtx,
582) -> Result<[f64; 3], MeasureError> {
583    let gast = get_gast(frame, ctx)?;
584    let lon = get_longitude(frame)?;
585    let last = gast + lon;
586
587    // applyPolarMotion: negate y, then apply R
588    let (_, ut2) = get_ut1_jd(frame, ctx)?;
589    let (xp, yp) = get_polar_motion_rad(frame, ut2)?;
590    let r = polar_motion_euler(-xp, -yp, last);
591    let negated = [cosines[0], -cosines[1], cosines[2]];
592    let rotated = rotate(&r, &negated);
593
594    // Undo diurnal aberration
595    let geo_lat = get_geocentric_latitude(frame)?;
596    let radius = get_geocentric_radius(frame)?;
597    let v_c = diurnal_aberration_factor(radius);
598    let aberr_dir = sofars::vm::s2c(last, geo_lat);
599    let unshifted = [
600        rotated[0] - v_c * aberr_dir[0],
601        rotated[1] - v_c * aberr_dir[1],
602        rotated[2] - v_c * aberr_dir[2],
603    ];
604    let (_, unit) = sofars::vm::pn(&unshifted);
605    Ok(unit)
606}

608// ---------------------------------------------------------------------------
609// Public helper: get J2000 direction (used by frequency module)
610// ---------------------------------------------------------------------------
611
612impl MDirection {
613    /// Evaluate geocentric apparent place for the horizon-crossing calculation.
614    ///
615    /// Casacore uses its direct FK4 precession, nutation, and aberration
616    /// for a B1950 input. Other references use the normal APP conversion.
617    pub fn to_apparent(&self, frame: &MeasFrame) -> Result<Self, MeasureError> {
618        if self.refer != DirectionRef::B1950 {
619            return self.convert_to(DirectionRef::APP, frame);
620        }
621        let cosines =
622            vector_rotation::fk4_apparent_direction(self.cosines, DirectionRef::APP, frame)?;
623        Ok(Self {
624            refer: DirectionRef::APP,
625            cosines,
626        })
627    }
628
629    /// Returns this direction converted to J2000 cosines.
630    ///
631    /// This is used internally by frequency conversions to compute the
632    /// dot product with velocity vectors defined in J2000.
633    pub fn to_j2000(&self, frame: &MeasFrame) -> Result<[f64; 3], MeasureError> {
634        if self.refer == DirectionRef::J2000 {
635            return Ok(self.cosines);
636        }
637        let j2000 = self.convert_to(DirectionRef::J2000, frame)?;
638        Ok(j2000.cosines)
639    }
