14fn axes(coordinates: &CoordinateSystem, kind: &str) -> Option<Vec<usize>> {
15    if kind.is_empty() {
16        return Some((0..coordinates.n_pixel_axes()).collect());
17    }
18    let index = coordinates.find_coordinate(coordinate_type(kind))?;
19    Some(
20        coordinates
21            .coordinate_pixel_axes(index)?
22            .into_iter()
23            .flatten()
24            .collect(),
25    )
26}

28#[pymethods]
29impl PyCoordsys {
30    #[pyo3(signature = (r#type = ""), text_signature = "($self, type='')")]
31    fn referencepixel(&self, py: Python<'_>, r#type: &str) -> PyResult<Py<PyAny>> {
32        let value = py.detach(|| {
33            let state = self.state();
34            let Some(range) = axes(&state.coordinates, r#type) else {
35                return CasaValue::empty_record();
36            };
37            let all = axis_values(&state.coordinates, false, CoordinateModel::reference_pixel);
38            let pixels = range.into_iter().map(|axis| all[axis]).collect();
39            CasaValue::record([
40                ("numeric", CasaValue::floats(pixels)),
41                ("pw_type", CasaValue::Text("pixel".into())),
42                ("ar_type", CasaValue::Text("absolute".into())),
43            ])
44        });
45        value.into_python(py)
46    }
47
48    #[pyo3(signature = (value, r#type = ""), text_signature = "($self, value, type='')")]
49    fn setreferencepixel(&self, py: Python<'_>, value: DoubleList, r#type: &str) -> PyResult<bool> {
50        py.detach(|| {
51            let mut state = self.state();
52            let range = axes(&state.coordinates, r#type).ok_or_else(|| {
53                PyRuntimeError::new_err(format!(
54                    "A coordinate of type {type_name} does not exist",
55                    type_name = r#type
56                ))
57            })?;
58            let mut record = state.editable_record()?;
59            for (axis, value) in range.into_iter().zip(value.0) {
60                state.edit_record_axis(&mut record, axis, TabularEdit::ReferencePixel(value))?;
61            }
62            state.apply_edited_record(record)?;
63            Ok(true)
64        })
65    }
66
67    fn torecord(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
68        py.detach(|| {
69            crate::tools::table::values::record(&self.state().record, std::path::Path::new("."))
70        })
71        .into_python(py)
72    }
73
74    fn done(&self, py: Python<'_>) -> bool {
75        py.detach(|| {
76            let mut state = self.lock_state();
77            *state = super::CoordinateState::empty();
78            state.parent_name.clear();
79        });
80        true
81    }
82
83    fn r#type(&self) -> &'static str {
84        "coordsys"
85    }
86}
