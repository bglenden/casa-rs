2//! Sample rate-registered Fringe Jones solutions on visibility frequencies.
3
4use super::*;
5
6pub(super) fn sample(
7    grid: &GainGrid,
8    receptor: usize,
9    channel: usize,
10    context: &CalibrationSamplingContext<'_>,
11) -> Result<GainSample, ApplyExecutionError> {
12    let frequencies = context.data_channels.frequencies_hz;
13    let frequency =
14        *frequencies
15            .get(channel)
16            .ok_or_else(|| ApplyExecutionError::UnsupportedInterpolation {
17                path: context.path.display().to_string(),
18                interp: context.interp,
19                reason: "Fringe Jones data channel is outside the frequency grid".into(),
20            })?;
21    let index = 4 * receptor;
22    let phase = f64::from(grid.values[[index, 0]].re);
23    let delay = f64::from(grid.values[[index + 1, 0]].re);
24    let dispersive = grid.values[[index + 3, 0]].re;
25    let phase =
26        phase + std::f64::consts::TAU * delay * (frequency - context.cal_ref_frequency_hz) / 1e9;
27    let correction = if dispersive == 0. {
28        0.
29    } else {
30        let low = (frequencies
31            .iter()
32            .copied()
33            .reduce(f64::min)
34            .expect("visibility frequencies")
35            / 1e9) as f32;
36        let high = (frequencies
37            .iter()
38            .copied()
39            .reduce(f64::max)
40            .expect("visibility frequencies")
41            / 1e9) as f32;
42        let center = (frequency / 1e9) as f32;
43        let basis = 1. / f64::from(center) + f64::from((center - low - high) / (low * high));
44        f64::from(dispersive * (1e-3 * std::f64::consts::TAU * basis) as f32)
45    };
46    let (sine, cosine) = (phase + correction).sin_cos();
47    Ok(GainSample {
48        value: Complex32::new(cosine as f32, sine as f32),
49        flagged: grid.flags[[index, 0]],
50    })
51}
