34/// The compiled spectral axis and the source channels it reads.
35pub(super) struct PreparedSpectralAxis<'a> {
36    /// Each store's source channels the axis reads, in store order.
37    pub(super) selected_source_channels: Cow<'a, [WindowChannels]>,
38    /// How each store's windows read beyond their selected channels at
39    /// every time ([`WindowRead`]), in store order.
40    pub(super) window_reads: Vec<WindowReads>,
41    pub(super) source_frame: FrequencyFrame,
42    pub(super) output_frequency_reference: FrequencyRef,
43    pub(super) output_frame: FrequencyFrame,
44    pub(super) anchor: SpectralFrameAnchor,
45    pub(super) wcs: SpectralWcs,
46    pub(super) rest_frequency: RestFrequency,
47    /// The spectral axis the image header records: the gridded axis, but
48    /// for an MFS image's frame and rest frequencies.
49    pub(super) image: ImageSpectralCoordinate,
50    pub(super) doppler: DopplerConvention,
51    pub(super) sampling: SpectralSamplingLaw,
52    pub(super) basis: ReconstructionBasis,
53    pub(super) output_channels: usize,
54    /// Centre of the first output channel.
55    pub(super) reference_frequency_hz: f64,
56    /// Channel increment; a continuum plane's full band.
57    pub(super) increment_hz: f64,
58}

60impl PreparedSpectralAxis<'_> {
61    /// The model's spectral coordinate of this axis.
62    pub(super) fn coordinate(&self) -> SpectralCoordinateSpec {
63        SpectralCoordinateSpec::new(
64            self.source_frame,
65            self.output_frame,
66            self.anchor,
67            self.wcs.clone(),
68            self.rest_frequency,
69            self.doppler,
70        )
71    }
72
73    /// The source channels the axis reads from surveyed window `spw_id` of
74    /// store `store`.
75    pub(super) fn window_channels(&self, store: usize, spw_id: usize) -> &[usize] {
76        self.selected_source_channels[store]
77            .get(&spw_id)
78            .expect("the spectral axis selects channels of every surveyed window")
79    }
80}

144/// The request's spectral axis over the surveyed selection of `stores`,
145/// laid on store `reference`'s metadata (CASA's `msobj`): a continuum
146/// plane over `continuum`, the band of every store, or a cube on the
147/// reference store's window that reads each store's window where it reaches
148/// the cube.
149pub(super) fn prepare_axis<'a>(
150    request: &ImagingRequest,
151    (stores, reference): (&[SurveyedStore], usize),
152    continuum: Option<&'a ContinuumBand>,
153    frame: &FrameContext<'_>,
154    moving_rest_frame: Option<&MeasFrame>,
155    continuum_reference_hz: Option<f64>,
156) -> Result<PreparedSpectralAxis<'a>, PrepareError> {
157    let store = &stores[reference];
158    let (msv2, catalogue) = (
159        store
160            .msv2()
161            .map(|observation| observation.measurement_set()),
162        store.catalogue(),
163    );
164    let survey = &store.survey;
165    let source_reference = survey.source_frequency_reference;
166    let law = match request.specmode {
167        SpecMode::Mfs => continuum_law(
168            request,
169            (msv2, catalogue),
170            survey,
171            continuum.expect("prepare spans the band of every continuum run"),
172            continuum_reference_hz,
173        )?,
174        SpecMode::Cube | SpecMode::Cubedata => cube_law(request, (stores, reference), frame)?,
175        SpecMode::Cubesource => {
176            super::stores::one_store(stores, "a source-frame cube (specmode cubesource)")?;
177            let frame_with_velocity = moving_rest_frame
178                .expect("prepare resolves the moving rest frame of every source-frame cube");
179            source_frame_cube_law(request, store, frame, frame_with_velocity)?
180        }
181    };
182    let source_frame = imaging_frequency_frame(source_reference)?;
183    let output_frame = imaging_frequency_frame(law.output_frequency_reference)?;
184    Ok(PreparedSpectralAxis {
185        selected_source_channels: law.selected_source_channels,
186        window_reads: law.window_reads,
187        source_frame,
188        output_frequency_reference: law.output_frequency_reference,
189        output_frame,
190        anchor: spectral_frame_anchor(source_frame, output_frame, request.specmode, frame)?,
191        wcs: SpectralWcs::Linear {
192            channels: law.output_channels,
193            reference_pixel: 0.0,
194            reference_frequency_hz: law.reference_frequency_hz,
195            increment_hz: law.increment_hz,
196        },
197        rest_frequency: law.rest_frequency,
198        // CASA lists tclean's further rest frequencies after the image's.
199        image: ImageSpectralCoordinate {
200            additional_rest_frequencies_hz: request.restfreq.iter().skip(1).copied().collect(),
201            ..law.image
202        },
203        doppler: law.doppler,
204        sampling: law.sampling,
205        basis: law.basis,
206        output_channels: law.output_channels,
207        reference_frequency_hz: law.reference_frequency_hz,
208        increment_hz: law.increment_hz,
209    })
210}

212/// One plane over the selected band's envelope over every store (`band`),
213/// gridded in LSRK as CASA's `FTMachine::matchChannel` grids it whatever
214/// the image's `outframe`. Its header records the envelope in `outframe`
215/// (`buildCoordinateSystem`'s `MSUtil::getFreqRangeInSpw`) and `restfreq`'s
216/// first rest frequency, else the reference store's (`survey`'s) first
217/// visibility buffer's source line ([`continuum_rest_frequency_hz`] on an
218/// MSv2, whose `SOURCE` and `DOPPLER` tables CASA reads; the catalogue's
219/// first line on an MSv4), else the recorded band's midpoint (`getImFreq`).
220fn continuum_law<'a>(
221    request: &ImagingRequest,
222    (msv2, catalogue): (Option<&MeasurementSet>, &Catalogue),
223    survey: &Survey,
224    band: &'a ContinuumBand,
225    continuum_reference_hz: Option<f64>,
226) -> Result<AxisLaw<'a>, PrepareError> {
227    let (channels, envelope, recorded) = (&band.channels, &band.gridded, &band.recorded);
228    let [lower_hz, upper_hz] = envelope.edges_hz();
229    let [recorded_lower_hz, recorded_upper_hz] = recorded.edges_hz();
230    // `getImFreq`: a given reffreq is the reference at pixel 0, with the
231    // band widened by twice its offset from the band's centre (CAS-9518).
232    let reference = |midpoint_hz: f64, width_hz: f64| match continuum_reference_hz {
233        Some(reference_hz) => (
234            reference_hz,
235            width_hz + 2.0 * (reference_hz - midpoint_hz).abs(),
236        ),
237        None => (midpoint_hz, width_hz),
238    };
239    if continuum_reference_hz.is_some() && request.outframe != FrequencyRef::LSRK {
240        return Err(PrepareError::ReferenceFrequencyOutsideLsrk {
241            outframe: request.outframe,
242        });
243    }
244    let (reference_frequency_hz, increment_hz) =
245        reference(envelope.midpoint_hz(), upper_hz - lower_hz);
246    let (image_reference_hz, image_increment_hz) = reference(
247        recorded.midpoint_hz(),
248        recorded_upper_hz - recorded_lower_hz,
249    );
250    let field_id = survey.first_visibility_field_id;
251    let spw_id = survey.spectral_windows[0].spw_id;
252    let image_rest_frequency_hz = match (request.restfreq.first(), msv2) {
253        (Some(&hertz), _) => hertz,
254        (None, Some(ms)) => {
255            continuum_rest_frequency_hz(ms, field_id, spw_id, recorded.midpoint_hz())?
256        }
257        (None, None) => catalogue
258            .source_records(
259                FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id")),
260                WindowId(u32::try_from(spw_id).expect("a window id fits u32")),
261            )
262            .first()
263            .and_then(|record| record.rest_frequencies.as_ref()?.first())
264            .map(casa_types::measures::frequency::MFrequency::hz)
265            .filter(|value| *value != 0.0)
266            .unwrap_or(recorded.midpoint_hz()),
267    };
268    Ok(AxisLaw {
269        selected_source_channels: continuum_channels(channels),
270        window_reads: vec![WindowReads::new(); channels.len()],
271        output_frequency_reference: FrequencyRef::LSRK,
272        reference_frequency_hz,
273        increment_hz,
274        output_channels: 1,
275        rest_frequency: RestFrequency::NotApplicable,
276        image: ImageSpectralCoordinate {
277            frequency_reference: request.outframe,
278            reference_frequency_hz: image_reference_hz,
279            increment_hz: image_increment_hz,
280            rest_frequency_hz: image_rest_frequency_hz,
281            additional_rest_frequencies_hz: Vec::new(),
282            undefined_frame: false,
283        },
284        doppler: DopplerConvention::NotApplicable,
285        sampling: SpectralSamplingLaw::IDENTITY,
286        basis: ReconstructionBasis::Constant,
287    })
288}

290/// CASA `SynthesisParamsImage::buildCoordinateSystemCore` asks
291/// `MSDopplerUtil::dopplerInfo` for the first selected SPW and field. Its
292/// first rest frequency wins; an absent or zero value uses the MFS band's
293/// midpoint. This header label does not change the continuum/Taylor basis.
294fn continuum_rest_frequency_hz(
295    ms: &MeasurementSet,
296    field_id: usize,
297    spw_id: usize,
298    band_midpoint_hz: f64,
299) -> Result<f64, PrepareError> {
300    let source_id = ms.field()?.source_id(field_id)?;
301    let window = ms
302        .subtable(SubtableId::SpectralWindow)
303        .expect("required SPW table");
304    let doppler_id = if window
305        .schema()
306        .is_some_and(|schema| schema.contains_column("DOPPLER_ID"))
307    {
308        let scalar = window
309            .cell_accessor(spw_id, "DOPPLER_ID")
310            .map_err(casa_ms::MsError::from)?
311            .scalar()
312            .map_err(casa_ms::MsError::from)?;
313        match scalar {
314            casa_types::ScalarValue::Int32(value) => *value,
315            _ => {
316                return Err(
317                    casa_ms::MsError::InvalidInput("DOPPLER_ID must be Int32".into()).into(),
318                );
319            }
320        }
321    } else {
322        -1
323    };
324    if doppler_id >= 0 && ms.subtable(SubtableId::Doppler).is_some() {
325        return Err(casa_ms::MsError::InvalidInput(
326            "MFS rest-frequency metadata with a DOPPLER table is not supported".into(),
327        )
328        .into());
329    }
330    if ms.subtable(SubtableId::Source).is_none() {
331        return Ok(band_midpoint_hz);
332    }
333    let resolved = continuum_source_line(ms, source_id, spw_id)?;
334    Ok(resolved
335        .filter(|value| *value != 0.0)
336        .unwrap_or(band_midpoint_hz))
337}

339/// The first SOURCE line within a matching key group; -1 is a wildcard.
340/// CASA binary-searches sorted unique keys using its wildcard comparator.
341/// With multiple key groups, wildcards can select or hide a matching group;
342/// refuse that unresolved case instead of merging physical row order.
343fn continuum_source_line(
344    ms: &MeasurementSet,
345    source_id: i32,
346    spw_id: usize,
347) -> Result<Option<f64>, PrepareError> {
348    let source = ms.source()?;
349    let spw_id = i32::try_from(spw_id).expect("SPW ids are stored i32 values");
350    let mut first_key = None;
351    let mut multiple_keys = false;
352    let mut wildcard = false;
353    let mut matching_key = None;
354    let mut resolved = None;
355    for row in 0..source.row_count() {
356        let row_source = source.i32(row, "SOURCE_ID")?;
357        let row_spw = source.i32(row, "SPECTRAL_WINDOW_ID")?;
358        let key = (row_source, row_spw);
359        multiple_keys |= first_key.is_some_and(|prior| prior != key);
360        first_key.get_or_insert(key);
361        wildcard |= row_source == -1 || row_spw == -1;
362        if (row_source != source_id && row_source != -1) || (row_spw != spw_id && row_spw != -1) {
363            continue;
364        }
365        if matching_key.is_some_and(|prior| prior != key) {
366            return Err(casa_ms::MsError::InvalidInput(
367                "overlapping SOURCE key groups for MFS rest-frequency metadata are not supported"
368                    .into(),
369            )
370            .into());
371        }
372        matching_key = Some(key);
373        if resolved.is_some() {
374            continue;
375        }
376        let Some(ArrayValue::Float64(values)) = source.optional_array(row, "REST_FREQUENCY")?
377        else {
378            continue;
379        };
380        if values.is_empty() {
381            continue;
382        }
383        resolved = values.first().copied();
384    }
385    if wildcard && multiple_keys {
386        return Err(casa_ms::MsError::InvalidInput(
387            "multiple SOURCE key groups with wildcard ids for MFS rest-frequency metadata are not supported".into(),
388        ).into());
389    }
390    Ok(resolved)
391}

393/// Whether CASA tunes a `cube`'s frequency selection to the image's band
394/// (`SynthesisImagerVi2::tuneSelectData`): when the window of its first
395/// visibility buffer has 30 or more channels, or when the model is saved to
396/// `MODEL_DATA`, `SynthesisImager::tuneSelectData` narrows each window's
397/// bands to the LSRK frequencies from pixel −1.5 to `nchan + 0.5` of the
398/// image axis ([`frequency_bands::cube_bands`]); a `cubedata` or
399/// `cubesource` cube keeps the selection. The tuned band keeps every
400/// channel a plane is interpolated from, since CASA refuses planes
401/// narrower than a channel (`calcChanFreqs`), so tuning changes which
402/// channels are read but not the image: a window the axis misses is still
403/// read (its bands are kept where the image's overlaps none of them), and
404/// the density `VisImagingWeight` forms is the untuned selection's.
405/// casa-rs's visibility-domain continuum fit (`fitspw`, which CASA's
406/// tclean lacks) keeps the named channels as its output channels.
407fn tunes_selection(request: &ImagingRequest, first_window: &SourceSpectralWindow) -> bool {
408    request.specmode == SpecMode::Cube
409        && request.fitspw.is_none()
410        && (first_window.frequencies_hz.len() >= 30 || request.savemodel == SaveModel::ModelColumn)
411}

413/// CASA's cube axis over the reference store's selected windows
414/// (`buildCoordinateSystemCore` on `msobj`'s windows): one window's
415/// channels, or several windows' merged into one axis
416/// ([`casa_ms::combine_spectral_windows`]), with `axis`'s grid laid over
417/// them; and the band the `spw` selection spans over `band_surveys`, in each
418/// one's data frame ([`casa_ms::casa_selected_band_hz`]; CASA's `gdatafstart`
419/// and `gdatafend`). Every window's samples are
420/// gridded onto the one axis (`FTMachine::matchChannel`).
421fn cube_setup(
422    request: &ImagingRequest,
423    (survey, band_surveys): (&Survey, &[&Survey]),
424    frame: &FrameContext<'_>,
425    mut axis: CubeAxisConfig,
426) -> Result<(CubeSpectralSetup, [f64; 2]), PrepareError> {
427    fn selected<'w>(
428        survey: &'w Survey,
429        window: &'w SourceSpectralWindow,
430    ) -> (&'w [f64], &'w [f64], Option<&'w [usize]>) {
431        (
432            window.frequencies_hz.as_slice(),
433            window.channel_widths_hz.as_slice(),
434            survey.spw_channels.get(&window.spw_id).map(Vec::as_slice),
435        )
436    }
437    let windows = &survey.spectral_windows;
438    // Name the window whose selection spans no band.
439    if let Some(window) = band_surveys
440        .iter()
441        .flat_map(|survey| {
442            survey
443                .spectral_windows
444                .iter()
445                .map(move |window| (*survey, window))
446        })
447        .find(|(survey, window)| {
448            casa_ms::casa_selected_band_hz([selected(survey, window)]).is_none()
449        })
450        .map(|(_, window)| window)
451    {
452        return Err(PrepareError::SpectralWindowCoordinates {
453            spw_id: window.spw_id,
454        });
455    }
456    // Over every MeasurementSet (`buildCoordinateSystem`'s `gdatafstart` and
457    // `gdatafend`), though the axis lies on `survey`'s channels.
458    let band_hz = casa_ms::casa_selected_band_hz(band_surveys.iter().flat_map(|survey| {
459        survey
460            .spectral_windows
461            .iter()
462            .map(move |window| selected(survey, window))
463    }))
464    .ok_or(PrepareError::NoSelectedSamples)?;
465    axis.selected_band_hz = Some(band_hz);
466    let (frequencies_hz, widths_hz) = match windows.as_slice() {
467        [window] => (
468            window.frequencies_hz.clone(),
469            window.channel_widths_hz.clone(),
470        ),
471        _ => casa_ms::combine_spectral_windows(
472            &windows
473                .iter()
474                .map(|window| {
475                    (
476                        window.frequencies_hz.as_slice(),
477                        window.channel_widths_hz.as_slice(),
478                    )
479                })
480                .collect::<Vec<_>>(),
481        )?,
482    };
483    let setup = CubeSpectralSetup::for_casa_cube_axis(
484        survey.source_frequency_reference,
485        &frequencies_hz,
486        &widths_hz,
487        request.cube_output_channel_count(),
488        &axis,
489        frame.anchor_time_mjd_seconds,
490        frame.field_id,
491        Some(frame.phase.clone()),
492        frame.engine,
493    )?;
494    Ok((setup, band_hz))
495}

497/// What a cube on `setup` reads of each store, in store order: the
498/// channels of each window, and the bands of the windows whose times read
499/// only some of them ([`store_reads`]). Whether CASA tunes the selection
500/// to the image's band is decided once, from the iterator's first buffer,
501/// the first store's ([`tunes_selection`]), and applies to every store
502/// (`SynthesisImager::tuneSelectData`).
503///
504/// The reference store's windows are those the axis is laid on; another
505/// store's one window is read where it reaches the axis, as CASA's cube
506/// gridding selects each MeasurementSet's channels in the image's band.
507///
508/// # Errors
509///
510/// An axis none of the reference store's windows reaches; another store
511/// of several windows, or of another source frame than the reference
512/// store's; a frame conversion that fails.
513fn cube_store_reads(
514    request: &ImagingRequest,
515    (stores, reference): (&[SurveyedStore], usize),
516    frame: &FrameContext<'_>,
517    setup: &CubeSpectralSetup,
518) -> Result<(Vec<WindowChannels>, Vec<WindowReads>), PrepareError> {
519    let first = &stores[0].survey;
520    let first_window = first
521        .spectral_windows
522        .iter()
523        .find(|window| window.spw_id == first.first_visibility_spw_id)
524        .expect("the first visibility is in a surveyed window");
525    let tuned = tunes_selection(request, first_window);
526    let image_band = image_band(frame, setup, tuned)?;
527    let mut channels = Vec::with_capacity(stores.len());
528    let mut bands = Vec::with_capacity(stores.len());
529    for (index, store) in stores.iter().enumerate() {
530        let survey = &store.survey;
531        let field = if index == reference {
532            frame.field_id
533        } else {
534            one_window(&survey.spectral_windows, "native cube imaging")?;
535            if survey.source_frequency_reference != setup.source_freq_ref {
536                return Err(PrepareError::MixedSourceFrames);
537            }
538            usize::try_from(*survey.fields.first().expect("a surveyed store's field"))
539                .expect("a surveyed field is a FIELD row")
540        };
541        let reads = store_reads(
542            request,
543            ((store.catalogue(), &store.engine), survey, field),
544            setup,
545            (image_band, tuned),
546        )?;
547        if index == reference && !reads.reaches_axis {
548            return Err(PrepareError::NoCommonCubeChannels);
549        }
550        channels.push(reads.channels);
551        bands.push(reads.windows);
552    }
553    Ok((channels, bands))
554}

568/// frequency bands, tuned to the image's band `image_band` where CASA
569/// tunes the selection ([`tunes_selection`]), select at some epoch of its
570/// samples ([`cube_bands`]); a `fitspw` fit, which has no bands, reads
571/// those `spw` names. Where the bands do not
572/// select every channel read at every epoch, the window carries them and
573/// each time reads only the channels they select then, as CASA's iterator
574/// does.
575///
576/// A window none of whose channels reaches the axis is read as `spw`
577/// selects it, tuned or not, as CASA does, by the two passes that need it:
578/// its samples fall outside every plane, but `VisImagingWeight`'s density
579/// counts them, since CASA forms it over the untuned selection of the imager
580/// that defines the cube, and a final pass writes the cube model's
581/// prediction there, zero, into its `MODEL_DATA` (test_modelvis_29). The
582/// other passes skip it where their source confirms this test, at a few
583/// epochs and over the whole window's channels, for its rows
584/// ([`WindowRead::density_only`]).
585fn store_reads(
586    request: &ImagingRequest,
587    ((catalogue, engine), survey, field): ((&Catalogue, &MsCalEngine), &Survey, usize),
588    setup: &CubeSpectralSetup,
589    (image_band, tuned): (Option<[f64; 2]>, bool),
590) -> Result<StoreReads, PrepareError> {
591    let bands = cube_bands(request, (catalogue, engine), survey, image_band)?;
592    let density_bands = if whole_selection_density(request) {
593        cube_bands(request, (catalogue, engine), survey, None)?
594    } else {
595        None
596    };
597    let prediction_bands =
598        if request.savemodel == SaveModel::ModelColumn && request.predicts_final_model() {
599            cube_bands(request, (catalogue, engine), survey, None)?
600        } else {
601            None
602        };
603    let predicts = request.savemodel == SaveModel::ModelColumn && request.predicts_final_model();
604    let mut reads = StoreReads {
605        channels: WindowChannels::new(),
606        windows: WindowReads::new(),
607        reaches_axis: false,
608    };
609    for window in &survey.spectral_windows {
610        let mut channels = setup
611            .source_channel_support(
612                &window.frequencies_hz,
613                &window.channel_widths_hz,
614                survey.time_bounds_mjd_seconds,
615                field,
616                engine,
617            )?
618            .indices;
619        let selection = match &bands {
620            Some(bands) => bands.epoch_selection((catalogue, engine), survey, window)?,
621            None => None,
622        };
623        let explicit = survey.spw_channels.get(&window.spw_id);
624        match (&selection, explicit) {
625            (Some(selection), _) => channels.retain(|channel| selection.ever(*channel)),
626            (None, Some(explicit)) if !tuned => {
627                channels.retain(|channel| explicit.contains(channel));
628            }
629            _ => {}
630        }
631        if channels.is_empty() {
632            // `spw`'s channels, or the whole window when it names none.
633            channels = explicit
634                .cloned()
635                .unwrap_or_else(|| (0..window.frequencies_hz.len()).collect());
636            reads.windows.entry(window.spw_id).or_default().density_only = true;
637        } else {
638            reads.reaches_axis = true;
639            if let (Some(bands), Some(selection)) = (&bands, &selection)
640                && channels.iter().any(|channel| !selection.always(*channel))
641                && let Some(window_bands) = bands.frequency_bands(window.spw_id)
642            {
643                reads.windows.entry(window.spw_id).or_default().bands = Some(window_bands.clone());
644            }
645        }
646        if let Some(density) = &density_bands
647            && let Some(read) =
648                window_density((catalogue, engine), survey, window, density, &channels)?
649        {
650            reads.windows.entry(window.spw_id).or_default().density = Some(read);
651        }
652        if predicts {
653            let prediction = match &prediction_bands {
654                Some(bands) => {
655                    let selection = bands.epoch_selection((catalogue, engine), survey, window)?;
656                    match selection {
657                        Some(selection) => {
658                            let read = (0..window.frequencies_hz.len())
659                                .filter(|channel| selection.ever(*channel))
660                                .collect::<Vec<_>>();
661                            let by_time = (!read.iter().all(|channel| selection.always(*channel)))
662                                .then(|| bands.frequency_bands(window.spw_id).cloned())
663                                .flatten();
664                            WindowDensity {
665                                channels: read,
666                                bands: by_time,
667                            }
668                        }
669                        None => WindowDensity {
670                            channels: Vec::new(),
671                            bands: None,
672                        },
673                    }
674                }
675                None => WindowDensity {
676                    channels: explicit
677                        .cloned()
678                        .unwrap_or_else(|| (0..window.frequencies_hz.len()).collect()),
679                    bands: None,
680                },
681            };
682            reads.windows.entry(window.spw_id).or_default().prediction = Some(prediction);
683        }
684        reads.channels.insert(window.spw_id, channels);
685    }
686    Ok(reads)
687}

689/// What a density over the whole selection counts of `window`, where that
690/// isn't simply `channels`: the channels `bands`, untuned, select at some
691/// time, so a range with a stride whole and a time's channels moving with its
692/// Doppler shift; None for a window without bands.
693fn window_density(
694    (catalogue, engine): (&Catalogue, &MsCalEngine),
695    survey: &Survey,
696    window: &SourceSpectralWindow,
697    bands: &frequency_bands::CubeBands,
698    channels: &[usize],
699) -> Result<Option<WindowDensity>, PrepareError> {
700    let Some(selection) = bands.epoch_selection((catalogue, engine), survey, window)? else {
701        return Ok(None);
702    };
703    let read = (0..window.frequencies_hz.len())
704        .filter(|channel| selection.ever(*channel))
705        .collect::<Vec<_>>();
706    let by_time = (!read.iter().all(|channel| selection.always(*channel)))
707        .then(|| bands.frequency_bands(window.spw_id).cloned())
708        .flatten();
709    Ok(
710        (read != channels || by_time.is_some()).then_some(WindowDensity {
711            channels: read,
712            bands: by_time,
713        }),
714    )
715}

717/// Whether a cube's weights divide by one density over the whole selection:
718/// `SynthesisImagerVi2::weight` forms `VisImagingWeight` on the parent
719/// imager's iterator before it tunes the selection, which at each time reads
720/// what the bands of `spw` select whatever the axis reaches, unless
721/// `perchanweightdensity` asks for a density per output channel. `fitspw`,
722/// which CASA's tclean lacks, keeps its named channels.
723fn whole_selection_density(request: &ImagingRequest) -> bool {
724    request.fitspw.is_none()
725        && request.weighting.uses_density()
726        && !(request.perchanweightdensity && request.weighting.per_channel_density())
727}

748/// The bands a cube's gridding selects one store's windows' channels by at
749/// each time ([`frequency_bands::cube_bands`]): a `cube`'s in `outframe`,
750/// or tuned to the image's band `image_band` where CASA tunes the
751/// selection; in the data's frame where tclean's selection frame is
752/// `Undefined` (`cubedata`) or `REST` (`cubesource`, or a `cube` whose
753/// `outframe` is `REST`), which select each named range's channels at
754/// every time, a range with a stride whole. None for a `fitspw` fit,
755/// which reads its channels as named.
756fn cube_bands(
757    request: &ImagingRequest,
758    (catalogue, engine): (&Catalogue, &MsCalEngine),
759    survey: &Survey,
760    image_band: Option<[f64; 2]>,
761) -> Result<Option<frequency_bands::CubeBands>, PrepareError> {
762    use frequency_bands::SelectionFrame;
763    if request.fitspw.is_some() {
764        return Ok(None);
765    }
766    let frame = match request.specmode {
767        SpecMode::Cube if image_band.is_some() => SelectionFrame::Converted(FrequencyRef::LSRK),
768        SpecMode::Cube if request.outframe != FrequencyRef::REST => {
769            SelectionFrame::Converted(request.outframe)
770        }
771        SpecMode::Mfs | SpecMode::Cube | SpecMode::Cubedata | SpecMode::Cubesource => {
772            SelectionFrame::Data
773        }
774    };
775    frequency_bands::cube_bands((catalogue, engine), survey, frame, image_band).map(Some)
776}

778/// A cube in its output frame (`cube`) or the data's (`cubedata`), laid
779/// on the windows of store `reference` (CASA's `msobj`), reading each
780/// store's channels where they reach it ([`cube_store_reads`]).
781fn cube_law(
782    request: &ImagingRequest,
783    (stores, reference): (&[SurveyedStore], usize),
784    frame: &FrameContext<'_>,
785) -> Result<AxisLaw<'static>, PrepareError> {
786    let store = &stores[reference];
787    let (catalogue, survey) = (store.catalogue(), &store.survey);
788    let band_surveys = stores.iter().map(|store| &store.survey).collect::<Vec<_>>();
789    let (setup, band_hz) =
790        cube_setup(request, (survey, &band_surveys), frame, cube_axis(request)?)?;
791    let (selected_source_channels, window_reads) =
792        cube_store_reads(request, (stores, reference), frame, &setup)?;
793    let output_channels = setup.output_channel_frequencies_hz.len();
794    let (reference_frequency_hz, increment_hz) = first_and_increment(&setup, output_channels, 1.0)?;
795    let source_rest_hz =
796        source_rest_frequency(catalogue, frame.field_id, &survey.spectral_windows)?;
797    let (resolved_rest_hz, image_rest_frequency_hz) =
798        cube_rest_frequency_hz(request.restfreq.first().copied(), source_rest_hz, band_hz);
799    let (rest_frequency, doppler) = match resolved_rest_hz {
800        None => (
801            RestFrequency::NotApplicable,
802            DopplerConvention::NotApplicable,
803        ),
804        Some(hertz) => (
805            RestFrequency::Line { hertz },
806            doppler_convention(request.veltype)?,
807        ),
808    };
809    Ok(AxisLaw {
810        selected_source_channels: Cow::Owned(selected_source_channels),
811        window_reads,
812        output_frequency_reference: setup.output_freq_ref,
813        reference_frequency_hz,
814        increment_hz,
815        output_channels,
816        rest_frequency,
817        image: ImageSpectralCoordinate {
818            frequency_reference: setup.output_freq_ref,
819            reference_frequency_hz,
820            increment_hz,
821            rest_frequency_hz: image_rest_frequency_hz,
822            additional_rest_frequencies_hz: Vec::new(),
823            undefined_frame: request.specmode == SpecMode::Cubedata
824                && setup.output_freq_ref != FrequencyRef::REST,
825        },
826        doppler,
827        sampling: sampling_law(setup.interpolation),
828        basis: ReconstructionBasis::ChannelLocal {
829            channels: output_channels,
830        },
831    })
832}

834/// A cube in the rest frame of a moving source: the data-frame axis
835/// scaled into the source's rest frame (`cubesource`), on `store`'s one
836/// window, reading its channels as [`store_reads`] reads a `cubedata`
837/// cube's.
838///
839/// # Errors
840///
841/// Several selected windows: CASA's source-frame cube over several
842/// windows is not yet ported (#1059).
843fn source_frame_cube_law(
844    request: &ImagingRequest,
845    store: &SurveyedStore,
846    frame: &FrameContext<'_>,
847    moving_rest_frame: &MeasFrame,
848) -> Result<AxisLaw<'static>, PrepareError> {
849    let (catalogue, survey) = (store.catalogue(), &store.survey);
850    if survey.spectral_windows.len() > 1 {
851        return Err(PrepareError::SourceFrameCubeWindows {
852            selected: survey.spectral_windows.len(),
853        });
854    }
855    let source_reference = survey.source_frequency_reference;
856    let axis = CubeAxisConfig {
857        specmode: CubeSpecMode::Cubedata,
858        outframe: source_reference,
859        ..cube_axis(request)?
860    };
861    let (setup, _) = cube_setup(request, (survey, &[survey]), frame, axis)?;
862    let reads = store_reads(
863        request,
864        ((catalogue, &store.engine), survey, frame.field_id),
865        &setup,
866        (None, false),
867    )?;
868    if !reads.reaches_axis {
869        return Err(PrepareError::NoCommonCubeChannels);
870    }
871    let output_channels = setup.output_channel_frequencies_hz.len();
872    let factor = casa_ms::convert_frequency_to_frame_with_frame(
873        source_reference,
874        FrequencyRef::REST,
875        1.0,
876        Some(moving_rest_frame),
877    )?;
878    let (reference_frequency_hz, increment_hz) =
879        first_and_increment(&setup, output_channels, factor)?;
880    let rest_frequency_hz = request
881        .restfreq
882        .first()
883        .copied()
884        .or(source_rest_frequency(
885            catalogue,
886            frame.field_id,
887            &survey.spectral_windows,
888        )?)
889        .ok_or(PrepareError::RestFrequencyMissing)?;
890    Ok(AxisLaw {
891        selected_source_channels: Cow::Owned(vec![reads.channels]),
892        window_reads: vec![reads.windows],
893        output_frequency_reference: FrequencyRef::REST,
894        reference_frequency_hz,
895        increment_hz,
896        output_channels,
897        rest_frequency: RestFrequency::Line {
898            hertz: rest_frequency_hz,
899        },
900        image: ImageSpectralCoordinate {
901            frequency_reference: FrequencyRef::REST,
902            reference_frequency_hz,
903            increment_hz,
904            rest_frequency_hz,
905            additional_rest_frequencies_hz: Vec::new(),
906            undefined_frame: false,
907        },
908        doppler: doppler_convention(request.veltype)?,
909        sampling: sampling_law(request.interpolation),
910        basis: ReconstructionBasis::ChannelLocal {
911            channels: output_channels,
912        },
913    })
914}

916/// CASA's cube axis of the request: `start` (else `channel_start`) and
917/// `width` in the request's velocity convention; [`cube_setup`] sets the
918/// band `spw` selects.
919fn cube_axis(request: &ImagingRequest) -> Result<CubeAxisConfig, PrepareError> {
920    let start = match (&request.start, request.channel_start) {
921        (Some(start), _) => Some(CubeAxisValue::parse(start, request.veltype)?),
922        (None, Some(channel)) => Some(CubeAxisValue::Channel(
923            i32::try_from(channel).map_err(|_| PrepareError::CubeStartChannel { channel })?,
924        )),
925        (None, None) => None,
926    };
927    let width = request
928        .width
929        .as_deref()
930        .map(|width| CubeAxisValue::parse(width, request.veltype))
931        .transpose()?;
932    Ok(CubeAxisConfig {
933        specmode: if request.specmode == SpecMode::Cubedata {
934            CubeSpecMode::Cubedata
935        } else {
936            CubeSpecMode::Cube
937        },
938        outframe: request.outframe,
939        veltype: request.veltype,
940        interpolation: request.interpolation,
941        rest_frequency_hz: request.restfreq.first().copied(),
942        start,
943        width,
944        selected_band_hz: None,
945    })
946}

948/// The first output channel's centre and the channel increment, scaled by
949/// `factor`; a one-channel cube's increment is its width.
950fn first_and_increment(
951    setup: &CubeSpectralSetup,
952    output_channels: usize,
953    factor: f64,
954) -> Result<(f64, f64), PrepareError> {
955    let first_hz = setup.output_channel_frequencies_hz[0];
956    let increment_hz = if output_channels > 1 {
957        setup.output_channel_frequencies_hz[1] - first_hz
958    } else {
959        setup.output_channel_widths_hz[0]
960    } * factor;
961    if !increment_hz.is_finite() || increment_hz == 0.0 {
962        return Err(PrepareError::CubeIncrement { increment_hz });
963    }
964    Ok((first_hz * factor, increment_hz))
965}

967/// The model's Doppler convention of CASA `veltype`.
968fn doppler_convention(veltype: DopplerRef) -> Result<DopplerConvention, PrepareError> {
969    match veltype {
970        DopplerRef::RADIO => Ok(DopplerConvention::Radio),
971        DopplerRef::Z => Ok(DopplerConvention::Optical),
972        DopplerRef::BETA => Ok(DopplerConvention::Relativistic),
973        DopplerRef::RATIO | DopplerRef::GAMMA => Err(PrepareError::UnsupportedDoppler {
974            convention: veltype,
975        }),
976    }
977}

987/// The image's epoch, reference direction and observatory, where the
988/// rows' frame and the output frame differ, or where a `cube` is gridded
989/// through LSRK ([`FrequencyFrame::gridded_through_lsrk`]): CASA grids a
990/// `cube` (not `cubedata`, whose rows keep their frequencies) in LSRK
991/// whatever its `outframe`, so a TOPO cube of TOPO rows is fixed in LSRK at
992/// the first time and each row is Doppler tracked from it.
993fn spectral_frame_anchor(
994    source_frame: FrequencyFrame,
995    output_frame: FrequencyFrame,
996    specmode: SpecMode,
997    frame: &FrameContext<'_>,
998) -> Result<SpectralFrameAnchor, PrepareError> {
999    let tracked = specmode == SpecMode::Cube && output_frame.gridded_through_lsrk();
1000    if source_frame == output_frame && !tracked {
1001        return Ok(SpectralFrameAnchor::NotApplicable);
1002    }
1003    let [x_metres, y_metres, z_metres] = frame.engine.observatory_position().as_itrf();
1004    Ok(SpectralFrameAnchor::Conversion {
1005        // As `MEpoch::from_mjd_seconds` reads a row's time, so a row at the
1006        // first time is at the anchor's epoch exactly.
1007        epoch: Epoch::new(
1008            frame.anchor_time_mjd_seconds * (1.0 / 86_400.0),
1009            imaging_time_scale(frame.engine.time_reference())?,
1010        ),
1011        direction: frame.direction.reference_direction(),
1012        observatory_position: ItrfPosition::new(x_metres, y_metres, z_metres),
1013    })
1014}

1038/// The rest frequency the field's source states over the selected windows
1039/// (MSv2 `SOURCE::REST_FREQUENCY`, MSv4 `LINE_REST_FREQUENCY`), if any: the
1040/// first line of each record for a selected window or for every window
1041/// ([`Catalogue::source_records`]), which must agree.
1042fn source_rest_frequency(
1043    catalogue: &Catalogue,
1044    field_id: usize,
1045    spectral_windows: &[SourceSpectralWindow],
1046) -> Result<Option<f64>, PrepareError> {
1047    let field = FieldId(u32::try_from(field_id).expect("a FIELD row is an u32 id"));
1048    let mut resolved = None;
1049    for window in spectral_windows {
1050        let window = WindowId(u32::try_from(window.spw_id).expect("a window id fits u32"));
1051        for record in catalogue.source_records(field, window) {
1052            let Some(value) = record
1053                .rest_frequencies
1054                .as_ref()
1055                .and_then(|lines| lines.first())
1056                .map(casa_types::measures::frequency::MFrequency::hz)
1057                .filter(|value| *value > 0.0)
1058            else {
1059                continue;
1060            };
1061            if resolved.is_some_and(|prior: f64| prior.to_bits() != value.to_bits()) {
1062                return Err(PrepareError::RestFrequencyConflict);
1063            }
1064            resolved = Some(value);
1065        }
1066    }
1067    Ok(resolved)
1068}

1070/// CASA's precedence for a cube's rest frequency: the request's, else the
1071/// source's; the header records the centre of the selected band `band_hz`
1072/// when neither exists (`buildCoordinateSystemCore`'s provisional rest
1073/// frequency, `(datafend + datafstart)/2`).
1074fn cube_rest_frequency_hz(
1075    explicit_hz: Option<f64>,
1076    source_hz: Option<f64>,
1077    band_hz: [f64; 2],
1078) -> (Option<f64>, f64) {
1079    let resolved = explicit_hz.or(source_hz);
1080    (
1081        resolved,
1082        resolved.unwrap_or((band_hz[1] + band_hz[0]) / 2.0),
1083    )
1084}
