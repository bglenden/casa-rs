3//! Thin PyO3 `ia.brightnessunit` / `ia.coordsys` / `ia.history` wrappers delegating to [`casa_imageanalysis::tool::metadata`].
4
5use casa_imageanalysis::tool::metadata::{
6    brightness_unit, coordsys_for_axes, image_history, set_brightness_unit, set_coordinate_record,
7};
8use pyo3::prelude::*;
9use pyo3::types::PyDict;
10
11use crate::convert::IntList;
12use crate::tools::coordsys::PyCoordsys;
13use crate::tools::table::casa_record;
14
15use super::PyImageTool;
16use super::handle::raised;
17
18#[pymethods]
19impl PyImageTool {
20    fn brightnessunit(&self, py: Python<'_>) -> String {
21        py.detach(|| {
22            self.state()
23                .image
24                .as_ref()
25                .map_or_else(String::new, brightness_unit)
26        })
27    }
28
29    fn setbrightnessunit(&self, py: Python<'_>, unit: String) -> PyResult<bool> {
30        py.detach(|| {
31            let state = self.state();
32            let Some(attachment) = state.image.as_ref() else {
33                return Ok(false);
34            };
35            set_brightness_unit(attachment, &unit, state.history).map_err(raised)
36        })
37    }
38
39    #[pyo3(signature = (axes=None), text_signature = "($self, axes=[-1])")]
40    fn coordsys(&self, py: Python<'_>, axes: Option<IntList>) -> PyResult<Py<PyAny>> {
41        let tool = py.detach(|| {
42            let state = self.state();
43            let Some(attachment) = state.image.as_ref() else {
44                return Ok(PyCoordsys::empty());
45            };
46            let axes = axes.as_ref().map_or(&[][..], |arg| arg.0.as_slice());
47            let (coordinates, record) = coordsys_for_axes(attachment, axes).map_err(raised)?;
48            PyCoordsys::from_image_coordinates(&coordinates, record)
49        })?;
50        Ok(Py::new(py, tool)?.into_any())
51    }
52
53    fn setcoordsys(&self, py: Python<'_>, csys: &Bound<'_, PyDict>) -> PyResult<bool> {
54        let record = casa_record(py, Some(csys))?;
55        py.detach(|| {
56            let state = self.state();
57            let Some(attachment) = state.image.as_ref() else {
58                return Ok(false);
59            };
60            set_coordinate_record(attachment, record, state.history).map_err(raised)
61        })
62    }
63
64    #[pyo3(signature = (list=true), text_signature = "($self, list=True)")]
65    fn history(&self, py: Python<'_>, list: bool) -> PyResult<Vec<String>> {
66        let entries = py.detach(|| {
67            let state = self.state();
68            let Some(attachment) = state.image.as_ref() else {
69                return Ok(Vec::new());
70            };
71            image_history(attachment).map_err(raised)
72        })?;
73        if list {
74            crate::tools::emit(
75                py,
76                "casars.tools.image",
77                &entries
78                    .iter()
79                    .cloned()
80                    .map(crate::tools::LogLine::info)
81                    .collect::<Vec<_>>(),
82            )?;
83        }
84        Ok(entries)
85    }
86}
