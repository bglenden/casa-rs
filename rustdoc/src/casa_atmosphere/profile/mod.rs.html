4//! The layered atmosphere above a site, built from ground weather.
5//!
6//! ATM describes the atmosphere as a stack of horizontal layers. Each has a
7//! thickness and mean pressure, temperature, water-vapour density and number
8//! densities of the minor gases. [`AtmosphereProfile::new`] builds the stack
9//! the way ATM's `AtmProfile` does from the ground pressure, temperature and
10//! relative humidity at the site:
11//!
12//! - In the troposphere the pressure falls in steps that grow by a constant
13//!   factor (ATM fixes it at 1.075), the height of each step follows from the
14//!   hypsometric equation with height-dependent gravity, the temperature
15//!   follows the tropospheric lapse rate, and the water-vapour density falls
16//!   exponentially with the water-vapour scale height.
17//! - Above the tropopause the levels come from the reference atmosphere
18//!   chosen by [`AtmosphereType`].
19//! - The minor gases (O3, N2O, CO, NO2, SO2, HCl, HCN) are interpolated from
20//!   the US Standard 1976 family of reference profiles at each layer's mid
21//!   height, with ATM's empirical corrections for Chajnantor.
22//!
23//! Each layer's mean temperature is the mean of its boundary temperatures;
24//! its mean pressure and water-vapour density are geometric means. The
25//! column of precipitable water is the sum of density times thickness
26//! ([`AtmosphereProfile::ground_water_column_m`]).

35/// ATM's reference atmospheres, which set the profile above the tropopause
36/// and the minor-gas abundances.
37///
38/// The discriminants are ATM's `typeAtm` numbers. CASA's `at` tool accepts
39/// 1 to 5. The Chajnantor variants (11 to 14) are the tropical atmosphere
40/// with ATM's seasonal and diurnal ozone corrections for the ALMA site.
41#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
42#[repr(u32)]
43pub enum AtmosphereType {
44    /// Tropical (ATM's default for ALMA).
45    Tropical = 1,
46    /// Mid-latitude summer.
47    MidlatitudeSummer = 2,
48    /// Mid-latitude winter.
49    MidlatitudeWinter = 3,
50    /// Subarctic summer.
51    SubarcticSummer = 4,
52    /// Subarctic winter.
53    SubarcticWinter = 5,
54    /// US Standard Atmosphere 1976.
55    UsStandard1976 = 6,
56    /// Tropical, with the Chajnantor winter daytime ozone profile.
57    ChajnantorWinterDay = 11,
58    /// Tropical, with the Chajnantor winter night-time ozone profile.
59    ChajnantorWinterNight = 12,
60    /// Tropical, with the Chajnantor summer daytime ozone profile.
61    ChajnantorSummerDay = 13,
62    /// Tropical, with the Chajnantor summer night-time ozone profile.
63    ChajnantorSummerNight = 14,
64}
65
66impl AtmosphereType {
67    /// The type for ATM's `typeAtm` number, if ATM defines one.
68    #[must_use]
69    pub fn from_number(number: u32) -> Option<Self> {
70        Some(match number {
71            1 => Self::Tropical,
72            2 => Self::MidlatitudeSummer,
73            3 => Self::MidlatitudeWinter,
74            4 => Self::SubarcticSummer,
75            5 => Self::SubarcticWinter,
76            6 => Self::UsStandard1976,
77            11 => Self::ChajnantorWinterDay,
78            12 => Self::ChajnantorWinterNight,
79            13 => Self::ChajnantorSummerDay,
80            14 => Self::ChajnantorSummerNight,
81            _ => return None,
82        })
83    }
84
85    /// The reference profile (1 to 6), the seasonal/diurnal ozone model
86    /// (1 to 4), and the pressure in hPa below which the tropospheric steps
87    /// give way to the reference levels.
88    fn model(self) -> (usize, u32, f64) {
89        match self {
90            Self::Tropical => (1, 1, 125.0),
91            Self::ChajnantorWinterDay => (1, 1, 125.0),
92            Self::ChajnantorWinterNight => (1, 2, 125.0),
93            Self::ChajnantorSummerDay => (1, 3, 125.0),
94            Self::ChajnantorSummerNight => (1, 4, 125.0),
95            Self::MidlatitudeSummer => (2, 1, 100.0),
96            Self::MidlatitudeWinter => (3, 1, 100.0),
97            Self::SubarcticSummer => (4, 1, 100.0),
98            Self::SubarcticWinter => (5, 1, 100.0),
99            Self::UsStandard1976 => (6, 1, 100.0),
100        }
101    }
102}

104/// The ground weather and model choices an [`AtmosphereProfile`] is built
105/// from, in the units ATM's `AtmProfile` constructor takes.
106///
107/// [`Default`] gives CASA's `at.initAtmProfile` defaults, which are also the
108/// simulator's: the ALMA site at 5,000 m, 560 hPa, 270 K, 20% humidity.
109#[derive(Clone, Debug, PartialEq)]
110pub struct ProfileParameters {
111    /// Site altitude above sea level, metres.
112    pub altitude_m: f64,
113    /// Ground pressure, hPa (mbar).
114    pub ground_pressure_hpa: f64,
115    /// Ground temperature, kelvin.
116    pub ground_temperature_k: f64,
117    /// Tropospheric temperature lapse rate, K/km (negative when the
118    /// temperature falls with height).
119    pub lapse_rate_k_per_km: f64,
120    /// Ground relative humidity, percent.
121    pub relative_humidity_percent: f64,
122    /// Water-vapour scale height, kilometres.
123    pub water_vapour_scale_height_km: f64,
124    /// Primary pressure step, hPa. ATM replaces a step outside 5 to 12.5 hPa
125    /// with 10 hPa.
126    pub pressure_step_hpa: f64,
127    /// Top of the profile above sea level, kilometres.
128    pub top_altitude_km: f64,
129    /// The reference atmosphere.
130    pub atmosphere: AtmosphereType,
131}

149/// One horizontal layer of an [`AtmosphereProfile`].
150///
151/// Pressures are in hPa, temperatures in kelvin, water-vapour densities in
152/// kg/m³ and the minor gases in molecules/m³. "Bottom" and "top" are the
153/// layer's boundary values; the unqualified values are the layer means.
154#[derive(Clone, Debug, PartialEq)]
155pub struct Layer {
156    /// Thickness, metres.
157    pub thickness_m: f64,
158    /// Mean temperature.
159    pub temperature_k: f64,
160    /// Temperature at the bottom boundary.
161    pub bottom_temperature_k: f64,
162    /// Temperature at the top boundary.
163    pub top_temperature_k: f64,
164    /// Mean pressure (geometric mean of the boundaries).
165    pub pressure_hpa: f64,
166    /// Pressure at the bottom boundary.
167    pub bottom_pressure_hpa: f64,
168    /// Pressure at the top boundary.
169    pub top_pressure_hpa: f64,
170    /// Mean water-vapour mass density.
171    pub water_vapour_kg_m3: f64,
172    /// Water-vapour mass density at the bottom boundary.
173    pub bottom_water_vapour_kg_m3: f64,
174    /// Water-vapour mass density at the top boundary.
175    pub top_water_vapour_kg_m3: f64,
176    /// Ozone number density.
177    pub o3_per_m3: f64,
178    /// Carbon monoxide number density.
179    pub co_per_m3: f64,
180    /// Nitrous oxide number density.
181    pub n2o_per_m3: f64,
182    /// Nitrogen dioxide number density.
183    pub no2_per_m3: f64,
184    /// Sulphur dioxide number density.
185    pub so2_per_m3: f64,
186    /// Hydrogen chloride number density.
187    pub hcl_per_m3: f64,
188    /// Hydrogen cyanide number density.
189    pub hcn_per_m3: f64,
190}

192/// The layered atmosphere above a site (ATM's `AtmProfile`).
193#[derive(Clone, Debug, PartialEq)]
194pub struct AtmosphereProfile {
195    parameters: ProfileParameters,
196    layers: Vec<Layer>,
197    tropopause_layer: usize,
198    tropopause_temperature_k: f64,
199    tropopause_altitude_m: f64,
200}
201
202impl AtmosphereProfile {
203    /// Builds the profile for `parameters` as ATM's `AtmProfile` constructor
204    /// does.
205    ///
206    /// # Errors
207    ///
208    /// [`AtmosphereError::InvalidParameter`] if a parameter is not finite,
209    /// the ground pressure, temperature or scale height is not positive, or
210    /// the top of the profile is not above the site.
211    pub fn new(parameters: &ProfileParameters) -> Result<Self, AtmosphereError> {
212        let mut parameters = parameters.clone();
213        validate(&parameters)?;
214        if !(5.0..=12.5).contains(&parameters.pressure_step_hpa) {
215            parameters.pressure_step_hpa = 10.0;
216        }
217        let levels = Levels::build(&parameters);
218        Ok(levels.into_profile(parameters))
219    }
220
221    /// As [`AtmosphereProfile::new`], then replaces the mean temperature of
222    /// each layer that contains user-given levels with their mean, as ATM's
223    /// `AtmProfile` constructor with a user temperature profile does.
224    ///
225    /// `levels` are (altitude above sea level in metres, temperature in
226    /// kelvin). A level on a boundary between two layers counts in both.
227    ///
228    /// # Errors
229    ///
230    /// As [`AtmosphereProfile::new`], and
231    /// [`AtmosphereError::InvalidParameter`] if a level's altitude or
232    /// temperature is not finite, a temperature is not positive, or a
233    /// layer's mean overflows (#894 review: ATM takes them, and the profile
234    /// would carry them into every refractivity).
235    pub fn with_temperature_levels(
236        parameters: &ProfileParameters,
237        levels: &[(f64, f64)],
238    ) -> Result<Self, AtmosphereError> {
239        if let Some((altitude, temperature)) = levels.iter().find(|(altitude, temperature)| {
240            !(altitude.is_finite() && temperature.is_finite() && *temperature > 0.0)
241        }) {
242            return Err(AtmosphereError::InvalidParameter(format!(
243                "temperature level ({altitude} m, {temperature} K) is not a finite altitude and \
244                 a positive temperature"
245            )));
246        }
247        let mut profile = Self::new(parameters)?;
248        let mut bottom = profile.parameters.altitude_m;
249        for layer in &mut profile.layers {
250            let top = bottom + layer.thickness_m;
251            let (sum, count) = levels
252                .iter()
253                .filter(|(altitude, _)| bottom <= *altitude && top >= *altitude)
254                .fold((0.0, 0.0), |(sum, count), (_, temperature)| {
255                    (sum + temperature, count + 1.0)
256                });
257            if sum > 0.0 {
258                let mean = sum / count;
259                if !mean.is_finite() {
260                    return Err(AtmosphereError::InvalidParameter(format!(
261                        "the temperature levels between {bottom} m and {top} m overflow their \
262                         mean"
263                    )));
264                }
265                layer.temperature_k = mean;
266            }
267            bottom = top;
268        }
269        Ok(profile)
270    }

272    /// The parameters the profile was built from, after ATM's corrections
273    /// (an out-of-range pressure step becomes 10 hPa).
274    #[must_use]
275    pub fn parameters(&self) -> &ProfileParameters {
276        &self.parameters
277    }

279    /// The layers, from the ground up.
280    #[must_use]
281    pub fn layers(&self) -> &[Layer] {
282        &self.layers
283    }
284
285    /// The index of the last tropospheric layer.
286    #[must_use]
287    pub fn tropopause_layer(&self) -> usize {
288        self.tropopause_layer
289    }
290
291    /// The temperature at the tropopause, kelvin.
292    #[must_use]
293    pub fn tropopause_temperature_k(&self) -> f64 {
294        self.tropopause_temperature_k
295    }
296
297    /// The altitude of the tropopause above sea level, metres.
298    #[must_use]
299    pub fn tropopause_altitude_m(&self) -> f64 {
300        self.tropopause_altitude_m
301    }
302
303    /// The zenith column of precipitable water above the site, metres
304    /// (ATM's `getGroundWH2O`).
305    #[must_use]
306    pub fn ground_water_column_m(&self) -> f64 {
307        let column_mm = self.layers.iter().fold(0.0, |sum, layer| {
308            sum + layer.water_vapour_kg_m3 * layer.thickness_m
309        });
310        column_mm * 1e-3
311    }

313    /// Releases spare vector capacity, so the profile keeps exactly its
314    /// layers (what an absorption profile's memory estimate counts).
315    pub(crate) fn shrink_to_fit(&mut self) {
316        self.layers.shrink_to_fit();
317    }

320fn validate(parameters: &ProfileParameters) -> Result<(), AtmosphereError> {
321    let values = [
322        ("altitude_m", parameters.altitude_m),
323        ("ground_pressure_hpa", parameters.ground_pressure_hpa),
324        ("ground_temperature_k", parameters.ground_temperature_k),
325        ("lapse_rate_k_per_km", parameters.lapse_rate_k_per_km),
326        (
327            "relative_humidity_percent",
328            parameters.relative_humidity_percent,
329        ),
330        (
331            "water_vapour_scale_height_km",
332            parameters.water_vapour_scale_height_km,
333        ),
334        ("pressure_step_hpa", parameters.pressure_step_hpa),
335        ("top_altitude_km", parameters.top_altitude_km),
336    ];
337    if let Some((name, _)) = values.iter().find(|(_, value)| !value.is_finite()) {
338        return Err(AtmosphereError::InvalidParameter(format!(
339            "{name} is not finite"
340        )));
341    }
342    for (name, value) in [
343        ("ground_pressure_hpa", parameters.ground_pressure_hpa),
344        ("ground_temperature_k", parameters.ground_temperature_k),
345        (
346            "water_vapour_scale_height_km",
347            parameters.water_vapour_scale_height_km,
348        ),
349    ] {
350        if value <= 0.0 {
351            return Err(AtmosphereError::InvalidParameter(format!(
352                "{name} must be positive, not {value}"
353            )));
354        }
355    }
356    if parameters.top_altitude_km * 1000.0 <= parameters.altitude_m {
357        return Err(AtmosphereError::InvalidParameter(format!(
358            "top_altitude_km {} is not above the site at {} m",
359            parameters.top_altitude_km, parameters.altitude_m
360        )));
361    }
362    Ok(())
363}

365/// Absolute humidity in g/m³ for a temperature (K), relative humidity (%)
366/// and pressure (hPa): ATM's `rwat`.
367fn absolute_humidity_g_m3(temperature_k: f64, humidity_percent: f64, pressure_hpa: f64) -> f64 {
368    if pressure_hpa <= 0.0 || temperature_k <= 0.0 || humidity_percent <= 0.0 {
369        return 0.0;
370    }
371    let t = temperature_k;
372    let saturation = 6.105 * (25.22 / t * (t - 273.0) - 5.31 * (t / 273.0).ln()).exp();
373    let factor = 1.0 - (1.0 - humidity_percent / 100.0) * saturation / pressure_hpa;
374    let partial = saturation * humidity_percent / 100.0 / factor;
375    partial * 216.502 / t
376}

378/// The level boundaries of `mkAtmProfile` before they become layers:
379/// pressure (hPa), temperature (K), height above sea level (m) and water
380/// vapour (g/m³) at each level.
381struct Levels {
382    pressure: Vec<f64>,
383    temperature: Vec<f64>,
384    height: Vec<f64>,
385    water: Vec<f64>,
386    tropopause: usize,
387    reference: usize,
388    ozone_model: u32,
389}
390
391const EARTH_RADIUS_KM: f64 = 6371.2;
392const SURFACE_GRAVITY: f64 = 9.80665;
393// ATM fixes the factor between consecutive pressure steps (05JUN2024).
394const PRESSURE_STEP_FACTOR: f64 = 1.075;
395
396impl Levels {
397    /// The level loop of ATM's `mkAtmProfile`.
398    fn build(parameters: &ProfileParameters) -> Self {
399        let (reference, ozone_model, mut pressure_limit) = parameters.atmosphere.model();
400        let (px, tx) = (&PX[reference - 1], &TX[reference - 1]);
401        let altitude_km = parameters.altitude_m / 1000.0;
402        let scale_height_km = parameters.water_vapour_scale_height_km;
403        let top_m = parameters.top_altitude_km * 1000.0;
404        let step = parameters.pressure_step_hpa;
405
406        let ground_water = absolute_humidity_g_m3(
407            parameters.ground_temperature_k,
408            parameters.relative_humidity_percent,
409            parameters.ground_pressure_hpa,
410        );
411        let sea_level_water = ground_water * (altitude_km / scale_height_km).exp();
412        let water_at =
413            |height_m: f64| sea_level_water * (-height_m / (1000.0 * scale_height_km)).exp();
414
415        let mut levels = Self {
416            pressure: vec![parameters.ground_pressure_hpa],
417            temperature: vec![parameters.ground_temperature_k],
418            height: vec![altitude_km * 1000.0],
419            water: vec![ground_water],
420            tropopause: 0,
421            reference,
422            ozone_model,
423        };
424        let mut gravity = SURFACE_GRAVITY;
425        let mut troposphere = true;
426        let mut reference_level = 0usize;
427        let mut i = 0usize;
428        loop {
429            i += 1;
430            let next = levels.pressure[i - 1] - step * PRESSURE_STEP_FACTOR.powf((i - 1) as f64);
431            if next <= pressure_limit {
432                if troposphere {
433                    // The reference level nearest to, and well below, the
434                    // next tropospheric pressure.
435                    let mut nearest = 20000.0;
436                    for (k, &level) in px.iter().enumerate() {
437                        if next.abs() > 1.05 * level && (next - level).abs() <= nearest {
438                            reference_level = k;
439                            nearest = (next - level).abs();
440                        }
441                    }
442                    let k = reference_level.saturating_sub(1);
443                    levels.pressure.push(px[k]);
444                    levels.temperature.push(tx[k]);
445                    let dh = levels.step_height(i, gravity);
446                    levels.height.push(levels.height[i - 1] + dh);
447                } else {
448                    reference_level += 1;
449                    levels.pressure.push(px[reference_level - 1]);
450                    levels.temperature.push(tx[reference_level - 1]);
451                    if reference_level < HX.len() {
452                        let dh = (HX[reference_level] - HX[reference_level - 1]) * 1000.0;
453                        levels.height.push(dh + levels.height[i - 1]);
454                    }
455                }
456                // ATM reads past the heights when the reference levels run
457                // out; that level's water is never used.
458                levels
459                    .water
460                    .push(levels.height.get(i).map_or(f64::NAN, |&h| water_at(h)));
461                if troposphere {
462                    levels.tropopause = i - 1;
463                    troposphere = false;
464                }
465                if reference_level >= HX.len() || levels.height[i] > top_m {
466                    break;
467                }
468            } else {
469                levels.pressure.push(next);
470                gravity = SURFACE_GRAVITY
471                    * (1.0 + (levels.height[i - 1] / 1000.0) / EARTH_RADIUS_KM).powf(-2.0);
472                let dh = levels.step_height(i, gravity);
473                levels.height.push(levels.height[i - 1] + dh);
474                levels
475                    .temperature
476                    .push(levels.temperature[i - 1] + parameters.lapse_rate_k_per_km * dh / 1000.0);
477                levels.water.push(water_at(levels.height[i]));
478                if levels.temperature[i] <= tx[4] && levels.pressure[i] <= px[4] {
479                    pressure_limit = levels.pressure[i];
480                }
481                if levels.height[i] > top_m {
482                    break;
483                }
484            }
485        }
486        levels.pressure.truncate(i);
487        levels.temperature.truncate(i);
488        levels.height.truncate(i);
489        levels.water.truncate(i);
490        levels
491    }
492
493    /// The hypsometric height of the step from level `i - 1` to level `i`.
494    fn step_height(&self, i: usize, gravity: f64) -> f64 {
495        let water_kg_m3 = self.water[i - 1] / 1000.0;
496        288.6948
497            * self.temperature[i - 1]
498            * (1.0 + 0.61 * water_kg_m3 / 1000.0)
499            * (self.pressure[i - 1] / self.pressure[i]).ln()
500            / gravity
501    }
502
503    /// The layers between consecutive levels, as the end of `mkAtmProfile`
504    /// makes them.
505    fn into_profile(self, parameters: ProfileParameters) -> AtmosphereProfile {
506        let count = self.pressure.len() - 1;
507        let mut layers: Vec<Layer> = Vec::with_capacity(count);
508        let mut mid_height_km = parameters.altitude_m / 1000.0;
509        for j in 0..count {
510            let thickness_m = self.height[j + 1] - self.height[j];
511            mid_height_km += thickness_m / 2.0e3;
512            let water_vapour_kg_m3 = if mid_height_km < 40.0 || j == 0 {
513                1.0e-3 * ((self.water[j + 1].ln() + self.water[j].ln()) / 2.0).exp()
514            } else {
515                // ATM keeps the mesospheric water constant (12FEB2024).
516                layers[j - 1].water_vapour_kg_m3
517            };
518            let MinorGases {
519                o3,
520                n2o,
521                co,
522                no2,
523                so2,
524                hcl,
525                hcn,
526            } = minor_gases(mid_height_km, self.reference, self.ozone_model);
527            layers.push(Layer {
528                thickness_m,
529                temperature_k: (self.temperature[j + 1] + self.temperature[j]) / 2.0,
530                bottom_temperature_k: self.temperature[j],
531                top_temperature_k: self.temperature[j + 1],
532                pressure_hpa: ((self.pressure[j + 1].ln() + self.pressure[j].ln()) / 2.0).exp(),
533                bottom_pressure_hpa: self.pressure[j],
534                top_pressure_hpa: self.pressure[j + 1],
535                water_vapour_kg_m3,
536                bottom_water_vapour_kg_m3: 1.0e-3 * self.water[j],
537                top_water_vapour_kg_m3: 1.0e-3 * self.water[j + 1],
538                o3_per_m3: 1.0e6 * o3,
539                co_per_m3: 1.0e6 * co,
540                n2o_per_m3: 1.0e6 * n2o,
541                no2_per_m3: 1.0e6 * no2,
542                so2_per_m3: 1.0e6 * so2,
543                hcl_per_m3: 1.0e6 * hcl,
544                hcn_per_m3: 1.0e6 * hcn,
545            });
546            mid_height_km += thickness_m / 2.0e3;
547        }
548        AtmosphereProfile {
549            tropopause_temperature_k: self.temperature[self.tropopause],
550            tropopause_altitude_m: self.height[self.tropopause],
551            tropopause_layer: self.tropopause,
552            parameters,
553            layers,
554        }
555    }
