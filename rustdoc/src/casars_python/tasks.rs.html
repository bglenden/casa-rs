24/// One application-catalog entry: its identity and how to launch it.
25#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
26pub(crate) struct ApplicationView {
27    pub(crate) id: String,
28    pub(crate) category: String,
29    pub(crate) display_name: String,
30    pub(crate) executable: String,
31    pub(crate) cargo_package: String,
32    pub(crate) override_env: String,
33    pub(crate) dataset_kinds: Vec<String>,
34    pub(crate) include_in_suite: bool,
35}
36
37/// Every application in the catalog, with the surface names of its task.
38pub(crate) fn applications() -> Result<Vec<ApplicationView>, String> {
39    let catalog = builtin_application_catalog()?;
40    catalog
41        .applications
42        .iter()
43        .map(|application| {
44            let bundle = application
45                .surface_bundle()
46                .map_err(|reason| format!("resolve application {}: {reason}", application.id))?;
47            Ok(ApplicationView {
48                id: application.id.clone(),
49                display_name: bundle.surface.display_name().to_string(),
50                category: bundle.surface.category().to_string(),
51                executable: application.launch.executable.clone(),
52                cargo_package: application.launch.cargo_package.clone(),
53                override_env: application.launch.override_env.clone(),
54                dataset_kinds: application.dataset_kinds.clone(),
55                include_in_suite: application.include_in_suite,
56            })
57        })
58        .collect()
59}

61/// One product a completed task declared.
62#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
63pub(crate) struct ProductView {

75/// A successful task's summary, products and diagnostics.
76#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
77pub(crate) struct CompletionView {
78    pub(crate) surface_id: String,
79    pub(crate) summary: String,
80    pub(crate) products: Vec<ProductView>,
81    pub(crate) diagnostics: Vec<String>,
82}

84/// Decode one successful provider result into its contract-declared
85/// products. `values` are the task's parameters as display text.
86pub(crate) fn completion(
87    surface_id: &str,
88    stdout: &str,
89    workspace: &Path,
90    values: Vec<(String, String)>,
91) -> Result<CompletionView, String> {
92    let bundle = builtin_surface_bundle(surface_id)?;
93    let values = values
94        .into_iter()
95        .map(|(name, value)| TaskOutputValue { name, value })
96        .collect::<Vec<_>>();
97    let completion = decode_task_completion(&bundle, stdout, workspace, &values)
98        .map_err(|error| error.to_string())?;
99    let products = completion
100        .products
101        .into_iter()
102        .map(|product| ProductView {
103            id: product.id,
104            role: match product.role {
105                RunProductRole::Primary => "PRIMARY",
106                RunProductRole::Auxiliary => "AUXILIARY",
107                RunProductRole::Preview => "PREVIEW",
108            },
109            resource_kind: match product.resource_kind {
110                RunProductKind::MeasurementSet => "MEASUREMENT_SET",
111                RunProductKind::CasaImage => "CASA_IMAGE",
112                RunProductKind::CasaTable => "CASA_TABLE",
113                RunProductKind::FitsImage => "FITS_IMAGE",
114                RunProductKind::File => "FILE",
115            },
116            label: product.label,
117            path: product.path.to_string_lossy().into_owned(),
118            exists: product.exists,
119            preview_path: product
120                .preview_path
121                .map(|path| path.to_string_lossy().into_owned()),
122            preview_exists: product.preview_exists,
123            diagnostic: product.diagnostic,
124        })
125        .collect();
126    Ok(CompletionView {
127        surface_id: completion.surface_id,
128        summary: completion.summary,
129        products,
130        diagnostics: completion.diagnostics,
131    })
132}

134/// The executable that runs `surface_id`: its catalog launch entry,
135/// resolved against `environment` and the installed suite whose Python
136/// environment is `prefix`, if it is one.
137pub(crate) fn executable(
138    surface_id: &str,
139    environment: &HashMap<String, String>,
140    prefix: &Path,
141) -> Result<PathBuf, String> {
142    let application = builtin_application_catalog()?
143        .application(surface_id)
144        .ok_or_else(|| format!("no task {surface_id:?} in the application catalog"))?;
145    resolve_task_executable(
146        &application.launch,
147        &|name| environment.get(name).map(OsString::from),
148        installed_suite_root(prefix).as_deref(),
149    )
150}

152/// How one task process ended, as Python reports it. The statuses are
153/// [`TaskOutcome`]'s, less a launch failure, which Python sees itself.
154#[derive(Debug, Clone, PartialEq, Serialize)]
155#[serde(tag = "status", rename_all = "snake_case")]
156pub(crate) enum OutcomeView {
157    /// The task succeeded: CASA's return value and the declared products.
158    /// The value reaches Python as CASA's own types, apart from this view.
159    Succeeded {
160        #[serde(skip)]
161        value: CasaValue,
162        completion: CompletionView,
163    },
164    /// The task refused its parameters before it changed anything.
165    Refused {
166        message: String,
167        parameter: Option<String>,
168    },
169    /// The task started and failed.
170    Failed {
171        message: String,
172        parameter: Option<String>,
173    },
174    /// The task caught an interrupt and stopped.
175    Interrupted { message: String },
176    /// A signal ended the task process.
177    Signalled { signal: i32 },
178    /// The process's reply is missing, invalid, or disagrees with its exit.
179    MalformedReply {
180        exit_code: Option<i32>,
181        reason: String,
182    },
183}
184
185/// Classify the process that ran `surface_id` from its return code (as
186/// Python's `subprocess` gives it: negative for a signal) and its stdout
187/// reply, and decode a success's products. `values` are the task's
188/// parameters as display text. A success whose result the contract's
189/// decoder refuses is an error.
190pub(crate) fn outcome(
191    surface_id: &str,
192    returncode: i32,
193    stdout: &str,
194    workspace: &Path,
195    values: Vec<(String, String)>,
196) -> Result<OutcomeView, String> {
197    let exit = if returncode < 0 {
198        TaskExit {
199            code: None,
200            signal: Some(-returncode),
201        }
202    } else {
203        TaskExit {
204            code: Some(returncode),
205            signal: None,
206        }
207    };
208    let mismatch = |surface: String| OutcomeView::MalformedReply {
209        exit_code: exit.code,
210        reason: format!("the task replied for {surface:?}, not {surface_id:?}"),
211    };
212    Ok(match classify_task_exit(exit, stdout) {
213        TaskOutcome::Succeeded { surface, .. }
214        | TaskOutcome::Refused { surface, .. }
215        | TaskOutcome::Failed { surface, .. }
216        | TaskOutcome::Interrupted { surface, .. }
217            if surface != surface_id =>
218        {
219            mismatch(surface)
220        }
221        TaskOutcome::Succeeded { value, result, .. } => {
222            let result = serde_json::to_string(&result)
223                .map_err(|error| format!("serialize the {surface_id} result: {error}"))?;
224            OutcomeView::Succeeded {
225                value,
226                completion: completion(surface_id, &result, workspace, values)?,
227            }
228        }
229        TaskOutcome::Refused {
230            message, parameter, ..
231        } => OutcomeView::Refused { message, parameter },
232        TaskOutcome::Failed {
233            message, parameter, ..
234        } => OutcomeView::Failed { message, parameter },
235        TaskOutcome::Interrupted { message, .. } => OutcomeView::Interrupted { message },
236        TaskOutcome::Signalled { signal } => OutcomeView::Signalled { signal },
237        TaskOutcome::MalformedReply { exit_code, reason } => {
238            OutcomeView::MalformedReply { exit_code, reason }
239        }
240        TaskOutcome::LaunchFailed { message } => OutcomeView::MalformedReply {
241            exit_code: exit.code,
242            reason: message,
243        },
244    })
245}

247/// The application catalog's launch entries.
248#[pyfunction]
249pub(crate) fn application_catalog(py: Python<'_>) -> PyResult<Bound<'_, PyAny>> {
250    let applications = applications().map_err(|reason| reason_error(py, reason))?;
251    to_python(py, &applications)
252}

254/// The executable that runs `surface_id`, found in `environ` (Python's
255/// `os.environ`) and the suite of Python environment `prefix`
256/// (`sys.prefix`). Raises `FileNotFoundError` naming what is missing.
257#[pyfunction]
258pub(crate) fn task_executable(
259    py: Python<'_>,
260    surface_id: &str,
261    environ: HashMap<String, String>,
262    prefix: PathBuf,
263) -> PyResult<PathBuf> {
264    py.detach(|| executable(surface_id, &environ, &prefix))
265        .map_err(PyFileNotFoundError::new_err)
266}

268/// How a task process ended, with a success's products. Releases the GIL
269/// while it checks them.
270#[pyfunction]
271pub(crate) fn task_outcome<'py>(
272    py: Python<'py>,
273    surface_id: &str,
274    returncode: i32,
275    stdout: &str,
276    workspace: PathBuf,
277    values: Vec<(String, String)>,
278) -> PyResult<Bound<'py, PyAny>> {
279    let outcome = py
280        .detach(|| outcome(surface_id, returncode, stdout, &workspace, values))
281        .map_err(|reason| reason_error(py, reason))?;
282    let view = to_python(py, &outcome)?;
283    if let OutcomeView::Succeeded { value, .. } = &outcome {
284        view.set_item("value", reply_value(py, value)?)?;
285    }
286    Ok(view)
287}
