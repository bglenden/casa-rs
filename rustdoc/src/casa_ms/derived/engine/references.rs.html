5/// Return the stored FIELD phase-direction measure without changing its frame.
6///
7/// Storage-owned spectral-range evaluation needs the original measure reference
8/// because CASA applies dynamic references such as ITRF at every selected row
9/// epoch instead of freezing them at the FIELD reference epoch.
10pub fn raw_field_phase_direction(ms: &MeasurementSet, field_id: usize) -> MsResult<MDirection> {
11    let field = ms.field()?;
12    let raw = field.phase_dir(field_id)?;
13    let (longitude, latitude) = phase_dir_constant(raw)?;
14    let reference = resolve_direction_reference(field.table(), "FIELD", "PHASE_DIR", field_id)?;
15    Ok(MDirection::from_angles(longitude, latitude, reference))
16}
17
18/// Return the stored FIELD reference direction in its original measure frame.
19pub fn raw_field_reference_direction(ms: &MeasurementSet, field_id: usize) -> MsResult<MDirection> {
20    let field = ms.field()?;
21    let (longitude, latitude) = phase_dir_constant(field.reference_dir(field_id)?)?;
22    let reference = resolve_direction_reference(field.table(), "FIELD", "REFERENCE_DIR", field_id)?;
23    Ok(MDirection::from_angles(longitude, latitude, reference))
24}

26/// The epoch of `field_id`'s FIELD row, its `TIME` in that column's
27/// reference (UTC when it names none): the origin of its direction
28/// polynomial and the time casacore evaluates its ephemeris at when asked
29/// for its direction at no time
30/// ([`MsCalEngine::moving_field_reference_direction`]).
31pub(super) fn field_reference_epoch(ms: &MeasurementSet, field_id: usize) -> MsResult<MEpoch> {
32    let field = ms.field()?;
33    Ok(MEpoch::from_mjd(
34        selected_f64_value(field.table(), field_id, "TIME")? / 86_400.0,
35        detect_epoch_reference(field.table(), "TIME", EpochRef::UTC),
36    ))
37}

39pub(super) enum BorrowedMeasureReference<'a> {
40    Fixed(&'a str),
41    VariableInt {
42        ref_column: &'a str,
43        tab_ref_types: &'a ArrayD<String>,
44        tab_ref_codes: BorrowedReferenceCodes<'a>,
45    },
46    VariableString {
47        ref_column: &'a str,
48    },
49}
50
51#[derive(Clone, Copy)]
52pub(super) enum BorrowedReferenceCodes<'a> {
53    Int32(&'a ArrayD<i32>),
54    UInt32(&'a ArrayD<u32>),
55    Int64(&'a ArrayD<i64>),
56}
57
58impl BorrowedReferenceCodes<'_> {
59    fn position(self, code: i32) -> Option<usize> {
60        match self {
61            Self::Int32(values) => values.iter().position(|candidate| *candidate == code),
62            Self::UInt32(values) => values
63                .iter()
64                .position(|candidate| *candidate as i32 == code),
65            Self::Int64(values) => values
66                .iter()
67                .position(|candidate| *candidate as i32 == code),
68        }
69    }
70}

72pub(super) fn borrowed_measure_reference<'a>(
73    table: &'a Table,
74    column: &str,
75) -> Option<BorrowedMeasureReference<'a>> {
76    let Value::Record(measinfo) = table.column_keywords(column)?.get("MEASINFO")? else {
77        return None;
78    };
79    let Value::Scalar(ScalarValue::String(measure_type)) = measinfo.get("type")? else {
80        return None;
81    };
82    if ![
83        "epoch",
84        "direction",
85        "position",
86        "frequency",
87        "doppler",
88        "radialvelocity",
89    ]
90    .iter()
91    .any(|candidate| measure_type.eq_ignore_ascii_case(candidate))
92    {
93        return None;
94    }
95
96    if let Some(Value::Scalar(ScalarValue::String(ref_column))) = measinfo.get("VarRefCol") {
97        let tab_ref_types = match measinfo.get("TabRefTypes") {
98            Some(Value::Array(ArrayValue::String(values))) => Some(values),
99            _ => None,
100        };
101        let tab_ref_codes = match measinfo.get("TabRefCodes") {
102            Some(Value::Array(ArrayValue::Int32(values))) => {
103                Some(BorrowedReferenceCodes::Int32(values))
104            }
105            Some(Value::Array(ArrayValue::UInt32(values))) => {
106                Some(BorrowedReferenceCodes::UInt32(values))
107            }
108            Some(Value::Array(ArrayValue::Int64(values))) => {
109                Some(BorrowedReferenceCodes::Int64(values))
110            }
111            _ => None,
112        };
113        return match (tab_ref_types, tab_ref_codes) {
114            (Some(tab_ref_types), Some(tab_ref_codes)) => {
115                Some(BorrowedMeasureReference::VariableInt {
116                    ref_column,
117                    tab_ref_types,
118                    tab_ref_codes,
119                })
120            }
121            _ => Some(BorrowedMeasureReference::VariableString { ref_column }),
122        };
123    }
124    match measinfo.get("Ref") {
125        Some(Value::Scalar(ScalarValue::String(refer))) => {
126            Some(BorrowedMeasureReference::Fixed(refer))
127        }
128        _ => None,
129    }
130}

136/// Detect the epoch reference frame from a table column's measure metadata.
137pub fn detect_epoch_reference(table: &Table, column: &str, default: EpochRef) -> EpochRef {
138    let Some(reference) = borrowed_measure_reference(table, column) else {
139        return default;
140    };
141    match reference {
142        BorrowedMeasureReference::Fixed(refer) => refer.parse::<EpochRef>().unwrap_or(default),
143        BorrowedMeasureReference::VariableInt { .. }
144        | BorrowedMeasureReference::VariableString { .. } => default,
145    }
146}

148pub(super) fn phase_dir_constant(dir: &ArrayValue) -> MsResult<(f64, f64)> {
149    match dir {
150        ArrayValue::Float64(arr) => {
151            let shape = arr.shape();
152            if shape.len() != 2 || shape[0] != 2 || shape[1] == 0 {
153                return Err(MsError::ColumnTypeMismatch {
154                    column: "PHASE_DIR".to_string(),
155                    table: "FIELD".to_string(),
156                    expected: "Float64 array shaped [2, nPoly+1]".to_string(),
157                    found: format!("Float64 array with shape {shape:?}"),
158                });
159            }
160            Ok((arr[[0, 0]], arr[[1, 0]]))
161        }
162        other => Err(MsError::ColumnTypeMismatch {
163            column: "PHASE_DIR".to_string(),
164            table: "FIELD".to_string(),
165            expected: "Float64 array shaped [2, nPoly+1]".to_string(),
166            found: format!("{:?}", other.primitive_type()),
167        }),
168    }
169}

171pub(super) fn selected_scalar_value(
172    table: &Table,
173    row: usize,
174    column: &str,
175) -> MsResult<ScalarValue> {
176    table
177        .column_accessor(column)?
178        .scalar_cells_owned_for_rows(&[row])?
179        .pop()
180        .flatten()
181        .ok_or_else(|| {
182            MsError::InvalidInput(format!(
183                "required selected metadata {column} row {row} is undefined"
184            ))
185        })
186}

212/// Resolve the direction reference frame for a selected row without reading all rows.
213pub fn resolve_direction_reference_selected(
214    table: &Table,
215    table_name: &str,
216    column: &str,
217    row: usize,
218) -> MsResult<DirectionRef> {
219    let Some(description) = borrowed_measure_reference(table, column) else {
220        return Ok(DirectionRef::J2000);
221    };
222    let reference = match description {
223        BorrowedMeasureReference::Fixed(refer) => Cow::Borrowed(refer),
224        BorrowedMeasureReference::VariableInt {
225            ref_column,
226            tab_ref_types,
227            tab_ref_codes,
228        } => {
229            let code = match selected_scalar_value(table, row, ref_column)? {
230                ScalarValue::Int32(value) => value,
231                ScalarValue::Int64(value) => {
232                    i32::try_from(value).map_err(|_| MsError::ColumnTypeMismatch {
233                        column: ref_column.to_string(),
234                        table: table_name.to_string(),
235                        expected: "an integer direction reference in the Int32 domain".to_string(),
236                        found: value.to_string(),
237                    })?
238                }
239                ScalarValue::UInt32(value) => {
240                    i32::try_from(value).map_err(|_| MsError::ColumnTypeMismatch {
241                        column: ref_column.to_string(),
242                        table: table_name.to_string(),
243                        expected: "an integer direction reference in the Int32 domain".to_string(),
244                        found: value.to_string(),
245                    })?
246                }
247                other => {
248                    return Err(MsError::ColumnTypeMismatch {
249                        column: ref_column.to_string(),
250                        table: table_name.to_string(),
251                        expected: "Int scalar".to_string(),
252                        found: format!("{other:?}"),
253                    });
254                }
255            };
256            let Some(index) = tab_ref_codes.position(code) else {
257                return Err(MsError::InvalidMeasureCode {
258                    table: table_name.to_string(),
259                    column: ref_column.to_string(),
260                    code,
261                });
262            };
263            tab_ref_types
264                .iter()
265                .nth(index)
266                .map(|reference| Cow::Borrowed(reference.as_str()))
267                .ok_or_else(|| MsError::ColumnTypeMismatch {
268                    column: column.to_string(),
269                    table: table_name.to_string(),
270                    expected: "TabRefTypes index in bounds".to_string(),
271                    found: format!("missing entry for TabRefCodes[{index}]"),
272                })?
273        }
274        BorrowedMeasureReference::VariableString { ref_column } => {
275            Cow::Owned(selected_string_value(table, row, ref_column)?)
276        }
277    };
278    reference
279        .parse::<DirectionRef>()
280        .map_err(|_| MsError::ColumnTypeMismatch {
281            column: column.to_string(),
282            table: table_name.to_string(),
283            expected: "a supported direction reference".to_string(),
284            found: reference.into_owned(),
285        })
286}

288pub(super) fn resolve_observatory_position(
289    ms: &MeasurementSet,
290    antenna_positions: &[MPosition],
291    measures: &dyn MeasuresProvider,
292) -> MPosition {
293    let telescope = ms.observation().ok().and_then(|observation| {
294        (0..observation.row_count()).find_map(|row| observation.string(row, "TELESCOPE_NAME").ok())
295    });
296    MPosition::casa_array_reference(telescope.as_deref(), antenna_positions.first(), measures)
297}

299pub(super) fn resolve_field_phase_direction_j2000_with_observatory(
300    dir: MDirection,
301    field: &crate::subtables::MsField<'_>,
302    field_id: usize,
303    observatory_position: &MPosition,
304    measures: Arc<dyn MeasuresProvider>,
305) -> MsResult<MDirection> {
306    if dir.refer() == DirectionRef::J2000 {
307        return Ok(dir);
308    }
309    let epoch_ref = detect_epoch_reference(field.table(), "TIME", EpochRef::UTC);
310    let epoch = MEpoch::from_mjd(field.time(field_id)? / 86400.0, epoch_ref);
311    let frame = MeasFrame::new()
312        .with_epoch(epoch)
313        .with_position(observatory_position.clone())
314        .with_measures(measures);
315    Ok(dir.convert_to(DirectionRef::J2000, &frame)?)
316}

318/// Resolve the direction reference frame for one row of a table direction column.
319pub fn resolve_direction_reference(
320    table: &Table,
321    table_name: &str,
322    column: &str,
323    row: usize,
324) -> MsResult<DirectionRef> {
325    let Some(desc) = TableMeasDesc::reconstruct(table, column) else {
326        return Ok(DirectionRef::J2000);
327    };
328    let refer = resolve_reference_string(table, table_name, &desc, row)?;
329    refer
330        .parse::<DirectionRef>()
331        .map_err(|_| MsError::ColumnTypeMismatch {
332            column: column.to_string(),
333            table: table_name.to_string(),
334            expected: "a supported direction reference".to_string(),
335            found: refer,
336        })
337}

339pub(super) fn resolve_reference_string(
340    table: &Table,
341    table_name: &str,
342    desc: &TableMeasDesc,
343    row: usize,
344) -> MsResult<String> {
345    match desc.ref_desc() {
346        MeasRefDesc::Fixed { refer } => Ok(refer.clone()),
347        MeasRefDesc::VariableInt {
348            ref_column,
349            tab_ref_types,
350            tab_ref_codes,
351        } => {
352            let code = match table.cell_accessor(row, ref_column)?.scalar()? {
353                &ScalarValue::Int32(value) => value,
354                &ScalarValue::Int64(value) => {
355                    i32::try_from(value).map_err(|_| MsError::ColumnTypeMismatch {
356                        column: ref_column.clone(),
357                        table: table_name.to_string(),
358                        expected: "an integer direction reference in the Int32 domain".to_string(),
359                        found: value.to_string(),
360                    })?
361                }
362                &ScalarValue::UInt32(value) => {
363                    i32::try_from(value).map_err(|_| MsError::ColumnTypeMismatch {
364                        column: ref_column.clone(),
365                        table: table_name.to_string(),
366                        expected: "an integer direction reference in the Int32 domain".to_string(),
367                        found: value.to_string(),
368                    })?
369                }
370                other => {
371                    return Err(MsError::ColumnTypeMismatch {
372                        column: ref_column.clone(),
373                        table: table_name.to_string(),
374                        expected: "Int scalar".to_string(),
375                        found: format!("{other:?}"),
376                    });
377                }
378            };
379            for (index, candidate) in tab_ref_codes.iter().enumerate() {
380                if *candidate == code {
381                    return tab_ref_types.get(index).cloned().ok_or_else(|| {
382                        MsError::ColumnTypeMismatch {
383                            column: desc.column_name().to_string(),
384                            table: table_name.to_string(),
385                            expected: "TabRefTypes index in bounds".to_string(),
386                            found: format!("missing entry for TabRefCodes[{index}]"),
387                        }
388                    });
389                }
390            }
391            Err(MsError::InvalidMeasureCode {
392                table: table_name.to_string(),
393                column: ref_column.clone(),
394                code,
395            })
396        }
397        MeasRefDesc::VariableString { ref_column } => {
398            match table.cell_accessor(row, ref_column)?.scalar()? {
399                ScalarValue::String(value) => Ok(value.clone()),
400                other => Err(MsError::ColumnTypeMismatch {
401                    column: ref_column.clone(),
402                    table: table_name.to_string(),
403                    expected: "String scalar".to_string(),
404                    found: format!("{other:?}"),
405                }),
406            }
407        }
408    }
