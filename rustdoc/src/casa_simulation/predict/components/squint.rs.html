50impl BeamContext {
51    pub(super) fn advance(
52        &mut self,
53        time_s: f64,
54        phase: &MDirection,
55        frame: &MeasFrame,
56        increment: f64,
57        alt_az: bool,
58    ) -> Result<f64, ComponentError> {
59        let changed = self.history.key != Some(self.key) || self.history.angle.is_none();
60        if changed || (self.history.time_s != Some(time_s) && increment < std::f64::consts::TAU) {
61            let angle = if alt_az {
62                let frame = frame
63                    .clone()
64                    .with_epoch(MEpoch::from_mjd(time_s / 86400., EpochRef::UTC));
65                f64::from(phase.parallactic_angle(&frame).map_err(error)? as f32)
66            } else {
67                0.
68            };
69            if changed
70                || self
71                    .history
72                    .angle
73                    .is_some_and(|previous| (angle - previous).abs() > increment)
74            {
75                self.history.angle = Some(angle);
76            }
77        }
78        self.history.key = Some(self.key);
79        self.history.time_s = Some(time_s);
80        Ok(self
81            .history
82            .angle
83            .expect("the first row establishes its feed angle"))
84    }
85}

87impl Squint {
88    pub(super) fn bytes(workers: usize, channels: usize) -> u64 {
89        size_of::<Self>() as u64
90            + BeamPattern::BYTES
91            + (workers
92                * (size_of::<Mutex<Vec<Complex64>>>() + channels * 4 * size_of::<Complex64>()))
93                as u64
94    }
95
96    pub(super) fn resident_bytes(&self) -> u64 {
97        Self::bytes(self.workspaces.len(), self.channels)
98    }
99
100    pub(super) fn new(
101        config: Config,
102        pattern: &BeamPattern,
103        workers: usize,
104        channels: usize,
105    ) -> Self {
106        Self {
107            config,
108            pattern: pattern.clone(),
109            history: Mutex::default(),
110            workspaces: (0..workers)
111                .map(|_| Mutex::new(vec![Complex64::new(0., 0.); channels * 4]))
112                .collect(),
113            channels,
114        }
115    }
116
117    pub(super) fn workspace(
118        &self,
119        worker: usize,
120    ) -> Result<std::sync::MutexGuard<'_, Vec<Complex64>>, ComponentError> {
121        self.workspaces[worker].lock().map_err(error)
122    }
123
124    pub(super) fn prepare_job(
125        &self,
126        key: (usize, usize),
127        times: impl Iterator<Item = f64>,
128        phase: &MDirection,
129        frame: &MeasFrame,
130    ) -> Result<BeamContext, ComponentError> {
131        let mut history = self.history.lock().map_err(error)?;
132        let initial = BeamContext {
133            history: *history,
134            key,
135        };
136        let mut context = initial;
137        for time in times {
138            context.advance(time, phase, frame, self.config.2, self.config.3)?;
139        }
140        *history = context.history;
141        Ok(initial)
142    }
143
144    pub(super) fn flux(
145        &self,
146        circular: [Complex64; 4],
147        direction: [f64; 2],
148        pointing: [f64; 2],
149        angle: f64,
150        frequency: f64,
151        basis: Polarisation,
152    ) -> [Complex64; 4] {
153        let powers = [true, false].map(|right| {
154            let [dx, dy] =
155                self.config
156                    .0
157                    .squint_displacement_rad(self.config.1, frequency, angle, right);
158            let shifted = shifted(pointing, dx, dy);
159            f64::from(
160                self.pattern
161                    .power_at(separation_rad(direction, shifted), frequency),
162            )
163        });
164        let centre = f64::from(
165            self.pattern
166                .power_at(separation_rad(direction, pointing), frequency),
167        );
168        let [rr, rl, lr, ll] = circular;
169        let [rr, ll] = [rr.re * powers[0], ll.re * powers[1]];
170        stokes_in(
171            basis,
172            [
173                (rr + ll) / 2.,
174                (rl.re + lr.re) * centre / 2.,
175                (rl.im - lr.im) * centre / 2.,
176                (rr - ll) / 2.,
177            ],
178        )
179    }
180}

182/// MVDirection::shift(dx,dy,true): Euler(-dx,Z,dec+dy,Y,-ra,Z).
183fn shifted([ra, dec]: [f64; 2], dx: f64, dy: f64) -> [f64; 2] {
184    let (sin, cos) = (dec + dy).sin_cos();
185    [ra + dx.sin().atan2(dx.cos() * cos), (dx.cos() * sin).asin()]
186}
