2//! Select compatible native spectral axes and admit the grid before forming it.
3use super::{TotalPowerError, TotalPowerRequest, coordinates::rest_frequency, failure};
4use crate::RunContext;
5use casa_coordinates::{
6    Coordinate, CoordinateSystem, SpectralCoordinate, StokesCoordinate, StokesType,
7};
8use casa_imaging_runtime::{Demand, Reservation, admit};
9use casa_imaging_singledish::{FlagMerge, GridFunction, GridShape, KernelTable, TotalPowerGrid};
10use casa_numerics::PrimaryBeam;
11use casa_observation::{Catalogue, Observation, PartitionId, WindowId};
12
13pub(super) fn selected_partitions(
14    catalogue: &Catalogue,
15    intent: &str,
16    windows: &[WindowId],
17) -> Result<Vec<PartitionId>, TotalPowerError> {
18    let mut partitions = if intent.is_empty() {
19        catalogue.partition_ids()
20    } else {
21        catalogue.partitions_with_intent(intent).map_err(failure)?
22    };
23    partitions
24        .retain(|id| windows.is_empty() || windows.contains(&catalogue.partition(*id).window));
25    if partitions.is_empty() {
26        return Err(failure(
27            "No match found for the requested intent and spectral windows",
28        ));
29    }
30    Ok(partitions)
31}
32
33pub(super) struct Plan {
34    pub partitions: Vec<PartitionId>,
35    pub shape: GridShape,
36    pub coordinates: CoordinateSystem,
37}
38
39impl Plan {
40    pub fn new(
41        observation: &dyn Observation,
42        request: &TotalPowerRequest,
43    ) -> Result<Self, TotalPowerError> {
44        let catalogue = observation.catalogue();
45        let partitions = selected_partitions(catalogue, &request.intent, &request.windows)?;
46        let first = *partitions
47            .first()
48            .ok_or_else(|| failure("No match found for the requested intent"))?;
49        let entry = catalogue.partition(first);
50        if request.channels.is_empty() || request.channels.end > entry.frequencies_hz.len() {
51            return Err(failure(
52                "channel selection is outside the input spectral window",
53            ));
54        }
55        let frequencies = &entry.frequencies_hz[request.channels.clone()];
56        let step = entry.channel_widths_hz[request.channels.start];
57        if frequencies
58            .windows(2)
59            .any(|v| ((v[1] - v[0]) - step).abs() > step.abs() * 1e-8)
60        {
61            return Err(failure(
62                "nonuniform spectral channels require regridding; see #938",
63            ));
64        }
65        for &id in &partitions {
66            let current = catalogue.partition(id);
67            if current
68                .fields
69                .iter()
70                .any(|field| catalogue.fields[field.0 as usize].ephemeris.is_some())
71            {
72                return Err(TotalPowerError::Unsupported {
73                    behavior: "moving-source total-power imaging",
74                    issue: 938,
75                });
76            }
77            if current.frequency_frame != entry.frequency_frame
78                || current.frequencies_hz != entry.frequencies_hz
79            {
80                return Err(failure(
81                    "spectral regridding between different windows is not implemented; see #938",
82                ));
83            }
84            if !(1..=2).contains(&current.correlations.len())
85                || current.correlations.iter().any(|c| {
86                    !matches!(
87                        c,
88                        casa_core::Correlation::XX
89                            | casa_core::Correlation::YY
90                            | casa_core::Correlation::RR
91                            | casa_core::Correlation::LL
92                            | casa_core::Correlation::I
93                    )
94                })
95            {
96                return Err(failure(
97                    "single-dish Stokes I imaging requires one or two parallel hands",
98                ));
99            }
100        }
101        let shape = GridShape {
102            nx: request.size[0],
103            ny: request.size[1],
104            planes: 1,
105            channels: frequencies.len(),
106        };
107        if shape.nx == 0
108            || shape.ny == 0
109            || shape
110                .nx
111                .checked_mul(shape.ny)
112                .is_none_or(|n| n > u32::MAX as usize || n.checked_mul(shape.channels).is_none())
113        {
114            return Err(failure("image dimensions are zero or too large"));
115        }
116        if request
117            .direction
118            .increment()
119            .iter()
120            .any(|v| !v.is_finite() || *v == 0.)
121        {
122            return Err(failure("image cells must be finite and nonzero"));
123        }
124        let mut coordinates = CoordinateSystem::new();
125        coordinates.add_coordinate(request.direction.clone());
126        coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
127        coordinates.add_coordinate(SpectralCoordinate::new(
128            entry.frequency_frame,
129            frequencies[0],
130            step,
131            0.,
132            rest_frequency(
133                catalogue,
134                first,
135                request.rest_frequency_hz,
136                !request.windows.is_empty(),
137            )?,
138        ));
139        Ok(Self {
140            partitions,
141            shape,
142            coordinates,
143        })
144    }
145
146    pub fn grid(
147        &self,
148        observation: &dyn Observation,
149        request: &TotalPowerRequest,
150        context: &RunContext,
151    ) -> Result<(TotalPowerGrid, Reservation), TotalPowerError> {
152        let catalogue = observation.catalogue();
153        let entry = catalogue.partition(self.partitions[0]);
154        let frequency = entry.frequencies_hz[request.channels.start];
155        let primary = if request.kernel == GridFunction::PrimaryBeam {
156            Some(
157                PrimaryBeam::for_telescope(
158                    &catalogue.observation(entry.observation).telescope,
159                    frequency,
160                )
161                .ok_or_else(|| failure("unsupported telescope primary beam"))?,
162            )
163        } else {
164            None
165        };
166        let pixels = primary.as_ref().map_or(0, |beam| {
167            beam.pattern()
168                .support_pixels(request.direction.increment()[0], frequency)
169        });
170        if request.kernel.support().unwrap_or(pixels) > (i32::MAX as usize / 300) {
171            return Err(failure("kernel support exceeds CASA's index range"));
172        }
173        let mut forming = admit(
174            &context.host,
175            &context.policy,
176            &Demand {
177                phase: "total-power kernel",
178                memory: request.kernel.forming_bytes(pixels),
179            },
180        )
181        .map_err(failure)?;
182        let table = if let Some(primary) = primary {
183            KernelTable::primary_beam(
184                &primary.pattern(),
185                request.direction.increment()[0],
186                frequency,
187            )
188        } else {
189            KernelTable::new(request.kernel).map_err(failure)?
190        };
191        forming.retain(table.bytes());
192        let grid_memory = admit(
193            &context.host,
194            &context.policy,
195            &Demand {
196                phase: "total-power grid",
197                memory: TotalPowerGrid::bytes(self.shape, &table, self.shape.channels),
198            },
199        )
200        .map_err(failure)?;
201        let grid = TotalPowerGrid::new(self.shape, table, FlagMerge::AnyCorrelation, grid_memory)
202            .map_err(failure)?;
203        // The grid's reservation includes its kernel; no allocation escapes admission.
204        forming.retain(0);
205        Ok((grid, forming))
206    }
207}
