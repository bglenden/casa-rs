24pub(crate) fn validate_solve_interval(
25    solve_interval: GainSolveInterval,
26) -> Result<(), GainSolveError> {
27    match solve_interval {
28        GainSolveInterval::Infinite | GainSolveInterval::Integration => Ok(()),
29        GainSolveInterval::Seconds(seconds) if seconds > 0.0 => Ok(()),
30        _ => Err(GainSolveError::UnsupportedSolveInterval { solve_interval }),
31    }
32}

34pub(crate) fn validate_smodel(smodel: [f32; 4]) -> Result<(), GainSolveError> {
35    if smodel[0] <= 0.0 || smodel[1] != 0.0 || smodel[2] != 0.0 || smodel[3] != 0.0 {
36        return Err(GainSolveError::UnsupportedSkyModel { smodel });
37    }
38    Ok(())
39}

41pub(crate) fn load_preapplied_rows(
42    ms: &MeasurementSet,
43    selection: &MsSelection,
44    prior_calibration_tables: &[ApplyCalibrationTableSpec],
45    parang: bool,
46) -> Result<Option<HashMap<usize, EvaluatedApplyRow>>, GainSolveError> {
47    if prior_calibration_tables.is_empty() && !parang {
48        return Ok(None);
49    }
50    let plan = plan_apply(
51        ms,
52        &ApplyPlanRequest {
53            selection: selection.clone(),
54            apply_mode: crate::ApplyMode::CalFlag,
55            parang,
56            calibration_tables: prior_calibration_tables.to_vec(),
57        },
58    )
59    .map_err(|source| GainSolveError::PriorCalibrationPlan {
60        source: Box::new(source),
61    })?;
62    evaluate_apply_rows(ms, &plan).map(Some).map_err(|source| {
63        GainSolveError::PriorCalibrationApply {
64            source: Box::new(source),
65        }
66    })
67}

69pub(crate) fn build_solve_groups(
70    ms: &MeasurementSet,
71    rows: &[SelectedSolveRow],
72    preapplied_rows: Option<&HashMap<usize, EvaluatedApplyRow>>,
73    options: SolveGroupOptions,
74) -> Result<BTreeMap<(SolveBaseKey, SolveBucketKey), SolveAccumulator>, GainSolveError> {
75    let mut sorted_rows = rows.to_vec();
76    sorted_rows.sort_by_key(|row| {
77        (
78            row.field_id,
79            row.data_spw_id,
80            row.observation_id,
81            row.scan_number,
82            row.time_seconds.to_bits(),
83            row.interval_seconds.to_bits(),
84            row.row_index,
85        )
86    });
87
88    let mut groups = BTreeMap::<(SolveBaseKey, SolveBucketKey), SolveAccumulator>::new();
89    let mut bucket_starts = BTreeMap::<SolveBaseKey, f64>::new();
90
91    for row in sorted_rows {
92        if row.antenna1 == row.antenna2 {
93            continue;
94        }
95        let base_key = SolveBaseKey {
96            field_id: if options.combine.fields {
97                0
98            } else {
99                row.field_id
100            },
101            spw_id: row.data_spw_id,
102            observation_id: row.observation_id,
103            scan_number: if options.combine.scans {
104                0
105            } else {
106                row.scan_number
107            },
108        };
109        let bucket_key = match options.solve_interval {
110            GainSolveInterval::Infinite => SolveBucketKey::Infinite,
111            GainSolveInterval::Integration => SolveBucketKey::Integration {
112                time_bits: row.time_seconds.to_bits(),
113                interval_bits: row.interval_seconds.to_bits(),
114            },
115            GainSolveInterval::Seconds(seconds) => {
116                let bucket_start = bucket_starts.entry(base_key).or_insert(row.time_seconds);
117                let bucket_index = ((row.time_seconds - *bucket_start) / seconds)
118                    .floor()
119                    .max(0.0);
120                SolveBucketKey::Seconds(bucket_index as u64)
121            }
122        };
123
124        let group = groups.entry((base_key, bucket_key)).or_insert_with(|| {
125            SolveAccumulator::new(row.field_id, row.data_spw_id, row.observation_id)
126        });
127        let preapplied = preapplied_rows.and_then(|rows| rows.get(&row.row_index));
128        group.observe(ms, &row, preapplied, &options)?;
129    }
130
131    Ok(groups)
132}

134pub(crate) fn all_antenna_ids(ms: &MeasurementSet) -> Result<BTreeSet<i32>, GainSolveError> {
135    let antenna = ms
136        .antenna()
137        .map_err(|source| GainSolveError::OpenMeasurementSet {
138            path: ms
139                .path()
140                .map(|path| path.display().to_string())
141                .unwrap_or_else(|| "<in-memory>".to_string()),
142            source,
143        })?;
144    Ok((0..antenna.row_count())
145        .map(|row| i32::try_from(row).expect("antenna row count should fit in i32"))
146        .collect())
147}

149pub(crate) fn resolve_refant(
150    ms: &MeasurementSet,
151    selector: &RefAntSelector,
152) -> Result<i32, GainSolveError> {
153    match selector {
154        RefAntSelector::AntennaId(id) => Ok(*id),
155        RefAntSelector::AntennaName(name) => {
156            let antenna = ms
157                .antenna()
158                .map_err(|error| GainSolveError::ResolveRefAnt {
159                    selector: name.clone(),
160                    reason: error.to_string(),
161                })?;
162            for row in 0..antenna.row_count() {
163                if antenna
164                    .name(row)
165                    .map_err(|error| GainSolveError::ResolveRefAnt {
166                        selector: name.clone(),
167                        reason: error.to_string(),
168                    })?
169                    == *name
170                {
171                    return Ok(row as i32);
172                }
173            }
174            Err(GainSolveError::ResolveRefAnt {
175                selector: name.clone(),
176                reason: "no ANTENNA.NAME match".to_string(),
177            })
178        }
179    }
180}

182pub(crate) fn collect_selected_rows(
183    ms: &MeasurementSet,
184    selection: &MsSelection,
185) -> Result<Vec<SelectedSolveRow>, GainSolveError> {
186    let selected_rows = resolve_calibration_selection(ms, selection).map_err(|error| {
187        GainSolveError::OpenMeasurementSet {
188            path: ms
189                .path()
190                .map(|path| path.display().to_string())
191                .unwrap_or_else(|| "<in-memory>".to_string()),
192            source: match error {
193                casa_ms::MsSelectionError::Domain(source) => source,
194                other => MsError::InvalidInput(other.to_string()),
195            },
196        }
197    })?;
198    if selected_rows.selected_rows.is_empty() {
199        return Err(GainSolveError::EmptySelection);
200    }
201
202    selected_rows
203        .selected_rows
204        .into_iter()
205        .map(|row| {
206            Ok(SelectedSolveRow {
207                row_index: row.row_index,
208                field_id: row.field_id,
209                observation_id: get_i32(ms.main_table(), row.row_index, "OBSERVATION_ID")?,
210                data_desc_id: row.data_desc_id,
211                data_spw_id: row.spectral_window_id,
212                channel_selection: selection.channel_selection_for_spw(row.spectral_window_id),
213                antenna1: row.antenna1_id,
214                antenna2: row.antenna2_id,
215                time_seconds: row.time_mjd_seconds,
216                time_centroid_seconds: get_f64(ms.main_table(), row.row_index, "TIME_CENTROID")?,
217                interval_seconds: get_f64(ms.main_table(), row.row_index, "INTERVAL")?,
218                scan_number: get_i32(ms.main_table(), row.row_index, "SCAN_NUMBER")?,
219                flag_row: row.flag_row,
220            })
221        })
222        .collect()
223}

225pub(crate) fn correlation_types_for_ddid(
226    ms: &MeasurementSet,
227    data_desc_id: i32,
228) -> Result<Vec<i32>, GainSolveError> {
229    let dd = ms
230        .data_description()
231        .map_err(|source| GainSolveError::OpenMeasurementSet {
232            path: ms
233                .path()
234                .map(|path| path.display().to_string())
235                .unwrap_or_else(|| "<in-memory>".to_string()),
236            source,
237        })?;
238    let pol = ms
239        .polarization()
240        .map_err(|source| GainSolveError::OpenMeasurementSet {
241            path: ms
242                .path()
243                .map(|path| path.display().to_string())
244                .unwrap_or_else(|| "<in-memory>".to_string()),
245            source,
246        })?;
247    let row = usize::try_from(data_desc_id).map_err(|_| {
248        GainSolveError::UnsupportedCorrelationLayout {
249            data_desc_id,
250            correlation_types: Vec::new(),
251        }
252    })?;
253    let pol_id = dd
254        .polarization_id(row)
255        .map_err(|source| GainSolveError::OpenMeasurementSet {
256            path: ms
257                .path()
258                .map(|path| path.display().to_string())
259                .unwrap_or_else(|| "<in-memory>".to_string()),
260            source,
261        })?;
262    pol.corr_type(usize::try_from(pol_id).unwrap_or(usize::MAX))
263        .map_err(|source| GainSolveError::OpenMeasurementSet {
264            path: ms
265                .path()
266                .map(|path| path.display().to_string())
267                .unwrap_or_else(|| "<in-memory>".to_string()),
268            source,
269        })
270}

383/// One MAIN row's arrays for the solve, `[correlation, channel]`.
384struct SolveRowInputs<'a> {
385    data: &'a ArrayD<Complex32>,
386    flags: Cow<'a, ArrayD<bool>>,
387    weights: ArrayD<f32>,
388    model: Option<&'a ArrayD<Complex32>>,
389}
390
391impl SolveAccumulator {
392    pub(crate) fn new(field_id: i32, spw_id: i32, observation_id: i32) -> Self {
393        Self {
394            field_id,
395            spw_id,
396            observation_id,
397            min_time: f64::INFINITY,
398            max_time: f64::NEG_INFINITY,
399            unique_time_sum: 0.0,
400            unique_time_count: 0,
401            total_interval: 0.0,
402            sample_rows: 0,
403            field_ids: BTreeSet::new(),
404            scan_numbers: BTreeSet::new(),
405            antenna_ids: BTreeSet::new(),
406            unique_time_bits: BTreeSet::new(),
407            solution_channels: 1,
408            channel_graphs: Vec::new(),
409            solution_time: 0.0,
410            baseline_centroids: HashMap::new(),
411            receptor_accumulations: Vec::new(),
412        }
413    }
414
415    /// Add one MAIN row: its unflagged, positively weighted samples divided by
416    /// the model, each weighted by its solve weight times |model|² as
417    /// `VisEquation::divideCorrByModel` does, and its TIME_CENTROID.
418    pub(crate) fn observe(
419        &mut self,
420        ms: &MeasurementSet,
421        row: &SelectedSolveRow,
422        preapplied_row: Option<&EvaluatedApplyRow>,
423        options: &SolveGroupOptions,
424    ) -> Result<(), GainSolveError> {
425        let inputs = read_row_inputs(ms, row, preapplied_row, options.model_source)?;
426        let correlation_types = correlation_types_for_ddid(ms, row.data_desc_id)?;
427        let channel_count = inputs.data.shape()[1];
428        let channels = if options.channelized {
429            if self.sample_rows > 0 && self.solution_channels != channel_count {
430                return Err(GainSolveError::UnsupportedParameterShape {
431                    path: "<measurement-set DATA>".to_string(),
432                    shape: inputs.data.shape().to_vec(),
433                });
434            }
435            self.solution_channels = channel_count;
436            (0..channel_count).collect()
437        } else {
438            selected_channels(row, channel_count)?
439        };
440
441        let graph_count = match options.gain_type {
442            GainType::G => 2,
443            GainType::T => 1,
444        };
445        if self.receptor_accumulations.len() < graph_count {
446            self.receptor_accumulations
447                .resize_with(graph_count, HashMap::new);
448        }
449        self.observe_metadata(row);
450
451        let mut row_weight = 0.0_f64;
452        for (corr_index, &correlation_type) in correlation_types.iter().enumerate() {
453            let Some(receptors) = correlation_receptors(correlation_type) else {
454                return Err(GainSolveError::UnsupportedCorrelationLayout {
455                    data_desc_id: row.data_desc_id,
456                    correlation_types: correlation_types
457                        .iter()
458                        .map(|code| stokes_name(*code).to_string())
459                        .collect(),
460                });
461            };
462            if receptors.0 != receptors.1 {
463                continue;
464            }
465            let graph_index = match options.gain_type {
466                GainType::G => receptors.0,
467                GainType::T => 0,
468            };
469            let mut channel_weights = Vec::with_capacity(channels.len());
470            for &chan_index in &channels {
471                let weight = self.accumulate_sample(
472                    graph_index,
473                    row,
474                    &inputs,
475                    options,
476                    corr_index,
477                    chan_index,
478                );
479                channel_weights.push(weight);
480            }
481            row_weight += if options.channelized {
482                f64::from(median_with_even_mean(&mut channel_weights))
483            } else {
484                channel_weights
485                    .iter()
486                    .map(|weight| f64::from(*weight))
487                    .sum()
488            };
489        }
490        if row_weight > 0.0 {
491            let centroid = self
492                .baseline_centroids
493                .entry((row.antenna1, row.antenna2))
494                .or_default();
495            centroid.weighted_sum += row_weight * row.time_centroid_seconds;
496            centroid.weight += row_weight;
497        }
498        Ok(())
499    }
500
501    fn observe_metadata(&mut self, row: &SelectedSolveRow) {
502        self.min_time = self.min_time.min(row.time_seconds);
503        self.max_time = self.max_time.max(row.time_seconds);
504        if self.unique_time_bits.insert(row.time_seconds.to_bits()) {
505            self.unique_time_sum += row.time_seconds;
506            self.unique_time_count += 1;
507        }
508        self.total_interval += row.interval_seconds;
509        self.sample_rows += 1;
510        self.field_ids.insert(row.field_id);
511        self.scan_numbers.insert(row.scan_number);
512        self.antenna_ids.insert(row.antenna1);
513        self.antenna_ids.insert(row.antenna2);
514    }
515
516    /// Accumulate one sample if it can enter the solve; return its effective
517    /// weight, zero when it is flagged or unusable.
518    fn accumulate_sample(
519        &mut self,
520        graph_index: usize,
521        row: &SelectedSolveRow,
522        inputs: &SolveRowInputs<'_>,
523        options: &SolveGroupOptions,
524        corr_index: usize,
525        chan_index: usize,
526    ) -> f32 {
527        let weight = inputs.weights[[corr_index, chan_index]];
528        if weight <= 0.0 || solve_sample_flagged(&inputs.flags, options.gain_type, chan_index) {
529            return 0.0;
530        }
531        let model = match inputs.model {
532            Some(model_data) => model_data[[corr_index, chan_index]],
533            None => Complex32::new(options.stokes_i, 0.0),
534        };
535        let model_norm_sqr = model.norm_sqr();
536        if model_norm_sqr <= f32::EPSILON {
537            return 0.0;
538        }
539        let sample = inputs.data[[corr_index, chan_index]] / model;
540        if sample.norm() <= f32::EPSILON {
541            return 0.0;
542        }
543        let effective_weight = weight * model_norm_sqr;
544        let accumulation = self.receptor_accumulations[graph_index]
545            .entry(SolveAccumulationKey {
546                baseline: (row.antenna1, row.antenna2),
547                correlation_index: corr_index,
548                channel_index: if options.channelized { chan_index } else { 0 },
549            })
550            .or_default();
551        accumulation.weighted_sum += sample * Complex32::new(effective_weight, 0.0);
552        accumulation.weight += effective_weight;
553        accumulation.weighted_sample_power +=
554            f64::from(effective_weight) * f64::from(sample.norm_sqr());
555        accumulation.sample_count += 1;
556        effective_weight
557    }
558
559    /// Collapse each baseline's samples per correlation and solution channel
560    /// into the solve graphs, and set the solution TIME.
561    ///
562    /// A phase-only solve sees each collapsed sample at unit amplitude with
563    /// its weight times its squared amplitude
564    /// (`SolveDataBuffer::enforceAPonData`). TIME is
565    /// `SDBList::aggregateTimeCentroid`: each baseline's TIME_CENTROID,
566    /// averaged over its rows by row weight as time averaging does, weighted
567    /// by the baseline's solve weight after that phase-only scaling.
568    pub(crate) fn finalize_for_solve(&mut self, solve_mode: GainSolveMode) {
569        self.channel_graphs = vec![
570            vec![SolveGraph::default(); self.receptor_accumulations.len()];
571            self.solution_channels
572        ];
573        let mut baseline_weights = HashMap::<(i32, i32), f64>::new();
574        for (receptor, accumulations) in self.receptor_accumulations.iter().enumerate() {
575            for (key, accumulation) in accumulations {
576                if accumulation.weight <= f32::EPSILON {
577                    continue;
578                }
579                let averaged = accumulation.weighted_sum / Complex32::new(accumulation.weight, 0.0);
580                let (sample, weight) = match solve_mode {
581                    GainSolveMode::Phase => {
582                        let amplitude = averaged.norm();
583                        if amplitude <= f32::EPSILON {
584                            continue;
585                        }
586                        (
587                            averaged / Complex32::new(amplitude, 0.0),
588                            accumulation.weight * amplitude * amplitude,
589                        )
590                    }
591                    GainSolveMode::AmplitudePhase => (averaged, accumulation.weight),
592                };
593                self.channel_graphs[key.channel_index][receptor].accumulate(
594                    key.baseline,
595                    weight,
596                    sample * Complex32::new(weight, 0.0),
597                    SolveEdgeStats {
598                        weighted_sample_power: f64::from(weight) * f64::from(sample.norm_sqr()),
599                        raw_weighted_sample_power: accumulation.weighted_sample_power,
600                        sample_count: accumulation.sample_count,
601                        collapsed_count: 1,
602                    },
603                );
604                *baseline_weights.entry(key.baseline).or_default() += f64::from(weight);
605            }
606        }
607
608        let mut solution_time = WeightedTime::default();
609        for (baseline, weight) in baseline_weights {
610            let Some(centroid) = self.baseline_centroids.get(&baseline) else {
611                continue;
612            };
613            if centroid.weight > 0.0 {
614                solution_time.weighted_sum += weight * centroid.weighted_sum / centroid.weight;
615                solution_time.weight += weight;
616            }
617        }
618        self.solution_time = if solution_time.weight > 0.0 {
619            solution_time.weighted_sum / solution_time.weight
620        } else {
621            self.aggregate_time()
622        };
623    }
624
625    /// The mean of the interval's distinct TIMEs (`SDBList::aggregateTime`,
626    /// CASA's TIME when no sample carries weight).
627    pub(crate) fn aggregate_time(&self) -> f64 {
628        if self.unique_time_count > 0 {
629            self.unique_time_sum / self.unique_time_count as f64
630        } else {
631            (self.min_time + self.max_time) / 2.0
632        }
633    }
634}

636/// Read one MAIN row's data, flags, model and solve weights, from the row or
637/// from its pre-applied calibration.
638///
639/// The weights are CASA's (`CalSolvingVi2::calibrateCurrentVB`, which calls
640/// `VisBufferImpl2::resetWeightsUsingSigma`): 1/σ² from SIGMA_SPECTRUM when
641/// the MeasurementSet has WEIGHT_SPECTRUM and SIGMA_SPECTRUM, otherwise from
642/// SIGMA broadcast over channels; σ ≤ 0 gives no weight. WEIGHT and
643/// WEIGHT_SPECTRUM are not read. Pre-applied `calwt` tables then scale them.
644fn read_row_inputs<'a>(
645    ms: &'a MeasurementSet,
646    row: &SelectedSolveRow,
647    preapplied_row: Option<&'a EvaluatedApplyRow>,
648    model_source: GainSolveModelSource,
649) -> Result<SolveRowInputs<'a>, GainSolveError> {
650    let (data, flags) = match preapplied_row {
651        Some(preapplied) => (&preapplied.corrected_data, &preapplied.flags),
652        None => (
653            main_cell(ms, row.row_index, "DATA")?,
654            main_cell(ms, row.row_index, "FLAG")?,
655        ),
656    };
657    let ArrayValue::Complex32(data) = data else {
658        return Err(unsupported_shape("DATA", data));
659    };
660    let ArrayValue::Bool(flags) = flags else {
661        return Err(unsupported_shape("FLAG", flags));
662    };
663    if data.ndim() != 2 || data.shape() != flags.shape() {
664        return Err(GainSolveError::UnsupportedParameterShape {
665            path: "<measurement-set row>".to_string(),
666            shape: data.shape().to_vec(),
667        });
668    }
669    let model = match model_source {
670        GainSolveModelSource::ModelColumn => {
671            let model = main_cell(ms, row.row_index, "MODEL_DATA")?;
672            match model {
673                ArrayValue::Complex32(model) if model.shape() == data.shape() => Some(model),
674                other => return Err(unsupported_shape("MODEL_DATA", other)),
675            }
676        }
677        GainSolveModelSource::PointSource => None,
678    };
679    let mut weights = solve_weights(ms, row.row_index, data.shape())?;
680    if let Some(factors) = preapplied_row.and_then(|row| row.weight_factors.as_ref()) {
681        if factors.shape() != weights.shape() {
682            return Err(GainSolveError::UnsupportedParameterShape {
683                path: "<pre-applied weight factors>".to_string(),
684                shape: factors.shape().to_vec(),
685            });
686        }
687        weights *= factors;
688    }
689    Ok(SolveRowInputs {
690        data,
691        flags: solve_flags(row, flags),
692        weights,
693        model,
694    })
695}

697fn solve_weights(
698    ms: &MeasurementSet,
699    row_index: usize,
700    shape: &[usize],
701) -> Result<ArrayD<f32>, GainSolveError> {
702    let has_column = |name| {
703        ms.main_table()
704            .schema()
705            .is_some_and(|schema| schema.contains_column(name))
706    };
707    let inverse_square = |sigma: f32| {
708        if sigma > 0.0 {
709            1.0 / (sigma * sigma)
710        } else {
711            0.0
712        }
713    };
714    if has_column("WEIGHT_SPECTRUM") && has_column("SIGMA_SPECTRUM") {
715        let sigma = main_cell(ms, row_index, "SIGMA_SPECTRUM")?;
716        return match sigma {
717            ArrayValue::Float32(sigma) if sigma.shape() == shape => Ok(sigma.mapv(inverse_square)),
718            other => Err(unsupported_shape("SIGMA_SPECTRUM", other)),
719        };
720    }
721    let sigma = main_cell(ms, row_index, "SIGMA")?;
722    let ArrayValue::Float32(sigma) = sigma else {
723        return Err(unsupported_shape("SIGMA", sigma));
724    };
725    if sigma.ndim() != 1 || sigma.shape()[0] != shape[0] {
726        return Err(GainSolveError::UnsupportedParameterShape {
727            path: "<measurement-set SIGMA>".to_string(),
728            shape: sigma.shape().to_vec(),
729        });
730    }
731    Ok(ArrayD::from_shape_fn(shape, |index| {
732        inverse_square(sigma[[index[0]]])
733    }))
734}
735
736fn main_cell<'a>(
737    ms: &'a MeasurementSet,
738    row_index: usize,
739    column: &str,
740) -> Result<&'a ArrayValue, GainSolveError> {
741    ms.main_table()
742        .cell_accessor(row_index, column)
743        .and_then(|cell| cell.array())
744        .map_err(|source| GainSolveError::OpenMeasurementSet {
745            path: ms
746                .path()
747                .map(|path| path.display().to_string())
748                .unwrap_or_else(|| "<in-memory>".to_string()),
749            source: MsError::from(source),
750        })
751}
752
753fn unsupported_shape(column: &str, value: &ArrayValue) -> GainSolveError {
754    GainSolveError::UnsupportedParameterShape {
755        path: format!("<measurement-set {column}>"),
756        shape: value.shape().to_vec(),
757    }
758}

760/// The median, the mean of the middle two for an even count, as casacore's
761/// `partialMedians(..., takeEvenMean=True)` that CASA uses for a row's
762/// channel-independent weight.
763fn median_with_even_mean(values: &mut [f32]) -> f32 {
764    if values.is_empty() {
765        return 0.0;
766    }
767    values.sort_by(f32::total_cmp);
768    let middle = values.len() / 2;
769    if values.len().is_multiple_of(2) {
770        (values[middle - 1] + values[middle]) / 2.0
771    } else {
772        values[middle]
773    }
774}

776fn selected_channels(
777    row: &SelectedSolveRow,
778    channel_count: usize,
779) -> Result<Vec<usize>, GainSolveError> {
780    match &row.channel_selection {

784                .map_err(|source| GainSolveError::OpenMeasurementSet {

786                    source,
787                })
788        }
789        None => Ok((0..channel_count).collect()),
790    }

793/// The sample flags a solve uses for `row`: its `FLAG` cell, or every sample
794/// flagged when the row's `FLAG_ROW` is set, whatever `FLAG` holds. CASA's
795/// solve does the same: `CalSolvingVi2::calibrateCurrentVB` calls
796/// `VisEquation::collapse2`, whose `divideCorrByModel` flags the whole cube
797/// of such a row and zeroes its weights before G, T or B accumulates it.
798pub(crate) fn solve_flags<'a>(
799    row: &SelectedSolveRow,
800    flags: &'a ArrayD<bool>,
801) -> Cow<'a, ArrayD<bool>> {
802    if row.flag_row {
803        Cow::Owned(ArrayD::from_elem(flags.raw_dim(), true))
804    } else {
805        Cow::Borrowed(flags)
806    }
807}
808
809/// Correlation-independent flags, as `CalSolvingVi2::corrIndepFlags` makes
810/// them: a channel any correlation flags is flagged for all.
811fn solve_sample_flagged(flags: &ArrayD<bool>, gain_type: GainType, chan_index: usize) -> bool {
812    match gain_type {
813        GainType::G | GainType::T => {
814            (0..flags.shape()[0]).any(|corr_index| flags[[corr_index, chan_index]])
815        }
816    }
817}
