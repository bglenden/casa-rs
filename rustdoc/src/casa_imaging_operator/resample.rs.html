2//! The single spectral resampler: native rows to placements.
3//!
4//! A constant or Taylor basis grids every native channel on plane 0. A
5//! channel-local basis follows CASA `FTMachine`: `nearest` maps a native
6//! channel to the output channel its rounded spectral pixel names
7//! (`matchChannel`); `linear` interpolates adjacent native visibilities onto
8//! CASA's fine frequency grid, keeps the nearer channel's weight and ORs the
9//! flags (`interpolateFrequencyTogrid`), except that a one-channel image or
10//! a one-channel row bypasses interpolation and maps like `nearest`.
11
12use num_complex::{Complex32, Complex64};

15use crate::error::OperatorError;
16use crate::operator::{Basis, MeasurementOperator};
17use crate::sample::{Placement, SampleBuffer};
18use crate::weighting::WeightingGeneration;

20const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;
21
22/// One native MeasurementSet row: every selected channel and correlation.
23///
24/// Arrays are `[channel][correlation]`, correlation fastest, over the
25/// selected correlations in the operator's routing order. Frequencies are
26/// finite and strictly monotonic; `uvw_m` is finite.
27#[derive(Clone, Copy, Debug)]
28pub struct NativeRow<'a> {
29    /// Baseline `[u, v, w]` in metres, in the image phase-centre frame.
30    pub uvw_m: [f64; 3],
31    /// Path-length shift to the image phase centre in metres; the phasor
32    /// `e^{i·2π·shift·ν/c}` is applied to the data.
33    pub phase_shift_m: f64,
34    /// Native channel centres in Hz.
35    pub frequencies_hz: &'a [f64],
36    /// Visibilities.
37    pub values: &'a [Complex32],
38    /// Input weights (`WEIGHT_SPECTRUM` or `WEIGHT` repeated per channel).
39    pub weights: &'a [f32],
40    /// Flags.
41    pub flags: &'a [bool],
42    /// Row flag.
43    pub row_flag: bool,
44    /// Per-row context for kernel-set cell selection.
45    pub context: RowContext,
46}

48/// CASA cube channel mapping rule.
49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50pub enum SpectralKernel {
51    /// Nearest output channel by rounded spectral pixel.
52    Nearest,
53    /// Linear interpolation onto CASA's fine frequency grid.
54    Linear,
55}

57/// The uniform output spectral axis of a channel-local basis: first channel
58/// centre, signed channel width and channel count. The width matters even
59/// for one channel, because it decides which native channels the channel
60/// accepts.
61#[derive(Clone, Copy, Debug, PartialEq)]
62pub struct SpectralAxis {
63    first_hz: f64,
64    increment_hz: f64,
65    channels: u32,
66}
67
68impl SpectralAxis {
69    /// An axis of `channels` channels of width `increment_hz` starting at
70    /// `first_hz`; the frequency must be finite and positive, the increment
71    /// finite and non-zero, and there must be at least one channel.
72    pub fn new(first_hz: f64, increment_hz: f64, channels: u32) -> Result<Self, OperatorError> {
73        if !first_hz.is_finite() || first_hz <= 0.0 {
74            return Err(OperatorError::SpectralAxis {
75                reason: "the first channel centre must be finite and positive",
76            });
77        }
78        if !increment_hz.is_finite() || increment_hz == 0.0 {
79            return Err(OperatorError::SpectralAxis {
80                reason: "the channel width must be finite and non-zero",
81            });
82        }
83        if channels == 0 {
84            return Err(OperatorError::SpectralAxis {
85                reason: "the axis has no channels",
86            });
87        }
88        Ok(Self {
89            first_hz,
90            increment_hz,
91            channels,
92        })
93    }
94
95    /// Centre of the first channel in Hz.
96    #[must_use]
97    pub const fn first_hz(self) -> f64 {
98        self.first_hz
99    }
100
101    /// Signed channel width in Hz.
102    #[must_use]
103    pub const fn increment_hz(self) -> f64 {
104        self.increment_hz
105    }
106
107    /// Number of channels.
108    #[must_use]
109    pub const fn channels(self) -> u32 {
110        self.channels
111    }
112
113    /// Centre of `channel` in Hz.
114    #[must_use]
115    pub fn centre_hz(self, channel: u32) -> f64 {
116        self.first_hz + f64::from(channel) * self.increment_hz
117    }
118
119    fn last_hz(self) -> f64 {
120        self.centre_hz(self.channels - 1)
121    }
122
123    /// `FTMachine::matchChannel`: the channel whose rounded spectral pixel
124    /// `floor((ν − ν₀)/Δν + 0.5)` lies on the axis.
125    #[must_use]
126    pub fn nearest_channel(self, frequency_hz: f64) -> Option<u32> {
127        let pixel = ((frequency_hz - self.first_hz) / self.increment_hz + 0.5).floor();
128        (pixel >= 0.0 && pixel < f64::from(self.channels)).then_some(pixel as u32)
129    }
130}

132/// CASA's fine frequency grid for linear interpolation: the output centres
133/// when output channels are no wider than native ones, otherwise
134/// `floor(width ratio)` points per output channel.
135#[derive(Clone, Copy, Debug, PartialEq)]
136struct FineGrid {
137    start_hz: f64,
138    increment_hz: f64,
139    per_output: usize,
140    output: SpectralAxis,
141}
142
143impl FineGrid {
144    fn compile(output: SpectralAxis, native_increment_hz: f64) -> Option<Self> {
145        let output_increment_hz = output.increment_hz;
146        if native_increment_hz == 0.0 || !native_increment_hz.is_finite() {
147            return None;
148        }
149        let width = output_increment_hz.abs() / native_increment_hz.abs();
150        if width <= 1.0 {
151            let increment_hz = output_increment_hz.abs().copysign(native_increment_hz);
152            let start_hz = if increment_hz.signum() == output_increment_hz.signum() {
153                output.first_hz
154            } else {
155                output.last_hz()
156            };
157            return Some(Self {
158                start_hz,
159                increment_hz,
160                per_output: 1,
161                output,
162            });
163        }
164        let per_output = width.floor() as usize;
165        let fine_abs = output_increment_hz.abs() / per_output as f64;
166        let first_edge = output.first_hz - output_increment_hz / 2.0;
167        let last_edge = output.last_hz() + output_increment_hz / 2.0;
168        let low_edge = first_edge.min(last_edge);
169        let high_edge = first_edge.max(last_edge);
170        let increment_hz = fine_abs.copysign(native_increment_hz);
171        let start_hz = if increment_hz > 0.0 {
172            low_edge + fine_abs / 2.0
173        } else {
174            high_edge - fine_abs / 2.0
175        };
176        Some(Self {
177            start_hz,
178            increment_hz,
179            per_output,
180            output,
181        })
182    }
183
184    fn count(self) -> usize {
185        self.per_output * self.output.channels as usize
186    }
187
188    fn frequency_hz(self, ordinal: usize) -> f64 {
189        self.start_hz + ordinal as f64 * self.increment_hz
190    }
191
192    fn output_channel(self, ordinal: usize) -> u32 {
193        let output_ordinal = (ordinal / self.per_output) as u32;
194        if self.increment_hz.signum() == self.output.increment_hz.signum() {
195            output_ordinal
196        } else {
197            self.output.channels - 1 - output_ordinal
198        }
199    }
200}

202#[derive(Clone, Copy, Debug, PartialEq)]
203enum Sampling {
204    Direct,
205    Nearest(SpectralAxis),
206    Linear(SpectralAxis),
207}
208
209/// One spectral sample of a row: its native source.
210#[derive(Clone, Copy, Debug)]
211enum Source {
212    Channel(usize),
213    Pair { left: usize, right_factor: f64 },
214}

216/// Turns native rows into placed, weighted samples for one operator.
217#[derive(Clone, Copy, Debug, PartialEq)]
218pub struct SpectralResampler {
219    sampling: Sampling,
220    basis: Basis,
221}
222
223impl SpectralResampler {
224    /// Resampler for a constant or Taylor basis: every native channel is a
225    /// sample on plane 0.
226    pub fn direct(basis: Basis) -> Result<Self, OperatorError> {
227        if matches!(basis, Basis::ChannelLocal { .. }) {
228            return Err(OperatorError::SpectralAxis {
229                reason: "a channel-local basis needs its output axis and kernel",
230            });
231        }
232        Ok(Self {
233            sampling: Sampling::Direct,
234            basis,
235        })
236    }
237
238    /// Resampler for the channel-local basis whose planes are the channels
239    /// of `output`, mapped by `kernel`.
240    #[must_use]
241    pub fn channel_local(output: SpectralAxis, kernel: SpectralKernel) -> Self {
242        Self {
243            sampling: match kernel {
244                SpectralKernel::Nearest => Sampling::Nearest(output),
245                SpectralKernel::Linear => Sampling::Linear(output),
246            },
247            basis: Basis::ChannelLocal {
248                planes: output.channels,
249            },
250        }
251    }
252
253    /// The basis the resampler produces placements for.
254    #[must_use]
255    pub const fn basis(&self) -> Basis {
256        self.basis
257    }
258
259    /// Place one row's unflagged samples with imaging weights into `out`.
260    ///
261    /// Each sample's value is `W · V · e^{iφ}` and its weight `W` for every
262    /// polarization; samples whose kernel support leaves the padded grid or
263    /// whose imaging weight is zero are dropped.
264    pub fn place(
265        &self,
266        operator: &MeasurementOperator,
267        weighting: &WeightingGeneration,
268        row: &NativeRow<'_>,
269        out: &mut SampleBuffer,
270    ) -> Result<(), OperatorError> {
271        let npol = self.validate(operator, row, out)?;
272        let cf = operator.cf();
273        let geometry = operator.geometry();
274        let mut values = vec![Complex32::default(); npol];
275        let mut weights = vec![0.0_f32; npol];
276        self.for_each_sample(row, |plane, frequency_hz, source| {
277            let Some(input_weight) = self.input_weight(row, npol, source) else {
278                return;
279            };
280            let scale = frequency_hz / SPEED_OF_LIGHT_M_PER_S;
281            let w = row.uvw_m[2] * scale;
282            let key = cf.key(&row.context, frequency_hz, w);
283            let taps = cf.taps(key);
284            let u = row.uvw_m[0] * scale;
285            let v = row.uvw_m[1] * scale;
286            let location = geometry.locate(u, v, taps.oversampling());
287            if !geometry.fits(location, taps.half_support()) {
288                return;
289            }
290            let placement = Placement {
291                u,
292                v,
293                w,
294                phase: std::f64::consts::TAU * row.phase_shift_m * scale,
295                plane,
296                spectral: self.basis.spectral(frequency_hz),
297                cf: key,
298                gradient: [0.0, 0.0],
299            };
300            let weight = weighting.imaging_weight(&placement, input_weight);
301            if weight <= 0.0 {
302                return;
303            }
304            let phasor = Complex64::from_polar(f64::from(weight), placement.phase);
305            for pol in 0..npol {
306                let value = self.value(row, npol, source, pol);
307                let scaled = Complex64::new(f64::from(value.re), f64::from(value.im)) * phasor;
308                values[pol] = Complex32::new(scaled.re as f32, scaled.im as f32);
309                weights[pol] = weight;
310            }
311            out.push(placement, &values, &weights);
312        });
313        Ok(())
314    }

316    /// Place one row's density-pass samples: one polarization carrying
317    /// CASA's unpolarized input weight, zero values, no support test.
318    pub fn place_density(
319        &self,
320        operator: &MeasurementOperator,
321        row: &NativeRow<'_>,
322        out: &mut SampleBuffer,
323    ) -> Result<(), OperatorError> {
324        let npol = operator.polarization().correlations().len();
325        self.validate_row(operator, row, npol)?;
326        if out.npol() != 1 {
327            return Err(OperatorError::NativeRow {
328                reason: "density buffers carry one polarization",
329            });
330        }
331        let cf = operator.cf();
332        self.for_each_sample(row, |plane, frequency_hz, source| {
333            let Some(input_weight) = self.input_weight(row, npol, source) else {
334                return;
335            };
336            let scale = frequency_hz / SPEED_OF_LIGHT_M_PER_S;
337            let w = row.uvw_m[2] * scale;
338            let placement = Placement {
339                u: row.uvw_m[0] * scale,
340                v: row.uvw_m[1] * scale,
341                w,
342                phase: 0.0,
343                plane,
344                spectral: self.basis.spectral(frequency_hz),
345                cf: cf.key(&row.context, frequency_hz, w),
346                gradient: [0.0, 0.0],
347            };
348            out.push(placement, &[Complex32::default()], &[input_weight]);
349        });
350        Ok(())
351    }
352
353    fn validate(
354        &self,
355        operator: &MeasurementOperator,
356        row: &NativeRow<'_>,
357        out: &SampleBuffer,
358    ) -> Result<usize, OperatorError> {
359        let npol = operator.polarization().correlations().len();
360        if out.npol() != npol {
361            return Err(OperatorError::NativeRow {
362                reason: "buffer polarizations must match the operator",
363            });
364        }
365        self.validate_row(operator, row, npol)?;
366        Ok(npol)
367    }
368
369    fn validate_row(
370        &self,
371        operator: &MeasurementOperator,
372        row: &NativeRow<'_>,
373        npol: usize,
374    ) -> Result<(), OperatorError> {
375        assert_eq!(
376            self.basis,
377            operator.basis(),
378            "the resampler and the operator must share a basis"
379        );
380        let channels = row.frequencies_hz.len();
381        if channels == 0 {
382            return Err(OperatorError::NativeRow {
383                reason: "row has no channels",
384            });
385        }
386        if row.values.len() != channels * npol
387            || row.weights.len() != channels * npol
388            || row.flags.len() != channels * npol
389        {
390            return Err(OperatorError::NativeRow {
391                reason: "values, weights and flags must be channels × correlations",
392            });
393        }
394        Ok(())
395    }

397    /// Visit every spectral sample of the row as `(plane, frequency, source)`.
398    fn for_each_sample(&self, row: &NativeRow<'_>, mut visit: impl FnMut(u32, f64, Source)) {
399        let native = row.frequencies_hz;
400        let nearest = |axis: SpectralAxis, visit: &mut dyn FnMut(u32, f64, Source)| {
401            for (channel, frequency_hz) in native.iter().enumerate() {
402                if let Some(plane) = axis.nearest_channel(*frequency_hz) {
403                    visit(plane, *frequency_hz, Source::Channel(channel));
404                }
405            }
406        };
407        match self.sampling {
408            Sampling::Direct => {
409                for (channel, frequency_hz) in native.iter().enumerate() {
410                    visit(0, *frequency_hz, Source::Channel(channel));
411                }
412            }
413            Sampling::Nearest(axis) => nearest(axis, &mut visit),
414            // CASA bypasses interpolation for a one-channel image or row.
415            Sampling::Linear(axis) if axis.channels == 1 || native.len() == 1 => {
416                nearest(axis, &mut visit);
417            }
418            Sampling::Linear(axis) => {
419                let Some(grid) = FineGrid::compile(axis, native[1] - native[0]) else {
420                    return;
421                };
422                let count = grid.count();
423                let ascending = grid.increment_hz > 0.0;
424                let mut next = 0;
425                for left in 0..native.len() - 1 {
426                    let left_hz = native[left];
427                    let right_hz = native[left + 1];
428                    let span = right_hz - left_hz;
429                    while next < count {
430                        let frequency_hz = grid.frequency_hz(next);
431                        let before_left = if ascending {
432                            frequency_hz < left_hz
433                        } else {
434                            frequency_hz > left_hz
435                        };
436                        if before_left {
437                            next += 1;
438                            continue;
439                        }
440                        let after_right = if ascending {
441                            frequency_hz > right_hz
442                        } else {
443                            frequency_hz < right_hz
444                        };
445                        if after_right {
446                            break;
447                        }
448                        let right_factor = ((frequency_hz - left_hz) / span).clamp(0.0, 1.0);
449                        visit(
450                            grid.output_channel(next),
451                            frequency_hz,
452                            Source::Pair { left, right_factor },
453                        );
454                        next += 1;
455                    }
456                }
457            }
458        }
459    }

461    /// CASA's unpolarized input weight `(w_first + w_last)/2` of a sample,
462    /// or `None` when the row or any selected correlation is flagged.
463    fn input_weight(&self, row: &NativeRow<'_>, npol: usize, source: Source) -> Option<f32> {
464        if row.row_flag {
465            return None;
466        }
467        let (channel, right_factor) = match source {
468            Source::Channel(channel) => (channel, None),
469            Source::Pair { left, right_factor } => (left, Some(right_factor)),
470        };
471        let flagged = |channel: usize| {
472            row.flags[channel * npol..(channel + 1) * npol]
473                .iter()
474                .any(|flag| *flag)
475        };
476        let unpolarized = |channel: usize| {
477            let weights = &row.weights[channel * npol..(channel + 1) * npol];
478            (weights[0] + weights[npol - 1]) / 2.0
479        };
480        let (flag, weight) = match right_factor {
481            None => (flagged(channel), unpolarized(channel)),
482            Some(right_factor) => {
483                let left_flag = flagged(channel);
484                let right_flag = flagged(channel + 1);
485                let flag = if right_factor <= f64::EPSILON {
486                    left_flag
487                } else if right_factor >= 1.0 - f64::EPSILON {
488                    right_flag
489                } else {
490                    left_flag || right_flag
491                };
492                // CASA nearest-weight interpolation keeps the left element at a tie.
493                let nearest = if right_factor > 0.5 {
494                    channel + 1
495                } else {
496                    channel
497                };
498                (flag, unpolarized(nearest))
499            }
500        };
501        if flag || !weight.is_finite() || weight <= 0.0 {
502            return None;
503        }
504        Some(weight)
505    }
506
507    fn value(&self, row: &NativeRow<'_>, npol: usize, source: Source, pol: usize) -> Complex32 {
508        match source {
509            Source::Channel(channel) => row.values[channel * npol + pol],
510            Source::Pair { left, right_factor } => {
511                let left_value = row.values[left * npol + pol];
512                let right_value = row.values[(left + 1) * npol + pol];
513                let left_factor = (1.0 - right_factor) as f32;
514                left_value * left_factor + right_value * right_factor as f32
515            }
516        }
517    }
