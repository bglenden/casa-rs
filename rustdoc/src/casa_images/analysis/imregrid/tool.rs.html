24/// How the target grid is constructed from a coordinate record or selected image.
25#[derive(Clone, Copy)]
26pub enum ImageRegridTarget<'a> {
27    /// Target CASA record. An empty target retains each source axis and length.
28    Coordinates(Option<&'a RecordValue>),
29    /// Rotate the selected direction or linear coordinate by this many radians.
30    Rotation(f64),
31}
32
33/// Output coordinates, selection and sampling controls for the public image tools.
34pub struct ImageRegridOptions<'a> {
35    /// Destination pixel lengths, or the source shape when omitted.
36    pub shape: &'a [i64],
37    /// The coordinate record or rotation used to construct the target grid.
38    pub target: ImageRegridTarget<'a>,
39    /// Pixel axes to regrid, or the compatible nondegenerate axes by default.
40    pub axes: &'a [i64],
41    /// Input region read before interpolation.
42    pub region: Option<SubimageRegion<'a>>,
43    /// Lattice expression contributing to the input validity mask.
44    pub mask: Option<&'a str>,
45    /// Interpolation method as accepted by casacore.
46    pub method: &'a str,
47    /// Sampling interval for computing the plane's coordinate map.
48    pub decimate: i64,
49    /// Replicate by relative pixel position.
50    pub replicate: bool,
51    /// Convert direction and spectral reference frames during sampling.
52    pub convert_reference: bool,
53    /// Remove singleton axes from the input region before regridding.
54    pub dropdeg: bool,
55    /// Interpolate even when the two grids already match.
56    pub force: bool,
57    /// Interpolate the spectral coordinate in velocity space.
58    pub as_velocity: bool,
59    /// Broadcast a conformant mask over matching singleton axes.
60    pub stretch: bool,
61    /// Public output destination, used to guard named mask operands.
62    pub outfile: Option<&'a Path>,
63}

65/// Regridded pixels and the native coordinate and restoring-beam metadata.
66pub struct ImageRegridProduct<T> {
67    /// Output values, validity and typed coordinate system.
68    pub regridded: Regridded<T>,
69    /// Output coordinate record for temporary attachments and native storage.
70    pub coordinate_record: RecordValue,
71    /// Image type and the selected subset of the input restoring beams.
72    pub image_info: ImageInfo,
73    /// Original channel and polarization indices for the output beams.
74    pub beam_planes: [Vec<usize>; 2],
75}

77/// Regrid an open image onto a CASA coordinate record.
78/// # Errors
79/// An invalid output grid, unsupported region, unreadable mask or pixels, or
80/// a destination whose grid has no overlap with the selected input.
81pub fn regrid_image<T: RegridPixel>(
82    input: &dyn ImageInterface<T>,
83    native: RecordValue,
84    beam_record: &RecordValue,
85    options: ImageRegridOptions<'_>,
86) -> Result<ImageRegridProduct<T>, ImageError> {
87    let rotating = matches!(options.target, ImageRegridTarget::Rotation(_));
88    if !rotating && is_pv_grid(input.coordinates()) {
89        return Err(invalid("PV images are not supported by regrid"));
90    }
91    let shape_specified = options.shape.first().is_some_and(|&size| size >= 0);
92    let shape: Vec<i64> = if !shape_specified {
93        input
94            .shape()
95            .iter()
96            .copied()
97            .filter(|&size| !(rotating && options.dropdeg && size == 1))
98            .map(|size| size as i64)
99            .collect()
100    } else {
101        options.shape.to_vec()
102    };
103    let output_shape = shape
104        .iter()
105        .map(|&size| {
106            usize::try_from(size)
107                .ok()
108                .filter(|&size| size > 0)
109                .ok_or_else(|| invalid("output shape lengths must be positive"))
110        })
111        .collect::<Result<Vec<_>, _>>()?;
112    let explicit_target = match options.target {
113        ImageRegridTarget::Coordinates(Some(record)) => {
114            Some(CoordinateSystem::from_record(record)?)
115        }
116        _ => None,
117    };
118    let info = ImageInfo::from_record(beam_record)?;
119    if let (Some(output), Some(mask)) = (options.outfile, options.mask) {
120        for operand in ImageToolExpressionInputs::resolve(mask)?.paths() {
121            ensure_outfile_does_not_alias_input(output, operand)?;
122        }
123    }
124
125    let selection = (options.region.is_some() || options.mask.is_some() || options.dropdeg)
126        .then(|| {
127            SubimagePlan::new_with_beams(
128                input,
129                Some(native.clone()),
130                SubimageOptions {
131                    region: options.region,
132                    dropdeg: options.dropdeg,
133                    keepaxes: &[],
134                },
135                Some(beam_record),
136            )
137        })
138        .transpose()?;
139    if let Some(selection) = &selection {
140        selection.require_single_region()?;
141    }
142    let (selected_coords, selected_shape) = selection
143        .as_ref()
144        .map_or((input.coordinates(), input.shape()), |selection| {
145            (&selection.coordinates, selection.shape.as_slice())
146        });
147    let ndim = selected_shape.len();
148    if rotating && is_pv_grid(selected_coords) {
149        return Err(invalid("PV images are not supported by rotate"));
150    }
151    let has_rotation_shape = rotating && !options.shape.is_empty() && options.shape != [-1];
152    if output_shape.len() != ndim || (has_rotation_shape && options.shape.len() != ndim) {
153        return Err(invalid(
154            "The number of pixel axes in the output shape and Coordinate System must be the same",
155        ));
156    }
157    let rotation_axes;
158    let (target, axes) = match options.target {
159        ImageRegridTarget::Coordinates(_) => (
160            explicit_target.unwrap_or_else(CoordinateSystem::new),
161            options.axes,
162        ),
163        ImageRegridTarget::Rotation(angle) => {
164            let (target, axes) =
165                rotated_grid(selected_coords, angle, selected_shape, &output_shape)?;
166            rotation_axes = axes;
167            (target, rotation_axes.as_slice())
168        }
169    };
170    let all_axes = axes.is_empty() || axes == [-1];
171    if info.beam_set.is_multi()
172        && target.find_coordinate(CoordinateType::Spectral).is_some()
173        && let Some(axis) = selected_coords
174            .find_coordinate(CoordinateType::Spectral)
175            .and_then(|index| selected_coords.coordinate_pixel_axes(index))
176            .and_then(|axes| axes.into_iter().flatten().next())
177        && (all_axes || axes.contains(&(axis as i64)))
178    {
179        return Err(invalid(
180            "An image with multiple beams cannot be regridded along the spectral axis",
181        ));
182    }
183    let selected = selection
184        .as_ref()
185        .map(|selection| {
186            let mut image = TempImage::new(
187                selection.shape.clone(),
188                selection.coordinates.clone(),
189                TempStoragePolicy::Auto {
190                    memory_budget_bytes: 64 * 1024 * 1024,
191                    scratch: ScratchSpace::SystemTemp,
192                },
193            )?;
194            selection.copy_to(input, &mut image, options.mask, options.stretch)?;
195            Ok::<_, ImageError>(image)
196        })
197        .transpose()?;
198    let source = selected
199        .as_ref()
200        .map_or(input, |image| image as &dyn ImageInterface<T>);
201    let template = RegridTemplate {
202        shape: template_shape(source.coordinates(), &target, &output_shape),
203        coordinates: target,
204    };
205    let regridded = regrid_task(
206        source,
207        &template,
208        TaskRegridOptions {
209            axes,
210            shape: &shape,
211            as_velocity: options.as_velocity,
212            decimate: options.decimate.max(0),
213            replicate: options.replicate,
214            method: RegridMethod::from_image_tool(options.method)?,
215            image_tool: Some(ImageToolSampling {
216                force: options.force,
217                convert_reference: options.convert_reference,
218                shape_specified,
219                half_turn: matches!(options.target, ImageRegridTarget::Rotation(angle) if angle.rem_euclid(TAU) == PI),
220            }),
221        },
222    )?;
223    let calculated_record = regridded.coordinates.to_record();
224    let coordinate_record = if calculated_record == source.coordinates().to_record() {
225        if let Some(selection) = &selection {
226            selection
227                .coordinate_record
228                .clone()
229                .unwrap_or(calculated_record)
230        } else {
231            native
232        }
233    } else {
234        calculated_record
235    };
236    let (image_info, beam_planes) = output_beams(input, &regridded, info, selection.as_ref())?;
237    Ok(ImageRegridProduct {
238        regridded,
239        coordinate_record,
240        image_info,
241        beam_planes,
242    })
243}

245fn is_pv_grid(coordinates: &CoordinateSystem) -> bool {
246    coordinates
247        .find_coordinate(CoordinateType::Spectral)
248        .and_then(|index| coordinates.coordinate_pixel_axes(index))
249        .is_some_and(|axes| axes.into_iter().any(|axis| axis.is_some()))
250        && coordinates
251            .find_coordinate(CoordinateType::Linear)
252            .is_some_and(|index| {
253                coordinates
254                    .coordinate_pixel_axes(index)
255                    .expect("existing coordinate")
256                    .into_iter()
257                    .zip(coordinates.coordinate(index).axis_names())
258                    .any(|(axis, name)| axis.is_some() && name == "Offset")
259            })
260}

262fn template_shape(
263    input: &CoordinateSystem,
264    target: &CoordinateSystem,
265    shape: &[usize],
266) -> Vec<usize> {
267    let mut target_shape = vec![1; target.n_pixel_axes()];
268    for index in 0..target.n_coordinates() {
269        let Some(input_index) = input.find_coordinate(target.coordinate(index).coordinate_type())
270        else {
271            continue;
272        };
273        let input_axes = input
274            .coordinate_pixel_axes(input_index)
275            .expect("input coordinate exists");
276        let target_axes = target
277            .coordinate_pixel_axes(index)
278            .expect("target coordinate exists");
279        for (from, to) in input_axes.into_iter().zip(target_axes) {
280            if let (Some(from), Some(to)) = (from, to) {
281                target_shape[to] = shape[from];
282            }
283        }
284    }
285    target_shape
286}

288fn output_beams<T: RegridPixel>(
289    input: &dyn ImageInterface<T>,
290    product: &Regridded<T>,
291    mut info: ImageInfo,
292    selection: Option<&SubimagePlan>,
293) -> Result<(ImageInfo, [Vec<usize>; 2]), ImageError> {
294    let mut planes = if let Some(plan) = selection {
295        let section = &plan.selection;
296        beam_section_indices(
297            &info.beam_set,
298            input.coordinates(),
299            input.ndim(),
300            &section.start,
301            &section.packed_shape(),
302            section.stride.as_deref(),
303        )
304    } else {
305        [
306            (0..info.beam_set.n_channels()).collect(),
307            (0..info.beam_set.n_stokes()).collect(),
308        ]
309    };
310    if info.beam_set.is_multi() {
311        let output_stokes = product
312            .coordinates
313            .find_coordinate(CoordinateType::Stokes)
314            .and_then(|index| product.coordinates.coordinate_pixel_axes(index))
315            .and_then(|axes| axes.into_iter().flatten().next());
316        if let (Some(input_axis), Some(output_axis)) =
317            (input.find_axis(CoordinateType::Stokes), output_stokes)
318            && info.beam_set.n_stokes() > 1
319        {
320            let original = stokes_for_axis(input.coordinates(), input_axis)?;
321            planes[1] = stokes_for_axis(&product.coordinates, output_axis)?
322                .iter()
323                .map(|pol| {
324                    original
325                        .iter()
326                        .position(|value| value == pol)
327                        .ok_or_else(|| invalid("output polarization has no input restoring beam"))
328                })
329                .collect::<Result<_, _>>()?;
330        }
331        info.beam_set = info.beam_set.subset(&planes[0], &planes[1])?;
332    }
333    Ok((info, planes))
334}
