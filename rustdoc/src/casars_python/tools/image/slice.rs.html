15struct SliceVector {
16    values: Vec<f64>,
17    origin: VectorOrigin,
18}
19
20enum VectorOrigin {
21    Sequence,
22    Scalar,
23    Text,
24    OtherScalar,
25}
26
27fn vector_error() -> PyErr {
28    PySystemError::new_err(
29        "<built-in function image_getslice> returned a result with an exception set",
30    )
31}
32
33impl SliceVector {
34    fn default_values(values: &[f64]) -> Self {
35        Self {
36            values: values.to_vec(),
37            origin: VectorOrigin::Sequence,
38        }
39    }
40
41    fn indices(self, argument: &str) -> PyResult<Vec<i64>> {
42        match self.origin {
43            VectorOrigin::Text => {
44                return Err(PyTypeError::new_err(format!(
45                    "argument _{argument} must not be a string"
46                )));
47            }
48            VectorOrigin::OtherScalar => return Err(vector_error()),
49            _ => (),
50        }
51        let mut values: Vec<_> = self
52            .values
53            .into_iter()
54            .map(|value| i64::from(value as i32))
55            .collect();
56        if argument == "axes" && matches!(self.origin, VectorOrigin::Scalar) {
57            // Scalar assignment replaces the first default axis only.
58            values.push(1);
59        }
60        Ok(values)
61    }
62}

64impl FromPyObject<'_, '_> for SliceVector {
65    type Error = PyErr;
66
67    fn extract(value: Borrowed<'_, '_, PyAny>) -> PyResult<Self> {
68        if value.is_instance_of::<PyTuple>() {
69            return Err(vector_error());
70        }
71        if value.is_none() {
72            return Ok(Self::default_values(&[]));
73        }
74        if value.is_instance_of::<PyString>() || value.is_instance_of::<PyBytes>() {
75            return Ok(Self {
76                values: vec![-1.0],
77                origin: VectorOrigin::Text,
78            });
79        }
80        let flat = if value.is_instance_of::<PyUntypedArray>() {
81            value.call_method1("ravel", ("F",))?
82        } else if let Ok(list) = value.cast::<PyList>() {
83            if list
84                .iter()
85                .any(|item| item.is_instance_of::<PyList>() || item.is_instance_of::<PyTuple>())
86            {
87                let py = value.py();
88                let args = PyDict::new(py);
89                args.set_item("dtype", "object")?;
90                py.import("numpy")?
91                    .call_method("asarray", (list,), Some(&args))?
92                    .call_method1("ravel", ("F",))?
93            } else {
94                value.to_owned()
95            }
96        } else {
97            let origin = if value.is_instance_of::<PyInt>() || value.is_instance_of::<PyFloat>() {
98                VectorOrigin::Scalar
99            } else {
100                VectorOrigin::OtherScalar
101            };
102            return Ok(Self {
103                values: vec![value.extract::<f64>()?],
104                origin,
105            });
106        };
107        let values = flat
108            .try_iter()?
109            .map(|item| item.map(|item| double_vector_element(&item).unwrap_or(0.0)))
110            .collect::<PyResult<Vec<_>>>()?;
111        Ok(Self {
112            values,
113            origin: VectorOrigin::Sequence,
114        })
115    }
116}

118struct SliceMethod(String);
119
120impl SliceMethod {
121    fn parse(&self) -> PyResult<RegridMethod> {
122        match self.0.as_bytes().first().map(u8::to_ascii_uppercase) {
123            Some(b'N') => Ok(RegridMethod::Nearest),
124            Some(b'L') => Ok(RegridMethod::Linear),
125            Some(b'C') => Ok(RegridMethod::Cubic),
126            _ => Err(raised("Illegal interpolation method")),
127        }
128    }
129}
130
131impl FromPyObject<'_, '_> for SliceMethod {
132    type Error = PyErr;
133
134    fn extract(value: Borrowed<'_, '_, PyAny>) -> PyResult<Self> {
135        if let Ok(bytes) = value.cast::<PyBytes>() {
136            return Ok(Self(String::from_utf8_lossy(bytes.as_bytes()).into_owned()));
137        }
138        value
139            .extract::<String>()
140            .map(Self)
141            .map_err(|_| PyTypeError::new_err("argument _method must be a string"))
142    }
143}

145fn record<T>(
146    slice: PixelSlice<T>,
147    axes: [i64; 2],
148    pixels: impl FnOnce(Vec<T>) -> CasaValue,
149) -> CasaValue {
150    let count = slice.positions.len();
151    CasaValue::record([
152        (
153            "axes",
154            CasaValue::Ints {
155                shape: vec![2],
156                values: axes.to_vec(),
157            },
158        ),
159        (
160            "distance",
161            CasaValue::floats(slice.distance.into_iter().map(f64::from).collect()),
162        ),
163        (
164            "mask",
165            CasaValue::Bools {
166                shape: vec![count],
167                values: slice.mask,
168            },
169        ),
170        ("pixel", pixels(slice.pixels)),
171        (
172            "xpos",
173            CasaValue::floats(
174                slice
175                    .positions
176                    .iter()
177                    .map(|point| f64::from(point[0]))
178                    .collect(),
179            ),
180        ),
181        (
182            "ypos",
183            CasaValue::floats(
184                slice
185                    .positions
186                    .into_iter()
187                    .map(|point| f64::from(point[1]))
188                    .collect(),
189            ),
190        ),
191    ])
192}

194fn complex(values: Vec<Complex64>) -> CasaValue {
195    CasaValue::Complexes {
196        shape: vec![values.len()],
197        values,
198    }
199}

201#[pymethods]
202impl PyImageTool {
203    #[allow(clippy::too_many_arguments)]
204    #[pyo3(signature = (x, y, axes=SliceVector::default_values(&[0.0, 1.0]), coord=SliceVector::default_values(&[-1.0]), npts=0, method=SliceMethod("linear".into())), text_signature = "($self, x, y, axes=[0, 1], coord=[-1], npts=0, method='linear')")]
205    fn getslice(
206        &self,
207        py: Python<'_>,
208        x: SliceVector,
209        y: SliceVector,
210        axes: SliceVector,
211        coord: SliceVector,
212        npts: i32,
213        method: SliceMethod,
214    ) -> PyResult<Py<PyAny>> {
215        let axes = axes.indices("axes")?;
216        let axes = [
217            axes.first().copied().unwrap_or(-1),
218            axes.get(1).copied().unwrap_or(-1),
219        ];
220        let coord = coord.indices("coord")?;
221        let output = py.detach(|| {
222            let state = self.state();
223            let Some(attachment) = state.image.as_ref() else {
224                return Ok(CasaValue::empty_record());
225            };
226            let npts = usize::try_from(npts).map_err(|_| raised("npts must be non-negative"))?;
227            let method = method.parse()?;
228            let mut image = attachment.lock();
229            attachment.require_pixel_read_lock(&mut image)?;
230            macro_rules! read {
231                ($view:expr, $pixels:expr) => {
232                    get_slice($view, &x.values, &y.values, axes, &coord, npts, method)
233                        .map(|slice| record(slice, axes, $pixels))
234                        .map_err(raised)
235                };
236            }
237            let result = match &*image {
238                ImageHandle::Float(Storage::Temp {
239                    image,
240                    fits: Some(fits),
241                    ..
242                }) => {
243                    read!(&fits.image_view(image), |pixels: Vec<f32>| {
244                        CasaValue::floats(pixels.into_iter().map(f64::from).collect())
245                    })
246                }
247                ImageHandle::Float(storage) => with_storage!(storage, view => {
248                    read!(view.as_ref(), |pixels: Vec<f32>| {
249                        CasaValue::floats(pixels.into_iter().map(f64::from).collect())
250                    })
251                }),
252                ImageHandle::Double(storage) => with_storage!(storage, view => {
253                    read!(view.as_ref(), CasaValue::floats)
254                }),
255                ImageHandle::Complex(storage) => with_storage!(storage, view => {
256                    read!(view.as_ref(), |pixels: Vec<Complex32>| {
257                        complex(pixels.into_iter().map(|value| {
258                            Complex64::new(f64::from(value.re), f64::from(value.im))
259                        }).collect())
260                    })
261                }),
262                ImageHandle::DComplex(storage) => with_storage!(storage, view => {
263                    read!(view.as_ref(), complex)
264                }),
265            };
266            let release = attachment.release_requested_lock(&mut image);
267            result.and_then(|value| release.map(|()| value))
268        })?;
269        output.into_python(py)
270    }
271}
