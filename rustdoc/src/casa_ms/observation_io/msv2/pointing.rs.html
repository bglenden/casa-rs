2//! The MSv2 adapter's pointing: the `POINTING` rows of the requested
3//! antennas over the requested times (and each antenna's nearest row on
4//! either side), read once as the production selected-observation source
5//! reads them (`selected_pointing.rs`), as the boundary's series.
6
7use casa_observation::{ObservationError, PointingDirection, PointingRequest, PointingSeries};
8use casa_types::measures::direction::DirectionRef;
9
10use super::{Msv2Observation, storage};
11use crate::selected_pointing::SCALAR_SCAN_BYTES_PER_ROW;
12use crate::{MsError, PointingDirectionColumn, PointingReadPlan, SelectedPointingQueryDomain};

14/// The most rows of `POINTING` scanned per block while reading. Each block
15/// re-reads the storage manager's index (#949), so blocks are as large as a
16/// sixteenth of the request's budget allows, up to this.
17const SCAN_ROWS: usize = 1 << 16;
18
19/// The polynomial terms a `POINTING` row may have (`NUM_POLY` + 1): the
20/// bound the production imaging source plans with.
21pub(crate) const POLYNOMIAL_TERMS: usize = 4;
22
23/// The rows of `POINTING` scanned per block for a request of `budget`
24/// bytes: a sixteenth of it, up to [`SCAN_ROWS`].
25pub(crate) fn scan_rows(budget: usize) -> usize {
26    (budget / 16 / SCALAR_SCAN_BYTES_PER_ROW).clamp(1, SCAN_ROWS)
27}
28
29impl Msv2Observation {
30    /// The most bytes a pointing query of this observation holds while it
31    /// is read and once it is answered, whatever its antennas and times:
32    /// every `POINTING` row taken as a candidate at four polynomial terms
33    /// (the production source's bound), with one scan block of at most
34    /// every row, and the series it gives. A [`PointingRequest`] with this
35    /// budget is never refused for its size, and reads `POINTING` in blocks
36    /// as large as a query may scan. Zero without a `POINTING` table.
37    ///
38    /// # Errors
39    ///
40    /// When `POINTING` cannot be read.
41    pub fn pointing_bytes(&self) -> Result<usize, ObservationError> {
42        Ok(self.pointing_bound()?.0)
43    }
44
45    /// [`Self::pointing_bytes`], and the rows of the scan block it holds:
46    /// `POINTING`'s, up to [`SCAN_ROWS`].
47    fn pointing_bound(&self) -> Result<(usize, usize), ObservationError> {
48        let mut domain = SelectedPointingQueryDomain::builder();
49        for antenna in 0..self.catalogue.antennas.len() {
50            let id = i32::try_from(antenna).map_err(|_| ObservationError::InvalidStore {
51                reason: format!("antenna {antenna} beyond the MSv2 identifier range"),
52            })?;
53            domain.observe_row(id, id, 0.0).map_err(storage)?;
54        }
55        let Some(domain) = domain.finish().map_err(storage)? else {
56            return Ok((0, 1));
57        };
58        let (catalog, scan_bytes_per_row) = self
59            .ms
60            .selected_pointing_catalog_requirements(&domain, POLYNOMIAL_TERMS)
61            .map_err(storage)?;
62        // A scan block holds at most the table's rows.
63        let rows = SCAN_ROWS.min(catalog.source_rows_scanned()).max(1);
64        // The series holds a time, an interval, an origin and two
65        // coefficients a term for each retained row, at most the catalog.
66        let bytes = catalog
67            .construction_peak_bytes()
68            .saturating_add(rows.saturating_mul(scan_bytes_per_row))
69            .saturating_add(catalog.retained_bytes());
70        Ok((bytes, rows))
71    }
72
73    /// [`casa_observation::Observation::pointing`]: an MSv2's times are its
74    /// `TIME` values (its partitions' time origin is MJD 0), so the
75    /// request's times are `POINTING`'s.
76    pub(super) fn read_pointing(
77        &self,
78        request: &PointingRequest,
79    ) -> Result<PointingSeries, ObservationError> {
80        self.partition(request.partition)?;
81        let (start, end) = (*request.times.start(), *request.times.end());
82        if !(start.is_finite() && end.is_finite() && start <= end) {
83            return Err(ObservationError::InvalidRequest {
84                reason: format!("pointing times {start}..={end} s are not a finite span"),
85            });
86        }
87        let mut domain = SelectedPointingQueryDomain::builder();
88        for antenna in &request.antennas {
89            let id = i32::try_from(antenna.0)
90                .ok()
91                .filter(|_| (antenna.0 as usize) < self.catalogue.antennas.len())
92                .ok_or_else(|| ObservationError::InvalidRequest {
93                    reason: format!("no antenna {antenna:?}"),
94                })?;
95            for time in [start, end] {
96                domain.observe_row(id, id, time).map_err(storage)?;
97            }
98        }
99        let store_has_pointing = self
100            .ms
101            .subtable(crate::SubtableId::Pointing)
102            .is_some_and(|table| table.row_count() > 0);
103        let Some(domain) = domain.finish().map_err(storage)? else {
104            return Ok(PointingSeries {
105                store_has_pointing,
106                ..PointingSeries::empty(DirectionRef::J2000)
107            });
108        };
109        let column = match request.direction {
110            PointingDirection::Beam => PointingDirectionColumn::Direction,
111            PointingDirection::Target => PointingDirectionColumn::Target,
112        };
113        // A budget holding the whole bound holds its scan block, every row
114        // up to `SCAN_ROWS`; a smaller one, a sixteenth of itself.
115        let (bound, rows) = self.pointing_bound()?;
116        let scan = if request.budget_bytes >= bound {
117            rows
118        } else {
119            scan_rows(request.budget_bytes)
120        };
121        let plan = PointingReadPlan::new(scan, POLYNOMIAL_TERMS, request.budget_bytes.max(1))
122            .map_err(storage)?;
123        self.ms
124            .prepare_selected_pointing_catalog(column, &domain, plan)
125            .map_err(|error| match error {
126                MsError::OverBudget {
127                    required,
128                    available,
129                    ..
130                } => ObservationError::OverBudget {
131                    what: "an MSv2 pointing query",
132                    required,
133                    budget: available,
134                },
135                error => ObservationError::InvalidStore {
136                    reason: format!("POINTING: {error}"),
137                },
138            })?
139            .series()
140            .map(|series| PointingSeries {
141                store_has_pointing,
142                ..series
143            })
144    }
145}

208    /// One `POINTING` row as the table states it.
209    struct Row {
210        antenna: i32,
211        time_s: f64,
212        interval_s: f64,
213        origin_s: f64,
214        /// `DIRECTION`'s terms, `[2, NUM_POLY + 1]`.
215        terms: ndarray::Array2<f64>,
216    }
217
218    /// Every `POINTING` row of `path`, in table order.
219    fn table_rows(path: &std::path::Path) -> Vec<Row> {
220        let ms = MeasurementSet::open(path).expect("ms");
221        let pointing = ms.pointing().expect("POINTING");
222        (0..pointing.row_count())
223            .map(|row| {
224                let ArrayValue::Float64(terms) =
225                    pointing.array(row, "DIRECTION").expect("DIRECTION").clone()
226                else {
227                    panic!("DIRECTION is Float64");
228                };
229                Row {
230                    antenna: pointing.i32(row, "ANTENNA_ID").expect("ANTENNA_ID"),
231                    time_s: pointing.f64(row, "TIME").expect("TIME"),
232                    interval_s: pointing.f64(row, "INTERVAL").expect("INTERVAL"),
233                    origin_s: pointing.f64(row, "TIME_ORIGIN").expect("TIME_ORIGIN"),
234                    terms: terms.into_dimensionality().expect("[2, terms]"),
235                }
236            })
237            .collect()
238    }
239
240    /// `row`'s direction at `time_s`: its polynomial about `TIME_ORIGIN`.
241    fn direction_at(row: &Row, time_s: f64) -> [f64; 2] {
242        let dt = time_s - row.origin_s;
243        [0, 1].map(|axis| {
244            row.terms
245                .row(axis)
246                .iter()
247                .rev()
248                .fold(0.0, |sum, term| sum * dt + term)
249        })
250    }
251
252    /// `antenna`'s bracket at `time_s` read from `rows` one by one, as the
253    /// boundary defines it ([`casa_observation::PointingBracket`]): the row
254    /// whose interval holds the time, nearest it and then first in time
255    /// order; the first row of the latest time at or before it; and the
256    /// first row at or after it. Each is its time and its direction there.
257    fn table_bracket(rows: &[Row], antenna: i32, time_s: f64) -> [Option<(f64, [f64; 2])>; 3] {
258        let mut own: Vec<&Row> = rows.iter().filter(|row| row.antenna == antenna).collect();
259        own.sort_by(|left, right| left.time_s.total_cmp(&right.time_s));
260        let covering = own
261            .iter()
262            .filter(|row| {
263                row.interval_s > 0.0 && (row.time_s - time_s).abs() <= 0.5 * row.interval_s
264            })
265            .min_by(|left, right| {
266                (left.time_s - time_s)
267                    .abs()
268                    .total_cmp(&(right.time_s - time_s).abs())
269            });
270        let latest = own
271            .iter()
272            .rfind(|row| row.time_s <= time_s)
273            .map(|row| row.time_s);
274        let before = latest.and_then(|latest| own.iter().find(|row| row.time_s == latest));
275        let after = own.iter().find(|row| row.time_s >= time_s);
276        [covering, before, after].map(|row| row.map(|row| (row.time_s, direction_at(row, time_s))))
277    }

279    /// The boundary's MSv2 pointing brackets each (antenna, time) as the
280    /// table states it, read row by row: the same covering, preceding and
281    /// following rows, evaluated at the time to the same directions. The
282    /// rows here are in time order for each antenna.
283    #[test]
284    fn brackets_are_the_tables_rows() {
285        let root = tempfile::tempdir().expect("root");
286        let path = fixture(root.path());
287        let observation = open(&path, &OpenOptions::default()).expect("open");
288        let grid = observation.grid(PartitionId(0), "base").expect("grid");
289        let mut queries: Vec<f64> = grid
290            .times_s
291            .iter()
292            .flat_map(|time| [time - 7.0, time - 5.0, *time, time + 3.0, time + 5.0])
293            .collect();
294        queries.extend([
295            grid.times_s[0] - 30.0,
296            grid.times_s[grid.times_s.len() - 1] + 30.0,
297        ]);
298        queries.sort_by(f64::total_cmp);
299        let antennas: Vec<AntennaId> = (0..3).map(AntennaId).collect();
300        let series = observation
301            .pointing(&PointingRequest {
302                partition: PartitionId(0),
303                direction: PointingDirection::Beam,
304                antennas: antennas.clone(),
305                times: queries[0]..=queries[queries.len() - 1],
306                budget_bytes: 1 << 20,
307            })
308            .expect("pointing");
309        assert_eq!(series.terms, 2);
310        let rows = table_rows(&path);
311        let mut covered = [0; 3];
312        for antenna in antennas {
313            for time in &queries {
314                let boundary = series.bracket(antenna, *time).expect("boundary");
315                let boundary = [boundary.covering, boundary.before, boundary.after]
316                    .map(|candidate| candidate.map(|c| (c.time_s, c.direction_rad)));
317                let table = table_bracket(&rows, antenna.0 as i32, *time);
318                for (got, want) in boundary.iter().zip(&table) {
319                    match (got, want) {
320                        (Some((got_time, got)), Some((want_time, want))) => {
321                            assert_eq!(got_time, want_time, "{antenna:?} at {time}");
322                            for (got, want) in got.iter().zip(want) {
323                                assert!(
324                                    (got - want).abs() <= 1e-15,
325                                    "{antenna:?} at {time}: {got} against {want}"
326                                );
327                            }
328                        }
329                        (None, None) => {}
330                        _ => panic!("{antenna:?} at {time}: {boundary:?} against {table:?}"),
331                    }
332                }
333                covered[antenna.0 as usize] += usize::from(table[0].is_some());
334            }
335        }
336        assert!(
337            covered[0] > 0 && covered[2] > 0,
338            "rows cover queries: {covered:?}"
339        );
340        assert_eq!(covered[1], 0, "rows of no interval cover nothing");
341    }
