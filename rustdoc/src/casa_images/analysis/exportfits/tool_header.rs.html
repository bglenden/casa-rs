31/// Build a FITS header for an attached image without reading its pixels.
32///
33/// The shape is supplied separately so that virtual and native attachments
34/// use the same header path.
35///
36/// # Errors
37/// Returns an error for invalid WCS, metadata or an unreadable clock.
38pub fn image_fits_header(
39    image: &dyn ImageInterface<f32>,
40    shape: &[usize],
41    name: &str,
42    history: &[ImageHistoryEntry],
43) -> Result<ImageFitsHeader, ImageError> {
44    image_fits_header_with_beams(image, shape, name, history, None)
45}

47/// Build an attached image's FITS header using its saved restoring beam view.
48///
49/// An open tool can retain beam metadata independently of shared native pixels.
50pub fn image_fits_header_with_beams(
51    image: &dyn ImageInterface<f32>,
52    shape: &[usize],
53    name: &str,
54    history: &[ImageHistoryEntry],
55    beam_info: Option<&RecordValue>,
56) -> Result<ImageFitsHeader, ImageError> {
57    let coords = image.coordinates();
58    let plan = ExportAxisPlan::build(coords, shape, false, false, false, false, false)?;
59    let standard_wcs = || {
60        let base = try_to_fits_header(coords, shape)?;
61        let mut wcs = build_export_keywords(coords, &base, &plan, true, true, None)?
62            .into_iter()
63            .collect();
64        normalize_wcs(coords, &mut wcs)?;
65        Ok::<_, ImageError>(wcs)
66    };
67    let (mut wcs, projection) = match standard_wcs() {
68        Ok(wcs) => (wcs, Some(projection_parameters(coords, Some))),
69        Err(_) => (linear_fallback(coords, shape.len())?, None),
70    };
71    let mut header = ImageFitsHeader(Vec::new());
72    header.add("SIMPLE", FitsValue::Logical(true), Some("Standard FITS"));
73    header.add(
74        "BITPIX",
75        FitsValue::Integer(-32),
76        Some("Floating point (32 bit)"),
77    );
78    header.integer("NAXIS", shape.len() as i64);
79    for (index, length) in shape.iter().enumerate() {
80        header.integer(format!("NAXIS{}", index + 1), *length as i64);
81    }
82    header.add("EXTEND", FitsValue::Logical(true), None);
83    header.add(
84        "BSCALE",
85        FitsValue::Float(1.0),
86        Some("PHYSICAL = PIXEL*BSCALE + BZERO"),
87    );
88    header.add("BZERO", FitsValue::Float(0.0), None);
89    let mut info = image.image_info()?;
90    if let Some(record) = beam_info {
91        info.beam_set = ImageInfo::from_record(record)?.beam_set;
92    }
93    header.image_metadata(image.units(), &info, beam_info)?;
94    let standard = projection.is_some();
95    header.axes(shape.len(), &mut wcs, projection);
96    for (name, comment) in [
97        ("RESTFRQ", "Rest Frequency (Hz)"),
98        ("SPECSYS", "Spectral reference frame"),
99        ("ALTRVAL", "Alternate frequency reference value"),
100        ("ALTRPIX", "Alternate frequency reference pixel"),
101        ("VELREF", "1 LSR, 2 HEL, 3 OBS, +256 Radio"),
102    ] {
103        if let Some(value) = wcs.remove(name) {
104            header.add(name, value, Some(comment));
105            if name == "VELREF" {
106                header.text(
107                    "COMMENT",
108                    "casacore non-standard usage: 4 LSD, 5 GEO, 6 SOU, 7 GAL",
109                );
110            }
111        }
112    }
113    if standard {
114        if image
115            .coordinates()
116            .obs_info()
117            .date
118            .as_ref()
119            .is_none_or(|epoch| epoch.value().as_mjd() == 0.0)
120        {
121            wcs.remove("DATE-OBS");
122            wcs.insert("TIMESYS".into(), FitsValue::String("UTC".into()));
123        }
124        header.observation(image, &mut wcs)?;
125    } else {
126        header.text("TIMESYS", "UTC");
127    }
128    header.extend_misc(image.misc_info())?;
129    header.add(
130        "DATE",
131        FitsValue::String(format_time(
132            ImageHistoryEntry::current_time()? / 86_400.0,
133            FITS,
134            12,
135        )),
136        Some("Date FITS file was written"),
137    );
138    header.text("ORIGIN", "casacore-3.8.0");
139    if info.beam_set.is_multi() {
140        header.add(
141            "CASAMBM",
142            FitsValue::Logical(true),
143            Some("CASA multiple BEAMS table present"),
144        );
145    }
146    if !history.is_empty() {
147        visit_history_cards(history, false, |text| {
148            header.text("HISTORY", text);
149            Ok(())
150        })?;
151    }
152    header.text("END", "");
153    let start = name.len().saturating_sub(68);
154    let suffix = name.get(start..).unwrap_or_else(|| {
155        let boundary = name.char_indices().find(|(i, _)| *i >= start);
156        &name[boundary.map_or(name.len(), |(i, _)| i)..]
157    });
158    header.text("IMAGENME", suffix);
159    Ok(header)
160}

162pub(super) fn image_export_cards<T: ImagePixel>(
163    image: &dyn ImageInterface<T>,
164    info: &ImageInfo,
165    wcs: &[(String, FitsValue)],
166) -> Result<Vec<String>, ImageError> {
167    let mut header = ImageFitsHeader(Vec::new());
168    header.image_metadata(image.units(), info, None)?;
169    header.0.retain(|card| match card.keyword.name.as_str() {
170        "BMAJ" | "BMIN" | "BPA" => false,
171        "BUNIT" => image.units().is_empty(),
172        "OBJECT" => info.object_name.is_empty(),
173        _ => true,
174    });
175    header.observation(image, &mut BTreeMap::new())?;
176    let mut misc = ImageFitsHeader(Vec::new());
177    misc.extend_misc(image.misc_info())?;
178    let keep_misc = |name: &str| {
179        !matches!(
180            name,
181            "SIMPLE"
182                | "BITPIX"
183                | "NAXIS"
184                | "EXTEND"
185                | "EXTNAME"
186                | "END"
187                | "BSCALE"
188                | "BZERO"
189                | "BLANK"
190                | "DATAMIN"
191                | "DATAMAX"
192                | "BMAJ"
193                | "BMIN"
194                | "BPA"
195                | "CASAMBM"
196                | "BTYPE"
197                | "BUNIT"
198                | "OBJECT"
199                | "DATE"
200                | "ORIGIN"
201        ) && !(name.starts_with("NAXIS") && name[5..].bytes().all(|byte| byte.is_ascii_digit()))
202            && !wcs.iter().any(|(key, _)| key == name)
203    };
204    header.0.extend(
205        misc.0
206            .into_iter()
207            .filter(|card| keep_misc(&card.keyword.name)),
208    );
209    header.text(
210        "DATE",
211        format_time(ImageHistoryEntry::current_time()? / 86_400.0, FITS, 12),
212    );
213    header.text("ORIGIN", "casa-rs");
214    Ok(header
215        .0
216        .iter()
217        .map(|card| render_card(&card.keyword))
218        .collect())
219}

221impl ImageFitsHeader {
222    fn add(&mut self, name: impl Into<String>, value: FitsValue, comment: Option<&str>) {
223        let name = name.into();
224        let group = group_name(&name);
225        self.0.push(HeaderCard {
226            keyword: FitsKeyword {
227                name,
228                value,
229                comment: comment.map(str::to_owned),
230            },
231            group,
232        });
233    }
234
235    fn text(&mut self, name: impl Into<String>, text: impl Into<String>) {
236        self.add(name, FitsValue::String(text.into()), None);
237    }
238
239    fn integer(&mut self, name: impl Into<String>, number: i64) {
240        self.add(name, FitsValue::Integer(number), None);
241    }
242
243    fn image_metadata(
244        &mut self,
245        units: &str,
246        info: &ImageInfo,
247        beam_info: Option<&RecordValue>,
248    ) -> Result<(), ImageError> {
249        if let Some(beam) = info.beam_set.single_beam().filter(|beam| !beam.is_null()) {
250            self.add("BMAJ", FitsValue::Float(beam.major_in("deg")?), None);
251            self.add("BMIN", FitsValue::Float(beam.minor_in("deg")?), None);
252            let angle = match beam_info.and_then(ImageInfo::first_beam_record) {
253                Some(stored) => {
254                    stored_position_angle_degrees(&StoredGaussianBeam::from_record(stored)?)?
255                }
256                _ => wrapped_beam_position_angle(beam.position_angle_in("deg")?),
257            };
258            self.add("BPA", FitsValue::Float(angle), None);
259        }
260        let image_type = if info.image_type == ImageType::Undefined {
261            ImageType::Intensity
262        } else {
263            info.image_type
264        };
265        self.text("BTYPE", image_type.to_string());
266        self.text("OBJECT", &info.object_name);
267        self.add(
268            "BUNIT",
269            FitsValue::String(units.to_owned()),
270            Some("Brightness (pixel) unit"),
271        );
272        Ok(())
273    }
274
275    fn axes(
276        &mut self,
277        rank: usize,
278        wcs: &mut BTreeMap<String, FitsValue>,
279        projection: Option<Vec<(String, FitsValue)>>,
280    ) {
281        for keyword in ["EQUINOX", "RADESYS", "LONPOLE", "LATPOLE"] {
282            if let Some(value) = wcs.remove(keyword) {
283                self.add(keyword, value, None);
284            }
285        }
286        if projection.is_some() {
287            for col in 1..=rank {
288                for row in 1..=rank {
289                    let name = format!("PC{row}_{col}");
290                    let value = wcs
291                        .remove(&name)
292                        .unwrap_or(FitsValue::Float(f64::from(row == col)));
293                    self.add(name, value, None);
294                }
295            }
296        }
297        let projection = projection.unwrap_or_default();
298        let interleave_projection = projection.len() == rank;
299        let mut projection = projection.into_iter();
300        for axis in 1..=rank {
301            for key in ["CTYPE", "CRVAL", "CDELT", "CRPIX", "CUNIT"] {
302                let name = format!("{key}{axis}");
303                if let Some(value) = wcs.remove(&name) {
304                    self.add(name, value, None);
305                } else if key == "CUNIT" {
306                    self.text(name, "");
307                }
308            }
309            if interleave_projection && let Some((name, value)) = projection.next() {
310                self.add(name, value, None);
311            }
312        }
313        for (name, value) in projection {
314            self.add(name, value, None);
315        }
316    }
317
318    fn observation<T: ImagePixel>(
319        &mut self,
320        image: &dyn ImageInterface<T>,
321        wcs: &mut BTreeMap<String, FitsValue>,
322    ) -> Result<(), ImageError> {
323        for keyword in ["TELESCOP", "OBSERVER", "DATE-OBS", "TIMESYS"] {
324            if let Some(value) = wcs.remove(keyword) {
325                self.add(keyword, value, None);
326            }
327        }
328        let obs = image.coordinates().obs_info();
329        if let Some(position) = &obs.telescope_position {
330            let xyz = position
331                .convert_to(PositionRef::ITRF)
332                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?
333                .values();
334            for (key, value) in ["OBSGEO-X", "OBSGEO-Y", "OBSGEO-Z"].into_iter().zip(xyz) {
335                self.add(key, FitsValue::Float(value), None);
336            }
337        }
338        if !obs.pointing_center_initial {
339            for (key, angle) in ["OBSRA", "OBSDEC"].into_iter().zip(obs.pointing_center_rad) {
340                self.add(key, FitsValue::Float(angle.to_degrees()), None);
341            }
342        }
343        Ok(())
344    }
345
346    fn extend_misc(&mut self, info: RecordValue) -> Result<(), ImageError> {
347        let mut fields = info.fields().iter().collect::<Vec<_>>();
348        fields.sort_by_key(|field| field.name.to_ascii_uppercase());
349        let mut cards_by_field = Vec::new();
350        for field in fields {
351            let name = field.name.to_ascii_uppercase();
352            if name.is_empty() || !name.is_ascii() {
353                continue;
354            }
355            let values = misc_values(&name, &field.value)?;
356            if values.is_empty() {
357                continue;
358            }
359            let shape = match &field.value {
360                Value::Array(array) => array.shape().to_vec(),
361                _ => Vec::new(),
362            };
363            cards_by_field.push((name, shape, values));
364        }
365        // casacore interleaves arrays of the same rank and element count,
366        // naming matrix cards using the last array's row count in the run.
367        for run in cards_by_field.chunk_by(|(_, first, values), (_, next, other)| {
368            first.len() == next.len() && values.len() == other.len()
369        }) {
370            let shape = &run.last().expect("nonempty run").1;
371            let mut fields = Vec::with_capacity(run.len());
372            for (name, _, values) in run {
373                let names = misc_keyword_names(name, shape, values.len())?;
374                fields.push(names.into_iter().zip(values));
375            }
376            for _ in 0..run[0].2.len() {
377                for cards in &mut fields {
378                    let (mut key, value) = cards.next().expect("one card per value");
379                    if key == "COMMENT"
380                        && matches!(value, FitsValue::String(text) if text.is_empty())
381                    {
382                        key.clear();
383                    }
384                    self.0.push(HeaderCard {
385                        keyword: FitsKeyword {
386                            name: key,
387                            value: value.clone(),
388                            comment: None,
389                        },
390                        group: None,
391                    });
392                }
393            }
394        }
395        Ok(())
396    }

398    /// The FITS header as consecutive 80-byte cards, with no block padding.
399    pub fn card_string(&self) -> String {
400        self.0
401            .iter()
402            .map(|card| render_card(&card.keyword))
403            .collect()
404    }

406    /// The FITS header as a CASA record with the given entry names omitted.
407    pub fn record(&self, exclude: &[String]) -> RecordValue {
408        let mut grouped = BTreeMap::<String, Vec<FitsValue>>::new();
409        for card in &self.0 {
410            let keyword = &card.keyword;
411            let name = card.group.unwrap_or(&keyword.name);
412            let name = if name.is_empty() { "        " } else { name };
413            grouped
414                .entry(name.to_owned())
415                .or_default()
416                .push(keyword.value.clone());
417        }
418        let mut record = RecordValue::default();
419        for (name, values) in grouped {
420            if !exclude.contains(&name) {
421                let axis_vector = matches!(
422                    name.as_str(),
423                    "NAXIS" | "CTYPE" | "CRVAL" | "CRPIX" | "CDELT"
424                );
425                let value = if !axis_vector && values.len() == 1 {
426                    scalar(&name, &values[0]).into()
427                } else {
428                    Value::Array(array(&name, &values))
429                };
430                record.upsert(name, value);
431            }
432        }
433        record
434    }
435}

437fn linear_fallback(
438    coords: &CoordinateSystem,
439    rank: usize,
440) -> Result<BTreeMap<String, FitsValue>, ImageError> {
441    if coords.n_world_axes() != rank
442        || (0..rank).any(|axis| coords.pixel_axis_to_world_axis(axis) != Some(axis))
443    {
444        return Err(header_creation_error());
445    }
446    let mut wcs = BTreeMap::new();
447    for index in 0..coords.n_coordinates() {
448        let coordinate = coords.coordinate(index);
449        let (reference, increment, units) = match coordinate {
450            CoordinateModel::Direction(direction) => {
451                let (reference, increment) = direction.reference_and_increment_in_world_units();
452                (reference, increment, direction.visible_world_units())
453            }
454            _ => (
455                coordinate.reference_value(),
456                coordinate.increment(),
457                coordinate.axis_units(),
458            ),
459        };
460        let names = coordinate.axis_names();
461        let pixels = coordinate.reference_pixel();
462        for (local, axis) in coords
463            .coordinate_pixel_axes(index)
464            .expect("existing coordinate")
465            .into_iter()
466            .enumerate()
467        {
468            let Some(axis) = axis else {
469                continue;
470            };
471            let axis = axis + 1;
472            let mut name = names[local].to_ascii_uppercase();
473            let mut unit = units[local].clone();
474            truncate_bytes(&mut name, 8);
475            truncate_bytes(&mut unit, 8);
476            for (key, value) in [
477                ("CTYPE", FitsValue::String(name)),
478                ("CRVAL", FitsValue::Float(reference[local])),
479                ("CDELT", FitsValue::Float(increment[local])),
480                ("CRPIX", FitsValue::Float(pixels[local] + 1.0)),
481                ("CUNIT", FitsValue::String(unit)),
482            ] {
483                wcs.insert(format!("{key}{axis}"), value);
484            }
485        }
486    }
487    Ok(wcs)
488}

490fn normalize_wcs(
491    coords: &CoordinateSystem,
492    wcs: &mut BTreeMap<String, FitsValue>,
493) -> Result<(), ImageError> {
494    if !wcs.contains_key("RESTFRQ") {
495        wcs.remove("SPECSYS");
496    }
497    for index in 0..coords.n_coordinates() {
498        let coordinate = coords.coordinate(index);
499        let axes = coords
500            .coordinate_world_axes(index)
501            .expect("existing coordinate");
502        match coordinate {
503            CoordinateModel::Direction(direction) => {
504                let frame = direction.direction_ref();
505                let (longitude, latitude) = match frame {
506                    DirectionRef::GALACTIC => ("GLON", "GLAT"),
507                    DirectionRef::SUPERGAL => ("SLON", "SLAT"),
508                    DirectionRef::ECLIPTIC | DirectionRef::MECLIPTIC | DirectionRef::TECLIPTIC => {
509                        ("ELON", "ELAT")
510                    }
511                    DirectionRef::J2000
512                    | DirectionRef::JMEAN
513                    | DirectionRef::APP
514                    | DirectionRef::B1950
515                    | DirectionRef::B1950Vla
516                    | DirectionRef::BMEAN
517                    | DirectionRef::BTRUE
518                    | DirectionRef::ICRS
519                    | DirectionRef::HADEC => ("RA--", "DEC-"),
520                    _ => ("??LN", "??LT"),
521                };
522                let projection = direction.projection().name();
523                for (axis, name) in axes.iter().zip([longitude, latitude]) {
524                    let Some(axis) = axis else {
525                        continue;
526                    };
527                    wcs.insert(
528                        format!("CTYPE{}", axis + 1),
529                        FitsValue::String(format!("{name}-{projection}")),
530                    );
531                }
532                wcs.remove("EQUINOX");
533                wcs.remove("RADESYS");
534                let reference = match frame {
535                    DirectionRef::J2000 => Some(("FK5", Some(2000.0))),
536                    DirectionRef::B1950 => Some(("FK4", Some(1950.0))),
537                    DirectionRef::B1950Vla => Some(("FK4", Some(1979.9))),
538                    DirectionRef::ICRS => Some(("ICRS", None)),
539                    _ => None,
540                };
541                if let Some((system, equinox)) = reference {
542                    wcs.insert("RADESYS".into(), FitsValue::String(system.into()));
543                    if let Some(equinox) = equinox {
544                        wcs.insert("EQUINOX".into(), FitsValue::Float(equinox));
545                    }
546                }
547            }
548            CoordinateModel::Stokes(stokes) => {
549                if let [Some(axis)] = axes.as_slice()
550                    && let [Some(pixel)] = coords
551                        .coordinate_pixel_axes(index)
552                        .expect("existing coordinate")
553                        .as_slice()
554                {
555                    wcs.extend(stokes_axis_keywords(stokes, axis + 1, pixel + 1)?);
556                }
557            }
558            _ => {}
559        }
560    }
561    for (key, value) in wcs.iter_mut() {
562        if matches!(split_axis_keyword(key), Some(("CTYPE" | "CUNIT", _)))
563            && let FitsValue::String(text) = value
564        {
565            truncate_bytes(text, 8);
566        }
567    }
568    Ok(())
569}

571pub(super) fn wrapped_beam_position_angle(angle: f64) -> f64 {
572    let angle = angle % 180.0;
573    if angle > 90.0 {
574        angle - 180.0
575    } else if angle <= -90.0 {
576        angle + 180.0
577    } else {
578        angle
579    }
580}

582pub(super) fn stokes_axis_keywords(
583    stokes: &StokesCoordinate,
584    axis: usize,
585    pixel: usize,
586) -> Result<[(String, FitsValue); 3], ImageError> {
587    let (reference, increment) =
588        fits_stokes_axis(stokes.stokes()).ok_or_else(header_creation_error)?;
589    Ok([
590        (format!("CRVAL{axis}"), FitsValue::Float(reference)),
591        (format!("CDELT{axis}"), FitsValue::Float(increment)),
592        (format!("CRPIX{pixel}"), FitsValue::Float(1.0)),
593    ])
594}

596pub(super) fn projection_parameters(
597    coords: &CoordinateSystem,
598    output_axis: impl Fn(usize) -> Option<usize>,
599) -> Vec<(String, FitsValue)> {
600    let mut result = Vec::new();
601    for index in 0..coords.n_coordinates() {
602        let axis = coords.coordinate(index);
603        if let CoordinateModel::Direction(direction) = axis
604            && let [_, Some(latitude)] = coords
605                .coordinate_world_axes(index)
606                .expect("existing coordinate")
607                .as_slice()
608            && let Some(latitude_axis) = output_axis(latitude + 1)
609        {
610            let projection = direction.projection();
611            let params = projection.parameters();
612            let start = projection.projection_type().fits_parameter_start();
613            let count = match projection.projection_type() {
614                ProjectionType::SIN => 2,
615                ProjectionType::ZPN => params
616                    .len()
617                    .min(projection.projection_type().fits_parameter_defaults().len()),
618                _ => params.len(),
619            };
620            for index in 0..count {
621                result.push((
622                    format!("PV{}_{}", latitude_axis, index + start),
623                    FitsValue::Float(params.get(index).copied().unwrap_or(0.0)),
624                ));
625            }
626        }
627    }
628    result
629}

635fn misc_keyword_names(
636    name: &str,
637    shape: &[usize],
638    count: usize,
639) -> Result<Vec<String>, ImageError> {
640    match shape {
641        [] => Ok(vec![if name.starts_with("COMMENT") {
642            "COMMENT".into()
643        } else if name.starts_with("HISTORY") {
644            "HISTORY".into()
645        } else {
646            name[..name.len().min(8)].to_owned()
647        }]),
648        [length] if *length <= 999 && name.len() + length.to_string().len() <= 8 => {
649            Ok((1..=count).map(|index| format!("{name}{index}")).collect())
650        }
651        [rows, cols] if name.len() <= 2 && *rows <= 99 && *cols <= 99 => {
652            let width = if *rows > 9 { 2 } else { 1 };
653            Ok((0..count)
654                .map(|index| {
655                    let row = index % rows + 1;
656                    let col = index / rows + 1;
657                    format!("{name}{row:0width$}_{col:0width$}")
658                })
659                .collect())
660        }
661        _ => Err(header_creation_error()),
662    }
663}

665fn misc_values(name: &str, value: &Value) -> Result<Vec<FitsValue>, ImageError> {
666    let card_integer = |value| FitsValue::Integer(i64::from(value as i32));
667    macro_rules! array {
668        ($values:expr, $variant:ident, $map:expr) => {
669            $values
670                .t()
671                .iter()
672                .map(|value| FitsValue::$variant($map(value)))
673                .collect()
674        };
675    }
676    let values = match value {
677        Value::Scalar(value) => {
678            let scalar = match value {
679                ScalarValue::Bool(value) => FitsValue::Logical(*value),
680                ScalarValue::UInt8(value) => FitsValue::Integer(i64::from(*value)),
681                ScalarValue::UInt16(value) => FitsValue::Integer(i64::from(*value)),
682                ScalarValue::UInt32(value) => card_integer(i64::from(*value)),
683                ScalarValue::Int16(value) => FitsValue::Integer(i64::from(*value)),
684                ScalarValue::Int32(value) => FitsValue::Integer(i64::from(*value)),
685                ScalarValue::Int64(value) => card_integer(*value),
686                ScalarValue::Float32(value) => FitsValue::Float(f64::from(*value)),
687                ScalarValue::Float64(value) => FitsValue::Float(*value),
688                ScalarValue::String(value) => {
689                    let mut text = value.clone();
690                    if name.starts_with("COMMENT") || name.starts_with("HISTORY") {
691                        truncate_bytes(&mut text, 72);
692                        return Ok(vec![FitsValue::String(text)]);
693                    }
694                    FitsValue::String(text)
695                }
696                ScalarValue::Complex32(_) | ScalarValue::Complex64(_) => {
697                    return Err(header_creation_error());
698                }
699            };
700            vec![scalar]
701        }
702        Value::Array(ArrayValue::Complex32(_) | ArrayValue::Complex64(_)) => Vec::new(),
703        Value::Array(value) if value.shape().len() > 2 => return Err(header_creation_error()),
704        Value::Array(value) => match value {
705            ArrayValue::Bool(values) => array!(values, Logical, |value: &bool| *value),
706            ArrayValue::UInt8(values) => array!(values, Integer, |value: &u8| i64::from(*value)),
707            ArrayValue::UInt16(values) => array!(values, Integer, |value: &u16| i64::from(*value)),
708            ArrayValue::UInt32(values) => {
709                array!(values, Integer, |value: &u32| i64::from(*value as i32))
710            }
711            ArrayValue::Int16(values) => array!(values, Integer, |value: &i16| i64::from(*value)),
712            ArrayValue::Int32(values) => array!(values, Integer, |value: &i32| i64::from(*value)),
713            ArrayValue::Int64(values) => {
714                array!(values, Integer, |value: &i64| i64::from(*value as i32))
715            }
716            ArrayValue::Float32(values) => array!(values, Float, |value: &f32| f64::from(*value)),
717            ArrayValue::Float64(values) => array!(values, Float, |value: &f64| *value),
718            ArrayValue::String(values) => array!(values, String, Clone::clone),
719            _ => Vec::new(),
720        },
721        _ => Vec::new(),
722    };
723    if values
724        .iter()
725        .any(|value| matches!(value, FitsValue::String(text) if text.len() > 68))
726    {
727        return Err(header_creation_error());
728    }
729    Ok(values)
730}

732fn group_name(name: &str) -> Option<&'static str> {
733    if name == "NAXIS" {
734        return Some("NAXIS");
735    }
736    if let Some((base, _)) = split_axis_keyword(name) {
737        return (base != "CUNIT").then_some(base);
738    }
739    name.strip_prefix("NAXIS")
740        .filter(|axis| !axis.is_empty() && axis.bytes().all(|ch| ch.is_ascii_digit()))
741        .map(|_| "NAXIS")
742}

744fn padded_text(name: &str, value: &str) -> String {
745    if matches!(name, "        " | "END" | "HISTORY" | "COMMENT") {
746        value.to_owned()
747    } else {
748        let mut text = value.to_owned();
749        text.extend(std::iter::repeat_n(' ', 8_usize.saturating_sub(text.len())));
750        text
751    }
752}
753
754fn scalar(name: &str, value: &FitsValue) -> ScalarValue {
755    match value {
756        FitsValue::String(value) => ScalarValue::String(padded_text(name, value)),
757        FitsValue::Integer(value) => ScalarValue::Int64(*value),
758        FitsValue::Float(value) => ScalarValue::Float64(*value),
759        FitsValue::Logical(value) => ScalarValue::Bool(*value),
760    }
761}

763fn array(name: &str, values: &[FitsValue]) -> ArrayValue {
764    match values.first().expect("at least one FITS card in a group") {
765        FitsValue::String(_) => ArrayValue::from_string_vec(
766            values
767                .iter()
768                .filter_map(|value| match value {
769                    FitsValue::String(text) => Some(padded_text(name, text)),
770                    _ => None,
771                })
772                .collect(),
773        ),
774        FitsValue::Integer(_) => ArrayValue::Int64(
775            ndarray::Array1::from_iter(values.iter().filter_map(|value| match value {
776                FitsValue::Integer(integer) => Some(*integer),
777                _ => None,
778            }))
779            .into_dyn(),
780        ),
781        FitsValue::Float(_) => ArrayValue::from_f64_vec(
782            values
783                .iter()
784                .filter_map(|value| match value {
785                    FitsValue::Float(float) => Some(*float),
786                    _ => None,
787                })
788                .collect(),
789        ),
790        FitsValue::Logical(_) => ArrayValue::Bool(
791            ndarray::Array1::from_iter(values.iter().filter_map(|value| match value {
792                FitsValue::Logical(flag) => Some(*flag),
793                _ => None,
794            }))
795            .into_dyn(),
796        ),
797    }
798}

800fn render_card(keyword: &FitsKeyword) -> String {
801    let name = &keyword.name;
802    let mut card = match &keyword.value {
803        FitsValue::String(text) if matches!(name.as_str(), "" | "END" | "HISTORY" | "COMMENT") => {
804            format!("{name:<8}{text}")
805        }
806        value => {
807            let field = match value {
808                FitsValue::String(text) => {
809                    format!("'{}'", padded_text(name, text).replace('\'', "''"))
810                }
811                FitsValue::Integer(integer) => format!("{integer:>20}"),
812                FitsValue::Logical(flag) => format!("{:>20}", if *flag { 'T' } else { 'F' }),
813                FitsValue::Float(number) => {
814                    let token = format!("{number:.12E}");
815                    let field = token.split_once('E').map_or_else(
816                        || token.clone(),
817                        |(mantissa, exponent)| {
818                            format!(
819                                "{mantissa}E{:+03}",
820                                exponent.parse::<i32>().expect("decimal exponent")
821                            )
822                        },
823                    );
824                    format!("{field:>20}")
825                }
826            };
827            format!("{name:<8}= {field}")
828        }
829    };
830    if let Some(comment) = &keyword.comment {
831        card.extend(std::iter::repeat_n(
832            ' ',
833            30_usize.saturating_sub(card.len()),
834        ));
835        card.push_str(" /");
836        card.push_str(comment);
837    }
838    truncate_bytes(&mut card, 80);
839    card.extend(std::iter::repeat_n(' ', 80 - card.len()));
840    card
841}
842
843fn truncate_bytes(text: &mut String, limit: usize) {
844    if text.len() > limit {
845        let boundary = (0..=limit)
846            .rev()
847            .find(|index| text.is_char_boundary(*index))
848            .expect("zero is a boundary");
849        text.truncate(boundary);
850    }
851}
