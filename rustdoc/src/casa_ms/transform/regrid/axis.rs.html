13pub(super) fn output_axis(
14    options: &MsRegridOptions,
15    centres: &[f64],
16    widths: &[f64],
17    maximum_channels: usize,
18) -> Result<CasaChannelFrequencies, MsTransformError> {
19    let error = MsTransformError::Spectrum;
20    let mode = options.mode.as_str();
21    let length = centres.len();
22    if length == 0 || length != widths.len() {
23        return Err(error(
24            "Regridding needs matching non-empty frequency and width arrays.".into(),
25        ));
26    }
27    let count = usize::try_from(options.nchan)
28        .ok()
29        .filter(|&n| n > 0)
30        .map(|channels| {
31            channels
32                .checked_mul(options.nspw.max(1))
33                .filter(|&count| count <= i32::MAX as usize)
34                .ok_or_else(|| {
35                    error("Regridding channel count exceeds the MeasurementSet limit.".into())
36                })
37        })
38        .transpose()?;
39    let rest_frequency_hz = if options.restfreq.trim().is_empty() {
40        if mode == "velocity" {
41            return Err(MsTransformError::RegridParameter {
42                parameter: "restfreq",
43                reason: "Parameter restfreq must be set when mode='velocity'".into(),
44            });
45        }
46        0.0
47    } else {
48        parse_rest_frequency_hz(&options.restfreq).map_err(|source| error(source.to_string()))?
49    };
50    let mut request = RegridRequest {
51        quantity: RegridQuantity::Frequency,
52        start: None,
53        width: None,
54        nchan: count,
55        rest_frequency_hz,
56    };
57    match mode {
58        "channel" => {
59            let start = channel_arg(&options.start, 0);
60            let step = channel_arg(&options.width, 1);
61            let descending = centres.first() > centres.last();
62            let first = if descending { length - 1 } else { 0 };
63            let start = usize::try_from(start).ok();
64            let selected = start.filter(|&value| value < length).unwrap_or(first);
65            let mut width = widths[first].abs() * f64::from(step.unsigned_abs().max(1));
66            let mut start_is_end = (step < 0) ^ descending;
67            if start.is_some_and(|index| index >= length) {
68                width = width.abs();
69                start_is_end = false;
70            }
71            let edge = if start_is_end {
72                centres[selected] + widths[selected].abs() / 2.
73            } else {
74                centres[selected] - widths[selected].abs() / 2.
75            };
76            request.start = start.map(|_| {
77                if start_is_end {
78                    edge - width / 2.
79                } else {
80                    edge + width / 2.
81                }
82            });
83            request.width = Some(if start_is_end { -width } else { width });
84        }
85        "channel_b" => return channel_bounds(options, centres, widths, count),
86        "frequency" => {
87            request.start = quantity_arg("start", &options.start, "Hz", "0")?;
88            request.width = quantity_arg("width", &options.width, "Hz", "1")?;
89        }
90        "velocity" => {
91            request.quantity = match options.veltype.to_ascii_lowercase().as_str() {
92                "optical" => RegridQuantity::OpticalVelocity,
93                "radio" => RegridQuantity::RadioVelocity,
94                _ => return Err(super::invalid_choice("veltype", &options.veltype)),
95            };
96            request.start = quantity_arg("start", &options.start, "m/s", "0")?;
97            request.width = quantity_arg("width", &options.width, "m/s", "1")?;
98        }
99        _ => return Err(super::invalid_choice("mode", mode)),
100    }
101    casa_data_channel_frequencies(&request, centres, widths, maximum_channels)
102        .map_err(|source| error(source.to_string()))
103}

105fn channel_arg(value: &str, default: i32) -> i32 {
106    let value = value.trim();
107    if value.is_empty() || value == "[]" {
108        default
109    } else {
110        value.parse().unwrap_or_default()
111    }
112}
113
114fn quantity_arg(
115    parameter: &'static str,
116    value: &str,
117    unit: &str,
118    default: &str,
119) -> Result<Option<f64>, MsTransformError> {
120    let value = value.trim();
121    if value.is_empty() || value == "[]" || value == default {
122        return Ok(None);
123    }
124    let unit = Unit::new(unit).expect("regridding uses a known frequency or velocity unit");
125    let valid = Quantity::read(value)
126        .ok()
127        .and_then(|quantity| quantity.get_value_in(&unit).ok())
128        .filter(|number| number.is_finite());
129    valid.map(Some).ok_or_else(|| {
130        let dimension = if unit.name() == "Hz" {
131            "frequency"
132        } else {
133            "velocity"
134        };
135        MsTransformError::RegridParameter {
136            parameter,
137            reason: format!("Parameter {parameter} does not have valid {dimension} units"),
138        }
139    })
140}

142fn channel_bounds(
143    options: &MsRegridOptions,
144    frequencies: &[f64],
145    widths: &[f64],
146    count: Option<usize>,
147) -> Result<CasaChannelFrequencies, MsTransformError> {
148    let channel_count = frequencies.len() as i64;
149    let step = channel_arg(&options.width, 1);
150    let requested_step = i64::from(step.unsigned_abs().max(1));
151    let requested_start = channel_arg(&options.start, 0);
152    let descending = frequencies.first() > frequencies.last();
153    let phys = |index: i64| {
154        if descending {
155            (channel_count - 1 - index) as usize
156        } else {
157            index as usize
158        }
159    };
160    let starts_high = (step < 0) ^ descending;
161    let mut span = count.map_or(channel_count, |channels| channels as i64 * requested_step);
162    let from = i64::from(requested_start);
163    if from < 0 || from >= channel_count {
164        return Err(MsTransformError::Spectrum(format!(
165            "SPW start {requested_start} outside valid range 0–{}.",
166            channel_count - 1
167        )));
168    }
169    let centre = if starts_high {
170        phys(from) as i64 - span / 2
171    } else {
172        phys(from) as i64 + span / 2
173    };
174    if centre - span / 2 < 0 {
175        span = 2 * centre + 1;
176    }
177    if centre + span / 2 > channel_count {
178        span = 2 * (channel_count - centre);
179    }
180    if span <= 0 {
181        return Err(MsTransformError::Spectrum(
182            "Requested channel bounds leave an empty SPW.".into(),
183        ));
184    }
185    let step = requested_step.min(span);
186    let lower = centre - span / 2;
187    let upper = (lower + span).min(channel_count);
188    if lower < 0 || lower >= upper {
189        return Err(MsTransformError::Spectrum(
190            "Requested channel bounds leave an empty SPW.".into(),
191        ));
192    }
193    let central_bin = if (span / step) % 2 == 1 {
194        centre - step / 2
195    } else {
196        centre
197    };
198    let central_bin = central_bin.clamp(lower, upper);
199    let mut bins = Vec::new();
200    let mut right = central_bin;
201    while right > lower {
202        let left = lower.max(right - step);
203        bins.push((left, right - 1));
204        right = left;
205    }
206    bins.reverse();
207    for left in (central_bin..upper).step_by(step as usize) {
208        bins.push((left, upper.min(left + step) - 1));
209    }
210    if descending {
211        bins.reverse();
212    }
213    let (centres_hz, steps_hz) = bins
214        .into_iter()
215        .map(|(low, high)| {
216            let lo = frequencies[phys(low)] - widths[phys(low)].abs() / 2.;
217            let hi = frequencies[phys(high)] + widths[phys(high)].abs() / 2.;
218            ((hi + lo) / 2., hi - lo)
219        })
220        .unzip();
221    Ok(CasaChannelFrequencies {
222        centres_hz,
223        steps_hz,
224    })
225}
