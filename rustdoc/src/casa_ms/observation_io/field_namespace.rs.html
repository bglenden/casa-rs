19/// Field selection in an existing observation's format.
20pub enum FieldNamespace<'a> {
21    /// Resolve selectors in the source MSv2's FIELD numbering.
22    Msv2(Msv2FieldSelection<'a>),
23    /// Resolve selectors in a calibration table and bind them to the processing set.
24    Msv4(&'a Msv4FieldSelection),
25}
26
27/// A selector backed by the MSv2's FIELD subtable.
28pub struct Msv2FieldSelection<'a>(&'a MeasurementSet);
29
30impl Msv2FieldSelection<'_> {
31    /// Resolve a CASA selector to FIELD indices. Empty text selects every
32    /// field and returns an empty list.
33    ///
34    /// # Errors
35    /// When the selector or FIELD metadata is invalid.
36    pub fn select(&self, selector: &str) -> MsResult<Vec<i32>> {
37        self.0.resolve_field_selector(selector)
38    }
39}

41/// The MSv4 field identities needed to validate calibration table selectors.

48struct ObservationField {

56    code: Option<String>,

61/// Open the field namespace of either store for `query`. MSv2 resolves
62/// selectors over its FIELD subtable. MSv4 keeps its field identities and
63/// releases the rest of the admitted catalogue before invoking the caller.
64///
65/// # Errors
66/// When the store cannot be opened or the MSv4 catalogue cannot be admitted.
67pub fn with_field_namespace<T>(
68    source: &Path,
69    query: impl FnOnce(FieldNamespace<'_>) -> T,
70) -> Result<T, ObservationError> {
71    if super::is_msv4_store(source) {
72        #[cfg(all(feature = "msv4", unix))]
73        {
74            let fields =
75                Msv4FieldSelection::open(source, super::creation::default_memory_bytes()?)?;
76            return Ok(query(FieldNamespace::Msv4(&fields)));
77        }
78        #[cfg(not(all(feature = "msv4", unix)))]
79        return Err(ObservationError::Unsupported {
80            what: "an MSv4 field namespace without casa-ms's msv4 feature",
81        });
82    }
83    let ms = MeasurementSet::open(source).map_err(storage)?;
84    Ok(query(FieldNamespace::Msv2(Msv2FieldSelection(&ms))))
85}

87impl Msv4FieldSelection {
88    #[cfg(all(feature = "msv4", unix))]
89    fn open(source: &Path, budget: usize) -> Result<Self, ObservationError> {
90        let mut admit = |required| admit(budget, required, "MSv4 field catalogue opening");
91        let observation = super::Msv4Observation::open_granted(
92            source,
93            &super::OpenOptions::default(),
94            &mut admit,
95        )?;
96        let catalogue = &observation.catalogue().fields;
97        let kept = catalogue.len() * size_of::<ObservationField>()
98            + catalogue
99                .iter()
100                .map(|field| field.name.len())
101                .sum::<usize>();
102        admit(observation.held_bytes().saturating_add(kept))?;
103        let fields = catalogue
104            .iter()
105            .map(|field| ObservationField {
106                name: field.name.clone(),
107                phase_centre: field.phase_centre.clone(),
108                tracks_ephemeris: field.ephemeris.is_some(),
109            })
110            .collect();
111        Ok(Self {
112            fields,
113            kept,
114            budget,
115        })
116    }
117
118    /// Resolve a CASA selector in the calibration table's own FIELD
119    /// numbering, requiring a unique MSv4 identity for every selected row.
120    /// Empty text selects every calibration field and returns an empty list.
121    ///
122    /// # Errors
123    /// When the FIELD subtable or selector is invalid, the fields cannot be
124    /// matched uniquely, a selected identity moves with time, or the
125    /// calibration FIELD read cannot be admitted.
126    pub fn select(&self, directory: &Path, selector: &str) -> Result<Vec<i32>, ObservationError> {
127        if selector.trim().is_empty() {
128            return Ok(Vec::new());
129        }
130        let rows = self.read_calibration_fields(directory, selector)?;
131        let fields: Vec<_> = rows
132            .iter()
133            .map(|field| FieldSelectorRow {
134                name: &field.name,
135                code: field.code.as_deref(),
136                flagged: field.flagged,
137            })
138            .collect();
139        let selected = resolve_field_selector_rows(&fields, selector).map_err(storage)?;
140        for &id in &selected {
141            self.bind_field(&rows, id as usize)?;
142        }
143        Ok(selected)
144    }
145
146    fn read_calibration_fields(
147        &self,
148        directory: &Path,
149        selector: &str,
150    ) -> Result<Vec<CalibrationField>, ObservationError> {
151        let path = directory.join("FIELD");
152        let opening_bytes = self.kept.saturating_add(held::opening_bytes(&path));
153        let what = "MSv4 calibration FIELD binding";
154        admit(self.budget, opening_bytes, what)?;
155        let layout = Table::immediate_part_names(&path, &path)
156            .map_err(|error| storage(MsError::from(error)))?;
157        if !matches!(layout, TablePartNames::Plain(_)) {
158            return Err(ObservationError::Unsupported {
159                what: "binding a calibration FIELD reference or concatenation to MSv4",
160            });
161        }
162        let table = Table::open(TableOptions::new(path)).map_err(|e| storage(MsError::from(e)))?;
163        if table.row_count() > i32::MAX as usize {
164            return Err(ObservationError::InvalidStore {
165                reason: "calibration FIELD row numbers exceed the signed selector range".to_owned(),
166            });
167        }
168        let required = opening_bytes
169            // The opened table's row and column caches plus its projected NAME and CODE.
170            .saturating_add(held::loaded_bytes(&table).saturating_mul(2))
171            .saturating_add(binding_scratch_bytes(table.row_count(), selector));
172        admit(self.budget, required, what)?;
173        require_units(&table, "PHASE_DIR", "rad")?;
174        let field = MsField::new(&table);
175        (0..table.row_count())
176            .map(|row| {
177                Ok(CalibrationField {
178                    name: field.name(row).map_err(storage)?,
179                    code: optional_string(&table, row, "CODE").map_err(storage)?,
180                    flagged: optional_bool(&table, row, "FLAG_ROW")
181                        .map_err(storage)?
182                        .unwrap_or_default(),
183                    fixed_direction: fixed_direction(&table, &field, row)?,
184                })
185            })
186            .collect()
187    }
188
189    fn bind_field(&self, rows: &[CalibrationField], id: usize) -> Result<(), ObservationError> {
190        let field = &rows[id];
191        let Some(direction) = &field.fixed_direction else {
192            return Err(ObservationError::Unsupported {
193                what: "binding a moving or polynomial calibration FIELD to an MSv4 field",
194            });
195        };
196        let mut matched = None;
197        for entry in self
198            .fields
199            .iter()
200            .filter(|entry| names_agree(&entry.name, &field.name))
201        {
202            // An ephemeris-tracking field stores an offset here. Even a zero offset
203            // cannot identify a fixed direction with the same value.
204            if entry.tracks_ephemeris {
205                return Err(ObservationError::Unsupported {
206                    what: "binding an ephemeris-tracking MSv4 field to a calibration FIELD",
207                });
208            }
209            if same_direction(&entry.phase_centre, direction) && matched.replace(entry).is_some() {
210                return Err(ambiguous(&field.name));
211            }
212        }
213        let Some(entry) = matched else {
214            return Err(ObservationError::InvalidRequest {
215                reason: format!(
216                    "calibration table field {} is absent from the MSv4 store",
217                    field.name
218                ),
219            });
220        };
221        if rows.iter().enumerate().any(|(row, other)| {
222            row != id
223                && names_agree(&entry.name, &other.name)
224                && other
225                    .fixed_direction
226                    .as_ref()
227                    .is_some_and(|centre| same_direction(&entry.phase_centre, centre))
228        }) {
229            return Err(ambiguous(&field.name));
230        }
231        Ok(())
232    }
233}

235fn binding_scratch_bytes(rows: usize, selector: &str) -> usize {
236    let terms = selector.bytes().filter(|byte| *byte == b',').count() + 1;
237    // Each term can append every row. Account for the growing vector and
238    // temporary matches, numeric ranges and two possible numeric aliases per row.
239    let ids = terms.saturating_add(4).saturating_mul(3 * size_of::<i32>());
240    let aliases = 2 * (size_of::<(i32, i32)>() + 4 * size_of::<usize>());
241    let per_row = size_of::<CalibrationField>() + size_of::<FieldSelectorRow>() + aliases;
242    rows.saturating_mul(per_row.saturating_add(ids))
243        .saturating_add(terms.saturating_mul(3 * size_of::<(i32, i32)>()))
244        .saturating_add(selector.len())
245}
246
247fn admit(budget: usize, required: usize, what: &'static str) -> Result<(), ObservationError> {
248    if required > budget {
249        Err(ObservationError::OverBudget {
250            what,
251            required,
252            budget,
253        })
254    } else {
255        Ok(())
256    }
257}

259pub(super) fn fixed_direction(
260    table: &Table,
261    field: &MsField<'_>,
262    row: usize,
263) -> Result<Option<MDirection>, ObservationError> {
264    if field
265        .ephemeris_id(row)
266        .map_err(storage)?
267        .is_some_and(|id| id >= 0)
268        || optional_i32(table, row, "NUM_POLY")
269            .map_err(storage)?
270            .is_some_and(|terms| terms > 0)
271        || field
272            .phase_dir(row)
273            .map_err(storage)?
274            .shape()
275            .get(1)
276            .is_some_and(|&terms| terms > 1)
277    {
278        return Ok(None);
279    }
280    let (lon, lat) = DirectionColumn::phase_dir(table)
281        .get_angles(row)
282        .map_err(storage)?;
283    let frame =
284        resolve_direction_reference_selected(table, "FIELD", "PHASE_DIR", row).map_err(storage)?;
285    Ok(Some(MDirection::from_angles(lon, lat, frame)))
286}

288fn names_agree(msv4: &str, calibration: &str) -> bool {
289    msv4 == calibration
290        || msv4
291            .rsplit_once('_')
292            .is_some_and(|(base, suffix)| base == calibration && suffix.parse::<u32>().is_ok())
293}

295pub(super) fn same_direction(a: &MDirection, b: &MDirection) -> bool {
296    // Compare the unit vectors to allow roundoff from encoding the same
297    // angles in the two stores, including longitude wrap.
298    const ROUNDOFF: f64 = 8.0 * f64::EPSILON;
299    a.refer() == b.refer()
300        && a.cosines()
301            .into_iter()
302            .zip(b.cosines())
303            .map(|(left, right)| (left - right).powi(2))
304            .sum::<f64>()
305            <= ROUNDOFF * ROUNDOFF
306}
307
308fn ambiguous(name: &str) -> ObservationError {
309    ObservationError::InvalidRequest {
310        reason: format!("calibration table field {name} has an ambiguous MSv4 identity"),
311    }

314#[cfg(all(test, feature = "msv4", unix))]
315mod tests {
316    use super::*;
317    use casa_core::{ArrayValue, PrimitiveType, RecordField, RecordValue, ScalarValue, Value};
318    use casa_tables::{ColumnSchema, TableSchema};
319
320    #[test]
321    fn a_small_msv4_catalogue_does_not_admit_a_large_calibration_field_index() {
322        let root = tempfile::tempdir().unwrap();
323        let archive = include_bytes!("../../tests/fixtures/ms_trial/mssel_test_small.ps.zarr.tgz");
324        tar::Archive::new(flate2::read::GzDecoder::new(&archive[..]))
325            .unpack(root.path())
326            .unwrap();
327        let source = root.path().join("mssel_test_small.ps.zarr");
328        let mut opening_peak = 0;
329        let observation = super::super::Msv4Observation::open_granted(
330            &source,
331            &super::super::OpenOptions::default(),
332            &mut |required| {
333                opening_peak = opening_peak.max(required);
334                Ok(())
335            },
336        )
337        .unwrap();
338        let budget = opening_peak.max(observation.held_bytes()) + (128 << 10);
339        let selected = &observation.catalogue().fields[0];
340        let (lon, lat) = selected.phase_centre.as_angles();
341        let fields = Msv4FieldSelection::open(&source, budget).unwrap();
342
343        let directory = root.path().join("caltable");
344        std::fs::create_dir(&directory).unwrap();
345        let path = directory.join("FIELD");
346        let schema = TableSchema::new(vec![
347            ColumnSchema::scalar("NAME", PrimitiveType::String),
348            ColumnSchema::array_fixed("PHASE_DIR", PrimitiveType::Float64, vec![2, 1]),
349        ])
350        .unwrap();
351        let mut table = Table::with_schema(schema);
352        let record = |name: String| {
353            RecordValue::new(vec![
354                RecordField::new("NAME", Value::Scalar(ScalarValue::String(name))),
355                RecordField::new(
356                    "PHASE_DIR",
357                    Value::Array(ArrayValue::Float64(
358                        ndarray::arr2(&[[lon], [lat]]).into_dyn(),
359                    )),
360                ),
361            ])
362        };
363        table.add_row(record(selected.name.clone())).unwrap();
364        table.save(TableOptions::new(&path)).unwrap();
365        assert_eq!(fields.select(&directory, "0").unwrap(), [0]);
366
367        let unselected = budget / binding_scratch_bytes(1, "0") + 1;
368        for _ in 0..unselected {
369            table.add_row(record("unselected".to_owned())).unwrap();
370        }
371        table.save(TableOptions::new(&path)).unwrap();
372        assert!(matches!(
373            fields.select(&directory, "0"),
374            Err(ObservationError::OverBudget {
375                what: "MSv4 calibration FIELD binding",
376                required,
377                budget: admitted,
378            }) if required > admitted && admitted == budget
379        ));
380    }
