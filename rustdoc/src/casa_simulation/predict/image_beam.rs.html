2//! Image-plane beams that mix the sky's Stokes planes before gridding.
3use super::{BeamPattern, PredictError, SkyModel, jones::JonesImage};
4use casa_imaging_runtime::Admitted;
5use ndarray::Array2;
6use std::sync::Arc;
7
8pub(crate) enum ImageBeam {
9    Jones(Arc<JonesImage>),
10    Radial(Arc<Admitted<BeamPattern>>),
11    Squinted {
12        pattern: Arc<Admitted<BeamPattern>>,
13        offset: [f64; 2],
14        reference_hz: f64,
15    },
16}
17
18impl ImageBeam {
19    pub(crate) fn apply_plane(
20        &self,
21        model: &SkyModel,
22        plane: usize,
23        images: Vec<Array2<f32>>,
24        pointing: [f64; 2],
25        angle: f64,
26        frame: Option<&casa_core::measures::MeasFrame>,
27    ) -> Result<Vec<Array2<f32>>, PredictError> {
28        let pixel_of = |direction| {
29            frame.map_or_else(
30                || model.pixel_of(direction),
31                |frame| model.pixel_of_with_frame(direction, frame),
32            )
33        };
34        match self {
35            Self::Jones(image) => image.apply_plane(model, plane, images, pointing, angle, frame),
36            Self::Radial(pattern) => {
37                let centre = pixel_of(pointing)?;
38                Ok(radial_plane(
39                    model,
40                    plane,
41                    images,
42                    &pattern.value,
43                    [centre; 3],
44                ))
45            }
46            Self::Squinted {
47                pattern,
48                offset,
49                reference_hz,
50            } => {
51                let frequency = model.plane_frequency_hz(plane);
52                // PBMath1D converts POINTING before applying the squint;
53                // displacement axes belong to the image's native frame.
54                let centre = frame.map_or_else(
55                    || Ok(pointing),
56                    |frame| model.native_pointing_with_frame(pointing, frame),
57                )?;
58                let [right, left] = [1., -1.].map(|sign| {
59                    // BeamSquint::getPointingDirection quantizes the
60                    // frequency scale before rotating the saved AZEL offset.
61                    let scale = f64::from((-reference_hz / frequency) as f32) * sign;
62                    let [sx, sy] = offset.map(|v| v * scale);
63                    let (sin, cos) = (-angle).sin_cos();
64                    let (dx, dy) = (-(sx * cos + sy * sin), -sx * sin + sy * cos);
65                    let [ra, dec] = centre;
66                    let direction = [
67                        ra + dx.sin().atan2(dx.cos() * (dec + dy).cos()),
68                        (dx.cos() * (dec + dy).sin()).asin(),
69                    ];
70                    if frame.is_some() {
71                        model.pixel_of_native(direction)
72                    } else {
73                        model.pixel_of(direction)
74                    }
75                });
76                let centres = [right?, left?, pixel_of(pointing)?];
77                Ok(radial_plane(model, plane, images, &pattern.value, centres))
78            }
79        }
80    }
81}

83fn radial_plane(
84    model: &SkyModel,
85    plane: usize,
86    images: Vec<Array2<f32>>,
87    pattern: &BeamPattern,
88    centres: [[f64; 2]; 3],
89) -> Vec<Array2<f32>> {
90    use casa_imaging_model::PolarizationCoordinate::{StokesI, StokesQ, StokesU, StokesV};
91    let frequency = model.plane_frequency_hz(plane);
92    let at = pattern.at_frequency(frequency);
93    let increments = model.increment_rad.map(f64::to_degrees);
94    let [nx, ny] = model.shape();
95    let mut out: Vec<_> = (0..4).map(|_| Array2::zeros((ny, nx))).collect();
96    for y in 0..ny {
97        for x in 0..nx {
98            let powers = centres.map(|centre| {
99                let rx = ((x as f64 - centre[0]) * increments[0]).powi(2) as f32;
100                let ry = ((y as f64 - centre[1]) * increments[1]).powi(2) as f32;
101                let radius = (f64::from((rx + ry).sqrt()) * 60. * frequency / 1e9) as f32;
102                at.power(f64::from(radius))
103            });
104            let mut stokes = [0.; 4];
105            for (pol, image) in model.polarizations().iter().zip(&images) {
106                let slot = match pol {
107                    StokesI => 0,
108                    StokesQ => 1,
109                    StokesU => 2,
110                    StokesV => 3,
111                    _ => unreachable!("sky models contain Stokes planes"),
112                };
113                stokes[slot] = image[[y, x]];
114            }
115            let [i, q, u, v] = stokes;
116            let rr = (i + v) * powers[0];
117            let ll = (i - v) * powers[1];
118            // PBMath1D applies the central beam to RL/LR rather than
119            // the product of displaced voltage patterns.
120            for (output, value) in
121                out.iter_mut()
122                    .zip([(rr + ll) / 2., q * powers[2], u * powers[2], (rr - ll) / 2.])
123            {
124                output[[y, x]] = value;
125            }
126        }
127    }
128    out
129}
