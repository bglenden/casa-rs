2//! The `table` tool's write side for keywords and the table info, as CASA
3//! 6.7.6.14's `tb` does it (`tools/casa-compat/table_tool_oracle.py`).
4//!
5//! Every write is made through to the table on disk and touches only the
6//! table's metadata: `Table::save_metadata_only` rewrites `table.dat`'s
7//! keywords and `table.info` and leaves the storage managers as they are
8//! (`Table::flush` or a plain unlock would rewrite the table as
9//! StManAipsIO). A table opened with an automatic lock (`default`, `auto`,
10//! `autonoread`) takes the write lock for the write and releases it after;
11//! a table opened with `user` locking must hold the write lock, and one
12//! opened with `permanent` locking holds it. A table opened read-only
13//! refuses. With no table open a write logs and returns `False`.
14
15use casa_tables::{LockType, Table, TableError, TableOptions};
16use casa_types::{RecordField, RecordValue, Value};

18use super::keyword_value::Incoming;
19use super::read::KeywordRef;
20use super::state::{Fail, State, no_table};
21use crate::convert::value::CasaValue;

23type Reply = Result<CasaValue, Fail>;

25/// What a write changes; it names the refusal of a table not open for
26/// update.
27#[derive(Clone, Copy)]
28pub(super) enum Target {
29    TableKeywords,
30    ColumnKeywords,
31    Info,
32    /// Cells: a table not open for update answers `False`, not an error.
33    Cells,
34    /// The table's rows or columns, as the casacore operation named here
35    /// refuses a table not open for update.
36    Structure(&'static str),
37}
38
39/// What a change leaves to write to the disk.
40pub(super) enum Persist {
41    /// The table's keywords and info.
42    Metadata,
43    /// The cells of `rows` in `column`, in place in the column's storage
44    /// manager.
45    Rows { column: String, rows: Vec<usize> },
46    /// Nothing: the call selected no cell.
47    Nothing,
48    /// The change has written itself to the disk (a structural change in
49    /// place, or a table written in full).
50    Done,
51}

53/// What `putinfo` changes: the fields its record has.
54#[derive(Debug, Default)]

56    pub(crate) table_type: Option<String>,
57    pub(crate) sub_type: Option<String>,

61fn runtime<T>(message: String) -> Result<T, Fail> {
62    Err(Fail::Runtime(message))
63}

65/// The record a dotted `path` names inside `record`, made if `make` is set.
66fn descend<'a>(
67    record: &'a mut RecordValue,
68    path: &str,
69    make: bool,
70) -> Result<&'a mut RecordValue, Fail> {
71    let mut current = record;
72    for part in path.split('.') {
73        if current.get(part).is_none() {
74            if !make {
75                return runtime(format!("RecordInterface: field {part} is unknown"));
76            }
77            current.push(RecordField::new(
78                part,
79                Value::Record(RecordValue::new(Vec::new())),
80            ));
81        }
82        match current.get_mut(part) {
83            Some(Value::Record(inner)) => current = inner,
84            _ => return runtime(format!("RecordInterface: field {part} is unknown")),
85        }
86    }
87    Ok(current)
88}

90/// Defines `name` in `record`: a new field takes the value's natural type,
91/// an existing one keeps its own.
92fn store(record: &mut RecordValue, name: &str, value: Incoming) -> Result<(), Fail> {
93    match record.get_mut(name) {
94        Some(old) => {
95            let new = value.like(old, name)?;
96            *old = new;
97        }
98        None => record.push(RecordField::new(name, value.natural()?)),
99    }
100    Ok(())
101}

103/// `putkeyword`'s rule: by (dotted) name or by 1-based position.
104fn put(
105    record: &mut RecordValue,
106    key: &KeywordRef,
107    value: Incoming,
108    make_subrecord: bool,
109) -> Result<(), Fail> {
110    match key {
111        KeywordRef::Whole => Ok(()),
112        KeywordRef::Name(name) => match name.rsplit_once('.') {
113            Some((parent, last)) => store(descend(record, parent, make_subrecord)?, last, value),
114            None => store(record, name, value),
115        },
116        KeywordRef::Index(index) => {
117            let slot = usize::try_from(*index)
118                .ok()
119                .filter(|slot| (1..=record.len()).contains(slot));
120            let Some(slot) = slot else {
121                return runtime("RecordInterface::define - new fieldNumber exceeds #fields".into());
122            };
123            let field = &mut record.fields_mut()[slot - 1];
124            let new = value.like(&field.value, &field.name)?;
125            field.value = new;
126            Ok(())
127        }
128    }
129}

131/// `removekeyword`'s rule; `missing` words the refusal of an absent name.
132fn remove(
133    record: &mut RecordValue,
134    key: &KeywordRef,
135    missing: impl Fn(&str) -> String,
136) -> Result<(), Fail> {
137    match key {
138        KeywordRef::Whole => Ok(()),
139        KeywordRef::Name(name) => {
140            let (parent, last) = name.rsplit_once('.').unwrap_or(("", name));
141            let target = if parent.is_empty() {
142                record
143            } else {
144                descend(record, parent, false).map_err(|_| Fail::Runtime(missing(name)))?
145            };
146            match target.remove(last) {
147                Some(_) => Ok(()),
148                None => runtime(missing(name)),
149            }
150        }
151        KeywordRef::Index(index) => {
152            let name = usize::try_from(*index)
153                .ok()
154                .and_then(|slot| slot.checked_sub(1))
155                .and_then(|slot| record.fields().get(slot))
156                .map(|field| field.name.clone());
157            match name {
158                Some(name) => {
159                    record.remove(&name);
160                    Ok(())
161                }
162                None => runtime("RecordRep::deleteDataField".into()),
163            }
164        }
165    }
166}

168/// The column `column`'s keywords, edited by `edit`.
169fn edit_column(
170    table: &mut Table,
171    column: &str,
172    edit: impl FnOnce(&mut RecordValue) -> Result<(), Fail>,
173) -> Result<(), Fail> {
174    if table
175        .schema()
176        .and_then(|schema| schema.column(column))
177        .is_none()
178    {
179        return runtime(format!("Table column {column} is unknown"));
180    }
181    let mut keywords = table
182        .column_keywords(column)
183        .cloned()
184        .unwrap_or_else(|| RecordValue::new(Vec::new()));
185    edit(&mut keywords)?;
186    table.set_column_keywords(column, keywords);
187    Ok(())
188}

190impl State {
191    /// Applies `change` to the open table and saves its metadata, under the
192    /// lock the table's lock options call for.
193    fn write(
194        &mut self,
195        target: Target,
196        log: &mut Vec<crate::tools::LogLine>,
197        change: impl FnOnce(&mut Table) -> Result<(), Fail>,
198    ) -> Reply {
199        self.write_with(target, log, move |table| {
200            change(table).map(|()| Persist::Metadata)
201        })
202    }
203
204    /// As [`write`](Self::write), for a change that says what to persist.
205    pub(super) fn write_with(
206        &mut self,
207        target: Target,
208        log: &mut Vec<crate::tools::LogLine>,
209        change: impl FnOnce(&mut Table) -> Result<Persist, Fail>,
210    ) -> Reply {
211        let Some(open) = self.open.as_mut() else {
212            no_table(log);
213            return Ok(CasaValue::Bool(false));
214        };
215        let name = open.name.display().to_string();
216        if !open.writable {
217            if matches!(target, Target::Cells) {
218                return Ok(CasaValue::Bool(false));
219            }
220            return runtime(match target {
221                Target::Cells => String::new(),
222                Target::TableKeywords => {
223                    format!("Table::rwKeywordSet cannot be used: table {name} is not writable")
224                }
225                Target::ColumnKeywords => {
226                    format!(
227                        "TableColumn::rwKeywordSet cannot be used: table {name} is not writable"
228                    )
229                }
230                Target::Info => format!("Table {name} is not writable"),
231                Target::Structure(operation) => {
232                    format!("Invalid Table operation: {operation}; table {name} is not writable")
233                }
234            });
235        }
236        let automatic = open.lock.automatic();
237        if automatic {
238            if !open.table.lock(LockType::Write, open.lock.attempts())? {
239                return runtime(format!("Table {name} could not be write locked"));
240            }
241        } else if !open.table.has_lock(LockType::Write) {
242            return runtime(format!(
243                "ColumnSet::doLock: table {name} should be locked when using UserLocking"
244            ));
245        }
246        // Whether the change wrote itself and the handle holds what it wrote.
247        let mut written = false;
248        let saved = change(&mut open.table).and_then(|persist| match persist {
249            Persist::Nothing => Ok(()),
250            Persist::Done => {
251                written = true;
252                Ok(())
253            }
254            Persist::Metadata => open
255                .table
256                .save_metadata_only(TableOptions::new(&open.name))
257                .map_err(Fail::from),
258            Persist::Rows { column, rows } => {
259                let partial = open
260                    .table
261                    .prepare_write()
262                    .save_selected_rows(&[column.as_str()], &rows);
263                match partial {
264                    // StManAipsIO keeps its table in one file and writes it
265                    // in full, as casacore does. A table that is one such
266                    // group is written in full with the group's name and
267                    // byte order kept, which is the layout it has. Any
268                    // other table must not be rewritten (it would lose its
269                    // managers), so the refusal stands.
270                    Err(TableError::Storage(message))
271                        if message.starts_with("partial in-place save does not support") =>
272                    {
273                        match open.table.single_stman_aipsio_options()? {
274                            Some(options) => open.table.save(options).map_err(Fail::from),
275                            None => Err(Fail::from(TableError::Storage(message))),
276                        }
277                    }
278                    other => other.map_err(Fail::from),
279                }
280            }
281        });
282        // A table that holds the write lock and has changes it has not
283        // flushed is written in full, as StManAipsIO, when it is closed or
284        // dropped: a `user` table closed without an unlock, a `permanent`
285        // one always (a `casa-tables` defect, #1035; this reload is the
286        // workaround and goes with its fix). The change is on disk, or part
287        // of it is if the save failed part way; either way the table is
288        // reloaded as the disk has it, so that nothing is left to flush, and
289        // what reached the disk is still published when the lock is
290        // released. A reload that fails marks the table clean all the same
291        // (`resync_or_discard`): the full write must never be left to a
292        // drop. An automatic table releases through
293        // `unlock_metadata_only_or_discard`, which does the same when the
294        // release fails. The release comes last whatever else went wrong,
295        // so an error never leaves the write lock held with changes. A
296        // change that wrote itself needs no reload: the handle is what it
297        // wrote (a full save marks it flushed), and a reload under the lock
298        // would read the row count last published, which a removal of rows
299        // has lowered.
300        let reloaded = if saved.is_err() || (!automatic && !written) {
301            open.table.resync_or_discard().map_err(Fail::from)
302        } else {
303            Ok(())
304        };
305        let released = if automatic {
306            open.table
307                .unlock_metadata_only_or_discard()
308                .map_err(Fail::from)
309        } else {
310            Ok(())
311        };
312        saved?;
313        reloaded?;
314        released?;
315        Ok(CasaValue::Bool(true))
316    }

318    /// `putkeyword(keyword, value, makesubrecord)`.
319    pub(super) fn putkeyword(
320        &mut self,
321        key: KeywordRef,
322        value: Incoming,
323        make_subrecord: bool,
324        log: &mut Vec<crate::tools::LogLine>,
325    ) -> Reply {
326        if matches!(&key, KeywordRef::Whole) {
327            return Ok(CasaValue::Bool(false));
328        }
329        self.write(Target::TableKeywords, log, move |table| {
330            put(table.keywords_mut(), &key, value, make_subrecord)
331        })
332    }
333
334    /// `putkeywords(value)`: each field is defined as `putkeyword` does.
335    pub(super) fn putkeywords(
336        &mut self,
337        fields: Vec<(String, Incoming)>,
338        log: &mut Vec<crate::tools::LogLine>,
339    ) -> Reply {
340        self.write(Target::TableKeywords, log, move |table| {
341            for (name, value) in fields {
342                put(table.keywords_mut(), &KeywordRef::Name(name), value, false)?;
343            }
344            Ok(())
345        })
346    }

348    /// `removekeyword(keyword)`.
349    pub(super) fn removekeyword(
350        &mut self,
351        key: KeywordRef,
352        log: &mut Vec<crate::tools::LogLine>,
353    ) -> Reply {
354        self.write(Target::TableKeywords, log, move |table| {
355            remove(table.keywords_mut(), &key, |name| {
356                format!("Table keyword {name} does not exist")
357            })
358        })
359    }

410    /// `putinfo(value)`: the fields `value` has replace the table's type,
411    /// subtype and readme; the readme text is kept as lines.
412    pub(super) fn putinfo(
413        &mut self,
414        update: InfoUpdate,
415        log: &mut Vec<crate::tools::LogLine>,
416    ) -> Reply {
417        self.write(Target::Info, log, move |table| {
418            let mut info = table.info().clone();
419            if let Some(table_type) = update.table_type {
420                info.table_type = table_type;
421            }
422            if let Some(sub_type) = update.sub_type {
423                info.sub_type = sub_type;
424            }
425            if let Some(readme) = update.readme {
426                info.readme = readme.split('\n').map(str::to_string).collect();
427            }
428            table.set_info(info);
429            Ok(())
430        })
431    }
432
433    /// `addreadmeline(value)`: appends a line to the readme.
434    pub(super) fn addreadmeline(
435        &mut self,
436        line: String,
437        log: &mut Vec<crate::tools::LogLine>,
438    ) -> Reply {
439        self.write(Target::Info, log, move |table| {

441            info.readme.push(line);
442            table.set_info(info);
443            Ok(())
444        })
445    }
