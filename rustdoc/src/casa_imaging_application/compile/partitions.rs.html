2//! The selection resolved to the observation's partitions (#932).
3//!
4//! Compile names the partitions that hold the request's samples
5//! ([`candidates`]): a partition's window must be selected, its fields must
6//! include a selected one, and, under an intent selection, its observing
7//! mode must be a selected one ([`super::intents`]). The survey then keeps
8//! those that hold a selected sample, and the compiled problem names them
9//! ([`casa_imaging_model::PartitionSelection`]). The run reads the same
10//! partitions ([`resolve_partitions`]).

30/// Speed of light in m/s, for a window's reference wavelength.
31const SPEED_OF_LIGHT_M_S: f64 = 299_792_458.0;

33/// Why the selection cannot be read through the observation's partitions.
34#[derive(Debug, thiserror::Error)]
35pub enum PartitionSelectionError {
36    /// What the passes do not read.
37    #[error("native imaging does not read {0}")]
38    Unsupported(&'static str),
39    /// A selected product the partition does not hold, or a window
40    /// without channels or reference frequency.
41    #[error("the selection is inconsistent with the observation: {0}")]
42    Inconsistent(String),
43}
44
45/// One partition the request selects, before the survey reads its samples.
46pub(super) struct Candidate {
47    pub(super) id: PartitionId,
48    /// Its spectral window number (the catalogue's window; an MSv2
49    /// `SPECTRAL_WINDOW` row).
50    pub(super) spectral_window: usize,
51    /// Its window's reference wavelength, when a UV-distance interval is in
52    /// wavelengths.
53    pub(super) wavelength_m: Option<f64>,
54}

56/// The partitions a request selects, and the data group holding the
57/// visibility column it images.
58pub(super) struct Candidates {
59    pub(super) data_group: String,
60    pub(super) partitions: Vec<Candidate>,
61}

63/// The partitions of `store` in the windows `windows` selects that `rows`
64/// may select, in partition order: a partition's fields must include a
65/// selected one and, under an intent selection, its observing mode must be
66/// a selected one. A state number in the selection (`states`) must select
67/// every state of its partition. A partition's UV distances and times are
68/// the survey's to test.
69///
70/// # Errors
71///
72/// [`PartitionSelectionError::Unsupported`] for an intent selection by
73/// state that splits an `OBS_MODE`.
74pub(super) fn selected(
75    store: &OpenedStore,
76    windows: &WindowChoice,
77    (rows, states): (&RowSelection, &BTreeSet<StateId>),
78) -> Result<Vec<PartitionId>, PartitionSelectionError> {
79    let fields = field_selection(rows.fields());
80    let modes = match rows.intents() {
81        IntentSelection::All => None,
82        IntentSelection::Only(modes) => Some(modes.iter().cloned().collect::<BTreeSet<_>>()),
83    };
84    let catalogue = store.catalogue();
85    let mut selected = Vec::new();
86    for id in catalogue.partition_ids() {
87        let entry = catalogue.partition(id);
88        if !windows.holds(store, id) {
89            continue;
90        }
91        if let Some(modes) = &modes {
92            if !modes.contains(&intents::observing_mode(catalogue, entry)) {
93                continue;
94            }
95            if !entry.states.is_empty() && !entry.states.iter().any(|state| states.contains(state))
96            {
97                continue;
98            }
99            if intents::split_state(entry, states) {
100                return Err(PartitionSelectionError::Unsupported(
101                    "an intent selection that takes some of the states sharing an OBS_MODE (the \
102                     MSv2 adapter partitions their rows together)",
103                ));
104            }
105        }
106        if entry.fields.iter().any(|field| fields.holds(*field)) {
107            selected.push(id);
108        }
109    }
110    Ok(selected)
111}

113/// The `selected` partitions of `store` ([`selected`]) with the data group
114/// holding `visibility`, which each must have.
115///
116/// # Errors
117///
118/// [`PartitionSelectionError::Unsupported`] for what the passes do not
119/// read: `FLOAT_DATA` (no request names it) and a partition without the data
120/// group. [`PartitionSelectionError::Inconsistent`] for a window without a
121/// positive reference frequency under a UV-distance interval in
122/// wavelengths.
123pub(super) fn candidates(
124    store: &OpenedStore,
125    selected: &[PartitionId],
126    rows: &RowSelection,
127    visibility: VisibilityColumn,
128) -> Result<Candidates, PartitionSelectionError> {
129    if visibility == VisibilityColumn::FloatData {
130        return Err(PartitionSelectionError::Unsupported("FLOAT_DATA"));
131    }
132    let group = data_group(visibility);
133    let catalogue = store.catalogue();
134    let mut partitions = Vec::new();
135    for &id in selected {
136        let entry = catalogue.partition(id);
137        if entry.data_group(group).is_none() {
138            return Err(PartitionSelectionError::Unsupported(
139                "a selected partition without the imaged data group",
140            ));
141        }
142        partitions.push(Candidate {
143            id,
144            spectral_window: entry.window.0 as usize,
145            wavelength_m: wavelength(catalogue, entry, rows)?,
146        });
147    }
148    Ok(Candidates {
149        data_group: group.to_string(),
150        partitions,
151    })
152}

154/// `problem`'s selection resolved to `observation`'s partitions, in
155/// partition order: the partitions compile chose, the channels of their
156/// windows, their correlations, the data group holding the visibility
157/// column, and the per-sample predicates.
158///
159/// # Errors
160///
161/// [`PartitionSelectionError::Unsupported`] for several MeasurementSets,
162/// `FLOAT_DATA` and partitions whose column belongs to different data
163/// groups; [`PartitionSelectionError::Inconsistent`] for a window the
164/// selection names no channel of, a correlation a partition does not hold,
165/// and a window without a positive reference frequency under a UV-distance
166/// interval in wavelengths.
167pub(crate) fn resolve_partitions(
168    problem: &CompiledProblem,
169    observation: &dyn Observation,
170) -> Result<PartitionSelection, PartitionSelectionError> {
171    let [source] = problem.observation_transaction().read_set().sources() else {
172        return Err(PartitionSelectionError::Unsupported(
173            "more than one MeasurementSet",
174        ));
175    };
176    let selection = source.selection();
177    let visibility = source.selected_columns().visibility();
178    if visibility == VisibilityColumn::FloatData {
179        return Err(PartitionSelectionError::Unsupported("FLOAT_DATA"));
180    }
181    let group = data_group(visibility);
182    let catalogue = observation.catalogue();
183    let rows = selection.rows_filter();
184    let mut partitions = Vec::with_capacity(selection.partitions().len());
185    for selected in selection.partitions() {
186        let entry = catalogue.partition(selected.partition());
187        if entry.data_group(group).is_none() {
188            return Err(PartitionSelectionError::Unsupported(
189                "a selected partition without the imaged data group",
190            ));
191        }
192        let window = selected.spectral_window_id();
193        let channels = selection
194            .spectral_windows()
195            .iter()
196            .find(|selected| selected.spectral_window_id() == window)
197            .map(|selected| selected.channel_indices().to_vec())
198            .filter(|channels| !channels.is_empty())
199            .ok_or_else(|| {
200                PartitionSelectionError::Inconsistent(format!(
201                    "spectral window {window} has no selected channel"
202                ))
203            })?;
204        let correlations: Vec<Correlation> = selected
205            .products()
206            .iter()
207            .map(|product| product.correlation_type())
208            .collect();
209        if let Some(missing) = correlations
210            .iter()
211            .find(|product| entry.correlation_index(**product).is_none())
212        {
213            return Err(PartitionSelectionError::Inconsistent(format!(
214                "spectral window {window} lacks {missing:?}"
215            )));
216        }
217        partitions.push(SelectedPartition {
218            id: selected.partition(),
219            spectral_window: window,
220            channels,
221            correlations,
222            wavelength_m: wavelength(catalogue, entry, rows)?,
223        });
224    }
225    if partitions.is_empty() {
226        return Err(PartitionSelectionError::Unsupported(
227            "a selection no partition holds",
228        ));
229    }
230    Ok(PartitionSelection {
231        data_group: group.to_string(),
232        partitions,
233        fields: field_selection(rows.fields()),
234        uv: rows.uv_distances().clone(),
235    })
236}

238/// The fields `fields` selects, as the boundary names them.
239pub(super) fn field_selection(fields: &IdSelection) -> FieldSelection {
240    match fields {
241        IdSelection::All => FieldSelection::All,
242        IdSelection::Only(ids) => FieldSelection::Only(ids.iter().copied().map(FieldId).collect()),
243    }
244}

246/// `entry`'s window's reference wavelength, when `rows` tests a UV distance
247/// in wavelengths.
248fn wavelength(
249    catalogue: &Catalogue,
250    entry: &PartitionEntry,
251    rows: &RowSelection,
252) -> Result<Option<f64>, PartitionSelectionError> {
253    if !rows.uv_distances().needs_wavelengths() {
254        return Ok(None);
255    }
256    let window = entry.window.0;
257    catalogue
258        .windows
259        .get(window as usize)
260        .and_then(|window| window.reference_frequency.as_ref())
261        .map(casa_types::measures::frequency::MFrequency::hz)
262        .filter(|hz| hz.is_finite() && *hz > 0.0)
263        .map(|hz| Some(SPEED_OF_LIGHT_M_S / hz))
264        .ok_or_else(|| {
265            PartitionSelectionError::Inconsistent(format!(
266                "spectral window {window} has no positive reference frequency"
267            ))
268        })
269}

271/// The weights the partitions `ids` of `store` hold beside `visibility`:
272/// `WEIGHT_SPECTRUM` when any of them reads per-channel weights, else
273/// `WEIGHT`. The MSv2 adapter decides each partition's by CASA's rule, at
274/// the first of its rows; an MSv4's weights are per channel. The passes
275/// read each partition's own weights; this names them in the problem's
276/// manifest.
277pub(super) fn stored_weights(
278    store: &OpenedStore,
279    ids: impl IntoIterator<Item = PartitionId>,
280    visibility: VisibilityColumn,
281) -> WeightColumn {
282    let group = data_group(visibility);
283    let catalogue = store.catalogue();
284    let spectral = store.msv2().is_none()
285        || ids.into_iter().any(|id| {
286            catalogue
287                .partition(id)
288                .data_group(group)
289                .is_some_and(|(_, entry)| entry.weights == "WEIGHT_SPECTRUM")
290        });
291    if spectral {
292        WeightColumn::WeightSpectrum
293    } else {
294        WeightColumn::Weight
295    }
