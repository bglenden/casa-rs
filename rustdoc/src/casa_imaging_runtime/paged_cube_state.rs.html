22/// Geometry and exact physical values layout for one reconstruction-owned array.
23///
24/// Scalar cardinality comes from the reconstruction owner. Cache tiles and the
25/// maximum accessed window are independent: sequential I/O can retain one
26/// plane while the caller holds a multi-plane window.
27#[derive(Debug, Clone)]
28pub(crate) struct CubeArrayLayout {
29    logical_scalars: usize,
30    window_scalars: usize,
31    axis0: usize,
32    axis1: usize,
33    values: TiledArrayStorageLayout,
34}
35
36impl CubeArrayLayout {
37    pub(crate) fn new(
38        logical_scalars: usize,
39        plane_scalars: usize,
40        window_scalars: usize,
41        cache_tiles: usize,
42    ) -> Result<Self, String> {
43        Self::new_spatial(
44            logical_scalars,
45            plane_scalars,
46            1,
47            window_scalars,
48            cache_tiles,
49        )
50    }
51
52    /// Axis zero is the contiguous native y coordinate; axis one is x (and any
53    /// interleaved scalar/polarization factor). The third axis holds planes.
54    pub(crate) fn new_spatial(
55        logical_scalars: usize,
56        axis0: usize,
57        axis1: usize,
58        window_scalars: usize,
59        cache_tiles: usize,
60    ) -> Result<Self, String> {
61        if logical_scalars == 0
62            || axis0 == 0
63            || axis1 == 0
64            || window_scalars == 0
65            || cache_tiles == 0
66        {
67            return Err("cube array dimensions, window and cache must be positive".into());
68        }
69        let plane_scalars = axis0.checked_mul(axis1).ok_or("cube plane size overflow")?;
70        let planes = logical_scalars.div_ceil(plane_scalars);
71        let cache_tiles = cache_tiles.min(planes);
72        let cache_bytes = plane_scalars
73            .checked_mul(cache_tiles)
74            .and_then(|n| n.checked_mul(size_of::<f64>()))
75            .ok_or("cube array cache capacity overflow")?;
76        let shape = TiledShape::with_tile_shape(vec![axis0, axis1, planes], vec![axis0, axis1, 1])
77            .map_err(|error| error.to_string())?;
78        let values = PagedArray::<f64>::storage_layout(shape, cache_bytes)
79            .map_err(|error| error.to_string())?;
80        Ok(Self {
81            logical_scalars,
82            window_scalars: window_scalars.min(logical_scalars),
83            axis0,
84            axis1,
85            values,
86        })
87    }
88
89    fn support_layout(&self) -> Result<TiledArrayStorageLayout, String> {
90        let shape = TiledShape::with_tile_shape(
91            self.values.cube_shape().to_vec(),
92            self.values.tile_shape().to_vec(),
93        )
94        .map_err(|error| error.to_string())?;
95        PagedArray::<bool>::storage_layout(
96            shape,
97            self.values.cache_budget_bytes() / size_of::<f64>(),
98        )
99        .map_err(|error| error.to_string())
100    }
101
102    pub(crate) fn normal_ledger(&self, parent: &Path) -> Result<CubeArrayLedger, String> {
103        let directory = planned_directory(parent, NORMAL_DIRECTORY_PREFIX);
104        let values = directory.join("values");
105        let retained = PagedArray::<f64>::planned_persistent_heap_bytes(&self.values, &values)
106            .map_err(|error| error.to_string())?;
107        let window = checked_bytes(self.window_scalars, size_of::<f64>())?;
108        Ok(CubeArrayLedger {
109            retained_bytes: checked_sum(&[
110                size_of::<PagedNormalArray>(),
111                directory.as_os_str().len(),
112                retained,
113            ])?,
114            cache_bytes: self
115                .values
116                .cache_payload_bytes()
117                .map_err(|error| error.to_string())?,
118            cache_index_bytes: self
119                .values
120                .cache_metadata_bytes()
121                .map_err(|error| error.to_string())?,
122            read_scratch_bytes: checked_sum(&[
123                window,
124                self.values
125                    .slice_scratch_bytes()
126                    .map_err(|error| error.to_string())?,
127            ])?,
128            write_scratch_bytes: self
129                .values
130                .slice_scratch_bytes()
131                .map_err(|error| error.to_string())?,
132            flush_scratch_bytes: self
133                .values
134                .flush_scratch_bytes()
135                .map_err(|error| error.to_string())?,
136            storage_bytes: self
137                .values
138                .storage_bytes()
139                .map_err(|error| error.to_string())?,
140            file_handles: 1,
141        })
142    }
143}

145fn transfer_range<E>(
146    axis0: usize,
147    axis1: usize,
148    start: usize,
149    len: usize,
150    mut transfer: impl FnMut([usize; 3], [usize; 3], std::ops::Range<usize>) -> Result<(), E>,
151) -> Result<(), E> {
152    let plane = axis0 * axis1;
153    let mut offset = start;
154    let end = start + len;
155    while offset < end {
156        let within_plane = offset % plane;
157        let y = within_plane % axis0;
158        let x = within_plane / axis0;
159        let channel = offset / plane;
160        let remaining = end - offset;
161        let (shape, count) = if within_plane == 0 && remaining >= plane {
162            let channels = remaining / plane;
163            ([axis0, axis1, channels], channels * plane)
164        } else if y == 0 && remaining >= axis0 {
165            let rows = (remaining / axis0).min(axis1 - x);
166            ([axis0, rows, 1], rows * axis0)
167        } else {
168            let row_values = remaining.min(axis0 - y);
169            ([row_values, 1, 1], row_values)
170        };
171        transfer(
172            [y, x, channel],
173            shape,
174            (offset - start)..(offset - start + count),
175        )?;
176        offset += count;
177    }
178    Ok(())
179}

181/// Complete owned payload/metadata projection for a created private backing.
182///
183/// Prepared factory metadata is counted separately by the factory. Read/write
184/// scratch excludes the reconstruction owner's input/destination window; it
185/// includes the remaining model-value/support conversion vectors and underlying
186/// typed tile conversion buffers. File lengths include standard metadata and packed edge-tile padding,
187/// not filesystem inode/block-allocation overhead or allocator bookkeeping.
188#[derive(Debug, Clone, Copy, PartialEq, Eq)]
189pub(crate) struct CubeArrayLedger {
190    pub(crate) retained_bytes: usize,
191    pub(crate) cache_bytes: usize,
192    pub(crate) cache_index_bytes: usize,
193    pub(crate) read_scratch_bytes: usize,
194    pub(crate) write_scratch_bytes: usize,
195    pub(crate) flush_scratch_bytes: usize,
196    pub(crate) storage_bytes: usize,
197    pub(crate) file_handles: usize,
198}

247impl CubeBackingMetrics {
248    pub(crate) fn snapshot(&self) -> CubeBackingSnapshot {
249        *self.0.lock().expect("cube backing metrics lock poisoned")
250    }

252    pub(crate) fn register(self: &Arc<Self>, ledger: CubeArrayLedger) -> BackingObservation {
253        let mut state = self.0.lock().expect("cube backing metrics lock poisoned");
254        state.live_owned_bytes += ledger.retained_bytes;
255        state.peak_owned_bytes = state.peak_owned_bytes.max(state.live_owned_bytes);
256        state.live_cache_bytes += ledger.cache_bytes;
257        state.peak_cache_bytes = state.peak_cache_bytes.max(state.live_cache_bytes);
258        state.live_cache_index_bytes += ledger.cache_index_bytes;
259        state.peak_cache_index_bytes = state
260            .peak_cache_index_bytes
261            .max(state.live_cache_index_bytes);
262        state.live_storage_bytes += ledger.storage_bytes;
263        state.peak_storage_bytes = state.peak_storage_bytes.max(state.live_storage_bytes);
264        state.live_file_handles += ledger.file_handles;
265        state.peak_file_handles = state.peak_file_handles.max(state.live_file_handles);
266        state.live_backings += 1;
267        state.peak_backings = state.peak_backings.max(state.live_backings);
268        BackingObservation {
269            metrics: self.clone(),
270            ledger,
271        }
272    }
273}

278    ledger: CubeArrayLedger,
279}

281impl BackingObservation {
282    pub(crate) fn record(&self, scalars: usize, delta: TiledFileIoStats) {
283        let mut state = self
284            .metrics
285            .0
286            .lock()
287            .expect("cube backing metrics lock poisoned");
288        state.max_access_scalars = state.max_access_scalars.max(scalars);
289        state.read_bytes +=
290            delta.flat_bulk_read_bytes + delta.lru_read_bytes + delta.lru_batch_load_bytes;
291        state.write_bytes += delta.flat_flush_write_bytes
292            + delta.lru_flush_write_bytes
293            + delta.lru_batch_flush_bytes
294            + delta.direct_tile_write_bytes;
295    }
296}

298impl Drop for BackingObservation {
299    fn drop(&mut self) {
300        let mut state = self
301            .metrics
302            .0
303            .lock()
304            .expect("cube backing metrics lock poisoned");
305        state.live_owned_bytes -= self.ledger.retained_bytes;
306        state.live_cache_bytes -= self.ledger.cache_bytes;
307        state.live_cache_index_bytes -= self.ledger.cache_index_bytes;
308        state.live_storage_bytes -= self.ledger.storage_bytes;
309        state.live_file_handles -= self.ledger.file_handles;
310        state.live_backings -= 1;
311    }
312}

314fn array_file_bytes(directory: &Path) -> Result<usize, String> {
315    ["table.dat", "table.info", "table.f0", "table.f0_TSM0"]
316        .iter()
317        .try_fold(0usize, |sum, name| {
318            let bytes = std::fs::metadata(directory.join(name))
319                .map_err(|error| error.to_string())?
320                .len();
321            let bytes =
322                usize::try_from(bytes).map_err(|_| "cube backing file length exceeds usize")?;
323            checked_sum(&[sum, bytes])
324        })
325}
326
327fn planned_directory(parent: &Path, prefix: &str) -> PathBuf {
328    parent.join(format!("{prefix}{}", "x".repeat(DIRECTORY_RANDOM_CHARS)))
329}
330
331fn checked_bytes(count: usize, bytes: usize) -> Result<usize, String> {
332    count
333        .checked_mul(bytes)
334        .ok_or_else(|| "cube backing byte count overflow".into())
335}
336
337fn checked_sum(values: &[usize]) -> Result<usize, String> {
338    values.iter().try_fold(0usize, |sum, &value| {
339        sum.checked_add(value)
340            .ok_or_else(|| "cube backing byte count overflow".into())
341    })
342}

355    pub(crate) fn new(
356        parent: &Path,

358        retentions: Box<[(usize, Arc<dyn std::fmt::Debug + Send + Sync>)]>,

371    pub(crate) fn owned_metadata_bytes(&self) -> Result<usize, String> {
372        let mut bytes = checked_sum(&[
373            size_of::<Self>(),
374            self.parent.as_os_str().len(),
375            checked_bytes(self.layouts.len(), size_of::<(usize, CubeArrayLayout)>())?,
376            size_of_val(self.retentions.as_ref()),
377        ])?;
378        for (_, layout) in &self.layouts {
379            bytes = checked_sum(&[
380                bytes,
381                layout
382                    .values
383                    .owned_heap_bytes()
384                    .map_err(|error| error.to_string())?,
385            ])?;
386        }
387        Ok(bytes)
388    }

391impl NormalStorageFactory for PagedNormalStorageFactory {
392    fn scalar_sensitivity(&self) -> bool {
393        self.scalar_sensitivity
394    }
395    fn create(
396        &self,
397        allocation_ordinal: usize,
398        scalars: usize,
399    ) -> Result<Box<dyn NormalArrayStorage>, SpectralOperatorError> {
400        let layout = self
401            .layouts
402            .iter()
403            .find_map(|(ordinal, layout)| (*ordinal == allocation_ordinal).then_some(layout))
404            .filter(|layout| scalars > 0 && scalars <= layout.logical_scalars)
405            .ok_or_else(|| {
406                normal_storage_error("normal backing does not match its domain layout")
407            })?;
408        Ok(Box::new(PagedNormalArray::create(
409            &self.parent,
410            layout,
411            scalars,
412            self.retentions
413                .iter()
414                .find(|(ordinal, _)| *ordinal == allocation_ordinal)
415                .ok_or_else(|| normal_storage_error("normal allocation lacks its retention owner"))?
416                .1
417                .clone(),
418            self.metrics.clone(),
419        )?))
420    }
421}

423/// Exact scalar backing with a fixed tile cache and private file lifetime.
424///
425/// Reads allocate one returned `f64` window; writes borrow the caller's slice.
426/// The backend retains one codec tile. Construction borrows pre-encoded standard
427/// table metadata. The caller must admit the returned window and bound its size.
428#[derive(Debug)]
429pub(crate) struct PagedNormalArray {
430    array: Mutex<PagedArray<f64>>,
431    scalars: usize,
432    window_scalars: usize,
433    axes: [usize; 2],
434    // Fields drop in declaration order: close the array before removing its files.
435    #[allow(
436        dead_code,
437        reason = "RAII guard removing the private backing files on drop"
438    )]
439    directory: TempDir,
440    observation: BackingObservation,
441    _retention: Arc<dyn std::fmt::Debug + Send + Sync>,
442}
443
444impl PagedNormalArray {
445    pub(crate) fn create(
446        parent: &Path,
447        layout: &CubeArrayLayout,
448        scalars: usize,
449        retention: Arc<dyn std::fmt::Debug + Send + Sync>,
450        metrics: Arc<CubeBackingMetrics>,
451    ) -> Result<Self, SpectralOperatorError> {
452        if scalars == 0 || scalars > layout.logical_scalars {
453            return Err(normal_storage_error(
454                "normal backing exceeds its planned scalar capacity",
455            ));
456        }
457        let directory = tempfile::Builder::new()
458            .prefix(NORMAL_DIRECTORY_PREFIX)
459            .rand_bytes(DIRECTORY_RANDOM_CHARS)
460            .tempdir_in(parent)
461            .map_err(normal_storage_error)?;
462        let array = PagedArray::create_planned(&layout.values, directory.path().join("values"))
463            .map_err(normal_storage_error)?;
464        let mut ledger = layout.normal_ledger(parent).map_err(normal_storage_error)?;
465        ledger.retained_bytes = checked_sum(&[
466            size_of::<Self>(),
467            directory.path().as_os_str().len(),
468            array
469                .owned_persistent_heap_bytes()
470                .map_err(normal_storage_error)?,
471        ])
472        .map_err(normal_storage_error)?;
473        ledger.storage_bytes =
474            array_file_bytes(&directory.path().join("values")).map_err(normal_storage_error)?;
475        ledger.file_handles = array.owned_file_handles();
476        Ok(Self {
477            array: Mutex::new(array),
478            scalars,
479            window_scalars: layout.window_scalars.min(scalars),
480            axes: [layout.axis0, layout.axis1],
481            directory,
482            observation: metrics.register(ledger),
483            _retention: retention,
484        })
485    }
486
487    #[cfg(test)]
488    pub(crate) fn measurements(&self) -> Result<CubeArrayMeasurements, SpectralOperatorError> {
489        let array = self.array.lock().map_err(normal_storage_error)?;
490        Ok(CubeArrayMeasurements {
491            owned_bytes: checked_sum(&[
492                size_of::<Self>(),
493                self.directory.path().as_os_str().len(),
494                array
495                    .owned_persistent_heap_bytes()
496                    .map_err(normal_storage_error)?,
497            ])
498            .map_err(normal_storage_error)?,
499            storage_bytes: array_file_bytes(&self.directory.path().join("values"))
500                .map_err(normal_storage_error)?,
501            file_handles: array.owned_file_handles(),
502            values_io: array.io_stats(),
503            support_io: None,
504        })
505    }
506
507    fn check_window(&self, start: usize, len: usize) -> Result<(), SpectralOperatorError> {
508        let end = start
509            .checked_add(len)
510            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
511        if end > self.scalars || len > self.window_scalars {
512            return Err(SpectralOperatorError::InvalidSlab);
513        }
514        Ok(())
515    }
516}

518impl NormalArrayStorage for PagedNormalArray {
519    fn len(&self) -> usize {
520        self.scalars
521    }
522
523    fn read(
524        &self,
525        start: usize,
526        len: usize,
527    ) -> Result<std::borrow::Cow<'_, [f64]>, SpectralOperatorError> {
528        self.check_window(start, len)?;
529        if len == 0 {
530            return Ok(std::borrow::Cow::Owned(Vec::new()));
531        }
532        let array = self.array.lock().map_err(normal_storage_error)?;
533        let before = array.io_stats();
534        let mut values = vec![0.0; len];
535        transfer_range(
536            self.axes[0],
537            self.axes[1],
538            start,
539            len,
540            |position, shape, range| {
541                array
542                    .read_slice_into(&position, &shape, &mut values[range])
543                    .map_err(normal_storage_error)
544            },
545        )?;
546        self.observation
547            .record(len, array.io_stats().delta_since(before));
548        Ok(std::borrow::Cow::Owned(values))
549    }
550
551    fn write(&mut self, start: usize, values: &[f64]) -> Result<(), SpectralOperatorError> {
552        self.check_window(start, values.len())?;
553        if values.is_empty() {
554            return Ok(());
555        }
556        let array = self.array.get_mut().map_err(normal_storage_error)?;
557        let before = array.io_stats();
558        let result = transfer_range(
559            self.axes[0],
560            self.axes[1],
561            start,
562            values.len(),
563            |position, shape, range| {
564                array
565                    .write_slice_from(&position, &shape, &values[range])
566                    .map_err(normal_storage_error)
567            },
568        );
569        self.observation
570            .record(values.len(), array.io_stats().delta_since(before));
571        result
572    }
573}

575impl Drop for PagedNormalArray {
576    fn drop(&mut self) {
577        if let Ok(array) = self.array.get_mut() {
578            let before = array.io_stats();
579            let _ = array.flush();
580            self.observation
581                .record(0, array.io_stats().delta_since(before));
582        }
583    }
584}

617    pub(crate) fn ledger(&self) -> Result<CubeArrayLedger, String> {
618        let directory = planned_directory(&self.parent, MODEL_DIRECTORY_PREFIX);
619        let values = PagedArray::<f64>::planned_persistent_heap_bytes(
620            &self.layout.values,
621            &directory.join("values"),
622        )
623        .map_err(|error| error.to_string())?;
624        let support = PagedArray::<bool>::planned_persistent_heap_bytes(
625            &self.support,
626            &directory.join("support"),
627        )
628        .map_err(|error| error.to_string())?;
629        let value_window = checked_bytes(self.layout.window_scalars, size_of::<f64>())?;
630        let support_window = checked_bytes(self.layout.window_scalars, size_of::<bool>())?;
631        let value_scratch = self
632            .layout
633            .values
634            .slice_scratch_bytes()
635            .map_err(|error| error.to_string())?;
636        let support_scratch = self
637            .support
638            .slice_scratch_bytes()
639            .map_err(|error| error.to_string())?;
640        Ok(CubeArrayLedger {
641            retained_bytes: checked_sum(&[
642                size_of::<PagedModelSamples>(),
643                directory.as_os_str().len(),
644                values,
645                support,
646            ])?,
647            cache_bytes: checked_sum(&[
648                self.layout
649                    .values
650                    .cache_payload_bytes()
651                    .map_err(|error| error.to_string())?,
652                self.support
653                    .cache_payload_bytes()
654                    .map_err(|error| error.to_string())?,
655            ])?,
656            cache_index_bytes: checked_sum(&[
657                self.layout
658                    .values
659                    .cache_metadata_bytes()
660                    .map_err(|error| error.to_string())?,
661                self.support
662                    .cache_metadata_bytes()
663                    .map_err(|error| error.to_string())?,
664            ])?,
665            read_scratch_bytes: checked_sum(&[value_window, value_scratch])?.max(checked_sum(&[
666                value_window,
667                support_window,
668                support_scratch,
669            ])?),
670            write_scratch_bytes: checked_sum(&[value_window, value_scratch])?
671                .max(checked_sum(&[support_window, support_scratch])?),
672            flush_scratch_bytes: self
673                .layout
674                .values
675                .flush_scratch_bytes()
676                .map_err(|error| error.to_string())?
677                .max(
678                    self.support
679                        .flush_scratch_bytes()
680                        .map_err(|error| error.to_string())?,
681                ),
682            storage_bytes: checked_sum(&[
683                self.layout
684                    .values
685                    .storage_bytes()
686                    .map_err(|error| error.to_string())?,
687                self.support
688                    .storage_bytes()
689                    .map_err(|error| error.to_string())?,
690            ])?,
691            file_handles: 2,
692        })
693    }

732/// One logical model retains its private files, not its full pixel payload.
733/// Values and support remain independently represented at full precision.
734#[derive(Debug)]
735pub(crate) struct PagedModelSamples {
736    arrays: Mutex<ModelArrays>,
737    samples: usize,
738    window_samples: usize,
739    axes: [usize; 2],
740    // Delete after the array handles have flushed and closed.
741    #[allow(
742        dead_code,
743        reason = "RAII guard removing the private backing files on drop"
744    )]
745    directory: TempDir,
746    observation: BackingObservation,
747    _retention: Arc<dyn std::fmt::Debug + Send + Sync>,
748}
749
750#[derive(Debug)]
751struct ModelArrays {
752    values: PagedArray<f64>,
753    support: PagedArray<bool>,
754}
755
756impl PagedModelSamples {
757    pub(crate) fn create(
758        directory: &Path,
759        layout: &CubeArrayLayout,
760        support_layout: &TiledArrayStorageLayout,
761        retention: Arc<dyn std::fmt::Debug + Send + Sync>,
762        metrics: Arc<CubeBackingMetrics>,
763        mut ledger: CubeArrayLedger,
764    ) -> Result<Self, ModelLifecycleError> {
765        let directory = tempfile::Builder::new()
766            .prefix(MODEL_DIRECTORY_PREFIX)
767            .rand_bytes(DIRECTORY_RANDOM_CHARS)
768            .tempdir_in(directory)
769            .map_err(storage_error)?;
770        let values = PagedArray::create_planned(&layout.values, directory.path().join("values"))
771            .map_err(storage_error)?;
772        let support = PagedArray::create_planned(support_layout, directory.path().join("support"))
773            .map_err(storage_error)?;
774        ledger.retained_bytes = checked_sum(&[
775            size_of::<Self>(),
776            directory.path().as_os_str().len(),
777            values
778                .owned_persistent_heap_bytes()
779                .map_err(storage_error)?,
780            support
781                .owned_persistent_heap_bytes()
782                .map_err(storage_error)?,
783        ])
784        .map_err(storage_error)?;
785        ledger.storage_bytes = checked_sum(&[
786            array_file_bytes(&directory.path().join("values")).map_err(storage_error)?,
787            array_file_bytes(&directory.path().join("support")).map_err(storage_error)?,
788        ])
789        .map_err(storage_error)?;
790        ledger.file_handles = values.owned_file_handles() + support.owned_file_handles();
791        Ok(Self {
792            arrays: Mutex::new(ModelArrays { values, support }),
793            samples: layout.logical_scalars,
794            window_samples: layout.window_scalars,
795            axes: [layout.axis0, layout.axis1],
796            directory,
797            observation: metrics.register(ledger),
798            _retention: retention,
799        })
800    }

831impl ModelSampleStorage for PagedModelSamples {
832    fn sample_count(&self) -> usize {
833        self.samples
834    }
835
836    fn read(
837        &self,
838        start: usize,
839        destination: &mut [ModelSample],
840    ) -> Result<(), ModelLifecycleError> {
841        if start
842            .checked_add(destination.len())
843            .is_none_or(|end| end > self.samples || destination.len() > self.window_samples)
844        {
845            return Err(ModelLifecycleError::CellOutsideShape);
846        }
847        if destination.is_empty() {
848            return Ok(());
849        }
850        let arrays = self.arrays.lock().map_err(storage_error)?;
851        let before_values = arrays.values.io_stats();
852        let before_support = arrays.support.io_stats();
853        let mut values = vec![0.0; destination.len()];
854        let mut support = vec![false; destination.len()];
855        transfer_range(
856            self.axes[0],
857            self.axes[1],
858            start,
859            destination.len(),
860            |position, shape, range| {
861                arrays
862                    .values
863                    .read_slice_into(&position, &shape, &mut values[range.clone()])
864                    .map_err(storage_error)?;
865                arrays
866                    .support
867                    .read_slice_into(&position, &shape, &mut support[range])
868                    .map_err(storage_error)
869            },
870        )?;
871        self.observation.record(
872            destination.len(),
873            arrays.values.io_stats().delta_since(before_values),
874        );
875        self.observation.record(
876            destination.len(),
877            arrays.support.io_stats().delta_since(before_support),
878        );
879        for ((destination, value), supported) in destination
880            .iter_mut()
881            .zip(values.iter().copied())
882            .zip(support.iter().copied())
883        {
884            *destination = if supported {
885                ModelSample::valid(ModelValue::new(value)?)
886            } else if value == 0.0 {
887                ModelSample::invalid()
888            } else {
889                return Err(ModelLifecycleError::InvalidSupportPayload);
890            };
891        }
892        Ok(())
893    }
894
895    fn apply_updates(
896        &self,
897        updates: &[casa_imaging_reconstruction::ModelSampleUpdate],
898        precision: casa_imaging_model::NumericPrecision,
899        bound: f64,
900    ) -> Result<f64, ModelLifecycleError> {
901        let Some(first) = updates.first() else {
902            return Ok(0.0);
903        };
904        let start = first.index();
905        let len = updates.last().expect("nonempty updates").index() - start + 1;
906        if len > self.window_samples || start.checked_add(len).is_none_or(|end| end > self.samples)
907        {
908            return Err(ModelLifecycleError::CellOutsideShape);
909        }
910        let mut arrays = self.arrays.lock().map_err(storage_error)?;
911        let before_values = arrays.values.io_stats();
912        let before_support = arrays.support.io_stats();
913        let mut values = vec![0.0; len];
914        let mut support = vec![false; len];
915        transfer_range(
916            self.axes[0],
917            self.axes[1],
918            start,
919            len,
920            |position, shape, range| {
921                arrays
922                    .values
923                    .read_slice_into(&position, &shape, &mut values[range.clone()])
924                    .map_err(storage_error)?;
925                arrays
926                    .support
927                    .read_slice_into(&position, &shape, &mut support[range])
928                    .map_err(storage_error)
929            },
930        )?;
931        let mut maximum: f64 = 0.0;
932        for update in updates {
933            let index = update.index() - start;
934            let sample = if support[index] {
935                ModelSample::valid(ModelValue::new(values[index])?)
936            } else {
937                ModelSample::invalid()
938            };
939            values[index] = update.apply(sample, precision, bound)?.value().value();
940            maximum = maximum.max(values[index].abs());
941        }
942        transfer_range(
943            self.axes[0],
944            self.axes[1],
945            start,
946            len,
947            |position, shape, range| {
948                arrays
949                    .values
950                    .write_slice_from(&position, &shape, &values[range])
951                    .map_err(storage_error)
952            },
953        )?;
954        self.observation
955            .record(len, arrays.values.io_stats().delta_since(before_values));
956        self.observation
957            .record(len, arrays.support.io_stats().delta_since(before_support));
958        Ok(maximum)
959    }
960
961    fn write(&mut self, start: usize, samples: &[ModelSample]) -> Result<(), ModelLifecycleError> {
962        if start
963            .checked_add(samples.len())
964            .is_none_or(|end| end > self.samples || samples.len() > self.window_samples)
965        {
966            return Err(ModelLifecycleError::CellOutsideShape);
967        }
968        if samples.is_empty() {
969            return Ok(());
970        }
971        let arrays = self.arrays.get_mut().map_err(storage_error)?;
972        let before_values = arrays.values.io_stats();
973        let before_support = arrays.support.io_stats();
974        let values: Vec<f64> = samples
975            .iter()
976            .map(|sample| sample.value().value())
977            .collect();
978        transfer_range(
979            self.axes[0],
980            self.axes[1],
981            start,
982            samples.len(),
983            |position, shape, range| {
984                arrays
985                    .values
986                    .write_slice_from(&position, &shape, &values[range])
987                    .map_err(storage_error)
988            },
989        )?;
990        drop(values);
991        let support: Vec<bool> = samples
992            .iter()
993            .map(|sample| sample.support() == ModelSupport::Valid)
994            .collect();
995        transfer_range(
996            self.axes[0],
997            self.axes[1],
998            start,
999            samples.len(),
1000            |position, shape, range| {
1001                arrays
1002                    .support
1003                    .write_slice_from(&position, &shape, &support[range])
1004                    .map_err(storage_error)
1005            },
1006        )?;
1007        self.observation.record(
1008            samples.len(),
1009            arrays.values.io_stats().delta_since(before_values),
1010        );
1011        self.observation.record(
1012            samples.len(),
1013            arrays.support.io_stats().delta_since(before_support),
1014        );
1015        Ok(())
1016    }
1017}
1018
1019impl Drop for PagedModelSamples {
1020    fn drop(&mut self) {
1021        if let Ok(arrays) = self.arrays.get_mut() {
1022            let before = arrays.values.io_stats();
1023            let _ = arrays.values.flush();
1024            self.observation
1025                .record(0, arrays.values.io_stats().delta_since(before));
1026            let before = arrays.support.io_stats();
1027            let _ = arrays.support.flush();
1028            self.observation
1029                .record(0, arrays.support.io_stats().delta_since(before));
1030        }
1031    }
1032}

1035    ModelLifecycleError::Storage(error.to_string())
1036}
