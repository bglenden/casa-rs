3use std::alloc::Layout;
4use std::sync::{Arc, atomic::AtomicUsize};
5
6use casa_imaging_model::{CompiledProblem, LogicalIdentity, ReferenceDataKind};
7use casa_types::measures::{MeasuresProvider, MeasuresProviderState};
8use thiserror::Error;

10/// One explicitly acquired Measures provider bound to its immutable reference snapshot.
11///
12/// Construction rejects providers that cannot eagerly stabilize and project all
13/// retained cache state. The bound observation owns this capability once and
14/// passes shared provider references inward to its geometry engines.
15#[derive(Debug)]
16pub struct SelectedObservationMeasures {
17    provider: Arc<dyn MeasuresProvider>,
18    provider_state: MeasuresProviderState,
19    retained_bytes: usize,
20}
21
22impl SelectedObservationMeasures {
23    /// Acquire the authoritative logical snapshot owned by one provider.
24    pub fn new(
25        provider: Arc<dyn MeasuresProvider>,
26    ) -> Result<Self, SelectedObservationMeasuresError> {
27        let provider_state = provider
28            .prepare_bounded_state()
29            .map_err(SelectedObservationMeasuresError::ProviderPreparation)?
30            .ok_or(SelectedObservationMeasuresError::UnaccountedProvider)?;
31        let retained_bytes = arc_allocation_bytes(provider.as_ref())
32            .ok_or(SelectedObservationMeasuresError::ByteOverflow)?
33            .checked_add(provider_state.retained_heap_bytes())
34            .ok_or(SelectedObservationMeasuresError::ByteOverflow)?;
35        Ok(Self {
36            provider,
37            provider_state,
38            retained_bytes,
39        })
40    }
41
42    /// Return the provider-owned reference-data identity.
43    #[must_use]
44    pub const fn identity(&self) -> LogicalIdentity {
45        LogicalIdentity::from_sha256(self.provider_state.identity_sha256())
46    }
47
48    pub(crate) const fn retained_bytes(&self) -> usize {
49        self.retained_bytes
50    }
51
52    pub(crate) const fn provider_state(&self) -> MeasuresProviderState {
53        self.provider_state
54    }
55
56    pub(crate) fn provider(&self) -> Arc<dyn MeasuresProvider> {
57        Arc::clone(&self.provider)
58    }
59
60    pub(crate) fn validate_problem(
61        &self,
62        problem: &CompiledProblem,
63    ) -> Result<(), SelectedObservationMeasuresError> {
64        let expected = problem
65            .inputs()
66            .reference_data()
67            .iter()
68            .find_map(|(kind, identity)| {
69                (*kind == ReferenceDataKind::Measures).then_some(*identity)
70            })
71            .ok_or(SelectedObservationMeasuresError::MissingMeasuresReference)?;
72        let actual = self.identity();
73        if actual != expected {
74            return Err(
75                SelectedObservationMeasuresError::ReferenceIdentityMismatch { expected, actual },
76            );
77        }
78        self.verify_state()
79    }
80
81    pub(crate) fn verify_state(&self) -> Result<(), SelectedObservationMeasuresError> {
82        let actual = self
83            .provider
84            .prepare_bounded_state()
85            .map_err(SelectedObservationMeasuresError::ProviderPreparation)?;
86        if actual != Some(self.provider_state) {
87            return Err(SelectedObservationMeasuresError::ProviderStateChanged {
88                expected: self.provider_state,
89                actual,
90            });
91        }
92        Ok(())
93    }
94}
95
96fn arc_allocation_bytes(provider: &dyn MeasuresProvider) -> Option<usize> {
97    let header = Layout::array::<AtomicUsize>(2).ok()?;
98    let (allocation, _) = header.extend(Layout::for_value(provider)).ok()?;
99    Some(allocation.pad_to_align().size())
100}

102/// Failure to bind a Measures provider into bounded Selected Observation access.
103#[derive(Debug, Error)]
104pub enum SelectedObservationMeasuresError {
105    /// The compiled input snapshot omitted the Measures reference used by geometry evaluation.
106    #[error("compiled selected observation has no Measures reference-data identity")]
107    MissingMeasuresReference,
108    /// The acquired provider belongs to a different immutable Measures snapshot.
109    #[error("Measures provider identity {actual} does not match compiled identity {expected}")]
110    ReferenceIdentityMismatch {
111        /// Identity required by the compiled problem.
112        expected: LogicalIdentity,
113        /// Identity supplied by the provider acquisition owner.
114        actual: LogicalIdentity,
115    },
116    /// The provider could not stabilize its bounded cache state.
117    #[error("Measures provider bounded preparation failed: {0}")]
118    ProviderPreparation(String),
119    /// The provider exposes opaque retained state and cannot enter a bounded operation.
120    #[error("Measures provider does not expose bounded retained residency")]
121    UnaccountedProvider,
122    /// Provider scientific identity or cache residency changed after acquisition.
123    #[error("Measures provider state changed from {expected:?} to {actual:?}")]
124    ProviderStateChanged {
125        /// Provider-owned state admitted during acquisition.
126        expected: MeasuresProviderState,
127        /// Provider-owned state observed on the later stability check.
128        actual: Option<MeasuresProviderState>,
129    },
130    /// The provider allocation and cache projection overflowed the host byte domain.
131    #[error("Measures provider retained-residency projection overflowed")]
132    ByteOverflow,
133}

135#[cfg(test)]
136pub(crate) fn test_selected_observation_measures(
137    problem: &CompiledProblem,
138) -> Result<SelectedObservationMeasures, SelectedObservationMeasuresError> {
139    let identity = problem
140        .inputs()
141        .reference_data()
142        .iter()
143        .find_map(|(kind, identity)| (*kind == ReferenceDataKind::Measures).then_some(*identity))
144        .ok_or(SelectedObservationMeasuresError::MissingMeasuresReference)?;
145    SelectedObservationMeasures::new(
146        casa_test_support::deterministic_measures_provider_for_identity(identity.as_bytes()),
147    )
