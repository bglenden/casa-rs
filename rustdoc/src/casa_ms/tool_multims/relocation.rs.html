18/// Plan rewrites for symbolic entries whose lexical target would change.
19/// Do not follow a symbolic directory while walking the physical member.
20pub(super) fn filesystem_links(
21    member: &Member,
22    locations: &Locations,
23    output: &Path,
24) -> MsResult<FilesystemLinks> {
25    let mut directories = vec![member.physical.clone()];
26    let mut links = Vec::new();
27    while let Some(directory) = directories.pop() {
28        for entry in directory.read_dir().map_err(invalid)? {
29            let entry = entry.map_err(invalid)?;
30            let kind = entry.file_type().map_err(invalid)?;
31            let path = entry.path();
32            if kind.is_dir() {
33                directories.push(path);
34            } else if kind.is_symlink() {
35                let payload = path.read_link().map_err(invalid)?;
36                let old = lexical_absolute(&directory.join(&payload));
37                if old.starts_with(output)
38                    || path
39                        .canonicalize()
40                        .is_ok_and(|referent| referent.starts_with(output))
41                {
42                    return Err(invalid(format!(
43                        "MultiMS destination is used by {}",
44                        path.display()
45                    )));
46                }
47                let new = member
48                    .target
49                    .join(path.strip_prefix(&member.physical).unwrap());
50                let parent = new.parent().expect("staged link parent");
51                let referent = relocated_path(&member.physical, &member.target, &old, locations);
52                if lexical_absolute(&parent.join(&payload)) != referent {
53                    let stage = member.target.parent().expect("staged member parent");
54                    let payload = if referent.starts_with(stage) {
55                        relative_file_path(parent, &referent)
56                    } else {
57                        referent
58                    };
59                    links.push((new, payload));
60                }
61            }
62        }
63    }
64    Ok(links)
65}

67fn relative_file_path(mut from: &Path, target: &Path) -> PathBuf {
68    let mut path = PathBuf::new();
69    while !target.starts_with(from) {
70        path.push("..");
71        from = from.parent().expect("paths share the staging directory");
72    }
73    let suffix = target.strip_prefix(from).unwrap();
74    if path.as_os_str().is_empty() && suffix.as_os_str().is_empty() {
75        path.push(".");
76    } else {
77        path.push(suffix);
78    }
79    path
80}

82pub(super) fn apply_filesystem_links(links: FilesystemLinks) -> MsResult<()> {
83    for (path, target) in links {
84        #[cfg(unix)]
85        {
86            fs::remove_file(&path).map_err(invalid)?;
87            std::os::unix::fs::symlink(target, path).map_err(invalid)?;
88        }
89        #[cfg(not(unix))]
90        {
91            let _ = (path, target);
92            return Err(invalid("symbolic-link MultiMS input is unsupported"));
93        }
94    }
95    Ok(())
96}

98/// Return the top-level member containing the base when the entire
99/// reference chain moves into the private MultiMS stage.
100pub(super) fn owned_reference_base(
101    path: &Path,
102    locations: &Locations,
103) -> MsResult<Option<PathBuf>> {
104    let mut path = path.to_path_buf();
105    for _ in 0..64 {
106        match Table::immediate_part_names(&path, &path)? {
107            TablePartNames::Reference(parent) => {
108                path = parent.canonicalize().map_err(invalid)?;
109                if !locations.keys().any(|member| path.starts_with(member)) {
110                    return Ok(None);
111                }
112            }
113            TablePartNames::Plain(_) => {
114                return Ok(locations
115                    .keys()
116                    .find(|member| path.starts_with(member))
117                    .cloned());
118            }
119            TablePartNames::Concatenation { .. } => return Ok(None),
120        }
121    }
122    Err(invalid("member reference chain is too deep"))
123}

125pub(super) enum MemberEdit {
126    None,
127    Parts(Vec<PathBuf>),
128    Keywords {
129        keywords: RecordValue,
130        columns: Vec<(String, RecordValue)>,
131    },
132}
133
134impl MemberEdit {
135    pub(super) fn prepare(member: &Member, locations: &Locations) -> MsResult<Self> {
136        let source = &member.physical;
137        let target = &member.target;
138        match Table::immediate_part_names(source, source)? {
139            TablePartNames::Plain(_) => {
140                let table = Table::open(TableOptions::new(source))?;
141                let mut keywords = table.keywords().clone();
142                let changed = relocate_links(&mut keywords, source, target, locations);
143                let mut columns = Vec::new();
144                for column in table
145                    .schema()
146                    .into_iter()
147                    .flat_map(|schema| schema.columns())
148                {
149                    let name = column.name();
150                    if let Some(keys) = table.column_keywords(name) {
151                        let mut keys = keys.clone();
152                        if relocate_links(&mut keys, source, target, locations) {
153                            columns.push((name.to_owned(), keys));
154                        }
155                    }
156                }
157                Ok(if changed || !columns.is_empty() {
158                    Self::Keywords { keywords, columns }
159                } else {
160                    Self::None
161                })
162            }
163            original => {
164                let before = parts(original);
165                let after = parts(Table::immediate_part_names(source, target)?);
166                let mapped = before
167                    .iter()
168                    .map(|part| relocated_path(source, target, part, locations))
169                    .collect::<Vec<_>>();
170                Ok(if mapped == after {
171                    Self::None
172                } else {
173                    Self::Parts(mapped)
174                })
175            }
176        }
177    }
178
179    pub(super) fn is_empty(&self) -> bool {
180        matches!(self, Self::None)
181    }
182
183    pub(super) fn apply(self, target: &Path) -> MsResult<()> {
184        match self {
185            Self::None => {}
186            Self::Parts(parts) => Table::set_staged_table_parts(target, &parts)?,
187            Self::Keywords { keywords, columns } => {
188                let mut table = Table::open(TableOptions::new(target))?;
189                *table.keywords_mut() = keywords;
190                for (column, keys) in columns {
191                    table.set_column_keywords(column, keys);
192                }
193                table.save_metadata_only(TableOptions::new(target))?;
194            }
195        }
196        Ok(())
197    }
198}

200fn parts(layout: TablePartNames) -> Vec<PathBuf> {
201    match layout {
202        TablePartNames::Reference(path) => vec![path],
203        TablePartNames::Concatenation { parts, .. } => parts,
204        TablePartNames::Plain(_) => Vec::new(),
205    }
206}

208fn relocate_links(
209    record: &mut RecordValue,
210    source: &Path,
211    target: &Path,
212    locations: &Locations,
213) -> bool {
214    let mut changed = false;
215    for field in record.fields_mut() {
216        match &mut field.value {
217            Value::Record(nested) => changed |= relocate_links(nested, source, target, locations),
218            Value::TableRef(link) => {
219                let old = add_directory(link, source);
220                let relocated = relocated_path(source, target, &old, locations);
221                if lexical_absolute(&add_directory(link, target)) != relocated {
222                    *link = strip_directory(&relocated, target);
223                    changed = true;
224                }
225            }
226            _ => {}
227        }
228    }
229    changed
230}

232fn relocated_path(source: &Path, target: &Path, path: &Path, locations: &Locations) -> PathBuf {
233    let path = lexical_absolute(path);
234    if let Ok(relative) = path.strip_prefix(source) {
235        return target.join(relative);
236    }
237    locations
238        .iter()
239        .find_map(|(source, target)| {
240            path.strip_prefix(source)
241                .ok()
242                .map(|relative| target.join(relative))
243        })
244        .unwrap_or(path)
245}

247pub(super) fn stage_alias(member: &Member, locations: &Locations) -> MsResult<()> {
248    let referent = locations.get(&member.physical).unwrap_or(&member.physical);
249    let parent = member.target.parent().expect("member staging directory");
250    let referent = referent.strip_prefix(parent).unwrap_or(referent);
251    #[cfg(unix)]
252    std::os::unix::fs::symlink(referent, &member.target).map_err(invalid)?;
253    #[cfg(not(unix))]
254    {
255        let _ = referent;
256        return Err(invalid("symbolic-link MultiMS input is unsupported"));
257    }
258    #[cfg(unix)]
259    Ok(())
260}

262pub(super) fn materialize_member(member: &Member, locations: &Locations) -> MsResult<()> {
263    // Reference-table keywords are written through to their parent. Give a
264    // staged copy its own data and keywords before replacing any subtable.
265    let mut table = Table::open(TableOptions::new(&member.physical))?;
266    let columns = table
267        .schema()
268        .into_iter()
269        .flat_map(|schema| schema.columns())
270        .map(|column| column.name().to_owned())
271        .collect::<Vec<_>>();
272    relocate_links(
273        table.keywords_mut(),
274        &member.physical,
275        &member.target,
276        locations,
277    );
278    for column in columns {
279        if let Some(keys) = table.column_keywords(&column) {
280            let mut keys = keys.clone();
281            if relocate_links(&mut keys, &member.physical, &member.target, locations) {
282                table.set_column_keywords(column, keys);
283            }
284        }
285    }
286    table.deep_copy(TableOptions::new(&member.target))?;
287    Ok(())
288}

290pub(super) fn privatize_descriptor(target: &Path) -> MsResult<()> {
291    for name in ["table.dat", "table.info"] {
292        let file = target.join(name);
293        if file
294            .symlink_metadata()
295            .is_ok_and(|entry| entry.is_symlink())
296        {
297            let replacement = tempfile::NamedTempFile::new_in(target).map_err(invalid)?;
298            fs::copy(&file, replacement.path()).map_err(invalid)?;
299            replacement.persist(file).map_err(invalid)?;
300        }
301    }
302    Ok(())
