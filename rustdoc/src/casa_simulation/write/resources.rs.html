2//! What `close` holds while it writes MAIN, admitted against the session's
3//! resource policy before anything is written (plan §4: the runtime's
4//! budget, not a second one).
5//!
6//! In order:
7//! 1. **The writer.** The MeasurementSet write plan is sized from half of
8//!    what the policy leaves free, and its resident bytes are admitted with
9//!    the subtable columns its save generates (POINTING's runs, a few
10//!    hundred bytes a scan). Its batches are tile bands of about
11//!    [`casa_ms::TARGET_BATCH_BYTES`], so the budget is a ceiling that only
12//!    a small budget reaches.
13//! 2. **The pipeline.** The batch being filled, the geometry of the
14//!    integrations it spans, each worker's scratch and, when the session
15//!    predicts, the predictor, its block of native rows and (on Metal) the
16//!    device copies of the model grids. A block is the whole batch when
17//!    that fits with the policy's streaming workers; otherwise blocks are
18//!    small, and as many workers as fit what is then free are admitted
19//!    (#737's rule: workers shrink first). With one worker that does not
20//!    fit, `close` is refused.
21//!
22//! The model grids themselves were admitted by `predict`, and are held.
23//! A refusal is a typed [`Admission`] carried by the session error.
24

40/// The writer's phase, as an admission refusal names it.
41pub(super) const WRITER_PHASE: &str = "the simulator's MeasurementSet writer";
42
43/// The pipeline's phase, as an admission refusal names it.
44pub(super) const PIPELINE_PHASE: &str = "the simulator's geometry, prediction and noise";
45
46/// Batches the writer holds at once: the one it writes and one queued.
47const WRITER_LIVE_BATCHES: usize = 2;
48
49/// A writer budget larger than any plan's needs, to tell a budget too small
50/// from a plan that cannot be made.
51const UNBOUNDED_WRITER_BYTES: usize = 1 << 50;
52
53/// Scratch of one worker's integration, per antenna: the antennas' uvw,
54/// their shadow flags and elevations, and for `tsys-manual` noise their
55/// airmasses (and elevations at a moved pointing's phase centre), with room
56/// for the measures frames.
57const ANTENNA_SCRATCH_BYTES: u64 = 64;

59/// The MAIN table `close` writes.
60#[derive(Debug, Clone, Copy)]
61pub(super) struct MainShape<'a> {
62    pub(super) rows: usize,
63    pub(super) correlations: usize,
64    pub(super) channels: usize,
65    pub(super) telescope: &'a str,
66    /// Rows of the session's largest integration.
67    pub(super) largest_integration: usize,
68    /// Antennas of the session's largest scan.
69    pub(super) antennas: usize,
70    /// Heap of the subtable columns the save generates (POINTING's), which
71    /// grows with the scans.
72    pub(super) generated_bytes: u64,
73    /// What a batch's `tsys-atm` noise holds beside the cells.
74    pub(super) noise_scales: NoiseScales,
75}

77/// What a batch's `tsys-atm` noise and `settrop` terms hold beside the
78/// cells: their antenna scales and phasors ([`BatchScales`]), and each
79/// worker's σ of a row ([`NoiseScratch`]).
80#[derive(Debug, Clone, Copy, Default)]
81pub(super) struct NoiseScales {
82    /// The plan's `tsys-atm` terms.
83    terms: usize,
84    /// The plan's `settrop` terms.
85    tropospheric_terms: usize,
86    /// Channels of the cells.
87    channels: usize,
88    /// Rows of the smallest integration, and antennas of the largest, when
89    /// the rows come in whole integrations ([`BatchScales::most_entries`]).
90    integrations: Option<(usize, usize)>,
91}
92
93impl NoiseScales {
94    /// What `plan`'s corruptions hold, for cells of `channels` channels and
95    /// the rows' `integrations`. Every `corrupt` covers scan 0, so its
96    /// corruptions are all the plan's.
97    pub(super) fn of(
98        plan: &DataPlan,
99        channels: usize,
100        integrations: Option<(usize, usize)>,
101    ) -> Self {
102        let mut models: Vec<*const TsysAtm> = plan
103            .noise_for_scan(0)
104            .iter()
105            .filter_map(|application| application.tsys_atm().map(std::ptr::from_ref))
106            .collect();
107        models.sort_unstable();
108        models.dedup();
109        let mut tropospheric: Vec<*const JonesTerm> = plan
110            .corruptions_for_scan(0)
111            .flat_map(|(_, jones)| jones)
112            .filter(|term| term.tropospheric_phase().is_some())
113            .map(|term| std::ptr::from_ref::<JonesTerm>(term))
114            .collect();
115        tropospheric.sort_unstable();
116        tropospheric.dedup();
117        Self {
118            terms: models.len(),
119            tropospheric_terms: tropospheric.len(),
120            channels,
121            integrations,
122        }
123    }
124
125    /// The most antenna-scale and phasor entries a batch of `rows` rows
126    /// needs: the room its table reserves.
127    pub(super) fn entries(&self, rows: usize) -> (usize, usize) {
128        (
129            BatchScales::most_entries(self.terms, rows, self.integrations),
130            BatchScales::most_entries(self.tropospheric_terms, rows, self.integrations),
131        )
132    }
133
134    /// The most a batch of `rows` rows holds with `workers` workers.
135    pub(super) fn bytes(&self, rows: usize, workers: usize) -> u64 {
136        let scratch = if self.terms == 0 {
137            0
138        } else {
139            workers as u64 * NoiseScratch::bytes(self.channels)
140        };
141        BatchScales::bytes(self.entries(rows), self.channels) + scratch
142    }
143}

145/// What `close` was admitted: the write plan, the workers that fit, and the
146/// memory, held until the write ends.
147#[derive(Debug)]
148pub(super) struct MainAdmission {
149    pub(super) plan: MeasurementSetWritePlan,
150    pub(super) cores: CoreAllocation,
151    /// The policy's allocation, before admission shrank it.
152    pub(super) policy_cores: CoreAllocation,
153    /// Rows of a prediction block: the write batch's, or fewer when the
154    /// budget is small.
155    pub(super) block_rows: usize,
156    pub(super) reservation: Reservation,
157}
158
159/// Admit `close`'s writer and pipeline for `shape` within `resources`, the
160/// pipeline predicting with `prediction` when the session predicts.
161///
162/// # Errors
163///
164/// [`SimulatorError::admission`] when the writer, or the pipeline with one
165/// worker, does not fit what the policy leaves free; a raised error when the
166/// MAIN write cannot be planned at all.
167pub(super) fn admit_main_write(
168    resources: &SessionResources,
169    shape: &MainShape<'_>,
170    prediction: Option<&PredictionStage<'_>>,
171) -> SimulatorResult<MainAdmission> {
172    let (host, policy) = (resources.host(), resources.policy());
173    let free = free_memory(host, policy);
174    let plan = writer_plan(shape, to_usize(free / 2)).map_err(|error| {
175        let required =
176            minimum_writer_bytes(shape).map_or(u64::MAX, |bytes| (bytes as u64).saturating_mul(2));
177        refusal_or(error, WRITER_PHASE, required, free)
178    })?;
179    let mut reservation = admit(
180        host,
181        policy,
182        &Demand {
183            phase: WRITER_PHASE,
184            memory: plan.maximum_resident_bytes() as u64 + shape.generated_bytes,
185        },
186    )
187    .map_err(SimulatorError::admission)?;
188    let policy_cores = resources
189        .close_cores(prediction.is_some_and(|stage| stage.backend() == BackendChoice::Metal));
190    let free = free_memory(host, policy);
191    let batch_rows = plan.batch_rows();
192    let demand = |cores: CoreAllocation, block_rows: usize| {
193        pipeline_bytes(shape, batch_rows, cores)
194            + prediction.map_or(0, |stage| stage.bytes(block_rows, cores.streaming_workers))
195    };
196    // A batch predicted as one block when that fits with the policy's
197    // workers; else small blocks, and the workers shrink first (#737).
198    let (cores, block_rows) = if demand(policy_cores, batch_rows) <= free {
199        (policy_cores, batch_rows)
200    } else {
201        let small = prediction.map_or(batch_rows, |stage| stage.small_block_rows(batch_rows));
202        let cores = (1..=policy_cores.streaming_workers)
203            .rev()
204            .map(|workers| policy_cores.at_most(workers))
205            .find(|cores| demand(*cores, small) <= free)
206            .unwrap_or_else(|| policy_cores.at_most(1));
207        (cores, small)
208    };
209    let pipeline = admit(
210        host,
211        policy,
212        &Demand {
213            phase: PIPELINE_PHASE,
214            memory: demand(cores, block_rows),
215        },
216    )
217    .map_err(SimulatorError::admission)?;
218    reservation.join(pipeline);
219    Ok(MainAdmission {
220        plan,
221        cores,
222        policy_cores,
223        block_rows,
224        reservation,
225    })
226}

228/// The MAIN write plan with `available_bytes` for the writer.
229pub(super) fn writer_plan(
230    shape: &MainShape<'_>,
231    available_bytes: usize,
232) -> Result<MeasurementSetWritePlan, MeasurementSetWriteError> {
233    MeasurementSetWritePlan::visibility_creation(
234        shape.rows,
235        shape.correlations,
236        shape.channels,
237        shape.telescope,
238        MeasurementSetWriteResources {
239            available_bytes,
240            maximum_live_batches: WRITER_LIVE_BATCHES,
241            tiled_column_buffer_bytes: STREAMING_TILED_COLUMN_BUFFER_BYTES,
242        },
243    )
244}

246/// The fewest writer bytes that plan `shape`'s MAIN write; `None` when no
247/// budget does.
248fn minimum_writer_bytes(shape: &MainShape<'_>) -> Option<usize> {
249    writer_plan(shape, UNBOUNDED_WRITER_BYTES).ok()?;
250    // Whether a budget plans is monotone in the budget.
251    let (mut refused, mut planned) = (0, UNBOUNDED_WRITER_BYTES);
252    while planned - refused > 1 {
253        let middle = refused + (planned - refused) / 2;
254        if writer_plan(shape, middle).is_ok() {
255            planned = middle;
256        } else {
257            refused = middle;
258        }
259    }
260    Some(planned)
261}

263/// A plan refused for its budget as an [`Admission`] of `required` bytes
264/// against `free`, and any other planning failure as raised.
265fn refusal_or(
266    error: MeasurementSetWriteError,
267    phase: &'static str,
268    required: u64,
269    free: u64,
270) -> SimulatorError {
271    if required == u64::MAX {
272        return SimulatorError::raised(format!("plan the MAIN write: {error}"));
273    }
274    SimulatorError::admission(Admission {
275        phase,
276        required,
277        available: free,
278    })
279}

281/// Bytes the pipeline holds beside the writer, with `cores`' streaming
282/// workers: the batch being filled, the geometry of the integrations it
283/// spans (with the unwritten rows of the one before), the list of those
284/// integrations, one integration's scalar runs, each worker's scratch, and
285/// the batch's `tsys-atm` antenna scales.
286pub(super) fn pipeline_bytes(
287    shape: &MainShape<'_>,
288    batch_rows: usize,
289    cores: CoreAllocation,
290) -> u64 {
291    let samples = (shape.correlations * shape.channels) as u64;
292    let rows = batch_rows as u64;
293    let largest = shape.largest_integration as u64;
294    let size = |bytes: usize| bytes as u64;
295    // DATA, FLAG_ROW, UVW, WEIGHT and SIGMA.
296    let row_bytes = samples * size(size_of::<Complex32>()) + 1 + 24 + 4 + 4;
297    let batch = rows * row_bytes;
298    let geometry = (rows + 2 * largest) * size(size_of::<RowGeometry>());
299    let integrations = (rows + 2) * size(size_of::<Integration>());
300    let scalars = largest * (4 + 4 + 1);
301    let workers = cores.streaming_workers as u64 * shape.antennas as u64 * ANTENNA_SCRATCH_BYTES;
302    let noise = shape
303        .noise_scales
304        .bytes(batch_rows, cores.streaming_workers);
305    batch + geometry + integrations + scalars + workers + noise
306}
307
308fn to_usize(bytes: u64) -> usize {
309    usize::try_from(bytes).unwrap_or(usize::MAX)
310}

312#[cfg(test)]
313mod tests {
314    use casa_imaging_runtime::{HostResources, ResourcePolicy};
315
316    use super::*;
317    use crate::session::BackendRequest;
318
319    fn shape() -> MainShape<'static> {
320        MainShape {
321            rows: 100_000,
322            correlations: 2,
323            channels: 64,
324            telescope: "ALMA",
325            largest_integration: 1_000,
326            antennas: 44,
327            generated_bytes: 0,
328            noise_scales: NoiseScales::default(),
329        }
330    }
331
332    fn resources(memory: u64, workers: usize) -> SessionResources {
333        SessionResources::pinned(
334            HostResources {
335                threads: 8,
336                performance_cores: 8,
337                available_memory: 8 << 30,
338                metal: false,
339            },
340            ResourcePolicy::Explicit { workers, memory },
341            BackendRequest::Cpu,
342        )
343        .expect("resources")
344    }
345
346    #[test]
347    fn the_writer_takes_half_the_budget_and_the_pipeline_fits_the_rest() {
348        let admitted = admit_main_write(&resources(1 << 30, 5), &shape(), None).expect("fits");
349        assert_eq!(admitted.cores.streaming_workers, 4);
350        let writer = admitted.plan.maximum_resident_bytes() as u64;
351        assert!(writer <= 1 << 29);
352        assert_eq!(
353            admitted.reservation.memory(),
354            writer + pipeline_bytes(&shape(), admitted.plan.batch_rows(), admitted.cores)
355        );
356    }
357
358    #[test]
359    fn a_budget_too_small_for_the_writer_is_a_typed_refusal() {
360        let error =
361            admit_main_write(&resources(1 << 20, 4), &shape(), None).expect_err("too small");
362        let refusal = error.admission_refused().expect("an admission refusal");
363        assert_eq!(refusal.phase, WRITER_PHASE);
364        assert_eq!(refusal.available, 1 << 20);
365        let minimum = minimum_writer_bytes(&shape()).expect("plannable") as u64;
366        assert_eq!(refusal.required, 2 * minimum);
367        assert!(writer_plan(&shape(), to_usize(minimum)).is_ok());
368        assert!(writer_plan(&shape(), to_usize(minimum - 1)).is_err());
369    }
370
371    /// Admission state is the process's, so these laws assume no other
372    /// test holds memory at once (nextest runs each test in its own
373    /// process).
374    #[test]
375    fn workers_shrink_before_the_pipeline_is_refused() {
376        // A budget of twice the writer's minimum, and arrays wide enough
377        // that each worker's scratch is a third of what the writer leaves:
378        // two workers fit, four do not.
379        let writer = minimum_writer_bytes(&shape()).expect("plannable");
380        let plan = writer_plan(&shape(), writer).expect("plan");
381        let budget = 2 * writer as u64;
382        let left = budget - plan.maximum_resident_bytes() as u64;
383        let shape = MainShape {
384            antennas: to_usize(left / 3 / ANTENNA_SCRATCH_BYTES),
385            ..shape()
386        };
387        let four = CoreAllocation {
388            workers: 5,
389            streaming_workers: 4,
390        };
391        let two = pipeline_bytes(&shape, plan.batch_rows(), four.at_most(2));
392        assert!(two <= left, "two workers fit: {two} of {left}");
393        assert!(pipeline_bytes(&shape, plan.batch_rows(), four.at_most(3)) > left);
394        let admitted = admit_main_write(&resources(budget, 5), &shape, None).expect("two fit");
395        assert_eq!(admitted.policy_cores, four);
396        assert_eq!(admitted.cores, four.at_most(2));
397        drop(admitted);
398        let error = admit_main_write(
399            &resources(budget, 5),
400            &MainShape {
401                antennas: shape.antennas * 4,
402                ..shape
403            },
404            None,
405        )
406        .expect_err("one worker does not fit");
407        let refusal = error.admission_refused().expect("an admission refusal");
408        assert_eq!(refusal.phase, PIPELINE_PHASE);
409    }
