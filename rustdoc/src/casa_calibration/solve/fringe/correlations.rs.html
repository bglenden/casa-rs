4use std::collections::{BTreeMap, HashMap};
5
6use super::{FringeBaseline, FringeCorrelation, FringeSample, SolveAccumulator, Window};

8pub(super) fn has_stokes_data(group: &SolveAccumulator, windows: &BTreeMap<i32, Window>) -> bool {
9    let mut first = HashMap::new();
10    for spectrum in group
11        .delay_spectra()
12        .filter(|spectrum| spectrum.parallel_hand)
13    {
14        if windows[&spectrum.buffer.0].single_hand {
15            if spectrum.channels().any(|(_, weight)| weight > 0.) {
16                return true;
17            }
18            continue;
19        }
20        let key = (spectrum.buffer, spectrum.antennas);
21        if spectrum.receptor == 0 {
22            first.insert(key, spectrum);
23        } else if first.get(&key).is_some_and(|other| {
24            spectrum
25                .channels()
26                .zip(other.channels())
27                .any(|((_, a), (_, b))| a > 0. && b > 0.)
28        }) {
29            return true;
30        }
31    }
32    false
33}

35pub(super) fn combine(
36    hands: [Vec<FringeBaseline>; 2],
37    combination: FringeCorrelation,
38) -> Vec<FringeBaseline> {
39    let mut by_baseline = BTreeMap::<_, BTreeMap<_, [Option<FringeSample>; 2]>>::new();
40    for (hand, baselines) in hands.into_iter().enumerate() {
41        for baseline in baselines {
42            let by_channel = by_baseline.entry(baseline.antennas).or_default();
43            for sample in baseline.samples {
44                by_channel.entry(sample.buffer_channel_key()).or_default()[hand] = Some(sample);
45            }
46        }
47    }
48    by_baseline
49        .into_iter()
50        .filter_map(|(antennas, channels)| {
51            let samples = channels
52                .into_values()
53                .filter_map(|hands| combine_sample(hands, combination))
54                .collect::<Vec<_>>();
55            samples
56                .iter()
57                .any(|sample| sample.weight > 0.)
58                .then_some(FringeBaseline { antennas, samples })
59        })
60        .collect()
61}

63fn combine_sample(
64    hands: [Option<FringeSample>; 2],
65    combination: FringeCorrelation,
66) -> Option<FringeSample> {
67    let empty = hands[0].or(hands[1]).map(|sample| FringeSample {
68        weight: 0.,
69        visibility: Default::default(),
70        ..sample
71    });
72    match hands.map(|hand| hand.filter(|sample| sample.weight > 0.)) {
73        [Some(left), Some(right)] => {
74            let (visibility, weight) = match combination {
75                FringeCorrelation::Stokes => (
76                    (left.visibility + right.visibility) * 0.5,
77                    4. / (left.weight.recip() + right.weight.recip()),
78                ),
79                FringeCorrelation::Parallel => {
80                    let total = left.weight + right.weight;
81                    (
82                        (left.visibility * left.weight + right.visibility * right.weight) / total,
83                        total,
84                    )
85                }
86                FringeCorrelation::Separate => unreachable!("combining mode"),
87            };
88            Some(FringeSample {
89                visibility,
90                weight,
91                ..left
92            })
93        }
94        [Some(value), None] | [None, Some(value)] if combination == FringeCorrelation::Parallel => {
95            Some(value)
96        }
97        _ => empty,
98    }
99}
