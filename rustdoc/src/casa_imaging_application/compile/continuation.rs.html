2//! What a run continuing on an earlier run's products reads of them,
3//! checked when the request compiles (#944, #980): every product on the
4//! requested grid, the model as the start model, a CLEAN mask to reuse,
5//! and the PSF, sums of weights and residual a run reads rather than
6//! forms ([`crate::restart`]). Compile reads headers and the mask; the
7//! model and the normal-state images are read when the run uses them.
8
9use std::path::PathBuf;
10
11use casa_coordinates::CoordinateSystem;
12use casa_images::RegridTemplate;
13use casa_imaging_products::CasaImageWriter;
14use casa_imaging_runtime::{Demand, Reservation, admit};
15
16use super::domains::{DomainMask, PreparedImageDomain};
17use super::{PrepareError, StartModel};
18use crate::restart::{self, ExistingProducts, Family};
19use crate::{ImagingRequest, RunContext, UseMask};

21/// What a run reads of the products at its `imagename`.
22pub(crate) struct Continuation {
23    /// The earlier model, on the image's own grid.
24    pub(crate) start_model: Option<StartModel>,
25    /// Existing mosaic PSF coordinates when CASA grids into that image.
26    pub(crate) psf_coordinates: Option<CoordinateSystem>,
27    /// Charges for reused mask and image metadata, held through publication.
28    pub(crate) mask_resident: Reservation,
29    /// Earlier products or their metadata, including images read in place of a pass.
30    pub(crate) existing: Option<ExistingProducts>,
31}
32
33/// Bind the coordinates of a re-formed PSF and the model for the first pass.
34pub(super) fn prepare(
35    request: &ImagingRequest,
36    main: &mut PreparedImageDomain,
37    (channels, reconstruction_planes): (usize, usize),
38    context: &RunContext,
39    writer: &mut CasaImageWriter,
40) -> Result<Continuation, PrepareError> {
41    let mut continuation = read(request, main, channels, context)?;
42    if let Some(coordinates) = continuation.psf_coordinates.take() {
43        writer.set_main_psf_coordinates(coordinates);
44    }
45    if request.has_start_model() {
46        let terms = match request.deconvolver {
47            crate::Deconvolver::Mtmfs => request.nterms,
48            _ => 1,
49        };
50        continuation.start_model = Some(StartModel::prepare(
51            &request.startmodel,
52            main,
53            request.stokes.len(),
54            (reconstruction_planes, terms),
55        )?);
56    }
57    Ok(continuation)
58}

60/// Check every product at the main domain's prefix on its grid, and take
61/// what the run continues on: the model, the CLEAN mask (into `main`'s
62/// mask) when the run cleans with no new mask, and the PSF and residual
63/// that `calcpsf=False` and `calcres=False` read. `channels` are the
64/// image's output channels. A request that does not continue
65/// (`tclean_restart` unset, lower-level execution) reads nothing.
66fn read(
67    request: &ImagingRequest,
68    main: &mut PreparedImageDomain,
69    channels: usize,
70    context: &RunContext,
71) -> Result<Continuation, PrepareError> {
72    let mut continuation = Continuation {
73        start_model: None,
74        psf_coordinates: None,
75        mask_resident: Reservation::none(),
76        existing: None,
77    };
78    if !request.tclean_restart {
79        restart::check_inputs(request, false)?;
80        return Ok(continuation);
81    }
82    let family = Family::of(request);
83    let [nx, ny] = main.image_size;
84    let polarizations = request.stokes.len();
85    let planes = match family {
86        Family::Single => channels,
87        Family::Taylor { .. } => 1,
88    };
89    let template = RegridTemplate {
90        coordinates: main.coordinates.clone(),
91        shape: vec![nx, ny, polarizations, planes],
92    };
93    let saved_mosaic_psf = request.supports_separate_psf().then_some(main.direction);
94    for path in restart::products(&request.imagename)? {
95        let suffix = restart::suffix(&request.imagename, &path);
96        let psf_coordinates = restart::check_grid(&path, &suffix, &template, saved_mosaic_psf)?;
97        if request.calcpsf
98            && !request.uses_separate_psf()
99            && let Some(coordinates) = psf_coordinates
100        {
101            // Without a separate PSF request CASA makes new samples in
102            // the existing image store, keeping its coordinates and beam.
103            let bytes = coordinates
104                .to_record()
105                .retained_heap_bytes()
106                .and_then(|bytes| u64::try_from(bytes).ok())
107                .unwrap_or(u64::MAX);
108            continuation.mask_resident.join(admit(
109                &context.host,
110                &context.policy,
111                &Demand {
112                    phase: "saved mosaic PSF coordinates",
113                    memory: bytes,
114                },
115            )?);
116            continuation.psf_coordinates = Some(coordinates);
117        }
118        // CASA's setupMask reuses an existing mask only when no mask is
119        // given; `restart::check` refused the other cleaning cases.
120        if suffix == "mask"
121            && request.niter > 0
122            && request.usemask == UseMask::User
123            && main.mask == DomainMask::FullPlane
124        {
125            let input = restart::MaskInput::open(&path)?;
126            let mut charge = admit(
127                &context.host,
128                &context.policy,
129                &Demand {
130                    phase: "restart CLEAN mask",
131                    memory: input.reading_bytes(),
132                },
133            )?;
134            let (planes, support) = input.read()?;
135            charge.retain(restart::MaskInput::retained_bytes(planes, nx * ny));
136            main.mask = DomainMask::PixelSupport {
137                planes,
138                support: support.into(),
139            };
140            continuation.mask_resident.join(charge);
141        }
142    }
143    let models = family
144        .names("model")
145        .into_iter()
146        .map(|name| {
147            let path = restart::product_path(&request.imagename, &name);
148            path.exists().then_some(path)
149        })
150        .collect::<Box<[Option<PathBuf>]>>();
151    if models.iter().any(Option::is_some) {
152        continuation.start_model = Some(StartModel::restart(
153            models,
154            RegridTemplate {
155                shape: vec![
156                    nx,
157                    ny,
158                    polarizations,
159                    match family {
160                        Family::Single => channels,
161                        Family::Taylor { terms } => terms,
162                    },
163                ],
164                ..template
165            },
166        ));
167    }
168    let residual = restart::reads_residual(request);
169    // CASA retains an existing residual's beam even when it forms new pixels.
170    let residual_name = family
171        .names("residual")
172        .into_iter()
173        .next()
174        .expect("a residual name");
175    let psf_name = family.names("psf").into_iter().next().expect("a PSF name");
176    // CubeMajorCycleAlgorithm forms a temporary PSF without the old image
177    // metadata, whereas MFS makes its PSF in the existing image store.
178    let psf_beam = (!request.calcpsf || request.specmode == crate::SpecMode::Mfs)
179        && !(request.calcpsf && request.uses_separate_psf())
180        && restart::product_path(&request.imagename, &psf_name).exists();
181    let mask_beam = request.usemask == UseMask::AutoMultithresh
182        && request.niter > 0
183        && restart::product_path(&request.imagename, &residual_name).exists();
184    if !request.calcpsf || residual || mask_beam || psf_beam {
185        let mut existing = ExistingProducts::new(
186            &request.imagename,
187            family,
188            ([nx, ny], polarizations, channels),
189            (!request.calcpsf, residual),
190            restart::reuses_direction_dependent_psf(request),
191        )?;
192        if psf_beam {
193            let mut charge = admit(
194                &context.host,
195                &context.policy,
196                &Demand {
197                    phase: "restart PSF beam metadata",
198                    memory: ExistingProducts::beam_reading_bytes(planes, polarizations),
199                },
200            )?;
201            existing.read_psf_beams(channels, main.direction.increment_rad()[0].abs())?;
202            charge.retain(existing.beam_resident_bytes());
203            continuation.mask_resident.join(charge);
204        }
205        if mask_beam {
206            let mut charge = admit(
207                &context.host,
208                &context.policy,
209                &Demand {
210                    phase: "restart residual beam metadata",
211                    memory: ExistingProducts::beam_reading_bytes(planes, polarizations),
212                },
213            )?;
214            let before = existing.beam_resident_bytes();
215            existing.read_mask_beams(channels, main.direction.increment_rad()[0].abs())?;
216            charge.retain(existing.beam_resident_bytes() - before);
217            continuation.mask_resident.join(charge);
218        }
219        continuation.existing = Some(existing);
220    }
221    Ok(continuation)
222}
