18fn main() -> ExitCode {
19    TaskFrontEnd::new(
20        "flagmanager",
21        flagmanager_task_schema_bundle(),
22        &["flagmanager"],
23        run,
24    )
25    .expect("the flagmanager contract carries its surface")
26    .main(
27        std::env::args_os().skip(1),
28        CasaLoggingConfig::from_env(),
29        std::env::set_current_dir,
30        &mut std::io::stdout(),
31        std::io::stderr(),
32    )
33}

35fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
36    let vis = task.path("vis")?;
37    let request = FlagManagerTaskRequest {
38        vis: vis.clone(),
39        mode: task.required_text("mode")?.to_ascii_lowercase(),
40        versionname: task.text("versionname")?,
41        oldname: task.text("oldname")?,
42        comment: task.text("comment")?,
43        merge: task_flag_merge(task, "merge")?,
44    };
45    let result = execute(request)?;
46    // CASA's `list` returns the versions; the other modes return None.
47    let value = match &result {
48        FlagManagerTaskResult::Versions(versions) => flagmanager_list_record(versions, &vis),
49        FlagManagerTaskResult::Mutation(_) => CasaValue::None,
50    };
51    ms_task_success(value, &result)
52}
53
54fn execute(request: FlagManagerTaskRequest) -> Result<FlagManagerTaskResult, TaskFailure> {
55    let failed = |error: casa_ms::FlaggingError| TaskFailure::execution(error.to_string());
56    let required = |name: &str, value: Option<String>| {
57        value.ok_or_else(|| {
58            TaskFailure::validation(format!(
59                "flagmanager mode {:?} requires {name}",
60                request.mode
61            ))
62            .at(name)
63        })
64    };
65    let mut ms = MeasurementSet::open(&request.vis)
66        .map_err(|error| TaskFailure::execution(error.to_string()))?;
67    let mutation = |mode: &str, versionname, oldname| {
68        Ok(FlagManagerTaskResult::Mutation(FlagManagerMutationResult {
69            mode: mode.to_string(),
70            versionname,
71            oldname,
72        }))
73    };
74    match request.mode.as_str() {
75        "list" => Ok(FlagManagerTaskResult::Versions(
76            list_flag_versions(&ms).map_err(failed)?,
77        )),
78        "save" => {
79            let name = required("versionname", request.versionname.clone())?;
80            let comment = request.comment.as_deref().unwrap_or("");
81            keep_existing_version(&ms, &name)?;
82            save_flag_version(&ms, &name, comment, request.merge).map_err(failed)?;
83            mutation("save", Some(name), None)
84        }
85        "restore" => {
86            let name = required("versionname", request.versionname.clone())?;
87            restore_flag_version(&mut ms, &name, request.merge).map_err(failed)?;
88            ms.save_main_table_only()
89                .map_err(|error| TaskFailure::execution(error.to_string()))?;
90            mutation("restore", Some(name), None)
91        }
92        "delete" => {
93            let name = required("versionname", request.versionname.clone())?;
94            delete_flag_version(&ms, &name).map_err(failed)?;
95            mutation("delete", Some(name), None)
96        }
97        "rename" => {
98            let old = required("oldname", request.oldname.clone())?;
99            let name = required("versionname", request.versionname.clone())?;
100            let comment = request.comment.as_deref().unwrap_or("");
101            rename_flag_version(&ms, &old, &name, comment).map_err(failed)?;
102            mutation("rename", Some(name), Some(old))
103        }
104        other => Err(TaskFailure::validation(format!(
105            "unsupported flagmanager mode {other:?}; expected list, save, restore, delete or rename"
106        ))
107        .at("mode")),
108    }
109}

111/// CASA's task keeps a version that a save would replace: it is renamed
112/// `<name>.old.<seconds since 1970>` first, so the save adds a new version
113/// and the old flags stay restorable (`task_flagmanager.py`). CASA writes
114/// the new comment onto the renamed version; casa-rs keeps its own (see
115/// `docs/CASA (C++) bugs.md`).
116fn keep_existing_version(ms: &MeasurementSet, name: &str) -> Result<(), TaskFailure> {
117    let failed = |error: casa_ms::FlaggingError| TaskFailure::execution(error.to_string());
118    // `main`, the working copy, is listed but is no saved version.
119    let Some(existing) = list_flag_versions(ms)
120        .map_err(failed)?
121        .into_iter()
122        .skip(1)
123        .find(|version| version.name == name)
124    else {
125        return Ok(());
126    };
127    let seconds = std::time::SystemTime::now()
128        .duration_since(std::time::UNIX_EPOCH)
129        .map_or(0, |elapsed| elapsed.as_secs());
130    let kept = format!("{name}.old.{seconds}");
131    tracing::warn!("Version name '{name}' already exist. Will rename it to {kept}");
132    rename_flag_version(ms, name, &kept, &existing.comment).map_err(failed)
133}
