8/// Result of running one casacore `hclean` minor-cycle call on a single plane.
9#[derive(Debug, Clone, PartialEq)]
10pub struct HogbomMinorCycle2d {
11    /// CASA-reported iteration count returned by `hclean`.
12    pub iterdone: usize,
13    /// Peak absolute residual after the minor-cycle call.
14    pub peak_residual_jy_per_beam: f32,
15    /// Updated model plane in canonical x-major `(x, y)` order.
16    pub model: Vec<f32>,
17    /// Updated residual plane in canonical x-major `(x, y)` order.
18    pub residual: Vec<f32>,
19}

44/// Typed Rust-facing access to casacore's Hogbom oracle.
45pub struct HogbomOracle;
46
47#[cfg_attr(not(has_casacore_cpp), allow(unused_variables))]
48impl HogbomOracle {
49    /// Run one casacore `hclean` minor-cycle call on a single residual/PSF plane.
50    ///
51    /// Rust imaging planes are x-major and may place the PSF peak anywhere.
52    /// The Fortran oracle consumes column-major, unit-peak planes and treats
53    /// `[nx / 2, ny / 2]` as the PSF reference pixel. This bridge normalizes
54    /// the working planes and embeds them in a zero-padded canvas, translating
55    /// the PSF peak to that reference pixel without wrapping finite support.
56    /// Returned model and residual planes are cropped back into the caller's
57    /// canonical coordinates; production coordinates are never changed to
58    /// accommodate the test oracle.
59    pub fn clean_minor_cycle_2d(
60        psf: &[f32],
61        residual: &[f32],
62        shape: [usize; 2],
63        gain: f32,
64        threshold: f32,
65        cycle_niter: usize,
66    ) -> Result<HogbomMinorCycle2d, OracleError> {
67        Self::clean_minor_cycle_2d_with_mask(
68            psf,
69            residual,
70            shape,
71            None,
72            gain,
73            threshold,
74            cycle_niter,
75        )
76    }
77
78    /// Run one casacore `hclean` call with an explicit x-major CLEAN mask.
79    pub fn clean_minor_cycle_2d_masked(
80        psf: &[f32],
81        residual: &[f32],
82        shape: [usize; 2],
83        mask: &[bool],
84        gain: f32,
85        threshold: f32,
86        cycle_niter: usize,
87    ) -> Result<HogbomMinorCycle2d, OracleError> {
88        Self::clean_minor_cycle_2d_with_mask(
89            psf,
90            residual,
91            shape,
92            Some(mask),
93            gain,
94            threshold,
95            cycle_niter,
96        )
97    }
98
99    fn clean_minor_cycle_2d_with_mask(
100        psf: &[f32],
101        residual: &[f32],
102        shape: [usize; 2],
103        mask: Option<&[bool]>,
104        gain: f32,
105        threshold: f32,
106        cycle_niter: usize,
107    ) -> Result<HogbomMinorCycle2d, OracleError> {
108        oracle_operation!("hogbom.clean_minor_cycle_2d", {
109            let [nx, ny] = shape;
110            if psf.len() != nx * ny || residual.len() != nx * ny {
111                return Err(OracleError::InvalidInput {
112                    context: "hogbom planes",
113                    message: format!(
114                        "expected {} pixels for shape {:?}, got psf={} residual={}",
115                        nx * ny,
116                        shape,
117                        psf.len(),
118                        residual.len()
119                    ),
120                });
121            }
122            if mask.is_some_and(|mask| mask.len() != nx * ny) {
123                return Err(OracleError::InvalidInput {
124                    context: "hogbom mask",
125                    message: format!("expected {} mask pixels for shape {shape:?}", nx * ny),
126                });
127            }
128
129            let mut psf_peak = None::<(usize, f32)>;
130            for (index, value) in psf.iter().copied().enumerate() {
131                let magnitude = value.abs();
132                if value.is_finite() && psf_peak.is_none_or(|(_, best)| magnitude > best) {
133                    psf_peak = Some((index, magnitude));
134                }
135            }
136            let (psf_peak, magnitude) = psf_peak.ok_or_else(|| OracleError::InvalidInput {
137                context: "hogbom PSF plane",
138                message: "PSF has no finite peak".to_owned(),
139            })?;
140            if magnitude == 0.0 {
141                return Err(OracleError::InvalidInput {
142                    context: "hogbom PSF plane",
143                    message: "PSF peak is zero".to_owned(),
144                });
145            }
146            let psf_peak = [psf_peak / ny, psf_peak % ny];
147            let padded_shape = [
148                nx.checked_mul(2).ok_or_else(|| OracleError::InvalidInput {
149                    context: "hogbom plane shape",
150                    message: "padded width overflow".to_owned(),
151                })?,
152                ny.checked_mul(2).ok_or_else(|| OracleError::InvalidInput {
153                    context: "hogbom plane shape",
154                    message: "padded height overflow".to_owned(),
155                })?,
156            ];
157            let offset = [nx - psf_peak[0], ny - psf_peak[1]];
158            let normalized_psf = psf
159                .iter()
160                .map(|value| *value / magnitude)
161                .collect::<Vec<_>>();
162            let normalized_residual = residual
163                .iter()
164                .map(|value| *value / magnitude)
165                .collect::<Vec<_>>();
166            let psf = pack_fortran_padded(&normalized_psf, [nx, ny], padded_shape, offset);
167            let residual =
168                pack_fortran_padded(&normalized_residual, [nx, ny], padded_shape, offset);
169            let mask = mask.map(|mask| {
170                let values = mask
171                    .iter()
172                    .map(|value| u8::from(*value))
173                    .collect::<Vec<_>>();
174                pack_fortran_padded(&values, [nx, ny], padded_shape, offset)
175            });
176            let padded_cells = padded_shape[0] * padded_shape[1];
177            let mut model_out = vec![0.0f32; padded_cells];
178            let mut residual_out = vec![0.0f32; padded_cells];
179            let mut iterdone = 0i32;
180            let mut peak = 0.0f32;
181            let mut error: *mut std::ffi::c_char = std::ptr::null_mut();
182            let rc = unsafe {
183                ffi_cpp_hogbom_clean_minor_cycle_2d(
184                    padded_shape[0] as i32,
185                    padded_shape[1] as i32,
186                    gain,
187                    threshold / magnitude,
188                    cycle_niter as i32,
189                    psf.as_ptr(),
190                    residual.as_ptr(),
191                    mask.as_ref().map_or(std::ptr::null(), Vec::as_ptr),
192                    model_out.as_mut_ptr(),
193                    residual_out.as_mut_ptr(),
194                    padded_cells as i32,
195                    &mut iterdone,
196                    &mut peak,
197                    &mut error,
198                )
199            };
200            unsafe {
201                CasacoreOracleRuntime::cpp_status(
202                    "hogbom.clean_minor_cycle_2d",
203                    rc,
204                    error,
205                    cpp_table_free_error,
206                )?;
207            }
208            Ok(HogbomMinorCycle2d {
209                iterdone: iterdone as usize,
210                peak_residual_jy_per_beam: peak * magnitude,
211                model: unpack_fortran_padded(&model_out, [nx, ny], padded_shape, offset),
212                residual: unpack_fortran_padded(&residual_out, [nx, ny], padded_shape, offset)
213                    .into_iter()
214                    .map(|value| value * magnitude)
215                    .collect(),
216            })
217        })
218    }
219}

221#[cfg(has_casacore_cpp)]
222fn pack_fortran_padded<T: Copy + Default>(
223    input: &[T],
224    shape: [usize; 2],
225    padded_shape: [usize; 2],
226    offset: [usize; 2],
227) -> Vec<T> {
228    let [nx, ny] = shape;
229    let [padded_nx, padded_ny] = padded_shape;
230    let mut packed = vec![T::default(); padded_nx * padded_ny];
231    for source_x in 0..nx {
232        for source_y in 0..ny {
233            let oracle_x = source_x + offset[0];
234            let oracle_y = source_y + offset[1];
235            packed[oracle_x + padded_nx * oracle_y] = input[source_x * ny + source_y];
236        }
237    }
238    packed
239}

241#[cfg(has_casacore_cpp)]
242fn unpack_fortran_padded(
243    input: &[f32],
244    shape: [usize; 2],
245    padded_shape: [usize; 2],
246    offset: [usize; 2],
247) -> Vec<f32> {
248    let [nx, ny] = shape;
249    let [padded_nx, _] = padded_shape;
250    let mut unpacked = vec![0.0; nx * ny];
251    for source_x in 0..nx {
252        for source_y in 0..ny {
253            let oracle_x = source_x + offset[0];
254            let oracle_y = source_y + offset[1];
255            unpacked[source_x * ny + source_y] = input[oracle_x + padded_nx * oracle_y];
256        }
257    }
258    unpacked
259}
