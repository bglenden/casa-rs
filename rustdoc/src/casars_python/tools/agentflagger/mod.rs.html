35#[derive(Default)]
36struct Session {
37    attached: Option<Attachment>,
38    pending: Vec<Agent>,
39    initialized: Vec<Agent>,
40}
41
42struct Attachment {
43    path: PathBuf,
44    selection: Option<MsSelection>,
45    column_names: Vec<String>,
46    time_chunk: Option<f64>,
47}
48
49#[pyclass(name = "agentflagger", module = "casars.tools", frozen)]
50pub(crate) struct PyAgentFlagger {
51    session: Mutex<Session>,
52}
53
54impl PyAgentFlagger {
55    fn with_session<T: Send>(
56        &self,
57        py: Python<'_>,
58        operation: impl FnOnce(&mut Session) -> PyResult<T> + Send,
59    ) -> PyResult<T> {
60        py.detach(|| {
61            let mut session = self.session.lock().map_err(runtime_error)?;
62            operation(&mut session)
63        })
64    }
65
66    fn queue(&self, py: Python<'_>, parameters: &Parameters, specialized: bool) -> PyResult<bool> {
67        self.with_session(py, |session| {
68            let Some(table) = &session.attached else {
69                return Ok(true);
70            };
71            let Some(mut agent) = agent(parameters, table.time_chunk) else {
72                return Ok(false);
73            };
74            agent.check_column(&table.column_names);
75            if !specialized && !agent.configurable {
76                return Ok(false);
77            }
78            session.pending.push(agent);
79            Ok(true)
80        })
81    }
82}

93    #[pyo3(signature = (msname = "", ntime = 0.0), text_signature = "($self, msname='', ntime=0.0)")]
94    fn open(&self, py: Python<'_>, msname: &str, ntime: f64) -> PyResult<bool> {
95        if msname.is_empty() {
96            return Ok(false);
97        }
98        let path = absolute_path(Path::new(msname))?;
99        self.with_session(py, |session| {
100            let mut table = open_table(&path)?;
101            if table.main_table().info().table_type != "Calibration" {
102                return Err(PyRuntimeError::new_err(
103                    "agentflagger requires a calibration table (#1527); MeasurementSet support is tracked in #978",
104                ));
105            }
106            let main = table.main_table();
107            if main.schema().is_some_and(|schema| !schema.contains_column("OBSERVATION_ID"))
108                || main.keywords().get("OBSERVATION").is_none()
109            {
110                // CASA's writable NewCalTable open upgrades legacy tables.
111                drop(table);
112                casa_calibration::table::update_calibration_table(&path).map_err(runtime_error)?;
113                table = open_table(&path)?;
114            }
115            let column_names = table.main_table().schema().into_iter()
116                .flat_map(casa_tables::TableSchema::columns)
117                .map(|column| column.name().to_owned())
118                .collect();
119            *session = Session {
120                attached: Some(Attachment {
121                    path, selection: None, column_names,
122                    time_chunk: (ntime > 0.0 && ntime.is_finite()).then_some(ntime),
123                }),
124                ..Session::default()
125            };
126            Ok(true)
127        })
128    }
129
130    fn done(&self, py: Python<'_>) -> PyResult<bool> {
131        self.with_session(py, |session| {
132            *session = Session::default();
133            Ok(true)
134        })
135    }

137    #[pyo3(signature = (config = None, field = "", spw = "", array = "", feed = "", scan = "", antenna = "", uvrange = "", timerange = "", correlation = "", intent = "", observation = ""), text_signature = "($self, config={}, field='', spw='', array='', feed='', scan='', antenna='', uvrange='', timerange='', correlation='', intent='', observation='')")]
138    #[allow(clippy::too_many_arguments)]
139    fn selectdata(
140        &self,
141        py: Python<'_>,
142        #[pyo3(from_py_with = supplied)] config: Option<Bound<'_, PyAny>>,
143        field: &str,
144        spw: &str,
145        array: &str,
146        feed: &str,
147        scan: &str,
148        antenna: &str,
149        uvrange: &str,
150        timerange: &str,
151        correlation: &str,
152        intent: &str,
153        observation: &str,
154    ) -> PyResult<bool> {
155        let config = config
156            .map(|value| record(&value, "config"))
157            .transpose()?
158            .unwrap_or_default();
159        let requested = if config.is_empty() {
160            Selectors {
161                field,
162                spw,
163                array,
164                feed,
165                scan,
166                antenna,
167                uvrange,
168                timerange,
169                correlation,
170                intent,
171                observation,
172            }
173            .selection()
174        } else {
175            selection(&config)?
176        };
177        self.with_session(py, |session| {
178            if let Some(attached) = &mut session.attached {
179                let table = open_table(&attached.path)?;
180                flagdata_calibration_agent_rows(&table, &requested).map_err(runtime_error)?;
181                attached.selection = Some(requested);
182            }
183            Ok(true)
184        })
185    }

187    fn parseagentparameters(&self, py: Python<'_>, aparams: &Bound<'_, PyAny>) -> PyResult<bool> {
188        self.queue(py, &record(aparams, "aparams")?, false)
189    }
190
191    fn init(&self, py: Python<'_>) -> PyResult<bool> {
192        self.with_session(py, |session| {
193            if session
194                .attached
195                .as_ref()
196                .and_then(|attached| attached.selection.as_ref())
197                .is_none()
198            {
199                return Ok(false);
200            }
201            let configs = session
202                .initialized
203                .iter()
204                .chain(&session.pending)
205                .filter(|agent| agent.configurable)
206                .map(|agent| (agent.mode.as_str(), agent.channelavg, agent.timeavg));
207            validate_calibration_flag_averaging(configs).map_err(runtime_error)?;
208            let initialized = session.initialized.len();
209            session
210                .initialized
211                .extend(session.pending.drain(..).filter(Agent::can_init));
212            Ok(initialized != session.initialized.len())
213        })
214    }
215
216    #[pyo3(signature = (writeflags = true, sequential = true), text_signature = "($self, writeflags=True, sequential=True)")]
217    fn run(&self, py: Python<'_>, writeflags: bool, sequential: bool) -> PyResult<Py<PyAny>> {
218        let value = self.with_session(py, |session| {
219            let agents = std::mem::take(&mut session.initialized);
220            let Some(attached) = &session.attached else {
221                return Ok(empty_record());
222            };
223            let Some(selection) = &attached.selection else {
224                return Ok(empty_record());
225            };
226            if agents.is_empty() {
227                return Ok(empty_record());
228            }
229            let table = open_table(&attached.path)?;
230            let rows = flagdata_calibration_agent_rows(&table, selection).map_err(runtime_error)?;
231            let requests: Vec<FlagDataRequest> =
232                agents.iter().map(|agent| agent.request.clone()).collect();
233            let reports =
234                flagdata_calibration_agents(table, &requests, &rows, writeflags, sequential)
235                    .map_err(runtime_error)?;
236            Ok(program_record(&reports, &agents))
237        })?;
238        reply_value(py, &value)
239    }
240
241    fn printflagselection(&self, py: Python<'_>) -> PyResult<bool> {
242        let line = self.with_session(py, |session| {
243            Ok(session
244                .attached
245                .as_ref()
246                .and_then(|target| target.selection.as_ref())
247                .map_or_else(
248                    || LogLine::info("No flag selection configured."),
249                    |selection| LogLine::info(format!("{selection:?}")),
250                ))
251        })?;
252        emit(py, LOGGER, &[line])?;
253        Ok(true)
254    }
255}

261fn absolute_path(path: &Path) -> PyResult<PathBuf> {

265fn open_table(path: &Path) -> PyResult<MeasurementSet> {
266    if !path.exists() {
267        return Err(PyRuntimeError::new_err(format!(
268            "Table {} does not exist",
269            path.display()
270        )));
271    }
272    MeasurementSet::open(path).map_err(runtime_error)
273}

275fn record(value: &Bound<'_, PyAny>, name: &str) -> PyResult<Parameters> {
276    let dict = value
277        .cast::<PyDict>()
278        .map_err(|_| PyTypeError::new_err(format!("_{name} is not a dictionary")))?;
279    dict.iter()
280        .map(|(key, value)| Ok((key.extract::<String>()?, parameter_value(&value)?)))
281        .collect()
282}

284#[allow(clippy::unnecessary_wraps)] // PyO3's extractor distinguishes a default from explicit None.
285fn supplied<'py>(value: &Bound<'py, PyAny>) -> PyResult<Option<Bound<'py, PyAny>>> {
286    Ok(Some(value.clone()))
287}

289fn empty_record() -> CasaValue {
290    CasaValue::dict(std::iter::empty::<(&str, CasaValue)>())

293fn program_record(reports: &[Option<FlagDataReport>], agents: &[Agent]) -> CasaValue {
294    if reports.iter().all(Option::is_none) {
295        return empty_record();
296    }
297    let array = |rows: &[[f64; 3]]| CasaValue::FloatArray {
298        dtype: FloatDtype::Float64,
299        shape: vec![rows.len(), 3],
300        data: (0..3)
301            .flat_map(|column| rows.iter().map(move |row| CasaFloat(row[column])))
302            .collect(),
303    };
304    let reports = reports
305        .iter()
306        .zip(agents)
307        .filter_map(|(report, agent)| {
308            let report = report.as_ref()?;
309            if let Some(summary) = &report.summary {
310                Some(flagdata_named_summary_record(summary, &agent.name))
311            } else {
312                let thresholds = report.rflag_report.as_ref()?;
313                if thresholds.timedev.is_empty() && thresholds.freqdev.is_empty() {
314                    return None;
315                }
316                Some(CasaValue::dict([
317                    ("name", CasaValue::string(&agent.name)),
318                    ("type", CasaValue::string("rflag")),
319                    ("timedev", array(&thresholds.timedev)),
320                    ("freqdev", array(&thresholds.freqdev)),
321                ]))
322            }
323        })
324        .collect::<Vec<_>>();
325    let count = reports.len() as i64;
326    let mut record = BTreeMap::from([
327        ("type".to_owned(), CasaValue::string("list")),
328        ("nreport".to_owned(), CasaValue::Int { value: count }),
329    ]);
330    record.extend(
331        reports
332            .into_iter()
333            .enumerate()
334            .map(|(i, report)| (format!("report{i}"), report)),
335    );
336    CasaValue::dict(record)
