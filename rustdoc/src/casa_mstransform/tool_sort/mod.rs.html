17/// Write a sorted, independent MeasurementSet at a staging path.
18///
19/// The caller owns publication and holds the guards for `held_tables`.
20pub fn write_sorted_measurement_set(
21    input: &MeasurementSet,
22    staged_output: &Path,
23    columns: Option<&[String]>,
24    in_place: bool,
25    input_name: &Path,
26    held_tables: &BTreeSet<PathBuf>,
27) -> MsResult<()> {
28    let main = input.main_table();
29    let keys = columns.map_or_else(
30        || vec!["TIME"],
31        |columns| columns.iter().map(String::as_str).collect(),
32    );
33    let rows = ordered_rows(main, &keys)?;
34    write_selected_measurement_set_rows(
35        main,
36        &rows,
37        staged_output,
38        EndianFormat::default(),
39        held_tables,
40    )?;
41    let (origin, names) = match columns {
42        None => ("ms::timesort()", "TIME".to_string()),
43        Some(columns) => ("ms::sort()", format!("[{}]", columns.join(","))),
44    };
45    let message = if in_place {
46        format!("Sorted by {names} in ascending order.")
47    } else {
48        let expression = if columns.is_some() {
49            format!(" {names} ")
50        } else {
51            names
52        };
53        format!(
54            "Generated from {} by sorting by {expression} in ascending order.",
55            input_name.display()
56        )
57    };
58    append_history_messages(staged_output, origin, [message])
59        .map_err(|error| MsError::InvalidInput(error.to_string()))
60}

62/// Write an independent MeasurementSet with the specified MAIN row order.
63///
64/// The caller owns publication and holds the guards for `held_tables`.
65///
66/// # Errors
67/// Returns an error if the projected rows or linked subtables cannot be saved.
68pub fn write_selected_measurement_set_rows(
69    main: &Table,
70    rows: &[usize],
71    staged_output: &Path,
72    endian: EndianFormat,
73    held_tables: &BTreeSet<PathBuf>,
74) -> MsResult<()> {
75    let mut ordered = main.to_memory_rows(rows)?;
76    // These optional iterator caches refer to the input's row order.
77    let mut keywords = main.keywords().clone();
78    keywords.remove("SORTED_TABLE");
79    subtables::copy_links(
80        main,
81        &mut keywords,
82        &mut ordered,
83        staged_output,
84        held_tables,
85    )?;
86    *ordered.keywords_mut() = keywords;
87    let bindings = measurement_set_main_table_bindings(&ordered);
88    ordered.prepare_write().save_with_column_overrides(
89        measurement_set_table_options(staged_output).with_endian_format(endian),
90        &bindings,
91        &ColumnOverrides::default(),
92    )?;
93    Ok(())
94}

128/// Capture the filesystem dependencies to protect when writing a named
129/// result later. The flag identifies a linked subtable whose descendants
130/// are protected as well as its own path.
131#[doc(hidden)]
132pub fn capture_named_output_inputs(input: &Path) -> MsResult<Vec<(PathBuf, bool)>> {
133    let mut paths = std::collections::BTreeMap::new();
134    subtables::visit_source_tables(input, None, |path, subtable| {
135        let stored = paths.entry(path.to_owned()).or_insert(false);
136        *stored |= subtable;
137        Ok(())
138    })?;
139    Ok(paths.into_iter().collect())
140}

142fn ordered_rows(table: &Table, names: &[&str]) -> MsResult<Vec<usize>> {
143    let mut rows = (0..table.row_count()).collect::<Vec<_>>();
144    if names.is_empty() {
145        return Ok(rows);
146    }
147    for name in names {
148        let column = table
149            .schema()
150            .and_then(|schema| schema.column(name))
151            .ok_or_else(|| MsError::InvalidInput(format!("Table column {name} is unknown")))?;
152        if *column.column_type() != ColumnType::Scalar {
153            return Err(MsError::InvalidInput(format!(
154                "Invalid table sort: Sort column {name} is not a scalar"
155            )));
156        }
157    }
158    // Reading keys through ordinary cell access loads every visibility
159    // column before the sorted copy can read it again.
160    let values = table.scalar_columns_owned(names)?;
161    let keys: Vec<_> = names.iter().map(|name| &values[*name]).collect();
162    rows.sort_by(|&a, &b| {
163        keys.iter()
164            .map(|column| compare_keys(column[a].as_ref(), column[b].as_ref()))
165            .find(|order| *order != Ordering::Equal)
166            .unwrap_or(Ordering::Equal)
167    });
168    Ok(rows)
169}
170
171fn compare_keys(a: Option<&ScalarValue>, b: Option<&ScalarValue>) -> Ordering {
172    match (a, b) {
173        (Some(ScalarValue::Complex32(a)), Some(ScalarValue::Complex32(b))) => {
174            a.norm_sqr().total_cmp(&b.norm_sqr())
175        }
176        (Some(ScalarValue::Complex64(a)), Some(ScalarValue::Complex64(b))) => {
177            a.norm_sqr().total_cmp(&b.norm_sqr())
178        }
179        // CASA treats the two zero representations as equal sort keys.
180        (Some(ScalarValue::Float32(a)), Some(ScalarValue::Float32(b))) => {
181            a.partial_cmp(b).unwrap_or_else(|| a.total_cmp(b))
182        }
183        (Some(ScalarValue::Float64(a)), Some(ScalarValue::Float64(b))) => {
184            a.partial_cmp(b).unwrap_or_else(|| a.total_cmp(b))
185        }
186        (Some(a), Some(b)) => a.sort_cmp(b).unwrap_or(Ordering::Equal),
187        (Some(_), None) => Ordering::Less,
188        (None, Some(_)) => Ordering::Greater,
189        (None, None) => Ordering::Equal,
190    }
191}
