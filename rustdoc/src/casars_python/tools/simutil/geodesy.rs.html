2//! `simutil`'s geodesy: its datum table and its conversions between UTM,
3//! geodetic and Earth-centred coordinates, over
4//! [`casa_simulation::readantenna::geodesy`], which `readantenna` uses and
5//! which records where it corrects CASA.
6//!
7//! Each returns CASA 6.7.6.14's tuple of Python numbers, for scalar
8//! arguments:
9//! - `getdatum(datumcode)`: `(dx, dy, dz, er, rf)`, the offsets `int` where
10//!   CASA's table writes them so;
11//! - `tconpc`: `(eps, r, so, v0, v2, v4, v6)`; `tmgeod`:
12//!   `(lat, lon, conv, kp)`, the longitude west-positive;
13//! - `utm2long`: `(lon, lat)`; `xyz2long`: `(lon, lat, height)`;
14//!   `xyz2long_old`: `(lon, lat)`; `long2xyz`, `utm2xyz` and
15//!   `locxyz2itrf`: `(x, y, z)`;
16//! - `itrf2locname`: `itrf2loc`'s arrays about the named observatory.
17//!
18//! An unknown datum raises CASA's "unknown datum" as a `RuntimeError`.
19//! CASA's `utm2long` with `verbose` also logs the position it found; casa-rs
20//! does not.
21

25};
26use casa_types::measures::{MPosition, PositionRef};

35/// The datum CASA's `getdatum` names `code`, or its refusal.
36fn named_datum(code: &str) -> PyResult<&'static Datum> {
37    geodesy::datum(code).ok_or_else(|| PyRuntimeError::new_err(format!("unknown datum {code}")))
38}
39
40/// A datum offset as CASA's table holds it: an `int` where it is whole.
41fn offset(value: f64) -> CasaValue {
42    if value.fract() == 0.0 && value.abs() < 1e15 {
43        #[allow(clippy::cast_possible_truncation)]
44        CasaValue::Int(value as i64)
45    } else {
46        CasaValue::Float(value)
47    }
48}
49
50fn floats(values: &[f64]) -> CasaValue {
51    CasaValue::Tuple(values.iter().copied().map(CasaValue::Float).collect())
52}

54#[pymethods]
55impl PySimutil {
56    /// The offsets from ITRF and the ellipsoid of a datum: `(dx, dy, dz,
57    /// er, rf)`, metres and the reciprocal flattening.
58    #[pyo3(signature = (datumcode, verbose = false), text_signature = "($self, datumcode, verbose=False)")]
59    fn getdatum(&self, py: Python<'_>, datumcode: &str, verbose: bool) -> PyResult<Py<PyAny>> {
60        let datum = named_datum(datumcode)?;
61        if verbose {
62            emit(
63                py,
64                LOGGER,
65                &[LogLine::info(format!(
66                    "Using {} datum with {} ellipsoid",
67                    datum.name, datum.ellipsoid.name
68                ))],
69            )?;
70        }
71        CasaValue::Tuple(vec![
72            offset(datum.dx_m),
73            offset(datum.dy_m),
74            offset(datum.dz_m),
75            CasaValue::Float(datum.ellipsoid.equatorial_radius_m),
76            CasaValue::Float(datum.ellipsoid.inverse_flattening),
77        ])
78        .into_python(py)
79    }

81    /// The transverse Mercator zone constants `(eps, r, so, v0, v2, v4,
82    /// v6)`.
83    #[pyo3(signature = (sf, orlim, er, esq, rf), text_signature = "($self, sf, orlim, er, esq, rf)")]
84    fn tconpc(
85        &self,
86        py: Python<'_>,
87        sf: f64,
88        orlim: f64,
89        er: f64,
90        esq: f64,
91        rf: f64,
92    ) -> PyResult<Py<PyAny>> {
93        floats(&zone_constants(sf, orlim, er, esq, rf)).into_python(py)
94    }

96    /// Grid coordinates to `(lat, lon, conv, kp)`: geodetic latitude and
97    /// west-positive longitude (radians), convergence and point scale
98    /// factor.
99    #[pyo3(
100        signature = (n, e, eps, cm, fe, sf, so, r, v0, v2, v4, v6, r#fn, er, esq),
101        text_signature = "($self, n, e, eps, cm, fe, sf, so, r, v0, v2, v4, v6, fn, er, esq)"
102    )]
103    // CASA's signature: one value each.
104    #[allow(clippy::too_many_arguments)]
105    fn tmgeod(
106        &self,
107        py: Python<'_>,
108        n: f64,
109        e: f64,
110        eps: f64,
111        cm: f64,
112        fe: f64,
113        sf: f64,
114        so: f64,
115        r: f64,
116        v0: f64,
117        v2: f64,
118        v4: f64,
119        v6: f64,
120        r#fn: f64,
121        er: f64,
122        esq: f64,
123    ) -> PyResult<Py<PyAny>> {
124        let found = grid_to_geodetic(
125            n,
126            e,
127            eps,
128            cm,
129            fe,
130            sf,
131            so,
132            r,
133            [v0, v2, v4, v6],
134            r#fn,
135            er,
136            esq,
137        );
138        floats(&found).into_python(py)
139    }

141    /// UTM easting and northing in `zone` to `(lon, lat)`, radians,
142    /// longitude east-positive; `nors` is `'N'` or `'S'`.
143    #[pyo3(signature = (east, north, zone, datum, nors), text_signature = "($self, east, north, zone, datum, nors)")]
144    fn utm2long(
145        &self,
146        py: Python<'_>,
147        east: f64,
148        north: f64,
149        zone: i64,
150        datum: &str,
151        nors: &str,
152    ) -> PyResult<Py<PyAny>> {
153        let datum = named_datum(datum)?;
154        let (lon, lat) = utm_to_geodetic(east, north, zone, datum, nors == "S");
155        floats(&[lon, lat]).into_python(py)
156    }
157
158    /// Geodetic longitude, latitude (radians) and elevation on `datum` to
159    /// Earth-centred `(x, y, z)`, metres.
160    #[pyo3(signature = (lon, lat, elevation, datum), text_signature = "($self, lon, lat, elevation, datum)")]
161    fn long2xyz(
162        &self,
163        py: Python<'_>,
164        lon: f64,
165        lat: f64,
166        elevation: f64,
167        datum: &str,
168    ) -> PyResult<Py<PyAny>> {
169        floats(&geodetic_to_xyz(lon, lat, elevation, named_datum(datum)?)).into_python(py)
170    }
171
172    /// Earth-centred x, y, z to `(lon, lat, height)` on `datum`; CASA's
173    /// `(-1, -1, -1)` for an ellipsoid it cannot use.
174    #[pyo3(signature = (x, y, z, datum), text_signature = "($self, x, y, z, datum)")]
175    fn xyz2long(&self, py: Python<'_>, x: f64, y: f64, z: f64, datum: &str) -> PyResult<Py<PyAny>> {
176        match xyz_to_geodetic(x, y, z, named_datum(datum)?) {
177            Some((lon, lat, height)) => floats(&[lon, lat, height]).into_python(py),
178            None => CasaValue::Tuple(vec![CasaValue::Int(-1); 3]).into_python(py),
179        }
180    }
181
182    /// Earth-centred x, y, z to `(lon, lat)` on `datum`, by CASA's older
183    /// iteration.
184    #[pyo3(signature = (x, y, z, datum), text_signature = "($self, x, y, z, datum)")]
185    fn xyz2long_old(
186        &self,
187        py: Python<'_>,
188        x: f64,
189        y: f64,
190        z: f64,
191        datum: &str,
192    ) -> PyResult<Py<PyAny>> {
193        let (lon, lat) = xyz_to_geodetic_old(x, y, z, named_datum(datum)?);
194        floats(&[lon, lat]).into_python(py)
195    }

197    /// UTM easting, northing and elevation to Earth-centred `(x, y, z)`.
198    #[pyo3(
199        signature = (easting, northing, elevation, zone, datum, nors),
200        text_signature = "($self, easting, northing, elevation, zone, datum, nors)"
201    )]
202    // CASA's signature: one value each.
203    #[allow(clippy::too_many_arguments)]
204    fn utm2xyz(
205        &self,
206        py: Python<'_>,
207        easting: f64,
208        northing: f64,
209        elevation: f64,
210        zone: i64,
211        datum: &str,
212        nors: &str,
213    ) -> PyResult<Py<PyAny>> {
214        let datum = named_datum(datum)?;
215        floats(&utm_to_xyz(
216            easting,
217            northing,
218            elevation,
219            zone,
220            datum,
221            nors == "S",
222        ))
223        .into_python(py)
224    }

226    /// Local offsets `(locx, locy, locz)` (east, north, up, metres) about a
227    /// point at `lat`, `longitude` (degrees) and `alt` to Earth-centred
228    /// `(x, y, z)`.
229    #[pyo3(
230        signature = (lat, longitude, alt, locx = 0.0, locy = 0.0, locz = 0.0),
231        text_signature = "($self, lat, longitude, alt, locx=0.0, locy=0.0, locz=0.0)"
232    )]
233    // CASA's signature: one value each.
234    #[allow(clippy::too_many_arguments)]
235    fn locxyz2itrf(
236        &self,
237        py: Python<'_>,
238        lat: f64,
239        longitude: f64,
240        alt: f64,
241        locx: f64,
242        locy: f64,
243        locz: f64,
244    ) -> PyResult<Py<PyAny>> {
245        floats(&local_to_itrf(lat, longitude, alt, [locx, locy, locz])).into_python(py)
246    }

248    /// Earth-centred positions relative to a known observatory: `itrf2loc`
249    /// about the observatory's WGS84 position, as CASA finds it with
250    /// `me.observatory` and `long2xyz`.
251    #[pyo3(signature = (x, y, z, obsname), text_signature = "($self, x, y, z, obsname)")]
252    fn itrf2locname(
253        &self,
254        py: Python<'_>,
255        x: DoubleList,
256        y: DoubleList,
257        z: DoubleList,
258        obsname: &str,
259    ) -> PyResult<Py<PyAny>> {
260        let centre = py
261            .detach(|| {
262                let measures = casa_ms::open_measures_runtime()
263                    .map_err(|error| format!("measures data: {error}"))?;
264                let position = MPosition::from_observatory_name(obsname, &*measures)
265                    .map_err(|error| error.to_string())?
266                    .ok_or_else(|| format!("Unknown observatory {obsname}"))?;
267                let wgs84 = position
268                    .convert_to(PositionRef::WGS84)
269                    .map_err(|error| error.to_string())?;
270                let [lon, lat, height] = wgs84.values();
271                let datum = geodesy::datum("WGS84").expect("WGS84 is one of casa-rs's datums");
272                Ok::<_, String>(geodetic_to_xyz(lon, lat, height, datum))
273            })
274            .map_err(PyRuntimeError::new_err)?;
275        self.itrf2loc(py, x, y, z, centre[0], centre[1], centre[2])
276    }
