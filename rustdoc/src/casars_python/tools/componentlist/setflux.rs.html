2//! CASA's component flux setter, preserving values it would silently discard.
3
4use casa_components::{Flux, Polarisation};
5use num_complex::Complex64;
6use pyo3::prelude::*;
7
8use crate::convert::{ConvertError, ConvertResult, Variant, casacore_unit, complex_values};

10/// Preserve NumPy scalar and empty-array distinctions in the CASA binding.
11pub(super) fn argument(value: Option<Bound<'_, PyAny>>) -> PyResult<(Variant, bool)> {
12    if let Some(ref object) = value {
13        let ndim = object.getattr("ndim").and_then(|v| v.extract::<usize>());
14        if ndim.is_ok() && object.is_instance(&object.py().import("numpy")?.getattr("ndarray")?)? {
15            if matches!(ndim, Ok(0)) {
16                return super::setrefdir_args::argument(Some(object.call_method0("item")?));
17            }
18            let size = object.getattr("size")?.extract::<usize>()?;
19            let kind = object
20                .getattr("dtype")?
21                .getattr("kind")?
22                .extract::<String>()?;
23            if size == 0 && matches!(kind.as_str(), "f" | "i" | "u" | "c") {
24                return Ok((Variant::Numbers(Vec::new()), false));
25            }
26        }
27    }
28    super::setrefdir_args::argument(value)
29}

31/// CASA flattens numeric arrays in column-major order before setting flux.
32fn numeric(value: &Variant) -> Option<Vec<Complex64>> {
33    match value {
34        Variant::Array { values, .. } => {
35            Some(values.iter().map(|&v| Complex64::new(v, 0.0)).collect())
36        }
37        Variant::ComplexArray { values, .. } => Some(values.clone()),
38        _ => complex_values(value),
39    }
40}

42/// Prepare and validate the whole flux before changing any component.
43pub(super) fn prepare(
44    value: &Variant,
45    unit: &str,
46    polarization: &str,
47    error: &Variant,
48) -> ConvertResult<Flux> {
49    let pol = Polarisation::parse_casa(polarization).map_err(|_| {
50        ConvertError::runtime(
51            "Unknown polarization. Possible values are:\n 'Stokes'  'Linear'  'Circular' ",
52        )
53    })?;
54    let target = casacore_unit(unit).map_err(ConvertError::runtime)?;
55    let jy = casacore_unit("Jy").map_err(ConvertError::runtime)?;
56    if !target.conformant(&jy) {
57        return Err(ConvertError::runtime(
58            "The flux units must have the same dimensions as the Jansky\nFlux not changed on any components",
59        ));
60    }
61    let values = if value.is_empty_sequence() {
62        vec![Complex64::new(0.0, 0.0); 4]
63    } else {
64        let values = numeric(value).ok_or_else(|| {
65            ConvertError::runtime("Could not understand the type of flux value variable")
66        })?;
67        if values.is_empty() {
68            vec![Complex64::new(0.0, 0.0); 4]
69        } else {
70            values
71        }
72    };
73    let errors = numeric(error);
74    if errors
75        .as_ref()
76        .is_some_and(|errors| errors.len() != values.len())
77    {
78        return Err(ConvertError::runtime(
79            "Error array must have same length as value array",
80        ));
81    }
82    if !matches!((values.len(), pol), (1, Polarisation::Stokes) | (4, _)) {
83        return Err(ConvertError::runtime(
84            "The flux must have one or four elements,\none element can only be used if the polarization is 'Stokes'.\nFlux not changed on any components",
85        ));
86    }
87    for item in values.iter().chain(errors.iter().flatten()) {
88        if !item.re.is_finite() || !item.im.is_finite() {
89            return Err(ConvertError::runtime(
90                "non-finite flux or error refused (#1053)",
91            ));
92        }
93        if item.im != 0.0 {
94            return Err(ConvertError::runtime(
95                "imaginary flux or error would be discarded by CASA (#1053)",
96            ));
97        }
98    }
99    let mut result_values = [Complex64::new(0.0, 0.0); 4];
100    result_values[..values.len()].copy_from_slice(&values);
101    let mut result_errors = [Complex64::new(0.0, 0.0); 4];
102    if let Some(errors) = errors {
103        result_errors[..errors.len()].copy_from_slice(&errors);
104    }
105    // CASA's setter ultimately stores the values as Stokes, regardless of pol.
106    Flux::new(result_values, unit, Polarisation::Stokes)
107        .map(|flux| flux.with_errors(result_errors))
108        .map_err(|error| ConvertError::runtime(error.to_string()))
109}
