16pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
17    let outfile = required(task, "outfile")?;
18    let requested_path = Path::new(&outfile);
19    let destination = task.workspace().join(requested_path);
20    let overwrite = task.flag("overwrite")?;
21    if !overwrite && destination.exists() {
22        return Err(exists(requested_path));
23    }
24    let hosts = task.texts("hosts")?;
25    match hosts.first().map(String::as_str) {
26        None => return Err(invalid("hosts", "Parameter hosts must be specified")),
27        Some("") => {
28            return Err(invalid(
29                "hosts",
30                "The first element of the hosts list must be specified",
31            ));
32        }
33        Some(_) => {}
34    }
35    let asdm = required(task, "asdm")?;
36    let tw = task.required_text("tw")?;
37    let search = task.required_text("search")?;
38    let first = task.flag("firstintegration")?;
39    let mut metadata = Map::from_iter([
40        ("caltype".into(), json!("ALMA antenna positions")),
41        (
42            "description".into(),
43            json!("ALMA ITRF antenna positions in meters"),
44        ),
45        ("product_code".into(), json!("antposalma")),
46        ("outfile".into(), json!(outfile)),
47        ("hosts".into(), json!(hosts)),
48    ]);
49    let mut query = vec![("asdm", Value::from(asdm))];
50    if !tw.is_empty() {
51        validate_window(&tw)?;
52        query.push(("tw", Value::from(tw)));
53    }
54    if let Some(snr) = signal_to_noise(task.value("snr"))? {
55        query.push(("snr", snr));
56    }
57    if !search.is_empty() {
58        query.push(("search", Value::from(search)));
59    }
60    query.push(("firstIntegration", Value::from(first)));
61    let attempts = Retry::read(task)?;
62    metadata.insert("nretry".into(), json!(attempts.count));
63    metadata.insert("rdelay".into(), json!(attempts.delay_seconds));
64    let (url, data) = retrieve(&hosts, &query_string(&query), attempts)?;
65    metadata.extend(
66        query
67            .into_iter()
68            .map(|(name, value)| (name.to_string(), value)),
69    );
70    metadata.insert("successful_url".into(), Value::from(url));
71    metadata.insert("timestamp".into(), Value::from(timestamp()));
72    publish(
73        &destination,
74        requested_path,
75        overwrite,
76        &json!({"data": data, "metadata": metadata}),
77    )?;
78    TaskSuccess::new(CasaValue::None, &())
79}

93        "A file or directory named {} already exists and overwrite is False, so exiting. \
94         Either rename the existing file or directory, change the value of overwrite to True, or both.",
95        path.display()

99fn validate_window(tw: &str) -> Result<(), TaskFailure> {
100    let Some((begin, end)) = tw.split_once(',').filter(|(_, end)| !end.contains(',')) else {
101        return Err(invalid(
102            "tw",
103            "Parameter tw should contain exactly one comma that separates two times",
104        ));
105    };
106    let days = Unit::new("d").expect("known time unit");
107    let angles = Unit::new("deg").expect("known angle unit");
108    let parse = |label, text: &str| {
109        Quantity::read(text).ok().and_then(|time| {
110            time.get_value_in(&days).or_else(|_| {
111                time.get_value_in(&angles).map(|degrees| degrees / 360.0)
112            }).ok()
113        })
114            // The pinned CASA task compares timestamps formatted as FITS
115            // with whole seconds, including when the input is an angle.
116            .map(|days| (days * 86_400.0).round())
117            .ok_or_else(|| invalid("tw", format!("{label} time {text} does not appear to have a valid format. The correct format is of the form YYYY-MM-DDThh:mm:ss.")))
118    };
119    if parse("Begin", begin)? >= parse("End", end)? {
120        return Err(invalid(
121            "tw",
122            format!("Parameter tw, start time ({begin}) must be less than end time ({end})."),
123        ));
124    }
125    Ok(())
126}

128fn signal_to_noise(value: Option<&ParameterValue>) -> Result<Option<Value>, TaskFailure> {
129    match value {
130        Some(ParameterValue::String(text)) if text == "default" => Ok(None),
131        Some(ParameterValue::String(_)) => Err(invalid(
132            "snr",
133            "If snr is a string, its only permissible value is 'default'",
134        )),
135        Some(ParameterValue::Integer(number)) if *number >= 0 => Ok(Some(json!(number))),
136        Some(ParameterValue::Float(number)) if *number >= 0.0 => Ok(Some(json!(number))),
137        Some(ParameterValue::Bool(value)) => Ok(Some(json!(value))),
138        Some(ParameterValue::Integer(number)) => Err(invalid(
139            "snr",
140            format!("If a number, parameter snr ({number}) must be non-negative."),
141        )),
142        Some(ParameterValue::Float(number)) => Err(invalid(
143            "snr",
144            format!(
145                "If a number, parameter snr ({}) must be non-negative.",
146                python_float(*number)
147            ),
148        )),
149        _ => Err(TaskFailure::execution("snr must be a string or number").at("snr.type")),
150    }
151}

153#[derive(Clone, Copy)]
154struct Retry {
155    count: i64,
156    delay_seconds: f64,
157}
158
159impl Retry {
160    fn read(task: &TaskInvocation<'_>) -> Result<Self, TaskFailure> {
161        let count = task.integer("nretry")?;
162        let mut delay_seconds = task.number("rdelay")?;
163        if count < 0 {
164            return Err(invalid(
165                "nretry",
166                format!("Parameter nretry (={count}) must be non-negative."),
167            ));
168        }
169        if count > 0 {
170            if delay_seconds <= 0.0 || Duration::try_from_secs_f64(delay_seconds).is_err() {
171                return Err(invalid(
172                    "rdelay",
173                    format!(
174                        "If nretry > 0, rdelay (={}) must be positive.",
175                        python_float(delay_seconds)
176                    ),
177                ));
178            }
179            if delay_seconds < 0.5 {
180                tracing::warn!(
181                    original = delay_seconds,
182                    "Setting the retry delay to 0.5 seconds"
183                );
184                delay_seconds = 0.5;
185            }
186        }
187        Ok(Self {
188            count,
189            delay_seconds,
190        })
191    }
192}

194fn query_string(parameters: &[(&str, Value)]) -> String {
195    parameters
196        .iter()
197        .map(|(name, value)| {
198            let text = match value {
199                Value::String(text) => text.clone(),
200                Value::Bool(true) => "True".to_string(),
201                Value::Bool(false) => "False".to_string(),
202                Value::Number(value) if value.is_f64() => {
203                    python_float(value.as_f64().expect("known JSON float"))
204                }
205                value => value.to_string(),
206            };
207            let mut encoded = format!("{name}=");
208            for byte in text.bytes() {
209                match byte {
210                    b' ' => encoded.push('+'),
211                    b'-' | b'_' | b'.' | b'~' => encoded.push(char::from(byte)),
212                    byte if byte.is_ascii_alphanumeric() => encoded.push(char::from(byte)),
213                    byte => {
214                        use std::fmt::Write as _;
215                        write!(&mut encoded, "%{byte:02X}").expect("writing to a string succeeds");
216                    }
217                }
218            }
219            encoded
220        })
221        .collect::<Vec<_>>()
222        .join("&")
223}

225fn python_float(value: f64) -> String {
226    let magnitude = value.abs();
227    if magnitude != 0.0 && !(1.0e-4..1.0e16).contains(&magnitude) {
228        let scientific = format!("{value:e}");
229        let (mantissa, exponent) = scientific.split_once('e').expect("exponential format");
230        let exponent = exponent.parse::<i32>().expect("integer exponent");
231        return format!("{mantissa}e{exponent:+03}");
232    }
233    let mut text = value.to_string();
234    if !text.contains('.') {
235        text.push_str(".0");
236    }
237    text
238}

240fn retrieve(
241    hosts: &[String],
242    query: &str,
243    attempts: Retry,
244) -> Result<(String, Value), TaskFailure> {
245    let agent = crate::calibration_service::agent(Duration::from_secs(400))
246        .map_err(|error| TaskFailure::execution(error.to_string()).at("hosts"))?;
247    for attempt in 0..=attempts.count {
248        if attempt > 0 {
249            std::thread::sleep(Duration::from_secs_f64(attempts.delay_seconds));
250        }
251        for host in hosts {
252            if host
253                .parse::<ureq::http::Uri>()
254                .ok()
255                .is_none_or(|uri| uri.scheme().is_none() || uri.authority().is_none())
256            {
257                return Err(invalid(
258                    "hosts",
259                    format!("Parameter hosts: {host} is not a valid host expressed as a URL."),
260                ));
261            }
262            let url = format!("{host}/?{query}");
263            let text = match query_host(&agent, &url) {
264                Ok(Some(text)) => text,
265                Ok(None) => continue,
266                Err(error) => {
267                    tracing::warn!(%error, "ALMA antenna position request failed");
268                    continue;
269                }
270            };
271            let data: Value = serde_json::from_str(&text)
272                .map_err(|error| TaskFailure::execution(error.to_string()).at("hosts.json"))?;
273            if response_has_data(&data) {
274                return Ok((url, data));
275            }
276            break;
277        }
278    }
279    Err(TaskFailure::execution(
280        "All URLs failed to return an antenna position list.",
281    ))
282}

284fn response_has_data(data: &Value) -> bool {
285    match data {
286        Value::Null | Value::Bool(false) => false,
287        Value::Bool(true) => true,
288        Value::Number(number) => number.as_f64().is_some_and(|number| number != 0.0),
289        Value::String(text) => !text.is_empty(),
290        Value::Array(items) => !items.is_empty(),
291        Value::Object(items) => !items.is_empty(),
292    }
293}

295fn query_host(agent: &ureq::Agent, url: &str) -> Result<Option<String>, ureq::Error> {
296    let mut response = agent.get(url).call()?;
297    if response.status() != 200 {
298        return Ok(None);
299    }
300    let text = response.body_mut().read_to_string()?;
301    Ok((!text.is_empty()).then_some(text))
302}

304fn timestamp() -> String {
305    let now = OffsetDateTime::now_local().unwrap_or_else(|_| OffsetDateTime::now_utc());
306    format!(
307        "{} {:02}:{:02}:{:02}.{:06}",
308        now.date(),
309        now.hour(),
310        now.minute(),
311        now.second(),
312        now.microsecond(),
313    )
314}

316fn publish(
317    path: &Path,
318    requested_path: &Path,
319    overwrite: bool,
320    document: &Value,
321) -> Result<(), TaskFailure> {
322    let parent = path.parent().unwrap_or_else(|| Path::new("."));
323    let failed = |error: io::Error| {
324        TaskFailure::execution(format!("{}: {error}", path.display())).at("outfile")
325    };
326    let mut stage = NamedTempFile::new_in(parent).map_err(failed)?;
327    serde_json::to_writer(stage.as_file_mut(), document)
328        .map_err(|error| TaskFailure::execution(error.to_string()).at("outfile"))?;
329    stage.as_file().sync_all().map_err(failed)?;
330    if !overwrite {
331        stage.persist_noclobber(path).map_err(|error| {
332            if error.error.kind() == io::ErrorKind::AlreadyExists {
333                exists(requested_path)
334            } else {
335                failed(error.error)
336            }
337        })?;
338    } else if path.is_dir() {
339        if fs::symlink_metadata(path).map_err(failed)?.is_symlink() {
340            return Err(
341                TaskFailure::execution("cannot replace a symlink to a directory").at("outfile"),
342            );
343        }
344        let backup = Builder::new()
345            .prefix(".getantposalma-replaced-")
346            .tempdir_in(parent)
347            .map_err(failed)?;
348        let recovered = backup.path().join("previous");
349        fs::rename(path, &recovered).map_err(failed)?;
350        if let Err(error) = stage.persist_noclobber(path) {
351            if fs::rename(&recovered, path).is_err() {
352                let saved = backup.keep();
353                return Err(TaskFailure::execution(format!(
354                    "{}: {}; prior product left at {}",
355                    path.display(),
356                    error.error,
357                    saved.display()
358                ))
359                .at("outfile"));
360            }
361            return Err(failed(error.error));
362        }
363        backup.close().map_err(failed)?;
364    } else {
365        stage.persist(path).map_err(|error| failed(error.error))?;
366    }
367    Ok(())
