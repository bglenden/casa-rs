56pub(crate) fn centered_transform_f64(

63
64pub(crate) fn centered_transform_f64_owned(
65    mut output: Array2<Complex64>,
66    direction: FftDirection,
67    use_case: FftUseCase,
68) -> Result<(Array2<Complex64>, FftTiming), &'static str> {
69    if output.shape().len() != 2
70        || output.shape()[0] == 0
71        || output.shape()[1] == 0
72        || output.shape()[0] & 1 != 0
73        || output.shape()[1] & 1 != 0
74        || output.shape()[0] > c_int::MAX as usize
75        || output.shape()[1] > c_int::MAX as usize
76    {
77        return Err("fftw_local_requires_non_empty_even_int_range_axes");
78    }
79    let api = api()?;
80    let rows = output.shape()[0];
81    let columns = output.shape()[1];
82    let spec = Fft2Spec::centered_c2c(
83        rows,
84        columns,
85        FftPrecision::F64,
86        direction,
87        use_case,
88        FftBackendChoice::FftwLocalBench,
89    );
90    let selection = select_fft_backend(spec);
91    if !selection.requested_backend_supported {
92        return Err("fftw_local_backend_not_supported");
93    }
94    let mut timing = FftTiming::new(spec, selection);
95    timing.plan_cache_hit = false;
96    let total_started = Instant::now();
97
98    let pack_started = Instant::now();
99    if !shift_quadrants_in_place(&mut output) {
100        return Err("fftw_local_requires_contiguous_even_axes");
101    }
102    timing.pack += pack_started.elapsed();
103
104    let dimensions = [rows as c_int, columns as c_int];
105    let sign = match direction {
106        FftDirection::Forward => FFTW_FORWARD,
107        FftDirection::Inverse => FFTW_BACKWARD,
108    };
109    let plan_started = Instant::now();
110    let output_pointer = output
111        .as_slice_memory_order_mut()
112        .ok_or("fftw_local_requires_contiguous_input")?
113        .as_mut_ptr();
114    let plan = {
115        let _guard = FFTW_PLAN_LOCK
116            .lock()
117            .map_err(|_| "fftw_local_plan_lock_poisoned")?;
118        // SAFETY: the dynamically loaded symbols use FFTW's documented C ABI.
119        // Planning is serialized because FFTW's planner and thread count are
120        // global. The input is contiguous and remains alive until destruction.
121        unsafe {
122            if (api.init_threads)() == 0 {
123                return Err("fftw_local_init_threads_failed");
124            }
125            (api.plan_with_threads)(thread_count());
126            (api.plan_dft)(
127                2,
128                dimensions.as_ptr(),
129                output_pointer,
130                output_pointer,
131                sign,
132                FFTW_ESTIMATE,
133            )
134        }
135    };
136    timing.plan = plan_started.elapsed();
137    if plan.is_null() {
138        return Err("fftw_local_plan_failed");
139    }
140
141    let exec_started = Instant::now();
142    // SAFETY: `plan` was created for `output`, which remains allocated and
143    // unmoved for the duration of execution.
144    unsafe {
145        (api.execute)(plan);
146    }
147    if direction == FftDirection::Inverse {
148        let scale = 1.0 / (rows * columns) as f64;
149        output.mapv_inplace(|value| value * scale);
150    }
151    timing.exec = exec_started.elapsed();
152
153    let pack_started = Instant::now();
154    let shifted = shift_quadrants_in_place(&mut output);
155    debug_assert!(shifted);
156    timing.pack += pack_started.elapsed();
157    timing.total = total_started.elapsed();
158
159    // SAFETY: `plan` is a live FFTW plan and is destroyed exactly once after
160    // its bound execution has completed.
161    unsafe {
162        (api.destroy_plan)(plan);
163    }
164    Ok((output, timing))
165}

167fn api() -> Result<&'static FftwApi, &'static str> {
168    if let Some(api) = FFTW_API.get() {
169        return Ok(api);
170    }
171    let loaded = load_api()?;
172    let _ = FFTW_API.set(loaded);
173    FFTW_API.get().ok_or("fftw_local_api_initialization_race")
174}
175
176fn load_api() -> Result<FftwApi, &'static str> {
177    let (core_path, threads_path) = library_paths().ok_or("set_CASA_RS_FFTW_LIBRARY_DIR")?;
178    // SAFETY: loading is explicit-only from an operator-supplied directory.
179    let core =
180        unsafe { Library::new(core_path) }.map_err(|_| "fftw_local_core_library_load_failed")?;
181    // SAFETY: same explicit local-only experiment boundary as the core library.
182    let threads = unsafe { Library::new(threads_path) }
183        .map_err(|_| "fftw_local_threads_library_load_failed")?;
184
185    // SAFETY: symbol names and signatures match FFTW 3's public C API. The
186    // `Library` owners are retained in `FftwApi` for all copied symbols.
187    unsafe {
188        let init_threads = *threads
189            .get::<FftwInitThreads>(b"fftw_init_threads\0")
190            .map_err(|_| "fftw_local_missing_init_threads")?;
191        let plan_with_threads = *threads
192            .get::<FftwPlanWithThreads>(b"fftw_plan_with_nthreads\0")
193            .map_err(|_| "fftw_local_missing_plan_with_nthreads")?;
194        let plan_dft = *core
195            .get::<FftwPlanDft>(b"fftw_plan_dft\0")
196            .map_err(|_| "fftw_local_missing_plan_dft")?;
197        let execute = *core
198            .get::<FftwExecute>(b"fftw_execute\0")
199            .map_err(|_| "fftw_local_missing_execute")?;
200        let destroy_plan = *core
201            .get::<FftwDestroyPlan>(b"fftw_destroy_plan\0")
202            .map_err(|_| "fftw_local_missing_destroy_plan")?;
203        Ok(FftwApi {
204            _core: core,
205            _threads: threads,
206            init_threads,
207            plan_with_threads,
208            plan_dft,
209            execute,
210            destroy_plan,
211        })
212    }
213}

215fn library_paths() -> Option<(PathBuf, PathBuf)> {
216    let directory = env::var_os("CASA_RS_FFTW_LIBRARY_DIR").map(PathBuf::from)?;
217    #[cfg(target_os = "macos")]
218    let candidates = [
219        ("libfftw3.dylib", "libfftw3_threads.dylib"),
220        ("libfftw3.3.dylib", "libfftw3_threads.3.dylib"),
221    ];
222    #[cfg(all(unix, not(target_os = "macos")))]
223    let candidates = [
224        ("libfftw3.so", "libfftw3_threads.so"),
225        ("libfftw3.so.3", "libfftw3_threads.so.3"),
226    ];
227    #[cfg(not(unix))]
228    return None;
229    candidates
230        .into_iter()
231        .map(|names| (directory.join(names.0), directory.join(names.1)))
232        .find(|(core, threads)| core.is_file() && threads.is_file())
233}

235fn thread_count() -> c_int {
236    env::var("CASA_RS_FFTW_THREADS")
237        .ok()
238        .and_then(|value| value.parse::<c_int>().ok())
239        .filter(|&value| value > 0)
240        .unwrap_or_else(|| {
241            std::thread::available_parallelism()
242                .map(|count| count.get().min(c_int::MAX as usize) as c_int)
243                .unwrap_or(1)
244        })
245}

247fn shift_quadrants_in_place(input: &mut Array2<Complex64>) -> bool {
248    let rows = input.shape()[0];
249    let columns = input.shape()[1];
250    if rows & 1 != 0 || columns & 1 != 0 {
251        return false;
252    }
253    let Some(values) = input.as_slice_memory_order_mut() else {
254        return false;
255    };
256    let half_rows = rows / 2;
257    let half_columns = columns / 2;
258    for row in 0..half_rows {
259        for column in 0..half_columns {
260            let q00 = row * columns + column;
261            let q11 = (row + half_rows) * columns + column + half_columns;
262            values.swap(q00, q11);
263
264            let q10 = (row + half_rows) * columns + column;
265            let q01 = row * columns + column + half_columns;
266            values.swap(q10, q01);
267        }
268    }
269    true
270}
