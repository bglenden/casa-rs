4//! MeasurementSet into compiler input, and write selected visibilities back.

6//! A MeasurementSet is read and written as CASA does. casacore's table locks

35/// Explicit production inputs for resolving one selected MeasurementSet.

140/// Bounded in-place selected-visibility writer following ordinary casacore semantics.
141///
142/// The writer holds the MAIN write lock until it completes. Completion flushes
143/// every selected destination and releases the lock. A failure may leave
144/// partially written derived values, as an interrupted CASA write does; the
145/// next run recomputes them. No backup, staging column, rollback, snapshot,
146/// marker, or content digest is created.
147#[cfg(unix)]
148pub struct SelectedVisibilityWrite {
149    measurement_set: Option<MeasurementSet>,
150    targets: SelectedVisibilityWriteTargets,
151    pending_cell: Option<PendingVisibilityCells>,
152    pending_rows: Vec<usize>,
153    completed: bool,
154}
155
156struct PendingVisibilityCells {
157    row: usize,
158    model_data: Option<ndarray::ArrayD<Complex32>>,
159    corrected_data: Option<ndarray::ArrayD<Complex32>>,
160}
161
162#[cfg(unix)]
163impl SelectedVisibilityWrite {
164    /// Take the MAIN write lock and start an in-place write. `MODEL_DATA` is
165    /// created, zero-filled, when MAIN lacks it, as CASA does.
166    ///
167    /// The MeasurementSet is opened with retained read locks, and MAIN's is
168    /// then upgraded to the write lock. Both wait, as casacore's do, for a
169    /// conflicting lock another process holds. When another process holding
170    /// a read lock waits to upgrade it too, the kernel refuses one of the
171    /// two waits; if it is this one, the write fails with
172    /// [`ObservationOwnerError::WriteLockUnavailable`] and releases its read
173    /// locks.
174    pub fn begin(
175        path: impl AsRef<Path>,
176        targets: SelectedVisibilityWriteTargets,
177    ) -> Result<Self, ObservationOwnerError> {
178        if !targets.model_data && !targets.corrected_data {
179            return Err(ObservationOwnerError::EmptyWriteTargets);
180        }
181        let mut measurement_set = MeasurementSet::open_retained_read(path.as_ref())?;
182        if !measurement_set.main_table_mut().lock(LockType::Write, 0)? {
183            return Err(ObservationOwnerError::WriteLockUnavailable);
184        }
185        let has_model = {
186            let schema = measurement_set.main_table().schema().ok_or_else(|| {
187                MsError::InvalidInput("MeasurementSet MAIN has no schema".to_string())
188            })?;
189            schema.contains_column("MODEL_DATA")
190        };
191        let has_corrected = measurement_set
192            .main_table()
193            .schema()
194            .is_some_and(|schema| schema.contains_column("CORRECTED_DATA"));
195        if targets.corrected_data && !has_corrected {
196            return Err(ObservationOwnerError::MissingCorrectedDataDestination);
197        }
198        if targets.model_data && !has_model {
199            measurement_set.main_table_mut().add_column(
200                ColumnSchema::array_variable("MODEL_DATA", PrimitiveType::Complex32, Some(2)),
201                None,
202            )?;
203            measurement_set
204                .main_table_mut()
205                .prepare_write()
206                .add_tiled_column_clone("DATA", "MODEL_DATA", "TiledModelData")?;
207            let row_count = measurement_set.main_table().row_count();
208            let mut pending_rows = Vec::with_capacity(VISIBILITY_WRITE_BATCH_ROWS as usize);
209            for row in 0..row_count {
210                let data_description_id = main_data_description_id(&measurement_set, row)?;
211                let shape = model_cell_shape(&measurement_set, data_description_id)?;
212                queue_visibility_cell(
213                    measurement_set.main_table_mut(),
214                    "MODEL_DATA",
215                    row,
216                    ArrayValue::Complex32(ndarray::ArrayD::from_elem(
217                        ndarray::IxDyn(&shape),
218                        Complex32::new(0.0, 0.0),
219                    )),
220                )?;
221                pending_rows.push(row);
222                if pending_rows.len() == VISIBILITY_WRITE_BATCH_ROWS as usize {
223                    persist_visibility_rows(
224                        measurement_set.main_table_mut(),
225                        &["MODEL_DATA"],
226                        &pending_rows,
227                    )?;
228                    pending_rows.clear();
229                }
230            }
231            if !pending_rows.is_empty() {
232                persist_visibility_rows(
233                    measurement_set.main_table_mut(),
234                    &["MODEL_DATA"],
235                    &pending_rows,
236                )?;
237            }
238        }
239        Ok(Self {
240            measurement_set: Some(measurement_set),
241            targets,
242            pending_cell: None,
243            pending_rows: Vec::with_capacity(VISIBILITY_WRITE_BATCH_ROWS as usize),
244            completed: false,
245        })
246    }
247
248    /// Write one selected prediction at its physical row/channel/correlation.
249    pub fn write(
250        &mut self,
251        column: MsColumnKind,
252        row: u64,
253        channel: u32,
254        correlation: u32,
255        value: Complex32,
256    ) -> Result<(), ObservationOwnerError> {
257        let row = usize::try_from(row).map_err(|_| ObservationOwnerError::PredictionAddress)?;
258        if self.pending_cell.as_ref().map(|cell| cell.row) != Some(row) {
259            self.flush_pending_cell()?;
260            let measurement_set = self
261                .measurement_set
262                .as_ref()
263                .ok_or(ObservationOwnerError::TransactionClosed)?;
264            let load = |name: &str| -> Result<ndarray::ArrayD<Complex32>, ObservationOwnerError> {
265                let current = measurement_set
266                    .main_table()
267                    .column_accessor(name)?
268                    .get(row)?
269                    .cloned()
270                    .ok_or(ObservationOwnerError::PredictionAddress)?;
271                let Value::Array(ArrayValue::Complex32(values)) = current else {
272                    return Err(ObservationOwnerError::PredictionAddress);
273                };
274                Ok(values)
275            };
276            self.pending_cell = Some(PendingVisibilityCells {
277                row,
278                model_data: self
279                    .targets
280                    .model_data
281                    .then(|| load("MODEL_DATA"))
282                    .transpose()?,
283                corrected_data: self
284                    .targets
285                    .corrected_data
286                    .then(|| load("CORRECTED_DATA"))
287                    .transpose()?,
288            });
289        }
290        let pending = self
291            .pending_cell
292            .as_mut()
293            .ok_or(ObservationOwnerError::TransactionClosed)?;
294        let values = match column {
295            MsColumnKind::ModelData if self.targets.model_data => pending.model_data.as_mut(),
296            MsColumnKind::CorrectedData if self.targets.corrected_data => {
297                pending.corrected_data.as_mut()
298            }
299            _ => None,
300        }
301        .ok_or(ObservationOwnerError::UnselectedWriteTarget)?;
302        let index = [correlation as usize, channel as usize];
303        let Some(cell) = values.get_mut(index) else {
304            return Err(ObservationOwnerError::PredictionAddress);
305        };
306        *cell = value;
307        Ok(())
308    }
309
310    /// Flush the in-place write and release the MAIN write lock.
311    ///
312    /// The selected cells were persisted in bounded row batches, so only the
313    /// table metadata (a newly created `MODEL_DATA`) remains to flush.
314    pub fn complete(mut self) -> Result<(), ObservationOwnerError> {
315        self.flush_pending_cell()?;
316        self.persist_pending_rows()?;
317        let measurement_set = self
318            .measurement_set
319            .as_mut()
320            .ok_or(ObservationOwnerError::TransactionClosed)?;
321        measurement_set.main_table_mut().unlock_metadata_only()?;
322        self.measurement_set = None;
323        self.completed = true;
324        Ok(())
325    }
326
327    fn flush_pending_cell(&mut self) -> Result<(), ObservationOwnerError> {
328        let Some(cell) = self.pending_cell.take() else {
329            return Ok(());
330        };
331        let measurement_set = self
332            .measurement_set
333            .as_mut()
334            .ok_or(ObservationOwnerError::TransactionClosed)?;
335        if let Some(values) = cell.model_data {
336            queue_visibility_cell(
337                measurement_set.main_table_mut(),
338                "MODEL_DATA",
339                cell.row,
340                ArrayValue::Complex32(values),
341            )?;
342        }
343        if let Some(values) = cell.corrected_data {
344            queue_visibility_cell(
345                measurement_set.main_table_mut(),
346                "CORRECTED_DATA",
347                cell.row,
348                ArrayValue::Complex32(values),
349            )?;
350        }
351        self.pending_rows.push(cell.row);
352        if self.pending_rows.len() == VISIBILITY_WRITE_BATCH_ROWS as usize {
353            self.persist_pending_rows()?;
354        }
355        Ok(())
356    }
357
358    fn persist_pending_rows(&mut self) -> Result<(), ObservationOwnerError> {
359        if self.pending_rows.is_empty() {
360            return Ok(());
361        }
362        let columns = match (self.targets.model_data, self.targets.corrected_data) {
363            (true, true) => &["MODEL_DATA", "CORRECTED_DATA"][..],
364            (true, false) => &["MODEL_DATA"][..],
365            (false, true) => &["CORRECTED_DATA"][..],
366            (false, false) => return Err(ObservationOwnerError::EmptyWriteTargets),
367        };
368        let measurement_set = self
369            .measurement_set
370            .as_mut()
371            .ok_or(ObservationOwnerError::TransactionClosed)?;
372        persist_visibility_rows(
373            measurement_set.main_table_mut(),
374            columns,
375            &self.pending_rows,
376        )?;
377        self.pending_rows.clear();
378        Ok(())
379    }
380}
381
382#[cfg(unix)]
383impl Drop for SelectedVisibilityWrite {
384    /// An abandoned write releases the MAIN write lock without flushing the
385    /// cells still queued: the batches persisted so far stay, as after an
386    /// interrupted CASA write, and MAIN is not rewritten.
387    fn drop(&mut self) {
388        if !self.completed {
389            if let Some(measurement_set) = self.measurement_set.as_mut() {
390                let _ = measurement_set.main_table_mut().unlock_metadata_only();
391            }
392            self.measurement_set = None;
393        }
394    }
395}

463/// Resolve a MeasurementSet, as CASA wrote it, into compiler input and access.
464///
465/// The physical schema, selected rows, Measures provider, and resource binding
466/// are all evaluated while one retained MeasurementSet read capability is
467/// alive. The selected data, flag, and weight columns must exist.
468#[cfg(unix)]
469pub fn resolve_selected_observation(
470    request: SelectedObservationResolutionRequest,
471) -> Result<ResolvedSelectedObservation, ObservationOwnerError> {
472    if request.reference_data.iter().any(|(kind, _)| {
473        matches!(
474            kind,
475            ReferenceDataKind::Measures | ReferenceDataKind::Ephemeris
476        )
477    }) {
478        return Err(ObservationOwnerError::MeasuresReferenceIsOwnerSupplied);
479    }
480    let measurement_set = MeasurementSet::open_retained_read(&request.locator)?;
481    let schema = measurement_set.main_table().schema().ok_or_else(|| {
482        MsError::InvalidInput("MeasurementSet MAIN table has no schema".to_string())
483    })?;
484    let visibility_column = match request.visibility {
485        VisibilityColumn::Data => "DATA",
486        VisibilityColumn::CorrectedData => "CORRECTED_DATA",
487        VisibilityColumn::FloatData => "FLOAT_DATA",
488    };
489    let weight_column = match request.weights {
490        WeightColumn::Weight => "WEIGHT",
491        WeightColumn::WeightSpectrum => "WEIGHT_SPECTRUM",
492    };
493    for column in [visibility_column, "FLAG", "FLAG_ROW", weight_column] {
494        if !schema.contains_column(column) {
495            return Err(ObservationOwnerError::MissingColumn { column });
496        }
497    }
498    let corrected_data_present = schema.contains_column("CORRECTED_DATA");
499    let selection = bind_physical_spectral_coordinates(
500        &measurement_set,
501        &request.selection,
502        request.content_budget,
503    )?;
504    let pointing_query_domain =
505        selected_pointing_query_domain(&measurement_set, &selection, request.content_budget)?;
506    let source = ObservationSourceInput::new(
507        ObservationSourceProvenance::new(request.locator, request.selection_request),
508        selection,
509        SelectedColumns::new(
510            request.visibility,
511            FlagPolicy::FlagOrFlagRow,
512            request.weights,
513        ),
514        corrected_data_present,
515    );
516    let measures = SelectedObservationMeasures::new(request.measures_provider)?;
517    let mut reference_data = request.reference_data;
518    reference_data.push((ReferenceDataKind::Measures, measures.identity()));
519    if let Some(ephemeris) = request.ephemeris.as_ref() {
520        reference_data.push((ReferenceDataKind::Ephemeris, ephemeris.identity()));
521    }
522    let snapshot_input = ObservationSnapshotInput::new(vec![source], reference_data, request.model);
523    // The one resolved source is the snapshot's first.
524    let binding = ObservationSourceBinding::new(0, request.content_budget)
525        .with_ephemeris(request.ephemeris)
526        .with_pointing_query_domain(pointing_query_domain);
527    Ok(ResolvedSelectedObservation {
528        snapshot_input,
529        access: ResolvedSelectedObservationAccess { binding, measures },
530    })
531}

533fn bind_physical_spectral_coordinates(
534    measurement_set: &MeasurementSet,
535    selection: &ObservationSelection,
536    content_budget: SelectedObservationContentBudget,
537) -> Result<ObservationSelection, ObservationOwnerError> {
538    let spectral_window = measurement_set.spectral_window()?;
539    let mut retained_catalog_bytes = 0usize;
540    let mut spectral_windows = Vec::with_capacity(selection.spectral_windows().len());
541    for selected in selection.spectral_windows() {
542        let row = usize::try_from(selected.spectral_window_id())
543            .map_err(|_| ObservationOwnerError::PhysicalSelectionMismatch)?;
544        let frequency_shape = spectral_window
545            .table()
546            .array_shape(row, "CHAN_FREQ")?
547            .filter(|shape| shape.len() == 1 && shape[0] > 0)
548            .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
549        let width_shape = spectral_window
550            .table()
551            .array_shape(row, "CHAN_WIDTH")?
552            .filter(|shape| shape.len() == 1 && shape[0] > 0)
553            .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
554        if frequency_shape != width_shape {
555            return Err(ObservationOwnerError::PhysicalSelectionMismatch);
556        }
557        let values = frequency_shape[0];
558        let simultaneous_array_bytes = values
559            .checked_mul(2)
560            .and_then(|values| values.checked_mul(std::mem::size_of::<f64>()))
561            .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
562        let catalog_bytes = values
563            .checked_mul(std::mem::size_of::<f64>())
564            .and_then(|bytes| bytes.checked_add(2 * std::mem::size_of::<usize>()))
565            .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
566        // Arc<[f64]> construction briefly overlaps the source Vec with the
567        // destination allocation. CHAN_WIDTH is dropped first, so both the
568        // two-array read peak and the Arc conversion peak are exactly covered.
569        let construction_peak_bytes = retained_catalog_bytes
570            .checked_add(simultaneous_array_bytes)
571            .and_then(|bytes| bytes.checked_add(2 * std::mem::size_of::<usize>()))
572            .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
573        retained_catalog_bytes = retained_catalog_bytes
574            .checked_add(catalog_bytes)
575            .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
576        let required_bytes = construction_peak_bytes.max(retained_catalog_bytes);
577        if required_bytes > content_budget.available_bytes() {
578            return Err(ObservationOwnerError::SpectralCoordinateCatalogBudget {
579                required_bytes,
580                available_bytes: content_budget.available_bytes(),
581            });
582        }
583        let frequencies_hz = spectral_window.chan_freq(row)?;
584        let widths_hz = spectral_window.chan_width(row)?;
585        if frequencies_hz.len() != values || widths_hz.len() != values {
586            return Err(ObservationOwnerError::PhysicalSelectionMismatch);
587        }
588        let first_channel_width_hz = widths_hz[0];
589        drop(widths_hz);
590        let coordinate_catalog =
591            SpectralWindowCoordinateCatalog::new(frequencies_hz, first_channel_width_hz)
592                .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)?;
593        spectral_windows.push(
594            SpectralWindowSelection::new(
595                selected.spectral_window_id(),
596                selected.channel_indices().to_vec(),
597            )
598            .with_coordinate_catalog(coordinate_catalog),
599        );
600    }
601    Ok(ObservationSelection::new(
602        selection.rows().clone(),
603        selection.rows_filter().clone(),
604        selection.data_descriptions().to_vec(),
605        spectral_windows,
606        selection.correlations().to_vec(),
607    ))
608}

610/// Failure to resolve a selected MeasurementSet or write its visibilities.
611#[derive(Debug, Error)]
612pub enum ObservationOwnerError {
613    /// MeasurementSet or subtable access failed.
614    #[error(transparent)]
615    MeasurementSet(#[from] MsError),
616    /// MAIN-table locking or persistence failed.
617    #[error(transparent)]
618    Table(#[from] casa_tables::TableError),
619    /// Measures acquisition or bounded-state preparation failed.
620    #[error(transparent)]
621    Measures(#[from] SelectedObservationMeasuresError),
622    /// The compiled physical coordinates do not match the retained MS.
623    #[error(transparent)]
624    SelectedCoordinates(#[from] BoundObservationSourceError),
625    /// A selected data, flag, or weight column is not in MAIN.
626    #[error("the MeasurementSet has no {column} column")]
627    MissingColumn {
628        /// Standard MAIN column name.
629        column: &'static str,
630    },
631    /// The MeasurementSet cannot realize the requested selection: a selected
632    /// SPECTRAL_WINDOW row is missing or its CHAN_FREQ and CHAN_WIDTH are
633    /// empty or of unequal length, or no MAIN row is selected.
634    #[error(
635        "the MeasurementSet's spectral windows or MAIN rows do not realize the observation selection"
636    )]
637    PhysicalSelectionMismatch,
638    /// The complete physical SPW coordinate catalog cannot fit its explicit source budget.
639    #[error(
640        "spectral coordinate catalog requires {required_bytes} bytes but the content budget has {available_bytes} bytes"
641    )]

678/// The antennas and times of the selected rows, the domain the POINTING
679/// catalog is prepared for.
680fn selected_pointing_query_domain(
681    measurement_set: &MeasurementSet,
682    selection: &ObservationSelection,
683    content_budget: SelectedObservationContentBudget,
684) -> Result<SelectedPointingQueryDomain, ObservationOwnerError> {
685    let row_selection = SelectedObservationRowSelection::from_compiled(selection);
686    let mut pointing_query_domain = SelectedPointingQueryDomain::builder();
687    let mut pointing_domain_error = None;
688    measurement_set.visit_selected_observation_rows(
689        &row_selection,
690        content_budget.row_io_budget(),
691        |row| {
692            if pointing_domain_error.is_none() {
693                pointing_domain_error = pointing_query_domain
694                    .observe_row(
695                        row.antenna1(),
696                        row.antenna2(),
697                        row.time_mjd_seconds(),
698                        row.time_centroid_mjd_seconds(),
699                    )
700                    .err();
701            }
702        },
703    )?;
704    if let Some(error) = pointing_domain_error {
705        return Err(error.into());
706    }
707    pointing_query_domain
708        .finish()?
709        .ok_or(ObservationOwnerError::PhysicalSelectionMismatch)
710}

737    measurement_set: &MeasurementSet,
738    data_description_id: usize,
739) -> Result<[usize; 2], ObservationOwnerError> {
740    let data_description = measurement_set.data_description()?;
741    let spectral_window_id =
742        usize::try_from(data_description.spectral_window_id(data_description_id)?)
743            .map_err(|_| ObservationOwnerError::PredictionAddress)?;
744    let polarization_id = usize::try_from(data_description.polarization_id(data_description_id)?)
745        .map_err(|_| ObservationOwnerError::PredictionAddress)?;
746    let channels = usize::try_from(
747        measurement_set

749            .num_chan(spectral_window_id)?,

753        .map_err(|_| ObservationOwnerError::PredictionAddress)?;
754    Ok([correlations, channels])
755}
