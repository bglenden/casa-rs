12pub(super) struct Ionosphere {
13    height_m: f64,
14    field_cache: Mutex<HashMap<(u64, i32, i32), f64>>,
15}
16
17impl Ionosphere {
18    pub(super) fn for_plan(
19        table: &Table,
20        plan: &ApplyCalibrationTablePlan,
21    ) -> Result<Option<Self>, ApplyExecutionError> {
22        if plan.summary.table_subtype != "F Jones" {
23            return Ok(None);
24        }
25        let path = &plan.spec.path;
26        // FJones::setApply reads the keyword. CLPatchPanel setup calls the
27        // base setCallib path and keeps the default magnetic shell height.
28        let keyword = if plan.spec.callibrary.is_none() {
29            table.keywords().get(HEIGHT_KEYWORD)
30        } else {
31            None
32        };
33        let height = match keyword {
34            Some(Value::Scalar(ScalarValue::Float64(height))) => *height,
35            Some(Value::Scalar(ScalarValue::Float32(height))) => f64::from(*height),
36            None => DEFAULT_HEIGHT_KM,
37            _ => {
38                return Err(invalid(
39                    path,
40                    "F Jones ionosphere height is not a real scalar",
41                ));
42            }
43        };
44        if !height.is_finite() {
45            return Err(invalid(path, "F Jones ionosphere height must be finite"));
46        }
47        Ok(Some(Self {
48            height_m: 1000. * height,
49            field_cache: Mutex::default(),
50        }))
51    }
52
53    fn magnetic_field(
54        &self,
55        row: &ApplyRowPlan,
56        antenna: i32,
57        context: &CalibrationSamplingContext<'_>,
58    ) -> Result<f64, ApplyExecutionError> {
59        let key = (row.time_seconds.to_bits(), row.field_id, antenna);
60        let cache = || {
61            self.field_cache
62                .lock()
63                .unwrap_or_else(|error| error.into_inner())
64        };
65        if let Some(value) = cache().get(&key) {
66            return Ok(*value);
67        }
68        let engine = context.engine.ok_or_else(|| {
69            invalid(
70                context.path,
71                "F Jones apply requires MeasurementSet geometry",
72            )
73        })?;
74        let failed = |source| ApplyExecutionError::MutateMeasurementSet {
75            path: context.path.display().to_string(),
76            source,
77        };
78        let direction = engine
79            .reference_phase_direction(row.field_id as usize, row.time_seconds)
80            .map_err(failed)?;
81        let frame = engine
82            .antenna_frame(row.time_seconds, antenna as usize)
83            .map_err(failed)?
84            .with_epoch(MEpoch::from_mjd_seconds(row.time_seconds, EpochRef::UTC));
85        let sample = calculate_igrf_at_shell_offset(self.height_m, &direction, &frame)
86            .map_err(|error| failed(error.into()))?;
87        let field_gauss = sample.los_field_nt * 1e-5;
88        let mut cache = cache();
89        if cache.len() >= 8192 {
90            cache.clear();
91        }
92        cache.insert(key, field_gauss);
93        Ok(field_gauss)
94    }

96    pub(super) fn apply(
97        &self,
98        row: &ApplyRowPlan,
99        grids: [&CalibrationGrid; 2],
100        context: &CalibrationSamplingContext<'_>,
101        output: &mut ApplyBuffer<'_>,
102        forward: bool,
103    ) -> Result<(), ApplyExecutionError> {
104        let (left, left_flag) = electrons(grids[0], context.path)?;
105        let (right, right_flag) = electrons(grids[1], context.path)?;
106        let values = [left, right];
107        let missing = [left_flag, right_flag];
108        let mut rotation = [0.; 2];
109        for (side, antenna) in [row.antenna1, row.antenna2].into_iter().enumerate() {
110            if !missing[side] {
111                rotation[side] =
112                    2.3649 * values[side] * self.magnetic_field(row, antenna, context)?;
113            }
114        }
115        let circular = output
116            .correlation_types
117            .iter()
118            .all(|code| matches!(code, 5..=8));
119        let linear_order = (!circular)
120            .then(|| polarization::coherency_order(row.data_desc_id, output.correlation_types))
121            .transpose()?;
122        let sign = if forward { 1. } else { -1. };
123        let write_flags = output.propagate_flags;
124        for (channel, &frequency) in context
125            .data_channels
126            .frequencies_hz
127            .iter()
128            .enumerate()
129            .take(output.corrected.shape()[1])
130        {
131            let delay = values.map(|tec| -8.4483e-7 * tec / frequency);
132            let angle = rotation.map(|term| term / frequency / frequency);
133            let mut invalid = missing.into_iter().any(|flag| flag);
134            if let Some(order) = linear_order {
135                let dispersion = unit_phase(sign * (delay[0] - delay[1]));
136                let rotations = angle.map(|r| (sign * r) as f32).map(f32::sin_cos);
137                let samples = order.map(|index| output.corrected[[index, channel]]);
138                for (index, value) in order
139                    .into_iter()
140                    .zip(parang::rotate_linear(samples, rotations))
141                {
142                    invalid |= write_flags && output.flags[[index, channel]];
143                    output.corrected[[index, channel]] = dispersion * value;
144                }
145            } else {
146                let phases = std::array::from_fn::<_, 2, _>(|side| {
147                    [delay[side] + angle[side], delay[side] - angle[side]]
148                });
149                for (index, &code) in output.correlation_types.iter().enumerate() {
150                    let (a, b) = correlation_receptors(code).expect("circular correlation");
151                    output.corrected[[index, channel]] *=
152                        unit_phase(sign * (phases[0][a] - phases[1][b]));
153                }
154            }
155            if invalid && write_flags {
156                for index in 0..output.correlation_types.len() {
157                    if !output.flags[[index, channel]] {
158                        output.flags[[index, channel]] = true;
159                        *output.newly_flagged_samples += 1;
160                    }
161                }
162            }
163        }
164        Ok(())
165    }
166}

168fn electrons(grid: &CalibrationGrid, path: &Path) -> Result<(f64, bool), ApplyExecutionError> {
169    let CalibrationGrid::Complex(grid) = grid else {
170        return Err(invalid(path, "F Jones requires scalar FPARAM rows"));
171    };
172    let Some(gain) = grid.values.first() else {
173        return Err(invalid(
174            path,
175            "F Jones requires one line-of-sight TEC value",
176        ));
177    };
178    let invalid = grid.flags.first().copied().unwrap_or(true) || !gain.re.is_finite();
179    Ok((if invalid { 0. } else { f64::from(gain.re) }, invalid))
180}
181
182fn unit_phase(angle: f64) -> Complex32 {
183    let (sin, cos) = angle.sin_cos();
184    Complex32::new(cos as f32, sin as f32)
185}
186
187fn invalid(path: &Path, message: &str) -> ApplyExecutionError {
188    ApplyExecutionError::UnsupportedCalibrationTable {
189        path: path.display().to_string(),
190        reason: message.to_string(),
191    }
