7#[derive(Debug, Clone)]
8pub(crate) struct Selection {
9    pub(super) start: Vec<usize>,
10    pub(super) shape: Vec<usize>,
11    /// The spectral axis its named channels belong to.
12    pub(super) channel_axis: Option<usize>,
13    pub(super) channel_indices: Option<Vec<usize>>,
14    /// The Stokes axis its named planes belong to.
15    pub(super) stokes_axis: Option<usize>,
16    /// The image's Stokes-axis indices the selection holds, when a Stokes
17    /// selection named them.
18    pub(super) stokes_indices: Option<Vec<usize>>,
19    /// The image axes a box or polygon's x and y run along: the direction
20    /// plane's.
21    pub(super) plane: [usize; 2],
22}
23
24impl Selection {
25    pub(super) fn new<T: ImagePixel>(
26        image: &impl ImageInterface<T>,
27        region: Option<&PixelRegion>,
28        chans: Option<&str>,
29        stokes: Option<&str>,
30    ) -> Result<Self, ImageError> {
31        let mut start = vec![0; image.ndim()];
32        let mut shape = image.shape().to_vec();
33        let plane = region.map_or(Ok([0, 1]), |_| direction_plane_axes(image))?;
34        if let Some(region) = region {
35            let [x0, y0, x1, y1] = region.bounding_box();
36            let [x, y] = plane;
37            if image.ndim() < 2 || x1 >= shape[x] || y1 >= shape[y] {
38                return Err(ImageError::ShapeMismatch {
39                    expected: image.shape().to_vec(),
40                    got: vec![x0, y0, x1, y1],
41                });
42            }
43            start[x] = x0;
44            start[y] = y0;
45            shape[x] = x1 - x0 + 1;
46            shape[y] = y1 - y0 + 1;
47        }
48        let (channel_axis, channel_indices) =
49            if let Some(chans) = chans.filter(|text| !text.trim().is_empty()) {
50                let spectral_axis = image.find_axis(CoordinateType::Spectral).ok_or_else(|| {
51                    ImageError::InvalidMetadata(
52                        "channel selection requires a spectral axis".to_string(),
53                    )
54                })?;
55                let indices = parse_image_channel_selection(chans, image.shape()[spectral_axis])?;
56                start[spectral_axis] = indices[0];
57                shape[spectral_axis] = indices[indices.len() - 1] - indices[0] + 1;
58                (Some(spectral_axis), Some(indices))
59            } else {
60                (None, None)
61            };
62        let (stokes_axis, stokes_indices) =
63            if let Some(text) = stokes.filter(|text| !text.trim().is_empty()) {
64                let indices = parse_stokes_selection(image, text)?;
65                let axis = image
66                    .find_axis(CoordinateType::Stokes)
67                    .expect("parse_stokes_selection found the axis");
68                start[axis] = indices[0];
69                shape[axis] = indices[indices.len() - 1] - indices[0] + 1;
70                (Some(axis), Some(indices))
71            } else {
72                (None, None)
73            };
74        Ok(Self {
75            start,
76            shape,
77            channel_axis,
78            channel_indices,
79            stokes_axis,
80            stokes_indices,
81            plane,
82        })
83    }
84
85    /// The image-axis indices this selection holds along `axis`, in
86    /// order: a named set's, or the bounding rectangle's span.
87    pub(super) fn axis_positions(&self, axis: usize) -> Vec<usize> {
88        if Some(axis) == self.channel_axis
89            && let Some(indices) = &self.channel_indices
90        {
91            return indices.clone();
92        }
93        if Some(axis) == self.stokes_axis
94            && let Some(indices) = &self.stokes_indices
95        {
96            return indices.clone();
97        }
98        (self.start[axis]..self.start[axis] + self.shape[axis]).collect()
99    }
100
101    /// Whether the named Stokes set is not a contiguous span, which the
102    /// streaming classical path cannot express.
103    pub(super) fn stokes_has_gaps(&self) -> bool {
104        Self::named_set_has_gaps(self.stokes_axis, &self.stokes_indices)
105    }
106
107    /// Whether the named channel set is not a contiguous span, which the
108    /// streaming classical path cannot express.
109    pub(super) fn channels_have_gaps(&self) -> bool {
110        Self::named_set_has_gaps(self.channel_axis, &self.channel_indices)
111    }
112
113    /// Whether a named selection set skips indices between its first and
114    /// last.
115    fn named_set_has_gaps(axis: Option<usize>, indices: &Option<Vec<usize>>) -> bool {
116        match (axis, indices) {
117            (Some(_), Some(indices)) => {
118                indices.len() > 1 && indices[indices.len() - 1] - indices[0] + 1 != indices.len()
119            }
120            _ => false,
121        }
122    }
123}

125/// Parse CASA's `stokes` selection: comma-separated Stokes plane names,
126/// concatenated as in `QU` or `IQUV`. Numeric forms are not names.
127pub(crate) fn parse_stokes_selection<T: ImagePixel>(
128    image: &(impl ImageInterface<T> + ?Sized),
129    text: &str,
130) -> Result<Vec<usize>, ImageError> {
131    let axis = image.find_axis(CoordinateType::Stokes).ok_or_else(|| {
132        ImageError::InvalidMetadata("Stokes selection requires a Stokes axis".to_string())
133    })?;
134    parse_stokes_names(image.coordinates(), axis, text)
135}
136
137pub(super) fn parse_stokes_names(
138    coordinates: &CoordinateSystem,
139    axis: usize,
140    text: &str,
141) -> Result<Vec<usize>, ImageError> {
142    let Some(CoordinateModel::Stokes(stokes)) = coordinate_for_axis(coordinates, axis) else {
143        return Err(ImageError::InvalidMetadata(
144            "Stokes selection requires a Stokes coordinate".to_string(),
145        ));
146    };
147    let mut indices = Vec::new();
148    for part in text.split(',') {
149        let part = part.trim();
150        if part.is_empty() {
151            continue;
152        }
153        let mut rest = part;
154        while !rest.is_empty() {
155            let name = if rest.len() >= 2 && StokesType::from_name(&rest[..2]).is_some() {
156                rest[..2].to_string()
157            } else if let Some(kind) = rest
158                .chars()
159                .next()
160                .and_then(|first| StokesType::from_name(&first.to_string()))
161            {
162                let _ = kind;
163                rest[..1].to_string()
164            } else {
165                return Err(ImageError::CasaSelection(format!(
166                    "(Sub)String {part} in stokes specification part {part} does not match a known polarization"
167                )));
168            };
169            let kind =
170                StokesType::from_name(&name).expect("the name was matched against the table");
171            let Some(index) = stokes.stokes().iter().position(|known| *known == kind) else {
172                return Err(ImageError::CasaSelection(format!(
173                    "Stokes {name} is not one of this image's Stokes planes"
174                )));
175            };
176            indices.push(index);
177            rest = &rest[name.len()..];
178        }
179    }
180    if indices.is_empty() {
181        return Err(ImageError::InvalidMetadata(
182            "the Stokes selection names no plane".to_string(),
183        ));
184    }
185    indices.sort_unstable();
186    indices.dedup();
187    Ok(indices)
188}

190/// The `axes` selection: all axes, or the named axes to collapse.
191#[derive(Debug, Clone, PartialEq, Eq)]
192pub(super) enum AxisSelection {
193    /// Every axis: the statistics of the whole selection.
194    All,
195    /// Collapse these pixel axes; the statistics vary along the rest.
196    Collapsed(Vec<usize>),
197}
198
199/// Parse CASA's `axes` argument: a pixel-axis list, `-1` for all of them,
200/// or an empty list for the default. A negative axis other than a lone
201/// `-1` is invalid, with CASA's wording.
202pub(super) fn parse_axes(axes: Option<&[i64]>, n_dim: usize) -> Result<AxisSelection, String> {
203    let Some(axes) = axes else {
204        return Ok(AxisSelection::All);
205    };
206    if axes.is_empty() {
207        return Ok(AxisSelection::All);
208    }
209    if axes == [-1] {
210        return Ok(AxisSelection::All);
211    }
212    let mut named = Vec::with_capacity(axes.len());
213    for &axis in axes {
214        if !(0..n_dim as i64).contains(&axis) {
215            return Err(format!("Invalid cursor axes: {axes:?}."));
216        }
217        if named.contains(&(axis as usize)) {
218            return Err(format!(
219                "Invalid cursor axes: {axes:?}: an axis is named twice; CASA 6.7.6.14 aborts on it"
220            ));
221        }
222        named.push(axis as usize);
223    }
224    named.sort_unstable();
225    Ok(AxisSelection::Collapsed(named))
226}

228#[derive(Clone)]
229pub(crate) enum PixelRegion {
230    /// One or more rectangular boxes: CASA's task `box` parameter takes
231    /// comma-separated quadruplets, and CRTF strings name one box.
232    Boxes(Vec<[usize; 4]>),
233    Polygon {
234        bounds: [usize; 4],
235        vertices: Vec<[f64; 2]>,
236    },
237}
238
239impl PixelRegion {
240    pub(super) fn bounding_box(&self) -> [usize; 4] {
241        match self {
242            Self::Boxes(boxes) => {
243                let min_x = boxes.iter().map(|b| b[0]).min().unwrap_or(0);
244                let min_y = boxes.iter().map(|b| b[1]).min().unwrap_or(0);
245                let max_x = boxes.iter().map(|b| b[2]).max().unwrap_or(min_x);
246                let max_y = boxes.iter().map(|b| b[3]).max().unwrap_or(min_y);
247                [min_x, min_y, max_x, max_y]
248            }
249            Self::Polygon { bounds, .. } => *bounds,
250        }
251    }
252}

254pub(crate) fn region_pixels<T: ImagePixel>(
255    image: &impl ImageInterface<T>,
256    region: Option<&str>,
257) -> Result<Option<PixelRegion>, ImageError> {
258    let Some(region) = region.map(str::trim).filter(|text| !text.is_empty()) else {
259        return Ok(None);
260    };
261    if let Some(parsed) = crtf_box_pixels(image, region)? {
262        return Ok(Some(PixelRegion::Boxes(vec![parse_box(&parsed)?])));
263    }
264    if let Some(polygon) = crtf_polygon_region(image, region)? {
265        return Ok(Some(polygon));
266    }
267    if let Some(box_text) = region.strip_prefix("box:") {
268        let parsed = parse_box(box_text.trim())?;
269        return Ok(Some(PixelRegion::Boxes(vec![parsed])));
270    }
271    if let Some(contents) = region
272        .strip_prefix("pixelbox(")
273        .and_then(|text| text.strip_suffix(')'))
274    {
275        let parsed = parse_box(contents.trim())?;
276        return Ok(Some(PixelRegion::Boxes(vec![parsed])));
277    }
278    if looks_like_box(region) {
279        let parsed = parse_box(region)?;
280        return Ok(Some(PixelRegion::Boxes(vec![parsed])));
281    }
282
283    let region_path = Path::new(region);
284    let contents = fs::read_to_string(region_path).map_err(|error| {
285        ImageError::Io(format!(
286            "read region file {}: {error}",
287            region_path.display()
288        ))
289    })?;
290    if let Some(box_pixels) = crtf_box_pixels(image, &contents)? {
291        return Ok(Some(PixelRegion::Boxes(vec![parse_box(&box_pixels)?])));
292    }
293    if let Some(polygon) = crtf_polygon_region(image, &contents)? {
294        return Ok(Some(polygon));
295    }
296    let format = if contents.trim_start().starts_with('{') {
297        "casa-rs JSON region files are not a persistent interchange format; use CASA CRTF"
298    } else {
299        "expected CASA CRTF box or polygon, for example box[[100pix,100pix],[150pix,150pix]] or poly [[100pix,100pix],...]"
300    };
301    Err(ImageError::InvalidMetadata(format!(
302        "unsupported region file {}: {format}",
303        region_path.display()
304    )))
305}

307pub(crate) fn region_pixel_mask(
308    region: &PixelRegion,
309    selection: &Selection,
310) -> Result<ArrayD<bool>, ImageError> {
311    let mask = match region {
312        PixelRegion::Boxes(boxes) => {
313            if boxes.len() < 2 {
314                // One box is the selection itself; every pixel is in it.
315                ArrayD::from_elem(IxDyn(&selection.shape).f(), true)
316            } else {
317                // Several boxes: the selection is their bounding box, and
318                // a pixel is in the region when some box holds it.
319                let [x_axis, y_axis] = selection.plane;
320                ArrayD::from_shape_fn(IxDyn(&selection.shape).f(), |idx| {
321                    let x = selection.start[x_axis] + idx[x_axis];
322                    let y = selection.start[y_axis] + idx[y_axis];
323                    boxes
324                        .iter()
325                        .any(|b| b[0] <= x && x <= b[2] && b[1] <= y && y <= b[3])
326                })
327            }
328        }
329        PixelRegion::Polygon { vertices, .. } => {
330            let [x_axis, y_axis] = selection.plane;
331            ArrayD::from_shape_fn(IxDyn(&selection.shape).f(), |idx| {
332                let x = selection.start[x_axis] + idx[x_axis];
333                let y = selection.start[y_axis] + idx[y_axis];
334                crtf_polygon_contains(vertices, x, y)
335            })
336        }
337    };
338    Ok(mask)
339}

341pub(super) fn crtf_box_pixels<T: ImagePixel>(
342    image: &impl ImageInterface<T>,
343    text: &str,
344) -> Result<Option<String>, ImageError> {
345    let Some(parts) = crtf_box_parts(text)? else {
346        return Ok(None);
347    };
348    let pixels = parts
349        .iter()
350        .map(|part| parse_crtf_pixel_quantity(part))
351        .collect::<Result<Vec<_>, ImageError>>()?;
352    if pixels.iter().all(Option::is_some) {
353        let values = pixels.into_iter().flatten().collect::<Vec<_>>();
354        return Ok(Some(format!(
355            "{},{},{},{}",
356            values[0], values[1], values[2], values[3]
357        )));
358    }
359    if pixels.iter().any(Option::is_some) {
360        return Err(ImageError::InvalidMetadata(
361            "cannot mix pix and non-pix coordinates in a CRTF box region".to_string(),
362        ));
363    }
364
365    let world = parts
366        .iter()
367        .map(|part| parse_crtf_world_quantity(part))
368        .collect::<Result<Vec<_>, ImageError>>()?;
369    Ok(Some(crtf_world_box_pixels(
370        image,
371        [world[0], world[1], world[2], world[3]],
372    )?))
373}

375fn crtf_polygon_region<T: ImagePixel>(
376    image: &impl ImageInterface<T>,
377    text: &str,
378) -> Result<Option<PixelRegion>, ImageError> {
379    let Some(pairs) = crtf_polygon_parts(text)? else {
380        return Ok(None);
381    };
382    if pairs.len() < 3 {
383        return Err(ImageError::InvalidMetadata(format!(
384            "invalid CASA CRTF polygon region {text:?}: expected at least three vertices"
385        )));
386    }
387    // Evaluate the original fractional vertices at integer pixel centres,
388    // and report the bounds of the pixels the polygon actually selects.
389    let vertices = crtf_coordinate_pairs_to_pixels(image, &pairs)?;
390    let bounds = crtf_polygon_bounds(image, &vertices)?;
391    Ok(Some(PixelRegion::Polygon { bounds, vertices }))
392}

394fn crtf_polygon_bounds<T: ImagePixel>(
395    image: &impl ImageInterface<T>,
396    vertices: &[[f64; 2]],
397) -> Result<[usize; 4], ImageError> {
398    let ranges = vertices.iter().fold(
399        [
400            f64::INFINITY,
401            f64::INFINITY,
402            f64::NEG_INFINITY,
403            f64::NEG_INFINITY,
404        ],
405        |[min_x, min_y, max_x, max_y], &[x, y]| {
406            [min_x.min(x), min_y.min(y), max_x.max(x), max_y.max(y)]
407        },
408    );
409    let shape = image.shape();
410    let [x, y] = direction_plane_axes(image)?;
411    let [width, height] = match [x, y].map(|axis| shape.get(axis).copied()) {
412        [Some(width), Some(height)] if x != y && width > 0 && height > 0 => [width, height],
413        _ => {
414            return Err(ImageError::ShapeMismatch {
415                expected: shape.to_vec(),
416                got: ranges.map(|value| value.round() as usize).to_vec(),
417            });
418        }
419    };
420    let [min_x, min_y, max_x, max_y] = ranges;
421    let xs = min_x.floor() as usize..=(max_x.ceil() as usize).min(width - 1);
422    let ys = min_y.floor() as usize..=(max_y.ceil() as usize).min(height - 1);
423    let no_centres = || {
424        ImageError::InvalidMetadata("CRTF polygon contains no pixel centres in the image".into())
425    };
426    let [x0, x1] = occupied_polygon_axis(xs.clone(), &ys, |x, y| {
427        crtf_polygon_contains(vertices, x, y)
428    })
429    .ok_or_else(no_centres)?;
430    let [y0, y1] =
431        occupied_polygon_axis(ys, &(x0..=x1), |y, x| crtf_polygon_contains(vertices, x, y))
432            .ok_or_else(no_centres)?;
433    Ok([x0, y0, x1, y1])
434}

436/// Inspect only as far into each end of an axis as needed to find an
437/// occupied row or column. Interior pixels are evaluated when masking.
438fn occupied_polygon_axis(
439    axis: RangeInclusive<usize>,
440    cross_axis: &RangeInclusive<usize>,
441    contains: impl Fn(usize, usize) -> bool,
442) -> Option<[usize; 2]> {
443    let mut occupied = axis.filter(|&index| cross_axis.clone().any(|cross| contains(index, cross)));
444    let first = occupied.next()?;
445    Some([first, occupied.next_back().unwrap_or(first)])
446}

448pub(super) fn crtf_polygon_contains(vertices: &[[f64; 2]], x: usize, y: usize) -> bool {
449    // Allow projection roundoff in pixel units. The width of the boundary
450    // stays fixed even when an edge spans thousands of pixels.
451    const EDGE_TOLERANCE_PIXELS: f64 = 1.0e-6;
452    let (x, y) = (x as f64, y as f64);
453    let mut inside = false;
454    let mut previous = vertices[vertices.len() - 1];
455    for &[x1, y1] in vertices {
456        let [x0, y0] = previous;
457        if (x0.min(x1) - EDGE_TOLERANCE_PIXELS..=x0.max(x1) + EDGE_TOLERANCE_PIXELS).contains(&x)
458            && (y0.min(y1) - EDGE_TOLERANCE_PIXELS..=y0.max(y1) + EDGE_TOLERANCE_PIXELS)
459                .contains(&y)
460        {
461            let dx = x1 - x0;
462            let dy = y1 - y0;
463            let cross = (x - x0) * dy - (y - y0) * dx;
464            if cross.abs() <= EDGE_TOLERANCE_PIXELS * dx.hypot(dy) {
465                return true;
466            }
467        }
468        if (y0 > y) != (y1 > y) && x <= (x1 - x0) * (y - y0) / (y1 - y0) + x0 {
469            inside = !inside;
470        }
471        previous = [x1, y1];
472    }
473    inside
474}

476pub(super) fn crtf_polygon_parts(text: &str) -> Result<Option<Vec<[String; 2]>>, ImageError> {
477    let normalized = text.trim();
478    let Some(after) = find_crtf_body(normalized, "poly", "[") else {
479        return Ok(None);
480    };
481    let end = after.rfind(']').ok_or_else(|| {
482        ImageError::InvalidMetadata(format!(
483            "invalid CASA CRTF polygon region {text:?}: missing closing ]"
484        ))
485    })?;
486    let body = &after[..=end];
487    let body = body
488        .strip_prefix('[')
489        .and_then(|body| body.strip_suffix(']'))
490        .unwrap_or(body);
491    let mut pairs = Vec::new();
492    let mut cursor = 0usize;
493    while let Some(relative_start) = body[cursor..].find('[') {
494        let pair_start = cursor + relative_start + 1;
495        let pair_end = body[pair_start..].find(']').ok_or_else(|| {
496            ImageError::InvalidMetadata(format!(
497                "invalid CASA CRTF polygon region {text:?}: missing vertex closing ]"
498            ))
499        })? + pair_start;
500        let pair = body[pair_start..pair_end]
501            .split(',')
502            .map(|part| part.trim().to_string())
503            .filter(|part| !part.is_empty())
504            .collect::<Vec<_>>();
505        if pair.len() == 2 {
506            pairs.push([pair[0].clone(), pair[1].clone()]);
507        }
508        cursor = pair_end + 1;
509    }
510    if pairs.is_empty() {
511        return Err(ImageError::InvalidMetadata(format!(
512            "invalid CASA CRTF polygon region {text:?}: expected poly [[x0,y0],...]"
513        )));
514    }
515    Ok(Some(pairs))
516}

518fn crtf_coordinate_pairs_to_pixels<T: ImagePixel>(
519    image: &impl ImageInterface<T>,
520    pairs: &[[String; 2]],
521) -> Result<Vec<[f64; 2]>, ImageError> {
522    let pixel_pairs = pairs
523        .iter()
524        .map(|pair| {
525            Ok([
526                parse_crtf_pixel_coordinate(&pair[0])?,
527                parse_crtf_pixel_coordinate(&pair[1])?,
528            ])
529        })
530        .collect::<Result<Vec<_>, ImageError>>()?;
531    if pixel_pairs
532        .iter()
533        .all(|pair| pair[0].is_some() && pair[1].is_some())
534    {
535        return Ok(pixel_pairs
536            .into_iter()
537            .map(|pair| [pair[0].unwrap(), pair[1].unwrap()])
538            .collect());
539    }
540    if pixel_pairs
541        .iter()
542        .any(|pair| pair[0].is_some() || pair[1].is_some())
543    {
544        return Err(ImageError::InvalidMetadata(
545            "cannot mix pix and non-pix coordinates in a CRTF polygon region".to_string(),
546        ));
547    }
548
549    let coordinates = image.coordinates();
550    if coordinates.n_pixel_axes() < 2 || coordinates.n_world_axes() < 2 {
551        return Err(ImageError::InvalidMetadata(
552            "world-coordinate CRTF polygon regions require an image coordinate system with at least two axes"
553                .to_string(),
554        ));
555    }
556    let [x, y] = direction_plane_axes(image)?;
557    let [longitude, latitude] = world_axes_for_plane(coordinates, [x, y])?;
558    let base_pixel = vec![0.0; coordinates.n_pixel_axes()];
559    let base_world = coordinates.to_world(&base_pixel)?;
560    pairs
561        .iter()
562        .map(|pair| {
563            let mut world = base_world.clone();
564            world[longitude] = parse_crtf_world_quantity(&pair[0])?;
565            world[latitude] = parse_crtf_world_quantity(&pair[1])?;
566            let pixel = coordinates.to_pixel(&world)?;
567            Ok([
568                nonnegative_pixel_value(pixel[x], "CRTF polygon world x")?,
569                nonnegative_pixel_value(pixel[y], "CRTF polygon world y")?,
570            ])
571        })
572        .collect()
573}

575pub(super) fn crtf_box_parts(text: &str) -> Result<Option<[String; 4]>, ImageError> {
576    let normalized = text.trim();
577    let Some(after) = find_crtf_body(normalized, "box", "[[").and_then(|s| s.strip_prefix("[["))
578    else {
579        return Ok(None);
580    };
581    let end = after.find("]]").ok_or_else(|| {
582        ImageError::InvalidMetadata(format!(
583            "invalid CASA CRTF box region {text:?}: missing closing ]]"
584        ))
585    })?;
586    let body = &after[..end];
587    let body = body
588        .replace("], [", ",")
589        .replace("],[", ",")
590        .replace(['[', ']'], "");
591    let parts = body
592        .split(',')
593        .map(|part| part.trim().to_string())
594        .filter(|part| !part.is_empty())
595        .collect::<Vec<_>>();
596    if parts.len() != 4 {
597        return Err(ImageError::InvalidMetadata(format!(
598            "invalid CASA CRTF box region {text:?}: expected four coordinates"
599        )));
600    }
601    Ok(Some([
602        parts[0].clone(),
603        parts[1].clone(),
604        parts[2].clone(),
605        parts[3].clone(),
606    ]))
607}

609pub(super) fn find_crtf_body<'a>(text: &'a str, keyword: &str, open: &str) -> Option<&'a str> {
610    let mut offset = 0usize;
611    while let Some(rel) = text[offset..].find(keyword) {
612        let pos = offset + rel;
613        let boundary = pos == 0
614            || text[..pos]
615                .chars()
616                .next_back()
617                .is_none_or(|ch| !ch.is_ascii_alphanumeric() && ch != '_');
618        let rest = &text[pos + keyword.len()..];
619        if boundary {
620            let trimmed = rest.trim_start();
621            if trimmed.starts_with(open) {
622                return Some(trimmed);
623            }
624        }
625        offset = pos + keyword.len();
626    }
627    None
628}

634fn parse_crtf_pixel_coordinate(text: &str) -> Result<Option<f64>, ImageError> {
635    let text = text.trim();
636    let lowercase = text.to_ascii_lowercase();
637    let Some(value) = lowercase.strip_suffix("pix") else {

639    };
640    let value = value.trim().parse::<f64>().map_err(|error| {

646pub(super) fn parse_crtf_world_quantity(text: &str) -> Result<f64, ImageError> {
647    let text = text.trim();
648    let lowercase = text.to_ascii_lowercase();
649    let (value, scale) = if let Some(value) = lowercase.strip_suffix("arcsec") {
650        (value, std::f64::consts::PI / 180.0 / 3600.0)
651    } else if let Some(value) = lowercase.strip_suffix("arcmin") {
652        (value, std::f64::consts::PI / 180.0 / 60.0)
653    } else if let Some(value) = lowercase.strip_suffix("deg") {
654        (value, std::f64::consts::PI / 180.0)
655    } else if let Some(value) = lowercase.strip_suffix("rad") {
656        (value, 1.0)
657    } else {
658        (lowercase.as_str(), 1.0)
659    };
660    let parsed = value.trim().parse::<f64>().map_err(|error| {
661        ImageError::InvalidMetadata(format!("invalid CRTF world coordinate {text:?}: {error}"))
662    })?;
663    if !parsed.is_finite() {
664        return Err(ImageError::InvalidMetadata(format!(
665            "invalid CRTF world coordinate {text:?}: value is not finite"
666        )));
667    }
668    Ok(parsed * scale)
669}

671pub(super) fn crtf_world_box_pixels<T: ImagePixel>(
672    image: &impl ImageInterface<T>,
673    world_box: [f64; 4],
674) -> Result<String, ImageError> {
675    let coordinates = image.coordinates();
676    if coordinates.n_pixel_axes() < 2 || coordinates.n_world_axes() < 2 {
677        return Err(ImageError::InvalidMetadata(
678            "world-coordinate CRTF box regions require an image coordinate system with at least two axes"
679                .to_string(),
680        ));
681    }
682    let [x, y] = direction_plane_axes(image)?;
683    let base_pixel = vec![0.0; coordinates.n_pixel_axes()];
684    let mut first_world = coordinates.to_world(&base_pixel)?;
685    let [longitude, latitude] = world_axes_for_plane(coordinates, [x, y])?;
686    let mut second_world = first_world.clone();
687    first_world[longitude] = world_box[0];
688    first_world[latitude] = world_box[1];
689    second_world[longitude] = world_box[2];
690    second_world[latitude] = world_box[3];
691    let first_pixel = coordinates.to_pixel(&first_world)?;
692    let second_pixel = coordinates.to_pixel(&second_world)?;
693    let x0 = rounded_nonnegative_pixel_value(first_pixel[x], "first CRTF world x")?;
694    let y0 = rounded_nonnegative_pixel_value(first_pixel[y], "first CRTF world y")?;
695    let x1 = rounded_nonnegative_pixel_value(second_pixel[x], "second CRTF world x")?;
696    let y1 = rounded_nonnegative_pixel_value(second_pixel[y], "second CRTF world y")?;
697    Ok(format!(
698        "{},{},{},{}",
699        x0.min(x1),
700        y0.min(y1),
701        x0.max(x1),
702        y0.max(y1)
703    ))
704}

708    label: &str,

713fn nonnegative_pixel_value(value: f64, label: &str) -> Result<f64, ImageError> {
714    if !value.is_finite() || value < -1.0e-6 {
715        return Err(ImageError::InvalidMetadata(format!(
716            "invalid CRTF pixel coordinate {label:?}: value is not a non-negative finite number"
717        )));
718    }
719    Ok(value.max(0.0))
720}

722pub(super) fn looks_like_box(text: &str) -> bool {
723    let mut count = 0usize;
724    for part in text.split(',') {
725        let part = part.trim();
726        if part.is_empty() || part.parse::<usize>().is_err() {
727            return false;
728        }
729        count += 1;
730    }
731    count == 4
732}

734pub(super) fn parse_box(text: &str) -> Result<[usize; 4], ImageError> {
735    let values = text
736        .split(',')
737        .map(|part| {
738            part.trim().parse::<usize>().map_err(|error| {
739                ImageError::InvalidMetadata(format!("invalid box {text:?}: {error}"))
740            })
741        })
742        .collect::<Result<Vec<_>, _>>()?;
743    if values.len() != 4 || values[2] < values[0] || values[3] < values[1] {
744        return Err(ImageError::InvalidMetadata(format!(
745            "box must be x0,y0,x1,y1 with inclusive upper bounds: {text:?}"
746        )));
747    }
748    Ok([values[0], values[1], values[2], values[3]])
749}

751/// Parse CASA's task `box` parameter: comma-separated quadruplets of
752/// inclusive pixel corners, one box per quadruplet. CASA's
753/// `CasacRegionManager` refuses any other count with the same wording.
754pub(super) fn parse_boxes(text: &str) -> Result<Vec<[usize; 4]>, ImageError> {
755    let values = text
756        .split(',')
757        .map(|part| {
758            part.trim().parse::<usize>().map_err(|error| {
759                ImageError::InvalidMetadata(format!("invalid box {text:?}: {error}"))
760            })
761        })
762        .collect::<Result<Vec<_>, _>>()?;
763    if values.is_empty() || values.len() % 4 != 0 {
764        return Err(ImageError::CasaSelection(
765            "box not specified correctly.".to_string(),
766        ));
767    }
768    Ok(values
769        .as_chunks::<4>()
770        .0
771        .iter()
772        .map(|b| [b[0], b[1], b[2], b[3]])
773        .collect())
774}

776/// A task `box` parameter as a pixel region, validated as CASA's
777/// `CasacRegionManager::fromBCS` validates it: each quadruplet's trc
778/// inside the direction plane, and each box's blc no greater than its
779/// trc, with CASA's own refusals. A box's x and y run along the
780/// direction plane's axes wherever they are.
781pub(crate) fn parse_box_region(
782    text: &str,
783    image_shape: &[usize],
784    plane: [usize; 2],
785) -> Result<PixelRegion, ImageError> {
786    let boxes = parse_boxes(text)?;
787    let [x, y] = plane;
788    for [x0, y0, x1, y1] in &boxes {
789        if image_shape.len() < 2 || *x1 >= image_shape[x] || *y1 >= image_shape[y] {
790            return Err(ImageError::CasaSelection(
791                "trc in box spec is greater than or equal to number of direction \
792                 coordinate pixels in the image"
793                    .to_string(),
794            ));
795        }
796        if x1 < x0 || y1 < y0 {
797            let mut blc = vec![0; image_shape.len()];
798            blc[x] = *x0;
799            blc[y] = *y0;
800            let mut trc: Vec<usize> = image_shape.iter().map(|axis| axis - 1).collect();
801            trc[x] = *x1;
802            trc[y] = *y1;
803            return Err(ImageError::CasaSelection(format!(
804                "LCBox::LCBox - blc {blc:?} must be <= trc {trc:?}"
805            )));
806        }
807    }
808    Ok(PixelRegion::Boxes(boxes))
809}

811/// CASA imstat's `flux` over the selection, as casacore's
812/// `ImageStatistics::_flux` computes it: a brightness in kelvin times the
813/// pixel area in square arcseconds; a brightness per beam divided by the
814/// beam's area in pixels (one beam; per-plane beams are not computed). Over
815/// a selection of more than one channel the flux is integrated along the
816/// spectral axis: times the channel width in km/s (radio, casacore's default
817/// velocity) when the axis has a rest frequency, else its frequency
818/// increment in Hz.
819pub(super) fn image_flux<T: ImagePixel>(
820    image: &PagedImage<T>,
821    sum: f64,
822    selection: &Selection,
823) -> Option<f64> {
824    image_flux_with_integration(image, sum, selection, true)
825}
826
827pub(super) fn image_flux_with_integration<T: ImagePixel>(
828    image: &PagedImage<T>,
829    sum: f64,
830    selection: &Selection,
831    spectral_axis_collapsed: bool,
832) -> Option<f64> {
833    let spectral_axis = spectral_axis_collapsed
834        .then(|| image.find_axis(CoordinateType::Spectral))
835        .flatten()
836        .filter(|&axis| selection.shape[axis] > 1);
837    let channels = spectral_axis.map(|axis| [selection.start[axis], selection.start[axis] + 1]);
838    let beam = image
839        .units()
840        .contains("/beam")
841        .then(|| image.image_info().ok()?.beam_set.single_beam())
842        .flatten();
843    statistics_flux_scale(image.coordinates(), image.units(), beam, channels)
844        .map(|factor| sum * factor)
845}
