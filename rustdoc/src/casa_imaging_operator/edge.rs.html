2//! Where a placement's taps land on the grid under its set's [`GridEdge`]:
3//! wholly, on a box of cells only, or not at all, and whether its weight
4//! counts in `sumwt` ([`Reach`]). Both backends grid by it.
5//!
6//! A set that drops a sample whose support overruns the grid
7//! ([`GridEdge::Whole`]) grids every placement the resampler keeps wholly.
8//! W projection's sectioned edge ([`GridEdge::Sectioned`]) clips the data
9//! and the PSF to the same sectors as the mosaic's, and every sample with a
10//! tap on them weighs, by the kernel sum of those taps
11//! ([`clipped_kernel_sum`]).
12//! The mosaic's clipped edge ([`GridEdge::Clipped`]) follows `MosaicFT::put`:
13//! - the data and the PSF go through `sectgmosd2` over the sectors
14//!   `FTMachine::findGridSector` cuts, which cover every row and column of
15//!   the grid but its last (the last sector is cut one cell short of the
16//!   1-based grid's end): a support is clipped to that extent, and its weight
17//!   counts when its anchor cell lies in it (`onmosgrid`'s `centin`);
18//! - the weight image goes through `gmoswgtd` over the whole grid: the
19//!   weight taps are placed at the uv origin, but only those whose offsets
20//!   the sample's own support keeps on the grid (`onmosgrid` at the sample's
21//!   uv cell), and every sample that reaches the grid counts.
22//!
23//! Prediction (`sectdmos2`) keeps the whole support on the whole grid
24//! (`omos`), [`GridGeometry::fits`].
25
26use crate::accumulator::Mode;
27use crate::convolution::{ConvolutionFunctionSet, GridEdge};
28use crate::geometry::{CellLocation, GridGeometry};
29use crate::sample::Placement;
30
31/// Grid cells `[x0, x1) × [y0, y1)`, `[[x0, x1], [y0, y1]]`, in grid
32/// coordinates.
33pub type CellBox = [[i64; 2]; 2];
34
35/// How one placement's taps of a mode land on the grid.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub enum Reach {
38    /// Every tap lands on the grid, and the weight counts in `sumwt`.
39    Whole,
40    /// Only the taps on `cells` are gridded; the weight counts in `sumwt`
41    /// when `weighs`.
42    Clipped {
43        /// The cells the taps may land on.
44        cells: CellBox,
45        /// Whether the weight counts in `sumwt`.
46        weighs: bool,
47    },
48    /// Nothing is gridded.
49    Off,
50}
51
52impl Reach {
53    /// The reach of `placement`'s taps of `mode`, `half` taps each side of
54    /// `location` (the uv origin for [`Mode::Weight`]), located at
55    /// `oversampling` fine offsets a cell, under `cf`'s [`GridEdge`] on
56    /// `geometry`'s grid.
57    #[must_use]
58    pub fn of(
59        cf: &dyn ConvolutionFunctionSet,
60        geometry: &GridGeometry,
61        (mode, placement): (Mode, &Placement),
62        location: CellLocation,
63        (half, oversampling): ([u16; 2], u16),
64    ) -> Self {
65        if cf.grid_edge() == GridEdge::Whole {
66            return if geometry.fits(location, half) {
67                Self::Whole
68            } else {
69                Self::Off
70            };
71        }
72        let [nx, ny] = geometry.grid_shape().map(|n| n as i64);
73        match mode {
74            Mode::Data | Mode::Psf => {
75                let extent = [[0, nx - 1], [0, ny - 1]];
76                let support = support_box(location, half);
77                match overlap(support, extent) {
78                    None => Self::Off,
79                    Some(cells) if cells == support => Self::Whole,
80                    Some(cells) => Self::Clipped {
81                        cells,
82                        weighs: cf.grid_edge() == GridEdge::Sectioned || inside(location, extent),
83                    },
84                }
85            }
86            Mode::Weight => {
87                let sample = geometry.locate(placement.u, placement.v, oversampling);
88                let support = support_box(sample, half);
89                match overlap(support, [[0, nx], [0, ny]]) {
90                    None => Self::Off,
91                    Some(cells) if cells == support => Self::Whole,
92                    Some(cells) => {
93                        let shift = [location.x - sample.x, location.y - sample.y];
94                        Self::Clipped {
95                            cells: [0, 1].map(|axis| cells[axis].map(|edge| edge + shift[axis])),
96                            weighs: true,
97                        }
98                    }
99                }
100            }
101        }
102    }
103}

105/// `Σ (conjugate ? conj(t) : t)` over the taps of Mueller plane `mueller`
106/// at `location`'s fine offset that land on `cells`: what a clipped
107/// placement of a [`GridEdge::Sectioned`] set adds to `sumwt` per unit
108/// weight (its real part, `wprojgrid.f` `sectgwgridd`'s `norm`), where a
109/// whole one adds the kernel sum over its support.
110#[must_use]
111pub fn clipped_kernel_sum(
112    taps: &crate::TapLayout<'_>,
113    location: CellLocation,
114    cells: CellBox,
115    mueller: u8,
116    conjugate: bool,
117) -> num_complex::Complex64 {
118    crate::cpu::kernel::clipped_norm(taps, location, cells, mueller, conjugate)
119}
120
121/// The cells of a support of `half` taps each side of `location`.
122fn support_box(location: CellLocation, half: [u16; 2]) -> CellBox {
123    let [hx, hy] = half.map(i64::from);
124    [
125        [location.x - hx, location.x + hx + 1],
126        [location.y - hy, location.y + hy + 1],
127    ]
128}
129
130/// The cells `a` and `b` share, if any.
131fn overlap(a: CellBox, b: CellBox) -> Option<CellBox> {
132    let cells = [0, 1].map(|axis| [a[axis][0].max(b[axis][0]), a[axis][1].min(b[axis][1])]);
133    cells.iter().all(|[low, high]| low < high).then_some(cells)
134}
135
136/// Whether `location`'s cell lies in `cells`.
137fn inside(location: CellLocation, cells: CellBox) -> bool {
138    (cells[0][0]..cells[0][1]).contains(&location.x)
139        && (cells[1][0]..cells[1][1]).contains(&location.y)
140}

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    #[test]
147    fn boxes_overlap_in_the_cells_they_share() {
148        let a = [[-3, 4], [10, 17]];
149        assert_eq!(overlap(a, [[0, 63], [0, 63]]), Some([[0, 4], [10, 17]]));
150        assert_eq!(overlap(a, [[4, 63], [0, 63]]), None);
151        assert!(inside(
152            CellLocation {
153                x: 0,
154                y: 62,
155                ox: 0,
156                oy: 0
157            },
158            [[0, 63], [0, 63]]
159        ));
160        assert!(!inside(
161            CellLocation {
162                x: 63,
163                y: 0,
164                ox: 0,
165                oy: 0
166            },
167            [[0, 63], [0, 63]]
168        ));
169    }
