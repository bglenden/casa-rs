33/// The pixels a domain's minor cycle may update.
34#[derive(Clone, Debug, PartialEq)]
35pub(super) enum DomainMask {
36    /// Every valid model pixel.
37    FullPlane,
38    /// The union of inclusive pixel boxes `[x0, y0, x1, y1]`.
39    Boxes(Vec<[usize; 4]>),
40    /// The non-zero pixels of a CASA image, reprojected onto the domain.
41    Image(PathBuf),
42    /// Row-major support over the domain's own grid.
43    PixelSupport(Box<[bool]>),
44    /// CASA auto-multithresh support from the current normal state.
45    AutoMultithresh(AutoMultithreshControls),
46}

48/// One image domain of the run.
49pub(super) struct PreparedImageDomain {
50    pub(super) role: ImageDomainRole,
51    pub(super) output: PathBuf,
52    /// Pixels along x and y.
53    pub(super) image_size: [usize; 2],
54    pub(super) direction: DirectionCoordinateSpec,
55    pub(super) coordinates: CoordinateSystem,
56    pub(super) mask: DomainMask,
57    /// CASA facets the first image field only (`SynthesisImager`'s
58    /// `createFacetImageStoreList`); outliers are single.
59    pub(super) facets: FacetLayout,
60}
61
62impl PreparedImageDomain {
63    /// The model's specification of this domain: its facets, axes
64    /// direction, polarization, spectral.
65    pub(super) fn spec(&self) -> ImageDomainSpec {
66        ImageDomainSpec::new(
67            self.role.clone(),
68            ImageShape::new(self.image_size[0], self.image_size[1]),
69            self.direction,
70            self.facets,
71            AxisOrder::new([
72                ImageAxis::DirectionLongitude,
73                ImageAxis::DirectionLatitude,
74                ImageAxis::Polarization,
75                ImageAxis::Spectral,
76            ]),
77        )
78    }
79
80    /// Where the image writer writes this domain's images.
81    pub(super) fn output(&self) -> CasaImageDomainOutput {
82        CasaImageDomainOutput::new(
83            self.role.clone(),
84            self.output.clone(),
85            self.coordinates.clone(),
86        )
87    }
88}

90/// The main domain and the outlier file's, ordered by role, each writing
91/// to its own output.
92pub(super) fn prepare_domains(
93    request: &ImagingRequest,
94    centre: &Centre,
95    spectral: ImageSpectralCoordinate,
96    observation: &ObsInfo,
97) -> Result<Vec<PreparedImageDomain>, PrepareError> {
98    let polarizations = &request.stokes;
99    let facets = match request.gridder {
100        Gridder::Standard { facets } if facets.get() > 1 => FacetLayout::Regular {
101            columns: facets.get(),
102            rows: facets.get(),
103        },
104        _ => FacetLayout::Single,
105    };
106    let mut domains = vec![PreparedImageDomain {
107        role: ImageDomainRole::Main,
108        output: request.imagename.clone(),
109        image_size: request.imsize,
110        direction: centre.direction,
111        coordinates: image_coordinates(
112            centre.direction,
113            request.imsize,
114            polarizations,
115            spectral,
116            observation,
117        ),
118        mask: request_mask(request),
119        facets,
120    }];
121    if let Some(path) = request.outlierfile.as_deref() {
122        for outlier in read_outlier_domains(path, request.imsize, request.cell)? {
123            let centre = parse_phase_center_direction(&outlier.phase_center)?;
124            let direction = direction_spec(
125                outlier.image_size,
126                outlier.cell_arcsec,
127                DirectionFrame::J2000,
128                centre.longitude_rad(),
129                centre.latitude_rad(),
130            );
131            domains.push(PreparedImageDomain {
132                role: ImageDomainRole::Outlier(outlier.name),
133                output: outlier.output,
134                image_size: outlier.image_size,
135                direction,
136                coordinates: image_coordinates(
137                    direction,
138                    outlier.image_size,
139                    polarizations,
140                    spectral,
141                    observation,
142                ),
143                mask: outlier.mask,
144                facets: FacetLayout::Single,
145            });
146        }
147    }
148    // Distinct outputs give distinct roles: an outlier's output is its name
149    // resolved beside the outlier file.
150    let mut outputs = BTreeSet::new();
151    for domain in &domains {
152        if !outputs.insert(domain.output.as_path()) {
153            return Err(PrepareError::DuplicateOutput {
154                path: domain.output.clone(),
155            });
156        }
157    }
158    domains.sort_by(|left, right| left.role.cmp(&right.role));
159    for domain in &domains {
160        warn_inefficient_size(domain.image_size);
161    }
162    Ok(domains)
163}

165/// `tclean`'s warning for a side that is not CASA's efficient size
166/// (`SynthesisParamsImage::verify`): the image is still made at the size
167/// asked for. Like CASA, it names y only when it differs from x.
168fn warn_inefficient_size(image_size: [usize; 2]) {
169    let [x, y] = image_size;
170    let [best_x, best_y] = image_size.map(optimum_image_size);
171    if best_x != x {
172        tracing::warn!(
173            "imsize with {x} pixels is not an efficient imagesize. Try {best_x} instead."
174        );
175    }
176    if x != y && best_y != y {
177        tracing::warn!(
178            "imsize with {y} pixels is not an efficient imagesize. Try {best_y} instead."
179        );
180    }
181}

183/// The main domain's mask: CASA auto-multithresh, the user mask image or
184/// boxes, or every pixel. The auto-multithresh controls the catalog does
185/// not carry take CASA's defaults (`smoothfactor = 1`, `cutthreshold =
186/// 0.01`, `minpercentchange = -1`).
187fn request_mask(request: &ImagingRequest) -> DomainMask {
188    match (request.usemask, &request.mask_image) {
189        (UseMask::AutoMultithresh, _) => DomainMask::AutoMultithresh(AutoMultithreshControls {
190            sidelobe_factor: request.sidelobethreshold,
191            noise_factor: request.noisethreshold,
192            low_noise_factor: request.lownoisethreshold,
193            negative_factor: request.negativethreshold,
194            minimum_beam_fraction: request.minbeamfrac,
195            smooth_factor: 1.0,
196            cut_threshold: 0.01,
197            grow_iterations: request.growiterations,
198            minimum_percent_change: -1.0,
199        }),
200        (UseMask::User, Some(path)) => DomainMask::Image(path.clone()),
201        (UseMask::User, None) if request.mask_box.is_empty() => DomainMask::FullPlane,
202        (UseMask::User, None) => DomainMask::Boxes(request.mask_box.clone()),
203    }
204}

206/// Each domain's mask plan, bound to its direction coordinate.
207pub(super) fn mask_plans(
208    domains: &[PreparedImageDomain],
209) -> Result<ImageDomainReconstructionMaskPlans, PrepareError> {
210    Ok(ImageDomainReconstructionMaskPlans::new(
211        domains
212            .iter()
213            .map(|domain| mask_plan(domain.mask.clone(), domain.direction, domain.image_size))
214            .collect::<Result<Vec<_>, _>>()?,
215    )?)
216}
217
218fn mask_plan(
219    mask: DomainMask,
220    coordinate: DirectionCoordinateSpec,
221    image_size: [usize; 2],
222) -> Result<ReconstructionMaskPlan, PrepareError> {
223    Ok(match mask {
224        DomainMask::FullPlane => ReconstructionMaskPlan::FullPlane { coordinate },
225        DomainMask::Boxes(boxes) => ReconstructionMaskPlan::Boxes {
226            coordinate,
227            boxes: boxes
228                .into_iter()
229                .map(|[x0, y0, x1, y1]| MaskBox::new([x0, y0], [x1, y1]))
230                .collect::<Result<Vec<_>, _>>()?,
231        },
232        DomainMask::Image(path) => reproject_image_mask(&path, coordinate, image_size)?,
233        DomainMask::PixelSupport(support) => {
234            assert_eq!(
235                image_size[0].checked_mul(image_size[1]),
236                Some(support.len()),
237                "an outlier circle mask covers exactly its image domain"
238            );
239            ReconstructionMaskPlan::Reprojected {
240                coordinate,
241                support,
242            }
243        }
244        DomainMask::AutoMultithresh(controls) => ReconstructionMaskPlan::AutoMultithresh {
245            coordinate,
246            controls,
247            completed_major_cycles: 0,
248            cycle_threshold_reached: false,
249            previous: None,
250            evolution_stopped: false,
251        },
252    })
253}

255/// The non-zero, unmasked, finite pixels of a one-plane CASA image,
256/// reprojected onto `target_spec`.
257fn reproject_image_mask(
258    path: &Path,
259    target_spec: DirectionCoordinateSpec,
260    target_size: [usize; 2],
261) -> Result<ReconstructionMaskPlan, PrepareError> {
262    let (source_shape, source_coordinates, source_support) = match AnyPagedImage::open(path)? {
263        AnyPagedImage::Float32(image) => {
264            let mask = image.get_mask()?;
265            let support = image
266                .get()?
267                .iter()
268                .enumerate()
269                .map(|(index, value)| {
270                    value.is_finite()
271                        && *value != 0.0
272                        && mask.as_ref().is_none_or(|mask| mask[index])
273                })
274                .collect::<Vec<_>>();
275            (image.shape().to_vec(), image.coordinates().clone(), support)
276        }
277        AnyPagedImage::Float64(image) => {
278            let mask = image.get_mask()?;
279            let support = image
280                .get()?
281                .iter()
282                .enumerate()
283                .map(|(index, value)| {
284                    value.is_finite()
285                        && *value != 0.0
286                        && mask.as_ref().is_none_or(|mask| mask[index])
287                })
288                .collect::<Vec<_>>();
289            (image.shape().to_vec(), image.coordinates().clone(), support)
290        }
291        AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
292            return Err(PrepareError::ComplexMaskImage);
293        }
294    };
295    let index = source_coordinates
296        .find_coordinate(CoordinateType::Direction)
297        .ok_or(PrepareError::MaskImageDirection)?;
298    // The plane is taken from the first two axes, which are the direction's
299    // only when it is the first coordinate: a degenerate transposed mask
300    // passes the shape check below with its plane transposed.
301    if index != 0 {
302        return Err(PrepareError::MaskImageDirectionAxes {
303            axes: (0..source_coordinates.n_coordinates())
304                .flat_map(|k| source_coordinates.coordinate(k).axis_names())
305                .collect(),
306        });
307    }
308    if source_shape.len() < 2
309        || source_shape[2..].iter().any(|extent| *extent != 1)
310        || source_support.len() != source_shape[0] * source_shape[1]
311    {
312        return Err(PrepareError::MaskImageShape {
313            shape: source_shape,
314        });
315    }
316    let source_spec = direction_model_spec(source_coordinates.coordinate(index))?;
317    let support = casa_imaging_deconvolution::reproject_mask_support(
318        source_spec,
319        [source_shape[0], source_shape[1]],
320        &source_support,
321        target_spec,
322        target_size,
323    )?;
324    Ok(ReconstructionMaskPlan::Reprojected {
325        coordinate: target_spec,
326        support,
327    })
328}

330fn direction_model_spec(
331    coordinate: &CoordinateModel,
332) -> Result<DirectionCoordinateSpec, PrepareError> {
333    let CoordinateModel::Direction(direction) = coordinate else {
334        unreachable!("find_coordinate(Direction) names a direction coordinate");
335    };
336    let projection = direction.projection().projection_type();
337    if projection != ProjectionType::SIN {
338        return Err(PrepareError::MaskImageProjection { projection });
339    }
340    let frame = match direction.direction_ref() {
341        DirectionRef::J2000 => DirectionFrame::J2000,
342        DirectionRef::B1950 => DirectionFrame::B1950,
343        DirectionRef::GALACTIC => DirectionFrame::Galactic,
344        DirectionRef::ICRS => DirectionFrame::Icrs,
345        frame => return Err(PrepareError::MaskImageFrame { frame }),
346    };
347    let reference = coordinate.reference_value();
348    let pixel = coordinate.reference_pixel();
349    let increment = coordinate.increment();
350    let pc = direction.pc_matrix();
351    Ok(DirectionCoordinateSpec::new(
352        Projection::Sin,
353        SkyDirection::new(frame, reference[0], reference[1]),
354        [pixel[0], pixel[1]],
355        [increment[0], increment[1]],
356        [[pc[[0, 0]], pc[[0, 1]]], [pc[[1, 0]], pc[[1, 1]]]],
357        [
358            direction.longpole().to_degrees(),
359            direction.latpole().to_degrees(),
360        ],
361    ))
362}

364/// Pixels a model plane holds across every domain, times `planes` and
365/// `polarizations`.
366pub(super) fn model_samples(
367    domains: &[PreparedImageDomain],
368    planes: usize,
369    polarizations: &[PolarizationCoordinate],
370) -> Result<usize, PrepareError> {
371    domains
372        .iter()
373        .try_fold(0_usize, |total, domain| {
374            domain.image_size[0]
375                .saturating_mul(domain.image_size[1])
376                .checked_add(total)
377        })
378        .and_then(|samples| samples.checked_mul(planes))
379        .and_then(|samples| samples.checked_mul(polarizations.len()))
380        .ok_or(PrepareError::ImageSize)
381}

393    fn direction() -> DirectionCoordinate {
394        DirectionCoordinate::new(
395            DirectionRef::J2000,
396            CoordinateProjection::new(ProjectionType::SIN),
397            [1.2, -0.4],
398            [-1e-5, 1e-5],
399            [1.5, 0.0],
400        )
401    }
402
403    /// A one-plane mask of ones on a 4 by 1 direction plane, with the
404    /// direction first or (as a transposed CASA mask reads) after Stokes.
405    fn mask(path: &Path, direction_first: bool) -> DirectionCoordinateSpec {
406        let mut coordinates = CoordinateSystem::new();
407        let stokes = StokesCoordinate::new(vec![StokesType::I]);
408        let shape = if direction_first {
409            coordinates.add_coordinate(direction());
410            coordinates.add_coordinate(stokes);
411            vec![4, 1, 1, 1]
412        } else {
413            coordinates.add_coordinate(stokes);
414            coordinates.add_coordinate(direction());
415            vec![1, 4, 1, 1]
416        };
417        coordinates.add_coordinate(SpectralCoordinate::new(
418            FrequencyRef::LSRK,
419            1.0e9,
420            1.0e6,
421            0.0,
422            1.0e9,
423        ));
424        let mut image = PagedImage::<f32>::create(shape, coordinates, path).unwrap();
425        image.set(1.0).unwrap();
426        image.save().unwrap();
427        direction_model_spec(&CoordinateModel::Direction(direction())).unwrap()
428    }

430    #[test]
431    fn a_mask_whose_direction_is_not_on_its_first_two_axes_is_refused() {
432        // [Stokes, RA, Dec, Freq] of shape [1, 4, 1, 1] passes the
433        // one-plane shape check, but its first two axes are Stokes and RA:
434        // read as the direction plane, its 4 by 1 row would become a column.
435        let dir = tempfile::tempdir().unwrap();
436        let transposed = dir.path().join("transposed.mask");
437        let target = mask(&transposed, false);
438        let error = reproject_image_mask(&transposed, target, [4, 1]).unwrap_err();
439        assert!(
440            matches!(&error, PrepareError::MaskImageDirectionAxes { axes }
441                if axes[..2] == ["Stokes", "Right Ascension"]),
442            "{error}"
443        );
444
445        let first = dir.path().join("first.mask");
446        let target = mask(&first, true);
447        assert!(reproject_image_mask(&first, target, [4, 1]).is_ok());
448    }
