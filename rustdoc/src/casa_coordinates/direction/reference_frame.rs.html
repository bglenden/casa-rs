3use casa_core::measures::direction::DirectionRef;
4
5use crate::{CoordinateError, ProjectionType};
6
7use super::DirectionCoordinate;
8
9impl DirectionCoordinate {
10    /// Rebuild the projection in a native frame. When supplied, `reference`
11    /// gives the adjusted reference angles in radians. Increments and PC
12    /// are retained.
13    ///
14    /// # Errors
15    /// Returns an error if the PC matrix cannot be normalized or the
16    /// reference angles cannot be represented with the projection's pole.
17    pub fn with_reassigned_reference_frame(
18        &self,
19        frame: DirectionRef,
20        reference: Option<[f64; 2]>,
21    ) -> Result<Self, CoordinateError> {
22        let mut coordinate = Self::new(
23            frame,
24            self.projection.clone(),
25            self.crval,
26            self.cdelt,
27            self.crpix,
28        )
29        .with_normalized_pc_matrix(self.pc.clone())?;
30        if let Some(mut reference) = reference {
31            if self.projection.projection_type() == ProjectionType::SFL
32                && coordinate.cdelt[1] != 0.0
33            {
34                // Casacore puts an adjusted SFL reference on the equator and
35                // shifts the reference pixel to preserve its latitude offset.
36                coordinate.crpix[1] -= reference[1] / coordinate.cdelt[1];
37                reference[1] = 0.0;
38            }
39            coordinate = coordinate.with_reference_value(reference);
40            coordinate.validate_native_pole().map_err(|_| {
41                CoordinateError::InvalidRecord(
42                    "wcs wcsset_error: Ill-conditioned coordinate transformation parameter".into(),
43                )
44            })?;
45        }
46        coordinate.latpole = coordinate.native_pole[1];
47        Ok(coordinate)
48    }
49
50    /// Rebuild the direction coordinate in a new native reference frame.
51    ///
52    /// Keep the projection, pixel origin, and angular increments. The new
53    /// coordinate starts with an identity PC matrix. The returned angle in
54    /// radians describes the old pixel plane's rotation relative to the new
55    /// frame, with clockwise positive as in casacore's `convert`.
56    ///
57    /// # Errors
58    /// Returns an error for nonsquare pixels, an unavailable frame conversion,
59    /// or an invalid projected offset used to measure the rotation.
60    pub fn convert_reference_frame(
61        &self,
62        frame: DirectionRef,
63    ) -> Result<(Self, f64), CoordinateError> {
64        // Casacore restores direction records with unit-length PC rows. Their
65        // scale belongs in the increments before checking for square pixels.
66        let mut old = self
67            .clone()
68            .with_normalized_pc_matrix(self.pc.clone())?
69            .with_conversion_direction_ref(frame);
70        let [first, second] = old.in_world_units(old.cdelt).map(f64::abs);
71        if (first - second).abs() > 2e-10 * first.max(second) {
72            return Err(CoordinateError::ConversionFailed(
73                "Coordinate rotation can only be performed on a coordinate that has square pixels"
74                    .into(),
75            ));
76        }
77        let world = old.pixel_to_world(old.crpix)?;
78        let mut converted = Self::new(frame, self.projection.clone(), world, old.cdelt, old.crpix);
79        // CASA saves the native pole selected by the WCS constructor.
80        converted.latpole = converted.native_pole[1];
81
82        // Measure the new frame's positive latitude direction on the old
83        // projection. A one-degree offset toward the equator avoids crossing
84        // a pole, and a zero reference pixel avoids cancellation.
85        old.crpix = [0.0; 2];
86        old.cdelt = old.cdelt.map(|step| 1.0_f64.to_radians().copysign(step));
87        let toward_equator: f64 = if world[1] < 0.0 { 1.0 } else { -1.0 };
88        let [x, y] = old.world_to_pixel([world[0], world[1] + toward_equator.to_radians()])?;
89        let (x, y) = (x * toward_equator, y * toward_equator);
90        let length = x.hypot(y);
91        if !length.is_finite() || length == 0.0 {
92            return Err(CoordinateError::ConversionFailed(
93                "Unable to measure the rotation of the converted direction coordinate".into(),
94            ));
95        }
96        let angle = (y / length).clamp(-1.0, 1.0).acos();
97        // WCSLIB's degree trigonometry gives zero at the negative meridian.
98        // Radian roundoff at that meridian must leave the half-turn positive.
99        let on_meridian = angle == std::f64::consts::PI && x.abs() <= 8.0 * f64::EPSILON * length;
100        Ok((
101            converted,
102            if x > 0.0 && !on_meridian {
103                -angle
104            } else {
105                angle
106            },
107        ))
108    }
