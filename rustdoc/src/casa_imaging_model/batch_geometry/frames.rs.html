19/// The observatory, antennas and fields of one observation, in the frames
20/// geometry needs, and the parallactic angles computed so far.
21pub(super) struct Frames {
22    measures: Arc<dyn MeasuresProvider>,
23    observatory: MPosition,
24    antennas: Vec<Antenna>,
25    /// Each field's phase centre in J2000.
26    fields: Vec<MDirection>,
27    /// Each antenna's parallactic angle by time and field, NaN where not
28    /// yet computed: an antenna's angle at a time is the same in every
29    /// partition, and the batches visit the partitions one after another.
30    parallactic: Mutex<ParallacticAngles>,
31}
32
33/// Every antenna's parallactic angle by (time bits, time scale, field).
34type ParallacticAngles = HashMap<ParallacticKey, Box<[f64]>>;
35
36type ParallacticKey = (u64, EpochRef, FieldId);
37
38struct Antenna {
39    position: MPosition,
40    alt_az: bool,
41    dish_diameter_m: f64,
42}
43
44impl Frames {
45    /// The frames of `catalogue`: the observatory named by its first
46    /// observation's telescope ([`MPosition::casa_observatory`], else the
47    /// first antenna's position), and its fields' phase centres in J2000.
48    /// A centre in another frame is converted from the observatory at its
49    /// reference epoch, or with no epoch where the field states none (an
50    /// MSv4 field: ICRS or another frame whose tie to J2000 is fixed).
51    pub(super) fn new(
52        catalogue: &Catalogue,
53        measures: Arc<dyn MeasuresProvider>,
54    ) -> Result<Self, BatchGeometryError> {
55        let metres = Unit::new("m").expect("metres are a unit");
56        let antennas = catalogue
57            .antennas
58            .iter()
59            .map(|antenna| {
60                Ok(Antenna {
61                    position: antenna.position.clone(),
62                    alt_az: antenna.alt_az(),
63                    dish_diameter_m: antenna.dish_diameter.get_value_in(&metres).map_err(|_| {
64                        BatchGeometryError::Catalogue("states a dish diameter that is not a length")
65                    })?,
66                })
67            })
68            .collect::<Result<Vec<_>, BatchGeometryError>>()?;
69        let observatory = MPosition::casa_array_reference(
70            catalogue
71                .observations
72                .first()
73                .map(|observation| observation.telescope.as_str()),
74            antennas.first().map(|antenna| &antenna.position),
75            measures.as_ref(),
76        );
77        let fields = catalogue
78            .fields
79            .iter()
80            .map(|field| {
81                if field.phase_centre.refer() == DirectionRef::J2000 {
82                    return Ok(field.phase_centre.clone());
83                }
84                let mut frame = MeasFrame::new();
85                if let Some(epoch) = field.reference_epoch.clone() {
86                    frame = frame.with_epoch(epoch);
87                }
88                let frame = frame
89                    .with_position(observatory.clone())
90                    .with_measures(Arc::clone(&measures));
91                Ok(field.phase_centre.convert_to(DirectionRef::J2000, &frame)?)
92            })
93            .collect::<Result<Vec<_>, BatchGeometryError>>()?;
94        Ok(Self {
95            measures,
96            observatory,
97            antennas,
98            fields,
99            parallactic: Mutex::new(HashMap::new()),
100        })
101    }
102
103    /// `field`'s phase centre in J2000.
104    pub(super) fn field(&self, field: FieldId) -> &MDirection {
105        &self.fields[field.0 as usize]
106    }
107
108    /// The observatory's frame at `time_mjd_s` on `scale`.
109    pub(super) fn observatory_at(&self, time_mjd_s: f64, scale: EpochRef) -> MeasFrame {
110        MeasFrame::new()
111            .with_epoch(MEpoch::from_mjd_seconds(time_mjd_s, scale))
112            .with_position(self.observatory.clone())
113            .with_measures(Arc::clone(&self.measures))
114    }
115
116    /// `angles_rad` in `frame` converted to J2000 at `time_mjd_s`, from the
117    /// observatory; J2000 angles unchanged.
118    pub(super) fn j2000(
119        &self,
120        time_mjd_s: f64,
121        scale: EpochRef,
122        angles_rad: [f64; 2],
123        frame: DirectionRef,
124    ) -> Result<[f64; 2], BatchGeometryError> {
125        if frame == DirectionRef::J2000 {
126            return Ok(angles_rad);
127        }
128        let direction = MDirection::from_angles(angles_rad[0], angles_rad[1], frame);
129        let context = self
130            .observatory_at(time_mjd_s, scale)
131            .with_direction(direction.clone());
132        let (longitude, latitude) = direction
133            .convert_to(DirectionRef::J2000, &context)?
134            .as_angles();
135        Ok([longitude, latitude])
136    }

138    /// Whether Measures names `name` and has a model of it
139    /// ([`MDirection::check_source_name`]); [`Self::named`] can still fail
140    /// at an epoch that model does not span.
141    pub(super) fn check_named(&self, name: &str) -> Result<(), MeasureError> {
142        let frame = MeasFrame::new()
143            .with_position(self.observatory.clone())
144            .with_measures(Arc::clone(&self.measures));
145        MDirection::check_source_name(name, &frame)
146    }
147
148    /// The J2000 direction of the solar-system body or catalogued source
149    /// `name` at `time_mjd_s`, seen from the observatory.
150    pub(super) fn named(
151        &self,
152        time_mjd_s: f64,
153        scale: EpochRef,
154        name: &str,
155    ) -> Result<MDirection, BatchGeometryError> {
156        let frame = self.observatory_at(time_mjd_s, scale);
157        Ok(MDirection::from_source_name(name, &frame)?.convert_to(DirectionRef::J2000, &frame)?)
158    }

160    /// The parallactic angles of `field` at `time_mjd_s` for every antenna,
161    /// computed for `antennas` (zero for a mount that is not alt-az; NaN
162    /// for an antenna no batch has asked for at this time):
163    /// [`MDirection::parallactic_angle`] in each antenna's frame. The cache
164    /// is consulted once a time, not once a sample.
165    pub(super) fn parallactic_angles(
166        &self,
167        time_mjd_s: f64,
168        scale: EpochRef,
169        field: FieldId,
170        antennas: &[AntennaId],
171    ) -> Result<Box<[f64]>, BatchGeometryError> {
172        let key = (time_mjd_s.to_bits(), scale, field);
173        let mut angles = self
174            .parallactic
175            .lock()
176            .expect("parallactic angle cache poisoned")
177            .get(&key)
178            .cloned()
179            .unwrap_or_else(|| vec![f64::NAN; self.antennas.len()].into_boxed_slice());
180        let mut computed = false;
181        for antenna in antennas {
182            let slot = antenna.0 as usize;
183            if !angles[slot].is_nan() {
184                continue;
185            }
186            let site = &self.antennas[slot];
187            angles[slot] = if site.alt_az {
188                let frame = MeasFrame::new()
189                    .with_epoch(MEpoch::from_mjd_seconds(time_mjd_s, scale))
190                    .with_position(site.position.clone())
191                    .with_measures(Arc::clone(&self.measures));
192                self.field(field).parallactic_angle(&frame)?
193            } else {
194                0.0
195            };
196            computed = true;
197        }
198        if computed {
199            self.parallactic
200                .lock()
201                .expect("parallactic angle cache poisoned")
202                .insert(key, angles.clone());
203        }
204        Ok(angles)
205    }

207    /// At most the bytes the parallactic angles computed so far hold: one
208    /// angle an antenna for each (time, field) a batch has asked for, and
209    /// the table that finds them. The cache holds every (time, field) of
210    /// the observation once its batches have all been read, and no more.
211    pub(super) fn parallactic_bytes(&self) -> usize {
212        let cache = self
213            .parallactic
214            .lock()
215            .expect("parallactic angle cache poisoned");
216        let angles = cache.len() * self.antennas.len() * size_of::<f64>();
217        let capacity = cache.capacity();
218        if capacity == 0 {
219            return angles;
220        }
221        // The standard table keeps at most capacity × 8/7 + 1 buckets, each
222        // an entry and a control byte, plus one group of control bytes and
223        // its alignment (at most 16 bytes each).
224        let buckets = (capacity * 8).div_ceil(7) + 1;
225        angles + buckets * (size_of::<(ParallacticKey, Box<[f64]>)>() + 1) + 32
226    }

228    /// CASA's aperture class of each antenna
229    /// ([`AntennaResponseClass::casa_alma_aca`]) when every observation is
230    /// an ALMA or ACA one; `None` throughout for another telescope.
231    pub(super) fn alma_aca_classes(
232        &self,
233        catalogue: &Catalogue,
234    ) -> Vec<Option<AntennaResponseClass>> {
235        let alma = !catalogue.observations.is_empty()
236            && catalogue.observations.iter().all(|observation| {
237                AntennaResponseClass::casa_alma_aca_telescope(&observation.telescope)
238            });
239        self.antennas
240            .iter()
241            .map(|antenna| {
242                if alma {
243                    AntennaResponseClass::casa_alma_aca(antenna.dish_diameter_m)
244                } else {
245                    None
246                }
247            })
248            .collect()
249    }
