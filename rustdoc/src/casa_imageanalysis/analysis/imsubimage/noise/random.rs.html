7// Seed expansion and word generation follow casacore's MLCG in
8// casa/BasicMath/Random.cc. Its legacy 32-bit arithmetic determines
9// the sequence visible to scripts that supply two seeds.
10const SEED_TABLE: [u32; 32] = [
11    0xbdcc47e5, 0x54aea45d, 0xec0df859, 0xda84637b, 0xc8c6cb4f, 0x35574b01, 0x28260b7d, 0x0d07fdbf,
12    0x9faaeeb0, 0x613dd169, 0x5ce2d818, 0x85b9e706, 0xab2469db, 0xda02b0dc, 0x45c60d6e, 0xffe49d10,
13    0x7224fea3, 0xf9684fc9, 0xfc7ee074, 0x326ce92a, 0x366d13b5, 0x17aaa731, 0xeb83a675, 0x7781cb32,
14    0x4ec7c92d, 0x7f187521, 0x2cf346b4, 0xad13310f, 0xb89cff2b, 0x12164de1, 0xa865168d, 0x32b56cdf,
15];
16
17pub(super) struct Generator {
18    first: i32,
19    second: i32,
20    spare_normal: Option<f64>,
21}
22
23impl Generator {
24    pub(super) fn new(seeds: &[i64]) -> Self {
25        let expand = |seed: i32, period| {
26            let seed = if seed < 0 {
27                (seed + 2_147_483_561).abs()
28            } else {
29                seed
30            };
31            let entry = SEED_TABLE[seed as usize & (SEED_TABLE.len() - 1)] as i32;
32            let seed = if seed < SEED_TABLE.len() as i32 {
33                entry
34            } else {
35                seed ^ entry
36            };
37            (seed % period) + 1
38        };
39        let (first, second) = match seeds {
40            [first, second, ..] => (*first as i32, *second as i32),
41            _ => {
42                let elapsed = SystemTime::now()
43                    .duration_since(UNIX_EPOCH)
44                    .map_or(0.0, |elapsed| elapsed.as_secs_f64());
45                let base = (40_587.0 + elapsed / 86_400.0) * 1e7;
46                let second = base as i64 as i32;
47                let first = seeds.first().map_or_else(
48                    || (1e7 * (base - f64::from(second))) as i64 as i32,
49                    |&seed| seed as i32,
50                );
51                (first, second)
52            }
53        };
54        Self {
55            first: expand(first, 2_147_483_561),
56            second: expand(second, 2_147_483_397),
57            spare_normal: None,
58        }
59    }
60
61    fn word(&mut self) -> u32 {
62        let advance = |seed: i32, multiplier: i32, divisor: i32, correction: i32, period: i32| {
63            let residue = seed % divisor;
64            let next = multiplier
65                .wrapping_mul(seed.wrapping_sub(residue.wrapping_mul(divisor)))
66                .wrapping_sub(residue.wrapping_mul(correction));
67            if next < 0 { next + period } else { next }
68        };
69        self.first = advance(self.first, 40_014, 53_668, 12_211, 2_147_483_563);
70        self.second = advance(self.second, 40_692, 52_774, 3_791, 2_147_483_399);
71        let value = self.first.wrapping_sub(self.second);
72        (if value < 1 {
73            value + 2_147_483_562
74        } else {
75            value
76        }) as u32
77    }
78
79    fn uniform(&mut self) -> f64 {
80        let high = u64::from(self.word() & 0xfffff) << 32;
81        let low = u64::from(self.word());
82        f64::from_bits(1.0_f64.to_bits() | high | low) - 1.0
83    }
84
85    fn normal(&mut self) -> f64 {
86        if let Some(spare) = self.spare_normal.take() {
87            return spare;
88        }
89        loop {
90            let first = 2.0 * self.uniform() - 1.0;
91            let second = 2.0 * self.uniform() - 1.0;
92            let distance = first * first + second * second;
93            if distance <= 1.0 {
94                let scale = (-2.0 * distance.ln() / distance).sqrt();
95                self.spare_normal = Some(second * scale);
96                return first * scale;
97            }
98        }
99    }
100}

102pub(super) enum Distribution {
103    Binomial(u32, f64),
104    DiscreteUniform(i32, f64),
105    Erlang(i32, f64),
106    Geometric(f64),
107    HyperGeometric(f64, f64),
108    Normal(f64, f64),
109    LogNormal(f64, f64),
110    NegativeExponential(f64),
111    Poisson(f64),
112    Uniform(f64, f64),
113    Weibull(f64, f64),
114}
115
116impl Distribution {
117    pub(super) fn parse(name: &str, parameters: &[f64]) -> Result<Self, ImageError> {
118        let name = name.to_ascii_uppercase();
119        let nonzero = |value: f64| value.abs() > 1e-13;
120        let probability = |value: f64| (0.0..=1.0).contains(&value);
121        let illegal =
122            || ImageError::CasaSelection("The distribution parameters are illegal".into());
123        let distribution = match (name.as_str(), parameters) {
124            ("BINOMIAL", &[trials, chance]) if trials >= 0.5 && probability(chance) => {
125                let trials = if trials < -(i64::MIN as f64) {
126                    trials as i64 as u32
127                } else {
128                    0
129                };
130                if trials == 0 {
131                    return Err(ImageError::CasaSelection(
132                        "Failed AlwaysAssert newN > 0".into(),
133                    ));
134                }
135                Self::Binomial(trials, chance)
136            }
137            ("DISCRETEUNIFORM", &[low, high]) if low <= high => {
138                let low = legacy_i32(low);
139                let high = legacy_i32(high);
140                if low > high {
141                    return Err(ImageError::CasaSelection(
142                        "Failed AlwaysAssert low <= high".into(),
143                    ));
144                }
145                Self::DiscreteUniform(low, f64::from(high.wrapping_sub(low).wrapping_add(1)))
146            }
147            ("ERLANG", &[mean, variance]) if nonzero(mean) && variance > 0.0 => {
148                let count = legacy_i32(mean * mean / variance + 0.5).max(1);
149                Self::Erlang(count, f64::from(count) / mean)
150            }
151            ("GEOMETRIC", &[0.0]) => return Err(illegal()),
152            ("GEOMETRIC", &[chance]) if (0.0..1.0).contains(&chance) => Self::Geometric(chance),
153            ("HYPERGEOMETRIC", &[mean, variance])
154                if nonzero(mean) && variance > 0.0 && mean * mean <= variance =>
155            {
156                // The mean setter first checks against the constructor's unit variance.
157                if mean.abs() > 1.0 {
158                    return Err(ImageError::CasaSelection(
159                        "Failed AlwaysAssert itsMean*itsMean <= itsVariance".into(),
160                    ));
161                }
162                let ratio = variance / (mean * mean);
163                let chance = 0.5 * (1.0 - ((ratio - 1.0) / (ratio + 1.0)).sqrt());
164                Self::HyperGeometric(mean, chance)
165            }
166            ("NORMAL", &[mean, variance]) if variance > 0.0 => Self::Normal(mean, variance.sqrt()),
167            ("LOGNORMAL", &[mean, variance]) if nonzero(mean) && variance > 0.0 => {
168                let mean_squared = mean * mean;
169                // The mean setter first tries to build a Normal distribution
170                // with the constructor's original lognormal variance of one.
171                let initial_ratio = (1.0 + mean_squared) / mean_squared;
172                if initial_ratio <= 1.0 || initial_ratio.is_nan() {
173                    return Err(ImageError::CasaSelection(
174                        "Failed AlwaysAssert itsVariance > 0.0".into(),
175                    ));
176                }
177                let scale = ((variance + mean_squared) / mean_squared).ln().sqrt();
178                if scale.is_nan() || scale <= 0.0 {
179                    return Err(illegal());
180                }
181                Self::LogNormal(
182                    (mean_squared / (variance + mean_squared).sqrt()).ln(),
183                    scale,
184                )
185            }
186            ("NEGATIVEEXPONENTIAL", &[mean]) => Self::NegativeExponential(mean),
187            ("POISSON", &[mean]) if mean >= 0.0 => {
188                let bound = (-mean).exp();
189                if bound == 0.0 {
190                    return Err(illegal());
191                }
192                Self::Poisson(bound)
193            }
194            ("UNIFORM", &[low, high]) if low < high => Self::Uniform(low, high - low),
195            ("WEIBULL", &[alpha, beta]) if nonzero(alpha) && beta > 0.0 => {
196                Self::Weibull(1.0 / alpha, beta)
197            }
198            (
199                "BINOMIAL"
200                | "DISCRETEUNIFORM"
201                | "ERLANG"
202                | "GEOMETRIC"
203                | "HYPERGEOMETRIC"
204                | "NORMAL"
205                | "LOGNORMAL"
206                | "NEGATIVEEXPONENTIAL"
207                | "POISSON"
208                | "UNIFORM"
209                | "WEIBULL",
210                _,
211            ) => return Err(illegal()),
212            _ => {
213                return Err(ImageError::CasaSelection(
214                    "Exception: You have not yet called function 'set'.".into(),
215                ));
216            }
217        };
218        Ok(distribution)
219    }
220
221    pub(super) fn draw(&self, generator: &mut Generator) -> f64 {
222        match *self {
223            Self::Binomial(trials, chance) => {
224                let successes = (0..trials).filter(|_| generator.uniform() < chance).count();
225                successes as f64
226            }
227            Self::DiscreteUniform(low, range) => {
228                f64::from(low.wrapping_add((range * generator.uniform()).floor() as i32))
229            }
230            Self::Erlang(count, scale) => {
231                let product: f64 = (0..count).map(|_| generator.uniform()).product();
232                -product.ln() / scale
233            }
234            Self::Geometric(chance) => {
235                let mut count = 0u32;
236                while generator.uniform() > chance {
237                    count = count.wrapping_add(1);
238                }
239                f64::from(count)
240            }
241            Self::HyperGeometric(mean, chance) => {
242                let selected = if generator.uniform() > chance {
243                    1.0 - chance
244                } else {
245                    chance
246                };
247                -mean * generator.uniform().ln() / (2.0 * selected)
248            }
249            Self::Normal(mean, scale) => generator.normal() * scale + mean,
250            Self::LogNormal(mean, scale) => {
251                std::f64::consts::E.powf(generator.normal() * scale + mean)
252            }
253            Self::NegativeExponential(mean) => -mean * generator.uniform().ln(),
254            Self::Poisson(bound) => {
255                let mut product = generator.uniform();
256                let mut count = 0;
257                while product >= bound {
258                    product *= generator.uniform();
259                    count += 1;
260                }
261                f64::from(count)
262            }
263            Self::Uniform(low, width) => low + width * generator.uniform(),
264            Self::Weibull(power, beta) => (beta * -(1.0 - generator.uniform()).ln()).powf(power),
265        }
266    }
267}

269// The pinned CASA's signed double-to-int conversion returns INT_MIN
270// whenever truncation cannot produce a signed 32-bit integer.
271fn legacy_i32(value: f64) -> i32 {
272    let value = value.trunc();
273    if (f64::from(i32::MIN)..-f64::from(i32::MIN)).contains(&value) {
274        value as i32
275    } else {
276        i32::MIN
277    }
278}
