3//! Bounded CASA MVC cube-to-Taylor PSF and residual sums.
4
5use crate::ProductsError;
6
7/// Accumulate the two-term MVC Taylor PSF, residual and sum-weight moments.
8///
9/// The caller owns and admits all storage. `psf` holds three contiguous
10/// x-major planes, `residual` two, and `mean_pb` one: no channel cube or
11/// channel plane is retained. Add published, cutoff-applied cube PB planes,
12/// normalized cube PSFs, PB-adjusted cube residuals and their published Float
13/// sum weights in channel order. Finish divides by their total weight and,
14/// for images, by the supplied mean PB above the strict cutoff.
15///
16/// An error poisons the accumulator; discard the partially written buffers.
17pub struct TaylorMomentAccumulator<'a> {
18    mean_pb: &'a [f32],
19    psf: &'a mut [f32],
20    residual: &'a mut [f32],
21    sumwt: &'a mut [f32; 3],
22    reference_hz: f64,
23    cutoff: f32,
24    weight: f32,
25    failed: bool,
26}
27
28impl<'a> TaylorMomentAccumulator<'a> {
29    /// Clear and borrow the caller's two-term output planes.
30    ///
31    /// # Errors
32    ///
33    /// The shape, reference frequency, cutoff or common PB is invalid, or
34    /// the common PB has no positive pixel (CASA refuses a zero maximum).
35    pub fn new(
36        mean_pb: &'a [f32],
37        psf: &'a mut [f32],
38        residual: &'a mut [f32],
39        sumwt: &'a mut [f32; 3],
40        reference_hz: f64,
41        cutoff: f32,
42    ) -> Result<Self, ProductsError> {
43        let cells = mean_pb.len();
44        if cells == 0
45            || cells.checked_mul(3) != Some(psf.len())
46            || cells.checked_mul(2) != Some(residual.len())
47            || !reference_hz.is_finite()
48            || reference_hz <= 0.0
49            || !cutoff.is_finite()
50            || cutoff < 0.0
51            || mean_pb
52                .iter()
53                .any(|value| !value.is_finite() || *value < 0.0)
54            || !mean_pb.iter().any(|value| *value > 0.0)
55        {
56            return Err(ProductsError::InvalidTaylorMomentReduction);
57        }
58        psf.fill(0.0);
59        residual.fill(0.0);
60        sumwt.fill(0.0);
61        Ok(Self {
62            mean_pb,
63            psf,
64            residual,
65            sumwt,
66            reference_hz,
67            cutoff,
68            weight: 0.0,
69            failed: false,
70        })
71    }
72
73    /// Add one normalized cube channel and its published weight and frequency.
74    ///
75    /// # Errors
76    ///
77    /// Shapes, weight, frequency, pixels or published PB support are invalid,
78    /// arithmetic is non-finite, or an earlier call failed.
79    pub fn add(
80        &mut self,
81        psf: &[f32],
82        residual: &[f32],
83        pb: &[f32],
84        weight: f32,
85        frequency_hz: f64,
86    ) -> Result<(), ProductsError> {
87        let cells = self.mean_pb.len();
88        if self.failed
89            || psf.len() != cells
90            || residual.len() != cells
91            || pb.len() != cells
92            || !weight.is_finite()
93            || weight < 0.0
94            || !frequency_hz.is_finite()
95            || frequency_hz <= 0.0
96        {
97            self.failed = true;
98            return Err(ProductsError::InvalidTaylorMomentReduction);
99        }
100        let x = (frequency_hz - self.reference_hz) / self.reference_hz;
101        let factors = [1.0_f64, x, x * x].map(|factor| (factor * f64::from(weight)) as f32);
102        let total = self.weight + weight;
103        if !total.is_finite() || factors.iter().any(|factor| !factor.is_finite()) {
104            self.failed = true;
105            return Err(ProductsError::InvalidTaylorMomentReduction);
106        }
107        for (term, &factor) in factors.iter().enumerate() {
108            self.sumwt[term] += factor * weight;
109            if !self.sumwt[term].is_finite() {
110                self.failed = true;
111                return Err(ProductsError::InvalidTaylorMomentReduction);
112            }
113        }
114        for pixel in 0..cells {
115            let channel_pb = pb[pixel];
116            if !channel_pb.is_finite()
117                || channel_pb < 0.0
118                || (channel_pb > 0.0 && channel_pb <= self.cutoff)
119                || !psf[pixel].is_finite()
120                || !residual[pixel].is_finite()
121            {
122                self.failed = true;
123                return Err(ProductsError::InvalidTaylorMomentReduction);
124            }
125            let psf_pb = channel_pb * psf[pixel];
126            let residual_pb = channel_pb * residual[pixel];
127            for (term, &factor) in factors.iter().enumerate() {
128                let psf = &mut self.psf[term * cells + pixel];
129                *psf += factor * psf_pb;
130                if !psf.is_finite() {
131                    self.failed = true;
132                    return Err(ProductsError::InvalidTaylorMomentReduction);
133                }
134                if term < 2 {
135                    let residual = &mut self.residual[term * cells + pixel];
136                    *residual += factor * residual_pb;
137                    if !residual.is_finite() {
138                        self.failed = true;
139                        return Err(ProductsError::InvalidTaylorMomentReduction);
140                    }
141                }
142            }
143        }
144        self.weight = total;
145        Ok(())
146    }
147
148    /// Normalize the output moments, blanking pixels at or below the PB cutoff.
149    ///
150    /// # Errors
151    ///
152    /// No positive weight was added, a previous channel failed, or the
153    /// normalization generated a non-finite value.
154    pub fn finish(self) -> Result<(), ProductsError> {
155        if self.failed || self.weight <= 0.0 {
156            return Err(ProductsError::InvalidTaylorMomentReduction);
157        }
158        let cells = self.mean_pb.len();
159        for (index, value) in self
160            .psf
161            .iter_mut()
162            .chain(self.residual.iter_mut())
163            .enumerate()
164        {
165            let beam = self.mean_pb[index % cells];
166            *value = if beam > self.cutoff {
167                *value / self.weight / beam
168            } else {
169                0.0
170            };
171            if !value.is_finite() {
172                return Err(ProductsError::InvalidTaylorMomentReduction);
173            }
174        }
175        for value in self.sumwt {
176            *value /= self.weight;
177        }
178        Ok(())
179    }
180}
