2//! Stage fully defined weight spectra in tiled storage before publishing them.
3
4use std::collections::BTreeMap;
5
6use casa_core::{ArrayValue, PrimitiveType};
7use casa_ms::TARGET_BATCH_BYTES;
8use casa_ms::{MsError, MsResult};
9use casa_tables::streamed_tiled::{
10    StreamedTiledShapeCubeLayout, StreamedTiledShapeValueType, StreamingTiledShapeWriter,
11};
12use casa_tables::{ColumnSchema, Table, TableError, TableWriteLock};
13
14pub(crate) const WEIGHT_SPECTRUM: &str = "WEIGHT_SPECTRUM";
15pub(crate) const SIGMA_SPECTRUM: &str = "SIGMA_SPECTRUM";
16
17pub(crate) fn has_column(table: &Table, name: &str) -> bool {
18    table
19        .schema()
20        .is_some_and(|schema| schema.contains_column(name))
21}
22
23#[derive(Default)]
24pub(crate) struct NewSpectra {
25    writers: BTreeMap<&'static str, StreamingTiledShapeWriter>,
26    _staging: Option<tempfile::TempDir>,
27}
28
29impl NewSpectra {
30    pub(crate) fn prepare(
31        main: &Table,
32        columns: &[&'static str],
33        row_shapes: impl Iterator<Item = [usize; 2]>,
34    ) -> MsResult<Self> {
35        let added: Vec<_> = columns
36            .iter()
37            .copied()
38            .filter(|name| {
39                matches!(*name, WEIGHT_SPECTRUM | SIGMA_SPECTRUM) && !has_column(main, name)
40            })
41            .collect();
42        if added.is_empty() {
43            return Ok(Self::default());
44        }
45        let mut counts = BTreeMap::<_, usize>::new();
46        for shape in row_shapes {
47            *counts.entry(shape).or_default() += 1;
48        }
49        let layouts: Vec<_> = counts
50            .into_iter()
51            .map(|([correlations, channels], count)| {
52                StreamedTiledShapeCubeLayout::new(
53                    vec![correlations, channels],
54                    count,
55                    vec![correlations, channels.min(100), 100],
56                )
57            })
58            .collect();
59        let path = main.path().ok_or_else(|| {
60            MsError::InvalidInput("weight spectrum needs an on-disk MeasurementSet".into())
61        })?;
62        let staging = tempfile::Builder::new()
63            .prefix(".weight-spectrum-")
64            .tempdir_in(path)
65            .map_err(|error| {
66                TableError::Storage(format!(
67                    "stage weight spectrum for {}: {error}",
68                    path.display()
69                ))
70            })?;
71        let big_endian = main
72            .stored_endian_format()?
73            .unwrap_or_default()
74            .is_big_endian();
75        // The writer may need one three-element row-map record per row if
76        // different cell shapes alternate. Allow this with the tile buffers.
77        let budget = TARGET_BATCH_BYTES.saturating_add(main.row_count().saturating_mul(12));
78        let writers = added
79            .into_iter()
80            .map(|name| {
81                let writer = StreamingTiledShapeWriter::create(
82                    staging.path().join(name),
83                    StreamedTiledShapeValueType::Float32,
84                    layouts.clone(),
85                    budget,
86                    big_endian,
87                )
88                .map_err(TableError::from)?;
89                Ok((name, writer))
90            })
91            .collect::<MsResult<_>>()?;
92        Ok(Self {
93            writers,
94            _staging: Some(staging),
95        })
96    }
97
98    pub(crate) fn contains(&self, name: &str) -> bool {
99        self.writers.contains_key(name)
100    }
101
102    pub(crate) fn append(&mut self, name: &str, value: &ArrayValue) -> MsResult<bool> {
103        let Some(writer) = self.writers.get_mut(name) else {
104            return Ok(false);
105        };
106        let ArrayValue::Float32(value) = value else {
107            return Err(MsError::InvalidInput(format!(
108                "{name} requires float32 cells"
109            )));
110        };
111        let data = value
112            .as_slice_memory_order()
113            .ok_or_else(|| MsError::InvalidInput(format!("{name} requires contiguous cells")))?;
114        writer
115            .push_f32_row(value.shape(), data)
116            .map_err(TableError::from)?;
117        Ok(true)
118    }
119
120    pub(crate) fn append_unity(&mut self, shape: [usize; 2]) -> MsResult<()> {
121        if !self.writers.is_empty() {
122            let ones = vec![1.0; shape[0] * shape[1]];
123            for writer in self.writers.values_mut() {
124                writer
125                    .push_f32_row(&shape, &ones)
126                    .map_err(TableError::from)?;
127            }
128        }
129        Ok(())
130    }
131
132    pub(crate) fn publish(self, main: &mut Table, lock: &mut TableWriteLock) -> MsResult<()> {
133        let finished = self
134            .writers
135            .into_iter()
136            .map(|(name, writer)| Ok((name, writer.finish().map_err(TableError::from)?)))
137            .collect::<MsResult<Vec<_>>>()?;
138        for (name, spectrum) in finished {
139            let comment = if name == WEIGHT_SPECTRUM {
140                "weight spectrum"
141            } else {
142                "sigma spectrum"
143            };
144            main.add_column(
145                ColumnSchema::array_variable(name, PrimitiveType::Float32, Some(2))
146                    .with_comment(comment),
147                None,
148            )?;
149            lock.record_write();
150            main.prepare_write()
151                .add_streamed_tiled_shape_column(name, spectrum)?;
152        }
153        Ok(())
154    }
155}
