6pub(crate) type PatchGroup = [Vec<i32>; 6];

8pub(super) fn groups(
9    ms: &MeasurementSet,
10    plan: &ApplyCalibrationTablePlan,
11    selected: &[ApplyRowPlan],
12    contexts: &HashMap<(i32, i32, i32, i32), BTreeSet<i32>>,
13) -> Result<Vec<PatchGroup>, ApplyPlanError> {
14    let library = plan.spec.callibrary.as_ref().expect("library");
15    let raw = resolve_calibration_selection(ms, &library.selection).map_err(|error| {
16        ApplyPlanError::Callibrary {
17            message: error.to_string(),
18        }
19    })?;
20    let mut indices = raw.row_indices().collect::<Vec<_>>();
21    indices.sort_unstable();
22    let rows = selected
23        .iter()
24        .filter(|row| indices.binary_search(&row.row_index).is_ok())
25        .collect::<Vec<_>>();
26    if rows.is_empty() {
27        return Ok(Vec::new());
28    }
29    let requested = [
30        library
31            .selection
32            .observation
33            .as_ref()
34            .map(|_| unique_i32(rows.iter().map(|row| row.observation_id))),
35        library
36            .selection
37            .scan
38            .as_ref()
39            .map(|_| unique_i32(rows.iter().map(|row| row.scan_number))),
40        library
41            .selection
42            .field
43            .as_ref()
44            .map(|_| unique_i32(rows.iter().map(|row| row.field_id))),
45        if library.selection.intent.is_some() {
46            Some(unique_i32(
47                rows.iter()
48                    .map(|row| get_i32(ms.main_table(), row.row_index, "STATE_ID"))
49                    .collect::<Result<Vec<_>, _>>()?,
50            ))
51        } else {
52            None
53        },
54    ];
55    let observation_count = ms
56        .observation()
57        .map_err(|source| ApplyPlanError::Selection { source })?
58        .row_count() as i32;
59    let field_count = ms
60        .field()
61        .map_err(|source| ApplyPlanError::Selection { source })?
62        .row_count() as i32;
63    let antenna_count = ms
64        .antenna()
65        .map_err(|source| ApplyPlanError::Selection { source })?
66        .row_count() as i32;
67    let scan_count = selected
68        .iter()
69        .map(|row| row.scan_number)
70        .max()
71        .unwrap_or(-1)
72        + 1;
73    let windows = unique_i32(rows.iter().map(|row| row.data_spw_id));
74    let mappings = resolve_spw_mapping(&plan.spec, &[], 0, &windows)?;
75    let mut groups = Vec::new();
76    for (&(observation, scan, field, window), antennas) in contexts {
77        let fields = if !library.field_map.is_empty() {
78            restrict(
79                library
80                    .field_map
81                    .iter()
82                    .enumerate()
83                    .filter(|(_, target)| **target == field)
84                    .map(|(id, _)| id as i32)
85                    .collect(),
86                requested[2].as_deref(),
87            )
88        } else if !plan.resolved_nearest_gainfields.is_empty() {
89            restrict(
90                plan.resolved_nearest_gainfields
91                    .iter()
92                    .filter(|mapping| mapping.calibration_field_id == field)
93                    .map(|mapping| mapping.measurement_set_field_id)
94                    .collect(),
95                requested[2].as_deref(),
96            )
97        } else if plan.resolved_gainfields.len() == 1 && plan.spec.gainfield.is_some() {
98            restrict((0..field_count).collect(), requested[2].as_deref())
99        } else {
100            requested[2].clone().unwrap_or_else(|| vec![-1])
101        };
102        let group = [
103            reverse(
104                &library.observation_map,
105                observation,
106                observation_count,
107                requested[0].as_deref(),
108            ),
109            reverse(&library.scan_map, scan, scan_count, requested[1].as_deref()),
110            fields,
111            requested[3].clone().unwrap_or_else(|| vec![-1]),
112            mappings
113                .iter()
114                .filter(|mapping| mapping.calibration_spw_id == window)
115                .map(|mapping| mapping.data_spw_id)
116                .collect(),
117            (0..antenna_count)
118                .filter(|&id| {
119                    let target = library
120                        .antenna_map
121                        .get(id as usize)
122                        .or_else(|| library.antenna_map.last())
123                        .copied()
124                        .filter(|id| *id >= 0)
125                        .unwrap_or(id);
126                    antennas.contains(&target)
127                })
128                .collect(),
129        ];
130        if group.iter().all(|axis| !axis.is_empty()) {
131            groups.push(group);
132        }
133    }
134    Ok(groups)
135}

137fn reverse(
138    map: &CallibraryContextMap,
139    target: i32,
140    count: i32,
141    requested: Option<&[i32]>,
142) -> Vec<i32> {
143    match map {
144        CallibraryContextMap::All => requested.map_or_else(|| vec![-1], <[i32]>::to_vec),
145        CallibraryContextMap::SelfMap => restrict(
146            if (0..count).contains(&target) {
147                vec![target]
148            } else {
149                vec![]
150            },
151            requested,
152        ),
153        CallibraryContextMap::Indices(ids) => restrict(
154            ids.iter()
155                .enumerate()
156                .filter(|(_, id)| **id == target)
157                .map(|(id, _)| id as i32)
158                .collect(),
159            requested,
160        ),
161    }
162}
163
164fn restrict(mut ids: Vec<i32>, requested: Option<&[i32]>) -> Vec<i32> {
165    if let Some(requested) = requested {
166        ids.retain(|id| requested.contains(id));
167    }
168    ids
169}

171pub(super) fn overlap(left: &PatchGroup, right: &PatchGroup) -> bool {
172    left.iter().zip(right).all(|(a, b)| {
173        a.iter().any(|id| *id < 0)
174            || b.iter().any(|id| *id < 0)
175            || a.iter().any(|id| b.contains(id))
176    })
177}
