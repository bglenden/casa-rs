3//! Exact failure reasons for continuum product construction and output.
4
5use std::path::PathBuf;
6
7use casa_imaging_model::{ProductRole, ProductTerm};
8use thiserror::Error;
9
10/// Why product planning, generation or writing failed.
11#[derive(Debug, Error)]
12pub enum ProductsError {
13    /// A product image could not be created, written or saved.
14    #[error("writing {}: {source}", path.display())]
15    Image {
16        /// The image's published path.
17        path: PathBuf,
18        /// What the image layer reported.
19        #[source]
20        source: casa_images::ImageError,
21    },
22    /// Staging or replacing a product image failed.
23    #[error("{operation} {}: {source}", path.display())]
24    Io {
25        /// What the writer was doing.
26        operation: &'static str,
27        /// The directory or image it was doing it to.
28        path: PathBuf,
29        /// What the system reported.
30        #[source]
31        source: std::io::Error,
32    },
33    /// A product window exceeded its shape or admitted capacity.
34    #[error("product window exceeds its admitted shape or capacity")]
35    InvalidWindow,
36    /// Ordered product coverage was missing, repeated, or reordered.
37    #[error("product window coverage requires channel {expected}, found {actual}")]
38    WindowCoverage {
39        /// Next required channel, or complete channel count at output finish.
40        expected: usize,
41        /// Actual start or completed channel count.
42        actual: usize,
43    },
44    /// An authoritative Normal State window could not be read.
45    #[error(transparent)]
46    NormalAccess(#[from] casa_imaging_deconvolution::NormalStateError),
47    /// An authoritative model plane could not be read for product generation.
48    #[error(transparent)]
49    ModelAccess(#[from] casa_imaging_deconvolution::ModelLifecycleError),
50    /// A direction-dependent sensitivity plane has no positive value: no
51    /// weighted sample reached it.
52    #[error("the sensitivity plane has no positive support")]
53    UnsupportedSensitivity,
54    /// Direction-dependent image units need flat-noise or flat-sky
55    /// normalization.
56    #[error("direction-dependent image units need flat-noise or flat-sky normalization")]
57    UnsupportedNormalization,
58    /// Production controls were outside their validated ranges.
59    #[error("continuum production controls are invalid")]
60    InvalidControls,
61    /// The compiled problem asks for a product rule these products do not
62    /// implement.
63    #[error("the compiled problem asks for a product rule the products do not implement")]
64    UnsupportedProblem,
65    /// A supplied Taylor primary beam is invalid or does not belong to a Taylor family.
66    #[error(
67        "a supplied Taylor primary beam must be finite, non-negative, and belong to a Taylor family"
68    )]
69    InvalidTaylorPrimaryBeam,
70    /// The supplied cube correction beam or its normal state is invalid.
71    #[error(
72        "a cube Taylor primary beam needs a finite non-negative plane and a single-domain, single-polarization cube"
73    )]
74    InvalidCubeTaylorPrimaryBeam,
75    /// The saved primary beam contains an invalid value.
76    #[error("the saved mosaic primary beam contains a negative or non-finite value")]
77    InvalidSavedMosaicBeam,
78    /// The saved mosaic PB does not match the dimensions of the normal state.
79    #[error("saved mosaic PB shape {saved:?} does not match the image grid {expected:?}")]
80    SavedMosaicBeamShape {
81        /// Saved CASA image axes.
82        saved: [usize; 4],
83        /// Expected CASA image axes.
84        expected: [usize; 4],
85    },
86    /// The saved primary beam image or its default mask could not be read.
87    #[error("reading saved mosaic primary beam {}: {source}", path.display())]
88    SavedMosaicBeam {
89        /// Path to the saved CASA image.
90        path: PathBuf,
91        /// Image or mask read error.
92        #[source]
93        source: casa_images::ImageError,
94    },
95    /// A supplied Taylor residual normalization divisor is invalid.
96    #[error("Taylor residual normalization needs a positive finite divisor and a Taylor family")]
97    InvalidTaylorResidualSumWeight,
98    /// A cube-to-Taylor image reduction received invalid data.
99    #[error(
100        "Taylor moment reduction needs valid matching planes, frequencies, weights and positive support"
101    )]
102    InvalidTaylorMomentReduction,
103    /// A cube-to-Taylor primary-beam reduction received invalid data.
104    #[error(
105        "Taylor primary-beam reduction needs matching finite non-negative planes and weights and a positive finite total"
106    )]
107    InvalidTaylorBeamReduction,
108    /// A PB product or PB-masked member was planned without a primary-beam
109    /// model in the controls.
110    #[error("a primary-beam product or mask needs a primary-beam model")]
111    MissingPrimaryBeamModel,
112    /// Restoring beams that do not form the channel × Stokes grid casacore
113    /// requires of an image's beam set (`ImageInfo::checkBeamSet`).
114    #[error("{beams} restoring beams do not form a grid of {channels} channels")]
115    BeamGrid {
116        /// Beams given for the image's planes.
117        beams: usize,
118        /// The image's channels.
119        channels: usize,
120    },
121    /// The selected beam law cannot represent an output channel's frequency.
122    #[error(
123        "primary-beam model {model:?} does not support output channel {output_channel} at {frequency_hz} Hz"
124    )]
125    UnsupportedPrimaryBeamFrequency {
126        /// Explicitly selected primary-beam law.
127        model: crate::AnalyticPrimaryBeamModel,
128        /// Zero-based output channel, after spectral coordinate conversion.
129        output_channel: usize,
130        /// Output channel centre, in Hz.
131        frequency_hz: f64,
132    },
133    /// The product list publishes a role the products cannot produce.
134    #[error("product role {role:?} cannot be produced")]
135    UnsupportedProductRole {
136        /// The first unsupported role encountered.
137        role: ProductRole,
138    },
139    /// A Taylor term other than the zeroth coefficient was requested.
140    #[error("Taylor term {term:?} exceeds the nterms=1 continuum algorithm catalog")]
141    UnsupportedTaylorTerm {
142        /// The offending Taylor term.
143        term: ProductTerm,
144    },
145    /// A Taylor run's principal sum of weights, normal or published, is not
146    /// positive and finite. Every Stokes plane takes the first gridded
147    /// correlation's sum (`MultiTermFTNew`), so this is a run whose first
148    /// correlation has no weight: CASA writes every plane as zeros, and
149    /// casa-rs refuses it until #1109.
150    #[error(
151        "Taylor products need a positive principal sum of weights, not {sum_weight}: every \
152         Stokes plane takes the first gridded correlation's sum, which has no weight; casa-rs \
153         does not image this yet (#1109)"
154    )]
155    PrincipalSumWeight {
156        /// The principal sum of weights.
157        sum_weight: f64,
158    },
159    /// A produced payload length disagreed with its declared shape.
160    #[error("member payload requires {expected} values but carries {actual}")]
161    PayloadLengthMismatch {
162        /// Shape-derived value count.
163        expected: usize,
164        /// Actual payload length.
165        actual: usize,
166    },
167    /// Beam fitting failed for products that carry fitted beam metadata.
168    #[error("restoring-beam fitting failed: {0}")]
169    BeamFitFailed(String),
170    /// Solver arithmetic produced a non-finite product value.
171    #[error("product arithmetic generated a non-finite value")]
172    GeneratedNonfinite,
173    /// Checked product-generation resource arithmetic exceeded `u64`.
174    #[error("continuum generation demand overflowed while calculating {0}")]
175    ResourceDemandOverflow(&'static str),
176}
