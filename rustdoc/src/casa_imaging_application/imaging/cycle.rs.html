2//! The loop of major and minor cycles on the major-cycle pass.
3//!
4//! The initial pass grids the data and the PSF; each later pass grids the
5//! residual `V − A·m` of the model the preceding minor cycle produced.
6//! Imaging weights are generated once, before the initial pass, and reused
7//! by every pass. The final pass also writes the model column when the run
8//! asks for it.

175/// Run every major and minor cycle of `inputs.problem`.
176///
177/// The loop is tclean's: after each major cycle the controller decides from
178/// the fresh residual whether cleaning is complete (`cleanComplete`);
179/// otherwise a minor cycle runs under its controls and a major cycle
180/// refreshes the residual. A minor cycle that cleans nothing ends the run
181/// without another major cycle. Each cycle, and each minor cycle's
182/// preparation (its masks and residual statistics), is a phase of
183/// `summary`, so every step that can fail is inside one; `summary` also
184/// takes the resolved precision, the minor-cycle totals as each cycle
185/// completes and the stop reason. A cancelled run stops at the next block
186/// boundary of a pass or before the next phase.
187pub(crate) fn run(
188    inputs: ImagingInputs<'_>,
189    summary: &mut RunSummary,
190) -> Result<ImagingOutcome, ImagingError> {
191    let problem = inputs.problem;
192    let controls = problem.reconstruction().controls();
193    let cleaning = controls.max_minor_iterations() > 0;
194    let setup = minor_setup(problem, inputs.image_response, &inputs.masks);
195    let mut mask_plans = inputs.masks.clone();
196    let cancel = inputs.cancel.clone();
197    let diagnostics = inputs.diagnostics;
198    // What a minor cycle holds on a team of so many workers, from the
199    // layout the first pass forms, before any image exists (#737): the masks
200    // it keeps and the larger of its two halves.
201    let layout = MinorLayout::planned(problem);
202    let iterations = Controller::new(&controls).cycle_iterations();
203    let minor = |workers: usize| {
204        if cleaning {
205            2 * layout.mask_bytes()
206                + prepare_bytes_for(&layout, &mask_plans, &setup, true, workers)
207                    .max(run_bytes_for(&layout, &setup, iterations, 0, workers))
208        } else {
209            0
210        }
211    };
212    // What a refresh pass holds beside its wave (#763 review): every cycle
213    // before it keeps its record, up to the last refresh `nmajor` allows,
214    // and the minor cycle's terms, which its solve holds too (#778 review).
215    let refresh = cleaning.then(|| RefreshDemand {
216        masks: 2 * layout.mask_bytes(),
217        cached: layout.refresh_bytes(&setup),
218        results: layout.result_bytes(&setup, iterations),
219        records: layout.record_bytes() * controls.maximum_major_cycles().unwrap_or(1).max(1) as u64,
220    });
221    // Operators, the workers, the source, the imaging weights and the paged
222    // cube state. The choice of workers is recorded as it is made, so a
223    // setup that then fails still reports it.
224    let mut chosen = None;
225    let opened = run_phase("setup", &cancel, diagnostics, summary, || {
226        Run::open(inputs, (&minor, refresh), &mut chosen)
227    });
228    if let Some(workers) = chosen {
229        summary.workers = workers.cores.workers;
230        summary.workers_policy = workers.policy.workers;
231        summary.workers_reason = workers.reason;
232    }
233    let mut run = opened?;
234    summary.precision = run
235        .domains
236        .first()
237        .map(|domain| domain.operator.precision());
238    let major = run_phase("major cycle 1", &cancel, diagnostics, summary, || {
239        run.initial(!cleaning)
240    })?;
241    let mut outcome = ImagingOutcome::initial(major, &run);
242    if !cleaning {
243        outcome.retained = std::mem::replace(&mut run.retained, Reservation::none());
244        return Ok(outcome);
245    }
246    let mut controller = Controller::new(&controls);
247    let mut cache = PsfCache::default();
248    // What outlives each minor cycle: the masks it forms (the latest, and the
249    // one an automatic mask evolves from) and a Clark refresh it caches.
250    let mut held = Reservation::none();
251    // tclean skips the major cycle after a minor cycle that cleans nothing,
252    // updates the mask and still asks `hasConverged` for the stop it reports.
253    let mut cleaned_nothing = false;
254    loop {
255        let ordinal = outcome.minor_cycles.len() + 1;
256        let name = format!("minor-cycle preparation {ordinal}");
257        // A plane preparation keeps for the solve stays charged until the
258        // solve, which the minor cycle's charge then leaves out; the solve
259        // drops the plane, and the charge goes with the minor cycle.
260        let (prepared, kept) = run_phase(name, &cancel, diagnostics, summary, || {
261            if ordinal == 1 {
262                held = run.admit("reconstruction masks", 2 * mask_bytes(&outcome.scientific))?;
263            }
264            let mut preparing = run.admit(
265                "minor-cycle preparation",
266                prepare_bytes(
267                    &outcome.scientific,
268                    &mask_plans,
269                    &setup,
270                    &cache,
271                    run.teams.compute().workers(),
272                ),
273            )?;
274            let prepared = prepare_minor_cycle(
275                &outcome.scientific,
276                &mask_plans,
277                &setup,
278                &mut cache,
279                run.teams.compute(),
280            )?;
281            let kept = preparing.split(prepared.retained_bytes());
282            Ok::<_, ImagingError>((prepared, kept))
283        })?;
284        let statistics = prepared.statistics;
285        if diagnostics.science_trace {
286            trace::residual(outcome.major_cycle_count, &statistics);
287        }
288        let stop = controller.clean_complete(&statistics);
289        if let Some(stop) = stop {
290            tracing::info!("imaging stopped: {stop:?} (stopcode {})", stop.code());
291        }
292        if cleaned_nothing {
293            outcome.stop = stop;
294            outcome.masks = Some(ReconstructionMaskSet::Domains(prepared.masks().clone()));
295            break;
296        }
297        if stop.is_some() {
298            outcome.stop = stop;
299            outcome
300                .masks
301                .get_or_insert_with(|| ReconstructionMaskSet::Domains(prepared.masks().clone()));
302            break;
303        }
304        let cycle_controls = controller.cycle_controls(&statistics);
305        if diagnostics.science_trace {
306            trace::cycle_controls(ordinal, &cycle_controls);
307        }
308        let started = Instant::now();
309        let name = format!("minor cycle {ordinal}");
310        let (minor, next_plans) = run_phase(name, &cancel, diagnostics, summary, || {
311            let cached = cache.refresh_bytes(&outcome.scientific);
312            let mut solving = run.admit(
313                "minor cycle",
314                run_bytes(
315                    &outcome.scientific,
316                    &setup,
317                    &cycle_controls,
318                    &cache,
319                    run.teams.compute().workers(),
320                )
321                .saturating_sub(prepared.retained_bytes()),
322            )?;
323            let minor = run_minor_cycle(
324                prepared,
325                &outcome.scientific,
326                &setup,
327                &cycle_controls,
328                &mut cache,
329                run.teams.compute(),
330                (&run.host, &run.policy),
331            )?;
332            let refresh = cache.refresh_bytes(&outcome.scientific);
333            if refresh > cached {
334                held.join(solving.split(refresh - cached));
335            }
336            let plans = next_masks(
337                &mask_plans,
338                &minor,
339                ordinal,
340                cycle_controls.threshold_reached,
341            )?;
342            Ok::<_, ImagingError>((minor, plans))
343        })?;
344        drop(kept);
345        mask_plans = next_plans;
346        let global_threshold = controls
347            .threshold_jy_per_beam()
348            .max(statistics.nsigma_threshold);
349        // The outcome keeps every cycle's record, with a copy of its mask's
350        // support and its traced components (#706).
351        held.join(run.admit("minor-cycle record", record_bytes(&minor))?);
352        let cycle = account_minor_cycle(
353            &mut outcome,
354            &mut controller,
355            &minor,
356            &cycle_controls,
357            global_threshold,
358            started,
359        );
360        summary.minor_cycles = cycle;
361        summary.minor_iterations = outcome.total_minor_iterations;
362        if diagnostics.science_trace {
363            trace::minor_cycle(&outcome.minor_cycles[cycle - 1]);
364        }
365        let last = controller.budget_spent() || controller.last_cycle(&statistics);
366        if minor.summary.iterations == 0 {
367            cleaned_nothing = true;
368            continue;
369        }
370        let name = format!("major cycle {}", outcome.major_cycle_count + 1);
371        let scientific = outcome.scientific;
372        let major = run_phase(name, &cancel, diagnostics, summary, || {
373            run.refresh(scientific, (minor.terms, minor.results), last)
374        })?;
375        controller.end_major_cycle();
376        outcome.scientific = major.completion;
377        outcome.visibility_products = major.visibility;
378        outcome.major_cycle_count += 1;
379    }
380    summary.stop = outcome.stop;
381    if run.visibility_write.is_some() && outcome.visibility_products.is_none() {
382        outcome = run_phase("model prediction", &cancel, diagnostics, summary, || {
383            predict_final_model(&mut run, outcome)
384        })?;
385    }
386    outcome.planes_per_wave = run.planes_per_wave;
387    outcome.retained = std::mem::replace(&mut run.retained, Reservation::none());
388    outcome.retained.join(held);
389    Ok(outcome)
390}

392/// Fold one minor cycle into the run's totals and its controller
393/// (`mergeMinorCycleSummary`), record and log it; returns its ordinal.
394fn account_minor_cycle(
395    outcome: &mut ImagingOutcome,
396    controller: &mut Controller,
397    minor: &MinorCycleOutcome,
398    cycle_controls: &CycleControls,
399    global_threshold: f64,
400    started: Instant,
401) -> usize {
402    let entering = (
403        outcome.total_minor_iterations,
404        outcome.total_actual_minor_iterations,
405    );
406    controller.record_minor_cycle(minor.summary.iterations, minor.summary.peak);
407    outcome.total_minor_iterations = controller.iterations();
408    outcome.total_actual_minor_iterations += minor.summary.components;
409    let cycle = outcome.minor_cycles.len() + 1;
410    let record = minor_cycle_record(
411        cycle,
412        minor,
413        cycle_controls,
414        global_threshold,
415        entering,
416        (
417            outcome.total_minor_iterations,
418            outcome.total_actual_minor_iterations,
419        ),
420    );
421    tracing::info!(
422        "imaging minor cycle {cycle}: {} iterations ({} total), peak {:.6} -> {:.6} Jy, \
423         cycle threshold {:.6} Jy, stop {:?}, {:.2} s",
424        record.iterations,
425        record.total_iterations,
426        record.initial_peak_flux,
427        record.final_peak_flux,
428        record.effective_threshold,
429        record.stop_reason,
430        started.elapsed().as_secs_f64(),
431    );
432    outcome.minor_cycles.push(record);
433    outcome.masks = Some(ReconstructionMaskSet::Domains(minor.masks.clone()));
434    cycle
435}

437/// Predict the final model once more to write the model column: the pass
438/// that ended the run was not known to be the last when it ran.
439fn predict_final_model(
440    run: &mut Run<'_>,
441    mut outcome: ImagingOutcome,
442) -> Result<ImagingOutcome, ImagingError> {
443    let major = run.refresh(outcome.scientific, (Vec::new(), Reservation::none()), true)?;
444    outcome.scientific = major.completion;
445    outcome.visibility_products = major.visibility;
446    outcome.major_cycle_count += 1;
447    Ok(outcome)
448}

450/// The minor cycle's view of the run.
451fn minor_setup(
452    problem: &CompiledProblem,
453    response: Option<MinorCycleImageResponse>,
454    masks: &ImageDomainReconstructionMaskPlans,

469impl<'a> Run<'a> {
470    /// The operators, worker teams, source and imaging weights of a run.
471    /// The teams take the most workers, up to the policy's, whose charges
472    /// fit beside the operators and the source ([`setup::fit_workers`],
473    /// #737); a paged kernel set is admitted for the count chosen, and each
474    /// later phase is admitted as it starts. `minor` is what a minor cycle
475    /// holds on a team of so many workers and `refresh` what a refresh pass
476    /// holds beside its wave. `chosen` receives the choice as it is made.
477    fn open(
478        inputs: ImagingInputs<'a>,
479        (minor, refresh): (&dyn Fn(usize) -> u64, Option<RefreshDemand>),
480        chosen: &mut Option<WorkerChoice>,
481    ) -> Result<Self, ImagingError> {
482        let problem = inputs.problem;
483        let backend = inputs.backend;
484        let correlations = selected_correlations(problem)?;
485        let selected = inputs
486            .access
487            .into_deferred()
488            .open(problem)
489            .map_err(|error| ImagingError::Observation(Box::new(error)))?;
490        let dish_classes = selected.antenna_response_classes();
491        let (host, policy) = (inputs.host, inputs.policy);
492        // The passes' workers hold the kernels: at most the policy's.
493        let ceiling = policy.cores(&host);
494        let admission = KernelAdmission {
495            host: &host,
496            policy: &policy,
497            workers: ceiling.streaming_workers,
498        };
499        // One operator per chart: each image domain, or each facet of a
500        // faceted one.
501        let charts = charts(problem);
502        let mut domains = charts
503            .iter()
504            .map(|chart| {
505                domain_operator(
506                    problem,
507                    chart.extent(&problem.geometry().domains()[chart.domain])?,
508                    &correlations,
509                    backend,
510                    inputs.grid_precision,
511                    inputs.aw_catalog.as_ref(),
512                    &dish_classes,
513                    admission,
514                )
515            })
516            .collect::<Result<Vec<_>, _>>()?;
517        let weight_image = domains.iter().any(|domain| domain.weight_image);
518        if inputs.visibility_write.is_some() && problem.geometry().domains().len() > 1 {
519            return Err(ImagingError::Unsupported {
520                reason: "visibilities are written back for one image domain",
521            });
522        }
523        let cancel = inputs.cancel;
524        let channel_local = matches!(domains[0].operator.basis(), Basis::ChannelLocal { .. });
525        let mut source = MeasurementSetSource::new(
526            problem,
527            selected,
528            domains[0].operator.basis().planes(),
529            plane_bounds(problem),
530            domains
531                .iter()
532                .any(|domain| domain.operator.cf().pointing_ramp()),
533            dish_classes,
534        );
535        let stitch = ChartStitch::new(problem, charts.clone());
536        let visibility_write_bytes = inputs
537            .visibility_write
538            .as_ref()
539            .map(VisibilityWriteTarget::retained_bytes)
540            .transpose()
541            .map_err(|error| ImagingError::Pass(PassError::VisibilityWrite(error)))?
542            .unwrap_or(0);
543        let initial = initial_modes(weight_image);
544        let demand = SetupDemand {
545            problem,
546            domains: &domains,
547            stitch: &stitch,
548            block: source.maximum_block(),
549            native_spacing_hz: native_spacing_hz(problem),
550            backend,
551            modes: initial,
552            // A cube's model is paged with its state.
553            model_bytes: if channel_local {
554                0
555            } else {
556                ModelLifecycle::new(problem, ModelStoragePlan::resident(usize::MAX)?)
557                    .resident_bytes()
558            },
559            write: inputs.visibility_write.as_ref().map(|_| {
560                let (modes, with_model) = write_pass(problem, initial);
561                (modes, with_model, visibility_write_bytes)
562            }),
563            spill_directory: inputs.spill_directory,
564            weight_image,
565            minor,
566            refresh,
567            kept_weights: kept_weight_bytes(
568                problem,
569                &domains,
570                source.maximum_block().channels(),
571                free_memory(&host, &policy),
572            )
573            .map_or(0, |(_, bytes)| bytes),
574            free: free_memory(&host, &policy),
575        };
576        let choice = setup::fit_workers(ceiling, demand.free, |cores| demand.bytes(cores))?;
577        *chosen = Some(choice.clone());
578        if let Some(reason) = &choice.reason {
579            tracing::warn!(
580                "imaging on {} workers, not the resource policy's {}: {reason}",
581                choice.cores.workers,
582                choice.policy.workers
583            );
584        }
585        tracing::info!(
586            "workers: {} computing, {} streaming beside the source's producer",
587            choice.cores.workers,
588            choice.cores.streaming_workers
589        );
590        let formed = WorkerTeams::new(choice.cores).map_err(ImagingError::Pass)?;
591        let teams = inputs.teams.get_or_init(move || formed);
592        // An automatic AW cell cache takes what the chosen workers leave
593        // free (#449).
594        let mut spare = demand.free.saturating_sub(demand.bytes(choice.cores)?);
595        for domain in &mut domains {
596            domain.admit_kernels(admission, choice.cores.streaming_workers, &mut spare)?;
597        }
598        let streaming = teams.streaming();
599        let started = Instant::now();
600        let (weighting, weights) = setup::weighting(
601            problem,
602            &domains[0],
603            &mut source,
604            streaming,
605            &cancel,
606            (&host, &policy),
607        )?;
608        tracing::info!(
609            "imaging weights: {} workers, {:.2} s",
610            streaming.workers(),
611            started.elapsed().as_secs_f64()
612        );
613        // The minor cycles, the widest team, access the cube state's planes.
614        let cube = channel_local
615            .then(|| {
616                setup::cube_state(
617                    problem,
618                    inputs.spill_directory,
619                    (teams.compute().workers(), weight_image),
620                    &host,
621                    &policy,
622                )
623            })
624            .transpose()?;
625        let lifecycle = ModelLifecycle::new(
626            problem,
627            match &cube {
628                Some(cube) => cube.model_storage()?,
629                None => ModelStoragePlan::resident(usize::MAX)?,
630            },
631        );
632        let run = Self {
633            problem,
634            domains,
635            stitch,
636            charts,
637            source,
638            weighting,
639            _weights: weights,
640            teams,
641            cancel,
642            host,
643            policy,
644            cube,
645            lifecycle,
646            native_spacing_hz: native_spacing_hz(problem),
647            backend,
648            metal_stage_timing: inputs.diagnostics.metal_stage_timing,
649            visibility_write: inputs.visibility_write,
650            visibility_write_bytes,
651            planes_per_wave: None,
652            weight_image,
653            retained: Reservation::none(),
654            kept_weights: None,
655            row_work: None,
656        };
657        if run.visibility_write.is_some() {
658            // The pass that writes must hold every plane, and is refused now
659            // rather than after the cycles that precede it.
660            let (modes, with_model) = write_pass(problem, initial);
661            if run
662                .admit_pass(modes, with_model, run.visibility_write_bytes, true)?
663                .value
664                .residency
665                != Residency::All
666            {
667                return Err(ImagingError::Pass(PassError::VisibilityWriteWaves));
668            }
669        }
670        Ok(run)
671    }

673    /// The initial major cycle: data and PSF, from an empty model; it writes
674    /// visibilities when it is also final.
675    fn initial(&mut self, last: bool) -> Result<Major, ImagingError> {
676        self.measure_row_work()?;
677        // The model is admitted with the pass, before it is allocated.
678        let Admitted {
679            value: plan,
680            reservation: pass,
681        } = self.admit_pass(
682            self.initial_modes(),
683            false,
684            self.lifecycle.resident_bytes() + self.write_bytes(last),
685            self.writes(last),
686        )?;
687        let model = self
688            .lifecycle
689            .prepare_final_model(self.lifecycle.initial_empty()?, [])?;
690        let cycle = MajorCycle::initial(self.problem, model, self.normal_storage(plan.residency)?)?;
691        let major = self.reconcile(cycle, true, last, plan)?;
692        self.retain(&major, pass)?;
693        Ok(major)
694    }

696    /// The major cycle after a minor cycle: the residual of the model updated
697    /// by `terms`.
698    fn refresh(
699        &mut self,
700        completion: MajorCycleCompletion,
701        (terms, results): (Vec<ModelDeltaTerm>, Reservation),
702        last: bool,
703    ) -> Result<Major, ImagingError> {
704        // The retained charge still covers the normal state and model the
705        // pass refreshes; the pass adds its scratch and the new residual,
706        // and the minor cycle's charge covers its terms until the model
707        // holds them.
708        let (normal_state, model) = completion.into_parts();
709        let Admitted {
710            value: plan,
711            reservation: mut pass,
712        } = self.admit_pass(
713            ModeSet::DATA,
714            true,
715            self.write_bytes(last),
716            self.writes(last),
717        )?;
718        let model = self.lifecycle.prepare_final_model(model, terms)?;
719        let cycle = MajorCycle::refresh(
720            self.problem,
721            normal_state,
722            model,
723            self.normal_storage(plan.residency)?,
724        )?;
725        let major = self.reconcile(cycle, false, last, plan)?;
726        pass.join(results);
727        self.retain(&major, pass)?;
728        Ok(major)
729    }

745        Ok(admit(&self.host, &self.policy, &Demand { phase, memory })?)

748    /// Hand the pass's charge on to what `major` holds resident: join it to
749    /// the retained charge and keep only the completion's resident bytes.
750    fn retain(&mut self, major: &Major, pass: Reservation) -> Result<(), ImagingError> {
751        let resident = major.completion.resident_bytes()?;
752        self.retained.join(pass);
753        debug_assert!(
754            resident <= self.retained.memory(),
755            "the pass admitted what it hands on: {resident} bytes resident, {} charged",
756            self.retained.memory()
757        );
758        if resident > self.retained.memory() {
759            // The pass's plan undercounted what it hands on: admitted after
760            // the fact, so said.
761            tracing::warn!(
762                resident,
763                charged = self.retained.memory(),
764                "a pass handed on more than it admitted"
765            );
766            self.retained.join(admit(
767                &self.host,
768                &self.policy,
769                &Demand {
770                    phase: "retained normal state",
771                    memory: resident - self.retained.memory(),
772                },
773            )?);
774        }
775        self.retained.retain(resident);
776        Ok(())
777    }

779    /// Run `cycle`'s pass and finish it. The initial pass grids the data;
780    /// every refresh grids the residual of the cycle's model.
781    fn reconcile(
782        &mut self,
783        mut cycle: MajorCycle,
784        initial: bool,
785        last: bool,
786        plan: PassPlan,
787    ) -> Result<Major, ImagingError> {
788        let residency = plan.residency;
789        let modes = if initial {
790            self.initial_modes()
791        } else {
792            ModeSet::DATA
793        };
794        let transform = self.problem.visibility_transform();
795        let mut writer = self
796            .visibility_write
797            .as_ref()
798            .filter(|_| last)
799            .map(|target| VisibilityWriter::begin(target, transform))
800            .transpose()
801            .map_err(|error| ImagingError::Pass(PassError::VisibilityWrite(error)))?;
802        if let Residency::Waves { planes_per_wave } = residency {
803            self.planes_per_wave = Some(
804                self.planes_per_wave
805                    .map_or(planes_per_wave, |planes| planes.min(planes_per_wave)),
806            );
807        }
808        let started = Instant::now();
809        let (model, state) = cycle.parts();
810        let summary = self.pass(
811            modes,
812            (!initial).then_some(model),
813            plan,
814            state,
815            initial,
816            writer.as_mut(),
817        )?;
818        let pass_seconds = started.elapsed().as_secs_f64();
819        let visibility = writer
820            .map(VisibilityWriter::complete)
821            .transpose()
822            .map_err(|error| ImagingError::Pass(PassError::VisibilityWrite(error)))?
823            .map(VisibilityProductCompletion::new);
824        let completion = cycle.finish(summary.samples, summary.blocks)?;
825        tracing::info!(
826            "imaging major cycle: {} samples, pass {pass_seconds:.2} s, total {:.2} s",
827            summary.samples,
828            started.elapsed().as_secs_f64(),
829        );
830        self.keep_row_work(MeasuredRowWork {
831            rows: summary.row_work,
832            modes: Vec::new(),
833        });
834        Ok(Major {
835            completion,
836            visibility,
837        })
838    }

858    /// The waves of a pass accumulating `modes`, with a model when
859    /// `with_model`, that fit the free memory beside `besides` bytes the
860    /// pass also allocates (an initial model) and the faceted domains'
861    /// stitched images ([`ChartStitch::pass_bytes`]), and the reservation of
862    /// all three: planned once per major cycle for the pass and its normal
863    /// storage. A pass that `writes` visibilities keeps one set of chunks,
864    /// since it does not pipeline its owners.
865    fn admit_pass(
866        &self,
867        modes: ModeSet,
868        with_model: bool,
869        besides: u64,
870        writes: bool,
871    ) -> Result<Admitted<PassPlan>, ImagingError> {
872        let besides = besides + self.stitch.pass_bytes(modes.psf);
873        let workers = self.teams.streaming().workers();
874        let domains = pass_domains(&self.domains, workers, self.row_work(modes));
875        let mut demand = WaveDemand {
876            domains: &domains,
877            modes,
878            with_model,
879            native_spacing_hz: self.native_spacing_hz,
880            workers,
881            backend: self.backend,
882            block: self.source.maximum_block(),
883            chunk_placements: CHUNK_PLACEMENTS,
884            chunk_sets: 1,
885        };
886        let free = free_memory(&self.host, &self.policy).saturating_sub(besides);
887        let residency = Residency::plan(&demand, free);
888        let planes = match residency {
889            Residency::All => demand.planes(),
890            Residency::Waves { planes_per_wave } => planes_per_wave,
891        };
892        // Region owners pipeline over a second set of chunks, and share
893        // larger slices more evenly; take what fits beside the waves just
894        // planned (#736, #449).
895        (demand.chunk_placements, demand.chunk_sets) = demand.plan_chunks(planes, free, writes);
896        let reservation = admit(
897            &self.host,
898            &self.policy,
899            &Demand {
900                phase: "major-cycle pass",
901                memory: demand.bytes(planes) + besides,
902            },
903        )?;
904        Ok(Admitted {
905            value: PassPlan {
906                residency,
907                chunk_placements: demand.chunk_placements,
908                chunk_sets: demand.chunk_sets,
909            },
910            reservation,
911        })
912    }

914    /// Keep the imaging weights of every sample across passes (#659) when a
915    /// single-wave pass places one image domain's rows a channel at a time
916    /// and the slots, four bytes per selected (row, channel), take at most a
917    /// sixteenth of the free memory; charged before they are allocated.
918    /// Wider runs weigh every pass.
919    fn keep_weights(&self, residency: Residency) -> Option<(WeightCache, Reservation)> {
920        if residency != Residency::All {
921            return None;
922        }
923        let (samples, bytes) = kept_weight_bytes(
924            self.problem,
925            &self.domains,
926            self.source.maximum_block().channels(),
927            free_memory(&self.host, &self.policy),
928        )?;
929        let charge = self.admit("kept imaging weights", bytes).ok()?;
930        Some((WeightCache::new(samples), charge))
931    }

933    /// Measure each region-split domain's gridding work by anchor row before
934    /// the first pass, over a traversal of flags and weights only
935    /// ([`run_row_work_pass`]), so that the first pass cuts balanced strips
936    /// too (#736). A run without several region owners measures nothing,
937    /// and the pass is charged while it runs.
938    fn measure_row_work(&mut self) -> Result<(), ImagingError> {
939        if self.row_work.is_some() {
940            return Ok(());
941        }
942        let team = self.teams.streaming();
943        let work = {
944            let domains = pass_domains(&self.domains, team.workers(), None);
945            if !needs_row_work(&domains) {
946                return Ok(());
947            }
948            let _charge = self.admit(
949                "region row-work pass",
950                row_work_pass_bytes(&domains, team.workers(), self.weight_image),
951            )?;
952            let started = Instant::now();
953            let work = run_row_work_pass(
954                &domains,
955                &self.weighting,
956                &mut self.source,
957                team,
958                &self.cancel,
959            )?;
960            tracing::info!(
961                "imaging region row work: measured in {:.2} s",
962                started.elapsed().as_secs_f64()
963            );
964            work
965        };
966        self.keep_row_work(work);
967        Ok(())
968    }

970    /// The kept row work ([`Self::keep_row_work`]), for a pass of `modes`.
971    fn row_work(&self, modes: ModeSet) -> Option<(&MeasuredRowWork, ModeSet)> {
972        self.row_work.as_ref().map(|(work, _)| (work, modes))
973    }
974
975    /// Keep the first pass's gridding work by anchor row, and the work in
976    /// each mode a row-work pass measured, for the passes' strips
977    /// ([`Partition::regions_balanced`]), charged before it is kept. Every
978    /// pass places the same samples, so one histogram serves the run.
979    /// Without a region-split domain, or without room, the later passes
980    /// keep strips of equal height.
981    fn keep_row_work(&mut self, row_work: MeasuredRowWork) {
982        if self.row_work.is_some() || row_work.rows.iter().all(Option::is_none) {
983            return;
984        }
985        let bytes = row_work.bytes();
986        match self.admit("region row work", bytes) {
987            Ok(charge) => self.row_work = Some((row_work, charge)),
988            Err(_) => tracing::info!(
989                "imaging region row work: no room to keep {bytes} bytes; \
990                 later passes keep strips of equal height"
991            ),
992        }
993    }

995    /// One pass over every image domain, appending its images to `state`.
996    fn pass(
997        &mut self,
998        modes: ModeSet,
999        model: Option<&ModelGeneration>,
1000        plan: PassPlan,
1001        mut state: PassAppender<'_>,
1002        initial: bool,
1003        mut writer: Option<&mut VisibilityWriter<'_>>,
1004    ) -> Result<PassSummary, ImagingError> {
1005        let PassPlan {
1006            residency,
1007            chunk_placements,
1008            chunk_sets,
1009        } = plan;
1010        if self.kept_weights.is_none() {
1011            self.kept_weights = self.keep_weights(residency);
1012        }
1013        let team = self.teams.streaming();
1014        let domains = pass_domains(&self.domains, team.workers(), self.row_work(modes));
1015        let charts = &self.charts;
1016        let prepare = |chart: usize, planes: PlaneRange| {
1017            let generation = model.expect("the closure is installed only with a model");
1018            let Chart { domain, window } = charts[chart];
1019            prepare_model(
1020                &self.domains[chart].operator,
1021                generation,
1022                (domain, window),
1023                planes,
1024            )
1025            .map_err(|error| PassError::Model(Box::new(error)))
1026        };
1027        let pass = MajorCyclePass {
1028            domains: &domains,
1029            weighting: &self.weighting,
1030            modes,
1031            model: model.map(|_| &prepare as &ModelPreparation<'_>),
1032            residency,
1033            native_spacing_hz: self.native_spacing_hz,
1034            backend: self.backend,
1035            metal_stage_timing: self.metal_stage_timing,
1036            weights: self.kept_weights.as_ref().map(|(cache, _)| cache),
1037            chunk_placements,
1038            chunk_sets,
1039        };
1040        let predictions = writer.as_deref().map(VisibilityWriter::needs_predictions);
1041        let mut write = |block: &_, predictions: &[_]| match writer.as_deref_mut() {
1042            Some(writer) => writer.write(block, predictions),
1043            None => Ok(()),
1044        };
1045        let mut sink = predictions.map(|predictions| VisibilitySink {
1046            predictions,
1047            write: &mut write,
1048        });
1049        Ok(run_major_cycle(
1050            &pass,
1051            &mut self.source,
1052            team,
1053            &self.cancel,
1054            &mut |chart, images| {
1055                let basis = self.domains[chart].operator.basis();
1056                let images = self
1057                    .stitch
1058                    .accept(chart, pass_images(chart, &images, initial, basis))
1059                    .map_err(|error| PassError::Images(error.into()))?;
1060                match images {
1061                    Some(images) => state
1062                        .append(images)
1063                        .map_err(|error| PassError::Images(Box::new(error))),
1064                    None => Ok(()),
1065                }
1066            },
1067            sink.as_mut(),
1068        )?)
1069    }
1070}

1072/// The samples and bytes of the imaging weights a run keeps across its
1073/// passes ([`Run::keep_weights`]): one image domain placed a channel at a
1074/// time, `channels` per row, whose slots take at most a sixteenth of
1075/// `free`; `None` when the run weighs every pass instead.
1076fn kept_weight_bytes(
1077    problem: &CompiledProblem,
1078    domains: &[DomainOperator],
1079    channels: usize,
1080    free: u64,
1081) -> Option<(usize, u64)> {
1082    if domains.len() != 1 || !domains[0].resampler.is_direct() {
1083        return None;
1084    }
1085    let rows = problem
1086        .observation()
1087        .sources()
1088        .iter()
1089        .map(|source| source.selection().rows().selected_row_count())
1090        .sum::<u64>();
1091    let samples = usize::try_from(rows).ok()?.checked_mul(channels)?;
1092    let bytes = WeightCache::bytes(samples);
1093    (bytes <= free / 16).then_some((samples, bytes))
1094}

1096/// Every image domain's operator, resampler and partition: planes split
1097/// among `workers` for a channel-local basis, grid strips otherwise, cut to
1098/// equal shares of an earlier pass's gridding work by anchor row when the
1099/// run has it (`row_work`, per domain; #736), for a pass of the modes given
1100/// with it: one that grids the weight image adds that work to the centre
1101/// row's owner ([`casa_imaging_runtime::pass::ModeWork::weighed`]).
1102///
1103/// That weighed histogram is a transient of 8 bytes per grid row for each
1104/// region domain, dropped once the strips are cut. It goes uncharged, like
1105/// the prefix sums [`Partition::regions_balanced`] builds while cutting:
1106/// 32 KB at 4096².
1107pub(super) fn pass_domains<'d>(
1108    domains: &'d [DomainOperator],
1109    workers: usize,
1110    row_work: Option<(&MeasuredRowWork, ModeSet)>,
1111) -> Vec<PassDomain<'d>> {
1112    domains
1113        .iter()
1114        .enumerate()
1115        .map(|(index, domain)| PassDomain {
1116            operator: &domain.operator,
1117            resampler: &domain.resampler,
1118            partition: match domain.operator.basis() {
1119                Basis::ChannelLocal { .. } => Partition::Planes { owners: workers },
1120                Basis::Constant | Basis::Taylor { .. } => {
1121                    match row_work.and_then(|(work, modes)| {
1122                        let rows = work.rows.get(index)?.as_ref()?;
1123                        Some((
1124                            rows,
1125                            work.modes.get(index).and_then(|w| w.weighed(rows, modes)),
1126                        ))
1127                    }) {
1128                        Some((rows, weighed)) => Partition::regions_balanced(
1129                            &domain.operator,
1130                            workers,
1131                            weighed.as_deref().unwrap_or(rows),
1132                        ),
1133                        None => Partition::regions(&domain.operator, workers),
1134                    }
1135                }
1136            },
1137        })
1138        .collect()
1139}

1178/// The output-frame frequency envelope of a plane range of a channel-local
1179/// basis: from the low edge of its first channel to the high edge of its
1180/// last.
1181fn plane_bounds(problem: &CompiledProblem) -> Option<PlaneBounds> {
1182    if !matches!(
1183        problem.reconstruction().basis(),
1184        casa_imaging_model::ReconstructionBasis::ChannelLocal { .. }
1185    ) {
1186        return None;
1187    }
1188    let SpectralWcs::Linear {
1189        reference_pixel,
1190        reference_frequency_hz,
1191        increment_hz,
1192        ..
1193    } = *problem.geometry().spectral().wcs()
1194    else {
1195        return None;
1196    };
1197    let centre = move |channel: u32| {
1198        reference_frequency_hz + (f64::from(channel) - reference_pixel) * increment_hz
1199    };
1200    Some(Box::new(move |planes: PlaneRange| {
1201        let half = increment_hz.abs() / 2.0;
1202        let [first, last] = [centre(planes.start), centre(planes.end - 1)];
1203        [first.min(last) - half, first.max(last) + half]
1204    }))
1205}
