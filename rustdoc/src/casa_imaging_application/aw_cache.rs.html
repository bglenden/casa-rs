180        decoded_count(&self.imaging)
181            .and_then(|left| decoded_count(&self.weight).and_then(|right| left.checked_add(right)))
182            .and_then(|count| count.checked_mul(std::mem::size_of::<Complex32>()))

185    /// Conservative encoded workspace required while producing the decoded cell.
186    #[must_use]
187    pub fn decoder_workspace_bytes(&self) -> Option<usize> {
188        [self.imaging.shape, self.weight.shape]
189            .into_iter()
190            .try_fold(0_usize, |bytes, shape| {
191                shape[0]
192                    .checked_mul(shape[1])
193                    .and_then(|elements| elements.checked_mul(std::mem::size_of::<Complex32>()))
194                    .and_then(|plane| bytes.checked_add(plane))
195            })?
196            .checked_mul(2)
197    }

214    /// Bind this validated CASA pair as a plan-listed structured load source.
215    pub fn import_source(
216        &self,
217        cache: &CasaAwCache,
218        storage_domain: &StorageDomain,
219        producer: WorkNodeId,
220    ) -> Result<PreparedArtifactImportSource, PreparedArtifactError> {
221        let entry = cache
222            .entries
223            .get(&self.stable_key)
224            .filter(|entry| entry.identity == self.metadata.identity())
225            .ok_or(PreparedArtifactError::SourceIdentityMismatch)?;
226        let source_identity = ArtifactIdentity::from_sha256(entry.identity.as_bytes());
227        let segment = |name: &str, metadata: &KernelMetadata| {
228            let width = u64::try_from(metadata.shape[0])
229                .map_err(|_| PreparedArtifactError::ArtifactTooLarge)?;
230            let height = u64::try_from(metadata.shape[1])
231                .map_err(|_| PreparedArtifactError::ArtifactTooLarge)?;
232            let elements = width
233                .checked_mul(height)
234                .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
235            let bytes = elements
236                .checked_mul(8)
237                .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
238            let operations = elements
239                .checked_add(1)
240                .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
241            PreparedArtifactImportSegment::new(
242                name,
243                metadata.path.clone(),
244                source_identity,
245                bytes,
246                operations,
247                storage_domain,
248            )
249        };
250        PreparedArtifactImportSource::new(
251            &self.descriptor,
252            producer,
253            vec![
254                segment("imaging", &entry.imaging)?,
255                segment("weight", &entry.weight)?,
256            ],
257        )
258    }

324impl PreparedPoolState {
325    fn ensure_available(&self) -> Result<(), AwOperatorError> {
326        if self.closed || self.aborted {
327            Err(AwOperatorError::PreparedCellUnavailable)
328        } else {
329            Ok(())
330        }
331    }
332
333    fn finish_decoded(
334        &mut self,
335        decoded: Result<Arc<AwConvolutionCell>, PreparedArtifactError>,
336    ) -> Result<Arc<AwConvolutionCell>, AwOperatorError> {
337        match decoded {
338            Ok(cell) => {
339                self.ensure_available()?;
340                Ok(cell)
341            }
342            Err(_) => {
343                self.invalidate();
344                Err(AwOperatorError::PreparedCellUnavailable)
345            }
346        }
347    }
348
349    fn invalidate(&mut self) {
350        self.closed = true;
351        self.aborted = true;
352        self.cells.clear();
353        self.resident = 0;
354    }

358    /// Validate and index one standard CASA cache directory without reading pixels.
359    pub fn open(root: impl AsRef<Path>) -> Result<Self, CasaAwCacheError> {
360        let root = root.as_ref();
361        let mut imaging = BTreeMap::new();
362        let mut weight = BTreeMap::new();
363        for directory in
364            fs::read_dir(root).map_err(|error| fail(root, format!("cannot list cache: {error}")))?
365        {
366            let directory = directory
367                .map_err(|error| fail(root, format!("cannot inspect cache entry: {error}")))?;
368            let path = directory.path();
369            if !path.is_dir() {
370                continue;
371            }
372            let Some(name) = path.file_name().and_then(|name| name.to_str()) else {
373                continue;
374            };
375            let destination = if name.starts_with(WEIGHT_PREFIX) && name.ends_with(".im") {
376                &mut weight
377            } else if name.starts_with(IMAGING_PREFIX) && name.ends_with(".im") {
378                &mut imaging
379            } else {
380                continue;
381            };
382            let (key, metadata) = read_metadata(&path)?;
383            if destination.insert(StableKey::from(key), metadata).is_some() {
384                return Err(fail(
385                    &path,
386                    format!("duplicate scientific cell key {key:?}"),
387                ));
388            }
389        }
390        if imaging.is_empty() && weight.is_empty() {
391            return Err(fail(root, "no CFS_/WTCFS_ image-table directories found"));
392        }
393        let mut entries = BTreeMap::new();
394        for (stable, imaging_metadata) in imaging {
395            let weight_metadata = weight
396                .remove(&stable)
397                .ok_or_else(|| fail(&imaging_metadata.path, "paired WTCFS_ cell is missing"))?;
398            let key = key_from(stable);
399            validate_pair(key, &imaging_metadata, &weight_metadata)?;
400            let identity = mint_identity(key, &imaging_metadata)?;
401            entries.insert(
402                stable,
403                Entry {
404                    key,
405                    identity,
406                    imaging: imaging_metadata,
407                    weight: weight_metadata,
408                },
409            );
410        }
411        if let Some(metadata) = weight.into_values().next() {
412            return Err(fail(&metadata.path, "paired CFS_ cell is missing"));
413        }
414        let inventory = validate_axes(root, &entries)?;
415        let identities = entries
416            .iter()
417            .map(|(key, entry)| (entry.identity.as_bytes(), *key))
418            .collect();
419        Ok(Self {
420            root: root.to_path_buf(),
421            entries,
422            identities,
423            inventory,
424        })
425    }

451    /// and its bounded T50 manifest.
452    pub fn prepared_cache_bytes(&self) -> Result<u64, CasaAwCacheError> {
453        self.entries.values().try_fold(0_u64, |total, entry| {
454            let payload = [entry.imaging.shape, entry.weight.shape]
455                .into_iter()
456                .try_fold(0_usize, |bytes, shape| {
457                    shape[0]
458                        .checked_mul(shape[1])
459                        .and_then(|elements| elements.checked_mul(std::mem::size_of::<Complex32>()))
460                        .and_then(|plane| bytes.checked_add(plane))
461                })
462                .and_then(|bytes| u64::try_from(bytes).ok())
463                .ok_or_else(|| fail(&entry.imaging.path, "prepared payload size overflowed"))?;
464            total
465                .checked_add(payload)
466                .and_then(|bytes| bytes.checked_add(16 * 1024))
467                .ok_or_else(|| fail(&entry.imaging.path, "prepared cache budget overflowed"))
468        })

472    pub fn prepared_cell<R: ImplementationRegistry>(
473        &self,
474        metadata: &AwPreparedCellMetadata,
475        store: &PreparedArtifactStore,
476        registry: &R,
477        implementation: &WorkImplementationId,
478        problem: &casa_imaging_model::CompiledProblem,
479    ) -> Result<CasaAwPreparedCell, CasaAwCacheError> {
480        let stable_key = *self
481            .identities
482            .get(&metadata.identity().as_bytes())
483            .ok_or_else(|| {
484                fail(
485                    &self.root,
486                    "prepared metadata is not owned by this CASA cache",
487                )
488            })?;
489        let entry = self
490            .entries
491            .get(&stable_key)
492            .ok_or_else(|| fail(&self.root, "prepared CASA cell disappeared from the index"))?;
493        let descriptor = PreparedArtifactDescriptor::convolution_function(
494            store,
495            registry,
496            implementation,
497            problem,
498            entry.identity,
499            plane_descriptor(&entry.imaging)?,
500            plane_descriptor(&entry.weight)?,
501        )
502        .map_err(|error| fail(&self.root, format!("cannot compile private cell: {error}")))?;
503        Ok(CasaAwPreparedCell {
504            metadata: metadata.clone(),
505            descriptor,
506            stable_key,
507            imaging: entry.imaging.clone(),
508            weight: entry.weight.clone(),
509        })

548    fn fill_segment(
549        &mut self,
550        segment: &PreparedArtifactSegmentDescriptor,
551        byte_offset: u64,
552        output: &mut [u8],
553    ) -> Result<u64, PreparedArtifactError> {
554        let (name, metadata) = match segment.name() {
555            "imaging" => ("imaging", &self.entry.imaging),
556            "weight" => ("weight", &self.entry.weight),
557            _ => return Err(PreparedArtifactError::SegmentMismatch),
558        };
559        let opened = self.loaded.as_ref().map(|loaded| loaded.name) != Some(name);
560        if opened {
561            self.loaded = Some(LoadedCasaPlane {
562                name,
563                image: PagedImage::<Complex32>::open(&metadata.path)
564                    .map_err(|_| PreparedArtifactError::SourceIdentityMismatch)?,
565                last: None,
566            });
567        }
568        let reads = encode_complex32_range(
569            self.loaded.as_mut().expect("loaded above"),
570            metadata.shape,
571            byte_offset,
572            output,
573        )?;
574        reads
575            .checked_add(u64::from(opened))
576            .ok_or(PreparedArtifactError::ArtifactTooLarge)
577    }

580impl PreparedAwCellProvider {
581    /// Bind a fresh decoded-cell LRU to one inactive prepared reader session.
582    pub fn new(
583        resident_byte_ceiling: usize,
584        reader: Arc<PreparedArtifactReader>,
585        prepared: Vec<CasaAwPreparedCell>,
586    ) -> Result<Self, AwOperatorError> {
587        if resident_byte_ceiling == 0 || prepared.is_empty() {
588            return Err(AwOperatorError::ResidencyCeilingExceeded);
589        }
590        let workspace_ceiling = usize::try_from(reader.plan().decoder_workspace_bytes())
591            .map_err(|_| AwOperatorError::MeasurementOverflow)?;
592        if u64::try_from(resident_byte_ceiling).ok() != Some(reader.plan().decoded_resident_bytes())
593            || prepared
594                .iter()
595                .map(CasaAwPreparedCell::decoder_workspace_bytes)
596                .try_fold(0_usize, |maximum, bytes| {
597                    bytes.map(|bytes| maximum.max(bytes))
598                })
599                .filter(|required| *required > 0 && *required <= workspace_ceiling)
600                .is_none()
601        {
602            return Err(AwOperatorError::ResidencyCeilingExceeded);
603        }
604        let prepared = prepared
605            .into_iter()
606            .map(|cell| (cell.metadata.identity().as_bytes(), cell))
607            .collect::<BTreeMap<_, _>>();
608        let reload_probe = reader
609            .take_reload_cost_probe()
610            .map_err(|_| AwOperatorError::PreparedCellUnavailable)?;
611        Ok(Self {
612            pool: Arc::new(PreparedPool {
613                reader,
614                prepared,
615                state: Mutex::new(PreparedPoolState {
616                    ceiling: resident_byte_ceiling,
617                    workspace_ceiling,
618                    reload_probe,
619                    ..PreparedPoolState::default()
620                }),
621                available: Condvar::new(),
622            }),
623        })
624    }
625
626    /// Exact decoded bytes currently retained across all prepared cells.
627    pub fn resident_bytes(&self) -> Result<usize, AwOperatorError> {
628        self.pool
629            .state
630            .lock()
631            .map(|state| state.resident)
632            .map_err(|_| AwOperatorError::PreparedCellUnavailable)
633    }
634
635    /// Number of exact prepared pairs currently available to operators.
636    pub fn resident_cells(&self) -> Result<usize, AwOperatorError> {
637        self.pool
638            .state
639            .lock()
640            .map(|state| state.cells.len())
641            .map_err(|_| AwOperatorError::PreparedCellUnavailable)
642    }
643}

645impl AwPreparedCellProvider for PreparedAwCellProvider {
646    fn load(
647        &mut self,
648        metadata: &AwPreparedCellMetadata,
649        ceiling: usize,
650    ) -> Result<AwPreparedCellLease, AwOperatorError> {
651        let identity = metadata.identity().as_bytes();
652        let prepared = self
653            .pool
654            .prepared
655            .get(&identity)
656            .ok_or(AwOperatorError::PreparedCellUnavailable)?;
657        let bytes = prepared
658            .decoded_resident_bytes()
659            .ok_or(AwOperatorError::MeasurementOverflow)?;
660        let workspace = prepared
661            .decoder_workspace_bytes()
662            .ok_or(AwOperatorError::MeasurementOverflow)?;
663        if bytes > ceiling
664            || bytes
665                > self
666                    .pool
667                    .state
668                    .lock()
669                    .map_err(|_| AwOperatorError::PreparedCellUnavailable)?
670                    .ceiling
671        {
672            return Err(AwOperatorError::ResidencyCeilingExceeded);
673        }
674        let mut evicted = 0_usize;
675        let mut reload_sample = None;
676        'load: loop {
677            let mut state = self
678                .pool
679                .state
680                .lock()
681                .map_err(|_| AwOperatorError::PreparedCellUnavailable)?;
682            state.ensure_available()?;
683            if state.cells.contains_key(&identity) {
684                state.clock = state
685                    .clock
686                    .checked_add(1)
687                    .ok_or(AwOperatorError::MeasurementOverflow)?;
688                let clock = state.clock;
689                let cell = {
690                    let resident = state
691                        .cells
692                        .get_mut(&identity)
693                        .expect("resident identity was checked");
694                    resident.last_use = clock;
695                    Arc::clone(&resident.cell)
696                };
697                state.hits = state
698                    .hits
699                    .checked_add(1)
700                    .ok_or(AwOperatorError::MeasurementOverflow)?;
701                observe_pinned(&mut state)?;
702                drop(state);
703                return Ok(self.lease(cell, AwPreparedCellDisposition::Resident, evicted, 0));
704            }
705            if reload_sample.is_none() {
706                if let Some(probe) = state.reload_probe.as_mut() {
707                    reload_sample = Some(
708                        probe
709                            .begin(prepared.descriptor().identity().as_bytes(), bytes)
710                            .map_err(|_| AwOperatorError::MeasurementOverflow)?,
711                    );
712                }
713            }
714            if state.loading.contains(&identity) {
715                observe_reload(&mut reload_sample, ReloadStage::LoadingWait, || {
716                    drop(
717                        self.pool
718                            .available
719                            .wait(state)
720                            .map_err(|_| AwOperatorError::PreparedCellUnavailable)?,
721                    );
722                    Ok::<_, AwOperatorError>(())
723                })?;
724                continue;
725            }
726            if state
727                .reserved_workspace
728                .checked_add(workspace)
729                .ok_or(AwOperatorError::MeasurementOverflow)?
730                > state.workspace_ceiling
731            {
732                observe_reload(&mut reload_sample, ReloadStage::WorkspaceWait, || {
733                    drop(
734                        self.pool
735                            .available
736                            .wait(state)
737                            .map_err(|_| AwOperatorError::PreparedCellUnavailable)?,
738                    );
739                    Ok::<_, AwOperatorError>(())
740                })?;
741                continue;
742            }
743            while state
744                .resident
745                .checked_add(state.reserved)
746                .and_then(|retained| retained.checked_add(bytes))
747                .ok_or(AwOperatorError::MeasurementOverflow)?
748                > state.ceiling
749            {
750                let eviction_started = reload_sample
751                    .as_ref()
752                    .and_then(|sample: &ReloadSample| sample.cost.start());
753                let victim = state
754                    .cells
755                    .iter()
756                    .filter(|(_, resident)| Arc::strong_count(&resident.cell) == 1)
757                    .min_by_key(|(victim, resident)| (resident.last_use, **victim))
758                    .map(|(victim, _)| *victim);
759                let Some(victim) = victim else {
760                    finish_reload(&mut reload_sample, ReloadStage::Eviction, eviction_started);
761                    observe_reload(&mut reload_sample, ReloadStage::PinWait, || {
762                        drop(
763                            self.pool
764                                .available
765                                .wait(state)
766                                .map_err(|_| AwOperatorError::PreparedCellUnavailable)?,
767                        );
768                        Ok::<_, AwOperatorError>(())
769                    })?;
770                    continue 'load;
771                };
772                if let Some(probe) = state.reload_probe.as_mut() {
773                    let descriptor = self
774                        .pool
775                        .prepared
776                        .get(&victim)
777                        .ok_or(AwOperatorError::PreparedCellUnavailable)?
778                        .descriptor();
779                    probe
780                        .evicted(descriptor.identity().as_bytes())
781                        .map_err(|_| AwOperatorError::MeasurementOverflow)?;
782                }
783                let victim = state
784                    .cells
785                    .remove(&victim)
786                    .expect("selected resident victim exists");
787                state.resident = state
788                    .resident
789                    .checked_sub(victim.bytes)
790                    .ok_or(AwOperatorError::MeasurementOverflow)?;
791                evicted = evicted
792                    .checked_add(victim.bytes)
793                    .ok_or(AwOperatorError::MeasurementOverflow)?;
794                state.evicted_bytes = state
795                    .evicted_bytes
796                    .checked_add(
797                        u64::try_from(victim.bytes)
798                            .map_err(|_| AwOperatorError::MeasurementOverflow)?,
799                    )
800                    .ok_or(AwOperatorError::MeasurementOverflow)?;
801                drop(victim);
802                finish_reload(&mut reload_sample, ReloadStage::Eviction, eviction_started);
803            }
804            if state.cells.contains_key(&identity) || state.loading.contains(&identity) {
805                continue;
806            }
807            state.reserved = state
808                .reserved
809                .checked_add(bytes)
810                .ok_or(AwOperatorError::MeasurementOverflow)?;
811            state.reserved_workspace = state
812                .reserved_workspace
813                .checked_add(workspace)
814                .ok_or(AwOperatorError::MeasurementOverflow)?;
815            state.peak_workspace = state.peak_workspace.max(state.reserved_workspace);
816            state.loading.insert(identity);
817            state.loads = state
818                .loads
819                .checked_add(1)
820                .ok_or(AwOperatorError::MeasurementOverflow)?;
821            if let (Some(probe), Some(sample)) =
822                (state.reload_probe.as_mut(), reload_sample.as_mut())
823            {
824                probe
825                    .admit(sample)
826                    .map_err(|_| AwOperatorError::MeasurementOverflow)?;
827            }
828            drop(state);
829
830            let selected = reload_sample
831                .as_ref()
832                .is_some_and(|sample| sample.cost.enabled());
833            let mut load_cost = ReloadCost::new(selected);
834            let decoded = (|| {
835                let mut decoder = load_cost.measure(ReloadStage::DecoderAllocate, || {
836                    PreparedCellDecoder::new(prepared, selected)
837                })?;
838                let read = load_cost.measure(ReloadStage::Reader, || {
839                    self.pool
840                        .reader
841                        .read(prepared.descriptor().identity(), &mut decoder)
842                });
843                load_cost.merge(&decoder.reload_cost);
844                read?;
845                let cell = decoder
846                    .finish(prepared, &mut load_cost)
847                    .map_err(|_| PreparedArtifactError::InvalidLayout)?;
848                if cell.resident_bytes() != bytes {
849                    return Err(PreparedArtifactError::InvalidLayout);
850                }
851                Ok::<_, PreparedArtifactError>(Arc::new(cell))
852            })();
853
854            let completion_started = load_cost.start();
855            let mut state = self
856                .pool
857                .state
858                .lock()
859                .map_err(|_| AwOperatorError::PreparedCellUnavailable)?;
860            if !state.loading.remove(&identity) {
861                return Err(AwOperatorError::PreparedCellUnavailable);
862            }
863            state.reserved = state
864                .reserved
865                .checked_sub(bytes)
866                .ok_or(AwOperatorError::MeasurementOverflow)?;
867            state.reserved_workspace = state
868                .reserved_workspace
869                .checked_sub(workspace)
870                .ok_or(AwOperatorError::MeasurementOverflow)?;
871            let cell = match state.finish_decoded(decoded) {
872                Ok(cell) => cell,
873                Err(error) => {
874                    self.pool.available.notify_all();
875                    drop(state);
876                    load_cost.finish(ReloadStage::PoolComplete, completion_started);
877                    if let Some(mut sample) = reload_sample {
878                        sample.cost.merge(&load_cost);
879                        sample.emit(false);
880                    }
881                    return Err(error);
882                }
883            };
884            state.resident = state
885                .resident
886                .checked_add(bytes)
887                .ok_or(AwOperatorError::MeasurementOverflow)?;
888            if state.resident > state.ceiling {
889                return Err(AwOperatorError::ResidencyCeilingExceeded);
890            }
891            state.clock = state
892                .clock
893                .checked_add(1)
894                .ok_or(AwOperatorError::MeasurementOverflow)?;
895            let clock = state.clock;
896            state.copied_bytes = state
897                .copied_bytes
898                .checked_add(
899                    u64::try_from(bytes).map_err(|_| AwOperatorError::MeasurementOverflow)?,
900                )
901                .ok_or(AwOperatorError::MeasurementOverflow)?;
902            state.peak_resident = state.peak_resident.max(state.resident);
903            let lease_cell = Arc::clone(&cell);
904            if state
905                .cells
906                .insert(
907                    identity,
908                    ResidentPreparedCell {
909                        cell,
910                        bytes,
911                        last_use: clock,
912                    },
913                )
914                .is_some()
915            {
916                return Err(AwOperatorError::PreparedCellUnavailable);
917            }
918            observe_pinned(&mut state)?;
919            self.pool.available.notify_all();
920            drop(state);
921            load_cost.finish(ReloadStage::PoolComplete, completion_started);
922            let lease = load_cost.measure(ReloadStage::Lease, || {
923                self.lease(
924                    lease_cell,
925                    AwPreparedCellDisposition::Loaded,
926                    evicted,
927                    bytes,
928                )
929            });
930            if let Some(mut sample) = reload_sample {
931                sample.cost.merge(&load_cost);
932                sample.emit(true);
933            }
934            return Ok(lease);
935        }
936    }
937}

997impl PreparedArtifactReaderResidency for PreparedAwCellProvider {
998    fn close(&self) -> Result<PreparedArtifactResidencyMeasurements, PreparedArtifactError> {
999        let mut state = self
1000            .pool
1001            .state
1002            .lock()
1003            .map_err(|_| PreparedArtifactError::PoisonedStore)?;
1004        state.closed = true;
1005        if state.reserved != 0
1006            || state.reserved_workspace != 0
1007            || !state.loading.is_empty()
1008            || pinned_bytes(&state)? != 0
1009        {
1010            return Err(PreparedArtifactError::ReaderStillInUse);
1011        }
1012        if let Some(probe) = state.reload_probe.take() {
1013            probe.emit(state.aborted);
1014        }
1015        Self::measurements(&state)
1016    }
1017
1018    fn release(&self) -> Result<PreparedArtifactResidencyMeasurements, PreparedArtifactError> {
1019        let mut state = self
1020            .pool
1021            .state
1022            .lock()
1023            .map_err(|_| PreparedArtifactError::PoisonedStore)?;
1024        state.closed = true;
1025        if state.reserved != 0 || state.reserved_workspace != 0 || !state.loading.is_empty() {
1026            return Err(PreparedArtifactError::ReaderStillInUse);
1027        }
1028        if let Some(probe) = state.reload_probe.take() {
1029            probe.emit(state.aborted);
1030        }
1031        let measurements = Self::measurements(&state)?;
1032        state.cells.clear();
1033        state.resident = 0;
1034        self.pool.available.notify_all();
1035        Ok(measurements)
1036    }
1037
1038    fn abort(&self) {
1039        if let Ok(mut state) = self.pool.state.lock() {
1040            state.invalidate();
1041            self.pool.available.notify_all();
1042        }
1043    }
1044}

1048        .cells
1049        .values()
1050        .filter(|resident| Arc::strong_count(&resident.cell) > 1)
1051        .try_fold(0_usize, |total, resident| {

1064struct PreparedCellDecoder {
1065    imaging: Vec<u8>,
1066    weight: Vec<u8>,
1067    imaging_expected: usize,
1068    weight_expected: usize,
1069    reload_cost: ReloadCost,
1070}

1072impl PreparedCellDecoder {
1073    fn new(
1074        prepared: &CasaAwPreparedCell,
1075        observe_reload: bool,
1076    ) -> Result<Self, PreparedArtifactError> {
1077        let expected = |segment: &PreparedArtifactSegmentDescriptor| {
1078            segment
1079                .shape()
1080                .iter()
1081                .try_fold(1_u64, |count, extent| count.checked_mul(*extent))
1082                .and_then(|count| count.checked_mul(8))
1083                .and_then(|bytes| usize::try_from(bytes).ok())
1084                .ok_or(PreparedArtifactError::ArtifactTooLarge)
1085        };
1086        let imaging = prepared
1087            .descriptor
1088            .imaging_plane()
1089            .ok_or(PreparedArtifactError::SegmentMismatch)?;
1090        let weight = prepared
1091            .descriptor
1092            .weight_plane()
1093            .ok_or(PreparedArtifactError::SegmentMismatch)?;
1094        let imaging_expected = expected(imaging)?;
1095        let weight_expected = expected(weight)?;
1096        Ok(Self {
1097            imaging: Vec::with_capacity(imaging_expected),
1098            weight: Vec::with_capacity(weight_expected),
1099            imaging_expected,
1100            weight_expected,
1101            reload_cost: ReloadCost::new(observe_reload),
1102        })
1103    }
1104
1105    fn finish(
1106        self,
1107        prepared: &CasaAwPreparedCell,
1108        cost: &mut ReloadCost,
1109    ) -> Result<AwConvolutionCell, CasaAwCacheError> {
1110        if self.imaging.len() != self.imaging_expected || self.weight.len() != self.weight_expected
1111        {
1112            return Err(fail(
1113                &prepared.imaging.path,
1114                "private payload ended before its declared shape",
1115            ));
1116        }
1117        let (imaging_plane, weight_plane) = cost.measure(ReloadStage::DecodePlanes, || {
1118            Ok::<_, CasaAwCacheError>((
1119                decode_complex32_plane(self.imaging, &prepared.imaging)?,
1120                decode_complex32_plane(self.weight, &prepared.weight)?,
1121            ))
1122        })?;
1123        cost.measure(ReloadStage::ConstructKernels, || {
1124            let imaging = adapt_kernel_from_plane(&prepared.imaging, imaging_plane)?;
1125            let weight = adapt_kernel_from_plane(&prepared.weight, weight_plane)?;
1126            AwConvolutionCell::new(prepared.metadata.identity(), imaging, weight)
1127                .map_err(|error| fail(&prepared.imaging.path, error.to_string()))
1128        })
1129    }
1130}

1183fn encode_complex32_range(
1184    loaded: &mut LoadedCasaPlane,
1185    shape: [usize; 2],
1186    byte_offset: u64,
1187    output: &mut [u8],
1188) -> Result<u64, PreparedArtifactError> {
1189    let offset =
1190        usize::try_from(byte_offset).map_err(|_| PreparedArtifactError::ArtifactTooLarge)?;
1191    let elements = shape[0]
1192        .checked_mul(shape[1])
1193        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
1194    let scalars = elements
1195        .checked_mul(2)
1196        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
1197    let total = scalars
1198        .checked_mul(4)
1199        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
1200    if offset
1201        .checked_add(output.len())
1202        .filter(|end| *end <= total)
1203        .is_none()
1204    {
1205        return Err(PreparedArtifactError::SegmentMismatch);
1206    }
1207    let mut reads = 0_u64;
1208    for (output_index, byte) in output.iter_mut().enumerate() {
1209        let position = offset + output_index;
1210        let scalar_index = position / 4;
1211        let element = scalar_index / 2;
1212        if loaded.last.map(|(index, _)| index) != Some(element) {
1213            let x = element / shape[1];
1214            let y = element % shape[1];
1215            let complex = loaded
1216                .image
1217                .get_at(&[x, y, 0, 0])
1218                .map_err(|_| PreparedArtifactError::SourceIdentityMismatch)?;
1219            loaded.last = Some((element, complex));
1220            reads = reads
1221                .checked_add(1)
1222                .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
1223        }
1224        let complex = loaded.last.expect("loaded element").1;
1225        let scalar = if scalar_index % 2 == 0 {
1226            complex.re
1227        } else {
1228            complex.im
1229        };
1230        *byte = scalar.to_le_bytes()[position % 4];
1231    }
1232    Ok(reads)
1233}

1235fn decode_complex32_plane(
1236    bytes: Vec<u8>,
1237    metadata: &KernelMetadata,
1238) -> Result<Array2<Complex32>, CasaAwCacheError> {
1239    let chunks = bytes.chunks_exact(8);
1240    if !chunks.remainder().is_empty() {
1241        return Err(fail(
1242            &metadata.path,
1243            "private complex payload is not element aligned",
1244        ));
1245    }
1246    let values = chunks
1247        .map(|chunk| {
1248            Complex32::new(
1249                f32::from_le_bytes(chunk[..4].try_into().expect("four bytes")),
1250                f32::from_le_bytes(chunk[4..].try_into().expect("four bytes")),
1251            )
1252        })
1253        .collect::<Vec<_>>();
1254    Array2::from_shape_vec((metadata.shape[0], metadata.shape[1]), values).map_err(|error| {
1255        fail(
1256            &metadata.path,
1257            format!("cannot decode private plane: {error}"),
1258        )
1259    })
1260}

1278fn adapt_kernel_from_plane(
1279    metadata: &KernelMetadata,
1280    plane: Array2<Complex32>,

1293fn take_plane_storage(
1294    metadata: &KernelMetadata,
1295    plane: Array2<Complex32>,
1296) -> Result<Vec<Complex32>, CasaAwCacheError> {
1297    if plane.shape() != metadata.shape || !plane.is_standard_layout() {
1298        return Err(fail(
1299            &metadata.path,
1300            "decoded plane has an unexpected shape or layout",
1301        ));
1302    }
1303    let (taps, offset) = plane.into_raw_vec_and_offset();
1304    if offset != Some(0) || metadata.shape[0].checked_mul(metadata.shape[1]) != Some(taps.len()) {
1305        return Err(fail(
1306            &metadata.path,
1307            "decoded plane does not own exactly its logical storage",
1308        ));
1309    }
1310    Ok(taps)
1311}

1313fn kernel_layout(metadata: &KernelMetadata) -> Result<AwKernelLayout, CasaAwCacheError> {
1314    let reference = metadata.uv.reference_pixel.map(f64::from_bits);
1315    let rounded = reference.map(|value| value.round());
1316    if reference
1317        .into_iter()
1318        .zip(rounded)
1319        .any(|(value, rounded)| value.to_bits() != rounded.to_bits())
1320    {
1321        return Err(fail(
1322            &metadata.path,
1323            "UU/VV reference pixels must be integral",
1324        ));
1325    }
1326    let center = rounded.map(|value| value as usize);
1327    AwKernelLayout::new(metadata.support, metadata.sampling, metadata.shape, center)
1328        .map_err(|error| fail(&metadata.path, error.to_string()))
1329}

1335fn read_metadata(path: &Path) -> Result<(CasaAwCellKey, KernelMetadata), CasaAwCacheError> {
1336    let image = PagedImage::<Complex32>::open(path)
1337        .map_err(|error| fail(path, format!("cannot open Complex32 image: {error}")))?;
1338    let shape = image.shape();
1339    if shape.len() != 4 || shape[0] == 0 || shape[1] == 0 || shape[2..] != [1, 1] {
1340        return Err(fail(
1341            path,
1342            format!("expected non-empty [nx, ny, 1, 1], got {shape:?}"),
1343        ));
1344    }
1345    let (uv, polarization, frequency_hz) = validate_coordinates(path, image.coordinates())?;
1346    let misc = image.misc_info();
1347    let mueller = required_i32(&misc, "MuellerElement", path)?;
1348    let key = CasaAwCellKey {
1349        frequency_hz,
1350        w_value_lambda: required_f64(&misc, "WValue", path)?,
1351        mueller_element: u32::try_from(mueller)
1352            .map_err(|_| fail(path, "MuellerElement must be non-negative"))?,
1353        parallactic_angle_deg: required_f64(&misc, "ParallacticAngle", path)?,
1354    };
1355    if !key.w_value_lambda.is_finite()
1356        || !key.parallactic_angle_deg.is_finite()
1357        || key.mueller_element >= 16
1358    {
1359        return Err(fail(path, "invalid W, Mueller, or parallactic-angle key"));
1360    }
1361    let sampling = positive_usize(required_f64(&misc, "Sampling", path)?, "Sampling", path)?;
1362    let support = [
1363        positive_i32(required_i32(&misc, "Xsupport", path)?, "Xsupport", path)?,
1364        positive_i32(required_i32(&misc, "Ysupport", path)?, "Ysupport", path)?,
1365    ];
1366    let conjugate_polarization = u32::try_from(required_i32(&misc, "ConjPoln", path)?)
1367        .map_err(|_| fail(path, "ConjPoln must be non-negative"))?;
1368    let metadata = KernelMetadata {
1369        path: path.to_path_buf(),
1370        shape: [shape[0], shape[1]],
1371        sampling,
1372        support,
1373        uv,
1374        telescope: required_string(&misc, "TelescopeName", path)?,
1375        band: required_string(&misc, "BandName", path)?,
1376        diameter_m: required_f64(&misc, "Diameter", path)?,
1377        conjugate_frequency_hz: required_f64(&misc, "ConjFreq", path)?,
1378        conjugate_polarization,
1379        polarization,
1380        w_increment: required_f64(&misc, "WIncr", path)?,
1381        rotationally_symmetric: required_bool(&misc, "OpCode", path)?,
1382    };
1383    validate_kernel(&metadata)?;
1384    Ok((key, metadata))
1385}

1387fn validate_coordinates(
1388    path: &Path,
1389    coordinates: &CoordinateSystem,
1390) -> Result<(UvCoordinate, u32, f64), CasaAwCacheError> {
1391    if coordinates.n_pixel_axes() != 4 {
1392        return Err(fail(path, "expected exactly four CF pixel axes"));
1393    }
1394    let linear_index = coordinates
1395        .find_coordinate(CoordinateType::Linear)
1396        .ok_or_else(|| fail(path, "missing UV linear coordinate"))?;
1397    let model = coordinates.coordinate(linear_index);
1398    if model.axis_names() != ["UU", "VV"] || model.axis_units() != ["lambda", "lambda"] {
1399        return Err(fail(path, "expected UU/VV axes in lambda"));
1400    }
1401    let CoordinateModel::Linear(linear) = model else {
1402        return Err(fail(path, "UV coordinate has wrong model type"));
1403    };
1404    if linear.n_pixel_axes() != 2 || linear.n_world_axes() != 2 {
1405        return Err(fail(path, "UU/VV coordinate must have two axes"));
1406    }
1407    let reference_value = two(linear.reference_value(), path, "UV reference value")?;
1408    let reference_pixel = two(linear.reference_pixel(), path, "UV reference pixel")?;
1409    let increment = two(linear.increment(), path, "UV increment")?;
1410    let pc_matrix = linear.pc_matrix();
1411    let pc = [
1412        [pc_matrix[[0, 0]], pc_matrix[[0, 1]]],
1413        [pc_matrix[[1, 0]], pc_matrix[[1, 1]]],
1414    ];
1415    if reference_value
1416        .into_iter()
1417        .chain(reference_pixel)
1418        .chain(increment)
1419        .chain(pc.into_iter().flatten())
1420        .any(|value| !value.is_finite())
1421        || increment.contains(&0.0)
1422    {
1423        return Err(fail(
1424            path,
1425            "invalid non-finite or zero-increment UV coordinate",
1426        ));
1427    }
1428    if pc[0][0] * pc[1][1] - pc[0][1] * pc[1][0] == 0.0 {
1429        return Err(fail(path, "UV PC matrix must be invertible"));
1430    }
1431    let stokes_index = coordinates
1432        .find_coordinate(CoordinateType::Stokes)
1433        .ok_or_else(|| fail(path, "missing Stokes coordinate"))?;
1434    let CoordinateModel::Stokes(stokes) = coordinates.coordinate(stokes_index) else {
1435        return Err(fail(path, "Stokes coordinate has wrong model type"));
1436    };
1437    let [polarization] = stokes.stokes() else {
1438        return Err(fail(path, "expected one degenerate correlation plane"));
1439    };
1440    let spectral_index = coordinates
1441        .find_coordinate(CoordinateType::Spectral)
1442        .ok_or_else(|| fail(path, "missing spectral coordinate"))?;
1443    let frequency = coordinates
1444        .coordinate(spectral_index)
1445        .reference_value()
1446        .first()
1447        .copied()
1448        .ok_or_else(|| fail(path, "spectral coordinate has no reference frequency"))?;
1449    if !frequency.is_finite() || frequency <= 0.0 {
1450        return Err(fail(path, "spectral frequency must be finite and positive"));
1451    }
1452    Ok((
1453        UvCoordinate {
1454            reference_value: reference_value.map(f64::to_bits),
1455            reference_pixel: reference_pixel.map(f64::to_bits),
1456            increment: increment.map(f64::to_bits),
1457            pc: pc.map(|row| row.map(f64::to_bits)),
1458        },
1459        (*polarization).code() as u32,
1460        frequency,
1461    ))
1462}

1469        || metadata.diameter_m <= 0.0
1470        || !metadata.conjugate_frequency_hz.is_finite()
1471        || metadata.conjugate_frequency_hz <= 0.0
1472        || !metadata.w_increment.is_finite()
1473        || metadata.w_increment <= 0.0

1475        return Err(fail(
1476            &metadata.path,

1479    }

1481    Ok(())

1484fn validate_pair(
1485    key: CasaAwCellKey,
1486    imaging: &KernelMetadata,
1487    weight: &KernelMetadata,
1488) -> Result<(), CasaAwCacheError> {
1489    let common = imaging.sampling == weight.sampling
1490        && imaging.telescope == weight.telescope
1491        && imaging.band == weight.band
1492        && imaging.diameter_m.to_bits() == weight.diameter_m.to_bits()
1493        && imaging.conjugate_frequency_hz.to_bits() == weight.conjugate_frequency_hz.to_bits()
1494        && imaging.conjugate_polarization == weight.conjugate_polarization
1495        && imaging.polarization == weight.polarization
1496        && imaging.w_increment.to_bits() == weight.w_increment.to_bits()
1497        && imaging.rotationally_symmetric == weight.rotationally_symmetric;
1498    if !common {
1499        return Err(fail(
1500            &weight.path,
1501            format!("metadata does not match paired imaging cell for {key:?}"),
1502        ));
1503    }
1504    if imaging.uv.reference_value != weight.uv.reference_value
1505        || imaging.uv.pc != weight.uv.pc
1506        || !same_world_window(imaging, weight)
1507    {
1508        return Err(fail(
1509            &weight.path,
1510            format!("UV coordinate does not cover paired imaging world window for {key:?}"),
1511        ));
1512    }
1513    Ok(())
1514}

1516fn same_world_window(left: &KernelMetadata, right: &KernelMetadata) -> bool {
1517    (0..2).all(|axis| {
1518        let left_size = left.shape[axis] as f64;
1519        let right_size = right.shape[axis] as f64;
1520        nearly_equal(
1521            f64::from_bits(left.uv.reference_pixel[axis]) / left_size,
1522            f64::from_bits(right.uv.reference_pixel[axis]) / right_size,
1523        ) && nearly_equal(
1524            f64::from_bits(left.uv.increment[axis]) * left_size,
1525            f64::from_bits(right.uv.increment[axis]) * right_size,
1526        )
1527    })
1528}

1530fn validate_axes(
1531    root: &Path,
1532    entries: &BTreeMap<StableKey, Entry>,
1533) -> Result<CasaAwCacheInventory, CasaAwCacheError> {
1534    let first = entries
1535        .values()
1536        .next()
1537        .ok_or_else(|| fail(root, "cache has no paired cells"))?;
1538    if entries
1539        .values()
1540        .any(|entry| entry.imaging.w_increment.to_bits() != first.imaging.w_increment.to_bits())
1541    {
1542        return Err(fail(root, "WIncr must be identical across the cache"));
1543    }
1544    let frequency_bits = entries
1545        .values()
1546        .map(|entry| entry.key.frequency_hz.to_bits())
1547        .collect::<BTreeSet<_>>();
1548    let w_bits = entries
1549        .values()
1550        .map(|entry| entry.key.w_value_lambda.to_bits())
1551        .collect::<BTreeSet<_>>();
1552    let mueller_elements = entries
1553        .values()
1554        .map(|entry| entry.key.mueller_element)
1555        .collect::<BTreeSet<_>>();
1556    let pa_bits = entries
1557        .values()
1558        .map(|entry| entry.key.parallactic_angle_deg.to_bits())
1559        .collect::<BTreeSet<_>>();
1560    let expected = frequency_bits
1561        .len()
1562        .checked_mul(w_bits.len())
1563        .and_then(|n| n.checked_mul(mueller_elements.len()))
1564        .and_then(|n| n.checked_mul(pa_bits.len()))
1565        .ok_or_else(|| fail(root, "axis cardinality overflow"))?;
1566    if entries.len() != expected {
1567        return Err(fail(
1568            root,
1569            format!(
1570                "cache is not a complete frequency x W x Mueller x PA product: found {}, expected {expected}",
1571                entries.len()
1572            ),
1573        ));
1574    }
1575    let frequencies_hz = sorted_bits(frequency_bits);
1576    let w_values_lambda = sorted_bits(w_bits);
1577    let parallactic_angles_deg = sorted_bits(pa_bits);
1578    for (index, actual) in w_values_lambda.iter().copied().enumerate() {
1579        let expected = (index * index) as f64 / first.imaging.w_increment;
1580        if !nearly_equal_loose(actual, expected) {
1581            return Err(fail(
1582                root,
1583                format!("W plane {index} is {actual}, expected CASA quadratic value {expected}"),
1584            ));
1585        }
1586    }
1587    Ok(CasaAwCacheInventory {
1588        paired_cells: entries.len(),
1589        frequencies_hz,
1590        w_values_lambda,
1591        mueller_elements: mueller_elements.into_iter().collect(),
1592        parallactic_angles_deg,
1593    })
1594}
