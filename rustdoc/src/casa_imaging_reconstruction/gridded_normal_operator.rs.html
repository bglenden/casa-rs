33const RECORD_DOMAIN: &[u8] = b"casa-rs-gridded-normal-operator";
34const RECORD_VERSION: u32 = 2;
35const TAP_KEY_BITS: u32 = 38;
36const TAP_KEY_MASK: u64 = (1_u64 << TAP_KEY_BITS) - 1;
37const CHANNEL_KEY_BITS: u32 = 24;
38const CHANNEL_KEY_MASK: u64 = (1_u64 << CHANNEL_KEY_BITS) - 1;
39const GROUP_END_BIT: u64 = 1_u64 << (TAP_KEY_BITS + CHANNEL_KEY_BITS);
40const RECORD_KEY_MASK: u64 = (GROUP_END_BIT << 1) - 1;
41
42/// Width of every opaque gridded normal-operator record.

138impl GriddedNormalOperatorCompiler {
139    /// Compile the first vertical record owner for a supported reconstruction basis.
140    pub fn new(problem: &CompiledProblem) -> Result<Self, SpectralOperatorError> {
141        require_supported_basis(&problem.reconstruction().basis())?;
142        let specification = SpectralOperatorSpecification::new(problem)?;
143        validate_record_geometry(&specification)?;
144        let geometry = specification.operator_geometry();
145        let binding = static_binding(&specification);
146        Ok(Self {
147            finite_values: specification.finite_values(),
148            gridder: StandardConvolution::new(&geometry),
149            specification,
150            binding,
151            next_block_sequence: 0,
152            sample_count: 0,
153            record_count: 0,
154            coverage: CoverageEncoder::new(),
155            descriptors: Vec::new(),
156        })
157    }

159    /// Reduce one complete canonical weighting block and return its opaque bytes.
160    pub fn compile_block(
161        &mut self,
162        block: &WeightingReplayChunk,
163    ) -> Result<GriddedNormalOperatorBlock, SpectralOperatorError> {
164        if block.sequence() != self.next_block_sequence {
165            return Err(SpectralOperatorError::BlockSequence);
166        }
167        self.coverage.adopt(block.coverage_checkpoint());
168        let mut groups = BTreeMap::<Vec<ReducedRecordKey>, Vec<f64>>::new();
169        let mut measurements = GriddedNormalOperatorBlockMeasurements::default();
170        for weighted in block.samples() {
171            let selected = weighted.selected();
172            if !accept_weighted_input(selected, self.finite_values)?
173                || !selected
174                    .address()
175                    .correlation_type
176                    .contributes_to_stokes_i()
177            {
178                continue;
179            }
180            let uvw_m = selected.transformed_uvw_m();
181            if !selected.phase_shift_m().is_finite() || uvw_m.iter().any(|value| !value.is_finite())
182            {
183                return Err(SpectralOperatorError::InvalidSample);
184            }
185            let mut group = Vec::new();
186            let mut has_positive_weight = false;
187            for spectral in weighted.spectral_values() {
188                let contribution = spectral.contribution();
189                let output_channel = usize::try_from(contribution.output_channel())
190                    .map_err(|_| SpectralOperatorError::InvalidSample)?;
191                let frequency_hz = contribution.evaluation_frequency_hz();
192                if output_channel >= self.specification.slab().total_channels()
193                    || !frequency_hz.is_finite()
194                    || frequency_hz <= 0.0
195                    || !spectral.imaging_weight().is_finite()
196                    || spectral.imaging_weight() < 0.0
197                    || !contribution.factor().is_finite()
198                    || contribution.factor() == 0.0
199                {
200                    return Err(SpectralOperatorError::InvalidSample);
201                }
202                let scale = frequency_hz / SPEED_OF_LIGHT_M_PER_S;
203                let Some(taps) = self.gridder.taps([uvw_m[0] * scale, uvw_m[1] * scale]) else {
204                    continue;
205                };
206                let phase_angle = std::f64::consts::TAU * selected.phase_shift_m() * frequency_hz
207                    / SPEED_OF_LIGHT_M_PER_S;
208                let forward_scale = Complex64::from_polar(contribution.factor(), -phase_angle);
209                if !forward_scale.re.is_finite() || !forward_scale.im.is_finite() {
210                    return Err(SpectralOperatorError::GeneratedNonfinite);
211                }
212                has_positive_weight |= spectral.imaging_weight() > 0.0;
213                let old_capacity = group.capacity();
214                group.push(ReducedRecordKey {
215                    output_channel: contribution.output_channel(),
216                    taps: encode_taps(taps)?,
217                    forward_real: canonical_zero_bits(forward_scale.re),
218                    forward_imaginary: canonical_zero_bits(forward_scale.im),
219                    imaging_weight: canonical_zero_bits(spectral.imaging_weight()),
220                });
221                record_vector_growth(
222                    old_capacity,
223                    group.capacity(),
224                    size_of::<ReducedRecordKey>(),
225                    &mut measurements.source_group_vector_allocations,
226                    &mut measurements.source_group_capacity_growth_bytes,
227                )?;
228            }
229            if !group.is_empty() && has_positive_weight {
230                let multiplicities = match groups.entry(group) {
231                    Entry::Occupied(entry) => entry.into_mut(),
232                    Entry::Vacant(entry) => {
233                        measurements.reduction_map_entry_insertions = measurements
234                            .reduction_map_entry_insertions
235                            .checked_add(1)
236                            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
237                        entry.insert(Vec::new())
238                    }
239                };
240                let old_capacity = multiplicities.capacity();
241                multiplicities.push(1.0);
242                record_vector_growth(
243                    old_capacity,
244                    multiplicities.capacity(),
245                    size_of::<f64>(),
246                    &mut measurements.multiplicity_vector_allocations,
247                    &mut measurements.multiplicity_capacity_growth_bytes,
248                )?;
249            }
250        }
251        let encoded = encode_reduced(groups)?;
252        if !encoded.is_empty() {
253            measurements.encoded_buffer_allocations = 1;
254        }
255        measurements.encoded_buffer_bytes =
256            u64::try_from(encoded.len()).map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
257        let source_samples = u64::try_from(block.samples().len())
258            .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
259        let record_count = u64::try_from(encoded.len() / GRIDDED_NORMAL_OPERATOR_RECORD_BYTES)
260            .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
261        self.sample_count = self
262            .sample_count
263            .checked_add(source_samples)
264            .ok_or(SpectralOperatorError::CoverageOverflow)?;
265        self.record_count = self
266            .record_count
267            .checked_add(record_count)
268            .ok_or(SpectralOperatorError::CoverageOverflow)?;
269        let old_descriptor_capacity = self.descriptors.capacity();
270        self.descriptors.push(BlockDescriptor {
271            source_samples,
272            record_count,
273            digest: Sha256::digest(&encoded).into(),
274        });
275        record_vector_growth(
276            old_descriptor_capacity,
277            self.descriptors.capacity(),
278            size_of::<BlockDescriptor>(),
279            &mut measurements.descriptor_vector_allocations,
280            &mut measurements.descriptor_capacity_growth_bytes,
281        )?;
282        let result = GriddedNormalOperatorBlock {
283            sequence: self.next_block_sequence,
284            encoded,
285            measurements,
286        };
287        self.next_block_sequence = self
288            .next_block_sequence
289            .checked_add(1)
290            .ok_or(SpectralOperatorError::CoverageOverflow)?;
291        Ok(result)
292    }

294    /// Seal exhaustive coverage and exact per-block byte identities.
295    pub fn complete(
296        self,
297        replay: &WeightingReplaySummary,
298        selected_generation: SelectedObservationGenerationId,
299        continuum_transform_generation: Option<ContinuumTransformGenerationId>,
300    ) -> Result<GriddedNormalOperatorProgram, SpectralOperatorError> {
301        if self.sample_count != replay.sample_count()
302            || self.next_block_sequence != replay.block_count()
303            || self.descriptors.len()
304                != usize::try_from(replay.block_count())
305                    .map_err(|_| SpectralOperatorError::CoverageOverflow)?
306        {
307            return Err(SpectralOperatorError::IncompleteCoverage);
308        }
309        let (coverage, _) = self
310            .coverage
311            .finish(replay.weighting_generation(), self.sample_count);
312        if coverage != replay.coverage() {
313            return Err(SpectralOperatorError::IncompleteCoverage);
314        }
315        let identity = program_identity(
316            self.binding,
317            replay,
318            selected_generation,
319            continuum_transform_generation,
320            self.record_count,
321            &self.descriptors,
322        );
323        Ok(GriddedNormalOperatorProgram {
324            manifest: Arc::new(GriddedNormalOperatorManifest {
325                identity,
326                specification: self.specification,
327                weighting_generation: replay.weighting_generation(),
328                replay: replay.replay_id(),
329                coverage,
330                selected_generation,
331                continuum_transform_generation,
332                sample_count: replay.sample_count(),
333                record_count: self.record_count,
334                descriptors: self.descriptors.into_boxed_slice(),
335            }),
336        })
337    }
338}

340fn record_vector_growth(
341    old_capacity: usize,
342    new_capacity: usize,
343    element_bytes: usize,
344    allocation_operations: &mut u64,
345    capacity_growth_bytes: &mut u64,
346) -> Result<(), SpectralOperatorError> {
347    if new_capacity == old_capacity {
348        return Ok(());
349    }
350    *allocation_operations = allocation_operations
351        .checked_add(1)
352        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
353    let growth = new_capacity
354        .checked_sub(old_capacity)
355        .and_then(|elements| elements.checked_mul(element_bytes))
356        .and_then(|bytes| u64::try_from(bytes).ok())
357        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
358    *capacity_growth_bytes = capacity_growth_bytes
359        .checked_add(growth)
360        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
361    Ok(())
362}

421    /// Bind a model and prior invariant normal state to the gridded apply owner.
422    pub fn begin_apply(
423        &self,
424        problem: &CompiledProblem,
425        model: &ModelGeneration,
426        prior: FinalNormalState,
427        prepared: PreparedSpectralOperator,
428    ) -> Result<GriddedNormalOperatorApply, SpectralOperatorError> {
429        require_supported_basis(&problem.reconstruction().basis())?;
430        let (prepared_specification, workload, fft) = prepared.into_parts();
431        if problem.problem_id() != self.manifest.specification.problem_id()
432            || prepared_specification != self.manifest.specification
433            || workload.pass() != SpectralOperatorPass::ResidualRefresh
434            || prior.problem_id() != self.manifest.specification.problem_id()
435            || prior.geometry_id() != self.manifest.specification.geometry_id()
436            || prior.numerics_id() != self.manifest.specification.numerics_id()
437            || prior.weighting_commitment_id()
438                != self.manifest.specification.weighting_commitment_id()
439            || prior.weighting_generation() != self.manifest.weighting_generation
440            || prior.replay_id() != self.manifest.replay
441            || prior.coverage() != self.manifest.coverage
442            || prior.selected_generation() != self.manifest.selected_generation
443            || prior.continuum_transform_generation()
444                != self.manifest.continuum_transform_generation
445            || prior.sample_count() != self.manifest.sample_count
446            || prior.block_count() != self.block_count()
447            || prior.catalog()
448                != if self.manifest.specification.slab().total_channels() == 1 {
449                    crate::NormalStateCatalog::UnnormalizedPlaneV1
450                } else {
451                    crate::NormalStateCatalog::UnnormalizedChannelSlabV1
452                }
453            || prior.channel_count() != self.manifest.specification.slab().core_depth()
454        {
455            return Err(SpectralOperatorError::GriddedRecordMismatch);
456        }
457        let model_generation = model.generation_id();
458        let mut operator = SpectralSlabOperator::new(prepared_specification, workload, fft);
459        operator.prepare_residual_model(model, Some(prior.into_reusable()))?;
460        Ok(GriddedNormalOperatorApply {
461            program: self.clone(),
462            operator,
463            model_generation,
464            next_block_sequence: 0,
465            applied_records: 0,
466        })
467    }

480impl GriddedNormalOperatorApply {
481    /// Validate and apply one borrowed runtime frame without copying its payload.
482    pub fn apply_encoded_block(
483        &mut self,
484        sequence: u64,
485        encoded: &[u8],
486    ) -> Result<(), SpectralOperatorError> {
487        let descriptor = self
488            .program
489            .manifest
490            .descriptors
491            .get(
492                usize::try_from(sequence)
493                    .map_err(|_| SpectralOperatorError::GriddedRecordMismatch)?,
494            )
495            .ok_or(SpectralOperatorError::GriddedRecordMismatch)?;
496        if sequence != self.next_block_sequence {
497            return Err(SpectralOperatorError::BlockSequence);
498        }
499        validate_encoded_block(descriptor, encoded)?;
500        self.apply_validated_encoded(sequence, encoded)
501    }
502
503    fn apply_validated_encoded(
504        &mut self,
505        sequence: u64,
506        encoded: &[u8],
507    ) -> Result<(), SpectralOperatorError> {
508        if sequence != self.next_block_sequence {
509            return Err(SpectralOperatorError::BlockSequence);
510        }
511        let mut group_start = 0;
512        while group_start < encoded.len() {
513            let mut group_end = None;
514            let mut predicted = Complex64::default();
515            for (record_offset, bytes) in encoded[group_start..]
516                .chunks_exact(GRIDDED_NORMAL_OPERATOR_RECORD_BYTES)
517                .enumerate()
518            {
519                let record = decode_record(
520                    bytes,
521                    self.program.manifest.specification.grid_shape(),
522                    self.program.manifest.specification.slab().total_channels(),
523                )?;
524                predicted += self.operator.predict_gridded_normal(
525                    record.output_channel,
526                    record.taps,
527                    record.forward_scale,
528                )?;
529                if record.group_end {
530                    group_end = Some(
531                        group_start + (record_offset + 1) * GRIDDED_NORMAL_OPERATOR_RECORD_BYTES,
532                    );
533                    break;
534                }
535            }
536            let group_end = group_end.ok_or(SpectralOperatorError::InvalidGriddedRecord)?;
537            if !predicted.re.is_finite() || !predicted.im.is_finite() {
538                return Err(SpectralOperatorError::GeneratedNonfinite);
539            }
540            for bytes in
541                encoded[group_start..group_end].chunks_exact(GRIDDED_NORMAL_OPERATOR_RECORD_BYTES)
542            {
543                let record = decode_record(
544                    bytes,
545                    self.program.manifest.specification.grid_shape(),
546                    self.program.manifest.specification.slab().total_channels(),
547                )?;
548                self.operator.apply_gridded_normal(
549                    record.output_channel,
550                    record.taps,
551                    predicted,
552                    record.forward_scale.conj() * record.imaging_weight,
553                )?;
554            }
555            group_start = group_end;
556        }
557        let record_count = u64::try_from(encoded.len() / GRIDDED_NORMAL_OPERATOR_RECORD_BYTES)
558            .map_err(|_| SpectralOperatorError::CoverageOverflow)?;
559        self.applied_records = self
560            .applied_records
561            .checked_add(record_count)
562            .ok_or(SpectralOperatorError::CoverageOverflow)?;
563        self.next_block_sequence = self
564            .next_block_sequence
565            .checked_add(1)
566            .ok_or(SpectralOperatorError::CoverageOverflow)?;
567        Ok(())
568    }
569
570    /// Finish `dirty - A* W A x` and return ordinary Major-Cycle input.
571    pub fn finish(self) -> Result<CompleteDataOwnerResult, SpectralOperatorError> {
572        if self.next_block_sequence != self.program.block_count()
573            || self.applied_records != self.program.manifest.record_count
574        {
575            return Err(SpectralOperatorError::IncompleteCoverage);
576        }
577        let primitives = self.operator.finish_gridded_normal(self.model_generation)?;
578        let primitive_catalog = if self.program.manifest.specification.slab().total_channels() == 1
579        {
580            SpectralPrimitiveCatalog::UnnormalizedPlaneV1
581        } else {
582            SpectralPrimitiveCatalog::UnnormalizedChannelSlabV1
583        };
584        Ok(CompleteDataOwnerResult {
585            primitives,
586            completion: CompleteDataOwnerCompletion {
587                problem: self.program.manifest.specification.problem_id(),
588                geometry: self.program.manifest.specification.geometry_id(),
589                numerics: self.program.manifest.specification.numerics_id(),
590                weighting_commitment: self
591                    .program
592                    .manifest
593                    .specification
594                    .weighting_commitment_id(),
595                weighting_generation: self.program.manifest.weighting_generation,
596                replay: self.program.manifest.replay,
597                coverage: self.program.manifest.coverage,
598                coverage_proof_bytes: 0,
599                coverage_proof_hash_calls: 0,
600                primitives: primitive_catalog,
601                selected_generation: self.program.manifest.selected_generation,
602                continuum_transform_generation: self
603                    .program
604                    .manifest
605                    .continuum_transform_generation,
606                sample_count: self.program.manifest.sample_count,
607                block_count: self.program.block_count(),
608            },
609        })
610    }

613fn require_supported_basis(basis: &ReconstructionBasis) -> Result<(), SpectralOperatorError> {
614    if matches!(
615        basis,
616        ReconstructionBasis::Constant | ReconstructionBasis::ChannelLocal { .. }
617    ) {
618        Ok(())
619    } else {
620        Err(SpectralOperatorError::UnsupportedGriddedReplay)
621    }
622}
623
624fn validate_record_geometry(
625    specification: &SpectralOperatorSpecification,
626) -> Result<(), SpectralOperatorError> {
627    if specification.slab().core_depth() != specification.slab().total_channels()
628        || specification.slab().resident_depth() != specification.slab().total_channels()
629        || specification.slab().total_channels() > 1 << CHANNEL_KEY_BITS
630        || specification
631            .grid_shape()
632            .into_iter()
633            .any(|extent| extent > 1 << 12)
634    {
635        return Err(SpectralOperatorError::UnsupportedGriddedReplay);
636    }
637    Ok(())
638}

640fn static_binding(specification: &SpectralOperatorSpecification) -> LogicalIdentity {
641    let mut encoder = Encoder::new(RECORD_DOMAIN, RECORD_VERSION);
642    encoder.identity(specification.problem_id().as_bytes());
643    encoder.identity(specification.geometry_id().as_bytes());
644    encoder.identity(specification.numerics_id().as_bytes());
645    encoder.identity(specification.weighting_commitment_id().as_bytes());
646    encoder.usize(specification.grid_shape()[0]);
647    encoder.usize(specification.grid_shape()[1]);
648    encoder.usize(specification.slab().total_channels());
649    encoder.usize(GRIDDED_NORMAL_OPERATOR_RECORD_BYTES);
650    LogicalIdentity::from_sha256(encoder.finish())
651}

653fn program_identity(
654    binding: LogicalIdentity,
655    replay: &WeightingReplaySummary,
656    selected_generation: SelectedObservationGenerationId,
657    continuum_transform_generation: Option<ContinuumTransformGenerationId>,
658    record_count: u64,
659    descriptors: &[BlockDescriptor],
660) -> LogicalIdentity {
661    let mut encoder = Encoder::new(RECORD_DOMAIN, RECORD_VERSION + 1);
662    encoder.identity(binding.as_bytes());
663    encoder.identity(replay.weighting_generation().as_bytes());
664    encoder.identity(replay.replay_id().as_bytes());
665    encoder.identity(replay.coverage().as_bytes());
666    encoder.identity(selected_generation.as_bytes());
667    match continuum_transform_generation {
668        Some(generation) => {
669            encoder.u8(1);
670            encoder.identity(generation.as_bytes());
671        }
672        None => encoder.u8(0),
673    }
674    encoder.u64(replay.sample_count());
675    encoder.u64(replay.block_count());
676    encoder.u64(record_count);
677    for descriptor in descriptors {
678        encoder.u64(descriptor.source_samples);
679        encoder.u64(descriptor.record_count);
680        encoder.identity(descriptor.digest);
681    }
682    LogicalIdentity::from_sha256(encoder.finish())
683}

685fn encode_reduced(
686    groups: BTreeMap<Vec<ReducedRecordKey>, Vec<f64>>,
687) -> Result<Box<[u8]>, SpectralOperatorError> {
688    let record_count = groups.keys().try_fold(0_usize, |total, records| {
689        total
690            .checked_add(records.len())
691            .ok_or(SpectralOperatorError::ResidencyOverflow)
692    })?;
693    let capacity = record_count
694        .checked_mul(GRIDDED_NORMAL_OPERATOR_RECORD_BYTES)
695        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
696    let mut encoded = Vec::with_capacity(capacity);
697    for (records, mut multiplicities) in groups {
698        if records.is_empty() {
699            return Err(SpectralOperatorError::InvalidGriddedRecord);
700        }
701        multiplicities.sort_by(f64::total_cmp);
702        let multiplicity = compensated_sum(&multiplicities)?;
703        let last = records.len() - 1;
704        for (index, record) in records.into_iter().enumerate() {
705            let output_channel = u64::from(record.output_channel);
706            if output_channel > CHANNEL_KEY_MASK || record.taps & !TAP_KEY_MASK != 0 {
707                return Err(SpectralOperatorError::InvalidGriddedRecord);
708            }
709            let key = record.taps
710                | (output_channel << TAP_KEY_BITS)
711                | if index == last { GROUP_END_BIT } else { 0 };
712            let forward_real = f64::from_bits(record.forward_real);
713            let forward_imaginary = f64::from_bits(record.forward_imaginary);
714            let imaging_weight = f64::from_bits(record.imaging_weight) * multiplicity;
715            if !forward_real.is_finite()
716                || !forward_imaginary.is_finite()
717                || (forward_real == 0.0 && forward_imaginary == 0.0)
718                || !imaging_weight.is_finite()
719                || imaging_weight.is_sign_negative()
720            {
721                return Err(SpectralOperatorError::GeneratedNonfinite);
722            }
723            encoded.extend_from_slice(&key.to_le_bytes());
724            encoded.extend_from_slice(&forward_real.to_le_bytes());
725            encoded.extend_from_slice(&forward_imaginary.to_le_bytes());
726            encoded.extend_from_slice(&imaging_weight.to_le_bytes());
727        }
728    }
729    Ok(encoded.into_boxed_slice())
730}

732fn canonical_zero_bits(value: f64) -> u64 {
733    if value == 0.0 { 0 } else { value.to_bits() }
734}
735
736fn compensated_sum(values: &[f64]) -> Result<f64, SpectralOperatorError> {
737    let mut sum = 0.0;
738    let mut compensation = 0.0;
739    for value in values {
740        let corrected = *value - compensation;
741        let updated = sum + corrected;
742        compensation = (updated - sum) - corrected;
743        sum = updated;
744    }
745    if sum.is_finite() && !sum.is_sign_negative() {
746        Ok(sum)
747    } else {
748        Err(SpectralOperatorError::GeneratedNonfinite)
749    }
750}

752fn encode_taps(taps: SampleTaps) -> Result<u64, SpectralOperatorError> {
753    if taps.x.start >= 1 << 12
754        || taps.y.start >= 1 << 12
755        || taps.x.weight_index >= 1 << 7
756        || taps.y.weight_index >= 1 << 7
757    {
758        return Err(SpectralOperatorError::InvalidGriddedRecord);
759    }
760    Ok(taps.x.start as u64
761        | ((taps.y.start as u64) << 12)
762        | ((taps.x.weight_index as u64) << 24)
763        | ((taps.y.weight_index as u64) << 31))
764}

766fn decode_record(
767    encoded: &[u8],
768    grid_shape: [usize; 2],
769    output_channels: usize,
770) -> Result<DecodedRecord, SpectralOperatorError> {
771    if encoded.len() != GRIDDED_NORMAL_OPERATOR_RECORD_BYTES {
772        return Err(SpectralOperatorError::InvalidGriddedRecord);
773    }
774    let key = u64::from_le_bytes(
775        encoded[..8]
776            .try_into()
777            .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?,
778    );
779    let forward_real = f64::from_le_bytes(
780        encoded[8..16]
781            .try_into()
782            .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?,
783    );
784    let forward_imaginary = f64::from_le_bytes(
785        encoded[16..24]
786            .try_into()
787            .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?,
788    );
789    let imaging_weight = f64::from_le_bytes(
790        encoded[24..]
791            .try_into()
792            .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?,
793    );
794    let output_channel = usize::try_from((key >> TAP_KEY_BITS) & CHANNEL_KEY_MASK)
795        .map_err(|_| SpectralOperatorError::InvalidGriddedRecord)?;
796    if key & !RECORD_KEY_MASK != 0
797        || output_channel >= output_channels
798        || !forward_real.is_finite()
799        || !forward_imaginary.is_finite()
800        || (forward_real == 0.0 && forward_imaginary == 0.0)
801        || !imaging_weight.is_finite()
802        || imaging_weight.is_sign_negative()
803    {
804        return Err(SpectralOperatorError::InvalidGriddedRecord);
805    }
806    let tap_key = key & TAP_KEY_MASK;
807    let taps = SampleTaps {
808        x: TapSpan {
809            start: (tap_key & 0x0fff) as usize,
810            weight_index: ((tap_key >> 24) & 0x7f) as usize,
811        },
812        y: TapSpan {
813            start: ((tap_key >> 12) & 0x0fff) as usize,
814            weight_index: ((tap_key >> 31) & 0x7f) as usize,
815        },
816    };
817    if taps.x.weight_index > OVERSAMPLING
818        || taps.y.weight_index > OVERSAMPLING
819        || taps
820            .x
821            .start
822            .checked_add(2 * SUPPORT)
823            .is_none_or(|end| end >= grid_shape[0])
824        || taps
825            .y
826            .start
827            .checked_add(2 * SUPPORT)
828            .is_none_or(|end| end >= grid_shape[1])
829    {
830        return Err(SpectralOperatorError::InvalidGriddedRecord);
831    }
832    Ok(DecodedRecord {
833        output_channel,
834        taps,
835        forward_scale: Complex64::new(forward_real, forward_imaginary),
836        imaging_weight,
837        group_end: key & GROUP_END_BIT != 0,
838    })
839}
840
841fn validate_encoded_block(
842    descriptor: &BlockDescriptor,
843    encoded: &[u8],
844) -> Result<(), SpectralOperatorError> {
845    let expected_bytes = usize::try_from(descriptor.record_count)
846        .ok()
847        .and_then(|records| records.checked_mul(GRIDDED_NORMAL_OPERATOR_RECORD_BYTES))
848        .ok_or(SpectralOperatorError::GriddedRecordMismatch)?;
849    if encoded.len() != expected_bytes
850        || <[u8; 32]>::from(Sha256::digest(encoded)) != descriptor.digest
851    {
852        return Err(SpectralOperatorError::GriddedRecordMismatch);
853    }
854    Ok(())
855}

930    fn grouped_record_apply_matches_scalar_normal_operator_below_ceiling() {
931        let geometry = geometry();
932        let gridder = StandardConvolution::new(&geometry);
933        let shape = (geometry.grid_shape[0], geometry.grid_shape[1]);
934        let mut model_grid = Array2::<Complex64>::zeros(shape);
935        for ((x, y), value) in model_grid.indexed_iter_mut() {
936            *value = Complex64::new(
937                (x * shape.1 + y) as f64 * 0.013 - 0.4,
938                x as f64 * -0.017 + y as f64 * 0.009,
939            );
940        }
941        let first = gridder.taps([0.0, 0.0]).expect("central taps");
942        let second = gridder
943            .taps([20.0, -14.0])
944            .expect("offset taps remain on grid");
945        let contributions = [
946            (first, 0.75),
947            (first, 1.25),
948            (second, 0.4),
949            (second, 0.6),
950            (first, 0.125),
951            (first, 0.375),
952        ];
953        let mut direct = Array2::<Complex64>::zeros(shape);
954        let mut direct_compensation = Array2::<Complex64>::zeros(shape);
955        for (taps, coefficient) in contributions {
956            let predicted = gridder.degrid(&model_grid, taps);
957            gridder.grid_compensated(
958                &mut direct,
959                &mut direct_compensation,
960                taps,
961                predicted * coefficient,
962            );
963        }
964        let encoded = encode_reduced(scalar_groups(contributions)).expect("reduce records");
965        let mut grouped = Array2::<Complex64>::zeros(shape);
966        let mut grouped_compensation = Array2::<Complex64>::zeros(shape);
967        for record in encoded.chunks_exact(GRIDDED_NORMAL_OPERATOR_RECORD_BYTES) {
968            let record = decode_record(record, geometry.grid_shape, 1).expect("decode record");
969            let predicted = gridder.degrid(&model_grid, record.taps) * record.forward_scale;
970            gridder.grid_compensated(
971                &mut grouped,
972                &mut grouped_compensation,
973                record.taps,
974                predicted * record.forward_scale.conj() * record.imaging_weight,
975            );
976        }
977        let squared_error = direct
978            .iter()
979            .zip(&grouped)
980            .map(|(expected, actual)| (*actual - *expected).norm_sqr())
981            .sum::<f64>();
982        let squared_reference = direct.iter().map(Complex64::norm_sqr).sum::<f64>();
983        let normalized_rms = (squared_error / squared_reference.max(f64::MIN_POSITIVE)).sqrt();
984        assert!(normalized_rms <= 0.001, "normalized RMS {normalized_rms}");
985
986        let dirty = Complex64::new(4.0, -0.5);
987        let normal = Complex64::new(1.25, 0.75);
988        assert_eq!(dirty - normal, Complex64::new(2.75, -1.25));
989    }
