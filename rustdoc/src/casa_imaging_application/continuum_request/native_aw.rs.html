13/// Bounded metadata acquisition; pixel generation remains in the admitted phase.
14#[allow(clippy::too_many_arguments)]
15pub(super) fn resolve(
16    request: &ContinuumImagingRequest,
17    controls: &NativeEvlaAwCache,
18    ms: &MeasurementSet,
19    windows: &[SourceSpectralWindow],
20    spectral: &PreparedSpectralAxis,
21    first_row: SelectedObservationRow,
22    geometry_engine: &casa_ms::derived::engine::MsCalEngine,
23) -> Result<NativeAwRequestInput, crate::ApplicationError> {
24    let aw = request.aw_projection.as_ref().expect("native AW source");
25    let observation = ms.observation()?;
26    if observation
27        .string(first_row.observation_id() as usize, "TELESCOPE_NAME")?
28        .trim()
29        != "EVLA"
30    {
31        return Err(boxed(
32            "native aperture generation requires an EVLA observation",
33        ));
34    }
35    let antenna = ms.antenna()?;
36    for row in 0..antenna.row_count() {
37        if antenna.dish_diameter(row)? != 25.0 {
38            return Err(boxed(
39                "native EVLA generation requires the explicit homogeneous 25 m dish model",
40            ));
41        }
42    }
43    // The reference data is an explicit input with bounded acquisition, never
44    // discovered through a CASA installation or downloaded during execution.
45    let mut bytes = Vec::new();
46    std::fs::File::open(&controls.surface)?
47        .take(1_048_577)
48        .read_to_end(&mut bytes)?;
49    if bytes.len() > 1_048_576 {
50        return Err(boxed(
51            "native EVLA surface exceeds the 1 MiB reference-data bound",
52        ));
53    }
54    let surface = EvlaDishSurface::from_surface_text(std::str::from_utf8(&bytes)?)?;
55    let mut frequencies = windows
56        .iter()
57        .map(|window| {
58            let selected = spectral
59                .selected_source_channels
60                .get(&window.spw_id)
61                .ok_or_else(|| boxed("native AW selected SPW is absent from the spectral owner"))?;
62            let channel_frequencies_hz = selected
63                .iter()
64                .map(|index| {
65                    window
66                        .frequencies_hz
67                        .get(*index)
68                        .copied()
69                        .ok_or_else(|| boxed("native AW selected channel is outside its SPW"))
70                })
71                .collect::<Result<Vec<_>, _>>()?;
72            // TransformMachines2::makeFreqValList selects each SPW's high endpoint.
73            let cf_frequency_hz = channel_frequencies_hz
74                .iter()
75                .copied()
76                .reduce(f64::max)
77                .ok_or_else(|| boxed("native AW SPW has no selected frequencies"))?;
78            Ok(NativeAwFrequencyGroup {
79                spectral_window: u32::try_from(window.spw_id)?,
80                channel_frequencies_hz,
81                cf_frequency_hz,
82            })
83        })
84        .collect::<Result<Vec<_>, crate::ApplicationError>>()?;
85    frequencies.sort_by(|left, right| left.cf_frequency_hz.total_cmp(&right.cf_frequency_hz));
86    if !aw.wideband {
87        let selected_frequencies = frequencies
88            .iter()
89            .flat_map(|g| g.channel_frequencies_hz.iter().copied())
90            .collect();
91        frequencies = vec![NativeAwFrequencyGroup {
92            spectral_window: frequencies[0].spectral_window,
93            channel_frequencies_hz: selected_frequencies,
94            cf_frequency_hz: spectral.reference_frequency_hz,
95        }];
96    }
97    let planes = aw.w_plane_count.expect("validated native AW W-plane count");
98    let sky_cell = request.cell_arcsec.to_radians() / 3600.0;
99    // CASA AWConvFunc derives its W grid from the requested field of view,
100    // not the observed W envelope: maxUVW=1/(4*sky_increment), w=i²/wScale.
101    let max_w = 1.0 / (sky_cell * 4.0);
102    let w_increment = ((planes - 1)
103        .checked_mul(planes - 1)
104        .ok_or_else(|| boxed("native AW W-grid size overflowed"))? as f32)
105        as f64
106        / max_w;
107    let w_values = (0..planes)
108        .map(|index| (index * index) as f64 / w_increment)
109        .collect();
110    let working_cell = sky_cell * controls.oversampling as f64 * request.image_size as f64
111        / controls.working_size as f64;
112    let pa = geometry_engine.parallactic_angle(
113        first_row.time_mjd_seconds(),
114        first_row.field_id() as usize,
115        0,
116    )? as f32;
117    let feed_angle = receptor_zero_angle(
118        ms,
119        first_row.time_mjd_seconds(),
120        frequencies[0].spectral_window,
121    )? as f32;
122    let pa = f64::from(pa + feed_angle);
123    let mueller_elements = match request.polarizations.as_slice() {
124        [PolarizationCoordinate::CircularRr] => vec![0],
125        [PolarizationCoordinate::CircularLl] => vec![15],
126        [PolarizationCoordinate::StokesI] => vec![0, 15],
127        _ => {
128            return Err(boxed(
129                "native EVLA AW currently supports Stokes I or one circular parallel hand",
130            ));
131        }
132    };
133    Ok(NativeAwRequestInput {
134        surface,
135        antenna_diameter_m: 25.0,
136        frequencies,
137        w_values,
138        w_increment,
139        pa_values: vec![pa],
140        mueller_elements,
141        reference_frequency_hz: spectral.reference_frequency_hz,
142        grid: NativeAwGrid {
143            size: controls.working_size,
144            sky_increment_rad: [-working_cell, working_cell],
145            oversampling: controls.oversampling,
146        },
147        terms: NativeAwTerms {
148            aperture: aw.a_term,
149            w_term: true,
150            prolate_spheroidal: aw.ps_term,
151            wideband: aw.wideband,
152            conjugate_beams: aw.conjugate_beams,
153        },
154        maximum_cells: controls.maximum_cells,
155    })
156}

158fn receptor_zero_angle(
159    ms: &MeasurementSet,
160    time: f64,
161    spw: u32,
162) -> Result<f64, crate::ApplicationError> {
163    let feed = ms.feed()?;
164    let mut angle = None;
165    for row in 0..feed.row_count() {
166        if feed.i32(row, "ANTENNA_ID")? != 0 || feed.i32(row, "FEED_ID")? != 0 {
167            continue;
168        }
169        let spectral_window = feed.i32(row, "SPECTRAL_WINDOW_ID")?;
170        if spectral_window != -1 && spectral_window != spw as i32 {
171            continue;
172        }
173        let center = feed.f64(row, "TIME")?;
174        let interval = feed.f64(row, "INTERVAL")?;
175        if interval != 0.0 && (time - center).abs() > interval / 2.0 {
176            continue;
177        }
178        let ArrayValue::Float64(values) = feed.array(row, "RECEPTOR_ANGLE")? else {
179            return Err(boxed(
180                "native EVLA feed receptor angles require Float64 metadata",
181            ));
182        };
183        let value = values
184            .iter()
185            .next()
186            .copied()
187            .filter(|value| value.is_finite())
188            .ok_or_else(|| boxed("native EVLA feed has no finite receptor-zero angle"))?;
189        if angle.is_some_and(|previous: f64| previous.to_bits() != value.to_bits()) {
190            return Err(boxed(
191                "native EVLA receptor-zero angle is ambiguous for the selected epoch/SPW",
192            ));
193        }
194        angle = Some(value);
195    }
196    angle.ok_or_else(|| boxed("native EVLA request has no applicable antenna-zero feed metadata"))
197}
