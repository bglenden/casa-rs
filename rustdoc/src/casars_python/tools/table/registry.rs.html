2//! The `table` tools of this process, as casacore's table cache lists the
3//! tables a process has open.
4//!
5//! casacore's tools share one table object per table, so a call on one tool
6//! can see another's lock, refuse to replace a table another tool holds
7//! (`prepareCopyRename`) and release every automatic lock of the process
8//! (`Table::relinquishAutoLocks`). Here each tool has a handle of its own;
9//! this registry lets a call visit the other tools.
10//!
11//! A call holds its own tool's mutex while it visits the others, so the
12//! others are visited with `try_lock`: one that is busy in another thread is
13//! reported as such, never waited for (two tools visiting each other would
14//! deadlock).
15
16use std::sync::{Arc, Mutex, PoisonError, TryLockError, Weak};
17
18use super::state::State;
19
20/// Every tool made in this process that is still alive, with its
21/// `State::id`.
22static TOOLS: Mutex<Vec<(u64, Weak<Mutex<State>>)>> = Mutex::new(Vec::new());
23
24/// `state` as a tool's shared state, listed with the process's tools.
25pub(super) fn track(state: State) -> Arc<Mutex<State>> {
26    let id = state.id;
27    let tool = Arc::new(Mutex::new(state));
28    let mut tools = TOOLS.lock().unwrap_or_else(PoisonError::into_inner);
29    tools.retain(|(_, other)| other.strong_count() > 0);
30    tools.push((id, Arc::downgrade(&tool)));
31    tool
32}
33
34/// Calls `visit` on the state of every tool but `me` (a tool's `State::id`,
35/// the caller's, whose mutex the caller holds): with `Some(state)` for a tool
36/// that is free, with `None` for one that is busy in another call.
37pub(super) fn for_each_other(me: u64, mut visit: impl FnMut(Option<&mut State>)) {
38    let tools: Vec<Arc<Mutex<State>>> = TOOLS
39        .lock()
40        .unwrap_or_else(PoisonError::into_inner)
41        .iter()
42        .filter(|(id, _)| *id != me)
43        .filter_map(|(_, tool)| tool.upgrade())
44        .collect();
45    for tool in tools {
46        match tool.try_lock() {
47            Ok(mut state) => visit(Some(&mut state)),
48            Err(TryLockError::Poisoned(poisoned)) => visit(Some(&mut poisoned.into_inner())),
49            Err(TryLockError::WouldBlock) => visit(None),
50        }
51    }
52}
