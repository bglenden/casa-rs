25/// A column's `MEASINFO` record, if it has one.
26fn measinfo<'t>(table: &'t Table, column: &str) -> Option<&'t casa_types::RecordValue> {
27    match table.column_keywords(column)?.get("MEASINFO")? {
28        Value::Record(record) => Some(record),
29        _ => None,
30    }
31}
32
33/// A column's fixed `MEASINFO.Ref`, if it names one. A variable reference
34/// (`VarRefCol`) is refused: the trial's views carry one reference per
35/// batch.
36pub(crate) fn measure_reference(
37    table: &Table,
38    column: &str,
39) -> Result<Option<String>, ObservationError> {
40    let Some(record) = measinfo(table, column) else {
41        return Ok(None);
42    };
43    if record.get("VarRefCol").is_some() {
44        return Err(ObservationError::Unsupported {
45            what: "an MSv2 measure column with a variable reference (VarRefCol)",
46        });
47    }
48    Ok(match record.get("Ref") {
49        Some(Value::Scalar(ScalarValue::String(reference))) => Some(reference.clone()),
50        _ => None,
51    })
52}

54/// Require a column's `QuantumUnits`, when present, to be `expected` in
55/// every position. An absent keyword is the MeasurementSet definition's
56/// unit, which is `expected`.
57pub(crate) fn require_units(
58    table: &Table,
59    column: &str,
60    expected: &str,
61) -> Result<(), ObservationError> {
62    let Some(units) = quantum_units(table, column) else {
63        return Ok(());
64    };
65    if units.iter().all(|unit| unit == expected) {
66        Ok(())
67    } else {
68        Err(invalid(format!(
69            "{column} is in {units:?}; the trial adapter reads only {expected} and does not convert"
70        )))
71    }
72}
73
74fn quantum_units(table: &Table, column: &str) -> Option<Vec<String>> {
75    match table.column_keywords(column)?.get("QuantumUnits")? {
76        Value::Array(ArrayValue::String(units)) => Some(units.iter().cloned().collect()),
77        Value::Scalar(ScalarValue::String(unit)) => Some(vec![unit.clone()]),
78        _ => None,
79    }
80}

82/// What visibility column `column`'s `QuantumUnits` declares: janskys
83/// only when it says `Jy`.
84pub(super) fn data_units(table: &Table, column: &str) -> VisibilityUnits {
85    match quantum_units(table, column).as_deref() {
86        None | Some([]) => VisibilityUnits::Undeclared,
87        Some([unit, ..]) if unit == "Jy" => VisibilityUnits::Jansky,
88        Some([unit, ..]) => VisibilityUnits::Declared(unit.clone()),
89    }
90}

92/// The time scale of MAIN `TIME` and `TIME_CENTROID`, in seconds. The
93/// MeasurementSet definition's default reference is UTC; another fixed
94/// reference is kept, a variable one refused, and the two columns must
95/// agree.
96pub(super) fn time_scale(main: &Table) -> Result<EpochRef, ObservationError> {
97    let mut scales = Vec::new();
98    for column in ["TIME", "TIME_CENTROID"] {
99        require_units(main, column, "s")?;
100        let scale = match measure_reference(main, column)? {
101            Some(name) => name
102                .parse::<EpochRef>()
103                .map_err(|_| invalid(format!("unknown {column} reference {name}")))?,
104            None => EpochRef::UTC,
105        };
106        scales.push(scale);
107    }
108    for column in ["INTERVAL", "EXPOSURE"] {
109        require_units(main, column, "s")?;
110    }
111    if scales[0] == scales[1] {
112        Ok(scales[0])
113    } else {
114        Err(ObservationError::Unsupported {
115            what: "MSv2 TIME and TIME_CENTROID in different time scales",
116        })
117    }
118}

129/// Each `FIELD` row's `PHASE_DIR` frame: fixed for the column, or
130/// variable per row (`VarRefCol`, integer or string codes, as ALMA's
131/// `PhaseDir_Ref`).
132pub(crate) fn phase_frames(table: &Table) -> Result<Vec<DirectionRef>, ObservationError> {
133    require_units(table, "PHASE_DIR", "rad")?;
134    (0..table.row_count())
135        .map(|row| {
136            resolve_direction_reference_selected(table, "FIELD", "PHASE_DIR", row).map_err(storage)
137        })
138        .collect()
139}

141/// The constant term of FIELD `column` at `row`, a direction polynomial of
142/// shape `[2, NUM_POLY + 1]` (the row's `num_poly`), refused at any other
143/// shape; `None` where the column is absent or the cell undefined.
144fn field_angles(
145    table: &Table,
146    column: &str,
147    row: usize,
148    num_poly: i32,
149) -> Result<Option<(f64, f64)>, ObservationError> {
150    let Some(value) = crate::subtables::optional_array(table, row, column).map_err(storage)? else {
151        return Ok(None);
152    };
153    let ArrayValue::Float64(values) = value else {
154        return Err(invalid(format!("FIELD {column} is not a Float64 array")));
155    };
156    let terms = usize::try_from(num_poly).ok().map(|order| order + 1);
157    if values.shape() != [2, terms.unwrap_or(0)] || terms.is_none() {
158        return Err(invalid(format!(
159            "FIELD {column} of row {row} has shape {:?}; NUM_POLY {num_poly} makes it [2, {}]",
160            values.shape(),
161            i64::from(num_poly) + 1
162        )));
163    }
164    Ok(Some((values[[0, 0]], values[[1, 0]])))
165}

167/// The fields, their ephemerides, and the UVW convention; `sources` gives
168/// each stored `SOURCE_ID`'s logical source.
169pub(super) fn fields_and_uvw(
170    ms: &MeasurementSet,
171    sources: &BTreeMap<i32, casa_observation::SourceId>,
172) -> Result<Fields, ObservationError> {
173    let field = ms.field().map_err(storage)?;
174    let field_table = ms
175        .subtable(SubtableId::Field)
176        .ok_or_else(|| invalid("no FIELD subtable"))?;
177    // The fields share one frame: a data group has one UVW frame, and MSv4
178    // one phase-centre frame. Without fields it is J2000, the definition's
179    // default.
180    let frames = phase_frames(field_table)?;
181    let phase_frame = frames.first().copied().unwrap_or(DirectionRef::J2000);
182    if frames.iter().any(|frame| *frame != phase_frame) {
183        return Err(ObservationError::Unsupported {
184            what: "MSv2 fields whose phase centres are in different frames",
185        });
186    }
187    require_units(field_table, "TIME", "s")?;
188    let epoch_scale = match measure_reference(field_table, "TIME")? {
189        Some(name) => name
190            .parse::<EpochRef>()
191            .map_err(|_| invalid(format!("unknown FIELD TIME reference {name}")))?,
192        None => EpochRef::UTC,
193    };
194    let ephemerides = (0..field.row_count())
195        .map(|row| field.ephemeris_id(row).map_err(storage))
196        .collect::<Result<Vec<_>, ObservationError>>()?;
197    // What each field follows, read as today's imaging reads it: a field
198    // whose `NUM_POLY` is above zero follows its polynomial instead. The
199    // boundary holds every field, so a table that cannot be read refuses
200    // the opening.
201    let attached = crate::ephemeris::field_motions(ms)
202        .map_err(storage)?
203        .iter()
204        .map(|motion| match motion {
205            crate::ephemeris::FieldMotion::Fixed => Ok(None),
206            crate::ephemeris::FieldMotion::Ephemeris(series) => Ok(Some(series.clone())),
207            crate::ephemeris::FieldMotion::Unreadable(reason) => {
208                Err(invalid(format!("the ephemeris a field follows: {reason}")))
209            }
210        })
211        .collect::<Result<Vec<_>, ObservationError>>()?;
212    for column in ["DELAY_DIR", "REFERENCE_DIR"] {
213        require_units(field_table, column, "rad")?;
214    }
215    // A delay or reference centre the row leaves undefined is unstated.
216    let centre = |row: usize, column: &str, num_poly: i32| {
217        let Some((longitude, latitude)) = field_angles(field_table, column, row, num_poly)? else {
218            return Ok(None);
219        };
220        let frame = resolve_direction_reference_selected(field_table, "FIELD", column, row)
221            .map_err(storage)?;
222        Ok::<_, ObservationError>(Some(MDirection::from_angles(longitude, latitude, frame)))
223    };
224    let fields = (0..field.row_count())
225        .map(|row| {
226            let num_poly = field.num_poly(row).map_err(storage)?;
227            let (longitude, latitude) = field_angles(field_table, "PHASE_DIR", row, num_poly)?
228                .ok_or_else(|| invalid(format!("FIELD PHASE_DIR of row {row} is undefined")))?;
229            let source = field.source_id(row).map_err(storage)?;
230            let time_s = field.time(row).map_err(storage)?;
231            Ok(FieldEntry {
232                name: field.name(row).map_err(storage)?,
233                phase_centre: MDirection::from_angles(longitude, latitude, frames[row]),
234                reference_epoch: Some(MEpoch::from_mjd(time_s / 86_400.0, epoch_scale)),
235                source_name: source_name(ms, source)?,
236                source: sources.get(&source).copied(),
237                code: crate::subtables::optional_string(field_table, row, "CODE")
238                    .map_err(storage)?,
239                delay_centre: centre(row, "DELAY_DIR", num_poly)?,
240                reference_centre: centre(row, "REFERENCE_DIR", num_poly)?,
241                flagged: crate::subtables::optional_bool(field_table, row, "FLAG_ROW")
242                    .map_err(storage)?,
243                ephemeris: attached[row].clone(),
244            })
245        })
246        .collect::<Result<_, ObservationError>>()?;
247    let main = ms.main_table();
248    require_units(main, "UVW", "m")?;
249    let stored_label = measure_reference(main, "UVW")?
250        .map(|name| {
251            name.parse::<DirectionRef>()
252                .map_err(|_| invalid(format!("unknown UVW reference {name}")))
253        })
254        .transpose()?;
255    // CASA's convention: UVW are in the phase centre's frame. An ITRF label
256    // (casacore's default, which simulators and fillers write) is the known
257    // mislabelling; any other label must name the phase centres' frame.
258    match stored_label {
259        None | Some(DirectionRef::ITRF) => {}
260        Some(label) if label == phase_frame => {}
261        Some(_) => {
262            return Err(ObservationError::Unsupported {
263                what: "MSv2 UVW labelled with a celestial frame other than the phase centres'",
264            });
265        }
266    }
267    Ok(Fields {
268        entries: fields,
269        ephemerides,
270        uvw_frame: UvwFrame {
271            effective: phase_frame,
272            stored_label,
273        },
274    })
275}

277/// Every `OBSERVATION` row. A MeasurementSet without one gets a single
278/// empty entry, which every row's `OBSERVATION_ID` 0 then names.
279pub(super) fn read_observations(
280    ms: &MeasurementSet,
281) -> Result<Vec<ObservationEntry>, ObservationError> {
282    let observation = ms.observation().map_err(storage)?;
283    if observation.row_count() == 0 {
284        return Ok(vec![ObservationEntry::default()]);
285    }
286    let table = observation.table();
287    let range_scale = super::metadata::epoch_scale(table, "TIME_RANGE")?;
288    let release_scale = super::metadata::epoch_scale(table, "RELEASE_DATE")?;
289    (0..observation.row_count())
290        .map(|row| {
291            let schedule: Vec<String> = match observation
292                .optional_array(row, "SCHEDULE")
293                .map_err(storage)?
294            {
295                Some(ArrayValue::String(lines)) => lines.iter().cloned().collect(),
296                _ => Vec::new(),
297            };
298            let (scheduling_block, execution_block) = schedule_identifiers(&schedule);
299            Ok(ObservationEntry {
300                telescope: observation.string(row, "TELESCOPE_NAME").map_err(storage)?,
301                observers: vec![observation.string(row, "OBSERVER").map_err(storage)?],
302                project: observation.string(row, "PROJECT").map_err(storage)?,
303                execution_block,
304                session_reference: None,
305                scheduling_block,
306                time_range: super::metadata::floats(
307                    observation
308                        .optional_array(row, "TIME_RANGE")
309                        .map_err(storage)?,
310                    "TIME_RANGE",
311                )?
312                .map(|range| match range.as_slice() {
313                    [start, end] => Ok([
314                        MEpoch::from_mjd_seconds(*start, range_scale),
315                        MEpoch::from_mjd_seconds(*end, range_scale),
316                    ]),
317                    _ => Err(invalid(format!(
318                        "OBSERVATION TIME_RANGE holds {} values, not 2",
319                        range.len()
320                    ))),
321                })
322                .transpose()?,
323                release_date: observation
324                    .optional_f64(row, "RELEASE_DATE")
325                    .map_err(storage)?
326                    .map(|seconds| MEpoch::from_mjd_seconds(seconds, release_scale)),
327                flagged: observation
328                    .optional_bool(row, "FLAG_ROW")
329                    .map_err(storage)?,
330            })
331        })
332        .collect()
333}

335/// The scheduling and execution block UIDs a `SCHEDULE` cell states: those
336/// of its two lines, as importasdm writes them ([`scheduled_uid`]); none
337/// for a cell of another length.
338pub(crate) fn schedule_identifiers(lines: &[String]) -> (Option<String>, Option<String>) {
339    match lines {
340        [scheduling, execution] => (
341            scheduled_uid(scheduling, "SchedulingBlock"),
342            scheduled_uid(execution, "ExecBlock"),
343        ),
344        _ => (None, None),
345    }
346}

348/// The `SCHEDULE` cell importasdm writes for block UIDs `scheduling` and
349/// `execution`, which [`schedule_identifiers`] reads back: its two lines,
350/// or no line without both.
351pub(crate) fn schedule_lines(scheduling: Option<&str>, execution: Option<&str>) -> Vec<String> {
352    match (scheduling, execution) {
353        (Some(scheduling), Some(execution)) => vec![
354            format!("[SchedulingBlock {scheduling},"),
355            format!("ExecBlock {execution}]"),
356        ],
357        _ => Vec::new(),
358    }
359}

361/// The UID after `label` in a `SCHEDULE` line, as XRADIO's converter reads
362/// one `importasdm` writes (`[SchedulingBlock uid://A001/X3571/X122,
363/// ExecBlock uid://A002/X1003af4/X75a3]`, one entry per line): the word
364/// characters, `/` and `:` that follow `label` and a space.
365fn scheduled_uid(line: &str, label: &str) -> Option<String> {
366    let start = line.find(&format!("{label} "))? + label.len() + 1;
367    let uid: String = line[start..]
368        .chars()
369        .take_while(|c| c.is_alphanumeric() || matches!(c, '_' | '/' | ':'))
370        .collect();
371    (!uid.is_empty()).then_some(uid)
372}

384/// The intents an `OBS_MODE` string lists, sorted and deduplicated.
385pub(super) fn intents(mode: &str) -> Vec<String> {
386    let set: BTreeSet<String> = mode
387        .split(',')
388        .map(str::trim)
389        .filter(|intent| !intent.is_empty())
390        .map(str::to_string)
391        .collect();

395/// Every `ANTENNA` row.
396pub(super) fn read_antennas(ms: &MeasurementSet) -> Result<Vec<AntennaEntry>, ObservationError> {
397    let antenna = ms.antenna().map_err(storage)?;
398    let antenna_table = ms
399        .subtable(SubtableId::Antenna)
400        .ok_or_else(|| invalid("no ANTENNA subtable"))?;
401    require_units(antenna_table, "POSITION", "m")?;
402    require_units(antenna_table, "DISH_DIAMETER", "m")?;
403    match measure_reference(antenna_table, "POSITION")?.as_deref() {
404        None | Some("ITRF") => {}
405        Some(_) => {
406            return Err(ObservationError::Unsupported {
407                what: "MSv2 antenna positions in other than ITRF",
408            });
409        }
410    }
411    (0..antenna.row_count())
412        .map(|row| {
413            let [x, y, z] = antenna.position(row).map_err(storage)?;
414            Ok(AntennaEntry {
415                name: antenna.name(row).map_err(storage)?,
416                station: antenna.station(row).map_err(storage)?,
417                position: MPosition::new_itrf(x, y, z),
418                mount: antenna.mount(row).map_err(storage)?,
419                dish_diameter: metres(antenna.dish_diameter(row).map_err(storage)?),
420                flagged: crate::subtables::optional_bool(antenna_table, row, "FLAG_ROW")
421                    .map_err(storage)?,
422            })
423        })
424        .collect()
425}

427/// Every `FEED` row's receptors, by (antenna, feed, spectral window), the
428/// window -1 standing for every window, in table order.
429pub(crate) struct Feeds {
430    rows: BTreeMap<(i32, i32, i32), Vec<Receptors>>,
431    /// Some row names a window: casacore's `MSIter` then reads only each
432    /// window's own rows, and XRADIO's converter likewise.
433    window_dependent: bool,
434}
435
436/// The `FEED` rows.
437pub(crate) fn read_feeds(ms: &MeasurementSet) -> Result<Feeds, ObservationError> {
438    let feed = ms.feed().map_err(storage)?;
439    if let Some(feed_table) = ms.subtable(SubtableId::Feed) {
440        require_units(feed_table, "RECEPTOR_ANGLE", "rad")?;
441    }
442    let mut rows = BTreeMap::<(i32, i32, i32), Vec<Receptors>>::new();
443    for row in 0..feed.row_count() {
444        let key = (
445            feed.i32(row, "ANTENNA_ID").map_err(storage)?,
446            feed.i32(row, "FEED_ID").map_err(storage)?,
447            feed.i32(row, "SPECTRAL_WINDOW_ID").map_err(storage)?,
448        );
449        let types = match feed.array(row, "POLARIZATION_TYPE").map_err(storage)? {
450            ArrayValue::String(types) => types.iter().cloned().collect(),
451            _ => return Err(invalid("FEED POLARIZATION_TYPE is not a string array")),
452        };
453        let angles = match feed.array(row, "RECEPTOR_ANGLE").map_err(storage)? {
454            ArrayValue::Float64(angles) => angles.iter().map(|angle| radians(*angle)).collect(),
455            _ => return Err(invalid("FEED RECEPTOR_ANGLE is not a Float64 array")),
456        };
457        rows.entry(key)
458            .or_default()
459            .push(Receptors { types, angles });
460    }
461    let window_dependent = rows.keys().any(|(_, _, window)| *window != -1);
462    Ok(Feeds {
463        rows,
464        window_dependent,
465    })
466}
467
468impl Feeds {
469    /// The retained feed lookup and all rows of receptor values it owns.
470    pub(super) fn held_bytes(&self) -> usize {
471        casa_types::memory::btree_node_bytes::<(i32, i32, i32), Vec<Receptors>>(self.rows.len())
472            + self
473                .rows
474                .values()
475                .map(|rows| heap::entries(rows, heap::receptors_bytes))
476                .sum::<usize>()
477    }
478
479    /// Antenna `antenna`'s receptors through feed `feed` in spectral window
480    /// `window`, as casacore's `MSIter::setFeedInfo` and XRADIO's converter
481    /// select them: the window's own rows when any row names a window, else
482    /// the rows for every window (-1). A partition's receptors are one set
483    /// (MSv4 holds no time dependence), so where rows of one antenna, feed
484    /// and window differ, `FEED` stating receptors per time, the last row
485    /// in table order is taken, as casacore's `MSIter` takes it (with a
486    /// warning); XRADIO's converter refuses such rows, and a conversion
487    /// reports them as a loss ([`Self::varying`]). `None` without a row.
488    pub(super) fn receptors(&self, antenna: i32, feed: i32, window: i32) -> Option<&Receptors> {
489        let window = if self.window_dependent { window } else { -1 };
490        self.rows
491            .get(&(antenna, feed, window))
492            .and_then(|rows| rows.last())
493    }
494
495    /// The most heap a partition's copy of one antenna's receptors holds
496    /// ([`heap::receptors_bytes`]).
497    pub(super) fn largest_receptors_bytes(&self) -> usize {
498        self.rows
499            .values()
500            .flatten()
501            .map(heap::receptors_bytes)
502            .max()
503            .unwrap_or(0)
504    }
505
506    /// The first (antenna, feed, window) whose rows differ, among the rows
507    /// [`Self::receptors`] reads: `FEED` states receptors that vary in
508    /// time, of which it keeps the last. Rows for every window (-1) beside
509    /// window rows are never read, and do not count.
510    pub(crate) fn varying(&self) -> Option<(i32, i32, i32)> {
511        self.rows
512            .iter()
513            .filter(|((_, _, window), _)| !self.window_dependent || *window != -1)
514            .find(|(_, rows)| rows.iter().any(|row| row != &rows[0]))
515            .map(|(key, _)| *key)
516    }
517}

527/// The `SOURCE` name of source `id`, if the MeasurementSet has one.
528fn source_name(ms: &MeasurementSet, id: i32) -> Result<Option<String>, ObservationError> {
529    if id < 0 {
530        return Ok(None);
531    }
532    let Ok(source) = ms.source() else {
533        return Ok(None);
534    };
535    for row in 0..source.row_count() {
536        if source.i32(row, "SOURCE_ID").map_err(storage)? == id {
537            return source.string(row, "NAME").map(Some).map_err(storage);
538        }
539    }
540    Ok(None)
541}

543/// The scans: MAIN's distinct `SCAN_NUMBER`s in ascending order, named by
544/// their decimal number as XRADIO names them, with the `STATE::OBS_MODE`
545/// intents of their rows' states (`states_by_number`, each scan's
546/// `STATE_ID`s; `modes`, each `STATE` row's `OBS_MODE`); and the scan of
547/// each `SCAN_NUMBER`.
548pub(super) fn scans(
549    states_by_number: BTreeMap<i32, BTreeSet<i32>>,
550    modes: &[String],
551) -> (Vec<ScanEntry>, BTreeMap<i32, ScanId>) {
552    let index: BTreeMap<i32, ScanId> = states_by_number
553        .keys()
554        .enumerate()
555        .map(|(index, number)| (*number, ScanId(index as u32)))
556        .collect();
557    let scans = states_by_number
558        .into_iter()
559        .map(|(number, states)| {
560            let intents: BTreeSet<String> = states
561                .into_iter()
562                .filter_map(|state| {
563                    usize::try_from(state)
564                        .ok()
565                        .and_then(|state| modes.get(state))
566                })
567                .flat_map(|mode| intents(mode))
568                .collect();
569            ScanEntry {
570                name: number.to_string(),
571                number: Some(number),
572                intents: intents.into_iter().collect(),
573            }
574        })
575        .collect();
576    (scans, index)
577}

579/// The spectral window and correlation products of one data description.
580pub(super) struct SpectralSetup {
581    /// The `SPECTRAL_WINDOW` row.
582    pub(super) spectral_window_id: u32,
583    pub(super) spectral_window: String,
584    pub(super) frequencies_hz: Vec<f64>,
585    pub(super) frequency_frame: FrequencyRef,
586    pub(super) channel_widths_hz: Vec<f64>,
587    pub(super) correlations: Vec<Correlation>,
588}
589
590impl SpectralSetup {
591    /// The heap a copy of the setup holds.
592    pub(super) fn heap_bytes(&self) -> usize {
593        self.spectral_window.len()
594            + size_of_val(self.frequencies_hz.as_slice())
595            + size_of_val(self.channel_widths_hz.as_slice())
596            + size_of_val(self.correlations.as_slice())
597    }
598}
599
600/// The most [`spectral_setups`] holds: each `DATA_DESCRIPTION` row's
601/// copies of its window's name, channels and widths (`NUM_CHAN` of each)
602/// and its polarization's products (`NUM_CORR`), read from the rows the
603/// opening loads.
604pub(super) fn setups_bytes(ms: &MeasurementSet) -> Result<usize, ObservationError> {
605    let descriptions = ms.data_description().map_err(storage)?;
606    let windows = ms.spectral_window().map_err(storage)?;
607    let polarizations = ms.polarization().map_err(storage)?;
608    let mut bytes = 0;
609    for row in 0..descriptions.row_count() {
610        let window = descriptions.spectral_window_id(row).map_err(storage)?;
611        let polarization = descriptions.polarization_id(row).map_err(storage)?;
612        let channels = usize::try_from(window)
613            .ok()
614            .filter(|window| *window < windows.row_count())
615            .map(|window| {
616                Ok::<_, ObservationError>((
617                    windows.num_chan(window).map_err(storage)?,
618                    windows.name(window).map_err(storage)?.len(),
619                ))
620            })
621            .transpose()?
622            .map_or(0, |(channels, name)| {
623                2 * usize::try_from(channels).unwrap_or(0) * size_of::<f64>() + name
624            });
625        let products = usize::try_from(polarization)
626            .ok()
627            .filter(|polarization| *polarization < polarizations.row_count())
628            .map(|polarization| polarizations.num_corr(polarization).map_err(storage))
629            .transpose()?
630            .map_or(0, |products| {
631                usize::try_from(products).unwrap_or(0) * size_of::<Correlation>()
632            });
633        bytes += size_of::<SpectralSetup>() + channels + products;
634    }
635    Ok(bytes)
636}
637
638/// Every `DATA_DESCRIPTION` row's spectral setup.
639pub(super) fn spectral_setups(ms: &MeasurementSet) -> Result<Vec<SpectralSetup>, ObservationError> {
640    let descriptions = ms.data_description().map_err(storage)?;
641    let windows = ms.spectral_window().map_err(storage)?;
642    if let Some(window_table) = ms.subtable(SubtableId::SpectralWindow) {
643        require_units(window_table, "CHAN_FREQ", "Hz")?;
644        require_units(window_table, "CHAN_WIDTH", "Hz")?;
645    }
646    let polarizations = ms.polarization().map_err(storage)?;
647    (0..descriptions.row_count())
648        .map(|row| {
649            let window = usize::try_from(descriptions.spectral_window_id(row).map_err(storage)?)
650                .map_err(|_| invalid("negative SPECTRAL_WINDOW_ID"))?;
651            let polarization = usize::try_from(descriptions.polarization_id(row).map_err(storage)?)
652                .map_err(|_| invalid("negative POLARIZATION_ID"))?;
653            let frame_code = windows.meas_freq_ref(window).map_err(storage)?;
654            let frequency_frame = FrequencyRef::from_casacore_code(frame_code)
655                .ok_or_else(|| invalid(format!("unknown MEAS_FREQ_REF {frame_code}")))?;
656            let correlations = polarizations
657                .corr_type(polarization)
658                .map_err(storage)?
659                .into_iter()
660                .map(|code| {
661                    Correlation::from_casacore_code(code)
662                        .ok_or_else(|| invalid(format!("unknown CORR_TYPE {code}")))
663                })
664                .collect::<Result<_, _>>()?;
665            Ok(SpectralSetup {
666                spectral_window_id: window as u32,
667                spectral_window: windows.name(window).map_err(storage)?,
668                frequencies_hz: windows.chan_freq(window).map_err(storage)?,
669                frequency_frame,
670                channel_widths_hz: windows.chan_width(window).map_err(storage)?,
671                correlations,
672            })
673        })
674        .collect()
675}
