2//! How a resolved run starts its task binary.
3//!
4//! Every task binary runs on the shared task front end (#931), so every run
5//! is one invocation shape: the surface's fixed arguments, which name the
6//! surface a family binary hosts, then `--json-run -` with a
7//! [`TaskRunRequest`] of the values the caller set. The binary resolves them
8//! against the same contract, so defaults are not repeated.
9
10use casa_provider_contracts::{ParameterValue, ProviderInvocation, TaskRunRequest, ValueAdapter};
11use thiserror::Error;
12
13use crate::ParameterSession;
14
15/// The invocation of `session`'s task binary: the surface's fixed
16/// arguments, then `--json-run -` with a [`TaskRunRequest`] holding the
17/// values the caller set, from the base profile and the overrides, that are
18/// active.
19pub fn project_provider_invocation(
20    session: &ParameterSession,
21) -> Result<ProviderInvocation, ProviderInvocationError> {
22    let values = session
23        .states()
24        .iter()
25        .filter(|(_, state)| state.explicit && state.active)
26        .filter_map(|(name, state)| Some((name.clone(), state.value.as_ref()?.to_plain_json())))
27        .collect();
28    let request = TaskRunRequest {
29        surface: session.bundle().surface.id().to_string(),
30        values,
31    };
32    let stdin = serde_json::to_string(&request)
33        .map_err(|error| ProviderInvocationError::Request(error.to_string()))?;
34    let mut args = session.bundle().surface.execution().fixed_args.clone();
35    args.extend(["--json-run".to_string(), "-".to_string()]);
36    Ok(ProviderInvocation {
37        args,
38        stdin: Some(stdin),
39    })
40}
41
42/// Failure to project a resolved parameter session into a task process.
43#[derive(Debug, Error)]
44pub enum ProviderInvocationError {
45    /// The task request could not be written.
46    #[error("cannot write the task request: {0}")]
47    Request(String),
48}
49
50/// Whether `value` is the explicit absence its provider adapter omits:
51/// `"none"`, `"auto"` or CASA's empty string under `omit_none`, or an empty
52/// list under any.
53pub fn parameter_value_is_omitted(value: &ParameterValue, adapter: Option<&ValueAdapter>) -> bool {
54    matches!(value, ParameterValue::Array(values) if values.is_empty())
55        || (matches!(adapter, Some(ValueAdapter::OmitNone)) && is_explicit_absence(value))
56}
57
58fn is_explicit_absence(value: &ParameterValue) -> bool {
59    match value {
60        ParameterValue::String(state) => state.is_empty() || state == "none" || state == "auto",
61        ParameterValue::Array(values) => {
62            !values.is_empty() && values.iter().all(is_explicit_absence)
63        }
64        _ => false,
65    }
66}

76    /// The task request `session` sends its binary.
77    fn request(session: &ParameterSession) -> TaskRunRequest {
78        let invocation = project_provider_invocation(session).unwrap();
79        serde_json::from_str(invocation.stdin.as_deref().unwrap()).unwrap()
80    }

82    #[test]
83    fn a_run_is_one_json_run_of_the_values_set() {
84        let mut session = ParameterSession::defaults(builtin_surface_bundle("feather").unwrap())
85            .expect("feather defaults");
86        for (name, path) in [
87            ("imagename", "products/feathered.image"),
88            ("highres", "high.image"),
89            ("lowres", "low.image"),
90        ] {
91            session
92                .set(name, ParameterValue::String(path.into()))
93                .unwrap();
94        }
95        let invocation = project_provider_invocation(&session).unwrap();
96        assert_eq!(invocation.args, ["--json-run", "-"]);
97        let feather = request(&session);
98        assert_eq!(feather.surface, "feather");
99        // The values set, and no defaults such as sdfactor or overwrite.
100        assert_eq!(
101            feather.values.keys().collect::<Vec<_>>(),
102            ["highres", "imagename", "lowres"]
103        );
104
105        // A family binary's surface is named by its fixed arguments too.
106        let mut imstat =
107            ParameterSession::defaults(builtin_surface_bundle("imstat").unwrap()).unwrap();
108        imstat
109            .set("imagename", ParameterValue::String("in.image".into()))
110            .unwrap();
111        let invocation = project_provider_invocation(&imstat).unwrap();
112        assert_eq!(invocation.args, ["imstat", "--json-run", "-"]);
113        let mut gaincal =
114            ParameterSession::defaults(builtin_surface_bundle("gaincal").unwrap()).unwrap();
115        gaincal
116            .set("vis", ParameterValue::String("in.ms".into()))
117            .unwrap();
118        let invocation = project_provider_invocation(&gaincal).unwrap();
119        assert_eq!(invocation.args, ["gaincal", "--json-run", "-"]);
120        assert_eq!(request(&gaincal).surface, "gaincal");
121    }

141    #[test]
142    fn a_vlass_awproject_profile_is_one_complete_request() {
143        let bundle = builtin_surface_bundle("imager").unwrap();
144        let mut session = ParameterSession::defaults(bundle).unwrap();
145        let set = [
146            ("vis", ParameterValue::String("vlass.ms".into())),
147            ("imagename", ParameterValue::String("products/vlass".into())),
148            ("field", ParameterValue::String("1525".into())),
149            ("spw", ParameterValue::String("9".into())),
150            ("uvrange", ParameterValue::String("<12km".into())),
151            ("intent", ParameterValue::String("*TARGET*".into())),
152            ("stokes", ParameterValue::String("I".into())),
153            ("gridder", ParameterValue::String("awproject".into())),
154            ("cfcache", ParameterValue::String("/cache/vlass".into())),
155            ("deconvolver", ParameterValue::String("mtmfs".into())),
156            ("scales", ParameterValue::String("0,5,12".into())),
157            ("normtype", ParameterValue::String("flatnoise".into())),
158            ("parallel", ParameterValue::Bool(false)),
159            ("usepointing", ParameterValue::Bool(true)),
160            ("write_pb", ParameterValue::Bool(true)),
161            ("nterms", ParameterValue::Integer(2)),
162            ("wprojplanes", ParameterValue::Integer(32)),
163            ("cf_resident_mb", ParameterValue::Integer(256)),
164            ("pblimit", ParameterValue::Float(0.0001)),
165        ];
166        for (name, value) in &set {
167            session
168                .set(*name, value.clone())
169                .unwrap_or_else(|error| panic!("set VLASS profile parameter {name}: {error}"));
170        }
171        let request = request(&session);
172        assert_eq!(request.surface, "imager");
173        for (name, _) in &set {
174            assert_eq!(
175                request.values.get(*name),
176                session.states()[*name]
177                    .value
178                    .as_ref()
179                    .map(ParameterValue::to_plain_json)
180                    .as_ref(),
181                "{name}"
182            );
183        }
184        assert_eq!(request.values["vis"], serde_json::json!(["vlass.ms"]));
185    }

205    #[test]
206    fn auto_and_none_sentinels_are_omitted_but_real_values_are_kept() {
207        let omit_none = Some(&ValueAdapter::OmitNone);
208        for sentinel in ["auto", "none"] {
209            let value = ParameterValue::String(sentinel.into());
210            assert!(parameter_value_is_omitted(&value, omit_none), "{sentinel}");
211            assert!(!parameter_value_is_omitted(
212                &value,
213                Some(&ValueAdapter::Identity)
214            ));
215        }
216        assert!(!parameter_value_is_omitted(
217            &ParameterValue::String("spectral".into()),
218            omit_none
219        ));
220        let pair = ParameterValue::Array(vec![
221            ParameterValue::Integer(256),
222            ParameterValue::Integer(128),
223        ]);
224        assert!(!parameter_value_is_omitted(&pair, omit_none));
225    }
