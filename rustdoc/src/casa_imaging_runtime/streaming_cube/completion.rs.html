3//! Native-band transfer into the existing runtime-bound normal-state fold.

5use super::*;
6use casa_imaging_reconstruction::SpectralOperatorPrimitives;
7use casa_imaging_reconstruction::runtime_adapter::{CubeNormalRefresh, CubeResidual};
8use casa_imaging_reconstruction::{FinalNormalState, ModelGenerationId};
9
10pub(crate) struct PendingCubeRefresh {
11    evidence: CubeNormalRefresh,
12    binding: CompleteDataExecutionBinding,
13}
14
15impl PendingCubeRefresh {
16    #[allow(clippy::too_many_arguments)]
17    pub(crate) fn new(
18        context: WorkExecutionContext<'_>,
19        reconciliation_node: &WorkNodeId,
20        imported_node: &WorkNodeId,
21        specification: &SpectralOperatorSpecification,
22        previous: &FinalNormalState,
23        model: ModelGenerationId,
24        original: &WeightingReplayCompletion,
25        storage: &NormalStoragePlan,
26    ) -> Result<Self, CompleteDataOperatorError> {
27        if context.node().id != *reconciliation_node
28            || context.node().kind != WorkKind::Compute
29            || context.compiled().problem_id() != original.problem_id()
30            || !context
31                .node()
32                .dependencies
33                .contains(&WorkDependency::Fence(FenceId::new(
34                    imported_node.clone(),
35                    FenceKind::Io,
36                )))
37        {
38            return Err(CompleteDataOperatorError::ExecutionBinding);
39        }
40        let evidence = previous.begin_streaming_cube_refresh(
41            specification,
42            original.reconstruction_summary(),
43            original.selected_generation(),
44            original
45                .continuum_transform()
46                .map(|value| value.generation_id()),
47            model,
48            storage,
49        )?;
50        Ok(Self {
51            evidence,
52            binding: CompleteDataExecutionBinding {
53                problem: original.problem_id(),
54                attempt: context.attempt_id(),
55                replay_node: imported_node.clone(),
56                reconciliation_node: reconciliation_node.clone(),
57                lease_epoch: context.lease_epoch(),
58                observation_predecessor_required: false,
59            },
60        })
61    }
62
63    pub(crate) fn append(
64        &mut self,
65        residual: CubeResidual,
66    ) -> Result<(), CompleteDataOperatorError> {
67        self.evidence.append(residual)?;
68        Ok(())
69    }
70
71    pub(crate) fn complete(self) -> Result<CompleteDataOperatorResult, CompleteDataOperatorError> {
72        Ok(CompleteDataOperatorResult {
73            evidence: self.evidence.finish()?,
74            attempt: self.binding.attempt,
75            replay_node: self.binding.replay_node,
76            reconciliation_node: self.binding.reconciliation_node,
77            lease_epoch: self.binding.lease_epoch,
78            observation_predecessor_required: false,
79        })
80    }
81}

83impl CompleteDataSlabResult {
84    /// Adopt an initial band only after the selected-source I/O fence settled.
85    /// The reconciliation node comes from the composed phase plan, not the band.
86    pub(crate) fn from_streaming_cube(
87        context: WorkExecutionContext<'_>,
88        reconciliation_node: &WorkNodeId,
89        specification: &SpectralOperatorSpecification,
90        primitives: SpectralOperatorPrimitives,
91        replay: &WeightingReplayCompletion,
92    ) -> Result<Self, CompleteDataOperatorError> {
93        let predecessor = context
94            .predecessor_observation_completion(replay.owner_node())
95            .ok_or(CompleteDataOperatorError::ExecutionBinding)?;
96        if context.node().id != *reconciliation_node
97            || context.node().kind != WorkKind::Compute
98            || !context
99                .node()
100                .dependencies
101                .contains(&WorkDependency::Fence(FenceId::new(
102                    replay.owner_node().clone(),
103                    FenceKind::Io,
104                )))
105            || context.compiled().problem_id() != replay.problem_id()
106            || context.attempt_id() != replay.attempt_id()
107            || context.lease_epoch() != replay.lease_epoch()
108            || predecessor.attempt_id() != replay.attempt_id()
109            || predecessor.lease_epoch() != replay.lease_epoch()
110            || predecessor.owner_node() != replay.owner_node()
111            || !predecessor.settled_fences().contains(&FenceKind::Io)
112            || predecessor.owner_completion().generation_id() != replay.selected_generation()
113            || predecessor.owner_completion().sample_count() != replay.sample_count()
114        {
115            return Err(CompleteDataOperatorError::ExecutionBinding);
116        }
117        let evidence = CompleteDataOwnerResult::from_streaming_cube(
118            specification,
119            primitives,
120            replay.reconstruction_summary(),
121            replay.selected_generation(),
122            replay
123                .continuum_transform()
124                .map(|value| value.generation_id()),
125        )?;
126        if evidence.completion().problem_id() != replay.problem_id() {
127            return Err(CompleteDataOperatorError::ExecutionBinding);
128        }
129        Ok(Self {
130            evidence,
131            binding: CompleteDataExecutionBinding {
132                problem: replay.problem_id(),
133                attempt: replay.attempt_id(),
134                replay_node: replay.owner_node().clone(),
135                reconciliation_node: reconciliation_node.clone(),
136                lease_epoch: replay.lease_epoch(),
137                observation_predecessor_required: true,
138            },
139        })
140    }
