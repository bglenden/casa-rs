3//! The restoring beam a problem's restored products are published with.
4//!
5//! CASA's `tclean` restores each plane with the Gaussian fitted to its PSF
6//! (`restoringbeam=''`), with one beam enclosing every plane's fitted beam
7//! (`'common'`), or with a beam the user gives by its axes and angle
8//! (`restoringbeam=['10arcsec','8arcsec','45deg']`). In the last two the
9//! residual is first brought to the restoring beam's resolution
10//! (`SIImageStore::restore`, `rescaleResolution`).
11
12use thiserror::Error;
13
14/// Restoring-beam requirement for published products.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum RestoringBeamPolicy {
17    /// Do not create a restored product.
18    None,
19    /// Fit and use an independent beam for each plane.
20    PerPlane,
21    /// Use one common enclosing beam across all spectral planes.
22    Common,
23    /// Use the given beam for every plane, as CASA restores with a
24    /// `restoringbeam` given by its axes and angle.
25    Fixed(FixedRestoringBeam),
26}
27
28/// A restoring beam given by the request: its full widths at half maximum
29/// and position angle, in radians.
30///
31/// Equality compares the three values bit for bit, so the beam can be part
32/// of a policy that is compared exactly.
33#[derive(Debug, Clone, Copy)]
34pub struct FixedRestoringBeam {
35    major_fwhm_rad: f64,
36    minor_fwhm_rad: f64,
37    position_angle_rad: f64,
38}
39
40/// Why a restoring beam's axes and angle do not make a beam.
41#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
42#[error(
43    "a restoring beam needs finite, positive axes with the major axis at least the minor, \
44     and a finite position angle"
45)]
46pub struct InvalidRestoringBeam;
47
48impl FixedRestoringBeam {
49    /// The beam with these full widths at half maximum and position angle
50    /// (east from north), in radians.
51    ///
52    /// # Errors
53    ///
54    /// [`InvalidRestoringBeam`] when an axis is not finite and positive, the
55    /// minor axis exceeds the major, or the angle is not finite: casacore's
56    /// `GaussianBeam::setMajorMinor` refuses the same axes, and a zero axis
57    /// is its null beam, which restores with the fitted beams instead.
58    pub fn new(
59        major_fwhm_rad: f64,
60        minor_fwhm_rad: f64,
61        position_angle_rad: f64,
62    ) -> Result<Self, InvalidRestoringBeam> {
63        let positive = |axis: f64| axis.is_finite() && axis > 0.0;
64        if !(positive(major_fwhm_rad)
65            && positive(minor_fwhm_rad)
66            && major_fwhm_rad >= minor_fwhm_rad
67            && position_angle_rad.is_finite())
68        {
69            return Err(InvalidRestoringBeam);
70        }
71        Ok(Self {
72            major_fwhm_rad,
73            minor_fwhm_rad,
74            position_angle_rad,
75        })
76    }
77
78    /// Major-axis full width at half maximum in radians.
79    #[must_use]
80    pub const fn major_fwhm_rad(self) -> f64 {
81        self.major_fwhm_rad
82    }
83
84    /// Minor-axis full width at half maximum in radians.
85    #[must_use]
86    pub const fn minor_fwhm_rad(self) -> f64 {
87        self.minor_fwhm_rad
88    }
89
90    /// Position angle, east from north, in radians, as given.
91    #[must_use]
92    pub const fn position_angle_rad(self) -> f64 {
93        self.position_angle_rad
94    }
95}
96
97impl PartialEq for FixedRestoringBeam {
98    fn eq(&self, other: &Self) -> bool {
99        self.major_fwhm_rad.to_bits() == other.major_fwhm_rad.to_bits()
100            && self.minor_fwhm_rad.to_bits() == other.minor_fwhm_rad.to_bits()
101            && self.position_angle_rad.to_bits() == other.position_angle_rad.to_bits()
102    }
103}
104
105impl Eq for FixedRestoringBeam {}

107#[cfg(test)]
108mod tests {
109    use super::*;
110
111    #[test]
112    fn a_fixed_beam_takes_casas_axes_and_compares_exactly() {
113        let beam = FixedRestoringBeam::new(2.0e-5, 1.0e-5, 0.5).expect("a beam");
114        assert_eq!(
115            (
116                beam.major_fwhm_rad(),
117                beam.minor_fwhm_rad(),
118                beam.position_angle_rad()
119            ),
120            (2.0e-5, 1.0e-5, 0.5)
121        );
122        assert_eq!(
123            RestoringBeamPolicy::Fixed(beam),
124            RestoringBeamPolicy::Fixed(FixedRestoringBeam::new(2.0e-5, 1.0e-5, 0.5).unwrap())
125        );
126        assert_ne!(
127            RestoringBeamPolicy::Fixed(beam),
128            RestoringBeamPolicy::Fixed(FixedRestoringBeam::new(2.0e-5, 1.0e-5, 0.0).unwrap())
129        );
130        assert_ne!(
131            RestoringBeamPolicy::Fixed(beam),
132            RestoringBeamPolicy::Common
133        );
134        // A circular beam is a beam.
135        FixedRestoringBeam::new(1.0e-5, 1.0e-5, 0.0).expect("a circular beam");
136        for (major, minor, angle) in [
137            (1.0e-5, 2.0e-5, 0.0),
138            (0.0, 0.0, 0.0),
139            (-1.0e-5, -2.0e-5, 0.0),
140            (f64::INFINITY, 1.0e-5, 0.0),
141            (1.0e-5, f64::NAN, 0.0),
142            (2.0e-5, 1.0e-5, f64::NAN),
143        ] {
144            assert_eq!(
145                FixedRestoringBeam::new(major, minor, angle),
146                Err(InvalidRestoringBeam),
147                "{major} {minor} {angle}"
148            );
149        }
150    }
