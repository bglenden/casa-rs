7pub(super) fn load(
8    plan: &ApplyCalibrationTablePlan,
9    engine: Option<&Arc<MsCalEngine>>,
10    ms: &MeasurementSet,
11) -> Result<LoadedCalibrationTable, ApplyExecutionError> {
12    let family = TimeParameterFamily::from_subtype(&plan.summary.table_subtype);
13    // Interpolate the real FPARAM coefficients before constructing the Jones
14    // matrix. Antenna positions, opacity and gain-curve coefficients share
15    // the same real time operator as delays.
16    let mut table = load_parameters(plan, engine, family.is_some())?;
17    table.time_parameter_family = family;
18    table.callibrary = plan.spec.callibrary.clone();
19    if !plan.resolved_gainfields.is_empty() {
20        table.solutions.retain(|(field, _, _), _| {
21            plan.resolved_gainfields
22                .iter()
23                .any(|selected| selected.field_id == *field)
24        });
25    }
26    if let Some(library) = plan
27        .spec
28        .callibrary
29        .as_ref()
30        .filter(|library| !library.field_map.is_empty())
31    {
32        table
33            .solutions
34            .retain(|(field, _, _), _| library.field_map.contains(field));
35    }
36    // One flagged template per window handles missing antenna/context slices
37    // without copying a full parameter grid for each baseline.
38    for (&(_, window, _), rows) in &table.solutions {
39        if let std::collections::hash_map::Entry::Vacant(entry) = table.missing_grids.entry(window)
40            && let Some(row) = rows.first()
41        {
42            let grid = physical_grid(&table.path, family, Arc::clone(&row.grid))?;
43            entry.insert(flagged(&grid));
44        }
45    }
46    let nearest_field = plan.summary.table_subtype != "BPOLY"
47        && (matches!(plan.spec.gainfield, Some(crate::GainFieldSelector::Nearest))
48            || plan
49                .spec
50                .callibrary
51                .as_ref()
52                .is_some_and(|library| !library.field_map.is_empty()));
53    let sample = table
54        .solutions
55        .values()
56        .flatten()
57        .next()
58        .map(|row| row.grid.as_ref());
59    let curve = matches!(
60        sample,
61        Some(CalibrationGrid::Complex(_) | CalibrationGrid::Delay(_))
62    );
63    if curve {
64        let observation_count = ms
65            .observation()
66            .map_err(|source| ApplyExecutionError::MutateMeasurementSet {
67                path: display_ms_path(ms),
68                source,
69            })?
70            .row_count();
71        let mut options = plan.interp.time_options(observation_count);
72        // CLPatchPanel partitions through explicit maps, not these traditional
73        // interpolation-string modifiers. Library maps are handled separately.
74        if plan.spec.callibrary.is_some() {
75            options.per_observation = plan
76                .spec
77                .callibrary
78                .as_ref()
79                .is_some_and(|library| library.observation_map.active());
80            options.per_scan = plan
81                .spec
82                .callibrary
83                .as_ref()
84                .is_some_and(|library| library.scan_map.active());
85        }
86        // CASA's legacy BPOLY conversion emits unpartitioned modern headers.
87        if plan.summary.table_subtype == "BPOLY" {
88            options.per_scan = false;
89            options.per_observation = false;
90            options.phase_delay = false;
91        }
92        // CASA dispatches scalar PD by Jones family, even for a one-channel B table.
93        options.phase_delay &= matches!(plan.summary.table_subtype.as_str(), "G Jones" | "T Jones");
94        table.phase_delay = options.phase_delay;
95        let mut context = TimePlan::apply_context(&table, options)?;
96        if let Some(library) = &plan.spec.callibrary {
97            context.mapped_contexts(library.observation_map.active(), library.scan_map.active());
98        }
99        let mut groups = HashMap::<_, HashMap<_, _>>::new();
100        for (key, rows) in std::mem::take(&mut table.solutions) {
101            groups
102                .entry(if nearest_field { key.0 } else { -1 })
103                .or_default()
104                .insert(key, rows);
105        }
106        for (field, solutions) in groups {
107            let selected = LoadedCalibrationTable {
108                path: table.path.clone(),
109                interp: table.interp,
110                supports_calwt: false,
111                engine: None,
112                solutions,
113                callibrary: None,
114                time_plans: HashMap::new(),
115                missing_grids: HashMap::new(),
116                time_parameter_family: None,
117                phase_delay: false,
118            };
119            table.time_plans.insert(
120                field,
121                TimePlan::for_apply(&selected, options, family.is_some(), &context)?,
122            );
123        }
124    } else if !nearest_field {
125        table.solutions = crate::lookup::across_fields(&table)
126            .into_iter()
127            .map(|((window, antenna), series)| ((-1, window, antenna), series))
128            .collect();
129    }
130    Ok(table)
131}

133pub(super) fn lookup(
134    table: &LoadedCalibrationTable,
135    field: i32,
136    window: i32,
137    antenna: i32,
138    row: &ApplyRowPlan,
139    data_window: &crate::plan::SpectralWindowPlan,
140) -> Result<Option<Arc<CalibrationGrid>>, ApplyExecutionError> {
141    let antenna = table
142        .callibrary
143        .as_ref()
144        .and_then(|library| {
145            library
146                .antenna_map
147                .get(antenna as usize)
148                .or_else(|| library.antenna_map.last())
149        })
150        .copied()
151        .filter(|id| *id >= 0)
152        .unwrap_or(antenna);
153    let Some(plan) = table.time_plans.get(&field) else {
154        if !table.time_plans.is_empty() {
155            return Ok(table.missing_grids.get(&window).cloned());
156        }
157        return table
158            .lookup(field, window, antenna, row.time_seconds, table.interp)
159            .map(|grid| grid.or_else(|| table.missing_grids.get(&window).cloned()));
160    };
161    let frequencies = &data_window.channel_frequencies_hz;
162    let frequency_hz = table
163        .phase_delay
164        .then(|| frequencies.iter().sum::<f64>() / frequencies.len() as f64);
165    let query = TimeQuery {
166        observation: table
167            .callibrary
168            .as_ref()
169            .map_or(row.calibration_observation_id, |library| {
170                library.observation_map.map(row.calibration_observation_id)
171            }),
172        scan: table
173            .callibrary
174            .as_ref()
175            .map_or(row.scan_number, |library| {
176                library.scan_map.map(row.scan_number)
177            }),
178        window,
179        antenna,
180        time: row.time_seconds,
181        frequency_hz,
182    };
183    if let Some(grid) = plan.registered_grid(&query) {
184        return Ok(Some(grid));
185    }
186    let Some(solution) = plan.solution(&table.path, query)? else {
187        return Ok(table.missing_grids.get(&window).cloned());
188    };
189    let shape = IxDyn(&[solution.receptors, solution.channels]);
190    let values =
191        ArrayD::from_shape_vec(shape.clone(), solution.values).expect("time grid dimensions");
192    let flags = ArrayD::from_shape_vec(shape, solution.flags).expect("time flag dimensions");
193    let raw = Arc::new(CalibrationGrid::Complex(GainGrid {
194        receptor_count: solution.receptors,
195        channel_count: solution.channels,
196        values,
197        flags,
198    }));
199    physical_grid(&table.path, table.time_parameter_family, raw).map(Some)
200}

202/// FPARAM interpolation operates on coefficients, not on their geometry-dependent gains.
203fn physical_grid(
204    path: &Path,
205    family: Option<TimeParameterFamily>,
206    raw: Arc<CalibrationGrid>,
207) -> Result<Arc<CalibrationGrid>, ApplyExecutionError> {
208    let Some(family) = family else {
209        return Ok(raw);
210    };
211    let CalibrationGrid::Complex(grid) = raw.as_ref() else {
212        unreachable!("real parameter carrier");
213    };
214    let values = ArrayValue::Float32(grid.values.mapv(|value| value.re));
215    let flags = ArrayValue::Bool(grid.flags.clone());
216    Ok(Arc::new(match family {
217        TimeParameterFamily::Delay => {
218            CalibrationGrid::Delay(DelayGrid::from_arrays(path, &values, &flags)?)
219        }
220        TimeParameterFamily::Antpos => {
221            CalibrationGrid::Antpos(AntposGrid::from_arrays(path, &values, &flags)?)
222        }
223        TimeParameterFamily::GainCurve { power } => {
224            CalibrationGrid::GainCurve(GainCurveGrid::from_arrays(path, &values, &flags, power)?)
225        }
226        TimeParameterFamily::Opacity => {
227            CalibrationGrid::Opacity(OpacityGrid::from_arrays(path, &values, &flags)?)
228        }
229    }))
230}

232/// A missing slice has unity effect with flags, retaining its partner's gain.
233pub(super) fn flagged(grid: &CalibrationGrid) -> Arc<CalibrationGrid> {
234    let mut grid = grid.clone();
235    match &mut grid {
236        CalibrationGrid::Complex(grid) => grid.flags.fill(true),
237        CalibrationGrid::Delay(grid) => grid.flags.fill(true),
238        CalibrationGrid::Antpos(grid) => grid.flagged = true,
239        CalibrationGrid::GainCurve(grid) => grid.flags.fill(true),
240        CalibrationGrid::Opacity(grid) => grid.flagged = true,
241    }
242    Arc::new(grid)
243}
