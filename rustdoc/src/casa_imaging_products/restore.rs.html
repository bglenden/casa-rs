3//! Restoration and normalization algorithms for continuum image products.
4//!
5//! The restoring kernel is evaluated in physical radians: fitted beam widths
6//! and the image cell scale share one unit system, so a multi-pixel beam
7//! stays a multi-pixel kernel at any cell size.
8
9use ndarray::{Array2, Axis};
10use num_complex::Complex64;
11use rustfft::FftPlanner;
12
13use casa_imaging_model::ProductNormalization;
14
15use crate::beam::RestoringBeam;
16use crate::error::ProductsError;

18const FWHM_TO_SIGMA: f64 = 1.0 / 2.354_820_045_030_949_3;
19
20/// Normalize one unnormalized plane to its compiled product normalization.
21///
22/// `UnitResponse` and the currently supported scalar-response `FlatNoise`
23/// path divide by the scalar sensitivity (sum weight). `UnitResponse` does
24/// not apply direction-dependent sensitivity division. When no usable scalar
25/// sensitivity exists every pixel blanks to NaN. `FlatSky` is not produced by
26/// this catalog version.
27///
28/// # Errors
29///
30/// Fails when the normalization is not representable by this catalog.
31pub fn normalize_plane(
32    values: &[f32],
33    normalization: ProductNormalization,
34    sensitivity: f64,
35) -> Result<Vec<f32>, ProductsError> {
36    match normalization {
37        ProductNormalization::UnitResponse | ProductNormalization::FlatNoise => {
38            if !(sensitivity.is_finite() && sensitivity > 0.0) {
39                return Ok(vec![f32::NAN; values.len()]);
40            }
41            Ok(values
42                .iter()
43                .map(|value| value / sensitivity as f32)
44                .collect())
45        }
46        ProductNormalization::FlatSky => Err(ProductsError::UnsupportedProductRole {
47            role: casa_imaging_model::ProductRole::Sensitivity,
48            catalog: crate::CONTINUUM_ALGORITHM_CATALOG_VERSION,
49        }),
50    }
51}

53/// Build the unit-peak elliptical-Gaussian restoring kernel image.
54///
55/// `cell_size_rad` is the image cell scale in radians per pixel on each
56/// axis: the fitted beam widths are radians, so pixel offsets must be
57/// converted to radians before they are divided by the Gaussian sigmas.
58#[must_use]
59pub fn gaussian_beam_image(
60    shape: [usize; 2],
61    beam: &RestoringBeam,
62    cell_size_rad: [f64; 2],
63) -> Array2<f32> {
64    let [width, height] = shape;
65    let sigma_major = (beam.major_fwhm_rad() * FWHM_TO_SIGMA).abs();
66    let sigma_minor = (beam.minor_fwhm_rad() * FWHM_TO_SIGMA).abs();
67    let cos_pa = beam.position_angle_rad().cos();
68    let sin_pa = beam.position_angle_rad().sin();
69    let centre_x = width as f64 / 2.0;
70    let centre_y = height as f64 / 2.0;
71    let mut kernel = Array2::<f32>::zeros((width, height));
72    for x in 0..width {
73        for y in 0..height {
74            let dx = (x as f64 - centre_x) * cell_size_rad[0];
75            let dy = (y as f64 - centre_y) * cell_size_rad[1];
76            let u = dx * cos_pa + dy * sin_pa;
77            let v = -dx * sin_pa + dy * cos_pa;
78            kernel[(x, y)] =
79                (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2))).exp() as f32;
80        }
81    }
82    let peak = kernel.iter().copied().fold(f32::NEG_INFINITY, f32::max);
83    if peak > 0.0 {
84        kernel.mapv_inplace(|value| value / peak);
85    }
86    kernel
87}

89/// Result of rescaling one normalized residual plane to a selected beam.
90///
91/// The diagnostics are the first-divergence seam used by the focused CASA
92/// comparator: fitted and common beams are compared before any final restored
93/// image can hide whether selection or smoothing first diverged.
94#[derive(Debug, Clone, PartialEq)]
95pub struct ResidualBeamScaling {
96    values: Vec<f32>,
97    smoothing_beam: Option<RestoringBeam>,
98    area_ratio: f64,
99    applied: bool,
100}
101
102impl ResidualBeamScaling {
103    /// Borrow the scaled residual values.
104    #[must_use]
105    pub fn values(&self) -> &[f32] {
106        &self.values
107    }
108
109    /// Return the deconvolved smoothing beam, if the beams differed.
110    #[must_use]
111    pub const fn smoothing_beam(&self) -> Option<RestoringBeam> {
112        self.smoothing_beam
113    }
114
115    /// Return the target/source beam-area ratio.
116    #[must_use]
117    pub const fn area_ratio(&self) -> f64 {
118        self.area_ratio
119    }
120
121    /// Return whether CASA's pixel-width gate applied smoothing and scaling.
122    #[must_use]
123    pub const fn applied(&self) -> bool {
124        self.applied
125    }
126
127    pub(crate) fn into_values(self) -> Vec<f32> {
128        self.values
129    }
130}

132/// Rescale a normalized residual plane from its fitted beam to a selected beam.
133///
134/// This follows CASA `SIImageStore::rescaleResolution`: deconvolve the fitted
135/// beam from the selected beam, skip effectively identical or sub-pixel
136/// smoothing, otherwise convolve with unit volume and multiply by the
137/// selected/fitted beam-area ratio.
138///
139/// # Errors
140///
141/// Returns [`ProductsError::BeamFitFailed`] when beam deconvolution fails.
142pub fn rescale_residual_to_beam(
143    residual: &[f32],
144    shape: [usize; 2],
145    cell_size_rad: [f64; 2],
146    fitted: RestoringBeam,
147    selected: RestoringBeam,
148) -> Result<ResidualBeamScaling, ProductsError> {
149    let area_ratio = selected.area_sr() / fitted.area_sr();
150    let smoothing_beam = selected
151        .deconvolving_beam(fitted)
152        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))?;
153    let Some(smoothing_beam) = smoothing_beam else {
154        return Ok(ResidualBeamScaling {
155            values: residual.to_vec(),
156            smoothing_beam: None,
157            area_ratio,
158            applied: false,
159        });
160    };
161    if smoothing_beam.minor_fwhm_rad() <= cell_size_rad[0].hypot(cell_size_rad[1]) {
162        return Ok(ResidualBeamScaling {
163            values: residual.to_vec(),
164            smoothing_beam: Some(smoothing_beam),
165            area_ratio,
166            applied: false,
167        });
168    }
169
170    let mut kernel = gaussian_beam_image(shape, &smoothing_beam, cell_size_rad);
171    let volume = f64::from(kernel.sum());
172    if !(volume.is_finite() && volume > 0.0) {
173        return Err(ProductsError::GeneratedNonfinite);
174    }
175    kernel.mapv_inplace(|value| (f64::from(value) / volume) as f32);
176    let mut values = fft_convolve(
177        residual,
178        kernel.as_slice().expect("Gaussian kernel is contiguous"),
179        shape,
180    );
181    for value in &mut values {
182        *value *= area_ratio as f32;
183    }
184    Ok(ResidualBeamScaling {
185        values,
186        smoothing_beam: Some(smoothing_beam),
187        area_ratio,
188        applied: true,
189    })
190}

192/// Convolve one real plane with an equal-shape kernel through even-shifted
193/// FFTs, matching the reconstruction owner's transform conventions.
194#[must_use]
195pub fn fft_convolve(plane: &[f32], kernel: &[f32], shape: [usize; 2]) -> Vec<f32> {
196    let cells = shape[0] * shape[1];
197    let mut planner = FftPlanner::<f64>::new();
198    let mut signal = Array2::<Complex64>::from_shape_vec(
199        (shape[0], shape[1]),
200        plane
201            .iter()
202            .map(|v| Complex64::new(f64::from(*v), 0.0))
203            .collect(),
204    )
205    .expect("shape matches payload");
206    let mut response = Array2::<Complex64>::from_shape_vec(
207        (shape[0], shape[1]),
208        kernel
209            .iter()
210            .map(|v| Complex64::new(f64::from(*v), 0.0))
211            .collect(),
212    )
213    .expect("shape matches kernel");
214
215    for data in [&mut signal, &mut response] {
216        shift_even(data);
217        for axis in 0..2 {
218            let plan = planner.plan_fft_forward(data.len_of(Axis(axis)));
219            let scratch_len = plan.get_inplace_scratch_len();
220            let length = data.len_of(Axis(axis));
221            let mut lane = vec![Complex64::default(); length];
222            let mut scratch = vec![Complex64::default(); scratch_len];
223            for mut view in data.lanes_mut(Axis(axis)) {
224                lane.iter_mut()
225                    .zip(view.iter())
226                    .for_each(|(target, source)| {
227                        *target = *source;
228                    });
229                plan.process_with_scratch(&mut lane, &mut scratch);
230                lane.iter()
231                    .zip(view.iter_mut())
232                    .for_each(|(source, target)| *target = *source);
233            }
234        }
235        shift_even(data);
236    }
237    for (signal, response) in signal.iter_mut().zip(response.iter()) {
238        *signal *= *response;
239    }
240    shift_even(&mut signal);
241    for axis in 0..2 {
242        let plan = planner.plan_fft_inverse(signal.len_of(Axis(axis)));
243        let scratch_len = plan.get_inplace_scratch_len();
244        let length = signal.len_of(Axis(axis));
245        let mut lane = vec![Complex64::default(); length];
246        let mut scratch = vec![Complex64::default(); scratch_len];
247        for mut view in signal.lanes_mut(Axis(axis)) {
248            lane.iter_mut()
249                .zip(view.iter())
250                .for_each(|(target, source)| *target = *source);
251            plan.process_with_scratch(&mut lane, &mut scratch);
252            lane.iter()
253                .zip(view.iter_mut())
254                .for_each(|(source, target)| *target = *source);
255        }
256    }
257    shift_even(&mut signal);
258
259    let scale = 1.0 / cells as f64;
260    signal
261        .iter()
262        .map(|value| (value.re * scale) as f32)
263        .collect()
264}
265
266fn shift_even(data: &mut Array2<Complex64>) {
267    let [width, height] = [data.shape()[0], data.shape()[1]];
268    debug_assert_eq!(width % 2, 0);
269    debug_assert_eq!(height % 2, 0);
270    for x in 0..width / 2 {
271        for y in 0..height / 2 {
272            data.swap((x, y), (x + width / 2, y + height / 2));
273            data.swap((x + width / 2, y), (x, y + height / 2));
274        }
275    }
276}

278#[cfg(test)]
279mod tests {
280    use super::*;
281
282    #[test]
283    fn identical_beams_leave_the_residual_bit_exact() {
284        let beam = RestoringBeam::new(4.0e-6, 3.0e-6, 0.2).expect("beam");
285        let residual = vec![0.0, 1.0, -2.0, 3.0];
286        let scaling = rescale_residual_to_beam(&residual, [2, 2], [1.0e-6; 2], beam, beam).unwrap();
287        assert!(!scaling.applied());
288        assert_eq!(scaling.smoothing_beam(), None);
289        assert_eq!(scaling.area_ratio(), 1.0);
290        assert_eq!(scaling.values(), residual);
291    }
292
293    #[test]
294    fn common_beam_scaling_uses_unit_volume_and_the_beam_area_ratio() {
295        let source = RestoringBeam::new(4.0e-6, 3.0e-6, 0.0).expect("source beam");
296        let target = RestoringBeam::new(8.0e-6, 6.0e-6, 0.0).expect("target beam");
297        let shape = [32, 32];
298        let mut residual = vec![0.0_f32; shape[0] * shape[1]];
299        residual[(shape[0] / 2) * shape[1] + shape[1] / 2] = 1.0;
300
301        let scaling =
302            rescale_residual_to_beam(&residual, shape, [1.0e-6; 2], source, target).unwrap();
303        assert!(scaling.applied());
304        assert!(scaling.smoothing_beam().is_some());
305        assert!((scaling.area_ratio() - 4.0).abs() < 1.0e-12);
306        let sum = scaling
307            .values()
308            .iter()
309            .map(|value| f64::from(*value))
310            .sum::<f64>();
311        assert!((sum - scaling.area_ratio()).abs() < 1.0e-5, "{sum}");
312    }
313
314    #[test]
315    fn smaller_selected_beam_fails_instead_of_silently_skipping_scaling() {
316        let fitted = RestoringBeam::new(8.0e-6, 6.0e-6, 0.0).expect("fitted beam");
317        let selected = RestoringBeam::new(4.0e-6, 3.0e-6, 0.0).expect("selected beam");
318        let error = rescale_residual_to_beam(&[1.0; 16], [4, 4], [1.0e-6; 2], fitted, selected)
319            .expect_err("smaller target must fail");
320        assert!(matches!(error, ProductsError::BeamFitFailed(_)));
321    }
