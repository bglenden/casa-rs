2//! Place spectra through the observation's POINTING window, in the image frame.
3use super::{TotalPowerError, failure};
4use casa_coordinates::{Coordinate, DirectionCoordinate};
5use casa_core::measures::{direction::MDirection, frame::MeasFrame};
6use casa_imaging_singledish::Pointing;
7use casa_observation::{Observation, PointingDirection, PointingRequest, VisibilityBatch};
8
9type ProjectedPointings = (Vec<Option<[f64; 2]>>, Vec<Pointing>);
10
11pub(super) fn pixels(
12    observation: &dyn Observation,
13    batch: &VisibilityBatch<'_>,
14    coordinate: &DirectionCoordinate,
15    measures: &std::sync::Arc<dyn casa_core::measures::MeasuresProvider>,
16    selected_antennas: &[casa_observation::AntennaId],
17) -> Result<ProjectedPointings, TotalPowerError> {
18    let times = batch.time.offsets_s();
19    let antennas = batch
20        .baselines
21        .iter()
22        .map(|baseline| baseline.antenna1)
23        .collect::<std::collections::BTreeSet<_>>()
24        .into_iter()
25        .collect();
26    let pointing = observation
27        .pointing(&PointingRequest {
28            partition: batch.partition,
29            direction: PointingDirection::Beam,
30            antennas,
31            times: times[0]..=*times.last().ok_or_else(|| failure("empty batch"))?,
32            budget_bytes: 8 << 20,
33        })
34        .map_err(failure)?;
35    let mut pixels = Vec::with_capacity(batch.samples());
36    let mut sampling = Vec::new();
37    for row in 0..batch.samples() {
38        let sample = batch.sample(row);
39        if !sample.baseline.is_auto()
40            || (!selected_antennas.is_empty()
41                && !selected_antennas.contains(&sample.baseline.antenna1))
42        {
43            pixels.push(None);
44            continue;
45        }
46        let field = &batch.catalogue.fields[sample.field.0 as usize];
47        if field.ephemeris.is_some() {
48            return Err(TotalPowerError::Unsupported {
49                behavior: "moving-source total-power imaging",
50                issue: 938,
51            });
52        }
53        let direction = if !pointing.store_has_pointing {
54            Some(field.phase_centre.clone())
55        } else {
56            let series = pointing.antennas.get(&sample.baseline.antenna1);
57            let sample_interval = batch.interval_s(row).unwrap_or(0.);
58            if let Some(series) = series {
59                selected_direction(
60                    series,
61                    sample.time_offset_s,
62                    sample_interval,
63                    pointing.frame,
64                )?
65            } else {
66                None
67            }
68        };
69        let Some(direction) = direction else {
70            pixels.push(None);
71            continue;
72        };
73        let frame = MeasFrame::new()
74            .with_measures(measures.clone())
75            .with_epoch(batch.time.get(row / batch.samples_per_time()))
76            .with_position(
77                batch.catalogue.antennas[sample.baseline.antenna1.0 as usize]
78                    .position
79                    .clone(),
80            );
81        let direction = direction
82            .convert_to(coordinate.direction_ref(), &frame)
83            .map_err(failure)?;
84        let (longitude, latitude) = direction.as_angles();
85        let pixel = coordinate
86            .to_pixel(&[longitude, latitude])
87            .map_err(failure)?;
88        pixels.push(Some([pixel[0], pixel[1]]));
89        sampling.push(Pointing {
90            time_s: sample.time_offset_s,
91            direction_rad: [longitude, latitude],
92        });
93    }
94    Ok((pixels, sampling))
95}

97/// CASA's two covering tolerances; interval-less MSv4 pointings use the
98/// nearest sample within half a MAIN integration. No interpolation is guessed.
99fn selected_direction(
100    series: &casa_observation::AntennaPointing,
101    time: f64,
102    interval: f64,
103    frame: casa_core::measures::direction::DirectionRef,
104) -> Result<Option<MDirection>, TotalPowerError> {
105    let epsilon = time.abs() * f64::EPSILON;
106    let selected = if let Some(intervals) = series.intervals_s() {
107        [epsilon, interval / 2.].into_iter().find_map(|tolerance| {
108            series
109                .times_s()
110                .iter()
111                .enumerate()
112                .find(|(i, knot)| {
113                    intervals[*i] > 0. && (**knot - time).abs() <= intervals[*i] / 2. + tolerance
114                })
115                .map(|(i, _)| i)
116        })
117    } else {
118        series
119            .times_s()
120            .iter()
121            .enumerate()
122            .min_by(|(_, a), (_, b)| (*a - time).abs().total_cmp(&(*b - time).abs()))
123            .filter(|(_, knot)| (**knot - time).abs() <= interval / 2. + epsilon)
124            .map(|(i, _)| i)
125    };
126    let Some(index) = selected else {
127        return Ok(None);
128    };
129    if series.intervals_s().is_some_and(|v| interval < v[index])
130        && (series.times_s()[index] - time).abs() > epsilon
131    {
132        return Err(TotalPowerError::Unsupported {
133            behavior: "POINTING time interpolation",
134            issue: 938,
135        });
136    }
137    let direction = series.direction_at(index, time);
138    if direction.iter().any(|v| !v.is_finite()) {
139        return Err(failure("nonfinite pointing direction"));
140    }
141    Ok(Some(MDirection::from_angles(
142        direction[0],
143        direction[1],
144        frame,
145    )))
146}
