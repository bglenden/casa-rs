35pub(crate) fn solve(
36    ms: &MeasurementSet,
37    request: &GainSolveRequest,
38    controls: GainSplineControls,
39    mode: CalibrationOutputMode,
40    tool: Option<ToolOptions<'_>>,
41) -> Result<GainSolveReport, GainSolveError> {
42    let tool_mode = tool.map(|settings| settings.mode);
43    let all_fields = tool.is_some_and(|settings| settings.all_fields);
44    let input = crate::timing::Phase::start("gspline_select_average");
45    let context = prepare_calibration_solve(
46        ms,
47        &request.selection,
48        &request.refant,
49        &request.prior_calibration_tables,
50        request.parang,
51        true,
52        PriorCalibrationMode::GainSpline,
53    )?;
54    let preaverage = if request.preaverage_seconds > 0. && request.preaverage_seconds.is_finite() {
55        request.preaverage_seconds
56    } else {
57        0.1
58    };
59    let selection = SolveGroupOptions {
60        input_rows: Default::default(),
61        gain_type: GainType::GSpline,
62        model_source: request.model_source,
63        point_stokes: if request.model_source == GainSolveModelSource::PointSource {
64            request.smodel
65        } else {
66            [1., 0., 0., 0.]
67        },
68        rotate_model: false,
69        solve_interval: GainSolveInterval::Infinite,
70        // GSPLINE's old iterator sorts by array, field and data description.
71        combine: GainSolveCombine {
72            scans: true,
73            observations: true,
74            ..GainSolveCombine::default()
75        },
76        preaverage_max_span: Some(preaverage),
77        correlation_dependent_flags: false,
78        robust_seed: false,
79        legacy_g_receptors: false,
80        channelized: false,
81        polynomial_bandpass: false,
82    };
83    let groups = build_solve_groups(
84        ms,
85        &context.rows,
86        context.preapplied_rows.as_ref(),
87        &BTreeMap::new(),
88        selection,
89    )?;
90    if groups.is_empty() {
91        return Err(GainSolveError::EmptySelection);
92    }
93    let fields = if all_fields {
94        let count = ms
95            .field()
96            .map_err(|source| GainSolveError::OpenMeasurementSet {
97                path: ms
98                    .path()
99                    .map_or_else(|| "<in-memory>".into(), |path| path.display().to_string()),
100                source,
101            })?
102            .row_count();
103        (0..count as i32).collect()
104    } else {
105        context
106            .rows
107            .iter()
108            .map(|row| row.field_id)
109            .collect::<BTreeSet<_>>()
110    };
111    let windows = context
112        .rows
113        .iter()
114        .map(|row| row.data_spw_id)
115        .collect::<BTreeSet<_>>();
116    let descriptions = context
117        .rows
118        .iter()
119        .map(|row| row.data_desc_id)
120        .collect::<BTreeSet<_>>();
121    let correlations = correlation_counts(ms, descriptions)?;
122    let hands = correlations
123        .values()
124        .copied()
125        .max()
126        .unwrap_or(1)
127        .clamp(1, 2);
128    let (times, series) = time_series(groups.values(), &correlations, hands);
129    drop(input);
130    let fit_phase = crate::timing::Phase::start("gspline_fit");
131    let fit_amplitude = tool_mode.map_or(request.solve_mode != GainSolveMode::Phase, |mode| {
132        mode.contains('A')
133    });
134    let fit_phase_component = tool_mode
135        .map_or(request.solve_mode != GainSolveMode::Amplitude, |mode| {
136            mode.contains('P')
137        });
138    let poly_mode = match tool_mode.unwrap_or(match request.solve_mode {
139        GainSolveMode::Amplitude => "A",
140        GainSolveMode::Phase => "P",
141        GainSolveMode::AmplitudePhase => "AP",
142    }) {
143        "A" => "AMP",
144        "P" => "PHAS",
145        "AP" => "PHASAMP",
146        mode => mode,
147    };
148    let coefficients = fit(
149        &times,
150        &series,
151        context.available_antennas.len(),
152        &SplineKernelOptions {
153            amplitude: fit_amplitude,
154            phase: fit_phase_component,
155            hands,
156            spacing_seconds: controls.splinetime,
157            phase_points: controls.npointaver.max(2) as usize,
158            phase_wrap_radians: controls.phasewrap.to_radians(),
159            reference_antenna: context.refant_id.max(0) as usize,
160        },
161    );
162    drop(fit_phase);
163    let output = crate::timing::Phase::start("gspline_write");
164    let solution_row_count = if let Some(coefficients) = &coefficients {
165        let frequency = reference_frequency(ms, &context.rows, context.preapplied_rows.as_ref())?;
166        let solved_antennas = series
167            .iter()
168            .filter(|baseline| baseline.samples.iter().flatten().copied().any(usable))
169            .flat_map(|baseline| baseline.antennas)
170            .collect::<BTreeSet<_>>();
171        let count = writer::write(
172            ms,
173            request,
174            coefficients,
175            (mode, all_fields),
176            &fields,
177            (context.refant_id, &solved_antennas, frequency),
178            poly_mode,
179        )?;
180        diagnostics::write_logs(&request.output_table, &times, &series, coefficients, hands);
181        count
182    } else {
183        0
184    };
185    drop(output);
186    Ok(GainSolveReport {
187        output_table: request.output_table.clone(),
188        gain_type: GainType::GSpline,
189        refant_antenna_id: context.refant_id,
190        field_ids: fields.into_iter().collect(),
191        spectral_window_ids: windows.into_iter().collect(),
192        solution_row_count,
193        selected_data: context.selected_data,
194        solve_stats: Default::default(),
195    })
196}

198fn reference_frequency(
199    ms: &MeasurementSet,
200    rows: &[SelectedSolveRow],
201    preapplied: Option<&PreappliedRows>,
202) -> Result<f64, GainSolveError> {
203    let ms_error = |source| GainSolveError::OpenMeasurementSet {
204        path: ms.path().map_or_else(
205            || "<in-memory>".to_owned(),
206            |path| path.display().to_string(),
207        ),
208        source,
209    };
210    let spectral = ms.spectral_window().map_err(ms_error)?;
211    // The old accumulator takes coordinates from the first usable averaged
212    // buffer within each array, field and data-description chunk.
213    let mut chunks = BTreeMap::<_, Vec<&SelectedSolveRow>>::new();
214    for row in rows {
215        chunks
216            .entry((row.array_id, row.field_id, row.data_desc_id))
217            .or_default()
218            .push(row);
219    }
220    let mut sum = 0.;
221    let mut usable_chunks = 0;
222    for chunk in chunks.values_mut() {
223        let first = chunk[0];
224        let frequencies = spectral
225            .chan_freq(first.data_spw_id as usize)
226            .map_err(ms_error)?;
227        let channels = selected_channels(first.channel_selection.as_ref(), frequencies.len())?;
228        if channels.len() == 1 {
229            sum += frequencies[channels[0]];
230            usable_chunks += 1;
231            continue;
232        }
233        if !chunk.is_sorted_by(|left, right| left.time_seconds <= right.time_seconds) {
234            chunk.sort_by(|left, right| left.time_seconds.total_cmp(&right.time_seconds));
235        }
236        for buffer in chunk.chunk_by(|left, right| left.time_seconds == right.time_seconds) {
237            let mut buffer_sum = 0.;
238            let mut count = 0;
239            for row in buffer.iter().filter(|row| !row.flag_row) {
240                let flags = if let Some(applied) =
241                    preapplied.and_then(|rows| rows.data.get(&row.row_index))
242                {
243                    &applied.flags
244                } else {
245                    ms.main_table()
246                        .cell_accessor(row.row_index, "FLAG")
247                        .and_then(|cell| cell.array())
248                        .map_err(|error| ms_error(MsError::from(error)))?
249                };
250                let ArrayValue::Bool(flags) = flags else {
251                    return Err(ms_error(MsError::InvalidInput(
252                        "FLAG is not a boolean array".into(),
253                    )));
254                };
255                for &channel in &channels {
256                    if !flags
257                        .index_axis(ndarray::Axis(1), channel)
258                        .iter()
259                        .any(|flagged| *flagged)
260                    {
261                        buffer_sum += frequencies[channel];
262                        count += 1;
263                    }
264                }
265            }
266            if count > 0 {
267                sum += buffer_sum / count as f64;
268                usable_chunks += 1;
269                break;
270            }
271        }
272    }
273    Ok(sum / usable_chunks.max(1) as f64)
274}

276fn correlation_counts(
277    ms: &MeasurementSet,
278    ids: BTreeSet<i32>,
279) -> Result<BTreeMap<i32, usize>, GainSolveError> {
280    let read = || {
281        let descriptions = ms.data_description()?;
282        let polarization = ms.polarization()?;
283        ids.into_iter()
284            .map(|id| {
285                let row = descriptions.polarization_id(id as usize)?;
286                let count = polarization.num_corr(row as usize)?;
287                Ok((id, count as usize))
288            })
289            .collect::<casa_ms::MsResult<BTreeMap<_, _>>>()
290    };
291    read().map_err(|source: MsError| GainSolveError::OpenMeasurementSet {
292        path: ms.path().map_or_else(
293            || "<in-memory>".to_owned(),
294            |path| path.display().to_string(),
295        ),
296        source,
297    })
298}

300fn time_series<'a>(
301    groups: impl Iterator<Item = &'a SolveAccumulator>,
302    correlations: &BTreeMap<i32, usize>,
303    hands: usize,
304) -> (Vec<f64>, Vec<BaselineSeries>) {
305    let mut timestamps = BTreeMap::<i64, (usize, f64)>::new();
306    let mut baselines = BTreeMap::<[usize; 2], BTreeMap<i64, [WeightedVisibility; 2]>>::new();
307    for (chunk, group) in groups.enumerate() {
308        for spectrum in group
309            .delay_spectra()
310            .filter(|spectrum| spectrum.parallel_hand && spectrum.antennas.0 != spectrum.antennas.1)
311        {
312            let Some((sum, weight)) = spectrum.channels().next() else {
313                continue;
314            };
315            if weight <= 0. {
316                continue;
317            }
318            let timestamp = (spectrum.time_seconds * 10.).round() as i64;
319            let first = timestamps
320                .entry(timestamp)
321                .or_insert((chunk, spectrum.time_seconds));
322            // CASA retains the first chronological sample of the first chunk
323            // contributing to a tenth-second bin, even when later samples
324            // happen to round to the same bin.
325            if first.0 == chunk {
326                first.1 = first.1.min(spectrum.time_seconds);
327            }
328            let [first, second] = [spectrum.antennas.0, spectrum.antennas.1].map(|id| id as usize);
329            let baseline = baselines.entry([first, second]).or_default();
330            let samples = baseline.entry(timestamp).or_default();
331            let update = |sample: &mut WeightedVisibility| {
332                sample.sum += Complex64::new(f64::from(sum.re), f64::from(sum.im));
333                sample.weight += f64::from(weight);
334            };
335            if correlations.get(&spectrum.buffer.0) == Some(&1) {
336                for sample in &mut samples[..hands] {
337                    update(sample);
338                }
339            } else {
340                update(&mut samples[spectrum.receptor]);
341            }
342        }
343    }
344    let times = timestamps
345        .values_mut()
346        .enumerate()
347        .map(|(index, (slot, time))| {
348            *slot = index;
349            *time
350        })
351        .collect::<Vec<_>>();
352    let series = baselines
353        .into_iter()
354        .map(|(antennas, by_time)| {
355            let mut samples = vec![[WeightedVisibility::default(); 2]; times.len()];
356            for (time, value) in by_time {
357                samples[timestamps[&time].0] = value;
358            }
359            BaselineSeries { antennas, samples }
360        })
361        .collect();
362    (times, series)
363}

366pub(super) struct WeightedVisibility {
367    pub(super) sum: Complex64,
368    pub(super) weight: f64,
369}
370
371impl WeightedVisibility {
372    fn value(self, phase: bool) -> f64 {
373        if phase {
374            self.sum.arg()
375        } else {
376            let amplitude = self.sum.norm() / self.weight;
377            if amplitude > 0. { amplitude.ln() } else { 0. }
378        }
379    }
380}

397pub(super) struct SplineCoefficients {
398    pub(super) knots: Vec<f64>,
399    /// Receptor, then antenna, then cubic spline control point.

401    pub(super) phase: Option<[Vec<Vec<f64>>; 2]>,

404pub(super) fn fit(
405    times: &[f64],
406    baselines: &[BaselineSeries],
407    antenna_count: usize,
408    options: &SplineKernelOptions,
409) -> Option<SplineCoefficients> {
410    if baselines.iter().any(|baseline| {
411        baseline.samples.len() != times.len()
412            || baseline
413                .antennas
414                .iter()
415                .any(|&antenna| antenna >= antenna_count)
416            || baseline.antennas[0] == baseline.antennas[1]
417    }) {
418        return None;
419    }
420    let knots = basis::knots(times, options.spacing_seconds)?;
421    let basis = times
422        .iter()
423        .map(|&time| basis::sample(&knots, time))
424        .collect::<Option<Vec<_>>>()?;
425    let size = (antenna_count, knots.len() - 4);
426    let log_amplitude = if !options.amplitude {
427        None
428    } else {
429        Some(fit_component(&basis, baselines, size, options, false)?)
430    };
431    let phase = if !options.phase {
432        None
433    } else {
434        Some(fit_component(&basis, baselines, size, options, true)?)
435    };
436    Some(SplineCoefficients {
437        knots,
438        log_amplitude,
439        phase,
440    })
441}

443fn fit_component(
444    basis: &[SampleBasis],
445    baselines: &[BaselineSeries],
446    (antenna_count, controls): (usize, usize),
447    options: &SplineKernelOptions,
448    phase: bool,
449) -> Option<[Vec<Vec<f64>>; 2]> {
450    let fit_hand = |hand: usize| {
451        let active = antenna_indices(
452            baselines,
453            antenna_count,
454            hand,
455            phase.then_some(options.reference_antenna),
456        )?;
457        let count = active.iter().flatten().count();
458        let size = controls.checked_mul(count)?;
459        let mut normal = NormalBand::new(size, count.checked_mul(4)?)?;
460        let mut values = vec![0.; basis.len()];
461        let mut phases = Vec::new();
462        for (index, baseline) in baselines.iter().enumerate() {
463            let [first, second] = baseline.antennas;
464            if active[first].is_none() && active[second].is_none() {
465                continue;
466            }
467            for (value, samples) in values.iter_mut().zip(&baseline.samples) {
468                *value = if usable(samples[hand]) {
469                    samples[hand].value(phase)
470                } else {
471                    0.
472                };
473            }
474            if phase {
475                unwrap_phases(
476                    &mut values,
477                    options.phase_points,
478                    options.phase_wrap_radians,
479                );
480            }
481            for ((sample, value), &basis) in baseline.samples.iter().zip(&values).zip(basis) {
482                if !usable(sample[hand]) {
483                    continue;
484                }
485                let terms = basis_terms(basis, [active[first], active[second]], count, phase);
486                normal.accumulate(&terms.0[..terms.1], *value, sample[hand].weight);
487            }
488            if phase {
489                phases.push((index, values.clone()));
490            }
491        }
492        normal.factor()?;
493        let mut solved = normal.solve()?.to_vec();
494        // The temporal unwrap seeds the linear fit. Refine it with circular
495        // residuals, so baselines displaced by whole turns constrain the same
496        // antenna phases. The weighted design matrix stays fixed.
497        for _ in 1..100 {
498            if !phase
499                || !refine_phase(&mut normal, &mut solved, &phases, |index, time| {
500                    let baseline = &baselines[index];
501                    let antennas = baseline.antennas.map(|antenna| active[antenna]);
502                    let terms = basis_terms(basis[time], antennas, count, true);
503                    (terms, baseline.samples[time][hand])
504                })?
505            {
506                break;
507            }
508        }
509        Some(
510            active
511                .into_iter()
512                .map(|index| {
513                    index.map_or_else(
514                        || vec![0.; controls],
515                        |index| solved.iter().skip(index).step_by(count).copied().collect(),
516                    )
517                })
518                .collect(),
519        )
520    };
521    let first = fit_hand(0)?;
522    let second = if options.hands == 1 {
523        vec![vec![0.; controls]; antenna_count]
524    } else {
525        fit_hand(1)?
526    };
527    Some([first, second])
528}

530fn refine_phase(
531    normal: &mut NormalBand,
532    solution: &mut [f64],
533    phases: &[(usize, Vec<f64>)],
534    sample_at: impl Fn(usize, usize) -> (([Term; 8], usize), WeightedVisibility),
535) -> Option<bool> {
536    normal.clear_rhs();
537    for (baseline, values) in phases {
538        for (time, measured) in values.iter().enumerate() {
539            let ((terms, size), sample) = sample_at(*baseline, time);
540            if !usable(sample) {
541                continue;
542            }
543            let terms = &terms[..size];
544            let predicted = terms
545                .iter()
546                .map(|term| term.coefficient * solution[term.index])
547                .sum::<f64>();
548            normal.add_rhs(terms, (measured - predicted).sin(), sample.weight);
549        }
550    }
551    let step = normal.solve()?;
552    for (control, correction) in solution.iter_mut().zip(step) {
553        *control += correction;
554    }
555    Some(step.iter().map(|value| value * value).sum::<f64>() > 1e-14)
556}

558fn antenna_indices(
559    baselines: &[BaselineSeries],
560    antenna_count: usize,
561    hand: usize,
562    reference: Option<usize>,
563) -> Option<Vec<Option<usize>>> {
564    let mut active = vec![false; antenna_count];
565    for baseline in baselines {
566        if baseline.samples.iter().any(|sample| usable(sample[hand])) {
567            for antenna in baseline.antennas {
568                *active.get_mut(antenna)? = true;
569            }
570        }
571    }
572    if let Some(reference) = reference {
573        let present = active.get_mut(reference)?;
574        if !*present {
575            return None;
576        }
577        *present = false;
578    }
579    if !active.iter().any(|&present| present) {
580        return None;
581    }
582    let mut next = 0;
583    Some(
584        active
585            .into_iter()
586            .map(|present| {
587                present.then(|| {
588                    let index = next;
589                    next += 1;
590                    index
591                })
592            })
593            .collect(),
594    )
595}
596
597fn usable(value: WeightedVisibility) -> bool {
598    value.weight.is_finite() && value.weight > 0. && value.sum.is_finite()
599}
600
601fn basis_terms(
602    basis: SampleBasis,
603    antennas: [Option<usize>; 2],
604    count: usize,
605    phase: bool,
606) -> ([Term; 8], usize) {
607    let mut terms = [Term {
608        index: 0,
609        coefficient: 0.,
610    }; 8];
611    let mut size = 0;
612    // CASA's stored baseline phase convention is antenna2 minus antenna1.
613    let signs = [if phase { -1. } else { 1. }, 1.];
614    for (antenna, sign) in antennas.into_iter().zip(signs) {
615        if let Some(antenna) = antenna {
616            for (offset, weight) in basis.weights.into_iter().enumerate() {
617                terms[size] = Term {
618                    index: (basis.start + offset) * count + antenna,
619                    coefficient: sign * weight,
620                };
621                size += 1;
622            }
623        }
624    }
625    (terms, size)
626}

628fn unwrap_phases(values: &mut [f64], points: usize, threshold: f64) {
629    let points = points.max(2);
630    if values.len() <= points.saturating_mul(2) {
631        return;
632    }
633    let end = values.len() - points;
634    let mut scratch = vec![0.; points];
635    for index in ((points..end).step_by(points / 2)).chain([end]) {
636        let difference = phase_step(values, &mut scratch, index);
637        remove_wrap(&mut values[index..], difference, threshold);
638    }
639    let difference = phase_step(values, &mut scratch, points);
640    remove_wrap(&mut values[..points], -difference, threshold);
641}
642
643fn phase_step(values: &[f64], scratch: &mut [f64], index: usize) -> f64 {
644    let width = scratch.len();
645    let mut median = |slice: &[f64]| {
646        scratch.copy_from_slice(slice);
647        scratch.sort_unstable_by(f64::total_cmp);
648        let lower = scratch[(width - 1) / 2];
649        // casacore's default median averages the middle pair only up to 100.
650        if width > 100 {
651            lower
652        } else {
653            0.5 * (lower + scratch[width / 2])
654        }
655    };
656    median(&values[index..index + width]) - median(&values[index - width..index])
657}
658
659fn remove_wrap(samples: &mut [f64], difference: f64, threshold: f64) {
660    let wraps = i32::from(difference < -threshold) - i32::from(difference > threshold);
661    if wraps == 0 {
662        return;
663    }
664    let offset = f64::from(wraps) * TAU;
665    for sample in samples {
666        *sample += offset;
667    }
668}
