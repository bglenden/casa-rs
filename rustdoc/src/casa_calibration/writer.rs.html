24/// Common on-disk identity for one calibration-table family.
25pub(crate) struct CalibrationTableDescriptor<'a> {
26    pub output: &'a Path,
27    pub schema: TableSchema,
28    pub subtype: &'a str,
29    pub parameter_type: Option<&'a str>,
30    /// The MeasurementSet solved on, as a path or a name; the table records
31    /// its base name as `MSName`, as CASA's `NewCalTable::setMetaInfo`
32    /// does, and CASA appends only to a table whose `MSName` is the one it
33    /// would write.
34    pub measurement_set_name: String,
35    pub include_polarization_basis: bool,
36    pub time_extra_precision_column: Option<&'a str>,
37}

39/// Common calibration-table writer failure.
40#[derive(Debug, Error)]
41pub enum CalibrationTableWriteError {
42    #[error("failed to prepare output path {path}: {reason}")]
43    Prepare { path: String, reason: String },
44    #[error("failed to append calibration row to {path}: {source}")]
45    Append {
46        path: String,
47        #[source]
48        source: Box<TableError>,
49    },
50    #[error("failed to save calibration table {path}: {source}")]
51    Save {
52        path: String,
53        #[source]
54        source: Box<TableError>,
55    },
56    #[error("failed to copy {subtable} subtable into {path}: {source}")]
57    CopySubtable {
58        subtable: String,
59        path: String,
60        #[source]
61        source: Box<TableError>,
62    },
63    #[error("failed to publish staged calibration table {staging} as {output}: {reason}")]
64    Publish {
65        staging: String,
66        output: String,
67        reason: String,
68    },
69}

71/// Typed owner of common calibration-table construction and final persistence.
72pub(crate) struct CalibrationTableWriter<'a> {
73    measurement_set: &'a MeasurementSet,
74    output: CalibrationOutput,
75    table: Table,
76    mode: CalibrationOutputMode,
77    sort_for_casa: bool,
78}
79
80impl<'a> CalibrationTableWriter<'a> {
81    pub(crate) fn create_simulated(
82        measurement_set: &'a MeasurementSet,
83        descriptor: CalibrationTableDescriptor<'_>,
84    ) -> Result<Self, CalibrationTableWriteError> {
85        let output = CalibrationOutput::prepare_simulated(descriptor.output)?;
86        let mut writer = Self::staged(
87            measurement_set,
88            descriptor,
89            output,
90            CalibrationOutputMode::Replace,
91        );
92        // Simulation supplies its established row order and budgets one
93        // materialized table. Do not duplicate it for task storeNCT sorting.
94        writer.preserve_row_order();
95        Ok(writer)
96    }
97
98    pub(crate) fn preserve_row_order(&mut self) {
99        self.sort_for_casa = false;
100    }
101
102    pub(crate) fn with_mode(
103        measurement_set: &'a MeasurementSet,
104        descriptor: CalibrationTableDescriptor<'_>,
105        mode: CalibrationOutputMode,
106    ) -> Result<Self, CalibrationTableWriteError> {
107        let output = CalibrationOutput::prepare(descriptor.output, mode)?;
108        Ok(Self::staged(measurement_set, descriptor, output, mode))
109    }
110
111    pub(crate) fn staged(
112        measurement_set: &'a MeasurementSet,
113        descriptor: CalibrationTableDescriptor<'_>,
114        output: CalibrationOutput,
115        mode: CalibrationOutputMode,
116    ) -> Self {
117        let mut table = Table::with_schema(descriptor.schema);
118        table.set_info(TableInfo {
119            table_type: TABLE_INFO_TYPE.to_string(),
120            sub_type: descriptor.subtype.to_string(),
121            readme: Vec::new(),
122        });
123        let measurement_set_name = Path::new(&descriptor.measurement_set_name)
124            .file_name()
125            .map_or(descriptor.measurement_set_name.clone(), |name| {
126                name.to_string_lossy().into_owned()
127            });
128        let producer = if descriptor.subtype == "G Jones" {
129            concat!("casa-rs ", env!("CARGO_PKG_VERSION"))
130        } else {
131            "casa-rs"
132        };
133        for (key, value) in [
134            (KEY_VIS_CAL, descriptor.subtype.to_string()),
135            (KEY_MS_NAME, measurement_set_name),
136            (KEY_CASA_VERSION, producer.to_string()),
137        ] {
138            table
139                .keywords_mut()
140                .upsert(key, Value::Scalar(ScalarValue::String(value)));
141        }
142        if let Some(parameter_type) = descriptor.parameter_type {
143            table.keywords_mut().upsert(
144                KEY_PAR_TYPE,
145                Value::Scalar(ScalarValue::String(parameter_type.to_string())),
146            );
147        }
148        if descriptor.include_polarization_basis {
149            table.keywords_mut().upsert(
150                KEY_POL_BASIS,
151                Value::Scalar(ScalarValue::String("unknown".to_string())),
152            );
153        }
154        set_fixed_unit_keyword(&mut table, COL_TIME, &["s"]);
155        set_measinfo_keyword(&mut table, COL_TIME, "epoch", Some("UTC"));
156        set_fixed_unit_keyword(&mut table, COL_INTERVAL, &["s"]);
157        if let Some(column) = descriptor.time_extra_precision_column {
158            set_fixed_unit_keyword(&mut table, column, &["s"]);
159        }
160        for name in STANDARD_SUBTABLE_KEYWORDS {
161            table.keywords_mut().upsert(
162                *name,
163                Value::table_ref(strip_directory(&output.staging.join(name), &output.staging)),
164            );
165        }
166        Self {
167            measurement_set,
168            output,
169            table,
170            mode,
171            sort_for_casa: true,
172        }
173    }
174
175    /// Staging root used for family-specific subtables before publication.
176    pub(crate) fn output_path(&self) -> &Path {
177        &self.output.staging
178    }
179
180    pub(crate) fn table_mut(&mut self) -> &mut Table {
181        &mut self.table
182    }
183
184    pub(crate) fn append(&mut self, row: RecordValue) -> Result<(), CalibrationTableWriteError> {
185        self.table
186            .add_row(row)
187            .map_err(|source| CalibrationTableWriteError::Append {
188                path: self.output.staging.display().to_string(),
189                source: Box::new(source),
190            })
191    }
192
193    pub(crate) fn copy_subtables(
194        &self,
195        subtables: &[(SubtableId, &'static str)],
196    ) -> Result<(), CalibrationTableWriteError> {
197        for &(id, name) in subtables {
198            self.measurement_set
199                .subtable(id)
200                .expect("required calibration subtable available")
201                .save(TableOptions::new(self.output.staging.join(name)))
202                .map_err(|source| CalibrationTableWriteError::CopySubtable {
203                    subtable: name.to_string(),
204                    path: self.output.staging.display().to_string(),
205                    source: Box::new(source),
206                })?;
207        }
208        Ok(())
209    }
210
211    pub(crate) fn save_main(mut self) -> Result<(), CalibrationTableWriteError> {
212        if self.mode == CalibrationOutputMode::Append && self.output.destination.exists() {
213            if matches!(
214                self.table.info().sub_type.as_str(),
215                "BPOLY" | "GSPLINE" | "EP Jones"
216            ) {
217                let mut old = output_merge::open(&self.output.destination)?;
218                output_merge::append_legacy(
219                    &mut old,
220                    &mut self.table,
221                    &self.output.destination,
222                    &self.output.staging,
223                )?;
224                self.table = old;
225            } else {
226                let spw = output_merge::merge_spw(&self.output.destination, &self.output.staging)?;
227                let mut old = output_merge::open(&self.output.destination)?;
228                output_merge::append(&mut old, &self.table, &self.output.destination)?;
229                output_merge::preserve_subtables(&self.output.destination, &self.output.staging)?;
230                spw.save(TableOptions::new(
231                    self.output.staging.join("SPECTRAL_WINDOW"),
232                ))
233                .map_err(|e| output_merge::invalid(&self.output.staging, e.to_string()))?;
234                self.table = old;
235            }
236        }
237        if self.sort_for_casa {
238            output_merge::save_sorted(&mut self.table, &self.output.staging)?;
239        } else {
240            self.table
241                .save(
242                    TableOptions::new(&self.output.staging)
243                        .with_data_manager(casa_tables::DataManagerKind::StandardStMan),
244                )
245                .map_err(|source| CalibrationTableWriteError::Save {
246                    path: self.output.staging.display().to_string(),
247                    source: Box::new(source),
248                })?;
249        }
250        self.output.commit()
251    }
252}

254pub(crate) fn set_fixed_unit_keyword(table: &mut Table, column: &str, units: &[&str]) {
255    let mut keywords = table.column_keywords(column).cloned().unwrap_or_default();
256    keywords.upsert(
257        "QuantumUnits",
258        Value::Array(ArrayValue::from_string_vec(
259            units.iter().map(|unit| (*unit).to_string()).collect(),
260        )),
261    );
262    table.set_column_keywords(column, keywords);
263}
264
265pub(crate) fn set_measinfo_keyword(
266    table: &mut Table,
267    column: &str,
268    measure_type: &str,
269    measure_ref: Option<&str>,
270) {
271    let mut keywords = table.column_keywords(column).cloned().unwrap_or_default();
272    let mut fields = vec![RecordField::new(
273        "type",
274        Value::Scalar(ScalarValue::String(measure_type.to_string())),
275    )];
276    if let Some(measure_ref) = measure_ref {
277        fields.push(RecordField::new(
278            "Ref",
279            Value::Scalar(ScalarValue::String(measure_ref.to_string())),
280        ));
281    }
282    keywords.upsert("MEASINFO", Value::Record(RecordValue::new(fields)));
283    table.set_column_keywords(column, keywords);
284}

286pub(crate) enum SingleChannelFrequency {
287    MiddleChannel,
288    Mean,
289}
290
291/// Reduce every spectral window of the SPECTRAL_WINDOW subtable at `path`
292/// to one channel, as CASA's `NewCalTable::makeSpwSingleChan` does for a
293/// term that does not depend on frequency: the middle channel's frequency
294/// (`CHAN_FREQ[n/2]`), and CHAN_WIDTH, EFFECTIVE_BW and RESOLUTION the
295/// window's TOTAL_BANDWIDTH. CASA's `applycal` refuses a table whose
296/// windows keep the MeasurementSet's channels.
297pub(crate) fn make_windows_single_channel(
298    path: &Path,
299    frequency: SingleChannelFrequency,
300) -> Result<(), CalibrationTableWriteError> {
301    let failed = |source: TableError| CalibrationTableWriteError::Save {
302        path: path.display().to_string(),
303        source: Box::new(source),
304    };
305    let mut table = Table::open(TableOptions::new(path)).map_err(failed)?;
306    for row in 0..table.row_count() {
307        let value = |column: &str| -> Result<Value, TableError> {
308            Ok(table
309                .cell_accessor(row, column)?
310                .value()?
311                .cloned()
312                .unwrap_or_else(|| Value::Scalar(ScalarValue::Int32(0))))
313        };
314        let frequencies = match value("CHAN_FREQ").map_err(failed)? {
315            Value::Array(ArrayValue::Float64(frequencies)) => frequencies,
316            _ => continue,
317        };
318        let channels = frequencies.len();
319        if channels <= 1 {
320            continue;
321        }
322        let bandwidth = match value("TOTAL_BANDWIDTH").map_err(failed)? {
323            Value::Scalar(ScalarValue::Float64(bandwidth)) => bandwidth,
324            _ => continue,
325        };
326        let one = |value: f64| {
327            Value::Array(ArrayValue::Float64(
328                ArrayD::from_shape_vec(IxDyn(&[1]), vec![value]).expect("one channel"),
329            ))
330        };
331        let frequency = match frequency {
332            SingleChannelFrequency::MiddleChannel => frequencies
333                .iter()
334                .nth(channels / 2)
335                .copied()
336                .unwrap_or_default(),
337            SingleChannelFrequency::Mean => frequencies.iter().sum::<f64>() / channels as f64,
338        };
339        for (column, cell) in [
340            ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(1))),
341            ("CHAN_FREQ", one(frequency)),
342            ("CHAN_WIDTH", one(bandwidth)),
343            ("EFFECTIVE_BW", one(bandwidth)),
344            ("RESOLUTION", one(bandwidth)),
345        ] {
346            table
347                .cell_accessor_mut(row, column)
348                .and_then(|mut accessor| accessor.set(cell))
349                .map_err(failed)?;
350        }
351    }
352    table.flush().map_err(failed)
353}

355pub(crate) fn new_calibration_table_schema(complex: bool) -> TableSchema {
356    let (column, parameter) = if complex {
357        (COL_CPARAM, PrimitiveType::Complex32)
358    } else {
359        (COL_FPARAM, PrimitiveType::Float32)
360    };
361    TableSchema::new(vec![
362        ColumnSchema::scalar(COL_TIME, PrimitiveType::Float64),
363        ColumnSchema::scalar(COL_FIELD_ID, PrimitiveType::Int32),
364        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, PrimitiveType::Int32),
365        ColumnSchema::scalar(COL_ANTENNA1, PrimitiveType::Int32),
366        ColumnSchema::scalar(COL_ANTENNA2, PrimitiveType::Int32),
367        ColumnSchema::scalar(COL_INTERVAL, PrimitiveType::Float64),
368        ColumnSchema::scalar(COL_SCAN_NUMBER, PrimitiveType::Int32),
369        ColumnSchema::scalar(COL_OBSERVATION_ID, PrimitiveType::Int32),
370        ColumnSchema::array_variable(column, parameter, None),
371        ColumnSchema::array_variable(COL_PARAMERR, PrimitiveType::Float32, None),
372        ColumnSchema::array_variable(COL_FLAG, PrimitiveType::Bool, None),
373        ColumnSchema::array_variable(COL_SNR, PrimitiveType::Float32, None),
374        ColumnSchema::array_variable(COL_WEIGHT, PrimitiveType::Float32, None),
375    ])
376    .expect("valid simulated calibration schema")
377}

380mod tests {
381
382    use tempfile::tempdir;
383
384    use crate::output::CalibrationOutput;
385    use crate::output::CalibrationOutputMode;
386    use std::fs;

388    #[test]
389    fn output_collision_is_rejected_without_modifying_existing_data() {
390        let temp = tempdir().expect("tempdir");
391        let output = temp.path().join("gain.cal");
392        fs::create_dir(&output).expect("create existing output");
393        let sentinel = output.join("sentinel");
394        fs::write(&sentinel, b"keep").expect("write sentinel");
395
396        let error = CalibrationOutput::prepare(&output, CalibrationOutputMode::Create)
397            .expect_err("collision must fail");
398
399        assert!(error.to_string().contains("output already exists"));
400        assert_eq!(fs::read(&sentinel).expect("sentinel remains"), b"keep");
401    }
402
403    #[test]
404    fn interrupted_staging_output_is_visible_and_blocks_reuse() {
405        let temp = tempdir().expect("tempdir");
406        let output = temp.path().join("bandpass.cal");
407        let staging = CalibrationOutput::prepare(&output, CalibrationOutputMode::Create)
408            .expect("reserve staging output");
409        fs::write(staging.staging.join("partial"), b"incomplete").expect("write partial output");
410
411        let error = CalibrationOutput::prepare(&output, CalibrationOutputMode::Create)
412            .expect_err("staging collision must fail");
413
414        assert!(!output.exists());
415        assert_eq!(
416            fs::read(staging.staging.join("partial")).expect("partial remains"),
417            b"incomplete"
418        );
419        assert!(
420            error
421                .to_string()
422                .contains("another writer or interrupted publication")
423        );
424    }
