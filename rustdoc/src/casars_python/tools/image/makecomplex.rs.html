19fn combine<T>(
20    real: &dyn ImageInterface<T>,
21    imaginary: &dyn ImageInterface<T>,
22    real_record: &RecordValue,
23    region: Option<&Region>,
24    output_path: Option<&Path>,
25    wrap: impl FnOnce(Storage<Complex<T>>) -> ImageHandle,
26) -> PyResult<(ImageHandle, Vec<usize>)>
27where
28    T: ImagePixel + Copy,
29    Complex<T>: ImagePixel + Copy,
30{
31    let selection = SubimageOptions {
32        region: region.map(Region::as_region),
33        dropdeg: false,
34        keepaxes: &[],
35    };
36    let plan = SubimagePlan::new(real, Some(real_record.clone()), selection).map_err(raised)?;
37    let mut storage = Storage::create(
38        plan.shape.clone(),
39        plan.coordinates.clone(),
40        output_path,
41        None,
42    )
43    .map_err(raised)?;
44    with_storage!(&mut storage, output => plan.make_complex_to(real, imaginary, output.as_mut(), region.map(Region::as_region)))
45        .map_err(raised)?;
46    let mut result = wrap(storage);
47    let origin = plan.bounds().0.to_vec();
48    if let Some(record) = plan.coordinate_record {
49        result.set_coordinate_record(record)?;
50    }
51    Ok((result, origin))
52}

54fn preserve_beam_records(
55    real: &ImageHandle,
56    output: &mut ImageHandle,
57    coordinates: &RecordValue,
58    origin: &[usize],
59) -> PyResult<()> {
60    let Some(stored) = image_info_record(real) else {
61        return Ok(());
62    };
63    let source = image_info(real).map_err(raised)?.beam_set;
64    let result = image_info(output).map_err(raised)?.beam_set;
65    let mut record = stored.clone();
66    record.remove("restoringbeam");
67    record.remove("perplanebeams");
68    let (spectral, stokes) = plane_axes(coordinates)?;
69    let offset = |axis: Option<usize>, length: usize| {
70        axis.and_then(|axis| origin.get(axis).copied())
71            .filter(|_| length > 1)
72            .unwrap_or(0)
73    };
74    let first_channel = offset(spectral, source.n_channels());
75    let first_stokes = offset(stokes, source.n_stokes());
76    let beam = |channel: usize, stokes: usize| {
77        let channel = (first_channel + channel).min(source.n_channels().saturating_sub(1));
78        let stokes = (first_stokes + stokes).min(source.n_stokes().saturating_sub(1));
79        Value::Record(stored_beam_record(Some(stored), &source, channel, stokes))
80    };
81    if result.is_single() {
82        record.upsert("restoringbeam", beam(0, 0));
83    } else if result.is_multi() {
84        let mut grid = result.to_record();
85        for polarization in 0..result.n_stokes() {
86            for channel in 0..result.n_channels() {
87                let index = polarization * result.n_channels() + channel;
88                grid.upsert(format!("*{index}"), beam(channel, polarization));
89            }
90        }
91        record.upsert("perplanebeams", Value::Record(grid));
92    }
93    with_pixels!(output, storage => with_storage!(storage, image => image.set_image_info_record(record))).map_err(raised)
94}

96#[pymethods]
97impl PyImageTool {
98    #[pyo3(signature = (outfile, imag, region=None, overwrite=false), text_signature = "($self, outfile, imag, region={}, overwrite=False)")]
99    fn makecomplex(
100        &self,
101        py: Python<'_>,
102        outfile: &str,
103        imag: &str,
104        region: Option<&Bound<'_, PyAny>>,
105        overwrite: bool,
106    ) -> PyResult<bool> {
107        if self.state().image.is_none() {
108            return Ok(false);
109        }
110        let region = Region::argument(py, region, "makecomplex")?;
111        py.detach(|| {
112            let state = self.state();
113            let Some(attachment) = &state.image else {
114                return Ok(false);
115            };
116            let product = publish_image(outfile, overwrite, |path| {
117                let mut real = attachment.lock();
118                if !matches!(*real, ImageHandle::Float(_) | ImageHandle::Double(_)) {
119                    return Err(raised("The attached image must be float valued"));
120                }
121                attachment.require_pixel_read_lock(&mut real)?;
122                let imaginary = ImageHandle::open_read(Path::new(imag)).map_err(raised)?;
123                if !matches!(imaginary, ImageHandle::Float(_) | ImageHandle::Double(_)) {
124                    return Err(raised(format!("{imag} does not have supported real valued pixels")));
125                }
126                if real.kind() != imaginary.kind() {
127                    return Err(raised("Real and imaginary images do not have the same precision"));
128                }
129                if !outfile.is_empty() {
130                    for source in [real.path(), imaginary.path()].into_iter().flatten() {
131                        ensure_outfile_does_not_alias_input(Path::new(outfile), source)
132                            .map_err(raised)?;
133                    }
134                }
135                if real.shape() != imaginary.shape() {
136                    return Err(raised("Image shapes are not identical"));
137                }
138                let real_record = real.coordinate_record();
139                if !coordinates_conform(&real_record, &imaginary.coordinate_record(), false)? {
140                    return Err(raised("Image Coordinate systems are not conformant"));
141                }
142                let (mut output, origin) = match (&*real, &imaginary) {
143                    (ImageHandle::Float(left), ImageHandle::Float(right)) => {
144                        left.with_float_view(|left| right.with_float_view(|right| {
145                            combine(left, right, &real_record, region.as_ref(), path, ImageHandle::Complex)
146                        }))?
147                    }
148                    (ImageHandle::Double(left), ImageHandle::Double(right)) => {
149                        with_storage!(left, left => with_storage!(right, right => {
150                            combine(left.as_ref(), right.as_ref(), &real_record, region.as_ref(), path, ImageHandle::DComplex)
151                        }))?
152                    }
153                    _ => unreachable!("real images with matching precision"),
154                };
155                preserve_beam_records(&real, &mut output, &real_record, &origin)?;
156                let rows = with_pixels!(&*real, storage => match storage {
157                    Storage::Paged(image) => image_history_rows(image.name().expect("persistent image")),
158                    Storage::Temp { image, .. } => Ok(image.history_rows().map_err(raised)?.into_owned()),
159                })
160                .map_err(raised)?;
161                with_pixels!(&mut output, storage => with_storage!(storage, image => image.append_history_entries(&rows)))
162                    .map_err(raised)?;
163                if state.history {
164                    let region = region.as_ref().map_or_else(|| "{}".into(), Region::history);
165                    output.record_history("makecomplex", format!(
166                        "ia.makecomplex(outfile={outfile:?}, imag={imag:?}, region={region}, overwrite={overwrite})"
167                    ))?;
168                }
169                output.save().map_err(raised)?;
170                Ok(output)
171            })?;
172            drop(product);
173            Ok(true)
174        })
175    }
