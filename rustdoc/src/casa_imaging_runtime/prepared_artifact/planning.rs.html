16/// Canonical plan fragment for one cold-generate, cold-load, or warm-reuse operation.
17pub struct PreparedArtifactPlanFragment<'a> {
18    descriptor: &'a PreparedArtifactDescriptor,
19    store: &'a PreparedArtifactStore,
20    operation: PreparedArtifactOperation,
21    producer: WorkNodeId,
22    publication_commit: WorkNodeId,
23    release_implementation: WorkImplementationId,
24    load_source: Option<&'a PreparedArtifactLoadSource>,
25}
26
27impl<'a> PreparedArtifactPlanFragment<'a> {
28    /// Bind a prepared operation to its producer, terminal publication gate, and release owner.
29    #[must_use]
30    pub fn new(
31        descriptor: &'a PreparedArtifactDescriptor,
32        store: &'a PreparedArtifactStore,
33        operation: PreparedArtifactOperation,
34        producer: WorkNodeId,
35        publication_commit: WorkNodeId,
36        release_implementation: WorkImplementationId,
37    ) -> Self {
38        Self {
39            descriptor,
40            store,
41            operation,
42            producer,
43            publication_commit,
44            release_implementation,
45            load_source: None,
46        }
47    }
48
49    /// Bind the exact predecessor-owned source for a cold load.
50    #[must_use]
51    pub const fn with_load_source(mut self, source: &'a PreparedArtifactLoadSource) -> Self {
52        self.load_source = Some(source);
53        self
54    }
55
56    /// Compose this fragment with already validated T17/T28 physical work.
57    pub fn compose(
58        self,
59        base: &PhysicalWorkBinding,
60    ) -> Result<PhysicalWorkBinding, PreparedArtifactPlanError> {
61        if matches!(self.operation, PreparedArtifactOperation::Load) != self.load_source.is_some() {
62            return Err(PreparedArtifactPlanError::InvalidSourceBinding);
63        }
64        let reservation = self.store.reservation(self.descriptor, self.operation)?;
65        let suffix = match self.operation {
66            PreparedArtifactOperation::Generate => "generate",
67            PreparedArtifactOperation::Load => "load",
68            PreparedArtifactOperation::Reuse => "reuse",
69        };
70        let cell_identity = self.descriptor.identity();
71        let allocation_id =
72            AllocationId::new(format!("prepared-resident-buffer-{suffix}-{cell_identity}"));
73        let slot_id =
74            PhysicalSlotId::new(format!("prepared-resident-slot-{suffix}-{cell_identity}"));
75        let compatibility = SlotCompatibility {
76            memory_domain: CapacityDomainId::new("host-memory"),
77            views: BTreeSet::from([CapacityViewId::new("host-memory")]),
78            alignment_bytes: 8,
79            storage_mode: StorageMode::Host,
80            layout: AllocationLayout::new("prepared-artifact-streaming-buffer"),
81            initialization: InitializationPolicy::OverwriteBeforeRead,
82            access: AllocationAccess::ReadWrite,
83        };
84        let demand_id = self.descriptor.storage_demand_id();
85        let mut alternative: DemandAlternative =
86            base.execution_dag().resource_alternative().clone();
87        alternative.id = AlternativeId::new(format!(
88            "{}-prepared-{suffix}-{cell_identity}",
89            alternative.id.as_str()
90        ));
91        alternative.demand.memory.push(MemoryDemand {
92            allocation_id: allocation_id.as_str().to_string(),
93            hard_bytes: reservation.resident_buffer_bytes(),
94            preferred_bytes: reservation.resident_buffer_bytes(),
95            views: vec![CapacityViewId::new("host-memory")],
96        });
97        alternative.demand.locks = combine_count(alternative.demand.locks, 2);
98        alternative.demand.file_descriptors = combine_count(
99            alternative.demand.file_descriptors,
100            reservation.file_descriptors(),
101        );
102        alternative.demand.io_buffers.storage_manager_bytes = alternative
103            .demand
104            .io_buffers
105            .storage_manager_bytes
106            .max(reservation.resident_buffer_bytes());
107        if let Some(cache_demand) = alternative
108            .demand
109            .storage
110            .iter_mut()
111            .find(|demand| demand.demand_id == demand_id)
112        {
113            if cache_demand.domain != *self.store.storage_domain() {
114                return Err(PreparedArtifactError::CachePolicyMismatch.into());
115            }
116            cache_demand.temporary_bytes = cache_demand
117                .temporary_bytes
118                .max(reservation.temporary_staging_bytes());
119            cache_demand.persistent_cache_bytes = cache_demand
120                .persistent_cache_bytes
121                .max(reservation.persistent_cache_bytes());
122            cache_demand.read_rate = combine_count(cache_demand.read_rate, 1);
123            cache_demand.write_rate = combine_count(cache_demand.write_rate, 1);
124            cache_demand.operations_rate = combine_count(cache_demand.operations_rate, 1);
125            cache_demand.queue_slots = combine_count(cache_demand.queue_slots, 1);
126        } else {
127            alternative.demand.storage.push(StorageDemand {
128                demand_id: demand_id.clone(),
129                domain: self.store.storage_domain().clone(),
130                temporary_bytes: reservation.temporary_staging_bytes(),
131                staged_output_bytes: 0,
132                final_output_bytes: 0,
133                persistent_cache_bytes: reservation.persistent_cache_bytes(),
134                read_rate: CountDemand::new(1, 1),
135                write_rate: CountDemand::new(1, 1),
136                operations_rate: CountDemand::new(1, 1),
137                queue_slots: CountDemand::new(1, 1),
138            });
139        }
140
141        let source_demands = self
142            .load_source
143            .map(PreparedArtifactLoadSource::storage_demands)
144            .unwrap_or_default();
145        alternative.demand.storage.extend(source_demands.iter().map(
146            |(source_demand_id, source_domain)| StorageDemand {
147                demand_id: source_demand_id.clone(),
148                domain: source_domain.clone(),
149                temporary_bytes: 0,
150                staged_output_bytes: 0,
151                final_output_bytes: 0,
152                persistent_cache_bytes: 0,
153                read_rate: CountDemand::new(1, 1),
154                write_rate: CountDemand::zero(),
155                operations_rate: CountDemand::new(1, 1),
156                queue_slots: CountDemand::new(1, 1),
157            },
158        ));
159
160        let mut claims = vec![
161            claim(LeaseResource::Workers, 1),
162            claim(LeaseResource::Locks, 1),
163            claim(
164                LeaseResource::FileDescriptors,
165                reservation.file_descriptors(),
166            ),
167            claim(
168                LeaseResource::Storage {
169                    demand_id: demand_id.clone(),
170                    use_kind: StorageUseKind::PersistentCache,
171                },
172                reservation.persistent_cache_bytes(),
173            ),
174            claim(
175                LeaseResource::IoBuffer(IoBufferKind::StorageManager),
176                reservation.resident_buffer_bytes(),
177            ),
178            claim(
179                LeaseResource::StorageReadRate {
180                    demand_id: demand_id.clone(),
181                },
182                1,
183            ),
184            claim(
185                LeaseResource::StorageWriteRate {
186                    demand_id: demand_id.clone(),
187                },
188                1,
189            ),
190            claim(
191                LeaseResource::StorageOperationsRate {
192                    demand_id: demand_id.clone(),
193                },
194                1,
195            ),
196            claim(
197                LeaseResource::StorageQueue {
198                    demand_id: demand_id.clone(),
199                },
200                1,
201            ),
202        ];
203        if reservation.temporary_staging_bytes() > 0 {
204            claims.push(claim(
205                LeaseResource::Storage {
206                    demand_id,
207                    use_kind: StorageUseKind::Temporary,
208                },
209                reservation.temporary_staging_bytes(),
210            ));
211        }
212        for source_demand_id in source_demands.keys() {
213            claims.extend([
214                claim(
215                    LeaseResource::StorageReadRate {
216                        demand_id: source_demand_id.clone(),
217                    },
218                    1,
219                ),
220                claim(
221                    LeaseResource::StorageOperationsRate {
222                        demand_id: source_demand_id.clone(),
223                    },
224                    1,
225                ),
226                claim(
227                    LeaseResource::StorageQueue {
228                        demand_id: source_demand_id.clone(),
229                    },
230                    1,
231                ),
232            ]);
233        }
234        let prepared_node = WorkNode {
235            id: self.descriptor.work_node_id(self.operation),
236            kind: WorkKind::Cache,
237            domain: WorkDomain::Cpu,
238            implementation: self.descriptor.work_implementation_id(self.operation),
239            dependencies: BTreeSet::from([WorkDependency::Work(self.producer)]),
240            claims,
241            allocations: vec![AllocationUse {
242                allocation: allocation_id.clone(),
243                lifetime: ClaimLifetime::Work,
244            }],
245            fences: BTreeSet::new(),
246            quiescence_after: BTreeSet::new(),
247        };
248        let release_id = WorkNodeId::new(format!(
249            "prepared-release-{suffix}-{}",
250            self.descriptor.identity()
251        ));
252        let release_node = WorkNode {
253            id: release_id.clone(),
254            kind: WorkKind::Release,
255            domain: WorkDomain::Cpu,
256            implementation: self.release_implementation,
257            dependencies: BTreeSet::from([WorkDependency::Work(prepared_node.id.clone())]),
258            claims: vec![
259                claim(LeaseResource::Workers, 1),
260                claim(
261                    LeaseResource::IoBuffer(IoBufferKind::StorageManager),
262                    reservation.resident_buffer_bytes(),
263                ),
264            ],
265            allocations: vec![AllocationUse {
266                allocation: allocation_id.clone(),
267                lifetime: ClaimLifetime::Work,
268            }],
269            fences: BTreeSet::new(),
270            quiescence_after: BTreeSet::new(),
271        };
272        let mut nodes = base
273            .execution_dag()
274            .nodes()
275            .values()
276            .cloned()
277            .collect::<Vec<_>>();
278        nodes
279            .iter_mut()
280            .find(|node| node.id == self.publication_commit)
281            .ok_or(PreparedArtifactPlanError::MissingPublicationCommit)?
282            .dependencies
283            .insert(WorkDependency::Work(release_id.clone()));
284        nodes.extend([prepared_node.clone(), release_node]);
285
286        let allocation = LogicalAllocation {
287            id: allocation_id.clone(),
288            bytes: reservation.resident_buffer_bytes(),
289            purpose: AllocationPurpose::IoBuffer(IoBufferKind::StorageManager),
290            compatibility: compatibility.clone(),
291            physical_slot: slot_id.clone(),
292            lifetime: AllocationLifetime {
293                acquire_at: prepared_node.id.clone(),
294                release_after: BTreeSet::from([WorkDependency::Work(release_id.clone())]),
295            },
296        };
297        let slot = PhysicalSlot {
298            id: slot_id,
299            lease_resource: LeaseResource::Memory {
300                allocation_id: allocation_id.as_str().to_string(),
301            },
302            capacity_bytes: reservation.resident_buffer_bytes(),
303            compatibility,
304        };
305        let dag = ExecutionDag::new(ExecutionDagSpecification {
306            required_resource_capabilities: base
307                .execution_dag()
308                .required_resource_capabilities()
309                .clone(),
310            resource_alternative: alternative,
311            nodes,
312            logical_allocations: base
313                .execution_dag()
314                .logical_allocations()
315                .values()
316                .cloned()
317                .chain([allocation])
318                .collect(),
319            physical_slots: base
320                .execution_dag()
321                .physical_slots()
322                .values()
323                .cloned()
324                .chain([slot])
325                .collect(),
326            initial_knobs: base.execution_dag().initial_knobs().clone(),
327            adaptations: base
328                .execution_dag()
329                .adaptations()
330                .values()
331                .cloned()
332                .collect(),
333        })?;
334        let prepared_stage =
335            StagePrediction::new(prepared_node.id, 1_000).with_io(vec![IoPrediction::new(
336                IoBufferKind::StorageManager,
337                reservation.persistent_cache_bytes().max(
338                    reservation
339                        .entry_bytes()
340                        .checked_add(reservation.source_read_bytes())
341                        .ok_or(PreparedArtifactError::ArtifactTooLarge)?,
342                ),
343                10_000,
344            )]);
345        let release_stage = StagePrediction::new(release_id, 100).with_io(vec![IoPrediction::new(
346            IoBufferKind::StorageManager,
347            reservation.resident_buffer_bytes(),
348            1,
349        )]);
350        let prediction = PlanPrediction::new(
351            base.prediction().elapsed_nanos()
352                + prepared_stage.elapsed_nanos()
353                + release_stage.elapsed_nanos(),
354            base.prediction().confidence(),
355            base.prediction().uncertainty().to_vec(),
356            base.prediction()
357                .stages()
358                .values()
359                .cloned()
360                .chain([prepared_stage, release_stage])
361                .collect(),
362        )?;
363        let mut artifacts = base.artifacts().to_vec();
364        if let Some(source) = self.load_source {
365            artifacts.push(source.planned_artifact());
366        }
367        artifacts.extend([
368            self.descriptor.planned_artifact(self.operation),
369            self.descriptor.eviction_artifact(self.operation),
370        ]);
371        Ok(PhysicalWorkBinding::new(
372            dag,
373            prediction,
374            artifacts,
375            base.observation_transaction().clone(),
376            base.publication_layouts().clone(),
377        )?)
378    }
379}

381fn claim(resource: LeaseResource, amount: u64) -> ResourceClaim {
382    ResourceClaim {
383        resource,
384        amount,
385        lifetime: ClaimLifetime::Work,
386    }

389fn combine_count(base: CountDemand, fragment: u64) -> CountDemand {
390    CountDemand::new(base.hard().max(fragment), base.preferred().max(fragment))
391}

393/// Failure to compose a prepared operation with an existing physical plan.
394#[derive(Debug)]
395pub enum PreparedArtifactPlanError {
396    /// The store could not derive a bounded reservation.
397    Prepared(PreparedArtifactError),
398    /// The composed execution DAG is invalid.
399    Execution(ExecutionError),
400    /// The complete physical binding is inconsistent.
401    Binding(PhysicalWorkBindingError),
402    /// A cold load omitted its source or another operation supplied one.
403    InvalidSourceBinding,
404    /// The named publication commit was absent from the base plan.
405    MissingPublicationCommit,
406}
407
408impl fmt::Display for PreparedArtifactPlanError {
409    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
410        match self {
411            Self::Prepared(error) => error.fmt(formatter),
412            Self::Execution(error) => error.fmt(formatter),
413            Self::Binding(error) => error.fmt(formatter),
414            Self::InvalidSourceBinding => {
415                formatter.write_str("invalid prepared load-source binding")
416            }
417            Self::MissingPublicationCommit => {
418                formatter.write_str("prepared fragment publication commit is absent")
419            }
420        }
421    }
422}
423
424impl Error for PreparedArtifactPlanError {}
425
426impl From<PreparedArtifactError> for PreparedArtifactPlanError {
427    fn from(error: PreparedArtifactError) -> Self {
428        Self::Prepared(error)
429    }
430}
431
432impl From<ExecutionError> for PreparedArtifactPlanError {
433    fn from(error: ExecutionError) -> Self {
434        Self::Execution(error)
435    }
436}
437
438impl From<PhysicalWorkBindingError> for PreparedArtifactPlanError {
439    fn from(error: PhysicalWorkBindingError) -> Self {
440        Self::Binding(error)
441    }
