41struct TimeAverageStatistics {
42    partition: [i32; 4],
43    first: Integration,
44    last: Integration,
45    last_usable: Option<Integration>,
46    weights: Vec<ChannelWeightSum>,
47    channel_bin_widths: Vec<usize>,
48    visibility_shape: Option<(usize, usize)>,
49}
50
51struct ChannelWeightSum {
52    flagged: bool,
53    sum: f32,
54}
55
56impl Default for ChannelWeightSum {
57    fn default() -> Self {
58        Self {
59            flagged: true,
60            sum: 0.0,
61        }
62    }
63}
64
65impl ChannelWeightSum {
66    fn sample(&self) -> (bool, f32) {
67        (self.flagged, self.sum)
68    }
69
70    fn observe(&mut self, flagged: bool, weight: f32) {
71        if flagged == self.flagged {
72            self.sum += weight;
73        } else if self.flagged {
74            self.flagged = false;
75            self.sum = weight;
76        }
77    }
78}
79
80impl TimeAverageStatistics {
81    fn integration(&self) -> Integration {
82        let last = self.last_usable.unwrap_or(self.last);
83        let (time, interval) = centered_time_and_interval(
84            (self.first.time, self.first.interval),
85            (last.time, last.interval),
86        );
87        Integration { time, interval }
88    }
89
90    fn baseline_weight(&self) -> f64 {
91        let weight = if self.channel_bin_widths.is_empty() {
92            mean_weight_preferring_unflagged(self.weights.iter().map(ChannelWeightSum::sample))
93        } else {
94            let mut channels = self.weights.as_slice();
95            mean_weight_preferring_unflagged(self.channel_bin_widths.iter().map(|&width| {
96                let (bin, rest) = channels.split_at(width);
97                channels = rest;
98                plotms_columns::sum_channel_weights(
99                    bin.iter().map(ChannelWeightSum::sample),
100                    plotms_columns::ChannelWeightMask::Positive,
101                )
102            }))
103        };
104        plotms_columns::baseline_weight(f64::from(weight))
105    }
106}

108impl TimeBaselineStatistics {
109    pub(super) fn observe(&mut self, row: TimeBaselineRow<'_>) {
110        let group = self
111            .groups
112            .entry(row.time_group)
113            .or_insert_with(|| TimeAverageStatistics {
114                partition: row.partition,
115                first: row.integration,
116                last: row.integration,
117                last_usable: None,
118                weights: row.first_correlation.map_or_else(Vec::new, |_| {
119                    row.bins
120                        .iter()
121                        .flat_map(ChannelBin::channels)
122                        .map(|_| ChannelWeightSum::default())
123                        .collect()
124                }),
125                channel_bin_widths: if row.first_correlation.is_some()
126                    && row.bins.iter().any(|bin| bin.averaged)
127                {
128                    row.bins.iter().map(|bin| bin.channels().count()).collect()
129                } else {
130                    Vec::new()
131                },
132                visibility_shape: row.visibility_shape,
133            });
134        if row.integration.time < group.first.time {
135            group.first = row.integration;
136        }
137        if row.integration.time > group.last.time {
138            group.last = row.integration;
139        }
140        if row.has_unflagged_visibility
141            && group
142                .last_usable
143                .is_none_or(|last| row.integration.time > last.time)
144        {
145            group.last_usable = Some(row.integration);
146        }
147        if let Some(correlation) = row.first_correlation {
148            for (channel, sample) in row
149                .bins
150                .iter()
151                .flat_map(ChannelBin::channels)
152                .zip(&mut group.weights)
153            {
154                let weight = row.weights.map_or(1.0, |grid| {
155                    grid.values[correlation * grid.chan_count + channel] as f32
156                });
157                sample.observe(row.flags[(correlation, channel)], weight);
158            }
159        }
160    }

162    pub(super) fn plan(
163        self,
164        time_plan: &PlotTimeGroups,
165        rows: &[usize],
166        averaging: &MsAverageSpec,
167    ) -> Result<TimeBaselinePlan, String> {
168        let interval = averaging.avgtime.unwrap_or(0.0);
169        let mut averages: Vec<BaselineAverage> = Vec::new();
170        let mut rows_to_average = HashMap::with_capacity(self.groups.len());
171        let mut shapes = VisibilityAverageShapes::default();
172        for buffer in time_plan.buffers() {
173            let Some(anchor) = self.groups.get(&buffer.start) else {
174                continue;
175            };
176            let bounds = anchor.integration();
177            let [observation, scan, field, window] = anchor.partition;
178            let partition = [
179                observation,
180                if averaging.avgscan { 0 } else { scan },
181                if averaging.avgfield { 0 } else { field },
182                if averaging.avgspw { 0 } else { window },
183            ];
184            let (start, end) = (
185                bounds.time - bounds.interval / 2.0,
186                bounds.time + bounds.interval / 2.0,
187            );
188            let starts_new_average = averages.last().is_none_or(|average| {
189                let elapsed = bounds.time - average.anchor_time;
190                elapsed < 0.0 || elapsed > interval || average.partition != partition
191            });
192            if starts_new_average {
193                averages.push(BaselineAverage {
194                    partition,
195                    scan,
196                    field,
197                    anchor_time: bounds.time,
198                    row: rows[time_plan.first_row_slot(buffer.start)],
199                    first_buffer_group: buffer.start,
200                    last_buffer_group: buffer.start,
201                    start,
202                    end,
203                });
204            } else if let Some(average) = averages.last_mut() {
205                average.start = average.start.min(start);
206                average.end = average.end.max(end);
207                average.last_buffer_group = buffer.start;
208                if average.scan != scan {
209                    average.scan = -1;
210                }
211            }
212            let output = averages.len() - 1;
213            if !averaging.avgspw
214                && let Some((correlations, channels)) = anchor.visibility_shape
215            {
216                shapes.check_shape(
217                    "vector antenna time averaging",
218                    output,
219                    correlations,
220                    channels,
221                )?;
222            }
223            for group in buffer.clone() {
224                if let Some(statistics) = self.groups.get(&group) {
225                    if averaging.avgspw
226                        && let Some((correlations, channels)) = statistics.visibility_shape
227                    {
228                        shapes.check_shape(
229                            "vector time and spectral window averaging",
230                            output,
231                            correlations,
232                            channels,
233                        )?;
234                    }
235                    rows_to_average.insert(group, (output, statistics.baseline_weight()));
236                }
237            }
238        }
239        Ok(TimeBaselinePlan {
240            averages,
241            rows_to_average,
242        })
243    }
244}

246struct BaselineAverage {
247    partition: [i32; 4],
248    scan: i32,
249    field: i32,
250    anchor_time: f64,
251    row: usize,
252    first_buffer_group: usize,
253    last_buffer_group: usize,
254    start: f64,
255    end: f64,
256}
257
258pub(super) struct TimeBaselinePlan {
259    averages: Vec<BaselineAverage>,
260    rows_to_average: HashMap<usize, (usize, f64)>,
261}

263impl TimeBaselinePlan {
264    pub(super) fn output_group(&self, time_group: usize) -> Option<usize> {
265        self.rows_to_average
266            .get(&time_group)
267            .map(|&(group, _)| group)
268    }
269
270    pub(super) fn scan_and_field(&self, average: usize) -> Option<(i32, i32)> {
271        self.averages
272            .get(average)
273            .map(|average| (average.scan, average.field))
274    }
275
276    pub(super) fn center(&self, average: usize) -> Option<f64> {
277        self.averages
278            .get(average)
279            .map(|average| average.start + (average.end - average.start) / 2.0)
280    }
281
282    pub(super) fn spectral_window_buffers(
283        &self,
284    ) -> impl Iterator<Item = (usize, usize, (i32, i32))> + '_ {
285        self.averages.iter().map(|average| {
286            (
287                average.first_buffer_group,
288                average.last_buffer_group,
289                (average.scan, average.field),
290            )
291        })
292    }

294    /// Each point here is a normalized complex time average for one baseline.
295    /// Only project to the displayed amplitude or phase after this second mean.
296    pub(super) fn combine_series(
297        &self,
298        series: &mut AveragedSeriesAccumulator,
299        retain_baselines: bool,
300    ) {
301        let mean = |samples: &AveragedVisibility| {
302            if retain_baselines {
303                samples
304                    .transformed_vector_sample()
305                    .map(|(value, _)| Complex64::new(f64::from(value.re), f64::from(value.im)))
306            } else {
307                samples.vector_mean()
308            }
309        };
310        for (mut key, mut point) in std::mem::take(&mut series.points) {
311            if !point.has_usable_weight() {
312                continue;
313            }
314            let Some(&(group, weight)) = key
315                .time_key
316                .group()
317                .and_then(|time_group| self.rows_to_average.get(&time_group))
318            else {
319                continue;
320            };
321            let Some(y) = mean(&point.y_visibility) else {
322                continue;
323            };
324            let x = mean(&point.x_visibility);
325            key.time_key = AveragingTimeKey::Group(group);
326            let output = series.points.entry(key).or_default();
327            if output.representative.is_none() {
328                if let Some(mut representative) = point.representative.take() {
329                    if !retain_baselines {
330                        representative.row = self.averages[group].row;
331                    }
332                    output.representative = Some(representative);
333                }
334                output.display_channel = point.display_channel;
335                output.numeric_axis_samples = point.numeric_axis_samples;
336            }
337            for (output, sample) in [
338                (&mut output.x_visibility, x),
339                (&mut output.y_visibility, Some(y)),
340            ] {
341                if let Some(sample) = sample {
342                    if retain_baselines {
343                        output.add_normalized_vector_sample(
344                            Complex32::new(sample.re as f32, sample.im as f32),
345                            weight as f32,
346                        );
347                    } else {
348                        output.add_weighted_value(sample, weight);
349                    }
350                }
351            }
352        }
353    }

355    /// Reduce the complex means from each time-averaged baseline with its
356    /// transformed spectrum. The second antenna receives the conjugate.
357    pub(super) fn combine_antenna_series(&self, series: &mut AveragedSeriesAccumulator) {
358        for (key, point) in std::mem::take(&mut series.points) {
359            let AveragingBaselineToken::Baseline(first, second) = key.baseline else {
360                continue;
361            };
362            let Some(group) = key
363                .time_key
364                .group()
365                .and_then(|group| self.output_group(group))
366            else {
367                continue;
368            };
369            let Some((y, channel_weight)) = point.y_visibility.transformed_vector_sample() else {
370                continue;
371            };
372            let x = point
373                .x_visibility
374                .transformed_vector_sample()
375                .map(|(value, _)| value);
376            let weight = channel_weight
377                .map(f64::from)
378                .or(point.combined_primary_weight);
379            let weight = plotms_columns::baseline_weight(weight.unwrap_or(1.0)) as f32;
380            for (antenna, x, y) in [
381                (first, x, y),
382                (second, x.map(|value| value.conj()), y.conj()),
383            ] {
384                let key = AveragingPointKey {
385                    time_key: AveragingTimeKey::Group(group),
386                    baseline: AveragingBaselineToken::Antenna(antenna),
387                    ..key.clone()
388                };
389                let output = series.points.entry(key).or_default();
390                if output.representative.is_none() {
391                    output.representative = point.representative.clone();
392                    output.display_channel = point.display_channel;
393                    output.numeric_axis_samples = point.numeric_axis_samples.clone();
394                }
395                if let Some(x) = x {
396                    output.x_visibility.add_normalized_vector_sample(x, weight);
397                }
398                output.y_visibility.add_normalized_vector_sample(y, weight);
399            }
400        }
401    }
