19fn hex(bytes: [u8; 32]) -> String {
20    use std::fmt::Write;
21
22    let mut encoded = String::with_capacity(64);
23    for byte in bytes {
24        write!(&mut encoded, "{byte:02x}").expect("writing to String cannot fail");
25    }
26    encoded
27}

29pub(super) fn execute(config: &CliConfig) -> Result<RunSummary, String> {
30    let started = Instant::now();
31    let result =
32        execute_continuum(application_request(config)?).map_err(|error| error.to_string())?;
33    let minor_cycles = result.minor_cycles.clone();
34    let native = result.outcome.output;
35    let visibility_products = native.visibility_products.map(|completion| {
36        crate::task_contract::ImagerVisibilityProductDiagnostic {
37            problem_id: hex(completion.problem_id().as_bytes()),
38            final_model_generation: hex(completion.final_model().as_bytes()),
39            selected_generation: hex(completion.selected_generation().as_bytes()),
40            weighting_generation: hex(completion.weighting_generation().as_bytes()),
41            model_product: hex(completion.model_product().as_bytes()),
42            residual_product: hex(completion.residual_product().as_bytes()),
43            sample_count: completion.sample_count(),
44        }
45    });
46    let clean_stop_reason = result.minor_stop_reason.map(|reason| match reason {
47        ContinuumStopReason::ThresholdReached => CleanStopReason::GlobalThresholdReached,
48        ContinuumStopReason::IterationBound => CleanStopReason::IterationLimitReached,
49        ContinuumStopReason::StalenessBound => CleanStopReason::MajorCycleLimitReached,
50        ContinuumStopReason::MultiscaleDivergence => CleanStopReason::DivergenceDetected,
51    });
52    Ok(RunSummary {
53        warnings: Vec::new(),
54        gridded_samples: usize::try_from(native.scientific.normal_state().sample_count())
55            .map_err(|_| "native selected-sample count exceeds usize".to_string())?,
56        major_cycles: native.major_cycle_count,
57        minor_iterations: result.minor_iterations,
58        clean_stop_reason,
59        minor_cycles,
60        visibility_products,
61        elapsed: started.elapsed(),
62        output_products: result.product_names,
63    })
64}

66fn application_request(config: &CliConfig) -> Result<ContinuumImagingRequest, String> {
67    let maximum_model_update_jy = if config.dirty_only || config.niter == 0 {
68        1.0
69    } else {
70        config
71            .maximum_model_update_jy
72            .ok_or_else(|| "native deconvolution requires --maximum-model-update-jy".to_string())?
73    };
74    Ok(ContinuumImagingRequest {
75        measurement_set: config.ms.clone(),
76        image_name: config.imagename.clone(),
77        image_size: config.imsize,
78        cell_arcsec: config.cell_arcsec,
79        field_ids: config.field_ids.clone(),
80        uv_range: config.uvrange.clone(),
81        intent: config.intent.clone(),
82        data_description: config.ddid,
83        spectral_window: config
84            .spw_selector
85            .clone()
86            .or_else(|| config.spw.map(|spw| spw.to_string())),
87        channel_start: config.channel_start,
88        channel_count: config.channel_count,
89        data_column: config.datacolumn.clone(),
90        algorithm: if config.dirty_only || config.niter == 0 {
91            ContinuumAlgorithm::Dirty
92        } else {
93            match config.deconvolver {
94                Deconvolver::Hogbom => ContinuumAlgorithm::Hogbom,
95                Deconvolver::Clark => ContinuumAlgorithm::Clark,
96                Deconvolver::Multiscale => ContinuumAlgorithm::Multiscale {
97                    scales_px: config
98                        .multiscale_scales
99                        .iter()
100                        .copied()
101                        .map(f64::from)
102                        .collect(),
103                    small_scale_bias: f64::from(config.small_scale_bias),
104                },
105                Deconvolver::Mtmfs => ContinuumAlgorithm::Mtmfs {
106                    terms: config.nterms,
107                },
108            }
109        },
110        weighting: match config.weighting {
111            WeightingMode::Natural => ContinuumWeighting::Natural,
112            WeightingMode::Uniform => ContinuumWeighting::Uniform,
113            WeightingMode::Briggs { robust } => ContinuumWeighting::Briggs(f64::from(robust)),
114            WeightingMode::BriggsBwTaper { robust } => {
115                ContinuumWeighting::BriggsBandwidthTaper(f64::from(robust))
116            }
117        },
118        iterations: config.niter,
119        cycle_iterations: config.minor_cycle_length.min(config.niter.max(1)),
120        maximum_major_cycles: config.nmajor.unwrap_or(1),
121        maximum_model_update_jy: f64::from(maximum_model_update_jy),
122        noise_sigma: (config.nsigma > 0.0).then_some(f64::from(config.nsigma)),
123        cycle_factor: f64::from(config.cyclefactor),
124        minimum_psf_fraction: f64::from(config.min_psf_fraction),
125        maximum_psf_fraction: f64::from(config.max_psf_fraction),
126        gain: f64::from(config.gain),
127        threshold_jy: f64::from(config.threshold_jy),
128        psf_cutoff: config.psf_cutoff,
129        beam_policy: match config.restoring_beam_mode {
130            RestoringBeamMode::PerPlane => ContinuumBeamPolicy::PerPlane,
131            RestoringBeamMode::Common => ContinuumBeamPolicy::Common,
132        },
133        mask: match (&config.mask_image, config.use_mask) {
134            (Some(path), CleanMaskMode::User) => ContinuumMask::Image(path.clone()),
135            (_, CleanMaskMode::AutoMultiThreshold) => {
136                ContinuumMask::AutoMultithresh(ContinuumAutoMaskControls {
137                    sidelobe_factor: f64::from(config.auto_mask.sidelobe_threshold),
138                    noise_factor: f64::from(config.auto_mask.noise_threshold),
139                    low_noise_factor: f64::from(config.auto_mask.low_noise_threshold),
140                    negative_factor: f64::from(config.auto_mask.negative_threshold),
141                    minimum_beam_fraction: f64::from(config.auto_mask.min_beam_frac),
142                    smooth_factor: f64::from(config.auto_mask.smooth_factor),
143                    cut_threshold: f64::from(config.auto_mask.cut_threshold),
144                    grow_iterations: config.auto_mask.grow_iterations,
145                    minimum_percent_change: f64::from(config.auto_mask.min_percent_change),
146                })
147            }
148            (None, CleanMaskMode::User) if config.mask_boxes.is_empty() => ContinuumMask::FullPlane,
149            (None, CleanMaskMode::User) => ContinuumMask::Boxes(
150                config
151                    .mask_boxes
152                    .iter()
153                    .map(|region| ContinuumMaskBox {
154                        blc: [region[0], region[1]],
155                        trc: [region[2], region[3]],
156                    })
157                    .collect(),
158            ),
159        },
160        save_model_column: config.save_model == SaveModelMode::ModelColumn,
161        task_requirements: task_requirements(config),
162    })
163}

165fn task_requirements(config: &CliConfig) -> Vec<TaskRequirement> {
166    let mut requirements = vec![match config.spectral_mode {
167        SpectralMode::Mfs => TaskRequirement::SerialCpu,
168        SpectralMode::Cube => TaskRequirement::SpectralCube,
169        SpectralMode::Cubedata => TaskRequirement::SpectralCubedata,
170    }];
171    if config.aw_project.is_some() {
172        requirements.push(TaskRequirement::AwProjection);
173    } else if config.w_term_mode == WTermMode::WProject {
174        requirements.push(TaskRequirement::WProjection);
175    } else if config.use_pointing && !config.force_standard_gridder {
176        requirements.push(TaskRequirement::MosaicGridder);
177    }
178    if config.outlier_file.is_some() {
179        requirements.push(TaskRequirement::FacetsOutliers);
180    }
181    if config.use_mask == CleanMaskMode::AutoMultiThreshold {
182        requirements.push(TaskRequirement::Automasking);
183    }
184    if config.use_mask == CleanMaskMode::AutoMultiThreshold
185        || !config.mask_boxes.is_empty()
186        || config.mask_image.is_some()
187    {
188        requirements.push(TaskRequirement::MaskProduct);
189    }
190    if config.start_model.is_some() {
191        requirements.push(TaskRequirement::StartModel);
192    }
193    if config.save_model != SaveModelMode::None {
194        requirements.push(TaskRequirement::ModelColumnWrite);
195    }
196    requirements.extend(backend_requirements(config));
197    if unsupported_native_controls(config) {
198        requirements.push(TaskRequirement::UnsupportedControls);
199    }
200    requirements
201}

203fn backend_requirements(config: &CliConfig) -> Vec<TaskRequirement> {
204    let mut requirements = Vec::new();
205    match config.standard_mfs_acceleration {
206        StandardMfsAccelerationPolicy::Auto => {
207            requirements.push(TaskRequirement::ExecutionAuto);
208        }
209        StandardMfsAccelerationPolicy::Cpu => {}
210        StandardMfsAccelerationPolicy::MultiCpu => {
211            requirements.push(TaskRequirement::FixedTileCpu);
212        }
213        StandardMfsAccelerationPolicy::Metal => {
214            requirements.push(TaskRequirement::MetalRowRunGroupedGridder);
215        }
216    }
217    if let Some(backend) = config.standard_mfs_backend.as_deref() {
218        requirements.push(match backend {
219            "cpu" | "serial" | "serial-cpu" => TaskRequirement::SerialCpu,
220            "fixed-tile" | "fixed-tile-cpu" => TaskRequirement::FixedTileCpu,
221            "metal" | "metal-gridder" => TaskRequirement::MetalGridder,
222            "metal-row-run" | "metal-row-run-gridder" => TaskRequirement::MetalRowRunGridder,
223            "metal-row-run-grouped" | "metal-row-run-grouped-gridder" => {
224                TaskRequirement::MetalRowRunGroupedGridder
225            }
226            _ => TaskRequirement::UnsupportedControls,
227        });
228    }
229    match config.imaging_fft_backend {
230        ImagingFftBackendPolicy::RustFft => {}
231        ImagingFftBackendPolicy::Auto => requirements.push(TaskRequirement::FftAuto),
232        ImagingFftBackendPolicy::Accelerate => requirements.push(TaskRequirement::Accelerate),
233        ImagingFftBackendPolicy::MetalMpsGraph => requirements.push(TaskRequirement::MetalMpsGraph),
234        ImagingFftBackendPolicy::Fftw => requirements.push(TaskRequirement::Fftw),
235    }
236    requirements
237}

239fn unsupported_native_controls(config: &CliConfig) -> bool {
240    config.phasecenter_field.is_some()
241        || config.phasecenter.is_some()
242        || config
243            .correlation
244            .as_deref()
245            .is_some_and(|plane| !plane.eq_ignore_ascii_case("I"))
246        || config.uv_taper.is_some()
247        || config.hogbom_iteration_mode != super::HogbomIterationMode::Strict
248        || config.fullsummary
249        || config.pbcor
250        || config.write_pb
251        || config.chanchunks.is_some()
252        || config.per_channel_weight_density
253        || config.w_project_planes.is_some()
254        || config.standard_mfs_grid_threads.is_some()
255        || config.standard_mfs_tile_anchor.is_some()
256        || config.standard_mfs_residual_backend.is_some()
257        || config.standard_mfs_initial_dirty_backend.is_some()
258        || config.standard_mfs_metal_minor_cycle_chunk.is_some()
259        || config.standard_mfs_metal_grouped_input_cache.is_some()
260        || config.standard_mfs_memory_target_mb.is_some()
261        || config.standard_mfs_prepare_buffer_mb.is_some()
262        || config.imaging_memory_target_mb.is_some()
263        || config.imaging_memory_pressure_policy != ImagingMemoryPressurePolicy::Auto
264        || config.imaging_prepare_buffer_mb.is_some()
265        || config.imaging_row_block_rows.is_some()
266        || config.imaging_prepare_workers.is_some()
267        || config.imaging_read_ahead_blocks.is_some()
268        || config.imaging_fft_precision != ImagingFftPrecisionPolicy::Auto
269        || config.write_preview_pngs
270}
