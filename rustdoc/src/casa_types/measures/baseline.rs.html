2//! Interferometer baselines to uvw coordinates, as casacore computes them.
3//!
4//! [`UvwFrame`] is casacore's `MBaseline` conversion from ITRF to J2000
5//! followed by the `MVuvw` projection on a J2000 phase centre: the chain
6//! `NewMSSimulator::calcAntUVW` and `MSUVWGenerator` use to give each antenna
7//! its uvw. The conversion routes the baseline direction through HADEC, TOPO,
8//! APP and JNAT, applying Earth rotation (local apparent sidereal time),
9//! polar motion, diurnal aberration, precession and nutation (IAU 1976/1980),
10//! annual aberration and solar light deflection. The baseline's length is
11//! unchanged.
12//!
13//! An antenna's uvw is that of the baseline from the observatory to the
14//! antenna (observatory minus antenna, casacore's `MVBaseline(first, second)`
15//! convention). A row's uvw is the second antenna's minus the first's, so the
16//! observatory term cancels.
17//!
18//! The epoch is the caller's. casacore's simulator reads a row's stored time
19//! as UT1; this module converts whatever epoch it is given.
20

30/// The ITRF-to-uvw rotation of interferometer baselines at one epoch, for one
31/// observatory and one J2000 phase centre.
32///
33/// Building a frame does the epoch-dependent work once; [`UvwFrame::baseline_uvw`]
34/// then rotates and projects each baseline.
35#[derive(Debug, Clone)]
36pub struct UvwFrame {
37    phase_center: MDirection,
38    observatory_longitude_sin_cos: (f64, f64),
39    polar_motion_rotation: [[f64; 3]; 3],
40    diurnal_aberration_rotation: [[f64; 3]; 3],
41    precession_nutation_inverse: [[f64; 3]; 3],
42    inverse_aberration_rotation: [[f64; 3]; 3],
43    inverse_solar_deflection_rotation: [[f64; 3]; 3],
44}
45
46impl UvwFrame {
47    /// The frame at `epoch` for baselines at `observatory`, projected on the
48    /// J2000 direction `phase_center_rad` (longitude, latitude in radians).
49    ///
50    /// # Errors
51    ///
52    /// An epoch or direction conversion the measures data cannot make, or a
53    /// failed ephemeris lookup.
54    pub fn new(
55        phase_center_rad: [f64; 2],
56        epoch: MEpoch,
57        observatory: &MPosition,
58        measures: Arc<dyn MeasuresProvider>,
59    ) -> Result<Self, MeasureError> {
60        let phase_center = MDirection::from_angles(
61            phase_center_rad[0],
62            phase_center_rad[1],
63            DirectionRef::J2000,
64        );
65        let frame = MeasFrame::new()
66            .with_epoch(epoch)
67            .with_position(observatory.clone())
68            .with_direction(phase_center.clone())
69            .with_measures(measures);
70        let longitude = observatory.longitude_rad();
71        let observatory_longitude_sin_cos = longitude.sin_cos();
72        let last = local_apparent_sidereal_time(&frame)?;
73        let tdb_mjd = epoch_mjd(&frame, EpochRef::TDB)?;
74        let (xp, yp) = polar_motion_rad(&frame, tdb_mjd)?;
75        let polar_motion_rotation = polar_motion_euler(-xp, -yp, last);
76        let radius = vector_norm(observatory.as_itrf());
77        let v_c = diurnal_aberration_factor(radius);
78        let aberration_direction =
79            spherical_to_cartesian(last, observatory.geocentric_latitude_rad());
80        let diurnal_aberration_shift = scale_vector(aberration_direction, -v_c);
81        let app_phase_center = phase_center.convert_to(DirectionRef::APP, &frame)?;
82        let tt = epoch_jd_pair(&frame, EpochRef::TT)?;
83        let nutation = sofars::pnp::nutm80(tt.0, tt.1);
84        let precession = sofars::pnp::pmat76(tt.0, tt.1);
85        let precession_nutation_inverse =
86            precession_nutation_inverse_matrix(&nutation, &precession);
87        let phase_center_cosines = phase_center.cosines();
88        let inverse_aberration_rotation = rotate_shift_matrix(
89            inverse_aberration_shift(&phase_center, &frame)?,
90            phase_center_cosines,
91        );
92        let inverse_solar_deflection_rotation = rotate_shift_matrix(
93            inverse_solar_deflection_shift(&phase_center, &frame)?,
94            phase_center_cosines,
95        );
96
97        Ok(Self {
98            phase_center,
99            observatory_longitude_sin_cos,
100            polar_motion_rotation,
101            diurnal_aberration_rotation: rotate_shift_matrix(
102                diurnal_aberration_shift,
103                app_phase_center.cosines(),
104            ),
105            precession_nutation_inverse,
106            inverse_aberration_rotation,
107            inverse_solar_deflection_rotation,
108        })
109    }
110
111    /// The uvw, in metres, of the ITRF baseline vector `baseline_itrf_m`
112    /// (first position minus second): its J2000 form projected on the phase
113    /// centre, with w toward the source.
114    pub fn baseline_uvw(&self, baseline_itrf_m: [f64; 3]) -> [f64; 3] {
115        let baseline_len = vector_norm(baseline_itrf_m);
116        if baseline_len == 0.0 {
117            return [0.0, 0.0, 0.0];
118        }
119
120        let mut unit = scale_vector(baseline_itrf_m, 1.0 / baseline_len);
121        unit = self.itrf_to_hadec(unit);
122        unit = self.hadec_to_topo(unit);
123        unit = rotate(&self.precession_nutation_inverse, &unit);
124        unit = rotate(&self.inverse_aberration_rotation, &unit);
125        unit = rotate(&self.inverse_solar_deflection_rotation, &unit);
126        j2000_baseline_uvw(scale_vector(unit, baseline_len), &self.phase_center)
127    }
128
129    fn itrf_to_hadec(&self, vector: [f64; 3]) -> [f64; 3] {
130        let (s, c) = self.observatory_longitude_sin_cos;
131        let negated_y = -vector[1];
132        [
133            c * vector[0] - s * negated_y,
134            s * vector[0] + c * negated_y,
135            vector[2],
136        ]
137    }
138
139    fn hadec_to_topo(&self, vector: [f64; 3]) -> [f64; 3] {
140        let vector = rotate(
141            &self.polar_motion_rotation,
142            &[vector[0], -vector[1], vector[2]],
143        );
144        rotate(&self.diurnal_aberration_rotation, &vector)
145    }
146}

148/// casacore's `MVuvw` projection: the uvw, in metres, of the J2000 baseline
149/// vector `baseline_j2000_m` on the J2000 direction `phase_center`, with u
150/// east, v north and w toward the source.
151pub fn j2000_baseline_uvw(baseline_j2000_m: [f64; 3], phase_center: &MDirection) -> [f64; 3] {
152    let (ra, dec) = phase_center.as_angles();
153    let (sin_ra, cos_ra) = ra.sin_cos();
154    let (sin_dec, cos_dec) = dec.sin_cos();
155    let [x, y, z] = baseline_j2000_m;
156
157    [
158        -(sin_ra * x - cos_ra * y),
159        -sin_dec * cos_ra * x - sin_dec * sin_ra * y + cos_dec * z,
160        cos_dec * cos_ra * x + cos_dec * sin_ra * y + sin_dec * z,
161    ]
162}

164fn ephemeris_error(what: &str) -> MeasureError {
165    MeasureError::ModelError {
166        model: "sofars epv00",
167        reason: format!("{what} ephemeris lookup failed"),
168    }
169}

171fn inverse_aberration_shift(
172    phase_center: &MDirection,
173    frame: &MeasFrame,
174) -> Result<[f64; 3], MeasureError> {
175    const C_AU_PER_DAY: f64 = 173.144_632_674_240_34;
176
177    let tt = epoch_jd_pair(frame, EpochRef::TT)?;
178    let (_, earth_bary) =
179        sofars::eph::epv00(tt.0, tt.1).ok_or_else(|| ephemeris_error("aberration"))?;
180    let velocity = scale_vector(earth_bary[1], 1.0 / C_AU_PER_DAY);
181    let beta_inv = (1.0 - dot(velocity, velocity)).sqrt();
182    let source = phase_center.cosines();
183    let mut solution = subtract_vectors(source, velocity);
184
185    loop {
186        let dot_sv = dot(solution, velocity);
187        let mut trial = scale_vector(
188            add_vectors(
189                scale_vector(solution, beta_inv),
190                scale_vector(velocity, 1.0 + dot_sv / (1.0 + beta_inv)),
191            ),
192            1.0 / (1.0 + dot_sv),
193        );
194        trial = normalize(trial);
195        let residual = subtract_vectors(trial, source);
196        if vector_norm(residual) <= 1.0e-10 {
197            break;
198        }
199        for idx in 0..3 {
200            let component = velocity[idx];
201            solution[idx] -= residual[idx]
202                / (((beta_inv + component * component / (1.0 + beta_inv))
203                    - component * trial[idx])
204                    / (1.0 + dot_sv));
205        }
206    }
207
208    Ok(subtract_vectors(solution, source))
209}

211fn inverse_solar_deflection_shift(
212    phase_center: &MDirection,
213    frame: &MeasFrame,
214) -> Result<[f64; 3], MeasureError> {
215    let tt = epoch_jd_pair(frame, EpochRef::TT)?;
216    let (earth_helio, _) =
217        sofars::eph::epv00(tt.0, tt.1).ok_or_else(|| ephemeris_error("solar-position"))?;
218    let sun_vector = [-earth_helio[0][0], -earth_helio[0][1], -earth_helio[0][2]];
219    let sun_distance = vector_norm(sun_vector);
220    let sun = scale_vector(sun_vector, 1.0 / sun_distance);
221    let source = phase_center.cosines();
222    let mut dot_solution_sun = dot(source, sun);
223    let strength = -1.974e-8 / sun_distance;
224    let mut solution = source;
225
226    loop {
227        let correction = scale_vector(
228            subtract_vectors(sun, scale_vector(solution, dot_solution_sun)),
229            strength / (1.0 - dot_solution_sun),
230        );
231        for idx in 0..3 {
232            let component = sun[idx];
233            solution[idx] -= (correction[idx] + solution[idx] - source[idx])
234                / (1.0
235                    + (component * correction[idx]
236                        - strength * (dot_solution_sun + component * solution[idx]))
237                        / (1.0 - dot_solution_sun));
238        }
239        dot_solution_sun = dot(solution, sun);
240        let residual = add_vectors(correction, subtract_vectors(solution, source));
241        if vector_norm(residual) <= 1.0e-10 {
242            break;
243        }
244    }
245
246    Ok(subtract_vectors(solution, source))
247}

249/// The local apparent sidereal time, in radians and not reduced to a turn, at
250/// the frame's epoch and position, as casacore's UT1-to-LAST conversion gives
251/// it: GMST1 plus the equation of the equinoxes of its legacy `STANDARD`
252/// nutation, plus the position's longitude.
253///
254/// # Errors
255///
256/// A frame without an epoch or a position, or an epoch it cannot convert to
257/// UT1.
258pub fn local_apparent_sidereal_time(frame: &MeasFrame) -> Result<f64, MeasureError> {
259    let (ut1_a, ut1_b) = epoch_jd_pair(frame, EpochRef::UT1)?;
260    let ut1_mjd = epoch_mjd(frame, EpochRef::UT1)?;
261    let centuries = (ut1_mjd - 51_544.5) / 36_525.0;
262    let gmst0_seconds =
263        24_110.548_41 + 8_640_184.812_866 * centuries + 0.093_104 * centuries * centuries
264            - 6.2e-6 * centuries * centuries * centuries;
265    let gmst0_turns = (ut1_mjd + gmst0_seconds / 86_400.0 + 6_713.0).fract();
266    let equation_of_equinoxes =
267        sofars::vm::anpm(sofars::erst::gst94(ut1_a, ut1_b) - sofars::erst::gmst82(ut1_a, ut1_b));
268    // Matches casacore's legacy `Nutation::STANDARD` equation of equinoxes
269    // used by `MCEpoch::UT1_GAST`; SOFA `gst94` differs by about 0.1 ms.
270    const CASACORE_STANDARD_NUTATION_GAST_OFFSET_RAD: f64 = 7.719e-9;
271    let gast = gmst0_turns * std::f64::consts::TAU
272        + equation_of_equinoxes
273        + CASACORE_STANDARD_NUTATION_GAST_OFFSET_RAD;
274    let position = frame
275        .position()
276        .ok_or(MeasureError::MissingFrameData { what: "position" })?;
277    Ok(gast + position.longitude_rad())
278}

294fn polar_motion_rad(frame: &MeasFrame, epoch_mjd: f64) -> Result<(f64, f64), MeasureError> {
295    const ARCSEC_TO_RAD: f64 = std::f64::consts::PI / (180.0 * 3600.0);
296    match frame.polar_motion_for_mjd(epoch_mjd)? {
297        Some((xp_arcsec, yp_arcsec)) => Ok((xp_arcsec * ARCSEC_TO_RAD, yp_arcsec * ARCSEC_TO_RAD)),
298        None => Ok((0.0, 0.0)),
299    }
300}

302fn polar_motion_euler(xp: f64, yp: f64, last: f64) -> [[f64; 3]; 3] {
303    let mut matrix = [[0.0; 3]; 3];
304    for (idx, row) in matrix.iter_mut().enumerate() {
305        row[idx] = 1.0;
306    }
307    casacore_apply_single_euler(&mut matrix, xp, 2);
308    casacore_apply_single_euler(&mut matrix, yp, 1);
309    casacore_apply_single_euler(&mut matrix, last, 3);
310    matrix
311}
312
313fn casacore_apply_single_euler(matrix: &mut [[f64; 3]; 3], angle: f64, axis: usize) {
314    if angle == 0.0 || axis == 0 {
315        return;
316    }
317    let mut single = [[0.0; 3]; 3];
318    for (idx, row) in single.iter_mut().enumerate() {
319        row[idx] = 1.0;
320    }
321    let i = axis % 3;
322    let j = (i + 1) % 3;
323    let (sin_angle, cos_angle) = angle.sin_cos();
324    single[i][i] = cos_angle;
325    single[j][j] = cos_angle;
326    single[i][j] = -sin_angle;
327    single[j][i] = sin_angle;
328
329    let original = *matrix;
330    for row in 0..3 {
331        for col in 0..3 {
332            matrix[row][col] = original[row][0] * single[0][col]
333                + original[row][1] * single[1][col]
334                + original[row][2] * single[2][col];
335        }
336    }
337}

339fn diurnal_aberration_factor(radius_m: f64) -> f64 {
340    const C: f64 = 299_792_458.0;
341    const SIDEREAL_RATIO: f64 = 1.002_737_909_35;
342    (2.0 * std::f64::consts::PI * radius_m) / 86_400.0 * SIDEREAL_RATIO / C
343}

345fn rotate_shift_matrix(shift: [f64; 3], reference_direction: [f64; 3]) -> [[f64; 3]; 3] {
346    let reference = normalize(reference_direction);
347    let reference_long = reference[1].atan2(reference[0]);
348    let reference_lat = reference[2].asin();
349
350    let mut rot = [[0.0; 3]; 3];
351    for (idx, row) in rot.iter_mut().enumerate() {
352        row[idx] = 1.0;
353    }
354    casacore_apply_single_euler(&mut rot, -std::f64::consts::FRAC_PI_2 + reference_lat, 2);
355    casacore_apply_single_euler(&mut rot, -reference_long, 3);
356
357    let shifted_once = rotate(&rot, &shift);
358    let shifted_long = shifted_once[1].atan2(shifted_once[0]);
359    let mut long_rot = [[0.0; 3]; 3];
360    for (idx, row) in long_rot.iter_mut().enumerate() {
361        row[idx] = 1.0;
362    }
363    casacore_apply_single_euler(&mut long_rot, -shifted_long, 3);
364    rot = multiply_matrices(&long_rot, &rot);
365
366    let shifted_twice = rotate(&rot, &shift);
367    let mut correction_rot = [[0.0; 3]; 3];
368    for (idx, row) in correction_rot.iter_mut().enumerate() {
369        row[idx] = 1.0;
370    }
371    casacore_apply_single_euler(&mut correction_rot, shifted_twice[0], 2);
372    let corrected = multiply_matrices(&correction_rot, &rot);
373    multiply_transpose_left(&rot, &corrected)
374}

376fn precession_nutation_inverse_matrix(
377    nutation: &[[f64; 3]; 3],
378    precession: &[[f64; 3]; 3],
379) -> [[f64; 3]; 3] {
380    let mut result = [[0.0; 3]; 3];
381    for col in 0..3 {
382        let mut basis = [0.0; 3];
383        basis[col] = 1.0;
384        let transformed = rotate_t(precession, &rotate_t(nutation, &basis));
385        for row in 0..3 {
386            result[row][col] = transformed[row];
387        }
388    }
389    result
390}
391
392fn rotate(matrix: &[[f64; 3]; 3], vector: &[f64; 3]) -> [f64; 3] {
393    [
394        matrix[0][0] * vector[0] + matrix[0][1] * vector[1] + matrix[0][2] * vector[2],
395        matrix[1][0] * vector[0] + matrix[1][1] * vector[1] + matrix[1][2] * vector[2],
396        matrix[2][0] * vector[0] + matrix[2][1] * vector[1] + matrix[2][2] * vector[2],
397    ]
398}
399
400fn rotate_t(matrix: &[[f64; 3]; 3], vector: &[f64; 3]) -> [f64; 3] {
401    [
402        matrix[0][0] * vector[0] + matrix[1][0] * vector[1] + matrix[2][0] * vector[2],
403        matrix[0][1] * vector[0] + matrix[1][1] * vector[1] + matrix[2][1] * vector[2],
404        matrix[0][2] * vector[0] + matrix[1][2] * vector[1] + matrix[2][2] * vector[2],
405    ]
406}

408fn multiply_matrices(left: &[[f64; 3]; 3], right: &[[f64; 3]; 3]) -> [[f64; 3]; 3] {
409    let mut result = [[0.0; 3]; 3];
410    for row in 0..3 {
411        for col in 0..3 {
412            result[row][col] = left[row][0] * right[0][col]
413                + left[row][1] * right[1][col]
414                + left[row][2] * right[2][col];
415        }
416    }
417    result
418}
419
420fn multiply_transpose_left(left: &[[f64; 3]; 3], right: &[[f64; 3]; 3]) -> [[f64; 3]; 3] {
421    let mut result = [[0.0; 3]; 3];
422    for row in 0..3 {
423        for col in 0..3 {
424            result[row][col] = left[0][row] * right[0][col]
425                + left[1][row] * right[1][col]
426                + left[2][row] * right[2][col];
427        }
428    }
429    result
430}

432fn spherical_to_cartesian(lon: f64, lat: f64) -> [f64; 3] {
433    let (sin_lon, cos_lon) = lon.sin_cos();
434    let (sin_lat, cos_lat) = lat.sin_cos();
435    [cos_lat * cos_lon, cos_lat * sin_lon, sin_lat]
436}
437
438fn vector_norm(vector: [f64; 3]) -> f64 {
439    dot(vector, vector).sqrt()
440}
441
442fn normalize(vector: [f64; 3]) -> [f64; 3] {
443    let norm = vector_norm(vector);
444    if norm == 0.0 {
445        vector
446    } else {
447        scale_vector(vector, 1.0 / norm)
448    }
449}
450
451fn dot(left: [f64; 3], right: [f64; 3]) -> f64 {
452    left[0] * right[0] + left[1] * right[1] + left[2] * right[2]
453}
454
455fn add_vectors(left: [f64; 3], right: [f64; 3]) -> [f64; 3] {
456    [left[0] + right[0], left[1] + right[1], left[2] + right[2]]
457}
458
459fn subtract_vectors(left: [f64; 3], right: [f64; 3]) -> [f64; 3] {
460    [left[0] - right[0], left[1] - right[1], left[2] - right[2]]
461}
462
463fn scale_vector(vector: [f64; 3], scale: f64) -> [f64; 3] {
464    [vector[0] * scale, vector[1] * scale, vector[2] * scale]
465}
