3//! Stream a cube's published normals into CASA's two-term Taylor images.
4
5use crate::{
6    ContinuumProductInputs, PlannedContinuumGeneration, ProductsError, TaylorMomentAccumulator,
7    TaylorPrimaryBeamAccumulator, read_mvc_cube_plane, read_mvc_cube_primary_beam,
8};
9
10/// Reduce a cube normal state into CASA MVC's two-term Taylor products.
11///
12/// The caller owns and admits every output plane and each channel's transient
13/// workspace. `mean_pb` is one x-major plane, `psf` three contiguous planes,
14/// and `residual` two. The first pass reads only one PB and weight at a time;
15/// the second reads one PSF, PB and corrected residual at a time. No cube-wide
16/// image cache is created. `reference_hz` is the Taylor expansion reference,
17/// which is independent of the Taylor image WCS.
18///
19/// # Errors
20///
21/// Invalid shapes or data, absent cube products, failed bounded reads, or
22/// non-finite arithmetic. A failure leaves partial outputs to be discarded.
23pub fn reduce_mvc_cube(
24    planned: &PlannedContinuumGeneration,
25    inputs: &ContinuumProductInputs<'_>,
26    reference_hz: f64,
27    mean_pb: &mut [f32],
28    psf: &mut [f32],
29    residual: &mut [f32],
30    sumwt: &mut [f32; 3],
31) -> Result<(), ProductsError> {
32    let [width, height] = inputs.normal_state().shape();
33    let cells = width
34        .checked_mul(height)
35        .ok_or(ProductsError::InvalidTaylorMomentReduction)?;
36    if cells == 0
37        || mean_pb.len() != cells
38        || cells.checked_mul(3) != Some(psf.len())
39        || cells.checked_mul(2) != Some(residual.len())
40        || !reference_hz.is_finite()
41        || reference_hz <= 0.0
42    {
43        return Err(ProductsError::InvalidTaylorMomentReduction);
44    }
45    let channels = inputs.normal_state().channel_count();
46    let mut beams = TaylorPrimaryBeamAccumulator::new(mean_pb)?;
47    for channel in 0..channels {
48        let beam = read_mvc_cube_primary_beam(planned, inputs, channel)?;
49        beams.add(&beam.pixels, beam.sum_weight)?;
50    }
51    let mean_pb = beams.finish()?;
52    let corrected = inputs.clone().with_cube_taylor_primary_beam(mean_pb)?;
53    let cutoff = inputs
54        .problem()
55        .products()
56        .validity()
57        .primary_beam()
58        .cutoff();
59    let mut moments =
60        TaylorMomentAccumulator::new(mean_pb, psf, residual, sumwt, reference_hz, cutoff)?;
61    for channel in 0..channels {
62        let plane = read_mvc_cube_plane(planned, &corrected, channel)?;
63        moments.add(
64            &plane.psf,
65            &plane.residual,
66            &plane.primary_beam,
67            plane.sum_weight,
68            plane.frequency_hz,
69        )?;
70    }
71    moments.finish()
72}
