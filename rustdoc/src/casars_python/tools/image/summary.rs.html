22fn dimension_list(values: &[usize]) -> CasaValue {
23    CasaValue::Ints {
24        shape: vec![values.len()],
25        values: values.iter().map(|&value| value as i64).collect(),
26    }
27}
28
29fn text_list(values: Vec<String>) -> CasaValue {
30    if values.iter().all(String::is_empty) {
31        CasaValue::TextArray {
32            shape: vec![values.len()],
33            values,
34        }
35    } else {
36        CasaValue::Texts(values)
37    }
38}

40fn mask_description(masks: &[String], default: &str, has_mask: bool) -> String {
41    match masks {
42        [] if has_mask => "Parent is masked".into(),
43        [] => "None".into(),
44        [only] if only == default => default.into(),
45        _ => {
46            let other = masks
47                .iter()
48                .filter(|name| name.as_str() != default)
49                .cloned()
50                .collect::<Vec<_>>()
51                .join(", ");
52            format!("{default} [{other}]")
53        }
54    }
55}

57fn describe(
58    image: &ImageHandle,
59    beam_view: &BeamView,
60    doppler: DopplerRef,
61    list: bool,
62    pixel_order: bool,
63    verbose: bool,
64    logs: &mut Vec<LogLine>,
65) -> PyResult<CasaValue> {
66    let shape = image.shape();
67    let tile = if let Some(fits) = image.fits() {
68        fits.tile_shape.clone()
69    } else {
70        with_pixels!(image, storage => match storage {
71            Storage::Paged(image) => image.tile_shape().to_vec(),
72            Storage::Temp { image, .. } if image.is_in_memory() => image.shape().to_vec(),
73            Storage::Temp { image, .. } => {
74                let lattice: &dyn ImageInterface<_> = image.as_ref();
75                lattice.nice_cursor_shape()
76            }
77        })
78    };
79    let (has_mask, default, masks, units) = with_pixels!(image, storage =>
80    with_storage!(storage, image => (
81        image.has_pixel_mask(),
82        image.default_mask_name().unwrap_or_default(),
83        image.mask_names(),
84        image.units().to_owned(),
85    )));
86    let has_mask = image.fits().map_or(has_mask, |fits| fits.has_mask);
87    let regions = with_pixels!(image, storage => match storage {
88        Storage::Paged(image) => image.region_names(),
89        Storage::Temp { .. } => Vec::new(),
90    });
91    let name = image
92        .path()
93        .and_then(|path| path.file_name())
94        .map_or_else(|| "Temporary_Image".into(), |name| name.to_string_lossy());
95    let kind = image.image_type_name();
96    let info = image_info(image).map_err(raised)?;
97    let image_type = match info.image_type {
98        ImageType::Undefined => ImageType::Intensity,
99        other => other,
100    }
101    .to_string();
102    let masks_text = mask_description(&masks, &default, has_mask);
103    let regions_text = if regions.is_empty() {
104        "None".into()
105    } else {
106        regions.join(", ")
107    };
108    let mut header = format!(
109        "\nImage name       : {name}\nObject name      : {}\nImage type       : {kind}\nImage quantity   : {image_type}\nPixel mask(s)    : {masks_text}\nRegion(s)        : {regions_text}",
110        info.object_name
111    );
112    if !units.is_empty() {
113        write!(header, "\nImage units      : {units}").unwrap();
114    }
115    let record = image.coordinate_record();
116    let coords = CoordinateSystem::from_record(&record).map_err(raised)?;
117    let mut fields = coordinates::fields(&record, pixel_order)?;
118    let beams = beam_view.beam_set().map_err(raised)?;
119    if let Some(beam) = beams.single_beam() {
120        let width_unit = if beam.minor_in("deg").map_err(raised)? < 1.0 {
121            "arcsec"
122        } else {
123            "deg"
124        };
125        let major = format_stream_value(beam.major_in(width_unit).map_err(raised)?);
126        let minor = format_stream_value(beam.minor_in(width_unit).map_err(raised)?);
127        let pa = format_stream_value(plane_beams::normalized_angle(&beam)?);
128        write!(
129            header,
130            "\nRestoring Beam   : {major} {width_unit}, {minor} {width_unit}, {pa} deg"
131        )
132        .unwrap();
133        fields.push((
134            "restoringbeam",
135            restoring_beam(beam_view, || record.clone(), -1, -1, "list")?,
136        ));
137    } else if beams.is_multi() {
138        plane_beams::append(&mut header, &record, &coords, &beams, list && verbose)?;
139        fields.push((
140            "perplanebeams",
141            restoring_beam(beam_view, || record.clone(), -1, -1, "list")?,
142        ));
143    }
144
145    let mut messages = vec![
146        header,
147        coordinates::report(&record, &coords, shape, &tile, doppler)?,
148    ];
149    if list {
150        logs.extend(messages.drain(..).map(LogLine::info));
151    }
152    fields.extend([
153        ("messages", text_list(messages)),
154        ("ndim", CasaValue::Int(shape.len() as i64)),
155        ("shape", dimension_list(shape)),
156        ("tileshape", dimension_list(&tile)),
157        ("unit", CasaValue::Text(units)),
158        ("hasmask", CasaValue::Bool(has_mask)),
159        ("defaultmask", CasaValue::Text(default)),
160        ("masks", text_list(masks)),
161        ("imagetype", CasaValue::Text(image_type)),
162    ]);
163    Ok(CasaValue::record(fields))
164}

166#[pymethods]
167impl PyImageTool {
168    #[pyo3(signature = (doppler="RADIO", list=true, pixelorder=true, verbose=false), text_signature = "($self, doppler='RADIO', list=True, pixelorder=True, verbose=False)")]
169    fn summary(
170        &self,
171        py: Python<'_>,
172        doppler: &str,
173        list: bool,
174        pixelorder: bool,
175        verbose: bool,
176    ) -> PyResult<Py<PyAny>> {
177        let mut logs = Vec::new();
178        let result = py.detach(|| {
179            let state = self.state();
180            let Some(attachment) = &state.image else {
181                return Ok(CasaValue::empty_record());
182            };
183            let doppler = if doppler.eq_ignore_ascii_case("TRUE") {
184                DopplerRef::BETA
185            } else {
186                doppler.parse().unwrap_or_else(|_| {
187                    logs.push(LogLine::warning("Illegal velocity type, using RADIO"));
188                    DopplerRef::RADIO
189                })
190            };
191            let mut image = attachment.lock();
192            attachment.require_lock(&mut image, casa_tables::LockType::Read)?;
193            describe(
194                &image,
195                &attachment.beam_view(),
196                doppler,
197                list,
198                pixelorder,
199                verbose,
200                &mut logs,
201            )
202        });
203        emit(py, "casars.tools.image", &logs)?;
204        result?.into_python(py)
205    }
206}
