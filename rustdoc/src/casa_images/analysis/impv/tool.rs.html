21/// A position-velocity image and its CASA axis-map record.
22pub struct ImagePvProduct {
23    /// Masked Float image with offset and surviving non-direction axes.
24    pub image: TempImage<f32>,
25    /// Native axis maps for both temporary attachments and on-disk products.
26    pub coordinate_record: RecordValue,
27    /// Channel and polarization indices for retaining the original beam records.
28    pub beam_planes: [Vec<usize>; 2],
29}

31/// Create a position-velocity image from an attached Float image.
32///
33/// # Errors
34/// Invalid endpoints, unit, region, width, mask, or failed image I/O.
35pub fn generate_image_pv(
36    image: &dyn ImageInterface<f32>,
37    native: RecordValue,
38    beam_record: &RecordValue,
39    request: &ImpvRequest,
40    region: Option<SubimageRegion<'_>>,
41) -> Result<ImagePvProduct, ImageError> {
42    let plane = direction_plane_axes(image)?;
43    let direction = direction(image.coordinates())?;
44    let width = position::width(&request.width, direction)?;
45    let endpoints = position::endpoints(request, direction, false)?;
46    let strip = geometry::Strip::new(
47        direction,
48        endpoints,
49        plane.map(|axis| image.shape()[axis]),
50        width,
51        true,
52    )?;
53    let selection = resolve_selection(image, request, plane, region, true)?;
54    if selection
55        .stride
56        .as_ref()
57        .is_some_and(|stride| plane.iter().any(|&axis| stride[axis] != 1))
58    {
59        return Err(ImageError::CasaSelection(
60            "You are not permitted to make a region selection in the direction coordinate".into(),
61        ));
62    }
63    let (coordinates, coordinate_record, secondary) = output_coordinates(
64        image,
65        Some(native),
66        &selection,
67        plane,
68        &strip,
69        &request.unit,
70    )?;
71    let mut info = image.image_info()?;
72    let packed = selection.packed_shape();
73    let beams = ImageInfo::from_record(beam_record)?.beam_set;
74    let beam_planes = beam_section_indices(
75        &beams,
76        image.coordinates(),
77        image.ndim(),
78        &selection.start,
79        &packed,
80        selection.stride.as_deref(),
81    );
82    info.beam_set = if beams.is_multi() {
83        beams.subset(&beam_planes[0], &beam_planes[1])?
84    } else {
85        beams
86    };
87    let mut misc = image.misc_info();
88    misc.upsert("secondary_coordinates", Value::Record(secondary));
89    let [x, y] = plane;
90    let mut shape = packed.clone();
91    shape[x] = strip.shape[0];
92    shape.remove(y);
93    let mut result = TempImage::<f32>::new(
94        shape,
95        coordinates,
96        TempStoragePolicy::Auto {
97            memory_budget_bytes: 64 * 1024 * 1024,
98            scratch: ScratchSpace::SystemTemp,
99        },
100    )?;
101    result.set_units(image.units())?;
102    result.set_image_info(&info)?;
103    result.set_misc_info(misc)?;
104
105    let ones = vec![1; packed.len()];
106    let section = StridedSection {
107        start: &selection.start,
108        shape: &packed,
109        stride: selection.stride.as_deref().unwrap_or(&ones),
110    };
111    let windows = PixelSections {
112        image,
113        selection: &selection,
114        mask: PreparedMask::new_for_outputs(
115            request.mask.as_deref(),
116            image.shape(),
117            image.coordinates(),
118            request.stretch,
119            (!request.outfile.as_os_str().is_empty()).then_some(request.outfile.as_path()),
120        )?,
121    };
122    let sampler = StripSampler {
123        section,
124        windows,
125        strip,
126        plane,
127        order: [x, y]
128            .into_iter()
129            .chain((0..packed.len()).filter(|axis| *axis != x && *axis != y))
130            .collect(),
131    };
132    sampler.copy_to(&mut result)?;
133    Ok(ImagePvProduct {
134        image: result,
135        coordinate_record,
136        beam_planes,
137    })
138}

140// A square tile in strip coordinates maps to at most a sqrt(2)-wider
141// square on the input plane. Tile both axes to bound memory for long or
142// wide diagonal strips, including the cubic interpolation halo.
143const STRIP_TILE: usize = 128;
144
145struct StripSampler<'a> {
146    section: StridedSection<'a>,
147    windows: PixelSections<'a, f32>,
148    strip: geometry::Strip,
149    plane: [usize; 2],
150    order: Vec<usize>,
151}
152
153impl StripSampler<'_> {
154    fn copy_to(&self, result: &mut TempImage<f32>) -> Result<(), ImageError> {
155        let [x, y] = self.plane;
156        let offset_axis = x - usize::from(y < x);
157        let mut grid = self.section.shape.to_vec();
158        grid[x] = 1;
159        grid[y] = 1;
160        let mut shape = vec![1; grid.len() - 1];
161        let planes =
162            TraversalCursorIter::new(grid, vec![1; self.order.len()], TraversalSpec::tiles());
163        let mut any_good = false;
164        let mut has_mask = false;
165        for plane in planes {
166            let plane_start = plane?.position;
167            let mut output_start = plane_start.clone();
168            output_start.remove(y);
169            for along in tile_ranges(self.strip.shape[0]) {
170                output_start[offset_axis] = along.start;
171                shape[offset_axis] = along.len();
172                let samples = self.reduce_rows(&plane_start, along)?;
173                let (values, valid): (Vec<_>, Vec<_>) = samples
174                    .into_iter()
175                    .map(|(sum, count)| {
176                        let good = count != 0;
177                        any_good |= good;
178                        (
179                            if good {
180                                (sum / count as f64) as f32
181                            } else {
182                                0.
183                            },
184                            good,
185                        )
186                    })
187                    .unzip();
188                let values =
189                    ArrayD::from_shape_vec(IxDyn(&shape).f(), values).expect("PV row length");
190                result.put_slice(&values, &output_start)?;
191                if valid.iter().any(|&good| !good) {
192                    if !has_mask {
193                        result.make_mask("mask0", true, true)?;
194                        has_mask = true;
195                    }
196                    let mask = ArrayD::from_shape_vec(IxDyn(&shape).f(), valid)
197                        .expect("PV mask row length");
198                    result.put_mask_slice("mask0", &mask, &output_start)?;
199                }
200            }
201        }
202        if !any_good {
203            return Err(ImageError::CasaSelection(
204                "All selected pixels are masked".into(),
205            ));
206        }
207        Ok(())
208    }
209
210    fn reduce_rows(
211        &self,
212        plane_start: &[usize],
213        along: Range<usize>,
214    ) -> Result<Vec<(f64, usize)>, ImageError> {
215        let [x, y] = self.plane;
216        let sky_shape = self.plane.map(|axis| self.section.shape[axis]);
217        let method = if self.strip.rotated() {
218            RegridMethod::Cubic
219        } else {
220            RegridMethod::Nearest
221        };
222        let mut reduced = vec![(0., 0); along.len()];
223        for across in tile_ranges(self.strip.shape[1]) {
224            let points = along
225                .clone()
226                .flat_map(|step| {
227                    across
228                        .clone()
229                        .map(move |width| self.strip.input_pixel(step, width))
230                })
231                .collect::<Result<Vec<_>, _>>()?;
232            let (origin, extent) = section_bounds(&points, sky_shape, self.strip.rotated());
233            let mut start = plane_start.to_vec();
234            let mut shape = vec![1; plane_start.len()];
235            start[x] = origin[0];
236            start[y] = origin[1];
237            shape[x] = extent[0];
238            shape[y] = extent[1];
239            let (pixels, mask) = self.windows.read(
240                &self.section.source_start(&start),
241                &shape,
242                self.section.stride,
243            )?;
244            let pixels = pixels.permuted_axes(self.order.clone()).reversed_axes();
245            let pixels: Vec<_> = pixels.iter().map(|&value| f64::from(value)).collect();
246            let mask = mask.map(|valid| {
247                valid
248                    .permuted_axes(self.order.clone())
249                    .reversed_axes()
250                    .iter()
251                    .copied()
252                    .collect::<Vec<_>>()
253            });
254            let sky = Plane {
255                data: &pixels,
256                mask: mask.as_deref(),
257                shape: extent,
258            };
259            for ((sum, count), row) in reduced.iter_mut().zip(points.chunks_exact(across.len())) {
260                for &[px, py] in row {
261                    let at = [px - origin[0] as f64, py - origin[1] as f64];
262                    if let Some(sample) = sky.sample(method, at) {
263                        *sum += f64::from(sample as f32);
264                        *count += 1;
265                    }
266                }
267            }
268        }
269        Ok(reduced)
270    }
271}

273fn tile_ranges(length: usize) -> impl Iterator<Item = Range<usize>> {
274    (0..length)
275        .step_by(STRIP_TILE)
276        .map(move |start| start..(start + STRIP_TILE).min(length))
277}

279fn section_bounds(points: &[[f64; 2]], shape: [usize; 2], cubic: bool) -> ([usize; 2], [usize; 2]) {
280    let mut lower = [f64::INFINITY; 2];
281    let mut upper = [f64::NEG_INFINITY; 2];
282    for point in points {
283        for axis in 0..2 {
284            lower[axis] = lower[axis].min(point[axis]);
285            upper[axis] = upper[axis].max(point[axis]);
286        }
287    }
288    let halo = if cubic { 2. } else { 1. };
289    let lower: [usize; 2] =
290        std::array::from_fn(|axis| ((lower[axis].floor() - halo) as usize).min(shape[axis] - 1));
291    let upper: [usize; 2] =
292        std::array::from_fn(|axis| ((upper[axis].ceil() + halo) as usize).min(shape[axis] - 1));
293    let extent = std::array::from_fn(|axis| upper[axis] + 1 - lower[axis]);
294    (lower, extent)
295}
