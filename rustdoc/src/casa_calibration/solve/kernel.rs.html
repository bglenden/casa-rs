2//! Numerical solve kernel for `gaincal` and `bandpass`.
3//!
4//! Every solve is CASA's `VisCalSolver2::solve`: a Levenberg–Marquardt
5//! minimisation of the weighted complex residual Σ w |V − g_i conj(g_j)|²
6//! over antenna gains, with every receptor of one solution solved together
7//! and a first guess from `GJones::guessPar` or `TJones::guessPar`. A
8//! phase-only solve fits data already reduced to unit amplitude
9//! (`SolveDataBuffer::enforceAPonData`) and normalises the gains afterwards,
10//! as `SolvableVisCal::enforceAPonSoln` does; an amplitude-and-phase solve
11//! keeps them. Within one solution the visibilities of each baseline are
12//! collapsed to their weighted sum, which leaves the chi-square unchanged up
13//! to a constant because the gains are constant over the solution.
14

46/// One receptor's collapsed visibilities for one solution. For each directed
47/// baseline (i, j) `edges` holds the weighted sum Σ w V_ij of its samples and
48/// `weights` the weight sum Σ w; (j, i) holds the conjugate sum. `stats`
49/// carries what the absolute chi-square needs (Σ w |V|²) and the sample
50/// counts behind the parameter errors.
51#[derive(Debug, Clone, Default)]
52pub(crate) struct SolveGraph {
53    pub(crate) edges: HashMap<(i32, i32), Complex32>,
54    pub(crate) weights: HashMap<(i32, i32), f32>,
55    pub(crate) stats: HashMap<(i32, i32), SolveEdgeStats>,
56    pub(crate) robust_samples: Vec<RobustSample>,
57}

71impl SolveGraph {
72    /// Add one collapsed sample of baseline `(antenna1, antenna2)`: its
73    /// weight, its weighted visibility and its statistics.
74    pub(crate) fn accumulate(
75        &mut self,
76        (antenna1, antenna2): (i32, i32),
77        weight: f32,
78        weighted_sample: Complex32,
79        edge_stats: SolveEdgeStats,
80    ) {
81        for (baseline, value) in [
82            ((antenna1, antenna2), weighted_sample),
83            ((antenna2, antenna1), weighted_sample.conj()),
84        ] {
85            *self.edges.entry(baseline).or_default() += value;
86            *self.weights.entry(baseline).or_default() += weight;
87            self.stats
88                .entry(baseline)
89                .and_modify(|stats| stats.add(edge_stats))
90                .or_insert(edge_stats);
91        }
92    }
93
94    fn weight(&self, baseline: (i32, i32)) -> f32 {
95        self.weights.get(&baseline).copied().unwrap_or_default()
96    }
97
98    /// The baselines between antennas that meet the solve constraints.
99    fn restricted_to(&self, antennas: &BTreeSet<i32>) -> Self {
100        let keep = |&(antenna1, antenna2): &(i32, i32)| {
101            antennas.contains(&antenna1) && antennas.contains(&antenna2)
102        };
103        Self {
104            edges: self
105                .edges
106                .iter()
107                .filter(|(baseline, _)| keep(baseline))
108                .map(|(baseline, value)| (*baseline, *value))
109                .collect(),
110            weights: self
111                .weights
112                .iter()
113                .filter(|(baseline, _)| keep(baseline))
114                .map(|(baseline, value)| (*baseline, *value))
115                .collect(),
116            stats: self
117                .stats
118                .iter()
119                .filter(|(baseline, _)| keep(baseline))
120                .map(|(baseline, value)| (*baseline, *value))
121                .collect(),
122            robust_samples: self
123                .robust_samples
124                .iter()
125                .copied()
126                .filter(|sample| keep(&sample.baseline))
127                .collect(),
128        }
129    }
130
131    /// Antennas with at least one positive-weight baseline.
132    pub(crate) fn antennas(&self) -> BTreeSet<i32> {
133        self.weights
134            .iter()
135            .filter(|((antenna1, antenna2), weight)| {
136                antenna1 != antenna2 && **weight > f32::EPSILON
137            })
138            .map(|((antenna1, _), _)| *antenna1)
139            .collect()
140    }
141}

207/// Solve one `gaincal` solution interval, or `None` where CASA keeps no
208/// solution for it (see [`solve_graphs`]). Every antenna of the array or
209/// the interval gets a row; antennas without a solution are flagged, as are
210/// solutions at or below `min_snr`.
211pub(crate) fn solve_group(
212    group: SolveAccumulator,
213    available_antennas: &BTreeSet<i32>,
214    gain_type: GainType,
215    solve_mode: GainSolveMode,
216    min_snr: f32,
217    options: SolveGraphOptions<'_>,
218) -> Option<Vec<SolutionRow>> {
219    let refant_id = options.refant_id;
220    let scan_number = *group.scan_numbers.iter().next().unwrap_or(&0);
221    let graphs = group.channel_graphs.first()?;
222    let solved = solve_graphs(
223        graphs,
224        gain_type,
225        solve_mode,
226        SolveGraphOptions {
227            scalar_seed: (gain_type == GainType::T).then_some(&group),
228            defer_reference: true,
229            ..options
230        },
231        None,
232    )?;
233
234    let mut antenna_ids = available_antennas.clone();
235    antenna_ids.extend(group.antenna_ids.iter().copied());
236    if refant_id >= 0 {
237        antenna_ids.insert(refant_id);
238    }
239    let receptors = match gain_type {
240        GainType::G => &solved.receptors[..],
241        GainType::T => &solved.receptors[..1],
242    };
243    Some(
244        antenna_ids
245            .into_iter()
246            .map(|antenna_id| {
247                let mut row = SolutionRow {
248                    time_seconds: group.solution_time,
249                    field_id: group.field_id,
250                    spw_id: group.spw_id,
251                    antenna_id,
252                    scan_number,
253                    observation_id: group.observation_id,
254                    refant_id: if refant_id < 0 { -1 } else { solved.refant_id },
255                    gains: Vec::with_capacity(receptors.len()),
256                    flags: Vec::with_capacity(receptors.len()),
257                    param_errors: Vec::with_capacity(receptors.len()),
258                    snrs: Vec::with_capacity(receptors.len()),
259                    weights: Vec::with_capacity(receptors.len()),
260                };
261                for receptor in receptors {
262                    let snr = receptor.snr.get(&antenna_id).copied().unwrap_or_default();
263                    let param_error = receptor
264                        .param_error
265                        .get(&antenna_id)
266                        .copied()
267                        .unwrap_or_default();
268                    let flagged = !receptor.solved.contains(&antenna_id)
269                        || snr <= min_snr
270                        || !param_error.is_finite();
271                    row.gains.push(if flagged {
272                        Complex32::new(1.0, 0.0)
273                    } else {
274                        receptor.gains[&antenna_id]
275                    });
276                    row.flags.push(flagged);
277                    row.param_errors.push(param_error);
278                    row.snrs.push(snr);
279                    row.weights.push(
280                        receptor
281                            .hessian
282                            .get(&antenna_id)
283                            .copied()
284                            .unwrap_or_default(),
285                    );
286                }
287                row
288            })
289            .collect(),
290    )
291}

293/// Solve the receptor graphs of one solution together, as
294/// `VisCalSolver2::solve` does, or `None` where CASA records no solution:
295/// no antenna keeps `min_baselines_per_antenna` baselines with data
296/// (`SolvableVisCal::verifyConstraints`), or the solve has not converged
297/// after [`MAX_ITERATIONS`]. CASA also fails a solve whose chi-square is
298/// exactly zero; here an exact fit is a solution (`docs/CASA (C++) bugs.md`).
299///
300/// `start`, when given, seeds the antennas it solved in place of the first
301/// guess, as CASA's B solve starts each channel from the one solved before.
302pub(crate) fn solve_graphs(
303    graphs: &[SolveGraph],
304    gain_type: GainType,
305    solve_mode: GainSolveMode,
306    options: SolveGraphOptions<'_>,
307    start: Option<&SolvedGraphs>,
308) -> Option<SolvedGraphs> {
309    let constrained = baseline_constrained_antennas(graphs, options.min_baselines_per_antenna);
310    let refant_id = if constrained.contains(&options.refant_id) {
311        options.refant_id
312    } else {
313        *constrained.first()?
314    };
315    let active = graphs
316        .iter()
317        .map(|graph| graph.restricted_to(&constrained))
318        .collect::<Vec<_>>();
319    let mut solved_antennas = active
320        .iter()
321        .map(|graph| {
322            let with_data = graph.antennas();
323            if options.correlation_dependent_flags {
324                with_data
325            } else {
326                reachable_antennas(graph, refant_id)
327                    .into_iter()
328                    .filter(|antenna_id| with_data.contains(antenna_id))
329                    .collect::<BTreeSet<_>>()
330            }
331        })
332        .collect::<Vec<_>>();
333    if solved_antennas.iter().all(BTreeSet::is_empty) {
334        return None;
335    }
336    let mut first = if let Some(seed) = options.scalar_seed
337        && let Some(guess) =
338            seed.scalar_initial_gains(&solved_antennas[0], options.refant_id, solve_mode)
339    {
340        vec![guess]
341    } else if options.robustness.is_some() {
342        let first_graphs = robust::first_graphs(&active);
343        let first_reference = first_graphs
344            .iter()
345            .flat_map(SolveGraph::antennas)
346            .min_by_key(|antenna| (*antenna != options.refant_id, *antenna))
347            .unwrap_or(refant_id);
348        first_guess(&first_graphs, &solved_antennas, gain_type, first_reference)
349    } else {
350        first_guess(&active, &solved_antennas, gain_type, refant_id)
351    };
352    if let Some(start) = start {
353        for (guess, receptor) in first.iter_mut().zip(&start.receptors) {
354            for (antenna_id, value) in guess.iter_mut() {
355                if receptor.solved.contains(antenna_id) {
356                    *value = complex32_to_64(receptor.gains[antenna_id]);
357                }
358            }
359        }
360    }
361    let (gains, active) = if let Some(robustness) = options.robustness {
362        robust::solve(active, &mut solved_antennas, first, refant_id, robustness)?
363    } else {
364        (
365            solve_lm(
366                &active,
367                &mut solved_antennas,
368                first,
369                refant_id,
370                SolveIteration::Ordinary,
371            )?,
372            active,
373        )
374    };
375    let mut receptors = active
376        .iter()
377        .zip(solved_antennas)
378        .zip(gains)
379        .map(|((graph, mut solved), gains)| {
380            if options.robustness.is_some() {
381                let with_weight = graph.antennas();
382                solved.retain(|id| with_weight.contains(id));
383            }
384            finish_receptor(
385                graph,
386                solved,
387                gains,
388                solve_mode,
389                if options.defer_reference || options.refant_id < 0 {
390                    -1
391                } else {
392                    refant_id
393                },
394            )
395        })
396        .collect::<Vec<_>>();
397    apply_group_reduced_chi_square(&mut receptors);
398    Some(SolvedGraphs {
399        refant_id,
400        receptors,
401    })
402}

404/// The antennas that keep at least `min_baselines_per_antenna` baselines
405/// with data once antennas below the threshold, and their baselines, are
406/// removed in turn (`SolvableVisCal::verifyConstraints`). A baseline counts
407/// when any receptor has weight on it.
408pub(crate) fn baseline_constrained_antennas(
409    graphs: &[SolveGraph],
410    min_baselines_per_antenna: usize,
411) -> BTreeSet<i32> {
412    let baselines = graphs
413        .iter()
414        .flat_map(|graph| graph.stats.iter())
415        .filter(|((antenna1, antenna2), stats)| antenna1 < antenna2 && stats.collapsed_count > 0)
416        .map(|(baseline, _)| *baseline)
417        .collect::<BTreeSet<_>>();
418    let mut active = baselines
419        .iter()
420        .flat_map(|&(antenna1, antenna2)| [antenna1, antenna2])
421        .collect::<BTreeSet<_>>();
422    loop {
423        let mut counts = HashMap::<i32, usize>::new();
424        for &(antenna1, antenna2) in &baselines {
425            if active.contains(&antenna1) && active.contains(&antenna2) {
426                *counts.entry(antenna1).or_default() += 1;
427                *counts.entry(antenna2).or_default() += 1;
428            }
429        }
430        let before = active.len();
431        active.retain(|antenna_id| {
432            counts.get(antenna_id).copied().unwrap_or_default() >= min_baselines_per_antenna.max(1)
433        });
434        if active.len() == before {
435            return active;
436        }
437    }
438}

440/// CASA's first guess (`GJones::guessPar`, `TJones::guessPar`): each antenna
441/// starts from its baseline to the reference antenna, conjugated (G) or as a
442/// unit phasor (T), divided by the square root of those baselines' mean
443/// amplitude, which is also the reference antenna's start. Antennas without
444/// a baseline to the reference start there too. Phase-only data have unit
445/// amplitude, so the guess is then the baselines' phases.
446fn first_guess(
447    graphs: &[SolveGraph],
448    antennas: &[BTreeSet<i32>],
449    gain_type: GainType,
450    refant_id: i32,
451) -> Vec<HashMap<i32, Complex64>> {
452    let mut amplitude_sum = 0.0_f64;
453    let mut amplitude_count = 0_usize;
454    let mut guesses = graphs
455        .iter()
456        .zip(antennas)
457        .map(|(graph, antennas)| {
458            let seed = if antennas.contains(&refant_id) {
459                refant_id
460            } else {
461                antennas.first().copied().unwrap_or(refant_id)
462            };
463            antennas
464                .iter()
465                .filter(|antenna_id| **antenna_id != seed)
466                .filter_map(|&antenna_id| {
467                    let weight = f64::from(graph.weight((seed, antenna_id)));
468                    let edge = graph.edges.get(&(seed, antenna_id))?;
469                    let mean = complex32_to_64(*edge) / weight;
470                    let amplitude = mean.norm();
471                    (weight > f64::EPSILON && amplitude > 0.0).then(|| {
472                        amplitude_sum += amplitude;
473                        amplitude_count += 1;
474                        let guess = match gain_type {
475                            GainType::G => mean.conj(),
476                            GainType::T => mean.conj() / amplitude,
477                        };
478                        (antenna_id, guess)
479                    })
480                })
481                .collect::<HashMap<_, _>>()
482        })
483        .collect::<Vec<_>>();
484    let scale = if amplitude_count > 0 {
485        (amplitude_sum / amplitude_count as f64).sqrt()
486    } else {
487        1.0
488    };
489    for (guess, antennas) in guesses.iter_mut().zip(antennas) {
490        for value in guess.values_mut() {
491            *value /= scale;
492        }
493        for antenna_id in antennas {
494            guess
495                .entry(*antenna_id)
496                .or_insert(Complex64::new(scale, 0.0));
497        }
498    }
499    guesses
500}

504    Ordinary,
505    Rms,
506    L1(f32),

509/// The Levenberg–Marquardt iteration of `VisCalSolver2::solve`, with its
510/// step size optimisation (`optStepSize2`) and convergence test
511/// (`converged`): a step counts towards convergence when chi-square has not
512/// risen by more than 0.1% and changed by less than 10%, and six such steps
513/// converge. `None` when [`MAX_ITERATIONS`] pass without converging.
514fn solve_lm(
515    graphs: &[SolveGraph],
516    antennas: &mut [BTreeSet<i32>],
517    mut gains: Vec<HashMap<i32, Complex64>>,
518    refant_id: i32,
519    strategy: SolveIteration,
520) -> Option<Vec<HashMap<i32, Complex64>>> {
521    let mut last_gains = gains.clone();
522    let mut last_chi_square = f64::MAX;
523    let mut convergence_count = 0_i32;
524    let robust_iteration = !matches!(strategy, SolveIteration::Ordinary);
525    for iteration in 0..=MAX_ITERATIONS {
526        let weighted = match strategy {
527            SolveIteration::L1(clamp) => Some(robust::working_graphs(graphs, &gains, clamp)),
528            _ => None,
529        };
530        let graphs = weighted.as_deref().unwrap_or(graphs);
531        if robust_iteration && graphs.iter().all(|graph| graph.weights.is_empty()) {
532            return None;
533        }
534        let mut chi_square = chi_square_graphs(graphs, &gains);
535        if chi_square <= 0.0 {
536            return Some(gains);
537        }
538        let delta_chi_square = chi_square - last_chi_square;
539        let fractional_delta = delta_chi_square / chi_square;
540        if fractional_delta <= 0.001 {
541            if delta_chi_square.abs() < 0.1 * chi_square {
542                convergence_count += 1;
543            }
544            if convergence_count > 5 {
545                return Some(gains);
546            }
547        } else if delta_chi_square.abs() > 0.1 * chi_square {
548            convergence_count = 0;
549        } else {
550            convergence_count = (convergence_count - 1).max(0);
551        }
552
553        if delta_chi_square <= 0.0 || robust_iteration {
554            last_chi_square = chi_square;
555        } else {
556            gains.clone_from(&last_gains);
557            chi_square = chi_square_graphs(graphs, &gains);
558        }
559
560        let mut gradients = Vec::with_capacity(graphs.len());
561        let mut deltas = Vec::with_capacity(graphs.len());
562        for ((graph, antennas), receptor_gains) in
563            graphs.iter().zip(antennas.iter_mut()).zip(&gains)
564        {
565            let (gradient, hessian) = gradient_hessian(graph, antennas, receptor_gains);
566            if robust_iteration {
567                // solveGradHess clears parOK permanently, but residualate2
568                // and getErrors still use the last parameter value.
569                antennas.retain(|id| hessian.get(id).is_some_and(|value| *value > 0.0));
570            }
571            let delta = antennas
572                .iter()
573                .map(|antenna_id| {
574                    let hessian = hessian.get(antenna_id).copied().unwrap_or_default();
575                    let value = if hessian > f64::EPSILON {
576                        -gradient[antenna_id] / hessian / 2.0
577                    } else {
578                        Complex64::new(0.0, 0.0)
579                    };
580                    (*antenna_id, value)
581                })
582                .collect::<HashMap<_, _>>();
583            gradients.push((gradient, hessian));
584            deltas.push(delta);
585        }
586
587        last_gains.clone_from(&gains);
588        let step = (!matches!(strategy, SolveIteration::L1(_)))
589            .then(|| optimize_step(graphs, &last_gains, &mut gains, &mut deltas, chi_square));
590        for (receptor_gains, receptor_delta) in gains.iter_mut().zip(&deltas) {
591            for (antenna_id, gain) in receptor_gains.iter_mut() {
592                *gain += receptor_delta.get(antenna_id).copied().unwrap_or_default();
593            }
594        }
595        if let Some(step) = &step {
596            for (receptor, (antennas, (gradient, hessian))) in
597                antennas.iter().zip(&gradients).enumerate()
598            {
599                trace::trace_phase_solver_iteration(PhaseSolverTraceEvent {
600                    iteration,
601                    refant_id,
602                    chi_square,
603                    last_chi_square,
604                    delta_chi_square,
605                    fractional_delta,
606                    convergence_count,
607                    step,
608                    antenna_ids: antennas,
609                    last_gains: &last_gains[receptor],
610                    gains: &gains[receptor],
611                    gradient,
612                    hessian,
613                    delta: &deltas[receptor],
614                });
615            }
616        }
617    }
618    None
619}

621/// Refer one receptor's converged gains to the reference antenna's phase and
622/// derive its diagnostics. Errors and SNRs come from the solved gains, before
623/// a phase-only solve normalises them (`SolvableVisCal::formSolveSNR` runs
624/// before `enforceAPonSoln`).
625fn finish_receptor(
626    graph: &SolveGraph,
627    solved: BTreeSet<i32>,
628    mut gains: HashMap<i32, Complex64>,
629    solve_mode: GainSolveMode,
630    refant_id: i32,
631) -> SolvedReceptor {
632    rereference_complex64(&mut gains, refant_id, false);
633    let error_gains = gains
634        .iter()
635        .map(|(antenna_id, gain)| (*antenna_id, complex64_to_32(*gain)))
636        .collect::<HashMap<_, _>>();
637    let snr_amplitude = gains
638        .iter()
639        .map(|(antenna_id, gain)| (*antenna_id, gain.norm() as f32))
640        .collect::<HashMap<_, _>>();
641    let error_chi_square = if graph.robust_samples.is_empty() {
642        chi_square_for_errors(graph, &error_gains)
643    } else {
644        robust::chi_square(graph, &gains)
645    };
646    if solve_mode == GainSolveMode::Phase {
647        rereference_complex64(&mut gains, refant_id, true);
648    }
649    let gains = gains
650        .into_iter()
651        .map(|(antenna_id, gain)| (antenna_id, complex64_to_32(gain)))
652        .collect::<HashMap<_, _>>();
653
654    let hessian = hessian(graph, &error_gains, &solved);
655    let good_count = solved.len();
656    let sample_count = undirected_collapsed_count(&graph.stats);
657    let degrees_of_freedom = (2 * sample_count.saturating_sub(good_count)).max(1);
658    let param_error = param_errors(&hessian, error_chi_square / degrees_of_freedom as f64);
659    let snr = snrs(&gains, &solved, &param_error, &snr_amplitude);
660
661    SolvedReceptor {
662        gains,
663        solved,
664        hessian,
665        param_error,
666        snr,
667        error_chi_square,
668        sample_count,
669        good_count,
670        snr_amplitude,
671    }
672}

674/// Errors from the reduced chi-square of all receptors together, as one
675/// `VisCalSolver2::getErrors` covers every parameter of the solve.
676fn apply_group_reduced_chi_square(solved: &mut [SolvedReceptor]) {
677    let error_chi_square = solved
678        .iter()
679        .map(|per_receptor| per_receptor.error_chi_square)
680        .sum::<f64>();
681    let sample_count = solved
682        .iter()
683        .map(|per_receptor| per_receptor.sample_count)
684        .sum::<usize>();
685    let good_count = solved
686        .iter()
687        .map(|per_receptor| per_receptor.good_count)
688        .sum::<usize>();
689    let degrees_of_freedom = (2 * sample_count.saturating_sub(good_count)).max(1);
690    let reduced_chi_square = error_chi_square / degrees_of_freedom as f64;
691
692    for per_receptor in solved {
693        per_receptor.param_error = param_errors(&per_receptor.hessian, reduced_chi_square);
694        per_receptor.snr = snrs(
695            &per_receptor.gains,
696            &per_receptor.solved,
697            &per_receptor.param_error,
698            &per_receptor.snr_amplitude,
699        );
700    }
701}

703fn snrs(
704    gains: &HashMap<i32, Complex32>,
705    solved: &BTreeSet<i32>,
706    param_error: &HashMap<i32, f32>,
707    snr_amplitude: &HashMap<i32, f32>,
708) -> HashMap<i32, f32> {
709    gains
710        .iter()
711        .map(|(antenna_id, gain)| {
712            let error = param_error.get(antenna_id).copied().unwrap_or_default();
713            let amplitude = snr_amplitude
714                .get(antenna_id)
715                .copied()
716                .unwrap_or_else(|| gain.norm());
717            let value = if !solved.contains(antenna_id) {
718                0.0
719            } else if error > 0.0 {
720                amplitude / error
721            } else {
722                9_999_999.0
723            };
724            (*antenna_id, value)
725        })
726        .collect()
727}

729/// `VisCalSolver2::optStepSize2`: scale the Gauss–Newton step by trying it,
730/// doubling it while chi-square falls or halving it until it does, then
731/// taking the vertex of the parabola through the last three chi-squares.
732/// `gains` returns equal to `last_gains`; `deltas` returns scaled.
733fn optimize_step(
734    graphs: &[SolveGraph],
735    last_gains: &[HashMap<i32, Complex64>],
736    gains: &mut [HashMap<i32, Complex64>],
737    deltas: &mut [HashMap<i32, Complex64>],
738    current_chi_square: f64,
739) -> PhaseStepTrace {
740    let mut step = 1.0_f64;
741    let mut iterations = 0_usize;
742    let x0 = current_chi_square;
743    apply_step(gains, last_gains, deltas, 1.0);
744    let mut x1 = chi_square_graphs(graphs, gains);
745    let expanded;
746
747    let x2 = if x1 < x0 {
748        expanded = true;
749        apply_step(gains, last_gains, deltas, 2.0 * step);
750        let mut trial_x2 = chi_square_graphs(graphs, gains);
751        while trial_x2 < x1 {
752            iterations += 1;
753            step *= 2.0;
754            x1 = trial_x2;
755            apply_step(gains, last_gains, deltas, 2.0 * step);
756            trial_x2 = chi_square_graphs(graphs, gains);
757        }
758        trial_x2
759    } else {
760        expanded = false;
761        step *= 0.5;
762        apply_step(gains, last_gains, deltas, step);
763        let mut trial_x2 = x1;
764        x1 = chi_square_graphs(graphs, gains);
765        while x1 > x0 {
766            iterations += 1;
767            step *= 0.5;
768            trial_x2 = x1;
769            apply_step(gains, last_gains, deltas, step);
770            x1 = chi_square_graphs(graphs, gains);
771        }
772        trial_x2
773    };
774
775    let denominator = x0 - 2.0 * x1 + x2;
776    let opt_factor = if denominator.abs() > 0.0 {
777        step * (1.5 - (x2 - x1) / denominator)
778    } else {
779        0.0
780    };
781
782    for (gain, last_gain) in gains.iter_mut().zip(last_gains) {
783        gain.clone_from(last_gain);
784    }
785    if opt_factor > 0.0 {
786        for receptor_delta in deltas {
787            for value in receptor_delta.values_mut() {
788                *value *= opt_factor;
789            }
790        }
791    }
792    PhaseStepTrace {
793        x0,
794        x1,
795        x2,
796        step,
797        opt_factor,
798        expanded,
799        iterations,
800    }
801}

803fn apply_step(
804    gains: &mut [HashMap<i32, Complex64>],
805    bases: &[HashMap<i32, Complex64>],
806    deltas: &[HashMap<i32, Complex64>],
807    factor: f64,
808) {
809    for ((receptor_gains, base), delta) in gains.iter_mut().zip(bases).zip(deltas) {
810        for (antenna_id, gain) in receptor_gains.iter_mut() {
811            *gain = base[antenna_id] + factor * delta.get(antenna_id).copied().unwrap_or_default();
812        }
813    }
814}

816fn chi_square_graphs(graphs: &[SolveGraph], gains: &[HashMap<i32, Complex64>]) -> f64 {
817    graphs
818        .iter()
819        .zip(gains)
820        .map(|(graph, gains)| chi_square_complex64(graph, gains))
821        .sum()
822}
823
824fn chi_square_complex64(graph: &SolveGraph, gains: &HashMap<i32, Complex64>) -> f64 {
825    if !graph.robust_samples.is_empty() {
826        return robust::chi_square(graph, gains);
827    }
828    let mut total = 0.0_f64;
829    for (&(antenna1, antenna2), stat) in &graph.stats {
830        if antenna1 > antenna2 {
831            continue;
832        }
833        let weight = f64::from(graph.weight((antenna1, antenna2)));
834        let gain1 = gains
835            .get(&antenna1)
836            .copied()
837            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
838        let gain2 = gains
839            .get(&antenna2)
840            .copied()
841            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
842        let observed = graph
843            .edges
844            .get(&(antenna1, antenna2))
845            .copied()
846            .map(complex32_to_64)
847            .unwrap_or_default();
848        let model = gain1 * gain2.conj();
849        let cross = (model.conj() * observed).re;
850        let contribution = stat.weighted_sample_power - 2.0 * cross + weight * model.norm_sqr();
851        total += contribution.max(0.0);
852    }
853    total
854}

856/// The chi-square's gradient and diagonal Hessian with respect to each gain
857/// of `antennas` (`VisCalSolver2::accGradHess2`).
858fn gradient_hessian(
859    graph: &SolveGraph,
860    antennas: &BTreeSet<i32>,
861    gains: &HashMap<i32, Complex64>,
862) -> (HashMap<i32, Complex64>, HashMap<i32, f64>) {
863    let mut gradient = antennas
864        .iter()
865        .map(|antenna_id| (*antenna_id, Complex64::new(0.0, 0.0)))
866        .collect::<HashMap<_, _>>();
867    let mut hessian = antennas
868        .iter()
869        .map(|antenna_id| (*antenna_id, 0.0_f64))
870        .collect::<HashMap<_, _>>();
871
872    for (&(antenna1, antenna2), edge) in &graph.edges {
873        if antenna1 > antenna2 {
874            continue;
875        }
876        let weight = f64::from(graph.weight((antenna1, antenna2)));
877        if weight <= f64::EPSILON {
878            continue;
879        }
880        let gain1 = gains
881            .get(&antenna1)
882            .copied()
883            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
884        let gain2 = gains
885            .get(&antenna2)
886            .copied()
887            .unwrap_or_else(|| Complex64::new(1.0, 0.0));
888        let residual_sum = weight * gain1 * gain2.conj() - complex32_to_64(*edge);
889        let deriv1 = gain2.conj();
890        let deriv2 = gain1;
891        if let Some(value) = gradient.get_mut(&antenna1) {
892            *value += residual_sum * deriv1.conj();
893            *hessian.entry(antenna1).or_default() += weight * deriv1.norm_sqr();
894        }
895        if let Some(value) = gradient.get_mut(&antenna2) {
896            *value += deriv2 * residual_sum.conj();
897            *hessian.entry(antenna2).or_default() += weight * deriv2.norm_sqr();
898        }
899    }
900
901    (gradient, hessian)
902}

904/// Rotate every gain by the reference antenna's phase; with `phase_only`
905/// also normalise each to unit amplitude.
906fn rereference_complex64(gains: &mut HashMap<i32, Complex64>, refant_id: i32, phase_only: bool) {
907    let ref_gain = gains
908        .get(&refant_id)
909        .copied()
910        .unwrap_or_else(|| Complex64::new(1.0, 0.0));
911    let ref_norm = ref_gain.norm();
912    if ref_norm > f64::EPSILON {
913        let anchor = ref_gain / ref_norm;
914        for gain in gains.values_mut() {
915            *gain /= anchor;
916        }
917    }
918    if phase_only {
919        for gain in gains.values_mut() {
920            let norm = gain.norm();
921            if norm > f64::EPSILON {
922                *gain /= norm;
923            }
924        }
925        if let Some(gain) = gains.get_mut(&refant_id) {
926            *gain = Complex64::new(1.0, 0.0);
927        }
928    }
929}

939fn hessian(
940    graph: &SolveGraph,
941    gains: &HashMap<i32, Complex32>,
942    solved: &BTreeSet<i32>,
943) -> HashMap<i32, f32> {
944    let mut hessian = HashMap::new();
945    for &(antenna_id, other_id) in graph.edges.keys() {
946        let included = if graph.robust_samples.is_empty() {
947            solved.contains(&antenna_id) && solved.contains(&other_id)
948        } else {
949            gains.contains_key(&antenna_id) && gains.contains_key(&other_id)
950        };
951        if !included {
952            continue;
953        }
954        let weight = graph.weight((antenna_id, other_id));
955        if weight <= f32::EPSILON {
956            continue;
957        }
958        let other_power = gains
959            .get(&other_id)
960            .map(|gain| gain.norm_sqr())
961            .unwrap_or(1.0);
962        *hessian.entry(antenna_id).or_insert(0.0) += weight * other_power;
963    }
964    hessian
965}

967fn chi_square_for_errors(graph: &SolveGraph, gains: &HashMap<i32, Complex32>) -> f64 {
968    let mut total = 0.0_f64;
969    for (&(antenna1, antenna2), stat) in &graph.stats {
970        if antenna1 > antenna2 {
971            continue;
972        }
973        let gain1 = gains
974            .get(&antenna1)
975            .copied()
976            .unwrap_or_else(|| Complex32::new(1.0, 0.0));
977        let gain2 = gains
978            .get(&antenna2)
979            .copied()
980            .unwrap_or_else(|| Complex32::new(1.0, 0.0));
981        let observed = graph
982            .edges
983            .get(&(antenna1, antenna2))
984            .copied()
985            .unwrap_or_default();
986        let weight = f64::from(graph.weight((antenna1, antenna2)));
987        let model = gain1 * gain2.conj();
988        let cross = (model.conj() * observed).re;
989        let contribution = stat.weighted_sample_power - 2.0 * f64::from(cross)
990            + weight * f64::from(model.norm_sqr());
991        total += contribution.max(0.0);
992    }
993    total
994}

996fn undirected_collapsed_count(stats: &HashMap<(i32, i32), SolveEdgeStats>) -> usize {
997    stats
998        .iter()
999        .filter_map(|(&(antenna1, antenna2), stat)| {
1000            (antenna1 < antenna2).then_some(stat.collapsed_count)
1001        })
1002        .sum()
1003}

1005fn param_errors(hessian: &HashMap<i32, f32>, reduced_chi_square: f64) -> HashMap<i32, f32> {
1006    hessian
1007        .iter()
1008        .map(|(antenna_id, hessian)| {
1009            let value = if *hessian > 0.0 && reduced_chi_square.is_finite() {
1010                (2.0 * reduced_chi_square / f64::from(*hessian)).sqrt() as f32
1011            } else {
1012                0.0
1013            };
1014            (*antenna_id, value)
1015        })
1016        .collect()
1017}

1019fn reachable_antennas(graph: &SolveGraph, refant_id: i32) -> BTreeSet<i32> {
1020    let mut reachable = BTreeSet::from([refant_id]);
1021    let mut queue = VecDeque::from([refant_id]);
1022    while let Some(antenna_id) = queue.pop_front() {
1023        for (from, to) in graph.edges.keys() {
1024            if *from == antenna_id && reachable.insert(*to) {
1025                queue.push_back(*to);
1026            }
1027        }
1028    }
1029    reachable
1030}
