48impl PreparedAwPhase {
49    pub fn bind_plan(&self) -> Result<PreparedAwPlanBinding, ApplicationError> {
50        let reader = self.reader.session();
51        let provider = PreparedAwCellProvider::new(
52            self.resident_bytes,
53            Arc::clone(&reader),
54            self.prepared.clone(),
55        )?;
56        let projection = PreparedAwProjection::new(
57            self.catalog.clone(),
58            provider.clone(),
59            self.conjugate_beams,
60            self.resident_bytes,
61        )?;
62        Ok(PreparedAwPlanBinding {
63            projection,
64            execution: PreparedArtifactExecutionBinding::new(reader, provider),
65        })
66    }

123impl WorkImplementation for PhaseImplementation {
124    type Error = io::Error;
125
126    fn implementation_id(&self) -> &WorkImplementationId {
127        match self {
128            Self::Base { id, .. } => id,
129            Self::Catalog(op) => &op.id,
130        }
131    }
132
133    fn execute(&self, context: WorkExecutionContext<'_>) -> Result<WorkMeasurements, Self::Error> {
134        let op = match self {
135            Self::Base { sources, .. } => return base_measurements(context, sources),
136            Self::Catalog(op) => op,
137        };
138        let (result, measurements) = match &op.input {
139            CatalogOperation::Exact {
140                descriptors,
141                cold: Some(cold),
142            } => {
143                let (artifacts, measurements) = op
144                    .store
145                    .import_catalog(&context, descriptors, &cold.sources, |index| {
146                        cold.cache.importer(&cold.cells[index]).map_err(|_| {
147                            casa_imaging_runtime::PreparedArtifactError::SourceIdentityMismatch
148                        })
149                    })
150                    .map_err(io::Error::other)?;
151                (CatalogOutcome::Imported(artifacts), measurements)
152            }
153            CatalogOperation::Exact {
154                descriptors,
155                cold: None,
156            } => {
157                let (outcome, measurements) = op
158                    .store
159                    .reuse_catalog(&context, descriptors)
160                    .map_err(io::Error::other)?;
161                (CatalogOutcome::Reused(outcome), measurements)
162            }
163            CatalogOperation::Native { request, operation } => {
164                let mut generator = native::NativeGenerator::new(request.request());
165                let (outcome, measurements) = op.store.prepare_native_catalog(&context, request, *operation,
166                (*operation != casa_imaging_runtime::PreparedArtifactNativeOperation::Reuse)
167                    .then_some(&mut generator as &mut dyn casa_imaging_runtime::PreparedArtifactNativeGenerator))
168                .map_err(io::Error::other)?;
169                (CatalogOutcome::Native(outcome), measurements)
170            }
171        };
172        *op.result
173            .lock()
174            .map_err(|_| io::Error::other("AW catalog result state poisoned"))? = Some(result);
175        Ok(measurements)
176    }
177
178    fn failure_measurements<'a>(&'a self, error: &'a Self::Error) -> Option<&'a WorkMeasurements> {
179        error
180            .get_ref()?
181            .downcast_ref::<casa_imaging_runtime::PreparedArtifactError>()?
182            .work_measurements()
183    }
184
185    fn wait_for_fence(
186        &self,
187        _: WorkExecutionContext<'_>,
188        _: FenceKind,
189    ) -> Result<WorkMeasurements, Self::Error> {
190        Ok(WorkMeasurements::default())
191    }
192
193    fn complete_observation_read(
194        &self,
195        _: ObservationReadCompletionContext,
196    ) -> Result<AttemptBoundObservationCompletion, Self::Error> {
197        Err(io::Error::other("AW pre-phase is source-free"))
198    }
199
200    fn publish(&self, _: WorkExecutionContext<'_>) -> Result<(), Self::Error> {
201        Ok(())
202    }
203}
204
205fn base_measurements(
206    context: WorkExecutionContext<'_>,
207    sources: &[(casa_imaging_runtime::ArtifactIdentity, u64)],
208) -> Result<WorkMeasurements, io::Error> {
209    let resources = context
210        .resources()
211        .iter()
212        .map(|capability| {
213            ResourceMeasurement::new(
214                capability.resource().clone(),
215                capability.lifetime().clone(),
216                capability.amount(),
217            )
218        })
219        .collect();
220    let io = match context.node().id.as_str() {
221        "prepared-phase-commit" => vec![IoMeasurement::new(IoBufferKind::Publication, 0, 0)],
222        node if node.starts_with("prepared-release-") => {
223            vec![IoMeasurement::new(IoBufferKind::StorageManager, 0, 0)]
224        }
225        _ => vec![],
226    };
227    let artifacts = (context.node().id.as_str() == "prepared-phase-producer")
228        .then(|| {
229            sources
230                .iter()
231                .map(|&(identity, bytes)| {
232                    ArtifactMeasurement::new(
233                        identity,
234                        Some(identity),
235                        ArtifactDisposition::Loaded,
236                        bytes,
237                        None,
238                    )
239                    .map_err(io::Error::other)
240                })
241                .collect::<Result<Vec<_>, _>>()
242        })
243        .transpose()?
244        .unwrap_or_default();
245    Ok(WorkMeasurements::new(resources, io, artifacts))
246}

248struct PhaseRegistry {
249    id: ImplementationRegistryId,
250    metadata: ImplementationContractMetadata,
251    implementations: BTreeMap<WorkImplementationId, PhaseImplementation>,
252    prepared_artifact: PreparedArtifactRegistration,

255impl ImplementationRegistry for PhaseRegistry {
256    type Implementation = PhaseImplementation;
257    fn registry_id(&self) -> ImplementationRegistryId {
258        self.id
259    }
260    fn resolve(&self, id: &WorkImplementationId) -> Option<&Self::Implementation> {
261        self.implementations.get(id)
262    }
263    fn implementation_contract(
264        &self,
265        id: &WorkImplementationId,
266    ) -> Option<ImplementationContractMetadata> {
267        self.implementations
268            .contains_key(id)
269            .then(|| self.metadata.clone())
270    }
271    fn prepared_artifact_registration(
272        &self,
273        implementation: &WorkImplementationId,
274    ) -> Option<&PreparedArtifactRegistration> {
275        (implementation == self.prepared_artifact.implementation())
276            .then_some(&self.prepared_artifact)
277    }

280impl PhaseRegistry {
281    fn catalog(&self) -> &CatalogAdapter {
282        self.implementations
283            .values()
284            .find_map(|implementation| match implementation {
285                PhaseImplementation::Catalog(op) => Some(op.as_ref()),
286                PhaseImplementation::Base { .. } => None,
287            })
288            .expect("AW phase registry contains exactly one catalog")
289    }
290
291    fn into_result(self) -> Result<CatalogPhaseResult, ApplicationError> {
292        let catalog = self
293            .implementations
294            .into_values()
295            .find_map(|implementation| match implementation {
296                PhaseImplementation::Catalog(op) => Some(op),
297                PhaseImplementation::Base { .. } => None,
298            })
299            .expect("AW phase registry contains exactly one catalog");
300        let result = catalog
301            .result
302            .into_inner()
303            .map_err(|_| boxed("AW catalog result state poisoned"))?
304            .ok_or_else(|| boxed("AW catalog operation did not complete"))?;
305        match (catalog.input, result) {
306            (CatalogOperation::Exact { cold: None, .. }, CatalogOutcome::Reused(outcome)) => {
307                Ok(CatalogPhaseResult::Reused(outcome))
308            }
309            (
310                CatalogOperation::Exact {
311                    cold: Some(cold), ..
312                },
313                CatalogOutcome::Imported(artifacts),
314            ) => {
315                if cold.cells.len() != artifacts.len()
316                    || cold
317                        .cells
318                        .iter()
319                        .zip(&artifacts)
320                        .any(|(cell, artifact)| cell.descriptor().identity() != artifact.identity())
321                {
322                    return Err(boxed("AW cold catalog omitted or reordered an artifact"));
323                }
324                Ok(CatalogPhaseResult::Imported(
325                    cold.cells.into_iter().zip(artifacts).collect(),
326                ))
327            }
328            (CatalogOperation::Native { .. }, CatalogOutcome::Native(outcome)) => {
329                Ok(CatalogPhaseResult::Native(outcome))
330            }
331            _ => Err(boxed(
332                "AW catalog result did not match its planned operation",
333            )),
334        }
335    }
336}

338pub(crate) fn prepare_aw_projection(
339    problem: &CompiledProblem,
340    deployment: ApplicationAwPreparation,
341    runtime: &ApplicationRuntime,
342) -> Result<PreparedAwPhase, ApplicationError> {
343    if matches!(
344        deployment.source,
345        crate::ApplicationAwSource::NativeEvla { .. }
346    ) {
347        return native::prepare(problem, deployment, runtime);
348    }
349    std::fs::create_dir_all(&deployment.private_root)?;
350    let crate::ApplicationAwSource::CasaImport(path) = &deployment.source else {
351        unreachable!("native source dispatched above")
352    };
353    let cache = Arc::new(CasaAwCache::open(path)?);
354    let catalog = cache.prepared_catalog()?;
355    let inventory = cache.inventory();
356    let entries = inventory.paired_cells;
357    let prepared_cache_bytes = cache.prepared_cache_bytes()?;
358    eprintln!(
359        "imaging_aw_cache_inventory_summary paired_cells={} frequencies={} w_values={} mueller_elements={} parallactic_angles={} prepared_cache_bytes={} decoded_resident_ceiling_bytes={}",
360        entries,
361        inventory.frequencies_hz.len(),
362        inventory.w_values_lambda.len(),
363        inventory.mueller_elements.len(),
364        inventory.parallactic_angles_deg.len(),
365        prepared_cache_bytes,
366        deployment.resident_bytes,
367    );
368    let budget =
369        casa_imaging_runtime::PreparedArtifactBudget::new(prepared_cache_bytes, entries, 8 << 20)?;
370    let store = Arc::new(PreparedArtifactStore::open(
371        &deployment.private_root,
372        &deployment.storage_domain,
373        budget,
374    )?);
375    let owner =
376        super::PlanningRegistry::new(runtime.registry, runtime.implementation.clone(), problem);
377    let prepared = cache.prepared_cells(&store, &owner, &runtime.implementation, problem)?;
378    let largest_cell = prepared
379        .iter()
380        .map(PreparedAwCell::decoded_resident_bytes)
381        .collect::<Option<Vec<_>>>()
382        .and_then(|bytes| bytes.into_iter().max())
383        .ok_or_else(|| boxed("AW decoded cell residency overflowed"))?;
384    if largest_cell > deployment.resident_bytes {
385        return Err(Box::new(io::Error::other(format!(
386            "largest AW decoded cell requires {largest_cell} bytes, exceeding the {} byte ceiling",
387            deployment.resident_bytes
388        ))));
389    }
390    let decoder_workspace_bytes = prepared
391        .iter()
392        .map(PreparedAwCell::decoder_workspace_bytes)
393        .collect::<Option<Vec<_>>>()
394        .and_then(|bytes| bytes.into_iter().max())
395        .ok_or_else(|| boxed("AW decoder workspace overflowed"))?;
396    let mut receipts = Vec::new();
397
398    let mut ordered = prepared.to_vec();
399    ordered.sort_by_key(|cell| cell.descriptor().identity().as_bytes());
400    let descriptors = ordered
401        .iter()
402        .map(|cell| cell.descriptor().clone())
403        .collect::<Vec<_>>();
404    let (catalog_outcome, catalog_receipt) = run_catalog(
405        problem,
406        runtime,
407        Arc::clone(&store),
408        CatalogInput::Reuse(descriptors),
409    )?;
410    receipts.push(catalog_receipt);
411
412    let mut artifacts = Vec::new();
413    let mut reusable = BTreeSet::new();
414    let CatalogPhaseResult::Reused(catalog_outcome) = catalog_outcome else {
415        return Err(boxed("AW reuse phase returned cold artifacts"));
416    };
417    let outcomes = catalog_outcome.into_entries();
418    if outcomes.len() != ordered.len() {
419        return Err(boxed("AW catalog reuse omitted an entry"));
420    }
421    for (cell, outcome) in ordered.iter().zip(outcomes) {
422        if outcome.identity() != cell.descriptor().identity() {
423            return Err(boxed("AW catalog reuse changed descriptor order"));
424        }
425        match outcome.into_outcome() {
426            PreparedArtifactReuseOutcome::Reused(artifact) => {
427                reusable.insert(artifact.identity());
428                artifacts.push((cell.clone(), artifact));
429            }
430            PreparedArtifactReuseOutcome::Rejected(_) => {}
431        }
432    }
433    if reusable.len() != ordered.len() {
434        let (loaded, receipt) = run_catalog(
435            problem,
436            runtime,
437            Arc::clone(&store),
438            CatalogInput::Import {
439                cache: Arc::clone(&cache),
440                cells: ordered,
441                reusable,
442                source_domain: &deployment.storage_domain,
443            },
444        )?;
445        receipts.push(receipt);
446        let CatalogPhaseResult::Imported(loaded) = loaded else {
447            return Err(boxed("AW cold phase returned reuse outcomes"));
448        };
449        artifacts = loaded;
450    }
451    artifacts.sort_by_key(|(cell, _)| cell.metadata().identity().as_bytes());
452    if artifacts.len() != entries {
453        return Err(boxed("AW preparation omitted a catalog artifact"));
454    }
455    let reader = PreparedArtifactReaderFactory::new(
456        Arc::clone(&store),
457        artifacts
458            .into_iter()
459            .map(|(cell, artifact)| (cell.descriptor().clone(), artifact))
460            .collect(),
461        runtime.implementation.clone(),
462        u64::try_from(deployment.resident_bytes)?,
463        u64::try_from(decoder_workspace_bytes)?,
464    )?;
465    Ok(PreparedAwPhase {
466        catalog,
467        prepared,
468        reader,
469        conjugate_beams: deployment.conjugate_beams,
470        resident_bytes: deployment.resident_bytes,
471        receipts,
472    })
473}

475fn run_catalog(
476    problem: &CompiledProblem,
477    runtime: &ApplicationRuntime,
478    store: Arc<PreparedArtifactStore>,
479    input: CatalogInput<'_>,
480) -> Result<(CatalogPhaseResult, ExecutionReceipt), ApplicationError> {
481    let producer = casa_imaging_runtime::WorkNodeId::new("prepared-phase-producer");
482    let commit = casa_imaging_runtime::WorkNodeId::new("prepared-phase-commit");
483    let (descriptors, cold) = match input {
484        CatalogInput::Reuse(descriptors) => (descriptors, None),
485        CatalogInput::Import {
486            cache,
487            cells,
488            reusable,
489            source_domain,
490        } => {
491            let sources = cells
492                .iter()
493                .map(|cell| {
494                    (!reusable.contains(&cell.descriptor().identity()))
495                        .then(|| cache.import_source(cell, source_domain, producer.clone()))
496                        .transpose()
497                })
498                .collect::<Result<Vec<_>, _>>()?;
499            let descriptors = cells.iter().map(|cell| cell.descriptor().clone()).collect();
500            (
501                descriptors,
502                Some(ColdCatalog {
503                    cache,
504                    cells,
505                    sources,
506                }),
507            )
508        }
509    };
510    let reuse = cold.is_none();
511    let mut fragment = PreparedArtifactCatalogPlanFragment::new(
512        &descriptors,
513        &store,
514        producer.clone(),
515        commit.clone(),
516        runtime.implementation.clone(),
517    )?;
518    if let Some(cold) = &cold {
519        fragment = fragment.with_import_sources(&cold.sources)?;
520    }
521    let id = fragment.work_implementation_id()?;
522    let sources = cold
523        .iter()
524        .flat_map(|cold| &cold.sources)
525        .flatten()
526        .map(|source| (source.identity(), source.source_read_bytes()))
527        .collect();
528    let registry = PhaseRegistry {
529        id: runtime.registry,
530        metadata: ImplementationContractMetadata::new(
531            problem.problem_id(),
532            problem.numerics_id(),
533            problem.required_capabilities().clone(),
534        ),
535        implementations: BTreeMap::from([
536            (
537                id.clone(),
538                PhaseImplementation::Catalog(Box::new(CatalogAdapter {
539                    id,
540                    store: Arc::clone(&store),
541                    input: CatalogOperation::Exact { descriptors, cold },
542                    result: Mutex::new(None),
543                })),
544            ),
545            (
546                runtime.implementation.clone(),
547                PhaseImplementation::Base {
548                    id: runtime.implementation.clone(),
549                    sources,
550                },
551            ),
552        ]),
553        prepared_artifact: crate::prepared_aw_registration(runtime.implementation.clone()),
554    };
555    let catalog = registry.catalog();
556    let CatalogOperation::Exact { descriptors, cold } = &catalog.input else {
557        unreachable!("exact catalog constructed above")
558    };
559    let base = PreparedArtifactPlanFragment::standalone_base(
560        problem,
561        &registry,
562        runtime.implementation.clone(),
563        &descriptors[0],
564        &store,
565        runtime.stage_nanos,
566        runtime.confidence_parts_per_million,
567    )?;
568    let mut fragment = PreparedArtifactCatalogPlanFragment::new(
569        descriptors,
570        &store,
571        producer,
572        commit,
573        runtime.implementation.clone(),
574    )?;
575    if let Some(cold) = cold {
576        fragment = fragment.with_import_sources(&cold.sources)?;
577    }
578    let physical = fragment.compose(&base)?;
579    run_phase(
580        problem,
581        runtime,
582        registry,
583        physical,
584        reuse,
585        u64::from(!reuse),
586    )
587}

589fn run_phase(
590    problem: &CompiledProblem,
591    runtime: &ApplicationRuntime,
592    registry: PhaseRegistry,
593    physical: casa_imaging_runtime::PhysicalWorkBinding,
594    reuse: bool,
595    phase: u64,
596) -> Result<(CatalogPhaseResult, ExecutionReceipt), ApplicationError> {
597    let execution_plan = plan(
598        problem,
599        PlanningBindings::new(
600            runtime.registry,
601            runtime.resource_policy.clone(),
602            runtime.cost_model,
603        ),
604        &runtime.authority,
605        &registry,
606        &runtime.receipts,
607        move |_, _| Ok::<_, std::convert::Infallible>(vec![physical]),
608    )?;
609    let executable =
610        casa_imaging_reconstruction::ExecutableModelProblem::from_compiled(problem.clone())?;
611    let current = RunBindings::new(
612        problem.inputs().clone(),
613        &runtime.resource_policy,
614        runtime.cost_model.profile_id(),
615    );
616    let attempt = aw_attempt(runtime.attempts[0], phase);
617    let execution = run(
618        &executable,
619        &execution_plan,
620        &current,
621        &registry,
622        &runtime.authority,
623        &mut RunToCompletion,
624        runtime
625            .receipts
626            .bind(ExecutionProvenance::new(attempt, runtime.build)),
627    );
628    if let Err(error) = execution
629        && !(reuse
630            && matches!(
631                error,
632                RunError::Evidence(ExecutionEvidenceError::RejectedArtifact { .. })
633            ))
634    {
635        return Err(Box::new(error));
636    }
637    Ok((registry.into_result()?, runtime.receipts.open(attempt)?))
638}

640fn aw_attempt(base: ExecutionAttemptId, phase: u64) -> ExecutionAttemptId {
641    use sha2::{Digest, Sha256};
642    let mut hash = Sha256::new();
643    hash.update(b"casa-rs:aw-preparation-attempt:v1");
644    hash.update(base.as_bytes());
645    hash.update(phase.to_le_bytes());
646    ExecutionAttemptId::from_sha256(hash.finalize().into())
647}
