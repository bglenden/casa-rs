3//! The robust noise of a residual plane, as CASA's `tclean` estimates it.
4//!
5//! tclean estimates the noise its `nsigma` threshold and its automatic mask
6//! read in one of two ways, chosen by `fastnoise`:
7//!
8//! * **fast** (`fastnoise=True`, `SDMaskHandler::calcImageStatistics`):
9//!   casacore's classical median and median absolute deviation of every
10//!   valid pixel;
11//! * **robust** (`fastnoise=False`,
12//!   `SDMaskHandler::calcRobustImageStatistics`): a plane with no clean
13//!   mask, or whose mask covers the whole first plane, takes casacore's
14//!   Chauvenet-criterion statistics of its valid pixels
15//!   (`ImageStatsCalculator::configureChauvenet(-1, 5)`); any other plane
16//!   takes the classical statistics of the valid pixels outside its mask.
17//!
18//! The noise is 1.4826 times the median absolute deviation, CASA's literal
19//! factor (`SIImageStore::calcRobustRMS`).
20
21use crate::RobustNoise;
22pub use casa_imaging_model::NoiseEstimate;
23
24/// CASA's factor from the median absolute deviation to a Gaussian RMS
25/// (`mads*1.4826` in `SDMaskHandler` and `SIImageStore::calcRobustRMS`).
26pub const CASA_MAD_TO_RMS: f64 = 1.4826;
27
28/// Chauvenet iterations tclean allows (`configureChauvenet(-1.0, 5)`).
29const CHAUVENET_ITERATIONS: usize = 5;
30
31/// casacore `ZScoreCalculator`'s table of the largest sample size each
32/// z-score is the Chauvenet criterion of.
33const MAX_ZSCORES: [(u64, f64); 14] = [
34    (0, 0.5),
35    (1, 1.0),
36    (3, 1.5),
37    (10, 2.0),
38    (40, 2.5),
39    (185, 3.0),
40    (1_074, 3.5),
41    (7_893, 4.0),
42    (73_579, 4.5),
43    (872_138, 5.0),
44    (13_165_126, 5.5),
45    (253_398_672, 6.0),
46    (6_225_098_696, 6.5),
47    (195_341_107_722, 7.0),
48];
49
50/// The classical median and scaled median absolute deviation of
51/// `samples`, which it reorders; `None` when there are none.
52#[must_use]
53pub fn classical_noise(samples: &mut [f64]) -> Option<RobustNoise> {
54    let median = median_in_place(samples)?;
55    for sample in samples.iter_mut() {
56        *sample = (*sample - median).abs();
57    }
58    let deviation = median_in_place(samples)?;
59    Some(RobustNoise {
60        median,
61        rms: CASA_MAD_TO_RMS * deviation,
62    })
63}

65/// casacore's `ChauvenetCriterionStatistics` with tclean's configuration
66/// (z-score from the sample size, five iterations): the median and scaled
67/// median absolute deviation of the samples left inside
68/// `mean ± z·stddev` once clipping converges or the iterations run out.
69/// `None` when there are no samples.
70#[must_use]
71pub fn chauvenet_noise(mut samples: Vec<f64>) -> Option<RobustNoise> {
72    // `ChauvenetCriterionStatistics::_setRange`: the classical statistics
73    // of every sample set the first range; each later pass recomputes them
74    // inside the range until the count stops changing.
75    let mut range: Option<(f64, f64)> = None;
76    let mut previous_count = None;
77    for _ in 0..=CHAUVENET_ITERATIONS {
78        let (count, mean, stddev) = moments(&samples, range);
79        if previous_count == Some(count) {
80            break;
81        }
82        let z = max_zscore(count as u64);
83        range = Some((mean - z * stddev, mean + z * stddev));
84        previous_count = Some(count);
85    }
86    if let Some((low, high)) = range {
87        samples.retain(|value| *value >= low && *value <= high);
88    }
89    classical_noise(&mut samples)
90}

92/// tclean's noise of one plane: the fast estimate of the `valid` values,
93/// or the robust one with `mask` the plane's clean mask (`None` or empty
94/// for none) and `mask_full` whether the mask covers the image's whole
95/// first plane (`SDMaskHandler::calcRobustImageStatistics`). `None` when
96/// no value counts.
97#[must_use]
98pub fn plane_noise(
99    values: &[f64],
100    valid: &[bool],
101    estimate: NoiseEstimate,
102    mask: Option<&[bool]>,
103    mask_full: bool,
104) -> Option<RobustNoise> {
105    let selected = |outside: Option<&[bool]>| {
106        values
107            .iter()
108            .enumerate()
109            .filter(|(index, _)| valid[*index] && outside.is_none_or(|mask| !mask[*index]))
110            .map(|(_, value)| *value)
111            .collect::<Vec<_>>()
112    };
113    match (estimate, mask) {
114        (NoiseEstimate::Fast, _) => classical_noise(&mut selected(None)),
115        (NoiseEstimate::Robust, Some(mask)) if !mask_full && mask.iter().any(|set| *set) => {
116            classical_noise(&mut selected(Some(mask)))
117        }
118        (NoiseEstimate::Robust, _) => chauvenet_noise(selected(None)),
119    }
120}

122/// casacore `ZScoreCalculator::getMaxZScore`: the z-score beyond which
123/// fewer than half a sample of `count` normal samples is expected, found
124/// by bisection between the table's neighbours of `count`, as a fresh
125/// process finds it (casacore caches each answer in its table, so a long
126/// process may land elsewhere in the same one-count interval).
127fn max_zscore(count: u64) -> f64 {
128    if let Some((_, z)) = MAX_ZSCORES.iter().find(|(n, _)| *n == count) {
129        return *z;
130    }
131    let (mut low, mut high) = match MAX_ZSCORES.iter().position(|(n, _)| *n > count) {
132        Some(upper) => (MAX_ZSCORES[upper - 1].1, MAX_ZSCORES[upper].1),
133        None => {
134            let last = MAX_ZSCORES[MAX_ZSCORES.len() - 1].1;
135            let mut z = last + 0.5;
136            loop {
137                let reached = zscore_to_count(z);
138                if reached == count {
139                    return z;
140                }
141                if reached > count {
142                    break (last, z);
143                }
144                z += 0.5;
145            }
146        }
147    };
148    let mut z = 0.5 * (low + high);
149    loop {
150        let reached = zscore_to_count(z);
151        if reached == count || near(low, high) {
152            return z;
153        }
154        if reached > count {
155            high = z;
156        } else {
157            low = z;
158        }
159        z = 0.5 * (low + high);
160    }
161}

163/// casacore `ZScoreCalculator::zscoreToNpts`.
164fn zscore_to_count(z: f64) -> u64 {
165    (0.5 / libm::erfc(z / std::f64::consts::SQRT_2)) as u64
166}

168/// casacore `near(a, b)` at its default relative tolerance of 1e-13, for
169/// positive arguments.
170fn near(a: f64, b: f64) -> bool {
171    a == b || (a - b).abs() <= 1.0e-13 * a.abs().max(b.abs())
172}
173
174/// The count, mean and sample standard deviation (casacore's `stddev`,
175/// over `n − 1`) of the samples inside the inclusive `range`.
176fn moments(samples: &[f64], range: Option<(f64, f64)>) -> (usize, f64, f64) {
177    let (mut count, mut mean, mut scatter) = (0_usize, 0.0, 0.0);
178    for value in samples {
179        if range.is_some_and(|(low, high)| *value < low || *value > high) {
180            continue;
181        }
182        count += 1;
183        let previous = mean;
184        mean += (value - previous) / count as f64;
185        scatter += (value - previous) * (value - mean);
186    }
187    let variance = if count > 1 {
188        scatter / (count - 1) as f64
189    } else {
190        0.0
191    };
192    (count, mean, variance.sqrt())
193}

195/// casacore's median: the middle value, or the mean of the two middle
196/// values of an even count. Reorders `values`.
197fn median_in_place(values: &mut [f64]) -> Option<f64> {
198    if values.is_empty() {
199        return None;
200    }
201    let even = values.len().is_multiple_of(2);
202    let middle = values.len() / 2;
203    let (below, upper, _) = values.select_nth_unstable_by(middle, f64::total_cmp);
204    let upper = *upper;
205    if !even {
206        return Some(upper);
207    }
208    let lower = below.iter().copied().fold(f64::NEG_INFINITY, f64::max);
209    Some(0.5 * (lower + upper))
210}

212#[cfg(test)]
213mod tests {
214    use super::*;
215
216    #[test]
217    fn classical_noise_averages_the_two_middle_values_of_an_even_count() {
218        let mut values = [4.0, 1.0, 3.0, 2.0];
219        let noise = classical_noise(&mut values).expect("samples");
220        assert_eq!(noise.median, 2.5);
221        // Deviations 1.5, 1.5, 0.5, 0.5: their median is 1.
222        assert_eq!(noise.rms, CASA_MAD_TO_RMS);
223        assert_eq!(classical_noise(&mut []), None);
224    }
225
226    #[test]
227    fn max_zscore_follows_casacores_table_and_bisection() {
228        assert_eq!(max_zscore(185), 3.0);
229        assert_eq!(max_zscore(0), 0.5);
230        // Between table entries the z-score is the criterion's crossing.
231        for count in [2_u64, 50, 10_000, 1_000_000, 300_000_000_000] {
232            let z = max_zscore(count);
233            assert!(zscore_to_count(z) == count || zscore_to_count(z * (1.0 + 1e-9)) >= count);
234            assert!(zscore_to_count(z * (1.0 - 1e-6)) <= count, "{count}");
235        }
236        assert!(max_zscore(10_000) > 4.0 && max_zscore(10_000) < 4.5);
237    }
238
239    #[test]
240    fn chauvenet_clips_outliers_before_the_median() {
241        // A symmetric core and one far outlier: the classical median moves,
242        // Chauvenet's does not.
243        let mut values = (0..101)
244            .map(|i| f64::from(i - 50) * 0.01)
245            .collect::<Vec<_>>();
246        values.push(1000.0);
247        let clipped = chauvenet_noise(values.clone()).expect("samples");
248        assert_eq!(clipped.median, 0.0);
249        let classical = classical_noise(&mut values).expect("samples");
250        assert!(classical.median > 0.0);
251    }
252
253    #[test]
254    fn robust_noise_excludes_the_mask_unless_it_is_empty_or_full() {
255        let values = [1.0, 2.0, 3.0, 100.0, 200.0];
256        let valid = [true; 5];
257        let mask = [false, false, false, true, true];
258        let outside = plane_noise(&values, &valid, NoiseEstimate::Robust, Some(&mask), false)
259            .expect("samples");
260        assert_eq!(outside.median, 2.0);
261        let fast =
262            plane_noise(&values, &valid, NoiseEstimate::Fast, Some(&mask), false).expect("samples");
263        assert_eq!(fast.median, 3.0);
264        // A full mask, or none, takes Chauvenet over every valid value.
265        let full = plane_noise(&values, &valid, NoiseEstimate::Robust, Some(&mask), true);
266        let none = plane_noise(&values, &valid, NoiseEstimate::Robust, None, false);
267        assert_eq!(full, none);
268        assert_eq!(none, chauvenet_noise(values.to_vec()));
269    }
