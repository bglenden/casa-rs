2//! `setoptions(ftmachine='sd')`: total power, as CASA's simulator predicts
3//! it (`SDGrid`, built by `Simulator::createSkyEquation`), by sampling the
4//! model in the image plane.
5//!
6//! For a row pointed at pixel `(x, y)` of the model, `SDGrid::get` calls
7//! `dgridsd` (`fgridsd.f`), the model summed through the kernel around the
8//! pointing's nearest pixel; the visibility is the sum's conjugate.
9//! `casa-imaging-singledish` owns the kernels and that sum, which the
10//! single-dish gridder shares ([`KernelTable::degrid`]):
11//! - `BOX`: 1 out to a pixel, reaching the nearest pixel only;
12//! - `SF`: CASA's spheroid `(1 − ν²) grdsf(ν)` over 3 pixels;
13//! - `PB`: the telescope's power pattern along the x axis, at the frequency
14//!   of model plane 0 for every channel ([`KernelTable::primary_beam`]).
15//!
16//! Stokes planes become the selected receptor products through the shared
17//! polarization route, and SDGrid conjugates each sampled coherency.
18//! Channels select their nearest model plane; off-axis channels stay zero.
19//! Fixed frames share samples at identical pointings. Local image frames
20//! use each row's epoch and antenna position; saved image/squinted beams
21//! retain their orientation at CASA's requested angle threshold.
22//!
23//! Only autocorrelations are predicted, and flagged rows are not. CASA
24//! skips a row with a positive uvw component, which leaves a cross
25//! correlation with no positive component predicted as if it were one dish;
26//! casa-rs predicts autocorrelations only (`docs/CASA (C++) bugs.md`).
27

38/// The `SDGrid` kernel `setoptions(gridfunction=…)` names (`SDGrid::init`).
39#[derive(Debug, Clone, Copy, PartialEq, Eq)]
40pub enum SingleDishKernel {
41    /// `PB`: the telescope's primary beam.
42    Pb,
43    /// `BOX`: the nearest pixel.
44    Box,
45    /// `SF`: CASA's prolate spheroid over 3 pixels.
46    Spheroidal,
47    /// CASA's default Gaussian kernel.
48    Gauss,
49    /// CASA's default Gaussian-times-jinc kernel.
50    Gjinc,
51}
52
53impl SingleDishKernel {
54    /// The kernel `gridfunction` names, in any case; `None` for the others
55    /// (an unknown name is refused at prediction).
56    #[must_use]
57    pub fn from_grid_function(grid_function: &str) -> Option<Self> {
58        match grid_function.to_lowercase().as_str() {
59            "pb" => Some(Self::Pb),
60            "box" => Some(Self::Box),
61            "sf" => Some(Self::Spheroidal),
62            "gauss" => Some(Self::Gauss),
63            "gjinc" => Some(Self::Gjinc),
64            _ => None,
65        }
66    }
67
68    /// The kernel with the simulator's support, CASA's default.
69    fn grid_function(self) -> GridFunction {
70        match self {
71            Self::Pb => GridFunction::PrimaryBeam,
72            Self::Box => GridFunction::BOX,
73            Self::Spheroidal => GridFunction::SPHEROIDAL,
74            Self::Gauss => GridFunction::Gauss {
75                hwhm: std::f64::consts::LN_2.sqrt(),
76                truncate: (3. * std::f64::consts::LN_2.sqrt()) as f32,
77            },
78            Self::Gjinc => GridFunction::Gjinc {
79                hwhm: 2.52 * std::f64::consts::LN_2.sqrt(),
80                jwidth: 1.55,
81                truncate: None,
82            },
83        }
84    }
85
86    /// The kernel's table on `model`'s pixels; `beam` is the `PB` kernel's
87    /// pattern.
88    fn table(self, model: &SkyModel, beam: Option<&BeamPattern>) -> KernelTable {
89        match (self.grid_function(), beam) {
90            (GridFunction::PrimaryBeam, Some(beam)) => {
91                KernelTable::primary_beam(beam, model.increment_rad[0], model.plane_frequency_hz(0))
92            }
93            (GridFunction::PrimaryBeam, None) => unreachable!("a PB kernel has a beam"),
94            (function, _) => KernelTable::new(function).expect("a kernel of fixed support"),
95        }
96    }
97}

103    pub(crate) frequencies_hz: Vec<f64>,

108/// Where a total-power prediction samples the model: each distinct pointing
109/// direction of the scans it covers, and each scan's pointing. A scan
110/// points at its phase centre, which `observemany` can move off its field
111/// (the POINTING table casa-rs writes follows it).
112#[derive(Debug)]
113pub(crate) struct SingleDishPointings {
114    /// Each distinct pointing's J2000 direction, rad.
115    pub(crate) directions_rad: Vec<[f64; 2]>,
116    /// `[scan]`: the scan's pointing, an index into `directions_rad`.
117    pub(crate) of_scan: Vec<usize>,
118    /// Epoch/antenna-dependent pixels, where a local image frame needs them.
119    pub(crate) pixels: Option<Vec<[f64; 2]>>,
120    /// Sorted (scan, stored time bits, antenna) to prepared sample.
121    pub(crate) visits: Vec<((usize, u64, usize), usize)>,
122    pub(crate) angles: Vec<f64>,
123}
124
125impl SingleDishPointings {
126    /// The pointings of scans pointed at `phase_centres_rad`, in scan order.
127    pub(crate) fn of_scans(phase_centres_rad: impl ExactSizeIterator<Item = [f64; 2]>) -> Self {
128        let mut directions_rad: Vec<[f64; 2]> = Vec::new();
129        let of_scan = phase_centres_rad
130            .map(|centre| {
131                let same = |known: &[f64; 2]| known.map(f64::to_bits) == centre.map(f64::to_bits);
132                directions_rad.iter().position(same).unwrap_or_else(|| {
133                    directions_rad.push(centre);
134                    directions_rad.len() - 1
135                })
136            })
137            .collect();
138        Self {
139            directions_rad,
140            of_scan,
141            pixels: None,
142            visits: Vec::new(),
143            angles: Vec::new(),
144        }
145    }
146}

148/// A model prepared for total-power prediction: each pointing's samples in
149/// each window.
150#[derive(Debug)]
151pub(crate) struct SingleDishPrediction {
152    /// `[pointing][window]`: `[channel][correlation]`.
153    samples: Vec<Vec<Vec<Complex32>>>,
154    /// `[scan]`: the scan's pointing.
155    of_scan: Vec<usize>,
156    visits: Vec<((usize, u64, usize), usize)>,
157    /// The bytes [`Self::bytes`] counts.
158    _reservation: Reservation,
159}
160
161impl SingleDishPrediction {
162    /// Bytes a prediction of `windows` at `pointings` holds, known before it
163    /// is formed: the samples and the scans' pointings.
164    pub(crate) fn bytes(windows: &[SingleDishWindow], pointings: &SingleDishPointings) -> u64 {
165        let per_pointing: usize = size_of::<Vec<Vec<Complex32>>>()
166            + windows
167                .iter()
168                .map(|window| {
169                    size_of::<Vec<Complex32>>()
170                        + window.frequencies_hz.len()
171                            * window.correlations.len()
172                            * size_of::<Complex32>()
173                })
174                .sum::<usize>();
175        (pointings.directions_rad.len() * per_pointing
176            + pointings.of_scan.len() * size_of::<usize>()
177            + pointings.visits.capacity() * size_of::<((usize, u64, usize), usize)>())
178            as u64
179    }
180
181    /// Bytes preparing the prediction holds beside it:
182    /// - the kernel, as tabulated and as kept (`beam` is the `PB` kernel's
183    ///   pattern, `None` for the others);
184    /// - a model plane read, with the beam's power plane and their product
185    ///   when `setvp(dovp=True)` applies it first, and the image's tile
186    ///   cache ([`SkyModel::cache_bytes`]);
187    /// - each window channel's plane, and each pointing's direction, pixel
188    ///   and sample of the plane read.
189    pub(crate) fn preparation_bytes(
190        model: &SkyModel,
191        kind: SingleDishKernel,
192        transformed: bool,
193        beam: Option<&BeamPattern>,
194        (windows, pointings): (&[SingleDishWindow], &SingleDishPointings),
195    ) -> u64 {
196        // The PB kernel's extent, or the selected fixed kernel's support.
197        let kernel = match beam {
198            Some(beam) => GridFunction::PrimaryBeam.forming_bytes(
199                beam.support_pixels(model.increment_rad[0], model.plane_frequency_hz(0)),
200            ),
201            None => kind.grid_function().forming_bytes(0),
202        };
203        let channels: usize = windows.iter().map(|w| w.frequencies_hz.len()).sum();
204        let planes_of =
205            windows.len() * size_of::<Vec<Option<usize>>>() + channels * size_of::<Option<usize>>();
206        let per_pointing = 2 * size_of::<[f64; 2]>() + size_of::<f32>() + size_of::<f64>();
207        kernel
208            + (planes_of + pointings.directions_rad.len() * per_pointing) as u64
209            + if transformed { 4 } else { 2 } * model.plane_bytes()
210            + model.cache_bytes()
211            + windows.len() as u64 * 4096 // At most four products and Stokes coordinates per route.
212    }
213
214    /// Prepare `model` with `kernel` for `windows` at `pointings`, holding
215    /// `reservation`, its admission of [`Self::bytes`]. `beams` are the
216    /// telescope's pattern, which the `PB` kernel needs, and the pattern
217    /// `setvp(dovp=True)` applies to the model first, centred on each
218    /// pointing, as CASA's `SkyEquation` does whatever the kernel (so a
219    /// `PB` kernel then weights the model by the power pattern twice).
220    ///
221    /// # Errors
222    ///
223    /// The model's pixels could not be read, or a pointing could not be
224    /// placed on them.
225    pub(crate) fn new(
226        model: &SkyModel,
227        kernel: SingleDishKernel,
228        (beam, voltage_pattern, image_beam): (
229            Option<&BeamPattern>,
230            Option<&BeamPattern>,
231            Option<&ImageBeam>,
232        ),
233        (windows, pointings): (&[SingleDishWindow], SingleDishPointings),
234        reservation: Reservation,
235    ) -> Result<Self, PredictError> {
236        let SingleDishPointings {
237            directions_rad,
238            of_scan,
239            pixels: supplied_pixels,
240            visits,
241            angles,
242        } = pointings;
243        let kernel = kernel.table(model, beam);
244        if model.shape().iter().any(|&n| n <= 2 * kernel.support()) {
245            return Err(PredictError::InvalidModel(format!(
246                "number of image pixels {:?} is smaller than or equal to gridding support {}; use a larger image",
247                model.shape(),
248                2 * kernel.support()
249            )));
250        }
251        let axis = SpectralAxis::new(
252            model.first_frequency_hz,
253            model.frequency_increment_hz,
254            u32::try_from(model.planes())
255                .map_err(|_| PredictError::UnsupportedModel("too many model planes".into()))?,
256        )?;
257        // Each window channel's model plane (`matchChannel`), or none.
258        let planes_of: Vec<Vec<Option<usize>>> = windows
259            .iter()
260            .map(|window| {
261                window
262                    .frequencies_hz
263                    .iter()
264                    .map(|&frequency| axis.nearest_channel(frequency).map(|plane| plane as usize))
265                    .collect()
266            })
267            .collect();
268        let pixels = match supplied_pixels {
269            Some(pixels) => pixels,
270            None => directions_rad
271                .iter()
272                .map(|direction| model.pixel_of(*direction))
273                .collect::<Result<_, _>>()?,
274        };
275        // The samples, zero until a plane fills them. Sized exactly: they
276        // are kept, and charged by their length.
277        let mut samples: Vec<Vec<Vec<Complex32>>> = directions_rad
278            .iter()
279            .map(|_| {
280                windows
281                    .iter()
282                    .map(|window| {
283                        vec![
284                            Complex32::new(0.0, 0.0);
285                            window.frequencies_hz.len() * window.correlations.len()
286                        ]
287                    })
288                    .collect()
289            })
290            .collect();
291        // One plane read at a time, each pointing's sample of it written
292        // into the channels it feeds, so no plane × pointing matrix is held
293        // (#874 review).
294        let full_stokes = [
295            PolarizationCoordinate::StokesI,
296            PolarizationCoordinate::StokesQ,
297            PolarizationCoordinate::StokesU,
298            PolarizationCoordinate::StokesV,
299        ];
300        let stokes = if image_beam.is_some() {
301            &full_stokes[..]
302        } else {
303            model.polarizations()
304        };
305        let routing = windows
306            .iter()
307            .map(|window| {
308                let correlations = window
309                    .correlations
310                    .iter()
311                    .map(|&code| {
312                        Correlation::from_casacore_code(code)
313                            .expect("validated simulator correlation")
314                    })
315                    .collect::<Vec<_>>();
316                PolarizationRouting::compile(&correlations, stokes)
317            })
318            .collect::<Result<Vec<_>, _>>()?;
319        for plane in 0..model.planes() {
320            if !planes_of
321                .iter()
322                .flatten()
323                .any(|&mapped| mapped == Some(plane))
324            {
325                continue;
326            }
327            let sky = model.read_polarizations(plane)?;
328            for (pointing, (&direction, &pixel)) in directions_rad.iter().zip(&pixels).enumerate() {
329                let apparent;
330                let values = match image_beam {
331                    Some(beam) => {
332                        apparent = beam.apply_plane(
333                            model,
334                            plane,
335                            sky.clone(),
336                            direction,
337                            angles[pointing],
338                            None,
339                        )?;
340                        &apparent
341                    }
342                    None => &sky,
343                };
344                let mut sampled = [0_f32; 4];
345                for (sample, values) in sampled.iter_mut().zip(values) {
346                    *sample = match voltage_pattern {
347                        Some(pattern) => {
348                            let apparent =
349                                values * &super::beam::power_plane(pattern, model, plane, pixel);
350                            kernel.degrid(apparent.view(), pixel)
351                        }
352                        None => kernel.degrid(values.view(), pixel),
353                    };
354                }
355                for (((cells, window), planes), route) in samples[pointing]
356                    .iter_mut()
357                    .zip(windows)
358                    .zip(&planes_of)
359                    .zip(&routing)
360                {
361                    let correlations = window.correlations.len();
362                    for (channel, _) in planes
363                        .iter()
364                        .enumerate()
365                        .filter(|(_, mapped)| **mapped == Some(plane))
366                    {
367                        for (cell, &gpol) in cells[channel * correlations..][..correlations]
368                            .iter_mut()
369                            .zip(route.pol_map())
370                        {
371                            if let Some(gpol) = gpol {
372                                for (polarization, &value) in
373                                    sampled.iter().take(stokes.len()).enumerate()
374                                {
375                                    // dgridsd conjugates the sampled complex model.
376                                    let coefficient = route
377                                        .from_requested(usize::from(gpol), polarization)
378                                        .conj();
379                                    *cell += Complex32::new(
380                                        coefficient.re as f32,
381                                        coefficient.im as f32,
382                                    ) * value;
383                                }
384                            }
385                        }
386                    }
387                }
388            }
389        }
390        tracing::debug!(
391            support = kernel.support(),
392            bytes = kernel.bytes(),
393            "single-dish kernel"
394        );
395        Ok(Self {
396            samples,
397            of_scan,
398            visits,
399            _reservation: reservation,
400        })
401    }
402
403    pub(crate) fn needs_epoch(&self) -> bool {
404        !self.visits.is_empty()
405    }
406
407    pub(crate) fn samples_at(
408        &self,
409        scan: usize,
410        window: usize,
411        time: f64,
412        antenna: usize,
413    ) -> &[Complex32] {
414        if self.visits.is_empty() {
415            return self.samples(scan, window);
416        }
417        self.visits
418            .binary_search_by_key(&(scan, time.to_bits(), antenna), |v| v.0)
419            .ok()
420            .map_or(&[], |index| &self.samples[self.visits[index].1][window])
421    }
422
423    /// The samples of scan `scan`'s pointing in window `window`,
424    /// `[channel][correlation]`; empty for a scan not prepared.
425    pub(crate) fn samples(&self, scan: usize, window: usize) -> &[Complex32] {
426        self.of_scan
427            .get(scan)
428            .map_or(&[], |&pointing| self.samples[pointing][window].as_slice())
429    }
