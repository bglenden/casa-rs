11#[derive(Clone, Copy, Debug, Default)]
12pub(super) struct StoredLineFinder {
13    pub(super) enabled: bool,
14    pub(super) threshold: f32,
15    pub(super) average: u32,
16    pub(super) edges: [u32; 2],

19pub(super) struct SpectrumParameters {
20    pub(super) mask: String,
21    pub(super) settings: FitSettings,
22    pub(super) line_finder: StoredLineFinder,
23}
24
25impl SpectrumParameters {
26    pub(super) fn describe_basis(&self) -> String {
27        match &self.settings.basis {
28            SingleDishBaselineBasis::Polynomial { order, chebyshev } => {
29                let family = if *chebyshev { "chebyshev" } else { "poly" };
30                format!("{family} order={order}")
31            }
32            SingleDishBaselineBasis::CubicSpline { pieces } => format!("cspline npiece={pieces}"),
33            SingleDishBaselineBasis::Sinusoid(sinusoid) => {
34                format!("sinusoid nwave={:?}", sinusoid.variable_waves())
35            }
36        }
37    }
38}

40type SpectrumKey = (usize, usize);
41type ContextLimit = (usize, usize);
42
43pub(super) struct VariableParameters {
44    entries: BTreeMap<SpectrumKey, SpectrumParameters>,
45    context_limits: [Option<usize>; 4],
46}
47
48impl VariableParameters {
49    pub(super) fn open(path: &Path) -> Result<Self, String> {
50        let reader = BufReader::new(File::open(path).map_err(|error| {
51            format!(
52                "Cannot open baseline parameter file '{}': {error}",
53                path.display()
54            )
55        })?);
56        let mut entries = BTreeMap::new();
57        let mut context_limits = [None::<usize>; 4];
58        for (index, line) in reader.lines().enumerate() {
59            let line = line.map_err(|error| error.to_string())?;
60            if line.is_empty() || line.starts_with('#') {
61                continue;
62            }
63            let (key, parameters, (family, order)) =
64                parse_line(&line).map_err(|error| format!("line {}: {error}", index + 1))?;
65            context_limits[family] =
66                Some(context_limits[family].map_or(order, |old| old.max(order)));
67            entries.insert(key, parameters);
68        }
69        Ok(Self {
70            entries,
71            context_limits,
72        })
73    }
74
75    pub(super) fn get(&self, row: usize, polarization: usize) -> Option<&SpectrumParameters> {
76        self.entries.get(&(row, polarization))
77    }
78
79    pub(super) fn validate_contexts(&self, channels: usize) -> Result<(), String> {
80        let [polynomial, chebyshev, spline, wave] = self.context_limits;
81        if polynomial.is_some_and(|order| order >= channels)
82            || chebyshev.is_some_and(|order| order >= channels)
83            || spline.is_some_and(|pieces| pieces == 0 || pieces + 3 > channels)
84        {
85            return Err("sakura_CreateLSQFitContextFloat() failed -- InvalidArgument".into());
86        }
87        if let Some(wave) = wave {
88            sinusoid::validate_context(wave, channels)?;
89        }
90        Ok(())
91    }
92}

94fn parse_line(line: &str) -> Result<(SpectrumKey, SpectrumParameters, ContextLimit), String> {
95    let mut values = line.split(',');
96    let mut field = || values.next().unwrap_or_default();
97    let row = number(field(), "row")?;
98    let polarization = number(field(), "polarization")?;
99    let mask = field().to_owned();
100    let iterations = field();
101    if iterations.is_empty() {
102        return Err("Number of maximum clip iteration is mandatory".into());
103    }
104    let iterations: u16 = number(iterations, "clip iteration")?;
105    let clip_sigma = f64::from(optional(field(), "clip threshold", 3.0_f32)?);
106    let enabled = field() == "true";
107    let threshold = field();
108    let left = field();
109    let right = field();
110    let average = field();
111    let line_finder = if enabled {
112        StoredLineFinder {
113            enabled,
114            threshold: optional(threshold, "line threshold", 5.)?,
115            average: optional(average, "channel average limit", 0)?,
116            edges: [
117                optional(left, "left edge", 0)?,
118                optional(right, "right edge", 0)?,
119            ],
120        }
121    } else {
122        StoredLineFinder {
123            threshold: 5.,
124            ..Default::default()
125        }
126    };
127    let family = field();
128    let order = field();
129    let pieces = field();
130    let (basis, context) = match family {
131        "cspline" => {
132            if pieces.is_empty() {
133                return Err("Baseline type 'cspline' requires num_piece value.".into());
134            }
135            let pieces: usize = number(pieces, "num_piece")?;
136            if pieces > usize::from(u16::MAX) {
137                return Err("num_piece is larger than max of uint16_t".into());
138            }
139            (SingleDishBaselineBasis::CubicSpline { pieces }, (2, pieces))
140        }
141        "sinusoid" => {
142            let waves = bracketed_waves(line)?;
143            // Sakura's context takes a uint16_t maximum wavenumber.
144            let maximum = *waves.last().expect("nonempty wavenumbers") as u16;
145            (
146                SingleDishBaselineBasis::Sinusoid(SingleDishSinusoid::from_variable_waves(waves)),
147                (3, usize::from(maximum)),
148            )
149        }
150        _ => {
151            if order.is_empty() {
152                return Err("Baseline type 'poly' and 'chebyshev' require order value.".into());
153            }
154            let order: u16 = number(order, "polynomial order")?;
155            let order = usize::from(order);
156            let chebyshev = family == "chebyshev";
157            (
158                SingleDishBaselineBasis::Polynomial { order, chebyshev },
159                (usize::from(chebyshev), order),
160            )
161        }
162    };
163    Ok((
164        (row, polarization),
165        SpectrumParameters {
166            mask,
167            settings: FitSettings {
168                basis,
169                clip_sigma,
170                max_fits: usize::from(iterations.wrapping_add(1)),
171            },
172            line_finder,
173        },
174        context,
175    ))
176}

178fn number<T: std::str::FromStr>(text: &str, name: &str) -> Result<T, String> {
179    text.trim()
180        .parse()
181        .map_err(|_| format!("Invalid {name}: {text:?}"))
182}
183
184fn optional<T: std::str::FromStr>(text: &str, name: &str, default: T) -> Result<T, String> {
185    if text.is_empty() {
186        Ok(default)
187    } else {
188        number(text, name)
189    }
190}

192fn bracketed_waves(line: &str) -> Result<Vec<usize>, String> {
193    for tail in line.split('[').skip(1) {
194        let Some((contents, _)) = tail.split_once(']') else {
195            continue;
196        };
197        let mut waves = Vec::new();
198        if contents.split(',').all(|word| {
199            if !word.bytes().all(|byte| byte.is_ascii_digit()) {
200                return false;
201            }
202            match word.parse::<usize>() {
203                Ok(wave) => {
204                    waves.push(wave);
205                    true
206                }
207                Err(_) => false,
208            }
209        }) {
210            waves.sort_unstable();
211            waves.dedup();
212            return Ok(waves);
213        }
214    }
215    Err("Incorrect format or empty nwave list given. Use a list such as [1,2].".into())
216}
