29/// The pixels a domain's minor cycle may update.
30#[derive(Clone, Debug, PartialEq)]
31pub(super) enum DomainMask {
32    /// Every valid model pixel.
33    FullPlane,
34    /// The union of inclusive pixel boxes `[x0, y0, x1, y1]`.
35    Boxes(Vec<[usize; 4]>),
36    /// The non-zero pixels of a CASA image, reprojected onto the domain.
37    Image(PathBuf),
38    /// Row-major support over the domain's own grid.
39    PixelSupport(Box<[bool]>),
40    /// CASA auto-multithresh support from the current normal state.
41    AutoMultithresh(AutoMultithreshControls),
42}

44/// One image domain of the run.
45pub(super) struct PreparedImageDomain {
46    pub(super) role: ImageDomainRole,
47    pub(super) output: PathBuf,
48    pub(super) image_size: usize,
49    pub(super) direction: DirectionCoordinateSpec,
50    pub(super) coordinates: CoordinateSystem,
51    pub(super) mask: DomainMask,
52}
53
54impl PreparedImageDomain {
55    /// The model's specification of this domain: one facet, axes
56    /// direction, polarization, spectral.
57    pub(super) fn spec(&self) -> ImageDomainSpec {
58        ImageDomainSpec::new(
59            self.role.clone(),
60            ImageShape::new(self.image_size, self.image_size),
61            self.direction,
62            FacetLayout::Single,
63            AxisOrder::new([
64                ImageAxis::DirectionLongitude,
65                ImageAxis::DirectionLatitude,
66                ImageAxis::Polarization,
67                ImageAxis::Spectral,
68            ]),
69        )
70    }
71
72    /// Where the product sink writes this domain's images.
73    pub(super) fn output(&self) -> CasaImageDomainOutput {
74        CasaImageDomainOutput::new(
75            self.role.clone(),
76            self.output.clone(),
77            self.coordinates.clone(),
78        )
79    }
80}

82/// The main domain and the outlier file's, ordered by role.
83pub(super) fn prepare_domains(
84    request: &ImagingRequest,
85    centre: &Centre,
86    spectral: ImageSpectralCoordinate,
87    observation: &ObsInfo,
88) -> Result<Vec<PreparedImageDomain>, ApplicationError> {
89    let polarizations = &request.stokes;
90    let mut domains = vec![PreparedImageDomain {
91        role: ImageDomainRole::Main,
92        output: request.imagename.clone(),
93        image_size: request.imsize,
94        direction: centre.direction,
95        coordinates: image_coordinates(
96            centre.direction,
97            request.imsize,
98            polarizations,
99            spectral,
100            observation,
101        ),
102        mask: request_mask(request),
103    }];
104    if let Some(path) = request.outlierfile.as_deref() {
105        for outlier in read_outlier_domains(path, request.imsize, request.cell)? {
106            let centre = parse_phase_center_direction(&outlier.phase_center)?;
107            let direction = direction_spec(
108                outlier.image_size,
109                outlier.cell_arcsec,
110                DirectionFrame::J2000,
111                centre.longitude_rad(),
112                centre.latitude_rad(),
113            );
114            domains.push(PreparedImageDomain {
115                role: ImageDomainRole::Outlier(outlier.name),
116                output: outlier.output,
117                image_size: outlier.image_size,
118                direction,
119                coordinates: image_coordinates(
120                    direction,
121                    outlier.image_size,
122                    polarizations,
123                    spectral,
124                    observation,
125                ),
126                mask: outlier.mask,
127            });
128        }
129    }
130    domains.sort_by(|left, right| left.role.cmp(&right.role));
131    Ok(domains)
132}

134/// The main domain's mask: CASA auto-multithresh, the user mask image or
135/// boxes, or every pixel. The auto-multithresh controls the catalog does
136/// not carry take CASA's defaults (`smoothfactor = 1`, `cutthreshold =
137/// 0.01`, `minpercentchange = -1`).
138fn request_mask(request: &ImagingRequest) -> DomainMask {
139    match (request.usemask, &request.mask_image) {
140        (UseMask::AutoMultithresh, _) => DomainMask::AutoMultithresh(AutoMultithreshControls {
141            sidelobe_factor: request.sidelobethreshold,
142            noise_factor: request.noisethreshold,
143            low_noise_factor: request.lownoisethreshold,
144            negative_factor: request.negativethreshold,
145            minimum_beam_fraction: request.minbeamfrac,
146            smooth_factor: 1.0,
147            cut_threshold: 0.01,
148            grow_iterations: request.growiterations,
149            minimum_percent_change: -1.0,
150        }),
151        (UseMask::User, Some(path)) => DomainMask::Image(path.clone()),
152        (UseMask::User, None) if request.mask_box.is_empty() => DomainMask::FullPlane,
153        (UseMask::User, None) => DomainMask::Boxes(request.mask_box.clone()),
154    }
155}

157/// Each domain's mask plan, bound to its direction coordinate.
158pub(super) fn mask_plans(
159    domains: &[PreparedImageDomain],
160) -> Result<ImageDomainReconstructionMaskPlans, ApplicationError> {
161    Ok(ImageDomainReconstructionMaskPlans::new(
162        domains
163            .iter()
164            .map(|domain| mask_plan(domain.mask.clone(), domain.direction, domain.image_size))
165            .collect::<Result<Vec<_>, _>>()?,
166    )?)
167}

169fn mask_plan(
170    mask: DomainMask,
171    coordinate: DirectionCoordinateSpec,
172    image_size: usize,
173) -> Result<ReconstructionMaskPlan, ApplicationError> {
174    Ok(match mask {
175        DomainMask::FullPlane => ReconstructionMaskPlan::FullPlane { coordinate },
176        DomainMask::Boxes(boxes) => ReconstructionMaskPlan::Boxes {
177            coordinate,
178            boxes: boxes
179                .into_iter()
180                .map(|[x0, y0, x1, y1]| MaskBox::new([x0, y0], [x1, y1]))
181                .collect::<Result<Vec<_>, _>>()?,
182        },
183        DomainMask::Image(path) => reproject_image_mask(&path, coordinate, image_size)?,
184        DomainMask::PixelSupport(support) => {
185            if image_size
186                .checked_mul(image_size)
187                .is_none_or(|expected| support.len() != expected)
188            {
189                return Err(boxed("pixel mask support does not match its image domain"));
190            }
191            ReconstructionMaskPlan::Reprojected {
192                coordinate,
193                source_coordinate: coordinate,
194                source_shape: [image_size, image_size],
195                support,
196            }
197        }
198        DomainMask::AutoMultithresh(controls) => ReconstructionMaskPlan::AutoMultithresh {
199            coordinate,
200            controls,
201            completed_major_cycles: 0,
202            cycle_threshold_reached: false,
203            previous: None,
204            evolution_stopped: false,
205        },
206    })
207}

209/// The non-zero, unmasked, finite pixels of a one-plane CASA image,
210/// reprojected onto `target_spec`.
211fn reproject_image_mask(
212    path: &Path,
213    target_spec: DirectionCoordinateSpec,
214    target_size: usize,
215) -> Result<ReconstructionMaskPlan, ApplicationError> {
216    let (source_shape, source_coordinates, source_support) = match AnyPagedImage::open(path)? {
217        AnyPagedImage::Float32(image) => {
218            let mask = image.get_mask()?;
219            let support = image
220                .get()?
221                .iter()
222                .enumerate()
223                .map(|(index, value)| {
224                    value.is_finite()
225                        && *value != 0.0
226                        && mask.as_ref().is_none_or(|mask| mask[index])
227                })
228                .collect::<Vec<_>>();
229            (image.shape().to_vec(), image.coordinates().clone(), support)
230        }
231        AnyPagedImage::Float64(image) => {
232            let mask = image.get_mask()?;
233            let support = image
234                .get()?
235                .iter()
236                .enumerate()
237                .map(|(index, value)| {
238                    value.is_finite()
239                        && *value != 0.0
240                        && mask.as_ref().is_none_or(|mask| mask[index])
241                })
242                .collect::<Vec<_>>();
243            (image.shape().to_vec(), image.coordinates().clone(), support)
244        }
245        AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
246            return Err(boxed("reconstruction masks require a real CASA image"));
247        }
248    };
249    if source_shape.len() < 2
250        || source_shape[2..].iter().any(|extent| *extent != 1)
251        || source_support.len() != source_shape[0] * source_shape[1]
252    {
253        return Err(boxed(
254            "reconstruction mask must contain one two-dimensional direction plane",
255        ));
256    }
257    let index = source_coordinates
258        .find_coordinate(CoordinateType::Direction)
259        .ok_or_else(|| boxed("mask image has no direction coordinate"))?;
260    let source_spec = direction_model_spec(source_coordinates.coordinate(index))?;
261    let support = casa_imaging_reconstruction::reproject_mask_support(
262        source_spec,
263        [source_shape[0], source_shape[1]],
264        &source_support,
265        target_spec,
266        [target_size, target_size],
267    )?;
268    Ok(ReconstructionMaskPlan::Reprojected {
269        coordinate: target_spec,
270        source_coordinate: source_spec,
271        source_shape: [source_shape[0], source_shape[1]],
272        support,
273    })
274}

276fn direction_model_spec(
277    coordinate: &CoordinateModel,
278) -> Result<DirectionCoordinateSpec, ApplicationError> {
279    let CoordinateModel::Direction(direction) = coordinate else {
280        return Err(boxed("mask direction-coordinate lookup was inconsistent"));
281    };
282    if direction.projection().projection_type() != ProjectionType::SIN {
283        return Err(boxed(
284            "native mask reprojection currently requires SIN coordinates",
285        ));
286    }
287    let frame = match direction.direction_ref() {
288        DirectionRef::J2000 => DirectionFrame::J2000,
289        DirectionRef::B1950 => DirectionFrame::B1950,
290        DirectionRef::GALACTIC => DirectionFrame::Galactic,
291        DirectionRef::ICRS => DirectionFrame::Icrs,
292        _ => {
293            return Err(boxed(
294                "mask direction frame is not supported by native imaging",
295            ));
296        }
297    };
298    let reference = coordinate.reference_value();
299    let pixel = coordinate.reference_pixel();
300    let increment = coordinate.increment();
301    let pc = direction.pc_matrix();
302    Ok(DirectionCoordinateSpec::new(
303        Projection::Sin,
304        SkyDirection::new(frame, reference[0], reference[1]),
305        [pixel[0], pixel[1]],
306        [increment[0], increment[1]],
307        [[pc[[0, 0]], pc[[0, 1]]], [pc[[1, 0]], pc[[1, 1]]]],
308        [
309            direction.longpole().to_degrees(),
310            direction.latpole().to_degrees(),
311        ],
312    ))
313}

315/// Pixels a model plane holds across every domain, times `planes` and
316/// `polarizations`.
317pub(super) fn model_samples(
318    domains: &[PreparedImageDomain],
319    planes: usize,
320    polarizations: &[PolarizationCoordinate],
321) -> Result<usize, ApplicationError> {
322    domains
323        .iter()
324        .try_fold(0_usize, |total, domain| {
325            domain
326                .image_size
327                .saturating_mul(domain.image_size)
328                .checked_add(total)
329        })
330        .and_then(|samples| samples.checked_mul(planes))
331        .and_then(|samples| samples.checked_mul(polarizations.len()))
332        .ok_or_else(|| boxed("reconstruction model sample count overflowed"))
333}
