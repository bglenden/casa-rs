2//! Writing back: what an update session may write, and the values one
3//! delivered batch's samples receive (`docs/measurement-set/contracts.md`,
4//! section 7, "Updates").
5//!
6//! An update addresses the samples of a batch the same session delivered:
7//! the store maps them to its own native units (MAIN rows, Zarr chunks),
8//! and a sample with no native address (MSv2 padding) receives nothing. A
9//! batch's address ([`BatchAddress`]) outlives its buffers, so a consumer
10//! that hands a batch's samples to another thread writes them back without
11//! keeping the batch. The targets are declared when the session opens, so
12//! the structural preflight (the variable exists, the data groups sharing
13//! it are accepted) runs before any payload. Values are checked against the
14//! batch before anything is written; a refused write writes nothing of its
15//! batch.
16//!
17//! **Trial status.** This is the write-back trial (`docs/measurement-set/
18//! trial/README.md`, "Write-back"): one updater per traversal, written
19//! batch for batch in any order, boolean flags only; the slices of a
20//! [`TimeSplit`](crate::TimeSplit) each have their own updater, live at
21//! once.

33/// A role a session writes, of one data group.
34#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
35pub enum WritableRole {
36    /// The group's visibilities (MSv2 `DATA`, `CORRECTED_DATA` or
37    /// `MODEL_DATA`; MSv4 the group's correlated-data variable).
38    Visibilities,
39    /// The group's element flags (`FLAG`). Row flags (`FLAG_ROW`) and flag
40    /// categories are never written: a flag change cannot clear them.
41    Flags,
42    /// The group's weights (`WEIGHT`, `WEIGHT_SPECTRUM` or MSv4 `WEIGHT`).
43    Weights,
44}

46/// Whether a target may write a variable other data groups share: data
47/// groups commonly share their flags and weights, so writing one group's
48/// flags changes every group's.
49#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
50pub enum AliasScope {
51    /// Refuse the target when another group shares its variable.
52    #[default]
53    Refuse,
54    /// Write the shared variable, for every group that shares it.
55    Accept,
56}

58/// One thing an update session may write, declared when it opens.
59#[derive(Clone, Debug, PartialEq, Eq)]
60pub struct UpdateTarget {
61    /// The data group, by name.
62    pub data_group: String,
63    /// The role.
64    pub role: WritableRole,
65    /// Whether the variable may be shared with other groups.
66    pub aliases: AliasScope,
67    /// Create the group's visibility variable when the store lacks it,
68    /// initialized as CASA initializes it: `corrected` from the `base`
69    /// group's visibilities (`CORRECTED_DATA` from `DATA`), so what a pass
70    /// skips reads as uncorrected; `model` with a unit point source at the
71    /// phase centre (1, and 0 in the cross hands), so what a pass skips
72    /// reads as CASA's default model. Stores that cannot add a variable
73    /// refuse.
74    pub create: bool,
75}

77impl UpdateTarget {
78    /// A target that writes `role` of `data_group`, refusing aliases and
79    /// creating nothing.
80    #[must_use]
81    pub fn new(data_group: impl Into<String>, role: WritableRole) -> Self {
82        Self {
83            data_group: data_group.into(),
84            role,
85            aliases: AliasScope::Refuse,
86            create: false,
87        }
88    }

106/// What a store's updates guarantee on failure.
107#[derive(Clone, Copy, Debug, PartialEq, Eq)]
108pub struct UpdateGuarantees {
109    /// Each physical unit (an MSv4 chunk object) is replaced whole: a
110    /// failure or a killed process leaves the old unit or the new one,
111    /// never a mixture. Units already replaced stay replaced: there is no
112    /// transaction across units.
113    pub atomic_units: bool,
114    /// Another process is excluded while any of the session's updaters is
115    /// live (casacore's write lock on MAIN for MSv2). MSv4 has no
116    /// cross-process lock protocol.
117    pub cross_process_lock: bool,
118}

120/// How a weight patch treats channels the native weight shares: an MSv2
121/// `WEIGHT` is one value per row and correlation that every channel reads.
122#[derive(Clone, Copy, Debug, PartialEq, Eq)]
123pub enum WeightChannels {
124    /// Change the delivered channels only. Where the native weight is
125    /// shared by channels, this is refused unless the batch delivers every
126    /// channel and gives each sample and product one value: the patch must
127    /// not change a channel it does not address.
128    Delivered,
129    /// Change every channel of the partition: the patch gives each sample
130    /// and product one value, which every channel takes.
131    All,
132}

134/// The new values of one target for the samples, channels and products of
135/// the batch they are written against, each `[sample, channel, product]`.
136#[derive(Clone, Copy, Debug)]
137pub enum PatchValues<'a> {
138    /// Replace the visibilities where `written` holds, or every one when it
139    /// is `None`. An element it leaves keeps its stored value, and a
140    /// sample it leaves wholly is not touched: a pass that images some of
141    /// a batch's samples and channels writes back exactly those.
142    Visibilities {
143        /// The visibilities.
144        values: ArrayView3<'a, Complex32>,
145        /// Which elements are written.
146        written: Option<ArrayView3<'a, bool>>,
147    },
148    /// Change element flags: `set` sets and `clear` clears where true;
149    /// elements neither names keep their stored flag. A sample may not be
150    /// both set and cleared.
151    Flags {
152        /// Flags to set.
153        set: Option<ArrayView3<'a, bool>>,
154        /// Flags to clear.
155        clear: Option<ArrayView3<'a, bool>>,
156    },
157    /// Replace the weights.
158    Weights {
159        /// The weights.
160        values: ArrayView3<'a, f32>,
161        /// Which channels they change.
162        channels: WeightChannels,
163    },
164}
165
166impl<'a> PatchValues<'a> {
167    /// Replace every visibility.
168    #[must_use]
169    pub fn visibilities(values: ArrayView3<'a, Complex32>) -> Self {
170        Self::Visibilities {
171            values,
172            written: None,
173        }
174    }
175
176    /// The role these values write.
177    #[must_use]
178    pub fn role(&self) -> WritableRole {
179        match self {
180            Self::Visibilities { .. } => WritableRole::Visibilities,
181            Self::Flags { .. } => WritableRole::Flags,
182            Self::Weights { .. } => WritableRole::Weights,
183        }
184    }
185
186    /// Whether checking these values needs the batch's payload, not only
187    /// its address: clearing a flag or writing a weight must not turn
188    /// padding into an acquisition, and an MSv4 store knows its padding
189    /// only from the batch's weights ([`Updater::write_at`]).
190    #[must_use]
191    pub fn needs_payload(&self) -> bool {
192        match self {
193            Self::Visibilities { .. } => false,
194            Self::Flags { clear, .. } => clear.is_some(),
195            Self::Weights { .. } => true,
196        }
197    }
198
199    /// Whether the patch writes anything of `sample`: always for flags and
200    /// weights, and for visibilities when `written` holds one of its
201    /// elements.
202    #[must_use]
203    pub fn writes_sample(&self, sample: usize) -> bool {
204        match self {
205            Self::Visibilities {
206                written: Some(written),
207                ..
208            } => written
209                .index_axis(Axis(0), sample)
210                .iter()
211                .any(|written| *written),
212            _ => true,
213        }
214    }
215}

217/// Where a delivered batch's samples lie, without its payload: its
218/// partition, times, channels and correlation products, and the catalogue
219/// and weight rule it was delivered with ([`VisibilityBatch::address`]).
220///
221/// The address is owned, so it outlives the buffers the batch borrowed: a
222/// consumer that packs a batch's samples and hands them to another thread
223/// writes them back through [`Updater::write_at`] once those buffers hold
224/// the next batch. Only a delivered batch gives one, so an address names
225/// samples a store delivered, as a batch does.
226#[derive(Clone, Debug)]
227pub struct BatchAddress {
228    catalogue: Arc<Catalogue>,
229    partition: PartitionId,
230    time_scale: EpochRef,
231    time_origin: MjdHighPrec,
232    offsets_s: Vec<f64>,
233    first_channel: usize,
234    channels: usize,
235    correlations: Vec<Correlation>,
236    weight_rule: WeightRule,
237}
238
239impl VisibilityBatch<'_> {
240    /// The batch's address, which outlives its buffers.
241    #[must_use]
242    pub fn address(&self) -> BatchAddress {
243        BatchAddress {
244            catalogue: Arc::clone(self.catalogue),
245            partition: self.partition,
246            time_scale: self.time.reference(),
247            time_origin: self.time.origin(),
248            offsets_s: self.time.offsets_s().to_vec(),
249            first_channel: self.first_channel,
250            channels: self.channel_count(),
251            correlations: self.correlations.to_vec(),
252            weight_rule: self.group.weight_rule,
253        }
254    }
255}
256
257impl BatchAddress {
258    /// The catalogue the batch was delivered with: the store's opening that
259    /// delivered it.
260    #[must_use]
261    pub fn catalogue(&self) -> &Arc<Catalogue> {
262        &self.catalogue
263    }
264
265    /// The partition.
266    #[must_use]
267    pub fn partition(&self) -> PartitionId {
268        self.partition
269    }
270
271    /// The batch's times, in the partition's scale and origin.
272    #[must_use]
273    pub fn time(&self) -> EpochView<'_> {
274        EpochView::new(
275            self.time_scale,
276            self.time_origin,
277            ArrayView1::from(self.offsets_s.as_slice()),
278        )
279    }
280
281    /// Number of times.
282    #[must_use]
283    pub fn times(&self) -> usize {
284        self.offsets_s.len()
285    }
286
287    /// Samples per time: the partition's baselines.
288    #[must_use]
289    pub fn samples_per_time(&self) -> usize {
290        self.catalogue.partition(self.partition).baselines.len()
291    }
292
293    /// Number of samples.
294    #[must_use]
295    pub fn samples(&self) -> usize {
296        self.times() * self.samples_per_time()
297    }
298
299    /// The partition's native channels the batch holds.
300    #[must_use]
301    pub fn channels(&self) -> Range<usize> {
302        self.first_channel..self.first_channel + self.channels
303    }
304
305    /// The delivered correlation products, in the partition's order.
306    #[must_use]
307    pub fn correlations(&self) -> &[Correlation] {
308        &self.correlations
309    }
310
311    /// What the batch's weights were: stored, or made from the sigmas.
312    #[must_use]
313    pub fn weight_rule(&self) -> WeightRule {
314        self.weight_rule
315    }
316
317    /// The shape of every patch array: `[samples, channels, correlations]`.
318    #[must_use]
319    pub fn shape(&self) -> [usize; 3] {
320        [self.samples(), self.channels, self.correlations.len()]
321    }
322}

324/// What an updater wrote, for the evidence.
325#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
326pub struct UpdateReport {
327    /// Batches written.
328    pub batches: usize,
329    /// Samples given a native address and written.
330    pub samples: usize,
331    /// Samples with no native address (MSv2 padding): nothing was written
332    /// for them.
333    pub unaddressed: usize,
334    /// Native units written: MSv2 cells, MSv4 chunk objects.
335    pub units_written: usize,
336    /// Native units read back to merge a change into what they hold: an
337    /// MSv2 cell a patch covers only partly, an MSv4 chunk a flush covers
338    /// only partly. A unit a patch supplies whole is never read.
339    pub units_read: usize,
340    /// The most bytes the updater holds between batches: MSv4, one
341    /// chunk-row of times of each target variable over the channel chunks
342    /// the traversal reaches (every channel for weights); MSv2, none.
343    pub held_bytes: usize,
344}

346/// Writes one traversal's batches back to its store.
347///
348/// An updater is bound to a traversal and to some of its session's
349/// targets (`casa_ms::observation_io::UpdateSession::updater`); each call
350/// gives the values of those targets, in that order, for one delivered
351/// batch. It takes the traversal's batches in any order, and any of them:
352/// the times it is not given keep what they hold, as the elements and
353/// samples a visibility patch leaves do.
354///
355/// Only [`Self::finish`] completes an update: an updater dropped without
356/// it leaves what it wrote and discards what it held (an MSv4 updater's
357/// window of up to one chunk-row of times), and reports neither.
358pub trait Updater: Send {
359    /// Write `values`, one per bound target, for `batch`'s samples.
360    ///
361    /// # Errors
362    ///
363    /// When a value disagrees with its target or the batch, the batch is
364    /// not of the traversal (another observation's, other channels, times
365    /// outside its selection), or a change is refused (a weight shared by
366    /// channels the patch does not address, a flag cleared or a finite
367    /// weight written where the store holds padding). Nothing of `batch` is
368    /// then written. A storage failure leaves what the store's guarantees
369    /// state.
370    fn write(
371        &mut self,
372        batch: &VisibilityBatch<'_>,
373        values: &[PatchValues<'_>],
374    ) -> Result<(), ObservationError>;
375
376    /// Write `values` for the samples of the delivered batch `address`
377    /// names, as [`Self::write`] writes them for the batch, once its
378    /// buffers are gone. A change whose check reads the batch's payload
379    /// ([`PatchValues::needs_payload`]: a flag cleared, a weight written)
380    /// is refused, on every store: an MSv4 store knows its padding only from
381    /// the batch's weights.
382    ///
383    /// # Errors
384    ///
385    /// As [`Self::write`], and when a value needs the batch's payload.
386    fn write_at(
387        &mut self,
388        address: &BatchAddress,
389        values: &[PatchValues<'_>],
390    ) -> Result<(), ObservationError>;
391
392    /// Finish: write what is still held and report.
393    ///
394    /// # Errors
395    ///
396    /// When storage fails.
397    fn finish(self: Box<Self>) -> Result<UpdateReport, ObservationError>;
398}

400/// Refuse, for [`Updater::write_at`], values whose check needs the batch's
401/// payload ([`PatchValues::needs_payload`]).
402///
403/// # Errors
404///
405/// [`ObservationError::InvalidRequest`] naming the first such value.
406pub fn refuse_payload_checks(values: &[PatchValues<'_>]) -> Result<(), ObservationError> {
407    match values.iter().position(PatchValues::needs_payload) {
408        Some(index) => Err(ObservationError::InvalidRequest {
409            reason: format!(
410                "patch value {index} clears flags or writes weights, which must not turn \
411                 padding into an acquisition: write it against the batch, whose weights mark \
412                 padding, not its address"
413            ),
414        }),
415        None => Ok(()),
416    }
417}

419/// Check `values` against the bound `roles` and the batch at `address`
420/// before anything is written: one value per role, in order, each of the
421/// batch's shape (a visibility patch's mask too); no sample both set and
422/// cleared; and a weight patch for every channel giving each sample and
423/// product one value.
424///
425/// # Errors
426///
427/// [`ObservationError::InvalidRequest`] naming the first disagreement.
428pub fn check_values(
429    address: &BatchAddress,
430    roles: &[WritableRole],
431    values: &[PatchValues<'_>],
432) -> Result<(), ObservationError> {
433    let refuse = |reason: String| Err(ObservationError::InvalidRequest { reason });
434    if values.len() != roles.len() {
435        return refuse(format!(
436            "{} patch values for {} bound targets",
437            values.len(),
438            roles.len()
439        ));
440    }
441    let shape = address.shape();
442    for (index, (value, role)) in values.iter().zip(roles).enumerate() {
443        if value.role() != *role {
444            return refuse(format!(
445                "patch value {index} writes {:?}; its target is {role:?}",
446                value.role()
447            ));
448        }
449        let shapes: Vec<&[usize]> = match value {
450            PatchValues::Visibilities { values, written } => std::iter::once(values.shape())
451                .chain(written.iter().map(|mask| mask.shape()))
452                .collect(),
453            PatchValues::Flags { set, clear } => {
454                set.iter().chain(clear).map(|mask| mask.shape()).collect()
455            }
456            PatchValues::Weights { values, .. } => vec![values.shape()],
457        };
458        if let Some(wrong) = shapes.iter().find(|actual| **actual != shape.as_slice()) {
459            return refuse(format!(
460                "patch value {index} has shape {wrong:?}; the batch's is {shape:?}"
461            ));
462        }
463        match value {
464            PatchValues::Flags {
465                set: Some(set),
466                clear: Some(clear),
467            } if set.iter().zip(clear).any(|(set, clear)| *set && *clear) => {
468                return refuse(format!("patch value {index} both sets and clears a flag"));
469            }
470            // Weights patched from weights made from the sigmas would write
471            // sigmas back as weights, unannounced.
472            PatchValues::Weights { .. } if address.weight_rule != WeightRule::Stored => {
473                return refuse(format!(
474                    "patch value {index} writes weights for a batch whose weights were made \
475                     from the sigmas ({:?}); patch weights from a traversal of the stored \
476                     weights",
477                    address.weight_rule
478                ));
479            }
480            PatchValues::Weights {
481                values,
482                channels: WeightChannels::All,
483            } if !uniform_over_channels(values) => {
484                return refuse(format!(
485                    "patch value {index} changes every channel's weight but gives a sample's \
486                     channels different values"
487                ));
488            }
489            _ => {}
490        }
491    }
492    Ok(())
493}

495/// Whether every sample and product has one value across channels.
496#[must_use]
497pub fn uniform_over_channels(values: &ArrayView3<'_, f32>) -> bool {
498    values.axis_iter(Axis(0)).all(|sample| {
499        let first = sample.index_axis(Axis(0), 0);
500        sample.axis_iter(Axis(0)).all(|channel| {
501            channel
502                .iter()
503                .zip(&first)
504                .all(|(a, b)| a.to_bits() == b.to_bits())
505        })
506    })
