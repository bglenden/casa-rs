11/// The measurements of one PSF plane that every minor cycle reuses: they
12/// change only when the PSF does, so a run measures them once.
13#[derive(Clone, Copy, Debug, PartialEq)]
14pub struct PsfSummary {
15    peak: usize,
16    beam: RestoringBeam,
17    sidelobe: f64,
18    clark: ClarkPatch,
19    off_centre: bool,
20    unsettled_beam: bool,
21}
22
23/// Clark's PSF patch (`SDAlgorithmClarkClean2`): the square the active
24/// pixels are updated within, and the largest PSF value outside it.
25#[derive(Clone, Copy, Debug, PartialEq)]
26pub struct ClarkPatch {
27    /// Half the patch width per axis; an even patch ends one pixel early on
28    /// its positive side, as in casacore.
29    pub radius: [usize; 2],
30    /// The patch width per axis.
31    pub size: [usize; 2],
32    /// The largest PSF magnitude outside the patch, by casacore's
33    /// asymmetric box about the plane centre; 0 when the patch is the plane.
34    pub exterior_sidelobe: f64,
35}
36
37impl PsfSummary {
38    /// Measure a PSF plane normalised to peak 1.
39    ///
40    /// The peak is found as `MatrixCleaner::findPSFMaxAbs` finds it; the
41    /// beam is CASA's main-lobe fit in pixel units with `psfcutoff` 0.35;
42    /// the sidelobe is CASA's fitted sidelobe level
43    /// (`SIImageStore::getPSFSidelobeLevel`).
44    ///
45    /// # Errors
46    ///
47    /// When the PSF has no positive finite peak or the main lobe cannot be
48    /// fitted.
49    pub fn new(psf: &[f64], shape: PlaneShape) -> Result<Self, Error> {
50        Self::with_cutoff(psf, shape, DEFAULT_PSF_FIT_CUTOFF)
51    }
52
53    /// Measure a PSF using the requested CASA main-lobe fit cutoff.
54    ///
55    /// # Errors
56    ///
57    /// When the PSF has no positive finite peak or the main lobe cannot be fitted.
58    pub fn with_cutoff(psf: &[f64], shape: PlaneShape, cutoff: f32) -> Result<Self, Error> {
59        Self::measure(psf, shape, cutoff, None, [1.0, 1.0])
60    }
61
62    /// Measure a PSF with a fitted beam recovered from its image metadata.
63    /// The beam widths use pixels, as in [`Self::beam`].
64    ///
65    /// # Errors
66    ///
67    /// When the PSF has no positive finite peak.
68    pub fn with_saved_beam(
69        psf: &[f64],
70        shape: PlaneShape,
71        beam: RestoringBeam,
72    ) -> Result<Self, Error> {
73        Self::measure(psf, shape, DEFAULT_PSF_FIT_CUTOFF, Some(beam), [1.0, 1.0])
74    }
75
76    /// Measure with direction increments expressed in first-axis pixels.
77    ///
78    /// # Errors
79    /// The PSF has no positive finite peak or its main lobe cannot be fitted.
80    pub fn with_cutoff_and_cell(
81        psf: &[f64],
82        shape: PlaneShape,
83        cutoff: f32,
84        cell: [f64; 2],
85    ) -> Result<Self, Error> {
86        Self::measure(psf, shape, cutoff, None, cell)
87    }
88
89    /// Measure a saved beam with direction increments in first-axis pixels.
90    ///
91    /// # Errors
92    /// The PSF has no positive finite peak.
93    pub fn with_saved_beam_and_cell(
94        psf: &[f64],
95        shape: PlaneShape,
96        beam: RestoringBeam,
97        cell: [f64; 2],
98    ) -> Result<Self, Error> {
99        Self::measure(psf, shape, DEFAULT_PSF_FIT_CUTOFF, Some(beam), cell)
100    }

102    /// A blank plane's saved beam for automatic-mask common-beam selection.
103    /// No solver may use its peak or patch: the plane has no PSF samples.
104    #[must_use]
105    pub fn blank_with_saved_beam(beam: RestoringBeam) -> Self {
106        Self {
107            peak: 0,
108            beam,
109            sidelobe: 0.0,
110            clark: ClarkPatch {
111                radius: [0, 0],
112                size: [0, 0],
113                exterior_sidelobe: 0.0,
114            },
115            off_centre: false,
116            unsettled_beam: false,
117        }
118    }
119
120    /// Copy this summary with the beam used by automatic masking. The PSF's
121    /// sidelobe and Clark patch remain those measured from the PSF beam.
122    #[must_use]
123    pub const fn with_mask_beam(mut self, beam: RestoringBeam) -> Self {
124        self.beam = beam;
125        self
126    }
127
128    fn measure(
129        psf: &[f64],
130        shape: PlaneShape,
131        cutoff: f32,
132        saved_beam: Option<RestoringBeam>,
133        cell: [f64; 2],
134    ) -> Result<Self, Error> {
135        let peak = psf_peak(psf, shape).ok_or(Error::PsfPeak)?;
136        if !(psf[peak].is_finite() && psf[peak] > 0.0) {
137            return Err(Error::PsfPeak);
138        }
139        let single = psf.iter().map(|value| *value as f32).collect::<Vec<_>>();
140        let axes = [shape.nx, shape.ny];
141        // An off-centre plane without a saved beam takes its domain's largest
142        // later. With a saved beam, CASA subtracts that beam at the image
143        // centre even if it overlaps the off-centre PSF peak.
144        let off_centre = peak_outside_inner_quarter(&single, axes)?;
145        let (beam, sidelobe) = if off_centre {
146            match saved_beam {
147                Some(beam) => (
148                    beam,
149                    fitted_psf_sidelobe_fraction_with_cell(&single, axes, beam, cell)?,
150                ),
151                None => (
152                    RestoringBeam::new(1.0, 1.0, 0.0)?,
153                    psf.iter()
154                        .fold(0.0_f64, |level, value| level.max(value.abs())),
155                ),
156            }
157        } else {
158            let beam = match saved_beam {
159                Some(beam) => beam,
160                None => fit_main_lobe(&single, axes, cell, cutoff)?,
161            };
162            let sidelobe = fitted_psf_sidelobe_fraction_with_cell(&single, axes, beam, cell)?;
163            (beam, sidelobe)
164        };
165        Ok(Self {
166            peak,
167            beam,
168            sidelobe,
169            clark: ClarkPatch::new(psf, shape, beam),
170            off_centre,
171            unsettled_beam: off_centre && saved_beam.is_none(),
172        })
173    }
174
175    /// Whether the plane's PSF peaks outside its inner quarter, so it has no
176    /// beam of its own (`SIImageStore::makeImageBeamSet`).
177    #[must_use]
178    pub const fn off_centre(&self) -> bool {
179        self.off_centre
180    }
181
182    /// This summary of the normalised PSF `psf` with its domain's largest
183    /// beam, and Clark's patch for it, as CASA gives a plane with none of its
184    /// own (`SDAlgorithmClarkClean2` reads the patch from the beam set). The
185    /// sidelobe stays as measured.
186    #[must_use]
187    pub fn with_domain_beam(self, psf: &[f64], shape: PlaneShape, beam: RestoringBeam) -> Self {
188        Self {
189            beam,
190            clark: ClarkPatch::new(psf, shape, beam),
191            ..self
192        }
193    }
194
195    /// Whether the off-centre plane still needs its domain's beam.
196    #[must_use]
197    pub const fn needs_domain_beam(&self) -> bool {
198        self.unsettled_beam
199    }
200
201    /// Set the domain's beam and remeasure an off-centre PSF's sidelobe.
202    /// `cell` gives the direction increments in first-axis pixels.
203    ///
204    /// # Errors
205    ///
206    /// The PSF shape or peak is invalid.
207    pub fn settle_domain_beam(
208        self,
209        psf: &[f64],
210        shape: PlaneShape,
211        beam: RestoringBeam,
212        cell: [f64; 2],
213    ) -> Result<Self, Error> {
214        let single = psf.iter().map(|value| *value as f32).collect::<Vec<_>>();
215        let sidelobe =
216            fitted_psf_sidelobe_fraction_with_cell(&single, [shape.nx, shape.ny], beam, cell)?;
217        Ok(Self {
218            sidelobe,
219            unsettled_beam: false,
220            ..self.with_domain_beam(psf, shape, beam)
221        })
222    }
223
224    /// Storage index of the PSF peak.
225    #[must_use]
226    pub const fn peak(&self) -> usize {
227        self.peak
228    }
229
230    /// The fitted main lobe, in pixels.
231    #[must_use]
232    pub const fn beam(&self) -> RestoringBeam {
233        self.beam
234    }
235
236    /// The fitted sidelobe level the cycle threshold scales by.
237    #[must_use]
238    pub const fn sidelobe(&self) -> f64 {
239        self.sidelobe
240    }
241
242    /// Clark's patch.
243    #[must_use]
244    pub const fn clark(&self) -> ClarkPatch {
245        self.clark
246    }
247}

249impl ClarkPatch {
250    /// At least four pixels, otherwise the ceiling of the fitted widths;
251    /// a `3·ncent + 1` square capped by the plane.
252    fn new(psf: &[f64], shape: PlaneShape, beam: RestoringBeam) -> Self {
253        let central = 4_usize
254            .max(beam.major_fwhm_rad().ceil() as usize)
255            .max(beam.minor_fwhm_rad().ceil() as usize);
256        let requested = 3 * central + 1;
257        let size = [requested.min(shape.nx), requested.min(shape.ny)];
258        let radius = [size[0] / 2, size[1] / 2];
259        // ClarkCleanLatModel::absMaxBeyondDist about the PSF origin
260        // (psfShape/2): rows beyond ±d, and in the other rows the columns
261        // left of −d and from +d on. A patch as large as the plane leaves no
262        // exterior (getPsfPatch's user value, 0).
263        let [cx, cy] = shape.centre();
264        let exterior_sidelobe = if size == [shape.nx, shape.ny] {
265            0.0
266        } else {
267            psf.iter()
268                .enumerate()
269                .filter(|(index, _)| {
270                    let [x, y] = shape.pixel(*index);
271                    y + radius[1] < cy
272                        || y > cy + radius[1]
273                        || x + radius[0] < cx
274                        || x >= cx + radius[0]
275                })
276                .fold(0.0_f64, |maximum, (_, value)| maximum.max(value.abs()))
277        };
278        Self {
279            radius,
280            size,
281            exterior_sidelobe,
282        }
283    }

290    /// A smooth main lobe at `centre` of a 64-pixel plane, x-major.
291    fn lobe(centre: [usize; 2]) -> Vec<f64> {
292        let mut psf = vec![0.0; 64 * 64];
293        for x in 0..64 {
294            for y in 0..64 {
295                let dx = x as f64 - centre[0] as f64;
296                let dy = y as f64 - centre[1] as f64;
297                psf[x * 64 + y] = (-(dx * dx + dy * dy) / 8.0).exp();
298            }
299        }
300        psf
301    }

303    /// `getPSFSidelobeLevel` subtracts a Gaussian at the image's centre, so a
304    /// plane peaking far from it keeps that peak as its sidelobe, where one
305    /// peaking at the centre subtracts its main lobe.
306    #[test]
307    fn a_plane_peaking_off_centre_has_its_peak_as_its_sidelobe() {
308        let shape = PlaneShape::new(64, 64);
309        let centred = PsfSummary::new(&lobe([32, 32]), shape).expect("a summary");
310        assert!(centred.sidelobe() < 0.2, "{}", centred.sidelobe());
311        let off = PsfSummary::new(&lobe([8, 40]), shape).expect("a summary");
312        assert_eq!(off.sidelobe(), 1.0);
313        // No fit, so a spike too narrow for one is no failure.
314        let mut spike = vec![0.0; 64 * 64];
315        spike[32 * 64 + 32] = 0.99;
316        spike[8 * 64 + 40] = 1.0;
317        assert_eq!(
318            PsfSummary::new(&spike, shape)
319                .expect("a summary")
320                .sidelobe(),
321            1.0
322        );
323    }
324    /// The Gaussian uses both world-axis increments (CASA CAS-7171).
325    #[test]
326    fn a_saved_beam_sidelobe_uses_both_cell_sizes() {
327        let shape = PlaneShape::new(64, 64);
328        let mut psf = vec![0.0; 64 * 64];
329        for x in 0..64 {
330            for y in 0..64 {
331                let dx = x as f64 - 32.0;
332                let dy = y as f64 - 32.0;
333                psf[x * 64 + y] =
334                    (-4.0 * std::f64::consts::LN_2 * (dx * dx + dy * dy) / 16.0).exp();
335            }
336        }
337        let beam = RestoringBeam::new(4.0, 4.0, 0.0).unwrap();
338        let summary = PsfSummary::with_saved_beam_and_cell(&psf, shape, beam, [1.0, 2.0]).unwrap();
339        assert!(
340            (summary.sidelobe() - 0.4375).abs() < 1e-6,
341            "{}",
342            summary.sidelobe()
343        );
344    }
345    /// A PSF circular in world coordinates spans half as many pixels along
346    /// an axis whose increment is twice as large.
347    #[test]
348    fn a_fresh_psf_fit_uses_both_cell_sizes() {
349        let shape = PlaneShape::new(64, 64);
350        let mut psf = vec![0.0; shape.len()];
351        for x in 0..64 {
352            for y in 0..64 {
353                let dx = x as f64 - 32.0;
354                let dy = (y as f64 - 32.0) * 2.0;
355                psf[x * 64 + y] =
356                    (-4.0 * std::f64::consts::LN_2 * (dx * dx + dy * dy) / 64.0).exp();
357            }
358        }
359        let summary = PsfSummary::with_cutoff_and_cell(&psf, shape, 0.35, [1.0, 2.0]).unwrap();
360        let beam = summary.beam();
361        assert!(
362            (beam.major_fwhm_rad() / beam.minor_fwhm_rad() - 1.0).abs() < 0.05,
363            "{beam:?}"
364        );
365        assert!(summary.sidelobe() < 0.03, "{}", summary.sidelobe());
366    }
367    /// A domain beam's Gaussian is evaluated using both pixel increments.
368    #[test]
369    fn an_off_centre_domain_beam_uses_the_actual_pixel_aspect_ratio() {
370        let shape = PlaneShape::new(64, 64);
371        let mut psf = vec![0.0; shape.len()];
372        psf[32 * 64 + 15] = 1.0;
373        let beam = RestoringBeam::new(40.0, 40.0, 0.0).unwrap();
374        let summary = PsfSummary::new(&psf, shape)
375            .unwrap()
376            .settle_domain_beam(&psf, shape, beam, [1.0, 2.0])
377            .unwrap();
378        let expected = 1.0 - (-4.0 * std::f64::consts::LN_2 * (34.0_f64 / 40.0).powi(2)).exp();
379        assert!(
380            (summary.sidelobe() - expected).abs() < 1e-6,
381            "{}",
382            summary.sidelobe()
383        );
384        assert!(!summary.needs_domain_beam());
385    }
386    /// CASA subtracts the saved beam at the image centre even when the PSF
387    /// peaks outside the inner quarter (CASA 6.7.6.14: 0.39395368).

395        assert!(summary.off_centre());
396        assert!(
397            (summary.sidelobe() - 0.39395368).abs() < 1e-6,
398            "{}",
399            summary.sidelobe()
400        );
