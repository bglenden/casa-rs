16pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
17    let standalone = task.surface() == "getjyperkalma";
18    let caltable = task.required_file_text("caltable")?;
19    if caltable.is_empty() {
20        return Err(TaskFailure::validation("A caltable name must be specified").at("caltable"));
21    }
22    let endpoint = Endpoint::parse(&task.required_text("endpoint")?).ok_or_else(|| {
23        let message = if standalone {
24            "endpoint must be 'asdm', 'interpolation', or 'model-fit'"
25        } else {
26            "When the caltype is jyperk, endpoint must be one of asdm, interpolation or model-fit"
27        };
28        TaskFailure::validation(message).at("endpoint")
29    })?;
30    let measurement_set = task.path("vis")?;
31    if !measurement_set.exists() {
32        if standalone {
33            return Err(TaskFailure::execution(format!(
34                "Visibility data not found: {}",
35                measurement_set.display()
36            ))
37            .at("jyperk.vis"));
38        }
39        return Err(TaskFailure::validation(
40            "Visibility data set not found - please verify the name",
41        )
42        .at("vis"));
43    }
44    let request = GencalRequest {
45        measurement_set,
46        output_table: caltable.into(),
47        caltype: GencalType::Jyperk,
48        antenna: String::new(),
49        spw: task.text("spw")?.unwrap_or_default(),
50        parameter: Vec::new(),
51        gaincurve_table: None,
52    };
53    let infile = task
54        .text("infile")?
55        .filter(|name| !name.is_empty())
56        .map(PathBuf::from);
57    let report = super::read::report(task)?;
58    if let Some(path) = report.output.as_deref() {
59        super::refuse_existing_report(path, report.overwrite)?;
60        let products = vec![
61            request.measurement_set.clone(),
62            request.output_table.clone(),
63        ];
64        super::refuse_report_in_products(
65            path,
66            products.into_iter().chain(infile.clone()).collect(),
67            task,
68        )?;
69    }
70    let options = ServiceOptions {
71        endpoint,
72        timeout: task.integer("timeout")?,
73        attempts: task.integer("retry")?,
74        delay: task.integer("retry_wait_time")?,
75    };
76    let backup_hosts = standalone.then(|| task.texts("backup_hosts")).transpose()?;
77    let calibration = jyperk::calibrate(
78        &request,
79        infile.as_deref(),
80        &options,
81        backup_hosts.as_deref(),
82        CalibrationOutputMode::Update,
83    )
84    .map_err(|error| input_failure(error, standalone))?;
85    super::report_success(
86        &report,
87        &CalibrationTaskResult::Gencal(calibration),
88        CasaValue::None,
89    )
90}
91
92fn input_failure(error: JyperkError, standalone: bool) -> TaskFailure {
93    match error {
94        JyperkError::NoFactors if !standalone => {
95            TaskFailure::execution("There is no factor.").at("jyperk.no_factors")
96        }
97        JyperkError::MissingFile(_) | JyperkError::ReadFile { .. } => {
98            TaskFailure::execution(error.to_string()).at("jyperk.os")
99        }
100        JyperkError::InvalidNumber(_) | JyperkError::InvalidFactor { .. } => {
101            TaskFailure::validation(error.to_string()).at("infile")
102        }
103        _ => TaskFailure::execution(error.to_string()),
104    }
105}
