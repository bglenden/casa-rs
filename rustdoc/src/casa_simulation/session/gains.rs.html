2//! `sm.setgain` and `sm.setleakage`: antenna gains and polarization
3//! leakage, which `corrupt` applies (#841 Phase 3).
4//!
5//! As in CASA's `Simulator`, each call simulates its term over the rows
6//! observed so far, with the seed in force (`sm.setseed`), and adds it to
7//! the terms every later `corrupt` applies, until `reset`. The simulation
8//! and its differences from CASA are the `corrupt::jones` module's.
9//!
10//! A term grows with the integrations observed, so each call counts them
11//! first, admits what the term keeps and what forming it holds against the
12//! session's resource policy, and only then forms it; the term keeps its
13//! share for as long as it, a corruption recorded with it, or a table still
14//! to be written from it lives.
15

34/// The phase a gain or leakage term is admitted as, at `setgain` or
35/// `setleakage`: what it keeps for the session, and what forming it holds.
36const JONES_TERM_PHASE: &str = "a simulated gain or leakage term";
37
38/// The phase writing the calibration tables of the terms is admitted as, at
39/// `close`: the largest table built in memory and saved.
40const JONES_TABLE_PHASE: &str = "writing a simulated calibration table";
41
42/// The arguments of `sm.setgain`.
43#[derive(Debug, Clone, PartialEq)]
44pub struct GainSpec {
45    /// `mode`: `fbm`, gains drifting as fractional Brownian motion, or
46    /// `random`, gains drawn afresh each integration about one (CASA's
47    /// scatter about zero, `docs/CASA (C++) bugs.md`); `table` CASA
48    /// refuses.
49    pub mode: String,
50    /// `table`: the calibration table the simulated gains are written to,
51    /// or `''` for none. CASA writes it at once; casa-rs at `close`.
52    pub table: String,
53    /// `amplitude`: one value `a` for the complex scale `(a, a)`, or two,
54    /// `(a₀, a₁)`. The fluctuations' scale is its modulus.
55    pub amplitude: Vec<f64>,
56}
57
58/// The arguments of `sm.setleakage`.
59#[derive(Debug, Clone, PartialEq)]
60pub struct LeakageSpec {
61    /// `table`: the calibration table the simulated leakages are written
62    /// to, or `''` for none. CASA writes it at once; casa-rs at `close`.
63    /// (`mode` CASA does not read.)
64    pub table: String,
65    /// `amplitude`: one value `a` for the complex scale `(a, a)` of the
66    /// leakages' real and imaginary parts, or two.
67    pub amplitude: Vec<f64>,
68    /// `offset`: one value `o` for the offset `(o, o)` added to them, or
69    /// two.
70    pub offset: Vec<f64>,
71}
72
73/// CASA's complex pair from a one- or two-element vector (`setgain`,
74/// `setleakage`).
75fn pair(values: &[f64], what: &str) -> SimulatorResult<Complex64> {
76    match values {
77        [value] => Ok(Complex64::new(*value, *value)),
78        [real, imaginary, ..] => Ok(Complex64::new(*real, *imaginary)),
79        [] => Err(SimulatorError::refused(format!(
80            "Caught exception: {what} needs at least one value"
81        ))),
82    }
83}

85impl Simulator {
86    /// `sm.setgain(...)`: simulate antenna gains over the rows observed so
87    /// far and add them to the terms `corrupt` applies.
88    ///
89    /// # Errors
90    ///
91    /// - `mode='table'`: CASA logs "Cannot yet read from table" and returns
92    ///   `False`; another unknown mode is refused as CASA's "unsupported
93    ///   mode".
94    /// - No rows observed yet, or no `amplitude`: refused.
95    /// - A `table` path that cannot be made absolute, or that is, lies
96    ///   inside or contains the session's MeasurementSet: raised.
97    /// - The term's memory does not fit the session's resource policy: an
98    ///   admission error, before anything is formed.
99    pub fn set_gain(&mut self, spec: &GainSpec) -> SimulatorResult<()> {
100        let random = match spec.mode.as_str() {
101            "fbm" => false,
102            "random" => true,
103            "table" => {
104                return Err(SimulatorError::refused("Cannot yet read from table"));
105            }
106            other => {
107                return Err(SimulatorError::refused(format!(
108                    "Caught exception: unsupported mode {other} in setgain()"
109                )));
110            }
111        };
112        let table = self.table_path(&spec.table)?;
113        let camp = pair(&spec.amplitude, "setgain's amplitude")?;
114        tracing::info!("Gain corruption with complex RMS amplitude = {camp}");
115        self.require_rows()?;
116        tracing::info!("Creating G JONES Calibration structure for data corruption.");
117        let antennas = self.antennas.len();
118        let (per_window, span) = self.integration_counts();
119        let bytes = if random {
120            TermBytes::per_chunk(&per_window, antennas)
121        } else {
122            TermBytes::fbm(&per_window, screen_slots(span), antennas)
123        };
124        let reservation = self.admit_term(bytes)?;
125        let chunks = self.integration_chunks(&per_window);
126        drop(per_window);
127        let term = if random {
128            random_gains(self.seed, camp, antennas, chunks)
129        } else {
130            let mut scale = camp.norm();
131            if scale >= MAXIMUM_GAIN_SCALE {
132                tracing::warn!(" decreasing gain fluctuations from {scale} to 0.9 ");
133                scale = MAXIMUM_GAIN_SCALE;
134            }
135            fbm_gains(self.seed, scale, antennas, chunks)
136        };
137        self.hold_term(term, (reservation, bytes), table);
138        Ok(())
139    }

141    /// `sm.setleakage(...)`: simulate polarization leakages over the rows
142    /// observed so far and add them to the terms `corrupt` applies.
143    ///
144    /// # Errors
145    ///
146    /// No rows observed yet, or no `amplitude` or `offset`: refused. A
147    /// `table` path that cannot be made absolute, or that is, lies inside
148    /// or contains the session's MeasurementSet: raised. The term's memory
149    /// does not fit the session's resource policy: an admission error,
150    /// before anything is formed.
151    pub fn set_leakage(&mut self, spec: &LeakageSpec) -> SimulatorResult<()> {
152        let table = self.table_path(&spec.table)?;
153        let camp = pair(&spec.amplitude, "setleakage's amplitude")?;
154        let offset = pair(&spec.offset, "setleakage's offset")?;
155        self.require_rows()?;
156        tracing::info!("Creating D Calibration structure for data corruption.");
157        let antennas = self.antennas.len();
158        let first = self.first_integrations();
159        let per_window: Vec<usize> = first
160            .iter()
161            .map(|first| usize::from(first.is_some()))
162            .collect();
163        let bytes = TermBytes::per_chunk(&per_window, antennas);
164        let reservation = self.admit_term(bytes)?;
165        let chunks = window_chunks(&first, &per_window);
166        let term = leakages(self.seed, (camp, offset), antennas, chunks);
167        self.hold_term(term, (reservation, bytes), table);
168        Ok(())
169    }

171    /// Admit a term's memory before it is formed.
172    fn admit_term(&self, bytes: TermBytes) -> SimulatorResult<Reservation> {
173        let resources = &self.options.resources;
174        admit(
175            resources.host(),
176            resources.policy(),
177            &Demand {
178                phase: JONES_TERM_PHASE,
179                memory: bytes.kept + bytes.forming,
180            },
181        )
182        .map_err(SimulatorError::admission)
183    }
184
185    /// Add `term` to those `corrupt` applies, keeping of its reservation
186    /// what it keeps for as long as it lives, and to the calibration tables
187    /// `close` writes when it names one. CASA writes the table at once;
188    /// casa-rs writes it at `close`, beside the MeasurementSet whose
189    /// subtables it copies. `reset` does not remove it.
190    fn hold_term(
191        &mut self,
192        term: JonesTerm,
193        (mut reservation, bytes): (Reservation, TermBytes),
194        table: Option<PathBuf>,
195    ) {
196        reservation.retain(bytes.kept);
197        let term = Arc::new(HeldTerm::new(term, reservation));
198        if let Some(path) = table {
199            tracing::info!("Writing calTable = {} at close", path.display());
200            self.jones_tables.push((path, Arc::clone(&term)));
201        }
202        self.jones_terms.push(term);
203    }

205    /// CASA's `create_corrupt` cannot size a term on an empty
206    /// MeasurementSet.
207    fn require_rows(&self) -> SimulatorResult<()> {
208        if self.planned_rows() == 0 {
209            return Err(SimulatorError::refused(
210                "Caught exception: Error in Simulator::create_corrupt: MSIter::construct - \
211                 Input MeasurementSet.has zero selected rows",
212            ));
213        }
214        Ok(())
215    }

217    /// Write the calibration tables `setgain` and `setleakage` named, from
218    /// the MeasurementSet `close` wrote: one row per solution and antenna,
219    /// by spectral window, time and antenna, as CASA's
220    /// `NewCalTable::writeToDisk` sorts them.
221    ///
222    /// The terms hold their own admission; a table is built in memory and
223    /// saved, one at a time, so the largest table's build is admitted
224    /// before any is written, and each row is formed from its term as it
225    /// is added (#891 review).
226    pub(super) fn write_jones_tables(&self) -> SimulatorResult<()> {
227        let Some(rows) = self
228            .jones_tables
229            .iter()
230            .map(|(_, term)| term.table_rows())
231            .max()
232        else {
233            return Ok(());
234        };
235        let resources = &self.options.resources;
236        let _building = admit(
237            resources.host(),
238            resources.policy(),
239            &Demand {
240                phase: JONES_TABLE_PHASE,
241                memory: simulated_table_bytes(rows),
242            },
243        )
244        .map_err(SimulatorError::admission)?;
245        let ms = MeasurementSet::open(self.path()).map_err(|error| {
246            SimulatorError::raised(format!("reopen the MeasurementSet: {error}"))
247        })?;
248        for (path, term) in &self.jones_tables {
249            let kind = match term.kind {
250                JonesKind::Gain => SimulatedJones::Gain,
251                JonesKind::Leakage => SimulatedJones::Leakage,
252            };
253            let rows = term.solutions().flat_map(|(window, solution, values)| {
254                values
255                    .iter()
256                    .enumerate()
257                    .map(move |(antenna, &receptors)| SimulatedSolution {
258                        time_s: solution.time_s,
259                        field_id: solution.field as i32,
260                        spectral_window_id: window as i32,
261                        antenna: antenna as i32,
262                        receptors,
263                    })
264            });
265            write_simulated_jones_table(&ms, path, kind, rows).map_err(|error| {
266                SimulatorError::raised(format!(
267                    "write the calibration table {}: {error}",
268                    path.display()
269                ))
270            })?;
271        }
272        Ok(())
273    }

275    /// `table` as the absolute path `close` writes it to, or `None` for
276    /// none; refused when it is, lies inside or contains the session's
277    /// MeasurementSet, which writing it would replace.
278    fn table_path(&self, table: &str) -> SimulatorResult<Option<PathBuf>> {
279        if table.is_empty() {
280            return Ok(None);
281        }
282        let path = std::path::absolute(table)
283            .map_err(|error| SimulatorError::raised(format!("table {table}: {error}")))?;
284        check_simulated_table_destination(self.path(), &path)
285            .map_err(|error| SimulatorError::raised(error.to_string()))?;
286        Ok(Some(path))
287    }

289    /// Scan `scan`'s integrations, their TIMEs in order: as planned, or as
290    /// the opened MeasurementSet has them.
291    fn integration_times_s(&self, scan: usize) -> impl Iterator<Item = f64> + '_ {
292        let planned = self.scans[scan].planned();
293        let opened: &[f64] = if planned.is_some() {
294            &[]
295        } else {
296            self.opened
297                .as_ref()
298                .and_then(|opened| opened.integration_times_s.get(scan))
299                .map_or(&[], Vec::as_slice)
300        };
301        planned
302            .map(|planned| planned.integration_times_s())
303            .into_iter()
304            .flatten()
305            .chain(opened.iter().copied())
306    }

308    /// The integrations of the scans observed so far in each spectral
309    /// window, and the span of their times: what a gain term's chunks
310    /// number, counted before they are formed.
311    fn integration_counts(&self) -> (Vec<usize>, Option<[f64; 2]>) {
312        let mut per_window = Vec::new();
313        let mut span: Option<[f64; 2]> = None;
314        for (index, scan) in self.scans.iter().enumerate() {
315            let window = scan.spectral_window_id;
316            if per_window.len() <= window {
317                per_window.resize(window + 1, 0);
318            }
319            for time_s in self.integration_times_s(index) {
320                per_window[window] += 1;
321                span = Some(span.map_or([time_s, time_s], |[low, high]| {
322                    [low.min(time_s), high.max(time_s)]
323                }));
324            }
325        }
326        (per_window, span)
327    }

329    /// The solution chunks of a gain term: one per integration of each scan
330    /// observed so far, at its TIME, in its spectral window and field,
331    /// counted `per_window` ([`Self::integration_counts`]).
332    fn integration_chunks(&self, per_window: &[usize]) -> Chunks {
333        let mut chunks = Chunks::with_capacity(per_window);
334        for (index, scan) in self.scans.iter().enumerate() {
335            for time_s in self.integration_times_s(index) {
336                chunks.push(scan.spectral_window_id, time_s, scan.field_id);
337            }
338        }
339        chunks.sorted()
340    }

342    /// Each spectral window's earliest TIME among the scans observed so far
343    /// and the field of that integration, if any. CASA's infinite leakage
344    /// solution interval spans every scan and field of a window: its table
345    /// has one solution per window and antenna.
346    fn first_integrations(&self) -> Vec<Option<(f64, usize)>> {
347        let mut first: Vec<Option<(f64, usize)>> = Vec::new();
348        for (index, scan) in self.scans.iter().enumerate() {
349            let Some(time_s) = self.integration_times_s(index).next() else {
350                continue;
351            };
352            let window = scan.spectral_window_id;
353            if first.len() <= window {
354                first.resize(window + 1, None);
355            }
356            if first[window].is_none_or(|(earlier, _)| time_s < earlier) {
357                first[window] = Some((time_s, scan.field_id));
358            }
359        }
360        first
361    }
362}

364/// The solution chunks of a leakage term: one per spectral window observed
365/// so far, at its earliest TIME and on that integration's field (`first`),
366/// counted `per_window`.
367fn window_chunks(first: &[Option<(f64, usize)>], per_window: &[usize]) -> Chunks {
368    let mut chunks = Chunks::with_capacity(per_window);
369    for (window, first) in first.iter().enumerate() {
370        if let Some((time_s, field)) = *first {
371            chunks.push(window, time_s, field);
372        }
373    }
374    chunks
