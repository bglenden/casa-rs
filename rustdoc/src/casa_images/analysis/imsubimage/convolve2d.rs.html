18/// Requested Gaussian convolution geometry and scale.
19pub struct GaussianConvolutionOptions {
20    /// Pixel axes in CASA argument order. An empty list uses axes zero and one.
21    pub axes: Vec<i64>,
22    /// Full widths and position angle in the supplied quantity units.
23    pub widths: [Quantity; 3],
24    /// Positive manual multiplier, or nonpositive for unit-dependent scaling.
25    pub scale: f64,
26    /// Interpret the widths as the final resolution.
27    pub target_resolution: bool,
28}

49    /// Resolve the selected image, physical Gaussian, and convolution scaling.
50    ///
51    /// # Errors
52    /// The axis pair, region, kernel dimensions, or requested resolution is
53    /// incompatible with the image.
54    pub fn new<T: ConvolutionPixel>(
55        image: &dyn ImageInterface<T>,
56        native: RecordValue,
57        beams: &RecordValue,
58        region: Option<SubimageRegion<'_>>,
59        request: GaussianConvolutionOptions,
60    ) -> Result<Self, ImageError> {
61        let mut axes = parse_axes(&request.axes, image.ndim())?;
62        require_square(image.coordinates(), axes)?;
63        let mut section = SubimagePlan::new_with_beams(
64            image,
65            Some(native),
66            SubimageOptions {
67                region,
68                dropdeg: false,
69                keepaxes: &[],
70            },
71            Some(beams),
72        )?;
73        let coordinates = &section.coordinates;
74        let has_sky = direction_pair(coordinates).is_some_and(|sky| {
75            if sky.iter().all(|axis| axes.contains(axis)) {
76                axes = sky;
77                true
78            } else {
79                false
80            }
81        });
82        if !has_sky {
83            axes.sort_unstable();
84        }
85        if request.target_resolution && !has_sky {
86            let x = increment(coordinates, axes[0])?;
87            if !x.unit().conformant(&Unit::new("rad").expect("radians")) {
88                return Err(invalid(format!(
89                    "Major axis must have angular units ({} is not).",
90                    x.unit().name()
91                )));
92            }
93        }
94        let geometry = Sampling::new(coordinates, axes, has_sky, &request.widths)?;
95        let beam_set = &section.image_info.beam_set;
96        let channels = beam_set.n_channels().max(1);
97        let polarizations = beam_set.n_stokes().max(1);
98        let multiple = beam_set.is_multi();
99        let beam_axes = [CoordinateType::Spectral, CoordinateType::Stokes].map(|kind| {
100            multiple
101                .then(|| image.find_axis(kind))
102                .flatten()
103                .filter(|axis| !axes.contains(axis))
104        });
105        let per_pixel = has_sky && image.units().to_ascii_uppercase().contains("/PIXEL");
106        let per_beam = image.units().to_ascii_uppercase().contains("BEAM");
107        let mut planes = Vec::with_capacity(channels * polarizations);
108        for pol in 0..polarizations {
109            for channel in 0..channels {
110                planes.push(geometry.plane(
111                    if beam_set.is_empty() {
112                        GaussianBeam::default()
113                    } else {
114                        *beam_set.beam(channel, pol)
115                    },
116                    request.target_resolution,
117                    multiple,
118                    request.scale,
119                    per_pixel,
120                    per_beam,
121                )?);
122            }
123        }
124        let unit = if per_pixel && planes.last().is_some_and(|plane| plane.samples.is_some()) {
125            "Jy/beam"
126        } else {
127            image.units()
128        }
129        .to_owned();
130        let keep_requested = has_sky && (per_pixel || request.target_resolution);
131        section.image_info.beam_set = if !has_sky {
132            ImageBeamSet::default()
133        } else if request.target_resolution {
134            let beam = match planes.first() {
135                Some(plane) if !plane.beam.is_null() => geometry.request,
136                _ => GaussianBeam::default(),
137            };
138            ImageBeamSet::new(beam)
139        } else if multiple {
140            ImageBeamSet::from_grid(
141                (0..channels)
142                    .map(|channel| {
143                        (0..polarizations)
144                            .map(|pol| planes[pol * channels + channel].beam)
145                            .collect()
146                    })
147                    .collect(),
148            )
149        } else {
150            ImageBeamSet::new(planes[0].beam)
151        };
152        let mut output_info = section.image_info.to_record();
153        if keep_requested && output_info.get("restoringbeam").is_some() {
154            output_info.upsert("restoringbeam", Value::Record(geometry.beam_record()?));
155        } else if per_pixel && let Some(Value::Record(grid)) = output_info.get("perplanebeams") {
156            let mut grid = grid.clone();
157            let record = geometry.beam_record()?;
158            for index in 0..planes.len() {
159                grid.upsert(format!("*{index}"), Value::Record(record.clone()));
160            }
161            output_info.upsert("perplanebeams", Value::Record(grid));
162        }
163        Ok(Self {
164            inner: ImageConvolution { section },
165            axes,
166            beam_axes,
167            channels,
168            planes,
169            unit,
170            output_info,
171        })
172    }

184    /// Convolve the selected planes into a writable image of the planned shape.
185    ///
186    /// # Errors
187    /// Input or output storage fails, or a requested mask is incompatible.
188    pub fn copy_to<T: ConvolutionPixel>(
189        &self,
190        image: &dyn ImageInterface<T>,
191        options: ConvolutionOptions<'_>,
192        output: &mut dyn MutableImageInterface<T>,
193    ) -> Result<(), ImageError> {
194        let shape = &self.section().shape;
195        let active = (0..shape.len())
196            .map(|axis| shape[axis] > 1 && self.axes.contains(&axis))
197            .collect::<Vec<_>>();
198        let plan = |plane: Plane| {
199            plane
200                .samples
201                .map(|samples| KernelPlan::gaussian(samples, self.axes, shape, plane.scale))
202                .transpose()
203        };
204        if self.beam_axes.iter().all(Option::is_none)
205            && let Some(workspace) = plan(self.planes[0])?
206        {
207            self.inner.copy_groups(
208                image,
209                options,
210                ConvolutionGroups {
211                    active: &active,
212                    plane_axes: &[],
213                    workspace: KernelWorkspaces::Shared(&workspace),
214                },
215                output,
216            )?;
217        } else {
218            let beam_axes = self.beam_axes.into_iter().flatten().collect::<Vec<_>>();
219            let plane_kernel = |position: &[usize]| {
220                let [channel, pol] = self
221                    .beam_axes
222                    .map(|axis| axis.map_or(0, |axis| position[axis]));
223                let channel = if self.channels == 1 { 0 } else { channel };
224                let pol = if self.planes.len() == self.channels {
225                    0
226                } else {
227                    pol
228                };
229                plan(self.planes[pol * self.channels + channel])
230            };
231            self.inner.copy_groups(
232                image,
233                options,
234                ConvolutionGroups {
235                    active: &active,
236                    plane_axes: &beam_axes,
237                    workspace: KernelWorkspaces::PerPlane(&plane_kernel),
238                },
239                output,
240            )?;
241        }
242        output.set_units_string(self.unit.clone())
243    }
244}

250fn parse_axes(axes: &[i64], rank: usize) -> Result<[usize; 2], ImageError> {
251    let pair = match axes {
252        [] => [0, 1],
253        &[x, y] => [x, y],
254        _ => return Err(invalid("Number of axes to convolve must be exactly 2")),
255    };
256    if pair[0] == pair[1] {
257        return Err(invalid("Axes must be different"));
258    }
259    if pair
260        .iter()
261        .any(|&axis| usize::try_from(axis).map_or(true, |axis| axis >= rank))
262    {
263        return Err(invalid(
264            "Axis value must be less than number of axes in image",
265        ));
266    }
267    Ok(pair.map(|axis| axis as usize))
268}

270fn direction_pair(coordinates: &CoordinateSystem) -> Option<[usize; 2]> {
271    let index = coordinates.find_coordinate(CoordinateType::Direction)?;
272    if let [Some(x), Some(y)] = coordinates.coordinate_pixel_axes(index)?.as_slice() {
273        Some([*x, *y])
274    } else {
275        None
276    }
277}

279fn increment(coordinates: &CoordinateSystem, axis: usize) -> Result<Quantity, ImageError> {
280    let (owner, local) = coordinates
281        .pixel_axis_owner(axis)
282        .ok_or_else(|| invalid("Selected axis has no coordinate"))?;
283    let model = coordinates.coordinate(owner);
284    let (value, unit) = match model {
285        CoordinateModel::Direction(direction) => (
286            direction.reference_and_increment_in_world_units().1[local],
287            direction.world_units()[local].clone(),
288        ),
289        _ => (model.increment()[local], model.axis_units()[local].clone()),
290    };
291    Quantity::new(value.abs(), &unit).map_err(|error| invalid(error.to_string()))
292}
293
294fn require_square(coordinates: &CoordinateSystem, axes: [usize; 2]) -> Result<(), ImageError> {
295    let x = increment(coordinates, axes[0])?;
296    let y = increment(coordinates, axes[1])?;
297    if y.get_value_in(x.unit())
298        .is_ok_and(|y| (x.value() - y).abs() <= 1e-13 * x.value().max(y))
299    {
300        Ok(())
301    } else {
302        Err(invalid(
303            "Pixels must be square, please regrid your image so that they are",
304        ))
305    }
306}

308struct Sampling {
309    matrix: [[f64; 2]; 2],
310    sky: bool,
311    request: GaussianBeam,
312    quantities: [Quantity; 3],
313}
314
315impl Sampling {
316    fn new(
317        coordinates: &CoordinateSystem,
318        axes: [usize; 2],
319        sky: bool,
320        widths: &[Quantity; 3],
321    ) -> Result<Self, ImageError> {
322        let x = increment(coordinates, axes[0])?;
323        let y = increment(coordinates, axes[1])?;
324        let pixel_width = |q: &Quantity| matches!(q.unit().name(), "pix" | "pixel");
325        let both_pixels = pixel_width(&widths[0]) && pixel_width(&widths[1]);
326        if pixel_width(&widths[0]) != pixel_width(&widths[1]) {
327            return Err(invalid(format!(
328                "Quantum::operator< unequal units '{}', '{}'",
329                widths[0].unit().name(),
330                widths[1].unit().name()
331            )));
332        }
333        let radians = Unit::new("rad").expect("radians");
334        let angle = widths[2]
335            .get_value_in(&radians)
336            .map_err(|_| invalid("The units of the position angle must be angular"))?;
337        let (matrix, values, quantities) = if sky {
338            let direction = coordinates
339                .find_coordinate(CoordinateType::Direction)
340                .expect("sky axes");
341            let CoordinateModel::Direction(direction) = coordinates.coordinate(direction) else {
342                unreachable!()
343            };
344            let matrix = sky_per_pixel(direction)?;
345            let quantities = if both_pixels {
346                [
347                    Quantity::with_unit(widths[0].value() * x.value(), x.unit().clone()),
348                    Quantity::with_unit(widths[1].value() * y.value(), y.unit().clone()),
349                    widths[2].clone(),
350                ]
351            } else {
352                widths.clone()
353            };
354            let values = [
355                quantities[0]
356                    .get_value_in(&radians)
357                    .map_err(|_| invalid("The units of the major axis must be angular"))?,
358                quantities[1]
359                    .get_value_in(&radians)
360                    .map_err(|_| invalid("The units of the minor axis must be angular"))?,
361            ];
362            (matrix, values, quantities)
363        } else if both_pixels {
364            (
365                [[0., 1.], [1., 0.]],
366                [widths[0].value(), widths[1].value()],
367                widths.clone(),
368            )
369        } else {
370            let convert = |q: &Quantity| {
371                q.get_value_in(x.unit()).map_err(|_| invalid(format!("The units of the world length ({}) are not consistent with those of Coordinate System ({})", q.unit().name(), x.unit().name())))
372            };
373            (
374                [[0., convert(&y)?], [x.value(), 0.]],
375                [convert(&widths[0])?, convert(&widths[1])?],
376                widths.clone(),
377            )
378        };
379        if values[0] < values[1] {
380            return Err(invalid("Major axis is less than minor axis"));
381        }
382        if values[0] <= 0. {
383            return Err(invalid(
384                "Gaussian2DParam<T>::setMajorAxis(const T &width) - width must be positive",
385            ));
386        }
387        if values[1] <= 0. {
388            return Err(invalid("axialRatio must be between (0,1]"));
389        }
390        Ok(Self {
391            matrix,
392            sky,
393            request: GaussianBeam::new(values[0], values[1], angle),
394            quantities,
395        })
396    }

398    fn beam_record(&self) -> Result<RecordValue, ImageError> {
399        let mut beam = RecordValue::default();
400        for (i, label) in ["major", "minor", "positionangle"].into_iter().enumerate() {
401            let mut q = self.quantities[i].clone();
402            if i == 2 {
403                q = Quantity::new(
404                    unwrap_position_angle(self.request.position_angle).to_degrees(),
405                    "deg",
406                )
407                .map_err(|error| invalid(error.to_string()))?;
408            }
409            let mut value = RecordValue::default();
410            value.upsert("value", Value::Scalar(ScalarValue::Float64(q.value())));
411            value.upsert(
412                "unit",
413                Value::Scalar(ScalarValue::String(q.unit().name().to_owned())),
414            );
415            beam.upsert(label, Value::Record(value));
416        }
417        Ok(beam)
418    }

420    fn plane(
421        &self,
422        input: GaussianBeam,
423        target: bool,
424        multiple: bool,
425        scale: f64,
426        per_pixel: bool,
427        per_beam: bool,
428    ) -> Result<Plane, ImageError> {
429        let input_beam = self.sky && !input.is_null();
430        if target && input_beam && multiple && beams_near(self.request, input) {
431            return Ok(Plane {
432                samples: None,
433                scale: 1.,
434                beam: input,
435            });
436        }
437        let beam = if target && input_beam {
438            self.request.deconvolving_beam(input)
439                .ok().flatten().ok_or_else(|| invalid("Unable to reach target resolution. Input image beam is identical to or larger than the output beam size"))?
440        } else {
441            self.request
442        };
443        let samples = GaussianSamples::new(beam, self.matrix)?;
444        let automatic = scale <= 0.;
445        let (scale, beam) = if per_pixel {
446            (if automatic { 1. } else { scale }, self.request)
447        } else if input_beam {
448            let input_samples = GaussianSamples::new(input, self.matrix)?;
449            let (peak, fitted) = sampled_beam_convolution(input_samples, samples, self.matrix)?;
450            let scale = if automatic { peak.recip() } else { scale };
451            let scale = if per_beam {
452                scale
453            } else {
454                scale * input.area() / fitted.area()
455            };
456            let beam = if target { self.request } else { fitted };
457            (scale, beam)
458        } else {
459            let scale = if automatic {
460                samples.volume().recip()
461            } else {
462                scale
463            };
464            (scale, GaussianBeam::default())
465        };
466        Ok(Plane {
467            samples: Some(samples),
468            scale,
469            beam,
470        })
471    }
472}

474fn beams_near(a: GaussianBeam, b: GaussianBeam) -> bool {
475    let close_width = |a: f64, b: f64| (a - b).abs() <= 1e-5 * a.abs().max(b.abs());
476    let angular_tolerance = (0.01_f64 / 3600.).to_radians();
477    close_width(a.major, b.major)
478        && close_width(a.minor, b.minor)
479        && (unwrap_position_angle(a.position_angle) - unwrap_position_angle(b.position_angle)).abs()
480            <= angular_tolerance
481}
