2//! An MSv2 destination's `POINTING`: the samples each input's plan's reads
3//! gather through the boundary ([`Observation::pointing`],
4//! [`PointingPlan::reads`]), one row per sample, input after input, each
5//! input's in (time, antenna) order as casacore's
6//! `MSPointingColumns::pointingIndex` expects it and its antennas the
7//! output's. An MSv4 source states no
8//! interval, so its rows have `INTERVAL` 0 (which casacore reads as covering
9//! half the gap to the antenna's next row, and casa-rs's reader as covering
10//! nothing); it holds no target, so `TARGET` repeats the direction, as MSv2
11//! writers without a separate target write it.

27/// One gathered `POINTING` row, its times in MJD seconds; its coefficients,
28/// `[axis][term]`, are `terms` per axis from `start` in the gathered
29/// coefficients.
30#[derive(Clone, Copy, Debug)]
31struct Row {
32    /// The time's bit pattern, so that equal times compare equal.
33    time_bits: u64,
34    antenna: AntennaId,
35    /// The row's place among its antenna's samples at its time.
36    repeat: u32,
37    interval_s: f64,
38    origin_s: f64,
39    start: usize,
40    terms: usize,
41}
42
43/// About the bytes a `POINTING` row holds in the created MeasurementSet's
44/// table until it is saved (#950), as the counting laws measure it.
45pub(super) const TABLE_ROW_BYTES: usize = 2_560;
46
47/// The bytes the writer holds for `samples` gathered samples of
48/// `coefficients` coefficients in all: it reserves exactly that and
49/// removes repeats in place.
50pub(super) fn gathered_bytes(samples: usize, coefficients: usize) -> usize {
51    samples
52        .saturating_mul(size_of::<Row>())
53        .saturating_add(coefficients.saturating_mul(size_of::<f64>()))
54}

56/// One input's pointing: its opening, the reads its plan carries, and the
57/// output's `ANTENNA` row of each of its antennas.
58pub(super) struct Pointed<'a> {
59    pub(super) observation: &'a dyn Observation,
60    pub(super) plan: &'a PointingPlan,
61    pub(super) antennas: &'a [i32],
62}
63
64/// One input's gathered rows and their coefficients.
65struct Gathered {
66    rows: Vec<Row>,
67    coefficients: Vec<f64>,
68}

70/// Write `inputs`' pointing, as their plans carry it, into `ms`'s
71/// `POINTING`, input after input, each input's antennas as the output
72/// numbers them. Reads of one input that hold the same sample (an MSv4's
73/// datasets each hold their antennas' pointing) write it once: a sample is
74/// the k-th at its antenna and time, and different samples there are
75/// refused. Samples of different inputs are all written. Every input's
76/// directions must be in one frame.
77pub(super) fn write(
78    ms: &mut MeasurementSet,
79    inputs: &[Pointed<'_>],
80) -> Result<(), ObservationError> {
81    let mut gathered = Vec::with_capacity(inputs.len());
82    let mut frame = None;
83    for input in inputs {
84        let (own, own_frame) = gather_input(input)?;
85        if let Some(own_frame) = own_frame {
86            if frame.is_some_and(|known| known != own_frame) {
87                return Err(ObservationError::Unsupported {
88                    what: "writing MSv2 pointing of inputs in different frames",
89                });
90            }
91            frame = Some(own_frame);
92        }
93        gathered.push(own);
94    }
95    let Some(frame) = frame else {
96        return Ok(());
97    };
98    let terms = gathered
99        .iter()
100        .flat_map(|own| own.rows.iter().map(|row| row.terms))
101        .max()
102        .unwrap_or(1);
103    let table = subtable_mut(ms, SubtableId::Pointing).map_err(storage)?;
104    if frame != DirectionRef::J2000 {
105        for column in ["DIRECTION", "TARGET"] {
106            TableMeasDesc::new_fixed(column, MeasureType::Direction, frame.as_str())
107                .write(table)
108                .map_err(storage)?;
109        }
110    }
111    for own in &gathered {
112        for row in &own.rows {
113            // `[2, terms]`, each axis's terms padded to the most any row has.
114            let mut direction = Vec::with_capacity(2 * terms);
115            for axis in 0..2 {
116                let start = row.start + axis * row.terms;
117                direction.extend_from_slice(&own.coefficients[start..start + row.terms]);
118                direction.extend(std::iter::repeat_n(0.0, terms - row.terms));
119            }
120            let direction = f64_array(&direction, vec![2, terms]);
121            let values = row_from_defs(
122                schema::pointing::REQUIRED_COLUMNS,
123                &[
124                    ("ANTENNA_ID", i(row.antenna.0 as i32)),
125                    ("TIME", f(f64::from_bits(row.time_bits))),
126                    ("INTERVAL", f(row.interval_s)),
127                    ("NAME", s("")),
128                    ("NUM_POLY", i(terms as i32 - 1)),
129                    ("TIME_ORIGIN", f(row.origin_s)),
130                    ("DIRECTION", direction.clone()),
131                    ("TARGET", direction),
132                    ("TRACKING", b(true)),
133                ],
134            );
135            table.add_row(values).map_err(storage)?;
136        }
137    }
138    Ok(())
139}

141/// One input's rows, in (time, antenna) order with its repeats removed and
142/// its antennas the output's, and their frame (`None` for no pointing).
143fn gather_input(input: &Pointed<'_>) -> Result<(Gathered, Option<DirectionRef>), ObservationError> {
144    let (observation, plan) = (input.observation, input.plan);
145    let catalogue = observation.catalogue();
146    let (samples, held) = plan.gathered();
147    let mut rows = Vec::with_capacity(samples);
148    let mut coefficients = Vec::with_capacity(held);
149    let mut frame = None;
150    for read in plan.reads() {
151        let series = observation.pointing(read)?;
152        if series.antennas.is_empty() {
153            continue;
154        }
155        if frame.is_some_and(|known| known != series.frame) {
156            return Err(ObservationError::Unsupported {
157                what: "pointing in different frames in different partitions",
158            });
159        }
160        frame = Some(series.frame);
161        let entry = catalogue.partition(read.partition);
162        gather(&series, entry, &mut rows, &mut coefficients);
163    }
164    rows.sort_by(|a, b| {
165        f64::from_bits(a.time_bits)
166            .total_cmp(&f64::from_bits(b.time_bits))
167            .then(a.antenna.cmp(&b.antenna))
168            .then(a.repeat.cmp(&b.repeat))
169    });
170    // Remove repeats in place: a sample two reads gathered is written once.
171    let key = |row: &Row| (row.time_bits, row.antenna, row.repeat);
172    let mut kept = 0;
173    for index in 0..rows.len() {
174        if kept > 0 && key(&rows[kept - 1]) == key(&rows[index]) {
175            if !same_sample(&rows[kept - 1], &rows[index], &coefficients) {
176                return Err(ObservationError::Unsupported {
177                    what: "partitions with different pointing at one antenna and time",
178                });
179            }
180            continue;
181        }
182        rows[kept] = rows[index];
183        kept += 1;
184    }
185    rows.truncate(kept);
186    for row in &mut rows {
187        row.antenna = AntennaId(input.antennas[row.antenna.0 as usize] as u32);
188    }
189    Ok((Gathered { rows, coefficients }, frame))
190}

192/// Add `series`' samples (in `entry`'s time system) to `rows`, their
193/// coefficients to `coefficients`.
194fn gather(
195    series: &PointingSeries,
196    entry: &casa_observation::PartitionEntry,
197    rows: &mut Vec<Row>,
198    coefficients: &mut Vec<f64>,
199) {
200    // Times in the MSv2's own system: MJD seconds.
201    let mjd = |times: &[f64]| -> Vec<f64> {
202        let view = EpochView::new(entry.time_scale, entry.time_origin, ArrayView1::from(times));
203        (0..times.len())
204            .map(|time| view.seconds_since(time, MjdHighPrec::new(0.0, 0.0)))
205            .collect()
206    };
207    for (antenna, samples) in &series.antennas {
208        let times = mjd(samples.times_s());
209        let origins = mjd(samples.origins_s());
210        let terms = samples.terms();
211        let mut repeat = 0;
212        for (sample, time) in times.iter().enumerate() {
213            repeat = if sample > 0 && times[sample - 1].to_bits() == time.to_bits() {
214                repeat + 1
215            } else {
216                0
217            };
218            rows.push(Row {
219                time_bits: time.to_bits(),
220                antenna: *antenna,
221                repeat,
222                interval_s: samples
223                    .intervals_s()
224                    .map_or(0.0, |intervals| intervals[sample]),
225                origin_s: origins[sample],
226                start: coefficients.len(),
227                terms,
228            });
229            coefficients.extend_from_slice(
230                &samples.coefficients()[sample * 2 * terms..(sample + 1) * 2 * terms],
231            );
232        }
233    }
234}

236/// Whether two gathered rows are one sample: the same interval, origin and
237/// polynomial, terms beyond either's own being zero (a read pads each
238/// antenna's terms to the most any of its antennas has).
239fn same_sample(a: &Row, b: &Row, coefficients: &[f64]) -> bool {
240    let term = |row: &Row, axis: usize, term: usize| {
241        if term < row.terms {
242            coefficients[row.start + axis * row.terms + term]
243        } else {
244            0.0
245        }
246    };
247    let terms = a.terms.max(b.terms);
248    a.interval_s.to_bits() == b.interval_s.to_bits()
249        && a.origin_s.to_bits() == b.origin_s.to_bits()
250        && (0..2).all(|axis| {
251            (0..terms).all(|index| term(a, axis, index).to_bits() == term(b, axis, index).to_bits())
252        })
253}

255#[cfg(test)]
256mod tests {
257    use casa_observation::AntennaId;
258
259    use super::{Row, same_sample};
260
261    /// One sample read with its antenna's own terms and with terms padded
262    /// to another antenna's is the same sample; a different term is not.
263    #[test]
264    fn padding_does_not_make_a_different_sample() {
265        let row = |start, terms| Row {
266            time_bits: 1.0_f64.to_bits(),
267            antenna: AntennaId(1),
268            repeat: 0,
269            interval_s: 1.0,
270            origin_s: 0.5,
271            start,
272            terms,
273        };
274        // [ra, dec]; [ra, 0, dec, 0]; [ra, 1e-6, dec, 0].
275        let coefficients = [1.0, -0.5, 1.0, 0.0, -0.5, 0.0, 1.0, 1e-6, -0.5, 0.0];
276        assert!(same_sample(&row(0, 1), &row(2, 2), &coefficients));
277        assert!(!same_sample(&row(0, 1), &row(6, 2), &coefficients));
278    }
