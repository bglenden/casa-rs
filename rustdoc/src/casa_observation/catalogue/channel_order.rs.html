2//! Which order two stores keep one spectral window's channels in.
3//!
4//! A spectral window whose channels descend in frequency (negative channel
5//! widths, as lower-sideband data are often stored) keeps that order in an
6//! MSv2 and in casa-rs's own MSv4 conversion, which follow CASA's channel
7//! numbering. XRADIO's converter writes every window in ascending frequency,
8//! reversing the channel axis and making the widths positive. Both
9//! conventions are allowed (#1014): the same window then holds the same
10//! channels in opposite orders, and a [`Binding`](super::Binding) states
11//! which, so a consumer carrying per-channel results from one store to the
12//! other maps channel `c` to [`ChannelOrder::channel`].
13
14use std::ops::Range;
15
16use super::PartitionEntry;
17
18/// How a partition's channels lie along another store's partition of the
19/// same content ([`PartitionEntry::channel_order`],
20/// [`Binding::channel_orders`](super::Binding::channel_orders)).
21#[derive(Clone, Copy, Debug, PartialEq, Eq)]
22pub enum ChannelOrder {
23    /// Channel `c` is the other partition's channel `c`.
24    Same,
25    /// Channel `c` of `n` is the other partition's channel `n - 1 - c`.
26    Reversed,
27}
28
29impl ChannelOrder {
30    /// The other partition's channel holding channel `channel` of `count`.
31    ///
32    /// # Panics
33    ///
34    /// When `channel` is not below `count`.
35    #[must_use]
36    pub fn channel(self, channel: usize, count: usize) -> usize {
37        assert!(channel < count, "channel {channel} of {count}");
38        match self {
39            Self::Same => channel,
40            Self::Reversed => count - 1 - channel,
41        }
42    }
43
44    /// The other partition's channels holding `channels` of `count`: the
45    /// same channels, which run the other way along its axis when
46    /// reversed.
47    ///
48    /// # Panics
49    ///
50    /// When `channels` does not lie within `count`.
51    #[must_use]
52    pub fn channels(self, channels: Range<usize>, count: usize) -> Range<usize> {
53        assert!(
54            channels.start <= channels.end && channels.end <= count,
55            "channels {channels:?} of {count}"
56        );
57        match self {
58            Self::Same => channels,
59            Self::Reversed => count - channels.end..count - channels.start,
60        }
61    }
62}
63
64impl PartitionEntry {
65    /// Whether `other` holds this partition's spectral and product content
66    /// and in which channel order. Its frequency frame, correlation
67    /// products (in order) and intents must be this partition's. Its
68    /// channels are then in the same order when their frequencies and
69    /// signed widths are this partition's, bit for bit
70    /// ([`Self::same_content`]); reversed when its frequencies are this
71    /// partition's in the reverse order and its widths have the same
72    /// magnitudes in that order, bit for bit. XRADIO makes a descending
73    /// window's widths positive as it reverses it, so a reversed window's
74    /// widths may differ in sign; a window kept in its order must keep
75    /// their sign too, as every producer does. The same order is
76    /// preferred, so a partition of one channel, or of equal frequencies,
77    /// is never reversed. `None` when the content differs.
78    #[must_use]
79    pub fn channel_order(&self, other: &Self) -> Option<ChannelOrder> {
80        if self.frequency_frame != other.frequency_frame
81            || self.correlations != other.correlations
82            || self.intents != other.intents
83        {
84            return None;
85        }
86        // Whether `theirs` is `ours` (or `ours` reversed) by `bits`.
87        let matches = |ours: &[f64], theirs: &[f64], reversed: bool, bits: fn(f64) -> u64| {
88            ours.len() == theirs.len()
89                && if reversed {
90                    ours.iter()
91                        .rev()
92                        .zip(theirs)
93                        .all(|(a, b)| bits(*a) == bits(*b))
94                } else {
95                    ours.iter().zip(theirs).all(|(a, b)| bits(*a) == bits(*b))
96                }
97        };
98        let magnitude = |width: f64| width.abs().to_bits();
99        if matches(
100            &self.frequencies_hz,
101            &other.frequencies_hz,
102            false,
103            f64::to_bits,
104        ) && matches(
105            &self.channel_widths_hz,
106            &other.channel_widths_hz,
107            false,
108            f64::to_bits,
109        ) {
110            Some(ChannelOrder::Same)
111        } else if matches(
112            &self.frequencies_hz,
113            &other.frequencies_hz,
114            true,
115            f64::to_bits,
116        ) && matches(
117            &self.channel_widths_hz,
118            &other.channel_widths_hz,
119            true,
120            magnitude,
121        ) {
122            Some(ChannelOrder::Reversed)
123        } else {
124            None
125        }
126    }
127}

137    /// `partition`'s three channels from 676 GHz at −1 MHz, and the same
138    /// window as XRADIO writes it: ascending, its widths positive.
139    fn descending_and_ascending() -> (PartitionEntry, PartitionEntry) {
140        let mut descending = partition(0, &[0], &[0]);
141        descending.frequencies_hz = vec![676.0e9, 676.0e9 - 1.0e6, 676.0e9 - 2.0e6];
142        descending.channel_widths_hz = vec![-1.0e6; 3];
143        let mut ascending = descending.clone();
144        ascending.frequencies_hz.reverse();
145        ascending.channel_widths_hz = vec![1.0e6; 3];
146        (descending, ascending)
147    }
148
149    /// A window and its channels reversed bind either way, reversed; a
150    /// window and itself in the same order; and anything else of the
151    /// content not at all.
152    #[test]
153    fn a_window_binds_to_its_reversed_channels() {
154        let (descending, ascending) = descending_and_ascending();
155        assert_eq!(
156            descending.channel_order(&ascending),
157            Some(ChannelOrder::Reversed)
158        );
159        assert_eq!(
160            ascending.channel_order(&descending),
161            Some(ChannelOrder::Reversed)
162        );
163        assert_eq!(
164            descending.channel_order(&descending),
165            Some(ChannelOrder::Same)
166        );
167        let differs = |change: fn(&mut PartitionEntry)| {
168            let mut other = ascending.clone();
169            change(&mut other);
170            descending.channel_order(&other)
171        };
172        assert_eq!(differs(|p| p.channel_widths_hz[0] = 2.0e6), None);
173        assert_eq!(differs(|p| p.frequencies_hz[1] += 1.0), None);
174        assert_eq!(differs(|p| p.frequency_frame = FrequencyRef::TOPO), None);
175        assert_eq!(differs(|p| p.correlations.reverse()), None);
176        assert_eq!(differs(|p| p.intents.push("CAL".to_string())), None);
177        assert_eq!(differs(|p| p.frequencies_hz.truncate(2)), None);
178        // One channel is the same channel either way: never reversed.
179        let mut one = descending.clone();
180        one.frequencies_hz.truncate(1);
181        one.channel_widths_hz.truncate(1);
182        assert_eq!(one.channel_order(&one), Some(ChannelOrder::Same));
183    }

185    /// A catalogue binds to another holding its descending window
186    /// ascending, either way round, and the binding records each
187    /// partition's order. A catalogue holding the window both ways cannot
188    /// tell which is meant, and the binding is refused.
189    #[test]
190    fn a_binding_records_each_partitions_channel_order() {
191        let (descending, ascending) = descending_and_ascending();
192        let other_window = partition(0, &[1], &[1]);
193        let ours = observed(
194            vec![session("p", None)],
195            vec![descending.clone(), other_window.clone()],
196        );
197        let theirs = observed(
198            vec![session("p", None)],
199            vec![other_window.clone(), ascending.clone()],
200        );
201        // Each binding lists the orders of the catalogue bound from.
202        for (a, b, orders) in [
203            (&ours, &theirs, [ChannelOrder::Reversed, ChannelOrder::Same]),
204            (&theirs, &ours, [ChannelOrder::Same, ChannelOrder::Reversed]),
205        ] {
206            let binding = a.bind_to(b, &BindingTolerance::EXACT).expect("binds");
207            assert_eq!(binding.partitions, [PartitionId(1), PartitionId(0)]);
208            assert_eq!(binding.channel_orders, orders);
209        }
210        let both = observed(vec![session("p", None)], vec![descending, ascending]);
211        let refused = both
212            .bind_to(&both, &BindingTolerance::EXACT)
213            .expect_err("ambiguous");
214        assert!(refused.to_string().contains("more than one"), "{refused}");
215    }

217    /// Channel `c` of `n` maps to `n - 1 - c` reversed; a range maps to the
218    /// same channels, running the other way.
219    #[test]
220    fn channels_map_through_the_order() {
221        assert_eq!(ChannelOrder::Same.channel(2, 64), 2);
222        assert_eq!(ChannelOrder::Reversed.channel(0, 64), 63);
223        assert_eq!(ChannelOrder::Reversed.channel(63, 64), 0);
224        assert_eq!(ChannelOrder::Same.channels(2..6, 64), 2..6);
225        assert_eq!(ChannelOrder::Reversed.channels(2..6, 64), 58..62);
226        assert_eq!(ChannelOrder::Reversed.channels(0..64, 64), 0..64);
227        assert_eq!(ChannelOrder::Reversed.channels(5..5, 8), 3..3);
228        for order in [ChannelOrder::Same, ChannelOrder::Reversed] {
229            let mapped = order.channels(2..6, 64);
230            assert!((2..6).all(|channel| mapped.contains(&order.channel(channel, 64))));
231            assert_eq!(mapped.len(), 4);
232        }
233    }
