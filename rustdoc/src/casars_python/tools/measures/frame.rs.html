2//! The frame and conversions: `me.doframe`, `me.framenow`, `me.measure`,
3//! `me.separation`, `me.posangle` and `me.done`.
4//!
5//! A conversion runs in casa-types with a frame of the epoch, position and
6//! direction `doframe` set and the measures data. Divergence: casatools'
7//! tool starts with a frame at MJD 0 at the geocentre, so a conversion that
8//! needs a frame element converts with those values when `doframe` has not
9//! set it; here the conversion fails, with CASA's "Cannot convert due to
10//! missing frame information" (as CASA's does after `me.done()`).
11

25/// casacore's `C::degree`.
26const DEGREE: f64 = std::f64::consts::PI / 180.0;

28/// Reads a measure argument, logging casacore's messages (prefixed to CASA's
29/// `context`) when it is not one.
30fn read(record: &RecordArg, context: &str, lines: &mut Vec<LogLine>) -> Option<Measure> {
31    match Measure::from_record(&record.0) {
32        Ok(read) => {
33            if let Some(warning) = read.warning {
34                lines.push(LogLine::warning(warning));
35            }
36            Some(read.measure)
37        }
38        Err(error) => {
39            lines.push(LogLine::error(format!("{}{context}", error.message())));
40            None
41        }
42    }
43}

45/// `me.doframe(v)`: an epoch, position or direction joins the frame
46/// (`True`); other measures are not frame elements (`False`).
47pub(super) fn doframe(state: &mut State, v: &RecordArg, lines: &mut Vec<LogLine>) -> CasaValue {
48    if matches!(v.0.get("type"), Some(crate::convert::Variant::Text(kind)) if kind.eq_ignore_ascii_case("radialvelocity"))
49    {
50        return match super::record::Measures::read(v) {
51            Ok(values) => {
52                if let Some(warning) = &values.warning {
53                    lines.push(LogLine::warning(warning.clone()));
54                }
55                state.frame.radial_velocity = Some(values.into_reference_origin());
56                state.frame.rendered = None;
57                CasaValue::Bool(true)
58            }
59            Err(error) => {
60                lines.push(LogLine::error(format!(
61                    "{}Non-measure type v in measure conversion",
62                    error.message()
63                )));
64                CasaValue::Bool(false)
65            }
66        };
67    }
68    let Some(measure) = read(v, "Non-measure type v in measure conversion", lines) else {
69        return CasaValue::Bool(false);
70    };
71    let frame = &mut state.frame;
72    let slot = match measure.kind() {
73        MeasureKind::Epoch => &mut frame.epoch,
74        MeasureKind::Position => &mut frame.position,
75        MeasureKind::Direction => &mut frame.direction,
76        MeasureKind::RadialVelocity => unreachable!("handled with reference offsets above"),
77        MeasureKind::Frequency
78        | MeasureKind::Doppler
79        | MeasureKind::Baseline
80        | MeasureKind::Uvw
81        | MeasureKind::EarthMagnetic => return CasaValue::Bool(false),
82    };
83    *slot = Some(measure);
84    frame.rendered = None;
85    CasaValue::Bool(true)
86}

88/// `me.framenow()`: the frame's epoch is the system clock's UTC.
89pub(super) fn framenow(state: &mut State) -> CasaValue {
90    /// MJD of the Unix epoch, 1970-01-01.
91    const UNIX_EPOCH_MJD: f64 = 40_587.0;
92    let since_unix_s = std::time::SystemTime::now()
93        .duration_since(std::time::UNIX_EPOCH)
94        .map_or(0.0, |elapsed| elapsed.as_secs_f64());
95    let now = CasaMeasure::Epoch(MEpoch::from_mjd(
96        UNIX_EPOCH_MJD + since_unix_s / 86_400.0,
97        EpochRef::UTC,
98    ));
99    state.frame.epoch = Some(Measure::from_casa(&now));
100    state.frame.rendered = None;
101    CasaValue::Bool(true)
102}

104/// `me.done()`: the frame is emptied.
105pub(super) fn done(state: &mut State) -> CasaValue {
106    state.frame = super::Frame::default();
107    CasaValue::Bool(true)
108}

110/// `me.measure(v, rf, off)`: `v` converted to the code `rf` names (its
111/// kind's default if it names none).
112pub(super) fn measure(
113    state: &mut State,
114    v: &RecordArg,
115    rf: &str,
116    off: &RecordArg,
117    lines: &mut Vec<LogLine>,
118) -> PyResult<CasaValue> {
119    if !off.0.is_empty() {
120        lines.push(LogLine::error(
121            "measure offsets (off) are not yet implemented in casa-rs",
122        ));
123        return Ok(CasaValue::empty_record());
124    }
125    let measure = match super::record::Measures::read(v) {
126        Ok(measure) => measure,
127        Err(error) => {
128            lines.push(LogLine::error(format!(
129                "{}Non-measure type in measure conversion",
130                error.message()
131            )));
132            return Ok(CasaValue::empty_record());
133        }
134    };
135    if let Some(warning) = &measure.warning {
136        lines.push(LogLine::warning(warning.clone()));
137    }
138    let target = measure.values[0].kind().codes().resolve_or_default(rf);
139    match measure.resolve(state, target)? {
140        Ok(converted) => Ok(converted.to_value()),
141        Err(reason) => {
142            lines.push(LogLine::warning(format!(
143                "Cannot convert due to missing frame information\nCannot convert measure: {reason}"
144            )));
145            Ok(CasaValue::empty_record())
146        }
147    }
148}

150/// `measure` in the code `target` of its kind: itself when it is already
151/// there, else converted in casa-types with the frame. The outer error is the
152/// measures data failing to load; the inner one, the conversion failing.
153pub(super) fn convert(
154    state: &mut State,
155    measure: &Measure,
156    target: &'static str,
157) -> PyResult<Result<Measure, String>> {
158    if measure.refer == target {
159        return Ok(Ok(*measure));
160    }
161    if matches!(
162        measure.value,
163        MeasureValue::Baseline(_) | MeasureValue::Uvw(_) | MeasureValue::EarthMagnetic(_)
164    ) {
165        return super::geometry::convert(state, measure, target);
166    }
167    if let MeasureValue::Doppler(value) = measure.value {
168        let converted = casa_types::measures::doppler::MDoppler::new(
169            value,
170            crate::convert::holder::doppler_ref(measure.refer)
171                .map_err(pyo3::exceptions::PyRuntimeError::new_err)?,
172        )
173        .convert_to(
174            crate::convert::holder::doppler_ref(target)
175                .map_err(pyo3::exceptions::PyRuntimeError::new_err)?,
176            &MeasFrame::new(),
177        );
178        return Ok(converted
179            .map(|value| Measure {
180                refer: target,
181                value: MeasureValue::Doppler(value.value()),
182            })
183            .map_err(|error| error.to_string()));
184    }
185    let frame = match native_frame(state)? {
186        Ok(frame) => frame,
187        Err(error) => return Ok(Err(error)),
188    };
189    let converted = (|| {
190        let unsupported =
191            |kind: &str| format!("the {target} {kind} frame is not yet implemented in casa-rs");
192        let error = |error: casa_types::measures::MeasureError| error.to_string();
193        Ok::<_, String>(match measure.to_casa()? {
194            CasaMeasure::Epoch(epoch) => CasaMeasure::Epoch(
195                epoch
196                    .convert_to(
197                        EpochRef::from_str(target).map_err(|_| unsupported("epoch"))?,
198                        &frame,
199                    )
200                    .map_err(error)?,
201            ),
202            CasaMeasure::Direction(direction) => CasaMeasure::Direction(
203                convert_direction(
204                    state,
205                    &direction,
206                    DirectionRef::from_str(target).map_err(|_| unsupported("direction"))?,
207                    &frame,
208                )
209                .map_err(error)?,
210            ),
211            CasaMeasure::Position(position) => CasaMeasure::Position(
212                position
213                    .convert_to(PositionRef::from_str(target).map_err(|_| unsupported("position"))?)
214                    .map_err(error)?,
215            ),
216            CasaMeasure::Doppler(value) => CasaMeasure::Doppler(
217                value
218                    .convert_to(crate::convert::holder::doppler_ref(target)?, &frame)
219                    .map_err(error)?,
220            ),
221            CasaMeasure::RadialVelocity(value) => CasaMeasure::RadialVelocity(
222                value
223                    .convert_to(
224                        target.parse().map_err(|_| unsupported("radialvelocity"))?,
225                        &frame,
226                    )
227                    .map_err(error)?,
228            ),
229            CasaMeasure::Frequency(frequency) => CasaMeasure::Frequency(
230                frequency
231                    .convert_to(
232                        FrequencyRef::from_str(target).map_err(|_| unsupported("frequency"))?,
233                        &frame,
234                    )
235                    .map_err(error)?,
236            ),
237        })
238    })();
239    Ok(converted.map(|converted| Measure::from_casa(&converted)))
240}

242pub(super) fn convert_direction(
243    state: &mut State,
244    direction: &MDirection,
245    target: DirectionRef,
246    frame: &MeasFrame,
247) -> Result<MDirection, casa_types::measures::MeasureError> {
248    use casa_types::measures::MeasureError;
249
250    let refer = direction.refer();
251    if refer == DirectionRef::COMET && target != refer && state.frame.comet.is_some() {
252        return Err(MeasureError::NotYetImplemented {
253            route: format!("COMET → {target} (casa-rs #1252)"),
254        });
255    }
256    if refer == target {
257        return direction.convert_to(target, frame);
258    }
259    let (resolved, distance_m) = resolve_direction(state, direction, frame)?;
260    resolved.convert_with_parallax(target, frame, distance_m)
261}

263pub(super) fn resolve_direction(
264    state: &mut State,
265    direction: &MDirection,
266    frame: &MeasFrame,
267) -> Result<(MDirection, f64), casa_types::measures::MeasureError> {
268    use casa_measures_data::PlanetaryTable;
269    use casa_types::measures::MeasureError;
270
271    let refer = direction.refer();
272    if refer.casacore_code() < 32 || refer == DirectionRef::COMET {
273        return Ok((direction.clone(), 0.0));
274    }
275    if frame.epoch().is_none() {
276        return Err(MeasureError::MissingFrameData { what: "epoch" });
277    }
278    let model_error = |error: casa_measures_data::MeasuresDataError| MeasureError::ModelError {
279        model: "DE200",
280        reason: error.to_string(),
281    };
282    if state.planets.is_none() {
283        state.planets = Some(PlanetaryTable::open_discovered().map_err(model_error)?);
284    }
285    state
286        .planets
287        .as_ref()
288        .expect("the planetary table has been opened")
289        .geocentric_direction(direction, frame)
290        .map_err(model_error)
291}

293/// Build the explicit native frame once for a conversion or vector plan.
294pub(super) fn native_frame(state: &mut State) -> PyResult<Result<MeasFrame, String>> {
295    let measures = state.measures().map_err(no_measures)?;
296    Ok(materialize(
297        &state.frame,
298        MeasFrame::new().with_measures(measures),
299    ))
300}
301
302pub(super) fn materialize(
303    elements: &super::Frame,
304    mut frame: MeasFrame,
305) -> Result<MeasFrame, String> {
306    for element in [elements.epoch, elements.position, elements.direction]
307        .into_iter()
308        .flatten()
309    {
310        frame = match element.to_casa()? {
311            CasaMeasure::Epoch(epoch) => frame.with_epoch(epoch),
312            CasaMeasure::Position(position) => frame.with_position(position),
313            CasaMeasure::Direction(direction) => frame.with_direction(direction),
314            CasaMeasure::RadialVelocity(value) => frame.with_radial_velocity(value),
315            CasaMeasure::Frequency(_) | CasaMeasure::Doppler(_) => frame,
316        };
317    }
318    if let Some(velocity) = &elements.radial_velocity {
319        let value = velocity.radial_velocity(&frame)?;
320        frame = frame.with_radial_velocity(value);
321    }
322    Ok(frame)
323}

325/// What `me.separation` and `me.posangle` compute.
326#[derive(Debug, Clone, Copy, PartialEq, Eq)]
327pub(super) enum Between {
328    /// The angle between two directions.
329    Separation,
330    /// The position angle of the second from the first, north through east.
331    PositionAngle,
332}
333
334/// `me.separation(m1, m2)` and `me.posangle(m1, m2)`, in degrees, with `m2`
335/// converted to `m1`'s code when they differ. A failure is logged and gives
336/// casatools' `{'unit': 'deg', 'value': array([0.])}`.
337pub(super) fn between(
338    state: &mut State,
339    m1: &RecordArg,
340    m2: &RecordArg,
341    what: Between,
342    lines: &mut Vec<LogLine>,
343) -> PyResult<CasaValue> {
344    let zero = || {
345        CasaValue::record([
346            ("unit", CasaValue::Text("deg".to_string())),
347            ("value", CasaValue::floats(vec![0.0])),
348        ])
349    };
350    let mut directions = Vec::with_capacity(2);
351    for (name, record) in [("m1", m1), ("m2", m2)] {
352        let context = format!("Non-measure type {name} in measure conversion\n");
353        let Some(measure) = read(record, context.trim_end(), lines) else {
354            return Ok(zero());
355        };
356        if measure.kind() != MeasureKind::Direction {
357            lines.push(LogLine::error(
358                "Trying to convert non-direction to direction",
359            ));
360            return Ok(zero());
361        }
362        directions.push(measure);
363    }
364    let (first, second) = (directions[0], directions[1]);
365    let second = match convert(state, &second, first.refer)? {
366        Ok(second) => second,
367        Err(reason) => {
368            lines.push(LogLine::error(format!("Exception Reports: {reason}")));
369            return Ok(zero());
370        }
371    };
372    let (MeasureValue::Direction(a), MeasureValue::Direction(b)) = (first.value, second.value)
373    else {
374        unreachable!("both are directions");
375    };
376    let radians = match what {
377        Between::Separation => separation(a, b),
378        Between::PositionAngle => position_angle(a, b),
379    };
380    Ok(CasaValue::quantity(radians * (1.0 / DEGREE), "deg"))
381}

383/// `MVDirection::separation`.
384fn separation(a: [f64; 3], b: [f64; 3]) -> f64 {
385    let half = ((a[0] - b[0]).powi(2) + (a[1] - b[1]).powi(2) + (a[2] - b[2]).powi(2)).sqrt() / 2.0;
386    2.0 * half.min(1.0).asin()
387}
388
389/// `MVDirection::positionAngle`.
390fn position_angle(a: [f64; 3], b: [f64; 3]) -> f64 {
391    let longitude_difference = direction_angles(a).0 - direction_angles(b).0;
392    let (sin_lat1, sin_lat2) = (a[2], b[2]);
393    let cos_lat2 = (1.0 - sin_lat2 * sin_lat2).abs().sqrt();
394    let s1 = -cos_lat2 * longitude_difference.sin();
395    let c1 = (1.0 - sin_lat1 * sin_lat1).abs().sqrt() * sin_lat2
396        - sin_lat1 * cos_lat2 * longitude_difference.cos();
397    if s1 != 0.0 || c1 != 0.0 {
398        s1.atan2(c1)
399    } else {
400        0.0
401    }
