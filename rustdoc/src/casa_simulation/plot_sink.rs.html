3//! Domain-owned sink trait for simulation plot figure assembly.
4//!
5//! `casa-simulation` computes all physical data and passes it to an optional
6//! [`SimulationPlotSink`] implemented by presentation crates (`casa-plot-data`),
7//! keeping plot-document schemas out of the simulation library.
8
9use std::path::{Path, PathBuf};
10use std::sync::Arc;
11
12use casa_types::measures::{MPosition, MeasuresProvider};
13
14use crate::SessionResources;
15use crate::simobserve::SimobserveError;
16
17/// Physical inputs computed by `simobserve` for figure preparation.
18pub struct SimobservePlotInput<'a> {
19    /// Requested CASA graphics mode (`"file"`, `"screen"`, `"both"`).
20    pub graphics_mode: &'a str,
21    /// Project directory (`fileroot`).
22    pub directory: &'a Path,
23    /// Product stem (`<project>.<configuration>`).
24    pub stem: &'a str,
25    /// Whether observations were predicted (`int` or `sd`).
26    pub predict: bool,
27    /// Whether the observation is interferometric (`true`) or total power (`false`).
28    pub uvmode: bool,
29    /// Whether the sky model is component-only without a flat image input.
30    pub components_only: bool,
31    /// Component-list table path when supplied.
32    pub complist: &'a str,
33    /// Flattened sky model image path.
34    pub sky_flat: &'a Path,
35    /// Sky model centre direction string (`simutil` format).
36    pub sky_refdir: &'a str,
37    /// Reference frequency of the sky model, in GHz.
38    pub sky_specrefval_ghz: f64,
39    /// Pointing offsets relative to the sky model centre, in degrees.
40    pub pointing_offsets_deg: Vec<[f64; 2]>,
41    /// Median pointing centre direction string (`simutil` format).
42    pub pointing_centre: String,
43    /// Primary beam FWHM, in arcseconds.
44    pub primary_beam_arcsec: f64,
45    /// Antenna ITRF positions `[x, y, z]` in metres.
46    pub antenna_positions: Vec<[f64; 3]>,
47    /// Antenna dish diameters in metres.
48    pub antenna_diameters: &'a [f64],
49    /// Antenna pad/station names.
50    pub antenna_names: &'a [String],
51    /// Observatory reference position.
52    pub observatory: &'a MPosition,
53    /// Measures provider for ephemeris evaluation.
54    pub measures: Arc<dyn MeasuresProvider>,
55    /// Reference epoch MJD (TAI days).
56    pub refdate_mjd: f64,
57    /// Path to the MeasurementSet observed by this run.
58    pub measurement_set: PathBuf,
59    /// Path to the numerical `.quick.psf` image when produced.
60    pub quick_psf: PathBuf,
61    /// Execution resource policy for memory admission.
62    pub resources: SessionResources,
63}

65/// Physical inputs computed by `simanalyze` for figure preparation.
66pub struct SimanalyzePlotInput<'a> {
67    /// Requested CASA graphics mode (`"file"`, `"screen"`, `"both"`).
68    pub graphics_mode: &'a str,
69    /// Project directory.
70    pub directory: &'a Path,
71    /// Active product stem (with `.feather` suffix stripped when naming plots).
72    pub stem: &'a str,
73    /// Root path prefix (`directory.join(stem)`).
74    pub root: &'a Path,
75    /// Sky model path (`plan.model`).
76    pub model: &'a Path,
77    /// MeasurementSet path (`plan.ms`).
78    pub ms: &'a Path,
79    /// Flattened sky model image path.
80    pub model_flat: &'a Path,
81    /// Whether synthesis inputs were present.
82    pub has_synthesis: bool,
83    /// Whether `image` was requested.
84    pub image: bool,
85    /// Whether `analyze` was requested.
86    pub analyze: bool,
87    /// Panel selection flags.
88    pub showuv: bool,
89    /// Panel selection flags.
90    pub showpsf: bool,
91    /// Panel selection flags.
92    pub showmodel: bool,
93    /// Panel selection flags.
94    pub showconvolved: bool,
95    /// Panel selection flags.
96    pub showclean: bool,
97    /// Panel selection flags.
98    pub showresidual: bool,
99    /// Panel selection flags.
100    pub showdifference: bool,
101    /// Panel selection flags.
102    pub showfidelity: bool,
103    /// Optional explicit output file path (used by `simalma` TP analysis).
104    pub output_file_override: Option<PathBuf>,
105    /// Execution resource policy for memory admission.
106    pub resources: SessionResources,
107}

109/// Inputs for the four-panel ALMA/ACA/TP combination summary figure.
110pub struct SimalmaCombinationInput<'a> {
111    /// Project directory.
112    pub directory: &'a Path,
113    /// Project name.
114    pub project: &'a str,
115    /// Stem of the highest-resolution interferometer configuration.
116    pub best_stem: String,
117    /// Stem of the synthesis image used in combination.
118    pub high_stem: String,
119    /// Model preparation memory estimate in bytes.
120    pub model_preparation_bytes: u64,
121    /// Execution resource policy for memory admission.
122    pub resources: SessionResources,
123}

125/// Finalization inputs emitted at the end of a `simalma` run.
126pub struct SimalmaFinishInput<'a> {
127    /// Requested CASA graphics mode (`"file"`, `"screen"`, `"both"`).
128    pub graphics_mode: &'a str,
129    /// Optional combination figure inputs when total-power + imaging ran.
130    pub combination: Option<SimalmaCombinationInput<'a>>,
131}

133/// Callback sink receiving simulation figure inputs from `simobserve`,
134/// `simanalyze`, and `simalma`.
135pub trait SimulationPlotSink {
136    /// Record figure inputs from a `simobserve` run.
137    fn record_simobserve(&mut self, input: SimobservePlotInput<'_>) -> Result<(), SimobserveError>;
138
139    /// Record figure inputs from a `simanalyze` run.
140    fn record_simanalyze(&mut self, input: SimanalyzePlotInput<'_>) -> Result<(), SimobserveError>;
141
142    /// Finalize figures at the end of a `simalma` run (including optional
143    /// combination panel and `file`/`screen`/`both` normalization).
144    fn finish_simalma(&mut self, input: SimalmaFinishInput<'_>) -> Result<(), SimobserveError>;
145}
