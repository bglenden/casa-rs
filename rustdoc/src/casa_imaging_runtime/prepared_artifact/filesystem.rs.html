5pub(super) fn prepare_private_root(path: &Path) -> Result<PathBuf, PreparedArtifactError> {
6    let path = if path.is_absolute() {
7        path.to_path_buf()
8    } else {
9        std::env::current_dir()?.join(path)
10    };
11    reject_casa_visible_root(&path)?;
12    let mut missing = Vec::new();
13    let mut cursor = path.as_path();
14    loop {
15        match cursor.symlink_metadata() {
16            Ok(_) => {
17                let existing = validate_existing_private_ancestors(cursor, missing.is_empty())?;
18                let root = create_private_missing(existing, &missing)?;
19                let canonical = fs::canonicalize(&root)?;
20                reject_casa_visible_root(&canonical)?;
21                reject_casa_cache_contents(&canonical)?;
22                if !canonical.is_dir() {
23                    return Err(PreparedArtifactError::UnknownCacheEntry(canonical));
24                }
25                return Ok(canonical);
26            }
27            Err(error) if error.kind() == io::ErrorKind::NotFound => {
28                let component = cursor
29                    .file_name()
30                    .ok_or_else(|| PreparedArtifactError::UnknownCacheEntry(path.clone()))?;
31                missing.push(component.to_owned());
32                cursor = cursor
33                    .parent()
34                    .filter(|parent| !parent.as_os_str().is_empty())
35                    .unwrap_or_else(|| Path::new("."));
36            }
37            Err(error) => return Err(error.into()),
38        }
39    }
40}

42pub(super) fn validate_existing_private_ancestors(
43    path: &Path,
44    inventory_candidate: bool,
45) -> Result<PathBuf, PreparedArtifactError> {
46    let mut nearest = None;
47    for (index, ancestor) in path.ancestors().enumerate() {
48        let metadata = ancestor.symlink_metadata()?;
49        let canonical = fs::canonicalize(ancestor)?;
50        reject_casa_visible_root(&canonical)?;
51        if !canonical.is_dir() {
52            return Err(PreparedArtifactError::UnknownCacheEntry(
53                ancestor.to_path_buf(),
54            ));
55        }
56        if index == 0 && inventory_candidate {
57            reject_casa_cache_contents(&canonical)?;
58        } else {
59            reject_casacore_table_directory(&canonical)?;
60        }
61        if index == 0 && metadata.file_type().is_symlink() {
62            return Err(PreparedArtifactError::UnknownCacheEntry(
63                ancestor.to_path_buf(),
64            ));
65        }
66        nearest.get_or_insert(canonical);
67    }
68    nearest.ok_or_else(|| PreparedArtifactError::UnknownCacheEntry(path.to_path_buf()))
69}

71pub(super) fn reject_casacore_table_directory(path: &Path) -> Result<(), PreparedArtifactError> {
72    for entry in fs::read_dir(path)? {
73        let child = entry?.path();
74        let name = child
75            .file_name()
76            .ok_or_else(|| PreparedArtifactError::UnknownCacheEntry(child.clone()))?;
77        if casacore_table_marker_name(&name.to_string_lossy()) {
78            return Err(PreparedArtifactError::CasaVisiblePath(child));
79        }
80    }
81    Ok(())
82}

84pub(super) fn create_private_missing(
85    mut parent: PathBuf,
86    missing: &[std::ffi::OsString],
87) -> Result<PathBuf, PreparedArtifactError> {
88    for component in missing.iter().rev() {
89        let child = parent.join(component);
90        match child.symlink_metadata() {
91            Ok(metadata) => {
92                let canonical = fs::canonicalize(&child)?;
93                reject_casa_visible_root(&canonical)?;
94                if metadata.file_type().is_symlink() || !metadata.file_type().is_dir() {
95                    return Err(PreparedArtifactError::UnknownCacheEntry(child));
96                }
97            }
98            Err(error) if error.kind() == io::ErrorKind::NotFound => {
99                fs::create_dir(&child)?;
100            }
101            Err(error) => return Err(error.into()),
102        }
103        parent = child;
104    }
105    Ok(parent)
106}

108pub(super) fn ensure_private_child_directory(
109    parent: &Path,
110    child: &Path,
111) -> Result<(), PreparedArtifactError> {
112    match child.symlink_metadata() {
113        Ok(metadata) => {
114            let canonical = fs::canonicalize(child)?;
115            reject_casa_visible_root(&canonical)?;
116            if metadata.file_type().is_symlink() || canonical != child {
117                return Err(PreparedArtifactError::UnknownCacheEntry(
118                    child.to_path_buf(),
119                ));
120            }
121            if !metadata.file_type().is_dir() {
122                return Err(PreparedArtifactError::UnknownCacheEntry(
123                    child.to_path_buf(),
124                ));
125            }
126        }
127        Err(error) if error.kind() == io::ErrorKind::NotFound => {
128            if fs::canonicalize(parent)? != parent {
129                return Err(PreparedArtifactError::UnknownCacheEntry(
130                    parent.to_path_buf(),
131                ));
132            }
133            fs::create_dir(child)?;
134            let canonical = fs::canonicalize(child)?;
135            reject_casa_visible_root(&canonical)?;
136            if canonical != child {
137                return Err(PreparedArtifactError::UnknownCacheEntry(
138                    child.to_path_buf(),
139                ));
140            }
141        }
142        Err(error) => return Err(error.into()),
143    }
144    Ok(())
145}

147pub(super) fn reject_casa_visible_root(path: &Path) -> Result<(), PreparedArtifactError> {
148    if path.components().any(|component| {
149        let name = component.as_os_str().to_string_lossy();
150        casa_visible_name(&name)
151    }) {
152        Err(PreparedArtifactError::CasaVisiblePath(path.to_path_buf()))
153    } else {
154        Ok(())
155    }
156}

158pub(super) fn reject_casa_source_path(
159    path: &Path,
160    evidence: &mut ValidationEvidence,
161    demand_id: &str,
162) -> Result<(), PreparedArtifactError> {
163    reject_casa_visible_root(path)?;
164    for ancestor in path.ancestors().skip(1) {
165        let table_dat = ancestor.join("table.dat");
166        let table_info = ancestor.join("table.info");
167        evidence.source_read_operation(demand_id);
168        let has_table_dat = match table_dat.symlink_metadata() {
169            Ok(metadata) => metadata.file_type().is_file(),
170            Err(error) if error.kind() == io::ErrorKind::NotFound => false,
171            Err(error) => return Err(error.into()),
172        };
173        evidence.source_read_operation(demand_id);
174        let has_table_info = match table_info.symlink_metadata() {
175            Ok(metadata) => metadata.file_type().is_file(),
176            Err(error) if error.kind() == io::ErrorKind::NotFound => false,
177            Err(error) => return Err(error.into()),
178        };
179        if has_table_dat || has_table_info {
180            return Err(PreparedArtifactError::CasaVisiblePath(
181                ancestor.to_path_buf(),
182            ));
183        }
184    }
185    Ok(())
186}

188pub(super) fn reject_casa_cache_contents(root: &Path) -> Result<(), PreparedArtifactError> {
189    for entry in fs::read_dir(root)? {
190        let path = entry?.path();
191        let name = path
192            .file_name()
193            .ok_or_else(|| PreparedArtifactError::UnknownCacheEntry(path.clone()))?;
194        let name = name.to_string_lossy();
195        if casa_visible_name(&name) || casacore_table_marker_name(&name) {
196            return Err(PreparedArtifactError::CasaVisiblePath(path));
197        }
198    }
199    Ok(())
200}

202pub(super) fn casacore_table_marker_name(name: &str) -> bool {
203    let name = name.to_ascii_lowercase();
204    matches!(name.as_str(), "table.dat" | "table.info" | "table.lock")
205        || name
206            .strip_prefix("table.f")
207            .and_then(|suffix| suffix.as_bytes().first())
208            .is_some_and(u8::is_ascii_digit)
209}

211pub(super) fn casa_visible_name(name: &str) -> bool {
212    let name = name.to_ascii_lowercase();
213    name.ends_with(".im")
214        || name.ends_with(".ms")
215        || name.starts_with("cfs_")
216        || name.starts_with("wtcfs_")
217}

219pub(super) fn valid_segment_name(name: &str) -> bool {
220    valid_identifier(name)
221        && name != MANIFEST_FILE
222        && name != PAYLOAD_FILE
223        && !name.starts_with("CFS_")
224        && !name.starts_with("WTCFS_")
225        && !casa_visible_name(name)
226}

228pub(super) fn valid_identifier(value: &str) -> bool {
229    !value.is_empty()
230        && value.len() <= MAX_IDENTIFIER_BYTES
231        && value
232            .bytes()
233            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.' | b'@'))
234}

236pub(super) fn directory_size_counted(
237    path: &Path,
238    evidence: &mut ValidationEvidence,
239) -> Result<u64, PreparedArtifactError> {
240    with_directory_paths_counted(path, evidence, MAX_ENTRY_FILES, |evidence, paths| {
241        paths.iter().try_fold(0_u64, |total, entry| {
242            evidence.store_read_operation();
243            let metadata = entry.symlink_metadata()?;
244            if !metadata.file_type().is_file() {
245                return Err(PreparedArtifactError::UnknownCacheEntry(
246                    entry.to_path_buf(),
247                ));
248            }
249            total
250                .checked_add(metadata.len())
251                .ok_or(PreparedArtifactError::ArtifactTooLarge)
252        })
253    })
254}

256pub(super) fn with_directory_paths_counted<T>(
257    path: &Path,
258    evidence: &mut ValidationEvidence,
259    limit: usize,
260    use_paths: impl FnOnce(&mut ValidationEvidence, &[Box<Path>]) -> Result<T, PreparedArtifactError>,
261) -> Result<T, PreparedArtifactError> {
262    evidence.store_read_operation();
263    let entries = fs::read_dir(path)?;
264    evidence.observe_file_descriptors(2);
265    let mut paths = Vec::with_capacity(limit);
266    let inventory = (|| {
267        for entry in entries {
268            evidence.store_read_operation();
269            if paths.len() == limit {
270                return Err(PreparedArtifactError::CacheEntryBudgetExceeded {
271                    required: limit.saturating_add(1),
272                    budget: limit,
273                });
274            }
275            let path = entry?.path();
276            let component = path
277                .file_name()
278                .ok_or_else(|| PreparedArtifactError::UnknownCacheEntry(path.clone()))?;
279            if component.as_encoded_bytes().len() > MAX_CACHE_COMPONENT_BYTES {
280                return Err(PreparedArtifactError::UnknownCacheEntry(path));
281            }
282            paths.push(path.into_boxed_path());
283        }
284        paths.sort_unstable();
285        Ok(())
286    })();
287    let resident_bytes = observed_path_inventory_bytes(&paths, paths.capacity());
288    evidence.with_resident(resident_bytes, |evidence| {
289        inventory?;
290        use_paths(evidence, &paths)
291    })
292}

294pub(super) fn root_inventory_limit(
295    budget: PreparedArtifactBudget,
296) -> Result<usize, PreparedArtifactError> {
297    budget
298        .entries
299        .checked_mul(2)
300        .and_then(|entries| entries.checked_add(1))
301        .ok_or(PreparedArtifactError::ArtifactTooLarge)
302}

304pub(super) fn inventory_resident_reservation(
305    cache: &Path,
306    budget: PreparedArtifactBudget,
307) -> Result<u64, PreparedArtifactError> {
308    let cache_path_bytes = cache.as_os_str().as_encoded_bytes().len();
309    let root_paths = path_inventory_resident_reservation(
310        cache_path_bytes,
311        root_inventory_limit(budget)?,
312        MAX_CACHE_COMPONENT_BYTES,
313    )?;
314    let entry_directory_bytes = cache_path_bytes
315        .checked_add(1 + MAX_CACHE_COMPONENT_BYTES)
316        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
317    let entry_paths = path_inventory_resident_reservation(
318        entry_directory_bytes,
319        MAX_ENTRY_FILES,
320        MAX_CACHE_COMPONENT_BYTES,
321    )?;
322    let cache_entries = fixed_vec_resident_reservation::<CacheInventoryEntry>(budget.entries)?;
323    let evictions = fixed_vec_resident_reservation::<(ArtifactIdentity, u64)>(budget.entries)?;
324    root_paths
325        .checked_add(entry_paths)
326        .and_then(|bytes| bytes.checked_add(cache_entries))
327        .and_then(|bytes| bytes.checked_add(evictions))
328        .ok_or(PreparedArtifactError::ArtifactTooLarge)
329}

331pub(super) fn source_descriptor_reservation(segments: usize) -> Result<u64, PreparedArtifactError> {
332    let input_bytes = size_of::<PreparedArtifactSourceSegment>()
333        .checked_add(MAX_IDENTIFIER_BYTES)
334        .and_then(|bytes| bytes.checked_add(MAX_SOURCE_PATH_BYTES))
335        .and_then(|bytes| bytes.checked_add(MAX_SOURCE_PATH_BYTES))
336        .and_then(|bytes| bytes.checked_mul(segments))
337        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
338    let canonical_path_bytes = size_of::<PathBuf>()
339        .checked_add(MAX_SOURCE_PATH_BYTES)
340        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
341    let source_counters = size_of::<SourceIoCounter>()
342        .checked_add(MAX_IDENTIFIER_BYTES)
343        .and_then(|bytes| bytes.checked_add(3 * 64))
344        .and_then(|bytes| bytes.checked_mul(segments))
345        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
346    u64::try_from(
347        input_bytes
348            .checked_add(canonical_path_bytes)
349            .and_then(|bytes| bytes.checked_add(source_counters))
350            .ok_or(PreparedArtifactError::ArtifactTooLarge)?,
351    )
352    .map_err(|_| PreparedArtifactError::ArtifactTooLarge)
353}

355pub(super) fn path_inventory_resident_reservation(
356    directory_bytes: usize,
357    entries: usize,
358    component_bytes: usize,
359) -> Result<u64, PreparedArtifactError> {
360    let path_bytes = directory_bytes
361        .checked_add(1)
362        .and_then(|bytes| bytes.checked_add(component_bytes))
363        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
364    let entry_bytes = size_of::<Box<Path>>()
365        .checked_add(path_bytes)
366        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
367    let allocation = entries
368        .checked_mul(entry_bytes)
369        .and_then(|bytes| bytes.checked_add(size_of::<Vec<Box<Path>>>()))
370        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
371    u64::try_from(allocation).map_err(|_| PreparedArtifactError::ArtifactTooLarge)
372}

374pub(super) fn fixed_vec_resident_reservation<T>(
375    entries: usize,
376) -> Result<u64, PreparedArtifactError> {
377    let allocation = entries
378        .checked_mul(size_of::<T>())
379        .and_then(|bytes| bytes.checked_add(size_of::<Vec<T>>()))
380        .ok_or(PreparedArtifactError::ArtifactTooLarge)?;
381    u64::try_from(allocation).map_err(|_| PreparedArtifactError::ArtifactTooLarge)
382}

384pub(super) fn sync_directory_counted(
385    path: &Path,
386    evidence: &mut ValidationEvidence,
387) -> Result<(), PreparedArtifactError> {
388    evidence.store_control_operation();
389    let directory = File::open(path)?;
390    evidence.observe_file_descriptors(2);
391    evidence.store_write_operation();
392    directory.sync_all()?;
393    Ok(())
394}

396pub(super) fn remove_staging_counted(
397    path: &Path,
398    evidence: &mut ValidationEvidence,
399) -> Result<(), PreparedArtifactError> {
400    evidence.store_control_operation();
401    match path.symlink_metadata() {
402        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
403        Err(error) => Err(error.into()),
404        Ok(metadata) if metadata.file_type().is_dir() => {
405            evidence.store_write_operation();
406            fs::remove_dir_all(path)?;
407            let parent = path
408                .parent()
409                .ok_or_else(|| PreparedArtifactError::UnknownCacheEntry(path.to_path_buf()))?;
410            sync_directory_counted(parent, evidence)
411        }
412        Ok(_) => Err(PreparedArtifactError::UnknownCacheEntry(path.to_path_buf())),
413    }
414}

416pub(super) fn map_incomplete(error: io::Error) -> PreparedArtifactError {
417    if error.kind() == io::ErrorKind::UnexpectedEof || error.kind() == io::ErrorKind::NotFound {
418        PreparedArtifactError::IncompleteArtifact
419    } else {
420        error.into()
421    }
