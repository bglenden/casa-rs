7// Share CTPatchedInterp's curves, flag registration, phase unwrapping and
8// short-axis rules with simulator application.
9fn resample(
10    grid: &GainGrid,
11    input: &[f64],
12    output: &[f64],
13    mode: crate::table::FrequencyInterpolation,
14    relative: bool,
15    parameter_family: Option<TimeParameterFamily>,
16    ordered_tsys: bool,
17) -> Result<GainGrid, ApplyExecutionError> {
18    let (relative_input, relative_output);
19    let (input, output) = if relative {
20        relative_input = relative_axis(input);
21        relative_output = relative_axis(output);
22        (relative_input.as_slice(), relative_output.as_slice())
23    } else {
24        (input, output)
25    };
26    let solution = crate::table::ComplexSolution::of(grid);
27    if matches!(
28        parameter_family,
29        Some(TimeParameterFamily::Sky | TimeParameterFamily::SpectralOpacity)
30    ) {
31        return solution
32            .resample_real_array_frequency(input, output, mode)
33            .map(crate::table::ComplexSolution::into_gain_grid);
34    }
35    if !matches!(
36        parameter_family,
37        Some(TimeParameterFamily::Tsys | TimeParameterFamily::PositionAngle)
38    ) {
39        return solution
40            .resample_frequency(input, output, mode)
41            .map(crate::table::ComplexSolution::into_gain_grid);
42    }
43
44    let mut sampled = solution
45        .resample_real_frequency(input, output, mode)?
46        .into_gain_grid();
47    if matches!(parameter_family, Some(TimeParameterFamily::PositionAngle)) {
48        return Ok(sampled);
49    }
50    // StandardTsys uses sqrt(Tsys) as each station's correction. The regular
51    // apply path represents the same factor as a divisor.
52    for (gain, flagged) in sampled.values.iter_mut().zip(&mut sampled.flags) {
53        let tsys = gain.re;
54        *flagged |= tsys <= 0.0 || tsys.is_nan();
55        *gain = if *flagged {
56            Complex32::new(1.0, 0.0)
57        } else if ordered_tsys {
58            Complex32::new(tsys.sqrt(), 0.0)
59        } else {
60            Complex32::new(1.0 / tsys.sqrt(), 0.0)
61        };
62    }
63    Ok(sampled)
64}

66pub(super) fn prepare(
67    grid: Arc<CalibrationGrid>,
68    input: &[f64],
69    output: &crate::plan::SpectralWindowPlan,
70    cache: Option<&mut RowGeometryCache>,
71    table: &LoadedCalibrationTable,
72    transient_grid: bool,
73    ordered_gain: bool,
74) -> Result<Arc<CalibrationGrid>, ApplyExecutionError> {
75    let interpolation = table.interp;
76    let path = &table.path;
77    let (gain, opacity) = match grid.as_ref() {
78        CalibrationGrid::Complex(gain) => (gain, false),
79        CalibrationGrid::SpectralOpacity(gain) => (gain, true),
80        _ => return Ok(grid),
81    };
82    if input.len() != gain.channel_count {
83        return Err(ApplyExecutionError::UnsupportedParameterShape {
84            path: path.display().to_string(),
85            shape: gain.values.shape().to_vec(),
86        });
87    }
88    let key = MaterializedGridKey {
89        row: RowGeometryKey::new(0., -1, -1),
90        kind: 4,
91        data_spw_id: output.spw_id,
92        grid_id: Arc::as_ptr(&grid) as usize,
93    };
94    let cache = cache.map(|cache| {
95        if transient_grid {
96            &mut cache.transient_frequency_grids
97        } else {
98            &mut cache.materialized_grids
99        }
100    });
101    if let Some(prepared) = cache.as_ref().and_then(|grids| grids.get(&key)) {
102        return Ok(prepared.clone());
103    }
104    // A scalar is broadcast during application. On an aligned, unflagged
105    // axis CASA copies the time interpolation result without reconstructing
106    // it again. In both cases the ordered applicator can reuse this grid.
107    let ordered_tsys =
108        ordered_gain && matches!(table.time_parameter_family, Some(TimeParameterFamily::Tsys));
109    let unchanged = ordered_gain
110        && !ordered_tsys
111        && (gain.channel_count == 1
112            || (input == output.channel_frequencies_hz && gain.flags.iter().all(|flag| !flag)));
113    let prepared = if unchanged {
114        grid
115    } else {
116        let sampled = resample(
117            gain,
118            input,
119            &output.channel_frequencies_hz,
120            if matches!(table.time_parameter_family, Some(TimeParameterFamily::Sky)) {
121                interpolation.sky_frequency_mode()
122            } else {
123                interpolation.frequency_mode()
124            },
125            matches!(
126                interpolation,
127                ApplyInterpolationMode::Casa {
128                    relative_frequency: true,
129                    ..
130                }
131            ),
132            if opacity {
133                Some(TimeParameterFamily::SpectralOpacity)
134            } else {
135                table.time_parameter_family
136            },
137            ordered_tsys,
138        )?;
139        Arc::new(if opacity {
140            CalibrationGrid::SpectralOpacity(sampled)
141        } else {
142            CalibrationGrid::Complex(sampled)
143        })
144    };
145    if let Some(cache) = cache {
146        cache.insert(key, prepared.clone());
147    }
148    Ok(prepared)
149}

151// BJones::calcWtScale uses squared RMS amplitude of the unflagged full-band
152// time-interpolated solution, before any selection/frequency interpolation.
153fn mean_power(grid: &GainGrid, receptor: usize) -> f32 {
154    let receptor = receptor.min(grid.receptor_count.saturating_sub(1));
155    let mut total = 0_f32;
156    let mut count = 0;
157    for channel in 0..grid.channel_count {
158        if !grid.flag_at(receptor, channel) {
159            let amplitude = grid.value_at(receptor, channel).norm();
160            total += amplitude * amplitude;
161            count += 1;
162        }
163    }
164    if count == 0 {
165        1.
166    } else {
167        let rms = (total / count as f32).sqrt();
168        rms * rms
169    }
170}

172#[derive(Default)]
173pub(super) struct PowerCache {
174    recent: HashMap<(usize, i32), (Arc<CalibrationGrid>, [f32; 2])>,
175}
176
177impl PowerCache {
178    fn powers(&mut self, table: usize, antenna: (i32, &Arc<CalibrationGrid>)) -> Option<[f32; 2]> {
179        let (id, grid) = antenna;
180        let CalibrationGrid::Complex(values) = grid.as_ref() else {
181            return None;
182        };
183        let make_powers = || [mean_power(values, 0), mean_power(values, 1)];
184        // One solution per table and antenna suffices for a visibility chunk.
185        // Keep the owning Arc so pointer identity cannot survive address reuse.
186        let entry = self
187            .recent
188            .entry((table, id))
189            .or_insert_with(|| (Arc::clone(grid), make_powers()));
190        if !Arc::ptr_eq(&entry.0, grid) {
191            *entry = (Arc::clone(grid), make_powers());
192        }
193        Some(entry.1)
194    }
195
196    pub(super) fn factors(
197        &mut self,
198        table: usize,
199        first: (i32, &Arc<CalibrationGrid>),
200        second: (i32, &Arc<CalibrationGrid>),
201        correlations: &[i32],
202    ) -> Option<Vec<f32>> {
203        let first = self.powers(table, first)?;
204        let second = self.powers(table, second)?;
205        (0..correlations.len())
206            .map(|index| {
207                let (left, right) = receptors(correlations, index)?;
208                Some(first[left] * second[right])
209            })
210            .collect()
211    }
212}

214pub(super) fn temperature_weight_factors(
215    first: &CalibrationGrid,
216    second: &CalibrationGrid,
217    correlations: &[i32],
218    channels: DataChannelGrid<'_>,
219    ordered_tsys: bool,
220) -> Option<Vec<f32>> {
221    let (CalibrationGrid::Complex(first), CalibrationGrid::Complex(second)) = (first, second)
222    else {
223        return None;
224    };
225    (0..correlations.len())
226        .map(|index| {
227            let (left, right) = receptors(correlations, index)?;
228            Some(
229                inverse_mean_tsys(first, left, channels, ordered_tsys)
230                    * inverse_mean_tsys(second, right, channels, ordered_tsys),
231            )
232        })
233        .collect()
234}
235
236fn inverse_mean_tsys(
237    grid: &GainGrid,
238    receptor: usize,
239    channels: DataChannelGrid<'_>,
240    ordered_tsys: bool,
241) -> f32 {
242    let receptor = receptor.min(grid.receptor_count.saturating_sub(1));
243    let mut total = 0.0_f64;
244    let mut count = 0;
245    for index in 0..channels.frequencies_hz.len() {
246        let channel = channels.original_index(index);
247        let gain = f64::from(grid.value_at(receptor, channel).re);
248        if !grid.flag_at(receptor, channel) && gain > 0.0 {
249            let power = gain * gain;
250            total += if ordered_tsys { power } else { power.recip() };
251            count += 1;
252        }
253    }
254    if count == 0 {
255        1.0
256    } else {
257        (f64::from(count) / total) as f32
258    }
259}

261pub(super) fn receptors(correlations: &[i32], index: usize) -> Option<(usize, usize)> {
262    let pair = correlation_receptors(correlations[index])?;
263    // CASA uses the first receptor for single-correlation bandpass and
264    // switched-power application, including a sole YY or LL product.
265    Some(if correlations.len() == 1 {
266        (0, 0)
267    } else {
268        pair
269    })
270}

358        let output = crate::apply::SpectralWindowPlan {

434    #[test]
435    fn missing_bandpass_bin_interpolates_amplitude_phase_and_uses_full_band_power() {
436        // Two valid endpoints: amplitude 1/phase 0 and amplitude 3/phase pi/2.
437        // The missing middle is amplitude 2/phase pi/4, independent of sideband.
438        for descending in [false, true] {
439            let mut frequencies = vec![1., 2., 3.];
440            let mut samples = vec![
441                Complex32::new(1., 0.),
442                Complex32::new(100., 0.),
443                Complex32::new(0., 3.),
444            ];
445            if descending {
446                frequencies.reverse();
447                samples.reverse();
448            }
449            let mut gain = GainGrid {
450                receptor_count: 1,
451                channel_count: 3,
452                values: ArrayD::from_shape_vec(IxDyn(&[1, 3]), samples).unwrap(),
453                flags: ArrayD::from_shape_vec(IxDyn(&[1, 3]), vec![false, true, false]).unwrap(),
454            };
455            let result = resample(
456                &gain,
457                &frequencies,
458                &[0., 1., 2., 3., 4.],
459                crate::table::FrequencyInterpolation::Linear,
460                false,
461                None,
462                false,
463            )
464            .unwrap();
465            let expected = [
466                Complex32::new(1., 0.),
467                Complex32::new(1., 0.),
468                Complex32::new(2_f32.sqrt(), 2_f32.sqrt()),
469                Complex32::new(0., 3.),
470                Complex32::new(0., 3.),
471            ];
472            for (value, expected) in result.values.iter().zip(expected) {
473                assert!((*value - expected).norm() < 1e-6);
474            }
475            assert!(result.flags.iter().all(|flag| !flag));
476            assert_eq!(mean_power(&gain, 0), 5.);
477            let mut other = gain.clone();
478            other.values.mapv_inplace(|value| value * 2.);
479            let original = Arc::new(CalibrationGrid::Complex(gain.clone()));
480            let scaled = Arc::new(CalibrationGrid::Complex(other));
481            let mut powers = PowerCache::default();
482            assert_eq!(
483                powers.factors(0, (0, &original), (1, &scaled), &[9]),
484                Some(vec![100.])
485            );
486            assert_eq!(
487                powers.factors(0, (0, &scaled), (1, &scaled), &[9]),
488                Some(vec![400.])
489            );
490            gain.flags.fill(true);
491            let missing = resample(
492                &gain,
493                &frequencies,
494                &[2.],
495                crate::table::FrequencyInterpolation::Linear,
496                false,
497                None,
498                false,
499            )
500            .unwrap();
501            assert!(missing.flags.iter().all(|flag| *flag));
502            assert!(
503                missing
504                    .values
505                    .iter()
506                    .all(|value| *value == Complex32::new(0., 0.))
507            );
508            assert_eq!(mean_power(&gain, 0), 1.);
509        }
510    }
