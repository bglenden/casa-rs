3use std::borrow::Cow;
4
5/// Expand casacore path variables and the current user's home directory.
6/// Returns `None` when repeated expansion exceeds casacore's limit.
7pub(crate) fn expand_path(name: &str) -> Option<Cow<'_, str>> {
8    expand(name, false)
9}
10
11/// Also resolve named home directories for ASDM index path display.
12pub(crate) fn expand_asdm_path(name: &str) -> Option<Cow<'_, str>> {
13    expand(name, true)
14}
15
16fn expand(name: &str, named_home: bool) -> Option<Cow<'_, str>> {
17    if !name.starts_with('~') && !name.contains('$') {
18        return Some(Cow::Borrowed(name));
19    }
20    let mut path = Cow::Borrowed(name);
21    // casacore Path::expandName follows environment references recursively
22    // and stops expanding if the substitutions form a cycle.
23    for _ in 0..25 {
24        if let Some(tail) = path.strip_prefix('~') {
25            let user = tail.split('/').next().unwrap_or_default();
26            if (user.is_empty() || named_home)
27                && let Some(home) = home_directory(user).filter(|home| !home.is_empty())
28            {
29                path = Cow::Owned(format!("{home}{}", &tail[user.len()..]));
30            }
31        }
32        let mut changed = false;
33        let next = path
34            .split('/')
35            .map(|part| match expand_variable(part) {
36                Some(expanded) => {
37                    changed = true;
38                    Cow::Owned(expanded)
39                }
40                None => Cow::Borrowed(part),
41            })
42            .collect::<Vec<_>>()
43            .join("/");
44        if !changed {
45            return Some(path);
46        }
47        path = Cow::Owned(next);
48    }
49    None
50}

52fn home_directory(user: &str) -> Option<String> {
53    if user.is_empty() {
54        return std::env::var("HOME").ok();
55    }
56    #[cfg(unix)]
57    {
58        use std::ffi::{CStr, CString};
59        use std::mem::MaybeUninit;
60        use std::ptr::null_mut;
61
62        let user = CString::new(user).ok()?;
63        let mut scratch = vec![0; 16 * 1024];
64        let mut account = MaybeUninit::<libc::passwd>::uninit();
65        loop {
66            let mut result = null_mut();
67            // SAFETY: The buffers are live and writable throughout the lookup.
68            let status = unsafe {
69                libc::getpwnam_r(
70                    user.as_ptr(),
71                    account.as_mut_ptr(),
72                    scratch.as_mut_ptr(),
73                    scratch.len(),
74                    &mut result,
75                )
76            };
77            if status == libc::ERANGE {
78                scratch.resize(scratch.len().checked_mul(2)?, 0);
79                continue;
80            }
81            if status != 0 || result.is_null() {
82                return None;
83            }
84            // SAFETY: A successful lookup with a non-null result initialized
85            // the account and its NUL-terminated home path in the buffer.
86            return unsafe { CStr::from_ptr((*result).pw_dir) }
87                .to_str()
88                .ok()
89                .map(str::to_owned);
90        }
91    }
92    #[cfg(not(unix))]
93    None
94}

96fn expand_variable(part: &str) -> Option<String> {
97    let (prefix, expression) = part.split_once('$')?;
98    let (variable, suffix) = match expression.strip_prefix('{') {
99        Some(braced) => braced.split_once('}')?,
100        None => (expression, ""),
101    };
102    let value = std::env::var(variable)
103        .ok()
104        .filter(|value| !value.is_empty())?;
105    Some(format!("{prefix}{value}{suffix}"))
106}
