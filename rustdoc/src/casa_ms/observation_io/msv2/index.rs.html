2//! The MSv2 adapter's opening index: every MAIN row's partition and its
3//! cell on that partition's time × baseline grid, built once at opening so
4//! that estimates are arithmetic and every read is addressed by cell.
5//!
6//! Rows are partitioned the way XRADIO's converter does by default
7//! (`DATA_DESC_ID` × `OBS_MODE` × `OBSERVATION_ID` × `EPHEMERIS_ID`), plus
8//! the feed pair, which MSv4 has no axis for, plus whatever the
9//! [`PartitionScheme`] adds. Within a partition the grid is the unique
10//! times (exact values; no tolerance merges jittered times) by the unique
11//! ordered baselines; a cell two rows claim is refused, naming both rows,
12//! and a cell no row fills is padding.
13//!
14//! MAIN's key columns are read [`ROWS_PER_CHUNK`] rows at a time, in two
15//! passes: the first gathers each partition's times, baselines and
16//! processors and each scan's states, the second places every row in its
17//! cell. Opening therefore holds the grids (four bytes a cell) and one
18//! chunk of columns, whatever the MeasurementSet's size; a MAIN that fits
19//! in one chunk is read once.
20

30/// The cell of a grid no row fills.
31pub(super) const NO_ROW: u32 = u32::MAX;
32
33/// MAIN rows whose key columns are read at once while the index is built:
34/// about 15 MiB of columns and row numbers.
35pub(super) const ROWS_PER_CHUNK: usize = 1 << 18;
36
37/// One chunk of MAIN's key columns, refilled for every chunk.
38#[derive(Default)]
39struct KeyColumns {
40    /// The chunk's MAIN rows.
41    rows: Vec<usize>,
42    data_description: Vec<i32>,
43    time: Vec<f64>,
44    field: Vec<i32>,
45    scan_number: Vec<i32>,
46    state: Vec<i32>,
47    observation: Vec<i32>,
48    antenna1: Vec<i32>,
49    antenna2: Vec<i32>,
50    feed1: Vec<i32>,
51    feed2: Vec<i32>,
52    processor: Vec<i32>,
53}
54
55impl KeyColumns {
56    /// Read the key columns of MAIN rows `rows`.
57    fn read(&mut self, main: &Table, rows: Range<usize>) -> Result<(), ObservationError> {
58        let Self {
59            rows: indices,
60            data_description,
61            time,
62            field,
63            scan_number,
64            state,
65            observation,
66            antenna1,
67            antenna2,
68            feed1,
69            feed2,
70            processor,
71        } = self;
72        indices.clear();
73        indices.extend(rows);
74        let mut destinations = vec![RequiredScalarColumnDestination::new(
75            "TIME",
76            RequiredScalarColumnValuesMut::Float64(time),
77        )];
78        for (name, values) in [
79            ("DATA_DESC_ID", data_description),
80            ("FIELD_ID", field),
81            ("SCAN_NUMBER", scan_number),
82            ("STATE_ID", state),
83            ("OBSERVATION_ID", observation),
84            ("ANTENNA1", antenna1),
85            ("ANTENNA2", antenna2),
86            ("FEED1", feed1),
87            ("FEED2", feed2),
88            ("PROCESSOR_ID", processor),
89        ] {
90            destinations.push(RequiredScalarColumnDestination::new(
91                name,
92                RequiredScalarColumnValuesMut::Int32(values),
93            ));
94        }
95        main.required_scalar_columns_for_rows_into(indices, &mut destinations)
96            .map_err(table_error)
97    }
98
99    /// The baseline of chunk row `i`, MAIN row `row`.
100    fn baseline(&self, i: usize, row: usize) -> Result<OrderedBaseline, ObservationError> {
101        let antenna = |value: i32| {
102            u32::try_from(value)
103                .map(AntennaId)
104                .map_err(|_| invalid(format!("row {row} has a negative antenna {value}")))
105        };
106        Ok(OrderedBaseline {
107            antenna1: antenna(self.antenna1[i])?,
108            antenna2: antenna(self.antenna2[i])?,
109        })
110    }
111}

113/// What defines a partition of MAIN rows.
114#[derive(Clone, Debug, PartialEq, Eq, PartialOrd, Ord)]
115pub(super) struct PartitionKey {
116    pub(super) data_description: i32,
117    /// `STATE::OBS_MODE` of the rows' state; empty without a state.
118    pub(super) obs_mode: String,
119    pub(super) observation: i32,
120    /// The rows' field's `EPHEMERIS_ID`, when the column exists.
121    pub(super) ephemeris: Option<i32>,
122    pub(super) feeds: [i32; 2],
123    /// `FIELD_ID`, when the scheme splits by field.
124    pub(super) field: Option<i32>,
125    /// `SCAN_NUMBER`, when the scheme splits by scan.
126    pub(super) scan: Option<i32>,
127}

129/// A row's [`PartitionKey`] with its `OBS_MODE` as the mode's rank among
130/// the distinct modes: it orders as the key does and costs no allocation.
131#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
132struct RowKey {
133    data_description: i32,
134    obs_mode: usize,
135    observation: i32,
136    ephemeris: Option<i32>,
137    feeds: [i32; 2],
138    field: Option<i32>,
139    scan: Option<i32>,
140}
141
142/// The subtables' part of every row's key.
143struct Keys<'a> {
144    scheme: PartitionScheme,
145    /// Each `FIELD` row's `EPHEMERIS_ID`.
146    field_ephemeris: &'a [Option<i32>],
147    /// The distinct `OBS_MODE`s and the empty mode, ascending.
148    modes: Vec<String>,
149    /// Each `STATE` row's mode, by its rank in `modes`.
150    state_mode: Vec<usize>,
151    /// The empty mode's rank: a row's without a state.
152    no_mode: usize,
153}
154
155impl<'a> Keys<'a> {
156    fn new(
157        scheme: PartitionScheme,
158        state_modes: &[String],
159        field_ephemeris: &'a [Option<i32>],
160    ) -> Self {
161        let modes: Vec<String> = state_modes
162            .iter()
163            .cloned()
164            .chain([String::new()])
165            .collect::<BTreeSet<_>>()
166            .into_iter()
167            .collect();
168        let rank = |mode: &str| {
169            modes
170                .binary_search_by(|known| known.as_str().cmp(mode))
171                .expect("every mode is collected")
172        };
173        let state_mode = state_modes.iter().map(|mode| rank(mode)).collect();
174        let no_mode = rank("");
175        Self {
176            scheme,
177            field_ephemeris,
178            modes,
179            state_mode,
180            no_mode,
181        }
182    }
183
184    /// The key and field of chunk row `i`, MAIN row `row`.
185    fn of(
186        &self,
187        columns: &KeyColumns,
188        i: usize,
189        row: usize,
190    ) -> Result<(RowKey, FieldId), ObservationError> {
191        let field = columns.field[i];
192        let ephemeris = usize::try_from(field)
193            .ok()
194            .and_then(|index| self.field_ephemeris.get(index))
195            .ok_or_else(|| invalid(format!("row {row} has FIELD_ID {field} outside FIELD")))?;
196        let obs_mode = usize::try_from(columns.state[i])
197            .ok()
198            .and_then(|state| self.state_mode.get(state))
199            .copied()
200            .unwrap_or(self.no_mode);
201        let key = RowKey {
202            data_description: columns.data_description[i],
203            obs_mode,
204            observation: columns.observation[i],
205            ephemeris: *ephemeris,
206            feeds: [columns.feed1[i], columns.feed2[i]],
207            field: self.scheme.fields.then_some(field),
208            scan: self.scheme.scans.then_some(columns.scan_number[i]),
209        };
210        Ok((key, FieldId(field.unsigned_abs())))
211    }
212
213    fn partition_key(&self, key: RowKey) -> PartitionKey {
214        PartitionKey {
215            data_description: key.data_description,
216            obs_mode: self.modes[key.obs_mode].clone(),
217            observation: key.observation,
218            ephemeris: key.ephemeris,
219            feeds: key.feeds,
220            field: key.field,
221            scan: key.scan,
222        }
223    }
224}

226/// What the first pass learns of one partition.
227#[derive(Default)]
228struct Gathered {
229    /// Distinct `TIME`s, by their bits.
230    times: BTreeSet<u64>,
231    /// The last row's `TIME`: rows of one time usually follow each other.
232    last_time: Option<u64>,
233    baselines: BTreeSet<OrderedBaseline>,
234    processors: BTreeSet<i32>,
235    rows: usize,
236}
237
238/// What the first pass learns of all of MAIN.
239#[derive(Default)]
240struct Survey {
241    partitions: BTreeMap<RowKey, Gathered>,
242    /// Each `SCAN_NUMBER`'s `STATE_ID`s.
243    scan_states: BTreeMap<i32, BTreeSet<i32>>,
244}
245
246impl Survey {
247    /// Gather the rows of `columns`, MAIN rows from `first`.
248    fn gather(
249        &mut self,
250        keys: &Keys<'_>,
251        columns: &KeyColumns,
252        first: usize,
253    ) -> Result<(), ObservationError> {
254        let mut last_scan = None;
255        for i in 0..columns.time.len() {
256            let row = first + i;
257            let (key, _) = keys.of(columns, i, row)?;
258            let partition = self.partitions.entry(key).or_default();
259            let time = columns.time[i].to_bits();
260            if partition.last_time != Some(time) {
261                partition.times.insert(time);
262                partition.last_time = Some(time);
263            }
264            partition.baselines.insert(columns.baseline(i, row)?);
265            partition.processors.insert(columns.processor[i]);
266            partition.rows += 1;
267            let scan = (columns.scan_number[i], columns.state[i]);
268            if last_scan != Some(scan) {
269                self.scan_states.entry(scan.0).or_default().insert(scan.1);
270                last_scan = Some(scan);
271            }
272        }
273        Ok(())
274    }
275}

277/// One partition's grid while the second pass places its rows.
278struct Placing {
279    times_s: Vec<f64>,
280    baselines: Vec<OrderedBaseline>,
281    cells: Vec<u32>,
282    field_per_time: Vec<Option<FieldId>>,
283    scan_per_time: Vec<Option<ScanId>>,
284    rows: usize,
285    /// The last row's `TIME` bits and time index.
286    last_time: Option<(u64, usize)>,
287}
288
289impl Placing {
290    /// The empty grid of a partition the first pass gathered.
291    ///
292    /// # Errors
293    ///
294    /// When the partition's rows have several `PROCESSOR_ID`s (XRADIO
295    /// requires one per partition).
296    fn new(gathered: Gathered) -> Result<Self, ObservationError> {
297        if gathered.processors.len() > 1 {
298            return Err(ObservationError::Unsupported {
299                what: "rows of one partition with several PROCESSOR_IDs",
300            });
301        }
302        let mut times_s: Vec<f64> = gathered.times.into_iter().map(f64::from_bits).collect();
303        times_s.sort_by(f64::total_cmp);
304        let baselines: Vec<OrderedBaseline> = gathered.baselines.into_iter().collect();
305        let times = times_s.len();
306        Ok(Self {
307            cells: vec![NO_ROW; times * baselines.len()],
308            field_per_time: vec![None; times],
309            scan_per_time: vec![None; times],
310            times_s,
311            baselines,
312            rows: gathered.rows,
313            last_time: None,
314        })
315    }
316
317    /// Place chunk row `i`, MAIN row `row`, of field `field`.
318    ///
319    /// # Errors
320    ///
321    /// When its cell already holds a row, or its time already holds another
322    /// field or scan.
323    fn place(
324        &mut self,
325        columns: &KeyColumns,
326        i: usize,
327        row: usize,
328        field: FieldId,
329        scans: &BTreeMap<i32, ScanId>,
330    ) -> Result<(), ObservationError> {
331        let bits = columns.time[i].to_bits();
332        let t = match self.last_time {
333            Some((last, t)) if last == bits => t,
334            _ => {
335                let t = self
336                    .times_s
337                    .binary_search_by(|time| time.total_cmp(&columns.time[i]))
338                    .expect("every row's time is on the grid");
339                self.last_time = Some((bits, t));
340                t
341            }
342        };
343        let b = self
344            .baselines
345            .binary_search(&columns.baseline(i, row)?)
346            .expect("every row's baseline is on the grid");
347        let cell = &mut self.cells[t * self.baselines.len() + b];
348        if *cell != NO_ROW {
349            return Err(ObservationError::InvalidStore {
350                reason: format!(
351                    "rows {} and {row} are two acquisitions of one time and baseline; \
352                     the partition keys do not separate them",
353                    *cell
354                ),
355            });
356        }
357        *cell = u32::try_from(row).expect("rows are counted below NO_ROW");
358        let scan = *scans
359            .get(&columns.scan_number[i])
360            .expect("every SCAN_NUMBER is catalogued");
361        one_per_time(
362            &mut self.field_per_time[t],
363            field,
364            "two fields at one time of one partition",
365        )?;
366        one_per_time(
367            &mut self.scan_per_time[t],
368            scan,
369            "two scans at one time of one partition",
370        )
371    }
372
373    fn finish(self) -> Grid {
374        Grid {
375            times_s: self.times_s,
376            field_per_time: self
377                .field_per_time
378                .into_iter()
379                .map(|field| field.expect("every time has a row"))
380                .collect(),
381            scan_per_time: self
382                .scan_per_time
383                .into_iter()
384                .map(|scan| scan.expect("every time has a row"))
385                .collect(),
386            baselines: self.baselines,
387            cells: self.cells,
388            rows: self.rows,
389        }
390    }
391}

393/// The opening index of a MAIN table.
394#[derive(Debug, PartialEq)]
395pub(super) struct Index {
396    /// Every partition's key and grid, in key order.
397    pub(super) partitions: Vec<(PartitionKey, Grid)>,
398    /// The scans, by ascending `SCAN_NUMBER`.
399    pub(super) scans: Vec<ScanEntry>,
400}
401
402impl Index {
403    /// Partition `main`'s rows and place them on their grids, reading the
404    /// key columns `rows_per_chunk` rows at a time.
405    ///
406    /// `state_modes` is each `STATE` row's `OBS_MODE` and `field_ephemeris`
407    /// each `FIELD` row's `EPHEMERIS_ID`.
408    ///
409    /// # Errors
410    ///
411    /// When a row's field is not in `FIELD` or an antenna is negative, two
412    /// rows share a (time, baseline) cell, a time holds rows of two fields
413    /// or two scans, a partition's rows have several `PROCESSOR_ID`s, or
414    /// MAIN has too many rows to number in 32 bits.
415    pub(super) fn build(
416        main: &Table,
417        scheme: PartitionScheme,
418        state_modes: &[String],
419        field_ephemeris: &[Option<i32>],
420        rows_per_chunk: usize,
421    ) -> Result<Self, ObservationError> {
422        let rows = main.row_count();
423        if rows >= NO_ROW as usize {
424            return Err(invalid("more than 2^32 - 1 MAIN rows"));
425        }
426        let chunks: Vec<Range<usize>> = (0..rows)
427            .step_by(rows_per_chunk)
428            .map(|start| start..start.saturating_add(rows_per_chunk).min(rows))
429            .collect();
430        let keys = Keys::new(scheme, state_modes, field_ephemeris);
431        let mut columns = KeyColumns::default();
432        let mut survey = Survey::default();
433        for chunk in &chunks {
434            columns.read(main, chunk.clone())?;
435            survey.gather(&keys, &columns, chunk.start)?;
436        }
437        let (scans, scan_ids) = catalogue::scans(survey.scan_states, state_modes);
438        let order: Vec<RowKey> = survey.partitions.keys().copied().collect();
439        let mut grids = survey
440            .partitions
441            .into_values()
442            .map(Placing::new)
443            .collect::<Result<Vec<_>, _>>()?;
444        for chunk in &chunks {
445            if chunks.len() > 1 {
446                columns.read(main, chunk.clone())?;
447            }
448            let mut last: Option<(RowKey, usize)> = None;
449            for i in 0..columns.time.len() {
450                let row = chunk.start + i;
451                let (key, field) = keys.of(&columns, i, row)?;
452                let partition = match last {
453                    Some((known, partition)) if known == key => partition,
454                    _ => {
455                        let partition = order
456                            .binary_search(&key)
457                            .expect("the first pass saw every key");
458                        last = Some((key, partition));
459                        partition
460                    }
461                };
462                grids[partition].place(&columns, i, row, field, &scan_ids)?;
463            }
464        }
465        Ok(Self {
466            partitions: order
467                .into_iter()
468                .zip(grids)
469                .map(|(key, grid)| (keys.partition_key(key), grid.finish()))
470                .collect(),
471            scans,
472        })
473    }
474}

476/// One partition's time × baseline grid and the row at each cell.
477#[derive(Debug, PartialEq)]
478pub(super) struct Grid {
479    /// Unique times, ascending (MAIN `TIME`, seconds from MJD 0).
480    pub(super) times_s: Vec<f64>,
481    pub(super) field_per_time: Vec<FieldId>,
482    pub(super) scan_per_time: Vec<ScanId>,
483    /// Unique ordered baselines, ascending.
484    pub(super) baselines: Vec<OrderedBaseline>,
485    /// The MAIN row at cell `t × baselines.len() + b`, or [`NO_ROW`].
486    pub(super) cells: Vec<u32>,
487    /// Rows placed: cells that are not padding.
488    pub(super) rows: usize,
489}
490
491impl Grid {
492    /// The cells of times `times`.
493    pub(super) fn cells(&self, times: Range<usize>) -> &[u32] {
494        let per_time = self.baselines.len();
495        &self.cells[times.start * per_time..times.end * per_time]
496    }
497}

499/// Record `value` at a time, which every row of that time must share.
500fn one_per_time<T: Copy + PartialEq>(
501    slot: &mut Option<T>,
502    value: T,
503    what: &'static str,
504) -> Result<(), ObservationError> {
505    if slot.is_some_and(|known| known != value) {
506        return Err(ObservationError::Unsupported { what });
507    }
508    *slot = Some(value);
509    Ok(())
510}

512#[cfg(all(test, unix))]
513mod tests {
514    use std::fs::File;
515    use std::path::Path;
516
517    use super::*;
518    use crate::MeasurementSet;
519
520    /// The index does not depend on how many rows each chunk reads, with
521    /// either partition scheme: one row, a ragged last chunk, or all of
522    /// MAIN at once.
523    #[test]
524    fn the_index_does_not_depend_on_the_chunk_size() {
525        let root = tempfile::tempdir().expect("root");
526        let archive =
527            Path::new(env!("CARGO_MANIFEST_DIR")).join("tests/fixtures/ms_trial/ms_trial.ms.tgz");
528        tar::Archive::new(flate2::read::GzDecoder::new(
529            File::open(archive).expect("fixture"),
530        ))
531        .unpack(root.path())
532        .expect("unpack");
533        let ms = MeasurementSet::open_retained_read(root.path().join("ms_trial.ms")).expect("open");
534        let modes = catalogue::read_state_modes(&ms).expect("modes");
535        let ephemerides = catalogue::fields_and_uvw(&ms).expect("fields").ephemerides;
536        let main = ms.main_table();
537        assert!(main.row_count() > 64, "several chunks of 7 and 64");
538        for scheme in [
539            PartitionScheme::default(),
540            PartitionScheme {
541                fields: true,
542                scans: true,
543            },
544        ] {
545            let build = |rows_per_chunk| {
546                Index::build(main, scheme, &modes, &ephemerides, rows_per_chunk).expect("index")
547            };
548            let whole = build(usize::MAX);
549            assert!(!whole.partitions.is_empty() && !whole.scans.is_empty());
550            for rows_per_chunk in [1, 7, 64] {
551                assert_eq!(
552                    build(rows_per_chunk),
553                    whole,
554                    "{scheme:?}, {rows_per_chunk} rows"
555                );
556            }
557        }
558    }
