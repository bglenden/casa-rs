3//! Thin PyO3 `ia.summary` wrapper delegating to [`casa_imageanalysis::tool::summary`].
4
5use casa_imageanalysis::tool::summary::{SummaryOptions, summarize_image};
6use pyo3::prelude::*;
7
8use crate::convert::value::CasaValue;
9
10use super::PyImageTool;
11use super::handle::{emit_tool_logs, raised};
12
13fn dimension_list(values: &[usize]) -> CasaValue {
14    CasaValue::Ints {
15        shape: vec![values.len()],
16        values: values.iter().map(|&value| value as i64).collect(),
17    }
18}
19
20fn text_list(values: Vec<String>) -> CasaValue {
21    if values.iter().all(String::is_empty) {
22        CasaValue::TextArray {
23            shape: vec![values.len()],
24            values,
25        }
26    } else {
27        CasaValue::Texts(values)
28    }
29}
30
31#[pymethods]
32impl PyImageTool {
33    #[pyo3(signature = (doppler="RADIO", list=true, pixelorder=true, verbose=false), text_signature = "($self, doppler='RADIO', list=True, pixelorder=True, verbose=False)")]
34    fn summary(
35        &self,
36        py: Python<'_>,
37        doppler: &str,
38        list: bool,
39        pixelorder: bool,
40        verbose: bool,
41    ) -> PyResult<Py<PyAny>> {
42        let (value, logs) = py.detach(|| {
43            let state = self.state();
44            let Some(attachment) = &state.image else {
45                return Ok((CasaValue::empty_record(), Vec::new()));
46            };
47            let options = SummaryOptions {
48                doppler: doppler.to_owned(),
49                list,
50                pixel_order: pixelorder,
51                velocity: false,
52                optical: false,
53                verbose,
54            };
55            let (mut report, logs) = summarize_image(attachment, &options).map_err(raised)?;
56            if list {
57                report.messages.clear();
58            }
59            let mut fields = vec![
60                ("refpix", CasaValue::floats(report.refpix)),
61                ("axisnames", text_list(report.axisnames)),
62                ("axisunits", text_list(report.axisunits)),
63                ("refval", CasaValue::floats(report.refval)),
64                ("incr", CasaValue::floats(report.incr)),
65                ("messages", text_list(report.messages)),
66                ("ndim", CasaValue::Int(report.ndim as i64)),
67                ("shape", dimension_list(&report.shape)),
68                ("tileshape", dimension_list(&report.tileshape)),
69                ("unit", CasaValue::Text(report.unit)),
70                ("hasmask", CasaValue::Bool(report.hasmask)),
71                ("defaultmask", CasaValue::Text(report.defaultmask)),
72                ("masks", text_list(report.masks)),
73                ("imagetype", CasaValue::Text(report.imagetype)),
74            ];
75            if let Some(beam) = &report.restoringbeam {
76                fields.push(("restoringbeam", CasaValue::from_record(beam)));
77            }
78            if let Some(beams) = &report.perplanebeams {
79                fields.push(("perplanebeams", CasaValue::from_record(beams)));
80            }
81            Ok::<_, PyErr>((CasaValue::record(fields), logs))
82        })?;
83        emit_tool_logs(py, &logs)?;
84        value.into_python(py)
85    }
86}
