17/// Options for importing a casacore ASCII file.
18#[derive(Debug, Clone)]
19pub struct AsciiInput<'a> {
20    /// A separate header file, or the data file itself if omitted.
21    pub header: Option<&'a Path>,
22    /// Infer column types and names from the first row.
23    pub auto_header: bool,
24    /// Field separator (exactly one ASCII byte).
25    pub separator: &'a str,
26    /// Ignore lines matched at the beginning by this regular expression.
27    pub comment_marker: &'a str,
28    /// First physical data line, numbered from one; zero means the beginning.
29    pub first_line: i64,
30    /// Last physical data line, inclusive; nonpositive means through EOF.
31    pub last_line: i64,
32    /// Names supplied in lieu of a header.
33    pub column_names: &'a [String],
34    /// Types supplied with the column names.
35    pub data_types: &'a [String],
36}
37
38type ReadResult<T> = Result<T, AsciiError>;
39
40fn format_error(message: impl Into<String>) -> AsciiError {
41    AsciiError::Format(format!("ReadAsciiTable: {}", message.into()))
42}
43
44fn lines<'a>(
45    path: &'a Path,
46    first: i64,
47    last: i64,
48    comment: Option<&'a Regex>,
49) -> ReadResult<impl Iterator<Item = ReadResult<String>> + 'a> {
50    let io_error = |source| AsciiError::Io {
51        path: path.into(),
52        source,
53    };
54    let first = usize::try_from(first.max(1) - 1).unwrap_or(usize::MAX);
55    let end = if last <= 0 {
56        usize::MAX
57    } else {
58        usize::try_from(last).unwrap_or(usize::MAX)
59    };
60    let file = BufReader::new(File::open(path).map_err(io_error)?);
61    Ok(file
62        .lines()
63        .take(end)
64        .skip(first)
65        .filter_map(move |line| match line {
66            Ok(line)
67                if comment
68                    .and_then(|re| re.find(&line))
69                    .is_some_and(|m| m.start() == 0) =>
70            {
71                None
72            }
73            other => Some(other.map_err(io_error)),
74        }))
75}

77struct Fields<'a> {
78    input: &'a str,
79    separator: char,
80    pending_empty: bool,
81}
82
83impl<'a> Fields<'a> {
84    fn new(input: &'a str, separator: char) -> Self {
85        Self {
86            input,
87            separator,
88            pending_empty: false,
89        }
90    }
91}
92
93impl Iterator for Fields<'_> {
94    type Item = String;
95
96    fn next(&mut self) -> Option<Self::Item> {
97        let (mut found, mut quoted) = (false, false);
98        let mut field = String::new();
99        for (offset, ch) in self.input.char_indices() {
100            match ch {
101                '"' => {
102                    quoted = !quoted;
103                    found = true;
104                }
105                _ if quoted => field.push(ch),
106                _ => {
107                    let ch = if ch == '\t' { ' ' } else { ch };
108                    if ch == self.separator && (found || self.separator != ' ') {
109                        self.input = &self.input[offset + ch.len_utf8()..];
110                        self.pending_empty = ch != ' ';
111                        return Some(field);
112                    }
113                    if ch != ' ' {
114                        field.push(ch);
115                        found = true;
116                    }
117                }
118            }
119        }
120        self.input = "";
121        (found | std::mem::take(&mut self.pending_empty)).then_some(field)
122    }
123}

126enum Encoding {

128    Polar,

133struct DataType {
134    primitive: PrimitiveType,
135    encoding: Encoding,
136    shape: Vec<usize>,
137    variable_axis: Option<usize>,
138}
139
140impl DataType {
141    fn parse(spec: &str) -> ReadResult<Self> {
142        use Encoding::*;
143        use PrimitiveType::*;
144        let spec = spec.to_ascii_uppercase();
145        let end_type = spec
146            .find(|ch: char| ch.is_ascii_digit())
147            .unwrap_or(spec.len());
148        if spec[..end_type].contains('[') {
149            return Err(format_error(format!(
150                "invalid bracketed type string '{spec}'"
151            )));
152        }
153        let (name, dimensions) = spec.split_at(end_type);
154        let (primitive, encoding) = match name {
155            "B" => (Bool, Plain),
156            "S" => (Int16, Plain),
157            "I" => (Int32, Plain),
158            "R" => (Float32, Plain),
159            "D" => (Float64, Plain),
160            "DMS" => (Float64, Dms),
161            "HMS" => (Float64, Hms),
162            "A" => (String, Plain),
163            "X" => (Complex32, Plain),
164            "Z" => (Complex32, Polar),
165            "DX" => (Complex64, Plain),
166            "DZ" => (Complex64, Polar),
167            _ => return Err(format_error(format!("invalid type specifier '{name}'"))),
168        };
169        let mut shape = Vec::new();
170        let mut variable_axis = None;
171        if !dimensions.is_empty() {
172            for dim in dimensions.split(',') {
173                let extent = dim.parse::<usize>().map_err(|_| {
174                    format_error(format!(
175                        "invalid shape value '{dim}' in type string '{spec}'"
176                    ))
177                })?;
178                if extent == 0 && variable_axis.replace(shape.len()).is_some() {
179                    return Err(format_error(format!(
180                        "multiple variable axes in type string '{spec}'"
181                    )));
182                }
183                shape.push(extent.max(1));
184            }
185        }
186        Ok(Self {
187            primitive,
188            encoding,
189            shape,
190            variable_axis,
191        })
192    }
193
194    fn column(&self, name: &str) -> ReadResult<ColumnSchema> {
195        Ok(if self.shape.is_empty() {
196            ColumnSchema::scalar(name, self.primitive)
197        } else if self.variable_axis.is_some() {
198            ColumnSchema::array_variable(name, self.primitive, None)
199        } else {
200            ColumnSchema::array_fixed(name, self.primitive, self.shape.clone()).with_options(
201                ColumnOptions {
202                    direct: true,
203                    undefined: false,
204                },
205            )?
206        })
207    }
208
209    fn value(&self, fields: &mut Fields<'_>, keyword: bool) -> ReadResult<Value> {
210        if self.shape.is_empty() && !keyword {
211            return Ok(Value::Scalar(
212                self.scalar(fields.next().unwrap_or_default(), fields),
213            ));
214        }
215        let mut shape = if self.shape.is_empty() {
216            vec![1]
217        } else {
218            self.shape.clone()
219        };
220        let axis = if self.shape.is_empty() {
221            Some(0)
222        } else {
223            self.variable_axis
224        };
225        let size = shape
226            .iter()
227            .try_fold(1usize, |n, d| n.checked_mul(*d))
228            .ok_or_else(|| format_error("array shape is too large"))?;
229        let mut values = Vec::new();
230        while axis.is_some() || values.len() < size {
231            let Some(field) = fields.next() else { break };
232            values.push(self.scalar(field, fields));
233        }
234        if let Some(axis) = axis {
235            shape[axis] = values.len().div_ceil(size);
236        }
237        if keyword && self.shape.is_empty() && values.len() == 1 {
238            return Ok(Value::Scalar(values.pop().expect("one value")));
239        }
240        let padded = size
241            .checked_mul(axis.map_or(1, |i| shape[i]))
242            .ok_or_else(|| format_error("array shape is too large"))?;
243        values
244            .try_reserve(padded.saturating_sub(values.len()))
245            .map_err(|_| format_error("array shape is too large"))?;
246        values.resize_with(padded, || {
247            self.scalar(String::new(), &mut Fields::new("", ' '))
248        });
249        Ok(Value::Array(make_array(self.primitive, &shape, values)?))
250    }
251
252    fn scalar(&self, text: String, fields: &mut Fields<'_>) -> ScalarValue {
253        use ScalarValue as S;
254        match self.primitive {
255            PrimitiveType::Bool => {
256                S::Bool(!text.is_empty() && text != "0" && !text.starts_with(['F', 'f', 'N', 'n']))
257            }
258            PrimitiveType::Int16 => {
259                S::Int16(parse_int(&text).clamp(i16::MIN.into(), i16::MAX.into()) as i16)
260            }
261            PrimitiveType::Int32 => {
262                S::Int32(parse_int(&text).clamp(i32::MIN.into(), i32::MAX.into()) as i32)
263            }
264            PrimitiveType::Float32 => S::Float32(parse_ascii(&text)),
265            PrimitiveType::Float64 => {
266                S::Float64(if matches!(self.encoding, Encoding::Dms | Encoding::Hms) {
267                    angle(&text, self.encoding == Encoding::Dms)
268                } else {
269                    parse_ascii(&text)
270                })
271            }
272            PrimitiveType::String => S::String(text),
273            PrimitiveType::Complex32 => {
274                let first: f32 = parse_ascii(&text);
275                let second: f32 = parse_ascii(&fields.next().unwrap_or_default());
276                S::Complex32(if self.encoding == Encoding::Polar {
277                    let radians = (f64::from(second) * CASA_DEGREE) as f32;
278                    Complex32::from_polar(first, radians)
279                } else {
280                    Complex32::new(first, second)
281                })
282            }
283            PrimitiveType::Complex64 => {
284                let first: f64 = parse_ascii(&text);
285                let second: f64 = parse_ascii(&fields.next().unwrap_or_default());
286                S::Complex64(if self.encoding == Encoding::Polar {
287                    Complex64::from_polar(first, second * CASA_DEGREE)
288                } else {
289                    Complex64::new(first, second)
290                })
291            }
292            _ => unreachable!("primitive not used by ASCII table columns"),
293        }
294    }
295}

297// ReadAsciiTable uses this decimal approximation for a polar complex value.
298const CASA_DEGREE: f64 = 314_159_265.0 / 100_000_000.0 / 180.0;

300fn parse_ascii<T: std::str::FromStr + Default>(text: &str) -> T {
301    if text
302        .as_bytes()
303        .iter()
304        .all(|b| matches!(b, b'0'..=b'9' | b'+' | b'-' | b'.' | b'e' | b'E'))
305    {
306        text.parse().unwrap_or_default()
307    } else {
308        T::default()
309    }
310}

312fn parse_int(text: &str) -> i64 {
313    let (sign, digits) = match text.as_bytes().first() {
314        Some(b'-') => (-1_i64, &text.as_bytes()[1..]),
315        Some(b'+') => (1_i64, &text.as_bytes()[1..]),
316        _ => (1_i64, text.as_bytes()),
317    };
318    digits
319        .iter()
320        .copied()
321        .take_while(u8::is_ascii_digit)
322        .fold(0_i64, |acc, b| {
323            acc.saturating_mul(10)
324                .saturating_add(sign * i64::from(b - b'0'))
325        })
326}

328fn angle(text: &str, dms: bool) -> f64 {
329    // ReadAsciiTable::stringToPos tolerates whitespace around units and
330    // accepts spaces between unmarked sexagesimal components.
331    let mut input = String::with_capacity(text.len());
332    let mut blank = false;
333    let mut needs_value = true;
334    for ch in text.trim().chars().map(|ch| ch.to_ascii_lowercase()) {
335        if ch.is_ascii_whitespace() {
336            blank = true;
337            continue;
338        }
339        if matches!(ch, ':' | '.' | 'h' | 'd' | 'm') {
340            if needs_value {
341                input.push('0');
342            }
343            needs_value = true;
344        } else if !matches!(ch, '+' | '-') {
345            if !needs_value && blank {
346                input.push(':');
347            }
348            needs_value = false;
349            blank = false;
350        }
351        input.push(ch);
352    }
353    let radians = input.parse::<f64>().unwrap_or_else(|_| {
354        Quantity::read(&input)
355            .ok()
356            .and_then(|q| mvangle_radians(&q).ok())
357            .unwrap_or(0.0)
358    });
359    if dms && input.contains(':') {
360        radians / 15.0
361    } else {
362        radians
363    }
364}

366fn make_array(
367    primitive: PrimitiveType,
368    shape: &[usize],
369    values: Vec<ScalarValue>,
370) -> ReadResult<ArrayValue> {
371    macro_rules! build {
372        ($variant:ident) => {{
373            let elements = values
374                .into_iter()
375                .map(|value| match value {
376                    ScalarValue::$variant(v) => v,
377                    _ => unreachable!("ASCII column changes type"),
378                })
379                .collect::<Vec<_>>();
380            ArrayD::from_shape_vec(IxDyn(shape).f(), elements).map(ArrayValue::$variant)
381        }};
382    }
383    match primitive {
384        PrimitiveType::Bool => build!(Bool),
385        PrimitiveType::Int16 => build!(Int16),
386        PrimitiveType::Int32 => build!(Int32),
387        PrimitiveType::Float32 => build!(Float32),
388        PrimitiveType::Float64 => build!(Float64),
389        PrimitiveType::Complex32 => build!(Complex32),
390        PrimitiveType::Complex64 => build!(Complex64),
391        PrimitiveType::String => build!(String),
392        _ => unreachable!("primitive not used by ASCII table columns"),
393    }
394    .map_err(|error| format_error(error.to_string()))
395}

397fn header_fields(line: &str, separator: char) -> Vec<String> {
398    let separator = if line.contains(separator) {
399        separator
400    } else {
401        ' '
402    };
403    Fields::new(line, separator).collect()
404}

406fn infer(line: &str, separator: char) -> (Vec<String>, Vec<String>) {
407    Fields::new(line, separator)
408        .enumerate()
409        .map(|(i, field)| {
410            let kind = if field.parse::<i64>().is_ok() {
411                "I"
412            } else if field.parse::<f64>().is_ok_and(f64::is_finite) {
413                "D"
414            } else {
415                "A"
416            };
417            (format!("Column{}", i + 1), kind.into())
418        })
419        .unzip()
420}

422fn keyword_sets(
423    mut line: String,
424    lines: &mut impl Iterator<Item = ReadResult<String>>,
425    separator: char,
426) -> ReadResult<(String, BTreeMap<String, RecordValue>)> {
427    let mut sets = BTreeMap::new();
428    while line.starts_with(".key") {
429        let column = Fields::new(&line, ' ').nth(1).unwrap_or_default();
430        let mut record = RecordValue::default();
431        loop {
432            let line = lines
433                .next()
434                .transpose()?
435                .ok_or_else(|| format_error("no .endkey line"))?;
436            if line.starts_with(".endkey") {
437                break;
438            }
439            let mut fields = Fields::new(&line, ' ');
440            let name = fields
441                .next()
442                .ok_or_else(|| format_error("no keyword name"))?;
443            let kind = fields
444                .next()
445                .ok_or_else(|| format_error("no keyword type"))?;
446            if record.get(&name).is_none() {
447                fields.separator = separator;
448                let value = DataType::parse(&kind)?.value(&mut fields, true)?;
449                record.push(RecordField::new(name, value));
450            }
451        }
452        sets.entry(column).or_insert(record);
453        line = lines.next().transpose()?.unwrap_or_default();
454    }
455    Ok((line, sets))
456}

458fn populate(
459    mut table: Table,
460    names: &[String],
461    types: &[DataType],
462    mut keys: BTreeMap<String, RecordValue>,
463    data: impl Iterator<Item = ReadResult<String>>,
464    separator: char,
465) -> ReadResult<Table> {
466    if let Some(record) = keys.remove("") {
467        *table.keywords_mut() = record;
468    }
469    for (name, kind) in names.iter().zip(types) {
470        let mut record = keys.remove(name).unwrap_or_default();
471        if !kind.shape.is_empty() && matches!(kind.encoding, Encoding::Dms | Encoding::Hms) {
472            record.upsert(
473                "QuantumUnits",
474                Value::Array(ArrayValue::from_string_vec(vec!["rad".into()])),
475            );
476        }
477        if !record.fields().is_empty() {
478            table.set_column_keywords(name, record);
479        }
480    }
481    for line in data {
482        let line = line?;
483        let mut fields = Fields::new(&line, separator);
484        let row = names
485            .iter()
486            .zip(types)
487            .map(|(name, kind)| Ok(RecordField::new(name, kind.value(&mut fields, false)?)))
488            .collect::<ReadResult<Vec<_>>>()?;
489        table.add_row(RecordValue::new(row))?;
490    }
491    Ok(table)
492}

494/// Parse an ASCII table into a native table with typed rows and keyword sets.
495///
496/// Reads data in physical line order. A variable array axis is accepted in the
497/// final column and is sized from the values left on each line. The caller
498/// chooses the output location and storage manager when saving the result.
499pub fn read_ascii(path: &Path, options: &AsciiInput<'_>) -> ReadResult<Table> {
500    let mut chars = options.separator.chars();
501    let separator = chars
502        .next()
503        .filter(|c| c.is_ascii() && chars.next().is_none())
504        .ok_or_else(|| AsciiError::Format("tablefromascii : separator must be 1 char".into()))?;
505    let comment = (!options.comment_marker.is_empty())
506        .then(|| Regex::new(options.comment_marker))
507        .transpose()
508        .map_err(|error| format_error(error.to_string()))?;
509    let header = options.header.unwrap_or(path);
510    let separate = header != path;
511    let explicit_names = options.column_names;
512    let explicit_types = options.data_types;
513    let explicit =
514        options.header.is_none() && !(explicit_names.is_empty() && explicit_types.is_empty());
515    let infer_types = options.header.is_none() && !explicit && options.auto_header;
516    let (start, end) = if separate {
517        (1, -1)
518    } else {
519        (options.first_line, options.last_line)
520    };
521    let mut header_lines = lines(header, start, end, comment.as_ref())?;
522    let first = header_lines.next().transpose()?.ok_or_else(|| {
523        format_error(format!(
524            "cannot read first header line of {}",
525            header.display()
526        ))
527    })?;
528    let (first, keys) = keyword_sets(first, &mut header_lines, separator)?;
529    let (names, specs, first_data) = if explicit {
530        (explicit_names.to_vec(), explicit_types.to_vec(), true)
531    } else if infer_types {
532        let (names, types) = infer(&first, separator);
533        (names, types, true)
534    } else {
535        if first.is_empty() {
536            return Err(format_error(format!(
537                "no COLUMN NAMES line in {}",
538                header.display()
539            )));
540        }
541        let second = header_lines
542            .next()
543            .transpose()?
544            .ok_or_else(|| format_error(format!("no COLUMN TYPES line in {}", header.display())))?;
545        (
546            header_fields(&first, separator),
547            header_fields(&second, separator),
548            false,
549        )
550    };
551    if names.len() != specs.len() || names.iter().chain(&specs).any(String::is_empty) {
552        return Err(if explicit {
553            format_error("vector of columnNames and dataTypes should have equal length")
554        } else {
555            format_error(format!(
556                "mismatching COLUMN NAMES and TYPES lines in {}",
557                header.display()
558            ))
559        });
560    }
561    let types: Vec<_> = specs
562        .iter()
563        .map(|spec| DataType::parse(spec))
564        .collect::<ReadResult<_>>()?;
565    if types
566        .iter()
567        .rev()
568        .skip(1)
569        .any(|kind| kind.variable_axis.is_some())
570    {
571        return Err(format_error(
572            "only last column can have variable shaped arrays",
573        ));
574    }
575    let columns = names
576        .iter()
577        .zip(&types)
578        .map(|(name, kind)| kind.column(name))
579        .collect::<ReadResult<_>>()?;
580    let table = Table::with_schema(TableSchema::new(columns)?);
581    if separate {
582        populate(
583            table,
584            &names,
585            &types,
586            keys,
587            lines(
588                path,
589                options.first_line,
590                options.last_line,
591                comment.as_ref(),
592            )?,
593            separator,
594        )
595    } else {
596        let saved = (first_data && !first.is_empty()).then_some(Ok(first));
597        populate(
598            table,
599            &names,
600            &types,
601            keys,
602            saved.into_iter().chain(header_lines),
603            separator,
604        )
605    }
606}
