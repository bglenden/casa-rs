40/// Why a native run did not complete.
41#[derive(Debug, thiserror::Error)]
42pub(crate) enum NativeError {
43    /// A phase's memory did not fit the resource policy.
44    #[error(transparent)]
45    Admission(#[from] Admission),
46    /// The run was cancelled.
47    #[error(transparent)]
48    Cancelled(#[from] Cancelled),
49    /// Any other failure.
50    #[error("{0}")]
51    Other(ApplicationError),
52}
53
54impl From<ApplicationError> for NativeError {
55    fn from(error: ApplicationError) -> Self {
56        Self::Other(error)
57    }
58}
59
60impl From<ImagingError> for NativeError {
61    fn from(error: ImagingError) -> Self {
62        match error {
63            ImagingError::Admission(admission) => Self::Admission(admission),
64            ImagingError::Cancelled(cancelled) => Self::Cancelled(cancelled),
65            ImagingError::Pass(error) => error.into(),
66            error => Self::Other(Box::new(error)),
67        }
68    }
69}
70
71impl From<PassError> for NativeError {
72    fn from(error: PassError) -> Self {
73        match error {
74            PassError::Cancelled => Self::Cancelled(Cancelled),
75            error => Self::Other(Box::new(error)),
76        }
77    }
78}

101/// Run the major-cycle passes, the minor cycles between them and any
102/// visibility write, then publish the products.
103///
104/// When the runtime names a summary target, the run summary is written
105/// there whether the run completes or fails; a cancelled run writes
106/// nothing.
107pub(crate) fn run_native(
108    problem: &CompiledProblem,
109    input: NativeInput,
110    native: Result<ApplicationNative, ApplicationError>,
111) -> Result<NativeApplicationOutcome, NativeError> {
112    let native = native?;
113    let target = native.runtime.summary.clone();
114    let mut summary = RunSummary {
115        backend: native.runtime.backend,
116        ..RunSummary::default()
117    };
118    if let Some(target) = &target {
119        summary.request = target.request.clone();
120    }
121    let result = run(problem, input, native, &mut summary);
122    let Some(target) = target else {
123        return result;
124    };
125    match result {
126        Ok(outcome) => {
127            outcome.summary.write(&target.path)?;
128            Ok(outcome)
129        }
130        Err(NativeError::Cancelled(cancelled)) => Err(NativeError::Cancelled(cancelled)),
131        Err(error) => {
132            summary.error = Some(error.to_string());
133            if let Err(write) = summary.write(&target.path) {
134                tracing::warn!("the failed run's summary could not be written: {write}");
135            }
136            Err(error)
137        }
138    }
139}

141/// The run itself; `summary` collects its phases, and a completed run's
142/// outcome takes it.
143fn run(
144    problem: &CompiledProblem,
145    input: NativeInput,
146    native: ApplicationNative,
147    summary: &mut RunSummary,
148) -> Result<NativeApplicationOutcome, NativeError> {
149    let ApplicationNative {
150        runtime,
151        publication,
152        aw_catalog,
153    } = native;
154    publication.controls.validate_for_problem(problem)?;
155    let visibility_write = (input.write_model_column || input.write_corrected_data)
156        .then(|| {
157            Ok::<_, ApplicationError>(imaging::VisibilityWriteTarget {
158                path: PathBuf::from(input.observation.locator()),
159                expected: input.initial_access.source_state().clone(),
160                selection: visibility_write_selection(problem, input.observation.selection())?,
161                model_data: input.write_model_column,
162                corrected_data: input.write_corrected_data,
163            })
164        })
165        .transpose()?;
166    let (access, source) = finalize_source_access(
167        problem,
168        input.initial_access,
169        &runtime.host,
170        &runtime.resource_policy,
171    )?;
172    let team = WorkerTeam::new(runtime.resource_policy.workers(&runtime.host))?;
173    summary.workers = team.workers();
174    let outcome = imaging::run(
175        imaging::ImagingInputs {
176            problem,
177            access,
178            masks: input.masks,
179            image_response: input.minor_cycle_image_response,
180            visibility_write,
181            host: runtime.host,
182            policy: runtime.resource_policy,
183            team: &team,
184            cancel: runtime.cancel.clone(),
185            spill_directory: &runtime.spill_directory,
186            aw_catalog,
187            backend: runtime.backend,
188        },
189        summary,
190    )?;
191    drop(source);
192    summary.minor_cycles = outcome.minor_cycles.len();
193    summary.minor_iterations = outcome.total_minor_iterations;
194    let products = run_phase("products", &runtime.cancel, summary, || {
195        publish_products(
196            problem,
197            outcome.scientific,
198            outcome.masks,
199            &runtime,
200            publication,
201            &team,
202        )
203    })?;
204    summary.products = products
205        .planned
206        .members()
207        .iter()
208        .map(|member| member.name().to_string())
209        .collect();
210    Ok(NativeApplicationOutcome {
211        problem: problem.clone(),
212        minor_cycles: outcome.minor_cycles,
213        stop: outcome.stop,
214        major_cycle_count: outcome.major_cycle_count,
215        total_minor_iterations: outcome.total_minor_iterations,
216        total_actual_minor_iterations: outcome.total_actual_minor_iterations,
217        workers: team.workers(),
218        planes_per_wave: outcome.planes_per_wave,
219        visibility_products: outcome.visibility_products,
220        summary: std::mem::take(summary),
221        scientific: products.scientific,
222        planned_products: products.planned,
223        products: products.published,
224    })
225}

227/// The products of a run and the state they were generated from.
228struct Products {
229    planned: PlannedContinuumGeneration,
230    scientific: MajorCycleCompletion,
231    published: PublishedContinuumGeneration,
232}
233
234/// Generate every planned product member into the sink's private staging,
235/// one output channel per window on the run's team, and publish them; a
236/// cancelled run publishes nothing.
237fn publish_products(
238    problem: &CompiledProblem,
239    scientific: MajorCycleCompletion,
240    reconstruction_masks: Option<ReconstructionMaskSet>,
241    runtime: &ApplicationRuntime,
242    publication: ApplicationPublication,
243    team: &WorkerTeam,
244) -> Result<Products, NativeError> {
245    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific)?;
246    if let Some(masks) = reconstruction_masks.as_ref() {
247        inputs = match masks {
248            ReconstructionMaskSet::Shared(mask) => inputs.with_reconstruction_mask(mask)?,
249            ReconstructionMaskSet::Domains(masks) => {
250                inputs.with_domain_reconstruction_masks(masks)?
251            }
252        };
253    }
254    let planned = PlannedContinuumGeneration::new(&inputs, &publication.controls)?;
255    let window = ProductStoragePlan::new(1, team.workers())?;
256    let demand = planned.demand(&inputs, window)?;
257    let held = admit(
258        &runtime.host,
259        &runtime.resource_policy,
260        &Demand {
261            phase: "products",
262            memory: demand.peak_residency_bytes()
263                + publication.sink.residency(&planned, &demand)?,
264        },
265    )?;
266    let published = produce_continuum_members(&planned, &inputs, window, team, &publication.sink)?;
267    drop(held);
268    if runtime.cancel.is_cancelled() {
269        // The staged members are removed when the sink drops.
270        return Err(Cancelled.into());
271    }
272    publication.sink.publish()?;
273    Ok(Products {
274        planned,
275        scientific,
276        published,
277    })
