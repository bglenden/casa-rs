2//! An owner's weight image, summed by contribution before it is gridded
3//! (#787).
4//!
5//! `Mode::Weight` places every sample's weight taps at the uv origin
6//! (`CpuBackend`'s gridding, the Metal records), so a sample's contribution
7//! is fixed by its kernel cell, plane, spectral factor, pointing gradient
8//! and the sign of its w, and, for a mosaic sample whose support overruns
9//! the grid, by the taps its support keeps on it ([`Reach`]); it scales
10//! with its weights alone. Samples that agree on those add the same taps
11//! scaled by their weights: summing
12//! their weights in f64 first and gridding each group once gives the same
13//! image, to rounding, for a lookup and an add per sample rather than a
14//! kernel stamp.
15//!
16//! What it saves, in work and in precision, depends on how many groups the
17//! samples share:
18//! - **Fixed pointing.** A mosaic with fixed pointings shares a few groups
19//!   per pointing and plane: B10, the 1024² ALMA mosaic, sums 4.19 M
20//!   placements into 14 (#787). Its weight image then costs a lookup per
21//!   sample. On f32 grids (Metal) it also keeps the grid's
22//!   own precision: millions of adds into the origin's cells had lost about
23//!   1e-3, and a few adds per group do not.
24//! - **A pointing that drifts row by row.** AW projection's pointing does,
25//!   which leaves about two samples a group: B3, the 63-field VLASS clean,
26//!   has 3.36 M groups for 6.42 M placements. The saving is then about
27//!   half, and f32 grids keep most of their per-sample drift.
28//!
29//! The row-work pass counts the groups the same way ([`GroupCount`]), so
30//! the strips charge the weight owner for the stamps it will make.
31

40/// Groups an owner's table holds before it grids them and starts again: a
41/// mosaic's first pass has a few per pointing and plane (14 on B10), so the
42/// table grids once a wave unless the pointing drifts row by row.
43pub(in crate::pass) const WEIGHT_GROUPS: usize = 1 << 14;
44
45/// Correlations a placement carries at most.
46const MAX_POL: usize = 4;
47
48/// What fixes one sample's weight-image contribution apart from its
49/// weights: every input `Mode::Weight` reads but the weights.
50///
51/// Both backends locate the weight taps at the uv origin, so the sample's
52/// `u` and `v` are read only for the taps a clipping set keeps
53/// (`clipped`). Only the sign of `w` is read (the Mueller table and the tap
54/// conjugation), and `phase` and `prediction_w_positive` are read only by
55/// prediction. Floats are kept as their bits, so two samples share a group
56/// only when their contributions are the same bits.
57#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
58struct Contribution {
59    cf: CfKey,
60    plane: u32,
61    spectral: u32,
62    gradient: [u32; 2],
63    w_positive: bool,
64    /// The cells a mosaic sample whose support overruns the grid keeps
65    /// (`Reach::Clipped`).
66    clipped: Option<CellBox>,
67}
68
69impl Contribution {
70    fn of(
71        placement: &Placement,
72        (cf, geometry): (&dyn ConvolutionFunctionSet, &GridGeometry),
73    ) -> Self {
74        Self {
75            cf: placement.cf,
76            plane: placement.plane,
77            spectral: placement.spectral.to_bits(),
78            gradient: placement.gradient.map(f32::to_bits),
79            w_positive: placement.w > 0.0,
80            clipped: weight_cells(placement, cf, geometry),
81        }
82    }
83
84    /// A placement with this contribution: at `uv` (a sample of the group's,
85    /// whose support fixes the taps a clipping set keeps; any for the
86    /// others, which place the taps at the uv origin), with a w of the
87    /// group's sign.
88    fn placement(self, uv: [f64; 2]) -> Placement {
89        Placement {
90            u: uv[0],
91            v: uv[1],
92            w: if self.w_positive { 1.0 } else { -1.0 },
93            prediction_w_positive: self.w_positive,
94            phase: 0.0,
95            plane: self.plane,
96            spectral: f32::from_bits(self.spectral),
97            cf: self.cf,
98            gradient: self.gradient.map(f32::from_bits),
99        }
100    }
101}

103/// The cells `placement`'s weight taps keep under a set that clips
104/// supports to the grid, when its support overruns it; `None` for every
105/// other placement and set.
106fn weight_cells(
107    placement: &Placement,
108    cf: &dyn ConvolutionFunctionSet,
109    geometry: &GridGeometry,
110) -> Option<CellBox> {
111    if cf.grid_edge() == GridEdge::Whole {
112        return None;
113    }
114    let half = cf
115        .weight_half_support(placement.cf)
116        .expect("a set that clips has weight cells");
117    let oversampling = cf.gridding_oversampling(placement.cf);
118    let origin = geometry.locate(0.0, 0.0, oversampling);
119    match Reach::of(
120        cf,
121        geometry,
122        (Mode::Weight, placement),
123        origin,
124        (half, oversampling),
125    ) {
126        Reach::Clipped { cells, .. } => Some(cells),
127        Reach::Whole | Reach::Off => None,
128    }
129}

131/// The groups a [`WeightSums`] of `capacity` groups grids for the
132/// placements it is shown, counted without summing: the row-work pass
133/// charges the weight owner a kernel stamp for each
134/// ([`ModeWork`](crate::pass::ModeWork)). Its set is allocated whole on the
135/// first placement, so a pass without a weight image holds nothing.
136pub(in crate::pass) struct GroupCount {
137    capacity: usize,
138    seen: HashSet<Contribution>,
139    last: Option<Contribution>,
140}
141
142impl GroupCount {
143    /// A counter that forgets every group when `capacity` are held, as a
144    /// full [`WeightSums`] grids and starts again.
145    pub(in crate::pass) fn new(capacity: usize) -> Self {
146        assert!(capacity > 0, "a group count holds at least one group");
147        Self {
148            capacity,
149            seen: HashSet::new(),
150            last: None,
151        }
152    }
153
154    /// Bytes a counter of `capacity` groups holds once it has seen a
155    /// placement: its hash set's buckets, a key and a control byte each.
156    #[must_use]
157    pub(in crate::pass) fn bytes(capacity: usize) -> u64 {
158        let buckets = (capacity * 8).div_ceil(7).next_power_of_two().max(4);
159        (buckets * (size_of::<Contribution>() + 1) + 16) as u64
160    }
161
162    /// Whether `placement` starts a group that [`WeightSums::add`] would
163    /// grid for `cf` on `geometry`: its contribution is not among those
164    /// held since the last time the table filled.
165    pub(in crate::pass) fn starts(
166        &mut self,
167        placement: &Placement,
168        operator: (&dyn ConvolutionFunctionSet, &GridGeometry),
169    ) -> bool {
170        let key = Contribution::of(placement, operator);
171        if self.last == Some(key) {
172            return false;
173        }
174        self.last = Some(key);
175        if self.seen.capacity() == 0 {
176            self.seen.reserve(self.capacity);
177        }
178        if self.seen.contains(&key) {
179            return false;
180        }
181        if self.seen.len() == self.capacity {
182            self.seen.clear();
183        }
184        self.seen.insert(key);
185        true
186    }
187}

189/// One owner's weight-image contributions, summed in f64 by
190/// [`Contribution`] in the order they first appear, and gridded as one
191/// placement per group ([`Self::flush`]) when the table fills and when the
192/// wave ends.
193pub(in crate::pass) struct WeightSums {
194    npol: usize,
195    capacity: usize,
196    index: HashMap<Contribution, usize>,
197    /// Each group's contribution and the uv of its first placement.
198    groups: Vec<(Contribution, [f64; 2])>,
199    /// `npol` weights per group.
200    sums: Vec<f64>,
201    /// The groups as placements, for the backend.
202    buffer: SampleBuffer,
203    /// Placements summed, groups gridded and grids made since the table
204    /// was made, for the wave's log ([`Self::totals`]).
205    totals: [u64; 3],
206}
207
208impl WeightSums {
209    /// An empty table of `capacity` groups of `npol` weights, allocated
210    /// whole ([`Self::bytes`]).
211    pub(in crate::pass) fn new(npol: usize, capacity: usize) -> Self {
212        assert!(
213            (1..=MAX_POL).contains(&npol),
214            "a placement carries one to four correlations"
215        );
216        assert!(capacity > 0, "a weight table holds at least one group");
217        Self {
218            npol,
219            capacity,
220            index: HashMap::with_capacity(capacity),
221            groups: Vec::with_capacity(capacity),
222            sums: Vec::with_capacity(capacity * npol),
223            buffer: SampleBuffer::with_capacity(npol, capacity),
224            totals: [0; 3],
225        }
226    }
227
228    /// Placements summed, groups gridded and grids made so far.
229    pub(in crate::pass) const fn totals(&self) -> [u64; 3] {
230        self.totals
231    }
232
233    /// Bytes a table of `capacity` groups of `npol` weights holds: the
234    /// hash index (its buckets, each a key, an index and a control byte,
235    /// at the standard library's 7/8 load), the groups, their sums and
236    /// the buffer they are gridded from.
237    #[must_use]
238    pub(in crate::pass) fn bytes(npol: usize, capacity: usize) -> u64 {
239        let buckets = (capacity * 8).div_ceil(7).next_power_of_two().max(4);
240        let index = buckets * (size_of::<(Contribution, usize)>() + 1) + 16;
241        (index
242            + capacity * size_of::<(Contribution, [f64; 2])>()
243            + capacity * npol * size_of::<f64>()) as u64
244            + SampleBuffer::bytes(npol, capacity)
245    }
246
247    /// Bytes the table holds now, from its parts' capacities.
248    #[cfg(test)]
249    fn capacity_bytes(&self) -> u64 {
250        let index = self.index.capacity();
251        let buckets = (index * 8).div_ceil(7).next_power_of_two().max(4);
252        (buckets * (size_of::<(Contribution, usize)>() + 1)
253            + 16
254            + self.groups.capacity() * size_of::<(Contribution, [f64; 2])>()
255            + self.sums.capacity() * size_of::<f64>()) as u64
256            + self.buffer.capacity_bytes()
257    }

259    /// Add the weights of every placement of `block` to its group for
260    /// `operator`'s set and grid, gridding the groups through `grid` first
261    /// whenever a new group finds the table full.
262    pub(in crate::pass) fn add<E>(
263        &mut self,
264        block: &SampleBlock<'_>,
265        operator: (&dyn ConvolutionFunctionSet, &GridGeometry),
266        grid: &mut dyn FnMut(&SampleBlock<'_>) -> Result<(), E>,
267    ) -> Result<(), E> {
268        debug_assert_eq!(block.npol, self.npol, "the table's correlations");
269        self.totals[0] += block.len() as u64;
270        // Rows place their channels together, so successive samples share
271        // a group almost always: look the hash up only when it changes.
272        let mut last: Option<(Contribution, usize)> = None;
273        for (index, placement) in block.placements.iter().enumerate() {
274            let key = Contribution::of(placement, operator);
275            let group = match last {
276                Some((held, group)) if held == key => group,
277                _ => {
278                    let group = match self.index.get(&key) {
279                        Some(&group) => group,
280                        None => {
281                            if self.groups.len() == self.capacity {
282                                self.flush(grid)?;
283                            }
284                            let group = self.groups.len();
285                            self.index.insert(key, group);
286                            self.groups.push((key, [placement.u, placement.v]));
287                            self.sums.resize(self.sums.len() + self.npol, 0.0);
288                            group
289                        }
290                    };
291                    last = Some((key, group));
292                    group
293                }
294            };
295            let sums = &mut self.sums[group * self.npol..(group + 1) * self.npol];
296            for (sum, weight) in sums.iter_mut().zip(block.weights_of(index)) {
297                *sum += f64::from(*weight);
298            }
299        }
300        Ok(())
301    }
302
303    /// Grid every group through `grid`, as one placement of its summed
304    /// weights each, and empty the table. Nothing when it is empty.
305    pub(in crate::pass) fn flush<E>(
306        &mut self,
307        grid: &mut dyn FnMut(&SampleBlock<'_>) -> Result<(), E>,
308    ) -> Result<(), E> {
309        if self.groups.is_empty() {
310            return Ok(());
311        }
312        self.buffer.clear();
313        let values = [Complex32::default(); MAX_POL];
314        let mut weights = [0.0_f32; MAX_POL];
315        for (group, (key, uv)) in self.groups.iter().enumerate() {
316            let sums = &self.sums[group * self.npol..(group + 1) * self.npol];
317            for (weight, sum) in weights.iter_mut().zip(sums) {
318                *weight = *sum as f32;
319            }
320            self.buffer.push(
321                key.placement(*uv),
322                &values[..self.npol],
323                &weights[..self.npol],
324            );
325        }
326        self.totals[1] += self.groups.len() as u64;
327        self.totals[2] += 1;
328        self.index.clear();
329        self.groups.clear();
330        self.sums.clear();
331        grid(&self.buffer.block())
332    }
333}

353    /// The groups a table hands to `grid`, with their summed weights.
354    fn flushed(sums: &mut WeightSums) -> Vec<(Placement, Vec<f32>)> {
355        let mut out = Vec::new();
356        sums.flush(&mut |block: &SampleBlock<'_>| {
357            for (index, placement) in block.placements.iter().enumerate() {
358                out.push((*placement, block.weights_of(index).to_vec()));
359            }
360            Ok::<_, ()>(())
361        })
362        .expect("flush");
363        out
364    }

366    /// Samples that differ in any input of `Mode::Weight` but their weights
367    /// keep apart, and the rest merge with their weights summed (#787).
368    #[test]
369    fn only_samples_with_one_contribution_share_a_group() {
370        let base = placement(1, 0, 0.25, [0.5, -0.5], 10.0);
371        let variants = [
372            base,
373            placement(2, 0, 0.25, [0.5, -0.5], 10.0),
374            placement(1, 1, 0.25, [0.5, -0.5], 10.0),
375            placement(1, 0, 0.5, [0.5, -0.5], 10.0),
376            placement(1, 0, 0.25, [0.25, -0.5], 10.0),
377            placement(1, 0, 0.25, [0.5, -0.25], 10.0),
378            placement(1, 0, 0.25, [0.5, -0.5], -10.0),
379        ];
380        // The base again at another uv, w magnitude and phase, which
381        // `Mode::Weight` does not read.
382        let same = Placement {
383            u: -9.0,
384            v: 8.0,
385            w: 3.0,
386            phase: -1.0,
387            ..base
388        };
389        let mut buffer = SampleBuffer::with_capacity(2, 16);
390        for placement in variants.iter().chain([&same]) {
391            buffer.push(*placement, &[Complex32::default(); 2], &[1.5, 2.5]);
392        }
393        let operator = spheroidal_operator();
394        let set = (operator.cf(), operator.geometry());
395        let mut sums = WeightSums::new(2, 16);
396        sums.add(&buffer.block(), set, &mut |_: &SampleBlock<'_>| {
397            Ok::<_, ()>(())
398        })
399        .expect("add");
400        let groups = flushed(&mut sums);
401        assert_eq!(groups.len(), variants.len());
402        assert_eq!(groups[0].1, [3.0, 5.0], "the base merged with its twin");
403        for (group, variant) in groups.iter().zip(&variants).skip(1) {
404            assert_eq!(group.1, [1.5, 2.5]);
405            assert_eq!(
406                Contribution::of(&group.0, set),
407                Contribution::of(variant, set)
408            );
409        }
410        assert!(flushed(&mut sums).is_empty(), "a flush empties the table");
411    }

532    /// `samples` weight placements over three pointings, three spectral
533    /// factors and both signs of w, with random weights, in random order.
534    fn weight_samples(
535        operator: &casa_imaging_operator::MeasurementOperator,
536        samples: usize,
537    ) -> SampleBuffer {
538        let gradients = [[0.0, 0.0], [0.6, -0.2], [-0.3, 0.9]].map(|offset: [f64; 2]| {
539            operator
540                .geometry()
541                .pointing_gradient([offset[0] * 1.0e-4, offset[1] * 1.0e-4])
542        });
543        let spectral = [-0.01_f32, 0.0, 0.02];
544        let mut state = 0x9e37_79b9_7f4a_7c15_u64;
545        let mut next = || {
546            state ^= state << 13;
547            state ^= state >> 7;
548            state ^= state << 17;
549            state
550        };
551        let mut buffer = SampleBuffer::with_capacity(2, samples);
552        for _ in 0..samples {
553            let draw = next();
554            let weight = |shift: u32| 0.5 + ((draw >> shift) & 0xffff) as f32 / 65536.0;
555            buffer.push(
556                Placement {
557                    u: 0.0,
558                    v: 0.0,
559                    w: if draw & 1 == 0 { 25.0 } else { -25.0 },
560                    prediction_w_positive: true,
561                    phase: 0.0,
562                    plane: 0,
563                    spectral: spectral[(draw >> 1) as usize % 3],
564                    cf: CfKey::default(),
565                    gradient: gradients[(draw >> 3) as usize % 3],
566                },
567                &[Complex32::default(); 2],
568                &[weight(16), weight(32)],
569            );
570        }
571        buffer
572    }

584    /// [`weight_image`] and the finished images' `sumwt`.
585    fn weight_image_and_sums(
586        operator: &casa_imaging_operator::MeasurementOperator,
587        samples: &SampleBuffer,
588        per_sample: bool,
589    ) -> (Vec<f32>, Vec<f64>) {
590        use casa_imaging_operator::{CpuBackend, GridBackend, Mode, ModeSet, PlaneRange, Work};
591        let mut acc = operator.accumulator(PlaneRange::single(0), None, ModeSet::ALL);
592        let mut backend = CpuBackend::new();
593        let cf = operator.cf();
594        let mut grid = |block: &SampleBlock<'_>| {
595            backend.apply(
596                block,
597                cf,
598                Work::Grid {
599                    mode: Mode::Weight,
600                    acc: &mut acc,
601                },
602            )
603        };
604        if per_sample {
605            grid(&samples.block()).expect("grid");
606        } else {
607            let mut sums = WeightSums::new(2, WEIGHT_GROUPS);
608            sums.add(&samples.block(), (cf, operator.geometry()), &mut grid)
609                .expect("add");
610            sums.flush(&mut grid).expect("flush");
611        }
612        let images = operator.finish(acc).expect("finish");
613        let plane = &images.planes[0];
614        (
615            plane
616                .weight
617                .iter()
618                .flat_map(|image| image.iter().copied())
619                .collect(),

624    /// The largest difference between `a` and `b`, relative to `b`'s peak.
625    fn relative(a: &[f32], b: &[f32]) -> f64 {
626        let peak = b
627            .iter()
628            .fold(0.0_f64, |peak, v| peak.max(f64::from(v.abs())));
629        a.iter()
630            .zip(b)
631            .map(|(a, b)| f64::from((a - b).abs()))
632            .fold(0.0, f64::max)
633            / peak
634    }

637    /// set with pointing ramps and Taylor terms, what gridding each sample
638    /// grids, to rounding in f64 grids (#787). On f32 grids it stays at the
639    /// f32 image's precision while the per-sample f32 weight image drifts
640    /// with the sample count.

642    fn summed_weights_grid_what_each_sample_grids() {

644        let f64_operator = mosaic_operator(GridPrecision::F64);

667        assert!(summed_f32 <= 1.0e-6, "summed on f32 grids: {summed_f32}");

724    /// A kernel set routed as AW routes four linear correlations (#810, for
725    /// #653's AW rows). `AWVisResampler` grids correlation `e` (XX, XY, YX
726    /// and YY are AW's elements 0, 5, 10 and 15, here the cells' Mueller
727    /// planes 0 to 3) through its own plane for w > 0 and through its
728    /// partner's, 15 − e (XX ↔ YY, XY ↔ YX), otherwise, reading the
729    /// visibility of that plane's hand. Each key lends one dense cell of
730    /// complex taps as its imaging and its weight taps, normalised as AW's
731    /// (`KernelSum`), with AW's pointing ramp.
732    struct AwRouted {
733        cells: Vec<casa_imaging_operator::DenseCell>,
734        mueller: casa_imaging_operator::MuellerRouting,
735        correction: casa_imaging_operator::ImageCorrection,
736    }
737
738    /// Taps per axis and fine offsets per cell of [`AwRouted`]'s cells.
739    const AW_SUPPORT: u16 = 5;
740    const AW_OVERSAMPLING: u16 = 4;
741
742    impl AwRouted {
743        /// `keys` cells of random complex taps, from `seed`.
744        fn new(keys: usize, mut seed: u64) -> Self {
745            use casa_imaging_operator::{DenseCell, ImageCorrection};
746            let mut signed = move || {
747                seed ^= seed << 13;
748                seed ^= seed >> 7;
749                seed ^= seed << 17;
750                ((seed >> 11) as f64 / (1_u64 << 53) as f64 * 2.0 - 1.0) as f32
751            };
752            let fine = usize::from(AW_OVERSAMPLING) + 1;
753            let taps = fine * fine * 4 * usize::from(AW_SUPPORT).pow(2);
754            let cells = (0..keys)
755                .map(|_| {
756                    DenseCell::new(
757                        (0..taps)
758                            .map(|_| Complex32::new(signed(), signed()))
759                            .collect(),
760                        [AW_SUPPORT; 2],
761                        AW_OVERSAMPLING,
762                        4,
763                    )
764                })
765                .collect();
766            let [nx, ny] = aw_geometry().grid_shape();
767            Self {
768                cells,
769                mueller: aw_routing(),
770                correction: ImageCorrection::new(vec![1.0; nx], vec![1.0; ny]),
771            }
772        }
773
774        /// These kernels with every tap replaced by its modulus: gridding a
775        /// sample with them, without its ramp and at the magnitude of its
776        /// spectral factor, adds `|W · s^t · tap|` to each cell.
777        fn magnitudes(&self) -> Self {
778            use casa_imaging_operator::DenseCell;
779            Self {
780                cells: self
781                    .cells
782                    .iter()
783                    .map(|cell| {
784                        DenseCell::new(
785                            cell.data()
786                                .iter()
787                                .map(|tap| Complex32::new(tap.norm(), 0.0))
788                                .collect(),
789                            cell.support(),
790                            cell.oversampling(),
791                            cell.mueller_planes(),
792                        )
793                    })
794                    .collect(),
795                mueller: self.mueller.clone(),
796                correction: self.correction.clone(),
797            }
798        }
799    }

801    /// AW's routes for XX, XY, YX and YY ([`AwRouted`]): for w > 0 grid
802    /// correlation `g` takes visibility `g` through plane `g`; otherwise it
803    /// takes its partner `3 − g` through plane `3 − g`. The forward tables
804    /// swap the signs (`GridToData`) and read the grid of the plane's hand.
805    fn aw_routing() -> casa_imaging_operator::MuellerRouting {
806        let table = |route: fn(usize, usize) -> Option<usize>| -> Vec<Vec<Option<u8>>> {
807            (0..4)
808                .map(|grid| {
809                    (0..4)
810                        .map(|vis| route(grid, vis).map(|plane| plane as u8))
811                        .collect()
812                })
813                .collect()
814        };
815        casa_imaging_operator::MuellerRouting {
816            adjoint: [
817                table(|grid, vis| (vis == 3 - grid).then_some(3 - grid)),
818                table(|grid, vis| (vis == grid).then_some(grid)),
819            ],
820            forward: [
821                table(|grid, vis| (vis == grid).then_some(grid)),
822                table(|grid, vis| (vis == 3 - grid).then_some(grid)),
823            ],
824        }
825    }

913    /// `samples` weight placements over `keys` cells, three pointing
914    /// gradients, three spectral factors and both signs of w, in random
915    /// order, each with random weights on XX, YX and YY and none on XY.
916    fn aw_routed_samples(
917        operator: &casa_imaging_operator::MeasurementOperator,
918        keys: u16,
919        samples: usize,
920    ) -> SampleBuffer {
921        let gradients = [[0.0, 0.0], [0.6, -0.2], [-0.3, 0.9]].map(|offset: [f64; 2]| {
922            operator
923                .geometry()
924                .pointing_gradient([offset[0] * 1.0e-4, offset[1] * 1.0e-4])
925        });
926        let spectral = [-0.01_f32, 0.0, 0.02];
927        let mut state = 0x2545_f491_4f6c_dd1d_u64;
928        let mut next = || {
929            state ^= state << 13;
930            state ^= state >> 7;
931            state ^= state << 17;
932            state
933        };
934        let mut buffer = SampleBuffer::with_capacity(4, samples);
935        for _ in 0..samples {
936            let draw = next();
937            // Fractions of 2^-16, so a group's sum of a thousand needs more
938            // bits than f32 keeps.
939            let weight = |shift: u32| 0.5 + ((draw >> shift) & 0xffff) as f32 / 65536.0;
940            buffer.push(
941                Placement {
942                    u: 0.0,
943                    v: 0.0,
944                    w: if draw & 1 == 0 { 25.0 } else { -25.0 },
945                    prediction_w_positive: true,
946                    phase: 0.0,
947                    plane: 0,
948                    spectral: spectral[(draw >> 1) as usize % 3],
949                    cf: CfKey {
950                        group: u16::try_from((draw >> 3) % u64::from(keys)).expect("a key"),
951                        cube: 0,
952                    },
953                    gradient: gradients[(draw >> 7) as usize % 3],
954                },
955                &[Complex32::default(); 4],
956                &[weight(16), 0.0, weight(32), weight(48)],
957            );
958        }
959        buffer
960    }

976    /// The weight blocks, cell by cell, and the weight `sumwt` that
977    /// `samples` grid on `operator`'s f64 accumulator: one sample at a time
978    /// (`capacity` `None`), or summed by contribution in a table of
979    /// `capacity` groups, with the grids that table made.
980    fn weight_grids(
981        operator: &casa_imaging_operator::MeasurementOperator,
982        samples: &SampleBuffer,
983        capacity: Option<usize>,
984    ) -> (Vec<num_complex::Complex64>, Vec<f64>, u64) {
985        use casa_imaging_operator::{CpuBackend, GridBackend, Mode, ModeSet, PlaneRange, Work};
986        let mut acc = operator.accumulator(PlaneRange::single(0), None, ModeSet::ALL);
987        let mut backend = CpuBackend::new();
988        let cf = operator.cf();
989        let mut grid = |block: &SampleBlock<'_>| {
990            backend.apply(
991                block,
992                cf,
993                Work::Grid {
994                    mode: Mode::Weight,
995                    acc: &mut acc,
996                },
997            )
998        };
999        let grids = match capacity {
1000            None => {
1001                grid(&samples.block()).expect("grid");
1002                0
1003            }
1004            Some(capacity) => {
1005                let mut sums = WeightSums::new(samples.block().npol, capacity);
1006                sums.add(&samples.block(), (cf, operator.geometry()), &mut grid)
1007                    .expect("add");
1008                sums.flush(&mut grid).expect("flush");
1009                sums.totals()[2]
1010            }
1011        };
1012        let layout = acc.layout();
1013        let terms = layout.term_range(Mode::Weight).expect("weight terms");
1014        let mut cells = Vec::new();
1015        let mut sumwt = Vec::new();
1016        for pol in 0..layout.pols() {
1017            for term in terms.clone() {
1018                cells.extend_from_slice(acc.block::<f64>(0, pol, term));
1019                sumwt.push(acc.sumwt_at(0, pol, term));
1020            }
1021        }
1022        (cells, sumwt, grids)
1023    }

1025    /// #792 review (Astra), #810, for #653's AW rows: under AW's routing
1026    /// ([`AwRouted`]) summing the weight image by contribution grids what
1027    /// gridding each sample grids, to the rounding of each group's sum to
1028    /// f32. The routes change with the sign of w and cross the hands, the
1029    /// cells are complex and dense with four Mueller planes, XY weighs
1030    /// nothing in either route that reads it, and the samples carry ramps
1031    /// and Taylor terms: in a table that grids once, and in one of seven
1032    /// groups that fills and grids every few samples.
1033    ///
1034    /// The bound is the arithmetic's, cell by cell (Astra's probes saw
1035    /// 6.7e-9 of the peak). A group's weights sum in f64 and each grid of
1036    /// the group rounds its partial sum to f32, within 2^-24 of it; the
1037    /// weights are positive, so a cell's summed value is within 2^-24 of
1038    /// `Σ |W · s^t · tap|` of the one gridded sample by sample, and a
1039    /// weight `sumwt` within 2^-24 of `Σ W · |s|^t · |N|`, both besides
1040    /// f64's own rounding over the samples.
1041    #[test]
1042    fn summed_weights_grid_what_each_sample_grids_under_aw_routing() {
1043        const KEYS: u16 = 3;
1044        const SAMPLES: usize = 60_000;
1045        let kernels = AwRouted::new(usize::from(KEYS), 0x9e37_79b9_7f4a_7c15);
1046        let magnitude_operator = aw_routed_operator(kernels.magnitudes());
1047        let operator = aw_routed_operator(kernels);
1048        let samples = aw_routed_samples(&operator, KEYS, SAMPLES);
1049        let (each, each_sumwt, _) = weight_grids(&operator, &samples, None);
1050        let unsigned = |placement: Placement| Placement {
1051            spectral: placement.spectral.abs(),
1052            ..placement
1053        };
1054        let (magnitudes, _, _) = weight_grids(
1055            &magnitude_operator,
1056            &mapped(&samples, |placement| Placement {
1057                gradient: [0.0, 0.0],
1058                ..unsigned(placement)
1059            }),
1060            None,
1061        );
1062        let (_, sumwt_magnitudes, _) = weight_grids(&operator, &mapped(&samples, unsigned), None);
1063        let rounding = F32_ROUNDING + 4.0 * SAMPLES as f64 * f64::EPSILON;
1064        let block = each.len() / each_sumwt.len();
1065        for (index, cells) in each.chunks(block).enumerate() {
1066            assert!(
1067                cells.iter().any(|cell| cell.norm() > 0.0),
1068                "weight block {index} has weight"
1069            );
1070        }
1071        for capacity in [WEIGHT_GROUPS, 7] {
1072            let (summed, summed_sumwt, grids) = weight_grids(&operator, &samples, Some(capacity));
1073            assert!(
1074                capacity == WEIGHT_GROUPS || grids > 1,
1075                "a table of {capacity} groups fills"
1076            );
1077            for (index, ((summed, each), magnitude)) in
1078                summed.iter().zip(&each).zip(&magnitudes).enumerate()
1079            {
1080                assert!(
1081                    (summed - each).norm() <= rounding * magnitude.re,
1082                    "{capacity} groups, cell {index}: {summed} against {each}, \
1083                     magnitude {magnitude}"
1084                );
1085            }
1086            for ((summed, each), magnitude) in
1087                summed_sumwt.iter().zip(&each_sumwt).zip(&sumwt_magnitudes)
1088            {
1089                assert!(
1090                    (summed - each).abs() <= rounding * magnitude,
1091                    "{capacity} groups, sumwt: {summed} against {each}, magnitude {magnitude}"
1092                );
1093            }
1094        }
1095    }
