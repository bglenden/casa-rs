17const ARCSEC: f64 = std::f64::consts::PI / (180. * 3600.);

19/// Naming, overwrite, mask and brightness policy for a total-power image.
20pub struct TotalPowerProducts {
21    /// CASA's base `outfile`; products append `.image` and `.weight`.
22    pub outfile: PathBuf,
23    /// Replace the task's previous CASA images after preparation succeeds.
24    pub overwrite: bool,
25    /// Fraction of the median positive weight below which pixels are masked.
26    pub minweight: f64,
27    /// Image brightness label, normally `Jy/beam` or `K`.
28    pub brightness_unit: String,
29}

31/// Published products and the beam needed for simulation flux scaling.
32pub struct PublishedTotalPower {
33    /// Gridded image.
34    pub image: PathBuf,
35    /// Sum-of-weights image.
36    pub weight: PathBuf,
37    /// Circular raster/kernel beam, when sampling is measurable.
38    pub beam: Option<GaussianBeam>,
39    _metadata: Reservation,
40}

42/// Grid and publish both CASA products, retaining previous results through
43/// preparation, gridding and beam estimation.
44///
45/// # Errors
46/// The gridding errors, invalid output/mask settings or image I/O failure.
47pub fn image_observation(
48    observation: &dyn Observation,
49    request: &TotalPowerRequest,
50    products: &TotalPowerProducts,
51    context: &RunContext,
52) -> Result<PublishedTotalPower, TotalPowerError> {
53    let prepared = prepare(products, context)?;
54    let result = grid_observation(observation, request, context)?;
55    publish_prepared(observation, request, products, context, result, prepared)
56}

58/// Publish an admitted grid using the task's normal beam, mask and safety checks.
59/// This also supports callers that retain a grid before choosing its output.
60///
61/// # Errors
62/// Invalid publication settings, resource admission, beam estimation or image I/O.
63pub fn publish_total_power(
64    observation: &dyn Observation,
65    request: &TotalPowerRequest,
66    products: &TotalPowerProducts,
67    context: &RunContext,
68    result: TotalPowerOutcome,
69) -> Result<PublishedTotalPower, TotalPowerError> {
70    let prepared = prepare(products, context)?;
71    publish_prepared(observation, request, products, context, result, prepared)
72}

74struct PreparedPublication {
75    metadata: Reservation,
76    image: PathBuf,
77    weight: PathBuf,
78}
79
80fn prepare(
81    products: &TotalPowerProducts,
82    context: &RunContext,
83) -> Result<PreparedPublication, TotalPowerError> {
84    let metadata = admit(
85        &context.host,
86        &context.policy,
87        &Demand {
88            phase: "total-power product metadata",
89            memory: 32 << 10,
90        },
91    )
92    .map_err(failure)?;
93    if !products.minweight.is_finite() || products.minweight < 0. {
94        return Err(failure("minweight must be finite and nonnegative"));
95    }
96    let outfile = std::path::absolute(&products.outfile).map_err(failure)?;
97    let target = |suffix: &str| {
98        let mut p = outfile.as_os_str().to_os_string();
99        p.push(suffix);
100        PathBuf::from(p)
101    };
102    let image = target(".image");
103    let weight = target(".weight");
104    for path in [&image, &weight] {
105        if path.try_exists().map_err(failure)? {
106            if !products.overwrite {
107                return Err(failure(format!("{} already exists", path.display())));
108            }
109            // An overwrite never replaces a different scientific dataset.
110            PagedImage::<f32>::open(path).map_err(failure)?;
111        }
112    }
113    Ok(PreparedPublication {
114        metadata,
115        image,
116        weight,
117    })
118}

120fn publish_prepared(
121    observation: &dyn Observation,
122    request: &TotalPowerRequest,
123    products: &TotalPowerProducts,
124    context: &RunContext,
125    result: TotalPowerOutcome,
126    prepared: PreparedPublication,
127) -> Result<PublishedTotalPower, TotalPowerError> {
128    let PreparedPublication {
129        metadata,
130        image,
131        weight,
132    } = prepared;
133    let cells = result.arrays.value.image.len() as u64;
134    let _scratch = admit(
135        &context.host,
136        &context.policy,
137        &Demand {
138            phase: "total-power masks, beam and publication",
139            memory: cells
140                .saturating_mul(5)
141                .saturating_add(16 << 20)
142                .saturating_add(casa_imaging_singledish::raster_sampling_working_bytes(
143                    result.sampling.len(),
144                )),
145        },
146    )
147    .map_err(failure)?;
148    let beam = beam(observation, request, &result)?;
149    if context.cancel.is_cancelled() {
150        return Err(failure("cancelled before publication"));
151    }
152    let parent = image
153        .parent()
154        .ok_or_else(|| failure("output has no parent"))?;
155    let staging = tempfile::Builder::new()
156        .prefix(".tsdimaging-")
157        .tempdir_in(parent)
158        .map_err(failure)?;
159    let staged_image = staging.path().join("image");
160    let staged_weight = staging.path().join("weight");
161    write(&staged_image, &result, products, beam)?;
162    let mut weights = PagedImage::<f32>::create(
163        result.arrays.value.weight.shape().to_vec(),
164        result.coordinates.clone(),
165        &staged_weight,
166    )
167    .map_err(failure)?;
168    weights
169        .put_slice(&result.arrays.value.weight, &[0; 4])
170        .map_err(failure)?;
171    weights.save().map_err(failure)?;
172    drop(weights);
173    if context.cancel.is_cancelled() {
174        return Err(failure("cancelled before publication"));
175    }
176    publish_staged_image(&staged_image, &image).map_err(failure)?;
177    publish_staged_image(&staged_weight, &weight).map_err(failure)?;
178    Ok(PublishedTotalPower {
179        image,
180        weight,
181        beam,
182        _metadata: metadata,
183    })
184}

186fn write(
187    path: &Path,
188    result: &TotalPowerOutcome,
189    products: &TotalPowerProducts,
190    beam: Option<GaussianBeam>,
191) -> Result<(), TotalPowerError> {
192    let data = &result.arrays.value;
193    let mut image = PagedImage::<f32>::create(
194        data.image.shape().to_vec(),
195        result.coordinates.clone(),
196        path,
197    )
198    .map_err(failure)?;
199    image.put_slice(&data.image, &[0; 4]).map_err(failure)?;
200    image
201        .set_units(&products.brightness_unit)
202        .map_err(failure)?;
203    if let Some(beam) = beam {
204        image
205            .set_image_info(&ImageInfo {
206                beam_set: ImageBeamSet::new(beam),
207                ..Default::default()
208            })
209            .map_err(failure)?;
210    }
211    if let Some(threshold) = weight_mask_threshold(
212        data.weight
213            .as_slice_memory_order()
214            .expect("contiguous grid"),
215        products.minweight,
216    ) {
217        let mask = data.weight.mapv(|value| f64::from(value) > threshold);
218        image.put_mask("mask0", &mask).map_err(failure)?;
219        image.set_default_mask("mask0").map_err(failure)?;
220    }
221    image.save().map_err(failure)
222}

224fn beam(
225    observation: &dyn Observation,
226    request: &TotalPowerRequest,
227    result: &TotalPowerOutcome,
228) -> Result<Option<GaussianBeam>, TotalPowerError> {
229    let Some(raster) = raster_sampling(&result.sampling) else {
230        return Ok(None);
231    };
232    let Some(sampling) = beam_sampling(raster.interval_rad.map(|v| v / ARCSEC)) else {
233        return Ok(None);
234    };
235    let catalogue = observation.catalogue();
236    let entry = catalogue.partition(result.source_partition);
237    let antenna = &catalogue.antennas[entry
238        .baselines
239        .iter()
240        .find(|b| {
241            b.is_auto() && (request.antennas.is_empty() || request.antennas.contains(&b.antenna1))
242        })
243        .ok_or_else(|| failure("no selected autocorrelation"))?
244        .antenna1
245        .0 as usize];
246    let CoordinateModel::Spectral(spectral) = result.coordinates.coordinate(2) else {
247        return Err(failure("no spectral coordinate"));
248    };
249    let frequency = spectral.rest_frequency();
250    let kernel = if request.kernel == GridFunction::PrimaryBeam {
251        let primary = PrimaryBeam::for_telescope(
252            &catalogue.observation(entry.observation).telescope,
253            spectral.reference_value()[0],
254        )
255        .ok_or_else(|| failure("unsupported primary beam"))?;
256        KernelTable::primary_beam(
257            &primary.pattern(),
258            request.direction.increment()[0],
259            spectral.reference_value()[0],
260        )
261    } else {
262        KernelTable::new(request.kernel).map_err(failure)?
263    };
264    let fwhm = theoretical_beam_arcsec(&BeamSetup {
265        diameter_m: antenna
266            .dish_diameter
267            .get_value_in(&casa_core::quanta::Unit::new("m").map_err(failure)?)
268            .map_err(failure)?,
269        obscuration_m: if ["PM", "DV", "DA", "CM"]
270            .iter()
271            .any(|prefix| antenna.name.starts_with(prefix))
272        {
273            0.75
274        } else {
275            0.
276        },
277        frequency_hz: frequency,
278        cell_arcsec: [
279            request.direction.increment()[0].abs() / ARCSEC,
280            request.direction.increment()[1].abs() / ARCSEC,
281        ],
282        kernel: (request.kernel, &kernel),
283        sampling_arcsec: &sampling,
284    });
285    if !fwhm.is_finite() || fwhm <= 0. {
286        return Err(failure("invalid theoretical beam"));
287    }
288    Ok(Some(GaussianBeam::new(fwhm * ARCSEC, fwhm * ARCSEC, 0.)))
289}
