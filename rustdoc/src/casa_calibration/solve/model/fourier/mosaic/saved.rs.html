11pub(super) fn rules(machine: &RecordValue, beam: &RecordValue) -> MosaicRules {
12    // The older HetArrayConvFunc keys its cache by MS and SPW. The newer
13    // Fourier machines include the selected native channel count as well.
14    // Some tclean records carry the older machine name as an alias.
15    if record_text(machine, "name").is_ok_and(|name| name == "MosaicFT")
16        && record_text(beam, "key0").map_or_else(
17            |_| machine.get("usepointingtable").is_none(),
18            |key| key.split('_').count() == 2,
19        )
20    {
21        MosaicRules::Simulator
22    } else {
23        MosaicRules::Tclean
24    }
25}

27pub(super) fn planes<'a>(
28    machine: &RecordValue,
29    beam: &'a RecordValue,
30    window: &MosaicWindow,
31) -> Result<Option<SavedMosaicPlanes<'a>>, GainSolveError> {
32    let key = format!(
33        "0_{}_{}",
34        window.spectral_window,
35        window.selected_frequencies_hz.len()
36    );
37    let count = integer(beam, "ndefined")?;
38    if count < 0 || count as usize > beam.fields().len() {
39        return Err(model_error("invalid virtual mosaic cache count"));
40    }
41    for entry in 0..count {
42        if record_text(beam, &format!("key{entry}"))? == key {
43            let index =
44                usize::try_from(integer(beam, &format!("val{entry}"))?).map_err(model_error)?;
45            return cached_planes(machine, beam, index);
46        }
47    }
48    Ok(None)
49}

51fn cached_planes<'a>(
52    machine: &RecordValue,
53    beam: &'a RecordValue,
54    index: usize,
55) -> Result<Option<SavedMosaicPlanes<'a>>, GainSolveError> {
56    let invalid = || model_error("invalid saved virtual mosaic convolution");
57    let Some(Value::Array(ArrayValue::Bool(done))) = beam.get("donemainconv") else {
58        return Err(invalid());
59    };
60    if !done
61        .get(ndarray::IxDyn(&[index]))
62        .copied()
63        .ok_or_else(invalid)?
64    {
65        return Ok(None);
66    }
67    let (shape, strides, values) = match beam.get(&format!("convfunctions{index}")) {
68        Some(Value::Array(ArrayValue::Complex32(grid))) => (
69            grid.shape(),
70            grid.strides(),
71            Cow::Borrowed(grid.as_slice_memory_order().ok_or_else(invalid)?),
72        ),
73        Some(Value::Array(ArrayValue::Complex64(grid))) => (
74            grid.shape(),
75            grid.strides(),
76            Cow::Owned(
77                grid.as_slice_memory_order()
78                    .ok_or_else(invalid)?
79                    .iter()
80                    .map(|value| Complex32::new(value.re as f32, value.im as f32))
81                    .collect(),
82            ),
83        ),
84        _ => return Err(invalid()),
85    };
86    let &[side, height, 1, cells, pairs] = shape else {
87        return Err(invalid());
88    };
89    let dishes = (1..=256)
90        .find(|dishes| dishes * (dishes + 1) / 2 == pairs)
91        .ok_or_else(invalid)?;
92    let support = match beam.get(&format!("convsupportblock{index}")) {
93        Some(Value::Array(ArrayValue::Int32(values))) => {
94            values.iter().copied().max().map(i64::from)
95        }
96        Some(Value::Array(ArrayValue::Int64(values))) => values.iter().copied().max(),
97        _ => None,
98    };
99    if side != height || strides[0] != 1 {
100        return Err(invalid());
101    }
102    let half_support = support
103        .and_then(|max| u16::try_from(max).ok())
104        .ok_or_else(invalid)?;
105    let sampling = u16::try_from(integer(machine, "convsampling")?).map_err(model_error)?;
106    Ok(Some(SavedMosaicPlanes {
107        values,
108        side,
109        cells,
110        dishes,
111        half_support,
112        sampling,
113    }))
114}

116pub(super) fn integer(record: &RecordValue, field: &str) -> Result<i32, GainSolveError> {
117    match record.get(field) {
118        Some(Value::Scalar(ScalarValue::Int32(value))) => Ok(*value),
119        Some(Value::Scalar(ScalarValue::Int64(value))) => {
120            i32::try_from(*value).map_err(model_error)
121        }
122        _ => Err(model_error(format!(
123            "virtual mosaic has no {field} integer"
124        ))),
125    }
126}
