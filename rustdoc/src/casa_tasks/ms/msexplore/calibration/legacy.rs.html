37pub(super) fn materialize(
38    source: Table,
39    path: &Path,
40    selection: &Expressions<'_>,
41    averaged: bool,
42    polarization: &str,
43) -> Result<(Table, Option<SplineDisplay>), String> {
44    let subtype = source.info().sub_type.as_str();
45    if !matches!(subtype, "BPOLY" | "GSPLINE") {
46        return Ok((source, None));
47    }
48    if averaged {
49        return Err(format!(
50            "Averaging not supported for cal table type {subtype}"
51        ));
52    }
53    if subtype == "BPOLY" && polarization == "/" {
54        return Err("Polarization ratio plots not supported for BPOLY tables.".into());
55    }
56    let input = LegacySource::open(&source, path)?;
57    let mut virtual_table = input.virtual_table(subtype)?;
58    let spline = if subtype == "BPOLY" {
59        input.bandpass(&source, &mut virtual_table)?;
60        None
61    } else {
62        Some(input.spline(&source, selection, &mut virtual_table)?)
63    };
64    Ok((virtual_table, spline))
65}

67pub(super) fn check_spline_axes(axes: impl IntoIterator<Item = Axis>) -> Result<(), String> {
68    use Axis::*;
69    for axis in axes {
70        if !matches!(
71            axis,
72            Scan | Field
73                | Time
74                | Window
75                | Interval
76                | Receptor
77                | Antenna1
78                | Feed1
79                | Flag
80                | Amplitude
81                | GainAmplitude
82                | Phase
83                | GainPhase
84                | Real
85                | GainReal
86                | Imaginary
87                | GainImaginary
88                | Observation
89        ) {
90            return Err(format!(
91                "GSPLINE plotting does not support {} axis",
92                axis.label()
93            ));
94        }
95    }
96    Ok(())
97}

99struct LegacySource {
100    ms: MeasurementSet,
101    path: PathBuf,
102    windows: Vec<Vec<i32>>,
103}

105impl LegacySource {
106    fn open(table: &Table, path: &Path) -> Result<Self, String> {
107        let description = subtable(table, path, "CAL_DESC")?;
108        let name = text(&description, 0, "MS_NAME")
109            .filter(|name| !name.is_empty())
110            .ok_or("Associated MS is not available, cannot plot solutions.")?;
111        let path = path.parent().unwrap_or_else(|| Path::new(".")).join(name);
112        let path = std::path::absolute(path).map_err(display)?;
113        let ms = MeasurementSet::open(&path)
114            .map_err(|_| "Associated MS is not available, cannot plot solutions.")?;
115        let windows = (0..description.row_count())
116            .map(
117                |row| match array(&description, row, "SPECTRAL_WINDOW_ID")? {
118                    ArrayValue::Int32(values) if !values.is_empty() => {
119                        Ok(values.iter().copied().collect())
120                    }
121                    ArrayValue::Int32(_) => Err("CAL_DESC has no spectral window".into()),
122                    _ => Err("CAL_DESC spectral-window IDs must be integers".into()),
123                },
124            )
125            .collect::<Result<_, String>>()?;
126        Ok(Self { ms, path, windows })
127    }
128
129    fn virtual_table(&self, subtype: &str) -> Result<Table, String> {
130        let columns = ["TIME", "INTERVAL"]
131            .into_iter()
132            .map(|name| ColumnSchema::scalar(name, PrimitiveType::Float64))
133            .chain(
134                [
135                    "SCAN_NUMBER",
136                    "FIELD_ID",
137                    "SPECTRAL_WINDOW_ID",
138                    "ANTENNA1",
139                    "ANTENNA2",
140                    "FEED1",
141                    "OBSERVATION_ID",
142                ]
143                .into_iter()
144                .map(|name| ColumnSchema::scalar(name, PrimitiveType::Int32)),
145            )
146            .chain(
147                [
148                    ("CPARAM", PrimitiveType::Complex32),
149                    ("FLAG", PrimitiveType::Bool),
150                    ("SNR", PrimitiveType::Float32),
151                ]
152                .into_iter()
153                .map(|(name, kind)| ColumnSchema::array_variable(name, kind, Some(2))),
154            )
155            .collect();
156        let mut result = Table::with_schema(TableSchema::new(columns).map_err(display)?);
157        result.set_info(TableInfo {
158            table_type: "Calibration".into(),
159            sub_type: subtype.into(),
160            ..TableInfo::default()
161        });
162        for id in [
163            SubtableId::Antenna,
164            SubtableId::Field,
165            SubtableId::SpectralWindow,
166            SubtableId::Observation,
167        ] {
168            let path = subtable_path_from_main(self.ms.main_table(), &self.path, id)
169                .ok_or_else(|| format!("associated MS has no {} table", id.name()))?;
170            result.keywords_mut().push(RecordField::new(
171                id.name(),
172                Value::TableRef(path.to_string_lossy().into_owned()),
173            ));
174        }
175        result.keywords_mut().push(RecordField::new(
176            "MSName",
177            Value::Scalar(ScalarValue::String(
178                self.path.to_string_lossy().into_owned(),
179            )),
180        ));
181        Ok(result)
182    }

184    fn bandpass(&self, source: &Table, output: &mut Table) -> Result<(), String> {
185        let spectral_window = self.ms.spectral_window().map_err(display)?;
186        let frequencies = (0..spectral_window.row_count())
187            .map(|row| spectral_window.chan_freq(row).map_err(display))
188            .collect::<Result<Vec<_>, _>>()?;
189        for row in 0..source.row_count() {
190            let description = integer(source, row, "CAL_DESC_ID")
191                .and_then(|id| usize::try_from(id).ok())
192                .and_then(|id| self.windows.get(id))
193                .map(|ids| ids[0])
194                .ok_or("invalid legacy CAL_DESC_ID")?;
195            let frequencies = usize::try_from(description)
196                .ok()
197                .and_then(|id| frequencies.get(id))
198                .ok_or("legacy spectral window is missing from the associated MS")?;
199            let domain: [f64; 2] = array_values(source, row, "VALID_DOMAIN")?
200                .try_into()
201                .map_err(|_| "BPOLY needs two domain frequencies")?;
202            if !domain[0].is_finite() || !domain[1].is_finite() || domain[0] > domain[1] {
203                return Err("invalid BPOLY frequency domain".into());
204            }
205            let amp = coefficients(source, row, "N_POLY_AMP", "POLY_COEFF_AMP")?;
206            let phase = coefficients(source, row, "N_POLY_PHASE", "POLY_COEFF_PHASE")?;
207            let scale = gain_scale(source, row)?;
208            let active = [0, 1].map(|pol| {
209                amp[pol]
210                    .iter()
211                    .chain(&phase[pol])
212                    .any(|&value| value != 0.0)
213            });
214            let values = Array2::from_shape_fn((2, frequencies.len()), |(pol, channel)| {
215                let frequency = frequencies[channel];
216                if !active[pol] || frequency < domain[0] || frequency > domain[1] {
217                    return Complex32::new(1.0, 0.0);
218                }
219                let amplitude =
220                    legacy_bpoly_chebyshev_value(&amp[pol], domain[0], domain[1], frequency).exp();
221                let angle =
222                    legacy_bpoly_chebyshev_value(&phase[pol], domain[0], domain[1], frequency);
223                scale * Complex32::from_polar(amplitude as f32, angle as f32)
224            });
225            let flags = Array2::from_shape_fn(values.dim(), |(pol, _)| !active[pol]);
226            let meta = RowMetadata::read(source, row, description)?;
227            write_row(
228                output,
229                RowMetadata {
230                    observation: meta.observation.max(0),
231                    ..meta
232                },
233                values,
234                flags,
235            )?;
236        }
237        Ok(())
238    }

240    fn selected_spline_rows(
241        &self,
242        source: &Table,
243        selection: &Expressions<'_>,
244    ) -> Result<Vec<(usize, RowMetadata)>, String> {
245        if !selection.timerange.is_empty() {
246            tracing::warn!("Time selection not supported for GSPLINE calibration tables");
247        }
248        let fields = self
249            .ms
250            .resolve_field_selector(selection.field)
251            .map_err(display)?;
252        let mut antennas = BTreeSet::new();
253        let mut excluded = BTreeSet::new();
254        for clause in selection.antenna.split(';').map(str::trim) {
255            let (set, expr) = clause
256                .strip_prefix('!')
257                .map_or_else(|| (&mut antennas, clause), |expr| (&mut excluded, expr));
258            set.extend(self.ms.resolve_reference_antennas(expr).map_err(display)?);
259        }
260        let windows = self
261            .ms
262            .resolve_spw_selector(selection.spw)
263            .map_err(display)?;
264        let scans = parse_numeric_id_selector(selection.scan, "scan").map_err(display)?;
265        let observations =
266            parse_numeric_id_selector(selection.observation, "observation").map_err(display)?;
267        let raw_rows = if selection.taql.is_empty() {
268            (0..source.row_count()).collect::<Vec<_>>()
269        } else {
270            source
271                .query(&format!("SELECT * WHERE ({})", selection.taql))
272                .map_err(display)?
273                .row_numbers()
274                .to_vec()
275        };
276        let mut matched = Vec::new();
277        for row in raw_rows {
278            let index = integer(source, row, "CAL_DESC_ID")
279                .and_then(|id| usize::try_from(id).ok())
280                .ok_or("invalid GSPLINE CAL_DESC_ID")?;
281            let description = self
282                .windows
283                .get(index)
284                .ok_or("missing GSPLINE CAL_DESC row")?;
285            if !windows.selects_all()
286                && !description
287                    .iter()
288                    .any(|&id| usize::try_from(id).is_ok_and(|id| windows.spw_ids().contains(&id)))
289            {
290                continue;
291            }
292            let metadata = RowMetadata::read(source, row, description[0])?;
293            let antenna = metadata.antenna1 as usize;
294            if (fields.is_empty() || fields.contains(&metadata.field))
295                && (scans.is_empty() || scans.contains(&metadata.scan))
296                && (observations.is_empty() || observations.contains(&metadata.observation))
297                && (antennas.is_empty() || antennas.contains(&antenna))
298                && !excluded.contains(&antenna)
299            {
300                matched.push((row, metadata));
301            }
302        }
303        if matched.is_empty() {
304            return Err("calibration selection contains no rows".into());
305        }
306        Ok(matched)
307    }

309    fn spline(
310        &self,
311        source: &Table,
312        selection: &Expressions<'_>,
313        output: &mut Table,
314    ) -> Result<SplineDisplay, String> {
315        let matched = self.selected_spline_rows(source, selection)?;
316        let selected_fields: BTreeSet<_> =
317            matched.iter().map(|(_, metadata)| metadata.field).collect();
318        let field = *selected_fields.first().expect("matched rows");
319        let window = matched[0].1.window;
320        let display_window = common_value(self.windows.iter().flatten().copied(), -1);
321        let scan = common_value(matched.iter().map(|(_, meta)| meta.scan), -1);
322        let interval = common_value(matched.iter().map(|(_, meta)| meta.interval), -1.0);
323        let mut signals = BTreeMap::new();
324        for (row, meta) in matched
325            .into_iter()
326            .filter(|(_, metadata)| metadata.field == field)
327        {
328            if let std::collections::btree_map::Entry::Vacant(slot) = signals.entry(meta.antenna1) {
329                slot.insert((meta, Spline::read(source, row)?));
330            }
331        }
332        let first = &signals
333            .first_key_value()
334            .ok_or("GSPLINE has no solutions")?
335            .1
336            .1;
337        let (start, stop) = first.range()?;
338        let reference_frequency = first.reference_frequency;
339        let step = (stop - start) / 1000.0;
340        for index in 0..1000 {
341            let time = start + 0.5 * step + f64::from(index) * step;
342            for (meta, spline) in signals.values() {
343                let [first, second] = spline.evaluate(time, reference_frequency)?;
344                let values =
345                    Array2::from_shape_vec((2, 1), vec![first, second]).expect("two receptors");
346                let flags = Array2::from_elem((2, 1), false);
347                write_row(
348                    output,
349                    RowMetadata {
350                        time,
351                        window: window.max(0),
352                        ..*meta
353                    },
354                    values,
355                    flags,
356                )?;
357            }
358        }
359        Ok(SplineDisplay {
360            field: if selected_fields.len() == 1 {
361                field
362            } else {
363                -1
364            },
365            window: display_window,
366            scan,
367            interval,
368        })
369    }
370}

372#[derive(Clone, Copy)]
373struct RowMetadata {
374    time: f64,
375    interval: f64,
376    scan: i32,
377    field: i32,
378    window: i32,
379    antenna1: i32,
380    antenna2: i32,
381    feed1: i32,
382    observation: i32,
383}
384
385impl RowMetadata {
386    fn read(source: &Table, row: usize, window: i32) -> Result<Self, String> {
387        let antenna = integer(source, row, "ANTENNA1")
388            .filter(|&id| id >= 0)
389            .ok_or("legacy calibration row has no valid antenna")?;
390        let reference = array(source, row, "REF_ANT")
391            .ok()
392            .and_then(|values| match values {
393                ArrayValue::Int32(values) => values.iter().copied().next(),
394                _ => None,
395            })
396            .unwrap_or(-1);
397        Ok(Self {
398            time: number(source, row, "TIME").ok_or("legacy calibration row has no TIME")?,
399            interval: number(source, row, "INTERVAL").unwrap_or_default(),
400            scan: integer(source, row, "SCAN_NUMBER").unwrap_or(-1),
401            field: integer(source, row, "FIELD_ID").unwrap_or(-1),
402            window,
403            antenna1: antenna,
404            antenna2: reference,
405            feed1: integer(source, row, "FEED1").unwrap_or(-1),
406            observation: integer(source, row, "OBSERVATION_ID").unwrap_or(-1),
407        })
408    }
409}

411fn write_row(
412    output: &mut Table,
413    row: RowMetadata,
414    values: Array2<Complex32>,
415    flags: Array2<bool>,
416) -> Result<(), String> {
417    let scalar = |name, value| RecordField::new(name, Value::Scalar(value));
418    let array = |name, value| RecordField::new(name, Value::Array(value));
419    let snr = Array2::from_elem(values.dim(), 1.0_f32);
420    output
421        .add_row(RecordValue::new(vec![
422            scalar("TIME", ScalarValue::Float64(row.time)),
423            scalar("INTERVAL", ScalarValue::Float64(row.interval)),
424            scalar("SCAN_NUMBER", ScalarValue::Int32(row.scan)),
425            scalar("FIELD_ID", ScalarValue::Int32(row.field)),
426            scalar("SPECTRAL_WINDOW_ID", ScalarValue::Int32(row.window)),
427            scalar("ANTENNA1", ScalarValue::Int32(row.antenna1)),
428            scalar("ANTENNA2", ScalarValue::Int32(row.antenna2)),
429            scalar("FEED1", ScalarValue::Int32(row.feed1)),
430            scalar("OBSERVATION_ID", ScalarValue::Int32(row.observation)),
431            array("CPARAM", ArrayValue::Complex32(values.into_dyn())),
432            array("FLAG", ArrayValue::Bool(flags.into_dyn())),
433            array("SNR", ArrayValue::Float32(snr.into_dyn())),
434        ]))
435        .map_err(display)
436}

438fn common_value<T: Copy + PartialEq>(mut values: impl Iterator<Item = T>, ambiguous: T) -> T {
439    let Some(first) = values.next() else {
440        return ambiguous;
441    };
442    if values.all(|value| value == first) {
443        first
444    } else {
445        ambiguous
446    }
447}

449struct Spline {
450    amplitude: Option<SplineComponent>,
451    phase: Option<SplineComponent>,
452    scale: Complex32,
453    reference_frequency: f64,
454}
455
456impl Spline {
457    fn read(table: &Table, row: usize) -> Result<Self, String> {
458        let mode = text(table, row, "POLY_MODE").ok_or("GSPLINE row has no POLY_MODE")?;
459        let amplitude = (mode.contains("AMP") || mode.contains("A&P"))
460            .then(|| SplineComponent::read(table, row, "AMP"))
461            .transpose()?;
462        let phase = (mode.contains("PHAS") || mode.contains("A&P"))
463            .then(|| SplineComponent::read(table, row, "PHASE"))
464            .transpose()?;
465        let reference_frequency = array_values(table, row, "REF_FREQUENCY")?
466            .first()
467            .copied()
468            .ok_or("GSPLINE row has no reference frequency")?;
469        Ok(Self {
470            amplitude,
471            phase,
472            scale: gain_scale(table, row)?,
473            reference_frequency,
474        })
475    }
476
477    fn range(&self) -> Result<(f64, f64), String> {
478        let knots = &self
479            .amplitude
480            .as_ref()
481            .or(self.phase.as_ref())
482            .ok_or("GSPLINE has no active coefficients")?
483            .knots;
484        Ok((knots[0], *knots.last().expect("valid knots")))
485    }
486
487    fn evaluate(&self, time: f64, frequency: f64) -> Result<[Complex32; 2], String> {
488        let ratio = if self.reference_frequency == 0.0 {
489            1.0
490        } else {
491            frequency / self.reference_frequency
492        };
493        let value = |pol| -> Result<Complex32, String> {
494            let amp = self
495                .amplitude
496                .as_ref()
497                .map_or(Ok(0.0), |part| part.value(time, pol))?;
498            let phase = self
499                .phase
500                .as_ref()
501                .map_or(Ok(0.0), |part| part.value(time, pol))?;
502            Ok(self.scale * Complex32::from_polar(amp.exp() as f32, (-phase * ratio) as f32))
503        };
504        Ok([value(0)?, value(1)?])
505    }
506}

508struct SplineComponent {
509    knots: Vec<f64>,
510    coefficients: [Vec<f64>; 2],
511}
512
513impl SplineComponent {
514    fn read(table: &Table, row: usize, component: &str) -> Result<Self, String> {
515        let count = integer(table, row, &format!("N_KNOTS_{component}"))
516            .and_then(|count| usize::try_from(count).ok())
517            .ok_or("invalid GSPLINE knot count")?;
518        let knots = array_values(table, row, &format!("SPLINE_KNOTS_{component}"))?;
519        let knots = knots
520            .get(..count)
521            .ok_or("GSPLINE knot count exceeds its array length")?
522            .to_vec();
523        if knots.len() < 8
524            || knots.iter().any(|value| !value.is_finite())
525            || knots.windows(2).any(|pair| pair[0] > pair[1])
526            || knots[0] >= knots[knots.len() - 1]
527        {
528            return Err("invalid GSPLINE knot vector".into());
529        }
530        let coefficients = coefficients(
531            table,
532            row,
533            &format!("N_POLY_{component}"),
534            &format!("POLY_COEFF_{component}"),
535        )?;
536        Ok(Self {
537            knots,
538            coefficients,
539        })
540    }
541
542    fn value(&self, time: f64, pol: usize) -> Result<f64, String> {
543        legacy_cubic_spline(&self.knots, &self.coefficients[pol], time)
544            .ok_or_else(|| "cannot evaluate legacy cubic spline".into())
545    }
546}
547
548fn coefficients(
549    table: &Table,
550    row: usize,
551    count: &str,
552    name: &str,
553) -> Result<[Vec<f64>; 2], String> {
554    let count = integer(table, row, count)
555        .and_then(|count| usize::try_from(count).ok())
556        .ok_or_else(|| format!("invalid legacy coefficient count for {name}"))?;
557    if count == 0 {
558        return Ok([Vec::new(), Vec::new()]);
559    }
560    let values = array_values(table, row, name)?;
561    let first = values
562        .get(..count)
563        .ok_or_else(|| format!("truncated legacy coefficients in {name}"))?;
564    let second = values
565        .get(count..2 * count)
566        .ok_or_else(|| format!("missing second receptor coefficients in {name}"))?;
567    Ok([first.to_vec(), second.to_vec()])
568}
569
570fn array_values(table: &Table, row: usize, name: &str) -> Result<Vec<f64>, String> {
571    match array(table, row, name)? {
572        ArrayValue::Float64(values) => Ok(values.iter().copied().collect()),
573        ArrayValue::Float32(values) => Ok(values.iter().copied().map(f64::from).collect()),
574        _ => Err(format!(
575            "legacy calibration column {name} is not a floating array"
576        )),
577    }
578}
579
580fn gain_scale(table: &Table, row: usize) -> Result<Complex32, String> {
581    match table
582        .cell_accessor(row, "SCALE_FACTOR")
583        .and_then(|cell| cell.scalar())
584        .map_err(display)?
585    {
586        ScalarValue::Complex32(value) => Ok(*value),
587        ScalarValue::Complex64(value) => Ok(Complex32::new(value.re as f32, value.im as f32)),
588        _ => Err("legacy calibration SCALE_FACTOR is not complex".into()),
589    }
590}
