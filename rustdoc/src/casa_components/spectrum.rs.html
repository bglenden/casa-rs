2//! A component's spectral model, CASA's `SpectralModel` family.
3//!
4//! CASA's spectral models are a class hierarchy under `SpectralModel`, which
5//! holds the reference frequency and its error. Here one [`SpectralModel`]
6//! holds those, and a [`SpectralKind`] enum carries what each model adds.
7//!
8//! # Parameters
9//!
10//! A component list stores a spectrum as its name (`Spectrum_Shape`), its
11//! reference frequency (`Reference_Frequency`, `Frequency_Frame`), that
12//! frequency's error (`Frequency_Error`, `Frequency_Error_Units`), two vectors
13//! laid out as CASA's `parameters()` and `errors()`
14//! ([`SpectralModel::parameters`], [`SpectralModel::errors`]), and the record
15//! CASA's `toRecord` writes (`Spectral_Record`), which carries what the vectors
16//! cannot:
17//!
18//! | model | parameters | errors | only in `Spectral_Record` |
19//! |---|---|---|---|
20//! | constant | none | `[0]` | nothing |
21//! | spectral index | α | α's error | the four Stokes indices |
22//! | tabular | none | none | the frame, frequencies and Stokes fluxes of the table |
23//! | power-log-polynomial | the coefficients | their errors | nothing |
24//!
25//! (CASA's constant spectrum reports one zero error although it has no
26//! parameters; casa-rs writes the same.)
27
28use casa_types::measures::frequency::{FrequencyRef, MFrequency};
29use casa_types::quanta::{Quantity, Unit};

34/// A spectral index: the flux scales as (ν/ν₀)^α.
35///
36/// CASA's `SpectralIndex` keeps α and, optionally, indices for all four
37/// Stokes parameters, `[α_I, α_pol, RM, α_V]`: I scales with α_I, Q and U with
38/// α_I + α_pol (so the fractional linear polarisation scales with α_pol) and
39/// are rotated by the rotation measure RM (rad m⁻²), and V scales with
40/// α_I + α_V.
41///
42/// Without the four, CASA keeps a one-element Stokes index, which it reports
43/// (`stokesIndex()`, and a list's `Spectral_Record`) but never uses: α alone
44/// scales the flux. Setting α (`setIndex`) sets it to α, but setting the
45/// parameters (`setParameters`, as `readTable` does) leaves it, so a list
46/// read from a table without `Spectral_Record` reports 0. casa-rs keeps it as
47/// CASA does, so that what it writes and reports matches CASA.
48#[derive(Debug, Clone, PartialEq)]
49pub struct SpectralIndex {
50    index: f64,
51    error: f64,
52    stokes_index: StokesIndex,
53}
54
55/// CASA's `itsStokesIndex`: one value (reported, not used) or the four.
56#[derive(Debug, Clone, Copy, PartialEq)]
57enum StokesIndex {
58    Single(f64),
59    Full([f64; 4]),
60}
61
62impl SpectralIndex {
63    /// A spectral index α with zero error and no Stokes indices
64    /// (`SpectralIndex(refFreq, exponent)`).
65    pub fn new(index: f64) -> Self {
66        Self {
67            index,
68            error: 0.0,
69            stokes_index: StokesIndex::Single(index),
70        }
71    }
72
73    /// Replace CASA's Stokes index as `SpectralIndex::fromRecord` does from a
74    /// record's `stokesindex`: one value or four, leaving α alone. Other
75    /// lengths are ignored, as `fromRecord` refuses them.
76    pub(crate) fn with_casa_stokes_index(mut self, values: &[f64]) -> Self {
77        match *values {
78            [single] => self.stokes_index = StokesIndex::Single(single),
79            [i, pol, rm, v] => self.stokes_index = StokesIndex::Full([i, pol, rm, v]),
80            _ => {}
81        }
82        self
83    }
84
85    /// This index with α's error set (`setErrors`).
86    ///
87    /// # Errors
88    ///
89    /// [`ComponentError::Spectrum`] when the error is negative, which CASA
90    /// refuses.
91    pub fn with_error(mut self, error: f64) -> Result<Self, ComponentError> {
92        self.error = non_negative_error("spectral index", error)?;
93        Ok(self)
94    }
95
96    /// This index with the four Stokes indices `[α_I, α_pol, RM, α_V]` set
97    /// (`setStokesIndex`). As in CASA, α becomes α_I.
98    pub fn with_stokes_indices(mut self, stokes_indices: [f64; 4]) -> Self {
99        self.index = stokes_indices[0];
100        self.stokes_index = StokesIndex::Full(stokes_indices);
101        self
102    }
103
104    /// α (`index`).
105    pub fn index(&self) -> f64 {
106        self.index
107    }
108
109    /// α's error.
110    pub fn error(&self) -> f64 {
111        self.error
112    }
113
114    /// The four Stokes indices, when they were set (`stokesIndex` of length
115    /// four).
116    pub fn stokes_indices(&self) -> Option<[f64; 4]> {
117        match self.stokes_index {
118            StokesIndex::Full(indices) => Some(indices),
119            StokesIndex::Single(_) => None,
120        }
121    }
122
123    /// CASA's `stokesIndex()`: the four indices, or the one-element vector
124    /// CASA reports without them.
125    pub fn casa_stokes_index(&self) -> Vec<f64> {
126        match self.stokes_index {
127            StokesIndex::Full(indices) => indices.to_vec(),
128            StokesIndex::Single(single) => vec![single],
129        }
130    }
131}

133/// Flux tabulated against frequency, CASA's `TabularSpectrum`.
134///
135/// The frequencies are in Hz in the tabulation frame ([`Self::frame`]), and
136/// each flux is Stokes I, Q, U and V. CASA normalises the tabulated flux to its
137/// value at the reference frequency expressed in the tabulation frame.
138#[derive(Debug, Clone, PartialEq)]
139pub struct TabularSpectrum {
140    frame: FrequencyRef,
141    frequencies_hz: Vec<f64>,
142    stokes_flux: Vec<[f64; 4]>,
143}
144
145impl TabularSpectrum {
146    /// A tabulation in `frame`.
147    ///
148    /// # Errors
149    ///
150    /// [`ComponentError::Spectrum`] when the frequencies and fluxes differ in
151    /// length (`TabularSpectrum::setValues`).
152    pub fn new(
153        frame: FrequencyRef,
154        frequencies_hz: Vec<f64>,
155        stokes_flux: Vec<[f64; 4]>,
156    ) -> Result<Self, ComponentError> {
157        if frequencies_hz.len() != stokes_flux.len() {
158            return Err(ComponentError::Spectrum {
159                spectrum: "tabular",
160                reason: format!(
161                    "{} frequencies but {} fluxes",
162                    frequencies_hz.len(),
163                    stokes_flux.len()
164                ),
165            });
166        }
167        Ok(Self {
168            frame,
169            frequencies_hz,
170            stokes_flux,
171        })
172    }
173
174    /// The empty tabulation in LSRK of CASA's default `TabularSpectrum`, which
175    /// a table row without a `Spectral_Record` reads as.
176    pub fn empty() -> Self {
177        Self {
178            frame: FrequencyRef::LSRK,
179            frequencies_hz: Vec::new(),
180            stokes_flux: Vec::new(),
181        }
182    }
183
184    /// The frame the frequencies are in (`freqRef`).
185    pub fn frame(&self) -> FrequencyRef {
186        self.frame
187    }
188
189    /// The tabulated frequencies, Hz (`tabFreqVal`).
190    pub fn frequencies_hz(&self) -> &[f64] {
191        &self.frequencies_hz
192    }
193
194    /// The tabulated Stokes I, Q, U and V fluxes (`ival`, `qval`, `uval`,
195    /// `vval`).
196    pub fn stokes_flux(&self) -> &[[f64; 4]] {
197        &self.stokes_flux
198    }
199
200    /// The highest and lowest tabulated frequencies, Hz (`maxFreq`,
201    /// `minFreq`), or zeros for an empty tabulation as CASA's default.
202    pub fn frequency_range_hz(&self) -> (f64, f64) {
203        if self.frequencies_hz.is_empty() {
204            return (0.0, 0.0);
205        }
206        let max = self
207            .frequencies_hz
208            .iter()
209            .copied()
210            .fold(f64::NEG_INFINITY, f64::max);
211        let min = self
212            .frequencies_hz
213            .iter()
214            .copied()
215            .fold(f64::INFINITY, f64::min);
216        (max, min)
217    }
218
219    /// The reference frequency expressed in the tabulation frame, Hz
220    /// (CASA's `referenceFreq`, from `SpectralModel::refFreqInFrame`).
221    ///
222    /// CASA takes the value unconverted when the frames agree, or when either
223    /// is REST, and otherwise converts it with an empty measures frame, which
224    /// fails for every pair of moving frames.
225    ///
226    /// # Errors
227    ///
228    /// [`ComponentError::Spectrum`] when the frames differ and neither is
229    /// REST.
230    pub fn reference_frequency_in_frame(
231        &self,
232        reference: &MFrequency,
233    ) -> Result<f64, ComponentError> {
234        let unconverted = reference.refer() == self.frame
235            || reference.refer() == FrequencyRef::REST
236            || self.frame == FrequencyRef::REST;
237        if unconverted {
238            Ok(reference.hz())
239        } else {
240            Err(ComponentError::Spectrum {
241                spectrum: "tabular",
242                reason: format!(
243                    "the reference frequency is in {} but the tabulation is in {}; \
244                     CASA cannot convert between them without a measures frame",
245                    reference.refer().as_str(),
246                    self.frame.as_str()
247                ),
248            })
249        }
250    }
251}

253/// A power law whose exponent is a polynomial in the logarithm of the
254/// frequency ratio, CASA's `PowerLogPoly`: with x = ν/ν₀ the flux scales as
255/// x^(c₀ + c₁ ln x + c₂ (ln x)² + …).
256#[derive(Debug, Clone, PartialEq)]
257pub struct PowerLogPolynomial {
258    coefficients: Vec<f64>,
259    errors: Vec<f64>,
260}
261
262impl PowerLogPolynomial {
263    /// Coefficients c₀, c₁, … with zero errors (`PowerLogPoly(refFreq, coeffs)`).
264    pub fn new(coefficients: Vec<f64>) -> Self {
265        let errors = vec![0.0; coefficients.len()];
266        Self {
267            coefficients,
268            errors,
269        }
270    }
271
272    /// This polynomial with its coefficients' errors set (`setErrors`).
273    ///
274    /// As in CASA, the coefficients are resized to the number of errors:
275    /// those that remain keep their values and new ones are zero.
276    ///
277    /// # Errors
278    ///
279    /// [`ComponentError::Spectrum`] when an error is negative, which CASA
280    /// refuses; the polynomial is then unchanged.
281    pub fn with_errors(mut self, errors: Vec<f64>) -> Result<Self, ComponentError> {
282        self.set_errors(errors)?;
283        Ok(self)
284    }
285
286    /// `PowerLogPoly::setErrors`: refuse negative errors, then take the errors
287    /// and resize the coefficients to match.
288    fn set_errors(&mut self, errors: Vec<f64>) -> Result<(), ComponentError> {
289        for error in &errors {
290            non_negative_error("power-log-polynomial", *error)?;
291        }
292        self.coefficients.resize(errors.len(), 0.0);
293        self.errors = errors;
294        Ok(())
295    }
296
297    /// The coefficients c₀, c₁, … (`coeffs`).
298    pub fn coefficients(&self) -> &[f64] {
299        &self.coefficients
300    }
301
302    /// The coefficients' errors.
303    pub fn errors(&self) -> &[f64] {
304        &self.errors
305    }
306}

308/// What each spectral model adds to the reference frequency.
309#[derive(Debug, Clone, PartialEq)]
310pub enum SpectralKind {
311    /// A flat spectrum.
312    Constant,
313    /// A power law (spectral index).
314    SpectralIndex(SpectralIndex),
315    /// A tabulated spectrum.
316    Tabular(TabularSpectrum),
317    /// A power law with a log-polynomial exponent.
318    PowerLogPolynomial(PowerLogPolynomial),
319}
320
321impl SpectralKind {
322    /// The spectral model's type.
323    pub fn spectral_type(&self) -> SpectralType {
324        match self {
325            Self::Constant => SpectralType::Constant,
326            Self::SpectralIndex(_) => SpectralType::SpectralIndex,
327            Self::Tabular(_) => SpectralType::Tabular,
328            Self::PowerLogPolynomial(_) => SpectralType::PowerLogPolynomial,
329        }
330    }
331
332    /// The model CASA's default constructor of `spectral_type` makes, before
333    /// a table row's values are applied.
334    pub(crate) fn casa_default(spectral_type: SpectralType) -> Self {
335        match spectral_type {
336            SpectralType::Constant => Self::Constant,
337            SpectralType::SpectralIndex => Self::SpectralIndex(SpectralIndex::new(0.0)),
338            SpectralType::Tabular => Self::Tabular(TabularSpectrum::empty()),
339            SpectralType::PowerLogPolynomial => {
340                Self::PowerLogPolynomial(PowerLogPolynomial::new(Vec::new()))
341            }
342        }
343    }
344}

346/// A component's spectral model: its reference frequency, that frequency's
347/// error, and what the model adds ([`SpectralKind`]).
348///
349/// This is CASA's `SpectralModel` with its subclasses folded into
350/// [`SpectralKind`]. The reference frequency is a casa-types [`MFrequency`]
351/// carrying its frame; a component list stores it in that frame
352/// (`Frequency_Frame`), unconverted. The error is a quantity whose unit the
353/// table stores too (`Frequency_Error_Units`); CASA's default is 0 GHz.
354#[derive(Debug, Clone)]
355pub struct SpectralModel {
356    reference_frequency: MFrequency,
357    reference_frequency_error: Quantity,
358    kind: SpectralKind,
359}
360
361impl SpectralModel {
362    /// A model at `reference_frequency` with a zero error (0 GHz, CASA's
363    /// default).
364    pub fn new(reference_frequency: MFrequency, kind: SpectralKind) -> Self {
365        Self {
366            reference_frequency,
367            reference_frequency_error: zero_frequency_error(),
368            kind,
369        }
370    }
371
372    /// A flat spectrum at `reference_frequency`.
373    pub fn constant(reference_frequency: MFrequency) -> Self {
374        Self::new(reference_frequency, SpectralKind::Constant)
375    }
376
377    /// This model with its reference frequency's error replaced
378    /// (`setRefFrequencyError`).
379    ///
380    /// # Errors
381    ///
382    /// [`ComponentError::BadError`] when the error is negative or not a
383    /// frequency, which CASA refuses.
384    pub fn with_reference_frequency_error(
385        mut self,
386        error: Quantity,
387    ) -> Result<Self, ComponentError> {
388        self.set_reference_frequency_error(error)?;
389        Ok(self)
390    }
391
392    /// The reference frequency, in its own frame (`refFrequency`).
393    pub fn reference_frequency(&self) -> &MFrequency {
394        &self.reference_frequency
395    }
396
397    /// Replace the reference frequency (`setRefFrequency`).
398    pub fn set_reference_frequency(&mut self, reference_frequency: MFrequency) {
399        self.reference_frequency = reference_frequency;
400    }
401
402    /// The reference frequency's error (`refFrequencyError`).
403    pub fn reference_frequency_error(&self) -> &Quantity {
404        &self.reference_frequency_error
405    }
406
407    /// Replace the reference frequency's error (`setRefFrequencyError`).
408    ///
409    /// # Errors
410    ///
411    /// [`ComponentError::BadError`] when the error is negative or not a
412    /// frequency (`SpectralModel::badError`).
413    pub fn set_reference_frequency_error(&mut self, error: Quantity) -> Result<(), ComponentError> {
414        let hz = Unit::new("Hz").expect("Hz is a unit");
415        if !error.unit().conformant(&hz) || error.value() < 0.0 {
416            return Err(ComponentError::BadError {
417                what: "reference frequency error",
418                kind: "frequency",
419                value: error.value(),
420                unit: error.unit().name().to_string(),
421            });
422        }
423        self.reference_frequency_error = error;
424        Ok(())
425    }
426
427    /// What the model adds to the reference frequency.
428    pub fn kind(&self) -> &SpectralKind {
429        &self.kind
430    }
431
432    /// Mutable access to what the model adds.
433    pub fn kind_mut(&mut self) -> &mut SpectralKind {
434        &mut self.kind
435    }
436
437    /// The model's type.
438    pub fn spectral_type(&self) -> SpectralType {
439        self.kind.spectral_type()
440    }
441
442    /// The parameters, laid out as CASA's `parameters()`.
443    pub fn parameters(&self) -> Vec<f64> {
444        match &self.kind {
445            SpectralKind::Constant | SpectralKind::Tabular(_) => Vec::new(),
446            SpectralKind::SpectralIndex(index) => vec![index.index],
447            SpectralKind::PowerLogPolynomial(polynomial) => polynomial.coefficients.clone(),
448        }
449    }
450
451    /// The parameters' errors, laid out as CASA's `errors()`; a constant
452    /// spectrum reports one zero.
453    pub fn errors(&self) -> Vec<f64> {
454        match &self.kind {
455            SpectralKind::Constant => vec![0.0],
456            SpectralKind::Tabular(_) => Vec::new(),
457            SpectralKind::SpectralIndex(index) => vec![index.error],
458            SpectralKind::PowerLogPolynomial(polynomial) => polynomial.errors.clone(),
459        }
460    }
461
462    /// Apply a parameter vector as CASA's `setParameters` does.
463    ///
464    /// A constant spectrum ignores it; a spectral index takes α from its first
465    /// value; a power-log polynomial takes all of them as coefficients,
466    /// keeping its errors for the coefficients it still has and zero for new
467    /// ones.
468    ///
469    /// # Errors
470    ///
471    /// [`ComponentError::ParameterCount`] for a spectral index without a
472    /// value or a tabular spectrum with any (CASA asserts it has none).
473    pub fn set_parameters(&mut self, parameters: &[f64]) -> Result<(), ComponentError> {
474        match &mut self.kind {
475            SpectralKind::Constant => {}
476            SpectralKind::SpectralIndex(index) => {
477                index.index = *parameters.first().ok_or(ComponentError::ParameterCount {
478                    what: "spectral index parameters",
479                    expected: 1,
480                    actual: 0,
481                })?;
482            }
483            SpectralKind::Tabular(_) => {
484                if !parameters.is_empty() {
485                    return Err(ComponentError::ParameterCount {
486                        what: "tabular spectrum parameters",
487                        expected: 0,
488                        actual: parameters.len(),
489                    });
490                }
491            }
492            SpectralKind::PowerLogPolynomial(polynomial) => {
493                polynomial.coefficients = parameters.to_vec();
494                polynomial.errors.resize(parameters.len(), 0.0);
495            }
496        }
497        Ok(())
498    }
499
500    /// Apply an error vector as CASA's `setErrors` does.
501    ///
502    /// A constant spectrum ignores it; a spectral index takes α's error from
503    /// its first value; a power-log polynomial takes all of them and resizes
504    /// its coefficients to match, keeping those it still has and zero for
505    /// new ones (`PowerLogPoly::setErrors`).
506    ///
507    /// # Errors
508    ///
509    /// [`ComponentError::Spectrum`] for a negative spectral-index or
510    /// power-log-polynomial error, and [`ComponentError::ParameterCount`] for
511    /// a spectral index without a value or a tabular spectrum with any.
512    pub fn set_errors(&mut self, errors: &[f64]) -> Result<(), ComponentError> {
513        match &mut self.kind {
514            SpectralKind::Constant => {}
515            SpectralKind::SpectralIndex(index) => {
516                let error = *errors.first().ok_or(ComponentError::ParameterCount {
517                    what: "spectral index errors",
518                    expected: 1,
519                    actual: 0,
520                })?;
521                index.error = non_negative_error("spectral index", error)?;
522            }
523            SpectralKind::Tabular(_) => {
524                if !errors.is_empty() {
525                    return Err(ComponentError::ParameterCount {
526                        what: "tabular spectrum errors",
527                        expected: 0,
528                        actual: errors.len(),
529                    });
530                }
531            }
532            SpectralKind::PowerLogPolynomial(polynomial) => {
533                polynomial.set_errors(errors.to_vec())?
534            }
535        }
536        Ok(())
537    }
538}
539
540impl PartialEq for SpectralModel {
541    fn eq(&self, other: &Self) -> bool {
542        self.reference_frequency.refer() == other.reference_frequency.refer()
543            && self.reference_frequency.hz() == other.reference_frequency.hz()
544            && self.reference_frequency_error == other.reference_frequency_error
545            && self.kind == other.kind
546    }
547}

549/// CASA's default reference-frequency error, 0 GHz.
550pub(crate) fn zero_frequency_error() -> Quantity {
551    Quantity::with_unit(0.0, Unit::new("GHz").expect("GHz is a unit"))
552}
553
554fn non_negative_error(spectrum: &'static str, error: f64) -> Result<f64, ComponentError> {
555    if error < 0.0 {
556        Err(ComponentError::Spectrum {
557            spectrum,
558            reason: format!("the errors must be non-negative, got {error}"),
559        })
560    } else {
561        Ok(error)
562    }
563}

565#[cfg(test)]
566mod tests {
567    use super::*;
568
569    #[test]
570    fn power_log_polynomial_errors_resize_the_coefficients() {
571        // `PowerLogPoly::setErrors` resizes the coefficients to the errors'
572        // length, keeping those it has; casacore's resize zeroes new ones.
573        let mut model = SpectralModel::new(
574            MFrequency::new(1.4e9, FrequencyRef::LSRK),
575            SpectralKind::PowerLogPolynomial(PowerLogPolynomial::new(vec![1.0, 2.0, 3.0])),
576        );
577        model.set_errors(&[0.1, 0.2]).expect("setErrors");
578        assert_eq!(model.parameters(), [1.0, 2.0]);
579        assert_eq!(model.errors(), [0.1, 0.2]);
580        model.set_errors(&[0.1, 0.2, 0.3]).expect("setErrors");
581        assert_eq!(model.parameters(), [1.0, 2.0, 0.0]);
582        assert_eq!(model.errors(), [0.1, 0.2, 0.3]);
583        // A negative error is refused before anything changes.
584        assert!(model.set_errors(&[0.1, -0.2]).is_err());
585        assert_eq!(model.parameters(), [1.0, 2.0, 0.0]);
586        assert_eq!(model.errors(), [0.1, 0.2, 0.3]);
587
588        let polynomial = PowerLogPolynomial::new(vec![1.0, 2.0, 3.0])
589            .with_errors(vec![0.1, 0.2])
590            .expect("setErrors");
591        assert_eq!(polynomial.coefficients(), [1.0, 2.0]);
592        assert_eq!(polynomial.errors(), [0.1, 0.2]);
593    }
