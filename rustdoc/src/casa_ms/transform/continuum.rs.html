13/// Weighted polynomial fitting for the public mstransform continuum stage.
14#[derive(Debug, Clone)]
15pub struct ContinuumFitOptions {
16    /// Input SPWs and line-free channels. Empty fits all selected channels.
17    pub fit_spw: String,
18    /// Desired polynomial order, reduced when there are fewer usable channels.
19    pub order: usize,
20    /// Write the fitted continuum in place of the residuals.
21    pub want_continuum: bool,
22    /// Use CASA's iterative reweighting mode. The legacy fit uses one pass.
23    pub reweight: bool,
24    /// Number of least-squares passes when iterative reweighting is active.
25    pub iterations: usize,
26    /// Maximum number of distinct rows fitted concurrently.
27    pub threads: usize,
28}

41impl ContinuumFit {
42    pub(super) fn new(
43        ms: &MeasurementSet,
44        selected: &BTreeMap<i32, TransformChannelSelection>,
45        options: &ContinuumFitOptions,
46    ) -> Result<Self, MsTransformError> {
47        let spec = options.fit_spw.trim();
48        let mut windows = BTreeMap::new();
49        if spec == "NONE" {
50            return Ok(Self {
51                windows,
52                options: options.clone(),
53            });
54        }
55        let fit_channels = (!spec.is_empty())
56            .then(|| {
57                let spws = ms.resolve_spw_selector(spec).map_err(|source| {
58                    MsTransformError::Spectrum(format!("invalid fitspw {spec:?}: {source}"))
59                })?;
60                super::resolve_transform_channels(ms, &spws, spec).map_err(|source| {
61                    MsTransformError::Spectrum(format!("invalid fitspw {spec:?}: {source}"))
62                })
63            })
64            .transpose()?;
65        let spectral_windows = ms.spectral_window().map_err(spectrum_error)?;
66        for (&id, window) in selected {
67            let admissible = if let Some(fit_channels) = &fit_channels {
68                let Some(fit) = fit_channels.get(&id) else {
69                    continue;
70                };
71                Some(
72                    fit.channels
73                        .iter()
74                        .copied()
75                        .collect::<std::collections::BTreeSet<_>>(),
76                )
77            } else {
78                None
79            };
80            let full_frequencies = spectral_windows
81                .chan_freq(id as usize)
82                .map_err(spectrum_error)?;
83            let frequencies = window
84                .channels
85                .iter()
86                .map(|&channel| full_frequencies.get(channel).copied())
87                .collect::<Option<Vec<_>>>()
88                .ok_or_else(|| {
89                    MsTransformError::Spectrum(format!("SPW {id} has a truncated CHAN_FREQ array"))
90                })?;
91            let (minimum, maximum) = frequencies.iter().copied().fold(
92                (f64::INFINITY, f64::NEG_INFINITY),
93                |(low, high), frequency| (low.min(frequency), high.max(frequency)),
94            );
95            let midpoint = (minimum + maximum) * 0.5;
96            let frequencies = frequencies
97                .into_iter()
98                .map(|frequency| {
99                    if minimum == maximum {
100                        0.0
101                    } else {
102                        (frequency - midpoint) / (minimum - midpoint)
103                    }
104                })
105                .collect();
106            windows.insert(
107                id,
108                FitWindow {
109                    channels: window.channels.clone(),
110                    frequencies,
111                    fit_channels: window
112                        .channels
113                        .iter()
114                        .map(|channel| admissible.as_ref().is_none_or(|fit| fit.contains(channel)))
115                        .collect(),
116                },
117            );
118        }
119        Ok(Self {
120            windows,
121            options: options.clone(),
122        })
123    }
124
125    pub(super) fn working_bytes_per_sample(&self) -> usize {
126        let terms = self
127            .windows
128            .values()
129            .map(|window| window.channels.len())
130            .max()
131            .unwrap_or(0)
132            .min(self.options.order.saturating_add(1));
133        // Per-channel bases, weighted design, SVD factors, and row vectors
134        // can coexist. The SVD's square factor has at most this many terms.
135        if terms == 0 {
136            0
137        } else {
138            64_usize.saturating_add(48_usize.saturating_mul(terms))
139        }
140    }

142    pub(super) fn apply(
143        &self,
144        block: &mut MainArrayBlock,
145        visibility: &VisibilityColumns,
146        spws: impl Iterator<Item = i32>,
147        spectrum_names: &[&str],
148        projected: bool,
149    ) -> Result<(), MsTransformError> {
150        if self.windows.is_empty() {
151            return Ok(());
152        }
153        let windows = spws.map(|id| self.windows.get(&id)).collect::<Vec<_>>();
154        let workers = self
155            .options
156            .threads
157            .max(1)
158            .min(std::thread::available_parallelism().map_or(1, std::num::NonZeroUsize::get));
159        for (copy, cells) in visibility.copies().iter().zip(&mut block.visibility) {
160            if copy.source == OptionalMainColumn::LagData {
161                continue;
162            }
163            let sigma = copy.source == OptionalMainColumn::Data;
164            let (name, scalars) = if sigma {
165                ("SIGMA_SPECTRUM", &block.sigmas)
166            } else {
167                ("WEIGHT_SPECTRUM", &block.weights)
168            };
169            let spectra = spectrum_names
170                .iter()
171                .position(|&candidate| candidate == name)
172                .map(|slot| &block.spectra[slot]);
173            let flags = &block.flags;
174            let fit = |start: usize, cells: &mut [Option<ArrayValue>]| {
175                for (offset, cell) in cells.iter_mut().enumerate() {
176                    let row = start + offset;
177                    if let (Some(window), Some(values), Some(flags)) =
178                        (windows[row], cell, &flags[row])
179                    {
180                        let noise = Noise {
181                            spectrum: spectra.and_then(|cells| cells[row].as_ref()),
182                            scalar: scalars[row].as_ref(),
183                            sigma,
184                        };
185                        window.apply(values, flags, noise, &self.options, projected)?;
186                    }
187                }
188                Ok(())
189            };
190            if workers == 1 || cells.len() < 2 {
191                fit(0, cells)?;
192            } else {
193                let size = cells.len().div_ceil(workers);
194                std::thread::scope(|scope| {
195                    let handles = cells
196                        .chunks_mut(size)
197                        .enumerate()
198                        .map(|(index, cells)| scope.spawn(move || fit(index * size, cells)))
199                        .collect::<Vec<_>>();
200                    for handle in handles {
201                        handle.join().map_err(|_| {
202                            MsTransformError::Spectrum("continuum fit worker panicked".into())
203                        })??;
204                    }
205                    Ok::<_, MsTransformError>(())
206                })?;
207            }
208        }
209        Ok(())
210    }
211}

213#[derive(Clone, Copy)]
214struct Noise<'a> {
215    spectrum: Option<&'a ArrayValue>,
216    scalar: Option<&'a ArrayValue>,
217    sigma: bool,
218}
219
220impl Noise<'_> {
221    fn weight(self, correlation: usize, channel: usize) -> f64 {
222        let value = match self.spectrum {
223            Some(ArrayValue::Float32(spectrum)) => spectrum.get([correlation, channel]),
224            _ => match self.scalar {
225                Some(ArrayValue::Float32(scalar)) => scalar.get([correlation]),
226                _ => None,
227            },
228        }
229        .copied()
230        .unwrap_or(0.0);
231        if self.sigma && value > 0.0 {
232            f64::from(1.0 / (value * value))
233        } else if !self.sigma {
234            f64::from(value)
235        } else {
236            0.0
237        }
238    }
239}

241impl FitWindow {
242    fn apply(
243        &self,
244        visibility: &mut ArrayValue,
245        flags: &ArrayValue,
246        noise: Noise<'_>,
247        options: &ContinuumFitOptions,
248        projected: bool,
249    ) -> Result<(), MsTransformError> {
250        let ArrayValue::Bool(flags) = flags else {
251            return Err(MsTransformError::Spectrum(
252                "continuum fitting requires a boolean FLAG array".into(),
253            ));
254        };
255        match visibility {
256            ArrayValue::Complex32(values) => self.fit_rows(
257                values,
258                flags,
259                noise,
260                options,
261                projected,
262                |v| v.re,
263                |v| v.im,
264                Complex32::new,
265            ),
266            ArrayValue::Float32(values) => self.fit_rows(
267                values,
268                flags,
269                noise,
270                options,
271                projected,
272                |v| v,
273                |_| 0.0,
274                |real, _| real,
275            ),
276            _ => Err(MsTransformError::Spectrum(
277                "continuum fitting requires complex or float visibilities".into(),
278            )),
279        }
280    }
281
282    #[allow(clippy::too_many_arguments)]
283    fn fit_rows<T: Copy>(
284        &self,
285        values: &mut ArrayD<T>,
286        flags: &ArrayD<bool>,
287        noise: Noise<'_>,
288        options: &ContinuumFitOptions,
289        projected: bool,
290        real: impl Fn(T) -> f32,
291        imaginary: impl Fn(T) -> f32,
292        assemble: impl Fn(f32, f32) -> T,
293    ) -> Result<(), MsTransformError> {
294        let mut values = values
295            .view_mut()
296            .into_dimensionality::<Ix2>()
297            .map_err(spectrum_error)?;
298        let flags = flags
299            .view()
300            .into_dimensionality::<Ix2>()
301            .map_err(spectrum_error)?;
302        if values.shape() != flags.shape() {
303            return Err(MsTransformError::Spectrum(
304                "continuum visibility and FLAG shapes differ".into(),
305            ));
306        }
307        let index = |position| {
308            if projected {
309                position
310            } else {
311                self.channels[position]
312            }
313        };
314        if (0..self.channels.len()).any(|position| index(position) >= values.ncols()) {
315            return Err(MsTransformError::Spectrum(
316                "continuum channel is outside the visibility array".into(),
317            ));
318        }
319        let iterations = if options.reweight {
320            options.iterations
321        } else {
322            1
323        };
324        for correlation in 0..values.nrows() {
325            let mut spectrum = values.slice_mut(s![correlation, ..]);
326            let admissible = self
327                .fit_channels
328                .iter()
329                .enumerate()
330                .filter_map(|(position, &include)| {
331                    let channel = index(position);
332                    let weight = noise.weight(correlation, channel);
333                    (include
334                        && !flags[[correlation, channel]]
335                        && weight.is_finite()
336                        && weight > 0.0)
337                        .then_some((position, weight))
338                })
339                .collect::<Vec<_>>();
340            let terms = admissible.len().min(options.order.saturating_add(1));
341            let fit = |component: &dyn Fn(T) -> f32| {
342                let data = |position| f64::from(component(spectrum[index(position)]));
343                polynomial_fit(&admissible, &self.frequencies, data, terms, iterations)
344            };
345            let real_fit = fit(&real);
346            let imaginary_fit = fit(&imaginary);
347            for (position, &frequency) in self.frequencies.iter().enumerate() {
348                let sample = &mut spectrum[index(position)];
349                let estimate = |coefficients: &[f64]| evaluate(coefficients, frequency) as f32;
350                let re = estimate(&real_fit);
351                let im = estimate(&imaginary_fit);
352                *sample = if options.want_continuum {
353                    assemble(re, im)
354                } else {
355                    assemble(real(*sample) - re, imaginary(*sample) - im)
356                };
357            }
358        }
359        Ok(())
360    }
361}

363fn polynomial_fit(
364    admitted: &[(usize, f64)],
365    frequencies: &[f64],
366    value: impl Fn(usize) -> f64,
367    terms: usize,
368    iterations: usize,
369) -> Vec<f64> {
370    if terms == 0 || iterations == 0 {
371        return Vec::new();
372    }
373    // A common weight scale leaves the fit unchanged. Normalize before
374    // applying the shared solver's absolute singular-value cutoff.
375    let weight_scale = admitted
376        .iter()
377        .map(|&(_, weight)| weight)
378        .fold(0.0, f64::max);
379    let mut rows = admitted
380        .iter()
381        .map(|&(position, weight)| {
382            let coordinate = frequencies[position];
383            let basis = std::iter::successors(Some(1.0), move |power| Some(power * coordinate))
384                .take(terms)
385                .collect::<Vec<_>>();
386            (basis, value(position), weight / weight_scale)
387        })
388        .collect::<Vec<_>>();
389    let scale = rows
390        .iter()
391        .map(|(_, value, _)| value.abs())
392        .fold(0.0, f64::max);
393    let mut coefficients = Vec::new();
394    for pass in 0..iterations {
395        coefficients = solve_weighted_least_squares(&rows, terms).unwrap_or_default();
396        if pass + 1 == iterations {
397            break;
398        }
399        let mut largest = 0.0_f64;
400        for ((_, observed, weight), &(position, _)) in rows.iter_mut().zip(admitted) {
401            let residual = if *weight > f64::MIN_POSITIVE {
402                (*observed - evaluate(&coefficients, frequencies[position])).abs()
403            } else {
404                0.0
405            };
406            largest = largest.max(residual);
407            *weight = residual;
408        }
409        if largest <= scale * 64.0 * f64::EPSILON {
410            break;
411        }
412        for (_, _, weight) in &mut rows {
413            *weight = if *weight > 0.0 && *weight < largest {
414                (largest - *weight) / largest
415            } else {
416                // The shared SVD solver requires a positive weight. Using
417                // the smallest normal number retains the zero-weight rows
418                // for rank-deficient fits without moving a finite solution.
419                f64::MIN_POSITIVE
420            };
421        }
422    }
423    coefficients
424}

426pub(super) fn row_flag(flags: Option<&ArrayValue>, channels: Option<&[usize]>) -> bool {
427    let Some(ArrayValue::Bool(flags)) = flags else {
428        return false;
429    };
430    let Some(channels) = channels else {
431        return flags.iter().all(|&flag| flag);
432    };
433    let Ok(flags) = flags.view().into_dimensionality::<Ix2>() else {
434        return false;
435    };
436    flags
437        .rows()
438        .into_iter()
439        .all(|row| channels.iter().all(|&channel| row[channel]))
440}

442/// FreqAxisTVI reports the mean of its spectrum over unflagged channels.
443/// When an entire correlation is flagged, use the mean of all its channels.
444pub(super) fn mean_noise(
445    scalar: &mut ArrayValue,
446    spectrum: Option<&ArrayValue>,
447    flags: &ArrayValue,
448    channels: Option<&[usize]>,
449) {
450    let (Some(ArrayValue::Float32(spectrum)), ArrayValue::Bool(flags)) = (spectrum, flags) else {
451        return;
452    };
453    let (Ok(spectrum), Ok(flags)) = (
454        spectrum.view().into_dimensionality::<Ix2>(),
455        flags.view().into_dimensionality::<Ix2>(),
456    ) else {
457        return;
458    };
459    if spectrum.shape() != flags.shape() {
460        return;
461    }
462    let channel_count = channels.map_or(spectrum.ncols(), <[usize]>::len);
463    let means = Array1::from_shape_fn(spectrum.nrows(), |correlation| {
464        let (mut full, mut good, mut count) = (0.0_f32, 0.0_f32, 0);
465        for index in 0..channel_count {
466            let channel = channels.map_or(index, |channels| channels[index]);
467            let value = spectrum[[correlation, channel]];
468            full += value;
469            if !flags[[correlation, channel]] {
470                good += value;
471                count += 1;
472            }
473        }
474        if count == 0 {
475            full / channel_count.max(1) as f32
476        } else {
477            good / count as f32
478        }
479    });
480    *scalar = ArrayValue::Float32(means.into_dyn());
481}
482
483fn evaluate(coefficients: &[f64], x: f64) -> f64 {
484    coefficients
485        .iter()
486        .rev()
487        .fold(0.0, |sum, coefficient| sum * x + coefficient)
488}
