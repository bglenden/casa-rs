5pub(super) fn channel_edges_from_centers(all_source_frequencies_hz: &[f64]) -> MsResult<Vec<f64>> {
6    if all_source_frequencies_hz.is_empty() {
7        return Err(MsError::VersionError(
8            "cannot build cube channel axis from an empty SPW".to_string(),
9        ));
10    }
11    if all_source_frequencies_hz.len() == 1 {
12        let center = all_source_frequencies_hz[0];
13        return Ok(vec![center - 0.5, center + 0.5]);
14    }
15
16    let mut edges = Vec::with_capacity(all_source_frequencies_hz.len() + 1);
17    let first_step = all_source_frequencies_hz[1] - all_source_frequencies_hz[0];
18    edges.push(all_source_frequencies_hz[0] - first_step / 2.0);
19    for pair in all_source_frequencies_hz.windows(2) {
20        edges.push(0.5 * (pair[0] + pair[1]));
21    }
22    let last = all_source_frequencies_hz.len() - 1;
23    let last_step = all_source_frequencies_hz[last] - all_source_frequencies_hz[last - 1];
24    edges.push(all_source_frequencies_hz[last] + last_step / 2.0);
25    Ok(edges)
26}

28pub(super) fn channel_widths_from_centers(channel_frequencies_hz: &[f64]) -> MsResult<Vec<f64>> {
29    let edges = channel_edges_from_centers(channel_frequencies_hz)?;
30    Ok(edges
31        .windows(2)
32        .map(|pair| (pair[1] - pair[0]).abs())
33        .collect())
34}

36pub(super) fn channel_mode_single_output_width_hz(
37    source_frequencies_hz: &[f64],
38    source_channel_widths_hz: &[f64],
39    start_channel: i32,
40    width_channels: i32,
41) -> MsResult<f64> {
42    if width_channels == 0 {
43        return Err(MsError::VersionError(
44            "cube channel width must not be zero".to_string(),
45        ));
46    }
47    let nchannels = i32::try_from(source_frequencies_hz.len()).map_err(|_| {
48        MsError::VersionError("spectral window channel count exceeds i32".to_string())
49    })?;
50    if source_channel_widths_hz.len() != source_frequencies_hz.len() {
51        return Err(MsError::VersionError(format!(
52            "channel-mode cube axis requires matching frequency/width arrays, got {} and {}",
53            source_frequencies_hz.len(),
54            source_channel_widths_hz.len()
55        )));
56    }
57    let (low, high) = if width_channels > 0 {
58        let low = start_channel;
59        let high = low + width_channels;
60        (low, high)
61    } else {
62        let high = start_channel + 1;
63        let low = high + width_channels;
64        (low, high)
65    };
66    if low < 0 || high < 0 || low >= nchannels || high > nchannels || low >= high {
67        return Err(MsError::VersionError(format!(
68            "cube channel axis bin [{low}, {high}) is outside the SPW with {nchannels} channels"
69        )));
70    }
71    let selected = usize::try_from(low).expect("validated low bound")
72        ..usize::try_from(high).expect("validated high bound");
73    let low_edge_hz = selected
74        .clone()
75        .map(|index| source_frequencies_hz[index] - source_channel_widths_hz[index].abs() / 2.0)
76        .reduce(f64::min)
77        .expect("validated non-empty channel range");
78    let high_edge_hz = selected
79        .map(|index| source_frequencies_hz[index] + source_channel_widths_hz[index].abs() / 2.0)
80        .reduce(f64::max)
81        .expect("validated non-empty channel range");
82    Ok((high_edge_hz - low_edge_hz).abs())
83}

102            .map(|width_hz| width_hz.abs())
103            .collect()),
104    }

107/// CASA's channel-mode `chanFreq` vector: the regridder's output channel
108/// centres, ordered as `SynthesisParamsImage::getImFreq` leaves them after
109/// its final reversal.
110///
111/// This is a step-for-step port of `MSTransformRegridder::regridChanBounds`
112/// and the `(lo + hi)/2` centres of `MSTransformRegridder::calcChanFreqs`.
113/// These are not the frequencies CASA grids on:
114/// [`casa_image_spectral_axis_hz`] turns them into the image spectral axis.
115/// An unset `nchan` (`-1`) counts the output channels as `regridChanBounds`
116/// does when only the start is set: the span from the start channel's edge
117/// to the band's far edge in whole output widths, with a 1% tolerance.
118pub(super) fn casa_transformed_channel_mode_output_centers(
119    transformed_source_frequencies_hz: &[f64],
120    transformed_source_channel_widths_hz: &[f64],
121    start_channel: i32,
122    width_channels: i32,
123    nchan: Option<usize>,
124) -> MsResult<Vec<f64>> {
125    if width_channels == 0 {
126        return Err(MsError::VersionError(
127            "cube channel width must not be zero".to_string(),
128        ));
129    }
130    if transformed_source_frequencies_hz.len() != transformed_source_channel_widths_hz.len() {
131        return Err(MsError::VersionError(format!(
132            "CASA transformed cube axis requires matching frequency/width arrays, got {} and {}",
133            transformed_source_frequencies_hz.len(),
134            transformed_source_channel_widths_hz.len()
135        )));
136    }
137    if transformed_source_frequencies_hz.is_empty() {
138        return Err(MsError::VersionError(
139            "cannot build CASA transformed cube axis from an empty SPW".to_string(),
140        ));
141    }
142
143    let nchannels = i32::try_from(transformed_source_frequencies_hz.len()).map_err(|_| {
144        MsError::VersionError("spectral window channel count exceeds i32".to_string())
145    })?;
146    let mut transformed_frequencies_hz = transformed_source_frequencies_hz.to_vec();
147    let mut transformed_widths_hz = transformed_source_channel_widths_hz
148        .iter()
149        .map(|width_hz| width_hz.abs())
150        .collect::<Vec<_>>();
151    let mut first_channel = start_channel;
152    let mut start_is_end = width_channels < 0;
153    let descending = transformed_frequencies_hz.first() > transformed_frequencies_hz.last();
154
155    if descending {
156        transformed_frequencies_hz.reverse();
157        transformed_widths_hz.reverse();
158        first_channel = nchannels - 1 - first_channel;
159        start_is_end = !start_is_end;
160    }
161
162    if first_channel < 0 || first_channel >= nchannels {
163        return Err(MsError::VersionError(format!(
164            "cube channel start {start_channel} is outside the SPW with {nchannels} channels"
165        )));
166    }
167
168    let output_channel_width_hz =
169        transformed_widths_hz[0] * f64::from(width_channels.unsigned_abs());
170    let first_channel = usize::try_from(first_channel).expect("validated channel index");
171    let nchan = match nchan {
172        Some(nchan) => nchan,
173        None => {
174            let last = transformed_frequencies_hz.len() - 1;
175            let span_hz = if start_is_end {
176                transformed_frequencies_hz[first_channel]
177                    + transformed_widths_hz[first_channel] / 2.0
178                    - (transformed_frequencies_hz[0] - transformed_widths_hz[0] / 2.0)
179            } else {
180                transformed_frequencies_hz[last] + transformed_widths_hz[last] / 2.0
181                    - (transformed_frequencies_hz[first_channel]
182                        - transformed_widths_hz[first_channel] / 2.0)
183            };
184            let count =
185                ((span_hz + output_channel_width_hz * 0.01) / output_channel_width_hz).floor();
186            if !(count >= 1.0 && count < u32::MAX as f64) {
187                return Err(MsError::VersionError(format!(
188                    "no output channel of {output_channel_width_hz} Hz fits from cube channel \
189                     start {start_channel}"
190                )));
191            }
192            count as usize
193        }
194    };
195    let output_bandwidth_hz = output_channel_width_hz * nchan as f64;
196    // CASA `MSTransformRegridder::regridChanBounds` ("freq" regridding)
197    // moves the start edge to the band centre, steps the channel bounds out
198    // from the centre channel by repeated subtraction and addition of the
199    // width, and `calcChanFreqs` takes each centre as `(lo + hi)/2`. The
200    // rounding of the first two centres fixes the image axis, and so
201    // whether an output channel lies exactly on a transformed source
202    // channel, which `FTMachine::interpolateFrequencyTogrid` keeps and a
203    // frequency one ulp below it discards.
204    let centre_hz = if start_is_end {
205        transformed_frequencies_hz[first_channel] + transformed_widths_hz[first_channel] / 2.0
206            - output_bandwidth_hz / 2.0
207    } else {
208        transformed_frequencies_hz[first_channel] - transformed_widths_hz[first_channel] / 2.0
209            + output_bandwidth_hz / 2.0
210    };
211    // An even channel count puts the band centre on the lower edge of the
212    // centre channel, an odd one on its centre.
213    let centre_channel = if nchan % 2 == 1 {
214        (
215            centre_hz - output_channel_width_hz / 2.0,
216            centre_hz + output_channel_width_hz / 2.0,
217        )
218    } else {
219        (centre_hz, centre_hz + output_channel_width_hz)
220    };
221    let below = nchan / 2;
222    let mut bounds_hz = vec![centre_channel; nchan];
223    for index in (0..below).rev() {
224        let high_hz = bounds_hz[index + 1].0;
225        bounds_hz[index] = (high_hz - output_channel_width_hz, high_hz);
226    }
227    for index in below + 1..nchan {
228        let low_hz = bounds_hz[index - 1].1;
229        bounds_hz[index] = (low_hz, low_hz + output_channel_width_hz);
230    }
231    let mut centers_hz = bounds_hz
232        .iter()
233        .map(|(low_hz, high_hz)| (low_hz + high_hz) / 2.0)
234        .collect::<Vec<_>>();
235    if descending ^ (width_channels < 0) {
236        centers_hz.reverse();
237    }
238    Ok(centers_hz)
239}

241/// The image spectral axis CASA builds from its `chanFreq` vector and grids
242/// on.
243///
244/// `SynthesisParamsImage::buildCoordinateSystemCore` makes a linear
245/// `SpectralCoordinate` with `crval = chanFreq[0]`,
246/// `cdelt = chanFreq[1] − chanFreq[0]` and the reference pixel 0 set by
247/// `getImFreq`, and `FTMachine::initMaps` takes plane `k` at
248/// `toWorld(k)`. Only the first two entries of `chan_freqs_hz` survive; the
249/// rest of the axis is [`linear_spectral_axis_hz`], which can differ from
250/// the remaining `chanFreq` entries by a few ulps.
251///
252/// For a non-linear velocity convention (`OPTICAL`/`Z`, `BETA`/`RELATIVISTIC`
253/// or `GAMMA`) `buildCoordinateSystemCore` makes a tabular coordinate from
254/// `chanFreq` instead, which reproduces the table at integer pixels up to the
255/// rounding of its channel corrector; the table is returned unchanged.
256pub(super) fn casa_image_spectral_axis_hz(
257    chan_freqs_hz: Vec<f64>,
258    veltype: DopplerRef,
259) -> Vec<f64> {
260    if casa_tabular_convention(veltype) || chan_freqs_hz.len() < 2 {
261        return chan_freqs_hz;
262    }
263    let reference_hz = chan_freqs_hz[0];
264    linear_spectral_axis_hz(
265        reference_hz,
266        chan_freqs_hz[1] - reference_hz,
267        chan_freqs_hz.len(),
268    )
269}

271/// Whether `buildCoordinateSystemCore` makes a tabular spectral coordinate
272/// for `veltype`: the non-linear conventions `OPTICAL`/`Z`,
273/// `BETA`/`RELATIVISTIC` and `GAMMA`.
274pub(super) fn casa_tabular_convention(veltype: DopplerRef) -> bool {
275    matches!(
276        veltype,
277        DopplerRef::Z | DopplerRef::BETA | DopplerRef::GAMMA
278    )
279}

281/// CASA's frequency- or velocity-mode grid for `axis_config`
282/// (`SynthesisParamsImage::findSpecMode` and
283/// `MSTransformRegridder::convertGridPars`).
284///
285/// A velocity `start` or `width` makes a velocity grid, spaced in optical
286/// velocity for `veltype='optical'` and in radio velocity otherwise, since
287/// `convertGridPars` turns every other convention into radio; a frequency
288/// one makes a frequency grid. Like `SynthesisParamsImage`, this refuses a
289/// `start` and `width` of different kinds. A dimensionless Doppler value is
290/// taken as the velocity it stands for in the grid's convention.
291pub(super) fn casa_regrid_request(
292    axis_config: &CubeAxisConfig,
293    nchan: Option<usize>,
294) -> MsResult<regrid::RegridRequest> {
295    let kind = |value: &Option<CubeAxisValue>| match value {
296        None => None,
297        Some(CubeAxisValue::Channel(_)) => Some("channel"),
298        Some(CubeAxisValue::FrequencyHz { .. }) => Some("frequency"),
299        Some(CubeAxisValue::VelocityMs { .. } | CubeAxisValue::Doppler { .. }) => Some("velocity"),
300    };
301    let quantity = match (kind(&axis_config.start), kind(&axis_config.width)) {
302        (Some(start), Some(width)) if start != width => {
303            return Err(MsError::VersionError(format!(
304                "Cannot mix start and width with different unit types: a {start} start and a \
305                 {width} width"
306            )));
307        }
308        (Some("frequency"), _) | (_, Some("frequency")) => regrid::RegridQuantity::Frequency,
309        (Some("velocity"), _) | (_, Some("velocity")) if axis_config.veltype == DopplerRef::Z => {
310            regrid::RegridQuantity::OpticalVelocity
311        }
312        (Some("velocity"), _) | (_, Some("velocity")) => regrid::RegridQuantity::RadioVelocity,
313        _ => {
314            return Err(MsError::VersionError(
315                "a frequency or velocity cube axis needs a frequency or velocity start or width"
316                    .to_string(),
317            ));
318        }
319    };
320    let convention = match quantity {
321        regrid::RegridQuantity::OpticalVelocity => DopplerRef::Z,
322        _ => DopplerRef::RADIO,
323    };
324    let in_grid_unit = |value: &CubeAxisValue| match value {
325        CubeAxisValue::FrequencyHz { hz, .. } => Ok(*hz),
326        CubeAxisValue::VelocityMs { ms, .. } => Ok(*ms),
327        CubeAxisValue::Doppler {
328            value,
329            convention: from,
330        } => MDoppler::new(*value, *from)
331            .convert_to(convention, &MeasFrame::new())
332            .map(|doppler| doppler.value() * SPEED_OF_LIGHT_M_S)
333            .map_err(|error| {
334                MsError::VersionError(format!(
335                    "convert cube-axis doppler value {value} to {convention}: {error}"
336                ))
337            }),
338        CubeAxisValue::Channel(channel) => Err(MsError::VersionError(format!(
339            "expected a frequency or velocity cube axis value, found channel {channel}"
340        ))),
341    };
342    let rest_frequency_hz = if quantity == regrid::RegridQuantity::Frequency {
343        0.0
344    } else {
345        required_rest_frequency_hz(axis_config.rest_frequency_hz, &axis_config.start)?
346    };
347    Ok(regrid::RegridRequest {
348        quantity,
349        start: axis_config.start.as_ref().map(in_grid_unit).transpose()?,
350        width: axis_config.width.as_ref().map(in_grid_unit).transpose()?,
351        nchan,
352        rest_frequency_hz,
353    })
354}

356/// The `nchan` planes of the image spectral axis
357/// `SynthesisParamsImage::buildCoordinateSystemCore` makes from a
358/// frequency- or velocity-mode `chanFreq` vector; `None` (tclean's
359/// `nchan=-1`) takes as many planes as `chanFreq` holds (`getImFreq`).
360///
361/// For a linear convention it is a linear coordinate with
362/// `crval = chanFreq[0]` and `cdelt = chanFreq[1] − chanFreq[0]`
363/// (`chanFreqStep[0]` for a single channel), however many channels the
364/// regridder made. For a non-linear convention
365/// ([`casa_tabular_convention`]) CASA makes a tabular coordinate over
366/// `chanFreq`, whose header records `chanFreq[0]` and the table's mean step
367/// `(chanFreq[n−1] − chanFreq[0])/(n − 1)`. The optical-velocity grid
368/// retains each centre and extrapolates additional planes by the last
369/// table interval. The other conventions use a uniform frequency grid.
370pub(super) fn casa_regridded_image_axis_hz(
371    channels: &regrid::CasaChannelFrequencies,
372    request: &regrid::RegridRequest,
373    veltype: DopplerRef,
374) -> Vec<f64> {
375    let centres_hz = &channels.centres_hz;
376    let nchan = request.nchan.unwrap_or(centres_hz.len());
377    let first_hz = centres_hz[0];
378    let last = centres_hz.len() - 1;
379    if request.quantity == regrid::RegridQuantity::OpticalVelocity && last > 0 {
380        let step = centres_hz[last] - centres_hz[last - 1];
381        return (0..nchan)
382            .map(|pixel| {
383                if pixel <= last {
384                    centres_hz[pixel]
385                } else {
386                    centres_hz[last] + (pixel - last) as f64 * step
387                }
388            })
389            .collect();
390    }
391    let increment_hz = if last == 0 {
392        channels.steps_hz[0]
393    } else if casa_tabular_convention(veltype) {
394        (centres_hz[last] - first_hz) / last as f64
395    } else {
396        centres_hz[1] - first_hz
397    };
398    linear_spectral_axis_hz(first_hz, increment_hz, nchan)
399}

401/// Frequencies of `nchan` planes on a linear spectral axis whose reference
402/// pixel is 0.
403///
404/// This is casacore `SpectralCoordinate::toWorld` for a linear `FREQ` axis
405/// in Hz: wcslib's `linp2x` forms `cdelt·(k − crpix)` and `spcx2s` adds
406/// `crval`, so plane `k` lies at `reference_hz + k·increment_hz` rounded
407/// after the product and after the sum.
408pub(super) fn linear_spectral_axis_hz(
409    reference_hz: f64,
410    increment_hz: f64,
411    nchan: usize,
412) -> Vec<f64> {
413    (0..nchan)
414        .map(|pixel| reference_hz + pixel as f64 * increment_hz)
415        .collect()
416}

418/// Resolve a simple contiguous source-channel selection.
419pub fn resolve_contiguous_channel_selection(
420    all_frequencies_hz: &[f64],
421    start: Option<usize>,
422    count: Option<usize>,
423) -> MsResult<ResolvedChannelSelection> {
424    let start = start.unwrap_or(0);
425    if start >= all_frequencies_hz.len() {
426        return Err(MsError::VersionError(format!(
427            "channel start {start} is outside spectral window with {} channels",
428            all_frequencies_hz.len()
429        )));
430    }
431    let count = count.unwrap_or(all_frequencies_hz.len() - start);
432    if count == 0 || start + count > all_frequencies_hz.len() {
433        return Err(MsError::VersionError(format!(
434            "channel selection [{start}, {}) is outside spectral window with {} channels",
435            start + count,
436            all_frequencies_hz.len()
437        )));
438    }
439    Ok(ResolvedChannelSelection {
440        indices: (start..start + count).collect(),
441        frequencies_hz: all_frequencies_hz[start..start + count].to_vec(),
442    })
443}

445/// Resolve an explicit CASA channel selector such as `4~9;12~14` or `0~10^2`.
446pub fn resolve_channel_selector_selection(
447    all_frequencies_hz: &[f64],
448    selector: &ChannelSelection,
449) -> MsResult<ResolvedChannelSelection> {
450    let indices = selector.indices(all_frequencies_hz.len())?;
451    if indices.is_empty() {
452        return Err(MsError::VersionError(
453            "channel selector resolved to zero channels".to_string(),
454        ));
455    }
456    Ok(ResolvedChannelSelection {
457        frequencies_hz: indices
458            .iter()
459            .map(|&index| all_frequencies_hz[index])
460            .collect(),
461        indices,
462    })
