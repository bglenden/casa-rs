4use std::fmt::Debug;

6/// Immutable identity and retained residency of one prepared Measures provider.
7///
8/// The identity must commit every scientific value exposed by the provider
9/// after preparation. A provider entering a bounded operation must eagerly
10/// stabilize all caches first, then return the same state on every later call.
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
12pub struct MeasuresProviderState {
13    identity_sha256: [u8; 32],
14    retained_heap_bytes: usize,
15}
16
17impl MeasuresProviderState {
18    /// Construct a provider-owned state from its scientific-content digest and
19    /// exact retained heap residency.
20    #[must_use]
21    pub const fn new(identity_sha256: [u8; 32], retained_heap_bytes: usize) -> Self {
22        Self {
23            identity_sha256,
24            retained_heap_bytes,
25        }
26    }
27
28    /// Return the authoritative SHA-256 identity of the prepared scientific state.
29    #[must_use]
30    pub const fn identity_sha256(self) -> [u8; 32] {
31        self.identity_sha256
32    }
33
34    /// Return exact heap bytes retained beyond the concrete provider object.
35    #[must_use]
36    pub const fn retained_heap_bytes(self) -> usize {
37        self.retained_heap_bytes
38    }
39}

41/// Interpolated Earth-orientation values at one UTC epoch.
42#[derive(Debug, Clone, Copy, PartialEq)]
43pub struct EopValues {
44    /// UT1 minus UTC, in seconds.
45    pub dut1_seconds: f64,
46    /// Polar-motion X component, in arcseconds.
47    pub x_arcsec: f64,
48    /// Polar-motion Y component, in arcseconds.
49    pub y_arcsec: f64,
50    /// Celestial-pole dX offset, in milliarcseconds.
51    pub dx_mas: f64,
52    /// Celestial-pole dY offset, in milliarcseconds.
53    pub dy_mas: f64,
54    /// Whether either interpolation endpoint is predicted.
55    pub is_predicted: bool,
56}

58/// Observatory coordinates returned by a measures-data provider.
59#[derive(Debug, Clone, PartialEq)]
60pub enum ObservatoryPosition {
61    /// Geocentric ITRF coordinates, in metres.
62    Itrf { x_m: f64, y_m: f64, z_m: f64 },
63    /// WGS84 geodetic coordinates.
64    Wgs84 {
65        longitude_rad: f64,
66        latitude_rad: f64,
67        height_m: f64,
68    },
69}

71/// Fixed named-source direction returned by a measures-data provider.
72#[derive(Debug, Clone, PartialEq)]

82/// Pure, immutable input-data boundary used by measure algorithms.
83///
84/// Implementations may load files or cache data, but `casa-types` depends only
85/// on these scientific values and never performs discovery or I/O itself.
86pub trait MeasuresProvider: Debug + Send + Sync {
87    /// Prepare this provider for a bounded retained operation and return its
88    /// authoritative immutable state and exact retained residency.
89    ///
90    /// Returning `Some` promises that every cache any later provider method can
91    /// retain has been materialized, the identity commits every scientific
92    /// value those methods can expose, and repeated calls return the same
93    /// state. Bounded owners reject the default `None` rather than accepting
94    /// opaque, relabeled, or subsequently changing provider state.
95    fn prepare_bounded_state(&self) -> Result<Option<MeasuresProviderState>, String> {
96        Ok(None)
97    }
98
99    /// Interpolate EOP values at a UTC MJD.
100    fn eop_values(&self, _utc_mjd: f64) -> Result<Option<EopValues>, String> {
101        Ok(None)
102    }
103
104    /// Return TAI minus UTC in seconds at a UTC MJD.
105    fn tai_minus_utc_seconds(&self, _utc_mjd: f64) -> Result<f64, String> {
106        Err("TAI-UTC data is unavailable from this measures provider".to_string())
107    }
108
109    /// Convert a TAI MJD to UTC MJD.
110    fn utc_from_tai_mjd(&self, _tai_mjd: f64) -> Result<f64, String> {
111        Err("TAI-UTC data is unavailable from this measures provider".to_string())
112    }
113
114    /// Return IGRF coefficients and maximum spherical-harmonic degree.
115    fn igrf_coefficients(&self, _decimal_year: f64) -> Result<(Vec<f64>, usize), String> {
116        Err("IGRF data is unavailable from this measures provider".to_string())
117    }
118
119    /// Look up an observatory by case-insensitive name.
120    fn observatory(&self, _name: &str) -> Result<Option<ObservatoryPosition>, String> {
121        Ok(None)
122    }
123
124    /// Look up a fixed source by case-insensitive name.
125    fn source(&self, _name: &str) -> Result<Option<NamedSourceDirection>, String> {
126        Ok(None)
127    }
128
129    /// Look up a rest-line frequency by case-insensitive name.
130    fn spectral_line_hz(&self, _name: &str) -> Result<Option<f64>, String> {
131        Ok(None)
132    }
133}

139#[cfg(test)]
140impl MeasuresProvider for TestMeasuresProvider {
141    fn prepare_bounded_state(&self) -> Result<Option<MeasuresProviderState>, String> {
142        Ok(Some(MeasuresProviderState::new([0; 32], 0)))
143    }
144
145    fn eop_values(&self, _utc_mjd: f64) -> Result<Option<EopValues>, String> {
146        Ok(Some(EopValues {
147            dut1_seconds: 0.0,
148            x_arcsec: 0.0,
149            y_arcsec: 0.0,
150            dx_mas: 0.0,
151            dy_mas: 0.0,
152            is_predicted: false,
153        }))
154    }
155
156    fn tai_minus_utc_seconds(&self, _utc_mjd: f64) -> Result<f64, String> {
157        Ok(32.0)
158    }
159
160    fn utc_from_tai_mjd(&self, tai_mjd: f64) -> Result<f64, String> {
161        Ok(tai_mjd - 32.0 / 86_400.0)
162    }
163
164    fn igrf_coefficients(&self, decimal_year: f64) -> Result<(Vec<f64>, usize), String> {
165        if !(1900.0..=2025.999).contains(&decimal_year) {
166            return Err("test IGRF date is outside 1900 through 2025".to_string());
167        }
168        let mut coefficients = vec![0.0; 13 * 15];
169        coefficients[0] = -29_440.0;
170        coefficients[1] = -1_500.0;
171        coefficients[2] = 4_650.0;
172        Ok((coefficients, 13))
173    }
174
175    fn observatory(&self, name: &str) -> Result<Option<ObservatoryPosition>, String> {
176        Ok(match name.to_ascii_uppercase().as_str() {
177            "ALMA" => Some(ObservatoryPosition::Wgs84 {
178                longitude_rad: -67.754_929_f64.to_radians(),
179                latitude_rad: -23.022_886_f64.to_radians(),
180                height_m: 5056.8,
181            }),
182            "VLA" => Some(ObservatoryPosition::Itrf {
183                x_m: -1_601_185.0,
184                y_m: -5_041_977.0,
185                z_m: 3_554_875.0,
186            }),
187            "WSRT" => Some(ObservatoryPosition::Itrf {
188                x_m: 3_826_577.0,
189                y_m: 461_022.0,
190                z_m: 5_064_892.0,
191            }),
192            _ => None,
193        })
194    }
195
196    fn source(&self, name: &str) -> Result<Option<NamedSourceDirection>, String> {
197        Ok(match name.to_ascii_uppercase().as_str() {
198            "CASA" => Some(NamedSourceDirection {
199                reference: "J2000".to_string(),
200                longitude_rad: 6.123_487_680_622_104,
201                latitude_rad: 1.026_515_399_560_464_8,
202            }),
203            "0002-478" => Some(NamedSourceDirection {
204                reference: "ICRS".to_string(),
205                longitude_rad: 0.020_046_3,
206                latitude_rad: -0.830_872,
207            }),
208            _ => None,
209        })
210    }
211
212    fn spectral_line_hz(&self, name: &str) -> Result<Option<f64>, String> {
213        Ok(name.eq_ignore_ascii_case("HI").then_some(1.420_405_752e9))
214    }
215}
