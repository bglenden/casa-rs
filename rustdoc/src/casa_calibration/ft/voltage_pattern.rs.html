16/// A radial voltage pattern selected for the imager's next prediction.
17#[derive(Clone, Debug, Default)]
18pub struct ImagerVoltagePattern {
19    /// Custom pattern table. With no table, use the telescope's default.
20    pub table: Option<PathBuf>,
21}
22
23#[derive(Clone)]
24pub(super) struct PreparedVoltagePattern {
25    pub(super) pattern: BeamPattern,
26    centre: MDirection,
27    frame: MeasFrame,
28}
29
30impl ImagerVoltagePattern {
31    pub(super) fn prepare(
32        &self,
33        ms: &MeasurementSet,
34        selected: &ResolvedMsSelection,
35    ) -> MsResult<PreparedVoltagePattern> {
36        let main = ms.main_table();
37        let mut first = None;
38        for row in &selected.selected_rows {
39            let ScalarValue::Int32(array) =
40                main.cell_accessor(row.row_index, "ARRAY_ID")?.scalar()?
41            else {
42                return Err(invalid("ARRAY_ID must be Int32"));
43            };
44            let key = (*array, row.field_id, row.data_desc_id, row.time_mjd_seconds);
45            if first.as_ref().is_none_or(|(previous, _)| key < *previous) {
46                first = Some((key, row));
47            }
48        }
49        let Some(((array, field, _, time), first)) = first else {
50            return Err(invalid("a voltage pattern needs a selected visibility"));
51        };
52        let telescope = ms
53            .observation()?
54            .string(usize::try_from(array).map_err(invalid)?, "TELESCOPE_NAME")?;
55        let pattern = if let Some(path) = &self.table {
56            let table = voltage_pattern::VoltagePatternTable::open(path).map_err(invalid)?;
57            let record = table
58                .default_record(&telescope)
59                .map_err(invalid)?
60                .ok_or_else(|| invalid(format!("no voltage pattern for {telescope}")))?;
61            voltage_pattern::radial_pattern(&record)
62                .map_err(invalid)?
63                .pattern()
64        } else {
65            let frequency = ms.spectral_window()?.ref_frequency(0)?;
66            match PrimaryBeam::for_telescope(&telescope, frequency) {
67                Some(pattern) => pattern.pattern(),
68                None => BeamPattern::for_dish_diameter(ms.antenna()?.dish_diameter(0)?)
69                    .ok_or_else(|| {
70                        invalid("no usable dish diameter for a default voltage pattern")
71                    })?,
72            }
73        };
74        let engine = MsCalEngine::new(ms)?;
75        let pointing = if ms.subtable(SubtableId::Pointing).is_some() {
76            // A homogeneous BeamSkyJones shares one entry for both antennas.
77            // Its update stores the second antenna's direction last.
78            ms.pointing()?.legacy_direction(first.antenna2_id, time)?
79        } else {
80            None
81        };
82        let centre = match pointing {
83            Some(centre) => centre,
84            None => engine
85                .observation_direction_j2000(time, usize::try_from(field).map_err(invalid)?)?,
86        };
87        let frame = engine.spectral_frame_observatory_direction(time, centre.clone())?;
88        Ok(PreparedVoltagePattern {
89            pattern,
90            centre,
91            frame,
92        })
93    }
94}

96impl PreparedVoltagePattern {
97    pub(super) fn separations(
98        &self,
99        direction: &DirectionCoordinate,
100        [nx, ny]: [usize; 2],
101    ) -> MsResult<Array2<f64>> {
102        let centre = self
103            .centre
104            .convert_to(direction.direction_ref(), &self.frame)
105            .map_err(invalid)?;
106        let (lon0, lat0) = centre.as_angles();
107        Ok(Array2::from_shape_fn((ny, nx), |(y, x)| {
108            let Ok([lon, lat]) = direction.pixel_to_world([x as f64, y as f64]) else {
109                return f64::INFINITY;
110            };
111            let separation = ((lat - lat0) * 0.5).sin().powi(2)
112                + lat.cos() * lat0.cos() * ((lon - lon0) * 0.5).sin().powi(2);
113            2.0 * separation.clamp(0., 1.).sqrt().asin()
114        }))
115    }
116}
