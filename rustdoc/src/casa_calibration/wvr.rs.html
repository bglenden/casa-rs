29/// Parameters of the native radiometer calibration operation.
30#[derive(Debug, Clone, Serialize, Deserialize)]
31pub struct WvrRequest {
32    /// Input MeasurementSet.
33    pub measurement_set: PathBuf,
34    /// Destination T Jones table.
35    pub output_table: PathBuf,
36    /// Translate radiometer timestamps by this many seconds before fitting.
37    pub time_offset_seconds: f64,
38    /// Optional table of temperatures to subtract from raw measurements.
39    pub temperature_offsets: Option<PathBuf>,
40    /// Apply the frequency-dependent wet-path dispersion correction.
41    pub disperse: bool,
42    /// Restrict the four-channel windows used as radiometers.
43    pub radiometer_windows: Vec<i32>,
44    /// Restrict output windows. Empty means every window, including radiometers.
45    pub output_windows: Vec<i32>,
46    /// Reverse the phase sign on every selected spectral window.
47    pub reverse: bool,
48    /// Spectral windows whose phases are individually reversed.
49    pub reversed_windows: Vec<i32>,
50    /// Fit once for each contiguous interval of a source or tied group.
51    pub segment_sources: bool,
52    /// Number of fitted points in the fixed-interval retrieval.
53    pub solution_count: usize,
54    /// Source IDs grouped into a common fit when they occur consecutively.
55    pub tied_sources: Vec<Vec<i32>>,
56    /// Source IDs excluded from fitting and flagged in the output.
57    pub flagged_sources: Vec<i32>,
58    /// Source IDs included in RMS diagnostics. Empty means all sources.
59    pub statistic_sources: Vec<i32>,
60    /// Field IDs included in RMS diagnostics when no source selection is present.
61    pub statistic_fields: Vec<i32>,
62    /// Force use of the FIELD geometry in place of recorded antenna pointing.
63    pub use_field_table: bool,
64    /// Fit the optional continuum-opacity term.
65    pub continuum: bool,
66    /// Seed used by the ordinary atmospheric retrieval.
67    pub seed: u32,
68    /// Number of radiometer samples in the time smoother.
69    pub smoothing_samples: usize,
70    /// Factor applied to paths after computing the diagnostics.
71    pub scale: f64,
72    /// Antennas whose direct radiometer samples should be replaced.
73    pub flagged_antennas: Vec<usize>,
74    /// Candidate reference antennas in preference order.
75    pub reference_antennas: Vec<usize>,
76    /// Greatest distance to an eligible replacement antenna, in meters.
77    pub max_distance_meters: f64,
78    /// Number of available neighbors required to replace a radiometer.
79    pub minimum_neighbors: usize,
80    /// Minimum fraction of valid measurements for a direct radiometer.
81    pub minimum_good_fraction: f64,
82}

114/// Failure before publishing a radiometer calibration table.
115#[derive(Debug, thiserror::Error)]
116pub enum WvrError {
117    /// Failed to read the MeasurementSet or its geometry.
118    #[error(transparent)]
119    Input(#[from] MsError),
120    /// Input selection or retrieval could not furnish a useful path.
121    #[error("{0}")]
122    Invalid(String),
123    /// No available measurements could furnish a solution.
124    #[error("{0}")]
125    NoSolution(String),
126    /// The reference antenna produced zero evidence for every eligible fit.
127    #[error("No finite WVR retrieval for antenna {0}")]
128    ZeroEvidence(usize),
129    /// The requested reference antennas could not provide or interpolate a solution.
130    #[error("No usable WVR reference antenna")]
131    ReferenceUnavailable,
132    /// Requested input window is outside the four-channel radiometer collection.
133    #[error("Spectral window {0} is not a WVR spectral window")]
134    RadiometerWindow(i32),
135    /// Requested output window does not exist in the MeasurementSet.
136    #[error("Invalid output spectral window {0}")]
137    OutputWindow(i32),
138    /// Requested reversal has no selected output spectral window.
139    #[error("Reversal requested for unselected spectral window {0}")]
140    ReverseWindow(i32),
141    /// The on-disk dispersion reference could not be read.
142    #[error(transparent)]
143    Io(#[from] std::io::Error),
144    /// Failed to publish the calibration table.
145    #[error(transparent)]
146    Output(#[from] crate::CalibrationTableWriteError),
147}

149/// Generate atmospheric phase corrections from a MeasurementSet's radiometer data.
150pub fn execute_wvr(request: &WvrRequest) -> Result<WvrReport, WvrError> {
151    execute(request, crate::CalibrationOutputMode::Create, None)
152}

154fn execute(
155    request: &WvrRequest,
156    mode: crate::CalibrationOutputMode,
157    logfile: Option<&Path>,
158) -> Result<WvrReport, WvrError> {
159    if request.solution_count == 0 {
160        return Err(WvrError::Invalid("nsol must be at least one".into()));
161    }
162    if !request.scale.is_finite() || request.scale <= 0.0 {
163        return Err(WvrError::Invalid(
164            "scale must be positive and finite".into(),
165        ));
166    }
167    if !request.time_offset_seconds.is_finite() {
168        return Err(WvrError::Invalid("toffset must be finite".into()));
169    }
170    if request.smoothing_samples == 0 {
171        return Err(WvrError::Invalid("smooth must be at least one".into()));
172    }
173    if request.max_distance_meters.is_nan() || request.max_distance_meters < 0.0 {
174        return Err(WvrError::Invalid("maxdistm must be nonnegative".into()));
175    }
176    if request.continuum && (request.segment_sources || request.solution_count > 1) {
177        return Err(WvrError::Invalid(
178            "continuum requires a single fixed-interval fit".into(),
179        ));
180    }
181    if !request.segment_sources
182        && (!request.flagged_sources.is_empty() || !request.tied_sources.is_empty())
183    {
184        return Err(WvrError::Invalid(
185            "sourceflag and tie require segsource".into(),
186        ));
187    }
188    if mode == crate::CalibrationOutputMode::Create
189        && request.output_table.symlink_metadata().is_ok()
190    {
191        return Err(WvrError::Invalid(format!(
192            "Output calibration table {} already exists",
193            request.output_table.display()
194        )));
195    }
196    let ms = MeasurementSet::open(&request.measurement_set)?;
197    let offsets = request
198        .temperature_offsets
199        .as_ref()
200        .map(|path| Table::open(TableOptions::new(path)))
201        .transpose()
202        .map_err(MsError::from)?;
203    let dispersion_path = request
204        .disperse
205        .then(dispersion::DispersionTable::reference_path)
206        .transpose()?;
207    let products = std::iter::once(request.output_table.as_path()).chain(logfile);
208    for product in products {
209        protect_inputs(&ms, product, offsets.as_ref(), dispersion_path.as_deref())?;
210    }
211    let windows = phase_windows(&ms, request, dispersion_path.as_deref())?;
212    let fitted = match fit_observations(&ms, request, offsets.as_ref(), None) {
213        Err(WvrError::ZeroEvidence(antenna)) if !request.continuum => {
214            fit_observations(&ms, request, offsets.as_ref(), Some(antenna))?
215        }
216        other => other?,
217    };
218    let FittedObservations {
219        data,
220        elevations,
221        replacement,
222        antenna,
223        coefficients,
224        fits,
225    } = fitted;
226    let antenna_table = ms.antenna()?;
227    let names = (0..antenna_table.row_count())
228        .map(|id| antenna_table.name(id))
229        .collect::<Result<Vec<_>, _>>()?;
230    let statistic_intervals = series::statistics_intervals(&data, request);
231    let mut diagnostics = series::PathDiagnostics::new(names, &replacement);
232    let mut integrations = Vec::with_capacity(data.integrations.len());
233    for (point, elevation) in data.integrations.iter().zip(elevations) {
234        let derivative = coefficients
235            .at(point.time_seconds)
236            .temperature_per_path_k_per_meter;
237        let weights = path_weights(derivative);
238        let selected = statistic_intervals
239            .iter()
240            .any(|&[start, end]| (start..=end).contains(&point.time_seconds));
241        let source_flagged = request.flagged_sources.contains(&point.source_id);
242        let mut meters: Vec<_> = point
243            .temperatures
244            .iter()
245            .enumerate()
246            .map(|(id, &reading)| {
247                let path = if source_flagged {
248                    0.0
249                } else {
250                    antenna_path_meters(reading, weights)
251                };
252                if selected {
253                    diagnostics.observe(id, path, elevation, reading, derivative);
254                }
255                path
256            })
257            .collect();
258        if selected {
259            diagnostics.baselines(&meters);
260        }
261        meters.iter_mut().for_each(|path| *path *= request.scale);
262        integrations.push(AntennaPaths {
263            time_seconds: point.time_seconds,
264            field_id: point.field_id,
265            meters,
266        });
267    }
268    writer::publish(&ms, request, &integrations, &windows, mode)?;
269    let (antennas, largest_baseline_meters) = diagnostics.finish();
270    let log = series::render_log(
271        &antennas,
272        &fits,
273        antenna,
274        &statistic_intervals,
275        data.observations[0].time_seconds,
276        coefficients.representative(),
277        largest_baseline_meters,
278    );
279    Ok(WvrReport {
280        output_table: request.output_table.clone(),
281        status: if request.continuum { -2 } else { 0 },
282        antennas,
283        log,
284    })
285}

287/// Run the direct tool with its independently located diagnostic log.
288pub fn execute_wvr_with_log(request: &WvrRequest, logfile: &Path) -> Result<WvrReport, WvrError> {
289    let target = crate::paths::normalized(logfile)?;
290    let paths = [&request.measurement_set, &request.output_table]
291        .into_iter()
292        .chain(request.temperature_offsets.iter());
293    for path in paths {
294        if crate::paths::overlap(&target, path)? {
295            return Err(WvrError::Invalid(
296                "WVR logfile path overlaps a scientific table".into(),
297            ));
298        }
299    }
300    if target.is_dir() {
301        return Err(WvrError::Invalid("WVR logfile must be a file".into()));
302    }
303    let directory = target.parent().expect("an absolute file path has a parent");
304    let mut file = tempfile::NamedTempFile::new_in(directory)?;
305    let outcome = execute(
306        request,
307        crate::CalibrationOutputMode::Replace,
308        Some(&target),
309    );
310    match &outcome {
311        Ok(report) => file.write_all(report.log.as_bytes())?,
312        Err(
313            error @ (WvrError::NoSolution(_)
314            | WvrError::ZeroEvidence(_)
315            | WvrError::ReferenceUnavailable),
316        ) => {
317            writeln!(file, "WVR phase calibration\n{error}")?;
318        }
319        Err(_) => return outcome,
320    }
321    file.persist(target).map_err(|error| error.error)?;
322    outcome
323}

325fn protect_inputs(
326    ms: &MeasurementSet,
327    product: &Path,
328    offsets: Option<&Table>,
329    dispersion: Option<&Path>,
330) -> Result<(), WvrError> {
331    let mut inputs = ms
332        .path()
333        .into_iter()
334        .chain(dispersion)
335        .map(Path::to_path_buf)
336        .collect::<Vec<_>>();
337    let subtables = SubtableId::ALL_REQUIRED
338        .iter()
339        .chain(SubtableId::ALL_OPTIONAL)
340        .filter_map(|&subtable| ms.subtable(subtable));
341    for table in std::iter::once(ms.main_table())
342        .chain(subtables)
343        .chain(offsets)
344    {
345        inputs.extend(table.path().map(Path::to_path_buf));
346        if table.is_reference_table().map_err(MsError::from)? {
347            inputs.extend(table.cache_plain_table_paths().map_err(MsError::from)?);
348        }
349    }
350    for input in inputs {
351        if crate::paths::overlap(product, &input)? {
352            return Err(WvrError::Invalid(format!(
353                "WVR product {} overlaps input {}",
354                product.display(),
355                input.display()
356            )));
357        }
358    }
359    Ok(())
360}

362fn sky_states(ms: &MeasurementSet) -> Result<Vec<i32>, WvrError> {
363    let states = ms
364        .subtable(SubtableId::State)
365        .ok_or_else(|| MsError::MissingSubtable("STATE".into()))?;
366    let mut ids = Vec::new();
367    for row in 0..states.row_count() {
368        let cell = states
369            .cell_accessor(row, "OBS_MODE")
370            .and_then(|cell| cell.scalar())
371            .map_err(MsError::from)?;
372        let ScalarValue::String(mode) = cell else {
373            continue;
374        };
375        if !mode.contains("CALIBRATE_ATMOSPHERE")
376            && ["ON_SOURCE", "REFERENCE", "SIGNAL"]
377                .iter()
378                .any(|intent| mode.contains(intent))
379        {
380            ids.push(row as i32);
381        }
382    }
383    Ok(ids)
384}

386fn phase_windows(
387    ms: &MeasurementSet,
388    request: &WvrRequest,
389    dispersion_path: Option<&Path>,
390) -> Result<Vec<PhaseWindow>, WvrError> {
391    let windows = ms.spectral_window()?;
392    for &window in &request.radiometer_windows {
393        if window < 0
394            || window as usize >= windows.row_count()
395            || windows.num_chan(window as usize)? != 4
396        {
397            return Err(WvrError::RadiometerWindow(window));
398        }
399    }
400    for &window in &request.output_windows {
401        if window < 0 || window as usize >= windows.row_count() {
402            return Err(WvrError::OutputWindow(window));
403        }
404    }
405    let dispersion = dispersion_path
406        .map(dispersion::DispersionTable::load)
407        .transpose()?;
408    let indices: Vec<_> = if request.output_windows.is_empty() {
409        (0..windows.row_count() as i32).collect()
410    } else {
411        request.output_windows.clone()
412    };
413    if !request.reverse {
414        for &id in &request.reversed_windows {
415            if !indices.contains(&id) {
416                return Err(WvrError::ReverseWindow(id));
417            }
418        }
419    }
420    indices
421        .into_iter()
422        .map(|id| {
423            let channels = windows.chan_freq(id as usize)?;
424            let frequency = channels
425                .get(channels.len() / 2)
426                .copied()
427                .filter(|frequency| frequency.is_finite() && *frequency > 0.0)
428                .ok_or_else(|| {
429                    WvrError::Invalid(format!(
430                        "No valid middle-channel frequency in spectral window {id}"
431                    ))
432                })?;
433            Ok(PhaseWindow {
434                id,
435                frequency_hz: frequency,
436                reverse: request.reverse || request.reversed_windows.contains(&id),
437                dispersion: dispersion
438                    .as_ref()
439                    .map_or(Ok(0.0), |table| table.at(frequency))?,
440            })
441        })
442        .collect()
443}

445fn elevations(
446    ms: &MeasurementSet,
447    points: &[RadiometerIntegration],
448    use_field_table: bool,
449) -> Result<Vec<f64>, WvrError> {
450    if !use_field_table && let Some(result) = pointing_elevations(ms, points) {
451        return Ok(result);
452    }
453    let geometry = MsCalEngine::for_calibration(ms)?;
454    points
455        .iter()
456        .map(|point| {
457            let (_, elevation) = geometry.azel(point.time_seconds, point.field_id as usize, 0)?;
458            Ok(elevation)
459        })
460        .collect()
461}

463fn pointing_elevations(ms: &MeasurementSet, points: &[RadiometerIntegration]) -> Option<Vec<f64>> {
464    const BATCH_ROWS: usize = 32_768;
465    let pointing = ms.subtable(SubtableId::Pointing)?;
466    let columns = pointing.required_scalar_columns_owned(&["TIME"]).ok()?;
467    let RequiredScalarColumnValues::Float64(times) = columns.get("TIME")? else {
468        return None;
469    };
470    let last = times.len().checked_sub(1)?;
471    let directions = pointing.column_accessor("DIRECTION").ok()?;
472    let mut index = 0;
473    let mut selected = Vec::with_capacity(points.len());
474    for point in points {
475        while index < last && times[index] < point.time_seconds {
476            index += 1;
477        }
478        selected.push(index);
479    }
480    let mut selected = selected.into_iter().peekable();
481    let mut elevations = Vec::with_capacity(points.len());
482    // CASA falls back to FIELD when any POINTING direction cannot be read.
483    // Validate in physical order without caching every direction in the table.
484    for start in (0..times.len()).step_by(BATCH_ROWS) {
485        let end = start.saturating_add(BATCH_ROWS).min(times.len());
486        let rows = (start..end).collect::<Vec<_>>();
487        let values = directions.array_cells_owned_uncached(&rows).ok()?;
488        for (row, direction) in rows.into_iter().zip(values) {
489            let Some(ArrayValue::Float64(direction)) = direction else {
490                return None;
491            };
492            let elevation = *direction.get([1, 0])?;
493            while selected.next_if_eq(&row).is_some() {
494                elevations.push(elevation);
495            }
496        }
497    }
498    Some(elevations)
499}

501struct FittedObservations {
502    data: RadiometerSeries,
503    elevations: Vec<f64>,
504    replacement: AntennaReplacement,
505    antenna: usize,
506    coefficients: CoefficientTimeline,
507    fits: Vec<Posterior>,
508}
509
510fn fit_observations(
511    ms: &MeasurementSet,
512    request: &WvrRequest,
513    offsets: Option<&Table>,
514    failed_reference: Option<usize>,
515) -> Result<FittedObservations, WvrError> {
516    let mut data = RadiometerSeries::read(ms, &request.radiometer_windows, offsets)?;
517    series::smooth(&mut data.integrations, request.smoothing_samples);
518    let sky = sky_states(ms)?;
519    data.integrations
520        .retain(|point| sky.contains(&point.state_id));
521    if data.integrations.len() < 2 {
522        return Err(WvrError::NoSolution(
523            "Fewer than two radiometer integrations point at the sky".into(),
524        ));
525    }
526    let elevations = elevations(ms, &data.integrations, request.use_field_table)?;
527    for point in &mut data.integrations {
528        point.time_seconds += request.time_offset_seconds;
529    }
530    let replacement = series::replace_bad_antennas(ms, &mut data, request, failed_reference)?;
531    let antenna = series::reference_antenna(&data, &replacement, request)?;
532    let (coefficients, fits) = fit_timeline(
533        &data.integrations,
534        &data.observations,
535        &elevations,
536        antenna,
537        request,
538    )?;
539    Ok(FittedObservations {
540        data,
541        elevations,
542        replacement,
543        antenna,
544        coefficients,
545        fits,
546    })
547}

549fn fit_timeline(
550    points: &[RadiometerIntegration],
551    observations: &[Observation],
552    elevations: &[f64],
553    reference: usize,
554    request: &WvrRequest,
555) -> Result<(CoefficientTimeline, Vec<Posterior>), WvrError> {
556    let mut knots = Vec::new();
557    let mut fits = Vec::new();
558    let mut attempted = false;
559    for interval in series::fit_intervals(points, observations, request) {
560        let point = &points[interval.reference];
561        if request.flagged_sources.contains(&point.source_id) {
562            continue;
563        }
564        let Some(temperatures) = point.temperatures[reference] else {
565            continue;
566        };
567        if temperatures[0] <= 0.0 {
568            continue;
569        }
570        attempted = true;
571        let Some(fit) = retrieval::fit(
572            temperatures,
573            point.time_seconds,
574            elevations[interval.reference],
575            request.seed,
576            request.continuum,
577        ) else {
578            continue;
579        };
580        for time_seconds in interval.boundaries {
581            knots.push(Retrieval {
582                time_seconds,
583                ..fit.coefficients
584            });
585        }
586        fits.push(fit);
587    }
588    let timeline = CoefficientTimeline::new(knots).ok_or_else(|| {
589        if attempted {
590            WvrError::ZeroEvidence(reference)
591        } else {
592            WvrError::NoSolution("No usable WVR retrievals".into())
593        }
594    })?;
595    Ok((timeline, fits))
596}
