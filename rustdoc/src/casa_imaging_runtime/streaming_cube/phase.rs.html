98    /// Identify the migrated capability from compiled science and source shape.
99    /// Other scientific modes retain their own execution owners; a failed native
100    /// phase never retries through another implementation.
101    pub fn supports(problem: &CompiledProblem) -> io::Result<bool> {
102        if problem.weighting().scheme() != casa_imaging_model::WeightingScheme::Natural
103            || !matches!(
104                problem.model_lifecycle().input(),
105                casa_imaging_model::ModelInputCommitment::Empty
106            )
107            || problem.visibility_transform().is_some()
108            || !matches!(
109                problem.reconstruction().basis(),
110                casa_imaging_model::ReconstructionBasis::ChannelLocal { .. }
111            )
112        {
113            return Ok(false);
114        }
115        let [source] = problem.selected_observation().read_set().sources() else {
116            return Ok(false);
117        };
118        let ([dd], [spw], [pol]) = (
119            source.selection().data_descriptions(),
120            source.selection().spectral_windows(),
121            source.selection().correlations(),
122        ) else {
123            return Ok(false);
124        };
125        if dd.spectral_window_id() != spw.spectral_window_id()
126            || dd.polarization_id() != pol.polarization_id()
127            || spw.channel_indices().len() < 2
128        {
129            return Ok(false);
130        }
131        let specification =
132            SpectralOperatorSpecification::for_slab(problem, 0, 1).map_err(io::Error::other)?;
133        Ok(BandPlan::supports(&specification))
134    }

194    /// Refresh the residual from the owned native store and immutable model epoch.
195    #[allow(clippy::too_many_arguments)]
196    pub fn refresh(
197        problem: CompiledProblem,
198        registry: &impl ImplementationRegistry,
199        policy: SpectralCycleExecutionPolicy,
200        storage: ManagedSpillStorage,
201        retained: NativeReplay,
202        input: FinalMajorPhaseInput,
203        ordinal: u32,
204        minor: Option<(ImageDomainReconstructionMaskPlans, MinorCycleProgram)>,
205    ) -> io::Result<(PhysicalWorkBinding, Self)> {
206        let workers = Self::workers(&problem, &policy)?;
207        let prior = input.evidence().normal_state();
208        if ordinal == 0
209            || retained.replay.problem_id() != problem.problem_id()
210            || prior.problem_id() != problem.problem_id()
211            || prior.weighting_generation() != retained.replay.weighting_generation()
212            || prior.selected_generation() != retained.replay.selected_generation()
213            || prior.continuum_transform_generation()
214                != retained
215                    .replay
216                    .continuum_transform()
217                    .map(|v| v.generation_id())
218        {
219            return Err(io::Error::other(
220                "native refresh changed source, weighting or problem association",
221            ));
222        }
223        Self::build(
224            problem,
225            registry,
226            policy,
227            storage,
228            None,
229            Some(retained),
230            Some(input),
231            ordinal,
232            workers,
233            1,
234            0,
235            minor,
236        )
237    }

239    fn workers(
240        problem: &CompiledProblem,
241        policy: &SpectralCycleExecutionPolicy,
242    ) -> io::Result<usize> {
243        let workers = policy
244            .authority
245            .planning_worker_capacity(&policy.resource_policy)
246            .map_err(io::Error::other)?
247            .min(problem.geometry().spectral().output_channels() as u64);
248        if workers == 0 {
249            return Err(io::Error::other(
250                "native cube has no admitted worker capacity",
251            ));
252        }
253        usize::try_from(workers).map_err(io::Error::other)
254    }

256    #[allow(clippy::too_many_arguments)]
257    fn build(
258        problem: CompiledProblem,
259        registry: &impl ImplementationRegistry,
260        policy: SpectralCycleExecutionPolicy,
261        storage: ManagedSpillStorage,
262        selected: Option<DeferredSelectedObservationAccess>,
263        retained: Option<NativeReplay>,
264        pass_input: Option<FinalMajorPhaseInput>,
265        ordinal: u32,
266        workers: usize,
267        source_slots: usize,
268        enclosing_owner_bytes: u64,
269        minor: Option<(ImageDomainReconstructionMaskPlans, MinorCycleProgram)>,
270    ) -> io::Result<(PhysicalWorkBinding, Self)> {
271        if policy.visibility_write.is_some()
272            || policy.aw_projection.is_some()
273            || policy.aw_reader.is_some()
274        {
275            return Err(io::Error::other(
276                "native cube comparison does not support AW or visibility writes",
277            ));
278        }
279        let imported = retained.is_some();
280        let phase_input_artifact = pass_input.as_ref().map(|input| {
281            (
282                input.identity(),
283                (input.pending_delta_terms() * size_of::<casa_imaging_model::ModelDeltaTerm>())
284                    as u64,
285            )
286        });
287        let pass = SpectralPassIdentity::new(
288            if imported {
289                SpectralPassPhase::FinalMajor
290            } else {
291                SpectralPassPhase::InitialMajor
292            },
293            ordinal,
294        );
295        let (base, source) = crate::spectral_cycle_plan::base_physical(
296            &problem,
297            registry,
298            &policy,
299            pass,
300            pass_input.as_ref().map(FinalMajorPhaseInput::identity),
301        )
302        .map_err(io::Error::other)?;
303        let read = base
304            .execution_dag()
305            .nodes()
306            .values()
307            .find(|node| node.kind == WorkKind::ObservationRead)
308            .ok_or_else(|| io::Error::other("native cube plan lacks source read"))?
309            .id
310            .clone();
311        let prepare = base
312            .observation_transaction()
313            .final_model_preparation()
314            .ok_or_else(|| io::Error::other("native cube plan lacks model preparation"))?
315            .clone();
316        let reconcile = base
317            .observation_transaction()
318            .post_replay_reconciliation()
319            .ok_or_else(|| io::Error::other("native cube plan lacks reconciliation"))?
320            .clone();
321        let weighting =
322            plan_weighting(&problem, policy.weighting_limits).map_err(io::Error::other)?;
323        let preparation = if !imported {
324            Some(
325                crate::weighting::NativePreparationPlan::new(
326                    &problem,
327                    &weighting,
328                    workers,
329                    NativePhasePlan::source_store(&base, &problem)?.block_rows,
330                )
331                .map_err(io::Error::other)?,
332            )
333        } else {
334            None
335        };
336        let physical = if imported {
337            base
338        } else {
339            WeightingPlanFragment::streaming_for_pass(
340                &weighting,
341                read.clone(),
342                source.clone(),
343                policy.implementation.clone(),
344                pass,
345                WeightingStreamingMode::NaturalInitial,
346                None,
347            )
348            .with_native_preparation(preparation.expect("initial native source plan"))
349            .compose(&base)
350            .map_err(io::Error::other)?
351        };
352        let minor = minor.map(|(masks, program)| {
353            (
354                WorkNodeId::new(format!("native-cube-minor-{ordinal}")),
355                masks,
356                program,
357            )
358        });
359        let physical = if let Some((node, _, _)) = &minor {
360            let resources = crate::spectral_cycle_plan::MinorCycleResources::for_worker_count(
361                ReconstructionPlaneWorkspace::for_problem(&problem).map_err(io::Error::other)?,
362                &policy,
363                workers as u64,
364            )
365            .map_err(io::Error::other)?;
366            crate::spectral_cycle_plan::append_minor(
367                registry, physical, &policy, node, resources, 0,
368            )
369            .map_err(io::Error::other)?
370        } else {
371            physical
372        };
373        // The initial grain is one complete output plane, including its normal
374        // write window. Wave sizing groups these grains under the shared budget.
375        let state_terminal = minor
376            .as_ref()
377            .map_or(&reconcile, |(node, _, _)| node)
378            .clone();
379        let mut cube_state = CubeStatePlan::streaming_cube(
380            &problem,
381            &storage,
382            1,
383            prepare.clone(),
384            state_terminal.clone(),
385            false,
386            imported,
387        )?;
388        let before_cube_state = physical;
389        let mut physical = cube_state
390            .compose(
391                registry,
392                policy.implementation.clone(),
393                &storage,
394                before_cube_state.clone(),
395                &read,
396                &reconcile,
397            )
398            .map_err(io::Error::other)?;
399        let spectral = problem.geometry().spectral();
400        let output_hz = (0..spectral.output_channels())
401            .map(|i| {
402                spectral
403                    .channel_centre_hz(i)
404                    .ok_or_else(|| io::Error::other("native cube output axis is incomplete"))
405            })
406            .collect::<io::Result<Vec<_>>>()?;
407        let bands = if let Some(retained) = &retained {
408            retained
409                .native
410                .bands
411                .iter()
412                .map(BandPlan::residual_refresh)
413                .collect()
414        } else {
415            (0..output_hz.len())
416                .map(|channel| {
417                    BandPlan::new(
418                        &SpectralOperatorSpecification::for_slab(&problem, channel, 1)?,
419                        SpectralOperatorPass::InitialMajor,
420                    )
421                })
422                .collect::<Result<Vec<_>, _>>()
423                .map_err(io::Error::other)?
424        };
425        // Unit fixtures pass their controlled enclosing-owner bound explicitly.
426        #[cfg(not(test))]
427        let enclosing_owner_bytes = enclosing_owner_bytes.max(super::enclosing_memory_bytes(
428            pass_input
429                .as_ref()
430                .map_or(Ok(0), |input| {
431                    input.evidence().normal_state().retained_resident_bytes()
432                })
433                .map_err(io::Error::other)?,
434        )?);
435        let shared_bytes = enclosing_owner_bytes
436            .checked_add(Self::shared_bytes(&problem, output_hz.capacity())?)
437            .ok_or_else(|| io::Error::other("native cube owner residency overflow"))?;
438        let plan_native = |physical: &PhysicalWorkBinding| {
439            if let Some(retained) = &retained {
440                NativePhasePlan::new_for_pass(
441                    physical,
442                    &policy.authority,
443                    &policy.resource_policy,
444                    &storage,
445                    retained.native.store.plan,
446                    &bands,
447                    shared_bytes,
448                    workers,
449                    source_slots,
450                    true,
451                )
452            } else {
453                NativePhasePlan::for_initial_source(
454                    physical,
455                    &policy.authority,
456                    &policy.resource_policy,
457                    &storage,
458                    &problem,
459                    &bands,
460                    shared_bytes,
461                    workers,
462                    source_slots,
463                )
464            }
465        };
466        let mut native_plan = plan_native(&physical)?;
467        let resident = CubeStatePlan::streaming_cube(
468            &problem,
469            &storage,
470            1,
471            prepare.clone(),
472            state_terminal,
473            true,
474            imported,
475        )?;
476        let resident_physical = resident
477            .compose(
478                registry,
479                policy.implementation.clone(),
480                &storage,
481                before_cube_state,
482                &read,
483                &reconcile,
484            )
485            .map_err(io::Error::other)?;
486        // Compare the complete storage reservation before allocating payloads.
487        // Residency must leave a full worker wave, not just a runnable band.
488        let resident_available = match policy.authority.remaining_planning_memory_bytes(
489            &policy.resource_policy,
490            resident_physical.execution_dag().resource_alternative(),
491        ) {
492            Ok(bytes) => bytes,
493            Err(crate::ResourceError::Infeasible { resource, .. })
494                if resource.starts_with("memory-domain:") =>
495            {
496                0
497            }
498            Err(error) => return Err(io::Error::other(error)),
499        };
500        if resident_available >= native_plan.worker_wave_bytes {
501            physical = resident_physical;
502            native_plan = plan_native(&physical)?;
503            cube_state = resident;
504            eprintln!(
505                "streaming_cube_normal_storage ordinal={ordinal} backing=resident retained_bytes={}",
506                cube_state.retained_memory_bytes()
507            );
508        } else {
509            eprintln!(
510                "streaming_cube_normal_storage ordinal={ordinal} backing=paged retained_bytes={}",
511                cube_state.retained_memory_bytes()
512            );
513        }
514        let physical = native_plan.compose(physical, &storage)?;
515        let (native, replay) = retained.map_or((None, None), |mut retained| {
516            retained.native.bands = bands.iter().map(BandPlan::residual_refresh).collect();
517            (Some(retained.native), Some(retained.replay))
518        });
519        Ok((
520            physical,
521            Self {
522                problem,
523                implementation: policy.implementation,
524                weighting,
525                preparation,
526                source,
527                pass,
528                read,
529                prepare,
530                reconcile,
531                output_hz,
532                storage,
533                native_plan,
534                cube_state,
535                minor,
536                imported,
537                phase_input_artifact,
538                #[cfg(test)]
539                failure: Failure::None,
540                state: Mutex::new(State {
541                    selected,
542                    weighting: WeightingExecutionState::new(),
543                    native,
544                    bands: if imported { Vec::new() } else { bands },
545                    lifecycle: None,
546                    model: None,
547                    complete: None,
548                    pass_input,
549                    prior: None,
550                    masks: None,
551                    replay,
552                    minor: None,
553                    #[cfg(test)]
554                    fenced: false,
555                    #[cfg(test)]
556                    published: false,
557                }),
558            },
559        ))
560    }

583    fn shared_bytes(problem: &CompiledProblem, output_capacity: usize) -> io::Result<u64> {
584        let [source] = problem.selected_observation().read_set().sources() else {
585            return Err(io::Error::other("native cube requires one selected source"));
586        };
587        let [spw] = source.selection().spectral_windows() else {
588            return Err(io::Error::other("native cube requires one spectral window"));
589        };
590        // Stokes I with at most four correlations stays inside the polarization
591        // operator's inline SmallVec storage; no matrix heap is omitted here.
592        let bytes = spw
593            .channel_indices()
594            .len()
595            .checked_mul(size_of::<u32>())
596            .and_then(|n| n.checked_add(output_capacity.checked_mul(size_of::<f64>())?))
597            .and_then(|n| {
598                n.checked_add(
599                    size_of::<Self>()
600                        + size_of::<PolarizationOperator>()
601                        + size_of::<[CorrelationType; 4]>(),
602                )
603            })
604            .ok_or_else(|| io::Error::other("native cube owner residency overflow"))?;
605        Ok(bytes as u64)
606    }

622impl WorkImplementation for InitialCube {
623    type Error = io::Error;
624    fn implementation_id(&self) -> &WorkImplementationId {
625        &self.implementation
626    }
627    fn execute(&self, context: WorkExecutionContext<'_>) -> io::Result<WorkMeasurements> {
628        let started = std::time::Instant::now();
629        let mut state = self.state.lock().unwrap();
630        if context.node().id == self.prepare {
631            let executable = ExecutableModelProblem::from_compiled(self.problem.clone())
632                .map_err(io::Error::other)?;
633            let attempt = ModelExecutionAttemptId::new(LogicalIdentity::from_sha256(
634                context.attempt_id().as_bytes(),
635            ));
636            let storage = self.cube_state.model_storage(context)?;
637            let (lifecycle, named, terms) = if let Some(input) = state.pass_input.take() {
638                let (terms, continuation, prior, masks) = input.into_execution_parts();
639                let (lifecycle, named) = ModelLifecycle::continue_from(
640                    executable,
641                    attempt,
642                    context.lease_epoch(),
643                    continuation,
644                    storage,
645                )
646                .map_err(io::Error::other)?;
647                state.prior = Some(prior);
648                state.masks = Some(masks);
649                (lifecycle, named, Some(terms))
650            } else {
651                if self.imported {
652                    return Err(io::Error::other(
653                        "native refresh lacks accepted model input",
654                    ));
655                }
656                let lifecycle =
657                    ModelLifecycle::bind(executable, attempt, context.lease_epoch(), storage)
658                        .map_err(io::Error::other)?;
659                let named = lifecycle.initial_empty().map_err(io::Error::other)?;
660                (lifecycle, named, None)
661            };
662            let delta = terms
663                .filter(|terms| !terms.is_empty())
664                .map(|terms| {
665                    lifecycle
666                        .compile_delta(&named, terms.iter().copied())
667                        .map_err(io::Error::other)
668                })
669                .transpose()?;
670            state.model = Some(
671                MajorCyclePreparation::prepare(&lifecycle, named, delta)
672                    .map_err(io::Error::other)?,
673            );
674            state.lifecycle = Some(lifecycle);
675        } else if context.node().id == self.read {
676            if self.imported {
677                if state.native.is_none()
678                    || state
679                        .replay
680                        .as_ref()
681                        .is_none_or(|replay| replay.problem_id() != self.problem.problem_id())
682                {
683                    return Err(io::Error::other("native import lacks its owned source"));
684                }
685            } else {
686                let selected = state
687                    .selected
688                    .take()
689                    .ok_or_else(|| io::Error::other("native source already consumed"))?;
690                let fragment = self.fragment();
691                fragment
692                    .authorize_source_observation(
693                        context,
694                        &self.problem,
695                        &selected
696                            .certify_residency(&self.problem)
697                            .map_err(io::Error::other)?,
698                    )
699                    .map_err(io::Error::other)?;
700                let bands = std::mem::take(&mut state.bands);
701                let input = NativePreparation::new(
702                    &self.storage,
703                    self.native_plan.store,
704                    &self.problem,
705                    &self.output_hz,
706                    bands,
707                )?;
708                let source_open = std::time::Instant::now();
709                let selected = selected.open(&self.problem).map_err(io::Error::other)?;
710                let source_open_nanos = source_open.elapsed().as_nanos();
711                state.native = Some(input.traverse(
712                    context,
713                    &fragment,
714                    &self.problem,
715                    selected,
716                    &mut state.weighting,
717                )?);
718                if std::env::var_os("CASA_RS_TRACE_IMAGING_STAGE_TIMING").is_some() {
719                    eprintln!(
720                        "streaming_cube_source open_nanos={source_open_nanos} traversal={:?} stream={:?}",
721                        state.weighting.latest_traversal_measurements(),
722                        state.weighting.latest_stream_measurements()
723                    );
724                }
725                #[cfg(test)]
726                assert!(state.weighting.replay_completion().is_none());
727            }
728        } else if context.node().id == self.reconcile {
729            #[cfg(test)]
730            assert!(self.imported || state.fenced);
731            let PreparedNative {
732                mut store,
733                bands,
734                layout,
735            } = state
736                .native
737                .take()
738                .ok_or_else(|| io::Error::other("native source is not prepared"))?;
739            let model = state
740                .model
741                .take()
742                .ok_or_else(|| io::Error::other("native model is not prepared"))?;
743            let mut correlations = [CorrelationType::StokesI; 4];
744            for (target, (_, correlation)) in correlations.iter_mut().zip(&layout.correlations) {
745                *target = *correlation;
746            }
747            let polarization = PolarizationOperator::compile(
748                self.problem.reconstruction().polarization().coordinates(),
749                &correlations[..layout.correlations.len()],
750                [0.0; 2],
751                MuellerMatrix::identity(),
752            )
753            .map_err(io::Error::other)?;
754            let replay = state
755                .replay
756                .as_ref()
757                .or_else(|| state.weighting.replay_completion())
758                .ok_or_else(|| io::Error::other("native source fence is incomplete"))?;
759            let retained_bands = bands.iter().map(BandPlan::residual_refresh).collect();
760            let mut fold: Option<PendingCompleteDataSlabFold> = None;
761            let reconciliation = &self.reconcile;
762            #[cfg(test)]
763            let reconciliation = if self.failure == Failure::ReconciliationNode {
764                &self.read
765            } else {
766                reconciliation
767            };
768            let mut refresh = if self.imported {
769                Some(
770                    PendingCubeRefresh::new(
771                        context,
772                        reconciliation,
773                        &self.read,
774                        &SpectralOperatorSpecification::new(&self.problem)
775                            .map_err(io::Error::other)?,
776                        state
777                            .prior
778                            .as_ref()
779                            .ok_or_else(|| io::Error::other("native refresh lacks prior normal"))?,
780                        model.final_model().generation_id(),
781                        replay,
782                        &self.cube_state.normal_storage()?,
783                    )
784                    .map_err(io::Error::other)?,
785                )
786            } else {
787                None
788            };
789            let mut pending = bands.into_iter();
790            while !pending.as_slice().is_empty() {
791                let count = WavePlan::prefix(
792                    store.plan,
793                    pending.as_slice(),
794                    self.native_plan.workers,
795                    self.native_plan.source_slots,
796                    self.native_plan.shared_bytes,
797                    self.native_plan.workspace_bytes,
798                )?;
799                let jobs = pending.by_ref().take(count).collect::<Vec<_>>();
800                let mut measurements = None;
801                let wave = execute::execute(
802                    &mut store,
803                    jobs,
804                    model.final_model(),
805                    &layout,
806                    &self.output_hz,
807                    &polarization,
808                    self.native_plan.workers,
809                    self.native_plan.source_slots,
810                    self.native_plan.shared_bytes,
811                    self.native_plan.workspace_bytes,
812                    self.pass.ordinal(),
813                    &mut measurements,
814                )?;
815                if let Some(measurements) = measurements {
816                    eprintln!(
817                        "streaming_cube_wave ordinal={} bands={} workers={} measured={measurements:?} io={:?}",
818                        self.pass.ordinal(),
819                        wave.bands.len(),
820                        self.native_plan.workers,
821                        wave.source
822                    );
823                }
824                for normal in wave.bands {
825                    let normal = match normal {
826                        BandResult::Residual(residual) => {
827                            refresh
828                                .as_mut()
829                                .ok_or_else(|| {
830                                    io::Error::other("initial phase returned residual-only output")
831                                })?
832                                .append(residual)
833                                .map_err(io::Error::other)?;
834                            continue;
835                        }
836                        BandResult::Initial(normal) if !self.imported => normal,
837                        BandResult::Initial(_) => {
838                            return Err(io::Error::other("refresh returned initial normal fields"));
839                        }
840                    };
841                    let core = normal.slab().core_range();
842                    let spec = SpectralOperatorSpecification::for_slab(
843                        &self.problem,
844                        core.start,
845                        core.len(),
846                    )
847                    .map_err(io::Error::other)?;
848                    let band = CompleteDataSlabResult::from_streaming_cube(
849                        context,
850                        reconciliation,
851                        &spec,
852                        normal,
853                        replay,
854                    )
855                    .map_err(io::Error::other)?;
856                    fold = Some(
857                        match fold {
858                            None => band.begin_fold(&self.cube_state.normal_storage()?),
859                            Some(prefix) => prefix.fold(band),
860                        }
861                        .map_err(io::Error::other)?,
862                    );
863                }
864            }
865            let complete = if let Some(refresh) = refresh {
866                refresh.complete()
867            } else {
868                fold.ok_or_else(|| io::Error::other("native cube produced no bands"))?
869                    .complete(replay)
870            }
871            .map_err(io::Error::other)?;
872            state.prior = None;
873            state.native = Some(PreparedNative {
874                store,
875                bands: retained_bands,
876                layout,
877            });
878            let mut operator =
879                MajorCycleOperatorState::begin(complete, model).map_err(io::Error::other)?;
880            if let Some(masks) = state.masks.take() {
881                operator = operator
882                    .bind_reconstruction_masks(&masks)
883                    .map_err(io::Error::other)?;
884            }
885            state.complete = Some(
886                operator
887                    .reconcile(
888                        context,
889                        state
890                            .lifecycle
891                            .as_mut()
892                            .ok_or_else(|| io::Error::other("native model lifecycle is absent"))?,
893                    )
894                    .map_err(io::Error::other)?,
895            );
896        } else if let Some((_, masks, program)) = self
897            .minor
898            .as_ref()
899            .filter(|(node, _, _)| context.node().id == *node)
900        {
901            let complete = state
902                .complete
903                .take()
904                .ok_or_else(|| io::Error::other("native minor lacks normal state"))?;
905            let lifecycle = state
906                .lifecycle
907                .as_ref()
908                .ok_or_else(|| io::Error::other("native minor lacks lifecycle"))?;
909            let mut measurements = None;
910            let minor = complete.run_reconstruction_cycle(
911                lifecycle,
912                masks,
913                program.clone(),
914                context,
915                self.pass.ordinal(),
916                &mut measurements,
917            )?;
918            if let Some(measurements) = measurements {
919                eprintln!(
920                    "streaming_cube_minor ordinal={} measured={measurements:?}",
921                    self.pass.ordinal()
922                );
923            }
924            state.minor = Some(minor);
925        } else if !self.imported && context.node().id == *self.fragment().release_node() {
926            if context.is_cleanup() {
927                state
928                    .weighting
929                    .release(context, &self.fragment())
930                    .map_err(io::Error::other)?;
931            } else {
932                state.replay = Some(
933                    state
934                        .weighting
935                        .release_retaining_replay(context, &self.fragment())
936                        .map_err(io::Error::other)?,
937                );
938            }
939        }
940        eprintln!(
941            "streaming_cube_stage ordinal={} node={} elapsed_nanos={}",
942            self.pass.ordinal(),
943            context.node().id.as_str(),
944            started.elapsed().as_nanos()
945        );
946        self.cube_state.log_measurements(&context.node().id);
947        Ok(WorkMeasurements::new(
948            context
949                .node()
950                .claims
951                .iter()
952                .map(|claim| {
953                    ResourceMeasurement::new(
954                        claim.resource.clone(),
955                        claim.lifetime.clone(),
956                        claim.amount,
957                    )
958                })
959                .collect(),
960            context
961                .stage_prediction()
962                .io()
963                .iter()
964                .map(|prediction| {
965                    IoMeasurement::new(
966                        prediction.kind(),
967                        prediction.bytes(),
968                        prediction.operations(),
969                    )
970                })
971                .collect(),
972            self.phase_input_artifact
973                .filter(|_| context.node().id == self.prepare)
974                .map(|(identity, bytes)| {
975                    crate::ArtifactMeasurement::new(
976                        identity,
977                        Some(identity),
978                        crate::ArtifactDisposition::Loaded,
979                        bytes,
980                        None,
981                    )
982                    .map_err(io::Error::other)
983                })
984                .transpose()?
985                .into_iter()
986                .collect(),
987        ))
988    }
989    fn failure_measurements<'a>(&'a self, _: &'a io::Error) -> Option<&'a WorkMeasurements> {
990        None
991    }
992    fn wait_for_fence(
993        &self,
994        _context: WorkExecutionContext<'_>,
995        _fence: FenceKind,
996    ) -> io::Result<WorkMeasurements> {
997        #[cfg(test)]
998        if _context.node().id == self.read && _fence == FenceKind::Io {
999            if self.failure == Failure::SourceFence {
1000                return Err(io::Error::from_raw_os_error(5));
1001            }
1002            self.state.lock().unwrap().fenced = true;
1003        }
1004        Ok(WorkMeasurements::default())
1005    }
1006    fn complete_observation_read(
1007        &self,
1008        completion: ObservationReadCompletionContext,
1009    ) -> io::Result<AttemptBoundObservationCompletion> {
1010        #[cfg(test)]
1011        assert!(completion.settled_fences().contains(&FenceKind::Io));
1012        self.state
1013            .lock()
1014            .unwrap()
1015            .weighting
1016            .complete_replay(completion)
1017            .map_err(io::Error::other)
1018    }
1019    fn publish(&self, context: WorkExecutionContext<'_>) -> io::Result<()> {
1020        let ready = {
1021            let state = self.state.lock().unwrap();
1022            state.complete.is_some() || state.minor.is_some()
1023        };
1024        if context.node().kind != WorkKind::Publication || context.publication().is_none() || !ready
1025        {
1026            return Err(io::Error::other(
1027                "native cube commit lacks completed transaction authority",
1028            ));
1029        }
1030        #[cfg(test)]
1031        {
1032            self.state.lock().unwrap().published = true;
1033        }
1034        Ok(())
1035    }
1036
1037    fn retain_artifact_resources(
1038        &self,
1039        owner: &WorkNodeId,
1040        permit: RetainedArtifactPermit,
1041    ) -> io::Result<bool> {
1042        if self.cube_state.retains_at(owner) {
1043            self.cube_state.retain(owner, permit)?;
1044            return Ok(true);
1045        }
1046        if !self.native_plan.retains_at(owner) {
1047            return Ok(false);
1048        }
1049        let mut state = self.state.lock().unwrap();
1050        self.native_plan.retain(
1051            &mut state
1052                .native
1053                .as_mut()
1054                .ok_or_else(|| io::Error::other("native store is absent at retention"))?
1055                .store,
1056            permit,
1057        )?;
1058        Ok(true)
1059    }
1060
1061    fn abort_node_io(&self, owner: &WorkNodeId) -> io::Result<()> {
1062        if owner == &self.read {
1063            let mut state = self.state.lock().unwrap();
1064            state.native = None;
1065        }
1066        Ok(())
1067    }
