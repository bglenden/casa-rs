60#[derive(Debug, Clone, PartialEq, Eq, Error)]
61pub(crate) enum RowSelectionEvaluationError {
62    #[error(
63        "selected DATA_DESC_ID {data_description_id} has no positive finite reference wavelength"
64    )]
65    MissingReferenceWavelength { data_description_id: u32 },
66}
67
68pub(crate) struct CompiledRowPredicate {
69    catalog: PredicateCatalog,
70    wavelengths_by_ddid: Box<[(u32, f64)]>,
71}
72
73enum PredicateCatalog {
74    SharedSource(ObservationSource),
75    OwnedProjection {
76        selection: RowSelection,
77        data_descriptions: Box<[DataDescriptionSelection]>,
78    },
79}
80
81impl PredicateCatalog {
82    fn selection(&self) -> &RowSelection {
83        match self {
84            Self::SharedSource(source) => source.selection().rows_filter(),
85            Self::OwnedProjection { selection, .. } => selection,
86        }
87    }
88
89    fn data_descriptions(&self) -> &[DataDescriptionSelection] {
90        match self {
91            Self::SharedSource(source) => source.selection().data_descriptions(),
92            Self::OwnedProjection {
93                data_descriptions, ..
94            } => data_descriptions,
95        }
96    }
97}
98
99impl CompiledRowPredicate {
100    pub(crate) fn shared_retained_heap_bytes(source: &ObservationSource) -> Option<usize> {
101        let wavelength_bytes = if needs_reference_wavelengths(source.selection().rows_filter()) {
102            source
103                .selection()
104                .data_descriptions()
105                .len()
106                .checked_mul(size_of::<(u32, f64)>())?
107        } else {
108            0
109        };
110        source
111            .provenance()
112            .locator()
113            .len()
114            .checked_add(wavelength_bytes)
115    }
116
117    pub(crate) fn new_shared(
118        source: &ObservationSource,
119        reference_wavelength: impl FnMut(u32) -> Option<f64>,
120    ) -> Result<Self, RowSelectionEvaluationError> {
121        Self::from_catalog(
122            PredicateCatalog::SharedSource(source.clone()),
123            reference_wavelength,
124        )
125    }
126
127    pub(crate) fn new(
128        selection: &RowSelection,
129        data_descriptions: &[DataDescriptionSelection],
130        reference_wavelength: impl FnMut(u32) -> Option<f64>,
131    ) -> Result<Self, RowSelectionEvaluationError> {
132        Self::from_catalog(
133            PredicateCatalog::OwnedProjection {
134                selection: selection.clone(),
135                data_descriptions: data_descriptions.into(),
136            },
137            reference_wavelength,
138        )
139    }
140
141    fn from_catalog(
142        catalog: PredicateCatalog,
143        mut reference_wavelength: impl FnMut(u32) -> Option<f64>,
144    ) -> Result<Self, RowSelectionEvaluationError> {
145        let mut wavelengths_by_ddid = Vec::new();
146        if needs_reference_wavelengths(catalog.selection()) {
147            let data_descriptions = catalog.data_descriptions();
148            wavelengths_by_ddid.reserve(data_descriptions.len());
149            for description in data_descriptions {
150                let data_description_id = description.data_description_id();
151                let Some(wavelength_m) = reference_wavelength(data_description_id)
152                    .filter(|value| value.is_finite() && *value > 0.0)
153                else {
154                    return Err(RowSelectionEvaluationError::MissingReferenceWavelength {
155                        data_description_id,
156                    });
157                };
158                wavelengths_by_ddid.push((data_description_id, wavelength_m));
159            }
160        }
161        Ok(Self {
162            catalog,
163            wavelengths_by_ddid: wavelengths_by_ddid.into_boxed_slice(),
164        })
165    }
166
167    pub(crate) fn matches(&self, row: StoredMainRow) -> bool {
168        let selection = self.catalog.selection();
169        let data_descriptions = self.catalog.data_descriptions();
170        let Ok(data_description_id) = u32::try_from(row.data_description_id) else {
171            return false;
172        };
173        data_descriptions
174            .iter()
175            .any(|description| description.data_description_id() == data_description_id)
176            && id_matches(selection.fields(), row.field_id)
177            && time_matches(selection.times(), row.time_mjd_seconds)
178            && uv_matches(
179                selection.uv_distances(),
180                row.uvw_m,
181                data_description_id,
182                &self.wavelengths_by_ddid,
183            )
184            && antenna_matches(selection.antennas(), row.antenna1, row.antenna2)
185            && id_matches(selection.scans(), row.scan_number)
186            && id_matches(selection.observations(), row.observation_id)
187            && intent_matches(selection.intents(), row.state_id)
188            && id_matches(selection.arrays(), row.array_id)
189    }
190
191    pub(crate) fn requires_every_source_row(&self, data_description_count: usize) -> bool {
192        let selection = self.catalog.selection();
193        let data_descriptions = self.catalog.data_descriptions();
194        matches!(selection.fields(), IdSelection::All)
195            && matches!(selection.times(), TimeSelection::All)
196            && matches!(selection.uv_distances(), UvSelection::All)
197            && matches!(selection.antennas(), AntennaSelection::All)
198            && matches!(selection.scans(), IdSelection::All)
199            && matches!(selection.observations(), IdSelection::All)
200            && matches!(selection.intents(), IntentSelection::All)
201            && matches!(selection.arrays(), IdSelection::All)
202            && data_descriptions.len() == data_description_count
203            && data_descriptions
204                .iter()
205                .enumerate()
206                .all(|(index, description)| {
207                    usize::try_from(description.data_description_id()).ok() == Some(index)
208                })
209    }
210}
211
212fn needs_reference_wavelengths(selection: &RowSelection) -> bool {
213    matches!(selection.uv_distances(), UvSelection::Ranges(ranges) if ranges.iter().any(|range| range.unit() == UvDistanceUnit::Wavelengths))
214}
215
216fn id_matches(selection: &IdSelection, value: i32) -> bool {
217    match selection {
218        IdSelection::All => true,
219        IdSelection::Only(ids) => u32::try_from(value)
220            .ok()
221            .is_some_and(|value| ids.binary_search(&value).is_ok()),
222    }
223}
224
225fn time_matches(selection: &TimeSelection, value: f64) -> bool {
226    match selection {
227        TimeSelection::All => true,
228        TimeSelection::Ranges(ranges) => {
229            value.is_finite()
230                && ranges.iter().any(|range| {
231                    lower_matches(range.lower(), value) && upper_matches(range.upper(), value)
232                })
233        }
234    }
235}
236
237fn uv_matches(
238    selection: &UvSelection,
239    uvw_m: [f64; 3],
240    data_description_id: u32,
241    wavelengths_by_ddid: &[(u32, f64)],
242) -> bool {
243    match selection {
244        UvSelection::All => true,
245        UvSelection::Ranges(ranges) => {
246            let distance_m = uvw_m[0].hypot(uvw_m[1]);
247            distance_m.is_finite()
248                && ranges.iter().any(|range| {
249                    let value = match range.unit() {
250                        UvDistanceUnit::Meters => distance_m,
251                        UvDistanceUnit::Wavelengths => {
252                            let Some((_, wavelength_m)) = wavelengths_by_ddid
253                                .iter()
254                                .find(|(ddid, _)| *ddid == data_description_id)
255                            else {
256                                return false;
257                            };
258                            distance_m / wavelength_m
259                        }
260                    };
261                    lower_matches(range.lower(), value) && upper_matches(range.upper(), value)
262                })
263        }
264    }
265}
266
267fn lower_matches(bound: Option<SelectionBound>, value: f64) -> bool {
268    bound.is_none_or(|bound| {
269        if bound.is_inclusive() {
270            value >= bound.value()
271        } else {
272            value > bound.value()
273        }
274    })
275}
276
277fn upper_matches(bound: Option<SelectionBound>, value: f64) -> bool {
278    bound.is_none_or(|bound| {
279        if bound.is_inclusive() {
280            value <= bound.value()
281        } else {
282            value < bound.value()
283        }
284    })
285}
286
287fn antenna_matches(selection: &AntennaSelection, antenna1: i32, antenna2: i32) -> bool {
288    match selection {
289        AntennaSelection::All => true,
290        AntennaSelection::Only(baselines) => {
291            let (Ok(antenna1), Ok(antenna2)) = (u32::try_from(antenna1), u32::try_from(antenna2))
292            else {
293                return false;
294            };
295            let pair = if antenna1 <= antenna2 {
296                [antenna1, antenna2]
297            } else {
298                [antenna2, antenna1]
299            };
300            baselines
301                .binary_search_by_key(&pair, |baseline| baseline.antennas())
302                .is_ok()
303        }
304    }
305}
306
307fn intent_matches(selection: &IntentSelection, state_id: i32) -> bool {
308    match selection {
309        IntentSelection::All => true,
310        IntentSelection::Only(intents) => u32::try_from(state_id).ok().is_some_and(|state_id| {
311            intents
312                .binary_search_by_key(&state_id, |intent| intent.state_id())
313                .is_ok()
314        }),
315    }
316}

359    #[test]
360    fn compiled_row_predicate_evaluates_every_resolved_selector() {
361        let descriptions = [DataDescriptionSelection::new(6, 8, 10)];
362        let selection = exact_selection(UvSelection::Ranges(vec![UvDistanceRange::new(
363            Some(SelectionBound::inclusive(5.0)),
364            Some(SelectionBound::inclusive(5.0)),
365            UvDistanceUnit::Meters,
366        )]));
367        let predicate = CompiledRowPredicate::new(&selection, &descriptions, |_| None)
368            .expect("metre UV selection needs no wavelength metadata");
369        let row = matching_row();
370        assert!(predicate.matches(row));
371
372        for rejected in [
373            StoredMainRow {
374                data_description_id: 4,
375                ..row
376            },
377            StoredMainRow { field_id: 4, ..row },
378            StoredMainRow { antenna2: 2, ..row },
379            StoredMainRow {
380                time_mjd_seconds: 20.0,
381                ..row
382            },
383            StoredMainRow {
384                scan_number: 8,
385                ..row
386            },
387            StoredMainRow { state_id: 6, ..row },
388            StoredMainRow {
389                observation_id: 8,
390                ..row
391            },
392            StoredMainRow { array_id: 1, ..row },
393            StoredMainRow {
394                uvw_m: [6.0, 0.0, 0.0],
