5/// The image mask of a selection, combined with its region's and a LEL
6/// mask expression's. The expression is CASA's full lattice expression
7/// language, evaluated over the image and extended to its shape when
8/// `stretch` allows.
9pub(super) fn selection_pixel_mask<T>(
10    image: &PagedImage<T>,
11    effective_region: Option<&PixelRegion>,
12    selection: &Selection,
13    mask_expr: Option<&str>,
14    stretch: bool,
15) -> Result<Option<ndarray::ArrayD<bool>>, ImageError>
16where
17    T: ImagePixel + ImageExprValue + PartialOrd + ExprValueConvert,
18{
19    let mut pixel_mask =
20        image.get_mask_slice(&selection.start, &selection.shape, &vec![1; image.ndim()])?;
21    if let Some(region) = effective_region {
22        let region_mask = region_pixel_mask(region, selection)?;
23        pixel_mask = Some(match pixel_mask {
24            Some(mut mask) => {
25                Zip::from(&mut mask)
26                    .and(&region_mask)
27                    .for_each(|existing, region_valid| *existing = *existing && *region_valid);
28                mask
29            }
30            None => region_mask,
31        });
32    }
33    if let Some(expr) = mask_expr.filter(|text| !text.trim().is_empty()) {
34        let expr_mask = image_mask_expression(expr, image, selection, stretch)?;
35        pixel_mask = Some(match pixel_mask {
36            Some(mut mask) => {
37                Zip::from(&mut mask)
38                    .and(&expr_mask)
39                    .for_each(|existing, expr_valid| *existing = *existing && *expr_valid);
40                mask
41            }
42            None => expr_mask,
43        });
44    }
45    Ok(pixel_mask)
46}

48/// A CASA LEL mask expression over the image: parsed by the expression
49/// engine, evaluated over the whole image, extended to the image's shape
50/// when `stretch` allows it, and cut to the selection.
51pub(super) fn image_mask_expression<T>(
52    expr: &str,
53    image: &PagedImage<T>,
54    selection: &Selection,
55    stretch: bool,
56) -> Result<ndarray::ArrayD<bool>, ImageError>
57where
58    T: ImagePixel + ImageExprValue + PartialOrd + ExprValueConvert,
59{
60    // Open each referenced image as the
61    // target's pixel type and hand them to the parser's resolver.
62    let names = casa_images::expr_parser::referenced_images(expr).map_err(|error| {
63        ImageError::CasaSelection(format!(
64            "Input mask specification is incorrect: {}",
65            mask_error_wording(error)
66        ))
67    })?;
68    let mut opened: Vec<PagedImage<T>> = Vec::with_capacity(names.len());
69    for name in &names {
70        let opened_image = PagedImage::<T>::open(name).map_err(|_| {
71            ImageError::CasaSelection(format!(
72                "Input mask specification is incorrect: ImageExprParse: '{name}' \
73                 is an unknown lattice or image or it is an unqualified region"
74            ))
75        })?;
76        opened.push(opened_image);
77    }
78    let sources: std::collections::HashMap<String, &dyn ImageInterface<T>> = names
79        .iter()
80        .zip(opened.iter())
81        .map(|(name, opened)| (name.clone(), opened as &dyn ImageInterface<T>))
82        .collect();
83    let resolver = HashMapResolver(sources);
84    let expression = parse_mask_expr(expr, &resolver).map_err(|error| {
85        ImageError::CasaSelection(format!(
86            "Input mask specification is incorrect: {}",
87            mask_error_wording(error)
88        ))
89    })?;
90    let mask = expression.get()?;
91    let mask_shape = mask.shape().to_vec();
92    let target_shape = image.shape().to_vec();
93    let extended = if mask_shape == target_shape {
94        mask
95    } else {
96        let extendable = mask_shape.len() <= target_shape.len()
97            && mask_shape
98                .iter()
99                .zip(&target_shape)
100                .all(|(mask_axis, image_axis)| mask_axis == image_axis || *mask_axis == 1);
101        if !extendable {
102            return Err(ImageError::CasaSelection(
103                "Unable to extend mask: ExtendSpecifier - lengths of old axis mismatch".to_string(),
104            ));
105        }
106        if !stretch {
107            return Err(ImageError::CasaSelection(format!(
108                "The input image shape ({target_shape:?}) and mask shape ({mask_shape:?}) are \
109                 different, and it was specified that the mask should not be extended, so the \
110                 mask cannot be applied to the (sub)image. Specifying that the mask should be \
111                 extended may resolve the issue"
112            )));
113        }
114        extend_mask_to_shape(mask, &target_shape)?
115    };
116    let start = &selection.start;
117    let shape = &selection.shape;
118    let cut = extended.slice_each_axis(|axis| {
119        ndarray::Slice::from(start[axis.axis.0]..start[axis.axis.0] + shape[axis.axis.0])
120    });
121    Ok(cut.to_owned())
122}

124/// Tile a mask whose axes prefix the image's, each equal or degenerate,
125/// to the image's full shape: casacore extends along the image's later
126/// axes and broadcasts degenerate ones.
127pub(super) fn extend_mask_to_shape(
128    mask: ndarray::ArrayD<bool>,
129    target_shape: &[usize],
130) -> Result<ndarray::ArrayD<bool>, ImageError> {
131    let mask_shape = mask.shape().to_vec();
132    let tiled = ndarray::ArrayD::<bool>::from_shape_fn(ndarray::IxDyn(target_shape), |index| {
133        let target: Vec<usize> = index.slice().to_vec();
134        let source: Vec<usize> = mask_shape
135            .iter()
136            .enumerate()
137            .map(|(axis, &mask_axis)| target.get(axis).copied().unwrap_or(0).min(mask_axis - 1))
138            .collect();
139        mask[ndarray::IxDyn(&source)]
140    });
141    Ok(tiled)
142}

144/// The quoted image names a LEL expression carries.
145pub fn quoted_expression_names(expr: &str) -> Vec<String> {
146    let mut names = Vec::new();
147    let characters: Vec<char> = expr.chars().collect();
148    let mut index = 0;
149    while index < characters.len() {
150        let quote = characters[index];
151        if (quote == '"' || quote == '\'')
152            && let Some(end) = characters[index + 1..].iter().position(|&c| c == quote)
153        {
154            let name: String = characters[index + 1..index + 1 + end].iter().collect();
155            if !name.is_empty() && !names.contains(&name) {
156                names.push(name);
157            }
158            index += end + 2;
159            continue;
160        }
161        index += 1;
162    }
163    names
164}

166/// The selection's pixel values as a robust-algorithm sample, with each
167/// value's Fortran-linear position inside the selection, filtered by
168/// the pixel mask, an inclusive pixel-value range, and the named
169/// channel and Stokes sets.
170#[allow(clippy::too_many_arguments)]
171pub(super) fn sample_points<T>(
172    pixels: &ndarray::ArrayD<T>,
173    pixel_mask: Option<&ndarray::ArrayD<bool>>,
174    includepix: Option<[f64; 2]>,
175    channel_axis: Option<usize>,
176    channel_indices: Option<&[usize]>,
177    stokes_axis: Option<usize>,
178    stokes_indices: Option<&[usize]>,
179    start: &[usize],
180    cancellation: Option<&Cancellation>,
181) -> Result<Vec<SamplePoint>, ImageError>
182where
183    T: ImagePixel + StatsElement,
184{
185    let shape = pixels.shape();
186    let mut strides = Vec::with_capacity(shape.len());
187    let mut stride = 1u64;
188    for &axis_shape in shape {
189        strides.push(stride);
190        stride *= axis_shape as u64;
191    }
192    let mut sample = Vec::new();
193    for (visited, (index, value)) in pixels.indexed_iter().enumerate() {
194        if visited.is_multiple_of(4096) {
195            check_imstat_cancelled(cancellation)?;
196        }
197        if let Some(mask) = pixel_mask {
198            let position: &[usize] = index.slice();
199            if !mask[position] {
200                continue;
201            }
202        }
203        let number = value.to_f64_stats();
204        if let Some([min, max]) = includepix
205            && !(min <= number && number <= max)
206        {
207            continue;
208        }
209        let in_named_set = |axis: Option<usize>, indices: Option<&[usize]>| match (axis, indices) {
210            (Some(axis), Some(indices)) => {
211                let absolute = index[axis] + start[axis];
212                indices.contains(&absolute)
213            }
214            _ => true,
215        };
216        if !in_named_set(channel_axis, channel_indices)
217            || !in_named_set(stokes_axis, stokes_indices)
218        {
219            continue;
220        }
221        let mut linear = 0u64;
222        for (axis, &axis_stride) in strides.iter().enumerate() {
223            linear += index[axis] as u64 * axis_stride;
224        }
225        sample.push(SamplePoint {
226            value: number,
227            position: linear,
228        });
229    }
230    // casacore walks its arrays in Fortran order, so a tie's reported
231    // position is the first in that order.
232    check_imstat_cancelled(cancellation)?;
233    sample.sort_by_key(|point| point.position);
234    Ok(sample)
235}

237/// The position of a Fortran-linear index inside a selection shape.
238pub(super) fn selection_position(linear: u64, shape: &[usize]) -> Vec<usize> {
239    let mut position = Vec::with_capacity(shape.len());
240    let mut remainder = linear;
241    for &axis_shape in shape {
242        position.push((remainder % axis_shape as u64) as usize);
243        remainder /= axis_shape.max(1) as u64;
244    }
245    position
246}

248/// The inclusive top-right corner of a selection, as pixels.
249pub(super) fn trc_pixels(start: &[usize], shape: &[usize]) -> Vec<usize> {
250    start
251        .iter()
252        .zip(shape)
253        .map(|(start, shape)| start + shape - 1)
254        .collect()
255}

257/// One statistic of the selection. With no valid pixel, a count or a sum
258/// is 0 and every other statistic NaN; CASA returns each as an empty array.
259pub(super) fn scalar_stat<T: StatsElement>(
260    stats: &LatticeStatistics<'_, T>,
261    stat: Statistic,
262) -> Result<f64, ImageError> {
263    let values = stats.get_statistic(stat)?;
264    Ok(values.iter().next().copied().unwrap_or(match stat {
265        Statistic::Npts | Statistic::Sum | Statistic::SumSq => 0.0,
266        _ => f64::NAN,
267    }))
268}

271    start
272        .iter()
273        .zip(pos)
274        .map(|(base, offset)| base + offset)

278pub(super) fn combined_selection_mask<T: ImagePixel>(
279    image: &dyn ImageInterface<T>,
280    selection: &imsubimage::region::SubimageSelection,
281    mask_expr: Option<&str>,
282    stretch: bool,
283) -> Result<Option<ArrayD<bool>>, ImageError> {
284    let mut base =
285        image.get_mask_slice(&selection.start, &selection.shape, &vec![1; image.ndim()])?;
286    let expression = lel_mask::evaluate_mask(
287        mask_expr,
288        image.shape(),
289        image.coordinates(),
290        stretch,
291        (&selection.start, &selection.shape),
292    )?;
293    if let Some(expression) = expression {
294        if let Some(base) = &mut base {
295            Zip::from(base)
296                .and(&expression)
297                .for_each(|valid, expression| *valid &= *expression);
298        } else {
299            base = Some(expression);
300        }
301    }
302    Ok(base)
