2//! Complex-valued nodes used by the mixed-pixel LEL mask parser.
3
4use super::{
5    Complex64, ExprNode, ExprParseError, ExprValueConvert, ImageExprBinaryOp, ImageExprBuilder,
6    ImageExprCompareOp, ImageExprUnaryOp, ImageExprValue, MaskExprBuilder,
7};
8use crate::ImageError;
9
10fn error(position: usize) -> impl FnOnce(ImageError) -> ExprParseError {
11    move |error| ExprParseError {
12        message: error.to_string(),
13        position,
14    }
15}
16
17fn promote<T: ImageExprValue + PartialOrd + ExprValueConvert>(
18    node: ExprNode<'_, T>,
19    pos: usize,
20) -> Result<ImageExprBuilder<'_, Complex64>, ExprParseError> {
21    match node {
22        ExprNode::Complex(expr) => Ok(*expr),
23        ExprNode::Scalar(value) => Ok(ImageExprBuilder::scalar(Complex64::new(value, 0.0))),
24        other => other
25            .into_numeric(pos)?
26            .map_pixel_type(|value| Complex64::new(value.to_f64(), 0.0))
27            .map_err(error(pos)),
28    }
29}
30
31pub(super) fn arithmetic_complex<'a, T: ImageExprValue + PartialOrd + ExprValueConvert>(
32    lhs: ExprNode<'a, T>,
33    rhs: ExprNode<'a, T>,
34    op: ImageExprBinaryOp,
35    pos: usize,
36) -> Result<ExprNode<'a, T>, ExprParseError> {
37    promote(lhs, pos)?
38        .binary_expr(promote(rhs, pos)?, op)
39        .map(|expr| ExprNode::Complex(Box::new(expr)))
40        .map_err(error(pos))
41}
42
43pub(super) fn compare_complex<'a, T: ImageExprValue + PartialOrd + ExprValueConvert>(
44    lhs: ExprNode<'a, T>,
45    rhs: ExprNode<'a, T>,
46    op: ImageExprCompareOp,
47    pos: usize,
48) -> Result<ExprNode<'a, T>, ExprParseError> {
49    let mask = promote(lhs, pos)?
50        .compare_expr(promote(rhs, pos)?, op)
51        .map_err(error(pos))?;
52    let data = mask
53        .compile()
54        .and_then(|mask| mask.get())
55        .map_err(error(pos))?;
56    Ok(ExprNode::Mask(MaskExprBuilder::from_constant(data)))
57}
58
59pub(super) fn complex_function<'a, T: ImageExprValue + PartialOrd + ExprValueConvert>(
60    name: &str,
61    expr: ImageExprBuilder<'a, Complex64>,
62    op: Option<ImageExprUnaryOp>,
63    pos: usize,
64) -> Result<ExprNode<'a, T>, ExprParseError> {
65    let convert: fn(Complex64) -> f64 = match name {
66        "abs" | "amplitude" => Complex64::norm,
67        "real" => |value| value.re,
68        "imag" => |value| value.im,
69        "arg" => Complex64::arg,
70        "value" => return Ok(ExprNode::Complex(Box::new(expr.strip_mask()))),
71        "mask" => {
72            let mask = expr
73                .source_mask()
74                .map_err(error(pos))?
75                .unwrap_or_else(|| ndarray::ArrayD::from_elem(ndarray::IxDyn(expr.shape()), true));
76            return Ok(ExprNode::Mask(MaskExprBuilder::from_constant(mask)));
77        }
78        _ if op.is_some() => return Ok(ExprNode::Complex(Box::new(expr.unary(op.unwrap())))),
79        _ => {
80            return Err(ExprParseError {
81                message: format!("unsupported complex argument to '{name}'"),
82                position: pos,
83            });
84        }
85    };
86    let expr = expr
87        .map_pixel_type(move |value| T::from_f64(convert(value)))
88        .map_err(error(pos))?;
89    Ok(ExprNode::Numeric(Box::new(expr)))
90}
