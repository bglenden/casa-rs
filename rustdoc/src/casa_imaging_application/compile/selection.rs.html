67/// Per-row facts gathered by the pass.
68struct RowFacts {
69    rows: SelectedRowsBuilder,
70    rows_error: Option<casa_imaging_model::SelectedRowSequenceError>,
71    ddids: BTreeSet<i32>,
72    fields: BTreeSet<i32>,
73    observation_ids: BTreeSet<i32>,
74    first_time_mjd_seconds: Option<f64>,
75    time_bounds_mjd_seconds: [f64; 2],
76    first_cross_row: Option<SelectedObservationRow>,
77    w: WRange,
78}
79
80impl RowFacts {
81    /// Nothing observed yet, over `row_count` MAIN rows and `ddids`
82    /// candidate data descriptions.
83    fn new(row_count: usize, ddids: usize) -> Result<Self, ApplicationError> {
84        Ok(Self {
85            rows: SelectedRowsBuilder::with_data_description_capacity(
86                u64::try_from(row_count).map_err(|_| boxed("MS row count exceeds u64"))?,
87                ddids,
88            ),
89            rows_error: None,
90            ddids: BTreeSet::new(),
91            fields: BTreeSet::new(),
92            observation_ids: BTreeSet::new(),
93            first_time_mjd_seconds: None,
94            time_bounds_mjd_seconds: [f64::INFINITY, f64::NEG_INFINITY],
95            first_cross_row: None,
96            w: WRange {
97                maximum_abs_m: 0.0,
98                minimum_abs_m: f64::INFINITY,
99                sum_squares_m2: 0.0,
100                rows: 0,
101            },
102        })
103    }
104
105    fn observe(&mut self, row: SelectedObservationRow) {
106        self.ddids.insert(row.data_description_id());
107        self.fields.insert(row.field_id());
108        self.observation_ids.insert(row.observation_id());
109        self.first_time_mjd_seconds
110            .get_or_insert(row.time_mjd_seconds());
111        if !row.flag_row() && row.antenna1() != row.antenna2() {
112            self.first_cross_row.get_or_insert(row);
113        }
114        self.time_bounds_mjd_seconds[0] =
115            self.time_bounds_mjd_seconds[0].min(row.time_mjd_seconds());
116        self.time_bounds_mjd_seconds[1] =
117            self.time_bounds_mjd_seconds[1].max(row.time_mjd_seconds());
118        let [_, _, w_m] = row.uvw_m();
119        self.w.maximum_abs_m = self.w.maximum_abs_m.max(w_m.abs());
120        self.w.minimum_abs_m = self.w.minimum_abs_m.min(w_m.abs());
121        self.w.sum_squares_m2 += w_m * w_m;
122        self.w.rows += 1;
123        if self.rows_error.is_none() {
124            self.rows_error = self
125                .rows
126                .push(SelectedMainRow::new(
127                    u64::try_from(row.physical_row()).expect("row bounded by MS row count"),
128                    u32::try_from(row.data_description_id()).expect("validated nonnegative DDID"),
129                ))
130                .err();
131        }
132    }
133}

135/// Resolve the request's selection and survey its rows in one pass.
136pub(super) fn survey(
137    request: &ImagingRequest,
138    ms: &MeasurementSet,
139    budget: SelectedObservationContentBudget,
140    frame_engine: &casa_ms::derived::engine::MsCalEngine,
141) -> Result<Survey, ApplicationError> {
142    let data_description = ms.data_description()?;
143    let ddids = selected_data_descriptions(request, &data_description)?;
144    let row_selection = ms.selected_observation_row_selection(
145        &ddids,
146        request.field.as_deref(),
147        request.uvrange.as_deref(),
148        request.intent.as_deref(),
149    )?;
150    let mut spectral_windows = candidate_windows(ms, &data_description, &ddids)?;
151    let mut continuum = continuum_envelope(
152        request,
153        ms,
154        &row_selection,
155        &spectral_windows,
156        frame_engine,
157        budget,
158    )?;
159    let mut io = budget.row_io_budget();
160    io.available_bytes = io
161        .available_bytes
162        .checked_sub(
163            continuum
164                .as_ref()
165                .map_or(0, |(_, reducer)| reducer.retained_bytes()),
166        )
167        .ok_or_else(|| boxed("selected spectral envelope exhausts the row traversal budget"))?;
168    let mut facts = RowFacts::new(ms.row_count(), ddids.len())?;
169    let main_table = ms.main_table();
170    let mut weight_spectrum_complete = main_table.column_accessor("WEIGHT_SPECTRUM").is_ok();
171    let mut row_error = None;
172    ms.visit_selected_observation_rows(&row_selection, io, |row| {
173        if row_error.is_none()
174            && let Some((_, reducer)) = continuum.as_mut()
175        {
176            row_error = reducer
177                .observe(row)
178                .err()
179                .map(|error| Box::new(error) as ApplicationError);
180        }
181        if weight_spectrum_complete {
182            match main_table
183                .column_accessor("WEIGHT_SPECTRUM")
184                .and_then(|column| column.array_cell_is_defined_uncached(row.physical_row()))
185            {
186                Ok(defined) => weight_spectrum_complete = defined,
187                Err(error) => row_error = Some(Box::new(error)),
188            }
189        }
190        facts.observe(row);
191    })?;
192    if let Some(error) = facts.rows_error {
193        return Err(Box::new(error));
194    }
195    if let Some(error) = row_error {
196        return Err(error);
197    }
198    let rows = facts.rows.finish();
199    if rows.selected_row_count() == 0 {
200        return Err(boxed("selection resolved to no rows"));
201    }
202    let bindings = facts
203        .ddids
204        .iter()
205        .map(|ddid| data_description_binding(&data_description, *ddid))
206        .collect::<Result<Vec<_>, _>>()?;
207    let spw_ids = bindings
208        .iter()
209        .map(|(spw_id, _)| *spw_id)
210        .collect::<BTreeSet<_>>();
211    spectral_windows.retain(|window| spw_ids.contains(&window.spw_id));
212    let source_frequency_reference = one_source_frame(&spectral_windows)?;
213    let continuum = continuum
214        .map(|(mut selected, reducer)| {
215            selected.retain(|spw_id, _| spw_ids.contains(spw_id));
216            reducer.finish().map(|envelope| (selected, envelope))
217        })
218        .transpose()?;
219    Ok(Survey {
220        row_selection,
221        rows,
222        bindings,
223        spectral_windows,
224        source_frequency_reference,
225        fields: facts.fields,
226        observation_ids: facts.observation_ids,
227        first_time_mjd_seconds: facts
228            .first_time_mjd_seconds
229            .expect("nonempty selected row traversal"),
230        time_bounds_mjd_seconds: facts.time_bounds_mjd_seconds,
231        first_cross_row: facts.first_cross_row,
232        w: facts.w,
233        weight_spectrum_complete,
234        continuum,
235    })
236}

238/// A continuum run's selected channels per candidate window, with the
239/// reducer of their LSRK envelope that the row pass feeds; `None` for a
240/// cube.
241fn continuum_envelope<'a>(
242    request: &ImagingRequest,
243    ms: &'a MeasurementSet,
244    row_selection: &'a SelectedObservationRowSelection,
245    spectral_windows: &[SourceSpectralWindow],
246    frame_engine: &'a casa_ms::derived::engine::MsCalEngine,
247    budget: SelectedObservationContentBudget,
248) -> Result<
249    Option<(
250        WindowChannels,
251        SelectedObservationSpectralEnvelopeReducer<'a>,
252    )>,
253    ApplicationError,
254> {
255    if request.specmode != SpecMode::Mfs {
256        return Ok(None);
257    }
258    let selected = spectral_windows
259        .iter()
260        .map(|window| {
261            Ok((
262                window.spw_id,
263                selected_channels(request, window.spw_id, &window.frequencies_hz)?,
264            ))
265        })
266        .collect::<Result<BTreeMap<_, _>, ApplicationError>>()?;
267    let reducer = ms.selected_observation_spectral_envelope_reducer(
268        row_selection,
269        spectral_windows.iter().map(|window| {
270            SelectedObservationSpectralWindow::borrow_selected(
271                u32::try_from(window.spw_id).expect("nonnegative i32 SPW fits u32"),
272                window.frequency_reference,
273                &window.frequencies_hz,
274                &window.channel_widths_hz,
275                selected
276                    .get(&window.spw_id)
277                    .expect("selected every candidate spectral window"),
278            )
279        }),
280        FrequencyRef::LSRK,
281        frame_engine,
282        budget.available_bytes(),
283    )?;
284    Ok(Some((selected, reducer)))
285}

287/// The spectral windows of the candidate data descriptions.
288fn candidate_windows(
289    ms: &MeasurementSet,
290    data_description: &casa_ms::MsDataDescription<'_>,
291    ddids: &[i32],
292) -> Result<Vec<SourceSpectralWindow>, ApplicationError> {
293    let spectral_window = ms.spectral_window()?;
294    let spw_ids = ddids
295        .iter()
296        .map(|ddid| data_description_binding(data_description, *ddid).map(|(spw_id, _)| spw_id))
297        .collect::<Result<BTreeSet<_>, _>>()?;
298    spw_ids
299        .into_iter()
300        .map(|spw_id| {
301            Ok(SourceSpectralWindow {
302                spw_id,
303                frequency_reference: FrequencyRef::from_casacore_code(
304                    spectral_window.meas_freq_ref(spw_id)?,
305                )
306                .ok_or_else(|| boxed("selected SPW has an unsupported frequency frame"))?,
307                frequencies_hz: spectral_window.chan_freq(spw_id)?,
308                channel_widths_hz: spectral_window.chan_width(spw_id)?,
309            })
310        })
311        .collect()
312}

314/// The one source frame every selected window shares.
315fn one_source_frame(windows: &[SourceSpectralWindow]) -> Result<FrequencyRef, ApplicationError> {
316    let mut frames = windows.iter().map(|window| window.frequency_reference);
317    let first = frames
318        .next()
319        .ok_or_else(|| boxed("selection holds no spectral window"))?;
320    if frames.any(|frame| frame != first) {
321        return Err(boxed(
322            "selected spectral windows use different source frequency frames",
323        ));
324    }
325    Ok(first)
326}

328/// The one selected spectral window of a stage that images one; `stage`
329/// names it in the error.
330pub(super) fn one_window<'a>(
331    windows: &'a [SourceSpectralWindow],
332    stage: &str,
333) -> Result<&'a SourceSpectralWindow, ApplicationError> {
334    match windows {
335        [window] => Ok(window),
336        _ => Err(boxed(format!(
337            "{stage} requires exactly one selected spectral window"
338        ))),
339    }

342fn selected_data_descriptions(
343    request: &ImagingRequest,
344    table: &casa_ms::MsDataDescription<'_>,
345) -> Result<Vec<i32>, ApplicationError> {
346    if let Some(ddid) = request.ddid {
347        return Ok(vec![ddid]);
348    }
349    let selected_spws = request
350        .spw
351        .as_deref()
352        .map(parse_spw_selector)
353        .transpose()?
354        .unwrap_or_default()
355        .into_iter()
356        .map(|selector| selector.spw_id)
357        .collect::<BTreeSet<_>>();
358    let mut ddids = Vec::new();
359    for row in 0..table.row_count() {
360        let spw = table.spectral_window_id(row)?;
361        let polarization = table.polarization_id(row)?;
362        if spw >= 0
363            && polarization >= 0
364            && (selected_spws.is_empty() || selected_spws.contains(&spw))
365        {
366            ddids.push(i32::try_from(row).map_err(|_| boxed("DDID exceeds i32"))?);
367        }
368    }
369    if ddids.is_empty() {
370        return Err(boxed("selection resolved to no data descriptions"));
371    }
372    Ok(ddids)
373}
374
375fn data_description_binding(
376    table: &casa_ms::MsDataDescription<'_>,
377    ddid: i32,
378) -> Result<(usize, usize), ApplicationError> {
379    let ddid = usize::try_from(ddid).map_err(|_| boxed("selected DATA_DESC_ID is negative"))?;
380    let spw = table.spectral_window_id(ddid)?;
381    let polarization = table.polarization_id(ddid)?;
382    Ok((
383        usize::try_from(spw).map_err(|_| boxed("selected DDID has a negative SPW id"))?,
384        usize::try_from(polarization)
385            .map_err(|_| boxed("selected DDID has a negative polarization id"))?,
386    ))
387}

389/// The channels of `spw_id` the request selects: those its `spw`
390/// selector names, else `channel_start` and `channel_count`.
391pub(super) fn selected_channels(
392    request: &ImagingRequest,
393    spw_id: usize,
394    frequencies: &[f64],
395) -> Result<Vec<usize>, ApplicationError> {
396    if let Some(channels) = explicit_spw_channels(request, spw_id, frequencies)? {
397        return Ok(channels);
398    }
399    let start = request.channel_start.unwrap_or(0);
400    let count = request
401        .channel_count
402        .unwrap_or_else(|| frequencies.len().saturating_sub(start));
403    let end = start
404        .checked_add(count)
405        .ok_or_else(|| boxed("channel range overflows usize"))?;
406    if count == 0 || end > frequencies.len() {
407        return Err(boxed("selected channel range is empty or out of bounds"));
408    }
409    Ok((start..end).collect())
410}

412/// The channels of `spw_id` the request's `spw` selector names, if it
413/// names any.
414pub(super) fn explicit_spw_channels(
415    request: &ImagingRequest,
416    spw_id: usize,
417    frequencies: &[f64],
418) -> Result<Option<Vec<usize>>, ApplicationError> {
419    let Some(text) = request.spw.as_deref() else {
420        return Ok(None);
421    };
422    let Some(selector) = parse_spw_selector(text)?
423        .into_iter()
424        .find(|selector| usize::try_from(selector.spw_id).ok() == Some(spw_id))
425        .and_then(|selector| selector.channels)
426    else {
427        return Ok(None);
428    };
429    Ok(Some(
430        resolve_channel_selector_selection(frequencies, &selector)?.indices,
431    ))
432}

434/// The observation selection: the surveyed rows, `channels` per window and
435/// every correlation of the selected polarization setups.
436pub(super) fn observation_selection(
437    ms: &MeasurementSet,
438    survey: Survey,
439    channels: &WindowChannels,
440) -> Result<ObservationSelection, ApplicationError> {
441    let polarization = ms.polarization()?;
442    let spectral_windows = channels
443        .iter()
444        .map(|(spw_id, channels)| {
445            Ok(SpectralWindowSelection::new(
446                u32::try_from(*spw_id).map_err(|_| boxed("SPW id exceeds u32"))?,
447                channels
448                    .iter()
449                    .map(|channel| {
450                        u32::try_from(*channel).map_err(|_| boxed("channel exceeds u32"))
451                    })
452                    .collect::<Result<Vec<_>, _>>()?,
453            ))
454        })
455        .collect::<Result<Vec<_>, ApplicationError>>()?;
456    let correlations = survey
457        .bindings
458        .iter()
459        .map(|(_, polarization_id)| *polarization_id)
460        .collect::<BTreeSet<_>>()
461        .into_iter()
462        .map(|polarization_id| {
463            let correlations = polarization
464                .corr_type(polarization_id)?
465                .iter()
466                .enumerate()
467                .map(|(index, code)| {
468                    Ok(CorrelationProduct::new(
469                        u32::try_from(index).map_err(|_| boxed("correlation index exceeds u32"))?,
470                        correlation_type(*code)?,
471                    ))
472                })
473                .collect::<Result<Vec<_>, ApplicationError>>()?;
474            Ok(CorrelationSelection::new(
475                u32::try_from(polarization_id).map_err(|_| boxed("polarization id exceeds u32"))?,
476                correlations,
477            ))
478        })
479        .collect::<Result<Vec<_>, ApplicationError>>()?;
480    Ok(ObservationSelection::new(
481        survey.rows,
482        survey.row_selection.rows().clone(),
483        survey.row_selection.data_descriptions().to_vec(),
484        spectral_windows,
485        correlations,
486    ))
487}

489/// The visibility column imaged: the requested one, else `CORRECTED_DATA`
490/// when present, else `DATA`.
491pub(super) fn visibility_column(
492    ms: &MeasurementSet,
493    requested: Option<DataColumn>,
494) -> Result<OwnerVisibilityColumn, ApplicationError> {
495    Ok(match requested {
496        Some(DataColumn::Data) => OwnerVisibilityColumn::Data,
497        Some(DataColumn::Corrected) => OwnerVisibilityColumn::CorrectedData,
498        None if ms.data_column(VisibilityDataColumn::CorrectedData).is_ok() => {
499            OwnerVisibilityColumn::CorrectedData
500        }
501        None if ms.data_column(VisibilityDataColumn::Data).is_ok() => OwnerVisibilityColumn::Data,
502        None => return Err(boxed("MS has neither CORRECTED_DATA nor DATA")),
503    })
504}

506fn correlation_type(code: i32) -> Result<CorrelationType, ApplicationError> {
507    use CorrelationType::*;
508    Ok(match code {
509        1 => StokesI,
510        2 => StokesQ,
511        3 => StokesU,
512        4 => StokesV,
513        5 => CircularRr,
514        6 => CircularRl,
515        7 => CircularLr,
516        8 => CircularLl,
517        9 => LinearXx,
518        10 => LinearXy,
519        11 => LinearYx,
520        12 => LinearYy,
521        13 => MixedRx,
522        14 => MixedRy,
523        15 => MixedLx,
524        16 => MixedLy,
525        17 => MixedXr,
526        18 => MixedXl,
527        19 => MixedYr,
528        20 => MixedYl,
529        21 => QuasiOrthogonalPp,
530        22 => QuasiOrthogonalPq,
531        23 => QuasiOrthogonalQp,
532        24 => QuasiOrthogonalQq,
533        25 => RightCircular,
534        26 => LeftCircular,
535        27 => Linear,
536        28 => PolarizedIntensity,
537        29 => LinearPolarizedIntensity,
538        30 => FractionalPolarizedIntensity,
539        31 => FractionalLinearPolarizedIntensity,
540        32 => PolarizationAngle,
541        _ => return Err(boxed(format!("unsupported correlation code {code}"))),
542    })
