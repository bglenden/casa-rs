33pub(super) fn check_cancelled(cancellation: Option<&Cancellation>) -> Result<(), ImageError> {
34    if cancellation.is_some_and(Cancellation::is_cancelled) {
35        return Err(ImageError::Interrupted);
36    }
37    Ok(())
38}

40pub(super) fn run_paged<T>(
41    image: &PagedImage<T>,
42    request: &ImmomentsRequest,
43    cancellation: Option<&Cancellation>,
44) -> Result<MomentMapSummary, ImageError>
45where
46    T: ImagePixel + Into<f64> + From<f32> + Copy,
47{
48    let record = match image.table().keywords().get("coords") {
49        Some(Value::Record(record)) => Some(record),
50        _ => None,
51    };
52    check_cancelled(cancellation)?;
53    let history = image_history_entries(&request.imagename)?;
54    run(image, record, &history, request, cancellation)
55}

57pub(super) fn run<T>(
58    image: &dyn ImageInterface<T>,
59    record: Option<&RecordValue>,
60    history: &[ImageHistoryEntry],
61    request: &ImmomentsRequest,
62    cancellation: Option<&Cancellation>,
63) -> Result<MomentMapSummary, ImageError>
64where
65    T: ImagePixel + Into<f64> + From<f32> + Copy,
66{
67    check_cancelled(cancellation)?;
68    let clip = PixelClip::from_request(request)?;
69    let destinations = output::destinations(request, clip)?;
70    let selection = SubimageSelection::resolve(
71        image,
72        request.box_pixels.as_deref(),
73        request.region.as_deref().map(SubimageRegion::Text),
74        request.chans.as_deref(),
75        request.stokes.as_deref(),
76    )?;
77    let axis = axis::resolve(image, request.axis.as_ref(), &selection.shape)?;
78    check_cancelled(cancellation)?;
79    let mut input = image.get_slice(&selection.start, &selection.shape, &vec![1; image.ndim()])?;
80    check_cancelled(cancellation)?;
81    let mut mask =
82        combined_selection_mask(image, &selection, request.mask.as_deref(), request.stretch)?;
83    selection.apply_selection_masks(&mut mask);
84    check_cancelled(cancellation)?;
85    let mut shape = selection.shape.clone();
86    shape[axis] = 1;
87    let coordinates = subimage_coordinates_with_stokes(
88        image.coordinates(),
89        &selection.start,
90        &selection.shape,
91        selection.selected_stokes.as_deref(),
92    )?;
93    let record = record.cloned().and_then(|record| {
94        subimage_coordinate_record(
95            record,
96            &selection.start,
97            &selection.shape,
98            selection.selected_stokes.as_deref(),
99        )
100    });
101    let mut info = image.image_info()?;
102    info.beam_set = subset_beam_set(
103        &info.beam_set,
104        image.coordinates(),
105        image.ndim(),
106        &selection.start,
107        &selection.shape,
108    )?;
109    let brightness = if image.find_axis(casa_coordinates::CoordinateType::Spectral) == Some(axis)
110        && info.beam_set.is_multi()
111    {
112        let (target, brightness) = crate::convolve2d::convolve_to_common_beam(
113            &mut input,
114            mask.as_ref(),
115            &coordinates,
116            image.units(),
117            &info.beam_set,
118            || check_cancelled(cancellation),
119        )?;
120        info.beam_set = ImageBeamSet::new(target);
121        brightness
122    } else {
123        image.units().to_owned()
124    };
125    if selection.stokes_axis == Some(axis) && info.beam_set.n_stokes() > 1 {
126        return Err(ImageError::CasaSelection(
127            "Number of polarizations is not consistent with the restoring beam set".into(),
128        ));
129    }
130    let axis = axis::MomentAxis::new(&coordinates, axis)?;
131    let positions = axis.profiles(&selection.shape, cancellation)?;
132    let results = collapse::collapse_moments(
133        &input,
134        mask.as_ref(),
135        axis.index,
136        &positions,
137        axis.integrated_scale()?,
138        clip,
139        &request.moments,
140        cancellation,
141    )?;
142    let misc = image.misc_info();
143    let command = [
144        concat!("version: casa-rs ", env!("CARGO_PKG_VERSION")).into(),
145        task_history(request),
146    ];
147    check_cancelled(cancellation)?;
148    let stage = destinations.stage()?;
149    let mut products = Vec::with_capacity(destinations.paths.len());
150    for (i, (&moment, result)) in request.moments.iter().zip(results).enumerate() {
151        check_cancelled(cancellation)?;
152        let units = moment_units(&brightness, &axis.unit, moment);
153        let mut output = TempImage::<f32>::new(
154            shape.clone(),
155            coordinates.clone(),
156            casa_lattices::TempStoragePolicy::Auto {
157                memory_budget_bytes: 64 * 1024 * 1024,
158                scratch: casa_lattices::ScratchSpace::SystemTemp,
159            },
160        )?;
161        output.set_units(units.clone())?;
162        output.set_image_info(&info)?;
163        output.set_misc_info(misc.clone())?;
164        output.put_slice(&result.values, &vec![0; shape.len()])?;
165        output.put_mask("mask0", &result.valid)?;
166        output.set_default_mask("mask0")?;
167        let staged = stage.path().join(i.to_string());
168        output.save_as(&staged)?;
169        if let Some(record) = &record {
170            patch_saved_coords_keyword(&staged, record.clone())?;
171        }
172        casa_images::append_image_history_entries(&staged, history)?;
173        casa_images::add_image_history(&staged, "immoments", &command)?;
174        products.push(MomentImageSummary {
175            outfile: destinations.paths[i].display().to_string(),
176            moment,
177            units,
178            valid_profiles: result.valid.iter().filter(|&&value| value).count(),
179        });
180    }
181    #[cfg(feature = "immoments-test-hooks")]
182    observe(Checkpoint::BeforePublication);
183    // The complete product set is committed together by the existing
184    // publication and rollback sequence once this check accepts the run.
185    check_cancelled(cancellation)?;
186    destinations.publish(stage)?;
187    Ok(MomentMapSummary {
188        imagename: request.imagename.display().to_string(),
189        shape,
190        products,
191    })
192}

194fn moment_units(brightness: &str, axis: &str, moment: i32) -> String {
195    match moment {
196        0 if !brightness.is_empty() && !axis.is_empty() => format!("{brightness}.{axis}"),
197        0 => String::new(),
198        2 if axis != "km/s" && !axis.is_empty() => format!("{axis}.{axis}"),
199        1 | 2 | 4 | 9 | 11 => axis.to_owned(),
200        _ => brightness.to_owned(),
201    }
202}
203
204fn task_history(request: &ImmomentsRequest) -> String {
205    let option = |value: Option<&str>| value.unwrap_or("").to_owned();
206    let axis = match &request.axis {
207        Some(MomentAxis::Index(index)) => index.to_string(),
208        Some(MomentAxis::Name(name)) => format!("{name:?}"),
209        None => "-1".to_string(),
210    };
211    format!(
212        "immoments(imagename={:?}, moments={:?}, axis={}, region={:?}, box={:?}, chans={:?}, stokes={:?}, mask={:?}, includepix={:?}, excludepix={:?}, outfile={:?}, stretch={})",
213        request.imagename.display().to_string(),
214        request.moments,
215        axis,
216        option(request.region.as_deref()),
217        option(request.box_pixels.as_deref()),
218        option(request.chans.as_deref()),
219        option(request.stokes.as_deref()),
220        option(request.mask.as_deref()),
221        request.includepix,
222        request.excludepix,
223        request.outfile.display().to_string(),
224        if request.stretch { "True" } else { "False" },
225    )
