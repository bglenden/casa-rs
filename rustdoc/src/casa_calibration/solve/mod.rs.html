67/// Supported first-wave gain solve families.
68#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
69pub enum GainType {
70    /// Per-receptor complex gains.
71    G,
72    /// Polarization-collapsed scalar complex gains.
73    T,
74    /// Per-receptor delay in nanoseconds.
75    K,
76    /// Multi-band K delay table.
77    KMbd,
78    /// Shared cross-hand delay in nanoseconds.
79    #[serde(rename = "KCROSS")]
80    KCross,
81    /// Direct KAntPos table with the antenna delay fit in its first two terms.
82    KAntPos,
83    /// Time-domain cubic-spline gains per receptor.
84    #[serde(rename = "GSPLINE")]
85    GSpline,
86}
87
88impl GainType {
89    pub(crate) fn vis_cal(self) -> &'static str {
90        match self {
91            Self::G => "G Jones",
92            Self::T => "T Jones",
93            Self::K => "K Jones",
94            Self::KMbd => "KMBD Jones",
95            Self::KCross => "Kcross Jones",
96            Self::KAntPos => "KAntPos Jones",
97            Self::GSpline => "GSPLINE",
98        }
99    }
100
101    /// The polarizations a solution holds: one gain per receptor for G, one
102    /// for both for T.
103    pub(crate) fn polarizations(self) -> usize {
104        match self {
105            Self::G | Self::K | Self::KMbd | Self::KCross | Self::GSpline => 2,
106            Self::KAntPos => 3,
107            Self::T => 1,
108        }
109    }
110
111    pub(crate) fn is_delay(self) -> bool {
112        matches!(self, Self::K | Self::KMbd | Self::KCross | Self::KAntPos)
113    }
114}

158/// Reweighting and outlier rejection for a routine gain solve.
159#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize, JsonSchema)]
160pub struct GainSolveRobustness {
161    /// Reweight the samples by their complex residuals in three L1 passes.
162    pub l1: bool,
163    /// Successive rejection thresholds in multiples of each correlation's RMS.
164    pub rms_thresholds: Vec<f32>,
165}
166
167impl GainSolveRobustness {
168    /// Parse CASA's documented `solmode` and optional `rmsthresh` sequence.
169    #[must_use]
170    pub fn from_casa(mode: &str, thresholds: Vec<f64>) -> Option<Self> {
171        let (l1, reject) = match mode {
172            "" => (false, false),
173            "L1" => (true, false),
174            "R" => (false, true),
175            "L1R" => (true, true),
176            _ => return None,
177        };
178        let rms_thresholds = if !reject {
179            Vec::new()
180        } else if thresholds.is_empty() {
181            vec![7.0, 5.0, 4.0, 3.5, 3.0, 2.8, 2.6, 2.4, 2.2, 2.5]
182        } else {
183            thresholds.into_iter().map(|value| value as f32).collect()
184        };
185        Some(Self { l1, rms_thresholds })
186    }
187
188    pub(crate) fn enabled(&self) -> bool {
189        self.l1 || !self.rms_thresholds.is_empty()
190    }
191}

213/// Supported first-wave gain solution intervals.
214#[derive(Debug, Clone, Copy, Default, PartialEq, Serialize, Deserialize, JsonSchema)]
215pub enum GainSolveInterval {
216    /// Solve one solution per `(observation, field, spw, scan)` group.
217    #[default]
218    Infinite,
219    /// Solve one solution per integration timestamp inside each scan.
220    Integration,
221    /// Solve one solution per contiguous time bucket of the given duration, in seconds.
222    Seconds(f64),
223}
224
225impl GainSolveInterval {
226    /// Read CASA's time component of `solint`, accepting quanta and seconds.
227    #[must_use]
228    pub fn from_casa(text: &str) -> Option<Self> {
229        use casa_core::quanta::{Quantity, Unit};
230        let upper = text.to_ascii_uppercase();
231        if upper.is_empty() || upper.contains("INF") {
232            return Some(Self::Infinite);
233        }
234        if upper.contains("INT") {
235            return Some(Self::Integration);
236        }
237        let quantity = text.parse::<Quantity>().ok()?;
238        let seconds = if quantity.unit().name().is_empty() {
239            quantity.value()
240        } else {
241            quantity.get_value_in(&Unit::new("s").ok()?).ok()?
242        };
243        if !seconds.is_finite() {
244            return None;
245        }
246        Some(if seconds > 0.0 {
247            Self::Seconds(seconds)
248        } else if seconds == 0.0 {
249            Self::Infinite
250        } else {
251            Self::Integration
252        })
253    }
254}

573/// Solve a limited `gaincal` request from an already-open MeasurementSet.
574pub(crate) fn solve_gain(
575    ms: &MeasurementSet,
576    request: &GainSolveRequest,
577    mode: crate::output::CalibrationOutputMode,
578    activity: Option<&mut ToolSolveActivity>,
579    legacy_iterator: bool,
580) -> Result<GainSolveReport, GainSolveError> {
581    if request.gain_type == GainType::GSpline {
582        return gspline::solve(ms, request, GainSplineControls::default(), mode, None);
583    }
584    let solved =
585        if legacy_iterator && matches!(request.gain_type, GainType::KMbd | GainType::KAntPos) {
586            delay::solve(ms, request, activity, true)?
587        } else {
588            solve_gain_rows(ms, request, activity, mode)?
589        };
590    write_gain_caltable(ms, request, &solved, mode)
591}

625fn tf_multi_bin_windows(
626    ms: &MeasurementSet,
627    rows: &[grouping::SelectedSolveRow],
628    widths: &BTreeMap<i32, usize>,
629) -> Result<BTreeSet<i32>, GainSolveError> {
630    let fail = |error: MsError| GainSolveError::FrequencyScalar(error.to_string());
631    let selections = rows
632        .iter()
633        .map(|row| (row.data_spw_id, row.channel_selection.as_ref()))
634        .collect::<BTreeMap<_, _>>();
635    let spectral = ms.spectral_window().map_err(fail)?;
636    let mut multiple = BTreeSet::new();
637    for (spw, selection) in selections {
638        let count = spectral.num_chan(spw as usize).map_err(fail)?;
639        let selected = grouping::selected_channels(selection, count as usize)?.len();
640        if selected > widths.get(&spw).copied().unwrap_or(1) {
641            multiple.insert(spw);
642        }
643    }
644    Ok(multiple)
645}

669    activity: Option<&mut ToolSolveActivity>,
670    mode: CalibrationOutputMode,
671) -> Result<SolvedGains, GainSolveError> {

673        return delay::solve(ms, request, activity, false);
674    }
675    let legacy_g_receptors = legacy_gain_receptors(request, mode)?;
676    solve_gain_rows_with_settings(ms, request, activity, None, None, legacy_g_receptors)
677        .map(|(solved, _)| solved)
678}

682    mode: CalibrationOutputMode,

686        && request.output_table.exists()

696            Some(Value::Scalar(ScalarValue::String(tag)))
697                if tag == crate::constants::LEGACY_NATIVE_G_VERSION

701    }

741
742fn solve_gain_rows_with_settings(
743    ms: &MeasurementSet,
744    request: &GainSolveRequest,
745    activity: Option<&mut ToolSolveActivity>,
746    tf: Option<TfSolveSettings>,
747    gather_only: Option<GatherOnlySolveSettings>,
748    legacy_g_receptors: bool,
749) -> Result<(SolvedGains, Option<TfSpectrum>), GainSolveError> {
750    let point_stokes = solve_point_stokes(ms, request)?;
751    validate_solve_interval(request.solve_interval)?;
752    // P Jones is later in CASA's Jones chain than G, so the rotation
753    // corrupts the MeasurementSet model or a polarized explicit point
754    // before a diagonal gain is fitted.
755    let rotate_model = request.parang
756        && request.gain_type == GainType::G
757        && request.model_source.needs_model_rotation(request.smodel);
758    let input_phase = crate::timing::Phase::start("gain_select_read_average");
759    let context = prepare_calibration_solve(
760        ms,
761        &request.selection,
762        &request.refant,
763        &request.prior_calibration_tables,
764        request.parang,
765        true,
766        grouping::PriorCalibrationMode::Gain {
767            scalar: request.gain_type == GainType::T,
768            rotate_model,
769            share_input_flags: !request.correlation_dependent_flags,
770        },
771    )?;
772    let tf_widths = tf
773        .map(|settings| {
774            crate::bandpass::solution_frequency_bin_widths(ms, &context.rows, settings.frequency)
775        })
776        .transpose()
777        .map_err(|error| GainSolveError::FrequencyScalar(error.to_string()))?;
778    let multi_bin_windows = tf_widths
779        .as_ref()
780        .map(|widths| tf_multi_bin_windows(ms, &context.rows, widths))
781        .transpose()?
782        .unwrap_or_default();
783    if let Some(settings) = &gather_only
784        && (settings.check_append || request.combine.spectral_windows)
785    {
786        crate::bandpass::check_gather_frequency(
787            ms,
788            settings
789                .check_append
790                .then_some(request.output_table.as_path()),
791            &context.rows,
792            settings.layout,
793            request.combine.spectral_windows,
794        )
795        .map_err(|error| GainSolveError::FrequencyScalar(error.to_string()))?;
796    }
797    let input_rows = if let Some(settings) = &gather_only {
798        grouping::SolveInputRows::GatherOnlyTool {
799            legacy: settings.legacy,
800        }
801    } else {
802        grouping::SolveInputRows::gain(
803            activity.is_some() && matches!(request.gain_type, GainType::G | GainType::T),
804        )
805    };
806    // CalSolvingVi2 shares raw flags before applying priors. The solver
807    // keeps flags introduced by a prior on their corresponding receptors
808    // through reference selection, which may need another antenna.
809    let correlation_dependent_flags = request.correlation_dependent_flags
810        || (context.preapplied_rows.is_some()
811            && gather_only.is_none()
812            && matches!(request.gain_type, GainType::G | GainType::T));
813    let groups = build_solve_groups(
814        ms,
815        &context.rows,
816        context.preapplied_rows.as_ref(),
817        &BTreeMap::new(),
818        SolveGroupOptions {
819            input_rows,
820            gain_type: request.gain_type,
821            model_source: request.model_source,
822            point_stokes,
823            rotate_model,
824            solve_interval: request.solve_interval,
825            combine: request.combine,
826            // VbAvg applies this boundary to either source of the interval.
827            preaverage_max_span: preaverage_span(
828                request.solve_interval,
829                request.preaverage_seconds,
830            ),
831            // Polarized preapply combines raw flags before applying priors.
832            // A newly flagged cross-hand must not invalidate parallel hands.
833            correlation_dependent_flags: correlation_dependent_flags
834                || context
835                    .preapplied_rows
836                    .as_ref()
837                    .is_some_and(|rows| rows.polarized),
838            robust_seed: request.robustness.enabled(),
839            legacy_g_receptors,
840            channelized: false,
841            polynomial_bandpass: false,
842        },
843    )?;
844
845    if groups.is_empty() {
846        return Err(GainSolveError::EmptySelection);
847    }
848
849    drop(input_phase);
850    let kernel_phase = crate::timing::Phase::start("gain_solve");
851    let mut solution_rows = Vec::new();
852    let mut attempted_spw_ids = BTreeSet::new();
853    let mut legacy_refant_slots = LegacyRefantSlots::new();
854    let mut solve_stats = SolveStatsBuilder::new(
855        gather_only.map_or_else(
856            || request.gain_type.polarizations(),
857            |settings| settings.parameters,
858        ),
859        spectral_window_ids(ms)?,
860        &context.available_antennas,
861    );
862    for ((base_key, bucket_key), mut group) in groups {
863        if (group.attempted || group.gather_attempted())
864            && (multi_bin_windows.contains(&group.spw_id)
865                || (request.combine.spectral_windows && !multi_bin_windows.is_empty()))
866        {
867            return Err(GainSolveError::FrequencyScalar(format!(
868                "SPW {} has more than one frequency bin",
869                group.spw_id,
870            )));
871        }
872        if group.attempted {
873            attempted_spw_ids.insert(group.spw_id);
874        }
875        record_legacy_refant_slots(&mut legacy_refant_slots, &group);
876        group.finalize_for_solve(request.solve_mode, request.robustness.enabled());
877        trace::trace_group(&base_key, &bucket_key, &group, request);
878        let Some(group_rows) = solve_group_with_stats(
879            group,
880            &context.available_antennas,
881            request,
882            context.refant_id,
883            correlation_dependent_flags,
884            &mut solve_stats,
885        ) else {
886            continue;
887        };
888        trace::trace_solution_rows(
889            &base_key,
890            &bucket_key,
891            &group_rows,
892            request,
893            context.refant_id,
894        );
895        solution_rows.extend(group_rows);
896    }
897    apply_requested_reference(
898        ms,
899        &mut solution_rows,
900        &request.refant,
901        &context,
902        &legacy_refant_slots,
903    )?;
904    finish_gain_amplitudes(&mut solution_rows, request);
905    tally_references(&solution_rows, &mut solve_stats);
906    drop(kernel_phase);
907    let mut channel_selections = BTreeMap::new();
908    for row in &context.rows {
909        if let Some(channels) = &row.channel_selection {
910            channel_selections
911                .entry(row.data_spw_id)
912                .or_insert_with(|| channels.clone());
913        }
914    }
915    solve_stats.track_activity(activity);
916    let solved = SolvedGains {
917        rows: solution_rows,
918        refant_id: context.refant_id,
919        attempted_spw_ids,
920        channel_selections,
921        selected_data: context.selected_data,
922        solve_stats: solve_stats.finish(),
923    };
924    Ok((
925        solved,
926        tf_widths.map(|bin_widths| TfSpectrum {
927            bin_widths,
928            selected_rows: context.rows,
929        }),
930    ))
931}

933fn finish_gain_amplitudes(rows: &mut [kernel::SolutionRow], request: &GainSolveRequest) {
934    // CASA enforces amplitudes on the final gains after referencing and
935    // before normalizing across all intervals (`globalPostSolveTinker`).
936    if request.solve_mode != GainSolveMode::AmplitudePhase {
937        for row in rows.iter_mut() {
938            for (gain, flagged) in row.gains.iter_mut().zip(&row.flags) {
939                if *flagged {
940                    *gain = Complex32::new(1.0, 0.0);
941                } else if request.solve_mode == GainSolveMode::Amplitude {
942                    let amplitude = gain.norm();
943                    *gain = Complex32::new(if amplitude == 0.0 { 1.0 } else { amplitude }, 0.0);
944                }
945            }
946        }
947    }
948    if request.solve_mode != GainSolveMode::Phase && request.normalize_average_amplitude {
949        normalize_gain_solution_amplitudes(rows, request.normalization_type);
950    }
951}

953/// Gather at the shorter positive user preaverage or solution interval.
954pub(crate) fn preaverage_span(interval: GainSolveInterval, requested: f64) -> Option<f64> {
955    let requested = requested as f32;
956    let preaverage = (requested > f32::EPSILON && requested.is_finite()).then_some(requested);
957    let seconds = match interval {
958        GainSolveInterval::Infinite => preaverage,
959        GainSolveInterval::Integration => None,
960        GainSolveInterval::Seconds(seconds) => {
961            let solint = seconds as f32;
962            Some(preaverage.map_or(solint, |preavg| preavg.min(solint)))
963        }
964    };
965    seconds.map(|seconds| f64::from(seconds) * 0.999)
966}

984fn apply_requested_reference(
985    ms: &MeasurementSet,
986    rows: &mut [kernel::SolutionRow],
987    request: &RefAntSelector,
988    context: &crate::session::CalibrationSolveContext,
989    legacy_refant_slots: &LegacyRefantSlots,
990) -> Result<(), GainSolveError> {
991    let preferred = context.refant_id;
992    if preferred < 0 {
993        return Ok(());
994    }
995    let strict = matches!(request, RefAntSelector::Expression { strict: true, .. });
996    let mut alternates = context.refant_alternates.clone();
997    if !strict {
998        let last_preferred = alternates.last().copied().unwrap_or(preferred);
999        let mut nearby = refant_distance_order(ms, last_preferred)?;
1000        nearby.retain(|id| *id != preferred && !alternates.contains(id));
1001        alternates.extend(nearby);
1002    }
1003    apply_reference_antenna(rows, preferred, &alternates, strict, legacy_refant_slots);
1004    Ok(())
1005}

1089/// The antennas other than `refant_id`, nearest it first: the alternates
1090/// `SolvableVisJones::applyRefAnt` tries when the reference antenna is flagged.
1091pub(crate) fn refant_distance_order(
1092    ms: &MeasurementSet,
1093    refant_id: i32,
1094) -> Result<Vec<i32>, GainSolveError> {
1095    let antenna = ms
1096        .antenna()
1097        .map_err(|source| GainSolveError::OpenMeasurementSet {
1098            path: ms
1099                .path()
1100                .map(|path| path.display().to_string())
1101                .unwrap_or_else(|| "<in-memory>".to_string()),
1102            source,
1103        })?;
1104    let positions = (0..antenna.row_count())
1105        .map(|row| antenna.position(row))
1106        .collect::<Result<Vec<_>, _>>()
1107        .map_err(|source| GainSolveError::OpenMeasurementSet {
1108            path: "ANTENNA POSITION".to_string(),
1109            source,
1110        })?;
1111    Ok(nearest_first(&positions, refant_id))
1112}

1114/// The antennas other than `refant_id`, nearest it first, of antennas at
1115/// `positions` (ITRF metres, indexed by antenna ID).
1116pub(crate) fn nearest_first(positions: &[[f64; 3]], refant_id: i32) -> Vec<i32> {
1117    let origin = usize::try_from(refant_id)
1118        .ok()
1119        .and_then(|row| positions.get(row))
1120        .copied()
1121        .unwrap_or_default();
1122    let mut order = (0..positions.len())
1123        .filter(|row| i32::try_from(*row).ok() != Some(refant_id))
1124        .map(|row| {
1125            let distance = positions[row]
1126                .iter()
1127                .zip(origin)
1128                .map(|(value, origin)| (value - origin).powi(2))
1129                .sum::<f64>();
1130            (distance, row)
1131        })
1132        .collect::<Vec<_>>();
1133    order.sort_by(|left, right| left.0.total_cmp(&right.0).then(left.1.cmp(&right.1)));
1134    order
1135        .into_iter()
1136        .map(|(_, row)| i32::try_from(row).expect("antenna row count fits in i32"))
1137        .collect()
1138}

1140/// Refer each solution interval's phases to a reference antenna, as
1141/// `SolvableVisJones::applyRefAnt` does with `refantmode='flex'`. Intervals
1142/// go in time order within each spectral window. The reference is the first
1143/// of: the preferred antenna, the one used last, then the others nearest the
1144/// preferred first, that has every receptor unflagged both here and in the
1145/// last referred interval (only the first receptor when the window's second
1146/// is never solved, or only the second for a legacy YY/LL-only window).
1147/// Its phase change since that interval is removed from
1148/// every antenna, so an alternate is held at constant phase. An interval
1149/// with no such antenna is flagged entirely and leaves the last referred
1150/// interval in place for the next.
1151pub(crate) fn apply_reference_antenna(
1152    rows: &mut [kernel::SolutionRow],
1153    preferred_refant_id: i32,
1154    alternates: &[i32],
1155    strict: bool,
1156    legacy_refant_slots: &LegacyRefantSlots,
1157) {
1158    // CASA `applyRefAnt` iterates by window and exact caltable time, even
1159    // where separate solve intervals contributed rows at the same time.
1160    let mut group_indices = BTreeMap::<_, Vec<_>>::new();
1161    for (index, row) in rows.iter().enumerate() {
1162        group_indices
1163            .entry((row.spw_id, row.time_seconds.to_bits()))
1164            .or_default()
1165            .push(index);
1166    }
1167    let mut receptor_counts = BTreeMap::<i32, usize>::new();
1168    for row in rows.iter() {
1169        let count = receptor_counts.entry(row.spw_id).or_insert(1);
1170        if row.flags.get(1).is_some_and(|flag| !*flag) {
1171            *count = 2;
1172        }
1173    }
1174
1175    let mut choices = Vec::with_capacity(alternates.len() + 2);
1176    choices.extend([preferred_refant_id, preferred_refant_id]);
1177    choices.extend_from_slice(alternates);
1178
1179    let mut previous = None::<Vec<usize>>;
1180    let mut last_spw = None::<i32>;
1181    for ((spw_id, _), current) in group_indices {
1182        if last_spw != Some(spw_id) {
1183            previous = None;
1184            last_spw = Some(spw_id);
1185        }
1186        let receptors = if legacy_refant_slots.get(&spw_id) == Some(&[false, true]) {
1187            1..2
1188        } else {
1189            0..receptor_counts[&spw_id]
1190        };
1191        let previous_indices = previous.as_ref().unwrap_or(&current);
1192        let candidates = if strict { &choices[..1] } else { &choices[..] };
1193        let Some((current_ref_index, previous_ref_index)) =
1194            find_flex_refant(rows, candidates, &current, previous_indices, receptors)
1195        else {
1196            for index in &current {
1197                rows[*index].flags.iter_mut().for_each(|flag| *flag = true);
1198                rows[*index].refant_id = -1;
1199            }
1200            continue;
1201        };
1202
1203        let first = previous.is_none();
1204        let ref_phasor = rows[current_ref_index]
1205            .gains
1206            .iter()
1207            .zip(rows[previous_ref_index].gains.iter())
1208            .map(|(current_gain, previous_gain)| {
1209                let current_phase = phase_only(*current_gain);
1210                if first {
1211                    current_phase
1212                } else {
1213                    current_phase / phase_only(*previous_gain)
1214                }
1215            })
1216            .collect::<Vec<_>>();
1217
1218        let current_refant = rows[current_ref_index].antenna_id;
1219        choices[1] = current_refant;
1220        for index in &current {
1221            for (gain, phasor) in rows[*index].gains.iter_mut().zip(ref_phasor.iter()) {
1222                if phasor.norm() > f32::EPSILON {
1223                    *gain /= *phasor;
1224                }
1225            }
1226            rows[*index].refant_id = current_refant;
1227        }
1228        previous = Some(current);
1229    }
1230}
1231
1232fn find_flex_refant(
1233    rows: &[kernel::SolutionRow],
1234    choices: &[i32],
1235    current: &[usize],
1236    previous: &[usize],
1237    receptors: Range<usize>,
1238) -> Option<(usize, usize)> {
1239    let usable = |indices: &[usize], antenna_id: i32| {
1240        indices
1241            .iter()
1242            .copied()
1243            .find(|&index| rows[index].antenna_id == antenna_id)
1244            .filter(|&index| {
1245                rows[index]
1246                    .flags
1247                    .get(receptors.clone())
1248                    .is_some_and(|flags| flags.iter().all(|flag| !*flag))
1249            })
1250    };
1251    choices
1252        .iter()
1253        .find_map(|&antenna_id| Some((usable(current, antenna_id)?, usable(previous, antenna_id)?)))
1254}

1256pub(crate) fn phase_only(gain: Complex32) -> Complex32 {
1257    let norm = gain.norm();
1258    if norm > f32::EPSILON {
1259        gain / Complex32::new(norm, 0.0)
1260    } else {
1261        Complex32::new(1.0, 0.0)
1262    }
1263}
1264
1265/// Divide every gain of each window by the root mean or median power of the window's
1266/// unflagged solutions, taken over all its intervals, antennas and
1267/// receptors together, as CASA's `solnorm` does
1268/// (`SolvableVisJones::normalize`, `SolvableVisCal::calcPowerNorm`). A
1269/// window with one unflagged solution or none stays, as `normSolnArray`
1270/// normalizes only more than one.
1271fn normalize_gain_solution_amplitudes(rows: &mut [kernel::SolutionRow], method: GainNormalization) {
1272    let mut power = BTreeMap::<i32, (f64, usize, Vec<f32>)>::new();
1273    for row in rows.iter() {
1274        let (sum, count, samples) = power.entry(row.spw_id).or_default();
1275        for (gain, flagged) in row.gains.iter().zip(&row.flags) {
1276            if !flagged {
1277                let power = gain.norm_sqr();
1278                match method {
1279                    GainNormalization::Mean => *sum += f64::from(power),
1280                    GainNormalization::Median => samples.push(power),
1281                }
1282                *count += 1;
1283            }
1284        }
1285    }
1286    let factors: BTreeMap<_, _> = power
1287        .into_iter()
1288        .filter_map(|(spw, (sum, count, mut samples))| {
1289            if count <= 1 {
1290                return None;
1291            }
1292            let factor = match method {
1293                GainNormalization::Mean => (sum / count as f64).sqrt() as f32,
1294                GainNormalization::Median => grouping::median_with_even_mean(&mut samples).sqrt(),
1295            };
1296            (factor.is_finite() && factor > 0.0).then_some((spw, factor))
1297        })
1298        .collect();
1299    for row in rows.iter_mut() {
1300        let Some(&factor) = factors.get(&row.spw_id) else {
1301            continue;
1302        };
1303        for gain in &mut row.gains {
1304            *gain /= Complex32::new(factor, 0.0);
1305        }
1306    }
1307}

1360/// Convert the CASA point-model Stokes vector to the solver's single precision.
1361/// Short vectors are padded with zero. An explicit empty vector is a zero
1362/// model, as in `calibrater.setptmodel`, while tasks choose their stored model
1363/// before calling this conversion when `smodel` is empty.
1364pub fn point_model_stokes(values: &[f64]) -> Option<[f32; 4]> {
1365    let mut stokes = [0.0; 4];
1366    for (target, &value) in stokes.get_mut(..values.len())?.iter_mut().zip(values) {
1367        *target = value as f32;
1368    }
1369    stokes
1370        .iter()
1371        .all(|value| value.is_finite())
1372        .then_some(stokes)
1373}

1375pub(crate) fn validate_smodel(smodel: [f32; 4], polarized: bool) -> Result<(), GainSolveError> {
1376    if smodel.iter().any(|value| !value.is_finite())
1377        || (!polarized && smodel[1..].iter().any(|value| *value != 0.0))
1378    {
1379        return Err(GainSolveError::UnsupportedSkyModel { smodel });
1380    }
1381    Ok(())
1382}
