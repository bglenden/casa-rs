3//! The imaging request: the provider-contracts catalog's `imager` surface,
4//! resolved, as one typed record (plan section 5.8, decision D8).
5//!
6//! Field names and choices are the catalog's, and so are the defaults: a
7//! request is deserialized only from resolved catalog values (every route,
8//! the command line and `--json-run` alike, resolves through the catalog),
9//! so nothing here restates a default and a missing field is an error. The
10//! catalog spells an unset optional value `"none"`. Parameters that are
11//! active only for one gridder live in that gridder's variant.
12

224/// The route the passes read the selected visibilities by, while the
225/// imaging port (#932) holds both: the selected-observation route, which
226/// reads MAIN rows, or the observation's batches through the shared
227/// `casa-observation` boundary. Both image the same samples; the cutover
228/// removes the first.
229#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
230pub enum SourceRoute {
231    /// MAIN rows, through `casa-ms`'s selected-observation layer.
232    #[default]
233    SelectedRows,
234    /// The observation's batches, with geometry over batches. It writes
235    /// nothing back yet: a run that writes a model or corrected column is
236    /// refused (#932, step 5).
237    ObservationBatches,
238}

240/// The device a request asks to grid on (`backend`).
241///
242/// `auto` takes the host's GPU for what it implements and has been checked
243/// on, end to end, against the CPU; everything else grids on the CPU. The
244/// GPU accumulates in `f32` (plan decision D2), so choosing it accepts
245/// single-precision grids, and an explicit `gridprecision = f64` keeps
246/// `auto` on the CPU.
247#[derive(Clone, Copy, Debug, Default, Deserialize, PartialEq, Eq)]
248#[serde(rename_all = "lowercase")]
249pub enum BackendRequest {
250    /// The GPU where it runs the problem, else the CPU.
251    #[default]
252    Auto,
253    /// The CPU.
254    Cpu,
255    /// The host's GPU (Metal on macOS); a problem it cannot run is refused.
256    Gpu,
257}

268impl FromStr for DataColumn {
269    type Err = String;
270
271    fn from_str(text: &str) -> Result<Self, Self::Err> {
272        match text.to_ascii_uppercase().as_str() {
273            "DATA" => Ok(Self::Data),
274            "CORRECTED" | "CORRECTED_DATA" => Ok(Self::Corrected),
275            _ => Err(format!(
276                "datacolumn {text:?} is neither DATA nor CORRECTED_DATA"
277            )),
278        }

439impl ImagingRequest {
440    /// Check the relations between parameters that no single parameter's
441    /// catalog domain can state.
442    pub fn validate(&self) -> Result<(), InvalidRequest> {
443        let invalid = |parameters, reason| Err(InvalidRequest { parameters, reason });
444        if self.phasecenter.is_some() && self.phasecenter_field.is_some() {
445            return invalid(
446                "phasecenter, phasecenter_field",
447                "name one image centre, not two",
448            );
449        }
450        if self.imsize.contains(&0) || self.cell.iter().any(|cell| *cell <= 0.0) {
451            return invalid("imsize, cell", "the image must have positive extent");
452        }
453        if self.psfcutoff <= 0.0 || self.pblimit == 0.0 || self.pblimit.abs() >= 1.0 {
454            return invalid(
455                "psfcutoff, pblimit",
456                "psfcutoff must be positive and |pblimit| in (0, 1)",
457            );
458        }
459        if self.nsigma < 0.0 || self.scales.iter().any(|scale| *scale < 0.0) {
460            return invalid("nsigma, scales", "must not be negative");
461        }
462        if self.startmodel.is_some() && self.deconvolver == Deconvolver::Mtmfs {
463            return invalid(
464                "startmodel, deconvolver",
465                "an mtmfs start model is one image per Taylor term, which is not supported yet",
466            );
467        }
468        if self.startmodel.is_some() && self.outlierfile.is_some() {
469            return invalid(
470                "startmodel, outlierfile",
471                "a start model with outlier fields is not supported yet: an outlier would own the \
472                 main field's overlap with no start model of its own",
473            );
474        }
475        if self.nterms == 0 || (self.nterms > 1) != (self.deconvolver == Deconvolver::Mtmfs) {
476            return invalid(
477                "nterms, deconvolver",
478                "nterms > 1 exactly when deconvolver = mtmfs",
479            );
480        }
481        if self.deconvolver == Deconvolver::Mtmfs && self.specmode != SpecMode::Mfs {
482            return invalid("deconvolver, specmode", "mtmfs images a continuum (mfs)");
483        }
484        if self.solves() && (self.minor_cycle_length == 0 || self.nmajor == Some(0)) {
485            return invalid(
486                "minor_cycle_length, nmajor",
487                "a solving run needs minor-cycle iterations and a major cycle",
488            );
489        }
490        if self.savemodel && !self.solves() && self.startmodel.is_none() {
491            return invalid(
492                "savemodel, niter",
493                "MODEL_DATA needs a solved model or a start model",
494            );
495        }
496        if (self.mask_image.is_some() && !self.mask_box.is_empty())
497            || (self.usemask == UseMask::AutoMultithresh
498                && (self.mask_image.is_some() || !self.mask_box.is_empty()))
499        {
500            return invalid(
501                "usemask, mask_image, mask_box",
502                "name one mask: auto-multithresh, a mask image or boxes",
503            );
504        }
505        if self.save_continuum_residual && self.fitspw.is_none() {
506            return invalid(
507                "save_continuum_residual, fitspw",
508                "the residual of a continuum fit needs the fit",
509            );
510        }
511        if self.fitspw.is_some() && self.specmode == SpecMode::Mfs {
512            return invalid(
513                "fitspw, specmode",
514                "continuum subtraction feeds a channel-local cube",
515            );
516        }
517        if self.outlierfile.is_some()
518            && (self.weighting != Weighting::Natural
519                || (self.solves() && self.deconvolver != Deconvolver::Hogbom))
520        {
521            return invalid(
522                "outlierfile",
523                "outlier domains are imaged with natural weighting, dirty or Högbom",
524            );
525        }
526        if let Gridder::Awproject(aw) = &self.gridder {
527            aw.validate()?;
528        }
529        Ok(())
530    }
531
532    /// Whether the run solves for a model: `mtmfs` always, the other
533    /// solvers unless the run is dirty (`dirty_only` or `niter = 0`).
534    #[must_use]
535    pub fn solves(&self) -> bool {
536        self.deconvolver == Deconvolver::Mtmfs || (!self.dirty_only && self.niter > 0)
537    }
538
539    /// Total minor-cycle iterations: none for `dirty_only`.
540    #[must_use]
541    pub const fn iterations(&self) -> usize {
542        if self.dirty_only { 0 } else { self.niter }
543    }
544
545    /// The resource policy `parallel` asks for: the balanced share of the
546    /// host, or one worker with no memory ceiling of its own.
547    #[must_use]
548    pub const fn resource_policy(&self) -> ResourcePolicy {
549        if self.parallel {
550            ResourcePolicy::Balanced
551        } else {
552            ResourcePolicy::Explicit {
553                workers: 1,
554                memory: u64::MAX,
555            }
556        }
557    }
558}

560impl AwProjection {
561    fn validate(&self) -> Result<(), InvalidRequest> {
562        let invalid = |parameters, reason| Err(InvalidRequest { parameters, reason });
563        // The EVLA cache contract CASA's AWProjectFT builds: 32 W planes.
564        if self.wprojplanes.map(NonZeroUsize::get) != Some(32) {
565            return invalid(
566                "wprojplanes",
567                "AW projection uses CASA's 32-plane EVLA cache",
568            );
569        }
570        if self
571            .pointingoffsetsigdev
572            .iter()
573            .any(|threshold| *threshold < 0.0)
574        {
575            return invalid("pointingoffsetsigdev", "thresholds must not be negative");
576        }
577        if let AwCfSource::NativeEvla {
578            native_cf_working_size,
579            native_cf_oversampling,
580            native_cf_cache_bytes,
581            native_cf_maximum_cells,
582            ..
583        } = &self.cf_source
584        {
585            let size = *native_cf_working_size;
586            if size < 8
587                || !size.is_multiple_of(2)
588                || size
589                    .checked_mul(size)
590                    .and_then(|cells| cells.checked_mul(48))
591                    .is_none()
592                || *native_cf_oversampling == 0
593                || *native_cf_oversampling > size / 4
594            {
595                return invalid(
596                    "native_cf_working_size, native_cf_oversampling",
597                    "the aperture grid must be even, at least 8, and oversampled at most a \
598                     quarter of its side",
599                );
600            }
601            if *native_cf_cache_bytes == 0 || *native_cf_maximum_cells < 32 {
602                return invalid(
603                    "native_cf_cache_bytes, native_cf_maximum_cells",
604                    "the cache must hold every W plane",
605                );
606            }
607        }
608        Ok(())
609    }

732/// CASA's `uvtaper` (`synthesisimager::setweighting`, then
733/// `VisImagingWeight::setFilter`): `[bmaj]`, `[bmaj, bmin]` or
734/// `[bmaj, bmin, bpa]`, with `bmin = bmaj` and `bpa = 0` when not given,
735/// and no taper when the list or its first entry is empty. In wavelengths
736/// (`lambda`, `klambda`, …) the axes are the taper's uv-plane HWHM; as
737/// angles they are its image-plane FWHM θ, whose uv-plane HWHM is
738/// 2 ln 2 / (π θ). CASA reads an axis with no unit as radians and tapers
739/// every weight to zero; casa-rs refuses it (`docs/CASA (C++) bugs.md`).
740fn uv_taper<'de, D: Deserializer<'de>>(deserializer: D) -> Result<Option<UvTaper>, D::Error> {
741    let entries = match Value::deserialize(deserializer)? {
742        Value::Array(items) => items,
743        value => vec![value],
744    };
745    let texts = entries
746        .iter()
747        .map(|entry| {
748            entry
749                .as_str()
750                .map(str::trim)
751                .ok_or_else(|| unexpected(entry, "a quantity"))
752        })
753        .collect::<Result<Vec<_>, D::Error>>()?;
754    let taper = match texts.as_slice() {
755        [] | ["", ..] => return Ok(None),
756        [bmaj] => uv_taper_axes(bmaj, bmaj, None),
757        [bmaj, bmin] => uv_taper_axes(bmaj, bmin, None),
758        [bmaj, bmin, bpa] => uv_taper_axes(bmaj, bmin, Some(bpa)),
759        _ => Err("uvtaper takes at most [bmaj, bmin, bpa]".to_string()),
760    };
761    taper.map(Some).map_err(D::Error::custom)
762}
763
764/// One `uvtaper` axis: its uv-plane HWHM in wavelengths, and whether it was
765/// given as an angle.
766fn uv_taper_axis(text: &str) -> Result<(f64, bool), String> {
767    let refused = || {
768        format!(
769            "uvtaper {text:?}: give an angle (the image-plane FWHM) or wavelengths (the uv-plane \
770             HWHM, lambda or klambda)"
771        )
772    };
773    let quantity: Quantity = text.parse().map_err(|_| refused())?;
774    let unit = |name: &str| Unit::from_str(name).expect("a casacore unit");
775    let hwhm = if let Ok(wavelengths) = quantity.get_value_in(&unit("lambda")) {
776        (wavelengths, false)
777    } else if let Ok(fwhm) = quantity.get_value_in(&unit("rad")) {
778        (
779            2.0 * std::f64::consts::LN_2 / (std::f64::consts::PI * fwhm),
780            true,
781        )
782    } else {
783        return Err(refused());
784    };
785    if quantity.unit().name().is_empty() || !(hwhm.0.is_finite() && hwhm.0 > 0.0) {
786        return Err(refused());
787    }
788    Ok(hwhm)
789}
790
791fn uv_taper_axes(bmaj: &str, bmin: &str, bpa: Option<&str>) -> Result<UvTaper, String> {
792    let (major, major_angle) = uv_taper_axis(bmaj)?;
793    let (minor, minor_angle) = uv_taper_axis(bmin)?;
794    if major_angle != minor_angle {
795        return Err(format!(
796            "uvtaper {bmaj:?}, {bmin:?}: give both axes as angles or both in wavelengths"
797        ));
798    }
799    let position_angle_rad = match bpa {
800        None => 0.0,
801        Some(text) => text
802            .parse::<Quantity>()
803            .ok()
804            .filter(|angle| !angle.unit().name().is_empty())
805            .and_then(|angle| angle.get_value_in(&Unit::from_str("rad").ok()?).ok())
806            .ok_or_else(|| format!("uvtaper position angle {text:?} needs an angle unit"))?,
807    };
808    Ok(UvTaper::new(major, minor, position_angle_rad))
809}

815/// Comma-separated values; an unset list is empty.
816fn comma_list<T>(text: &str) -> Result<Vec<T>, String>
817where
818    T: FromStr,
819    T::Err: fmt::Display,
820{
821    if unset(text) {
822        return Ok(Vec::new());
823    }
824    text.split(',')
825        .map(|item| {
826            item.trim()
827                .parse()
828                .map_err(|error| format!("{item:?}: {error}"))
829        })
830        .collect()
831}

838        value => return Err(unexpected(&value, "comma-separated numbers")),
839    };
840    if values.iter().all(|value: &f64| value.is_finite()) {
841        Ok(values)
842    } else {
843        Err(D::Error::custom("numbers must be finite"))
844    }
845}

891/// CASA `stokes`: Stokes parameters (`I`, `IV`, `IQUV`, …), circular
892/// correlations (`RR`, `RRLL`, …) or linear ones (`XXYY`, `XXXYYXYY`, …),
893/// one kind only; sorted and without repeats.
894fn stokes<'de, D: Deserializer<'de>>(
895    deserializer: D,
896) -> Result<Vec<PolarizationCoordinate>, D::Error> {
897    use PolarizationCoordinate::*;
898    let text = choice(deserializer)?;
899    let mut coordinates = Vec::new();
900    let mut rest = text.as_str();
901    while !rest.is_empty() {
902        let (coordinate, width) = match rest.as_bytes()[0] {
903            b'I' => (StokesI, 1),
904            b'Q' => (StokesQ, 1),
905            b'U' => (StokesU, 1),
906            b'V' => (StokesV, 1),
907            _ => match rest.get(..2) {
908                Some("RR") => (CircularRr, 2),
909                Some("RL") => (CircularRl, 2),
910                Some("LR") => (CircularLr, 2),
911                Some("LL") => (CircularLl, 2),
912                Some("XX") => (LinearXx, 2),
913                Some("XY") => (LinearXy, 2),
914                Some("YX") => (LinearYx, 2),
915                Some("YY") => (LinearYy, 2),
916                _ => {
917                    return Err(D::Error::custom(format!(
918                        "stokes {text:?} is not a list of I, Q, U, V or of RR, RL, LR, LL, XX, \
919                         XY, YX, YY"
920                    )));
921                }
922            },
923        };
924        coordinates.push(coordinate);
925        rest = &rest[width..];
926    }
927    coordinates.sort_unstable();
928    coordinates.dedup();
929    let kind = |coordinate: &PolarizationCoordinate| match coordinate {
930        StokesI | StokesQ | StokesU | StokesV => 0,
931        LinearXx | LinearXy | LinearYx | LinearYy => 1,
932        CircularRr | CircularRl | CircularLr | CircularLl => 2,
933    };
934    match coordinates.first() {
935        Some(first) if coordinates.iter().all(|other| kind(other) == kind(first)) => {
936            Ok(coordinates)
937        }
938        _ => Err(D::Error::custom(format!(
939            "stokes {text:?} names no coordinate, or more than one of Stokes parameters, \
940             linear and circular correlations"
941        ))),
942    }
943}

1151            .expect("a taper");
1152        close(circular.major_lambda(), 2.0 * LN_2 / (PI * 10.0 * arcsec));
1153        close(circular.minor_lambda(), circular.major_lambda());
1154        assert_eq!(circular.position_angle_rad(), 0.0);
1155        let rbmaj = PI * PI * (10.0 * arcsec).powi(2) / (4.0 * LN_2);
1156        close(LN_2 / circular.major_lambda().powi(2), rbmaj);
1157        let uv = parsed(json!(["5klambda", "3klambda", "45deg"]))
1158            .expect("parsed")
1159            .expect("a taper");
1160        close(uv.major_lambda(), 5000.0);
1161        close(uv.minor_lambda(), 3000.0);
1162        close(uv.position_angle_rad(), PI / 4.0);
1163        let two = parsed(json!(["80arcsec", "40arcsec"]))
1164            .expect("parsed")
1165            .expect("a taper");
1166        close(two.minor_lambda(), 2.0 * two.major_lambda());
1167        // CASA reads a bare number as radians and tapers every weight to
1168        // zero; mixed axes and a unitless angle are refused too.
1169        for value in [
1170            json!(["500.0"]),
1171            json!(["10arcsec", "3klambda"]),
1172            json!(["10arcsec", "5arcsec", "30"]),
1173            json!(["1Jy"]),
1174        ] {
1175            assert!(parsed(value.clone()).is_err(), "{value}");
1176        }
