24pub(crate) fn write_gain_caltable(
25    ms: &MeasurementSet,
26    request: &GainSolveRequest,
27    solved: &SolvedGains,
28) -> Result<GainSolveReport, GainSolveError> {
29    let name = ms
30        .path()
31        .map(|path| path.display().to_string())
32        .unwrap_or_else(|| "<in-memory>".to_string());
33    write_gain_caltable_named(ms, name, request, solved)
34}

36/// [`write_gain_caltable`], recording `measurement_set_name` as the table's
37/// `MSName`: the store the subtables of `ms` describe, when `ms` holds
38/// them in memory.
39pub(crate) fn write_gain_caltable_named(
40    ms: &MeasurementSet,
41    measurement_set_name: String,
42    request: &GainSolveRequest,
43    solved: &SolvedGains,
44) -> Result<GainSolveReport, GainSolveError> {
45    let rows = &solved.rows;
46    let schema = TableSchema::new(vec![
47        ColumnSchema::scalar(COL_TIME, casa_types::PrimitiveType::Float64),
48        ColumnSchema::scalar(COL_FIELD_ID, casa_types::PrimitiveType::Int32),
49        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, casa_types::PrimitiveType::Int32),
50        ColumnSchema::scalar(COL_ANTENNA1, casa_types::PrimitiveType::Int32),
51        ColumnSchema::scalar(COL_ANTENNA2, casa_types::PrimitiveType::Int32),
52        ColumnSchema::scalar(COL_INTERVAL, casa_types::PrimitiveType::Float64),
53        ColumnSchema::scalar(COL_SCAN_NUMBER, casa_types::PrimitiveType::Int32),
54        ColumnSchema::scalar(COL_OBSERVATION_ID, casa_types::PrimitiveType::Int32),
55        ColumnSchema::scalar(COL_ARRAY_ID, casa_types::PrimitiveType::Int32),
56        ColumnSchema::scalar(COL_TIME_EXTRA_PREC, casa_types::PrimitiveType::Float64),
57        ColumnSchema::array_variable(COL_CPARAM, casa_types::PrimitiveType::Complex32, Some(2)),
58        ColumnSchema::array_variable(COL_PARAMERR, casa_types::PrimitiveType::Float32, Some(2)),
59        ColumnSchema::array_variable(COL_FLAG, casa_types::PrimitiveType::Bool, Some(2)),
60        ColumnSchema::array_variable(COL_SNR, casa_types::PrimitiveType::Float32, Some(2)),
61        ColumnSchema::array_variable(COL_WEIGHT, casa_types::PrimitiveType::Float32, Some(2)),
62    ])
63    .expect("valid gain caltable schema");
64    let mut writer = CalibrationTableWriter::create(
65        ms,
66        CalibrationTableDescriptor {
67            output: &request.output_table,
68            schema,
69            subtype: request.gain_type.vis_cal(),
70            parameter_type: Some("Complex"),
71            measurement_set_name,
72            include_polarization_basis: true,
73            time_extra_precision_column: Some(COL_TIME_EXTRA_PREC),
74        },
75    )?;
76
77    for row in rows {
78        writer.append(RecordValue::new(vec![
79            RecordField::new(
80                COL_TIME,
81                Value::Scalar(ScalarValue::Float64(row.time_seconds)),
82            ),
83            RecordField::new(
84                COL_FIELD_ID,
85                Value::Scalar(ScalarValue::Int32(row.field_id)),
86            ),
87            RecordField::new(
88                COL_SPECTRAL_WINDOW_ID,
89                Value::Scalar(ScalarValue::Int32(row.spw_id)),
90            ),
91            RecordField::new(
92                COL_ANTENNA1,
93                Value::Scalar(ScalarValue::Int32(row.antenna_id)),
94            ),
95            RecordField::new(
96                COL_ANTENNA2,
97                Value::Scalar(ScalarValue::Int32(row.refant_id)),
98            ),
99            // CASA writes every solution's INTERVAL as 0 (`SolvableVisCal::keepNCT`).
100            RecordField::new(COL_INTERVAL, Value::Scalar(ScalarValue::Float64(0.0))),
101            RecordField::new(
102                COL_SCAN_NUMBER,
103                Value::Scalar(ScalarValue::Int32(row.scan_number)),
104            ),
105            RecordField::new(
106                COL_OBSERVATION_ID,
107                Value::Scalar(ScalarValue::Int32(row.observation_id)),
108            ),
109            RecordField::new(COL_ARRAY_ID, Value::Scalar(ScalarValue::Int32(0))),
110            RecordField::new(
111                COL_TIME_EXTRA_PREC,
112                Value::Scalar(ScalarValue::Float64(0.0)),
113            ),
114            RecordField::new(
115                COL_CPARAM,
116                Value::Array(ArrayValue::Complex32(
117                    ArrayD::from_shape_vec(IxDyn(&[row.gains.len(), 1]).f(), row.gains.clone())
118                        .expect("gain vector should reshape to receptor x channel"),
119                )),
120            ),
121            RecordField::new(
122                COL_PARAMERR,
123                Value::Array(ArrayValue::Float32(
124                    ArrayD::from_shape_vec(
125                        IxDyn(&[row.gains.len(), 1]).f(),
126                        row.param_errors.clone(),
127                    )
128                    .expect("paramerr vector should reshape to receptor x channel"),
129                )),
130            ),
131            RecordField::new(
132                COL_FLAG,
133                Value::Array(ArrayValue::Bool(
134                    ArrayD::from_shape_vec(IxDyn(&[row.flags.len(), 1]).f(), row.flags.clone())
135                        .expect("flag vector should reshape to receptor x channel"),
136                )),
137            ),
138            RecordField::new(
139                COL_SNR,
140                Value::Array(ArrayValue::Float32(
141                    ArrayD::from_shape_vec(IxDyn(&[row.gains.len(), 1]).f(), row.snrs.clone())
142                        .expect("snr vector should reshape to receptor x channel"),
143                )),
144            ),
145            RecordField::new(
146                COL_WEIGHT,
147                Value::Array(ArrayValue::Float32(
148                    ArrayD::from_shape_vec(IxDyn(&[row.gains.len(), 1]).f(), row.weights.clone())
149                        .expect("weight vector should reshape to receptor x channel"),
150                )),
151            ),
152        ]))?;
153    }
154
155    writer.copy_subtables(&[
156        (SubtableId::Observation, "OBSERVATION"),
157        (SubtableId::Antenna, "ANTENNA"),
158        (SubtableId::Field, "FIELD"),
159        (SubtableId::History, "HISTORY"),
160    ])?;
161    write_gain_spectral_window_subtable(ms, writer.output_path(), request, solved)?;
162    writer.save_main()?;
163
164    let field_ids = rows.iter().map(|row| row.field_id).collect::<BTreeSet<_>>();
165    let spw_ids = rows.iter().map(|row| row.spw_id).collect::<BTreeSet<_>>();
166    Ok(GainSolveReport {
167        output_table: request.output_table.clone(),
168        gain_type: request.gain_type,
169        refant_antenna_id: solved.refant_id,
170        field_ids: field_ids.iter().copied().collect(),
171        spectral_window_ids: spw_ids.iter().copied().collect(),
172        solution_row_count: rows.len(),
173    })
174}

176/// Write the table's `SPECTRAL_WINDOW` as CASA's `gaincal` writes a G or T
177/// table's: the MeasurementSet's rows, each collapsed to the one channel a
178/// solution holds ([`CollapsedWindow`]), the windows without solution rows
179/// flagged (`NewCalTable::flagAbsentSpws`).
180fn write_gain_spectral_window_subtable(
181    ms: &MeasurementSet,
182    output_root: &Path,
183    request: &GainSolveRequest,
184    solved: &SolvedGains,
185) -> Result<(), GainSolveError> {
186    let destination = output_root.join("SPECTRAL_WINDOW");
187    let failed = |source| GainSolveError::CopySubtable {
188        subtable: "SPECTRAL_WINDOW".to_string(),
189        path: output_root.display().to_string(),
190        source: Box::new(source),
191    };
192    ms.subtable(SubtableId::SpectralWindow)
193        .expect("required subtable available")
194        .save(TableOptions::new(&destination))
195        .map_err(failed)?;
196    let mut table = Table::open(TableOptions::new(&destination)).map_err(failed)?;
197
198    let with_rows = solved
199        .rows
200        .iter()
201        .map(|row| row.spw_id)
202        .collect::<BTreeSet<_>>();
203    if let Some(missing) = with_rows
204        .iter()
205        .chain(&solved.attempted_spw_ids)
206        .find(|spw_id| usize::try_from(**spw_id).map_or(true, |row| row >= table.row_count()))
207    {
208        return Err(GainSolveError::UnsupportedParameterShape {
209            path: format!("SPECTRAL_WINDOW rows, which lack the solved window {missing}"),
210            shape: vec![table.row_count()],
211        });
212    }
213    for row in 0..table.row_count() {
214        let spw_id = i32::try_from(row).ok();
215        let has_rows = spw_id.is_some_and(|spw_id| with_rows.contains(&spw_id));
216        let attempted =
217            has_rows || spw_id.is_some_and(|spw_id| solved.attempted_spw_ids.contains(&spw_id));
218        let frequencies = get_f64_array(&table, row, "CHAN_FREQ")?;
219        let widths = get_f64_array(&table, row, "CHAN_WIDTH")?;
220        if frequencies.len() != widths.len() {
221            return Err(GainSolveError::UnsupportedParameterShape {
222                path: format!("SPECTRAL_WINDOW row {row} CHAN_FREQ and CHAN_WIDTH"),
223                shape: vec![frequencies.len(), widths.len()],
224            });
225        }
226        let collapsed = if attempted {
227            let selection = request.selection.channel_selection_for_spw(row as i32);
228            let channels = selected_channels(selection.as_ref(), frequencies.len())?;
229            let window =
230                CollapsedWindow::solved(&frequencies, &widths, channels).ok_or_else(|| {
231                    GainSolveError::UnsupportedParameterShape {
232                        path: format!(
233                            "SPECTRAL_WINDOW row {row}, of which the solve selects no channel"
234                        ),
235                        shape: vec![frequencies.len()],
236                    }
237                })?;
238            Some(window)
239        } else {
240            CollapsedWindow::unsolved(
241                &frequencies,
242                &widths,
243                &get_f64_array(&table, row, "EFFECTIVE_BW")?,
244                &get_f64_array(&table, row, "RESOLUTION")?,
245            )
246        };
247        let mut cells = vec![("FLAG_ROW", Value::Scalar(ScalarValue::Bool(!has_rows)))];
248        if let Some(window) = collapsed {
249            cells.extend([
250                ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(1))),
251                ("CHAN_FREQ", Value::Array(f64_array(&[window.frequency]))),
252                ("CHAN_WIDTH", Value::Array(f64_array(&[window.width]))),
253                (
254                    "EFFECTIVE_BW",
255                    Value::Array(f64_array(&[window.effective_bw])),
256                ),
257                ("RESOLUTION", Value::Array(f64_array(&[window.resolution]))),
258            ]);
259            if let Some(total) = window.total_bandwidth {
260                cells.push((
261                    "TOTAL_BANDWIDTH",
262                    Value::Scalar(ScalarValue::Float64(total)),
263                ));
264            }
265        }
266        for (column, value) in cells {
267            table
268                .cell_accessor_mut(row, column)
269                .and_then(|mut cell| cell.set(value))
270                .map_err(failed)?;
271        }
272    }
273
274    table
275        .save(TableOptions::new(&destination))
276        .map_err(failed)?;
277
278    Ok(())
279}

281/// The one channel of a window's row in a G or T gain table: what CASA's
282/// `gaincal` writes, measured with CASA 6.7.5 (#917).
283#[derive(Clone, Copy, Debug, PartialEq)]
284struct CollapsedWindow {
285    frequency: f64,
286    width: f64,
287    effective_bw: f64,
288    resolution: f64,
289    /// `None` keeps the MeasurementSet's.
290    total_bandwidth: Option<f64>,
291}
292
293impl CollapsedWindow {
294    /// A window a solve reached, over `channels` of `frequencies` and
295    /// `widths` (as many of each): the arithmetic mean of their
296    /// frequencies, unweighted whatever their widths, and the sum of their
297    /// widths' magnitudes. CASA's solve averages and sums them before a G or
298    /// T solve (`ChannelAverageTVI`, over the channels in ascending order;
299    /// `VisibilityIteratorImpl2` holds widths positive), and
300    /// `SolvableVisCal::setCTFrequencies` writes the sum as `CHAN_WIDTH` and
301    /// `EFFECTIVE_BW` and the frequency as `RESOLUTION` and
302    /// `TOTAL_BANDWIDTH`. casa-rs writes those two as CASA does, though they
303    /// are meant to be widths: CASA appends to a table only if they match
304    /// its own (`docs/CASA (C++) bugs.md`). `None` when `channels` is empty
305    /// or names a channel the window lacks.
306    fn solved(frequencies: &[f64], widths: &[f64], mut channels: Vec<usize>) -> Option<Self> {
307        if channels.is_empty() || channels.iter().any(|channel| *channel >= widths.len()) {
308            return None;
309        }
310        channels.sort_unstable();
311        let frequency = channels
312            .iter()
313            .map(|channel| frequencies[*channel])
314            .sum::<f64>()
315            / channels.len() as f64;
316        let width = channels
317            .iter()
318            .map(|channel| widths[*channel].abs())
319            .sum::<f64>();
320        Some(Self {
321            frequency,
322            width,
323            effective_bw: width,
324            resolution: frequency,
325            total_bandwidth: Some(frequency),
326        })
327    }
328
329    /// A window no solve reached, as `SolvableVisCal::createMemCalTable2`
330    /// collapses every window of a table whose parameters do not depend on
331    /// frequency: the mean of all its frequencies and the sums of its
332    /// `CHAN_WIDTH`, `EFFECTIVE_BW` and `RESOLUTION`, signs kept, its
333    /// `TOTAL_BANDWIDTH` the MeasurementSet's. `None` for a window without
334    /// channels, left as it is.
335    fn unsolved(
336        frequencies: &[f64],
337        widths: &[f64],
338        effective_bw: &[f64],
339        resolution: &[f64],
340    ) -> Option<Self> {
341        (!frequencies.is_empty()).then(|| Self {
342            frequency: frequencies.iter().sum::<f64>() / frequencies.len() as f64,
343            width: widths.iter().sum(),
344            effective_bw: effective_bw.iter().sum(),
345            resolution: resolution.iter().sum(),
346            total_bandwidth: None,
347        })
348    }
349}
350
351fn get_f64_array(table: &Table, row: usize, column: &str) -> Result<Vec<f64>, GainSolveError> {
352    match table
353        .cell_accessor(row, column)
354        .and_then(|cell| cell.array())
355        .map_err(MsError::from)
356        .map_err(|source| GainSolveError::OpenMeasurementSet {
357            path: column.to_string(),
358            source,
359        })? {
360        ArrayValue::Float64(values) => Ok(values.iter().copied().collect()),
361        other => Err(GainSolveError::UnsupportedParameterShape {
362            path: format!("{column} array"),
363            shape: other.shape().to_vec(),
364        }),
365    }
366}
367
368fn f64_array(values: &[f64]) -> ArrayValue {
369    ArrayValue::Float64(
370        ArrayD::from_shape_vec(IxDyn(&[values.len()]).f(), values.to_vec())
371            .expect("f64 vector should reshape to 1-D array"),
372    )
373}

375#[cfg(test)]
376mod tests {
377    use super::CollapsedWindow;
378
379    /// `alma/multiobs.ms` (casatestdata): window 1, four channels of
380    /// decreasing frequency, each −117.1875 MHz wide.
381    const LOWER: [f64; 4] = [
382        217_281_281_250.0,
383        217_164_093_750.0,
384        217_046_906_250.0,
385        216_929_718_750.0,
386    ];
387    const LOWER_WIDTH: f64 = -117_187_500.0;
388
389    /// What CASA 6.7.5's `gaincal` (G, `calmode='ap'`) wrote for that
390    /// window: solved, at the arithmetic mean of its selected channels
391    /// with their widths' summed magnitude, the mean again as resolution
392    /// and total bandwidth; without solutions (another window solved), at
393    /// the mean of all its channels with each column's signed sum.
394    #[test]
395    fn windows_collapse_as_casa_gaincal_wrote_them() {
396        let widths = [LOWER_WIDTH; 4];
397        let magnitudes = [-LOWER_WIDTH; 4];
398        for (channels, frequency, width) in [
399            (vec![0, 1, 2, 3], 217_105_500_000.0, 468_750_000.0),
400            (vec![1, 2], 217_105_500_000.0, 234_375_000.0),
401        ] {
402            assert_eq!(
403                CollapsedWindow::solved(&LOWER, &widths, channels.clone()),
404                Some(CollapsedWindow {
405                    frequency,
406                    width,
407                    effective_bw: width,
408                    resolution: frequency,
409                    total_bandwidth: Some(frequency),
410                }),
411                "{channels:?}"
412            );
413        }
414        assert_eq!(
415            CollapsedWindow::unsolved(&LOWER, &widths, &magnitudes, &magnitudes),
416            Some(CollapsedWindow {
417                frequency: 217_105_500_000.0,
418                width: -468_750_000.0,
419                effective_bw: 468_750_000.0,
420                resolution: 468_750_000.0,
421                total_bandwidth: None,
422            })
423        );
424    }
425
426    /// The mean is unweighted whatever the widths: CASA's frequency for
427    /// ngc5921 with its widths rewritten to grow threefold across the
428    /// window stayed the plain mean, 130 kHz from the width-weighted one.
429    /// It sums the channels in ascending order, whatever order the
430    /// selection names them in.
431    #[test]
432    fn the_mean_ignores_the_widths_and_the_selections_order() {
433        let frequencies = [1.0e9, 1.001e9, 1.003e9];
434        let widths = [1.0e6, 3.0e6, 1.0e6];
435        let window = CollapsedWindow::solved(&frequencies, &widths, vec![0, 1]).expect("solved");
436        assert_eq!((window.frequency, window.width), (1.0005e9, 4.0e6));
437        assert_eq!(
438            CollapsedWindow::solved(&frequencies, &widths, vec![2, 0, 1]),
439            CollapsedWindow::solved(&frequencies, &widths, vec![0, 1, 2])
440        );
441    }
442
443    /// No channels, or a channel the window lacks, collapse to nothing.
444    #[test]
445    fn a_selection_outside_the_window_is_refused() {
446        assert_eq!(CollapsedWindow::solved(&[1.0e9], &[1.0e6], vec![]), None);
447        assert_eq!(CollapsedWindow::solved(&[1.0e9], &[1.0e6], vec![1]), None);
448        assert_eq!(CollapsedWindow::unsolved(&[], &[], &[], &[]), None);
449    }
