4//! Metal implementation of the imaging gridding backend.
5//!
6//! This crate is the only place a device API appears in the imaging stack.
7//! [`MetalBackend`] implements the `casa-imaging-operator` backend trait for
8//! both tap layouts and every [`Work`](casa_imaging_operator::Work) variant
9//! with support-generic kernels: one SIMD group per sample, its lanes over
10//! the taps, `f32` grids updated with atomic adds (D2). The host locates
11//! every sample with the operator's rounding rule and accumulates `sumwt`
12//! in `f64`, so tap selection and `sumwt` equal the CPU backend's; only the
13//! order of the `f32` grid additions differs.
14//!
15//! Accumulators live in memory the device addresses
16//! ([`MetalBackend::accumulator`]). Every dispatch completes before `apply`
17//! returns, except grid work a backend made with
18//! [`MetalBackend::with_deferred_grid_completion`] leaves running; the
19//! accumulator's cells wait for that work, so the host reads them like any
20//! other accumulator. Without a
21//! unified-memory Metal 3 device, constructors return
22//! [`DeviceFailure::Unavailable`](casa_imaging_operator::DeviceFailure).
23//!
24//! The design is `docs/imaging-architecture/imaging-foundation-plan-20261007.md`
25//! sections 5.3 and 5.9, ADR-0016, and the IF-4 deviations on #653.

49/// first start to the last end). A pass also logs, once per wave, its

53/// Ring slots of one backend: the host prepares one while the device runs
54/// the others. Twice the sub-blocks of one apply ([`SPLIT`]), so that an
55/// apply whose grid work is left in flight
56/// (`MetalBackend::with_deferred_grid_completion`) seldom waits for the apply
57/// before it.
58const RING: usize = 2 * SPLIT;
59/// Sub-blocks one apply is split into, so the host prepares one while the
60/// device runs another.
61const SPLIT: usize = 3;
62/// The most samples in one ring sub-block.
63const MAX_SUB: usize = 32_768;
64/// Device bytes per sample of a ring slot: the record, its native endpoints
65/// (or a native term), and per visibility polarization a value, a weight,
66/// an inverse norm and an output.
67const SAMPLE_BYTES: usize = 40;
68const SOURCE_BYTES: usize = 24;
69const POLARIZATION_BYTES: usize = 8 + 4 + 8 + 8;
70
71/// The most device memory one backend's ring holds for blocks of `npol`
72/// visibility polarizations; its kernel copies add [`kernel_bytes`], and a
73/// native prediction its batch ([`batch_bytes`]).
74#[must_use]
75pub const fn ring_bytes(npol: usize) -> u64 {
76    (RING * MAX_SUB * (SAMPLE_BYTES + SOURCE_BYTES + npol * POLARIZATION_BYTES)) as u64
77}
78
79/// Device bytes of one kernel table record (`records::TableRecord`).
80const TABLE_RECORD_BYTES: u64 = 16;
81/// Host bytes a backend keeps per kernel cell beside its norms: its table
82/// record, its `KnownTable` and their map entry, at the map's load factor
83/// and growth.
84const HOST_TABLE_BYTES: u64 = 256;
85/// Host and device bytes a backend keeps for its kernel copies before it
86/// meets a cell: its empty arenas and table buffer (16 bytes each) and its
87/// map's first allocation.
88const KERNEL_BASE_BYTES: u64 = 1024;
89
90/// The most memory one backend holds for its copies of `cf`'s kernel
91/// cells, which a pass admits for every backend it makes (#793 review):
92/// the taps of every cell the set can lend, in a device arena reserved
93/// whole from the set's footprint at the first upload, so it never grows
94/// past them; a device table record per cell; and the host's norm tables
95/// and table map ([`ConvolutionFunctionSet::tap_footprint`]). A backend
96/// that meets only some cells holds less; `MetalBackend::kernel_bytes_held`
97/// says how much.
98#[must_use]
99pub fn kernel_bytes(cf: &dyn ConvolutionFunctionSet) -> u64 {
100    let footprint = cf.tap_footprint();
101    footprint.tap_bytes
102        + footprint.norm_bytes
103        + footprint.cells as u64 * (TABLE_RECORD_BYTES + HOST_TABLE_BYTES)
104        + KERNEL_BASE_BYTES
105}

107/// Bytes of every device buffer this crate holds now: kernel copies, rings,
108/// accumulators, model copies and native cells of every live backend. Zero
109/// on a host without Metal.
110#[must_use]
111pub fn device_bytes() -> u64 {
112    #[cfg(all(target_os = "macos", not(coverage)))]
113    {
114        device::live_bytes()
115    }
116    #[cfg(not(all(target_os = "macos", not(coverage))))]
117    {
118        0
119    }
120}

122/// The device memory a backend's predictions of a native-prediction batch
123/// of `samples` samples of `npol` polarizations hold.
124#[must_use]
125pub const fn batch_bytes(npol: usize, samples: usize) -> u64 {
126    (samples * npol * size_of::<num_complex::Complex32>()) as u64
127}
128
129/// The most device memory a backend holds for cell noise
130/// (`MetalBackend::add_cell_noise`) of at most `rows` rows and `terms`
131/// terms: the two buffers it copies them to, made at its first noise and
132/// grown to the largest, each at least a buffer's 16 bytes. A backend that
133/// adds no noise holds none.
134#[must_use]
135pub const fn noise_bytes(rows: usize, terms: usize) -> u64 {
136    (buffer_bytes(rows * size_of::<MetalNoiseRow>())
137        + buffer_bytes(terms * size_of::<MetalNoiseTerm>())) as u64
138}
139
140/// The bytes of a device buffer made for `bytes`: at least 16.
141const fn buffer_bytes(bytes: usize) -> usize {
142    if bytes > 16 { bytes } else { 16 }
143}
