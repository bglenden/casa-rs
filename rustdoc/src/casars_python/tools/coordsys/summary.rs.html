39fn reference_frame(result: &mut String, label: &str, native: &str, current: &str) {
40    let _ = write!(result, "{label} reference : {native}");
41    if native != current {
42        let _ = write!(result, " (-> {current})");
43    }
44    result.push('\n');
45}

47fn metadata(state: &CoordinateState, numbers: &[usize], name: &str) -> String {
48    let coordinates = &state.coordinates;
49    let obs = coordinates.obs_info();
50    let first = |kind| {
51        numbers
52            .iter()
53            .enumerate()
54            .filter(|(index, _)| coordinates.coordinate(*index).coordinate_type() == kind)
55            .min_by_key(|(_, number)| *number)
56            .map(|(index, _)| coordinates.coordinate(index))
57    };
58    let direction = first(CoordinateType::Direction).and_then(|model| match model {
59        CoordinateModel::Direction(direction) => Some(direction),
60        _ => None,
61    });
62    let mut output = String::from("\n");
63    if let Some(direction) = direction {
64        reference_frame(
65            &mut output,
66            "Direction",
67            direction.direction_ref().as_str(),
68            direction.conversion_direction_ref().as_str(),
69        );
70    }
71    if let Some(CoordinateModel::Spectral(spectral)) = first(CoordinateType::Spectral) {
72        let native = spectral.frequency_ref();
73        reference_frame(
74            &mut output,
75            "Spectral ",
76            native.as_str(),
77            spectral
78                .conversion_frequency_ref()
79                .unwrap_or(native)
80                .as_str(),
81        );
82        let _ = writeln!(output, "Velocity  type      : {name}");
83        let rest = spectral.rest_frequency();
84        if rest > 0.0 {
85            let _ = write!(
86                output,
87                "Rest frequency      : {}",
88                format_stream_value(rest)
89            );
90            if spectral.rest_frequencies().len() > 1 {
91                let others = spectral
92                    .rest_frequencies()
93                    .iter()
94                    .filter(|&&value| (value - rest).abs() > 1e-13 * value.abs().max(rest))
95                    .map(|&value| format_stream_value(value))
96                    .collect::<Vec<_>>();
97                let _ = write!(output, " [{}]", others.join(", "));
98            }
99            let _ = writeln!(output, " {}", spectral.axis_units()[0]);
100        }
101    }
102    if let Some(direction) = direction
103        && !obs.pointing_center_initial
104    {
105        let [longitude, latitude] = obs.pointing_center_rad;
106        let _ = writeln!(
107            output,
108            "Pointing center     :  {}  {}",
109            sky_reference(direction, 0, longitude, 6),
110            sky_reference(direction, 1, latitude, 6),
111        );
112    }
113    let _ = writeln!(output, "Telescope           : {}", obs.telescope);
114    let _ = writeln!(output, "Observer            : {}", obs.observer);
115    let date = obs
116        .date
117        .as_ref()
118        .map(|epoch| epoch.value().as_mjd())
119        .filter(|mjd| mjd.floor() != 0.0)
120        .map_or_else(|| "UNKNOWN".into(), |mjd| format_time(mjd, YMD, 12));
121    let _ = writeln!(output, "Date observation    : {date}");
122    if let Some(position) = &obs.telescope_position {
123        let [x, y, z] = position.values();
124        let values = [x, y, z].map(format_stream_value);
125        let units = if position.refer() == PositionRef::ITRF {
126            ["m"; 3]
127        } else {
128            ["rad", "rad", "m"]
129        };
130        let _ = writeln!(
131            output,
132            "Telescope position: [{}{}, {}{}, {}{}] ({})",
133            values[0],
134            units[0],
135            values[1],
136            units[1],
137            values[2],
138            units[2],
139            position.refer(),
140        );
141    }
142    output.push('\n');
143    output
144}

146fn sky_reference(
147    direction: &DirectionCoordinate,
148    axis: usize,
149    value: f64,
150    decimals: u32,
151) -> String {
152    let frame = direction.direction_ref();
153    let degree_frame = matches!(
154        frame,
155        DirectionRef::GALACTIC
156            | DirectionRef::SUPERGAL
157            | DirectionRef::ECLIPTIC
158            | DirectionRef::MECLIPTIC
159            | DirectionRef::TECLIPTIC
160            | DirectionRef::AZEL
161            | DirectionRef::AZELSW
162            | DirectionRef::AZELGEO
163            | DirectionRef::AZELSWGEO
164    );
165    if degree_frame {
166        let mut degrees = value.to_degrees();
167        if axis == 0 {
168            degrees = if matches!(
169                frame,
170                DirectionRef::ECLIPTIC | DirectionRef::MECLIPTIC | DirectionRef::TECLIPTIC
171            ) {
172                (degrees + 180.0).rem_euclid(360.0) - 180.0
173            } else {
174                degrees.rem_euclid(360.0)
175            };
176        }
177        return format!("{degrees:.digits$}", digits = decimals as usize);
178    }
179    let format = match axis {
180        0 if frame == DirectionRef::HADEC => TIME | DIG2,
181        0 => TIME,
182        _ => ANGLE | DIG2,
183    };
184    format_angle(value, format, 6 + decimals)
185}

187fn scientific(value: f64, precision: usize) -> String {
188    if !value.is_finite() {
189        return format_stream_value(value);
190    }
191    let scientific = format!("{value:.precision$e}");
192    let (mantissa, exponent) = scientific.split_once('e').expect("scientific notation");
193    let exponent = exponent.parse::<i32>().expect("decimal exponent");
194    format!("{mantissa}e{exponent:+03}")
195}

197fn coordinate_precision(value: f64, increment: f64) -> usize {
198    if !value.is_finite() || !increment.is_finite() || value.fract() == 0.0 || increment == 0.0 {
199        return 6;
200    }
201    let magnitude = value.abs();
202    let step = increment.abs();
203    let mut needed = 1;
204    let mut threshold = 1.0;
205    while magnitude > threshold {
206        needed += 1;
207        threshold *= 10.0;
208    }
209    if step < 1.0 {
210        let mut threshold = 0.1;
211        while step < threshold {
212            needed += 1;
213            threshold /= 10.0;
214        }
215    } else {
216        let step_digits = step.log10().ceil() as usize + 1;
217        if step_digits < needed {
218            needed -= step_digits;
219        }
220    }
221    if needed > 5 { needed + 1 } else { 6 }
222}

224struct AxisRow {
225    fields: [String; 9],
226    visible: bool,
227    labels_only: bool,
228}
229
230impl AxisRow {
231    fn write(&self, result: &mut String, widths: &[usize; 9]) {
232        for (index, field) in self.fields.iter().enumerate() {
233            if self.labels_only && index >= 6 {
234                break;
235            }
236            let width = widths[index];
237            let _ = if index < 4 {
238                write!(result, "{field:<width$}")
239            } else if index < 8 {
240                write!(result, "{field:>width$}")
241            } else {
242                write!(result, "{field}")
243            };
244        }
245        result.push('\n');
246    }
247
248    fn update_widths(&self, widths: &mut [usize; 9]) {
249        for (width, field) in widths.iter_mut().zip(&self.fields) {
250            *width = (*width).max(field.len());
251        }
252    }
253}

255fn axis_rows(
256    state: &CoordinateState,
257    numbers: &[usize],
258    doppler: DopplerRef,
259) -> PyResult<Vec<AxisRow>> {
260    let cs = &state.coordinates;
261    let mut rows = Vec::new();
262    for pixel in 0..cs.n_pixel_axes() {
263        let (index, local) = cs.pixel_axis_owner(pixel).expect("public pixel axis");
264        let number = numbers[index];
265        rows.extend(coordinate_rows(
266            cs,
267            index,
268            number,
269            local,
270            Some(pixel),
271            doppler,
272        )?);
273        let pixels = cs.coordinate_pixel_axes(index).expect("placed coordinate");
274        if pixels.iter().flatten().copied().max() == Some(pixel) {
275            let worlds = cs.coordinate_world_axes(index).expect("placed coordinate");
276            for (local, (pixel, world)) in pixels.iter().zip(worlds).enumerate() {
277                if pixel.is_none() && world.is_some() {
278                    rows.extend(coordinate_rows(cs, index, number, local, None, doppler)?);
279                }
280            }
281        }
282    }
283    for (index, &number) in numbers.iter().enumerate() {
284        let pixels = cs.coordinate_pixel_axes(index).expect("placed coordinate");
285        if pixels.iter().all(Option::is_none) {
286            let worlds = cs.coordinate_world_axes(index).expect("placed coordinate");
287            for (local, world) in worlds.into_iter().enumerate() {
288                if world.is_some() {
289                    let mut hidden = coordinate_rows(cs, index, number, local, None, doppler)?;
290                    for row in &mut hidden {
291                        row.visible = false;
292                    }
293                    rows.extend(hidden);
294                }
295            }
296        }
297    }
298    Ok(rows)
299}

301fn coordinate_rows(
302    cs: &casa_coordinates::CoordinateSystem,
303    index: usize,
304    number: usize,
305    local: usize,
306    pixel: Option<usize>,
307    doppler: DopplerRef,
308) -> PyResult<Vec<AxisRow>> {
309    let model = cs.coordinate(index);
310    if matches!(model, CoordinateModel::System(_)) && model.n_pixel_axes() != model.n_world_axes() {
311        return Err(PyRuntimeError::new_err(
312            "Cannot summarize a nested coordinate system with unequal pixel and world axes",
313        ));
314    }
315    let mut fields = std::array::from_fn(|_| String::new());
316    fields[0] = pixel.map_or_else(|| "..".into(), |axis| axis.to_string());
317    fields[1] = number.to_string();
318    fields[2] = model.coordinate_type().to_string();
319    fields[3] = model.axis_names()[local].clone();
320    let reference_pixel = model.reference_pixel();
321    let fixed = cs
322        .local_pixel_replacements(index)
323        .unwrap_or(&reference_pixel);
324    let ref_pixel = if pixel.is_some() {
325        reference_pixel[local]
326    } else {
327        fixed[local]
328    };
329    let labels = match model {
330        CoordinateModel::Stokes(stokes) => Some(if pixel.is_some() {
331            stokes
332                .stokes()
333                .iter()
334                .map(|label| label.name())
335                .collect::<Vec<_>>()
336                .join(" ")
337        } else {
338            model
339                .to_world(fixed)
340                .ok()
341                .and_then(|world| StokesType::from_code(world[local] as i32))
342                .map_or_else(|| "?".into(), |label| label.name().into())
343        }),
344        CoordinateModel::Quality(quality) => Some(if pixel.is_some() {
345            quality
346                .values()
347                .iter()
348                .map(|label| label.name())
349                .collect::<Vec<_>>()
350                .join(" ")
351        } else {
352            model.to_world(fixed).ok().map_or_else(
353                || "?".into(),
354                |world| QualityType::from_world(world[local]).name().into(),
355            )
356        }),
357        _ => None,
358    };
359    if let Some(labels) = labels {
360        fields[5] = labels;
361        return Ok(vec![AxisRow {
362            fields,
363            visible: true,
364            labels_only: true,
365        }]);
366    }
367    let value = if pixel.is_some() {
368        model.reference_value()[local]
369    } else {
370        model
371            .to_world(fixed)
372            .map_err(|err| PyRuntimeError::new_err(err.to_string()))?[local]
373    };
374    let increment = model.increment()[local];
375    fields[6] = format!("{ref_pixel:.2}");
376    let mut row_increment = increment;
377    let mut unit = model.axis_units()[local].clone();
378    let mut precision = 6;
379    let mut formatted_increment = None;
380    match model {
381        CoordinateModel::Direction(direction) => {
382            fields[4] = if direction.is_ncp() {
383                "NCP".into()
384            } else {
385                direction.projection().projection_type().to_string()
386            };
387            fields[5] = sky_reference(direction, local, value, 3);
388            let hours = direction.direction_ref() == DirectionRef::HADEC && local == 0;
389            row_increment = if hours {
390                increment.to_degrees() / 15.0
391            } else {
392                increment.to_degrees() * 3600.0
393            };
394            unit = if hours { "h".into() } else { "arcsec".into() };
395        }
396        CoordinateModel::Spectral(spectral) => {
397            if !spectral.has_linear_mapping() {
398                fields[3].push_str(" (tab)");
399            }
400            fields[5] = format_stream_value(value);
401        }
402        _ => {
403            fields[5] = format_stream_value_with_precision(
404                value,
405                coordinate_precision(value, model.increment()[0]),
406            );
407            precision = coordinate_precision(increment, model.increment()[0]);
408            if let Some((value, increment, units)) =
409                nested_axis_display(model, local, value, increment)?
410            {
411                fields[5] = value;
412                formatted_increment = Some(increment);
413                unit = units;
414            }
415        }
416    }
417    fields[7] = if pixel.is_some() {
418        formatted_increment.unwrap_or_else(|| scientific(row_increment, precision))
419    } else {
420        String::new()
421    };
422    if pixel.is_none() && matches!(model, CoordinateModel::Direction(_)) {
423        unit.clear();
424    }
425    fields[8] = format!(" {unit}");
426    let mut rows = vec![AxisRow {
427        fields,
428        visible: true,
429        labels_only: false,
430    }];
431    if let CoordinateModel::Spectral(spectral) = model
432        && spectral.rest_frequency() > 0.0
433    {
434        let velocity_unit = Unit::new("km/s").expect("standard velocity unit");
435        let conversion = VelocityConversion::for_spectral_axis(spectral, &velocity_unit, doppler)?;
436        let mut fields = std::array::from_fn(|_| String::new());
437        fields[3] = "Velocity".into();
438        fields[5] = conversion
439            .mixed_velocity(spectral.reference_value()[0])
440            .map_or_else(|_| "Fail".into(), format_stream_value);
441        if pixel.is_some() {
442            fields[6] = format!("{ref_pixel:.2}");
443            let velocity_at = |offset| {
444                let world = spectral.to_world(&[ref_pixel + offset]).ok()?[0];
445                conversion.mixed_velocity(world).ok()
446            };
447            fields[7] = velocity_at(0.5)
448                .zip(velocity_at(-0.5))
449                .map_or_else(|| "Fail".into(), |(high, low)| scientific(high - low, 6));
450            fields[8] = " km/s".into();
451        }
452        rows.push(AxisRow {
453            fields,
454            visible: true,
455            labels_only: false,
456        });
457    }
458    Ok(rows)
459}

461fn coordinate_summary(
462    state: &CoordinateState,
463    doppler: DopplerRef,
464    name: &str,
465) -> PyResult<String> {
466    let placements = state
467        .coordinates
468        .coordinates_in_record(&state.record)
469        .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
470    let numbers = placements.iter().map(coordinate_index).collect::<Vec<_>>();
471    let mut output = metadata(state, &numbers, name);
472    let rows = axis_rows(state, &numbers, doppler)?;
473    let header = AxisRow {
474        fields: COLUMNS.map(str::to_owned),
475        visible: true,
476        labels_only: false,
477    };
478    let mut widths = COLUMNS.map(str::len);
479    for row in &rows {
480        row.update_widths(&mut widths);
481    }
482    for width in &mut widths[..8] {
483        *width += 1;
484    }
485    header.write(&mut output, &widths);
486    let _ = writeln!(output, "{} ", "-".repeat(widths.iter().sum::<usize>() - 1));
487    for row in rows.into_iter().filter(|row| row.visible) {
488        row.write(&mut output, &widths);
489    }
490    output.pop();
491    Ok(output)
492}

494struct ListFlag(bool);
495
496impl FromPyObject<'_, '_> for ListFlag {
497    type Error = PyErr;
498
499    fn extract(value: Borrowed<'_, '_, PyAny>) -> PyResult<Self> {
500        value
501            .cast::<PyBool>()
502            .map(|flag| Self(flag.is_true()))
503            .map_err(|_| {
504                PyTypeError::new_err("in method 'coordsys_summary', argument 3 of type 'bool'")
505            })
506    }
507}

509#[pymethods]
510impl PyCoordsys {
511    #[pyo3(signature = (doppler=FrameName::radio_default(), list=ListFlag(true)), text_signature = "($self, doppler='RADIO', list=True)")]
512    fn summary(&self, py: Python<'_>, doppler: FrameName, list: ListFlag) -> PyResult<Vec<String>> {
513        let name = doppler.into_name("doppler")?;
514        let canonical = DOPPLER_CODES.resolve(&name);
515        let mut logs = Vec::new();
516        if canonical.is_none() {
517            logs.push(LogLine::warning("Illegal doppler type, using RADIO"));
518        }
519        let name = canonical.unwrap_or("RADIO");
520        let convention = doppler_ref(name).map_err(|err| PyTypeError::new_err(err.to_string()))?;
521        let text = py.detach(|| coordinate_summary(&self.state(), convention, name))?;
522        if list.0 {
523            logs.push(LogLine::info(text));
524            emit(py, "casars.tools.coordsys", &logs)?;
525            Ok(vec!["T".into()])
526        } else {
527            emit(py, "casars.tools.coordsys", &logs)?;
528            Ok(vec![text])
529        }
530    }
