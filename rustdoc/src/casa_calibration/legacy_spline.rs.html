13/// Legacy time-dependent diagonal gains, keyed by antenna and field.
14pub struct LegacySplineTable {
15    rows: HashMap<(i32, usize), Row>,
16    windows: usize,
17}
18
19struct Row {
20    antenna: i32,
21    field: i32,
22    scale: Complex32,
23    reference_hz: f64,
24    amplitude: Option<Curve>,
25    phase: Option<Curve>,
26}
27
28struct Curve {
29    knots: Vec<f64>,
30    coefficients: Vec<f64>,
31    stride: usize,
32}
33
34impl LegacySplineTable {
35    /// Number of data windows covered at the time the table was configured.
36    pub fn windows(&self) -> usize {
37        self.windows
38    }

40    /// Inspect the coefficient cell shapes without loading the arrays.
41    ///
42    /// # Errors
43    /// Returns a string describing an invalid table or a size overflow.
44    pub fn layout(path: &Path, windows: usize) -> Result<CalibrationTableLayout, String> {
45        let table = Table::open(TableOptions::new(path)).map_err(|error| error.to_string())?;
46        let rows = table.row_count();
47        let mut values = 0usize;
48        for name in [
49            "SPLINE_KNOTS_AMP",
50            "SPLINE_KNOTS_PHASE",
51            "POLY_COEFF_AMP",
52            "POLY_COEFF_PHASE",
53        ] {
54            let column = table
55                .column_accessor(name)
56                .map_err(|error| error.to_string())?;
57            for start in (0..rows).step_by(512) {
58                let selected: Vec<_> = (start..rows.min(start + 512)).collect();
59                for shape in column
60                    .array_shapes_uncached(&selected)
61                    .map_err(|error| error.to_string())?
62                    .into_iter()
63                    .flatten()
64                {
65                    let size = shape
66                        .into_iter()
67                        .try_fold(1usize, usize::checked_mul)
68                        .ok_or_else(|| "GSPLINE cell size overflows usize".to_owned())?;
69                    values = values
70                        .checked_add(size)
71                        .ok_or_else(|| "GSPLINE value count overflows usize".to_owned())?;
72                }
73            }
74        }
75        Ok(CalibrationTableLayout {
76            rows,
77            values,
78            max_cell_values: 2,
79            frequency_values: 0,
80            frequency_windows: windows,
81        })
82    }

84    /// Load the first matching solution for each antenna and field in disk order.
85    ///
86    /// # Errors
87    /// Returns a string describing a malformed or inaccessible table.
88    pub fn open(path: &Path, windows: usize) -> Result<Self, String> {
89        let table = Table::open(TableOptions::new(path)).map_err(|error| error.to_string())?;
90        let mut rows = HashMap::with_capacity(table.row_count());
91        for index in 0..table.row_count() {
92            let mode = match table
93                .cell_accessor(index, "POLY_MODE")
94                .map_err(|error| error.to_string())?
95                .scalar()
96                .map_err(|error| error.to_string())?
97            {
98                ScalarValue::String(mode) => mode,
99                _ => return Err("GSPLINE POLY_MODE is not a string".to_owned()),
100            };
101            let scale = match table
102                .cell_accessor(index, "SCALE_FACTOR")
103                .map_err(|error| error.to_string())?
104                .scalar()
105                .map_err(|error| error.to_string())?
106            {
107                ScalarValue::Complex32(value) => *value,
108                _ => return Err("GSPLINE SCALE_FACTOR is not complex".to_owned()),
109            };
110            let reference = floats(&table, index, "REF_FREQUENCY")?;
111            let row = Row {
112                antenna: integer(&table, index, "ANTENNA1")?,
113                field: integer(&table, index, "FIELD_ID")?,
114                scale,
115                reference_hz: reference.first().copied().unwrap_or(0.),
116                amplitude: (mode.contains("AMP") || mode.contains("A&P"))
117                    .then(|| Curve::open(&table, index, "AMP"))
118                    .transpose()?,
119                phase: (mode.contains("PHAS") || mode.contains("A&P"))
120                    .then(|| Curve::open(&table, index, "PHASE"))
121                    .transpose()?,
122            };
123            rows.entry((row.antenna, row.field as usize)).or_insert(row);
124        }
125        Ok(Self { rows, windows })
126    }

128    /// Evaluate a two-receptor gain at the requested time and mean data frequency.
129    /// No solution exists when the table has no row for the antenna and field.
130    pub fn solution(
131        &self,
132        antenna: i32,
133        field: usize,
134        time: f64,
135        frequency_hz: Option<f64>,
136    ) -> Option<ComplexSolution> {
137        // The legacy buffer matches antenna AND field and takes its first
138        // matching row. Interpolation arguments do not change these splines.
139        let row = self.rows.get(&(antenna, field))?;
140        let ratio = frequency_hz
141            .filter(|_| row.reference_hz.abs() > 0.)
142            .map_or(1., |frequency| frequency / row.reference_hz);
143        let values = (0..2)
144            .map(|receptor| {
145                let amplitude = row
146                    .amplitude
147                    .as_ref()
148                    .map_or(1., |c| c.value(time, receptor).exp());
149                // GILDAS phases have the opposite sign; CASA also scales them
150                // by mean observing frequency / calibration reference frequency.
151                let phase = row
152                    .phase
153                    .as_ref()
154                    .map_or(0., |c| -c.value(time, receptor) * ratio);
155                row.scale * Complex32::from_polar(amplitude as f32, phase as f32)
156            })
157            .collect();
158        Some(ComplexSolution {
159            receptors: 2,
160            channels: 1,
161            values,
162            flags: vec![false; 2],
163        })
164    }
165}

167impl Curve {
168    fn open(table: &Table, row: usize, suffix: &str) -> Result<Self, String> {
169        let stride = usize::try_from(integer(table, row, &format!("N_POLY_{suffix}"))?)
170            .map_err(|_| "GSPLINE coefficient count is negative".to_owned())?;
171        let count = usize::try_from(integer(table, row, &format!("N_KNOTS_{suffix}"))?)
172            .map_err(|_| "GSPLINE knot count is negative".to_owned())?;
173        let mut knots = floats(table, row, &format!("SPLINE_KNOTS_{suffix}"))?;
174        let mut coefficients = floats(table, row, &format!("POLY_COEFF_{suffix}"))?;
175        let used_coefficients = stride
176            .checked_mul(2)
177            .ok_or("GSPLINE coefficient count overflow")?;
178        // A table assembled by append can have a common physical size for
179        // rows with different logical lengths. The tails are unused.
180        knots.truncate(count);
181        coefficients.truncate(used_coefficients);
182        if count < 8
183            || knots.len() != count
184            || stride < count - 4
185            || coefficients.len() != used_coefficients
186            || knots.iter().chain(&coefficients).any(|v| !v.is_finite())
187            || knots.windows(2).any(|w| w[0] > w[1])
188            || knots[3] >= knots[count - 4]
189        {
190            return Err("invalid GSPLINE cubic knot/coefficient layout".to_owned());
191        }
192        Ok(Self {
193            knots,
194            coefficients,
195            stride,
196        })
197    }
198
199    fn value(&self, time: f64, receptor: usize) -> f64 {
200        let knots = &self.knots;
201        let count = knots.len() - 4;
202        // GETBSPL returns zero outside the fitted interval. Consequently
203        // the gain retains SCALE_FACTOR, with unit amplitude and zero phase.
204        if time < knots[3] || time > knots[count] {
205            return 0.;
206        }
207        let span = knots
208            .partition_point(|&knot| knot <= time)
209            .saturating_sub(1)
210            .clamp(3, count - 1);
211        let coefficients = &self.coefficients[receptor * self.stride..];
212        let mut values: [f64; 4] = std::array::from_fn(|j| coefficients[span - 3 + j]);
213        // de Boor's cubic recurrence, retaining only four values on the stack.
214        for order in 1..=3 {
215            for j in (order..=3).rev() {
216                let i = span - 3 + j;
217                let width = knots[i + 4 - order] - knots[i];
218                let fraction = if width > 0. {
219                    (time - knots[i]) / width
220                } else {
221                    0.
222                };
223                values[j] = (1. - fraction) * values[j - 1] + fraction * values[j];
224            }
225        }
226        values[3]
227    }
228}

230fn integer(table: &Table, row: usize, name: &str) -> Result<i32, String> {
231    match table
232        .cell_accessor(row, name)
233        .map_err(|error| error.to_string())?
234        .scalar()
235        .map_err(|error| error.to_string())?
236    {
237        ScalarValue::Int32(value) => Ok(*value),
238        _ => Err(format!("GSPLINE {name} is not an integer")),
239    }
240}
241
242fn floats(table: &Table, row: usize, name: &str) -> Result<Vec<f64>, String> {
243    match table
244        .cell_accessor(row, name)
245        .map_err(|error| error.to_string())?
246        .array()
247        .map_err(|error| error.to_string())?
248    {
249        ArrayValue::Float64(values) => Ok(values.iter().copied().collect()),
250        _ => Err(format!("GSPLINE {name} is not a double array")),
251    }
252}
