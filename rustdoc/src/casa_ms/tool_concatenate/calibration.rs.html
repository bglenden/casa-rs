2//! Concatenate optional weather, calibration, and Earth-orientation rows.
3
4use std::collections::BTreeMap;
5
6use casa_tables::Table;
7use casa_types::{RecordValue, ScalarValue, Value};
8
9use super::{TableRowsChange, main_rows::sparse_id};
10use crate::{MeasurementSet, MsError, MsResult, SubtableId};
11
12pub(super) fn reconcile<'a>(
13    target: &'a MeasurementSet,
14    donor: &MeasurementSet,
15    extra: [&'a BTreeMap<&str, Table>; 2],
16    antenna_ids: &[usize],
17    spectral_window_ids: &[usize],
18    remap_spectral_windows: bool,
19    observation_ids: [&[usize]; 2],
20) -> MsResult<Vec<(&'a Table, TableRowsChange)>> {
21    let [attached_observation_ids, donor_observation_ids] = observation_ids;
22    let mut changes = Vec::new();
23    for (id, name) in [
24        (Some(SubtableId::SysCal), "SYSCAL"),
25        (Some(SubtableId::Weather), "WEATHER"),
26        (None, "GAIN_CURVE"),
27        (None, "PHASE_CAL"),
28        (None, "EARTH_ORIENTATION"),
29    ] {
30        let Some(target_table) = linked_table(target, extra[0], id, name) else {
31            continue;
32        };
33        if target_table.row_count() == 0 {
34            continue;
35        }
36        let mut records = linked_table(donor, extra[1], id, name)
37            .filter(|table| table.row_count() > 0)
38            .map(|table| table.rows().map(|rows| rows.to_vec()))
39            .transpose()?
40            .unwrap_or_default();
41        for record in &mut records {
42            if name == "EARTH_ORIENTATION" {
43                remap(record, name, "OBSERVATION_ID", donor_observation_ids, true)?;
44                continue;
45            }
46            remap(record, name, "ANTENNA_ID", antenna_ids, name == "WEATHER")?;
47            if remap_spectral_windows && name != "WEATHER" {
48                remap(
49                    record,
50                    name,
51                    "SPECTRAL_WINDOW_ID",
52                    spectral_window_ids,
53                    false,
54                )?;
55            }
56        }
57        let updated = if name == "EARTH_ORIENTATION" && !attached_observation_ids.is_empty() {
58            remap_attached_observations(target_table, attached_observation_ids)?
59        } else {
60            Vec::new()
61        };
62        if !records.is_empty() || !updated.is_empty() {
63            changes.push((
64                target_table,
65                TableRowsChange {
66                    appended: records,
67                    updated,
68                    ..TableRowsChange::default()
69                },
70            ));
71        }
72    }
73    Ok(changes)
74}

76fn remap_attached_observations(
77    table: &Table,
78    ids: &[usize],
79) -> MsResult<Vec<(usize, RecordValue)>> {
80    let name = "OBSERVATION_ID";
81    table
82        .column_accessor(name)?
83        .scalar_cells_owned()?
84        .into_iter()
85        .enumerate()
86        .filter_map(|(row, value)| {
87            let Some(ScalarValue::Int32(old)) = value else {
88                return Some(Err(MsError::InvalidInput(
89                    "EARTH_ORIENTATION.OBSERVATION_ID requires an integer".into(),
90                )));
91            };
92            match sparse_id(ids, old) {
93                Ok(new) if new == old => None,
94                Ok(new) => Some(Ok((
95                    row,
96                    RecordValue::new(vec![casa_types::RecordField::new(
97                        name,
98                        Value::Scalar(ScalarValue::Int32(new)),
99                    )]),
100                ))),
101                Err(error) => Some(Err(error)),
102            }
103        })
104        .collect()
105}

107fn linked_table<'a>(
108    ms: &'a MeasurementSet,
109    extra: &'a BTreeMap<&str, Table>,
110    id: Option<SubtableId>,
111    name: &str,
112) -> Option<&'a Table> {
113    match id {
114        Some(id) => ms.subtable(id),
115        None => extra.get(name),
116    }
117}
118
119fn remap(
120    row: &mut RecordValue,
121    table: &str,
122    column: &str,
123    ids: &[usize],
124    allow_undefined: bool,
125) -> MsResult<()> {
126    let Some(Value::Scalar(ScalarValue::Int32(index))) = row.get(column) else {
127        return Err(MsError::InvalidInput(format!(
128            "{table}.{column} requires an integer"
129        )));
130    };
131    if *index == -1 && allow_undefined {
132        return Ok(());
133    }
134    let replacement = usize::try_from(*index)
135        .ok()
136        .and_then(|index| ids.get(index))
137        .and_then(|&id| i32::try_from(id).ok())
138        .ok_or_else(|| {
139            MsError::InvalidInput(format!(
140                "{table}.{column} contains invalid donor ID {index}"
141            ))
142        })?;
143    row.upsert(column, Value::Scalar(ScalarValue::Int32(replacement)));
144    Ok(())
145}
