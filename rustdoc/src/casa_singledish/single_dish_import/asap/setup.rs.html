2//! Header and spectral-window metadata from an ASAP scan table.
3
4use std::collections::BTreeMap;
5use std::path::Path;
6
7use casa_core::measures::frequency::CasaFrequencyFrame;
8use casa_tables::Table;
9
10use super::super::{Antenna, ImportHeader, Window, antenna_diameter};
11use super::asap_row::{doubles, flags, integer, invalid, number, string, subtable};
12use casa_ms::MsResult;

14pub(super) fn header(path: &Path, table: &Table) -> MsResult<ImportHeader> {
15    let keywords = table.keywords();
16    let raw_name = string(keywords, "AntennaName")?;
17    let (observatory, antenna) = raw_name.split_once("//").unwrap_or_else(|| {
18        let (observatory, _) = raw_name.split_once('@').unwrap_or((raw_name, ""));
19        (observatory, raw_name)
20    });
21    let (name, station) = antenna.split_once('@').unwrap_or((antenna, ""));
22    let position = doubles(keywords, "AntennaPosition")?
23        .try_into()
24        .map_err(|_| invalid("AntennaPosition"))?;
25    let times = (0..table.row_count())
26        .map(|index| number(table.row_accessor().row(index)?, "TIME"))
27        .collect::<MsResult<Vec<_>>>()?;
28    let range = times
29        .into_iter()
30        .fold([f64::INFINITY, f64::NEG_INFINITY], |[min, max], time| {
31            [min.min(time), max.max(time)]
32        });
33    Ok(ImportHeader {
34        antenna: vec![Antenna {
35            name: name.into(),
36            station: station.into(),
37            position,
38            diameter: antenna_diameter(name),
39        }],
40        direction_frame: "J2000",
41        telescope: observatory.into(),
42        observer: string(keywords, "Observer")?.into(),
43        project: string(keywords, "Project")?.into(),
44        range: range.map(|time| time * super::DAY),
45        data_unit: string(keywords, "FluxUnit")?.into(),
46        windows: windows(path, table)?,
47    })
48}

50fn windows(path: &Path, table: &Table) -> MsResult<BTreeMap<i32, Window>> {
51    let frequencies = subtable(path, table.keywords(), "FREQUENCIES")?;
52    let frame = string(frequencies.keywords(), "BASEFRAME")?.parse::<CasaFrequencyFrame>();
53    let frame_code = match frame {
54        Ok(CasaFrequencyFrame::Defined(frame)) => frame.casacore_code(),
55        Ok(CasaFrequencyFrame::Undefined) => 64,
56        Err(_) => 9,
57    };
58    let by_id = (0..frequencies.row_count())
59        .map(|index| {
60            let row = frequencies.row_accessor().row(index)?;
61            Ok((integer(row, "ID")?, row))
62        })
63        .collect::<MsResult<BTreeMap<_, _>>>()?;
64    let mut iter = BTreeMap::new();
65    for index in 0..table.row_count() {
66        let row = table.row_accessor().row(index)?;
67        let window = integer(row, "IFNO")?;
68        if let std::collections::btree_map::Entry::Vacant(entry) = iter.entry(window) {
69            let frequency = by_id
70                .get(&integer(row, "FREQ_ID")?)
71                .ok_or_else(|| invalid("FREQ_ID"))?;
72            entry.insert(Window {
73                channels: flags(row, "FLAGTRA")?.len(),
74                frame: frame_code,
75                reference_channel: number(frequency, "REFPIX")?,
76                reference_frequency: number(frequency, "REFVAL")?,
77                increment: number(frequency, "INCREMENT")?,
78            });
79        }
80    }
81    Ok(iter)
82}
