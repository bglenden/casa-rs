2//! The loop of major and minor cycles on the major-cycle pass.
3//!
4//! The initial pass grids the data and the PSF (the residual of a start
5//! model when there is one); each later pass grids the residual `V − A·m`
6//! of the model the preceding minor cycle produced. Imaging weights are
7//! generated once, before the initial pass, and reused by every pass. The
8//! final pass also writes the model column when the run asks for it.

125/// Run every major and minor cycle of `inputs.problem`.
126///
127/// The loop is tclean's: after each major cycle the controller decides from
128/// the fresh residual whether cleaning is complete (`cleanComplete`);
129/// otherwise a minor cycle runs under its controls and a major cycle
130/// refreshes the residual. A minor cycle that cleans nothing ends the run
131/// without another major cycle. Each cycle is a phase of `summary`; a
132/// cancelled run stops at the next block boundary of a pass or before the
133/// next phase.
134pub(crate) fn run(
135    inputs: ImagingInputs<'_>,
136    summary: &mut RunSummary,
137) -> Result<ImagingOutcome, ImagingError> {
138    let problem = inputs.problem;
139    let controls = problem.reconstruction().controls();
140    let cleaning = controls.max_minor_iterations() > 0;
141    let setup = minor_setup(problem, inputs.image_response, &inputs.masks);
142    let mut mask_plans = inputs.masks.clone();
143    let cancel = inputs.cancel.clone();
144    // Operators, the source, the imaging weights and the paged cube state.
145    let mut run = run_phase("setup", &cancel, summary, || Run::open(inputs))?;
146    let major = run_phase("major cycle 1", &cancel, summary, || run.initial(!cleaning))?;
147    let mut outcome = ImagingOutcome::initial(major, &run);
148    if !cleaning {
149        return Ok(outcome);
150    }
151    let mut controller = Controller::new(&controls);
152    let mut cache = PsfCache::default();
153    loop {
154        let prepared = prepare_minor_cycle(
155            &outcome.scientific,
156            &mask_plans,
157            &setup,
158            &mut cache,
159            run.team,
160        )?;
161        let statistics = prepared.statistics;
162        if let Some(stop) = controller.clean_complete(&statistics) {
163            tracing::info!("imaging stopped: {stop:?} (stopcode {})", stop.code());
164            outcome.stop = Some(stop);
165            outcome
166                .masks
167                .get_or_insert_with(|| ReconstructionMaskSet::Domains(prepared.masks().clone()));
168            break;
169        }
170        let cycle_controls = controller.cycle_controls(&statistics);
171        let started = Instant::now();
172        let name = format!("minor cycle {}", outcome.minor_cycles.len() + 1);
173        let minor = run_phase(name, &cancel, summary, || {
174            Ok::<_, ImagingError>(run_minor_cycle(
175                prepared,
176                &outcome.scientific,
177                &setup,
178                &cycle_controls,
179                &mut cache,
180                run.team,
181            )?)
182        })?;
183        let global_threshold = controls
184            .threshold_jy_per_beam()
185            .max(statistics.nsigma_threshold);
186        let cycle = account_minor_cycle(
187            &mut outcome,
188            &mut controller,
189            &minor,
190            &cycle_controls,
191            global_threshold,
192            started,
193        );
194        let last = controller.budget_spent() || controller.last_cycle(&statistics);
195        mask_plans = next_masks(&mask_plans, &minor, cycle, cycle_controls.threshold_reached)?;
196        if minor.summary.iterations == 0 {
197            // tclean skips the major cycle, updates the mask and still asks
198            // `hasConverged` for the stop it reports.
199            let prepared = prepare_minor_cycle(
200                &outcome.scientific,
201                &mask_plans,
202                &setup,
203                &mut cache,
204                run.team,
205            )?;
206            outcome.stop = controller.clean_complete(&prepared.statistics);
207            outcome.masks = Some(ReconstructionMaskSet::Domains(prepared.masks().clone()));
208            break;
209        }
210        let name = format!("major cycle {}", outcome.major_cycle_count + 1);
211        let scientific = outcome.scientific;
212        let major = run_phase(name, &cancel, summary, || {
213            run.refresh(scientific, minor.terms, &minor.masks, last)
214        })?;
215        controller.end_major_cycle();
216        outcome.scientific = major.completion;
217        outcome.visibility_products = major.visibility;
218        outcome.major_cycle_count += 1;
219    }
220    if run.visibility_write.is_some() && outcome.visibility_products.is_none() {
221        outcome = run_phase("model prediction", &cancel, summary, || {
222            predict_final_model(&mut run, outcome)
223        })?;
224    }
225    outcome.planes_per_wave = run.planes_per_wave;
226    Ok(outcome)
227}

229/// Fold one minor cycle into the run's totals and its controller
230/// (`mergeMinorCycleSummary`), record and log it; returns its ordinal.
231fn account_minor_cycle(
232    outcome: &mut ImagingOutcome,
233    controller: &mut Controller,
234    minor: &MinorCycleOutcome,
235    cycle_controls: &CycleControls,
236    global_threshold: f64,
237    started: Instant,
238) -> usize {
239    let entering = (
240        outcome.total_minor_iterations,
241        outcome.total_actual_minor_iterations,
242    );
243    controller.record_minor_cycle(minor.summary.iterations, minor.summary.peak);
244    outcome.total_minor_iterations = controller.iterations();
245    outcome.total_actual_minor_iterations += minor.summary.components;
246    let cycle = outcome.minor_cycles.len() + 1;
247    let record = minor_cycle_record(
248        cycle,
249        minor,
250        cycle_controls,
251        global_threshold,
252        entering,
253        (
254            outcome.total_minor_iterations,
255            outcome.total_actual_minor_iterations,
256        ),
257    );
258    tracing::info!(
259        "imaging minor cycle {cycle}: {} iterations ({} total), peak {:.6} -> {:.6} Jy, \
260         cycle threshold {:.6} Jy, stop {:?}, {:.2} s",
261        record.iterations,
262        record.total_iterations,
263        record.initial_peak_flux,
264        record.final_peak_flux,
265        record.effective_threshold,
266        record.stop_reason,
267        started.elapsed().as_secs_f64(),
268    );
269    outcome.minor_cycles.push(record);
270    outcome.masks = Some(ReconstructionMaskSet::Domains(minor.masks.clone()));
271    cycle
272}

274/// Predict the final model once more to write the model column: the pass
275/// that ended the run was not known to be the last when it ran.
276fn predict_final_model(
277    run: &mut Run<'_>,
278    mut outcome: ImagingOutcome,
279) -> Result<ImagingOutcome, ImagingError> {
280    let masks = match &outcome.masks {
281        Some(ReconstructionMaskSet::Domains(masks)) => masks.clone(),
282        _ => unreachable!("a cleaning run has formed its masks"),
283    };
284    let major = run.refresh(outcome.scientific, Vec::new(), &masks, true)?;
285    outcome.scientific = major.completion;
286    outcome.visibility_products = major.visibility;
287    outcome.major_cycle_count += 1;
288    Ok(outcome)
289}

291/// The minor cycle's view of the run.
292fn minor_setup(
293    problem: &CompiledProblem,
294    response: Option<MinorCycleImageResponse>,
295    masks: &ImageDomainReconstructionMaskPlans,
296) -> MinorCycleSetup {
297    let controls = problem.reconstruction().controls();
298    MinorCycleSetup {
299        algorithm: problem.reconstruction().algorithm().clone(),
300        accounting: controls.hogbom_iteration_accounting(),
301        response,
302        nsigma: controls.noise_sigma().unwrap_or(0.0),
303        automask: matches!(
304            masks.primary(),
305            ReconstructionMaskPlan::AutoMultithresh { .. }
306        ),
307    }
308}

310impl<'a> Run<'a> {
311    /// The operators, worker team, source and imaging weights of a run.
312    fn open(inputs: ImagingInputs<'a>) -> Result<Self, ImagingError> {
313        let problem = inputs.problem;
314        let backend = inputs.backend;
315        if backend == BackendChoice::Metal && !inputs.host.metal {
316            return Err(ImagingError::Unsupported {
317                reason: "the Metal backend needs a unified-memory Metal 3 device",
318            });
319        }
320        let correlations = selected_correlations(problem)?;
321        let selected = inputs
322            .access
323            .into_deferred()
324            .open(problem)
325            .map_err(|error| ImagingError::Observation(Box::new(error)))?;
326        let dish_classes = selected.antenna_response_classes();
327        let domains = problem
328            .geometry()
329            .domains()
330            .iter()
331            .map(|domain| {
332                domain_operator(
333                    problem,
334                    domain,
335                    &correlations,
336                    backend,
337                    inputs.aw_catalog.as_ref(),
338                    &dish_classes,
339                )
340            })
341            .collect::<Result<Vec<_>, _>>()?;
342        let weight_image = domains.iter().any(|domain| domain.weight_image);
343        if inputs.visibility_write.is_some() && domains.len() > 1 {
344            return Err(ImagingError::Unsupported {
345                reason: "visibilities are written back for one image domain",
346            });
347        }
348        let team = inputs.team;
349        let workers = team.workers();
350        let cancel = inputs.cancel;
351        let main = &domains[0].operator;
352        let mut source = MeasurementSetSource::new(
353            problem,
354            selected,
355            main.basis().planes(),
356            plane_bounds(problem),
357            domains
358                .iter()
359                .any(|domain| domain.operator.cf().pointing_ramp()),
360            dish_classes,
361        );
362        let started = Instant::now();
363        let weighting = weighting(problem, &domains[0], &mut source, team, &cancel)?;
364        tracing::info!(
365            "imaging weights: {workers} workers, {:.2} s",
366            started.elapsed().as_secs_f64()
367        );
368        let (host, policy) = (inputs.host, inputs.policy);
369        let cube = matches!(main.basis(), Basis::ChannelLocal { .. })
370            .then(|| cube_state(problem, inputs.spill_directory, workers, &host, &policy))
371            .transpose()?;
372        let run = Self {
373            problem,
374            domains,
375            source,
376            weighting,
377            weighting_id: WeightingGenerationId::next(),
378            team,
379            cancel,
380            host,
381            policy,
382            cube,
383            native_spacing_hz: native_spacing_hz(problem),
384            backend,
385            attempts: 0,
386            visibility_write: inputs.visibility_write,
387            planes_per_wave: None,
388            weight_image,
389        };
390        if run.visibility_write.is_some() {
391            // The pass that writes is the initial one without cleaning and a
392            // residual refresh otherwise; it must hold every plane, and is
393            // refused now rather than after the cycles that precede it.
394            let cleaning = problem.reconstruction().controls().max_minor_iterations() > 0;
395            let (modes, with_model) = if cleaning {
396                (ModeSet::DATA, true)
397            } else {
398                (run.initial_modes(), start_model(problem))
399            };
400            if run.admit_pass(modes, with_model)?.0 != Residency::All {
401                return Err(ImagingError::Pass(PassError::VisibilityWriteWaves));
402            }
403        }
404        Ok(run)
405    }

407    /// The initial major cycle: data and PSF, from the start model when the
408    /// problem has one; it writes visibilities when it is also final.
409    fn initial(&mut self, last: bool) -> Result<Major, ImagingError> {
410        let mut lifecycle = ModelLifecycle::bind(
411            ExecutableModelProblem::from_compiled(self.problem.clone())?,
412            self.next_attempt(),
413            1,
414            self.model_storage()?,
415        )?;
416        let start_model = !matches!(lifecycle.contract().input(), ModelInputCommitment::Empty);
417        let named = if start_model {
418            lifecycle.initial_reprojected()?
419        } else {
420            lifecycle.initial_empty()?
421        };
422        let preparation = MajorCyclePreparation::prepare(&lifecycle, named, None)?;
423        let (residency, _pass) = self.admit_pass(self.initial_modes(), start_model)?;
424        let state = PassNormalState::initial(
425            self.problem,
426            self.weighting_id,
427            preparation.final_model_generation(),
428            self.normal_storage(residency)?,
429        )?;
430        self.reconcile(lifecycle, state, preparation, None, last, residency)
431    }
432
433    /// The major cycle after a minor cycle: the residual of the model updated
434    /// by `terms`, cleaned within `masks`.
435    fn refresh(
436        &mut self,
437        completion: MajorCycleCompletion,
438        terms: Vec<ModelDeltaTerm>,
439        masks: &ImageDomainReconstructionMasks,
440        last: bool,
441    ) -> Result<Major, ImagingError> {
442        let (normal_state, continuation) = completion.into_continuation();
443        let attempt = self.next_attempt();
444        let (lifecycle, named) = ModelLifecycle::continue_from(
445            ExecutableModelProblem::from_compiled(self.problem.clone())?,
446            attempt,
447            1,
448            continuation,
449            self.model_storage()?,
450        )?;
451        let delta = (!terms.is_empty())
452            .then(|| lifecycle.compile_delta(&named, terms))
453            .transpose()?;
454        let preparation = MajorCyclePreparation::prepare(&lifecycle, named, delta)?;
455        let (residency, _pass) = self.admit_pass(ModeSet::DATA, true)?;
456        let state = PassNormalState::refresh(
457            self.problem,
458            normal_state,
459            preparation.final_model_generation(),
460            self.normal_storage(residency)?,
461        )?;
462        let masks = ReconstructionMaskSet::Domains(masks.clone());
463        self.reconcile(lifecycle, state, preparation, Some(&masks), last, residency)
464    }

466    /// Run one pass into `state` and reconcile it with the prepared model.
467    /// The initial pass grids the start model's residual only when there is
468    /// a start model; every refresh grids the residual.
469    fn reconcile(
470        &mut self,
471        mut lifecycle: ModelLifecycle,
472        mut state: PassNormalState,
473        preparation: MajorCyclePreparation,
474        masks: Option<&ReconstructionMaskSet>,
475        last: bool,
476        residency: Residency,
477    ) -> Result<Major, ImagingError> {
478        let initial = masks.is_none();
479        let modes = if initial {
480            self.initial_modes()
481        } else {
482            ModeSet::DATA
483        };
484        let with_model = !initial || start_model(self.problem);
485        let model = with_model.then(|| preparation.final_model());
486        let transform = self.problem.visibility_transform();
487        let mut writer = self
488            .visibility_write
489            .as_ref()
490            .filter(|_| last)
491            .map(|target| VisibilityWriter::begin(target, transform))
492            .transpose()
493            .map_err(|error| ImagingError::Pass(PassError::VisibilityWrite(error)))?;
494        if let Residency::Waves { planes_per_wave } = residency {
495            self.planes_per_wave = Some(
496                self.planes_per_wave
497                    .map_or(planes_per_wave, |planes| planes.min(planes_per_wave)),
498            );
499        }
500        let started = Instant::now();
501        let summary = self.pass(
502            modes,
503            model,
504            residency,
505            &mut state,
506            initial,
507            writer.as_mut(),
508        )?;
509        let pass_seconds = started.elapsed().as_secs_f64();
510        let final_model = preparation.final_model_generation();
511        let visibility = writer
512            .map(|writer| writer.complete(final_model))
513            .transpose()
514            .map_err(|error| ImagingError::Pass(PassError::VisibilityWrite(error)))?
515            .map(|samples| {
516                VisibilityProductCompletion::new(
517                    self.problem.problem_id(),
518                    final_model,
519                    self.weighting_id,
520                    samples,
521                )
522            });
523        let normal = state.finish(summary.samples, summary.blocks)?;
524        let mut owner = MajorCycleOwner::from_complete_data(normal, preparation)?;
525        if let Some(masks) = masks {
526            owner = owner.bind_reconstruction_masks(masks)?;
527        }
528        let completion = owner.reconcile(&mut lifecycle)?;
529        tracing::info!(
530            "imaging major cycle: {} samples, pass {pass_seconds:.2} s, total {:.2} s",
531            summary.samples,
532            started.elapsed().as_secs_f64(),
533        );
534        Ok(Major {
535            completion,
536            visibility,
537        })
538    }

540    fn next_attempt(&mut self) -> casa_imaging_model::ModelExecutionAttemptId {
541        self.attempts += 1;
542        let mut identity = [0_u8; 32];
543        identity[0] = 1;
544        identity[24..].copy_from_slice(&self.attempts.to_be_bytes());
545        casa_imaging_model::ModelExecutionAttemptId::new(
546            casa_imaging_model::LogicalIdentity::from_bytes(identity),
547        )
548    }

557    /// Normal storage whose paged window holds the planes of one wave of
558    /// `residency`.
559    fn normal_storage(&self, residency: Residency) -> Result<NormalStoragePlan, ImagingError> {
560        let planes = self.domains[0].operator.basis().planes() as usize;
561        Ok(match &self.cube {
562            Some(cube) => cube.normal_storage(match residency {
563                Residency::All => planes,
564                Residency::Waves { planes_per_wave } => planes_per_wave as usize,
565            })?,
566            None => NormalStoragePlan::resident(planes)?,
567        })

580    /// The waves of a pass accumulating `modes`, with a model when
581    /// `with_model`, that fit the free memory, and the reservation of one
582    /// wave: planned once per major cycle for both the pass and its normal
583    /// storage, and held while the pass runs.
584    fn admit_pass(
585        &self,
586        modes: ModeSet,
587        with_model: bool,
588    ) -> Result<(Residency, Reservation), ImagingError> {
589        let domains = pass_domains(&self.domains, self.team.workers());
590        let demand = WaveDemand {
591            domains: &domains,
592            modes,
593            with_model,
594            native_spacing_hz: self.native_spacing_hz,
595            workers: self.team.workers(),
596            backend: self.backend,
597        };
598        let residency = Residency::plan(&demand, free_memory(&self.host, &self.policy));
599        let planes = match residency {
600            Residency::All => demand.planes(),
601            Residency::Waves { planes_per_wave } => planes_per_wave,
602        };
603        let reservation = admit(
604            &self.host,
605            &self.policy,
606            &Demand {
607                phase: "major-cycle pass",
608                memory: demand.bytes(planes),
609            },
610        )?;
611        Ok((residency, reservation))
612    }

614    /// One pass over every image domain, appending its images to `state`.
615    fn pass(
616        &mut self,
617        modes: ModeSet,
618        model: Option<&ModelGeneration>,
619        residency: Residency,
620        state: &mut PassNormalState,
621        initial: bool,
622        mut writer: Option<&mut VisibilityWriter<'_>>,
623    ) -> Result<PassSummary, ImagingError> {
624        let domains = pass_domains(&self.domains, self.team.workers());
625        let prepare = |domain: usize, planes: PlaneRange| {
626            let generation = model.expect("the closure is installed only with a model");
627            prepare_model(&self.domains[domain].operator, generation, domain, planes)
628                .map_err(|error| PassError::Model(Box::new(error)))
629        };
630        let pass = MajorCyclePass {
631            domains: &domains,
632            weighting: &self.weighting,
633            modes,
634            model: model.map(|_| &prepare as &ModelPreparation<'_>),
635            residency,
636            native_spacing_hz: self.native_spacing_hz,
637            backend: self.backend,
638        };
639        let predictions = writer.as_deref().map(VisibilityWriter::needs_predictions);
640        let mut write = |block: &_, predictions: &[_]| match writer.as_deref_mut() {
641            Some(writer) => writer.write(block, predictions),
642            None => Ok(()),
643        };
644        let mut sink = predictions.map(|predictions| VisibilitySink {
645            predictions,
646            write: &mut write,
647        });
648        Ok(run_major_cycle(
649            &pass,
650            &mut self.source,
651            self.team,
652            &self.cancel,
653            &mut |domain, images| {
654                let basis = self.domains[domain].operator.basis();
655                state
656                    .append(pass_images(domain, &images, initial, basis))
657                    .map_err(|error| PassError::Images(Box::new(error)))
658            },
659            sink.as_mut(),
660        )?)
661    }
662}

664/// Every image domain's operator, resampler and partition: planes split
665/// among `workers` for a channel-local basis, grid strips otherwise.
666fn pass_domains(domains: &[DomainOperator], workers: usize) -> Vec<PassDomain<'_>> {
667    domains
668        .iter()
669        .map(|domain| PassDomain {
670            operator: &domain.operator,
671            resampler: &domain.resampler,
672            partition: match domain.operator.basis() {
673                Basis::ChannelLocal { .. } => Partition::Planes { owners: workers },
674                Basis::Constant | Basis::Taylor { .. } => {
675                    Partition::regions(&domain.operator, workers)
676                }
677            },
678        })
679        .collect()
680}

706/// CASA's imaging weights for the run: the input weights (natural) or the
707/// density weights of one traversal on the main domain's cells, which every
708/// domain shares (`SynthesisImagerVi2::weight`).
709fn weighting(
710    problem: &CompiledProblem,
711    main: &DomainOperator,
712    source: &mut MeasurementSetSource<'_>,
713    team: &WorkerTeam,
714    cancel: &Cancel,
715) -> Result<WeightingGeneration, ImagingError> {
716    let contract = problem.weighting();
717    let taper = contract.uv_taper();
718    let (robust, bandwidth) = match contract.scheme() {
719        WeightingScheme::Natural => return Ok(WeightingGeneration::Natural { taper }),
720        WeightingScheme::Uniform => (None, None),
721        WeightingScheme::Briggs { robust } => (Some(robust), None),
722        WeightingScheme::BriggsBandwidthTaper { robust } => {
723            let spectral = problem.geometry().spectral();
724            let last = spectral.output_channels().saturating_sub(1);
725            let (Some(first_hz), Some(last_hz)) = (
726                spectral.channel_centre_hz(0),
727                spectral.channel_centre_hz(last),
728            ) else {
729                return Err(ImagingError::Unsupported {
730                    reason: "the image spectral axis has no channels",
731                });
732            };
733            (
734                Some(robust),
735                Some(BandwidthTaper::from_frequency_range(first_hz, last_hz)?),
736            )
737        }
738    };
739    let shape = density_shape(problem, &problem.geometry().domains()[0])?;
740    let grid = run_density_pass(&main.operator, &main.resampler, shape, source, team, cancel)?;
741    Ok(WeightingGeneration::density(
742        grid, robust, bandwidth, taper,
743    )?)
744}

746/// The output-frame frequency envelope of a plane range of a channel-local
747/// basis: from the low edge of its first channel to the high edge of its
748/// last.
749fn plane_bounds(problem: &CompiledProblem) -> Option<PlaneBounds> {
750    if !matches!(
751        problem.reconstruction().basis(),
752        casa_imaging_model::ReconstructionBasis::ChannelLocal { .. }
753    ) {
754        return None;
755    }
756    let SpectralWcs::Linear {
757        reference_pixel,
758        reference_frequency_hz,
759        increment_hz,
760        ..
761    } = *problem.geometry().spectral().wcs()
762    else {
763        return None;
764    };
765    let centre = move |channel: u32| {
766        reference_frequency_hz + (f64::from(channel) - reference_pixel) * increment_hz
767    };
768    Some(Box::new(move |planes: PlaneRange| {
769        let half = increment_hz.abs() / 2.0;
770        let [first, last] = [centre(planes.start), centre(planes.end - 1)];
771        [first.min(last) - half, first.max(last) + half]
772    }))
773}

775/// The paged cube state of a channel-local run, with a cache as large as
776/// the whole cube when a quarter of the free memory allows; the state holds
777/// the cache's reservation for as long as its storage lives.
778fn cube_state(
779    problem: &CompiledProblem,
780    directory: &Path,
781    workers: usize,
782    host: &HostResources,
783    policy: &ResourcePolicy,
784) -> Result<CubeState, ImagingError> {
785    let target = problem.model_lifecycle().target();
786    let domains = target
787        .domains()
788        .iter()
789        .map(|domain| domain.pixels())
790        .collect::<Vec<_>>();
791    let planes = target.coefficients() * target.polarizations();
792    let (minimum, full) = CubeState::cache_limits(directory, &domains, planes, workers)?;
793    let quarter = usize::try_from(free_memory(host, policy) / 4).unwrap_or(usize::MAX);
794    let cache = full.min(minimum.max(quarter));
795    let reservation = admit(
796        host,
797        policy,
798        &Demand {
799            phase: "paged cube state",
800            memory: cache as u64,
801        },
802    )?;
803    Ok(CubeState::new(directory, &domains, planes, reservation)?)
804}
