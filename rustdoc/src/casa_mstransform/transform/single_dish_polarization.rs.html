23/// How to combine a selected pair of single-dish parallel hands.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25pub enum SingleDishPolarizationAverage {
26    /// Average the available samples with equal coefficients.
27    Stokes,
28    /// Average the available samples using spectral or scalar weights.
29    Geometric,
30}
31
32impl SingleDishPolarizationAverage {
33    fn weight(self, a: f32, b: f32) -> f32 {
34        match self {
35            Self::Stokes => (4. / (f64::from(a).recip() + f64::from(b).recip())) as f32,
36            Self::Geometric => a + b,
37        }
38    }
39}

41struct Projection<'a> {
42    hands: [usize; 2],
43    flags: ndarray::ArrayView2<'a, bool>,
44    spectral_weight: Option<ndarray::ArrayView2<'a, f32>>,
45    weight: [f32; 2],
46    mode: SingleDishPolarizationAverage,
47    flag_row: bool,
48}
49
50impl Projection<'_> {
51    fn weight_for(&self, hand: usize, channel: usize) -> f32 {
52        if self.mode == SingleDishPolarizationAverage::Stokes {
53            1.
54        } else if let Some(weight) = self.spectral_weight {
55            weight[[self.hands[hand], channel]]
56        } else {
57            self.weight[hand]
58        }
59    }
60
61    fn data<T>(&self, input: ArrayD<T>) -> Result<ArrayD<T>, TableError>
62    where
63        T: Copy + Default + Add<Output = T> + Mul<f32, Output = T> + Div<f32, Output = T>,
64    {
65        let input = input
66            .into_dimensionality::<Ix2>()
67            .map_err(|_| invalid("visibility"))?;
68        if input.shape() != self.flags.shape() {
69            return Err(invalid("visibility"));
70        }
71        let channels = self.flags.ncols();
72        let result = Array2::from_shape_fn((1, channels), |(_, channel)| {
73            let mut summed = T::default();
74            let mut weight = 0.;
75            for hand in 0..2 {
76                let coordinate = [self.hands[hand], channel];
77                if !self.flag_row && !self.flags[coordinate] {
78                    let w = self.weight_for(hand, channel);
79                    summed = summed + input[coordinate] * w;
80                    weight += w;
81                }
82            }
83            if weight > 0. { summed / weight } else { summed }
84        });
85        Ok(result.into_dyn())
86    }
87
88    fn flags(&self) -> ArrayD<bool> {
89        Array2::from_shape_fn((1, self.flags.ncols()), |(_, channel)| {
90            self.flag_row
91                || (self.flags[[self.hands[0], channel]] && self.flags[[self.hands[1], channel]])
92        })
93        .into_dyn()
94    }
95
96    fn projected_scalar_weight(&self) -> f32 {
97        self.mode.weight(self.weight[0], self.weight[1])
98    }
99}
100
101fn invalid(name: &str) -> TableError {
102    TableError::Schema(format!(
103        "Selected single-dish {name} has inconsistent axes."
104    ))
105}

107pub(super) struct PolarizationProjection {
108    hands: BTreeMap<i32, [usize; 2]>,
109    mode: SingleDishPolarizationAverage,
110}
111
112impl PolarizationProjection {
113    pub(super) fn resolve(
114        input: &MeasurementSet,
115        selection: Option<&CorrelationSelection>,
116        ddids: &[i32],
117        mode: SingleDishPolarizationAverage,
118    ) -> MsResult<Option<Self>> {
119        let mut hands = parallel_hands(input, selection, ddids)?;
120        for (&description, indices) in &mut hands {
121            if let Some(axes) = selection.and_then(|selection| selection.axes(description)) {
122                *indices = indices.map(|index| axes[index]);
123            }
124        }
125        Ok((!hands.is_empty()).then_some(Self { hands, mode }))
126    }
127
128    pub(super) fn contains(&self, description: i32) -> bool {
129        self.hands.contains_key(&description)
130    }
131
132    pub(super) fn write_metadata(
133        &self,
134        request: &MsTransformRequest,
135        descriptions: &BTreeMap<i32, i32>,
136    ) -> Result<(), MsTransformError> {
137        let polarization = request.output_ms.join(SubtableId::Polarization.name());
138        let description = request.output_ms.join(SubtableId::DataDescription.name());
139        let error = |path: &std::path::Path, reason: String| MsTransformError::SpectralMetadata {
140            path: path.display().to_string(),
141            reason,
142        };
143        let stokes = ensure_stokes_i_polarization(&polarization)?;
144        let mut dd = Table::open(TableOptions::new(&description))
145            .map_err(|source| error(&description, source.to_string()))?;
146        let mut metadata = MsDataDescriptionMut::new(&mut dd);
147        for old in self.hands.keys() {
148            let row = descriptions.get(old).ok_or_else(|| {
149                error(
150                    &description,
151                    format!("No output row for selected DATA_DESC_ID {old}."),
152                )
153            })?;
154            metadata
155                .set_i32(*row as usize, "POLARIZATION_ID", stokes as i32)
156                .map_err(|source| error(&description, source.to_string()))?;
157        }
158        dd.save(TableOptions::new(&description))
159            .map_err(|source| error(&description, source.to_string()))?;
160        Ok(())
161    }
162
163    pub(super) fn apply(
164        &self,
165        block: &mut MainArrayBlock,
166        (rows, row_flags): (&[i32], &[Option<ScalarValue>]),
167        channel_sources: (
168            &BTreeMap<i32, i32>,
169            &BTreeMap<i32, TransformChannelSelection>,
170        ),
171        spectra: &[&str],
172    ) -> Result<(), TableError> {
173        let (windows, channels) = channel_sources;
174        let weight_index = spectra.iter().position(|&name| name == "WEIGHT_SPECTRUM");
175        let sigma_index = spectra.iter().position(|&name| name == "SIGMA_SPECTRUM");
176        for (index, description) in rows.iter().enumerate() {
177            let Some(&hands) = self.hands.get(description) else {
178                continue;
179            };
180            let Some(ArrayValue::Bool(flags)) = &block.flags[index] else {
181                return Err(invalid("flags"));
182            };
183            let flags = flags
184                .view()
185                .into_dimensionality::<Ix2>()
186                .map_err(|_| invalid("flags"))?;
187            if hands.into_iter().any(|axis| axis >= flags.nrows()) {
188                return Err(invalid("parallel hands"));
189            }
190            let Some(ArrayValue::Float32(weight)) = &block.weights[index] else {
191                return Err(invalid("scalar weights"));
192            };
193            let weight = weight
194                .as_slice()
195                .filter(|weight| weight.len() == flags.nrows())
196                .ok_or_else(|| invalid("scalar weights"))?;
197            let spectral = weight_index.and_then(|column| block.spectra[column][index].as_ref());
198            let spectral_weight = match spectral {
199                Some(ArrayValue::Float32(values)) if values.is_empty() => None,
200                None => None,
201                Some(ArrayValue::Float32(values)) if values.shape() == flags.shape() => Some(
202                    values
203                        .view()
204                        .into_dimensionality::<Ix2>()
205                        .map_err(|_| invalid("weight spectra"))?,
206                ),
207                _ => return Err(invalid("weight spectra")),
208            };
209            let projection = Projection {
210                hands,
211                flags,
212                spectral_weight,
213                weight: hands.map(|i| weight[i]),
214                mode: self.mode,
215                flag_row: matches!(row_flags[index], Some(ScalarValue::Bool(true))),
216            };
217            for column in &mut block.visibility {
218                column[index] = column[index]
219                    .take()
220                    .map(|value| match value {
221                        ArrayValue::Float32(data) => projection.data(data).map(ArrayValue::Float32),
222                        ArrayValue::Complex32(data) => {
223                            projection.data(data).map(ArrayValue::Complex32)
224                        }
225                        _ => Err(invalid("visibility")),
226                    })
227                    .transpose()?;
228            }
229            let flags = projection.flags();
230            let projected = projection.spectral_weight.map(|values| {
231                Array2::from_shape_fn((1, values.ncols()), |(_, channel)| {
232                    self.mode
233                        .weight(values[[hands[0], channel]], values[[hands[1], channel]])
234                })
235                .into_dyn()
236            });
237            let mut weight = projection.projected_scalar_weight();
238            let mut sigma = sigma_from_weight(weight);
239            if let Some(projected) = &projected {
240                let selected = &channels[&windows[description]].channels;
241                (weight, sigma) = spectrum_statistics(projected, &flags, selected);
242                if let Some(column) = sigma_index {
243                    block.spectra[column][index] =
244                        Some(ArrayValue::Float32(projected.mapv(sigma_from_weight)));
245                }
246            } else if let Some(column) = sigma_index {
247                block.spectra[column][index] = None;
248            }
249            if let Some(column) = weight_index {
250                block.spectra[column][index] = projected.map(ArrayValue::Float32);
251            }
252            block.weights[index] = Some(ArrayValue::Float32(
253                Array1::from_vec(vec![weight]).into_dyn(),
254            ));
255            block.sigmas[index] = Some(ArrayValue::Float32(
256                Array1::from_vec(vec![sigma]).into_dyn(),
257            ));
258            block.flags[index] = Some(ArrayValue::Bool(flags));
259            if let Some(ArrayValue::Bool(values)) = &mut block.categories[index]
260                && !values.is_empty()
261            {
262                let category = values
263                    .view()
264                    .into_dimensionality::<Ix3>()
265                    .map_err(|_| invalid("flag categories"))?;
266                if hands.into_iter().any(|axis| axis >= category.shape()[0]) {
267                    return Err(invalid("flag categories"));
268                }
269                let shape = (1, category.shape()[1], category.shape()[2]);
270                *values = Array3::from_shape_fn(shape, |(_, channel, category_id)| {
271                    category[[hands[0], channel, category_id]]
272                        && category[[hands[1], channel, category_id]]
273                })
274                .into_dyn();
275            }
276        }
277        Ok(())
278    }
279}

281/// Return the first Stokes I row in a private staged POLARIZATION table,
282/// appending one if needed. Preserve existing rows and widen fixed shapes.
283///
284/// # Errors
285/// Returns a metadata error if the staged table cannot be read or replaced.
286pub fn ensure_stokes_i_polarization(path: &Path) -> Result<usize, MsTransformError> {
287    const COLUMNS: [&str; 2] = ["CORR_TYPE", "CORR_PRODUCT"];
288    let error = |reason: String| MsTransformError::SpectralMetadata {
289        path: path.display().to_string(),
290        reason,
291    };
292    let mut table =
293        Table::open(TableOptions::new(path)).map_err(|source| error(source.to_string()))?;
294    let rows = MsPolarization::new(&table);
295    let stokes = Correlation::I.casacore_code();
296    for row in 0..rows.row_count() {
297        if !rows
298            .flag_row(row)
299            .map_err(|source| error(source.to_string()))?
300            && rows
301                .corr_type(row)
302                .map_err(|source| error(source.to_string()))?
303                == [stokes]
304        {
305            return Ok(row);
306        }
307    }
308    let row = table.row_count();
309    let fixed = COLUMNS.iter().any(|name| {
310        table
311            .schema()
312            .and_then(|schema| schema.column(name))
313            .is_some_and(|column| {
314                matches!(
315                    column.column_type(),
316                    ColumnType::Array(ArrayShapeContract::Fixed { .. })
317                )
318            })
319    });
320    let temporary = fixed
321        .then(|| tempfile::tempdir_in(path.parent().unwrap_or_else(|| Path::new("."))))
322        .transpose()
323        .map_err(|source| error(source.to_string()))?;
324    let widened = temporary
325        .as_ref()
326        .map(|directory| directory.path().join("polarization"));
327    let write_to = widened.as_deref().unwrap_or(path);
328    if fixed {
329        let indices = (0..row).collect::<Vec<_>>();
330        rewrite_table_rows_with_variable_arrays(path, write_to, &indices, &COLUMNS, |_, row| {
331            Ok(row)
332        })?;
333        table =
334            Table::open(TableOptions::new(write_to)).map_err(|source| error(source.to_string()))?;
335    }
336    table
337        .add_row(casa_ms::subtables::polarization::stokes_i_row())
338        .map_err(|source| error(source.to_string()))?;
339    table
340        .save(TableOptions::new(write_to))
341        .map_err(|source| error(source.to_string()))?;
342    drop(table);
343    if let Some(widened) = widened {
344        fs::remove_dir_all(path)
345            .and_then(|()| fs::rename(widened, path))
346            .map_err(|source| error(source.to_string()))?;
347    }
348    // A copied casacore lock still describes the table before it gained a row.
349    if let Err(source) = fs::remove_file(path.join("table.lock"))
350        && source.kind() != ErrorKind::NotFound
351    {
352        return Err(error(source.to_string()));
353    }
354    Ok(row)
355}
356
357fn sigma_from_weight(weight: f32) -> f32 {
358    weight.sqrt().recip()
359}
360
361fn spectrum_statistics(
362    weights: &ArrayD<f32>,
363    flags: &ArrayD<bool>,
364    channels: &[usize],
365) -> (f32, f32) {
366    let all_flagged = channels.iter().all(|&channel| flags[[0, channel]]);
367    let mut count = 0;
368    let mut weight = 0.;
369    let mut sigma = 0.;
370    for &channel in channels {
371        if all_flagged || !flags[[0, channel]] {
372            let value = weights[[0, channel]];
373            count += 1;
374            weight += value;
375            sigma += sigma_from_weight(value);
376        }
377    }
378    if count == 0 {
379        (0., -1.)
380    } else {
381        (weight / count as f32, sigma / count as f32)
382    }
