3//! A cube's per-time frequency selection: the bands CASA's visibility
4//! iterator selects each window's channels by at each time
5//! (`vi::FrequencySelectionUsingFrame`), and the channels they select.
6//!
7//! `SynthesisImagerVi2::selectData` states each channel range `spw` names
8//! as a band in the selection's frame, tclean's `outframe`: the range's
9//! lowest and highest channel centres converted at every epoch of the
10//! window's samples ([`selection_band`]). A selection frame of `REST` or
11//! `Undefined` (`ignoreframe`, as tclean sets it for `cubedata` and
12//! `cubesource`) takes the range's centres as they are, in the data's
13//! frame, so each range selects its channels at every time; a range with a
14//! stride is read whole there too. A cube is gridded by
15//! `CubeMajorCycleAlgorithm`'s own imager, which selects the same way and,
16//! where `SynthesisImagerVi2::tuneSelectData` tunes the selection
17//! ([`super::spectral`]), narrows each band that the image's LSRK band
18//! (pixel −1.5 to `nchan + 0.5`) overlaps to their intersection, adding
19//! the image's band to a window none of whose bands it overlaps
20//! (`SynthesisImagerVi2::andFreqSelection`). At each time the iterator
21//! converts the bands to the window's frame, towards the time's field, and
22//! reads the channels whose extent meets one
23//! (`VisibilityIteratorImpl2::makeChannelSelectorF`), so where the Doppler
24//! shift across an observation is a sizeable part of a channel, the
25//! channels a time reads move with it, and a channel range with a stride
26//! is read whole.
27//!
28//! Compile ([`CubeBands::epoch_selection`]) and the gridding passes'
29//! packer test a channel by the same predicate,
30//! [`FrequencyBands::meets`]; they differ only in where the frame
31//! conversion is evaluated: compile at each surveyed epoch of the window,
32//! towards its field ([`epoch_frame`]), the packer at each row's time and
33//! field through the run's geometry. A channel a time's bands miss is read
34//! as flagged: no sample of it, or interpolated from it, is imaged. A
35//! density over the whole selection selects by the same bands untuned, as
36//! CASA forms it before it tunes the selection; a density per plane counts
37//! every selected channel. A final pass on CASA's tuned selection leaves its
38//! `MODEL_DATA` as it is; `predictModel` writes it (`observation_source`'s
39//! write-back says which).
40//!
41//! casa-rs departs from CASA in two ways, both recorded in
42//! `docs/CASA (C++) bugs.md`: a tuned band is narrowed by the selection's
43//! bands in LSRK, where CASA compares numbers in `outframe` with numbers in
44//! LSRK; and the bands of one window's channel ranges do not narrow each
45//! other where a Doppler shift makes them overlap, as CASA's do.
46

71/// The frame a cube's per-time frequency selection is stated in.
72#[derive(Clone, Copy, Debug, PartialEq, Eq)]
73pub(super) enum SelectionFrame {
74    /// Each range's channel centres converted to this frame at every
75    /// epoch, and narrowed by the image's band where it is given.
76    Converted(FrequencyRef),
77    /// The data's frame, each range's channel centres as they are (CASA's
78    /// `ignoreframe` for a selection frame of `REST` or `Undefined`).
79    Data,
80}

82/// The bands a cube's gridding selects one store's windows' channels by at
83/// each time ([`cube_bands`]): a `cube`'s in `outframe`,
84/// or tuned to the image's band `image_band` where CASA tunes the
85/// selection; in the data's frame where tclean's selection frame is
86/// `Undefined` (`cubedata`) or `REST` (`cubesource`, or a `cube` whose
87/// `outframe` is `REST`), which select each named range's channels at
88/// every time, a range with a stride whole. None for a `fitspw` fit,
89/// which reads its channels as named.
90pub(super) fn cube_read_bands(
91    request: &ImagingRequest,
92    (catalogue, engine): (&Catalogue, &MsCalEngine),
93    survey: &Survey,
94    image_band: Option<[f64; 2]>,
95) -> Result<Option<CubeBands>, PrepareError> {
96    if request.fitspw.is_some() {
97        return Ok(None);
98    }
99    let frame = match request.specmode {
100        SpecMode::Cube if image_band.is_some() => SelectionFrame::Converted(FrequencyRef::LSRK),
101        SpecMode::Cube if request.outframe != FrequencyRef::REST => {
102            SelectionFrame::Converted(request.outframe)
103        }
104        SpecMode::Mfs | SpecMode::Cube | SpecMode::Cubedata | SpecMode::Cubesource => {
105            SelectionFrame::Data
106        }
107    };
108    cube_bands((catalogue, engine), survey, frame, image_band).map(Some)
109}

111/// The image's band in LSRK as `SynthesisImager::tuneSelectData` states
112/// it: its endpoints at pixels −1.5 and `channels + 0.5` in frame
113/// `output`, converted to LSRK through the image's conversion layer
114/// at its anchor (`setSpectralConversion("LSRK")`).
115pub(super) fn image_band_lsrk(
116    frame: &FrameContext<'_>,
117    setup: &CubeSpectralSetup,
118) -> Result<[f64; 2], PrepareError> {
119    let channels = setup.output_channel_frequencies_hz.len();
120    let (first_hz, increment_hz) = first_and_increment(setup, channels)?;
121    let ends = image_band_frequencies(
122        first_hz,
123        increment_hz,
124        channels,
125        setup.output_channel_table_hz.as_deref(),
126    );
127    let output = setup.output_freq_ref;
128    let factor = if output == FrequencyRef::LSRK {
129        1.0
130    } else {
131        let reference = frame.direction.reference_direction();
132        let direction = MDirection::from_angles(
133            reference.longitude_rad(),
134            reference.latitude_rad(),
135            reference.frame().direction_ref(),
136        );
137        let anchor = frame
138            .engine
139            .spectral_frame_observatory_direction(frame.anchor_time_mjd_seconds, direction)?;
140        PreparedFrequencyConversion::new(output, FrequencyRef::LSRK, Some(&anchor), Some(&anchor))?
141            .convert_hz(1.0)
142    };
143    let ends = ends.map(|hz| tuned_selection_hz(hz * factor));
144    Ok([ends[0].min(ends[1]), ends[0].max(ends[1])])
145}

147/// World frequencies at the two pixels that bound tuneSelectData.
148fn image_band_frequencies(
149    first_hz: f64,
150    increment_hz: f64,
151    channels: usize,
152    table_hz: Option<&[f64]>,
153) -> [f64; 2] {
154    // A one-entry CASA table maps every pixel to the same frequency.
155    // The iterator's inclusive channel extents still meet that point.
156    if let Some([frequency_hz]) = table_hz {
157        return [*frequency_hz; 2];
158    }
159    let far_pixel = channels as f64 + 0.5;
160    let far_hz = if let Some(table) = table_hz {
161        let left = channels.min(table.len() - 2);
162        table[left] + (far_pixel - left as f64) * (table[left + 1] - table[left])
163    } else {
164        first_hz + far_pixel * increment_hz
165    };
166    [first_hz - 1.5 * increment_hz, far_hz]
167}

169/// CASA's tuneSelectData sends each LSRK endpoint through std::to_string
170/// and parses the resulting frequency quantity. Its six fractional digits
171/// can put the endpoint exactly on a stored channel's inclusive edge.
172fn tuned_selection_hz(hertz: f64) -> f64 {
173    format!("{hertz:.6}")
174        .parse()
175        .expect("a formatted frequency is a number")
176}

178/// The bands a cube's gridding selects each surveyed window's channels by:
179/// those of the channel ranges `spw` names (a window it names without
180/// channels is one range) in `frame`, narrowed to the image's LSRK band
181/// where `tuned_image_band` gives it (`frame` is then LSRK).
182///
183/// # Errors
184///
185/// A frame conversion that fails, or a frame the image model does not
186/// hold.
187pub(super) fn cube_bands(
188    (catalogue, engine): (&Catalogue, &MsCalEngine),
189    survey: &Survey,
190    frame: SelectionFrame,
191    tuned_image_band: Option<[f64; 2]>,
192) -> Result<CubeBands, PrepareError> {
193    let reference = match frame {
194        SelectionFrame::Converted(reference) => reference,
195        SelectionFrame::Data => survey.source_frequency_reference,
196    };
197    let model_frame = super::spectral::imaging_frequency_frame(reference)?;
198    let mut by_window = BTreeMap::new();
199    for window in &survey.spectral_windows {
200        let whole = [[0, window.frequencies_hz.len().saturating_sub(1)]];
201        let ranges = survey
202            .spw_ranges
203            .get(&window.spw_id)
204            .map_or(&whole[..], Vec::as_slice);
205        let mut bands = Vec::with_capacity(ranges.len() + 1);
206        for &range in ranges {
207            let band = match frame {
208                SelectionFrame::Converted(reference) => {
209                    selection_band((catalogue, engine), (survey, window), range, reference)?
210                }
211                SelectionFrame::Data => Some(centre_span(window, range)?),
212            };
213            bands.extend(band);
214        }
215        if let Some(image) = tuned_image_band {
216            and_band(&mut bands, image);
217        }
218        if !bands.is_empty() {
219            by_window.insert(window.spw_id, FrequencyBands::new(model_frame, bands)?);
220        }
221    }
222    Ok(CubeBands { bands: by_window })
223}

225/// The lowest and highest centre of `window`'s channels `[first, last]`, in
226/// its own frame.
227fn centre_span(
228    window: &SourceSpectralWindow,
229    [first, last]: [usize; 2],
230) -> Result<[f64; 2], PrepareError> {
231    let centres =
232        window
233            .frequencies_hz
234            .get(first..=last)
235            .ok_or(PrepareError::SpectralWindowCoordinates {
236                spw_id: window.spw_id,
237            })?;
238    Ok(centres
239        .iter()
240        .fold([f64::INFINITY, f64::NEG_INFINITY], |[lower, upper], hz| {
241            [lower.min(*hz), upper.max(*hz)]
242        }))
243}

245/// `SynthesisImagerVi2::andFreqSelection` of `[lower, upper]` on one
246/// window's bands: each band it overlaps, ends excluded, narrows to their
247/// intersection; with none, it is added.
248fn and_band(bands: &mut Vec<[f64; 2]>, [lower, upper]: [f64; 2]) {
249    let mut overlapped = false;
250    for band in bands.iter_mut() {
251        if lower < band[1] && upper > band[0] {
252            overlapped = true;
253            band[0] = band[0].max(lower);
254            band[1] = band[1].min(upper);
255        }
256    }
257    if !overlapped {
258        bands.push([lower, upper]);
259    }
260}

262/// What a window's bands select over the epochs of its samples: for each
263/// channel, at how many epochs, of how many.
264pub(super) struct EpochSelection {
265    counts: Vec<u32>,
266    epochs: u32,
267}
268
269impl EpochSelection {
270    /// Whether some epoch selects `channel`.
271    pub(super) fn ever(&self, channel: usize) -> bool {
272        self.counts.get(channel).is_some_and(|count| *count > 0)
273    }
274
275    /// Whether every epoch selects `channel`.
276    pub(super) fn always(&self, channel: usize) -> bool {
277        self.counts
278            .get(channel)
279            .is_some_and(|count| *count == self.epochs)
280    }
281}

283/// Keep the selected channels whose cubic stencils can reach the
284/// source-frame cube band. The coarse output centres alone can miss a
285/// native pair reached by a fine sample between a centre and its edge.
286/// Two neighbours on either side also cover the four-point stencil when
287/// the other side reaches the end of the selected channels.
288pub(super) fn cubic_read_support(
289    retained: &mut Vec<usize>,
290    frequencies: &[f64],
291    [lower, upper]: [f64; 2],
292    epochs: Option<&EpochSelection>,
293    explicit: Option<&[usize]>,
294) -> Option<NativeSpacingSelection> {
295    let selected = |channel| match epochs {
296        Some(epochs) => epochs.ever(channel),
297        None => explicit.is_none_or(|indices| indices.contains(&channel)),
298    };
299    let channels = 0..frequencies.len();
300    let mut previous: Option<usize> = None;
301    let mut support: Option<(usize, usize)> = None;
302    for right in channels.clone().filter(|&index| selected(index)) {
303        if let Some(left) = previous {
304            let [a, b] = [frequencies[left], frequencies[right]];
305            if a.min(b) <= upper && a.max(b) >= lower {
306                support = Some((support.map_or(left, |(first, _)| first), right));
307            }
308        }
309        previous = Some(right);
310    }
311    let (first, last) = support?;
312    let first = (0..first)
313        .rev()
314        .filter(|&index| selected(index))
315        .take(2)
316        .last()
317        .unwrap_or(first);
318    let last = (last + 1..channels.end)
319        .filter(|&index| selected(index))
320        .take(2)
321        .last()
322        .unwrap_or(last);
323    if let Some((&retained_first, &retained_last)) = retained.first().zip(retained.last()) {
324        retained.extend(
325            (first..retained_first)
326                .chain(retained_last + 1..=last)
327                .filter(|&index| selected(index)),
328        );
329        retained.sort_unstable();
330    } else {
331        retained.extend((first..=last).filter(|&index| selected(index)));
332    }
333    // Cubesource selects in the stored frame, so the leading pair is
334    // the same at every epoch (its converted centres still move).
335    let mut physical = channels.filter(|&index| selected(index));
336    let pair = [physical.next()?, physical.next()?];
337    (!retained.starts_with(&pair)).then(|| {
338        NativeSpacingSelection::Pair(
339            pair.map(|index| u32::try_from(index).expect("a CHAN_FREQ index fits u32")),
340        )
341    })
342}

344/// Match the physical selection at the edges of cube support. The numeric
345/// support brackets the image on the full catalogue, so when its outer
346/// channel is missing from a selection, retain the nearest selected
347/// partner outside that cut. Moving bands may need more than one choice.
348/// Keep the physical leading pair independently of the payload extent.
349pub(super) fn linear_read_support(
350    channel_count: usize,
351    (retained, (lower, upper)): (&mut Vec<usize>, (usize, usize)),
352    explicit: Option<&[usize]>,
353    by_band: Option<(&EpochSelection, &FrequencyBands)>,
354) -> Option<NativeSpacingSelection> {
355    let ever = |channel: usize| match by_band {
356        Some((epochs, _)) => epochs.ever(channel),
357        None => explicit.is_none_or(|channels| channels.contains(&channel)),
358    };
359    let always = |channel: usize| by_band.is_none_or(|(epochs, _)| epochs.always(channel));
360    let previous_len = retained.len();
361    let first = retained[0];
362    let last = *retained.last().expect("nonempty retained support");
363    if first > lower || !always(first) {
364        extend_support(
365            retained,
366            (0..first).rev().filter(|&index| ever(index)),
367            always,
368        );
369    }
370    if last < upper || !always(last) {
371        extend_support(
372            retained,
373            (last + 1..channel_count).filter(|&index| ever(index)),
374            always,
375        );
376    }
377    if retained.len() != previous_len {
378        retained.sort_unstable();
379    }
380    let mut possible = (0..channel_count).filter(|&channel| ever(channel));
381    let pair = [possible.next()?, possible.next()?];
382    if by_band.is_none_or(|(epochs, _)| pair.iter().all(|&channel| epochs.always(channel))) {
383        return (!retained.starts_with(&pair)).then(|| {
384            NativeSpacingSelection::Pair(
385                pair.map(|channel| u32::try_from(channel).expect("a CHAN_FREQ index fits u32")),
386            )
387        });
388    }
389    let (epochs, bands) = by_band.expect("the pair can move only with bands");
390    let mut always = 0;
391    for channel in pair.into_iter().chain(possible) {
392        if retained.binary_search(&channel).is_err() {
393            return Some(NativeSpacingSelection::Bands(bands.clone()));
394        }
395        always += usize::from(epochs.always(channel));
396        if always == 2 {
397            break;
398        }
399    }
400    None
401}
402
403fn extend_support(
404    retained: &mut Vec<usize>,
405    candidates: impl Iterator<Item = usize>,
406    always: impl Fn(usize) -> bool,
407) {
408    for candidate in candidates {
409        retained.push(candidate);
410        if always(candidate) {
411            break;
412        }
413    }
414}

416impl CubeBands {
417    /// What `window`'s bands select at each epoch of its samples: at each,
418    /// the bands converted to the window's frame towards the epoch's field
419    /// ([`epoch_frame`]), as the visibility iterator converts them, select
420    /// the channels whose extent meets one ([`FrequencyBands::meets`]).
421    /// `None` for a window without bands or epochs.
422    pub(super) fn epoch_selection(
423        &self,
424        (catalogue, engine): (&Catalogue, &MsCalEngine),
425        survey: &Survey,
426        window: &SourceSpectralWindow,
427    ) -> Result<Option<EpochSelection>, PrepareError> {
428        let Some(bands) = self.bands.get(&window.spw_id) else {
429            return Ok(None);
430        };
431        let epochs = survey.window_epochs(window.spw_id);
432        if epochs.is_empty() {
433            return Ok(None);
434        }
435        let (source, frame) = (window.frequency_reference, bands.frame().frequency_ref());
436        let mut counts = vec![0_u32; window.frequencies_hz.len()];
437        for epoch in epochs {
438            let factor = if source == frame || source == FrequencyRef::REST {
439                1.0
440            } else {
441                let at = epoch_frame((catalogue, engine), *epoch)?;
442                PreparedFrequencyConversion::new(frame, source, Some(&at), Some(&at))?
443                    .convert_hz(1.0)
444            };
445            for (count, (centre, width)) in counts
446                .iter_mut()
447                .zip(window.frequencies_hz.iter().zip(&window.channel_widths_hz))
448            {
449                let half = width.abs() / 2.0;
450                if bands.meets(factor, [centre - half, centre + half]) {
451                    *count += 1;
452                }
453            }
454        }
455        Ok(Some(EpochSelection {
456            counts,
457            epochs: u32::try_from(epochs.len()).expect("a window's epochs fit u32"),
458        }))
459    }
460
461    /// The bands of window `spw_id`, for the gridding to select a time's
462    /// channels by; none for a window CASA selects nothing of by frequency.
463    pub(super) fn frequency_bands(&self, spw_id: usize) -> Option<&FrequencyBands> {
464        self.bands.get(&spw_id)
465    }
466}

472    #[test]
473    fn the_tuned_band_evaluates_beyond_the_stored_tabular_centres() {
474        let table = [140.0, 120.0, 107.0, 100.0];

477            [170.0, 61.5]

485    #[test]
486    fn tuning_a_constant_table_selects_the_channels_that_contain_its_frequency() {
487        let ends = image_band_frequencies(120.0, 50.0, 1, Some(&[120.0]));
488        let band = FrequencyBands::new(casa_imaging_model::FrequencyFrame::Lsrk, vec![ends])
489            .expect("a constant frequency is a valid band");
490        assert_eq!(ends, [120.0, 120.0]);
491        assert!(!band.meets(1.0, [90.0, 110.0]));
492        assert!(band.meets(1.0, [110.0, 130.0]));
493        assert!(!band.meets(1.0, [130.0, 150.0]));
494    }

496    #[test]
497    fn a_tuned_band_includes_only_channel_edges_reached_after_casa_rounding() {
498        let edge: f64 = 1_525_000_000.0;
499        let below = edge.next_down();
500        let mut band = FrequencyBands::new(
501            casa_imaging_model::FrequencyFrame::Lsrk,
502            vec![[1.025e9, tuned_selection_hz(below)]],
503        )
504        .expect("ordered band");
505        assert!(band.meets(1.0, [edge, edge + 50.0e6]));
506        band = FrequencyBands::new(
507            casa_imaging_model::FrequencyFrame::Lsrk,
508            vec![[1.025e9, tuned_selection_hz(edge - 1.0e-6)]],
509        )
510        .expect("ordered band");
511        assert!(!band.meets(1.0, [edge, edge + 50.0e6]));
512        assert_eq!(tuned_selection_hz(edge + 2.4e-6), edge + 2.0e-6);
513    }

516    fn a_band_narrows_the_bands_it_overlaps_and_is_added_where_none() {
517        let mut bands = vec![[1.0, 4.0], [6.0, 9.0]];
518        and_band(&mut bands, [3.0, 7.0]);
519        assert_eq!(bands, vec![[3.0, 4.0], [6.0, 7.0]]);
