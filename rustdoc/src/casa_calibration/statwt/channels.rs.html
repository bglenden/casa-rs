2//! Channel bins and the independent channel selection for variance estimates.
3
4use std::collections::BTreeMap;
5use std::ops::Range;
6
7use casa_ms::{MeasurementSet, MsError, MsResult, ResolvedMsSelectionRow};
8
9use super::{StatWtChannelBin, StatWtOptions};
10
11pub(super) struct Channels {
12    pub(super) bins: Vec<Range<usize>>,
13    pub(super) fit: Vec<bool>,
14}
15
16pub(super) fn prepare(
17    ms: &MeasurementSet,
18    selection: &[ResolvedMsSelectionRow],
19    options: &StatWtOptions,
20) -> MsResult<BTreeMap<i32, Channels>> {
21    let fitted = ms.resolve_spw_selector(&options.fit_spw)?;
22    let windows = ms.spectral_window()?;
23    let mut channels = BTreeMap::new();
24    for row in selection {
25        let spw = row.spectral_window_id;
26        let std::collections::btree_map::Entry::Vacant(entry) = channels.entry(spw) else {
27            continue;
28        };
29        let index = usize::try_from(spw)
30            .map_err(|_| MsError::InvalidInput(format!("invalid spectral window {spw}")))?;
31        let count = windows.num_chan(index)? as usize;
32        let bins = match options.channel_bin {
33            StatWtChannelBin::Window => std::iter::once(0..count).collect(),
34            StatWtChannelBin::Channels(width) => (0..count)
35                .step_by(width)
36                .map(|start| start..start.saturating_add(width).min(count))
37                .collect(),
38            StatWtChannelBin::Hertz(width) => {
39                let frequencies = windows.chan_freq(index)?;
40                let mut bins = Vec::new();
41                let mut start = 0;
42                for (channel, &frequency) in frequencies.iter().enumerate() {
43                    if (frequency - frequencies[start]).abs() > width {
44                        bins.push(start..channel);
45                        start = channel;
46                    }
47                }
48                bins.push(start..frequencies.len());
49                bins
50            }
51        };
52        let fit = if let Some(fitted) = fitted.channels_of(index)? {
53            let mut mask = vec![options.exclude_channels; count];
54            for channel in fitted {
55                if let Some(value) = mask.get_mut(channel) {
56                    *value = !options.exclude_channels;
57                }
58            }
59            mask
60        } else {
61            vec![true; count]
62        };
63        entry.insert(Channels { bins, fit });
64    }
65    Ok(channels)
66}
