4//! The command-line and task surface of casa-rs's simulation tasks:
5//! `simobserve` (simulation Phase 5c, #841), with `simanalyze` and
6//! `simalma` to come, on the shared task front end (#931).
7//!
8//! As for the imager, the front end resolves the parameters through the
9//! provider-contracts catalog: command-line flags and a `--json-run` request
10//! become the same resolved values, CASA's parameters with CASA's defaults,
11//! and those values, and nothing else, make the [`SimobserveRequest`] the
12//! task runs. The host adds what CASA's task takes from its environment:
13//! CASA's data directories, where `alma/simmos` holds the antenna
14//! configurations, and the measures data. A run's reply carries its
15//! [`SimobserveRunTaskResult`]; simobserve returns None, as CASA's does.
16

43/// The task's parameters from `resolved`, every simobserve parameter.
44fn parameters_from(
45    resolved: Map<String, Value>,
46) -> Result<(SimobserveTaskParameters, SimobserveParameters), String> {
47    let missing = |name: &str, what: &str| format!("simobserve parameter {name}: {what}");
48    let text = |name: &str| {
49        resolved
50            .get(name)
51            .and_then(Value::as_str)
52            .map(str::to_string)
53            .ok_or_else(|| missing(name, "a string"))
54    };
55    let texts = |name: &str| -> Result<Vec<String>, String> {
56        match resolved.get(name) {
57            Some(Value::String(one)) => Ok(vec![one.clone()]),
58            Some(Value::Array(many)) => many
59                .iter()
60                .map(|item| item.as_str().map(str::to_string))
61                .collect::<Option<_>>()
62                .ok_or_else(|| missing(name, "strings")),
63            _ => Err(missing(name, "a list of strings")),
64        }
65    };
66    // The contract resolves a direction string to a list of one, so one
67    // direction is taken as a string, and several as the pointings.
68    let directions = |name: &str| -> Result<Directions, String> {
69        let mut many = texts(name)?;
70        Ok(match many.len() {
71            0 => Directions::One(String::new()),
72            1 => Directions::One(many.remove(0)),
73            _ => Directions::List(many),
74        })
75    };
76    let flag = |name: &str| {
77        resolved
78            .get(name)
79            .and_then(Value::as_bool)
80            .ok_or_else(|| missing(name, "a boolean"))
81    };
82    let integer = |name: &str| {
83        resolved
84            .get(name)
85            .and_then(Value::as_i64)
86            .ok_or_else(|| missing(name, "an integer"))
87    };
88    let number = |name: &str| {
89        resolved
90            .get(name)
91            .and_then(Value::as_f64)
92            .ok_or_else(|| missing(name, "a number"))
93    };
94    let parameters = SimobserveParameters {
95        project: text("project")?,
96        skymodel: text("skymodel")?,
97        inbright: text("inbright")?,
98        indirection: text("indirection")?,
99        incell: text("incell")?,
100        incenter: text("incenter")?,
101        inwidth: text("inwidth")?,
102        complist: text("complist")?,
103        compwidth: text("compwidth")?,
104        comp_nchan: integer("comp_nchan")?,
105        setpointings: flag("setpointings")?,
106        ptgfile: text("ptgfile")?,
107        integration: text("integration")?,
108        direction: directions("direction")?,
109        mapsize: texts("mapsize")?,
110        maptype: text("maptype")?,
111        pointingspacing: text("pointingspacing")?,
112        caldirection: text("caldirection")?,
113        calflux: text("calflux")?,
114        obsmode: text("obsmode")?,
115        refdate: text("refdate")?,
116        hourangle: text("hourangle")?,
117        totaltime: text("totaltime")?,
118        antennalist: text("antennalist")?,
119        sdantlist: text("sdantlist")?,
120        sdant: integer("sdant")?,
121        outframe: text("outframe")?,
122        thermalnoise: text("thermalnoise")?,
123        user_pwv: number("user_pwv")?,
124        t_ground: number("t_ground")?,
125        t_sky: number("t_sky")?,
126        tau0: number("tau0")?,
127        seed: integer("seed")?,
128        leakage: number("leakage")?,
129        correlator: text("correlator")?,
130        graphics: text("graphics")?,
131        verbose: flag("verbose")?,
132        overwrite: flag("overwrite")?,
133    };
134    Ok((SimobserveTaskParameters(resolved), parameters))
135}

137/// Resolve `parameters`, catalog-named and sparse, over CASA's defaults:
138/// the resolved parameters and the task's.
139pub fn resolve_request(
140    parameters: &SimobserveTaskParameters,
141) -> Result<(SimobserveTaskParameters, SimobserveParameters), String> {
142    let resolved = resolve_plain_values(simobserve_surface()?, workspace()?, &parameters.0)
143        .map_err(|error| format!("resolve simobserve parameters: {error}"))?;
144    parameters_from(resolved)
145}

147/// CASA's data directories: the measures data's candidates, which hold
148/// `alma/simmos` in a CASA data tree.
149fn data_roots() -> Result<Vec<PathBuf>, String> {
150    casa_measures_data::MeasuresRuntime::discover_candidates()
151        .map(|candidates| {
152            candidates
153                .into_iter()
154                .map(|candidate| candidate.path)
155                .collect()
156        })
157        .map_err(|error| format!("CASA's data directories: {error}"))
158}

160/// Run `task`, resolved as `parameters`, with the host's measures data,
161/// CASA data directories and resources.
162pub fn run(
163    parameters: SimobserveTaskParameters,
164    task: SimobserveParameters,
165) -> Result<SimobserveRunTaskResult, String> {
166    let started = Instant::now();
167    let resources = SessionResources::from_controls(&ExecutionControls::default())
168        .map_err(|error| error.to_string())?;
169    let request = SimobserveRequest {
170        parameters: task,
171        data_roots: data_roots()?,
172        options: SimulatorOptions::from_system(resources).map_err(|error| error.to_string())?,
173    };
174    let outcome = simobserve(&request).map_err(|error| error.to_string())?;
175    Ok(SimobserveRunTaskResult {
176        request: parameters,
177        run: SimobserveRunReport {
178            stem: outcome.stem.clone(),
179            directory: outcome.directory.display().to_string(),
180            warnings: outcome.warnings.clone(),
181            elapsed_ns: u64::try_from(started.elapsed().as_nanos()).unwrap_or(u64::MAX),
182        },
183        artifacts: outcome
184            .products
185            .iter()
186            .map(|path| artifact(&outcome.stem, path))
187            .collect(),
188    })
189}

191/// The product at `path`, named by its suffix after `stem`.
192fn artifact(stem: &str, path: &Path) -> SimobserveArtifact {
193    let name = path
194        .file_name()
195        .map(|name| name.to_string_lossy().into_owned())
196        .unwrap_or_default();
197    let kind = name
198        .strip_prefix(stem)
199        .and_then(|suffix| suffix.strip_prefix('.'))
200        .unwrap_or(&name)
201        .to_string();
202    let resource_kind = if kind == "ms" || kind.ends_with(".ms") {
203        RunProductKind::MeasurementSet
204    } else if kind.ends_with("skymodel") || kind.ends_with(".flat") || kind == "newmodel" {
205        RunProductKind::CasaImage
206    } else if kind.ends_with(".txt") {
207        RunProductKind::File
208    } else {
209        RunProductKind::CasaTable
210    };
211    SimobserveArtifact {
212        label: name,
213        kind,
214        path: path.display().to_string(),
215        exists: path.exists(),
216        resource_kind,
217    }
218}

220/// The front end of the `simobserve` binary, which hosts `simobserve`.
221pub fn task_front_end()
222-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
223    TaskFrontEnd::new(
224        "simobserve",
225        simobserve_task_schema_bundle(),
226        &["simobserve"],
227        run_task,
228    )
229    .expect("the simobserve contract carries its surface")
230}
231
232/// Run one resolved simobserve request.
233fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
234    let (parameters, simulation) =
235        parameters_from(task.plain_values()).map_err(TaskFailure::validation)?;
236    let result = run(parameters, simulation).map_err(TaskFailure::execution)?;
237    let text = format!(
238        "Wrote {} products of {} in {}\n",
239        result.artifacts.len(),
240        result.run.stem,
241        result.run.directory
242    );
243    Ok(TaskSuccess::new(casa_task_runtime::CasaValue::None, &result)?.with_text(text))
244}

254    /// A command line as the front end resolves it, read into the task's
255    /// parameters.
256    fn resolve_cli_request(
257        args: &[OsString],
258    ) -> Result<(SimobserveTaskParameters, SimobserveParameters), String> {
259        let bundle = simobserve_surface()?;
260        let line = TaskCommandLine::parse(&bundle, args)?;
261        let session = ParameterRuntime::default()
262            .open_session(line.open_session_request(bundle))
263            .map_err(|error| error.to_string())?;
264        parameters_from(TaskInvocation::new(&session, Path::new(".")).plain_values())
265    }

274    #[test]
275    fn flags_and_json_resolve_alike() {
276        let flags: Vec<OsString> = [
277            "--project",
278            "p",
279            "--antennalist",
280            "alma.out10.cfg",
281            "--direction",
282            "J2000 19h00m00 -23d00m00",
283            "--no-overwrite",
284        ]
285        .into_iter()
286        .map(OsString::from)
287        .collect();
288        let (_, from_flags) = resolve_cli_request(&flags).expect("flags");
289        let json = serde_json::json!({
290            "project": "p",
291            "antennalist": "alma.out10.cfg",
292            "direction": "J2000 19h00m00 -23d00m00",
293            "overwrite": false,
294        });
295        let (_, from_json) = resolve_request(&SimobserveTaskParameters(
296            json.as_object().expect("an object").clone(),
297        ))
298        .expect("JSON");
299        assert_eq!(from_flags, from_json);
300        assert_eq!(
301            from_json.direction,
302            Directions::One("J2000 19h00m00 -23d00m00".to_string())
303        );
304        assert!(!from_json.overwrite);
305        assert_eq!(from_json.ptgfile, "$project.ptg.txt");
306        // A list of one is its direction; several are the pointings.
307        let directions = |value: Value| {
308            let mut json = json.as_object().expect("an object").clone();
309            json.insert("direction".to_string(), value);
310            resolve_request(&SimobserveTaskParameters(json))
311                .expect("JSON")
312                .1
313                .direction
314        };
315        assert_eq!(
316            directions(serde_json::json!(["J2000 19h00m00 -23d00m00"])),
317            from_json.direction
318        );
319        assert_eq!(
320            directions(serde_json::json!([
321                "J2000 19h00m00 -23d00m00",
322                "J2000 19h00m01 -23d00m00"
323            ])),
324            Directions::List(vec![
325                "J2000 19h00m00 -23d00m00".to_string(),
326                "J2000 19h00m01 -23d00m00".to_string()
327            ])
328        );
329        assert_eq!(
330            directions(serde_json::json!([])),
331            Directions::One(String::new())
332        );
333    }
