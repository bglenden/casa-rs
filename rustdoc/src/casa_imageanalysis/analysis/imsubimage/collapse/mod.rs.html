29/// Parameters for a CASA image collapse over specified pixel axes.
30pub struct ImageCollapseOptions<'a> {
31    /// Statistic computed for each output cell.
32    pub function: CollapseFunction,
33    /// Axes to reduce to length one.
34    pub axes: &'a [usize],
35    /// Optional saved or textual region.
36    pub region: Option<SubimageRegion<'a>>,
37    /// Rectangular pixel bounds when supplied instead of a region.
38    pub spatial: Option<&'a str>,
39    /// Spectral channels selected before reducing.
40    pub chans: Option<&'a str>,
41    /// Stokes planes selected before reducing.
42    pub stokes: Option<&'a str>,
43    /// Additional lattice validity expression.
44    pub mask: Option<&'a str>,
45    /// Permit stretch across compatible mask axes.
46    pub stretch: bool,
47    /// Final destination, if any, to protect named mask inputs.
48    pub destination: Option<&'a Path>,
49}

80    /// Resolve the selected region and build the retained grid.
81    ///
82    /// # Errors
83    /// The axes, region, units, or coordinate record cannot describe this operation.
84    pub fn new<T: CollapsePixel>(
85        source: &dyn ImageInterface<T>,
86        native: RecordValue,
87        options: ImageCollapseOptions<'a>,
88        beams: &RecordValue,
89    ) -> Result<Self, ImageError> {
90        let mut reduced = vec![false; source.ndim()];
91        for &axis in options.axes {
92            let flag = reduced.get_mut(axis).ok_or_else(|| {
93                invalid(format!(
94                    "Specified zero-based axis ({axis}) must be less than the number of axes ({})",
95                    source.ndim()
96                ))
97            })?;
98            if std::mem::replace(flag, true) {
99                return Err(invalid("Collapse axes must be distinct"));
100            }
101        }
102        let mut selection = SubimageSelection::resolve_section(
103            source,
104            options.spatial,
105            options.region,
106            options.chans,
107            options.stokes,
108        )?;
109        if options.region.is_none()
110            && options.spatial.is_none()
111            && let Some(axes) = implicit_positional_axes(source)
112        {
113            for axis in axes {
114                selection.shape[axis] = 1;
115            }
116        }
117        let tabular = source
118            .coordinates()
119            .find_coordinate(CoordinateType::Tabular)
120            .map(|_| native.clone());
121        let mut section =
122            SubimagePlan::from_selection(source, Some(native), selection, false, &[], Some(beams))?;
123        if let Some(tabular) = &tabular {
124            apply_decimation(&mut section, Some(tabular), 0, 1, DecimationMethod::Copy)?;
125        }
126        section.preserve_beams(source.coordinates(), beams)?;
127        let units = if options.function == CollapseFunction::Flux {
128            flux_units(source, &section, options.axes)?
129        } else {
130            source.units().to_owned()
131        };
132        let spectral = source.find_axis(CoordinateType::Spectral);
133        let stokes = source.find_axis(CoordinateType::Stokes);
134        let beam_pixels = if options.function == CollapseFunction::SqrtSumNpixBeam {
135            if let Some(beam) = section.image_info.beam_set.single_beam() {
136                statistics_flux_scale(&section.coordinates, "Jy/beam", Some(beam), None)
137                    .filter(|&scale| scale > 0.0)
138                    .map(|scale| 1.0 / scale)
139                    .ok_or_else(|| invalid("Image has no direction coordinate for the beam area"))?
140            } else {
141                1.0
142            }
143        } else {
144            1.0
145        };
146        if section.image_info.beam_set.is_multi()
147            && [spectral, stokes]
148                .into_iter()
149                .flatten()
150                .any(|axis| reduced[axis] && section.shape[axis] > 1)
151        {
152            section.keep_first_beam();
153        }
154        collapse_coordinates(&mut section, options.axes)?;
155        Ok(Self {
156            section,
157            options,
158            reduced,
159            units,
160            spectral,
161            stokes,
162            beam_pixels,
163        })
164    }

166    /// Reduce valid pixels in bounded source sections and write the result.
167    ///
168    /// # Errors
169    /// Output shape or I/O failed, or all selected source pixels are invalid.
170    pub fn copy_to<T: CollapsePixel>(
171        &self,
172        source: &dyn ImageInterface<T>,
173        output: &mut dyn MutableImageInterface<T>,
174    ) -> Result<(), ImageError> {
175        self.section.copy_metadata(source, output)?;
176        output.set_units_string(self.units.clone())?;
177        let mut has_mask = false;
178        self.visit_tiles(source, |position, results| {
179            output.put_slice(&results.values, position)?;
180            if results.mask.iter().any(|&present| !present) {
181                if !has_mask {
182                    output.create_filled_mask("mask0", true)?;
183                    output.set_default_mask("mask0")?;
184                    has_mask = true;
185                }
186                output.put_mask_slice("mask0", &results.mask, position)?;
187            }
188            Ok(())
189        })
190    }

192    pub(super) fn collect_profile<T: CollapsePixel>(
193        &self,
194        source: &dyn ImageInterface<T>,
195        axis: usize,
196    ) -> Result<CollapsedProfile<T>, ImageError> {
197        let length = self.section.shape[axis];
198        let mut result = CollapsedProfile {
199            values: vec![T::default(); length],
200            mask: vec![false; length],
201            counts: vec![0; length],
202        };
203        self.visit_tiles(source, |position, tile| {
204            let start = position[axis];
205            let end = start + tile.counts.len();
206            for (slot, &value) in result.values[start..end].iter_mut().zip(&tile.values) {
207                *slot = value;
208            }
209            for (slot, &value) in result.mask[start..end].iter_mut().zip(&tile.mask) {
210                *slot = value;
211            }
212            result.counts[start..end].copy_from_slice(&tile.counts);
213            Ok(())
214        })?;
215        Ok(result)
216    }

222    fn visit_tiles<T: CollapsePixel>(
223        &self,
224        source: &dyn ImageInterface<T>,
225        mut visit: impl FnMut(&[usize], ReducedTile<T>) -> Result<(), ImageError>,
226    ) -> Result<(), ImageError> {
227        let selection = &self.section.selection;
228        let reader = PixelSections {
229            image: source,
230            selection,
231            mask: PreparedMask::new_for_output(
232                self.options.mask,
233                source.shape(),
234                source.coordinates(),
235                self.options.stretch,
236                self.options.destination,
237            )?,
238        };
239        let ones = vec![1; source.ndim()];
240        let packed = selection.packed_shape();
241        let stride = selection.stride.as_deref().unwrap_or(&ones);
242        let groups = Groups {
243            reader: &reader,
244            section: StridedSection {
245                start: &selection.start,
246                shape: &packed,
247                stride,
248            },
249            reduced: &self.reduced,
250        };
251        let group_size = self
252            .reduced
253            .iter()
254            .zip(&packed)
255            .filter_map(|(&reduce, &length)| reduce.then_some(length))
256            .fold(1usize, usize::saturating_mul);
257        let median = self.options.function.uses_median();
258        let outer_budget = if median {
259            (MAX_TILE_SAMPLES / group_size).max(1)
260        } else {
261            4096
262        };
263        let cursor = bounded_cursor(&self.section.shape, stride, outer_budget);
264        let mut has_data = false;
265        for tile in
266            TraversalCursorIter::new(self.section.shape.clone(), cursor, TraversalSpec::tiles())
267        {
268            let tile = tile?;
269            let values = self.reduce_tile(&groups, &tile.position, &tile.shape, group_size)?;
270            has_data |= values.counts.iter().any(|&count| count != 0);
271            visit(&tile.position, values)?;
272        }
273        if !has_data {
274            return Err(invalid("All selected pixels are masked"));
275        }
276        Ok(())
277    }

279    fn reduce_tile<T: CollapsePixel>(
280        &self,
281        groups: &Groups<'_, T>,
282        start: &[usize],
283        shape: &[usize],
284        group_size: usize,
285    ) -> Result<ReducedTile<T>, ImageError> {
286        let size = shape.iter().product();
287        let function = self.options.function;
288        let (values, validity, counts) = if function.uses_median() {
289            if group_size > MAX_TILE_SAMPLES {
290                debug_assert_eq!(size, 1);
291                let (value, count) = reduce_large(function, |visit| {
292                    groups.for_each(start, shape, |_, _, value| visit(value))
293                })?;
294                (
295                    vec![value.unwrap_or_default()],
296                    vec![value.is_some()],
297                    vec![count],
298                )
299            } else {
300                let mut by_group = vec![vec![None; group_size]; size];
301                groups.for_each(start, shape, |index, position, sample| {
302                    by_group[index][position] = Some(sample);
303                    Ok(())
304                })?;
305                let (values, counts): (Vec<_>, Vec<_>) = by_group
306                    .into_iter()
307                    .map(|group| {
308                        let samples: Vec<_> = group.into_iter().flatten().collect();
309                        let count = samples.len();
310                        (reduce(function, samples), count)
311                    })
312                    .unzip();
313                let validity = values.iter().map(Option::is_some).collect();
314                (
315                    values.into_iter().map(Option::unwrap_or_default).collect(),
316                    validity,
317                    counts,
318                )
319            }
320        } else {
321            let mut cells = (0..size).map(|_| Summary::default()).collect::<Vec<_>>();
322            let mut retained = self
323                .reduced
324                .iter()
325                .enumerate()
326                .filter_map(|(axis, &reduced)| (!reduced).then_some(axis));
327            if !matches!(function, CollapseFunction::Min | CollapseFunction::Max)
328                && let (Some(axis), None) = (retained.next(), retained.next())
329            {
330                groups.for_each_plane(start, shape, axis, |index, sample| {
331                    cells[index].add(function, sample, 0)
332                })?;
333            } else {
334                groups.for_each(start, shape, |index, position, sample| {
335                    cells[index].add(function, sample, position);
336                    Ok(())
337                })?;
338            }
339            let counts = cells.iter().map(|cell| cell.count).collect();
340            let (values, validity) = cells
341                .iter()
342                .enumerate()
343                .map(|(index, cell)| {
344                    let scale = if function == CollapseFunction::Flux {
345                        self.flux_scale(groups.reader.image.units(), start, shape, index)?
346                    } else {
347                        1.0
348                    };
349                    Ok(cell.finish(function, scale, self.beam_pixels))
350                })
351                .collect::<Result<Vec<_>, ImageError>>()?
352                .into_iter()
353                .unzip();
354            (values, validity, counts)
355        };
356        Ok(ReducedTile {
357            values: ArrayD::from_shape_vec(IxDyn(shape), values)
358                .expect("one result per output cell"),
359            mask: ArrayD::from_shape_vec(IxDyn(shape), validity)
360                .expect("one validity per output cell"),
361            counts,
362        })
363    }

365    fn flux_scale(
366        &self,
367        brightness: &str,
368        start: &[usize],
369        shape: &[usize],
370        index: usize,
371    ) -> Result<f64, ImageError> {
372        let planes = |axis: Option<usize>| {
373            axis.map_or(0, |axis| {
374                let stride = shape[axis + 1..].iter().product::<usize>();
375                start[axis] + (index / stride) % shape[axis]
376            })
377        };
378        let beams = &self.section.image_info.beam_set;
379        let beam =
380            (!beams.is_empty()).then(|| *beams.beam(planes(self.spectral), planes(self.stokes)));
381        statistics_flux_scale(&self.section.coordinates, brightness, beam, None)
382            .ok_or_else(|| invalid("Unable to calculate flux density"))
383    }
384}

386fn flux_units<T: CollapsePixel>(
387    source: &dyn ImageInterface<T>,
388    section: &SubimagePlan,
389    axes: &[usize],
390) -> Result<String, ImageError> {
391    if source.find_axis(CoordinateType::Direction).is_none() {
392        return Err(invalid(
393            "Image has no direction coordinate. Cannot do flux density calculation.",
394        ));
395    }
396    let brightness = source.units();
397    if brightness.contains("beam") && section.image_info.beam_set.is_empty() {
398        return Err(invalid(
399            "Image has no beam. Cannot do flux density calculation.",
400        ));
401    }
402    let direction = direction_plane_axes(source)?;
403    for &axis in axes {
404        if !direction.contains(&axis) && section.shape[axis] > 1 {
405            return Err(invalid(format!(
406                "Specified axis {axis} is not a direction axis but has length > 1. Cannot do flux density calculation."
407            )));
408        }
409    }
410    if brightness.contains('K') {
411        Ok(format!("{brightness}.arcsec2"))
412    } else if let Some((prefix, suffix)) = brightness.split_once("/beam") {
413        Ok(format!("{prefix}{suffix}"))
414    } else {
415        Err(invalid("Unable to calculate flux density"))
416    }
417}

419fn collapse_coordinates(section: &mut SubimagePlan, axes: &[usize]) -> Result<(), ImageError> {
420    let coordinates = &section.coordinates;
421    let record = section
422        .coordinate_record
423        .take()
424        .unwrap_or_else(|| coordinates.to_record());
425    let record = collapsed_coordinate_record(coordinates, &section.shape, record, axes)?;
426    for &axis in axes {
427        section.shape[axis] = 1;
428    }
429    section.coordinates = CoordinateSystem::from_record(&record)?;
430    section.coordinate_record = Some(record);
431    Ok(())
432}

434pub(super) fn collapsed_coordinate_record(
435    coordinates: &CoordinateSystem,
436    shape: &[usize],
437    mut record: RecordValue,
438    axes: &[usize],
439) -> Result<RecordValue, ImageError> {
440    let first = coordinates.to_world(&vec![0.0; shape.len()])?;
441    let end = shape
442        .iter()
443        .map(|&length| (length - 1) as f64)
444        .collect::<Vec<_>>();
445    let last = coordinates.to_world(&end)?;
446    let mut interior = None;
447    for &axis in axes {
448        let world = coordinates
449            .pixel_axis_to_world_axis(axis)
450            .ok_or_else(|| invalid("Collapsed pixel axis has no corresponding world axis"))?;
451        let (index, local) = coordinates
452            .pixel_axis_owner(axis)
453            .ok_or_else(|| invalid("Collapsed pixel axis has no coordinate"))?;
454        let model = coordinates.coordinate(index);
455        let average = if let CoordinateModel::Direction(direction) = model {
456            let midpoint = if direction.longitude_axis() == Some(local) {
457                let circle = std::f64::consts::TAU;
458                let half = std::f64::consts::PI;
459                if interior.is_none() {
460                    let center = end.iter().map(|&length| length * 0.5).collect::<Vec<_>>();
461                    interior = Some(coordinates.to_world(&center)?);
462                }
463                let center = interior.as_ref().expect("interior longitude")[world];
464                let offset = |longitude: f64| (longitude - center + half).rem_euclid(circle) - half;
465                (center + 0.5 * (offset(first[world]) + offset(last[world]))).rem_euclid(circle)
466            } else {
467                0.5 * (first[world] + last[world])
468            };
469            let stored = direction.reference_and_increment_in_world_units().1[local];
470            midpoint * (stored / model.increment()[local])
471        } else {
472            0.5 * (first[world] + last[world])
473        };
474        CoordinateSystem::set_reference_value_in_record(&mut record, axis, average)?;
475        CoordinateSystem::set_reference_pixel_in_record(&mut record, axis, 0.0)?;
476    }
477    Ok(record)
478}

480fn bounded_cursor(shape: &[usize], stride: &[usize], mut budget: usize) -> Vec<usize> {
481    shape
482        .iter()
483        .zip(stride)
484        .map(|(&length, &step)| {
485            let width = budget.div_ceil(step).min(length);
486            budget = (budget / ((width - 1) * step + 1)).max(1);
487            width
488        })
489        .collect()
490}

492struct Groups<'a, T: CollapsePixel> {
493    reader: &'a PixelSections<'a, T>,
494    section: StridedSection<'a>,
495    reduced: &'a [bool],
496}
497
498impl<T: CollapsePixel> Groups<'_, T> {
499    fn for_each(
500        &self,
501        start: &[usize],
502        shape: &[usize],
503        mut visit: impl FnMut(usize, usize, T) -> Result<(), ImageError>,
504    ) -> Result<(), ImageError> {
505        let mut group_step = 1;
506        let group_weights: Vec<_> = shape
507            .iter()
508            .rev()
509            .map(|&length| {
510                let weight = group_step;
511                group_step *= length;
512                weight
513            })
514            .collect();
515        let mut collapse_step = 1;
516        let collapse_weights: Vec<_> = self
517            .reduced
518            .iter()
519            .zip(self.section.shape)
520            .map(|(&reduce, &length)| {
521                if !reduce {
522                    return 0;
523                }
524                let weight = collapse_step;
525                collapse_step *= length;
526                weight
527            })
528            .collect();
529        self.visit_windows(start, shape, |tile_position, pixels, valid| {
530            let pixels = pixels.reversed_axes();
531            let valid = valid.map(ArrayD::reversed_axes);
532            for (index, &pixel) in pixels.indexed_iter() {
533                if valid.as_ref().is_some_and(|valid| !valid[&index]) {
534                    continue;
535                }
536                let mut group = 0;
537                let mut position = 0;
538                for (axis, (&local, &tile_start)) in
539                    index.slice().iter().rev().zip(tile_position).enumerate()
540                {
541                    if self.reduced[axis] {
542                        position += (tile_start + local) * collapse_weights[axis];
543                    } else {
544                        group += local * group_weights[shape.len() - 1 - axis];
545                    }
546                }
547                visit(group, position, pixel)?;
548            }
549            Ok(())
550        })
551    }
552
553    fn for_each_plane(
554        &self,
555        start: &[usize],
556        shape: &[usize],
557        retained_axis: usize,
558        mut visit: impl FnMut(usize, T),
559    ) -> Result<(), ImageError> {
560        self.visit_windows(start, shape, |tile_position, values, mask| {
561            let axis = Axis(retained_axis);
562            let offset = tile_position[retained_axis];
563            for (index, plane) in values.axis_iter(axis).enumerate() {
564                let group = offset + index;
565                if let Some(mask) = &mask {
566                    Zip::from(plane).and(mask.index_axis(axis, index)).for_each(
567                        |&value, &valid| {
568                            if valid {
569                                visit(group, value);
570                            }
571                        },
572                    );
573                } else {
574                    for &value in &plane {
575                        visit(group, value);
576                    }
577                }
578            }
579            Ok(())
580        })
581    }
582
583    fn visit_windows(
584        &self,
585        start: &[usize],
586        shape: &[usize],
587        mut visit: impl FnMut(&[usize], ArrayD<T>, Option<ArrayD<bool>>) -> Result<(), ImageError>,
588    ) -> Result<(), ImageError> {
589        let full = self.section.shape;
590        let stride = self.section.stride;
591        let request: Vec<_> = self
592            .reduced
593            .iter()
594            .enumerate()
595            .map(
596                |(axis, &reduce)| {
597                    if reduce { full[axis] } else { shape[axis] }
598                },
599            )
600            .collect();
601        let dense_outer: usize = shape
602            .iter()
603            .zip(stride)
604            .map(|(&length, &step)| (length - 1) * step + 1)
605            .product();
606        let mut budget = MAX_TILE_SAMPLES / dense_outer;
607        let cursor = self
608            .reduced
609            .iter()
610            .enumerate()
611            .map(|(axis, &reduce)| {
612                if reduce {
613                    let width = budget.div_ceil(stride[axis]).min(full[axis]);
614                    budget = if width < full[axis] {
615                        1
616                    } else {
617                        (budget / ((width - 1) * stride[axis] + 1)).max(1)
618                    };
619                    width
620                } else {
621                    shape[axis]
622                }
623            })
624            .collect();
625        for tile in TraversalCursorIter::new(request, cursor, TraversalSpec::tiles()) {
626            let tile = tile?;
627            let packed: Vec<_> = tile
628                .position
629                .iter()
630                .zip(start)
631                .map(|(&offset, &origin)| offset + origin)
632                .collect();
633            let source = self.section.source_start(&packed);
634            if !self
635                .reader
636                .selection
637                .overlaps_window(&source, &self.section.dense_shape(&tile.shape))
638            {
639                continue;
640            }
641            let (pixels, valid) = self.reader.read(&source, &tile.shape, stride)?;
642            visit(&tile.position, pixels, valid)?;
643        }
644        Ok(())
645    }
