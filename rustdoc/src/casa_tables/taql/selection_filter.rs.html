12/// Find the first physical row satisfying a prepared selection's WHERE
13/// expression. This is used for MeasurementSet time defaults, for which
14/// ORDER BY and LIMIT are not part of the selection grammar.
15/// # Errors
16/// The predicate cannot be prepared or evaluated for a candidate row.
17#[doc(hidden)]
18pub fn first_matching_row(
19    sel: &SelectStatement,
20    table: &crate::Table,
21) -> Result<Option<usize>, TaqlError> {
22    let Some(mut predicate) = sel.where_clause.clone() else {
23        return Ok((table.row_count() > 0).then_some(0));
24    };
25    super::mscal_lookup::bind(&mut predicate, table, sel.style)?;
26    super::mscal_help::fold_filter(&mut predicate, table, sel.style)?;
27    let mut columns = HashSet::new();
28    let mut all_columns = false;
29    collect_filter_columns(&mut predicate, &mut columns, &mut all_columns);
30    let schema = table.schema();
31    let columns = schema.map(|schema| {
32        schema
33            .columns()
34            .iter()
35            .filter(|column| all_columns || columns.contains(column.name()))
36            .collect::<Vec<_>>()
37    });
38    for row_index in 0..table.row_count() {
39        let row;
40        let fields = if let Some(columns) = &columns {
41            row = read_expression_row(table, row_index, columns)?;
42            &row
43        } else {
44            table
45                .row_accessor()
46                .row(row_index)
47                .map_err(|error| TaqlError::Table(error.to_string()))?
48        };
49        let context = EvalContext {
50            row: fields,
51            row_index,
52            style: sel.style,
53            measures: table.measures_provider(),
54        };
55        if matches_where(&predicate, &context)? {
56            return Ok(Some(row_index));
57        }
58    }
59    Ok(None)
60}

62fn collect_filter_columns(expr: &mut Expr, names: &mut HashSet<String>, all_columns: &mut bool) {
63    match expr {
64        Expr::ColumnRef(column) => {
65            names.insert(column.column.clone());
66        }
67        Expr::BoundRowFunction(_) => *all_columns = true,
68        Expr::FunctionCall { name, .. } if super::functions::has_udf(name) => *all_columns = true,
69        _ => {}
70    }
71    let Ok(()) = expr.try_for_each_child_mut(&mut |child| {
72        collect_filter_columns(child, names, all_columns);
73        Ok::<(), std::convert::Infallible>(())
74    });
75}

77fn read_expression_row(
78    table: &crate::Table,
79    row_index: usize,
80    columns: &[&crate::ColumnSchema],
81) -> Result<RecordValue, TaqlError> {
82    use crate::ColumnType;
83    let convert_error = |error: crate::TableError| TaqlError::Table(error.to_string());
84    let fields = columns
85        .iter()
86        .map(|column| {
87            let name = column.name();
88            let accessor = table.column_accessor(name).map_err(convert_error)?;
89            let value = match column.column_type() {
90                ColumnType::Scalar => accessor
91                    .scalar_cells_owned_for_rows(&[row_index])
92                    .map_err(convert_error)?
93                    .pop()
94                    .flatten()
95                    .map(Value::Scalar),
96                ColumnType::Array(_) => accessor
97                    .array_cells_owned_uncached(&[row_index])
98                    .map_err(convert_error)?
99                    .pop()
100                    .flatten()
101                    .map(Value::Array),
102                ColumnType::Record => Some(Value::Record(
103                    accessor.record_cell(row_index).map_err(convert_error)?,
104                )),
105            };
106            Ok(value.map(|value| RecordField {
107                name: name.into(),
108                value,
109            }))
110        })
111        .collect::<Result<Vec<_>, _>>()?;
112    Ok(RecordValue::new(fields.into_iter().flatten().collect()))
113}

115fn matches_where(expr: &Expr, context: &EvalContext<'_>) -> Result<bool, TaqlError> {
116    match expr {
117        Expr::Binary {
118            left,
119            op: BinaryOp::Or,
120            right,
121        } => Ok(matches_where(left, context)? || matches_where(right, context)?),
122        Expr::Binary {
123            left,
124            op: BinaryOp::And,
125            right,
126        } => Ok(matches_where(left, context)? && matches_where(right, context)?),
127        _ => eval_expr(expr, context)?.to_bool(),
128    }
129}
