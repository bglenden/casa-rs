21pub(super) fn lines(
22    ms: &MeasurementSet,
23    spec: &MsExploreSpec,
24    input: &PlotMsInput<'_>,
25) -> Vec<String> {
26    use MsPageHeaderItem::*;
27
28    if spec.header_items.is_empty() {
29        return Vec::new();
30    }
31    let main = ms.main_table();
32    let observation = input.header_row.and_then(|row| {
33        let id = usize::try_from(main_ids::observation_id(main).get(row).ok()?).ok()?;
34        Some((ms.observation().ok()?, id))
35    });
36    let read_obs = |name| {
37        observation
38            .as_ref()
39            .and_then(|(table, id)| table.string(*id, name).ok())
40            .unwrap_or_default()
41    };
42    let field = input.header_row.and_then(|row| {
43        let id = usize::try_from(main_ids::field_id(main).get(row).ok()?).ok()?;
44        Some((ms.field().ok()?, id))
45    });
46    let start = spec
47        .header_items
48        .iter()
49        .any(|item| matches!(item, ObsDate | ObsTime))
50        .then(|| {
51            observation
52                .as_ref()
53                .and_then(|(table, id)| observation_start(table, *id))
54        })
55        .flatten();
56    let segments = spec
57        .header_items
58        .iter()
59        .map(|item| match item {
60            Filename => format!(
61                "File: {}",
62                spec.ms_path
63                    .file_name()
64                    .unwrap_or(spec.ms_path.as_os_str())
65                    .to_string_lossy()
66            ),
67            YColumn => format!("Y Column(s): {}", y_columns(spec, input)),
68            ObsDate => format!(
69                "Observation date: {}",
70                start.map(observation_date).unwrap_or_default()
71            ),
72            ObsTime => format!(
73                "Observation time: {}",
74                start
75                    .map(|time| format!("{} UTC", format_time(time.as_mjd_days(), TIME | CLEAN, 4)))
76                    .unwrap_or_default()
77            ),
78            Observer => format!("Observer: {}", read_obs("OBSERVER")),
79            ProjId => format!("Project ID: {}", read_obs("PROJECT")),
80            Telescope => format!("Telescope: {}", read_obs("TELESCOPE_NAME")),
81            TargName => format!(
82                "Target name: {}",
83                field
84                    .as_ref()
85                    .and_then(|(table, id)| table.name(*id).ok())
86                    .unwrap_or_default()
87            ),
88            TargDir => format!(
89                "Target RA,Dec: {}",
90                field
91                    .as_ref()
92                    .and_then(|(table, id)| target_direction(ms, input.header_row?, table, *id))
93                    .unwrap_or_default()
94            ),
95        })
96        .collect::<Vec<_>>();
97    wrap_header_segments(&segments)
98}

100fn observation_start(table: &MsObservation<'_>, row: usize) -> Option<MvTime> {
101    let ArrayValue::Float64(range) = table.array(row, "TIME_RANGE").ok()? else {
102        return None;
103    };
104    let seconds = *range.iter().next()?;
105    if !seconds.is_finite() {
106        return None;
107    }
108    let reference = epoch_reference(table.table(), "TIME_RANGE", row, 0).ok()?;
109    let start = MEpoch::from_mjd_seconds(seconds, reference);
110    let start = if reference == EpochRef::UTC {
111        start
112    } else {
113        let frame = MeasFrame::new().with_measures(casa_ms::open_measures_runtime().ok()?);
114        start.convert_to(EpochRef::UTC, &frame).ok()?
115    };
116    Some(MvTime::from_mjd_days(start.value().as_mjd()))
117}

119fn observation_date(time: MvTime) -> String {
120    let date = time.format_dmy_date();
121    let (_, month_year) = date.split_once('-').expect("DMY has a day separator");
122    format!("{} {}", time.month_day(), month_year.replacen('-', " ", 1))
123}

125fn y_columns(spec: &MsExploreSpec, input: &PlotMsInput<'_>) -> String {
126    spec.plots
127        .iter()
128        .flat_map(|plot| {
129            plot.y_axes.iter().enumerate().map(|(index, &axis)| {
130                let column = match index {
131                    0 => plot.data_column,
132                    1 => input.secondary_y_data_column.unwrap_or(plot.data_column),
133                    _ => input
134                        .additional_ordinates
135                        .get(index - 2)
136                        .map_or(plot.data_column, |entry| entry.1),
137                };
138                let name = axis_name(axis);
139                if axis.is_visibility_math() {
140                    format!("{name}:{}", column_name(column, input.prefer_float_data))
141                } else {
142                    name.to_owned()
143                }
144            })
145        })
146        .collect::<Vec<_>>()
147        .join(", ")
148}

150fn column_name(column: MsDataColumn, prefer_float: bool) -> &'static str {
151    use MsDataColumn::*;
152
153    match column {
154        Data if prefer_float => "float",
155        CorrectedMinusModel => "corrected-model_vector",
156        CorrectedMinusModelScalar => "corrected-model_scalar",
157        DataMinusModel => "data-model_vector",
158        DataMinusModelScalar => "data-model_scalar",
159        CorrectedDivModel => "corrected/model_vector",
160        CorrectedDivModelScalar => "corrected/model_scalar",
161        DataDivModel => "data/model_vector",
162        DataDivModelScalar => "data/model_scalar",
163        stored => stored.as_str(),
164    }
165}

167fn target_direction(
168    ms: &MeasurementSet,
169    main_row: usize,
170    field: &MsField<'_>,
171    id: usize,
172) -> Option<String> {
173    let source = ms.source().ok()?;
174    let source_id = field.source_id(id).ok()?;
175    let data_desc = main_ids::data_desc_id(ms.main_table()).get(main_row).ok()?;
176    let spw = ms
177        .data_description()
178        .ok()?
179        .spectral_window_id(usize::try_from(data_desc).ok()?)
180        .ok()?;
181    let (row, _) = (0..source.row_count())
182        .filter_map(|row| {
183            let id = source.i32(row, "SOURCE_ID").ok()?;
184            let window = source.i32(row, "SPECTRAL_WINDOW_ID").ok()?;
185            if id != source_id || (window != -1 && window != spw) {
186                return None;
187            }
188            Some((row, source.f64(row, "TIME").ok()?))
189        })
190        .min_by(|(_, left), (_, right)| left.total_cmp(right))?;
191    let ArrayValue::Float64(values) = source.array(row, "DIRECTION").ok()? else {
192        return None;
193    };
194    let [lon, lat] = values.as_slice()? else {
195        return None;
196    };
197    if !lon.is_finite() || !lat.is_finite() {
198        return None;
199    }
200    let reference = resolve_direction_reference(source.table(), "SOURCE", "DIRECTION", row).ok()?;
201    let (lon, lat) = MDirection::from_angles(*lon, *lat, reference).as_angles();
202    Some(direction_label(lon, lat, reference))
203}

205fn direction_label(lon: f64, lat: f64, reference: DirectionRef) -> String {
206    use DirectionRef::*;
207
208    let angles = if matches!(
209        reference,
210        ICRS | J2000 | JMEAN | JTRUE | APP | B1950 | B1950Vla | BMEAN | BTRUE
211    ) {
212        let sign = if lat < 0.0 { "-" } else { "" };
213        format!(
214            "{} {sign}{}",
215            format_angle(lon, TIME, 12),
216            format_angle(lat.abs(), ANGLE | CLEAN, 11).trim()
217        )
218    } else {
219        format!(
220            "{} {}",
221            format_angle(lon, ANGLE, 11),
222            format_angle(lat, ANGLE, 11)
223        )
224    };
225    if reference == ICRS {
226        angles
227    } else {
228        format!("{angles} {reference}")
229    }
