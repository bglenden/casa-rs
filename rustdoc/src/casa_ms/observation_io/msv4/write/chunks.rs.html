2//! Whole-chunk writes. zarrs stores a chunk directly only when one write
3//! covers all of it; any other write decodes the stored chunk, merges and
4//! re-encodes it, once per batch that touches it. So the times of a batch
5//! that cover a chunk only partly are held here until the chunk's other
6//! times arrive, and every chunk of a dataset written in time order is
7//! encoded exactly once, whatever the batch size. Whole chunks of a batch
8//! are stored straight from its views.
9
10use std::ops::Range;
11
12use num_complex::Complex32;
13
14/// One write's elements of every payload variable, times outermost: a
15/// batch's, from its views, or the held times'. `values` holds one slice
16/// per data group the dataset declares, in declaration order.
17#[derive(Clone)]
18pub(super) struct Columns<'a> {
19    pub(super) values: Vec<&'a [Complex32]>,
20    pub(super) flags: &'a [bool],
21    pub(super) weights: &'a [f32],
22    pub(super) uvw: &'a [f64],
23    pub(super) centroid: &'a [f64],
24    pub(super) exposure: &'a [f64],
25}
26
27/// The elements one time holds in each variable: `cube` in each group's
28/// visibilities and in the flags and weights, three per baseline in the
29/// UVW, one per baseline in the centroids and exposures.
30#[derive(Clone, Copy, Debug)]
31pub(super) struct PerTime {
32    pub(super) baselines: usize,
33    pub(super) cube: usize,
34    /// Visibility variables: one per data group.
35    pub(super) groups: usize,
36}
37
38impl PerTime {
39    /// Elements of every variable at one time.
40    pub(super) fn elements(self) -> usize {
41        (self.groups + 2) * self.cube + 5 * self.baselines
42    }
43
44    /// Bytes of every variable at one time.
45    pub(super) fn bytes(self) -> usize {
46        let cube = self.groups * size_of::<Complex32>() + size_of::<bool>() + size_of::<f32>();
47        self.cube * cube + self.baselines * 5 * size_of::<f64>()
48    }
49}
50
51impl<'a> Columns<'a> {
52    /// The elements of `times`, counted from the first time these hold.
53    pub(super) fn times(&self, times: &Range<usize>, per: PerTime) -> Self {
54        let span = |n: usize| times.start * n..times.end * n;
55        Self {
56            values: self
57                .values
58                .iter()
59                .map(|values| &values[span(per.cube)])
60                .collect(),
61            flags: &self.flags[span(per.cube)],
62            weights: &self.weights[span(per.cube)],
63            uvw: &self.uvw[span(3 * per.baselines)],
64            centroid: &self.centroid[span(per.baselines)],
65            exposure: &self.exposure[span(per.baselines)],
66        }
67    }
68}

70/// Times held toward one whole chunk: contiguous dataset times inside one
71/// chunk, and their elements of every variable. It holds at most one
72/// chunk of times, reserved when it first holds any.
73#[derive(Default)]
74pub(super) struct Held {
75    times: Range<usize>,
76    values: Vec<Vec<Complex32>>,
77    flags: Vec<bool>,
78    weights: Vec<f32>,
79    uvw: Vec<f64>,
80    centroid: Vec<f64>,
81    exposure: Vec<f64>,
82}
83
84impl Held {
85    /// The dataset times held; empty when none are.
86    pub(super) fn times(&self) -> Range<usize> {
87        self.times.clone()
88    }
89
90    /// Hold `columns`, the elements of dataset times `times`, after those
91    /// already held, in a holding of at most `chunk` times.
92    pub(super) fn hold(
93        &mut self,
94        times: Range<usize>,
95        columns: &Columns<'_>,
96        per: PerTime,
97        chunk: usize,
98    ) {
99        if self.times.is_empty() {
100            if self.flags.capacity() == 0 {
101                self.values = (0..per.groups)
102                    .map(|_| Vec::with_capacity(chunk * per.cube))
103                    .collect();
104                self.flags.reserve_exact(chunk * per.cube);
105                self.weights.reserve_exact(chunk * per.cube);
106                self.uvw.reserve_exact(chunk * 3 * per.baselines);
107                self.centroid.reserve_exact(chunk * per.baselines);
108                self.exposure.reserve_exact(chunk * per.baselines);
109            }
110            self.times = times;
111        } else {
112            debug_assert_eq!(self.times.end, times.start, "held times are contiguous");
113            self.times.end = times.end;
114        }
115        for (held, values) in self.values.iter_mut().zip(&columns.values) {
116            held.extend_from_slice(values);
117        }
118        self.flags.extend_from_slice(columns.flags);
119        self.weights.extend_from_slice(columns.weights);
120        self.uvw.extend_from_slice(columns.uvw);
121        self.centroid.extend_from_slice(columns.centroid);
122        self.exposure.extend_from_slice(columns.exposure);
123    }
124
125    /// The held elements.
126    pub(super) fn columns(&self) -> Columns<'_> {
127        Columns {
128            values: self.values.iter().map(Vec::as_slice).collect(),
129            flags: &self.flags,
130            weights: &self.weights,
131            uvw: &self.uvw,
132            centroid: &self.centroid,
133            exposure: &self.exposure,
134        }
135    }
136
137    /// Hold nothing, keeping the storage for the next chunk.
138    pub(super) fn clear(&mut self) {
139        self.times = 0..0;
140        self.values.iter_mut().for_each(Vec::clear);
141        self.flags.clear();
142        self.weights.clear();
143        self.uvw.clear();
144        self.centroid.clear();
145        self.exposure.clear();
146    }
147}

149/// One write of a batch's times, or of the held ones.
150#[derive(Clone, Debug, PartialEq, Eq)]
151pub(super) enum Step {
152    /// Copy these dataset times of the batch after the held ones.
153    Hold(Range<usize>),
154    /// Store these dataset times of the batch straight from its views:
155    /// whole chunks only.
156    Store(Range<usize>),
157    /// Store the held times and hold none.
158    Release,
159}
160
161/// The chunk of `chunk` times holding dataset time `time`, in a dataset of
162/// `total` times: the last chunk ends at `total`.
163pub(super) fn chunk_of(time: usize, chunk: usize, total: usize) -> Range<usize> {
164    let start = time / chunk * chunk;
165    start..(start + chunk).min(total)
166}
167
168/// How to write the dataset times `batch`, with `held` already held, in
169/// chunks of `chunk` times of a dataset of `total`: whole chunks straight
170/// from the batch, the rest held until its chunk is complete and then
171/// released whole. Held times the batch does not continue are released
172/// first, covering their chunk only partly.
173pub(super) fn plan(
174    held: Range<usize>,
175    batch: Range<usize>,
176    chunk: usize,
177    total: usize,
178) -> Vec<Step> {
179    let mut steps = Vec::new();
180    let mut held = held;
181    let mut stored: Option<Range<usize>> = None;
182    let mut start = batch.start;
183    while start < batch.end {
184        let whole = chunk_of(start, chunk, total);
185        let end = whole.end.min(batch.end);
186        if !held.is_empty() && (held.end != start || chunk_of(held.start, chunk, total) != whole) {
187            steps.push(Step::Release);
188            held = 0..0;
189        }
190        if held.is_empty() && start == whole.start && end == whole.end {
191            stored = Some(stored.map_or(start..end, |run| run.start..end));
192        } else {
193            if let Some(run) = stored.take() {
194                steps.push(Step::Store(run));
195            }
196            steps.push(Step::Hold(start..end));
197            held = if held.is_empty() {
198                start..end
199            } else {
200                held.start..end
201            };
202            if held == whole {
203                steps.push(Step::Release);
204                held = 0..0;
205            }
206        }
207        start = end;
208    }
209    if let Some(run) = stored {
210        steps.push(Step::Store(run));
211    }
212    steps
213}

215#[cfg(test)]
216mod tests {
217    use super::Step::{Hold, Release, Store};
218    use super::plan;
219
220    #[test]
221    fn whole_chunks_are_stored_and_the_rest_held_until_its_chunk_completes() {
222        // Five-time chunks of a twelve-time dataset: 0..5, 5..10, 10..12.
223        assert_eq!(plan(0..0, 0..12, 5, 12), [Store(0..12)]);
224        assert_eq!(plan(0..0, 0..4, 5, 12), [Hold(0..4)]);
225        assert_eq!(plan(0..4, 4..8, 5, 12), [Hold(4..5), Release, Hold(5..8)]);
226        assert_eq!(
227            plan(5..8, 8..12, 5, 12),
228            [Hold(8..10), Release, Store(10..12)],
229            "the short last chunk is whole"
230        );
231    }
232
233    #[test]
234    fn held_times_a_batch_does_not_continue_are_released_partly() {
235        assert_eq!(plan(5..8, 0..5, 5, 12), [Release, Store(0..5)]);
236        assert_eq!(plan(0..2, 3..4, 5, 12), [Release, Hold(3..4)]);
237        assert_eq!(plan(3..5, 5..7, 5, 12), [Release, Hold(5..7)]);
238        assert_eq!(
239            plan(1..3, 3..12, 5, 12),
240            [Hold(3..5), Release, Store(5..12)],
241            "1..5 never completes its chunk, so 5..10 is not held after it"
242        );
243    }
