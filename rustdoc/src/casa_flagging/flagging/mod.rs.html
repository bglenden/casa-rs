126/// Input visibility column used by automatic flagging modes.
127#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
128#[serde(rename_all = "snake_case")]
129pub enum FlagDataColumn {
130    /// MAIN.DATA.
131    #[default]
132    Data,
133    /// MAIN.CORRECTED_DATA.
134    CorrectedData,
135    /// MAIN.FLOAT_DATA.
136    FloatData,
137    /// MAIN.MODEL_DATA.
138    ModelData,
139    /// Corrected visibilities minus the model.
140    Residual,
141    /// Observed visibilities minus the model.
142    ResidualData,
143    /// The stored weight spectrum or channel-repeated scalar weights.
144    Weight,
145    /// MAIN.WEIGHT_SPECTRUM.
146    WeightSpectrum,
147    /// Complex calibration parameters.
148    Cparam,
149    /// Real calibration parameters.
150    Fparam,
151    /// Calibration parameter uncertainties.
152    ParamErr,
153    /// Calibration solution signal-to-noise.
154    Snr,
155}
156
157impl FlagDataColumn {
158    /// The physical column name used by the flagging data source.
159    pub fn name(self) -> &'static str {
160        match self {
161            Self::Data => VisibilityDataColumn::Data.name(),
162            Self::CorrectedData => VisibilityDataColumn::CorrectedData.name(),
163            Self::FloatData => OptionalMainColumn::FloatData.name(),
164            Self::ModelData => VisibilityDataColumn::ModelData.name(),
165            Self::Residual => "RESIDUAL",
166            Self::ResidualData => "RESIDUAL_DATA",
167            Self::Weight => "WEIGHT",
168            Self::WeightSpectrum => OptionalMainColumn::WeightSpectrum.name(),
169            Self::Cparam => "CPARAM",
170            Self::Fparam => "FPARAM",
171            Self::ParamErr => "PARAMERR",
172            Self::Snr => "SNR",
173        }
174    }
175
176    fn is_real(self) -> bool {
177        matches!(
178            self,
179            Self::FloatData
180                | Self::Weight
181                | Self::WeightSpectrum
182                | Self::Fparam
183                | Self::ParamErr
184                | Self::Snr
185        )
186    }
187}

189/// Native `flagdata` mode.
190#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
191#[serde(rename_all = "snake_case")]
192pub enum FlagDataMode {
193    /// Manual selected-row/sample flagging.
194    Manual,
195    /// Clip visibility samples or signed floating-point values.
196    Clip,
197    /// CASA quack-style scan-edge flagging.
198    Quack,
199    /// Flag selected baselines outside the antenna elevation limits.
200    Elevation,
201    /// Flag selected baselines involving a shadowed antenna.
202    Shadow,
203    /// Propagate flags from a reference antenna to selected integrations.
204    Antint,
205    /// CASA TFCrop-family robust time/frequency outlier flagging.
206    Tfcrop,
207    /// CASA RFlag-family two-pass robust time/frequency flagging.
208    Rflag,
209    /// Extend existing flags in polarization, time, or frequency.
210    Extend,
211    /// Summarize flags without mutating the MS.
212    Summary,
213}

608fn persist_changed_flags(
609    ms: &mut MeasurementSet,
610    changes: &ChangeSet,
611    ms_path: &str,
612) -> Result<(), FlaggingError> {
613    let mut maximum_flag_bytes = 0usize;
614    for &row in &changes.changed_row_indices {
615        maximum_flag_bytes = maximum_flag_bytes.max(clone_flag_matrix(ms, row)?.len());
616    }
617    let mut columns = vec![MeasurementSetWriteColumnPlan {
618        name: FLAG_COLUMN.to_string(),
619        bytes_per_row: maximum_flag_bytes,
620        mode: MeasurementSetColumnWriteMode::Replace,
621        storage_manager: MeasurementSetColumnStorage::Persisted,
622        tile_shape: None,
623        create_source_column: None,
624    }];
625    if changes.changed_flag_row_rows > 0 {
626        columns.push(MeasurementSetWriteColumnPlan {
627            name: FLAG_ROW_COLUMN.to_string(),
628            bytes_per_row: 1,
629            mode: MeasurementSetColumnWriteMode::Replace,
630            storage_manager: MeasurementSetColumnStorage::Persisted,
631            tile_shape: None,
632            create_source_column: None,
633        });
634    }
635    let resources = MeasurementSetWriteResources::from_system_memory(2).map_err(|error| {
636        FlaggingError::MutateFlags {
637            path: ms_path.to_owned(),
638            reason: format!("plan changed MAIN flag rows: {error}"),
639        }
640    })?;
641    let include_flag_row = changes.changed_flag_row_rows > 0;
642    let write_plan = MeasurementSetWritePlan::selected_row_mutation(
643        changes.changed_row_indices.clone(),
644        columns,
645        resources,
646    )
647    .map_err(|error| FlaggingError::MutateFlags {
648        path: ms_path.to_owned(),
649        reason: format!("plan changed MAIN flag rows: {error}"),
650    })?;
651    let batch_rows = write_plan.batch_rows();
652    let mut write_session = MeasurementSetWriteSession::start_selected_row_mutation(ms, write_plan)
653        .map_err(|error| FlaggingError::MutateFlags {
654            path: ms_path.to_owned(),
655            reason: format!("start changed MAIN flag rows: {error}"),
656        })?;
657    for rows in changes.changed_row_indices.chunks(batch_rows) {
658        let flag_values = rows
659            .iter()
660            .map(|&row| clone_flag_matrix(ms, row).map(ArrayValue::Bool))
661            .collect::<Result<Vec<_>, _>>()?;
662        let mut batch_columns = vec![MeasurementSetMutationColumnBatch {
663            name: FLAG_COLUMN.to_string(),
664            values: MeasurementSetMutationColumnValues::Arrays(flag_values),
665        }];
666        if include_flag_row {
667            let values = load_bool_cells(ms.main_table(), rows, FLAG_ROW_COLUMN, ms_path)?
668                .into_iter()
669                .map(ScalarValue::Bool)
670                .collect();
671            batch_columns.push(MeasurementSetMutationColumnBatch {
672                name: FLAG_ROW_COLUMN.to_string(),
673                values: MeasurementSetMutationColumnValues::Scalars(values),
674            });
675        }
676        write_session
677            .write_mutation_batch(
678                ms,
679                MeasurementSetMutationBatch {
680                    row_indices: rows.to_vec(),
681                    columns: batch_columns,
682                },
683            )
684            .map_err(|error| FlaggingError::MutateFlags {
685                path: ms_path.to_owned(),
686                reason: format!("write changed MAIN flag rows: {error}"),
687            })?;
688    }
689    write_session
690        .finish_mutation()
691        .map_err(|error| FlaggingError::MutateFlags {
692            path: ms_path.to_owned(),
693            reason: format!("save changed MAIN flag rows: {error}"),
694        })?;
695    Ok(())
696}

811fn report_after(
812    ms: &MeasurementSet,
813    request: &FlagDataRequest,
814    rows: &[usize],
815    changes: ChangeSet,
816    backup_version: Option<String>,
817    thresholds: Option<(f64, f64)>,
818    time_selection: FlagTimeSelection,
819) -> Result<FlagDataReport, FlaggingError> {
820    let summary = match changes.summary {
821        Some(summary) => summary,
822        None if request.casa_row_selection
823            && (time_selection != FlagTimeSelection::Global
824                || (request.selection.timerange.is_none()
825                    && request.selection.msselect.is_none())) =>
826        {
827            summarize_flags(ms, rows)?
828        }
829        None => {
830            let summary_rows = selected_rows(ms, request, time_selection)?;
831            summarize_flags(ms, &summary_rows.rows)?
832        }
833    };
834    Ok(FlagDataReport {
835        mode: request.mode,
836        selected_rows: changes.selected_rows.unwrap_or(rows.len()),
837        changed_rows: changes.changed_rows,
838        changed_samples: changes.changed_samples,
839        flagged_samples: summary.flagged_samples,
840        total_samples: summary.total_samples,
841        backup_version,
842        timedev: thresholds.map(|pair| pair.0),
843        freqdev: thresholds.map(|pair| pair.1),
844        rflag_report: changes.rflag_report,
845        summary: None,
846    })
847}

849fn summarize_flags(
850    ms: &MeasurementSet,
851    selected_rows: &[usize],
852) -> Result<FlagSummary, FlaggingError> {
853    let mut flagged_samples = 0usize;
854    let mut total_samples = 0usize;
855    for &row in selected_rows {
856        let flags = clone_flag_matrix(ms, row)?;
857        flagged_samples += flags.iter().filter(|flag| **flag).count();
858        total_samples += flags.len();
859    }
860    Ok(FlagSummary {
861        flagged_samples,
862        total_samples,
863    })
864}

866fn summary_after_flag_mask(
867    samples: &[Sample],
868    changed_samples: usize,
869    flag_value: bool,
870) -> FlagSummary {
871    let flagged = samples.iter().filter(|sample| sample.flag).count();
872    FlagSummary {
873        flagged_samples: if flag_value {
874            flagged + changed_samples
875        } else {
876            flagged.saturating_sub(changed_samples)
877        },
878        total_samples: samples.len(),
879    }
880}

882fn percent_modified(samples: &[Sample], modified: &[bool]) -> f64 {
883    if samples.is_empty() {
884        return 0.0;
885    }
886    100.0
887        * samples
888            .iter()
889            .filter(|sample| sample.flag || modified[sample.index])
890            .count() as f64
891        / samples.len() as f64
892}

910fn flag_mask_counts_by_spw_corr(mask: &[bool], samples: &[Sample]) -> BTreeMap<String, usize> {
911    let mut counts = BTreeMap::<String, usize>::new();
912    for (sample, flag) in samples.iter().zip(mask.iter()) {
913        if *flag {
914            *counts
915                .entry(format!("spw{}_corr{}", sample.spw, sample.corr))
916                .or_default() += 1;
917        }
918    }
