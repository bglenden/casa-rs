6    #[allow(clippy::too_many_arguments)]
7    #[expect(
8        clippy::too_many_lines,
9        reason = "predates the function-length limit; #705"
10    )]
11    pub(crate) fn save_borrowed(
12        &self,
13        control: StagedTableDat<'_>,
14        rows: &[RecordValue],
15        undefined_cells: &[HashSet<String>],
16        keywords: &RecordValue,
17        column_keywords: &HashMap<String, RecordValue>,
18        schema: Option<&TableSchema>,
19        table_info: &TableInfo,
20        virtual_columns: &HashSet<String>,
21        virtual_bindings: &[virtual_engine::VirtualColumnBinding],
22        dm_kind: crate::table::DataManagerKind,
23        dm_group_name: Option<&str>,
24        casacore_default_ssm: bool,
25        big_endian: bool,
26        tile_shape: Option<&[usize]>,
27    ) -> Result<(), StorageError> {
28        use crate::table::DataManagerKind;
29
30        let table_path = control.table_path();
31
32        let schema = schema.ok_or_else(|| {
33            StorageError::FormatMismatch("cannot save without schema".to_string())
34        })?;
35
36        let filtered_rows_storage;
37        let filtered_rows: &[RecordValue] = if undefined_cells.iter().any(|set| !set.is_empty()) {
38            filtered_rows_storage = filter_rows_for_save(rows, undefined_cells);
39            &filtered_rows_storage
40        } else {
41            rows
42        };
43        let nrrow = filtered_rows.len() as u64;
44        let data_path = table_path.join(format!("{}0", TABLE_DATA_FILE_PREFIX));
45        let has_virtual = !virtual_bindings.is_empty();
46
47        let stored_rows: Vec<RecordValue>;
48        let rows_for_data = if has_virtual {
49            stored_rows = rows
50                .iter()
51                .zip(undefined_cells)
52                .map(|(row, undef)| {
53                    let stored_fields: Vec<_> = row
54                        .fields()
55                        .iter()
56                        .filter(|f| !virtual_columns.contains(&f.name) && !undef.contains(&f.name))
57                        .cloned()
58                        .collect();
59                    RecordValue::new(stored_fields)
60                })
61                .collect();
62            &stored_rows
63        } else {
64            filtered_rows
65        };
66
67        let dm_type_name;
68        let dm_data;
69        match dm_kind {
70            DataManagerKind::StManAipsIO => {
71                dm_type_name = "StManAipsIO".to_string();
72                dm_data = Vec::new();
73                let mut table_dat = if has_virtual {
74                    TableDatContents::from_snapshot_with_virtual(
75                        schema,
76                        keywords,
77                        column_keywords,
78                        nrrow,
79                        &dm_type_name,
80                        &dm_data,
81                        big_endian,
82                        virtual_bindings,
83                        control.keyword_table_path(),
84                    )
85                } else {
86                    TableDatContents::from_snapshot(
87                        schema,
88                        keywords,
89                        column_keywords,
90                        nrrow,
91                        &dm_type_name,
92                        &dm_data,
93                        big_endian,
94                    )
95                };
96                apply_physical_dm_descriptor_group(
97                    &mut table_dat,
98                    &dm_type_name,
99                    dm_group_name.unwrap_or(&dm_type_name),
100                    virtual_columns,
101                );
102                control.publish(&table_dat)?;
103                let stored_col_descs: Vec<_> = table_dat
104                    .table_desc
105                    .columns
106                    .iter()
107                    .filter(|c| !virtual_columns.contains(&c.col_name))
108                    .cloned()
109                    .collect();
110                write_stman_file(
111                    &data_path,
112                    0,
113                    &stored_col_descs,
114                    rows_for_data,
115                    ByteOrder::BigEndian,
116                )?;
117            }
118            DataManagerKind::StandardStMan => {
119                dm_type_name = "StandardStMan".to_string();
120                let table_dat_tmp = if has_virtual {
121                    TableDatContents::from_snapshot_with_virtual(
122                        schema,
123                        keywords,
124                        column_keywords,
125                        nrrow,
126                        "StandardStMan",
127                        &[],
128                        big_endian,
129                        virtual_bindings,
130                        control.keyword_table_path(),
131                    )
132                } else {
133                    TableDatContents::from_snapshot(
134                        schema,
135                        keywords,
136                        column_keywords,
137                        nrrow,
138                        "StandardStMan",
139                        &[],
140                        big_endian,
141                    )
142                };
143                let stored_col_descs: Vec<_> = table_dat_tmp
144                    .table_desc
145                    .columns
146                    .iter()
147                    .filter(|c| !virtual_columns.contains(&c.col_name))
148                    .cloned()
149                    .collect();
150                let group_name = dm_group_name.unwrap_or(&dm_type_name);
151                let existing_name = control.existing_ssm_name(0, group_name);
152                dm_data = write_ssm_file_with_options(
153                    &data_path,
154                    &stored_col_descs,
155                    rows_for_data,
156                    big_endian,
157                    casacore_default_ssm,
158                    existing_name.as_deref().unwrap_or(group_name),
159                )?;
160                let mut table_dat = if has_virtual {
161                    TableDatContents::from_snapshot_with_virtual(
162                        schema,
163                        keywords,
164                        column_keywords,
165                        nrrow,
166                        &dm_type_name,
167                        &dm_data,
168                        big_endian,
169                        virtual_bindings,
170                        control.keyword_table_path(),
171                    )
172                } else {
173                    TableDatContents::from_snapshot(
174                        schema,
175                        keywords,
176                        column_keywords,
177                        nrrow,
178                        &dm_type_name,
179                        &dm_data,
180                        big_endian,
181                    )
182                };
183                apply_physical_dm_descriptor_group(
184                    &mut table_dat,
185                    &dm_type_name,
186                    group_name,
187                    virtual_columns,
188                );
189                control.publish(&table_dat)?;
190            }
191            DataManagerKind::IncrementalStMan => {
192                dm_type_name = "IncrementalStMan".to_string();
193                let table_dat_tmp = TableDatContents::from_snapshot(
194                    schema,
195                    keywords,
196                    column_keywords,
197                    nrrow,
198                    "IncrementalStMan",
199                    &[],
200                    big_endian,
201                );
202                dm_data = write_ism_file(
203                    &data_path,
204                    &table_dat_tmp.table_desc.columns,
205                    rows,
206                    big_endian,
207                )?;
208                let mut table_dat = TableDatContents::from_snapshot(
209                    schema,
210                    keywords,
211                    column_keywords,
212                    nrrow,
213                    &dm_type_name,
214                    &dm_data,
215                    big_endian,
216                );
217                apply_physical_dm_descriptor_group(
218                    &mut table_dat,
219                    &dm_type_name,
220                    dm_group_name.unwrap_or(&dm_type_name),
221                    virtual_columns,
222                );
223                control.publish(&table_dat)?;
224            }
225            DataManagerKind::TiledColumnStMan
226            | DataManagerKind::TiledShapeStMan
227            | DataManagerKind::TiledCellStMan
228            | DataManagerKind::TiledDataStMan => {
229                dm_type_name = match dm_kind {
230                    DataManagerKind::TiledColumnStMan => "TiledColumnStMan",
231                    DataManagerKind::TiledShapeStMan => "TiledShapeStMan",
232                    DataManagerKind::TiledCellStMan => "TiledCellStMan",
233                    DataManagerKind::TiledDataStMan => "TiledDataStMan",
234                    _ => unreachable!(),
235                }
236                .to_string();
237                dm_data = Vec::new();
238                let mut table_dat = TableDatContents::from_snapshot(
239                    schema,
240                    keywords,
241                    column_keywords,
242                    nrrow,
243                    &dm_type_name,
244                    &dm_data,
245                    big_endian,
246                );
247                let default_dm_group_name = table_dat
248                    .table_desc
249                    .columns
250                    .first()
251                    .map(|c| c.col_name.clone())
252                    .unwrap_or_default();
253                let dm_group_name = dm_group_name.unwrap_or(&default_dm_group_name);
254                for desc in &mut table_dat.table_desc.columns {
255                    desc.data_manager_type = dm_type_name.clone();
256                    desc.data_manager_group = dm_group_name.to_string();
257                }
258                control.publish(&table_dat)?;
259                let first_col_name = table_dat
260                    .table_desc
261                    .columns
262                    .first()
263                    .map(|c| c.col_name.as_str())
264                    .unwrap_or("");
265                tiled_stman::save_tiled_columns(
266                    table_path,
267                    0,
268                    &dm_type_name,
269                    &table_dat.table_desc.columns,
270                    rows,
271                    big_endian,
272                    tile_shape,
273                    first_col_name,
274                    &table_dat.table_desc.private_keywords,
275                )?;
276            }
277        }
278
279        let info_path = table_path.join(TABLE_INFO_FILE);
280        fs::write(&info_path, table_info.to_string())?;
281        Ok(())
282    }
283}

285impl CompositeStorage {
286    /// Rewrite the keywords in `table_path`'s `table.dat` and its
287    /// `table.info`, keeping the data-manager layout on disk. The snapshot's
288    /// `row_count` is the count the table's sync data publishes, which the
289    /// file's row counts are raised to (#880); its rows are not written.
290    pub(crate) fn save_metadata_only(
291        &self,
292        control: StagedTableDat<'_>,
293        snapshot: &StorageSnapshot,
294    ) -> Result<(), StorageError> {
295        let table_path = control.table_path();
296        let control_path = table_path.join(TABLE_CONTROL_FILE);
297
298        match read_table_dat_dispatch(&control_path)? {
299            TableDatResult::Plain(mut table_dat) => {
300                table_dat.raise_row_count(snapshot.row_count as u64);
301                table_dat.table_desc.table_keywords = snapshot.keywords.clone();
302                for column in &mut table_dat.table_desc.columns {
303                    column.keywords = snapshot
304                        .column_keywords
305                        .get(&column.col_name)
306                        .cloned()
307                        .unwrap_or_default();
308                }
309                control.publish(&table_dat)?;
310                let info_path = table_path.join(TABLE_INFO_FILE);
311                fs::write(&info_path, snapshot.table_info.to_string())?;
312                Ok(())
313            }
314            TableDatResult::Ref(_) | TableDatResult::Concat(_) => {
315                Err(StorageError::FormatMismatch(
316                    "metadata-only save is only supported for PlainTable storage".to_string(),
317                ))
318            }
319        }
320    }
321}

325    ///
326    /// Columns listed in `bindings` use their specified DM; all other columns
327    /// use the default DM from `dm_kind`. Each DM group writes its own data
328    /// file (`table.f0`, `table.f1`, ...) and gets a separate entry in `table.dat`.
329    #[allow(clippy::too_many_arguments)]
330    pub(crate) fn save_with_bindings_borrowed(

367    #[allow(clippy::too_many_arguments)]
368    #[expect(
369        clippy::too_many_lines,
370        reason = "predates the function-length limit; #705"
371    )]
372    pub(crate) fn save_with_bindings_and_column_overrides_borrowed(
373        &self,
374        control: StagedTableDat<'_>,
375        rows: &[RecordValue],
376        undefined_cells: &[HashSet<String>],
377        keywords: &RecordValue,
378        column_keywords: &HashMap<String, RecordValue>,
379        schema: Option<&TableSchema>,
380        table_info: &TableInfo,
381        virtual_columns: &HashSet<String>,
382        virtual_bindings: &[virtual_engine::VirtualColumnBinding],
383        default_dm: crate::table::DataManagerKind,
384        default_dm_group_name: Option<&str>,
385        big_endian: bool,
386        default_tile_shape: Option<&[usize]>,
387        bindings: &std::collections::HashMap<String, crate::table::ColumnBinding>,
388        column_overrides: &ColumnOverrides,
389    ) -> Result<(), StorageError> {
390        use crate::table::DataManagerKind;
391
392        let table_path = control.table_path();
393
394        let mut profiler = StorageProfiler::start(format!(
395            "CompositeStorage::save_with_bindings path={}",
396            table_path.display()
397        ));
398
399        let schema = schema.ok_or_else(|| {
400            StorageError::FormatMismatch("cannot save without schema".to_string())
401        })?;
402        // A persisted projection may already omit every undefined cell
403        // from its records. Only copy records if a masked value must be removed.
404        let rows_need_filter = rows
405            .iter()
406            .zip(undefined_cells)
407            .any(|(row, undefined)| undefined.iter().any(|column| row.get(column).is_some()));
408        let filtered_rows_storage =
409            rows_need_filter.then(|| filter_rows_for_save(rows, undefined_cells));
410        let filtered_rows: &[RecordValue] = filtered_rows_storage.as_deref().unwrap_or(rows);
411        // These positions are hints. The indexed writers verify each field
412        // name and look it up when later rows have a different layout.
413        let row_field_positions = filtered_rows.first().map(|row| {
414            row.fields()
415                .iter()
416                .enumerate()
417                .map(|(idx, field)| (field.name.clone(), idx))
418                .collect::<HashMap<_, _>>()
419        });
420        let effective_row_count = column_overrides.row_count().unwrap_or(filtered_rows.len());
421        let nrrow = effective_row_count as u64;
422        let has_virtual = !virtual_bindings.is_empty();
423        if let Some(profiler) = profiler.as_mut() {
424            profiler.mark_with_detail(
425                "prepare_rows",
426                Some(format!(
427                    "rows={} columns={} bindings={} virtual={}",
428                    nrrow,
429                    schema.columns().len(),
430                    bindings.len(),
431                    has_virtual
432                )),
433            );
434        }
435
436        // Group columns by DM kind. Columns in `bindings` get their own DM;
437        // everything else goes into the default DM.
438        struct DmGroup {
439            dm_kind: DataManagerKind,
440            dm_type_name: String,
441            dm_group_name: String,
442            seq_nr: u32,
443            col_names: Vec<String>,
444            tile_shape: Option<Vec<usize>>,
445        }
446
447        let mut groups: Vec<DmGroup> = Vec::new();
448
449        // Default DM group (seq_nr 0).
450        let default_type_name = match default_dm {
451            DataManagerKind::StManAipsIO => "StManAipsIO",
452            DataManagerKind::StandardStMan => "StandardStMan",
453            DataManagerKind::IncrementalStMan => "IncrementalStMan",
454            DataManagerKind::TiledColumnStMan => "TiledColumnStMan",
455            DataManagerKind::TiledShapeStMan => "TiledShapeStMan",
456            DataManagerKind::TiledCellStMan => "TiledCellStMan",
457            DataManagerKind::TiledDataStMan => "TiledDataStMan",
458        };
459        groups.push(DmGroup {
460            dm_kind: default_dm,
461            dm_type_name: default_type_name.to_string(),
462            dm_group_name: default_dm_group_name
463                .unwrap_or(default_type_name)
464                .to_string(),
465            seq_nr: 0,
466            col_names: Vec::new(),
467            tile_shape: default_tile_shape.map(|s| s.to_vec()),
468        });
469
470        // Collect virtual column names for filtering.
471        let virtual_col_names: HashSet<&str> = virtual_columns.iter().map(|s| s.as_str()).collect();
472
473        // Build additional DM groups from bindings, then assign columns.
474        let mut binding_seq_map: std::collections::HashMap<String, usize> =
475            std::collections::HashMap::new();
476        let mut next_seq = 1u32;
477
478        for col in schema.columns() {
479            let col_name = col.name();
480            if virtual_col_names.contains(col_name) {
481                continue; // Virtual columns don't get stored DM entries.
482            }
483            if let Some(binding) = bindings.get(col_name) {
484                let group_key = if matches!(
485                    binding.data_manager,
486                    DataManagerKind::TiledColumnStMan
487                        | DataManagerKind::TiledShapeStMan
488                        | DataManagerKind::TiledCellStMan
489                        | DataManagerKind::TiledDataStMan
490                ) {
491                    format!("{:?}:{col_name}", binding.data_manager)
492                } else {
493                    format!("{:?}", binding.data_manager)
494                };
495                let group_idx = if let Some(&idx) = binding_seq_map.get(&group_key) {
496                    idx
497                } else {
498                    let dm_type = match binding.data_manager {
499                        DataManagerKind::StManAipsIO => "StManAipsIO",
500                        DataManagerKind::StandardStMan => "StandardStMan",
501                        DataManagerKind::IncrementalStMan => "IncrementalStMan",
502                        DataManagerKind::TiledColumnStMan => "TiledColumnStMan",
503                        DataManagerKind::TiledShapeStMan => "TiledShapeStMan",
504                        DataManagerKind::TiledCellStMan => "TiledCellStMan",
505                        DataManagerKind::TiledDataStMan => "TiledDataStMan",
506                    };
507                    let idx = groups.len();
508                    groups.push(DmGroup {
509                        dm_kind: binding.data_manager,
510                        dm_type_name: dm_type.to_string(),
511                        dm_group_name: dm_type.to_string(),
512                        seq_nr: next_seq,
513                        col_names: Vec::new(),
514                        tile_shape: binding.tile_shape.clone(),
515                    });
516                    next_seq += 1;
517                    binding_seq_map.insert(group_key, idx);
518                    idx
519                };
520                groups[group_idx].col_names.push(col_name.to_string());
521            } else {
522                groups[0].col_names.push(col_name.to_string());
523            }
524        }
525        if !groups[0].col_names.is_empty() && default_dm == DataManagerKind::StandardStMan {
526            let first_name = groups[0].dm_group_name.clone();
527            for group in groups.iter_mut().skip(1).filter(|group| {
528                group.dm_kind == DataManagerKind::StandardStMan && group.dm_group_name == first_name
529            }) {
530                if control
531                    .existing_ssm_name(group.seq_nr, &first_name)
532                    .is_none()
533                {
534                    group.dm_group_name = format!("{first_name}_1");
535                }
536            }
537        }
538        if let Some(profiler) = profiler.as_mut() {
539            profiler.mark_with_detail(
540                "group_columns",
541                Some(format!("groups={} next_seq={next_seq}", groups.len())),
542            );
543        }
544
545        // Build a combined table.dat with all DM entries.
546        // For now, use from_snapshot for the first DM and add extra DM entries manually.
547        // This is a simplified approach: we write each group's data separately.
548
549        // First, build a complete table_dat with all DMs.
550        let mut dm_entries = Vec::new();
551        let mut col_dm_map: std::collections::HashMap<String, u32> =
552            std::collections::HashMap::new();
553
554        for group in &groups {
555            dm_entries.push(table_control::DataManagerEntry {
556                type_name: group.dm_type_name.clone(),
557                seq_nr: group.seq_nr,
558                data: Vec::new(), // Will be filled in for SSM/ISM.
559            });
560            for col_name in &group.col_names {
561                col_dm_map.insert(col_name.clone(), group.seq_nr);
562            }
563        }
564
565        // Build initial table_dat (using from_snapshot but overriding DM entries).
566        let mut table_dat = if has_virtual {
567            TableDatContents::from_snapshot_with_virtual(
568                schema,
569                keywords,
570                column_keywords,
571                nrrow,
572                &groups[0].dm_type_name,
573                &[],
574                big_endian,
575                virtual_bindings,
576                control.keyword_table_path(),
577            )
578        } else {
579            TableDatContents::from_snapshot(
580                schema,
581                keywords,
582                column_keywords,
583                nrrow,
584                &groups[0].dm_type_name,
585                &[],
586                big_endian,
587            )
588        };
589        if let Some(profiler) = profiler.as_mut() {
590            profiler.mark("build_table_dat");
591        }
592
593        // Override DM entries with our multi-DM setup.
594        // Preserve any virtual DM entries that from_snapshot_with_virtual added.
595        let virtual_dm_entries: Vec<_> = table_dat
596            .column_set
597            .data_managers
598            .iter()
599            .filter(|e| e.seq_nr > 0 && is_virtual_engine(&e.type_name))
600            .cloned()
601            .collect();
602
603        // Update each PlainColumnEntry's dm_seq_nr.
604        for pc in &mut table_dat.column_set.columns {
605            if let Some(&seq) = col_dm_map.get(&pc.original_name) {
606                pc.dm_seq_nr = seq;
607            }
608        }
609
610        // Override DM entries, filtering out empty groups that have no
611        // columns and no data file to avoid "missing table.f<N>" errors.
612        let non_empty_groups: Vec<&DmGroup> =
613            groups.iter().filter(|g| !g.col_names.is_empty()).collect();
614        table_dat.column_set.data_managers = non_empty_groups
615            .iter()
616            .map(|g| table_control::DataManagerEntry {
617                type_name: g.dm_type_name.clone(),
618                seq_nr: g.seq_nr,
619                data: Vec::new(),
620            })
621            .collect();
622        let group_by_seq: HashMap<u32, &DmGroup> = non_empty_groups
623            .iter()
624            .map(|group| (group.seq_nr, *group))
625            .collect();
626        for desc in &mut table_dat.table_desc.columns {
627            if let Some(seq) = col_dm_map.get(&desc.col_name)
628                && let Some(group) = group_by_seq.get(seq)
629            {
630                desc.data_manager_type = group.dm_type_name.clone();
631                desc.data_manager_group = group.dm_group_name.clone();
632            }
633        }
634        table_dat
635            .column_set
636            .data_managers
637            .extend(virtual_dm_entries);
638        table_dat.column_set.seq_count = table_dat
639            .column_set
640            .data_managers
641            .iter()
642            .map(|e| e.seq_nr + 1)
643            .max()
644            .unwrap_or(1);
645        if let Some(profiler) = profiler.as_mut() {
646            profiler.mark_with_detail(
647                "finalize_table_dat",
648                Some(format!(
649                    "non_empty_groups={} seq_count={}",
650                    non_empty_groups.len(),
651                    table_dat.column_set.seq_count
652                )),
653            );
654        }
655
656        // Write data files per group.
657        for group in &groups {
658            if group.col_names.is_empty() {
659                continue;
660            }
661            let data_path = table_path.join(format!("{}{}", TABLE_DATA_FILE_PREFIX, group.seq_nr));
662
663            let group_col_set: HashSet<&str> = group.col_names.iter().map(|s| s.as_str()).collect();
664            let group_col_descs: Vec<_> = table_dat
665                .table_desc
666                .columns
667                .iter()
668                .filter(|c| group_col_set.contains(c.col_name.as_str()))
669                .cloned()
670                .collect();
671            let group_col_indices = row_field_positions.as_ref().and_then(|positions| {
672                let indices: Vec<usize> = group_col_descs
673                    .iter()
674                    .filter_map(|desc| positions.get(&desc.col_name).copied())
675                    .collect();
676                (indices.len() == group_col_descs.len()).then_some(indices)
677            });
678            // A group holding a generated array column is written from its
679            // sources, or refused before any of its rows are projected.
680            let generated_array_sources = generated_array_group_sources(
681                group.dm_kind,
682                &group_col_descs,
683                column_overrides,
684                effective_row_count,
685            )?;
686            let deferred_group = group_col_descs.iter().all(|desc| {
687                matches!(
688                    column_overrides.get(&desc.col_name),
689                    Some(ColumnOverride::Deferred)
690                )
691            });
692            let use_borrowed_tiled_values = matches!(
693                group.dm_kind,
694                DataManagerKind::TiledColumnStMan
695                    | DataManagerKind::TiledShapeStMan
696                    | DataManagerKind::TiledCellStMan
697                    | DataManagerKind::TiledDataStMan
698            ) && group_col_descs.len() == 1;
699            let override_columns: Vec<_> = group_col_descs
700                .iter()
701                .filter_map(|desc| {
702                    column_overrides
703                        .contains_key(&desc.col_name)
704                        .then_some(desc.col_name.as_str())
705                })
706                .collect();
707            let use_direct_indexed_rows = matches!(
708                group.dm_kind,
709                DataManagerKind::StandardStMan | DataManagerKind::IncrementalStMan
710            ) && filtered_rows.len() == effective_row_count
711                && group_col_indices.is_some()
712                && override_columns.is_empty();
713            let group_col_index = group_col_indices
714                .as_ref()
715                .and_then(|indices| (indices.len() == 1).then_some(indices[0]));
716            let group_values = if use_borrowed_tiled_values {
717                match column_overrides.get(&group_col_descs[0].col_name) {
718                    Some(ColumnOverride::Values(override_values)) => {
719                        Some(override_values.iter().map(|value| value.as_ref()).collect())
720                    }
721                    Some(ColumnOverride::GeneratedScalar(_)) => {
722                        return Err(StorageError::FormatMismatch(format!(
723                            "tiled column override {} cannot be generated scalar",
724                            group_col_descs[0].col_name
725                        )));
726                    }
727                    Some(ColumnOverride::GeneratedArray(_)) => {
728                        return Err(StorageError::FormatMismatch(format!(
729                            "tiled column override {} cannot be generated array",
730                            group_col_descs[0].col_name
731                        )));
732                    }
733                    Some(ColumnOverride::StreamedScalar(_)) => {
734                        return Err(StorageError::FormatMismatch(format!(
735                            "tiled column override {} cannot be streamed scalar",
736                            group_col_descs[0].col_name
737                        )));
738                    }
739                    Some(ColumnOverride::Deferred) => None,
740                    None => {
741                        if filtered_rows.len() != effective_row_count {
742                            return Err(StorageError::FormatMismatch(format!(
743                                "column {} has no row values or override for {effective_row_count} generated rows",
744                                group_col_descs[0].col_name
745                            )));
746                        }
747                        Some(project_column_values_for_group(
748                            filtered_rows,
749                            &group_col_descs[0].col_name,
750                            group_col_index,
751                        ))
752                    }
753                }
754            } else {
755                None
756            };
757            let override_sources = match generated_array_sources {
758                Some(sources) => Some(sources),
759                None if matches!(
760                    group.dm_kind,
761                    DataManagerKind::StandardStMan | DataManagerKind::IncrementalStMan
762                ) =>
763                {
764                    override_columns_for_group(
765                        group.dm_kind,
766                        &group_col_descs,
767                        column_overrides,
768                        effective_row_count,
769                        filtered_rows,
770                    )?
771                }
772                None => None,
773            };
774            if deferred_group {
775                if let Some(profiler) = profiler.as_mut() {
776                    profiler.mark_with_detail(
777                        "group_projection",
778                        Some(format!(
779                            "seq={} dm={} cols={} rows={} indexed_projection={} mode=deferred",
780                            group.seq_nr,
781                            group.dm_type_name,
782                            group_col_descs.len(),
783                            effective_row_count,
784                            group_col_indices.is_some()
785                        )),
786                    );
787                    profiler.mark_with_detail(
788                        "group_save",
789                        Some(format!(
790                            "seq={} dm={} cols={} rows={} deferred=true",
791                            group.seq_nr,
792                            group.dm_type_name,
793                            group_col_descs.len(),
794                            effective_row_count
795                        )),
796                    );
797                }
798                continue;
799            }
800            let group_rows =
801                if group_values.is_none() && !use_direct_indexed_rows && override_sources.is_none()
802                {
803                    Some(project_rows_for_group(
804                        filtered_rows,
805                        effective_row_count,
806                        &group_col_descs,
807                        group_col_indices.as_deref(),
808                        column_overrides,
809                    ))
810                } else {
811                    None
812                };
813            if let Some(profiler) = profiler.as_mut() {
814                profiler.mark_with_detail(
815                    "group_projection",
816                    Some(format!(
817                        "seq={} dm={} cols={} rows={} indexed_projection={} mode={}",
818                        group.seq_nr,
819                        group.dm_type_name,
820                        group_col_descs.len(),
821                        effective_row_count,
822                        group_col_indices.is_some(),
823                        if group_values.is_some() {
824                            "borrowed_values"
825                        } else if use_direct_indexed_rows {
826                            "direct_rows"
827                        } else if override_sources.is_some() {
828                            "override_sources"
829                        } else {
830                            "projected_rows"
831                        }
832                    )),
833                );
834            }
835
836            match group.dm_kind {
837                DataManagerKind::StManAipsIO => {
838                    write_stman_file(
839                        &data_path,
840                        group.seq_nr,
841                        &group_col_descs,
842                        group_rows.as_ref().expect("row projection for StManAipsIO"),
843                        ByteOrder::BigEndian,
844                    )?;
845                }
846                DataManagerKind::StandardStMan => {
847                    let existing_name =
848                        control.existing_ssm_name(group.seq_nr, &group.dm_group_name);
849                    let name = existing_name.as_deref().unwrap_or(&group.dm_group_name);
850                    let dm_data = if let Some(override_sources) = override_sources.as_ref() {
851                        write_ssm_file_column_sources(
852                            &data_path,
853                            &group_col_descs,
854                            override_sources,
855                            big_endian,
856                            name,
857                        )?
858                    } else if let Some(group_col_indices) = group_col_indices.as_ref() {
859                        write_ssm_file_indexed(
860                            &data_path,
861                            &group_col_descs,
862                            filtered_rows,
863                            group_col_indices,
864                            big_endian,
865                            name,
866                        )?
867                    } else {
868                        write_ssm_file_with_options(
869                            &data_path,
870                            &group_col_descs,
871                            group_rows
872                                .as_ref()
873                                .expect("row projection for StandardStMan"),
874                            big_endian,
875                            false,
876                            name,
877                        )?
878                    };
879                    // Update the DM blob in the table_dat.
880                    if let Some(entry) = table_dat
881                        .column_set
882                        .data_managers
883                        .iter_mut()
884                        .find(|e| e.seq_nr == group.seq_nr)
885                    {
886                        entry.data = dm_data;
887                    }
888                }
889                DataManagerKind::IncrementalStMan => {
890                    let dm_data = if let Some(override_sources) = override_sources.as_ref() {
891                        write_ism_file_column_sources(
892                            &data_path,
893                            &group_col_descs,
894                            override_sources,
895                            big_endian,
896                        )?
897                    } else if let Some(group_col_indices) = group_col_indices.as_ref() {
898                        write_ism_file_indexed(
899                            &data_path,
900                            &group_col_descs,
901                            filtered_rows,
902                            group_col_indices,
903                            big_endian,
904                        )?
905                    } else {
906                        write_ism_file(
907                            &data_path,
908                            &group_col_descs,
909                            group_rows
910                                .as_ref()
911                                .expect("row projection for IncrementalStMan"),
912                            big_endian,
913                        )?
914                    };
915                    if let Some(entry) = table_dat
916                        .column_set
917                        .data_managers
918                        .iter_mut()
919                        .find(|e| e.seq_nr == group.seq_nr)
920                    {
921                        entry.data = dm_data;
922                    }
923                }
924                DataManagerKind::TiledColumnStMan
925                | DataManagerKind::TiledShapeStMan
926                | DataManagerKind::TiledCellStMan
927                | DataManagerKind::TiledDataStMan => {
928                    // Use the first column name as the hypercolumn/DM name.
929                    let first_col = group.col_names.first().map(|s| s.as_str()).unwrap_or("");
930                    if let Some(group_values) = group_values.as_ref() {
931                        tiled_stman::save_tiled_single_column_values(
932                            table_path,
933                            group.seq_nr,
934                            &group_col_descs[0],
935                            group_values,
936                            tiled_stman::SingleColumnTiledSaveOptions {
937                                dm_type_name: &group.dm_type_name,
938                                big_endian,
939                                default_tile_shape: group.tile_shape.as_deref(),
940                                dm_name: first_col,
941                            },
942                        )?;
943                    } else {
944                        tiled_stman::save_tiled_columns(
945                            table_path,
946                            group.seq_nr,
947                            &group.dm_type_name,
948                            &group_col_descs,
949                            group_rows
950                                .as_ref()
951                                .expect("row projection for tiled storage"),
952                            big_endian,
953                            group.tile_shape.as_deref(),
954                            first_col,
955                            &table_dat.table_desc.private_keywords,
956                        )?;
957                    }
958                }
959            }
960            if let Some(profiler) = profiler.as_mut() {
961                profiler.mark_with_detail(
962                    "group_save",
963                    Some(format!(
964                        "seq={} dm={} cols={} rows={}",
965                        group.seq_nr,
966                        group.dm_type_name,
967                        group_col_descs.len(),
968                        effective_row_count
969                    )),
970                );
971            }
972        }
973
974        control.publish(&table_dat)?;
975        let info_path = table_path.join(TABLE_INFO_FILE);
976        fs::write(&info_path, table_info.to_string())?;
977        if let Some(profiler) = profiler.as_mut() {
978            profiler.mark("write_control_files");
979        }
980
981        Ok(())
982    }

985    #[allow(clippy::too_many_arguments)]
986    pub(crate) fn save_with_bindings(

1001            snapshot.schema.as_ref(),
1002            &snapshot.table_info,
1003            &snapshot.virtual_columns,
1004            &snapshot.virtual_bindings,
1005            default_dm,
1006            None,
1007            big_endian,
1008            default_tile_shape,
1009            bindings,
1010        )
1011    }
