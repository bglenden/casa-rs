2//! The image-plane gridder: total-power spectra spread over the pixels
3//! around their pointings and normalised by the weight they put there, as
4//! CASA's `SDGrid::put` (`ggridsd`) and `SDGrid::getImage` do.
5//!
6//! For each unflagged spectrum, at each data channel with a positive weight
7//! `w` that maps to an image channel, and each unflagged correlation that
8//! maps to an image plane, every pixel the kernel reaches
9//! ([`KernelTable::taps`]) with weight `k` accumulates
10//! - `w · value · k` in the data grid, and
11//! - `w · k` in the weight grid.
12//!
13//! The image is the data grid over the weight grid where the weight is
14//! positive, and zero elsewhere. A Stokes I image takes both parallel
15//! hands into its one plane (`FTMachine::matchPol`), so it is their mean,
16//! and its weight counts each spectrum's weight once per hand. A Stokes
17//! image flags a channel in every correlation when any is flagged
18//! (`FTMachine::setSpectralFlag`), so it never takes one hand alone; a
19//! pseudo-Stokes image takes each correlation's flags as they are
20//! ([`FlagMerge`]).
21//!
22//! Both grids are held whole, as CASA holds them, in single precision.
23//! Total-power data are real, so the grids are real where CASA's data grid
24//! is complex. The image channels are gridded in parallel on the worker
25//! team, each summing its pixels in CASA's order (spectrum, then data
26//! channel, then correlation), so a channel's sums are CASA's. The image
27//! and weight [`TotalPowerGrid::finish`] makes are those grids, so they keep
28//! the grids' share of the admission until they drop.
29

37/// The most bytes the gridder's per-pass tap scratch holds: the kernel's
38/// taps of the spectra gridded in one pass over the channels.
39const TAP_SCRATCH_BYTES: usize = 4 << 20;

41/// The shape of an image the gridder fills, in CASA's axis order.
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub struct GridShape {
44    /// Pixels along x.
45    pub nx: usize,
46    /// Pixels along y.
47    pub ny: usize,
48    /// Polarization planes.
49    pub planes: usize,
50    /// Channels.
51    pub channels: usize,
52}
53
54impl GridShape {
55    /// Pixels in one channel's plane of one polarization.
56    #[must_use]
57    pub const fn plane_pixels(self) -> usize {
58        self.nx * self.ny
59    }
60
61    /// Pixels in all of the image.
62    #[must_use]
63    pub const fn pixels(self) -> usize {
64        self.plane_pixels() * self.planes * self.channels
65    }
66}

68/// How a channel's flags combine across its correlations
69/// (`FTMachine::setSpectralFlag`).
70#[derive(Debug, Clone, Copy, PartialEq, Eq)]
71pub enum FlagMerge {
72    /// A Stokes image (`stokes='I'`, the default): a value flagged in any
73    /// correlation flags the channel in all of them.
74    AnyCorrelation,
75    /// A pseudo-Stokes image (`stokes` of `XX`, `YY`, `XXYY`, `RR`, `LL` or
76    /// `RRLL`): each correlation keeps its own flags.
77    PerCorrelation,
78}

80/// A block of total-power spectra to grid, as one `VisBuffer` hands them to
81/// `SDGrid::put`.
82#[derive(Debug, Clone, Copy)]
83pub struct SpectrumBlock<'a> {
84    /// `[spectrum]`: the zero-based fractional image pixel of the spectrum's
85    /// pointing, or `None` where it could not be placed (`getXYPos`).
86    pub pixels: &'a [Option<[f64; 2]>],
87    /// `[spectrum][channel][correlation]`: the spectra.
88    pub values: ArrayView3<'a, f32>,
89    /// `[spectrum][channel][correlation]`: whether a value is flagged.
90    pub flags: ArrayView3<'a, bool>,
91    /// `[spectrum]`: whether the whole spectrum is flagged (`FLAG_ROW`).
92    pub row_flags: &'a [bool],
93    /// `[spectrum][channel]`: the spectrum's weight
94    /// ([`crate::weights_from_rows`], [`crate::weights_from_spectra`]).
95    pub weights: ArrayView2<'a, f32>,
96    /// `[channel]`: the image channel each data channel is gridded into
97    /// (`chanMap`), or `None`.
98    pub channel_map: &'a [Option<usize>],
99    /// `[correlation]`: the image plane each correlation is gridded into
100    /// (`polMap`), or `None`.
101    pub correlation_map: &'a [Option<usize>],
102}
103
104impl SpectrumBlock<'_> {
105    fn check(&self, shape: GridShape) -> Result<(), SingleDishError> {
106        let (spectra, channels, correlations) = self.values.dim();
107        let agrees = self.flags.dim() == (spectra, channels, correlations)
108            && self.weights.dim() == (spectra, channels)
109            && self.pixels.len() == spectra
110            && self.row_flags.len() == spectra
111            && self.channel_map.len() == channels
112            && self.correlation_map.len() == correlations;
113        if !agrees {
114            return Err(SingleDishError::Shape(format!(
115                "a block of {spectra} spectra of {channels} channels and {correlations} \
116                 correlations needs flags, weights, pixels, row flags and maps of those \
117                 lengths"
118            )));
119        }
120        let outside = |map: &[Option<usize>], size: usize| map.iter().flatten().any(|&i| i >= size);
121        if outside(self.channel_map, shape.channels) || outside(self.correlation_map, shape.planes)
122        {
123            return Err(SingleDishError::Shape(format!(
124                "a channel or correlation maps outside an image of {} channels and {} planes",
125                shape.channels, shape.planes
126            )));
127        }
128        Ok(())
129    }
130}

132/// The gridded image and its weight.
133#[derive(Debug, Clone, PartialEq)]
134pub struct TotalPowerImage {
135    /// `[x, y, plane, channel]`: the image, zero where there is no weight.
136    pub image: ArrayD<f32>,
137    /// `[x, y, plane, channel]`: the weight each pixel accumulated
138    /// (`SDGrid::getWeightImage`).
139    pub weight: ArrayD<f32>,
140}
141
142impl TotalPowerImage {
143    /// Bytes the image and weight of `shape` hold.
144    #[must_use]
145    pub const fn bytes(shape: GridShape) -> u64 {
146        (2 * shape.pixels() * size_of::<f32>()) as u64
147    }
148}

150/// The data and weight grids of an image being gridded.
151#[derive(Debug)]
152pub struct TotalPowerGrid {
153    shape: GridShape,
154    kernel: KernelTable,
155    flags: FlagMerge,
156    /// `[channel][plane][y][x]`.
157    data: Vec<f32>,
158    /// `[channel][plane][y][x]`.
159    weight: Vec<f32>,
160    /// The admission of [`Self::bytes`].
161    _reservation: Reservation,
162}
163
164impl TotalPowerGrid {
165    /// Bytes a grid of `shape` holds with `kernel`, known before it is
166    /// formed, while it grids blocks of at most `data_channels` channels:
167    /// the two grids, the kernel, the taps of one pass's spectra, and each
168    /// pass's per-channel bookkeeping.
169    #[must_use]
170    pub fn bytes(shape: GridShape, kernel: &KernelTable, data_channels: usize) -> u64 {
171        let pass = shape.channels
172            * (size_of::<ChannelGrids<'_>>() + size_of::<Result<(), SingleDishError>>());
173        TotalPowerImage::bytes(shape)
174            + (Taps::bytes(kernel) + Sources::bytes(shape.channels, data_channels) + pass) as u64
175            + kernel.bytes()
176    }
177
178    /// An empty grid of `shape`, merging flags as `flags` says, holding
179    /// `reservation`, its admission of [`Self::bytes`].
180    ///
181    /// # Errors
182    ///
183    /// An image too large to index.
184    pub fn new(
185        shape: GridShape,
186        kernel: KernelTable,
187        flags: FlagMerge,
188        reservation: Reservation,
189    ) -> Result<Self, SingleDishError> {
190        if shape.plane_pixels() > u32::MAX as usize {
191            return Err(SingleDishError::Shape(format!(
192                "an image plane of {} × {} pixels is too large to grid",
193                shape.nx, shape.ny
194            )));
195        }
196        Ok(Self {
197            shape,
198            kernel,
199            flags,
200            data: vec![0.0; shape.pixels()],
201            weight: vec![0.0; shape.pixels()],
202            _reservation: reservation,
203        })
204    }
205
206    /// The image's shape.
207    #[must_use]
208    pub const fn shape(&self) -> GridShape {
209        self.shape
210    }
211
212    /// Grid `block`, its image channels in parallel on `team`.
213    ///
214    /// # Errors
215    ///
216    /// A block whose arrays and maps disagree in shape, or that maps outside
217    /// the image.
218    pub fn grid(
219        &mut self,
220        block: &SpectrumBlock<'_>,
221        team: &WorkerTeam,
222    ) -> Result<(), SingleDishError> {
223        block.check(self.shape)?;
224        let sources = Sources::of(block.channel_map, self.shape.channels);
225        let per_channel = self.shape.planes * self.shape.plane_pixels();
226        let rows = tap_scratch_rows(&self.kernel);
227        let mut taps = Taps::new(&self.kernel);
228        let mut start = 0;
229        while start < block.pixels.len() {
230            let end = block.pixels.len().min(start + rows);
231            taps.fill(&self.kernel, block, start..end, self.shape);
232            let mut channels: Vec<ChannelGrids<'_>> = self
233                .data
234                .chunks_mut(per_channel)
235                .zip(self.weight.chunks_mut(per_channel))
236                .enumerate()
237                .map(|(channel, (data, weight))| ChannelGrids {
238                    data,
239                    weight,
240                    sources: sources.of_channel(channel),
241                })
242                .collect();
243            let plane_pixels = self.shape.plane_pixels();
244            let flags = self.flags;
245            team.for_each_mut(&mut channels, |_, grids| {
246                grids.accumulate(block, &taps, (plane_pixels, flags));
247                Ok::<_, SingleDishError>(())
248            })?;
249            start = end;
250        }
251        Ok(())
252    }
253
254    /// The image and its weight, as `tsdimaging` forms them: the data grid
255    /// over the weight grid where the weight is positive and zero
256    /// elsewhere (`SDGrid::getImage` unnormalised, then
257    /// `SIImageStore::divideResidualByWeightSD`), and the weight grid
258    /// (`SDGrid::getWeightImage`). `tsdimaging`'s `minweight` then masks
259    /// the image ([`crate::weight_mask_threshold`]).
260    ///
261    /// The image and weight are the grids, so they keep the grids'
262    /// [`TotalPowerImage::bytes`] of the admission; the kernel's and the
263    /// passes' bytes are released. A grid formed without an admission
264    /// ([`Reservation::none`]) hands nothing on.
265    #[must_use]
266    pub fn finish(self) -> Admitted<TotalPowerImage> {
267        let Self {
268            shape,
269            mut data,
270            weight,
271            _reservation: mut reservation,
272            ..
273        } = self;
274        reservation.retain(reservation.memory().min(TotalPowerImage::bytes(shape)));
275        for (value, &weight) in data.iter_mut().zip(&weight) {
276            *value = if weight > 0.0 { *value / weight } else { 0.0 };
277        }
278        let dims = IxDyn(&[shape.nx, shape.ny, shape.planes, shape.channels]).f();
279        let array = |values| ArrayD::from_shape_vec(dims.clone(), values).expect("grid shape");
280        Admitted {
281            value: TotalPowerImage {
282                image: array(data),
283                weight: array(weight),
284            },
285            reservation,
286        }
287    }
288}

290/// A tap of a pointing's kernel as an index into an image plane.
291#[derive(Debug, Clone, Copy)]
292struct PlaneTap {
293    pixel: u32,
294    weight: f32,
295}
296
297/// The taps of a run of spectra: each spectrum's, in order.
298#[derive(Debug)]
299struct Taps {
300    /// The taps, spectrum after spectrum.
301    taps: Vec<PlaneTap>,
302    /// `[spectrum in the run + 1]`: where each spectrum's taps start; a
303    /// spectrum not gridded has none.
304    starts: Vec<usize>,
305    /// The first spectrum of the run.
306    first: usize,
307    /// One spectrum's taps as the kernel gives them.
308    scratch: Vec<Tap>,
309}
310
311impl Taps {
312    /// Room for the taps of a pass's spectra, held at once so that filling
313    /// it never grows it.
314    fn new(kernel: &KernelTable) -> Self {
315        let rows = tap_scratch_rows(kernel);
316        let per_spectrum = taps_per_spectrum(kernel);
317        Self {
318            taps: Vec::with_capacity(rows * per_spectrum),
319            starts: Vec::with_capacity(rows + 1),
320            first: 0,
321            scratch: Vec::with_capacity(per_spectrum),
322        }
323    }
324
325    /// The bytes [`Self::new`] holds.
326    fn bytes(kernel: &KernelTable) -> usize {
327        let rows = tap_scratch_rows(kernel);
328        let per_spectrum = taps_per_spectrum(kernel);
329        rows * per_spectrum * size_of::<PlaneTap>()
330            + (rows + 1) * size_of::<usize>()
331            + per_spectrum * size_of::<Tap>()
332    }
333
334    /// The taps of `block`'s spectra `spectra`.
335    fn fill(
336        &mut self,
337        kernel: &KernelTable,
338        block: &SpectrumBlock<'_>,
339        spectra: std::ops::Range<usize>,
340        shape: GridShape,
341    ) {
342        self.taps.clear();
343        self.starts.clear();
344        self.first = spectra.start;
345        self.starts.push(0);
346        for spectrum in spectra {
347            if let (false, Some(pixel)) = (block.row_flags[spectrum], block.pixels[spectrum]) {
348                self.scratch.clear();
349                kernel.taps(pixel, (shape.nx, shape.ny), &mut self.scratch);
350                self.taps.extend(self.scratch.iter().map(|tap| PlaneTap {
351                    pixel: (tap.y * shape.nx + tap.x) as u32,
352                    weight: tap.weight,
353                }));
354            }
355            self.starts.push(self.taps.len());
356        }
357    }
358
359    /// The spectra of the run, each with its taps.
360    fn spectra(&self) -> impl Iterator<Item = (usize, &[PlaneTap])> {
361        self.starts
362            .windows(2)
363            .enumerate()
364            .map(|(i, range)| (self.first + i, &self.taps[range[0]..range[1]]))
365    }
366}

368/// The data channels each image channel takes, in order: a counting sort
369/// of `chanMap`.
370struct Sources {
371    /// `[image channel + 1]`: where each image channel's data channels
372    /// start in `order`.
373    starts: Vec<usize>,
374    /// The mapped data channels, by image channel.
375    order: Vec<usize>,
376}
377
378impl Sources {
379    fn of(channel_map: &[Option<usize>], channels: usize) -> Self {
380        let mut starts = vec![0; channels + 1];
381        for &target in channel_map.iter().flatten() {
382            starts[target + 1] += 1;
383        }
384        for channel in 0..channels {
385            starts[channel + 1] += starts[channel];
386        }
387        let mut next = starts.clone();
388        let mut order = vec![0; starts[channels]];
389        for (channel, target) in channel_map.iter().enumerate() {
390            if let Some(target) = *target {
391                order[next[target]] = channel;
392                next[target] += 1;
393            }
394        }
395        Self { starts, order }
396    }
397
398    /// The bytes [`Self::of`] holds at most, with its scratch.
399    const fn bytes(channels: usize, data_channels: usize) -> usize {
400        (2 * (channels + 1) + data_channels) * size_of::<usize>()
401    }
402
403    fn of_channel(&self, channel: usize) -> &[usize] {
404        &self.order[self.starts[channel]..self.starts[channel + 1]]
405    }
406}

408/// One image channel's grids, and the data channels gridded into it.
409struct ChannelGrids<'a> {
410    data: &'a mut [f32],
411    weight: &'a mut [f32],
412    sources: &'a [usize],
413}
414
415impl ChannelGrids<'_> {
416    /// `ggridsd` for this channel, over the spectra of `taps`.
417    fn accumulate(
418        &mut self,
419        block: &SpectrumBlock<'_>,
420        taps: &Taps,
421        (plane_pixels, flags): (usize, FlagMerge),
422    ) {
423        if self.sources.is_empty() {
424            return;
425        }
426        for (spectrum, spectrum_taps) in taps.spectra() {
427            if spectrum_taps.is_empty() {
428                continue;
429            }
430            for &channel in self.sources {
431                let weight = block.weights[[spectrum, channel]];
432                // `ggridsd` grids where `weight > 0`, which a NaN is not.
433                if weight.is_nan() || weight <= 0.0 {
434                    continue;
435                }
436                let merged = || {
437                    block
438                        .flags
439                        .slice(ndarray::s![spectrum, channel, ..])
440                        .iter()
441                        .any(|&f| f)
442                };
443                if flags == FlagMerge::AnyCorrelation && merged() {
444                    continue;
445                }
446                for (correlation, plane) in block.correlation_map.iter().enumerate() {
447                    let Some(plane) = plane else { continue };
448                    if block.flags[[spectrum, channel, correlation]] {
449                        continue;
450                    }
451                    let value = weight * block.values[[spectrum, channel, correlation]];
452                    let offset = plane * plane_pixels;
453                    let data = &mut self.data[offset..offset + plane_pixels];
454                    let grid_weight = &mut self.weight[offset..offset + plane_pixels];
455                    for tap in spectrum_taps {
456                        let pixel = tap.pixel as usize;
457                        data[pixel] += value * tap.weight;
458                        grid_weight[pixel] += weight * tap.weight;
459                    }
460                }
461            }
462        }
463    }
464}

466/// The most taps the kernel gives one spectrum.
467fn taps_per_spectrum(kernel: &KernelTable) -> usize {
468    (2 * kernel.support() + 1).pow(2)
469}
470
471/// Spectra whose taps one pass holds: as many as fit
472/// [`TAP_SCRATCH_BYTES`], at least one.
473fn tap_scratch_rows(kernel: &KernelTable) -> usize {
474    (TAP_SCRATCH_BYTES / (taps_per_spectrum(kernel) * size_of::<PlaneTap>())).max(1)
475}
