14/// CASA's `uvtaper` (`synthesisimager::setweighting`, then
15/// `VisImagingWeight::setFilter`): `[bmaj]`, `[bmaj, bmin]` or
16/// `[bmaj, bmin, bpa]`, with `bmin = bmaj` and `bpa = 0` when not given,
17/// and no taper when the list or its first entry is empty. In wavelengths
18/// (`lambda`, `klambda`, …) the axes are the taper's uv-plane HWHM; as
19/// angles they are its image-plane FWHM θ, whose uv-plane HWHM is
20/// 2 ln 2 / (π θ). CASA reads an axis with no unit as radians and tapers
21/// every weight to zero; casa-rs refuses it (`docs/CASA (C++) bugs.md`).
22pub(super) fn uv_taper<'de, D: Deserializer<'de>>(
23    deserializer: D,
24) -> Result<Option<UvTaper>, D::Error> {
25    let entries = match Value::deserialize(deserializer)? {
26        Value::Array(items) => items,
27        value => vec![value],
28    };
29    let texts = entries
30        .iter()
31        .map(|entry| {
32            entry
33                .as_str()
34                .map(str::trim)
35                .ok_or_else(|| unexpected(entry, "a quantity"))
36        })
37        .collect::<Result<Vec<_>, D::Error>>()?;
38    let taper = match texts.as_slice() {
39        [] | ["", ..] => return Ok(None),
40        [bmaj] => uv_taper_axes(bmaj, bmaj, None),
41        [bmaj, bmin] => uv_taper_axes(bmaj, bmin, None),
42        [bmaj, bmin, bpa] => uv_taper_axes(bmaj, bmin, Some(bpa)),
43        _ => Err("uvtaper takes at most [bmaj, bmin, bpa]".to_string()),
44    };
45    taper.map(Some).map_err(D::Error::custom)
46}

48/// One `uvtaper` axis: its uv-plane HWHM in wavelengths, and whether it was
49/// given as an angle.
50fn uv_taper_axis(text: &str) -> Result<(f64, bool), String> {
51    let refused = || {
52        format!(
53            "uvtaper {text:?}: give an angle (the image-plane FWHM) or wavelengths (the uv-plane \
54             HWHM, lambda or klambda)"
55        )
56    };
57    let quantity: Quantity = text.parse().map_err(|_| refused())?;
58    let unit = |name: &str| Unit::from_str(name).expect("a casacore unit");
59    let hwhm = if let Ok(wavelengths) = quantity.get_value_in(&unit("lambda")) {
60        (wavelengths, false)
61    } else if let Ok(fwhm) = quantity.get_value_in(&unit("rad")) {
62        (
63            2.0 * std::f64::consts::LN_2 / (std::f64::consts::PI * fwhm),
64            true,
65        )
66    } else {
67        return Err(refused());
68    };
69    if quantity.unit().name().is_empty() || !(hwhm.0.is_finite() && hwhm.0 > 0.0) {
70        return Err(refused());
71    }
72    Ok(hwhm)
73}
74
75fn uv_taper_axes(bmaj: &str, bmin: &str, bpa: Option<&str>) -> Result<UvTaper, String> {
76    let (major, major_angle) = uv_taper_axis(bmaj)?;
77    let (minor, minor_angle) = uv_taper_axis(bmin)?;
78    if major_angle != minor_angle {
79        return Err(format!(
80            "uvtaper {bmaj:?}, {bmin:?}: give both axes as angles or both in wavelengths"
81        ));
82    }
83    let position_angle_rad = match bpa {
84        None => 0.0,
85        Some(text) => text
86            .parse::<Quantity>()
87            .ok()
88            .filter(|angle| !angle.unit().name().is_empty())
89            .and_then(|angle| angle.get_value_in(&Unit::from_str("rad").ok()?).ok())
90            .ok_or_else(|| format!("uvtaper position angle {text:?} needs an angle unit"))?,
91    };
92    Ok(UvTaper::new(major, minor, position_angle_rad))
93}
