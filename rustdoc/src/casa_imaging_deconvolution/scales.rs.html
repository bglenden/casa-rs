11/// Heap bytes of one `f64` plane of `shape`.
12pub(crate) const fn plane_bytes(shape: PlaneShape) -> u64 {
13    (shape.len() * size_of::<f64>()) as u64
14}
15
16/// Heap bytes of one half spectrum of `shape` ([`PlaneFft::forward`]).
17pub(crate) const fn spectrum_bytes(shape: PlaneShape) -> u64 {
18    match RealFft2::<f64>::spectrum_len([shape.nx, shape.ny]) {
19        Some(len) => (len * size_of::<Complex64>()) as u64,
20        None => u64::MAX,
21    }
22}
23
24/// Heap bytes planning the half-spectrum transform of `shape` holds while
25/// FFTW plans it ([`casa_fft::planning_bytes`]).
26pub(crate) const fn planning_bytes(shape: PlaneShape) -> u64 {
27    match RealFft2::<f64>::spectrum_len([shape.nx, shape.ny]) {
28        Some(len) => casa_fft::planning_bytes::<f64>(len),
29        None => u64::MAX,
30    }
31}
32
33/// Heap bytes one [`ScaleBank::convolve`] holds while it runs: the product
34/// spectrum, then the plane it returns.
35pub(crate) const fn convolve_bytes(shape: PlaneShape) -> u64 {
36    spectrum_bytes(shape) + plane_bytes(shape)
37}

39/// Real two-dimensional transforms of one plane shape, in double precision
40/// on one thread.
41pub(crate) struct PlaneFft {
42    shape: PlaneShape,
43    fft: RealFft2<f64>,
44}
45
46impl PlaneFft {
47    pub(crate) fn new(shape: PlaneShape) -> Result<Self, Error> {
48        Ok(Self {
49            shape,
50            fft: RealFft2::with_threads([shape.nx, shape.ny], 1)?,
51        })
52    }
53
54    /// The half spectrum of `plane`.
55    pub(crate) fn forward(&mut self, plane: &[f64]) -> Result<Vec<Complex64>, Error> {
56        let mut storage = vec![Complex64::default(); self.fft.storage_len()];
57        let stride = self.fft.real_row_stride();
58        let real: &mut [f64] = bytemuck::cast_slice_mut(&mut storage);
59        for x in 0..self.shape.nx {
60            real[x * stride..][..self.shape.ny]
61                .copy_from_slice(&plane[x * self.shape.ny..][..self.shape.ny]);
62        }
63        self.fft.forward(&mut storage)?;
64        Ok(storage)
65    }
66
67    /// The plane of a half spectrum, normalised by the pixel count.
68    pub(crate) fn inverse(&mut self, mut spectrum: Vec<Complex64>) -> Result<Vec<f64>, Error> {
69        self.fft.inverse(&mut spectrum)?;
70        let stride = self.fft.real_row_stride();
71        let scale = 1.0 / self.shape.len() as f64;
72        let real: &[f64] = bytemuck::cast_slice(&spectrum);
73        let mut plane = vec![0.0; self.shape.len()];
74        for x in 0..self.shape.nx {
75            for (value, &sample) in plane[x * self.shape.ny..][..self.shape.ny]
76                .iter_mut()
77                .zip(&real[x * stride..][..self.shape.ny])
78            {
79                *value = sample * scale;
80            }
81        }
82        Ok(plane)
83    }
84}

86/// `MatrixCleaner::spheroidal`: the rational approximation to the
87/// prolate spheroidal function the scales are tapered by.
88#[must_use]
89pub fn spheroidal(nu: f64) -> f64 {
90    if nu <= 0.0 {
91        return 1.0;
92    }
93    if nu >= 1.0 {
94        return 0.0;
95    }
96    let (p, q, end) = if nu < 0.75 {
97        (
98            [
99                8.203343e-2,
100                -3.644705e-1,
101                6.278660e-1,
102                -5.335581e-1,
103                2.312756e-1,
104            ],
105            [1.0, 8.212018e-1, 2.078043e-1],
106            0.75,
107        )
108    } else {
109        (
110            [
111                4.028559e-3,
112                -3.697768e-2,
113                1.021332e-1,
114                -1.201436e-1,
115                6.412774e-2,
116            ],
117            [1.0, 9.599102e-1, 2.918724e-1],
118            1.0,
119        )
120    };
121    let delta = nu * nu - end * end;
122    let top = p
123        .iter()
124        .rev()
125        .fold(0.0, |sum, coefficient| sum * delta + coefficient);
126    let bottom = q
127        .iter()
128        .rev()
129        .fold(0.0, |sum, coefficient| sum * delta + coefficient);
130    if bottom == 0.0 { 0.0 } else { top / bottom }
131}

133/// One scale function: `(1 − r²)·spheroidal(r)` inside radius `size`
134/// pixels, normalised to unit sum, as its nonzero offsets from the centre.
135#[derive(Clone, Debug, PartialEq)]
136pub(crate) struct ScaleFunction {
137    /// The scale size in pixels.
138    pub(crate) size: f64,
139    /// `(dx, dy, value)` of every nonzero sample.
140    pub(crate) samples: Vec<(isize, isize, f64)>,
141}
142
143impl ScaleFunction {
144    /// `MatrixCleaner::makeScale` centred at `centre` of `shape`: a delta for
145    /// size 0, otherwise the tapered disc sampled within the plane.
146    pub(crate) fn new(size: f64, shape: PlaneShape) -> Self {
147        let [cx, cy] = shape.centre();
148        if size == 0.0 {
149            return Self {
150                size,
151                samples: vec![(0, 0, 1.0)],
152            };
153        }
154        let low = |centre: usize| ((centre as f64 - size) as isize).max(0);
155        let high = |centre: usize, extent: usize| {
156            ((centre as f64 + size) as isize).min(extent as isize - 1)
157        };
158        let mut samples = Vec::new();
159        let mut volume = 0.0;
160        for y in low(cy)..=high(cy, shape.ny) {
161            let y_part = ((cy as f64 - y as f64) / size).powi(2);
162            for x in low(cx)..=high(cx, shape.nx) {
163                let r2 = y_part + ((cx as f64 - x as f64) / size).powi(2);
164                if r2 < 1.0 {
165                    let value = (1.0 - r2) * spheroidal(r2.max(0.0).sqrt());
166                    volume += value;
167                    samples.push((x - cx as isize, y - cy as isize, value));
168                }
169            }
170        }
171        for sample in &mut samples {
172            sample.2 /= volume;
173        }
174        samples.retain(|sample| sample.2 != 0.0);
175        Self { size, samples }
176    }
177
178    /// The function as a plane centred at the origin, wrapping negative
179    /// offsets, so that a product of transforms is a centred convolution.
180    fn wrapped(&self, shape: PlaneShape) -> Vec<f64> {
181        let mut plane = vec![0.0; shape.len()];
182        for &(dx, dy, value) in &self.samples {
183            let x = dx.rem_euclid(shape.nx as isize) as usize;
184            let y = dy.rem_euclid(shape.ny as isize) as usize;
185            plane[shape.index(x, y)] += value;
186        }
187        plane
188    }
189}

191/// The scales of one solve with their transforms and small-scale biases.
192pub(crate) struct ScaleBank {
193    shape: PlaneShape,
194    pub(crate) functions: Vec<ScaleFunction>,
195    pub(crate) bias: Vec<f64>,
196    transforms: Vec<Vec<Complex64>>,
197    pub(crate) fft: PlaneFft,
198}
199
200impl ScaleBank {
201    /// Heap bytes a bank of `sizes` on `shape` holds, with what building
202    /// it holds besides: each scale's spectrum and samples, and one
203    /// wrapped plane and its spectrum at a time, beside planning their
204    /// transform.
205    pub(crate) fn bytes(sizes: &[f64], shape: PlaneShape) -> u64 {
206        let samples = sizes
207            .iter()
208            .map(|&size| {
209                let count = Self::scale_cells(size, shape).next_power_of_two();
210                (count * size_of::<(isize, isize, f64)>()) as u64
211            })
212            .sum::<u64>();
213        sizes.len() as u64 * spectrum_bytes(shape)
214            + samples
215            + plane_bytes(shape)
216            + spectrum_bytes(shape)
217            + planning_bytes(shape)
218    }
219
220    /// Samples one scale function of `size` pixels has at most on `shape`:
221    /// the disc's bounding square, as [`ScaleFunction::new`] scans it.
222    fn scale_cells(size: f64, shape: PlaneShape) -> usize {
223        let side = 2 * size as usize + 3;
224        (side * side).min(shape.len())
225    }
226
227    /// Pixels a component of any of `sizes` covers at most on `shape`.
228    pub(crate) fn support_cells(sizes: &[f64], shape: PlaneShape) -> usize {
229        sizes
230            .iter()
231            .map(|&size| Self::scale_cells(size, shape))
232            .max()
233            .unwrap_or(1)
234    }
235
236    /// Heap bytes [`Self::masks`] holds while it runs, besides the masks it
237    /// returns: the support as a plane, its spectrum and one convolution.
238    pub(crate) const fn masks_bytes(shape: PlaneShape) -> u64 {
239        plane_bytes(shape) + spectrum_bytes(shape) + convolve_bytes(shape)
240    }
241
242    /// The scales `sizes` (ascending) with CASA's bias
243    /// `1 − bias·size/largest` (1 for a single scale); `small_scale_bias`
244    /// is clamped to [−1, 1] as `SDAlgorithmMSClean` does.
245    pub(crate) fn new(
246        sizes: &[f64],
247        small_scale_bias: f64,
248        shape: PlaneShape,
249    ) -> Result<Self, Error> {
250        let small_scale_bias = small_scale_bias.clamp(-1.0, 1.0);
251        let largest = sizes.last().copied().unwrap_or(0.0);
252        let bias = sizes
253            .iter()
254            .map(|size| {
255                if sizes.len() > 1 {
256                    1.0 - small_scale_bias * size / largest
257                } else {
258                    1.0
259                }
260            })
261            .collect();
262        let functions = sizes
263            .iter()
264            .map(|&size| ScaleFunction::new(size, shape))
265            .collect::<Vec<_>>();
266        let mut fft = PlaneFft::new(shape)?;
267        let transforms = functions
268            .iter()
269            .map(|function| fft.forward(&function.wrapped(shape)))
270            .collect::<Result<Vec<_>, _>>()?;
271        Ok(Self {
272            shape,
273            functions,
274            bias,
275            transforms,
276            fft,
277        })
278    }
279
280    /// Number of scales.
281    pub(crate) fn len(&self) -> usize {
282        self.functions.len()
283    }
284
285    /// `spectrum` times the transforms of the listed scales, back in the
286    /// image domain: the centred circular convolution with each.
287    pub(crate) fn convolve(
288        &mut self,
289        spectrum: &[Complex64],
290        scales: &[usize],
291    ) -> Result<Vec<f64>, Error> {
292        let mut product = spectrum.to_vec();
293        for &scale in scales {
294            for (value, kernel) in product.iter_mut().zip(&self.transforms[scale]) {
295                *value *= kernel;
296            }
297        }
298        self.fft.inverse(product)
299    }
300
301    /// The support convolved with each scale, above `threshold`, with the
302    /// `border(size)` pixels at every edge removed (`makeScaleMasks`,
303    /// `setupUserMask`). `skip_point_border` keeps the edge of the point
304    /// scale, as the multi-term cleaner does.
305    pub(crate) fn masks(
306        &mut self,
307        support: &Support,
308        threshold: f64,
309        skip_point_border: bool,
310    ) -> Result<Vec<Support>, Error> {
311        let shape = self.shape;
312        let plane = support
313            .as_slice()
314            .iter()
315            .map(|supported| f64::from(u8::from(*supported)))
316            .collect::<Vec<_>>();
317        let spectrum = self.fft.forward(&plane)?;
318        (0..self.len())
319            .map(|scale| {
320                let smoothed = self.convolve(&spectrum, &[scale])?;
321                let border = if skip_point_border && scale == 0 {
322                    None
323                } else {
324                    Some((self.functions[scale].size * 1.5) as usize)
325                };
326                let pixels = smoothed
327                    .iter()
328                    .enumerate()
329                    .map(|(index, value)| {
330                        let [x, y] = shape.pixel(index);
331                        let inside = border.is_none_or(|border| {
332                            x > border
333                                && y > border
334                                && x + border + 1 < shape.nx
335                                && y + border + 1 < shape.ny
336                        });
337                        inside && *value > threshold
338                    })
339                    .collect();
340                Ok(Support::new(shape, pixels))
341            })
342            .collect()
343    }
344}

350    #[test]
351    fn scale_functions_have_unit_sum_and_the_casa_radius() {
352        let shape = PlaneShape::new(64, 48);
353        for size in [0.0, 1.5, 6.0, 10.0] {
354            let function = ScaleFunction::new(size, shape);
355            let sum = function.samples.iter().map(|sample| sample.2).sum::<f64>();
356            assert!((sum - 1.0).abs() < 1e-12, "size {size}: sum {sum}");
357            assert!(
358                function
359                    .samples
360                    .iter()
361                    .all(|&(dx, dy, _)| ((dx * dx + dy * dy) as f64) < size * size || size == 0.0)
362            );
363        }
364        assert_eq!(spheroidal(0.0), 1.0);
365        assert_eq!(spheroidal(1.0), 0.0);
366        // Continuous across the 0.75 break.
367        assert!((spheroidal(0.749_999_9) - spheroidal(0.75)).abs() < 1e-5);
368    }
369
370    /// Transform products are centred convolutions: a delta convolved with a
371    /// scale is the scale centred on the delta, wrapping at the edges.
372    #[test]
373    fn convolution_is_centred_and_circular() {
374        let shape = PlaneShape::new(15, 12);
375        let mut bank = ScaleBank::new(&[0.0, 3.0], 0.6, shape).unwrap();
376        assert_eq!(bank.bias, vec![1.0, 0.4]);
377        let mut delta = vec![0.0; shape.len()];
378        delta[shape.index(1, 10)] = 2.0;
379        let spectrum = bank.fft.forward(&delta).unwrap();
380        let convolved = bank.convolve(&spectrum, &[1]).unwrap();
381        let mut expected = vec![0.0; shape.len()];
382        for &(dx, dy, value) in &bank.functions[1].samples {
383            let x = (1 + dx).rem_euclid(15) as usize;
384            let y = (10 + dy).rem_euclid(12) as usize;
385            expected[shape.index(x, y)] += 2.0 * value;
386        }
387        for (a, e) in convolved.iter().zip(&expected) {
388            assert!((a - e).abs() < 1e-12);
389        }
390    }

392    #[test]
393    fn scale_masks_drop_the_border_and_partial_overlaps() {
394        let shape = PlaneShape::new(32, 32);
395        let mut bank = ScaleBank::new(&[0.0, 4.0], 0.0, shape).unwrap();
396        let support = Support::new(
397            shape,
398            (0..shape.len())
399                .map(|index| {
400                    let [x, y] = shape.pixel(index);
401                    (8..24).contains(&x) && (8..24).contains(&y)
402                })
403                .collect(),
404        );
405        let masks = bank.masks(&support, 0.9, false).unwrap();
406        // The point scale keeps the box; a full-image support loses its ring.
407        assert_eq!(masks[0], support);
408        let full = bank.masks(&Support::full(shape), 0.9, false).unwrap();
409        assert!(!full[0].contains(shape.index(0, 5)) && full[0].contains(shape.index(1, 1)));
410        assert!(bank.masks(&Support::full(shape), 0.1, true).unwrap()[0].contains(0));
411        // The 4-pixel scale needs 90% of its weight inside the box.
412        assert!(masks[1].contains(shape.index(16, 16)));
413        assert!(!masks[1].contains(shape.index(8, 16)));
414        assert!(masks[1].count() < support.count());
415    }
