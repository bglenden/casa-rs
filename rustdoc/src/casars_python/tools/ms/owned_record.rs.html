4use casa_types::{ArrayValue, RecordValue, Value};

10pub(super) fn visibility_record(record: RecordValue) -> CasaValue {
11    CasaValue::record(record.into_fields().into_iter().map(|field| {
12        let value = match field.value {
13            Value::Array(array) => visibility_array(array),
14            Value::Record(record) => visibility_record(record),
15            other => CasaValue::from_value(&other),
16        };
17        (field.name, value)
18    }))
19}
20
21fn visibility_array(array: ArrayValue) -> CasaValue {
22    macro_rules! move_array {
23        ($array:expr, $variant:ident) => {{
24            let (shape, values) = take_fortran($array);
25            CasaValue::$variant { shape, values }
26        }};
27        ($array:expr, $variant:ident, $convert:expr) => {{
28            let (shape, values) = take_fortran($array);
29            CasaValue::$variant {
30                shape,
31                values: values.into_iter().map($convert).collect(),
32            }
33        }};
34    }
35    match array {
36        ArrayValue::Bool(array) => move_array!(array, Bools),
37        ArrayValue::UInt8(array) => move_array!(array, Ints, i64::from),
38        ArrayValue::UInt16(array) => move_array!(array, Ints, i64::from),
39        ArrayValue::UInt32(array) => move_array!(array, Ints, i64::from),
40        ArrayValue::Int16(array) => move_array!(array, Ints, i64::from),
41        ArrayValue::Int32(array) => move_array!(array, Ints, i64::from),
42        ArrayValue::Int64(array) => move_array!(array, Ints),
43        ArrayValue::Float32(array) => move_array!(array, Floats, f64::from),
44        ArrayValue::Float64(array) => move_array!(array, Floats),
45        ArrayValue::Complex32(array) => move_array!(array, Complexes, |value| Complex64::new(
46            f64::from(value.re),
47            f64::from(value.im)
48        )),
49        ArrayValue::Complex64(array) => move_array!(array, Complexes),
50        ArrayValue::String(array) => move_array!(array, TextArray),
51    }
52}
53
54fn take_fortran<T>(array: ArrayD<T>) -> (Vec<usize>, Vec<T>) {
55    let shape = array.shape().to_vec();
56    let array = array.reversed_axes();
57    let length = array.len();
58    if array.is_standard_layout() {
59        let (mut values, offset) = array.into_raw_vec_and_offset();
60        if let Some(offset) = offset {
61            values.truncate(offset + length);
62            values.drain(..offset);
63        } else {
64            values.clear();
65        }
66        (shape, values)
67    } else {
68        (shape, array.into_iter().collect())
69    }
70}
