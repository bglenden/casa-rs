19const LIGHT_SPEED: f64 = 299_792_458.;
20const PLANCK: f64 = 6.626_075_5e-34;
21const BOLTZMANN: f64 = 1.380_657_798_751_064_7e-23;

23pub struct Solar2010 {
24    direction: MDirection,
25    diameter_rad: f64,
26    temperature: Temperature,
27}
28
29impl Solar2010 {
30    pub fn open(workspace: &Path, body: &str, epoch: &MEpoch, frame: &MeasFrame) -> MsResult<Self> {
31        let mut temperature = Temperature::for_body(body).ok_or_else(|| {
32            invalid(format!(
33                "Solar System object {body:?} is not supported by Butler-JPL-Horizons 2010"
34            ))
35        })?;
36        let path = find_table(workspace, body, epoch, frame)?;
37        let table = Table::open(TableOptions::new(path))?;
38        let radius = quantity_keyword(&table, "meanrad", "AU")?
39            .ok_or_else(|| invalid("ephemeris has no meanrad"))?;
40        let interpolation = EphemerisSample::new(&table, epoch.value().as_mjd())?;
41        let distance = interpolation.sample(&table, "Rho")?;
42        let has_direction = table.schema().is_some_and(|schema| {
43            ["RA", "DEC"]
44                .iter()
45                .all(|name| schema.column(name).is_some())
46        });
47        let direction = if has_direction {
48            MDirection::from_angles(
49                interpolation.sample(&table, "RA")?.to_radians(),
50                interpolation.sample(&table, "DEC")?.to_radians(),
51                DirectionRef::J2000,
52            )
53        } else {
54            MDirection::from_angles(0., std::f64::consts::FRAC_PI_2, DirectionRef::J2000)
55        };
56        if let Temperature::Ephemeris = temperature {
57            temperature = Temperature::Constant(
58                quantity_keyword(&table, "T_mean", "K")?
59                    .filter(|value| *value > 0.)
60                    .ok_or_else(|| {
61                        invalid(format!(
62                            "the ephemeris has no brightness temperature for {body}"
63                        ))
64                    })?,
65            );
66        }
67        if !radius.is_finite() || radius <= 0. || !distance.is_finite() || distance <= 0. {
68            return Err(invalid("invalid Solar System radius or range"));
69        }
70        Ok(Self {
71            direction,
72            diameter_rad: 2. * radius / distance,
73            temperature,
74        })
75    }
76
77    pub fn component(
78        &self,
79        position: Option<MDirection>,
80        frequencies: &[MFrequency],
81    ) -> MsResult<(SkyComponent, [f64; 4])> {
82        // Explicit FIELD references are retained even at the north pole.
83        let direction = position.unwrap_or_else(|| self.direction.clone());
84        let shape = ComponentShape::from_parameters(
85            ShapeType::Disk,
86            direction,
87            &[self.diameter_rad, self.diameter_rad, 0.],
88            None,
89        )
90        .map_err(invalid)?;
91        let exponent = PLANCK / BOLTZMANN;
92        let area = std::f64::consts::PI * (self.diameter_rad * 0.5 / LIGHT_SPEED).powi(2);
93        let fluxes = frequencies
94            .iter()
95            .map(|frequency| {
96                let hz = frequency.hz();
97                if !hz.is_finite() || hz <= 0. {
98                    return Err(invalid(
99                        "Solar System flux needs positive finite frequencies",
100                    ));
101                }
102                let temp = self.temperature.at(hz);
103                let flux = 2.0e26 * PLANCK * area * hz.powi(3) / (exponent * hz / temp).exp_m1();
104                Ok([flux, 0., 0., 0.])
105            })
106            .collect::<MsResult<Vec<_>>>()?;
107        sampled_shape(shape, frequencies, fluxes)
108    }
109}

111fn find_table(
112    workspace: &Path,
113    body: &str,
114    epoch: &MEpoch,
115    frame: &MeasFrame,
116) -> MsResult<PathBuf> {
117    let mut roots = vec![workspace.to_path_buf()];
118    for runtime in MeasuresRuntime::discover_candidates().unwrap_or_default() {
119        roots.push(runtime.path.join("ephemerides/JPL-Horizons"));
120    }
121    let prefix = format!("{}_", body.to_ascii_lowercase());
122    for root in roots {
123        let candidates = std::fs::read_dir(root)
124            .into_iter()
125            .flatten()
126            .flatten()
127            .filter(|entry| entry.path().is_dir());
128        // CASA accepts the first covering table in directory iteration order.
129        for candidate in candidates {
130            let name = candidate.file_name();
131            let lower = name.to_string_lossy().to_ascii_lowercase();
132            let Some(range) = lower
133                .strip_prefix(&prefix)
134                .and_then(|name| name.strip_suffix(".tab"))
135            else {
136                continue;
137            };
138            if covers(range, epoch, frame) {
139                return Ok(candidate.path());
140            }
141        }
142    }
143    Err(invalid(format!(
144        "no JPL-Horizons ephemeris for {body} at MJD {}",
145        epoch.value().as_mjd()
146    )))
147}

149fn covers(range: &str, epoch: &MEpoch, frame: &MeasFrame) -> bool {
150    let Some(separator) = range
151        .get(1..)
152        .and_then(|range| range.find('-'))
153        .map(|index| index + 1)
154    else {
155        return false;
156    };
157    let (start, rest) = range.split_at(separator);
158    let rest = &rest[1..];
159    let Some(unit) = rest.find(['d', 'h', 'm', 's', 'y']) else {
160        return false;
161    };
162    let (end, time) = rest.split_at(unit);
163    let (Ok(start), Ok(end), Ok(unit), Ok(reference)) = (
164        start.parse::<f64>(),
165        end.parse::<f64>(),
166        Unit::new(&time[..1]),
167        time[1..].parse::<EpochRef>(),
168    ) else {
169        return false;
170    };
171    let Ok(time) = epoch.convert_to(reference, frame) else {
172        return false;
173    };
174    Quantity::new(time.value().as_mjd(), "d")
175        .and_then(|date| date.get_value_in(&unit))
176        .is_ok_and(|value| (start..=end).contains(&value))
177}

179fn quantity_keyword(table: &Table, name: &str, unit: &str) -> MsResult<Option<f64>> {
180    let Some(Value::Record(record)) = table.keywords().get(name) else {
181        return Ok(None);
182    };
183    let (
184        Some(Value::Scalar(ScalarValue::Float64(value))),
185        Some(Value::Scalar(ScalarValue::String(input_unit))),
186    ) = (record.get("value"), record.get("unit"))
187    else {
188        return Err(invalid(format!("invalid ephemeris quantity {name}")));
189    };
190    let unit = Unit::new(unit).map_err(invalid)?;
191    Quantity::new(*value, input_unit)
192        .and_then(|quantity| quantity.get_value_in(&unit))
193        .map(Some)
194        .map_err(invalid)
195}

197struct EphemerisSample {
198    indices: [usize; 3],
199    offsets: [f64; 2],
200    f: f64,
201}
202
203impl EphemerisSample {
204    fn new(table: &Table, epoch: f64) -> MsResult<Self> {
205        let count = table.row_count();
206        if count == 0 {
207            return Err(invalid("empty JPL-Horizons ephemeris"));
208        }
209        let column = table.column_accessor("MJD")?;
210        let nearest = legacy_sample_center(&column, count, epoch)?;
211        let indices = [
212            nearest.saturating_sub(1),
213            nearest,
214            (nearest + 1).min(count - 1),
215        ];
216        let [a, b, c] = indices.map(|row| scalar(&column, row));
217        let [a, b, c] = [a?, b?, c?];
218        Ok(Self {
219            indices,
220            offsets: [b - a, c - b],
221            f: epoch - b,
222        })
223    }
224
225    fn sample(&self, table: &Table, name: &str) -> MsResult<f64> {
226        let column = table.column_accessor(name)?;
227        let [prior, middle, next] = self.indices.map(|row| scalar(&column, row));
228        let (prior, middle, next) = (prior?, middle?, next?);
229        let [left, right] = self.offsets;
230        let span = left + right;
231        if span == 0. {
232            return Ok(middle);
233        }
234        let phase = self.f / span;
235        let curvature = if left > 0. && right > 0. {
236            span * ((next - middle) / right - (middle - prior) / left)
237        } else {
238            0.
239        };
240        Ok(middle + phase * (next - prior + phase * curvature))
241    }
242}

244// The 2010 standard starts from the first grid interval, then doubles or
245// halves its stride. It can center the quadratic on either bracketing row.
246fn legacy_sample_center(
247    column: &casa_tables::TableColumn<'_>,
248    count: usize,
249    epoch: f64,
250) -> MsResult<usize> {
251    let first = scalar(column, 0)?;
252    let last = scalar(column, count - 1)?;
253    if !(first..=last).contains(&epoch) {
254        return Err(invalid("observation is outside the JPL-Horizons ephemeris"));
255    }
256    if epoch == first {
257        return Ok(0);
258    }
259    if epoch == last {
260        return Ok(count - 1);
261    }
262    let mut anchor = 1;
263    while anchor < count - 1 && scalar(column, anchor)? == first {
264        anchor += 1;
265    }
266    let interval = (scalar(column, anchor)? - first) / anchor as f64;
267    let mut lower = 0_isize;
268    let mut upper = (count - 1) as isize;
269    let mut row = (((epoch - first) / interval).round_ties_even() as isize).clamp(lower, upper);
270    let mut forward = scalar(column, row as usize)? < epoch;
271    let mut step = 1;
272    for _ in 0..count {
273        let earlier = scalar(column, row as usize)? < epoch;
274        if earlier {
275            lower = lower.max(row);
276        } else {
277            upper = upper.min(row);
278        }
279        step = if earlier == forward {
280            step * 2
281        } else {
282            step / 2
283        };
284        forward = earlier;
285        let limit = if forward { upper - row } else { row - lower };
286        if step > limit {
287            step = (limit - 1).max(0);
288        }
289        if step == 0 {
290            return Ok(row as usize);
291        }
292        row += if forward { step } else { -step };
293    }
294    Err(invalid(
295        "could not bracket the JPL-Horizons observation epoch",
296    ))
297}

299pub(super) fn scalar(column: &casa_tables::TableColumn<'_>, row: usize) -> Result<f64, MsError> {
300    match column.scalar_cell(row)? {
301        ScalarValue::Float64(value) => Ok(*value),
302        _ => Err(invalid("expected a double-precision ephemeris column")),
303    }
304}

306enum Temperature {
307    Constant(f64),
308    Ephemeris,
309    Venus,
310    Jupiter,
311    Uranus,
312    Neptune,
313}
314
315impl Temperature {
316    fn for_body(body: &str) -> Option<Self> {
317        Some(match body.to_ascii_lowercase().as_str() {
318            "venus" => Self::Venus,
319            "jupiter" => Self::Jupiter,
320            "uranus" => Self::Uranus,
321            "neptune" => Self::Neptune,
322            "io" | "ganymede" => Self::Constant(110.),
323            "callisto" => Self::Constant(134.),
324            "europa" => Self::Constant(109.),
325            "titan" => Self::Constant(76.6),
326            "triton" => Self::Constant(38.),
327            "pluto" => Self::Constant(35.),
328            "ceres" => Self::Constant(167.),
329            "pallas" => Self::Constant(164.),
330            "juno" => Self::Constant(163.),
331            "vesta" => Self::Constant(160.),
332            "mercury" | "mars" | "victoria" | "davida" => Self::Ephemeris,
333            _ => return None,
334        })
335    }
336
337    fn at(&self, hz: f64) -> f64 {
338        let wavelength_cm = 100. * LIGHT_SPEED / hz;
339        match self {
340            Self::Constant(temperature) => *temperature,
341            Self::Ephemeris => {
342                unreachable!("the ephemeris temperature was read when opening the model")
343            }
344            Self::Venus => venus(hz).max(2.7),
345            Self::Jupiter => {
346                let wave = wavelength_cm.clamp(0.1, 6.2);
347                if wave < 0.44 {
348                    170.
349                } else if wave < 0.7 {
350                    160. + 21.537539 * (0.7 / wave).ln()
351                } else if wave < 1.3 {
352                    130. + 48.462196889 * (1.3 / wave).ln()
353                } else {
354                    130. + 65.38532335444 * (wave / 1.3).log10()
355                }
356            }
357            Self::Uranus => {
358                let wave = wavelength_cm.clamp(0.07, 6.2);
359                if wave < 0.4 {
360                    90. + 32.46063842 * (10. * wave).ln()
361                } else if wave < 1. {
362                    135.
363                } else {
364                    135. + 105. * wave.log10()
365                }
366            }
367            Self::Neptune => {
368                let ghz = (hz * 1e-9).clamp(4., 1000.);
369                if ghz < 70. {
370                    140. - 30.083556662 * (ghz / 70.).ln()
371                } else {
372                    240. - 34.93815 * (ghz / 4.).ln()
373                }
374            }
375        }
376    }
377}

379fn venus(hz: f64) -> f64 {
380    // The published 2010 grid uses Float frequencies and a four-point cubic.
381    let ghz = f64::from((hz * 1e-9) as f32);
382    let first = VENUS
383        .partition_point(|&(freq, _)| f64::from(freq) <= ghz)
384        .saturating_sub(2)
385        .min(VENUS.len() - 4);
386    let samples = &VENUS[first..first + 4];
387    samples
388        .iter()
389        .enumerate()
390        .map(|(i, &(x, y))| {
391            let x = f64::from(x);
392            samples.iter().enumerate().filter(|(j, _)| *j != i).fold(
393                y,
394                |weight, (_, &(other, _))| {
395                    let other = f64::from(other);
396                    weight * (ghz - other) / (x - other)
397                },
398            )
399        })
400        .sum()
401}
