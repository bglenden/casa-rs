216/// Stable application spelling of the scientific minor-cycle terminal reason.
217#[derive(Clone, Copy, Debug, PartialEq, Eq)]
218pub enum NativeMinorCycleStopReason {
219    /// The normalized residual peak fell below the requested threshold.
220    ThresholdReached,
221    /// The bounded minor-cycle iteration budget was exhausted.
222    IterationBound,
223    /// The next update would exceed the frozen-approximation envelope.
224    StalenessBound,
225    /// The multiscale residual trajectory diverged after accepted progress.
226    MultiscaleDivergence,
227}
228
229impl From<MinorCycleStopReason> for NativeMinorCycleStopReason {
230    fn from(value: MinorCycleStopReason) -> Self {
231        match value {
232            MinorCycleStopReason::ThresholdReached => Self::ThresholdReached,
233            MinorCycleStopReason::IterationBound => Self::IterationBound,
234            MinorCycleStopReason::StalenessBound => Self::StalenessBound,
235            MinorCycleStopReason::MultiscaleDivergence => Self::MultiscaleDivergence,
236        }
237    }
238}

240impl NativeMinorCycleOutcome {
241    /// Return the first exact component mismatch against a CASA/parity baseline.
242    #[must_use]
243    pub fn first_component_divergence(
244        &self,
245        baseline: &[casa_imaging_reconstruction::MinorCycleComponent],
246    ) -> Option<(
247        usize,
248        Option<casa_imaging_reconstruction::MinorCycleComponent>,
249        Option<casa_imaging_reconstruction::MinorCycleComponent>,
250    )> {
251        let shared = baseline.len().min(self.recorded_components.len());
252        for (index, (expected, actual)) in
253            baseline.iter().zip(&self.recorded_components).enumerate()
254        {
255            if expected != actual {
256                return Some((index, Some(*expected), Some(*actual)));
257            }
258        }
259        (baseline.len() != self.recorded_components.len()).then(|| {
260            (
261                shared,
262                baseline.get(shared).copied(),
263                self.recorded_components.get(shared).copied(),
264            )
265        })
266    }
267}

275/// Execute one imaging request through the sole installed implementation.
276/// Unsupported requirements fail typed before physical planning or execution.
277pub fn execute<S>(
278    request: ApplicationRequest<S>,
279) -> Result<ApplicationOutcome, ApplicationDispatchError>
280where
281    S: SerialProductPublicationSink + Send + 'static,
282    S::Error: Send + Sync,
283{
284    let resolved = resolve_selected_observation(request.observation.clone())
285        .map_err(|error| ApplicationDispatchError::Preparation(Box::new(error)))?;
286    let (snapshot, access) = resolved.into_parts();
287    let observation = compile_observation(snapshot)
288        .map_err(|error| ApplicationDispatchError::Preparation(Box::new(error)))?;
289    let imaging = ImagingRequest::new(
290        request.specification,
291        request.geometry,
292        ProblemInputIdentities::new(observation),
293        request.model_lifecycle,
294    );
295    let problem = compile(imaging).map_err(ApplicationDispatchError::Compile)?;
296    validate_installed_implementation(&problem, request.task_requirements)
297        .map_err(ApplicationDispatchError::Unavailable)?;
298    let input = NativeInput {
299        observation: request.observation,
300        initial_access: access,
301        write_model_column: request.write_model_column,
302        write_corrected_data: request.write_corrected_data,
303        mask: request.mask,
304        native: request.native,
305    };
306    let output = run_native(&problem, input).map_err(ApplicationDispatchError::Native)?;
307    Ok(ApplicationOutcome {
308        output: Box::new(output),
309    })
310}

312struct NativeInput<S> {
313    observation: SelectedObservationResolutionRequest,
314    initial_access: ResolvedSelectedObservationAccess,

321fn run_native<S>(
322    problem: &CompiledProblem,
323    input: NativeInput<S>,
324) -> Result<NativeApplicationOutcome, ApplicationError>
325where
326    S: SerialProductPublicationSink + Send + 'static,
327    S::Error: Send + Sync,
328{
329    let ApplicationNative {
330        runtime,
331        publication,
332    } = input.native?;
333    let algorithm = problem.reconstruction().algorithm().clone();
334    let initial_access = input.initial_access;
335    let residency = initial_access.certify_residency(problem)?;
336    let write_targets =
337        SelectedVisibilityWriteTargets::new(input.write_model_column, input.write_corrected_data);
338    let visibility_write_requested = write_targets.model_data() || write_targets.corrected_data();
339    let initial_write =
340        matches!(algorithm, ReconstructionAlgorithm::Dirty) && visibility_write_requested;
341    let planning_registry =
342        PlanningRegistry::new(runtime.registry, runtime.implementation.clone(), problem);
343    let mut policy = execution_policy(&runtime, residency.clone());
344    if initial_write {
345        policy = policy
346            .with_visibility_write(initial_access.selected_visibility_storage_plan(write_targets)?);
347    }
348    let planned = match algorithm {
349        ReconstructionAlgorithm::Dirty => {
350            SpectralCyclePlan::dirty(problem, &planning_registry, policy)?
351        }
352        ReconstructionAlgorithm::Hogbom
353        | ReconstructionAlgorithm::Clark
354        | ReconstructionAlgorithm::Multiscale { .. } => {
355            SpectralCyclePlan::initial(problem, &planning_registry, policy)?
356        }
357        _ => unreachable!("native validation admits only continuum minor-cycle solvers"),
358    };
359    let minor_node = planned.minor_cycle_node().cloned();
360    let SpectralCyclePlanParts {
361        physical,
362        weighting,
363        complete_data: complete,
364        source_resources: resources,
365        pass,
366        gridded_normal: planned_gridded_normal,
367        ..
368    } = planned.into_parts();
369    let replay_proof_bytes = initial_access.replay_proof_retained_heap_bytes(problem)?;
370    let frozen_reservation = (!matches!(algorithm, ReconstructionAlgorithm::Dirty))
371        .then(|| {
372            FrozenWeightingReservation::acquire(
373                &runtime.authority,
374                runtime.resource_policy.clone(),
375                weighting.planned_residency(),
376                replay_proof_bytes,
377            )
378        })
379        .transpose()?;
380    let initial_source_state = initial_access.source_state().clone();
381    let selected = initial_access.open(problem)?;
382    let mut executor = SpectralCycleExecutor::new(
383        runtime.implementation.clone(),
384        problem.clone(),
385        weighting,
386        resources,
387        pass,
388        complete,
389        selected,
390        ExecutableModelProblem::from_compiled(problem.clone())?,
391        SpectralCyclePassInput::Initial,
392    );
393    if !matches!(algorithm, ReconstructionAlgorithm::Dirty) {
394        executor = executor.with_frozen_weighting_reservation(
395            frozen_reservation.expect("non-dirty execution reserves frozen weighting"),
396        );
397        executor = executor.with_planned_gridded_normal_binding(
398            planned_gridded_normal
399                .ok_or_else(|| boxed("clean initial plan omitted gridded replay binding"))?,
400        )?;
401        let program = MinorCycleProgram::for_algorithm(
402            algorithm.clone(),
403            problem.reconstruction().controls(),
404        )?
405        .record_component_sequence(64)?;
406        executor = executor.with_reconstruction_cycle(
407            minor_node.ok_or_else(|| boxed("initial plan omitted its minor-cycle node"))?,
408            input.mask.clone(),
409            program,
410        );
411    }
412    let mut initial_terminal_replay = None;
413    if initial_write {
414        let (replay, sink) = FinalVisibilityReplay::with_visibility_write(
415            std::path::PathBuf::from(input.observation.locator()),
416            initial_source_state,
417            visibility_write_selection(problem, input.observation.selection())?,
418            write_targets,
419        )?;
420        executor = executor.with_final_visibility_sink(sink);
421        initial_terminal_replay = Some(replay);
422    }
423    let registry = SpectralCycleRegistry::new(
424        runtime.registry,
425        runtime.implementation.clone(),
426        problem,
427        executor,
428    );
429    let initial_plan = plan(
430        problem,
431        PlanningBindings::new(
432            runtime.registry,
433            runtime.resource_policy.clone(),
434            runtime.cost_model,
435        ),
436        &runtime.authority,
437        &registry,
438        &runtime.receipts,
439        move |_, _| Ok::<_, std::convert::Infallible>(vec![physical]),
440    )?;
441    run_phase(
442        problem,
443        &initial_plan,
444        &registry,
445        &runtime,
446        runtime.attempts[0],
447    )?;
448    let initial_receipt = runtime.receipts.open(runtime.attempts[0])?;
449
450    let (
451        scientific,
452        final_reconstruction_mask,
453        final_major_receipt,
454        minor_cycles,
455        major_cycle_count,
456        total_minor_iterations,
457        total_actual_minor_iterations,
458        visibility_products,
459        visibility_replay,
460        visibility_output_receipt,
461    ) = match algorithm {
462        ReconstructionAlgorithm::Dirty => {
463            let result = registry
464                .implementation()
465                .take_completion()
466                .ok_or_else(|| boxed("dirty execution omitted final major-cycle evidence"))?;
467            let visibility_products = initial_terminal_replay
468                .as_ref()
469                .map(FinalVisibilityReplay::completion)
470                .transpose()?;
471            (
472                result.into_completion(),
473                None,
474                None,
475                Vec::new(),
476                1,
477                0,
478                0,
479                visibility_products,
480                initial_terminal_replay,
481                None,
482            )
483        }
484        ReconstructionAlgorithm::Hogbom
485        | ReconstructionAlgorithm::Clark
486        | ReconstructionAlgorithm::Multiscale { .. } => {
487            let mut frozen_weighting = registry
488                .implementation()
489                .take_frozen_weighting()
490                .ok_or_else(|| boxed("initial major omitted frozen weighting"))?;
491            let mut gridded_replay = registry
492                .implementation()
493                .take_gridded_normal_replay()
494                .ok_or_else(|| boxed("initial major omitted sealed gridded-normal replay"))?;
495            let mut minor = registry
496                .implementation()
497                .take_reconstruction_cycle_completion()
498                .ok_or_else(|| boxed("reconstruction cycle omitted scientific evidence"))?;
499            let controls = problem.reconstruction().controls();
500            // CASA's `nmajor=-1` leaves the major-cycle count unbounded, but
501            // every productive cycle consumes at least one of the total
502            // minor-iteration budget. That budget is therefore the finite
503            // execution ceiling when no explicit major-cycle limit exists.
504            let maximum_cycles = controls
505                .maximum_major_cycles()
506                .unwrap_or(controls.max_minor_iterations());
507            let mut cycle = 1_usize;
508            let mut total_iterations = 0_usize;
509            let mut total_actual_iterations = 0_usize;
510            let mut mask_plan = input.mask.clone();
511            let mut minor_outcomes = Vec::new();
512            loop {
513                let iterations_entering = total_iterations;
514                let actual_iterations_entering = total_actual_iterations;
515                total_iterations = total_iterations
516                    .checked_add(minor.evidence().controller_iterations())
517                    .ok_or_else(|| boxed("minor-cycle iteration count overflowed"))?;
518                total_actual_iterations = total_actual_iterations
519                    .checked_add(minor.evidence().iterations())
520                    .ok_or_else(|| boxed("actual minor-cycle iteration count overflowed"))?;
521                let minor_outcome = NativeMinorCycleOutcome {
522                    cycle,
523                    iterations_entering,
524                    iterations: minor.evidence().controller_iterations(),
525                    total_iterations,
526                    actual_iterations_entering,
527                    actual_iterations: minor.evidence().iterations(),
528                    total_actual_iterations,
529                    total_flux: minor.evidence().total_flux(),
530                    initial_peak_flux: minor.evidence().initial_peak_flux(),
531                    final_peak_flux: minor.evidence().final_peak_flux(),
532                    noise_rms: minor.evidence().noise_rms(),
533                    effective_threshold: minor.evidence().effective_threshold(),
534                    global_threshold: minor.evidence().global_threshold(),
535                    cycle_threshold: minor.evidence().cycle_threshold(),
536                    stop_reason: minor.evidence().stop_reason().into(),
537                    clark_refreshes: minor.evidence().clark_refreshes(),
538                    associated_replay_ordinal: cycle,
539                    recorded_components: minor.evidence().recorded_components().copied().collect(),
540                    mask_support: minor.mask().support().to_vec(),
541                    mask_generation: minor.mask().generation_id(),
542                    mask_model_generation: minor.mask().model_generation(),
543                    mask_normal_state: minor.mask().normal_state_completion(),
544                    auto_mask: minor.auto_mask_evidence(),
545                };
546                eprintln!(
547                    "imaging_minor_cycle_summary cycle={} associated_replay_ordinal={} controller_iterations_entering={} controller_iterations={} controller_iterations_total={} actual_iterations_entering={} actual_iterations={} actual_iterations_total={} initial_peak_flux={} final_peak_flux={} model_update_abs_flux={} global_threshold={} effective_threshold={} cycle_threshold={} stop_reason={:?} clark_refreshes={}",
548                    minor_outcome.cycle,
549                    minor_outcome.associated_replay_ordinal,
550                    minor_outcome.iterations_entering,
551                    minor_outcome.iterations,
552                    minor_outcome.total_iterations,
553                    minor_outcome.actual_iterations_entering,
554                    minor_outcome.actual_iterations,
555                    minor_outcome.total_actual_iterations,
556                    minor_outcome.initial_peak_flux,
557                    minor_outcome.final_peak_flux,
558                    minor_outcome.total_flux,
559                    minor_outcome.global_threshold,
560                    minor_outcome.effective_threshold,
561                    minor_outcome
562                        .cycle_threshold
563                        .map_or_else(|| "none".to_owned(), |value| value.to_string()),
564                    minor_outcome.stop_reason,
565                    minor_outcome.clark_refreshes,
566                );
567                let continue_cleaning = cycle < maximum_cycles
568                    && total_iterations < controls.max_minor_iterations()
569                    && minor.evidence().requests_reconciliation();
570                let next_mask = mask_plan.next_cycle(
571                    minor.mask(),
572                    cycle,
573                    minor.evidence().cycle_threshold_is_global(),
574                    minor_outcome
575                        .auto_mask
576                        .is_some_and(|evidence| evidence.channel_stopped),
577                );
578                let applied_mask = minor.mask().clone();
579                minor_outcomes.push(minor_outcome);
580                let final_input = minor.into_final_major_input();
581                let final_policy = execution_policy(&runtime, residency.clone());
582                let ordinal =
583                    u32::try_from(cycle).map_err(|_| boxed("major-cycle ordinal exceeds u32"))?;
584                let final_planned = if continue_cleaning {
585                    SpectralCyclePlan::continuing_major(
586                        problem,
587                        &planning_registry,
588                        final_policy,
589                        &final_input,
590                        ordinal,
591                        gridded_replay,
592                    )?
593                } else {
594                    SpectralCyclePlan::final_major_at(
595                        problem,
596                        &planning_registry,
597                        final_policy,
598                        &final_input,
599                        ordinal,
600                        gridded_replay,
601                    )?
602                };
603                let SpectralCyclePlanParts {
604                    physical,
605                    weighting,
606                    complete_data: complete,
607                    pass,
608                    minor_cycle_node: minor_node,
609                    gridded_normal: planned_gridded_normal,
610                    ..
611                } = final_planned.into_parts();
612                let mut executor = SpectralCycleExecutor::new_gridded(
613                    runtime.implementation.clone(),
614                    problem.clone(),
615                    weighting,
616                    pass,
617                    complete,
618                    ExecutableModelProblem::from_compiled(problem.clone())?,
619                    SpectralCyclePassInput::FinalMajor(final_input),
620                    planned_gridded_normal
621                        .ok_or_else(|| boxed("later-major plan omitted gridded replay binding"))?,
622                )?
623                .with_frozen_weighting(frozen_weighting);
624                if continue_cleaning {
625                    let remaining = controls
626                        .max_minor_iterations()
627                        .saturating_sub(total_iterations);
628                    let program = MinorCycleProgram::for_algorithm(algorithm.clone(), controls)?
629                        .record_component_sequence(64)?
630                        .limit_iterations(remaining)?;
631                    executor = executor.with_reconstruction_cycle(
632                        minor_node.ok_or_else(|| boxed("continuing plan omitted minor node"))?,
633                        next_mask.clone(),
634                        program,
635                    );
636                }
637                let registry = SpectralCycleRegistry::new(
638                    runtime.registry,
639                    runtime.implementation.clone(),
640                    problem,
641                    executor,
642                );
643                let final_plan = plan(
644                    problem,
645                    PlanningBindings::new(
646                        runtime.registry,
647                        runtime.resource_policy.clone(),
648                        runtime.cost_model,
649                    ),
650                    &runtime.authority,
651                    &registry,
652                    &runtime.receipts,
653                    move |_, _| Ok::<_, std::convert::Infallible>(vec![physical]),
654                )?;
655                let attempt = major_cycle_attempt(runtime.attempts[1], ordinal);
656                run_phase(problem, &final_plan, &registry, &runtime, attempt)?;
657                let receipt = runtime.receipts.open(attempt)?;
658                frozen_weighting = registry
659                    .implementation()
660                    .take_frozen_weighting()
661                    .ok_or_else(|| boxed("later major omitted reusable frozen weighting"))?;
662                gridded_replay = registry
663                    .implementation()
664                    .take_gridded_normal_replay()
665                    .ok_or_else(|| boxed("later major omitted reusable gridded-normal replay"))?;
666                if continue_cleaning {
667                    minor = registry
668                        .implementation()
669                        .take_reconstruction_cycle_completion()
670                        .ok_or_else(|| boxed("continuing major omitted cycle evidence"))?;
671                    mask_plan = next_mask;
672                    cycle += 1;
673                    continue;
674                }
675                let completion = registry
676                    .implementation()
677                    .take_completion()
678                    .ok_or_else(|| boxed("final-major execution omitted scientific evidence"))?
679                    .into_completion();
680                if !visibility_write_requested {
681                    break (
682                        completion,
683                        Some(applied_mask),
684                        Some(receipt),
685                        minor_outcomes,
686                        cycle + 1,
687                        total_iterations,
688                        total_actual_iterations,
689                        None,
690                        None,
691                        None,
692                    );
693                }
694                let resolved = resolve_selected_observation(input.observation.clone())?;
695                let (_, access) = resolved.into_parts();
696                let output_residency = access.certify_residency(problem)?;
697                let source_state = access.source_state().clone();
698                let output_policy = execution_policy(&runtime, output_residency)
699                    .with_visibility_write(access.selected_visibility_storage_plan(write_targets)?);
700                let output_planned = SpectralCyclePlan::selected_output(
701                    problem,
702                    &planning_registry,
703                    output_policy,
704                    ordinal,
705                )?;
706                let SpectralCyclePlanParts {
707                    physical: output_physical,
708                    weighting: output_weighting,
709                    complete_data: output_complete,
710                    source_resources: output_resources,
711                    pass: output_pass,
712                    ..
713                } = output_planned.into_parts();
714                let selected = frozen_weighting.rebind_selected(access, problem)?;
715                let (visibility_replay, sink) = FinalVisibilityReplay::with_visibility_write(
716                    std::path::PathBuf::from(input.observation.locator()),
717                    source_state,
718                    visibility_write_selection(problem, input.observation.selection())?,
719                    write_targets,
720                )?;
721                let output_executor = SpectralCycleExecutor::new_selected_output(
722                    runtime.implementation.clone(),
723                    problem.clone(),
724                    output_weighting,
725                    output_resources,
726                    output_pass,
727                    output_complete,
728                    selected,
729                    completion,
730                    frozen_weighting,
731                )
732                .with_final_visibility_sink(sink);
733                let output_registry = SpectralCycleRegistry::new(
734                    runtime.registry,
735                    runtime.implementation.clone(),
736                    problem,
737                    output_executor,
738                );
739                let output_plan = plan(
740                    problem,
741                    PlanningBindings::new(
742                        runtime.registry,
743                        runtime.resource_policy.clone(),
744                        runtime.cost_model,
745                    ),
746                    &runtime.authority,
747                    &output_registry,
748                    &runtime.receipts,
749                    move |_, _| Ok::<_, std::convert::Infallible>(vec![output_physical]),
750                )?;
751                let output_attempt = selected_output_attempt(attempt);
752                run_phase(
753                    problem,
754                    &output_plan,
755                    &output_registry,
756                    &runtime,
757                    output_attempt,
758                )?;
759                let output_receipt = runtime.receipts.open(output_attempt)?;
760                let completion = output_registry
761                    .implementation()
762                    .take_selected_output_completion()
763                    .ok_or_else(|| boxed("selected-output traversal omitted scientific state"))?;
764                break (
765                    completion,
766                    Some(applied_mask),
767                    Some(receipt),
768                    minor_outcomes,
769                    cycle + 1,
770                    total_iterations,
771                    total_actual_iterations,
772                    Some(visibility_replay.completion()?),
773                    Some(visibility_replay),
774                    Some(output_receipt),
775                );
776            }
777        }
778        _ => unreachable!("native validation admits only dirty or Högbom"),
779    };
780
781    publish_products(
782        problem,
783        scientific,
784        final_reconstruction_mask,
785        runtime,
786        publication,
787        PriorPhaseOutcome {
788            initial_receipt,
789            final_major_receipt,
790            minor_cycles,
791            major_cycle_count,
792            total_minor_iterations,
793            total_actual_minor_iterations,
794            visibility_products,
795            visibility_replay,
796            visibility_output_receipt,
797        },
798    )
799}

801fn visibility_write_selection(
802    problem: &CompiledProblem,
803    selected: Arc<ObservationSelection>,
804) -> Result<Arc<ObservationSelection>, ApplicationError> {
805    let Some(transform) = problem.visibility_transform() else {
806        return Ok(selected);
807    };
808    let spectral_windows = selected
809        .spectral_windows()
810        .iter()
811        .map(|selection| {
812            let output_channels = selection
813                .channel_indices()
814                .iter()
815                .copied()
816                .filter(|channel| {
817                    transform.rules().iter().any(|rule| {
818                        rule.spectral_window_id() == selection.spectral_window_id()
819                            && rule
820                                .channel_use(*channel)
821                                .is_some_and(|role| role.contributes_to_output())
822                    })
823                })
824                .collect::<Vec<_>>();
825            if output_channels.is_empty() {
826                return Err(boxed(
827                    "continuum transform selected no visibility-write output channels",
828                ));
829            }
830            Ok(SpectralWindowSelection::new(
831                selection.spectral_window_id(),
832                output_channels,
833            ))
834        })
835        .collect::<Result<Vec<_>, ApplicationError>>()?;
836    Ok(Arc::new(ObservationSelection::new(
837        selected.rows().clone(),
838        selected.rows_filter().clone(),
839        selected.data_descriptions().to_vec(),
840        spectral_windows,
841        selected.correlations().to_vec(),
842    )))
843}

873fn run_phase(
874    problem: &CompiledProblem,
875    execution_plan: &casa_imaging_runtime::ExecutionPlan,
876    registry: &SpectralCycleRegistry<SpectralCycleExecutor>,
877    runtime: &ApplicationRuntime,
878    attempt: ExecutionAttemptId,
879) -> Result<(), ApplicationError> {
880    let executable = ExecutableModelProblem::from_compiled(problem.clone())?;
881    let current = RunBindings::new(
882        problem.inputs().clone(),
883        &runtime.resource_policy,
884        runtime.cost_model.profile_id(),
885    );
886    let mut controller = RunToCompletion;
887    run(
888        &executable,
889        execution_plan,
890        &current,
891        registry,
892        &runtime.authority,
893        &mut controller,
894        runtime
895            .receipts
896            .bind(ExecutionProvenance::new(attempt, runtime.build)),
897    )?;
898    Ok(())
899}

901fn major_cycle_attempt(base: ExecutionAttemptId, ordinal: u32) -> ExecutionAttemptId {
902    if ordinal == 1 {
903        return base;
904    }
905    let mut hash = Sha256::new();
906    hash.update(b"casa-rs:imaging:major-cycle-attempt:v1");
907    hash.update(base.as_bytes());
908    hash.update(ordinal.to_le_bytes());
909    ExecutionAttemptId::from_sha256(hash.finalize().into())
910}
911
912fn selected_output_attempt(science: ExecutionAttemptId) -> ExecutionAttemptId {
913    let mut hash = Sha256::new();
914    hash.update(b"casa-rs:imaging:selected-output-attempt:v1");
915    hash.update(science.as_bytes());
916    ExecutionAttemptId::from_sha256(hash.finalize().into())
917}

919fn publish_products<S>(
920    problem: &CompiledProblem,
921    scientific: MajorCycleCompletion,
922    reconstruction_mask: Option<casa_imaging_reconstruction::ReconstructionMask>,
923    runtime: ApplicationRuntime,
924    publication_config: ApplicationPublication<S>,
925    prior: PriorPhaseOutcome,
926) -> Result<NativeApplicationOutcome, ApplicationError>
927where
928    S: SerialProductPublicationSink + Send + 'static,
929    S::Error: Send + Sync,
930{
931    let sources = ContinuumSourceCatalog::from_major_cycle_with_mask(
932        problem,
933        &scientific,
934        reconstruction_mask.as_ref(),
935    )?;
936    let authority = ProductGenerationAuthority::bind(problem);
937    let planned_products = authority.plan(&sources, &publication_config.controls)?;
938
939    let visibility_write_receipt = prior
940        .visibility_replay
941        .as_ref()
942        .is_some_and(FinalVisibilityReplay::has_visibility_write)
943        .then(|| {
944            prior
945                .visibility_output_receipt
946                .clone()
947                .or_else(|| prior.final_major_receipt.clone())
948                .unwrap_or_else(|| prior.initial_receipt.clone())
949        });
950    let planning_registry =
951        PlanningRegistry::new(runtime.registry, runtime.implementation.clone(), problem);
952    // The ordinary publication plan is deliberately constructed before member
953    // production and sealing; the executor later presents the completed
954    // projection to the runtime for authorization at the commit gate.
955    let publication_plan = SerialProductPublicationPlan::new(
956        problem,
957        &planned_products,
958        &planning_registry,
959        SerialProductPublicationPolicy::new(
960            runtime.implementation.clone(),
961            runtime.storage_io.clone(),
962            runtime.stage_nanos,
963            runtime.confidence_parts_per_million,
964        ),
965    )?;
966    let mut inputs = ContinuumProductInputs::from_major_cycle(problem, &scientific)?;
967    if let Some(mask) = reconstruction_mask.as_ref() {
968        inputs = inputs.with_reconstruction_mask(mask)?;
969    }
970    let produced = produce_continuum_members(&planned_products, &inputs)?;
971    let sealed = authority.authorize(&planned_products, &produced)?;
972    let (physical, publication) = publication_plan.into_parts();
973    let executor = SerialProductPublicationExecutor::new(
974        runtime.implementation.clone(),
975        publication,
976        sealed,
977        publication_config.sink,
978    )?;
979    let registry = SerialProductPublicationRegistry::new(
980        runtime.registry,
981        runtime.implementation.clone(),
982        problem,
983        executor,
984    );
985    let execution_plan = plan(
986        problem,
987        PlanningBindings::new(
988            runtime.registry,
989            runtime.resource_policy.clone(),
990            runtime.cost_model,
991        ),
992        &runtime.authority,
993        &registry,
994        &runtime.receipts,
995        move |_, _| Ok::<_, std::convert::Infallible>(vec![physical]),
996    )?;
997    let executable = ExecutableModelProblem::from_compiled(problem.clone())?;
998    let current = RunBindings::new(
999        problem.inputs().clone(),
1000        &runtime.resource_policy,
1001        runtime.cost_model.profile_id(),
1002    );
1003    let mut controller = RunToCompletion;
1004    run(
1005        &executable,
1006        &execution_plan,
1007        &current,
1008        &registry,
1009        &runtime.authority,
1010        &mut controller,
1011        runtime
1012            .receipts
1013            .bind(ExecutionProvenance::new(runtime.attempts[2], runtime.build)),
1014    )?;
1015    let publication_receipt = runtime.receipts.open(runtime.attempts[2])?;
1016    let products = registry
1017        .implementation()
1018        .take_sealed_generation()
1019        .ok_or_else(|| boxed("publication execution omitted its sealed product generation"))?;
1020    Ok(NativeApplicationOutcome {
1021        initial_receipt: prior.initial_receipt,
1022        final_major_receipt: prior.final_major_receipt,
1023        minor_cycles: prior.minor_cycles,
1024        major_cycle_count: prior.major_cycle_count,
1025        total_minor_iterations: prior.total_minor_iterations,
1026        total_actual_minor_iterations: prior.total_actual_minor_iterations,
1027        visibility_products: prior.visibility_products,
1028        publication_receipt,
1029        visibility_write_receipt,
1030        scientific,
1031        planned_products,
1032        products,
1033    })
1034}

1114/// Failure before or within the installed whole-run implementation.
1115#[derive(Debug)]
1116pub enum ApplicationDispatchError {
1117    /// MeasurementSet resolution or request preparation failed before availability checking.
1118    Preparation(ApplicationError),
1119    /// Backend-independent request compilation failed.
1120    Compile(CompileProblemError),
1121    /// No installed implementation satisfies the compiled and task contract.
1122    Unavailable(ImplementationUnavailable),
1123    /// The sole installed implementation failed.
1124    Native(ApplicationError),
1125}
1126
1127impl fmt::Display for ApplicationDispatchError {
1128    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
1129        match self {
1130            Self::Preparation(error) => {
1131                write!(formatter, "imaging application preparation failed: {error}")
1132            }
1133            Self::Compile(error) => {
1134                write!(formatter, "imaging request compilation failed: {error}")
1135            }
1136            Self::Unavailable(error) => error.fmt(formatter),
1137            Self::Native(error) => write!(formatter, "native imaging run failed: {error}"),
1138        }
1139    }
1140}
1141
1142impl Error for ApplicationDispatchError {}

1145    Box::new(io::Error::other(message))
