34/// CASA `imstat` request.
35#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
36pub struct ImstatRequest {
37    /// Input CASA image path.
38    pub imagename: PathBuf,
39    /// CASA inclusive pixel box, formatted as `x0,y0,x1,y1`.
40    #[serde(default)]
41    pub box_pixels: Option<String>,
42    /// Region file path or inline CASA CRTF syntax, for example
43    /// `box[[100pix,100pix],[150pix,150pix]]` or a world-coordinate CRTF box.
44    #[serde(default)]
45    pub region: Option<String>,
46    /// CASA channel expression, supporting tutorial forms like `0~4`.
47    #[serde(default)]
48    pub chans: Option<String>,
49    /// Inclusive pixel-value range to include.
50    #[serde(default)]
51    pub includepix: Option<[f64; 2]>,
52    /// CASA `algorithm` selection, minimum-matched: `biweight`,
53    /// `chauvenet`, `classic`, `fit-half` or `hinges-fences`.
54    #[serde(default = "default_algorithm")]
55    pub algorithm: String,
56    /// The `hinges-fences` fence multiplier.
57    #[serde(default = "default_fence")]
58    pub fence: f64,
59    /// The `fit-half` center, minimum-matched: `mean`, `median` or `zero`.
60    #[serde(default = "default_center")]
61    pub center: String,
62    /// Whether `fit-half` uses the values at or above the center.
63    #[serde(default = "default_true")]
64    pub lside: bool,
65    /// The fixed `chauvenet` z-score, or a negative value for
66    /// Chauvenet's criterion.
67    #[serde(default = "default_zscore")]
68    pub zscore: f64,
69    /// The maximum `chauvenet` iterations; negative means until
70    /// convergence.
71    #[serde(default = "default_maxiter")]
72    pub maxiter: i64,
73    /// CASA `clmethod` selection for the classical method: `auto`,
74    /// `tiled` or `framework`, minimum-matched. The strategy does not
75    /// change the statistics.
76    #[serde(default = "default_clmethod")]
77    pub clmethod: String,
78    /// The maximum `biweight` iterations; negative takes the fast pass.
79    #[serde(default = "default_niter")]
80    pub niter: i64,
81    /// CASA `axes` selection: the axes to evaluate the statistics over,
82    /// as a pixel-axis list or `-1` for all of them. The default is all
83    /// axes.
84    #[serde(default)]
85    pub axes: Option<Vec<i64>>,
86    /// CASA `stokes` selection: Stokes plane names, as in `I`, `QU` or
87    /// `IQUV`, comma-separated and concatenated.
88    #[serde(default)]
89    pub stokes: Option<String>,
90    /// CASA `mask` expression: an image threshold expression, as in
91    /// `'"image.im" > 100'`, selecting the pixels the statistics use.
92    #[serde(default)]
93    pub mask: Option<String>,
94    /// Whether CASA would stretch a mask whose shape differs; the
95    /// expressions this surface reaches are same-shaped, so the value is
96    /// accepted and inert.
97    #[serde(default = "default_false")]
98    pub stretch: bool,
99    /// CASA `logfile`: a file to write the statistics block to, in
100    /// addition to the logger.
101    #[serde(default)]
102    pub logfile: Option<String>,
103    /// Whether the logfile is appended to rather than overwritten.
104    #[serde(default = "default_true")]
105    pub append: bool,
106    /// Whether the statistics block is printed to the logger; the
107    /// returned record is the same either way.
108    #[serde(default = "default_true")]
109    pub listit: bool,
110    /// Whether additional messages are printed to the logger; the
111    /// returned record is the same either way.
112    #[serde(default = "default_true")]
113    pub verbose: bool,
114}
115
116fn default_false() -> bool {
117    false
118}

181/// One `imstat` per-plane outcome: statistics evaluated over the
182/// collapsed axes, one value per display position, in display order.
183#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
184pub struct ImstatPlanesSummary {
185    /// Input image path.
186    pub imagename: String,
187    /// Applied zero-based origin of the selection.
188    pub blc: Vec<usize>,
189    /// Applied zero-based inclusive top-right corner of the selection.
190    pub trc: Vec<usize>,
191    /// CASA-formatted world coordinates of the applied origin.
192    pub blcf: String,
193    /// CASA-formatted world coordinates of the applied top-right corner.
194    pub trcf: String,
195    /// The display shape, with CASA's degenerate display axes dropped.
196    pub display_shape: Vec<usize>,
197    /// Text table of plane coordinates and statistics for the task log.
198    #[serde(skip)]
199    pub plane_log: String,
200    /// Number of points in each plane.
201    pub npts: Vec<f64>,
202    /// Each plane's minimum.
203    pub min: Vec<f64>,
204    /// Each plane's maximum.
205    pub max: Vec<f64>,
206    /// Each plane's sum.
207    pub sum: Vec<f64>,
208    /// Each plane's sum of squares.
209    pub sumsq: Vec<f64>,
210    /// Each plane's mean.
211    pub mean: Vec<f64>,
212    /// Each plane's root-mean-square.
213    pub rms: Vec<f64>,
214    /// Each plane's Bessel-corrected standard deviation.
215    pub sigma: Vec<f64>,
216    /// Each plane's median.
217    pub median: Vec<f64>,
218    /// Each plane's median absolute deviation.
219    pub medabsdevmed: Vec<f64>,
220    /// Each plane's first quartile.
221    pub q1: Vec<f64>,
222    /// Each plane's third quartile.
223    pub q3: Vec<f64>,
224    /// Each plane's interquartile range.
225    pub quartile: Vec<f64>,
226    /// Each plane's flux density, when the image carries a beam.
227    pub flux: Option<Vec<f64>>,
228    /// Whether the biweight's location and scale replace the mean and
229    /// sigma, with no other statistics in the record.
230    pub biweight: bool,
231}

989impl ImageHeaderSummary {
990    /// Render a compact, human-readable summary of the image header.
991    ///
992    /// This is the casa-rs text presentation, rather than the detailed CASA
993    /// coordinate-system log emitted by `imhead` summary mode.
994    pub fn format_text(&self) -> String {
995        use std::fmt::Write as _;
996
997        let mut out = String::new();
998        let _ = writeln!(out, "Image name       : {}", self.imagename);
999        let _ = writeln!(out, "Image type       : {}", self.image_type);
1000        let _ = writeln!(out, "Pixel type       : {}", self.pixel_type);
1001        let _ = writeln!(out, "Image units      : {}", self.units);
1002        let _ = writeln!(out, "Shape            : {:?}", self.shape);
1003        if let Some(beam) = &self.restoring_beam {
1004            let _ = writeln!(
1005                out,
1006                "Restoring Beam   : {:.6} arcsec, {:.6} arcsec, {:.6} deg",
1007                beam.major_arcsec, beam.minor_arcsec, beam.position_angle_deg
1008            );
1009        }
1010        for axis in &self.axes {
1011            let _ = writeln!(
1012                out,
1013                "Axis {} {:<12} {:<18} shape={} refpix={:.3} crval={:.12e} cdelt={:.12e} {}",
1014                axis.axis,
1015                axis.coordinate_type,
1016                axis.name,
1017                axis.shape,
1018                axis.reference_pixel,
1019                axis.reference_value,
1020                axis.increment,
1021                axis.unit
1022            );
1023        }
1024        out
1025    }
1026}
