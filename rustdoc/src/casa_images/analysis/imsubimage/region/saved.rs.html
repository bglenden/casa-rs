15pub(super) fn load_saved_region(
16    image_path: &Path,
17    region_name: &str,
18    target_coords: &CoordinateSystem,
19    full_shape: &[usize],
20    plane: [usize; 2],
21) -> Result<ParsedRegion, ImageError> {
22    let rec = match AnyPagedImage::open(image_path)? {
23        AnyPagedImage::Float32(img) => img.get_region_record(region_name)?,
24        AnyPagedImage::Float64(img) => img.get_region_record(region_name)?,
25        AnyPagedImage::Complex32(img) => img.get_region_record(region_name)?,
26        AnyPagedImage::Complex64(img) => img.get_region_record(region_name)?,
27    };
28    parse_saved_region(region_name, &rec, target_coords, full_shape, plane)
29}

31pub(super) fn parse_saved_region(
32    region_name: &str,
33    rec: &RecordValue,
34    target_coords: &CoordinateSystem,
35    full_shape: &[usize],
36    plane: [usize; 2],
37) -> Result<ParsedRegion, ImageError> {
38    let class_name = match rec.get("name") {
39        Some(Value::Scalar(ScalarValue::String(name))) => name.as_str(),
40        _ => "",
41    };
42
43    match class_name {
44        "WCPolygon" | "WCUnion" => {
45            let poly = PolygonRegion::from_record(region_name, rec, target_coords)?;
46            let mask = poly.mask(target_coords, full_shape)?;
47            let mut min_pt = full_shape.to_vec();
48            let mut max_pt = vec![0usize; full_shape.len()];
49            let mut any_true = false;
50            for (idx, &valid) in mask.indexed_iter() {
51                if valid {
52                    any_true = true;
53                    for (ax, &pos) in idx.slice().iter().enumerate() {
54                        min_pt[ax] = min_pt[ax].min(pos);
55                        max_pt[ax] = max_pt[ax].max(pos);
56                    }
57                }
58            }
59            if !any_true {
60                return Err(ImageError::CasaSelection(format!(
61                    "saved region '{region_name}' selects no pixels in image"
62                )));
63            }
64            let shape: Vec<usize> = min_pt
65                .iter()
66                .zip(max_pt.iter())
67                .map(|(lo, hi)| hi - lo + 1)
68                .collect();
69            Ok(ParsedRegion {
70                shapes: Vec::new(),
71                saved_polygon_mask: Some(mask),
72                saved_bounds: Some((min_pt, shape)),
73            })
74        }
75        "LCSlicer" | "LCBox" => {
76            let (blc, trc) = if class_name == "LCSlicer" {
77                decode_lc_slicer(region_name, rec, target_coords, full_shape)?
78            } else {
79                decode_lc_box(region_name, rec, full_shape)?
80            };
81            let shapes = if full_shape.len() >= 2 {
82                vec![
83                    SpatialShape::Box([blc[plane[0]], blc[plane[1]], trc[plane[0]], trc[plane[1]]])
84                        .into(),
85                ]
86            } else {
87                Vec::new()
88            };
89            let shape: Vec<usize> = blc
90                .iter()
91                .zip(trc.iter())
92                .map(|(lo, hi)| hi - lo + 1)
93                .collect();
94            Ok(ParsedRegion {
95                shapes,
96                saved_polygon_mask: None,
97                saved_bounds: Some((blc, shape)),
98            })
99        }
100        "WCBox" => {
101            let (blc, trc) = decode_wc_box(region_name, rec, target_coords, full_shape)?;
102            let shapes = if full_shape.len() >= 2 {
103                vec![
104                    SpatialShape::Box([blc[plane[0]], blc[plane[1]], trc[plane[0]], trc[plane[1]]])
105                        .into(),
106                ]
107            } else {
108                Vec::new()
109            };
110            let shape: Vec<usize> = blc
111                .iter()
112                .zip(trc.iter())
113                .map(|(lo, hi)| hi - lo + 1)
114                .collect();
115            Ok(ParsedRegion {
116                shapes,
117                saved_polygon_mask: None,
118                saved_bounds: Some((blc, shape)),
119            })
120        }
121        "WCEllipsoid" => Ok(ParsedRegion {
122            shapes: vec![
123                decode_wc_ellipsoid(region_name, rec, target_coords, full_shape, plane)?.into(),
124            ],
125            saved_polygon_mask: None,
126            saved_bounds: None,
127        }),
128        other => Err(ImageError::InvalidMetadata(format!(
129            "unsupported saved region '{region_name}' of class {other:?}"
130        ))),
131    }
132}

134fn decode_lc_slicer(
135    name: &str,
136    rec: &RecordValue,
137    coords: &CoordinateSystem,
138    full_shape: &[usize],
139) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
140    let numeric = |field| {
141        read_numeric_vec(rec.get(field)).ok_or_else(|| {
142            ImageError::InvalidMetadata(format!("saved region '{name}' missing numeric {field}"))
143        })
144    };
145    let one_rel = matches!(
146        rec.get("oneRel"),
147        Some(Value::Scalar(ScalarValue::Bool(true)))
148    );
149    let refpix = (0..coords.n_coordinates())
150        .flat_map(|i| coords.coordinate(i).reference_pixel())
151        .collect::<Vec<_>>();
152    let [lower, upper, step] = [numeric("blc")?, numeric("trc")?, numeric("inc")?];
153    let ndim = lower.len().max(upper.len()).max(step.len());
154    if ndim > full_shape.len() {
155        return Err(ImageError::ShapeMismatch {
156            expected: vec![full_shape.len()],
157            got: vec![ndim],
158        });
159    }
160    let resolve = |values: &[f64], fraction_key, mode_key, low| {
161        let fractions = read_bool_vec(rec.get(fraction_key), values.len());
162        let modes = numeric(mode_key)?;
163        if modes.len() != values.len() {
164            return Err(ImageError::InvalidMetadata(format!(
165                "saved region '{name}' has incorrect {mode_key} length"
166            )));
167        }
168        full_shape
169            .iter()
170            .enumerate()
171            .map(|(axis, &len)| {
172                let default = if low { 0 } else { len - 1 } as i64;
173                let Some(&stored) = values.get(axis) else {
174                    return Ok(default);
175                };
176                if is_mimic_source(stored) {
177                    return Ok(default);
178                }
179                let fractional = fractions[axis];
180                let mut pixel = stored as f32;
181                if fractional {
182                    pixel = pixel * len as f32 - if low { 0.0 } else { 1.0 };
183                }
184                pixel += match modes[axis] {
185                    1.0 if one_rel && !fractional => -1.0,
186                    1.0 => 0.0,
187                    2.0 => *refpix.get(axis).ok_or_else(|| {
188                        ImageError::InvalidMetadata(format!(
189                            "saved region '{name}' has no reference pixel on axis {axis}"
190                        ))
191                    })? as f32,
192                    3.0 => len as f32 / 2.0,
193                    _ => {
194                        return Err(ImageError::InvalidMetadata(format!(
195                            "saved region '{name}' has invalid {mode_key} on axis {axis}"
196                        )));
197                    }
198                };
199                if !pixel.is_finite() {
200                    return Err(ImageError::InvalidMetadata(format!(
201                        "saved region '{name}' has non-finite bounds on axis {axis}"
202                    )));
203                }
204                let pixel = (pixel + 0.5) as i64;
205                Ok(if low {
206                    pixel.max(0)
207                } else {
208                    pixel.min(default)
209                })
210            })
211            .collect::<Result<Vec<_>, _>>()
212    };
213    let blc = resolve(&lower, "fracblc", "arblc", true)?;
214    let trc = resolve(&upper, "fractrc", "artrc", false)?;
215    if blc.iter().zip(&trc).any(|(lo, hi)| lo > hi) {
216        return Err(ImageError::CasaSelection(format!(
217            "saved region '{name}' lower bounds {blc:?} exceed upper bounds {trc:?}"
218        )));
219    }
220    let fractional_step = read_bool_vec(rec.get("fracinc"), step.len());
221    for (axis, (&stored, fractional)) in step.iter().zip(fractional_step).enumerate() {
222        let stride = if is_mimic_source(stored) {
223            1.0
224        } else if fractional {
225            stored as f32 * full_shape[axis] as f32
226        } else {
227            stored as f32
228        };
229        if !(1.0..2.0).contains(&(stride + 0.5)) {
230            return Err(ImageError::InvalidMetadata(format!(
231                "saved region '{name}' has non-unit stride {stride} on axis {axis}"
232            )));
233        }
234    }
235    let as_indices = |values: Vec<i64>| values.into_iter().map(|value| value as usize).collect();
236    Ok((as_indices(blc), as_indices(trc)))
237}

239fn is_mimic_source(value: f64) -> bool {
240    // The integer sentinel is imprecise in the record's Float32 representation.
241    value < -2_147_483_646.0 + 10.0
242}

244fn decode_lc_box(
245    name: &str,
246    rec: &RecordValue,
247    full_shape: &[usize],
248) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
249    let one_rel = matches!(
250        rec.get("oneRel"),
251        Some(Value::Scalar(ScalarValue::Bool(true)))
252    );
253    let blc_raw = read_numeric_vec(rec.get("blc")).ok_or_else(|| {
254        ImageError::InvalidMetadata(format!("saved region '{name}' missing numeric blc"))
255    })?;
256    let trc_raw = read_numeric_vec(rec.get("trc")).ok_or_else(|| {
257        ImageError::InvalidMetadata(format!("saved region '{name}' missing numeric trc"))
258    })?;
259    if let Some(inc) = read_numeric_vec(rec.get("inc"))
260        && inc.iter().any(|&v| (v - 1.0).abs() > 1e-6)
261    {
262        return Err(ImageError::InvalidMetadata(format!(
263            "saved region '{name}' has non-unit stride {inc:?}"
264        )));
265    }
266    let ndim = full_shape.len();
267    if blc_raw.len() != ndim || trc_raw.len() != ndim {
268        return Err(ImageError::ShapeMismatch {
269            expected: vec![ndim],
270            got: vec![blc_raw.len()],
271        });
272    }
273    let frac_blc = read_bool_vec(rec.get("fracblc"), ndim);
274    let frac_trc = read_bool_vec(rec.get("fractrc"), ndim);
275    let mut blc = Vec::with_capacity(ndim);
276    let mut trc = Vec::with_capacity(ndim);
277    for axis in 0..ndim {
278        let len = full_shape[axis];
279        let lo = if frac_blc[axis] {
280            (blc_raw[axis] * (len.saturating_sub(1)) as f64).round()
281        } else if one_rel {
282            (blc_raw[axis] - 1.0).round()
283        } else {
284            blc_raw[axis].round()
285        };
286        let hi = if frac_trc[axis] {
287            (trc_raw[axis] * (len.saturating_sub(1)) as f64).round()
288        } else if one_rel {
289            (trc_raw[axis] - 1.0).round()
290        } else {
291            trc_raw[axis].round()
292        };
293        if !lo.is_finite() || !hi.is_finite() || lo < 0.0 || hi < lo || (hi as usize) >= len {
294            return Err(ImageError::CasaSelection(format!(
295                "saved region '{name}' bounds [{lo}, {hi}] out of range for axis {axis} (len {len})"
296            )));
297        }
298        blc.push(lo as usize);
299        trc.push(hi as usize);
300    }
301    Ok((blc, trc))
302}

304fn decode_wc_box(
305    name: &str,
306    rec: &RecordValue,
307    coords: &CoordinateSystem,
308    full_shape: &[usize],
309) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
310    let (axes, origin) = wc_pixel_axes(name, "WCBox", rec, coords)?;
311    let mut modes = read_numeric_vec(rec.get("absrel")).unwrap_or_default();
312    if modes.is_empty() {
313        modes.resize(axes.len(), 1.0);
314    }
315    let mut bounds = Vec::new();
316    for (key, low) in [("blc", true), ("trc", false)] {
317        let quantities = read_quantum_list(rec.get(key))?;
318        if quantities.len() != axes.len() || modes.len() != axes.len() {
319            return Err(ImageError::InvalidMetadata(format!(
320                "WCBox {name:?} dimension mismatch"
321            )));
322        }
323        bounds.push(wc_box_corner(
324            coords,
325            full_shape,
326            &axes,
327            &quantities,
328            &modes,
329            origin,
330            low,
331        )?);
332    }
333    let trc = bounds.pop().expect("upper corner");
334    let blc = bounds.pop().expect("lower corner");
335    if blc.iter().zip(&trc).any(|(lower, upper)| lower > upper) {
336        return Err(ImageError::CasaSelection(format!(
337            "WCBox {name:?} has reversed bounds"
338        )));
339    }
340    Ok((blc, trc))
341}

343fn wc_pixel_axes(
344    name: &str,
345    class: &str,
346    rec: &RecordValue,
347    coords: &CoordinateSystem,
348) -> Result<(Vec<usize>, usize), ImageError> {
349    let origin = usize::from(matches!(
350        rec.get("oneRel"),
351        Some(Value::Scalar(ScalarValue::Bool(true)))
352    ));
353    let Some(Value::Record(source_record)) = rec.get("coordinates") else {
354        return Err(ImageError::InvalidMetadata(format!(
355            "{class} has no coordinate system"
356        )));
357    };
358    let saved_coords = CoordinateSystem::from_record(source_record)?;
359    let names = |system: &CoordinateSystem| {
360        (0..system.n_coordinates())
361            .flat_map(|i| system.coordinate(i).axis_names())
362            .collect::<Vec<_>>()
363    };
364    let saved_names = names(&saved_coords);
365    let target_names = names(coords);
366    let axes = read_numeric_vec(rec.get("pixelAxes"))
367        .ok_or_else(|| ImageError::InvalidMetadata(format!("{class} has no pixelAxes")))?
368        .into_iter()
369        .map(|axis| {
370            (axis as usize)
371                .checked_sub(origin)
372                .and_then(|i| saved_names.get(i))
373                .and_then(|name| target_names.iter().position(|candidate| candidate == name))
374                .ok_or_else(|| {
375                    ImageError::InvalidMetadata(format!(
376                        "{class} {name:?} has an unmatched pixel axis {axis}"
377                    ))
378                })
379        })
380        .collect::<Result<Vec<_>, _>>()?;
381    Ok((axes, origin))
382}

384fn decode_wc_ellipsoid(
385    name: &str,
386    rec: &RecordValue,
387    coords: &CoordinateSystem,
388    shape: &[usize],
389    plane: [usize; 2],
390) -> Result<SpatialShape, ImageError> {
391    let invalid =
392        || ImageError::InvalidMetadata(format!("invalid direction-plane WCEllipsoid {name:?}"));
393    let (axes, origin) = wc_pixel_axes(name, "WCEllipsoid", rec, coords)?;
394    if axes != plane
395        || !matches!(
396            rec.get("absrel"),
397            Some(Value::Scalar(ScalarValue::Int32(1)))
398        )
399    {
400        return Err(invalid());
401    }
402    let [x, y] = plane;
403    let center = read_quantum_list(rec.get("center"))?;
404    let [first, second] = center.as_slice() else {
405        return Err(invalid());
406    };
407    let rad = Unit::new("rad").expect("radians are a built-in unit");
408    let in_radians = |quantity: &Quantity| quantity.get_value_in(&rad).map_err(|_| invalid());
409    let [cx, cy] = if matches!(first.unit().name(), "pix" | "frac")
410        && matches!(second.unit().name(), "pix" | "frac")
411    {
412        let pixel = |q: &Quantity, axis: usize| match q.unit().name() {
413            "frac" => q.value() * shape[axis] as f64,
414            _ => q.value() - origin as f64,
415        };
416        [pixel(first, x), pixel(second, y)]
417    } else {
418        let mut world = (0..coords.n_coordinates())
419            .flat_map(|i| coords.coordinate(i).reference_value())
420            .collect::<Vec<_>>();
421        world[x] = in_radians(first)?;
422        world[y] = in_radians(second)?;
423        let pixel = coords.to_pixel(&world)?;
424        [pixel[x], pixel[y]]
425    };
426    let scales = super::direction_pixel_scales(coords, plane).ok_or_else(invalid)?;
427    let radius = |q: &Quantity, axis: usize| {
428        let size = if q.unit().name() == "pix" {
429            q.value()
430        } else {
431            in_radians(q)? / scales[axis]
432        };
433        if !size.is_finite() || size <= 0.0 {
434            return Err(invalid());
435        }
436        Ok(size)
437    };
438    match rec.get("type") {
439        Some(Value::Scalar(ScalarValue::Int32(1))) => {
440            let Some(Value::Record(value)) = rec.get("radius") else {
441                return Err(invalid());
442            };
443            let r = radius(&crate::beam::quantity_from_record(value)?, 0)?;
444            Ok(SpatialShape::Circle {
445                cx,
446                cy,
447                rx: r,
448                ry: r,
449            })
450        }
451        Some(Value::Scalar(ScalarValue::Int32(0 | 2))) => {
452            let values = read_quantum_list(rec.get("radii"))?;
453            let [major, minor] = values.as_slice() else {
454                return Err(invalid());
455            };
456            let (a, b) = (radius(major, 0)?, radius(minor, 1)?);
457            if matches!(rec.get("type"), Some(Value::Scalar(ScalarValue::Int32(0)))) {
458                return Ok(SpatialShape::Circle {
459                    cx,
460                    cy,
461                    rx: a,
462                    ry: b,
463                });
464            }
465            let Some(Value::Record(value)) = rec.get("theta") else {
466                return Err(invalid());
467            };
468            let pa = in_radians(&crate::beam::quantity_from_record(value)?)?
469                - std::f64::consts::FRAC_PI_2;
470            Ok(SpatialShape::Ellipse {
471                cx,
472                cy,
473                semi_major: a,
474                semi_minor: b,
475                cos_pa: pa.cos(),
476                sin_pa: pa.sin(),
477            })
478        }
479        _ => Err(invalid()),
480    }
481}

483fn wc_box_corner(
484    coords: &CoordinateSystem,
485    shape: &[usize],
486    axes: &[usize],
487    quantities: &[Quantity],
488    modes: &[f64],
489    origin: usize,
490    low: bool,
491) -> Result<Vec<usize>, ImageError> {
492    let mut world = (0..coords.n_coordinates())
493        .flat_map(|i| coords.coordinate(i).reference_value())
494        .collect::<Vec<_>>();
495    let reference_pixels = (0..coords.n_coordinates())
496        .flat_map(|i| coords.coordinate(i).reference_pixel())
497        .collect::<Vec<_>>();
498    let units = (0..coords.n_coordinates())
499        .flat_map(|i| coords.coordinate(i).axis_units())
500        .collect::<Vec<_>>();
501    for ((&axis, quantity), &mode) in axes.iter().zip(quantities).zip(modes) {
502        if !matches!(quantity.unit().name(), "pix" | "frac" | "default") {
503            if mode != 1.0 {
504                return Err(ImageError::InvalidMetadata(
505                    "relative WCBox world bounds are unsupported".into(),
506                ));
507            }
508            let unit = Unit::new(&units[axis])
509                .map_err(|err| ImageError::InvalidMetadata(err.to_string()))?;
510            world[axis] = quantity
511                .get_value_in(&unit)
512                .map_err(|err| ImageError::InvalidMetadata(err.to_string()))?;
513        }
514    }
515    let converted = coords.to_pixel(&world)?;
516    let mut result = shape
517        .iter()
518        .map(|len| if low { 0 } else { len - 1 })
519        .collect::<Vec<_>>();
520    for ((&axis, quantity), &mode) in axes.iter().zip(quantities).zip(modes) {
521        let unit = quantity.unit().name();
522        if unit == "default" {
523            continue;
524        }
525        let limit = (shape[axis] - 1) as f64;
526        let relative = match mode {
527            1.0 => 0.0,
528            2.0 => reference_pixels[axis],
529            3.0 => shape[axis] as f64 * 0.5,
530            _ => return Err(ImageError::InvalidMetadata("invalid WCBox absrel".into())),
531        };
532        let pixel = match unit {
533            "pix" => quantity.value() + relative - if mode == 1.0 { origin as f64 } else { 0.0 },
534            "frac" => quantity.value() * shape[axis] as f64 + relative - f64::from(!low),
535            _ => converted[axis],
536        };
537        result[axis] = pixel.round().clamp(0.0, limit) as usize;
538    }
539    Ok(result)
540}

542fn read_quantum_list(val: Option<&Value>) -> Result<Vec<Quantity>, ImageError> {
543    let Some(Value::Record(rec)) = val else {
544        return Err(ImageError::InvalidMetadata(
545            "saved region is missing a quantity list".into(),
546        ));
547    };
548    rec.fields()
549        .iter()
550        .map(|field| match &field.value {
551            Value::Record(value) => crate::beam::quantity_from_record(value),
552            _ => Err(ImageError::InvalidMetadata(
553                "invalid saved region quantity".into(),
554            )),
555        })
556        .collect()
557}
558
559fn read_numeric_vec(val: Option<&Value>) -> Option<Vec<f64>> {
560    match val? {
561        Value::Array(ArrayValue::Float64(arr)) => Some(arr.iter().copied().collect()),
562        Value::Array(ArrayValue::Float32(arr)) => Some(arr.iter().map(|&v| f64::from(v)).collect()),
563        Value::Array(ArrayValue::Int32(arr)) => Some(arr.iter().map(|&v| f64::from(v)).collect()),
564        Value::Array(ArrayValue::Int64(arr)) => Some(arr.iter().map(|&v| v as f64).collect()),
565        _ => None,
566    }
567}

569fn read_bool_vec(val: Option<&Value>, ndim: usize) -> Vec<bool> {
570    if let Some(Value::Array(ArrayValue::Bool(arr))) = val
571        && arr.len() == ndim
572    {
573        return arr.iter().copied().collect();
574    }
575    vec![false; ndim]
