20/// A single-dish visibility convolution and its noise propagation kernel.
21pub trait SpectrumConvolution {
22    /// Initialize kernels for the original channel counts of the selected windows.
23    fn prepare(&mut self, channel_counts: &[usize]) -> Result<(), String>;
24
25    /// Replace a selected visibility cell with its convolved spectrum.
26    fn apply(&mut self, data: &mut ArrayValue, flags: &ArrayD<bool>) -> Result<(), String>;
27
28    /// Coefficients used to propagate channel weights and errors.
29    fn weight_kernel(&self) -> &[f32];
30}
31
32pub(super) enum SpectrumOperator<'a> {
33    Convolution(&'a mut dyn SpectrumConvolution),
34    Baseline(&'a mut dyn SpectrumBaseline),
35}
36
37pub(super) struct SpectralOperation<'a> {
38    pub(super) kernel: SpectrumOperator<'a>,
39    pub(super) reindex: bool,
40}

42pub(super) fn auto_correlations(
43    input: &MeasurementSet,
44    selector: Option<&str>,
45) -> Result<String, MsTransformError> {
46    let Some(selector) = selector.filter(|text| !text.is_empty()) else {
47        return Ok("*&&&".into());
48    };
49    let ids = input
50        .signed_antenna1_indices(selector)
51        .map_err(|source| MsTransformError::Spectrum(source.to_string()))?;
52    let rewritten = ids
53        .iter()
54        .map(|antenna| format!("{antenna}&&&"))
55        .collect::<Vec<_>>()
56        .join(";");
57    if let Some(index) = rewritten.find('-') {
58        return Err(MsTransformError::Spectrum(format!(
59            "Antenna Expression: Parse error at or near '-'\n(near char. {index} in string \"{rewritten}\")"
60        )));
61    }
62    if rewritten.is_empty() {
63        return Ok("*&&&".into());
64    }
65    Ok(rewritten)
66}

68pub(super) fn statistics(
69    weights: ArrayValue,
70    sigmas: ArrayValue,
71    flags: &ArrayD<bool>,
72    kernel: &[f32],
73    corrected: bool,
74) -> Result<AveragedStatistics, TableError> {
75    let input = if corrected { weights } else { sigmas };
76    let ArrayValue::Float32(input) = input else {
77        return Err(TableError::Schema(
78            "Spectral weights must be Float32.".into(),
79        ));
80    };
81    let &[correlations, channels] = flags.shape() else {
82        return Err(TableError::Schema(
83            "Spectral flags must have two axes.".into(),
84        ));
85    };
86    let mut weight_spectrum = Array2::zeros((correlations, channels));
87    let mut sigma_spectrum = Array2::zeros((correlations, channels));
88    let mut weight = Array1::zeros(correlations);
89    let mut sigma = Array1::zeros(correlations);
90    let half = kernel.len() / 2;
91    for (correlation, mask) in flags.axis_iter(Axis(0)).enumerate() {
92        let incoming = input.index_axis(Axis(0), correlation);
93        let values = incoming
94            .iter()
95            .map(|&value| {
96                if corrected {
97                    value
98                } else {
99                    weight_from_sigma(value)
100                }
101            })
102            .collect::<Vec<_>>();
103        let mut valid = 0;
104        let mut weight_sum = 0.;
105        let mut sigma_sum = 0.;
106        for (channel, &original) in values.iter().enumerate() {
107            let interior = half <= channel && channel < channels.saturating_sub(half);
108            let (weighted, flag) = if interior {
109                let from = channel - half;
110                let mut inverse = 0.;
111                let mut flagged = false;
112                for (offset, &coefficient) in kernel.iter().enumerate() {
113                    let value = values[from + offset];
114                    if value > f32::MIN_POSITIVE {
115                        inverse += coefficient * coefficient / value;
116                        flagged |= mask[from + offset];
117                    } else {
118                        flagged = true;
119                    }
120                }
121                (
122                    if inverse > f32::MIN_POSITIVE {
123                        inverse.recip()
124                    } else {
125                        inverse
126                    },
127                    flagged,
128                )
129            } else {
130                (original, true)
131            };
132            let deviation = sigma_from_weight(weighted);
133            weight_spectrum[[correlation, channel]] = weighted;
134            sigma_spectrum[[correlation, channel]] = deviation;
135            if !flag {
136                weight_sum += weighted;
137                sigma_sum += deviation;
138                valid += 1;
139            }
140        }
141        if valid == 0 {
142            weight_sum = weight_spectrum.row(correlation).sum();
143            sigma_sum = sigma_spectrum.row(correlation).sum();
144            valid = channels;
145        }
146        weight[correlation] = weight_sum / valid as f32;
147        sigma[correlation] = sigma_sum / valid as f32;
148    }
149    Ok(AveragedStatistics {
150        weight: ArrayValue::Float32(weight.into_dyn()),
151        sigma: ArrayValue::Float32(sigma.into_dyn()),
152        weight_spectrum: ArrayValue::Float32(weight_spectrum.into_dyn()),
153        sigma_spectrum: ArrayValue::Float32(sigma_spectrum.into_dyn()),
154    })
155}

157/// Convolve the selected autocorrelation spectra and publish a complete MS.
158///
159/// An existing output is held for a recoverable exchange, leaving it readable
160/// if the selection or convolution fails.
161pub fn convolve_spectra(
162    request: &MsTransformRequest,
163    options: &MsTransformTaskOptions,
164    reindex: bool,
165    overwrite: bool,
166    kernel: &mut dyn SpectrumConvolution,
167) -> Result<MsTransformReport, MsTransformError> {
168    stage_spectra(
169        request,
170        options,
171        SpectralOperation {
172            kernel: SpectrumOperator::Convolution(kernel),
173            reindex,
174        },
175        overwrite,
176    )
177}

201fn stage_spectra(
202    request: &MsTransformRequest,
203    options: &MsTransformTaskOptions,
204    operation: SpectralOperation<'_>,
205    overwrite: bool,
206) -> Result<MsTransformReport, MsTransformError> {
207    let destination = &request.output_ms;
208    let path_error = |reason: String| MsTransformError::PrepareOutput {
209        path: destination.display().to_string(),
210        reason,
211    };
212    let name = destination
213        .file_name()
214        .ok_or_else(|| path_error("The output path must end in a name.".into()))?;
215    let parent = destination
216        .parent()
217        .filter(|path| !path.as_os_str().is_empty())
218        .unwrap_or_else(|| Path::new("."));
219    let parent = fs::canonicalize(parent).map_err(|error| path_error(error.to_string()))?;
220    let final_path = parent.join(name);
221    let input =
222        fs::canonicalize(&request.input_ms).map_err(|error| path_error(error.to_string()))?;
223    let existing = final_path.symlink_metadata().is_ok();
224    let output = fs::canonicalize(&final_path).unwrap_or_else(|_| final_path.clone());
225    if input.starts_with(&output) || output.starts_with(&input) {
226        return Err(MsTransformError::SameInputOutput {
227            path: destination.display().to_string(),
228        });
229    }
230    if existing && !overwrite {
231        return Err(MsTransformError::OutputExists {
232            path: destination.display().to_string(),
233        });
234    }
235    let replacement = existing
236        .then(|| Replacement::hold(&final_path))
237        .transpose()
238        .map_err(|error| path_error(error.to_string()))?;
239    let stage = Stage::reserve(&final_path, "casa-rs-spectral")
240        .map_err(|error| path_error(error.to_string()))?;
241    let mut physical = request.clone();
242    physical.output_ms = stage.path().to_path_buf();
243    let product = super::transform_product(
244        &physical,
245        RowGrammar::Transform,
246        options,
247        None,
248        Some(operation),
249    );
250    let commit = product.and_then(|mut report| {
251        if let Some(replacement) = replacement {
252            replacement
253                .publish(stage.path())
254                .map_err(|error| path_error(error.to_string()))?;
255        } else {
256            stage
257                .publish_new(&final_path)
258                .map_err(|error| path_error(error.to_string()))?;
259        }
260        report.output_ms = destination.clone();
261        Ok(report)
262    });
263    if commit.is_ok() {
264        stage.release();
265    } else {
266        stage.remove();
267    }
268    commit
269}
