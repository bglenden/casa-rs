2//! The `table` tool's slices of array cells and columns (`getcellslice`,
3//! `getcolslice`, `putcellslice`, `putcolslice`), as CASA 6.7.6.14's `tb`
4//! does them (`tools/casa-compat/table_tool_oracle.py`, `slices` and
5//! `ms_slices`).
6//!
7//! A slice is given by its bottom-left corner `blc`, top-right corner `trc`
8//! (both inclusive) and step `incr`, one element of each per axis. CASA
9//! aborts its whole process when a corner has the wrong rank or a slice
10//! starts outside the array; casa-rs raises a `RuntimeError` for those
11//! (`docs/CASA (C++) bugs.md`).
12

25type Reply = Result<CasaValue, Fail>;

27fn runtime<T>(message: String) -> Result<T, Fail> {
28    Err(Fail::Runtime(message))
29}

31/// A slice's corners and step, as the caller gave them.
32#[derive(Debug, Clone)]
33pub(crate) struct Corners {
34    pub(crate) blc: Vec<i64>,
35    pub(crate) trc: Vec<i64>,
36    pub(crate) incr: Vec<i64>,
37}
38
39impl Corners {
40    /// The slicer over an array of `shape` and the shape the slice has.
41    /// `trc` may be `-1` (the last element) where `end_ok`, as `getcolslice`
42    /// accepts it and `getcellslice` does not.
43    fn over(&self, shape: &[usize], end_ok: bool) -> Result<(Slicer, Vec<usize>), Fail> {
44        let rank = shape.len();
45        // The default `incr=[1]` steps every axis by one (CASA aborts on it
46        // for an array of more than one axis).
47        let mut incr = self.incr.clone();
48        if incr.len() == 1 && rank > 1 {
49            incr = vec![incr[0]; rank];
50        }
51        if self.blc.len() != rank || self.trc.len() != rank || incr.len() != rank {
52            return runtime(format!(
53                "blc, trc and incr need {rank} elements each, one per axis of the array"
54            ));
55        }
56        let mut start = Vec::with_capacity(rank);
57        let mut end = Vec::with_capacity(rank);
58        let mut stride = Vec::with_capacity(rank);
59        let mut result = Vec::with_capacity(rank);
60        for axis in 0..rank {
61            let dim = shape[axis];
62            let (blc, trc, incr) = (self.blc[axis], self.trc[axis], incr[axis]);
63            if blc < 0 {
64                return runtime(
65                    "Exception: All elements of blc must be greater than or equal to 0.".into(),
66                );
67            }
68            let trc = if trc == -1 && end_ok {
69                i64::try_from(dim).unwrap_or(i64::MAX) - 1
70            } else if trc < 0 {
71                return runtime(
72                    "Exception: All elements of trc must be greater than or equal to 0.".into(),
73                );
74            } else {
75                trc
76            };
77            if incr < 1 {
78                return runtime("Exception: All elements of incr must be greater than 0.".into());
79            }
80            let (blc, trc, incr) = (blc as usize, trc as usize, incr as usize);
81            if blc > trc || trc >= dim {
82                return runtime(format!(
83                    "the slice {blc}..={trc} lies outside an axis of {dim} elements"
84                ));
85            }
86            start.push(blc);
87            end.push(trc + 1);
88            stride.push(incr);
89            result.push((trc - blc) / incr + 1);
90        }
91        Ok((Slicer::new(start, end, stride)?, result))
92    }
93}

95impl Open {
96    /// The slice of one array cell.
97    fn cell_slice(
98        &self,
99        column: &str,
100        row: usize,
101        corners: &Corners,
102        end_ok: bool,
103    ) -> Result<(CasaValue, Slicer), Fail> {
104        let shape = self
105            .table
106            .array_shape(row, column)?
107            .ok_or_else(|| Fail::Runtime(format!(
108                "Table DataManager error: Invalid operation: no array in row {row} of column {column}"
109            )))?;
110        let (slicer, _) = corners.over(&shape, end_ok)?;
111        let Value::Array(array) = self.table.get_cell_slice(column, row, &slicer)? else {
112            return runtime(format!("row {row} of column {column} is not an array"));
113        };
114        Ok((values::array(&array), slicer))
115    }
116
117    fn require_array_column(&self, column: &str, what: &str) -> Result<(), Fail> {
118        let schema = self
119            .table
120            .schema()
121            .and_then(|schema| schema.column(column))
122            .ok_or_else(|| Fail::Runtime(format!("Table column {column} is unknown")))?;
123        if matches!(schema.column_type(), ColumnType::Array(_)) {
124            Ok(())
125        } else {
126            runtime(format!(
127                "TableProxy::{what}: column {column} is not an array column"
128            ))
129        }
130    }
131
132    fn getcellslice(&self, column: &str, row: i64, corners: &Corners) -> Reply {
133        self.require_array_column(column, "getCellSlice")?;
134        let nrows = self.table.row_count();
135        let row = usize::try_from(row)
136            .ok()
137            .filter(|row| *row < nrows)
138            .ok_or_else(|| {
139                Fail::Runtime(format!(
140                    "TableColumn: row number {} exceeds #rows {nrows} in table {}",
141                    row as u64,
142                    self.name.display()
143                ))
144            })?;
145        Ok(self.cell_slice(column, row, corners, false)?.0)
146    }
147
148    fn getcolslice(&self, column: &str, corners: &Corners, selection: (i64, i64, i64)) -> Reply {
149        self.require_array_column(column, "getColumnSlice")?;
150        let selected = rows(
151            self.table.row_count(),
152            selection.0,
153            selection.1,
154            selection.2,
155            "TableProxy::getColumnSlice: no such row",
156        )?;
157        if selected.is_empty() {
158            let data_type = self
159                .table
160                .schema()
161                .and_then(|schema| schema.column(column))
162                .and_then(|schema| schema.data_type());
163            return Ok(values::empty_column(data_type));
164        }
165        let cells = selected
166            .iter()
167            .map(|row| Ok(self.cell_slice(column, *row, corners, true)?.0))
168            .collect::<Result<Vec<_>, Fail>>()?;
169        values::stack_rows(&cells)
170    }
171}

173/// The array a slice write holds, from the value coerced to the column.
174fn slice_data(
175    table: &Table,
176    column: &str,
177    wide: super::keyword_value::Wide,
178) -> Result<ArrayValue, Fail> {
179    let ty = table
180        .schema()
181        .and_then(|schema| schema.column(column))
182        .and_then(|schema| schema.data_type())
183        .ok_or_else(|| Fail::Runtime(format!("Table column {column} is unknown")))?;
184    match wide.into_cell(ty, true)? {
185        Value::Array(array) => Ok(array),
186        _ => runtime(format!("a slice of column {column} needs an array")),
187    }
188}
189
190fn wide_of(value: Incoming) -> Result<super::keyword_value::Wide, Fail> {
191    match value {
192        Incoming::Wide(wide) => Ok(wide),
193        Incoming::Record(_) => runtime("a record cannot be written into a slice".into()),
194    }
195}

197impl State {
198    /// `getcellslice(columnname, rownr, blc, trc, incr)`.
199    pub(super) fn getcellslice(
200        &self,
201        column: &str,
202        row: i64,
203        corners: &Corners,
204        log: &mut Vec<LogLine>,
205    ) -> Reply {
206        match self.table(log) {
207            Some(open) => open.getcellslice(column, row, corners),
208            None => Ok(CasaValue::Bools {
209                shape: vec![0],
210                values: Vec::new(),
211            }),
212        }
213    }
214
215    /// `getcolslice(columnname, blc, trc, incr, startrow, nrow, rowincr)`.
216    pub(super) fn getcolslice(
217        &self,
218        column: &str,
219        corners: &Corners,
220        selection: (i64, i64, i64),
221        log: &mut Vec<LogLine>,
222    ) -> Reply {
223        match self.table(log) {
224            Some(open) => open.getcolslice(column, corners, selection),
225            None => Ok(CasaValue::Bools {
226                shape: vec![0],
227                values: Vec::new(),
228            }),
229        }
230    }
231
232    /// `putcellslice(columnname, rownr, value, blc, trc, incr)`.
233    pub(super) fn putcellslice(
234        &mut self,
235        column: String,
236        row: i64,
237        value: Incoming,
238        corners: Corners,
239        log: &mut Vec<LogLine>,
240    ) -> Reply {
241        self.write_with(Target::Cells, log, move |table| {
242            let wide = wide_of(value)?;
243            let array_column = table
244                .schema()
245                .and_then(|schema| schema.column(&column))
246                .map(|schema| matches!(schema.column_type(), ColumnType::Array(_)));
247            match array_column {
248                None => return Err(put_column_error(&format!("column {column} does not exist"))),
249                Some(false) => {
250                    return runtime(format!("TableProxy::putCellSlice: column {column} is not an array column"));
251                }
252                Some(true) => {}
253            }
254            let row = usize::try_from(row)
255                .ok()
256                .filter(|row| *row < table.row_count())
257                .ok_or_else(|| put_column_error("no such row"))?;
258            let shape = table
259                .array_shape(row, &column)?
260                .ok_or_else(|| Fail::Runtime(format!(
261                    "Table DataManager error: Invalid operation: no array in row {row} of column {column}"
262                )))?;
263            let (slicer, expected) = corners.over(&shape, false)?;
264            let data = slice_data(table, &column, wide)?;
265            if data.shape() != expected.as_slice() {
266                return Err(conformance_slice(row, &column, data.shape(), &expected));
267            }
268            table.put_cell_slice(&column, row, &slicer, &data)?;
269            Ok(Persist::Rows {
270                column,
271                rows: vec![row],
272            })
273        })
274    }
275
276    /// `putcolslice(columnname, value, blc, trc, incr, startrow, nrow, rowincr)`.
277    pub(super) fn putcolslice(
278        &mut self,
279        column: String,
280        value: Incoming,
281        corners: Corners,
282        selection: (i64, i64, i64),
283        log: &mut Vec<LogLine>,
284    ) -> Reply {
285        self.write_with(Target::Cells, log, move |table| {
286            let wide = wide_of(value)?;
287            let array_column = table
288                .schema()
289                .and_then(|schema| schema.column(&column))
290                .map(|schema| matches!(schema.column_type(), ColumnType::Array(_)));
291            match array_column {
292                None => return Err(put_column_error(&format!("column {column} does not exist"))),
293                Some(false) => {
294                    return runtime(format!("TableProxy::putColumnSlice: column {column} is not an array column"));
295                }
296                Some(true) => {}
297            }
298            let selected = rows(
299                table.row_count(),
300                selection.0,
301                selection.1,
302                selection.2,
303                "TableProxy::TableProxy::putColumn: no such row",
304            )?;
305            let given = wide.shape.as_ref().and_then(|shape| shape.last()).copied().unwrap_or(1);
306            let per_row = split_rows(wide, true, &column, selected.len()).map_err(|_| {
307                Fail::Runtime(format!(
308                    "ArrayColumn::putColumn(slicer) - column {column} has {}, but array has {given} rows: \
309                     Table array conformance error",
310                    selected.len()
311                ))
312            })?;
313            for (row, cell) in selected.iter().zip(per_row) {
314                let shape = table
315                    .array_shape(*row, &column)?
316                    .ok_or_else(|| Fail::Runtime(format!(
317                        "Table DataManager error: Invalid operation: no array in row {row} of column {column}"
318                    )))?;
319                let (slicer, expected) = corners.over(&shape, false)?;
320                let data = slice_data(table, &column, cell)?;
321                if data.shape() != expected.as_slice() {
322                    return Err(conformance_slice(*row, &column, data.shape(), &expected));
323                }
324                table.put_cell_slice(&column, *row, &slicer, &data)?;
325            }
326            if selected.is_empty() {
327                return Ok(Persist::Nothing);
328            }
329            Ok(Persist::Rows {
330                column,
331                rows: selected,
332            })
333        })
334    }
335}

337/// The refusal of a slice value of the wrong shape.
338fn conformance_slice(row: usize, column: &str, shape: &[usize], expected: &[usize]) -> Fail {
339    let text = conformance(row, column, shape, expected);
340    match text {
341        Fail::Runtime(message) => {
342            Fail::Runtime(message.replace("ArrayColumn::put ", "ArrayColumn::putSlice "))
343        }
344        other => other,
345    }
