2//! Resolve ALMA absolute antenna positions into offsets from the MS positions.

6use casa_ms::{MeasurementSet, schema::SubtableId};
7use serde_json::Value;
8use thiserror::Error;
9
10use super::{GencalError, GencalRequest, get_string};

12/// Errors reading an ALMA antenna-position input file.
13#[derive(Debug, Error)]
14pub enum AntposInputError {
15    /// The source could not be read.
16    #[error("Cannot open {path}: {source}")]
17    Io {
18        /// Source path.
19        path: String,
20        /// Filesystem failure.
21        source: io::Error,
22    },
23    /// The source did not contain JSON.
24    #[error(transparent)]
25    Json(#[from] serde_json::Error),
26    /// A required JSON object key is absent.
27    #[error("{0}")]
28    Key(&'static str),
29    /// One antenna position lacks one of the first three coordinates.
30    #[error("Antenna {0} position needs three coordinates")]
31    Index(String),
32    /// A value does not support the operation used to read it.
33    #[error("{0}")]
34    Type(String),
35    /// The positions container has no object entries.
36    #[error("The data value must have named antenna positions")]
37    Attribute,
38    /// The positions are incompatible with the request.
39    #[error("{0}")]
40    Invalid(String),
41}
42
43pub(super) fn resolve(
44    ms: &MeasurementSet,
45    request: &GencalRequest,
46    polarization: &str,
47    path: &Path,
48) -> Result<GencalRequest, GencalError> {
49    let invalid = |reason| GencalError::from(AntposInputError::Invalid(reason));
50    if !request.antenna.is_empty() || !polarization.is_empty() || !request.parameter.is_empty() {
51        return Err(invalid(
52            "When using infile for ALMA the caltype is antpos, antenna, pol and parameter must be empty"
53                .into(),
54        ));
55    }
56    let observation = ms
57        .subtable(SubtableId::Observation)
58        .ok_or_else(|| invalid("Missing OBSERVATION subtable".into()))?;
59    let telescope = get_string(observation, 0, "TELESCOPE_NAME")?;
60    let file = fs::read(path).map_err(|source| AntposInputError::Io {
61        path: path.display().to_string(),
62        source,
63    })?;
64    let positions: Value = serde_json::from_slice(&file).map_err(AntposInputError::from)?;
65    if telescope != "ALMA" {
66        return Err(invalid(format!(
67            "Antenna positions are from ALMA but MS is from {telescope} telescope"
68        )));
69    }
70    if object_key(object_key(&positions, "metadata")?, "product_code")? != "antposalma" {
71        return Err(invalid(
72            "JSON file with antenna positions is not from ALMA. This is currently not supported"
73                .into(),
74        ));
75    }
76    let data = object_key(&positions, "data")?
77        .as_object()
78        .ok_or(AntposInputError::Attribute)?;
79    if data.is_empty() {
80        return Err(invalid(
81            "The list of antenna positions in JSON file is empty".into(),
82        ));
83    }
84    let antennas = ms.antenna().map_err(|error| invalid(error.to_string()))?;
85    let mut nominal_rows = HashMap::new();
86    for row in 0..antennas.row_count() {
87        let name = antennas
88            .name(row)
89            .map_err(|error| invalid(error.to_string()))?;
90        nominal_rows.entry(name).or_insert(row);
91    }
92    let mut resolved = request.clone();
93    let mut names = Vec::with_capacity(data.len());
94    for (name, corrected) in data {
95        let row = nominal_rows
96            .get(name)
97            .ok_or_else(|| invalid(format!("Antenna {name} could not be found in input MS")))?;
98        let nominal = antennas
99            .position(*row)
100            .map_err(|error| invalid(error.to_string()))?;
101        let coordinates = corrected.as_array().ok_or_else(|| {
102            AntposInputError::Type(format!("Antenna {name} position must contain numbers"))
103        })?;
104        for (index, nominal) in nominal.into_iter().enumerate() {
105            let coordinate = coordinates
106                .get(index)
107                .ok_or_else(|| AntposInputError::Index(name.clone()))?;
108            let value = coordinate
109                .as_f64()
110                .or_else(|| coordinate.as_bool().map(|value| f64::from(u8::from(value))))
111                .ok_or_else(|| {
112                    AntposInputError::Type(format!("Antenna {name} position must contain numbers"))
113                })?;
114            resolved.parameter.push(value - nominal);
115        }
116        names.push(name.as_str());
117    }
118    resolved.antenna = names.join(",");
119    Ok(resolved)
120}
121
122fn object_key<'a>(value: &'a Value, key: &'static str) -> Result<&'a Value, AntposInputError> {
123    value
124        .as_object()
125        .ok_or_else(|| {
126            AntposInputError::Type(format!("Cannot read {key} from a non-object value"))
127        })?
128        .get(key)
129        .ok_or(AntposInputError::Key(key))
130}
