2//! Complex normal equations for two-receptor leakage on a baseline.
3
4use num_complex::{Complex32, Complex64};
5
6use crate::solve::kernel::complex32_to_64;
7
8pub(super) struct LeakageFit {
9    diagonal: Vec<f64>,
10    // Only opposite receptors are coupled. Store the X-by-Y block of the
11    // Hermitian matrix and reconstruct its conjugate transpose on demand.
12    cross: Vec<Complex64>,
13    rhs: Vec<Complex64>,
14    antennas: Vec<bool>,
15}
16
17impl LeakageFit {
18    pub(super) fn new(antennas: usize) -> Self {
19        let parameters = 2 * antennas;
20        Self {
21            diagonal: vec![0.0; parameters],
22            cross: vec![Complex64::ZERO; antennas * antennas],
23            rhs: vec![Complex64::ZERO; parameters],
24            antennas: vec![false; antennas],
25        }
26    }
27
28    pub(super) fn add_baseline(
29        &mut self,
30        [first, second]: [usize; 2],
31        [_, xy, yx, _]: [Complex32; 4],
32        model: [Complex32; 4],
33        weights: [f32; 4],
34    ) {
35        let [mxx, mxy, myx, myy] = model;
36        let coefficients = [mxx, myy].map(complex32_to_64);
37        let cross_hands = [
38            complex32_to_64(xy - mxy),
39            complex32_to_64((yx - myx).conj()),
40        ];
41
42        // Compute D_x and conj(D_y) together. The two measured cross-hands
43        // couple the X receptor at one antenna to the Y receptor at the other.
44        let left = self.add_equation(
45            [2 * second + 1, 2 * first],
46            coefficients,
47            cross_hands[0],
48            weights[1],
49        );
50        let right = self.add_equation(
51            [2 * first + 1, 2 * second],
52            coefficients,
53            cross_hands[1],
54            weights[2],
55        );
56        if left || right {
57            self.antennas[first] = true;
58            self.antennas[second] = true;
59        }
60    }
61
62    fn add_equation(
63        &mut self,
64        indices: [usize; 2],
65        elements: [Complex64; 2],
66        observed: Complex64,
67        weight: f32,
68    ) -> bool {
69        if weight <= 0.0
70            || !weight.is_finite()
71            || !observed.is_finite()
72            || elements.iter().any(|coefficient| !coefficient.is_finite())
73        {
74            return false;
75        }
76        let weight = f64::from(weight);
77        for (row, left) in indices.into_iter().zip(elements) {
78            self.rhs[row] += left.conj() * weight * observed;
79            self.diagonal[row] += left.norm_sqr() * weight;
80        }
81        let [y, x] = indices.map(|index| index / 2);
82        self.cross[x * self.antennas.len() + y] += elements[1].conj() * elements[0] * weight;
83        true
84    }
85
86    pub(super) fn has_antenna(&self, antenna: i32) -> bool {
87        self.antennas.get(antenna as usize) == Some(&true)
88    }
89
90    pub(super) fn solve(
91        self,
92        reference: i32,
93        reference_value: Complex32,
94    ) -> Option<Vec<Option<[Complex32; 2]>>> {
95        let mut active = self
96            .antennas
97            .iter()
98            .enumerate()
99            .filter(|(_, valid)| **valid)
100            .flat_map(|(antenna, _)| [2 * antenna, 2 * antenna + 1])
101            .collect::<Vec<_>>();
102        if active.is_empty() {
103            return None;
104        }
105        let reference = usize::try_from(reference)
106            .ok()
107            .filter(|&antenna| self.antennas.get(antenna) == Some(&true));
108        let target = complex32_to_64(reference_value);
109        let (mut solution, rank) = self.solve_active(&active, None)?;
110        let mut full = vec![Complex64::ZERO; self.rhs.len()];
111        if let Some(antenna) = reference.filter(|_| rank < active.len()) {
112            // Fix the reference within the measured equations. Subtracting an
113            // offset afterward can change a baseline whose parallel model
114            // visibilities have different amplitudes.
115            let parameter = 2 * antenna;
116            active.retain(|&active_parameter| active_parameter != parameter);
117            solution = self.solve_active(&active, Some((parameter, target)))?.0;
118            full[parameter] = target;
119        }
120        for (index, value) in active.into_iter().zip(solution) {
121            full[index] = value;
122        }
123        let offset = reference.map_or(Complex64::ZERO, |antenna| full[2 * antenna] - target);
124        let narrow = |value: Complex64| Complex32::new(value.re as f32, value.im as f32);
125        Some(
126            full.as_chunks::<2>()
127                .0
128                .iter()
129                .zip(self.antennas)
130                .map(|(params, valid)| {
131                    valid.then(|| {
132                        [
133                            narrow(params[0] - offset),
134                            narrow((params[1] + offset).conj()),
135                        ]
136                    })
137                })
138                .collect(),
139        )
140    }
141
142    fn coefficient(&self, row: usize, column: usize) -> Complex64 {
143        let antennas = self.antennas.len();
144        match (row % 2, column % 2) {
145            (0, 1) => self.cross[(row / 2) * antennas + column / 2],
146            (1, 0) => self.cross[(column / 2) * antennas + row / 2].conj(),
147            _ if row == column => self.diagonal[row].into(),
148            _ => Complex64::ZERO,
149        }
150    }
151
152    fn solve_active(
153        &self,
154        active: &[usize],
155        constraint: Option<(usize, Complex64)>,
156    ) -> Option<(Vec<Complex64>, usize)> {
157        let coefficients = active
158            .iter()
159            .flat_map(|&row| {
160                active
161                    .iter()
162                    .map(move |&column| self.coefficient(row, column))
163            })
164            .collect::<Vec<_>>();
165        let observations = active
166            .iter()
167            .map(|&parameter| {
168                self.rhs[parameter]
169                    - constraint.map_or(Complex64::ZERO, |(fixed, value)| {
170                        self.coefficient(parameter, fixed) * value
171                    })
172            })
173            .collect::<Vec<_>>();
174        casa_numerics::solve_complex_normal_equations(coefficients, observations)
175    }
176}
