3//! CASA's automatic multi-threshold mask over every plane of an image
4//! domain: each channel and polarization masked on its own, as tclean masks
5//! a cube's channels and an image's Stokes planes
6//! (`SDMaskHandler::autoMaskByMultiThreshold`'s loop over channels, then
7//! polarizations).
8//!
9//! tclean masks a cube block by block of channels, each block an image
10//! store of its own (`SynthesisDeconvolver::executeCubeMinorCycle` with
11//! `CubeMinorCycleAlgorithm`, blocks from `numblockchans` on one process),
12//! and a continuum image whole. A block shares
13//!
14//! * the beam its masks are smoothed and pruned by: the common beam of the
15//!   block's PSF beams (`CasaImageBeamSet::getCommonBeam`), each CASA's fit
16//!   of its plane's PSF. A plane with no data holds the largest beam of
17//!   the set (`SIImageStore::makeImageBeamSet`), which leaves the common
18//!   beam as it is;
19//! * whether its robust noise ignores the mask: when the inherited mask
20//!   covers the block's whole first plane, every plane of the block takes
21//!   Chauvenet statistics (`calcRobustImageStatistics`).
22//!
23//! Every plane shares the image's PSF sidelobe level, the largest of its
24//! planes' (`SIImageStore::getPSFSidelobeLevel`, which the cube's
25//! controller hands each block), and the planes of a channel share its
26//! stop: `SDMaskHandler::skipChannels` reads the first polarization's
27//! threshold and counts the mask over every polarization of the channel,
28//! and a stopped channel skips all of them.
29//!
30//! A plane with no data has a zero residual in CASA: its statistics and
31//! thresholds are zero and its mask stays as it was.
32

47/// One plane of an image domain: the domain, the channel within the normal
48/// state (from its slab's first) and the polarization.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
50pub struct MaskPlaneKey {
51    /// The image domain's ordinal.
52    pub domain: usize,
53    /// The channel, counted from the normal state's first.
54    pub channel: usize,
55    /// The polarization plane.
56    pub polarization: usize,
57}

59/// One plane an automatic mask is made from, as the minor cycle reads it.
60#[derive(Debug, Clone, PartialEq)]
61pub struct AutoMaskPlane {
62    /// The residual in the units the minor cycle cleans, CASA's
63    /// `.residual` plane, x-major.
64    pub residual: Vec<f64>,
65    /// The plane's valid pixels, x-major.
66    pub valid: Vec<bool>,
67}

69/// A plane's residual and valid pixels; `None` for a plane with no data.
70pub type AutoMaskRead<'a, E> = dyn Fn(MaskPlaneKey) -> Result<Option<AutoMaskPlane>, E> + Sync + 'a;
71
72/// A plane's PSF measurements; `None` for a plane with no data.
73pub type AutoMaskPsf<'a> = dyn Fn(MaskPlaneKey) -> Option<PsfSummary> + Sync + 'a;

75/// What an automatic mask reads of one major cycle's planes, from the
76/// owner of the minor cycle: each plane as [`AutoMaskRead`] gives it, read
77/// once per boundary, and its PSF's measurements ([`AutoMaskPsf`]), which
78/// do not change; and how many channels it masks at once.
79pub struct AutoMaskSource<'a, E> {
80    /// Each plane's residual and valid pixels.
81    pub read: &'a AutoMaskRead<'a, E>,
82    /// Each plane's PSF measurements.
83    pub psf: &'a AutoMaskPsf<'a>,
84    /// The channels masked at once: more than one masks a block's channels
85    /// side by side on the current rayon pool (the owner's worker team),
86    /// one masks them one after another.
87    pub workers: usize,
88}

90/// One image domain's automatic mask at one boundary: a row of CASA's
91/// automask summary for each plane, channel-major.
92#[derive(Debug, Clone, PartialEq)]
93pub struct AutoMaskEvidence {
94    polarizations: usize,
95    planes: Box<[AutoMultithreshEvidence]>,
96}
97
98impl AutoMaskEvidence {
99    /// The polarizations of each channel.
100    #[must_use]
101    pub const fn polarizations(&self) -> usize {
102        self.polarizations
103    }
104
105    /// Every plane's row, channel after channel.
106    #[must_use]
107    pub fn planes(&self) -> &[AutoMultithreshEvidence] {
108        &self.planes
109    }
110
111    /// The row of `channel` and `polarization`.
112    #[must_use]
113    pub fn plane(&self, channel: usize, polarization: usize) -> Option<&AutoMultithreshEvidence> {
114        (polarization < self.polarizations)
115            .then(|| self.planes.get(channel * self.polarizations + polarization))
116            .flatten()
117    }
118
119    /// The first plane's row.
120    #[must_use]
121    pub fn primary(&self) -> &AutoMultithreshEvidence {
122        &self.planes[0]
123    }
124
125    /// Whether each channel stops evolving at later boundaries.
126    pub fn stopped_channels(&self) -> impl ExactSizeIterator<Item = bool> + '_ {
127        self.planes
128            .chunks(self.polarizations)
129            .map(|channel| channel[0].channel_stopped)
130    }
131}

144/// A channel's place in the domain's new mask: its supports, polarization
145/// after polarization, and its rows of the automask summary, which it
146/// fills.
147struct ChannelSlots<'a> {
148    channel: usize,
149    support: &'a mut [bool],
150    rows: &'a mut [AutoMultithreshEvidence],
151}
152
153/// A row before its plane is masked.
154const UNMASKED: AutoMultithreshEvidence = AutoMultithreshEvidence {
155    median: 0.0,
156    robust_rms: 0.0,
157    peak: 0.0,
158    positive_threshold: 0.0,
159    low_noise_threshold: 0.0,
160    negative_threshold: None,
161    threshold: AutoMaskThreshold::Sidelobe,
162    masked: false,
163    regions: 0,
164    pruned_regions: 0,
165    grow_regions: 0,
166    grow_pruned_regions: 0,
167    negative_pixels: 0,
168    changed_pixels: 0,
169    channel_stopped: false,
170};

172/// What every channel of one block is masked with.
173struct Block<'a, 'b, E> {
174    state: &'a AutoMaskState<'a>,
175    source: &'a AutoMaskSource<'b, E>,
176    domain: usize,
177    shape: [usize; 2],
178    polarizations: usize,
179    beam: Option<AutoMaskBeam>,
180    first_plane_full: bool,
181    /// An empty plane, the mask a plane inherits before the first boundary.
182    nothing: &'a [bool],
183}
184
185impl<E: From<MaskError>> Block<'_, '_, E> {
186    /// Mask `slots`' channel into its slots, and apply its stop
187    /// (`SDMaskHandler::skipChannels`), which every plane of the channel
188    /// shares: whether it added a positive (emission) region.
189    fn channel(&self, slots: ChannelSlots<'_>) -> Result<bool, E> {
190        let ChannelSlots {
191            channel,
192            support,
193            rows,
194        } = slots;
195        let state = self.state;
196        let inherited = |polarization| {
197            state
198                .previous
199                .and_then(|mask| mask.plane(channel, polarization))
200                .unwrap_or(self.nothing)
201        };
202        let cycle = AutoMaskCycle {
203            completed_major_cycles: state.completed_major_cycles,
204            threshold_reached: state.threshold_reached,
205            stopped: state
206                .stopped_channels
207                .get(channel)
208                .copied()
209                .unwrap_or(false),
210        };
211        let cells = self.shape[0] * self.shape[1];
212        let mut positive = false;
213        for (polarization, (out, row)) in support.chunks_mut(cells).zip(rows.iter_mut()).enumerate()
214        {
215            let plane = (self.source.read)(MaskPlaneKey {
216                domain: self.domain,
217                channel,
218                polarization,
219            })?;
220            let made = plane_mask(&PlaneInputs {
221                plane: plane.as_ref(),
222                inherited: inherited(polarization),
223                shape: self.shape,
224                first_plane_full: self.first_plane_full,
225                cycle,
226                controls: state.controls,
227                beam: self.beam,
228            })?;
229            out.copy_from_slice(&made.support);
230            *row = made.evidence;
231            positive |= made.positive;
232        }
233        let count = |support: &[bool]| support.iter().filter(|set| **set).count();
234        let before = (0..self.polarizations)
235            .map(|polarization| count(inherited(polarization)))
236            .sum();
237        let stopped = skip_channel(
238            before,
239            count(support),
240            rows[0].threshold,
241            cycle,
242            state.controls,
243        );
244        for row in rows.iter_mut() {
245            row.channel_stopped = stopped;
246        }
247        Ok(positive)
248    }
249}

251/// The automatic mask of every plane of `domain`, `shape` pixels by
252/// `planes` channels and polarizations, from the mask it inherits. Each
253/// channel writes its planes and rows straight into the domain's new mask
254/// and summary; a block's channels are masked side by side on the current
255/// rayon pool when `source` masks more than one at once.
256pub(super) fn auto_mask_domain<E>(
257    state: &AutoMaskState<'_>,
258    domain: usize,
259    shape: [usize; 2],
260    planes: MaskPlanes,
261    source: &AutoMaskSource<'_, E>,
262) -> Result<(ReconstructionMask, AutoMaskEvidence), E>
263where
264    E: From<MaskError> + Send,
265{
266    validate_auto_controls(state.controls)?;
267    let cells = shape[0] * shape[1];
268    if state.previous.is_some_and(|mask| {
269        mask.shape != shape || mask.coordinate != state.coordinate || mask.planes != planes
270    }) {
271        return Err(MaskError::ShapeMismatch.into());
272    }
273    let polarizations = planes.polarizations();
274    let psfs = (0..planes.channels())
275        .flat_map(|channel| {
276            (0..polarizations).map(move |polarization| MaskPlaneKey {
277                domain,
278                channel,
279                polarization,
280            })
281        })
282        .map(source.psf)
283        .collect::<Vec<_>>();
284    let sidelobe = psfs
285        .iter()
286        .flatten()
287        .map(PsfSummary::sidelobe)
288        .fold(0.0_f64, f64::max);
289    let nothing = vec![false; cells];
290    // One allocation each, written in place: the support is shared as it
291    // is by the mask (a trusted-length iterator collects into the `Arc`
292    // without a staging copy).
293    let mut support = (0..planes.count() * cells)
294        .map(|_| false)
295        .collect::<Arc<[bool]>>();
296    let mut rows = vec![UNMASKED; planes.count()];
297    let slots = Arc::get_mut(&mut support).expect("a support no one else holds yet");
298    let per_channel = polarizations * cells;
299    for block in channel_blocks(planes.channels(), shape) {
300        let masking = Block {
301            state,
302            source,
303            domain,
304            shape,
305            polarizations,
306            beam: block_beam(
307                &psfs[block.start * polarizations..block.end * polarizations],
308                sidelobe,
309                state.coordinate,
310            )?,
311            first_plane_full: state
312                .previous
313                .and_then(|mask| mask.plane(block.start, 0))
314                .is_some_and(|plane| plane.iter().all(|set| *set)),
315            nothing: &nothing,
316        };
317        let block_support = &mut slots[block.start * per_channel..block.end * per_channel];
318        let block_rows = &mut rows[block.start * polarizations..block.end * polarizations];
319        let positive = if source.workers > 1 {
320            block_support
321                .par_chunks_mut(per_channel)
322                .zip(block_rows.par_chunks_mut(polarizations))
323                .zip(block.clone().into_par_iter())
324                .map(|((support, rows), channel)| {
325                    masking.channel(ChannelSlots {
326                        channel,
327                        support,
328                        rows,
329                    })
330                })
331                .try_reduce(|| false, |left, right| Ok(left || right))?
332        } else {
333            let mut positive = false;
334            for ((support, rows), channel) in block_support
335                .chunks_mut(per_channel)
336                .zip(block_rows.chunks_mut(polarizations))
337                .zip(block.clone())
338            {
339                positive |= masking.channel(ChannelSlots {
340                    channel,
341                    support,
342                    rows,
343                })?;
344            }
345            positive
346        };
347        report_block(
348            state,
349            block.start,
350            &rows[block.start * polarizations..block.end * polarizations],
351            positive,
352            polarizations,
353        );
354    }
355    Ok((
356        ReconstructionMask::from_plane_support(state.coordinate, shape, planes, support)?,
357        AutoMaskEvidence {
358            polarizations,
359            planes: rows.into_boxed_slice(),
360        },
361    ))
362}

364/// The blocks of channels tclean masks a cube in on one process
365/// (`SynthesisDeconvolver::numblockchans`): blocks of `8e6` pixels divided
366/// by the plane's, at least ten channels, or the whole cube when it has no
367/// more channels than that; the last block holds what remains.
368pub(super) fn channel_blocks(channels: usize, shape: [usize; 2]) -> Vec<Range<usize>> {
369    let optimum = ((8.0e6 / shape[0] as f64 / shape[1] as f64) as usize).max(10);
370    let size = if channels > optimum {
371        optimum
372    } else {
373        channels
374    }
375    .max(1);
376    (0..channels)
377        .step_by(size)
378        .map(|start| start..(start + size).min(channels))
379        .collect()
380}

382/// The beam a block's masks are smoothed and pruned by, with the image's
383/// sidelobe level: the common beam of its planes' PSF beams; `None` when no
384/// plane of the block has data. CASA converts the beam to pixels by the
385/// first direction axis's increment; the common beam is found in radians,
386/// as for restoration.
387fn block_beam(
388    psfs: &[Option<PsfSummary>],
389    sidelobe: f64,
390    coordinate: DirectionCoordinateSpec,
391) -> Result<Option<AutoMaskBeam>, MaskError> {
392    let cell = coordinate.increment_rad()[0].abs();
393    let beams = psfs
394        .iter()
395        .flatten()
396        .map(|summary| {
397            let beam = summary.beam();
398            RestoringBeam::new(
399                beam.major_fwhm_rad() * cell,
400                beam.minor_fwhm_rad() * cell,
401                beam.position_angle_rad(),
402            )
403        })
404        .collect::<Result<Vec<_>, _>>()
405        .map_err(|_| MaskError::InvalidBeamArea)?;
406    if beams.is_empty() {
407        return Ok(None);
408    }
409    let common = RestoringBeam::common_enclosing(&beams).map_err(|_| MaskError::InvalidBeamArea)?;
410    let beam = AutoMaskBeam::new(
411        common.major_fwhm_rad() / cell,
412        common.minor_fwhm_rad() / cell,
413        common.position_angle_rad(),
414        sidelobe,
415    );
416    validate_beam(&beam)?;
417    Ok(Some(beam))
418}

420/// What tclean logs of one block's mask, its `rows` channel-major from
421/// channel `first`: at the first boundary its warnings
422/// (`autoMaskByMultiThreshold`, `noMaskCheck`) when it added no positive
423/// region, and with `verbose` CASA's automask summary, a row per channel
424/// (`printAutomaskSummary`).
425fn report_block(
426    state: &AutoMaskState<'_>,
427    first: usize,
428    rows: &[AutoMultithreshEvidence],
429    positive: bool,
430    polarizations: usize,
431) {
432    let channels = rows.len() / polarizations;
433    if state.completed_major_cycles == 0 {
434        if state.controls.cut_threshold >= 0.2 {
435            tracing::warn!(
436                "Faint regions may not be included in the final mask. Consider decreasing \
437                 cutthreshold."
438            );
439        }
440        if !positive {
441            warn_no_regions(rows);
442            let pruned = rows
443                .iter()
444                .filter(|row| row.regions > 0 && row.pruned_regions == row.regions)
445                .count();
446            if pruned > 0 {
447                tracing::warn!(
448                    "{pruned} of {channels} channels had all regions removed by pruning. Try \
449                     decreasing minbeamfrac to remove fewer regions"
450                );
451            }
452        }
453    }
454    if state.controls.verbose {
455        log_summary(first, rows, polarizations);
456    }
457}

459/// `SDMaskHandler::noMaskCheck` of a block whose positive mask is empty.
460fn warn_no_regions(rows: &[AutoMultithreshEvidence]) {
461    tracing::warn!("No regions found by automasking");
462    let [first, ..] = rows else {
463        return;
464    };
465    if rows.len() == 1 {
466        let kind = first.threshold.name();
467        tracing::warn!(
468            "Used {kind} threshold to mask, but no emission was found.Try decreasing your \
469             {kind}threshold parameter if you want to capture emission in these channels."
470        );
471        return;
472    }
473    for kind in [AutoMaskThreshold::Sidelobe, AutoMaskThreshold::Noise] {
474        let count = rows.iter().filter(|row| row.threshold == kind).count();
475        if count > 0 {
476            let name = kind.name();
477            tracing::warn!(
478                "{count} of {} channels used the {name} threshold to mask, but no emission was \
479                 found. Try decreasing your {name}threshold parameter if you want to capture \
480                 emission in these channels.",
481                rows.len()
482            );
483        }
484    }
485}

487/// CASA's `printAutomaskSummary` of one block's `rows` from channel
488/// `first_channel`, through `tracing`: a row per channel, each value a list
489/// over its polarizations when it has several.
490fn log_summary(first_channel: usize, rows: &[AutoMultithreshEvidence], polarizations: usize) {
491    let list = |values: Vec<String>| {
492        if polarizations == 1 {
493            values.into_iter().next().unwrap_or_default()
494        } else {
495            format!("[{}]", values.join(", "))
496        }
497    };
498    tracing::info!("========== automask summary ==========");
499    tracing::info!(
500        "chan masking? median   RMS         peak   thresh_type   thresh_value   N_reg N_pruned \
501         N_grow N_grow_pruned N_neg_pix"
502    );
503    for (offset, rows) in rows.chunks(polarizations).enumerate() {
504        let column =
505            |value: fn(&AutoMultithreshEvidence) -> String| list(rows.iter().map(value).collect());
506        // A stopped channel was skipped before this boundary: CASA prints
507        // its median, RMS and threshold as `--`.
508        let masked = rows[0].masked;
509        let skipped = |text: String| if masked { text } else { "--".to_string() };
510        tracing::info!(
511            "[C{}] {}        {}  {}  {}  {}  {}        {}  {}  {}  {}  {}",
512            first_channel + offset,
513            if masked { "T" } else { "F" },
514            skipped(column(|row| row.median.to_string())),
515            skipped(column(|row| row.robust_rms.to_string())),
516            column(|row| row.peak.to_string()),
517            column(|row| row.threshold.name().to_string()),
518            skipped(column(|row| row.positive_threshold.to_string())),
519            column(|row| row.regions.to_string()),
520            column(|row| row.pruned_regions.to_string()),
521            column(|row| row.grow_regions.to_string()),
522            column(|row| row.grow_pruned_regions.to_string()),
523            column(|row| row.negative_pixels.to_string()),
524        );
525    }
526    tracing::info!("========== END of automask summary ==========");
527}

529#[cfg(test)]
530mod tests {
531    use super::*;
532
533    #[test]
534    fn a_cube_is_masked_in_casa_channel_blocks() {
535        // 100 by 100: blocks of 800 channels, so a small cube is one block.
536        assert_eq!(channel_blocks(3, [100, 100]), vec![0..3]);
537        assert_eq!(channel_blocks(1, [100, 100]), vec![0..1]);
538        // 2000 by 2000: 2 channels by size, at least ten.
539        assert_eq!(channel_blocks(25, [2000, 2000]), [0..10, 10..20, 20..25]);
540        assert_eq!(channel_blocks(10, [2000, 2000]), vec![0..10]);
541        // 1000 by 1000: eight channels by size, so ten.
542        assert_eq!(channel_blocks(21, [1000, 1000]), [0..10, 10..20, 20..21]);
543        // 900 by 900: 9.87 truncates to nine, so ten.
544        assert_eq!(channel_blocks(11, [900, 900]), [0..10, 10..11]);
545        // 400 by 400: blocks of fifty.
546        assert_eq!(channel_blocks(120, [400, 400]), [0..50, 50..100, 100..120]);
547    }
