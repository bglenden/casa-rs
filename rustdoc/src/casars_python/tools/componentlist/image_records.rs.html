16pub(in crate::tools) struct ImageComponentRecord {
17    pub(in crate::tools) shape: ShapeType,
18    pub(in crate::tools) gaussian_axes: Option<[Quantity; 3]>,
19    pub(in crate::tools) value: CasaValue,
20}

22/// Read a CASA list for the image tool, retaining its record units and offsets.
23pub(in crate::tools) fn image_component_records<'a>(
24    record: &'a RecordValue,
25    logs: &mut Vec<LogLine>,
26) -> ConvertResult<impl ExactSizeIterator<Item = ConvertResult<ImageComponentRecord>> + use<'a>> {
27    let components = State::image_components(record, logs, |detail| {
28        ConvertError::runtime(format!(
29            "Exception: Input dictionary is not a valid component list: {detail}."
30        ))
31    })?;
32    Ok((0..components.length() as usize).map(move |index| {
33        let shape = components.checked(index as i64)?.shape().shape_type();
34        let gaussian_axes = if shape == ShapeType::Gaussian {
35            let Some(casa_core::Value::Record(raw_shape)) =
36                records::item(record, index)?.get("shape")
37            else {
38                unreachable!("validated Gaussian record");
39            };
40            let [major, minor, pa] = ["majoraxis", "minoraxis", "positionangle"]
41                .map(|key| records::angular(raw_shape, key).map_err(ConvertError::runtime));
42            Some([major?, minor?, pa?])
43        } else {
44            None
45        };
46        Ok(ImageComponentRecord {
47            shape,
48            gaussian_axes,
49            value: components.getcomponent(index as i64)?,
50        })
51    }))
52}

54/// Decode an image component model with its direction and frequency offsets.
55pub(in crate::tools) fn image_component_model(
56    record: &RecordValue,
57    logs: &mut Vec<LogLine>,
58) -> ConvertResult<ComponentList> {
59    State::image_components(record, logs, |_| {
60        ConvertError::runtime("Exception: model is an invalid componentlist record.")
61    })?
62    .into_image_model()
63}

65pub(in crate::tools) fn component_image_creation_model(
66    record: &RecordValue,
67    logs: &mut Vec<LogLine>,
68    persistent: bool,
69) -> ConvertResult<(ComponentList, BTreeMap<usize, RecordValue>)> {
70    let components = State::image_components(record, logs, ConvertError::runtime)?;
71    let spectral_records = if persistent {
72        components
73            .spectral_records()
74            .map_err(ConvertError::runtime)?
75    } else {
76        BTreeMap::new()
77    };
78    Ok((components.into_image_model()?, spectral_records))
79}
