24fn copy(
25    image: &dyn ImageInterface<f32>,
26    coordinates: RecordValue,
27    options: ImageCropOptions<'_>,
28    beams: &BeamView,
29    path: Option<&Path>,
30) -> PyResult<Storage<f32>> {
31    let mut crop =
32        ImageCrop::new(image, Some(coordinates), options, Some(beams.record())).map_err(raised)?;
33    let record = crop.section.coordinate_record.take();
34    let beam_set = beams.beam_set().map_err(raised)?;
35    let mut info = crop.section.image_info().to_record();
36    if beam_set.is_multi() {
37        let channels = crop.beam_planes(
38            image.find_axis(CoordinateType::Spectral),
39            beam_set.n_channels(),
40        );
41        let polarizations =
42            crop.beam_planes(image.find_axis(CoordinateType::Stokes), beam_set.n_stokes());
43        subset_beam_record(
44            &mut info,
45            Some(beams.record()),
46            &beam_set,
47            &channels,
48            &polarizations,
49        )
50        .map_err(raised)?;
51    } else {
52        beams.apply_to(&mut info);
53    }
54    let mut product = Storage::create(
55        crop.section.shape.clone(),
56        crop.section.coordinates.clone(),
57        path,
58        None,
59    )
60    .map_err(raised)?;
61    with_storage!(&mut product, image_out => crop.copy_to(image, image_out.as_mut()))
62        .map_err(raised)?;
63    with_storage!(&mut product, image_out => image_out.set_image_info_record(info))
64        .map_err(raised)?;
65    if let Some(record) = record {
66        match &mut product {
67            Storage::Paged(image) => image.set_coordinate_record(record).map_err(raised)?,
68            Storage::Temp {
69                record: coordinates,
70                ..
71            } => *coordinates = record,
72        }
73    }
74    Ok(product)
75}

77#[pymethods]
78impl PyImageTool {
79    #[allow(clippy::too_many_arguments)]
80    #[pyo3(signature = (outfile="", axes=None, overwrite=false, region=None, r#box="", chans="", stokes="", mask="", stretch=false, wantreturn=true), text_signature = "($self, outfile='', axes=[], overwrite=False, region={}, box='', chans='', stokes='', mask='', stretch=False, wantreturn=True)")]
81    fn crop(
82        &self,
83        py: Python<'_>,
84        outfile: &str,
85        axes: Option<IntList>,
86        overwrite: bool,
87        region: Option<&Bound<'_, PyAny>>,
88        r#box: &str,
89        chans: &str,
90        stokes: &str,
91        mask: &str,
92        stretch: bool,
93        wantreturn: bool,
94    ) -> PyResult<Py<Self>> {
95        let record_argument = region.is_none_or(|value| value.is_instance_of::<PyDict>());
96        let omitted_history = match region {
97            _ if record_argument => "{...}",
98            Some(value) if value.is_instance_of::<PyString>() => "\"\"",
99            _ => "[]",
100        };
101        let region = Region::capture(py, region);
102        let result = py.detach(|| {
103            let state = self.state();
104            let Some(attached) = state.image.as_ref() else {
105                return Ok(None);
106            };
107            let region = region?;
108            publish_image(outfile, overwrite, |path| {
109                let mut original = attached.lock();
110                if !matches!(&*original, ImageHandle::Float(_)) {
111                    return Err(raised("This method only supports Float valued images"));
112                }
113                if record_argument || region.is_some() {
114                    if !r#box.is_empty() {
115                        return Err(raised("box and regionPtr cannot be simultaneously specified"));
116                    }
117                    if !chans.is_empty() || !stokes.is_empty() {
118                        return Err(raised(
119                            "regionPtr and regionName, chans, and/or stokes cannot be simultaneously specified",
120                        ));
121                    }
122                }
123                if !outfile.is_empty() && let Some(source) = original.path() {
124                    ensure_outfile_does_not_alias_input(Path::new(outfile), source).map_err(raised)?;
125                }
126                attached.require_pixel_read_lock(&mut original)?;
127                let coordinates = original.coordinate_record();
128                let ImageHandle::Float(source) = &*original else {
129                    unreachable!("only float images accept crop")
130                };
131                let axes = axes.as_ref().map_or(&[][..], |axes| axes.0.as_slice());
132                let options = ImageCropOptions {
133                    axes,
134                    box_selection: (!r#box.is_empty()).then_some(r#box),
135                    region: region.as_ref().map(Region::as_region),
136                    channels: (!chans.is_empty()).then_some(chans),
137                    stokes: (!stokes.is_empty()).then_some(stokes),
138                    mask: (!mask.is_empty()).then_some(mask),
139                    stretch,
140                };
141                let beams = attached.beam_view().clone();
142                let mut cropped = match source {
143                    Storage::Temp { image, fits: Some(fits), .. } => {
144                        copy(&fits.image_view(image), coordinates, options, &beams, path)
145                    }
146                    _ => with_storage!(source, image => copy(image.as_ref(), coordinates, options, &beams, path)),
147                }?;
148                let history = match source {
149                    Storage::Paged(image) => image_history_rows(image.name().expect("native image path")).map_err(raised)?,
150                    Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
151                };
152                with_storage!(&mut cropped, image => image.append_history_entries(&history)).map_err(raised)?;
153                let mut cropped = ImageHandle::Float(cropped);
154                if state.history {
155                    let axes = axes.iter().map(i64::to_string).collect::<Vec<_>>().join(",");
156                    let region = region.as_ref().map_or_else(|| omitted_history.into(), Region::history);
157                    cropped.record_history("crop", format!("ia.crop(outfile={outfile:?}, axes=[{axes}], overwrite={overwrite}, region={region}, box={box:?}, chans={chans:?}, stokes={stokes:?}, mask={mask:?}, stretch={stretch}, wantreturn={wantreturn})", box = r#box))?;
158                }
159                let source = original.path().map_or_else(
160                    || "Temporary_Image".into(),
161                    |path| path.display().to_string(),
162                );
163                cropped.append_history_at(
164                    "::ImageCropper::_reportOldNewImageShapes",
165                    &[
166                        format!("Original {source} size => {:?}", original.shape()),
167                        format!("New {outfile} size => {:?}", cropped.shape()),
168                    ],
169                )?;
170                cropped.save().map_err(raised)?;
171                attached.release_requested_lock(&mut original)?;
172                Ok(cropped)
173            }).map(Some)
174        })?;
175        match result {
176            Some(attached) if wantreturn => Py::new(py, Self::attached(attached)),
177            other => {
178                py.detach(|| drop(other));
179                Py::new(py, Self::new())
180            }
181        }
182    }
183}
