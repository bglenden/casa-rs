28/// Adds synthetic random noise to an attached image (`ia.addnoise`).
29pub fn add_noise(
30    attached: &AttachedImage,
31    kind: &str,
32    parameters: &[f64],
33    seeds: &[i64],
34    region: Option<SubimageRegion<'_>>,
35    zero: bool,
36    history_command: Option<String>,
37) -> ImageToolResult<bool> {
38    let mut image = attached.lock();
39    if image.fits().is_some() {
40        return Err(ImageToolError::new(
41            "LatticeIterator cannot be constructed; lattice is not writable",
42        ));
43    }
44    let mut noise = ImageNoise::new(kind, parameters, seeds)?;
45    attached.require_lock(&mut image, LockType::Write)?;
46    match &mut *image {
47        ImageHandle::Float(storage) => with_storage!(storage, target => {
48            noise.apply(target.as_mut(), region, zero, |value, gen_draw| {
49                (f64::from(value) + gen_draw.draw()) as f32
50            })
51        }),
52        ImageHandle::Double(storage) => with_storage!(storage, target => {
53            noise.apply(target.as_mut(), region, zero, |value, gen_draw| {
54                value + gen_draw.draw()
55            })
56        }),
57        ImageHandle::Complex(storage) => with_storage!(storage, target => {
58            noise.apply(target.as_mut(), region, zero, |value, gen_draw| {
59                Complex32::new(
60                    (f64::from(value.re) + gen_draw.draw()) as f32,
61                    (f64::from(value.im) + gen_draw.draw()) as f32,
62                )
63            })
64        }),
65        ImageHandle::DComplex(storage) => with_storage!(storage, target => {
66            noise.apply(target.as_mut(), region, zero, |value, gen_draw| {
67                Complex64::new(value.re + gen_draw.draw(), value.im + gen_draw.draw())
68            })
69        }),
70    }?;
71    if let Some(cmd) = history_command {
72        image.record_history("addnoise", cmd)?;
73    }
74    image.save()?;
75    Ok(true)
76}

78/// Inserts pixels from `infile` into `attached` (`ia.insert`).
79pub fn insert_image(
80    attached: &AttachedImage,
81    infile: &str,
82    region: Option<SubimageRegion<'_>>,
83    locate: &[f64],
84    verbose: bool,
85    history_command: Option<String>,
86) -> ImageToolResult<(bool, Vec<ToolLogLine>)> {
87    attached.lock().check_single_precision("insert")?;
88    let source_path = Path::new(infile);
89    let lock_order = attached
90        .native_path()
91        .zip(source_path.canonicalize().ok())
92        .map(|(target, source)| target.cmp(&source));
93
94    let (mut target, copied) = if lock_order == Some(Ordering::Equal) {
95        let mut target = attached.lock();
96        attached.require_write_lock(&mut target, "insert")?;
97        let snapshot = snapshot_single_precision_handle(&target)?;
98        let copied = copy_input_handle(&mut target, &snapshot, infile, region, locate);
99        let release = attached.release_requested_lock(&mut target);
100        let copied = copied?;
101        release?;
102        (target, copied)
103    } else {
104        let source = AttachedImage::open_for_ordered_read(source_path)?;
105        let (mut target, mut source_image) = if lock_order == Some(Ordering::Less) {
106            let mut target = attached.lock();
107            let mut source_image = source.lock();
108            attached.require_write_lock(&mut target, "insert")?;
109            source.require_pixel_read_lock(&mut source_image)?;
110            (target, source_image)
111        } else {
112            let mut source_image = source.lock();
113            let mut target = attached.lock();
114            source.require_pixel_read_lock(&mut source_image)?;
115            attached.require_write_lock(&mut target, "insert")?;
116            (target, source_image)
117        };
118        let copied = copy_input_handle(&mut target, &source_image, infile, region, locate);
119        let release_source = source.release_requested_lock(&mut source_image);
120        let release_target = attached.release_requested_lock(&mut target);
121        let copied = copied?;
122        release_source?;
123        release_target?;
124        (target, copied)
125    };
126
127    let mut logs = Vec::new();
128    if verbose && copied {
129        logs.push(ToolLogLine::info(format!("Inserted pixels from {infile}")));
130    }
131    if let Some(cmd) = history_command {
132        target.record_history("insert", cmd)?;
133    }
134    target.save()?;
135    Ok((true, logs))
136}

138fn snapshot_single_precision_handle(source: &ImageHandle) -> ImageToolResult<ImageHandle> {
139    let shape = source.shape().to_vec();
140    let coordinates = source.coordinates().clone();
141    let record = source.coordinate_record();
142    match source {
143        ImageHandle::Float(storage) => {
144            let mut temp = TempImage::<f32>::new(shape, coordinates, TempStoragePolicy::Memory)?;
145            storage.with_float_view(|view| super::create::copy_typed_storage(view, &mut temp))?;
146            Ok(ImageHandle::Float(Storage::Temp {
147                image: Box::new(temp),
148                record,
149                fits: None,
150            }))
151        }
152        ImageHandle::Complex(storage) => {
153            let mut temp =
154                TempImage::<Complex32>::new(shape, coordinates, TempStoragePolicy::Memory)?;
155            with_storage!(storage, img => {
156                super::create::copy_typed_storage(img.as_ref(), &mut temp)
157            })?;
158            Ok(ImageHandle::Complex(Storage::Temp {
159                image: Box::new(temp),
160                record,
161                fits: None,
162            }))
163        }
164        ImageHandle::Double(_) | ImageHandle::DComplex(_) => Err(ImageToolError::new(
165            "Exception: Unsupported image data type.",
166        )),
167    }
168}

170fn copy_input_handle(
171    target: &mut ImageHandle,
172    source: &ImageHandle,
173    infile: &str,
174    region: Option<SubimageRegion<'_>>,
175    locate: &[f64],
176) -> ImageToolResult<bool> {
177    match (target, source) {
178        (ImageHandle::Float(target), ImageHandle::Float(source)) => {
179            Ok(source.with_float_view(|src| {
180                if matches!(target, Storage::Temp { fits: Some(_), .. }) {
181                    target.with_float_view(|dst| {
182                        if ImageInsertion::new(src, dst, region, locate)?.is_some() {
183                            return Err(ImageError::CasaSelection(
184                                "This image is not writable".into(),
185                            ));
186                        }
187                        Ok(false)
188                    })
189                } else {
190                    copy_into(target, src, region, locate)
191                }
192            })?)
193        }
194        (ImageHandle::Complex(target), ImageHandle::Complex(source)) => {
195            Ok(with_storage!(source, image => {
196                copy_into(target, image.as_ref(), region, locate)
197            })?)
198        }
199        (_, ImageHandle::Double(_) | ImageHandle::DComplex(_)) => Err(ImageToolError::new(
200            "Exception: Unsupported image data type.",
201        )),
202        _ => Err(ImageToolError::new(format!(
203            "Attached image pixel data type differs from that of {infile}"
204        ))),
205    }
206}

208fn copy_into<T: ImagePixel>(
209    target: &mut Storage<T>,
210    source: &dyn ImageInterface<T>,
211    region: Option<SubimageRegion<'_>>,
212    locate: &[f64],
213) -> Result<bool, ImageError> {
214    let plan = with_storage!(target, image => ImageInsertion::new(source, image.as_ref(), region, locate))?;
215    if let Some(plan) = plan {
216        with_storage!(target, image => plan.copy(source, image.as_mut()))?;
217        Ok(true)
218    } else {
219        Ok(false)
220    }
221}

223/// Writes pixel and/or mask arrays into a region of `attached` (`ia.putregion`).
224#[allow(clippy::too_many_arguments)]
225pub fn put_region(
226    attached: &AttachedImage,
227    pixels: Option<&PixelChunk>,
228    pixelmask: Option<&ArrayD<bool>>,
229    region: Option<SubimageRegion<'_>>,
230    list: bool,
231    usemask: bool,
232    replicate: bool,
233    history_command: Option<String>,
234) -> ImageToolResult<Vec<ToolLogLine>> {
235    let mut handle = attached.lock();
236    if handle.read_only_file_type().is_some() {
237        return Err(ImageToolError::new("This image is not writable"));
238    }
239    if let Some(pixels) = pixels {
240        let complex = matches!(*handle, ImageHandle::Complex(_) | ImageHandle::DComplex(_));
241        let input_complex = matches!(pixels, PixelChunk::Complex(_) | PixelChunk::DComplex(_));
242        if input_complex != complex {
243            return Err(ImageToolError::new(if complex {
244                "Real values cannot be put in images with complex valued pixels"
245            } else {
246                "Complex values cannot be put in images with real valued pixels"
247            }));
248        }
249    }
250    validate_put_region_shapes(pixels.map(PixelChunk::shape), pixelmask.map(ArrayD::shape))?;
251    attached.require_lock(&mut handle, LockType::Write)?;
252    let fresh_mask_name = pixelmask.map(|_| {
253        (0..)
254            .map(|index| format!("mask{index}"))
255            .find(|name| !has_region_or_mask(&handle, name))
256            .expect("an unused mask name")
257    });
258    let mask_name = fresh_mask_name.as_deref().unwrap_or("");
259    let mask = pixelmask.cloned();
260    match &mut *handle {
261        ImageHandle::Float(storage) => {
262            let arr = pixels_as_f32(pixels)?;
263            with_storage!(storage, image => put_image_region(image.as_mut(), arr, mask, region, mask_name, usemask, replicate))?;
264        }
265        ImageHandle::Double(storage) => {
266            let arr = pixels_as_f64(pixels)?;
267            with_storage!(storage, image => put_image_region(image.as_mut(), arr, mask, region, mask_name, usemask, replicate))?;
268        }
269        ImageHandle::Complex(storage) => {
270            let arr = pixels_as_c32(pixels)?;
271            with_storage!(storage, image => put_image_region(image.as_mut(), arr, mask, region, mask_name, usemask, replicate))?;
272        }
273        ImageHandle::DComplex(storage) => {
274            let arr = pixels_as_c64(pixels)?;
275            with_storage!(storage, image => put_image_region(image.as_mut(), arr, mask, region, mask_name, usemask, replicate))?;
276        }
277    }
278    if let Some(cmd) = history_command {
279        handle.record_history("putregion", cmd)?;
280    }
281    handle.save()?;
282    attached.release_requested_lock(&mut handle)?;
283    let logs = list
284        .then(|| ToolLogLine::info("Wrote the selected image region"))
285        .into_iter()
286        .collect();
287    Ok(logs)
288}

290fn pixels_as_f32(pixels: Option<&PixelChunk>) -> ImageToolResult<Option<ArrayD<f32>>> {
291    match pixels {
292        None => Ok(None),
293        Some(PixelChunk::Float(a)) => Ok(Some(a.clone())),
294        Some(PixelChunk::Double(a)) => Ok(Some(a.mapv(|v| v as f32))),
295        _ => Err(ImageToolError::new("Expected real pixel array.")),
296    }
297}
298
299fn pixels_as_f64(pixels: Option<&PixelChunk>) -> ImageToolResult<Option<ArrayD<f64>>> {
300    match pixels {
301        None => Ok(None),
302        Some(PixelChunk::Double(a)) => Ok(Some(a.clone())),
303        Some(PixelChunk::Float(a)) => Ok(Some(a.mapv(f64::from))),
304        _ => Err(ImageToolError::new("Expected real pixel array.")),
305    }
306}

308fn pixels_as_c32(pixels: Option<&PixelChunk>) -> ImageToolResult<Option<ArrayD<Complex32>>> {
309    match pixels {
310        None => Ok(None),
311        Some(PixelChunk::Complex(a)) => Ok(Some(a.clone())),
312        Some(PixelChunk::DComplex(a)) => {
313            Ok(Some(a.mapv(|v| Complex32::new(v.re as f32, v.im as f32))))
314        }
315        _ => Err(ImageToolError::new("Expected complex pixel array.")),
316    }
317}

319fn pixels_as_c64(pixels: Option<&PixelChunk>) -> ImageToolResult<Option<ArrayD<Complex64>>> {
320    match pixels {
321        None => Ok(None),
322        Some(PixelChunk::DComplex(a)) => Ok(Some(a.clone())),
323        Some(PixelChunk::Complex(a)) => {
324            Ok(Some(a.mapv(|v| Complex64::new(v.re.into(), v.im.into()))))
325        }
326        _ => Err(ImageToolError::new("Expected complex pixel array.")),
327    }
328}

330/// Replaces masked pixel values in `attached` (`ia.replacemaskedpixels`).
331#[allow(clippy::too_many_arguments)]
332pub fn replace_masked_pixels(
333    attached: &AttachedImage,
334    expression: &str,
335    region: Option<SubimageRegion<'_>>,
336    mask: &str,
337    update: bool,
338    list: bool,
339    stretch: bool,
340    history_command: Option<String>,
341) -> ImageToolResult<(Vec<ToolLogLine>, bool)> {
342    let mut image = attached.lock();
343    image.check_single_precision("replacemaskedpixels")?;
344    if expression.is_empty() {
345        return Err(ImageToolError::new("You must specify an expression"));
346    }
347    if image.fits().is_some() {
348        return Err(ImageToolError::new(
349            "This image is not writable.  It is probably a reference or expression virtual image",
350        ));
351    }
352    if let Some(SubimageRegion::Text(text)) = &region {
353        with_pixels!(&*image, storage => with_storage!(storage, pixels => {
354            validate_mask_replacement_region(pixels.as_ref(), SubimageRegion::Text(text))
355        }))?;
356    }
357    attached.require_lock(&mut image, LockType::Write)?;
358    let options = MaskReplacementOptions {
359        expression,
360        region,
361        mask,
362        update,
363        stretch,
364    };
365    let applied = match &mut *image {
366        ImageHandle::Float(storage) => with_storage!(storage, pixels => {
367            replace_masked_real_pixels(pixels.as_mut(), options)
368        }),
369        ImageHandle::Complex(storage) => with_storage!(storage, pixels => {
370            replace_masked_complex_pixels(pixels.as_mut(), options)
371        }),
372        _ => unreachable!("single precision checked"),
373    }?;
374    if matches!(applied, MaskReplacementOutcome::MissingNativeMask) {
375        image.save()?;
376        return Err(ImageToolError::new(
377            "SubLattice::pixelMask - no pixelmask available",
378        ));
379    }
380    let mut logs = Vec::new();
381    if list {
382        logs.push(ToolLogLine::info(format!(
383            "Replaced values of masked pixels by {expression}"
384        )));
385    }
386    if let Some(cmd) = history_command {
387        image.record_history("replacemaskedpixels", cmd)?;
388    }
389    image.save()?;
390    Ok((logs, true))
391}

393/// Fills pixels and/or pixel mask over a region with scalar values (`ia.set`).
394#[allow(clippy::too_many_arguments)]
395pub fn set_pixels_and_mask(
396    attached: &AttachedImage,
397    pixels_expr: Option<&str>,
398    pixelmask: i64,
399    region: Option<SubimageRegion<'_>>,
400    list: bool,
401    history_command: Option<String>,
402) -> ImageToolResult<(bool, Vec<ToolLogLine>)> {
403    let mut handle = attached.lock();
404    handle.check_single_precision("set")?;
405    let set_mask = pixelmask != -1;
406    let has_pixels = pixels_expr.is_some();
407    let ImageHandle::Float(storage) = &*handle else {
408        return Err(ImageToolError::new("set only supports float-valued images"));
409    };
410    let text_selection = if let Some(SubimageRegion::Text(text)) = region
411        && handle.fits().is_none()
412    {
413        Some(with_storage!(storage, image => {
414            ImagePixelSelection::new(image.as_ref(), Some(SubimageRegion::Text(text)))
415        })?)
416    } else {
417        None
418    };
419    if !has_pixels && !set_mask {
420        return Ok((
421            false,
422            vec![ToolLogLine::warning(
423                "You must specify at least either the pixels or the mask to set",
424            )],
425        ));
426    }
427    if let Some(kind) = handle.read_only_file_type() {
428        if kind == "MIRIADImage" {
429            return Err(ImageToolError::new(if has_pixels {
430                "Failed AlwaysAssert to.isWritable()"
431            } else {
432                "SubLattice::pixelMask - no pixelmask available"
433            }));
434        }
435        return Err(ImageToolError::new(if set_mask {
436            format!("Cannot make requested mask for this image type which is {kind}")
437        } else {
438            format!("{kind} is not writable")
439        }));
440    }
441    attached.require_lock(&mut handle, LockType::Write)?;
442    let ImageHandle::Float(storage) = &*handle else {
443        unreachable!("float attachment checked");
444    };
445    let default = with_storage!(storage, image => image.active_pixel_mask_name());
446    let mask_name = if set_mask {
447        if let Some(name) = default {
448            Some(name)
449        } else {
450            let name = (0..)
451                .map(|index| format!("mask{index}"))
452                .find(|name| !has_region_or_mask(&handle, name))
453                .expect("a fresh mask name");
454            let mask = ArrayD::from_elem(IxDyn(handle.shape()), true);
455            let ImageHandle::Float(storage) = &mut *handle else {
456                unreachable!();
457            };
458            with_storage!(storage, image => image.put_mask(&name, &mask))?;
459            with_storage!(storage, image => image.set_default_mask(&name))?;
460            handle.save()?;
461            Some(name)
462        }
463    } else {
464        None
465    };
466    let mask = mask_name.as_deref().map(|name| (name, pixelmask > 0));
467    let ImageHandle::Float(storage) = &mut *handle else {
468        unreachable!();
469    };
470    let (start, shape) = with_storage!(storage, image => {
471        let selection = match text_selection {
472            Some(selection) => selection,
473            None => ImagePixelSelection::new(image.as_ref(), region)?,
474        };
475        let value = pixels_expr.map(evaluate_pixel_expression).transpose()?;
476        selection.apply(image.as_mut(), value, mask)
477    })?;
478    let mut logs = Vec::new();
479    if list {
480        logs.push(ToolLogLine::info(format!(
481            "Selected region at {start:?} with shape {shape:?}"
482        )));
483    }
484    if let Some(cmd) = history_command {
485        handle.record_history("set", cmd)?;
486    }
487    handle.save()?;
488    Ok((true, logs))
489}
