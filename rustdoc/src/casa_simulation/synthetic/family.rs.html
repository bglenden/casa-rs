72pub(super) fn resolve_family_preset(request: &FamilyRequest) -> Result<FamilyPreset, String> {
73    let telescope = request.telescope.trim().to_ascii_lowercase();
74    let array_config = request.array_config.trim().to_ascii_lowercase();
75    let start_frequency_hz = resolve_band_frequency_hz(&telescope, &request.band)?;
76    let phase_center_rad = [4.712_391_234_768_306, -0.401_423_788_703_971_4];
77    let start_time_mjd_seconds = 4_895_229_577.784_943;
78
79    if telescope == "vla" {
80        let (antennas, array_config_source) = match array_config.as_str() {
81            "a" | "vla.a" | "vla.a.cfg" => {
82                load_real_family_config("VLA", "vla.a.cfg", &request.data_roots)?.unwrap_or_else(
83                    || {
84                        (
85                            tutorial_vla_a_antennas(),
86                            "embedded CASA vla.a.cfg coordinates".to_string(),
87                        )
88                    },
89                )
90            }
91            "b" | "vla.b" | "vla.b.cfg" => {
92                require_real_family_config("VLA", "vla.b.cfg", &request.data_roots)?
93            }
94            "c" | "vla.c" | "vla.c.cfg" => {
95                require_real_family_config("VLA", "vla.c.cfg", &request.data_roots)?
96            }
97            "d" | "vla.d" | "vla.d.cfg" => {
98                require_real_family_config("VLA", "vla.d.cfg", &request.data_roots)?
99            }
100            "synthetic-vla-b" | "synthetic-b" => (
101                scaled_vla_antennas(0.35),
102                "synthetic scaled VLA A".to_string(),
103            ),
104            "synthetic-vla-c" | "synthetic-c" => (
105                scaled_vla_antennas(0.11),
106                "synthetic scaled VLA A".to_string(),
107            ),
108            "synthetic-vla-d" | "synthetic-d" => (
109                scaled_vla_antennas(0.035),
110                "synthetic scaled VLA A".to_string(),
111            ),
112            _ => match load_real_family_config(
113                "VLA",
114                request.array_config.trim(),
115                &request.data_roots,
116            )? {
117                Some(loaded) => loaded,
118                None => {
119                    return Err(format!(
120                        "unsupported VLA array_config {:?}; expected A, B, C, D, a CASA .cfg path/name, or synthetic-vla-b/c/d",
121                        request.array_config
122                    ));
123                }
124            },
125        };
126        return Ok(FamilyPreset {
127            telescope_name: "VLA".to_string(),
128            field_name: "synthetic-vla".to_string(),
129            antennas,
130            array_config_source,
131            phase_center_rad,
132            start_time_mjd_seconds,
133            start_frequency_hz,
134            channel_width_hz: 1.0e6,
135            dish_diameter_m: 25.0,
136            polarization_basis: PolarizationBasis::Circular,
137        });
138    }
139
140    if telescope == "alma" || telescope == "aca" {
141        let (antennas, array_config_source, aca_only, simalma) = match array_config.as_str() {
142            "aca" | "aca.cycle10" | "aca.cycle10.cfg" | "aca.cycle10.named.cfg"
143                if telescope == "aca" =>
144            {
145                let loaded =
146                    load_real_family_config("ACA", "aca.cycle10.cfg", &request.data_roots)?.or(
147                        load_real_family_config(
148                            "ACA",
149                            "aca.cycle10.named.cfg",
150                            &request.data_roots,
151                        )?,
152                    );
153                match loaded {
154                    Some((antennas, source)) => (antennas, source, true, false),
155                    None => {
156                        return Err(real_config_not_found_message("ACA", "aca.cycle10.cfg"));
157                    }
158                }
159            }
160            "alma.cycle8.5.cfg" | "alma.cycle10.5" | "alma.cycle10.5.cfg" => {
161                let config_name = if array_config == "alma.cycle8.5.cfg" {
162                    "alma.cycle8.5.cfg"
163                } else {
164                    "alma.cycle10.5.cfg"
165                };
166                let (antennas, source) =
167                    require_real_family_config("ALMA", config_name, &request.data_roots)?;
168                (antennas, source, false, false)
169            }
170            "synthetic-alma-compact" | "synthetic-12m" | "synthetic-compact" => (
171                synthetic_alma_antennas("DA", 16, 12.0, 450.0),
172                "synthetic ALMA 12m compact".to_string(),
173                false,
174                false,
175            ),
176            "synthetic-aca" => (
177                synthetic_alma_antennas("CM", 10, 7.0, 120.0),
178                "synthetic ACA 7m compact".to_string(),
179                true,
180                false,
181            ),
182            "synthetic-simalma" => {
183                let mut antennas = synthetic_alma_antennas("DA", 12, 12.0, 450.0);
184                antennas.extend(synthetic_alma_antennas("CM", 10, 7.0, 120.0));
185                (
186                    antennas,
187                    "synthetic ALMA 12m plus ACA 7m".to_string(),
188                    false,
189                    true,
190                )
191            }
192            _ => match load_real_family_config(
193                &request.telescope,
194                request.array_config.trim(),
195                &request.data_roots,
196            )? {
197                Some((antennas, source)) => (
198                    antennas,
199                    source,
200                    telescope == "aca" || array_config.contains("aca"),
201                    false,
202                ),
203                None => {
204                    return Err(format!(
205                        "unsupported ALMA/ACA array_config {:?}; expected a CASA .cfg path/name such as alma.cycle10.5.cfg or aca.cycle10.cfg, or synthetic-alma-compact/synthetic-aca/synthetic-simalma",
206                        request.array_config
207                    ));
208                }
209            },
210        };
211        let dish_diameter_m = if aca_only { 7.0 } else { 12.0 };
212        return Ok(FamilyPreset {
213            telescope_name: if aca_only {
214                "ACA".to_string()
215            } else {
216                "ALMA".to_string()
217            },
218            field_name: if simalma {
219                "synthetic-simalma".to_string()
220            } else if aca_only {
221                "synthetic-aca".to_string()
222            } else {
223                "synthetic-alma".to_string()
224            },
225            antennas,
226            array_config_source,
227            phase_center_rad,
228            start_time_mjd_seconds,
229            start_frequency_hz,
230            channel_width_hz: 1.0e6,
231            dish_diameter_m,
232            polarization_basis: PolarizationBasis::Linear,
233        });
234    }
235
236    Err(format!(
237        "unsupported telescope {:?}; expected VLA, ALMA, or ACA",
238        request.telescope
239    ))
240}

257pub(super) fn load_real_family_config(
258    telescope: &str,
259    config_label: &str,
260    roots: &[PathBuf],
261) -> Result<Option<(Vec<Antenna>, String)>, String> {
262    let label = config_label.trim();
263    if label.is_empty() {
264        return Ok(None);
265    }
266    let looks_like_cfg = label.ends_with(".cfg") || Path::new(label).is_file();
267    if !looks_like_cfg {
268        return Ok(None);
269    }
270    for candidate in family_config_candidates(label, roots) {
271        if candidate.is_file() {
272            let antennas = parse_casa_array_config(telescope, &candidate)?;
273            return Ok(Some((antennas, candidate.display().to_string())));
274        }
275    }
276    Ok(None)
277}

279pub(super) fn family_config_candidates(label: &str, roots: &[PathBuf]) -> Vec<PathBuf> {
280    let path = Path::new(label);
281    if path.is_absolute() || path.components().count() > 1 {
282        return vec![path.to_path_buf()];
283    }
284
285    let mut candidates = vec![path.to_path_buf()];
286    for root in roots {
287        push_family_config_candidates(&mut candidates, root, label);
288    }
289    candidates
290}

292pub(super) fn push_family_config_candidates(
293    candidates: &mut Vec<PathBuf>,
294    root: &Path,
295    label: &str,
296) {
297    candidates.push(root.join(label));
298    candidates.push(root.join("alma").join("simmos").join(label));
299    candidates.push(
300        root.join("__data__")
301            .join("alma")
302            .join("simmos")
303            .join(label),
304    );
305}

307pub(super) fn parse_casa_array_config(
308    _telescope: &str,
309    path: &Path,
310) -> Result<Vec<Antenna>, String> {
311    let measures = casa_ms::open_measures_runtime().map_err(|e| e.to_string())?;
312    let config = crate::readantenna::read_antenna_config(path, measures.as_ref())
313        .map_err(|e| e.to_string())?;
314    Ok((0..config.x.len())
315        .map(|i| Antenna {
316            name: config.antenna_names[i].clone(),
317            station: config.pad_names[i].clone(),
318            position_m: [config.x[i], config.y[i], config.z[i]],
319            dish_diameter_m: config.dish_diameter[i],
320        })
321        .collect())
322}

324pub(super) fn resolve_band_frequency_hz(telescope: &str, band: &str) -> Result<f64, String> {
325    let band = band.trim().to_ascii_lowercase().replace(' ', "");
326    match (telescope, band.as_str()) {
327        ("vla", "l") => Ok(1.5e9),
328        ("vla", "s") => Ok(3.0e9),
329        ("vla", "c") => Ok(6.0e9),
330        ("vla", "x") => Ok(10.0e9),
331        ("vla", "ku") => Ok(15.0e9),
332        ("vla", "k") => Ok(22.0e9),
333        ("vla", "ka") => Ok(33.0e9),
334        ("vla", "q") => Ok(44.0e9),
335        ("alma" | "aca", "3" | "band3" | "band_3") => Ok(100.0e9),
336        ("alma" | "aca", "6" | "band6" | "band_6") => Ok(230.0e9),
337        ("alma" | "aca", "7" | "band7" | "band_7") => Ok(345.0e9),
338        ("alma" | "aca", "9" | "band9" | "band_9") => Ok(690.0e9),
339        ("vla", other) => Err(format!(
340            "unsupported VLA band {other:?}; expected L, S, C, X, Ku, K, Ka, or Q"
341        )),
342        (_, other) => Err(format!(
343            "unsupported ALMA/ACA band {other:?}; expected Band 3, 6, 7, or 9"
344        )),
345    }
346}

371pub(super) fn family_antennas_for_mode(
372    antennas: Vec<Antenna>,
373    observation_mode: ObservationMode,
374    total_power_antenna_index: Option<usize>,
375) -> Result<Vec<Antenna>, String> {
376    match observation_mode {
377        ObservationMode::Interferometric => Ok(antennas),
378        ObservationMode::TotalPower => {
379            let selected_index = total_power_antenna_index.unwrap_or(0);
380            antennas
381                .into_iter()
382                .nth(selected_index)
383                .map(|antenna| vec![antenna])
384                .ok_or_else(|| {
385                    format!(
386                        "total_power_antenna_index {selected_index} is out of range for array_config"
387                    )
388                })
389        }
390    }
391}

418pub(super) fn scaled_vla_antennas(scale: f64) -> Vec<Antenna> {
419    let antennas = tutorial_vla_a_antennas();
420    if (scale - 1.0).abs() < f64::EPSILON {
421        return antennas;
422    }
423    let center = antenna_centroid(&antennas);
424    antennas
425        .into_iter()
426        .map(|antenna| Antenna {
427            position_m: [
428                center[0] + (antenna.position_m[0] - center[0]) * scale,
429                center[1] + (antenna.position_m[1] - center[1]) * scale,
430                center[2] + (antenna.position_m[2] - center[2]) * scale,
431            ],
432            ..antenna
433        })
434        .collect()
435}

437pub(super) fn antenna_centroid(antennas: &[Antenna]) -> [f64; 3] {
438    let mut sum = [0.0; 3];
439    for antenna in antennas {
440        sum[0] += antenna.position_m[0];
441        sum[1] += antenna.position_m[1];
442        sum[2] += antenna.position_m[2];
443    }
444    let count = antennas.len().max(1) as f64;
445    [sum[0] / count, sum[1] / count, sum[2] / count]
446}

448pub(super) fn synthetic_alma_antennas(
449    name_prefix: &str,
450    count: usize,
451    dish_diameter_m: f64,
452    max_radius_m: f64,
453) -> Vec<Antenna> {
454    let center = [-2_223_990.194, -5_440_045.461, -2_481_682.086];
455    (0..count)
456        .map(|index| {
457            let radius = if count <= 1 {
458                0.0
459            } else {
460                max_radius_m * ((index + 1) as f64 / count as f64).sqrt()
461            };
462            let angle = index as f64 * 2.399_963_229_728_653;
463            let east_m = radius * angle.cos();
464            let north_m = radius * angle.sin();
465            let name = format!("{name_prefix}{:02}", index + 1);
466            Antenna {
467                name: name.clone(),
468                station: name,
469                position_m: offset_itrf_position(center, east_m, north_m),
470                dish_diameter_m,
471            }
472        })
473        .collect()
474}

476pub(super) fn offset_itrf_position(center: [f64; 3], east_m: f64, north_m: f64) -> [f64; 3] {
477    offset_itrf_position_enu(center, east_m, north_m, 0.0)
478}
479
480pub(super) fn offset_itrf_position_enu(
481    center: [f64; 3],
482    east_m: f64,
483    north_m: f64,
484    up_m: f64,
485) -> [f64; 3] {
486    let up = normalized(center);
487    let east = normalized([-center[1], center[0], 0.0]);
488    let north = [
489        up[1] * east[2] - up[2] * east[1],
490        up[2] * east[0] - up[0] * east[2],
491        up[0] * east[1] - up[1] * east[0],
492    ];
493    [
494        center[0] + east[0] * east_m + north[0] * north_m + up[0] * up_m,
495        center[1] + east[1] * east_m + north[1] * north_m + up[1] * up_m,
496        center[2] + east[2] * east_m + north[2] * north_m + up[2] * up_m,
497    ]
498}

500pub(super) fn normalized(vector: [f64; 3]) -> [f64; 3] {
501    let norm = (vector[0] * vector[0] + vector[1] * vector[1] + vector[2] * vector[2]).sqrt();
502    if norm == 0.0 {
503        return [0.0, 0.0, 0.0];
504    }
505    [vector[0] / norm, vector[1] / norm, vector[2] / norm]
506}

508pub(super) fn family_fields(
509    pointing_count: usize,
510    phase_center_rad: [f64; 2],
511    field_phase_centers_rad: Option<&[[f64; 2]]>,
512    imaging_mode: &str,
513    reference_frequency_hz: f64,
514    dish_diameter_m: f64,
515) -> Vec<Field> {
516    if let Some(field_phase_centers_rad) = field_phase_centers_rad {
517        if field_phase_centers_rad.len() == 1 {
518            return Vec::new();
519        }
520        return field_phase_centers_rad
521            .iter()
522            .enumerate()
523            .map(|(index, phase_center_rad)| Field {
524                name: format!("P{:03}", index),
525                phase_center_rad: *phase_center_rad,
526            })
527            .collect();
528    }
529    if pointing_count <= 1 || !mode_uses_mosaic(imaging_mode) {
530        return Vec::new();
531    }
532    let wavelength_m = 299_792_458.0 / reference_frequency_hz;
533    let primary_beam_fwhm_rad = 1.13 * wavelength_m / dish_diameter_m;
534    let spacing_rad = 0.45 * primary_beam_fwhm_rad;
535    mosaic_offsets(pointing_count, spacing_rad)
536        .into_iter()
537        .enumerate()
538        .map(|(index, [l_rad, m_rad])| Field {
539            name: format!("P{:03}", index),
540            phase_center_rad: [
541                phase_center_rad[0] + l_rad / phase_center_rad[1].cos().max(1.0e-6),
542                phase_center_rad[1] + m_rad,
543            ],
544        })
545        .collect()
546}

548pub(super) fn mode_uses_mosaic(imaging_mode: &str) -> bool {
549    let mode = imaging_mode.trim().to_ascii_lowercase();
550    matches!(
551        mode.as_str(),
552        "mosaic" | "mosaic_mfs" | "mosaic_cube" | "simalma" | "aca"
553    )
554}
555
556pub(super) fn mosaic_offsets(pointing_count: usize, spacing_rad: f64) -> Vec<[f64; 2]> {
557    let mut offsets = vec![[0.0, 0.0]];
558    let mut ring = 1usize;
559    while offsets.len() < pointing_count {
560        let samples = ring * 6;
561        for sample in 0..samples {
562            if offsets.len() == pointing_count {
563                break;
564            }
565            let angle = std::f64::consts::TAU * sample as f64 / samples as f64;
566            offsets.push([
567                ring as f64 * spacing_rad * angle.cos(),
568                ring as f64 * spacing_rad * angle.sin(),
569            ]);
570        }
571        ring += 1;
572    }
573    offsets
574}

582pub(super) fn directory_size_bytes(path: &Path) -> std::io::Result<u64> {
583    let metadata = fs::metadata(path)?;
584    if metadata.is_file() {
585        return Ok(metadata.len());
586    }
587    let mut total = 0;
588    for entry in fs::read_dir(path)? {
589        total += directory_size_bytes(&entry?.path())?;
590    }
591    Ok(total)
