11pub(in crate::storage) fn stage_int32_replacement(
12    original: &Path,
13    staged: &Path,
14    columns: &[ColumnDescContents],
15    replacements: &BTreeMap<String, Vec<Option<Value>>>,
16    rows: u64,
17) -> Result<bool, StorageError> {
18    let mut selected = Vec::with_capacity(columns.len());
19    for column in columns {
20        let values = replacements.get(&column.col_name).map(Vec::as_slice);
21        if values.is_some() && column.data_type != CasacoreDataType::TpInt {
22            return Ok(false);
23        }
24        selected.push(values);
25    }
26
27    let mut input = File::open(original)?;
28    let header = parse_ism_header(&mut input)?;
29    let index = parse_ism_index(&mut input, &header)?;
30    if index.used == 0 {
31        return Ok(false);
32    }
33    if index.rows.first() != Some(&0)
34        || index.rows.get(index.used) != Some(&rows)
35        || index.bucket_nrs.len() < index.used
36    {
37        return Err(invalid(
38            "IncrementalStMan interval index does not cover the table",
39        ));
40    }
41    let mut changed = BTreeMap::new();
42    let mut preserved = BTreeSet::new();
43    for interval in 0..index.used {
44        let start =
45            usize::try_from(index.rows[interval]).map_err(|_| invalid("ISM row overflow"))?;
46        let end =
47            usize::try_from(index.rows[interval + 1]).map_err(|_| invalid("ISM row overflow"))?;
48        let bucket_number = index.bucket_nrs[interval];
49        if start >= end || bucket_number >= header.nr_buckets {
50            return Err(invalid("invalid IncrementalStMan interval"));
51        }
52        let raw = read_ism_bucket(&mut input, &header, bucket_number)?;
53        let bucket = parse_ism_bucket(&raw, columns.len(), header.big_endian)?;
54        let origin = ISM_HEADER_SIZE + u64::from(bucket_number) * u64::from(header.bucket_size) + 4;
55        if !record_offsets(
56            &bucket,
57            &selected,
58            start..end,
59            origin,
60            header.big_endian,
61            &mut changed,
62            &mut preserved,
63        )? {
64            return Ok(false);
65        }
66    }
67    if changed.keys().any(|address| preserved.contains(address)) {
68        return Ok(false);
69    }
70
71    fs::copy(original, staged)?;
72    let mut output = File::options().write(true).open(staged)?;
73    for (address, value) in changed {
74        let words = if header.big_endian {
75            value.to_be_bytes()
76        } else {
77            value.to_le_bytes()
78        };
79        output.seek(SeekFrom::Start(address))?;
80        output.write_all(&words)?;
81    }
82    Ok(true)
83}

85fn record_offsets(
86    bucket: &IsmBucket,
87    selected: &[Option<&[Option<Value>]>],
88    rows: std::ops::Range<usize>,
89    origin: u64,
90    big_endian: bool,
91    changed: &mut BTreeMap<u64, i32>,
92    preserved: &mut BTreeSet<u64>,
93) -> Result<bool, StorageError> {
94    if bucket.col_indices.len() != selected.len() {
95        return Ok(false);
96    }
97    for (index, column) in bucket.col_indices.iter().zip(selected) {
98        let Some(column) = column else {
99            preserved.extend(
100                index
101                    .offsets
102                    .iter()
103                    .map(|&offset| origin + u64::from(offset)),
104            );
105            continue;
106        };
107        if index.row_nrs.first() != Some(&0) {
108            return Ok(false);
109        }
110        for (entry, (&start, &offset)) in index.row_nrs.iter().zip(&index.offsets).enumerate() {
111            let start = start as usize;
112            if start >= rows.len() {
113                // One physical bucket can cover a prefix of its logical rows.
114                break;
115            }
116            let end = index
117                .row_nrs
118                .get(entry + 1)
119                .map_or(rows.len(), |&row| rows.len().min(row as usize));
120            let Some(values) = column.get(rows.start + start..rows.start + end) else {
121                return Err(invalid("ISM replacement range is out of bounds"));
122            };
123            let Some((first @ Some(Value::Scalar(ScalarValue::Int32(value))), rest)) =
124                values.split_first()
125            else {
126                return Ok(false);
127            };
128            if rest.iter().any(|other| other != first) {
129                return Ok(false);
130            }
131            let bytes = bucket
132                .data
133                .get(offset as usize..offset as usize + 4)
134                .ok_or_else(|| invalid("ISM replacement value offset is out of bounds"))?;
135            let old = if big_endian {
136                read_i32_be(bytes)
137            } else {
138                read_i32_le(bytes)
139            };
140            let address = origin + u64::from(offset);
141            if *value == old {
142                preserved.insert(address);
143            } else if changed
144                .insert(address, *value)
145                .is_some_and(|prior| prior != *value)
146            {
147                return Ok(false);
148            }
149        }
150    }
151    Ok(true)
152}

154fn invalid(message: &str) -> StorageError {
155    StorageError::FormatMismatch(message.into())
156}

158#[cfg(all(test, unix))]
159mod tests {
160    use super::*;
161    use crate::{
162        ColumnSchema, DataManagerKind, EndianFormat, Table, TableOptions, TableSchema,
163        TableWriteLock,
164    };
165    use casa_types::{PrimitiveType, RecordField, RecordValue};
166
167    #[test]
168    fn encoded_replacement_preserves_other_cells_and_falls_back_when_a_run_splits() {
169        let scratch = tempfile::tempdir().unwrap();
170        for endian in [EndianFormat::BigEndian, EndianFormat::LittleEndian] {
171            let path = scratch.path().join(format!("{endian:?}"));
172            let schema = TableSchema::new(vec![
173                ColumnSchema::scalar("id", PrimitiveType::Int32),
174                ColumnSchema::scalar("keep", PrimitiveType::Int32),
175            ])
176            .unwrap();
177            let mut table = Table::with_schema(schema);
178            for row in 0..8 {
179                table
180                    .add_row(RecordValue::new(vec![
181                        RecordField::new("id", Value::Scalar(ScalarValue::Int32(row / 4))),
182                        RecordField::new("keep", Value::Scalar(ScalarValue::Int32(10 + row))),
183                    ]))
184                    .unwrap();
185            }
186            table
187                .save(
188                    TableOptions::new(&path)
189                        .with_data_manager(DataManagerKind::IncrementalStMan)
190                        .with_endian_format(endian),
191                )
192                .unwrap();
193            let mut lock = TableWriteLock::acquire_to_reread(&path, 1).unwrap();
194            let table = Table::open(TableOptions::new(&path)).unwrap();
195            let data = path.join("table.f0");
196            let initial = fs::read(&data).unwrap();
197            for (ids, encoded) in [
198                ([4, 4, 4, 4, 5, 5, 5, 5], true),
199                ([4, 5, 6, 7, 8, 9, 10, 11], false),
200            ] {
201                let requested = BTreeMap::from([(
202                    "id".to_owned(),
203                    ids.map(|id| Some(Value::Scalar(ScalarValue::Int32(id))))
204                        .into(),
205                )]);
206                let target = scratch.path().join(format!("raw-{endian:?}-{encoded}"));
207                let description =
208                    crate::storage::table_control::read_table_dat(&path.join("table.dat")).unwrap();
209                assert_eq!(
210                    stage_int32_replacement(
211                        &data,
212                        &target,
213                        &description.table_desc.columns,
214                        &requested,
215                        8
216                    )
217                    .unwrap(),
218                    encoded
219                );
220                if encoded {
221                    let staged_raw = fs::read(target).unwrap();
222                    assert_eq!(initial.len(), staged_raw.len());
223                    assert_eq!(
224                        initial
225                            .iter()
226                            .zip(&staged_raw)
227                            .filter(|(a, b)| a != b)
228                            .count(),
229                        2
230                    );
231                }
232                let mut edit = table
233                    .prepare_column_replacement(&requested, None, &mut lock)
234                    .unwrap();
235                edit.publish().unwrap();
236                edit.synchronize().unwrap();
237                let read = Table::open(TableOptions::new(&path))
238                    .unwrap()
239                    .scalar_columns_owned(&["id", "keep"])
240                    .unwrap();
241                assert_eq!(read["id"], ids.map(|id| Some(ScalarValue::Int32(id))));
242                assert_eq!(
243                    read["keep"],
244                    (10..18)
245                        .map(|id| Some(ScalarValue::Int32(id)))
246                        .collect::<Vec<_>>()
247                );
248                edit.restore().unwrap();
249                assert_eq!(fs::read(&data).unwrap(), initial);
250            }
251        }
252    }
