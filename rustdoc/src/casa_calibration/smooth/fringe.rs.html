2//! Track fringe phase using the solution rates before filtering the parameters.
3
4use casa_types::ArrayValue;
5use ndarray::ArrayD;
6
7use super::{Smoother, smooth_series};
8
9pub(super) fn apply(
10    parameters: &mut [ArrayValue],
11    flags: &mut [ArrayD<bool>],
12    rows: &[(usize, f64)],
13    windows: &[&[usize]],
14    smoother: Smoother,
15    reference_frequency: f64,
16    ratesmooth: bool,
17) -> Result<(), String> {
18    if flags[0].shape() != [8, 1] {
19        return Err("fringe smoothing requires two sets of four parameters at one channel".into());
20    }
21    let mut data = parameters
22        .iter_mut()
23        .map(|row| match row {
24            ArrayValue::Float32(row) => Ok(row),
25            _ => Err("fringe solutions must have a float32 FPARAM array".to_string()),
26        })
27        .collect::<Result<Vec<_>, _>>()?;
28    let tau = std::f64::consts::TAU;
29    for phase_index in [0, 4] {
30        let rate_index = phase_index + 2;
31        let phases = data
32            .iter()
33            .map(|row| row[[phase_index, 0]])
34            .collect::<Vec<_>>();
35        let rates = data
36            .iter()
37            .map(|row| row[[rate_index, 0]])
38            .collect::<Vec<_>>();
39        let valid_rates = flags
40            .iter()
41            .map(|row| !row[[rate_index, 0]])
42            .collect::<Vec<_>>();
43        let initial_rates = smooth_series(&rates, &valid_rates, windows, smoother);
44        for (row, rate) in data.iter_mut().zip(initial_rates) {
45            row[[rate_index, 0]] = rate;
46        }
47        let mut cycles = 0.0_f64;
48        let mut unwrapped = vec![phases[0]];
49        for index in 1..phases.len() {
50            // CASA's fringe predictor stores the epoch and estimates in float32.
51            let step = rows[index].1 as f32 - rows[index - 1].1 as f32;
52            let previous = f64::from(phases[index - 1]) / tau;
53            let current = f64::from(phases[index]) / tau;
54            let predict = |rate: f32| {
55                if ratesmooth {
56                    f64::from(rate) * reference_frequency * f64::from(step) * 2.0
57                } else {
58                    0.0
59                }
60            };
61            let forward = (previous + predict(rates[index - 1]) - current) as f32;
62            let backward = (current - predict(rates[index]) - previous) as f32;
63            let difference = (forward - backward) / 2.0;
64            if difference > 0.5 {
65                cycles += (f64::from(difference) + 0.5).floor();
66            } else if difference < 0.5 {
67                cycles -= (f64::from(difference.abs()) + 0.5).floor();
68            }
69            unwrapped.push((f64::from(phases[index]) + tau * cycles) as f32);
70        }
71        let valid_phase = flags
72            .iter()
73            .map(|row| !row[[phase_index, 0]])
74            .collect::<Vec<_>>();
75        let smoothed = smooth_series(&unwrapped, &valid_phase, windows, smoother);
76        for (i, (row, selected)) in data.iter_mut().zip(windows).enumerate() {
77            if selected.iter().any(|&index| valid_phase[index]) {
78                row[[phase_index, 0]] = wrap_phase(smoothed[i]);
79                flags[i][[phase_index, 0]] = false;
80            } else {
81                flags[i][[phase_index, 0]] = true;
82            }
83        }
84    }
85    for parameter in [1, 2, 3, 5, 6, 7] {
86        let cell = [parameter, 0];
87        let valid = flags.iter().map(|row| !row[cell]).collect::<Vec<_>>();
88        let values = data.iter().map(|row| row[cell]).collect::<Vec<_>>();
89        let smoothed = smooth_series(&values, &valid, windows, smoother);
90        for (((row, flag), window), value) in data
91            .iter_mut()
92            .zip(flags.iter_mut())
93            .zip(windows)
94            .zip(smoothed)
95        {
96            row[cell] = value;
97            flag[cell] = !window.iter().any(|&i| valid[i]);
98        }
99    }
100    Ok(())
101}
102
103fn wrap_phase(phase: f32) -> f32 {
104    let pi = std::f64::consts::PI;
105    let tau = std::f64::consts::TAU;
106    if !phase.is_finite() {
107        return phase;
108    }
109    let mut wrapped = phase.abs();
110    while f64::from(wrapped) > pi {
111        let current = f64::from(wrapped);
112        let next = (current - tau) as f32;
113        if next == wrapped {
114            // CASA's repeated subtraction makes no progress at this spacing.
115            return ((f64::from(phase) + pi).rem_euclid(tau) - pi) as f32;
116        }
117        // The rounded subtraction has a constant step within a float32
118        // exponent range. Leave a turn at the boundary for the finer spacing.
119        let boundary = f64::from(f32::from_bits(wrapped.to_bits() & 0x7f80_0000));
120        let step = current - f64::from(next);
121        let turns = ((current - boundary) / step).floor() - 1.0;
122        wrapped = if turns > 1.0 {
123            (current - turns * step) as f32
124        } else {
125            next
126        };
127    }
128    wrapped * phase.signum()
129}
