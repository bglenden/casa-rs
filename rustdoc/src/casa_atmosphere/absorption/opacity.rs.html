4//! Zenith opacities, phase delays and path lengths of an
5//! [`AbsorptionProfile`].
6//!
7//! Channels are addressed by their flat index in the grid
8//! ([`SpectralGrid::flat_index`](crate::SpectralGrid::flat_index)); an index
9//! past the grid panics, as slice indexing does. Wet quantities take the
10//! water column (metres) they are wanted for, and scale the profile's own
11//! by it, as ATM's `RefractiveIndexProfile` accessors with an
12//! `integratedwatercolumn` do.
13
14use super::{
15    AbsorptionChannel, AbsorptionProfile, ChannelAbsorption, LayerAbsorption, LayerTotals,
16};
17use num_complex::Complex64;
18
19/// Degrees per radian as ATM writes it.
20const DEGREES_PER_RADIAN: f64 = 57.29578;
21/// The speed of light, m/s.
22const SPEED_OF_LIGHT: f64 = 299_792_458.0;
23
24/// A species family whose opacity, phase delay or path length a full
25/// profile gives on its own (ATM's per-species accessors). Wet families are
26/// for the profile's own water column.
27#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
28pub enum Absorber {
29    /// All O2 lines.
30    O2Lines,
31    /// (16)O(16)O ground-state lines.
32    O2Ground,
33    /// (16)O(16)O first-vibrational-state lines.
34    O2Vibrational,
35    /// (16)O(18)O lines.
36    O2Isotope18,
37    /// (16)O(17)O lines.
38    O2Isotope17,
39    /// The dry continuum.
40    DryContinuum,
41    /// All O3 lines.
42    O3Lines,
43    /// (16)O3 ground-state lines.
44    O3Ground,
45    /// (16)O3 ν1 lines.
46    O3Nu1,
47    /// (16)O3 ν2 lines.
48    O3Nu2,
49    /// (16)O3 ν3 lines.
50    O3Nu3,
51    /// (16)O(16)O(18)O lines.
52    O3Isotope668,
53    /// (16)O(18)O(16)O lines.
54    O3Isotope686,
55    /// (16)O(16)O(17)O lines.
56    O3Isotope667,
57    /// (16)O(17)O(16)O lines.
58    O3Isotope676,
59    /// CO lines.
60    CoLines,
61    /// N2O lines.
62    N2oLines,
63    /// NO2 lines.
64    No2Lines,
65    /// SO2 lines.
66    So2Lines,
67    /// HCl lines.
68    HclLines,
69    /// HCN lines.
70    HcnLines,
71    /// All H2O lines.
72    H2oLines,
73    /// H2(16)O ground-state lines.
74    H2oGround,
75    /// H2(16)O bending-state lines.
76    H2oBending,
77    /// H2(17)O lines.
78    H2oIsotope17,
79    /// H2(18)O lines.
80    H2oIsotope18,
81    /// HDO lines.
82    Hdo,
83    /// The water continuum.
84    H2oContinuum,
85}

87impl Absorber {
88    /// This family's refractivity in a layer.
89    #[must_use]
90    pub fn of(self, layer: &LayerAbsorption) -> Complex64 {
91        match self {
92            Self::O2Lines => layer.o2,
93            Self::O2Ground => layer.o2_16o16o,
94            Self::O2Vibrational => layer.o2_vibrational,
95            Self::O2Isotope18 => layer.o2_16o18o,
96            Self::O2Isotope17 => layer.o2_16o17o,
97            Self::DryContinuum => layer.dry_continuum,
98            Self::O3Lines => layer.o3,
99            Self::O3Ground => layer.o3_ground,
100            Self::O3Nu1 => layer.o3_nu1,
101            Self::O3Nu2 => layer.o3_nu2,
102            Self::O3Nu3 => layer.o3_nu3,
103            Self::O3Isotope668 => layer.o3_668,
104            Self::O3Isotope686 => layer.o3_686,
105            Self::O3Isotope667 => layer.o3_667,
106            Self::O3Isotope676 => layer.o3_676,
107            Self::CoLines => layer.co,
108            Self::N2oLines => layer.n2o,
109            Self::No2Lines => layer.no2,
110            Self::So2Lines => layer.so2,
111            Self::HclLines => layer.hcl,
112            Self::HcnLines => layer.hcn,
113            Self::H2oLines => layer.h2o,
114            Self::H2oGround => layer.hh16o,
115            Self::H2oBending => layer.hh16o_bending,
116            Self::H2oIsotope17 => layer.hh17o,
117            Self::H2oIsotope18 => layer.hh18o,
118            Self::Hdo => layer.hdo,
119            Self::H2oContinuum => layer.h2o_continuum,
120        }
121    }
122}

124impl<C: AbsorptionChannel> AbsorptionProfile<C> {
125    /// Σ f(layer totals)·Δz over the layers of `channel`.
126    fn integrate(&self, channel: usize, f: impl Fn(&LayerTotals) -> f64) -> f64 {
127        let column = &self.channels[channel];
128        let mut sum = 0.0;
129        for (j, layer) in self.atmosphere.layers().iter().enumerate() {
130            sum += f(&column.layer_totals(j)) * layer.thickness_m;
131        }
132        sum
133    }
134
135    /// The ratio of `water_column_m` to the profile's own column.
136    fn water_scale(&self, water_column_m: f64) -> f64 {
137        water_column_m / self.ground_water_column_m
138    }
139
140    /// The wavelength of `channel`, metres.
141    fn wavelength_m(&self, channel: usize) -> f64 {
142        SPEED_OF_LIGHT / self.channels[channel].frequency_hz()
143    }
144
145    /// The zenith dry opacity of `channel`, nepers (ATM's `getDryOpacity`).
146    #[must_use]
147    pub fn dry_opacity(&self, channel: usize) -> f64 {
148        self.integrate(channel, LayerTotals::dry_absorption)
149    }
150
151    /// The zenith dry-continuum opacity, nepers (`getDryContOpacity`).
152    #[must_use]
153    pub fn dry_continuum_opacity(&self, channel: usize) -> f64 {
154        self.integrate(channel, |l| l.dry_continuum.im)
155    }
156
157    /// The zenith wet opacity for `water_column_m`, nepers
158    /// (`getWetOpacity(integratedwatercolumn, …)`).
159    #[must_use]
160    pub fn wet_opacity(&self, channel: usize, water_column_m: f64) -> f64 {
161        self.integrate(channel, LayerTotals::wet_absorption) * self.water_scale(water_column_m)
162    }
163
164    /// The zenith H2O-lines opacity for `water_column_m`, nepers.
165    #[must_use]
166    pub fn h2o_lines_opacity(&self, channel: usize, water_column_m: f64) -> f64 {
167        self.integrate(channel, |l| l.h2o_lines.im) * self.water_scale(water_column_m)
168    }
169
170    /// The zenith water-continuum opacity for `water_column_m`, nepers.
171    #[must_use]
172    pub fn h2o_continuum_opacity(&self, channel: usize, water_column_m: f64) -> f64 {
173        self.integrate(channel, |l| l.h2o_continuum.im) * self.water_scale(water_column_m)
174    }
175
176    /// The zenith dispersive (H2O lines) wet phase delay for
177    /// `water_column_m`, degrees (`getDispersiveH2OPhaseDelay`).
178    #[must_use]
179    pub fn dispersive_h2o_phase_delay_deg(&self, channel: usize, water_column_m: f64) -> f64 {
180        self.integrate(channel, |l| l.h2o_lines.re)
181            * self.water_scale(water_column_m)
182            * DEGREES_PER_RADIAN
183    }
184
185    /// The zenith non-dispersive (water continuum) wet phase delay for
186    /// `water_column_m`, degrees (`getNonDispersiveH2OPhaseDelay`).
187    #[must_use]
188    pub fn non_dispersive_h2o_phase_delay_deg(&self, channel: usize, water_column_m: f64) -> f64 {
189        self.integrate(channel, |l| l.h2o_continuum.re)
190            * self.water_scale(water_column_m)
191            * DEGREES_PER_RADIAN
192    }
193
194    /// The zenith dispersive dry phase delay (the dry lines), degrees
195    /// (`getDispersiveDryPhaseDelay`).
196    #[must_use]
197    pub fn dispersive_dry_phase_delay_deg(&self, channel: usize) -> f64 {
198        self.integrate(channel, |l| l.dry_lines.re) * DEGREES_PER_RADIAN
199    }
200
201    /// The zenith non-dispersive dry phase delay (the dry continuum),
202    /// degrees (`getNonDispersiveDryPhaseDelay`).
203    #[must_use]
204    pub fn non_dispersive_dry_phase_delay_deg(&self, channel: usize) -> f64 {
205        self.integrate(channel, |l| l.dry_continuum.re) * DEGREES_PER_RADIAN
206    }
207
208    /// The path length a phase delay of `phase_deg` in `channel` amounts
209    /// to, metres: (λ/360°)·φ, as ATM converts.
210    #[must_use]
211    pub fn path_length_m(&self, channel: usize, phase_deg: f64) -> f64 {
212        (self.wavelength_m(channel) / 360.0) * phase_deg
213    }
214
215    /// The zenith dispersive wet path length for `water_column_m`, metres.
216    #[must_use]
217    pub fn dispersive_h2o_path_length_m(&self, channel: usize, water_column_m: f64) -> f64 {
218        self.path_length_m(
219            channel,
220            self.dispersive_h2o_phase_delay_deg(channel, water_column_m),
221        )
222    }
223
224    /// The zenith non-dispersive wet path length for `water_column_m`,
225    /// metres.
226    #[must_use]
227    pub fn non_dispersive_h2o_path_length_m(&self, channel: usize, water_column_m: f64) -> f64 {
228        self.path_length_m(
229            channel,
230            self.non_dispersive_h2o_phase_delay_deg(channel, water_column_m),
231        )
232    }
233
234    /// The zenith dispersive dry path length, metres.
235    #[must_use]
236    pub fn dispersive_dry_path_length_m(&self, channel: usize) -> f64 {
237        self.path_length_m(channel, self.dispersive_dry_phase_delay_deg(channel))
238    }
239
240    /// The zenith non-dispersive dry path length, metres.
241    #[must_use]
242    pub fn non_dispersive_dry_path_length_m(&self, channel: usize) -> f64 {
243        self.path_length_m(channel, self.non_dispersive_dry_phase_delay_deg(channel))
244    }
245
246    /// The dry and wet absorption coefficients (1/m) of layer `layer` in
247    /// `channel`, the wet one for the profile's own water (ATM's
248    /// `getAbsTotalDry` and `getAbsTotalWet`).
249    #[must_use]
250    pub fn layer_absorption(&self, channel: usize, layer: usize) -> (f64, f64) {
251        let totals = self.channels[channel].layer_totals(layer);
252        (totals.dry_absorption(), totals.wet_absorption())
253    }
254
255    /// The mean of `value` over the channels of spectral window `window`,
256    /// as ATM's `getAverage…` accessors take it (the sum over the channels
257    /// divided by their number).
258    ///
259    /// # Panics
260    ///
261    /// If `window` is not a window of the grid.
262    pub fn window_average(&self, window: usize, value: impl Fn(usize) -> f64) -> f64 {
263        let first = self
264            .grid
265            .flat_index(window, 0)
266            .expect("a spectral window of the grid");
267        let count = self.grid.windows()[window].num_channels();
268        let sum = (first..first + count).fold(0.0, |sum, channel| sum + value(channel));
269        sum / count as f64
270    }
271}

273impl AbsorptionProfile<ChannelAbsorption> {
274    /// The refractivities of layer `layer` in `channel`, by species.
275    #[must_use]
276    pub fn layer(&self, channel: usize, layer: usize) -> &LayerAbsorption {
277        &self.channels[channel].layers()[layer]
278    }
279
280    fn integrate_absorber(
281        &self,
282        channel: usize,
283        absorber: Absorber,
284        part: impl Fn(Complex64) -> f64,
285    ) -> f64 {
286        let column = self.channels[channel].layers();
287        let mut sum = 0.0;
288        for (layer, atmosphere) in column.iter().zip(self.atmosphere.layers()) {
289            sum += part(absorber.of(layer)) * atmosphere.thickness_m;
290        }
291        sum
292    }
293
294    /// The zenith opacity of one species family, nepers (ATM's
295    /// `getO2LinesOpacity` and its siblings; a wet family for the profile's
296    /// own water column).
297    #[must_use]
298    pub fn absorber_opacity(&self, channel: usize, absorber: Absorber) -> f64 {
299        self.integrate_absorber(channel, absorber, |n| n.im)
300    }
301
302    /// The zenith phase delay of one species family, degrees (ATM's
303    /// `getO2LinesPhaseDelay` and its siblings).
304    #[must_use]
305    pub fn absorber_phase_delay_deg(&self, channel: usize, absorber: Absorber) -> f64 {
306        self.integrate_absorber(channel, absorber, |n| n.re) * DEGREES_PER_RADIAN
307    }
308
309    /// The zenith path length of one species family, metres (ATM's
310    /// `getO2LinesPathLength` and its siblings).
311    #[must_use]
312    pub fn absorber_path_length_m(&self, channel: usize, absorber: Absorber) -> f64 {
313        self.path_length_m(channel, self.absorber_phase_delay_deg(channel, absorber))
314    }
