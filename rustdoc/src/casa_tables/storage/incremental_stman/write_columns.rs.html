15/// Each column's element type and element count, and the bucket size, of
16/// an IncrementalStMan file: room
17/// for [`ISM_BUCKET_TARGET_ROWS`] rows of every value and its index entry,
18/// a string counted as 12 bytes. Indirect arrays occupy one Int64 offset.
19fn ism_source_layout(
20    col_descs: &[ColumnDescContents],
21) -> Result<(Vec<(CasacoreDataType, usize)>, u32), StorageError> {
22    if let Some(col_desc) = col_descs.iter().find(|col_desc| col_desc.is_record()) {
23        return Err(StorageError::FormatMismatch(format!(
24            "ISM column source writer stores scalars and fixed-shape arrays, not column {}",
25            col_desc.col_name
26        )));
27    }
28    let col_info: Vec<(CasacoreDataType, usize)> = col_descs
29        .iter()
30        .map(|c| {
31            if is_ism_indirect_array(c) {
32                return Ok((CasacoreDataType::TpInt64, 1));
33            }
34            let dt = CasacoreDataType::from_primitive_type(c.require_primitive_type()?, false);
35            let nrelem = c.shape.iter().map(|&dim| dim as usize).product::<usize>();
36            Ok((dt, if c.is_array { nrelem } else { 1 }))
37        })
38        .collect::<Result<_, StorageError>>()?;
39    let fixed_bytes_per_row: usize = col_info
40        .iter()
41        .map(|&(dt, nrelem)| {
42            if dt == CasacoreDataType::TpBool {
43                nrelem.div_ceil(8)
44            } else {
45                let elem = ism_element_size(dt);
46                if elem > 0 { elem * nrelem } else { nrelem * 12 }
47            }
48        })
49        .sum();
50    let index_overhead_per_row = col_descs.len() * 8;
51    let bytes_per_row = fixed_bytes_per_row + index_overhead_per_row;
52    let bucket_size = if bytes_per_row == 0 {
53        128u32
54    } else {
55        let target = (bytes_per_row * ISM_BUCKET_TARGET_ROWS + 4) as u32;
56        target.clamp(128, 327680)
57    };
58    Ok((col_info, bucket_size))
59}

61/// Write an IncrementalStMan file from column sources, a block of cells at
62/// a time: scalars and fixed-shape arrays, in buckets of
63/// [`ism_source_layout`]'s size. A bucket starts with each column's value
64/// at its first row and then holds each value that changes; a row whose
65/// values do not fit an empty bucket is refused.
66pub(crate) fn write_ism_file_column_sources(
67    file_path: &Path,
68    col_descs: &[ColumnDescContents],
69    scalar_columns: &ColumnSources<'_>,
70    big_endian: bool,
71) -> Result<Vec<u8>, StorageError> {
72    write_ism_column_sources_with_layout(file_path, col_descs, scalar_columns, big_endian, None, 1)
73}

75pub(super) fn write_ism_column_sources_with_layout(
76    file_path: &Path,
77    col_descs: &[ColumnDescContents],
78    scalar_columns: &ColumnSources<'_>,
79    big_endian: bool,
80    bucket: Option<u32>,
81    cache_size: u32,
82) -> Result<Vec<u8>, StorageError> {
83    if scalar_columns.column_count() != col_descs.len() {
84        return Err(StorageError::FormatMismatch(format!(
85            "ISM column source count mismatch: expected {}, got {}",
86            col_descs.len(),
87            scalar_columns.column_count()
88        )));
89    }
90    let (col_info, default_bucket) = ism_source_layout(col_descs)?;
91    let bucket_size = bucket.unwrap_or(default_bucket);
92    let nrrow = scalar_columns.row_count();
93    let ncol = col_descs.len();
94    if (bucket_size as usize) < 4 + 12 * ncol {
95        return Err(StorageError::FormatMismatch(format!(
96            "IncrementalStMan bucket size {bucket_size} cannot hold an index for {ncol} columns"
97        )));
98    }
99
100    if let Some(scalar_runs) = (0..ncol)
101        .map(|col_idx| scalar_columns.scalar_runs(col_idx))
102        .collect::<Option<Vec<_>>>()
103    {
104        return write_ism_file_scalar_run_columns(
105            file_path,
106            nrrow,
107            &col_info,
108            &scalar_runs,
109            bucket_size,
110            big_endian,
111            cache_size,
112        );
113    }
114
115    let mut arrays = col_descs
116        .iter()
117        .any(is_ism_indirect_array)
118        .then(|| {
119            let mut array_name = file_path.as_os_str().to_os_string();
120            array_name.push("i");
121            StManArrayFileWriter::create(Path::new(&array_name), big_endian, 1)
122        })
123        .transpose()?;
124
125    let mut bucket_start_rows: Vec<usize> = Vec::new();
126    let mut last_values: Vec<Vec<u8>> = vec![Vec::new(); ncol];
127    let mut file = File::create(file_path)?;
128    file.write_all(&vec![0u8; ISM_HEADER_SIZE as usize])?;
129    let mut nr_buckets = 0usize;
130
131    if nrrow > 0 {
132        let mut current = BucketBuilder {
133            data: Vec::new(),
134            col_indices: (0..ncol).map(|_| (Vec::new(), Vec::new())).collect(),
135        };
136        bucket_start_rows.push(0);
137
138        // Each column's cells are encoded a block of rows at a time; the
139        // bucket decisions then run row by row over the encoded bytes.
140        let mut cells = (0..ncol)
141            .map(|_| EncodedCells::default())
142            .collect::<Vec<_>>();
143        for block_start in (0..nrrow).step_by(SCALAR_BLOCK_CELLS) {
144            let block = block_start..nrrow.min(block_start + SCALAR_BLOCK_CELLS);
145            for (col_idx, cells) in cells.iter_mut().enumerate() {
146                let (dt, nrelem) = col_info[col_idx];
147                let desc = &col_descs[col_idx];
148                if is_ism_indirect_array(desc) {
149                    let writer = arrays.as_mut().expect("shared ISM array file");
150                    let element_type = CasacoreDataType::from_primitive_type(
151                        desc.require_primitive_type()?,
152                        false,
153                    );
154                    cells.fill_with(
155                        scalar_columns,
156                        col_idx,
157                        block.clone(),
158                        dt,
159                        big_endian,
160                        |value| {
161                            let offset = value
162                                .map(|v| writer.write_array(v, element_type))
163                                .transpose()?
164                                .unwrap_or(0);
165                            Ok(if big_endian {
166                                offset.to_be_bytes()
167                            } else {
168                                offset.to_le_bytes()
169                            }
170                            .to_vec())
171                        },
172                    )?;
173                } else {
174                    cells.fill(
175                        scalar_columns,
176                        col_idx,
177                        block.clone(),
178                        dt,
179                        big_endian,
180                        |value| encode_cell(value.into(), dt, nrelem, big_endian),
181                    )?;
182                }
183            }
184            for row_idx in block {
185                let cell = |col_idx: usize| cells[col_idx].cell(row_idx - block_start);
186                let rel_row = (row_idx - *bucket_start_rows.last().unwrap()) as u32;
187
188                let new_data_estimate: usize = (0..ncol)
189                    .map(cell)
190                    .zip(&last_values)
191                    .filter(|(encoded, last)| rel_row == 0 || encoded != last)
192                    .map(|(encoded, _)| encoded.len())
193                    .sum();
194
195                let current_index_size: usize = current
196                    .col_indices
197                    .iter()
198                    .map(|(row_nrs, _)| 4 + (row_nrs.len() + 1) * 4 + (row_nrs.len() + 1) * 4)
199                    .sum();
200                let total_estimate =
201                    4 + current.data.len() + new_data_estimate + current_index_size + ncol * 12;
202
203                let starts_bucket = rel_row > 0 && total_estimate > bucket_size as usize;
204                if starts_bucket {
205                    write_bucket(&mut file, &current, bucket_size, big_endian)?;
206                    nr_buckets += 1;
207                    current = BucketBuilder {
208                        data: Vec::new(),
209                        col_indices: (0..ncol).map(|_| (Vec::new(), Vec::new())).collect(),
210                    };
211                    bucket_start_rows.push(row_idx);
212                }
213                let rel_row = if starts_bucket { 0 } else { rel_row };
214                for (col_idx, last) in last_values.iter_mut().enumerate() {
215                    let encoded = cell(col_idx);
216                    if rel_row == 0 || encoded != last.as_slice() {
217                        let offset = current.data.len() as u32;
218                        current.data.extend_from_slice(encoded);
219                        current.col_indices[col_idx].0.push(rel_row);
220                        current.col_indices[col_idx].1.push(offset);
221                        last.clear();
222                        last.extend_from_slice(encoded);
223                    }
224                }
225            }
226        }
227
228        write_bucket(&mut file, &current, bucket_size, big_endian)?;
229        nr_buckets += 1;
230    }
231
232    let mut ism_rows: Vec<u64> = Vec::with_capacity(nr_buckets + 1);
233    let mut ism_bucket_nrs: Vec<u32> = Vec::with_capacity(nr_buckets);
234    for (i, &start) in bucket_start_rows.iter().enumerate() {
235        ism_rows.push(start as u64);
236        ism_bucket_nrs.push(i as u32);
237    }
238    ism_rows.push(nrrow as u64);
239
240    if let Some(arrays) = &mut arrays {
241        arrays.finish()?;
242    }
243    let index_data = serialize_ism_index(&ism_rows, &ism_bucket_nrs, big_endian)?;
244    let header_buf = serialize_ism_header(bucket_size, nr_buckets as u32, big_endian, cache_size)?;
245
246    file.write_all(&index_data)?;
247    file.seek(SeekFrom::Start(0))?;
248    file.write_all(&header_buf)?;
249
250    serialize_ism_dm_blob("ISM")
251}

253struct BucketBuilder {
254    data: Vec<u8>,
255    col_indices: Vec<(Vec<u32>, Vec<u32>)>,
256}
257
258fn write_bucket(
259    file: &mut File,
260    bucket: &BucketBuilder,
261    bucket_size: u32,
262    big_endian: bool,
263) -> Result<(), StorageError> {
264    let data_len = bucket.data.len();
265    let index_len: usize = bucket
266        .col_indices
267        .iter()
268        .map(|(row_nrs, _)| 4 + row_nrs.len() * 8)
269        .sum();
270    if 4 + data_len + index_len > bucket_size as usize {
271        return Err(StorageError::FormatMismatch(format!(
272            "ISM bucket requires {} bytes, exceeds planned bucket size {}",
273            4 + data_len + index_len,
274            bucket_size
275        )));
276    }
277    let mut raw = vec![0u8; bucket_size as usize];
278    raw[4..4 + data_len].copy_from_slice(&bucket.data);
279    let index_start = raw.len() - index_len;
280    let index_offset = index_start as u32;
281    if big_endian {
282        raw[0..4].copy_from_slice(&index_offset.to_be_bytes());
283    } else {
284        raw[0..4].copy_from_slice(&index_offset.to_le_bytes());
285    }
286    let mut cursor = index_start;
287    for (row_nrs, offsets) in &bucket.col_indices {
288        let count = row_nrs.len() as u32;
289        let encoded_count = if big_endian {
290            count.to_be_bytes()
291        } else {
292            count.to_le_bytes()
293        };
294        raw[cursor..cursor + 4].copy_from_slice(&encoded_count);
295        cursor += 4;
296        for &row in row_nrs {
297            let encoded = if big_endian {
298                row.to_be_bytes()
299            } else {
300                row.to_le_bytes()
301            };
302            raw[cursor..cursor + 4].copy_from_slice(&encoded);
303            cursor += 4;
304        }
305        for &offset in offsets {
306            let encoded = if big_endian {
307                offset.to_be_bytes()
308            } else {
309                offset.to_le_bytes()
310            };
311            raw[cursor..cursor + 4].copy_from_slice(&encoded);
312            cursor += 4;
313        }
314    }
315    file.write_all(&raw)?;
316    Ok(())
