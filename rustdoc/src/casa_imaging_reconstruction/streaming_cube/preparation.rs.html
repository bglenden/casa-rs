3//! Whole-row native preparation on the admitted imaging team. Projection and
4//! spectral evaluation feed the shared weighting primitives here; exact
5//! sums and source-coverage encoding stay with their worker until ordered join.
6//! No prepared-sample collection or weighted replay allocation is constructed.

19const NATIVE_COVERAGE_DOMAIN: &[u8] = b"casa-rs-native-weighting-row-coverage";
20const NATIVE_COVERAGE_VERSION: u32 = 1;
21
22fn invalid(message: &'static str) -> io::Error {
23    io::Error::new(io::ErrorKind::InvalidInput, message)
24}
25
26fn natural_sum(
27    problem: &CompiledProblem,
28    plan: &WeightingPlan,
29) -> io::Result<WeightingSumWeightPhase> {
30    if !matches!(problem.weighting().scheme(), WeightingScheme::Natural) {
31        return Err(invalid("native preparation requires natural weighting"));
32    }
33    begin_weighting_generation(problem, plan)
34        .and_then(|density| density.finish(problem))
35        .map_err(io::Error::other)
36}
37
38fn sum_required_bytes(plan: &WeightingPlan) -> io::Result<usize> {
39    exact_sum_capacity_bytes(plan.grid.output_planes)
40        .and_then(|bytes| {
41            bytes.checked_add(
42                plan.grid
43                    .output_planes
44                    .checked_mul(size_of::<ExactF64Sum>())?,
45            )
46        })
47        .and_then(|bytes| bytes.checked_add(plan.grid.planes.checked_mul(size_of::<f64>())?))
48        .and_then(|bytes| bytes.checked_add(size_of::<WeightingSumWeightPhase>()))
49        .ok_or_else(|| invalid("native sum-weight capacity overflow"))
50}

52/// Preparation completion owner. Whole rows are committed in canonical source
53/// order; numerical sums merge exact integer bins, independent of scheduling.
54pub struct NativeWeightingPreparation {
55    sum: WeightingSumWeightPhase,
56    coverage: Sha256,
57    work: coverage::CoverageProofWork,
58    previous_row: Option<u64>,
59    rows: u64,
60    failed: bool,
61}
62
63impl NativeWeightingPreparation {
64    /// Begin the single natural-weighting payload traversal without allocating
65    /// the historical weighted replay block.
66    pub fn new(problem: &CompiledProblem, plan: &WeightingPlan) -> io::Result<Self> {
67        let mut coverage = Sha256::new();
68        coverage.update(NATIVE_COVERAGE_DOMAIN);
69        coverage.update(NATIVE_COVERAGE_VERSION.to_be_bytes());
70        Ok(Self {
71            sum: natural_sum(problem, plan)?,
72            coverage,
73            work: coverage::CoverageProofWork {
74                bytes: (NATIVE_COVERAGE_DOMAIN.len() + size_of::<u32>()) as u64,
75                hash_calls: 2,
76            },
77            previous_row: None,
78            rows: 0,
79            failed: false,
80        })
81    }
82
83    /// Required coordinator state, including the conservative exact-bin bound.
84    pub fn coordinator_required_bytes(plan: &WeightingPlan) -> io::Result<usize> {
85        sum_required_bytes(plan)?
86            .checked_add(size_of::<Self>())
87            .ok_or_else(|| invalid("native coordinator capacity overflow"))
88    }

90    /// Worker allocation bound. Runtime additionally admits its projector and
91    /// spectral cache; those owners remain outside reconstruction.
92    pub fn worker_required_bytes(
93        problem: &CompiledProblem,
94        plan: &WeightingPlan,
95        maximum_rows: usize,
96        channels: usize,
97        correlations: usize,
98    ) -> io::Result<usize> {
99        if plan.problem != problem.problem_id()
100            || plan.commitment != problem.weighting().commitment_id()
101        {
102            return Err(invalid(
103                "native preparation plan belongs to another problem",
104            ));
105        }
106        NativeBlock::required_bytes(maximum_rows, channels, correlations)?
107            .checked_add(sum_required_bytes(plan)?)
108            .and_then(|bytes| bytes.checked_add(channels.checked_mul(size_of::<u32>())?))
109            .and_then(|bytes| bytes.checked_add(size_of::<NativeLayout>()))
110            .and_then(|bytes| bytes.checked_add(maximum_rows.checked_mul(size_of::<[u8; 32]>())?))
111            .and_then(|bytes| bytes.checked_add(size_of::<NativePreparationWorker>()))
112            .ok_or_else(|| invalid("native preparation worker capacity overflow"))
113    }
114
115    /// Make one bounded worker. The shared source layout is small; payload
116    /// ownership transfers into its flat buffer and never into sample objects.
117    pub fn worker(
118        &self,
119        problem: &CompiledProblem,
120        plan: &WeightingPlan,
121        layout: NativeLayout,
122        block: NativeBlock,
123        finite_values: FiniteValuePolicy,
124    ) -> io::Result<NativePreparationWorker> {
125        if self.failed || self.sum.problem != problem.problem_id() {
126            return Err(invalid("native weighting preparation failed or mismatched"));
127        }
128        let maximum_rows = block.metadata.len();
129        block.validate_shape(
130            maximum_rows,
131            layout.channels.len(),
132            layout.correlations.len(),
133        )?;
134        let sum = natural_sum(problem, plan)?;
135        Ok(NativePreparationWorker {
136            sum,
137            row_digests: Vec::with_capacity(maximum_rows),
138            block,
139            layout,
140            maximum_rows,
141            channel: 0,
142            previous_row: None,
143            finite_values,
144            output_frame: problem.geometry().spectral().output_frame(),
145            taper: problem.weighting().uv_taper(),
146            maximum_terms: maximum_spectral_terms(problem),
147            row_coverage: CoverageEncoder::new(),
148            work: coverage::CoverageProofWork {
149                bytes: 0,
150                hash_calls: 0,
151            },
152            batch_open: false,
153            batch_finished: false,
154            failed: false,
155        })
156    }
157
158    /// Join a completed worker in source order, merging exact bins and row
159    /// digests only. The native payload remains borrowed for the storage sink.
160    pub fn commit<'a>(
161        &mut self,
162        worker: &'a mut NativePreparationWorker,
163    ) -> io::Result<&'a NativeBlock> {
164        let result = self.commit_inner(worker);
165        if result.is_err() {
166            self.failed = true;
167            worker.failed = true;
168        }
169        result?;
170        Ok(worker.block())
171    }
172
173    fn commit_inner(&mut self, worker: &mut NativePreparationWorker) -> io::Result<()> {
174        if self.failed
175            || worker.failed
176            || !worker.batch_finished
177            || self.sum.problem != worker.sum.problem
178            || self.sum.commitment != worker.sum.commitment
179            || self.sum.sum_weights.len() != worker.sum.sum_weights.len()
180        {
181            return Err(invalid("unready or mismatched native preparation worker"));
182        }
183        let block = worker.block();
184        if block.metadata.len() != worker.row_digests.len()
185            || worker.sum.sum_sample_count != block.values.len() as u64
186        {
187            return Err(invalid("native preparation row or sample count mismatch"));
188        }
189        for (row, digest) in block.metadata.iter().zip(&worker.row_digests) {
190            if self
191                .previous_row
192                .is_some_and(|previous| row.physical_row <= previous)
193            {
194                return Err(invalid(
195                    "native preparation rows committed out of source order",
196                ));
197            }
198            self.coverage.update([1]);
199            self.coverage.update(row.physical_row.to_be_bytes());
200            self.coverage.update(digest);
201            self.work.bytes += 1 + 8 + 32;
202            self.work.hash_calls += 3;
203            self.previous_row = Some(row.physical_row);
204            self.rows = self
205                .rows
206                .checked_add(1)
207                .ok_or_else(|| invalid("native row count overflow"))?;
208        }
209        for (sum, partial) in self
210            .sum
211            .sum_weights
212            .iter_mut()
213            .zip(&mut worker.sum.sum_weights)
214        {
215            sum.merge(std::mem::take(partial))
216                .map_err(io::Error::other)?;
217        }
218        self.sum.sum_sample_count = self
219            .sum
220            .sum_sample_count
221            .checked_add(worker.sum.sum_sample_count)
222            .ok_or_else(|| invalid("native sample count overflow"))?;
223        self.work.bytes = self
224            .work
225            .bytes
226            .checked_add(worker.work.bytes)
227            .ok_or_else(|| invalid("native coverage byte count overflow"))?;
228        self.work.hash_calls = self
229            .work
230            .hash_calls
231            .checked_add(worker.work.hash_calls)
232            .ok_or_else(|| invalid("native coverage call count overflow"))?;
233        worker.sum.sum_sample_count = 0;
234        if let Some(rows) = &mut worker.sum.cube_rows {
235            *rows = CubeWeightRows::new();
236        }
237        worker.row_digests.clear();
238        worker.work = coverage::CoverageProofWork {
239            bytes: 0,
240            hash_calls: 0,
241        };
242        worker.batch_open = false;
243        worker.batch_finished = false;
244        Ok(())
245    }

247    /// Finish algorithmic state and native source coverage. Source traversal
248    /// completion and its run association must still be supplied by the runtime.
249    pub fn finish(mut self) -> io::Result<(WeightingAlgorithmState, WeightingReplaySummary)> {
250        if self.failed || self.rows == 0 {
251            return Err(invalid("failed or empty native weighting preparation"));
252        }
253        self.sum.density_sample_count = self.sum.sum_sample_count;
254        let sample_count = self.sum.sum_sample_count;
255        let state = self.sum.finish().map_err(io::Error::other)?;
256        state.next_replay.store(1, Ordering::Relaxed);
257        self.coverage.update([2]);
258        self.coverage.update(self.rows.to_be_bytes());
259        self.coverage.update(sample_count.to_be_bytes());
260        self.coverage.update(state.generation_id.as_bytes());
261        self.work.bytes += 1 + 8 + 8 + 32;
262        self.work.hash_calls += 4;
263        let coverage = WeightingReplayCoverageId(LogicalIdentity::from_sha256(
264            self.coverage.finalize().into(),
265        ));
266        // Rows, unlike buffer batches, are canonical across worker counts.
267        let block_count = self.rows;
268        let replay_id =
269            replay_identity(state.generation_id, coverage, sample_count, block_count, 0);
270        let mut residency = state.generation_residency;
271        residency.weighted_block_bytes = 0;
272        residency.weighted_sample_bytes = 0;
273        residency.simultaneous_selected_weighted_bytes = 0;
274        let summary = WeightingReplaySummary {
275            replay_id,
276            generation: state.generation_id,
277            coverage,
278            sample_count,
279            block_count,
280            replay_sequence: 0,
281            coverage_proof_bytes: self.work.bytes,
282            coverage_proof_hash_calls: self.work.hash_calls,
283            residency,
284        };
285        Ok((state, summary))
286    }

289/// Reusable flat native buffer plus exact sums for a disjoint whole-row range.
290pub struct NativePreparationWorker {
291    sum: WeightingSumWeightPhase,
292    block: NativeBlock,
293    layout: NativeLayout,
294    maximum_rows: usize,
295    channel: usize,
296    previous_row: Option<u64>,
297    finite_values: FiniteValuePolicy,
298    output_frame: FrequencyFrame,
299    taper: Option<UvTaper>,
300    maximum_terms: usize,
301    row_digests: Vec<[u8; 32]>,
302    row_coverage: CoverageEncoder,
303    work: coverage::CoverageProofWork,
304    batch_open: bool,
305    batch_finished: bool,
306    failed: bool,
307}

309impl NativePreparationWorker {
310    /// Begin filling after the preceding native block was synchronously written.
311    pub fn begin_batch(&mut self) -> io::Result<()> {
312        if self.failed || self.batch_open || self.batch_finished {
313            return Err(invalid("native preparation worker batch already open"));
314        }
315        self.block
316            .set_shape(self.maximum_rows, self.layout.channels.len())?;
317        self.batch_open = true;
318        Ok(())
319    }

321    /// Prepare one borrowed row/channel correlation slice directly into the
322    /// worker's arrays. Row geometry is installed once; group weighting and
323    /// spectral values are computed once per channel, with per-value validation.
324    pub fn consume_channel(
325        &mut self,
326        row: &SelectedObservationRunRow,
327        channel: SelectedObservationRunChannel,
328        correlations: &[SelectedObservationRunCorrelation],
329        geometry: SelectedRowSpectralGeometry,
330        output_frame_frequency_hz: f64,
331        contributions: SelectedSpectralContributions,
332    ) -> io::Result<()> {
333        let result: io::Result<()> = (|| {
334            if self.failed || !self.batch_open || self.batch_finished {
335                return Err(invalid(
336                    "native preparation worker is not accepting channels",
337                ));
338            }
339            let row_index = self.row_digests.len();
340            if row_index == self.maximum_rows
341                || correlations.len() != self.layout.correlations.len()
342                || channel.channel_index != self.layout.channels[self.channel]
343                || contributions.len() > self.maximum_terms
344                || !output_frame_frequency_hz.is_finite()
345                || output_frame_frequency_hz <= 0.0
346            {
347                return Err(invalid("native channel exceeds shape or stencil bound"));
348            }
349            let first = &correlations[0];
350            let view = SelectedObservationSampleView::from_run(row, &channel, first);
351            let address = view.address();
352            if !self.layout.contains_source(address) {
353                return Err(invalid("native source identity mismatch"));
354            }
355            if !geometry.matches_sample(view, self.output_frame)
356                || [Some(geometry.first()), geometry.second()]
357                    .into_iter()
358                    .flatten()
359                    .any(|(index, hz)| {
360                        index == channel.channel_index
361                            && hz.to_bits() != output_frame_frequency_hz.to_bits()
362                    })
363            {
364                return Err(io::Error::other(
365                    WeightingError::RowSpectralGeometryMismatch,
366                ));
367            }
368            let native_geometry = NativeRowSpectralGeometry {
369                channels: geometry.selected_channels(),
370                first: geometry.first(),
371                second: geometry.second(),
372                lattice_first_pair_hz: geometry.lattice_first_pair_hz(),
373            };
374            if self.channel == 0 {
375                if self
376                    .previous_row
377                    .is_some_and(|previous| row.physical_row <= previous)
378                    || native_geometry.channels != self.block.channels
379                    || native_geometry.first.0 != self.layout.channels[0]
380                    || native_geometry.second.map(|pair| pair.0) != Some(self.layout.channels[1])
381                    || row.domain_projections.len() != 1
382                {
383                    return Err(invalid("native row order or geometry mismatch"));
384                }
385                let projection = row
386                    .domain_projections
387                    .get(0)
388                    .ok_or_else(|| invalid("native primary projection missing"))?
389                    .model();
390                self.block.metadata[row_index] = RowMetadata {
391                    physical_row: row.physical_row,
392                    uvw_m: projection.transformed_uvw_m(),
393                    phase_shift_m: projection.phase_shift_m(),
394                    original_pair_hz: native_geometry
395                        .first_pair_hz()
396                        .ok_or_else(|| invalid("missing native frequency pair"))?,
397                };
398                self.previous_row = Some(row.physical_row);
399            } else if self.previous_row != Some(row.physical_row) {
400                return Err(invalid("native row ended before its selected channels"));
401            }
402            let group_flag = correlations.iter().any(|value| value.channel_flag);
403            let input = casa_unpolarized_input_weight(SelectedInputWeightGroup::correlation_run(
404                first.input_weight,
405                correlations.last().expect("nonempty layout").input_weight,
406                correlations.len(),
407            ));
408            let input_value = input_weight_value(
409                input,
410                group_flag
411                    || row.row_flag
412                    || correlations
413                        .iter()
414                        .all(|value| value.parallel_hand_group_flag),
415                self.finite_values,
416            )
417            .map_err(io::Error::other)?;
418            let natural_weight = |frequency_hz| -> io::Result<f64> {
419                if input_value == 0.0 {
420                    Ok(0.0)
421                } else {
422                    let uv = if self.taper.is_some() {
423                        uv_lambda_for_coordinates(row.coordinates.density_uvw_m, frequency_hz)
424                    } else {
425                        [0.0; 2]
426                    };
427                    apply_weight_taper(input_value, self.taper, uv).map_err(io::Error::other)
428                }
429            };
430            let base_weight = natural_weight(output_frame_frequency_hz)?;
431            let cell = row_index * self.block.channels + self.channel;
432            self.block.frequencies_hz[cell] = output_frame_frequency_hz;
433            let start = cell * correlations.len();
434            let end = start + correlations.len();
435            let spectral_values: SmallVec<[WeightingSpectralValue; 4]> = contributions
436                .iter()
437                .map(|contribution| {
438                    Ok(WeightingSpectralValue {
439                        contribution,
440                        imaging_weight: natural_weight(contribution.evaluation_frequency_hz())?,
441                    })
442                })
443                .collect::<io::Result<_>>()?;
444            let flagged_spectral_values: SmallVec<[WeightingSpectralValue; 4]> = if correlations
445                .iter()
446                .any(|sample| sample.parallel_hand_group_flag)
447            {
448                spectral_values
449                    .iter()
450                    .map(|value| WeightingSpectralValue {
451                        contribution: value.contribution,
452                        imaging_weight: 0.0,
453                    })
454                    .collect()
455            } else {
456                SmallVec::new()
457            };
458            for (ordinal, ((((sample, value), weight), flag), weight_flag)) in correlations
459                .iter()
460                .zip(&mut self.block.values[start..end])
461                .zip(&mut self.block.weights[start..end])
462                .zip(&mut self.block.flags[start..end])
463                .zip(&mut self.block.weight_flags[start..end])
464                .enumerate()
465            {
466                if (sample.correlation_index, sample.correlation_type)
467                    != self.layout.correlations[ordinal]
468                {
469                    return Err(invalid("native correlation order mismatch"));
470                }
471                *value = match sample.visibility {
472                    SelectedVisibilitySample::Float32(value) => {
473                        Complex64::new(f64::from(value), 0.0)
474                    }
475                    SelectedVisibilitySample::Complex32([re, im]) => {
476                        Complex64::new(f64::from(re), f64::from(im))
477                    }
478                };
479                *weight = if sample.parallel_hand_group_flag {
480                    0.0
481                } else {
482                    base_weight
483                };
484                *flag = !accept_polarization_value(
485                    sample.visibility,
486                    sample.input_weight,
487                    row.row_flag || sample.channel_flag,
488                    self.finite_values,
489                )
490                .map_err(io::Error::other)?;
491                *weight_flag = group_flag || sample.parallel_hand_group_flag || row.row_flag;
492                let spectral = if sample.parallel_hand_group_flag {
493                    flagged_spectral_values.as_slice()
494                } else {
495                    spectral_values.as_slice()
496                };
497                if let Some(value) = spectral.first() {
498                    self.sum.sum_weights[0]
499                        .add(value.imaging_weight)
500                        .map_err(io::Error::other)?;
501                }
502                let mut member_address = address;
503                member_address.correlation_index = sample.correlation_index;
504                member_address.correlation_type = sample.correlation_type;
505                self.row_coverage.push_parts(
506                    member_address,
507                    Some(native_geometry),
508                    output_frame_frequency_hz,
509                    spectral,
510                );
511            }
512            self.sum.sum_sample_count = self
513                .sum
514                .sum_sample_count
515                .checked_add(correlations.len() as u64)
516                .ok_or_else(|| invalid("native sample count overflow"))?;
517            self.channel += 1;
518            if self.channel == self.block.channels {
519                let encoder = std::mem::replace(&mut self.row_coverage, CoverageEncoder::new());
520                let (digest, work) =
521                    encoder.finish_row((self.block.channels * self.block.correlations) as u64);
522                self.row_digests.push(digest);
523                self.work.bytes += work.bytes;
524                self.work.hash_calls += work.hash_calls;
525                self.channel = 0;
526            }
527            Ok(())
528        })();
529        self.failed = result.is_err();
530        result
531    }

533    /// Finish a nonempty whole-row batch, including a shorter terminal batch.
534    pub fn finish_batch(&mut self) -> io::Result<&NativeBlock> {
535        if self.failed || !self.batch_open || self.batch_finished {
536            return Err(invalid("native preparation worker cannot finish its batch"));
537        }
538        let result: io::Result<()> = (|| {
539            if let Some(rows) = &mut self.sum.cube_rows {
540                rows.cursor.finish().map_err(io::Error::other)?;
541            }
542            if self.channel != 0 || self.row_digests.is_empty() {
543                return Err(invalid("empty or incomplete native batch"));
544            }
545            self.block
546                .set_shape(self.row_digests.len(), self.layout.channels.len())?;
547            Ok(())
548        })();
549        if result.is_err() {
550            self.failed = true;
551        }
552        result?;
553        self.batch_finished = true;
554        Ok(self.block())
555    }
