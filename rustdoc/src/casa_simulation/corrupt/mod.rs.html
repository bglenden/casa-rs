2//! Visibility corruption: thermal noise, CASA's `ANoise` term
3//! (`sm.setnoise(mode='simplenoise')` then `sm.corrupt()`).
4//!
5//! # What CASA adds
6//!
7//! Measured against CASA 6.7.6 (simulation Phase 1d probes, #841), each
8//! `corrupt` adds, for every noise term, to every DATA sample:
9//! - on a cross-correlation row, independent real and imaginary draws, each
10//!   N(0, σ²), where σ is the term's `simplenoise` in Jy;
11//! - on an autocorrelation row, a real draw only: the imaginary part is
12//!   unchanged;
13//! - in every correlation, the cross-hands of four-correlation data
14//!   included, each with its own draw;
15//! - whatever the flags: flagged rows and samples get noise too.
16//!
17//! WEIGHT and SIGMA are left as observed. A draw is made in double precision,
18//! scaled by σ held as a `Float`, and rounded to single precision before it
19//! is added, as `ANoiseCorruptor::simPar` computes `Complex(r·amp, i·amp)`.
20//!
21//! # Where the draws come from
22//!
23//! CASA draws from one `MLCG` stream per term, in the order its visibility
24//! iterator visits rows, channels and correlations, and the stream carries on
25//! across `corrupt` calls. casa-rs keys each draw instead
26//! ([`philox`], Philox4x32-10): the key is the term's seed and its
27//! application ordinal (how many `corrupt` calls the term had seen before),
28//! and the counter is the absolute MAIN row, the channel and the
29//! correlation pair. The two are statistically equivalent, not bit-identical
30//! (plan, "Random numbers"), and they share the structure scripts see:
31//! - two terms with the same seed draw the same values in the same
32//!   `corrupt`, so their noise adds coherently (variance 4σ² for two);
33//! - one term applied twice draws independent values (variance 2σ²);
34//! - a different seed draws different values.
35//!
36//! The result depends only on (seed, ordinal, row, channel, correlation), so
37//! it is bitwise the same for any batch size, worker count or backend.
38//!
39//! Normals come from Box–Muller on a pair of 32-bit words, in `f64`, with
40//! the logarithm, sine and cosine evaluated on the words ([`normal`]), so
41//! they are also bitwise the same on every platform. The largest draw is
42//! √(64 ln 2) ≈ 6.66σ, one in about 4·10⁹ beyond that in an exact normal.
43

54/// The top byte of the key's second word names what the stream is for, so
55/// later corruptions (gains, fBM screens) draw from streams disjoint from
56/// thermal noise under the same seed.
57const PURPOSE_THERMAL_NOISE: u32 = 0x01;
58
59/// Application ordinals share the key word with the purpose byte, so a term
60/// can be applied at most this many times.
61pub(crate) const MAXIMUM_APPLICATIONS: u32 = 1 << 24;
62
63/// One application of a noise term: the noise one `corrupt` adds for one
64/// `setnoise` term.
65#[derive(Debug, Clone, Copy, PartialEq)]
66pub(crate) struct NoiseApplication {
67    /// The seed in force when the term was set (`sm.setseed`).
68    pub(crate) seed: i32,
69    /// σ of each real and imaginary draw, Jy, held as CASA holds it.
70    pub(crate) sigma_jy: f32,
71    /// How many times the term had been applied before; below
72    /// [`MAXIMUM_APPLICATIONS`].
73    pub(crate) ordinal: u32,
74}
75
76impl NoiseApplication {
77    fn key(&self) -> Key {
78        debug_assert!(self.ordinal < MAXIMUM_APPLICATIONS);
79        [
80            self.seed as u32,
81            (PURPOSE_THERMAL_NOISE << 24) | self.ordinal,
82        ]
83    }
84
85    /// Add this application's noise to the cell of MAIN row `row`:
86    /// `correlations × channels` samples, correlation fastest. An
87    /// autocorrelation row gets real draws only.
88    pub(crate) fn add_to_row(
89        &self,
90        row: u64,
91        autocorrelation: bool,
92        correlations: usize,
93        cell: &mut [Complex32],
94    ) {
95        // Adding σ·N with σ = 0 changes nothing but the sign of a zero.
96        if self.sigma_jy == 0.0 || correlations == 0 {
97            return;
98        }
99        let sigma = f64::from(self.sigma_jy);
100        let key = self.key();
101        for (channel, samples) in cell.chunks_exact_mut(correlations).enumerate() {
102            for (pair, slots) in samples.chunks_mut(2).enumerate() {
103                let words = philox::philox4x32_10(counter(row, channel, pair), key);
104                for (slot, sample) in slots.iter_mut().enumerate() {
105                    let (re, im) = box_muller(words[2 * slot], words[2 * slot + 1]);
106                    sample.re += (re * sigma) as f32;
107                    if !autocorrelation {
108                        sample.im += (im * sigma) as f32;
109                    }
110                }
111            }
112        }
113    }
114}
115
116/// The counter of a draw: the MAIN row (two words), the channel, and the
117/// correlation pair (correlations 0 and 1 are pair 0, 2 and 3 pair 1).
118fn counter(row: u64, channel: usize, pair: usize) -> philox::Counter {
119    [row as u32, (row >> 32) as u32, channel as u32, pair as u32]
120}
121
122/// The four standard normals of one draw, slot 0's (real, imaginary) then
123/// slot 1's, as [`NoiseApplication::add_to_row`] uses them.
124#[cfg(test)]
125fn standard_normals(
126    application: &NoiseApplication,
127    row: u64,
128    channel: usize,
129    pair: usize,
130) -> [f64; 4] {
131    let words = philox::philox4x32_10(counter(row, channel, pair), application.key());
132    let (a, b) = box_muller(words[0], words[1]);
133    let (c, d) = box_muller(words[2], words[3]);
134    [a, b, c, d]
135}
