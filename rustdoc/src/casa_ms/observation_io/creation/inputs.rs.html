25/// Seconds per day, for times as MJD seconds.
26#[cfg(feature = "msv4")]
27const SECONDS_PER_DAY: f64 = 86_400.0;
28
29/// An input's opening, by store.
30enum Opening {
31    Msv2(Box<Msv2Observation>),
32    #[cfg(feature = "msv4")]
33    Msv4(Box<Msv4Observation>),
34}
35
36/// One opened input.
37pub(super) struct Input {
38    /// Its store.
39    pub(super) path: PathBuf,
40    opening: Opening,
41    /// The factor its weights are multiplied by.
42    pub(super) weight_scale: f32,
43    /// Its observing sessions in the order they first appear; a session
44    /// no partition holds comes last.
45    pub(super) appearance: Vec<ObservationId>,
46}
47
48impl Input {
49    /// Its opening.
50    pub(super) fn observation(&self) -> &dyn Observation {
51        match &self.opening {
52            Opening::Msv2(observation) => observation.as_ref(),
53            #[cfg(feature = "msv4")]
54            Opening::Msv4(observation) => observation.as_ref(),
55        }
56    }
57
58    /// Its opening when it is an MSv2's, whose rows state what an
59    /// observation does not (each row's `STATE_ID`).
60    pub(super) fn msv2(&self) -> Option<&Msv2Observation> {
61        match &self.opening {
62            Opening::Msv2(observation) => Some(observation),
63            #[cfg(feature = "msv4")]
64            Opening::Msv4(_) => None,
65        }
66    }
67
68    /// Open `input`, its weight scale checked, admitting to `ledger` each
69    /// increase its opening asks for before it is allocated (an MSv2's
70    /// tables, the subtables' rows its readers load and what they read from
71    /// them, its index and its partitions: [`Msv2Observation::open_granted`];
72    /// an MSv4's metadata and coordinates, as the MSv4 adapter's
73    /// `open_granted` asks), the charge then trimmed to what the opening
74    /// keeps.
75    pub(super) fn open(
76        input: &CreationInput,
77        ledger: &mut Ledger,
78    ) -> Result<Self, ObservationError> {
79        let weight_scale = weight_scale(input)?;
80        let path = input.path.clone();
81        if path.join("zarr.json").is_file() {
82            return msv4(path, weight_scale, ledger);
83        }
84        let mut granted = 0;
85        let observation =
86            Msv2Observation::open_granted(&path, &OpenOptions::default(), &mut |total| {
87                if total > granted {
88                    ledger.admit("an input's opening", total - granted)?;
89                    granted = total;
90                }
91                Ok(())
92            })?;
93        // The opening keeps its tables (their descriptions and the
94        // subtables' loaded rows), its index and its catalogue.
95        let kept = observation.tables_bytes()
96            + observation.index_bytes()
97            + catalogue_bytes(observation.catalogue());
98        ledger.trim("an input's opening", granted, kept)?;
99        let catalogue = observation.catalogue();
100        let mut first = vec![None; catalogue.observations.len()];
101        for id in catalogue.partition_ids() {
102            let session = catalogue.partition(id).observation.0 as usize;
103            if let (Some(row), Some(known)) = (observation.first_row(id), first.get_mut(session)) {
104                *known = Some(known.map_or(row, |earlier: u32| earlier.min(row)));
105            }
106        }
107        Ok(Self {
108            path,
109            appearance: in_order(&first),
110            opening: Opening::Msv2(Box::new(observation)),
111            weight_scale,
112        })
113    }
114}

116/// An MSv4 input, its sessions by earliest time, what its opening holds
117/// admitted to `ledger` as it asks ([`Msv4Observation::open_granted`]) and
118/// then trimmed to what it keeps.
119#[cfg(feature = "msv4")]
120fn msv4(path: PathBuf, weight_scale: f32, ledger: &mut Ledger) -> Result<Input, ObservationError> {
121    let mut granted = 0;
122    let observation =
123        Msv4Observation::open_granted(&path, &OpenOptions::default(), &mut |total| {
124            if total > granted {
125                ledger.admit("an input's opening", total - granted)?;
126                granted = total;
127            }
128            Ok(())
129        })?;
130    // The catalogue is shared with the opening, not copied.
131    let catalogue = Arc::clone(observation.catalogue());
132    let kept = observation.held_bytes() + catalogue_bytes(&catalogue);
133    ledger.trim("an input's opening", granted, kept)?;
134    let mut first: Vec<Option<f64>> = vec![None; catalogue.observations.len()];
135    for id in catalogue.partition_ids() {
136        let entry = catalogue.partition(id);
137        let Some(group) = entry.data_groups.first() else {
138            continue;
139        };
140        let bytes = grid_bytes(
141            Sizing::of(&observation, id, slice::from_ref(&group.name), WEIGHTS)?.times,
142            entry.baselines.len(),
143        );
144        ledger.admit("a partition's grid", bytes)?;
145        let grid = observation.grid(id, &group.name);
146        ledger.release(bytes);
147        let grid = grid?;
148        let Some(time) = grid.times_s.first() else {
149            continue;
150        };
151        let origin = grid.time_origin;
152        let time = (origin.day() + origin.frac()) * SECONDS_PER_DAY + time;
153        if let Some(known) = first.get_mut(entry.observation.0 as usize) {
154            *known = Some(known.map_or(time, |earlier: f64| earlier.min(time)));
155        }
156    }
157    Ok(Input {
158        path,
159        appearance: in_order(&first),
160        opening: Opening::Msv4(Box::new(observation)),
161        weight_scale,
162    })
163}

165/// Without the `msv4` feature an MSv4 cannot be opened.

172    Err(ObservationError::Unsupported {

174    })

177/// Sessions in the order of `first` (each one's first appearance), those
178/// without one last, in ID order.
179fn in_order<T: PartialOrd + Copy>(first: &[Option<T>]) -> Vec<ObservationId> {
180    let mut sessions: Vec<usize> = (0..first.len()).collect();
181    sessions.sort_by(|a, b| match (first[*a], first[*b]) {
182        (Some(x), Some(y)) => x.partial_cmp(&y).unwrap_or(std::cmp::Ordering::Equal),
183        (Some(_), None) => std::cmp::Ordering::Less,
184        (None, Some(_)) => std::cmp::Ordering::Greater,
185        (None, None) => a.cmp(b),
186    });
187    sessions
188        .into_iter()
189        .map(|session| ObservationId(session as u32))
190        .collect()
191}

193/// `input`'s weight scale, which must be finite and positive.
194fn weight_scale(input: &CreationInput) -> Result<f32, ObservationError> {
195    let scale = input.weight_scale as f32;
196    if scale.is_finite() && scale > 0.0 {
197        Ok(scale)
198    } else {
199        Err(request_error(format!(
200            "the weight scale of {} is {}; it must be finite and positive",
201            input.path.display(),
202            input.weight_scale
203        )))
204    }
205}
