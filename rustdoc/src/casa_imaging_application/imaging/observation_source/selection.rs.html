2//! What the batch route reads: the compiled selection resolved to the
3//! observation's partitions, and the per-sample predicate the request
4//! cannot state.

6use casa_imaging_model::{BaselineSelection, IdSelection, TimeSelection, UvSelection};
7use casa_imaging_runtime::pass::SourceError;
8use casa_observation::{
9    Correlation, FieldSelection, OrderedBaseline, PartitionId, VisibilityBatch,

13/// One partition the selection reads.
14#[derive(Clone, Debug, PartialEq)]
15pub(crate) struct SelectedPartition {
16    /// The partition.
17    pub(crate) id: PartitionId,
18    /// The spectral window number tasks select by (MSv2 `SPECTRAL_WINDOW`
19    /// row): what kernel sets and continuum rules are keyed on.
20    pub(crate) spectral_window: u32,
21    /// The selected native channels, ascending.
22    pub(crate) channels: Vec<u32>,
23    /// The window's per-time frequency selection, when a time reads only
24    /// some of `channels` ([`casa_imaging_model::FrequencyBands`]).
25    pub(crate) frequency_bands: Option<casa_imaging_model::FrequencyBands>,
26    /// The correlation products the operator routes, in its order.
27    pub(crate) correlations: Vec<Correlation>,
28    /// The window's reference wavelength, when a UV-distance interval is
29    /// in wavelengths.
30    pub(crate) wavelength_m: Option<f64>,
31}
32
33/// The compiled selection resolved to the observation's partitions: what
34/// [`super::ObservationSource`] binds and tests.
35///
36/// Windows, channels, correlations, the data group and fields map onto
37/// each partition's [`AccessRequest`](casa_observation::AccessRequest);
38/// intents onto the partitions chosen, since a partition's samples share
39/// theirs; scan per time, and baseline and UV distance per sample, as the
40/// source packs it.
41#[derive(Clone, Debug, PartialEq)]
42pub(crate) struct PartitionSelection {
43    /// The data group whose visibilities are imaged.
44    pub(crate) data_group: String,
45    /// The partitions, in traversal order.
46    pub(crate) partitions: Vec<SelectedPartition>,
47    /// The fields.
48    pub(crate) fields: FieldSelection,
49    /// The exact scan numbers; the batch catalogue resolves its ScanIds.
50    pub(crate) scans: IdSelection,
51    /// The baselines the antenna selection holds.
52    pub(crate) baselines: BaselineSelection,
53    /// The times the `timerange` selection holds.
54    pub(crate) times: TimeSelection,
55    /// The UV-distance intervals.
56    pub(crate) uv: UvSelection,
57}
58
59impl PartitionSelection {
60    /// Bytes owned by the run's per-time and per-sample predicates (scan
61    /// and baseline), including their fixed representations.
62    pub(crate) fn predicate_bytes(&self) -> usize {
63        size_of::<IdSelection>()
64            + self
65                .scans
66                .allocated_bytes()
67                .expect("allocated selector fits usize")
68            + size_of::<BaselineSelection>()
69            + self.baselines.allocated_bytes()
70            + size_of::<TimeSelection>()
71            + self.times.allocated_bytes()
72    }
73
74    /// Whether the antenna selection holds `baseline`.
75    pub(crate) fn baseline_holds(&self, baseline: OrderedBaseline) -> bool {
76        self.baselines
77            .holds(baseline.antenna1.0, baseline.antenna2.0)
78    }
79
80    /// Whether the scan and the time of one batch time are selected; a time
81    /// is tested as CASA's TaQL tests `TIME`, in seconds since MJD 0.
82    pub(crate) fn time_holds(
83        &self,
84        batch: &VisibilityBatch<'_>,
85        time: usize,
86    ) -> Result<bool, SourceError> {
87        if !self
88            .times
89            .holds(batch.time.seconds_since(time, MjdHighPrec::new(0.0, 0.0)))
90        {
91            return Ok(false);
92        }
93        if matches!(self.scans, IdSelection::All) {
94            return Ok(true);
95        }
96        let scans = batch
97            .scan
98            .as_ref()
99            .ok_or("the batch carries no requested scans")?;
100        Ok(batch.catalogue.scans[scans[time].0 as usize]
101            .number
102            .is_some_and(|number| self.scans.holds(i64::from(number))))
103    }
104
105    /// Whether a sample of partition `partition` with stored UVW `uvw_m` is
106    /// selected by UV distance.
107    pub(crate) fn uv_holds(&self, partition: &SelectedPartition, uvw_m: [f64; 3]) -> bool {
108        self.uv.holds(uvw_m, partition.wavelength_m)
109    }
110}

112#[cfg(test)]
113mod tests {
114    use super::*;
115
116    #[test]
117    fn predicate_charge_tracks_owned_capacity() {
118        let mut selection = PartitionSelection {
119            data_group: String::new(),
120            partitions: vec![],
121            fields: FieldSelection::All,
122            scans: IdSelection::All,
123            baselines: BaselineSelection::All,
124            times: TimeSelection::All,
125            uv: UvSelection::All,
126        };
127        let baseline = selection.predicate_bytes();
128        assert_eq!(
129            baseline,
130            size_of::<IdSelection>() + size_of::<BaselineSelection>() + size_of::<TimeSelection>()
131        );
132        let mut ids = Vec::with_capacity(19);
133        ids.extend([1, 3]);
134        let bytes = ids.capacity() * size_of::<u32>();
135        selection.scans = IdSelection::Only(ids);
136        assert_eq!(selection.predicate_bytes(), baseline + bytes);
137        let mut ranges = Vec::with_capacity(17);
138        ranges.push((2, i32::MAX as u32));
139        let bytes = ranges.capacity() * size_of::<(u32, u32)>();
140        selection.scans = IdSelection::Ranges(ranges);
141        assert_eq!(selection.predicate_bytes(), baseline + bytes);
142        selection.scans = IdSelection::All;
143        let pairs = BaselineSelection::from_pairs(5, |antenna1, antenna2| antenna1 != antenna2);
144        let bytes = pairs.allocated_bytes();
145        assert!(bytes >= 25);
146        selection.baselines = pairs;
147        assert_eq!(selection.predicate_bytes(), baseline + bytes);
148        let conditions = Vec::with_capacity(3);
149        let bytes = bytes + 3 * size_of::<casa_imaging_model::TimeCondition>();
150        selection.times = TimeSelection::AnyOf(conditions);
151        assert_eq!(selection.predicate_bytes(), baseline + bytes);
152    }
153}
