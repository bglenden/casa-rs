2//! The MSv2 adapter's descriptive metadata (#937): every spectral window,
3//! the logical sources with their records, the states and the processors,
4//! read once at opening. A column the table lacks, or a cell it leaves
5//! undefined, is `None`; a unit or measure reference outside the table's
6//! definition refuses the store, naming the column, and so does a defined
7//! cell of another shape than its definition's (a window's per-channel
8//! arrays of `NUM_CHAN` values, SOURCE's fixed vectors of their length,
9//! each on one axis). A SOURCE line array whose length is not the row's
10//! `NUM_LINES` is dropped with a warning, since such a table is readable
11//! otherwise.

34/// The `f64` values of `value`, a `column` cell.
35pub(super) fn floats(
36    value: Option<&ArrayValue>,
37    column: &str,
38) -> Result<Option<Vec<f64>>, ObservationError> {
39    match value {
40        None => Ok(None),
41        Some(ArrayValue::Float64(values)) => Ok(Some(values.iter().copied().collect())),
42        Some(_) => Err(invalid(format!("{column} is not a Float64 array"))),
43    }
44}
45
46/// The `f64` values of `value`, a `column` cell of one axis.
47fn vector(value: Option<&ArrayValue>, column: &str) -> Result<Option<Vec<f64>>, ObservationError> {
48    match value {
49        Some(ArrayValue::Float64(values)) if values.ndim() != 1 => Err(invalid(format!(
50            "{column} has shape {:?}, not one axis",
51            values.shape()
52        ))),
53        value => floats(value, column),
54    }
55}

57/// `value`, a `column` cell, as exactly `N` values on one axis.
58fn fixed<const N: usize>(
59    value: Option<&ArrayValue>,
60    column: &str,
61) -> Result<Option<[f64; N]>, ObservationError> {
62    vector(value, column)?
63        .map(|values| {
64            <[f64; N]>::try_from(values.as_slice())
65                .map_err(|_| invalid(format!("{column} holds {} values, not {N}", values.len())))
66        })
67        .transpose()
68}
69
70/// `value`, a `column` cell of a window of `channels` channels, as one
71/// value per channel.
72fn per_channel(
73    value: Option<&ArrayValue>,
74    column: &str,
75    channels: i32,
76) -> Result<Option<Vec<f64>>, ObservationError> {
77    vector(value, column)?
78        .map(|values| {
79            if i32::try_from(values.len()) == Ok(channels) {
80                Ok(values)
81            } else {
82                Err(invalid(format!(
83                    "{column} holds {} values for a window of NUM_CHAN {channels}",
84                    values.len()
85                )))
86            }
87        })
88        .transpose()
89}

91/// The time scale of a table's `column` of epochs, in seconds: its fixed
92/// `MEASINFO.Ref`, or UTC, the definition's default.
93pub(super) fn epoch_scale(table: &Table, column: &str) -> Result<EpochRef, ObservationError> {
94    require_units(table, column, "s")?;
95    match measure_reference(table, column)? {
96        Some(name) => name
97            .parse()
98            .map_err(|_| invalid(format!("unknown {column} reference {name}"))),
99        None => Ok(EpochRef::UTC),
100    }
101}

103/// A column's fixed measure reference parsed as `R`, or `default` without
104/// one.
105fn reference<R: std::str::FromStr>(
106    table: &Table,
107    column: &str,
108    default: Option<R>,
109) -> Result<Option<R>, ObservationError> {
110    match measure_reference(table, column)? {
111        Some(name) => name
112            .parse()
113            .map(Some)
114            .map_err(|_| invalid(format!("unknown {column} reference {name}"))),
115        None => Ok(default),
116    }
117}

119/// Every `SPECTRAL_WINDOW` row, used by MAIN or not.
120pub(super) fn read_windows(ms: &MeasurementSet) -> Result<Vec<WindowEntry>, ObservationError> {
121    let Some(table) = ms.subtable(SubtableId::SpectralWindow) else {
122        return Ok(Vec::new());
123    };
124    for column in [
125        "REF_FREQUENCY",
126        "TOTAL_BANDWIDTH",
127        "EFFECTIVE_BW",
128        "RESOLUTION",
129    ] {
130        require_units(table, column, "Hz")?;
131    }
132    (0..table.row_count())
133        .map(|row| {
134            let code = get_i32(table, row, "MEAS_FREQ_REF").map_err(storage)?;
135            let frame = FrequencyRef::from_casacore_code(code)
136                .ok_or_else(|| invalid(format!("unknown MEAS_FREQ_REF {code}")))?;
137            let channels = get_i32(table, row, "NUM_CHAN").map_err(storage)?;
138            Ok(WindowEntry {
139                name: get_string(table, row, "NAME").map_err(storage)?,
140                reference_frequency: optional_f64(table, row, "REF_FREQUENCY")
141                    .map_err(storage)?
142                    .map(|hz| MFrequency::new(hz, frame)),
143                total_bandwidth_hz: optional_f64(table, row, "TOTAL_BANDWIDTH").map_err(storage)?,
144                if_chain: optional_i32(table, row, "IF_CONV_CHAIN").map_err(storage)?,
145                net_sideband: optional_i32(table, row, "NET_SIDEBAND").map_err(storage)?,
146                frequency_group: optional_i32(table, row, "FREQ_GROUP").map_err(storage)?,
147                frequency_group_name: optional_string(table, row, "FREQ_GROUP_NAME")
148                    .map_err(storage)?,
149                effective_bandwidths_hz: per_channel(
150                    optional_array(table, row, "EFFECTIVE_BW").map_err(storage)?,
151                    "EFFECTIVE_BW",
152                    channels,
153                )?,
154                resolutions_hz: per_channel(
155                    optional_array(table, row, "RESOLUTION").map_err(storage)?,
156                    "RESOLUTION",
157                    channels,
158                )?,
159                flagged: optional_bool(table, row, "FLAG_ROW").map_err(storage)?,
160            })
161        })
162        .collect()
163}

165/// The logical sources and each stored `SOURCE_ID`'s identity.
166#[derive(Default)]
167pub(super) struct Sources {
168    pub(super) entries: Vec<SourceEntry>,
169    pub(super) ids: BTreeMap<i32, SourceId>,
170}
171
172/// The frames of `SOURCE`'s measure columns.
173struct SourceFrames {
174    time: EpochRef,
175    direction: DirectionRef,
176    rest: FrequencyRef,
177    velocity: Option<RadialVelocityRef>,
178}
179
180impl SourceFrames {
181    fn of(table: &Table) -> Result<Self, ObservationError> {
182        for (column, unit) in [
183            ("INTERVAL", "s"),
184            ("DIRECTION", "rad"),
185            ("PROPER_MOTION", "rad/s"),
186            ("POSITION", "m"),
187            ("REST_FREQUENCY", "Hz"),
188            ("SYSVEL", "m/s"),
189        ] {
190            require_units(table, column, unit)?;
191        }
192        if let Some(name) = measure_reference(table, "POSITION")?
193            && name != "ITRF"
194        {
195            return Err(ObservationError::Unsupported {
196                what: "MSv2 SOURCE positions in other than ITRF",
197            });
198        }
199        Ok(Self {
200            time: epoch_scale(table, "TIME")?,
201            direction: reference(table, "DIRECTION", Some(DirectionRef::J2000))?
202                .expect("a default"),
203            rest: reference(table, "REST_FREQUENCY", Some(FrequencyRef::LSRK))?.expect("a default"),
204            velocity: reference(table, "SYSVEL", None)?,
205        })
206    }
207}

209/// Every `SOURCE` row, as records of the logical sources their
210/// `SOURCE_ID`s name, in stored order; `windows` is how many spectral
211/// windows the store has. A record naming a window `SPECTRAL_WINDOW` does
212/// not hold is skipped with a warning.
213pub(super) fn read_sources(
214    ms: &MeasurementSet,
215    windows: usize,
216) -> Result<Sources, ObservationError> {
217    let Some(table) = ms.subtable(SubtableId::Source) else {
218        return Ok(Sources::default());
219    };
220    let frames = SourceFrames::of(table)?;
221    let mut sources = Sources::default();
222    for row in 0..table.row_count() {
223        let window = get_i32(table, row, "SPECTRAL_WINDOW_ID").map_err(storage)?;
224        let window = match usize::try_from(window) {
225            Err(_) => SourceWindow::All,
226            Ok(window) if window < windows => SourceWindow::Window(WindowId(window as u32)),
227            Ok(_) => {
228                tracing::warn!(
229                    "SOURCE row {row} names spectral window {window}, which SPECTRAL_WINDOW does \
230                     not hold; the record is not read"
231                );
232                continue;
233            }
234        };
235        let record = source_record(table, row, window, &frames)?;
236        let stored = get_i32(table, row, "SOURCE_ID").map_err(storage)?;
237        let next = SourceId(u32::try_from(sources.entries.len()).expect("few sources"));
238        let id = *sources.ids.entry(stored).or_insert(next);
239        if id == next {
240            sources.entries.push(SourceEntry {
241                name: get_string(table, row, "NAME").map_err(storage)?,
242                records: Vec::new(),
243            });
244        }
245        sources.entries[id.0 as usize].records.push(record);
246    }
247    Ok(sources)
248}

250/// One `SOURCE` row's record for `window`.
251fn source_record(
252    table: &Table,
253    row: usize,
254    window: SourceWindow,
255    frames: &SourceFrames,
256) -> Result<SourceRecord, ObservationError> {
257    let array = |column: &str| optional_array(table, row, column).map_err(storage);
258    let num_lines = optional_i32(table, row, "NUM_LINES")
259        .map_err(storage)?
260        .map_or(0, |lines| u32::try_from(lines).unwrap_or(0));
261    let lines = |column: &str, length: usize| {
262        let matches = length == num_lines as usize;
263        if !matches {
264            tracing::warn!(
265                "SOURCE row {row} holds {length} {column} values for {num_lines} lines; they are \
266                 not read"
267            );
268        }
269        matches
270    };
271    let transitions = match array("TRANSITION")? {
272        None => None,
273        Some(ArrayValue::String(values)) => Some(values.iter().cloned().collect::<Vec<_>>()),
274        Some(_) => return Err(invalid("SOURCE TRANSITION is not a String array")),
275    }
276    .filter(|values| lines("TRANSITION", values.len()));
277    let rest_frequencies = floats(array("REST_FREQUENCY")?, "REST_FREQUENCY")?
278        .filter(|values| lines("REST_FREQUENCY", values.len()))
279        .map(|values| {
280            values
281                .into_iter()
282                .map(|hz| MFrequency::new(hz, frames.rest))
283                .collect()
284        });
285    let systemic_velocities = floats(array("SYSVEL")?, "SYSVEL")?
286        .filter(|values| lines("SYSVEL", values.len()))
287        .map(|values| {
288            values
289                .into_iter()
290                .map(|value_m_s| Velocity {
291                    value_m_s,
292                    frame: frames.velocity,
293                })
294                .collect()
295        });
296    Ok(SourceRecord {
297        window,
298        time: optional_f64(table, row, "TIME")
299            .map_err(storage)?
300            .map(|seconds| MEpoch::from_mjd_seconds(seconds, frames.time)),
301        interval_s: optional_f64(table, row, "INTERVAL").map_err(storage)?,
302        code: optional_string(table, row, "CODE").map_err(storage)?,
303        calibration_group: optional_i32(table, row, "CALIBRATION_GROUP").map_err(storage)?,
304        direction: fixed::<2>(array("DIRECTION")?, "DIRECTION")?.map(|[longitude, latitude]| {
305            MDirection::from_angles(longitude, latitude, frames.direction)
306        }),
307        proper_motion_rad_s: fixed::<2>(array("PROPER_MOTION")?, "PROPER_MOTION")?,
308        position: fixed::<3>(array("POSITION")?, "POSITION")?
309            .map(|[x, y, z]| MPosition::new_itrf(x, y, z)),
310        pulsar_id: optional_i32(table, row, "PULSAR_ID").map_err(storage)?,
311        num_lines,
312        transitions,
313        rest_frequencies,
314        systemic_velocities,
315    })
316}

318/// Every `STATE` row; none without the table.
319pub(super) fn read_states(ms: &MeasurementSet) -> Result<Vec<StateEntry>, ObservationError> {
320    let Some(table) = ms.subtable(SubtableId::State) else {
321        return Ok(Vec::new());
322    };
323    for column in ["CAL", "LOAD"] {
324        require_units(table, column, "K")?;
325    }
326    (0..table.row_count())
327        .map(|row| {
328            Ok(StateEntry {
329                obs_mode: get_string(table, row, "OBS_MODE").map_err(storage)?,
330                cal_k: optional_f64(table, row, "CAL").map_err(storage)?,
331                load_k: optional_f64(table, row, "LOAD").map_err(storage)?,
332                signal: optional_bool(table, row, "SIG").map_err(storage)?,
333                reference: optional_bool(table, row, "REF").map_err(storage)?,
334                sub_scan: optional_i32(table, row, "SUB_SCAN").map_err(storage)?,
335                flagged: optional_bool(table, row, "FLAG_ROW").map_err(storage)?,
336            })
337        })
338        .collect()
339}

385        .expect("ms");

389                .iter()
390                .find(|definition| definition.name == name)

394                    ColumnSchema::array_variable(name, definition.data_type, None),
395                    None,
396                )
397                .expect("column");
398        }

403            .write(table)
404            .expect("SYSVEL MEASINFO");

408    /// A SOURCE row of `id` for `window` stating `lines` lines, with the
409    /// rest frequencies and velocities of `stated` when given.
410    fn source(
411        ms: &mut MeasurementSet,
412        (id, window, name): (i32, i32, &str),
413        lines: i32,
414        stated: Option<(&[f64], &[f64])>,
415    ) {
416        let int = |value| Value::Scalar(ScalarValue::Int32(value));
417        let mut fields = vec![
418            RecordField::new("CALIBRATION_GROUP", int(0)),
419            RecordField::new("CODE", Value::Scalar(ScalarValue::String("OBJ".into()))),
420            RecordField::new("DIRECTION", values(&[1.0, 0.5])),
421            RecordField::new("INTERVAL", Value::Scalar(ScalarValue::Float64(1e30))),
422            RecordField::new("NAME", Value::Scalar(ScalarValue::String(name.into()))),
423            RecordField::new("NUM_LINES", int(lines)),
424            RecordField::new("PROPER_MOTION", values(&[0.0, 0.0])),
425            RecordField::new("SOURCE_ID", int(id)),
426            RecordField::new("SPECTRAL_WINDOW_ID", int(window)),
427            RecordField::new("TIME", Value::Scalar(ScalarValue::Float64(4.8e9))),
428        ];
429        if let Some((rest, velocities)) = stated {
430            fields.push(RecordField::new("REST_FREQUENCY", values(rest)));
431            fields.push(RecordField::new("SYSVEL", values(velocities)));
432        }
433        ms.subtable_mut(SubtableId::Source)
434            .expect("SOURCE")
435            .add_row(RecordValue::new(fields))
436            .expect("row");
437    }

439    /// A logical source holds every record of its `SOURCE_ID`, each for its
440    /// own window or for all of them, in stored order; two sources of one
441    /// name stay two. A record for a window the store lacks is not read,
442    /// and a line array of another length than `NUM_LINES` is dropped.
443    #[test]
444    fn sources_are_logical_with_their_records_by_window() {
445        let mut ms = with_lines();
446        source(&mut ms, (0, 0, "M51"), 1, Some((&[330.0e9], &[10.0])));
447        source(&mut ms, (0, 1, "M51"), 1, Some((&[331.0e9], &[12.0])));
448        source(&mut ms, (0, -1, "M51"), 0, None);
449        source(&mut ms, (7, 0, "M51"), 1, Some((&[1.0e9], &[0.0])));
450        source(&mut ms, (3, 5, "stray"), 0, None);
451        source(&mut ms, (4, 0, "short"), 2, Some((&[2.0e9], &[0.0])));
452        let sources = read_sources(&ms, 2).expect("sources");
453        assert_eq!(
454            sources.ids,
455            BTreeMap::from([(0, SourceId(0)), (7, SourceId(1)), (4, SourceId(2))])
456        );
457        let m51 = &sources.entries[0];
458        assert_eq!(m51.name, "M51");
459        let windows: Vec<SourceWindow> = m51.records.iter().map(|record| record.window).collect();
460        assert_eq!(
461            windows,
462            [
463                SourceWindow::Window(WindowId(0)),
464                SourceWindow::Window(WindowId(1)),
465                SourceWindow::All
466            ]
467        );
468        let line = m51.records[1].rest_frequencies.as_ref().expect("lines")[0].clone();
469        assert_eq!((line.hz(), line.refer()), (331.0e9, FrequencyRef::BARY));
470        let velocity = m51.records[1].systemic_velocities.as_ref().expect("lines")[0];
471        assert_eq!(
472            (velocity.value_m_s, velocity.frame),
473            (12.0, Some(RadialVelocityRef::LSRK))
474        );
475        assert_eq!(m51.records[2].num_lines, 0);
476        assert!(m51.records[2].rest_frequencies.is_none(), "undefined cells");
477        assert_eq!(m51.records[0].code.as_deref(), Some("OBJ"));
478        assert_eq!(sources.entries[1].name, "M51", "another source of one name");
479        let short = &sources.entries[2].records[0];
480        assert_eq!(short.num_lines, 2);
481        assert!(short.rest_frequencies.is_none() && short.systemic_velocities.is_none());
482    }
