11/// Heap bytes of one `f64` plane of `shape`.
12pub(crate) const fn plane_bytes(shape: PlaneShape) -> u64 {
13    (shape.len() * size_of::<f64>()) as u64
14}
15
16/// Heap bytes of one half spectrum of `shape` ([`PlaneFft::forward`]).
17pub(crate) const fn spectrum_bytes(shape: PlaneShape) -> u64 {
18    match RealFft2::<f64>::spectrum_len([shape.nx, shape.ny]) {
19        Some(len) => (len * size_of::<Complex64>()) as u64,
20        None => u64::MAX,
21    }
22}

24/// C heap the plans of a [`PlaneFft`] on `threads` threads hold beside the
25/// process cache: a threaded transform owns its plans
26/// ([`casa_fft::plan_pair_bytes`], #780 review); one thread's are the
27/// cache's ([`casa_fft::PLAN_CACHE_BYTES`]).
28pub(crate) const fn owned_plan_bytes(threads: usize) -> u64 {
29    if threads > 1 {
30        casa_fft::plan_pair_bytes(threads) as u64
31    } else {
32        0
33    }
34}
35
36/// Heap bytes planning the half-spectrum transform of `shape` holds while
37/// FFTW plans it ([`casa_fft::planning_bytes`]).
38pub(crate) const fn planning_bytes(shape: PlaneShape) -> u64 {
39    match RealFft2::<f64>::spectrum_len([shape.nx, shape.ny]) {
40        Some(len) => casa_fft::planning_bytes::<f64>(len),
41        None => u64::MAX,
42    }
43}

45/// Heap bytes one [`ScaleBank::convolve`] holds while it runs: the product
46/// spectrum, then the plane it returns.
47pub(crate) const fn convolve_bytes(shape: PlaneShape) -> u64 {
48    spectrum_bytes(shape) + plane_bytes(shape)
49}

51/// Real two-dimensional transforms of one plane shape, in double precision
52/// on `threads` FFTW threads. Threaded transforms own their plans, outside
53/// the process cache ([`owned_plan_bytes`]).
54pub(crate) struct PlaneFft {
55    shape: PlaneShape,
56    fft: RealFft2<f64>,
57}
58
59impl PlaneFft {
60    pub(crate) fn new(shape: PlaneShape, threads: usize) -> Result<Self, Error> {
61        let fft = RealFft2::with_threads([shape.nx, shape.ny], threads.max(1))?;
62        Ok(Self {
63            shape,
64            fft: if threads > 1 {
65                fft.with_owned_plan()
66            } else {
67                fft
68            },
69        })
70    }
71
72    /// The half spectrum of `plane`.
73    pub(crate) fn forward(&mut self, plane: &[f64]) -> Result<Vec<Complex64>, Error> {
74        let mut storage = vec![Complex64::default(); self.fft.storage_len()];
75        let stride = self.fft.real_row_stride();
76        let real: &mut [f64] = bytemuck::cast_slice_mut(&mut storage);
77        for x in 0..self.shape.nx {
78            real[x * stride..][..self.shape.ny]
79                .copy_from_slice(&plane[x * self.shape.ny..][..self.shape.ny]);
80        }
81        self.fft.forward(&mut storage)?;
82        Ok(storage)
83    }
84
85    /// The plane of a half spectrum, normalised by the pixel count.
86    pub(crate) fn inverse(&mut self, mut spectrum: Vec<Complex64>) -> Result<Vec<f64>, Error> {
87        self.fft.inverse(&mut spectrum)?;
88        let stride = self.fft.real_row_stride();
89        let scale = 1.0 / self.shape.len() as f64;
90        let real: &[f64] = bytemuck::cast_slice(&spectrum);
91        let mut plane = vec![0.0; self.shape.len()];
92        for x in 0..self.shape.nx {
93            for (value, &sample) in plane[x * self.shape.ny..][..self.shape.ny]
94                .iter_mut()
95                .zip(&real[x * stride..][..self.shape.ny])
96            {
97                *value = sample * scale;
98            }
99        }
100        Ok(plane)
101    }
102}

104/// `MatrixCleaner::spheroidal`: the rational approximation to the
105/// prolate spheroidal function the scales are tapered by.
106#[must_use]
107pub fn spheroidal(nu: f64) -> f64 {
108    if nu <= 0.0 {
109        return 1.0;
110    }
111    if nu >= 1.0 {
112        return 0.0;
113    }
114    let (p, q, end) = if nu < 0.75 {
115        (
116            [
117                8.203343e-2,
118                -3.644705e-1,
119                6.278660e-1,
120                -5.335581e-1,
121                2.312756e-1,
122            ],
123            [1.0, 8.212018e-1, 2.078043e-1],
124            0.75,
125        )
126    } else {
127        (
128            [
129                4.028559e-3,
130                -3.697768e-2,
131                1.021332e-1,
132                -1.201436e-1,
133                6.412774e-2,
134            ],
135            [1.0, 9.599102e-1, 2.918724e-1],
136            1.0,
137        )
138    };
139    let delta = nu * nu - end * end;
140    let top = p
141        .iter()
142        .rev()
143        .fold(0.0, |sum, coefficient| sum * delta + coefficient);
144    let bottom = q
145        .iter()
146        .rev()
147        .fold(0.0, |sum, coefficient| sum * delta + coefficient);
148    if bottom == 0.0 { 0.0 } else { top / bottom }
149}

151/// One scale function: `(1 − r²)·spheroidal(r)` inside radius `size`
152/// pixels, normalised to unit sum, as its nonzero offsets from the centre.
153#[derive(Clone, Debug, PartialEq)]
154pub(crate) struct ScaleFunction {
155    /// The scale size in pixels.
156    pub(crate) size: f64,
157    /// `(dx, dy, value)` of every nonzero sample.
158    pub(crate) samples: Vec<(isize, isize, f64)>,
159}
160
161impl ScaleFunction {
162    /// `MatrixCleaner::makeScale` centred at `centre` of `shape`: a delta for
163    /// size 0, otherwise the tapered disc sampled within the plane.
164    pub(crate) fn new(size: f64, shape: PlaneShape) -> Self {
165        let [cx, cy] = shape.centre();
166        if size == 0.0 {
167            return Self {
168                size,
169                samples: vec![(0, 0, 1.0)],
170            };
171        }
172        let low = |centre: usize| ((centre as f64 - size) as isize).max(0);
173        let high = |centre: usize, extent: usize| {
174            ((centre as f64 + size) as isize).min(extent as isize - 1)
175        };
176        let mut samples = Vec::new();
177        let mut volume = 0.0;
178        for y in low(cy)..=high(cy, shape.ny) {
179            let y_part = ((cy as f64 - y as f64) / size).powi(2);
180            for x in low(cx)..=high(cx, shape.nx) {
181                let r2 = y_part + ((cx as f64 - x as f64) / size).powi(2);
182                if r2 < 1.0 {
183                    let value = (1.0 - r2) * spheroidal(r2.max(0.0).sqrt());
184                    volume += value;
185                    samples.push((x - cx as isize, y - cy as isize, value));
186                }
187            }
188        }
189        for sample in &mut samples {
190            sample.2 /= volume;
191        }
192        samples.retain(|sample| sample.2 != 0.0);
193        Self { size, samples }
194    }
195
196    /// The function as a plane centred at the origin, wrapping negative
197    /// offsets, so that a product of transforms is a centred convolution.
198    fn wrapped(&self, shape: PlaneShape) -> Vec<f64> {
199        let mut plane = vec![0.0; shape.len()];
200        for &(dx, dy, value) in &self.samples {
201            let x = dx.rem_euclid(shape.nx as isize) as usize;
202            let y = dy.rem_euclid(shape.ny as isize) as usize;
203            plane[shape.index(x, y)] += value;
204        }
205        plane
206    }
207}

209/// The scales of one solve with their transforms and small-scale biases.
210pub(crate) struct ScaleBank {
211    shape: PlaneShape,
212    pub(crate) functions: Vec<ScaleFunction>,
213    pub(crate) bias: Vec<f64>,
214    transforms: Vec<Vec<Complex64>>,
215    pub(crate) fft: PlaneFft,
216}
217
218impl ScaleBank {
219    /// Heap bytes a bank of `sizes` on `shape` with transforms on
220    /// `fft_threads` threads holds, with what building it holds besides:
221    /// each scale's spectrum and samples, its transform's own plans when
222    /// threaded ([`owned_plan_bytes`]), and one wrapped plane and its
223    /// spectrum at a time, beside planning their transform.
224    pub(crate) fn bytes(sizes: &[f64], shape: PlaneShape, fft_threads: usize) -> u64 {
225        Self::resident_bytes(sizes, shape)
226            + owned_plan_bytes(fft_threads)
227            + plane_bytes(shape)
228            + spectrum_bytes(shape)
229            + planning_bytes(shape)
230    }
231
232    /// Heap bytes a built bank of `sizes` on `shape` keeps: each scale's
233    /// spectrum and samples.
234    pub(crate) fn resident_bytes(sizes: &[f64], shape: PlaneShape) -> u64 {
235        let samples = sizes
236            .iter()
237            .map(|&size| {
238                let count = Self::scale_cells(size, shape).next_power_of_two();
239                (count * size_of::<(isize, isize, f64)>()) as u64
240            })
241            .sum::<u64>();
242        sizes.len() as u64 * spectrum_bytes(shape) + samples
243    }
244
245    /// Samples one scale function of `size` pixels has at most on `shape`:
246    /// the disc's bounding square, as [`ScaleFunction::new`] scans it.
247    fn scale_cells(size: f64, shape: PlaneShape) -> usize {
248        let side = 2 * size as usize + 3;
249        (side * side).min(shape.len())
250    }
251
252    /// Pixels a component of any of `sizes` covers at most on `shape`.
253    pub(crate) fn support_cells(sizes: &[f64], shape: PlaneShape) -> usize {
254        sizes
255            .iter()
256            .map(|&size| Self::scale_cells(size, shape))
257            .max()
258            .unwrap_or(1)
259    }
260
261    /// Heap bytes [`Self::masks`] holds while it runs, besides the masks it
262    /// returns: the support as a plane, its spectrum and one convolution.
263    pub(crate) const fn masks_bytes(shape: PlaneShape) -> u64 {
264        plane_bytes(shape) + spectrum_bytes(shape) + convolve_bytes(shape)
265    }
266
267    /// The scales `sizes` (ascending) with CASA's bias
268    /// `1 − bias·size/largest` (1 for a single scale); `small_scale_bias`
269    /// is clamped to [−1, 1] as `SDAlgorithmMSClean` does. Its transforms
270    /// run on `fft_threads` threads.
271    pub(crate) fn new(
272        sizes: &[f64],
273        small_scale_bias: f64,
274        shape: PlaneShape,
275        fft_threads: usize,
276    ) -> Result<Self, Error> {
277        let small_scale_bias = small_scale_bias.clamp(-1.0, 1.0);
278        let largest = sizes.last().copied().unwrap_or(0.0);
279        let bias = sizes
280            .iter()
281            .map(|size| {
282                if sizes.len() > 1 {
283                    1.0 - small_scale_bias * size / largest
284                } else {
285                    1.0
286                }
287            })
288            .collect();
289        let functions = sizes
290            .iter()
291            .map(|&size| ScaleFunction::new(size, shape))
292            .collect::<Vec<_>>();
293        let mut fft = PlaneFft::new(shape, fft_threads)?;
294        let transforms = functions
295            .iter()
296            .map(|function| fft.forward(&function.wrapped(shape)))
297            .collect::<Result<Vec<_>, _>>()?;
298        Ok(Self {
299            shape,
300            functions,
301            bias,
302            transforms,
303            fft,
304        })
305    }
306
307    /// Number of scales.
308    pub(crate) fn len(&self) -> usize {
309        self.functions.len()
310    }
311
312    /// `spectrum` times the transforms of the listed scales, back in the
313    /// image domain: the centred circular convolution with each.
314    pub(crate) fn convolve(
315        &mut self,
316        spectrum: &[Complex64],
317        scales: &[usize],
318    ) -> Result<Vec<f64>, Error> {
319        let mut product = spectrum.to_vec();
320        for &scale in scales {
321            for (value, kernel) in product.iter_mut().zip(&self.transforms[scale]) {
322                *value *= kernel;
323            }
324        }
325        self.fft.inverse(product)
326    }
327
328    /// The support convolved with each scale, above `threshold`, with the
329    /// `border(size)` pixels at every edge removed (`makeScaleMasks`,
330    /// `setupUserMask`). `skip_point_border` keeps the edge of the point
331    /// scale, as the multi-term cleaner does.
332    pub(crate) fn masks(
333        &mut self,
334        support: &Support,
335        threshold: f64,
336        skip_point_border: bool,
337    ) -> Result<Vec<Support>, Error> {
338        let shape = self.shape;
339        let plane = support
340            .as_slice()
341            .iter()
342            .map(|supported| f64::from(u8::from(*supported)))
343            .collect::<Vec<_>>();
344        let spectrum = self.fft.forward(&plane)?;
345        (0..self.len())
346            .map(|scale| {
347                let smoothed = self.convolve(&spectrum, &[scale])?;
348                let border = if skip_point_border && scale == 0 {
349                    None
350                } else {
351                    Some((self.functions[scale].size * 1.5) as usize)
352                };
353                let pixels = smoothed
354                    .iter()
355                    .enumerate()
356                    .map(|(index, value)| {
357                        let [x, y] = shape.pixel(index);
358                        let inside = border.is_none_or(|border| {
359                            x > border
360                                && y > border
361                                && x + border + 1 < shape.nx
362                                && y + border + 1 < shape.ny
363                        });
364                        inside && *value > threshold
365                    })
366                    .collect();
367                Ok(Support::new(shape, pixels))
368            })
369            .collect()
370    }
371}

377    #[test]
378    fn scale_functions_have_unit_sum_and_the_casa_radius() {
379        let shape = PlaneShape::new(64, 48);
380        for size in [0.0, 1.5, 6.0, 10.0] {
381            let function = ScaleFunction::new(size, shape);
382            let sum = function.samples.iter().map(|sample| sample.2).sum::<f64>();
383            assert!((sum - 1.0).abs() < 1e-12, "size {size}: sum {sum}");
384            assert!(
385                function
386                    .samples
387                    .iter()
388                    .all(|&(dx, dy, _)| ((dx * dx + dy * dy) as f64) < size * size || size == 0.0)
389            );
390        }
391        assert_eq!(spheroidal(0.0), 1.0);
392        assert_eq!(spheroidal(1.0), 0.0);
393        // Continuous across the 0.75 break.
394        assert!((spheroidal(0.749_999_9) - spheroidal(0.75)).abs() < 1e-5);
395    }

397    /// Transform products are centred convolutions: a delta convolved with a
398    /// scale is the scale centred on the delta, wrapping at the edges.
399    #[test]
400    fn convolution_is_centred_and_circular() {
401        let shape = PlaneShape::new(15, 12);
402        let mut bank = ScaleBank::new(&[0.0, 3.0], 0.6, shape, 1).unwrap();
403        assert_eq!(bank.bias, vec![1.0, 0.4]);
404        let mut delta = vec![0.0; shape.len()];
405        delta[shape.index(1, 10)] = 2.0;
406        let spectrum = bank.fft.forward(&delta).unwrap();
407        let convolved = bank.convolve(&spectrum, &[1]).unwrap();
408        let mut expected = vec![0.0; shape.len()];
409        for &(dx, dy, value) in &bank.functions[1].samples {
410            let x = (1 + dx).rem_euclid(15) as usize;
411            let y = (10 + dy).rem_euclid(12) as usize;
412            expected[shape.index(x, y)] += 2.0 * value;
413        }
414        for (a, e) in convolved.iter().zip(&expected) {
415            assert!((a - e).abs() < 1e-12);
416        }
417    }

419    #[test]
420    fn scale_masks_drop_the_border_and_partial_overlaps() {
421        let shape = PlaneShape::new(32, 32);
422        let mut bank = ScaleBank::new(&[0.0, 4.0], 0.0, shape, 1).unwrap();
423        let support = Support::new(
424            shape,
425            (0..shape.len())
426                .map(|index| {
427                    let [x, y] = shape.pixel(index);
428                    (8..24).contains(&x) && (8..24).contains(&y)
429                })
430                .collect(),
431        );
432        let masks = bank.masks(&support, 0.9, false).unwrap();
433        // The point scale keeps the box; a full-image support loses its ring.
434        assert_eq!(masks[0], support);
435        let full = bank.masks(&Support::full(shape), 0.9, false).unwrap();
436        assert!(!full[0].contains(shape.index(0, 5)) && full[0].contains(shape.index(1, 1)));
437        assert!(bank.masks(&Support::full(shape), 0.1, true).unwrap()[0].contains(0));
438        // The 4-pixel scale needs 90% of its weight inside the box.
439        assert!(masks[1].contains(shape.index(16, 16)));
440        assert!(!masks[1].contains(shape.index(8, 16)));
441        assert!(masks[1].count() < support.count());
442    }
