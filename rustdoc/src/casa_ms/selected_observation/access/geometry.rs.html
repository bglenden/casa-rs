36pub(super) fn evaluate_row_geometry(
37    source: &BoundObservationSource,
38    problem: &CompiledProblem,
39    stored: SelectedStoredRow,
40    observation_pointing: Option<SelectedPointingDirections>,
41) -> Result<EvaluatedRowGeometry, BoundObservationSourceError> {
42    let field_id = usize::try_from(stored.field_id())
43        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
44    let (longitude_rad, latitude_rad) = source
45        .geometry_engine
46        .observation_direction_j2000(stored.time_mjd_seconds(), field_id)?
47        .as_angles();
48    let observation_direction =
49        SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad);
50    let centres = problem.geometry().centres();
51    let phase_direction = match centres.phase_tracking() {
52        PhaseCentreLaw::Observation => observation_direction,
53        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction)?,
54        PhaseCentreLaw::Ephemeris(target) => {
55            let direction = source.geometry_engine.moving_direction_j2000(
56                stored.time_mjd_seconds(),
57                field_id,
58                target,
59            )?;
60            let (longitude_rad, latitude_rad) = direction.as_angles();
61            SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad)
62        }
63    };
64    let pointing_directions = match centres.pointing() {
65        PointingCentreLaw::PhaseTrackingCentre => SelectedPointingDirections {
66            antenna1: phase_direction,
67            antenna2: phase_direction,
68        },
69        PointingCentreLaw::FieldCentre => SelectedPointingDirections {
70            antenna1: observation_direction,
71            antenna2: observation_direction,
72        },
73        PointingCentreLaw::Fixed(direction) => {
74            let direction = require_fixed_j2000(*direction)?;
75            SelectedPointingDirections {
76                antenna1: direction,
77                antenna2: direction,
78            }
79        }
80        PointingCentreLaw::Observation(_) => observation_pointing
81            .ok_or(BoundObservationSourceError::MissingEvaluatedPointingDirections)?,
82    };
83    let domains = problem.geometry().domains();
84    let moving_direction_shift = if matches!(centres.phase_tracking(), PhaseCentreLaw::Ephemeris(_))
85    {
86        let anchor = domains
87            .first()
88            .ok_or(BoundObservationSourceError::InvalidRowGeometry)?
89            .model_phase_centre();
90        let anchor_j2000 = source.geometry_engine.direction_angles_j2000(
91            stored.time_mjd_seconds(),
92            [anchor.longitude_rad(), anchor.latitude_rad()],
93            anchor.frame().direction_ref(),
94        )?;
95        Some([
96            phase_direction.longitude_rad() - anchor_j2000[0],
97            phase_direction.latitude_rad() - anchor_j2000[1],
98        ])
99    } else {
100        None
101    };
102    let chart_count = domains.iter().try_fold(0_usize, |count, domain| {
103        count.checked_add(domain.facets().len())
104    });
105    let mut domain_projections = Vec::new();
106    domain_projections
107        .try_reserve_exact(chart_count.ok_or(BoundObservationSourceError::InvalidRowGeometry)?)
108        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
109    for (domain_ordinal, domain) in domains.iter().enumerate() {
110        let domain_ordinal = u32::try_from(domain_ordinal)
111            .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
112        let shared_psf = domain.psf_phase_centre() == domain.model_phase_centre();
113        let distinct_psf = if shared_psf {
114            None
115        } else {
116            Some(evaluate_phase_centre_projection(
117                source,
118                stored,
119                observation_direction,
120                PhaseCentreProjectionRequest {
121                    field_id,
122                    target_direction: domain.psf_phase_centre(),
123                    project_to_observation_plane: domain.facets().len() > 1,
124                    moving_direction_shift,
125                    uvw_law: problem.geometry().uvw(),
126                },
127            )?)
128        };
129        for (facet_ordinal, facet) in domain.facets().iter().enumerate() {
130            let facet_ordinal = u32::try_from(facet_ordinal)
131                .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
132            let model_phase_centre = if domain.facets().len() == 1 {
133                domain.model_phase_centre()
134            } else {
135                facet.phase_centre()
136            };
137            let model = evaluate_phase_centre_projection(
138                source,
139                stored,
140                observation_direction,
141                PhaseCentreProjectionRequest {
142                    field_id,
143                    target_direction: model_phase_centre,
144                    project_to_observation_plane: domain.facets().len() > 1,
145                    moving_direction_shift,
146                    uvw_law: problem.geometry().uvw(),
147                },
148            )?;
149            let projection = match distinct_psf {
150                Some(psf) => SelectedImageDomainProjection::new_facet(
151                    domain_ordinal,
152                    facet_ordinal,
153                    model,
154                    psf,
155                ),
156                None => SelectedImageDomainProjection::facet_with_shared_psf(
157                    domain_ordinal,
158                    facet_ordinal,
159                    model,
160                ),
161            };
162            let projection = if let Some(plan) = &source.aw_pointing_plan {
163                let epoch = AwPointingEpochKey {
164                    data_description_id: stored.data_description_id(),
165                    field_id: stored.field_id(),
166                    time_bits: stored.time_mjd_seconds().to_bits(),
167                };
168                let chart = domain_projections.len();
169                let pixels = plan
170                    .chart_pixels_by_epoch
171                    .get(&epoch)
172                    .and_then(|charts| charts.get(chart))
173                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
174                let antenna1 = pixels
175                    .get(&stored.antenna1())
176                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
177                let antenna2 = pixels
178                    .get(&stored.antenna2())
179                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
180                projection
181                    .with_aw_pointing_pixel([
182                        0.5 * (antenna1[0] + antenna2[0]),
183                        0.5 * (antenna1[1] + antenna2[1]),
184                    ])
185                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?
186            } else {
187                projection
188            };
189            domain_projections.push(projection);
190        }
191    }
192    let domain_projections = SelectedImageDomainProjections::new(domain_projections)
193        .ok_or(BoundObservationSourceError::InvalidRowGeometry)?;
194    Ok(EvaluatedRowGeometry {
195        domain_projections,
196        pointing_directions,
197    })
198}

200struct PhaseCentreProjectionRequest {
201    field_id: usize,
202    target_direction: SkyDirection,
203    project_to_observation_plane: bool,
204    moving_direction_shift: Option<[f64; 2]>,
205    uvw_law: casa_imaging_model::UvwCoordinateLaw,
206}
207
208fn evaluate_phase_centre_projection(
209    source: &BoundObservationSource,
210    stored: SelectedStoredRow,
211    observation_direction: SkyDirection,
212    request: PhaseCentreProjectionRequest,
213) -> Result<SelectedPhaseCentreProjection, BoundObservationSourceError> {
214    let mut target_angles = source.geometry_engine.direction_angles_j2000(
215        stored.time_mjd_seconds(),
216        [
217            request.target_direction.longitude_rad(),
218            request.target_direction.latitude_rad(),
219        ],
220        request.target_direction.frame().direction_ref(),
221    )?;
222    if let Some([longitude, latitude]) = request.moving_direction_shift {
223        target_angles[0] += longitude;
224        target_angles[1] += latitude;
225    }
226    let target_j2000 = SkyDirection::new(DirectionFrame::J2000, target_angles[0], target_angles[1]);
227    let (transformed_uvw_m, phase_shift_m) = if matches!(
228        request.uvw_law,
229        casa_imaging_model::UvwCoordinateLaw::MosaicPhaseTrackingCentre
230    ) {
231        if request.project_to_observation_plane {
232            return Err(BoundObservationSourceError::InvalidRowGeometry);
233        }
234        let target = MDirection::from_angles(
235            target_j2000.longitude_rad(),
236            target_j2000.latitude_rad(),
237            DirectionRef::J2000,
238        );
239        let (uvw_m, casa_dphase_m) = source
240            .geometry_engine
241            .reproject_raw_uvw_for_mosaic_to_direction(stored.uvw_m(), request.field_id, &target)?;
242        (uvw_m, -casa_dphase_m)
243    } else if target_j2000 == observation_direction && !request.project_to_observation_plane {
244        (stored.uvw_m(), 0.0)
245    } else if request.project_to_observation_plane {
246        source
247            .geometry_engine
248            .reproject_raw_uvw_for_faceted_gridft_between_j2000_directions(
249                stored.uvw_m(),
250                [
251                    observation_direction.longitude_rad(),
252                    observation_direction.latitude_rad(),
253                ],
254                target_angles,
255            )?
256    } else {
257        source
258            .geometry_engine
259            .reproject_raw_uvw_for_gridft_between_j2000_directions(
260                stored.uvw_m(),
261                [
262                    observation_direction.longitude_rad(),
263                    observation_direction.latitude_rad(),
264                ],
265                target_angles,
266            )?
267    };
268    SelectedPhaseCentreProjection::new(transformed_uvw_m, phase_shift_m)
269        .ok_or(BoundObservationSourceError::InvalidRowGeometry)
270}

272pub(super) fn evaluate_observation_pointings(
273    source: &BoundObservationSource,
274    problem: &CompiledProblem,
275    buffer: &SelectedObservationBuffer,
276) -> Result<Option<Vec<SelectedPointingDirections>>, BoundObservationSourceError> {
277    let PointingCentreLaw::Observation(law) = problem.geometry().centres().pointing() else {
278        return Ok(None);
279    };
280    let mut queries = Vec::with_capacity(buffer.row_count().saturating_mul(2));
281    let mut phase_directions = Vec::with_capacity(buffer.row_count());
282    for row in 0..buffer.row_count() {
283        let stored = buffer
284            .row(row)
285            .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
286        let time_mjd_seconds = match law.time_sampling() {
287            PointingTimeSampling::VisibilityTime => stored.time_mjd_seconds(),
288            PointingTimeSampling::VisibilityTimeCentroid => stored.time_centroid_mjd_seconds(),
289        };
290        queries.push(PointingDirectionQuery::new(
291            stored.antenna1(),
292            time_mjd_seconds,
293        )?);
294        queries.push(PointingDirectionQuery::new(
295            stored.antenna2(),
296            time_mjd_seconds,
297        )?);
298        phase_directions.push(evaluate_phase_direction(source, problem, stored)?);
299    }
300    let brackets = source
301        .pointing_catalog
302        .as_ref()
303        .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?
304        .direction_brackets(&source.geometry_engine, &queries)?;
305    let mut pointings = Vec::with_capacity(buffer.row_count());
306    for ((row, antenna_brackets), fallback) in brackets
307        .as_chunks::<2>()
308        .0
309        .iter()
310        .enumerate()
311        .zip(phase_directions)
312    {
313        pointings.push(SelectedPointingDirections {
314            antenna1: resolve_pointing_direction(
315                antenna_brackets[0],
316                queries[2 * row],
317                *law,
318                fallback,
319            )?,
320            antenna2: resolve_pointing_direction(
321                antenna_brackets[1],
322                queries[2 * row + 1],
323                *law,
324                fallback,
325            )?,
326        });
327    }
328    Ok(Some(pointings))
329}

331pub(super) fn build_aw_pointing_epoch_plan(
332    problem: &CompiledProblem,
333    source: &BoundObservationSource,
334) -> Result<AwPointingEpochPlan, BoundObservationSourceError> {
335    let grouping = AwPointingGrouping::new(problem)
336        .map_err(aw_pointing_error)?
337        .ok_or(BoundObservationSourceError::UnsupportedCentreLaw)?;
338    let law = match problem.geometry().centres().pointing() {
339        PointingCentreLaw::Observation(law) => *law,
340        _ => return Err(BoundObservationSourceError::UnsupportedCentreLaw),
341    };
342    let catalog = source
343        .pointing_catalog
344        .as_ref()
345        .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?;
346    let available_bytes = source
347        .content_plan
348        .rows_per_block()
349        .checked_mul(crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW)
350        .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
351    let scan_plan = MsReadPlan::new(
352        source.measurement_set.row_count(),
353        MsSelectionIoBudget {
354            available_bytes,
355            maximum_live_blocks: 1,
356            requested_bytes_per_row: crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW,
357            storage_alignment_rows: Some(source.content_plan.rows_per_block()),
358        },
359    )
360    .map_err(|error| MsError::InvalidInput(error.to_string()))?;
361    let mut cursor = source
362        .measurement_set
363        .main_row_selection_cursor(scan_plan)?;
364    let mut epochs = Vec::<AwPointingEpochInput>::new();
365    while let Some(row) = cursor.next(&source.measurement_set)? {
366        if !source.row_predicate.matches(row) {
367            continue;
368        }
369        let key = AwPointingEpochKey {
370            data_description_id: row.data_description_id(),
371            field_id: row.field_id(),
372            time_bits: row.time_mjd_seconds().to_bits(),
373        };
374        if epochs.last().is_none_or(|epoch| epoch.key != key) {
375            epochs.push(AwPointingEpochInput {
376                key,
377                row_count: 0,
378                antennas: BTreeSet::new(),
379            });
380        }
381        let epoch = epochs.last_mut().expect("AW pointing epoch was appended");
382        epoch.row_count = epoch
383            .row_count
384            .checked_add(1)
385            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
386        epoch.antennas.insert(row.antenna1());
387        epoch.antennas.insert(row.antenna2());
388    }
389
390    let mut chart_pixels_by_epoch = BTreeMap::new();
391    let mut groups = AwPointingEpochs::default();
392    for epoch in epochs {
393        let time = f64::from_bits(epoch.key.time_bits);
394        let fallback = pointing_epoch_phase_direction(source, problem, epoch.key.field_id, time)?;
395        let queries = epoch
396            .antennas
397            .iter()
398            .copied()
399            .map(|antenna| PointingDirectionQuery::new(antenna, time))
400            .collect::<Result<Vec<_>, _>>()?;
401        let brackets = catalog.direction_brackets(&source.geometry_engine, &queries)?;
402        let directions = epoch
403            .antennas
404            .iter()
405            .copied()
406            .zip(brackets)
407            .zip(queries.iter().copied())
408            .map(|((antenna, bracket), query)| {
409                resolve_pointing_direction(bracket, query, law, fallback)
410                    .map(|direction| (antenna, direction))
411            })
412            .collect::<Result<BTreeMap<_, _>, _>>()?;
413        let field = u32::try_from(epoch.key.field_id)
414            .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
415        let grouped = grouping
416            .pixels(&directions)
417            .and_then(|pixels| groups.next(&grouping, field, epoch.row_count, &pixels))
418            .map_err(aw_pointing_error)?;
419        if chart_pixels_by_epoch.insert(epoch.key, grouped).is_some() {
420            return Err(BoundObservationSourceError::InvalidRowGeometry);
421        }
422    }
423    let retained_byte_ceiling = size_of::<AwPointingEpochPlan>()
424        .checked_add(
425            chart_pixels_by_epoch
426                .values()
427                .try_fold(0_usize, |bytes, charts| {
428                    charts.iter().try_fold(bytes, |bytes, chart| {
429                        chart
430                            .len()
431                            .checked_mul(usize::BITS as usize)
432                            .and_then(|entries| bytes.checked_add(entries))
433                    })
434                })
435                .ok_or(BoundObservationSourceError::MeasurementOverflow)?,
436        )
437        .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
438    Ok(AwPointingEpochPlan {
439        chart_pixels_by_epoch,
440        retained_byte_ceiling,
441    })
442}

444fn aw_pointing_error(error: AwPointingError) -> BoundObservationSourceError {
445    match error {
446        AwPointingError::Overflow => BoundObservationSourceError::MeasurementOverflow,
447        _ => BoundObservationSourceError::InvalidRowGeometry,
448    }
449}

451fn pointing_epoch_phase_direction(
452    source: &BoundObservationSource,
453    problem: &CompiledProblem,
454    field_id: i32,
455    time_mjd_seconds: f64,
456) -> Result<SkyDirection, BoundObservationSourceError> {
457    let field =
458        usize::try_from(field_id).map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
459    let observation = source
460        .geometry_engine
461        .observation_direction_j2000(time_mjd_seconds, field)?;
462    let (longitude, latitude) = observation.as_angles();
463    let observation = SkyDirection::new(DirectionFrame::J2000, longitude, latitude);
464    match problem.geometry().centres().phase_tracking() {
465        PhaseCentreLaw::Observation => Ok(observation),
466        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction),
467        PhaseCentreLaw::Ephemeris(target) => {
468            let direction =
469                source
470                    .geometry_engine
471                    .moving_direction_j2000(time_mjd_seconds, field, target)?;
472            let (longitude, latitude) = direction.as_angles();
473            Ok(SkyDirection::new(
474                DirectionFrame::J2000,
475                longitude,
476                latitude,
477            ))
478        }
479    }
480}

482fn evaluate_phase_direction(
483    source: &BoundObservationSource,
484    problem: &CompiledProblem,
485    stored: SelectedStoredRow,
486) -> Result<SkyDirection, BoundObservationSourceError> {
487    let field_id = usize::try_from(stored.field_id())
488        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
489    let (longitude_rad, latitude_rad) = source
490        .geometry_engine
491        .observation_direction_j2000(stored.time_mjd_seconds(), field_id)?
492        .as_angles();
493    let observation = SkyDirection::new(DirectionFrame::J2000, longitude_rad, latitude_rad);
494    match problem.geometry().centres().phase_tracking() {
495        PhaseCentreLaw::Observation => Ok(observation),
496        PhaseCentreLaw::Fixed(direction) => require_fixed_j2000(*direction),
497        PhaseCentreLaw::Ephemeris(target) => {
498            let direction = source.geometry_engine.moving_direction_j2000(
499                stored.time_mjd_seconds(),
500                field_id,
501                target,
502            )?;
503            let (longitude_rad, latitude_rad) = direction.as_angles();
504            Ok(SkyDirection::new(
505                DirectionFrame::J2000,
506                longitude_rad,
507                latitude_rad,
508            ))
509        }
510    }
511}

513fn resolve_pointing_direction(
514    bracket: PointingDirectionBracket,
515    query: PointingDirectionQuery,
516    law: casa_imaging_model::ObservationPointingLaw,
517    fallback: SkyDirection,
518) -> Result<SkyDirection, BoundObservationSourceError> {
519    let sample = |candidate: Option<crate::selected_pointing::PointingDirectionCandidate>| {
520        candidate.map(|candidate| PointingSample {
521            sample: candidate.row_index(),
522            time_s: candidate.row_time_mjd_seconds(),
523            direction_rad: candidate.direction_j2000_rad(),
524        })
525    };
526    let samples = PointingSamples {
527        covering: sample(bracket.covering()),
528        before: sample(bracket.before()),
529        after: sample(bracket.after()),
530    };
531    law.resolve(samples, query.time_mjd_seconds(), fallback)
532        .map_err(|error| match error {
533            PointingLawError::OutsideCoverage => {
534                BoundObservationSourceError::PointingOutsideCoverage {
535                    antenna_id: query.antenna_id(),
536                    time_mjd_seconds: query.time_mjd_seconds(),
537                }
538            }
539            PointingLawError::Missing => BoundObservationSourceError::MissingPointingDirection {
540                antenna_id: query.antenna_id(),
541                time_mjd_seconds: query.time_mjd_seconds(),
542            },
543            PointingLawError::InvalidInterpolation => {
544                BoundObservationSourceError::InvalidPointingInterpolation
545            }
546        })
547}
548
549fn require_fixed_j2000(
550    direction: SkyDirection,
551) -> Result<SkyDirection, BoundObservationSourceError> {
552    if direction.frame() != DirectionFrame::J2000 {
553        return Err(BoundObservationSourceError::UnsupportedFixedCentreFrame {
554            frame: direction.frame(),
555        });
556    }
557    Ok(direction)
558}
