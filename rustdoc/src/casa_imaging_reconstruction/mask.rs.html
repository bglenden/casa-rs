3//! Generation-bound reconstruction masks.
4//!
5//! Reconstruction masks constrain where model updates may be placed. They are
6//! intentionally distinct from product-validity masks and consume immutable
7//! model/Normal-State lineage. Static user masks and auto-multithreshold masks
8//! mint the same closed generation type, so solvers need no mask-mode branch.

17/// Reproject decoded source support onto the target direction grid.
18///
19/// Storage/application code supplies only decoded samples and coordinates;
20/// reconstruction owns traversal, mapping, sampling, and uncovered support.
21pub fn reproject_mask_support(
22    source_coordinate: DirectionCoordinateSpec,
23    source_shape: [usize; 2],
24    source_support: &[bool],
25    target_coordinate: DirectionCoordinateSpec,
26    target_shape: [usize; 2],
27) -> Result<Box<[bool]>, MaskError> {
28    validate_shape(source_shape)?;
29    validate_shape(target_shape)?;
30    if source_support.len() != source_shape[0] * source_shape[1] {
31        return Err(MaskError::ShapeMismatch);
32    }
33    let mut target_support = vec![false; target_shape[0] * target_shape[1]];
34    for x in 0..target_shape[0] {
35        for y in 0..target_shape[1] {
36            let world = direction_pixel_to_world(target_coordinate, [x as f64, y as f64])?;
37            let source = direction_world_to_pixel(source_coordinate, world)?;
38            let sx = source[0].round();
39            let sy = source[1].round();
40            if sx >= 0.0
41                && sy >= 0.0
42                && sx < source_shape[0] as f64
43                && sy < source_shape[1] as f64
44                && source_support[sx as usize * source_shape[1] + sy as usize]
45            {
46                target_support[x * target_shape[1] + y] = true;
47            }
48        }
49    }
50    Ok(target_support.into_boxed_slice())
51}

53fn direction_pixel_to_world(
54    coordinate: DirectionCoordinateSpec,
55    pixel: [f64; 2],
56) -> Result<[f64; 2], MaskError> {
57    if coordinate.projection() != casa_imaging_model::Projection::Sin {
58        return Err(MaskError::UnsupportedReprojection);
59    }
60    let offset = [
61        pixel[0] - coordinate.reference_pixel()[0],
62        pixel[1] - coordinate.reference_pixel()[1],
63    ];
64    let pc = coordinate.pc();
65    let increment = coordinate.increment_rad();
66    let x = increment[0] * (pc[0][0] * offset[0] + pc[0][1] * offset[1]);
67    let y = increment[1] * (pc[1][0] * offset[0] + pc[1][1] * offset[1]);
68    let radius_squared = x * x + y * y;
69    if radius_squared > 1.0 + 1.0e-12 {
70        return Err(MaskError::UnsupportedReprojection);
71    }
72    let phi = if radius_squared == 0.0 {
73        0.0
74    } else {
75        x.atan2(-y)
76    };
77    let theta = (1.0 - radius_squared.min(1.0)).max(0.0).sqrt().asin();
78    let reference = coordinate.reference_direction();
79    let alpha_p = reference.longitude_rad();
80    let delta_p = reference.latitude_rad();
81    let phi_p = coordinate.pole_deg()[0].to_radians();
82    let dphi = phi - phi_p;
83    let sin_lat = theta.sin() * delta_p.sin() + theta.cos() * delta_p.cos() * dphi.cos();
84    let latitude = sin_lat.clamp(-1.0, 1.0).asin();
85    let longitude = alpha_p
86        + (-theta.cos() * dphi.sin())
87            .atan2(theta.sin() * delta_p.cos() - theta.cos() * delta_p.sin() * dphi.cos());
88    Ok([longitude, latitude])
89}

91fn direction_world_to_pixel(
92    coordinate: DirectionCoordinateSpec,
93    world: [f64; 2],
94) -> Result<[f64; 2], MaskError> {
95    if coordinate.projection() != casa_imaging_model::Projection::Sin {
96        return Err(MaskError::UnsupportedReprojection);
97    }
98    let reference = coordinate.reference_direction();
99    let alpha_p = reference.longitude_rad();
100    let delta_p = reference.latitude_rad();
101    let phi_p = coordinate.pole_deg()[0].to_radians();
102    let delta_alpha = world[0] - alpha_p;
103    let sin_theta =
104        world[1].sin() * delta_p.sin() + world[1].cos() * delta_p.cos() * delta_alpha.cos();
105    let theta = sin_theta.clamp(-1.0, 1.0).asin();
106    let phi = phi_p
107        + (-world[1].cos() * delta_alpha.sin()).atan2(
108            world[1].sin() * delta_p.cos() - world[1].cos() * delta_p.sin() * delta_alpha.cos(),
109        );
110    let x = theta.cos() * phi.sin();
111    let y = -theta.cos() * phi.cos();
112    let increment = coordinate.increment_rad();
113    let intermediate = [x / increment[0], y / increment[1]];
114    let pc = coordinate.pc();
115    let determinant = pc[0][0] * pc[1][1] - pc[0][1] * pc[1][0];
116    if !determinant.is_finite() || determinant.abs() < 1.0e-15 {
117        return Err(MaskError::UnsupportedReprojection);
118    }
119    let offset = [
120        (pc[1][1] * intermediate[0] - pc[0][1] * intermediate[1]) / determinant,
121        (-pc[1][0] * intermediate[0] + pc[0][0] * intermediate[1]) / determinant,
122    ];
123    let reference_pixel = coordinate.reference_pixel();
124    let pixel = [
125        reference_pixel[0] + offset[0],
126        reference_pixel[1] + offset[1],
127    ];
128    if pixel.iter().all(|value| value.is_finite()) {
129        Ok(pixel)
130    } else {
131        Err(MaskError::UnsupportedReprojection)
132    }
133}

167/// Deferred static mask construction evaluated at an exact major-cycle boundary.
168#[derive(Debug, Clone)]
169pub enum ReconstructionMaskPlan {
170    /// Materialize the independently committed continuum and line supports of
171    /// one joint reconstruction boundary.
172    Coupled {
173        /// Continuum-component spatial support.
174        continuum: Box<ReconstructionMaskPlan>,
175        /// Channel-local line-component spatial support.
176        line: Box<ReconstructionMaskPlan>,
177    },
178    /// Admit every model-support pixel.
179    FullPlane {
180        /// Target model direction coordinate.
181        coordinate: DirectionCoordinateSpec,
182    },
183    /// Admit the union of explicit target-grid boxes.
184    Boxes {
185        /// Target model direction coordinate.
186        coordinate: DirectionCoordinateSpec,
187        /// Inclusive target-grid regions.
188        boxes: Vec<MaskBox>,
189    },
190    /// Admit exact target-grid support reprojected from a named source WCS.
191    Reprojected {
192        /// Target model direction coordinate.
193        coordinate: DirectionCoordinateSpec,
194        /// Source mask direction coordinate retained as lineage evidence.
195        source_coordinate: DirectionCoordinateSpec,
196        /// Source mask direction-plane shape.
197        source_shape: [usize; 2],
198        /// Canonical target-grid support produced by exact WCS reprojection.
199        support: Box<[bool]>,
200    },
201    /// Generate CASA auto-multithreshold support from the current Normal State.
202    AutoMultithresh {
203        /// Target model direction coordinate.
204        coordinate: DirectionCoordinateSpec,
205        /// Thresholding, pruning, smoothing, and growth controls.
206        controls: AutoMultithreshControls,
207        /// Number of completed major cycles before this boundary.
208        completed_major_cycles: usize,
209        /// Whether the previous cycle reached its stopping threshold.
210        cycle_threshold_reached: bool,
211        /// Prior immutable mask generation, when another cycle already ran.
212        previous: Option<Box<ReconstructionMask>>,
213        /// Whether CASA's channel-stop rule froze further mask evolution.
214        evolution_stopped: bool,
215    },
216}

218impl ReconstructionMaskPlan {
219    /// Advance live automasking to the next major-cycle boundary.
220    #[must_use]
221    pub fn next_cycle(
222        &self,
223        current: &ReconstructionMask,
224        completed_major_cycles: usize,
225        cycle_threshold_reached: bool,
226        evolution_stopped: bool,
227    ) -> Self {
228        match self {
229            Self::Coupled { .. } => self.clone(),
230            Self::AutoMultithresh {
231                coordinate,
232                controls,
233                ..
234            } => Self::AutoMultithresh {
235                coordinate: *coordinate,
236                controls: *controls,
237                completed_major_cycles,
238                cycle_threshold_reached,
239                previous: Some(Box::new(current.clone())),
240                evolution_stopped,
241            },
242            other => other.clone(),
243        }
244    }
245
246    /// Materialize one immutable generation for the exact current model.
247    pub fn materialize(
248        &self,
249        base: &crate::ModelGeneration,
250        normal: &FinalNormalState,
251    ) -> Result<(ReconstructionMask, Option<AutoMultithreshEvidence>), MaskError> {
252        let problem = normal.problem_id();
253        let model_generation = base.generation_id();
254        let shape = normal.shape();
255        match self {
256            Self::Coupled { .. } => Err(MaskError::CoupledPlanRequired),
257            Self::FullPlane { coordinate } => Ok((
258                ReconstructionMask::full_plane(problem, model_generation, *coordinate, shape)?,
259                None,
260            )),
261            Self::Boxes { coordinate, boxes } => Ok((
262                ReconstructionMask::from_boxes(
263                    problem,
264                    model_generation,
265                    *coordinate,
266                    shape,
267                    boxes.iter().copied(),
268                )?,
269                None,
270            )),
271            Self::Reprojected {
272                coordinate,
273                source_coordinate,
274                source_shape,
275                support,
276            } => Ok((
277                ReconstructionMask::from_reprojected_support(
278                    problem,
279                    model_generation,
280                    *coordinate,
281                    shape,
282                    support,
283                    *source_coordinate,
284                    *source_shape,
285                )?,
286                None,
287            )),
288            Self::AutoMultithresh {
289                coordinate,
290                controls,
291                completed_major_cycles,
292                cycle_threshold_reached,
293                previous,
294                evolution_stopped,
295            } => {
296                let valid_support = base
297                    .samples()
298                    .iter()
299                    .map(|sample| sample.support() == casa_imaging_model::ModelSupport::Valid)
300                    .collect::<Vec<_>>();
301                let (mask, evidence) = auto_multithresh(
302                    problem,
303                    model_generation,
304                    *coordinate,
305                    normal,
306                    previous.as_deref(),
307                    &valid_support,
308                    *completed_major_cycles,
309                    *cycle_threshold_reached,
310                    *evolution_stopped,
311                    *controls,
312                )?;
313                Ok((mask, Some(evidence)))
314            }
315        }
316    }
317
318    /// Materialize the two immutable supports of a joint reconstruction.
319    pub fn materialize_coupled(
320        &self,
321        base: &crate::ModelGeneration,
322        normal: &FinalNormalState,
323    ) -> Result<
324        (
325            CoupledReconstructionMask,
326            [Option<AutoMultithreshEvidence>; 2],
327        ),
328        MaskError,
329    > {
330        let Self::Coupled { continuum, line } = self else {
331            return Err(MaskError::CoupledPlanRequired);
332        };
333        let (continuum, continuum_evidence) = continuum.materialize(base, normal)?;
334        let (line, line_evidence) = line.materialize(base, normal)?;
335        Ok((
336            CoupledReconstructionMask::new(continuum, line)?,
337            [continuum_evidence, line_evidence],
338        ))
339    }
340
341    /// Advance both live supports to the next joint major-cycle boundary.
342    #[must_use]
343    pub fn next_coupled_cycle(
344        &self,
345        current: &CoupledReconstructionMask,
346        completed_major_cycles: usize,
347        cycle_threshold_reached: bool,
348        evolution_stopped: [bool; 2],
349    ) -> Self {
350        let Self::Coupled { continuum, line } = self else {
351            return self.clone();
352        };
353        Self::Coupled {
354            continuum: Box::new(continuum.next_cycle(
355                current.continuum(),
356                completed_major_cycles,
357                cycle_threshold_reached,
358                evolution_stopped[0],
359            )),
360            line: Box::new(line.next_cycle(
361                current.line(),
362                completed_major_cycles,
363                cycle_threshold_reached,
364                evolution_stopped[1],
365            )),
366        }
367    }
368}

371struct AutoMaskBeam {
372    major_fwhm_pixels: f64,
373    minor_fwhm_pixels: f64,
374    position_angle_rad: f64,
375    area_pixels: f64,
376    sidelobe_fraction: f64,
377}
378
379fn fit_auto_mask_beam(normal: &FinalNormalState) -> Result<AutoMaskBeam, MaskError> {
380    let shape = normal.shape();
381    let psf = normal
382        .normal_approximation()
383        .iter()
384        .map(|sample| sample.re as f32)
385        .collect::<Vec<_>>();
386    let fitted = crate::fit_restoring_beam(&psf, shape, [1.0, 1.0], crate::DEFAULT_PSF_FIT_CUTOFF)
387        .map_err(|_| MaskError::InvalidBeamArea)?;
388    let major_fwhm_pixels = fitted.major_fwhm_rad();
389    let minor_fwhm_pixels = fitted.minor_fwhm_rad();
390    let area_pixels = std::f64::consts::PI * major_fwhm_pixels * minor_fwhm_pixels
391        / (4.0 * std::f64::consts::LN_2);
392    let sidelobe_fraction =
393        crate::psf_beam::fitted_psf_sidelobe_fraction_with_beam(&psf, shape, fitted)
394            .map_err(|_| MaskError::InvalidBeamArea)?;
395    if [
396        major_fwhm_pixels,
397        minor_fwhm_pixels,
398        area_pixels,
399        sidelobe_fraction,
400    ]
401    .into_iter()
402    .all(|value| value.is_finite() && value >= 0.0)
403        && major_fwhm_pixels > 0.0
404        && minor_fwhm_pixels > 0.0
405        && area_pixels > 0.0
406        && fitted.position_angle_rad().is_finite()
407    {
408        Ok(AutoMaskBeam {
409            major_fwhm_pixels,
410            minor_fwhm_pixels,
411            position_angle_rad: fitted.position_angle_rad(),
412            area_pixels,
413            sidelobe_fraction,
414        })
415    } else {
416        Err(MaskError::InvalidBeamArea)
417    }
418}

514        match self {
515            Self::Shared(_) | Self::Domains(_) => None,
516            Self::Coupled(masks) => Some(masks),

518    }

522    pub const fn domains(&self) -> Option<&ImageDomainReconstructionMasks> {
523        match self {
524            Self::Domains(masks) => Some(masks),
525            Self::Shared(_) | Self::Coupled(_) => None,
526        }
527    }

530impl ImageDomainReconstructionMaskPlans {
531    /// Adopt one non-empty plan for each image domain in compiled order.
532    pub fn new(plans: impl IntoIterator<Item = ReconstructionMaskPlan>) -> Result<Self, MaskError> {
533        let plans = plans.into_iter().collect::<Vec<_>>().into_boxed_slice();
534        if plans.is_empty() {
535            return Err(MaskError::DomainCardinalityMismatch);
536        }
537        Ok(Self { plans })
538    }
539
540    /// Return the number of canonical domain plans.
541    #[must_use]
542    pub const fn len(&self) -> usize {
543        self.plans.len()
544    }
545
546    /// Return whether no domain plans are present (always false for valid values).
547    #[must_use]
548    pub const fn is_empty(&self) -> bool {
549        self.plans.is_empty()
550    }
551
552    /// Borrow the primary-domain plan.
553    #[must_use]
554    pub const fn primary(&self) -> &ReconstructionMaskPlan {
555        &self.plans[0]
556    }
557
558    /// Advance every domain plan to the next shared major-cycle boundary.
559    ///
560    /// The T31 slice admits live auto-masking only for a one-domain
561    /// collection. Multi-domain auto-mask evolution fails typed instead of
562    /// silently sharing or independently inventing support.
563    pub fn next_cycle(
564        &self,
565        current: &ImageDomainReconstructionMasks,
566        completed_major_cycles: usize,
567        cycle_threshold_reached: bool,
568        evolution_stopped: &[bool],
569    ) -> Result<Self, MaskError> {
570        if self.plans.len() != current.len() || self.plans.len() != evolution_stopped.len() {
571            return Err(MaskError::DomainCardinalityMismatch);
572        }
573        if self.plans.len() > 1
574            && self
575                .plans
576                .iter()
577                .any(|plan| matches!(plan, ReconstructionMaskPlan::AutoMultithresh { .. }))
578        {
579            return Err(MaskError::UnsupportedMultiDomainPlan);
580        }
581        let plans = self
582            .plans
583            .iter()
584            .zip(current.iter())
585            .zip(evolution_stopped)
586            .map(|((plan, mask), stopped)| {
587                plan.next_cycle(
588                    mask,
589                    completed_major_cycles,
590                    cycle_threshold_reached,
591                    *stopped,
592                )
593            });
594        Self::new(plans)
595    }
596
597    /// Materialize every domain support against one shared Normal State.
598    pub fn materialize(
599        &self,
600        base: &crate::ModelGeneration,
601        normal: &FinalNormalState,
602    ) -> Result<ImageDomainMaskMaterialization, MaskError> {
603        if self.plans.len() != normal.domain_count()
604            || self.plans.len() != base.shape().domains().len()
605        {
606            return Err(MaskError::DomainCardinalityMismatch);
607        }
608        if self.plans.len() == 1 {
609            let (mask, evidence) = self.plans[0].materialize(base, normal)?;
610            return Ok(ImageDomainMaskMaterialization::new(
611                ImageDomainReconstructionMasks::new([mask])?,
612                vec![evidence].into_boxed_slice(),
613            ));
614        }
615        let mut masks = Vec::with_capacity(self.plans.len());
616        for (plan, domain) in self.plans.iter().zip(normal.domains()) {
617            let problem = normal.problem_id();
618            let model_generation = base.generation_id();
619            let shape = domain.shape();
620            let mask = match plan {
621                ReconstructionMaskPlan::FullPlane { coordinate } => {
622                    ReconstructionMask::full_plane(problem, model_generation, *coordinate, shape)?
623                }
624                ReconstructionMaskPlan::Boxes { coordinate, boxes } => {
625                    ReconstructionMask::from_boxes(
626                        problem,
627                        model_generation,
628                        *coordinate,
629                        shape,
630                        boxes.iter().copied(),
631                    )?
632                }
633                ReconstructionMaskPlan::Reprojected {
634                    coordinate,
635                    source_coordinate,
636                    source_shape,
637                    support,
638                } => ReconstructionMask::from_reprojected_support(
639                    problem,
640                    model_generation,
641                    *coordinate,
642                    shape,
643                    support,
644                    *source_coordinate,
645                    *source_shape,
646                )?,
647                ReconstructionMaskPlan::Coupled { .. }
648                | ReconstructionMaskPlan::AutoMultithresh { .. } => {
649                    return Err(MaskError::UnsupportedMultiDomainPlan);
650                }
651            };
652            masks.push(mask);
653        }
654        let evidence = vec![None; masks.len()].into_boxed_slice();
655        Ok(ImageDomainMaskMaterialization::new(
656            ImageDomainReconstructionMasks::new(masks)?,
657            evidence,
658        ))
659    }
660}

826            return Err(MaskError::EmptyMask);
827        }
828        Self::mint(
829            problem,
830            model_generation,
831            None,
832            coordinate,
833            shape,
834            support,
835            0,
836            &[],

1002/// Generate a new auto-multithreshold mask from immutable Normal State.
1003#[allow(clippy::too_many_arguments)]
1004pub fn auto_multithresh(
1005    problem: CompiledProblemId,
1006    model_generation: ModelGenerationId,
1007    coordinate: DirectionCoordinateSpec,
1008    normal: &FinalNormalState,
1009    previous: Option<&ReconstructionMask>,
1010    valid_support: &[bool],
1011    completed_major_cycles: usize,
1012    cycle_threshold_reached: bool,
1013    evolution_stopped: bool,
1014    controls: AutoMultithreshControls,
1015) -> Result<(ReconstructionMask, AutoMultithreshEvidence), MaskError> {
1016    let beam = fit_auto_mask_beam(normal)?;
1017    validate_auto_controls(controls, beam.area_pixels)?;
1018    let shape = normal.shape();
1019    if valid_support.len() != shape[0] * shape[1]
1020        || previous.is_some_and(|mask| {
1021            mask.problem != problem || mask.shape != shape || mask.coordinate != coordinate
1022        })
1023    {
1024        return Err(MaskError::ShapeMismatch);
1025    }
1026    // `FinalNormalState` retains the unnormalised normal-equation planes. CASA
1027    // derives automask thresholds from the displayed residual in model units,
1028    // so remove the accumulated normalisation exactly as the minor-cycle
1029    // solver does before applying Jy-valued controls.
1030    let psf_peak = normal
1031        .normal_approximation()
1032        .iter()
1033        .map(|value| value.re)
1034        .fold(f64::NEG_INFINITY, f64::max);
1035    if !psf_peak.is_finite() || psf_peak <= 0.0 {
1036        return Err(MaskError::InvalidBeamArea);
1037    }
1038    let residual = normal
1039        .residual()
1040        .iter()
1041        .map(|value| value.re / psf_peak)
1042        .collect::<Vec<_>>();
1043    if residual.iter().any(|value| !value.is_finite()) {
1044        return Err(MaskError::NonfiniteResidual);
1045    }
1046    let residual_median = median(&residual);
1047    let deviations = residual
1048        .iter()
1049        .map(|value| (value - residual_median).abs())
1050        .collect::<Vec<_>>();
1051    let robust_rms = 1.482_602_218_505_602 * median(&deviations);
1052    let absolute_peak = residual
1053        .iter()
1054        .map(|value| (value - residual_median).abs())
1055        .fold(0.0_f64, f64::max);
1056    let sidelobe_level = beam.sidelobe_fraction;
1057    let positive_offset = (controls.sidelobe_factor * sidelobe_level * absolute_peak)
1058        .max(controls.noise_factor * robust_rms);
1059    let low_offset = (controls.sidelobe_factor * sidelobe_level * absolute_peak)
1060        .max(controls.low_noise_factor * robust_rms);
1061    let positive_threshold = residual_median + positive_offset;
1062    let low_noise_threshold = residual_median + low_offset;
1063    let negative_threshold = (controls.negative_factor > 0.0).then(|| {
1064        residual_median
1065            - (controls.sidelobe_factor * sidelobe_level * absolute_peak)
1066                .max(controls.negative_factor * robust_rms)
1067    });
1068    let mut detected = residual
1069        .iter()
1070        .zip(valid_support)
1071        .map(|(value, valid)| {
1072            *valid
1073                && (*value >= positive_threshold
1074                    || negative_threshold.is_some_and(|threshold| *value <= threshold))
1075        })
1076        .collect::<Vec<_>>();
1077    if evolution_stopped {
1078        detected.fill(false);
1079    }
1080    if !evolution_stopped {
1081        prune_regions(
1082            &mut detected,
1083            shape,
1084            (controls.minimum_beam_fraction * beam.area_pixels).ceil() as usize,
1085        );
1086        detected = smooth_and_cut(
1087            &detected,
1088            shape,
1089            controls.smooth_factor,
1090            beam,
1091            controls.cut_threshold,
1092        );
1093        if completed_major_cycles > 0 {
1094            grow_cross(
1095                &mut detected,
1096                shape,
1097                &residual,
1098                low_noise_threshold,
1099                controls.grow_iterations,
1100            );
1101        }
1102    }
1103    let mut support =
1104        previous.map_or_else(|| vec![false; detected.len()], |mask| mask.support.to_vec());
1105    let before = support.clone();
1106    for index in 0..support.len() {
1107        support[index] = valid_support[index] && (support[index] || detected[index]);
1108    }
1109    let changed_pixels = support
1110        .iter()
1111        .zip(&before)
1112        .filter(|(left, right)| left != right)
1113        .count();
1114    let previous_pixels = before.iter().filter(|value| **value).count();
1115    let percent_change = if previous_pixels == 0 {
1116        if changed_pixels == 0 { 0.0 } else { 100.0 }
1117    } else {
1118        100.0 * changed_pixels as f64 / previous_pixels as f64
1119    };
1120    let channel_stopped = evolution_stopped
1121        || support.iter().all(|value| !*value)
1122        || (controls.minimum_percent_change >= 0.0
1123            && cycle_threshold_reached
1124            && percent_change <= controls.minimum_percent_change);
1125    let mut source = Encoder::new(b"casa-rs-auto-multithresh-evidence", 1);
1126    source.identity(normal.completion_id().as_bytes());
1127    source.u64(crate::canonical_f64_bits(positive_threshold));
1128    source.u64(crate::canonical_f64_bits(low_noise_threshold));
1129    source.usize(completed_major_cycles);
1130    source.u8(u8::from(cycle_threshold_reached));
1131    source.u8(u8::from(evolution_stopped));
1132    match previous {
1133        Some(previous) => {
1134            source.u8(1);
1135            source.identity(previous.generation_id().as_bytes());
1136        }
1137        None => source.u8(0),
1138    }
1139    for value in [
1140        controls.sidelobe_factor,
1141        controls.noise_factor,
1142        controls.low_noise_factor,
1143        controls.negative_factor,
1144        controls.minimum_beam_fraction,
1145        controls.smooth_factor,
1146        controls.cut_threshold,
1147        controls.minimum_percent_change,
1148        beam.major_fwhm_pixels,
1149        beam.minor_fwhm_pixels,
1150        beam.position_angle_rad,
1151        beam.area_pixels,
1152        beam.sidelobe_fraction,
1153    ] {
1154        source.u64(crate::canonical_f64_bits(value));
1155    }
1156    source.usize(controls.grow_iterations);
1157    let mask = ReconstructionMask::mint(
1158        problem,
1159        model_generation,
1160        Some(normal.completion_id()),
1161        coordinate,
1162        shape,
1163        support,
1164        2,
1165        &source.finish(),
1166    )?;
1167    Ok((
1168        mask,
1169        AutoMultithreshEvidence {
1170            previous_mask_generation: previous.map(ReconstructionMask::generation_id),
1171            median: residual_median,
1172            robust_rms,
1173            positive_threshold,
1174            low_noise_threshold,
1175            negative_threshold,
1176            changed_pixels,
1177            channel_stopped,
1178        },
1179    ))
1180}

1182/// Mask construction or lineage failure.
1183#[derive(Debug, Clone, PartialEq, Eq, Error)]
1184pub enum MaskError {
1185    /// Mask-plan cardinality does not equal the compiled image-domain count.
1186    #[error("reconstruction mask plans must name every compiled image domain exactly once")]
1187    DomainCardinalityMismatch,
1188    /// This multi-domain slice supports only static full-plane, box, or reprojected masks.
1189    #[error("multi-domain reconstruction does not support this mask plan")]
1190    UnsupportedMultiDomainPlan,
1191    /// A joint solve requires a plan carrying both component supports.
1192    #[error("joint reconstruction requires a coupled mask plan")]
1193    CoupledPlanRequired,
1194    /// Source and target direction grids cannot be mapped by the accepted law.
1195    #[error("mask direction grids require a supported exact reprojection")]
1196    UnsupportedReprojection,
1197    /// The current PSF cannot define a finite positive beam area.
1198    #[error("normal-state PSF cannot define a finite positive beam area")]
1199    InvalidBeamArea,
1200    /// A pixel box is inverted.
1201    #[error("mask box requires blc <= trc")]
1202    InvertedBox,
1203    /// A static box lies outside its named grid.
1204    #[error("mask box lies outside the target grid")]
1205    OutsideTarget,
1206    /// Shape or lineage does not match the target grid.
1207    #[error("mask shape or lineage mismatch")]
1208    ShapeMismatch,
1209    /// Shape is empty or overflows addressable storage.
1210    #[error("mask shape must be finite and non-empty")]
1211    InvalidShape,
1212    /// Empty reconstruction masks fail visibly; no full-image fallback exists.
1213    #[error("reconstruction mask is empty")]
1214    EmptyMask,
1215    /// Auto-mask residual state contained a non-finite sample.
1216    #[error("auto-mask residual contains a non-finite sample")]
1217    NonfiniteResidual,
1218    /// Auto-mask controls or beam area are invalid.
1219    #[error("auto-multithreshold controls are invalid")]
1220    InvalidAutoControls,
1221}

1223fn validate_shape(shape: [usize; 2]) -> Result<(), MaskError> {
1224    if shape[0] == 0 || shape[1] == 0 || shape[0].checked_mul(shape[1]).is_none() {
1225        Err(MaskError::InvalidShape)
1226    } else {
1227        Ok(())
1228    }

1231fn validate_auto_controls(
1232    controls: AutoMultithreshControls,
1233    beam_area_pixels: f64,
1234) -> Result<(), MaskError> {
1235    let values = [
1236        controls.sidelobe_factor,
1237        controls.noise_factor,
1238        controls.low_noise_factor,
1239        controls.negative_factor,
1240        controls.minimum_beam_fraction,
1241        controls.smooth_factor,
1242        controls.cut_threshold,
1243        controls.minimum_percent_change,
1244        beam_area_pixels,
1245    ];
1246    if values.iter().enumerate().any(|(index, value)| {
1247        !value.is_finite() || (*value < 0.0 && index != 7) || (index == 7 && *value < -1.0)
1248    }) || controls.cut_threshold > 1.0
1249        || beam_area_pixels == 0.0
1250    {
1251        Err(MaskError::InvalidAutoControls)
1252    } else {
1253        Ok(())
1254    }
1255}

1257fn median(values: &[f64]) -> f64 {
1258    let mut sorted = values.to_vec();
1259    sorted.sort_by(f64::total_cmp);
1260    let middle = sorted.len() / 2;
1261    if sorted.len() % 2 == 0 {
1262        (sorted[middle - 1] + sorted[middle]) * 0.5
1263    } else {
1264        sorted[middle]
1265    }
1266}

1268fn prune_regions(mask: &mut [bool], shape: [usize; 2], minimum: usize) {
1269    if minimum <= 1 {
1270        return;
1271    }
1272    let mut seen = vec![false; mask.len()];
1273    for seed in 0..mask.len() {
1274        if !mask[seed] || seen[seed] {
1275            continue;
1276        }
1277        let mut queue = VecDeque::from([seed]);
1278        let mut component = Vec::new();
1279        seen[seed] = true;
1280        while let Some(index) = queue.pop_front() {
1281            component.push(index);
1282            let pixel = [index / shape[1], index % shape[1]];
1283            for offset in [[-1, 0], [1, 0], [0, -1], [0, 1]] {
1284                if let Some(neighbor) = offset_mask_pixel(pixel, offset, shape) {
1285                    let flat = neighbor[0] * shape[1] + neighbor[1];
1286                    if mask[flat] && !seen[flat] {
1287                        seen[flat] = true;
1288                        queue.push_back(flat);
1289                    }
1290                }
1291            }
1292        }
1293        if component.len() < minimum {
1294            for index in component {
1295                mask[index] = false;
1296            }
1297        }
1298    }
1299}

1301fn smooth_and_cut(
1302    mask: &[bool],
1303    shape: [usize; 2],
1304    smooth_factor: f64,
1305    beam: AutoMaskBeam,
1306    cut: f64,
1307) -> Vec<bool> {
1308    if smooth_factor == 0.0 || mask.iter().all(|value| !*value) {
1309        return mask.to_vec();
1310    }
1311    let sigma_major = smooth_factor * beam.major_fwhm_pixels / 2.354_820_045_030_949_3;
1312    let sigma_minor = smooth_factor * beam.minor_fwhm_pixels / 2.354_820_045_030_949_3;
1313    // Image2DConvolver::_shapeOfKernel uses a square +/-5-sigma Gaussian
1314    // kernel (`_sizeOfGaussian(width, 5.0)`) and then forces an odd shape.
1315    let radius = (5.0 * sigma_major + 0.5) as isize + 1;
1316    let cos_pa = beam.position_angle_rad.cos();
1317    let sin_pa = beam.position_angle_rad.sin();
1318    let mut smoothed = vec![0.0; mask.len()];
1319    for x in 0..shape[0] {
1320        for y in 0..shape[1] {
1321            let mut value = 0.0;
1322            for dx in -radius..=radius {
1323                for dy in -radius..=radius {
1324                    if let Some(pixel) = offset_mask_pixel([x, y], [dx, dy], shape)
1325                        && mask[pixel[0] * shape[1] + pixel[1]]
1326                    {
1327                        let dx = dx as f64;
1328                        let dy = dy as f64;
1329                        let u = dx * cos_pa + dy * sin_pa;
1330                        let v = -dx * sin_pa + dy * cos_pa;
1331                        value +=
1332                            (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2))).exp();
1333                    }
1334                }
1335            }
1336            smoothed[x * shape[1] + y] = value;
1337        }
1338    }
1339    let peak = smoothed.iter().copied().fold(0.0_f64, f64::max);
1340    smoothed
1341        .into_iter()
1342        .map(|value| value >= cut * peak)
1343        .collect()
1344}

1346fn grow_cross(
1347    mask: &mut [bool],
1348    shape: [usize; 2],
1349    residual: &[f64],
1350    low_threshold: f64,
1351    iterations: usize,
1352) {
1353    for _ in 0..iterations {
1354        let previous = mask.to_owned();
1355        for index in 0..previous.len() {
1356            if previous[index] {
1357                continue;
1358            }
1359            let pixel = [index / shape[1], index % shape[1]];
1360            if residual[index] < low_threshold {
1361                continue;
1362            }
1363            mask[index] = [[-1, 0], [1, 0], [0, -1], [0, 1]]
1364                .into_iter()
1365                .any(|offset| {
1366                    offset_mask_pixel(pixel, offset, shape)
1367                        .is_some_and(|neighbor| previous[neighbor[0] * shape[1] + neighbor[1]])
1368                });
1369        }
1370    }
1371}

1373fn offset_mask_pixel(
1374    pixel: [usize; 2],
1375    offset: [isize; 2],
1376    shape: [usize; 2],
1377) -> Option<[usize; 2]> {
1378    let x = pixel[0].checked_add_signed(offset[0])?;
1379    let y = pixel[1].checked_add_signed(offset[1])?;
1380    (x < shape[0] && y < shape[1]).then_some([x, y])
1381}

1383#[allow(clippy::too_many_arguments)]
1384fn mask_identity(
1385    problem: CompiledProblemId,
1386    model_generation: ModelGenerationId,
1387    normal: Option<FinalNormalStateCompletionId>,
1388    coordinate: DirectionCoordinateSpec,
1389    shape: [usize; 2],
1390    support: &[bool],
1391    source_kind: u8,
1392    source_identity: &[u8],
1393) -> ReconstructionMaskGenerationId {
1394    let mut encoder = Encoder::new(MASK_DOMAIN, MASK_VERSION);
1395    encoder.identity(problem.as_bytes());
1396    encoder.identity(model_generation.as_bytes());
1397    match normal {
1398        None => encoder.u8(0),
1399        Some(normal) => {
1400            encoder.u8(1);
1401            encoder.identity(normal.as_bytes());
1402        }
1403    }
1404    encode_coordinate(&mut encoder, coordinate);
1405    encoder.usize(shape[0]);
1406    encoder.usize(shape[1]);
1407    encoder.u8(source_kind);
1408    encoder.bytes(source_identity);
1409    for value in support {
1410        encoder.u8(u8::from(*value));
1411    }
1412    ReconstructionMaskGenerationId(LogicalIdentity::from_sha256(encoder.finish()))
1413}
1414
1415fn encode_coordinate(encoder: &mut Encoder, coordinate: DirectionCoordinateSpec) {
1416    encoder.u8(coordinate.projection() as u8);
1417    encoder.u8(coordinate.reference_direction().frame() as u8);
1418    encoder.u64(crate::canonical_f64_bits(
1419        coordinate.reference_direction().longitude_rad(),
1420    ));
1421    encoder.u64(crate::canonical_f64_bits(
1422        coordinate.reference_direction().latitude_rad(),
1423    ));
1424    for value in coordinate.reference_pixel() {
1425        encoder.u64(crate::canonical_f64_bits(value));
1426    }
1427    for value in coordinate.increment_rad() {
1428        encoder.u64(crate::canonical_f64_bits(value));
1429    }
1430    for value in coordinate.pc().into_iter().flatten() {
1431        encoder.u64(crate::canonical_f64_bits(value));
1432    }
1433}
