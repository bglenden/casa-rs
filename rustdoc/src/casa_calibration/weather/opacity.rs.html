2//! VLA opacity interpolation for CASA's fixed 0–50 GHz reference model.
3
4const CHANNELS: usize = 2000;
5const START_HZ: f64 = 12_500_000.0;
6const STEP_HZ: f64 = 25_000_000.0;
7const COEFFICIENT_BYTES: usize = 2 * size_of::<f64>();
8
9// The model profiles and spectral grid are identical on every CASA call.
10// Regenerate the fixed coefficients with tools/casa-compat/generate_plotweather_opacity.py.
11const COEFFICIENTS: &[u8; CHANNELS * COEFFICIENT_BYTES] =
12    include_bytes!("vla_opacity_6_7_6_14.bin");
13
14pub(super) fn windows(windows_ghz: &[f64], water_mm: f64) -> Result<Vec<f64>, String> {
15    windows_ghz
16        .iter()
17        .map(|&frequency| interpolate(frequency, water_mm))
18        .collect()
19}
20
21pub(super) fn frequencies() -> Vec<f64> {
22    (0..CHANNELS).map(frequency_ghz).collect()
23}
24
25pub(super) fn spectrum(water_mm: f64) -> Vec<f64> {
26    (0..CHANNELS)
27        .map(|index| opacity(index, water_mm))
28        .collect()
29}
30
31fn frequency_ghz(index: usize) -> f64 {
32    (START_HZ + STEP_HZ * index as f64) / 1e9
33}
34
35fn interpolate(frequency: f64, water_mm: f64) -> Result<f64, String> {
36    if !frequency.is_finite() {
37        return Err("spectral-window frequency must be finite".into());
38    }
39    let sample = ((frequency * 1e9 - START_HZ) / STEP_HZ).clamp(0.0, (CHANNELS - 1) as f64);
40    let lower = sample.floor() as usize;
41    let upper = (lower + 1).min(CHANNELS - 1);
42    // CASA's four-point slice starts two channels before the nearest
43    // channel, with ties resolved in favour of the first index.
44    let nearest = if frequency - frequency_ghz(lower) > frequency_ghz(upper) - frequency {
45        upper
46    } else {
47        lower
48    };
49    if nearest < 2 {
50        return Err("array of sample points is empty".into());
51    }
52    let start = opacity(lower, water_mm);
53    if lower == upper || frequency == frequency_ghz(lower) {
54        return Ok(start);
55    }
56    let spacing = frequency_ghz(upper) - frequency_ghz(lower);
57    Ok(start + (frequency - frequency_ghz(lower)) * (opacity(upper, water_mm) - start) / spacing)
58}
59
60fn opacity(index: usize, water_mm: f64) -> f64 {
61    let offset = index * COEFFICIENT_BYTES;
62    let row = &COEFFICIENTS[offset..offset + COEFFICIENT_BYTES];
63    let intercept = f64::from_le_bytes(row[..8].try_into().expect("fixed opacity intercept"));
64    let slope = f64::from_le_bytes(row[8..].try_into().expect("fixed opacity slope"));
65    intercept + slope * water_mm
66}
