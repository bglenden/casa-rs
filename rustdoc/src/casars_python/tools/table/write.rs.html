2//! The `table` tool's write side for keywords and the table info, as CASA
3//! 6.7.6.14's `tb` does it (`tools/casa-compat/table_tool_oracle.py`).
4//!
5//! Every write is made through to the table on disk and touches only the
6//! table's metadata: `Table::save_metadata_only` rewrites `table.dat`'s
7//! keywords and `table.info` and leaves the storage managers as they are
8//! (`Table::flush` or a plain unlock would rewrite the table as
9//! StManAipsIO). A table opened with an automatic lock (`default`, `auto`,
10//! `autonoread`) takes the write lock for the write and releases it after;
11//! a table opened with `user` locking must hold the write lock, and one
12//! opened with `permanent` locking holds it. A table opened read-only
13//! refuses. With no table open a write logs and returns `False`.
14

23type Reply = Result<CasaValue, Fail>;

25/// What a write changes; it names the refusal of a table not open for
26/// update.
27#[derive(Clone, Copy)]
28pub(super) enum Target {
29    TableKeywords,
30    ColumnKeywords,
31    Info,
32    /// Cells: a table not open for update answers `False`, not an error.
33    Cells,
34    /// The table's rows or columns, as the casacore operation named here
35    /// refuses a table not open for update.
36    Structure(&'static str),
37    /// A `tablerow` handle writing to a read-only table raises a `RuntimeError`.
38    RowProxy,
39}
40
41/// What a change leaves to write to the disk.
42pub(super) enum Persist {
43    /// The table's keywords and info.
44    Metadata,
45    /// The cells of `rows` in `columns`, in place in each column's storage
46    /// manager.
47    Rows {
48        columns: Vec<String>,
49        rows: Vec<usize>,
50    },
51    /// Nothing: the call selected no cell.
52    Nothing,
53    /// The change has written itself to the disk (a structural change in
54    /// place, or a table written in full).
55    Done,
56}

62    pub(crate) sub_type: Option<String>,

66fn runtime<T>(message: String) -> Result<T, Fail> {
67    Err(Fail::Runtime(message))
68}

70/// The record a dotted `path` names inside `record`, made if `make` is set.
71fn descend<'a>(
72    record: &'a mut RecordValue,
73    path: &str,
74    make: bool,
75) -> Result<&'a mut RecordValue, Fail> {
76    let mut current = record;
77    for part in path.split('.') {
78        if current.get(part).is_none() {
79            if !make {
80                return runtime(format!("RecordInterface: field {part} is unknown"));
81            }
82            current.push(RecordField::new(
83                part,
84                Value::Record(RecordValue::new(Vec::new())),
85            ));
86        }
87        match current.get_mut(part) {
88            Some(Value::Record(inner)) => current = inner,
89            _ => return runtime(format!("RecordInterface: field {part} is unknown")),
90        }
91    }
92    Ok(current)
93}

95/// Defines `name` in `record`: a new field takes the value's natural type,
96/// an existing one keeps its own.
97fn store(record: &mut RecordValue, name: &str, value: Incoming) -> Result<(), Fail> {
98    match record.get_mut(name) {
99        Some(old) => {
100            let new = value.like(old, name)?;
101            *old = new;
102        }
103        None => record.push(RecordField::new(name, value.natural()?)),
104    }
105    Ok(())
106}

108/// `putkeyword`'s rule: by (dotted) name or by 1-based position.
109pub(super) fn put(
110    record: &mut RecordValue,
111    key: &KeywordRef,
112    value: Incoming,
113    make_subrecord: bool,
114) -> Result<(), Fail> {
115    match key {
116        KeywordRef::Whole => Ok(()),
117        KeywordRef::Name(name) => match name.rsplit_once('.') {
118            Some((parent, last)) => store(descend(record, parent, make_subrecord)?, last, value),
119            None => store(record, name, value),
120        },
121        KeywordRef::Index(index) => {
122            let slot = usize::try_from(*index)
123                .ok()
124                .filter(|slot| (1..=record.len()).contains(slot));
125            let Some(slot) = slot else {
126                return runtime("RecordInterface::define - new fieldNumber exceeds #fields".into());
127            };
128            let field = &mut record.fields_mut()[slot - 1];
129            let new = value.like(&field.value, &field.name)?;
130            field.value = new;
131            Ok(())
132        }
133    }
134}

136/// `removekeyword`'s rule; `missing` words the refusal of an absent name.
137fn remove(
138    record: &mut RecordValue,
139    key: &KeywordRef,
140    missing: impl Fn(&str) -> String,
141) -> Result<(), Fail> {
142    match key {
143        KeywordRef::Whole => Ok(()),
144        KeywordRef::Name(name) => {
145            let (parent, last) = name.rsplit_once('.').unwrap_or(("", name));
146            let target = if parent.is_empty() {
147                record
148            } else {
149                descend(record, parent, false).map_err(|_| Fail::Runtime(missing(name)))?
150            };
151            match target.remove(last) {
152                Some(_) => Ok(()),
153                None => runtime(missing(name)),
154            }
155        }
156        KeywordRef::Index(index) => {
157            let name = usize::try_from(*index)
158                .ok()
159                .and_then(|slot| slot.checked_sub(1))
160                .and_then(|slot| record.fields().get(slot))
161                .map(|field| field.name.clone());
162            match name {
163                Some(name) => {
164                    record.remove(&name);
165                    Ok(())
166                }
167                None => runtime("RecordRep::deleteDataField".into()),
168            }
169        }
170    }
171}

173/// The column `column`'s keywords, edited by `edit`.
174fn edit_column(
175    table: &mut Table,
176    column: &str,
177    edit: impl FnOnce(&mut RecordValue) -> Result<(), Fail>,
178) -> Result<(), Fail> {
179    if table
180        .schema()
181        .and_then(|schema| schema.column(column))
182        .is_none()
183    {
184        return runtime(format!("Table column {column} is unknown"));
185    }
186    let mut keywords = table
187        .column_keywords(column)
188        .cloned()
189        .unwrap_or_else(|| RecordValue::new(Vec::new()));
190    edit(&mut keywords)?;
191    table.set_column_keywords(column, keywords);
192    Ok(())
193}

195impl State {
196    /// Applies `change` to the open table and saves its metadata, under the
197    /// lock the table's lock options call for.
198    fn write(
199        &mut self,
200        target: Target,
201        log: &mut Vec<crate::tools::LogLine>,
202        change: impl FnOnce(&mut Table) -> Result<(), Fail>,
203    ) -> Reply {
204        self.write_with(target, log, move |table, _| {
205            change(table).map(|()| Persist::Metadata)
206        })
207    }
208
209    /// As [`write`](Self::write), for a change that says what to persist.
210    pub(super) fn write_with(
211        &mut self,
212        target: Target,
213        log: &mut Vec<crate::tools::LogLine>,
214        change: impl FnOnce(&mut Table, Option<&[usize]>) -> Result<Persist, Fail>,
215    ) -> Reply {
216        let Some(open) = self.open.as_ref() else {
217            no_table(log);
218            return Ok(CasaValue::Bool(false));
219        };
220        let name = open.name.display().to_string();
221        let row_map = open.row_map.clone();
222        let mut root = open.root();
223        write_root(&mut root, &name, target, row_map.as_deref(), change)
224    }
225}

227/// Applies `change` directly to `root` and persists what it modified under
228/// the table's lock policy. Shared by `State::write_with` and `PyTableRowTool::put`.
229pub(super) fn write_root(
230    root: &mut RootOpen,
231    name: &str,
232    target: Target,
233    row_map: Option<&[usize]>,
234    change: impl FnOnce(&mut Table, Option<&[usize]>) -> Result<Persist, Fail>,
235) -> Reply {
236    if !root.writable {
237        if matches!(target, Target::Cells) {
238            return Ok(CasaValue::Bool(false));
239        }
240        return runtime(match target {
241            Target::Cells => String::new(),
242            Target::TableKeywords => {
243                format!("Table::rwKeywordSet cannot be used: table {name} is not writable")
244            }
245            Target::ColumnKeywords => {
246                format!("TableColumn::rwKeywordSet cannot be used: table {name} is not writable")
247            }
248            Target::Info => format!("Table {name} is not writable"),
249            Target::Structure(operation) => {
250                format!("Invalid Table operation: {operation}; table {name} is not writable")
251            }
252            Target::RowProxy => "TableRowProxy: the given TableRow is not writable".into(),
253        });
254    }
255    super::state::publish_component_keywords(&root.name)?;
256    let named_memory = root.table.is_named_memory_manager();
257    if root.table.is_memory() && !named_memory {
258        change(&mut root.table, row_map)?;
259        return Ok(CasaValue::Bool(true));
260    }
261    let is_reference = root.table.is_reference_table()?;
262    if is_reference && matches!(target, Target::Structure(_)) {
263        return Err(Fail::Unsupported(
264            "structural edits of an opened reference table require a reference descriptor write"
265                .to_string(),
266        ));
267    }
268    let automatic = root.lock.automatic();
269    let attempts = root.lock.attempts();
270    if automatic {
271        if !root.table.lock(LockType::Write, attempts)? {
272            return runtime(format!("Table {name} could not be write locked"));
273        }
274    } else if !root.table.has_lock(LockType::Write) {
275        return runtime(format!(
276            "ColumnSet::doLock: table {name} should be locked when using UserLocking"
277        ));
278    }
279    let saved = change(&mut root.table, row_map).and_then(|persist| {
280        if named_memory {
281            let metadata = !matches!(target, Target::Cells | Target::RowProxy)
282                && !matches!(persist, Persist::Nothing);
283            return root
284                .table
285                .finish_memory_manager_write(metadata)
286                .map_err(Fail::from);
287        }
288        match persist {
289            Persist::Nothing | Persist::Done => Ok(()),
290            Persist::Metadata => root
291                .table
292                .save_metadata_only(TableOptions::new(&root.name))
293                .map_err(Fail::from),
294            Persist::Rows {
295                mut columns,
296                mut rows,
297            } => {
298                columns.sort_unstable();
299                columns.dedup();
300                if is_reference {
301                    // Different persisted view rows can refer to the same
302                    // root cell. Keep the last occurrence of each view row,
303                    // in mutation order, so the last write to the root wins.
304                    let mut seen = std::collections::HashSet::new();
305                    rows.reverse();
306                    rows.retain(|row| seen.insert(*row));
307                    rows.reverse();
308                } else {
309                    rows.sort_unstable();
310                    rows.dedup();
311                }
312                let col_refs: Vec<&str> = columns.iter().map(String::as_str).collect();
313                let partial = root
314                    .table
315                    .prepare_write()
316                    .save_selected_rows(&col_refs, &rows);
317                match partial {
318                    // StManAipsIO keeps its table in one file and writes it
319                    // in full, as casacore does. A table that is one such
320                    // group is written in full with the group's name and
321                    // byte order kept, which is the layout it has. Any
322                    // other table must not be rewritten (it would lose its
323                    // managers), so the refusal stands.
324                    Err(TableError::Storage(message))
325                        if message.starts_with("partial in-place save does not support") =>
326                    {
327                        match root.table.single_rewrite_options()? {
328                            Some(options) => root.table.save(options).map_err(Fail::from),
329                            None => Err(Fail::from(TableError::Storage(message))),
330                        }
331                    }
332                    other => other.map_err(Fail::from),
333                }
334            }
335        }
336    });
337    // Successful saves clear only the edits they persisted. After a failure,
338    // discard the unpersisted edits while still publishing anything that
339    // reached disk; do not let closing the handle retry a failed mutation.
340    let reloaded = if named_memory && saved.is_err() {
341        root.table
342            .finish_memory_manager_write(false)
343            .map_err(Fail::from)
344    } else if saved.is_err() || is_reference {
345        // Repeated reference rows can alias the very same cell.
346        root.table.resync_or_discard().map_err(Fail::from)
347    } else {
348        Ok(())
349    };
350    let released = if automatic {
351        root.table
352            .unlock_metadata_only_or_discard()
353            .map_err(Fail::from)
354    } else {
355        Ok(())
356    };
357    saved?;
358    reloaded?;
359    released?;
360    if matches!(
361        target,
362        Target::TableKeywords | Target::ColumnKeywords | Target::Info
363    ) {
364        // casacore shares the keyword object across its in-process handles.
365        // Native component caches reload only after a completed metadata write.
366        root.seen_keyword_epoch = root
367            .keyword_epoch
368            .fetch_add(1, std::sync::atomic::Ordering::Release)
369            .wrapping_add(1);
370    }
371    Ok(CasaValue::Bool(true))
372}

374impl State {
375    /// `putkeyword(keyword, value, makesubrecord)`.
376    pub(super) fn putkeyword(
377        &mut self,
378        key: KeywordRef,
379        value: Incoming,
380        make_subrecord: bool,
381        log: &mut Vec<crate::tools::LogLine>,
382    ) -> Reply {
383        if matches!(&key, KeywordRef::Whole) {
384            return Ok(CasaValue::Bool(false));
385        }
386        self.write(Target::TableKeywords, log, move |table| {
387            put(table.keywords_mut(), &key, value, make_subrecord)
388        })
389    }
390
391    /// `putkeywords(value)`: each field is defined as `putkeyword` does.
392    pub(super) fn putkeywords(
393        &mut self,
394        fields: Vec<(String, Incoming)>,
395        log: &mut Vec<crate::tools::LogLine>,
396    ) -> Reply {
397        self.write(Target::TableKeywords, log, move |table| {
398            for (name, value) in fields {
399                put(table.keywords_mut(), &KeywordRef::Name(name), value, false)?;
400            }
401            Ok(())
402        })
403    }
404
405    /// `removekeyword(keyword)`.
406    pub(super) fn removekeyword(
407        &mut self,
408        key: KeywordRef,
409        log: &mut Vec<crate::tools::LogLine>,
410    ) -> Reply {
411        self.write(Target::TableKeywords, log, move |table| {
412            remove(table.keywords_mut(), &key, |name| {
413                format!("Table keyword {name} does not exist")
414            })
415        })
416    }
417
418    /// `putcolkeyword(columnname, keyword, value)`.
419    pub(super) fn putcolkeyword(
420        &mut self,
421        column: String,
422        key: KeywordRef,
423        value: Incoming,
424        log: &mut Vec<crate::tools::LogLine>,
425    ) -> Reply {
426        if matches!(&key, KeywordRef::Whole) {
427            return Ok(CasaValue::Bool(false));
428        }
429        if let Some(open) = self.open.as_ref()
430            && !open.column_allowed(&column)
431        {
432            return runtime(format!("Table column {column} is unknown"));
433        }
434        self.write(Target::ColumnKeywords, log, move |table| {
435            edit_column(table, &column, |keywords| put(keywords, &key, value, false))
436        })
437    }
438
439    /// `putcolkeywords(columnname, value)`.
440    pub(super) fn putcolkeywords(
441        &mut self,
442        column: String,
443        fields: Vec<(String, Incoming)>,
444        log: &mut Vec<crate::tools::LogLine>,
445    ) -> Reply {
446        if let Some(open) = self.open.as_ref()
447            && !open.column_allowed(&column)
448        {
449            return runtime(format!("Table column {column} is unknown"));
450        }
451        self.write(Target::ColumnKeywords, log, move |table| {
452            edit_column(table, &column, |keywords| {
453                for (name, value) in fields {
454                    put(keywords, &KeywordRef::Name(name), value, false)?;
455                }
456                Ok(())
457            })
458        })
459    }
460
461    /// `removecolkeyword(columnname, keyword)`.
462    pub(super) fn removecolkeyword(
463        &mut self,
464        column: String,
465        key: KeywordRef,
466        log: &mut Vec<crate::tools::LogLine>,
467    ) -> Reply {
468        if let Some(open) = self.open.as_ref()
469            && !open.column_allowed(&column)
470        {
471            return runtime(format!("Table column {column} is unknown"));
472        }
473        self.write(Target::ColumnKeywords, log, move |table| {
474            edit_column(table, &column, |keywords| {
475                remove(keywords, &key, |name| {
476                    format!("Keyword {name} in column {column} does not exist")
477                })
478            })
479        })
480    }
481
482    /// `putinfo(value)`: the fields `value` has replace the table's type,
483    /// subtype and readme; the readme text is kept as lines.
484    pub(super) fn putinfo(
485        &mut self,
486        update: InfoUpdate,
487        log: &mut Vec<crate::tools::LogLine>,
488    ) -> Reply {
489        self.write(Target::Info, log, move |table| {
490            let mut info = table.info().clone();
491            if let Some(table_type) = update.table_type {
492                info.table_type = table_type;
493            }
494            if let Some(sub_type) = update.sub_type {
495                info.sub_type = sub_type;
496            }
497            if let Some(readme) = update.readme {
498                info.readme = readme.split('\n').map(str::to_string).collect();
499            }
500            table.set_info(info);
501            Ok(())
502        })
503    }
504
505    /// `addreadmeline(value)`: appends a line to the readme.
506    pub(super) fn addreadmeline(
507        &mut self,
508        line: String,
509        log: &mut Vec<crate::tools::LogLine>,
510    ) -> Reply {
511        self.write(Target::Info, log, move |table| {
512            let mut info = table.info().clone();
513            info.readme.push(line);
514            table.set_info(info);
515            Ok(())
516        })
517    }
