17/// State shared by the calibrater's model, apply and solve operations.
18#[derive(Default)]
19pub struct CalibraterLifecycle {
20    legacy_iterator: bool,
21    equation: Option<VisibilityEquation>,
22}
23
24struct VisibilityEquation {
25    attachment: CalibrationAttachment,
26    model: CalibraterModel,
27    apply: ApplyTerms,
28    solve: Option<(CalibrationSolveRequest, CalibrationOutputMode)>,
29}
30
31#[derive(Default)]
32struct ApplyTerms {
33    tables: Vec<ApplyCalibrationTableSpec>,
34    parang: bool,
35}
36
37#[derive(Clone, Copy, Default)]
38enum CalibraterModel {
39    #[default]
40    FromMeasurementSet,
41    Point([f32; 4]),
42}
43
44/// A stateful calibrater operation failed.
45#[derive(Debug, thiserror::Error)]
46pub enum CalibraterLifecycleError {
47    /// The operation needs an attached MeasurementSet.
48    #[error("calibrater has no open MeasurementSet")]
49    Unattached,
50    /// No solver is configured in the current visibility equation.
51    #[error("Please run setsolve before attempting to solve.")]
52    NoSolver,
53    /// A point model must contain at most four finite CASA Stokes values.
54    #[error("a point model requires up to four finite Stokes values")]
55    InvalidPointModel,
56    /// MeasurementSet attachment or visibility selection failed.
57    #[error(transparent)]
58    Attachment(#[from] AttachmentError),
59    /// The configured solver could not complete.
60    #[error(transparent)]
61    Solve(#[from] CalibrationError),
62}
63
64impl CalibraterLifecycle {
65    /// Choose the visibility iterator before opening a MeasurementSet.
66    ///
67    /// Returns false when a MeasurementSet is already attached.
68    #[must_use]
69    pub fn set_legacy_iterator(&mut self, legacy: bool) -> bool {
70        if self.equation.is_some() {
71            return false;
72        }
73        self.legacy_iterator = legacy;
74        true
75    }
76
77    /// Whether the calibrater is configured to open the legacy iterator.
78    #[must_use]
79    pub fn legacy_iterator(&self) -> bool {
80        self.legacy_iterator
81    }
82
83    /// Attach a MeasurementSet and start a new visibility equation.
84    ///
85    /// The equation reads its model from the attached MeasurementSet until an
86    /// explicit point model is installed.
87    ///
88    /// # Errors
89    ///
90    /// Returns an error when the MeasurementSet cannot be opened.
91    pub fn open(&mut self, path: &Path) -> Result<(), CalibraterLifecycleError> {
92        let attachment = CalibrationAttachment::attach(
93            path,
94            AttachmentMode::Open,
95            AttachmentOptions::default(),
96            MsSelection::default(),
97        )?;
98        self.equation = Some(VisibilityEquation {
99            attachment,
100            model: CalibraterModel::FromMeasurementSet,
101            apply: ApplyTerms::default(),
102            solve: None,
103        });
104        Ok(())
105    }
106
107    /// The path of the currently attached MeasurementSet.
108    #[must_use]
109    pub fn dataset_path(&self) -> Option<&Path> {
110        self.equation
111            .as_ref()
112            .map(|equation| equation.attachment.path())
113    }
114
115    /// Release the current equation and attachment. The iterator choice is
116    /// retained for the next call to `open`.
117    pub fn close(&mut self) {
118        self.equation = None;
119    }
120
121    /// Replace the selected rows and channels and reset apply and solve.
122    ///
123    /// A successful selection keeps the installed point model. A selection
124    /// error reinitializes the equation over the same MeasurementSet.
125    ///
126    /// # Errors
127    ///
128    /// Returns an error without an open dataset or if a selector fails.
129    pub fn select(&mut self, selection: MsSelection) -> Result<(), CalibraterLifecycleError> {
130        let equation = self.equation_mut()?;
131        let selected = equation.attachment.select(selection);
132        equation.apply = ApplyTerms::default();
133        equation.solve = None;
134        if let Err(error) = selected {
135            equation.attachment.reset_selection();
136            equation.model = CalibraterModel::FromMeasurementSet;
137            return Err(error.into());
138        }
139        Ok(())
140    }
141
142    /// Install an explicit constant Stokes model. Empty and four-zero
143    /// vectors both install zero flux.
144    ///
145    /// # Errors
146    ///
147    /// Returns an error without an open dataset or for an invalid vector.
148    pub fn set_point_model(&mut self, stokes: &[f64]) -> Result<(), CalibraterLifecycleError> {
149        let equation = self.equation_mut()?;
150        let stokes =
151            point_model_stokes(stokes).ok_or(CalibraterLifecycleError::InvalidPointModel)?;
152        equation.model = CalibraterModel::Point(stokes);
153        Ok(())
154    }
155
156    /// Append an applied calibration term in call order.
157    ///
158    /// # Errors
159    ///
160    /// Returns an error without an open dataset.
161    pub fn add_apply(
162        &mut self,
163        apply: ApplyCalibrationTableSpec,
164    ) -> Result<(), CalibraterLifecycleError> {
165        self.equation_mut()?.apply.tables.push(apply);
166        Ok(())
167    }
168
169    /// Set the parallactic-angle apply term in the current equation.
170    ///
171    /// # Errors
172    ///
173    /// Returns an error without an open dataset.
174    pub fn set_parallactic_angle(&mut self, apply: bool) -> Result<(), CalibraterLifecycleError> {
175        self.equation_mut()?.apply.parang = apply;
176        Ok(())
177    }
178
179    /// Reset the selected scratch calibration columns with the iterator
180    /// that opened this MeasurementSet.
181    ///
182    /// # Errors
183    ///
184    /// Returns an error without an open dataset, when the iterator does not
185    /// support the operation, or when the scratch update fails.
186    pub fn init_cal_set(&self, calset: i64) -> Result<(), CalibraterLifecycleError> {
187        let equation = self.equation()?;
188        let mut opened =
189            OpenedMs::open(equation.attachment.path()).map_err(AttachmentError::from)?;
190        opened.lock = reset_calibration_set(
191            &mut opened.measurement_set,
192            equation.attachment.selection(),
193            calset,
194            self.legacy_iterator,
195            opened.lock,
196        )
197        .map_err(AttachmentError::from)?;
198        opened.release().map_err(AttachmentError::from)?;
199        Ok(())
200    }
201
202    /// Configure the solver for the current visibility equation.
203    ///
204    /// # Errors
205    ///
206    /// Returns an error without an open dataset.
207    pub fn set_solve(
208        &mut self,
209        request: CalibrationSolveRequest,
210        mode: CalibrationOutputMode,
211    ) -> Result<(), CalibraterLifecycleError> {
212        self.equation_mut()?.solve = Some((request, mode));
213        Ok(())
214    }
215
216    /// Clear the requested terms in the current visibility equation.
217    ///
218    /// # Errors
219    ///
220    /// Returns an error without an open dataset.
221    pub fn reset(&mut self, apply: bool, solve: bool) -> Result<(), CalibraterLifecycleError> {
222        let equation = self.equation_mut()?;
223        if apply {
224            equation.apply = ApplyTerms::default();
225        }
226        if solve {
227            equation.solve = None;
228        }
229        Ok(())
230    }
231
232    /// Run the configured solver using this equation's model, selection and
233    /// prior calibration terms.
234    ///
235    /// # Errors
236    ///
237    /// Returns an error without an open dataset or configured solver, or
238    /// when the solve fails.
239    pub fn solve(&self) -> Result<CalibrationSolveResult, CalibraterLifecycleError> {
240        let equation = self.equation()?;
241        let (mut request, mode) = equation
242            .solve
243            .clone()
244            .ok_or(CalibraterLifecycleError::NoSolver)?;
245        let (model_source, stokes) = match equation.model {
246            CalibraterModel::FromMeasurementSet => {
247                (GainSolveModelSource::FromMeasurementSet, [1., 0., 0., 0.])
248            }
249            CalibraterModel::Point(stokes) => (GainSolveModelSource::PointSource, stokes),
250        };
251        let selection = equation.attachment.selection().clone();
252        let apply = equation.apply.tables.clone();
253        match &mut request {
254            CalibrationSolveRequest::Gain(gain) => {
255                gain.selection = selection;
256                gain.model_source = model_source;
257                gain.smodel = stokes;
258                gain.prior_calibration_tables = apply;
259                gain.parang = equation.apply.parang;
260            }
261            CalibrationSolveRequest::Bandpass(bandpass) => {
262                bandpass.selection = selection;
263                bandpass.model_source = model_source;
264                bandpass.smodel = stokes;
265                bandpass.prior_calibration_tables = apply;
266                bandpass.parang = equation.apply.parang;
267            }
268        }
269        Ok(solve_calibration_with_output_mode(
270            CalibrationDataset::path(equation.attachment.path()),
271            request,
272            mode,
273        )?)
274    }
275
276    fn equation(&self) -> Result<&VisibilityEquation, CalibraterLifecycleError> {
277        self.equation
278            .as_ref()
279            .ok_or(CalibraterLifecycleError::Unattached)
280    }
281
282    fn equation_mut(&mut self) -> Result<&mut VisibilityEquation, CalibraterLifecycleError> {
283        self.equation
284            .as_mut()
285            .ok_or(CalibraterLifecycleError::Unattached)
286    }
