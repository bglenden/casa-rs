2//! One hypercube of a streamed `TiledShapeStMan` column, written a row at a
3//! time.
4//!
5//! A row fills its slot in every tile across the cell, in the band of tiles
6//! that holds it along the row axis. Tiles are numbered with the row axis
7//! slowest, so a band is contiguous in the cube's file. A cube whose band
8//! fits the writer's budget assembles each band in memory and writes it with
9//! one call when its last row arrives. Otherwise each tile row is written in
10//! place as it is pushed, a Bool tile row by reading back the bytes it
11//! shares with its neighbours. The cubes of a column share one tile-row
12//! scratch, held by the column's writer ([`ShapeCubeWorkingSet`]).
13
14use std::fs::{File, OpenOptions};
15use std::io::{Read, Seek, SeekFrom, Write};
16use std::path::PathBuf;
17
18use super::{
19    StreamedPrimitiveTilePlan, StreamedTiledShapeCubeLayout, StreamedTiledShapeValueType,
20    compute_tile_layout, copy_cube_to_tile, linear_to_nd,
21};
22use crate::storage::StorageError;
23use crate::storage::canonical::write_bool_bits_from_bytes;

25#[derive(Debug)]
26pub(super) struct DirectShapeCubeWriter {
27    pub(super) temp_data_path: PathBuf,
28    pub(super) file: File,
29    pub(super) cell_shape: Vec<usize>,
30    pub(super) row_count: usize,
31    pub(super) rows_written: usize,
32    pub(super) tile_shape: Vec<usize>,
33    tile_plans: Vec<StreamedPrimitiveTilePlan>,
34    pub(super) non_row_tile_count: usize,
35    pub(super) rows_per_tile: usize,
36    tile_cell_nelem: usize,
37    pub(super) bucket_size: usize,
38    /// Every tile of the band being assembled; empty when the cube writes
39    /// each tile row in place.
40    band: Vec<u8>,
41}
42
43/// The bytes a cube works in while it writes a row.
44///
45/// The encoded row and a Bool tile row's packed bytes live only while the
46/// row is written; the tile row is the writer's scratch, shared by every
47/// cube and sized for the largest. A column holds, at once, its largest
48/// encoded row, its largest tile row and its largest packed Bool tile row,
49/// which may come from different cubes ([`Self::max`], [`Self::bytes`]).
50#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
51pub(super) struct ShapeCubeWorkingSet {
52    /// One row's cell, encoded.
53    pub(super) row_bytes: usize,
54    /// One tile row of the cell.
55    pub(super) tile_row_bytes: usize,
56    /// One Bool tile row's packed bits and the byte they may share.
57    pub(super) packed_bool_bytes: usize,
58}
59
60impl ShapeCubeWorkingSet {
61    /// The larger of each part of `self` and `other`.
62    pub(super) fn max(self, other: Self) -> Self {
63        Self {
64            row_bytes: self.row_bytes.max(other.row_bytes),
65            tile_row_bytes: self.tile_row_bytes.max(other.tile_row_bytes),
66            packed_bool_bytes: self.packed_bool_bytes.max(other.packed_bool_bytes),
67        }
68    }
69
70    /// Every part, held at once.
71    pub(super) fn bytes(self) -> Option<usize> {
72        self.row_bytes
73            .checked_add(self.tile_row_bytes)?
74            .checked_add(self.packed_bool_bytes)
75    }
76}
77
78impl DirectShapeCubeWriter {
79    /// Create the cube's file at `temp_data_path`, sized for every planned
80    /// row, and return the writer with the bytes it works in per row.
81    pub(super) fn create(
82        temp_data_path: PathBuf,
83        layout: StreamedTiledShapeCubeLayout,
84        value_type: StreamedTiledShapeValueType,
85    ) -> Result<(Self, ShapeCubeWorkingSet), StorageError> {
86        let elem_size = value_type.elem_size();
87        let overflow = |what: &str| {
88            StorageError::FormatMismatch(format!("streamed TiledShape {what} overflows usize"))
89        };
90        let product = |extents: &[usize], what: &str| {
91            extents
92                .iter()
93                .try_fold(1usize, |acc, &dim| acc.checked_mul(dim))
94                .ok_or_else(|| overflow(what))
95        };
96        let cell_rank = layout.cell_shape.len();
97        let cell_nelem = product(&layout.cell_shape, "cell size")?;
98        let tile_cell_nelem = product(&layout.tile_shape[..cell_rank], "tile size")?;
99        let row_bytes = cell_nelem
100            .checked_mul(elem_size)
101            .ok_or_else(|| overflow("row byte size"))?;
102        let tile_row_bytes = tile_cell_nelem
103            .checked_mul(elem_size)
104            .ok_or_else(|| overflow("tile-row size"))?;
105        let packed_bool_bytes = if value_type == StreamedTiledShapeValueType::Bool {
106            tile_cell_nelem.div_ceil(8) + 1
107        } else {
108            0
109        };
110        let working_set = ShapeCubeWorkingSet {
111            row_bytes,
112            tile_row_bytes,
113            packed_bool_bytes,
114        };
115        working_set.bytes().ok_or_else(|| overflow("working set"))?;
116
117        let mut cube_shape = layout.cell_shape.clone();
118        cube_shape.push(layout.row_count);
119        let tiles_per_dim: Vec<usize> = cube_shape
120            .iter()
121            .zip(layout.tile_shape.iter())
122            .map(|(&cube, &tile)| cube.div_ceil(tile))
123            .collect();
124        let cell_tiles_per_dim = &tiles_per_dim[..cell_rank];
125        let non_row_tile_count = cell_tiles_per_dim.iter().product::<usize>();
126        let tile_plans = (0..non_row_tile_count)
127            .map(|tile_index| {
128                let tile_pos = linear_to_nd(tile_index, cell_tiles_per_dim);
129                let cell_start: Vec<usize> = tile_pos
130                    .iter()
131                    .enumerate()
132                    .map(|(axis, &pos)| pos * layout.tile_shape[axis])
133                    .collect();
134                let actual_extent: Vec<usize> = cell_start
135                    .iter()
136                    .enumerate()
137                    .map(|(axis, &start)| {
138                        std::cmp::min(layout.tile_shape[axis], layout.cell_shape[axis] - start)
139                    })
140                    .collect();
141                StreamedPrimitiveTilePlan {
142                    cell_start,
143                    actual_extent,
144                }
145            })
146            .collect();
147        let (bucket_size, _) =
148            compute_tile_layout(&[value_type.casacore_type()], &layout.tile_shape);
149        let file_bytes = product(&tiles_per_dim, "tile count")?
150            .checked_mul(bucket_size)
151            .ok_or_else(|| overflow("file size"))?;
152        let file = OpenOptions::new()
153            .create(true)
154            .truncate(true)
155            .read(true)
156            .write(true)
157            .open(&temp_data_path)?;
158        file.set_len(file_bytes as u64)?;
159        let rows_per_tile = *layout.tile_shape.last().expect("validated tile row axis");
160        let cube = Self {
161            temp_data_path,
162            file,
163            cell_shape: layout.cell_shape,
164            row_count: layout.row_count,
165            rows_written: 0,
166            tile_shape: layout.tile_shape,
167            tile_plans,
168            non_row_tile_count,
169            rows_per_tile,
170            tile_cell_nelem,
171            bucket_size,
172            band: Vec::new(),
173        };
174        Ok((cube, working_set))
175    }
176
177    /// Bytes of one band: every tile across the cell, one tile along the row
178    /// axis.
179    pub(super) fn band_bytes(&self) -> usize {
180        self.non_row_tile_count * self.bucket_size
181    }
182
183    /// Assemble each band in memory from now on.
184    pub(super) fn buffer_bands(&mut self) {
185        self.band = vec![0; self.band_bytes()];
186    }
187
188    /// Write the next row's cell, `encoded` in the cube's cell order, one
189    /// tile row at a time through `tile_row`, the column's scratch of at
190    /// least this cube's tile-row bytes.
191    pub(super) fn write_row(
192        &mut self,
193        value_type: StreamedTiledShapeValueType,
194        encoded: &[u8],
195        tile_row: &mut [u8],
196    ) -> Result<(), StorageError> {
197        let elem_size = value_type.elem_size();
198        let cell_rank = self.cell_shape.len();
199        let row_tile_index = self.rows_written / self.rows_per_tile;
200        let row_in_tile = self.rows_written % self.rows_per_tile;
201        let tile_row_bytes = self.tile_cell_nelem * elem_size;
202        let tile_row = &mut tile_row[..tile_row_bytes];
203        for (cell_tile_index, plan) in self.tile_plans.iter().enumerate() {
204            tile_row.fill(0);
205            copy_cube_to_tile(
206                encoded,
207                &self.cell_shape,
208                tile_row,
209                &self.tile_shape[..cell_rank],
210                &plan.cell_start,
211                &plan.actual_extent,
212                elem_size,
213            );
214            let band_offset = cell_tile_index * self.bucket_size;
215            let bucket_start = row_tile_index * self.band_bytes() + band_offset;
216            if value_type == StreamedTiledShapeValueType::Bool {
217                let bit_offset = row_in_tile * self.tile_cell_nelem;
218                let byte_offset = bit_offset / 8;
219                let bit_in_byte = bit_offset % 8;
220                let byte_count = (bit_in_byte + self.tile_cell_nelem).div_ceil(8);
221                if self.band.is_empty() {
222                    let mut packed = vec![0u8; byte_count];
223                    self.file
224                        .seek(SeekFrom::Start((bucket_start + byte_offset) as u64))?;
225                    self.file.read_exact(&mut packed)?;
226                    write_bool_bits_from_bytes(&mut packed, bit_in_byte, tile_row);
227                    self.file
228                        .seek(SeekFrom::Start((bucket_start + byte_offset) as u64))?;
229                    self.file.write_all(&packed)?;
230                } else {
231                    let start = band_offset + byte_offset;
232                    write_bool_bits_from_bytes(
233                        &mut self.band[start..start + byte_count],
234                        bit_in_byte,
235                        tile_row,
236                    );
237                }
238            } else {
239                let row_offset = row_in_tile * tile_row_bytes;
240                if self.band.is_empty() {
241                    self.file
242                        .seek(SeekFrom::Start((bucket_start + row_offset) as u64))?;
243                    self.file.write_all(tile_row)?;
244                } else {
245                    let start = band_offset + row_offset;
246                    self.band[start..start + tile_row_bytes].copy_from_slice(tile_row);
247                }
248            }
249        }
250        self.rows_written += 1;
251        if !self.band.is_empty() && self.rows_written.is_multiple_of(self.rows_per_tile) {
252            self.write_band(row_tile_index)?;
253        }
254        Ok(())
255    }
256
257    /// Write the last band if rows ended inside it. Its rows past the last
258    /// are zero, as they are in the file.
259    pub(super) fn finish_band(&mut self) -> Result<(), StorageError> {
260        if !self.band.is_empty() && !self.rows_written.is_multiple_of(self.rows_per_tile) {
261            self.write_band(self.rows_written / self.rows_per_tile)?;
262        }
263        Ok(())
264    }
265
266    fn write_band(&mut self, row_tile_index: usize) -> Result<(), StorageError> {
267        self.file
268            .seek(SeekFrom::Start((row_tile_index * self.band.len()) as u64))?;
269        self.file.write_all(&self.band)?;
270        self.band.fill(0);
271        Ok(())
272    }
273}

282    /// Two shapes, each tiled in two tiles across the cell; the first ends
283    /// inside a band, and undefined rows fall between them.
284    fn layouts() -> Vec<StreamedTiledShapeCubeLayout> {
285        vec![
286            StreamedTiledShapeCubeLayout::new(vec![2, 5], 9, vec![2, 3, 3]),
287            StreamedTiledShapeCubeLayout::new(vec![2, 4], 5, vec![1, 4, 2]),
288        ]
289    }
290
291    /// The pushes: `Some(0)` a row of the first shape, `Some(1)` of the
292    /// second, `None` an undefined row.
293    const ORDER: [Option<usize>; 14] = [
294        Some(0),
295        Some(0),
296        Some(1),
297        None,
298        Some(0),
299        Some(1),
300        Some(1),
301        Some(0),
302        None,
303        Some(0),
304        Some(1),
305        Some(0),
306        None,
307        Some(0),
308    ];
309
310    fn write(
311        dir: &std::path::Path,
312        name: &str,
313        value_type: StreamedTiledShapeValueType,
314        budget: usize,
315    ) -> (StreamedTiledShapeColumn, Vec<Vec<u8>>) {
316        let mut writer =
317            StreamingTiledShapeWriter::create(dir.join(name), value_type, layouts(), budget, true)
318                .expect("create writer");
319        let shapes = [[2usize, 5], [2, 4]];
320        for (row, shape) in ORDER.iter().enumerate() {
321            let Some(shape) = shape else {
322                writer.push_undefined_row().expect("undefined row");
323                continue;
324            };
325            let cell = shapes[*shape];
326            let samples = cell[0] * cell[1];
327            if value_type == StreamedTiledShapeValueType::Bool {
328                let values = (0..samples)
329                    .map(|sample| (row * 7 + sample * 3) % 5 < 2)
330                    .collect::<Vec<_>>();
331                writer.push_bool_row(&cell, &values).expect("bool row");
332            } else {
333                let values = (0..samples)
334                    .map(|sample| Complex32::new(row as f32, sample as f32 + 0.5))
335                    .collect::<Vec<_>>();
336                writer
337                    .push_complex32_row(&cell, &values)
338                    .expect("complex row");
339            }
340        }
341        let column = writer.finish().expect("finish");
342        let files = column
343            .cubes
344            .iter()
345            .map(|cube| std::fs::read(&cube.temp_data_path).expect("cube file"))
346            .collect();
347        (column, files)
348    }

350    #[test]
351    fn band_buffered_cubes_write_the_files_of_in_place_rows() {
352        let dir = tempfile::tempdir().expect("tempdir");
353        for value_type in [
354            StreamedTiledShapeValueType::Complex32,
355            StreamedTiledShapeValueType::Bool,
356        ] {
357            // A budget that holds the working set and the largest row map,
358            // but not the bands, keeps every tile row in place.
359            let elem = value_type.elem_size();
360            let packed = if value_type == StreamedTiledShapeValueType::Bool {
361                6usize.div_ceil(8) + 1
362            } else {
363                0
364            };
365            let working = 10 * elem + 6 * elem + packed;
366            let in_place_budget = working + ORDER.len() * 3 * size_of::<u32>();
367            let (direct, direct_files) = write(
368                dir.path(),
369                &format!("{value_type:?}.direct"),
370                value_type,
371                in_place_budget,
372            );
373            let (banded, banded_files) = write(
374                dir.path(),
375                &format!("{value_type:?}.banded"),
376                value_type,
377                1 << 20,
378            );
379            assert!(
380                banded.max_resident_bytes() > direct.max_resident_bytes(),
381                "{value_type:?}: the large budget buffers bands"
382            );
383            assert_eq!(direct_files.len(), 2);
384            assert_eq!(direct_files, banded_files, "{value_type:?} cube files");
385            assert_eq!(direct.row_map, banded.row_map);
386            assert_eq!(direct.cube_map, banded.cube_map);
387            assert_eq!(direct.pos_map, banded.pos_map);
388        }
389    }
