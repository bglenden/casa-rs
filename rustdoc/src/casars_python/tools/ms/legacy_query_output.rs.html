66impl QueryOutputOptions {
67    pub(super) fn parse(giving: &GivingClause) -> PyResult<Self> {
68        let mut output = Self {
69            independent: false,
70            memory: false,
71            overwrite: true,
72            endian: None,
73            storage: None,
74            block_size: -2,
75        };
76        if let Some(mode) = &giving.output_type {
77            output.set_mode(mode)?;
78        }
79        if let Some(options) = &giving.output_options {
80            let mut fields: Vec<(&str, ExprValue, &str)> = Vec::new();
81            for field in options {
82                let (value, data_type) = option_value(field)?;
83                if let Some((_, existing, previous_type)) =
84                    fields.iter_mut().find(|(name, _, _)| *name == field.name)
85                {
86                    if *previous_type != data_type {
87                        return Err(runtime_error(format!(
88                            "Incorrect TaQL output option data type for field {}",
89                            field.name
90                        )));
91                    }
92                    *existing = value;
93                } else {
94                    fields.push((&field.name, value, data_type));
95                }
96            }
97            for (name, value, _) in fields {
98                output.set(name, &value)?;
99            }
100        }
101        if output.independent && !output.memory && giving.table_name.is_empty() {
102            return Err(runtime_error("TaQL output must name a MeasurementSet"));
103        }
104        Ok(output)
105    }
106
107    fn set_mode(&mut self, mode: &str) -> PyResult<()> {
108        self.memory = false;
109        match mode.to_ascii_lowercase().as_str() {
110            "plain" | "scratch" => (),
111            "plain_local" => self.endian = Some(EndianFormat::LocalEndian),
112            "plain_big" => self.endian = Some(EndianFormat::BigEndian),
113            "plain_little" => self.endian = Some(EndianFormat::LittleEndian),
114            "memory" => self.memory = true,
115            _ => {
116                return Err(runtime_error(format!(
117                    "{mode} is an invalid TaQL output type"
118                )));
119            }
120        }
121        self.independent = true;
122        Ok(())
123    }
124
125    fn set(&mut self, name: &str, value: &ExprValue) -> PyResult<()> {
126        let key = name.to_ascii_lowercase();
127        match (key.as_str(), value) {
128            ("type", ExprValue::String(mode)) => match mode.to_ascii_lowercase().as_str() {
129                "plain" | "scratch" | "memory" => self.set_mode(mode),
130                _ => Err(runtime_error(
131                    "TaQL output type must be plain, scratch or memory",
132                )),
133            },
134            ("endian", ExprValue::String(endian)) => {
135                self.endian = match endian.to_ascii_lowercase().as_str() {
136                    "aipsrc" => None,
137                    "local" => Some(EndianFormat::LocalEndian),
138                    "big" => Some(EndianFormat::BigEndian),
139                    "little" => Some(EndianFormat::LittleEndian),
140                    _ => return Err(runtime_error("Invalid TaQL output byte order")),
141                };
142                Ok(())
143            }
144            ("overwrite", ExprValue::Bool(overwrite)) => {
145                self.overwrite = *overwrite;
146                Ok(())
147            }
148            ("storage", ExprValue::String(storage)) => {
149                self.storage = match storage.to_ascii_lowercase().as_str() {
150                    "aipsrc" => None,
151                    "sepfile" | "default" => Some(QueryStorage::Separate),
152                    "multifile" => Some(QueryStorage::MultiFile),
153                    "multihdf5" => Some(QueryStorage::MultiHdf5),
154                    _ => return Err(runtime_error("Invalid TaQL output storage option")),
155                };
156                Ok(())
157            }
158            ("blocksize", ExprValue::Int(size)) => {
159                self.block_size = *size as i32;
160                Ok(())
161            }
162            (_, ExprValue::Bool(true)) => self.set_mode(&key),
163            _ => Err(runtime_error(format!("Invalid TaQL output option {name}"))),
164        }
165    }
166
167    fn disk_format(&self) -> PyResult<QueryDiskFormat> {
168        let defaults = query_defaults();
169        let storage = self.storage.unwrap_or(match defaults.storage.as_deref() {
170            Some("multifile") => QueryStorage::MultiFile,
171            Some("multihdf5") => QueryStorage::MultiHdf5,
172            _ => QueryStorage::Separate,
173        });
174        let multifile_block_size = match storage {
175            QueryStorage::Separate => None,
176            QueryStorage::MultiFile => {
177                let configured = if self.block_size <= -2 {
178                    defaults.block_size
179                } else {
180                    self.block_size
181                };
182                Some(if configured > 0 {
183                    configured as usize
184                } else {
185                    4 * 1024 * 1024
186                })
187            }
188            QueryStorage::MultiHdf5 => {
189                return Err(runtime_error("TaQL MultiHDF5 storage is unavailable"));
190            }
191        };
192        Ok(QueryDiskFormat {
193            endian: self.endian.unwrap_or(defaults.endian),
194            multifile_block_size,
195        })
196    }
197}

199fn option_value(field: &GivingOption) -> PyResult<(ExprValue, &'static str)> {
200    let value =
201        eval_constant(&field.value, &field.name, IndexStyle::default()).map_err(runtime_error)?;
202    let Some(data_type) = &field.data_type else {
203        let data_type = match value {
204            ExprValue::Bool(_) => "B",
205            ExprValue::String(_) => "S",
206            ExprValue::Int(_) => "I8",
207            _ => {
208                return Err(runtime_error(format!(
209                    "Unsupported value for TaQL output option {}",
210                    field.name
211                )));
212            }
213        };
214        return Ok((value, data_type));
215    };
216    let value = match (data_type.to_ascii_uppercase().as_str(), value) {
217        ("B", value @ ExprValue::Bool(_)) => (value, "B"),
218        ("S", value @ ExprValue::String(_)) => (value, "S"),
219        ("I2", ExprValue::Int(value)) => (ExprValue::Int(i64::from(value as i16)), "I2"),
220        ("I4", ExprValue::Int(value)) => (ExprValue::Int(i64::from(value as i32)), "I4"),
221        ("I8", value @ ExprValue::Int(_)) => (value, "I8"),
222        ("U1", ExprValue::Int(value)) => (ExprValue::Int(i64::from(value as u8)), "U1"),
223        ("U4", ExprValue::Int(value)) => (ExprValue::Int(i64::from(value as u32)), "U4"),
224        _ => {
225            return Err(runtime_error(format!(
226                "Invalid data type {data_type} for TaQL output option {}",
227                field.name
228            )));
229        }
230    };
231    Ok(value)
232}

234impl AttachedMs {
235    pub(super) fn stage_legacy_query(
236        &self,
237        table: &Table,
238        rows: &[usize],
239        name: &str,
240        options: QueryOutputOptions,
241        original: &Path,
242    ) -> PyResult<Option<StagedLegacyQuery>> {
243        if name.is_empty() {
244            return Ok(None);
245        }
246        let format = options
247            .independent
248            .then(|| options.disk_format())
249            .transpose()?
250            .or_else(|| self.memory_ms.as_ref().map(|_| QueryDiskFormat::default()));
251        let active = self.legacy_selection.query_parent(&self.path);
252        if active.is_none() && self.memory_ms.is_some() && QuerySubtable::has_implicit_parent(name)?
253        {
254            return Err(runtime_error(
255                "the active TaQL view has no persisted directory",
256            ));
257        }
258        let subtable = QuerySubtable::resolve(name, active, &self.path)?;
259        let output = if let Some(subtable) = &subtable {
260            subtable.output.clone()
261        } else {
262            canonical_named_output(name)?
263        };
264        let parent = output
265            .parent()
266            .ok_or_else(|| runtime_error("TaQL output has no parent directory"))?;
267        let copy_parent = self.check_query_destination(
268            &output,
269            original,
270            options.independent,
271            subtable.as_ref(),
272        )?;
273        crate::tools::table::check_target(&output).map_err(PyErr::from)?;
274        let destination = SortedDestination::inspect(&output, "TaQL selection")?;
275        if !options.overwrite && !matches!(destination, SortedDestination::Absent) {
276            return Err(runtime_error("TaQL output already exists"));
277        }
278        let stage = tempfile::Builder::new()
279            .prefix(".taql-ms-")
280            .tempdir_in(if format.is_some() {
281                &copy_parent
282            } else {
283                parent
284            })
285            .map_err(runtime_error)?;
286        let staged = if let Some(format) = format {
287            let staged = stage.path().join("selected.ms");
288            let source_locks = self
289                .read_locks
290                .keys()
291                .chain(self.write_locks.keys())
292                .cloned()
293                .collect();
294            if let Some(memory) = &self.memory_ms {
295                memory
296                    .to_memory_rows(rows)
297                    .and_then(|mut copy| copy.save_detached_in_stage(&staged, format.endian))
298                    .map_err(runtime_error)?;
299            } else {
300                write_selected_measurement_set_rows(
301                    table,
302                    rows,
303                    &staged,
304                    format.endian,
305                    &source_locks,
306                )
307                .map_err(runtime_error)?;
308            }
309            if let Some(block_size) = format.multifile_block_size {
310                Table::pack_staged_multifile(&staged, block_size).map_err(runtime_error)?;
311            }
312            staged
313        } else {
314            let staged = stage.path().to_owned();
315            table
316                .select_rows(rows)
317                .and_then(|view| view.save_relative_to(TableOptions::new(&staged), &output))
318                .map_err(runtime_error)?;
319            staged
320        };
321        self.working_metadata_link()
322            .verify_directory()
323            .map_err(runtime_error)?;
324        let stage_lock = format
325            .is_some()
326            .then(|| TableWriteLock::acquire_to_reread(&staged, 1))
327            .transpose()
328            .map_err(runtime_error)?;
329        Ok(Some(StagedLegacyQuery {
330            output,
331            staged,
332            stage,
333            destination,
334            subtable,
335            independent: options.independent,
336            _stage_lock: stage_lock,
337        }))
338    }
339
340    fn check_query_destination(
341        &self,
342        output: &Path,
343        original: &Path,
344        independent: bool,
345        subtable: Option<&QuerySubtable>,
346    ) -> PyResult<PathBuf> {
347        let current = self.path.as_path();
348        let active = self.legacy_selection.query_parent(current);
349        let inputs = std::iter::once(current)
350            .chain((original != current).then_some(original))
351            .chain(active.filter(|path| *path != current && *path != original));
352        let mut parent = output
353            .parent()
354            .expect("validated output has a parent")
355            .to_owned();
356        if self.memory_ms.is_some() {
357            for (source, linked) in &self.memory_source_paths {
358                if source.starts_with(output)
359                    || (*linked && output.starts_with(source) && subtable.is_none())
360                {
361                    return Err(runtime_error("TaQL output overlaps a source table"));
362                }
363                if *linked && parent.starts_with(source) {
364                    parent = source
365                        .parent()
366                        .ok_or_else(|| runtime_error("no staging parent outside the source table"))?
367                        .to_owned();
368                }
369            }
370            return Ok(parent);
371        }
372        if let Some(link) = subtable {
373            for input in inputs {
374                let outside =
375                    named_keyword_output_staging_parent(input, output, &link.owner, &link.keyword)
376                        .map_err(runtime_error)?;
377                if parent.starts_with(&outside) {
378                    parent = outside;
379                }
380            }
381        } else if independent {
382            for input in inputs {
383                if input.starts_with(output) {
384                    return Err(runtime_error(
385                        "cannot overwrite the attached MeasurementSet with a named copy",
386                    ));
387                }
388                check_named_output_path(input, output).map_err(runtime_error)?;
389            }
390        } else {
391            for input in inputs.chain(self.sources.iter().map(|path| path.as_path())) {
392                crate::tools::table::refuse_nested(output, input).map_err(PyErr::from)?;
393            }
394        }
395        Ok(parent)
396    }
397}

399impl StagedLegacyQuery {
400    pub(super) fn publish(
401        self,
402        retained: &mut BTreeMap<PathBuf, TableWriteLock>,
403    ) -> PyResult<PublishedLegacyQuery> {
404        let mut owner_lock = None;
405        let link_lock = if let Some(subtable) = &self.subtable {
406            let lock = match retained.get_mut(&subtable.owner) {
407                Some(lock) => lock,
408                None => owner_lock.insert(
409                    TableWriteLock::acquire_to_reread(&subtable.owner, 1).map_err(runtime_error)?,
410                ),
411            };
412            subtable.verify_lock(lock)?;
413            Some((subtable, lock))
414        } else {
415            None
416        };
417        publish_selection(
418            &self.output,
419            &self.staged,
420            self.stage,
421            self.destination,
422            |publication| {
423                let attached = self
424                    .independent
425                    .then(|| {
426                        AttachedMs::open_with_publication_guard(
427                            self.output.clone(),
428                            false,
429                            false,
430                            false,
431                            publication,
432                        )
433                    })
434                    .transpose()?;
435                if let Some((subtable, lock)) = link_lock {
436                    subtable.write_link(lock)?;
437                }
438                Ok(match attached {
439                    Some(attached) => PublishedLegacyQuery::Independent(Box::new(attached)),
440                    None => PublishedLegacyQuery::Reference(
441                        crate::tools::table::attach_table_paths(vec![self.output.clone()]),
442                    ),
443                })
444            },
445        )
446    }
447}

449fn canonical_named_output(name: &str) -> PyResult<PathBuf> {
450    let absolute = std::path::absolute(name).map_err(runtime_error)?;
451    let name = absolute
452        .file_name()
453        .ok_or_else(|| runtime_error("TaQL output must name a MeasurementSet"))?;
454    let parent = absolute
455        .parent()
456        .ok_or_else(|| runtime_error("TaQL output has no parent directory"))?
457        .canonicalize()
458        .map_err(runtime_error)?;
459    Ok(parent.join(name))
460}

462fn publish_selection<T>(
463    output: &Path,
464    staged: &Path,
465    stage: tempfile::TempDir,
466    destination: SortedDestination,
467    attach: impl FnOnce(&MutexGuard<'static, ()>) -> PyResult<T>,
468) -> PyResult<T> {
469    let publication = crate::tools::table::lock_attachment_publication();
470    let had_table = destination.is_table();
471    let _locks = if had_table {
472        if crate::tools::table::replacement_target_is_in_use(output) {
473            return Err(runtime_error("TaQL output is still open in this process"));
474        }
475        check_write_access(output).map_err(runtime_error)?;
476        lock_table_tree_for_replacement(output).map_err(runtime_error)?
477    } else {
478        Vec::new()
479    };
480    let empty_permissions = match &destination {
481        SortedDestination::Empty(metadata) => Some(metadata.permissions()),
482        _ => None,
483    };
484    destination.publish(staged, output, "TaQL selection")?;
485    match attach(&publication) {
486        Ok(attached) => Ok(attached),
487        Err(error) => match restore_destination(output, staged, had_table, empty_permissions) {
488            Ok(()) => Err(error),
489            Err(rollback) => Err(runtime_error(format!(
490                "could not attach TaQL output: {error}; rollback failed: {rollback}; staging retained at {}",
491                stage.keep().display()
492            ))),
493        },
494    }
495}
496
497fn restore_destination(
498    output: &Path,
499    staged: &Path,
500    had_table: bool,
501    empty_permissions: Option<fs::Permissions>,
502) -> io::Result<()> {
503    if had_table {
504        return exchange_directories(staged, output);
505    }
506    rename_without_replacing(output, staged)?;
507    if let Some(permissions) = empty_permissions {
508        fs::create_dir(output)?;
509        fs::set_permissions(output, permissions)?;
510    }
511    Ok(())
512}

514#[cfg(test)]
515mod tests {
516    use casa_tables::TableSchema;
517
518    use super::*;
519
520    #[test]
521    fn failed_attachment_restores_the_previous_table() {
522        let scratch = tempfile::tempdir().unwrap();
523        let stage = tempfile::tempdir_in(scratch.path()).unwrap();
524        let staged = stage.path().join("result.ms");
525        let output = scratch.path().join("output.ms");
526        for path in [&staged, &output] {
527            let schema = TableSchema::new(Vec::new()).unwrap();
528            Table::with_schema(schema)
529                .save(TableOptions::new(path))
530                .unwrap();
531        }
532        fs::write(output.join("original"), "preserve").unwrap();
533        fs::write(staged.join("replacement"), "staged").unwrap();
534        let destination = SortedDestination::inspect(&output, "TaQL selection").unwrap();
535        let mut reached_attachment = false;
536        let result = publish_selection(&output, &staged, stage, destination, |_| {
537            reached_attachment = true;
538            assert!(output.join("replacement").is_file());
539            Err::<(), _>(runtime_error("simulated failure attaching the new view"))
540        });
541        assert!(reached_attachment);
542        assert!(result.is_err());
543        assert_eq!(
544            fs::read_to_string(output.join("original")).unwrap(),
545            "preserve"
546        );
547        assert!(!output.join("replacement").exists());
548    }
