2//! Parallactic-angle rotation epochs for an imported aperture buffer.
3
4use std::sync::Arc;
5
6/// Feed angles at the selected observation times and the rotation threshold.
7#[derive(Clone, Debug, PartialEq)]
8pub struct AwPaRotation {
9    /// Minimum absolute change that triggers rotation, in radians.
10    pub step_rad: f64,
11    /// The selected times in increasing order, seconds since MJD 0, with
12    /// the corresponding antenna-0 feed PA in Float radians.
13    pub samples: Arc<[(f64, f32)]>,
14}
15
16impl AwPaRotation {
17    /// The input is shared with the deployment and stays allocated.
18    pub(super) fn samples_bytes(&self) -> usize {
19        self.samples.len() * size_of::<(f64, f32)>() + 2 * size_of::<usize>()
20    }
21
22    /// In the single-buffer case, record each rotation epoch. The cached
23    /// angle is retained until the next sample differs strictly more than
24    /// the requested threshold (`AWConvFunc::prepareConvFunction`).
25    pub(super) fn schedule(&self, base_deg: f64) -> Vec<(f64, f64)> {
26        let base = f64::from(base_deg as f32).to_radians();
27        let mut last = base;
28        let mut rotations = Vec::new();
29        for &(time, pa) in self.samples.iter() {
30            let pa = f64::from(pa);
31            if (last - pa).abs() > self.step_rad {
32                rotations.push((time, base - pa));
33                last = pa;
34            }
35        }
36        rotations
37    }
38}
39
40/// The inverse pixel transform for an axis-aligned linear CF plane.
41pub(super) fn pixel_rotation(angle_rad: f64) -> [[f64; 2]; 2] {
42    let (sin, cos) = angle_rad.sin_cos();
43    [[cos, -sin], [sin, cos]]
44}

46#[cfg(test)]
47mod tests {
48    use super::*;
49
50    #[test]
51    fn rotation_samples_the_inverse_linear_coordinate_transform() {
52        let [[xx, xy], [yx, yy]] = pixel_rotation(std::f64::consts::FRAC_PI_2);
53        assert!(xx.abs() < 1.0e-15 && yy.abs() < 1.0e-15);
54        assert_eq!([xy, yx], [-1.0, 1.0]);
55    }
56
57    #[test]
58    fn rotation_epochs_use_the_previous_angle_and_rotate_the_base() {
59        let rotation = AwPaRotation {
60            step_rad: 0.125,
61            samples: Arc::new([(1.0, 0.125), (2.0, 0.25), (3.0, 0.375), (4.0, 0.5)]),
62        };
63        assert_eq!(rotation.schedule(0.0), [(2.0, -0.25), (4.0, -0.5)]);
64    }
