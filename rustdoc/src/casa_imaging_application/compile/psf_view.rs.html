25pub(crate) struct PsfView {
26    specification: ProblemSpecification,
27    geometry: GeometryInput,
28    windows: Option<Vec<Vec<SpectralWindowSelection>>>,
29    reservation: Reservation,

32impl PsfView {
33    pub(crate) fn compile(
34        self,
35        main: &CompiledProblem,
36        lifecycle: ModelLifecycleRequirements,
37    ) -> Result<Admitted<CompiledProblem>, crate::ApplicationDispatchError> {
38        let observation = match self.windows {
39            Some(windows) => alternate_observation(main.observation(), windows)
40                .map_err(|error| crate::ApplicationDispatchError::Preparation(error.into()))?,
41            None => main.observation().clone(),
42        };
43        Ok(Admitted {
44            value: casa_imaging_model::compile(ProblemInput::new(
45                self.specification,
46                self.geometry,
47                observation,
48                lifecycle,
49            ))
50            .map_err(crate::ApplicationDispatchError::Compile)?,
51            reservation: self.reservation,
52        })
53    }
54}

56/// CASA forms the alternate PSF as a new mosaic view of the same selection.
57/// The alternate phase law participates in row packing and convolution, not
58/// just the image header. The other products retain the main view.
59pub(super) fn prepare(
60    request: &ImagingRequest,
61    main: &PreparedImageDomain,
62    tracks_source: bool,
63    surveyed: &Surveyed<'_>,
64    context: &RunContext,
65    specification: &ProblemSpecification,
66    writer: &mut CasaImageWriter,
67) -> Result<Option<PsfView>, PrepareError> {
68    let Some(text) = request.psfphasecenter.as_deref() else {
69        return Ok(None);
70    };
71    if !request.calcpsf || !matches!(request.gridder, Gridder::Mosaic { .. }) {
72        return Ok(None);
73    }
74    // A single selected channel leaves only the tiny frame-conversion
75    // changes to determine the Taylor slope of the tracked continuum.
76    if tracks_source
77        && request.deconvolver == Deconvolver::Mtmfs
78        && surveyed
79            .spectral
80            .selected_source_channels
81            .iter()
82            .flat_map(|windows| windows.values())
83            .map(Vec::len)
84            .sum::<usize>()
85            == 1
86    {
87        return Err(PrepareError::MovingTaylorPsfSingleChannel);
88    }
89    let phase = direction::parse_phase_center_direction(text)?;
90    let (longitude, latitude) = (phase.longitude_rad(), phase.latitude_rad());
91    direction::refuse_ncp_at_equator(request.projection, latitude)?;
92    let direction = direction::direction_spec(
93        main.image_size,
94        request.cell,
95        request.projection,
96        phase.frame(),
97        [longitude, latitude],
98    );
99    // Cover both compiled and uncompiled single-domain metadata during
100    // conversion. The alternate view shares the observation manifest
101    // when frame conversion leaves the row and channel reads unchanged.
102    let variable = (request.restfreq.len() as u64 + request.scales.len() as u64) * 32
103        + (main.coordinates.obs_info().telescope.len() as u64
104            + main.coordinates.obs_info().observer.len() as u64)
105            * 2;
106    let cube = request.specmode == SpecMode::Cube;
107    let mut reservation = admit(
108        &context.host,
109        &context.policy,
110        &Demand {
111            phase: "separate mosaic PSF view",
112            memory: 64 * 1024 + variable,
113        },
114    )?;
115    let mut axis = None;
116    let mut windows = None;
117    if cube {
118        let mut cube_charge = admit_cube_axis(surveyed, context)?;
119        let alternate = alternate_axis(request, surveyed, phase, direction)?;
120        let changed_selection = alternate.selected_source_channels
121            != surveyed.spectral.selected_source_channels
122            || alternate.window_reads != surveyed.spectral.window_reads;
123        if changed_selection {
124            windows = Some(alternate_windows(surveyed, &alternate)?);
125            cube_charge.retain(alternate_selection_bytes(surveyed));
126            reservation.join(cube_charge);
127        }
128        axis = Some(alternate);
129    }
130    let spectral = axis.as_ref().unwrap_or(surveyed.spectral);
131    let coordinates = coordinates(request, main, direction, spectral);
132    let geometry = GeometryInput::new(
133        vec![main.spec().with_direction(direction)],
134        CentreLaws::new(
135            PhaseCentreLaw::Fixed(phase),
136            instrument::pointing_centre_law(&request.gridder),
137        ),
138        UvwCoordinateLaw::MosaicPhaseTrackingCentre,
139        spectral.coordinate(),
140    );
141    writer.set_main_psf_coordinates(coordinates);
142    Ok(Some(PsfView {
143        specification: specification.clone(),
144        geometry,
145        windows,
146        reservation,
147    }))
148}

150fn alternate_axis(
151    request: &ImagingRequest,
152    surveyed: &Surveyed<'_>,
153    phase: SkyDirection,
154    direction: DirectionCoordinateSpec,
155) -> Result<PreparedSpectralAxis<'static>, PrepareError> {
156    let frame = FrameContext {
157        anchor_time_mjd_seconds: surveyed.survey.image_time_mjd_seconds,
158        field_id: *surveyed.survey.fields.first().expect("selected field") as usize,
159        phase: MDirection::from_angles(
160            phase.longitude_rad(),
161            phase.latitude_rad(),
162            DirectionRef::J2000,
163        ),
164        direction,
165        engine: surveyed.engine,
166    };
167    let spectral = spectral::prepare_axis(
168        request,
169        (surveyed.stores, surveyed.reference),
170        None,
171        &frame,
172        None,
173        request.reffreq,
174    )?;
175    if spectral.output_channels != surveyed.spectral.output_channels {
176        return Err(PrepareError::AlternatePsfChannelCount {
177            main: surveyed.spectral.output_channels,
178            psf: spectral.output_channels,
179        });
180    }
181    Ok(spectral)
182}

199/// Validate the restart against the independently calculated alternate grid
200/// at the saved direction. Other products still use the main grid.
201pub(super) fn saved_cube_grid(
202    path: &std::path::Path,
203    request: &ImagingRequest,
204    main: &PreparedImageDomain,
205    surveyed: &Surveyed<'_>,
206    context: &RunContext,
207) -> Result<Option<RegridTemplate>, PrepareError> {
208    let image = PagedImage::<f32>::open(path).map_err(|error| PrepareError::Restart {
209        path: path.to_path_buf(),
210        reason: error.to_string(),
211    })?;
212    let actual = image.coordinates();
213    if actual.n_coordinates() == 0 {
214        return Ok(None);
215    }
216    let CoordinateModel::Direction(saved) = actual.coordinate(0) else {
217        return Ok(None);
218    };
219    if saved.direction_ref() != DirectionRef::J2000 {
220        return Ok(None);
221    }
222    let values = saved.reference_value();
223    let main_values = main.coordinates.coordinate(0).reference_value();
224    if main.direction.reference_direction().frame() == DirectionFrame::J2000
225        && values
226            .iter()
227            .zip(main_values)
228            .all(|(a, b)| (a - b).abs() <= 1e-13)
229    {
230        return Ok(None);
231    }
232    let phase = SkyDirection::new(DirectionFrame::J2000, values[0], values[1]);
233    direction::refuse_ncp_at_equator(request.projection, values[1])?;
234    let direction = direction::direction_spec(
235        main.image_size,
236        request.cell,
237        request.projection,
238        phase.frame(),
239        [values[0], values[1]],
240    );
241    let _axis = admit_cube_axis(surveyed, context)?;
242    let spectral = alternate_axis(request, surveyed, phase, direction)?;
243    let [nx, ny] = main.image_size;
244    Ok(Some(RegridTemplate {
245        coordinates: coordinates(request, main, direction, &spectral),
246        shape: vec![nx, ny, request.stokes.len(), spectral.output_channels],
247    }))
248}

250fn alternate_windows(
251    surveyed: &Surveyed<'_>,
252    spectral: &PreparedSpectralAxis<'_>,
253) -> Result<Vec<Vec<SpectralWindowSelection>>, PrepareError> {
254    surveyed
255        .stores
256        .iter()
257        .zip(spectral.selected_source_channels.iter())
258        .zip(&spectral.window_reads)
259        .map(|((store, channels), reads)| {
260            let factors = envelope::first_buffer_beam_factors(
261                (store.catalogue(), &store.engine),
262                &store.survey,
263                None,
264            )?;
265            selection::window_selections(&store.survey, channels, &factors, reads)
266        })
267        .collect()
268}

270fn alternate_observation(
271    main: &ObservationSnapshot,
272    windows: Vec<Vec<SpectralWindowSelection>>,
273) -> Result<ObservationSnapshot, casa_imaging_model::CompileObservationError> {
274    let sources = main
275        .sources()
276        .iter()
277        .zip(windows)
278        .map(|(source, windows)| {
279            let rows = source.selection();
280            ObservationSourceInput::new(
281                source.provenance().clone(),
282                ObservationSelection::new(
283                    rows.samples(),
284                    rows.rows_filter().clone(),
285                    rows.partitions().to_vec(),
286                    windows,
287                ),
288                source.columns(),
289                source.corrected_data_present(),
290            )
291        })
292        .collect();
293    compile_observation(ObservationSnapshotInput::new(sources))
294}

296fn alternate_selection_bytes(surveyed: &Surveyed<'_>) -> u64 {
297    surveyed
298        .stores
299        .iter()
300        .map(|store| {
301            let windows = &store.survey.spectral_windows;
302            let channels: usize = windows
303                .iter()
304                .map(|window| window.frequencies_hz.len())
305                .sum();
306            // Each window holds the catalog, three indexes and up to three
307            // interval lists, each bounded by the number of physical channels.
308            let channel_bytes = size_of::<f64>() + 3 * (size_of::<u32>() + size_of::<[f64; 2]>());
309            let partitions =
310                store.survey.partitions.len() * size_of::<casa_imaging_model::PartitionSelection>();
311            (windows.len() * size_of::<SpectralWindowSelection>()
312                + channels * channel_bytes
313                + partitions) as u64
314        })
315        .sum()
316}

318fn admit_cube_axis(
319    surveyed: &Surveyed<'_>,
320    context: &RunContext,
321) -> Result<Reservation, PrepareError> {
322    // The alternate axis builder uses the same surveyed inputs and channel
323    // scratch as the main cube, while the main specification is live.
324    let workspace: u64 = surveyed
325        .stores
326        .iter()
327        .map(|store| store.survey_charge.memory())
328        .sum();
329    Ok(admit(
330        &context.host,
331        &context.policy,
332        &Demand {
333            phase: "separate mosaic cube spectral axis",
334            memory: workspace + alternate_selection_bytes(surveyed),
335        },
336    )?)
