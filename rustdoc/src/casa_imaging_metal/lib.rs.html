4//! Metal implementation of the imaging gridding backend.
5//!
6//! This crate is the only place a device API appears in the imaging stack.
7//! [`MetalBackend`] implements the `casa-imaging-operator` backend trait for
8//! both tap layouts and every [`Work`](casa_imaging_operator::Work) variant
9//! with support-generic kernels: one SIMD group per sample, its lanes over
10//! the taps, `f32` grids updated with atomic adds (D2). The host locates
11//! every sample with the operator's rounding rule and accumulates `sumwt`
12//! in `f64`, so tap selection and `sumwt` equal the CPU backend's; only the
13//! order of the `f32` grid additions differs.
14//!
15//! Accumulators live in memory the device addresses
16//! ([`MetalBackend::accumulator`]); every dispatch completes before `apply`
17//! returns, so the host reads them like any other accumulator. Without a
18//! unified-memory Metal 3 device, constructors return
19//! [`DeviceFailure::Unavailable`](casa_imaging_operator::DeviceFailure).
20//!
21//! The design is `docs/imaging-architecture/imaging-foundation-plan-20261007.md`
22//! sections 5.3 and 5.9, ADR-0016, and the IF-4 deviations on #653.
23

38/// Ring slots of one backend: the host prepares one while the device runs
39/// the others.
40const RING: usize = 3;
41/// The most samples in one ring sub-block.
42const MAX_SUB: usize = 32_768;
43/// Device bytes per sample of a ring slot: the record, and per visibility
44/// polarization a value, a weight, an inverse norm and an output.
45const SAMPLE_BYTES: usize = 40;
46const POLARIZATION_BYTES: usize = 8 + 4 + 8 + 8;
47
48/// The most device memory one backend's ring holds for blocks of `npol`
49/// visibility polarizations; kernel tables add one copy of the taps of the
50/// cells the backend has met.
51#[must_use]
52pub const fn ring_bytes(npol: usize) -> u64 {
53    (RING * MAX_SUB * (SAMPLE_BYTES + npol * POLARIZATION_BYTES)) as u64
54}

56/// Whether this host has a Metal device the backend can use: Metal 3 with
57/// memory shared with the host.
58#[must_use]
59pub fn available() -> bool {
60    #[cfg(all(target_os = "macos", not(coverage)))]
61    {
62        device::Device::present()
63    }
64    #[cfg(not(all(target_os = "macos", not(coverage))))]
65    {
66        false
67    }
