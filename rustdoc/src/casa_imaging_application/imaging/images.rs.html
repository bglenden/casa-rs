15/// The pass images of domain `domain` in the normal state's x-major layout.
16///
17/// Data images become the residual planes `[channel][term][pol]`; on an
18/// initial pass the PSF moments, their `sumwt` and the `sumwt` CASA
19/// publishes for `basis` come along.
20pub(crate) fn pass_images(
21    domain: usize,
22    images: &NormalImages,
23    initial: bool,
24    basis: Basis,
25) -> PassImages {
26    let (height, width) = images.planes[0].data[0].dim();
27    let first = images.first_plane as usize;
28    let mut residual = Vec::with_capacity(images.planes.len() * images.data_terms * images.pols);
29    for plane in &images.planes {
30        for image in &plane.data {
31            push_x_major(&mut residual, image);
32        }
33    }
34    let (psf, sum_weights, published_sum_weights, weight) = if initial {
35        let mut psf = Vec::new();
36        let mut sum_weights = Vec::new();
37        let mut published_sum_weights = Vec::new();
38        let mut weight = Vec::new();
39        for (index, plane) in images.planes.iter().enumerate() {
40            for image in &plane.psf {
41                push_x_major(&mut psf, image);
42            }
43            // CASA's `.sumwt`, which `SIImageStore` divides the residual
44            // and the weight image by. `FTMachine::finalizeToSkyNew`
45            // writes it once, from the PSF gridding, and keeps it through
46            // the data passes, whose own sums it discards (they differ for
47            // a set whose PSF kernel is not its data kernel, `AWProjectFT`'s
48            // `cfwts2_p`). `MultiTermFTNew::finalizeToSkyNew` instead
49            // rewrites each term's `.sumwt` from every data pass, so a
50            // Taylor residual divides by the data gridding's sum while its
51            // weight keeps the PSF's.
52            let taylor = matches!(basis, Basis::Taylor { .. });
53            for term in 0..images.psf_terms {
54                for pol in 0..images.pols {
55                    sum_weights.push(images.psf_sumwt(index, term, pol));
56                    published_sum_weights.push(if taylor && term < images.data_terms {
57                        images.data_sumwt(index, term, pol)
58                    } else {
59                        images.psf_sumwt(index, term, pol)
60                    });
61                }
62            }
63            for image in &plane.weight {
64                push_x_major(&mut weight, image);
65            }
66        }
67        // A kernel set with weight taps (mosaic, AW) gridded one sensitivity
68        // image per polarization and plane; the standard sets gridded none.
69        let weight = (!weight.is_empty()).then_some(weight);
70        (Some(psf), sum_weights, published_sum_weights, weight)
71    } else {
72        (None, Vec::new(), Vec::new(), None)
73    };
74    PassImages {
75        domain,
76        shape: [width, height],
77        channels: first..first + images.planes.len(),
78        polarizations: images.pols,
79        residual,
80        psf,
81        sum_weights,
82        published_sum_weights,
83        weight,
84    }
85}

87fn push_x_major(out: &mut Vec<f32>, image: &Array2<f32>) {
88    let (height, width) = image.dim();
89    out.reserve(width * height);
90    for x in 0..width {
91        for y in 0..height {
92            out.push(image[(y, x)]);
93        }
94    }
95}

97/// Model images of `planes` on domain `domain` of `generation`: one plane
98/// per output channel of a channel-local basis, the Taylor terms of one
99/// plane otherwise. Pixels outside the model's support predict nothing.
100pub(crate) fn model_images(
101    generation: &ModelGeneration,
102    domain: usize,
103    planes: PlaneRange,
104    channel_local: bool,
105) -> Result<ModelImages, ImagingError> {
106    let shape = generation.shape();
107    let [width, height] = shape.domains()[domain].pixels();
108    let pols = shape.polarizations();
109    let terms = if channel_local {
110        1
111    } else {
112        shape.coefficients()
113    };
114    let mut model_planes = Vec::with_capacity(planes.len());
115    for plane in planes.start..planes.end {
116        let mut images = Vec::with_capacity(terms * pols);
117        for term in 0..terms {
118            let coefficient = if channel_local { plane as usize } else { term };
119            for pol in 0..pols {
120                let samples = generation.read_plane(domain, coefficient, pol)?;
121                images.push(Array2::from_shape_fn((height, width), |(y, x)| {
122                    let sample = samples[y * width + x];
123                    match sample.support() {
124                        ModelSupport::Valid => sample.value().value() as f32,
125                        ModelSupport::Invalid => 0.0,
126                    }
127                }));
128            }
129        }
130        model_planes.push(ModelPlane { images });
131    }
132    Ok(ModelImages {
133        first_plane: planes.start,
134        planes: model_planes,
135    })
136}

138/// Prepared model grids of `planes` of domain `domain`.
139pub(crate) fn prepare_model(
140    operator: &MeasurementOperator,
141    generation: &ModelGeneration,
142    domain: usize,
143    planes: PlaneRange,
144) -> Result<PreparedModelGrids, ImagingError> {
145    let channel_local = matches!(operator.basis(), Basis::ChannelLocal { .. });
146    let images = model_images(generation, domain, planes, channel_local)?;
147    Ok(operator.prepare_model(&images, ModelPrescale::Unit)?)
148}

150#[cfg(test)]
151mod tests {
152    use casa_imaging_operator::NormalPlane;
153
154    use super::*;
155
156    /// One plane of `data_terms` data and `psf_terms` PSF images with one
157    /// polarization and a weight image; `sumwt` in the plane's order.
158    fn images(data_terms: usize, psf_terms: usize, sumwt: Vec<f64>) -> NormalImages {
159        let image = || Array2::<f32>::zeros((2, 3));
160        NormalImages {
161            first_plane: 0,
162            pols: 1,
163            data_terms,
164            psf_terms,
165            planes: vec![NormalPlane {
166                data: (0..data_terms).map(|_| image()).collect(),
167                psf: (0..psf_terms).map(|_| image()).collect(),
168                weight: vec![image()],
169                sumwt,
170            }],
171        }
172    }
173
174    #[test]
175    fn the_published_sumwt_follows_the_psf_gridding_unless_the_basis_is_taylor() {
176        let constant = pass_images(0, &images(1, 1, vec![3.0, 8.0, 9.0]), true, Basis::Constant);
177        assert_eq!(constant.sum_weights, vec![8.0]);
178        assert_eq!(constant.published_sum_weights, vec![8.0]);
179        let taylor = pass_images(
180            0,
181            &images(2, 3, vec![3.0, 3.1, 8.0, 8.1, 8.2, 9.0]),
182            true,
183            Basis::Taylor {
184                terms: 2,
185                reference_hz: 1.0e9,
186            },
187        );
188        assert_eq!(taylor.sum_weights, vec![8.0, 8.1, 8.2]);
189        assert_eq!(taylor.published_sum_weights, vec![3.0, 3.1, 8.2]);
190        let residual = pass_images(0, &images(1, 0, vec![3.0, 9.0]), false, Basis::Constant);
191        assert!(residual.sum_weights.is_empty() && residual.published_sum_weights.is_empty());
192    }
