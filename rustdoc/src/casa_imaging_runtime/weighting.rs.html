61fn transformed_spectral_contributions(
62    problem: &CompiledProblem,
63    transformed: &ContinuumTransformedSample,
64) -> Result<SelectedSpectralContributions, WeightingError> {
65    if !transformed.use_role().contributes_to_output() {
66        return Ok(SelectedSpectralContributions::empty());
67    }
68    Ok(compile_spectral_stencil(
69        problem,
70        transformed.selected(),
71        transformed.spectral_evaluation(),
72    )?
73    .contributions()
74    .clone())

77fn density_spectral_contributions(
78    problem: &CompiledProblem,
79    reported: &SelectedObservationTraversalSample,
80    continuum: &SequentialContinuumTransform,
81) -> Result<SelectedSpectralContributions, ContinuumDensityCallbackError> {
82    if let Some(rule) = continuum.rule(
83        reported.selected().metadata.field_id,
84        reported.selected().address.spectral_window_id,
85    ) {
86        let use_role = rule
87            .channel_use(reported.selected().address.channel_index)
88            .ok_or(ContinuumTransformError::UndeclaredChannel)?;
89        if !use_role.contributes_to_output() {
90            return Ok(SelectedSpectralContributions::empty());
91        }
92    }
93    compiled_spectral_contributions(problem, reported).map_err(ContinuumDensityCallbackError::Owner)
94}

96fn consume_transformed_weighted<E>(
97    problem: &CompiledProblem,
98    stream: &mut FusedWeightingPhase,
99    transformed: ContinuumTransformedSample,
100    emit: &mut impl FnMut(&ReconstructionWeightedBlock) -> Result<(), E>,
101) -> Result<bool, ReplayCallbackError<E>> {
102    let contributions = transformed_spectral_contributions(problem, &transformed)
103        .map_err(ReplayCallbackError::Owner)?;
104    let has_spectral_support = !contributions.is_empty();
105    if let Some(block) = stream
106        .consume(problem, *transformed.selected(), contributions)
107        .map_err(ReplayCallbackError::Owner)?
108    {
109        emit(&block).map_err(ReplayCallbackError::Consumer)?;
110    }
111    Ok(has_spectral_support)
112}

342    fn compose_legacy(
343        &self,
344        base: &PhysicalWorkBinding,
345    ) -> Result<PhysicalWorkBinding, WeightingPlanFragmentError> {
346        let source = base
347            .execution_dag()
348            .nodes()
349            .get(&self.source_read)
350            .ok_or_else(|| WeightingPlanFragmentError::MissingNode(self.source_read.clone()))?;
351        if source.kind != WorkKind::ObservationRead {
352            return Err(WeightingPlanFragmentError::InvalidSourceKind(
353                self.source_read.clone(),
354            ));
355        }
356        let reconciliation_id = base
357            .observation_transaction()
358            .post_replay_reconciliation()
359            .expect("weighting composition requires reconstruction");
360        let model_preparation_id = base
361            .observation_transaction()
362            .final_model_preparation()
363            .cloned();
364        let commit_id = base.observation_transaction().commit();
365        let reconciliation = base
366            .execution_dag()
367            .nodes()
368            .get(reconciliation_id)
369            .ok_or_else(|| WeightingPlanFragmentError::MissingNode(reconciliation_id.clone()))?;
370        if !base.execution_dag().nodes().contains_key(commit_id) {
371            return Err(WeightingPlanFragmentError::MissingNode(commit_id.clone()));
372        }
373
374        let source_contract = SourceTraversalContract::from_source(
375            base,
376            source,
377            &self.source_resources.residency,
378            &self.source_resources.allocations,
379            &self.source_resources.queue,
380            &self.ids.release_node,
381        )?;
382        let io_lifetime = ClaimLifetime::through_fence(FenceKind::Io);
383        let read_claims = source_contract
384            .traversal_claims
385            .iter()
386            .chain(&source_contract.retained_claims)
387            .cloned()
388            .collect::<Vec<_>>();
389        let read_allocations = source_contract.allocations.clone();
390        let generation = WorkNode {
391            id: self.ids.generation_node.clone(),
392            kind: WorkKind::ObservationRead,
393            domain: WorkDomain::Io,
394            implementation: self.generation_implementation.clone(),
395            dependencies: terminal_events(source),
396            claims: read_claims.clone(),
397            allocations: read_allocations
398                .iter()
399                .cloned()
400                .chain([
401                    allocation_use(&self.ids.frozen_allocation, io_lifetime.clone()),
402                    allocation_use(&self.ids.partial_allocation, io_lifetime.clone()),
403                    allocation_use(&self.ids.reduction_allocation, io_lifetime.clone()),
404                ])
405                .collect(),
406            fences: BTreeSet::from([FenceKind::Io]),
407            quiescence_after: BTreeSet::new(),
408        };
409        let replay = WorkNode {
410            id: self.ids.replay_node.clone(),
411            kind: WorkKind::ObservationRead,
412            domain: WorkDomain::Io,
413            implementation: self.replay_implementation.clone(),
414            dependencies: std::iter::once(WorkDependency::Fence(FenceId::new(
415                self.ids.generation_node.clone(),
416                FenceKind::Io,
417            )))
418            .chain(
419                model_preparation_id
420                    .iter()
421                    .cloned()
422                    .map(WorkDependency::Work),
423            )
424            .collect(),
425            claims: read_claims.clone(),
426            allocations: read_allocations
427                .into_iter()
428                .chain([
429                    allocation_use(&self.ids.frozen_allocation, io_lifetime.clone()),
430                    allocation_use(&self.ids.replay_read_allocation, io_lifetime.clone()),
431                    allocation_use(&self.ids.weighted_block_allocation, io_lifetime.clone()),
432                ])
433                .chain(self.continuum_row_bytes.map(|_| {
434                    allocation_use(&self.ids.continuum_row_allocation, io_lifetime.clone())
435                }))
436                .collect(),
437            fences: BTreeSet::from([FenceKind::Io]),
438            quiescence_after: BTreeSet::new(),
439        };
440        let release_claims = std::iter::once(ResourceClaim {
441            resource: LeaseResource::Workers,
442            amount: 1,
443            lifetime: ClaimLifetime::Work,
444        })
445        .chain(source_contract.retained_claims.iter().cloned())
446        .chain(source_contract.release_buffer_claims.iter().cloned())
447        .collect();
448        let release = WorkNode {
449            id: self.ids.release_node.clone(),
450            kind: WorkKind::Release,
451            domain: WorkDomain::Cpu,
452            implementation: self.release_implementation.clone(),
453            dependencies: terminal_events(reconciliation),
454            claims: release_claims,
455            allocations: std::iter::once(allocation_use(
456                &self.ids.frozen_allocation,
457                ClaimLifetime::Work,
458            ))
459            .chain(
460                source_contract
461                    .retained_allocations
462                    .iter()
463                    .map(|allocation| allocation_use(allocation, ClaimLifetime::Work)),
464            )
465            .collect(),
466            fences: BTreeSet::new(),
467            quiescence_after: BTreeSet::new(),
468        };
469
470        let mut nodes = base
471            .execution_dag()
472            .nodes()
473            .values()
474            .cloned()
475            .collect::<Vec<_>>();
476        let source_node = nodes
477            .iter_mut()
478            .find(|node| node.id == self.source_read)
479            .ok_or_else(|| WeightingPlanFragmentError::MissingNode(self.source_read.clone()))?;
480        source_node.claims = read_claims.clone();
481        source_node
482            .allocations
483            .push(allocation_use(&self.ids.frozen_allocation, io_lifetime));
484        nodes
485            .iter_mut()
486            .find(|node| &node.id == reconciliation_id)
487            .ok_or_else(|| WeightingPlanFragmentError::MissingNode(reconciliation_id.clone()))?
488            .dependencies
489            .insert(WorkDependency::Fence(FenceId::new(
490                self.ids.replay_node.clone(),
491                FenceKind::Io,
492            )));
493        nodes
494            .iter_mut()
495            .find(|node| &node.id == commit_id)
496            .ok_or_else(|| WeightingPlanFragmentError::MissingNode(commit_id.clone()))?
497            .dependencies
498            .insert(WorkDependency::Work(self.ids.release_node.clone()));
499        nodes.extend([generation.clone(), replay.clone(), release.clone()]);
500
501        let allocation_specs = self.allocation_specs()?;
502        let mut alternative = base.execution_dag().resource_alternative().clone();
503        alternative.id = AlternativeId::new(format!(
504            "{}-weighting-{}",
505            alternative.id.as_str(),
506            self.plan.commitment_id()
507        ));
508        alternative
509            .demand
510            .memory
511            .extend(allocation_specs.iter().map(AllocationSpec::memory_demand));
512        let dag = ExecutionDag::new(ExecutionDagSpecification {
513            required_resource_capabilities: base
514                .execution_dag()
515                .required_resource_capabilities()
516                .clone(),
517            resource_alternative: alternative,
518            nodes,
519            logical_allocations: base
520                .execution_dag()
521                .logical_allocations()
522                .values()
523                .cloned()
524                .map(|mut allocation| {
525                    if source_contract
526                        .retained_allocations
527                        .contains(&allocation.id)
528                    {
529                        allocation.lifetime.release_after =
530                            BTreeSet::from([WorkDependency::Work(self.ids.release_node.clone())]);
531                    } else if source_contract.allocation_ids.contains(&allocation.id) {
532                        allocation
533                            .lifetime
534                            .release_after
535                            .insert(WorkDependency::Fence(FenceId::new(
536                                self.ids.replay_node.clone(),
537                                FenceKind::Io,
538                            )));
539                    }
540                    allocation
541                })
542                .chain(
543                    allocation_specs
544                        .iter()
545                        .map(AllocationSpec::logical_allocation),
546                )
547                .collect(),
548            physical_slots: base
549                .execution_dag()
550                .physical_slots()
551                .values()
552                .cloned()
553                .chain(allocation_specs.iter().map(AllocationSpec::physical_slot))
554                .collect(),
555            initial_knobs: base.execution_dag().initial_knobs().clone(),
556            adaptations: base
557                .execution_dag()
558                .adaptations()
559                .values()
560                .cloned()
561                .collect(),
562        })?;
563
564        let source_prediction = base
565            .prediction()
566            .stages()
567            .get(&self.source_read)
568            .ok_or_else(|| WeightingPlanFragmentError::MissingNode(self.source_read.clone()))?;
569        let generation_prediction = scaled_prediction(source_prediction, generation.id, 2)?;
570        let replay_prediction = scaled_prediction(source_prediction, replay.id, 1)?;
571        let release_prediction =
572            StagePrediction::new(release.id, 0).with_io(vec![IoPrediction::new(
573                IoBufferKind::SourceReadAhead,
574                u64::try_from(self.source_resources.residency.aggregate_resident_bytes())
575                    .map_err(|_| WeightingPlanFragmentError::PredictionOverflow)?,
576                1,
577            )]);
578        let extra_elapsed = generation_prediction
579            .elapsed_nanos()
580            .checked_add(replay_prediction.elapsed_nanos())
581            .ok_or(WeightingPlanFragmentError::PredictionOverflow)?;
582        let prediction = PlanPrediction::new(
583            base.prediction()
584                .elapsed_nanos()
585                .checked_add(extra_elapsed)
586                .ok_or(WeightingPlanFragmentError::PredictionOverflow)?,
587            base.prediction().confidence(),
588            base.prediction().uncertainty().to_vec(),
589            base.prediction()
590                .stages()
591                .values()
592                .cloned()
593                .chain([generation_prediction, replay_prediction, release_prediction])
594                .collect(),
595        )?;
596        Ok(PhysicalWorkBinding::with_implementation_contract(
597            base.implementation_contract().for_execution_dag(&dag)?,
598            dag,
599            prediction,
600            base.artifacts().to_vec(),
601            base.observation_transaction().clone(),
602            base.publication_layouts().clone(),
603            base.product_publication_authority(),
604        )?)
605    }

607    fn compose_streaming(
608        &self,
609        base: &PhysicalWorkBinding,
610        mode: WeightingStreamingMode,
611    ) -> Result<PhysicalWorkBinding, WeightingPlanFragmentError> {
612        let legacy = self.compose_legacy(base)?;
613        let terminal = self.streaming_node().clone();
614        let removed = match mode {
615            WeightingStreamingMode::DensityInitial => {
616                BTreeSet::from([self.ids.replay_node.clone()])
617            }
618            WeightingStreamingMode::NaturalInitial | WeightingStreamingMode::Reuse => {
619                BTreeSet::from([
620                    self.ids.generation_node.clone(),
621                    self.ids.replay_node.clone(),
622                ])
623            }
624        };
625        let extra_allocations = legacy
626            .execution_dag()
627            .nodes()
628            .iter()
629            .filter(|(id, _)| removed.contains(*id))
630            .flat_map(|(_, node)| node.allocations.iter().cloned())
631            .collect::<Vec<_>>();
632        let model_preparation = base
633            .observation_transaction()
634            .final_model_preparation()
635            .cloned();
636        let reconciliation = base
637            .observation_transaction()
638            .post_replay_reconciliation()
639            .expect("streaming weighting composition requires reconstruction");
640        let mut nodes = legacy
641            .execution_dag()
642            .nodes()
643            .values()
644            .filter(|node| !removed.contains(&node.id))
645            .cloned()
646            .collect::<Vec<_>>();
647        for node in &mut nodes {
648            node.dependencies.retain(|dependency| match dependency {
649                WorkDependency::Work(id) => !removed.contains(id),
650                WorkDependency::Fence(fence) => !removed.contains(fence.node()),
651            });
652            if node.id == terminal {
653                if let Some(preparation) = &model_preparation {
654                    node.dependencies
655                        .insert(WorkDependency::Work(preparation.clone()));
656                }
657                node.allocations.extend(extra_allocations.iter().cloned());
658                node.allocations
659                    .sort_by(|left, right| left.allocation.as_str().cmp(right.allocation.as_str()));
660                node.allocations
661                    .dedup_by(|left, right| left.allocation == right.allocation);
662            }
663            if &node.id == reconciliation {
664                node.dependencies.insert(WorkDependency::Fence(FenceId::new(
665                    terminal.clone(),
666                    FenceKind::Io,
667                )));
668            }
669        }
670        let terminal_fence = WorkDependency::Fence(FenceId::new(terminal.clone(), FenceKind::Io));
671        let allocations = legacy
672            .execution_dag()
673            .logical_allocations()
674            .values()
675            .cloned()
676            .map(|mut allocation| {
677                if removed.contains(&allocation.lifetime.acquire_at) {
678                    allocation.lifetime.acquire_at = terminal.clone();
679                }
680                let mut release_after = BTreeSet::new();
681                for dependency in allocation.lifetime.release_after {
682                    let dependency = match dependency {
683                        WorkDependency::Work(id) if removed.contains(&id) => terminal_fence.clone(),
684                        WorkDependency::Fence(fence) if removed.contains(fence.node()) => {
685                            terminal_fence.clone()
686                        }
687                        other => other,
688                    };
689                    release_after.insert(dependency);
690                }
691                allocation.lifetime.release_after = release_after;
692                allocation
693            })
694            .collect();
695        let dag = ExecutionDag::new(ExecutionDagSpecification {
696            required_resource_capabilities: legacy
697                .execution_dag()
698                .required_resource_capabilities()
699                .clone(),
700            resource_alternative: legacy.execution_dag().resource_alternative().clone(),
701            nodes,
702            logical_allocations: allocations,
703            physical_slots: legacy
704                .execution_dag()
705                .physical_slots()
706                .values()
707                .cloned()
708                .collect(),
709            initial_knobs: legacy.execution_dag().initial_knobs().clone(),
710            adaptations: legacy
711                .execution_dag()
712                .adaptations()
713                .values()
714                .cloned()
715                .collect(),
716        })?;
717        let stages = legacy
718            .prediction()
719            .stages()
720            .values()
721            .filter(|stage| !removed.contains(stage.node()))
722            .cloned()
723            .map(|stage| {
724                if mode == WeightingStreamingMode::DensityInitial
725                    && stage.node() == &self.ids.generation_node
726                {
727                    scaled_prediction(
728                        base.prediction()
729                            .stages()
730                            .get(&self.source_read)
731                            .expect("source stage"),
732                        self.ids.generation_node.clone(),
733                        1,
734                    )
735                    .expect("one-pass prediction")
736                } else {
737                    stage
738                }
739            })
740            .collect::<Vec<_>>();
741        let elapsed_nanos = stages
742            .iter()
743            .try_fold(0_u64, |total, stage| {
744                total.checked_add(stage.elapsed_nanos())
745            })
746            .ok_or(WeightingPlanFragmentError::PredictionOverflow)?;
747        let prediction = PlanPrediction::new(
748            elapsed_nanos,
749            legacy.prediction().confidence(),
750            legacy.prediction().uncertainty().to_vec(),
751            stages,
752        )?;
753        Ok(PhysicalWorkBinding::with_implementation_contract(
754            legacy.implementation_contract().for_execution_dag(&dag)?,
755            dag,
756            prediction,
757            legacy.artifacts().to_vec(),
758            legacy.observation_transaction().clone(),
759            legacy.publication_layouts().clone(),
760            legacy.product_publication_authority(),
761        )?)
762    }

764    fn allocation_specs(&self) -> Result<Vec<AllocationSpec>, WeightingPlanFragmentError> {
765        let residency = self.plan.planned_residency();
766        let frozen_bytes = checked_sum([
767            residency.density_grid_bytes(),
768            residency.robust_factor_bytes(),
769            residency.sum_weight_bytes(),
770        ])?;

778        ))]);

780            AllocationSpec::new(

784                "weighting-frozen-generation",
785                self.source_read.clone(),

792                "weighting-density-partials",

795            )?,

797                self.ids.reduction_allocation.clone(),
798                self.ids.reduction_slot.clone(),
799                residency.reduction_scratch_bytes(),
800                "weighting-exact-reduction",
801                self.ids.generation_node.clone(),
802                generation_fence,
803            )?,
804            AllocationSpec::new(
805                self.ids.replay_read_allocation.clone(),
806                self.ids.replay_read_slot.clone(),
807                residency.replay_read_bytes(),
808                "weighting-replay-read",
809                self.ids.replay_node.clone(),
810                replay_fence.clone(),
811            )?,

813                self.ids.weighted_block_allocation.clone(),

817                self.ids.replay_node.clone(),

830            )?);
831        }
832        Ok(specs)
833    }

981        &self,
982        context: WorkExecutionContext<'_>,
983    ) -> Result<(), WeightingEvidenceError> {
984        let specs = self
985            .allocation_specs()
986            .map_err(|_| WeightingEvidenceError)?;
987        validate_work_authority(
988            context,
989            &self.ids.release_node,
990            &[&specs[0]],
991            WeightingWorkContract::Release,
992        )?;
993        let expected_bytes =
994            u64::try_from(self.source_resources.residency.aggregate_resident_bytes())
995                .map_err(|_| WeightingEvidenceError)?;
996        let selected = context
997            .allocations()
998            .iter()
999            .filter(|capability| {
1000                self.source_resources
1001                    .allocations
1002                    .contains(capability.allocation())
1003            })
1004            .collect::<Vec<_>>();
1005        let selected_bytes = selected
1006            .iter()
1007            .try_fold(0_u64, |total, capability| {
1008                total.checked_add(capability.capacity_bytes())
1009            })
1010            .ok_or(WeightingEvidenceError)?;
1011        let read_buffer_bytes = context
1012            .resources()
1013            .iter()
1014            .filter(|capability| {
1015                capability.resource() == &LeaseResource::IoBuffer(IoBufferKind::SourceReadAhead)
1016                    && capability.lifetime() == &ClaimLifetime::Work
1017            })
1018            .try_fold(0_u64, |total, capability| {
1019                total.checked_add(capability.amount())
1020            })
1021            .ok_or(WeightingEvidenceError)?;
1022        if selected.len() != self.source_resources.allocations.len()
1023            || selected_bytes != expected_bytes
1024            || read_buffer_bytes != expected_bytes
1025        {
1026            return Err(WeightingEvidenceError);
1027        }
1028        Ok(())
1029    }

1169        }

1248    fn traverse_initial_stream_impl<E>(
1249        &mut self,
1250        context: WorkExecutionContext<'_>,
1251        fragment: &WeightingPlanFragment<'_>,
1252        problem: &CompiledProblem,
1253        selected: Option<BoundSelectedObservation>,
1254        continuum: Option<&SequentialContinuumTransform>,
1255        mut emit: impl FnMut(&ReconstructionWeightedBlock) -> Result<(), E>,
1256    ) -> Result<(), WeightingReplayError<E>>
1257    where
1258        E: Error + 'static,
1259    {
1260        if !matches!(self.phase, WeightingExecutionPhase::Empty) {
1261            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1262        }
1263        let (mut selected, mut stream, binding) = match fragment.streaming {
1264            Some(WeightingStreamingMode::NaturalInitial) => {
1265                let selected =
1266                    selected.ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1267                fragment
1268                    .authorize_source_observation(
1269                        context,
1270                        problem,
1271                        selected.residency_certificate(),
1272                    )
1273                    .map_err(WeightingReplayError::Evidence)?;
1274                let stream = begin_natural_weighting_stream(problem, fragment.plan)
1275                    .map_err(WeightingReplayError::Owner)?;
1276                (selected, stream, None)
1277            }
1278            Some(WeightingStreamingMode::DensityInitial) => {
1279                if selected.is_some() {
1280                    return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1281                }
1282                let binding = fragment
1283                    .authorize_generation(context, problem)
1284                    .map_err(WeightingReplayError::Evidence)?;
1285                let retained = self
1286                    .retained_observation
1287                    .take()
1288                    .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1289                if retained.attempt_id != context.attempt_id()
1290                    || retained.lease_epoch != context.lease_epoch()
1291                {
1292                    return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1293                }
1294                let density = self
1295                    .density
1296                    .take()
1297                    .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1298                let stream = density
1299                    .finish_into_stream(problem, fragment.plan)
1300                    .map_err(WeightingReplayError::Owner)?;
1301                (retained.selected, stream, Some(binding))
1302            }
1303            Some(WeightingStreamingMode::Reuse) | None => {
1304                return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1305            }
1306        };
1307        let mut continuum_stream = match continuum {
1308            Some(contract) => {
1309                let plan = plan_continuum_transform_row(problem)
1310                    .map_err(WeightingReplayError::Transform)?
1311                    .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1312                Some(
1313                    ContinuumTransformStream::new(contract, plan)
1314                        .map_err(WeightingReplayError::Transform)?,
1315                )
1316            }
1317            None => None,
1318        };
1319        let mut spectral_support_sample_count = 0_u64;
1320        let owner_completion = selected
1321            .traverse(problem, |reported| {
1322                if let Some(transform) = &mut continuum_stream {
1323                    let completed = transform
1324                        .push(*reported.selected(), reported.spectral_evaluation())
1325                        .map_err(ReplayCallbackError::Transform)?;
1326                    for transformed in completed {
1327                        if consume_transformed_weighted(
1328                            problem,
1329                            &mut stream,
1330                            transformed,
1331                            &mut emit,
1332                        )? {
1333                            spectral_support_sample_count += 1;
1334                        }
1335                    }
1336                } else {
1337                    let contributions = compiled_spectral_contributions(problem, &reported)
1338                        .map_err(ReplayCallbackError::Owner)?;
1339                    if let Some(block) = stream
1340                        .consume(problem, *reported.selected(), contributions)
1341                        .map_err(ReplayCallbackError::Owner)?
1342                    {
1343                        emit(&block).map_err(ReplayCallbackError::Consumer)?;
1344                    }
1345                }
1346                Ok(())
1347            })
1348            .map_err(WeightingReplayError::Traversal)?;
1349        let continuum_completion = if let Some(mut transform) = continuum_stream {
1350            for transformed in transform
1351                .finish_rows()
1352                .map_err(WeightingReplayError::Transform)?
1353            {
1354                if consume_transformed_weighted(problem, &mut stream, transformed, &mut emit)
1355                    .map_err(|error| match error {
1356                        ReplayCallbackError::Owner(error) => WeightingReplayError::Owner(error),
1357                        ReplayCallbackError::Transform(error) => {
1358                            WeightingReplayError::Transform(error)
1359                        }
1360                        ReplayCallbackError::Consumer(error) => {
1361                            WeightingReplayError::Consumer(error)
1362                        }
1363                    })?
1364                {
1365                    spectral_support_sample_count = spectral_support_sample_count
1366                        .checked_add(1)
1367                        .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1368                }
1369            }
1370            Some(
1371                transform
1372                    .complete(owner_completion.generation_id())
1373                    .map_err(WeightingReplayError::Transform)?,
1374            )
1375        } else {
1376            None
1377        };
1378        let (final_block, state, summary) = stream.finish().map_err(WeightingReplayError::Owner)?;
1379        if let Some(block) = final_block {
1380            emit(&block).map_err(WeightingReplayError::Consumer)?;
1381        }
1382        let binding = binding.unwrap_or_else(|| WeightingGenerationBinding {
1383            attempt_id: context.attempt_id(),
1384            owner_node: context.node().id.clone(),
1385            lease_epoch: context.lease_epoch(),
1386            source_generation: owner_completion.generation_id(),
1387            source_sample_count: owner_completion.sample_count(),
1388        });
1389        if state.sample_count() != owner_completion.sample_count()
1390            || owner_completion.generation_id() != binding.source_generation
1391            || owner_completion.sample_count() != binding.source_sample_count
1392        {
1393            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1394        }
1395        let artifact = FrozenWeightingArtifact {
1396            state: Arc::new(state),
1397            source_generation: owner_completion.generation_id(),
1398            source_sample_count: owner_completion.sample_count(),
1399            continuum_transform: continuum_completion,
1400            cross_plan_reservation: None,
1401        };
1402        let frozen = FrozenWeightingGeneration {
1403            artifact,
1404            binding: WeightingGenerationBinding {
1405                attempt_id: binding.attempt_id,
1406                owner_node: binding.owner_node.clone(),
1407                lease_epoch: binding.lease_epoch,
1408                source_generation: binding.source_generation,
1409                source_sample_count: binding.source_sample_count,
1410            },
1411        };
1412        self.retained_observation = Some(RetainedWeightingObservation {
1413            selected,
1414            attempt_id: context.attempt_id(),
1415            owner_node: fragment.source_read.clone(),
1416            lease_epoch: context.lease_epoch(),
1417        });
1418        self.phase = WeightingExecutionPhase::PendingReplay {
1419            frozen,
1420            pending: PendingWeightingReplay {
1421                state: summary,
1422                owner_completion,
1423                binding,
1424                continuum_transform: continuum_completion,
1425                spectral_support_sample_count,
1426            },
1427        };
1428        Ok(())
1429    }

1453    fn traverse_reuse_stream_impl<E>(
1454        &mut self,
1455        context: WorkExecutionContext<'_>,
1456        fragment: &WeightingPlanFragment<'_>,
1457        mut selected: BoundSelectedObservation,
1458        problem: &CompiledProblem,
1459        continuum: Option<&SequentialContinuumTransform>,
1460        mut emit: impl FnMut(&ReconstructionWeightedBlock) -> Result<(), E>,
1461    ) -> Result<(), WeightingReplayError<E>>
1462    where
1463        E: Error + 'static,
1464    {
1465        if !matches!(self.phase, WeightingExecutionPhase::Empty)
1466            || self.retained_observation.is_some()
1467            || self.density.is_some()
1468            || fragment.streaming != Some(WeightingStreamingMode::Reuse)
1469        {
1470            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1471        }
1472        fragment
1473            .authorize_source_observation(context, problem, selected.residency_certificate())
1474            .map_err(WeightingReplayError::Evidence)?;
1475        let artifact = self
1476            .imported
1477            .take()
1478            .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1479        let mut replay = artifact
1480            .state
1481            .begin_replay(problem, fragment.plan)
1482            .map_err(WeightingReplayError::Owner)?;
1483        let mut continuum_stream = match continuum {
1484            Some(contract) => {
1485                let plan = plan_continuum_transform_row(problem)
1486                    .map_err(WeightingReplayError::Transform)?
1487                    .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1488                Some(
1489                    ContinuumTransformStream::new(contract, plan)
1490                        .map_err(WeightingReplayError::Transform)?,
1491                )
1492            }
1493            None => None,
1494        };
1495        let mut spectral_support_sample_count = 0_u64;
1496        let owner_completion = selected
1497            .traverse(problem, |reported| {
1498                if let Some(transform) = &mut continuum_stream {
1499                    let completed = transform
1500                        .push(*reported.selected(), reported.spectral_evaluation())
1501                        .map_err(ReplayCallbackError::Transform)?;
1502                    for transformed in completed {
1503                        let contributions =
1504                            transformed_spectral_contributions(problem, &transformed)
1505                                .map_err(ReplayCallbackError::Owner)?;
1506                        if !contributions.is_empty() {
1507                            spectral_support_sample_count += 1;
1508                        }
1509                        if let Some(block) = replay
1510                            .consume(problem, *transformed.selected(), contributions)
1511                            .map_err(ReplayCallbackError::Owner)?
1512                        {
1513                            emit(&block).map_err(ReplayCallbackError::Consumer)?;
1514                        }
1515                    }
1516                } else {
1517                    let contributions = compiled_spectral_contributions(problem, &reported)
1518                        .map_err(ReplayCallbackError::Owner)?;
1519                    if let Some(block) = replay
1520                        .consume(problem, *reported.selected(), contributions)
1521                        .map_err(ReplayCallbackError::Owner)?
1522                    {
1523                        emit(&block).map_err(ReplayCallbackError::Consumer)?;
1524                    }
1525                }
1526                Ok(())
1527            })
1528            .map_err(WeightingReplayError::Traversal)?;
1529        let continuum_completion = if let Some(mut transform) = continuum_stream {
1530            for transformed in transform
1531                .finish_rows()
1532                .map_err(WeightingReplayError::Transform)?
1533            {
1534                let contributions = transformed_spectral_contributions(problem, &transformed)
1535                    .map_err(WeightingReplayError::Owner)?;
1536                if !contributions.is_empty() {
1537                    spectral_support_sample_count = spectral_support_sample_count
1538                        .checked_add(1)
1539                        .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
1540                }
1541                if let Some(block) = replay
1542                    .consume(problem, *transformed.selected(), contributions)
1543                    .map_err(WeightingReplayError::Owner)?
1544                {
1545                    emit(&block).map_err(WeightingReplayError::Consumer)?;
1546                }
1547            }
1548            Some(
1549                transform
1550                    .complete(owner_completion.generation_id())
1551                    .map_err(WeightingReplayError::Transform)?,
1552            )
1553        } else {
1554            None
1555        };
1556        if owner_completion.generation_id() != artifact.source_generation
1557            || owner_completion.sample_count() != artifact.source_sample_count
1558            || continuum_completion != artifact.continuum_transform
1559        {
1560            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1561        }
1562        let (final_block, summary) = replay.finish().map_err(WeightingReplayError::Owner)?;
1563        if let Some(block) = final_block {
1564            emit(&block).map_err(WeightingReplayError::Consumer)?;
1565        }
1566        let binding = WeightingGenerationBinding {
1567            attempt_id: context.attempt_id(),
1568            owner_node: context.node().id.clone(),
1569            lease_epoch: context.lease_epoch(),
1570            source_generation: owner_completion.generation_id(),
1571            source_sample_count: owner_completion.sample_count(),
1572        };
1573        self.retained_observation = Some(RetainedWeightingObservation {
1574            selected,
1575            attempt_id: context.attempt_id(),
1576            owner_node: fragment.source_read.clone(),
1577            lease_epoch: context.lease_epoch(),
1578        });
1579        self.phase = WeightingExecutionPhase::PendingReplay {
1580            frozen: FrozenWeightingGeneration {
1581                artifact,
1582                binding: WeightingGenerationBinding {
1583                    attempt_id: binding.attempt_id,
1584                    owner_node: binding.owner_node.clone(),
1585                    lease_epoch: binding.lease_epoch,
1586                    source_generation: binding.source_generation,
1587                    source_sample_count: binding.source_sample_count,
1588                },
1589            },
1590            pending: PendingWeightingReplay {
1591                state: summary,
1592                owner_completion,
1593                binding,
1594                continuum_transform: continuum_completion,
1595                spectral_support_sample_count,
1596            },
1597        };
1598        Ok(())
1599    }

1602    pub fn traverse_generation(
1603        &mut self,
1604        context: WorkExecutionContext<'_>,
1605        fragment: &WeightingPlanFragment<'_>,
1606        problem: &CompiledProblem,
1607    ) -> Result<(), WeightingGenerationError> {
1608        if !matches!(self.phase, WeightingExecutionPhase::Empty) {
1609            return Err(WeightingGenerationError::Evidence(WeightingEvidenceError));
1610        }
1611        let retained = self
1612            .retained_observation
1613            .as_mut()
1614            .ok_or(WeightingGenerationError::Evidence(WeightingEvidenceError))?;
1615        if retained.attempt_id != context.attempt_id()
1616            || retained.owner_node != fragment.source_read
1617            || retained.lease_epoch != context.lease_epoch()
1618        {
1619            return Err(WeightingGenerationError::Evidence(WeightingEvidenceError));
1620        }
1621        self.phase = WeightingExecutionPhase::PendingGeneration(Box::new(
1622            traverse_weighting_generation(context, fragment, &mut retained.selected, problem)?,
1623        ));
1624        Ok(())
1625    }

1628    pub fn complete_generation(
1629        &mut self,

1631    ) -> Result<AttemptBoundObservationCompletion, WeightingGenerationCompletionError> {

1644    /// Drive the third exhaustive traversal while retaining the frozen W.
1645    pub fn traverse_replay<E>(
1646        &mut self,
1647        context: WorkExecutionContext<'_>,
1648        fragment: &WeightingPlanFragment<'_>,
1649        problem: &CompiledProblem,
1650        emit: impl FnMut(&WeightedObservationBlock) -> Result<(), E>,
1651    ) -> Result<(), WeightingReplayError<E>>
1652    where
1653        E: Error + 'static,
1654    {
1655        let phase = std::mem::take(&mut self.phase);
1656        let WeightingExecutionPhase::Frozen(frozen) = phase else {
1657            self.phase = phase;
1658            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1659        };
1660        let Some(retained) = self.retained_observation.as_mut() else {
1661            self.phase = WeightingExecutionPhase::Frozen(frozen);
1662            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1663        };
1664        if retained.attempt_id != context.attempt_id()
1665            || retained.owner_node != fragment.source_read
1666            || retained.lease_epoch != context.lease_epoch()
1667        {
1668            self.phase = WeightingExecutionPhase::Frozen(frozen);
1669            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
1670        }
1671        match frozen.replay(context, fragment, &mut retained.selected, problem, emit) {
1672            Ok(pending) => {
1673                self.phase = WeightingExecutionPhase::PendingReplay { frozen, pending };
1674                Ok(())
1675            }
1676            Err(error) => {
1677                self.phase = WeightingExecutionPhase::Frozen(frozen);
1678                Err(error)
1679            }
1680        }
1681    }

1683    /// Bind a successfully settled replay fence and retain its terminal proof.
1684    pub fn complete_replay(
1685        &mut self,
1686        context: ObservationReadCompletionContext,
1687    ) -> Result<AttemptBoundObservationCompletion, WeightingReplayCompletionError> {
1688        let phase = std::mem::take(&mut self.phase);
1689        let WeightingExecutionPhase::PendingReplay { frozen, pending } = phase else {
1690            self.phase = phase;
1691            return Err(WeightingReplayCompletionError::Evidence(
1692                WeightingEvidenceError,
1693            ));
1694        };
1695        match pending.bind(context) {
1696            Ok((completion, predecessor)) => {
1697                self.phase = WeightingExecutionPhase::Replayed { frozen, completion };
1698                Ok(predecessor)
1699            }
1700            Err(error) => {
1701                self.phase = WeightingExecutionPhase::Frozen(frozen);
1702                Err(error)
1703            }
1704        }
1705    }

1707    /// Return the terminal replay proof retained for reconciliation.
1708    #[must_use]
1709    pub const fn replay_completion(&self) -> Option<&WeightingReplayCompletion> {
1710        match &self.phase {
1711            WeightingExecutionPhase::Replayed { completion, .. } => Some(completion),
1712            WeightingExecutionPhase::Empty
1713            | WeightingExecutionPhase::PendingGeneration(_)
1714            | WeightingExecutionPhase::Frozen(_)
1715            | WeightingExecutionPhase::PendingReplay { .. } => None,

1719    /// Drop every retained phase only from the scheduler-issued Release node.
1720    ///
1721    /// A success-path release requires a completed replay. A draining release
1722    /// accepts any phase, including the pending values held between work and
1723    /// fence completion, after validating its attempt, lease, and allocation.
1724    pub fn release(
1725        &mut self,
1726        context: WorkExecutionContext<'_>,
1727        fragment: &WeightingPlanFragment<'_>,
1728    ) -> Result<(), WeightingEvidenceError> {
1729        fragment.authorize_release(context)?;
1730        if !self.matches_attempt(context, fragment)
1731            || !context.is_cleanup()
1732                && !matches!(self.phase, WeightingExecutionPhase::Replayed { .. })
1733        {
1734            return Err(WeightingEvidenceError);
1735        }
1736        self.phase = WeightingExecutionPhase::Empty;
1737        self.retained_observation = None;
1738        self.density = None;
1739        self.imported = None;
1740        Ok(())
1741    }

1743    /// Return whether the planned release has consumed all externally retained state.
1744    #[must_use]
1745    pub const fn is_empty(&self) -> bool {
1746        matches!(self.phase, WeightingExecutionPhase::Empty)
1747            && self.retained_observation.is_none()
1748            && self.density.is_none()
1749            && self.imported.is_none()
1750    }

1752    /// Return whether the actual read-locked selected-observation owner is live.
1753    #[must_use]
1754    pub const fn has_retained_observation(&self) -> bool {
1755        self.retained_observation.is_some()
1756    }

1758    fn matches_attempt(
1759        &self,
1760        context: WorkExecutionContext<'_>,
1761        fragment: &WeightingPlanFragment<'_>,
1762    ) -> bool {
1763        if self.retained_observation.as_ref().is_some_and(|retained| {
1764            retained.attempt_id != context.attempt_id()
1765                || retained.owner_node != fragment.source_read
1766                || retained.lease_epoch != context.lease_epoch()
1767        }) {
1768            return false;
1769        }
1770        let valid_binding = |binding: &WeightingGenerationBinding, owner: &WorkNodeId| {
1771            binding.attempt_id == context.attempt_id()
1772                && binding.lease_epoch == context.lease_epoch()
1773                && &binding.owner_node == owner
1774        };
1775        let (generation_owner, replay_owner) = match fragment.streaming {
1776            Some(_) => (fragment.streaming_node(), fragment.streaming_node()),
1777            None => (fragment.generation_node(), fragment.replay_node()),
1778        };
1779        match &self.phase {
1780            WeightingExecutionPhase::Empty => context.is_cleanup(),
1781            WeightingExecutionPhase::PendingGeneration(pending) => {
1782                valid_binding(&pending.binding, fragment.generation_node())
1783                    && pending.state.commitment_id() == fragment.plan.commitment_id()
1784            }
1785            WeightingExecutionPhase::Frozen(frozen) => {
1786                valid_binding(&frozen.binding, generation_owner)
1787                    && frozen.artifact.state.commitment_id() == fragment.plan.commitment_id()
1788            }
1789            WeightingExecutionPhase::PendingReplay { frozen, pending } => {
1790                valid_binding(&frozen.binding, generation_owner)
1791                    && valid_binding(&pending.binding, replay_owner)
1792                    && frozen.artifact.state.commitment_id() == fragment.plan.commitment_id()
1793                    && pending.state.weighting_generation() == frozen.artifact.state.generation_id()
1794            }
1795            WeightingExecutionPhase::Replayed { frozen, completion } => {
1796                valid_binding(&frozen.binding, generation_owner)
1797                    && valid_binding(&completion.binding, replay_owner)
1798                    && frozen.artifact.state.commitment_id() == fragment.plan.commitment_id()
1799                    && completion.state.weighting_generation()
1800                        == frozen.artifact.state.generation_id()
1801            }
1802        }
1803    }
1804}

1857struct SourceTraversalContract {
1858    traversal_claims: Vec<ResourceClaim>,
1859    retained_claims: Vec<ResourceClaim>,
1860    release_buffer_claims: Vec<ResourceClaim>,

1862    allocation_ids: BTreeSet<AllocationId>,

1865
1866impl SourceTraversalContract {
1867    fn from_source(
1868        base: &PhysicalWorkBinding,
1869        source: &WorkNode,
1870        residency: &SelectedObservationResidencyCertificate,
1871        selected_content_allocations: &BTreeSet<AllocationId>,
1872        queue: &LeaseResource,
1873        release: &WorkNodeId,
1874    ) -> Result<Self, WeightingPlanFragmentError> {
1875        if residency.aggregate_resident_bytes() == 0
1876            || residency.peak_live_blocks() == 0
1877            || residency.maximum_pointing_polynomial_terms() == 0
1878        {
1879            return Err(WeightingPlanFragmentError::InvalidSourceAuthority {
1880                node: source.id.clone(),
1881                reason: "selected-content budget is empty",
1882            });
1883        }
1884        if !source
1885            .claims
1886            .iter()
1887            .any(|claim| matches!(claim.resource, LeaseResource::Workers) && claim.amount > 0)
1888        {
1889            return Err(WeightingPlanFragmentError::InvalidSourceAuthority {
1890                node: source.id.clone(),
1891                reason: "selected traversal has no worker claim",
1892            });
1893        }
1894        let required_blocks = u64::try_from(residency.peak_live_blocks())
1895            .map_err(|_| WeightingPlanFragmentError::ResidencyOverflow)?;
1896        let mut queue_claims = source
1897            .claims
1898            .iter()
1899            .filter(|claim| &claim.resource == queue);
1900        let queue_claim_covers = queue_claims
1901            .next()
1902            .is_some_and(|claim| claim.amount == required_blocks)
1903            && queue_claims.next().is_none();
1904        if !is_selected_content_queue(queue)
1905            || !queue_claim_covers
1906            || !queue_demand_covers(
1907                base.execution_dag().resource_alternative(),
1908                queue,
1909                required_blocks,
1910            )
1911        {
1912            return Err(WeightingPlanFragmentError::InvalidSourceAuthority {
1913                node: source.id.clone(),
1914                reason: "selected traversal lacks its exact planned queue identity and capacity",
1915            });
1916        }
1917        let expected_bytes = u64::try_from(residency.aggregate_resident_bytes())
1918            .map_err(|_| WeightingPlanFragmentError::ResidencyOverflow)?;
1919        let claimed_bytes = source.claims.iter().try_fold(0_u64, |total, claim| {
1920            if claim.resource == LeaseResource::IoBuffer(IoBufferKind::SourceReadAhead) {
1921                total.checked_add(claim.amount)
1922            } else {
1923                Some(total)
1924            }
1925        });
1926        let retained_allocations = source
1927            .allocations
1928            .iter()

2161            !context.allocations().iter().any(|capability| {

2166        || expected_allocations.iter().any(|spec| {
2167            !context
2168                .allocations()
2169                .iter()
2170                .any(|capability| spec.matches_capability(capability, &lifetime))
2171        })
2172    {
2173        return Err(WeightingEvidenceError);
2174    }
2175    if let Some((residency, queue, source_allocations)) = selected_content_budget {
2176        if !residency
2177            .matches_problem(problem.expect("selected traversal always carries a compiled problem"))
2178        {

2180        }

2182            .map_err(|_| WeightingEvidenceError)?;

2190                total.checked_add(capability.capacity_bytes())

2192            .ok_or(WeightingEvidenceError)?;

2194            .resources()
2195            .iter()
2196            .filter(|capability| {
2197                capability.resource() == &LeaseResource::IoBuffer(IoBufferKind::SourceReadAhead)
2198            })
2199            .try_fold(0_u64, |total, capability| {
2200                total.checked_add(capability.amount())
2201            })
2202            .ok_or(WeightingEvidenceError)?;
2203        let mut queue_capabilities = context
2204            .resources()
2205            .iter()
2206            .filter(|capability| capability.resource() == queue);
2207        let queue_capability_covers = queue_capabilities
2208            .next()
2209            .is_some_and(|capability| capability.amount() == required_blocks)
2210            && queue_capabilities.next().is_none();

2212            || context
2213                .allocations()
2214                .iter()
2215                .filter(|capability| source_allocations.contains(capability.allocation()))
2216                .count()
2217                != source_allocations.len()

2219            || source_capacity != required_bytes
2220            || !queue_capability_covers
2221            || !queue_demand_covers(context.resource_alternative(), queue, required_blocks)
2222            || !context.resources().iter().any(|capability| {
2223                capability.resource() == &LeaseResource::Workers && capability.amount() > 0
2224            })

2645    fn replay<E>(
2646        &self,
2647        context: WorkExecutionContext<'_>,
2648        fragment: &WeightingPlanFragment<'_>,
2649        selected: &mut BoundSelectedObservation,
2650        problem: &CompiledProblem,
2651        mut emit: impl FnMut(&WeightedObservationBlock) -> Result<(), E>,
2652    ) -> Result<PendingWeightingReplay, WeightingReplayError<E>>
2653    where
2654        E: Error + 'static,
2655    {
2656        let replay_binding = fragment
2657            .authorize_replay(context, self, problem)
2658            .map_err(WeightingReplayError::Evidence)?;
2659        let predecessor = context
2660            .predecessor_observation_completion(fragment.generation_node())
2661            .ok_or(WeightingReplayError::Evidence(WeightingEvidenceError))?;
2662        if !selected.can_resume_after(predecessor.owner_completion()) {
2663            return Err(WeightingReplayError::Evidence(WeightingEvidenceError));
2664        }
2665        let mut phase = self
2666            .artifact
2667            .state
2668            .begin_replay(problem, fragment.plan)
2669            .map_err(WeightingReplayError::Owner)?;
2670        let owner_completion = selected
2671            .traverse(problem, |reported| {
2672                let contributions = compiled_spectral_contributions(problem, &reported)
2673                    .map_err(ReplayCallbackError::Owner)?;
2674                if let Some(block) = phase
2675                    .consume(problem, *reported.selected(), contributions)
2676                    .map_err(ReplayCallbackError::Owner)?
2677                {
2678                    let block = WeightedObservationBlock::authorize(self.generation_id(), block);
2679                    emit(&block).map_err(ReplayCallbackError::Consumer)?;
2680                }
2681                Ok(())
2682            })
2683            .map_err(WeightingReplayError::Traversal)?;
2684        validate_replay_completion(
2685            self.artifact.source_generation,
2686            self.artifact.source_sample_count,
2687            predecessor.owner_completion(),
2688            &owner_completion,
2689            &self.artifact.state,
2690        )
2691        .map_err(WeightingReplayError::Evidence)?;
2692        let (final_block, state) = phase.finish().map_err(WeightingReplayError::Owner)?;
2693        if let Some(block) = final_block {
2694            let block = WeightedObservationBlock::authorize(self.generation_id(), block);
2695            emit(&block).map_err(WeightingReplayError::Consumer)?;
2696        }
2697        let spectral_support_sample_count = state.sample_count();
2698        Ok(PendingWeightingReplay {
2699            state,
2700            owner_completion,
2701            binding: replay_binding,
2702            continuum_transform: None,
2703            spectral_support_sample_count,
2704        })
2705    }

2708/// Unbranded result of two exhaustive owner traversals.

2710struct PendingWeightingGeneration {
2711    state: WeightingAlgorithmState,
2712    density_completion: SelectedObservationCompletion,
2713    sum_weight_completion: SelectedObservationCompletion,
2714    binding: WeightingGenerationBinding,
2715}
2716
2717fn traverse_weighting_generation(
2718    context: WorkExecutionContext<'_>,
2719    fragment: &WeightingPlanFragment<'_>,
2720    selected: &mut BoundSelectedObservation,
2721    problem: &CompiledProblem,
2722) -> Result<PendingWeightingGeneration, WeightingGenerationError> {
2723    let binding = fragment
2724        .authorize_generation(context, problem)
2725        .map_err(WeightingGenerationError::Evidence)?;
2726    let source_completion = context
2727        .predecessor_observation_completion(&fragment.source_read)
2728        .ok_or(WeightingGenerationError::Evidence(WeightingEvidenceError))?
2729        .owner_completion();
2730    if !selected.can_resume_after(source_completion) {
2731        return Err(WeightingGenerationError::Evidence(WeightingEvidenceError));
2732    }
2733    let mut density = begin_weighting_generation(problem, fragment.plan)
2734        .map_err(WeightingGenerationError::Owner)?;
2735    let density_completion = selected
2736        .traverse(problem, |reported| {
2737            let contributions = compiled_spectral_contributions(problem, &reported)?;
2738            density.consume(problem, *reported.selected(), contributions)
2739        })
2740        .map_err(WeightingGenerationError::DensityTraversal)?;
2741    let sum_weight = density
2742        .finish(problem)
2743        .map_err(WeightingGenerationError::Owner)?;
2744    let mut sum_weight = sum_weight;
2745    let sum_weight_completion = selected
2746        .traverse(problem, |reported| {
2747            let contributions = compiled_spectral_contributions(problem, &reported)?;
2748            sum_weight.consume(problem, *reported.selected(), contributions)
2749        })
2750        .map_err(WeightingGenerationError::SumWeightTraversal)?;
2751    let state = sum_weight
2752        .finish()
2753        .map_err(WeightingGenerationError::Owner)?;
2754    validate_generation_completions(&density_completion, &sum_weight_completion)
2755        .map_err(WeightingGenerationError::Evidence)?;
2756    if state.sample_count() != density_completion.sample_count() {
2757        return Err(WeightingGenerationError::Evidence(WeightingEvidenceError));
2758    }
2759    if !source_completion.precedes(&density_completion)
2760        || density_completion.generation_id() != binding.source_generation
2761        || density_completion.sample_count() != binding.source_sample_count
2762    {
2763        return Err(WeightingGenerationError::Evidence(WeightingEvidenceError));
2764    }
2765    Ok(PendingWeightingGeneration {
2766        state,
2767        density_completion,
2768        sum_weight_completion,
2769        binding,
2770    })
2771}

2773fn complete_weighting_generation(
2774    pending: PendingWeightingGeneration,
2775    context: ObservationReadCompletionContext,
2776) -> Result<
2777    (FrozenWeightingGeneration, AttemptBoundObservationCompletion),
2778    WeightingGenerationCompletionError,
2779> {
2780    validate_generation_completions(&pending.density_completion, &pending.sum_weight_completion)
2781        .map_err(WeightingGenerationCompletionError::Evidence)?;
2782    if pending.state.sample_count() != pending.density_completion.sample_count() {
2783        return Err(WeightingGenerationCompletionError::Evidence(
2784            WeightingEvidenceError,
2785        ));
2786    }
2787    if context.attempt_id() != pending.binding.attempt_id
2788        || context.owner_node() != &pending.binding.owner_node
2789        || context.lease_epoch() != pending.binding.lease_epoch
2790    {
2791        return Err(WeightingGenerationCompletionError::Evidence(
2792            WeightingEvidenceError,
