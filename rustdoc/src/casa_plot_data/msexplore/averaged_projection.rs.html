71impl AveragedProjection<'_> {
72    fn row_frequencies(
73        &mut self,
74        group: Option<usize>,
75        channel_ordinal: usize,
76        planned_time: Option<f64>,
77    ) -> Result<RowFrequencies, String> {
78        let Self {
79            inputs:
80                AveragedProjectionInputs {
81                    time_spectra,
82                    time_reporting_frame,
83                    time_bins,
84                    selected_data_desc_ids,
85                    selected_field_ids,
86                    derived_engine,
87                    ms,
88                    selected_times,
89                    time_bounds,
90                    time_wavelengths,
91                    ..
92                },
93            converted_time_frequencies,
94            time_spectral_engine,
95            time_doppler,
96            time_rest_frequency,
97            ..
98        } = self;
99        Ok(
100            if let (Some(spectra), Some(frame), Some(bins), Some(group)) =
101                (time_spectra, *time_reporting_frame, *time_bins, group)
102            {
103                let slot = bins.first_row_slot(group);
104                let ddid = selected_data_desc_ids[slot];
105                let spectrum = spectra
106                    .get(&ddid)
107                    .ok_or("Missing selected frequency grid for PlotMS time averaging")?;
108                let mut frequency_at = |channel: usize, time: f64| -> Result<f64, String> {
109                    let frequency = *spectrum
110                        .frequencies_hz
111                        .get(channel)
112                        .ok_or("Missing selected channel frequency for PlotMS time averaging")?;
113                    if spectrum.frame == frame {
114                        return Ok(frequency);
115                    }
116                    use std::collections::hash_map::Entry;
117                    let field = usize::try_from(selected_field_ids[slot])
118                        .map_err(|_| "Invalid FIELD_ID for PlotMS time spectral conversion")?;
119                    let key = (ddid, field, channel, time.to_bits());
120                    match converted_time_frequencies.entry(key) {
121                        Entry::Occupied(entry) => Ok(*entry.get()),
122                        Entry::Vacant(entry) => {
123                            let engine = match derived_engine {
124                                Some(engine) => engine,
125                                None => time_spectral_engine
126                                    .get_or_insert_with(|| MsCalEngine::for_calibration(ms))
127                                    .as_ref()
128                                    .map_err(|error| error.to_string())?,
129                            };
130                            Ok(*entry.insert(
131                                convert_frequency_to_frame(
132                                    spectrum.frame,
133                                    frame,
134                                    frequency,
135                                    time,
136                                    field,
137                                    engine,
138                                )
139                                .map_err(|error| error.to_string())?,
140                            ))
141                        }
142                    }
143                };
144                let buffer_time = planned_time.unwrap_or(selected_times[slot]);
145                let frequency = frequency_at(channel_ordinal, buffer_time)?;
146                // CASA projects a wavelength at the individual baseline epoch;
147                // Frequency, Velocity and verbose metadata use buffer row zero.
148                let row_time = time_bounds
149                    .get(&group)
150                    .map_or(buffer_time, |bounds| bounds.buffer_integration().time);
151                let wavelength_frequency = if *time_wavelengths && row_time != buffer_time {
152                    frequency_at(channel_ordinal, row_time)?
153                } else {
154                    frequency
155                };
156                let velocity = time_doppler
157                    .map(|definition| {
158                        let rest = match *time_rest_frequency {
159                            Some(rest) => rest,
160                            None => frequency_at(spectrum.frequencies_hz.len() / 2, buffer_time)?,
161                        };
162                        velocity_ms_from_frequency_hz(frequency, rest, definition)
163                            .map(|velocity| velocity / 1000.0)
164                            .map_err(|error| error.to_string())
165                    })
166                    .transpose()?;
167                RowFrequencies {
168                    wavelength: Some(wavelength_frequency),
169                    reporting: Some(frequency),
170                    velocity,
171                }
172            } else {
173                RowFrequencies::default()
174            },
175        )
176    }

178    fn point_geometry(
179        &mut self,
180        group: Option<usize>,
181        channel_ordinal: usize,
182        accumulator: &AveragedPointAccumulator,
183    ) -> Result<Option<PointGeometry>, String> {
184        let Self {
185            inputs:
186                AveragedProjectionInputs {
187                    collapsed_uv,
188                    uv_channels,
189                    time_bounds,
190                    time_bins,
191                    selected_data_desc_ids,
192                    uv_selections,
193                    clear_uv_channels,
194                    phase_shift,
195                    data_description,
196                    selected_field_ids,
197                    selected_times,
198                    ms,
199                    scalar_intervals,
200                    scalar_time_bounds,
201                    ..
202                },
203            shifted_time_row,
204            time_uvws,
205            time_baseline_plan,
206            buffer_integrations,
207            ..
208        } = self;
209        if !*collapsed_uv && !accumulator.has_usable_weight() {
210            return Ok(None);
211        }
212        let Some(mut representative) = accumulator.representative.clone() else {
213            return Ok(None);
214        };
215        let mut display_channel = accumulator.display_channel;
216        if *collapsed_uv {
217            let group = group.expect("collapsed UV uses time groups");
218            if time_bounds
219                .get(&group)
220                .and_then(plotms_ordinates::VisibilityTimeBounds::midpoint)
221                .is_none()
222            {
223                return Ok(None);
224            }
225            if *uv_channels && !clear_uv_channels.contains(&(group, channel_ordinal)) {
226                return Ok(None);
227            }
228            let time = time_bins.expect("collapsed UV uses a buffer plan");
229            let ddid = selected_data_desc_ids[time.first_row_slot(group)] as usize;
230            let (correlations, channels) = &uv_selections[&ddid];
231            let (bins, channel) = if *uv_channels {
232                (channels.len(), channel_ordinal)
233            } else {
234                (1, 0)
235            };
236            let index = plotms_row_axes::averaged_uvw_cache_index(time, group, bins, channel);
237            let channel = &channels[index % channels.len()];
238            representative.corr = correlations[index % correlations.len()];
239            representative.chan_start = channel.start;
240            representative.chan_end = channel.end;
241            display_channel = channel.display_channel();
242        }
243        let channel_start = representative.chan_start;
244        // The phase-shifting TVI is above AveragingTvi2. Use the
245        // averaged baseline before projecting its complex visibilities.
246        let shifted_time = if let Some(shift) = phase_shift {
247            let group = group.ok_or("Missing PlotMS phasecenter time group")?;
248            let bins = time_bins.expect("phasecenter time groups");
249            let slot = bins.first_row_slot(group);
250            let uvw = match *shifted_time_row {
251                Some((previous, uvw)) if previous == group => uvw,
252                _ => {
253                    let ddid = usize::try_from(selected_data_desc_ids[slot])
254                        .map_err(|_| "Invalid DATA_DESC_ID for PlotMS phasecenter")?;
255                    let spw = data_description
256                        .spectral_window_id(ddid)
257                        .map_err(|error| error.to_string())?;
258                    let spw = usize::try_from(spw)
259                        .map_err(|_| "Invalid SPECTRAL_WINDOW_ID for PlotMS phasecenter")?;
260                    let reference = bins
261                        .phase_reference_slots()
262                        .expect("phasecenter time reference slots")[slot];
263                    let uvw = shift.prepare_row(
264                        ms,
265                        selected_field_ids[slot],
266                        spw,
267                        selected_times[reference],
268                        time_uvws.as_ref().expect("phasecenter averaged UVW")[&group],
269                    )?;
270                    *shifted_time_row = Some((group, uvw));
271                    uvw
272                }
273            };
274            Some((uvw, shift.phase(channel_start)?))
275        } else {
276            None
277        };
278        let planned_time = group.and_then(|group| {
279            if *scalar_intervals {
280                scalar_time_bounds
281                    .get(&group)
282                    .map(|(start, end)| start + (end - start) / 2.0)
283            } else if let Some(plan) = &time_baseline_plan {
284                plan.center(group)
285            } else if let Some(integrations) = &buffer_integrations {
286                integrations[group].map(|integration| integration.time)
287            } else {
288                time_bounds
289                    .get(&group)
290                    .and_then(plotms_ordinates::VisibilityTimeBounds::midpoint)
291            }
292        });
293
294        Ok(Some(PointGeometry {
295            representative,
296            display_channel,
297            shifted_time,
298            planned_time,
299        }))
300    }

302    pub(super) fn project_point(
303        &mut self,
304        group: Option<usize>,
305        order: Option<usize>,
306        spectral_channel: Option<&plotms_spw::SpectralChannel>,
307        channel_ordinal: usize,
308        accumulator: AveragedPointAccumulator,
309        y_projection_axis: MsAxis,
310    ) -> Result<Option<ProjectedPoint>, String> {
311        let Some(PointGeometry {
312            representative,
313            display_channel,
314            shifted_time,
315            planned_time,
316        }) = self.point_geometry(group, channel_ordinal, &accumulator)?
317        else {
318            return Ok(None);
319        };
320        let channel_start = representative.chan_start;
321        let RowFrequencies {
322            wavelength: wavelength_frequency,
323            reporting: time_reporting_frequency,
324            velocity: time_reporting_velocity,
325        } = self.row_frequencies(group, channel_ordinal, planned_time)?;
326        let Self {
327            inputs:
328                AveragedProjectionInputs {
329                    time_bins,
330                    time_reporting_axes,
331                    scalar_intervals,
332                    averaged_row_numbers,
333                    spec,
334                    average_statistics,
335                    weighted_time_uvw,
336                    scalar_visibility,
337                    calibration,
338                    requested_freqframe,
339                    ..
340                },
341            time_statistics,
342            buffer_integrations,
343            time_uvws,
344            x_projection_axis,
345            report_time_frequencies,
346            ..
347        } = self;
348        let statistics = time_statistics
349            .as_ref()
350            .and_then(|statistics| statistics.get(&(group?, representative.corr)));
351        let project = |axis: MsAxis, x_axis| {
352            if *time_reporting_axes && axis == MsAxis::Frequency {
353                time_reporting_frequency
354            } else if *time_reporting_axes && axis == MsAxis::Velocity {
355                time_reporting_velocity
356            } else if let Some(channel) = spectral_channel
357                && matches!(axis, MsAxis::Channel | MsAxis::Frequency)
358            {
359                Some(if axis == MsAxis::Channel {
360                    channel.number as f64
361                } else {
362                    channel.frequency_hz
363                })
364            } else if axis == MsAxis::Time && (time_bins.is_some() || *scalar_intervals) {
365                planned_time
366            } else if axis == MsAxis::Row {
367                Some(*(*averaged_row_numbers)?.get(group?)? as f64)
368            } else if axis == MsAxis::Interval {
369                Some(buffer_integrations.as_ref()?[group?]?.interval)
370            } else if axis == MsAxis::WeightedAmplitude {
371                let value = finalize_averaged_axis_value(
372                    MsAxis::Amplitude,
373                    &accumulator,
374                    spec.averaging.scalar,
375                    spec.averaging.avgtime.is_some(),
376                    x_axis,
377                    time_bins.is_some(),
378                )?;
379                Some(weight_amplitude(value, statistics?.weight))
380            } else if *average_statistics && axis.is_weight_axis() {
381                statistics?.value(axis, channel_start)
382            } else if let Some((_, phase)) = shifted_time
383                && axis.is_visibility_math()
384            {
385                let samples = if x_axis {
386                    &accumulator.x_visibility
387                } else {
388                    &accumulator.y_visibility
389                };
390                samples.project_shifted_time(axis, phase)
391            } else if *weighted_time_uvw && axis.uses_uvw() {
392                let uvw = shifted_time
393                    .map(|(uvw, _)| uvw)
394                    .or_else(|| time_uvws.as_ref()?.get(&order?).copied())?;
395                let metres = uvw_metres_coordinate(axis, uvw);
396                if axis.uses_wavelengths() {
397                    Some(metres_to_wavelengths(metres, wavelength_frequency?))
398                } else {
399                    Some(metres)
400                }
401            } else if *scalar_visibility && axis.is_visibility_math() {
402                let samples = if x_axis {
403                    &accumulator.x_visibility
404                } else {
405                    &accumulator.y_visibility
406                };
407                samples.project_scalar(axis)
408            } else {
409                finalize_averaged_axis_value(
410                    axis,
411                    &accumulator,
412                    spec.averaging.scalar,
413                    spec.averaging.avgtime.is_some(),
414                    x_axis,
415                    time_bins.is_some(),
416                )
417            }
418        };
419        let Some(x_value) = project(*x_projection_axis, true) else {
420            return Ok(None);
421        };
422        let Some(y_value) = project(y_projection_axis, false) else {
423            return Ok(None);
424        };
425
426        let (reporting_frequency, metadata) = if calibration.is_some() {
427            let reporting_frequency = if *report_time_frequencies {
428                time_reporting_frequency
429            } else if requested_freqframe.is_some() && *x_projection_axis == MsAxis::Frequency {
430                Some(x_value)
431            } else if requested_freqframe.is_some() && y_projection_axis == MsAxis::Frequency {
432                Some(y_value)
433            } else {
434                None
435            };
436            let metadata = MsScatterPlotMsMetadata {
437                channel: spectral_channel.map_or(display_channel, |channel| channel.number),
438                average_time: spec
439                    .averaging
440                    .avgtime
441                    .and_then(|_| project(MsAxis::Time, false)),
442                averaging_group: order,
443            };
444            (reporting_frequency, Some(metadata))
445        } else {
446            (None, None)
447        };
448        Ok(Some(ProjectedPoint {
449            point: (x_value, y_value),
450            representative,
451            flagged: accumulator.visibility_flagged == Some(true),
452            reporting_frequency,
453            metadata,
454        }))
455    }
