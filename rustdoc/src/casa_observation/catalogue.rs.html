2//! The catalogue of an opened observation: its antennas, fields, scans,
3//! observations and partitions, with identities scoped to that opening.
4//!
5//! Every per-partition constant a batch needs (channel frequencies and
6//! their frame, correlation products, the baseline table, the time system,
7//! the data groups with their units and UVW frame) lives here once; a batch
8//! links to it by [`PartitionId`] and copies none of it.

60}

70/// What a data group's visibilities are measured in.
71///
72/// Jy only where the store says so; anything else is kept as the store
73/// spells it and never assumed to be a calibrated flux density.
74#[derive(Clone, Debug, Default, PartialEq, Eq)]
75pub enum VisibilityUnits {
76    /// Janskys.

82    #[default]

173impl ObservationEntry {
174    /// Whether `other` can be the same session: the same telescope, project
175    /// and observers, and the same archive identifier wherever both state
176    /// one.
177    #[must_use]
178    pub fn same_session(&self, other: &Self) -> bool {
179        let agree = |a: &Option<String>, b: &Option<String>| match (a, b) {
180            (Some(a), Some(b)) => a == b,
181            _ => true,
182        };
183        self.telescope == other.telescope
184            && self.project == other.project
185            && self.observers == other.observers
186            && agree(&self.execution_block, &other.execution_block)
187            && agree(&self.session_reference, &other.session_reference)
188            && agree(&self.scheduling_block, &other.scheduling_block)
189    }
190}

192/// The frame a data group's UVW are in, and the label the store wrote.
193///
194/// The batch's [`UvwView`](casa_types::measures::UvwView) carries only the
195/// effective frame: an adapter resolves the store's convention once, at
196/// opening, and refuses a store whose label it cannot resolve. For MSv2
197/// that convention is CASA's: `UVW` is computed in the phase centre's frame
198/// whatever its `MEASINFO` says (simulators and fillers commonly label it
199/// `ITRF`, the casacore default; `fixvis` notes the label is usually wrong,
200/// and XRADIO's converter relabels it the same way). MSv4 states the frame.
201#[derive(Clone, Copy, Debug, PartialEq, Eq)]
202pub struct UvwFrame {
203    /// The frame the UVW values are in.
204    pub effective: DirectionRef,
205    /// The frame the store's label names, when it names one.
206    pub stored_label: Option<DirectionRef>,
207}

230/// One partition: a store unit and the content that defines it.
231///
232/// The content keys (spectral window, correlation products, intents,
233/// observation, feed pair) and the fields and scans the partition's
234/// samples visit are what two stores of one observation agree on; the MSv2
235/// adapter partitions rows the way XRADIO's converter does by default
236/// (`DATA_DESC_ID` × `OBS_MODE` × `OBSERVATION_ID` × `EPHEMERIS_ID`, and the
237/// feed pair, which MSv4 has no axis for), so the same observation opens as
238/// the same partition set from either store. Cross-store matching
239/// ([`Catalogue::bind_to`]) compares content, never list positions.
240#[derive(Clone, Debug)]
241pub struct PartitionEntry {
242    /// The spectral window's name as stored.
243    pub spectral_window: String,
244    /// Channel centres in hertz.
245    pub frequencies_hz: Vec<f64>,
246    /// Their spectral frame.
247    pub frequency_frame: FrequencyRef,
248    /// Signed channel widths in hertz.
249    pub channel_widths_hz: Vec<f64>,
250    /// The correlation products, in stored order.
251    pub correlations: Vec<Correlation>,
252    /// The scan intents of every sample (MSv2 `STATE::OBS_MODE`, MSv4
253    /// `scan_name.scan_intents`), sorted.
254    pub intents: Vec<String>,
255    /// The observing session.
256    pub observation: ObservationId,
257    /// The feed pair (MSv2 `FEED1`, `FEED2`) of every sample; `None` for a
258    /// store without a feed axis (MSv4).
259    pub feeds: Option<[u32; 2]>,
260    /// The fields the partition's samples are phased to, over every data
261    /// group, sorted.
262    pub fields: Vec<FieldId>,
263    /// The scans its samples belong to, sorted; empty for a store that
264    /// names none (an MSv4 dataset without `scan_name`).
265    pub scans: Vec<ScanId>,
266    /// The baseline table: sample `t × baselines.len() + b` of every batch
267    /// is baseline `b`.
268    pub baselines: Vec<OrderedBaseline>,
269    /// Time scale of the partition's times and time centroids.
270    pub time_scale: EpochRef,
271    /// Origin the partition's time offsets count from (MJD 0 for MSv2,
272    /// the Unix epoch for MSv4 `format: unix`).
273    pub time_origin: MjdHighPrec,
274    /// The partition's data groups.
275    pub data_groups: Vec<DataGroupEntry>,
276    /// The receptors of each antenna the baselines name, in this
277    /// partition's spectral window; an antenna the store states none for
278    /// is absent.
279    pub receptors: BTreeMap<AntennaId, Receptors>,
280}

282impl PartitionEntry {
283    /// Data group `name`, with its index in [`Self::data_groups`].
284    #[must_use]
285    pub fn data_group(&self, name: &str) -> Option<(usize, &DataGroupEntry)> {
286        self.data_groups
287            .iter()
288            .enumerate()
289            .find(|(_, group)| group.name == name)
290    }
291
292    /// Where `correlation` is along this partition's correlation axis.
293    #[must_use]
294    pub fn correlation_index(&self, correlation: Correlation) -> Option<usize> {
295        self.correlations.iter().position(|c| *c == correlation)
296    }
297
298    /// Whether `other` has the same spectral and product content: the same
299    /// channel frequencies and widths (bit for bit) in the same frame, the
300    /// same correlation products in the same order and the same intents.
301    /// Names, which converters rewrite, take no part; the keys that name
302    /// other entities (observation, fields, scans) and the feed pair are
303    /// compared by [`Catalogue::bind_to`] through the binding.
304    #[must_use]
305    pub fn same_content(&self, other: &Self) -> bool {
306        let bitwise = |a: &[f64], b: &[f64]| {
307            a.len() == b.len() && a.iter().zip(b).all(|(a, b)| a.to_bits() == b.to_bits())
308        };
309        self.frequency_frame == other.frequency_frame
310            && bitwise(&self.frequencies_hz, &other.frequencies_hz)
311            && bitwise(&self.channel_widths_hz, &other.channel_widths_hz)
312            && self.correlations == other.correlations
313            && self.intents == other.intents
314    }
315}

351/// One catalogue's entities as another catalogue numbers them: what

407    /// Bind this catalogue's entities to `other`'s by content: antennas by
408    /// station and position within `tolerance`, fields by phase centre
409    /// (frame and angles within `tolerance`), scans by casacore number (by
410    /// name when a scan has none), observing sessions by
411    /// [`ObservationEntry::same_session`], and partitions by
412    /// [`PartitionEntry::same_content`] together with the bound session, the
413    /// bound sets of fields and of scans (scans when both stores name
414    /// them) and the feed pair (when both stores have one). Names and list
415    /// positions decide nothing: converters rename antennas and fields.
416    ///
417    /// The binding is one to one: a key one store lacks (MSv4 has no feed
418    /// axis; an MSv2 may hold no archive identifiers) does not distinguish,
419    /// but two entities this catalogue keeps apart never bind to one entity
420    /// of `other`.
421    ///
422    /// # Errors
423    ///
424    /// When an entity of this catalogue matches none of `other`'s, or more
425    /// than one, or two of them match the same one: an ambiguity is
426    /// refused, never resolved by position.
427    pub fn bind_to(
428        &self,
429        other: &Self,
430        tolerance: &BindingTolerance,
431    ) -> Result<Binding, ObservationError> {
432        let within = |a: f64, b: f64, tolerance: f64| (a - b).abs() <= tolerance;
433        let antennas = bind_each("antenna", &self.antennas, &other.antennas, |a, b| {
434            a.station == b.station
435                && a.position.refer() == b.position.refer()
436                && a.position
437                    .values()
438                    .iter()
439                    .zip(b.position.values())
440                    .all(|(x, y)| within(*x, y, tolerance.position_m))
441        })?;
442        let fields = bind_each("field", &self.fields, &other.fields, |a, b| {
443            let (al, ab) = a.phase_centre.as_angles();
444            let (bl, bb) = b.phase_centre.as_angles();
445            a.phase_centre.refer() == b.phase_centre.refer()
446                && within(al, bl, tolerance.direction_rad)
447                && within(ab, bb, tolerance.direction_rad)
448        })?;
449        let scans = bind_each("scan", &self.scans, &other.scans, |a, b| match a.number {
450            Some(number) => b.number == Some(number),
451            None => b.number.is_none() && a.name == b.name,
452        })?;
453        let observations = bind_each(
454            "observation",
455            &self.observations,
456            &other.observations,
457            ObservationEntry::same_session,
458        )?;
459        let partitions = bind_each("partition", &self.partitions, &other.partitions, |a, b| {
460            a.same_content(b)
461                && observations.get(a.observation.0 as usize) == Some(&b.observation.0)
462                && (a.feeds.is_none() || b.feeds.is_none() || a.feeds == b.feeds)
463                && same_set(&a.fields, &b.fields, |field| field.0, &fields)
464                && (a.scans.is_empty()
465                    || b.scans.is_empty()
466                    || same_set(&a.scans, &b.scans, |scan| scan.0, &scans))
467        })?;
468        Ok(Binding {
469            antennas: antennas.into_iter().map(AntennaId).collect(),
470            fields: fields.into_iter().map(FieldId).collect(),
471            scans: scans.into_iter().map(ScanId).collect(),
472            observations: observations.into_iter().map(ObservationId).collect(),
473            partitions: partitions.into_iter().map(PartitionId).collect(),
474        })
475    }
476}

478/// Whether identities `ours`, renumbered through `binding`, are the sorted
479/// set `theirs`.
480fn same_set<T: Copy>(ours: &[T], theirs: &[T], id: fn(T) -> u32, binding: &[u32]) -> bool {
481    let renumbered: Option<Vec<u32>> = ours
482        .iter()
483        .map(|ours| binding.get(id(*ours) as usize).copied())
484        .collect();
485    renumbered.is_some_and(|mut renumbered| {
486        renumbered.sort_unstable();
487        renumbered
488            .into_iter()
489            .eq(theirs.iter().map(|theirs| id(*theirs)))
490    })
491}

493/// The index in `others` of the one entry `matches` each of `entries`, no
494/// two of them the same.
495fn bind_each<T>(
496    what: &str,
497    entries: &[T],
498    others: &[T],
499    matches: impl Fn(&T, &T) -> bool,
500) -> Result<Vec<u32>, ObservationError> {
501    let bound: Vec<u32> = entries
502        .iter()
503        .enumerate()
504        .map(|(index, entry)| {
505            let mut found = others
506                .iter()
507                .enumerate()
508                .filter(|(_, other)| matches(entry, other))
509                .map(|(index, _)| index);
510            match (found.next(), found.next()) {
511                (Some(one), None) => Ok(u32::try_from(one).expect("few entries")),
512                (None, _) => Err(ObservationError::InvalidRequest {
513                    reason: format!("{what} {index} matches nothing in the other catalogue"),
514                }),
515                (Some(_), Some(_)) => Err(ObservationError::InvalidRequest {
516                    reason: format!("{what} {index} matches more than one in the other catalogue"),
517                }),
518            }
519        })
520        .collect::<Result<_, _>>()?;
521    let mut first = std::collections::BTreeMap::new();
522    for (index, other) in bound.iter().enumerate() {
523        if let Some(earlier) = first.insert(*other, index) {
524            return Err(ObservationError::InvalidRequest {
525                reason: format!(
526                    "{what}s {earlier} and {index} both match {what} {other} of the other \
527                     catalogue, which does not keep them apart"
528                ),
529            });
530        }
531    }
532    Ok(bound)
533}

597    fn binds(a: &Catalogue, b: &Catalogue) -> Result<Vec<PartitionId>, ObservationError> {
598        a.bind_to(b, &BindingTolerance::EXACT)
599            .map(|binding| binding.partitions)

602    /// Equal spectral setups are told apart by everything else a partition
603    /// is keyed by: its session, the fields and scans it visits, its feed
604    /// pair where both stores have one, and its channel widths. A key one
605    /// store lacks does not distinguish, and two entities never bind to one.
606    #[test]
607    fn partitions_bind_by_session_fields_scans_feeds_and_widths() {
608        let one = observed(vec![session("p", None)], vec![partition(0, &[0], &[0])]);
609        assert_eq!(binds(&one, &one).expect("itself"), [PartitionId(0)]);
610
611        // Same spectrum, another session.
612        let other_project = observed(vec![session("q", None)], vec![partition(0, &[0], &[0])]);
613        assert!(binds(&one, &other_project).is_err(), "another project");
614        let two = observed(
615            vec![session("p", Some("uid://A")), session("p", Some("uid://B"))],
616            vec![partition(0, &[0], &[0]), partition(1, &[0], &[0])],
617        );
618        assert_eq!(
619            binds(&two, &two).expect("sessions apart"),
620            [PartitionId(0), PartitionId(1)]
621        );
622        let only_b = observed(
623            vec![session("p", Some("uid://B"))],
624            vec![partition(0, &[0], &[0])],
625        );
626        assert_eq!(
627            binds(&only_b, &two).expect("a subset binds to its own session"),
628            [PartitionId(1)]
629        );
630        assert!(
631            binds(&two, &only_b).is_err(),
632            "session A has no counterpart"
633        );
634        assert!(
635            binds(&two, &one).is_err(),
636            "two sessions told apart by their UIDs would collapse onto one"
637        );
638        assert_eq!(
639            binds(&one, &only_b).expect("a UID one store lacks does not distinguish"),
640            [PartitionId(0)]
641        );
642
643        // Same session and spectrum, split by field and by scan.
644        let by_field = observed(
645            vec![session("p", None)],
646            vec![partition(0, &[0], &[0, 1]), partition(0, &[1], &[0, 1])],
647        );
648        let both_fields = observed(
649            vec![session("p", None)],
650            vec![partition(0, &[0, 1], &[0, 1])],
651        );
652        assert!(binds(&by_field, &both_fields).is_err(), "field split");
653        assert!(binds(&both_fields, &by_field).is_err(), "field split");
654        let field_one = observed(vec![session("p", None)], vec![partition(0, &[1], &[0, 1])]);
655        assert_eq!(
656            binds(&field_one, &by_field).expect("asymmetric subset"),
657            [PartitionId(1)]
658        );
659        let by_scan = observed(
660            vec![session("p", None)],
661            vec![partition(0, &[0], &[0]), partition(0, &[0], &[1])],
662        );
663        assert_eq!(
664            binds(&by_scan, &by_scan).expect("scans apart"),
665            [PartitionId(0), PartitionId(1)]
666        );
667        let unscanned = observed(vec![session("p", None)], vec![partition(0, &[0], &[])]);
668        assert!(
669            binds(&by_scan, &unscanned).is_err(),
670            "two scan partitions would collapse onto one without scans"
671        );
672
673        // Feeds distinguish only where both stores have them.
674        let feeds = |pair: Option<[u32; 2]>| {
675            let mut entry = partition(0, &[0], &[0]);
676            entry.feeds = pair;
677            entry
678        };
679        let feed_split = observed(
680            vec![session("p", None)],
681            vec![feeds(Some([0, 0])), feeds(Some([1, 1]))],
682        );
683        assert_eq!(
684            binds(&feed_split, &feed_split).expect("feeds apart"),
685            [PartitionId(0), PartitionId(1)]
686        );
687        assert!(
688            binds(&feed_split, &one).is_err(),
689            "two feed pairs would collapse onto a store without feeds"
690        );
691        let feed_one = observed(vec![session("p", None)], vec![feeds(Some([1, 1]))]);
692        assert_eq!(
693            binds(&feed_one, &feed_split).expect("feed"),
694            [PartitionId(1)]
695        );
696        assert_eq!(
697            binds(&feed_one, &one).expect("no feed axis"),
698            [PartitionId(0)]
699        );
700
701        // Channel widths, bit for bit.
702        let mut narrow = observed(vec![session("p", None)], vec![partition(0, &[0], &[0])]);
703        narrow.partitions[0].channel_widths_hz[1] = 0.9e8;
704        assert!(binds(&narrow, &one).is_err(), "another channel width");
705    }

717    #[test]
718    fn binding_matches_content_and_refuses_ambiguity() {
719        let mut a = Catalogue {
720            antennas: vec![antenna("W1", 1.0), antenna("W2", 2.0)],
721            ..Catalogue::default()
722        };
723        let mut b = Catalogue {
724            antennas: vec![antenna("W2", 2.0), antenna("W1", 1.0 + 1e-4)],
725            ..Catalogue::default()
726        };
727        b.antennas[0].name = "renamed".to_string();
728        let tolerance = BindingTolerance {
729            position_m: 1e-3,
730            direction_rad: 0.0,
731        };
732        let binding = a.bind_to(&b, &tolerance).expect("binds");
733        assert_eq!(binding.antennas, vec![AntennaId(1), AntennaId(0)]);
734        assert!(
735            a.bind_to(&b, &BindingTolerance::EXACT).is_err(),
736            "1e-4 m off"
737        );
738        b.antennas.push(antenna("W1", 1.0));
739        assert!(
740            a.bind_to(&b, &tolerance).is_err(),
741            "two W1 within tolerance"
742        );
743        a.antennas.push(antenna("W9", 9.0));
744        assert!(a.bind_to(&b, &tolerance).is_err(), "W9 matches nothing");
745    }
