2//! CASA's extend mode (#931): existing flags grown across time, frequency
3//! and correlation, as CASA 6.7.6.14's extension agent grows them
4//! (`FlagAgentExtension.cc`, `computeAntennaPairFlags`).
5//!
6//! The agent works on one baseline of one visibility-iterator chunk at a time:
7//! 1. Grow from existing flags to neighboring times and channels.
8//! 2. Per correlation, a timestep whose flagged channels reach `growfreq`
9//!    percent is flagged in every channel, and a channel whose flagged
10//!    timesteps reach `growtime` percent is flagged at every timestep. Both
11//!    counts are from the flags after neighbor growth.
12//! 3. With `extendpols`, a channel and timestep flagged in one
13//!    correlation is flagged in every correlation.
14//!
15//! The agent reads flags only, so the data column plays no part.
16

25pub(super) fn apply(
26    ms: &mut MeasurementSet,
27    selected: &[usize],
28    channels: Option<&BTreeMap<i32, ChannelSelection>>,
29    correlations: Option<&CorrelationSelection>,
30    request: &FlagDataRequest,
31    path: &str,
32) -> Result<ChangeSet, FlaggingError> {
33    let mut changed = ChangeSet::default();
34    if !valid_request(request) {
35        changed.agent_skipped = true;
36        return Ok(changed);
37    }
38    let timed =
39        ordered_baseline_chunks(ms, selected, request.time_chunk, request.combinescans, path)?;
40    // Stage changed cells only. Later invalid rows must not leave a
41    // caller with a partially updated MeasurementSet open in memory.
42    let mut pending = BTreeMap::new();
43    for baseline in timed.chunk_by(|a, b| a.baseline_key() == b.baseline_key()) {
44        let rows = baseline.iter().map(|row| row.row).collect::<Vec<_>>();
45        let mut loaded = load_samples(
46            ms,
47            &rows,
48            channels,
49            None,
50            correlations,
51            request.casa_row_selection,
52            None,
53        )?;
54        let grown = grow(&loaded.samples, request);
55        let (touched, changed_samples) = stage_flag_mask(
56            &loaded.samples,
57            &grown,
58            &mut loaded.flags_by_row,
59            request.action.value(),
60        );
61        changed.changed_samples += changed_samples;
62        loaded.flags_by_row.retain(|row, _| touched.contains(row));
63        pending.append(&mut loaded.flags_by_row);
64    }
65    let touched = pending.keys().copied().collect();
66    write_touched_flag_updates(
67        ms,
68        pending,
69        touched,
70        |_| !request.casa_row_selection,
71        &mut changed,
72    )?;
73    Ok(changed)
74}

76pub(super) fn valid_request(request: &FlagDataRequest) -> bool {
77    // The CASA constructor uses these comparisons directly. A NaN disables
78    // its percentage test, so neighbor growth and polarization can still run.
79    let invalid = if request.extension_enabled {
80        [
81            ("growtime", request.growtime),
82            ("growfreq", request.growfreq),
83        ]
84        .into_iter()
85        .find(|(_, value)| !value.is_nan() && !(0.0..=100.0).contains(value))
86        .map(|(parameter, value)| {
87            format!("Unsupported value for {parameter}:{value}. Allowed range is 0.0 through 100.0")
88        })
89    } else {
90        Some("Parameter 'extendpols' must be of type 'bool'".to_owned())
91    };
92    if let Some(reason) = invalid {
93        tracing::warn!("Cannot create agent Extend: {reason}");
94        tracing::error!("There is no agent to run in list");
95        return false;
96    }
97    true
98}

100/// Flag mask for one baseline's selected samples in time order.
101pub(super) fn grow(samples: &[Sample], request: &FlagDataRequest) -> Vec<bool> {
102    let mut flags = samples.iter().map(|sample| sample.flag).collect::<Vec<_>>();
103    let value = request.action.value();
104    if value && (request.growaround || request.flagneartime || request.flagnearfreq) {
105        grow_neighbors(samples, request, &mut flags);
106    }
107    grow_across_time_and_frequency(samples, request, &mut flags);
108    if request.extendpols {
109        extend_across_correlations(samples, &mut flags, value);
110    }
111    if !value {
112        for (flag, sample) in flags.iter_mut().zip(samples) {
113            *flag = sample.flag && !*flag;
114        }
115    }
116    flags
117}

119fn grow_neighbors(samples: &[Sample], request: &FlagDataRequest, flags: &mut [bool]) {
120    // Channel slots follow the selection's order, including gaps and
121    // repeated references to the same stored cell in overlapping ranges.
122    let mut positions = Vec::with_capacity(samples.len());
123    for (time, row) in samples.chunk_by(|a, b| a.row == b.row).enumerate() {
124        for plane in row.chunk_by(|a, b| a.corr == b.corr) {
125            positions.extend((0..plane.len()).map(|channel| (plane[0].corr, time, channel)));
126        }
127    }
128    let original = samples
129        .iter()
130        .zip(&positions)
131        .filter_map(|(sample, &position)| sample.flag.then_some(position))
132        .collect::<BTreeSet<_>>();
133    let physical_cell = |sample: &Sample| (sample.row, sample.corr, sample.chan);
134    let mut grown = BTreeSet::new();
135    for (sample, &(corr, time, channel)) in samples.iter().zip(&positions) {
136        if sample.flag {
137            continue;
138        }
139        let neighbor = |dt: isize, dc: isize| {
140            time.checked_add_signed(dt)
141                .zip(channel.checked_add_signed(dc))
142                .is_some_and(|(t, c)| original.contains(&(corr, t, c)))
143        };
144        if (request.flagneartime && (neighbor(-1, 0) || neighbor(1, 0)))
145            || (request.flagnearfreq && (neighbor(0, -1) || neighbor(0, 1)))
146            || (request.growaround
147                && (-1..=1)
148                    .flat_map(|dt| (-1..=1).map(move |dc| (dt, dc)))
149                    .filter(|&(dt, dc)| neighbor(dt, dc))
150                    .count()
151                    > 4)
152        {
153            grown.insert(physical_cell(sample));
154        }
155    }
156    // All references to a flag see the same write before percentage growth.
157    for sample in samples {
158        flags[sample.index] |= grown.contains(&physical_cell(sample));
159    }
160}

162/// Percentage growth reads the original counts after the neighbor pass.
163fn grow_across_time_and_frequency(
164    baseline: &[Sample],
165    request: &FlagDataRequest,
166    flags: &mut [bool],
167) {
168    if !(request.growtime > 0.0 || request.growfreq > 0.0) {
169        return;
170    }
171    // The flagged counts per correlation and timestep, over its channels,
172    // and per correlation and channel, over its timesteps.
173    let mut by_timestep = BTreeMap::<(usize, usize), (usize, usize)>::new();
174    let mut by_channel = BTreeMap::<(usize, usize), (usize, usize)>::new();
175    for sample in baseline {
176        for (counts, key) in [
177            (&mut by_timestep, (sample.corr, sample.row)),
178            (&mut by_channel, (sample.corr, sample.chan)),
179        ] {
180            let (flagged, total) = counts.entry(key).or_default();
181            *flagged += usize::from(flags[sample.index]);
182            *total += 1;
183        }
184    }
185    let reaches = |(flagged, total): (usize, usize), percent: f64| {
186        percent > 0.0 && 100.0 * flagged as f64 / total as f64 >= percent
187    };
188    for sample in baseline {
189        if reaches(by_timestep[&(sample.corr, sample.row)], request.growfreq)
190            || reaches(by_channel[&(sample.corr, sample.chan)], request.growtime)
191        {
192            flags[sample.index] = request.action.value();
193        }
194    }
195}

197/// Flag every correlation of a channel and timestep with any flagged correlation.
198fn extend_across_correlations(baseline: &[Sample], flags: &mut [bool], value: bool) {
199    let mut flagged = BTreeMap::<(usize, usize), bool>::new();
200    for sample in baseline {
201        *flagged.entry((sample.row, sample.chan)).or_default() |= flags[sample.index];
202    }
203    for sample in baseline {
204        if flagged[&(sample.row, sample.chan)] {
205            flags[sample.index] = value;
206        }
207    }
208}

214    /// Two timesteps of one baseline, XX and YY by eight channels.
215    fn cube(flagged: &[(usize, usize, usize)]) -> Vec<Sample> {
216        let mut samples = Vec::new();
217        for row in 0..2 {
218            for corr in 0..2 {
219                for chan in 0..8 {
220                    samples.push(Sample {
221                        index: samples.len(),
222                        row,
223                        corr,
224                        chan,
225                        amp: 0.0,
226                        real: 0.0,
227                        imag: 0.0,
228                        flag: flagged.contains(&(row, corr, chan)),
229                        field: 0,
230                        spw: 0,
231                        scan: 1,
232                        ant1: 0,
233                        ant2: 1,
234                    });
235                }
236            }
237        }
238        samples
239    }

241    /// Flagged channels per correlation of `row`.
242    fn per_correlation(samples: &[Sample], flags: &[bool], row: usize) -> [usize; 2] {
243        let mut counts = [0; 2];
244        for sample in samples.iter().filter(|sample| sample.row == row) {
245            counts[sample.corr] += usize::from(flags[sample.index]);
246        }
247        counts
248    }

259    #[test]
260    fn growth_counts_each_correlation_on_its_own_then_extends_polarizations() {
261        // Astra's #997 F3: four of XX's eight channels flagged on row 0,
262        // none of YY's. CASA 6.7.6.14 grows XX's row to [8, 0], and with
263        // extendpols to [8, 8]; pooling the correlations gave [4, 0] and
264        // extending before growing gave [4, 4].
265        let samples = cube(&[(0, 0, 0), (0, 0, 1), (0, 0, 2), (0, 0, 3)]);
266        let grown = grow(&samples, &request(0.0, 50.0, false));
267        assert_eq!(per_correlation(&samples, &grown, 0), [8, 0]);
268        assert_eq!(per_correlation(&samples, &grown, 1), [0, 0]);
269        let grown = grow(&samples, &request(0.0, 50.0, true));
270        assert_eq!(per_correlation(&samples, &grown, 0), [8, 8]);
271        assert_eq!(per_correlation(&samples, &grown, 1), [0, 0]);
272    }

274    #[test]
275    fn growth_in_time_counts_each_correlations_timesteps() {
276        // Channel 5 of XX is flagged on one of the two timesteps: 50% of
277        // XX's timesteps, but 25% of the channel's samples over both
278        // correlations.
279        let samples = cube(&[(0, 0, 5)]);
280        let grown = grow(&samples, &request(50.0, 0.0, false));
281        let flagged = |row, corr, chan| {
282            let sample = samples
283                .iter()
284                .find(|sample| (sample.row, sample.corr, sample.chan) == (row, corr, chan))
285                .unwrap();
286            grown[sample.index]
287        };
288        assert!(flagged(1, 0, 5));
289        assert!(!flagged(1, 1, 5) && !flagged(0, 1, 5));
290        assert_eq!(grown.iter().filter(|flag| **flag).count(), 2);
291    }

293    #[test]
294    fn both_growths_count_the_flags_before_either_grows() {
295        // XX row 0 has 4 of 8 channels flagged, so growfreq=50 fills it;
296        // channel 7 was flagged on no timestep before, so growtime=50 must
297        // not then flag it on row 1.

301        // Channels 0 to 3 of XX reach 50% of timesteps and grow to row 1.
