18/// One sample's coordinates, assembled on the stack.

23    /// The baseline.
24    pub baseline: OrderedBaseline,
25    /// The field.
26    pub field: FieldId,
27}

29/// A per-sample quantity in the form the store shares it: one value per
30/// sample, one per time shared by its baselines, or one for the whole batch.
31/// [`Self::get`] resolves any of them for a sample, so a consumer needs no
32/// rule of either format.
33#[derive(Clone, Copy, Debug)]
34pub enum Shared<'a, T> {
35    /// One value per sample (MSv2 rows; MSv4 `[time, baseline]` variables).
36    PerSample(ArrayView1<'a, T>),
37    /// One value per time, shared by its baselines.
38    PerTime(ArrayView1<'a, T>),
39    /// One value for every sample (MSv4 `time.integration_time`).
40    Constant(T),
41}
42
43impl<'a, T: Copy> Shared<'a, T> {
44    /// The value of `sample`, in a batch with `samples_per_time` samples
45    /// per time ([`VisibilityBatch::samples_per_time`]).
46    ///
47    /// # Panics
48    ///
49    /// If `sample` is out of range.
50    #[must_use]
51    pub fn get(&self, sample: usize, samples_per_time: usize) -> T {
52        match self {
53            Self::PerSample(values) => values[sample],
54            Self::PerTime(values) => values[sample / samples_per_time],
55            Self::Constant(value) => *value,
56        }
57    }
58
59    fn window(&self, samples: Range<usize>, times: Range<usize>) -> Self {
60        match self {
61            Self::PerSample(values) => Self::PerSample(values.slice_move(s![samples])),
62            Self::PerTime(values) => Self::PerTime(values.slice_move(s![times])),
63            Self::Constant(value) => Self::Constant(*value),
64        }
65    }
66}

68/// The samples' time auxiliaries, each present when the request asked for
69/// it. Times are in the batch's time scale and origin.
70#[derive(Clone, Copy, Debug, Default)]
71pub struct SampleTiming<'a> {
72    /// Time centroid of each sample: one entry per sample, same scale and
73    /// origin as [`VisibilityBatch::time`].
74    pub centroid: Option<EpochView<'a>>,
75    /// Integration interval in seconds.
76    pub interval_s: Option<Shared<'a, f64>>,
77    /// Effective integration time in seconds.
78    pub exposure_s: Option<Shared<'a, f64>>,
79}
80
81impl SampleTiming<'_> {
82    fn window(&self, samples: Range<usize>, times: Range<usize>) -> Self {
83        Self {
84            centroid: self
85                .centroid
86                .map(|centroid| centroid.slice(samples.clone())),
87            interval_s: self
88                .interval_s
89                .map(|interval| interval.window(samples.clone(), times.clone())),
90            exposure_s: self
91                .exposure_s
92                .map(|exposure| exposure.window(samples, times)),
93        }
94    }
95}

97/// The borrowed payload of one data group.
98///
99/// A role is `None` only when the request did not ask for it; a role the
100/// request needs but the store lacks fails binding. Every array is
101/// `[sample, channel, correlation]` over the delivered correlations. The
102/// weights of a store that holds one weight per correlation (MSv2 `WEIGHT`)
103/// are the same view with a zero stride along the channel axis: no copy,
104/// and `as_slice()` is `None` on it. `flags` is the stored flag state with
105/// absent acquisitions folded in (MSv2 `FLAG_ROW` and padding; in MSv4 a
106/// non-finite stored weight); [`VisibilityBatch::validity`] is the one rule
107/// that decides whether a value is used.
108#[derive(Clone, Copy, Debug)]
109pub struct DataGroupView<'a> {
110    /// The group's catalogue entry: name, units, UVW frame.
111    pub entry: &'a DataGroupEntry,
112    /// Visibilities.
113    pub values: Option<ArrayView3<'a, Complex32>>,
114    /// Flags, absent acquisitions folded in; true excludes the value.
115    pub flags: Option<ArrayView3<'a, bool>>,
116    /// Weights.
117    pub weights: Option<ArrayView3<'a, f32>>,
118    /// UVW per sample, in the data group's effective frame
119    /// ([`UvwFrame`](crate::UvwFrame)).
120    pub uvw: Option<UvwView<'a>>,
121}
122
123impl DataGroupView<'_> {
124    fn window(&self, samples: Range<usize>, channels: Range<usize>) -> Self {
125        Self {
126            entry: self.entry,
127            values: self
128                .values
129                .map(|values| values.slice_move(s![samples.clone(), channels.clone(), ..])),
130            flags: self
131                .flags
132                .map(|flags| flags.slice_move(s![samples.clone(), channels.clone(), ..])),
133            weights: self
134                .weights
135                .map(|weights| weights.slice_move(s![samples.clone(), channels, ..])),
136            uvw: self.uvw.map(|uvw| uvw.slice(samples)),
137        }
138    }
139}

141/// Whether each value of a batch is used: the contract's single validity
142/// rule, the same for every store.
143///
144/// A value `(sample, channel, correlation)` is excluded when its delivered
145/// flag is set or, among the roles read, its visibility or weight is not
146/// finite. A delivered flag already marks every absent acquisition, in
147/// every store: MSv2 `FLAG_ROW` and the cells the MSv2 adapter pads (no row
148/// filled them: flagged, values and weights NaN) are flagged, and the MSv4 adapter
149/// folds a non-finite stored weight into the flag, because XRADIO pads a
150/// cell no row filled with NaN and leaves its `FLAG` false. So the rule
151/// gives one answer for one observation whichever store holds it and
152/// whichever roles are read, a flags-only read included, and a consumer
153/// applies no format-specific rule; the values and weights catch corrupt
154/// values besides.
155#[derive(Clone, Copy, Debug)]
156pub struct Validity<'a> {
157    flags: Option<ArrayView3<'a, bool>>,
158    values: Option<ArrayView3<'a, Complex32>>,
159    weights: Option<ArrayView3<'a, f32>>,
160}
161
162impl Validity<'_> {
163    /// Whether `(sample, channel, correlation)` is excluded.
164    ///
165    /// # Panics
166    ///
167    /// If an index is out of range.
168    #[must_use]
169    pub fn excluded(&self, sample: usize, channel: usize, correlation: usize) -> bool {
170        let at = [sample, channel, correlation];
171        self.flags.is_some_and(|flags| flags[at])
172            || self.values.is_some_and(|values| !values[at].is_finite())
173            || self.weights.is_some_and(|weights| !weights[at].is_finite())
174    }
175
176    /// The rule evaluated for every element into `mask`, replacing its
177    /// contents: `[sample, channel, correlation]` in standard order, true
178    /// where excluded. For a consumer that reads the batch more than once
179    /// and would otherwise evaluate the rule each time.
180    pub fn mask_into(&self, mask: &mut Vec<bool>, shape: [usize; 3]) {
181        mask.clear();
182        mask.resize(shape.iter().product(), false);
183        if let Some(flags) = self.flags {
184            for (slot, flag) in mask.iter_mut().zip(flags.iter()) {
185                *slot = *flag;
186            }
187        }
188        if let Some(values) = self.values {
189            for (slot, value) in mask.iter_mut().zip(values.iter()) {
190                *slot |= !value.is_finite();
191            }
192        }
193        if let Some(weights) = self.weights {
194            for (slot, weight) in mask.iter_mut().zip(weights.iter()) {
195                *slot |= !weight.is_finite();
196            }
197        }
198    }
199}

201/// Whole times of one partition's time × baseline grid, their coordinates
202/// and one data group's payload, borrowed from the caller's
203/// [`BatchBuffers`] and the observation's [`Catalogue`].
204///
205/// Sample `t × B + b` is time `t` of the batch and baseline `b` of the
206/// partition's table ([`Self::baselines`]); the coordinates cost `T + B`
207/// entries, not `T × B`. Both stores deliver this shape: MSv4 holds it, the
208/// MSv2 adapter places rows on it at opening and pads the cells no row
209/// filled (flag true, values and weights NaN). Field and scan are per time.
210///
211/// Every view is filled and checked before the batch is delivered: reading
212/// an element performs no I/O or decoding. The borrow keeps the buffers
213/// from being refilled while any view of them is alive.
214///
215/// [`BatchBuffers`]: crate::BatchBuffers
216/// [`Catalogue`]: crate::Catalogue
217#[derive(Clone, Copy, Debug)]
218pub struct VisibilityBatch<'a> {
219    /// The observation's catalogue, which the batch was checked against:
220    /// every identity in the batch (partition, field, scan, antenna) names
221    /// an entry of it.
222    pub catalogue: &'a Arc<Catalogue>,
223    /// The partition the samples belong to.
224    pub partition: PartitionId,
225    /// The batch's times, in the partition's scale and origin.
226    pub time: EpochView<'a>,
227    /// The partition's baseline table, from the catalogue.
228    pub baselines: &'a [OrderedBaseline],
229    /// The field at each time.
230    pub field: ArrayView1<'a, FieldId>,
231    /// The scan at each time, when requested.
232    pub scan: Option<ArrayView1<'a, ScanId>>,
233    /// The samples' time auxiliaries.
234    pub timing: SampleTiming<'a>,
235    /// The partition's native channel of this batch's first channel.
236    pub first_channel: usize,
237    /// The channels' frequencies: a window of the catalogue's.
238    pub frequency: FrequencyView<'a>,
239    /// The correlation products along the last axis: the delivered ones, in
240    /// the partition's order.
241    pub correlations: &'a [Correlation],
242    /// The traversal's correlation axis these are: what a packed piece
243    /// shares rather than copies.
244    pub(crate) axis: &'a CorrelationAxis,
245    /// The requested data group.
246    pub group: DataGroupView<'a>,
247    /// The scientific-group fragments in this batch, in sample order. Only
248    /// a delivered batch has them; its compute windows have none.
249    pub fragments: &'a [GroupFragment],
250}
251
252impl<'a> VisibilityBatch<'a> {
253    /// Number of times.
254    #[must_use]
255    pub fn times(&self) -> usize {
256        self.time.len()
257    }
258
259    /// Samples per time: the partition's baselines.
260    #[must_use]
261    pub fn samples_per_time(&self) -> usize {
262        self.baselines.len()
263    }
264
265    /// Number of samples.
266    #[must_use]
267    pub fn samples(&self) -> usize {
268        self.times() * self.samples_per_time()
269    }
270
271    /// Number of channels.
272    #[must_use]
273    pub fn channel_count(&self) -> usize {
274        self.frequency.len()
275    }
276
277    /// The partition's native channels this batch holds.
278    #[must_use]
279    pub fn channels(&self) -> Range<usize> {
280        self.first_channel..self.first_channel + self.channel_count()
281    }
282
283    /// The shape of every payload array: `[samples, channels, correlations]`.
284    #[must_use]
285    pub fn shape(&self) -> [usize; 3] {
286        [
287            self.samples(),
288            self.channel_count(),
289            self.correlations.len(),
290        ]
291    }
292
293    /// Sample `sample`'s coordinates.
294    ///
295    /// # Panics
296    ///
297    /// If `sample` is out of range.
298    #[must_use]
299    pub fn sample(&self, sample: usize) -> SampleCoordinate {
300        let per_time = self.samples_per_time();
301        let (time, baseline) = (sample / per_time, sample % per_time);
302        SampleCoordinate {
303            time_offset_s: self.time.offsets_s()[time],
304            baseline: self.baselines[baseline],
305            field: self.field[time],
306        }
307    }
308
309    /// Sample `sample`'s integration interval in seconds, if requested.
310    #[must_use]
311    pub fn interval_s(&self, sample: usize) -> Option<f64> {
312        self.timing
313            .interval_s
314            .map(|interval| interval.get(sample, self.samples_per_time()))
315    }
316
317    /// Sample `sample`'s effective integration time in seconds, if
318    /// requested.
319    #[must_use]
320    pub fn exposure_s(&self, sample: usize) -> Option<f64> {
321        self.timing
322            .exposure_s
323            .map(|exposure| exposure.get(sample, self.samples_per_time()))
324    }
325
326    /// Sample `sample`'s scan, if requested.
327    #[must_use]
328    pub fn scan(&self, sample: usize) -> Option<ScanId> {
329        self.scan.map(|scan| scan[sample / self.samples_per_time()])
330    }
331
332    /// The batch's validity rule ([`Validity`]).
333    #[must_use]
334    pub fn validity(&self) -> Validity<'a> {
335        Validity {
336            flags: self.group.flags,
337            values: self.group.values,
338            weights: self.group.weights,
339        }
340    }
341
342    /// A compute window of channels `range` (relative to this batch's
343    /// channels) of every sample: views of the same buffers, nothing copied.
344    /// The window carries no fragments: group completion stays with the
345    /// delivered batch, so splitting a batch among work units cannot
346    /// announce a group's end more than once.
347    ///
348    /// # Errors
349    ///
350    /// If `range` is outside the batch's channels.
351    pub fn channel_window(&self, range: Range<usize>) -> Result<Self, ObservationError> {
352        if range.start > range.end || range.end > self.channel_count() {
353            return Err(ObservationError::InvalidRequest {
354                reason: format!(
355                    "channel window {range:?} outside the batch's {} channels",
356                    self.channel_count()
357                ),
358            });
359        }
360        Ok(Self {
361            first_channel: self.first_channel + range.start,
362            frequency: self.frequency.slice(range.clone()),
363            group: self.group.window(0..self.samples(), range),
364            fragments: &[],
365            ..*self
366        })
367    }
368
369    /// A compute window of times `range` (relative to this batch's times),
370    /// every channel. Like a channel window it carries no fragments.
371    ///
372    /// # Errors
373    ///
374    /// If `range` is outside the batch's times.
375    pub fn time_window(&self, range: Range<usize>) -> Result<Self, ObservationError> {
376        if range.start > range.end || range.end > self.times() {
377            return Err(ObservationError::InvalidRequest {
378                reason: format!(
379                    "time window {range:?} outside the batch's {} times",
380                    self.times()
381                ),
382            });
383        }
384        let per_time = self.samples_per_time();
385        let samples = range.start * per_time..range.end * per_time;
386        Ok(Self {
387            time: self.time.slice(range.clone()),
388            field: self.field.slice_move(s![range.clone()]),
389            scan: self.scan.map(|scan| scan.slice_move(s![range.clone()])),
390            timing: self.timing.window(samples.clone(), range),
391            group: self.group.window(samples, 0..self.channel_count()),
392            fragments: &[],
393            ..*self
394        })
395    }
396
397    /// A compute window of samples `range`, which must be whole times (as a
398    /// [`GroupFragment`]'s samples are): [`Self::time_window`] in sample
399    /// units.
400    ///
401    /// # Errors
402    ///
403    /// If `range` is outside the batch or splits a time.
404    pub fn sample_window(&self, range: Range<usize>) -> Result<Self, ObservationError> {
405        let per_time = self.samples_per_time().max(1);
406        if !range.start.is_multiple_of(per_time) || !range.end.is_multiple_of(per_time) {
407            return Err(ObservationError::InvalidRequest {
408                reason: format!("sample window {range:?} splits a time of {per_time} samples"),
409            });
410        }
411        self.time_window(range.start / per_time..range.end / per_time)
412    }
