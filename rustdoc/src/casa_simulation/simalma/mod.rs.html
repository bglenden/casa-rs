19/// CASA-named task parameters and the simulator's explicit environment.
20pub struct SimalmaRequest {
21    /// Sparse CASA simalma parameters; defaults come from the shared catalog.
22    pub parameters: Map<String, Value>,
23    /// Antenna configuration repositories, as for simobserve.
24    pub data_roots: Vec<PathBuf>,
25    /// Measures, clock, memory, workers and backend.
26    pub options: SimulatorOptions,
27}
28
29fn failure(error: impl std::fmt::Display) -> SimobserveError {
30    SimobserveError::raised(format!("simalma: {error}"))
31}
32
33struct Parameters(Map<String, Value>);
34impl Parameters {
35    fn text(&self, name: &str) -> &str {
36        self.0[name].as_str().expect("resolved string")
37    }
38    fn flag(&self, name: &str) -> bool {
39        self.0[name].as_bool().expect("resolved flag")
40    }
41    fn integer(&self, name: &str) -> i64 {
42        self.0[name].as_i64().expect("resolved integer")
43    }
44    fn strings(&self, name: &str) -> Vec<String> {
45        match &self.0[name] {
46            Value::String(one) => vec![one.clone()],
47            Value::Array(many) => many
48                .iter()
49                .map(|v| v.as_str().expect("resolved string").to_owned())
50                .collect(),
51            _ => unreachable!("resolved strings"),
52        }
53    }
54}

56/// Simulate each requested array and TP antenna, image, and combine them.
57///
58/// # Errors
59/// Invalid input/configuration/time, an existing protected project or input,
60/// resource refusal, or a native observation/imaging failure.
61pub fn simalma(request: &SimalmaRequest) -> Result<SimobserveOutcome, SimobserveError> {
62    let resources = request.options.resources;
63    charge_process_caches(resources.host(), resources.policy()).map_err(failure)?;
64    let _planning = admit(
65        resources.host(),
66        resources.policy(),
67        &Demand {
68            phase: "ALMA simulation planning",
69            memory: (2 << 20)
70                + request
71                    .parameters
72                    .iter()
73                    .map(|(name, value)| {
74                        128 + name.len() as u64 + crate::simanalyze::parameter_bytes(value)
75                    })
76                    .sum::<u64>()
77                    .saturating_mul(4),
78        },
79    )
80    .map_err(failure)?;
81    let values = resolve_plain_values(
82        builtin_surface_bundle("simalma").map_err(failure)?,
83        std::env::current_dir().map_err(failure)?,
84        &request.parameters,
85    )
86    .map_err(failure)?;
87    let p = Parameters(values);
88    let mut plan = plan::prepare(&p, request)?;
89    let mut outcome = SimobserveOutcome {
90        directory: plan.directory.clone(),
91        stem: plan.project.clone(),
92        products: Vec::new(),
93        warnings: Vec::new(),
94        plots: Vec::new(),
95    };
96    if p.flag("dryrun") {
97        outcome.warnings.extend(plan.arrays.iter().map(|array| {
98            format!(
99                "dryrun: simulate {} for {:.6} minutes",
100                array.configuration, array.minutes
101            )
102        }));
103        if plan.tp_antennas > 0 {
104            outcome.warnings.push(format!(
105                "dryrun: simulate {} TP antennas for {} minutes each",
106                plan.tp_antennas,
107                plan.tp_minutes.trunc()
108            ));
109        }
110        return Ok(outcome);
111    }
112    if p.text("graphics") != "none" {
113        outcome.warnings.push("simalma graphics remain in simulation Phase 6i (#841); numerical products are complete independently of graphics".into());
114    }
115    plan::publish_directory(&p, &plan)?;
116    for array in &plan.arrays {
117        let _sky = admit(
118            resources.host(),
119            resources.policy(),
120            &Demand {
121                phase: "ALMA observation sky preparation",
122                memory: plan.model.preparation_bytes,
123            },
124        )
125        .map_err(failure)?;
126        let mut observation = plan.observation.clone();
127        observation.antennalist = array.configuration.clone();
128        observation.totaltime = format!("{:.6}min", array.minutes);
129        observation.correlator = array.correlator.clone();
130        let result = simobserve(&crate::simobserve::SimobserveRequest {
131            parameters: observation,
132            data_roots: request.data_roots.clone(),
133            options: request.options.clone(),
134        })?;
135        outcome.products.extend(result.products);
136        outcome.warnings.extend(result.warnings);
137    }
138    if plan.tp_antennas > 0 {
139        let (mut observation, _pointings) = pointings::observation(&p, &plan, resources)?;
140        for antenna in 0..plan.tp_antennas {
141            let _sky = admit(
142                resources.host(),
143                resources.policy(),
144                &Demand {
145                    phase: "ALMA TP sky preparation",
146                    memory: plan.model.preparation_bytes,
147                },
148            )
149            .map_err(failure)?;
150            observation.sdant = antenna as i64;
151            // Independent streams per dish. Random sample identity is not a
152            // CASA contract; the native simulator owns its seeded stream.
153            observation.seed = i64::from(crate::DEFAULT_SEED) + antenna as i64;
154            let result = simobserve(&crate::simobserve::SimobserveRequest {
155                parameters: observation.clone(),
156                data_roots: request.data_roots.clone(),
157                options: request.options.clone(),
158            })?;
159            for mut path in result.products {
160                if plan.tp_antennas > 1 && path.to_string_lossy().ends_with(".ms") {
161                    let mut target = path.as_os_str().to_os_string();
162                    target.push(format!(".Ant{antenna}"));
163                    let target = PathBuf::from(target);
164                    std::fs::rename(&path, &target).map_err(failure)?;
165                    path = target;
166                }
167                if !outcome.products.contains(&path) {
168                    outcome.products.push(path);
169                }
170            }
171            let noisy = if plan.noisy { ".noisy" } else { "" };
172            let suffix = if plan.tp_antennas > 1 {
173                format!(".Ant{antenna}")
174            } else {
175                String::new()
176            };
177            plan.tp_vis.push(
178                plan.directory
179                    .join(format!("{}{noisy}.sd.ms{suffix}", plan.tp_stem)),
180            );
181        }
182    }
183    if p.flag("image") {
184        outcome
185            .products
186            .extend(imaging::execute(&p, &plan, resources)?);
187    }
188    outcome.products.sort();
189    outcome.products.dedup();
190    std::fs::write(
191        plan.directory
192            .join(format!("{}.simalma.report.txt", plan.project)),
193        format!(
194            "simalma arrays={} tp_antennas={} products={}\n",
195            plan.arrays.len(),
196            plan.tp_antennas,
197            outcome.products.len()
198        ),
199    )
200    .map_err(failure)?;
201    Ok(outcome)
202}
