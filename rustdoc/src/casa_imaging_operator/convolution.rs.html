12/// Taps of one convolution-function cell in the layout the kernels read.
13///
14/// Fine-offset rows are indexed by [`CellLocation`](crate::CellLocation)'s
15/// `ox`/`oy`, which run over `0..=oversampling`, so every layout holds
16/// `oversampling + 1` rows per axis and `oversampling` must be even.
17#[derive(Clone, Copy, Debug)]
18pub enum TapLayout<'a> {
19    /// Separable real kernel (standard spheroidal). `rows[o * support + i]`
20    /// is tap `i`, at grid offset `i − support/2` from the anchor, for fine
21    /// offset row `o`; one fractional offset is one contiguous row
22    /// (oversampling-major, Obit `ConvFunc`).
23    SeparableReal {
24        /// `(oversampling + 1) × support` tap values.
25        rows: &'a [f32],
26        /// Taps per axis (odd).
27        support: u16,
28        /// Fine offsets per cell (even).
29        oversampling: u16,
30    },
31    /// Dense complex kernel (W, AW, mosaic) ordered
32    /// `[oy][ox][mueller][iy][ix]`, x fastest, with one contiguous
33    /// `mueller × sy × sx` tile per fractional offset. Tap `(ix, iy)` sits
34    /// at grid offset `(ix − sx/2, iy − sy/2)` from the anchor.
35    Dense {
36        /// `(oversampling + 1)² × mueller_planes × sy × sx` tap values.
37        data: &'a [Complex32],
38        /// Taps per axis `[sx, sy]` (odd).
39        support: [u16; 2],
40        /// Fine offsets per cell (even).
41        oversampling: u16,
42        /// Mueller planes per tile; 1 for a scalar kernel.
43        mueller_planes: u8,
44    },
45}
46
47impl TapLayout<'_> {
48    /// Taps on each side of the anchor, `[x, y]`.
49    #[must_use]
50    pub fn half_support(&self) -> [u16; 2] {
51        match self {
52            Self::SeparableReal { support, .. } => [support / 2, support / 2],
53            Self::Dense { support, .. } => [support[0] / 2, support[1] / 2],
54        }
55    }
56
57    /// Fine offsets per cell.
58    #[must_use]
59    pub fn oversampling(&self) -> u16 {
60        match self {
61            Self::SeparableReal { oversampling, .. } | Self::Dense { oversampling, .. } => {
62                *oversampling
63            }
64        }
65    }
66
67    /// Mueller planes per cell.
68    #[must_use]
69    pub fn mueller_planes(&self) -> u8 {
70        match self {
71            Self::SeparableReal { .. } => 1,
72            Self::Dense { mueller_planes, .. } => *mueller_planes,
73        }
74    }
75
76    /// The kernel norm `Σ (conjugate ? conj(t) : t)` of Mueller plane
77    /// `mueller` at the fine offsets of `location`, summed in `f64` without
78    /// the pointing ramp: the forward normalisation every backend divides
79    /// by and whose magnitude `sumwt` accumulates (CASA
80    /// `AWVisResampler::faccumulateFromGrid`).
81    #[must_use]
82    pub fn norm(&self, location: CellLocation, mueller: u8, conjugate: bool) -> Complex64 {
83        crate::cpu::kernel::norm(self, location, mueller, conjugate)
84    }
85}

87/// Dense taps of one cell in the [`TapLayout::Dense`] order, owned so a
88/// bounded cache can hand them out while it evicts others.
89#[derive(Clone, Debug, PartialEq)]
90pub struct DenseCell {
91    /// `(oversampling + 1)² × mueller_planes × sy × sx` tap values.
92    pub data: Box<[Complex32]>,
93    /// Taps per axis `[sx, sy]` (odd).
94    pub support: [u16; 2],
95    /// Fine offsets per cell (even).
96    pub oversampling: u16,
97    /// Mueller planes per tile.
98    pub mueller_planes: u8,
99}
100
101impl DenseCell {
102    /// The cell as a tap layout.
103    #[must_use]
104    pub fn layout(&self) -> TapLayout<'_> {
105        TapLayout::Dense {
106            data: &self.data,
107            support: self.support,
108            oversampling: self.oversampling,
109            mueller_planes: self.mueller_planes,
110        }
111    }
112
113    /// Bytes of the tap values.
114    #[must_use]
115    pub fn bytes(&self) -> usize {
116        std::mem::size_of_val(self.data.as_ref())
117    }
118}

120/// A caller-owned slot that keeps one cell of a bounded cache alive while
121/// the taps borrowed from it are in use.
122///
123/// Resident sets ignore it and lend their own storage. A set whose cells
124/// come and go ([`AwCatalog`](crate::AwCatalog)) parks the cell here, so
125/// the cache may drop its own reference without invalidating the borrow,
126/// and a worker holds at most one cell per slot beyond the cache's bound.
127#[derive(Clone, Debug, Default)]
128pub struct CellHold {
129    cell: Option<Arc<DenseCell>>,
130    /// The cell's key and the set's slot (imaging, weight, prediction cell)
131    /// when parked through [`Self::lend_keyed`]; a repeated key lends the
132    /// parked cell again without touching the set's cache.
133    keyed: Option<(CfKey, u8)>,
134}
135
136impl CellHold {
137    /// An empty slot.
138    #[must_use]
139    pub const fn new() -> Self {
140        Self {
141            cell: None,
142            keyed: None,
143        }
144    }
145
146    /// Park `cell` and lend its taps.
147    pub fn lend(&mut self, cell: Arc<DenseCell>) -> TapLayout<'_> {
148        self.keyed = None;
149        self.cell.insert(cell).layout()
150    }
151
152    /// Lend the taps of the cell parked for `key` in the set's `slot` (the
153    /// set numbers its kinds of cell), fetching it with `load` when another
154    /// cell is parked: rows of one key in sequence cost one fetch.
155    pub fn lend_keyed(
156        &mut self,
157        key: CfKey,
158        slot: u8,
159        load: impl FnOnce() -> Arc<DenseCell>,
160    ) -> TapLayout<'_> {
161        if self.keyed != Some((key, slot)) || self.cell.is_none() {
162            self.cell = Some(load());
163            self.keyed = Some((key, slot));
164        }
165        self.cell.as_ref().expect("a parked cell").layout()
166    }
167
168    /// Release the parked cell.
169    pub fn clear(&mut self) {
170        self.cell = None;
171        self.keyed = None;
172    }
173}

175/// How a set's kernels are normalised: what a prediction divides by and
176/// what `sumwt` accumulates, with `N` the sum of the w-conjugated taps a
177/// visibility polarization uses at its fine offset, without the pointing
178/// ramp (plan section 5.3; R1 on #650). CASA's machines differ, so the
179/// rule is a property of the set.
180#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
181pub enum KernelNormalisation {
182    /// Unit-sum taps: predictions are the gathered sums and `sumwt += W`
183    /// (CASA `GridFT`, `MosaicFT`: `fgridft.f`, `fmosaic.f` add the weight
184    /// without a kernel sum).
185    UnitSum,
186    /// Predictions are the gathered sums and `sumwt += W · Re N` (CASA
187    /// `WProjectFT`: `wprojgrid.f` accumulates `norm += real(cwt)` and adds
188    /// `weight · norm`; `dwgrid` does not divide).
189    RealSum,
190    /// A prediction divides once by `N` summed over its routed Mueller
191    /// planes and `sumwt += W · |N|` (CASA `AWVisResampler::DataToGrid`,
192    /// `GridToData` and `faccumulateFromGrid.f`).
193    KernelSum,
194}
195
196/// Which kernel Mueller plane serves each (grid polarization, visibility
197/// polarization) pair, per direction of the transform and sign of `w`.
198///
199/// Four `[grid pol][visibility pol]` tables; `None` skips a pair. CASA
200/// routes the two directions differently (`AWVisResampler`):
201/// `getConvFunc_p` selects the cell by the sign of `w` (`mNdx` for
202/// `w > 0`, `conjMNdx` otherwise; `GridToData` passes the two swapped),
203/// then `DataToGrid` reads the visibility of the selected cell's own hand
204/// (`muellerElement % nDataPol`) into the outer grid polarization, while
205/// `GridToData` reads the model grid of that hand into the outer
206/// visibility polarization. A forward table is therefore not a swap of an
207/// adjoint one. Tap conjugation is separate from the routing: adjoint
208/// taps are conjugated for `w > 0`, forward taps for `w ≤ 0`
209/// (`accumulateToGrid.inc`, `accumulateFromGrid.inc`).
210#[derive(Clone, Debug, PartialEq, Eq)]
211pub struct MuellerRouting {
212    /// Adjoint tables, indexed by `usize::from(w > 0)`.
213    pub adjoint: [Vec<Vec<Option<u8>>>; 2],
214    /// Forward tables, indexed by `usize::from(w > 0)`.
215    pub forward: [Vec<Vec<Option<u8>>>; 2],
216}
217
218impl MuellerRouting {
219    /// Scalar routing: Mueller plane 0 for every pair `pol_map` joins
220    /// (visibility polarization `p` feeds grid polarization `pol_map[p]`),
221    /// with identical tables for both directions and signs of w.
222    #[must_use]
223    pub fn scalar(pol_map: &[Option<u8>], grid_pols: usize) -> Self {
224        let direct = (0..grid_pols)
225            .map(|gpol| {
226                pol_map
227                    .iter()
228                    .map(|target| (*target == Some(gpol as u8)).then_some(0))
229                    .collect::<Vec<_>>()
230            })
231            .collect::<Vec<_>>();
232        Self {
233            adjoint: [direct.clone(), direct.clone()],
234            forward: [direct.clone(), direct],
235        }
236    }
237
238    /// Number of grid polarizations.
239    #[must_use]
240    pub fn grid_pols(&self) -> usize {
241        self.adjoint[0].len()
242    }
243
244    /// Number of visibility polarizations.
245    #[must_use]
246    pub fn visibility_pols(&self) -> usize {
247        self.adjoint[0].first().map_or(0, Vec::len)
248    }
249
250    /// The complete routing for one transform and sign of w.
251    #[must_use]
252    pub fn table(&self, w_positive: bool, forward: bool) -> &[Vec<Option<u8>>] {
253        if forward {
254            &self.forward[usize::from(w_positive)]
255        } else {
256            &self.adjoint[usize::from(w_positive)]
257        }
258    }
259}

261/// Per-row context a kernel set keys its cells on.
262#[derive(Clone, Copy, Debug, PartialEq)]
263pub struct RowContext {
264    /// The row's MeasurementSet `UVW` w in metres before the phase-centre
265    /// rotation, when the source rotated the row; `None` when the row's
266    /// uvw are the MeasurementSet's. The AW set keys its prediction cell
267    /// and parity on it ([`ConvolutionFunctionSet::prediction_w`]).
268    pub original_w_m: Option<f64>,
269    /// Row time in seconds (MeasurementSet `TIME`).
270    pub time_s: f64,
271    /// Antenna pair.
272    pub antennas: [u32; 2],
273    /// Antenna type of each antenna: an index into the kernel set's
274    /// antenna classes (CASA `HetArrayConvFunc` dish classes, AW baseline
275    /// types); 0 for a homogeneous array.
276    pub antenna_types: [u8; 2],
277    /// CASA's visibility polarization operator angle of each antenna in
278    /// radians: the negative of the physical parallactic angle.
279    pub parallactic_angle_rad: [f64; 2],
280    /// Field (pointing) identifier.
281    pub field: u32,
282    /// Spectral window of the row's data description; kernel sets with
283    /// per-window frequency cells key on it.
284    pub spectral_window: u32,
285}

287/// Image-domain gridding correction: one real vector per grid axis on each
288/// side of the operator, applied as `x[gx] · y[gy]` to the image the
289/// adjoint forms ([`Self::at`]) and to the model the forward transform
290/// reads ([`Self::model_at`]).
291///
292/// CASA's `GridFT` divides the model and the dirty image by the same
293/// response ([`Self::new`]); `MosaicFT` (`prepGridForDegrid`, `getImage`)
294/// and `WProjectFT` (`initializeToVis`, `getImage`) multiply the model by
295/// their sinc and divide the image by it, so the sides differ
296/// ([`Self::split`]).
297#[derive(Clone, Debug, PartialEq)]
298pub struct ImageCorrection {
299    x: Box<[f64]>,
300    y: Box<[f64]>,
301    model_x: Box<[f64]>,
302    model_y: Box<[f64]>,
303}
304
305impl ImageCorrection {
306    /// The same correction vectors on both sides, over the padded grid
307    /// axes.
308    #[must_use]
309    pub fn new(x: Vec<f64>, y: Vec<f64>) -> Self {
310        Self {
311            model_x: x.clone().into_boxed_slice(),
312            model_y: y.clone().into_boxed_slice(),
313            x: x.into_boxed_slice(),
314            y: y.into_boxed_slice(),
315        }
316    }
317
318    /// Different vectors for the image side (`x`, `y`) and the model side
319    /// (`model_x`, `model_y`).
320    #[must_use]
321    pub fn split(x: Vec<f64>, y: Vec<f64>, model_x: Vec<f64>, model_y: Vec<f64>) -> Self {
322        Self {
323            x: x.into_boxed_slice(),
324            y: y.into_boxed_slice(),
325            model_x: model_x.into_boxed_slice(),
326            model_y: model_y.into_boxed_slice(),
327        }
328    }
329
330    /// Image-side correction along the grid x axis.
331    #[must_use]
332    pub fn x(&self) -> &[f64] {
333        &self.x
334    }
335
336    /// Image-side correction along the grid y axis.
337    #[must_use]
338    pub fn y(&self) -> &[f64] {
339        &self.y
340    }
341
342    /// Model-side correction along the grid x axis.
343    #[must_use]
344    pub fn model_x(&self) -> &[f64] {
345        &self.model_x
346    }
347
348    /// Model-side correction along the grid y axis.
349    #[must_use]
350    pub fn model_y(&self) -> &[f64] {
351        &self.model_y
352    }
353
354    /// Image-side correction at grid cell `(gx, gy)`.
355    #[must_use]
356    pub fn at(&self, gx: usize, gy: usize) -> f64 {
357        self.x[gx] * self.y[gy]
358    }
359
360    /// Model-side correction at grid cell `(gx, gy)`.
361    #[must_use]
362    pub fn model_at(&self, gx: usize, gy: usize) -> f64 {
363        self.model_x[gx] * self.model_y[gy]
364    }
365}

367/// A kernel set: pure cell selection plus the taps, routing and correction
368/// the kernels read. Implementations: the standard spheroidal set, W-planes,
369/// the AW catalog and the mosaic primary beam.
370pub trait ConvolutionFunctionSet: Send + Sync {
371    /// Cell for one row at `freq_hz` and `w_lambda`. Pure; CASA-pinned
372    /// rounding rules for w-planes, frequency and parallactic-angle cells
373    /// live here.
374    fn key(&self, row: &RowContext, freq_hz: f64, w_lambda: f64) -> CfKey;
375
376    /// Imaging taps of a cell. A set that pages cells parks the cell in
377    /// `hold` and lends from it; resident sets lend their own storage.
378    fn taps<'s>(&'s self, key: CfKey, hold: &'s mut CellHold) -> TapLayout<'s>;
379
380    /// The largest [`TapLayout::half_support`] of any cell `key` can
381    /// return, `[x, y]`: the halo a tile needs so that every sample anchored
382    /// inside it keeps its whole support in the tile.
383    fn max_half_support(&self) -> [u16; 2];
384
385    /// `FT[PB²]` taps for the weight (sensitivity) image, placed at the uv
386    /// origin; `None` for sets without a weight image.
387    fn weight_taps<'s>(&'s self, key: CfKey, hold: &'s mut CellHold) -> Option<TapLayout<'s>>;
388
389    /// Mueller plane routing for every cell of the set.
390    fn mueller(&self) -> &MuellerRouting;
391
392    /// The w in wavelengths a prediction of `row`'s sample at `freq_hz`
393    /// keys its cell and tap conjugation on: the image-frame `w_lambda` the
394    /// gridding uses, for every set but AW. `AWVisResampler::GridToData`
395    /// reads the MeasurementSet w before the phase-centre rotation
396    /// (`vb_p->uvw()(2, irow)`) for the w-plane and the sign, while its u,
397    /// v and phasor are the rotated ones (`sgrid` on `uvw_p`);
398    /// `DataToGrid` reads the rotated w for both.
399    fn prediction_w(&self, _row: &RowContext, _freq_hz: f64, w_lambda: f64) -> f64 {
400        w_lambda
401    }
402
403    /// The paired image-domain correction.
404    fn image_correction(&self) -> &ImageCorrection;
405
406    /// The normalisation rule of the set's kernels.
407    fn normalisation(&self) -> KernelNormalisation;
408
409    /// Whether placements carry the pointing phase gradient
410    /// (`Placement::gradient`): the kernels encode a pointing offset from
411    /// the image centre as `e^{i(k·g)}` over the taps (mosaic, AW).
412    fn pointing_ramp(&self) -> bool;
413
414    /// The taps `Mode::Psf` grids a sample with at its uv position: the
415    /// imaging taps for `GridFT`, `WProjectFT` and `MosaicFT`; the AW
416    /// catalog grids its PSF with the weight cells (`AWProjectFT::
417    /// findConvFunction` maps `cfwts2_p` when `makingPSF`).
418    fn psf_taps<'s>(&'s self, key: CfKey, hold: &'s mut CellHold) -> TapLayout<'s> {
419        self.taps(key, hold)
420    }
421
422    /// The taps a prediction gathers with: the gridding taps for every set
423    /// but the AW catalog, whose `GridToData` reads the native-frequency
424    /// cell (`CFBuffer::nearestFreqNdx(spw, chan)` without `conjBeams`)
425    /// where `DataToGrid` read the conjugate one.
426    fn prediction_taps<'s>(&'s self, key: CfKey, hold: &'s mut CellHold) -> TapLayout<'s> {
427        self.taps(key, hold)
428    }
429
430    /// The half support a placement keyed `key` must keep inside the grid:
431    /// the largest of every tap layout the key can lend.
432    fn placement_half_support(&self, key: CfKey, hold: &mut CellHold) -> [u16; 2] {
433        self.taps(key, hold).half_support()
434    }
435
436    /// Whether a sample at `w_lambda` is gridded at all. `WProjectFT`'s
437    /// gridder drops a row whose plane index `nint(√(wScale·|w|))` lies
438    /// beyond the last plane (`wprojgrid.f` `owp` tests the unclamped
439    /// `loc(3)`); every other set accepts every `w` (the AW catalog clamps,
440    /// `CFBuffer::nearestWNdx`).
441    fn admits(&self, w_lambda: f64) -> bool {
442        let _ = w_lambda;
443        true
444    }
445
446    /// Whether the weight (sensitivity) image receives the image-side
447    /// correction. `AWProjectFT::getWeightImage` divides its average PB by
448    /// the sampling sinc as `getImage` does; `MosaicFT` publishes the
449    /// transform of its gridded weight functions as `skyCoverage_p` with
450    /// no correction at all, while its `getImage` divides the data and PSF
451    /// by the sinc.
452    fn corrects_weight_image(&self) -> bool {
453        true
454    }
