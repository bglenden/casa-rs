45/// A native A/W request or numerical calculation could not be completed.
46#[derive(Clone, Copy, Debug, Error, PartialEq, Eq)]
47pub enum NativeAwGenerationError {
48    /// Dish samples are not finite, regularly spaced, or physically valid.
49    #[error("EVLA dish surface requires finite, uniformly spaced radius/height/slope samples")]
50    InvalidSurface,
51    /// The requested receiver band is outside the implemented EVLA proof.
52    #[error("native EVLA aperture supports only L, S and C receiver bands (0.9 to 8 GHz)")]
53    UnsupportedFrequency,
54    /// Aperture geometry or numerical sampling is invalid.
55    #[error(
56        "native aperture grid requires a finite positive cell size, frequency, and even extent"
57    )]
58    InvalidGrid,
59    /// The caller did not supply the declared numerical workspace.
60    #[error("native A/W generation workspace does not fit the requested grid")]
61    WorkspaceMismatch,
62    /// A generated value or normalization is invalid.
63    #[error("native A/W generation produced an invalid numerical value")]
64    InvalidNumerics,
65}

68#[derive(Debug, Error)]
69pub enum AwCatalogError {

73        /// The directory or cell image.

75        /// What failed.
76        detail: String,
77    },

79    #[error("native EVLA generation: {0}")]

91fn cache_error(path: impl AsRef<Path>, detail: impl Into<String>) -> AwCatalogError {

108/// Header of one cell image, read at open (`CFCache::getCFParams`).
109#[derive(Clone, Debug)]
110struct CellHeader {
111    path: PathBuf,
112    /// `[nx, ny]` of the image.
113    shape: [usize; 2],
114    /// `Sampling`.
115    sampling: u16,
116    /// `Xsupport`, `Ysupport`.

118    /// `MuellerElement`.

140/// One (PA, frequency, w) group: its Mueller elements in the catalog's
141/// element order.
142#[derive(Clone, Debug)]
143struct Group {
144    cells: Vec<CellFiles>,
145    /// The largest declared half support over the group's imaging cells:
146    /// the imaging dense cell is padded to it, and `AWVisResampler`'s
147    /// `onGrid` test for a data row uses it (`CF Support: 4 (6)` in CASA's
148    /// log is the imaging support with the weight support in brackets).
149    imaging_half_support: [u16; 2],
150    /// The same over the weight cells, which the PSF and weight image use.
151    weight_half_support: [u16; 2],
152    /// The smallest declared half support over both kinds: the row filter
153    /// keeps every row some cell of the group fits.
154    smallest_half_support: [u16; 2],
155}
156
157impl Group {
158    fn new(cells: Vec<CellFiles>) -> Self {
159        let largest = |select: fn(&CellFiles) -> &CellHeader| {
160            cells.iter().map(select).fold([0_u16; 2], |half, header| {
161                [
162                    half[0].max(header.support[0]),
163                    half[1].max(header.support[1]),
164                ]
165            })
166        };
167        let imaging_half_support = largest(|cell| &cell.imaging);
168        let weight_half_support = largest(|cell| &cell.weight);
169        let smallest_half_support = cells
170            .iter()
171            .flat_map(|cell| [&cell.imaging, &cell.weight])
172            .fold([u16::MAX; 2], |half, header| {
173                [
174                    half[0].min(header.support[0]),
175                    half[1].min(header.support[1]),
176                ]
177            });
178        Self {
179            cells,
180            imaging_half_support,
181            weight_half_support,
182            smallest_half_support,
183        }
184    }
185}

187/// The support `locate_sample` tests a row against: the smallest cell the
188/// gridding group or the prediction group can select. Every mode re-tests
189/// the support of the cell it reads (`AWVisResampler::DataToGrid` and
190/// `GridToData` call `onGrid` with the selected cell's support), so the
191/// filter only has to keep every row CASA grids with any of them.
192fn placement_half_support(gridding: &Group, prediction: &Group) -> [u16; 2] {
193    let (g, p) = (
194        gridding.smallest_half_support,
195        prediction.smallest_half_support,
196    );
197    [g[0].min(p[0]), g[1].min(p[1])]
198}

214/// The bounded cell cache: least recently used eviction once the loaded
215/// cells exceed the bound; a cell a worker holds stays alive through its
216/// [`CellHold`] after eviction.

225/// The AW-projection kernel set over a CASA convolution-function cache.
226///
227/// Cells are keyed on the parallactic-angle cell, the frequency cell and
228/// the w-plane (`CfKey::group`); the Mueller elements live inside the cell
229/// as planes, routed by [`MuellerRouting`] with CASA's swapped tables for
230/// the conjugate baseline (`AWProjectFT::makeConjPolMap`). Prediction
231/// divides once by the kernel sum and `sumwt += W·|N|`
232/// ([`KernelNormalisation::KernelSum`]); the pointing ramp applies to
233/// every row ([`ConvolutionFunctionSet::pointing_ramp`]). A row whose
234/// parallactic angle lies between cells takes the nearest cell; CASA would
235/// rotate it (`AWConvFunc::rotateCF`, `rotatepastep`), which the frozen
236/// 360° steps of the EVLA cache make the same choice.
237///
238/// # Panics
239///
240/// [`ConvolutionFunctionSet::taps`] and [`ConvolutionFunctionSet::weight_taps`]
241/// are infallible: a cell image that was validated at open but cannot be
242/// read when a row first needs it is a catalog broken under the run, and
243/// the catalog panics with the cell's path rather than gridding without it.
244pub struct AwCatalog {
245    root: PathBuf,
246    indexing: AwIndexing,
247    pa_deg: Vec<f64>,
248    frequencies_hz: Vec<f64>,
249    w_values: Vec<f64>,
250    w_increment: f64,
251    mueller_elements: Vec<u32>,
252    groups: Vec<Group>,
253    oversampling: u16,
254    max_half_support: [u16; 2],
255    mueller: MuellerRouting,
256    correction: ImageCorrection,
257    lru: Mutex<Lru>,
258}

273impl AwCatalog {
274    /// Open the cache at `root`: read every `CFS_*`/`WTCFS_*` image's
275    /// coordinates and misc info (`CFCache::fillCFListFromDisk`), sort the
276    /// unique parallactic angles, frequencies, w values and Mueller
277    /// elements, and build the routing of `polarization`'s correlations
278    /// over `geometry`'s grid. Pixels are read on demand, at most
279    /// `bound_bytes` of cells resident.
280    ///
281    /// # Errors
282    ///
283    /// The directory cannot be listed, a cell is unreadable or malformed,
284    /// a cell lacks its partner, a (PA, frequency, w) group lacks a Mueller
285    /// element, or the catalog has no element for a selected correlation.
286    pub fn open_casa(
287        root: impl AsRef<Path>,
288        indexing: AwIndexing,
289        geometry: &GridGeometry,
290        polarization: &PolarizationRouting,
291        bound_bytes: usize,
292    ) -> Result<Self, AwCatalogError> {
293        let root = root.as_ref().to_path_buf();
294        let entries = std::fs::read_dir(&root)
295            .map_err(|error| cache_error(&root, format!("cannot list: {error}")))?;
296        let mut imaging = HashMap::new();
297        let mut weight = HashMap::new();
298        for entry in entries {
299            let entry =
300                entry.map_err(|error| cache_error(&root, format!("cannot list: {error}")))?;
301            let name = entry.file_name();
302            let Some(name) = name.to_str() else {
303                continue;
304            };
305            if let Some(rest) = name.strip_prefix(WEIGHT_PREFIX) {
306                weight.insert(rest.to_string(), read_header(&entry.path())?);
307            } else if let Some(rest) = name.strip_prefix(IMAGING_PREFIX) {
308                imaging.insert(rest.to_string(), read_header(&entry.path())?);
309            }
310        }
311        if imaging.is_empty() {
312            return Err(cache_error(&root, "no CFS_ cell images"));
313        }
314        // A native cell's working field is the image field times the
315        // oversampling: `working_size · working_increment = sampling · n · Δx`.
316        let [grid_n, _] = geometry.grid_shape();
317        let image_increment = geometry.image().increment_rad[0].abs();
318        for header in imaging.values() {
319            let Some(working_field) = header.working_field_rad else {
320                continue;
321            };
322            let expected = image_increment * f64::from(header.sampling) * grid_n as f64;
323            if (working_field - expected).abs() > 1.0e-6 * expected {
324                return Err(cache_error(
325                    &header.path,
326                    format!(
327                        "generated for a working field of {working_field} rad, the image needs \
328                         {expected} rad: the cache belongs to another image geometry"
329                    ),
330                ));
331            }
332        }
333        let mut cells = Vec::with_capacity(imaging.len());
334        for (rest, header) in imaging {
335            let partner = weight
336                .remove(&rest)
337                .ok_or_else(|| cache_error(&header.path, "paired WTCFS_ cell is missing"))?;
338            if partner.mueller != header.mueller
339                || partner.w_value.to_bits() != header.w_value.to_bits()
340                || partner.pa_deg.to_bits() != header.pa_deg.to_bits()
341                || partner.sampling != header.sampling
342            {
343                return Err(cache_error(
344                    &partner.path,
345                    "WTCFS_ cell disagrees with its CFS_ cell",
346                ));
347            }
348            cells.push(CellFiles {
349                imaging: header,
350                weight: partner,
351            });
352        }
353        if let Some((_, orphan)) = weight.into_iter().next() {
354            return Err(cache_error(&orphan.path, "paired CFS_ cell is missing"));
355        }
356        let unique = |values: Vec<f64>| {
357            let mut values = values;
358            values.sort_by(f64::total_cmp);
359            values.dedup_by(|a, b| a.to_bits() == b.to_bits());
360            values
361        };
362        let pa_deg = unique(cells.iter().map(|cell| cell.imaging.pa_deg).collect());
363        let frequencies_hz = unique(cells.iter().map(|cell| cell.imaging.frequency_hz).collect());
364        let w_values = unique(cells.iter().map(|cell| cell.imaging.w_value).collect());
365        let mut mueller_elements = cells
366            .iter()
367            .map(|cell| cell.imaging.mueller)
368            .collect::<Vec<_>>();
369        mueller_elements.sort_unstable();
370        mueller_elements.dedup();
371        let w_increment = cells[0].imaging.w_increment;
372        let oversampling = cells[0].imaging.sampling;
373        if cells.iter().any(|cell| {
374            cell.imaging.w_increment.to_bits() != w_increment.to_bits()
375                || cell.imaging.sampling != oversampling
376        }) {
377            return Err(cache_error(&root, "cells disagree on WIncr or Sampling"));
378        }
379        let index_of = |values: &[f64], value: f64| {
380            values
381                .iter()
382                .position(|listed| listed.to_bits() == value.to_bits())
383                .expect("listed value")
384        };
385        let groups_len = pa_deg.len() * frequencies_hz.len() * w_values.len();
386        let mut groups: Vec<Vec<Option<CellFiles>>> = (0..groups_len)
387            .map(|_| vec![None; mueller_elements.len()])
388            .collect();
389        let mut max_half_support = [0_u16; 2];
390        for cell in cells {
391            let group = (index_of(&pa_deg, cell.imaging.pa_deg) * frequencies_hz.len()
392                + index_of(&frequencies_hz, cell.imaging.frequency_hz))
393                * w_values.len()
394                + index_of(&w_values, cell.imaging.w_value);
395            let plane = mueller_elements
396                .binary_search(&cell.imaging.mueller)
397                .expect("listed element");
398            if groups[group][plane].is_some() {
399                return Err(cache_error(&cell.imaging.path, "duplicate cell key"));
400            }
401            for (axis, largest) in max_half_support.iter_mut().enumerate() {
402                *largest = (*largest)
403                    .max(cell.imaging.support[axis])
404                    .max(cell.weight.support[axis]);
405            }
406            groups[group][plane] = Some(cell);
407        }
408        let groups = groups
409            .into_iter()
410            .map(|planes| {
411                let cells = planes
412                    .into_iter()
413                    .collect::<Option<Vec<_>>>()
414                    .ok_or_else(|| {
415                        cache_error(&root, "a (PA, frequency, w) group lacks a Mueller element")
416                    })?;
417                Ok(Group::new(cells))
418            })
419            .collect::<Result<Vec<_>, AwCatalogError>>()?;
420        let mueller = routing(&mueller_elements, polarization).ok_or_else(|| {
421            cache_error(
422                &root,
423                "the catalog lacks a Mueller element of a selected correlation",
424            )
425        })?;
426        let [nx, ny] = geometry.grid_shape();
427        // `AWProjectFT::getImage` and `getWeightImage` divide every image
428        // by the sampling sinc `sin(x)/x`, `x = π(i − n/2)/(n·sampling)`;
429        // the spheroidal lives in the cells under `psterm`, and
430        // `initializeToVis` leaves the model uncorrected (its `sincConv`
431        // is reset to one).
432        let sampling_correction = |len: usize| {
433            (0..len)
434                .map(|index| {
435                    if index == len / 2 {
436                        return 1.0;
437                    }
438                    let x = std::f64::consts::PI * (index as f64 - (len / 2) as f64)
439                        / (len as f64 * f64::from(oversampling));
440                    x / x.sin()
441                })
442                .collect::<Vec<_>>()
443        };
444        Ok(Self {
445            root,
446            indexing,
447            pa_deg,
448            frequencies_hz,
449            w_values,
450            w_increment,
451            mueller_elements,
452            groups,
453            oversampling,
454            max_half_support,
455            mueller,
456            correction: ImageCorrection::split(
457                sampling_correction(nx),
458                sampling_correction(ny),
459                vec![1.0; nx],
460                vec![1.0; ny],
461            ),
462            lru: Mutex::new(Lru {
463                loaded: HashMap::new(),
464                clock: 0,
465                bytes: 0,
466                bound: bound_bytes,
467            }),
468        })
469    }

471    /// Generate every cell of `input` natively into `root` as CASA-format
472    /// images (`CFS_<pa>_0_CF_<f>_<w>_<m>.im` and the `WTCFS_` partner, one
473    /// baseline type) so [`AwCatalog::open_casa`] serves them. Existing
474    /// cells of the same names are replaced.
475    ///
476    /// # Errors
477    ///
478    /// Generation fails or an image cannot be written.
479    pub fn generate_native(
480        root: impl AsRef<Path>,
481        input: &NativeAwRequestInput,
482        only_missing: bool,
483    ) -> Result<(), AwCatalogError> {
484        let root = root.as_ref();
485        std::fs::create_dir_all(root)
486            .map_err(|error| cache_error(root, format!("cannot create: {error}")))?;
487        let model = EvlaApertureModel::new(input.surface.clone());
488        let grid = input.grid;
489        let mut workspace: Option<EvlaAwWorkspace> = None;
490        for (ipa, &pa) in input.pa_values.iter().enumerate() {
491            for (inu, group) in input.frequencies.iter().enumerate() {
492                let frequency = group.cf_frequency_hz;
493                let conjugate = conjugate_frequency(frequency, input.reference_frequency_hz);
494                // `AWConvFunc::makeConvFunction2`: the conjugate beam uses the
495                // nearest listed frequency to `√(2f₀² − f²)`, the first on a tie
496                // (`SynthesisUtils::nearestValue`).
497                let conjugate_frequency = if input.terms.conjugate_beams {
498                    nearest(
499                        &input
500                            .frequencies
501                            .iter()
502                            .map(|group| group.cf_frequency_hz)
503                            .collect::<Vec<_>>(),
504                        conjugate,
505                    )
506                    .1
507                } else {
508                    frequency
509                };
510                for (iw, &w) in input.w_values.iter().enumerate() {
511                    for (im, &mueller) in input.mueller_elements.iter().enumerate() {
512                        let suffix = format!("{ipa}_0_CF_{inu}_{iw}_{im}.im");
513                        let imaging_path = root.join(format!("{IMAGING_PREFIX}{suffix}"));
514                        let weight_path = root.join(format!("{WEIGHT_PREFIX}{suffix}"));
515                        if only_missing && imaging_path.is_dir() && weight_path.is_dir() {
516                            continue;
517                        }
518                        let request = EvlaAwCellRequest {
519                            size: grid.size,
520                            sky_increment_rad: grid.sky_increment_rad,
521                            frequency_hz: frequency,
522                            conjugate_frequency_hz: conjugate_frequency,
523                            w_wavelengths: w,
524                            parallactic_angle_rad: pa,
525                            mueller,
526                            oversampling: grid.oversampling,
527                            prolate_spheroidal: input.terms.prolate_spheroidal,
528                            aperture: input.terms.aperture,
529                        };
530                        let workspace = match &mut workspace {
531                            Some(workspace) => workspace,
532                            None => workspace.insert(EvlaAwWorkspace::new(request)?),
533                        };
534                        let pair = workspace.generate(&model, request)?;
535                        let misc = CellMisc {
536                            sampling: grid.oversampling as f64,
537                            pa_deg: pa.to_degrees(),
538                            mueller,
539                            w_value: w,
540                            w_increment: input.w_increment,
541                            conjugate_frequency,
542                            conjugate_mueller: MUELLER_ELEMENTS - 1 - mueller as u32,
543                            diameter_m: input.antenna_diameter_m,
544                            sky_increment_rad: grid.sky_increment_rad[1].abs(),
545                            working_size: grid.size,
546                        };
547                        write_cell(&imaging_path, &pair.imaging, frequency, grid, &misc)?;
548                        write_cell(&weight_path, &pair.weight, frequency, grid, &misc)?;
549                    }
550                }
551            }
552        }
553        Ok(())
554    }

556    /// How many of the cells `input` names exist under `root` (both the
557    /// `CFS_` and `WTCFS_` images), and how many it names.
558    pub fn native_cells_present(
559        root: impl AsRef<Path>,
560        input: &NativeAwRequestInput,
561    ) -> (usize, usize) {
562        let root = root.as_ref();
563        let mut present = 0;
564        let mut expected = 0;
565        for ipa in 0..input.pa_values.len() {
566            for inu in 0..input.frequencies.len() {
567                for iw in 0..input.w_values.len() {
568                    for im in 0..input.mueller_elements.len() {
569                        let suffix = format!("{ipa}_0_CF_{inu}_{iw}_{im}.im");
570                        expected += 1;
571                        if root.join(format!("{IMAGING_PREFIX}{suffix}")).is_dir()
572                            && root.join(format!("{WEIGHT_PREFIX}{suffix}")).is_dir()
573                        {
574                            present += 1;
575                        }
576                    }
577                }
578            }
579        }
580        (present, expected)
581    }

583    /// Remove every `CFS_` and `WTCFS_` cell under `root`, so a regenerated
584    /// catalog carries no member of an earlier request.
585    pub fn clear_native(root: impl AsRef<Path>) -> Result<(), AwCatalogError> {
586        let root = root.as_ref();
587        if !root.is_dir() {
588            return Ok(());
589        }
590        let entries = std::fs::read_dir(root)
591            .map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
592        for entry in entries {
593            let entry =
594                entry.map_err(|error| cache_error(root, format!("cannot list: {error}")))?;
595            let name = entry.file_name();
596            let Some(name) = name.to_str() else {
597                continue;
598            };
599            if name.starts_with(IMAGING_PREFIX) || name.starts_with(WEIGHT_PREFIX) {
600                std::fs::remove_dir_all(entry.path()).map_err(|error| {

662    /// The parallactic-angle cell of `pa_deg`: the nearest listed angle on
663    /// the circle (`CFCache::fillCFListFromDisk` selects cells within `dPA`
664    /// and `SynthesisUtils::stdNearestValue` picks among them).
665    #[must_use]
666    pub fn pa_cell(&self, pa_deg: f64) -> usize {
667        self.pa_deg
668            .iter()
669            .enumerate()
670            .fold((0, f64::INFINITY), |best, (index, listed)| {
671                let distance = circular_degrees(pa_deg - listed).abs();
672                if distance < best.1 {
673                    (index, distance)
674                } else {
675                    best
676                }
677            })
678            .0
679    }

681    /// The frequency cell of a row at `freq_hz`: the nearest listed
682    /// frequency to `√(2f₀² − f²)` under `conjbeams`, to `f` otherwise
683    /// (`CFBuffer::initMaps`, `nearestFreqNdx`).
684    #[must_use]
685    pub fn frequency_cell(&self, freq_hz: f64) -> usize {
686        let target = if self.indexing.conjugate_beams {
687            conjugate_frequency(freq_hz, self.indexing.image_reference_hz)
688        } else {
689            freq_hz
690        };
691        nearest(&self.frequencies_hz, target).0
692    }
693
694    /// The w-plane of `w_lambda` (`CFBuffer::nearestWNdx`, CAS-13191).
695    #[must_use]
696    pub fn w_cell(&self, w_lambda: f64) -> usize {
697        let index = (self.w_increment * w_lambda.abs()).sqrt().round() as usize;
698        index.min(self.w_values.len() - 1)
699    }

701    /// The group of a (PA, frequency, w) cell triple.
702    #[must_use]
703    pub fn group_index(&self, pa: usize, frequency: usize, w: usize) -> usize {
704        (pa * self.frequencies_hz.len() + frequency) * self.w_values.len() + w
705    }

707    /// Load group `group`'s cells, or fetch them from the cache, and park
708    /// them in `hold`.
709    fn loaded(&self, group: usize) -> (Arc<DenseCell>, Arc<DenseCell>) {
710        let mut lru = self.lru.lock().expect("cache lock");
711        lru.clock += 1;
712        let now = lru.clock;
713        if let Some(loaded) = lru.loaded.get_mut(&group) {
714            loaded.last_use = now;
715            return (loaded.imaging.clone(), loaded.weight.clone());
716        }
717        let mut loaded = load_group(&self.groups[group], self.oversampling)
718            .unwrap_or_else(|error| panic!("AW cell group {group} cannot be read: {error}"));
719        loaded.last_use = now;
720        let pair = (loaded.imaging.clone(), loaded.weight.clone());
721        lru.bytes += loaded.bytes();
722        lru.loaded.insert(group, loaded);
723        while lru.bytes > lru.bound && lru.loaded.len() > 1 {
724            let oldest = lru
725                .loaded
726                .iter()
727                .min_by_key(|(_, loaded)| loaded.last_use)
728                .map(|(group, _)| *group)
729                .expect("a loaded group");
730            if let Some(cell) = lru.loaded.remove(&oldest) {
731                lru.bytes -= cell.bytes();
732            }
733        }
734        pair
735    }
736}

814/// `AWProjectFT::makeCFPolMap` and `makeConjPolMap`: each visibility
815/// correlation maps to the plane of its own Mueller element (`4·i + j`
816/// over `RR, RL, LR, LL` or `XX, XY, YX, YY`), the conjugate table to the
817/// swapped hand (`RR ↔ LL`, `RL ↔ LR`), feeding the grid polarization
818/// `pol_map` names; `None` when an element is missing.
819fn routing(elements: &[u32], polarization: &PolarizationRouting) -> Option<MuellerRouting> {
820    let pol_map = polarization.pol_map();
821    let gpols = polarization.grid_pols();
822    let plane_of = |element: u32| {
823        elements
824            .binary_search(&element)
825            .ok()
826            .map(|plane| u8::try_from(plane).expect("planes fit u8"))
827    };
828    let mut direct = vec![vec![None; pol_map.len()]; gpols];
829    let mut conjugate = vec![vec![None; pol_map.len()]; gpols];
830    for (vpol, (correlation, target)) in polarization.correlations().iter().zip(pol_map).enumerate()
831    {
832        let Some(gpol) = *target else {
833            continue;
834        };
835        let element = mueller_element(*correlation)?;
836        direct[usize::from(gpol)][vpol] = Some(plane_of(element)?);
837        conjugate[usize::from(gpol)][vpol] = Some(plane_of(MUELLER_ELEMENTS - 1 - element)?);
838    }
839    Some(MuellerRouting { direct, conjugate })
840}
841
842/// The diagonal Mueller element of a parallel or cross hand.
843fn mueller_element(correlation: CorrelationType) -> Option<u32> {
844    Some(match correlation {
845        CorrelationType::CircularRr | CorrelationType::LinearXx => 0,
846        CorrelationType::CircularRl | CorrelationType::LinearXy => 5,
847        CorrelationType::CircularLr | CorrelationType::LinearYx => 10,
848        CorrelationType::CircularLl | CorrelationType::LinearYy => 15,
849        _ => return None,
850    })
851}

853/// `SynthesisUtils::conjFreq`: `√(2f₀² − f²)`.
854fn conjugate_frequency(frequency_hz: f64, reference_hz: f64) -> f64 {
855    (2.0 * reference_hz * reference_hz - frequency_hz * frequency_hz).sqrt()
856}

858/// `SynthesisUtils::nearestValue`: the first listed value at the smallest
859/// distance.
860fn nearest(values: &[f64], target: f64) -> (usize, f64) {
861    let mut best = (0, f64::INFINITY);
862    for (index, value) in values.iter().enumerate() {
863        let distance = (value - target).abs();
864        if distance < best.1 {
865            best = (index, distance);
866        }
867    }
868    (best.0, values[best.0])
869}

871fn circular_degrees(value: f64) -> f64 {
872    (value + 180.0).rem_euclid(360.0) - 180.0
873}

875/// `CFCache::getCFParams`: the cell's coordinates and misc info.
876fn read_header(path: &Path) -> Result<CellHeader, AwCatalogError> {
877    let image = PagedImage::<Complex32>::open(path)
878        .map_err(|error| cache_error(path, format!("cannot open Complex32 image: {error}")))?;
879    let shape = image.shape();
880    if shape.len() != 4 || shape[0] == 0 || shape[1] == 0 || shape[2..] != [1, 1] {
881        return Err(cache_error(
882            path,
883            format!("expected [nx, ny, 1, 1], got {shape:?}"),
884        ));
885    }
886    if shape[0] % 2 != 0 || shape[1] % 2 != 0 {
887        return Err(cache_error(path, "cell images must have even extents"));
888    }
889    let coordinates = image.coordinates();
890    let spectral = coordinates
891        .find_coordinate(CoordinateType::Spectral)
892        .ok_or_else(|| cache_error(path, "missing spectral coordinate"))?;
893    let frequency = coordinates
894        .coordinate(spectral)
895        .reference_value()
896        .first()
897        .copied()
898        .ok_or_else(|| cache_error(path, "spectral coordinate has no reference value"))?;
899    // CASA reads the reference value into a `Float` before listing it.
900    let frequency_hz = f64::from(frequency as f32);
901    if !frequency_hz.is_finite() || frequency_hz <= 0.0 {
902        return Err(cache_error(
903            path,
904            "cell frequency must be finite and positive",
905        ));
906    }
907    let misc = image.misc_info();
908    let sampling = required_f64(&misc, "Sampling", path)?;
909    if sampling.fract() != 0.0
910        || sampling < 2.0
911        || sampling > f64::from(u16::MAX)
912        || sampling % 2.0 != 0.0
913    {
914        return Err(cache_error(
915            path,
916            "Sampling must be a positive even integer",
917        ));
918    }
919    let support = [
920        required_support(&misc, "Xsupport", path)?,
921        required_support(&misc, "Ysupport", path)?,
922    ];
923    let mueller = required_i32(&misc, "MuellerElement", path)?;
924    let mueller = u32::try_from(mueller)
925        .ok()
926        .filter(|element| *element < MUELLER_ELEMENTS)
927        .ok_or_else(|| cache_error(path, "MuellerElement must lie in 0..16"))?;
928    let w_value = required_f64(&misc, "WValue", path)?;
929    let w_increment = required_f64(&misc, "WIncr", path)?;
930    let pa_deg = required_f64(&misc, "ParallacticAngle", path)?;
931    if !w_value.is_finite()
932        || w_value < 0.0
933        || !w_increment.is_finite()
934        || w_increment < 0.0
935        || !pa_deg.is_finite()
936    {
937        return Err(cache_error(
938            path,
939            "WValue, WIncr and ParallacticAngle must be finite and non-negative",
940        ));
941    }
942    let sampling = sampling as u16;
943    for axis in 0..2 {
944        if (2 * usize::from(support[axis]) + 1) * usize::from(sampling) > shape[axis] {
945            return Err(cache_error(path, "support does not fit the cell image"));
946        }
947    }
948    // Native cells carry the working sky increment they were generated on;
949    // CASA's own caches do not.
950    let sky_increment_rad = match misc.get("CasaRsSkyIncrement") {
951        Some(Value::Scalar(ScalarValue::Float64(value))) => Some(*value),
952        _ => None,
953    };
954    if sky_increment_rad.is_some_and(|value| !value.is_finite() || value <= 0.0) {
955        return Err(cache_error(
956            path,
957            "CasaRsSkyIncrement must be finite and positive",
958        ));
959    }
960    let working_size = match misc.get("CasaRsWorkingSize") {
961        Some(Value::Scalar(ScalarValue::Int32(value))) if *value > 0 => Some(*value as usize),
962        Some(_) => {
963            return Err(cache_error(path, "CasaRsWorkingSize must be positive"));
964        }
965        None => None,
966    };
967    let working_field = sky_increment_rad
968        .zip(working_size)
969        .map(|(increment, size)| increment * size as f64);
970    Ok(CellHeader {
971        path: path.to_path_buf(),
972        shape: [shape[0], shape[1]],
973        sampling,
974        support,
975        mueller,
976        w_value,
977        w_increment,
978        pa_deg,
979        frequency_hz,
980        working_field_rad: working_field,
981    })
982}

1017/// Read the pixels of `header`'s image as a `[y][x]` plane, x fastest.
1018fn read_plane(header: &CellHeader) -> Result<Vec<Complex32>, AwCatalogError> {
1019    let image = PagedImage::<Complex32>::open(&header.path)
1020        .map_err(|error| cache_error(&header.path, format!("cannot open: {error}")))?;
1021    let pixels = image
1022        .get()
1023        .map_err(|error| cache_error(&header.path, format!("cannot read: {error}")))?;
1024    let [nx, ny] = header.shape;
1025    if pixels.shape() != [nx, ny, 1, 1] {
1026        return Err(cache_error(
1027            &header.path,
1028            "cell image changed shape since open",
1029        ));
1030    }
1031    let mut plane = Vec::with_capacity(nx * ny);
1032    for y in 0..ny {
1033        for x in 0..nx {
1034            plane.push(pixels[[x, y, 0, 0]]);
1035        }
1036    }
1037    Ok(plane)
1038}

1040/// `AWConvFunc::cfArea`: `Σ` over `[−support, support)` of the integer taps
1041/// about `shape/2`.
1042fn cf_area(plane: &[Complex32], shape: [usize; 2], support: [u16; 2], sampling: u16) -> Complex64 {
1043    let origin = [shape[0] / 2, shape[1] / 2];
1044    let mut area = Complex64::default();
1045    for iy in -i64::from(support[1])..i64::from(support[1]) {
1046        for ix in -i64::from(support[0])..i64::from(support[0]) {
1047            let x = origin[0] as i64 + ix * i64::from(sampling);
1048            let y = origin[1] as i64 + iy * i64::from(sampling);
1049            if x < 0 || y < 0 || x as usize >= shape[0] || y as usize >= shape[1] {
1050                continue;
1051            }
1052            let value = plane[y as usize * shape[0] + x as usize];
1053            area += Complex64::new(f64::from(value.re), f64::from(value.im));
1054        }
1055    }
1056    area
1057}

1059/// Load one group's Mueller planes into an imaging and a weight cell:
1060/// every plane normalised by its own area (`AWConvFunc::cfArea`), tiled at
1061/// its kind's largest support with zero beyond a smaller plane's image.
1062fn load_group(group: &Group, sampling: u16) -> Result<Loaded, AwCatalogError> {
1063    let build = |select: fn(&CellFiles) -> &CellHeader,
1064                 half: [u16; 2]|
1065     -> Result<Arc<DenseCell>, AwCatalogError> {
1066        let headers = group.cells.iter().map(select).collect::<Vec<_>>();
1067        let mut planes = Vec::with_capacity(headers.len());
1068        for header in &headers {
1069            let mut plane = read_plane(header)?;
1070            let area = cf_area(&plane, header.shape, header.support, header.sampling);
1071            if !(area.norm().is_finite() && area.norm() > 0.0) {
1072                return Err(cache_error(&header.path, "cell area is not positive"));
1073            }
1074            let scale = Complex32::new((1.0 / area).re as f32, (1.0 / area).im as f32);
1075            for value in &mut plane {
1076                *value *= scale;
1077            }
1078            planes.push((plane, header.shape, header.support));
1079        }
1080        Ok(Arc::new(dense_cell_padded(&planes, half, sampling)))
1081    };
1082    Ok(Loaded {
1083        imaging: build(|cell| &cell.imaging, group.imaging_half_support)?,
1084        weight: build(|cell| &cell.weight, group.weight_half_support)?,
1085        last_use: 0,
1086    })
1087}

1089/// The dense cell of `planes` (each `[y][x]` with origin at `shape/2` and
1090/// its own declared half support) at `half` taps each side: tap `k` at
1091/// fine offset `off` reads pixel `origin + k·sampling + off`
1092/// (`accumulateToGrid.inc`: `convOrigin + (Int)(sampling·ix + off)`), zero
1093/// outside the plane and beyond the plane's own support (`AWVisResampler`
1094/// takes the support per Mueller cell; the crop buffer around it is not
1095/// part of the kernel).
1096fn dense_cell_padded(
1097    planes: &[(Vec<Complex32>, [usize; 2], [u16; 2])],
1098    half: [u16; 2],
1099    sampling: u16,
1100) -> DenseCell {
1101    let support = [2 * half[0] + 1, 2 * half[1] + 1];
1102    let [sx, sy] = [usize::from(support[0]), usize::from(support[1])];
1103    let rows = usize::from(sampling) + 1;
1104    let fine_half = i64::from(sampling / 2);
1105    let mut data = Vec::with_capacity(rows * rows * planes.len() * sx * sy);
1106    for oy in 0..rows {
1107        let off_y = oy as i64 - fine_half;
1108        for ox in 0..rows {
1109            let off_x = ox as i64 - fine_half;
1110            for (plane, shape, plane_half) in planes {
1111                let origin = [(shape[0] / 2) as i64, (shape[1] / 2) as i64];
1112                for iy in 0..sy {
1113                    let ky = iy as i64 - i64::from(half[1]);
1114                    let y = origin[1] + ky * i64::from(sampling) + off_y;
1115                    for ix in 0..sx {
1116                        let kx = ix as i64 - i64::from(half[0]);
1117                        let x = origin[0] + kx * i64::from(sampling) + off_x;
1118                        let inside = kx.abs() <= i64::from(plane_half[0])
1119                            && ky.abs() <= i64::from(plane_half[1])
1120                            && x >= 0
1121                            && y >= 0
1122                            && (x as usize) < shape[0]
1123                            && (y as usize) < shape[1];
1124                        data.push(if inside {
1125                            plane[y as usize * shape[0] + x as usize]
1126                        } else {
1127                            Complex32::default()
1128                        });
1129                    }
1130                }
1131            }
1132        }
1133    }
1134    DenseCell {
1135        data: data.into_boxed_slice(),
1136        support,
1137        oversampling: sampling,
1138        mueller_planes: u8::try_from(planes.len()).expect("Mueller planes fit u8"),
1139    }
1140}

1145    pa_deg: f64,
1146    mueller: usize,
1147    w_value: f64,
1148    w_increment: f64,
1149    conjugate_frequency: f64,
1150    conjugate_mueller: u32,
1151    diameter_m: f64,
1152    /// The working sky increment and size the cell was generated on:
1153    /// private keys (`CasaRsSkyIncrement`, `CasaRsWorkingSize`) the loader
1154    /// checks against the image geometry so a cache is never reused for
1155    /// another one.
1156    sky_increment_rad: f64,
1157    working_size: usize,
1158}

1160/// Write one generated plane as a CASA cell image (`CFCell::makePersistent`
1161/// with `SynthesisUtils::makeFTCoordSys`'s UU/VV coordinate).
1162fn write_cell(
1163    path: &Path,
1164    plane: &NativeAwPlane<'_>,
1165    frequency_hz: f64,
1166    grid: casa_imaging_model::NativeAwGrid,
1167    misc: &CellMisc,
1168) -> Result<(), AwCatalogError> {
1169    let write_error = |detail: String| AwCatalogError::Write {
1170        path: path.to_path_buf(),
1171        detail,
1172    };
1173    if path.exists() {
1174        std::fs::remove_dir_all(path)
1175            .map_err(|error| write_error(format!("cannot replace: {error}")))?;
1176    }
1177    let side = plane.size;
1178    // The uv increment of one oversampled pixel over the working field.
1179    let increment =
1180        1.0 / (grid.size as f64 * grid.sky_increment_rad[1].abs()) / grid.oversampling as f64;
1181    let mut coordinates = CoordinateSystem::new();
1182    coordinates.add_coordinate(
1183        LinearCoordinate::new(
1184            2,
1185            vec!["UU".to_string(), "VV".to_string()],
1186            vec!["lambda".to_string(), "lambda".to_string()],
1187        )
1188        .with_reference_value(vec![0.0, 0.0])
1189        .with_reference_pixel(vec![(side / 2) as f64, (side / 2) as f64])
1190        .with_increment(vec![-increment, increment]),
1191    );
1192    coordinates.add_coordinate(StokesCoordinate::new(vec![if misc.mueller == 0 {
1193        StokesType::RR
1194    } else {
1195        StokesType::LL
1196    }]));
1197    coordinates.add_coordinate(SpectralCoordinate::new(
1198        FrequencyRef::LSRK,
1199        frequency_hz,
1200        1.0,
1201        0.0,
1202        frequency_hz,
1203    ));
1204    let mut image = PagedImage::<Complex32>::create(vec![side, side, 1, 1], coordinates, path)
1205        .map_err(|error| write_error(format!("cannot create: {error}")))?;
1206    let mut pixels = ndarray::ArrayD::<Complex32>::zeros(ndarray::IxDyn(&[side, side, 1, 1]));
1207    for y in 0..side {
1208        for x in 0..side {
1209            pixels[[x, y, 0, 0]] = plane.values[y * side + x];
1210        }
1211    }
1212    image
1213        .put_slice(&pixels, &[0, 0, 0, 0])
1214        .map_err(|error| write_error(format!("cannot write pixels: {error}")))?;
1215    let name = path
1216        .file_name()
1217        .and_then(|name| name.to_str())
1218        .unwrap_or_default()
1219        .to_string();
1220    let field = |name: &str, value: ScalarValue| RecordField::new(name, Value::Scalar(value));
1221    image
1222        .set_misc_info(RecordValue::new(vec![
1223            field("Xsupport", ScalarValue::Int32(plane.support as i32)),
1224            field("Ysupport", ScalarValue::Int32(plane.support as i32)),
1225            field("Sampling", ScalarValue::Float64(misc.sampling)),
1226            field("ParallacticAngle", ScalarValue::Float64(misc.pa_deg)),
1227            field("MuellerElement", ScalarValue::Int32(misc.mueller as i32)),
1228            field("WValue", ScalarValue::Float64(misc.w_value)),
1229            field("WIncr", ScalarValue::Float64(misc.w_increment)),
1230            field("Name", ScalarValue::String(name)),
1231            field("ConjFreq", ScalarValue::Float64(misc.conjugate_frequency)),
1232            field(
1233                "ConjPoln",
1234                ScalarValue::Int32(misc.conjugate_mueller as i32),
1235            ),
1236            field("TelescopeName", ScalarValue::String("EVLA".to_string())),
1237            field("BandName", ScalarValue::String(String::new())),
1238            field("Diameter", ScalarValue::Float64(misc.diameter_m)),
1239            field("OpCode", ScalarValue::Bool(false)),
1240            field(
1241                "CasaRsSkyIncrement",
1242                ScalarValue::Float64(misc.sky_increment_rad),
1243            ),
1244            field(
1245                "CasaRsWorkingSize",
1246                ScalarValue::Int32(
1247                    i32::try_from(misc.working_size)
1248                        .map_err(|_| write_error("working size exceeds i32".to_string()))?,
1249                ),
1250            ),
1251        ]))
1252        .map_err(|error| write_error(format!("cannot write misc info: {error}")))?;
1253    image
1254        .save()
1255        .map_err(|error| write_error(format!("cannot save: {error}")))?;
1256    Ok(())
1257}

1328    fn padded_cells_read_the_oversampled_plane_and_zero_outside() {

1331        let side = 8;
1332        let values = (0..side * side)
1333            .map(|index| Complex32::new((index % side) as f32, (index / side) as f32))
