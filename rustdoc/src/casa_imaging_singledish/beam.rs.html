2//! The restoring beam `tsdimaging` gives its image: CASA's
3//! `sdbeamutil.TheoreticalBeam`, a one-dimensional model of the antenna's
4//! beam smeared by the gridding kernel and by the raster sampling.
5//!
6//! 1. The antenna's beam is a Gaussian whose full width is Baars' formula
7//!    for a 10 dB edge taper (Baars 2007, eq. 4.13) times Schroeder's
8//!    central-obstruction factor (*Astronomical Optics*, table 10.1),
9//!    `b λ/D`.
10//! 2. It is convolved with a cut through the gridding kernel and normalised
11//!    to a peak of 1, then convolved with a box as wide as the sampling
12//!    interval and normalised again.
13//! 3. The beam is the full width of the Gaussian, centred and of peak 1,
14//!    fitted to that by least squares. With two sampling intervals (or a
15//!    `BOX` kernel on non-square pixels), the second axis is fitted too and
16//!    the beam is the geometric mean of the two widths.
17//!
18//! The profiles are sampled every 0.02″, coarser for a wide beam or fine
19//! sampling, as CASA samples them, and convolved as `numpy.convolve(…,
20//! mode='same')` convolves them.
21//!
22//! One departure (`docs/CASA (C++) bugs.md`): CASA convolves with its own
23//! models of the kernels, not the kernels the gridder uses. Its `GAUSS`
24//! takes `gwidth` as a full width where the gridder takes a half width,
25//! and truncates at 1.5 `gwidth` where the gridder truncates at
26//! `truncate`; its `GJINC` ignores `truncate`; its `SF` is SciPy's prolate
27//! spheroid of `c = 5.356π/2`, which approximates the gridder's; its `PB`
28//! is an Airy pattern scaled to the beam of step 1, where the gridder
29//! takes the telescope's pattern; and its `BOX` is one pixel wide whatever
30//! `convsupport` is. casa-rs cuts through the gridder's own table
31//! ([`KernelTable`]), so the beam describes the image the gridder makes;
32//! a `BOX` of support `s` is `2s + 1` pixels wide.
33

36/// `primaryBeamArcsec`: the full width at half maximum, arcsec, of the beam
37/// of a dish of `diameter_m` with a central obscuration of `obscuration_m`
38/// and an edge taper of `taper_db`, at `frequency_hz`.
39#[must_use]
40pub fn primary_beam_fwhm_arcsec(
41    frequency_hz: f64,
42    diameter_m: f64,
43    obscuration_m: f64,
44    taper_db: f64,
45) -> f64 {
46    let wavelength_m = 2.997_924_58e8 / frequency_hz;
47    let factor =
48        baars_taper_factor(taper_db.abs()) * central_obstruction_factor(obscuration_m / diameter_m);
49    factor * wavelength_m * 3600.0 * 180.0 / (diameter_m * std::f64::consts::PI)
50}
51
52/// `baarsTaperFactor` (Baars 2007, eq. 4.13): the constant `b` in
53/// `b λ/D` for a parabolic illumination with an edge taper of `taper_db`.
54fn baars_taper_factor(taper_db: f64) -> f64 {
55    let tau = 10_f64.powf(-0.05 * taper_db);
56    1.269 - 0.566 * tau + 0.534 * tau * tau - 0.208 * tau * tau * tau
57}

59/// `centralObstructionFactor`: Schroeder's table 10.1 of an Airy pattern's
60/// width against the obscuration ratio `epsilon`, relative to the
61/// unobstructed width, interpolated as SciPy's `UnivariateSpline(s=0)`
62/// interpolates it: the not-a-knot cubic spline through the five points,
63/// whose one interior knot is at 0.2.
64fn central_obstruction_factor(epsilon: f64) -> f64 {
65    let (low, high) = schroeder_spline();
66    let [a, b, c, d] = if epsilon <= 0.2 { low } else { high };
67    let x = epsilon - 0.2;
68    (a + x * (b + x * (c + x * d))) / 1.22
69}
70
71/// Schroeder's points: obscuration ratio and relative width (× 1.22).
72const SCHROEDER: [(f64, f64); 5] = [
73    (0.0, 1.22),
74    (0.1, 1.205),
75    (0.2, 1.167),
76    (0.33, 1.098),
77    (0.4, 1.058),
78];
79
80/// The not-a-knot cubic through [`SCHROEDER`], as two cubics in powers of
81/// `ε − 0.2`, below and above the knot at 0.2, that agree there in value,
82/// slope and curvature: on each side `y = y₂ + s x + (c/2) x² + dᵢ x³`,
83/// with the shared slope `s` and curvature `c`, and each side's `dᵢ` fixed
84/// by its two other points.
85fn schroeder_spline() -> ([f64; 4], [f64; 4]) {
86    let y2 = SCHROEDER[2].1;
87    // Each other point gives `s x + (c/2) x² + d x³ = y − y₂`; the two on a
88    // side share `d`, so eliminating it leaves one equation in `s` and `c`
89    // per side.
90    let side = |i: usize, j: usize| {
91        let (xi, xj) = (SCHROEDER[i].0 - 0.2, SCHROEDER[j].0 - 0.2);
92        let (ri, rj) = (
93            (SCHROEDER[i].1 - y2) / (xi * xi * xi),
94            (SCHROEDER[j].1 - y2) / (xj * xj * xj),
95        );
96        // s (1/xi² − 1/xj²) + (c/2) (1/xi − 1/xj) = ri − rj
97        [
98            1.0 / (xi * xi) - 1.0 / (xj * xj),
99            0.5 * (1.0 / xi - 1.0 / xj),
100            ri - rj,
101        ]
102    };
103    let low = side(0, 1);
104    let high = side(3, 4);
105    let determinant = low[0] * high[1] - low[1] * high[0];
106    let slope = (low[2] * high[1] - low[1] * high[2]) / determinant;
107    let curvature = (low[0] * high[2] - low[2] * high[0]) / determinant;
108    let cube = |i: usize| {
109        let x = SCHROEDER[i].0 - 0.2;
110        (SCHROEDER[i].1 - y2 - slope * x - 0.5 * curvature * x * x) / (x * x * x)
111    };
112    (
113        [y2, slope, 0.5 * curvature, cube(0)],
114        [y2, slope, 0.5 * curvature, cube(4)],
115    )
116}

118/// What the theoretical beam is computed from.
119#[derive(Debug, Clone, Copy)]
120pub struct BeamSetup<'a> {
121    /// The antenna's diameter, m.
122    pub diameter_m: f64,
123    /// Its central obscuration's diameter, m (`tsdimaging`: 0.75 m for
124    /// ALMA's antennas, else 0).
125    pub obscuration_m: f64,
126    /// The frequency, Hz (`tsdimaging`: the rest frequency).
127    pub frequency_hz: f64,
128    /// The image's cell along each axis, arcsec, unsigned.
129    pub cell_arcsec: [f64; 2],
130    /// The kernel the image is gridded with, and its table.
131    pub kernel: (GridFunction, &'a KernelTable),
132    /// The sampling intervals, arcsec ([`crate::beam_sampling`]): one, or
133    /// along and across the scan.
134    pub sampling_arcsec: &'a [f64],
135}

137/// `TheoreticalBeam.get_beamsize_image`: the beam's full width at half
138/// maximum, arcsec, which `tsdimaging` sets as a circular restoring beam.
139#[must_use]
140pub fn theoretical_beam_arcsec(setup: &BeamSetup<'_>) -> f64 {
141    let fwhm = primary_beam_fwhm_arcsec(
142        setup.frequency_hz,
143        setup.diameter_m,
144        setup.obscuration_m,
145        10.0,
146    );
147    let step = profile_step(fwhm, setup.cell_arcsec, setup.sampling_arcsec);
148    let axis = arange(-3.0 * fwhm, 3.0 * fwhm + 0.5 * step, step);
149    let sigma = fwhm / 2.354_820_1;
150    let beam: Vec<f64> = axis
151        .iter()
152        .map(|x| (-(x * x) / (2.0 * sigma * sigma)).exp())
153        .collect();
154    let (function, table) = setup.kernel;
155    let cut = |cell: f64| kernel_cut(&axis, function, table, cell);
156    let mut gridded = convolve_same(&beam, &cut(setup.cell_arcsec[0]));
157    normalise(&mut gridded);
158    let along = box_of(&axis, setup.sampling_arcsec[0]);
159    let mut result = convolve_same(&gridded, &along);
160    normalise(&mut result);
161    let width = gaussian_fit(&axis, &result);
162    let box_kernel = matches!(function, GridFunction::Box { .. });
163    let square = setup.cell_arcsec[0] == setup.cell_arcsec[1];
164    if setup.sampling_arcsec.len() == 1 && (square || !box_kernel) {
165        return width;
166    }
167    // The second axis.
168    let across = match setup.sampling_arcsec.get(1) {
169        Some(&interval) => box_of(&axis, interval),
170        None => along,
171    };
172    if setup.sampling_arcsec.len() == 1 {
173        gridded = convolve_same(&beam, &cut(setup.cell_arcsec[1]));
174        normalise(&mut gridded);
175    }
176    let mut result = convolve_same(&gridded, &across);
177    normalise(&mut result);
178    (width * gaussian_fit(&axis, &result)).sqrt()
179}

181/// `get_antenna_beam`'s sample spacing, arcsec: 0.02″, finer for sampling
182/// below 0.1″, coarser when the smallest of the cells, the beam and the
183/// sampling exceeds 1,000 samples, or the beam 5,000.
184fn profile_step(fwhm: f64, cell: [f64; 2], sampling: &[f64]) -> f64 {
185    let mut step = 0.02;
186    let finest = sampling.iter().copied().fold(f64::INFINITY, f64::min);
187    if finest < 5.0 * step {
188        step = finest / 5.0;
189    }
190    let smallest = cell.iter().chain(sampling).copied().fold(fwhm, f64::min);
191    if smallest > 1000.0 * step {
192        step = smallest / 1000.0;
193    }
194    if fwhm > 5000.0 * step {
195        step = (fwhm / 5000.0).min(smallest / 10.0);
196    }
197    step
198}

200/// The kernel along `axis` (arcsec) on pixels of `cell` arcsec: the table at
201/// each radius truncated to a hundredth of a pixel, as the gridder reads
202/// it, and for `BOX` the `2s + 1` pixels a spectrum reaches.
203fn kernel_cut(axis: &[f64], function: GridFunction, table: &KernelTable, cell: f64) -> Vec<f64> {
204    axis.iter()
205        .map(|x| {
206            let pixels = x.abs() / cell;
207            if let GridFunction::Box { support } = function {
208                return f64::from(u8::from(pixels <= support as f64 + 0.5));
209            }
210            let index = (pixels * SAMPLING as f64) as usize;
211            f64::from(table.values().get(index).copied().unwrap_or(0.0))
212        })
213        .collect()
214}

216/// `get_box_kernel`: 1 where `|x| ≤ width/2`.
217fn box_of(axis: &[f64], width: f64) -> Vec<f64> {
218    axis.iter()
219        .map(|x| f64::from(u8::from(x.abs() <= width / 2.0)))
220        .collect()
221}

223/// `numpy.arange(start, stop, step)`.
224fn arange(start: f64, stop: f64, step: f64) -> Vec<f64> {
225    let count = ((stop - start) / step).ceil().max(0.0) as usize;
226    (0..count).map(|i| start + i as f64 * step).collect()
227}

229/// `numpy.convolve(signal, kernel, mode='same')` of two profiles of one
230/// length `n`: the full convolution's samples from `(n − 1)/2`. Only the
231/// kernel's nonzero samples are visited.
232fn convolve_same(signal: &[f64], kernel: &[f64]) -> Vec<f64> {
233    let n = signal.len();
234    let shift = (n - 1) / 2;
235    let taps: Vec<(usize, f64)> = kernel
236        .iter()
237        .copied()
238        .enumerate()
239        .filter(|&(_, value)| value != 0.0)
240        .collect();
241    (0..n)
242        .map(|i| {
243            // `full[i + shift] = Σⱼ signal[j] kernel[i + shift − j]`.
244            taps.iter()
245                .filter_map(|&(k, value)| {
246                    (i + shift)
247                        .checked_sub(k)
248                        .and_then(|j| signal.get(j))
249                        .map(|s| s * value)
250                })
251                .sum()
252        })
253        .collect()
254}

256/// Divide `profile` by its largest value.
257fn normalise(profile: &mut [f64]) {
258    let peak = profile.iter().copied().fold(f64::NEG_INFINITY, f64::max);
259    profile.iter_mut().for_each(|value| *value /= peak);
260}

262/// `gaussfit`: the full width of the Gaussian of peak 1 centred at 0 that
263/// fits `profile` on `axis` by least squares (`scipy.optimize.leastsq`
264/// from the half-power width), here by Gauss–Newton iteration to
265/// convergence.
266fn gaussian_fit(axis: &[f64], profile: &[f64]) -> f64 {
267    let mut width = half_power_width(axis, profile);
268    for _ in 0..100 {
269        let sigma = width / 2.354_820_1;
270        // r = y − g, ∂g/∂w = g x² / (σ² w).
271        let (mut gradient, mut curvature) = (0.0, 0.0);
272        for (x, y) in axis.iter().zip(profile) {
273            let g = (-(x * x) / (2.0 * sigma * sigma)).exp();
274            let slope = g * x * x / (sigma * sigma * width);
275            gradient += (y - g) * slope;
276            curvature += slope * slope;
277        }
278        if curvature == 0.0 {
279            break;
280        }
281        let change = gradient / curvature;
282        width += change;
283        if change.abs() <= 1.0e-14 * width {
284            break;
285        }
286    }
287    width
288}

290/// The width at half maximum of a profile with one peak, interpolated
291/// linearly between samples (or the axis's extent where the profile does
292/// not fall to half): the fit's starting point (`findFWHM`).
293fn half_power_width(axis: &[f64], profile: &[f64]) -> f64 {
294    let centre = profile.iter().enumerate().fold(
295        0,
296        |best, (i, value)| if *value > profile[best] { i } else { best },
297    );
298    let half = 0.5 * profile[centre];
299    let crossing = |outward: &mut dyn Iterator<Item = usize>| {
300        let mut previous = centre;
301        for i in outward {
302            if profile[i] <= half {
303                let t = (profile[previous] - half) / (profile[previous] - profile[i]);
304                return axis[previous] + t * (axis[i] - axis[previous]);
305            }
306            previous = i;
307        }
308        axis[previous]
309    };
310    let right = crossing(&mut (centre + 1..axis.len()));
311    let left = crossing(&mut (0..centre).rev());
312    right - left
313}
