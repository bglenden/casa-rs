15/// Which CASA cell arithmetic a density grid uses.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub enum DensityCellRule {
18    /// `VisImagingWeight`: single-precision coordinates truncated toward
19    /// zero, `v` not mirrored.
20    Standard,
21    /// `BriggsCubeWeightor`: rounded coordinates of the negated `u` and `v`,
22    /// built in double and looked up in single precision.
23    Cube,
24}
25
26/// A sample's density-cell coordinates in wavelengths, rounded the way CASA
27/// forms them before it picks a density cell (`VisImagingWeight::weightUniform`
28/// and its accumulation loop, `BriggsCubeWeightor::getWeightUniform`):
29/// `Float f = ν/c`, then `Float u = uvw_u·f` and `Float v = uvw_v·f`. At
30/// large image sizes one single-precision ulp is a visible fraction of a
31/// cell, so the rounding order decides which cell a boundary sample uses.
32/// Gridding and the uv taper keep the double coordinates of the
33/// [`Placement`].
34#[derive(Clone, Copy, Debug, PartialEq)]
35pub struct DensityUv {
36    /// u in wavelengths.
37    pub u: f32,
38    /// v in wavelengths.
39    pub v: f32,
40    /// CASA's `Float f = ν/c`, in wavelengths per metre.
41    pub wavelengths_per_m: f32,
42    /// The baseline's u and v in metres.
43    pub uv_m: [f64; 2],
44}
45
46impl DensityUv {
47    /// CASA's single-precision coordinates of baseline `uvw_m` at
48    /// `frequency_hz`.
49    #[must_use]
50    pub fn casa(uvw_m: [f64; 3], frequency_hz: f64) -> Self {
51        let per_m = (frequency_hz / SPEED_OF_LIGHT_M_PER_S) as f32;
52        let wavelengths_per_m = f64::from(per_m);
53        Self {
54            u: (uvw_m[0] * wavelengths_per_m) as f32,
55            v: (uvw_m[1] * wavelengths_per_m) as f32,
56            wavelengths_per_m: per_m,
57            uv_m: [uvw_m[0], uvw_m[1]],
58        }
59    }
60
61    /// The uv distance in wavelengths that CASA's radial weighting scales a
62    /// weight by (`VisImagingWeight::weightRadial`):
63    /// `f·sqrt(square(u) + square(v))`, the metre coordinates in double and
64    /// `f` the single-precision `ν/c`.
65    #[must_use]
66    pub fn radial_distance(self) -> f64 {
67        let [u, v] = self.uv_m;
68        f64::from(self.wavelengths_per_m) * (u * u + v * v).sqrt()
69    }
70}

72/// CASA `C::c`.
73pub(crate) const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

87/// Shape of a density grid: image-sized cells per plane.
88#[derive(Clone, Copy, Debug, PartialEq)]
89pub struct DensityGridShape {
90    /// Cells along u (the image width).
91    pub width: usize,
92    /// Cells along v (the image height).
93    pub height: usize,
94    /// Density planes of one generation: 1 for a continuum density; for a
95    /// per-channel generation the output channel count plus `padding` on
96    /// each side. A per-field generation holds them for each of its
97    /// [`Self::fields`] ([`Self::total_planes`]).
98    pub planes: usize,
99    /// Per-channel generation: density planes on each side of the output
100    /// axis (CASA `BriggsCubeWeightor::estimateSwingChanPad`), so native
101    /// channels just beyond the image still find their own plane. 0 for a
102    /// global generation.
103    pub padding: u32,
104    /// Half-width `b` of the box of `(2b + 1)²` cells about its own that
105    /// each sample adds to: CASA `superuniform`'s `uBox = vBox`
106    /// (`VisImagingWeight`) and a cube density's `superUniformBox`
107    /// (`BriggsCubeWeightor`), both from `npixels`. 0 adds to one cell.
108    pub box_half_width: usize,
109    /// Signed cell increments `[Δx, Δy]` in radians: the image's, or for
110    /// a square `npixels` grid CASA's `(n·Δ)/n` (`SynthesisImagerVi2::weight`).
111    pub increment_rad: [f64; 2],
112    /// Cell arithmetic.
113    pub rule: DensityCellRule,
114    /// A per-field generation (tclean's `mosweight`: the `multiField` of
115    /// `VisImagingWeight` on [`DensityCellRule::Standard`] cells and of
116    /// `BriggsCubeWeightor` on [`DensityCellRule::Cube`] cells): field key
117    /// `first + f` has the [`Self::planes`] after those of the `f` keys
118    /// before it, gridded from its samples alone, each plane with its own
119    /// Briggs factors, and a sample reads its own field's planes. `None`
120    /// when one generation covers every field.
121    pub fields: Option<DensityFields>,
122}
123
124impl DensityGridShape {
125    /// Density planes over every field: [`Self::planes`] for each of
126    /// [`Self::fields`], or [`Self::planes`] without them.
127    #[must_use]
128    pub const fn total_planes(&self) -> usize {
129        match self.fields {
130            Some(fields) => self.planes * fields.count.get() as usize,
131            None => self.planes,
132        }
133    }
134
135    /// Cells over every plane.
136    #[must_use]
137    pub const fn cells(&self) -> usize {
138        self.width * self.height * self.total_planes()
139    }
140
141    /// Bytes a [`DensityGrid`] of this shape holds: an `f64` per cell and a
142    /// weight sum per plane, over every field's planes. A weighting
143    /// generation keeps its grid for the whole run, so a per-field cube
144    /// density (`mosweight`) holds every field's `f64` cube at once,
145    /// `8·(nx·ny + 1)·F·(C + 2·padding)` bytes for `F` fields spanned and
146    /// `C` channels, where CASA holds one field's `Float` cube at a time
147    /// (`BriggsCubeWeightor::fillImgWeightCol`): 20 fields of 480 channels
148    /// at 800² hold about 50 GB, which admission refuses on most hosts
149    /// (#1085).
150    #[must_use]
151    pub const fn bytes(&self) -> u64 {
152        ((self.cells() + self.total_planes()) * size_of::<f64>()) as u64
153    }
154
155    /// The cell increment CASA gives `VisImagingWeight` for a uniform or
156    /// Briggs density of `cells` cells along an image axis of increment
157    /// `increment_rad`: `SynthesisImagerVi2::weight` forms the field of view
158    /// `cells·Δ` and divides it by `cells` again, in double, whether `cells`
159    /// is `npixels` or the image's own size.
160    #[must_use]
161    pub fn visimagingweight_increment(cells: usize, increment_rad: f64) -> f64 {
162        (cells as f64 * increment_rad) / cells as f64
163    }
164
165    /// The density plane a sample on plane `plane` of one generation, from
166    /// field key `field`, grids into and reads: under a per-field generation
167    /// `plane` of its field's planes (`None` outside the span or past the
168    /// planes), a continuum density's one plane whatever `plane` is; else
169    /// `plane`, which `plane_of` resolves.
170    #[must_use]
171    pub fn density_plane(&self, plane: u32, field: u32) -> Option<u32> {
172        let Some(fields) = self.fields else {
173            return Some(plane);
174        };
175        let index = field
176            .checked_sub(fields.first)
177            .filter(|index| *index < fields.count.get())?;
178        let plane = if self.planes == 1 { 0 } else { plane };
179        let planes = u32::try_from(self.planes).ok()?;
180        if plane >= planes {
181            return None;
182        }
183        index.checked_mul(planes)?.checked_add(plane)
184    }
185
186    /// Density plane of a placement: plane 0 for a global generation.
187    fn plane_of(&self, placement_plane: u32) -> Option<usize> {
188        let planes = self.total_planes();
189        if planes == 1 {
190            Some(0)
191        } else {
192            let plane = placement_plane as usize;
193            (plane < planes).then_some(plane)
194        }
195    }
196
197    /// CASA's integer cell origin `[nx/2, ny/2]` (`uorigin_p`, `vorigin_p`;
198    /// `GridFT`'s `offset_p` likewise): integer division, so an odd size
199    /// puts the origin on the lower of the two middle cells.
200    fn origin(&self) -> [usize; 2] {
201        [self.width / 2, self.height / 2]
202    }
203
204    /// CASA's `Float` cell scales `uscale_p = nx·Δx`, `vscale_p = ny·Δy`,
205    /// each rounded once from the double product
206    /// (`VisImagingWeight::VisImagingWeight`, `BriggsCubeWeightor::init`).
207    fn scale(&self) -> [f32; 2] {
208        [
209            (self.width as f64 * self.increment_rad[0]) as f32,
210            (self.height as f64 * self.increment_rad[1]) as f32,
211        ]
212    }
213
214    /// The first cell of the box a density placement adds to, the
215    /// `(2b + 1)²` cells about its own cell, when the whole box is accepted
216    /// (`VisImagingWeight`: `(ucell − uBox) > 0 && (ucell + uBox) < nx`, and
217    /// likewise in v; `GridFT`'s `onmygrid` likewise keeps a sample only
218    /// when its whole support lies on the grid). Standard placements carry
219    /// [`DensityUv`] coordinates; cube placements carry the double
220    /// coordinates CASA grids its weight density with (`GridFT::put`'s
221    /// `locuvw`: `nint(scale·u + offset + 1)` in Fortran indexing, of the
222    /// `u` and `v` that `FTMachine::negateUV` negated).
223    fn build_box(&self, plane: usize, u: f64, v: f64) -> Option<usize> {
224        let [x, y] = match self.rule {
225            DensityCellRule::Standard => self.standard_xy(u as f32, v as f32),
226            DensityCellRule::Cube => {
227                let [origin_x, origin_y] = self.origin().map(|origin| origin as f64);
228                let x = (-u * self.width as f64 * self.increment_rad[0] + origin_x + 1.0).round()
229                    as isize
230                    - 1;
231                let y = (-v * self.height as f64 * self.increment_rad[1] + origin_y + 1.0).round()
232                    as isize
233                    - 1;
234                [x, y]
235            }
236        };
237        let half = isize::try_from(self.box_half_width).unwrap_or(isize::MAX);
238        self.cell_index(plane, x.saturating_add(half), y.saturating_add(half))?;
239        self.cell_index(plane, x.saturating_sub(half), y.saturating_sub(half))
240    }
241
242    /// The share of a sample's weight each cell of its box takes.
243    /// `VisImagingWeight` adds the whole weight to every cell of the box;
244    /// `BriggsCubeWeightor` grids through `GridFT` with a convolution
245    /// function of ones over the box, which `sectggridd` divides by its sum
246    /// `(2b + 1)²`.
247    fn box_share(&self) -> f64 {
248        match self.rule {
249            DensityCellRule::Standard => 1.0,
250            DensityCellRule::Cube => {
251                let side = (2 * self.box_half_width + 1) as f64;
252                1.0 / (side * side)
253            }
254        }
255    }
256
257    /// The cell a lookup reads: `BriggsCubeWeightor::getWeightUniform`
258    /// rounds `uscale·u + uorigin` in `Float` with `u = −uvw_u·f` and
259    /// `v = −uvw_v·f`, as its grid was built. Evaluated in a mirrored
260    /// orientation, an interior cell differs only at an exact `Float` tie
261    /// (`k + ½`), where rounding half away from zero is not mirror-symmetric;
262    /// and on an odd width the accepted cells `0 < x < width` are not
263    /// mirror-symmetric about the origin `width / 2`, so the outermost
264    /// column would be read where CASA reads none, or the reverse.
265    fn lookup_cell(&self, plane: usize, uv: DensityUv) -> Option<usize> {
266        match self.rule {
267            DensityCellRule::Standard => self.standard_cell(plane, uv.u, uv.v),
268            DensityCellRule::Cube => {
269                let [scale_x, scale_y] = self.scale();
270                let [origin_x, origin_y] = self.origin().map(|origin| origin as f32);
271                let x = (scale_x * -uv.u + origin_x).round();
272                let y = (scale_y * -uv.v + origin_y).round();
273                self.cell_index(plane, x as isize, y as isize)
274            }
275        }
276    }
277
278    /// `VisImagingWeight`'s cell.
279    fn standard_cell(&self, plane: usize, u: f32, v: f32) -> Option<usize> {
280        let [x, y] = self.standard_xy(u, v);
281        self.cell_index(plane, x, y)
282    }
283
284    /// `VisImagingWeight`: `Int(uscale·u + uorigin)` in `Float`, truncated
285    /// toward zero.
286    fn standard_xy(&self, u: f32, v: f32) -> [isize; 2] {
287        let [scale_x, scale_y] = self.scale();
288        let [origin_x, origin_y] = self.origin().map(|origin| origin as f32);
289        [
290            (scale_x * u + origin_x) as isize,
291            (scale_y * v + origin_y) as isize,
292        ]
293    }
294
295    /// CASA accepts `0 < cell < extent` only.
296    fn cell_index(&self, plane: usize, x: isize, y: isize) -> Option<usize> {
297        if x <= 0 || y <= 0 || x >= self.width as isize || y >= self.height as isize {
298            return None;
299        }
300        Some((plane * self.height + y as usize) * self.width + x as usize)
301    }
302}

304/// Gridded weight density per plane, the input to uniform and Briggs
305/// weights.
306#[derive(Clone, Debug, PartialEq)]
307pub struct DensityGrid {
308    shape: DensityGridShape,
309    cells: Vec<f64>,
310    sum_weights: Vec<f64>,
311}
312
313impl DensityGrid {
314    /// An empty grid of `shape`, ready to [`DensityGrid::accumulate`] blocks.
315    #[must_use]
316    pub fn new(shape: DensityGridShape) -> Self {
317        Self {
318            cells: vec![0.0; shape.cells()],
319            sum_weights: vec![0.0; shape.total_planes()],
320            shape,
321        }
322    }
323
324    /// Add one block of density samples.
325    ///
326    /// Blocks carry one polarization whose weight is CASA's unpolarized input
327    /// weight; each sample adds its weight's box share
328    /// ([`DensityGridShape::box_half_width`]) to every cell of the box about
329    /// its cell and of the box about the conjugate cell, each when that whole
330    /// box lies inside the grid (CASA tests the two independently), and its
331    /// weight once to the plane's weight sum when its own box does.
332    pub fn accumulate(&mut self, block: &SampleBlock<'_>) {
333        assert_eq!(block.npol, 1, "density blocks carry one unpolarized weight");
334        let share = self.shape.box_share();
335        let side = 2 * self.shape.box_half_width + 1;
336        for (placement, weight) in block.placements.iter().zip(block.weights) {
337            if *weight <= 0.0 {
338                continue;
339            }
340            let Some(plane) = self.shape.plane_of(placement.plane) else {
341                continue;
342            };
343            let weight = f64::from(*weight);
344            let primary = self.shape.build_box(plane, placement.u, placement.v);
345            let conjugate = self.shape.build_box(plane, -placement.u, -placement.v);
346            for first in primary.into_iter().chain(conjugate) {
347                // CASA's loops: v outer, u inner.
348                for row in 0..side {
349                    let start = first + row * self.shape.width;
350                    for cell in &mut self.cells[start..start + side] {
351                        *cell += weight * share;
352                    }
353                }
354            }
355            if primary.is_some() {
356                self.sum_weights[plane] += weight;
357            }
358        }
359    }
360
361    /// The shape.
362    #[must_use]
363    pub const fn shape(&self) -> &DensityGridShape {
364        &self.shape
365    }
366
367    /// Every cell, plane-major then `[y][x]`.
368    #[must_use]
369    pub fn cells(&self) -> &[f64] {
370        &self.cells
371    }
372
373    /// Cells of one plane.
374    #[must_use]
375    pub fn plane(&self, plane: usize) -> &[f64] {
376        let cells = self.shape.width * self.shape.height;
377        &self.cells[plane * cells..(plane + 1) * cells]
378    }
379
380    /// Input weight summed once per sample whose own box lies inside the
381    /// grid, per plane.
382    #[must_use]
383    pub fn sum_weights(&self) -> &[f64] {
384        &self.sum_weights
385    }
386
387    /// Density at the lookup cell of `uv` on `plane`; `None` outside the
388    /// grid.
389    #[must_use]
390    pub fn lookup(&self, plane: usize, uv: DensityUv) -> Option<f64> {
391        self.shape
392            .lookup_cell(plane, uv)
393            .map(|cell| self.cells[cell])
394    }
395
396    /// Briggs factors `f² = (5·10^{−robust})² / (Σd² / Σd)` per plane, with
397    /// `Σd` the density sum (standard) or twice the input weight sum (cube),
398    /// and `d² = 1` (`rmode = "norm"`).
399    #[must_use]
400    pub fn robust_factors(&self, robust: f64) -> RobustFactors {
401        let cells = self.shape.width * self.shape.height;
402        let f2 = (0..self.shape.total_planes())
403            .map(|plane| {
404                let density = &self.cells[plane * cells..(plane + 1) * cells];
405                let square_sum = density.iter().map(|value| value * value).sum::<f64>();
406                let sum = match self.shape.rule {
407                    DensityCellRule::Standard => density.iter().sum::<f64>(),
408                    DensityCellRule::Cube => 2.0 * self.sum_weights[plane],
409                };
410                if sum > 0.0 && square_sum > 0.0 {
411                    (5.0 * 10_f64.powf(-robust)).powi(2) / (square_sum / sum)
412                } else {
413                    0.0
414                }
415            })
416            .collect();
417        RobustFactors {
418            robust,
419            f2,
420            d2: 1.0,
421        }
422    }
423
424    /// Absolute Briggs factors (CASA `briggsabs`, `rmode = "abs"`):
425    /// `f² = robust²` and `d² = 2·noise²` on every plane, each held in
426    /// `Float` (`VisImagingWeight`'s `f2_p` and `d2_p`; `BriggsCubeWeightor`
427    /// likewise).
428    #[must_use]
429    pub fn absolute_factors(&self, robust: f64, noise_jy: f64) -> RobustFactors {
430        let f2 = f64::from((robust * robust) as f32);
431        RobustFactors {
432            robust,
433            f2: vec![f2; self.shape.total_planes()],
434            d2: f64::from((2.0 * (noise_jy * noise_jy)) as f32),
435        }
436    }
437}

439/// CASA's Briggs factors for one robustness: `f²` per density plane and the
440/// offset `d²`, so that a sample whose cell has density `D` weighs
441/// `w / (D·f² + d²)`.
442#[derive(Clone, Debug, PartialEq)]
443pub struct RobustFactors {
444    robust: f64,
445    f2: Vec<f64>,
446    d2: f64,
447}
448
449impl RobustFactors {
450    /// The robustness parameter.
451    #[must_use]
452    pub const fn robust(&self) -> f64 {
453        self.robust
454    }
455
456    /// `f²` of `plane`.
457    #[must_use]
458    pub fn factor(&self, plane: usize) -> f64 {
459        self.f2[plane]
460    }
461
462    /// `d²`: 1 for normalized Briggs, `2·noise²` for absolute Briggs.
463    #[must_use]
464    pub const fn offset(&self) -> f64 {
465        self.d2
466    }
467}

469/// The density grid of every block from `source` (see
470/// [`DensityGrid::accumulate`]).
471pub fn build_density_grid<'a>(
472    source: impl Iterator<Item = SampleBlock<'a>>,
473    shape: DensityGridShape,
474) -> DensityGrid {
475    let mut grid = DensityGrid::new(shape);
476    for block in source {
477        grid.accumulate(&block);
478    }
479    grid

482/// CASA `briggsbwtaper` (CAS-13021): the fractional bandwidth
483/// `2(ν_last − ν_first)/(ν_last + ν_first)` of the image's spectral axis,
484/// which turns each sample's uv distance in cells into a divisor of the
485/// Briggs density term so every cube channel is weighted as if its uv
486/// coverage had the continuum's radial smearing.
487#[derive(Clone, Copy, Debug, PartialEq)]
488pub struct BandwidthTaper {
489    fractional_bandwidth: f64,
490}
491
492impl BandwidthTaper {
493    /// From the first and last channel centres of the image's spectral axis
494    /// (`SynthesisImagerVi2::weight`); the two must be finite, positive and
495    /// distinct.
496    pub fn from_frequency_range(first_hz: f64, last_hz: f64) -> Result<Self, OperatorError> {
497        let low = first_hz.min(last_hz);
498        let high = first_hz.max(last_hz);
499        if !(low.is_finite() && high.is_finite()) || low <= 0.0 || high <= low {
500            return Err(OperatorError::Weighting {
501                reason: "the bandwidth taper needs two distinct positive frequencies",
502            });
503        }
504        Ok(Self {
505            fractional_bandwidth: 2.0 * (high - low) / (high + low),
506        })
507    }
508
509    /// `2(ν_last − ν_first)/(ν_last + ν_first)`.
510    #[must_use]
511    pub const fn fractional_bandwidth(self) -> f64 {
512        self.fractional_bandwidth
513    }
514
515    /// CASA `uvDistanceFactor`: `n + 0.5`, or `(4 − n)/(4 − 2n)` when that is
516    /// below 1.5, with `n` the fractional bandwidth times the sample's
517    /// distance from the uv origin in grid cells.
518    fn uv_distance_factor(self, shape: &DensityGridShape, u: f64, v: f64) -> f64 {
519        let u_cells = u * shape.width as f64 * shape.increment_rad[0];
520        let v_cells = v * shape.height as f64 * shape.increment_rad[1];
521        let cells = self.fractional_bandwidth * u_cells.hypot(v_cells);
522        let factor = cells + 0.5;
523        if factor < 1.5 {
524            (4.0 - cells) / (4.0 - 2.0 * cells)
525        } else {
526            factor
527        }
528    }
529}

531/// The imaging-weight rule of one run.
532#[derive(Clone, Debug, PartialEq)]
533pub enum WeightingGeneration {
534    /// Input weight, optionally tapered.
535    Natural {
536        /// Gaussian uv taper.
537        taper: Option<Taper>,
538    },
539    /// Input weight times the sample's uv distance in wavelengths
540    /// ([`DensityUv::radial_distance`], CASA `radial`), optionally tapered.
541    Radial {
542        /// Gaussian uv taper.
543        taper: Option<Taper>,
544    },
545    /// Uniform (`robust` absent), Briggs, absolute Briggs or Briggs
546    /// bandwidth-taper density weighting.
547    Density {
548        /// Gridded weight density.
549        grid: DensityGrid,
550        /// Briggs factors; `None` for uniform weighting.
551        robust: Option<RobustFactors>,
552        /// CASA `briggsbwtaper`; needs `robust`.
553        bandwidth: Option<BandwidthTaper>,
554        /// Gaussian uv taper.
555        taper: Option<Taper>,
556    },
557}
558
559impl WeightingGeneration {
560    /// Uniform, Briggs or Briggs bandwidth-taper weighting over `grid`.
561    pub fn density(
562        grid: DensityGrid,
563        robust: Option<f64>,
564        bandwidth: Option<BandwidthTaper>,
565        taper: Option<Taper>,
566    ) -> Result<Self, OperatorError> {
567        let robust = match robust {
568            Some(robust) if !robust.is_finite() => {
569                return Err(OperatorError::Weighting {
570                    reason: "robustness is not finite",
571                });
572            }
573            Some(robust) => Some(grid.robust_factors(robust)),
574            None => None,
575        };
576        if bandwidth.is_some() && robust.is_none() {
577            return Err(OperatorError::Weighting {
578                reason: "the bandwidth taper is a Briggs mode and needs a robustness",
579            });
580        }
581        Ok(Self::Density {
582            grid,
583            robust,
584            bandwidth,
585            taper,
586        })
587    }
588
589    /// CASA `briggsabs` over `grid`: `w / (D·robust² + 2·noise²)`
590    /// ([`DensityGrid::absolute_factors`]). Refused when both terms vanish,
591    /// as `SynthesisImagerVi2::weight` refuses `robust = 0` with `noise = 0`.
592    pub fn absolute_density(
593        grid: DensityGrid,
594        robust: f64,
595        noise_jy: f64,
596        taper: Option<Taper>,
597    ) -> Result<Self, OperatorError> {
598        if !(robust.is_finite() && noise_jy.is_finite()) {
599            return Err(OperatorError::Weighting {
600                reason: "absolute Briggs robustness and noise must be finite",
601            });
602        }
603        let factors = grid.absolute_factors(robust, noise_jy);
604        if factors.d2 == 0.0 && factors.f2.iter().all(|f2| *f2 == 0.0) {
605            return Err(OperatorError::Weighting {
606                reason: "absolute Briggs with robust and noise both zero divides by zero",
607            });
608        }
609        Ok(Self::Density {
610            grid,
611            robust: Some(factors),
612            bandwidth: None,
613            taper,
614        })
615    }
616
617    /// The padding of a per-channel density generation (CASA
618    /// `BriggsCubeWeightor`), whose output samples take the nearest native
619    /// channel's weight; `None` when native weights are interpolated
620    /// linearly (`VisImagingWeight`).
621    pub(crate) fn cube_padding(&self) -> Option<u32> {
622        match self {
623            Self::Density { grid, .. } if grid.shape.rule == DensityCellRule::Cube => {
624                Some(grid.shape.padding)
625            }
626            Self::Natural { .. } | Self::Radial { .. } | Self::Density { .. } => None,
627        }
628    }
629
630    /// The density plane a sample on plane `plane` of one generation (its
631    /// output plane, or a per-channel generation's padded density channel)
632    /// from field key `field` reads ([`DensityGridShape::density_plane`]);
633    /// `plane` for a weighting without a density.
634    #[must_use]
635    pub fn density_plane(&self, plane: u32, field: u32) -> Option<u32> {
636        match self {
637            Self::Density { grid, .. } => grid.shape.density_plane(plane, field),
638            Self::Natural { .. } | Self::Radial { .. } => Some(plane),
639        }
640    }
641
642    /// Imaging weight of one placement from its unpolarized input weight;
643    /// `density_uv` picks the density cell on the density plane
644    /// `placement.plane` (ignored by a global generation; one of a field's
645    /// planes for a per-field one, [`Self::density_plane`]) and gives radial
646    /// weighting its uv distance. Pure; CASA's cell rules live in
647    /// [`DensityGrid::lookup`].
648    #[must_use]
649    pub fn imaging_weight(
650        &self,
651        placement: &Placement,
652        density_uv: DensityUv,
653        input_weight: f32,
654    ) -> f32 {
655        if input_weight <= 0.0 {
656            return 0.0;
657        }
658        let input = f64::from(input_weight);
659        let (weighted, taper) = match self {
660            Self::Natural { taper } => (input, *taper),
661            Self::Radial { taper } => (density_uv.radial_distance() * input, *taper),
662            Self::Density {
663                grid,
664                robust,
665                bandwidth,
666                taper,
667            } => {
668                let Some(plane) = grid.shape.plane_of(placement.plane) else {
669                    return 0.0;
670                };
671                // CASA weighs a sample whose density cell is empty or off
672                // the grid at zero, under uniform and Briggs alike
673                // (`VisImagingWeight::weightUniform` requires `a_gwt > 0`;
674                // `BriggsCubeWeightor::getWeightUniform` likewise).
675                let Some(density) = grid
676                    .lookup(plane, density_uv)
677                    .filter(|density| *density > 0.0)
678                else {
679                    return 0.0;
680                };
681                let weighted = match robust {
682                    None => input / density,
683                    Some(factors) => {
684                        let distance = bandwidth.map_or(1.0, |bandwidth| {
685                            bandwidth.uv_distance_factor(&grid.shape, placement.u, placement.v)
686                        });
687                        input / (density * factors.factor(plane) / distance + factors.offset())
688                    }
689                };
690                (weighted, *taper)
691            }
692        };
693        (weighted * gaussian_taper(taper, placement.u, placement.v)) as f32
694    }
695}
696
697/// CASA `VisImagingWeight::filter`: an elliptical Gaussian in the uv plane.
698fn gaussian_taper(taper: Option<Taper>, u: f64, v: f64) -> f64 {
699    let Some(taper) = taper else {
700        return 1.0;
701    };
702    let sine = taper.position_angle_rad().sin();
703    let cosine = taper.position_angle_rad().cos();
704    let rotated_u = sine * u + cosine * v;
705    let rotated_v = -cosine * u + sine * v;
706    let major = std::f64::consts::LN_2 / taper.major_lambda().powi(2);
707    let minor = std::f64::consts::LN_2 / taper.minor_lambda().powi(2);
708    (-major * rotated_u.powi(2) - minor * rotated_v.powi(2)).exp()
709}
