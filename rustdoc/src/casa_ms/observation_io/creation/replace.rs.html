2//! Publishing a created MeasurementSet: staged in a directory the plan
3//! reserves beside its destination ([`Stage`]), then renamed into a new
4//! path or exchanged with the MeasurementSet it replaces.
5//!
6//! The stage is a directory created atomically under a name of the plan's
7//! own (another name is tried while one is taken), holding the MeasurementSet
8//! being written, so a failure removes only what the plan made, never a
9//! directory or file that was there before. A new MeasurementSet is renamed
10//! into place without replacing anything that appeared there meanwhile.
11//!
12//! Replacing an existing MeasurementSet ([`super::Destination::Replace`],
13//! r3 §8) is how CASA's `concat` appends to an existing `concatvis`, which
14//! is then the plan's first input. The plan takes casacore's write lock on
15//! the MeasurementSet's MAIN when it is made, refusing one another process
16//! holds, and holds it through publication. The two directories are
17//! exchanged in one call (`renamex_np` with `RENAME_SWAP` on macOS,
18//! `renameat2` with `RENAME_EXCHANGE` on Linux), so an opening sees either
19//! whole; the old one, then in the stage, is deleted. A filesystem without
20//! the exchange (some network filesystems) refuses `Replace` before
21//! anything is written. A failure before the exchange removes the stage and
22//! leaves the MeasurementSet as it was; a failed exchange changes nothing; an
23//! old copy that cannot be deleted is reported
24//! ([`super::CreationReport::leftover`]). The same exchange publishes image
25//! products (`casa-imaging-products`' writer); the two could share one
26//! function if that crate's dependency policy (ADR-0016) took this one's.
27

39/// A MeasurementSet held for replacement: its path, and for a
40/// MeasurementSet v2 the write lock held on its MAIN until publication.
41///
42/// What replaces it is written beside it, on the same filesystem, and the
43/// two directories are exchanged in one atomic call ([`Replacement::publish`]),
44/// so an opening sees either whole. A plan replaces this way
45/// ([`super::Destination::Replace`]), and so may any other writer of a whole
46/// store, an MSv4 as well.
47#[derive(Debug)]
48pub struct Replacement {
49    destination: PathBuf,
50    lock: Option<TableWriteLock>,
51}
52
53impl Replacement {
54    /// Hold the MeasurementSet at `destination` for replacement. It must
55    /// be a whole MeasurementSet ([`crate::observation_io::is_measurement_set`]),
56    /// not a link to one, on a filesystem that exchanges directories (two
57    /// empty ones reserved beside it are exchanged to find out); a
58    /// MeasurementSet v2 must not be held for writing elsewhere, and its
59    /// MAIN's casacore write lock is then held until [`publish`](Self::publish).
60    ///
61    /// # Errors
62    ///
63    /// [`ObservationError::InvalidRequest`] for what is no MeasurementSet, a
64    /// link, or an MSv2 another process writes;
65    /// [`ObservationError::Unsupported`] for a filesystem without the
66    /// exchange.
67    pub fn hold(destination: &Path) -> Result<Self, ObservationError> {
68        refuse_unless_whole(destination)?;
69        probe(destination)?;
70        let lock = if crate::observation_io::is_msv4_store(destination) {
71            None
72        } else {
73            Some(TableWriteLock::acquire(destination, 1).map_err(|error| {
74                request_error(format!(
75                    "{} is held for writing elsewhere and cannot be replaced: {error}",
76                    destination.display()
77                ))
78            })?)
79        };
80        Ok(Self {
81            destination: destination.to_path_buf(),
82            lock,
83        })
84    }
85
86    /// The MeasurementSet held.
87    #[must_use]
88    pub fn path(&self) -> &Path {
89        &self.destination
90    }
91
92    /// Exchange the complete store at `replacement`, beside the one held, for
93    /// it in one atomic call, then delete the old store, which `replacement`
94    /// then names; its path when it could not be deleted. What is at the
95    /// held path is checked again just before the exchange.
96    ///
97    /// # Errors
98    ///
99    /// When the held path no longer holds a whole MeasurementSet, or the
100    /// exchange fails; either changes nothing.
101    pub fn publish(self, replacement: &Path) -> Result<Option<PathBuf>, ObservationError> {
102        refuse_unless_whole(&self.destination)?;
103        exchange_directories(replacement, &self.destination).map_err(storage)?;
104        // A lock was taken on the old MAIN, now at `replacement`.
105        if let Some(lock) = self.lock {
106            let _ = lock.release();
107        }
108        Ok(fs::remove_dir_all(replacement)
109            .is_err()
110            .then(|| replacement.to_path_buf()))
111    }
112}
113
114/// Refuse to replace what is at `destination` unless it is a whole
115/// MeasurementSet, and not a link to one.
116fn refuse_unless_whole(destination: &Path) -> Result<(), ObservationError> {
117    let link = fs::symlink_metadata(destination).is_ok_and(|meta| meta.file_type().is_symlink());
118    if link || !crate::observation_io::is_measurement_set(destination) {
119        return Err(request_error(format!(
120            "{} is not a MeasurementSet to replace{}",
121            destination.display(),
122            if link { ": it is a link" } else { "" }
123        )));
124    }
125    Ok(())
126}

128/// A directory the plan reserved beside its destination, holding the
129/// MeasurementSet being written (at the destination's own name).
130pub struct Stage {
131    reserved: PathBuf,
132    measurement_set: PathBuf,
133}
134
135impl Stage {
136    /// Reserve a directory beside `destination` for `purpose`, created
137    /// atomically: a name already taken, by anything, is passed over.
138    pub fn reserve(destination: &Path, purpose: &str) -> Result<Self, ObservationError> {
139        let name = destination
140            .file_name()
141            .ok_or_else(|| request_error("a MeasurementSet's path ends in a name"))?;
142        for attempt in 0..1000 {
143            let reserved = beside(destination, &format!("{purpose}-{attempt}"))?;
144            match fs::create_dir(&reserved) {
145                Ok(()) => {
146                    let measurement_set = reserved.join(name);
147                    return Ok(Self {
148                        reserved,
149                        measurement_set,
150                    });
151                }
152                Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {}
153                Err(error) => return Err(storage(error)),
154            }
155        }
156        Err(request_error(format!(
157            "no staging directory could be reserved beside {}",
158            destination.display()
159        )))
160    }
161
162    /// Where the MeasurementSet is written.
163    pub fn path(&self) -> &Path {
164        &self.measurement_set
165    }
166
167    /// Publish the staged MeasurementSet at `destination`, a new path,
168    /// without replacing anything that appeared there.
169    pub fn publish_new(&self, destination: &Path) -> Result<(), ObservationError> {
170        rename_without_replacing(&self.measurement_set, destination).map_err(|error| {
171            request_error(format!(
172                "the MeasurementSet could not be published at {}: {error}",
173                destination.display()
174            ))
175        })
176    }
177
178    /// Release the reservation once its MeasurementSet is published: the
179    /// directory, then empty (a replaced copy that could not be deleted
180    /// keeps it).
181    pub fn release(self) {
182        let _ = fs::remove_dir(&self.reserved);
183    }
184
185    /// Remove the reservation and whatever was written in it.
186    pub fn remove(self) {
187        let _ = fs::remove_dir_all(&self.reserved);
188    }
189}
190
191/// A path beside `path` for this process's `purpose`.
192fn beside(path: &Path, purpose: &str) -> Result<PathBuf, ObservationError> {
193    let name = path
194        .file_name()
195        .and_then(|name| name.to_str())
196        .ok_or_else(|| request_error("a MeasurementSet's path ends in a UTF-8 name"))?;
197    Ok(path.with_file_name(format!(".{name}.{purpose}-{}", std::process::id())))
198}

200/// Refuse a filesystem that cannot exchange two directories, by exchanging
201/// two empty ones reserved beside `destination`; only what this made is
202/// removed.
203fn probe(destination: &Path) -> Result<(), ObservationError> {
204    let first = Stage::reserve(destination, "casa-rs-probe")?;
205    let second = Stage::reserve(destination, "casa-rs-probe")?;
206    let exchanged = exchange_directories(&first.reserved, &second.reserved);
207    first.remove();
208    second.remove();
209    exchanged.map_err(|_| ObservationError::Unsupported {
210        what: "replacing a MeasurementSet on a filesystem that cannot exchange directories",
211    })
212}
