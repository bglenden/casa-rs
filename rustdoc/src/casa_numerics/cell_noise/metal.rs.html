2//! The cell noise on a Metal device: the same draws as [`super::CellNoise`],
3//! in Metal Shading Language, for a backend to compile beside its own
4//! kernels.
5//!
6//! The source follows the CPU code operation for operation: Philox4x32-10
7//! on 32-bit words, the logarithm from the word's leading zeros and an
8//! integer `m − 1`, the sine and cosine from the octant's low bits, each
9//! sign by a flip of its bit, and the same additions into the cell in the
10//! same order. Its constants are the CPU's, written as their bits, so the
11//! device and the host round the same values. A device compiles it with
12//! fast math off, so additions, multiplications, the division and the
13//! square root are IEEE-754's, and the source turns off the fusing of a
14//! product and a sum into one rounding (`#pragma METAL fp contract(off)`),
15//! which Metal does by default and the host never does. A term of σ = 0 adds
16//! nothing, as the host's returns at once, so a cell's signed zero stays.
17//! Draws then agree with the host's to the bit, for every term
18//! [`metal_draws_exactly`] admits (the backend's law checks whole cells).
19//! The one difference left is the device's: it flushes subnormal `f32`
20//! values to zero, which only a σ below [`METAL_EXACT_SIGMA`] can make.
21//!
22//! Every name carries the `cell_noise_` prefix, so the source can be
23//! concatenated with a backend's own. The pragma holds to the end of the
24//! source it is compiled in, so the noise goes after the backend's kernels,
25//! which keep their contraction.
26

29};
30use super::philox::{MULTIPLIERS, WEYL};

32/// The smallest nonzero |σ| whose draws a Metal device makes bit for bit as
33/// the host does: 2⁻⁴⁰, about 9·10⁻¹³.
34///
35/// The device flushes subnormal `f32` values to zero; the host keeps them.
36/// No intermediate of a draw is subnormal whatever σ is: the logarithm's
37/// series and the octant's polynomials are exactly zero or at least about
38/// 2⁻⁹⁸ (the logarithm's correction at u nearest 1). A draw itself,
39/// `r·σ·sin θ` or `r·σ·cos θ`, is zero or at least
40/// `r_min·s_min·|σ|`, where `r_min` ≈ 2.2·10⁻⁵ is the smallest nonzero
41/// radius (u = 1 − 2⁻³²) and `s_min` ≈ 1.5·10⁻⁹ the smallest nonzero sine
42/// (one step of 2⁻²⁹ of an octant): 3·10⁻¹⁴·|σ|. From |σ| ≥ 2⁻⁵⁵ every
43/// nonzero draw is at least 2⁻¹⁰⁰. A sum of it and a normal cell then has
44/// both addends' units in the last place at least 2⁻¹²⁴, or the cell far
45/// below the draw, so it is zero or normal. 2⁻⁴⁰ leaves a margin of 2¹⁵.
46/// Any physical noise is many orders of magnitude above it.
47pub const METAL_EXACT_SIGMA: f32 = 1.0 / (1_u64 << 40) as f32;
48
49/// Whether a term of σ `sigma` draws on a Metal device exactly as on the
50/// host: σ = 0 (no draws), or |σ| at least [`METAL_EXACT_SIGMA`]. A NaN
51/// draws NaN on both, but is not admitted.
52#[must_use]
53pub fn metal_draws_exactly(sigma: f32) -> bool {
54    sigma == 0.0 || sigma.abs() >= METAL_EXACT_SIGMA
55}

96/// The Metal source of `cell_noise_add`: one thread per (row, channel,
97/// correlation pair) of `params.rows × params.channels` cells of
98/// `params.correlations` correlations, interleaved complex `f32`, row by
99/// row, channel then correlation fastest. Buffers: 0 the cells (read and
100/// written), 1 the rows ([`MetalNoiseRow`]), 2 the terms
101/// ([`MetalNoiseTerm`]), 3 the params ([`MetalNoiseParams`]).
102#[must_use]
103pub fn metal_source() -> String {
104    let bits = |value: f32| format!("as_type<float>(0x{:08X}u)", value.to_bits());
105    let list = |values: &[f32]| {
106        values
107            .iter()
108            .map(|v| bits(*v))
109            .collect::<Vec<_>>()
110            .join(", ")
111    };
112    format!(
113        r#"
114// ---- cell noise (casa-numerics::cell_noise) ----
115#include <metal_stdlib>
116using namespace metal;
117#pragma METAL fp contract(off)
118
119struct cell_noise_row {{ uint row_low; uint row_high; uint autocorrelation; }};
120struct cell_noise_term {{ uint key0; uint key1; float sigma; }};
121struct cell_noise_params {{ uint rows; uint channels; uint correlations; uint terms; }};
122
123constant float cell_noise_word_scale = {word_scale}; // 2^-32
124constant float cell_noise_octant_scale = {octant_scale}; // 2^-29
125constant uint cell_noise_octant_words = {octant_words}u;
126constant uint cell_noise_sqrt_half_word = {sqrt_half}u;
127constant float cell_noise_ln2_high = {ln2_high};
128constant float cell_noise_ln2_low = {ln2_low};
129constant float cell_noise_atanh[4] = {{ {atanh} }};
130constant float cell_noise_sine[5] = {{ {sine} }};
131constant float cell_noise_cosine[5] = {{ {cosine} }};
132
133static inline uint4 cell_noise_philox(uint4 c, uint2 k) {{
134    for (uint round = 0; round < 10; ++round) {{
135        if (round > 0) {{
136            k = uint2(k.x + {weyl0}u, k.y + {weyl1}u);
137        }}
138        uint high0 = mulhi({mult0}u, c.x);
139        uint low0 = {mult0}u * c.x;
140        uint high1 = mulhi({mult1}u, c.z);
141        uint low1 = {mult1}u * c.z;
142        c = uint4(high1 ^ c.y ^ k.x, low1, high0 ^ c.w ^ k.y, low0);
143    }}
144    return c;
145}}
146
147static inline float cell_noise_negate_if(float x, bool negate) {{
148    return as_type<float>(as_type<uint>(x) ^ (negate ? 0x80000000u : 0u));
149}}
150
151static inline float cell_noise_ln_uniform(uint word) {{
152    uint v = word + 1u;
153    uint shift = clz(v) & 31u;
154    uint normal = v << shift;
155    bool upper = (normal - 1u) >= (cell_noise_sqrt_half_word - 1u);
156    uint numerator = upper ? (0u - normal) : (normal & 0x7FFFFFFFu);
157    float scale = upper ? -cell_noise_word_scale : 2.0f * cell_noise_word_scale;
158    float f = float(int(numerator)) * scale;
159    float k = float(-int(shift) - (upper ? 0 : 1));
160    float s = f / (2.0f + f);
161    float z = s * s;
162    float r = z * (cell_noise_atanh[0] + z * (cell_noise_atanh[1]
163        + z * (cell_noise_atanh[2] + z * cell_noise_atanh[3])));
164    float half_square = 0.5f * f * f;
165    return k * cell_noise_ln2_high
166        - ((half_square - (s * (half_square + r) + k * cell_noise_ln2_low)) - f);
167}}
168
169static inline float cell_noise_radius(uint word) {{
170    return sqrt(-2.0f * cell_noise_ln_uniform(word));
171}}
172
173static inline float2 cell_noise_sin_cos(uint word) {{
174    uint octant = word >> 29;
175    uint within = word & (cell_noise_octant_words - 1u);
176    bool odd = (octant & 1u) == 1u;
177    uint steps = odd ? (cell_noise_octant_words - within) : within;
178    float t = float(int(steps)) * cell_noise_octant_scale;
179    float t2 = t * t;
180    float sine = t * (cell_noise_sine[0] + t2 * (cell_noise_sine[1] + t2 * (cell_noise_sine[2]
181        + t2 * (cell_noise_sine[3] + t2 * cell_noise_sine[4]))));
182    float cosine = 1.0f + t2 * (cell_noise_cosine[0] + t2 * (cell_noise_cosine[1]
183        + t2 * (cell_noise_cosine[2] + t2 * (cell_noise_cosine[3] + t2 * cell_noise_cosine[4]))));
184    bool swapped = ((octant ^ (octant >> 1)) & 1u) == 1u;
185    float s = swapped ? cosine : sine;
186    float c = swapped ? sine : cosine;
187    return float2(cell_noise_negate_if(s, (octant >> 2) == 1u),
188                  cell_noise_negate_if(c, ((octant + 2u) >> 2) == 1u));
189}}
190
191kernel void {kernel}(device float2 *cells [[buffer(0)]],
192                     device const cell_noise_row *rows [[buffer(1)]],
193                     device const cell_noise_term *terms [[buffer(2)]],
194                     constant cell_noise_params &p [[buffer(3)]],
195                     uint index [[thread_position_in_grid]]) {{
196    uint pairs = (p.correlations + 1u) / 2u;
197    uint per_row = p.channels * pairs;
198    if (index >= p.rows * per_row) {{
199        return;
200    }}
201    uint position = index / per_row;
202    uint channel = (index % per_row) / pairs;
203    uint pair = index % pairs;
204    cell_noise_row row = rows[position];
205    bool both = p.correlations - 2u * pair >= 2u;
206    ulong at = (ulong(position) * p.channels + channel) * p.correlations + 2u * pair;
207    uint4 counter = uint4(row.row_low, row.row_high, channel, pair);
208    for (uint term = 0; term < p.terms; ++term) {{
209        cell_noise_term t = terms[term];
210        if (t.sigma == 0.0f) {{
211            continue;
212        }}
213        uint4 words = cell_noise_philox(counter, uint2(t.key0, t.key1));
214        float scaled0 = cell_noise_radius(words.x) * t.sigma;
215        float2 sc0 = cell_noise_sin_cos(words.y);
216        float2 sample0 = cells[at];
217        sample0.x += scaled0 * sc0.y;
218        if (row.autocorrelation == 0u) {{
219            sample0.y += scaled0 * sc0.x;
220        }}
221        cells[at] = sample0;
222        if (both) {{
223            float scaled1 = cell_noise_radius(words.z) * t.sigma;
224            float2 sc1 = cell_noise_sin_cos(words.w);
225            float2 sample1 = cells[at + 1];
226            sample1.x += scaled1 * sc1.y;
227            if (row.autocorrelation == 0u) {{
228                sample1.y += scaled1 * sc1.x;
229            }}
230            cells[at + 1] = sample1;
231        }}
232    }}
233}}
234"#,
235        word_scale = bits(WORD_SCALE),
236        octant_scale = bits(1.0 / OCTANT_WORDS as f32),
237        octant_words = OCTANT_WORDS,
238        sqrt_half = SQRT_HALF_WORD,
239        ln2_high = bits(LN_2_HIGH),
240        ln2_low = bits(LN_2_LOW),
241        atanh = list(&ATANH),
242        sine = list(&SINE),
243        cosine = list(&COSINE),
244        weyl0 = WEYL[0],
245        weyl1 = WEYL[1],
246        mult0 = MULTIPLIERS[0],
247        mult1 = MULTIPLIERS[1],
248        kernel = METAL_KERNEL,
249    )
