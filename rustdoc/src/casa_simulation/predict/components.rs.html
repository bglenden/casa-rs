2//! `sm.predict(complist=…)`: a component list's visibilities, as CASA's
3//! simulator predicts them (`SkyEquation::get` with its
4//! `SimpleComponentFTMachine`), analytically and with no gridding.
5//!
6//! For each row and component (`SimpleComponentFTMachine::get`):
7//! - the row's uvw is rotated from the scan's phase centre to the
8//!   component's reference direction through `ComponentFTMachine`'s
9//!   projected [`UvwMachine`] at the integration epoch and antenna 0;
10//! - the component's visibility is its flux, sampled by its spectral model
11//!   at the channel's frequency and converted to the window's polarisation
12//!   basis, times its shape's visibility at the rotated baseline
13//!   (`SkyCompRep::visibility`, [`SkyComponent::visibility`]);
14//! - with `setvp(dovp=True)`, the flux is scaled first by the primary beam's
15//!   power at the component's separation from the field's pointing, at each
16//!   channel's frequency (`BeamSkyJones::apply` on a component, with its
17//!   full-spectral option, as `SkyEquation::applySkyJones` calls it);
18//! - the rotation's phase multiplies it: `e^{−2πi s ν/c}` with `s` the
19//!   phase shift of [`UvwMachine::convert_projected`], in CASA's sign.
20//!
21//! What does not depend on the row is prepared once: each component's flux
22//! in each window's correlations and the beam's power, per channel; each
23//! shape's transform ([`ShapeTransform`]); and, for each observed field,
24//! each component's rotation (a scan is phased on its field's direction).
25//! Native-frame transforms are rebuilt once per integration, as CASA's
26//! component machine does, using the first antenna's position.
27//! A row then takes one rotation and one phasor per component, and each
28//! channel advances the phasor by the channel spacing's rotation: a
29//! window's channels are evenly spaced (`setspwindow`).
30//!
31//! The visibilities add to what DATA holds. Unlike an image's prediction,
32//! a component's is formed for flagged rows too: CASA's component machine
33//! does not read the flags.
34//!
35//! Divergences, recorded in `docs/CASA (C++) bugs.md`:
36//! - **A negative flux keeps its sign under the beam.** CASA's full-spectral
37//!   beam scales each channel by the flux at that channel divided by the
38//!   flux's magnitude, and so turns a negative flux positive.
39//! - **The beam is taken at the data's frequencies.** CASA tabulates the
40//!   beam-scaled flux at each channel's LSRK frequency and then samples that
41//!   table at the data's (here TOPO) frequencies, about 1e-4 of the
42//!   frequency away.
43//!
44//! Native direction references retain their axes when evaluating shapes.

59/// The speed of light, m/s.
60const SPEED_OF_LIGHT_M_PER_S: f64 = 299_792_458.0;

62/// A window's channels and correlations, as the components are predicted
63/// into them.
64#[derive(Debug, Clone)]
65pub(crate) struct ComponentWindow {
66    pub(crate) stored_frequencies: Option<Arc<Admitted<Vec<f64>>>>,
67    /// The first channel's frequency, Hz.
68    pub(crate) first_hz: f64,
69    /// The channel spacing, Hz, for the generated linear axis.
70    pub(crate) step_hz: f64,
71    pub(crate) channels: usize,
72    /// casacore `Stokes::StokesTypes` codes, one per correlation.
73    pub(crate) correlations: Vec<i32>,
74}
75
76impl ComponentWindow {
77    fn frequency_hz(&self, channel: usize) -> f64 {
78        self.stored_frequencies.as_ref().map_or_else(
79            || self.first_hz + channel as f64 * self.step_hz,
80            |frequencies| frequencies.value[channel],
81        )
82    }
83}

92/// What a component list is predicted into: the session's windows, the
93/// distinct pairs of visibility phase centre and antenna beam centre,
94/// each scan's pair, and the primary beam with `setvp(dovp=True)`.
95/// FIELD determines phase; stored POINTING independently determines beam
96/// attenuation in a resumed simulation.
97#[derive(Clone)]
98pub(crate) struct ComponentTargets<'a> {
99    pub(crate) windows: &'a [ComponentWindow],
100    /// Slot, visibility phase centre, and antenna beam centre, J2000 rad.
101    pub(crate) pointings: &'a [(usize, [f64; 2], [f64; 2])],
102    /// `[scan]`: the scan's pointing.
103    pub(crate) pointing_of_scan: &'a [usize],
104    pub(crate) beam: Option<&'a BeamPattern>,
105    pub(crate) frame: &'a MeasFrame,
106    pub(crate) squint: Option<squint::Config>,
107    pub(crate) workers: usize,
108}

110impl ComponentTargets<'_> {
111    /// Bytes a prediction of `components` components into these targets
112    /// holds ([`ComponentPrediction::bytes`]), known before it is formed.
113    pub(crate) fn bytes(&self, components: usize) -> u64 {
114        let samples: usize = self
115            .windows
116            .iter()
117            .map(|window| {
118                window.channels
119                    * if self.squint.is_some() {
120                        4
121                    } else {
122                        window.correlations.len()
123                    }
124            })
125            .sum();
126        let channels: usize = self.windows.iter().map(|window| window.channels).sum();
127        let field_slots = self
128            .pointings
129            .iter()
130            .map(|(id, _, _)| id + 1)
131            .max()
132            .unwrap_or(0);
133        let per_field = components * size_of::<UvwMachine>()
134            + self.beam.filter(|_| self.squint.is_none()).map_or(0, |_| {
135                self.windows.len() * size_of::<Vec<f32>>()
136                    + components * channels * size_of::<f32>()
137            });
138        (components
139            * (size_of::<ShapeTransform>()
140                + size_of::<MDirection>()
141                + samples * size_of::<Complex64>())
142            + self.windows.len() * size_of::<PreparedWindow>()
143            + field_slots * size_of::<Option<FieldTerms>>()
144            + self.pointings.len() * per_field
145            + size_of_val(self.pointing_of_scan)) as u64
146            + self.squint.as_ref().map_or(0, |config| {
147                Squint::bytes(
148                    self.workers,
149                    self.windows
150                        .iter()
151                        .map(|window| window.channels)
152                        .max()
153                        .unwrap_or(0),
154                    config.radial(),
155                )
156            })
157    }

159    /// Bytes [`ComponentPrediction::new`] holds only while it prepares
160    /// `components` components: their J2000 directions.
161    pub(crate) fn preparation_bytes(components: usize) -> u64 {
162        (components * size_of::<[f64; 2]>()) as u64
163    }
164}

209impl ComponentPrediction {
210    /// Prepare `list` for prediction into `targets`, holding `reservation`,
211    /// its admission of [`ComponentTargets::bytes`] for the list's length.
212    ///
213    /// # Errors
214    ///
215    /// [`ComponentRefusal::NotYetImplemented`]: a time-dependent component
216    /// beam needing separate observer context, correlations not all circular or all
217    /// linear, or a flux, spectrum or shape casa-components does not
218    /// evaluate.
219    pub(crate) fn new(
220        list: &ComponentList,
221        targets: ComponentTargets<'_>,
222        reservation: Reservation,
223    ) -> Result<Self, ComponentRefusal> {
224        let refused = ComponentRefusal::NotYetImplemented;
225        let components = list.components();
226        let mut directions_rad = Vec::with_capacity(components.len());
227        let mut transforms = Vec::with_capacity(components.len());
228        for (index, component) in components.iter().enumerate() {
229            let direction = component.shape().reference_direction();
230            if (targets.beam.is_some() || targets.squint.is_some())
231                && !matches!(
232                    direction.refer(),
233                    DirectionRef::J2000
234                        | DirectionRef::B1950
235                        | DirectionRef::B1950Vla
236                        | DirectionRef::GALACTIC
237                        | DirectionRef::ICRS
238                        | DirectionRef::SUPERGAL
239                        | DirectionRef::ECLIPTIC
240                )
241            {
242                return Err(refused(format!(
243                    "component {index} in {:?} needs its beam's actual array observer context (#1297)",
244                    direction.refer()
245                )));
246            }
247            let (ra, dec) = direction
248                .convert_to(DirectionRef::J2000, targets.frame)
249                .map_err(|error| refused(error.to_string()))?
250                .as_angles();
251            directions_rad.push([ra, dec]);
252            transforms.push(
253                component
254                    .transform()
255                    .map_err(|error| refused(format!("component {index}: {error}")))?,
256            );
257        }
258        let windows = targets
259            .windows
260            .iter()
261            .map(|window| prepare_window(components, window, targets.squint.is_some()))
262            .collect::<Result<Vec<_>, _>>()?;
263        let directions: Vec<MDirection> = components
264            .iter()
265            .map(|component| component.shape().reference_direction().clone())
266            .collect();
267        let slots = targets
268            .pointings
269            .iter()
270            .map(|(id, _, _)| id + 1)
271            .max()
272            .unwrap_or(0);
273        let mut fields: Vec<Option<FieldTerms>> = (0..slots).map(|_| None).collect();
274        for &(field, direction, beam_centre) in targets.pointings {
275            let phase = MDirection::from_angles(direction[0], direction[1], DirectionRef::J2000);
276            let projections = directions
277                .iter()
278                .map(|component| {
279                    UvwMachine::with_frame(component, &phase, targets.frame)
280                        .map_err(|error| refused(error.to_string()))
281                })
282                .collect::<Result<Vec<_>, _>>()?;
283            let powers = targets
284                .beam
285                .filter(|_| targets.squint.is_none())
286                .map(|pattern| {
287                    targets
288                        .windows
289                        .iter()
290                        .map(|window| {
291                            let mut powers =
292                                Vec::with_capacity(directions_rad.len() * window.channels);
293                            for component in &directions_rad {
294                                let separation = separation_rad(*component, beam_centre);
295                                powers.extend((0..window.channels).map(|channel| {
296                                    pattern.power_at(separation, window.frequency_hz(channel))
297                                }));
298                            }
299                            powers
300                        })
301                        .collect()
302                });
303            fields[field] = Some(FieldTerms {
304                phase,
305                beam_centre,
306                projections,
307                powers,
308            });
309        }
310        Ok(Self {
311            directions,
312            frame: targets.frame.clone(),
313            squint: targets.squint.map(|config| {
314                Box::new(Squint::new(
315                    config,
316                    targets.beam,
317                    targets.workers,
318                    targets
319                        .windows
320                        .iter()
321                        .map(|window| window.channels)
322                        .max()
323                        .unwrap_or(0),
324                ))
325            }),
326            transforms,
327            windows,
328            fields,
329            of_scan: targets.pointing_of_scan.to_vec(),
330            _reservation: reservation,
331        })
332    }

334    /// Bytes the prediction holds beside the list it was read from: its
335    /// fluxes, the beam's powers, the shapes' transforms, the pointings'
336    /// rotations and the scans' pointings.
337    pub(crate) fn bytes(&self) -> u64 {
338        let windows: usize = self
339            .windows
340            .iter()
341            .map(|window| {
342                size_of::<PreparedWindow>() + window.fluxes.len() * size_of::<Complex64>()
343            })
344            .sum();
345        let fields: usize = self
346            .fields
347            .iter()
348            .map(|field| {
349                size_of::<Option<FieldTerms>>()
350                    + field.as_ref().map_or(0, |terms| {
351                        let powers = terms.powers.as_ref().map_or(0, |windows| {
352                            windows
353                                .iter()
354                                .map(|powers| {
355                                    size_of::<Vec<f32>>() + powers.len() * size_of::<f32>()
356                                })
357                                .sum()
358                        });
359                        terms.projections.len() * size_of::<UvwMachine>() + powers
360                    })
361            })
362            .sum();
363        (self.transforms.len() * size_of::<ShapeTransform>()
364            + self.directions.len() * size_of::<MDirection>()
365            + windows
366            + fields
367            + self.of_scan.len() * size_of::<usize>()) as u64
368            + self
369                .squint
370                .as_ref()
371                .map_or(0, |squint| squint.resident_bytes())
372    }

390    pub(crate) fn prepare_beam_job(
391        &self,
392        scan: usize,
393        field: usize,
394        times: impl Iterator<Item = f64>,
395    ) -> Result<Option<BeamContext>, casa_components::ComponentError> {
396        self.squint
397            .as_ref()
398            .map(|squint| {
399                let pointing = self.of_scan[scan];
400                let terms = self.fields[pointing].as_ref().expect("prepared pointing");
401                squint.prepare_job((field, pointing), times, &terms.phase, &self.frame)
402            })
403            .transpose()

406    /// Add to `out` the visibilities of `rows` (each row's raw uvw in
407    /// metres) of scan `scan` with DATA_DESCRIPTION `description`, at the
408    /// scan's pointing. A description supplies both frequencies and products.
409    /// `out` holds the rows' samples back to back,
410    /// `[row][channel][correlation]`.
411    pub(crate) fn add_rows(
412        &self,
413        (scan, description): (usize, usize),
414        rows: impl Iterator<Item = ([f64; 3], f64)> + Clone,
415        out: &mut [Complex32],
416        context: Option<BeamContext>,
417        worker: usize,
418    ) -> Result<(), casa_components::ComponentError> {
419        let Some(terms) = self
420            .of_scan
421            .get(scan)
422            .and_then(|&pointing| self.fields.get(pointing))
423            .and_then(Option::as_ref)
424        else {
425            // A prediction covers the scans observed before it, whose
426            // pointings it prepared.
427            debug_assert!(false, "scan {scan} was not observed before the prediction");
428            return Ok(());
429        };
430        let prepared = &self.windows[description];
431        let (channels, correlations) = (prepared.channels, prepared.correlations);
432        let powers = terms
433            .powers
434            .as_ref()
435            .map(|powers| powers[description].as_slice());
436        let mut workspace = self
437            .squint
438            .as_ref()
439            .map(|squint| squint.workspace(worker))
440            .transpose()?;
441        for (component, (transform, initial_projection)) in
442            self.transforms.iter().zip(&terms.projections).enumerate()
443        {
444            let direction = &self.directions[component];
445            let physical = if self.squint.is_some() {
446                let (ra, dec) = direction
447                    .convert_to(DirectionRef::J2000, &self.frame)
448                    .map_err(|error| casa_components::ComponentError::Visibility {
449                        shape: "component beam",
450                        reason: error.to_string(),
451                    })?
452                    .as_angles();
453                [ra, dec]
454            } else {
455                [0.; 2]
456            };
457            let mut beam_context = context;
458            let mut flux_angle = None;
459            let mut projection = *initial_projection;
460            let mut epoch = None;
461            for (row, (uvw_m, time_s)) in rows.clone().enumerate() {
462                let angle = match (&self.squint, &mut beam_context) {
463                    (Some(squint), Some(context)) => context.advance(
464                        time_s,
465                        &terms.phase,
466                        &self.frame,
467                        squint.config.increment,
468                        squint.config.alt_az,
469                    )?,
470                    _ => 0.,
471                };
472                if direction.refer() != DirectionRef::J2000 && epoch != Some(time_s) {
473                    let current = self
474                        .frame
475                        .clone()
476                        .with_epoch(MEpoch::from_mjd(time_s / 86400., EpochRef::UTC));
477                    projection = UvwMachine::with_frame(direction, &terms.phase, &current)
478                        .map_err(|error| casa_components::ComponentError::Visibility {
479                            shape: "component direction",
480                            reason: error.to_string(),
481                        })?;
482                    epoch = Some(time_s);
483                }
484                let samples = &mut out[row * channels * correlations..][..channels * correlations];
485                let (rotated, phase) =
486                    projection.convert_projected([-uvw_m[0], -uvw_m[1], uvw_m[2]]);
487                let rotated = [-rotated[0], -rotated[1], rotated[2]];
488                let phase_shift_m = -phase;
489                let baseline = transform.at_baseline(rotated);
490                // e^{−2πi s ν/c}, advanced a channel at a time.
491                let radians_per_hz =
492                    -std::f64::consts::TAU * phase_shift_m / SPEED_OF_LIGHT_M_PER_S;
493                let mut phasor = Complex64::from_polar(1.0, radians_per_hz * prepared.first_hz);
494                let step = Complex64::from_polar(1.0, radians_per_hz * prepared.step_hz);
495                let stored = if self.squint.is_some() {
496                    4
497                } else {
498                    correlations
499                };
500                let fluxes = &prepared.fluxes[component * channels * stored..][..channels * stored];
501                let powers = powers.map(|powers| &powers[component * channels..][..channels]);
502                if let Some(squint) = &self.squint
503                    && flux_angle != Some(angle)
504                {
505                    let workspace = workspace.as_mut().expect("squint has a workspace");
506                    for channel in 0..channels {
507                        let frequency = prepared.stored_frequencies.as_ref().map_or_else(
508                            || prepared.first_hz + channel as f64 * prepared.step_hz,
509                            |values| values.value[channel],
510                        );
511                        let circular = std::array::from_fn(|product| fluxes[channel * 4 + product]);
512                        let flux = squint.flux(
513                            circular,
514                            physical,
515                            terms.beam_centre,
516                            angle,
517                            frequency,
518                            prepared.basis,
519                        );
520                        workspace[channel * 4..(channel + 1) * 4].copy_from_slice(&flux);
521                    }
522                    flux_angle = Some(angle);
523                }
524                for channel in 0..channels {
525                    let frequency_hz = prepared.stored_frequencies.as_ref().map_or_else(
526                        || prepared.first_hz + channel as f64 * prepared.step_hz,
527                        |frequencies| frequencies.value[channel],
528                    );
529                    let channel_phasor = if prepared.stored_frequencies.is_some() {
530                        Complex64::from_polar(1.0, radians_per_hz * frequency_hz)
531                    } else {
532                        phasor
533                    };
534                    let mut scale = baseline.at_frequency(frequency_hz)?;
535                    if let Some(powers) = powers {
536                        scale *= f64::from(powers[channel]);
537                    }
538                    let value = channel_phasor * scale;
539                    let cells = &mut samples[channel * correlations..(channel + 1) * correlations];
540                    if self.squint.is_some() {
541                        let flux = &workspace.as_ref().expect("squint workspace")
542                            [channel * 4..(channel + 1) * 4];
543                        for (cell, &product) in
544                            cells.iter_mut().zip(&prepared.products[..correlations])
545                        {
546                            let sample = flux[product] * value;
547                            *cell += Complex32::new(sample.re as f32, sample.im as f32);
548                        }
549                    } else {
550                        for (cell, flux) in cells
551                            .iter_mut()
552                            .zip(&fluxes[channel * stored..(channel + 1) * stored])
553                        {
554                            let sample = flux * value;
555                            *cell += Complex32::new(sample.re as f32, sample.im as f32);
556                        }
557                    }
558                    phasor *= step;
559                }
560            }
561        }
562        Ok(())
563    }
564}

566/// `window`'s components' fluxes in its correlations
567/// ([`PreparedWindow::fluxes`]).
568fn prepare_window(
569    components: &[SkyComponent],
570    window: &ComponentWindow,
571    squinted: bool,
572) -> Result<PreparedWindow, ComponentRefusal> {
573    let refused = ComponentRefusal::NotYetImplemented;
574    let basis = |code: i32| match code {
575        5..=8 => Some((Polarisation::Circular, (code - 5) as usize)),
576        9..=12 => Some((Polarisation::Linear, (code - 9) as usize)),
577        _ => None,
578    };
579    let mapped = window
580        .correlations
581        .iter()
582        .map(|&code| basis(code))
583        .collect::<Option<Vec<_>>>()
584        .filter(|mapped| mapped.windows(2).all(|pair| pair[0].0 == pair[1].0))
585        .ok_or_else(|| {
586            refused(format!(
587                "predicting components into the correlations {:?}: casa-rs predicts them into \
588                 circular or linear correlations",
589                window.correlations
590            ))
591        })?;
592    let polarisation = mapped
593        .first()
594        .map_or(Polarisation::Circular, |(basis, _)| *basis);
595    let stored = if squinted { 4 } else { mapped.len() };
596    let mut fluxes = Vec::with_capacity(components.len() * window.channels * stored);
597    for (index, component) in components.iter().enumerate() {
598        for channel in 0..window.channels {
599            let flux = component
600                .flux_in(
601                    window.frequency_hz(channel),
602                    if squinted {
603                        Polarisation::Circular
604                    } else {
605                        polarisation
606                    },
607                )
608                .map_err(|error| refused(format!("component {index}: {error}")))?;
609            if squinted {
610                fluxes.extend(flux);
611            } else {
612                fluxes.extend(mapped.iter().map(|&(_, basis)| flux[basis]));
613            }
614        }
615    }
616    Ok(PreparedWindow {
617        stored_frequencies: window.stored_frequencies.clone(),
618        first_hz: window.first_hz,
619        step_hz: window.step_hz,
620        channels: window.channels,
621        correlations: mapped.len(),
622        basis: polarisation,
623        products: std::array::from_fn(|index| mapped.get(index).map_or(0, |&(_, product)| product)),
624        fluxes,
625    })
626}

628/// The great-circle separation of two J2000 directions, radians
629/// (`MVDirection::separation`: twice the arcsine of half the chord).
630fn separation_rad(a: [f64; 2], b: [f64; 2]) -> f64 {
631    let unit = |[ra, dec]: [f64; 2]| [dec.cos() * ra.cos(), dec.cos() * ra.sin(), dec.sin()];
632    let (a, b) = (unit(a), unit(b));
633    let chord = ((a[0] - b[0]).powi(2) + (a[1] - b[1]).powi(2) + (a[2] - b[2]).powi(2)).sqrt();
634    2.0 * (chord / 2.0).asin()
635}
