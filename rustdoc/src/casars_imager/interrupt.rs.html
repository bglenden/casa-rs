2//! SIGINT cancels the run (plan section 5.4). The first interrupt sets the
3//! process's cancellation token: the run stops at the next block boundary of
4//! a pass or before its next phase, and publishes nothing. A system call the
5//! interrupt lands in is restarted, so a read in flight completes rather than
6//! failing with `EINTR`. The handler then resets itself, so a second
7//! interrupt terminates the process.
8
9use std::sync::OnceLock;
10
11use casa_imaging_application::Cancel;
12
13static TOKEN: OnceLock<Cancel> = OnceLock::new();
14
15/// The process's cancellation token, which every run of this process
16/// observes.
17pub fn token() -> &'static Cancel {
18    TOKEN.get_or_init(Cancel::new)
19}
20
21/// Install the SIGINT handler that sets [`token`].
22///
23/// # Errors
24///
25/// The handler could not be installed.
26pub fn install() -> std::io::Result<()> {
27    // The token exists before the handler can run, so the handler only
28    // loads it and stores a flag: both async-signal-safe.
29    token();
30    // SAFETY: `action` is fully initialised before `sigaction` reads it, and
31    // `interrupted` touches only an initialised atomic.
32    let status = unsafe {
33        let mut action: libc::sigaction = std::mem::zeroed();
34        action.sa_sigaction = interrupted as extern "C" fn(libc::c_int) as libc::sighandler_t;
35        action.sa_flags = libc::SA_RESETHAND | libc::SA_RESTART;
36        libc::sigemptyset(&mut action.sa_mask);
37        libc::sigaction(libc::SIGINT, &action, std::ptr::null_mut())
38    };
39    if status == 0 {
40        Ok(())
41    } else {
42        Err(std::io::Error::last_os_error())
43    }
44}
45
46extern "C" fn interrupted(_: libc::c_int) {
47    if let Some(token) = TOKEN.get() {
48        token.cancel();
49    }
50}
