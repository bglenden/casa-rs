122impl MetalExecutionDecision {
123    /// Bind a Metal runtime decision to exact plan and Resource Authority topology facts.
124    pub fn bind(
125        plan: &ExecutionDag,
126        topology: &ResourceTopology,
127    ) -> Result<Self, MetalRuntimeError> {
128        let metal_accelerators = topology
129            .accelerators
130            .iter()
131            .filter(|accelerator| accelerator.kind == AcceleratorKind::Metal)
132            .collect::<Vec<_>>();
133        let [accelerator] = metal_accelerators.as_slice() else {
134            return Err(MetalRuntimeError::Ineligible(
135                "the initial Metal runtime requires exactly one inventoried Metal device"
136                    .to_string(),
137            ));
138        };
139        let view = topology
140            .memory_views
141            .iter()
142            .find(|view| view.id == accelerator.memory_view)
143            .ok_or_else(|| {
144                MetalRuntimeError::InvalidPlan(
145                    "Metal accelerator names a missing memory view".to_string(),
146                )
147            })?;
148        let domain = topology
149            .memory_domains
150            .iter()
151            .find(|domain| domain.id == view.domain)
152            .ok_or_else(|| {
153                MetalRuntimeError::InvalidPlan(
154                    "Metal memory view names a missing capacity domain".to_string(),
155                )
156            })?;
157        if domain.kind != MemoryCapacityKind::Unified {
158            return Err(MetalRuntimeError::Ineligible(
159                "Metal execution requires one host/device unified capacity domain".to_string(),
160            ));
161        }
162        let queue = topology
163            .queue_resources
164            .iter()
165            .find(|queue| queue.id == accelerator.command_queue)
166            .ok_or_else(|| {
167                MetalRuntimeError::InvalidPlan(
168                    "Metal accelerator names a missing command queue".to_string(),
169                )
170            })?;
171        if queue.slots == 0 {
172            return Err(MetalRuntimeError::Ineligible(
173                "Metal command queue has no available occupancy".to_string(),
174            ));
175        }
176
177        let mut nodes = BTreeMap::new();
178        let mut used_allocations = BTreeSet::new();
179        for node in plan.nodes().values() {
180            let Some(demand_id) = node.metal_demand_id() else {
181                continue;
182            };
183            if !matches!(
184                node.kind,
185                WorkKind::Preparation
186                    | WorkKind::Jit
187                    | WorkKind::Compute
188                    | WorkKind::ObservationRead
189                    | WorkKind::ObservationReadWriteback
190                    | WorkKind::Spill
191            ) {
192                return Err(MetalRuntimeError::InvalidPlan(format!(
193                    "Metal node {} has runtime-incompatible work kind {:?}",
194                    node.id.as_str(),
195                    node.kind
196                )));
197            }
198            let fences = if node.domain == WorkDomain::Io {
199                BTreeSet::from([FenceKind::Io, FenceKind::Device])
200            } else {
201                BTreeSet::from([FenceKind::Device])
202            };
203            if node.fences != fences {
204                return Err(MetalRuntimeError::InvalidPlan(format!(
205                    "Metal node {} must declare its I/O and device completion fences",
206                    node.id.as_str()
207                )));
208            }
209            let demand = plan
210                .resource_alternative()
211                .demand
212                .accelerators
213                .iter()
214                .find(|candidate| candidate.demand_id == demand_id)
215                .ok_or_else(|| {
216                    MetalRuntimeError::InvalidPlan(format!(
217                        "Metal node {} names absent accelerator demand {demand_id}",
218                        node.id.as_str()
219                    ))
220                })?;
221            if demand.accelerator != accelerator.id
222                || demand.slots.hard() == 0
223                || demand.command_queue_slots.hard() == 0
224            {
225                return Err(MetalRuntimeError::Ineligible(format!(
226                    "Metal node {} is not charged to the inventoried device and command queue",
227                    node.id.as_str()
228                )));
229            }
230            let scheduled_allocations = node
231                .allocations
232                .iter()
233                .map(|use_| {
234                    let allocation = &plan.logical_allocations()[&use_.allocation];
235                    let slot = &plan.physical_slots()[&allocation.physical_slot];
236                    (
237                        allocation.id.clone(),
238                        (slot.id.clone(), slot.capacity_bytes),
239                    )
240                })
241                .collect();
242            let allocations = node
243                .allocations
244                .iter()
245                .filter(|use_| {
246                    plan.logical_allocations()[&use_.allocation]
247                        .compatibility
248                        .storage_mode
249                        == StorageMode::MetalShared
250                })
251                .map(|use_| use_.allocation.clone())
252                .collect::<Vec<_>>();
253            used_allocations.extend(allocations.iter().cloned());
254            nodes.insert(
255                node.id.clone(),
256                MetalNodeDecision {
257                    node: node.id.clone(),
258                    kind: node.kind,
259                    domain: node.domain.clone(),
260                    demand_id: demand_id.to_owned(),
261                    allocations,
262                    scheduled_allocations,
263                },
264            );
265        }
266        if nodes.is_empty() {
267            return Err(MetalRuntimeError::Ineligible(
268                "execution DAG contains no plan-selected Metal work".to_string(),
269            ));
270        }
271
272        let overhead = plan.resource_alternative().demand.overhead;
273        if overhead.driver_bytes == 0 || overhead.command_buffer_bytes == 0 {
274            return Err(MetalRuntimeError::InvalidPlan(
275                "Metal work requires positive driver and command-buffer envelopes".to_string(),
276            ));
277        }
278        if overhead.jit_bytes == 0 {
279            return Err(MetalRuntimeError::InvalidPlan(
280                "Metal pipeline preparation requires a positive JIT envelope".to_string(),
281            ));
282        }
283
284        let mut allocation_slots = BTreeMap::new();
285        let mut allocation_bytes = BTreeMap::new();
286        let mut physical_slots = BTreeMap::new();
287        for allocation_id in used_allocations {
288            let allocation = &plan.logical_allocations()[&allocation_id];
289            let slot = &plan.physical_slots()[&allocation.physical_slot];
290            if slot.compatibility.storage_mode != StorageMode::MetalShared
291                || slot.compatibility.memory_domain != domain.id
292                || !slot.compatibility.views.contains(&accelerator.memory_view)
293            {
294                return Err(MetalRuntimeError::InvalidPlan(format!(
295                    "Metal allocation {} is not backed by the inventoried unified memory view",
296                    allocation_id.as_str()
297                )));
298            }
299            physical_slots
300                .entry(slot.id.clone())
301                .and_modify(|bytes: &mut u64| *bytes = (*bytes).max(slot.capacity_bytes))
302                .or_insert(slot.capacity_bytes);
303            allocation_bytes.insert(allocation_id.clone(), allocation.bytes);
304            allocation_slots.insert(allocation_id, slot.id.clone());
305        }
306        physical_slots.values().try_fold(0_u64, |total, bytes| {
307            total
308                .checked_add(*bytes)
309                .ok_or(MetalRuntimeError::Overflow("Metal residency"))
310        })?;
311
312        let io = plan.resource_alternative().demand.io_buffers;
313        let transfers = plan
314            .resource_alternative()
315            .demand
316            .transfers
317            .iter()
318            .map(|demand| {
319                let link = topology
320                    .transfer_links
321                    .iter()
322                    .find(|link| link.id == demand.link)
323                    .ok_or_else(|| {
324                        MetalRuntimeError::InvalidPlan(format!(
325                            "Metal transfer demand {} names an absent link",
326                            demand.demand_id
327                        ))
328                    })?;
329                if link.source_view != accelerator.memory_view
330                    && link.destination_view != accelerator.memory_view
331                {
332                    return Err(MetalRuntimeError::InvalidPlan(format!(
333                        "Metal transfer demand {} does not reach the selected device view",
334                        demand.demand_id
335                    )));
336                }
337                Ok(link.id.clone())
338            })
339            .collect::<Result<Vec<_>, _>>()?;
340        if (io.bytes(IoBufferKind::HostToDeviceTransfer) > 0
341            || io.bytes(IoBufferKind::DeviceToHostTransfer) > 0)
342            && transfers.is_empty()
343        {
344            return Err(MetalRuntimeError::InvalidPlan(
345                "Metal staging bytes require an explicit transfer-link demand".to_string(),
346            ));
347        }
348        Ok(Self {
349            inventory: MetalRuntimeInventory {
350                accelerator: accelerator.id.clone(),
351                memory_domain: domain.id.clone(),
352                memory_view: accelerator.memory_view.clone(),
353                command_queue: accelerator.command_queue.clone(),
354            },
355            nodes,
356            allocation_slots,
357            allocation_bytes,
358            physical_slots,
359            transfers,
360            host_to_device_staging_bytes: io.bytes(IoBufferKind::HostToDeviceTransfer),
361            device_to_host_staging_bytes: io.bytes(IoBufferKind::DeviceToHostTransfer),
362            resident_cache_bytes: plan
363                .resource_alternative()
364                .demand
365                .caches
366                .hard_resident_bytes,
367            overhead,
368        })
369    }

519        };

559    #[cfg(all(target_os = "macos", not(coverage)))]
560    pub(crate) fn allocate_replay(
561        &self,
562        reservation: crate::ResourceLease,
563        bytes: usize,
564    ) -> Result<Option<MetalReplayBuffer>, MetalRuntimeError> {
565        let inner = self.lock()?;
566        let platform = inner.platform.as_ref().expect("prepared platform");
567        if bytes == 0 || bytes as u64 > reservation.demand().caches.hard_resident_bytes {
568            return Err(MetalRuntimeError::InvalidPlan(
569                "unadmitted replay buffer".into(),
570            ));
571        }
572        let resident: usize = platform.buffers.values().map(|b| b.length()).sum();
573        if bytes > platform.device.maxBufferLength()
574            || resident
575                .checked_add(bytes)
576                .is_none_or(|n| n as u64 > platform.device.recommendedMaxWorkingSetSize())
577        {
578            return Ok(None);
579        }
580        let buffer = platform
581            .device
582            .newBufferWithLength_options(bytes, MTLResourceOptions::StorageModeShared)
583            .ok_or_else(|| MetalRuntimeError::AllocationFailed {
584                slot: PhysicalSlotId::new("run-metal-normal-replay"),
585                bytes: bytes as u64,
586            })?;
587        Ok(Some(MetalReplayBuffer {
588            buffer,
589            device: platform.device.registryID(),
590            bytes,
591            _reservation: reservation,
592        }))
593    }

604    fn lock(&self) -> Result<MutexGuard<'_, MetalExecutionInner>, MetalRuntimeError> {
605        let inner = self
606            .runtime
607            .inner
608            .lock()
609            .map_err(|_| MetalRuntimeError::RuntimeStatePoisoned)?;
610        if inner.closed {
611            return Err(MetalRuntimeError::RuntimeClosed);
612        }
613        if inner.attempt_id != Some(self.attempt) {
614            return Err(MetalRuntimeError::LeaseMismatch);
615        }
616        if inner.nodes.get(&self.node).is_some_and(|p| p.finished) {
617            return Err(MetalRuntimeError::NodeFinished(self.node.clone()));
618        }
619        Ok(inner)
620    }

622    #[cfg(all(target_os = "macos", not(coverage)))]
623    pub(crate) fn with_bytes<R>(
624        &self,
625        region: MetalBufferRegion<'_>,
626        use_bytes: impl FnOnce(&mut [u8]) -> R,
627    ) -> Result<R, MetalRuntimeError> {
628        let inner = self.lock()?;
629        let platform = inner.platform.as_ref().expect("admitted prepared platform");
630        if platform.pending.values().any(|pending| {
631            pending.regions.iter().any(|(id, start, bytes)| {
632                id == region.allocation
633                    && region.offset < start + bytes
634                    && *start < region.offset + region.bytes
635            })
636        }) {
637            return Err(MetalRuntimeError::InvalidPlan(
638                "CPU access overlaps in-flight GPU work".into(),
639            ));
640        }
641        let (buffer, offset) = buffer_region(&self.runtime.decision, platform, &self.node, region)?;
642        let bytes = unsafe {
643            std::slice::from_raw_parts_mut(
644                buffer.contents().as_ptr().cast::<u8>().add(offset),
645                region.bytes,
646            )
647        };
648        Ok(use_bytes(bytes))
649    }

660    pub(crate) fn execute(
661        &self,
662        dispatches: &[CubeDispatch<'_>],
663    ) -> Result<MetalBatchStats, MetalRuntimeError> {
664        if dispatches.is_empty() {
665            return Err(MetalRuntimeError::InvalidPlan("empty compute batch".into()));
666        }
667        let mut inner = self.lock()?;
668        if inner.nodes[&self.node].failed {
669            return Err(MetalRuntimeError::InvalidPlan(
670                "failed Metal node cannot dispatch".into(),
671            ));
672        }
673        metal_call(|| drain_cube_commands(&self.runtime.decision, &mut inner, None))?;
674        let before = inner.nodes[&self.node].stats;
675        let ticket = metal_call(|| {
676            submit_platform_batch(&self.runtime.decision, &mut inner, &self.node, dispatches)
677        })?;
678        metal_call(|| drain_cube_commands(&self.runtime.decision, &mut inner, Some(ticket)))?;
679        let after = inner.nodes[&self.node].stats;
680        Ok(MetalBatchStats {
681            batches: after.batches - before.batches,
682            grid_samples: after.grid_samples - before.grid_samples,
683            degrid_samples: after.degrid_samples - before.degrid_samples,
684            gpu_seconds: after.gpu_seconds - before.gpu_seconds,
685            submit_wait_seconds: after.submit_wait_seconds - before.submit_wait_seconds,
686            profiled_batches: after.profiled_batches - before.profiled_batches,
687            prediction_gpu_seconds: after.prediction_gpu_seconds - before.prediction_gpu_seconds,
688            residual_gpu_seconds: after.residual_gpu_seconds - before.residual_gpu_seconds,
689        })
690    }

692    /// Enqueue a bounded batch, pinning its regions until the returned ticket settles.
693    pub(crate) fn submit(&self, dispatches: &[CubeDispatch<'_>]) -> Result<u64, MetalRuntimeError> {
694        if dispatches.is_empty() {
695            return Err(MetalRuntimeError::InvalidPlan("empty compute batch".into()));
696        }
697        let mut inner = self.lock()?;
698        if inner.nodes[&self.node].failed {
699            return Err(MetalRuntimeError::InvalidPlan(
700                "failed Metal node cannot dispatch".into(),
701            ));
702        }
703        metal_call(|| {
704            submit_platform_batch(&self.runtime.decision, &mut inner, &self.node, dispatches)
705        })
706    }

909    fn lock_for_work(
910        &self,
911        context: WorkExecutionContext<'_>,
912    ) -> Result<MutexGuard<'_, MetalExecutionInner>, MetalRuntimeError> {
913        validate_execution_context(&self.decision, context, false)?;
914        if context.lease_epoch() != self.lease_epoch {
915            return Err(MetalRuntimeError::LeaseMismatch);
916        }
917        let mut inner = self
918            .inner
919            .lock()
920            .map_err(|_| MetalRuntimeError::RuntimeStatePoisoned)?;
921        if inner.closed {
922            return Err(MetalRuntimeError::RuntimeClosed);
923        }
924        match inner.attempt_id {
925            Some(attempt_id) if attempt_id != context.attempt_id() => {
926                return Err(MetalRuntimeError::LeaseMismatch);
927            }
928            None => inner.attempt_id = Some(context.attempt_id()),
929            Some(_) => {}
930        }
931        if inner
932            .nodes
933            .get(&context.node().id)
934            .is_some_and(|node| node.finished)
935        {
936            return Err(MetalRuntimeError::NodeFinished(context.node().id.clone()));
937        }
938        Ok(inner)
939    }

941    /// Prepare pipelines and materialize only this admitted node's device slots.
942    /// No command is fabricated for preparation-only work.
943    pub(crate) fn prepare(
944        &self,
945        context: WorkExecutionContext<'_>,
946    ) -> Result<(), MetalRuntimeError> {
947        let mut inner = self.lock_for_work(context)?;
948        metal_call(|| prepare_platform(&self.decision, &mut inner, &context.node().id))?;
949        inner

951            .entry(context.node().id.clone())

957    /// Consume terminal device evidence once, after any number of drained
958    /// batches. This is distinct from batch completion and works in either
959    /// terminal I/O/device fence order.
960    pub(crate) fn finish(
961        &self,
962        context: WorkExecutionContext<'_>,
963    ) -> Result<WorkMeasurements, MetalRuntimeError> {
964        validate_execution_context(&self.decision, context, true)?;
965        let mut inner = self
966            .inner
967            .lock()
968            .map_err(|_| MetalRuntimeError::RuntimeStatePoisoned)?;
969        if inner.closed {
970            return Err(MetalRuntimeError::RuntimeClosed);
971        }
972        if context.lease_epoch() != self.lease_epoch
973            || inner.attempt_id != Some(context.attempt_id())
974        {
975            return Err(MetalRuntimeError::LeaseMismatch);
976        }
977        metal_call(|| drain_cube_commands(&self.decision, &mut inner, None))?;
978        let progress = inner
979            .nodes
980            .get_mut(&context.node().id)
981            .ok_or(MetalRuntimeError::FenceAlreadySettled)?;
982        if progress.finished {
983            return Err(MetalRuntimeError::FenceAlreadySettled);
984        }
985        if progress.failed {
986            return Err(MetalRuntimeError::InvalidPlan(
987                "failed Metal node cannot complete".into(),
988            ));
989        }
990        if !progress.prepared
991            || (requires_dispatch(context.node().kind)
992                && progress.stats.batches == 0
993                && !progress.empty_source)
994        {
995            return Err(MetalRuntimeError::InvalidPlan(
996                "Metal compute stage did not dispatch".into(),
997            ));
998        }
999        progress.finished = true;
1000        Ok(measurements(context))
1001    }

1003    pub(crate) fn submitted(&self, node: &WorkNodeId) -> Result<bool, MetalRuntimeError> {
1004        self.inner
1005            .lock()
1006            .map(|inner| {
1007                inner.nodes.get(node).is_some_and(|progress| {
1008                    if requires_dispatch(self.decision.nodes[node].kind) {
1009                        progress.stats.batches > 0 || progress.empty_source
1010                    } else {
1011                        progress.prepared
1012                    }
1013                })
1014            })
1015            .map_err(|_| MetalRuntimeError::RuntimeStatePoisoned)
1016    }

1019        let mut inner = self
1020            .inner
1021            .lock()
1022            .map_err(|_| MetalRuntimeError::RuntimeStatePoisoned)?;
1023        metal_call(|| drain_cube_commands(&self.decision, &mut inner, None))?;
1024        metal_call(|| close_platform(&mut inner))?;
1025        inner.closed = true;
1026        Ok(())
1027    }

1029    /// Called by the scheduler at the allocation ledger's actual release event,
1030    /// before its reservation can be reused by another logical allocation.
1031    pub(crate) fn release_allocation(
1032        &self,
1033        allocation: &AllocationId,
1034    ) -> Result<(), MetalRuntimeError> {
1035        let Some(slot) = self.decision.allocation_slots.get(allocation) else {
1036            return Ok(());
1037        };
1038        let mut inner = self
1039            .inner
1040            .lock()
1041            .map_err(|_| MetalRuntimeError::RuntimeStatePoisoned)?;
1042        metal_call(|| drain_cube_commands(&self.decision, &mut inner, None))?;
1043        #[cfg(all(target_os = "macos", not(coverage)))]
1044        if let Some(platform) = inner.platform.as_mut() {
1045            platform.buffers.remove(slot);
1046        }
1047        #[cfg(not(all(target_os = "macos", not(coverage))))]
1048        let _ = (slot, &mut inner);
1049        Ok(())
1050    }

1063/// Typed Metal eligibility, plan, allocation, or command failure.
1064#[derive(Clone, Debug, PartialEq, Eq)]
1065pub enum MetalRuntimeError {
1066    /// This build cannot execute Metal work.
1067    UnsupportedPlatform,
1068    /// The selected host or device does not satisfy the Metal contract.
1069    Ineligible(String),
1070    /// The immutable execution DAG is inconsistent with the Metal runtime.
1071    InvalidPlan(String),
1072    /// The work context belongs to another execution attempt or lease.
1073    LeaseMismatch,
1074    /// The execution residency was closed before another submission.
1075    RuntimeClosed,
1076    /// The runtime command state could not be observed safely.
1077    RuntimeStatePoisoned,
1078    /// A checked byte calculation overflowed.
1079    Overflow(&'static str),
1080    /// A plan-selected node was absent from the Metal decision.
1081    UnknownNode(WorkNodeId),
1082    /// Work attempted another batch after terminal device completion.
1083    NodeFinished(WorkNodeId),
1084    /// A pipeline or typed dispatch could not be encoded.
1085    Encoding(String),
1086    /// The selected device could not create its command queue or buffer.
1087    CommandQueueUnavailable,
1088    /// One resident allocation exceeds the device's buffer limit.
1089    BufferTooLarge {
1090        /// Exact plan-owned physical slot.
1091        slot: PhysicalSlotId,
1092        /// Required resident bytes.
1093        bytes: u64,
1094    },
1095    /// Metal could not allocate a plan-selected physical slot.
1096    AllocationFailed {
1097        /// Exact plan-owned physical slot.
1098        slot: PhysicalSlotId,
1099        /// Requested resident bytes.
1100        bytes: u64,
1101    },
1102    /// A command fence was consumed more than once.
1103    FenceAlreadySettled,
1104    /// The native command completed with an error state.
1105    CommandFailed {
1106        /// Exact plan-selected node.
1107        node: WorkNodeId,
1108        /// Native `MTLCommandBufferStatus` value.
1109        status: u64,
1110    },
1111}

1164#[allow(
1165    dead_code,
1166    reason = "T57 consumes the constrained Metal execution seam"
1167)]
1168fn validate_execution_context(
1169    decision: &MetalExecutionDecision,
1170    context: WorkExecutionContext<'_>,
1171    terminal: bool,
1172) -> Result<(), MetalRuntimeError> {
1173    let node = decision
1174        .nodes
1175        .get(&context.node().id)
1176        .ok_or_else(|| MetalRuntimeError::UnknownNode(context.node().id.clone()))?;
1177    if context.node().kind != node.kind
1178        || context.node().domain != node.domain
1179        || context.node().metal_demand_id() != Some(node.demand_id.as_str())
1180        || context.resources().len()
1181            != context
1182                .node()
1183                .claims
1184                .iter()
1185                .filter(|claim| !terminal || claim.lifetime.retains_fence(FenceKind::Device))
1186                .count()
1187        || context
1188            .node()
1189            .claims
1190            .iter()
1191            .filter(|claim| !terminal || claim.lifetime.retains_fence(FenceKind::Device))
1192            .any(|claim| {
1193                !context.resources().iter().any(|resource| {
1194                    resource.resource() == &claim.resource
1195                        && resource.amount() == claim.amount
1196                        && resource.lifetime() == &claim.lifetime
1197                })
1198            })
1199        || context.node().allocations.iter().any(|use_| {
1200            !context.allocations().iter().any(|allocation| {
1201                allocation.allocation() == &use_.allocation
1202                    && allocation.lifetime() == &use_.lifetime
1203            })
1204        })
1205    {
1206        return Err(MetalRuntimeError::LeaseMismatch);
1207    }
1208    let owns_accelerator = context.resources().iter().any(|resource| {
1209        resource.amount() == 1
1210            && matches!(
1211                resource.resource(),
1212                LeaseResource::Accelerator { demand_id } if demand_id == &node.demand_id
1213            )
1214    });
1215    let owns_queue = context.resources().iter().any(|resource| {
1216        resource.amount() == 1
1217            && matches!(
1218                resource.resource(),
1219                LeaseResource::AcceleratorCommandQueue { demand_id }
1220                    if demand_id == &node.demand_id
1221            )
1222    });
1223    if !owns_accelerator || !owns_queue {
1224        return Err(MetalRuntimeError::LeaseMismatch);
1225    }
1226    let scheduled_allocations = context
1227        .allocations()
1228        .iter()
1229        .map(|allocation| {
1230            (
1231                allocation.allocation().clone(),
1232                (
1233                    allocation.physical_slot().clone(),
1234                    allocation.capacity_bytes(),
1235                ),
1236            )
1237        })
1238        .collect::<BTreeMap<_, _>>();
1239    if scheduled_allocations != node.scheduled_allocations {
1240        return Err(MetalRuntimeError::LeaseMismatch);
1241    }
1242    Ok(())
1243}

1314#[cfg(all(target_os = "macos", not(coverage)))]
1315fn prepare_platform(
1316    decision: &MetalExecutionDecision,
1317    inner: &mut MetalExecutionInner,
1318    node: &WorkNodeId,
1319) -> Result<(), MetalRuntimeError> {
1320    if inner.platform.is_none() {
1321        inner.platform = Some(open_platform_state(decision)?);
1322    }
1323    let platform = inner.platform.as_mut().expect("platform initialized");
1324    let slots = decision.nodes[node]
1325        .allocations
1326        .iter()
1327        .map(|id| &decision.allocation_slots[id])
1328        .collect::<BTreeSet<_>>();
1329    let added_bytes: u64 = slots
1330        .iter()
1331        .filter(|slot| !platform.buffers.contains_key(**slot))
1332        .map(|slot| decision.physical_slots[*slot])
1333        .sum();
1334    let resident_bytes: u64 = platform
1335        .buffers
1336        .values()
1337        .map(|buffer| buffer.length() as u64)
1338        .sum();
1339    if resident_bytes
1340        .checked_add(added_bytes)
1341        .is_none_or(|bytes| bytes > platform.device.recommendedMaxWorkingSetSize())
1342    {
1343        return Err(MetalRuntimeError::Ineligible(
1344            "admitted live Metal slots exceed device working-set recommendation".into(),
1345        ));
1346    }
1347    for slot in slots {
1348        if platform.buffers.contains_key(slot) {
1349            continue;
1350        }
1351        let bytes = decision.physical_slots[slot];
1352        let maximum_buffer_bytes = platform.device.maxBufferLength() as u64;
1353        if bytes > maximum_buffer_bytes {
1354            return Err(MetalRuntimeError::BufferTooLarge {
1355                slot: slot.clone(),
1356                bytes,
1357            });
1358        }
1359        let buffer = platform
1360            .device
1361            .newBufferWithLength_options(bytes as usize, MTLResourceOptions::StorageModeShared)
1362            .ok_or_else(|| MetalRuntimeError::AllocationFailed {
1363                slot: slot.clone(),
1364                bytes,
1365            })?;
1366        platform.buffers.insert(slot.clone(), buffer);
1367    }
1368    if platform.kernels.is_none() {
1369        platform.kernels = Some(
1370            crate::metal_cube::MetalCubeKernels::compile(&platform.device)
1371                .map_err(MetalRuntimeError::Encoding)?,
1372        );
1373    }
1374    Ok(())
1375}

1387fn buffer_region<'p>(
1388    decision: &MetalExecutionDecision,
1389    platform: &'p MetalPlatformState,
1390    node: &WorkNodeId,
1391    region: MetalBufferRegion<'_>,
1392) -> Result<(&'p ProtocolObject<dyn MTLBuffer>, usize), MetalRuntimeError> {

1394        || region.bytes == 0
1395        || region
1396            .offset
1397            .checked_add(region.bytes)

1400        return Err(MetalRuntimeError::InvalidPlan(

1402        ));

1406        .get(&decision.allocation_slots[region.allocation])
1407        .ok_or_else(|| MetalRuntimeError::InvalidPlan("Metal allocation is not resident".into()))?;
1408    Ok((buffer.as_ref(), region.offset))

1411#[cfg(all(target_os = "macos", not(coverage)))]
1412fn submit_platform_batch(
1413    decision: &MetalExecutionDecision,
1414    inner: &mut MetalExecutionInner,
1415    node: &WorkNodeId,
1416    dispatches: &[CubeDispatch<'_>],
1417) -> Result<u64, MetalRuntimeError> {
1418    let platform = inner.platform.as_mut().expect("prepared platform");
1419    let kernels = platform.kernels.as_ref().expect("prepared pipelines");
1420    let command = platform
1421        .queue
1422        .commandBuffer()
1423        .ok_or(MetalRuntimeError::CommandQueueUnavailable)?;
1424    let mut stats = MetalBatchStats {
1425        batches: 1,
1426        ..MetalBatchStats::default()
1427    };
1428    let mut regions = Vec::with_capacity(dispatches.len() * 4);
1429    for dispatch in dispatches {
1430        let region = |r| buffer_region(decision, platform, node, r);
1431        let samples = region(dispatch.samples)?;
1432        let weights = region(dispatch.weights)?;
1433        let grid = region(dispatch.grid)?;
1434        let count = dispatch.count as usize;
1435        let output = match dispatch.kind {
1436            CubeDispatchKind::Grid => dispatch.grid,
1437            CubeDispatchKind::Degrid { predicted } => predicted,
1438        };
1439        regions.extend([dispatch.samples, dispatch.weights, dispatch.grid]);
1440        if let CubeDispatchKind::Degrid { predicted } = dispatch.kind {
1441            regions.push(predicted);
1442        }
1443        if count == 0
1444            || dispatch.width < 7
1445            || dispatch.height < 7
1446            || dispatch.samples.offset % 8 != 0
1447            || dispatch.grid.offset % 8 != 0
1448            || dispatch.weights.offset % 4 != 0
1449            || count
1450                .checked_mul(size_of::<crate::metal_cube::CubeTap>())
1451                .is_none_or(|bytes| bytes > dispatch.samples.bytes)
1452            || (dispatch.width as usize)
1453                .checked_mul(dispatch.height as usize)
1454                .and_then(|cells| cells.checked_mul(8))
1455                .is_none_or(|bytes| bytes > dispatch.grid.bytes)
1456            || dispatch.weights.bytes < 7 * 4
1457            || dispatch.width.checked_mul(dispatch.height).is_none()
1458            || output.overlaps(dispatch.samples)
1459            || output.overlaps(dispatch.weights)
1460            || (matches!(dispatch.kind, CubeDispatchKind::Degrid { .. })
1461                && output.overlaps(dispatch.grid))
1462        {
1463            return Err(MetalRuntimeError::InvalidPlan(
1464                "invalid cube dispatch shape or buffer capacity".into(),
1465            ));
1466        }
1467        // Validate the externally encoded tap layout before the GPU can access
1468        // it. This is a bounded input check, never a grid-content verification.
1469        if platform.pending.values().any(|pending| {
1470            pending.regions.iter().any(|(id, start, bytes)| {
1471                id == dispatch.samples.allocation
1472                    && dispatch.samples.offset < start + bytes
1473                    && *start < dispatch.samples.offset + dispatch.samples.bytes
1474            })
1475        }) {
1476            return Err(MetalRuntimeError::InvalidPlan(
1477                "CPU tap validation overlaps in-flight GPU work".into(),
1478            ));
1479        }
1480        let taps = unsafe {
1481            std::slice::from_raw_parts(
1482                samples
1483                    .0
1484                    .contents()
1485                    .as_ptr()
1486                    .cast::<u8>()
1487                    .add(samples.1)
1488                    .cast::<crate::metal_cube::CubeTap>(),
1489                count,
1490            )
1491        };
1492        let weight_rows = dispatch.weights.bytes / (7 * 4);
1493        if taps.iter().any(|tap| {
1494            tap.x > dispatch.width - 7
1495                || tap.y > dispatch.height - 7
1496                || tap.x_weights as usize >= weight_rows
1497                || tap.y_weights as usize >= weight_rows
1498        }) {
1499            return Err(MetalRuntimeError::InvalidPlan(
1500                "cube tap exceeds grid or convolution table".into(),
1501            ));
1502        }
1503        match dispatch.kind {
1504            CubeDispatchKind::Grid => {
1505                kernels
1506                    .encode_grid(
1507                        &command,
1508                        samples,
1509                        weights,
1510                        grid,
1511                        dispatch.count,
1512                        dispatch.width,
1513                        dispatch.height,
1514                    )
1515                    .map_err(MetalRuntimeError::Encoding)?;
1516                stats.grid_samples += u64::from(dispatch.count);
1517            }
1518            CubeDispatchKind::Degrid { predicted } => {
1519                if predicted.offset % 8 != 0
1520                    || count
1521                        .checked_mul(8)
1522                        .is_none_or(|bytes| bytes > predicted.bytes)
1523                {
1524                    return Err(MetalRuntimeError::InvalidPlan(
1525                        "invalid predicted visibility capacity".into(),
1526                    ));
1527                }
1528                kernels
1529                    .encode_degrid(
1530                        &command,
1531                        samples,
1532                        weights,
1533                        grid,
1534                        region(predicted)?,
1535                        dispatch.count,
1536                        dispatch.width,
1537                        dispatch.height,
1538                    )
1539                    .map_err(MetalRuntimeError::Encoding)?;
1540                stats.degrid_samples += u64::from(dispatch.count);
1541            }
1542        }
1543    }
1544    // Adjacent plane regions share one arena. Coalesce them so CPU mapping
1545    // checks stay proportional to live arenas, not to every plane dispatch.
1546    regions.sort_unstable_by(|left, right| {
1547        left.allocation
1548            .cmp(right.allocation)
1549            .then(left.offset.cmp(&right.offset))
1550    });
1551    let mut owned: Vec<(AllocationId, usize, usize)> = Vec::with_capacity(regions.len());
1552    for region in regions {
1553        if let Some((allocation, offset, bytes)) = owned.last_mut()
1554            && allocation == region.allocation
1555            && region.offset <= *offset + *bytes
1556        {
1557            *bytes = (*offset + *bytes).max(region.offset + region.bytes) - *offset;
1558        } else {
1559            owned.push((region.allocation.clone(), region.offset, region.bytes));
1560        }
1561    }
1562    let ticket = platform.next_ticket;
1563    platform.next_ticket = ticket
1564        .checked_add(1)
1565        .ok_or(MetalRuntimeError::Overflow("command ticket"))?;
1566    let started = Instant::now();
1567    command.commit();
1568    stats.submit_wait_seconds = started.elapsed().as_secs_f64();
1569    platform.pending.insert(
1570        ticket,
1571        PendingCubeCommand {
1572            command,
1573            node: node.clone(),
1574            regions: owned,
1575            status: None,
1576            stats,
1577            stage_profile: None,
1578            _normal_replay: None,
1579        },
1580    );
1581    Ok(ticket)
1582}

1845/// Drain every selected ticket even if one fails. Mutable access is released only
1846/// after the command fence; failure never permits retry into partly filled grids.
1847#[cfg(all(target_os = "macos", not(coverage)))]
1848fn drain_cube_commands(
1849    decision: &MetalExecutionDecision,
1850    inner: &mut MetalExecutionInner,
1851    ticket: Option<u64>,
1852) -> Result<(), MetalRuntimeError> {
1853    let Some(platform) = inner.platform.as_mut() else {
1854        return Ok(());
1855    };
1856    let tickets: Vec<_> = match ticket {
1857        Some(ticket) if platform.pending.contains_key(&ticket) => vec![ticket],
1858        Some(_) => return Err(MetalRuntimeError::FenceAlreadySettled),
1859        None => platform.pending.keys().copied().collect(),
1860    };
1861    let mut error = None;
1862    for ticket in tickets {
1863        let mut pending = platform.pending.remove(&ticket).expect("selected command");
1864        let mut failed = false;
1865        let started = Instant::now();
1866        pending.command.waitUntilCompleted();
1867        pending.stats.submit_wait_seconds += started.elapsed().as_secs_f64();
1868        if pending.command.status() != MTLCommandBufferStatus::Completed {
1869            failed = true;
1870            error.get_or_insert_with(|| MetalRuntimeError::CommandFailed {
1871                node: pending.node.clone(),
1872                status: pending.command.status().0 as u64,
1873            });
1874        } else if let Some(status) = &pending.status {
1875            let region = MetalBufferRegion {
1876                allocation: &status.allocation,
1877                offset: status.offset,
1878                bytes: status.bytes,
1879            };
1880            match buffer_region(decision, platform, &pending.node, region) {
1881                Ok((buffer, offset)) => {
1882                    let value = unsafe {
1883                        *buffer
1884                            .contents()
1885                            .as_ptr()
1886                            .cast::<u8>()
1887                            .add(offset)
1888                            .cast::<u32>()
1889                    };
1890                    if value != 0 {
1891                        failed = true;
1892                        error.get_or_insert_with(|| {
1893                            MetalRuntimeError::Encoding(format!(
1894                                "connected cube invalid geometry/numerical input (status {value})"
1895                            ))
1896                        });
1897                    }
1898                }
1899                Err(failure) => {
1900                    failed = true;
1901                    error.get_or_insert(failure);
1902                }
1903            }
1904        }
1905        pending.stats.gpu_seconds =
1906            (pending.command.GPUEndTime() - pending.command.GPUStartTime()).max(0.0);
1907        if !failed && let Some(profile) = pending.stage_profile {
1908            match profile.seconds() {
1909                Ok([prediction, residual]) => {
1910                    pending.stats.profiled_batches = 1;
1911                    pending.stats.prediction_gpu_seconds = prediction;
1912                    pending.stats.residual_gpu_seconds = residual;
1913                }
1914                Err(message) => {
1915                    failed = true;
1916                    error.get_or_insert(MetalRuntimeError::Encoding(message));
1917                }
1918            }
1919        }
1920        let progress = inner.nodes.get_mut(&pending.node).expect("prepared node");
1921        progress.failed |= failed;
1922        progress.stats.batches += pending.stats.batches;
1923        progress.stats.grid_samples += pending.stats.grid_samples;
1924        progress.stats.degrid_samples += pending.stats.degrid_samples;
1925        progress.stats.gpu_seconds += pending.stats.gpu_seconds;
1926        progress.stats.submit_wait_seconds += pending.stats.submit_wait_seconds;
1927        progress.stats.profiled_batches += pending.stats.profiled_batches;
1928        progress.stats.prediction_gpu_seconds += pending.stats.prediction_gpu_seconds;
1929        progress.stats.residual_gpu_seconds += pending.stats.residual_gpu_seconds;
1930    }
1931    error.map_or(Ok(()), Err)
1932}
