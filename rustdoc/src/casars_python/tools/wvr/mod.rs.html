19#[pymethods]
20impl PyWvr {
21    #[new]
22    fn new() -> Self {
23        Self
24    }
25
26    #[pyo3(signature = (vis = "", output = "", toffset = 0.0, nsol = 1, segsource = false, reverse = false, reversespw = None, disperse = false, cont = false, wvrflag = "", sourceflag = None, statfield = None, statsource = None, tie = None, smooth = 1, scale = 1.0, maxdistm = 500.0, minnumants = 2, mingoodfrac = 0.8, usefieldtab = false, spw = None, wvrspw = None, refant = "", offsets = "", rseed = 0, logfile = "wvrgcal.log"), text_signature = "($self, vis='', output='', toffset=0.0, nsol=1, segsource=False, reverse=False, reversespw=[], disperse=False, cont=False, wvrflag='', sourceflag=[], statfield=[], statsource=[], tie=[], smooth=1, scale=1.0, maxdistm=500.0, minnumants=2, mingoodfrac=0.8, usefieldtab=False, spw=[], wvrspw=[], refant='', offsets='', rseed=0, logfile='wvrgcal.log')")]
27    #[allow(clippy::too_many_arguments)]
28    fn gcal(
29        &self,
30        py: Python<'_>,
31        vis: &str,
32        output: &str,
33        toffset: f64,
34        nsol: i64,
35        segsource: bool,
36        reverse: bool,
37        reversespw: Option<&Bound<'_, PyAny>>,
38        disperse: bool,
39        cont: bool,
40        wvrflag: &str,
41        sourceflag: Option<&Bound<'_, PyAny>>,
42        statfield: Option<&Bound<'_, PyAny>>,
43        statsource: Option<&Bound<'_, PyAny>>,
44        tie: Option<&Bound<'_, PyAny>>,
45        smooth: i64,
46        scale: f64,
47        maxdistm: f64,
48        minnumants: i64,
49        mingoodfrac: f64,
50        usefieldtab: bool,
51        spw: Option<&Bound<'_, PyAny>>,
52        wvrspw: Option<&Bound<'_, PyAny>>,
53        refant: &str,
54        offsets: &str,
55        rseed: i64,
56        logfile: &str,
57    ) -> PyResult<i32> {
58        let Ok(seed) = u32::try_from(rseed) else {
59            return warn(py, "rseed must be an integer from 0 through 4294967295", -1);
60        };
61        let selectors = WvrSelectors {
62            flagged_antennas: vec![wvrflag.into()],
63            reference_antennas: vec![refant.into()],
64            flagged_sources: string_vector(sourceflag, "sourceflag")?,
65            tied_sources: string_vector(tie, "tie")?,
66            statistic_sources: string_vector(statsource, "statsource")?,
67            statistic_fields: string_vector(statfield, "statfield")?,
68            ..WvrSelectors::default()
69        };
70        if vis.is_empty() || output.is_empty() {
71            return warn(
72                py,
73                "An input MeasurementSet and output table are required",
74                -1,
75            );
76        }
77        if !segsource
78            && (!selectors.flagged_sources.is_empty() || !selectors.tied_sources.is_empty())
79        {
80            return warn(py, "sourceflag and tie require segsource", -1);
81        }
82        let mut request = WvrRequest {
83            measurement_set: PathBuf::from(vis),
84            output_table: PathBuf::from(output),
85            time_offset_seconds: toffset,
86            temperature_offsets: (!offsets.is_empty()).then(|| PathBuf::from(offsets)),
87            disperse,
88            radiometer_windows: integer_vector(wvrspw, "wvrspw")?,
89            output_windows: integer_vector(spw, "spw")?,
90            reverse,
91            reversed_windows: integer_vector(reversespw, "reversespw")?,
92            segment_sources: segsource,
93            solution_count: usize::try_from(nsol).unwrap_or(0),
94            tied_sources: Vec::new(),
95            flagged_sources: Vec::new(),
96            statistic_sources: Vec::new(),
97            statistic_fields: Vec::new(),
98            use_field_table: usefieldtab,
99            continuum: cont,
100            seed,
101            smoothing_samples: usize::try_from(smooth).unwrap_or(0),
102            scale,
103            flagged_antennas: Vec::new(),
104            reference_antennas: Vec::new(),
105            max_distance_meters: maxdistm,
106            minimum_neighbors: usize::try_from(minnumants).unwrap_or(usize::MAX),
107            minimum_good_fraction: mingoodfrac,
108        };
109        let logfile = if logfile.is_empty() {
110            "wvrgcal.log"
111        } else {
112            logfile
113        };
114        let outcome = py.detach(move || {
115            resolve_wvr_selectors(&mut request, &selectors)?;
116            let report = execute_wvr_with_log(&request, Path::new(logfile))?;
117            Ok::<_, WvrError>(report.status)
118        });
119        match outcome {
120            Ok(status) => Ok(status),
121            Err(WvrError::Invalid(message)) => warn(py, &message, -1),
122            Err(WvrError::Input(error)) => warn(py, &error.to_string(), 1),
123            Err(
124                error @ (WvrError::NoSolution(_)
125                | WvrError::ZeroEvidence(_)
126                | WvrError::ReverseWindow(_)),
127            ) => warn(py, &error.to_string(), 1),
128            Err(error @ WvrError::ReferenceUnavailable) => warn(py, &error.to_string(), -1),
129            Err(error @ WvrError::RadiometerWindow(_)) => warn(py, &error.to_string(), -10),
130            Err(error @ WvrError::OutputWindow(_)) => warn(py, &error.to_string(), -11),
131            Err(error) => Err(PyRuntimeError::new_err(error.to_string())),
132        }
133    }
134}

136fn warn(py: Python<'_>, message: &str, status: i32) -> PyResult<i32> {
137    emit(py, "casars.tools.wvr", &[LogLine::warning(message)])?;
138    Ok(status)
139}

141fn string_vector(value: Option<&Bound<'_, PyAny>>, name: &str) -> PyResult<Vec<String>> {
142    let Some(value) = value else {
143        return Ok(Vec::new());
144    };
145    if let Ok(text) = value.extract::<String>() {
146        return Ok(vec![text]);
147    }
148    if value.extract::<i64>().is_ok() {
149        return Ok(Vec::new());
150    }
151    value
152        .extract::<Vec<String>>()
153        .map_err(|_| PyTypeError::new_err(format!("{name} must contain strings")))
154}
155
156fn integer_vector(value: Option<&Bound<'_, PyAny>>, name: &str) -> PyResult<Vec<i32>> {
157    let Some(value) = value else {
158        return Ok(Vec::new());
159    };
160    if let Ok(integer) = value.extract::<i32>() {
161        return Ok(vec![integer]);
162    }
163    let error = || PyTypeError::new_err(format!("{name} must contain integers"));
164    let items = value
165        .extract::<Vec<Bound<'_, PyAny>>>()
166        .map_err(|_| error())?;
167    items
168        .iter()
169        .map(|item| {
170            item.extract::<i32>()
171                .or_else(|_| item.extract::<String>()?.parse().map_err(|_| error()))
172        })
173        .collect()
174}
