2//! Philox4x32-10, the counter-based generator of Salmon, Moraes, Dror and
3//! Shaw, "Parallel random numbers: as easy as 1, 2, 3" (SC11), as
4//! Random123's `philox4x32` implements it.
5//!
6//! A draw is a pure function of a 128-bit counter and a 64-bit key: ten
7//! rounds of two 32×32→64-bit multiplies, with the key bumped by the Weyl
8//! constants between rounds. There is no state, so draws can be computed in
9//! any order, on any number of workers, and are the same every time.
10//!
11//! Draws are made several counters at a time ([`philox4x32_lanes`]), held
12//! word by word: every lane takes the same steps, so the lanes' multiplies
13//! run side by side in vector registers, and each lane's words are exactly
14//! those of that counter drawn alone.
15//!
16//! The known-answer tests are Random123's own (`tests/kat_vectors` in
17//! <https://github.com/DEShawResearch/random123>, fetched 2026-10-21).
18
19/// Philox's 128-bit counter.
20#[cfg(test)]
21pub(crate) type Counter = [u32; 4];
22/// Philox's 64-bit key.
23pub(crate) type Key = [u32; 2];
24/// The counters of `LANES` draws, word by word: `lanes[word][lane]` is word
25/// `word` of lane `lane`'s 128-bit counter.
26pub(crate) type Lanes<const LANES: usize> = [[u32; LANES]; 4];
27
28/// `PHILOX_M4x32_0` and `PHILOX_M4x32_1`.
29pub(super) const MULTIPLIERS: [u32; 2] = [0xD251_1F53, 0xCD9E_8D57];
30/// `PHILOX_W32_0` and `PHILOX_W32_1`: the golden ratio and √3 − 1 as 32-bit
31/// fractions.
32pub(super) const WEYL: [u32; 2] = [0x9E37_79B9, 0xBB67_AE85];
33
34/// The high and low words of `a × b`.
35#[inline(always)]
36fn multiply_high_low(a: u32, b: u32) -> (u32, u32) {
37    let product = u64::from(a) * u64::from(b);
38    ((product >> 32) as u32, product as u32)
39}
40
41/// One round (`_philox4x32round`) of every lane.
42///
43/// Each lane's step is the scalar round's, `(c₀, c₁, c₂, c₃) →
44/// (hi(M₁c₂) ⊕ c₁ ⊕ k₀, lo(M₁c₂), hi(M₀c₀) ⊕ c₃ ⊕ k₁, lo(M₀c₀))`. With
45/// NEON, whole groups of four lanes take the step in vector registers (the
46/// `neon` module, compiled for aarch64 only) and any remaining lanes take it
47/// one at a time; elsewhere every lane does. Integer arithmetic is exact, so
48/// the two give the same words.
49#[inline(always)]
50fn round<const LANES: usize>(counters: &Lanes<LANES>, key: Key) -> Lanes<LANES> {
51    let mut next = [[0; LANES]; 4];
52    #[cfg(all(target_arch = "aarch64", target_feature = "neon"))]
53    let first_single = neon::round_fours(counters, key, &mut next);
54    #[cfg(not(all(target_arch = "aarch64", target_feature = "neon")))]
55    let first_single = 0;
56    for lane in first_single..LANES {
57        round_lane(counters, key, lane, &mut next);
58    }
59    next
60}
61
62/// One round of lane `lane`, into `next`.
63#[inline(always)]
64fn round_lane<const LANES: usize>(
65    counters: &Lanes<LANES>,
66    key: Key,
67    lane: usize,
68    next: &mut Lanes<LANES>,
69) {
70    let (high0, low0) = multiply_high_low(MULTIPLIERS[0], counters[0][lane]);
71    let (high1, low1) = multiply_high_low(MULTIPLIERS[1], counters[2][lane]);
72    next[0][lane] = high1 ^ counters[1][lane] ^ key[0];
73    next[1][lane] = low1;
74    next[2][lane] = high0 ^ counters[3][lane] ^ key[1];
75    next[3][lane] = low0;
76}

78/// The round on NEON, four lanes to a register.
79///
80/// The compiler does not vectorise the portable round's 32×32→64-bit
81/// multiplies (its cost model prices them as 64-bit vector multiplies,
82/// which NEON lacks), so this spells out the instructions: `umull` and
83/// `umull2` form the four 64-bit products of a register, and `uzp2` and
84/// `uzp1` gather their high and low halves. It runs a round of eight lanes
85/// in about 24 instructions, against about 70 one lane at a time.
86#[cfg(all(target_arch = "aarch64", target_feature = "neon"))]
87mod neon {
88    use std::arch::aarch64::{
89        uint32x4_t, vdupq_n_u32, veorq_u32, vget_low_u32, vld1q_u32, vmull_high_u32, vmull_u32,
90        vreinterpretq_u32_u64, vst1q_u32, vuzp1q_u32, vuzp2q_u32,
91    };
92
93    use super::{Key, Lanes, MULTIPLIERS};
94
95    /// One round of lanes `0..4n`, the whole groups of four, into `next`;
96    /// returns `4n`, the first lane left for the one-at-a-time step.
97    #[inline(always)]
98    pub(super) fn round_fours<const LANES: usize>(
99        counters: &Lanes<LANES>,
100        key: Key,
101        next: &mut Lanes<LANES>,
102    ) -> usize {
103        let whole = LANES - LANES % 4;
104        for base in (0..whole).step_by(4) {
105            // SAFETY: the intrinsics need NEON, which this module's `cfg`
106            // requires; `base + 4 <= LANES`, so each load reads and each
107            // store writes four words inside one lane array.
108            unsafe {
109                let load = |word: usize| vld1q_u32(counters[word].as_ptr().add(base));
110                let (c0, c1, c2, c3) = (load(0), load(1), load(2), load(3));
111                let (high0, low0) = multiply_high_low(MULTIPLIERS[0], c0);
112                let (high1, low1) = multiply_high_low(MULTIPLIERS[1], c2);
113                let n0 = veorq_u32(veorq_u32(high1, c1), vdupq_n_u32(key[0]));
114                let n2 = veorq_u32(veorq_u32(high0, c3), vdupq_n_u32(key[1]));
115                vst1q_u32(next[0].as_mut_ptr().add(base), n0);
116                vst1q_u32(next[1].as_mut_ptr().add(base), low1);
117                vst1q_u32(next[2].as_mut_ptr().add(base), n2);
118                vst1q_u32(next[3].as_mut_ptr().add(base), low0);
119            }
120        }
121        whole
122    }
123
124    /// The high and low words of `multiplier × c` in each of four lanes.
125    /// Read as 32-bit lanes, a register of two little-endian 64-bit products
126    /// is (low, high, low, high).
127    #[inline(always)]
128    fn multiply_high_low(multiplier: u32, counter: uint32x4_t) -> (uint32x4_t, uint32x4_t) {
129        // SAFETY: the intrinsics need NEON, which this module's `cfg`
130        // requires; they touch registers only.
131        unsafe {
132            let multiplier = vdupq_n_u32(multiplier);
133            let lower =
134                vreinterpretq_u32_u64(vmull_u32(vget_low_u32(counter), vget_low_u32(multiplier)));
135            let upper = vreinterpretq_u32_u64(vmull_high_u32(counter, multiplier));
136            (vuzp2q_u32(lower, upper), vuzp1q_u32(lower, upper))
137        }
138    }
139}

141/// Philox4x32 with `ROUNDS` rounds of each of `LANES` counters: the first
142/// round with `key`, each later one with the key bumped once more by the
143/// Weyl constants. Lane `l` of the result is the scalar generator's output
144/// for lane `l`'s counter.
145#[inline(always)]
146pub(crate) fn philox4x32_lanes<const ROUNDS: usize, const LANES: usize>(
147    counters: Lanes<LANES>,
148    key: Key,
149) -> Lanes<LANES> {
150    let mut counters = counters;
151    let mut key = key;
152    for index in 0..ROUNDS {
153        if index > 0 {
154            key = [key[0].wrapping_add(WEYL[0]), key[1].wrapping_add(WEYL[1])];
155        }
156        counters = round(&counters, key);
157    }
158    counters
159}

161/// Philox4x32-10, Random123's recommended round count, of `LANES` counters.
162#[inline(always)]
163pub(crate) fn philox4x32_10_lanes<const LANES: usize>(
164    counters: Lanes<LANES>,
165    key: Key,
166) -> Lanes<LANES> {
167    philox4x32_lanes::<10, LANES>(counters, key)
168}

170/// Philox4x32-10 of one 128-bit `counter` under the 64-bit `key`: the four
171/// words of one draw.
172///
173/// Philox4x32 (Salmon, Moraes, Dror and Shaw, "Parallel random numbers: as
174/// easy as 1, 2, 3", SC11) is ten rounds, as Random123's `philox4x32_10`,
175/// of `(c₀, c₁, c₂, c₃) → (hi(M₁c₂) ⊕ c₁ ⊕ k₀, lo(M₁c₂), hi(M₀c₀) ⊕ c₃ ⊕ k₁,
176/// lo(M₀c₀))`, with `M₀ = 0xD2511F53` and `M₁ = 0xCD9E8D57`, the key
177/// bumped by the Weyl constants `(0x9E3779B9, 0xBB67AE85)` before every
178/// round after the first. It keeps no state: the words are a pure function
179/// of the counter and the key, matching Random123's known answers.
180///
181/// This is the draw [`CellNoise::add_to_row`](super::CellNoise::add_to_row)
182/// makes for each correlation pair, sixteen channels at a time; one counter
183/// at a time it is the reference other implementations of the same noise
184/// (another batch width, a GPU kernel) are compared with, bit for bit.
185#[must_use]
186pub fn philox4x32_10(counter: [u32; 4], key: [u32; 2]) -> [u32; 4] {
187    let words = philox4x32_10_lanes(counter.map(|word| [word]), key);
188    words.map(|[word]| word)
189}

244    /// `vector`'s counter in lane `lane`, and distinct counters elsewhere.
245    fn lanes_with<const LANES: usize>(vector: &Vector, lane: usize) -> Lanes<LANES> {
246        let mut lanes: Lanes<LANES> =
247            std::array::from_fn(|word| std::array::from_fn(|other| (word * 31 + other) as u32));
248        for (word, value) in vector.counter.iter().enumerate() {
249            lanes[word][lane] = *value;
250        }
251        lanes
252    }
253
254    /// Every vector in every lane of a `LANES`-lane batch.
255    fn assert_known_answers<const LANES: usize>() {
256        for vector in &KAT {
257            for lane in 0..LANES {
258                let counters = lanes_with::<LANES>(vector, lane);
259                let output = match vector.rounds {
260                    7 => philox4x32_lanes::<7, LANES>(counters, vector.key),
261                    10 => philox4x32_10_lanes(counters, vector.key),
262                    other => panic!("no {other}-round vector"),
263                };
264                assert_eq!(
265                    output.map(|words| words[lane]),
266                    vector.expected,
267                    "philox4x32-{} of {:08x?} under {:08x?} in lane {lane} of {LANES}",
268                    vector.rounds,
269                    vector.counter,
270                    vector.key
271                );
272            }
273        }
274    }
