44impl Axis {
45    pub(super) fn parse(value: &str) -> Result<Self, String> {
46        use Axis::*;
47        let normalized = value.to_ascii_lowercase().replace([' ', '-'], "");
48        let axis = match normalized.as_str() {
49            "chan" | "channel" => Channel,
50            "freq" | "frequency" => Frequency,
51            "time" => Time,
52            "interval" | "timeint" | "timeinterval" | "time_interval" => Interval,
53            "scan" => Scan,
54            "field" => Field,
55            "spw" => Window,
56            "row" => Row,
57            "antenna" | "antenna1" | "ant" | "ant1" => Antenna1,
58            "antenna2" | "ant2" => Antenna2,
59            "feed1" => Feed1,
60            "poln" | "polarization" | "corr" | "correlation" => Receptor,
61            "observation" => Observation,
62            "amp" | "amplitude" => Amplitude,
63            "phase" => Phase,
64            "real" => Real,
65            "imag" | "imaginary" => Imaginary,
66            "gainamp" | "gamp" => GainAmplitude,
67            "gainphase" | "gphase" => GainPhase,
68            "gainreal" | "greal" => GainReal,
69            "gainimag" | "gimag" => GainImaginary,
70            "snr" => SignalToNoise,
71            "flag" => Flag,
72            "delay" | "del" => Delay,
73            "rate" | "delayrate" => Rate,
74            "disp" | "dispdelay" => DispersiveDelay,
75            "tsys" => SystemTemperature,
76            "opacity" | "opac" => Opacity,
77            "swp" | "swpower" | "switchedpower" | "spgain" => SwitchedPower,
78            "tec" => Tec,
79            "antennaposition" | "antpos" => AntennaPosition,
80            _ => return Err(format!("unknown calibration plot axis {value:?}")),
81        };
82        Ok(axis)
83    }

125    pub(in crate::ms::msexplore) fn unit(self) -> &'static str {
126        match self {
127            Self::Frequency => "GHz",
128            Self::Time => "MJD seconds",
129            Self::Interval => "s",
130            Self::Phase | Self::GainPhase => "deg",
131            Self::Delay => "ns",
132            Self::Rate => "ps/s",
133            Self::SystemTemperature | Self::SkyTemperature => "K",
134            Self::AtmosphericTransmission => "%",
135            Self::AntennaPosition => "m",
136            _ => "",
137        }
138    }

140    pub(super) fn value(self, metadata: &SampleCoordinates, value: Complex64, snr: f64) -> f64 {
141        use Axis::*;
142        match self {
143            Channel => metadata.channel as f64,
144            Frequency => metadata.frequency_ghz,
145            Time => metadata.time,
146            Interval => metadata.interval,
147            Scan => f64::from(metadata.scan),
148            Field => f64::from(metadata.field),
149            Window => f64::from(metadata.window),
150            Row => metadata.row as f64,
151            Antenna1 => f64::from(metadata.antennas[0]),
152            Antenna2 => f64::from(metadata.antennas[1]),
153            Feed1 => f64::from(metadata.feed1),
154            Receptor => metadata.receptor as f64,
155            Observation => f64::from(metadata.observation),
156            Amplitude | GainAmplitude => value.norm(),
157            Phase | GainPhase => value.arg().to_degrees(),
158            Imaginary | GainImaginary => value.im,
159            SignalToNoise => snr,
160            Real
161            | GainReal
162            | Delay
163            | Rate
164            | DispersiveDelay
165            | SystemTemperature
166            | Opacity
167            | SwitchedPower
168            | Tec
169            | AntennaPosition
170            | Flag
171            | AtmosphericTransmission
172            | SkyTemperature
173            | ImageSideband => value.re,
174        }
175    }

177    pub(super) fn uses_receptor(self) -> bool {
178        use Axis::*;
179        matches!(
180            self,
181            Receptor
182                | Amplitude
183                | Phase
184                | Real
185                | Imaginary
186                | GainAmplitude
187                | GainPhase
188                | GainReal
189                | GainImaginary
190                | SignalToNoise
191                | Flag
192        )
193    }
194
195    pub(super) fn uses_spline_antenna(self) -> bool {
196        self == Self::Antenna1 || (self != Self::Receptor && self.uses_receptor())
197    }
198}
