13/// The front end of the `impbcor` binary.
14pub fn impbcor_front_end()
15-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
16    image_task_front_end("impbcor", &["impbcor"], run)
17}
18
19pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
20    let pbimage = match task.value("pbimage") {
21        Some(ParameterValue::String(name)) if !name.is_empty() => {
22            PbcorBeam::Image(task.input_path("pbimage")?)
23        }
24        Some(ParameterValue::Array(pixels)) if !pixels.is_empty() => {
25            let values = parameter_array_to_casa(pixels).map_err(TaskFailure::execution)?;
26            let ArrayValue::Float64(array) = values else {
27                return Err(TaskFailure::execution(
28                    "pbimage pixels must be a floating-point array",
29                ));
30            };
31            PbcorBeam::Pixels(array)
32        }
33        _ => {
34            return Err(TaskFailure::validation(
35                "pbimage must name an image or supply floating-point beam pixels",
36            )
37            .at("pbimage"));
38        }
39    };
40    let region = parse_region_parameter(task.value("region"), false)?;
41    let mask = task.text("mask")?;
42    // CASA reads stretch only with a mask, and ignores it otherwise.
43    let stretch = mask.is_some() && task.flag("stretch")?;
44    let request = ImpbcorRequest {
45        imagename: task.input_path("imagename")?,
46        pbimage,
47        outfile: task.path("outfile")?,
48        mode: task.required_text("mode")?,
49        cutoff: cutoff(task.value("cutoff"))?,
50        box_selection: task.text("box")?,
51        region,
52        chans: task.text("chans")?,
53        stokes: task.text("stokes")?,
54        mask,
55        stretch,
56        overwrite: task.flag("overwrite")?,
57    };
58    image_task_success(impbcor(&request).map(ImageAnalysisTaskResult::Impbcor))
59}
60
61fn cutoff(value: Option<&ParameterValue>) -> Result<f64, TaskFailure> {
62    let value = match value {
63        Some(ParameterValue::NumpyScalar { value, .. }) => Some(value.as_ref()),
64        value => value,
65    };
66    let invalid = || TaskFailure::validation("cutoff must be numeric").at("cutoff");
67    match value {
68        Some(ParameterValue::Float(number)) => Ok(*number),
69        Some(ParameterValue::Integer(number)) => Ok(*number as f64),
70        Some(ParameterValue::String(text)) => text.parse().map_err(|_| invalid()),
71        _ => Err(invalid()),
72    }
73}
