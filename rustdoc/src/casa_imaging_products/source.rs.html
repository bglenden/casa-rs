21/// Where an analytic primary beam is centred on each image domain: the
22/// pixel position `[x, y]` of every selected field's phase direction.
23///
24/// CASA centres one beam on each field's `PHASE_DIR` and sums them on every
25/// image, outliers included (`SynthesisImagerVi2::makePrimaryBeam`,
26/// `SIMapper::addPB`).
27pub type PrimaryBeamCentres = BTreeMap<ImageDomainRole, Vec<[f64; 2]>>;

29/// Borrowed scientific payloads for producing planned members.
30///
31/// This value borrows one whole completion and its compiled problem, so the
32/// final normal state and model it carries come from the same reconciliation.
33/// Binding them checks once that they have the problem's shape; product
34/// generation then indexes them without checking again.
35#[derive(Clone, Debug)]
36pub struct ContinuumProductInputs<'a> {
37    problem: &'a CompiledProblem,
38    normal_state: &'a FinalNormalState,
39    final_model: &'a ModelGeneration,
40    reconstruction_mask: Option<&'a ReconstructionMask>,
41    domain_reconstruction_masks: Option<&'a ImageDomainReconstructionMasks>,
42    primary_beam_centres: Option<&'a PrimaryBeamCentres>,
43    taylor_primary_beam: Option<&'a [f32]>,
44    taylor_residual_sum_weight: Option<f64>,
45    cube_taylor_primary_beam: Option<&'a [f32]>,
46    vla_selection_hz: Option<f64>,
47    facet_sum_weights: Option<&'a FacetSumWeights>,
48    saved_psf_beams: Option<&'a [RestoringBeam]>,
49    saved_mosaic_beam: Option<&'a crate::SavedMosaicBeam>,
50}

52/// Each faceted image domain's facets' own sums of weights, by domain
53/// ordinal: per facet in CASA's order (facet `row·N + column`, facet 0 at
54/// the bottom left), per polarization. CASA's `.sumwt` of a faceted image
55/// holds each facet store's own sum (`SIImageStore::getSubImageStore`).
56pub type FacetSumWeights = std::collections::BTreeMap<usize, Vec<Vec<f64>>>;

58impl<'a> ContinuumProductInputs<'a> {
59    /// Bind the completion of a run of `problem`.
60    ///
61    /// # Panics
62    ///
63    /// If the completion does not have `problem`'s image domains, their
64    /// shapes, planes and Taylor terms: it came from another problem's run.
65    pub fn from_major_cycle(problem: &'a CompiledProblem, join: &'a MajorCycleCompletion) -> Self {
66        let normal = join.normal_state();
67        let model = join.final_model().shape();
68        let domains = problem.geometry().domains();
69        assert!(
70            normal.domain_count() == domains.len()
71                && model.domains().len() == domains.len()
72                && domains.iter().all(|domain| {
73                    let pixels = domain.shape().pixels();
74                    model
75                        .domain_roles()
76                        .iter()
77                        .position(|role| role == domain.role())
78                        .is_some_and(|ordinal| {
79                            model.domains()[ordinal].pixels() == pixels
80                                && normal.domain_shape(ordinal) == Some(pixels)
81                        })
82                }),
83            "a run's completion has its problem's image domains and their shapes"
84        );
85        match normal.basis() {
86            NormalBasis::Taylor => {
87                let ReconstructionBasis::Taylor { terms } = problem.reconstruction().basis() else {
88                    panic!("a Taylor normal state comes from a Taylor-basis problem");
89                };
90                assert!(
91                    terms >= 1
92                        && normal.coefficient_term_count() == terms
93                        && normal.normal_moment_count() == 2 * terms - 1
94                        && model.coefficients() == terms
95                        && model.polarizations() == normal.polarization_count()
96                        && domains.len() == 1,
97                    "a Taylor completion has its problem's terms, 2·terms − 1 moments, \
98                     polarizations and one domain"
99                );
100            }
101            NormalBasis::Constant | NormalBasis::ChannelLocal => {
102                let slab = normal.slab();
103                assert!(
104                    slab.core_range() == (0..slab.total_channels())
105                        && normal.channel_count() == slab.total_channels()
106                        && model.coefficients() == normal.channel_count()
107                        && model.polarizations() == normal.polarization_count(),
108                    "a run's final completion covers every output channel and polarization"
109                );
110            }
111        }
112        Self {
113            problem,
114            normal_state: normal,
115            final_model: join.final_model(),
116            reconstruction_mask: None,
117            domain_reconstruction_masks: None,
118            primary_beam_centres: None,
119            taylor_primary_beam: None,
120            taylor_residual_sum_weight: None,
121            cube_taylor_primary_beam: None,
122            vla_selection_hz: None,
123            facet_sum_weights: None,
124            saved_psf_beams: None,
125            saved_mosaic_beam: None,
126        }
127    }

129    /// Bind a precomputed primary beam for a Taylor family, in x-major order.
130    ///
131    /// This is the already-normalized beam for the one Taylor image domain,
132    /// such as a sumweight-weighted mean of the published cube-channel PB
133    /// planes. CASA MVC retains its values below the PB cutoff while masking
134    /// those pixels. The caller retains and accounts for the borrowed plane
135    /// until generation ends.
136    ///
137    /// # Errors
138    ///
139    /// The completion is not a Taylor family, the plane has the wrong size,
140    /// or it contains a negative or non-finite value.
141    pub fn with_taylor_primary_beam(
142        mut self,
143        beam: &'a [f32],
144    ) -> Result<Self, crate::ProductsError> {
145        if self.normal_state.basis() != NormalBasis::Taylor {
146            return Err(crate::ProductsError::InvalidTaylorPrimaryBeam);
147        }
148        let [width, height] = self.normal_state.shape();
149        let expected =
150            width
151                .checked_mul(height)
152                .ok_or(crate::ProductsError::ResourceDemandOverflow(
153                    "Taylor primary-beam plane",
154                ))?;
155        if beam.len() != expected {
156            return Err(crate::ProductsError::PayloadLengthMismatch {
157                expected,
158                actual: beam.len(),
159            });
160        }
161        if beam.iter().any(|value| !value.is_finite() || *value < 0.0) {
162            return Err(crate::ProductsError::InvalidTaylorPrimaryBeam);
163        }
164        self.taylor_primary_beam = Some(beam);
165        Ok(self)
166    }
167
168    /// Bind MVC's weighted Taylor beam for adjusting a cube's residual.
169    ///
170    /// The caller owns and admits the x-major plane until generation ends.
171    /// Its channel PB and the PB cutoff adjust only the cube residual, as
172    /// `SynthesisUtilMethods::removeFreqDepPB` does before Taylor reduction.
173    ///
174    /// # Errors
175    ///
176    /// The completion is not a single-domain, single-polarization cube, or
177    /// the beam's size or values are invalid.
178    pub fn with_cube_taylor_primary_beam(
179        mut self,
180        beam: &'a [f32],
181    ) -> Result<Self, crate::ProductsError> {
182        if !supports_cube_taylor_beam(
183            self.normal_state.basis(),
184            self.normal_state.domain_count(),
185            self.normal_state.polarization_count(),
186        ) {
187            return Err(crate::ProductsError::InvalidCubeTaylorPrimaryBeam);
188        }
189        let [width, height] = self.normal_state.shape();
190        let expected =
191            width
192                .checked_mul(height)
193                .ok_or(crate::ProductsError::ResourceDemandOverflow(
194                    "cube Taylor primary-beam plane",
195                ))?;
196        if beam.len() != expected {
197            return Err(crate::ProductsError::PayloadLengthMismatch {
198                expected,
199                actual: beam.len(),
200            });
201        }
202        if beam.iter().any(|value| !value.is_finite() || *value < 0.0) {
203            return Err(crate::ProductsError::InvalidCubeTaylorPrimaryBeam);
204        }
205        self.cube_taylor_primary_beam = Some(beam);
206        Ok(self)
207    }

213    /// Bind the divisor for an already-normalized Taylor residual family.
214    ///
215    /// CASA MVC's cube-to-Taylor conversion produces normalized residual
216    /// planes while retaining its distinct published sum-weight moments.
217    /// Supplying `1.0` preserves those Float pixels without multiplying and
218    /// dividing them; published sum weights remain unchanged.
219    ///
220    /// # Errors
221    ///
222    /// The completion is not Taylor-based, or the divisor is nonpositive or
223    /// nonfinite.
224    pub fn with_taylor_residual_sum_weight(
225        mut self,
226        divisor: f64,
227    ) -> Result<Self, crate::ProductsError> {
228        if self.normal_state.basis() != NormalBasis::Taylor
229            || !divisor.is_finite()
230            || divisor <= 0.0
231        {
232            return Err(crate::ProductsError::InvalidTaylorResidualSumWeight);
233        }
234        self.taylor_residual_sum_weight = Some(divisor);
235        Ok(self)
236    }

239        self.taylor_residual_sum_weight
240    }

247    /// Bind the complete PSF beam grid retained from an earlier run.
248    ///
249    /// # Errors
250    /// The grid does not cover the main image's channels and polarizations.
251    pub fn with_saved_psf_beams(
252        mut self,
253        beams: &'a [RestoringBeam],
254    ) -> Result<Self, crate::ProductsError> {
255        let channels = self.normal_state.channel_count();
256        let expected = channels * self.normal_state.polarization_count();
257        if beams.len() != expected {
258            return Err(crate::ProductsError::BeamGrid {
259                beams: beams.len(),
260                channels,
261            });
262        }
263        self.saved_psf_beams = Some(beams);
264        Ok(self)
265    }

267    /// Bind an earlier MFS mosaic's PB and pixel mask for the products of a
268    /// continuation, polarization-major with x-major image planes.
269    ///
270    /// # Errors
271    /// The completion is outside the single-term mosaic layout, or the
272    /// saved PB does not match its grid.
273    pub fn with_saved_mosaic_beam(
274        mut self,
275        beam: &'a crate::SavedMosaicBeam,
276    ) -> Result<Self, crate::ProductsError> {
277        let normal = self.normal_state;
278        let [width, height] = normal.shape();
279        let expected = [
280            width,
281            height,
282            normal.polarization_count(),
283            normal.channel_count(),
284        ];
285        if normal.basis() == NormalBasis::Taylor || normal.domain_count() != 1 {
286            return Err(crate::ProductsError::UnsupportedProblem);
287        }
288        if beam.shape() != expected {
289            return Err(crate::ProductsError::SavedMosaicBeamShape {
290                saved: beam.shape(),
291                expected,
292            });
293        }
294        self.saved_mosaic_beam = Some(beam);
295        Ok(self)
296    }

307    /// Bind each faceted domain's facets' own sums of weights, which its
308    /// `.sumwt` publishes one per facet pixel; an unfaceted domain publishes
309    /// its normal state's sum.
310    ///
311    /// # Panics
312    ///
313    /// If an entry is not a faceted domain of the problem with one sum per
314    /// facet and polarization.
315    #[must_use]
316    pub fn with_facet_sum_weights(mut self, sums: &'a FacetSumWeights) -> Self {
317        let domains = self.problem.geometry().domains();
318        let polarizations = self.normal_state.polarization_count();
319        assert!(
320            sums.iter().all(|(ordinal, facets)| {
321                domains.get(*ordinal).is_some_and(|domain| {
322                    domain.facets().len() > 1
323                        && facets.len() == domain.facets().len()
324                        && facets.iter().all(|facet| facet.len() == polarizations)
325                })
326            }),
327            "facet sums of weights cover faceted domains, one per facet and polarization"
328        );
329        self.facet_sum_weights = Some(sums);
330        self
331    }

333    /// Domain `ordinal`'s facets' own sums of weights when it is faceted.
334    #[must_use]
335    pub fn facet_sum_weights(&self, ordinal: usize) -> Option<&'a [Vec<f64>]> {
336        self.facet_sum_weights
337            .and_then(|sums| sums.get(&ordinal))
338            .map(Vec::as_slice)
339    }

341    /// Whether `mask` covers every plane of `domain`: its grid, and each of
342    /// the state's channels and polarizations (a Taylor family is one
343    /// channel).
344    fn covers(&self, mask: &ReconstructionMask, domain: &CompiledImageDomain) -> bool {
345        let normal = self.normal_state;
346        let channels = if normal.basis() == NormalBasis::Taylor {
347            1
348        } else {
349            normal.channel_count()
350        };
351        mask.shape() == domain.shape().pixels()
352            && mask.coordinate() == domain.direction()
353            && mask.planes().covers(channels, normal.polarization_count())
354    }

356    /// Bind the exact reconstruction mask used by the final bounded solve.
357    ///
358    /// # Panics
359    ///
360    /// If the mask does not cover every plane of the main domain.
361    #[must_use]
362    pub fn with_reconstruction_mask(mut self, mask: &'a ReconstructionMask) -> Self {
363        let main = self.domain(&ImageDomainRole::Main);
364        assert!(
365            self.covers(mask, main),
366            "the run's reconstruction mask covers its main domain"
367        );
368        self.reconstruction_mask = Some(mask);
369        self.domain_reconstruction_masks = None;
370        self
371    }
372
373    /// Bind the exact support of every plane of every canonical image
374    /// domain.
375    ///
376    /// # Panics
377    ///
378    /// If the masks are not one per domain, each covering its domain's
379    /// planes.
380    #[must_use]
381    pub fn with_domain_reconstruction_masks(
382        mut self,
383        masks: &'a ImageDomainReconstructionMasks,
384    ) -> Self {
385        let domains = self.problem.geometry().domains();
386        assert!(
387            masks.len() == domains.len()
388                && masks
389                    .iter()
390                    .zip(domains)
391                    .all(|(mask, domain)| self.covers(mask, domain)),
392            "the run's reconstruction masks cover its domains, one each"
393        );
394        self.reconstruction_mask = None;
395        self.domain_reconstruction_masks = Some(masks);
396        self
397    }

399    /// Centre an analytic primary beam on the selected fields' pointings.
400    /// Without them it is centred on each domain's reference pixel. A
401    /// domain with no pointing has a zero PB.
402    ///
403    /// # Panics
404    ///
405    /// If a domain of the problem has no entry.
406    #[must_use]
407    pub fn with_primary_beam_centres(mut self, centres: &'a PrimaryBeamCentres) -> Self {
408        assert!(
409            self.problem
410                .geometry()
411                .domains()
412                .iter()
413                .all(|domain| centres.contains_key(domain.role())),
414            "the pointings cover every image domain"
415        );
416        self.primary_beam_centres = Some(centres);
417        self
418    }

423    /// # Panics

433    /// The run's analytic primary beam of `model`, centred on the bound
434    /// pointings; `None` for the mosaic's, which is each plane's weight
435    /// image. Planning checked its frequencies
436    /// ([`crate::ContinuumProductControls::validate_for_problem`]).
437    pub(crate) fn analytic_primary_beam(
438        &self,
439        model: AnalyticPrimaryBeamModel,
440    ) -> Option<AnalyticPrimaryBeam<'a>> {
441        AnalyticPrimaryBeam::bind(self.problem, model, self.primary_beam_centres).map(|beam| {
442            self.vla_selection_hz.map_or(beam, |frequency| {
443                beam.with_vla_selection_frequency(frequency)
444            })
445        })
446    }

511/// The current MVC conversion owns one common beam for one domain and Stokes
512/// plane; accepting other shapes would silently reuse it for unrelated planes.
513pub(crate) fn supports_cube_taylor_beam(
514    basis: NormalBasis,
515    domains: usize,
516    polarizations: usize,
517) -> bool {
518    basis == NormalBasis::ChannelLocal && domains == 1 && polarizations == 1
519}
