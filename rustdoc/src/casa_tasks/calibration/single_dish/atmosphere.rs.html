25pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
26    let infile = task.required_file_text("infile")?;
27    let outfile = task.required_file_text("outfile")?;
28    let overwrite = task.flag("overwrite")?;
29    let (input, output) = check_paths(&infile, &outfile, overwrite)?;
30    let atmosphere = task.integer("atmtype")?;
31    if !(1..=5).contains(&atmosphere) {
32        return Err(TaskFailure::execution(format!(
33            "atmtype (={atmosphere}) should be any one of (1, 2, 3, 4, 5)."
34        ))
35        .at("atmtype"));
36    }
37    let humidity = task.number("humidity")?;
38    if humidity != -1. && !(0.0..=100.0).contains(&humidity) {
39        return Err(TaskFailure::execution(format!(
40            "humidity (={humidity}) should be in range 0~100"
41        ))
42        .at("humidity"));
43    }
44    let column = task.required_text("datacolumn")?;
45    if data_column(&column).is_err() {
46        return Err(TaskFailure::execution(format!(
47            "Specified column name ({column}) Unacceptable."
48        ))
49        .at("datacolumn"));
50    }
51
52    let ms = MeasurementSet::open(input).map_err(execution)?;
53    let antenna = selector(task, "antenna")?
54        .map(|text| include_autocorrelations(&ms, text))
55        .transpose()?;
56    let output_spw = task.required_text("outputspw")?;
57    let rows = MainRows::read(&ms).map_err(execution)?;
58    let (config, diagnostics) = configuration(task, &ms, &rows, &output_spw, atmosphere, humidity)?;
59    let selection = MsSelection {
60        spw: Some(output_spw),
61        field: selector(task, "field")?,
62        scan: selector(task, "scan")?,
63        antenna,
64        correlation: selector(task, "correlation")?,
65        timerange: selector(task, "timerange")?,
66        intent: selector(task, "intent")?,
67        observation: selector(task, "observation")?,
68        feed: selector(task, "feed")?,
69        msselect: selector(task, "msselect")?,
70        ..MsSelection::default()
71    };
72    let mut session = SingleDishSession::default();
73    session.open(input);
74    session
75        .set_selection(selection.clone(), false)
76        .map_err(execution)?;
77    diagnostics.start(task);
78    let result = session.atmcor_prepared(
79        &ms,
80        rows,
81        SingleDishAtmosphere {
82            config: &config,
83            datacolumn: &column,
84            outfile: output,
85            overwrite,
86        },
87    );
88    if diagnostics.has_uncorrected_windows() {
89        let written = result.as_ref().map_or_else(
90            |_| selected_windows_after_failure(&ms, selection),
91            |report| report.spectral_window_ids.iter().copied().collect(),
92        );
93        diagnostics.report_uncorrected(&written, task);
94    }
95    TaskSuccess::new(CasaValue::None, &result.map_err(execution)?)
96}

98struct Diagnostics {
99    requested: Vec<i32>,
100    corrected: BTreeSet<i32>,
101    output: BTreeSet<i32>,
102    threads: usize,
103}
104
105impl Diagnostics {
106    fn start(&self, task: &TaskInvocation<'_>) {
107        log_message(
108            tracing::Level::INFO,
109            "SDAtmosphereCorrectionTVI::SDAtmosphereCorrectionTVI",
110            &format!("Setting numThreads_ to {}", self.threads),
111            task,
112        );
113        log_message(
114            tracing::Level::INFO,
115            "SDAtmosphereCorrectionTVI::initializeAtmosphereCorrection",
116            &format!("processspw (input) = {:?}", self.requested),
117            task,
118        );
119    }
120
121    fn has_uncorrected_windows(&self) -> bool {
122        !self.output.is_subset(&self.corrected)
123    }
124
125    fn report_uncorrected(&self, written: &BTreeSet<i32>, task: &TaskInvocation<'_>) {
126        let windows = written
127            .difference(&self.corrected)
128            .copied()
129            .collect::<Vec<_>>();
130        let message = match windows.as_slice() {
131            [] => return,
132            [window] => format!("SPW {window} is output but not corrected"),
133            _ => {
134                let ids = windows.iter().map(i32::to_string).collect::<Vec<_>>();
135                format!("SPWs {} are output but not corrected", ids.join(" "))
136            }
137        };
138        log_message(
139            tracing::Level::WARN,
140            "SDAtmosphereCorrectionTVI::initializeAtmosphereCorrection",
141            &message,
142            task,
143        );
144    }
145}

147fn log_message(level: tracing::Level, location: &str, message: &str, task: &TaskInvocation<'_>) {
148    if task.default_stderr_logs() {
149        let unix_seconds = std::time::SystemTime::now()
150            .duration_since(std::time::UNIX_EPOCH)
151            .map_or(0.0, |elapsed| elapsed.as_secs_f64());
152        let timestamp = format::format_time(
153            40_587.0 + unix_seconds / 86_400.0,
154            format::FITS | format::USE_SPACE,
155            6,
156        );
157        task.write_default_log_line(&format!("{timestamp}\t{level}\t{location}\t{message}"));
158    } else if level == tracing::Level::WARN {
159        tracing::warn!(casa.location = location, "{message}");
160    } else {
161        tracing::info!(casa.location = location, "{message}");
162    }
163}

165fn selected_windows_after_failure(
166    ms: &MeasurementSet,
167    mut selection: MsSelection,
168) -> BTreeSet<i32> {
169    // A successful transform already reports the original window IDs it
170    // wrote. Only resolve metadata again when there is no transform report.
171    if selection.antenna.as_deref().is_none_or(str::is_empty) {
172        selection.antenna = Some("*&&&".into());
173    }
174    MsSelectionIoBudget::from_system_memory(1, std::mem::size_of::<ResolvedMsSelectionRow>(), None)
175        .ok()
176        .and_then(|budget| ms.resolve_casa_selection(&selection, budget).ok())
177        .map(|resolved| {
178            resolved
179                .selected_rows
180                .into_iter()
181                .map(|row| row.spectral_window_id)
182                .collect()
183        })
184        .unwrap_or_default()
185}

187fn include_autocorrelations(ms: &MeasurementSet, text: String) -> Result<String, TaskFailure> {
188    let has_auto = |expression: &str| {
189        let selected = ms.resolve_antenna_selector(expression).map_err(execution)?;
190        Ok::<_, TaskFailure>(
191            (0..selected.antenna_count()).any(|index| selected.holds(index, index)),
192        )
193    };
194    if text.is_empty() || has_auto(&text)? {
195        return Ok(text);
196    }
197    let rewritten = text
198        .split(';')
199        .flat_map(|clause| {
200            if clause.is_empty() || clause.ends_with("&&") {
201                return vec![clause.to_string()];
202            }
203            clause
204                .split('&')
205                .filter(|part| !part.is_empty())
206                .map(|part| format!("{part}&&&"))
207                .collect()
208        })
209        .collect::<Vec<_>>()
210        .join(";");
211    if !has_auto(&rewritten)? {
212        return Err(TaskFailure::execution(
213            "Cannot handle antenna selection properly. Abort.",
214        ));
215    }
216    Ok(rewritten)
217}

219fn check_paths<'a>(
220    infile: &'a str,
221    outfile: &'a str,
222    overwrite: bool,
223) -> Result<(&'a Path, &'a Path), TaskFailure> {
224    if infile.is_empty() {
225        return Err(TaskFailure::execution("infile MUST BE specified.").at("infile"));
226    }
227    if outfile.is_empty() {
228        return Err(TaskFailure::execution("outfile MUST BE specified.").at("outfile"));
229    }
230    let input = Path::new(infile);
231    let output = Path::new(outfile);
232    if casa_calibration::paths::overlap(input, output).map_err(execution)? {
233        return Err(TaskFailure::execution(
234            "You are attempting to write the output on your input file.",
235        )
236        .at("outfile"));
237    }
238    if !input.exists() {
239        return Err(TaskFailure::execution("Specified infile does not exist.").at("infile"));
240    }
241    if output.exists() && !overwrite {
242        return Err(TaskFailure::execution("Specified outfile already exist.").at("outfile"));
243    }
244    Ok((input, output))
245}

247fn configuration(
248    task: &TaskInvocation<'_>,
249    ms: &MeasurementSet,
250    raw: &MainRows,
251    output: &str,
252    atmosphere: i64,
253    humidity: f64,
254) -> Result<(RecordValue, Diagnostics), TaskFailure> {
255    let reference = inputs::default_antenna(ms, raw).map_err(execution)?;
256    let site = inputs::site_altitude(ms, reference).map_err(execution)?;
257    let detailed = task.flag("atmdetail")?;
258    let processes = task.required_text("spw")?;
259    let selected = resolve_windows(ms, &processes, "spw")?;
260    let written = resolve_windows(ms, output, "outputspw")?;
261    let output = written
262        .into_iter()
263        .filter(|window| raw.windows.contains(window))
264        .collect::<BTreeSet<_>>();
265    let windows = output
266        .iter()
267        .copied()
268        .filter(|window| selected.contains(window))
269        .collect::<Vec<_>>();
270    let corrected = windows
271        .iter()
272        .copied()
273        .filter(|window| raw.science.contains(window))
274        .collect();
275    let length = ms.spectral_window().map_err(execution)?.row_count();
276    let gain = gain_factors(task.value("gainfactor"), length)?;
277    let mut record = RecordValue::default();
278    record.upsert(
279        "processspw",
280        Value::Array(ArrayValue::Int32(
281            Array1::from_vec(windows.clone()).into_dyn(),
282        )),
283    );
284    record.upsert("gainfactor", doubles(gain));
285    record.upsert("refant", Value::Scalar(ScalarValue::Int32(reference)));
286    record.upsert(
287        "atmType",
288        Value::Scalar(ScalarValue::Int32(atmosphere as i32)),
289    );
290    let threads = std::env::var("OMP_NUM_THREADS")
291        .ok()
292        .map(|text| text.trim().parse::<i32>())
293        .transpose()
294        .map_err(execution)?
295        .unwrap_or(-1);
296    record.upsert("nthreads", Value::Scalar(ScalarValue::Int32(threads)));
297    record.upsert("maxAltitude", scalar(120.));
298    record.upsert(
299        "lapseRate",
300        scalar(quantity(task, "dtem_dh", "K/km")?.unwrap_or(-5.6)),
301    );
302    record.upsert(
303        "scaleHeight",
304        scalar(quantity(task, "h0", "km")?.unwrap_or(2.)),
305    );
306    record.upsert(
307        "pressureStep",
308        scalar(quantity(task, "dp", "mbar")?.unwrap_or(10.)),
309    );
310    let factor = task.number("dpm")?;
311    record.upsert(
312        "pressureStepFactor",
313        scalar(if factor == -1. { 1.2 } else { factor }),
314    );
315    let altitude = if detailed {
316        quantity(task, "altitude", "m")?.unwrap_or(site)
317    } else {
318        site
319    };
320    record.upsert("siteAltitude", scalar(altitude));
321    for (name, unit) in [("temperature", "K"), ("pressure", "mbar"), ("pwv", "mm")] {
322        if let Some(value) = quantity(task, name, unit)?
323            && detailed
324        {
325            record.upsert(name, scalar(value));
326        }
327    }
328    if detailed && humidity != -1. {
329        record.upsert("humidity", scalar(humidity));
330    }
331    for (parameter, field, unit) in [
332        ("layerboundaries", "layerBoundaries", "m"),
333        ("layertemperature", "layerTemperatures", "K"),
334    ] {
335        if let Some(values) = layers(task, parameter, unit)?
336            && detailed
337        {
338            record.upsert(field, doubles(values));
339        }
340    }
341    Ok((
342        record,
343        Diagnostics {
344            requested: windows,
345            corrected,
346            output,
347            threads: thread_count(Some(f64::from(threads))),
348        },
349    ))
350}

352fn resolve_windows(
353    ms: &MeasurementSet,
354    expression: &str,
355    parameter: &str,
356) -> Result<Vec<i32>, TaskFailure> {
357    let expression = if expression.is_empty() {
358        "*"
359    } else {
360        expression
361    };
362    ms.resolve_spw_selector(expression)
363        .map(|selection| {
364            selection
365                .spw_ids()
366                .iter()
367                .map(|&window| window as i32)
368                .collect()
369        })
370        .map_err(|error| execution(error).at(parameter))
371}

373fn gain_factors(value: Option<&ParameterValue>, windows: usize) -> Result<Vec<f64>, TaskFailure> {
374    let value = value.map(scalar_value);
375    let mut factors = vec![1.; windows];
376    match value {
377        Some(ParameterValue::Table(entries)) => {
378            for (key, value) in entries {
379                let window = key
380                    .parse::<i32>()
381                    .map_err(|error| execution(error).at("gainfactor"))?;
382                if let Some(factor) = factors.get_mut(window as usize) {
383                    *factor = numeric(value).ok_or_else(|| {
384                        TaskFailure::execution("Non-numeric gain factor.").at("gainfactor")
385                    })?;
386                }
387            }
388        }
389        Some(ParameterValue::String(path)) => {
390            for (window, factor) in calibration_gains(Path::new(path))? {
391                if let Some(value) = factors.get_mut(window as usize) {
392                    *value = factor;
393                }
394            }
395        }
396        Some(value) => factors.fill(
397            numeric(value)
398                .ok_or_else(|| TaskFailure::execution("Invalid gainfactor.").at("gainfactor"))?,
399        ),
400        None => {}
401    }
402    Ok(factors)
403}

405fn calibration_gains(path: &Path) -> Result<BTreeMap<i32, f64>, TaskFailure> {
406    if !path.exists() {
407        return Err(
408            TaskFailure::execution(format!("\"{}\" should exist.", path.display()))
409                .at("gainfactor"),
410        );
411    }
412    let table = Table::open(TableOptions::new(path)).map_err(execution)?;
413    let (column, spw) = ["FPARAM", "CPARAM"]
414        .iter()
415        .find_map(|&name| {
416            table
417                .column_accessor(name)
418                .ok()
419                .zip(table.column_accessor("SPECTRAL_WINDOW_ID").ok())
420        })
421        .ok_or_else(|| {
422            TaskFailure::execution(format!("{} is not a caltable", path.display())).at("gainfactor")
423        })?;
424    let mut sums = BTreeMap::<i32, (f64, usize)>::new();
425    for first in (0..table.row_count()).step_by(512) {
426        let rows = (first..(first + 512).min(table.row_count())).collect::<Vec<_>>();
427        let values = column
428            .array_cells_owned_uncached(&rows)
429            .map_err(execution)?;
430        let windows = spw.scalar_cells_owned_for_rows(&rows).map_err(execution)?;
431        for (values, window) in values.into_iter().zip(windows) {
432            let Some(ScalarValue::Int32(window)) = window else {
433                return Err(TaskFailure::execution("Invalid calibration window."));
434            };
435            let (sum, count) = sums.entry(window).or_default();
436            match values {
437                Some(ArrayValue::Float32(values)) => {
438                    *sum += reciprocal_squares(values.iter().copied());
439                    *count += values.len();
440                }
441                Some(ArrayValue::Complex32(values)) => {
442                    *sum += reciprocal_squares(values.iter().map(|value| value.re));
443                    *count += values.len();
444                }
445                _ => {
446                    return Err(
447                        TaskFailure::execution("Invalid calibration gain cells.").at("gainfactor")
448                    );
449                }
450            }
451        }
452    }
453    Ok(sums
454        .into_iter()
455        .map(|(window, (sum, count))| (window, sum / count as f64))
456        .collect())
457}
458
459fn reciprocal_squares(values: impl Iterator<Item = f32>) -> f64 {
460    values.map(|value| f64::from((value * value).recip())).sum()
461}

463fn quantity(task: &TaskInvocation<'_>, name: &str, unit: &str) -> Result<Option<f64>, TaskFailure> {
464    match task.value(name).map(scalar_value) {
465        None => Ok(None),
466        Some(ParameterValue::String(text)) if text.is_empty() => Ok(None),
467        Some(value) => parse_quantity(value, unit)
468            .map(Some)
469            .map_err(|error| execution(error).at(name)),
470    }
471}

473fn layers(
474    task: &TaskInvocation<'_>,
475    name: &str,
476    unit: &str,
477) -> Result<Option<Vec<f64>>, TaskFailure> {
478    let numbers = match task.value(name).map(scalar_value) {
479        None => return Ok(None),
480        Some(ParameterValue::String(text)) if text.is_empty() => return Ok(None),
481        Some(ParameterValue::String(text)) => text
482            .split(',')
483            .map(|item| ParameterValue::String(item.into()))
484            .collect::<Vec<_>>(),
485        Some(ParameterValue::Array(items)) => items.to_vec(),
486        Some(ParameterValue::EmptyArray { .. }) => Vec::new(),
487        Some(_) => {
488            return Err(TaskFailure::execution(
489                "user_param for parse_atm_list_params should be either list or str.",
490            )
491            .at(name));
492        }
493    };
494    numbers
495        .iter()
496        .map(|value| parse_quantity(value, unit))
497        .collect::<Result<Vec<_>, _>>()
498        .map(Some)
499        .map_err(|error| execution(error).at(name))
500}

502fn parse_quantity(value: &ParameterValue, unit: &str) -> Result<f64, String> {
503    if let Some(value) = numeric(value) {
504        return Ok(value);
505    }
506    let ParameterValue::String(text) = scalar_value(value) else {
507        return Err("The atmosphere parameter must be a quantity.".into());
508    };
509    let quantity = Quantity::read(text).map_err(|error| error.to_string())?;
510    if quantity.unit() == Quantity::dimensionless(0.).unit() {
511        return Ok(quantity.value());
512    }
513    let expected = Unit::new(unit).map_err(|error| error.to_string())?;
514    quantity.get_value_in(&expected).map_err(|_| {
515        format!("User input \"{text}\" should have the unit compatible with \"{unit}\"")
516    })
517}

519fn scalar_value(value: &ParameterValue) -> &ParameterValue {
520    match value {
521        ParameterValue::NumpyScalar { value, .. } | ParameterValue::ScalarArray(value) => {
522            scalar_value(value)
523        }
524        value => value,
525    }
526}
527
528fn numeric(value: &ParameterValue) -> Option<f64> {
529    match scalar_value(value) {
530        ParameterValue::Float(value) => Some(*value),
531        ParameterValue::Integer(value) => Some(*value as f64),
532        ParameterValue::Bool(value) => Some(f64::from(u8::from(*value))),
533        _ => None,
534    }
535}
536
537fn scalar(value: f64) -> Value {
538    Value::Scalar(ScalarValue::Float64(value))
539}
540
541fn doubles(values: Vec<f64>) -> Value {
542    Value::Array(ArrayValue::Float64(Array1::from_vec(values).into_dyn()))
543}
