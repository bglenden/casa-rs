11pub(crate) fn next_prefix(prefix: &Path) -> Result<PathBuf, PrepareError> {
12    // Match the actual publication directory, including a trailing slash.
13    let mut probe = prefix.as_os_str().to_os_string();
14    probe.push(".model");
15    let probe = PathBuf::from(probe);
16    let directory = probe
17        .parent()
18        .filter(|p| !p.as_os_str().is_empty())
19        .unwrap_or(Path::new("."));
20    let stem = probe
21        .file_name()
22        .map(|s| s.as_encoded_bytes())
23        .and_then(|s| s.strip_suffix(b".model"))
24        .filter(|s| !s.is_empty())
25        .ok_or_else(|| PrepareError::Restart {
26            path: prefix.into(),
27            reason: "restart=False requires a nonempty image basename (#944)".into(),
28        })?;
29    let entries = match fs::read_dir(directory) {
30        Ok(entries) => entries,
31        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(prefix.into()),
32        Err(source) => {
33            return Err(PrepareError::InspectOutputDirectory {
34                path: directory.into(),
35                source,
36            });
37        }
38    };
39    let mut maximum: Option<String> = None;
40    for entry in entries {
41        let entry = entry.map_err(|source| PrepareError::InspectOutputDirectory {
42            path: directory.into(),
43            source,
44        })?;
45        let name = entry.file_name();
46        let name = name.as_encoded_bytes();
47        let Some(id) = product_id(stem, name, &entry.path())? else {
48            continue;
49        };
50        let current = maximum.get_or_insert_with(|| "1".into());
51        if (id.len(), id.as_str()) > (current.len(), current.as_str()) {
52            *current = id;
53        }
54    }
55    let Some(maximum) = maximum else {
56        return Ok(prefix.into());
57    };
58    // Append to the original bytes, preserving relative and absolute directories.
59    let mut selected = prefix.as_os_str().to_os_string();
60    selected.push(format!("_{}", increment(&maximum)));
61    Ok(selected.into())
62}

64fn product_id(stem: &[u8], name: &[u8], path: &Path) -> Result<Option<String>, PrepareError> {
65    let Some((root, extension)) = split_extension(name) else {
66        return Ok(None);
67    };
68    if !matches!(
69        extension,
70        b"image" | b"residual" | b"model" | b"psf" | b"sumwt" | b"tt0"
71    ) {
72        return Ok(None);
73    }
74    if let Some(tail) = name.strip_prefix(stem).and_then(|s| s.strip_prefix(b"_")) {
75        let id = tail
76            .split(|&byte| byte == b'.')
77            .next()
78            .expect("split has first element");
79        // POSIX surrogateescaped bytes are not digits in CASA. Invalid bytes
80        // elsewhere in the filename must not suppress recognized products.
81        let Ok(id) = std::str::from_utf8(id) else {
82            return Ok(None);
83        };
84        if id.is_empty() || !id.chars().all(casa_digit) {
85            return Ok(None);
86        }
87        if !id.is_ascii() {
88            return Err(PrepareError::Restart {
89                path: path.into(),
90                reason: "CASA rejects Unicode digit image suffixes when evaluating their integer ID (#944)".into(),
91            });
92        }
93        // Python eval permits all-zero integer literals but rejects 02, 001, etc.
94        if id.len() > 1 && id.starts_with('0') && id.bytes().any(|b| b != b'0') {
95            return Err(PrepareError::Restart { path: path.into(), reason: "CASA rejects numbered image names with a nonzero leading-zero integer suffix (#944)".into() });
96        }
97        return Ok(Some(if id.bytes().all(|b| b == b'0') {
98            "0".into()
99        } else {
100            id.into()
101        }));
102    }
103    let root = if extension == b"tt0" {
104        split_extension(root).map_or(root, |(root, _)| root)
105    } else {
106        root
107    };
108    Ok((root == stem).then(|| "1".into()))
109}

111fn split_extension(name: &[u8]) -> Option<(&[u8], &[u8])> {
112    let index = name.iter().rposition(|&byte| byte == b'.')?;
113    // Python splitext treats leading dots as part of the basename.
114    if name[..index].iter().all(|&byte| byte == b'.') {
115        return None;
116    }
117    Some((&name[..index], &name[index + 1..]))
118}

120// Exact non-ASCII str.isdigit ranges from CASA 6.7.6.14's Python Unicode 15.0.0.
121// Numeric-but-not-digit text (e.g. 1/2 symbol) must remain ignored, not refused.
122fn casa_digit(character: char) -> bool {
123    character.is_ascii_digit()
124        || matches!(character as u32, 0xB2..=0xB3 | 0xB9 | 0x660..=0x669 | 0x6F0..=0x6F9 | 0x7C0..=0x7C9 | 0x966..=0x96F | 0x9E6..=0x9EF | 0xA66..=0xA6F | 0xAE6..=0xAEF | 0xB66..=0xB6F | 0xBE6..=0xBEF | 0xC66..=0xC6F | 0xCE6..=0xCEF | 0xD66..=0xD6F | 0xDE6..=0xDEF | 0xE50..=0xE59 | 0xED0..=0xED9 | 0xF20..=0xF29 | 0x1040..=0x1049 | 0x1090..=0x1099 | 0x1369..=0x1371 | 0x17E0..=0x17E9 | 0x1810..=0x1819 | 0x1946..=0x194F | 0x19D0..=0x19DA | 0x1A80..=0x1A89 | 0x1A90..=0x1A99 | 0x1B50..=0x1B59 | 0x1BB0..=0x1BB9 | 0x1C40..=0x1C49 | 0x1C50..=0x1C59 | 0x2070 | 0x2074..=0x2079 | 0x2080..=0x2089 | 0x2460..=0x2468 | 0x2474..=0x247C | 0x2488..=0x2490 | 0x24EA | 0x24F5..=0x24FD | 0x24FF | 0x2776..=0x277E | 0x2780..=0x2788 | 0x278A..=0x2792 | 0xA620..=0xA629 | 0xA8D0..=0xA8D9 | 0xA900..=0xA909 | 0xA9D0..=0xA9D9 | 0xA9F0..=0xA9F9 | 0xAA50..=0xAA59 | 0xABF0..=0xABF9 | 0xFF10..=0xFF19 | 0x104A0..=0x104A9 | 0x10A40..=0x10A43 | 0x10D30..=0x10D39 | 0x10E60..=0x10E68 | 0x11052..=0x1105A | 0x11066..=0x1106F | 0x110F0..=0x110F9 | 0x11136..=0x1113F | 0x111D0..=0x111D9 | 0x112F0..=0x112F9 | 0x11450..=0x11459 | 0x114D0..=0x114D9 | 0x11650..=0x11659 | 0x116C0..=0x116C9 | 0x11730..=0x11739 | 0x118E0..=0x118E9 | 0x11950..=0x11959 | 0x11C50..=0x11C59 | 0x11D50..=0x11D59 | 0x11DA0..=0x11DA9 | 0x11F50..=0x11F59 | 0x16A60..=0x16A69 | 0x16AC0..=0x16AC9 | 0x16B50..=0x16B59 | 0x1D7CE..=0x1D7FF | 0x1E140..=0x1E149 | 0x1E2F0..=0x1E2F9 | 0x1E4F0..=0x1E4F9 | 0x1E950..=0x1E959 | 0x1F100..=0x1F10A | 0x1FBF0..=0x1FBF9)
125}

127fn increment(number: &str) -> String {
128    let mut digits = number.as_bytes().to_vec();
129    for digit in digits.iter_mut().rev() {
130        if *digit < b'9' {
131            *digit += 1;
132            return String::from_utf8(digits).expect("ASCII decimal");
133        }
134        *digit = b'0';
135    }
136    digits.insert(0, b'1');
137    String::from_utf8(digits).expect("ASCII decimal")
138}

145    fn matches_pinned_casa_numbering_cases() {
146        for (names, expected) in [
147            (vec![], "image"),
148            (vec!["image.image"], "image_2"),
149            (
150                vec!["image.image", "image_2.psf", "image_7.residual"],
151                "image_8",
152            ),
153            (vec!["image.psf.tt0", "image_3.image.tt0"], "image_4"),
154            (vec!["image.psf.tt1", "image.pb", "image.mask"], "image"),
155            (vec!["image_8.sumwt"], "image_9"),
156            (
157                vec!["imageElse.image", "image_4x.model", "image.summary.json"],
158                "image",
159            ),
160            (vec!["image.image", "image_99.pb"], "image_2"),
161            (vec!["image_6.extra.image"], "image_7"),
162            (
163                vec!["image_99999999999999999999999999999999999999999.model"],
164                "image_100000000000000000000000000000000000000000",
165            ),
166        ] {
167            let directory = tempfile::tempdir().unwrap();
168            for name in &names {
169                fs::write(directory.path().join(name), "preserve").unwrap();
170            }
171            assert_eq!(
172                next_prefix(&directory.path().join("image")).unwrap(),
173                directory.path().join(expected)
174            );
175            for name in names {
176                assert_eq!(
177                    fs::read_to_string(directory.path().join(name)).unwrap(),
178                    "preserve"
179                );
180            }
181        }

241    #[test]
242    fn non_utf8_product_bytes_match_casa_surrogateescape_numbering() {
243        use std::os::unix::ffi::OsStrExt;
244        for (name, id, expected) in [
245            (b"image.\xff.tt0".as_slice(), Some("1"), "image_2"),
246            (b"image_6.\xff.image".as_slice(), Some("6"), "image_7"),
247            (b"image_\xff.image".as_slice(), None, "image"),
248        ] {
249            let directory = tempfile::tempdir().unwrap();
250            let product = directory.path().join(std::ffi::OsStr::from_bytes(name));
251            // Always test parsing, even on APFS which cannot store such names.
252            assert_eq!(product_id(b"image", name, &product).unwrap().as_deref(), id);
253            match fs::write(&product, "preserve") {
254                Ok(()) => {}
255                // macOS APFS rejects invalid filename bytes with EILSEQ (#944).
256                Err(error) if cfg!(target_os = "macos") && error.raw_os_error() == Some(92) => {
257                    continue;
258                }
259                Err(error) => panic!("creating byte-name fixture: {error}"),
260            }
261            assert_eq!(
262                next_prefix(&directory.path().join("image")).unwrap(),
263                directory.path().join(expected)
264            );
265            assert_eq!(fs::read_to_string(product).unwrap(), "preserve");
266        }
