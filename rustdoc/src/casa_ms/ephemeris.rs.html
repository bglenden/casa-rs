22/// The bytes a phase-centre ephemeris may hold once read: what the run
23/// admitted for it.
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub struct EphemerisBudget {
26    available_bytes: usize,
27}

29impl EphemerisBudget {

33        Self { available_bytes }

86/// The moving target a phase centre names (`TRACKFIELD`, a body, or an
87/// ephemeris table), its table rows read once within an admitted budget.
88/// Evaluation interpolates only the epoch asked for, so no sample-sized
89/// direction or velocity array is held.
90#[derive(Debug, Clone)]
91pub struct PhaseCentreEphemeris {
92    kind: EphemerisKind,
93}

95impl PhaseCentreEphemeris {
96    /// Bind a named CASA/casacore moving target, evaluated with the bound
97    /// Measures provider.
98    pub fn named(
99        target: impl AsRef<str>,
100        budget: EphemerisBudget,
101    ) -> Result<Self, PhaseCentreEphemerisError> {
102        let target = target.as_ref();
103        if target.trim().is_empty() {
104            return Err(PhaseCentreEphemerisError::InvalidTarget);
105        }
106        let retained_bytes = selected_ephemeris_allocation_bytes()?
107            .checked_add(target.len())
108            .ok_or(PhaseCentreEphemerisError::ByteOverflow)?;
109        admit_reference_data(budget, retained_bytes)?;
110        Ok(Self {
111            kind: EphemerisKind::Named {
112                target: target.into(),
113            },
114        })
115    }
116
117    /// Bind one external CASA ephemeris table.
118    pub fn external(
119        path: impl AsRef<Path>,
120        budget: EphemerisBudget,
121    ) -> Result<Self, PhaseCentreEphemerisError> {
122        let preflight = EphemerisSeriesPreflight::open(path.as_ref())?;
123        let retained_bytes = selected_ephemeris_allocation_bytes()?
124            .checked_add(preflight.retained_bytes)
125            .ok_or(PhaseCentreEphemerisError::ByteOverflow)?;
126        admit_reference_data(budget, retained_bytes)?;
127        let series = Arc::new(preflight.load()?);
128        Ok(Self {
129            kind: EphemerisKind::External { series },
130        })
131    }
132
133    /// Bind every selected FIELD-linked ephemeris.
134    pub fn tracked_fields(
135        measurement_set: &MeasurementSet,
136        field_ids: impl IntoIterator<Item = usize>,
137        budget: EphemerisBudget,
138    ) -> Result<Self, PhaseCentreEphemerisError> {
139        let ephemeris_ids = field_ephemeris_ids(measurement_set)?;
140        let root = measurement_set
141            .path()
142            .ok_or(PhaseCentreEphemerisError::UnbackedMeasurementSet)?;
143        let mut requested = Vec::new();
144        let mut selected = BTreeSet::new();
145        for field_id in field_ids {
146            if !selected.insert(field_id) {
147                continue;
148            }
149            let ephemeris_id = ephemeris_ids
150                .get(field_id)
151                .copied()
152                .flatten()
153                .ok_or(PhaseCentreEphemerisError::MissingTrackedField { field_id })?;
154            let path = find_field_ephemeris(root, ephemeris_id)?;
155            requested.push((field_id, path));
156        }
157        if requested.is_empty() {
158            return Err(PhaseCentreEphemerisError::EmptyTrackedFieldSet);
159        }
160
161        let mut preflights = BTreeMap::new();
162        for (_, path) in &requested {
163            if !preflights.contains_key(path) {
164                preflights.insert(path.clone(), EphemerisSeriesPreflight::open(path)?);
165            }
166        }
167        let field_binding_bytes = requested
168            .len()
169            .checked_mul(size_of::<FieldEphemerisBinding>())
170            .ok_or(PhaseCentreEphemerisError::ByteOverflow)?;
171        let retained_bytes = preflights
172            .values()
173            .try_fold(selected_ephemeris_allocation_bytes()?, |bytes, series| {
174                bytes.checked_add(series.retained_bytes)
175            })
176            .and_then(|bytes| bytes.checked_add(field_binding_bytes))
177            .ok_or(PhaseCentreEphemerisError::ByteOverflow)?;
178        admit_reference_data(budget, retained_bytes)?;
179
180        let unique = preflights
181            .into_iter()
182            .map(|(path, preflight)| preflight.load().map(|series| (path, Arc::new(series))))
183            .collect::<Result<BTreeMap<_, _>, _>>()?;
184        let mut bindings = requested
185            .into_iter()
186            .map(|(field_id, path)| FieldEphemerisBinding {
187                field_id,
188                series: Arc::clone(
189                    unique
190                        .get(&path)
191                        .expect("every requested ephemeris path was preflighted"),
192                ),
193            })
194            .collect::<Vec<_>>();
195        bindings.sort_by_key(|binding| binding.field_id);
196        Ok(Self {
197            kind: EphemerisKind::TrackedField {
198                fields: bindings.into_boxed_slice(),
199            },
200        })
201    }

203    /// Whether the moving target is an external ephemeris table, rather
204    /// than a body or source Measures names or the fields' own ephemerides.
205    #[must_use]
206    pub fn is_table(&self) -> bool {
207        matches!(self.kind, EphemerisKind::External { .. })

210    /// The external ephemeris table's samples, when the moving target is
211    /// one ([`Self::is_table`]).
212    #[must_use]
213    pub fn external_series(&self) -> Option<&Arc<FieldEphemeris>> {
214        match &self.kind {
215            EphemerisKind::External { series } => Some(series),
216            EphemerisKind::Named { .. } | EphemerisKind::TrackedField { .. } => None,
217        }
218    }

220    pub(crate) fn named_target(&self) -> Option<&str> {
221        match &self.kind {
222            EphemerisKind::Named { target } => Some(target),
223            EphemerisKind::External { .. } | EphemerisKind::TrackedField { .. } => None,
224        }
225    }
226
227    pub(crate) fn sample(
228        &self,
229        field_id: usize,
230        mjd_days: f64,
231    ) -> Result<EvaluatedEphemerisSample, PhaseCentreEphemerisError> {
232        match &self.kind {
233            EphemerisKind::Named { .. } => {
234                Err(PhaseCentreEphemerisError::NamedSampleRequiresMeasures)
235            }
236            EphemerisKind::External { series } => evaluate(series, mjd_days),
237            EphemerisKind::TrackedField { fields } => evaluate(
238                fields
239                    .binary_search_by_key(&field_id, |binding| binding.field_id)
240                    .ok()
241                    .map(|index| &fields[index].series)
242                    .ok_or(PhaseCentreEphemerisError::MissingTrackedField { field_id })?,
243                mjd_days,
244            ),
245        }
246    }

249/// `series` at `mjd_days` ([`FieldEphemeris::sample`]).
250pub(crate) fn evaluate(
251    series: &FieldEphemeris,
252    mjd_days: f64,
253) -> Result<EvaluatedEphemerisSample, PhaseCentreEphemerisError> {
254    let sample =
255        series
256            .sample(mjd_days)
257            .ok_or_else(|| PhaseCentreEphemerisError::OutsideCoverage {
258                target: series.name().to_string(),
259                mjd_days,
260            })?;
261    Ok(EvaluatedEphemerisSample {
262        geocentric_position_metres: sample.position_m,
263        position_reference: series.frame(),
264        radial_velocity_m_per_s: sample.radial_velocity_m_s,
265        velocity_reference: series.velocity_frame(),
266    })
267}

269/// The ephemeris attached to the fields of the MeasurementSet at `root`
270/// whose `EPHEMERIS_ID` is `ephemeris_id`: the one table
271/// `FIELD/EPHEM<id>_*.tab`, as the boundary holds it.
272///
273/// # Errors
274///
275/// When no table or several match, or the table is not one an ephemeris
276/// is read from ([`PhaseCentreEphemeris::external`]'s checks).
277pub(crate) fn attached_ephemeris(
278    root: &Path,
279    ephemeris_id: i32,
280) -> Result<FieldEphemeris, PhaseCentreEphemerisError> {
281    EphemerisSeriesPreflight::open(&find_field_ephemeris(root, ephemeris_id)?)?.load()
282}

284/// The `EPHEMERIS_ID` each FIELD row of `measurement_set` follows: `None`
285/// for none (an id below zero or no column) and for a row whose `NUM_POLY`
286/// is above zero, whose direction casacore's `phaseDirMeas` takes from its
287/// polynomial whatever its `EPHEMERIS_ID`.
288///
289/// The cells are read without filling the table's caches, so a retained
290/// read of a locked MeasurementSet keeps its byte accounting.
291pub(crate) fn field_ephemeris_ids(
292    measurement_set: &MeasurementSet,
293) -> Result<Vec<Option<i32>>, MsError> {
294    let field = measurement_set.field()?;
295    let table = field.table();
296    let rows = (0..table.row_count()).collect::<Vec<_>>();
297    if !table
298        .schema()
299        .is_some_and(|schema| schema.contains_column("EPHEMERIS_ID"))
300    {
301        return Ok(vec![None; rows.len()]);
302    }
303    let ids = table
304        .column_accessor("EPHEMERIS_ID")?
305        .scalar_cells_owned_for_rows(&rows)?;
306    let polynomials = table
307        .column_accessor("NUM_POLY")?
308        .scalar_cells_owned_for_rows(&rows)?;
309    Ok(ids
310        .into_iter()
311        .zip(polynomials)
312        .map(|(id, polynomial)| match (id, polynomial) {
313            (Some(ScalarValue::Int32(id)), Some(ScalarValue::Int32(0))) if id >= 0 => Some(id),
314            _ => None,
315        })
316        .collect())
317}

319/// The direction casacore's `MSFieldColumns::extractDirMeas` gives a field
320/// that follows `series` at `mjd_utc` (days): the body's geocentric
321/// direction, shifted by the field's `PHASE_DIR` offset (`offset_rad`) by
322/// its true angle, in the ephemeris's frame.
323///
324/// # Errors
325///
326/// A time the samples do not cover, or a non-finite offset direction.
327pub(crate) fn reference_direction(
328    series: &FieldEphemeris,
329    mjd_utc: f64,
330    offset_rad: [f64; 2],
331) -> Result<MDirection, MsError> {
332    let sample =
333        evaluate(series, mjd_utc).map_err(|error| MsError::InvalidInput(error.to_string()))?;
334    let body =
335        MDirection::from_cosines(sample.geocentric_position_metres, sample.position_reference);
336    Ok(casa_types::measures::ephemeris::offset_true_angle(
337        &body, offset_rad,
338    )?)
339}

341/// What a FIELD row's direction follows.
342#[derive(Clone, Debug)]
343pub(crate) enum FieldMotion {
344    /// Its `PHASE_DIR` (or `PHASE_DIR` polynomial).
345    Fixed,
346    /// Its attached ephemeris, `PHASE_DIR` then the offset from the body.
347    Ephemeris(Arc<FieldEphemeris>),
348    /// An attached ephemeris that could not be read, and why: refused
349    /// wherever the field's direction is asked for, not before.
350    Unreadable(Arc<str>),
351}
352
353/// What each FIELD row of `measurement_set` follows
354/// ([`field_ephemeris_ids`]), each table read once however many rows share
355/// it ([`attached_ephemeris`]). A table that cannot be read leaves its
356/// fields [`FieldMotion::Unreadable`]: casacore opens such a MeasurementSet,
357/// and a run that never asks for those fields' directions images as before.
358///
359/// # Errors
360///
361/// FIELD cannot be read.
362pub(crate) fn field_motions(
363    measurement_set: &MeasurementSet,
364) -> Result<Box<[FieldMotion]>, MsError> {
365    let ids = field_ephemeris_ids(measurement_set)?;
366    let mut read: BTreeMap<i32, FieldMotion> = BTreeMap::new();
367    Ok(ids
368        .iter()
369        .map(|id| {
370            let Some(id) = *id else {
371                return FieldMotion::Fixed;
372            };
373            read.entry(id)
374                .or_insert_with(|| {
375                    let series = measurement_set
376                        .path()
377                        .ok_or(PhaseCentreEphemerisError::UnbackedMeasurementSet)
378                        .and_then(|root| attached_ephemeris(root, id));
379                    match series {
380                        Ok(series) => FieldMotion::Ephemeris(Arc::new(series)),
381                        Err(error) => FieldMotion::Unreadable(Arc::from(error.to_string())),
382                    }
383                })
384                .clone()
385        })
386        .collect())
387}

389impl EphemerisSeriesPreflight {
390    fn open(path: &Path) -> Result<Self, PhaseCentreEphemerisError> {
391        let table = Table::open(TableOptions::new(path))?;
392        for column in ["MJD", "RA", "DEC", "Rho", "RadVel"] {
393            if !table
394                .schema()
395                .is_some_and(|schema| schema.contains_column(column))
396            {
397                return Err(PhaseCentreEphemerisError::MissingColumn {
398                    column: column.to_string(),
399                });
400            }
401        }
402        let name_bytes = keyword_str(&table, "NAME")?.len();
403        let position_reference_name = keyword_str(&table, "posrefsys")?;
404        let position_reference = if position_reference_name.eq_ignore_ascii_case("ICRF/ICRS")
405            || position_reference_name.eq_ignore_ascii_case("ICRS")
406        {
407            DirectionRef::ICRS
408        } else if position_reference_name.eq_ignore_ascii_case("ICRF/J2000.0") {
409            DirectionRef::J2000
410        } else {
411            return Err(PhaseCentreEphemerisError::UnsupportedPositionReference {
412                reference: position_reference_name.to_string(),
413            });
414        };
415        let observer_location = keyword_str(&table, "obsloc")?;
416        if !observer_location.eq_ignore_ascii_case("GEOCENTRIC") {
417            return Err(PhaseCentreEphemerisError::UnsupportedObserverLocation {
418                location: observer_location.to_string(),
419            });
420        }
421        let geo_distance_km = keyword_f64(&table, "GeoDist")?;
422        let velocity_reference = if geo_distance_km.abs() * 1_000.0 > 1.0 {
423            RadialVelocityRef::TOPO
424        } else {
425            RadialVelocityRef::GEO
426        };
427        let row_count = table.row_count();
428        let retained_bytes = arc_allocation_bytes::<FieldEphemeris>()?
429            .checked_add(name_bytes)
430            .and_then(|bytes| {
431                row_count
432                    .checked_mul(SAMPLE_BYTES)
433                    .and_then(|rows| bytes.checked_add(rows))
434            })
435            .ok_or(PhaseCentreEphemerisError::ByteOverflow)?;
436        Ok(Self {
437            table,
438            row_count,
439            position_reference,
440            velocity_reference,
441            retained_bytes,
442        })
443    }
444
445    fn load(self) -> Result<FieldEphemeris, PhaseCentreEphemerisError> {
446        let name = keyword_str(&self.table, "NAME")?.to_string();
447        let mut times = Vec::with_capacity(self.row_count);
448        let mut directions = Vec::with_capacity(self.row_count);
449        let mut distances = Vec::with_capacity(self.row_count);
450        let mut velocities = Vec::with_capacity(self.row_count);
451        for row in 0..self.row_count {
452            let mjd_days = scalar_f64(&self.table, row, "MJD")?;
453            let longitude_rad = scalar_f64(&self.table, row, "RA")?.to_radians();
454            let latitude_rad = scalar_f64(&self.table, row, "DEC")?.to_radians();
455            let distance_metres = scalar_f64(&self.table, row, "Rho")? * AU_METRES;
456            let radial_velocity_m_per_s =
457                scalar_f64(&self.table, row, "RadVel")? * AU_METRES / 86_400.0;
458            if !mjd_days.is_finite()
459                || !longitude_rad.is_finite()
460                || !latitude_rad.is_finite()
461                || !distance_metres.is_finite()
462                || distance_metres <= 0.0
463                || !radial_velocity_m_per_s.is_finite()
464                || times.last().is_some_and(|prior: &f64| *prior >= mjd_days)
465            {
466                return Err(PhaseCentreEphemerisError::InvalidRow { row });
467            }
468            times.push(mjd_days);
469            directions.push([longitude_rad, latitude_rad]);
470            distances.push(distance_metres);
471            velocities.push(radial_velocity_m_per_s);
472        }
473        if times.len() < 2 {
474            return Err(PhaseCentreEphemerisError::InsufficientRows);
475        }
476        Ok(FieldEphemeris::new(
477            name,
478            times,
479            directions,
480            self.position_reference,
481            distances,
482            (velocities, self.velocity_reference),
483        )?)
484    }
485}

487pub(crate) fn find_field_ephemeris(
488    root: &Path,
489    ephemeris_id: i32,
490) -> Result<PathBuf, PhaseCentreEphemerisError> {
491    let prefix = format!("EPHEM{ephemeris_id}_");
492    let mut matches = fs::read_dir(root.join("FIELD"))?
493        .filter_map(Result::ok)
494        .map(|entry| entry.path())
495        .filter(|path| {
496            path.file_name()
497                .and_then(|name| name.to_str())
498                .is_some_and(|name| name.starts_with(&prefix) && name.ends_with(".tab"))
499        })
500        .collect::<Vec<_>>();
501    matches.sort();
502    match matches.as_slice() {
503        [path] => Ok(path.clone()),
504        [] => Err(PhaseCentreEphemerisError::MissingEphemerisTable { ephemeris_id }),
505        _ => Err(PhaseCentreEphemerisError::AmbiguousEphemerisTable { ephemeris_id }),
506    }
507}

509fn keyword_str<'a>(table: &'a Table, name: &str) -> Result<&'a str, PhaseCentreEphemerisError> {
510    match table.keywords().get(name) {
511        Some(Value::Scalar(ScalarValue::String(value))) => Ok(value),
512        _ => Err(PhaseCentreEphemerisError::InvalidKeyword {
513            keyword: name.to_string(),
514        }),
515    }
516}
517
518fn arc_allocation_bytes<T>() -> Result<usize, PhaseCentreEphemerisError> {
519    let header =
520        Layout::array::<AtomicUsize>(2).map_err(|_| PhaseCentreEphemerisError::ByteOverflow)?;
521    let (allocation, _) = header
522        .extend(Layout::new::<T>())
523        .map_err(|_| PhaseCentreEphemerisError::ByteOverflow)?;
524    Ok(allocation.pad_to_align().size())
525}
526
527fn selected_ephemeris_allocation_bytes() -> Result<usize, PhaseCentreEphemerisError> {
528    arc_allocation_bytes::<PhaseCentreEphemeris>()
529}
530
531fn admit_reference_data(
532    budget: EphemerisBudget,
533    required_bytes: usize,
534) -> Result<(), PhaseCentreEphemerisError> {
535    if required_bytes > budget.available_bytes() {
536        return Err(PhaseCentreEphemerisError::InsufficientReferenceDataBudget {
537            required_bytes,
538            available_bytes: budget.available_bytes(),
539        });
540    }
541    Ok(())
542}
543
544fn keyword_f64(table: &Table, name: &str) -> Result<f64, PhaseCentreEphemerisError> {
545    match table.keywords().get(name) {
546        Some(Value::Scalar(ScalarValue::Float64(value))) if value.is_finite() => Ok(*value),
547        _ => Err(PhaseCentreEphemerisError::InvalidKeyword {
548            keyword: name.to_string(),
549        }),
550    }
551}
552
553fn scalar_f64(table: &Table, row: usize, column: &str) -> Result<f64, PhaseCentreEphemerisError> {
554    match table.cell_accessor(row, column)?.scalar()? {
555        ScalarValue::Float64(value) => Ok(*value),
556        _ => Err(PhaseCentreEphemerisError::InvalidColumnType {
557            column: column.to_string(),
558        }),
559    }
560}

562/// Failure to bind or evaluate one immutable moving-source reference.
563#[derive(Debug, Error)]
564pub enum PhaseCentreEphemerisError {
565    /// MeasurementSet or table storage failed.
566    #[error(transparent)]
567    Storage(#[from] MsError),
568    /// Generic casacore table access failed.
569    #[error(transparent)]
570    Table(#[from] casa_tables::TableError),
571    /// Ephemeris directory inspection failed.
572    #[error(transparent)]
573    Io(#[from] std::io::Error),
574    /// A named target was empty.
575    #[error("moving-source target must be non-empty")]
576    InvalidTarget,
577    /// TRACKFIELD was requested from an in-memory MeasurementSet.
578    #[error("TRACKFIELD requires a filesystem-backed MeasurementSet")]
579    UnbackedMeasurementSet,
580    /// No selected FIELD rows were supplied.
581    #[error("TRACKFIELD requires at least one selected FIELD row")]
582    EmptyTrackedFieldSet,
583    /// A selected FIELD does not bind an ephemeris.
584    #[error("selected FIELD_ID {field_id} has no attached ephemeris")]
585    MissingTrackedField {
586        /// Selected FIELD row without an attached ephemeris.
587        field_id: usize,
588    },
589    /// No attached ephemeris table matched EPHEMERIS_ID.
590    #[error("FIELD EPHEMERIS_ID {ephemeris_id} has no matching EPHEM table")]
591    MissingEphemerisTable {
592        /// FIELD EPHEMERIS_ID without a matching table.
593        ephemeris_id: i32,
594    },
595    /// Multiple attached tables matched one EPHEMERIS_ID.
596    #[error("FIELD EPHEMERIS_ID {ephemeris_id} matches multiple EPHEM tables")]
597    AmbiguousEphemerisTable {
598        /// FIELD EPHEMERIS_ID matching more than one table.
599        ephemeris_id: i32,
600    },
601    /// A required standard ephemeris column was absent.
602    #[error("ephemeris table is missing required column {column}")]
603    MissingColumn {
604        /// Required column name.
605        column: String,
606    },
607    /// A required ephemeris keyword was absent or invalid.
608    #[error("ephemeris table keyword {keyword} is missing or invalid")]
609    InvalidKeyword {
610        /// Required keyword name.
611        keyword: String,
612    },
613    /// The persisted ephemeris position reference is unsupported.
614    #[error("ephemeris position reference {reference:?} is unsupported")]
615    UnsupportedPositionReference {

731    #[test]
732    fn t41_external_ephemeris_binding_is_an_immutable_content_snapshot() {
733        let directory = tempfile::tempdir().expect("temporary ephemeris table");
734        let path = directory.path().join("external_ephemeris.tab");
735        save_ephemeris_table(
736            &path,
737            &[
738                [60_000.0, 10.0, -30.0, 2.0, 0.001],
739                [60_001.0, 11.0, -29.0, 2.1, 0.002],
740                [60_002.0, 12.0, -28.0, 2.2, 0.003],
741            ],
742        );
743
744        let budget = reference_budget(1 << 20);
745        let bound = PhaseCentreEphemeris::external(&path, budget).expect("bind snapshot");
746        let original_sample = sample_bits(bound.sample(0, 60_001.0).expect("sample snapshot"));
747
748        let mut backing = Table::open(TableOptions::new(&path)).expect("reopen backing table");
749        backing
750            .cell_accessor_mut(1, "RA")
751            .expect("mutable RA cell")
752            .set(Value::Scalar(ScalarValue::Float64(111.0)))
753            .expect("mutate persisted RA");
754        backing
755            .prepare_write()
756            .save_selected_columns(&["RA"])
757            .expect("persist mutated RA");
758
759        assert_eq!(
760            sample_bits(bound.sample(0, 60_001.0).expect("resample snapshot")),
761            original_sample,
762            "the bound series must not observe later backing-table mutation"
763        );
764
765        let rebound = PhaseCentreEphemeris::external(&path, budget).expect("bind changed table");
766        assert_ne!(
767            sample_bits(rebound.sample(0, 60_001.0).expect("sample changed table")),
768            original_sample,
769            "a new binding must observe changed scientific content"
770        );
771    }

775        let directory = tempfile::tempdir().expect("temporary oversized ephemeris table");
776        let path = directory.path().join("oversized.tab");

778            &path,
779            &[

781                [60_001.0, 11.0, -29.0, 2.1, 0.002],
782            ],
783        );
784        let preflight = EphemerisSeriesPreflight::open(&path).expect("preflight table metadata");
785        let required_bytes = selected_ephemeris_allocation_bytes()
786            .and_then(|bytes| {
787                bytes
788                    .checked_add(preflight.retained_bytes)
789                    .ok_or(PhaseCentreEphemerisError::ByteOverflow)
790            })
791            .expect("finite fixture reference-data charge");
792        drop(preflight);
793
794        assert!(matches!(
795            PhaseCentreEphemeris::external(
796                &path,
797                reference_budget(required_bytes - 1),
798            ),
799            Err(
800                PhaseCentreEphemerisError::InsufficientReferenceDataBudget {
801                    required_bytes: required,
802                    available_bytes,
803                }
804            ) if required == required_bytes && available_bytes == required_bytes - 1
805        ));
806        assert!(matches!(
807            PhaseCentreEphemeris::external(&path, reference_budget(required_bytes)),
808            Err(PhaseCentreEphemerisError::InvalidRow { row: 0 })
809        ));
