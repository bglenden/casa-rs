2//! Reads of the CASA-named calibration surfaces, `applycal`, `gaincal`,
3//! `bandpass`, `fluxscale` and `gencal`, into their requests (#931).
4//!
5//! These surfaces take CASA 6.7.6's parameters and values, read as CASA's
6//! task scripts read them:
7//!
8//! - `field`, `spw` and `intent` always select; the other selectors only
9//!   while `selectdata` is set, as CASA resets them otherwise.
10//! - The chain is `callib` when `docallib` is set, else `gaintable`. Its
11//!   per-table lists are positional: `gaintable[i]` takes `gainfield[i]`,
12//!   `interp[i]` and `spwmap[i]`, and a list shorter than `gaintable` leaves
13//!   the later tables CASA's defaults (every field, `'linear'`, no window
14//!   map). An empty `gaintable` entry is skipped. `calwt` with one value
15//!   holds for every table; a solve applies each table with `calwt=True`.
16//!
17//! A value CASA takes that casa-rs does not implement yet is refused naming
18//! #979; the session already refuses the parameters casa-rs does not
19//! implement at all unless they hold CASA's default.
20

37/// The request a CASA-named surface makes.
38pub(super) fn task_request(
39    task: &TaskInvocation<'_>,
40) -> Result<CalibrationTaskRequest, TaskFailure> {
41    Ok(match task.surface() {
42        "applycal" => CalibrationTaskRequest::ExecuteApply(applycal(task)?),
43        "gaincal" => CalibrationTaskRequest::SolveGain(gaincal(task)?),
44        "fringefit" => CalibrationTaskRequest::SolveGain(fringefit(task)?),
45        "accor" => CalibrationTaskRequest::SolveAccor(accor(task)?),
46        "blcal" => blcal(task)?,
47        "bandpass" => CalibrationTaskRequest::SolveBandpass(bandpass(task)?),
48        "fluxscale" => CalibrationTaskRequest::FluxScale(fluxscale(task)?),
49        "gencal" => CalibrationTaskRequest::Gencal(gencal(task)?),
50        other => {
51            return Err(TaskFailure::validation(format!(
52                "calibrate runs no surface {other:?}"
53            )));
54        }
55    })
56}

58fn applycal(task: &TaskInvocation<'_>) -> Result<ExecuteApplyTaskRequest, TaskFailure> {
59    let (calibration_tables, callibrary_error, argument_error) = if task.flag("docallib")? {
60        let source = task.text("callib")?.unwrap_or_default();
61        match load_apply_specs_from_callib(Path::new(&source)) {
62            Ok(tables) => (tables, None, None),
63            Err(error) => (Vec::new(), Some(error.to_string()), None),
64        }
65    } else {
66        let (tables, error) = traditional_chain(task, Chain::Apply)?;
67        (tables, None, error)
68    };
69    Ok(ExecuteApplyTaskRequest {
70        measurement_set: task.path("vis")?,
71        selection: selection(task)?,
72        calibration_tables,
73        callibrary_error,
74        argument_error,
75        apply_mode: applymode(task)?,
76        parang: task.flag("parang")?,
77        flagbackup: task.flag("flagbackup")?,
78    })
79}

81fn gaincal(task: &TaskInvocation<'_>) -> Result<SolveGainTaskRequest, TaskFailure> {
82    let gain_type = gaintype(task)?;
83    let spline = gain_type == GainType::GSpline;
84    let spline_controls = if spline {
85        Some(GainSplineControls {
86            splinetime: task.number("splinetime")?,
87            npointaver: task.integer("npointaver")?,
88            phasewrap: task.number("phasewrap")?,
89        })
90    } else {
91        None
92    };
93    let measurement_set = task.path("vis")?;
94    let min_snr = if spline { 0. } else { min_snr(task)? };
95    let (model_source, smodel) = model(task, &measurement_set, true)?;
96    let reference = task.required_text("refant")?;
97    let strict = match task.required_text("refantmode")?.as_str() {
98        _ if spline => false,
99        "flex" => false,
100        "strict" => true,
101        other => return Err(unknown("refantmode", other, "'flex' or 'strict'")),
102    };
103    Ok(SolveGainTaskRequest {
104        append: task.flag("append")?,
105        selection: selection(task)?,
106        output_table: PathBuf::from(task.required_text("caltable")?),
107        gain_type,
108        spline_controls,
109        fringe_controls: None,
110        solve_mode: calmode(task)?,
111        solve_interval: if spline {
112            GainSolveInterval::Infinite
113        } else {
114            solint(task)?
115        },
116        preaverage_seconds: task.number("preavg")?,
117        correlation_dependent_flags: task.flag("corrdepflags")?,
118        combine: if spline {
119            GainSolveCombine::default()
120        } else {
121            combine(task)?
122        },
123        refant: if reference.is_empty() {
124            RefAntSelector::Unspecified
125        } else {
126            RefAntSelector::Expression {
127                selection: reference,
128                strict,
129            }
130        },
131        prior_calibration_tables: chain(task, Chain::Solve)?,
132        parang: task.flag("parang")?,
133        model_source,
134        normalize_average_amplitude: task.flag("solnorm")?,
135        normalization_type: match task.required_text("normtype")?.as_str() {
136            "mean" => GainNormalization::Mean,
137            "median" => GainNormalization::Median,
138            other => return Err(unknown("normtype", other, "'mean' or 'median'")),
139        },
140        robustness: if spline {
141            Default::default()
142        } else {
143            robust_gain_solve(task)?
144        },
145        min_snr,
146        min_baselines_per_antenna: if spline {
147            0
148        } else {
149            count(task, "minblperant")?
150        },
151        smodel,
152        measurement_set,
153    })
154}

156fn fringefit(task: &TaskInvocation<'_>) -> Result<SolveGainTaskRequest, TaskFailure> {
157    let reference = task.required_text("refant")?;
158    let active_parameters = task.flags("paramactive")?;
159    let active_parameters = match active_parameters.as_slice() {
160        [] => [true, true, false],
161        [delay, rate, dispersion] => [*delay, *rate, *dispersion],
162        _ => {
163            return Err(TaskFailure::validation(
164                "paramactive vector must have exactly three entries",
165            ));
166        }
167    };
168    let search_window = |name| -> Result<[f64; 2], TaskFailure> {
169        Ok(task.numbers(name)?.try_into().unwrap_or([-1e6, 1e6]))
170    };
171    let iterations = usize::try_from(task.integer("niter")?).unwrap_or(usize::MAX);
172    let correlation_combination = match task.required_text("corrcomb")?.as_str() {
173        "none" => crate::FringeCorrelation::Separate,
174        "stokes" => crate::FringeCorrelation::Stokes,
175        "parallel" => crate::FringeCorrelation::Parallel,
176        other => return Err(unknown("corrcomb", other, "'none', 'stokes' or 'parallel'")),
177    };
178    Ok(SolveGainTaskRequest {
179        measurement_set: task.path("vis")?,
180        selection: selection(task)?,
181        output_table: PathBuf::from(task.required_text("caltable")?),
182        append: task.flag("append")?,
183        gain_type: GainType::K,
184        spline_controls: None,
185        fringe_controls: Some(crate::FringeControls {
186            delay_window: search_window("delaywindow")?,
187            rate_window: search_window("ratewindow")?,
188            global_solve: task.flag("globalsolve")?,
189            iterations,
190            active_parameters,
191            zero_rates: task.flag("zerorates")?,
192            correlation_combination,
193            concatenate_windows: task.flag("concatspws")?,
194        }),
195        solve_mode: GainSolveMode::Phase,
196        solve_interval: solint(task)?,
197        preaverage_seconds: 0.,
198        correlation_dependent_flags: task.flag("corrdepflags")?,
199        combine: combine(task)?,
200        refant: if reference.is_empty() {
201            RefAntSelector::Unspecified
202        } else {
203            RefAntSelector::Expression {
204                selection: reference,
205                strict: false,
206            }
207        },
208        prior_calibration_tables: chain(task, Chain::Solve)?,
209        parang: task.flag("parang")?,
210        model_source: GainSolveModelSource::FromMeasurementSet,
211        normalize_average_amplitude: false,
212        normalization_type: GainNormalization::Mean,
213        robustness: Default::default(),
214        min_snr: task.number("minsnr")? as f32,
215        min_baselines_per_antenna: 0,
216        smodel: [1., 0., 0., 0.],
217    })
218}

247fn blcal(task: &TaskInvocation<'_>) -> Result<CalibrationTaskRequest, TaskFailure> {
248    if task.flag("selectdata")?
249        && let Some(ParameterValue::Bool(observation)) = task.value("observation")
250    {
251        let name = if *observation { "True" } else { "False" };
252        let first = name.chars().next().expect("boolean text");
253        return Err(TaskFailure::execution(format!(
254            "Error in data selection specification: Scan Expression: Parse error at or near '{first}'\n(near char. 1 in string \"{name}\")"
255        )));
256    }
257    let frequency_dependent = task.flag("freqdep")?;
258    let output_table = task.required_text("caltable")?;
259    if output_table.is_empty() {
260        return Err(TaskFailure::execution("Error in Calibrater::setsolve."));
261    }
262    let supplied_interval = selector_text(task.value("solint")).unwrap_or_default();
263    let (time, frequency) = supplied_interval
264        .split_once(',')
265        .unwrap_or((&supplied_interval, ""));
266    let frequency_interval = if frequency_dependent {
267        parse_frequency_solint(frequency)?
268    } else {
269        None
270    };
271    Ok(CalibrationTaskRequest::SolveBaseline {
272        frequency_dependent,
273        frequency_interval,
274        gain: SolveGainTaskRequest {
275            measurement_set: task.path("vis")?,
276            selection: selection(task)?,
277            output_table: PathBuf::from(output_table),
278            append: false,
279            gain_type: GainType::G,
280            spline_controls: None,
281            fringe_controls: None,
282            solve_mode: calmode(task)?,
283            solve_interval: parse_time_solint(time)?,
284            preaverage_seconds: 0.,
285            correlation_dependent_flags: false,
286            combine: combine(task)?,
287            refant: RefAntSelector::Unspecified,
288            prior_calibration_tables: traditional_chain(task, Chain::Solve)?.0,
289            parang: task.flag("parang")?,
290            model_source: GainSolveModelSource::FromMeasurementSet,
291            normalize_average_amplitude: task.flag("solnorm")?,
292            normalization_type: GainNormalization::Mean,
293            robustness: Default::default(),
294            min_snr: 0.,
295            min_baselines_per_antenna: 0,
296            smodel: [1., 0., 0., 0.],
297        },
298    })
299}

301fn bandpass(task: &TaskInvocation<'_>) -> Result<SolveBandpassTaskRequest, TaskFailure> {
302    let measurement_set = task.path("vis")?;
303    let band_type = match task.required_text("bandtype")?.as_str() {
304        "B" => BandpassType::B,
305        "BPOLY" => BandpassType::BPoly,
306        other => return Err(unknown("bandtype", other, "'B' or 'BPOLY'")),
307    };
308    let (model_source, smodel) = model(task, &measurement_set, true)?;
309    let supplied_interval = selector_text(task.value("solint")).unwrap_or_default();
310    let (time_interval, frequency_interval) = supplied_interval
311        .split_once(',')
312        .unwrap_or((&supplied_interval, ""));
313    let combine = combine(task)?;
314    let degrees = [task.integer("degamp")?, task.integer("degphase")?];
315    // CASA reads maskedge only for BPOLY.
316    let mask_edge_percent = match band_type {
317        BandpassType::B => BandpassSolveCuts::CASA_DEFAULTS.mask_edge_percent,
318        BandpassType::BPoly => task.number("maskedge")? as f32,
319    };
320    Ok(SolveBandpassTaskRequest {
321        append: task.flag("append")?,
322        selection: selection(task)?,
323        output_table: PathBuf::from(task.required_text("caltable")?),
324        refant: bandpass_refant(&task.required_text("refant")?),
325        prior_calibration_tables: chain(task, Chain::Solve)?,
326        parang: task.flag("parang")?,
327        combine: BandpassSolveCombine {
328            scans: combine.scans,
329            fields: combine.fields,
330            observations: combine.observations,
331            spectral_windows: combine.spectral_windows,
332        },
333        band_type,
334        normalize_average_amplitude: task.flag("solnorm")?,
335        amplitude_degree: degrees[0].max(0) as usize,
336        phase_degree: degrees[1].max(0) as usize,
337        model_source,
338        solve_interval: parse_time_solint(time_interval)?,
339        frequency_interval: parse_frequency_solint(frequency_interval)?,
340        correlation_dependent_flags: task.flag("corrdepflags")?,
341        smodel,
342        cuts: Some(BandpassSolveCuts {
343            min_baselines_per_antenna: task.integer("minblperant")?.max(0) as usize,
344            min_snr: task.number("minsnr")? as f32,
345            fill_gaps: match band_type {
346                BandpassType::B => {
347                    usize::try_from(task.integer("fillgaps")?.max(0)).unwrap_or(usize::MAX)
348                }
349                BandpassType::BPoly => 0,
350            },
351            mask_edge_percent,
352            mask_center_channels: if band_type == BandpassType::BPoly {
353                usize::try_from(task.integer("maskcenter")?.max(0)).unwrap_or(usize::MAX)
354            } else {
355                0
356            },
357            normalize_visibilities: band_type == BandpassType::BPoly && task.flag("visnorm")?,
358            omit_polynomial: degrees.map(|degree| degree == -1),
359        }),
360        measurement_set,
361    })
362}

364pub(super) fn leakage_request(
365    task: &TaskInvocation<'_>,
366    vis: &Path,
367) -> Result<crate::solve::grouping::leakage::LeakageSolveRequest, TaskFailure> {
368    use crate::solve::grouping::leakage::{LeakageSolveKind, LeakageSolveRequest};
369
370    let gain = task.surface() == "gaincal";
371    let poltype = task.required_text(if gain { "gaintype" } else { "poltype" })?;
372    let kind = match poltype.as_str() {
373        "D" => LeakageSolveKind::D,
374        "Df" => LeakageSolveKind::Df,
375        "D+QU" => LeakageSolveKind::DQu,
376        "Df+QU" => LeakageSolveKind::DfQu,
377        "D+X" => LeakageSolveKind::DX,
378        "Df+X" => LeakageSolveKind::DfX,
379        "Dgen" => LeakageSolveKind::DGen,
380        "Dfgen" => LeakageSolveKind::DfGen,
381        "Dgen+QU" => LeakageSolveKind::DGenQu,
382        "Dfgen+QU" => LeakageSolveKind::DfGenQu,
383        "Dgen+X" => LeakageSolveKind::DGenX,
384        "Dfgen+X" => LeakageSolveKind::DfGenX,
385        "Xf" => LeakageSolveKind::Xf,
386        "Xj" => LeakageSolveKind::Xj,
387        "PosAng" => LeakageSolveKind::PosAng,
388        "Xparang+QU" => LeakageSolveKind::Xparang,
389        "Xfparang+QU" => LeakageSolveKind::Xfparang,
390        "XY+QU" if gain => LeakageSolveKind::XY,
391        "XYf+QU" if gain => LeakageSolveKind::XYf,
392        "X" => {
393            return Err(TaskFailure::execution(
394                "Can't do selfGatherAndSolve for X Mueller",
395            ));
396        }
397        "Dlls" => LeakageSolveKind::Dlls,
398        "Dflls" => LeakageSolveKind::Dflls,
399        _ => {
400            return Err(TaskFailure::validation(format!(
401                "poltype={poltype:?} is under implementation in #1522"
402            ))
403            .at("poltype"));
404        }
405    };
406    let (model_source, stokes) = model(task, vis, true)?;
407    let interval = selector_text(task.value("solint")).unwrap_or_default();
408    let (time, frequency) = interval.split_once(',').unwrap_or((&interval, ""));
409    Ok(LeakageSolveRequest {
410        output: PathBuf::from(task.required_text("caltable")?),
411        selection: selection(task)?,
412        kind,
413        model_source,
414        stokes,
415        priors: chain(task, Chain::Solve)?,
416        parang_terms: usize::from(!gain || task.flag("parang")?),
417        correlation_dependent_flags: gain && task.flag("corrdepflags")?,
418        solve_mode: if gain {
419            calmode(task)?
420        } else {
421            GainSolveMode::AmplitudePhase
422        },
423        normalization: None,
424        interval: parse_time_solint(time)?,
425        frequency_interval: parse_frequency_solint(frequency)?,
426        combine: combine(task)?,
427        preaverage_seconds: task.number("preavg")?,
428        min_baselines: task.integer("minblperant")?.max(0) as usize,
429        min_snr: task.number("minsnr")? as f32,
430        refant: bandpass_refant(&task.required_text("refant")?),
431        append: task.flag("append")?,
432    })
433}

435/// `minsnr`, which CASA takes as any number; casa-rs takes it finite and
436/// not negative.
437fn min_snr(task: &TaskInvocation<'_>) -> Result<f32, TaskFailure> {
438    let min_snr = task.number("minsnr")? as f32;
439    if !min_snr.is_finite() || min_snr < 0.0 {
440        return Err(
441            TaskFailure::validation("minsnr must be a finite non-negative number").at("minsnr"),
442        );
443    }
444    Ok(min_snr)
445}

447fn fluxscale(task: &TaskInvocation<'_>) -> Result<FluxScaleRequest, TaskFailure> {
448    // CASA opens `vis` for its field and window names; casa-rs reads them
449    // from the gain table, but a MeasurementSet that is not there is still
450    // an error, as it is to CASA.
451    let vis = task.path("vis")?;
452    if !vis.exists() {
453        return Err(TaskFailure::validation(format!(
454            "MeasurementSet {} does not exist",
455            vis.display()
456        ))
457        .at("vis"));
458    }
459    let reference_fields = fields(task, "reference")?;
460    if reference_fields.is_empty() {
461        return Err(
462            TaskFailure::validation("fluxscale requires at least one reference field")
463                .at("reference"),
464        );
465    }
466    let refspwmap = task
467        .integers("refspwmap")?
468        .into_iter()
469        .map(|id| i32::try_from(id).map_err(|_| out_of_range("refspwmap", id)))
470        .collect::<Result<Vec<_>, _>>()?;
471    let gainthreshold = task.number("gainthreshold")?;
472    Ok(FluxScaleRequest {
473        options: crate::FluxScaleOptions {
474            measurement_set: Some(vis),
475            antenna: task.text("antenna")?.unwrap_or_default(),
476            timerange: task.text("timerange")?.unwrap_or_default(),
477            scan: task.text("scan")?.unwrap_or_default(),
478            fitorder: i32::try_from(task.integer("fitorder")?).map_err(|_| {
479                TaskFailure::validation("fitorder exceeds CASA's integer range").at("fitorder")
480            })?,
481            listfile: task
482                .text("listfile")?
483                .filter(|text| !text.is_empty())
484                .map(PathBuf::from),
485            display: task.flag("display")?,
486        },
487        append: task.flag("append")?,
488        input_table: PathBuf::from(task.required_text("caltable")?),
489        output_table: PathBuf::from(task.required_text("fluxtable")?),
490        reference_fields,
491        transfer_fields: fields(task, "transfer")?,
492        // CASA's [-1] maps no window.
493        refspwmap: if refspwmap == [-1] {
494            Vec::new()
495        } else {
496            refspwmap
497        },
498        // CASA's -1 excludes no gain.
499        gainthreshold: (gainthreshold >= 0.0).then_some(gainthreshold),
500        incremental: task.flag("incremental")?,
501    })
502}

504fn gencal(task: &TaskInvocation<'_>) -> Result<GencalTaskRequest, TaskFailure> {
505    let caltype = match task.required_text("caltype")?.as_str() {
506        "" => {
507            return Err(TaskFailure::validation("gencal requires a caltype").at("caltype"));
508        }
509        "amp" => GencalType::Amp,
510        "ph" => GencalType::Ph,
511        "sbd" => GencalType::Sbd,
512        "mbd" => GencalType::Mbd,
513        "antpos" => GencalType::Antpos,
514        "antposvla" => GencalType::AntposVla,
515        "gc" => GencalType::Gc,
516        "gceff" => GencalType::Gceff,
517        "eff" => GencalType::Eff,
518        "tsys" => GencalType::Tsys,
519        "opac" => GencalType::Opac,
520        "swpow" => GencalType::Swpow,
521        "evlagain" => GencalType::Evlagain,
522        "swp/rq" => GencalType::SwpOverRq,
523        "swpwts" => GencalType::Swpwts,
524        "rq" => GencalType::Rq,
525        "tecim" => GencalType::Tecim,
526        "eop" => GencalType::Eop,
527        other => return Err(not_yet("caltype", other)),
528    };
529    let antenna = task.text("antenna")?.unwrap_or_default();
530    let infile = if caltype.is_gain_curve()
531        || matches!(
532            caltype,
533            GencalType::Antpos | GencalType::Tecim | GencalType::Eop
534        ) {
535        task.text("infile")?
536            .filter(|path| !path.is_empty() && !(caltype == GencalType::Antpos && path == "None"))
537            .map(PathBuf::from)
538    } else {
539        None
540    };
541    if caltype == GencalType::Tecim && !infile.as_ref().is_some_and(|path| path.exists()) {
542        return Err(
543            TaskFailure::validation("An existing tec map must be specified in infile").at("infile"),
544        );
545    }
546    Ok(GencalTaskRequest {
547        prior: GencalRequest {
548            measurement_set: task.path("vis")?,
549            output_table: PathBuf::from(task.required_text("caltable")?),
550            caltype,
551            antenna,
552            spw: task.text("spw")?.unwrap_or_default(),
553            parameter: task.numbers("parameter")?,
554            gaincurve_table: task.text("gaincurve_table")?.map(PathBuf::from),
555        },
556        polarization: task.text("pol")?.unwrap_or_default(),
557        gaincurve_infile: infile.clone().filter(|_| caltype.is_gain_curve()),
558        antpos_infile: infile.clone().filter(|_| caltype == GencalType::Antpos),
559        tec_image: infile.clone().filter(|_| caltype == GencalType::Tecim),
560        eop_infile: infile.filter(|_| caltype == GencalType::Eop),
561        ant_pos_time_limit: (caltype == GencalType::Antpos)
562            .then(|| task.integer("ant_pos_time_limit"))
563            .transpose()?,
564        uniform: (caltype == GencalType::Tsys)
565            .then(|| task.flag("uniform"))
566            .transpose()?,
567    })
568}

570/// CASA's selection: `field`, `spw` and `intent` always; the others are inactive,
571/// and so unset, while `selectdata` is not.
572fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
573    let text = |name: &str| {
574        task.text(name)
575            .map(|value| value.filter(|text| !text.is_empty()))
576    };
577    let mut selection = MsSelection {
578        selectdata: true,
579        field: text("field")?,
580        spw: text("spw")?,
581        antenna: text("antenna")?,
582        scan: text("scan")?,
583        observation: selector_text(task.value("observation")),
584        array: text("array")?,
585        timerange: text("timerange")?,
586        msselect: text("msselect")?,
587        ..MsSelection::default()
588    };
589    selection.intent = task.text("intent")?.filter(|text| !text.is_empty());
590    selection.uvrange = uv_selector_text(task.value("uvrange"));
591    // calibrater.selectvis converts these variants with toCasaString, which
592    // removes one outer bracket pair even when the input was a string.
593    for selector in [
594        &mut selection.field,
595        &mut selection.spw,
596        &mut selection.intent,
597        &mut selection.timerange,
598        &mut selection.uvrange,
599        &mut selection.antenna,
600        &mut selection.scan,
601        &mut selection.observation,
602    ] {
603        if let Some(text) = selector
604            && let Some(inner) = text
605                .strip_prefix('[')
606                .and_then(|text| text.strip_suffix(']'))
607        {
608            *selector = (!inner.is_empty()).then(|| inner.to_string());
609        }
610    }
611    // CASA accepts these parameters even when disabled, retaining their input
612    // values for HISTORY while passing empty selectors to calibrater.selectvis.
613    if !task.flag("selectdata")? {
614        selection.timerange = None;
615        selection.uvrange = None;
616        selection.antenna = None;
617        selection.scan = None;
618        selection.observation = None;
619        selection.msselect = None;
620    }
621    if matches!(task.surface(), "gaincal" | "bandpass" | "polcal") {
622        // The public task always adds this predicate, including when the
623        // optional selectors are disabled.
624        selection.msselect = Some(match selection.msselect {
625            Some(predicate) => format!("({predicate}) && ANTENNA1!=ANTENNA2"),
626            None => "ANTENNA1!=ANTENNA2".into(),
627        });
628    }
629    Ok(selection)
630}

632pub(super) fn selector_text(value: Option<&ParameterValue>) -> Option<String> {
633    let text = match value? {
634        ParameterValue::String(text) => text.clone(),
635        ParameterValue::NumpyScalar { value, .. } | ParameterValue::ScalarArray(value) => {
636            return selector_text(Some(value));
637        }
638        other => python_repr(other),
639    };
640    (!text.is_empty()).then_some(text)
641}
642
643pub(super) fn uv_selector_text(value: Option<&ParameterValue>) -> Option<String> {
644    match value? {
645        ParameterValue::Array(values) => {
646            let text = values
647                .iter()
648                .filter_map(|value| selector_text(Some(value)))
649                .collect::<Vec<_>>()
650                .join(",");
651            (!text.is_empty()).then_some(text)
652        }
653        ParameterValue::EmptyArray { .. } => None,
654        value => selector_text(Some(value)),
655    }
656}

658/// Whether a chain is applied, each table with its own `calwt`, or
659/// pre-applied while solving, each with `calwt=True`.
660#[derive(Debug, Clone, Copy, PartialEq, Eq)]
661enum Chain {
662    Apply,
663    Solve,
664}
665
666/// The chain `callib` names when `docallib` is set, else `gaintable` with
667/// its positional lists.
668fn chain(
669    task: &TaskInvocation<'_>,
670    chain: Chain,
671) -> Result<Vec<ApplyCalibrationTableSpec>, TaskFailure> {
672    if task.flag("docallib")? {
673        let callib = task.text("callib")?.ok_or_else(|| {
674            TaskFailure::validation("docallib requires callib, a calibration library file")
675                .at("callib")
676        })?;
677        return load_apply_specs_from_callib(Path::new(&callib))
678            .map_err(|error| TaskFailure::validation(error.to_string()).at("callib"));
679    }
680    traditional_chain(task, chain).map(|(tables, _)| tables)
681}
682
683fn traditional_chain(
684    task: &TaskInvocation<'_>,
685    chain: Chain,
686) -> Result<(Vec<ApplyCalibrationTableSpec>, Option<ApplyArgumentError>), TaskFailure> {
687    let tables = task.texts("gaintable")?;
688    let gainfield = task.texts("gainfield")?;
689    let interp = task.texts("interp")?;
690    let spwmap = match apply_spw_maps(task.value("spwmap")) {
691        Ok(maps) => maps,
692        Err(error) if chain == Chain::Apply => return Ok((Vec::new(), Some(error))),
693        Err(error) => return Err(TaskFailure::validation(error.to_string()).at("spwmap")),
694    };
695    let calwt = match chain {
696        Chain::Apply => task.flags("calwt")?,
697        Chain::Solve => vec![true],
698    };
699    let mut specs = Vec::new();
700    for (index, table) in tables.iter().enumerate() {
701        if table.is_empty() {
702            continue;
703        }
704        let weight = match calwt.as_slice() {
705            [one] => *one,
706            values => match values.get(index) {
707                Some(weight) => *weight,
708                // Python indexes calwt immediately before setapply for an active
709                // table. All earlier table setup still takes place first.
710                None => return Ok((specs, Some(ApplyArgumentError::MissingCalwt))),
711            },
712        };
713        let mut spec = ApplyCalibrationTableSpec::new(table);
714        spec.gainfield = gainfield
715            .get(index)
716            .and_then(|field| parse_gainfield_value(field));
717        spec.interp = match interp.get(index) {
718            Some(text) => {
719                parse_interp_value(text).unwrap_or(crate::ApplyInterpolationMode::Invalid)
720            }
721            None => crate::ApplyInterpolationMode::Linear,
722        };
723        if let Some(map) = spwmap.get(index) {
724            match apply_spw_map(map) {
725                Ok(map) => spec.spwmap = map,
726                Err(error) if chain == Chain::Apply => return Ok((specs, Some(error))),
727                Err(error) => return Err(TaskFailure::validation(error.to_string()).at("spwmap")),
728            }
729        }
730        spec.calwt = weight;
731        specs.push(spec);
732    }
733    Ok((specs, None))
734}
735
736fn apply_spw_maps(
737    value: Option<&ParameterValue>,
738) -> Result<Vec<ParameterValue>, ApplyArgumentError> {
739    Ok(match value {
740        Some(ParameterValue::Array(items))
741            if !items.is_empty() && !matches!(items[0], ParameterValue::Array(_)) =>
742        {
743            vec![ParameterValue::Array(items.clone())]
744        }
745        Some(ParameterValue::Array(items)) => items.clone(),
746        Some(ParameterValue::String(text)) if text.is_empty() => Vec::new(),
747        Some(ParameterValue::Table(entries)) if entries.is_empty() => Vec::new(),
748        Some(ParameterValue::Table(_)) => return Err(ApplyArgumentError::SpwMap("0".into())),
749        Some(ParameterValue::EmptyArray { .. }) | None => Vec::new(),
750        Some(value) => {
751            let ty = match value {
752                ParameterValue::Integer(_) => "int",
753                ParameterValue::Float(_) => "float",
754                ParameterValue::Bool(_) => "bool",
755                _ => {
756                    return Err(ApplyArgumentError::SpwMap(
757                        "expected an integer window map".into(),
758                    ));
759                }
760            };
761            return Err(ApplyArgumentError::SpwMap(format!(
762                "object of type '{ty}' has no len()"
763            )));
764        }
765    })
766}
767
768fn apply_spw_map(value: &ParameterValue) -> Result<Vec<i32>, ApplyArgumentError> {
769    let ParameterValue::Array(items) = value else {
770        return Err(ApplyArgumentError::SpwMap(
771            "expected an integer window map".into(),
772        ));
773    };
774    let ids = items
775        .iter()
776        .map(|item| match item {
777            ParameterValue::Integer(id) => Ok(*id),
778            ParameterValue::Bool(id) => Ok(i64::from(*id)),
779            ParameterValue::NumpyScalar { value, .. }
780                if matches!(value.as_ref(), ParameterValue::Integer(_)) =>
781            {
782                let ParameterValue::Integer(id) = value.as_ref() else {
783                    unreachable!()
784                };
785                Ok(*id)
786            }
787            _ => Err(ApplyArgumentError::SpwMap(
788                "expected an integer window map".into(),
789            )),
790        })
791        .collect::<Result<Vec<_>, _>>()?;
792    window_map(&ids).map_err(|failure| ApplyArgumentError::SpwMap(failure.message))
793}
794
795fn window_map(map: &[i64]) -> Result<Vec<i32>, TaskFailure> {
796    if map == [-1] {
797        return Ok(Vec::new());
798    }
799    map.iter()
800        .map(|&id| i32::try_from(id).map_err(|_| out_of_range("spwmap", id)))
801        .collect()
802}

804/// Parse the point-source Stokes vector, with omitted Stokes set to zero.
805/// Gaincal and channelized bandpass resolve an empty vector from the MS.
806fn model(
807    task: &TaskInvocation<'_>,
808    measurement_set: &Path,
809    use_stored_models: bool,
810) -> Result<(GainSolveModelSource, [f32; 4]), TaskFailure> {
811    let smodel = task.numbers("smodel")?;
812    match smodel.as_slice() {
813        [] if use_stored_models => {
814            std::fs::metadata(measurement_set).map_err(|error| {
815                TaskFailure::validation(format!(
816                    "open MeasurementSet {}: {error}",
817                    measurement_set.display()
818                ))
819                .at("vis")
820            })?;
821            Ok((
822                GainSolveModelSource::FromMeasurementSet,
823                [1.0, 0.0, 0.0, 0.0],
824            ))
825        }
826        [] => {
827            let ms = MeasurementSet::open(measurement_set).map_err(|error| {
828                TaskFailure::validation(format!(
829                    "open MeasurementSet {}: {error}",
830                    measurement_set.display()
831                ))
832                .at("vis")
833            })?;
834            let virtual_fields = casa_ms::scratch_columns::virtual_model_fields(&ms);
835            if !virtual_fields.is_empty() {
836                return Err(TaskFailure::validation(format!(
837                    "smodel=[] reads the virtual model of field(s) {virtual_fields:?} (setjy with \
838                     usescratch=False), which casa-rs does not implement yet (#979); give smodel, \
839                     or write MODEL_DATA"
840                ))
841                .at("smodel"));
842            }
843            let source = if has_model_column(&ms) {
844                GainSolveModelSource::ModelColumn
845            } else {
846                GainSolveModelSource::PointSource
847            };
848            Ok((source, [1.0, 0.0, 0.0, 0.0]))
849        }
850        &[_, ref polarized @ ..] if polarized.len() < 4 => {
851            if !use_stored_models && polarized.iter().any(|stokes| *stokes != 0.0) {
852                return Err(not_yet("smodel", &format!("{smodel:?}")));
853            }
854            let stokes = crate::point_model_stokes(&smodel).ok_or_else(|| {
855                TaskFailure::validation("smodel needs finite Stokes values representable as floats")
856                    .at("smodel")
857            })?;
858            Ok((GainSolveModelSource::PointSource, stokes))
859        }
860        _ => Err(TaskFailure::validation(format!(
861            "smodel takes [I, Q, U, V], or fewer with the rest zero, or [] for the MODEL_DATA \
862             column; found {} values",
863            smodel.len()
864        ))
865        .at("smodel")),
866    }
867}

871    ms.main_table()
872        .schema()
873        .is_some_and(|schema| schema.contains_column("MODEL_DATA"))
874}

919/// CASA's `solint`, read as `SolvableVisCal::setSolve` reads it: empty
920/// text, or text that contains `inf`, in any case, is one solution per
921/// scan; text that contains `int`, one per integration; anything else is a
922/// time, in seconds when it has no unit. A time of zero is CASA's `inf`,
923/// and a negative one its `int`. G and T have a single gain per spectral
924/// window, so CASA ignores the frequency interval after a comma.
925fn solint(task: &TaskInvocation<'_>) -> Result<GainSolveInterval, TaskFailure> {
926    let supplied = selector_text(task.value("solint")).unwrap_or_default();
927    let time = supplied
928        .split_once(',')
929        .map_or(supplied.as_str(), |(time, _)| time);
930    parse_time_solint(time)
931}

934    GainSolveInterval::from_casa(text)
935        .ok_or_else(|| unknown("solint", text, "'inf', 'int' or a time such as '60s'"))
936}
