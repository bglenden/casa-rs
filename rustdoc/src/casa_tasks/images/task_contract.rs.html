23/// Top-level image-analysis JSON task request.
24#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
25#[serde(tag = "kind", content = "request", rename_all = "snake_case")]
26pub enum ImageAnalysisTaskRequest {
27    /// CASA `imhead(..., mode="summary")` style metadata summary.
28    Imhead(ImheadRequest),
29    /// CASA `imstat` style image statistics.
30    Imstat(ImstatRequest),
31    /// CASA `immoments` style moment map generation.
32    Immoments(ImmomentsRequest),
33    /// CASA `impv` style position-velocity extraction.
34    Impv(ImpvTaskRequest),
35    /// CASA `imsubimage` style pixel-box/channel extraction.
36    Imsubimage(ImsubimageRequest),
37    /// CASA `immath(..., mode="evalexpr")` style image arithmetic.
38    Immath(ImmathRequest),
39    /// CASA `impbcor` style primary-beam correction.
40    Impbcor(ImpbcorTaskRequest),
41    /// CASA `imregrid` style template-image regridding.
42    Imregrid(ImregridTaskRequest),
43    /// CASA `feather` style Fourier-domain image combination.
44    Feather(FeatherRequest),
45    /// CASA `exportfits` style FITS image export.
46    Exportfits(ExportFitsRequest),
47    /// CASA `importfits` style FITS image import.
48    Importfits(ImportFitsRequest),
49}

79/// JSON/schema wire request for `impv`.
80#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
81pub struct ImpvTaskRequest {
82    /// Input CASA image path.
83    pub imagename: PathBuf,
84    /// Output CASA image path.
85    pub outfile: PathBuf,
86    /// CASA mode, selected by the first letter of `coords` or `length`.
87    #[serde(default = "default_impv_mode")]
88    pub mode: String,
89    /// First endpoint in `coords` mode.
90    #[serde(default)]
91    pub start: Option<PvPosition>,
92    /// Second endpoint in `coords` mode.
93    #[serde(default)]
94    pub end: Option<PvPosition>,
95    /// Midpoint in `length` mode.
96    #[serde(default)]
97    pub center: Option<PvPosition>,
98    /// Distance between endpoints in pixels or angular units.
99    #[serde(default)]
100    pub length: Option<PvTaskAmount>,
101    /// Sky position angle in `length` mode.
102    #[serde(default)]
103    pub pa: Option<PvTaskAmount>,
104    /// Odd strip width in pixels, or a width in angular units.
105    #[serde(default = "default_impv_width")]
106    pub width: PvTaskAmount,
107    /// Angular unit of the output offset coordinate.
108    #[serde(default = "default_pv_unit")]
109    pub unit: String,
110    /// CASA text, named, or record region. Only non-direction axes may be cropped.
111    #[serde(default)]
112    pub region: Option<ParameterValue>,
113    /// CASA channel expression, supporting tutorial forms like `4~12`.
114    #[serde(default)]
115    pub chans: Option<String>,
116    /// Stokes planes to include.
117    #[serde(default)]
118    pub stokes: Option<String>,
119    /// Optional LEL mask evaluated before extraction.
120    #[serde(default)]
121    pub mask: Option<String>,
122    /// Stretch a conforming mask over singleton or missing axes.
123    #[serde(default)]
124    pub stretch: bool,
125    /// Replace an existing output image.
126    #[serde(default)]
127    pub overwrite: bool,
128}

130/// Wire representation of a length, width, or angle supplied to `impv`.
131#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
132#[serde(untagged)]
133pub enum PvTaskAmount {
134    /// Whole pixels.
135    Whole(i64),
136    /// Fractional pixels for the slice length.
137    Fractional(f64),
138    /// An angular value supplied as text.
139    Angular(String),
140    /// A CASA quanta record containing one angular value.
141    Quantity(BTreeMap<String, ParameterValue>),
142}

164    pub region: Option<PbcorTaskRegion>,
165    /// Optional channel selector.

174    /// Extend a conforming mask over singleton or missing axes.
175    #[serde(default)]

177    /// Replace an existing output image.
178    #[serde(default)]
179    pub overwrite: bool,

202/// JSON/schema wire request for `imregrid`.
203#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
204pub struct ImregridTaskRequest {
205    /// Input CASA image path.
206    pub imagename: PathBuf,
207    /// A template image, the `get` selector, or a coordinate and shape record.
208    pub template: ImregridTaskTemplate,
209    /// Output CASA image path.
210    #[serde(default)]
211    pub output: PathBuf,
212    /// Interpolation method (`nearest`, `linear`, or `cubic`).
213    #[serde(default = "default_regrid_interpolation")]
214    pub interpolation: String,
215    /// Regrid channels at equal radio velocity using the two rest frequencies.
216    #[serde(default = "default_true")]
217    pub asvelocity: bool,
218    /// Zero-based input pixel axes, or `[-1]` for matching nondegenerate axes.
219    #[serde(default)]
220    pub axes: Vec<i64>,
221    /// Output lengths in input axis order, or `[-1]` for CASA's inferred shape.
222    #[serde(default = "default_regrid_shape")]
223    pub shape: Vec<i64>,
224    /// Direction coordinate sampling interval, with zero using each pixel.
225    #[serde(default = "default_regrid_decimate")]
226    pub decimate: i64,
227    /// Resample in pixel space instead of using the coordinate conversion.
228    #[serde(default)]
229    pub replicate: bool,
230    /// Replace an existing output image.
231    #[serde(default)]
232    pub overwrite: bool,
233}

238pub enum ImregridTaskTemplate {

240    Image(PathBuf),

242    Record(BTreeMap<String, ParameterValue>),

245/// Wire result returned by `imregrid`.
246#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
247#[serde(untagged)]
248pub enum ImregridTaskOutcome {

253}

320/// Build the current request/result schema bundle.
321pub fn image_analysis_task_schema_bundle() -> TaskProviderContract {
322    try_image_analysis_task_schema_bundle().unwrap_or_else(|error| panic!("{error}"))

327    let result_schema = schema_for!(ImageAnalysisTaskResult);

343                },
344                TaskOperationDescriptor {
345                    name: "immoments".to_string(),
346                    request_kind: "immoments".to_string(),
347                    result_kind: Some("immoments".to_string()),
348                },
349                TaskOperationDescriptor {
350                    name: "impv".to_string(),
351                    request_kind: "impv".to_string(),

369                TaskOperationDescriptor {
370                    name: "imregrid".to_string(),

375                    name: "feather".to_string(),
376                    request_kind: "feather".to_string(),
377                    result_kind: Some("feather".to_string()),
378                },
379                TaskOperationDescriptor {
380                    name: "exportfits".to_string(),
381                    request_kind: "exportfits".to_string(),

383                },

385                    name: "importfits".to_string(),

397                    json_schema: Some("--json-schema".to_string()),
398                    protocol_info: Some("--protocol-info".to_string()),
399                    json_run: Some("--json-run <SOURCE>".to_string()),
400                },
401            }),
402            python: Some(serde_json::json!({
403                "module": "casars.tasks.image_analysis",
404                "functions": [
405                    "imhead",
406                    "imstat",
407                    "immoments",
408                    "impv",
409                    "imsubimage",
410                    "immath",
411                    "impbcor",
412                    "imregrid",
413                    "feather",
414                    "exportfits",
415                    "importfits"
416                ],
417            })),
418        },
419        parameter_surfaces: IMAGE_ANALYSIS_TASK_SURFACES
420            .into_iter()
421            .map(|surface| {
422                builtin_surface_bundle(surface).map_err(|error| {
423                    format!("built-in image-analysis parameter surface {surface:?}: {error}")
424                })
425            })
426            .collect::<Result<_, _>>()?,
427        domain_schemas: TaskProviderSchemas {
428            request_schema,
429            result_schema,
430        },
431    })
432}

434#[cfg(test)]
435mod tests {
436    use std::path::PathBuf;
437
438    use casa_imageanalysis::analysis::ImheadRequest;
439
440    use super::{
441        IMAGE_ANALYSIS_TASK_PROTOCOL_NAME, ImageAnalysisTaskRequest,
442        image_analysis_task_schema_bundle,

444
445    #[test]
446    fn image_analysis_schema_and_ui_surfaces_advertise_task_contracts() {
447        let bundle = image_analysis_task_schema_bundle();
448        bundle.validate().expect("image-analysis provider contract");
449        assert_eq!(
450            bundle.protocol.protocol_name,
451            IMAGE_ANALYSIS_TASK_PROTOCOL_NAME
452        );
453        assert_eq!(bundle.semantic.operations.len(), 11);
454        assert_eq!(bundle.parameter_surfaces.len(), 11);
455        assert_eq!(bundle.parameter_surfaces[0].surface.id(), "imhead");
456        assert_eq!(bundle.parameter_surfaces[10].surface.id(), "importfits");
457        for surface in &bundle.parameter_surfaces {
458            surface.validate().unwrap_or_else(|errors| {
459                panic!(
460                    "invalid embedded image-analysis surface {}: {errors:?}",
461                    surface.surface.id()
462                )
463            });
464        }
465        assert_eq!(
466            serde_json::to_value(&bundle).unwrap()["parameter_surfaces"]
467                .as_array()
468                .unwrap()
469                .len(),
470            11
471        );
472        assert_eq!(
473            bundle
474                .projections
475                .cli
476                .as_ref()
477                .unwrap()
478                .machine_actions
479                .json_run
480                .as_deref(),
481            Some("--json-run <SOURCE>")
482        );
483        assert_eq!(
484            bundle.projections.python.as_ref().unwrap()["module"],
485            "casars.tasks.image_analysis"
486        );
487    }
