336    /// Bind the Compiled Problem to one non-zero execution attempt and epoch.
337    pub fn bind(
338        problem: ExecutableModelProblem,
339        attempt: ModelExecutionAttemptId,
340        epoch: u64,
341        storage: ModelStoragePlan,
342    ) -> Result<Self, ModelLifecycleError> {
343        if attempt.identity().as_bytes() == [0; 32] || epoch == 0 {
344            return Err(ModelLifecycleError::InvalidExecutionBinding);
345        }
346        let authority = lifecycle_authority(
347            problem.problem_id(),
348            problem.model_lifecycle(),
349            attempt,
350            epoch,
351        );
352        let seal = next_authority_seal();
353        let ExecutableModelProblem { problem, prepared } = problem;
354        Ok(Self {
355            problem: problem.problem_id(),
356            contract: problem.model_lifecycle().clone(),
357            attempt,
358            epoch,
359            authority,
360            seal,
361            final_authority: Some(FinalAuthority(seal)),
362            continuation: None,
363            prepared,
364            storage,
365            next_generation: AtomicU64::new(1),
366        })
367    }

369    /// Bind the next execution attempt by consuming one completed generation.
370    ///
371    /// This is the sole cross-attempt model handoff. The previous completion
372    /// and generation stay inseparable until this method validates their
373    /// problem, generation, and private owner binding. The returned generation is
374    /// accepted only by this newly bound lifecycle and remains affine.
375    pub fn continue_from(
376        problem: ExecutableModelProblem,
377        attempt: ModelExecutionAttemptId,
378        epoch: u64,
379        continuation: FinalModelContinuation,
380        storage: ModelStoragePlan,
381    ) -> Result<(Self, ModelGeneration), ModelLifecycleError> {
382        let mut lifecycle = Self::bind(problem, attempt, epoch, storage)?;
383        let (completion, mut generation) = continuation.into_parts();
384        lifecycle.validate_generation_shape_and_bounds(&generation)?;
385        let completion_identity = final_completion_id(
386            generation.authority,
387            completion.problem,
388            completion.attempt,
389            completion.epoch,
390            completion.base,
391            completion.delta,
392            completion.generation,
393        );
394        if completion.problem != lifecycle.problem
395            || completion.generation != generation.generation_id
396            || completion.seal != generation.seal
397            || completion_identity != completion.completion_id
398            || generation.shape != *lifecycle.contract.target()
399        {
400            return Err(ModelLifecycleError::ForeignModelLifecycle);
401        }
402        lifecycle.continuation = Some(ContinuationAuthority {
403            seal: generation.seal,
404            generation: generation.generation_id,
405        });
406        generation
407            .samples
408            .record_validated_bound(lifecycle.contract.bounds().max_absolute_model_value());
409        Ok((lifecycle, generation))
410    }

446    /// Establish the compiled empty initial generation.
447    pub fn initial_empty(&self) -> Result<ModelGeneration, ModelLifecycleError> {
448        self.ensure_open()?;
449        if !matches!(self.contract.input(), ModelInputCommitment::Empty) {
450            return Err(ModelLifecycleError::InitialModelKindMismatch);
451        }
452        let zero = ModelValue::new(0.0)?;
453        let mut samples = self.storage.create(self.contract.target().sample_count())?;
454        let window = vec![ModelSample::valid(zero); samples.window_samples()];
455        for start in (0..samples.len()).step_by(window.len()) {
456            samples.write(start, &window[..window.len().min(samples.len() - start)])?;
457        }
458        self.mint_stored_generation(samples, ModelGenerationOrigin::Empty)
459    }

461    /// Consume one fallible aligned seed stream in exact canonical target order.
462    ///
463    /// The outer result preserves a source-reader failure. The inner result
464    /// reports the lifecycle contract failure after the source stream has been
465    /// consumed. Only the resulting generation buffer is retained; no complete
466    /// source array, coverage bitmap, or canonicalization copy is allocated.
467    pub fn ingest_aligned<E>(
468        &self,
469        source: LogicalIdentity,
470        source_shape: &ModelSourceShape,
471        samples: impl IntoIterator<Item = Result<ModelSample, E>>,
472    ) -> Result<Result<ModelGeneration, ModelLifecycleError>, E> {
473        if let Err(error) = self.ensure_open() {
474            return Ok(Err(error));
475        }
476        let ModelInputCommitment::AlignedSeed {
477            source: expected_source,
478            support,
479        } = self.contract.input()
480        else {
481            return Ok(Err(ModelLifecycleError::InitialModelKindMismatch));
482        };
483        if source != *expected_source || source_shape != self.contract.target() {
484            return Ok(Err(ModelLifecycleError::SourceProvenanceMismatch));
485        }
486        let storage = match self.storage.create(self.contract.target().sample_count()) {
487            Ok(storage) => storage,
488            Err(error) => return Ok(Err(error)),
489        };
490        let (samples, actual_support) = match store_exact_samples(
491            samples,
492            storage,
493            self.contract.bounds().max_absolute_model_value(),
494        )? {
495            Ok(samples) => samples,
496            Err(error) => return Ok(Err(error)),
497        };
498        Ok((|| {
499            if actual_support != *support {
500                return Err(ModelLifecycleError::SupportIdentityMismatch);
501            }
502            self.mint_stored_generation(
503                samples,
504                ModelGenerationOrigin::Ingested {
505                    source,
506                    reprojection: None,
507                },
508            )
509        })())
510    }

512    /// Consume one reconstruction-derived reprojection after checking its compiled evidence.
513    pub fn initial_reprojected(&mut self) -> Result<ModelGeneration, ModelLifecycleError> {
514        self.ensure_open()?;
515        let prepared = self
516            .prepared
517            .take()
518            .ok_or(ModelLifecycleError::OwnerPreparationRequired)?;
519        let ModelInputCommitment::ReprojectedSeed(commitment) = self.contract.input() else {
520            return Err(ModelLifecycleError::InitialModelKindMismatch);
521        };
522        if prepared.projection.source() != commitment.source()
523            || prepared.projection.source_shape() != commitment.source_shape()
524            || prepared.target_shape != *self.contract.target()
525            || prepared.projection.preparation_contract() != commitment.preparation_contract()
526            || prepared.bounds != self.contract.bounds()
527            || prepared.precision != self.contract.arithmetic_precision()
528        {
529            return Err(ModelLifecycleError::SourceProvenanceMismatch);
530        }
531        if prepared.projection != *commitment {
532            return Err(ModelLifecycleError::ReprojectionIdentityMismatch);
533        }
534        let source = prepared.projection.source();
535        let reprojection = ModelReprojectionId(prepared.projection.reprojection());
536        self.mint_generation(
537            prepared.samples.into_vec(),
538            ModelGenerationOrigin::Ingested {
539                source,
540                reprojection: Some(reprojection),
541            },
542        )
543    }

566    /// Validate and name one canonical sparse Model Delta.
567    ///
568    /// Terms must arrive in strictly increasing canonical cell order, which
569    /// removes the former full sorting/canonicalization allocation.
570    /// Delta derivation is independent of the lifecycle's one-shot final-model
571    /// completion authority: a completed Major Cycle may derive the bounded
572    /// T21 update that will be consumed by the next lifecycle attempt.
573    pub fn compile_delta(
574        &self,
575        base: &ModelGeneration,
576        terms: impl IntoIterator<Item = ModelDeltaTerm>,
577    ) -> Result<ModelDelta, ModelLifecycleError> {
578        self.validate_base(base)?;
579        self.compile_delta_with_support(base.generation_id(), terms, |index| {
580            Ok(base.samples.read(index..index + 1)?[0].support())
581        })
582    }
583
584    fn compile_delta_with_support(
585        &self,
586        base: ModelGenerationId,
587        terms: impl IntoIterator<Item = ModelDeltaTerm>,
588        mut support: impl FnMut(usize) -> Result<ModelSupport, ModelLifecycleError>,
589    ) -> Result<ModelDelta, ModelLifecycleError> {
590        let terms = terms.into_iter();
591        let capacity = terms.size_hint().0.min(
592            self.contract
593                .bounds()
594                .max_delta_terms()
595                .min(self.contract.target().sample_count()),
596        );
597        let mut canonical = Vec::with_capacity(capacity);
598        let mut prior = None;
599        for term in terms {
600            if canonical.len() == self.contract.bounds().max_delta_terms() {
601                return Err(ModelLifecycleError::DeltaTermBoundExceeded {
602                    terms: canonical.len() + 1,
603                    bound: self.contract.bounds().max_delta_terms(),
604                });
605            }
606            let index = self
607                .contract
608                .target()
609                .flat_index(term.cell())
610                .ok_or(ModelLifecycleError::CellOutsideShape)?;
611            if prior.is_some_and(|previous| previous >= index) {
612                return Err(ModelLifecycleError::NonCanonicalDelta);
613            }
614            prior = Some(index);
615            if term.increment().value() == 0.0 {
616                return Err(ModelLifecycleError::ZeroDeltaTerm);
617            }
618            validate_model_value(
619                term.increment(),
620                self.contract.bounds().max_absolute_delta_value(),
621            )
622            .map_err(|_| ModelLifecycleError::DeltaValueBoundExceeded)?;
623            if support(index)? != ModelSupport::Valid {
624                return Err(ModelLifecycleError::DeltaOutsideValidSupport);
625            }
626            canonical.push(term);
627        }
628        if canonical.is_empty() {
629            return Err(ModelLifecycleError::EmptyDelta);
630        }
631        let delta_id = ModelDeltaId(
632            NEXT_MODEL_DELTA
633                .try_update(Ordering::Relaxed, Ordering::Relaxed, |ordinal| {
634                    ordinal.checked_add(1)
635                })
636                .expect("model update event space exhausted"),
637        );
638        Ok(ModelDelta {
639            delta_id,
640            authority: self.authority,
641            seal: self.seal,
642            base,
643            terms: canonical.into_boxed_slice(),
644        })
645    }

673    /// Prepare one final-model candidate without consuming final-completion
674    /// authority.
675    ///
676    /// This is the first phase of the Major-Cycle transaction. All model and
677    /// delta association checks happen here. Sparse arithmetic is performed by
678    /// the first consumer of each model window; completion also resolves any
679    /// unvisited updates before consuming the lifecycle authority. A named generation
680    /// carried from an earlier attempt is rebound in place to this lifecycle
681    /// even when no Model Delta changes its samples.
682    pub fn prepare_final_model(
683        &self,
684        named: ModelGeneration,
685        delta: Option<ModelDelta>,
686    ) -> Result<PreparedFinalModel, ModelLifecycleError> {
687        self.ensure_open()?;
688        let base = named.generation_id;
689        let (mut generation, delta) = match delta {
690            Some(delta) => {
691                let delta_id = delta.delta_id;
692                (self.apply_delta_inner(named, delta)?, Some(delta_id))
693            }
694            None => {
695                let generation = self.adopt_generation(named)?;
696                (generation, None)
697            }
698        };
699        self.restore_later_domain_overlap(&mut generation)?;
700        Ok(PreparedFinalModel {
701            generation,
702            authority: self.authority,
703            seal: self.seal,
704            base,
705            delta,
706        })
707    }

709    /// Restore CASA's canonical multi-domain model ownership after an update.
710    ///
711    /// Later compiled domains own shared sky pixels. CASA omits those pixels from
712    /// earlier-domain prediction and restores the later values into earlier models
713    /// before publication and continuation.
714    fn restore_later_domain_overlap(
715        &self,
716        generation: &mut ModelGeneration,
717    ) -> Result<(), ModelLifecycleError> {
718        let shape = self.contract.target();
719        if shape.domains().len() < 2 {
720            return Ok(());
721        }
722        for earlier in (0..shape.domains().len() - 1).rev() {
723            let [width, height] = shape.domains()[earlier].pixels();
724            let target_coordinate = shape
725                .direction(earlier)
726                .ok_or(ModelLifecycleError::UnsupportedDirectionConversion)?;
727            for y in 0..height {
728                for x in 0..width {
729                    let mut owner = None;
730                    for later in earlier + 1..shape.domains().len() {
731                        let source_coordinate = shape
732                            .direction(later)
733                            .ok_or(ModelLifecycleError::UnsupportedDirectionConversion)?;
734                        if let Some(pixel) = crate::mask::reprojected_pixel(
735                            source_coordinate,
736                            shape.domains()[later].pixels(),
737                            target_coordinate,
738                            [x, y],
739                        )
740                        .map_err(|_| ModelLifecycleError::UnsupportedDirectionConversion)?
741                        {
742                            owner = Some((later, pixel));
743                        }
744                    }
745                    let Some((owner_domain, owner_pixel)) = owner else {
746                        continue;
747                    };
748                    for coefficient in 0..shape.coefficients() {
749                        for polarization in 0..shape.polarizations() {
750                            let target = shape
751                                .flat_index(ModelCell::new(
752                                    earlier,
753                                    coefficient,
754                                    polarization,
755                                    [x, y],
756                                ))
757                                .ok_or(ModelLifecycleError::CellOutsideShape)?;
758                            let source = shape
759                                .flat_index(ModelCell::new(
760                                    owner_domain,
761                                    coefficient,
762                                    polarization,
763                                    owner_pixel,
764                                ))
765                                .ok_or(ModelLifecycleError::CellOutsideShape)?;
766                            let sample =
767                                match generation.samples.read(target..target + 1)?[0].support() {
768                                    ModelSupport::Valid => ModelSample::valid(
769                                        generation.samples.read(source..source + 1)?[0].value(),
770                                    ),
771                                    ModelSupport::Invalid => ModelSample::invalid(),
772                                };
773                            generation.samples.write(target, &[sample])?;
774                        }
775                    }
776                }
777            }
778        }
779        Ok(())
780    }

782    /// Commit a successfully reconciled final-model candidate and mint its
783    /// distinct completion evidence.
784    pub fn commit_final_model(
785        &mut self,
786        mut prepared: PreparedFinalModel,
787    ) -> Result<FinalModelUpdate, ModelLifecycleError> {
788        self.ensure_open()?;
789        if prepared.authority != self.authority || prepared.seal != self.seal {
790            return Err(ModelLifecycleError::ForeignModelLifecycle);
791        }
792        self.validate_named_generation(&prepared.generation)?;
793        prepared.generation.samples.complete_updates()?;
794        let generation_id = prepared.generation.generation_id;
795        let completion_id = final_completion_id(
796            self.authority,
797            self.problem,
798            self.attempt,
799            self.epoch,
800            prepared.base,
801            prepared.delta,
802            generation_id,
803        );
804        let final_authority = self
805            .final_authority
806            .take()
807            .expect("open lifecycle retains final authority");
808        let completion = FinalModelCompletion {
809            completion_id,
810            seal: final_authority.0,
811            problem: self.problem,
812            attempt: self.attempt,
813            epoch: self.epoch,
814            base: prepared.base,
815            delta: prepared.delta,
816            generation: generation_id,
817        };
818        debug_assert_eq!(completion.seal, self.seal);
819        Ok(FinalModelUpdate {
820            generation: prepared.generation,
821            completion,
822        })
823    }

840    pub fn apply_final_delta(

842        base: ModelGeneration,

849    fn validate_delta_update(
850        &self,
851        base: &ModelGeneration,
852        delta: &ModelDelta,
853    ) -> Result<(), ModelLifecycleError> {
854        self.validate_base(base)?;
855        if delta.seal != self.seal
856            || delta.authority != self.authority
857            || delta.base != base.generation_id
858        {
859            return Err(ModelLifecycleError::DeltaBaseMismatch);
860        }
861        Ok(())

864    fn apply_delta_inner(
865        &self,
866        mut base: ModelGeneration,
867        delta: ModelDelta,
868    ) -> Result<ModelGeneration, ModelLifecycleError> {
869        self.validate_delta_update(&base, &delta)?;
870        let terms = delta.terms.iter().map(|term| ModelSampleUpdate {
871            index: self
872                .contract
873                .target()
874                .flat_index(term.cell())
875                .expect("validated delta cell remains in range"),
876            increment: term.increment().value(),
877        });
878        base.samples.queue_updates(
879            terms,
880            self.contract.arithmetic_precision(),
881            self.contract.bounds().max_absolute_model_value(),
882        )?;
883        self.mint_stored_generation(
884            base.samples,
885            ModelGenerationOrigin::Delta {
886                base: base.generation_id,
887                delta: delta.delta_id,
888            },
889        )
890    }

892    fn adopt_generation(
893        &self,
894        mut generation: ModelGeneration,
895    ) -> Result<ModelGeneration, ModelLifecycleError> {
896        self.validate_base(&generation)?;
897        if generation.authority != self.authority || generation.seal != self.seal {
898            generation.authority = self.authority;
899            generation.seal = self.seal;
900            generation.generation_id = self.next_generation_id();
901        }
902        Ok(generation)
903    }

905    fn validate_base(&self, generation: &ModelGeneration) -> Result<(), ModelLifecycleError> {
906        self.validate_generation_shape_and_bounds(generation)?;
907        if generation.shape != *self.contract.target() {
908            return Err(ModelLifecycleError::ForeignModelSpace);
909        }
910        if generation.seal == self.seal {
911            return Ok(());
912        }
913        if self.continuation.is_some_and(|continuation| {
914            continuation.seal == generation.seal
915                && continuation.generation == generation.generation_id
916        }) {
917            return Ok(());
918        }
919        if matches!(
920            self.contract.input(),
921            ModelInputCommitment::Generation(expected)
922                if generation.generation_id.identity() == *expected
923        ) {
924            Ok(())
925        } else {
926            Err(ModelLifecycleError::ForeignModelLifecycle)
927        }
928    }
929
930    fn validate_generation_shape_and_bounds(
931        &self,
932        generation: &ModelGeneration,
933    ) -> Result<(), ModelLifecycleError> {
934        generation.samples.finish_updates()?;
935        if generation.samples.len() != generation.shape.sample_count() {
936            return Err(ModelLifecycleError::GenerationIdentityMismatch);
937        }
938        // A tighter scientific bound is a new constraint, unlike a routine
939        // transfer of an already validated model. Overlap restoration can lower
940        // the recorded maximum, so validate rather than reject this transition.
941        if generation.samples.maximum_magnitude()
942            <= self.contract.bounds().max_absolute_model_value()
943        {
944            return Ok(());
945        }
946        generation.samples.for_each_window(|_, samples| {
947            for sample in samples {
948                if sample.support() == ModelSupport::Valid {
949                    validate_model_value(
950                        sample.value(),
951                        self.contract.bounds().max_absolute_model_value(),
952                    )?;
953                } else if sample.value().value() != 0.0 {
954                    return Err(ModelLifecycleError::InvalidSupportPayload);
955                }
956            }
957            Ok(())
958        })?;
959        Ok(())
960    }
961
962    fn mint_generation(
963        &self,
964        samples: Vec<ModelSample>,
965        origin: ModelGenerationOrigin,
966    ) -> Result<ModelGeneration, ModelLifecycleError> {
967        if samples.len() != self.contract.target().sample_count() {
968            return Err(ModelLifecycleError::SampleCountMismatch {
969                expected: self.contract.target().sample_count(),
970                actual: samples.len(),
971            });
972        }
973        let mut stored = self.storage.create(samples.len())?;
974        for (index, window) in samples.chunks(stored.window_samples()).enumerate() {
975            stored.write(index * stored.window_samples(), window)?;
976        }
977        self.mint_stored_generation(stored, origin)
978    }
979
980    fn mint_stored_generation(
981        &self,
982        samples: ModelSamples,
983        origin: ModelGenerationOrigin,
984    ) -> Result<ModelGeneration, ModelLifecycleError> {
985        let generation_id = self.next_generation_id();
986        Ok(ModelGeneration {
987            generation_id,
988            authority: self.authority,
989            seal: self.seal,
990            shape: self.contract.target().clone(),
991            samples,
992            origin,
993        })
994    }
995
996    fn next_generation_id(&self) -> ModelGenerationId {
997        let ordinal = self
998            .next_generation
999            .try_update(Ordering::Relaxed, Ordering::Relaxed, |ordinal| {
1000                ordinal.checked_add(1)
1001            })
1002            .expect("model generation ordinal exhausted");
1003        let mut encoder = Encoder::new(GENERATION_DOMAIN, GENERATION_VERSION);
1004        encoder.identity(self.authority.as_bytes());
1005        encoder.u64(self.seal.0);
1006        encoder.u64(ordinal);
1007        ModelGenerationId(LogicalIdentity::from_bytes(encoder.finish()))
1008    }

1010    fn ensure_open(&self) -> Result<(), ModelLifecycleError> {
1011        if self.final_authority.is_some() {
1012            Ok(())
1013        } else {
1014            Err(ModelLifecycleError::FinalModelAlreadyCompleted)
1015        }
1016    }

1019/// Exact reason model lifecycle validation failed closed.
1020#[derive(Debug, Clone, PartialEq, Eq, Error)]
1021pub enum ModelLifecycleError {
1022    /// An admitted model backing or window could not be accessed.
1023    #[error("model storage: {0}")]
1024    Storage(String),
1025    /// A raw reprojected compiler projection was presented without owner preparation.
1026    #[error("reprojected model execution requires reconstruction-owned preparation")]
1027    OwnerPreparationRequired,
1028    /// A model schema value was invalid.
1029    #[error(transparent)]
1030    Contract(#[from] ModelContractError),
1031    /// The execution attempt identity or generation epoch was zero.
1032    #[error("model lifecycle requires a non-zero execution attempt and epoch")]
1033    InvalidExecutionBinding,
1034    /// The requested initialization path differed from the compiled input commitment.
1035    #[error("model initialization does not match the compiled input commitment")]
1036    InitialModelKindMismatch,
1037    /// A source stream length differed from its shape.
1038    #[error("model stream requires {expected} samples but received {actual}")]
1039    SampleCountMismatch {
1040        /// Shape-derived sample count.
1041        expected: usize,
1042        /// Supplied sample count.
1043        actual: usize,
1044    },
1045    /// Source validity evidence differed from the Compiled Problem.
1046    #[error("model source validity evidence differs from the Compiled Problem")]
1047    SupportIdentityMismatch,
1048    /// Source artifact or typed coordinate/coefficient-space provenance differed.
1049    #[error("model source provenance differs from the Compiled Problem")]
1050    SourceProvenanceMismatch,
1051    /// A cell lay outside its typed source or target shape.
1052    #[error("model cell lies outside its declared shape")]
1053    CellOutsideShape,
1054    /// Source and target image-domain inventories cannot be paired exactly.
1055    #[error("model reprojection requires matching canonical image-domain inventories")]
1056    UnsupportedDomainMapping,
1057    /// Source and target direction laws require an unsupported frame or tangent-point conversion.
1058    #[error("model reprojection supports exact affine mapping within one direction tangent plane")]
1059    UnsupportedDirectionConversion,
1060    /// Source and target spectral coefficient bases cannot be mapped exactly.
1061    #[error("model reprojection cannot convert between the declared coefficient bases")]
1062    UnsupportedBasisConversion,
1063    /// A requested target polarization coordinate is absent from the source.
1064    #[error("model reprojection cannot derive the requested polarization coordinate")]
1065    UnsupportedPolarizationConversion,
1066    /// The mapping exceeded its compiled term ceiling.
1067    #[error("model reprojection has {terms} terms, exceeding bound {bound}")]
1068    ReprojectionTermBoundExceeded {
1069        /// Exact term count.
1070        terms: usize,
1071        /// Compiled term ceiling.
1072        bound: usize,
1073    },
1074    /// Reprojection weights did not sum to one in the committed arithmetic precision.
1075    #[error("reprojection stencil weights must sum to one")]
1076    UnnormalizedReprojectionStencil,
1077    /// Applied mapping evidence differed from the Compiled Problem.
1078    #[error("model reprojection identity differs from the Compiled Problem")]
1079    ReprojectionIdentityMismatch,
1080    /// A model value exceeded the lifecycle ceiling.
1081    #[error("model value exceeds the compiled lifecycle bound")]
1082    ModelValueBoundExceeded,
1083    /// A Model Delta contained no terms.
1084    #[error("a Model Delta requires at least one term")]
1085    EmptyDelta,
1086    /// A Model Delta exceeded its compiled term ceiling.
1087    #[error("Model Delta has {terms} terms, exceeding bound {bound}")]
1088    DeltaTermBoundExceeded {
1089        /// Exact term count observed before failure.
1090        terms: usize,
1091        /// Compiled term ceiling.
1092        bound: usize,
1093    },
1094    /// Delta terms were not unique and strictly canonical.
1095    #[error("Model Delta terms must be unique and in canonical cell order")]
1096    NonCanonicalDelta,
1097    /// A Model Delta term exceeded its value ceiling.
1098    #[error("Model Delta term exceeds the compiled lifecycle bound")]
1099    DeltaValueBoundExceeded,
1100    /// A Model Delta term had no numerical update.
1101    #[error("Model Delta terms must be non-zero")]
1102    ZeroDeltaTerm,
1103    /// A Model Delta attempted to create a value outside valid support.
1104    #[error("Model Delta terms may update only valid model support")]
1105    DeltaOutsideValidSupport,
1106    /// A generation belonged to another model space.
1107    #[error("model generation belongs to a different model space")]
1108    ForeignModelSpace,
1109    /// A generation belonged to a separately constructed lifecycle owner.
1110    #[error("model generation belongs to a different lifecycle owner")]
1111    ForeignModelLifecycle,
1112    /// A generation did not match the named input or its declared shape.
1113    #[error("model generation does not match its named input or shape")]
1114    GenerationIdentityMismatch,
1115    /// Invalid support carried a numeric payload.
1116    #[error("invalid model support may not carry a numeric value")]
1117    InvalidSupportPayload,
1118    /// A Model Delta did not name this exact base and owner.
1119    #[error("Model Delta does not name this base generation and owner")]
1120    DeltaBaseMismatch,
1121    /// The lifecycle's affine final-completion authority was already consumed.
1122    #[error("final-model completion authority has already been consumed")]
1123    FinalModelAlreadyCompleted,
1124}

1149#[allow(clippy::too_many_arguments)]
1150fn final_completion_id(
1151    authority: LogicalIdentity,
1152    problem: CompiledProblemId,
1153    attempt: ModelExecutionAttemptId,
1154    epoch: u64,
1155    base: ModelGenerationId,
1156    delta: Option<ModelDeltaId>,
1157    generation: ModelGenerationId,
1158) -> FinalModelCompletionId {
1159    let mut encoder = Encoder::new(FINAL_COMPLETION_DOMAIN, FINAL_COMPLETION_VERSION);
1160    encoder.identity(authority.as_bytes());
1161    encoder.identity(problem.as_bytes());
1162    encoder.identity(attempt.identity().as_bytes());
1163    encoder.u64(epoch);
1164    encoder.identity(base.as_bytes());
1165    match delta {
1166        None => encoder.u8(0),
1167        Some(delta) => {
1168            encoder.u8(1);
1169            encoder.u64(delta.ordinal());
1170        }
1171    }
1172    encoder.identity(generation.as_bytes());
1173    FinalModelCompletionId(LogicalIdentity::from_bytes(encoder.finish()))
1174}

1176fn store_exact_samples<E>(
1177    samples: impl IntoIterator<Item = Result<ModelSample, E>>,
1178    mut storage: ModelSamples,
1179    bound: f64,
1180) -> Result<Result<(ModelSamples, LogicalIdentity), ModelLifecycleError>, E> {
1181    let expected = storage.len();
1182    let mut support = casa_imaging_model::ModelSourceSupportInspection::new();
1183    let mut values = Vec::with_capacity(storage.window_samples());
1184    let mut written = 0;
1185    let mut iterator = samples.into_iter();
1186    for index in 0..expected {
1187        let sample = match iterator.next() {
1188            None => {
1189                return Ok(Err(ModelLifecycleError::SampleCountMismatch {
1190                    expected,
1191                    actual: index,
1192                }));
1193            }
1194            Some(Err(error)) => return Err(error),
1195            Some(Ok(sample)) => sample,
1196        };
1197        if sample.support() == ModelSupport::Valid {
1198            if let Err(error) = validate_model_value(sample.value(), bound) {
1199                return Ok(Err(error));
1200            }
1201        } else if sample.value().value() != 0.0 {
1202            return Ok(Err(ModelLifecycleError::InvalidSupportPayload));
1203        }
1204        support.push(sample.support());
1205        values.push(sample);
1206        if values.len() == storage.window_samples() {
1207            if let Err(error) = storage.write(written, &values) {
1208                return Ok(Err(error));
1209            }
1210            written += values.len();
1211            values.clear();
1212        }
1213    }
1214    match iterator.next() {
1215        None => {}
1216        Some(Err(error)) => return Err(error),
1217        Some(Ok(_)) => {
1218            return Ok(Err(ModelLifecycleError::SampleCountMismatch {
1219                expected,
1220                actual: expected + 1,
1221            }));
1222        }
1223    }
1224    if !values.is_empty()
1225        && let Err(error) = storage.write(written, &values)
1226    {
1227        return Ok(Err(error));
1228    }
1229    Ok(Ok((storage, support.finish())))
1230}

1232pub(crate) fn validate_model_value(
1233    value: ModelValue,
1234    bound: f64,
1235) -> Result<(), ModelLifecycleError> {
1236    if value.value().abs() > bound {
1237        Err(ModelLifecycleError::ModelValueBoundExceeded)
1238    } else {
1239        Ok(())
1240    }
