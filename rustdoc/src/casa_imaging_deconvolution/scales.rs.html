11/// Real two-dimensional transforms of one plane shape, in double precision
12/// on one thread.
13pub(crate) struct PlaneFft {
14    shape: PlaneShape,
15    fft: RealFft2<f64>,
16}
17
18impl PlaneFft {
19    pub(crate) fn new(shape: PlaneShape) -> Result<Self, Error> {
20        Ok(Self {
21            shape,
22            fft: RealFft2::with_threads([shape.nx, shape.ny], 1)?,
23        })
24    }
25
26    /// The half spectrum of `plane`.
27    pub(crate) fn forward(&mut self, plane: &[f64]) -> Result<Vec<Complex64>, Error> {
28        let mut storage = vec![Complex64::default(); self.fft.storage_len()];
29        let stride = self.fft.real_row_stride();
30        let real: &mut [f64] = bytemuck::cast_slice_mut(&mut storage);
31        for x in 0..self.shape.nx {
32            real[x * stride..][..self.shape.ny]
33                .copy_from_slice(&plane[x * self.shape.ny..][..self.shape.ny]);
34        }
35        self.fft.forward(&mut storage)?;
36        Ok(storage)
37    }
38
39    /// The plane of a half spectrum, normalised by the pixel count.
40    pub(crate) fn inverse(&mut self, mut spectrum: Vec<Complex64>) -> Result<Vec<f64>, Error> {
41        self.fft.inverse(&mut spectrum)?;
42        let stride = self.fft.real_row_stride();
43        let scale = 1.0 / self.shape.len() as f64;
44        let real: &[f64] = bytemuck::cast_slice(&spectrum);
45        let mut plane = vec![0.0; self.shape.len()];
46        for x in 0..self.shape.nx {
47            for (value, &sample) in plane[x * self.shape.ny..][..self.shape.ny]
48                .iter_mut()
49                .zip(&real[x * stride..][..self.shape.ny])
50            {
51                *value = sample * scale;
52            }
53        }
54        Ok(plane)
55    }
56}

58/// `MatrixCleaner::spheroidal`: the rational approximation to the
59/// prolate spheroidal function the scales are tapered by.
60#[must_use]
61pub fn spheroidal(nu: f64) -> f64 {
62    if nu <= 0.0 {
63        return 1.0;
64    }
65    if nu >= 1.0 {
66        return 0.0;
67    }
68    let (p, q, end) = if nu < 0.75 {
69        (
70            [
71                8.203343e-2,
72                -3.644705e-1,
73                6.278660e-1,
74                -5.335581e-1,
75                2.312756e-1,
76            ],
77            [1.0, 8.212018e-1, 2.078043e-1],
78            0.75,
79        )
80    } else {
81        (
82            [
83                4.028559e-3,
84                -3.697768e-2,
85                1.021332e-1,
86                -1.201436e-1,
87                6.412774e-2,
88            ],
89            [1.0, 9.599102e-1, 2.918724e-1],
90            1.0,
91        )
92    };
93    let delta = nu * nu - end * end;
94    let top = p
95        .iter()
96        .rev()
97        .fold(0.0, |sum, coefficient| sum * delta + coefficient);
98    let bottom = q
99        .iter()
100        .rev()
101        .fold(0.0, |sum, coefficient| sum * delta + coefficient);
102    if bottom == 0.0 { 0.0 } else { top / bottom }
103}

105/// One scale function: `(1 − r²)·spheroidal(r)` inside radius `size`
106/// pixels, normalised to unit sum, as its nonzero offsets from the centre.
107#[derive(Clone, Debug, PartialEq)]
108pub(crate) struct ScaleFunction {
109    /// The scale size in pixels.
110    pub(crate) size: f64,
111    /// `(dx, dy, value)` of every nonzero sample.
112    pub(crate) samples: Vec<(isize, isize, f64)>,
113}
114
115impl ScaleFunction {
116    /// `MatrixCleaner::makeScale` centred at `centre` of `shape`: a delta for
117    /// size 0, otherwise the tapered disc sampled within the plane.
118    pub(crate) fn new(size: f64, shape: PlaneShape) -> Self {
119        let [cx, cy] = shape.centre();
120        if size == 0.0 {
121            return Self {
122                size,
123                samples: vec![(0, 0, 1.0)],
124            };
125        }
126        let low = |centre: usize| ((centre as f64 - size) as isize).max(0);
127        let high = |centre: usize, extent: usize| {
128            ((centre as f64 + size) as isize).min(extent as isize - 1)
129        };
130        let mut samples = Vec::new();
131        let mut volume = 0.0;
132        for y in low(cy)..=high(cy, shape.ny) {
133            let y_part = ((cy as f64 - y as f64) / size).powi(2);
134            for x in low(cx)..=high(cx, shape.nx) {
135                let r2 = y_part + ((cx as f64 - x as f64) / size).powi(2);
136                if r2 < 1.0 {
137                    let value = (1.0 - r2) * spheroidal(r2.max(0.0).sqrt());
138                    volume += value;
139                    samples.push((x - cx as isize, y - cy as isize, value));
140                }
141            }
142        }
143        for sample in &mut samples {
144            sample.2 /= volume;
145        }
146        samples.retain(|sample| sample.2 != 0.0);
147        Self { size, samples }
148    }
149
150    /// The function as a plane centred at the origin, wrapping negative
151    /// offsets, so that a product of transforms is a centred convolution.
152    fn wrapped(&self, shape: PlaneShape) -> Vec<f64> {
153        let mut plane = vec![0.0; shape.len()];
154        for &(dx, dy, value) in &self.samples {
155            let x = dx.rem_euclid(shape.nx as isize) as usize;
156            let y = dy.rem_euclid(shape.ny as isize) as usize;
157            plane[shape.index(x, y)] += value;
158        }
159        plane
160    }
161}

163/// The scales of one solve with their transforms and small-scale biases.
164pub(crate) struct ScaleBank {
165    shape: PlaneShape,
166    pub(crate) functions: Vec<ScaleFunction>,
167    pub(crate) bias: Vec<f64>,
168    transforms: Vec<Vec<Complex64>>,
169    pub(crate) fft: PlaneFft,
170}
171
172impl ScaleBank {
173    /// The scales `sizes` (ascending) with CASA's bias
174    /// `1 − bias·size/largest` (1 for a single scale); `small_scale_bias`
175    /// is clamped to [−1, 1] as `SDAlgorithmMSClean` does.
176    pub(crate) fn new(
177        sizes: &[f64],
178        small_scale_bias: f64,
179        shape: PlaneShape,
180    ) -> Result<Self, Error> {
181        let small_scale_bias = small_scale_bias.clamp(-1.0, 1.0);
182        let largest = sizes.last().copied().unwrap_or(0.0);
183        let bias = sizes
184            .iter()
185            .map(|size| {
186                if sizes.len() > 1 {
187                    1.0 - small_scale_bias * size / largest
188                } else {
189                    1.0
190                }
191            })
192            .collect();
193        let functions = sizes
194            .iter()
195            .map(|&size| ScaleFunction::new(size, shape))
196            .collect::<Vec<_>>();
197        let mut fft = PlaneFft::new(shape)?;
198        let transforms = functions
199            .iter()
200            .map(|function| fft.forward(&function.wrapped(shape)))
201            .collect::<Result<Vec<_>, _>>()?;
202        Ok(Self {
203            shape,
204            functions,
205            bias,
206            transforms,
207            fft,
208        })
209    }
210
211    /// Number of scales.
212    pub(crate) fn len(&self) -> usize {
213        self.functions.len()
214    }
215
216    /// `spectrum` times the transforms of the listed scales, back in the
217    /// image domain: the centred circular convolution with each.
218    pub(crate) fn convolve(
219        &mut self,
220        spectrum: &[Complex64],
221        scales: &[usize],
222    ) -> Result<Vec<f64>, Error> {
223        let mut product = spectrum.to_vec();
224        for &scale in scales {
225            for (value, kernel) in product.iter_mut().zip(&self.transforms[scale]) {
226                *value *= kernel;
227            }
228        }
229        self.fft.inverse(product)
230    }
231
232    /// The support convolved with each scale, above `threshold`, with the
233    /// `border(size)` pixels at every edge removed (`makeScaleMasks`,
234    /// `setupUserMask`). `skip_point_border` keeps the edge of the point
235    /// scale, as the multi-term cleaner does.
236    pub(crate) fn masks(
237        &mut self,
238        support: &Support,
239        threshold: f64,
240        skip_point_border: bool,
241    ) -> Result<Vec<Support>, Error> {
242        let shape = self.shape;
243        let plane = support
244            .as_slice()
245            .iter()
246            .map(|supported| f64::from(u8::from(*supported)))
247            .collect::<Vec<_>>();
248        let spectrum = self.fft.forward(&plane)?;
249        (0..self.len())
250            .map(|scale| {
251                let smoothed = self.convolve(&spectrum, &[scale])?;
252                let border = if skip_point_border && scale == 0 {
253                    None
254                } else {
255                    Some((self.functions[scale].size * 1.5) as usize)
256                };
257                let pixels = smoothed
258                    .iter()
259                    .enumerate()
260                    .map(|(index, value)| {
261                        let [x, y] = shape.pixel(index);
262                        let inside = border.is_none_or(|border| {
263                            x > border
264                                && y > border
265                                && x + border + 1 < shape.nx
266                                && y + border + 1 < shape.ny
267                        });
268                        inside && *value > threshold
269                    })
270                    .collect();
271                Ok(Support::new(shape, pixels))
272            })
273            .collect()
274    }
275}

281    #[test]
282    fn scale_functions_have_unit_sum_and_the_casa_radius() {
283        let shape = PlaneShape::new(64, 48);
284        for size in [0.0, 1.5, 6.0, 10.0] {
285            let function = ScaleFunction::new(size, shape);
286            let sum = function.samples.iter().map(|sample| sample.2).sum::<f64>();
287            assert!((sum - 1.0).abs() < 1e-12, "size {size}: sum {sum}");
288            assert!(
289                function
290                    .samples
291                    .iter()
292                    .all(|&(dx, dy, _)| ((dx * dx + dy * dy) as f64) < size * size || size == 0.0)
293            );
294        }
295        assert_eq!(spheroidal(0.0), 1.0);
296        assert_eq!(spheroidal(1.0), 0.0);
297        // Continuous across the 0.75 break.
298        assert!((spheroidal(0.749_999_9) - spheroidal(0.75)).abs() < 1e-5);
299    }

301    /// Transform products are centred convolutions: a delta convolved with a
302    /// scale is the scale centred on the delta, wrapping at the edges.
303    #[test]
304    fn convolution_is_centred_and_circular() {
305        let shape = PlaneShape::new(15, 12);
306        let mut bank = ScaleBank::new(&[0.0, 3.0], 0.6, shape).unwrap();
307        assert_eq!(bank.bias, vec![1.0, 0.4]);
308        let mut delta = vec![0.0; shape.len()];
309        delta[shape.index(1, 10)] = 2.0;
310        let spectrum = bank.fft.forward(&delta).unwrap();
311        let convolved = bank.convolve(&spectrum, &[1]).unwrap();
312        let mut expected = vec![0.0; shape.len()];
313        for &(dx, dy, value) in &bank.functions[1].samples {
314            let x = (1 + dx).rem_euclid(15) as usize;
315            let y = (10 + dy).rem_euclid(12) as usize;
316            expected[shape.index(x, y)] += 2.0 * value;
317        }
318        for (a, e) in convolved.iter().zip(&expected) {
319            assert!((a - e).abs() < 1e-12);
320        }
321    }

323    #[test]
324    fn scale_masks_drop_the_border_and_partial_overlaps() {
325        let shape = PlaneShape::new(32, 32);
326        let mut bank = ScaleBank::new(&[0.0, 4.0], 0.0, shape).unwrap();
327        let support = Support::new(
328            shape,
329            (0..shape.len())
330                .map(|index| {
331                    let [x, y] = shape.pixel(index);
332                    (8..24).contains(&x) && (8..24).contains(&y)
333                })
334                .collect(),
335        );
336        let masks = bank.masks(&support, 0.9, false).unwrap();
337        // The point scale keeps the box; a full-image support loses its ring.
338        assert_eq!(masks[0], support);
339        let full = bank.masks(&Support::full(shape), 0.9, false).unwrap();
340        assert!(!full[0].contains(shape.index(0, 5)) && full[0].contains(shape.index(1, 1)));
341        assert!(bank.masks(&Support::full(shape), 0.1, true).unwrap()[0].contains(0));
342        // The 4-pixel scale needs 90% of its weight inside the box.
343        assert!(masks[1].contains(shape.index(16, 16)));
344        assert!(!masks[1].contains(shape.index(8, 16)));
345        assert!(masks[1].count() < support.count());
346    }
