2//! Visibility prediction from an external sky model, as CASA's simulator
3//! tool predicts it (`sm.predict` with its default `GridFT`, or with
4//! `WProjectFT` after `setoptions(ftmachine='wproject')`).
5//!
6//! CASA's `Simulator::predict` degrids the model through the FTMachine
7//! `setoptions` chose ([`FtMachine`]):
8//! - `GridFT`'s grid is padded by `setoptions(padding=…)`, 1.3 by default.
9//!   `WProjectFT`'s is padded by 1, its constructor's default, and degrids
10//!   through `wprojplanes` W-projection planes.
11//! - The tangent point is the direction of pixel `(nx/2, ny/2)`, whatever
12//!   the image's reference pixel is (`FTMachine::initMaps`), and every row is
13//!   rotated to it (`FTMachine::rotateUVW`).
14//! - Each MeasurementSet channel takes the nearest model plane, with no
15//!   frequency-frame conversion.
16//! - `GridFT` predicts autocorrelations, as the model's zero spacing (its
17//!   `usezero` is on by default); `WProjectFT` is built with `usezero` off
18//!   and predicts them as zero.
19//! - After `setvp(dovp=True)`, a `VPSkyJones` multiplies the model by the
20//!   primary beam's power pattern in the image plane, centred on the
21//!   pointing.
22//!
23//! The model is prepared on the imaging measurement operator, one plane at
24//! a time ([`SkyModel`] reads a plane, the beam multiplies it, the operator
25//! transforms it into the plane's grids):
26//! - the spheroidal or W-projection kernel set on a
27//!   [`GridPadding::CasaComposite`] grid;
28//! - a channel-local basis with nearest mapping, predicting
29//!   autocorrelations through `GridFT`;
30//! - the beam as the plane's prescale factor.
31//!
32//! The session predicts the rows it writes from the same preparation, in
33//! blocks on the worker team; [`ModelPredictor`] predicts one row at a time
34//! and is the reference the session is tested against.
35//!
36//! The model must be Stokes I, J2000 and in the axis order `simobserve`
37//! writes.

59/// A failure to read a sky model or to predict from it.
60#[derive(Debug, Error)]
61pub enum PredictError {
62    /// The model image could not be read.
63    #[error("model image {path}: {source}")]
64    Image {
65        /// The image path.
66        path: String,
67        /// The image error.
68        #[source]
69        source: ImageError,
70    },
71    /// A pixel could not be converted to a world coordinate, or back.
72    #[error("model image coordinates: {0}")]
73    Coordinates(#[from] CoordinateError),
74    /// The model image has a shape or coordinate system this predictor does
75    /// not support yet.
76    #[error("unsupported model image: {0}")]
77    UnsupportedModel(String),
78    /// The measurement operator refused the model or a row.
79    #[error(transparent)]
80    Operator(#[from] OperatorError),
81}

83/// The FTMachine a prediction degrids through (`sm.setoptions(ftmachine=…)`,
84/// `Simulator::createSkyEquation`).
85#[derive(Clone, Copy, Debug, PartialEq, Eq)]
86pub enum FtMachine {
87    /// `GridFT` (`ftmachine='ft'`, the default): the spheroidal kernel on a
88    /// grid padded by `setoptions(padding=…)`, predicting autocorrelations
89    /// as the model's zero spacing.
90    Grid(PaddingFactor),
91    /// `WProjectFT` (`ftmachine='wproject'`) with `wprojplanes` planes, at
92    /// least two, reaching `|w| = 0.25/|Δx|`. CASA's simulator builds it
93    /// with its constructor's padding of 1, whatever `setoptions(padding=…)`
94    /// says, and with `usezero` false, so autocorrelations predict zero.
95    WProject(u32),
96}
97
98impl FtMachine {
99    /// Whether the Metal backend predicts through this machine: `GridFT`'s
100    /// spheroidal set only.
101    #[must_use]
102    pub const fn runs_on_metal(self) -> bool {
103        matches!(self, Self::Grid(_))
104    }
105}

107/// The grid, polarization routing and spectral resampler of a model's
108/// operator for one FTMachine, before its kernels are built: the session
109/// admits the kernels from it ([`Self::geometry`]) and builds them in stages,
110/// as imaging does; [`Self::kernels`] builds them in one.
111pub(crate) struct ModelLayout {
112    machine: FtMachine,
113    geometry: GridGeometry,
114    polarization: PolarizationRouting,
115    resampler: SpectralResampler,
116}
117
118impl ModelLayout {
119    /// The layout for `model` on rows with `correlations` through `machine`.
120    ///
121    /// # Errors
122    ///
123    /// A geometry the grid cannot hold, or correlations the Stokes I model
124    /// cannot feed.
125    pub(crate) fn new(
126        model: &SkyModel,
127        correlations: &[CorrelationType],
128        machine: FtMachine,
129    ) -> Result<Self, PredictError> {
130        let [nx, ny] = model.shape();
131        let padding = match machine {
132            FtMachine::Grid(padding) => padding,
133            FtMachine::WProject(_) => PaddingFactor::SIMULATOR_WPROJECT,
134        };
135        let geometry = GridGeometry::new(
136            ImageExtent {
137                shape: [nx, ny],
138                increment_rad: model.increment_rad,
139                reference_pixel: [nx / 2, ny / 2],
140            },
141            GridPadding::CasaComposite(padding),
142        )?;
143        let polarization =
144            PolarizationRouting::compile(correlations, &[PolarizationCoordinate::StokesI])?;
145        let planes = u32::try_from(model.planes())
146            .map_err(|_| PredictError::UnsupportedModel("too many model planes".into()))?;
147        let axis = SpectralAxis::new(
148            model.first_frequency_hz,
149            model.frequency_increment_hz,
150            planes,
151        )?;
152        let resampler = SpectralResampler::channel_local(axis, SpectralKernel::Nearest);
153        let resampler = match machine {
154            FtMachine::Grid(_) => resampler.predicting_autocorrelations(),
155            FtMachine::WProject(_) => resampler,
156        };
157        Ok(Self {
158            machine,
159            geometry,
160            polarization,
161            resampler,
162        })
163    }
164
165    /// The FTMachine.
166    pub(crate) const fn machine(&self) -> FtMachine {
167        self.machine
168    }
169
170    /// The padded grid the kernels are built for.
171    pub(crate) const fn geometry(&self) -> &GridGeometry {
172        &self.geometry
173    }
174
175    /// The polarization routing the kernels are built for.
176    pub(crate) const fn polarization(&self) -> &PolarizationRouting {
177        &self.polarization
178    }
179
180    /// The machine's kernel set, built in one step.
181    ///
182    /// # Errors
183    ///
184    /// The W-projection planes cannot be built.
185    pub(crate) fn kernels(&self) -> Result<Box<dyn ConvolutionFunctionSet>, PredictError> {
186        Ok(match self.machine {
187            FtMachine::Grid(_) => Box::new(Spheroidal::new(&self.geometry, &self.polarization)),
188            FtMachine::WProject(planes) => Box::new(WPlanes::new(
189                &self.geometry,
190                &self.polarization,
191                WPlaneCount::Fixed(planes),
192            )?),
193        })
194    }
195}

197/// The measurement operator and spectral resampler that predict one model
198/// on rows of one correlation set, as CASA's `GridFT` or `WProjectFT` does,
199/// and the preparation of the model's grids. The session and
200/// [`ModelPredictor`] share it, so both predict from the same grids.
201pub(crate) struct ModelOperator {
202    pub(crate) operator: MeasurementOperator,
203    pub(crate) resampler: SpectralResampler,
204    machine: FtMachine,
205}
206
207impl ModelOperator {
208    /// The operator of `layout` with `kernels` (its machine's set), in
209    /// `precision`.
210    pub(crate) fn new(
211        layout: ModelLayout,
212        kernels: Box<dyn ConvolutionFunctionSet>,
213        precision: GridPrecision,
214    ) -> Self {
215        let ModelLayout {
216            machine,
217            geometry,
218            polarization,
219            resampler,
220        } = layout;
221        let operator = MeasurementOperator::new(
222            geometry,
223            resampler.basis(),
224            polarization,
225            kernels,
226            precision,
227        );
228        Self {
229            operator,
230            resampler,
231            machine,
232        }
233    }
234
235    /// The FTMachine the operator degrids through.
236    pub(crate) const fn machine(&self) -> FtMachine {
237        self.machine
238    }
239
240    /// The planes of the model's grids: every plane of the basis.
241    fn planes(&self) -> PlaneRange {
242        PlaneRange::new(0, self.operator.basis().planes())
243    }
244
245    /// Bytes one set of the model's grids holds
246    /// ([`PreparedModelGrids::bytes`]), known before they are allocated.
247    pub(crate) fn grid_bytes(&self) -> u64 {
248        self.operator
249            .accumulator_layout(self.planes(), None, ModeSet::DATA)
250            .bytes(self.operator.precision()) as u64
251    }
252
253    /// Bytes one worker holds while it prepares a plane, beside the grids:
254    /// the plane read and its slice, the beam's power on it, and the
255    /// transform's padded plane.
256    pub(crate) fn plane_scratch_bytes(&self, model: &SkyModel) -> u64 {
257        let [nx, ny] = model.shape();
258        let [gx, gy] = self.operator.geometry().grid_shape();
259        let cell = match self.operator.precision() {
260            GridPrecision::F32 => size_of::<Complex32>(),
261            GridPrecision::F64 => 2 * size_of::<Complex32>(),
262        };
263        model.plane_bytes() + (nx * ny * size_of::<f32>() + 2 * gx * gy * cell) as u64
264    }
265
266    /// The model's grids: each plane read, multiplied by `beam`'s power
267    /// pattern centred on its pixel when there is one, and transformed into
268    /// the plane's grids. Planes are prepared on `team`'s workers, one plane
269    /// per job, or in turn on the caller without one; the grids do not
270    /// depend on which.
271    ///
272    /// # Errors
273    ///
274    /// A plane that cannot be read, or the operator's refusal.
275    pub(crate) fn prepare(
276        &self,
277        model: &SkyModel,
278        beam: Option<(&BeamPattern, [f64; 2])>,
279        team: Option<&WorkerTeam>,
280    ) -> Result<PreparedModelGrids, PredictError> {
281        let mut grids = self.operator.empty_model_grids(self.planes())?;
282        let mut jobs: Vec<Option<ModelPlaneGrids<'_>>> =
283            grids.planes_mut().into_iter().map(Some).collect();
284        let prepare = |plane: usize, target: &mut Option<ModelPlaneGrids<'_>>| {
285            let Some(target) = target.take() else {
286                return Ok(());
287            };
288            let images = vec![model.read_plane(plane)?];
289            let factor = beam.map(|(pattern, centre)| pattern.power_plane(model, plane, centre));
290            self.operator
291                .fill_model_plane(&ModelPlane { images }, factor.as_ref(), target)?;
292            Ok::<_, PredictError>(())
293        };
294        match team {
295            Some(team) => team.for_each_mut(&mut jobs, prepare)?,
296            None => jobs
297                .iter_mut()
298                .enumerate()
299                .try_for_each(|(plane, target)| prepare(plane, target))?,
300        }
301        drop(jobs);
302        debug_assert_eq!(grids.bytes(), self.grid_bytes());
303        Ok(grids)
304    }
305}

307/// How [`ModelPredictor`] grids, as `sm.setoptions` and `sm.setvp` choose.
308#[derive(Clone, Copy, Debug, PartialEq)]
309pub struct PredictOptions {
310    /// The FTMachine, with `GridFT`'s padding (`setoptions(ftmachine=…,
311    /// padding=…, wprojplanes=…)`).
312    pub ft_machine: FtMachine,
313    /// The primary beam, when `setvp(dovp=True)`.
314    pub primary_beam: Option<PrimaryBeam>,
315    /// Grid arithmetic. CASA's simulator FTMachines run in `Float`.
316    pub precision: GridPrecision,
317}
318
319impl Default for PredictOptions {
320    fn default() -> Self {
321        Self {
322            ft_machine: FtMachine::Grid(PaddingFactor::SIMULATOR),
323            primary_beam: None,
324            precision: GridPrecision::F32,
325        }
326    }
327}

329/// One MeasurementSet row to predict.
330#[derive(Clone, Copy, Debug)]
331pub struct PredictRow<'a> {
332    /// Baseline coordinates in metres towards the model's tangent point, in
333    /// GridFT's convention: the row's `UVW` rotated by
334    /// [`casa_ms::derived::GridFtProjection`].
335    pub uvw_m: [f64; 3],
336    /// GridFT's phase shift in metres (`−dphase`), from the same rotation.
337    pub phase_shift_m: f64,
338    /// Channel frequencies in Hz.
339    pub frequencies_hz: &'a [f64],
340    /// Sample flags, `[channel][correlation]`.
341    pub flags: &'a [bool],
342    /// The row flag.
343    pub row_flag: bool,
344    /// Antenna pair; an autocorrelation predicts the model's zero spacing
345    /// through `GridFT`, and zero through `WProjectFT`.
346    pub antennas: [u32; 2],
347}

349/// Reusable per-thread buffers for [`ModelPredictor::predict_row`].

358/// A model prepared for prediction on one correlation set, predicting a row
359/// at a time on the CPU: the reference the session's block prediction is
360/// tested against.
361pub struct ModelPredictor {
362    model: ModelOperator,
363    grids: PreparedModelGrids,
364}
365
366impl ModelPredictor {
367    /// Prepare `model` for rows with `correlations`, its primary beam (when
368    /// requested) centred on `pointing_rad`.
369    ///
370    /// # Errors
371    ///
372    /// A geometry the grid cannot hold, correlations the Stokes I model
373    /// cannot feed, a pointing outside the model's coordinate system, or a
374    /// plane that cannot be read.
375    pub fn new(
376        model: &SkyModel,
377        correlations: &[CorrelationType],
378        pointing_rad: [f64; 2],
379        options: &PredictOptions,
380    ) -> Result<Self, PredictError> {
381        let layout = ModelLayout::new(model, correlations, options.ft_machine)?;
382        let kernels = layout.kernels()?;
383        let operator = ModelOperator::new(layout, kernels, options.precision);
384        let pattern = options.primary_beam.map(PrimaryBeam::pattern);
385        let beam = match &pattern {
386            Some(pattern) => Some((pattern, model.pixel_of(pointing_rad)?)),
387            None => None,
388        };
389        let grids = operator.prepare(model, beam, None)?;
390        Ok(Self {
391            model: operator,
392            grids,
393        })
394    }
395
396    /// Correlations per sample.
397    #[must_use]
398    pub fn correlations(&self) -> usize {
399        self.model.operator.polarization().correlations().len()
400    }
401
402    /// The model visibility of every sample of `row`, written to `out` as
403    /// `[channel][correlation]`. Flagged samples and flagged rows predict
404    /// zero; an autocorrelation predicts the model's zero spacing through
405    /// `GridFT` and zero through `WProjectFT`.
406    ///
407    /// # Errors
408    ///
409    /// A row inconsistent with the predictor, or an `out` of the wrong
410    /// length.
411    pub fn predict_row(
412        &self,
413        workspace: &mut PredictionWorkspace,
414        row: &PredictRow<'_>,
415        out: &mut [Complex32],
416    ) -> Result<(), PredictError> {
417        let cells = row.frequencies_hz.len() * self.correlations();
418        workspace.values.resize(cells, Complex32::new(0.0, 0.0));
419        workspace.weights.resize(cells, 1.0);
420        let native = NativeRow {
421            uvw_m: row.uvw_m,
422            phase_shift_m: row.phase_shift_m,
423            pointing_offset_rad: [0.0; 2],
424            frequencies_hz: row.frequencies_hz,
425            values: &workspace.values,
426            weights: &workspace.weights,
427            flags: row.flags,
428            row_flag: row.row_flag,
429            context: RowContext {
430                original_uvw_m: None,
431                time_s: 0.0,
432                antennas: row.antennas,
433                antenna_types: [0, 0],
434                parallactic_angle_rad: [0.0, 0.0],
435                field: 0,
436                spectral_window: 0,
437            },
438        };
439        self.model.resampler.predict_row(
440            &self.model.operator,
441            &mut workspace.backend,
442            &self.grids,
443            &native,
444            &mut workspace.scratch,
445            out,
446        )?;
447        Ok(())
448    }
