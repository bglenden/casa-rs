8pub(super) fn imstat_typed<T>(
9    path: &Path,
10    image: &PagedImage<T>,
11    request: &ImstatRequest,
12    cancellation: Option<&Cancellation>,
13) -> Result<ImstatOutcome, ImageError>
14where
15    T: ImagePixel + ImageExprValue + PartialOrd + ExprValueConvert + StatsElement,
16{
17    check_imstat_cancelled(cancellation)?;
18    let algorithm = parse_imstat_algorithm(request).map_err(ImageError::InvalidMetadata)?;
19    let axes =
20        parse_axes(request.axes.as_deref(), image.ndim()).map_err(ImageError::CasaSelection)?;
21    let region_selection = region_pixels(image, request.region.as_deref())?;
22    if request
23        .box_pixels
24        .as_deref()
25        .is_some_and(|text| !text.trim().is_empty())
26        && region_selection.is_some()
27    {
28        return Err(ImageError::InvalidMetadata(
29            "imstat accepts either box or region, not both".to_string(),
30        ));
31    }
32    let explicit_box = match request
33        .box_pixels
34        .as_deref()
35        .filter(|text| !text.trim().is_empty())
36    {
37        Some(box_text) => Some(parse_box_region(
38            box_text,
39            image.shape(),
40            direction_plane_axes(image)?,
41        )?),
42        None => None,
43    };
44    let effective_region = explicit_box.as_ref().or(region_selection.as_ref());
45    let selection = Selection::new(
46        image,
47        effective_region,
48        request.chans.as_deref(),
49        request.stokes.as_deref(),
50    )?;
51    check_imstat_cancelled(cancellation)?;
52    let pixel_mask = selection_pixel_mask(
53        image,
54        effective_region,
55        &selection,
56        request.mask.as_deref(),
57        request.stretch,
58    )?;
59    let coordinates = image.coordinates();
60    let world_text = |position: &[usize]| -> Result<String, ImageError> {
61        let pixel: Vec<f64> = position.iter().map(|&value| value as f64).collect();
62        Ok(casa_coordinates::format_world_display(coordinates, &pixel)?)
63    };
64    let blc = selection.start.clone();
65    let trc = trc_pixels(&selection.start, &selection.shape);
66    let blcf = world_text(&blc)?;
67    let trcf = world_text(&trc)?;
68    check_imstat_cancelled(cancellation)?;
69
70    let display_axes: Vec<usize> = match &axes {
71        AxisSelection::All => Vec::new(),
72        AxisSelection::Collapsed(collapse) => (0..image.ndim())
73            .filter(|axis| !collapse.contains(axis))
74            .collect(),
75    };
76    let outcome = if display_axes.is_empty() {
77        let summary = imstat_global_summary(
78            path,
79            image,
80            request,
81            &algorithm,
82            &selection,
83            pixel_mask.as_ref(),
84            blc,
85            trc,
86            blcf,
87            trcf,
88            &world_text,
89            cancellation,
90        )?;
91        ImstatOutcome::Global(summary)
92    } else {
93        let summary = imstat_planes_summary(
94            path,
95            image,
96            request,
97            &algorithm,
98            &selection,
99            pixel_mask.as_ref(),
100            &display_axes,
101            blc,
102            trc,
103            blcf,
104            trcf,
105            cancellation,
106        )?;
107        ImstatOutcome::Planes(summary)
108    };
109    Ok(outcome)
110}

112/// The whole selection's statistics, as `imstat` reports them with every
113/// axis collapsed.
114#[allow(clippy::too_many_arguments)]
115pub(super) fn imstat_global_summary<T>(
116    path: &Path,
117    image: &PagedImage<T>,
118    request: &ImstatRequest,
119    algorithm: &StatisticsAlgorithm,
120    selection: &Selection,
121    pixel_mask: Option<&ndarray::ArrayD<bool>>,
122    blc: Vec<usize>,
123    trc: Vec<usize>,
124    blcf: String,
125    trcf: String,
126    world_text: &dyn Fn(&[usize]) -> Result<String, ImageError>,
127    cancellation: Option<&Cancellation>,
128) -> Result<ImageStatisticsSummary, ImageError>
129where
130    T: ImagePixel + StatsElement,
131{
132    // The statistics the algorithm reports, with the extremum positions
133    // relative to the original image.
134    let computed = match algorithm {
135        StatisticsAlgorithm::Classic
136            if !selection.stokes_has_gaps() && !selection.channels_have_gaps() =>
137        {
138            // The classical moments stream through the lattice
139            // statistics; a gapped Stokes set needs the sample route.
140            let sub = image.sub_image(selection.start.clone(), selection.shape.clone())?;
141            let mut stats = LatticeStatistics::new(&sub);
142            stats.set_cancellation(cancellation);
143            if let Some([min, max]) = request.includepix {
144                stats.set_include_range(min, max);
145            }
146            if let Some(mask) = pixel_mask.cloned() {
147                stats.set_pixel_mask(mask);
148            }
149            let (minpos, maxpos) = stats.get_min_max_pos()?;
150            ComputedStats {
151                npts: scalar_stat(&stats, Statistic::Npts)?,
152                sum: scalar_stat(&stats, Statistic::Sum)?,
153                min: scalar_stat(&stats, Statistic::Min)?,
154                max: scalar_stat(&stats, Statistic::Max)?,
155                sumsq: scalar_stat(&stats, Statistic::SumSq)?,
156                mean: scalar_stat(&stats, Statistic::Mean)?,
157                rms: scalar_stat(&stats, Statistic::Rms)?,
158                sigma: scalar_stat(&stats, Statistic::Sigma)?,
159                median: scalar_stat(&stats, Statistic::Median)?,
160                medabsdevmed: scalar_stat(&stats, Statistic::MedAbsDevMed)?,
161                q1: scalar_stat(&stats, Statistic::Q1)?,
162                q3: scalar_stat(&stats, Statistic::Q3)?,
163                quartile: scalar_stat(&stats, Statistic::Quartile)?,
164                minpos: minpos.map(|pos| offset_pos(&selection.start, pos)),
165                maxpos: maxpos.map(|pos| offset_pos(&selection.start, pos)),
166            }
167        }
168        robust => {
169            let pixels = image.get_slice(&selection.start, &selection.shape)?;
170            let sample = sample_points(
171                &pixels,
172                pixel_mask,
173                request.includepix,
174                selection.channel_axis,
175                selection.channel_indices.as_deref(),
176                selection.stokes_axis,
177                selection.stokes_indices.as_deref(),
178                &selection.start,
179                cancellation,
180            )?;
181            match compute_robust_stats_with_cancellation(&sample, robust, cancellation)? {
182                RobustStats::Sample(stats) => ComputedStats {
183                    npts: stats.npts,
184                    sum: stats.sum,
185                    min: stats.min,
186                    max: stats.max,
187                    sumsq: stats.sumsq,
188                    mean: stats.mean,
189                    rms: stats.rms,
190                    sigma: stats.sigma,
191                    median: stats.median,
192                    medabsdevmed: stats.medabsdevmed,
193                    q1: stats.q1,
194                    q3: stats.q3,
195                    quartile: stats.quartile,
196                    minpos: stats.minpos.map(|linear| {
197                        offset_pos(
198                            &selection.start,
199                            selection_position(linear, &selection.shape),
200                        )
201                    }),
202                    maxpos: stats.maxpos.map(|linear| {
203                        offset_pos(
204                            &selection.start,
205                            selection_position(linear, &selection.shape),
206                        )
207                    }),
208                },
209                RobustStats::Biweight { location, scale } => {
210                    // The biweight reports only its location and scale,
211                    // with the sample's count and extrema.
212                    let npts = sample.len() as f64;
213                    let (min, max) = sample_extrema(&sample, cancellation)?;
214                    ComputedStats {
215                        npts,
216                        sum: f64::NAN,
217                        min,
218                        max,
219                        sumsq: f64::NAN,
220                        mean: location,
221                        rms: f64::NAN,
222                        sigma: scale,
223                        median: f64::NAN,
224                        medabsdevmed: f64::NAN,
225                        q1: f64::NAN,
226                        q3: f64::NAN,
227                        quartile: f64::NAN,
228                        minpos: None,
229                        maxpos: None,
230                    }
231                }
232            }
233        }
234    };
235
236    let minposf = match &computed.minpos {
237        Some(position) => Some(world_text(position)?),
238        None => None,
239    };
240    let maxposf = match &computed.maxpos {
241        Some(position) => Some(world_text(position)?),
242        None => None,
243    };
244    let flux = if algorithm.kind() == StatisticsAlgorithmKind::Biweight {
245        None
246    } else {
247        image_flux(image, computed.sum, selection)
248    };
249    Ok(ImageStatisticsSummary {
250        blcf,
251        trcf,
252        npts: computed.npts,
253        min: computed.min,
254        max: computed.max,
255        sum: computed.sum,
256        sumsq: computed.sumsq,
257        mean: computed.mean,
258        rms: computed.rms,
259        sigma: computed.sigma,
260        median: computed.median,
261        medabsdevmed: computed.medabsdevmed,
262        q1: computed.q1,
263        q3: computed.q3,
264        quartile: computed.quartile,
265        minposf,
266        maxposf,
267        minpos: computed.minpos,
268        maxpos: computed.maxpos,
269        imagename: path.display().to_string(),
270        blc,
271        trc,
272        flux,
273        units: image.units().to_string(),
274    })
275}

277/// Each display plane's statistics, as `imstat` reports them with only
278/// some axes collapsed: one value per display position, in display
279/// order, with CASA's degenerate display axes dropped from the record's
280/// shape.
281#[allow(clippy::too_many_arguments)]
282pub(super) fn imstat_planes_summary<T>(
283    path: &Path,
284    image: &PagedImage<T>,
285    request: &ImstatRequest,
286    algorithm: &StatisticsAlgorithm,
287    selection: &Selection,
288    pixel_mask: Option<&ndarray::ArrayD<bool>>,
289    display_axes: &[usize],
290    blc: Vec<usize>,
291    trc: Vec<usize>,
292    blcf: String,
293    trcf: String,
294    cancellation: Option<&Cancellation>,
295) -> Result<ImstatPlanesSummary, ImageError>
296where
297    T: ImagePixel + StatsElement,
298{
299    let display_positions: Vec<Vec<usize>> = display_axes
300        .iter()
301        .map(|&axis| selection.axis_positions(axis))
302        .collect();
303    let display_counts: Vec<usize> = display_positions.iter().map(|p| p.len()).collect();
304    let display_shape: Vec<usize> = display_counts
305        .iter()
306        .copied()
307        .filter(|&count| count > 1)
308        .collect();
309    let plane_count: usize = display_counts.iter().product();
310
311    // One pass over the selection's pixels, bucketed by display
312    // position: each pixel's plane is its display coordinates' index.
313    let pixels = image.get_slice(&selection.start, &selection.shape)?;
314    check_imstat_cancelled(cancellation)?;
315    let mut buckets: Vec<Vec<SamplePoint>> = vec![Vec::new(); plane_count];
316    let stokes_collapsed = selection
317        .stokes_axis
318        .is_some_and(|axis| display_axes.iter().all(|&display| display != axis));
319    let mut strides = Vec::with_capacity(pixels.shape().len());
320    let mut stride = 1u64;
321    for &axis_length in pixels.shape() {
322        strides.push(stride);
323        stride *= axis_length as u64;
324    }
325    for (visited, (index, value)) in pixels.indexed_iter().enumerate() {
326        if visited.is_multiple_of(4096) {
327            check_imstat_cancelled(cancellation)?;
328        }
329        if let Some(mask) = pixel_mask {
330            let mask_index: &[usize] = index.slice();
331            if !mask[mask_index] {
332                continue;
333            }
334        }
335        let number = value.to_f64_stats();
336        if let Some([min, max]) = request.includepix
337            && !(min <= number && number <= max)
338        {
339            continue;
340        }
341        let mut linear = 0u64;
342        for (axis, &axis_stride) in strides.iter().enumerate() {
343            linear += index[axis] as u64 * axis_stride;
344        }
345        let mut bucket = 0usize;
346        // The display positions run in Fortran order, the order casacore
347        // keeps array data in and casa-rs's records carry.
348        let mut slot_stride = 1usize;
349        let mut in_selection = true;
350        for (slot, &axis) in display_axes.iter().enumerate() {
351            let absolute = index[axis] + selection.start[axis];
352            let Some(slot_index) = display_positions[slot]
353                .iter()
354                .position(|&position| position == absolute)
355            else {
356                in_selection = false;
357                break;
358            };
359            bucket += slot_index * slot_stride;
360            slot_stride *= display_counts[slot];
361        }
362        if !in_selection {
363            continue;
364        }
365        if stokes_collapsed {
366            let axis = selection.stokes_axis.expect("the axis was found above");
367            let kept = selection.stokes_indices.as_deref().unwrap_or(&[]);
368            let absolute = index[axis] + selection.start[axis];
369            if !kept.contains(&absolute) {
370                continue;
371            }
372        }
373        buckets[bucket].push(SamplePoint {
374            value: number,
375            position: linear,
376        });
377    }
378
379    let biweight = algorithm.kind() == StatisticsAlgorithmKind::Biweight;
380    let spectral_collapsed = image
381        .find_axis(CoordinateType::Spectral)
382        .is_some_and(|axis| display_axes.iter().all(|&display| display != axis));
383    // CASA's flux density exists per plane only when the direction plane
384    // is at least partly collapsed, as casacore beam-integrates the sky
385    // it sums: a plane that still varies over the whole sky has none.
386    let [plane_x, plane_y] = selection.plane;
387    let flux_over_sky = !display_axes.contains(&plane_x) || !display_axes.contains(&plane_y);
388    let mut planes = ImstatPlanesSummary {
389        imagename: path.display().to_string(),
390        blc,
391        trc,
392        blcf,
393        trcf,
394        display_shape,
395        plane_log: String::new(),
396        npts: Vec::with_capacity(plane_count),
397        min: Vec::with_capacity(plane_count),
398        max: Vec::with_capacity(plane_count),
399        sum: Vec::with_capacity(plane_count),
400        sumsq: Vec::with_capacity(plane_count),
401        mean: Vec::with_capacity(plane_count),
402        rms: Vec::with_capacity(plane_count),
403        sigma: Vec::with_capacity(plane_count),
404        median: Vec::with_capacity(plane_count),
405        medabsdevmed: Vec::with_capacity(plane_count),
406        q1: Vec::with_capacity(plane_count),
407        q3: Vec::with_capacity(plane_count),
408        quartile: Vec::with_capacity(plane_count),
409        flux: None,
410        biweight,
411    };
412    let mut iterations = Vec::new();
413    for mut sample in buckets {
414        check_imstat_cancelled(cancellation)?;
415        #[cfg(feature = "imstat-test-hooks")]
416        observe(Checkpoint::PlaneStarted);
417        sample.sort_by_key(|point| point.position);
418        match compute_robust_stats_with_cancellation(&sample, algorithm, cancellation)? {
419            RobustStats::Sample(stats) => {
420                iterations.push(stats.chauvenet_iterations);
421                planes.npts.push(stats.npts);
422                planes.min.push(stats.min);
423                planes.max.push(stats.max);
424                planes.sum.push(stats.sum);
425                planes.sumsq.push(stats.sumsq);
426                planes.mean.push(stats.mean);
427                planes.rms.push(stats.rms);
428                planes.sigma.push(stats.sigma);
429                planes.median.push(stats.median);
430                planes.medabsdevmed.push(stats.medabsdevmed);
431                planes.q1.push(stats.q1);
432                planes.q3.push(stats.q3);
433                planes.quartile.push(stats.quartile);
434                let flux = if flux_over_sky {
435                    image_flux_with_integration(image, stats.sum, selection, spectral_collapsed)
436                } else {
437                    None
438                };
439                if let Some(flux) = flux {
440                    planes
441                        .flux
442                        .get_or_insert_with(|| Vec::with_capacity(plane_count))
443                        .push(flux);
444                }
445            }
446            RobustStats::Biweight { location, scale } => {
447                let count = sample.len() as f64;
448                let (min, max) = sample_extrema(&sample, cancellation)?;
449                planes.npts.push(count);
450                planes.min.push(min);
451                planes.max.push(max);
452                planes.sum.push(f64::NAN);
453                planes.sumsq.push(f64::NAN);
454                planes.mean.push(location);
455                planes.rms.push(f64::NAN);
456                planes.sigma.push(scale);
457                planes.median.push(f64::NAN);
458                planes.medabsdevmed.push(f64::NAN);
459                planes.q1.push(f64::NAN);
460                planes.q3.push(f64::NAN);
461                planes.quartile.push(f64::NAN);
462            }
463        }
464        #[cfg(feature = "imstat-test-hooks")]
465        observe(Checkpoint::PlaneFinished);
466    }
467    if biweight {
468        planes.flux = None;
469    }
470    if request.listit
471        || request
472            .logfile
473            .as_deref()
474            .is_some_and(|name| !name.is_empty())
475    {
476        planes.plane_log = imstat_log::plane_table(
477            image,
478            &planes,
479            display_axes,
480            &display_positions,
481            &iterations,
482            cancellation,
483        )?;
484    }
485    Ok(planes)
486}

488fn sample_extrema(
489    sample: &[SamplePoint],
490    cancellation: Option<&Cancellation>,
491) -> Result<(f64, f64), ImageError> {
492    let (mut min, mut max) = (f64::NAN, f64::NAN);
493    for batch in sample.chunks(4096) {
494        check_imstat_cancelled(cancellation)?;
495        for point in batch {
496            min = min.min(point.value);
497            max = max.max(point.value);
498        }
499    }
500    Ok((min, max))
501}

523/// Converts a sum of image pixels to the flux reported by CASA statistics.
524///
525/// The factor uses square arcseconds for temperature and pixels per beam for
526/// per-beam brightness. If the spectral axis is collapsed over multiple
527/// channels, `channels` gives the first two measured spectral pixels.
528pub fn statistics_flux_scale(
529    coordinates: &CoordinateSystem,
530    units: &str,
531    beam: Option<GaussianBeam>,
532    channels: Option<[usize; 2]>,
533) -> Option<f64> {
534    let pixel_area = direction_pixel_area(coordinates)?;
535    let flux_factor = if units.contains('K') {
536        let arcsec_per_rad = 180.0 * 3600.0 / std::f64::consts::PI;
537        pixel_area * arcsec_per_rad * arcsec_per_rad
538    } else if units.contains("/beam") {
539        let beam_area = beam?.area();
540        if beam_area <= 0.0 {
541            return None;
542        }
543        pixel_area / beam_area
544    } else {
545        return None;
546    };
547    let channel_width = channels
548        .and_then(|pixels| spectral_integration(coordinates, pixels))
549        .unwrap_or(1.0);
550    Some(flux_factor * channel_width)
551}
552
553/// The width casacore integrates `flux` over: one channel of the selected
554/// spectral axis, when the selection holds more than one.
555pub(super) fn spectral_integration(
556    coordinates: &CoordinateSystem,
557    [first, second]: [usize; 2],
558) -> Option<f64> {
559    let coordinate = coordinates.coordinate(coordinates.find_coordinate(CoordinateType::Spectral)?);
560    let rest = spectral_rest_frequency(coordinates).unwrap_or(0.0);
561    if rest > 0.0 {
562        // The selection's own first two channels, as casacore's sub-image.
563        let f0 = coordinate.to_world(&[first as f64]).ok()?[0];
564        let f1 = coordinate.to_world(&[second as f64]).ok()?[0];
565        Some(SPEED_OF_LIGHT_KM_S * (f1 - f0).abs() / rest)
566    } else {
567        Some(coordinate.increment()[0] * second.abs_diff(first) as f64)
568    }
569}
570
571pub(super) fn direction_pixel_area(coords: &CoordinateSystem) -> Option<f64> {
572    for idx in 0..coords.n_coordinates() {
573        let coord = coords.coordinate(idx);
574        if coord.coordinate_type() == CoordinateType::Direction && coord.n_pixel_axes() >= 2 {
575            let increment = coord.increment();
576            return Some(increment[0].abs() * increment[1].abs());
577        }
578    }
579    None
