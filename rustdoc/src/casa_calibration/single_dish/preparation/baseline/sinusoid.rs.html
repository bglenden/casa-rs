12pub(super) const INVALID_WAVE: &str = "wrong value given for addwn/rejwn";
13const NO_WAVES: &str = "No effective wave number given for sinusoidal fitting.";
14pub(super) const INVALID_FIT: &str = "sakura_LSQFitSinusoidFloat() failed -- InvalidArgument";
15
16#[derive(Clone)]
17pub(super) enum WaveSelection {
18    Explicit(Vec<usize>),
19    Exact(Vec<usize>),
20    Interval { first: usize, last: usize },
21    ThroughNyquist(usize),
22}
23
24impl WaveSelection {
25    pub(super) fn explicit(numbers: impl IntoIterator<Item = i64>) -> Result<Self, String> {
26        numbers
27            .into_iter()
28            .map(|number| usize::try_from(number).map_err(|_| INVALID_WAVE.into()))
29            .collect::<Result<Vec<_>, _>>()
30            .map(Self::Explicit)
31    }
32
33    pub(super) fn from_task_text(text: &str) -> Result<Self, String> {
34        let index = |value: &str| value.trim().parse::<usize>().map_err(|_| INVALID_WAVE);
35        if text.contains(',') {
36            let indexes = text
37                .split(',')
38                .map(index)
39                .collect::<Result<BTreeSet<_>, _>>()?;
40            return Ok(Self::Explicit(indexes.into_iter().collect()));
41        }
42        if let Some(delimiter) = ['-', '~']
43            .into_iter()
44            .find(|&delimiter| text.contains(delimiter))
45        {
46            let bounds = text
47                .split(delimiter)
48                .map(index)
49                .collect::<Result<Vec<_>, _>>()?;
50            let [first, last, ..] = bounds.as_slice() else {
51                return Err(INVALID_WAVE.into());
52            };
53            let (first, last) = (*first, *last);
54            return Ok(Self::Interval {
55                first: first.min(last),
56                last: first.max(last),
57            });
58        }
59        if let Some(last) = text
60            .strip_prefix("<=")
61            .or_else(|| text.strip_prefix("=<"))
62            .or_else(|| text.strip_suffix(">="))
63            .or_else(|| text.strip_suffix("=>"))
64        {
65            return Ok(Self::Interval {
66                first: 0,
67                last: index(last)?,
68            });
69        }
70        if let Some(last) = text.strip_prefix('<').or_else(|| text.strip_suffix('>')) {
71            return Ok(Self::Interval {
72                first: 0,
73                last: index(last)?.checked_sub(1).ok_or(INVALID_WAVE)?,
74            });
75        }
76        if let Some(first) = text
77            .strip_prefix(">=")
78            .or_else(|| text.strip_prefix("=>"))
79            .or_else(|| text.strip_suffix("<="))
80            .or_else(|| text.strip_suffix("=<"))
81        {
82            return Ok(Self::ThroughNyquist(index(first)?));
83        }
84        if let Some(first) = text.strip_prefix('>').or_else(|| text.strip_suffix('<')) {
85            return Ok(Self::ThroughNyquist(
86                index(first)?.checked_add(1).ok_or(INVALID_WAVE)?,
87            ));
88        }
89        Ok(Self::Explicit(vec![index(text)?]))
90    }
91
92    fn from_tool_text(text: &str) -> Self {
93        let values = text
94            .split(',')
95            .filter(|term| !term.is_empty())
96            .map(c_integer)
97            .collect::<Vec<_>>();
98        if let [first, -999] = values.as_slice() {
99            return Self::ThroughNyquist(usize::try_from(*first).unwrap_or_default());
100        }
101        Self::Explicit(
102            values
103                .into_iter()
104                .filter_map(|value| usize::try_from(value).ok())
105                .collect(),
106        )
107    }
108
109    fn includes(&self, wave: usize) -> bool {
110        match self {
111            Self::Explicit(values) | Self::Exact(values) => values.contains(&wave),
112            Self::Interval { first, last } => (*first..=*last).contains(&wave),
113            Self::ThroughNyquist(first) => wave >= *first,
114        }
115    }
116
117    fn rejects_fft(&self, wave: usize) -> bool {
118        match self {
119            // CASA's merge sees the original two-element sentinel representation.
120            Self::ThroughNyquist(first) => wave == *first,
121            _ => self.includes(wave),
122        }
123    }
124}

126// The direct binding passes each comma-delimited token through atoi.
127fn c_integer(text: &str) -> i32 {
128    parse_integer_prefix(text).unwrap_or_default() as i32
129}
130
131fn parse_integer_prefix(text: &str) -> Option<i64> {
132    let text = text.trim_start();
133    let digits = text.strip_prefix(['-', '+']).unwrap_or(text);
134    let count = digits.bytes().take_while(u8::is_ascii_digit).count();
135    let sign = text.len() - digits.len();
136    text[..sign + count].parse().ok()
137}

139#[derive(Clone, Copy)]
140pub(super) enum FftThreshold {
141    Sigma(f32),
142    Top(usize),
143}
144
145impl FftThreshold {
146    pub(super) fn validate(text: &str) -> Result<(), String> {
147        const FORMAT: &str = "fftthresh has a wrong format.";
148        const POSITIVE: &str = "threshold given to fftthresh must be positive.";
149        let not_positive = if let Some(top) = text.strip_prefix("top") {
150            top.trim().parse::<i64>().map_err(|_| FORMAT)? <= 0
151        } else {
152            text.strip_suffix("sigma")
153                .unwrap_or(text)
154                .trim()
155                .parse::<f64>()
156                .map_err(|_| FORMAT)?
157                <= 0.
158        };
159        if not_positive {
160            Err(POSITIVE.into())
161        } else {
162            Ok(())
163        }
164    }
165
166    fn from_tool(text: &str) -> Result<Self, &'static str> {
167        const INVALID: &str = "fftthresh has a wrong value";
168        if let Some(count) = text.strip_prefix("top") {
169            return parse_integer_prefix(count)
170                .and_then(|count| i32::try_from(count).ok())
171                .map(|count| Self::Top(count.max(0) as usize))
172                .ok_or(INVALID);
173        }
174        let sigma = if let Some(sigma) = text.strip_suffix("sigma") {
175            sigma
176        } else if text.is_empty()
177            || !text
178                .bytes()
179                .take(text.len() - 1)
180                .all(|byte| byte.is_ascii_digit() || byte == b'.')
181        {
182            return Err(INVALID);
183        } else {
184            text
185        };
186        let sigma = sigma.trim_start();
187        let prefix = sigma
188            .bytes()
189            .take_while(|&byte| {
190                byte.is_ascii_digit() || matches!(byte, b'+' | b'-' | b'.' | b'e' | b'E')
191            })
192            .count();
193        sigma[..prefix]
194            .parse::<f32>()
195            .ok()
196            .filter(|value| value.is_finite())
197            .map(Self::Sigma)
198            .ok_or(INVALID)
199    }
200}

202/// Wave selection for the sinusoidal single-dish baseline.
203pub struct SingleDishSinusoid {
204    include: WaveSelection,
205    exclude: WaveSelection,
206    fft: Option<Result<FftThreshold, &'static str>>,
207}
208
209impl SingleDishSinusoid {
210    /// Interpret the comma-delimited wavenumbers from the direct CASA tool.
211    /// A final sentinel of `-999` denotes an interval through Nyquist.
212    pub fn from_tool(
213        addwn: &str,
214        rejwn: &str,
215        applyfft: bool,
216        fftmethod: &str,
217        fftthresh: &str,
218    ) -> Result<Self, String> {
219        if addwn.split(',').all(str::is_empty) {
220            return Err("addwn must contain at least one element.".into());
221        }
222        Ok(Self::new(
223            WaveSelection::from_tool_text(addwn),
224            WaveSelection::from_tool_text(rejwn),
225            applyfft,
226            fftmethod,
227            fftthresh,
228        ))
229    }
230
231    pub(super) fn from_variable_waves(waves: Vec<usize>) -> Self {
232        Self {
233            include: WaveSelection::Exact(waves),
234            exclude: WaveSelection::Explicit(Vec::new()),
235            fft: None,
236        }
237    }
238
239    pub(super) fn variable_waves(&self) -> &[usize] {
240        match &self.include {
241            WaveSelection::Exact(waves) => waves,
242            _ => &[],
243        }
244    }
245
246    pub(super) fn new(
247        include: WaveSelection,
248        exclude: WaveSelection,
249        applyfft: bool,
250        fftmethod: &str,
251        fftthresh: &str,
252    ) -> Self {
253        let fft = applyfft.then(|| {
254            if fftmethod != "fft" {
255                return Err("fftmethod must be 'fft' for now.");
256            }
257            FftThreshold::from_tool(fftthresh)
258        });
259        Self {
260            include,
261            exclude,
262            fft,
263        }
264    }
265
266    pub(super) fn fit(
267        &self,
268        spectrum: &[f32],
269        included: &[bool],
270        clip_sigma: f64,
271        max_fits: usize,
272        has_unflagged_data: bool,
273    ) -> Result<Option<(Fit, Vec<usize>)>, String> {
274        let maximum = (spectrum.len() / 2).checked_sub(1).ok_or(NO_WAVES)?;
275        let extended = matches!(self.exclude, WaveSelection::ThroughNyquist(_));
276        let first_rejected = match self.exclude {
277            WaveSelection::ThroughNyquist(first) => Some(first),
278            _ => None,
279        };
280        let mut waves = match (&self.include, first_rejected) {
281            (WaveSelection::Exact(waves), _) => waves.clone(),
282            (WaveSelection::Explicit(values), _) => values
283                .iter()
284                .copied()
285                .filter(|&wave| (extended || wave <= maximum) && !self.exclude.includes(wave))
286                .collect::<Vec<_>>(),
287            (WaveSelection::Interval { first, last }, Some(rejected)) => {
288                let last = (*last).min(rejected.checked_sub(1).ok_or(NO_WAVES)?);
289                if last < *first {
290                    return Err(NO_WAVES.into());
291                }
292                validate_context(last, spectrum.len())?;
293                (*first..=last).collect()
294            }
295            _ => (0..=maximum)
296                .filter(|&wave| self.include.includes(wave) && !self.exclude.includes(wave))
297                .collect(),
298        };
299        let base_maximum = *waves.last().ok_or(NO_WAVES)?;
300        if self.fft.is_none() || extended {
301            validate_context(base_maximum, spectrum.len())?;
302        }
303        if !has_unflagged_data {
304            return Ok(None);
305        }
306        let base_len = waves.len();
307        if let Some(choice) = self.fft {
308            let choice = choice?;
309            let selected = peaks(spectrum, included, maximum, choice)?;
310            for wave in selected {
311                if !self.exclude.rejects_fft(wave) && !waves[..base_len].contains(&wave) {
312                    waves.push(wave);
313                }
314            }
315            waves.sort_unstable();
316        }
317        let terms = terms(&waves);
318        if included.iter().filter(|&&value| value).count() < terms {
319            return Ok(None);
320        }
321        let largest = *waves.last().expect("nonempty waves");
322        if self.fft.is_some() && !extended {
323            validate_context(largest, spectrum.len())?;
324        }
325        if (extended && largest > base_maximum) || !waves.windows(2).all(|pair| pair[0] < pair[1]) {
326            return Err(INVALID_FIT.into());
327        }
328        let size = spectrum.len();
329        let basis = (0..size)
330            .map(|channel| basis(channel, size, &waves))
331            .collect();
332        let linear = LinearFit {
333            basis,
334            terms,
335            segment_ends: vec![size],
336            failure: "sakura_LSQFitSinusoidFloat() failed -- NG",
337        };
338        Ok(linear
339            .solve(spectrum, included, clip_sigma, max_fits, None)?
340            .map(|fit| (fit, waves)))
341    }
342}

344pub(super) fn validate_context(wave: usize, channels: usize) -> Result<(), String> {
345    if wave.saturating_mul(2).saturating_add(2) > channels {
346        Err("sakura_CreateLSQFitContextSinusoidFloat() failed -- InvalidArgument".into())
347    } else {
348        Ok(())
349    }
350}

352fn peaks(
353    spectrum: &[f32],
354    included: &[bool],
355    maximum: usize,
356    threshold: FftThreshold,
357) -> Result<Vec<usize>, String> {
358    let samples = interpolate(spectrum, included)?;
359    let mut fft = RealFft2::<f32>::with_threads([1, spectrum.len()], 1)
360        .map_err(|error| error.to_string())?
361        .with_estimated_plan();
362    let mut storage = vec![Complex32::new(0., 0.); fft.storage_len()];
363    for (i, sample) in samples.into_iter().enumerate() {
364        if i.is_multiple_of(2) {
365            storage[i / 2].re = sample;
366        } else {
367            storage[i / 2].im = sample;
368        }
369    }
370    fft.forward(&mut storage)
371        .map_err(|error| error.to_string())?;
372    let scale = (2. / spectrum.len() as f64) as f32;
373    let amplitudes = storage
374        .into_iter()
375        .map(|amplitude| amplitude.norm() * scale)
376        .collect::<Vec<_>>();
377    match threshold {
378        FftThreshold::Sigma(sigma) => {
379            let count = amplitudes.len() as f32;
380            let sum = amplitudes.iter().sum::<f32>();
381            let squared = amplitudes
382                .iter()
383                .map(|&amplitude| amplitude * amplitude)
384                .sum::<f32>();
385            let mean = sum / count;
386            let cutoff = mean + sigma * (squared / count - mean * mean).sqrt();
387            Ok(amplitudes
388                .iter()
389                .take(maximum + 1)
390                .enumerate()
391                .filter_map(|(wave, &amplitude)| (cutoff <= amplitude).then_some(wave))
392                .collect())
393        }
394        FftThreshold::Top(count) => {
395            let mut indexes = (0..=maximum)
396                .filter(|&index| amplitudes[index] > 0.)
397                .collect::<Vec<_>>();
398            indexes.sort_by(|&left, &right| amplitudes[right].total_cmp(&amplitudes[left]));
399            indexes.truncate(count);
400            indexes.extend(std::iter::repeat_n(
401                0,
402                count
403                    .saturating_sub(indexes.len())
404                    .min(spectrum.len().saturating_add(1)),
405            ));
406            Ok(indexes)
407        }
408    }
409}

411fn interpolate(spectrum: &[f32], included: &[bool]) -> Result<Vec<f32>, String> {
412    let mut values = spectrum.to_vec();
413    let mut previous = None;
414    let mut from = 0;
415    for (channel, &valid) in included.iter().enumerate() {
416        if !valid {
417            continue;
418        }
419        if from < channel {
420            let value = previous.map_or(spectrum[channel], |before| {
421                (before + spectrum[channel]) / 2.
422            });
423            values[from..channel].fill(value);
424        }
425        previous = Some(spectrum[channel]);
426        from = channel + 1;
427    }
428    let last = previous.ok_or("Bad data. all channels are masked.")?;
429    values[from..].fill(last);
430    Ok(values)
431}

433pub(super) fn terms(waves: &[usize]) -> usize {
434    2 * waves.len() - usize::from(waves.first() == Some(&0))
435}
436
437fn basis(channel: usize, channels: usize, waves: &[usize]) -> Vec<f64> {
438    let norm = TAU / channels.saturating_sub(1).max(1) as f64;
439    let mut result = Vec::with_capacity(terms(waves));
440    for &wave in waves {
441        if wave == 0 {
442            result.push(1.);
443        } else {
444            let (sine, cosine) = (channel as f64 * wave as f64 * norm).sin_cos();
445            result.extend([sine, cosine]);
446        }
447    }
448    result
449}

451pub(super) fn subtract(
452    spectrum: &mut [f32],
453    coefficients: &[f64],
454    waves: &[usize],
455) -> Result<(), String> {
456    if waves.is_empty() || coefficients.len() != terms(waves) {
457        return Err("Invalid sinusoidal baseline coefficients.".into());
458    }
459    let channels = spectrum.len();
460    for (channel, sample) in spectrum.iter_mut().enumerate() {
461        let terms = basis(channel, channels, waves);
462        let fitted = terms
463            .iter()
464            .zip(coefficients)
465            .map(|(basis, coefficient)| basis * coefficient)
466            .sum::<f64>();
467        *sample -= fitted as f32;
468    }
469    Ok(())
