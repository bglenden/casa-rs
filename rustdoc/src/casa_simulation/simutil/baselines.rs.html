10/// `simutil.itrf2loc(x, y, z, cx, cy, cz)`: positions relative to `centre`,
11/// as `[east, north, up]` metres in the plane tangent to the WGS84
12/// ellipsoid at the centre's latitude and longitude. (CASA returns the
13/// three as `lat, lon, el`.) The centre's latitude and longitude come from
14/// casa-rs's `xyz2long`, which corrects CASA's (see
15/// [`crate::readantenna::geodesy`]); the rotation is CASA's.
16pub fn itrf_to_local(
17    positions: &[[f64; 3]],
18    centre: [f64; 3],
19) -> Result<Vec<[f64; 3]>, SimutilError> {
20    let wgs84 = geodesy::datum("WGS84").expect("WGS84 is one of casa-rs's datums");
21    let (longitude, latitude, _) = geodesy::xyz_to_geodetic(centre[0], centre[1], centre[2], wgs84)
22        .ok_or_else(|| SimutilError::Arithmetic(format!("no WGS84 position for {centre:?}")))?;
23    let (sin_lon, cos_lon) = longitude.sin_cos();
24    let (sin_lat, cos_lat) = latitude.sin_cos();
25    Ok(positions
26        .iter()
27        .map(|position| {
28            let [x, y, z] = [
29                position[0] - centre[0],
30                position[1] - centre[1],
31                position[2] - centre[2],
32            ];
33            [
34                (-sin_lon * x) + (cos_lon * y),
35                (-sin_lat * cos_lon * x) - (sin_lat * sin_lon * y) + cos_lat * z,
36                (cos_lat * cos_lon * x) + (cos_lat * sin_lon * y) + sin_lat * z,
37            ]
38        })
39        .collect())
40}

42/// `simutil.baselineLengths(configfile)`: every antenna pair's distance,
43/// metres, pair `(i, j)` for `i < j` in file order. CASA puts the antennas
44/// in local coordinates about their mean position first, which leaves the
45/// distances as they are.
46pub fn baseline_lengths(config: &AntennaConfigFile) -> Result<Vec<f64>, SimutilError> {
47    let count = config.x.len();
48    let positions: Vec<[f64; 3]> = (0..count)
49        .map(|i| [config.x[i], config.y[i], config.z[i]])
50        .collect();
51    let mean = |axis: usize| positions.iter().map(|p| p[axis]).sum::<f64>() / count as f64;
52    let local = itrf_to_local(&positions, [mean(0), mean(1), mean(2)])?;
53    let mut lengths = Vec::with_capacity(count * count.saturating_sub(1) / 2);
54    for i in 0..count {
55        for j in i + 1..count {
56            let [x, y, z] = [
57                local[i][0] - local[j][0],
58                local[i][1] - local[j][1],
59                local[i][2] - local[j][2],
60            ];
61            lengths.push((x * x + y * y + z * z).sqrt());
62        }
63    }
64    Ok(lengths)
65}

67/// `simutil.approxBeam(configfile, freq)`: λ over the 90th-percentile
68/// baseline, arcsec, at `frequency_ghz`. The percentile is scipy's
69/// `scoreatpercentile`: the sorted lengths interpolated linearly at
70/// `0.9 × (n − 1)`.
71pub fn approximate_beam(
72    config: &AntennaConfigFile,
73    frequency_ghz: f64,
74) -> Result<f64, SimutilError> {
75    if !(frequency_ghz > 0.0 && frequency_ghz.is_finite()) {
76        return Err(SimutilError::Frequency(frequency_ghz));
77    }
78    let mut lengths = baseline_lengths(config)?;
79    if lengths.is_empty() {
80        return Err(SimutilError::Arithmetic(
81            "an array of one antenna has no baselines".to_string(),
82        ));
83    }
84    lengths.sort_by(f64::total_cmp);
85    let index = 90.0 / 100.0 * (lengths.len() - 1) as f64;
86    let fraction = index % 1.0;
87    let below = lengths[index as usize];
88    let ninety = if fraction == 0.0 {
89        below
90    } else {
91        below + (lengths[index as usize + 1] - below) * fraction
92    };
93    Ok(0.2997924 / frequency_ghz / ninety * 3600.0 * 180.0 / std::f64::consts::PI)
94}
