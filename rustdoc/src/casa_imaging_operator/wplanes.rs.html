19/// CASA `WPConvFunc`: fine offsets per cell when there is more than one
20/// plane (`convSampling_p = 4`).
21const W_OVERSAMPLING: u16 = 4;
22/// CASA `WProjectFT` grid padding (`padding_p`, tclean's 1.2).
23const PADDING: f64 = 1.2;
24/// `WPConvFunc`: a plane's support reaches the last fine pixel whose
25/// modulus exceeds this fraction of the peak-normalised plane.
26const SUPPORT_THRESHOLD: f64 = 1.0e-3;
27
28/// How many w-planes to build.
29#[derive(Clone, Copy, Debug, PartialEq)]
30pub enum WPlaneCount {
31    /// `wprojplanes` given by the request: the largest `|w|` the planes
32    /// reach is `0.25 / |Δx|` wavelengths (`WPConvFunc::findConvFunction`
33    /// with `maxW_p = −1`, as tclean constructs it).
34    Fixed(u32),
35    /// `wprojplanes = −1`: the count follows CASA's `wStat` of the selected
36    /// data at the top frequency.
37    Auto {
38        /// `min |w| · ν_min / c` over the selection, in wavelengths.
39        min_w: f64,
40        /// `max |w| · ν_max / c` over the selection, in wavelengths.
41        max_w: f64,
42        /// `√(mean w²) · ν_max / c` over the selection, in wavelengths.
43        rms_w: f64,
44    },
45}
46
47/// The W-projection kernel set: one dense complex cell per w-plane,
48/// `C ⋆ FT[e^{2πi w(√(1−l²−m²)−1)}]` sampled four times per cell, scalar
49/// Mueller routing and the paired `1/(grdsf · sinc)` correction.
50#[derive(Clone, Debug)]
51pub struct WPlanes {
52    planes: Vec<DenseCell>,
53    w_scale: f64,
54    screen_size: usize,
55    max_half_support: u16,
56    mueller: MuellerRouting,
57    correction: ImageCorrection,
58}
59
60impl WPlanes {
61    /// Build the planes for `geometry`'s image on its CASA-padded grid.
62    ///
63    /// # Errors
64    ///
65    /// The plane count resolves to zero, the FFT cannot be planned, or the
66    /// plane-0 kernel has no positive integral.
67    pub fn new(
68        geometry: &GridGeometry,
69        polarization: &PolarizationRouting,
70        count: WPlaneCount,
71    ) -> Result<Self, OperatorError> {
72        let image = geometry.image();
73        let [nx, ny] = image.shape;
74        let increment = image.increment_rad;
75        // `WPConvFunc::findConvFunction`: the plane count and the largest
76        // |w| the planes reach.
77        let (planes, max_uvw) = match count {
78            WPlaneCount::Fixed(planes) => (planes as usize, 0.25 / increment[0].abs()),
79            WPlaneCount::Auto {
80                min_w,
81                max_w,
82                rms_w,
83            } => {
84                let mean = 0.5 * (min_w + max_w);
85                let max_uvw = if rms_w < mean {
86                    1.05 * max_w
87                } else {
88                    rms_w / mean * 1.05 * max_w
89                };
90                let half_fov = increment[0].abs() * nx.max(ny) as f64 / 2.0;
91                let planes = (max_uvw * half_fov.sin().abs()) as usize;
92                (planes, max_uvw)
93            }
94        };
95        if planes == 0 {
96            return Err(OperatorError::ConvolutionFunction {
97                reason: "W projection needs at least one plane",
98            });
99        }
100        // `wScale = Float((wConvSize-1)*(wConvSize-1))/maxUVW`.
101        let w_scale = f64::from(((planes - 1) * (planes - 1)) as f32) / max_uvw;
102        let sampling = if planes > 1 { W_OVERSAMPLING } else { 1 };
103        // `convSize = max(Int(nx*padding), Int(ny*padding))`, stepped to
104        // `nextLargerEven` for more than one plane only. `CompositeNumber::
105        // nextLargerEven` returns the first even composite strictly above
106        // its argument, so a padded size that is itself composite (1200 →
107        // 1440) steps up (to 1458); the grid keeps 1440 because `FTMachine`
108        // asks for the composite above `Int(1.2 n − 0.5)`. One plane keeps
109        // the raw padded size; CASA transforms it odd as well, while the
110        // plane FFT here swaps quadrants of even sides only, so an odd raw
111        // size (101 → 121) takes the next even one (#652, record 43).
112        let padded = ((nx as f64 * PADDING) as usize).max((ny as f64 * PADDING) as usize);
113        let conv_size = if planes > 1 {
114            next_larger_even_composite(padded + 1)
115        } else {
116            padded + padded % 2
117        };
118        let inner = conv_size / usize::from(sampling);
119        // The screen's sky sampling: `incr · convSampling · padding·n/convSize`.
120        let screen_increment = [
121            increment[0] * f64::from(sampling) * PADDING * nx as f64 / conv_size as f64,
122            increment[1] * f64::from(sampling) * PADDING * ny as f64 / conv_size as f64,
123        ];
124        let quadrant_side = conv_size / 2 - 1;
125        let correction_axis = |len: usize| {
126            (0..len)
127                .map(|index| grdsf((index as f64 - (len / 2) as f64).abs() / (len / 2) as f64))
128                .collect::<Vec<_>>()
129        };
130        let taper = correction_axis(inner);
131        let mut fft = PlaneFft::<f64>::new([conv_size, conv_size], false)?;
132        let mut screen = vec![Complex64::default(); conv_size * conv_size];
133        let mut quadrants = Vec::with_capacity(planes);
134        let mut peak = 0.0;
135        for plane in 0..planes {
136            // Plane `i` holds `w = i²/wScale`; with one plane CASA's
137            // `wScale` is zero and `makeGWplane` divides zero by zero, so
138            // the one-plane set takes the zero-w kernel instead.
139            let w = if planes > 1 {
140                (plane * plane) as f64 / w_scale
141            } else {
142                0.0
143            };
144            make_w_plane(&mut screen, conv_size, inner, screen_increment, &taper, w);
145            fft.transform(&mut screen, false)?;
146            let centre = conv_size / 2;
147            if plane == 0 {
148                peak = screen[centre * conv_size + centre].norm();
149                if !peak.is_finite() || peak <= 0.0 {
150                    return Err(OperatorError::ConvolutionFunction {
151                        reason: "the W-projection plane-0 kernel has no peak",
152                    });
153                }
154            }
155            let mut quadrant = Vec::with_capacity(quadrant_side * quadrant_side);
156            for y in 0..quadrant_side {
157                for x in 0..quadrant_side {
158                    let value = screen[(centre + y) * conv_size + centre + x] / peak;
159                    quadrant.push(Complex32::new(value.re as f32, value.im as f32));
160                }
161            }
162            quadrants.push(quadrant);
163        }
164        // Support per plane from the peak-normalised quadrant.
165        let fallback = conv_size / 2 / usize::from(sampling) - 1;
166        let supports = quadrants
167            .iter()
168            .map(|quadrant| {
169                let mut support = fallback;
170                for trial in (1..=conv_size / 2 - 2).rev() {
171                    if f64::from(quadrant[trial].norm()) > SUPPORT_THRESHOLD
172                        || f64::from(quadrant[trial * quadrant_side].norm()) > SUPPORT_THRESHOLD
173                    {
174                        support = (0.5 + trial as f64 / f64::from(sampling)) as usize + 1;
175                        if support * usize::from(sampling) * 2 >= conv_size {
176                            support = fallback;
177                        }
178                        break;
179                    }
180                }
181                support
182            })
183            .collect::<Vec<_>>();
184        // Crop every plane to the last plane's support and normalise all
185        // of them by the plane-0 integral over its support.
186        let last_support = *supports.last().expect("at least one plane");
187        let cropped = (2 * (last_support + 2) * usize::from(sampling)).min(conv_size);
188        let cropped_side = cropped / 2 - 1;
189        let mut integral = 0.0;
190        for iy in -(supports[0] as i64)..=supports[0] as i64 {
191            for ix in -(supports[0] as i64)..=supports[0] as i64 {
192                let x = ix.unsigned_abs() as usize * usize::from(sampling);
193                let y = iy.unsigned_abs() as usize * usize::from(sampling);
194                integral += f64::from(quadrants[0][y * quadrant_side + x].re);
195            }
196        }
197        if !integral.is_finite() || integral <= 0.0 {
198            return Err(OperatorError::ConvolutionFunction {
199                reason: "the W-projection convolution function integral is not positive",
200            });
201        }
202        let scale = (1.0 / integral) as f32;
203        let cells = quadrants
204            .iter()
205            .zip(&supports)
206            .map(|(quadrant, support)| {
207                let mut cropped_quadrant = Vec::with_capacity(cropped_side * cropped_side);
208                for y in 0..cropped_side {
209                    for x in 0..cropped_side {
210                        cropped_quadrant.push(quadrant[y * quadrant_side + x] * scale);
211                    }
212                }
213                let half_support = u16::try_from(*support).expect("support fits u16");
214                dense_cell_from_quadrant(&cropped_quadrant, cropped_side, sampling, half_support)
215            })
216            .collect::<Vec<_>>();
217        // `WProjectFT`'s sinc of the oversampled kernel cell: `getImage`
218        // tabulates it over `max(nx, ny)` for both axes and divides the
219        // image by `correctX1D · sinc`; `initializeToVis` tabulates it per
220        // axis and divides the model by `correctX1D / sinc`, so the model is
221        // multiplied by the sinc the image is divided by (the `MosaicFT`
222        // asymmetry).
223        let [grid_nx, grid_ny] = geometry.grid_shape();
224        let sinc = |len: usize, index: usize| {
225            let x = std::f64::consts::PI * (index as f64 - (len / 2) as f64)
226                / (len as f64 * f64::from(sampling));
227            if index == len / 2 { 1.0 } else { x.sin() / x }
228        };
229        let spheroidal = |len: usize, index: usize| {
230            let nu = ((index as f64 - (len / 2) as f64).abs() / (len / 2) as f64).clamp(0.0, 1.0);
231            grdsf(nu)
232        };
233        let sinc_len = grid_nx.max(grid_ny);
234        let image_correction = |len: usize| {
235            (0..len)
236                .map(|index| {
237                    let value = spheroidal(len, index) * sinc(sinc_len, index);
238                    if value.abs() > 1.0e-6 {
239                        1.0 / value
240                    } else {
241                        0.0
242                    }
243                })
244                .collect::<Vec<_>>()
245        };
246        let model_correction = |len: usize| {
247            (0..len)
248                .map(|index| {
249                    let value = spheroidal(len, index);
250                    if value.abs() > 1.0e-6 {
251                        sinc(len, index) / value
252                    } else {
253                        0.0
254                    }
255                })
256                .collect::<Vec<_>>()
257        };
258        Ok(Self {
259            max_half_support: u16::try_from(last_support).expect("support fits u16"),
260            planes: cells,
261            w_scale,
262            screen_size: conv_size,
263            mueller: MuellerRouting::scalar(polarization.pol_map(), polarization.grid_pols()),
264            correction: ImageCorrection::split(
265                image_correction(grid_nx),
266                image_correction(grid_ny),
267                model_correction(grid_nx),
268                model_correction(grid_ny),
269            ),
270        })
271    }

273    /// Number of planes.
274    #[must_use]
275    pub fn planes(&self) -> usize {
276        self.planes.len()
277    }
278
279    /// CASA `wScale`: plane `i` holds `w = i² / wScale`.
280    #[must_use]
281    pub const fn w_scale(&self) -> f64 {
282        self.w_scale
283    }
284
285    /// CASA `convSize`: the side of the screen the planes were transformed
286    /// on, the padded image size stepped to the even composite strictly
287    /// above it for more than one plane.
288    #[must_use]
289    pub const fn screen_size(&self) -> usize {
290        self.screen_size
291    }
292
293    /// The plane of `w_lambda`: `nint(√(wScale · |w|))` (`wprojgrid.f`
294    /// `swp`), clamped to the planes for a key; a row beyond the last plane
295    /// is not gridded ([`Self::admits`]).
296    #[must_use]
297    pub fn plane_of(&self, w_lambda: f64) -> usize {
298        self.plane_index(w_lambda).min(self.planes.len() - 1)
299    }
300
301    /// `nint(√(wScale · |w|))` before clamping.
302    fn plane_index(&self, w_lambda: f64) -> usize {
303        (self.w_scale * w_lambda.abs()).sqrt().round() as usize
304    }
305
306    /// Whether `wprojgrid.f` grids a row at `w_lambda`: `owp` rejects a
307    /// plane index past the last plane (`loc(3) > wconvsize`).
308    #[must_use]
309    pub fn admits(&self, w_lambda: f64) -> bool {
310        self.plane_index(w_lambda) < self.planes.len()
311    }
312
313    /// Half support of plane `plane`.
314    #[must_use]
315    pub fn half_support(&self, plane: usize) -> u16 {
316        self.planes[plane].support[0] / 2
317    }
318
319    /// Bytes of the tap values.
320    #[must_use]
321    pub fn bytes(&self) -> usize {
322        self.planes.iter().map(DenseCell::bytes).sum()
323    }
324}

326impl ConvolutionFunctionSet for WPlanes {

328        CfKey {
329            group: u16::try_from(self.plane_of(w_lambda)).expect("plane index fits u16"),

331        }

334    fn taps<'s>(&'s self, key: CfKey, _hold: &'s mut CellHold) -> TapLayout<'s> {
335        self.planes[usize::from(key.group)].layout()

338    fn max_half_support(&self) -> [u16; 2] {
339        [self.max_half_support; 2]
340    }
341
342    fn weight_taps<'s>(&'s self, _key: CfKey, _hold: &'s mut CellHold) -> Option<TapLayout<'s>> {
343        None
344    }
345
346    fn mueller(&self) -> &MuellerRouting {
347        &self.mueller
348    }
349
350    fn image_correction(&self) -> &ImageCorrection {
351        &self.correction
352    }
353
354    /// `wprojgrid.f`: `norm += real(cwt)`, `sumwt += weight · norm`; the
355    /// degrid does not divide.
356    fn normalisation(&self) -> KernelNormalisation {
357        KernelNormalisation::RealSum
358    }
359
360    fn pointing_ramp(&self) -> bool {
361        false
362    }
363
364    fn admits(&self, w_lambda: f64) -> bool {
365        Self::admits(self, w_lambda)
366    }

369/// `WPConvFunc::makeGWplane`: the taper times the w-screen over the inner
370/// `inner × inner` pixels of a centred `conv_size × conv_size` plane, zero
371/// outside; `w` in wavelengths.
372fn make_w_plane(
373    screen: &mut [Complex64],
374    conv_size: usize,
375    inner: usize,
376    increment: [f64; 2],
377    taper: &[f64],
378    w: f64,
379) {
380    screen.fill(Complex64::default());
381    let centre = (conv_size / 2) as i64;
382    let half = (inner / 2) as i64;
383    let two_pi_w = std::f64::consts::TAU * w;
384    for iy in -half..half {
385        let m = increment[1] * iy as f64;
386        let m_sq = m * m;
387        let taper_y = taper[(iy + half) as usize];
388        for ix in -half..half {
389            let l = increment[0] * ix as f64;
390            let r_sq = l * l + m_sq;
391            if r_sq < 1.0 {
392                let phase = two_pi_w * ((1.0 - r_sq).sqrt() - 1.0);
393                let taper_x = taper[(ix + half) as usize];
394                screen[((iy + centre) * conv_size as i64 + ix + centre) as usize] =
395                    Complex64::from_polar(taper_x * taper_y, phase);
396            }
397        }
398    }
