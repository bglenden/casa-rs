19#[derive(Default)]
20pub(super) struct ImageTileVolume(pub(super) i32);
21
22impl<'a, 'py> FromPyObject<'a, 'py> for ImageTileVolume {
23    type Error = PyErr;
24
25    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
26        if value.is_instance_of::<PyFloat>()
27            || value.is_instance_of::<PyString>()
28            || value.is_instance_of::<PyBytes>()
29            || value.is_instance_of::<PyList>()
30            || value.is_instance_of::<PyDict>()
31        {
32            return Err(PyTypeError::new_err(
33                "argument _imagetilevol must be an integer",
34            ));
35        }
36        // The legacy Python wrapper reads a signed long and narrows it
37        // to the imager's signed Int before it computes a tile shape.
38        value
39            .extract::<i64>()
40            .map(|volume| Self(volume as i32))
41            .map_err(|_| {
42                PySystemError::new_err(
43                    "<built-in function imager_setoptions> returned a result with an exception set",
44                )
45            })
46    }
47}

49pub(super) struct TransformArguments<'a> {
50    pub(super) machine: &'a str,
51    pub(super) cache: i64,
52    pub(super) tile: i32,
53    pub(super) convolution: &'a str,
54    pub(super) location: &'a Variant,
55    pub(super) padding: f64,
56    pub(super) interpolation: &'a str,
57    pub(super) planes: i32,
58    pub(super) single_precision: bool,
59    pub(super) threads: i32,
60}
61
62impl TransformArguments<'_> {
63    pub(super) fn options(
64        self,
65        mut previous: ImagerTransformOptions,
66    ) -> PyResult<ImagerTransformOptions> {
67        let machine = self.machine.to_ascii_lowercase();
68        let convolution = self.convolution.to_ascii_lowercase();
69        previous.single_dish = if matches!(machine.as_str(), "sd" | "both") {
70            let kernel = ImagerSingleDishKernel::from_name(&convolution).ok_or_else(|| {
71                runtime_error(format!(
72                    "Unknown single-dish grid function {}",
73                    self.convolution
74                ))
75            })?;
76            Some(ImagerSingleDish {
77                kernel,
78                combined: machine == "both",
79            })
80        } else {
81            None
82        };
83        if machine != "sd" && matches!(convolution.as_str(), "gauss" | "gjinc") {
84            return Err(runtime_error(format!(
85                "Grid function {} is available only for single dish imaging",
86                self.convolution,
87            )));
88        }
89        previous.gridder = match machine.as_str() {
90            "wproject" => ImagerGridder::w_project(self.planes),
91            "mosaic" => ImagerGridder::Mosaic,
92            "sd" | "both" | "pbwproject" | "pbmosaic" | "awproject" | "mawproject" => {
93                ImagerGridder::Spheroidal
94            }
95            // CASA's rGridFT (nift) and GridFT give the same predictions.
96            _ => match convolution.as_str() {
97                "box" => ImagerGridder::Boxcar,
98                "sf" => ImagerGridder::Spheroidal,
99                _ => {
100                    return Err(runtime_error(format!(
101                        "Unknown interferometric grid function {}",
102                        self.convolution
103                    )));
104                }
105            },
106        };
107        if self.cache > 0 {
108            previous.cache_size = self.cache;
109        }
110        if self.tile > 0 {
111            previous.tile_size = self.tile;
112        }
113        if self.padding >= 1.0 {
114            previous.padding = self.padding as f32;
115        }
116        let interpolation = self.interpolation.to_ascii_lowercase();
117        previous.frequency_interpolation = if interpolation.contains("linear") {
118            1
119        } else if interpolation.contains("splin") {
120            3
121        } else if interpolation.contains("cub") {
122            2
123        } else {
124            0
125        };
126        previous.observatory = if self.location.is_empty_sequence()
127            || matches!(self.location, Variant::Text(value) if value.is_empty())
128        {
129            None
130        } else {
131            position(
132                self.location,
133                &casa_ms::open_measures_runtime().map_err(runtime_error)?,
134            )?
135            .map(|position| {
136                let itrf = position
137                    .convert_to(PositionRef::ITRF)
138                    .map_err(runtime_error)?;
139                let (longitude, latitude, radius) = itrf.as_spherical();
140                Ok::<_, PyErr>([longitude, latitude, radius])
141            })
142            .transpose()?
143        };
144        previous.threads = self.threads;
145        previous.use_single_precision(self.single_precision);
146        Ok(previous)
147    }
148}
