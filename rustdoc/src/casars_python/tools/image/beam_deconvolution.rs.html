3//! Thin PyO3 `ia.beamforconvolvedsize` / `ia.deconvolvefrombeam` wrapper delegating to [`casa_imageanalysis::tool::beam_ops`].
4
5use casa_core::quanta::Quantity;
6use casa_imageanalysis::tool::beam_ops::{beam_for_convolved_size, deconvolve_from_beam};
7use pyo3::prelude::*;
8
9use crate::convert::{Variant, read_quantity, value::CasaValue};
10
11use super::PyImageTool;
12use super::handle::raised;
13
14fn quantities(argument: Option<Variant>, label: &str) -> PyResult<Vec<Quantity>> {
15    let strings = match argument {
16        Some(Variant::Text(value)) => {
17            let separator = if value.contains(',') { ',' } else { ' ' };
18            if value.contains(separator) {
19                value
20                    .split(separator)
21                    .filter(|part| !part.is_empty() && *part != " ")
22                    .map(str::to_owned)
23                    .collect()
24            } else {
25                vec![value]
26            }
27        }
28        Some(Variant::Texts(values)) => values,
29        _ => return Err(raised(format!("Cannot understand {label} values"))),
30    };
31    strings
32        .iter()
33        .map(|value| {
34            if value.contains("pix") {
35                // CASA accepts this unit in the list reader. All beam
36                // components that use it fail the angular-unit check.
37                return Quantity::new(0.0, "pix").map_err(raised);
38            }
39            read_quantity(value, &format!(" In converting quantity {value}")).map_err(Into::into)
40        })
41        .collect()
42}
43
44#[pymethods]
45impl PyImageTool {
46    #[allow(clippy::unused_self)]
47    #[pyo3(signature = (source=None, convolved=None), text_signature = "($self, source=[], convolved=[])")]
48    fn beamforconvolvedsize(
49        &self,
50        py: Python<'_>,
51        source: Option<Variant>,
52        convolved: Option<Variant>,
53    ) -> PyResult<Py<PyAny>> {
54        py.detach(|| {
55            let source = quantities(source, "source")?;
56            let convolved = quantities(convolved, "target")?;
57            let record = beam_for_convolved_size(&source, &convolved).map_err(raised)?;
58            Ok::<_, PyErr>(CasaValue::from_record(&record))
59        })?
60        .into_python(py)
61    }
62
63    #[allow(clippy::unused_self)]
64    #[pyo3(signature = (source=None, beam=None), text_signature = "($self, source=[], beam=[])")]
65    fn deconvolvefrombeam(
66        &self,
67        py: Python<'_>,
68        source: Option<Variant>,
69        beam: Option<Variant>,
70    ) -> PyResult<Py<PyAny>> {
71        py.detach(|| {
72            let source = quantities(source, "source")?;
73            let beam = quantities(beam, "beam")?;
74            let record = deconvolve_from_beam(&source, &beam).map_err(raised)?;
75            Ok::<_, PyErr>(CasaValue::from_record(&record))
76        })?
77        .into_python(py)
78    }
79}
