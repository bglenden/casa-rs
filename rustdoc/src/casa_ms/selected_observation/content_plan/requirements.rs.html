12/// Payload-free, identity-bound memory requirements for one selected source.
13///
14/// The storage owner derives this curve under a short-lived read lock. It includes
15/// retained metadata, source-specific plans, POINTING construction, and overlapping
16/// block preparation. It retains neither table handles nor a prepared catalog.
17#[derive(Debug, Clone, Copy, PartialEq, Eq)]
18pub struct SelectedObservationContentRequirements {
19    pub(super) problem: CompiledProblemId,
20    pub(super) provenance: ObservationProvenanceId,
21    pub(super) source: MeasurementSetIdentity,
22    pub(super) retained_bytes: usize,
23    pub(super) initialization_scratch_bytes: usize,
24    pub(super) initialization_scan_bytes_per_row: usize,
25    pub(super) pointing_reference_scratch_bytes: usize,
26    pub(super) traversal_base_bytes: usize,
27    pub(super) traversal_pointing_reference_scratch_bytes: usize,
28    pub(super) resident_bytes_per_row: usize,
29    pub(super) fill_bytes_per_row: usize,
30    pub(super) preparation_bytes_per_row: usize,
31    pub(super) fill_fixed_bytes: usize,
32    pub(super) selected_rows: usize,
33    pub(super) maximum_pointing_polynomial_terms: usize,
34}
35
36impl SelectedObservationContentRequirements {
37    pub(crate) fn matches(
38        &self,
39        problem: &CompiledProblem,
40        source: MeasurementSetIdentity,
41    ) -> bool {
42        self.problem == problem.problem_id()
43            && self.provenance == problem.inputs().observation_snapshot().provenance_id()
44            && self.source == source
45    }
46
47    /// Return the smallest complete envelope admitting one selected row.
48    pub fn minimum_bytes(
49        self,
50        maximum_live_blocks: usize,
51    ) -> Result<usize, SelectedObservationContentPlanError> {
52        self.bytes_for_rows(1, maximum_live_blocks)
53    }
54
55    /// Return the full initialization/traversal envelope for a bounded block size.
56    ///
57    /// Rows beyond the selected source's row count do not enlarge the block.
58    pub fn bytes_for_rows(
59        self,
60        rows: usize,
61        maximum_live_blocks: usize,
62    ) -> Result<usize, SelectedObservationContentPlanError> {
63        if rows == 0 || self.selected_rows == 0 || maximum_live_blocks == 0 {
64            return Err(SelectedObservationContentPlanError::InvalidBudget);
65        }
66        let rows = rows.min(self.selected_rows);
67        let overflow = SelectedObservationContentPlanError::ByteOverflow;
68        let prior_resident = self
69            .resident_bytes_per_row
70            .checked_mul(maximum_live_blocks - 1)
71            .ok_or(overflow)?;
72        let replay = super::BoundObservationSource::row_replay_bytes_per_row();
73        let fill_per_row = prior_resident
74            .checked_add(self.fill_bytes_per_row)
75            .and_then(|bytes| bytes.checked_add(replay))
76            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
77        let prepare_per_row = prior_resident
78            .checked_add(self.preparation_bytes_per_row)
79            .and_then(|bytes| bytes.checked_add(replay))
80            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
81        let base = self.traversal_base(maximum_live_blocks)?;
82        let fill = rows
83            .checked_mul(fill_per_row)
84            .and_then(|bytes| bytes.checked_add(self.fill_fixed_bytes))
85            .and_then(|bytes| bytes.checked_add(base))
86            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
87        let prepare = rows
88            .checked_mul(prepare_per_row)
89            .and_then(|bytes| bytes.checked_add(self.traversal_pointing_reference_scratch_bytes))
90            .and_then(|bytes| bytes.checked_add(base))
91            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
92        let initialization = rows
93            .checked_mul(self.initialization_scan_bytes_per_row)
94            .and_then(|bytes| bytes.checked_add(self.initialization_scratch_bytes))
95            .and_then(|bytes| bytes.checked_add(self.retained_bytes))
96            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?;
97        Ok(initialization.max(fill).max(prepare))
98    }
99
100    fn traversal_base(
101        self,
102        maximum_live_blocks: usize,
103    ) -> Result<usize, SelectedObservationContentPlanError> {
104        maximum_live_blocks
105            .checked_add(1)
106            .and_then(|blocks| blocks.checked_mul(size_of::<BufferedObservationBlock>()))
107            .and_then(|bytes| bytes.checked_add(self.traversal_base_bytes))
108            .ok_or(SelectedObservationContentPlanError::ByteOverflow)
109    }
110
111    /// Return the POINTING polynomial bound used to derive this curve.
112    #[must_use]
113    pub const fn maximum_pointing_polynomial_terms(self) -> usize {
114        self.maximum_pointing_polynomial_terms
115    }
116
117    /// Select the largest bounded block that fits an explicit physical budget.
118    pub fn plan(
119        self,
120        budget: SelectedObservationContentBudget,
121    ) -> Result<SelectedObservationContentPlan, SelectedObservationContentPlanError> {
122        if budget.available_bytes() == 0
123            || budget.maximum_live_blocks() == 0
124            || budget.maximum_pointing_polynomial_terms() != self.maximum_pointing_polynomial_terms
125            || self.maximum_pointing_polynomial_terms == 0
126        {
127            return Err(SelectedObservationContentPlanError::InvalidBudget);
128        }
129        let fixed = self
130            .retained_bytes
131            .checked_add(self.initialization_scratch_bytes)
132            .ok_or(SelectedObservationContentPlanError::ByteOverflow)?
133            .max(self.traversal_base(budget.maximum_live_blocks())?);
134        if fixed > budget.available_bytes() {
135            return Err(
136                SelectedObservationContentPlanError::InsufficientRetainedBudget {
137                    required_bytes: fixed,
138                    available_bytes: budget.available_bytes(),
139                },
140            );
141        }
142        let minimum = self.minimum_bytes(budget.maximum_live_blocks())?;
143        if minimum > budget.available_bytes() {
144            return Err(SelectedObservationContentPlanError::InsufficientBudget {
145                required_bytes: minimum,
146                available_bytes: budget.available_bytes(),
147            });
148        }
149        let mut low = 1;
150        let mut high = self.selected_rows;
151        while low < high {
152            let middle = low + (high - low).div_ceil(2);
153            if self.bytes_for_rows(middle, budget.maximum_live_blocks())?
154                <= budget.available_bytes()
155            {
156                low = middle;
157            } else {
158                high = middle - 1;
159            }
160        }
161        let rows = low;
162        Ok(SelectedObservationContentPlan {
163            retained_bytes: self.retained_bytes,
164            initialization_scratch_bytes: self
165                .initialization_scan_bytes_per_row
166                .checked_mul(rows)
167                .and_then(|bytes| bytes.checked_add(self.initialization_scratch_bytes))
168                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
169            pointing_reference_scratch_bytes: self.pointing_reference_scratch_bytes,
170            resident_bytes_per_row: self.resident_bytes_per_row,
171            preparation_bytes_per_row: self.preparation_bytes_per_row,
172            rows_per_block: rows,
173            resident_bytes_per_block: rows
174                .checked_mul(self.resident_bytes_per_row)
175                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
176            preparation_bytes_per_block: rows
177                .checked_mul(self.preparation_bytes_per_row)
178                .ok_or(SelectedObservationContentPlanError::ByteOverflow)?,
179            maximum_resident_bytes: self.bytes_for_rows(rows, budget.maximum_live_blocks())?,
180            maximum_live_blocks: budget.maximum_live_blocks(),
181            maximum_pointing_polynomial_terms: self.maximum_pointing_polynomial_terms,
182        })
183    }
