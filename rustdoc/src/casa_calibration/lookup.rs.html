2//! Calibration-table lookup: a table's solution for an antenna, spectral
3//! window, field and time, found and interpolated in time as the apply
4//! executor finds it, for callers that apply a table to visibilities of
5//! their own (`casa-simulation`'s `sm.setapply`).
6//!
7//! [`CalibrationTableLookup`] loads a complex (`CPARAM`) table as the
8//! executor does and answers [`CalibrationTableLookup::solution`]: the
9//! solutions of the row's field, or of the table's sole field, nearest in
10//! time or interpolated linearly in amplitude and phase. With
11//! [`FieldSelection::AnyField`] it takes every field's solutions as one
12//! series in time instead, as CASA's simulator applies a table. The time
13//! interpolation the executor shares lives here.
14//!
15//! Loading reads the whole table; [`CalibrationTableLookup::rows_on_disk`],
16//! [`CalibrationTableLookup::loading_bytes`] and
17//! [`CalibrationTableLookup::kept_bytes`] size narrow, two-parameter rows.
18//! For wide tables, [`CalibrationTableLookup::layout_on_disk`] inspects
19//! shape headers and counts parameter/frequency values before their payloads
20//! are read, so callers can admit the entire lookup first.
21

37/// Which fields' solutions a row takes.
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub enum FieldSelection {
40    /// The row's field's, or the table's sole field's: the apply
41    /// executor's default.
42    RowField,
43    /// Every field's, as one series in time: CASA's `Simulator::setapply`
44    /// ignores fields and does not partition the table by them (#893
45    /// review).
46    AnyField,
47}
48
49/// Heap loading a complex table holds per row at most: the row as
50/// casa-tables reads the table, and the solution it becomes. Measured at
51/// about 3,300 bytes (the #893 review's 9,996-row probe, and the setapply
52/// law in `casa-simulation`'s `session_memory/apply.rs`), with a quarter
53/// more for the allocator.
54pub const LOOKUP_LOADING_ROW_BYTES: u64 = 4_096;
55
56/// Heap a loaded lookup keeps per row: the solution's grid, its place in
57/// the table's series, and in the series across fields. Measured at about
58/// 320 bytes with the loading figure, with a quarter more.
59pub const LOOKUP_KEPT_ROW_BYTES: u64 = 400;
60
61/// Payload dimensions used to admit a complex calibration-table lookup.
62///
63/// Row metadata and numerical values have separate costs: a wide table
64/// cannot be sized from its row count alone.
65#[derive(Debug, Clone, Copy, PartialEq, Eq)]
66pub struct CalibrationTableLayout {
67    /// Physical rows in the calibration table.
68    pub rows: usize,
69    /// Complex parameters across all defined CPARAM cells.
70    pub values: usize,
71    /// Largest CPARAM cell, for admitted interpolation scratch.
72    pub max_cell_values: usize,
73    /// CHAN_FREQ values needed for frequency-dependent solutions.
74    pub frequency_values: usize,
75    /// SPECTRAL_WINDOW rows used for mapping and frequency-grid metadata.
76    pub frequency_windows: usize,
77}
78
79impl CalibrationTableLayout {
80    fn extra_values(self) -> u64 {
81        self.values.saturating_sub(self.rows.saturating_mul(2)) as u64
82    }
83
84    /// Heap used temporarily while loading, beside the retained lookup.
85    /// Existing per-row admission includes up to two complex parameters.
86    #[must_use]
87    pub fn loading_bytes(self) -> u64 {
88        (self.rows as u64)
89            .saturating_mul(LOOKUP_LOADING_ROW_BYTES)
90            .saturating_add(self.extra_values().saturating_mul(24))
91            .saturating_add((self.frequency_values as u64).saturating_mul(16))
92            .saturating_add(
93                (self.frequency_windows as u64).saturating_mul(LOOKUP_LOADING_ROW_BYTES),
94            )
95    }
96
97    /// Heap retained by the lookup, including wide parameter grids.
98    #[must_use]
99    pub fn kept_bytes(self) -> u64 {
100        (self.rows as u64)
101            .saturating_mul(LOOKUP_KEPT_ROW_BYTES)
102            .saturating_add(self.extra_values().saturating_mul(12))
103            .saturating_add((self.frequency_values as u64).saturating_mul(12))
104            .saturating_add((self.frequency_windows as u64).saturating_mul(64))
105    }
106}

108/// A complex calibration table, loaded to look its solutions up.
109pub struct CalibrationTableLookup {
110    path: PathBuf,
111    table: LoadedCalibrationTable,
112    subtype: String,
113    interp: ApplyInterpolationMode,
114    /// For [`FieldSelection::AnyField`]: each (spectral window, antenna)'s
115    /// solutions across the fields, in time order.
116    across_fields: Option<HashMap<(i32, i32), Vec<CalibrationSolution>>>,
117}
118
119/// One antenna's solution: CPARAM's values and FLAG, per receptor and
120/// channel.
121#[derive(Debug, Clone, PartialEq)]
122pub struct ComplexSolution {
123    /// Receptors (CPARAM's first axis).
124    pub receptors: usize,
125    /// Channels (its second axis): one for a term that does not depend on
126    /// frequency.
127    pub channels: usize,
128    /// `[receptor · channels + channel]`.
129    pub values: Vec<Complex32>,
130    /// `[receptor · channels + channel]`.
131    pub flags: Vec<bool>,
132}
133
134impl ComplexSolution {
135    fn of(grid: &GainGrid) -> Self {
136        let (receptors, channels) = (grid.receptor_count, grid.channel_count);
137        let mut values = Vec::with_capacity(receptors * channels);
138        let mut flags = Vec::with_capacity(receptors * channels);
139        for receptor in 0..receptors {
140            for channel in 0..channels {
141                let (value, flag) = if grid.values.ndim() == 1 {
142                    (grid.values[[receptor]], grid.flags[[receptor]])
143                } else {
144                    (
145                        grid.values[[receptor, channel]],
146                        grid.flags[[receptor, channel]],
147                    )
148                };
149                values.push(value);
150                flags.push(flag);
151            }
152        }
153        Self {
154            receptors,
155            channels,
156            values,
157            flags,
158        }
159    }
160
161    /// The value of `receptor` in `channel`.
162    #[must_use]
163    pub fn value(&self, receptor: usize, channel: usize) -> Complex32 {
164        self.values[receptor * self.channels + channel]
165    }
166}

168impl CalibrationTableLookup {
169    /// Load the complex calibration table at `path` to look its solutions
170    /// up with `interp`, each row taking its own field's
171    /// ([`FieldSelection::RowField`]).
172    ///
173    /// # Errors
174    ///
175    /// The table cannot be read, or holds other than complex (`CPARAM`)
176    /// solutions.
177    pub fn open(
178        path: impl AsRef<Path>,
179        interp: ApplyInterpolationMode,
180    ) -> Result<Self, ApplyExecutionError> {
181        Self::open_with(path, interp, FieldSelection::RowField)
182    }
183
184    /// The rows of the table at `path`, read without loading them.
185    ///
186    /// # Errors
187    ///
188    /// The table's control file cannot be read.
189    pub fn rows_on_disk(path: impl AsRef<Path>) -> Result<usize, ApplyExecutionError> {
190        let path = path.as_ref();
191        Table::rows_on_disk(path).map_err(|error| {
192            ApplyExecutionError::UnsupportedCalibrationTable {
193                path: path.display().to_string(),
194                reason: error.to_string(),
195            }
196        })
197    }
198
199    /// Admit the root and spectral-window row metadata before inspecting
200    /// shapes. This reads control files only, never numerical payloads.
201    ///
202    /// # Errors
203    /// A table control file cannot be read.
204    pub fn metadata_bytes_on_disk(path: impl AsRef<Path>) -> Result<u64, ApplyExecutionError> {
205        let path = path.as_ref();
206        let rows = Self::rows_on_disk(path)?;
207        let windows = spectral_window_path(path)?
208            .map(Self::rows_on_disk)
209            .transpose()?
210            .unwrap_or(0);
211        Ok((rows as u64)
212            .saturating_add(windows as u64)
213            .saturating_mul(LOOKUP_LOADING_ROW_BYTES)
214            .max(64 * 1024))
215    }
216
217    /// Inspect CPARAM dimensions without reading numerical cell payloads.
218    ///
219    /// The caller admits [`Self::metadata_bytes_on_disk`] first. Shapes
220    /// are scanned in bounded batches; only scalar counts survive here.
221    ///
222    /// # Errors
223    /// A table's shape metadata cannot be read, or value counts overflow.
224    pub fn layout_on_disk(
225        path: impl AsRef<Path>,
226    ) -> Result<CalibrationTableLayout, ApplyExecutionError> {
227        let path = path.as_ref();
228        let error = |reason: String| ApplyExecutionError::UnsupportedCalibrationTable {
229            path: path.display().to_string(),
230            reason,
231        };
232        let (rows, values, max_cell_values, channels) =
233            shape_counts(path, "CPARAM").map_err(error)?;
234        let spectral = spectral_window_path(path)?;
235        let (frequency_windows, frequency_values) = if let Some(spectral) = spectral {
236            if channels > 1 {
237                let (windows, values, _, _) =
238                    shape_counts(&spectral, "CHAN_FREQ").map_err(error)?;
239                (windows, values)
240            } else {
241                (Self::rows_on_disk(spectral)?, 0)
242            }
243        } else {
244            (0, 0)
245        };
246        Ok(CalibrationTableLayout {
247            rows,
248            values,
249            max_cell_values,
250            frequency_values,
251            frequency_windows,
252        })
253    }
254
255    /// Heap loading a table of `rows` rows holds at most, beside what the
256    /// lookup then keeps.
257    #[must_use]
258    pub const fn loading_bytes(rows: usize) -> u64 {
259        rows as u64 * LOOKUP_LOADING_ROW_BYTES
260    }
261
262    /// Heap a lookup of a table of `rows` rows keeps.
263    #[must_use]
264    pub const fn kept_bytes(rows: usize) -> u64 {
265        rows as u64 * LOOKUP_KEPT_ROW_BYTES
266    }

268    /// [`Self::open`], each row taking the solutions `fields` selects.
269    ///
270    /// # Errors
271    ///
272    /// As [`Self::open`].
273    pub fn open_with(
274        path: impl AsRef<Path>,
275        interp: ApplyInterpolationMode,
276        fields: FieldSelection,
277    ) -> Result<Self, ApplyExecutionError> {
278        let path = path.as_ref();
279        let unsupported = |reason: String| ApplyExecutionError::UnsupportedCalibrationTable {
280            path: path.display().to_string(),
281            reason,
282        };
283        let summary = summarize_table(path).map_err(|error| unsupported(error.to_string()))?;
284        if summary.parameter_family != CalibrationParameterFamily::Complex
285            || summary.table_subtype == "BPOLY"
286        {
287            return Err(unsupported(format!(
288                "a {} table is not complex (CPARAM)",
289                summary.table_subtype
290            )));
291        }
292        let subtype = summary.table_subtype.clone();
293        let plan = ApplyCalibrationTablePlan {
294            spec: ApplyCalibrationTableSpec {
295                interp,
296                ..ApplyCalibrationTableSpec::new(path)
297            },
298            applicable_selected_row_count: 0,
299            summary,
300            resolved_gainfield: None,
301            resolved_nearest_gainfields: Vec::new(),
302            spw_mapping: Vec::new(),
303            calibration_spectral_windows: Vec::new(),
304            interp,
305            calwt: false,
306        };
307        let table = load_calibration_table(&plan, None)?;
308        let across_fields = (fields == FieldSelection::AnyField).then(|| across_fields(&table));
309        Ok(Self {
310            path: path.to_path_buf(),
311            table,
312            subtype,
313            interp,
314            across_fields,
315        })
316    }
317
318    /// Whether this spectral window has any loaded solution series.
319    ///
320    /// A missing antenna is distinct from a missing entire window: CASA
321    /// substitutes neutral antenna values in the former case, but skips a
322    /// whole corruption step in the latter.
323    #[must_use]
324    pub fn has_spectral_window(&self, window: i32) -> bool {
325        self.table
326            .solutions
327            .keys()
328            .any(|&(_, spw, _)| spw == window)
329    }
330
331    /// First spectral-window id with actual calibration solutions.
332    /// CASA's automatic `spwmap=[-999]` fans this window out to all data windows.
333    #[must_use]
334    pub fn first_spectral_window(&self) -> Option<i32> {
335        self.table
336            .solutions
337            .keys()
338            .map(|&(_, window, _)| window)
339            .filter(|&window| window >= 0)
340            .min()
341    }
342
343    /// The table's subtype (`G Jones`, `D Jones`, `B Jones`, …).
344    #[must_use]
345    pub fn subtype(&self) -> &str {
346        &self.subtype
347    }
348
349    /// The solution of `antenna` in spectral window `spw` for a row of
350    /// field `field` at `time_s`: from the field's solutions, or the
351    /// table's sole field's (or, with [`FieldSelection::AnyField`], every
352    /// field's), nearest in time or interpolated linearly in amplitude and
353    /// phase. `None` where the table has none.
354    ///
355    /// # Errors
356    ///
357    /// The solutions cannot be interpolated (shapes differ in time).
358    pub fn solution(
359        &self,
360        field: i32,
361        spw: i32,
362        antenna: i32,
363        time_s: f64,
364    ) -> Result<Option<ComplexSolution>, ApplyExecutionError> {
365        let grid = match &self.across_fields {
366            Some(series) => self.across_fields_grid(series, spw, antenna, time_s)?,
367            None => self
368                .table
369                .lookup(field, spw, antenna, time_s, self.interp)?,
370        };
371        let Some(grid) = grid else {
372            return Ok(None);
373        };
374        match grid.as_ref() {
375            CalibrationGrid::Complex(grid) => Ok(Some(ComplexSolution::of(grid))),
376            _ => Err(ApplyExecutionError::UnsupportedCalibrationTable {
377                path: String::new(),
378                reason: "a complex table gave other than complex solutions".to_string(),
379            }),
380        }
381    }
382}

384impl CalibrationTableLookup {
385    /// The grid of `antenna` in window `spw` at `time_s` from the series
386    /// across fields: the nearest solution (the earlier of two as near), or
387    /// the bracketing pair interpolated as the executor interpolates.
388    fn across_fields_grid(
389        &self,
390        series: &HashMap<(i32, i32), Vec<CalibrationSolution>>,
391        spw: i32,
392        antenna: i32,
393        time_s: f64,
394    ) -> Result<Option<Arc<CalibrationGrid>>, ApplyExecutionError> {
395        let Some(candidates) = series
396            .get(&(spw, antenna))
397            .or_else(|| series.get(&(spw, -1)))
398            .filter(|candidates| !candidates.is_empty())
399        else {
400            return Ok(None);
401        };
402        // The first solution at or after the time, and the last at or
403        // before it.
404        let after = candidates.partition_point(|solution| solution.time_seconds < time_s);
405        let through = candidates.partition_point(|solution| solution.time_seconds <= time_s);
406        match self.interp {
407            ApplyInterpolationMode::Nearest | ApplyInterpolationMode::NearestLinear => {
408                let nearest = match (after.checked_sub(1), candidates.get(after)) {
409                    (Some(before), Some(later))
410                        if (later.time_seconds - time_s).abs()
411                            < (time_s - candidates[before].time_seconds).abs() =>
412                    {
413                        after
414                    }
415                    (Some(before), _) => before,
416                    (None, _) => 0,
417                };
418                Ok(Some(Arc::clone(&candidates[nearest].grid)))
419            }
420            ApplyInterpolationMode::Linear => {
421                let last = candidates.len() - 1;
422                let lower = through.checked_sub(1).unwrap_or(after.min(last));
423                let upper = after.min(last).max(lower);
424                interpolate_time_linear(&self.path, &candidates[lower..=upper], time_s).map(Some)
425            }
426        }
427    }
428}

430/// Each (spectral window, antenna)'s solutions across the fields of
431/// `table`, in time order, fields in order at equal times.
432fn across_fields(table: &LoadedCalibrationTable) -> HashMap<(i32, i32), Vec<CalibrationSolution>> {
433    let mut by_window: HashMap<(i32, i32), Vec<(i32, &CalibrationSolution)>> = HashMap::new();
434    for (&(field, spw, antenna), solutions) in &table.solutions {
435        by_window
436            .entry((spw, antenna))
437            .or_default()
438            .extend(solutions.iter().map(|solution| (field, solution)));
439    }
440    by_window
441        .into_iter()
442        .map(|(key, mut solutions)| {
443            solutions.sort_by(|a, b| {
444                a.1.time_seconds
445                    .total_cmp(&b.1.time_seconds)
446                    .then(a.0.cmp(&b.0))
447            });
448            let series = solutions
449                .into_iter()
450                .map(|(_, solution)| CalibrationSolution {
451                    time_seconds: solution.time_seconds,
452                    grid: Arc::clone(&solution.grid),
453                })
454                .collect();
455            (key, series)
456        })
457        .collect()
458}

460pub(crate) fn interpolate_time_linear(
461    path: &Path,
462    candidates: &[CalibrationSolution],
463    time_seconds: f64,
464) -> Result<Arc<CalibrationGrid>, ApplyExecutionError> {
465    let mut sorted = candidates.iter().collect::<Vec<_>>();
466    sorted.sort_by(|a, b| a.time_seconds.total_cmp(&b.time_seconds));
467
468    let lower = sorted
469        .iter()
470        .rev()
471        .find(|solution| solution.time_seconds <= time_seconds)
472        .copied();
473    let upper = sorted
474        .iter()
475        .find(|solution| solution.time_seconds >= time_seconds)
476        .copied();
477
478    match (lower, upper) {
479        (Some(lower), Some(upper))
480            if (upper.time_seconds - lower.time_seconds).abs() > f64::EPSILON =>
481        {
482            let fraction = ((time_seconds - lower.time_seconds)
483                / (upper.time_seconds - lower.time_seconds)) as f32;
484            match (lower.grid.as_ref(), upper.grid.as_ref()) {
485                (CalibrationGrid::Complex(lower), CalibrationGrid::Complex(upper)) => {
486                    if lower.values.shape() != upper.values.shape() {
487                        return Err(ApplyExecutionError::UnsupportedInterpolation {
488                            path: path.display().to_string(),
489                            interp: ApplyInterpolationMode::Linear,
490                            reason: "time interpolation requires matching parameter shapes"
491                                .to_string(),
492                        });
493                    }
494                    let mut values = lower.values.clone();
495                    for (value, upper_value) in values.iter_mut().zip(upper.values.iter()) {
496                        *value = interpolate_gain_amplitude_phase(*value, *upper_value, fraction);
497                    }
498                    let mut flags = lower.flags.clone();
499                    for (flag, upper_flag) in flags.iter_mut().zip(upper.flags.iter()) {
500                        *flag = *flag || *upper_flag;
501                    }
502                    Ok(Arc::new(CalibrationGrid::Complex(GainGrid {
503                        receptor_count: lower.receptor_count,
504                        channel_count: lower.channel_count,
505                        values,
506                        flags,
507                    })))
508                }
509                (CalibrationGrid::Delay(lower), CalibrationGrid::Delay(upper)) => {
510                    if lower.values_ns.shape() != upper.values_ns.shape() {
511                        return Err(ApplyExecutionError::UnsupportedInterpolation {
512                            path: path.display().to_string(),
513                            interp: ApplyInterpolationMode::Linear,
514                            reason: "time interpolation requires matching parameter shapes"
515                                .to_string(),
516                        });
517                    }
518                    let mut values_ns = lower.values_ns.clone();
519                    for (value, upper_value) in values_ns.iter_mut().zip(upper.values_ns.iter()) {
520                        *value = *value + (*upper_value - *value) * fraction;
521                    }
522                    let mut flags = lower.flags.clone();
523                    for (flag, upper_flag) in flags.iter_mut().zip(upper.flags.iter()) {
524                        *flag = *flag || *upper_flag;
525                    }
526                    Ok(Arc::new(CalibrationGrid::Delay(DelayGrid {
527                        receptor_count: lower.receptor_count,
528                        channel_count: lower.channel_count,
529                        values_ns,
530                        flags,
531                    })))
532                }
533                (
534                    CalibrationGrid::Antpos(_)
535                    | CalibrationGrid::GainCurve(_)
536                    | CalibrationGrid::Opacity(_),
537                    _,
538                )
539                | (
540                    _,
541                    CalibrationGrid::Antpos(_)
542                    | CalibrationGrid::GainCurve(_)
543                    | CalibrationGrid::Opacity(_),
544                ) => Err(ApplyExecutionError::UnsupportedInterpolation {
545                    path: path.display().to_string(),
546                    interp: ApplyInterpolationMode::Linear,
547                    reason:
548                        "VLA prior tables are row-geometry dependent and use nearest table rows"
549                            .to_string(),
550                }),
551                _ => Err(ApplyExecutionError::UnsupportedInterpolation {
552                    path: path.display().to_string(),
553                    interp: ApplyInterpolationMode::Linear,
554                    reason: "time interpolation requires matching calibration parameter families"
555                        .to_string(),
556                }),
557            }
558        }
559        (Some(lower), _) => Ok(lower.grid.clone()),
560        (_, Some(upper)) => Ok(upper.grid.clone()),
561        _ => Err(ApplyExecutionError::UnsupportedInterpolation {
562            path: path.display().to_string(),
563            interp: ApplyInterpolationMode::Linear,
564            reason: "no calibration rows available for linear interpolation".to_string(),
565        }),
566    }
567}

569pub(crate) fn interpolate_gain_amplitude_phase(
570    lower: Complex32,
571    upper: Complex32,
572    fraction: f32,
573) -> Complex32 {
574    let lower_amp = lower.norm();
575    let upper_amp = upper.norm();
576    let amp = lower_amp + (upper_amp - lower_amp) * fraction;
577    let lower_phase = lower.arg();
578    let mut upper_phase = upper.arg();
579    while upper_phase > lower_phase + std::f32::consts::PI {
580        upper_phase -= 2.0 * std::f32::consts::PI;
581    }
582    while upper_phase < lower_phase - std::f32::consts::PI {
583        upper_phase += 2.0 * std::f32::consts::PI;
584    }
585    let phase = lower_phase + (upper_phase - lower_phase) * fraction;
586    Complex32::new(amp * phase.cos(), amp * phase.sin())
587}

589/// Row count, logical value count, largest cell and largest second axis;
590/// arrays themselves are never read or retained during this inspection.
591fn shape_counts(path: &Path, name: &str) -> Result<(usize, usize, usize, usize), String> {
592    let table =
593        Table::open(casa_tables::TableOptions::new(path)).map_err(|error| error.to_string())?;
594    let rows = table.row_count();
595    let column = table
596        .column_accessor(name)
597        .map_err(|error| error.to_string())?;
598    let (mut values, mut maximum, mut channels) = (0usize, 0usize, 0usize);
599    for start in (0..rows).step_by(512) {
600        let selected = (start..rows.min(start.saturating_add(512))).collect::<Vec<_>>();
601        for shape in column
602            .array_shapes_uncached(&selected)
603            .map_err(|error| error.to_string())?
604            .into_iter()
605            .flatten()
606        {
607            channels = channels.max(shape.get(1).copied().unwrap_or(1));
608            let size = shape
609                .into_iter()
610                .try_fold(1usize, |count, dimension| count.checked_mul(dimension))
611                .ok_or_else(|| "array shape size overflows usize".to_string())?;
612            maximum = maximum.max(size);
613            values = values
614                .checked_add(size)
615                .ok_or_else(|| "array value count overflows usize".to_string())?;
616        }
617    }
618    Ok((rows, values, maximum, channels))
619}

621/// Resolve the ordinary table keyword, including linked external subtables.
622pub(crate) fn spectral_window_path(path: &Path) -> Result<Option<PathBuf>, ApplyExecutionError> {
623    let table = Table::open(casa_tables::TableOptions::new(path)).map_err(|error| {
624        ApplyExecutionError::UnsupportedCalibrationTable {
625            path: path.display().to_string(),
626            reason: error.to_string(),
627        }
628    })?;
629    Ok(table
