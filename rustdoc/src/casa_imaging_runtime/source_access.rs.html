2//! The bounded content budget of the selected observation's source.
3
4use casa_imaging_model::CompiledProblem;
5use casa_ms::{
6    BoundSelectedObservationError, ResolvedSelectedObservationAccess,
7    SelectedObservationContentBudget, SelectedObservationContentPlanError,
8};
9
10use crate::{ResourceAuthority, ResourceError, ResourcePolicy};
11
12/// The budget that bounds source inspection before the compiled problem
13/// can quote its complete initialization and traversal requirements.
14#[must_use]
15pub const fn bootstrap_source_budget() -> SelectedObservationContentBudget {
16    SelectedObservationContentBudget::new(64 << 20, 2, 4)
17}
18
19/// Why a source's execution envelope could not be finalized.
20#[derive(Debug, thiserror::Error)]
21pub enum SourceAccessError {
22    /// The storage owner could not quote or bind the source's requirements.
23    #[error("selected-observation access: {0}")]
24    Access(#[from] BoundSelectedObservationError),
25    /// The requirement curve could not be evaluated or planned.
26    #[error("selected-observation content plan: {0}")]
27    ContentPlan(#[from] SelectedObservationContentPlanError),
28    /// The policy's memory cannot hold the source's minimum envelope.
29    #[error(transparent)]
30    Resource(#[from] ResourceError),
31    /// The preferred envelope does not fit the address space.
32    #[error("the selected-observation envelope overflows")]
33    Overflow,
34}
35
36/// Finalize an unopened source's bounded execution envelope.
37///
38/// The storage owner supplies the requirement curve; the runtime selects at
39/// most the bootstrap budget of preferred growth beyond its mandatory minimum
40/// under the current policy. The returned budget charges the actual bounded
41/// plan, not all available host memory.
42pub fn finalize_source_access(
43    problem: &CompiledProblem,
44    access: ResolvedSelectedObservationAccess,
45    authority: &ResourceAuthority,
46    policy: &ResourcePolicy,
47) -> Result<ResolvedSelectedObservationAccess, SourceAccessError> {
48    let requirements = access.content_requirements(problem)?;
49    let maximum_live_blocks = access
50        .source_binding()
51        .content_budget()
52        .maximum_live_blocks();
53    let minimum = requirements.minimum_bytes(maximum_live_blocks)?;
54    let available = authority.remaining_selected_source_memory_bytes(policy)?;
55    let required = u64::try_from(minimum).map_err(|_| SourceAccessError::Overflow)?;
56    if required > available {
57        return Err(ResourceError::Infeasible {
58            resource: "selected-observation host memory".to_string(),
59            required,
60            available,
61        }
62        .into());
63    }
64    let preferred = minimum
65        .checked_add(bootstrap_source_budget().available_bytes())
66        .ok_or(SourceAccessError::Overflow)?;
67    let budget = SelectedObservationContentBudget::new(
68        preferred.min(usize::try_from(available).unwrap_or(usize::MAX)),
69        maximum_live_blocks,
70        requirements.maximum_pointing_polynomial_terms(),
71    );
72    let planned = requirements.plan(budget)?;
73    let budget = SelectedObservationContentBudget::new(
74        planned.maximum_resident_bytes(),
75        maximum_live_blocks,
76        requirements.maximum_pointing_polynomial_terms(),
77    );
78    Ok(access.with_content_budget(problem, &requirements, budget)?)
79}
