2//! `setoptions(ftmachine='mosaic')`: what `predict` builds CASA's simulator
3//! mosaic machine from.
4//!
5//! `Simulator::createSkyEquation` builds `MosaicFTNew` with the
6//! heterogeneous-array beams of `TransformMachines/HetArrayConvFunc` when
7//! OBSERVATION row 0's telescope has PBMath's UNKNOWN, OVRO, ACA or ALMA
8//! beam. casa-rs builds it for ALMA, ACA and OVRO. ALMA and ACA dishes
9//! `findAntennaSizes` replaces by the 10.7 m and 6.25 m Airy apertures
10//! ([`AiryDish::casa_alma_simulator`]); OVRO uses each physical diameter
11//! and its scaled blockage. Other supported default telescope patterns use
12//! `MosaicFTNew`'s `SimplePBConvFunc`, one common beam for all antennas.
13//!
14//! - The dish classes are the distinct dish diameters in ANTENNA order, a
15//!   diameter compared as casacore's `String::toString` prints it, to six
16//!   significant digits.
17//! - Every channel of each window observed is selected.
18//! - A scan's beam uses POINTING independently of its FIELD phase centre;
19//!   the kernels' phase ramp is its offset from the model's pixel
20//!   `(nx/2, ny/2)`. CASA predicts nothing for a pointing outside the
21//!   model image (`checkPBOfField`); casa-rs does not predict one yet.
22
23use casa_imaging_operator::{AiryDish, MosaicWindow};
24
25use super::Simulator;
26use super::error::{SimulatorError, SimulatorResult};
27use crate::predict::{MosaicBeams, PredictError, PrimaryBeam, SkyModel};
28
29/// The telescopes whose mosaic casa-rs builds.
30const MOSAIC_TELESCOPES: [&str; 3] = ["ALMA", "ACA", "OVRO"];
31
32/// A dish diameter as `String::toString(Double)` prints it, which keys
33/// `HetArrayConvFunc`'s dish classes: six significant digits.
34fn diameter_key(diameter_m: f64) -> String {
35    format!("{diameter_m:.5e}")
36}
37
38/// The most a diameter key's text holds: `format!` grows its string from
39/// nothing, to 16 bytes for its nine to twelve characters.
40const DIAMETER_KEY_BYTES: usize = 32;
41
42/// One pointing the mosaic machine keeps: a phase centre's bits and its
43/// offset.
44pub(super) type MosaicPointing = ([u64; 2], [f64; 2]);
45
46impl Simulator {
47    /// Bytes the mosaic machine's state holds, known before it is formed:
48    /// what the prediction keeps (a pointing for each scan's phase centre at
49    /// most, and each antenna's dish class), and what builds the beams
50    /// until the kernels are built (a dish and a diameter key for each
51    /// antenna at most, and every window's channel frequencies twice).
52    pub(super) fn mosaic_state_bytes(&self) -> (u64, u64) {
53        let kept =
54            self.scans.len() * size_of::<MosaicPointing>() + self.antennas.len() * size_of::<u8>();
55        let antennas = self.antennas.len()
56            * (size_of::<PrimaryBeam>() + size_of::<String>() + DIAMETER_KEY_BYTES);
57        let windows = self.spectral_windows.len() * size_of::<MosaicWindow>()
58            + self
59                .spectral_windows
60                .iter()
61                .map(|window| 2 * window.channel_count * size_of::<f64>())
62                .sum::<usize>();
63        (kept as u64, (antennas + windows) as u64)
64    }
65
66    /// The mosaic machine's beams for the scans observed so far, and each
67    /// antenna's dish class.
68    ///
69    /// # Errors
70    ///
71    /// Not yet implemented: a telescope without a supported default beam,
72    /// a heterogeneous-array antenna without a dish, or more than 255 dish
73    /// diameters.
74    pub(super) fn mosaic_beams(
75        &self,
76        custom: Option<&super::voltage_patterns::BeamSource>,
77    ) -> SimulatorResult<(MosaicBeams, Vec<u8>)> {
78        let telescope = self.observing_telescope();
79        let common = custom.is_some() || !MOSAIC_TELESCOPES.contains(&telescope);
80        let default = match custom {
81            Some(beam) => beam.configured(),
82            None => PrimaryBeam::for_telescope(telescope, self.reference_frequency_hz()),
83        };
84        if common && default.is_none() && custom.is_none() {
85            return Err(SimulatorError::not_yet_implemented(format!(
86                "predict with setoptions(ftmachine='mosaic') for telescope {telescope} is not \
87                 yet implemented in casa-rs (#1297)"
88            )));
89        }
90        // At most one class an antenna: the capacities are the bounds
91        // `mosaic_state_bytes` admits.
92        let mut keys: Vec<String> = Vec::with_capacity(self.antennas.len());
93        let mut dishes = Vec::with_capacity(self.antennas.len());
94        let mut classes = Vec::with_capacity(self.antennas.len());
95        if common && let Some(default) = default {
96            dishes.push(default);
97        }
98        for antenna in &self.antennas {
99            if common {
100                classes.push(0);
101                continue;
102            }
103            // `findAntennaSizes` gives a dish class only to `diameter > 0`.
104            if antenna.diameter_m.is_nan() || antenna.diameter_m <= 0.0 {
105                return Err(SimulatorError::not_yet_implemented(format!(
106                    "predict with setoptions(ftmachine='mosaic') for antenna {} of diameter {} \
107                     m is not yet implemented in casa-rs (#1053)",
108                    antenna.name, antenna.diameter_m
109                )));
110            }
111            let key = diameter_key(antenna.diameter_m);
112            let class = keys
113                .iter()
114                .position(|known| *known == key)
115                .unwrap_or_else(|| {
116                    let dish = if matches!(telescope, "ALMA" | "ACA") {
117                        AiryDish::casa_alma_simulator(antenna.diameter_m)
118                    } else {
119                        AiryDish::casa_airy(antenna.diameter_m)
120                    };
121                    tracing::info!(
122                        "Overriding PB with Airy of diameter {} m starting with antenna {}",
123                        dish.aperture_m,
124                        classes.len()
125                    );
126                    keys.push(key);
127                    dishes.push(PrimaryBeam::from(dish));
128                    keys.len() - 1
129                });
130            classes.push(u8::try_from(class).map_err(|_| {
131                SimulatorError::not_yet_implemented(
132                    "predict with setoptions(ftmachine='mosaic') for more than 255 dish \
133                     diameters is not yet implemented in casa-rs (#1053)",
134                )
135            })?);
136        }
137        let mut windows: Vec<MosaicWindow> = Vec::with_capacity(self.spectral_windows.len());
138        for scan in &self.scans {
139            let id = scan.spectral_window_id;
140            if windows
141                .iter()
142                .any(|window| window.spectral_window as usize == id)
143            {
144                continue;
145            }
146            let window = &self.spectral_windows[id];
147            let frequencies = window.channel_frequencies_hz();
148            windows.push(MosaicWindow {
149                spectral_window: id as u32,
150                window_frequencies_hz: frequencies.clone(),
151                channel_width_hz: window.channel_width_hz,
152                selected_frequencies_hz: frequencies,
153            });
154        }
155        Ok((
156            MosaicBeams {
157                common,
158                custom_pattern: custom.cloned(),
159                dishes,
160                windows,
161            },
162            classes,
163        ))
164    }
165
166    /// Each distinct pointing of the scans observed so far (its bits), where
167    /// their antennas point, and its offset in `model`
168    /// ([`SkyModel::pointing_offset_rad`]), sorted by the bits. The capacity
169    /// is a pointing for each scan, the bound `mosaic_state_bytes` admits.
170    ///
171    /// # Errors
172    ///
173    /// Not yet implemented: a pointing outside the model image. Raised: a
174    /// pointing the model's coordinates cannot convert.
175    pub(super) fn mosaic_pointings(
176        &self,
177        model: &SkyModel,
178    ) -> SimulatorResult<Vec<MosaicPointing>> {
179        let mut pointings: Vec<MosaicPointing> = Vec::with_capacity(self.scans.len());
180        for scan in &self.scans {
181            let key = scan.pointing_rad.map(f64::to_bits);
182            let offset = model
183                .pointing_offset_rad(scan.pointing_rad)
184                .map_err(|error: PredictError| {
185                    SimulatorError::raised(format!("Caught exception: {error}"))
186                })?
187                .ok_or_else(|| {
188                    SimulatorError::not_yet_implemented(format!(
189                        "predict with setoptions(ftmachine='mosaic') for field {} pointed \
190                         outside the model image is not yet implemented in casa-rs (#1053); \
191                         CASA predicts nothing for it",
192                        self.fields[scan.field_id].name
193                    ))
194                })?;
195            pointings.push((key, offset));
196        }
197        pointings.sort_unstable_by_key(|(key, _)| *key);
198        pointings.dedup_by_key(|(key, _)| *key);
199        Ok(pointings)
200    }
201}

204mod tests {

207    #[test]
208    fn diameters_compare_to_six_significant_digits() {
209        assert_eq!(diameter_key(12.0), diameter_key(12.000_000_1));
210        assert_ne!(diameter_key(12.0), diameter_key(12.000_1));
211        assert_ne!(diameter_key(12.0), diameter_key(7.0));
212    }
