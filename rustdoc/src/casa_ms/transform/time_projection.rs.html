25pub(super) const TIME_OUTPUT_SCALARS: [&str; 5] =
26    ["TIME", "TIME_CENTROID", "INTERVAL", "EXPOSURE", "FLAG_ROW"];

29pub(super) struct TimeProjection<'a> {

60    pub(super) fn new(
61        table: &'a Table,
62        input_path: &'a Path,
63        spectra: &'a [&'static str],
64        visibility: &'a VisibilityColumns,
65        rows_per_block: usize,
66        view: Option<&'a dyn PreparedTransformInputView>,
67    ) -> Self {
68        Self {
69            table,
70            input_path,
71            spectra,
72            visibility,
73            available: BTreeMap::new(),
74            rows_per_block,
75            view,
76        }
77    }

79    pub(super) fn average_groups(
80        &mut self,
81        plans: &[TimeGroupRead<'_>],
82    ) -> Result<(MainArrayBlock, Vec<TimeCoordinates>), MsTransformError> {
83        let mut output = MainArrayBlock::for_capacity(
84            plans.len(),
85            self.visibility.copies().len(),
86            STANDARD_SPECTRA.len(),
87        );
88        let mut coordinates = Vec::with_capacity(plans.len());
89        let mut members = plans
90            .iter()
91            .enumerate()
92            .flat_map(|(index, plan)| plan.group.rows.iter().copied().map(move |row| (index, row)));
93        let mut active_group = 0;
94        let mut accumulator = TimeGroupAccumulator::default();
95        loop {
96            let block = members
97                .by_ref()
98                .take(self.rows_per_block.max(1))
99                .collect::<Vec<_>>();
100            if block.is_empty() {
101                break;
102            }
103            let rows = block
104                .iter()
105                .map(|(_, row)| row.physical)
106                .collect::<Vec<_>>();
107            let mut raw =
108                MainArrayBlock::read(self.table, &rows, self.visibility, self.spectra, self.view)
109                    .map_err(|source| self.table_error(source))?;
110            self.cache_spectrum_availability(&block, &raw)?;
111            let scalars = TIME_INPUT_SCALARS.map(|name| {
112                self.table
113                    .column_accessor(name)?
114                    .scalar_cells_owned_for_rows(&rows)
115            });
116            let [times, centroids, intervals, exposures] = scalars;
117            let scalars = [
118                times.map_err(|source| self.table_error(source))?,
119                centroids.map_err(|source| self.table_error(source))?,
120                intervals.map_err(|source| self.table_error(source))?,
121                exposures.map_err(|source| self.table_error(source))?,
122            ];
123            for (offset, (group, row)) in block.into_iter().enumerate() {
124                if group != active_group {
125                    std::mem::take(&mut accumulator)
126                        .finish()
127                        .materialize(&mut output, &mut coordinates);
128                    active_group = group;
129                }
130                self.add_row(
131                    &mut accumulator,
132                    &plans[group],
133                    row,
134                    offset,
135                    &mut raw,
136                    &scalars,
137                )?;
138            }
139        }
140        if !plans.is_empty() {
141            accumulator
142                .finish()
143                .materialize(&mut output, &mut coordinates);
144        }
145        Ok((output, coordinates))
146    }

148    fn cache_spectrum_availability(
149        &mut self,
150        block: &[(usize, TimeInputRow)],
151        raw: &MainArrayBlock,
152    ) -> Result<(), MsTransformError> {
153        if self.spectra.is_empty() {
154            return Ok(());
155        }
156        let offsets = block
157            .iter()
158            .enumerate()
159            .map(|(offset, (_, row))| (row.physical, offset))
160            .collect::<BTreeMap<_, _>>();
161        for (name, cells) in self.spectra.iter().copied().zip(&raw.spectra) {
162            let mut missing = BTreeSet::new();
163            for (_, row) in block {
164                let first = row.buffer_first;
165                if let Some(&offset) = offsets.get(&first) {
166                    self.available
167                        .insert((first, name), cells[offset].is_some());
168                } else if !self.available.contains_key(&(first, name)) {
169                    missing.insert(first);
170                }
171            }
172            if missing.is_empty() {
173                continue;
174            }
175            let missing = missing.into_iter().collect::<Vec<_>>();
176            let accessor = self
177                .table
178                .column_accessor(name)
179                .map_err(|source| self.table_error(source))?;
180            if let Ok(shapes) = accessor.array_shapes_uncached(&missing) {
181                for (first, shape) in missing.into_iter().zip(shapes) {
182                    self.available.insert((first, name), shape.is_some());
183                }
184            } else {
185                let cells = accessor
186                    .array_cells_owned_uncached(&missing)
187                    .map_err(|source| self.table_error(source))?;
188                for (first, value) in missing.into_iter().zip(cells) {
189                    self.available.insert((first, name), value.is_some());
190                }
191            }
192        }
193        Ok(())
194    }

196    fn add_row(
197        &self,
198        accumulator: &mut TimeGroupAccumulator,
199        plan: &TimeGroupRead<'_>,
200        row: TimeInputRow,
201        offset: usize,
202        raw: &mut MainArrayBlock,
203        scalars: &[Vec<Option<ScalarValue>>; 4],
204    ) -> Result<(), MsTransformError> {
205        let TimeGroupRead { channels, axes, .. } = *plan;
206        let require = |value: Option<ArrayValue>, column| {
207            value.ok_or_else(|| missing_column_error(self.input_path, row.physical, column))
208        };
209        let select = |value, name, on_channels| {
210            select_projection(value, name, channels, axes, on_channels)
211                .map_err(|source| self.table_error(source))
212        };
213        let flags = select(require(raw.flags[offset].take(), "FLAG")?, "FLAG", true)?;
214        let weight = select(
215            require(raw.weights[offset].take(), "WEIGHT")?,
216            "WEIGHT",
217            false,
218        )?;
219        let sigma = select(require(raw.sigmas[offset].take(), "SIGMA")?, "SIGMA", false)?;
220        let values = self
221            .visibility
222            .copies()
223            .iter()
224            .zip(&mut raw.visibility)
225            .map(|(copy, cells)| {
226                cells[offset]
227                    .take()
228                    .map(|value| {
229                        select(
230                            value,
231                            copy.source.name(),
232                            copy.source != OptionalMainColumn::LagData,
233                        )
234                    })
235                    .transpose()
236            })
237            .collect::<Result<Vec<_>, _>>()?;
238        let weight_spectrum = self.select_spectrum(raw, row, offset, "WEIGHT_SPECTRUM", plan)?;
239        let sigma_spectrum = self.select_spectrum(raw, row, offset, "SIGMA_SPECTRUM", plan)?;
240        let &[correlations, channel_count] = flags.shape() else {
241            return Err(self.table_error(TableError::Schema(
242                "FLAG must have correlation and channel axes".to_string(),
243            )));
244        };
245        let average = accumulator.visibility.get_or_insert_with(|| {
246            TimeVisibilityAverage::new(self.visibility, (correlations, channel_count))
247        });
248        let (flag_row, row_weight) = average
249            .add(TimeVisibilityRow {
250                values: &values,
251                flags: &flags,
252                weight: &weight,
253                sigma: &sigma,
254                spectra: SpectrumCells {
255                    weight: weight_spectrum.as_ref(),
256                    sigma: sigma_spectrum.as_ref(),
257                },
258            })
259            .map_err(|source| self.table_error(source))?;
260        let uvw = uvw_coordinates(raw.uvw[offset].as_ref())
261            .ok_or_else(|| missing_column_error(self.input_path, row.physical, "UVW"))?;
262        let position = TimeCoordinates {
263            time: scalar_f64(scalars[0][offset].as_ref(), "TIME", row.physical)?,
264            centroid: scalar_f64(scalars[1][offset].as_ref(), "TIME_CENTROID", row.physical)?,
265            interval: scalar_f64(scalars[2][offset].as_ref(), "INTERVAL", row.physical)?,
266            exposure: scalar_f64(scalars[3][offset].as_ref(), "EXPOSURE", row.physical)?,
267            flag_row,
268            uvw,
269        };
270        accumulator
271            .coordinates
272            .get_or_insert_with(|| TimeCoordinatesAverage::new(position))
273            .add(position, row_weight);
274        if let Some(cells) = raw.categories[offset]
275            .take()
276            .filter(|value| !value.is_empty())
277        {
278            let bins = accumulator.category_bins.get_or_insert_with(|| {
279                channels
280                    .iter()
281                    .map(|&channel| vec![channel])
282                    .collect::<Vec<_>>()
283            });
284            let selected = select_visibility_axes(cells, axes, "FLAG_CATEGORY")
285                .and_then(|cells| transform_flag_category_channels(cells, bins))
286                .map_err(|source| self.table_error(source))?;
287            include_categories(&mut accumulator.categories, selected)
288                .map_err(|source| self.table_error(source))?;
289        }
290        Ok(())
291    }
292
293    fn select_spectrum(
294        &self,
295        raw: &mut MainArrayBlock,
296        row: TimeInputRow,
297        offset: usize,
298        name: &'static str,
299        plan: &TimeGroupRead<'_>,
300    ) -> Result<Option<ArrayValue>, MsTransformError> {
301        let Some(index) = self.spectra.iter().position(|&input| input == name) else {
302            return Ok(None);
303        };
304        if !self.available[&(row.buffer_first, name)] {
305            return Ok(None);
306        }
307        let value = raw.spectra[index][offset].take().ok_or_else(|| {
308            MsTransformError::MissingSpectrumCell {
309                path: self.input_path.display().to_string(),
310                row: row.physical,
311                column: name,
312            }
313        })?;
314        select_projection(value, name, plan.channels, plan.axes, true)
315            .map(Some)
316            .map_err(|source| self.table_error(source))
317    }
318
319    fn table_error(&self, source: TableError) -> MsTransformError {
320        MsTransformError::MutateMeasurementSet {
321            path: self.input_path.display().to_string(),
322            source: Box::new(source),
323        }
324    }
325}

327impl TimeGroupAccumulator {
328    fn finish(self) -> AveragedMainRow {
329        AveragedMainRow {
330            data: self
331                .visibility
332                .expect("a time bin contains a baseline row")
333                .finish(),
334            coordinates: self
335                .coordinates
336                .expect("a time bin contains row coordinates")
337                .finish(),
338            categories: self.categories,
339        }
340    }
341}
342
343impl AveragedMainRow {
344    fn materialize(self, output: &mut MainArrayBlock, coordinates: &mut Vec<TimeCoordinates>) {
345        for (column, value) in output.visibility.iter_mut().zip(self.data.values) {
346            column.push(value);
347        }
348        output.flags.push(Some(self.data.flags));
349        output.categories.push(self.categories);
350        output.uvw.push(Some(ArrayValue::Float64(
351            Array1::from_vec(self.coordinates.uvw.to_vec()).into_dyn(),
352        )));
353        output.weights.push(Some(self.data.weight));
354        output.sigmas.push(Some(self.data.sigma));
355        output.spectra[0].push(Some(self.data.weight_spectrum));
356        output.spectra[1].push(Some(self.data.sigma_spectrum));
357        coordinates.push(self.coordinates);
358    }
359}

361impl TimeCoordinates {
362    pub(super) fn scalar_cells(self) -> [ScalarValue; 5] {
363        [
364            ScalarValue::Float64(self.time),
365            ScalarValue::Float64(self.centroid),
366            ScalarValue::Float64(self.interval),
367            ScalarValue::Float64(self.exposure),
368            ScalarValue::Bool(self.flag_row),
369        ]
370    }
371}
372
373fn select_projection(
374    value: ArrayValue,
375    name: &str,
376    channels: &[usize],
377    axes: Option<&[usize]>,
378    on_channels: bool,
379) -> Result<ArrayValue, TableError> {
380    let value = select_visibility_axes(value, axes, name)?;
381    if on_channels && !value.is_empty() {
382        select_channels(value, channels)
383    } else {
384        Ok(value)
385    }
386}
387
388fn include_categories(
389    current: &mut Option<ArrayValue>,
390    incoming: ArrayValue,
391) -> Result<(), TableError> {
392    let Some(ArrayValue::Bool(current)) = current.as_mut() else {
393        *current = Some(incoming);
394        return Ok(());
395    };
396    let ArrayValue::Bool(incoming) = incoming else {
397        return Err(TableError::Schema(
398            "FLAG_CATEGORY must hold Bool".to_string(),
399        ));
400    };
401    if current.shape() != incoming.shape() {
402        return Err(TableError::Schema(
403            "FLAG_CATEGORY shapes differ during time averaging".to_string(),
404        ));
405    }
406    for (accumulator, &flag) in current.iter_mut().zip(&incoming) {
407        *accumulator &= flag;
408    }
409    Ok(())
