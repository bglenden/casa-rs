23/// The operator of one image domain and the resampler that places rows on
24/// its planes.
25pub(crate) struct DomainOperator {
26    pub(crate) operator: MeasurementOperator,
27    pub(crate) resampler: SpectralResampler,
28    /// Whether the kernel set grids a sensitivity (weight) image
29    /// (`ConvolutionFunctionSet::weight_taps`): mosaic and AW.
30    pub(crate) weight_image: bool,
31    /// The charge of the kernel set, held while the operator lives.
32    _kernels: Reservation,
33}

35/// Correlation types every block delivers, in block order: the first
36/// source's selection, which every source must share.
37pub(crate) fn selected_correlations(
38    problem: &CompiledProblem,
39) -> Result<Vec<CorrelationType>, ImagingError> {
40    let mut layouts = problem
41        .observation_transaction()
42        .read_set()
43        .sources()
44        .iter()
45        .flat_map(|source| source.selection().correlations())
46        .map(|selection| {
47            selection
48                .products()
49                .iter()
50                .map(|product| product.correlation_type())
51                .collect::<Vec<_>>()
52        });
53    let first = layouts.next().ok_or(ImagingError::Unsupported {
54        reason: "the selection names no correlations",
55    })?;
56    if layouts.any(|layout| layout != first) {
57        return Err(ImagingError::Unsupported {
58            reason: "selected polarization setups differ in their correlations",
59        });
60    }
61    Ok(first)
62}

64/// The most samples a row of `problem`'s source places and predicts on any
65/// of its image domains, which share one resampler
66/// ([`SpectralResampler::samples_per_row`],
67/// [`SpectralResampler::prediction_samples_per_row`]).
68pub(crate) fn row_samples(problem: &CompiledProblem) -> Result<RowSamples, ImagingError> {
69    let resampler = resampler(problem, basis(problem)?)?;
70    let spectrum = row_layout(problem).spectrum;
71    Ok(RowSamples {
72        placed: resampler.samples_per_row(spectrum),
73        predicted: resampler.prediction_samples_per_row(spectrum.channels),
74    })
75}

77/// The grid basis of the compiled reconstruction basis. A Taylor basis
78/// expands about the image's reference frequency.
79pub(crate) fn basis(problem: &CompiledProblem) -> Result<Basis, ImagingError> {
80    Ok(match problem.reconstruction().basis() {
81        ReconstructionBasis::Constant => Basis::Constant,
82        ReconstructionBasis::Taylor { terms } => Basis::Taylor {
83            terms: terms as u32,
84            reference_hz: reference_frequency_hz(problem)?,
85        },
86        ReconstructionBasis::ChannelLocal { channels } => Basis::ChannelLocal {
87            planes: channels as u32,
88        },
89    })
90}

92/// The image's spectral reference frequency (CASA `reffreq`).
93pub(crate) fn reference_frequency_hz(problem: &CompiledProblem) -> Result<f64, ImagingError> {
94    match problem.geometry().spectral().wcs() {
95        SpectralWcs::Linear {
96            reference_frequency_hz,
97            ..
98        } => Ok(*reference_frequency_hz),
99        SpectralWcs::Tabular { .. } => Err(ImagingError::Unsupported {
100            reason: "a tabular spectral axis has no uniform channel width",
101        }),
102    }
103}

105/// The grid precision: the requested one, else plan decision D2's rule as
106/// the owner amended it on 2026-10-15 (`gridprecision = auto`): f64 on the
107/// CPU for every basis, since CASA accumulates its grids in double
108/// (`useDoublePrec`, `SynthesisUtilMethods.cc:3722`), and f32 on Metal,
109/// which accumulates in `f32` only.
110///
111/// A cube is no exception: an f32 running sum drifts with the number of
112/// samples a cell takes, and a mosaic's weight image takes every sample of
113/// a channel at the uv origin (#659).
114pub(crate) const fn precision(
115    backend: BackendChoice,
116    requested: Option<GridPrecision>,
117) -> GridPrecision {
118    if let Some(precision) = requested {
119        return precision;
120    }
121    match backend {
122        BackendChoice::Cpu => GridPrecision::F64,
123        BackendChoice::Metal => GridPrecision::F32,
124    }
125}

127/// Whether the problem grids with the standard kernel set, the one the
128/// Metal backend implements.
129pub(crate) fn standard_kernel_set(problem: &CompiledProblem) -> bool {
130    kernel_set_kind(problem) == KernelSetKind::Standard
131}

133/// Which kernel set the compiled problem's measurement equation names.
134#[derive(Clone, Copy, Debug, PartialEq, Eq)]
135enum KernelSetKind {
136    /// The standard spheroidal set (`GridFT`).
137    Standard,
138    /// W-projection planes (`WProjectFT`).
139    WPlanes,
140    /// The heterogeneous-array mosaic beams (`MosaicFT` with
141    /// `HetArrayConvFunc`).
142    Mosaic,
143    /// The AW catalog (`AWProjectFT`).
144    Aw,
145}
146
147fn kernel_set_kind(problem: &CompiledProblem) -> KernelSetKind {
148    let equation = problem.science().measurement_equation();
149    if equation.aw_projection().is_some() {
150        KernelSetKind::Aw
151    } else if equation.w_projection().is_some() {
152        KernelSetKind::WPlanes
153    } else if problem.science().instrument_model()
154        == Some(InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1)
155    {
156        KernelSetKind::Mosaic
157    } else {
158        KernelSetKind::Standard
159    }
160}

162/// The operator of a chart of `extent` (an image domain, or one of its
163/// facets) with the kernel set the problem names: the
164/// standard spheroidal set, W-projection planes sized by the W contract,
165/// the mosaic primary beams of the selected windows, or the AW catalog of
166/// `aw_catalog`, in the requested precision or the one `backend` grids at.
167/// The mosaic set takes one dish per selected aperture class in
168/// `dish_classes` (`HetArrayConvFunc::findAntennaSizes`), the order the
169/// rows' antenna types index. Mosaic and AW grid without padding (CASA
170/// `MosaicFT`, `AWProjectFT`); the others on CASA's composite-padded grid.
171/// The kernel set is admitted as [`kernel_set`] says, and its charge lives
172/// with the operator.
173#[allow(clippy::too_many_arguments)]
174pub(crate) fn domain_operator(
175    problem: &CompiledProblem,
176    extent: ImageExtent,
177    correlations: &[CorrelationType],
178    backend: BackendChoice,
179    requested_precision: Option<GridPrecision>,
180    aw_catalog: Option<&AwCatalogDeployment>,
181    dish_classes: &[AntennaResponseClass],
182    admission: KernelAdmission<'_>,
183) -> Result<DomainOperator, ImagingError> {
184    let kind = kernel_set_kind(problem);
185    // Each CASA facet's GridFT pads its own facet grid.
186    let padding = match kind {
187        KernelSetKind::Standard | KernelSetKind::WPlanes => GridPadding::CasaComposite,
188        KernelSetKind::Mosaic | KernelSetKind::Aw => GridPadding::None,
189    };
190    let geometry = GridGeometry::new(extent, padding)?;
191    let polarization = PolarizationRouting::compile(
192        correlations,
193        problem.reconstruction().polarization().coordinates(),
194    )?;
195    // An AW cube is refused by the gate (`Unsupported::CubeAwProjection`).
196    let basis = basis(problem)?;
197    let resampler = resampler(problem, basis)?;
198    let (cf, kernels) = kernel_set(
199        problem,
200        kind,
201        (&geometry, &polarization),
202        aw_catalog,
203        dish_classes,
204        admission,
205    )?;
206    Ok(DomainOperator {
207        operator: MeasurementOperator::new(
208            geometry,
209            basis,
210            polarization,
211            cf,
212            precision(backend, requested_precision),
213        ),
214        resampler,
215        weight_image: matches!(kind, KernelSetKind::Mosaic | KernelSetKind::Aw),
216        _kernels: kernels,
217    })
218}

220/// Where a run admits its kernel sets: its host and policy, and the
221/// workers that fetch cells from a paged set.
222#[derive(Clone, Copy)]
223pub(crate) struct KernelAdmission<'a> {
224    pub(crate) host: &'a HostResources,
225    pub(crate) policy: &'a ResourcePolicy,
226    pub(crate) workers: usize,
227}
228
229impl KernelAdmission<'_> {
230    fn admit(self, phase: &'static str, memory: u64) -> Result<Reservation, ImagingError> {
231        Ok(admit(self.host, self.policy, &Demand { phase, memory })?)
232    }
233}

235/// The kernel set of `kind` on `geometry` routing `polarization`, with the
236/// charge of what it holds, admitted before it is built. W-projection and
237/// mosaic sets are built in two stages, their screens and then the kernels
238/// cut from them, each admitted before it allocates; the screens' charge
239/// is released once the kernels are built. An AW catalog is charged its
240/// cache bound and what its workers can hold beyond it once its index is
241/// read ([`AwCatalog::charge_bytes`]).
242fn kernel_set(
243    problem: &CompiledProblem,
244    kind: KernelSetKind,
245    (geometry, polarization): (&GridGeometry, &PolarizationRouting),
246    aw_catalog: Option<&AwCatalogDeployment>,
247    dish_classes: &[AntennaResponseClass],
248    admission: KernelAdmission<'_>,
249) -> Result<(Box<dyn ConvolutionFunctionSet>, Reservation), ImagingError> {
250    const PHASE: &str = "convolution functions";
251    Ok(match kind {
252        KernelSetKind::Standard => {
253            let kernels = admission.admit(PHASE, Spheroidal::bytes(geometry))?;
254            (Box::new(Spheroidal::new(geometry, polarization)), kernels)
255        }
256        KernelSetKind::WPlanes => {
257            let contract = problem
258                .science()
259                .measurement_equation()
260                .w_projection()
261                .expect("the kind names a W contract");
262            let count = match (contract.planes(), contract.statistics()) {
263                (Some(planes), _) => WPlaneCount::Fixed(planes.get() as u32),
264                (None, Some(statistics)) => WPlaneCount::Auto {
265                    min_w: statistics.minimum_abs_w_lambda(),
266                    max_w: contract.maximum_abs_w_lambda(),
267                    rms_w: statistics.rms_w_lambda(),
268                },
269                (None, None) => {
270                    return Err(ImagingError::Unsupported {
271                        reason: "an automatic W-plane count needs the selection's w statistics",
272                    });
273                }
274            };
275            let screening = admission.admit(
276                "W-projection screens",
277                WPlanes::screens_bytes(geometry, count)?,
278            )?;
279            let screens = WPlanes::screens(geometry, polarization, count)?;
280            let mut kernels = admission.admit(PHASE, screens.kernel_bytes())?;
281            let set = screens.finish()?;
282            drop(screening);
283            kernels.retain(set.resident_bytes());
284            (Box::new(set), kernels)
285        }
286        KernelSetKind::Mosaic => {
287            if dish_classes.is_empty() {
288                return Err(ImagingError::Unsupported {
289                    reason: "the mosaic set needs ALMA or ACA dishes in the selection",
290                });
291            }
292            let dishes = dish_classes
293                .iter()
294                .map(|class| {
295                    AiryDish::casa_alma(
296                        match class {
297                            AntennaResponseClass::CasaAlma12m => 12.0,
298                            AntennaResponseClass::CasaAca7m => 7.0,
299                        },
300                        geometry,
301                    )
302                })
303                .collect::<Vec<_>>();
304            let frequency_hz = reference_frequency_hz(problem)?;
305            let windows = mosaic_windows(problem)?;
306            let screening = admission.admit(
307                "mosaic screens",
308                MosaicPb::screens_bytes(geometry, frequency_hz, &dishes, &windows)?,
309            )?;
310            let screens =
311                MosaicPb::screens(geometry, polarization, frequency_hz, &dishes, &windows)?;
312            let mut kernels = admission.admit(PHASE, screens.kernel_bytes())?;
313            let set = screens.finish()?;
314            drop(screening);
315            kernels.retain(set.resident_bytes());
316            (Box::new(set), kernels)
317        }
318        KernelSetKind::Aw => {
319            let deployment = aw_catalog.ok_or(ImagingError::Unsupported {
320                reason: "an A-projection run needs its convolution-function catalog",
321            })?;
322            let catalog = AwCatalog::open_casa(
323                &deployment.root,
324                deployment.indexing,
325                geometry,
326                polarization,
327                deployment.resident_bytes,
328            )?;
329            let kernels = admission.admit(PHASE, catalog.charge_bytes(admission.workers))?;
330            (Box::new(catalog), kernels)
331        }
332    })
333}

335/// The selected spectral windows as the mosaic set keys them: every
336/// channel of the window, its first channel width and the selected
337/// channels' frequencies (`HetArrayConvFunc` beams per window).
338fn mosaic_windows(problem: &CompiledProblem) -> Result<Vec<MosaicWindow>, ImagingError> {
339    let mut windows = Vec::new();
340    for source in problem.observation_transaction().read_set().sources() {
341        for window in source.selection().spectral_windows() {
342            if windows
343                .iter()
344                .any(|known: &MosaicWindow| known.spectral_window == window.spectral_window_id())
345            {
346                continue;
347            }
348            let catalog = window
349                .coordinate_catalog()
350                .ok_or(ImagingError::Unsupported {
351                    reason: "the mosaic set needs every window's channel frequencies",
352                })?;
353            let selected = window
354                .channel_indices()
355                .iter()
356                .map(|index| {
357                    catalog
358                        .channel_frequency_hz(*index as usize)
359                        .ok_or(ImagingError::Unsupported {
360                            reason: "a selected channel lies outside its window",
361                        })
362                })
363                .collect::<Result<Vec<_>, _>>()?;
364            windows.push(MosaicWindow {
365                spectral_window: window.spectral_window_id(),
366                window_frequencies_hz: catalog.channel_frequencies_hz().to_vec(),
367                channel_width_hz: catalog.first_channel_width_hz(),
368                selected_frequencies_hz: selected,
369            });
370        }
371    }
372    Ok(windows)
373}

375pub(crate) fn image_extent(domain: &CompiledImageDomain) -> Result<ImageExtent, ImagingError> {
376    let direction = domain.direction();
377    let reference = direction.reference_pixel();
378    if reference
379        .iter()
380        .any(|pixel| pixel.fract() != 0.0 || *pixel < 0.0)
381    {
382        return Err(ImagingError::Unsupported {
383            reason: "the image reference pixel must be a whole pixel",
384        });
385    }
386    Ok(ImageExtent {
387        shape: domain.shape().pixels(),
388        increment_rad: direction.increment_rad(),
389        reference_pixel: [reference[0] as usize, reference[1] as usize],
390    })
391}

393/// CASA `FTMachine` channel mapping for a channel-local basis; every native
394/// channel on plane 0 otherwise.
395fn resampler(problem: &CompiledProblem, basis: Basis) -> Result<SpectralResampler, ImagingError> {
396    let Basis::ChannelLocal { planes } = basis else {
397        return Ok(SpectralResampler::direct(basis)?);
398    };
399    let SpectralWcs::Linear {
400        reference_pixel,
401        reference_frequency_hz,
402        increment_hz,
403        ..
404    } = problem.geometry().spectral().wcs()
405    else {
406        return Err(ImagingError::Unsupported {
407            reason: "a tabular spectral axis has no uniform channel width",
408        });
409    };
410    let kernel = match problem.science().spectral().sampling().kernel() {
411        SpectralKernel::Nearest | SpectralKernel::Identity => {
412            casa_imaging_operator::SpectralKernel::Nearest
413        }
414        SpectralKernel::Linear => casa_imaging_operator::SpectralKernel::Linear,
415        SpectralKernel::Cubic | SpectralKernel::ChannelIntegration { .. } => {
416            return Err(ImagingError::Unsupported {
417                reason: "cube interpolation is nearest or linear",
418            });
419        }
420    };
421    let first_hz = reference_frequency_hz - reference_pixel * increment_hz;
422    Ok(SpectralResampler::channel_local(
423        SpectralAxis::new(first_hz, *increment_hz, planes)?,
424        kernel,
425    ))
426}

428/// The widest spacing between adjacent selected native channels of any
429/// selected spectral window, from the storage owner's `CHAN_FREQ` catalog,
430/// widened by [`FRAME_MARGIN`]. A window without a catalog makes it
431/// infinite, so every wave holds the whole model.
432pub(crate) fn native_spacing_hz(problem: &CompiledProblem) -> f64 {
433    let mut widest = 0.0_f64;
434    for window in problem
435        .observation_transaction()
436        .read_set()
437        .sources()
438        .iter()
439        .flat_map(|source| source.selection().spectral_windows())
440    {
441        let Some(catalog) = window.coordinate_catalog() else {
442            return f64::INFINITY;
443        };
444        for pair in window.channel_indices().windows(2) {
445            let (Some(first), Some(second)) = (
446                catalog.channel_frequency_hz(pair[0] as usize),
447                catalog.channel_frequency_hz(pair[1] as usize),
448            ) else {
449                return f64::INFINITY;
450            };
451            widest = widest.max((second - first).abs());
452        }
453    }
454    widest * (1.0 + FRAME_MARGIN)
455}

457/// Shape of the weight-density grid of `domain`: one global plane on CASA's
458/// `VisImagingWeight` cells, or one plane per output channel, plus the
459/// compiled padding planes on each side, on `BriggsCubeWeightor` cells.
460pub(crate) fn density_shape(
461    problem: &CompiledProblem,
462    domain: &CompiledImageDomain,
463) -> Result<DensityGridShape, ImagingError> {
464    let [width, height] = domain.shape().pixels();
465    let (planes, padding, rule) = match problem.weighting().density_scope() {
466        WeightDensityScope::PerOutputChannel => {
467            // CASA always pads the cube density axis
468            // (`BriggsCubeWeightor::estimateSwingChanPad`); a contract without
469            // the padding cannot reproduce it.
470            let padding = problem.weighting().casa_cube_density_padding().ok_or(
471                ImagingError::Unsupported {
472                    reason: "a per-channel weight density needs its compiled padding",
473                },
474            )?;
475            let planes = padding
476                .checked_mul(2)
477                .and_then(|padding| {
478                    padding.checked_add(problem.geometry().spectral().output_channels())
479                })
480                .ok_or(ImagingError::Unsupported {
481                    reason: "the padded density axis overflows",
482                })?;
483            let padding = u32::try_from(padding).map_err(|_| ImagingError::Unsupported {
484                reason: "the padded density axis overflows",
485            })?;
486            (planes, padding, DensityCellRule::Cube)
487        }
488        WeightDensityScope::NotApplicable | WeightDensityScope::GlobalSelection => {
489            (1, 0, DensityCellRule::Standard)
490        }
491    };
492    Ok(DensityGridShape {
493        width,
494        height,
495        planes,
496        padding,
497        increment_rad: domain.direction().increment_rad(),
498        rule,
499    })
