30/// Produce every planned member and write it to `output`'s staging.
31///
32/// Runs restoring-beam fitting, restoration, residual scaling, normalization,
33/// validity, and metadata exactly once per member, in the planned publication
34/// order. [`CasaImageWriter::publish`] then replaces the targets.
35///
36/// # Errors
37///
38/// Unsupported roles, failed beam fits, generated non-finite payloads, and
39/// image writes that fail. A failed generation discards every member it
40/// staged, so a later [`CasaImageWriter::publish`] publishes none of them.
41pub fn produce_continuum_members(
42    planned: &PlannedContinuumGeneration,
43    inputs: &ContinuumProductInputs<'_>,
44    storage_plan: ProductStoragePlan,
45    execution: &impl ProductWindowExecutor,
46    output: &CasaImageWriter,
47) -> Result<PublishedContinuumGeneration, ProductsError> {
48    let kept = output.staged_count();
49    let produced =
50        if inputs.normal_state().catalog() == NormalStateCatalog::UnnormalizedTaylorBlockV1 {
51            produce_taylor_members(planned, inputs, storage_plan, output)
52        } else {
53            produce_plane_members(planned, inputs, storage_plan, execution, output)
54        };
55    if produced.is_err() {
56        output.discard_staged_after(kept);
57    }
58    produced
59}

61fn produce_plane_members(
62    planned: &PlannedContinuumGeneration,
63    inputs: &ContinuumProductInputs<'_>,
64    storage_plan: ProductStoragePlan,
65    execution: &impl ProductWindowExecutor,
66    output: &CasaImageWriter,
67) -> Result<PublishedContinuumGeneration, ProductsError> {
68    let fitted_beams = if planned.requires_beam() {
69        fit_beams(planned, inputs, storage_plan, execution)?
70    } else {
71        Box::new([])
72    };
73    let restoring_beams = restoring_beams(inputs, &fitted_beams)?;
74    for member in &planned.members {
75        let windows = MemberWindows::new(
76            planned,
77            member,
78            inputs,
79            storage_plan,
80            &fitted_beams,
81            &restoring_beams,
82        )?;
83        let image = output.begin(
84            member,
85            windows.layout,
86            &planned.beams_for_member(member, &fitted_beams, &restoring_beams),
87            planned.uses_weight_image,
88        )?;
89        windows.write(ProductMemberWriter::new(windows.layout, image), execution)?;
90    }
91    Ok(planned.published(fitted_beams, restoring_beams))
92}

94/// Fit a restoring beam to every valid PSF plane, domain by domain, then
95/// channel by channel, then polarization, a wave of the team's workers at a
96/// time.
97fn fit_beams(
98    planned: &PlannedContinuumGeneration,
99    inputs: &ContinuumProductInputs<'_>,
100    storage_plan: ProductStoragePlan,
101    execution: &impl ProductWindowExecutor,
102) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
103    let normal_state = inputs.normal_state();
104    let channels = normal_state.channel_count();
105    let polarizations = normal_state.polarization_count();
106    let beam_count = normal_state
107        .domain_count()
108        .checked_mul(channels)
109        .and_then(|count| count.checked_mul(polarizations))
110        .ok_or(ProductsError::ResourceDemandOverflow(
111            "generation beam count",
112        ))?;
113    let mut fitted = Vec::with_capacity(beam_count);
114    for start in (0..beam_count).step_by(storage_plan.maximum_workers()) {
115        let count = storage_plan.maximum_workers().min(beam_count - start);
116        let mut slots = vec![None; count];
117        execution.prepare(&mut slots, &|index| {
118            let ordinal = start + index;
119            let domain =
120                &inputs.problem().geometry().domains()[ordinal / (channels * polarizations)];
121            let plane = normal_state.read_plane(
122                inputs.model_domain_ordinal(domain.role()),
123                (ordinal / polarizations) % channels,
124                ordinal % polarizations,
125            )?;
126            if plane.validity() != SpectralChannelValidity::Valid {
127                return Ok(None);
128            }
129            fit_restoring_beam(
130                &psf_real_plane(&plane)?,
131                plane.shape(),
132                inputs.cell_size_rad_for_domain(domain.role()),
133                planned.psf_cutoff(),
134            )
135            .map(Some)
136        })?;
137        fitted.extend(
138            slots
139                .into_iter()
140                .map(|slot| slot.expect("the executor fills every slot")),
141        );
142    }
143    Ok(fitted.into_boxed_slice())
144}

146/// The beams each plane is restored with, by the problem's restoring-beam
147/// policy: none, each plane's fitted beam, or one beam enclosing a domain's
148/// fitted beams.
149fn restoring_beams(
150    inputs: &ContinuumProductInputs<'_>,
151    fitted_beams: &[Option<RestoringBeam>],
152) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
153    Ok(match inputs.problem().products().restoring_beam() {
154        RestoringBeamPolicy::None => Box::new([]),
155        RestoringBeamPolicy::PerPlane => fitted_beams.into(),
156        RestoringBeamPolicy::Common => {
157            let normal_state = inputs.normal_state();
158            let planes = normal_state.channel_count() * normal_state.polarization_count();
159            let mut selected = Vec::with_capacity(fitted_beams.len());
160            for domain_beams in fitted_beams.chunks(planes) {
161                let valid = domain_beams.iter().flatten().copied().collect::<Vec<_>>();
162                if valid.is_empty() {
163                    selected.extend_from_slice(domain_beams);
164                } else {
165                    let common = RestoringBeam::common_enclosing(&valid)
166                        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))?;
167                    selected.extend(domain_beams.iter().map(|fitted| fitted.map(|_| common)));
168                }
169            }
170            selected.into_boxed_slice()
171        }
172    })
173}

175/// What generating one member's windows needs.
176struct MemberWindows<'a, 'inputs> {
177    planned: &'a PlannedContinuumGeneration,
178    member: &'a PlannedMember,
179    inputs: &'a ContinuumProductInputs<'inputs>,
180    layout: ProductWindowLayout,
181    domain_ordinal: usize,
182    plane_shape: [usize; 2],
183    fitted_beams: &'a [Option<RestoringBeam>],
184    restoring_beams: &'a [Option<RestoringBeam>],
185    workers: usize,
186    fft_threads: usize,
187}
188
189impl<'a, 'inputs> MemberWindows<'a, 'inputs> {
190    fn new(
191        planned: &'a PlannedContinuumGeneration,
192        member: &'a PlannedMember,
193        inputs: &'a ContinuumProductInputs<'inputs>,
194        storage_plan: ProductStoragePlan,
195        fitted_beams: &'a [Option<RestoringBeam>],
196        restoring_beams: &'a [Option<RestoringBeam>],
197    ) -> Result<Self, ProductsError> {
198        let domain_ordinal = inputs.model_domain_ordinal(member.axes().domain());
199        let layout = storage_plan.layout(member.axes())?;
200        let windows = inputs
201            .normal_state()
202            .channel_count()
203            .div_ceil(layout.maximum_channels());
204        Ok(Self {
205            planned,
206            member,
207            inputs,
208            layout,
209            domain_ordinal,
210            plane_shape: inputs.final_model().shape().domains()[domain_ordinal].pixels(),
211            fitted_beams,
212            restoring_beams,
213            workers: storage_plan.maximum_workers().min(windows),
214            fft_threads: storage_plan.fft_threads(windows),
215        })
216    }
217
218    /// Generate the member's windows a wave of workers at a time and write
219    /// them in channel order.
220    fn write(
221        &self,
222        mut writer: ProductMemberWriter<'_>,
223        execution: &impl ProductWindowExecutor,
224    ) -> Result<(), ProductsError> {
225        let channels = self.inputs.normal_state().channel_count();
226        let window_channels = self.layout.maximum_channels();
227        let windows = channels.div_ceil(window_channels);
228        for wave_start in (0..windows).step_by(self.workers) {
229            let mut slots: Vec<Option<ProductWindow>> = (0..self.workers.min(windows - wave_start))
230                .map(|_| None)
231                .collect();
232            execution.prepare(&mut slots, &|index| {
233                let start = (wave_start + index) * window_channels;
234                self.window(start..(start + window_channels).min(channels))
235            })?;
236            for slot in slots {
237                writer.write(slot.expect("the executor fills every slot"))?;
238            }
239        }
240        writer.finish()
241    }
242
243    /// One window: every polarization plane of its channels.
244    fn window(&self, channels: Range<usize>) -> Result<ProductWindow, ProductsError> {
245        let mut window = self.layout.window(channels.clone())?;
246        let reconstruction_only = matches!(
247            self.member.role,
248            ProductRole::Model(ProductTerm::Single | ProductTerm::Taylor(0))
249                | ProductRole::CleanMask
250        );
251        for channel in channels.clone() {
252            for polarization in 0..self.inputs.normal_state().polarization_count() {
253                let local = channel - channels.start;
254                if reconstruction_only {
255                    let plane = self.reconstruction_plane(channel, polarization)?;
256                    self.scatter(
257                        &mut window.payload,
258                        &window.shape,
259                        polarization,
260                        local,
261                        &plane,
262                    );
263                } else {
264                    self.normal_plane(&mut window, channel, polarization, local)?;
265                }
266            }
267        }
268        Ok(window)
269    }
270
271    fn scatter<T: Copy>(
272        &self,
273        target: &mut [T],
274        shape: &[usize; 4],
275        polarization: usize,
276        local_channel: usize,
277        plane: &[T],
278    ) {
279        scatter_image_polarization_plane(
280            target,
281            self.member.axes().order(),
282            *shape,
283            polarization,
284            local_channel,
285            self.plane_shape,
286            plane,
287        );
288    }
289
290    /// A member formed from the final model or the CLEAN mask alone.
291    fn reconstruction_plane(
292        &self,
293        channel: usize,
294        polarization: usize,
295    ) -> Result<Vec<f32>, ProductsError> {
296        let [width, height] = self.plane_shape;
297        if self.member.role == ProductRole::CleanMask {
298            return Ok(reconstruction_support_plane(
299                self.inputs,
300                self.member.axes().domain(),
301                width * height,
302            ));
303        }
304        let model = model_real_plane(
305            self.inputs.final_model(),
306            self.domain_ordinal,
307            channel,
308            polarization,
309            self.plane_shape,
310        )?;
311        if self.planned.primary_beam_model != Some(AnalyticPrimaryBeamModel::MosaicSensitivity) {
312            return Ok(model);
313        }
314        let plane =
315            self.inputs
316                .normal_state()
317                .read_plane(self.domain_ordinal, channel, polarization)?;
318        let products = self.inputs.problem().products();
319        apparent_model(
320            model,
321            &plane,
322            products.normalization(),
323            products.validity().primary_beam(),
324            self.planned.primary_beam_model,
325        )
326    }

328    /// A member formed from a normal-state plane: its values, and its pixel
329    /// mask where it stores one.
330    fn normal_plane(
331        &self,
332        window: &mut ProductWindow,
333        channel: usize,
334        polarization: usize,
335        local_channel: usize,
336    ) -> Result<(), ProductsError> {
337        let plane =
338            self.inputs
339                .normal_state()
340                .read_plane(self.domain_ordinal, channel, polarization)?;
341        if matches!(self.member.role, ProductRole::SumWeights(_)) {
342            // CASA's `.sumwt` is the PSF gridding's sum:
343            // `FTMachine::finalizeToSkyNew` seeds the image store's sumwt
344            // once, from `makePSF` (`refim_mawproject`: 8987.97 is the weight
345            // cell's norms, 0.24 % above the data gridding's). The flat-noise
346            // products divide by the raw sensitivity, where every such sum
347            // cancels, as CASA's `divideResidualByWeight` reduces to.
348            scatter_polarization_plane_state(
349                &mut window.payload,
350                self.member.axes(),
351                window.shape,
352                polarization,
353                local_channel,
354                plane.sum_weight() as f32,
355            );
356            return Ok(());
357        }
358        let planes = self.inputs.normal_state().polarization_count();
359        let beam_index =
360            (self.domain_ordinal * self.inputs.normal_state().channel_count() + channel) * planes
361                + polarization;
362        let mut values = produce_plane_member(&PlaneMemberRequest {
363            member: self.member,
364            inputs: self.inputs,
365            plane: &plane,
366            domain_ordinal: self.domain_ordinal,
367            polarization,
368            fitted_beam: self.fitted_beams.get(beam_index).copied().flatten(),
369            restoring_beam: self.restoring_beams.get(beam_index).copied().flatten(),
370            primary_beam_model: self.planned.primary_beam_model,
371            fft_threads: self.fft_threads,
372        })?;
373        let validity = |rule| {
374            product_plane_validity(
375                rule,
376                &plane,
377                self.planned.primary_beam_model,
378                self.inputs,
379                self.member.axes().domain(),
380            )
381        };
382        if self.member.validity != ProductValidityRule::All {
383            zero_invalid_plane_values(&mut values, &validity(self.member.validity)?);
384        }
385        if let ProductPixelMask::Explicit(rule) = self.member.storage.pixel_mask() {
386            let support = validity(rule)?;
387            self.scatter(
388                &mut window.validity,
389                &window.shape,
390                polarization,
391                local_channel,
392                &support,
393            );
394        }
395        self.scatter(
396            &mut window.payload,
397            &window.shape,
398            polarization,
399            local_channel,
400            &values,
401        );
402        Ok(())
403    }
404}

406fn produce_taylor_members(
407    planned: &PlannedContinuumGeneration,
408    inputs: &ContinuumProductInputs<'_>,
409    storage_plan: ProductStoragePlan,
410    output: &CasaImageWriter,
411) -> Result<PublishedContinuumGeneration, ProductsError> {
412    let products = TaylorProducts::build(inputs, planned.psf_cutoff, planned.primary_beam_model)?;
413    let fitted_beams = if planned.requires_beam() {
414        vec![products.fitted_beam()].into_boxed_slice()
415    } else {
416        Box::new([])
417    };
418    let restoring_beams = match inputs.problem().products().restoring_beam() {
419        RestoringBeamPolicy::None => Box::new([]),
420        RestoringBeamPolicy::PerPlane | RestoringBeamPolicy::Common => {
421            vec![products.restoring_beam()].into_boxed_slice()
422        }
423    };
424    for member in &planned.members {
425        let mut payload = products.payload(member.role)?;
426        // As for one plane: a member with a validity rule is zero outside it,
427        // as `SIImageStore::makePBImage` leaves `.pb.tt0` at pblimit.
428        if member.validity != ProductValidityRule::All {
429            zero_invalid_plane_values(&mut payload, &products.validity(member.validity)?);
430        }
431        let validity = match member.storage.pixel_mask() {
432            ProductPixelMask::Absent
433            | ProductPixelMask::Explicit(
434                ProductValidityRule::All | ProductValidityRule::FinalNormalState,
435            ) => {
436                vec![true; member.payload_values]
437            }
438            ProductPixelMask::Explicit(rule) => products.validity(rule)?,
439        };
440        if payload.len() != member.payload_values || validity.len() != member.payload_values {
441            return Err(ProductsError::PayloadLengthMismatch {
442                expected: member.payload_values,
443                actual: payload.len().max(validity.len()),
444            });
445        }
446        let layout = storage_plan.layout(member.axes())?;
447        let image = output.begin(
448            member,
449            layout,
450            &planned.beams_for_member(member, &fitted_beams, &restoring_beams),
451            planned.uses_weight_image,
452        )?;
453        let mut writer = ProductMemberWriter::new(layout, image);
454        writer.write_coupled(&payload, &validity)?;
455        writer.finish()?;
456    }
457    Ok(planned.published(fitted_beams, restoring_beams))
