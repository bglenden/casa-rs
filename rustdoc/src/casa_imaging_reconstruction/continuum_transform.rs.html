10/// One channel/correlation member supplied to the row-bounded transform.
11#[derive(Debug, Clone, Copy, PartialEq)]
12pub struct ContinuumSample {
13    visibility: Complex64,
14    flag: bool,
15    weight: f64,
16    use_role: ContinuumChannelUse,
17}
18
19impl ContinuumSample {
20    /// Construct one native visibility sample and its compiled channel role.
21    #[must_use]
22    pub const fn new(
23        visibility: Complex64,
24        flag: bool,
25        weight: f64,
26        use_role: ContinuumChannelUse,
27    ) -> Self {
28        Self {
29            visibility,
30            flag,
31            weight,
32            use_role,
33        }
34    }
35
36    /// Return the untransformed complex visibility.
37    #[must_use]
38    pub const fn visibility(self) -> Complex64 {
39        self.visibility
40    }
41
42    /// Return the input flag. The transform never changes it.
43    #[must_use]
44    pub const fn flag(self) -> bool {
45        self.flag
46    }
47
48    /// Return the input statistical weight. The transform never changes it.
49    #[must_use]
50    pub const fn weight(self) -> f64 {
51        self.weight
52    }
53
54    /// Return the compiled fit/application role.
55    #[must_use]
56    pub const fn use_role(self) -> ContinuumChannelUse {
57        self.use_role
58    }
59
60    /// Return this sample with its flag replaced.
61    #[must_use]
62    pub const fn with_flag(mut self, flag: bool) -> Self {
63        self.flag = flag;
64        self
65    }
66}

70pub struct ContinuumRowInput<'a> {

92/// Outcome of CASA's per-row/per-correlation valid-point policy.
93#[derive(Debug, Clone, Copy, PartialEq, Eq)]
94pub enum ContinuumFitStatus {
95    /// A fit was evaluated, possibly after reducing the requested order.
96    Fitted {
97        /// Polynomial order actually used for this row/correlation.
98        effective_order: u8,
99    },
100    /// No unflagged, positively weighted fit samples existed.
101    NoValidFitSamples,
102}

104/// CASA-compatible continuum prediction and residual for one correlation.
105#[derive(Debug, Clone, PartialEq)]
106pub struct ContinuumRowResult {
107    samples: Vec<ContinuumSample>,
108    normalized_frequencies: Vec<f64>,
109    fit_sample_indices: Vec<usize>,
110    coefficients: Vec<Complex64>,

112    residual: Vec<Complex64>,
113    status: ContinuumFitStatus,
114    chi_squared: [f64; 2],
115}

117impl ContinuumRowResult {
118    /// Return the immutable source metadata retained for every channel.

122    }
123
124    /// Return CASA's complete-row min/max normalized frequency coordinates.
125    ///

140    }

147
148    /// Return the fitted continuum at every supplied native frequency.

152    }
153
154    /// Return input minus prediction at every supplied native frequency.

156    pub fn residual(&self) -> &[Complex64] {

158    }
159
160    /// Return the valid-point disposition and effective order.
161    #[must_use]
162    pub const fn status(&self) -> ContinuumFitStatus {
163        self.status
164    }
165
166    /// Return weighted real and imaginary residual sums of squares on fit data.

170    }
171}

173/// Invalid row input or a singular continuum fit.
174#[derive(Debug, Clone, Copy, PartialEq, Eq, Error)]
175pub enum ContinuumFitError {
176    /// Frequency and sample arrays were empty or had different lengths.
177    #[error("continuum row frequencies and samples must be nonempty and equal-length")]
178    InvalidShape,
179    /// A frequency or visibility value was not finite.
180    #[error("continuum row contains a non-finite frequency or visibility")]
181    NonFiniteInput,
182    /// The valid fit samples did not span the effective polynomial basis.
183    #[error("continuum fit is singular for the effective polynomial order")]
184    SingularFit,
185}
186
187/// Fit and subtract a CASA-compatible polynomial continuum from one correlation.
188///
189/// Frequencies are normalized over the complete supplied row domain with
190/// `x = (frequency - midpoint) / (minimum - midpoint)`. Fit and application
191/// roles are independent; predictions are returned for every channel while
192/// callers decide which residual members enter line imaging.
193pub fn fit_and_subtract_continuum(
194    input: ContinuumRowInput<'_>,
195) -> Result<ContinuumRowResult, ContinuumFitError> {
196    if input.samples.is_empty() || input.samples.len() != input.frequencies_hz.len() {
197        return Err(ContinuumFitError::InvalidShape);
198    }
199    if input.frequencies_hz.iter().any(|value| !value.is_finite())
200        || input
201            .samples
202            .iter()
203            .any(|sample| !sample.visibility.re.is_finite() || !sample.visibility.im.is_finite())
204    {
205        return Err(ContinuumFitError::NonFiniteInput);
206    }
207
208    let normalized = casa_normalized_frequency_axis(input.frequencies_hz);
209    let valid_indices = input
210        .samples
211        .iter()
212        .enumerate()
213        .filter_map(|(index, sample)| {
214            (sample.use_role.contributes_to_fit()
215                && !sample.flag
216                && sample.weight.is_finite()
217                && sample.weight > 0.0)
218                .then_some(index)
219        })
220        .collect::<Vec<_>>();
221    if valid_indices.is_empty() {
222        return Ok(ContinuumRowResult {
223            samples: input.samples.to_vec(),
224            normalized_frequencies: normalized,
225            fit_sample_indices: valid_indices,
226            coefficients: Vec::new(),
227            prediction: vec![Complex64::new(0.0, 0.0); input.samples.len()],
228            residual: input
229                .samples
230                .iter()
231                .map(|sample| sample.visibility)
232                .collect(),
233            status: ContinuumFitStatus::NoValidFitSamples,
234            chi_squared: [f64::INFINITY; 2],
235        });
236    }
237
238    let effective_order = input
239        .requested_order
240        .min(u8::try_from(valid_indices.len() - 1).unwrap_or(u8::MAX));
241    let coefficient_count = usize::from(effective_order) + 1;
242    let rows = |component: fn(Complex64) -> f64| {
243        valid_indices
244            .iter()
245            .map(|&index| {
246                (
247                    monomial_basis(normalized[index], coefficient_count),
248                    component(input.samples[index].visibility),
249                    input.samples[index].weight,
250                )
251            })
252            .collect::<Vec<_>>()
253    };
254    let real = solve_weighted_least_squares(&rows(|value| value.re), coefficient_count)
255        .ok_or(ContinuumFitError::SingularFit)?;
256    let imaginary = solve_weighted_least_squares(&rows(|value| value.im), coefficient_count)
257        .ok_or(ContinuumFitError::SingularFit)?;
258    let coefficients = real
259        .into_iter()
260        .zip(imaginary)
261        .map(|(real, imaginary)| Complex64::new(real, imaginary))
262        .collect::<Vec<_>>();
263    let prediction = normalized
264        .iter()
265        .map(|&x| evaluate_polynomial(&coefficients, x))
266        .collect::<Vec<_>>();
267    let residual = input
268        .samples
269        .iter()
270        .zip(&prediction)
271        .map(|(sample, model)| sample.visibility - model)
272        .collect::<Vec<_>>();
273    let mut chi_squared = [0.0_f64; 2];
274    for &index in &valid_indices {
275        chi_squared[0] += input.samples[index].weight * residual[index].re.powi(2);
276        chi_squared[1] += input.samples[index].weight * residual[index].im.powi(2);
277    }
278    Ok(ContinuumRowResult {
279        samples: input.samples.to_vec(),
280        normalized_frequencies: normalized,
281        fit_sample_indices: valid_indices,
282        coefficients,
283        prediction,
284        residual,
285        status: ContinuumFitStatus::Fitted { effective_order },
286        chi_squared,
287    })
288}

290fn casa_normalized_frequency_axis(frequencies_hz: &[f64]) -> Vec<f64> {
291    let minimum = frequencies_hz
292        .iter()
293        .copied()
294        .min_by(f64::total_cmp)
295        .expect("validated nonempty frequencies");
296    let maximum = frequencies_hz
297        .iter()
298        .copied()
299        .max_by(f64::total_cmp)
300        .expect("validated nonempty frequencies");
301    let midpoint = 0.5 * (minimum + maximum);
302    let denominator = minimum - midpoint;
303    if denominator == 0.0 {
304        vec![0.0; frequencies_hz.len()]
305    } else {
306        frequencies_hz
307            .iter()
308            .map(|frequency| (frequency - midpoint) / denominator)
309            .collect()
310    }
311}

313fn monomial_basis(x: f64, count: usize) -> Vec<f64> {
314    let mut terms = Vec::with_capacity(count);
315    let mut value = 1.0;
316    for _ in 0..count {
317        terms.push(value);
318        value *= x;
319    }
320    terms
321}
322
323fn evaluate_polynomial(coefficients: &[Complex64], x: f64) -> Complex64 {
324    coefficients
325        .iter()
326        .rev()
327        .fold(Complex64::new(0.0, 0.0), |value, coefficient| {
328            value * x + coefficient
329        })
