51pub(super) fn decode(
52    name: &str,
53    record: &RecordValue,
54    coords: &CoordinateSystem,
55    full_shape: &[usize],
56    plane: [usize; 2],
57) -> Result<ParsedRegion, ImageError> {
58    let region = compile(name, record, coords, full_shape, plane)?;
59    let shape = region
60        .start
61        .iter()
62        .zip(&region.end)
63        .map(|(&start, &end)| end - start)
64        .collect::<Vec<_>>();
65    Ok(ParsedRegion {
66        is_compound: true,
67        is_crtf: false,
68        shapes: Vec::new(),
69        saved_bounds: Some((region.start.clone(), shape)),
70        saved_mask: Some(SavedRegionMask::Compound(region)),
71        stride: None,
72    })
73}

75pub(super) fn bounds(
76    name: &str,
77    record: &RecordValue,
78    coords: &CoordinateSystem,
79    full_shape: &[usize],
80    plane: [usize; 2],
81) -> Result<(Vec<usize>, Vec<usize>), ImageError> {
82    let region = compile(name, record, coords, full_shape, plane)?;
83    let count = region
84        .start
85        .iter()
86        .zip(region.end)
87        .map(|(&start, end)| end - start)
88        .collect();
89    Ok((region.start, count))
90}

92fn compile(
93    name: &str,
94    record: &RecordValue,
95    coords: &CoordinateSystem,
96    full_shape: &[usize],
97    plane: [usize; 2],
98) -> Result<Region, ImageError> {
99    let class = native_region_kind(name, record)?;
100    let operation = match class {
101        "WCUnion" => Operation::Union,
102        "WCIntersection" => Operation::Intersection,
103        "WCDifference" => Operation::Difference,
104        "WCComplement" => Operation::Complement,
105        "WCExtension" => return compile_extension(name, record, coords, full_shape, plane),
106        "WCConcatenation" => return compile_concatenation(name, record, coords, full_shape, plane),
107        "WCPolygon" => {
108            let ([x, y], local) = polygon_plane(name, record, coords, full_shape)?;
109            let (start, shape) = local.section(full_shape, [x, y]);
110            let end = start
111                .iter()
112                .zip(shape)
113                .map(|(&start, size)| start + size)
114                .collect();
115            let mut axes = vec![x, y];
116            axes.sort_unstable();
117            return Ok(Region {
118                start,
119                end,
120                axes,
121                selection: Selection::Polygon {
122                    axes: [x, y],
123                    plane: local,
124                },
125            });
126        }
127        "WCBox" | "WCEllipsoid" => {
128            let (mut axes, _) = wc_pixel_axes(name, class, record, coords)?;
129            axes.sort_unstable();
130            let leaf = parse_saved_region(name, record, coords, full_shape, plane)?;
131            let shapes = leaf
132                .shapes
133                .iter()
134                .map(|shape| &shape.spatial)
135                .collect::<Vec<_>>();
136            let resolved = resolve_spatial_and_saved_bounds(
137                full_shape,
138                full_shape.len(),
139                plane,
140                &shapes,
141                leaf.saved_bounds.as_ref(),
142                leaf.saved_mask.is_some(),
143            )?;
144            let end = resolved
145                .start
146                .iter()
147                .zip(&resolved.shape)
148                .map(|(start, size)| start + size)
149                .collect();
150            return Ok(Region {
151                start: resolved.start,
152                end,
153                axes,
154                selection: Selection::Spatial {
155                    axes: plane,
156                    shapes: resolved.spatial_shapes,
157                },
158            });
159        }
160        _ => {
161            return Err(ImageError::InvalidMetadata(format!(
162                "saved region '{name}' contains unsupported compound member {class:?}"
163            )));
164        }
165    };
166    let members = compound_children(name, record)?;
167    let valid_count = match operation {
168        Operation::Difference => members.len() == 2,
169        Operation::Complement => members.len() == 1,
170        Operation::Union | Operation::Intersection => !members.is_empty(),
171    };
172    if !valid_count {
173        return Err(ImageError::InvalidMetadata(format!(
174            "saved region '{name}' has an invalid number of compound members"
175        )));
176    }
177    let children = members
178        .into_iter()
179        .map(|child| compile(name, child, coords, full_shape, plane))
180        .collect::<Result<Vec<_>, _>>()?;
181    compound_region(name, operation, children, full_shape)
182}

184fn compound_region(
185    name: &str,
186    operation: Operation,
187    children: Vec<Region>,
188    full_shape: &[usize],
189) -> Result<Region, ImageError> {
190    // casacore's LC compounds define their bounds independently of the
191    // result mask. Difference can remove an entire edge of its first child.
192    let (start, end) = match operation {
193        Operation::Complement => (vec![0; full_shape.len()], full_shape.to_vec()),
194        Operation::Difference => (children[0].start.clone(), children[0].end.clone()),
195        Operation::Union | Operation::Intersection => {
196            let (start, end) =
197                combined_bounds(&children, matches!(operation, Operation::Intersection));
198            if start.iter().zip(&end).any(|(start, end)| start >= end) {
199                return Err(ImageError::CasaSelection(format!(
200                    "saved region '{name}': regions do not overlap"
201                )));
202            }
203            (start, end)
204        }
205    };
206    let axes = (0..full_shape.len())
207        .filter(|axis| children.iter().any(|child| child.axes.contains(axis)))
208        .collect();
209    Ok(Region {
210        start,
211        end,
212        axes,
213        selection: Selection::Compound(operation, children),
214    })
215}
216
217fn combined_bounds(children: &[Region], intersection: bool) -> (Vec<usize>, Vec<usize>) {
218    let mut start = children[0].start.clone();
219    let mut end = children[0].end.clone();
220    for child in &children[1..] {
221        for axis in 0..start.len() {
222            if intersection {
223                start[axis] = start[axis].max(child.start[axis]);
224                end[axis] = end[axis].min(child.end[axis]);
225            } else {
226                start[axis] = start[axis].min(child.start[axis]);
227                end[axis] = end[axis].max(child.end[axis]);
228            }
229        }
230    }
231    (start, end)
232}

234fn compile_extension(
235    name: &str,
236    record: &RecordValue,
237    coords: &CoordinateSystem,
238    full_shape: &[usize],
239    plane: [usize; 2],
240) -> Result<Region, ImageError> {
241    let members = compound_children(name, record)?;
242    let invalid = || {
243        ImageError::InvalidMetadata(format!(
244            "saved region '{name}': WCExtension requires a region and an extension WCBox"
245        ))
246    };
247    let [original, extent] = members.as_slice() else {
248        return Err(invalid());
249    };
250    if native_region_kind(name, extent)? != "WCBox" {
251        return Err(invalid());
252    }
253    let child = compile(name, original, coords, full_shape, plane)?;
254    let (box_axes, _) = wc_pixel_axes(name, "WCBox", extent, coords)?;
255    let mut stretch = Vec::new();
256    let mut start = child.start.clone();
257    let mut end = child.end.clone();
258    let belongs_to_child = |axis| child.axes.contains(&axis);
259    // WCExtension splits the WCBox before converting world bounds. Its
260    // shared axes stretch a one-pixel plane and the other axes extend it.
261    for shared in [true, false] {
262        let selected = |axis| belongs_to_child(axis) == shared;
263        if !box_axes.iter().copied().any(selected) {
264            continue;
265        }
266        let (low, high) = decode_wc_box_axes(name, extent, coords, full_shape, selected)?;
267        for axis in box_axes.iter().copied().filter(|&axis| selected(axis)) {
268            if shared {
269                if end[axis] - start[axis] != 1 {
270                    return Err(ImageError::CasaSelection(format!(
271                        "saved region '{name}': a stretch axis does not have length 1"
272                    )));
273                }
274                stretch.push((axis, start[axis]));
275            }
276            start[axis] = low[axis];
277            end[axis] = high[axis] + 1;
278        }
279    }
280    let mut axes = child.axes.clone();
281    axes.extend(box_axes);
282    axes.sort_unstable();
283    axes.dedup();
284    Ok(Region {
285        start,
286        end,
287        axes,
288        selection: Selection::Extension {
289            child: Box::new(child),
290            stretch,
291        },
292    })
293}

295fn compile_concatenation(
296    name: &str,
297    record: &RecordValue,
298    coords: &CoordinateSystem,
299    full_shape: &[usize],
300    plane: [usize; 2],
301) -> Result<Region, ImageError> {
302    let invalid = || {
303        ImageError::InvalidMetadata(format!(
304            "saved region '{name}': WCConcatenation requires a one-dimensional extendBox"
305        ))
306    };
307    let Some(Value::Record(extent)) = record.get("box") else {
308        return Err(invalid());
309    };
310    if native_region_kind(name, extent)? != "WCBox" {
311        return Err(invalid());
312    }
313    let (box_axes, _) = wc_pixel_axes(name, "WCBox", extent, coords)?;
314    let [axis] = box_axes.as_slice() else {
315        return Err(invalid());
316    };
317    let axis = *axis;
318    let (low, high) = decode_wc_box(name, extent, coords, full_shape)?;
319    let members = compound_children(name, record)?;
320    if members.len() != high[axis] - low[axis] + 1 {
321        return Err(ImageError::CasaSelection(format!(
322            "saved region '{name}': number of regions has to match the range specified in extendBox"
323        )));
324    }
325    let children = members
326        .into_iter()
327        .map(|child| compile(name, child, coords, full_shape, plane))
328        .collect::<Result<Vec<_>, _>>()?;
329    let mut axes = children[0].axes.clone();
330    if axes.contains(&axis) || children.iter().any(|child| child.axes != axes) {
331        return Err(ImageError::CasaSelection(format!(
332            "saved region '{name}': concatenated members must share axes distinct from the extendBox"
333        )));
334    }
335    let (mut start, mut end) = combined_bounds(&children, false);
336    start[axis] = low[axis];
337    end[axis] = high[axis] + 1;
338    axes.push(axis);
339    axes.sort_unstable();
340    Ok(Region {
341        start,
342        end,
343        axes,
344        selection: Selection::Concatenation { axis, children },
345    })
346}

348impl Region {
349    pub(in crate::analysis::imsubimage::region) fn overlaps(
350        &self,
351        start: &[usize],
352        shape: &[usize],
353    ) -> bool {
354        if start
355            .iter()
356            .zip(shape)
357            .zip(self.start.iter().zip(&self.end))
358            .any(|((&start, &length), (&min, &end))| start >= end || start + length <= min)
359        {
360            return false;
361        }
362        match &self.selection {
363            Selection::Compound(Operation::Union, children) => {
364                children.iter().any(|child| child.overlaps(start, shape))
365            }
366            Selection::Compound(Operation::Intersection, children) => {
367                children.iter().all(|child| child.overlaps(start, shape))
368            }
369            Selection::Compound(Operation::Difference, children) => {
370                children[0].overlaps(start, shape)
371            }
372            _ => true,
373        }
374    }
375
376    pub(in crate::analysis::imsubimage::region) fn contains(&self, pixel: &mut [usize]) -> bool {
377        if pixel
378            .iter()
379            .zip(self.start.iter().zip(&self.end))
380            .any(|(&index, (&start, &end))| index < start || index >= end)
381        {
382            return false;
383        }
384        match &self.selection {
385            Selection::Spatial {
386                axes: [x, y],
387                shapes,
388            } => {
389                shapes.is_empty()
390                    || shapes
391                        .iter()
392                        .any(|shape| shape_contains(shape, pixel[*x], pixel[*y]))
393            }
394            Selection::Polygon {
395                axes: [x, y],
396                plane,
397            } => plane.contains(pixel[*x], pixel[*y]),
398            Selection::Extension { child, stretch } => contains_stretched(child, stretch, pixel),
399            Selection::Concatenation { axis, children } => {
400                let plane_index = pixel[*axis] - self.start[*axis];
401                children[plane_index].contains(pixel)
402            }
403            Selection::Compound(operation, children) => match operation {
404                Operation::Union => children.iter().any(|child| child.contains(pixel)),
405                Operation::Intersection => children.iter().all(|child| child.contains(pixel)),
406                Operation::Difference => {
407                    children[0].contains(pixel) && !children[1].contains(pixel)
408                }
409                Operation::Complement => !children[0].contains(pixel),
410            },
411        }
412    }
413}
414
415fn contains_stretched(child: &Region, stretch: &[(usize, usize)], pixel: &mut [usize]) -> bool {
416    let Some((&(axis, source), rest)) = stretch.split_first() else {
417        return child.contains(pixel);
418    };
419    let saved = std::mem::replace(&mut pixel[axis], source);
420    let included = contains_stretched(child, rest, pixel);
421    pixel[axis] = saved;
422    included
423}
