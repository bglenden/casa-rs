9const COLUMNS: [(&str, &str, &str); 14] = [
10    ("MJD", "d", "Date__(UT)"),
11    ("RA", "deg", "R.A.___(ICRF)___DEC"),
12    ("DEC", "deg", "R.A.___(ICRF)___DEC"),
13    ("Rho", "AU", "delta"),
14    ("RadVel", "AU/d", "deldot"),
15    ("NP_ang", "deg", "NP.ang"),
16    ("NP_dist", "arcsec", "NP.dist"),
17    ("DiskLong", "deg", "ObsSub-LON"),
18    ("DiskLat", "deg", "ObsSub-LAT"),
19    ("Sl_lon", "deg", "SunSub-LON"),
20    ("Sl_lat", "deg", "SunSub-LAT"),
21    ("r", "AU", "r"),
22    ("rdot", "km/s", "rdot"),
23    ("phang", "deg", "S-T-O"),
24];

26pub(super) fn ephemeris_table(response: &str) -> Result<Table, String> {
27    let (preamble, body) = response
28        .split_once("$$SOE")
29        .ok_or("no ephemeris data found")?;
30    let (body, _) = body
31        .split_once("$$EOE")
32        .ok_or("ephemeris data has no end marker")?;
33    let heading = preamble
34        .lines()
35        .find(|line| line.trim_start().starts_with("Date__(UT)"))
36        .ok_or("no ephemeris data header was found")?;
37    let names: Vec<_> = heading.split_whitespace().collect();
38    let indexes = column_indexes(&names)?;
39    let radial_velocity_scale = Unit::new("AU/d")
40        .and_then(|unit| Quantity::new(1., "km/s")?.get_value_in(&unit))
41        .map_err(|error| error.to_string())?;
42    let mut rows: Vec<[f64; COLUMNS.len()]> = Vec::new();
43    for line in body.lines().filter(|line| !line.trim().is_empty()) {
44        let mut row = parse_row(line, &indexes, names.len() + 2)?;
45        row[4] *= radial_velocity_scale;
46        if rows.last().is_some_and(|previous| previous[0] == row[0]) {
47            return Err(format!(
48                "duplicated timestamp in ephemeris data: {}",
49                row[0]
50            ));
51        }
52        rows.push(row);
53    }
54    let first = rows.first().ok_or("no ephemeris data found")?[0];
55    let last = rows.last().expect("nonempty ephemeris")[0];
56    let metadata = metadata(preamble, &rows, first, last)?;
57    let schema = TableSchema::new(
58        COLUMNS
59            .iter()
60            .map(|(column, _, _)| ColumnSchema::scalar(*column, PrimitiveType::Float64))
61            .collect(),
62    )
63    .map_err(|error| error.to_string())?;
64    let mut table = Table::with_schema(schema);
65    for row in rows {
66        table
67            .add_row(RecordValue::new(
68                COLUMNS
69                    .iter()
70                    .zip(row)
71                    .map(|((column, _, _), value)| RecordField::new(*column, number(value)))
72                    .collect(),
73            ))
74            .map_err(|error| error.to_string())?;
75    }
76    *table.keywords_mut() = metadata;
77    for (column, unit, _) in COLUMNS {
78        table.set_column_keywords(
79            column,
80            RecordValue::new(vec![RecordField::new(
81                "QuantumUnits",
82                Value::Array(ArrayValue::from_string_vec(vec![unit.into()])),
83            )]),
84        );
85    }
86    table.set_info(casa_tables::TableInfo {
87        table_type: "IERS".to_owned(),
88        sub_type: "Comet".to_owned(),
89        readme: vec!["Derivied by jplhorizons-query from JPL-Horizons API (https://ssd.jpl.nasa.gov/api/horizons.api)".to_owned()],
90    });
91    Ok(table)
92}

94fn column_indexes(names: &[&str]) -> Result<[usize; COLUMNS.len()], String> {
95    let mut indexes = [0; COLUMNS.len()];
96    for (index, (column, _, heading)) in COLUMNS.into_iter().enumerate().skip(1) {
97        let position = names
98            .iter()
99            .position(|name| *name == heading)
100            .ok_or_else(|| format!("missing Horizons column {heading}"))?;
101        indexes[index] = position
102            + match column {
103                "RA" => 1,
104                _ => 2,
105            };
106    }
107    Ok(indexes)
108}

110fn parse_row(
111    line: &str,
112    indexes: &[usize; COLUMNS.len()],
113    expected: usize,
114) -> Result<[f64; COLUMNS.len()], String> {
115    let fields: Vec<_> = line.split_whitespace().collect();
116    // Topocentric data may carry a solar/lunar presence code immediately
117    // after its calendar date. It is optional even within the same table.
118    if !matches!(fields.len().checked_sub(expected), Some(0 | 1)) {
119        return Err(format!(
120            "unexpected number of Horizons data items: {} (expected {expected})",
121            fields.len()
122        ));
123    }
124    let offset = fields.len() - expected;
125    let mut row = [0.; COLUMNS.len()];
126    row[0] = calendar(fields[0], fields[1]).ok_or("invalid Horizons calendar row")?;
127    for (index, &source) in indexes.iter().enumerate().skip(1) {
128        let value = fields
129            .get(source + offset)
130            .ok_or("short Horizons data row")?;
131        row[index] = if *value == "n.a." {
132            -999.
133        } else {
134            value
135                .parse()
136                .map_err(|_| format!("invalid Horizons data item {value}"))?
137        };
138    }
139    Ok(row)
140}

142fn metadata(
143    preamble: &str,
144    rows: &[[f64; COLUMNS.len()]],
145    first: f64,
146    last: f64,
147) -> Result<RecordValue, String> {
148    let mut keys = RecordValue::new(Vec::new());
149    keys.upsert("VS_VERSION", text("0004.0001"));
150    keys.upsert("VS_TYPE", text("Table of comet/planetary positions"));
151    let whole_minutes = std::time::SystemTime::now()
152        .duration_since(std::time::UNIX_EPOCH)
153        .map_err(|error| error.to_string())?
154        .as_secs()
155        / 60;
156    let now = format_time(whole_minutes as f64 / 1_440. + 40_587., YMD, 6);
157    keys.upsert(
158        "VS_DATE",
159        text(now.rsplit_once(':').map_or(&now, |(minute, _)| minute)),
160    );
161    keys.upsert("posrefsys", text("ICRS"));
162    let mut start = first;
163    let mut stop = last;
164    let mut step = None;
165    let mut synchronous = false;
166    for line in preamble.lines().map(str::trim_start) {
167        if line.starts_with("Ephemeris / ") {
168            let words: Vec<_> = line.split_whitespace().collect();
169            if let [_, _, _, _, month, day, time, year, ..] = words.as_slice()
170                && let (Some(month), Ok(day), Some(time)) =
171                    (month_index(month), day.parse::<u32>(), time.get(..5))
172            {
173                keys.upsert(
174                    "VS_CREATE",
175                    text(&format!("{year}/{month:02}/{day:02}/{time}")),
176                );
177            }
178        } else if let Some((label, value)) = line.trim_start_matches('>').split_once(':') {
179            let value = value.trim();
180            match label.trim() {
181                "Target body name" => {
182                    let (name, source) = object_name(value);
183                    keys.upsert("NAME", text(name));
184                    if let Some(source) = source {
185                        keys.upsert("ephemeris_source", text(source));
186                    }
187                }
188                "Start time" => start = header_epoch(value).unwrap_or(start),
189                "Stop  time" => stop = header_epoch(value).unwrap_or(stop),
190                "Step-size" => step = Some(value),
191                "Center-site name" => {
192                    let name = if value.contains("ALMA") {
193                        "ALMA"
194                    } else if value.contains("VLA") {
195                        "VLA"
196                    } else if value.contains("Green Bank") {
197                        "GBT"
198                    } else {
199                        value
200                    };
201                    keys.upsert("obsloc", text(name));
202                }
203                "Center geodetic" => {
204                    if let Some([lon, lat, distance]) =
205                        triplet(value.split('{').next().unwrap_or(value))
206                    {
207                        keys.upsert("GeoLong", number(lon));
208                        keys.upsert("GeoLat", number(lat));
209                        keys.upsert("GeoDist", number(distance));
210                    }
211                }
212                "Target radii" => {
213                    if let Some(radii) = value.split_once("km").and_then(|(radii, _)| {
214                        triplet(radii)
215                            .or_else(|| radii.trim().parse().ok().map(|radius| [radius; 3]))
216                    }) {
217                        let eq = (radii[0] + radii[1]) / 2.;
218                        let mean = rows
219                            .iter()
220                            .map(|row| {
221                                // This uses CASA's historical latitude scaling.
222                                let latitude = if row[8] == -999. {
223                                    0.
224                                } else {
225                                    row[8].to_degrees()
226                                };
227                                (eq * (eq * latitude.sin()).hypot(radii[2] * latitude.cos())).sqrt()
228                            })
229                            .sum::<f64>()
230                            / rows.len() as f64;
231                        keys.upsert(
232                            "radii",
233                            quantity("km", Value::Array(ArrayValue::from_f64_vec(radii.to_vec()))),
234                        );
235                        keys.upsert("meanrad", quantity("km", number(mean)));
236                    }
237                }
238                _ => {}
239            }
240        }
241        if keys.get("orb_per").is_none()
242            && let Some((value, unit)) = period(
243                line,
244                &["Orbital period", "orb. per.,", "orb per", "orbit period"],
245            )
246        {
247            keys.upsert("orb_per", quantity(&unit, number(value)));
248        }
249        if line.contains("Rotational period") && line.contains("Synchronous") {
250            synchronous = true;
251        } else if let Some((value, unit)) = period(
252            &line.to_ascii_lowercase(),
253            &["rot. period", "rot. per.", "rotational period", "rotper"],
254        ) {
255            keys.upsert("rot_per", quantity(&unit, number(value)));
256        }
257        if let Some((value, _)) = period(line, &["Mean Temperature", "Mean temperature"]) {
258            keys.upsert("T_mean", quantity("K", number(value)));
259        }
260    }
261    if synchronous {
262        let period = keys
263            .get("orb_per")
264            .cloned()
265            .unwrap_or_else(|| text("Synchronous"));
266        keys.upsert("rot_per", period);
267    }
268    let step = step
269        .map(|step| step_days(step, start, stop))
270        .transpose()?
271        .unwrap_or_else(|| {
272            if rows.len() > 1 {
273                rows[1][0] - first
274            } else {
275                0.
276            }
277        });
278    if step <= 0. || !step.is_finite() {
279        return Err("Horizons ephemeris step must be positive".into());
280    }
281    keys.upsert("dMJD", number(step));
282    keys.upsert("MJD0", number(start - step));
283    keys.upsert("earliest", epoch(first));
284    keys.upsert("latest", epoch(last));
285    Ok(keys)
286}

288fn object_name(value: &str) -> (&str, Option<&str>) {
289    let (name, source) = value
290        .split_once("{source:")
291        .map_or((value, None), |(name, source)| {
292            (name, Some(source.trim().trim_end_matches('}').trim()))
293        });
294    let mut name = name.trim();
295    if let Some((first, second)) = name.trim_end_matches(')').rsplit_once(" (") {
296        name = if second.trim().parse::<f64>().is_ok() {
297            first.trim()
298        } else {
299            second.trim()
300        };
301    }
302    (name, source)
303}

305fn period(line: &str, labels: &[&str]) -> Option<(f64, String)> {
306    let label = labels
307        .iter()
308        .find_map(|label| line.split_once(label).map(|(_, tail)| tail))?;
309    let (prelude, value) = label.split_once(['=', '~'])?;
310    let value = value.trim();
311    if labels.contains(&"rotper")
312        && let Some(hours) = rotational_clock_hours(value)
313    {
314        return Some((hours, "h".into()));
315    }
316    let end = value
317        .find(|ch: char| !matches!(ch, '0'..='9' | '.' | '-'))
318        .unwrap_or(value.len());
319    let number = value[..end].parse().ok()?;
320    let unit = value[end..]
321        .split_whitespace()
322        .next()
323        .filter(|unit| unit.chars().all(char::is_alphabetic))
324        .or_else(|| {
325            prelude
326                .trim()
327                .chars()
328                .all(char::is_alphabetic)
329                .then_some(prelude.trim())
330                .filter(|unit| !unit.is_empty())
331        })
332        .unwrap_or(if labels.contains(&"rotper") { "h" } else { "d" });
333    Some((number, unit.to_owned()))
334}

336fn rotational_clock_hours(value: &str) -> Option<f64> {
337    let (hours, rest) = value.split_once('h')?;
338    let (minutes, rest) = rest.split_once('m')?;
339    let (seconds, _) = rest.split_once('s')?;
340    let parse = |value: &str| value.trim().parse::<f64>().ok();
341    Some(parse(hours)? + parse(minutes)? / 60. + parse(seconds)? / 3_600.)
342}

344fn step_days(value: &str, start: f64, stop: f64) -> Result<f64, String> {
345    let mut words = value.split_whitespace();
346    let count: f64 = words
347        .next()
348        .and_then(|value| value.parse().ok())
349        .ok_or("invalid Horizons step size")?;
350    let unit = words.next().ok_or("no Horizons step unit")?;
351    match unit {
352        "minutes" => Ok(count / 1_440.),
353        "hours" => Ok(count / 24.),
354        "days" => Ok(count),
355        "steps" => Ok((stop - start) / count),
356        "calendar" => Err("Unit of Step-size in calendar month or year is not supported".into()),
357        _ => Err(format!("Unit of Step-size, {unit} is not recognized")),
358    }
359}

361fn triplet(value: &str) -> Option<[f64; 3]> {
362    let mut fields = value
363        .split([',', 'x'])
364        .map(|value| value.trim().parse::<f64>());
365    Some([
366        fields.next()?.ok()?,
367        fields.next()?.ok()?,
368        fields.next()?.ok()?,
369    ])
370}

372fn header_epoch(value: &str) -> Option<f64> {
373    let mut fields = value.split_whitespace();
374    fields.next()?; // A.D.
375    calendar(fields.next()?, fields.next()?)
376}
377
378fn calendar(date: &str, time: &str) -> Option<f64> {
379    let (year_month, day) = date.rsplit_once('-')?;
380    let (year, month) = year_month.rsplit_once('-')?;
381    let month = month_index(month)?;
382    Quantity::read(&format!("{year}/{month:02}/{day}/{time}"))
383        .ok()
384        .map(|epoch| epoch.value())
385}
386
387fn month_index(month: &str) -> Option<u32> {
388    month.parse().ok().or_else(|| {
389        [
390            "Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
391        ]
392        .iter()
393        .position(|name| name.eq_ignore_ascii_case(month))
394        .map(|index| index as u32 + 1)
395    })
396}

398fn number(value: f64) -> Value {
399    Value::Scalar(ScalarValue::Float64(value))
400}
401
402fn text(value: &str) -> Value {
403    Value::Scalar(ScalarValue::String(value.into()))
404}
405
406fn quantity(unit: &str, value: Value) -> Value {
407    Value::Record(RecordValue::new(vec![
408        RecordField::new("unit", text(unit)),
409        RecordField::new("value", value),
410    ]))
411}
412
413fn epoch(mjd: f64) -> Value {
414    Value::Record(RecordValue::new(vec![
415        RecordField::new("m0", quantity("d", number(mjd))),
416        RecordField::new("refer", text("UTC")),
417        RecordField::new("type", text("epoch")),
418    ]))
