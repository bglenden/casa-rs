44impl ParallacticAverages {
45    pub(super) fn new(
46        channels: usize,
47        circular: bool,
48        gain_phase: bool,
49        mode: GainSolveMode,
50        legacy: bool,
51    ) -> Self {
52        Self {
53            circular,
54            gain_phase,
55            legacy,
56            mode,
57            segments: BTreeMap::new(),
58            rows: ChannelSamples::new(channels),
59        }
60    }
61
62    pub(super) fn add(
63        &mut self,
64        antennas: [i32; 2],
65        context: BaselineSegment,
66        channels: &mut [CoherencyMean],
67    ) -> Result<(), GainSolveError> {
68        let id = self.segments.len();
69        let key_time = if self.legacy && self.gain_phase {
70            context.start
71        } else {
72            context.emission
73        };
74        let segment = self
75            .segments
76            .entry((key_time.to_bits(), context.field, context.data_desc))
77            .or_insert_with(|| SegmentRows {
78                id,
79                context,
80                flags: Vec::new(),
81                usable: false,
82            });
83        let mut unflagged = vec![0; channels.len().div_ceil(64)];
84        let legacy_phase_normalization = (self.legacy && self.mode == GainSolveMode::Phase)
85            .then(|| super::cross_hand::legacy_phase_normalization(channels));
86        for (channel, mean) in channels.iter_mut().enumerate() {
87            let mean = std::mem::take(mean);
88            if mean.unflagged != 0 {
89                unflagged[channel / 64] |= 1 << (channel % 64);
90                segment.usable = true;
91            }
92            if (!self.gain_phase || antennas[0] != antennas[1])
93                && let Some((data, model, weights)) = mean.finish(false)
94                && (weights[1] > 0.0 || weights[2] > 0.0)
95            {
96                let [(xy, wx), (yx, wy)] = [1, 2].map(|corr| {
97                    let (value, weight) = solve_sample(self.mode, data[corr], weights[corr]);
98                    let weight = legacy_phase_normalization
99                        .map_or(weight, |normalization| normalization[corr] * weights[corr]);
100                    (value, weight)
101                });
102                let sample = PhaseSample {
103                    segment: segment.id as u32,
104                    antennas,
105                    data: [xy, yx],
106                    model: [model[1], model[2]],
107                    weights: if self.legacy && self.gain_phase {
108                        // VI1 weights both cross-hands by the sum of their weights.
109                        [0.5 * (wx + wy); 2]
110                    } else {
111                        [wx, wy]
112                    },
113                };
114                self.rows
115                    .push(channel, sample)
116                    .map_err(GainSolveError::PolarizationScratch)?;
117            }
118        }
119        segment.flags.push(BaselineFlags {
120            antennas,
121            unflagged,
122        });
123        Ok(())
124    }
125
126    pub(super) fn finish(
127        mut self,
128        angles: &ParallacticAngleState,
129        legacy_times: Option<&LegacyCrossHandAveraging>,
130    ) -> Result<(Vec<Option<Complex32>>, Complex32), GainSolveError> {
131        let channels = self.rows.len();
132        let mut positions = vec![None; self.segments.len()];
133        let mut usable = self.segments.into_values().filter(|segment| segment.usable);
134        let first = usable
135            .next()
136            .ok_or(GainSolveError::NoUnflaggedParallacticData)?;
137        // CASA starts with the baselines in the first usable segment,
138        // then extends their channel flags through all usable segments.
139        let mut accepted = first
140            .flags
141            .iter()
142            .map(|row| (row.antennas, row.unflagged.clone()))
143            .collect::<HashMap<_, _>>();
144        let mut samples = Vec::new();
145        for segment in std::iter::once(first).chain(usable) {
146            positions[segment.id] = Some(samples.len());
147            if !self.gain_phase {
148                for row in segment.flags {
149                    if let Some(accepted) = accepted.get_mut(&row.antennas) {
150                        for (keep, unflagged) in accepted.iter_mut().zip(row.unflagged) {
151                            *keep &= unflagged;
152                        }
153                    }
154                }
155            }
156            let context = segment.context;
157            let time = legacy_times
158                .and_then(|clock| clock.time(context))
159                .unwrap_or(context.start + 0.5 * (context.end - context.start));
160            let angle = if self.circular {
161                0.0
162            } else {
163                angles
164                    .feed_parallactic_angle(time, context.field, 0, 0, context.spectral_window)
165                    .map_err(prior_error)? as f32
166            };
167            samples.push(Segment {
168                angle: f64::from(angle),
169                channels: vec![SegmentChannel::default(); channels],
170            });
171        }
172        let required = if self.circular || self.gain_phase {
173            2
174        } else {
175            3
176        };
177        if self.legacy && self.gain_phase && samples.len() < required {
178            return Ok((vec![None; channels], Complex32::ZERO));
179        }
180        if samples.len() < required {
181            if self.gain_phase {
182                return Err(GainSolveError::InsufficientGainPhaseSegments);
183            }
184            return Err(GainSolveError::InsufficientParallacticCoverage { required });
185        }
186        (0..channels).try_for_each(|channel| {
187            let word = channel / 64;
188            let bit = 1 << (channel % 64);
189            self.rows
190                .drain(channel, |row| {
191                    if let Some(segment) = positions[row.segment as usize]
192                        && (self.gain_phase
193                            || accepted
194                                .get(&row.antennas)
195                                .is_some_and(|words| words[word] & bit != 0))
196                    {
197                        samples[segment].channels[channel].observe(
198                            row.data,
199                            row.model,
200                            row.weights,
201                        );
202                    }
203                })
204                .map_err(GainSolveError::PolarizationScratch)
205        })?;
206        if self.legacy
207            && self.gain_phase
208            && !samples
209                .iter()
210                .flat_map(|segment| &segment.channels)
211                .any(SegmentChannel::has_weight)
212        {
213            return Ok((vec![None; channels], Complex32::ZERO));
214        }
215        solve(&samples, channels, self.circular, !self.gain_phase)
216            .map_err(|_| GainSolveError::SingularSourcePolarization)
217    }
218}

220#[derive(Debug, PartialEq)]
221struct PhaseSample {
222    segment: u32,
223    antennas: [i32; 2],
224    data: [Complex32; 2],
225    model: [Complex32; 2],
226    weights: [f32; 2],
227}
228
229impl ScratchSample for PhaseSample {
230    const BYTES: usize = 52;
231
232    fn write_to(&self, bytes: &mut [u8]) {
233        let words = [
234            self.segment,
235            self.antennas[0] as u32,
236            self.antennas[1] as u32,
237        ]
238        .into_iter()
239        .chain(
240            self.data
241                .into_iter()
242                .chain(self.model)
243                .flat_map(|value| [value.re.to_bits(), value.im.to_bits()]),
244        )
245        .chain(self.weights.map(f32::to_bits));
246        for (destination, source) in bytes.as_chunks_mut::<4>().0.iter_mut().zip(words) {
247            *destination = source.to_le_bytes();
248        }
249    }
250
251    fn from_bytes(bytes: &[u8]) -> Self {
252        let words = bytes.as_chunks::<4>().0;
253        let word = |index| u32::from_le_bytes(words[index]);
254        let float = |index| f32::from_bits(word(index));
255        let complex = |index| Complex32::new(float(index), float(index + 1));
256        Self {
257            segment: word(0),
258            antennas: [word(1) as i32, word(2) as i32],
259            data: [complex(3), complex(5)],
260            model: [complex(7), complex(9)],
261            weights: [float(11), float(12)],
262        }
263    }
264}

266#[cfg(test)]
267mod tests {
268    use super::*;
269
270    #[test]
271    fn phase_sample_scratch_format_keeps_segment_baseline_and_cross_hands() {
272        let sample = PhaseSample {
273            segment: 0x1234,
274            antennas: [1, 2],
275            data: [Complex32::new(1.5, -0.75), Complex32::new(-3.0, 4.0)],
276            model: [Complex32::new(2.75, -1.0), Complex32::new(1.125, 0.25)],
277            weights: [3.125, 0.0],
278        };
279        let mut bytes = [0; PhaseSample::BYTES];
280        sample.write_to(&mut bytes);
281        assert_eq!(
282            &bytes[..16],
283            &[0x34, 0x12, 0, 0, 1, 0, 0, 0, 2, 0, 0, 0, 0, 0, 192, 63]
284        );
285        assert_eq!(PhaseSample::from_bytes(&bytes), sample);
286    }
