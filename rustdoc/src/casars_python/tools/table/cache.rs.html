17/// Protect a plain table held by a live table tool, including an underlying
18/// table opened through a reference. A busy handle cannot be inspected.
19pub(crate) fn path_is_in_use(path: &Path) -> bool {
20    let Ok(path) = table_cache_name(path) else {
21        return true;
22    };
23    let mut used = false;
24    registry::for_each_other(u64::MAX, |state| {
25        used |= match state {
26            Some(state) => state.open.as_ref().is_some_and(|open| {
27                open.root()
28                    .table
29                    .cache_plain_table_paths()
30                    .map_or(true, |paths| {
31                        paths
32                            .into_iter()
33                            .any(|name| name.canonicalize().unwrap_or(name).starts_with(&path))
34                    })
35            }),
36            None => true,
37        };
38    });
39    used
40}

42fn add_open(open: &Open, names: &mut BTreeMap<PathBuf, bool>) -> Result<(), Fail> {
43    let root = open.root();
44    let locked = root.table.has_lock(LockType::Read) || root.table.has_lock(LockType::Write);
45    for name in root.table.cache_plain_table_paths()? {
46        names
47            .entry(name)
48            .and_modify(|held| *held |= locked)
49            .or_insert(locked);
50    }
51    Ok(())
52}

54impl State {
55    fn cached_names(&self) -> Result<BTreeMap<PathBuf, bool>, Fail> {
56        let mut names = registry::attached_table_paths()
57            .into_iter()
58            .map(|name| (name, false))
59            .collect();
60        if let Some(open) = &self.open {
61            add_open(open, &mut names)?;
62        }
63        let mut failure = None;
64        registry::for_each_other(self.id, |other| {
65            if failure.is_some() {
66                return;
67            }
68            match other {
69                Some(other) => {
70                    if let Some(open) = &other.open
71                        && let Err(error) = add_open(open, &mut names)
72                    {
73                        failure = Some(error);
74                    }
75                }
76                None => {
77                    failure = Some(Fail::Runtime(
78                        "Cannot inspect the table cache while another tool is busy".into(),
79                    ))
80                }
81            }
82        });
83        match failure {
84            Some(error) => Err(error),
85            None => Ok(names),
86        }
87    }
88
89    fn showcache(&self, verbose: bool, log: &mut Vec<LogLine>) -> Result<CasaValue, Fail> {
90        if self.open.is_none() {
91            no_table(log);
92        }
93        let names = self.cached_names()?;
94        if verbose {
95            if names.is_empty() {
96                log.push(LogLine::info("The Table Cache is empty."));
97            } else {
98                log.push(LogLine::info(format!(
99                    "The Table Cache has the following {} entries:",
100                    names.len()
101                )));
102                for (index, name) in names.keys().enumerate() {
103                    log.push(LogLine::info(format!(
104                        "    {index}: \"{}\"",
105                        name.display()
106                    )));
107                }
108            }
109        }
110        Ok(CasaValue::List(
111            names
112                .into_keys()
113                .map(|name| CasaValue::Text(name.to_string_lossy().into_owned()))
114                .collect(),
115        ))
116    }
117}

119#[pymethods]
120impl PyTableTool {
121    /// Whether the process table cache contains `tablename`.
122    #[pyo3(signature = (tablename), text_signature = "($self, tablename)")]
123    fn isopened(&self, py: Python<'_>, tablename: &Bound<'_, PyAny>) -> PyResult<Py<PyAny>> {
124        let name = text(tablename, "tablename")?;
125        let path = py.import("os")?.getattr("path")?;
126        let expanded = path.call_method1("expandvars", (name,))?;
127        let expanded: String = path.call_method1("expanduser", (expanded,))?.extract()?;
128        let name = PathBuf::from(expanded);
129        self.call(py, move |state, log| {
130            if state.open.is_none() {
131                no_table(log);
132            }
133            let name = table_cache_name(&name)?;
134            Ok(CasaValue::Bool(state.cached_names()?.contains_key(&name)))
135        })
136    }
137
138    /// List the plain tables held in the process table cache.
139    #[pyo3(signature = (verbose = None), text_signature = "($self, verbose=True)")]
140    fn showcache(&self, py: Python<'_>, verbose: Option<&Bound<'_, PyAny>>) -> PyResult<Py<PyAny>> {
141        let verbose = flag(verbose, true)?;
142        self.call(py, move |state, log| state.showcache(verbose, log))
143    }
144
145    /// Print the process's locked tables.
146    #[pyo3(signature = (), text_signature = "($self)")]
147    fn listlocks(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
148        self.call(py, |state, _| {
149            let names = state.cached_names()?;
150            let stdout = std::io::stdout();
151            let mut output = stdout.lock();
152            writeln!(output, "Locked tables: ")
153                .map_err(|error| Fail::Runtime(error.to_string()))?;
154            for (name, locked) in names {
155                if locked {
156                    writeln!(output, "   {}", name.display())
157                        .map_err(|error| Fail::Runtime(error.to_string()))?;
158                }
159            }
160            writeln!(output, "--").map_err(|error| Fail::Runtime(error.to_string()))?;
161            Ok(CasaValue::Bool(true))
162        })
163    }
164}
