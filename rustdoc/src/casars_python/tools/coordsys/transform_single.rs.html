30pub(super) fn numeric_record_value(value: &Value) -> PyResult<Vec<f64>> {
31    match value {
32        Value::Scalar(ScalarValue::Float64(value)) => Ok(vec![*value]),
33        Value::Scalar(ScalarValue::Float32(value)) => Ok(vec![f64::from(*value)]),
34        Value::Array(ArrayValue::Float64(values))
35            if values.shape().iter().filter(|&&size| size != 1).count() <= 1 =>
36        {
37            Ok(values.iter().copied().collect())
38        }
39        Value::Array(ArrayValue::Float32(values))
40            if values.shape().iter().filter(|&&size| size != 1).count() <= 1 =>
41        {
42            Ok(values.iter().copied().map(f64::from).collect())
43        }
44        Value::Array(ArrayValue::Float32(_) | ArrayValue::Float64(_)) => {
45            Err(raised("Coordinate numeric record must be one-dimensional"))
46        }
47        _ => Err(raised(
48            "RecordRep::get_pointer - incorrect data type for field numeric",
49        )),
50    }
51}

53pub(super) fn bool_vector(value: &Bound<'_, PyAny>) -> PyResult<Option<bool>> {
54    if value.is_instance_of::<PyList>() || value.is_instance_of::<PyTuple>() {
55        let mut empty = true;
56        for item in value.try_iter()? {
57            if !item?.is_instance_of::<PyBool>() {
58                return Ok(None);
59            }
60            empty = false;
61        }
62        return Ok(Some(empty));
63    }
64    if let Ok(array) = value.cast::<PyUntypedArray>()
65        && !matches!(
66            array.dtype().char(),
67            b'b' | b'B'
68                | b'h'
69                | b'H'
70                | b'i'
71                | b'I'
72                | b'l'
73                | b'L'
74                | b'q'
75                | b'Q'
76                | b'f'
77                | b'd'
78                | b'g'
79                | b'F'
80                | b'D'
81                | b'G'
82                | b'S'
83                | b'U'
84        )
85    {
86        // Unsupported NumPy dtypes also reach CASA as the unset bool variant.
87        return Ok(Some(array.dtype().char() != b'?' || array.is_empty()));
88    }
89    Ok(None)
90}

104fn pixel_argument(
105    py: Python<'_>,
106    value: Option<&Bound<'_, PyAny>>,
107    record_error: &str,
108    type_error: &str,
109) -> PyResult<Vec<f64>> {
110    let Some(value) = value.filter(|value| !value.is_none()) else {
111        return Ok(Vec::new());
112    };
113    if let Ok(dict) = value.cast::<PyDict>() {
114        let record = casa_record(py, Some(dict))?;
115        return numeric_record_value(record.get("numeric").ok_or_else(|| raised(record_error))?);
116    }
117    if let Some(empty) = bool_vector(value)? {
118        return if empty {
119            Ok(Vec::new())
120        } else {
121            Err(raised(type_error))
122        };
123    }
124    if let Ok(array) = value.cast::<PyUntypedArray>()
125        && array.ndim() == 0
126        && matches!(array.dtype().kind(), b'i' | b'u' | b'f')
127    {
128        return Ok(vec![value.call_method0("item")?.extract()?]);
129    }
130    match value.extract::<Variant>()? {
131        Variant::Numbers(values) | Variant::Array { values, .. } => Ok(values),
132        Variant::Empty => Ok(Vec::new()),
133        _ => Err(raised(type_error)),
134    }
135}

137pub(in crate::tools) enum WorldArgument {
138    Unset,
139    Numeric(Vec<f64>),
140    Text(String),
141    Texts(Vec<String>),
142    Record(RecordValue),
143}
144
145impl WorldArgument {
146    pub(in crate::tools) fn capture(
147        py: Python<'_>,
148        value: Option<&Bound<'_, PyAny>>,
149    ) -> PyResult<Self> {
150        let Some(value) = value.filter(|value| !value.is_none()) else {
151            return Ok(Self::Unset);
152        };
153        if let Ok(dict) = value.cast::<PyDict>() {
154            return Ok(Self::Record(casa_record(py, Some(dict))?));
155        }
156        if bool_vector(value)?.is_some() {
157            return Ok(Self::Unset);
158        }
159        if let Ok(array) = value.cast::<PyUntypedArray>()
160            && matches!(array.dtype().char(), b'S' | b'U')
161        {
162            let flattened = value.call_method1("ravel", ("F",))?;
163            let texts = flattened
164                .try_iter()?
165                .map(|item| match item?.extract::<Variant>()? {
166                    Variant::Text(text) => Ok(text),
167                    _ => Err(raised("Unrecognized format for world coordinate")),
168                })
169                .collect::<PyResult<_>>()?;
170            return Ok(Self::Texts(texts));
171        }
172        match value.extract::<Variant>()? {
173            Variant::Number(number) if !value.is_instance_of::<PyInt>() => {
174                Ok(Self::Numeric(vec![number]))
175            }
176            Variant::Numbers(values) | Variant::Array { values, .. } => Ok(Self::Numeric(values)),
177            Variant::Empty => Ok(Self::Numeric(Vec::new())),
178            Variant::Texts(text) => Ok(Self::Texts(text)),
179            Variant::Text(text) => Ok(Self::Text(text)),
180            _ => Err(raised("Unrecognized format for world coordinate")),
181        }
182    }
183
184    pub(super) fn values(self, state: &CoordinateState) -> PyResult<Vec<f64>> {
185        self.values_with_defaults(state, None, world_reference(&state.coordinates))
186    }
187
188    pub(super) fn values_with_defaults(
189        self,
190        state: &CoordinateState,
191        selected: Option<(usize, &CoordinateState)>,
192        mut world: Vec<f64>,
193    ) -> PyResult<Vec<f64>> {
194        let coordinates = selected.map_or(&state.coordinates, |(_, part)| &part.coordinates);
195        let input = WorldInput {
196            state,
197            coordinates,
198            selected: selected.map(|(index, _)| index),
199            units: native_world_units(coordinates),
200        };
201        let decoded = match self {
202            Self::Unset => input.reference_prefix()?,
203            Self::Numeric(values) => input.numeric(values)?,
204            Self::Text(text) => {
205                let parts: Vec<_> = text.split_whitespace().take(50).collect();
206                if parts.len() == 1 {
207                    vec![read_quantity(
208                        parts[0],
209                        input.units.first().map_or("", String::as_str),
210                    )?]
211                } else {
212                    input.texts(parts.into_iter())?
213                }
214            }
215            Self::Texts(texts) => input.texts(texts.iter().map(String::as_str))?,
216            Self::Record(record) => world_record_values(&record, &input)?,
217        };
218        overwrite_prefix(&mut world, decoded);
219        Ok(world)
220    }
221}

223struct WorldInput<'a> {
224    state: &'a CoordinateState,
225    coordinates: &'a CoordinateSystem,
226    selected: Option<usize>,
227    units: Vec<String>,
228}
229
230impl WorldInput<'_> {
231    fn reference_prefix(&self) -> PyResult<Vec<f64>> {
232        if self.selected.is_none() {
233            return Ok(world_reference(&self.state.coordinates));
234        }
235        // CASA uses the full declared reference vector as a prefix for a
236        // selected coordinate's string vector or unset argument.
237        let declared = axis_values(
238            &self.state.coordinates,
239            true,
240            |coordinate| match coordinate {
241                CoordinateModel::Direction(direction) => {
242                    direction.reference_and_increment_in_world_units().0
243                }
244                _ => coordinate.reference_value(),
245            },
246        );
247        declared_to_native(declared, self.coordinates)
248    }
249
250    fn texts<'a>(&self, texts: impl Iterator<Item = &'a str>) -> PyResult<Vec<f64>> {
251        let mut values = self.reference_prefix()?;
252        let supplied = text_values(&self.state.coordinates, &self.units, texts)?;
253        overwrite_prefix(&mut values, supplied);
254        Ok(values)
255    }
256
257    fn numeric(&self, mut values: Vec<f64>) -> PyResult<Vec<f64>> {
258        if let Some(index) = self.selected {
259            let axes = self
260                .state
261                .coordinates
262                .coordinate_world_axes(index)
263                .expect("selected coordinate");
264            let visible = axes.iter().flatten().count();
265            // A typed edit can supply a full local coordinate or just its
266            // visible slots. Hidden components are never written.
267            if visible < axes.len() && values.len() == visible {
268                let mut compact = values.into_iter();
269                values = axes
270                    .into_iter()
271                    .map(|axis| {
272                        axis.map_or(0.0, |_| compact.next().expect("validated visible count"))
273                    })
274                    .collect();
275            }
276        }
277        declared_to_native(values, self.coordinates)
278    }
279
280    fn measures(&self, record: &RecordValue) -> PyResult<Vec<f64>> {
281        let system = &self.state.coordinates;
282        let mut values = world_reference(system);
283        update_measures(record, self.state, &mut values)?;
284        if let Some(index) = self.selected {
285            let model = system.coordinate(index);
286            let reference = model.reference_value();
287            return Ok(system
288                .coordinate_world_axes(index)
289                .expect("selected coordinate")
290                .into_iter()
291                .zip(reference)
292                .map(|(axis, reference)| axis.map_or(reference, |axis| values[axis]))
293                .collect());
294        }
295        Ok(values)
296    }
297}

312fn declared_to_native(values: Vec<f64>, coordinates: &CoordinateSystem) -> PyResult<Vec<f64>> {
313    let conversions = axis_values(coordinates, true, CoordinateModel::declared_axis_units)
314        .into_iter()
315        .zip(coordinates.world_axis_units())
316        .map(|(declared, internal)| (declared != internal).then_some(declared));
317    values
318        .into_iter()
319        .zip(conversions)
320        .map(|(value, unit)| match unit {
321            Some(unit) => Quantity::new(value, &unit)
322                .map(|quantity| quantity.get_si_value())
323                .map_err(raised),
324            None => Ok(value),
325        })
326        .collect()
327}

329fn read_quantity(text: &str, unit: &str) -> PyResult<f64> {
330    let quantity = Quantity::read(text).map_err(raised)?;
331    Ok(get_value(&quantity, &Unit::new(unit).map_err(raised)?))

334fn text_values<'a>(
335    coordinates: &CoordinateSystem,
336    units: &[String],
337    values: impl Iterator<Item = &'a str>,
338) -> PyResult<Vec<f64>> {
339    let mut types = axis_values(coordinates, true, |coordinate| {
340        vec![coordinate.coordinate_type(); coordinate.n_world_axes()]
341    })
342    .into_iter()
343    .zip(units);
344    values
345        .map(|text| {
346            let (kind, unit) = types.next().ok_or_else(|| {
347                raised("Failed AlwaysAssert axisInCoordinateSystem < nWorldAxes()")
348            })?;
349            if kind == CoordinateType::Stokes {
350                return Ok(StokesType::from_name(text)
351                    .unwrap_or(StokesType::Undefined)
352                    .code() as f64);
353            }
354            if kind == CoordinateType::Quality {
355                return Quantity::read(text)
356                    .map(|value| value.value())
357                    .map_err(|_| {
358                        raised(format!(
359                            "Failed to convert string formatted world [{text}] to double"
360                        ))
361                    });
362            }
363            if kind == CoordinateType::Direction {
364                let invalid = || {
365                    raised(format!(
366                        "Failed to convert string formatted world direction {text} to double"
367                    ))
368                };
369                let value = Quantity::read(text).map_err(|_| invalid())?;
370                if value.unit().name().is_empty() {
371                    return Ok(value.value());
372                }
373                let radians = mvangle_radians(&value).map_err(|_| invalid())?;
374                let angle = Quantity::new(radians, "rad").map_err(raised)?;
375                return Ok(get_value(&angle, &Unit::new(unit).map_err(raised)?));
376            }
377            read_quantity(text, unit)
378        })
379        .collect()
380}

396fn quantity_record(record: &RecordValue, target_unit: &str) -> PyResult<f64> {
397    let malformed = || raised("Illegal Quantum record in QuantumHolder::fromRecord");
398    let Some(Value::Scalar(ScalarValue::String(unit))) = record.get("unit") else {
399        return Err(malformed());
400    };
401    let value = match record.get("value") {
402        Some(Value::Scalar(ScalarValue::Float64(value))) => *value,
403        Some(Value::Scalar(ScalarValue::Float32(value))) => f64::from(*value),
404        Some(Value::Scalar(ScalarValue::Int32(value))) => f64::from(*value),
405        Some(Value::Scalar(ScalarValue::Int64(value))) => *value as f64,
406        Some(value @ Value::Array(array))
407            if !array.shape().is_empty() && array.shape().iter().all(|&length| length == 1) =>
408        {
409            if let CasaValue::Floats { values, .. } = CasaValue::from_value(value) {
410                values[0]
411            } else {
412                return Err(raised("Wrong QuantumHolder to convert asQuantumDouble"));
413            }
414        }
415        Some(Value::Array(_)) => {
416            return Err(raised("Wrong QuantumHolder to convert asQuantumDouble"));
417        }
418        _ => return Err(malformed()),
419    };
420    let quantity = Quantity::new(value, unit).map_err(raised)?;
421    Ok(get_value(
422        &quantity,
423        &Unit::new(target_unit).map_err(raised)?,
424    ))
425}

427fn quantity_vector(record: &RecordValue, units: &[String]) -> PyResult<Vec<f64>> {
428    if record.fields().len() != units.len() {
429        return Err(raised(
430            "Number of axes must equal number of fields in record!",
431        ));
432    }
433    let mut fields: Vec<_> = record.fields().iter().collect();
434    fields.sort_by(|left, right| left.name.cmp(&right.name));
435    fields
436        .into_iter()
437        .zip(units)
438        .map(|(field, unit)| quantity_record(subrecord(&field.value, "quantity")?, unit))
439        .collect()
440}

442fn world_record_values(record: &RecordValue, input: &WorldInput<'_>) -> PyResult<Vec<f64>> {
443    if let Some(kind) = record.get("pw_type") {
444        scalar_text(kind, "pw_type")?;
445    }
446    if matches!(
447        record.get("ar_type"),
448        Some(Value::Scalar(ScalarValue::String(value))) if value == "relative"
449    ) {
450        return Err(raised(
451            "The value is relative, not absolute', origin='coordsys.checkAbsRel",
452        ));
453    }
454    if let Some(value) = record.get("numeric") {
455        return input.numeric(numeric_record_value(value)?);
456    }
457    if let Some(value) = record.get("quantity") {
458        return quantity_vector(subrecord(value, "quantity")?, &input.units);
459    }
460    if let Some(value) = record.get("measure") {
461        return input.measures(subrecord(value, "measure")?);
462    }
463    if let Some(value) = record.get("string") {
464        return match value {
465            Value::Array(ArrayValue::String(values)) if values.ndim() == 1 => {
466                input.texts(values.iter().map(String::as_str))
467            }
468            Value::Scalar(ScalarValue::String(value)) => {
469                input.texts(std::iter::once(value.as_str()))
470            }
471            _ => Err(raised("Coordinate string record must be one-dimensional")),
472        };
473    }
474    input.measures(record)
475}

477fn update_measures(
478    record: &RecordValue,
479    state: &CoordinateState,
480    world: &mut [f64],
481) -> PyResult<()> {
482    let placements = state
483        .coordinates
484        .coordinates_in_record(&state.record)
485        .map_err(raised)?;
486    let mut seen = Vec::new();
487    for (coordinate, placement) in models(&state.coordinates).zip(placements) {
488        let kind = coordinate.coordinate_type();
489        if seen.contains(&kind) {
490            continue;
491        }
492        seen.push(kind);
493        let key = kind.to_string().to_ascii_lowercase();
494        let Some(value) = record.get(&key) else {
495            continue;
496        };
497        let values = match coordinate {
498            CoordinateModel::Direction(_) => {
499                let angle =
500                    direction_from_record(subrecord(value, "direction")?).map_err(raised)?;
501                vec![angle.longitude_rad(), angle.latitude_rad()]
502            }
503            CoordinateModel::Spectral(spectral) => {
504                let measure = subrecord(value, "spectral")?
505                    .get("frequency")
506                    .ok_or_else(|| {
507                        raised("This spectral record does not contain a frequency field")
508                    })?;
509                let frequency =
510                    frequency_from_record(subrecord(measure, "frequency")?).map_err(raised)?;
511                vec![spectral.value_from_hz(frequency.hz()).map_err(raised)?]
512            }
513            CoordinateModel::Stokes(stokes) => {
514                let name = scalar_text(value, "stokes")?;
515                let polarization = StokesType::from_name(name).unwrap_or(StokesType::Undefined);
516                if !stokes.stokes().contains(&polarization) {
517                    return Err(raised(format!("Illegal Stokes coordinate {name}")));
518                }
519                vec![f64::from(polarization.code())]
520            }
521            CoordinateModel::Linear(_) => {
522                let units = if placement.world_axes.len() == coordinate.n_world_axes() {
523                    coordinate.axis_units()
524                } else {
525                    placement.model(&state.record).map_err(raised)?.axis_units()
526                };
527                quantity_vector(subrecord(value, "linear")?, &units)?
528            }
529            CoordinateModel::Quality(_) | CoordinateModel::System(_) => continue,
530            CoordinateModel::Tabular(_) => {
531                vec![quantity_record(
532                    subrecord(value, "tabular")?,
533                    &coordinate.axis_units()[0],
534                )?]
535            }
536        };
537        for (axis, value) in placement.world_axes.into_iter().zip(values) {
538            if let Some(axis) = axis {
539                world[axis] = value;
540            }
541        }
542    }
543    Ok(())
544}

546fn overwrite_prefix<T>(target: &mut [T], values: impl IntoIterator<Item = T>) {
547    for (slot, value) in target.iter_mut().zip(values) {
548        *slot = value;
549    }
550}

552impl PyCoordsys {
553    pub(in crate::tools) fn image_world(
554        &self,
555        pixel: Vec<f64>,
556        format: &str,
557        velocities: bool,
558    ) -> PyResult<CasaValue> {
559        image_world_record(&self.state(), pixel, format, velocities, false)
560    }
561
562    pub(in crate::tools) fn image_measures(
563        &self,
564        pixel: Vec<f64>,
565        direction_frame: &str,
566        spectral_frame: &str,
567    ) -> PyResult<CasaValue> {
568        let state = self.state();
569        if direction_frame.eq_ignore_ascii_case("cl") && spectral_frame.eq_ignore_ascii_case("cl") {
570            return image_world_record(&state, pixel, "m", true, false);
571        }
572        let mut state = state.clone();
573        select_measure_frames(&mut state.coordinates, direction_frame, spectral_frame)?;
574        image_world_record(&state, pixel, "m", true, false)
575    }
576
577    pub(in crate::tools) fn image_pixel(&self, value: WorldArgument) -> PyResult<CasaValue> {
578        let state = self.state();
579        let world = value.values(&state)?;
580        let pixel = state.coordinates.to_pixel_or_nan(&world).map_err(raised)?;
581        Ok(CasaValue::record([
582            ("numeric", CasaValue::floats(pixel)),
583            ("pw_type", CasaValue::Text("pixel".into())),
584            ("ar_type", CasaValue::Text("absolute".into())),
585        ]))
586    }
587}

615fn image_world_record(
616    state: &CoordinateState,
617    pixel: Vec<f64>,
618    format: &str,
619    velocities: bool,
620    coordinate_tags: bool,
621) -> PyResult<CasaValue> {
622    let mut complete = axis_values(&state.coordinates, false, CoordinateModel::reference_pixel);
623    overwrite_prefix(&mut complete, pixel);
624    let world = state.coordinates.to_world(&complete).map_err(|error| {
625        raised(format!(
626            "Exception: Error converting to world coordinates: {error}"
627        ))
628    })?;
629    format_transformed_world(state, &world, format, velocities, coordinate_tags)
630}

632fn select_measure_frames(
633    coordinates: &mut CoordinateSystem,
634    direction_frame: &str,
635    spectral_frame: &str,
636) -> PyResult<()> {
637    if !direction_frame.eq_ignore_ascii_case("cl")
638        && let Some(index) = coordinates.find_coordinate(CoordinateType::Direction)
639        && let CoordinateModel::Direction(direction) = coordinates.coordinate(index)
640    {
641        let frame = if direction_frame.eq_ignore_ascii_case("native") {
642            direction.direction_ref()
643        } else {
644            casa_direction_frame_name(direction_frame)
645                .and_then(|name| name.parse::<DirectionRef>().ok())
646                .ok_or_else(|| {
647                    raised(format!(
648                        "Exception: Unknown direction reference frame {direction_frame}."
649                    ))
650                })?
651        };
652        coordinates.replace_coordinate(
653            index,
654            direction.clone().with_conversion_direction_ref(frame),
655        );
656    }
657    if !spectral_frame.eq_ignore_ascii_case("cl")
658        && let Some(index) = coordinates.find_coordinate(CoordinateType::Spectral)
659        && let CoordinateModel::Spectral(spectral) = coordinates.coordinate(index)
660    {
661        let replacement = if spectral_frame.eq_ignore_ascii_case("native") {
662            spectral.clone().without_conversion()
663        } else {
664            let reference = spectral_frame.parse::<CasaFrequencyFrame>().map_err(|_| {
665                raised(format!(
666                    "Exception: Unknown frequency reference frame {spectral_frame}."
667                ))
668            })?;
669            let frame = spectral.conversion_frame().cloned().unwrap_or_else(|| {
670                let mut frame = MeasFrame::new().with_direction(MDirection::from_angles(
671                    0.,
672                    std::f64::consts::FRAC_PI_2,
673                    DirectionRef::J2000,
674                ));
675                if let Some(epoch) = &coordinates.obs_info().date {
676                    frame = frame.with_epoch(epoch.clone());
677                }
678                if let Some(position) = &coordinates.obs_info().telescope_position {
679                    frame = frame.with_position(position.clone());
680                }
681                frame
682            });
683            let CasaFrequencyFrame::Defined(reference) = reference else {
684                return Err(raised(
685                    "Transformations to/from frame \"Undefined\" are not possible.",
686                ));
687            };
688            spectral.clone().with_conversion(reference, frame)
689        };
690        coordinates.replace_coordinate(index, replacement);
691    }
692    Ok(())
693}
