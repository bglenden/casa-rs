2//! The major-cycle pass (plan section 5.4): one traversal of the selected
3//! visibilities that accumulates the dirty or residual image terms, and the
4//! PSF on the initial pass, of every image domain, for every capability.
5//!
6//! A pass reads blocks of native rows, projected on every image domain, from
7//! a [`BoundedSource`] through a two-slot stream, places each row with each
8//! domain's spectral resampler and the imaging weights on the CPU worker
9//! team, routes the placements to the owners of the domain's [`Partition`]
10//! and accumulates them with each owner's backend ([`BackendChoice`]): the
11//! residual `V − Σ_d A_d·m_d` of every
12//! domain's model when a model is present (CASA `SIMapperCollection::degrid`
13//! sums every mapper's prediction before `grid` forms the residual), `V`
14//! otherwise. Each wave of planes ([`Residency`]) ends with the per-plane
15//! transforms, single-threaded per worker, and hands each domain's
16//! [`NormalImages`] to the caller before the next wave starts.

62/// A failure of a major-cycle or density pass.
63#[derive(Debug, thiserror::Error)]
64pub enum PassError {
65    /// A worker team was asked for no workers.
66    #[error("a worker team needs at least one worker")]
67    Workers,
68    /// A pass was given no image domain.
69    #[error("a pass needs at least one image domain")]
70    NoDomains,
71    /// The image domains of a pass do not share one plane axis.
72    #[error("the image domains of a pass must have the same planes")]
73    PlaneAxes,
74    /// A row's native channels lie further apart than the spacing that
75    /// sized the waves' model halo.
76    #[error("native channels {observed_hz} Hz apart exceed the planned spacing {bound_hz} Hz")]
77    NativeSpacing {
78        /// The widest spacing in the row.
79        observed_hz: f64,
80        /// The pass's `native_spacing_hz`.
81        bound_hz: f64,
82    },
83    /// The measurement operator rejected a row, model or accumulator.
84    #[error("measurement operator: {0}")]
85    Operator(#[from] OperatorError),
86    /// The visibility source failed.
87    #[error("visibility source: {0}")]
88    Source(#[source] SourceError),
89    /// The model grids of a wave could not be prepared.
90    #[error("model preparation: {0}")]
91    Model(#[source] SourceError),
92    /// The consumer of a wave's images failed.
93    #[error("wave images: {0}")]
94    Images(#[source] SourceError),
95    /// The visibility writer failed.
96    #[error("visibility write: {0}")]
97    VisibilityWrite(#[source] SourceError),
98    /// A writing pass must hold every plane at once: a native sample's
99    /// prediction can draw on any output channel, and each row is written
100    /// once.
101    #[error("writing visibilities needs every plane resident")]
102    VisibilityWriteWaves,
103    /// The pass was cancelled at a block boundary.
104    #[error("the pass was cancelled")]
105    Cancelled,
106    /// The source thread panicked or could not start.
107    #[error("the visibility source thread failed")]
108    ProducerPanicked,
109}

111/// Cooperative cancellation shared between the caller and running passes;
112/// a pass stops at the next block boundary.
113#[derive(Clone, Debug, Default)]
114pub struct Cancel(Arc<AtomicBool>);
115
116impl Cancel {
117    /// A token that is not cancelled.
118    #[must_use]
119    pub fn new() -> Self {
120        Self::default()
121    }
122
123    /// Request cancellation.
124    pub fn cancel(&self) {
125        self.0.store(true, Ordering::Release);
126    }
127
128    /// Whether cancellation was requested.
129    #[must_use]
130    pub fn is_cancelled(&self) -> bool {
131        self.0.load(Ordering::Acquire)
132    }
133}

135/// A re-traversable source of native rows.
136///
137/// Each wave of a pass begins one traversal. Rows are delivered in a fixed
138/// order, so repeated traversals see the same samples.
139pub trait BoundedSource: Send {
140    /// Start a traversal for a wave over `planes`. With `restrict`, the
141    /// source may deliver only the native channels whose samples reach
142    /// `planes`, with both interpolation partners of each; without it, it
143    /// delivers every selected channel of each row.
144    fn begin(&mut self, planes: PlaneRange, restrict: bool) -> Result<(), SourceError>;
145
146    /// Fill `block` with the next rows of the traversal; `Ok(false)` once
147    /// the traversal is exhausted.
148    fn fill(&mut self, block: &mut NativeBlock) -> Result<bool, SourceError>;
149
150    /// The largest block any traversal fills, known before one begins: a
151    /// pass allocates its per-block storage for it and admits that storage
152    /// first ([`WaveDemand::bytes`], [`density_pass_bytes`]).
153    fn maximum_block(&self) -> BlockShape;
154}

168/// The gridding backend every owner of a pass dispatches to.
169///
170/// With [`BackendChoice::Metal`] each owner grids on the shared Metal
171/// device into an `f32` accumulator in device memory (D2), so every
172/// operator of the pass must be `f32`; placement and the native-channel
173/// predictions of a multi-domain or linearly interpolated residual stay on
174/// the CPU workers.
175#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
176#[serde(rename_all = "lowercase")]
177pub enum BackendChoice {
178    /// [`casa_imaging_operator::CpuBackend`].
179    #[default]
180    Cpu,
181    /// [`casa_imaging_metal::MetalBackend`].
182    Metal,
183}

185/// One major-cycle pass over the selected visibilities.
186pub struct MajorCyclePass<'a> {
187    /// The image domains; rows arrive projected on each, in this order.
188    pub domains: &'a [PassDomain<'a>],
189    /// The run's imaging weights.
190    pub weighting: &'a WeightingGeneration,
191    /// Modes accumulated: data and PSF on the initial pass, data later.
192    pub modes: ModeSet,
193    /// Model grids of a domain's planes; `None` grids the data themselves
194    /// (an initial pass without a start model).
195    pub model: Option<&'a ModelPreparation<'a>>,
196    /// Planes held at once.
197    pub residency: Residency,
198    /// The widest spacing between adjacent selected native channels; with
199    /// a model it sizes each wave's model halo
200    /// ([`SpectralResampler::model_planes`]).
201    pub native_spacing_hz: f64,
202    /// Where the owners grid.
203    pub backend: BackendChoice,
204}

206/// Whether a pass over `domains` forms its residual at native channels: with
207/// a model, when several domains' predictions must be summed before
208/// subtracting, or when the resampler interpolates onto the output channels
209/// ([`SpectralResampler::forms_native_residuals`]).
210fn native_residuals(domains: &[PassDomain<'_>], with_model: bool) -> bool {
211    with_model
212        && (domains.len() > 1
213            || domains
214                .iter()
215                .any(|domain| domain.resampler.forms_native_residuals()))
216}
217
218impl MajorCyclePass<'_> {
219    fn native_residuals(&self) -> bool {
220        native_residuals(self.domains, self.model.is_some())
221    }
222
223    /// Whether a wave must read whole rows: a native-channel prediction under
224    /// linear interpolation depends on the whole row's channel map
225    /// ([`SpectralResampler::predict_row`]), not only the channels that
226    /// reach the wave.
227    fn whole_rows(&self) -> bool {
228        self.model.is_some()
229            && self
230                .domains
231                .iter()
232                .any(|domain| domain.resampler.forms_native_residuals())
233    }
234}

236/// What a pass traversed.
237#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
238pub struct PassSummary {
239    /// Samples placed on the first image domain, over every wave: the same
240    /// count for every pass over the same selection and weights.
241    pub samples: u64,
242    /// Source blocks consumed, over every wave.
243    pub blocks: u64,
244}

246/// Run `pass`, handing each wave's normal images of each domain to
247/// `images` (domain, images) in plane order and, on a final pass that
248/// writes visibilities, every block to `visibilities`.
249pub fn run_major_cycle(
250    pass: &MajorCyclePass<'_>,
251    source: &mut dyn BoundedSource,
252    team: &WorkerTeam,
253    cancel: &Cancel,
254    images: &mut dyn FnMut(usize, NormalImages) -> Result<(), PassError>,
255    mut visibilities: Option<&mut VisibilitySink<'_>>,
256) -> Result<PassSummary, PassError> {
257    if visibilities.is_some() && pass.residency != Residency::All {
258        return Err(PassError::VisibilityWriteWaves);
259    }
260    let Some(main) = pass.domains.first() else {
261        return Err(PassError::NoDomains);
262    };
263    let total = main.operator.basis().planes();
264    if pass
265        .domains
266        .iter()
267        .any(|domain| domain.operator.basis().planes() != total)
268    {
269        return Err(PassError::PlaneAxes);
270    }
271    let restrict = !pass.whole_rows();
272    let native_residuals = pass.native_residuals();
273    let mut summary = PassSummary::default();
274    for planes in pass.residency.waves(total) {
275        source.begin(planes, restrict).map_err(PassError::Source)?;
276        let models = pass
277            .model
278            .map(|prepare| {
279                pass.domains
280                    .iter()
281                    .enumerate()
282                    .map(|(index, domain)| {
283                        prepare(
284                            index,
285                            domain
286                                .resampler
287                                .model_planes(planes, pass.native_spacing_hz),
288                        )
289                    })
290                    .collect::<Result<Vec<_>, _>>()
291            })
292            .transpose()?;
293        let mut wave = Wave::new(
294            pass,
295            planes,
296            models.as_deref(),
297            native_residuals,
298            source.maximum_block(),
299            team.workers(),
300        )?
301        .checking_spacing(!restrict && pass.residency != Residency::All);
302        summary.blocks += stream_blocks(source, cancel, |block| {
303            wave.consume(block, team, visibilities.as_deref_mut())
304        })?;
305        summary.samples += wave.samples();
306        for (domain, domain_images) in wave.finish(team)?.into_iter().enumerate() {
307            images(domain, domain_images)?;
308        }
309    }
310    Ok(summary)
311}

313/// Bytes the passes hold for a source whose largest block is `block`: the
314/// native blocks a pass's stream double-buffers and the model visibilities
315/// of one block, which a pass with a model predicts. They grow with the
316/// source's block, so the source's admission holds them
317/// ([`crate::finalize_source_access`]).
318#[must_use]
319pub const fn source_block_bytes(block: BlockShape) -> u64 {
320    stream::bytes(block) + wave::prediction_bytes(block)
321}
322
323/// Row chunks per worker of a density pass.
324const DENSITY_CHUNKS_PER_WORKER: usize = 4;
325
326/// Placements a row chunk takes at once. A pass places each block in slices
327/// of chunks this size, so its sample buffers have one bound whatever rows
328/// the source puts in a block, while a slice still spans most of a large
329/// block and its workers meet once per slice.
330const CHUNK_PLACEMENTS: usize = 1 << 15;
331
332/// Placements a row chunk of a major-cycle pass takes at once when each
333/// image domain has at most `owners` owners. Each owner's buffer in a
334/// chunk can take every placement of the chunk, and a pass has four chunks
335/// per worker, so past four owners a chunk takes proportionally fewer: the
336/// chunks' storage then grows linearly with the workers rather than as
337/// their square, and a slice places as many samples whatever the workers.
338const fn chunk_placements(owners: usize) -> usize {
339    if owners <= 4 {
340        CHUNK_PLACEMENTS
341    } else {
342        4 * CHUNK_PLACEMENTS / owners
343    }
344}
345
346/// Rows a row chunk places at once, for rows of `row` placements each in
347/// blocks of at most `block_rows` rows split among `chunks` chunks taking
348/// `placements` placements each: as many as fit, at least one, and no more
349/// than a block's share.
350const fn chunk_rows(row: usize, block_rows: usize, chunks: usize, placements: usize) -> usize {
351    let fitting = placements / if row == 0 { 1 } else { row };
352    let share = block_rows.div_ceil(if chunks == 0 { 1 } else { chunks });
353    let rows = if fitting < share { fitting } else { share };
354    if rows == 0 { 1 } else { rows }
355}
356
357/// An upper bound of the bytes a pass's row chunks hold per row of the
358/// source's block, for rows of `layout` that each place at most `row`
359/// samples ([`SpectralResampler::samples_per_row`]) on `workers` workers,
360/// while a chunk's rows are a block's share rather than its placement cap:
361/// each owner's buffer and the weight owner's can take every placement of
362/// a row. The source sizes its blocks with it
363/// ([`crate::finalize_source_access`]); [`WaveDemand::bytes`] charges the
364/// chunks exactly.
365#[must_use]
366pub const fn chunk_bytes_per_row(layout: BlockShape, row: usize, workers: usize) -> u64 {
367    (workers as u64 + 1) * SampleBuffer::bytes(layout.correlations, row)
368}

370/// The slices of a block of `rows` rows split among `chunks` chunks of at
371/// most `chunk_rows` rows each, in row order.
372fn slices(
373    rows: usize,
374    chunks: usize,
375    chunk_rows: usize,
376) -> impl Iterator<Item = std::ops::Range<usize>> {
377    let step = chunks * chunk_rows;
378    (0..rows)
379        .step_by(step)
380        .map(move |start| start..(start + step).min(rows))
381}
382
383/// The chunks of `slice` among at most `chunks` chunks, in row order.
384fn chunk_ranges(
385    slice: std::ops::Range<usize>,
386    chunks: usize,
387) -> impl Iterator<Item = std::ops::Range<usize>> {
388    let rows = slice.len();
389    let count = chunks.min(rows.max(1));
390    (0..count).map(move |index| {
391        slice.start + index * rows / count..slice.start + (index + 1) * rows / count
392    })
393}

395/// Bytes a density pass over blocks of `block` on `workers` workers holds:
396/// its grid of `shape` ([`DensityGridShape::bytes`]), which it returns, and
397/// its row chunks' sample buffers, which it frees. The stream's native
398/// blocks are the source's ([`stream_bytes`]).
399#[must_use]
400pub fn density_pass_bytes(
401    resampler: &SpectralResampler,
402    shape: &DensityGridShape,
403    block: BlockShape,
404    workers: usize,
405) -> u64 {
406    let chunks = workers * DENSITY_CHUNKS_PER_WORKER;
407    let row = resampler.density_samples_per_row(shape, block.spectrum);
408    let placements = chunk_rows(row, block.rows, chunks, CHUNK_PLACEMENTS) * row;
409    shape.bytes() + chunks as u64 * SampleBuffer::bytes(1, placements)
410}

412/// Accumulate the weight-density grid of `shape` over one traversal of the
413/// first image domain's projection: each row is placed with
414/// [`SpectralResampler::place_density`] on row chunks and the chunks are
415/// added in row order. It holds [`density_pass_bytes`] while it runs.
416pub fn run_density_pass(
417    operator: &MeasurementOperator,
418    resampler: &SpectralResampler,
419    shape: DensityGridShape,
420    source: &mut dyn BoundedSource,
421    team: &WorkerTeam,
422    cancel: &Cancel,
423) -> Result<DensityGrid, PassError> {
424    source
425        .begin(PlaneRange::new(0, operator.basis().planes()), false)
426        .map_err(PassError::Source)?;
427    let mut grid = DensityGrid::new(shape);
428    let maximum = source.maximum_block();
429    let count = team.workers() * DENSITY_CHUNKS_PER_WORKER;
430    let row = resampler.density_samples_per_row(&shape, maximum.spectrum);
431    let rows_per_chunk = chunk_rows(row, maximum.rows, count, CHUNK_PLACEMENTS);
432    let placements = rows_per_chunk * row;
433    let mut chunks = (0..count)
434        .map(|_| (0..0, SampleBuffer::with_capacity(1, placements)))
435        .collect::<Vec<_>>();
436    stream_blocks(source, cancel, |block| {
437        for slice in slices(block.len(), count, rows_per_chunk) {
438            let mut used = 0;
439            for ((range, _), rows) in chunks.iter_mut().zip(chunk_ranges(slice, count)) {
440                *range = rows;
441                used += 1;
442            }
443            team.for_each_mut(&mut chunks[..used], |_, (range, buffer)| {
444                buffer.clear();
445                for row in range.clone() {
446                    resampler.place_density(operator, &block.row(0, row), &shape, buffer)?;
447                }
448                debug_assert!(
449                    buffer.capacity_bytes() == SampleBuffer::bytes(1, placements),
450                    "a density chunk places within the placements it was allocated for"
451                );
452                Ok::<_, PassError>(())
453            })?;
454            for (_, buffer) in &chunks[..used] {
455                grid.accumulate(&buffer.block());
456            }
457        }
458        Ok(())
459    })?;
460    Ok(grid)
