3//! Canonical projection of a sealed generation into publication.
4//!
5//! The projection is the only hand-off into the T08 publication
6//! choreography: it lists the exact sealed member set once, in canonical
7//! order, with the byte layout each member occupies in staging and final
8//! storage. Runtime adapters consume it when building their publication
9//! layout ledgers and receipt participants.
10
11use crate::authority::SealedContinuumGeneration;
12use crate::error::ProductsError;
13
14/// Publication stage of one projected member.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum PublicationStage {
17    /// Durably prepared under its staged identity; not yet visible.
18    Prepared,
19    /// Promoted through the sole visibility operation.
20    Published,
21}
22
23/// One sealed member projected for atomic publication.
24#[derive(Debug, Clone)]
25pub struct PublicationMemberProjection {
26    artifact_id: crate::authority::MemberArtifactId,
27    name: String,
28    payload_bytes: u64,
29}
30
31impl PublicationMemberProjection {
32    /// Return the sealed artifact identity.
33    #[must_use]
34    pub const fn artifact_id(&self) -> crate::authority::MemberArtifactId {
35        self.artifact_id
36    }
37
38    /// Return the compiled product name.
39    #[must_use]
40    pub fn name(&self) -> &str {
41        &self.name
42    }
43
44    /// Return the exact binary32 payload byte count.
45    #[must_use]
46    pub const fn payload_bytes(&self) -> u64 {
47        self.payload_bytes
48    }
49}
50
51/// Exact-once publication projection of one sealed generation.
52#[derive(Debug, Clone)]
53pub struct PublicationProjection {
54    seal_id: crate::authority::ContinuumSealId,
55    members: Box<[PublicationMemberProjection]>,
56}
57
58impl PublicationProjection {
59    /// Project the exact sealed member set in canonical publication order.
60    ///
61    /// # Errors
62    ///
63    /// Rejects a seal without members; the atomic store protocol requires at
64    /// least one published artifact.
65    pub fn from_sealed(sealed: &SealedContinuumGeneration) -> Result<Self, ProductsError> {
66        if sealed.members().is_empty() {
67            return Err(ProductsError::UnsupportedProblem);
68        }
69        let members = sealed
70            .members()
71            .iter()
72            .map(|member| PublicationMemberProjection {
73                artifact_id: member.artifact_id(),
74                name: member.name().to_string(),
75                payload_bytes: u64::try_from(member.payload().len())
76                    .expect("payload length fits in u64 on supported targets")
77                    * 4,
78            })
79            .collect::<Box<[_]>>();
80        Ok(Self {
81            seal_id: sealed.seal_id(),
82            members,
83        })
84    }
85
86    /// Return the seal this projection publishes exactly once.
87    #[must_use]
88    pub const fn seal_id(&self) -> crate::authority::ContinuumSealId {
89        self.seal_id
90    }
91
92    /// Return projected members in canonical publication order.
93    #[must_use]
94    pub const fn members(&self) -> &[PublicationMemberProjection] {
95        &self.members
96    }
97
98    /// Return aggregate staged (and final) payload bytes across all members.
99    #[must_use]
100    pub const fn total_payload_bytes(&self) -> u64 {
101        let mut total = 0_u64;
102        let mut index = 0;
103        while index < self.members.len() {
104            total += self.members[index].payload_bytes();
105            index += 1;
106        }
107        total
108    }
109}

111/// Fail-closed state of an in-flight atomic publication.
112///
113/// This mirrors the runtime's T08 choreography over the projection alone:
114/// preparing stages every member exactly once, visibility promotes the whole
115/// set or nothing, and promotion can never be re-entered after failure.
116#[derive(Debug, Default)]
117pub struct AtomicPublicationAttempt {
118    stage: Option<Stage>,
119}
120
121#[derive(Debug, Clone, Copy, PartialEq, Eq)]
122enum Stage {
123    Prepared,
124    Visible,
125    Failed,
126}
127
128impl AtomicPublicationAttempt {
129    /// Durably prepare every projected member before any visibility.
130    ///
131    /// # Errors
132    ///
133    /// Fails when the attempt already advanced past preparation.
134    pub fn prepare(
135        &mut self,
136        projection: &PublicationProjection,
137    ) -> Result<Vec<(crate::authority::MemberArtifactId, PublicationStage)>, ProductsError> {
138        if self.stage.is_some() {
139            return Err(ProductsError::ForeignPlannedGeneration);
140        }
141        self.stage = Some(Stage::Prepared);
142        Ok(projection
143            .members()
144            .iter()
145            .map(|member| (member.artifact_id(), PublicationStage::Prepared))
146            .collect())
147    }
148
149    /// Perform the sole visibility operation over the whole prepared set.
150    ///
151    /// # Errors
152    ///
153    /// Requires a prepared attempt; a failed attempt retains fail-closed
154    /// `Prepared` evidence and cannot retry.
155    pub fn make_visible(&mut self) -> Result<(), ProductsError> {
156        match self.stage {
157            Some(Stage::Prepared) => {
158                self.stage = Some(Stage::Visible);
159                Ok(())
160            }
161            Some(Stage::Failed) | None => Err(ProductsError::ForeignPlannedGeneration),
162            Some(Stage::Visible) => Err(ProductsError::ForeignPlannedGeneration),
163        }
164    }
165
166    /// Record that receipt promotion could not be confirmed.
167    ///
168    /// # Errors
169    ///
170    /// Requires a visible attempt.
171    pub fn fail_promotion(&mut self) -> Result<(), ProductsError> {
172        match self.stage {
173            Some(Stage::Visible) => {
174                self.stage = Some(Stage::Failed);
175                Ok(())
176            }
177            _ => Err(ProductsError::ForeignPlannedGeneration),
178        }
179    }
180
181    /// Whether the attempt retains fail-closed prepared evidence.
182    #[must_use]
183    pub const fn retains_prepared_evidence(&self) -> bool {
184        matches!(self.stage, Some(Stage::Failed))
185    }
186}
