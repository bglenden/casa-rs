35struct SpectralRewrite {

37    grid: LegacySpectralGrid,
38    phase: Option<Variant>,

48impl SpectralRewrite {
49    fn method(&self) -> &'static str {
50        match self.grid {
51            LegacySpectralGrid::Cvel => "cvel",
52            _ => "regridspw",
53        }
54    }
55
56    fn unchanged(&mut self, ms: &AttachedMs) -> PyResult<bool> {
57        let cvel = matches!(self.grid, LegacySpectralGrid::Cvel);
58        let link = ms.working_metadata_link();
59        link.verify_directory().map_err(runtime_error)?;
60        let target = if self.options.outframe.is_empty() {
61            None
62        } else if self.options.outframe.eq_ignore_ascii_case("SOURCE") {
63            return Ok(self.options.interpolation.is_empty());
64        } else {
65            match self.options.outframe.parse::<FrequencyRef>() {
66                Ok(FrequencyRef::Undefined) | Err(_) => return Ok(true),
67                Ok(target) => Some(target),
68            }
69        };
70        if self.options.interpolation.is_empty() && target != Some(FrequencyRef::REST) {
71            // SubMS declines an unknown interpolation without editing the grid.
72            return Ok(true);
73        }
74        if matches!(self.grid, LegacySpectralGrid::FrameOnly) && target.is_none() {
75            return Ok(true);
76        }
77
78        let tables = ms
79            .subtables
80            .spectral_grid_tables(&ms.main)
81            .map_err(runtime_error)?;
82        let mut unchanged =
83            !cvel || tables.windows.iter().map(Table::row_count).sum::<usize>() == 1;
84        for table in tables.windows.iter() {
85            let window = MsSpectralWindow::new(table);
86            for row in 0..table.row_count() {
87                let code = window.meas_freq_ref(row).map_err(runtime_error)?;
88                let reference = FrequencyRef::from_casacore_code(code)
89                    .ok_or_else(|| runtime_error(format!("Invalid MEAS_FREQ_REF {code}")))?;
90                if target.is_some_and(|target| target != reference) {
91                    if target == Some(FrequencyRef::REST) {
92                        return Err(runtime_error(
93                            "No MeasFrame specified for conversion of Frequency",
94                        ));
95                    }
96                    return Ok(false);
97                }
98                if unchanged && !self.options.interpolation.is_empty() {
99                    unchanged = if cvel {
100                        self.same_channels(&window, row, reference)?
101                    } else if matches!(self.grid, LegacySpectralGrid::FrameOnly) {
102                        true
103                    } else {
104                        self.grid.leaves_band_unchanged(
105                            &window.chan_freq(row).map_err(runtime_error)?,
106                            &window.chan_width(row).map_err(runtime_error)?,
107                        )
108                    };
109                }
110            }
111        }
112        link.verify_directory().map_err(runtime_error)?;
113        if target == Some(FrequencyRef::REST) {
114            self.options.outframe.clear();
115        }
116        Ok(unchanged || self.options.interpolation.is_empty())
117    }

119    fn same_channels(
120        &self,
121        window: &MsSpectralWindow<'_>,
122        row: usize,
123        reference: FrequencyRef,
124    ) -> PyResult<bool> {
125        let input = window.chan_freq(row).map_err(runtime_error)?;
126        let spectrum = ToolSpectralWindow {
127            centres_hz: input.clone(),
128            widths_hz: window.chan_width(row).map_err(runtime_error)?,
129            reference,
130        };
131        let budget = MsSelectionIoBudget::from_system_memory(1, size_of::<f64>(), None)
132            .map_err(runtime_error)?
133            .available_bytes;
134        let candidate =
135            tool_channel_frequencies(vec![spectrum], &self.options, budget, false, || {
136                unreachable!("checked the frequency reference before testing the grid")
137            });
138        Ok(candidate.is_ok_and(|centres| centres == input))
139    }
140
141    fn prepare_phase(&mut self, ms: &AttachedMs) -> PyResult<bool> {
142        if !matches!(self.grid, LegacySpectralGrid::Cvel) {
143            return Ok(true);
144        }
145        let phase = self.phase.take().unwrap_or(Variant::Empty);
146        let field = match &phase {
147            phase if phase.is_empty_sequence() => Some(0),
148            Variant::Text(text) if text.is_empty() => Some(0),
149            Variant::Number(field) => Some(*field as i32),
150            Variant::Numbers(fields) => fields.first().map(|&field| field as i32),
151            Variant::Array { values, .. } => values.first().map(|&field| field as i32),
152            _ => None,
153        };
154        if let Some(field) = field {
155            let tables = ms
156                .subtables
157                .spectral_grid_tables(&ms.main)
158                .map_err(runtime_error)?;
159            let rows = tables.fields.iter().map(Table::row_count).sum::<usize>();
160            if !(0..rows).contains(&(field as usize)) {
161                return Ok(false);
162            }
163            self.options.phasecenter = field.to_string();
164        } else {
165            let measures = ms.spectral_query_measures().map_err(runtime_error)?;
166            let Some(direction) = grid_direction(&phase, &measures) else {
167                return Ok(false);
168            };
169            let frame = MeasFrame::new().with_measures(measures);
170            let direction = direction.resolve_offset(&frame).map_err(runtime_error)?;
171            self.options.phasecenter = format!(
172                "{} {}rad {}rad",
173                direction.refer(),
174                direction.longitude_rad(),
175                direction.latitude_rad()
176            );
177        }
178        Ok(true)
179    }
180}

182#[pymethods]
183impl PyMsTool {
184    /// Regrid the attached MeasurementSet in place.
185    #[allow(clippy::too_many_arguments)]
186    #[pyo3(signature = (outframe = "LSRK".into(), mode = "chan".into(), restfreq = -3e30, interpolation = "LINEAR".into(), start = -3e30, center = -3e30, bandwidth = -1., chanwidth = -1., hanning = true), text_signature = "($self, outframe='LSRK', mode='chan', restfreq=-3e+30, interpolation='LINEAR', start=-3e+30, center=-3e+30, bandwidth=-1.0, chanwidth=-1.0, hanning=True)")]
187    fn regridspw(
188        &self,
189        py: Python<'_>,
190        outframe: String,
191        mode: String,
192        restfreq: f64,
193        interpolation: String,
194        start: f64,
195        center: f64,
196        bandwidth: f64,
197        chanwidth: f64,
198        #[pyo3(from_py_with = strict_bool)] hanning: bool,
199    ) -> PyResult<bool> {
200        let centre = if start > UNSET_CENTRE {
201            let derived = start + bandwidth / 2.;
202            let derived = if mode == "chan" {
203                derived.floor()
204            } else {
205                derived
206            };
207            if center > UNSET_CENTRE && derived != center {
208                return Ok(false);
209            }
210            derived
211        } else {
212            center
213        };
214        // Empty mode and the default channel band only transform the frame.
215        let grid = match mode.as_str() {
216            "" => LegacySpectralGrid::FrameOnly,
217            "chan" if centre < UNSET_CENTRE && bandwidth <= 0. && chanwidth <= 1. => {
218                LegacySpectralGrid::FrameOnly
219            }
220            "chan" => LegacySpectralGrid::CenteredChannels {
221                centre,
222                bandwidth,
223                channel_width: chanwidth,
224            },
225            _ => LegacySpectralGrid::CenteredNumeric {
226                quantity: match mode.as_str() {
227                    "freq" => LegacyRegridQuantity::Frequency,
228                    "vrad" => LegacyRegridQuantity::RadioVelocity,
229                    "vopt" => LegacyRegridQuantity::OpticalVelocity,
230                    "wave" => LegacyRegridQuantity::Wavelength,
231                    _ => return self.invalid_regridspw(),
232                },
233                centre,
234                bandwidth,
235                channel_width: chanwidth,
236                rest_hz: restfreq,
237            },
238        };
239        let interpolation = if matches!(grid, LegacySpectralGrid::FrameOnly) {
240            "linear"
241        } else {
242            let fft_allowed = !matches!(
243                grid,
244                LegacySpectralGrid::CenteredNumeric {
245                    quantity: LegacyRegridQuantity::OpticalVelocity
246                        | LegacyRegridQuantity::Wavelength,
247                    ..
248                }
249            );
250            rewrite_interpolation(&interpolation, fft_allowed)
251        };
252        self.rewrite_spectra(
253            py,
254            SpectralRewrite {
255                options: MsRegridOptions {
256                    mode: "channel".into(),
257                    nchan: -1,
258                    start: String::new(),
259                    width: String::new(),
260                    nspw: 1,
261                    interpolation: interpolation.into(),
262                    phasecenter: String::new(),
263                    restfreq: String::new(),
264                    outframe,
265                    veltype: "radio".into(),
266                    preaverage: false,
267                },
268                grid,
269                phase: None,
270                hanning,
271            },
272        )
273    }
274
275    /// Combine the input windows and write a new spectral grid in place.
276    #[allow(clippy::too_many_arguments)]
277    #[pyo3(signature = (mode = "channel".into(), nchan = -1, start = None, width = None, interp = "linear".into(), phasec = None, restfreq = None, outframe = String::new(), veltype = "radio".into(), hanning = true), text_signature = "($self, mode='channel', nchan=-1, start=[], width=[], interp='linear', phasec=[], restfreq=[], outframe='', veltype='radio', hanning=True)")]
278    fn cvel(
279        &self,
280        py: Python<'_>,
281        mode: String,
282        nchan: i64,
283        start: Option<Variant>,
284        width: Option<Variant>,
285        interp: String,
286        phasec: Option<Variant>,
287        restfreq: Option<Variant>,
288        outframe: String,
289        veltype: String,
290        #[pyo3(from_py_with = strict_bool)] hanning: bool,
291    ) -> PyResult<bool> {
292        let unit = match mode.as_str() {
293            "channel" | "channel_b" => None,
294            "frequency" => Some("Hz"),
295            "velocity" => Some("m/s"),
296            _ => return Ok(false),
297        };
298        let Some(restfreq) = cvel_quantity(axis_argument("", restfreq), "Hz") else {
299            return Ok(false);
300        };
301        if mode == "velocity" && matches!(restfreq.as_str(), "" | "0Hz" | "-0Hz") {
302            return Ok(false);
303        }
304        let start = axis_argument(&mode, start);
305        let width = axis_argument(&mode, width);
306        let (start, width) = if let Some(unit) = unit {
307            let (Some(start), Some(width)) =
308                (cvel_quantity(start, unit), cvel_quantity(width, unit))
309            else {
310                return Ok(false);
311            };
312            (start, width)
313        } else {
314            (start, width)
315        };
316        let restfreq = if mode == "velocity" {
317            restfreq
318        } else {
319            String::new()
320        };
321        let interpolation =
322            rewrite_interpolation(&interp, mode != "velocity" || veltype != "optical");
323        let veltype = if veltype == "optical" {
324            veltype
325        } else {
326            "radio".into()
327        };
328        self.rewrite_spectra(
329            py,
330            SpectralRewrite {
331                options: MsRegridOptions {
332                    start,
333                    width,
334                    mode,
335                    nchan,
336                    nspw: 1,
337                    interpolation: interpolation.into(),
338                    phasecenter: String::new(),
339                    restfreq,
340                    outframe,
341                    veltype,
342                    preaverage: false,
343                },
344                grid: LegacySpectralGrid::Cvel,
345                phase: phasec,
346                hanning,
347            },
348        )
349    }
350}

352fn cvel_quantity(text: String, unit: &str) -> Option<String> {
353    if matches!(text.as_str(), "" | "[]") {
354        return Some(String::new());
355    }
356    let quantity = Quantity::read(&text).ok()?;
357    let target = Unit::new(unit).expect("cvel uses built-in units");
358    // SubMS reads a QuantumHolder, then uses the non-conformant getValue overload.
359    let value = casacore::get_value(&quantity, &target);
360    value.is_finite().then(|| format!("{value}{unit}"))
361}

363fn rewrite_interpolation(interpolation: &str, fft_allowed: bool) -> &'static str {
364    let interpolation = interpolation.to_ascii_lowercase();
365    if interpolation.contains("nearest") {
366        "nearest"
367    } else if interpolation.contains("splin") {
368        "spline"
369    } else if interpolation.contains("cub") {
370        "cubic"
371    } else if interpolation.contains("fft") {
372        if fft_allowed { "fftshift" } else { "" }
373    } else if interpolation.is_empty() || interpolation.contains("linear") {
374        "linear"
375    } else {
376        ""
377    }
378}

380impl PyMsTool {
381    fn invalid_regridspw(&self) -> PyResult<bool> {
382        let attached = self.attached.lock().map_err(runtime_error)?;
383        if attached.as_ref().is_some_and(|ms| ms.writable) {
384            Err(runtime_error("Regridding failed."))
385        } else {
386            Ok(false)
387        }
388    }
389
390    fn rewrite_spectra(&self, py: Python<'_>, mut rewrite: SpectralRewrite) -> PyResult<bool> {
391        py.detach(|| {
392            let mut attached = self.attached.lock().map_err(runtime_error)?;
393            let cvel = rewrite.method() == "cvel";
394            let Some(ms) = attached.as_mut() else {
395                return if cvel {
396                    Err(runtime_error("Tool is not attached to an MS"))
397                } else {
398                    Ok(false)
399                };
400            };
401            if !ms.writable {
402                return Ok(false);
403            }
404            if cvel && ms.selected_rows.is_some() {
405                return Err(runtime_error(
406                    "In-place cvel requires the complete attached MeasurementSet",
407                ));
408            }
409            if !rewrite.prepare_phase(ms)? {
410                return Ok(false);
411            }
412            if cvel {
413                validate_cvel_windows(ms)?;
414            }
415            if ms.selected_rows.is_none()
416                && ms
417                    .working_metadata_link()
418                    .row_count()
419                    .map_err(runtime_error)?
420                    == 0
421            {
422                return Ok(true);
423            }
424            if !has_unflagged_antenna(ms)? || rewrite.unchanged(ms)? {
425                if cvel && rewrite.hanning {
426                    return ms.hanning_rows(HanningDataColumn::Both);
427                }
428                return Ok(true);
429            }
430            if ms.selected_rows.is_some() {
431                return Ok(false);
432            }
433            let read_locks = std::mem::take(&mut ms.read_locks).into_keys().collect();
434            let hanning = rewrite.hanning;
435            match rewrite_attached(ms, rewrite) {
436                Ok(RewriteOutcome::Complete(mut replacement)) => {
437                    if !replacement.write_locks.is_empty() {
438                        replacement.write_locks.extend(
439                            std::mem::take(&mut ms.write_locks)
440                                .into_iter()
441                                .filter(|(path, _)| !path.starts_with(&ms.path)),
442                        );
443                    }
444                    *attached = Some(replacement);
445                    Ok(true)
446                }
447                Ok(RewriteOutcome::Unchanged) => {
448                    ms.retain_tables(read_locks)?;
449                    if cvel && hanning {
450                        return ms.hanning_rows(HanningDataColumn::Both);
451                    }
452                    Ok(true)
453                }
454                Ok(RewriteOutcome::Refused) => {
455                    ms.retain_tables(read_locks)?;
456                    Ok(false)
457                }
458                Err(error) => {
459                    ms.retain_tables(read_locks)?;
460                    Err(error)
461                }
462            }
463        })
464    }
465}

467fn validate_cvel_windows(ms: &AttachedMs) -> PyResult<()> {
468    let link = ms.working_metadata_link();
469    link.verify_directory().map_err(runtime_error)?;
470    let tables = ms
471        .subtables
472        .spectral_grid_tables(&ms.main)
473        .map_err(runtime_error)?;
474    for table in tables.windows.iter() {
475        let windows = MsSpectralWindow::new(table);
476        for row in 0..table.row_count() {
477            let centres = windows.chan_freq(row).map_err(runtime_error)?;
478            let widths = windows.chan_width(row).map_err(runtime_error)?;
479            let sign = if centres.first() > centres.last() {
480                -1.
481            } else {
482                1.
483            };
484            // SubMS checks adjacent channel edges before combining windows.
485            if centres.len() != widths.len()
486                || centres
487                    .windows(2)
488                    .zip(widths.windows(2))
489                    .take(centres.len().saturating_sub(2))
490                    .any(|(centre, width)| {
491                        let upper = centre[0] + sign * width[0].abs() / 2.;
492                        let lower = centre[1] - sign * width[1].abs() / 2.;
493                        (upper - lower).abs() > 1.
494                    })
495            {
496                return Err(runtime_error(
497                    "Inconsistent SPECTRAL_WINDOW table in input MS.",
498                ));
499            }
500        }
501    }
502    link.verify_directory().map_err(runtime_error)
503}

505fn has_unflagged_antenna(ms: &AttachedMs) -> PyResult<bool> {
506    let id = SubtableId::Antenna;
507    let tables = ms
508        .subtables
509        .spectral_grid_geometry(&ms.main, &[id])
510        .map_err(runtime_error)?;
511    for table in tables[&id].iter() {
512        let antenna = MsAntenna::new(table);
513        for row in 0..table.row_count() {
514            if !antenna.flag_row(row).map_err(runtime_error)? {
515                return Ok(true);
516            }
517        }
518    }
519    Ok(false)
520}

522fn rewrite_attached(
523    ms: &AttachedMs,
524    rewrite: SpectralRewrite,
525) -> PyResult<RewriteOutcome<AttachedMs>> {
526    let operation = rewrite.method();
527    let interpolation = rewrite.options.interpolation.clone();
528    let link = ms.working_metadata_link();
529    link.verify_directory().map_err(runtime_error)?;
530    check_write_access(&ms.path).map_err(runtime_error)?;
531    let destination = SortedDestination::Table(ms.directory.clone());
532    let locks = lock_table_tree_for_replacement_with_held(&ms.path, &ms.write_locks)
533        .map_err(runtime_error)?;
534    let workspace = tempfile::tempdir_in(ms.path.parent().unwrap_or_else(|| Path::new(".")))
535        .map_err(runtime_error)?;
536    let staged = workspace.path().join("spectral.ms");
537    let frame_only = matches!(rewrite.grid, LegacySpectralGrid::FrameOnly);
538    if frame_only {
539        stage_frame_only_ms(&ms.path, &staged).map_err(runtime_error)?;
540        match reframe_staged_ms(&staged, &rewrite.options) {
541            Ok(true) => {}
542            Ok(false) | Err(MsTransformError::RegridParameter { .. }) => {
543                return Ok(RewriteOutcome::Refused);
544            }
545            Err(error) => return Err(runtime_error(error)),
546        }
547    } else {
548        match regrid_to_stage(ms, rewrite, &staged)? {
549            RewriteOutcome::Complete(()) => {}
550            RewriteOutcome::Unchanged => return Ok(RewriteOutcome::Unchanged),
551            RewriteOutcome::Refused => return Ok(RewriteOutcome::Refused),
552        }
553    }
554    let table = Table::open(TableOptions::new(&staged)).map_err(runtime_error)?;
555    let history = spectral_history(&table, &staged, &interpolation)?;
556    append_tool_history(
557        &table,
558        true,
559        &[format!("Transformed/regridded with {operation}")],
560        &history,
561        &format!("ms::{operation}()"),
562        "ms",
563        &mut Default::default(),
564    )
565    .map_err(runtime_error)?;
566    drop(table);
567    if !frame_only {
568        preserve_replacement_tree_entries(&ms.path, &staged).map_err(runtime_error)?;
569    }
570    link.verify_directory().map_err(runtime_error)?;
571    if !locks
572        .iter()
573        .all(|lock| lock.matches_table_path(lock.path()))
574    {
575        return Err(runtime_error("The input changed while regridding"));
576    }
577    let keep_lock = operation == "regridspw" && !ms.write_locks.is_empty();
578    let original_main = ms
579        .write_locks
580        .get(&ms.path)
581        .or_else(|| locks.iter().find(|lock| lock.path() == ms.path))
582        .ok_or_else(|| runtime_error("The original MeasurementSet has no write lock"))?;
583    let mut published_locks = lock_table_tree_for_replacement(&staged)
584        .map_err(runtime_error)?
585        .into_iter()
586        .map(|lock| {
587            let relative = lock
588                .path()
589                .strip_prefix(&staged)
590                .expect("lock under the stage");
591            (ms.path.join(relative), lock)
592        })
593        .collect::<Vec<_>>();
594    destination.publish(&staged, &ms.path, operation)?;
595    match reopen_rewritten_ms(ms, &mut published_locks, keep_lock) {
596        Ok(replacement) => Ok(RewriteOutcome::Complete(replacement)),
597        Err(error) => {
598            let rollback = if original_main.matches_table_path(&staged)
599                && published_locks
600                    .iter()
601                    .all(|(path, lock)| lock.matches_table_path(path))
602            {
603                exchange_directories(&staged, &ms.path)
604            } else {
605                Err(std::io::Error::other(
606                    "a locked directory changed at publication",
607                ))
608            };
609            if let Err(rollback) = rollback {
610                let backup = workspace.keep();
611                return Err(runtime_error(format!(
612                    "Could not reopen the regridded MS: {error}; rollback failed: {rollback}; original at {}",
613                    backup.join("spectral.ms").display()
614                )));
615            }
616            Err(error)
617        }
618    }
619}

621fn reopen_rewritten_ms(
622    ms: &AttachedMs,
623    published_locks: &mut Vec<(std::path::PathBuf, TableWriteLock)>,
624    keep_lock: bool,
625) -> PyResult<AttachedMs> {
626    for (path, lock) in published_locks.iter_mut() {
627        lock.follow_directory_rename(path).map_err(runtime_error)?;
628    }
629    let mut replacement = AttachedMs::open(ms.path.clone(), true, false, false)?;
630    replacement.name = ms.name.clone();
631    if keep_lock {
632        replacement.write_locks.extend(published_locks.drain(..));
633    }
634    Ok(replacement)
635}

637fn spectral_history(main: &Table, staged: &Path, interpolation: &str) -> PyResult<String> {
638    let path = subtable_path_from_main(main, staged, SubtableId::SpectralWindow)
639        .ok_or_else(|| runtime_error("Missing SPECTRAL_WINDOW."))?;
640    let table = Table::open(TableOptions::new(path)).map_err(runtime_error)?;
641    let windows = MsSpectralWindow::new(&table);
642    let mut history = String::new();
643    for row in 0..table.row_count() {
644        let code = windows.meas_freq_ref(row).map_err(runtime_error)?;
645        let reference = FrequencyRef::from_casacore_code(code)
646            .ok_or_else(|| runtime_error(format!("Invalid MEAS_FREQ_REF {code}")))?;
647        let channels = windows.num_chan(row).map_err(runtime_error)?;
648        let bandwidth = windows.total_bandwidth(row).map_err(runtime_error)?;
649        writeln!(
650            history,
651            "Regridded spectral window {row}\n output frame = {reference}\n Number of channels = {channels}\n Total width of SPW (in output frame) = {bandwidth} Hz\n Interpolation Method = {interpolation}"
652        )
653        .expect("writing to a String cannot fail");
654    }
655    Ok(history)
656}
657
658fn regrid_to_stage(
659    ms: &AttachedMs,
660    rewrite: SpectralRewrite,
661    staged: &Path,
662) -> PyResult<RewriteOutcome<()>> {
663    let operation = rewrite.method();
664    let link = ms.working_metadata_link();
665    let combine_spws =
666        operation == "cvel" && link.spectral_window_count().map_err(runtime_error)? > 1;
667    let request = MsTransformRequest {
668        input_ms: ms.path.clone(),
669        output_ms: staged.to_owned(),
670        spw: "*".into(),
671        width: 1,
672        data_column: TransformDataColumn::All,
673        selection: Default::default(),
674        keep_flags: true,
675    };
676    let options = MsTransformTaskOptions {
677        combine_spws,
678        preserve_ids: !combine_spws,
679        hanning: rewrite.hanning,
680        regrid: Some(rewrite.options),
681        legacy_tool: Some(LegacyMsTool::Regrid(rewrite.grid)),
682        ..Default::default()
683    };
684    match mstransform_task(&request, &options) {
685        Err(MsTransformError::LegacyRegridUnchanged) => return Ok(RewriteOutcome::Unchanged),
686        Err(MsTransformError::Spectrum(_)) if operation == "regridspw" => {
687            return Err(runtime_error("Regridding failed."));
688        }
689        Err(MsTransformError::RegridParameter { .. } | MsTransformError::Spectrum(_)) => {
690            return Ok(RewriteOutcome::Refused);
691        }
692        result => result.map_err(runtime_error)?,
693    };
694    Ok(RewriteOutcome::Complete(()))
695}
