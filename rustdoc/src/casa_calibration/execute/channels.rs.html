8/// The compact buffer's physical frequencies and positions in the original SPW.
9#[derive(Clone, Copy)]
10pub(super) struct DataChannelGrid<'a> {
11    pub frequencies_hz: &'a [f64],
12    indices: Option<&'a [usize]>,
13}
14
15impl<'a> DataChannelGrid<'a> {
16    pub fn all(frequencies_hz: &'a [f64]) -> Self {
17        Self {
18            frequencies_hz,
19            indices: None,
20        }
21    }
22
23    pub fn original_index(self, compact_index: usize) -> usize {
24        self.indices
25            .map_or(compact_index, |indices| indices[compact_index])
26    }
27}

29fn select(
30    value: &ArrayValue,
31    channels: &[usize],
32    column: &str,
33) -> Result<ArrayValue, ApplyExecutionError> {
34    if value.shape().len() != 2 || channels.iter().any(|&channel| channel >= value.shape()[1]) {
35        return Err(ApplyExecutionError::UnsupportedParameterShape {
36            path: format!("<measurement-set {column}>"),
37            shape: value.shape().to_vec(),
38        });
39    }
40    Ok(match value {
41        ArrayValue::Complex32(values) => ArrayValue::Complex32(values.select(Axis(1), channels)),
42        ArrayValue::Float32(values) => ArrayValue::Float32(values.select(Axis(1), channels)),
43        ArrayValue::Bool(values) => ArrayValue::Bool(values.select(Axis(1), channels)),
44        _ => {
45            return Err(ApplyExecutionError::UnsupportedParameterShape {
46                path: format!("<measurement-set {column}>"),
47                shape: value.shape().to_vec(),
48            });
49        }
50    })
51}

53fn scatter(
54    base: &ArrayValue,
55    selected: &ArrayValue,
56    channels: &[usize],
57    column: &str,
58) -> Result<ArrayValue, ApplyExecutionError> {
59    if base.shape().len() != 2
60        || selected.shape() != [base.shape()[0], channels.len()]
61        || channels.iter().any(|&channel| channel >= base.shape()[1])
62    {
63        return Err(ApplyExecutionError::UnsupportedParameterShape {
64            path: format!("<measurement-set {column}>"),
65            shape: base.shape().to_vec(),
66        });
67    }
68    macro_rules! copy {
69        ($base:expr, $selected:expr, $variant:ident) => {{
70            let mut output = $base.clone();
71            for (source, &target) in channels.iter().enumerate() {
72                output
73                    .index_axis_mut(Axis(1), target)
74                    .assign(&$selected.index_axis(Axis(1), source));
75            }
76            ArrayValue::$variant(output)
77        }};
78    }
79    Ok(match (base, selected) {
80        (ArrayValue::Complex32(base), ArrayValue::Complex32(selected)) => {
81            copy!(base, selected, Complex32)
82        }
83        (ArrayValue::Float32(base), ArrayValue::Float32(selected)) => {
84            copy!(base, selected, Float32)
85        }
86        (ArrayValue::Bool(base), ArrayValue::Bool(selected)) => copy!(base, selected, Bool),
87        _ => {
88            return Err(ApplyExecutionError::UnsupportedParameterShape {
89                path: format!("<measurement-set {column}>"),
90                shape: base.shape().to_vec(),
91            });
92        }
93    })
94}

96struct SelectedInputs {
97    data: ArrayValue,
98    flags: Option<ArrayValue>,
99    weights: Option<ArrayValue>,
100}
101
102impl SelectedInputs {
103    fn new(
104        input: ExecutionRowInputs<'_>,
105        selection: &ApplyChannelSelection,
106    ) -> Result<Self, ApplyExecutionError> {
107        Ok(Self {
108            data: select(input.data, &selection.indices, "DATA")?,
109            flags: input
110                .original_flags
111                .map(|value| select(value, &selection.indices, "FLAG"))
112                .transpose()?,
113            weights: input
114                .original_weight_spectrum
115                .map(|value| select(value, &selection.indices, "SIGMA_SPECTRUM"))
116                .transpose()?,
117        })
118    }
119
120    fn inputs<'a>(
121        &'a self,
122        input: ExecutionRowInputs<'a>,
123        selection: &'a ApplyChannelSelection,
124    ) -> ExecutionRowInputs<'a> {
125        ExecutionRowInputs {
126            data: &self.data,
127            original_flags: self.flags.as_ref(),
128            original_weight_spectrum: self.weights.as_ref(),
129            channel_grid: Some(DataChannelGrid {
130                frequencies_hz: &selection.frequencies_hz,
131                indices: Some(&selection.indices),
132            }),
133            ..input
134        }
135    }
136}

138pub(super) fn apply_row(
139    row: &ApplyRowPlan,
140    input: ExecutionRowInputs<'_>,
141    plan: &ApplyPlan,
142    chain: &ApplyChain,
143    geometry: Option<&mut RowGeometryCache>,
144    profile: Option<&mut ApplyRowComputeProfile>,
145    activity: Option<&mut ToolApplyActivity>,
146) -> Result<ExecutionRowResult, ApplyExecutionError> {
147    let Some(selection) = plan.channel_selections.get(&row.data_spw_id) else {
148        return apply_row_all_channels(row, input, plan, chain, geometry, profile, activity);
149    };
150    let selected = SelectedInputs::new(input, selection)?;
151    let mut result = apply_row_all_channels(
152        row,
153        selected.inputs(input, selection),
154        plan,
155        chain,
156        geometry,
157        profile,
158        activity,
159    )?;
160    result.corrected_data = scatter(
161        input.preserved_corrected_data.unwrap_or(input.data),
162        &result.corrected_data,
163        &selection.indices,
164        "CORRECTED_DATA",
165    )?;
166    if let Some(flags) = &result.updated_flags {
167        let original = input.original_flags.expect("CalFlag requires FLAG");
168        let full = scatter(original, flags, &selection.indices, "FLAG")?;
169        let ArrayValue::Bool(values) = &full else {
170            unreachable!("FLAG type checked above")
171        };
172        result.row_became_fully_flagged = values.iter().all(|flag| *flag);
173        result.updated_flags = Some(full);
174    }
175    if let Some(weights) = &result.updated_weight_spectrum {
176        let expanded;
177        let base = if let Some(value) = input
178            .preserved_weight_spectrum
179            .or(input.original_weight_spectrum)
180        {
181            value
182        } else {
183            let Some(ArrayValue::Float32(weight)) = input.original_weight else {
184                return Err(ApplyExecutionError::UnsupportedParameterShape {
185                    path: "<measurement-set WEIGHT>".into(),
186                    shape: vec![],
187                });
188            };
189            expanded =
190                ArrayValue::Float32(expand_weight_to_spectrum(weight, input.data.shape()[1]));
191            &expanded
192        };
193        result.updated_weight_spectrum = Some(scatter(
194            base,
195            weights,
196            &selection.indices,
197            "WEIGHT_SPECTRUM",
198        )?);
199    }
200    Ok(result)
201}
