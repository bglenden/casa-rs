2//! Trim outliers against a median absolute-deviation fit before least squares.
3
4pub(super) fn trim(
5    x: &[f64],
6    y: &[f64],
7    errors: &[f64],
8    flags: &mut [bool],
9    degree: usize,
10    weighted: bool,
11) -> Option<()> {
12    let points = x
13        .iter()
14        .zip(y)
15        .zip(flags.iter())
16        .filter_map(|((&x, &y), &flag)| (!flag).then_some((x, y)))
17        .collect::<Vec<_>>();
18    let slope = match degree {
19        0 => 0.,
20        1 => robust_slope(&points)?,
21        _ => return None,
22    };
23    let intercept = median(&mut offsets(&points, slope))?;
24    let residuals = x
25        .iter()
26        .zip(y)
27        .zip(errors)
28        .map(|((&x, &y), &error)| (y - (slope * x + intercept)) / if weighted { error } else { 1. })
29        .collect::<Vec<_>>();
30    let limit = 5. * average_deviation(&residuals);
31    for (flag, residual) in flags.iter_mut().zip(residuals) {
32        *flag |= residual.abs() >= limit;
33    }
34    Some(())
35}

37fn robust_slope(points: &[(f64, f64)]) -> Option<f64> {
38    let mut pairs = Vec::new();
39    for (index, &(first_x, first_y)) in points.iter().enumerate() {
40        for &(second_x, second_y) in &points[index + 1..] {
41            if first_x != second_x || first_y != second_y {
42                pairs.push((second_y - first_y) / (second_x - first_x));
43            }
44        }
45    }
46    let estimate = median(&mut pairs)?;
47    let width = 10. * average_deviation(&pairs);
48    let (mut low, mut high) = (estimate - width, estimate + width);
49    for iteration in 0..30 {
50        let step = (high - low) / 20.;
51        let initial = slope_sign(points, low)?;
52        let mut bracketed = false;
53        for index in 1..=20 {
54            let candidate = low + f64::from(index) * step;
55            let sign = slope_sign(points, candidate)?;
56            if sign == 0. {
57                return Some(candidate);
58            }
59            if sign != initial {
60                (low, high) = (candidate - step, candidate);
61                bracketed = true;
62                break;
63            }
64        }
65        if !bracketed && iteration == 0 {
66            return None;
67        }
68    }
69    Some(0.5 * (low + high))
70}

72fn slope_sign(points: &[(f64, f64)], slope: f64) -> Option<f64> {
73    let intercept = median(&mut offsets(points, slope))?;
74    Some(sign(
75        points
76            .iter()
77            .map(|&(x, y)| x * sign(y - slope * x - intercept))
78            .sum(),
79    ))
80}

82fn sign(value: f64) -> f64 {
83    if value > 0. {
84        1.
85    } else if value < 0. {
86        -1.
87    } else {
88        0.
89    }
90}

92fn offsets(points: &[(f64, f64)], slope: f64) -> Vec<f64> {
93    points.iter().map(|&(x, y)| y - slope * x).collect()
94}

96fn median(values: &mut [f64]) -> Option<f64> {
97    if values.is_empty() || values.iter().any(|value| !value.is_finite()) {
98        return None;
99    }
100    let even = values.len().is_multiple_of(2);
101    let (lower, middle, _) = values.select_nth_unstable_by(values.len() / 2, f64::total_cmp);
102    Some(if even {
103        (lower.iter().copied().fold(f64::NEG_INFINITY, f64::max) + *middle) / 2.
104    } else {
105        *middle
106    })
107}

109fn average_deviation(values: &[f64]) -> f64 {
110    let mean = values.iter().sum::<f64>() / values.len() as f64;
111    values.iter().map(|value| (value - mean).abs()).sum::<f64>() / values.len() as f64
112}
