2//! The saved additive field of `setnoise(OTF=False)`: reused on every corrupt.
3use casa_calibration::SimulatedNoiseSolution;

8#[derive(Debug)]
9struct Chunk {
10    time: f64,
11    field: i32,
12    offset: usize,
13}
14#[derive(Debug)]
15struct Window {
16    channels: usize,
17    chunks: Vec<Chunk>,
18}

20/// A complete A Noise field, with its storage admitted before construction.
21#[derive(Debug)]
22pub(crate) struct OfflineNoise {
23    pub(crate) metadata: casa_calibration::SimulatedTableMetadata,
24    pub(crate) antennas: usize,
25    windows: Vec<Window>,
26    values: Vec<Complex32>,
27    chunks: usize,
28    _reservation: Reservation,
29}
30
31impl OfflineNoise {
32    pub(crate) fn bytes(antennas: usize, counts: &[usize], channels: &[usize]) -> u64 {
33        let baselines = antennas * (antennas + 1) / 2;
34        let values = counts
35            .iter()
36            .zip(channels)
37            .map(|(n, c)| n * baselines * c * 4)
38            .sum::<usize>();
39        (counts.len() * size_of::<Window>()
40            + counts.iter().sum::<usize>() * size_of::<Chunk>()
41            + values * size_of::<Complex32>()
42            + 1024) as u64
43    }
44
45    pub(crate) fn new(
46        antennas: usize,
47        counts: &[usize],
48        channels: &[usize],
49        reservation: Reservation,
50        metadata: casa_calibration::SimulatedTableMetadata,
51    ) -> Self {
52        let baselines = antennas * (antennas + 1) / 2;
53        let values = counts
54            .iter()
55            .zip(channels)
56            .map(|(n, c)| n * baselines * c * 4)
57            .sum();
58        Self {
59            metadata,
60            antennas,
61            windows: counts
62                .iter()
63                .zip(channels)
64                .map(|(&count, &channels)| Window {
65                    channels,
66                    chunks: Vec::with_capacity(count),
67                })
68                .collect(),
69            values: Vec::with_capacity(values),
70            chunks: 0,
71            _reservation: reservation,
72        }
73    }
74
75    pub(crate) fn push(&mut self, window: usize, time: f64, field: i32, seed: i32, sigma: f32) {
76        let channels = self.windows[window].channels;
77        let baselines = self.antennas * (self.antennas + 1) / 2;
78        let offset = self.values.len();
79        self.values
80            .resize(offset + baselines * channels * 4, Complex32::new(0., 0.));
81        let noise = CellNoise::new([seed as u32, super::PURPOSE_THERMAL_NOISE << 24], sigma);
82        for p in 0..self.antennas {
83            for q in p..self.antennas {
84                let baseline = self.baseline(p, q);
85                let begin = offset + baseline * channels * 4;
86                noise.add_to_row(
87                    (self.chunks * baselines + baseline) as u64,
88                    p == q,
89                    4,
90                    &mut self.values[begin..begin + channels * 4],
91                );
92            }
93        }
94        self.windows[window].chunks.push(Chunk {
95            time,
96            field,
97            offset,
98        });
99        self.chunks += 1;
100    }
101
102    pub(crate) fn sorted(mut self) -> Self {
103        for window in &mut self.windows {
104            window
105                .chunks
106                .sort_unstable_by(|a, b| a.time.total_cmp(&b.time).then(a.offset.cmp(&b.offset)));
107        }
108        self
109    }
110
111    pub(crate) fn covers(&self, window: usize) -> bool {
112        self.windows
113            .get(window)
114            .is_some_and(|w| !w.chunks.is_empty())
115    }
116
117    fn baseline(&self, p: usize, q: usize) -> usize {
118        p * self.antennas - p * (p + 1) / 2 + q
119    }
120
121    pub(crate) fn add_to_cell(
122        &self,
123        window: usize,
124        time: f64,
125        antennas: (usize, usize),
126        receptors: &[[usize; 2]],
127        cell: &mut [Complex32],
128    ) {
129        self.add_scaled_to_cell((window, time), antennas, receptors, cell, |_| 1.);
130    }
131
132    pub(crate) fn add_scaled_to_cell(
133        &self,
134        (window, time): (usize, f64),
135        antennas: (usize, usize),
136        receptors: &[[usize; 2]],
137        cell: &mut [Complex32],
138        sigma: impl Fn(usize) -> f32,
139    ) {
140        let Some(window) = self.windows.get(window).filter(|w| !w.chunks.is_empty()) else {
141            return;
142        };
143        let after = window.chunks.partition_point(|c| c.time < time);
144        let index = match (after.checked_sub(1), window.chunks.get(after)) {
145            (Some(before), Some(later))
146                if later.time - time < time - window.chunks[before].time =>
147            {
148                after
149            }
150            (Some(before), _) => before,
151            _ => 0,
152        };
153        let (p, q) = antennas;
154        if p >= self.antennas || q >= self.antennas {
155            return;
156        }
157        let reverse = p > q;
158        let (p, q) = if reverse { (q, p) } else { (p, q) };
159        let offset = window.chunks[index].offset + self.baseline(p, q) * window.channels * 4;
160        for (index, value) in cell.iter_mut().enumerate() {
161            let [r1, r2] = receptors[index % receptors.len()];
162            let parameter = if reverse { 2 * r2 + r1 } else { 2 * r1 + r2 };
163            let noise = self.values[offset + index / receptors.len() * 4 + parameter];
164            // VisEquation corrects zero DATA by A before adding the model.
165            *value -= (if reverse { noise.conj() } else { noise }) * sigma(index / receptors.len());
166        }
167    }
168
169    pub(crate) fn table_rows(&self) -> u64 {
170        (self.chunks * self.antennas * (self.antennas + 1) / 2) as u64
171    }
172    pub(crate) fn table_values(&self) -> u64 {
173        self.values.len() as u64
174    }
175
176    pub(crate) fn solutions(&self) -> impl Iterator<Item = SimulatedNoiseSolution> + '_ {
177        self.windows
178            .iter()
179            .enumerate()
180            .flat_map(move |(id, window)| {
181                window.chunks.iter().flat_map(move |chunk| {
182                    (0..self.antennas).flat_map(move |p| {
183                        (p..self.antennas).map(move |q| {
184                            let offset = chunk.offset + self.baseline(p, q) * window.channels * 4;
185                            SimulatedNoiseSolution {
186                                time_s: chunk.time,
187                                field_id: chunk.field,
188                                spectral_window_id: id as i32,
189                                antennas: [p as i32, q as i32],
190                                values: self.values[offset..offset + window.channels * 4]
191                                    .as_chunks::<4>()
192                                    .0
193                                    .to_vec(),
194                            }
195                        })
196                    })
197                })
198            })
199    }
