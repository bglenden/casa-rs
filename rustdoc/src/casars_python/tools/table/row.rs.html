40impl RowHandle {
41    fn ensure_alive(&self) -> Result<(), Fail> {
42        if self.alive.load(Ordering::Acquire) {
43            Ok(())
44        } else {
45            Err(Fail::Runtime("use of uninitialized table row".into()))
46        }
47    }
48
49    fn view_nrows(&self, root: &RootOpen) -> usize {
50        self.row_map
51            .as_ref()
52            .map_or_else(|| root.table.row_count(), Vec::len)
53    }
54}
55
56fn runtime_err(message: impl Into<String>) -> Fail {
57    Fail::Runtime(message.into())
58}

60fn undefined_row_array(schema: &ColumnSchema, contract: &ArrayShapeContract) -> CasaValue {
61    let shape = match contract {
62        ArrayShapeContract::Fixed { shape } => vec![0; shape.len().max(1)],
63        ArrayShapeContract::Variable { ndim: Some(ndim) } => vec![0; (*ndim).max(1)],
64        ArrayShapeContract::Variable { ndim: None } => vec![0],
65    };
66    values::empty_array_of(schema.data_type(), shape)
67}

69fn read_row_dict(root: &RootOpen, handle: &RowHandle, view_row: usize) -> Result<CasaValue, Fail> {
70    let root_row = handle
71        .row_map
72        .as_ref()
73        .map_or(view_row, |map| map[view_row]);
74    let mut fields = Vec::with_capacity(handle.columns.len());
75    for col in &handle.columns {
76        let schema = root
77            .table
78            .schema()
79            .and_then(|s| s.column(col))
80            .ok_or_else(|| runtime_err(format!("Table column {col} is unknown")))?;
81        let val = match schema.column_type() {
82            ColumnType::Scalar => {
83                let scalar = scalar_cells(&root.table, col, &[root_row])?.remove(0);
84                CasaValue::from_scalar(&scalar)
85            }
86            ColumnType::Record => {
87                let record = root.table.column_accessor(col)?.record_cell(root_row)?;
88                values::record(&record, &root.name)
89            }
90            ColumnType::Array(contract) => match root
91                .table
92                .column_accessor(col)?
93                .array_cells_owned_uncached(&[root_row])?
94                .into_iter()
95                .next()
96                .flatten()
97            {
98                Some(array) => values::array(&array),
99                None => undefined_row_array(schema, contract),
100            },
101        };
102        fields.push((col.clone(), val));
103    }
104    Ok(CasaValue::record(fields))
105}

107fn strict_scalar_error(ty: PrimitiveType, field_name: &str) -> Fail {
108    let method = match ty {
109        PrimitiveType::Bool => "asBool",
110        PrimitiveType::UInt8
111        | PrimitiveType::UInt16
112        | PrimitiveType::UInt32
113        | PrimitiveType::Int16
114        | PrimitiveType::Int32 => "asInt",
115        PrimitiveType::Int64 => "asInt64",
116        PrimitiveType::Float32 => "asFloat",
117        PrimitiveType::Float64 => "asDouble",
118        PrimitiveType::Complex32 => "asComplex",
119        PrimitiveType::Complex64 => "asDComplex",
120        PrimitiveType::String => {
121            return runtime_err(format!(
122                "TableRecordRep::get_pointer - incorrect data type used for field {field_name}"
123            ));
124        }
125    };
126    runtime_err(format!("RecordInterface::{method} - invalid data type"))
127}

129fn data_compatible(ty: PrimitiveType, data: &Data) -> bool {
130    match ty {
131        PrimitiveType::Bool => matches!(data, Data::Bools(_) | Data::Ints(_) | Data::Floats(_)),
132        PrimitiveType::UInt8
133        | PrimitiveType::UInt16
134        | PrimitiveType::UInt32
135        | PrimitiveType::Int16
136        | PrimitiveType::Int32
137        | PrimitiveType::Int64 => matches!(data, Data::Ints(_)),
138        PrimitiveType::Float32 | PrimitiveType::Float64 => {
139            matches!(data, Data::Ints(_) | Data::Floats(_))
140        }
141        PrimitiveType::Complex32 | PrimitiveType::Complex64 => {
142            matches!(data, Data::Ints(_) | Data::Floats(_) | Data::Complexes(_))
143        }
144        PrimitiveType::String => matches!(data, Data::Texts(_)),
145    }
146}

148fn strict_into_cell(
149    field_name: &str,
150    incoming: Incoming,
151    schema: &ColumnSchema,
152) -> Result<Value, Fail> {
153    match (schema.column_type(), schema.data_type()) {
154        (ColumnType::Record, _) => match incoming {
155            Incoming::Record(_) => incoming.natural(),
156            Incoming::Wide(_) => Err(runtime_err(format!(
157                "TableRecordRep::get_pointer - incorrect data type used for field {field_name}"
158            ))),
159        },
160        (ColumnType::Scalar, Some(ty)) => match incoming {
161            Incoming::Record(_) => Err(strict_scalar_error(ty, field_name)),
162            Incoming::Wide(wide) => {
163                if !data_compatible(ty, &wide.data) {
164                    return Err(strict_scalar_error(ty, field_name));
165                }
166                wide.into_cell(ty, false)
167            }
168        },
169        (ColumnType::Array(_), Some(ty)) => match incoming {
170            Incoming::Record(_) => Err(runtime_err(format!(
171                "TableRecordRep::get_pointer - incorrect data type used for field {field_name}"
172            ))),
173            Incoming::Wide(wide) => {
174                if !data_compatible(ty, &wide.data) {
175                    return Err(runtime_err(format!(
176                        "TableRecordRep::get_pointer - incorrect data type used for field {field_name}"
177                    )));
178                }
179                wide.into_cell(ty, true)
180            }
181        },
182        (_, None) => Err(runtime_err(format!(
183            "TableRecordRep::get_pointer - incorrect data type used for field {field_name}"
184        ))),
185    }
186}

188fn check_matching_fields(fields: &[(String, Incoming)], columns: &[String]) -> Result<(), Fail> {
189    if fields.len() != columns.len() {
190        return Err(runtime_err("TableRow::put; names not conforming"));
191    }
192    for column in columns {
193        if !fields.iter().any(|(name, _)| name == column) {
194            return Err(runtime_err("TableRow::put; names not conforming"));
195        }
196    }
197    Ok(())
198}

200fn apply_row_put(
201    table: &mut Table,
202    row_map: Option<&[usize]>,
203    row: i64,
204    table_name: &str,
205    columns: &[String],
206    fields: Vec<(String, Incoming)>,
207    matching: bool,
208) -> Result<Persist, Fail> {
209    let nrows = row_map.map_or_else(|| table.row_count(), <[usize]>::len);
210    let view_row = usize::try_from(row)
211        .ok()
212        .filter(|&r| r < nrows)
213        .ok_or_else(|| {
214            runtime_err(format!(
215                "TableRow::put: row {} does not exist in table {table_name}",
216                row as u64
217            ))
218        })?;
219    if !matching {
220        check_matching_fields(&fields, columns)?;
221    }
222    let root_row = row_map.map_or(view_row, |map| map[view_row]);
223    let wanted: HashSet<&str> = columns.iter().map(String::as_str).collect();
224    let mut updates = Vec::new();
225    let mut updated_cols = Vec::new();
226    for (field_name, incoming) in fields {
227        if !wanted.contains(field_name.as_str()) {
228            continue;
229        }
230        let schema = table
231            .schema()
232            .and_then(|s| s.column(&field_name))
233            .ok_or_else(|| runtime_err(format!("Table column {field_name} is unknown")))?
234            .clone();
235        let cell = strict_into_cell(&field_name, incoming, &schema)?;
236        ShapeRule::of(table, &field_name).check(table, &field_name, root_row, &cell)?;
237        updated_cols.push(field_name.clone());
238        updates.push((field_name, cell));
239    }
240    if updates.is_empty() {
241        return Ok(Persist::Nothing);
242    }
243    for (col, cell) in updates {
244        set_cells(table, &col, vec![(root_row, cell)])?;
245    }
246    Ok(Persist::Rows {
247        columns: updated_cols,
248        rows: vec![root_row],
249    })
250}

252impl PyTableRowTool {
253    fn clone_handle(&self) -> Result<RowHandle, Fail> {
254        let guard = self.handle.lock().unwrap_or_else(PoisonError::into_inner);
255        let handle = guard
256            .as_ref()
257            .ok_or_else(|| runtime_err("use of uninitialized table row"))?;
258        handle.ensure_alive()?;
259        Ok(handle.clone())
260    }
261
262    fn read_normalized_index(&self, py: Python<'_>, idx: i64) -> PyResult<Py<PyAny>> {
263        let nrows = self.__len__(py)?;
264        let n_i64 = i64::try_from(nrows).unwrap_or(i64::MAX);
265        let normalized = if idx < 0 { idx + n_i64 } else { idx };
266        if normalized < 0 || normalized >= n_i64 {
267            return Err(PyIndexError::new_err(format!(
268                "row index {idx} out of range for table row with {nrows} rows"
269            )));
270        }
271        self.read_single_row(py, normalized)
272    }
273
274    fn read_single_row(&self, py: Python<'_>, row: i64) -> PyResult<Py<PyAny>> {
275        let mut lines = Vec::new();
276        let result = py.detach(|| -> Result<CasaValue, Fail> {
277            let handle = self.clone_handle()?;
278            let mut root = handle.root.lock().unwrap_or_else(PoisonError::into_inner);
279            synchronize_root(&mut root)?;
280            handle.ensure_alive()?;
281            let nrows = handle.view_nrows(&root);
282            let view_row = usize::try_from(row)
283                .ok()
284                .filter(|&r| r < nrows)
285                .ok_or_else(|| {
286                    let message = format!(
287                        "TableRow::get: row {} does not exist in table {}",
288                        row as u64,
289                        handle.name.display()
290                    );
291                    lines.push(LogLine::error(message.clone()));
292                    runtime_err(message)
293                })?;
294            read_row_dict(&root, &handle, view_row)
295        });
296        emit(py, LOGGER, &lines)?;
297        result?.into_python(py)
298    }
299
300    fn read_indexed_rows(&self, py: Python<'_>, indices: Vec<i64>) -> PyResult<Py<PyAny>> {
301        let dicts = py.detach(|| -> Result<Vec<CasaValue>, PyErr> {
302            let handle = self.clone_handle().map_err(PyErr::from)?;
303            let mut root = handle.root.lock().unwrap_or_else(PoisonError::into_inner);
304            synchronize_root(&mut root).map_err(PyErr::from)?;
305            handle.ensure_alive().map_err(PyErr::from)?;
306            let nrows = handle.view_nrows(&root);
307            let n_i64 = i64::try_from(nrows).unwrap_or(i64::MAX);
308            let mut out = Vec::with_capacity(indices.len());
309            for idx in indices {
310                let normalized = if idx < 0 { idx + n_i64 } else { idx };
311                let Some(view_row) = usize::try_from(normalized).ok().filter(|&r| r < nrows) else {
312                    return Err(PyIndexError::new_err(format!(
313                        "row index {idx} out of range for table row with {nrows} rows"
314                    )));
315                };
316                out.push(read_row_dict(&root, &handle, view_row).map_err(PyErr::from)?);
317            }
318            Ok(out)
319        })?;
320        let items = dicts
321            .into_iter()
322            .map(|d| d.into_python(py))
323            .collect::<PyResult<Vec<_>>>()?;
324        Ok(PyList::new(py, items)?.into_any().unbind())
325    }
326}

328pub(super) fn parse_columnnames(value: Option<&Bound<'_, PyAny>>) -> PyResult<Vec<String>> {
329    let Some(value) = value else {
330        return Ok(Vec::new());
331    };
332    if value.is_none() {
333        return Ok(Vec::new());
334    }
335    if let Ok(name) = value.extract::<String>() {
336        return Ok(vec![name]);
337    }
338    if let Ok(names) = value.extract::<Vec<String>>() {
339        return Ok(names);
340    }
341    if let Ok(list) = value.call_method0("tolist")
342        && let Ok(names) = list.extract::<Vec<String>>()
343    {
344        return Ok(names);
345    }
346    Err(PyTypeError::new_err(
347        "argument _columnnames must be a string or list of strings",
348    ))
349}

351/// Opens a [`PyTableRowTool`] from `state`, selecting `columnnames` (or
352/// excluding them when `exclude` is true).
353pub(super) fn create(
354    state: &mut State,
355    columnnames: Vec<String>,
356    exclude: bool,
357    log: &mut Vec<LogLine>,
358) -> Result<PyTableRowTool, Fail> {
359    let Some(open) = state.open.as_ref() else {
360        log.push(LogLine::error("Row access from unitialized table"));
361        return Err(runtime_err("row access from unitialized table"));
362    };
363    let root = open.root();
364    let schema = root
365        .table
366        .schema()
367        .ok_or_else(|| runtime_err("table has no schema"))?;
368    let schema_cols: Vec<String> = open
369        .visible_columns(&root.table)
370        .into_iter()
371        .map(|col| col.name().to_string())
372        .collect();
373    let selected = if exclude {
374        let excluded: HashSet<&str> = columnnames.iter().map(String::as_str).collect();
375        schema_cols
376            .into_iter()
377            .filter(|name| !excluded.contains(name.as_str()))
378            .collect()
379    } else if columnnames.is_empty() {
380        schema_cols
381    } else {
382        let mut seen = HashSet::with_capacity(columnnames.len());
383        for name in &columnnames {
384            if !open.column_allowed(name) || schema.column(name).is_none() {
385                return Err(runtime_err(format!("Table column {name} is unknown")));
386            }
387            if !seen.insert(name.as_str()) {
388                return Err(runtime_err(format!(
389                    "RecordDesc::addField() - field {name} already has been defined"
390                )));
391            }
392        }
393        columnnames
394    };
395    Ok(PyTableRowTool {
396        handle: Mutex::new(Some(RowHandle {
397            root: Arc::clone(&open.root),
398            alive: Arc::clone(&open.alive),
399            row_map: open.row_map.clone(),
400            name: open.name.clone(),
401            columns: selected,
402        })),
403    })
404}

408impl PyTableRowTool {
409    #[new]
410    #[pyo3(signature = (table = None, columnnames = None, exclude = None), text_signature = "(table=None, columnnames=[], exclude=False)")]
411    fn new(
412        py: Python<'_>,
413        table: Option<&Bound<'_, PyAny>>,
414        columnnames: Option<&Bound<'_, PyAny>>,
415        exclude: Option<&Bound<'_, PyAny>>,
416    ) -> PyResult<Self> {
417        let Some(table_arg) = table else {
418            return Err(PyRuntimeError::new_err(
419                "invalid table passed for parameter one",
420            ));
421        };
422        let Ok(py_table) = table_arg.extract::<PyRef<'_, PyTableTool>>() else {
423            return Err(PyRuntimeError::new_err(
424                "invalid table passed for parameter one",
425            ));
426        };
427        let cols = parse_columnnames(columnnames)?;
428        let excl = flag(exclude, false)?;
429        let state_arc = py_table.shared_state();
430        drop(py_table);
431        let mut lines = Vec::new();
432        let row = py.detach(|| {
433            let mut guard = state_arc.lock().unwrap_or_else(PoisonError::into_inner);
434            if guard.open.is_none() {
435                return Err(runtime_err("invalid table passed for parameter one"));
436            }
437            create(&mut guard, cols, excl, &mut lines)
438        });
439        emit(py, LOGGER, &lines)?;
440        Ok(row?)
441    }

472    /// Write column values from `value` to row `rownr`.
473    #[pyo3(signature = (rownr, value, matchingfields = None), text_signature = "($self, rownr, value, matchingfields=True)")]
474    fn put(
475        &self,
476        py: Python<'_>,
477        rownr: &Bound<'_, PyAny>,
478        value: &Bound<'_, PyAny>,
479        matchingfields: Option<&Bound<'_, PyAny>>,
480    ) -> PyResult<bool> {
481        let row = integer(Some(rownr), "rownr", 0)?;
482        let fields = record_fields(value)?;
483        let matching = flag(matchingfields, true)?;
484        let mut lines = Vec::new();
485        let result = py.detach(|| -> Result<(), Fail> {
486            let handle = self.clone_handle()?;
487            let table_name = handle.name.display().to_string();
488            let name_for_closure = table_name.clone();
489            let row_map = handle.row_map.clone();
490            let columns = handle.columns.clone();
491            let mut root = handle.root.lock().unwrap_or_else(PoisonError::into_inner);
492            handle.ensure_alive()?;
493            match write_root(
494                &mut root,
495                &table_name,
496                Target::RowProxy,
497                row_map.as_deref(),
498                move |table, row_map| {
499                    apply_row_put(
500                        table,
501                        row_map,
502                        row,
503                        &name_for_closure,
504                        &columns,
505                        fields,
506                        matching,
507                    )
508                },
509            ) {
510                Ok(_) => Ok(()),
511                Err(Fail::Runtime(message)) => {
512                    lines.push(LogLine::error(message.clone()));
513                    Err(Fail::Runtime(message))
514                }
515                Err(other) => Err(other),
516            }
517        });
518        emit(py, LOGGER, &lines)?;
519        result?;
520        Ok(true)
521    }

541    fn __len__(&self, py: Python<'_>) -> PyResult<usize> {
542        py.detach(|| -> Result<usize, Fail> {
543            let handle = self.clone_handle()?;
544            let root = handle.root.lock().unwrap_or_else(PoisonError::into_inner);

547        })

551    fn __getitem__(&self, py: Python<'_>, rownr: &Bound<'_, PyAny>) -> PyResult<Py<PyAny>> {
552        if !rownr.is_instance_of::<PyBool>()
553            && let Ok(idx) = rownr.extract::<i64>()
554        {
555            return self.read_normalized_index(py, idx);
556        }
557        if let Ok(slice) = rownr.cast::<PySlice>() {
558            let nrows = self.__len__(py)?;
559            let indices = slice.indices(isize::try_from(nrows).unwrap_or(isize::MAX))?;
560            let selected: Vec<i64> = (0..indices.slicelength)
561                .map(|step| (indices.start + (step as isize) * indices.step) as i64)
562                .collect();
563            return self.read_indexed_rows(py, selected);
564        }
565        if let Ok(indices) = rownr.extract::<Vec<i64>>() {
566            return self.read_indexed_rows(py, indices);
567        }
568        if let Ok(native) = rownr.call_method0("tolist") {
569            if !native.is_instance_of::<PyBool>()
570                && let Ok(idx) = native.extract::<i64>()
571            {
572                return self.read_normalized_index(py, idx);
573            }
574            if let Ok(indices) = native.extract::<Vec<i64>>() {
575                return self.read_indexed_rows(py, indices);
576            }
577        }
578        Err(PyTypeError::new_err(
579            "row indices must be integers, slices, or sequences of integers",
580        ))
581    }
