10#[derive(Clone, Copy)]
11pub(super) enum Method {
12    Clip,
13    Bosma,
14    GaussianWindow,
15    GaussianFit,
16}
17
18impl Method {
19    pub(super) fn new(
20        names: &[String],
21        clip: PixelClip,
22        smooth: bool,
23        moments: &[i32],
24    ) -> Result<Self, ImageError> {
25        let methods = names
26            .first()
27            .filter(|first| !first.is_empty())
28            .map_or_else(String::new, |_| names.join(" ").to_ascii_uppercase());
29        let window = methods.contains("WIN");
30        let fit = methods.contains("FIT");
31        let clipped = !matches!(clip, PixelClip::All);
32        if moments.contains(&4) && (smooth || window || fit || !clip.permits_median_coordinate()) {
33            return Err(ImageError::CasaSelection(
34                "The median coordinate moment requires basic reduction with a wholly positive or negative pixel range".into(),
35            ));
36        }
37        if ((window || fit) && clipped) || (smooth && !window && (fit || !clipped)) {
38            return Err(ImageError::CasaSelection(
39                "Invalid combination of methods requested".into(),
40            ));
41        }
42        Ok(match (window, fit) {
43            (false, false) => Self::Clip,
44            (true, false) => Self::Bosma,
45            (true, true) => Self::GaussianWindow,
46            (false, true) => Self::GaussianFit,
47        })
48    }
49}

51pub(super) enum Selection {
52    All,
53    None,
54    Window(RangeInclusive<usize>),
55    Model([f64; 4]),
56}
57
58impl Selection {
59    pub(super) fn new(
60        method: Method,
61        selection: &[f32],
62        valid: &[bool],
63        peak_snr: f32,
64        noise: f32,
65    ) -> Self {
66        if matches!(method, Method::Clip) {
67            return Self::All;
68        }
69        let usable = || {
70            selection
71                .iter()
72                .copied()
73                .zip(valid.iter().copied())
74                .enumerate()
75                .filter_map(|(index, (value, keep))| {
76                    (keep && value.is_finite()).then_some((index, value))
77                })
78        };
79        let Some((peak, maximum)) =
80            usable().reduce(|left, right| if left.1 < right.1 { right } else { left })
81        else {
82            return Self::None;
83        };
84        let minimum = usable().map(|(_, value)| value).fold(maximum, f32::min);
85        let signal = minimum.abs().max(maximum.abs());
86        if signal == 0. || signal / noise < peak_snr {
87            return Self::None;
88        }
89        if matches!(method, Method::Bosma) {
90            return bosma_window(selection.len(), peak, noise, usable)
91                .map_or(Self::None, Self::Window);
92        }
93        let (sum, count) = usable().fold((0., 0), |(sum, count), (_, value)| {
94            (sum + f64::from(value), count + 1)
95        });
96        let baseline = (sum / f64::from(count)) as f32;
97        let start = [f64::from(maximum - baseline), peak as f64, 5.];
98        let Some(parameters) = fit_gaussian_profile(
99            usable().map(|(index, value)| (index as f64, f64::from(value))),
100            start,
101            Some(f64::from(baseline)),
102        ) else {
103            return Self::None;
104        };
105        if matches!(method, Method::GaussianFit) {
106            return Self::Model(parameters);
107        }
108        let [_, center, fwhm, _] = parameters;
109        // CASA uses three times the fitted FWHM, despite calling this a sigma window.
110        let last = selection.len() - 1;
111        let limit = |position: f64| ((position + 0.5) as usize).min(last);
112        let lower = limit(center - 3. * fwhm);
113        let upper = limit(center + 3. * fwhm);
114        if upper.saturating_sub(lower) < 3 {
115            Self::None
116        } else {
117            Self::Window(lower..=upper)
118        }
119    }
120}

122fn bosma_window<S: Iterator<Item = (usize, f32)>>(
123    length: usize,
124    peak: usize,
125    noise: f32,
126    usable: impl Fn() -> S,
127) -> Option<RangeInclusive<usize>> {
128    let last = length - 1;
129    let mut lower = peak.saturating_sub(2);
130    let mut upper = (peak + 2).min(last);
131    let mut previous = None::<f32>;
132    loop {
133        let (sum, count) = usable()
134            .filter(|(index, _)| *index < lower || *index > upper)
135            .fold((0., 0_usize), |(sum, count), (_, value)| {
136                (sum + f64::from(value), count + 1)
137            });
138        if count > 0 {
139            let mean = (sum / count as f64) as f32;
140            let tolerance = noise / (length + 1 + lower - upper) as f32;
141            if previous.is_some_and(|previous| (previous - mean).abs() < tolerance) {
142                return Some(lower..=upper);
143            }
144            previous = Some(mean);
145        }
146        if lower == 0 && upper == last {
147            return None;
148        }
149        lower = lower.saturating_sub(2);
150        upper = (upper + 2).min(last);
151    }
152}
