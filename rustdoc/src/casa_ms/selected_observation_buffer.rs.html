12/// Exact MeasurementSet visibility column read for selected-observation input.
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14pub(crate) enum SelectedVisibilityColumn {
15    /// MAIN `DATA` complex visibility values.
16    Data,
17    /// MAIN `CORRECTED_DATA` complex visibility values.
18    CorrectedData,
19    /// MAIN `FLOAT_DATA` real visibility values.
20    FloatData,
21}
22
23impl SelectedVisibilityColumn {
24    const fn name(self) -> &'static str {
25        match self {
26            Self::Data => VisibilityDataColumn::Data.name(),
27            Self::CorrectedData => VisibilityDataColumn::CorrectedData.name(),
28            Self::FloatData => "FLOAT_DATA",
29        }
30    }
31}

117/// Heap residency of one selected-observation buffer and its fill operation.
118#[derive(Debug, Clone, Copy, PartialEq, Eq)]
119pub(crate) struct SelectedObservationBufferResidency {
120    /// Bytes retained after a successful fill.
121    pub(crate) resident_bytes: usize,
122    /// Peak bytes while the request and all fill temporaries are simultaneously live.
123    pub(crate) fill_peak_bytes: usize,
124}
125
126/// Project the allocations performed by [`MeasurementSet::fill_selected_observation_buffer`].
127pub(crate) fn selected_observation_buffer_residency(
128    rows: usize,
129    packed_samples: usize,
130    weight_values: usize,
131    visibility_bytes: usize,
132) -> Option<SelectedObservationBufferResidency> {
133    let row_indices = rows.checked_mul(size_of::<usize>())?;
134    let visibility = packed_samples.checked_mul(visibility_bytes)?;
135    let flags = packed_samples.checked_mul(size_of::<bool>())?;
136    let weights = weight_values.checked_mul(size_of::<f32>())?;
137    let scalar_payload =
138        rows.checked_mul(10 * size_of::<i32>() + 4 * size_of::<f64>() + size_of::<bool>())?;
139    let uvw = rows.checked_mul(size_of::<[f64; 3]>())?;
140    let resident_bytes = row_indices
141        .checked_add(visibility)?
142        .checked_add(flags)?
143        .checked_add(weights)?
144        .checked_add(scalar_payload)?
145        .checked_add(uvw)?;
146
147    // The typed scalar batch owns fifteen named vectors until they move into
148    // the destination buffer. Hashbrown keeps at most 7/8 load, so derive the
149    // allocated bucket count from the actual column count rather than hiding a
150    // fixture-specific allowance.
151    const SCALAR_COLUMNS: [&str; 15] = [
152        "DATA_DESC_ID",
153        "FIELD_ID",
154        "ANTENNA1",
155        "ANTENNA2",
156        "FEED1",
157        "FEED2",
158        "TIME",
159        "TIME_CENTROID",
160        "INTERVAL",
161        "EXPOSURE",
162        "SCAN_NUMBER",
163        "STATE_ID",
164        "OBSERVATION_ID",
165        "ARRAY_ID",
166        "FLAG_ROW",
167    ];
168    let minimum_buckets = SCALAR_COLUMNS
169        .len()
170        .checked_mul(8)?
171        .checked_add(6)?
172        .checked_div(7)?;
173    let scalar_buckets = minimum_buckets.checked_next_power_of_two()?;
174    let scalar_bucket_bytes = scalar_buckets.checked_mul(
175        size_of::<String>()
176            .checked_add(size_of::<RequiredScalarColumnValues>())?
177            .checked_add(1)?,
178    )?;
179    // casa-tables converts its internal typed map into the public typed map.
180    // The consumed source map keeps its bucket allocation until conversion
181    // ends, so both bucket arrays coexist even though String and Vec payloads
182    // move between them.
183    let scalar_map_bytes = scalar_bucket_bytes
184        .checked_mul(2)?
185        .checked_add(SCALAR_COLUMNS.iter().map(|name| name.len()).sum::<usize>())?;
186    // UVW conversion owns the typed flat input and the final [f64; 3] vector
187    // together; the final vector is already part of resident_bytes.
188    let uvw_conversion_input = rows.checked_mul(3 * size_of::<f64>())?;
189    let fill_peak_bytes = resident_bytes
190        .checked_add(row_indices)?
191        .checked_add(scalar_map_bytes)?
192        .checked_add(uvw_conversion_input)?;
193    Some(SelectedObservationBufferResidency {
194        resident_bytes,
195        fill_peak_bytes,
196    })
197}

228impl SelectedObservationBuffer {
229    /// Number of selected MAIN rows in this block.
230    #[must_use]
231    pub(crate) fn row_count(&self) -> usize {
232        self.row_indices.len()
233    }
234
235    /// Return one stored sample by channel, row, and correlation block offsets.
236    #[must_use]
237    pub(crate) fn sample(
238        &self,
239        channel_offset: usize,
240        row_offset: usize,
241        correlation_offset: usize,
242    ) -> Option<SelectedStoredSample> {
243        if channel_offset >= self.channel_range.count
244            || row_offset >= self.row_count()
245            || correlation_offset >= self.correlation_count
246        {
247            return None;
248        }
249        let sample_index = packed_sample_index(
250            channel_offset,
251            row_offset,
252            correlation_offset,
253            self.row_count(),
254            self.correlation_count,
255        );
256        let row_weight_index = row_offset * self.correlation_count + correlation_offset;
257        let visibility = match self.visibility.as_ref()? {
258            SelectedStoredVisibilities::Float32(values) => {
259                SelectedStoredVisibility::Float32(*values.get(sample_index)?)
260            }
261            SelectedStoredVisibilities::Complex32(values) => {
262                let value = values.get(sample_index)?;
263                SelectedStoredVisibility::Complex32([value.re, value.im])
264            }
265        };
266        let input_weight = match self.weights.as_ref()? {
267            SelectedStoredWeights::PerRow(values) => *values.get(row_weight_index)?,
268            SelectedStoredWeights::PerChannel(values) => *values.get(sample_index)?,
269        };
270        Some(SelectedStoredSample {
271            physical_row: *self.row_indices.get(row_offset)?,
272            data_description_id: *self.data_description_ids.get(row_offset)?,
273            visibility,
274            channel_flag: *self.flags.get(sample_index)?,
275            row_flag: *self.row_flag.get(row_offset)?,
276            input_weight,
277            uvw_m: *self.uvw_m.get(row_offset)?,
278            time_mjd_seconds: *self.time_mjd_seconds.get(row_offset)?,
279            time_centroid_mjd_seconds: *self.time_centroid_mjd_seconds.get(row_offset)?,
280            interval_seconds: *self.interval_seconds.get(row_offset)?,
281            exposure_seconds: *self.exposure_seconds.get(row_offset)?,
282            field_id: *self.field_ids.get(row_offset)?,
283            antenna1: *self.antenna1.get(row_offset)?,
284            antenna2: *self.antenna2.get(row_offset)?,
285            feed1: *self.feed1.get(row_offset)?,
286            feed2: *self.feed2.get(row_offset)?,
287            scan_number: *self.scan_numbers.get(row_offset)?,
288            state_id: *self.state_ids.get(row_offset)?,
289            observation_id: *self.observation_ids.get(row_offset)?,
290            array_id: *self.array_ids.get(row_offset)?,
291        })
292    }
293}

403impl MeasurementSet {
404    /// Fill one bounded block with the closed selected-observation storage column set.
405    ///
406    /// This selected-channel operation requires a lazily reopened disk-backed MeasurementSet with
407    /// no pending array-cell writes. It never falls back to cloning complete array cells.
408    pub(crate) fn fill_selected_observation_buffer(
409        &self,
410        request: &SelectedObservationBufferRequest,
411        buffer: &mut SelectedObservationBuffer,
412    ) -> MsResult<()> {
413        validate_request(self, request)?;
414        *buffer = SelectedObservationBuffer::default();
415        let visibility_cells = read_required_2d(
416            self,
417            request.visibility.name(),
418            &request.row_indices,
419            request.channel_range,
420        )?;
421        let (visibility, visibility_rows, visibility_channels, correlation_count) =
422            match (request.visibility, visibility_cells) {
423                (SelectedVisibilityColumn::FloatData, SelectedArray2DCells::Float32(values)) => {
424                    let shape = (
425                        values.row_count(),
426                        values.channel_count(),
427                        values.axis0_count(),
428                    );
429                    (
430                        SelectedStoredVisibilities::Float32(values.into_values()),
431                        shape.0,
432                        shape.1,
433                        shape.2,
434                    )
435                }
436                (
437                    SelectedVisibilityColumn::Data | SelectedVisibilityColumn::CorrectedData,
438                    SelectedArray2DCells::Complex32(values),
439                ) => {
440                    let shape = (
441                        values.row_count(),
442                        values.channel_count(),
443                        values.axis0_count(),
444                    );
445                    (
446                        SelectedStoredVisibilities::Complex32(values.into_values()),
447                        shape.0,
448                        shape.1,
449                        shape.2,
450                    )
451                }
452                (SelectedVisibilityColumn::FloatData, other) => {
453                    return Err(column_type_error(
454                        request.visibility.name(),
455                        "Float32 2-D array",
456                        &other,
457                    ));
458                }
459                (_, other) => {
460                    return Err(column_type_error(
461                        request.visibility.name(),
462                        "Complex32 2-D array",
463                        &other,
464                    ));
465                }
466            };
467        require_shape(
468            request.visibility.name(),
469            visibility_rows,
470            visibility_channels,
471            correlation_count,
472            request,
473            correlation_count,
474        )?;
475        let flag_cells =
476            read_required_2d(self, "FLAG", &request.row_indices, request.channel_range)?;
477        let SelectedArray2DCells::Bool(flags) = flag_cells else {
478            return Err(MsError::ColumnTypeMismatch {
479                column: "FLAG".to_string(),
480                table: "MAIN".to_string(),
481                expected: "Bool 2-D array".to_string(),
482                found: "different array primitive".to_string(),
483            });
484        };
485        require_shape(
486            "FLAG",
487            flags.row_count(),
488            flags.channel_count(),
489            flags.axis0_count(),
490            request,
491            correlation_count,
492        )?;
493        let flags = flags.into_values();
494        let weights = read_weights(self, request, correlation_count)?;
495        let mut scalars = self.main_table().required_scalar_columns_owned_for_rows(
496            &[
497                "DATA_DESC_ID",
498                "FIELD_ID",
499                "ANTENNA1",
500                "ANTENNA2",
501                "FEED1",
502                "FEED2",
503                "TIME",
504                "TIME_CENTROID",
505                "INTERVAL",
506                "EXPOSURE",
507                "SCAN_NUMBER",
508                "STATE_ID",
509                "OBSERVATION_ID",
510                "ARRAY_ID",
511                "FLAG_ROW",
512            ],
513            &request.row_indices,
514        )?;
515        let uvw_m = read_strict_uvw(self, &request.row_indices)?;
516        *buffer = SelectedObservationBuffer {
517            row_indices: request.row_indices.clone(),
518            channel_range: request.channel_range,
519            correlation_count,
520            visibility: Some(visibility),
521            flags,
522            weights: Some(weights),
523            row_flag: take_bool(&mut scalars, "FLAG_ROW")?,
524            uvw_m,
525            data_description_ids: take_i32(&mut scalars, "DATA_DESC_ID")?,
526            field_ids: take_i32(&mut scalars, "FIELD_ID")?,
527            antenna1: take_i32(&mut scalars, "ANTENNA1")?,
528            antenna2: take_i32(&mut scalars, "ANTENNA2")?,
529            feed1: take_i32(&mut scalars, "FEED1")?,
530            feed2: take_i32(&mut scalars, "FEED2")?,
531            time_mjd_seconds: take_f64(&mut scalars, "TIME")?,
532            time_centroid_mjd_seconds: take_f64(&mut scalars, "TIME_CENTROID")?,
533            interval_seconds: take_f64(&mut scalars, "INTERVAL")?,
534            exposure_seconds: take_f64(&mut scalars, "EXPOSURE")?,
535            scan_numbers: take_i32(&mut scalars, "SCAN_NUMBER")?,
536            state_ids: take_i32(&mut scalars, "STATE_ID")?,
537            observation_ids: take_i32(&mut scalars, "OBSERVATION_ID")?,
538            array_ids: take_i32(&mut scalars, "ARRAY_ID")?,
539        };
540        validate_buffer_lengths(buffer)?;
541        Ok(())
542    }
543}

545fn validate_request(
546    ms: &MeasurementSet,
547    request: &SelectedObservationBufferRequest,
548) -> MsResult<()> {
549    if request.row_indices.is_empty() {
550        return Err(invalid(
551            "selected-observation buffer requires at least one row",
552        ));
553    }
554    if request.channel_range.count == 0 {
555        return Err(invalid(
556            "selected-observation buffer requires at least one channel",
557        ));
558    }
559    if request.row_indices.iter().any(|row| *row >= ms.row_count()) {
560        return Err(invalid("selected-observation buffer row lies outside MAIN"));
561    }
562    Ok(())
563}
564
565fn read_required_2d(
566    ms: &MeasurementSet,
567    column: &str,
568    rows: &[usize],
569    channels: VisibilityChannelReadRange,
570) -> MsResult<SelectedArray2DCells> {
571    ms.main_table()
572        .column_accessor(column)?
573        .array_cells_2d_channel_range_typed_uncached(rows, channels.start, channels.count)?
574        .ok_or_else(|| invalid(format!("required MAIN {column} cells are undefined")))
575}
576
577fn read_weights(
578    ms: &MeasurementSet,
579    request: &SelectedObservationBufferRequest,
580    correlation_count: usize,
581) -> MsResult<SelectedStoredWeights> {
582    match request.weight {
583        SelectedWeightColumn::Weight => {
584            let cells = ms
585                .main_table()
586                .column_accessor("WEIGHT")?
587                .array_cells_1d_typed_uncached(&request.row_indices)?;
588            let SelectedArray1DCells::Float32(values) = cells else {
589                return Err(invalid("MAIN WEIGHT must be a Float32 1-D array"));
590            };
591            if values.row_count() != request.row_indices.len()
592                || values.axis0_count() != correlation_count
593            {
594                return Err(invalid(
595                    "MAIN WEIGHT shape differs from selected visibility shape",
596                ));
597            }
598            Ok(SelectedStoredWeights::PerRow(values.into_values()))
599        }
600        SelectedWeightColumn::WeightSpectrum => {
601            let cells = read_required_2d(
602                ms,
603                "WEIGHT_SPECTRUM",
604                &request.row_indices,
605                request.channel_range,
606            )?;
607            let SelectedArray2DCells::Float32(values) = cells else {
608                return Err(invalid("MAIN WEIGHT_SPECTRUM must be a Float32 2-D array"));
609            };
610            require_shape(
611                "WEIGHT_SPECTRUM",
612                values.row_count(),
613                values.channel_count(),
614                values.axis0_count(),
615                request,
616                correlation_count,
617            )?;
618            Ok(SelectedStoredWeights::PerChannel(values.into_values()))
619        }
620    }
621}
622
623fn require_shape(
624    column: &str,
625    rows: usize,
626    channels: usize,
627    correlations: usize,
628    request: &SelectedObservationBufferRequest,
629    expected_correlations: usize,
630) -> MsResult<()> {
631    if rows != request.row_indices.len()
632        || channels != request.channel_range.count
633        || correlations != expected_correlations
634    {
635        return Err(invalid(format!(
636            "MAIN {column} shape differs from selected visibility shape"
637        )));
638    }
639    Ok(())
640}

642fn read_strict_uvw(ms: &MeasurementSet, rows: &[usize]) -> MsResult<Vec<[f64; 3]>> {
643    let cells = ms
644        .main_table()
645        .column_accessor("UVW")?
646        .array_cells_1d_typed_uncached(rows)?;
647    let SelectedArray1DCells::Float64(values) = cells else {
648        return Err(invalid("MAIN UVW must be a Float64 1-D array"));
649    };
650    if values.row_count() != rows.len() || values.axis0_count() != 3 {
651        return Err(invalid("MAIN UVW shape must be exactly [row][3]"));
652    }
653    Ok(values
654        .into_values()
655        .chunks_exact(3)
656        .map(|uvw| [uvw[0], uvw[1], uvw[2]])
657        .collect())
658}

660fn validate_buffer_lengths(buffer: &SelectedObservationBuffer) -> MsResult<()> {
661    let rows = buffer.row_count();
662    let lengths = [
663        buffer.row_flag.len(),
664        buffer.uvw_m.len(),
665        buffer.data_description_ids.len(),
666        buffer.field_ids.len(),
667        buffer.antenna1.len(),
668        buffer.antenna2.len(),
669        buffer.feed1.len(),
670        buffer.feed2.len(),
671        buffer.time_mjd_seconds.len(),
672        buffer.time_centroid_mjd_seconds.len(),
673        buffer.interval_seconds.len(),
674        buffer.exposure_seconds.len(),
675        buffer.scan_numbers.len(),
676        buffer.state_ids.len(),
677        buffer.observation_ids.len(),
678        buffer.array_ids.len(),
679    ];
680    if lengths.into_iter().any(|length| length != rows) {
681        return Err(invalid("selected-observation scalar column lengths differ"));
682    }
683    Ok(())
684}

882    fn add_row(ms: &mut MeasurementSet, row_id: i32) {
883        let fields = ms
884            .main_table()
885            .schema()
886            .unwrap()
887            .columns()
888            .iter()
889            .map(|column| {
890                let value = match column.name() {
891                    "DATA" => Value::Array(ArrayValue::Complex32(
892                        ArrayD::from_shape_vec(vec![2, 4], complex_row(row_id)).unwrap(),
893                    )),
894                    "FLOAT_DATA" => Value::Array(ArrayValue::Float32(
895                        ArrayD::from_shape_vec(vec![2, 4], float_row(row_id)).unwrap(),
896                    )),
897                    "FLAG" => Value::Array(ArrayValue::Bool(
898                        ArrayD::from_shape_vec(vec![2, 4], flag_values(row_id)).unwrap(),
899                    )),
900                    "WEIGHT" => Value::Array(ArrayValue::Float32(
901                        ArrayD::from_shape_vec(vec![2], vec![1.0, 2.0]).unwrap(),
902                    )),
903                    "WEIGHT_SPECTRUM" => Value::Array(ArrayValue::Float32(
904                        ArrayD::from_shape_vec(vec![2, 4], spectrum_weights(row_id)).unwrap(),
905                    )),
906                    "UVW" => Value::Array(ArrayValue::Float64(
907                        ArrayD::from_shape_vec(
908                            vec![3],
909                            vec![
910                                row_id as f64 * 100.0 + 1.0,
911                                row_id as f64 * 100.0 + 2.0,
912                                row_id as f64 * 100.0 + 3.0,
913                            ],
914                        )
915                        .unwrap(),
916                    )),
917                    "ANTENNA1" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 1)),
918                    "ANTENNA2" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 2)),
919                    "DATA_DESC_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 3)),
920                    "FIELD_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 4)),
921                    "ARRAY_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 5)),
922                    "OBSERVATION_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 6)),
923                    "SCAN_NUMBER" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 7)),
924                    "STATE_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 8)),
925                    "FEED1" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 9)),
926                    "FEED2" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 10)),
927                    "FLAG_ROW" => Value::Scalar(ScalarValue::Bool(row_id == 1)),
928                    "TIME" => Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 1.0)),
929                    "TIME_CENTROID" => {
930                        Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 2.0))
931                    }
932                    "INTERVAL" => {
933                        Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 10.0))
934                    }
935                    "EXPOSURE" => {
936                        Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 20.0))
937                    }
938                    _ => default_value(column.name()),
939                };
940                RecordField::new(column.name(), value)
941            })
942            .collect();
943        ms.main_table_mut()
944            .add_row(RecordValue::new(fields))
945            .unwrap();
946    }

948    fn complex_row(row_id: i32) -> Vec<Complex32> {
949        (0..2)
950            .flat_map(|corr| {
951                (0..4).map(move |channel| {
952                    let value = row_id as f32 * 100.0 + channel as f32 * 10.0 + corr as f32;
953                    Complex32::new(value, -value)
954                })
955            })
956            .collect()
957    }
958
959    fn float_row(row_id: i32) -> Vec<f32> {
960        (0..2)
961            .flat_map(|corr| {
962                (0..4)
963                    .map(move |channel| row_id as f32 * 100.0 + channel as f32 * 10.0 + corr as f32)
964            })
965            .collect()
966    }
967
968    fn flag_values(row_id: i32) -> Vec<bool> {
969        (0..2)
970            .flat_map(|corr| (0..4).map(move |channel| (row_id + channel + corr) % 2 == 0))
971            .collect()
972    }
973
974    fn spectrum_weights(row_id: i32) -> Vec<f32> {
975        (0..2)
976            .flat_map(|corr| {
977                (0..4).map(move |channel| {
978                    row_id as f32 * 100.0 + channel as f32 * 10.0 + corr as f32 + 0.5
979                })
980            })
981            .collect()
982    }
