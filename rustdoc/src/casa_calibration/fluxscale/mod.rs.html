29/// Request for a CASA flux-scale transfer.
30#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
31pub struct FluxScaleRequest {
32    /// Selection and MeasurementSet metadata used by the CASA task surface.
33    #[serde(flatten)]
34    pub options: FluxScaleOptions,
35    /// Input gain table produced by `gaincal`.
36    pub input_table: PathBuf,
37    /// Output flux-scaled gain table.
38    pub output_table: PathBuf,
39    /// Reference fields in CASA FIELD selection grammar.
40    pub reference_fields: Vec<String>,
41    /// Transfer field selectors. Empty means all solved non-reference fields.
42    pub transfer_fields: Vec<String>,
43    /// Optional spectral-window remapping from transfer SPW id to reference SPW id.
44    pub refspwmap: Vec<i32>,
45    /// Optional fractional threshold around the median gain amplitude.
46    pub gainthreshold: Option<f64>,
47    /// Emit only multiplicative correction factors instead of a fully scaled table.
48    pub incremental: bool,
49    /// Append compatible solutions to an existing flux table.
50    #[serde(default)]
51    pub append: bool,
52}

54/// Optional metadata and solution selections for flux-scale estimation.
55#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
56#[serde(default)]
57pub struct FluxScaleOptions {
58    /// CASA resolves reference/transfer field names against this MeasurementSet.
59    /// Native table-only callers may use the gain table's FIELD metadata.
60    pub measurement_set: Option<PathBuf>,
61    /// CASA antenna/baseline expression on the calibration table.
62    pub antenna: String,
63    /// CASA time expression on calibration solution times.
64    pub timerange: String,
65    /// CASA scan expression on calibration solution scan numbers.
66    pub scan: String,
67    /// Polynomial degree in log-frequency/log-flux space (CASA default 1).
68    pub fitorder: i32,
69    /// Optional text report, overwritten for each successful invocation.
70    pub listfile: Option<PathBuf>,
71    /// CASA 6.7.6's display hooks are empty; this requests the same no-plot behavior.
72    pub display: bool,
73}
74
75impl Default for FluxScaleOptions {
76    fn default() -> Self {
77        Self {
78            measurement_set: None,
79            antenna: String::new(),
80            timerange: String::new(),
81            scan: String::new(),
82            fitorder: 1,
83            listfile: None,
84            display: false,
85        }
86    }
87}

139/// Errors returned by flux-scale estimation and persistence.
140#[derive(Debug, Error)]
141pub enum FluxScaleError {
142    /// Staging or publishing the complete scientific product failed.
143    #[error(transparent)]
144    CalibrationTable(#[from] crate::table::CalibrationTableWriteError),
145    /// Opening or validating the input caltable failed.
146    #[error("failed to summarize input calibration table: {source}")]
147    SummarizeInput {
148        /// Underlying summary error.
149        #[source]
150        source: CalibrationTableError,
151    },
152
153    /// The input table is outside the first supported `fluxscale` surface.
154    #[error("unsupported input calibration table {path}: {reason}")]
155    UnsupportedInput {
156        /// Input path.
157        path: String,
158        /// Error context.
159        reason: String,
160    },
161
162    /// The request did not specify any reference fields.
163    #[error("fluxscale requires at least one reference field selector")]
164    MissingReferenceFields,
165
166    /// A field selector did not match any field rows.
167    #[error("field selector {selector:?} did not match any FIELD rows")]
168    UnresolvedFieldSelector {
169        /// Original selector.
170        selector: String,
171    },
172
173    /// The transfer field selection produced no solved non-reference fields.
174    #[error("fluxscale transfer field selection produced no solved non-reference fields")]
175    EmptyTransferSelection,
176
177    /// No usable gain samples were found for a required `(field, spw)` pair.
178    #[error("no usable gain samples for field={field_id} spw={spw_id}")]
179    MissingGainSamples {
180        /// Field id.
181        field_id: i32,
182        /// SPW id.
183        spw_id: i32,
184    },
185
186    /// The input gain amplitudes did not permit a stable scale factor.
187    #[error("unstable fluxscale factor for field={field_id} spw={spw_id}")]
188    UnstableScaleFactor {
189        /// Field id.
190        field_id: i32,
191        /// SPW id.
192        spw_id: i32,
193    },
194
195    /// An output path operation failed.
196    #[error("failed to prepare output path {path}: {reason}")]
197    PrepareOutput {
198        /// Output path.
199        path: String,
200        /// Error context.
201        reason: String,
202    },
203
204    /// Writing direct calibrater provenance to the attached MS failed.
205    #[error("failed to update fluxscale HISTORY: {0}")]
206    History(String),
207
208    /// Copying the caltable tree failed.
209    #[error("failed to copy calibration table {source_path} -> {destination_path}: {reason}")]
210    CopyTableTree {
211        /// Source path.
212        source_path: String,
213        /// Destination path.
214        destination_path: String,
215        /// Error context.
216        reason: String,
217    },
218
219    /// Opening or mutating the copied output table failed.
220    #[error("failed to mutate output calibration table {path}: {source}")]
221    MutateOutput {
222        /// Output path.
223        path: String,
224        /// Underlying table error.
225        #[source]
226        source: Box<TableError>,
227    },
228}

277pub(crate) fn fluxscale_tool(
278    request: &FluxScaleRequest,
279    dataset: &Path,
280) -> Result<FluxScaleReport, String> {
281    if request.output_table.as_os_str().is_empty() {
282        return Err("output fluxscaled caltable name must be specified!".into());
283    }
284    if request.output_table.exists() && !request.append {
285        return Err(format!(
286            "output caltable name, {} exists. Please specify a different caltable name",
287            request.output_table.display()
288        ));
289    }
290    if request.reference_fields.is_empty() {
291        return Err("A reference field must be specified!".into());
292    }
293    let failure = |error: FluxScaleError| {
294        tracing::error!(%error, "Direct fluxscale failed");
295        "Error in Calibrater::fluxscale.".to_owned()
296    };
297    let ms = casa_ms::MeasurementSet::open(dataset).map_err(|error| {
298        failure(FluxScaleError::UnsupportedInput {
299            path: dataset.display().to_string(),
300            reason: error.to_string(),
301        })
302    })?;
303    let subtable = |id| {
304        ms.subtable(id).ok_or_else(|| {
305            failure(FluxScaleError::UnsupportedInput {
306                path: dataset.display().to_string(),
307                reason: format!("MeasurementSet has no {id:?} table"),
308            })
309        })
310    };
311    if let Some(path) = &request.options.listfile {
312        let linked = ms
313            .subtable_ids()
314            .into_iter()
315            .filter_map(|id| ms.subtable(id).and_then(Table::path));
316        listing::validate_products(path, linked).map_err(failure)?;
317    }
318    let info = read_field_info(subtable(casa_ms::SubtableId::Field)?).map_err(failure)?;
319    let reference = resolve_field_selectors(&request.reference_fields, &info)
320        .map_err(|error| tool_field_error(error, &request.reference_fields))?;
321    let transfer = (!request.transfer_fields.is_empty())
322        .then(|| resolve_field_selectors(&request.transfer_fields, &info))
323        .transpose()
324        .map_err(|error| tool_field_error(error, &request.transfer_fields))?;
325    let history = subtable(casa_ms::SubtableId::History)?
326        .path()
327        .ok_or_else(|| failure(FluxScaleError::History("no on-disk HISTORY table".into())))?;
328    let windows = subtable(casa_ms::SubtableId::SpectralWindow)?;
329    let spw_names = (0..windows.row_count())
330        .map(|row| get_string(windows, row, "NAME"))
331        .collect::<Result<_, _>>()
332        .map_err(failure)?;
333    let mode = if request.append {
334        crate::output::CalibrationOutputMode::Append
335    } else {
336        crate::output::CalibrationOutputMode::Create
337    };
338    let fields = FluxScaleFields {
339        info,
340        reference,
341        transfer,
342    };
343    let mut report =
344        fluxscale_with_fields(request, mode, Some((fields, history))).map_err(failure)?;
345    report.spw_names = spw_names;
346    Ok(report)
347}
348
349fn tool_field_error(error: FluxScaleError, selectors: &[String]) -> String {
350    let FluxScaleError::UnsupportedInput { reason, .. } = &error else {
351        return error.to_string();
352    };
353    let selector = selectors.join(",");
354    if let Some((_, text)) = reason.split_once("field selector name ")
355        && let Some(quoted) = text.strip_suffix(" did not match any FIELD rows")
356    {
357        let name = quoted.trim_matches('"');
358        let position = selector
359            .find(name)
360            .map_or(selector.len(), |start| start + name.len());
361        return format!(
362            "Field Expression: No match found for name {quoted}\n(near char. {position} in string {selector:?})"
363        );
364    }
365    format!("Field Expression: {reason}")
366}

370pub fn fluxscale_with_output_mode(
371    request: &FluxScaleRequest,

377struct FluxScaleFields {
378    info: HashMap<i32, FieldInfo>,
379    reference: BTreeSet<i32>,
380    transfer: Option<BTreeSet<i32>>,

382
383fn fluxscale_with_fields(
384    request: &FluxScaleRequest,
385    mode: crate::output::CalibrationOutputMode,
386    tool: Option<(FluxScaleFields, &Path)>,
387) -> Result<FluxScaleReport, FluxScaleError> {
388    let input_phase = crate::timing::Phase::start("fluxscale_read");
389    validate_request(request)?;
390    let input = Table::open(TableOptions::new(&request.input_table)).map_err(|error| {
391        FluxScaleError::UnsupportedInput {
392            path: request.input_table.display().to_string(),
393            reason: error.to_string(),
394        }
395    })?;
396    let history_path = tool.as_ref().map(|(_, history)| *history);
397    let field_root = request
398        .options
399        .measurement_set
400        .as_deref()
401        .unwrap_or(&request.input_table);
402    if history_path.is_none()
403        && let Some(vis) = &request.options.measurement_set
404    {
405        casa_ms::MeasurementSet::open(vis).map_err(|error| FluxScaleError::UnsupportedInput {
406            path: vis.display().to_string(),
407            reason: error.to_string(),
408        })?;
409    }
410    let FluxScaleFields {
411        info: field_info,
412        reference: reference_field_ids,
413        transfer: transfer_selection,
414    } = if let Some((fields, _)) = tool {
415        fields
416    } else {
417        let info = load_field_info(field_root)?;
418        let reference = resolve_field_selectors(&request.reference_fields, &info)?;
419        let transfer = (!request.transfer_fields.is_empty())
420            .then(|| resolve_field_selectors(&request.transfer_fields, &info))
421            .transpose()?;
422        FluxScaleFields {
423            info,
424            reference,
425            transfer,
426        }
427    };
428    let info = input.info();
429    let calibration_table = info.table_type == crate::constants::TABLE_INFO_TYPE;
430    if let Some(path) = history_path
431        && calibration_table
432    {
433        history::record_tool_type(path, request, &info.sub_type)?;
434        if !matches!(info.sub_type.as_str(), "G Jones" | "T Jones") {
435            return Err(FluxScaleError::UnsupportedInput {
436                path: request.input_table.display().to_string(),
437                reason: format!("Type {} not supported in fluxscale", info.sub_type),
438            });
439        }
440    }
441    // The CLI batches successful task HISTORY. Preserve its initial entry on
442    // error as well. The direct tool writes each entry at its CASA phase.
443    let result = (|| {
444        validate_input_table(&input, &request.input_table)?;
445        let spw_info = load_spw_info(&request.input_table)?;
446        let rows = input::read(&input, &request.input_table)?;
447        let transfer_field_ids =
448            resolve_transfer_fields(transfer_selection.as_ref(), &reference_field_ids, &rows)?;
449        let selected_fields = reference_field_ids
450            .union(&transfer_field_ids)
451            .copied()
452            .collect();
453        let masks = selection::sample_masks(
454            &request.input_table,
455            &rows,
456            &request.options,
457            &selected_fields,
458        )?;
459
460        drop(input_phase);
461        let compute_phase = crate::timing::Phase::start("fluxscale_estimate");
462        let estimates = stats::derive_selected_estimates(
463            &rows,
464            &reference_field_ids,
465            &transfer_field_ids,
466            request.gainthreshold,
467            &request.refspwmap,
468            &masks,
469        )?;
470
471        drop(compute_phase);
472        let publish = || {
473            let _phase = crate::timing::Phase::start("fluxscale_write");
474            write_scaled_output(request, &input, &rows, &estimates, mode, &masks)
475        };
476        if history_path.is_none() {
477            publish()?;
478        }
479        let report = build_report(
480            request,
481            &transfer_field_ids,
482            &field_info,
483            &spw_info,
484            &estimates,
485            &rows,
486            &masks,
487        );
488        if let Some(path) = &request.options.listfile {
489            listing::write(path, request, &report)?;
490        }
491        if request.options.display {
492            tracing::debug!("CASA fluxscale display hooks do not produce a plot");
493        }
494        if let Some(path) = history_path {
495            history::record_tool_output(path, request)?;
496            if let Some(parent) = request.output_table.parent()
497                && !parent.as_os_str().is_empty()
498                && !parent.is_dir()
499            {
500                return Err(FluxScaleError::PrepareOutput {
501                    path: request.output_table.display().to_string(),
502                    reason: "output directory does not exist".into(),
503                });
504            }
505            publish()?;
506        }
507        Ok(report)
508    })();
509    if history_path.is_none() && result.is_err() && calibration_table {
510        history::record_failure(request, &info.sub_type);
511    }
512    result
513}

515fn validate_request(request: &FluxScaleRequest) -> Result<(), FluxScaleError> {
516    if request.reference_fields.is_empty() {
517        return Err(FluxScaleError::MissingReferenceFields);
518    }
519    listing::validate(request)?;
520    Ok(())
521}

523fn validate_input_table(table: &Table, path: &Path) -> Result<(), FluxScaleError> {
524    let keyword = |name| match table.keywords().get(name) {
525        Some(Value::Scalar(ScalarValue::String(value))) => Some(value.as_str()),
526        _ => None,
527    };
528    if table.info().table_type != crate::constants::TABLE_INFO_TYPE {
529        return Err(FluxScaleError::UnsupportedInput {
530            path: path.display().to_string(),
531            reason: format!("table.info type is {:?}", table.info().table_type),
532        });
533    }
534    if keyword("ParType") == Some("Float")
535        || !table
536            .schema()
537            .is_some_and(|schema| schema.contains_column(COL_CPARAM))
538    {
539        return Err(FluxScaleError::UnsupportedInput {
540            path: path.display().to_string(),
541            reason: "fluxscale currently supports only complex CPARAM tables".to_string(),
542        });
543    }
544    match keyword("VisCal") {
545        Some("G Jones") | Some("T Jones") => Ok(()),
546        other => Err(FluxScaleError::UnsupportedInput {
547            path: path.display().to_string(),
548            reason: format!("expected VisCal to be G Jones or T Jones, found {other:?}"),
549        }),
550    }
551}

553fn load_field_info(root: &Path) -> Result<HashMap<i32, FieldInfo>, FluxScaleError> {
554    let table = Table::open(TableOptions::new(root.join("FIELD"))).map_err(|source| {
555        FluxScaleError::MutateOutput {
556            path: root.join("FIELD").display().to_string(),
557            source: Box::new(source),
558        }
559    })?;
560    read_field_info(&table)
561}
562
563fn read_field_info(table: &Table) -> Result<HashMap<i32, FieldInfo>, FluxScaleError> {
564    let mut out = HashMap::new();
565    for row in 0..table.row_count() {
566        let name = get_string(table, row, "NAME")?;
567        let code = get_string(table, row, "CODE").ok();
568        let flagged = matches!(
569            table
570                .cell_accessor(row, "FLAG_ROW")
571                .and_then(|cell| cell.value()),
572            Ok(Some(Value::Scalar(ScalarValue::Bool(true))))
573        );
574        out.insert(
575            row as i32,
576            FieldInfo {
577                name,
578                code,
579                flagged,
580            },
581        );
582    }
583    Ok(out)
584}

586fn load_spw_info(root: &Path) -> Result<HashMap<i32, SpwInfo>, FluxScaleError> {
587    let table = Table::open(TableOptions::new(root.join("SPECTRAL_WINDOW"))).map_err(|source| {
588        FluxScaleError::MutateOutput {
589            path: root.join("SPECTRAL_WINDOW").display().to_string(),
590            source: Box::new(source),
591        }
592    })?;
593    let mut out = HashMap::new();
594    for row in 0..table.row_count() {
595        let ref_frequency_hz = get_ref_frequency(&table, row)?;
596        let name = get_string(&table, row, "NAME").unwrap_or_default();
597        out.insert(
598            row as i32,
599            SpwInfo {
600                name,
601                ref_frequency_hz,
602            },
603        );
604    }
605    Ok(out)
606}

608fn resolve_field_selectors(
609    selectors: &[String],
610    field_info: &HashMap<i32, FieldInfo>,
611) -> Result<BTreeSet<i32>, FluxScaleError> {
612    let fields = (0..field_info.len())
613        .map(|id| casa_ms::selection::FieldSelectorRow {
614            name: &field_info[&(id as i32)].name,
615            code: field_info[&(id as i32)].code.as_deref(),
616            flagged: field_info[&(id as i32)].flagged,
617        })
618        .collect::<Vec<_>>();
619    casa_ms::selection::resolve_field_selector_rows(&fields, &selectors.join(","))
620        .map(|ids| ids.into_iter().collect())
621        .map_err(|error| FluxScaleError::UnsupportedInput {
622            path: "reference/transfer".into(),
623            reason: error.to_string(),
624        })
625}
626
627fn resolve_transfer_fields(
628    selection: Option<&BTreeSet<i32>>,
629    reference_field_ids: &BTreeSet<i32>,
630    rows: &[GainSampleRow],
631) -> Result<BTreeSet<i32>, FluxScaleError> {
632    let solved_fields = rows.iter().map(|row| row.field_id).collect::<BTreeSet<_>>();
633    let mut transfer = if let Some(selection) = selection {
634        selection.clone()
635    } else {
636        solved_fields
637            .difference(reference_field_ids)
638            .copied()
639            .collect::<BTreeSet<_>>()
640    };
641    transfer.retain(|field_id| {
642        !reference_field_ids.contains(field_id) && solved_fields.contains(field_id)
643    });
644    if transfer.is_empty() {
645        return Err(FluxScaleError::EmptyTransferSelection);
646    }
647    Ok(transfer)
648}

650fn write_scaled_output(
651    request: &FluxScaleRequest,
652    input: &Table,
653    rows: &[GainSampleRow],
654    estimates: &HashMap<(i32, i32), FluxScaleEstimate>,
655    mode: crate::output::CalibrationOutputMode,
656    masks: &selection::SampleMasks,
657) -> Result<(), FluxScaleError> {
658    let input_root = &request.input_table;
659    let output_root = &request.output_table;
660    let incremental = request.incremental;
661    if crate::paths::overlap(input_root, output_root).map_err(|e| {
662        FluxScaleError::PrepareOutput {
663            path: output_root.display().to_string(),
664            reason: e.to_string(),
665        }
666    })? {
667        return Err(FluxScaleError::PrepareOutput {
668            path: output_root.display().to_string(),
669            reason: "input_table and output_table must differ".to_string(),
670        });
671    }
672    if mode == crate::output::CalibrationOutputMode::Create && output_root.exists() {
673        return Err(FluxScaleError::PrepareOutput {
674            path: output_root.display().to_string(),
675            reason: "fluxtable exists; please specify a different name or append=True".to_string(),
676        });
677    }
678    let target = crate::output::CalibrationOutput::prepare(output_root, mode)?;
679    crate::output_merge::preserve_subtables(input_root, &target.staging)?;
680    let sorted = !incremental
681        && mode != crate::output::CalibrationOutputMode::Append
682        && full_output::ordered(rows);
683    if sorted && full_output::save(input, input_root, &target.staging, rows, estimates)? {
684        mark_output_windows(&target.staging, rows, false)?;
685        target.commit()?;
686        return Ok(());
687    }
688    let mut table = if incremental {
689        empty_output(input)
690    } else {
691        input
692            .to_memory()
693            .map_err(|source| FluxScaleError::MutateOutput {
694                path: output_root.display().to_string(),
695                source: Box::new(source),
696            })?
697    };
698    for name in crate::constants::STANDARD_SUBTABLE_KEYWORDS {
699        table.keywords_mut().upsert(
700            *name,
701            Value::table_ref(casa_tables::strip_directory(
702                &target.staging.join(name),
703                &target.staging,
704            )),
705        );
706    }
707    if incremental {
708        let metadata = request
709            .options
710            .measurement_set
711            .as_deref()
712            .unwrap_or(input_root);
713        if let Some(vis) = &request.options.measurement_set {
714            table.keywords_mut().upsert(
715                "MSName",
716                Value::Scalar(ScalarValue::String(
717                    vis.file_name()
718                        .unwrap_or_default()
719                        .to_string_lossy()
720                        .into_owned(),
721                )),
722            );
723        }
724        incremental::copy_metadata(metadata, &target.staging)?;
725        incremental::populate(&mut table, &target.staging, rows, estimates, masks)?;
726    } else {
727        for row in rows {
728            let Some(estimate) = estimates.get(&(row.field_id, row.spw_id)) else {
729                continue;
730            };
731            let scaled = row.gains.mapv(|value| value * estimate.correction_factor);
732            table
733                .cell_accessor_mut(row.row_index, COL_CPARAM)
734                .and_then(|mut cell| cell.set(Value::Array(ArrayValue::Complex32(scaled))))
735                .map_err(|source| FluxScaleError::MutateOutput {
736                    path: output_root.display().to_string(),
737                    source: Box::new(source),
738                })?;
739        }
740    }
741    if !incremental && rows.len() != table.row_count() {
742        let selected = rows
743            .iter()
744            .map(|row| row.row_index)
745            .collect::<BTreeSet<_>>();
746        let removed = (0..table.row_count())
747            .filter(|index| !selected.contains(index))
748            .collect::<Vec<_>>();
749        table
750            .remove_rows(&removed)
751            .map_err(|source| FluxScaleError::MutateOutput {
752                path: output_root.display().to_string(),
753                source: Box::new(source),
754            })?;
755    }
756    mark_output_windows(&target.staging, rows, incremental)?;
757    if mode == crate::output::CalibrationOutputMode::Append && output_root.exists() {
758        let spw = crate::output_merge::merge_spw(output_root, &target.staging)?;
759        let mut old = crate::output_merge::open(output_root)?;
760        crate::output_merge::append(&mut old, &table, output_root)?;
761        crate::output_merge::preserve_subtables(output_root, &target.staging)?;
762        spw.save(TableOptions::new(target.staging.join("SPECTRAL_WINDOW")))
763            .map_err(|e| crate::output_merge::invalid(output_root, e.to_string()))?;
764        table = old;
765    }
766    if sorted {
767        table
768            .save(
769                TableOptions::new(&target.staging)
770                    .with_data_manager(casa_tables::DataManagerKind::StandardStMan),
771            )
772            .map_err(|error| crate::output_merge::invalid(output_root, error.to_string()))?;
773    } else {
774        crate::output_merge::save_sorted(&mut table, &target.staging)?;
775    }
776    target.commit()?;
777    Ok(())
778}

780fn empty_output(input: &Table) -> Table {
781    let mut table =
782        Table::with_schema_memory(input.schema().expect("validated gain schema").clone());
783    table.set_info(input.info().clone());
784    *table.keywords_mut() = input.keywords().clone();
785    for column in input.schema().expect("validated gain schema").columns() {
786        if let Some(keywords) = input.column_keywords(column.name()) {
787            table.set_column_keywords(column.name(), keywords.clone());
788        }
789    }
790    table
791}
792
793fn mark_output_windows(
794    stage: &Path,
795    rows: &[GainSampleRow],
796    incremental: bool,
797) -> Result<(), FluxScaleError> {
798    let path = stage.join("SPECTRAL_WINDOW");
799    let failed = |error: TableError| crate::output_merge::invalid(&path, error.to_string());
800    let mut table = Table::open(TableOptions::new(&path)).map_err(failed)?;
801    let used = rows.iter().map(|row| row.spw_id).collect::<BTreeSet<_>>();
802    for index in 0..table.row_count() {
803        table
804            .cell_accessor_mut(index, "FLAG_ROW")
805            .and_then(|mut cell| {
806                cell.set(Value::Scalar(ScalarValue::Bool(
807                    !incremental && !used.contains(&(index as i32)),
808                )))
809            })
810            .map_err(failed)?;
811    }
812    table.save(TableOptions::new(&path)).map_err(failed)?;
813    Ok(())
814}

816fn build_report(
817    request: &FluxScaleRequest,
818    transfer_field_ids: &BTreeSet<i32>,
819    field_info: &HashMap<i32, FieldInfo>,
820    spw_info: &HashMap<i32, SpwInfo>,
821    estimates: &HashMap<(i32, i32), FluxScaleEstimate>,
822    rows: &[GainSampleRow],
823    masks: &selection::SampleMasks,
824) -> FluxScaleReport {
825    let mut fields = BTreeMap::new();
826    let mut spw_ids = BTreeSet::new();
827
828    for field_id in transfer_field_ids {
829        // The public CASA result omits a transfer field with no successful
830        // window, while retaining its unscaled rows in the output product.
831        if !estimates.keys().any(|(field, _)| field == field_id) {
832            continue;
833        }
834        let mut spw_results = BTreeMap::new();
835        let mut windows = Vec::new();
836        for &spw_id in spw_info.keys() {
837            spw_ids.insert(spw_id);
838            let Some(estimate) = estimates.get(&(*field_id, spw_id)) else {
839                spw_results.insert(
840                    spw_id,
841                    FluxScaleSpwResult {
842                        fluxd: [-1.; 4],
843                        fluxd_err: [-1.; 4],
844                        num_sol: [-1.; 4],
845                    },
846                );
847                continue;
848            };
849            spw_results.insert(
850                spw_id,
851                FluxScaleSpwResult {
852                    fluxd: [estimate.flux_density_jy, 0.0, 0.0, 0.0],
853                    fluxd_err: [estimate.flux_density_error_jy, 0.0, 0.0, 0.0],
854                    num_sol: [estimate.num_sol as f64, 0.0, 0.0, 0.0],
855                },
856            );
857            windows.push(fit::WindowFlux {
858                frequency_hz: spw_info
859                    .get(&spw_id)
860                    .map(|info| info.ref_frequency_hz)
861                    .unwrap_or(0.0),
862                flux_density_jy: estimate.flux_density_jy,
863                flux_density_error_jy: estimate.flux_density_error_jy,
864            });
865        }
866        windows.sort_by(|a, b| a.frequency_hz.total_cmp(&b.frequency_hz));
867        let fit = fit::spectral_fit_order(&windows, request.options.fitorder);
868
869        fields.insert(
870            *field_id,
871            FluxScaleFieldResult {
872                field_name: field_info
873                    .get(field_id)
874                    .map(|info| info.name.clone())
875                    .unwrap_or_default(),
876                spw_results,
877                fit_ref_frequency_hz: fit.ref_frequency_hz,
878                fit_fluxd: fit.fluxd,
879                fit_fluxd_err: fit.fluxd_err,
880                spidx: fit.spidx,
881                spidx_err: fit.spidx_err,
882                covar_mat: fit.covariance,
883            },
884        );
885    }
886
887    let sorted_spw_ids = spw_ids.into_iter().collect::<Vec<_>>();
888    let spw_names = sorted_spw_ids
889        .iter()
890        .map(|spw_id| {
891            spw_info
892                .get(spw_id)
893                .map(|info| info.name.clone())
894                .unwrap_or_default()
895        })
896        .collect();
897    let frequencies_hz = sorted_spw_ids
898        .iter()
899        .map(|spw_id| {
900            if !rows
901                .iter()
902                .zip(&masks.logical)
903                .any(|(row, selected)| *selected && row.spw_id == *spw_id)
904            {
905                return -1.;
906            }
907            spw_info
908                .get(spw_id)
909                .map(|info| info.ref_frequency_hz)
910                .unwrap_or(0.0)
911        })
912        .collect();
913
914    FluxScaleReport {
915        output_table: request.output_table.clone(),
916        spw_ids: sorted_spw_ids,
917        spw_names,
918        frequencies_hz,
919        fields,
920    }
921}

923fn get_string(table: &Table, row: usize, column: &str) -> Result<String, FluxScaleError> {
924    match table
925        .cell_accessor(row, column)
926        .and_then(|cell| cell.value())
927        .map_err(|source| FluxScaleError::MutateOutput {
928            path: column.to_string(),
929            source: Box::new(source),
930        })? {
931        Some(Value::Scalar(ScalarValue::String(value))) => Ok(value.clone()),
932        Some(other) => Err(FluxScaleError::UnsupportedInput {
933            path: column.to_string(),
934            reason: format!("expected String in {column}, found {:?}", other.kind()),
935        }),
936        None => Err(FluxScaleError::UnsupportedInput {
937            path: column.to_string(),
938            reason: format!("missing required column {column}"),
939        }),
940    }
941}

943fn get_ref_frequency(table: &Table, row: usize) -> Result<f64, FluxScaleError> {
944    let chan_freq = table
945        .cell_accessor(row, "CHAN_FREQ")
946        .and_then(|cell| cell.array())
947        .map_err(|source| FluxScaleError::MutateOutput {
948            path: "CHAN_FREQ".to_string(),
949            source: Box::new(source),
950        })?;
951    // CASA's solution frequency, `solFreq`, is the window's middle channel.
952    match chan_freq {
953        ArrayValue::Float64(values) if !values.is_empty() => {
954            Ok(values.iter().copied().nth(values.len() / 2).unwrap_or(0.0))
955        }
956        _ => match table
957            .cell_accessor(row, "REF_FREQUENCY")
958            .and_then(|cell| cell.value())
959            .map_err(|source| FluxScaleError::MutateOutput {
960                path: "REF_FREQUENCY".to_string(),
961                source: Box::new(source),
962            })? {
963            Some(Value::Scalar(ScalarValue::Float64(value))) => Ok(*value),
964            Some(Value::Array(ArrayValue::Float64(values))) if values.len() == 1 => {
965                Ok(values.iter().copied().next().unwrap_or(0.0))
966            }
967            Some(_) | None => Ok(0.0),
968        },
969    }
