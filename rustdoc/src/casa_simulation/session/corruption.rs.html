2//! `sm.setseed`, `sm.setnoise`, `sm.corrupt` and `sm.reset`: thermal noise.
3//!
4//! As in CASA's `Simulator`:
5//! - `setnoise` adds a noise term with the seed in force at the call; terms
6//!   accumulate, duplicates included, until `reset`.
7//! - `corrupt` applies every term to the rows observed so far. Scans
8//!   observed later are not corrupted by it.
9//! - `reset` removes the terms. Corruptions already applied stay, as they
10//!   are already in CASA's DATA; the seed stays too.
11//!
12//! The draws are not CASA's (the crate's `corrupt` module describes the
13//! generator and what is statistically the same). Two terms set with the same seed add
14//! coherently in one `corrupt`, as CASA's identical streams do, and a term
15//! applied again draws afresh.
16//!
17//! `setnoise(mode='simplenoise')` gives every draw one σ;
18//! `mode='tsys-manual'` gives each row the σ of a system-temperature model
19//! (the `corrupt` module's `tsys`). Each term adds its own noise: CASA
20//! instead scales the noise of every term by every `tsys-manual` term's
21//! σ, recorded with casa-rs's other departures in
22//! `docs/CASA (C++) bugs.md`.
23

30/// CASA's `Simulator` seed until `setseed` (`seed_p(11111)`). The Python

34/// A precipitable water vapour above this, mm, is refused: CASA takes it
35/// for a value given without units, which default to metres.
36const LARGEST_WATER_VAPOUR_MM: f64 = 100.0;
37
38/// A zenith Tsys above this, K, or not positive, is refused
39/// (`AtmosCorruptor::initialize`).
40const LARGEST_ZENITH_TSYS_K: f32 = 1e6;
41
42/// The arguments of `sm.setnoise` that casa-rs reads.
43#[derive(Debug, Clone, PartialEq)]
44pub struct NoiseSpec {
45    /// `mode`: `simplenoise` and `tsys-manual` are implemented; `tsys-atm`
46    /// is a later phase's (#841).
47    pub mode: String,
48    /// `simplenoise`: σ of each real and imaginary draw, Jy.
49    pub simple_noise_jy: f64,
50    /// `table`: the calibration table's name. CASA writes no table for a
51    /// `simplenoise` term set with `OTF`; for `tsys-manual` it writes the
52    /// `T` term's to `<table>.T.cal`, which casa-rs does not yet.
53    pub table: String,
54    /// `pwv`, mm: in the Tsys modes, a value above 100 mm is refused, as
55    /// CASA refuses it (`tsys-manual` uses no water vapour).
56    pub water_vapour_mm: f64,
57    /// `tsys-manual`'s atmosphere, antenna and receiver parameters.
58    pub tsys: TsysParameters,
59    /// `OTF`: corrupt on the fly. CASA writes no calibration table then,
60    /// whatever `table` says; without it, it writes one, which casa-rs does
61    /// not yet.
62    pub on_the_fly: bool,
63}
64
65impl Default for NoiseSpec {
66    /// `simplenoise` of 0 Jy with CASA's other `setnoise` defaults.
67    fn default() -> Self {
68        Self {
69            mode: "simplenoise".to_string(),
70            simple_noise_jy: 0.0,
71            table: String::new(),
72            water_vapour_mm: 0.0,
73            tsys: TsysParameters::default(),
74            on_the_fly: true,
75        }
76    }
77}

100impl Simulator {
101    /// `sm.setseed(seed)`: the seed of noise terms set from now on.
102    pub fn set_seed(&mut self, seed: i32) {
103        self.seed = seed;
104    }
105
106    /// `sm.setnoise(...)`: add a noise term with the current seed.
107    ///
108    /// # Errors
109    ///
110    /// - A mode CASA does not know: CASA logs "unsupported mode … in
111    ///   setnoise()" and returns `False`.
112    /// - `tsys-atm`, or `OTF=False`: not yet implemented; so is
113    ///   `tsys-manual` with a `table`, which CASA writes.
114    /// - No rows observed yet: CASA cannot size the term on an empty
115    ///   MeasurementSet and returns `False`, adding no term.
116    /// - `tsys-manual` with a water vapour above 100 mm, or a zenith Tsys
117    ///   that is not positive or above 10⁶ K: CASA returns `False`. It has
118    ///   added a unit-σ noise term by then, which casa-rs does not
119    ///   (`docs/CASA (C++) bugs.md`).
120    pub fn set_noise(&mut self, spec: &NoiseSpec) -> SimulatorResult<()> {
121        let tsys = match spec.mode.as_str() {
122            "simplenoise" => false,
123            "tsys-manual" => true,
124            "tsys-atm" => {
125                return Err(SimulatorError::not_yet_implemented(
126                    "setnoise(mode='tsys-atm') is not yet implemented in casa-rs \
127                     (simulation Phase 4)",
128                ));
129            }
130            other => {
131                return Err(SimulatorError::refused(format!(
132                    "Caught exception: unsupported mode {other} in setnoise()"
133                )));
134            }
135        };
136        if !spec.on_the_fly {
137            return Err(SimulatorError::not_yet_implemented(
138                "setnoise(OTF=False), which writes a calibration table, is not yet implemented \
139                 in casa-rs (simulation Phase 3)",
140            ));
141        }
142        if !tsys {
143            tracing::info!(
144                "Using simple noise model with noise level of {} Jy",
145                spec.simple_noise_jy
146            );
147        }
148        if self.planned_rows() == 0 {
149            return Err(SimulatorError::refused(
150                "Caught exception: Error in Simulator::create_corrupt: MSIter::construct - \
151                 Input MeasurementSet.has zero selected rows",
152            ));
153        }
154        let level = if tsys {
155            NoiseLevel::TsysManual(self.tsys_manual(spec)?)
156        } else {
157            NoiseLevel::Simple(spec.simple_noise_jy as f32)
158        };
159        tracing::info!("Creating A NOISE Calibration structure for data corruption.");
160        if tsys {
161            tracing::info!("Creating T Calibration structure for data corruption.");
162        }
163        self.noise_terms.push(NoiseTerm {
164            seed: self.seed,
165            level,
166            applications: 0,
167        });
168        Ok(())
169    }
170
171    /// The `tsys-manual` model of `spec`, checked as CASA checks it.
172    fn tsys_manual(&self, spec: &NoiseSpec) -> SimulatorResult<TsysManual> {
173        let parameters = &spec.tsys;
174        if parameters.receiver_type > 0 {
175            tracing::info!("User has requested Double Sideband Receiver");
176        } else if parameters.receiver_type < 0 {
177            tracing::info!("User has not set Rx type, using 2SB");
178        }
179        if parameters.sensitivity > 0.0 {
180            tracing::info!(
181                "adding noise with the sensitivity constant of {}",
182                parameters.sensitivity as f32
183            );
184        } else {
185            tracing::info!("adding noise with the sensitivity constant of 1/sqrt(2)");
186        }
187        if spec.water_vapour_mm > LARGEST_WATER_VAPOUR_MM {
188            return Err(SimulatorError::refused(format!(
189                "Caught exception:  you input PWV={}mm. The most common reason for this error \
190                 is forgetting to set units, which default to meters",
191                spec.water_vapour_mm
192            )));
193        }
194        if let Some(table) = calibration_table_root(&spec.table) {
195            return Err(SimulatorError::not_yet_implemented(format!(
196                "setnoise(mode='tsys-manual', table=...), which writes the T calibration table \
197                 {table}.T.cal, is not yet implemented in casa-rs (simulation Phase 3)"
198            )));
199        }
200        let model = TsysManual::new(parameters);
201        // CASA checks the zenith Tsys at the centre of the first window.
202        let window = self
203            .spectral_windows
204            .first()
205            .ok_or_else(|| SimulatorError::raised("tsys-manual noise needs a spectral window"))?;
206        let zenith = model.zenith_tsys(
207            window.reference_frequency_hz,
208            window.total_bandwidth_hz,
209            window.channel_count,
210        );
211        tracing::info!(
212            "manual tau set to {} Zenith Tsys at center of first Spectral Window = {zenith} \
213             tground={} spillover={}; tauscale={}",
214            model.tau(),
215            parameters.ground_k as f32,
216            parameters.spillover_efficiency as f32,
217            model.tau()
218        );
219        if zenith > LARGEST_ZENITH_TSYS_K || zenith <= 0.0 {
220            return Err(SimulatorError::refused(format!(
221                "Caught exception: error in ATM setup - Tsys {zenith} poorly defined - check \
222                 inputs"
223            )));
224        }
225        Ok(model)
226    }

228    /// `sm.corrupt()`: apply every noise term to the rows observed so far.
229    /// With no terms it does nothing.
230    ///
231    /// # Errors
232    ///
233    /// A scan observed so far with a correlation count other than 1, 2 or 4,
234    /// which CASA cannot corrupt ("No such VisVector type"), or a term
235    /// applied 2²⁴ times.
236    pub fn corrupt(&mut self) -> SimulatorResult<()> {
237        if self.noise_terms.is_empty() {
238            return Ok(());
239        }
240        self.check_visibility_correlations("corrupted")?;
241        if self
242            .noise_terms
243            .iter()
244            .any(|term| term.applications >= MAXIMUM_APPLICATIONS)
245        {
246            return Err(SimulatorError::raised(format!(
247                "a noise term cannot be applied more than {MAXIMUM_APPLICATIONS} times"
248            )));
249        }
250        tracing::info!("Doing visibility corruption.");
251        let applications = self.noise_terms.iter_mut().map(|term| {
252            let application = NoiseApplication {
253                seed: term.seed,
254                level: term.level,
255                ordinal: term.applications,
256            };
257            term.applications += 1;
258            application
259        });
260        self.data_plan.corrupt(applications, self.scans.len());
261        Ok(())
262    }
263
264    /// `sm.reset()`: remove every noise term. Corruptions already applied,
265    /// and the seed, stay.
266    pub fn reset(&mut self) {
267        tracing::info!("Resetting all visibility corruption components");
268        tracing::info!("Reset all image-plane corruption components");
269        self.noise_terms.clear();
270    }
271
272    /// MAIN rows of the scans observed so far.
273    pub(super) fn planned_rows(&self) -> usize {
274        self.scans.iter().map(|scan| scan.row_count()).sum()
275    }
276
277    /// CASA's visibility vectors hold 1, 2 or 4 correlations: a scan
278    /// observed so far with another count cannot be `what` (corrupted,
279    /// predicted).
280    ///
281    /// # Errors
282    ///
283    /// Raised: "No such VisVector type".
284    pub(super) fn check_visibility_correlations(&self, what: &str) -> SimulatorResult<()> {
285        for scan in &self.scans {
286            let correlations = self.spectral_windows[scan.spectral_window_id]
287                .correlations
288                .len();
289            if !matches!(correlations, 1 | 2 | 4) {
290                return Err(SimulatorError::raised(format!(
291                    "No such VisVector type: scan {} has {correlations} correlations; \
292                     only 1, 2 or 4 can be {what}",
293                    scan.scan_number
294                )));
295            }
296        }
297        Ok(())
298    }

301/// The root of the calibration table `setnoise` writes for `table`, as
302/// `Simulator::setnoise` trims it (blanks, then a trailing `cal`, `.` and
303/// `_`), or `None` when it writes none (a root of at most one character).
304fn calibration_table_root(table: &str) -> Option<&str> {
305    let mut root = table.trim();
306    if root.len() > 3 {
307        root = root.strip_suffix("cal").unwrap_or(root);
308    }
309    for suffix in [".", "_"] {
310        if root.len() > 1 {
311            root = root.strip_suffix(suffix).unwrap_or(root);
312        }
313    }
314    (root.len() > 1).then_some(root)
315}

317#[cfg(test)]
318mod tests {
319    use super::calibration_table_root;
320
321    #[test]
322    fn calibration_table_names_are_trimmed_as_casa_trims_them() {
323        assert_eq!(calibration_table_root(""), None);
324        assert_eq!(calibration_table_root(" x "), None);
325        assert_eq!(calibration_table_root("noisy.cal"), Some("noisy"));
326        assert_eq!(calibration_table_root("noisy_.cal"), Some("noisy"));
327        assert_eq!(calibration_table_root("noisy_"), Some("noisy"));
328        assert_eq!(calibration_table_root("cal"), Some("cal"));
329        assert_eq!(calibration_table_root("a.cal"), None);
330    }
