33/// A selected FITS image ready to copy into a CASA image.
34///
35/// The image tools and the `importfits` task share this single-precision
36/// reader.
37pub struct FitsToolImport {
38    source: PrimaryHdu,
39    /// Coordinate record as the CASA FITS image tool saves it.
40    pub coordinate_record: RecordValue,
41    /// Brightness unit as interpreted by casacore.
42    pub units: String,
43    /// Original image-info record, including units for each beam.
44    pub image_info: RecordValue,
45    /// FITS keywords that Casa did not consume to decode the image.
46    pub misc_info: RecordValue,
47    /// Original history entries, in FITS card order.
48    pub history: Vec<ImageHistoryRow>,
49}

51impl FitsToolImport {
52    /// Read the metadata from the requested image HDU and coordinate representation.
53    /// Negative HDU numbers select the first nonempty image.
54    ///
55    /// # Errors
56    /// Returns an error for an unreadable FITS file, a missing image HDU,
57    /// or invalid coordinate or beam metadata.
58    pub fn open(path: &Path, which_rep: usize, which_hdu: i64) -> Result<Self, ImageError> {
59        let mut fits = FitsFile::open(path).map_err(|_| {
60            ImageError::CasaSelection(format!(
61                "Exception: {} does not exist or is not readable.",
62                path.display()
63            ))
64        })?;
65        let missing_hdu = || {
66            ImageError::CasaSelection(format!(
67                "Exception: No image at specified location in file {}.",
68                path.display()
69            ))
70        };
71        let count = fits.num_hdus().map_err(fits_error)?;
72        let hdu = if which_hdu < 0 {
73            (0..count).find_map(|index| fits.hdu(index).ok().filter(has_image_data))
74        } else {
75            usize::try_from(which_hdu)
76                .ok()
77                .filter(|&index| index < count)
78                .and_then(|index| fits.hdu(index).ok())
79                .filter(has_image_data)
80        }
81        .ok_or_else(missing_hdu)?;
82        let HduInfo::ImageInfo {
83            ref shape,
84            image_type,
85        } = hdu.info
86        else {
87            unreachable!("image HDU selected above");
88        };
89        let cards = read_hdu_cards(&mut fits)?;
90        let physical = FitsHeader::from_cards(
91            &cards
92                .iter()
93                .filter(|card| card.as_bytes().get(8) == Some(&b'='))
94                .map(String::as_str)
95                .collect::<Vec<_>>(),
96        );
97        let mut header = select_wcs(&physical, which_rep);
98        let shape = if is_tile_compressed(&physical) {
99            // The physical NAXIS cards describe the compression table rows.
100            let shape: Vec<_> = shape.iter().rev().copied().collect();
101            header.set("NAXIS", FitsValue::Integer(shape.len() as i64));
102            for (axis, &length) in shape.iter().enumerate() {
103                header.set(
104                    format!("NAXIS{}", axis + 1),
105                    FitsValue::Integer(length as i64),
106                );
107            }
108            shape
109        } else {
110            read_fits_axis_shape(&mut fits, &hdu)?
111        };
112        if header.get("RADESYS").is_none() {
113            let system = match header.get_string("RADECSYS") {
114                Some(system) => system,
115                None => match header.get_float("EQUINOX") {
116                    Some(epoch) if epoch < 1984.0 => "FK4",
117                    Some(_) => "FK5",
118                    None => "ICRS",
119                },
120            };
121            header.set("RADESYS", FitsValue::String(system.into()));
122        }
123        let mut has_wcs = false;
124        for axis in 1..=shape.len() {
125            for (prefix, default) in [("CRVAL", 0.0), ("CRPIX", 0.0), ("CDELT", 1.0)] {
126                let name = format!("{prefix}{axis}");
127                if header.get(&name).is_none() {
128                    header.set(name, FitsValue::Float(default));
129                } else {
130                    has_wcs = true;
131                }
132            }
133            has_wcs |= header.get(&format!("CTYPE{axis}")).is_some();
134        }
135        let (mut coordinates, axis_order) = from_fits_header_with_axis_order(&header, &shape)?;
136        if !has_wcs {
137            let ndim = shape.len();
138            coordinates = CoordinateSystem::new().with_obs_info(coordinates.obs_info().clone());
139            coordinates.add_coordinate(
140                LinearCoordinate::new(ndim, vec![String::new(); ndim], vec![String::new(); ndim])
141                    .with_reference_pixel(vec![-1.0; ndim]),
142            );
143        }
144        let obs = coordinates.obs_info_mut();
145        if obs.observer.is_empty() {
146            obs.observer = "UNKNOWN".into();
147        }
148        if obs.telescope.is_empty() {
149            obs.telescope = "UNKNOWN".into();
150        }
151        if obs.date.is_none() {
152            obs.date = Some(MEpoch::from_mjd(0.0, EpochRef::LAST));
153        }
154        if let [Some(x), Some(y), Some(z)] =
155            ["OBSGEO-X", "OBSGEO-Y", "OBSGEO-Z"].map(|key| header.get_float(key))
156        {
157            obs.telescope_position = Some(MPosition::new_itrf(x, y, z));
158        }
159        let coordinate_record = fits_coordinate_record(&coordinates, &axis_order, &header)?;
160        let coordinates = CoordinateSystem::from_record(&coordinate_record).map_err(|error| {
161            if matches!(
162                error,
163                casa_coordinates::CoordinateError::UnsupportedAxisLayout(_)
164            ) {
165                ImageError::CasaSelection(format!("{error} (casa-rs issue #1002)"))
166            } else {
167                error.into()
168            }
169        })?;
170        let mut source = PrimaryHdu {
171            fits,
172            hdu,
173            image_type,
174            shape,
175            header,
176            coordinates,
177        };
178        let mut info = image_info_from_fits_header(&source.header).to_record();
179        if let Some((beams, records)) = source.beam_set()? {
180            info.remove("restoringbeam");
181            if beams.is_single() {
182                if let Some(beam) = records.get("*0") {
183                    info.upsert("restoringbeam", beam.clone());
184                }
185            } else {
186                let mut grid = beams.to_record();
187                for field in records.fields() {
188                    grid.upsert(&field.name, field.value.clone());
189                }
190                info.upsert("perplanebeams", Value::Record(grid));
191            }
192        }
193        Ok(Self {
194            coordinate_record,
195            units: brightness_unit(&source.header),
196            image_info: info,
197            misc_info: miscellaneous(&physical),
198            history: history_from_cards(&cards, true),
199            source,
200        })
201    }

237    /// Copy single-precision pixels and the FITS validity mask into an image.
238    /// The copy uses bounded contiguous blocks along the FITS axis order.
239    ///
240    /// # Errors
241    /// Returns an error when the image's shape disagrees or reading or
242    /// writing fails.
243    pub fn copy_pixels(
244        &mut self,
245        image: &mut impl MutableImageInterface<f32>,
246        zero_blanks: bool,
247    ) -> Result<(), ImageError> {
248        let axes: Vec<_> = (0..self.source.shape.len()).collect();
249        self.copy_pixels_with_axes(image, zero_blanks, &axes)
250    }
251
252    // Destination axes in source order, with singleton axes appended beyond
253    // the FITS rank before applying the same permutation to pixels and masks.
254    pub(crate) fn copy_pixels_with_axes(
255        &mut self,
256        image: &mut impl MutableImageInterface<f32>,
257        zero_blanks: bool,
258        axes: &[usize],
259    ) -> Result<(), ImageError> {
260        let mut shape = self.source.shape.clone();
261        let ndim = shape.len();
262        if axes.len() < ndim {
263            return Err(ImageError::InvalidMetadata(
264                "FITS axes were dropped during import".into(),
265            ));
266        }
267        let mut remaining = vec![true; axes.len()];
268        for &axis in axes {
269            if !remaining.get_mut(axis).is_some_and(std::mem::take) {
270                return Err(ImageError::InvalidMetadata(
271                    "invalid FITS axis order".into(),
272                ));
273            }
274        }
275        shape.resize(axes.len(), 1);
276        let expected: Vec<_> = axes.iter().map(|&axis| shape[axis]).collect();
277        if image.shape() != expected {
278            return Err(ImageError::ShapeMismatch {
279                expected,
280                got: image.shape().to_vec(),
281            });
282        }
283        let total = shape
284            .iter()
285            .try_fold(1usize, |product, &length| product.checked_mul(length))
286            .ok_or_else(|| ImageError::InvalidMetadata("FITS image shape is too large".into()))?;
287        let quota = 1_048_576;
288        let mut stride = 1;
289        let mut axis = 0;
290        while axis + 1 < ndim && stride * shape[axis] <= quota {
291            stride *= shape[axis];
292            axis += 1;
293        }
294        let mut block_shape = vec![1; axes.len()];
295        block_shape[..axis].copy_from_slice(&shape[..axis]);
296        let mut origin = vec![0; axes.len()];
297        let mut output_origin = origin.clone();
298        let mut start = 0;
299        let mut mask = None;
300        while start < total {
301            let mut offset = start;
302            for (position, length) in origin.iter_mut().zip(&shape) {
303                *position = offset % length;
304                offset /= length;
305            }
306            for (position, &source_axis) in output_origin.iter_mut().zip(axes) {
307                *position = origin[source_axis];
308            }
309            let run = (quota / stride).min(shape[axis] - origin[axis]);
310            block_shape[axis] = run;
311            let count = run * stride;
312            let (mut values, valid) = self.source.read_pixels(start, count)?;
313            if valid.iter().any(|bit| !bit) {
314                let mask = mask.get_or_insert_with(|| vec![true; total]);
315                mask[start..start + count].copy_from_slice(&valid);
316                for (value, valid) in values.iter_mut().zip(valid) {
317                    if !valid {
318                        *value = if zero_blanks { 0.0 } else { f32::NAN };
319                    }
320                }
321            }
322            let block = ArrayD::from_shape_vec(IxDyn(&block_shape).f(), values)
323                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
324            image.put_slice(&block.permuted_axes(IxDyn(axes)), &output_origin)?;
325            start += count;
326        }
327        if let Some(mask) = mask {
328            let mask = ArrayD::from_shape_vec(IxDyn(&shape).f(), mask)
329                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
330            image.put_mask("mask0", &mask.permuted_axes(IxDyn(axes)))?;
331            image.set_default_mask("mask0")?;
332        }
333        Ok(())
334    }
335}

341/// A read-only pixel stream for a specific FITS image HDU.
342pub struct FitsSectionReader {
343    file: ThreadsafeFitsFile,
344    shape: Vec<usize>,
345    bitpix: i32,
346}
347
348impl FitsSectionReader {
349    /// Open one image HDU and retain it for successive reads. `shape` is
350    /// the logical image shape, in FITS pixel axis order.
351    ///
352    /// # Errors
353    /// Returns an error for missing or incompatible image data, including
354    /// a source whose shape changed since its metadata was read.
355    pub fn open(path: &Path, hdu_index: usize, shape: &[usize]) -> Result<Self, ImageError> {
356        let mut file = FitsFile::open(path).map_err(fits_error)?;
357        let hdu = file.hdu(hdu_index).map_err(fits_error)?;
358        let HduInfo::ImageInfo { shape: actual, .. } = hdu.info else {
359            return Err(ImageError::InvalidMetadata(
360                "selected FITS HDU is not an image".into(),
361            ));
362        };
363        if shape.is_empty() || shape.len() > 9 || !actual.iter().rev().eq(shape) {
364            return Err(ImageError::InvalidMetadata(
365                "FITS pixel source has incompatible image dimensions".into(),
366            ));
367        }
368        let (mut bitpix, mut status) = (0, 0);
369        // SAFETY: the handle is on the selected image and both outputs are
370        // valid. CFITSIO reports the logical BITPIX for compressed images.
371        unsafe { fitsio::sys::ffgidt(file.as_raw(), &mut bitpix, &mut status) };
372        check_status(status).map_err(fits_error)?;
373        Ok(Self {
374            file: file.threadsafe(),
375            shape: shape.to_vec(),
376            bitpix,
377        })
378    }
379
380    /// Read a strided section as `f32`, converting integer blanks to NaN.
381    /// Only the requested region is allocated.
382    ///
383    /// # Errors
384    /// Returns an error if reading fails or the slice is outside the image.
385    pub fn read(
386        &self,
387        start: &[usize],
388        count: &[usize],
389        stride: &[usize],
390    ) -> Result<ArrayD<f32>, ImageError> {
391        let mut file = self.file.lock().map_err(fits_error)?;
392        read_section(&mut file, &self.shape, self.bitpix, start, count, stride)
393    }
394}

419where
420    F: Fn(&[usize], &[usize], &[usize]) -> Result<ArrayD<f32>, ImageError>,
421{
422    fn shape(&self) -> &[usize] {
423        self.shape
424    }

427        false
428    }

430    fn get_at(&self, position: &[usize]) -> Result<f32, LatticeError> {
431        let ones = vec![1; self.ndim()];
432        let data = self.get_slice(position, &ones, &ones)?;
433        Ok(*data.first().expect("read one FITS pixel"))
434    }

437        &self,

439        shape: &[usize],
440        stride: &[usize],
441    ) -> Result<ArrayD<f32>, LatticeError> {
442        (self.read)(start, shape, stride).map_err(|error| LatticeError::Table(error.to_string()))
443    }

493fn read_section(
494    file: &mut FitsFile,
495    shape: &[usize],
496    bitpix: i32,
497    start: &[usize],
498    count: &[usize],
499    stride: &[usize],
500) -> Result<ArrayD<f32>, ImageError> {
501    let invalid = || ImageError::InvalidMetadata("FITS pixel section is outside the image".into());
502    let dims = shape.len();
503    if start.len() != dims || count.len() != dims || stride.len() != dims {
504        return Err(invalid());
505    }
506    let mut begin = Vec::with_capacity(dims);
507    let mut end = Vec::with_capacity(dims);
508    let mut increment = Vec::with_capacity(dims);
509    let mut size = 1usize;
510    for axis in 0..dims {
511        if stride[axis] == 0 || count[axis] == 0 {
512            return Err(invalid());
513        }
514        let last = (count[axis] - 1)
515            .checked_mul(stride[axis])
516            .and_then(|width| start[axis].checked_add(width))
517            .filter(|last| *last < shape[axis])
518            .ok_or_else(invalid)?;
519        begin.push(c_long::try_from(start[axis] + 1).map_err(|_| invalid())?);
520        end.push(c_long::try_from(last + 1).map_err(|_| invalid())?);
521        increment.push(c_long::try_from(stride[axis]).map_err(|_| invalid())?);
522        size = size.checked_mul(count[axis]).ok_or_else(invalid)?;
523    }
524    let mut values = vec![0.0_f32; size];
525    let mut null = f32::NAN;
526    // FITS float pixels already encode blanks as NaN. Asking CFITSIO to
527    // substitute them would also turn valid infinities into blanks.
528    let null_value = if bitpix < 0 {
529        std::ptr::null_mut()
530    } else {
531        (&mut null as *mut f32).cast()
532    };
533    let (mut any_null, mut status) = (0, 0);
534    // SAFETY: the FITS handle is on the selected image and the coordinate
535    // vectors and output buffer match the checked section above.
536    unsafe {
537        fitsio::sys::ffgsv(
538            file.as_raw(),
539            fitsio::sys::TFLOAT as i32,
540            begin.as_mut_ptr(),
541            end.as_mut_ptr(),
542            increment.as_mut_ptr(),
543            null_value,
544            values.as_mut_ptr().cast(),
545            &mut any_null,
546            &mut status,
547        );
548    }
549    check_status(status).map_err(fits_error)?;
550    ArrayD::from_shape_vec(IxDyn(count).f(), values)
551        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
552}

554fn fits_coordinate_record(
555    coordinates: &CoordinateSystem,
556    axis_order: &[usize],
557    header: &FitsHeader,
558) -> Result<RecordValue, ImageError> {
559    let original = coordinates.to_record();
560    let mut record = original.clone();
561    let mut descriptors = CoordinateSystem::record_coordinates(&original)?;
562    let coordinate_kind = |field: &str| match field.trim_end_matches(|ch: char| ch.is_ascii_digit())
563    {
564        "direction" => 0,
565        "stokes" => 1,
566        "spectral" => 2,
567        _ => 3,
568    };
569    descriptors.sort_by_key(|coordinate| coordinate_kind(&coordinate.field));
570    let mapping_fields = ["pixelmap", "pixelreplace", "worldmap", "worldreplace"];
571    for descriptor in &descriptors {
572        let kind = descriptor
573            .field
574            .trim_end_matches(|ch: char| ch.is_ascii_digit());
575        let index = &descriptor.field[kind.len()..];
576        record.remove(&descriptor.field);
577        for field in mapping_fields {
578            record.remove(&format!("{field}{index}"));
579        }
580    }
581    for (index, coordinate) in descriptors.iter().enumerate() {
582        let kind = coordinate
583            .field
584            .trim_end_matches(|ch: char| ch.is_ascii_digit());
585        let old_index = &coordinate.field[kind.len()..];
586        for field in mapping_fields {
587            if let Some(value) = original.get(&format!("{field}{old_index}")) {
588                let value = if matches!(field, "pixelmap" | "worldmap")
589                    && let Value::Array(ArrayValue::Int32(map)) = value
590                {
591                    Value::Array(ArrayValue::Int32(
592                        map.mapv(|axis| axis_order[axis as usize] as i32),
593                    ))
594                } else {
595                    value.clone()
596                };
597                record.upsert(format!("{field}{index}"), value);
598            }
599        }
600        if let Some(Value::Record(mut fields)) = original.get(&coordinate.field).cloned() {
601            let pixel_axes = coordinate
602                .pixel_axes
603                .iter()
604                .map(|axis| axis.map(|axis| axis_order[axis]))
605                .collect::<Vec<_>>();
606            if kind == "direction" {
607                set_fits_pole_and_projection(&mut fields, &pixel_axes, header);
608                normalize_direction_matrix(&mut fields);
609            } else if kind == "linear" {
610                fields.remove("naxes");
611                fields.remove("coordinate_type");
612            } else if kind == "spectral"
613                && let Some(Some(axis)) = pixel_axes.first()
614            {
615                let ctype = header
616                    .get_string(&format!("CTYPE{}", axis + 1))
617                    .unwrap_or("");
618                if ctype.eq_ignore_ascii_case("VOPT") {
619                    fields.upsert("nativeType", Value::Scalar(ScalarValue::Int32(2)));
620                }
621                if let Some(Value::Record(tabular)) = fields.get_mut("tabular") {
622                    tabular.upsert(
623                        "pc",
624                        Value::Array(ArrayValue::Float64(ndarray::arr2(&[[1.0]]).into_dyn())),
625                    );
626                }
627                if header.get("SPECSYS").is_none()
628                    && header.get("VELREF").is_none()
629                    && !ctype.to_ascii_uppercase().ends_with("-LSR")
630                {
631                    fields.upsert(
632                        "system",
633                        Value::Scalar(ScalarValue::String("Undefined".into())),
634                    );
635                }
636                let system = fields.get("system").expect("spectral coordinate system");
637                let mut conversion = RecordValue::default();
638                conversion.upsert("system", system.clone());
639                conversion.upsert(
640                    "epoch",
641                    Value::Record(epoch_to_record(&MEpoch::from_mjd(0.0, EpochRef::LAST))),
642                );
643                conversion.upsert(
644                    "position",
645                    Value::Record(position_to_record(&MPosition::new_itrf(0.0, 0.0, 0.0))),
646                );
647                conversion.upsert(
648                    "direction",
649                    Value::Record(direction_to_record(&MDirection::from_angles(
650                        0.0,
651                        std::f64::consts::FRAC_PI_2,
652                        DirectionRef::J2000,
653                    ))),
654                );
655                fields.upsert("conversion", Value::Record(conversion));
656            }
657            record.upsert(format!("{kind}{index}"), Value::Record(fields));
658        }
659    }
660    Ok(record)
661}

663fn normalize_direction_matrix(record: &mut RecordValue) {
664    let Some(Value::Array(ArrayValue::Float64(mut matrix))) = record.get("pc").cloned() else {
665        return;
666    };
667    let Some(Value::Array(ArrayValue::Float64(mut increments))) = record.get("cdelt").cloned()
668    else {
669        return;
670    };
671    // The saved PC matrix is transposed with respect to the WCS transform.
672    // casacore moves each world-axis row's norm into its CDELT.
673    for (axis, increment) in matrix.axis_iter_mut(Axis(1)).zip(&mut increments) {
674        let norm = axis
675            .iter()
676            .map(|coefficient| coefficient * coefficient)
677            .sum::<f64>()
678            .sqrt();
679        if norm != 0.0 && norm != 1.0 {
680            *increment *= norm;
681            axis.into_iter()
682                .for_each(|coefficient| *coefficient /= norm);
683        }
684    }
685    record.upsert("pc", Value::Array(ArrayValue::Float64(matrix)));
686    record.upsert("cdelt", Value::Array(ArrayValue::Float64(increments)));
687}

689fn set_fits_pole_and_projection(
690    direction: &mut RecordValue,
691    pixel_axes: &[Option<usize>],
692    header: &FitsHeader,
693) {
694    let Some(Value::Scalar(ScalarValue::String(projection))) = direction.get("projection") else {
695        return;
696    };
697    let sin = projection == "SIN";
698    if sin || matches!(projection.as_str(), "TAN" | "STG" | "ARC" | "ZEA" | "AZP") {
699        // The pole for a zenithal WCS is the center of the projection.
700        if let Some(Value::Array(ArrayValue::Float64(reference))) = direction.get("crval")
701            && let Some(latitude) = reference.get(IxDyn(&[1]))
702        {
703            direction.upsert(
704                "latpole",
705                Value::Scalar(ScalarValue::Float64(latitude.to_degrees())),
706            );
707        }
708    }
709    if sin && let Some(Some(axis)) = pixel_axes.get(1) {
710        let values = [1, 2].map(|index| {
711            header
712                .get_float(&format!("PV{}_{index}", axis + 1))
713                .unwrap_or(0.0)
714        });
715        direction.upsert(
716            "projection_parameters",
717            Value::Array(ArrayValue::Float64(ndarray::arr1(&values).into_dyn())),
718        );
719    }
720}

730fn read_hdu_cards(fits: &mut FitsFile) -> Result<Vec<String>, ImageError> {
731    let (mut count, mut more, mut status) = (0, 0, 0);
732    // SAFETY: the fitsio file is open on the image HDU and all outputs are live.
733    unsafe { fitsio::sys::ffghsp(fits.as_raw(), &mut count, &mut more, &mut status) };
734    check_status(status).map_err(fits_error)?;
735    let mut cards = Vec::with_capacity(count as usize);
736    for index in 1..=count {
737        let mut buffer = [0; 81];
738        // SAFETY: CFITSIO writes one card, at most 80 bytes and a terminator.
739        unsafe { fitsio::sys::ffgrec(fits.as_raw(), index, buffer.as_mut_ptr(), &mut status) };
740        check_status(status).map_err(fits_error)?;
741        // SAFETY: a successful ffgrec NUL-terminates its output.
742        cards.push(
743            unsafe { CStr::from_ptr(buffer.as_ptr()) }
744                .to_string_lossy()
745                .into_owned(),
746        );
747    }
748    Ok(cards)
749}

751fn wcs_key(name: &str) -> bool {
752    let prefix = name.trim_end_matches(|ch: char| ch.is_ascii_digit() || ch == '_');
753    matches!(
754        prefix,
755        "CTYPE"
756            | "CRPIX"
757            | "CRVAL"
758            | "CDELT"
759            | "CUNIT"
760            | "CROTA"
761            | "PV"
762            | "PS"
763            | "PC"
764            | "CD"
765            | "WCSAXES"
766            | "WCSNAME"
767            | "RADESYS"
768            | "RADECSYS"
769            | "EQUINOX"
770            | "SPECSYS"
771            | "SSYSOBS"
772            | "LONPOLE"
773            | "LATPOLE"
774            | "RESTFRQ"
775            | "RESTFREQ"
776            | "RESTWAV"
777            | "VELREF"
778    )
779}

781fn alternate_wcs_key(name: &str) -> Option<(&str, u8)> {
782    let last = *name.as_bytes().last()?;
783    if !last.is_ascii_uppercase() {
784        return None;
785    }
786    let base = &name[..name.len() - 1];
787    wcs_key(base).then_some((base, last))
788}

790fn select_wcs(header: &FitsHeader, which_rep: usize) -> FitsHeader {
791    if which_rep == 0 {
792        return header.clone();
793    }
794    let mut alternatives: Vec<u8> = header
795        .iter()
796        .filter_map(|field| alternate_wcs_key(&field.name))
797        .filter_map(|(base, suffix)| base.starts_with("CTYPE").then_some(suffix))
798        .collect();
799    alternatives.sort_unstable();
800    alternatives.dedup();
801    let Some(suffix) = alternatives.get(which_rep.min(alternatives.len()).saturating_sub(1)) else {
802        return header.clone();
803    };
804    let mut selected = FitsHeader::new();
805    for field in header.iter() {
806        if let Some((base, candidate)) = alternate_wcs_key(&field.name) {
807            if candidate == *suffix {
808                selected.set(base, field.value.clone());
809            }
810        } else if !wcs_key(&field.name) {
811            selected.set(&field.name, field.value.clone());
812        }
813    }
814    selected
815}

817fn miscellaneous(header: &FitsHeader) -> RecordValue {
818    let mut record = RecordValue::default();
819    for field in header.iter() {
820        let name = &field.name;
821        if wcs_key(name)
822            || alternate_wcs_key(name).is_some()
823            || name.starts_with("NAXIS")
824            || matches!(
825                name.as_str(),
826                "BUNIT"
827                    | "BITPIX"
828                    | "DATAMAX"
829                    | "DATAMIN"
830                    | "ORIGIN"
831                    | "EXTEND"
832                    | "BLOCKED"
833                    | "BLANK"
834                    | "SIMPLE"
835                    | "BSCALE"
836                    | "BZERO"
837                    | "XTENSION"
838                    | "PCOUNT"
839                    | "GCOUNT"
840                    | "TELESCOP"
841                    | "OBSERVER"
842                    | "DATE"
843                    | "DATE-OBS"
844                    | "OBSGEO-X"
845                    | "OBSGEO-Y"
846                    | "OBSGEO-Z"
847                    | "MJD-OBS"
848                    | "TIMESYS"
849                    | "BMAJ"
850                    | "BMIN"
851                    | "BPA"
852                    | "IMTYPE"
853            )
854        {
855            continue;
856        }
857        let value = match &field.value {
858            FitsValue::Integer(value) => {
859                i32::try_from(*value).map_or(ScalarValue::Int64(*value), ScalarValue::Int32)
860            }
861            FitsValue::Float(value) => ScalarValue::Float64(*value),
862            FitsValue::Logical(value) => ScalarValue::Bool(*value),
863            FitsValue::String(value) if name == "OBJECT" && value.is_empty() => {
864                ScalarValue::String(" ".into())
865            }
866            FitsValue::String(value) => ScalarValue::String(value.clone()),
867        };
868        record.upsert(name.to_ascii_lowercase(), Value::Scalar(value));
869    }
870    record
871}
