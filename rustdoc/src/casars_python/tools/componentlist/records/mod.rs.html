27/// Reuse the CASA record typemap, including scalar/array and numeric kinds.
28pub(super) fn argument(
29    py: Python<'_>,
30    value: Option<&Bound<'_, PyAny>>,
31    name: &str,
32) -> PyResult<RecordValue> {
33    let Some(value) = value else {
34        return Ok(RecordValue::default());
35    };
36    if !value.is_instance_of::<PyDict>() {
37        return Err(PyTypeError::new_err(format!(
38            "argument _{name} must be a dictionary"
39        )));
40    }
41    let mut holder = RecordValue::default();
42    define_component_keyword(
43        &mut holder,
44        "record".into(),
45        component_keyword_value(py, Some(value))?,
46    )?;
47    match holder.get("record") {
48        Some(Value::Record(record)) => Ok(record.clone()),
49        _ => unreachable!("dictionary converts to a record"),
50    }
51}

53pub(super) fn count(record: &RecordValue) -> ConvertResult<usize> {
54    match record.get("nelements") {
55        None => Ok(0),
56        Some(Value::Scalar(ScalarValue::Int32(count))) => Ok(*count as u32 as usize),
57        Some(Value::Scalar(ScalarValue::Int64(count))) => Ok(*count as u32 as usize),
58        Some(Value::Scalar(ScalarValue::UInt32(count))) => Ok(*count as usize),
59        _ => Err(ConvertError::runtime(
60            "RecordInterface::asuInt - invalid data type",
61        )),
62    }
63}

65pub(super) fn item(record: &RecordValue, index: usize) -> ConvertResult<&RecordValue> {
66    let name = format!("component{index}");
67    match record.get(&name) {
68        None => Err(ConvertError::runtime(format!(
69            "RecordInterface: field {name} is unknown"
70        ))),
71        Some(Value::Record(record)) => Ok(record),
72        Some(value) => Err(ConvertError::runtime(format!(
73            "RecordRep::get_pointer - incorrect data type Record   used for field {name} with type {}",
74            type_name(value)
75        ))),
76    }
77}

79fn type_name(value: &Value) -> &'static str {
80    match value {
81        Value::Scalar(ScalarValue::Int32(_) | ScalarValue::Int64(_)) => "Int     ",
82        Value::Scalar(ScalarValue::Float64(_) | ScalarValue::Float32(_)) => "double  ",
83        Value::Scalar(ScalarValue::String(_)) => "String  ",
84        Value::Scalar(ScalarValue::Bool(_)) => "Bool    ",
85        _ => "Array   ",
86    }
87}

89pub(super) fn decode(input: &RecordValue, lines: &mut Vec<LogLine>) -> Result<Component, String> {
90    let defaults = SkyComponent::new(
91        Flux::stokes_jy(1.0, 0.0, 0.0, 0.0),
92        ComponentShape::point(MDirection::from_angles(
93            0.0,
94            std::f64::consts::FRAC_PI_2,
95            DirectionRef::J2000,
96        )),
97        SpectralModel::new(
98            MFrequency::new(1e9, FrequencyRef::LSRK),
99            SpectralKind::Constant,
100        ),
101    )
102    .to_record()
103    .map_err(|e| e.to_string())?;
104    let flux = match input.get("flux") {
105        None => {
106            lines.push(LogLine::warning("The component does not have a 'flux' field.\nThe default is 1.0 Jy in I and 0.0 in Q, U & V"));
107            defaults.get("flux").expect("default flux").clone()
108        }
109        Some(Value::Record(record)) => Value::Record(flux(record, lines).map_err(|e| {
110            if e.starts_with("Unit::check ") {
111                e
112            } else {
113                format!("{e}Problem parsing the 'flux' field\n")
114            }
115        })?),
116        Some(_) => return Err("The 'flux' field must be a record\n".into()),
117    };
118    let (shape, units) = match input.get("shape") {
119        None => {
120            lines.push(LogLine::warning("The component does not have a 'shape' field.\nThe default is a point component at the J2000 north pole"));
121            (defaults.get("shape").expect("default shape").clone(), None)
122        }
123        Some(Value::Record(record)) => {
124            let (record, units) = shape(record, lines)?;
125            (Value::Record(record), units)
126        }
127        Some(_) => return Err("\nThe 'shape' field must be a record".into()),
128    };
129    let spectrum = match input.get("spectrum") {
130        None => {
131            lines.push(LogLine::warning("The component does not have a 'spectrum' field.\nThe default is a constant spectrum"));
132            defaults.get("spectrum").expect("default spectrum").clone()
133        }
134        Some(Value::Record(record)) => Value::Record(spectrum(record, lines)?),
135        Some(_) => return Err("\nThe 'spectrum' field must be a record".into()),
136    };
137    let mut record = RecordValue::new(vec![
138        RecordField::new("flux", flux),
139        RecordField::new("shape", shape),
140        RecordField::new("spectrum", spectrum),
141    ]);
142    if let Some(label) = input.get("label") {
143        if !matches!(label, Value::Scalar(ScalarValue::String(_))) {
144            return Err("\nThe 'label' field must be a string".into());
145        }
146        record.push(RecordField::new("label", label.clone()));
147    }
148    if let Some(parameters) = input.get("optionalParameters") {
149        if !matches!(parameters, Value::Array(ArrayValue::Float64(_))) {
150            return Err("\nThe 'optionalParameters' field must be a Double Array".into());
151        }
152        record.push(RecordField::new("optionalParameters", parameters.clone()));
153    }
154    let offsets = super::ops::offsets::Offsets::extract(&mut record);
155    let mut value = SkyComponent::from_record(&record).map_err(|e| e.to_string())?;
156    value
157        .spectrum_mut()
158        .set_frequency_unit(Unit::new("GHz").expect("GHz"))
159        .map_err(|e| e.to_string())?;
160    Ok(Component {
161        value,
162        units,
163        offsets,
164    })
165}

167fn text<'a>(record: &'a RecordValue, key: &str) -> Option<&'a str> {
168    match record.get(key) {
169        Some(Value::Scalar(ScalarValue::String(text))) => Some(text),
170        _ => None,
171    }
172}
173fn scalar(value: &Value) -> Option<f64> {
174    match value {
175        Value::Scalar(ScalarValue::Float64(v)) => Some(*v),
176        Value::Scalar(ScalarValue::Float32(v)) => Some(f64::from(*v)),
177        Value::Scalar(ScalarValue::Int32(v)) => Some(f64::from(*v)),
178        _ => None,
179    }
180}
181fn numbers(value: &Value) -> Option<Vec<Complex64>> {
182    let real = |v: f64| Complex64::new(v, 0.0);
183    if let Some(number) = scalar(value) {
184        return Some(vec![real(number)]);
185    }
186    match value {
187        Value::Scalar(ScalarValue::Complex64(v)) => Some(vec![*v]),
188        Value::Array(ArrayValue::Float64(a)) => Some(a.iter().map(|&v| real(v)).collect()),
189        Value::Array(ArrayValue::Int32(a)) => Some(a.iter().map(|&v| real(f64::from(v))).collect()),
190        Value::Array(ArrayValue::Complex64(a)) => Some(a.iter().copied().collect()),
191        _ => None,
192    }
193}
194fn replace(record: &mut RecordValue, key: &str, value: Value) {
195    if let Some(old) = record.get_mut(key) {
196        *old = value;
197    } else {
198        record.push(RecordField::new(key, value));
199    }
200}
201fn flux(input: &RecordValue, lines: &mut Vec<LogLine>) -> Result<RecordValue, String> {
202    let basis=match input.get("polarisation") {
203        None=>{
204            lines.push(LogLine::warning("The flux does not have a 'polarisation' field.\nUsing the default of Stokes"));
205            "Stokes"
206        }
207        Some(Value::Scalar(ScalarValue::String(name)))=>match casa_components::Polarisation::parse_casa(name) {
208            Ok(basis)=>basis.casa_name(),
209            Err(_)=>return Err("The polarisation type is not known.\nAllowed values are 'Stokes', 'Linear' & 'Circular'\n".into()),
210        },
211        _=>return Err("The 'polarisation' field must be a record\n".into()),
212    };
213    let illegal =
214        "Illegal Quantum record in QuantumHolder::fromRecord\nCould not parse the flux record\n";
215    let unit = text(input, "unit").ok_or(illegal)?;
216    let values = input
217        .get("value")
218        .and_then(numbers)
219        .filter(|v| !v.is_empty())
220        .ok_or(illegal)?;
221    let scalar_value = matches!(input.get("value"), Some(Value::Scalar(_))) || values.len() == 1;
222    let values: [Complex64; 4] = if scalar_value && basis == "Stokes" {
223        if values[0].im.abs() > f64::MIN_POSITIVE {
224            return Err("I value cannot be complex\n".into());
225        }
226        lines.push(LogLine::warning(
227            "Only the I flux specified. Assuming Q, U, & V are zero\n",
228        ));
229        [
230            Complex64::new(values[0].re, 0.0),
231            Complex64::default(),
232            Complex64::default(),
233            Complex64::default(),
234        ]
235    } else {
236        values.try_into().map_err(|_|"Must specify all 4 flux values\nif the polarisation representation is not Stokes\n")?
237    };
238    let errors = match input.get("error") {
239        None => [Complex64::default(); 4],
240        Some(value) => {
241            let values = numbers(value)
242                .filter(|v| v.len() == 4)
243                .ok_or("The 'error' field must be a vector with 4 elements\n")?;
244            let real = matches!(value, Value::Array(ArrayValue::Float64(_)));
245            let complex = matches!(value, Value::Array(ArrayValue::Complex64(_)));
246            if !(complex || real && basis == "Stokes" && input.get("polarisation").is_some()) {
247                return Err("The 'error' field must be a complex vector with 4 elements\n".into());
248            }
249            values.try_into().expect("four errors")
250        }
251    };
252    let parsed = casa_core::quanta::casacore::casacore_unit(unit).map_err(|e| e.to_string())?;
253    if !parsed.conformant(&Unit::new("Jy").expect("Jy"))
254        && !parsed.conformant(&Unit::new("K.rad.rad").expect("K.rad.rad"))
255    {
256        return Err("The dimensions of the units must be same as the Jy\n".into());
257    }
258    let flux = Flux::new(
259        values,
260        unit,
261        casa_components::Polarisation::parse_casa(basis).expect("validated basis"),
262    )
263    .map_err(|e| e.to_string())?
264    .with_errors(errors);
265    Ok(flux.to_record())
266}

268pub(super) fn angular(record: &RecordValue, key: &str) -> Result<Quantity, String> {
269    let value = record
270        .get(key)
271        .ok_or_else(|| format!("The '{key}' field does not exist\n"))?;
272    let quantity=match value {
273        Value::Scalar(ScalarValue::String(text))=>Quantity::read(text).map_err(|_|format!("in QuantumHolder::fromString with input string \"{text}\": Illegal input units or format\nProblem parsing the '{key}' string\n"))?,
274        Value::Record(record)=>{
275            let unit = text(record, "unit");
276            let value = record.get("value").and_then(|value| match value {
277                Value::Array(ArrayValue::Float64(values)) if values.len() == 1 => values.iter().copied().next(),
278                Value::Array(ArrayValue::Float32(values)) if values.len() == 1 => values.iter().map(|&n| f64::from(n)).next(),
279                Value::Array(ArrayValue::Int32(values)) if values.len() == 1 => values.iter().map(|&n| f64::from(n)).next(),
280                scalar_value => scalar(scalar_value),
281            });
282            match (unit,value) {
283                (Some(unit),Some(value))=>Quantity::new(value,unit).map_err(|e|e.to_string())?,
284                _=>return Err(format!("Illegal Quantum record in QuantumHolder::fromRecord\nProblem parsing the '{key}' record\n")),
285            }
286        }
287        _=>return Err(format!("The '{key}' field must be a record\nor a string (but not a vector of strings)\n")),
288    };
289    if !quantity.unit().conformant(&Unit::new("rad").expect("rad")) {
290        return Err(format!("The '{key}' field must have angular units\n"));
291    }
292    Ok(quantity)
293}
294fn quantum(q: &Quantity) -> Value {
295    Value::Record(RecordValue::new(vec![
296        RecordField::new(
297            "unit",
298            Value::Scalar(ScalarValue::String(q.unit().name().into())),
299        ),
300        RecordField::new("value", Value::Scalar(ScalarValue::Float64(q.value()))),
301    ]))
302}
303fn shape(
304    input: &RecordValue,
305    lines: &mut Vec<LogLine>,
306) -> Result<(RecordValue, Option<[Unit; 6]>), String> {
307    let unknown = "Cannot create a component with a 'Unknown' shape\n";
308    let name = match input.get("type") {
309        None => {
310            return Err(format!(
311                "The 'shape' record does not have a 'type' field.\n{unknown}"
312            ));
313        }
314        Some(Value::Scalar(ScalarValue::String(name))) => name,
315        _ => {
316            return Err(format!(
317                "The 'type' field, in the shape record, must be a String\n{unknown}"
318            ));
319        }
320    };
321    let kind = ShapeType::parse_casa(name).map_err(|_| unknown)?;
322    let mut record = input.clone();
323    let direction = match input.get("direction") {
324        None => casa_core::measures::direction_to_record(&MDirection::from_angles(
325            0.0,
326            std::f64::consts::FRAC_PI_2,
327            DirectionRef::J2000,
328        )),
329        Some(Value::Record(direction)) => {
330            let mut direction = measures::normalize(direction, crate::convert::holder::MeasureKind::Direction, lines)
331                .map_err(|e| format!("{e}Could not parse the reference direction\nProblem parsing the 'shape' field\n"))?;
332            if let Some(error) = direction.get("error") {
333                let Value::Record(error) = error else {
334                    return Err(
335                        "The 'error' field must be a record\nProblem parsing the 'shape' field\n"
336                            .into(),
337                    );
338                };
339                let mut errors = error.clone();
340                for name in ["latitude", "longitude"] {
341                    let quantity = angular(error, name)?;
342                    if quantity.value() < 0.0 {
343                        return Err(
344                            "Exception: The errors must be non-negative angular quantities.".into(),
345                        );
346                    }
347                    replace(&mut errors, name, quantum(&quantity));
348                }
349                replace(&mut direction, "error", Value::Record(errors));
350            }
351            direction
352        }
353        _ => {
354            return Err(
355                "The 'direction' field must be a record\nProblem parsing the 'shape' field\n"
356                    .into(),
357            );
358        }
359    };
360    replace(&mut record, "direction", Value::Record(direction));
361    if kind == ShapeType::Point {
362        return Ok((record, None));
363    }
364    let names = [
365        "majoraxis",
366        "minoraxis",
367        "positionangle",
368        "majoraxiserror",
369        "minoraxiserror",
370        "positionangleerror",
371    ];
372    let mut units = Vec::with_capacity(6);
373    for (index, name) in names.iter().enumerate() {
374        let q = angular(input, name).map_err(|e| {
375            format!(
376                "{e}{}\nProblem parsing the 'shape' field\n",
377                if index < 3 {
378                    "Shape not changed"
379                } else {
380                    "Shape errors not changed"
381                }
382            )
383        })?;
384        if index >= 3 && q.value() < 0.0 {
385            return Err("The errors must be non-negative angular quantities.".into());
386        }
387        units.push(q.unit().clone());
388        replace(&mut record, name, quantum(&q));
389    }
390    Ok((record, Some(units.try_into().expect("six units"))))
391}
392fn spectrum(input: &RecordValue, lines: &mut Vec<LogLine>) -> Result<RecordValue, String> {
393    let unknown = "Cannot create a component with a 'Unknown' spectrum\n";
394    let name = match input.get("type") {
395        None => {
396            return Err(format!(
397                "\nThe record does not have a 'type' field.{unknown}"
398            ));
399        }
400        Some(Value::Scalar(ScalarValue::String(name))) => name,
401        _ => {
402            return Err(format!(
403                "\nThe 'type' field in the spectrum record,must be a String{unknown}"
404            ));
405        }
406    };
407    let kind = casa_components::SpectralType::parse_casa(name).map_err(|_| unknown)?;
408    let mut record = input.clone();
409    match input.get("frequency") {
410        None=>return Err("The 'frequency' field does not exist\n".into()),
411        Some(Value::Scalar(ScalarValue::String(value))) if value=="current"=>{
412            replace(&mut record,"frequency",Value::Record(casa_core::measures::frequency_to_record(&MFrequency::new(1e9,FrequencyRef::LSRK))));
413        }
414        Some(Value::Record(frequency))=>{
415            let frequency = measures::normalize(frequency, crate::convert::holder::MeasureKind::Frequency, lines)
416                .map_err(|e| format!("{e}Could not parse the reference frequency\n"))?;
417            replace(&mut record,"frequency",Value::Record(frequency));
418        },
419        _=>return Err("The 'frequency' field must be a record\nor the string 'current' (to use the current value)\n".into()),
420    }
421    spectrum::validate(&record, kind)?;
422    Ok(record)
423}

425/// C++ throws these errors directly. Each caller adds its own context to
426/// failures returned by `fromRecord`.
427pub(super) fn is_direct_import_error(message: &str) -> bool {
428    message.starts_with("RecordRep::")
429        || message.starts_with("RecordInterface::")
430        || message.starts_with("Unit::check ")
431        || message.ends_with("The errors must be non-negative angular quantities.")
432}
433
434pub(super) fn import_error(message: String, adding: bool) -> ConvertError {
435    if is_direct_import_error(&message) {
436        ConvertError::runtime(message)
437    } else if adding {
438        ConvertError::runtime(format!(
439            "Converting Component from record failed with error {message}"
440        ))
441    } else {
442        ConvertError::runtime(format!("Error {message} in converting from record"))
443    }
