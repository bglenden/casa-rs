2//! Select the head and tail of each time-separated raster strip.
3
4use casa_ms::MeasurementSet;
5
6use super::{Integration, error_text};
7
8pub(super) fn edge_windows(
9    ms: &MeasurementSet,
10    rows: &[Integration],
11    windows: &[usize],
12    channels: &[i32],
13    fraction: f64,
14    noff: i64,
15) -> Result<Vec<[f64; 2]>, String> {
16    let states = ms.state().map_err(error_text)?;
17    let on_source = (0..states.row_count())
18        .map(|row| {
19            states
20                .string(row, "OBS_MODE")
21                .map(|mode| mode.starts_with("OBSERVE_TARGET#ON_SOURCE"))
22        })
23        .collect::<Result<Vec<_>, _>>()
24        .map_err(error_text)?;
25    let mut nominal = vec![None; channels.len()];
26    for (description, &window) in windows.iter().enumerate() {
27        let slot = nominal
28            .get_mut(window)
29            .ok_or("Invalid spectral window ID.")?;
30        *slot = Some(description as i32);
31    }
32    let reference_description = nominal
33        .into_iter()
34        .zip(channels)
35        .filter(|&(_, &count)| count != 4)
36        .filter_map(|(description, _)| description)
37        .find(|&description| {
38            rows.iter().any(|row| {
39                row.antenna == 0
40                    && row.other_antenna == 0
41                    && row.description == description
42                    && on_source.get(row.state as usize) == Some(&true)
43            })
44        })
45        .ok_or("No on-source raster reference window for antenna 0.")?;
46    let mut scan = rows
47        .iter()
48        .filter(|row| {
49            row.antenna == 0
50                && row.other_antenna == 0
51                && row.feed == 0
52                && row.other_feed == 0
53                && row.description == reference_description
54        })
55        .collect::<Vec<_>>();
56    if scan.len() < 2 {
57        return Err("Not enough integrations in raster reference window.".into());
58    }
59    scan.sort_by(|left, right| left.time.total_cmp(&right.time));
60    let differences = scan.windows(2).map(|pair| pair[1].time - pair[0].time);
61    let mut sorted = differences.clone().collect::<Vec<_>>();
62    sorted.sort_unstable_by(f64::total_cmp);
63    let middle = sorted.len() / 2;
64    let lower = sorted[(sorted.len() - 1) / 2];
65    // casacore averages the middle pair only on the small-array path.
66    let median = if sorted.len() <= 100 {
67        (lower + sorted[middle]) / 2.
68    } else {
69        lower
70    };
71    let half_interval = scan
72        .iter()
73        .map(|row| row.interval)
74        .reduce(f64::min)
75        .unwrap()
76        / 2.;
77    let mut first = 0;
78    let mut result = Vec::new();
79    for last in differences
80        .enumerate()
81        .filter_map(|(index, difference)| (difference > 5. * median).then_some(index + 1))
82        .chain(std::iter::once(scan.len()))
83    {
84        let count = edge_count(last - first, fraction, noff)?;
85        result.extend([
86            [
87                scan[first].time - half_interval,
88                scan[first + count - 1].time + half_interval,
89            ],
90            [
91                scan[last - count].time - half_interval,
92                scan[last - 1].time + half_interval,
93            ],
94        ]);
95        first = last;
96    }
97    Ok(result)
98}

100fn edge_count(integrations: usize, fraction: f64, noff: i64) -> Result<usize, String> {
101    let count = if noff > 0 {
102        usize::try_from(noff).unwrap_or(usize::MAX)
103    } else if fraction > 0. {
104        ((integrations as f32 * fraction as f32) as usize).max(1)
105    } else {
106        (((integrations as f64 + 1.).sqrt() - 1.) as usize).max(1)
107    };
108    if count.saturating_mul(2) >= integrations {
109        return Err(format!(
110            "Too many edge points ({count} at each end of {integrations} points)."
111        ));
112    }
113    Ok(count)
114}
