13/// Failure to read a saved voltage-pattern table or interpret its record.
14#[derive(Debug, thiserror::Error)]
15#[error("{0}")]
16pub struct VoltagePatternError(String);
17
18impl VoltagePatternError {
19    fn read(error: impl ToString) -> Self {
20        Self(error.to_string())
21    }
22}
23
24/// The root of a CASA VPManager table and its telescope defaults subtable.
25pub struct VoltagePatternTable {
26    root: Table,
27    defaults: PathBuf,
28}
29
30impl VoltagePatternTable {
31    /// Estimate peak bytes needed to open one VPManager table. The caller
32    /// holds this reservation while its decoded records are in use.
33    ///
34    /// # Errors
35    /// The table header or directory entries could not be read.
36    pub fn reading_bytes(path: &Path) -> Result<u64, VoltagePatternError> {
37        let rows = Table::rows_on_disk(path).map_err(VoltagePatternError::read)? as u64;
38        let mut bytes = rows.saturating_mul(4096).saturating_add(64 << 10);
39        for entry in std::fs::read_dir(path).map_err(VoltagePatternError::read)? {
40            let metadata = entry
41                .and_then(|entry| entry.metadata())
42                .map_err(VoltagePatternError::read)?;
43            if metadata.is_file() {
44                bytes = bytes.saturating_add(metadata.len().saturating_mul(4));
45            }
46        }
47        Ok(bytes)
48    }
49
50    /// Open the descriptions and locate the defaults table via its CASA
51    /// keyword. This does not open the defaults table.
52    ///
53    /// # Errors
54    /// The root table is unreadable or has no `VPLIST_DEFAULTS` table keyword.
55    pub fn open(path: &Path) -> Result<Self, VoltagePatternError> {
56        let root = Table::open(TableOptions::new(path)).map_err(VoltagePatternError::read)?;
57        let Some(Value::TableRef(reference)) = root.keywords().get("VPLIST_DEFAULTS") else {
58            return Err(VoltagePatternError::read(
59                "missing VPLIST_DEFAULTS subtable",
60            ));
61        };
62        let defaults = path.join(reference);
63        Ok(Self { root, defaults })
64    }
65
66    /// Estimate peak bytes needed to read the defaults subtable.
67    ///
68    /// # Errors
69    /// The defaults subtable cannot be sized.
70    pub fn defaults_reading_bytes(&self) -> Result<u64, VoltagePatternError> {
71        Self::reading_bytes(&self.defaults)
72    }
73
74    /// Read the last matching default row, as `VPManager::loadfromtable`
75    /// updates the telescope map. A saved index of `-1` selects a built-in
76    /// pattern and returns `None`.
77    ///
78    /// # Errors
79    /// The telescope default is missing, invalid, or names an unreadable row.
80    pub fn default_record(
81        &self,
82        telescope: &str,
83    ) -> Result<Option<RecordValue>, VoltagePatternError> {
84        let defaults =
85            Table::open(TableOptions::new(&self.defaults)).map_err(VoltagePatternError::read)?;
86        let mut selected = None;
87        for row in 0..defaults.row_count() {
88            let name = defaults
89                .cell_accessor(row, "tel_and_anttype")
90                .and_then(|cell| cell.scalar())
91                .map_err(VoltagePatternError::read)?;
92            if matches!(name, ScalarValue::String(name) if name == telescope) {
93                let ScalarValue::Int32(index) = defaults
94                    .cell_accessor(row, "vplistnum")
95                    .and_then(|cell| cell.scalar())
96                    .map_err(VoltagePatternError::read)?
97                else {
98                    return Err(VoltagePatternError::read(
99                        "VPLIST_DEFAULTS.vplistnum is not an integer",
100                    ));
101                };
102                selected = Some(*index);
103            }
104        }
105        let selected = selected.ok_or_else(|| {
106            VoltagePatternError::read(format!("no default for telescope {telescope}"))
107        })?;
108        if selected == -1 {
109            return Ok(None);
110        }
111        let index = usize::try_from(selected).map_err(VoltagePatternError::read)?;
112        if index >= self.root.row_count() {
113            return Err(VoltagePatternError::read(
114                "VPLIST_DEFAULTS selects an absent beam",
115            ));
116        }
117        self.root
118            .cell_accessor(index, "pbdescription")
119            .and_then(|cell| cell.record())
120            .map(Some)
121            .map_err(VoltagePatternError::read)
122    }
123}

125/// Decode the circular Gaussian and Airy laws shared by imaging and
126/// simulation. CASA stores radii at the record's reference frequency.
127/// The result scales them to the arcmin-GHz argument of PBMath1D.
128///
129/// # Errors
130/// The record is invalid or names a law other than `GAUSS` or `AIRY`.
131pub fn radial_pattern(record: &RecordValue) -> Result<PrimaryBeam, VoltagePatternError> {
132    let Some(Value::Scalar(ScalarValue::String(name))) = record.get("name") else {
133        return Err(VoltagePatternError::read("missing string name"));
134    };
135    let ghz = quantity(record, "reffreq", "GHz")?;
136    let radius = quantity(record, "maxrad", "arcmin")? * ghz;
137    if !ghz.is_finite() || ghz <= 0. || !radius.is_finite() || radius <= 0. {
138        return Err(VoltagePatternError::read(
139            "reference frequency and support must be finite and positive",
140        ));
141    }
142    match name.as_str() {
143        "AIRY" => {
144            let diameter = quantity(record, "dishdiam", "m")?;
145            let blockage = quantity(record, "blockagediam", "m")?;
146            if !diameter.is_finite()
147                || diameter <= 0.
148                || !blockage.is_finite()
149                || blockage < 0.
150                || blockage >= diameter
151            {
152                return Err(VoltagePatternError::read(
153                    "invalid dish or blockage diameter",
154                ));
155            }
156            Ok(PrimaryBeam::Airy {
157                dish_diameter_m: diameter,
158                blockage_diameter_m: blockage,
159                maximum_radius_arcmin_ghz: radius,
160            })
161        }
162        "GAUSS" => {
163            let width = quantity(record, "halfwidth", "arcmin")? * ghz;
164            if !width.is_finite() || width <= 0. {
165                return Err(VoltagePatternError::read(
166                    "Gaussian HWHM must be finite and positive",
167                ));
168            }
169            let Some(Value::Scalar(ScalarValue::Bool(is_voltage))) = record.get("isthisvp") else {
170                return Err(VoltagePatternError::read("missing boolean isthisvp"));
171            };
172            Ok(PrimaryBeam::Gaussian {
173                half_width_arcmin_ghz: width,
174                maximum_radius_arcmin_ghz: radius,
175                is_voltage: *is_voltage,
176            })
177        }
178        _ => Err(VoltagePatternError::read(format!(
179            "saved radial law {name} (#1307)"
180        ))),
181    }
182}

184fn quantity(record: &RecordValue, field: &str, unit: &str) -> Result<f64, VoltagePatternError> {
185    let Some(Value::Record(value)) = record.get(field) else {
186        return Err(VoltagePatternError::read(format!(
187            "missing quantity {field}"
188        )));
189    };
190    let Some(Value::Scalar(ScalarValue::Float64(number))) = value.get("value") else {
191        return Err(VoltagePatternError::read(format!(
192            "non-scalar quantity {field}"
193        )));
194    };
195    let Some(Value::Scalar(ScalarValue::String(saved_unit))) = value.get("unit") else {
196        return Err(VoltagePatternError::read("missing string unit"));
197    };
198    Quantity::new(*number, saved_unit)
199        .and_then(|q| q.get_value_in(&Unit::new(unit)?))
200        .map_err(VoltagePatternError::read)
201}
