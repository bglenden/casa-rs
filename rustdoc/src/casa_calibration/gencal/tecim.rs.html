20pub(crate) const HEIGHT_KEYWORD: &str = "IonosphereHeight(km)";
21pub(crate) const DEFAULT_HEIGHT_KM: f64 = 450.;

23pub(super) fn height_km(parameters: &[f64]) -> f64 {
24    match parameters {
25        [height] => *height,
26        _ => DEFAULT_HEIGHT_KM,
27    }
28}

30pub(super) fn generate(
31    request: &GencalRequest,
32    image: Option<&Path>,
33    mode: CalibrationOutputMode,
34) -> Result<GencalReport, GencalError> {
35    let image = image.ok_or_else(|| invalid("An existing tec map must be specified in infile"))?;
36    refuse_prior_input_overlap(request, image)?;
37    let ms = open_measurement_set(&request.measurement_set)?;
38    // specifycal validates these expressions when assembling its input, then
39    // the F Jones algorithm derives the entire time and antenna grid.
40    if !matches!(request.spw.as_str(), "" | "*") {
41        ms.resolve_spw_selector(&request.spw).map_err(invalid)?;
42    }
43    if !matches!(request.antenna.as_str(), "" | "*") {
44        ms.resolve_specified_antennas(&request.antenna)
45            .map_err(invalid)?;
46    }
47    let height = height_km(&request.parameter);
48    if !height.is_finite() {
49        return Err(invalid("Ionosphere height in kilometres must be finite"));
50    }
51    let mut source = TecImage::open(image)?;
52    let engine = MsCalEngine::for_calibration(&ms).map_err(invalid)?;
53    let samples = samples(&ms)?;
54    let mode = if mode == CalibrationOutputMode::Update {
55        CalibrationOutputMode::Replace
56    } else {
57        mode
58    };
59    let output = CalibrationOutput::prepare(&request.output_table, mode)?;
60    let rows = solution_rows(&engine, &mut source, &samples, height * 1000.)?;
61    write_prior_caltable(&ms, request, "F Jones", &rows, output, mode, true)
62}

64fn solution_rows(
65    engine: &MsCalEngine,
66    source: &mut TecImage,
67    samples: &[Sample],
68    height_m: f64,
69) -> Result<Vec<PriorCalRow>, GencalError> {
70    let first = samples
71        .first()
72        .ok_or_else(|| invalid("MeasurementSet has no TEC solution times"))?;
73    let seconds_in_day = 86_400.;
74    let origin = if source.absolute_time {
75        0.
76    } else {
77        (first.time / seconds_in_day).floor() * seconds_in_day
78    };
79    let mut rows = Vec::with_capacity(samples.len() * engine.num_antennas());
80    for sample in samples {
81        let field = sample.field as usize;
82        let direction = engine
83            .observation_direction_j2000(sample.time, field)
84            .map_err(invalid)?;
85        for antenna in 0..engine.num_antennas() {
86            let frame = engine
87                .antenna_frame(sample.time, antenna)
88                .map_err(invalid)?
89                .with_epoch(MEpoch::from_mjd_seconds(sample.time, EpochRef::UTC));
90            let point = ionospheric_pierce_point(height_m, &direction, &frame).map_err(invalid)?;
91            let [x, y, z] = point.as_itrf();
92            let longitude = y.atan2(x).to_degrees();
93            let latitude = z.atan2(x.hypot(y)).to_degrees();
94            let vertical = source.sample([longitude, latitude, sample.time - origin])?;
95            let (_, elevation) = engine.azel(sample.time, field, antenna).map_err(invalid)?;
96            let zenith = std::f64::consts::FRAC_PI_2 - elevation;
97            let line_of_sight = (f64::from(vertical) / zenith.cos()) as f32;
98            let electrons = (f64::from(line_of_sight) * 1.0e16) as f32;
99            if !electrons.is_finite() {
100                return Err(invalid(
101                    "TEC image and source elevation produced a nonfinite correction",
102                ));
103            }
104            rows.push(PriorCalRow {
105                time_seconds: sample.time,
106                interval_seconds: 0.,
107                field_id: sample.field,
108                spw_id: 0,
109                antenna_id: antenna as i32,
110                observation_id: sample.observation,
111                scan_number: sample.scan,
112                parameters: f32_column(&[electrons]),
113                snr_value: 100.,
114                preserved: None,
115            });
116        }
117    }
118    Ok(rows)
119}

121struct Sample {
122    array: i32,
123    scan: i32,
124    time: f64,
125    field: i32,
126    description: i32,
127    observation: i32,
128}
129
130fn samples(ms: &MeasurementSet) -> Result<Vec<Sample>, GencalError> {
131    let columns = ms
132        .main_table()
133        .required_scalar_columns_owned(&[
134            "ARRAY_ID",
135            "SCAN_NUMBER",
136            "TIME",
137            "FIELD_ID",
138            "DATA_DESC_ID",
139            "OBSERVATION_ID",
140        ])
141        .map_err(invalid)?;
142    let (
143        Int32(array),
144        Int32(scan),
145        Float64(time),
146        Int32(field),
147        Int32(description),
148        Int32(observation),
149    ) = (
150        &columns["ARRAY_ID"],
151        &columns["SCAN_NUMBER"],
152        &columns["TIME"],
153        &columns["FIELD_ID"],
154        &columns["DATA_DESC_ID"],
155        &columns["OBSERVATION_ID"],
156    )
157    else {
158        return Err(invalid("Invalid MAIN column types for TEC sampling"));
159    };
160    let mut samples = (0..time.len())
161        .map(|row| Sample {
162            array: array[row],
163            scan: scan[row],
164            time: time[row],
165            field: field[row],
166            description: description[row],
167            observation: observation[row],
168        })
169        .collect::<Vec<_>>();
170    // Match the visibility iterator's sorting and first field at a timestamp.
171    samples.sort_by(|left, right| {
172        (left.array, left.scan)
173            .cmp(&(right.array, right.scan))
174            .then_with(|| left.time.total_cmp(&right.time))
175            .then_with(|| (left.field, left.description).cmp(&(right.field, right.description)))
176    });
177    samples.dedup_by(|left, right| left.time == right.time);
178    Ok(samples)
179}

181struct TecImage {
182    image: PagedImage<f32>,
183    absolute_time: bool,
184    increment: [f64; 3],
185    block: Option<TecBlock>,
186}
187
188struct TecBlock {
189    start: [usize; 3],
190    world: Vec<f64>,
191    pixels: ArrayD<f32>,
192}
193
194impl TecImage {
195    fn open(path: &Path) -> Result<Self, GencalError> {
196        let image = PagedImage::<f32>::open(path).map_err(invalid)?;
197        let coordinates = image.coordinates().reference_value();
198        if coordinates.len() != 3
199            || image.shape().len() != 3
200            || image.shape().iter().any(|&n| n < 2)
201        {
202            return Err(invalid(
203                "TEC image must have longitude, latitude and time axes with at least two pixels each",
204            ));
205        }
206        let coordinates_system = image.coordinates();
207        let mut increment = [0.; 3];
208        for (axis, step) in increment.iter_mut().enumerate() {
209            let (owner, local) = coordinates_system
210                .world_axis_owner(axis)
211                .ok_or_else(|| invalid("TEC image is missing a world axis"))?;
212            *step = coordinates_system.coordinate(owner).increment()[local];
213            if !step.is_finite() || *step == 0. {
214                return Err(invalid(
215                    "TEC image axes must have finite nonzero increments",
216                ));
217            }
218        }
219        Ok(Self {
220            image,
221            absolute_time: coordinates[2] > 0.,
222            increment,
223            block: None,
224        })
225    }
226
227    fn sample(&mut self, world: [f64; 3]) -> Result<f32, GencalError> {
228        let pixel = self.image.coordinates().to_pixel(&world).map_err(invalid)?;
229        let mut start = [0; 3];
230        for axis in 0..3 {
231            let corner = pixel[axis].trunc();
232            if !corner.is_finite()
233                || corner < 0.0
234                || corner as usize >= self.image.shape()[axis] - 1
235            {
236                return Err(invalid(format!(
237                    "TEC image does not cover longitude, latitude and epoch {world:?}"
238                )));
239            }
240            start[axis] = corner as usize;
241        }
242        if self.block.as_ref().is_none_or(|block| block.start != start) {
243            let pixels = self.image.get_slice(&start, &[2; 3]).map_err(invalid)?;
244            let point = start.map(|index| index as f64);
245            let world = self.image.coordinates().to_world(&point).map_err(invalid)?;
246            self.block = Some(TecBlock {
247                start,
248                world,
249                pixels,
250            });
251        }
252        let block = self.block.as_ref().expect("TEC interpolation block");
253        let values = &block.pixels;
254        let [x, y, t] = std::array::from_fn(|axis| {
255            f64::from(((world[axis] - block.world[axis]) / self.increment[axis]) as f32)
256        });
257        let plane = |time| {
258            ((1. - x) * (1. - y) * f64::from(values[[0, 0, time]])
259                + x * (1. - y) * f64::from(values[[1, 0, time]])
260                + x * y * f64::from(values[[1, 1, time]])
261                + (1. - x) * y * f64::from(values[[0, 1, time]])) as f32
262        };
263        Ok(((1. - t) * f64::from(plane(0)) + t * f64::from(plane(1))) as f32)
264    }
265}
