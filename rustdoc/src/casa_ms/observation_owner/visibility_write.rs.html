49/// Bounded in-place selected-visibility writer following ordinary casacore semantics.
50///
51/// The writer holds the MAIN write lock until it completes. Completion flushes
52/// every selected destination and releases the lock. A failure may leave
53/// partially written derived values, as an interrupted CASA write does; the
54/// next run recomputes them. No backup, staging column, rollback, snapshot,
55/// marker, or content digest is created.
56///
57/// Only destinations stored in `TiledShapeStMan` or `TiledColumnStMan`, the
58/// managers CASA and casa-rs store visibilities in, are written: their save
59/// patches the written rows' tiles in place. Any other manager's save
60/// rewrites its whole group from loaded cells, which holds every row of
61/// MAIN (ADR-0008), so [`retained_bytes`](Self::retained_bytes) and
62/// [`begin`](Self::begin) refuse such a destination before anything is
63/// changed.
64#[cfg(unix)]
65pub struct SelectedVisibilityWrite {
66    measurement_set: Option<MeasurementSet>,
67    targets: SelectedVisibilityWriteTargets,
68    pending_cell: Option<PendingVisibilityCells>,
69    pending_rows: Vec<usize>,
70    completed: bool,
71}
72
73struct PendingVisibilityCells {
74    row: usize,
75    model_data: Option<ndarray::ArrayD<Complex32>>,
76    corrected_data: Option<ndarray::ArrayD<Complex32>>,
77}
78
79#[cfg(unix)]
80impl SelectedVisibilityWrite {
81    /// Take the MAIN write lock and start an in-place write. `MODEL_DATA` is
82    /// created when MAIN lacks it and filled with CASA's initial model, as
83    /// tclean creates it ([`initial_model_visibility`]: 1, and 0 in the
84    /// cross hands), so rows the run does not predict hold that model.
85    ///
86    /// The MeasurementSet is opened with retained read locks, and MAIN's is
87    /// then upgraded to the write lock. Both wait, as casacore's do, for a
88    /// conflicting lock another process holds. When another process holding
89    /// a read lock waits to upgrade it too, the kernel refuses one of the
90    /// two waits; if it is this one, the write fails with
91    /// [`ObservationOwnerError::WriteLockUnavailable`] and releases its read
92    /// locks.
93    ///
94    /// A destination that cannot be written in bounded memory fails with
95    /// [`ObservationOwnerError::UnboundedWriteDestination`] before the lock
96    /// is taken.
97    pub fn begin(
98        path: impl AsRef<Path>,
99        targets: SelectedVisibilityWriteTargets,
100    ) -> Result<Self, ObservationOwnerError> {
101        if !targets.model_data && !targets.corrected_data {
102            return Err(ObservationOwnerError::EmptyWriteTargets);
103        }
104        let mut measurement_set = MeasurementSet::open_retained_read(path.as_ref())?;
105        require_bounded_destinations(measurement_set.main_table(), targets)?;
106        if !measurement_set.main_table_mut().lock(LockType::Write, 0)? {
107            return Err(ObservationOwnerError::WriteLockUnavailable);
108        }
109        let has_model = {
110            let schema = measurement_set.main_table().schema().ok_or_else(|| {
111                MsError::InvalidInput("MeasurementSet MAIN has no schema".to_string())
112            })?;
113            schema.contains_column("MODEL_DATA")
114        };
115        if targets.model_data && !has_model {
116            // Everything that could refuse the seeding is checked before the
117            // column exists: a refusal after it would leave MODEL_DATA
118            // holding DATA's clone, which a later write would not reseed.
119            let models = initial_models(&measurement_set)?;
120            let row_count = measurement_set.main_table().row_count();
121            let batch = VISIBILITY_WRITE_BATCH_ROWS as usize;
122            for start in (0..row_count).step_by(batch) {
123                let rows: Vec<usize> = (start..row_count.min(start + batch)).collect();
124                let descriptions = main_data_description_ids(&measurement_set, &rows)?;
125                if let Some((row, id)) = rows
126                    .iter()
127                    .zip(descriptions)
128                    .find(|(_, id)| *id >= models.len())
129                {
130                    return Err(MsError::InvalidInput(format!(
131                        "MAIN row {row} names DATA_DESC_ID {id}, which DATA_DESCRIPTION \
132                         does not hold"
133                    ))
134                    .into());
135                }
136            }
137            measurement_set.main_table_mut().add_column(
138                ColumnSchema::array_variable("MODEL_DATA", PrimitiveType::Complex32, Some(2)),
139                None,
140            )?;
141            measurement_set
142                .main_table_mut()
143                .prepare_write()
144                .add_tiled_column_clone("DATA", "MODEL_DATA", "TiledModelData")?;
145            for start in (0..row_count).step_by(batch) {
146                let rows: Vec<usize> = (start..row_count.min(start + batch)).collect();
147                let descriptions = main_data_description_ids(&measurement_set, &rows)?;
148                for (&row, description) in rows.iter().zip(descriptions) {
149                    queue_visibility_cell(
150                        measurement_set.main_table_mut(),
151                        "MODEL_DATA",
152                        row,
153                        models[description].cell(),
154                    )?;
155                }
156                persist_visibility_rows(measurement_set.main_table_mut(), &["MODEL_DATA"], &rows)?;
157            }
158        }
159        Ok(Self {
160            measurement_set: Some(measurement_set),
161            targets,
162            pending_cell: None,
163            pending_rows: Vec::with_capacity(VISIBILITY_WRITE_BATCH_ROWS as usize),
164            completed: false,
165        })
166    }
167
168    /// Heap bytes a write of `targets` to the MeasurementSet at `path`
169    /// holds at most, which the pass that writes admits before it begins.
170    ///
171    /// Per written column, that is a batch of queued cells of the
172    /// MeasurementSet's largest cell, the queue's list of them, the row
173    /// being written, and, while a batch is saved, the batch's cells twice
174    /// more (gathered for the save and staged by tile) with the column's tile
175    /// read and its tile written; the list holds its old and new buffers at
176    /// once while it grows. Beside them are the batch's row numbers and, while a missing
177    /// `MODEL_DATA` is created, their data-description ids as read and as
178    /// indices, and each data description's initial model (a value per
179    /// product). Tiles cached for reuse go to the process-wide table cache,
180    /// charged on its own.
181    ///
182    /// # Errors
183    ///
184    /// When the MeasurementSet's spectral-window, polarization or table
185    /// descriptions cannot be read, and, as [`begin`](Self::begin) would,
186    /// when a destination cannot be written in bounded memory or
187    /// `CORRECTED_DATA` is missing: the pass that writes then fails before
188    /// it runs.
189    pub fn retained_bytes(
190        path: impl AsRef<Path>,
191        targets: SelectedVisibilityWriteTargets,
192    ) -> Result<u64, ObservationOwnerError> {
193        /// Staging per batch row: the save's sorted row list and the cells
194        /// it gathers by tile; measured at about 140 bytes.
195        const STAGING_BYTES_PER_ROW: u64 = 192;
196        /// Staging whatever the batch: the MeasurementSet opened for the
197        /// write and the table description each save reads; measured at
198        /// about 240 KB.
199        const STAGING_BYTES: u64 = 1 << 20;
200        let measurement_set = MeasurementSet::open(path.as_ref())?;
201        let descriptions = measurement_set.data_description()?.row_count();
202        let (mut cell_values, mut products) = (0, 0);
203        for description in 0..descriptions {
204            let [correlations, channels] = model_cell_shape(&measurement_set, description)?;
205            cell_values = cell_values.max(correlations * channels);
206            products = products.max(correlations);
207        }
208        let main = measurement_set.main_table();
209        require_bounded_destinations(main, targets)?;
210        let tile = |column: &'static str| -> Result<u64, ObservationOwnerError> {
211            // A created MODEL_DATA clones DATA's tiled manager.
212            let stored = if column == "MODEL_DATA"
213                && !main
214                    .schema()
215                    .is_some_and(|schema| schema.contains_column(column))
216            {
217                "DATA"
218            } else {
219                column
220            };
221            let bytes = main.array_column_tile_bytes(stored)?.ok_or_else(|| {
222                ObservationOwnerError::UnboundedWriteDestination {
223                    column,
224                    reason: format!("{stored} has no tiles on disk"),
225                }
226            })?;
227            Ok(bytes as u64)
228        };
229        let mut tiles = 0;
230        if targets.model_data {
231            tiles += 2 * tile("MODEL_DATA")?;
232        }
233        if targets.corrected_data {
234            tiles += 2 * tile("CORRECTED_DATA")?;
235        }
236        let cell = (cell_values * size_of::<Complex32>()) as u64;
237        let batch = VISIBILITY_WRITE_BATCH_ROWS;
238        let entry = size_of::<(usize, ArrayValue)>() as u64;
239        let queue = 3 * batch * cell + batch.next_power_of_two() * entry * 3 / 2;
240        let columns = u64::from(targets.model_data) + u64::from(targets.corrected_data);
241        let rows = batch * (size_of::<usize>() as u64 + STAGING_BYTES_PER_ROW);
242        let creation = if targets.model_data {
243            let models =
244                descriptions * (size_of::<InitialModel>() + products * size_of::<Complex32>());
245            batch * (size_of::<Option<ScalarValue>>() + size_of::<usize>()) as u64 + models as u64
246        } else {
247            0
248        };
249        Ok(columns * (queue + cell) + tiles + rows + creation + STAGING_BYTES)
250    }

252    /// Write one selected prediction at its physical row/channel/correlation.
253    pub fn write(
254        &mut self,
255        column: MsColumnKind,
256        row: u64,
257        channel: u32,
258        correlation: u32,
259        value: Complex32,
260    ) -> Result<(), ObservationOwnerError> {
261        let row = usize::try_from(row).map_err(|_| ObservationOwnerError::PredictionAddress)?;
262        if self.pending_cell.as_ref().map(|cell| cell.row) != Some(row) {
263            self.flush_pending_cell()?;
264            let measurement_set = self
265                .measurement_set
266                .as_ref()
267                .ok_or(ObservationOwnerError::TransactionClosed)?;
268            // The row's one cell, read past the table's row cache: a cached
269            // read would load every row of MAIN (ADR-0008).
270            let load = |name: &str| -> Result<ndarray::ArrayD<Complex32>, ObservationOwnerError> {
271                let current = measurement_set
272                    .main_table()
273                    .column_accessor(name)?
274                    .array_cells_owned_uncached(&[row])?
275                    .pop()
276                    .flatten()
277                    .ok_or(ObservationOwnerError::PredictionAddress)?;
278                let ArrayValue::Complex32(values) = current else {
279                    return Err(ObservationOwnerError::PredictionAddress);
280                };
281                Ok(values)
282            };
283            self.pending_cell = Some(PendingVisibilityCells {
284                row,
285                model_data: self
286                    .targets
287                    .model_data
288                    .then(|| load("MODEL_DATA"))
289                    .transpose()?,
290                corrected_data: self
291                    .targets
292                    .corrected_data
293                    .then(|| load("CORRECTED_DATA"))
294                    .transpose()?,
295            });
296        }
297        let pending = self
298            .pending_cell
299            .as_mut()
300            .ok_or(ObservationOwnerError::TransactionClosed)?;
301        let values = match column {
302            MsColumnKind::ModelData if self.targets.model_data => pending.model_data.as_mut(),
303            MsColumnKind::CorrectedData if self.targets.corrected_data => {
304                pending.corrected_data.as_mut()
305            }
306            _ => None,
307        }
308        .ok_or(ObservationOwnerError::UnselectedWriteTarget)?;
309        let index = [correlation as usize, channel as usize];
310        let Some(cell) = values.get_mut(index) else {
311            return Err(ObservationOwnerError::PredictionAddress);
312        };
313        *cell = value;
314        Ok(())
315    }
316
317    /// Flush the in-place write and release the MAIN write lock.
318    ///
319    /// The selected cells were persisted in bounded row batches, so only the
320    /// table metadata (a newly created `MODEL_DATA`) remains to flush.
321    pub fn complete(mut self) -> Result<(), ObservationOwnerError> {
322        self.flush_pending_cell()?;
323        self.persist_pending_rows()?;
324        let measurement_set = self
325            .measurement_set
326            .as_mut()
327            .ok_or(ObservationOwnerError::TransactionClosed)?;
328        measurement_set.main_table_mut().unlock_metadata_only()?;
329        self.measurement_set = None;
330        self.completed = true;
331        Ok(())
332    }
333
334    fn flush_pending_cell(&mut self) -> Result<(), ObservationOwnerError> {
335        let Some(cell) = self.pending_cell.take() else {
336            return Ok(());
337        };
338        let measurement_set = self
339            .measurement_set
340            .as_mut()
341            .ok_or(ObservationOwnerError::TransactionClosed)?;
342        if let Some(values) = cell.model_data {
343            queue_visibility_cell(
344                measurement_set.main_table_mut(),
345                "MODEL_DATA",
346                cell.row,
347                ArrayValue::Complex32(values),
348            )?;
349        }
350        if let Some(values) = cell.corrected_data {
351            queue_visibility_cell(
352                measurement_set.main_table_mut(),
353                "CORRECTED_DATA",
354                cell.row,
355                ArrayValue::Complex32(values),
356            )?;
357        }
358        self.pending_rows.push(cell.row);
359        if self.pending_rows.len() == VISIBILITY_WRITE_BATCH_ROWS as usize {
360            self.persist_pending_rows()?;
361        }
362        Ok(())
363    }
364
365    fn persist_pending_rows(&mut self) -> Result<(), ObservationOwnerError> {
366        if self.pending_rows.is_empty() {
367            return Ok(());
368        }
369        let columns = match (self.targets.model_data, self.targets.corrected_data) {
370            (true, true) => &["MODEL_DATA", "CORRECTED_DATA"][..],
371            (true, false) => &["MODEL_DATA"][..],
372            (false, true) => &["CORRECTED_DATA"][..],
373            (false, false) => return Err(ObservationOwnerError::EmptyWriteTargets),
374        };
375        let measurement_set = self
376            .measurement_set
377            .as_mut()
378            .ok_or(ObservationOwnerError::TransactionClosed)?;
379        persist_visibility_rows(
380            measurement_set.main_table_mut(),
381            columns,
382            &self.pending_rows,
383        )?;
384        self.pending_rows.clear();
385        Ok(())
386    }
387}

389#[cfg(unix)]
390impl Drop for SelectedVisibilityWrite {
391    /// An abandoned write releases the MAIN write lock without flushing the
392    /// cells still queued: the batches persisted so far stay, as after an
393    /// interrupted CASA write, and MAIN is not rewritten.
394    fn drop(&mut self) {
395        if !self.completed {
396            if let Some(measurement_set) = self.measurement_set.as_mut() {
397                let _ = measurement_set.main_table_mut().unlock_metadata_only();
398            }
399            self.measurement_set = None;
400        }
401    }
402}

404/// Storage managers whose selected-row save patches the written rows' tiles
405/// in place: the managers CASA and casa-rs store visibilities in.
406#[cfg(unix)]
407const BOUNDED_WRITE_MANAGERS: [&str; 2] = ["TiledShapeStMan", "TiledColumnStMan"];
408
409/// Refuse, before anything is changed, a write whose save would not be
410/// bounded: an existing destination stored outside
411/// [`BOUNDED_WRITE_MANAGERS`], or a missing `MODEL_DATA` whose clone of
412/// DATA's manager would be or cannot be made. A missing `CORRECTED_DATA` is
413/// refused too, since it holds the output of a continuum transform.
414#[cfg(unix)]
415fn require_bounded_destinations(
416    main: &Table,
417    targets: SelectedVisibilityWriteTargets,
418) -> Result<(), ObservationOwnerError> {
419    let has = |column: &str| {
420        main.schema()
421            .is_some_and(|schema| schema.contains_column(column))
422    };
423    if targets.corrected_data && !has("CORRECTED_DATA") {
424        return Err(ObservationOwnerError::MissingCorrectedDataDestination);
425    }
426    let destinations = [
427        ("MODEL_DATA", targets.model_data),
428        ("CORRECTED_DATA", targets.corrected_data),
429    ];
430    for (column, written) in destinations {
431        if !written {
432            continue;
433        }
434        let created = !has(column);
435        let stored = if created { "DATA" } else { column };
436        let Some(manager) = main
437            .data_manager_info()
438            .iter()
439            .find(|manager| manager.columns.iter().any(|name| name == stored))
440        else {
441            return Err(ObservationOwnerError::UnboundedWriteDestination {
442                column,
443                reason: format!("{stored} has no storage manager on disk"),
444            });
445        };
446        let storage = if created {
447            format!(
448                "it would be created in a clone of DATA's {}",
449                manager.dm_type
450            )
451        } else {
452            format!("it is stored in {}", manager.dm_type)
453        };
454        if !BOUNDED_WRITE_MANAGERS.contains(&manager.dm_type.as_str()) {
455            return Err(ObservationOwnerError::UnboundedWriteDestination {
456                column,
457                reason: format!(
458                    "{storage}; only TiledShapeStMan and TiledColumnStMan columns are written in place"
459                ),
460            });
461        }
462        if created && manager.columns.len() != 1 {
463            return Err(ObservationOwnerError::UnboundedWriteDestination {
464                column,
465                reason: format!("{storage}, which DATA shares with other columns"),
466            });
467        }
468    }
469    Ok(())
470}

472/// The `DATA_DESC_ID` of each of `rows`, read without filling the table's
473/// row cache.
474fn main_data_description_ids(
475    measurement_set: &MeasurementSet,
476    rows: &[usize],
477) -> Result<Vec<usize>, ObservationOwnerError> {
478    measurement_set
479        .main_table()
480        .column_accessor("DATA_DESC_ID")?
481        .scalar_cells_owned_for_rows(rows)?
482        .into_iter()
483        .map(|value| match value {
484            Some(ScalarValue::Int32(id)) => {
485                usize::try_from(id).map_err(|_| ObservationOwnerError::PredictionAddress)
486            }
487            _ => Err(ObservationOwnerError::PredictionAddress),
488        })
489        .collect()
490}

492/// CASA's initial `MODEL_DATA` for one data description: each product's
493/// [`initial_model_visibility`], in `CORR_TYPE` order, over its channels.
494struct InitialModel {
495    products: Vec<Complex32>,
496    channels: usize,
497}
498
499impl InitialModel {
500    /// The `[correlation, channel]` cell.
501    fn cell(&self) -> ArrayValue {
502        ArrayValue::Complex32(ndarray::ArrayD::from_shape_fn(
503            ndarray::IxDyn(&[self.products.len(), self.channels]),
504            |at| self.products[at[0]],
505        ))
506    }
507}
508
509/// Every `DATA_DESCRIPTION` row's [`InitialModel`], by row.
510///
511/// # Errors
512///
513/// A description whose window or polarization cannot be read, or whose
514/// polarization's `CORR_TYPE` does not list `NUM_CORR` products.
515fn initial_models(
516    measurement_set: &MeasurementSet,
517) -> Result<Vec<InitialModel>, ObservationOwnerError> {
518    let descriptions = measurement_set.data_description()?;
519    let polarizations = measurement_set.polarization()?;
520    (0..descriptions.row_count())
521        .map(|description| {
522            let [correlations, channels] = model_cell_shape(measurement_set, description)?;
523            let polarization = descriptions.polarization_id(description)?;
524            let types = polarizations.corr_type(
525                usize::try_from(polarization)
526                    .map_err(|_| ObservationOwnerError::PredictionAddress)?,
527            )?;
528            if types.len() != correlations {
529                return Err(MsError::InvalidInput(format!(
530                    "POLARIZATION row {polarization} has NUM_CORR {correlations} but {} \
531                     CORR_TYPE entries",
532                    types.len()
533                ))
534                .into());
535            }
536            Ok(InitialModel {
537                products: types.into_iter().map(initial_model_visibility).collect(),
538                channels,
539            })
540        })
541        .collect()
542}
543
544fn model_cell_shape(
545    measurement_set: &MeasurementSet,
546    data_description_id: usize,
547) -> Result<[usize; 2], ObservationOwnerError> {
548    let data_description = measurement_set.data_description()?;
549    let spectral_window_id =
550        usize::try_from(data_description.spectral_window_id(data_description_id)?)
551            .map_err(|_| ObservationOwnerError::PredictionAddress)?;
552    let polarization_id = usize::try_from(data_description.polarization_id(data_description_id)?)
553        .map_err(|_| ObservationOwnerError::PredictionAddress)?;
554    let channels = usize::try_from(
555        measurement_set
556            .spectral_window()?
557            .num_chan(spectral_window_id)?,
558    )
559    .map_err(|_| ObservationOwnerError::PredictionAddress)?;
560    let correlations = usize::try_from(measurement_set.polarization()?.num_corr(polarization_id)?)
561        .map_err(|_| ObservationOwnerError::PredictionAddress)?;
562    Ok([correlations, channels])
563}

565#[cfg(unix)]
566fn queue_visibility_cell(
567    table: &mut Table,
568    column: &str,
569    row: usize,
570    value: ArrayValue,
571) -> Result<(), ObservationOwnerError> {
572    {
573        let mut prepared = table.row_accessor_mut().prepare(&[column])?;
574        prepared.seek(row)?;
575        prepared.set_value_at(0, Value::Array(value))?;
576    }
577    Ok(())
578}

580#[cfg(unix)]
581fn persist_visibility_rows(
582    table: &mut Table,
583    columns: &[&str],
584    rows: &[usize],
585) -> Result<(), ObservationOwnerError> {
586    table.prepare_write().save_selected_rows(columns, rows)?;
587    table.discard_persisted_cell_updates(columns, rows);
588    Ok(())
589}

679        let cells = main
680            .column_accessor("MODEL_DATA")
681            .and_then(|column| column.array_cells_owned_uncached(&rows))
682            .expect("MODEL_DATA");
683        let expected = |description: usize| match description {
684            0 => [1.0, 0.0, 0.0, 1.0],
685            _ => [1.0, 1.0, 0.0, 0.0],

687        let mut seen = [0, 0];

689            let Some(ArrayValue::Complex32(cell)) = cell else {
690                panic!("row {row} has a MODEL_DATA cell");

693            for ((correlation, _), value) in cell.indexed_iter().map(|(at, v)| ((at[0], at[1]), v))
694            {
695                assert_eq!(
696                    *value,
697                    Complex32::new(expected(description)[correlation], 0.0),
698                    "row {row}, description {description}, product {correlation}"
699                );
700            }
701            seen[description] += 1;
702        }
703        assert!(
704            seen[0] > 0 && seen[1] > 0,
705            "rows of both descriptions: {seen:?}"
706        );

709    /// A description the initial model cannot be built for (here a
710    /// polarization whose `NUM_CORR` is not its `CORR_TYPE`'s length) refuses
711    /// the write before `MODEL_DATA` exists, so no half-seeded column holding
712    /// `DATA` is left for a later write to take as seeded.
713    #[test]
714    fn an_unseedable_model_is_refused_before_model_data_exists() {
715        let root = tempfile::tempdir().expect("root");
716        let path = root.path().join("inconsistent.ms");
717        let mut request = SyntheticObservationRequest::vla_ppdisk(
718            root.path().join("unused.fits"),
719            &path,
720            tutorial_vla_a_antennas(),
721        );
722        request.predict_model = false;
723        request.duration_seconds = 4.0;
724        request.integration_seconds = 2.0;
725        generate_synthetic_observation_ms(&request).expect("simulate");
726        let mut ms = MeasurementSet::open(&path).expect("open");
727        ms.polarization_mut()
728            .expect("POLARIZATION")
729            .set_i32(0, "NUM_CORR", 3)
730            .expect("an inconsistent NUM_CORR");
731        ms.save().expect("save");
732        drop(ms);
733        let refused =
734            SelectedVisibilityWrite::begin(&path, SelectedVisibilityWriteTargets::new(true, false));
735        assert!(
736            matches!(&refused, Err(error) if error.to_string().contains("CORR_TYPE")),
737            "{:?}",
738            refused.err()
739        );
740        let ms = MeasurementSet::open(&path).expect("reopen");
741        assert!(
742            !ms.main_table()
743                .schema()
744                .expect("schema")
745                .contains_column("MODEL_DATA"),
746            "no MODEL_DATA was added"
747        );
