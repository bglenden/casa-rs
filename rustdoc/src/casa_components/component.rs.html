2//! One sky component, CASA's `SkyComponent`.
3
4use crate::component_type::ShapeType;
5use crate::flux::Flux;
6use crate::shape::ComponentShape;
7use crate::spectrum::SpectralModel;
8
9/// The limb-darkening exponent CASA's `LimbDarkenedDiskShape` starts with.
10pub const DEFAULT_LIMB_DARKENING_EXPONENT: f64 = 0.05;
11
12/// One component of a sky model: a flux, a shape, a spectral model, a label
13/// and optional parameters.
14///
15/// This is CASA's `SkyComponent` (`SkyCompRep`). The flux is the integrated
16/// flux at the spectral model's reference frequency; the shape places it on
17/// the sky and spreads it; the spectrum scales it with frequency.
18///
19/// # Optional parameters
20///
21/// CASA keeps a vector of optional parameters with the component, used today
22/// only by the limb-darkened disk, whose first optional parameter is its
23/// limb-darkening exponent n. A component list stores the vector in its
24/// `Optional_Parameters` column, which CASA adds only to lists holding a
25/// limb-darkened disk and fills only for those rows. CASA also copies the
26/// exponent into the shape when it reads a table; casa-rs keeps the one copy
27/// the table persists, here, and derives the exponent from it
28/// ([`Self::limb_darkening_exponent`]).
29#[derive(Debug, Clone, PartialEq)]
30pub struct SkyComponent {
31    flux: Flux,
32    shape: ComponentShape,
33    spectrum: SpectralModel,
34    label: String,
35    optional_parameters: Vec<f64>,
36}
37
38impl SkyComponent {
39    /// A component with an empty label and no optional parameters
40    /// (`SkyComponent(flux, shape, spectrum)`).
41    pub fn new(flux: Flux, shape: ComponentShape, spectrum: SpectralModel) -> Self {
42        Self {
43            flux,
44            shape,
45            spectrum,
46            label: String::new(),
47            optional_parameters: Vec::new(),
48        }
49    }
50
51    /// This component with its label replaced.
52    pub fn with_label(mut self, label: impl Into<String>) -> Self {
53        self.label = label.into();
54        self
55    }
56
57    /// This component with its optional parameters replaced
58    /// (`optionalParameters()`).
59    pub fn with_optional_parameters(mut self, optional_parameters: Vec<f64>) -> Self {
60        self.optional_parameters = optional_parameters;
61        self
62    }
63
64    /// The flux.
65    pub fn flux(&self) -> &Flux {
66        &self.flux
67    }
68
69    /// Mutable access to the flux.
70    pub fn flux_mut(&mut self) -> &mut Flux {
71        &mut self.flux
72    }
73
74    /// The shape.
75    pub fn shape(&self) -> &ComponentShape {
76        &self.shape
77    }
78
79    /// Mutable access to the shape.
80    pub fn shape_mut(&mut self) -> &mut ComponentShape {
81        &mut self.shape
82    }
83
84    /// The spectral model.
85    pub fn spectrum(&self) -> &SpectralModel {
86        &self.spectrum
87    }
88
89    /// Mutable access to the spectral model.
90    pub fn spectrum_mut(&mut self) -> &mut SpectralModel {
91        &mut self.spectrum
92    }
93
94    /// The user's label (`label()`).
95    pub fn label(&self) -> &str {
96        &self.label
97    }
98
99    /// Replace the label.
100    pub fn set_label(&mut self, label: impl Into<String>) {
101        self.label = label.into();
102    }
103
104    /// The optional parameters (`optionalParameters()`).
105    pub fn optional_parameters(&self) -> &[f64] {
106        &self.optional_parameters
107    }
108
109    /// Replace the optional parameters.
110    pub fn set_optional_parameters(&mut self, optional_parameters: Vec<f64>) {
111        self.optional_parameters = optional_parameters;
112    }
113
114    /// A limb-darkened disk's exponent n: the first optional parameter, or
115    /// [`DEFAULT_LIMB_DARKENING_EXPONENT`] when there is none. `None` for
116    /// other shapes.
117    ///
118    /// CASA's shape holds n as a single-precision float; prediction should
119    /// round it the same way.
120    pub fn limb_darkening_exponent(&self) -> Option<f64> {
121        (self.shape.shape_type() == ShapeType::LimbDarkenedDisk).then(|| {
122            self.optional_parameters
123                .first()
124                .copied()
125                .unwrap_or(DEFAULT_LIMB_DARKENING_EXPONENT)
126        })
127    }
128}
