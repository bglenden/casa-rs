3use std::cell::RefCell;

10fn open_read_locked_table(path: &Path, attempts: u32) -> Result<(Table, bool), ImageError> {
11    if TableUseLock::readonly_without_lock_file(path) {
12        return Ok((Table::open(TableOptions::new(path))?, true));
13    }
14    let table = Table::open_with_lock_attempts(
15        TableOptions::new(path),
16        LockOptions::new(LockMode::AutoLocking),
17        attempts,
18    )?;
19    Ok((table, false))
20}

22impl AnyPagedImage {
23    /// Opens an existing image and retains a read lock on its table.
24    ///
25    /// Uses casacore-compatible locking and keeps the tiled payload lazy.
26    /// The read lock is acquired before reading metadata. Applications can
27    /// acquire a write lock before a mutation with [`PagedImage::lock`].
28    pub fn open_with_lock(path: impl AsRef<Path>, attempts: u32) -> Result<Self, ImageError> {
29        let path = path.as_ref().to_path_buf();
30        let (table, advisory_locks) = open_read_locked_table(&path, attempts)?;
31        let kind = ImagePixelType::from_primitive_type(
32            PagedImage::<f32>::map_column_primitive_type(&table, None)?,
33        )?;
34        match kind {
35            ImagePixelType::Float32 => {
36                PagedImage::from_locked_disk(path, table, 0, advisory_locks).map(Self::Float32)
37            }
38            ImagePixelType::Float64 => {
39                PagedImage::from_locked_disk(path, table, 0, advisory_locks).map(Self::Float64)
40            }
41            ImagePixelType::Complex32 => {
42                PagedImage::from_locked_disk(path, table, 0, advisory_locks).map(Self::Complex32)
43            }
44            ImagePixelType::Complex64 => {
45                PagedImage::from_locked_disk(path, table, 0, advisory_locks).map(Self::Complex64)
46            }
47        }
48    }
49}

51impl<T: ImagePixel> PagedImage<T> {
52    pub(crate) fn open_read_locked_with_cache(
53        path: &Path,
54        max_cache_bytes: usize,
55    ) -> Result<Self, ImageError> {
56        let (table, advisory_locks) = open_read_locked_table(path, 0)?;
57        Self::from_locked_disk(path.to_path_buf(), table, max_cache_bytes, advisory_locks)
58    }
59
60    fn from_locked_disk(
61        path: PathBuf,
62        table: Table,
63        max_cache_bytes: usize,
64        advisory_locks: bool,
65    ) -> Result<Self, ImageError> {
66        let tiled_io = Self::open_tiled_io(&path, max_cache_bytes)
67            .ok()
68            .map(RefCell::new);
69        let mut image = Self::from_disk(path, table, tiled_io, max_cache_bytes)?;
70        image.advisory_locks = advisory_locks;
71        Ok(image)
72    }
73
74    /// Acquires a casacore-compatible table lock on a locking-enabled image.
75    ///
76    /// See [`Table::lock`] for the meaning of `nattempts`. A new table
77    /// generation refreshes the coordinates and invalidates image caches.
78    pub fn lock(&mut self, kind: LockType, nattempts: u32) -> Result<bool, ImageError> {
79        if self.advisory_locks {
80            return Ok(true);
81        }
82        if self.has_lock(kind) {
83            self.table.resync_locked()?;
84        } else if !self.table.lock(kind, nattempts)? {
85            return Ok(false);
86        }
87        let counter = self.table.locked_modify_counter()?;
88        if self.cached_modify_counter != Some(counter) {
89            self.refresh_image_cache()?;
90            self.cached_modify_counter = Some(counter);
91        }
92        Ok(true)
93    }
94
95    /// Reports whether this handle holds the given lock.
96    pub fn has_lock(&self, kind: LockType) -> bool {
97        self.advisory_locks || self.table.has_lock(kind)
98    }
99
100    /// Flushes a held write lock and releases this handle's table lock.
101    pub fn unlock(&mut self) -> Result<(), ImageError> {
102        if self.advisory_locks {
103            return Ok(());
104        }
105        if self.has_lock(LockType::Write) {
106            if self.pending_tile_writes || self.table.holds_unflushed_changes() {
107                self.save()?;
108            }
109            self.table.publish_writes()?;
110            self.cached_modify_counter = Some(self.table.locked_modify_counter()?);
111        }
112        if self.tiled_io.is_some() {
113            self.table.unlock_metadata_only()?;
114        } else {
115            self.table.unlock()?;
116        }
117        Ok(())
118    }
119
120    /// Yields a held automatic lock if another process requested it.
121    ///
122    /// Checks the casacore request list at the table's inspection interval.
123    pub fn release_requested_lock(&mut self) -> Result<(), ImageError> {
124        if self.table.auto_lock_release_requested()? {
125            self.unlock()?;
126        }
127        Ok(())
128    }
129
130    fn refresh_image_cache(&mut self) -> Result<(), ImageError> {
131        let (coords, units, misc_info) = Self::read_table_metadata(&self.table)?;
132        if let (Some(path), Some(tile)) = (&self.path, &mut self.tiled_io) {
133            let refreshed = Self::open_tiled_io(path, self.max_cache_bytes.get())?;
134            self.shape = refreshed.cube_shape().to_vec();
135            self.tile_shape = refreshed.tile_shape().to_vec();
136            *tile.get_mut() = refreshed;
137        } else {
138            self.shape = Self::map_column_shape(&self.table)?;
139        }
140        self.coords = coords;
141        self.units = units;
142        self.misc_info = misc_info;
143        self.persistent_masks.get_mut().clear();
144        Ok(())
145    }
146}
