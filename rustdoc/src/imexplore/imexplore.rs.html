2//! `imexplore` - the `imhead` and `imstat` task provider for persistent
3//! casacore images, on the shared task front end. Its first word names the
4//! surface (`imexplore imhead ...`, `imexplore imstat ...`); `imhead` runs
5//! otherwise.
6
7use std::fmt::{Display, Write as _};
8use std::path::Path;
9use std::process::ExitCode;
10
11use casa_images::{
12    ImageHeaderSummary, ImageStatisticsSummary, image_task_front_end, imhead, imhead_put, imstat,
13};
14use casa_task_runtime::{TaskFailure, TaskInvocation, TaskSuccess};

16fn main() -> ExitCode {
17    image_task_front_end("imexplore", &["imhead", "imstat"], run).main(
18        std::env::args_os().skip(1),
19        casa_task_runtime::CasaLoggingConfig::from_env(),
20        std::env::set_current_dir,
21        &mut std::io::stdout(),
22        std::io::stderr(),
23    )
24}

26fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
27    match task.surface() {
28        "imstat" => run_imstat(task),
29        _ => run_imhead(task),
30    }
31}
32
33/// CASA's `imhead` returns its header record, as `imstat` returns its
34/// statistics record; both are the reply's value.
35fn run_imhead(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
36    let path = task.path("imagename")?;
37    let mode = task.required_text("mode")?.to_ascii_lowercase();
38    let summary = match mode.as_str() {
39        "summary" | "list" => imhead(&path).map_err(|error| image_error("imhead", &path, error))?,
40        "put" => {
41            let hdkey = task.text("hdkey")?.ok_or_else(|| {
42                TaskFailure::validation("imhead mode put requires hdkey").at("hdkey")
43            })?;
44            let hdvalue = task.text("hdvalue")?.ok_or_else(|| {
45                TaskFailure::validation("imhead mode put requires hdvalue").at("hdvalue")
46            })?;
47            imhead_put(&path, &hdkey, &hdvalue)
48                .map_err(|error| image_error("imhead", &path, error))?
49        }
50        other => {
51            return Err(TaskFailure::validation(format!(
52                "unsupported imhead mode {other:?}; expected summary, list, or put"
53            ))
54            .at("mode"));
55        }
56    };
57    let text = header_text(&summary);
58    Ok(TaskSuccess::new(&summary, &summary)?.with_text(text))
59}
60
61fn run_imstat(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
62    let path = task.path("imagename")?;
63    let summary = imstat(
64        &path,
65        task.text("box")?.as_deref(),
66        task.text("region")?.as_deref(),
67        task.text("chans")?.as_deref(),
68        None,
69    )
70    .map_err(|error| image_error("imstat", &path, error))?;
71    let text = statistics_text(&summary);
72    Ok(TaskSuccess::new(&summary, &summary)?.with_text(text))
73}
74
75fn image_error(operation: &str, path: &Path, error: impl Display) -> TaskFailure {
76    TaskFailure::execution(format!(
77        "{operation}: cannot open or read CASA image '{}': {error}",
78        path.display()
79    ))
80}

82fn header_text(summary: &ImageHeaderSummary) -> String {
83    let mut out = String::new();
84    let _ = writeln!(out, "Image name       : {}", summary.imagename);
85    let _ = writeln!(out, "Image type       : {}", summary.image_type);
86    let _ = writeln!(out, "Pixel type       : {}", summary.pixel_type);
87    let _ = writeln!(out, "Image units      : {}", summary.units);
88    let _ = writeln!(out, "Shape            : {:?}", summary.shape);
89    if let Some(beam) = &summary.restoring_beam {
90        let _ = writeln!(
91            out,
92            "Restoring Beam   : {:.6} arcsec, {:.6} arcsec, {:.6} deg",
93            beam.major_arcsec, beam.minor_arcsec, beam.position_angle_deg
94        );
95    }
96    for axis in &summary.axes {
97        let _ = writeln!(
98            out,
99            "Axis {} {:<12} {:<18} shape={} refpix={:.3} crval={:.12e} cdelt={:.12e} {}",
100            axis.axis,
101            axis.coordinate_type,
102            axis.name,
103            axis.shape,
104            axis.reference_pixel,
105            axis.reference_value,
106            axis.increment,
107            axis.unit
108        );
109    }
110    out
111}

113fn statistics_text(summary: &ImageStatisticsSummary) -> String {
114    let mut out = String::new();
115    let _ = writeln!(out, "npts: {}", summary.npts);
116    let _ = writeln!(out, "min: {}", summary.min);
117    let _ = writeln!(out, "max: {}", summary.max);
118    let _ = writeln!(out, "sum: {}", summary.sum);
119    let _ = writeln!(out, "mean: {}", summary.mean);
120    let _ = writeln!(out, "rms: {}", summary.rms);
121    let _ = writeln!(out, "sigma: {}", summary.sigma);
122    if let Some(flux) = summary.flux {
123        let _ = writeln!(out, "flux: {flux}");
124    }
125    out
126}
