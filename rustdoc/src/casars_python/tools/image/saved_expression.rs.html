30pub(super) fn open(path: &Path) -> PyResult<AttachedImage> {
31    let (image, inputs) = compile_at(path, AttachedImage::open, false, &mut HashSet::new(), None)?;
32    let mut attachment = AttachedImage::with_saved_expression(image, inputs.directory);
33    attachment.retain_expression_inputs(inputs.retained, inputs.paths);
34    Ok(attachment)
35}

37/// Resolve the final output name to its staged image when reading `imagemd`.
38pub(super) struct StagedOutput<'a> {
39    pub(super) target: &'a Path,
40    pub(super) stage: &'a Path,
41    entry: PathBuf,
42}
43
44impl<'a> StagedOutput<'a> {
45    pub(super) fn new(target: &'a Path, stage: &'a Path) -> PyResult<Self> {
46        let entry = if let Ok(path) = target.canonicalize() {
47            path
48        } else {
49            let path = std::path::absolute(target).map_err(raised)?;
50            let parent = path
51                .parent()
52                .ok_or_else(|| raised("Missing output parent"))?;
53            let name = path
54                .file_name()
55                .ok_or_else(|| raised("Missing output name"))?;
56            parent.canonicalize().map_err(raised)?.join(name)
57        };
58        Ok(Self {
59            target,
60            stage,
61            entry,
62        })
63    }
64
65    pub(super) fn substitute(&self, path: &Path) -> Option<PathBuf> {
66        let path = std::path::absolute(path).ok()?;
67        let parent = path.parent()?.canonicalize().ok()?;
68        let path_entry = parent.join(path.file_name()?);
69        let same_entry = path_entry == self.entry;
70        if !same_entry && path_entry.starts_with(&self.entry) {
71            return None;
72        }
73        let same_directory = path.canonicalize().is_ok_and(|path| path == self.entry);
74        (same_entry || same_directory).then(|| self.stage.to_path_buf())
75    }
76}

118pub(super) fn check_output_dependencies<'a>(
119    output: &Path,
120    paths: impl IntoIterator<Item = &'a PathBuf>,
121    check_new_children: bool,
122) -> PyResult<()> {
123    let route = casa_tables::lexical_absolute(output);
124    let entry = route
125        .parent()
126        .and_then(|parent| parent.canonicalize().ok())
127        .zip(route.file_name())
128        .map(|(parent, name)| parent.join(name));
129    let referent = output.canonicalize().ok();
130    let replaces_entry = check_new_children || output.symlink_metadata().is_ok();
131    for path in paths {
132        if std::iter::once(&route)
133            .chain(entry.iter())
134            .chain(referent.iter())
135            .any(|output| path.starts_with(output) || (replaces_entry && output.starts_with(path)))
136        {
137            return Err(raised(format!(
138                "Cannot overwrite output {} because it overlaps a saved image expression input {}",
139                output.display(),
140                path.display(),
141            )));
142        }
143    }
144    Ok(())
145}

147pub(super) fn compile_inline(
148    inputs: &ImageToolExpressionInputs<'_>,
149    temporary_source_locks: bool,
150) -> PyResult<(ImageToolPixelExpression, Vec<AttachedImage>, Vec<PathBuf>)> {
151    let (graph, operands) = capture_inline(inputs, AttachedImage::open)?;
152    let (pixels, _) = inputs
153        .compile_with_bound_operands(&operands, temporary_source_locks)
154        .map_err(raised)?;
155    Ok((pixels, graph.retained, graph.paths))
156}

158pub(super) fn has_manifest_operands(inputs: &ImageToolExpressionInputs<'_>) -> bool {
159    inputs
160        .paths()
161        .any(|path| is_image_expr(path) || is_image_concat(path))
162}

164pub(super) fn compile_edit(expression: &str) -> PyResult<ImageToolPixelExpression> {
165    let inputs = ImageToolExpressionInputs::resolve(expression).map_err(raised)?;
166    if !has_manifest_operands(&inputs) {
167        return inputs.compile().map_err(raised);
168    }
169    // The caller already holds the target image mutex. Open independent
170    // source handles so a transitive reference to that target can be read.
171    let (_graph, operands) =
172        capture_inline(&inputs, AttachedImage::open_unregistered_expression_operand)?;
173    inputs
174        .compile_with_bound_operands(&operands, false)
175        .map(|(pixels, _)| pixels)
176        .map_err(raised)
177}

179pub(super) fn evaluate_mask(
180    expression: &str,
181    shape: &[usize],
182    overwrite_target: Option<&Path>,
183) -> PyResult<ArrayD<bool>> {
184    let inputs = ImageToolExpressionInputs::resolve(expression).map_err(raised)?;
185    if !has_manifest_operands(&inputs) {
186        return inputs
187            .evaluate_mask_with_bound_operands(&BoundImageToolOperands::new(), shape)
188            .map_err(raised);
189    }
190    let (graph, operands) =
191        capture_inline(&inputs, AttachedImage::open_unregistered_expression_operand)?;
192    if let Some(target) = overwrite_target {
193        check_output_dependencies(target, &graph.paths, false)?;
194    }
195    inputs
196        .evaluate_mask_with_bound_operands(&operands, shape)
197        .map_err(raised)
198}

200fn capture_inline(
201    inputs: &ImageToolExpressionInputs<'_>,
202    open_source: fn(&Path) -> PyResult<AttachedImage>,
203) -> PyResult<(CapturedGraph<'static>, BoundImageToolOperands)> {
204    let mut graph = CapturedGraph {
205        include_manifest_paths: true,
206        ..CapturedGraph::default()
207    };
208    let directory = std::env::current_dir().map_err(raised)?;
209    let operands = graph.capture(inputs, None, &directory, open_source, &mut HashSet::new())?;
210    Ok((graph, operands))
211}

213fn compile_at(
214    path: &Path,
215    open_source: fn(&Path) -> PyResult<AttachedImage>,
216    include_manifest_paths: bool,
217    active: &mut HashSet<PathBuf>,
218    output: Option<&StagedOutput<'_>>,
219) -> PyResult<(ImageHandle, SavedExpressionInputs)> {
220    let directory = std::env::current_dir().map_err(raised)?;
221    let root = path.canonicalize().map_err(raised)?;
222    let (captured, manifest) = SavedExpressionDirectory::open(path, root.clone())?;
223    let inputs = ImageToolExpressionInputs::resolve_with_image_paths(
224        manifest.expression.as_str(),
225        &directory,
226        |path| output.and_then(|output| output.substitute(path)),
227    )
228    .map_err(raised)?;
229    let mut graph = CapturedGraph {
230        include_manifest_paths,
231        output,
232        ..CapturedGraph::default()
233    };
234    graph.retain_manifest(&captured)?;
235    let operands = graph.capture(&inputs, Some(root), &directory, open_source, active)?;
236    let (pixels, single_precision) = compile_manifest(&inputs, &operands)?;
237    let misc = manifest.misc_info().map_err(raised)?;
238    captured.check_name(path)?;
239    let image = match pixels {
240        ImageToolPixelExpression::Real(expression) if single_precision => {
241            ImageHandle::Float(expression_storage(
242                expression
243                    .convert_pixels(|value| value as f32)
244                    .map_err(raised)?,
245                path,
246                misc,
247            )?)
248        }
249        ImageToolPixelExpression::Real(expression) => {
250            ImageHandle::Double(expression_storage(expression, path, misc)?)
251        }
252        ImageToolPixelExpression::Complex(expression) if single_precision => {
253            ImageHandle::Complex(expression_storage(
254                expression
255                    .convert_pixels(|value| Complex32::new(value.re as f32, value.im as f32))
256                    .map_err(raised)?,
257                path,
258                misc,
259            )?)
260        }
261        ImageToolPixelExpression::Complex(expression) => {
262            ImageHandle::DComplex(expression_storage(expression, path, misc)?)
263        }
264        ImageToolPixelExpression::Boolean(_) => unreachable!("saved image geometry is numeric"),
265    };
266    Ok((
267        image,
268        SavedExpressionInputs {
269            directory: captured,
270            retained: graph.retained,
271            paths: graph.paths,
272        },
273    ))
274}

276#[derive(Default)]
277struct CapturedGraph<'a> {
278    paths: Vec<PathBuf>,
279    seen_paths: HashSet<PathBuf>,
280    links: HashSet<PathBuf>,
281    retained: Vec<AttachedImage>,
282    include_manifest_paths: bool,
283    output: Option<&'a StagedOutput<'a>>,
284}
285
286impl CapturedGraph<'_> {
287    fn capture(
288        &mut self,
289        inputs: &ImageToolExpressionInputs<'_>,
290        root: Option<PathBuf>,
291        directory: &Path,
292        open_source: fn(&Path) -> PyResult<AttachedImage>,
293        active: &mut HashSet<PathBuf>,
294    ) -> PyResult<BoundImageToolOperands> {
295        if let Some(path) = &root
296            && !active.insert(path.clone())
297        {
298            return Err(expression_cycle(path));
299        }
300        let mut pending = Vec::new();
301        let mut top = inputs.paths_and_entries();
302        let mut compiled = BoundImageToolOperands::new();
303        loop {
304            let next = match pending.last_mut() {
305                Some(frame) => PendingExpression::next_path(frame),
306                None => top.next(),
307            };
308            if let Some((path, entry)) = next {
309                if active.contains(path) {
310                    return Err(expression_cycle(path));
311                }
312                if entry != path {
313                    self.retain_operand_aliases(entry)?;
314                }
315                if !self.seen_paths.insert(path.to_path_buf()) {
316                    continue;
317                }
318                self.paths.push(path.to_path_buf());
319                if is_image_concat(path) {
320                    let image = virtual_image::open_expression_leaf(path, active, self.output)?;
321                    self.paths.extend(image.concat_expression_operands());
322                    for operand in image.concat_operands() {
323                        self.retain_operand_aliases(&operand)?;
324                        self.retain_miriad_files(&operand)?;
325                    }
326                    let source = AttachedImage::temporary(image);
327                    compiled.insert(path.to_path_buf(), bound_pixels(&source).map_err(raised)?);
328                    self.retained.push(source);
329                } else if is_image_expr(path) {
330                    let path = path.to_path_buf();
331                    active.insert(path.clone());
332                    let (captured, manifest) = SavedExpressionDirectory::open(&path, path.clone())?;
333                    self.retain_manifest(&captured)?;
334                    let nested =
335                        PendingExpression::from_manifest(path, manifest, directory, self.output)?;
336                    pending.push(nested);
337                } else if !path.is_file() {
338                    self.retain_miriad_files(path)?;
339                    self.retained.push(open_source(path)?);
340                }
341            } else if let Some(frame) = pending.pop() {
342                let (pixels, single) = compile_manifest(&frame.inputs, &compiled)?;
343                let misc = frame.manifest.misc_info().map_err(raised)?;
344                active.remove(&frame.path);
345                let pixels = match misc {
346                    Some(misc) => replace_misc_info(pixels, misc),
347                    None => pixels,
348                };
349                compiled.insert(frame.path, (pixels, single));
350            } else {
351                if let Some(path) = &root {
352                    active.remove(path);
353                }
354                return Ok(compiled);
355            }
356        }
357    }
358
359    fn retain_manifest(&mut self, directory: &SavedExpressionDirectory) -> PyResult<()> {
360        if self.include_manifest_paths
361            && let Some((entry, referent)) = directory.linked_manifest_paths()
362        {
363            self.retain_operand_aliases(&entry)?;
364            self.paths.push(referent.to_owned());
365        }
366        Ok(())
367    }
368
369    fn retain_miriad_files(&mut self, path: &Path) -> PyResult<()> {
370        if ImageHandle::recognizes_miriad(path, path) {
371            for entry in MiriadImage::linked_input_files(path) {
372                self.retain_operand_aliases(&entry)?;
373                self.paths.push(entry.canonicalize().map_err(raised)?);
374            }
375        }
376        Ok(())
377    }
378
379    fn retain_operand_aliases(&mut self, entry: &Path) -> PyResult<()> {
380        self.paths.push(lexical_absolute(entry));
381        let mut routes = vec![entry.to_path_buf()];
382        while let Some(route) = routes.pop() {
383            for prefix in route.ancestors() {
384                if !prefix
385                    .symlink_metadata()
386                    .is_ok_and(|entry| entry.is_symlink())
387                {
388                    continue;
389                }
390                let (Some(parent), Some(name)) = (prefix.parent(), prefix.file_name()) else {
391                    continue;
392                };
393                let parent = parent.canonicalize().map_err(raised)?;
394                let link = parent.join(name);
395                if self.links.insert(link.clone()) {
396                    routes.push(parent.join(fs::read_link(&link).map_err(raised)?));
397                    self.paths.push(link);
398                }
399            }
400        }
401        Ok(())
402    }
403}

405fn expression_cycle(path: &Path) -> PyErr {
406    raised(format!(
407        "Cycle in saved image expression at {}",
408        path.display()
409    ))

412fn bound_pixels(source: &AttachedImage) -> Result<(ImageToolPixelExpression, bool), ImageError> {
413    let handle = source.lock();
414    macro_rules! bind {
415        ($variant:ident, $storage:ident) => {
416            bound_image(source, $storage, |handle| {
417                let ImageHandle::$variant(storage) = handle else {
418                    unreachable!("a retained source keeps its pixel type");
419                };
420                storage
421            })?
422        };
423    }
424    Ok(match &*handle {
425        ImageHandle::Float(storage) => (
426            ImageToolPixelExpression::Real(bind!(Float, storage).convert_pixels(f64::from)?),
427            true,
428        ),
429        ImageHandle::Double(storage) => (
430            ImageToolPixelExpression::Real(bind!(Double, storage)),
431            false,
432        ),
433        ImageHandle::Complex(storage) => (
434            ImageToolPixelExpression::Complex(bind!(Complex, storage).convert_pixels(|pixel| {
435                Complex64::new(f64::from(pixel.re), f64::from(pixel.im))
436            })?),
437            true,
438        ),
439        ImageHandle::DComplex(storage) => (
440            ImageToolPixelExpression::Complex(bind!(DComplex, storage)),
441            false,
442        ),
443    })
444}

446fn bound_image<T: ImageExprValue>(
447    source: &AttachedImage,
448    storage: &Storage<T>,
449    select: fn(&ImageHandle) -> &Storage<T>,
450) -> Result<ImageExpression<T>, ImageError> {
451    let pixels = source.share_image();
452    let mask = source.share_image();
453    let coordinate_record = match storage {
454        Storage::Temp { record, .. } => Some(record),
455        Storage::Paged(_) => None,
456    };
457    with_storage!(storage, image => ImageExprBuilder::from_image_sections(
458        image.as_ref(),
459        coordinate_record,
460        move |start, count, stride| {
461            let handle = pixels.lock();
462            let storage = select(&handle);
463            if let Storage::Temp { image, .. } = storage
464                && let Some(concat) = image.concat_source()
465            {
466                concat.check_directory()?;
467            }
468            Ok(with_storage!(storage, image => Lattice::get_slice(image.as_ref(), start, count, stride))?)
469        },
470        move |start, count, stride| {
471            let handle = mask.lock();
472            let storage = select(&handle);
473            if let Storage::Temp { image, .. } = storage
474                && let Some(concat) = image.concat_source()
475            {
476                concat.check_directory()?;
477            }
478            with_storage!(storage, image => image.get_mask_slice(start, count, stride))
479        },
480    ))?
481    .compile()
482}

484struct PendingExpression {
485    path: PathBuf,
486    manifest: ToolExpressionManifest,
487    inputs: ImageToolExpressionInputs<'static>,
488    next: usize,
489}
490
491impl PendingExpression {
492    fn from_manifest(
493        path: PathBuf,
494        manifest: ToolExpressionManifest,
495        directory: &Path,
496        output: Option<&StagedOutput<'_>>,
497    ) -> PyResult<Self> {
498        let inputs = ImageToolExpressionInputs::resolve_with_image_paths(
499            manifest.expression.clone(),
500            directory,
501            |path| output.and_then(|output| output.substitute(path)),
502        )
503        .map_err(raised)?;
504        Ok(Self {
505            path,
506            manifest,
507            inputs,
508            next: 0,
509        })
510    }
511
512    fn next_path(&mut self) -> Option<(&Path, &Path)> {
513        let paths = self.inputs.paths_and_entries().nth(self.next)?;
514        self.next += 1;
515        Some(paths)
516    }
517}

519fn compile_manifest(
520    inputs: &ImageToolExpressionInputs<'_>,
521    operands: &BoundImageToolOperands,
522) -> PyResult<(ImageToolPixelExpression, bool)> {
523    let (pixels, single) = inputs
524        .compile_with_bound_operands(operands, true)
525        .map_err(raised)?;
526    super::imagecalc::check_geometry(&pixels, inputs.paths().len() != 0)?;
527    Ok((pixels, single))
528}

530fn replace_misc_info(
531    pixels: ImageToolPixelExpression,
532    misc: RecordValue,
533) -> ImageToolPixelExpression {
534    match pixels {
535        ImageToolPixelExpression::Real(expression) => {
536            ImageToolPixelExpression::Real(expression.with_misc_info(misc))
537        }
538        ImageToolPixelExpression::Complex(expression) => {
539            ImageToolPixelExpression::Complex(expression.with_misc_info(misc))
540        }
541        ImageToolPixelExpression::Boolean(_) => unreachable!("saved image geometry is numeric"),
542    }
543}

545fn expression_storage<T: ImageExprValue>(
546    expression: ImageExpression<T>,
547    path: &Path,
548    misc: Option<RecordValue>,
549) -> PyResult<Storage<T>> {
550    let record = expression.coordinate_record();
551    let mut image =
552        TempImage::from_saved_expression(expression, path.to_path_buf()).map_err(raised)?;
553    if let Some(misc) = misc {
554        image.set_misc_info(misc).map_err(raised)?;
555    }
556    Ok(Storage::Temp {
557        image: Box::new(image),
558        record,
559        fits: None,
560    })
