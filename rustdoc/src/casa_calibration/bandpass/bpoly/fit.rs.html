21pub(super) fn fit_interval(
22    group: &SolveAccumulator,
23    frequencies: &BTreeMap<i32, (&[f64], f64)>,
24    antennas: &BTreeSet<i32>,
25    refant: i32,
26    request: &BandpassSolveRequest,
27    cuts: BandpassSolveCuts,
28) -> Result<Vec<BPolySolutionRow>, BandpassSolveError> {
29    let error = |reason: &str| BandpassSolveError::FitPolynomial {
30        antenna_id: refant,
31        field_id: group.field_id,
32        spw_id: group.spw_id,
33        reason: reason.to_string(),
34    };
35    let degrees = [request.amplitude_degree, request.phase_degree];
36    let [amp_count, phase_count] = std::array::from_fn(|index| {
37        if cuts.omit_polynomial[index] {
38            0
39        } else {
40            degrees[index] + 1
41        }
42    });
43    let count = amp_count.max(phase_count);
44    let grid = FrequencyGrid::new(frequencies)
45        .ok_or_else(|| error("at least two distinct channel frequencies are required"))?;
46    let mut indices = frequencies
47        .iter()
48        .map(|(&id, &(frequencies, _))| {
49            (
50                id,
51                frequencies
52                    .iter()
53                    .map(|&value| grid.index(value))
54                    .collect::<Vec<_>>(),
55            )
56        })
57        .collect::<BTreeMap<_, _>>();
58    let occupied = indices.values().flatten().copied().collect::<BTreeSet<_>>();
59    if occupied.len() < count {
60        return Err(error(
61            "not enough spectral channels for the requested degree",
62        ));
63    }
64
65    // The logical grid can span a wide gap between spectral windows.
66    // Only store basis and participation entries for selected channels.
67    let basis = occupied
68        .iter()
69        .copied()
70        .map(|channel| {
71            let mut values = vec![0.0; count];
72            let [first, last] = grid.domain();
73            legacy_bpoly_basis(first, last, grid.center(channel), &mut values);
74            values
75        })
76        .collect::<Vec<_>>();
77    let compact = occupied
78        .into_iter()
79        .enumerate()
80        .map(|(index, slot)| (slot, index))
81        .collect::<BTreeMap<_, _>>();
82    for slot in indices.values_mut().flatten() {
83        *slot = compact[slot];
84    }
85    let antenna_indices = antennas
86        .iter()
87        .enumerate()
88        .map(|(i, &id)| (id, i))
89        .collect::<BTreeMap<_, _>>();
90    let (spectra, participation) =
91        collect_spectra(group, &indices, basis.len(), &antenna_indices, cuts);
92    if spectra.iter().all(Vec::is_empty) {
93        return Ok(Vec::new());
94    }
95    let time = group
96        .polynomial
97        .as_ref()
98        .and_then(|accumulation| accumulation.time())
99        .unwrap_or_else(|| group.aggregate_time());
100    let mut rows = antennas
101        .iter()
102        .map(|&antenna_id| BPolySolutionRow {
103            time_seconds: (time / 86_400.0) * 86_400.0,
104            interval_seconds: 0.0,
105            field_id: group.field_id,
106            spw_id: group.spw_id,
107            antenna_id,
108            scan_number: -1,
109            observation_id: -1,
110            valid_domain_hz: grid.domain(),
111            scale_factor: Complex32::new(1.0, 0.0),
112            amp_coefficients: vec![vec![0.0; amp_count]; 2],
113            phase_coefficients: vec![vec![0.0; phase_count]; 2],
114        })
115        .collect::<Vec<_>>();
116
117    for (receptor, samples) in spectra.iter().enumerate() {
118        if samples.is_empty() {
119            continue;
120        }
121        if samples
122            .iter()
123            .map(|sample| sample.channel)
124            .collect::<BTreeSet<_>>()
125            .len()
126            < count
127        {
128            tracing::warn!(
129                "{}",
130                error("too few unflagged frequencies for a polynomial fit")
131            );
132            return Ok(Vec::new());
133        }
134        let active = samples
135            .iter()
136            .flat_map(|sample| sample.antennas)
137            .collect::<BTreeSet<_>>();
138        let phase_reference = antenna_indices
139            .get(&refant)
140            .filter(|index| active.contains(index))
141            .or_else(|| active.first())
142            .copied();
143        let Some((amplitudes, phases)) =
144            fit_coefficients(samples, &basis, &active, amp_count, None).zip(fit_coefficients(
145                samples,
146                &basis,
147                &active,
148                phase_count,
149                phase_reference,
150            ))
151        else {
152            tracing::warn!("{}", error("normal equations are singular"));
153            return Ok(Vec::new());
154        };
155        for (&index, amplitude) in active.iter().zip(amplitudes.chunks(amp_count.max(1))) {
156            rows[index].amp_coefficients[receptor].copy_from_slice(amplitude);
157        }
158        for (&index, phase) in active
159            .iter()
160            .filter(|&&index| Some(index) != phase_reference)
161            .zip(phases.chunks(phase_count.max(1)))
162        {
163            rows[index].phase_coefficients[receptor].copy_from_slice(phase);
164        }
165        if request.normalize_average_amplitude {
166            for &index in &active {
167                normalize_output(&mut rows[index], receptor, &basis, &participation[index]);
168            }
169        }
170    }
171    Ok(rows)
172}

174struct FrequencyGrid {
175    first: f64,
176    step: f64,
177    count: usize,
178}
179
180impl FrequencyGrid {
181    fn new(frequencies: &BTreeMap<i32, (&[f64], f64)>) -> Option<Self> {
182        let step = frequencies
183            .values()
184            .map(|&(channels, spacing)| {
185                channels
186                    .get(1)
187                    .map_or(spacing, |second| (second - channels[0]).abs())
188            })
189            .reduce(f64::max)?;
190        let values = frequencies
191            .values()
192            .flat_map(|(channels, _)| channels.iter().copied());
193        let first = values.clone().reduce(f64::min)?;
194        let last = values.reduce(f64::max)?;
195        (step.is_finite() && step > 0.0 && first.is_finite() && last.is_finite()).then(|| Self {
196            first,
197            step,
198            count: ((last - first) / step + 1.5) as usize,
199        })
200    }
201
202    fn index(&self, frequency: f64) -> usize {
203        (((frequency - self.first) / self.step + 0.5) as usize).min(self.count - 1)
204    }
205
206    fn center(&self, channel: usize) -> f64 {
207        self.first + channel as f64 * self.step
208    }
209
210    fn domain(&self) -> [f64; 2] {
211        [self.first, self.center(self.count - 1)]
212    }
213}

215fn collect_spectra(
216    group: &SolveAccumulator,
217    bins: &BTreeMap<i32, Vec<usize>>,
218    occupied_channel_count: usize,
219    antennas: &BTreeMap<i32, usize>,
220    cuts: BandpassSolveCuts,
221) -> ([Vec<VisibilitySample>; 2], Vec<Vec<usize>>) {
222    let mut sums = [BTreeMap::new(), BTreeMap::new()];
223    let mut means = [BTreeMap::new(), BTreeMap::new()];
224    let mut participation = vec![vec![0; occupied_channel_count]; antennas.len()];
225    for (description, [a, b], receptor, spectrum) in
226        group.polynomial.iter().flat_map(|data| data.spectra())
227    {
228        let bins = &bins[&description];
229        let (i, j) = (antennas[&a], antennas[&b]);
230        let weight = f64::from(spectrum.weight);
231        for (channel, (&bin, &sample)) in
232            bins.iter().zip(&spectrum.weighted_visibilities).enumerate()
233        {
234            if cuts.masks_polynomial_channel(channel, bins.len()) || sample.norm_sqr() == 0.0 {
235                continue;
236            }
237            let sample = Complex64::new(f64::from(sample.re), f64::from(sample.im));
238            let (sum, total) = sums[receptor]
239                .entry((i, j, bin))
240                .or_insert((Complex64::default(), 0.0));
241            *sum += sample;
242            *total += weight;
243            if cuts.normalize_visibilities {
244                let (sum, total) = means[receptor]
245                    .entry((i, j))
246                    .or_insert((Complex64::default(), 0.0));
247                *sum += sample;
248                *total += weight;
249            }
250            participation[i][bin] += 1;
251            participation[j][bin] += 1;
252        }
253    }
254    let [first, second] = sums;
255    let [first_means, second_means] = means;
256    let spectra = [(first, first_means), (second, second_means)].map(|(sum, means)| {
257        sum.into_iter()
258            .filter_map(|((a, b, channel), (visibility, weight))| {
259                let visibility = if let Some(&(sum, norm_weight)) = means.get(&(a, b)) {
260                    if sum.norm_sqr() == 0.0 {
261                        return None;
262                    }
263                    visibility / (sum / norm_weight)
264                } else {
265                    visibility
266                };
267                (participation[a][channel] > 1
268                    && participation[b][channel] > 1
269                    && visibility.norm_sqr() > 0.0)
270                    .then_some(VisibilitySample {
271                        antennas: [a, b],
272                        channel,
273                        visibility: visibility / weight,
274                        weight,
275                    })
276            })
277            .collect::<Vec<_>>()
278    });
279    (spectra, participation)
280}

282fn fit_coefficients(
283    samples: &[VisibilitySample],
284    bases: &[Vec<f64>],
285    active: &BTreeSet<usize>,
286    count: usize,
287    phase_reference: Option<usize>,
288) -> Option<Vec<f64>> {
289    let columns = active
290        .iter()
291        .filter(|&&index| Some(index) != phase_reference)
292        .enumerate()
293        .map(|(column, &index)| (index, column * count))
294        .collect::<BTreeMap<_, _>>();
295    let size = columns.len() * count;
296    if size == 0 {
297        return Some(Vec::new());
298    }
299    let phase = phase_reference.is_some();
300    let signs = [1.0, if phase { -1.0 } else { 1.0 }];
301    let samples = samples
302        .iter()
303        .map(|sample| FitSample {
304            basis: &bases[sample.channel][..count],
305            terms: sample
306                .antennas
307                .map(|antenna| columns.get(&antenna).copied()),
308            weight: sample.weight,
309            observed: if phase {
310                sample.visibility.arg()
311            } else {
312                sample.visibility.norm().ln()
313            },
314        })
315        .collect::<Vec<_>>();
316    let mut normal = vec![0.0; size * size];
317    for sample in &samples {
318        for (left, left_sign) in sample.terms(signs) {
319            for (i, &x) in sample.basis.iter().enumerate() {
320                let weighted = sample.weight * left_sign * x;
321                for (column, sign) in sample.terms(signs) {
322                    for (j, &y) in sample.basis.iter().enumerate() {
323                        normal[(left + i) * size + column + j] += weighted * sign * y;
324                    }
325                }
326            }
327        }
328    }
329    let inverse = invert_symmetric_positive_definite(&normal, size)?;
330    let mut coefficients = vec![0.0; size];
331    let mut right = vec![0.0; size];
332    // CLIC POLYANT starts phase coefficients at zero and uses the sine
333    // of the baseline residual. Its normal matrix is constant across steps.
334    for _ in 0..if phase { 100 } else { 1 } {
335        right.fill(0.0);
336        for sample in &samples {
337            let residual = if phase {
338                let predicted = sample
339                    .terms(signs)
340                    .map(|(offset, sign)| {
341                        sign * dot(&coefficients[offset..offset + count], sample.basis)
342                    })
343                    .sum::<f64>();
344                (sample.observed - predicted).sin()
345            } else {
346                sample.observed
347            };
348            for (offset, sign) in sample.terms(signs) {
349                for (index, &basis) in sample.basis.iter().enumerate() {
350                    right[offset + index] += sign * sample.weight * basis * residual;
351                }
352            }
353        }
354        let mut update_norm = 0.0;
355        for (coefficient, row) in coefficients.iter_mut().zip(inverse.chunks(size)) {
356            let update = dot(row, &right);
357            *coefficient += update;
358            update_norm += update * update;
359        }
360        if update_norm <= 1e-14 {
361            break;
362        }
363    }
364    Some(coefficients)
365}

367struct FitSample<'a> {
368    basis: &'a [f64],
369    terms: [Option<usize>; 2],
370    weight: f64,
371    observed: f64,
372}
373
374impl FitSample<'_> {
375    fn terms(&self, signs: [f64; 2]) -> impl Iterator<Item = (usize, f64)> {
376        self.terms
377            .into_iter()
378            .zip(signs)
379            .filter_map(|(offset, sign)| offset.map(|offset| (offset, sign)))
380    }
381}

383fn normalize_output(
384    row: &mut BPolySolutionRow,
385    receptor: usize,
386    bases: &[Vec<f64>],
387    participation: &[usize],
388) {
389    let amplitude = &mut row.amp_coefficients[receptor];
390    let phase = &mut row.phase_coefficients[receptor];
391    let mut total_amplitude = 0.0;
392    let mut phasor = Complex64::default();
393    let mut count = 0;
394    for (basis, &coverage) in bases.iter().zip(participation) {
395        if coverage > 3 {
396            total_amplitude += dot(amplitude, basis).exp();
397            phasor += Complex64::from_polar(1.0, dot(phase, basis));
398            count += 1;
399        }
400    }
401    if count > 0 {
402        if let Some(offset) = amplitude.first_mut() {
403            *offset -= 2.0 * (total_amplitude / f64::from(count)).ln();
404        }
405        if let Some(offset) = phase.first_mut() {
406            *offset -= 2.0 * phasor.arg();
407        }
408    }
409}

411fn dot(first: &[f64], second: &[f64]) -> f64 {
412    first.iter().zip(second).map(|(a, b)| a * b).sum()
