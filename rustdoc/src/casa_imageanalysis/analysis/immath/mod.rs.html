23impl ResolvedInputs {
24    fn all_paths(&self) -> Vec<PathBuf> {
25        self.imagename_paths
26            .iter()
27            .chain(self.extra_quoted.iter())
28            .cloned()
29            .collect()
30    }

33pub(super) fn run(request: &ImmathRequest) -> Result<ImageMathSummary, ImageError> {
34    let mode = request.mode.trim().to_ascii_lowercase();
35    let outfile = resolve_outfile_path(request)?;
36    let varnames = effective_varnames(request);
37    let inputs = resolve_inputs(request, &mode, &varnames)?;
38    let all_filenames = inputs.all_paths();
39    ensure_outfile_does_not_alias_inputs(&outfile, &all_filenames)?;
40    if let Some(mask) = request
41        .mask
42        .as_deref()
43        .filter(|mask| !mask.trim().is_empty())
44    {
45        let mask = selection::quote_var_aliases(mask, &varnames);
46        let mask_paths = referenced_images(&mask)
47            .map_err(|error| ImageError::CasaSelection(format!("Invalid image mask: {error}")))?
48            .into_iter()
49            .map(operand_path)
50            .filter(|path| path.exists())
51            .collect::<Vec<_>>();
52        ensure_outfile_does_not_alias_inputs(&outfile, &mask_paths)?;
53    }
54
55    let summary = match mode.as_str() {
56        "evalexpr" => {
57            let prec = parse_prec(&request.prec)?;
58            evalexpr::run_evalexpr(
59                request,
60                &outfile,
61                &inputs,
62                &varnames,
63                &request.expr,
64                false,
65                prec,
66            )?
67        }
68        "spix" => {
69            let prec = parse_prec(&request.prec)?;
70            if inputs.imagename_paths.len() != 2 {
71                return Err(ImageError::InvalidMetadata(
72                    "Requires two images at different frequencies".into(),
73                ));
74            }
75            let spix_expr = format!("spectralindex({}, {})", varnames[0], varnames[1]);
76            evalexpr::run_evalexpr(
77                request, &outfile, &inputs, &varnames, &spix_expr, true, prec,
78            )?
79        }
80        "pola" => pol::run_pola(request, &outfile, &inputs.imagename_paths, &varnames)?,
81        "poli" | "lpoli" | "tpoli" => {
82            pol::run_poli(request, &outfile, &inputs.imagename_paths, &varnames, &mode)?
83        }
84        _ => {
85            return Err(ImageError::InvalidMetadata(format!(
86                "Unsupported mode {}",
87                request.mode
88            )));
89        }
90    };
91
92    write_history(&outfile, request, &mode, &all_filenames)?;
93    Ok(summary)
94}

96fn resolve_outfile_path(request: &ImmathRequest) -> Result<PathBuf, ImageError> {
97    let raw = request.outfile.to_string_lossy();
98    let outfile = if raw.trim().is_empty() {
99        PathBuf::from("immath_results.im")
100    } else {
101        request.outfile.clone()
102    };
103    if outfile.exists() && !request.overwrite {
104        return Err(ImageError::Io(format!(
105            "Output file {} already exists; immath can not proceed, please remove it or change the output file name.",
106            outfile.display()
107        )));
108    }
109    Ok(outfile)
110}

112pub(super) fn remove_existing_outfile_if_overwrite(
113    outfile: &Path,
114    overwrite: bool,
115) -> Result<(), ImageError> {
116    if overwrite && outfile.exists() {
117        if outfile.is_dir() {
118            fs::remove_dir_all(outfile).map_err(|e| ImageError::Io(e.to_string()))?;
119        } else {
120            fs::remove_file(outfile).map_err(|e| ImageError::Io(e.to_string()))?;
121        }
122    }
123    Ok(())
124}

126fn ensure_outfile_does_not_alias_inputs(
127    outfile: &Path,
128    inputs: &[PathBuf],
129) -> Result<(), ImageError> {
130    let out_canon = fs::canonicalize(outfile).ok();
131    for input in inputs {
132        let same_lexical = outfile == input.as_path();
133        let same_canon = out_canon
134            .as_ref()
135            .zip(fs::canonicalize(input).ok().as_ref())
136            .is_some_and(|(a, b)| a == b);
137        if same_lexical || same_canon {
138            return Err(ImageError::InvalidMetadata(format!(
139                "Output file {} matches input image {} and cannot overwrite it.",
140                outfile.display(),
141                input.display()
142            )));
143        }
144    }
145    Ok(())
146}

148fn parse_prec(prec: &str) -> Result<ImagePixelType, ImageError> {
149    let trimmed = prec.trim().to_ascii_lowercase();
150    if !trimmed.is_empty() && "float".starts_with(&trimmed) {
151        Ok(ImagePixelType::Float32)
152    } else if !trimmed.is_empty() && "double".starts_with(&trimmed) {
153        Ok(ImagePixelType::Float64)
154    } else {
155        Err(ImageError::InvalidMetadata(
156            "Unsupported value for type, it must be 'float' or 'double'.".into(),
157        ))
158    }
159}

161fn effective_varnames(request: &ImmathRequest) -> Vec<String> {
162    let mut names: Vec<String> = request
163        .varnames
164        .iter()
165        .filter(|name| !name.is_empty())
166        .cloned()
167        .collect();
168    let n = request.imagename.len();
169    names.truncate(n);
170    for idx in names.len()..n {
171        names.push(format!("IM{idx}"));
172    }
173    names
174}

176fn resolve_inputs(
177    request: &ImmathRequest,
178    mode: &str,
179    varnames: &[String],
180) -> Result<ResolvedInputs, ImageError> {
181    for path in &request.imagename {
182        if !Path::new(path).exists() {
183            return Err(ImageError::Io(format!(
184                "Image data set not found - please verify {}",
185                path.display()
186            )));
187        }
188    }
189    let imagename_paths = request.imagename.clone();
190    let mut extra_quoted = Vec::new();
191    if mode == "evalexpr" {
192        let expression = selection::quote_var_aliases(&request.expr, varnames);
193        for name in referenced_images(&expression).map_err(|error| {
194            ImageError::InvalidMetadata(format!("Invalid image expression: {error}"))
195        })? {
196            if varnames.contains(&name) {
197                continue;
198            }
199            let p = operand_path(&name);
200            if !imagename_paths.contains(&p) && !extra_quoted.contains(&p) {
201                if !p.exists() {
202                    return Err(ImageError::Io(format!(
203                        "Image data set not found - please verify {name}"
204                    )));
205                }
206                extra_quoted.push(p);
207            }
208        }
209    }
210    if imagename_paths.is_empty() && extra_quoted.is_empty() {
211        return Err(ImageError::Io(
212            "Image data set not found - please verify ".into(),
213        ));
214    }
215    Ok(ResolvedInputs {
216        imagename_paths,
217        extra_quoted,
218    })
219}

221fn write_history(
222    outfile: &Path,
223    request: &ImmathRequest,
224    mode: &str,
225    filenames: &[PathBuf],
226) -> Result<(), ImageError> {
227    let inputs = filenames
228        .iter()
229        .map(|p| p.display().to_string())
230        .collect::<Vec<_>>();
231    let lines = [
232        concat!("version: casa-rs ", env!("CARGO_PKG_VERSION")).into(),
233        format!(
234            "immath(imagename={inputs:?}, mode={mode:?}, outfile={:?}, expr={:?}, varnames={:?}, sigma={:?}, polithresh={:?}, mask={:?}, region={:?}, box={:?}, chans={:?}, stokes={:?}, stretch={}, imagemd={:?}, prec={:?}, overwrite={})",
235            outfile.display().to_string(),
236            request.expr,
237            request.varnames,
238            request.sigma,
239            request.polithresh,
240            request.mask,
241            request.region,
242            request.box_pixels,
243            request.chans,
244            request.stokes,
245            request.stretch,
246            request.imagemd,
247            request.prec,
248            request.overwrite,
249        ),
250    ];
251    add_image_history(outfile, "immath", &lines)
252}
