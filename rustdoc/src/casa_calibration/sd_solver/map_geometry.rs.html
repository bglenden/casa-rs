2//! Field tracking and antenna-frame directions for single-dish maps.
3
4use std::collections::{BTreeMap, btree_map::Entry};
5
6use casa_core::measures::direction::DirectionRef;
7use casa_ms::derived::engine::MsCalEngine;
8use casa_ms::{MeasurementSet, PlanetaryEphemeris};
9
10pub(super) struct MapTargets {
11    field_sources: Vec<i32>,
12    source_names: BTreeMap<i32, String>,
13    body_tables: BTreeMap<String, PlanetaryEphemeris>,
14}
15
16pub(super) enum Target<'a> {
17    Fixed,
18    Comet,
19    Planet(&'a PlanetaryEphemeris),
20}
21
22pub(super) enum SourceRow {
23    Required,
24    Optional,
25}
26
27impl MapTargets {
28    pub(super) fn read(ms: &MeasurementSet) -> Result<Self, String> {
29        let fields = ms.field().map_err(error_text)?;
30        let source = ms.source().map_err(error_text)?;
31        let field_sources = (0..fields.row_count())
32            .map(|row| fields.source_id(row))
33            .collect::<Result<Vec<_>, _>>()
34            .map_err(error_text)?;
35        let mut source_names = BTreeMap::new();
36        for row in 0..source.row_count() {
37            let id = source.i32(row, "SOURCE_ID").map_err(error_text)?;
38            if let Entry::Vacant(entry) = source_names.entry(id) {
39                let name = source.string(row, "NAME").map_err(error_text)?;
40                entry.insert(name.to_ascii_uppercase());
41            }
42        }
43        Ok(Self {
44            field_sources,
45            source_names,
46            body_tables: BTreeMap::new(),
47        })
48    }
49
50    pub(super) fn field(&mut self, id: usize, source_row: SourceRow) -> Result<Target<'_>, String> {
51        let source = self.field_sources.get(id).ok_or("Invalid map FIELD_ID.")?;
52        let Some(name) = self.source_names.get(source) else {
53            return match source_row {
54                SourceRow::Required => Err("No SOURCE row for selected map field.".into()),
55                SourceRow::Optional => Ok(Target::Fixed),
56            };
57        };
58        if name == "COMET" {
59            return Ok(Target::Comet);
60        }
61        if !matches!(
62            name.as_str(),
63            "SUN"
64                | "MOON"
65                | "MERCURY"
66                | "VENUS"
67                | "MARS"
68                | "JUPITER"
69                | "SATURN"
70                | "URANUS"
71                | "NEPTUNE"
72                | "PLUTO"
73        ) {
74            return Ok(Target::Fixed);
75        }
76        let table = match self.body_tables.entry(name.clone()) {
77            Entry::Occupied(entry) => entry.into_mut(),
78            Entry::Vacant(entry) => {
79                entry.insert(PlanetaryEphemeris::open_discovered(name).map_err(error_text)?)
80            }
81        };
82        Ok(Target::Planet(table))
83    }
84}
85
86impl Target<'_> {
87    pub(super) fn direction(
88        &self,
89        engine: &MsCalEngine,
90        time: f64,
91        antenna: usize,
92        field: usize,
93    ) -> Result<Option<(f64, f64)>, String> {
94        match self {
95            Self::Fixed => Ok(None),
96            Self::Comet => engine
97                .tracked_body_direction_j2000(time, field)
98                .map(|direction| Some(direction.as_angles()))
99                .map_err(error_text),
100            Self::Planet(body) => {
101                let frame = engine.antenna_frame(time, antenna).map_err(error_text)?;
102                let body = body
103                    .direction(&frame)
104                    .map_err(error_text)?
105                    .convert_to(DirectionRef::J2000, &frame)
106                    .map_err(error_text)?;
107                Ok(Some(body.as_angles()))
108            }
109        }
110    }
111}
112
113fn error_text(error: impl std::fmt::Display) -> String {
114    error.to_string()
115}
