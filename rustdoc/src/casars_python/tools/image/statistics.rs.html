20fn stats_mask(value: Option<&Bound<'_, PyAny>>) -> PyResult<String> {
21    match value.map(|value| value.extract::<Variant>()).transpose()? {
22        None | Some(Variant::Empty | Variant::Bool(false)) => Ok(String::new()),
23        Some(Variant::Text(text)) if matches!(text.as_str(), "false" | "[]") => Ok(String::new()),
24        Some(Variant::Text(text)) => Ok(text),
25        Some(Variant::Bool(true)) => Ok("true".into()),
26        _ => Err(raised("Mask is not understood, try a valid LEL string")),
27    }
28}

30#[pymethods]
31impl PyImageTool {
32    #[allow(clippy::too_many_arguments)]
33    #[pyo3(signature = (axes=None, region=None, mask=None, includepix=None, excludepix=None, list=false, force=false, disk=false, robust=false, verbose=false, stretch=false, logfile="", append=true, algorithm="classic", fence=-1.0, center="mean", lside=true, zscore=-1.0, maxiter=-1, clmethod="auto", niter=3), text_signature = "($self, axes=[-1], region='', mask='', includepix=[-1.0], excludepix=[-1.0], list=False, force=False, disk=False, robust=False, verbose=False, stretch=False, logfile='', append=True, algorithm='classic', fence=-1.0, center='mean', lside=True, zscore=-1.0, maxiter=-1, clmethod='auto', niter=3)")]
34    fn statistics(
35        &self,
36        py: Python<'_>,
37        axes: Option<IntList>,
38        region: Option<&Bound<'_, PyAny>>,
39        mask: Option<&Bound<'_, PyAny>>,
40        includepix: Option<DoubleList>,
41        excludepix: Option<DoubleList>,
42        list: bool,
43        force: bool,
44        disk: bool,
45        robust: bool,
46        verbose: bool,
47        stretch: bool,
48        logfile: &str,
49        append: bool,
50        algorithm: &str,
51        fence: f64,
52        center: &str,
53        lside: bool,
54        zscore: f64,
55        maxiter: i64,
56        clmethod: &str,
57        niter: i64,
58    ) -> PyResult<Py<PyAny>> {
59        let _ = force;
60        let region = read_region_argument(py, region, "statistics");
61        let expression = stats_mask(mask);
62        let (value, logs) = py.detach(|| -> PyResult<_> {
63            let state = self.state();
64            let Some(attachment) = state.image.as_ref() else {
65                return Ok((CasaValue::empty_record(), Vec::new()));
66            };
67            let include = parse_pixel_range(includepix.as_ref().map(|v| v.0.as_slice()), "include")
68                .map_err(raised)?;
69            let exclude = parse_pixel_range(excludepix.as_ref().map(|v| v.0.as_slice()), "exclude")
70                .map_err(raised)?;
71            let region = region?;
72            let mask = expression?;
73            let rank = attachment.lock().shape().len();
74            let raw_axes = axes.map_or_else(Vec::new, |arg| arg.0);
75            let parsed_axes = parse_cursor_axes(&raw_axes, rank).map_err(raised)?;
76            let request = StatisticsRequest {
77                axes: Some(parsed_axes),
78                region: region
79                    .as_ref()
80                    .map(Region::as_region)
81                    .or(None::<SubimageRegion<'_>>),
82                mask: (!mask.is_empty()).then_some(mask.as_str()),
83                stretch,
84                includepix: include,
85                excludepix: exclude,
86                list,
87                robust,
88                logfile: (!logfile.is_empty()).then(|| Path::new(logfile)),
89                append,
90                algorithm,
91                fence,
92                center,
93                lside,
94                zscore,
95                maxiter: maxiter as i32,
96                clmethod,
97                niter: niter as i32,
98                disk,
99                verbose,
100            };
101            let result = image_statistics(attachment, &request).map_err(raised)?;
102            Ok((CasaValue::from_record(&result.record), result.logs))
103        })?;
104        emit_tool_logs(py, &logs)?;
105        value.into_python(py)
106    }
107}
