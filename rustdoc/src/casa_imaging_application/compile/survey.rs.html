3//! Compile's survey of the selected observation: one traversal of the
4//! partitions the request selects ([`super::partitions::candidates`]) that
5//! reads no visibilities, for the facts the later stages need.
6//!
7//! The traversal reads each partition's selected fields' times and, when a
8//! UV-distance interval or a W-projecting gridder needs them, the samples'
9//! UVW; nothing else. A time's field comes with the batch, and a
10//! partition's samples share one spectral window and observing session, so
11//! the survey learns the selected fields, observations, times and windows
12//! from the batches' coordinates. The epochs (time, field) of each window's
13//! selected samples are what its frequency conversions are later evaluated
14//! at ([`super::envelope`]).
15

32/// The least bytes one batch of the survey holds: enough for several times
33/// of a large array, so a partition's survey is not a batch per time.
34const SURVEY_BATCH_BYTES: usize = 1 << 18;
35
36/// The `|w|` envelope of the selected samples, in metres.
37#[derive(Clone, Copy)]
38pub(super) struct WRange {
39    pub(super) maximum_abs_m: f64,
40    pub(super) minimum_abs_m: f64,
41    pub(super) sum_squares_m2: f64,
42    pub(super) samples: u64,
43}
44
45impl WRange {
46    const EMPTY: Self = Self {
47        maximum_abs_m: 0.0,
48        minimum_abs_m: f64::INFINITY,
49        sum_squares_m2: 0.0,
50        samples: 0,
51    };
52
53    fn observe(&mut self, w_m: f64) {
54        self.maximum_abs_m = self.maximum_abs_m.max(w_m.abs());
55        self.minimum_abs_m = self.minimum_abs_m.min(w_m.abs());
56        self.sum_squares_m2 += w_m * w_m;
57        self.samples += 1;
58    }
59}

79/// What the survey establishes.
80pub(super) struct Survey {
81    /// The per-sample predicates.
82    pub(super) row_selection: RowSelection,
83    /// The partitions holding a selected sample, in partition order, with
84    /// every correlation each holds.
85    pub(super) partitions: Vec<PartitionSelection>,

91    pub(super) fields: BTreeSet<i32>,
92    pub(super) observation_ids: BTreeSet<i32>,
93    /// The earliest selected time: the anchor of the spectral frame and of
94    /// the image's observation date.
95    pub(super) first_time_mjd_seconds: f64,
96    pub(super) time_bounds_mjd_seconds: [f64; 2],
97    /// Read only for native AW ([`first_cross_sample`]).

127/// The facts gathered over every partition.
128struct Facts {
129    fields: BTreeSet<i32>,
130    observation_ids: BTreeSet<i32>,
131    time_bounds_mjd_seconds: [f64; 2],
132    w: WRange,
133    epochs: BTreeMap<usize, Vec<Epoch>>,

136impl SurveyPlan<'_> {
137    fn reads_uvw(&self) -> bool {
138        self.w_range || !matches!(self.row_selection.uv_distances(), UvSelection::All)
139    }
140
141    /// The request surveying `candidate`: its selected fields' times of its
142    /// first channel, with UVW when read.
143    fn request(&self, candidate: &Candidate, batch_bytes: usize) -> AccessRequest {
144        AccessRequest {
145            partitions: vec![candidate.id],
146            data_group: self.candidates.data_group.clone(),
147            roles: Roles {
148                values: false,
149                flags: false,
150                weights: false,
151                uvw: self.reads_uvw(),
152                time_centroid: false,
153                interval: false,
154                exposure: false,
155                scan: false,
156            },
157            weights: WeightRequest::Stored,
158            channels: 0..1,
159            times: TimeSelection::All,
160            fields: field_selection(self.row_selection.fields()),
161            correlations: CorrelationRequest {
162                deliver: Deliver::All,
163                flags_from: FlagsFrom::Delivered,
164            },
165            grouping: Grouping::None,
166            batch_bytes,
167        }
168    }
169
170    /// The bytes one time of `candidate`'s survey holds, with the cursor's
171    /// scratch.
172    fn bytes_per_time(&self, candidate: &Candidate) -> Result<usize, PrepareError> {
173        let estimate = self
174            .observation
175            .estimate(&self.request(candidate, usize::MAX))?;
176        Ok(estimate
177            .partitions
178            .first()
179            .map_or(0, casa_observation::PartitionEstimate::held_bytes_per_time))
180    }
181}

183/// Survey `plan`'s partitions, admitted on `host` under `policy`: the
184/// largest batch of the traversal (native AW's pass reads the same) and each
185/// window's epochs, one per selected time of each of its partitions at
186/// most. The charge is held until the problem compiles, since the epochs
187/// are read until then.
188///
189/// # Errors
190///
191/// When the admission is refused, a traversal fails, or the selection
192/// holds no sample.
193pub(super) fn survey(
194    plan: SurveyPlan<'_>,
195    spectral_windows: Vec<SourceSpectralWindow>,
196    (host, policy): (&HostResources, &ResourcePolicy),
197) -> Result<Admitted<Survey>, PrepareError> {
198    let catalogue = plan.observation.catalogue();
199    let mut batch_bytes = SURVEY_BATCH_BYTES;
200    let mut epoch_bound = BTreeMap::<usize, usize>::new();
201    for candidate in &plan.candidates.partitions {
202        batch_bytes = batch_bytes.max(plan.bytes_per_time(candidate)?);
203        let estimate = plan
204            .observation
205            .estimate(&plan.request(candidate, batch_bytes))?;
206        *epoch_bound.entry(candidate.spectral_window).or_default() += estimate
207            .partitions
208            .first()
209            .map_or(0, |partition| partition.times);
210    }
211    let epoch_bytes = epoch_bound.values().sum::<usize>() * size_of::<Epoch>();
212    // The survey's buffers grow to its largest batch, the cursor's scratch
213    // with them.
214    let reservation = admit(
215        host,
216        policy,
217        &Demand {
218            phase: "observation survey",
219            memory: (2 * batch_bytes + epoch_bytes) as u64,
220        },
221    )?;
222    let mut facts = Facts {
223        fields: BTreeSet::new(),
224        observation_ids: BTreeSet::new(),
225        time_bounds_mjd_seconds: [f64::INFINITY, f64::NEG_INFINITY],
226        w: WRange::EMPTY,
227        epochs: epoch_bound
228            .into_iter()
229            .map(|(window, bound)| (window, Vec::with_capacity(bound)))
230            .collect(),
231    };
232    let mut buffers = BatchBuffers::default();
233    let mut partitions = Vec::new();
234    let mut samples = 0;
235    let mut first_cross: Option<FirstCrossSample> = None;
236    for candidate in &plan.candidates.partitions {
237        let entry = catalogue.partition(candidate.id);
238        let request = plan.request(candidate, batch_bytes);
239        let selected = survey_partition(&plan, candidate, &request, &mut buffers, &mut facts)?;
240        if selected == 0 {
241            continue;
242        }
243        samples += selected;
244        if plan.first_cross
245            && let Some(sample) = first_cross_sample(&plan, candidate, &request, &mut buffers)?
246            && first_cross.is_none_or(|first| sample.time_mjd_seconds < first.time_mjd_seconds)
247        {
248            first_cross = Some(sample);
249        }
250        partitions.push(PartitionSelection::new(
251            candidate.id,
252            u32::try_from(candidate.spectral_window).expect("an SPW id is a stored i32"),
253            entry
254                .correlations
255                .iter()
256                .enumerate()
257                .map(|(index, correlation)| {
258                    CorrelationProduct::new(
259                        u32::try_from(index).expect("a CORR_TYPE index fits u32"),
260                        *correlation,
261                    )
262                })
263                .collect(),
264        ));
265    }
266    drop(buffers);
267    if samples == 0 {
268        return Err(PrepareError::NoSelectedSamples);
269    }
270    for epochs in facts.epochs.values_mut() {
271        epochs.sort_unstable_by(|a, b| {
272            a.time_mjd_seconds
273                .total_cmp(&b.time_mjd_seconds)
274                .then(a.field_id.cmp(&b.field_id))
275        });
276        epochs.dedup();
277    }
278    facts.epochs.retain(|_, epochs| !epochs.is_empty());
279    let mut spectral_windows = spectral_windows;
280    spectral_windows.retain(|window| facts.epochs.contains_key(&window.spw_id));
281    let source_frequency_reference = super::selection::one_source_frame(&spectral_windows)?;
282    Ok(Admitted {
283        value: Survey {
284            row_selection: plan.row_selection,
285            partitions,
286            samples,
287            spectral_windows,
288            source_frequency_reference,
289            fields: facts.fields,
290            observation_ids: facts.observation_ids,
291            first_time_mjd_seconds: facts.time_bounds_mjd_seconds[0],
292            time_bounds_mjd_seconds: facts.time_bounds_mjd_seconds,
293            first_cross,
294            w: facts.w,
295            epochs: facts.epochs,
296            continuum: None,
297        },
298        reservation,
299    })
300}

302/// Survey one partition into `facts`: the epochs, fields, observation,
303/// times and W range of its selected samples. Returns how many it holds.
304///
305/// Without a UV-distance interval every delivered sample but padding is
306/// selected, and the estimate counts them ([`casa_observation::PartitionEstimate::padded_fraction`]):
307/// a delivered time always holds one, since a time is on a partition's grid
308/// only when a sample of it was acquired. With one, the survey tests each
309/// sample whose UVW is finite (padding's is NaN).
310fn survey_partition(
311    plan: &SurveyPlan<'_>,
312    candidate: &Candidate,
313    request: &AccessRequest,
314    buffers: &mut BatchBuffers,
315    facts: &mut Facts,
316) -> Result<u64, PrepareError> {
317    let uv = plan.row_selection.uv_distances();
318    let tested = !matches!(uv, UvSelection::All);
319    let mut traversal = plan.observation.bind_read(request)?;
320    let estimate = traversal_samples(&traversal.estimate);
321    let mut samples = if tested { 0 } else { estimate };
322    let epochs = facts
323        .epochs
324        .get_mut(&candidate.spectral_window)
325        .expect("the survey bounds every candidate's window");
326    let origin = MjdHighPrec::new(0.0, 0.0);
327    while let Some(event) = traversal.next(buffers)? {
328        let ReadEvent::Batch(batch) = event else {
329            continue;
330        };
331        for time in 0..batch.times() {
332            let selected = if plan.reads_uvw() {
333                let held = observe_time(plan, candidate, &batch, time, &mut facts.w)?;
334                samples += if tested { held } else { 0 };
335                held > 0
336            } else {
337                true
338            };
339            if !selected {
340                continue;
341            }
342            let time_mjd_seconds = batch.time.seconds_since(time, origin);
343            let field_id = i32::try_from(batch.field[time].0).expect("a FIELD row is an i32 id");
344            facts.fields.insert(field_id);
345            let bounds = &mut facts.time_bounds_mjd_seconds;
346            bounds[0] = bounds[0].min(time_mjd_seconds);
347            bounds[1] = bounds[1].max(time_mjd_seconds);
348            epochs.push(Epoch {
349                time_mjd_seconds,
350                field_id,
351            });
352        }
353    }
354    if samples > 0 {
355        let entry = plan.observation.catalogue().partition(candidate.id);
356        facts
357            .observation_ids
358            .insert(i32::try_from(entry.observation.0).expect("an OBSERVATION row is an i32 id"));
359    }
360    Ok(samples)
361}

363/// The selected samples of time `time` of `batch` observed into `w`: those
364/// with finite UVW that `uv`, when tested, holds for its candidate's
365/// wavelength. Returns how many.
366fn observe_time(
367    plan: &SurveyPlan<'_>,
368    candidate: &Candidate,
369    batch: &VisibilityBatch<'_>,
370    time: usize,
371    w: &mut WRange,
372) -> Result<u64, PrepareError> {
373    let uv = plan.row_selection.uv_distances();
374    let uvw = batch.group.uvw.expect("the survey reads UVW");
375    let per_time = batch.samples_per_time();
376    let mut held = 0;
377    for sample in time * per_time..(time + 1) * per_time {
378        let uvw_m = uvw.metres_at(sample);
379        if !uv.holds(uvw_m, candidate.wavelength_m) {
380            continue;
381        }
382        if !uvw_m.iter().all(|value| value.is_finite()) {
383            let baseline = batch.baselines[sample % per_time];
384            let time_mjd_seconds = batch.time.seconds_since(time, MjdHighPrec::new(0.0, 0.0));
385            if plan
386                .observation
387                // MSv2 offsets are the stored MJD-zero seconds. Preserve
388                // their bits for the index, including a stored -0.0.
389                .main_row(candidate.id, batch.time.offsets_s()[time], baseline)
390                .is_some()
391            {
392                return Err(PrepareError::NonFiniteUvw {
393                    partition: candidate.id.0,
394                    time_mjd_seconds,
395                    antennas: [baseline.antenna1.0, baseline.antenna2.0],
396                });
397            }
398            continue;
399        }
400        w.observe(uvw_m[2]);
401        held += 1;
402    }
403    Ok(held)
404}

406/// The samples `estimate`'s one partition delivers that are not padding.
407fn traversal_samples(estimate: &casa_observation::Estimate) -> u64 {
408    let partition = estimate
409        .partitions
410        .first()
411        .expect("a survey estimates one partition");
412    let cells = partition.times * partition.baselines;
413    let padded = (cells as f64 * partition.padded_fraction).round() as usize;
414    u64::try_from(cells.saturating_sub(padded)).expect("a sample count fits u64")
415}

417/// The first sample of `candidate`'s selected times, in time order, that
418/// correlates two antennas, is selected by UV distance and whose MAIN row
419/// is not row-flagged: CASA's AW projection evaluates its parallactic angle
420/// at the first row of the first buffer that is a cross-correlation and not
421/// row-flagged (`getPA`), whatever its cell flags. Padding has no row. The
422/// traversal, `request`, is the survey's own; it stops there.
423fn first_cross_sample(
424    plan: &SurveyPlan<'_>,
425    candidate: &Candidate,
426    request: &AccessRequest,
427    buffers: &mut BatchBuffers,
428) -> Result<Option<FirstCrossSample>, PrepareError> {
429    let uv = plan.row_selection.uv_distances();
430    let mut traversal = plan.observation.bind_read(request)?;
431    let observation = plan
432        .observation
433        .catalogue()
434        .partition(candidate.id)
435        .observation;
436    let origin = MjdHighPrec::new(0.0, 0.0);
437    while let Some(event) = traversal.next(buffers)? {
438        let ReadEvent::Batch(batch) = event else {
439            continue;
440        };
441        for sample in 0..batch.samples() {
442            let baseline = batch.baselines[sample % batch.samples_per_time()];
443            if baseline.is_auto() {
444                continue;
445            }
446            if let Some(uvw) = batch.group.uvw
447                && !uv.holds(uvw.metres_at(sample), candidate.wavelength_m)
448            {
449                continue;
450            }
451            let time = sample / batch.samples_per_time();
452            let time_mjd_seconds = batch.time.seconds_since(time, origin);
453            if plan
454                .observation
455                .row_flagged(candidate.id, time_mjd_seconds, baseline)
456                .map_err(PrepareError::Opening)?
457                != Some(false)
458            {
459                continue;
460            }
461            return Ok(Some(FirstCrossSample {
462                time_mjd_seconds,
463                field_id: i32::try_from(batch.field[time].0).expect("a FIELD row is an i32 id"),
464                observation_id: i32::try_from(observation.0)
465                    .expect("an OBSERVATION row is an i32 id"),
466            }));
467        }
468    }
469    Ok(None)
470}

619                .expect("set");
620        }
621        for (row, (row_flag, all_flagged)) in rows.into_iter().enumerate() {
622            let Value::Array(ArrayValue::Bool(mut flags)) = ms
623                .main_table()
624                .column_accessor("FLAG")
625                .expect("FLAG")
626                .get(row)
627                .expect("a row")
628                .expect("a cell")
629                .clone()
630            else {
631                panic!("FLAG is a Bool array");
632            };
633            flags.fill(all_flagged);
634            for (column, value) in [
635                ("FLAG", Value::Array(ArrayValue::Bool(flags))),
636                ("FLAG_ROW", Value::Scalar(ScalarValue::Bool(row_flag))),
637                ("ANTENNA1", Value::Scalar(ScalarValue::Int32(0))),
638                ("ANTENNA2", Value::Scalar(ScalarValue::Int32(1))),
639                (
640                    "TIME",
641                    Value::Scalar(ScalarValue::Float64(T0 + row as f64 * 3600.0)),
642                ),
643            ] {
644                ms.main_table_mut()
645                    .cell_accessor_mut(row, column)
646                    .expect("a cell")
647                    .set(value)
648                    .expect("set");
649            }
650        }
651        ms.save().expect("saved");
652        drop(ms);
653        let observation = Msv2Observation::open(&path).expect("the cut MeasurementSet opens");
654        let id = observation.catalogue().partition_ids()[0];
655        let entry = observation.catalogue().partition(id);
656        let candidate = Candidate {
657            id,
658            spectral_window: entry.window.0 as usize,
659            wavelength_m: None,
660        };
661        let plan = SurveyPlan {
662            observation: &observation,
663            candidates: Candidates {
664                data_group: "base".into(),
665                partitions: vec![],
666            },
667            row_selection: RowSelection::new(
668                IdSelection::All,
669                UvSelection::All,
670                IntentSelection::All,
671            ),
672            w_range: false,
673            first_cross: true,
674        };
675        let request = plan.request(&candidate, 1 << 20);
676        first_cross_sample(&plan, &candidate, &request, &mut BatchBuffers::default())
677            .expect("the pass reads")
678            .map(|sample| sample.time_mjd_seconds)
679    }
