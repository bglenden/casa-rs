2//! Convolving an image's direction planes with a Gaussian, as the image
3//! tool's `convolve2d` does (CASA's `Image2DConvolver`).
4//!
5//! `simutil.convimage` convolves a Jy/pixel sky model with the clean beam to
6//! compare it with the clean image. For an image in Jy/pixel, or any flux
7//! per pixel, CASA's rules (`Image2DConvolver::_dealWithRestoringBeam`) are:
8//! - the kernel is the Gaussian at unit peak, unscaled, so a point of 1
9//!   Jy/pixel becomes a peak of 1 Jy/beam;
10//! - the output is in Jy/beam, with the Gaussian as its restoring beam.
11//!
12//! The kernel is sampled on a square of an odd number of pixels: ±5σ of
13//! the wider axis, rounded as `Image2DConvolver::_sizeOfGaussian` rounds
14//! it. It is centred on the middle pixel, with the beam's position angle
15//! (north through east) turned into the pixel frame at the reference pixel.
16//! The convolution is linear (casacore's `LatticeConvolver` with
17//! `ConvEnums::LINEAR`): the plane is padded with zeros, so flux does not
18//! wrap around the edges, and the output has the input's shape. Masked
19//! pixels count as zero, and the output keeps the input's mask.
20//!
21//! As in CASA, the pixels must be square. Images in other units (Jy/beam,
22//! whose beam CASA convolves numerically, or none, whose kernel it scales to
23//! unit volume) are refused: the simulation tasks convolve Jy/pixel images
24//! only.

40/// An image convolved with a Gaussian.
41#[derive(Clone, Debug)]
42pub struct Convolved {
43    /// The pixels, in the input's shape.
44    pub data: ArrayD<f32>,
45    /// The input's mask, when it has one.
46    pub mask: Option<ArrayD<bool>>,
47    /// The output's brightness unit, `Jy/beam`.
48    pub units: String,
49    /// The output's restoring beam: the Gaussian, its position angle in
50    /// [−90°, 90°] as casacore's `GaussianBeam` keeps it.
51    pub beam: GaussianBeam,
52}

54/// The direction coordinate and axes in longitude-latitude order.
55fn direction_axes(
56    coordinates: &CoordinateSystem,
57) -> Result<([usize; 2], &DirectionCoordinate), ImageError> {
58    let index = coordinates
59        .find_coordinate(CoordinateType::Direction)
60        .ok_or_else(|| {
61            ImageError::InvalidMetadata("convolve2d needs a direction coordinate".into())
62        })?;
63    let CoordinateModel::Direction(direction) = coordinates.coordinate(index) else {
64        unreachable!("find_coordinate found a direction coordinate");
65    };
66    let axes = coordinates
67        .coordinate_pixel_axes(index)
68        .expect("existing coordinate");
69    let [Some(longitude), Some(latitude)] = axes.as_slice() else {
70        return Err(ImageError::InvalidMetadata(
71            "convolve2d needs both direction axes".into(),
72        ));
73    };
74    Ok(([*longitude, *latitude], direction))
75}

77fn require_square_pixels(direction: &DirectionCoordinate) -> Result<(), ImageError> {
78    let increment = direction.increment();
79    let (x, y) = (increment[0].abs(), increment[1].abs());
80    if (x - y).abs() > 1e-13 * x.max(y) {
81        return Err(ImageError::InvalidMetadata(
82            "Pixels must be square, please regrid your image so that they are".into(),
83        ));
84    }
85    Ok(())
86}

88/// The sky offsets (east, north), rad, of a pixel step along each axis at
89/// the reference pixel: `[[east per x, east per y], [north per x, north per
90/// y]]`, by central differences of the coordinate's own conversion. It
91/// carries the increments, the PC matrix and the projection, as casacore's
92/// `GaussianShape::toPixel` does when `worldWidthsToPixel` turns a beam into
93/// pixels.
94pub(crate) fn sky_per_pixel(direction: &DirectionCoordinate) -> Result<[[f64; 2]; 2], ImageError> {
95    const STEP: f64 = 0.5;
96    let crpix = direction.reference_pixel();
97    let world = |dx: f64, dy: f64| direction.to_world(&[crpix[0] + dx, crpix[1] + dy]);
98    let reference = world(0.0, 0.0)?;
99    let (longitude, latitude) = (reference[0], reference[1]);
100    let (sin_lon, cos_lon) = longitude.sin_cos();
101    let (sin_lat, cos_lat) = latitude.sin_cos();
102    let east = [-sin_lon, cos_lon, 0.];
103    let north = [-sin_lat * cos_lon, -sin_lat * sin_lon, cos_lat];
104    let line_of_sight = |world: &[f64]| {
105        let (sin_lon, cos_lon) = world[0].sin_cos();
106        let (sin_lat, cos_lat) = world[1].sin_cos();
107        [cos_lat * cos_lon, cos_lat * sin_lon, sin_lat]
108    };
109    let mut jacobian = [[0.0; 2]; 2];
110    for (axis, (dx, dy)) in [(STEP, 0.0), (0.0, STEP)].into_iter().enumerate() {
111        let plus = line_of_sight(&world(dx, dy)?);
112        let minus = line_of_sight(&world(-dx, -dy)?);
113        let project = |basis: [f64; 3]| {
114            (0..3).map(|i| basis[i] * (plus[i] - minus[i])).sum::<f64>() / (2.0 * STEP)
115        };
116        jacobian[0][axis] = project(east);
117        jacobian[1][axis] = project(north);
118    }
119    Ok(jacobian)
120}

122/// `Image2DConvolver::_sizeOfGaussian`: twice (5σ rounded, plus one).
123fn size_of_gaussian(fwhm_pixels: f64) -> usize {
124    let sigma = fwhm_pixels / (8.0 * LN_2).sqrt();
125    ((5.0 * sigma + 0.5) as usize + 1) * 2
126}

128/// A sampled Gaussian defined by its two projected, normalized pixel axes.
129/// The footprint is odd and includes all pixels out to CASA's five-sigma
130/// cutoff. Its values can also be generated a tile at a time.
131#[derive(Clone, Copy)]
132pub(crate) struct GaussianSamples {
133    pub(crate) side: usize,
134    along: [f64; 2],
135    across: [f64; 2],
136    determinant_root: f64,
137    minor_pixels: f64,
138}
139
140impl GaussianSamples {
141    pub(crate) fn new(beam: GaussianBeam, sky: [[f64; 2]; 2]) -> Result<Self, ImageError> {
142        if !beam.major.is_finite()
143            || beam.major <= 0.0
144            || !beam.minor.is_finite()
145            || beam.minor < 0.0
146            || beam.minor > beam.major
147            || !beam.position_angle.is_finite()
148        {
149            return Err(ImageError::InvalidMetadata(
150                "invalid Gaussian kernel dimensions".into(),
151            ));
152        }
153        let (sin_pa, cos_pa) = beam.position_angle.sin_cos();
154        // A target-resolution difference may have rank one. Sample its
155        // limiting Gaussian without dividing by zero or rounding positive
156        // values onto the axis.
157        let minor = if beam.minor == 0.0 {
158            beam.major * 1e-12
159        } else {
160            beam.minor
161        };
162        let along = [
163            (sky[0][0] * sin_pa + sky[1][0] * cos_pa) / beam.major,
164            (sky[0][1] * sin_pa + sky[1][1] * cos_pa) / beam.major,
165        ];
166        let across = [
167            (sky[0][0] * cos_pa - sky[1][0] * sin_pa) / minor,
168            (sky[0][1] * cos_pa - sky[1][1] * sin_pa) / minor,
169        ];
170        let a = along[0].powi(2) + across[0].powi(2);
171        let b = along[1].powi(2) + across[1].powi(2);
172        let c = along[0] * along[1] + across[0] * across[1];
173        let determinant_root =
174            (sky[0][0] * sky[1][1] - sky[0][1] * sky[1][0]).abs() / beam.major / minor;
175        let largest_root = ((a + b + (a - b).hypot(2.0 * c)) / 2.0).sqrt();
176        let major_pixels = largest_root / determinant_root;
177        if !major_pixels.is_finite()
178            || major_pixels <= 0.0
179            || major_pixels >= (isize::MAX / 16) as f64
180        {
181            return Err(ImageError::InvalidMetadata(
182                "invalid Gaussian kernel footprint".into(),
183            ));
184        }
185        let side = size_of_gaussian(major_pixels) + 1;
186        Ok(Self {
187            side,
188            along,
189            across,
190            determinant_root,
191            minor_pixels: largest_root.recip(),
192        })
193    }
194
195    pub(crate) fn sample(&self, x: usize, y: usize) -> f64 {
196        let center = self.side / 2;
197        let dx = (x as isize - center as isize) as f64;
198        let dy = (y as isize - center as isize) as f64;
199        let major = self.along[0] * dx + self.along[1] * dy;
200        let minor = self.across[0] * dx + self.across[1] * dy;
201        (-4.0 * LN_2 * (major.powi(2) + minor.powi(2))).exp()
202    }
203
204    pub(crate) fn precision(&self) -> [[f64; 2]; 2] {
205        let entry =
206            |i: usize, j: usize| self.along[i] * self.along[j] + self.across[i] * self.across[j];
207        [[entry(0, 0), entry(0, 1)], [entry(0, 1), entry(1, 1)]]
208    }
209
210    pub(crate) fn well_sampled(&self) -> bool {
211        self.minor_pixels >= 8.
212    }
213
214    pub(crate) fn volume(&self) -> f64 {
215        if self.side > 129 {
216            if self.well_sampled() {
217                // At this sampling the discrete and continuous volumes
218                // differ only by the far Gaussian tails outside the stamp.
219                return PI / (4. * LN_2 * self.determinant_root);
220            }
221            let mid = (self.side / 2) as isize;
222            return lattice_gaussian_volume(
223                self.precision(),
224                self.determinant_root.powi(2),
225                [0., 0.],
226                [[-mid, mid]; 2],
227            );
228        }
229        let mut sum = 0.;
230        for y in 0..self.side {
231            for x in 0..self.side {
232                sum += self.sample(x, y);
233            }
234        }
235        sum
236    }
237
238    fn collect(self) -> Result<Vec<f64>, ImageError> {
239        let too_large =
240            |_| ImageError::InvalidMetadata("Gaussian kernel exceeds available storage".into());
241        let count = self
242            .side
243            .checked_mul(self.side)
244            .ok_or_else(|| too_large(()))?;
245        let mut values = Vec::new();
246        values.try_reserve_exact(count).map_err(|_| too_large(()))?;
247        for y in 0..self.side {
248            for x in 0..self.side {
249                values.push(self.sample(x, y));
250            }
251        }
252        Ok(values)
253    }
254}

256// Sum a possibly thin Gaussian by completing the square along the narrower
257// lattice direction. Expand each strip to a tail well below Float precision.
258fn lattice_gaussian_volume(
259    precision: [[f64; 2]; 2],
260    determinant: f64,
261    mean: [f64; 2],
262    bounds: [[isize; 2]; 2],
263) -> f64 {
264    const LIMIT: f64 = 14.;
265    let first = usize::from(precision[1][1] > precision[0][0]);
266    let second = 1 - first;
267    let a = precision[first][first];
268    let tilt = precision[first][second] / a;
269    let reduced = determinant / a;
270    let row_radius = (LIMIT / reduced).sqrt();
271    let row_first = ((mean[second] - row_radius).ceil() as isize).max(bounds[second][0]);
272    let row_last = ((mean[second] + row_radius).floor() as isize).min(bounds[second][1]);
273    let mut volume = 0.;
274    for row in row_first..=row_last {
275        let row_delta = row as f64 - mean[second];
276        let row_cost = reduced * row_delta * row_delta;
277        let center = mean[first] - tilt * row_delta;
278        let radius = ((LIMIT - row_cost).max(0.) / a).sqrt();
279        let first_pixel = ((center - radius).ceil() as isize).max(bounds[first][0]);
280        let last_pixel = ((center + radius).floor() as isize).min(bounds[first][1]);
281        for pixel in first_pixel..=last_pixel {
282            let form = a * (pixel as f64 - center).powi(2) + row_cost;
283            volume += (-4. * LN_2 * form).exp();
284        }
285    }
286    volume
287}

289fn kernel(beam: &GaussianBeam, sky: [[f64; 2]; 2]) -> Result<(usize, Vec<f64>), ImageError> {
290    let sampled = GaussianSamples::new(*beam, sky)?;
291    Ok((sampled.side, sampled.collect()?))
292}

294/// The smallest length at least `n` with no prime factor above 7, which
295/// FFTW transforms fastest.
296fn fast_length(n: usize) -> usize {
297    (n..)
298        .find(|&m| {
299            let mut rest = m;
300            for p in [2, 3, 5, 7] {
301                while rest % p == 0 {
302                    rest /= p;
303                }
304            }
305            rest == 1
306        })
307        .expect("a length with small prime factors exists")
308}

310/// The linear convolution of one plane (first axis fastest) with the
311/// kernel, cut to the plane's shape about the kernel's centre.
312struct PlaneConvolver {
313    plane: [usize; 2],
314    padded: [usize; 2],
315    kernel_side: usize,
316    kernel_transform: Vec<Complex64>,
317    fft: Fft2<f64>,
318}
319
320impl PlaneConvolver {
321    fn new(plane: [usize; 2], kernel_side: usize, kernel: &[f64]) -> Result<Self, ImageError> {
322        let padded = [
323            fast_length(plane[0] + kernel_side - 1),
324            fast_length(plane[1] + kernel_side - 1),
325        ];
326        // Row-major [rows = second axis][columns = first axis].
327        let mut fft = Fft2::<f64>::with_threads([padded[1], padded[0]], 1)
328            .map_err(|error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}")))?
329            .with_estimated_plan();
330        let mut kernel_transform = vec![Complex64::new(0.0, 0.0); padded[0] * padded[1]];
331        for j in 0..kernel_side {
332            for i in 0..kernel_side {
333                kernel_transform[i + j * padded[0]] =
334                    Complex64::new(kernel[i + j * kernel_side], 0.0);
335            }
336        }
337        fft.transform(&mut kernel_transform, false)
338            .map_err(|error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}")))?;
339        Ok(Self {
340            plane,
341            padded,
342            kernel_side,
343            kernel_transform,
344            fft,
345        })
346    }
347
348    fn convolve(&mut self, values: &[f64]) -> Result<Vec<f64>, ImageError> {
349        let [nx, ny] = self.plane;
350        let [px, py] = self.padded;
351        let mut grid = vec![Complex64::new(0.0, 0.0); px * py];
352        for j in 0..ny {
353            for i in 0..nx {
354                grid[i + j * px] = Complex64::new(values[i + j * nx], 0.0);
355            }
356        }
357        let fft_error = |error| ImageError::InvalidMetadata(format!("convolve2d FFT: {error}"));
358        self.fft.transform(&mut grid, false).map_err(fft_error)?;
359        for (value, kernel) in grid.iter_mut().zip(&self.kernel_transform) {
360            *value *= kernel;
361        }
362        self.fft.transform(&mut grid, true).map_err(fft_error)?;
363        let norm = (px * py) as f64;
364        let centre = (self.kernel_side - 1) / 2;
365        let mut out = vec![0.0; nx * ny];
366        for j in 0..ny {
367            for i in 0..nx {
368                out[i + j * nx] = grid[(i + centre) + (j + centre) * px].re / norm;
369            }
370        }
371        Ok(out)
372    }
373}

375/// `ia.convolve2d(major, minor, pa)` with a Gaussian on the direction plane
376/// of `input`, an image in a flux per pixel.
377///
378/// # Errors
379///
380/// An image without a direction coordinate, with pixels that are not square,
381/// or in a unit that is not a flux per pixel; a failure to read the image.
382pub fn convolve2d(
383    input: &impl ImageInterface<f32>,
384    beam: &GaussianBeam,
385) -> Result<Convolved, ImageError> {
386    let (axes, direction) = direction_axes(input.coordinates())?;
387    require_square_pixels(direction)?;
388    if !input.units().to_ascii_uppercase().contains("/PIXEL") {
389        return Err(ImageError::InvalidMetadata(format!(
390            "convolve2d: casa-rs convolves images in Jy/pixel; this one is in {:?}",
391            input.units()
392        )));
393    }
394    let shape = input.shape().to_vec();
395    let plane = axes.map(|axis| shape[axis]);
396    let (side, values) = kernel(beam, sky_per_pixel(direction)?)?;
397    let mut convolver = PlaneConvolver::new(plane, side, &values)?;
398    let mut data = input.get()?;
399    let mask = input.default_mask()?;
400    let others: Vec<usize> = (0..shape.len()).filter(|a| !axes.contains(a)).collect();
401    let planes: usize = others.iter().map(|a| shape[*a]).product();
402    let mut index = vec![0usize; shape.len()];
403    let mut values = vec![0.0f64; plane[0] * plane[1]];
404    for plane_number in 0..planes {
405        let mut rest = plane_number;
406        for a in &others {
407            index[*a] = rest % shape[*a];
408            rest /= shape[*a];
409        }
410        for j in 0..plane[1] {
411            for i in 0..plane[0] {
412                index[axes[0]] = i;
413                index[axes[1]] = j;
414                let good = mask
415                    .as_ref()
416                    .is_none_or(|mask| mask[ndarray::IxDyn(&index)]);
417                values[i + j * plane[0]] = if good {
418                    f64::from(data[ndarray::IxDyn(&index)])
419                } else {
420                    0.0
421                };
422            }
423        }
424        let convolved = convolver.convolve(&values)?;
425        for j in 0..plane[1] {
426            for i in 0..plane[0] {
427                index[axes[0]] = i;
428                index[axes[1]] = j;
429                data[ndarray::IxDyn(&index)] = convolved[i + j * plane[0]] as f32;
430            }
431        }
432    }
433    Ok(Convolved {
434        data,
435        mask,
436        units: "Jy/beam".into(),
437        beam: GaussianBeam::new(
438            beam.major,
439            beam.minor,
440            unwrap_position_angle(beam.position_angle),
441        ),
442    })
443}

445/// Homogenize a spectral moment's selected planes to their common beam.
446/// The pixel mask remains on the same positions as the input.
447pub(crate) fn convolve_to_common_beam<T>(
448    data: &mut ArrayD<T>,
449    mask: Option<&ArrayD<bool>>,
450    coordinates: &CoordinateSystem,
451    units: &str,
452    beams: &crate::ImageBeamSet,
453) -> Result<(GaussianBeam, String), ImageError>
454where
455    T: Into<f64> + From<f32> + Copy,
456{
457    let target = beams.common_beam()?;
458    let (axes, direction) = direction_axes(coordinates)?;
459    require_square_pixels(direction)?;
460    let sky = sky_per_pixel(direction)?;
461    let shape = data.shape().to_vec();
462    let plane = axes.map(|axis| shape[axis]);
463    let beam_axis = |kind| {
464        coordinates
465            .find_coordinate(kind)
466            .and_then(|i| coordinates.coordinate_pixel_axes(i))
467            .and_then(|axes| axes.first().copied().flatten())
468    };
469    let spectral = beam_axis(CoordinateType::Spectral);
470    let stokes = beam_axis(CoordinateType::Stokes);
471    let others: Vec<_> = (0..shape.len()).filter(|i| !axes.contains(i)).collect();
472    let count = others.iter().map(|&i| shape[i]).product();
473    let upper_units = units.to_ascii_uppercase();
474    let per_pixel = upper_units.contains("/PIXEL");
475    let per_beam = upper_units.contains("BEAM");
476    let mut units = units.to_owned();
477    let mut index = vec![0; shape.len()];
478    let mut values = vec![0.0; plane[0] * plane[1]];
479    for mut number in 0..count {
480        for &i in &others {
481            index[i] = number % shape[i];
482            number /= shape[i];
483        }
484        let channel = spectral.map_or(0, |i| index[i]);
485        let polarization = stokes.map_or(0, |i| index[i]);
486        let input = *beams.beam(channel, polarization);
487        let Some(smoothing) = target.deconvolving_beam(input)? else {
488            continue;
489        };
490        if smoothing.minor <= 0.0 {
491            return Err(ImageError::InvalidMetadata(
492                "the common-beam convolution requires a positive axial ratio".into(),
493            ));
494        }
495        let (side, mut weights) = kernel(&smoothing, sky)?;
496        let scale = if per_pixel {
497            // CASA loops over channels within each polarization. The last
498            // plane sets the output unit and keeps the input unit if copied.
499            if channel + 1 == beams.n_channels() && polarization + 1 == beams.n_stokes() {
500                units = "Jy/beam".into();
501            }
502            1.0
503        } else {
504            let (input_side, input_weights) = kernel(&input, sky)?;
505            let peak = central_gaussian_response(side, &weights, input_side, &input_weights);
506            if per_beam {
507                peak.recip()
508            } else {
509                input.area() / (target.area() * peak)
510            }
511        };
512        for weight in &mut weights {
513            *weight = f64::from((*weight * scale) as f32);
514        }
515        let mut convolver = PlaneConvolver::new(plane, side, &weights)?;
516        for j in 0..plane[1] {
517            index[axes[1]] = j;
518            for i in 0..plane[0] {
519                index[axes[0]] = i;
520                let at = ndarray::IxDyn(&index);
521                values[i + j * plane[0]] = if mask.is_none_or(|mask| mask[at.clone()]) {
522                    data[at].into()
523                } else {
524                    0.0
525                };
526            }
527        }
528        let convolved = convolver.convolve(&values)?;
529        for j in 0..plane[1] {
530            index[axes[1]] = j;
531            for i in 0..plane[0] {
532                index[axes[0]] = i;
533                data[ndarray::IxDyn(&index)] = T::from(convolved[i + j * plane[0]] as f32);
534            }
535        }
536    }
537    Ok((target, units))
538}

540fn central_gaussian_response(
541    left_side: usize,
542    left: &[f64],
543    right_side: usize,
544    right: &[f64],
545) -> f64 {
546    // Both odd-sized kernels are centred and even. The peak of their
547    // discrete convolution is their dot product over the central overlap.
548    let width = left_side.min(right_side);
549    let a = (left_side - width) / 2;
550    let b = (right_side - width) / 2;
551    let mut peak = 0.0;
552    for row in 0..width {
553        for col in 0..width {
554            peak += left[(row + a) * left_side + col + a] * right[(row + b) * right_side + col + b];
555        }
556    }
557    peak
558}

560/// A position angle in [−90°, 90°], as casacore's `GaussianBeam` keeps it.
561pub(crate) fn unwrap_position_angle(angle: f64) -> f64 {
562    use std::f64::consts::FRAC_PI_2;
563    let mut pa = angle % PI;
564    if pa > FRAC_PI_2 {
565        pa -= PI;
566    } else if pa < -FRAC_PI_2 {
567        pa += PI;
568    }
569    pa
570}

582    #[test]
583    fn convolution_stays_within_the_mapped_direction_plane() {
584        let step = 1.0e-6;
585        let mut coordinates = CoordinateSystem::new();
586        coordinates.add_coordinate(DirectionCoordinate::new(
587            DirectionRef::J2000,
588            Projection::new(ProjectionType::SIN),
589            [0.0, 0.0],
590            [-step, step],
591            [3.0, 3.0],
592        ));
593        coordinates.add_coordinate(casa_coordinates::LinearCoordinate::new(1, vec![], vec![]));
594        coordinates.transpose_axes(&[0, 1, 2], &[1, 2, 0]).unwrap();
595        let mut input = crate::TempImage::new(
596            vec![7, 2, 7],
597            coordinates,
598            casa_lattices::TempStoragePolicy::Memory,
599        )
600        .unwrap();
601        input.set_units("Jy/pixel").unwrap();
602        input.put_at(1.0, &[3, 1, 3]).unwrap();
603        let output = convolve2d(&input, &GaussianBeam::new(4.0 * step, 2.0 * step, 0.0)).unwrap();
604        for (index, expected) in [
605            ([3, 1, 3], 1.0),
606            ([5, 1, 3], 0.5),
607            ([3, 1, 4], 0.5),
608            ([3, 1, 5], 0.0625),
609            ([3, 0, 3], 0.0),
610        ] {
611            let value = output.data[index];
612            assert!((value - expected).abs() < 1e-5, "{index:?}: {value}");
613        }
614    }

624    #[test]
625    fn the_major_axis_lies_north_through_east_on_the_sky() {
626        // RA increases to the left (negative increment): a position angle of
627        // 90° (east) puts the major axis along −x, the same line as +x.
628        let beam = GaussianBeam::new(6.0e-6, 2.0e-6, std::f64::consts::FRAC_PI_2);
629        let (side, values) = kernel(&beam, UNROTATED).unwrap();
630        let c = (side - 1) / 2;
631        let along_x = values[(c + 3) + c * side];
632        let along_y = values[c + (c + 3) * side];
633        assert!(along_x > along_y, "{along_x} {along_y}");
634        // Peak 1 at the centre.
635        assert!((values[c + c * side] - 1.0).abs() < 1e-15);
636    }

638    #[test]
639    fn a_rotated_grid_turns_the_kernel_with_it() {
640        // The #941 review's case: PC [[0, −1], [1, 0]], 1″ pixels, a 6″ × 2″
641        // beam at PA 0°. Pixel x then runs north, so the major axis lies
642        // along x: half maximum 3 pixels out, as CASA 6.7.6 gives 0.5 there
643        // and 0.00195 three pixels along y.
644        let arcsec = PI / 648_000.0;
645        let rotated = DirectionCoordinate::new(
646            DirectionRef::J2000,
647            Projection::new(ProjectionType::SIN),
648            [1.0, 0.3],
649            [-arcsec, arcsec],
650            [10.0, 10.0],
651        )
652        .with_pc_matrix(ndarray::array![[0.0, -1.0], [1.0, 0.0]]);
653        let beam = GaussianBeam::new(6.0 * arcsec, 2.0 * arcsec, 0.0);
654        let (side, values) = kernel(&beam, sky_per_pixel(&rotated).unwrap()).unwrap();
655        let c = (side - 1) / 2;
656        assert!((values[(c + 3) + c * side] - 0.5).abs() < 1e-9);
657        assert!((values[c + (c + 3) * side] - 0.001_953_125).abs() < 1e-9);
658    }
