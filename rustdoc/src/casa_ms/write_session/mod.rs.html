48/// Resource inputs for a bounded writer.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
50pub struct MeasurementSetWriteResources {
51    /// Bytes available to all writer-owned batches.
52    pub available_bytes: usize,
53    /// Maximum simultaneously live batches, including the active writer batch.
54    pub maximum_live_batches: usize,
55    /// I/O buffer bytes reserved by each sequential tiled-column writer.
56    pub tiled_column_buffer_bytes: usize,
57}

59impl MeasurementSetWriteResources {
60    /// Derive an explicit writer budget from the same physical-memory policy as readers.
61    pub fn from_system_memory(
62        maximum_live_batches: usize,
63    ) -> Result<Self, MeasurementSetWriteError> {
64        let budget = MsSelectionIoBudget::from_system_memory(maximum_live_batches, 1, None)
65            .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?;
66        Ok(Self {
67            available_bytes: budget.available_bytes,
68            maximum_live_batches,
69            tiled_column_buffer_bytes: STREAMING_TILED_COLUMN_BUFFER_BYTES,
70        })
71    }
72}

74/// Bytes the visibility columns' streamed tiled writers hold for cells of
75/// `(correlations, channels)` with the visibility, UVW and weight tile
76/// shapes `tiles`: a band of tiles and an I/O buffer of `buffer_bytes`
77/// each, DATA's with `data_bands` bands, and the WEIGHT and SIGMA rows.
78fn visibility_writer_resident_bytes(
79    (correlations, channels): (usize, usize),
80    [visibility_tiles, uvw_tiles, weight_tiles]: [&[usize]; 3],
81    data_bands: usize,
82    buffer_bytes: usize,
83) -> Result<usize, MeasurementSetWriteError> {
84    let cell = [correlations, channels];
85    [
86        (&cell[..], visibility_tiles, 8 * data_bands),
87        (&cell[..], visibility_tiles, 1),
88        (&[3][..], uvw_tiles, 8),
89        (&[correlations][..], weight_tiles, 4),
90        (&[correlations][..], weight_tiles, 4),
91    ]
92    .into_iter()
93    .try_fold(0usize, |total, (cell_shape, tile_shape, element_bytes)| {
94        let bytes = sequential_tiled_writer_resident_bytes(
95            cell_shape,
96            tile_shape,
97            element_bytes,
98            buffer_bytes,
99        )?;
100        total
101            .checked_add(bytes)
102            .ok_or(MeasurementSetWriteError::ByteOverflow)
103    })?
104    .checked_add(
105        correlations
106            .checked_mul(2 * std::mem::size_of::<f32>())
107            .ok_or(MeasurementSetWriteError::ByteOverflow)?,
108    )
109    .ok_or(MeasurementSetWriteError::ByteOverflow)
110}
111
112fn sequential_tiled_writer_resident_bytes(
113    cell_shape: &[usize],
114    tile_shape: &[usize],
115    element_bytes: usize,
116    writer_buffer_bytes: usize,
117) -> Result<usize, MeasurementSetWriteError> {
118    if tile_shape.len() != cell_shape.len() + 1 || tile_shape.contains(&0) {
119        return Err(MeasurementSetWriteError::InvalidPlan(format!(
120            "tiled writer shape mismatch: cell={cell_shape:?} tile={tile_shape:?}"
121        )));
122    }
123    let non_row_tiles =
124        cell_shape
125            .iter()
126            .zip(tile_shape)
127            .try_fold(1usize, |total, (&cell, &tile)| {
128                total
129                    .checked_mul(cell.div_ceil(tile))
130                    .ok_or(MeasurementSetWriteError::ByteOverflow)
131            })?;
132    let tile_elements = tile_shape.iter().try_fold(1usize, |total, extent| {
133        total
134            .checked_mul(*extent)
135            .ok_or(MeasurementSetWriteError::ByteOverflow)
136    })?;
137    non_row_tiles
138        .checked_mul(tile_elements)
139        .and_then(|elements| elements.checked_mul(element_bytes))
140        .and_then(|tile_bytes| tile_bytes.checked_add(writer_buffer_bytes))
141        .ok_or(MeasurementSetWriteError::ByteOverflow)
142}

144/// Largest standard visibility cell implied by persisted MS metadata.
145///
146/// DATA_DESCRIPTION, POLARIZATION, and SPECTRAL_WINDOW are the authoritative
147/// shape relationship. This does not inspect MAIN payloads or assume that all
148/// rows share one casa-rs-created shape.
149#[doc(hidden)]
150pub fn maximum_visibility_cell_elements(
151    measurement_set: &MeasurementSet,
152) -> Result<usize, MeasurementSetWriteError> {
153    let data_description = measurement_set
154        .data_description()
155        .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?;
156    let polarization = measurement_set
157        .polarization()
158        .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?;
159    let spectral_window = measurement_set
160        .spectral_window()
161        .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?;
162    let mut maximum = 0usize;
163    for row in 0..data_description.row_count() {
164        let polarization_id = data_description
165            .polarization_id(row)
166            .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?;
167        let spectral_window_id = data_description
168            .spectral_window_id(row)
169            .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?;
170        let polarization_row = usize::try_from(polarization_id).map_err(|_| {
171            MeasurementSetWriteError::InvalidPlan(format!(
172                "DATA_DESCRIPTION row {row} has negative POLARIZATION_ID {polarization_id}"
173            ))
174        })?;
175        let spectral_window_row = usize::try_from(spectral_window_id).map_err(|_| {
176            MeasurementSetWriteError::InvalidPlan(format!(
177                "DATA_DESCRIPTION row {row} has negative SPECTRAL_WINDOW_ID {spectral_window_id}"
178            ))
179        })?;
180        let correlations = usize::try_from(
181            polarization
182                .num_corr(polarization_row)
183                .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?,
184        )
185        .map_err(|_| {
186            MeasurementSetWriteError::InvalidPlan(format!(
187                "POLARIZATION row {polarization_row} has a negative correlation count"
188            ))
189        })?;
190        let channels = usize::try_from(
191            spectral_window
192                .num_chan(spectral_window_row)
193                .map_err(|error| MeasurementSetWriteError::InvalidPlan(error.to_string()))?,
194        )
195        .map_err(|_| {
196            MeasurementSetWriteError::InvalidPlan(format!(
197                "SPECTRAL_WINDOW row {spectral_window_row} has a negative channel count"
198            ))
199        })?;
200        maximum = maximum.max(
201            correlations
202                .checked_mul(channels)
203                .ok_or(MeasurementSetWriteError::ByteOverflow)?,
204        );
205    }
206    Ok(maximum)
207}

536/// Failure from canonical MeasurementSet write planning or execution.
537#[derive(Debug, Error)]
538pub enum MeasurementSetWriteError {
539    /// Plan inputs are inconsistent or missing.
540    #[error("invalid MeasurementSet write plan: {0}")]
541    InvalidPlan(String),
542    /// One row cannot fit the stated batch budget.
543    #[error(
544        "one write row requires {bytes_per_row} bytes but the per-batch budget is {per_batch_budget} bytes"
545    )]
546    InsufficientBudget {
547        /// Exact writer-owned row bytes.
548        bytes_per_row: usize,
549        /// Bytes available to one batch.
550        per_batch_budget: usize,
551    },
552    /// Checked byte accounting overflowed.
553    #[error("MeasurementSet write byte accounting overflowed")]
554    ByteOverflow,
555    /// A column writer failed.
556    #[error("MeasurementSet column write failed: {0}")]
557    Column(String),
558    /// The background writer stopped or panicked.
559    #[error("MeasurementSet background writer failed: {0}")]
560    Background(String),
561    /// Installing a completed streamed column failed.
562    #[error("install MeasurementSet column {column}: {reason}")]
563    Install {
564        /// MAIN column.
565        column: String,
566        /// Storage error.
567        reason: String,
568    },
569    /// Preparing or publishing a new output failed.
570    #[error("MeasurementSet output {path}: {reason}")]
571    Output {
572        /// Final output path.
573        path: String,
574        /// Filesystem failure.
575        reason: String,
576    },
577    /// casacore's write lock on MAIN could not be taken or released.
578    ///
579    /// [`TableError::LockFailed`] means another handle in this process holds
580    /// it, another process wrote MAIN while this writer waited for it, or
581    /// waiting would deadlock; a lock another process holds is waited for.
582    #[error("casacore write lock on MeasurementSet {path}: {source}")]
583    WriteLock {
584        /// MeasurementSet (MAIN table) directory.
585        path: String,
586        /// Lock failure.
587        #[source]
588        source: TableError,
589    },
590    /// A mutation batch was given a MeasurementSet other than the one the
591    /// session locked and was started on.
592    #[error(
593        "selected-row mutation of MeasurementSet {session} was given MeasurementSet {batch}; \
594         a session writes only the MeasurementSet it locked"
595    )]
596    ForeignMeasurementSet {
597        /// MeasurementSet directory the session locked.
598        session: String,
599        /// MeasurementSet directory the batch was given.
600        batch: String,
601    },
602}

604/// Commit-last target for a newly created MeasurementSet.
605///
606/// The staging directory is not a snapshot or recovery generation. It is the
607/// sole new output, renamed into place only after every casacore file is complete.
608pub struct MeasurementSetCreateTarget {
609    final_path: PathBuf,
610    staging_path: PathBuf,
611    overwrite: bool,
612}
613
614impl MeasurementSetCreateTarget {
615    /// Reserve a detectable incomplete staging path for a new output.
616    pub fn prepare(
617        final_path: impl AsRef<Path>,
618        overwrite: bool,
619    ) -> Result<Self, MeasurementSetWriteError> {
620        let final_path = final_path.as_ref().to_path_buf();
621        if final_path.exists() && !overwrite {
622            return Err(MeasurementSetWriteError::Output {
623                path: final_path.display().to_string(),
624                reason: "output already exists".to_string(),
625            });
626        }
627        let file_name = final_path
628            .file_name()
629            .and_then(|name| name.to_str())
630            .ok_or_else(|| MeasurementSetWriteError::Output {
631                path: final_path.display().to_string(),
632                reason: "output must have a UTF-8 file name".to_string(),
633            })?;
634        let staging_path = final_path.with_file_name(format!(
635            ".{file_name}.casa-rs-incomplete-{}",
636            std::process::id()
637        ));
638        if staging_path.exists() {
639            return Err(MeasurementSetWriteError::Output {
640                path: final_path.display().to_string(),
641                reason: format!(
642                    "incomplete staging output already exists at {}",
643                    staging_path.display()
644                ),
645            });
646        }
647        Ok(Self {
648            final_path,
649            staging_path,
650            overwrite,
651        })
652    }
653
654    /// Path to use for all physical creation work.
655    pub fn staging_path(&self) -> &Path {
656        &self.staging_path
657    }

659    /// Publish the one completed output with a same-directory rename.
660    pub fn commit(self) -> Result<(), MeasurementSetWriteError> {
661        if self.final_path.exists() {
662            if !self.overwrite {
663                return Err(MeasurementSetWriteError::Output {
664                    path: self.final_path.display().to_string(),
665                    reason: "output appeared before publish".to_string(),
666                });
667            }
668            fs::remove_dir_all(&self.final_path)
669                .or_else(|_| fs::remove_file(&self.final_path))
670                .map_err(|error| MeasurementSetWriteError::Output {
671                    path: self.final_path.display().to_string(),
672                    reason: format!("remove existing output before publish: {error}"),
673                })?;
674        }
675        fs::rename(&self.staging_path, &self.final_path).map_err(|error| {
676            MeasurementSetWriteError::Output {
677                path: self.final_path.display().to_string(),
678                reason: format!("publish completed staging output: {error}"),
679            }
680        })
681    }
682}

731struct ScalarColumnFinalization {
732    overrides: ColumnOverrides,
733    paths: Vec<PathBuf>,
734    telemetry: Vec<MeasurementSetColumnWriteTelemetry>,
735    bytes_written: usize,
736}

738impl ScalarColumnFinalization {
739    fn empty(overrides: ColumnOverrides) -> Self {
740        Self {
741            overrides,
742            paths: Vec::new(),
743            telemetry: Vec::new(),
744            bytes_written: 0,
745        }
746    }

749fn complete_scalar_finalization(
750    mut telemetry: MeasurementSetWriteTelemetry,
751    scalar_columns: ScalarColumnFinalization,
752    scalar_finalize_seconds: f64,
753    save_seconds: f64,
754) -> Result<MeasurementSetWriteTelemetry, MeasurementSetWriteError> {
755    telemetry.bytes_written = telemetry
756        .bytes_written
757        .checked_add(scalar_columns.bytes_written)
758        .ok_or(MeasurementSetWriteError::ByteOverflow)?;
759    telemetry.columns.extend(scalar_columns.telemetry);
760    telemetry.write_seconds += save_seconds;
761    telemetry.finalize_seconds += scalar_finalize_seconds + save_seconds;
762    telemetry.producer_seconds =
763        (telemetry.producer_seconds - scalar_finalize_seconds - save_seconds).max(0.0);
764    let cleanup_started = Instant::now();
765    for path in scalar_columns.paths {
766        fs::remove_file(&path).map_err(|error| {
767            MeasurementSetWriteError::Column(format!(
768                "remove scalar construction file {}: {error}",
769                path.display()
770            ))
771        })?;
772    }
773    telemetry.finalize_seconds += cleanup_started.elapsed().as_secs_f64();
774    Ok(telemetry)
775}

788fn primitive_value_bytes(primitive_type: PrimitiveType) -> Option<usize> {
789    match primitive_type {
790        PrimitiveType::Bool | PrimitiveType::UInt8 => Some(1),
791        PrimitiveType::UInt16 | PrimitiveType::Int16 => Some(2),
792        PrimitiveType::UInt32 | PrimitiveType::Int32 | PrimitiveType::Float32 => Some(4),
793        PrimitiveType::Int64 | PrimitiveType::Float64 | PrimitiveType::Complex32 => Some(8),
794        PrimitiveType::Complex64 => Some(16),
795        PrimitiveType::String => None,
796    }
797}
798
799fn mutation_values_bytes(
800    values: &MeasurementSetMutationColumnValues,
801) -> Result<usize, MeasurementSetWriteError> {
802    match values {
803        MeasurementSetMutationColumnValues::Scalars(values) => {
804            values.iter().try_fold(0usize, |total, value| {
805                let bytes = match value {
806                    ScalarValue::String(value) => value.len(),
807                    other => primitive_value_bytes(other.primitive_type()).ok_or_else(|| {
808                        MeasurementSetWriteError::InvalidPlan(
809                            "unsupported scalar mutation value".to_string(),
810                        )
811                    })?,
812                };
813                total
814                    .checked_add(bytes)
815                    .ok_or(MeasurementSetWriteError::ByteOverflow)
816            })
817        }
818        MeasurementSetMutationColumnValues::Arrays(values) => {
819            values.iter().try_fold(0usize, |total, value| {
820                let element_bytes =
821                    primitive_value_bytes(value.primitive_type()).ok_or_else(|| {
822                        MeasurementSetWriteError::InvalidPlan(
823                            "string-array mutation byte accounting is unsupported".to_string(),
824                        )
825                    })?;
826                let bytes = value
827                    .len()
828                    .checked_mul(element_bytes)
829                    .ok_or(MeasurementSetWriteError::ByteOverflow)?;
830                total
831                    .checked_add(bytes)
832                    .ok_or(MeasurementSetWriteError::ByteOverflow)
833            })
834        }
835    }
836}

838#[allow(clippy::too_many_arguments)]
839fn write_visibility_batches(
840    receiver: mpsc::Receiver<MeasurementSetWriteBatch>,
841    returns: &mpsc::SyncSender<VisibilityColumns>,
842    mut data_writer: StreamingTiledShapeComplex32Writer,
843    mut flag_writer: StreamingTiledPrimitiveWriter,
844    mut uvw_writer: StreamingTiledPrimitiveWriter,
845    mut weight_writer: StreamingTiledPrimitiveWriter,
846    mut sigma_writer: StreamingTiledPrimitiveWriter,
847    correlation_count: usize,
848) -> Result<StreamedVisibilityColumns, MeasurementSetWriteError> {
849    let weight_row = vec![1.0f32; correlation_count];
850    let sigma_row = vec![1.0f32; correlation_count];
851    for batch in receiver {
852        match batch {
853            MeasurementSetWriteBatch::Columns(columns) => {
854                let row_count = columns.row_count();
855                data_writer
856                    .push_rows(&columns.data)
857                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
858                flag_writer
859                    .push_bool_fill_rows(&columns.flag_row)
860                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
861                uvw_writer
862                    .push_f64_rows(&columns.uvw)
863                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
864                visibility_columns::push_per_row_cells(
865                    &mut weight_writer,
866                    &columns.weight,
867                    &weight_row,
868                    row_count,
869                )?;
870                visibility_columns::push_per_row_cells(
871                    &mut sigma_writer,
872                    &columns.sigma,
873                    &sigma_row,
874                    row_count,
875                )?;
876                visibility_columns::recycle(returns, columns);
877            }
878            #[cfg(test)]
879            MeasurementSetWriteBatch::Repeated {
880                data_row,
881                flag_row,
882                uvw_row,
883                row_count,
884            } => {
885                data_writer
886                    .push_repeated_row(&data_row, row_count)
887                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
888                flag_writer
889                    .push_repeated_bool_fill_row(flag_row, row_count)
890                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
891                uvw_writer
892                    .push_repeated_f64_row(&uvw_row, row_count)
893                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
894                weight_writer
895                    .push_repeated_f32_row(&weight_row, row_count)
896                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
897                sigma_writer
898                    .push_repeated_f32_row(&sigma_row, row_count)
899                    .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?;
900            }
901        }
902    }
903    Ok(StreamedVisibilityColumns {
904        data: data_writer
905            .finish()
906            .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?,
907        flag: flag_writer
908            .finish()
909            .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?,
910        uvw: uvw_writer
911            .finish()
912            .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?,
913        weight: weight_writer
914            .finish()
915            .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?,
916        sigma: sigma_writer
917            .finish()
918            .map_err(|error| MeasurementSetWriteError::Column(error.to_string()))?,
919    })
920}

1003fn install_streamed_columns(
1004    main: &casa_tables::Table,
1005    output: &Path,
1006    plan: &MeasurementSetWritePlan,
1007    streamed: StreamedVisibilityColumns,
1008) -> Result<(), MeasurementSetWriteError> {
1009    let sequence = |column: &str| {
1010        crate::ms::measurement_set_main_data_manager_sequence(main, column).ok_or_else(|| {
1011            MeasurementSetWriteError::Install {
1012                column: column.to_string(),
1013                reason: "data-manager sequence is absent".to_string(),
1014            }
1015        })
1016    };
1017    install_streamed_tiled_shape_complex32_column(output, sequence("DATA")?, "DATA", streamed.data)
1018        .map_err(|error| install_error("DATA", error))?;
1019    install_streamed_tiled_shape_primitive_column(output, sequence("FLAG")?, "FLAG", streamed.flag)
1020        .map_err(|error| install_error("FLAG", error))?;
1021    // Undefined cells, laid out as casacore leaves a column no row has
1022    // written; the default tile is the one a later hypercube would get.
1023    let visibility_tile = &plan.visibility_tile_shape;
1024    install_undefined_tiled_shape_column(
1025        output,
1026        sequence("FLAG_CATEGORY")?,
1027        "FLAG_CATEGORY",
1028        plan.row_count,
1029        StreamedTiledPrimitiveType::Bool,
1030        3,
1031        &[
1032            visibility_tile[0],
1033            visibility_tile[1],
1034            1,
1035            visibility_tile[2],
1036        ],
1037    )
1038    .map_err(|error| install_error("FLAG_CATEGORY", error))?;
1039    install_streamed_tiled_column_primitive_column(output, sequence("UVW")?, "UVW", streamed.uvw)
1040        .map_err(|error| install_error("UVW", error))?;
1041    install_streamed_tiled_shape_primitive_column(
1042        output,
1043        sequence("WEIGHT")?,
1044        "WEIGHT",
1045        streamed.weight,
1046    )
1047    .map_err(|error| install_error("WEIGHT", error))?;
1048    install_streamed_tiled_shape_primitive_column(
1049        output,
1050        sequence("SIGMA")?,
1051        "SIGMA",
1052        streamed.sigma,
1053    )
1054    .map_err(|error| install_error("SIGMA", error))?;
1055    Ok(())
1056}
