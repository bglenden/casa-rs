19/// The cosmic background's temperature `simobserve` uses, K.
20const CMB_K: f64 = 2.725;
21
22/// The telescopes whose thermal noise CASA models.
23const KNOWN_TELESCOPES: [&str; 6] = ["ALMASD", "ALMA", "ACA", "SMA", "EVLA", "VLA"];
24
25/// The relative humidity `setnoise` takes by default, %.
26const RELATIVE_HUMIDITY_PERCENT: f64 = 20.0;

28/// Leakage, then thermal noise, as `leakage` and `thermalnoise` ask.
29pub(super) fn corrupt(run: &mut Run<'_>, sky: &Sky, array: &Array) -> Result<(), SimobserveError> {
30    let parameters = run.parameters;
31    if !run.uvmode && parameters.leakage > 0. {
32        tracing::warn!("Can't corrupt SD data with polarization leakage");
33    }
34    let fullpol = (parameters.leakage > 0. && run.uvmode && run.predict)
35        .then(|| leak(run))
36        .transpose()?;
37    if !parameters.thermalnoise.is_empty() {
38        add_noise(run, sky, array, fullpol.as_deref())?;
39    }
40    Ok(())
41}

43/// Copy the MeasurementSet to `<stem>.fullpol.ms` and corrupt it by
44/// leakages of amplitude `leakage`, written to `<stem>.fullpol.cal`.
45fn leak(run: &mut Run<'_>) -> Result<PathBuf, SimobserveError> {
46    let ms = run.measurement_set();
47    let fullpol = run.file(".fullpol.ms");
48    if ms.exists() {
49        tracing::warn!(
50            "Copying {} to {} and adding polarization leakage ...",
51            ms.display(),
52            fullpol.display()
53        );
54        remove(&fullpol)?;
55        copy_tree(&ms, &fullpol)?;
56        let table = run.file(".fullpol.cal");
57        make_way(&table)?;
58        let mut sm = Simulator::open_from_ms(&fullpol, run.request.options.clone())?;
59        sm.set_data(&[], &[], "")?;
60        sm.set_leakage(&LeakageSpec {
61            table: table.display().to_string(),
62            amplitude: vec![run.parameters.leakage],
63            offset: vec![0.0],
64        })?;
65        sm.corrupt()?;
66        sm.close()?;
67        run.wrote(&fullpol);
68        run.wrote(&table);
69    }
70    tracing::info!("Created {}", ms.display());
71    Ok(fullpol)
72}

74/// Copy the MeasurementSet (the leaked one when there is one) and add
75/// `thermalnoise`'s noise to the copy: Tsys from the receiver and
76/// efficiencies `noisetemp` gives, and the atmosphere `tsys-manual` gives
77/// or ATM's (`tsys-atm`).
78fn add_noise(
79    run: &mut Run<'_>,
80    sky: &Sky,
81    array: &Array,
82    fullpol: Option<&Path>,
83) -> Result<(), SimobserveError> {
84    let parameters = run.parameters;
85    tracing::warn!("Applying thermal noise ...");
86    let (source, noisy) = if !run.uvmode {
87        (run.file(".sd.ms"), run.file(".noisy.sd.ms"))
88    } else if let Some(fullpol) = fullpol {
89        (fullpol.to_path_buf(), run.file(".noisier.ms"))
90    } else {
91        (run.file(".ms"), run.file(".noisy.ms"))
92    };
93    if !source.exists() {
94        return Err(SimobserveError::raised(format!(
95            "Couldn't find {}",
96            source.display()
97        )));
98    }
99    tracing::warn!(
100        "Copying {} to {} and adding thermal noise ...",
101        source.display(),
102        noisy.display()
103    );
104    remove(&noisy)?;
105    copy_tree(&source, &noisy)?;
106    let telescope = if run.predict {
107        array.config.telescope.clone()
108    } else {
109        let telescope = observing_telescope(&noisy)?;
110        tracing::info!("telescopename read from {}: {telescope}", noisy.display());
111        telescope
112    };
113    if !KNOWN_TELESCOPES.contains(&telescope.as_str()) {
114        tracing::warn!("thermal noise only works properly for ALMA/ACA, (E)VLA, and SMA");
115    }
116    let correlator = Some(parameters.correlator.as_str()).filter(|name| !name.is_empty());
117    let receiver = noise_temperature(&telescope, &sky.model.specrefval, None, None, correlator)?;
118    let antenna = receiver.antenna();
119    // `senscoeff`: 1/√2 (CASA's default, asked for by -1) for an
120    // interferometer, 1 for a single dish.
121    let sensitivity = if run.uvmode { -1.0 } else { 1.0 };
122    if parameters.verbose {
123        tracing::info!("antenna efficiency    = {}", python(antenna));
124        tracing::info!("spillover efficiency  = {}", python(receiver.spillover));
125        tracing::info!("correlator name       = {}", parameters.correlator);
126        tracing::info!("correlator efficiency = {}", python(receiver.correlator));
127    }
128    let tsys = TsysParameters {
129        tau: parameters.tau0,
130        atmosphere_k: parameters.t_sky,
131        antenna_efficiency: antenna,
132        spillover_efficiency: receiver.spillover,
133        correlator_efficiency: receiver.correlator,
134        receiver_k: receiver.receiver_temperature_k,
135        ground_k: parameters.t_ground,
136        cmb_k: CMB_K,
137        sensitivity,
138        receiver_type: 0,
139    };
140    // `tsys-atm` names a table, `<noisy>.T.cal`, which CASA deletes for
141    // `.noisy` and `.noisy.sd` and leaves for `.noisier`; casa-rs writes
142    // none.
143    let spec = if parameters.thermalnoise == "tsys-manual" {
144        NoiseSpec::tsys_manual(tsys)
145    } else {
146        let defaults = TsysParameters::default();
147        NoiseSpec::tsys_atm(
148            TsysParameters {
149                tau: defaults.tau,
150                atmosphere_k: defaults.atmosphere_k,
151                ..tsys
152            },
153            parameters.user_pwv,
154            AtmosphereSpec {
155                relative_humidity_percent: RELATIVE_HUMIDITY_PERCENT,
156                ..AtmosphereSpec::default()
157            },
158        )
159    };
160    let seed = i32::try_from(parameters.seed).map_err(|_| {
161        SimobserveError::raised(format!(
162            "seed {} does not fit a 32-bit integer",
163            parameters.seed
164        ))
165    })?;
166    let mut sm = Simulator::open_from_ms(&noisy, run.request.options.clone())?;
167    sm.set_data(&[], &[], "")?;
168    sm.set_seed(seed);
169    sm.set_noise(&spec)?;
170    sm.corrupt()?;
171    sm.close()?;
172    run.wrote(&noisy);
173    if parameters.verbose {
174        tracing::info!("done corrupting with thermal noise");
175    }
176    Ok(())
177}

179/// OBSERVATION's first TELESCOPE_NAME.
180fn observing_telescope(ms: &Path) -> Result<String, SimobserveError> {
181    let failed = |reason: &dyn std::fmt::Display| SimobserveError::Table {
182        table: ms.join("OBSERVATION"),
183        reason: format!("TELESCOPE_NAME: {reason}"),
184    };
185    let set = MeasurementSet::open(ms).map_err(|error| failed(&error))?;
186    let observation = set.observation().map_err(|error| failed(&error))?;
187    observation
188        .string(0, "TELESCOPE_NAME")
189        .map_err(|error| failed(&error))
190}

192/// `simutil.ismstp` ([`analysis::is_total_power`]).
193pub(super) fn is_total_power(ms: &Path) -> Result<bool, SimobserveError> {
194    analysis::is_total_power(ms).map_err(|error| match error {
195        AnalysisError::Table { table, reason } => SimobserveError::Table { table, reason },
196        other => SimobserveError::Table {
197            table: ms.to_path_buf(),
198            reason: other.to_string(),
199        },
200    })
