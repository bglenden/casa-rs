3//! Thin PyO3 `ia.moments` wrapper delegating to [`casa_imageanalysis::tool::ops_transform::moments_image`].
4
5use casa_core::measures::doppler::DopplerRef;
6use casa_imageanalysis::tool::ops_transform::moments_image;
7use pyo3::prelude::*;
8
9use crate::convert::{DoubleList, IntList, StringList, Variant};
10use crate::tools::{LogLine, emit};
11
12use super::PyImageTool;
13use super::handle::{external_open_check, raised};
14use super::subimage::{RegionArgument, mask_argument, require_bool};
15
16#[pymethods]
17impl PyImageTool {
18    /// Collapse masked image profiles and return the first requested moment image.
19    #[allow(clippy::too_many_arguments)]
20    #[pyo3(signature = (moments=None, axis=-10, region=None, mask=None, method=None, smoothaxes=None, smoothtypes=None, smoothwidths=None, includepix=None, excludepix=None, peaksnr=3., stddev=0., doppler="RADIO", outfile="", smoothout="", overwrite=false, drop=true, stretch=false), text_signature = "($self, moments=[0], axis=-10, region={}, mask='', method=[], smoothaxes=[-1], smoothtypes=[], smoothwidths=[0.0], includepix=[-1.0], excludepix=[-1.0], peaksnr=3.0, stddev=0.0, doppler='RADIO', outfile='', smoothout='', overwrite=False, drop=True, stretch=False)")]
21    fn moments(
22        &self,
23        py: Python<'_>,
24        moments: Option<IntList>,
25        axis: i64,
26        region: Option<&Bound<'_, PyAny>>,
27        mask: Option<&Bound<'_, PyAny>>,
28        method: Option<StringList>,
29        smoothaxes: Option<IntList>,
30        smoothtypes: Option<&Bound<'_, PyAny>>,
31        smoothwidths: Option<DoubleList>,
32        includepix: Option<DoubleList>,
33        excludepix: Option<DoubleList>,
34        peaksnr: f64,
35        stddev: f64,
36        doppler: &str,
37        outfile: &str,
38        smoothout: &str,
39        #[pyo3(from_py_with = require_bool)] overwrite: bool,
40        #[pyo3(from_py_with = require_bool)] drop: bool,
41        #[pyo3(from_py_with = require_bool)] stretch: bool,
42    ) -> PyResult<Py<Self>> {
43        let moments = moments
44            .map_or_else(|| vec![0], |values| values.0)
45            .into_iter()
46            .map(i32::try_from)
47            .collect::<Result<Vec<_>, _>>()
48            .map_err(raised)?;
49        let region: RegionArgument =
50            region.map_or_else(|| Ok(Default::default()), |value| value.extract())?;
51        let mask = mask_argument(mask)?;
52        let methods = method.map_or_else(Vec::new, |value| value.0);
53        let axes = smoothaxes.map_or_else(|| vec![-1], |value| value.0);
54        let kernels = match smoothtypes
55            .map(|value| value.extract::<Variant>())
56            .transpose()?
57        {
58            Some(Variant::Text(name)) => name
59                .split(|character: char| character == ',' || character.is_ascii_whitespace())
60                .filter(|name| !name.is_empty())
61                .map(str::to_owned)
62                .collect(),
63            Some(Variant::Texts(names)) => names,
64            _ => Vec::new(),
65        };
66        let widths = smoothwidths
67            .filter(|value| !value.0.is_empty())
68            .map_or_else(|| vec![0.], |value| value.0);
69        let include = includepix.map_or_else(|| vec![-1.], |value| value.0);
70        let exclude = excludepix.map_or_else(|| vec![-1.], |value| value.0);
71        let parsed = if doppler.eq_ignore_ascii_case("TRUE") {
72            Ok(DopplerRef::BETA)
73        } else {
74            doppler.parse::<DopplerRef>()
75        };
76        let logs = (!doppler.is_empty() && parsed.is_err())
77            .then(|| LogLine::warning(format!("Illegal velocity type {doppler}. Using RADIO")))
78            .into_iter()
79            .collect::<Vec<_>>();
80        let history = format!(
81            "ia.moments(moments={moments:?}, axis={axis}, region={}, mask={mask:?}, method={methods:?}, smoothaxes={axes:?}, smoothtypes={kernels:?}, smoothwidths={widths:?}, includepix={include:?}, excludepix={exclude:?}, peaksnr={peaksnr}, stddev={stddev}, doppler={doppler:?}, outfile={outfile:?}, smoothout={smoothout:?}, overwrite={}, drop={}, stretch={})",
82            region.history(),
83            python_bool(overwrite),
84            python_bool(drop),
85            python_bool(stretch),
86        );
87        let output = py.detach(|| -> PyResult<_> {
88            let state = self.state();
89            let Some(input) = &state.image else {
90                return Ok(None);
91            };
92            moments_image(
93                input,
94                &moments,
95                axis,
96                region.selection(),
97                Some(&mask),
98                stretch,
99                drop,
100                &methods,
101                (&axes, &kernels, &widths),
102                &include,
103                &exclude,
104                peaksnr as f32,
105                stddev as f32,
106                parsed.unwrap_or(DopplerRef::RADIO),
107                outfile,
108                smoothout,
109                overwrite,
110                state.history.then_some(history),
111                external_open_check(),
112            )
113            .map(Some)
114            .map_err(raised)
115        })?;
116        if output.is_some() {
117            emit(py, "casars.tools.image", &logs)?;
118        }
119        Py::new(py, output.map_or_else(Self::new, Self::attached))
120    }
121}
122
123fn python_bool(value: bool) -> &'static str {
124    if value { "True" } else { "False" }
125}
