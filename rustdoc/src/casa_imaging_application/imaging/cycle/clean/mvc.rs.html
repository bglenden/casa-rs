6/// The cube completion and the Taylor PB and admission retained while the
7/// ordinary cleaning controller operates on the Taylor completion.
8pub(in crate::imaging::cycle) struct MvcState<'a> {
9    spec: MvcCycleSpec<'a>,
10    cube: Option<MajorCycleCompletion>,
11    mean_pb: Vec<f32>,
12    taylor_reservation: Reservation,
13}
14
15impl<'a> MvcState<'a> {
16    /// Reduce the first cube pass to the Taylor state used by the minor cycle.
17    pub(in crate::imaging::cycle) fn begin(
18        spec: MvcCycleSpec<'a>,
19        cube: MajorCycleCompletion,
20        run: &mut Run<'_>,
21    ) -> Result<(Self, MajorCycleCompletion), ImagingError> {
22        let (cube, model, seed_charge) = seed(spec, run, cube)?;
23        let taylor = crate::mvc::assemble_taylor_with_model(
24            crate::mvc::TaylorAssemblyInputs {
25                cube_problem: run.problem,
26                cube: &cube,
27                taylor_problem: spec.taylor_problem,
28                expansion_reference_hz: spec.expansion_reference_hz,
29                controls: spec.controls,
30                centres: spec.centres,
31            },
32            model,
33            (&run.host, &run.policy, &run.cancel),
34        )?;
35        drop(seed_charge);
36        Ok((
37            Self {
38                spec,
39                cube: Some(cube),
40                mean_pb: taylor.value.primary_beam,
41                taylor_reservation: taylor.reservation,
42            },
43            taylor.value.scientific,
44        ))
45    }
46
47    /// Apply a Taylor minor update, predict it through the cube, then reduce
48    /// the refreshed cube residual back around the same Taylor model.
49    pub(super) fn refresh(
50        &mut self,
51        run: &mut Run<'_>,
52        taylor: MajorCycleCompletion,
53        (terms, results): (Vec<ModelDeltaTerm>, Reservation),
54        last: bool,
55    ) -> Result<MajorCycleCompletion, ImagingError> {
56        let shape = taylor.normal_state().shape();
57        let cells = shape[0]
58            .checked_mul(shape[1])
59            .ok_or(ImagingError::Unsupported {
60                reason: "MVC Taylor model shape exceeds addressable size",
61            })?;
62        let lifecycle =
63            ModelLifecycle::new(self.spec.taylor_problem, ModelStoragePlan::resident(cells)?);
64        let (old_normal, model) = taylor.into_parts();
65        drop(old_normal);
66        let prepared = lifecycle.prepare_final_model(model, terms)?;
67        let cube = self.cube.as_ref().ok_or(ImagingError::Unsupported {
68            reason: "MVC cube state is absent",
69        })?;
70        let projected = crate::mvc::project_mvc_model(
71            crate::mvc::MvcModelProjectionInputs {
72                cube_problem: run.problem,
73                cube,
74                cube_lifecycle: &run.lifecycle,
75                taylor_problem: self.spec.taylor_problem,
76                taylor_model: prepared.generation(),
77                mean_pb: &self.mean_pb,
78                saved_pb: None,
79                expansion_reference_hz: self.spec.expansion_reference_hz,
80                controls: self.spec.controls,
81                centres: self.spec.centres,
82            },
83            (&run.host, &run.policy, &run.cancel),
84        )?;
85        let cube = self.cube.take().expect("the cube was borrowed above");
86        let major = run.refresh_with_projected_model(cube, projected, last)?;
87        let taylor = crate::mvc::assemble_taylor_with_model(
88            crate::mvc::TaylorAssemblyInputs {
89                cube_problem: run.problem,
90                cube: &major.completion,
91                taylor_problem: self.spec.taylor_problem,
92                expansion_reference_hz: self.spec.expansion_reference_hz,
93                controls: self.spec.controls,
94                centres: self.spec.centres,
95            },
96            Some(prepared),
97            (&run.host, &run.policy, &run.cancel),
98        )?;
99        // The minor result's charge covers its queued Taylor update until
100        // assembly has completed and released the prepared model.
101        drop(results);
102        self.cube = Some(major.completion);
103        self.mean_pb = taylor.value.primary_beam;
104        self.taylor_reservation = taylor.reservation;
105        Ok(taylor.value.scientific)
106    }
107
108    /// Return the cube as the major result and the Taylor family with its
109    /// own admission for staged publication.
110    pub(super) fn finish(
111        mut self,
112        taylor: MajorCycleCompletion,
113    ) -> (MajorCycleCompletion, Admitted<crate::mvc::TaylorCompletion>) {
114        let cube = self.cube.take().expect("an MVC cycle retains its cube");
115        (
116            cube,
117            Admitted {
118                value: crate::mvc::TaylorCompletion {
119                    scientific: taylor,
120                    primary_beam: self.mean_pb,
121                },
122                reservation: self.taylor_reservation,
123            },
124        )
125    }
126}
127
128/// Seed the first cube residual from the explicit Taylor images and
129/// previously published beams, before the minor cycle examines it.
130fn seed(
131    spec: MvcCycleSpec<'_>,
132    run: &mut Run<'_>,
133    cube: MajorCycleCompletion,
134) -> Result<
135    (
136        MajorCycleCompletion,
137        Option<casa_imaging_deconvolution::PreparedFinalModel>,
138        Reservation,
139    ),
140    ImagingError,
141> {
142    let Some(start) = spec.start else {
143        return Ok((cube, None, Reservation::none()));
144    };
145    let [nx, ny] = cube.normal_state().shape();
146    let lifecycle = ModelLifecycle::new(spec.taylor_problem, ModelStoragePlan::resident(nx * ny)?);
147    let held = run.admit("MVC start model", lifecycle.resident_bytes())?;
148    let model = {
149        let _reading = run.admit("MVC start model regrid", start.model.seeding_bytes())?;
150        let samples = start
151            .model
152            .main_samples()
153            .map_err(ImagingError::StartModel)?;
154        run.start_model_flux =
155            super::super::seeding::plane_fluxes(spec.taylor_problem, &start.model, &samples);
156        lifecycle.prepare_final_model(lifecycle.initial_from_start_model(&samples)?, [])?
157    };
158    let projected = {
159        // CASA reads the previous run's PB cube and the Taylor PB. Keep
160        // just the latter and one cube channel in the projection workspace.
161        let cells = (nx * ny) as u64;
162        let _reading = run.admit(
163            "saved MVC primary beams",
164            cells * (2 * size_of::<f32>()) as u64
165                + start.taylor_pb.reading_bytes()
166                + start.cube_pb.reading_bytes(),
167        )?;
168        let mean_pb = start
169            .taylor_pb
170            .open()
171            .map_err(ImagingError::Existing)?
172            .get()
173            .map_err(ImagingError::Existing)?
174            .into_iter()
175            .collect::<Vec<_>>();
176        let cube_pb = start.cube_pb.open().map_err(ImagingError::Existing)?;
177        crate::mvc::project_mvc_model(
178            crate::mvc::MvcModelProjectionInputs {
179                cube_problem: run.problem,
180                cube: &cube,
181                cube_lifecycle: &run.lifecycle,
182                taylor_problem: spec.taylor_problem,
183                taylor_model: model.generation(),
184                mean_pb: &mean_pb,
185                saved_pb: Some(&cube_pb),
186                expansion_reference_hz: spec.expansion_reference_hz,
187                controls: spec.controls,
188                centres: spec.centres,
189            },
190            (&run.host, &run.policy, &run.cancel),
191        )?
192    };
193    let seeded = run.refresh_with_projected_model(cube, projected, false)?;
194    Ok((seeded.completion, Some(model), held))
195}

197impl Run<'_> {
198    /// Refresh a cube using a separately admitted model projected from its
199    /// Taylor minor cycle, rather than applying cube-space sparse terms.
200    fn refresh_with_projected_model(
201        &mut self,
202        completion: MajorCycleCompletion,
203        projected: Admitted<ModelGeneration>,
204        last: bool,
205    ) -> Result<Major, ImagingError> {
206        let (normal, old_model) = completion.into_parts();
207        drop(old_model);
208        let model = self.lifecycle.prepare_final_model(projected.value, [])?;
209        let Admitted {
210            value: plan,
211            reservation: mut pass,
212        } = self.admit_pass(
213            ModeSet::DATA,
214            true,
215            self.write_bytes(last),
216            self.writes(last),
217        )?;
218        pass.join(projected.reservation);
219        let cycle = MajorCycle::refresh(
220            self.problem,
221            normal,
222            model,
223            self.normal_storage(plan.residency)?,
224        )?;
225        let major = self.reconcile(cycle, ModeSet::DATA, true, last, plan)?;
226        self.retain(&major, pass)?;
227        Ok(major)
228    }
