29impl AwCatalog {
30    /// Form CASA-compatible legacy L-band cells in an empty staging directory.
31    /// Assign the first new PA the supplied index so the caller can promote
32    /// the staged cells alongside an existing cache.
33    ///
34    /// # Errors
35    /// The request, PA index range, generation, or cell write is invalid.
36    pub fn generate_legacy_l_band_at(
37        root: &Path,
38        input: &NativeAwRequestInput,
39        band: &str,
40        spectral: &SpectralCoordinate,
41        legacy_vla_feed: bool,
42        workers: usize,
43        pa_index_offset: usize,
44    ) -> Result<(), AwCatalogError> {
45        input
46            .validate_legacy_vla()
47            .map_err(|error| cache_error(root, format!("invalid legacy A/W request: {error}")))?;
48        if pa_index_offset
49            .checked_add(input.pa_values.len())
50            .is_none_or(|end| end > usize::from(u16::MAX) + 1)
51        {
52            return Err(cache_error(
53                root,
54                "legacy A/W PA indices exceed the cache index range",
55            ));
56        }
57        let units = units(root, input, false, workers);
58        generate(
59            root,
60            input,
61            &units,
62            GenerationOptions {
63                only_missing: false,
64                legacy_coordinates: Some((band, spectral)),
65                legacy_vla_feed,
66                pa_index_offset,
67            },
68            workers,
69        )
70    }
71}

73/// One worker's share: the cells of one (parallactic angle, frequency,
74/// Mueller element), indices into `input`'s lists, at the w planes of a
75/// range, less those already present when only missing cells are made.
76#[derive(Clone, Debug, PartialEq)]
77pub(super) struct Unit {
78    pa: usize,
79    frequency: usize,
80    mueller: usize,
81    w_planes: Range<usize>,
82}

84/// Heap bytes [`units`] holds at most for `apertures` (parallactic angle,
85/// frequency, Mueller element) triples on `workers` workers: its count of
86/// cells per aperture and the units, of which there are fewer than
87/// `apertures + workers`.
88pub(super) const fn units_bytes(apertures: usize, workers: usize) -> usize {
89    apertures * size_of::<usize>() + (apertures + workers) * size_of::<Unit>()
90}

92/// Whether both images of the cell are in `root`.
93fn present(root: &Path, pa: usize, frequency: usize, w: usize, mueller: usize) -> bool {
94    let suffix = cell_suffix(pa, frequency, w, mueller);
95    root.join(format!("{IMAGING_PREFIX}{suffix}")).is_dir()
96        && root.join(format!("{WEIGHT_PREFIX}{suffix}")).is_dir()
97}

99/// The cells of `input` to generate into `root`, as units for `workers`:
100/// every cell, or with `only_missing` those without both images. Each
101/// (parallactic angle, frequency, Mueller element) is split into runs of w
102/// planes, in the serial generator's order, holding as near equal numbers
103/// of cells to generate as can be and none empty. Each aperture takes the
104/// same number of runs, or all its cells if fewer, the fewest that give
105/// every worker a unit while there are cells for them. A run prepares its
106/// aperture once.
107pub(super) fn units(
108    root: &Path,
109    input: &NativeAwRequestInput,
110    only_missing: bool,
111    workers: usize,
112) -> Vec<Unit> {
113    let planes = input.w_values.len();
114    let to_generate =
115        |pa, frequency, w, mueller| !(only_missing && present(root, pa, frequency, w, mueller));
116    let apertures = || {
117        (0..input.pa_values.len()).flat_map(move |pa| {
118            (0..input.frequencies.len()).flat_map(move |frequency| {
119                (0..input.mueller_elements.len()).map(move |mueller| (pa, frequency, mueller))
120            })
121        })
122    };
123    // Sized exactly, as `units_bytes` charges: a collect would grow it.
124    let mut counts = Vec::with_capacity(
125        input.pa_values.len() * input.frequencies.len() * input.mueller_elements.len(),
126    );
127    counts.extend(apertures().map(|(pa, frequency, mueller)| {
128        (0..planes)
129            .filter(|&w| to_generate(pa, frequency, w, mueller))
130            .count()
131    }));
132    let cells = counts.iter().sum::<usize>();
133    if cells == 0 {
134        return Vec::new();
135    }
136    let wanted = workers.clamp(1, cells);
137    let units_for = |runs: usize| counts.iter().map(|&count| count.min(runs)).sum::<usize>();
138    // At `wanted` runs there are at least `wanted` units, so this ends.
139    let runs = (1..=wanted)
140        .find(|&runs| units_for(runs) >= wanted)
141        .unwrap_or(wanted);
142    let mut units = Vec::with_capacity(units_for(runs));
143    for ((pa, frequency, mueller), &count) in apertures().zip(&counts) {
144        if count == 0 {
145            continue;
146        }
147        // Run j takes the cells to generate of rank j·count/runs up to
148        // (j + 1)·count/runs; the last also covers the planes after them.
149        let runs = count.min(runs);
150        let (mut start, mut rank, mut run) = (0, 0, 0);
151        for w in (0..planes).filter(|&w| to_generate(pa, frequency, w, mueller)) {
152            rank += 1;
153            if rank == (run + 1) * count / runs {
154                run += 1;
155                let end = if run == runs { planes } else { w + 1 };
156                units.push(Unit {
157                    pa,
158                    frequency,
159                    mueller,
160                    w_planes: start..end,
161                });
162                start = end;
163            }
164        }
165    }
166    units
167}

169/// Sets `failed` when a worker unwinds, so a panic stops the others before
170/// their next unit as an error does.
171struct StopOnPanic<'a>(&'a AtomicBool);
172
173impl Drop for StopOnPanic<'_> {
174    fn drop(&mut self) {
175        if std::thread::panicking() {
176            self.0.store(true, Ordering::Relaxed);
177        }
178    }
179}

181/// Generate `units` of `input` into `root` on up to `workers` threads, each
182/// taking the next unit until none is left, and skipping cells already
183/// present with `only_missing`. The first failure or panic stops the others
184/// before their next unit and is returned or resumed.
185pub(super) fn generate(
186    root: &Path,
187    input: &NativeAwRequestInput,
188    units: &[Unit],
189    options: GenerationOptions<'_>,
190    workers: usize,
191) -> Result<(), AwCatalogError> {
192    let model = EvlaApertureModel::new(input.surface.clone());
193    let next = AtomicUsize::new(0);
194    let failed = AtomicBool::new(false);
195    let work = || {
196        let _stop = StopOnPanic(&failed);
197        let mut workspace = None;
198        while !failed.load(Ordering::Relaxed) {
199            let Some(unit) = units.get(next.fetch_add(1, Ordering::Relaxed)) else {
200                break;
201            };
202            let generated = generate_unit(root, input, &model, unit, options, &mut workspace);
203            if let Err(error) = generated {
204                failed.store(true, Ordering::Relaxed);
205                return Err(error);
206            }
207        }
208        Ok(())
209    };
210    let workers = workers.clamp(1, units.len().max(1));
211    if workers == 1 {
212        return work();
213    }
214    let dispatch = tracing::dispatcher::get_default(Clone::clone);
215    std::thread::scope(|scope| {
216        let threads = (0..workers)
217            .map(|_| scope.spawn(|| tracing::dispatcher::with_default(&dispatch, work)))
218            .collect::<Vec<_>>();
219        threads
220            .into_iter()
221            .map(|thread| {
222                thread
223                    .join()
224                    .unwrap_or_else(|panic| std::panic::resume_unwind(panic))
225            })
226            .collect::<Vec<_>>()
227    })
228    .into_iter()
229    .collect()
230}

232/// Generate `unit`'s cells with `workspace`, made on its first cell, and
233/// write each pair of images; with `only_missing`, skip those present.
234fn generate_unit(
235    root: &Path,
236    input: &NativeAwRequestInput,
237    model: &EvlaApertureModel,
238    unit: &Unit,
239    GenerationOptions {
240        only_missing,
241        legacy_coordinates,
242        legacy_vla_feed,
243        pa_index_offset,
244    }: GenerationOptions<'_>,
245    workspace: &mut Option<EvlaAwWorkspace>,
246) -> Result<(), AwCatalogError> {
247    let grid = input.grid;
248    let pa = input.pa_values[unit.pa];
249    let frequency = input.frequencies[unit.frequency].cf_frequency_hz;
250    // `AWConvFunc::makeConvFunction2`: the conjugate beam uses the nearest
251    // listed frequency to `√(2f₀² − f²)`, the first on a tie
252    // (`SynthesisUtils::nearestValue`).
253    let conjugate_frequency = if input.terms.conjugate_beams {
254        let listed = input
255            .frequencies
256            .iter()
257            .map(|group| group.cf_frequency_hz)
258            .collect::<Vec<_>>();
259        nearest(
260            &listed,
261            conjugate_frequency(frequency, input.reference_frequency_hz),
262        )
263        .1
264    } else {
265        frequency
266    };
267    let mueller = input.mueller_elements[unit.mueller];
268    let conjugate_mueller = if legacy_coordinates.is_some() {
269        // The legacy writer stores the first Stokes code from the conjugate
270        // polarization axis in every imaging and weight cell.
271        (if input.mueller_elements[0] == 0 {
272            StokesType::LL
273        } else {
274            StokesType::RR
275        }) as u32
276    } else {
277        MUELLER_ELEMENTS - 1 - mueller as u32
278    };
279    for iw in unit.w_planes.clone() {
280        // No other worker writes this unit's cells, so what is present now
281        // is what was present when the units were made.
282        if only_missing && present(root, unit.pa, unit.frequency, iw, unit.mueller) {
283            continue;
284        }
285        let w = input.w_values[iw];
286        let request = EvlaAwCellRequest {
287            size: grid.size,
288            sky_increment_rad: grid.sky_increment_rad,
289            frequency_hz: frequency,
290            conjugate_frequency_hz: conjugate_frequency,
291            w_wavelengths: w,
292            parallactic_angle_rad: pa,
293            mueller,
294            oversampling: grid.oversampling,
295            prolate_spheroidal: input.terms.prolate_spheroidal,
296            aperture: input.terms.aperture,
297        };
298        let workspace = match workspace {
299            Some(workspace) => workspace,
300            None => workspace.insert(
301                EvlaAwWorkspace::new(model.clone(), request)?
302                    .with_casa_cell_layout(legacy_coordinates.is_some(), legacy_vla_feed),
303            ),
304        };
305        // Only the W screen depends on w: the aperture and the weight cell
306        // are formed once per unit.
307        workspace.prepare(request)?;
308        let pair = workspace.realize(request)?;
309        let misc = CellMisc {
310            sampling: grid.oversampling as f64,
311            pa_deg: pa.to_degrees(),
312            mueller,
313            w_value: w,
314            w_increment: input.w_increment,
315            conjugate_frequency,
316            conjugate_mueller,
317            diameter_m: input.antenna_diameter_m,
318            sky_increment_rad: grid.sky_increment_rad[1].abs(),
319            working_size: grid.size,
320            telescope: if legacy_vla_feed { "VLA" } else { "EVLA" },
321            legacy_coordinates,
322        };
323        let suffix = cell_suffix(unit.pa + pa_index_offset, unit.frequency, iw, unit.mueller);
324        write_cell(
325            &root.join(format!("{IMAGING_PREFIX}{suffix}")),
326            &pair.imaging,
327            frequency,
328            grid,
329            &misc,
330        )?;
331        write_cell(
332            &root.join(format!("{WEIGHT_PREFIX}{suffix}")),
333            &pair.weight,
334            frequency,
335            grid,
336            &misc,
337        )?;
338    }
339    Ok(())
340}

367                    channel_frequencies_hz: vec![1.4e9 + window as f64 * 1.0e8],
368                    cf_frequency_hz: 1.4e9 + window as f64 * 1.0e8,
369                })
370                .collect(),
371            w_values: (0..w_planes).map(|w| w as f64).collect(),
372            w_increment: 1.0,

374            mueller_elements: vec![0, 15],
375            reference_frequency_hz: 1.5e9,
376            grid: NativeAwGrid {

379                oversampling: 4,

382                aperture: true,
383                w_term: true,
384                prolate_spheroidal: false,
385                wideband: true,
386                conjugate_beams: false,
387            },

389        }

392    /// The cells `units` generates, in order: the planes of each unit's
393    /// range, less those present in `root` with `only_missing`.
394    fn cells(
395        root: &Path,
396        units: &[Unit],
397        only_missing: bool,
398    ) -> Vec<Vec<(usize, usize, usize, usize)>> {
399        units
400            .iter()
401            .map(|unit| {
402                unit.w_planes
403                    .clone()
404                    .filter(|&w| {
405                        !(only_missing && present(root, unit.pa, unit.frequency, w, unit.mueller))
406                    })
407                    .map(|w| (unit.pa, unit.frequency, unit.mueller, w))
408                    .collect()
409            })
410            .collect()
411    }

413    /// Every unit has a cell; there is a unit for each worker while there
414    /// are cells for them; the units' cells are `expected`, once each and
415    /// in order; and the runs of one aperture differ by at most one cell.
416    fn assert_units(
417        root: &Path,
418        input: &NativeAwRequestInput,
419        only_missing: bool,
420        expected: &[(usize, usize, usize, usize)],
421    ) {
422        for workers in [1, 2, 3, 4, 5, 8, 15, 16, 17, 20, 21, 64] {
423            let units = units(root, input, only_missing, workers);
424            let cells = cells(root, &units, only_missing);
425            assert!(
426                cells.iter().all(|cells| !cells.is_empty()),
427                "{workers} workers"
428            );
429            assert!(
430                units.len() >= workers.min(expected.len()),
431                "{workers} workers, {} units",
432                units.len()
433            );
434            assert_eq!(cells.concat(), expected, "{workers} workers");
435            let aperture = |cells: &Vec<(usize, usize, usize, usize)>| {
436                let (pa, frequency, mueller, _) = cells[0];
437                (pa, frequency, mueller)
438            };
439            for aperture in cells.chunk_by(|a, b| aperture(a) == aperture(b)) {
440                let sizes = aperture.iter().map(Vec::len);
441                let spread = sizes.clone().max().unwrap() - sizes.min().unwrap();
442                assert!(spread <= 1, "{workers} workers");
443            }
444        }
445    }

447    /// One worker takes every cell of an aperture in one unit, in the serial
448    /// generator's order; more workers split each aperture's w planes into
449    /// balanced runs until each worker has a unit, and every cell is in one
450    /// run. A fill makes units of the missing cells alone.
451    #[test]
452    fn units_cover_every_cell_once_in_serial_order() {
453        let root = tempfile::tempdir().expect("cache directory");
454        let input = input(2, 5);
455        let apertures = 2 * input.mueller_elements.len();
456        let serial = units(root.path(), &input, false, 1);
457        assert_eq!(serial.len(), apertures);
458        assert!(serial.iter().all(|unit| unit.w_planes == (0..5)));
459        let every = cells(root.path(), &serial, false).concat();
460        assert_units(root.path(), &input, false, &every);
461
462        // Present: every plane of the first aperture, the first three of the
463        // second, the last of the third and none of the fourth.
464        let held = |&(pa, frequency, mueller, w): &(usize, usize, usize, usize)| match (frequency, mueller) {
465            (0, 0) => true,
466            (0, 1) => w < 3,
467            (1, 0) => w == 4,
468            _ => false,
469        } && pa == 0;
470        for &(pa, frequency, mueller, w) in every.iter().filter(|cell| held(cell)) {
471            let suffix = cell_suffix(pa, frequency, w, mueller);
472            for prefix in [IMAGING_PREFIX, WEIGHT_PREFIX] {
473                std::fs::create_dir(root.path().join(format!("{prefix}{suffix}"))).expect("cell");
474            }
475        }
476        let missing = every
477            .iter()
478            .copied()
479            .filter(|cell| !held(cell))
480            .collect::<Vec<_>>();
481        assert_eq!(missing.len(), 11);
482        assert_units(root.path(), &input, true, &missing);
483    }
