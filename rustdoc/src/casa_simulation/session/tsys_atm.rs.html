2//! `sm.setnoise(mode='tsys-atm')`: the atmosphere the noise's Tsys comes
3//! from, built at `setnoise` (#841 Phase 4c). The model and where it
4//! departs from CASA are the `corrupt::tsys_atm` module's.
5//!
6//! As CASA's `AtmosCorruptor::initAtm`, `setnoise` builds the atmosphere
7//! for every spectral window defined then; a scan in a window defined later
8//! is refused at `corrupt`. Each window's channels are independent and are
9//! computed on the session's workers. What the windows keep (each
10//! channel's line of sight) is admitted, with what building the largest
11//! window holds, before anything is built, and stays charged while the
12//! term does.
13
14use casa_atmosphere::{AbsorptionProfile, AtmosphereProfile, ChannelTotals};
15use casa_imaging_runtime::pass::WorkerTeam;
16use casa_imaging_runtime::{Demand, admit};

18use super::Simulator;
19use super::corruption::{LARGEST_ZENITH_TSYS_K, NoiseSpec};
20use super::error::{SimulatorError, SimulatorResult};
21use crate::corrupt::TsysAtm;
22use crate::corrupt::tsys_atm::{self, WindowAtmosphere};

24/// The phase the atmosphere of a `tsys-atm` term is admitted as.
25const ATMOSPHERE_PHASE: &str = "the atmosphere of a tsys-atm noise term";
26
27/// Channels a worker computes at a time.
28const CHANNELS_PER_JOB: usize = 64;
29
30impl Simulator {
31    /// The `tsys-atm` model of `spec`, its atmosphere built for every
32    /// spectral window, checked as CASA checks it. `pwv_mm` is the water
33    /// column after `setnoise`'s default.
34    pub(super) fn tsys_atm(&self, spec: &NoiseSpec, pwv_mm: f64) -> SimulatorResult<TsysAtm> {
35        let parameters = spec.atmosphere.resolved(spec.tsys.ground_k);
36        log_defaults(spec);
37        tracing::info!("Initializing ATM");
38        tracing::info!(
39            "altitude={}m, Pground={}mb, Tground={}K, humidity= {}%, water scale height={}m",
40            parameters.altitude_m,
41            parameters.ground_pressure_hpa,
42            parameters.ground_temperature_k,
43            parameters.relative_humidity_percent,
44            parameters.water_vapour_scale_height_km * 1000.0
45        );
46        let atmosphere = AtmosphereProfile::new(&parameters).map_err(|error| {
47            SimulatorError::refused(format!("Caught exception: error in ATM setup - {error}"))
48        })?;
49        let layers = atmosphere.layers().len();
50        // The windows' vector, and each window's channels (#897 review).
51        let kept: u64 = (self.spectral_windows.len() * size_of::<Option<WindowAtmosphere>>())
52            as u64
53            + self
54                .spectral_windows
55                .iter()
56                .map(|window| WindowAtmosphere::bytes(window.channel_count, layers))
57                .sum::<u64>();
58        let building = self
59            .spectral_windows
60            .iter()
61            .map(|window| building_bytes(window.channel_count, layers))
62            .max()
63            .unwrap_or(0);
64        let resources = &self.options.resources;
65        let mut reservation = admit(
66            resources.host(),
67            resources.policy(),
68            &Demand {
69                phase: ATMOSPHERE_PHASE,
70                memory: kept + building,
71            },
72        )
73        .map_err(SimulatorError::admission)?;
74        let team = WorkerTeam::new(resources.cores().workers.max(1))
75            .map_err(|error| SimulatorError::raised(format!("the worker team: {error}")))?;
76        // In single precision, as the corruptor holds them.
77        let pwv_m = tsys_atm::single(pwv_mm) * 1e-3;
78        let coupling = (
79            tsys_atm::single(spec.tsys.spillover_efficiency),
80            tsys_atm::single(spec.tsys.ground_k),
81        );
82        let mut windows = Vec::with_capacity(self.spectral_windows.len());
83        for spectral_window in &self.spectral_windows {
84            let channels = spectral_window.channel_frequencies_hz();
85            let grid = tsys_atm::window_grid(&channels)
86                .map_err(|error| SimulatorError::refused(format!("Caught exception: {error}")))?;
87            let totals = window_totals(&team, &atmosphere, &channels)?;
88            let profile = AbsorptionProfile::from_channels(atmosphere.clone(), grid, totals)
89                .map_err(|error| SimulatorError::raised(format!("the atmosphere: {error}")))?;
90            let window = WindowAtmosphere::new(&profile, pwv_m, coupling).map_err(|error| {
91                SimulatorError::refused(format!("Caught exception: error in ATM setup - {error}"))
92            })?;
93            windows.push(Some(window));
94        }
95        reservation.retain(kept);
96        let model = TsysAtm::new(&spec.tsys, windows, reservation);
97        check_zenith_tsys(&model, spec, pwv_mm)?;
98        Ok(model)
99    }
100}

102/// CASA's check of the zenith Tsys at the centre of the first window,
103/// logged as CASA logs it. A Tsys that is not a number is refused too.
104fn check_zenith_tsys(model: &TsysAtm, spec: &NoiseSpec, pwv_mm: f64) -> SimulatorResult<()> {
105    if !model.covers(0) {
106        return Err(SimulatorError::raised(
107            "tsys-atm noise needs a spectral window",
108        ));
109    }
110    let window = model.window(0);
111    let centre = window.channels() / 2;
112    let zenith = model.tsys(window, centre, 1.0);
113    tracing::info!(
114        "Zenith Tsys at center of first Spectral Window = {zenith} tground={} spillover={}",
115        spec.tsys.ground_k as f32,
116        spec.tsys.spillover_efficiency as f32
117    );
118    tracing::info!(
119        " pwv={} tau={}(ch{centre})",
120        pwv_mm,
121        window.zenith_opacity(centre)
122    );
123    if zenith > LARGEST_ZENITH_TSYS_K || zenith <= 0.0 || zenith.is_nan() {
124        return Err(SimulatorError::refused(format!(
125            "Caught exception: error in ATM setup - Tsys {zenith} poorly defined - check \
126                 inputs"
127        )));
128    }
129    Ok(())
130}

132/// What building a window of `channels` channels holds beside what it
133/// keeps: its channel frequencies, the profile of its channels' totals
134/// (grid and the atmosphere's copy included), and the jobs, with their
135/// vectors and the one they are gathered into.
136fn building_bytes(channels: usize, layers: usize) -> u64 {
137    AbsorptionProfile::<ChannelTotals>::shape_bytes(1, channels, layers)
138        + (channels * (size_of::<f64>() + 2 * size_of::<ChannelTotals>())) as u64
139        + (channels.div_ceil(CHANNELS_PER_JOB) * size_of::<(&[f64], Vec<ChannelTotals>)>()) as u64
140}
141
142/// Each channel's totals of `atmosphere` at `channels`, computed on `team`
143/// in jobs of [`CHANNELS_PER_JOB`] channels.
144fn window_totals(
145    team: &WorkerTeam,
146    atmosphere: &AtmosphereProfile,
147    channels: &[f64],
148) -> SimulatorResult<Vec<ChannelTotals>> {
149    let mut jobs: Vec<(&[f64], Vec<ChannelTotals>)> = channels
150        .chunks(CHANNELS_PER_JOB)
151        .map(|chunk| (chunk, Vec::new()))
152        .collect();
153    team.for_each_mut(&mut jobs, |_, (chunk, totals)| {
154        *totals = tsys_atm::channel_totals(atmosphere, chunk);
155        Ok::<_, SimulatorError>(())
156    })?;
157    let mut totals = Vec::with_capacity(channels.len());
158    for (_, job) in jobs {
159        totals.extend(job);
160    }
161    Ok(totals)
162}

164/// CASA's log lines for the site values `setnoise` fills in.
165fn log_defaults(spec: &NoiseSpec) {
166    let site = &spec.atmosphere;
167    if site.ground_pressure_hpa <= 100.0 {
168        tracing::info!("User has not set ground pressure, using 560mb");
169    }
170    if site.relative_humidity_percent <= 0.0 {
171        tracing::info!("User has not set ground relative humidity, using 20%");
172    }
173    if site.altitude_m <= 0.0 {
174        tracing::info!("User has not set site altitude, using 5000m");
175    }
176    if site.water_scale_height_m <= 100.0 {
177        tracing::info!("User has not set water scale height, using 2km");
178    }
179}
