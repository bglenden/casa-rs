27impl Plan {
28    pub(super) fn file(&self, suffix: &str) -> PathBuf {
29        suffixed(&self.root, suffix)
30    }
31    pub(super) fn analysis_bytes(&self, full_analysis: bool) -> Result<u64, SimobserveError> {
32        if !full_analysis {
33            return super::memory::flat_images(
34                super::memory::ImageGrid::read(&self.model)?,
35                super::memory::ImageGrid::read(&self.file(".image"))?,
36            );
37        }
38        let cells = |path: &Path| -> Result<u64, SimobserveError> {
39            let image = PagedImage::<f32>::open(path).map_err(failure)?;
40            image.shape().iter().try_fold(1u64, |count, &axis| {
41                count
42                    .checked_mul(axis as u64)
43                    .ok_or_else(|| failure("image size overflows"))
44            })
45        };
46        analysis_memory(cells(&self.model)?, cells(&self.file(".image"))?)
47    }
48    pub(super) fn products(&self) -> Result<Vec<PathBuf>, SimobserveError> {
49        let mut paths: Vec<_> = [
50            ".psf",
51            ".model",
52            ".residual",
53            ".image",
54            ".image.pbcor",
55            ".pb",
56            ".sumwt",
57            ".weight",
58            ".mask",
59            ".image.flat",
60            ".diff",
61            ".fidelity",
62        ]
63        .iter()
64        .map(|suffix| self.file(suffix))
65        .filter(|path| path.exists())
66        .collect();
67        let flat = suffixed(&self.model, ".flat");
68        for path in [
69            flat.clone(),
70            suffixed(&flat, ".regrid"),
71            suffixed(&flat, ".regrid.conv"),
72        ] {
73            if path.exists() && !paths.contains(&path) {
74                paths.push(path);
75            }
76        }
77        paths.sort();
78        Ok(paths)
79    }
80}

82// Peak bound for flat-image/regrid/convolution and fidelity statistics.
83// Source: float input, flattened products, masks and interpolation work.
84// Output: retained regrid, convolution's complex FFT work, masks, difference,
85// fidelity and robust-statistic Float64 work. These phases run sequentially;
86// charging their sum is conservative. Table/cache headroom is fixed.
87fn analysis_memory(source: u64, output: u64) -> Result<u64, SimobserveError> {
88    source
89        .checked_mul(48)
90        .and_then(|bytes| {
91            output
92                .checked_mul(96)
93                .and_then(|out| bytes.checked_add(out))
94        })
95        .and_then(|bytes| bytes.checked_add(128 << 20))
96        .ok_or_else(|| failure("analysis memory overflows"))
97}

99pub(super) fn suffixed(path: &Path, suffix: &str) -> PathBuf {
100    let mut value = path.as_os_str().to_os_string();
101    value.push(suffix);
102    PathBuf::from(value)
103}
104fn files(directory: &Path) -> Result<Vec<PathBuf>, SimobserveError> {
105    let mut files = std::fs::read_dir(directory)
106        .map_err(failure)?
107        .map(|e| e.map(|e| e.path()))
108        .collect::<Result<Vec<_>, _>>()
109        .map_err(failure)?;
110    files.sort();
111    Ok(files)
112}
113pub(super) fn expand_project(directory: &Path, value: &str) -> String {
114    value.replace(
115        "$project",
116        &directory.file_name().unwrap_or_default().to_string_lossy(),
117    )
118}
119pub(super) fn locate(directory: &Path, input: &str) -> PathBuf {
120    let in_project = directory.join(input);
121    if in_project.exists() {
122        in_project
123    } else {
124        PathBuf::from(input)
125    }
126}

128pub(super) fn prepare(
129    p: &Parameters<'_>,
130    resources: SessionResources,
131) -> Result<Plan, SimobserveError> {
132    let directory = std::path::absolute(p.text("project")).map_err(failure)?;
133    if !directory.is_dir() {
134        return Err(failure(format!(
135            "{} directory doesn't exist - the task expects to find results from creating the datasets there, like the skymodel.",
136            directory.display()
137        )));
138    }
139    let candidates = files(&directory)?;
140    let inputs = inputs::discover(p, &directory, &candidates)?;
141    let concat_weights = if inputs.synthesis.len() > 1 {
142        inputs::concat_weights(&inputs.synthesis)?
143    } else {
144        Vec::new()
145    };
146    let project = directory.file_name().unwrap().to_string_lossy();
147    let ms = if inputs.synthesis.len() > 1 {
148        directory.join(format!("{project}.concat.ms"))
149    } else {
150        inputs
151            .synthesis
152            .first()
153            .or(inputs.total_power.as_ref())
154            .cloned()
155            .unwrap_or_default()
156    };
157    let stem = if p.flag("image") {
158        if inputs.synthesis.is_empty() {
159            format!("{project}.sd")
160        } else {
161            ms.file_name()
162                .unwrap()
163                .to_string_lossy()
164                .trim_end_matches(".ms")
165                .to_owned()
166        }
167    } else if matches!(p.text("imagename"), "" | "default") {
168        candidates
169            .iter()
170            .find(|p| p.to_string_lossy().ends_with(".image"))
171            .and_then(|p| p.file_name())
172            .ok_or_else(|| failure("can't find any image in project directory"))?
173            .to_string_lossy()
174            .trim_end_matches(".image")
175            .to_owned()
176    } else {
177        Path::new(p.text("imagename"))
178            .file_name()
179            .unwrap_or_default()
180            .to_string_lossy()
181            .trim_end_matches(".image")
182            .to_owned()
183    };
184    let root = directory.join(&stem);
185    let mut models: Vec<_> = candidates
186        .iter()
187        .filter(|path| {
188            [".skymodel", ".newmodel", ".compskymodel"]
189                .iter()
190                .any(|suffix| path.to_string_lossy().ends_with(suffix))
191        })
192        .cloned()
193        .collect();
194    let search = inputs
195        .synthesis
196        .first()
197        .or(inputs.total_power.as_ref())
198        .and_then(|path| path.file_name())
199        .map(|name| name.to_string_lossy().trim_end_matches(".ms").to_owned())
200        .unwrap_or_else(|| stem.clone());
201    models.sort_by_key(|path| {
202        (
203            !path
204                .file_name()
205                .unwrap()
206                .to_string_lossy()
207                .contains(&search),
208            path.clone(),
209        )
210    });
211    let model = if !p.text("skymodel").is_empty() {
212        locate(&directory, &expand_project(&directory, p.text("skymodel")))
213    } else {
214        models
215            .into_iter()
216            .next()
217            .ok_or_else(|| failure("Can't find a model image in your project directory"))?
218    };
219    let imaging = if p.flag("image") {
220        imaging(p, &inputs, &ms, &root, &model, &directory, resources)?
221    } else {
222        Map::new()
223    };
224    Ok(Plan {
225        directory,
226        stem,
227        root,
228        model,
229        ms,
230        imaging,
231        inputs,
232        concat_weights,
233    })
234}

236fn imaging(
237    p: &Parameters<'_>,
238    inputs: &inputs::Inputs,
239    ms_path: &Path,
240    root: &Path,
241    model_path: &Path,
242    directory: &Path,
243    resources: SessionResources,
244) -> Result<Map<String, Value>, SimobserveError> {
245    let model = PagedImage::<f32>::open(model_path).map_err(failure)?;
246    let model_cell = analysis::cell_size(model.coordinates()).map_err(failure)?;
247    let CoordinateModel::Direction(direction) = model.coordinates().coordinate(0) else {
248        return Err(failure("model has no direction axes"));
249    };
250    let world = direction
251        .to_world(&[model.shape()[0] as f64 / 2., model.shape()[1] as f64 / 2.])
252        .map_err(failure)?;
253    let refdir = format!(
254        "{} {} {}",
255        direction.direction_ref().as_str(),
256        qa::hms(&Quantity::new(world[0], "rad").map_err(failure)?).map_err(failure)?,
257        qa::dms(&Quantity::new(world[1], "rad").map_err(failure)?).map_err(failure)?
258    );
259    let provided = p.strings("cell");
260    let cell = if provided.is_empty() || provided[0].is_empty() {
261        model_cell
262    } else {
263        [0, 1]
264            .map(|a| qa::value_in(&qa::read(&provided[a.min(provided.len() - 1)])?, "rad"))
265            .into_iter()
266            .collect::<Result<Vec<_>, _>>()
267            .map_err(failure)?
268            .try_into()
269            .expect("two cells")
270    };
271    let cell = positive_cells(cell)?;
272    let mut size: Vec<_> = match &p.0["imsize"] {
273        Value::Array(items) => items.iter().map(|v| v.as_i64().expect("integer")).collect(),
274        single => vec![single.as_i64().expect("integer")],
275    };
276    if size.is_empty() {
277        size.push(0);
278    }
279    if size.len() == 1 {
280        size.push(size[0]);
281    }
282    let frequency = model.coordinates().coordinate(2).reference_value()[0];
283    let minimum = inputs
284        .synthesis
285        .iter()
286        .map(|path| minimum_size(path, cell[0], frequency, resources))
287        .try_fold(None::<f64>, |value, next| {
288            next.map(|next| Some(value.map_or(next, |old| old.min(next))))
289        })?
290        .unwrap_or(0.);
291    let size = planned_sizes(&size, model.shape(), model_cell, cell, minimum, resources)?;
292    let size = if inputs.synthesis.is_empty() {
293        size
294    } else {
295        size.map(casa_imaging_model::optimum_image_size)
296    };
297    let input = inputs
298        .synthesis
299        .first()
300        .or(inputs.total_power.as_ref())
301        .expect("image input");
302    let ms = MeasurementSet::open(input).map_err(failure)?;
303    let source = ms
304        .subtable(SubtableId::Source)
305        .ok_or_else(|| failure("MS has no SOURCE table"))?;
306    let codes = source.column_accessor("CODE").map_err(failure)?;
307    let fields = (0..source.row_count())
308        .filter(|&row| matches!(codes.scalar_cell(row), Ok(ScalarValue::String(v)) if v=="OBJ"))
309        .count();
310    let startmodel = expand_project(directory, p.text("modelimage"));
311    let startmodel = locate(directory, &startmodel);
312    let nchan = ms
313        .spectral_window()
314        .map_err(failure)?
315        .num_chan(0)
316        .map_err(failure)?;
317    Ok(json!({ "vis": ms_path, "selectdata": false, "imagename": root,
318        "cell": cell.map(|v|format!("{v}rad")), "imsize": size,
319        "phasecenter": if p.text("imdirection").is_empty() {refdir.as_str()} else {p.text("imdirection")},
320        "stokes": p.text("stokes"), "startmodel": if startmodel.exists() && !p.text("modelimage").is_empty() {startmodel.to_string_lossy().into_owned()} else {String::new()},
321        "specmode": if nchan==1 {"mfs"} else {"cube"}, "gridder": if fields==1 {"standard"} else {"mosaic"},
322        "deconvolver": "clark", "weighting": p.text("weighting"), "niter": p.0["niter"],
323        "threshold": if p.0["niter"].as_i64().unwrap_or(0)>0 {p.text("threshold")} else {"0Jy"},
324        "interactive": p.flag("interactive"), "usemask": "user", "mask": p.0["mask"],
325        "uvtaper": p.0["outertaper"], "pbcor": p.flag("pbcor"), "pbmask": 0.,
326        "restoringbeam": "common", "savemodel": "none", "calcres": true, "calcpsf": true,
327        "restoration": true, "restart": true, "fastnoise": true, "parallel": false
328    }).as_object().unwrap().clone())
329}

331fn minimum_size(
332    path: &Path,
333    cell: f64,
334    frequency: f64,
335    resources: SessionResources,
336) -> Result<f64, SimobserveError> {
337    casa_imaging_runtime::charge_process_caches(resources.host(), resources.policy())
338        .map_err(failure)?;
339    let ms = MeasurementSet::open(path).map_err(failure)?;
340    let uvw = UvwColumn::new(ms.main_table());
341    let mut maximum = 0f64;
342    for row in 0..ms.main_table().row_count() {
343        let [u, v, _] = uvw.get(row).map_err(failure)?;
344        maximum = maximum.max(u).max(v);
345    }
346    if maximum <= 0. {
347        return Err(failure("an interferometric image needs positive u or v"));
348    }
349    Ok(8. * (300_000_000. / frequency / maximum / cell).floor())
350}

352fn positive_cells(cell: [f64; 2]) -> Result<[f64; 2], SimobserveError> {
353    if cell.iter().any(|value| !value.is_finite() || *value == 0.) {
354        return Err(failure("cell sizes must be finite and nonzero"));
355    }
356    Ok(cell.map(f64::abs))
357}

359fn planned_sizes(
360    requested: &[i64],
361    shape: &[usize],
362    model_cell: [f64; 2],
363    cell: [f64; 2],
364    minimum: f64,
365    resources: SessionResources,
366) -> Result<[usize; 2], SimobserveError> {
367    if !minimum.is_finite() {
368        return Err(failure("cell implies an unrepresentable image size"));
369    }
370    let mut size = [0usize; 2];
371    for axis in 0..2 {
372        let extent = if requested[axis] <= 0 {
373            (shape[axis] as f64 * model_cell[axis].abs() / cell[axis]).ceil()
374        } else {
375            requested[axis] as f64
376        };
377        let extent = extent.max(minimum);
378        // CASA's image extents are Int32, and a float image must fit within
379        // the run's memory even before its FFT/statistic working buffers.
380        if !extent.is_finite() || !(0. ..=f64::from(i32::MAX)).contains(&extent) {
381            return Err(failure("cell/imsize implies an unrepresentable image size"));
382        }
383        size[axis] = extent as usize;
384    }
385    let bytes = (size[0] as u64)
386        .checked_mul(size[1] as u64)
387        .and_then(|cells| cells.checked_mul(4));
388    if bytes.is_none_or(|bytes| bytes > resources.policy().memory(resources.host())) {
389        return Err(failure("image dimensions exceed the run's memory budget"));
390    }
391    Ok(size)
392}

394#[cfg(test)]
395mod tests {
396    use super::{analysis_memory, planned_sizes, positive_cells};
397
398    #[test]
399    fn analysis_counting_law_scales_with_both_image_grids_and_refuses_overflow() {
400        let base = analysis_memory(0, 0).unwrap();
401        for cells in [1, 65_536, 16_777_216] {
402            assert_eq!(analysis_memory(cells, 0).unwrap() - base, cells * 48);
403            assert_eq!(analysis_memory(0, cells).unwrap() - base, cells * 96);
404            assert_eq!(analysis_memory(cells, cells).unwrap() - base, cells * 144);
405        }
406        assert!(analysis_memory(u64::MAX, 1).is_err());
407        assert!(analysis_memory(1, u64::MAX).is_err());
408    }
409
410    #[test]
411    fn invalid_cells_and_unrepresentable_dimensions_are_refused_before_fft_sizing() {
412        for value in [0., f64::NAN, f64::INFINITY, f64::NEG_INFINITY] {
413            assert!(positive_cells([value, 1.]).is_err());
414        }
415        assert_eq!(positive_cells([-2e-6, 2e-6]).unwrap(), [2e-6, 2e-6]);
416        let resources =
417            crate::SessionResources::from_controls(&crate::ExecutionControls::default()).unwrap();
418        assert!(
419            planned_sizes(
420                &[i64::MAX, 64],
421                &[64, 64],
422                [2e-6; 2],
423                [2e-6; 2],
424                0.,
425                resources
426            )
427            .is_err()
428        );
429        assert!(planned_sizes(&[0, 0], &[64, 64], [2e-6; 2], [1e-300; 2], 0., resources).is_err());
430    }
