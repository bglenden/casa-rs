2//! Shared column-keyword metadata helpers for MeasurementSet compatibility.
3
4use casa_types::{RecordField, RecordValue, ScalarValue, Value};
5
6use crate::column_def::ColumnDef;
7
8/// Whether `def` is one of casacore's Muvw columns, `UVW` and `UVW2`
9/// (`MeasurementSet.cc`'s `colMapDef` measure "uvw"), which `MeasureType`
10/// does not model. Like a position, a uvw has three components in metres.
11fn is_uvw(def: &ColumnDef) -> bool {
12    matches!(def.name, "UVW" | "UVW2")
13}
14
15pub(crate) fn quantum_units_for(def: &ColumnDef) -> Option<Vec<String>> {
16    if is_uvw(def) && !def.unit.is_empty() {
17        return Some(vec![
18            def.unit.to_string(),
19            def.unit.to_string(),
20            def.unit.to_string(),
21        ]);
22    }
23    if def.unit.is_empty() {
24        return None;
25    }
26    match def.measure_type {
27        Some(casa_tables::table_measures::MeasureType::Direction) => {
28            Some(vec![def.unit.to_string(), def.unit.to_string()])
29        }
30        Some(casa_tables::table_measures::MeasureType::Position) => Some(vec![
31            def.unit.to_string(),
32            def.unit.to_string(),
33            def.unit.to_string(),
34        ]),
35        _ => Some(vec![def.unit.to_string()]),
36    }
37}
38
39pub(crate) fn measure_type_name_for(def: &ColumnDef) -> Option<String> {
40    if is_uvw(def) {
41        Some("uvw".to_string())
42    } else {
43        def.measure_type
44            .map(|measure_type| measure_type.as_str().to_string())
45    }
46}
47
48pub(crate) fn measinfo_for(def: &ColumnDef) -> Option<RecordValue> {
49    let measure_type = measure_type_name_for(def)?;
50
51    let mut fields = vec![RecordField::new(
52        "type",
53        Value::Scalar(ScalarValue::String(measure_type)),
54    )];
55    if !def.measure_ref.is_empty() {
56        fields.push(RecordField::new(
57            "Ref",
58            Value::Scalar(ScalarValue::String(def.measure_ref.to_string())),
59        ));
60    }
61    Some(RecordValue::new(fields))
62}
