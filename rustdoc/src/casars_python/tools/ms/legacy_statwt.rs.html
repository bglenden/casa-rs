18impl PyMsTool {
19    /// Estimate raw or centered visibility scatter per integration and baseline.
20    #[allow(clippy::too_many_arguments)]
21    #[pyo3(signature = (dorms = false, byantenna = true, sepacs = true, fitspw = None, fitcorr = None, combine = String::new(), timebin = None, minsamp = 3, field = None, spw = None, antenna = None, timerange = String::new(), scan = None, intent = String::new(), array = None, correlation = String::new(), obs = String::new(), datacolumn = "corrected_data".to_owned()), text_signature = "($self, dorms=False, byantenna=True, sepacs=True, fitspw=[], fitcorr=[], combine='', timebin=[], minsamp=3, field=[], spw=[], antenna='', timerange='', scan=[], intent='', array=[], correlation='', obs='', datacolumn='corrected_data')")]
22    fn oldstatwt(
23        &self,
24        py: Python<'_>,
25        #[pyo3(from_py_with = strict_bool)] dorms: bool,
26        #[pyo3(from_py_with = strict_bool)] byantenna: bool,
27        #[pyo3(from_py_with = strict_bool)] sepacs: bool,
28        fitspw: Option<&Bound<'_, PyAny>>,
29        fitcorr: Option<Variant>,
30        #[pyo3(from_py_with = combine_argument)] combine: String,
31        timebin: Option<Variant>,
32        minsamp: i64,
33        field: Option<&Bound<'_, PyAny>>,
34        spw: Option<&Bound<'_, PyAny>>,
35        antenna: Option<&Bound<'_, PyAny>>,
36        #[pyo3(from_py_with = timerange_argument)] timerange: String,
37        scan: Option<&Bound<'_, PyAny>>,
38        #[pyo3(from_py_with = intent_argument)] intent: String,
39        array: Option<&Bound<'_, PyAny>>,
40        #[pyo3(from_py_with = correlation_argument)] correlation: String,
41        #[pyo3(from_py_with = obs_argument)] obs: String,
42        #[pyo3(from_py_with = datacolumn_argument)] datacolumn: String,
43    ) -> PyResult<bool> {
44        // The original algorithm leaves these switches unused. It parses
45        // the time quantum and always fits over one integration at a time.
46        let _ = (byantenna, sepacs, fitcorr);
47        let selection = MsSelection {
48            field: all_if_minus_one(field)?,
49            spw: all_if_minus_one(spw)?,
50            antenna: optional_expression(antenna)?,
51            timerange: nonempty(timerange),
52            scan: optional_expression(scan)?,
53            intent: nonempty(intent),
54            array: optional_expression(array)?,
55            correlation: nonempty(correlation.to_ascii_uppercase()),
56            observation: nonempty(obs),
57            ..Default::default()
58        };
59        let options = LegacyStatWtOptions {
60            rms: dorms,
61            minimum_samples: minsamp as u32,
62            data_column: datacolumn,
63            fit_spw: all_if_minus_one(fitspw)?.unwrap_or_default(),
64            combine,
65            selection,
66        };
67        py.detach(|| self.reweight_legacy(&options, timebin.as_ref()))
68    }
69}

71impl PyMsTool {
72    fn reweight_legacy(
73        &self,
74        options: &LegacyStatWtOptions,
75        timebin: Option<&Variant>,
76    ) -> PyResult<bool> {
77        let mut attached = self.attached.lock().map_err(runtime_error)?;
78        let ms = attached
79            .as_mut()
80            .ok_or_else(|| runtime_error("Table ... Null table object does not exist"))?;
81        if !ms.writable {
82            return Err(runtime_error("MeasurementSet is not writable"));
83        }
84        if let Some(bin) = timebin {
85            quantity(bin)?;
86        }
87        let link = ms.working_metadata_link();
88        let view = link.working_view().map_err(runtime_error)?;
89        let path = view.path();
90        let mut physical = StatWtMs::open_legacy(path).map_err(runtime_error)?;
91        let sources = physical.legacy_sources().map_err(runtime_error)?;
92        let mut transient = None;
93        let locks = if sources.iter().all(|path| ms.write_locks.contains_key(path)) {
94            &mut ms.write_locks
95        } else {
96            let locks = StatWtMs::write_locks(&sources).map_err(runtime_error)?;
97            physical = StatWtMs::open_legacy(path).map_err(runtime_error)?;
98            transient.insert(locks)
99        };
100        link.verify_directory().map_err(runtime_error)?;
101        let result = reweight_legacy(&mut physical, options, locks);
102        drop(physical);
103        if let Some(locks) = transient {
104            for lock in locks.into_values() {
105                lock.release().map_err(runtime_error)?;
106            }
107        }
108        result.map_err(runtime_error)
109    }
110}
