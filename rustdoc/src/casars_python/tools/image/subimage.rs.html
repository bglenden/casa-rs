18pub(super) enum Region {
19    Text(String),
20    Record(RecordValue),
21}
22
23impl Region {
24    pub(super) fn argument(
25        py: Python<'_>,
26        value: Option<&Bound<'_, PyAny>>,
27        method: &str,
28    ) -> PyResult<Option<Self>> {
29        let region = Self::capture(py, value)?;
30        if let Some(Self::Record(record)) = &region {
31            let Some(Value::Scalar(ScalarValue::String(name))) = record.get("name") else {
32                return Err(super::stub(method).unwrap_err());
33            };
34            match name.as_str() {
35                "LCSlicer" | "LCBox" | "WCBox" | "WCPolygon" | "WCEllipsoid" => {}
36                "WCUnion" | "WCDifference" | "WCIntersection" | "WCComplement" | "WCExtension"
37                | "WCConcatenation" => {
38                    if record.get("regions").is_none() {
39                        return Err(raised("RecordInterface: field regions is unknown"));
40                    }
41                }
42                _ => return Err(super::stub(method).unwrap_err()),
43            }
44        }
45        Ok(region)
46    }
47
48    pub(super) fn reject_compound_record(region: Option<&Self>) -> PyResult<()> {
49        if matches!(region, Some(Self::Record(record)) if record.get("regions").is_some()) {
50            return Err(raised(
51                "Exception: Only a single n-dimensional rectangular region is supported.",
52            ));
53        }
54        Ok(())
55    }
56
57    pub(super) fn capture(
58        py: Python<'_>,
59        value: Option<&Bound<'_, PyAny>>,
60    ) -> PyResult<Option<Self>> {
61        let Some(value) = value else {
62            return Ok(None);
63        };
64        if let Ok(record) = value.cast::<PyDict>() {
65            let record = casa_record(py, Some(record))?;
66            return Ok((!record.is_empty()).then_some(Self::Record(record)));
67        }
68        if let Ok(text) = value.cast::<PyString>() {
69            let text = text.to_str()?;
70            return Ok((!text.is_empty()).then(|| Self::Text(text.to_owned())));
71        }
72        if empty_variant(value)? {
73            return Ok(None);
74        }
75        Err(raised("Unsupported type for region"))
76    }
77
78    pub(super) fn as_region(&self) -> SubimageRegion<'_> {
79        match self {
80            Self::Text(text) => SubimageRegion::Text(text),
81            Self::Record(record) => SubimageRegion::Record(record),
82        }
83    }
84
85    pub(super) fn history(&self) -> String {
86        match self {
87            Self::Text(text) => format!("{text:?}"),
88            Self::Record(_) => "{...}".to_owned(),
89        }
90    }
91}

93/// Draw the same distinction CASA makes between its default record and an
94/// explicitly supplied empty text or array.
95pub(super) struct RegionArgument {
96    selection: Option<Region>,
97    record: bool,
98    empty_history: &'static str,
99}
100
101impl Default for RegionArgument {
102    fn default() -> Self {
103        Self {
104            selection: None,
105            record: true,
106            empty_history: "{...}",
107        }
108    }
109}
110
111impl<'a, 'py> FromPyObject<'a, 'py> for RegionArgument {
112    type Error = PyErr;
113
114    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
115        let record = value.is_instance_of::<PyDict>();
116        let empty_history = if record {
117            "{...}"
118        } else if value.is_instance_of::<PyString>() {
119            r#""""#
120        } else {
121            "[]"
122        };
123        Ok(Self {
124            selection: Region::capture(value.py(), (!value.is_none()).then_some(&value))?,
125            record,
126            empty_history,
127        })
128    }
129}
130
131impl RegionArgument {
132    pub(super) fn empty_text() -> Self {
133        Self {
134            selection: None,
135            record: false,
136            empty_history: r#""""#,
137        }
138    }
139
140    pub(super) fn selection(&self) -> Option<SubimageRegion<'_>> {
141        self.selection.as_ref().map(Region::as_region)
142    }
143
144    pub(super) fn validate_selectors(
145        &self,
146        spatial: &str,
147        chans: &str,
148        stokes: &str,
149    ) -> PyResult<()> {
150        if self.record || self.selection.is_some() {
151            if !spatial.is_empty() {
152                return Err(raised(
153                    "box and regionPtr cannot be simultaneously specified",
154                ));
155            }
156            if !chans.is_empty() || !stokes.is_empty() {
157                return Err(raised(
158                    "regionPtr and regionName, chans, and/or stokes cannot be simultaneously specified",
159                ));
160            }
161        }
162        Ok(())
163    }
164
165    pub(super) fn history(&self) -> String {
166        self.selection
167            .as_ref()
168            .map_or_else(|| self.empty_history.into(), Region::history)
169    }
170}

172pub(super) fn require_bool(value: &Bound<'_, PyAny>) -> PyResult<bool> {
173    value
174        .cast::<PyBool>()
175        .map(|value| value.is_true())
176        .map_err(|_| PyTypeError::new_err("expected bool"))
177}
178
179fn empty_variant(value: &Bound<'_, PyAny>) -> PyResult<bool> {
180    if value.is_instance_of::<PyList>() || value.is_instance_of::<PyTuple>() {
181        return Ok(value
182            .try_iter()?
183            .all(|item| item.is_ok_and(|item| item.is_instance_of::<PyBool>())));
184    }
185    Ok(value.extract::<Variant>()?.is_empty_sequence())
186}
187
188pub(super) fn mask_argument(value: Option<&Bound<'_, PyAny>>) -> PyResult<String> {
189    let Some(value) = value else {
190        return Ok(String::new());
191    };
192    if let Ok(text) = value.cast::<PyString>() {
193        return Ok(text.to_str()?.to_owned());
194    }
195    if empty_variant(value)? {
196        return Ok(String::new());
197    }
198    Err(raised("Mask is not understood, try a valid LEL string"))
199}

201#[pymethods]
202impl PyImageTool {
203    #[allow(clippy::too_many_arguments)]
204    #[pyo3(signature = (outfile="", region=None, mask=None, dropdeg=false, overwrite=false, list=true, stretch=false, wantreturn=true, keepaxes=None), text_signature = "($self, outfile='', region='', mask='', dropdeg=False, overwrite=False, list=True, stretch=False, wantreturn=True, keepaxes=[])")]
205    fn subimage(
206        &self,
207        py: Python<'_>,
208        outfile: &str,
209        region: Option<&Bound<'_, PyAny>>,
210        mask: Option<&Bound<'_, PyAny>>,
211        dropdeg: bool,
212        overwrite: bool,
213        list: bool,
214        stretch: bool,
215        wantreturn: bool,
216        keepaxes: Option<IntList>,
217    ) -> PyResult<Py<Self>> {
218        let region = Region::argument(py, region, "subimage")?;
219        let mask = mask_argument(mask)?;
220        let keepaxes = keepaxes.map_or_else(Vec::new, |axes| axes.0);
221        let (result, logs) = py.detach(|| {
222            let state = self.state();
223            let attached = state
224                .image
225                .as_ref()
226                .ok_or_else(|| raised("Unable to create image"))?;
227            let request = SubimageRequest {
228                outfile,
229                region: region.as_ref().map(Region::as_region),
230                mask: (!mask.is_empty()).then_some(mask.as_str()),
231                dropdeg,
232                overwrite,
233                list,
234                stretch,
235                wantreturn,
236                keepaxes: &keepaxes,
237            };
238            extract_subimage(attached, &request, state.history, external_open_check())
239                .map_err(raised)
240        })?;
241        emit_tool_logs(py, &logs)?;
242        Py::new(py, result.map_or_else(Self::new, Self::attached))
243    }
