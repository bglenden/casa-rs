3//! Normalization: CASA's rules that turn the
4//! unnormalized normal state into image units. The minor cycle and the
5//! published products both use them.
6//!
7//! | Quantity | Rule (CASA) | Here |
8//! |---|---|---|
9//! | PSF | each plane ÷ the term-0 PSF's peak | `psf_peak`, `normalized_psf_value` |
10//! | dirty or residual, standard gridders | ÷ sumwt | [`normalize_plane`] |
11//! | PB (mosaic, AW) | √\|weight\| ÷ pbmax, zero where PB ≤ pblimit | [`MosaicSensitivity::primary_beam`], [`MosaicSensitivity::weighted_primary_beam`] |
12//! | residual, flat noise | ÷ (pbmax·√\|weight\|) where that exceeds pblimit·pbmax², else 0 | [`MosaicSensitivity::normalize`], [`MosaicSensitivity::normalize_weighted_residual_sample`] |
13//! | residual, flat sky, one term | ÷ weight where it exceeds pblimit²·pbmax², else 0 | the same |
14//! | residual, flat sky, Taylor family | ÷ weight where it exceeds \|pblimit\|·pbmax², else 0 | [`MosaicSensitivity::normalize_weighted_residual_sample`] |
15//! | residual, pbsquare | ÷ pbmax² everywhere | the same |
16//! | model, flat noise | × PB before prediction, ÷ PB after the solve | [`MosaicSensitivity::physical_to_apparent`], [`MosaicSensitivity::apparent_to_physical`] |
17//! | model, pbsquare | × PB², ÷ PB², where PB² exceeds \|pblimit\|, else 0 | the same |
18//! | pbcor | restored ÷ PB where PB > pblimit | [`MosaicSensitivity::correct_primary_beam`] |
19//!
20//! The PB and residual rules are `SIImageStore::makePBFromWeight` and
21//! `divideResidualByWeight`. For one term both residual cutoffs are
22//! PB > pblimit where the weight is positive; at negative weights, which FFT
23//! ringing leaves at a mosaic's edges, flat noise keeps CASA's `abs` and
24//! flat sky has no support. `SIImageStoreMultiTerm::divideResidualByWeight`
25//! keeps the flat-noise cutoff for flat sky, so a Taylor family's flat-sky
26//! residual is cut at PB ≤ √pblimit ([`ResidualFamily`]).
27

34/// The principal PSF's measured peak, which publication divides every
35/// PSF plane by; coupled Taylor terms all use this one divisor.
36pub(crate) fn psf_peak(mut values: impl Iterator<Item = f32>) -> Result<f32, ProductsError> {
37    values.try_fold(0.0_f32, |peak, value| {
38        value
39            .is_finite()
40            .then_some(peak.max(value))
41            .ok_or(ProductsError::GeneratedNonfinite)
42    })
43}
44
45pub(crate) fn normalized_psf_value(value: f32, peak: f32) -> f32 {
46    if peak > 0.0 { value / peak } else { 0.0 }
47}

49/// Normalize a standard-gridder plane: divide it by its sum of weights.
50///
51/// A plane without a positive, finite sum of weights blanks to NaN.
52#[must_use]
53pub fn normalize_plane(values: &[f32], sum_weight: f64) -> Vec<f32> {
54    if !(sum_weight.is_finite() && sum_weight > 0.0) {
55        return vec![f32::NAN; values.len()];
56    }
57    values
58        .iter()
59        .map(|value| value / sum_weight as f32)
60        .collect()
61}

63/// Whether a residual is one term or a member of a Taylor family. CASA
64/// normalizes them in different image stores, whose flat-sky cutoffs
65/// differ: `SIImageStore::divideResidualByWeight` compares the weight with
66/// pblimit²·pbmax², `SIImageStoreMultiTerm::divideResidualByWeight` with
67/// |pblimit|·pbmax².
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub enum ResidualFamily {
70    /// One term per plane: an MFS image or a cube channel.
71    Single,
72    /// A Taylor coefficient of an MT-MFS family.
73    Taylor,
74}

76/// The image units a direction-dependent minor cycle solves in.
77///

86impl MinorCycleImageResponse {
87    /// Bind CASA's flat-noise, flat-sky or pbsquare image units and PB
88    /// support.
89    ///
90    /// # Errors
91    ///
92    /// Unit-response normalization has no direction-dependent model mapping.
93    pub fn new(
94        normalization: ProductNormalization,
95        policy: PrimaryBeamValidityPolicy,
96    ) -> Result<Self, ProductsError> {
97        if !matches!(
98            normalization,
99            ProductNormalization::FlatNoise
100                | ProductNormalization::FlatSky

102        ) {
103            return Err(ProductsError::UnsupportedNormalization);
104        }

107            policy,
108        })
109    }

121    }

124/// One plane's direction-dependent sensitivity, borrowed from the normal
125/// state.
126///
127/// It derives CASA's unit-peak PB, the flat-noise, flat-sky and pbsquare
128/// residuals, PB support and the apparent and physical model, through the
129/// PB (flat noise) or PB² (pbsquare). It owns no CLEAN mask and leaves the
130/// normal state unchanged.
131#[derive(Debug, Clone, Copy)]
132pub struct MosaicSensitivity<'a> {
133    values: &'a [f64],
134    peak: f64,
135    normal_sum_weight: f64,
136    model_pb_scale: f32,
137}
138
139impl<'a> MosaicSensitivity<'a> {
140    /// Bind finite sensitivity with positive support.
141    ///
142    /// # Errors
143    ///
144    /// Empty or non-finite sensitivity, or sensitivity with no positive
145    /// value: no weighted sample reached the plane.
146    pub fn new(values: &'a [f64]) -> Result<Self, ProductsError> {
147        Self::try_new(values)?.ok_or(ProductsError::UnsupportedSensitivity)
148    }
149
150    /// Bind finite sensitivity, or `None` for a plane no weighted sample
151    /// reached (a flagged or unmapped cube channel). CASA skips such a plane
152    /// and publishes zeros for it (`SIImageStore::makePBFromWeight`,
153    /// `divideResidualByWeight`, `divideModelByWeight` when its
154    /// `pbmax <= 0`).
155    ///
156    /// # Errors
157    ///
158    /// Empty or non-finite sensitivity.
159    pub fn try_new(values: &'a [f64]) -> Result<Option<Self>, ProductsError> {
160        if values.is_empty() || values.iter().any(|value| !value.is_finite()) {
161            return Err(ProductsError::GeneratedNonfinite);
162        }
163        let peak = values.iter().copied().fold(0.0_f64, f64::max);
164        Ok((peak > 0.0).then(|| Self {
165            values,
166            peak,
167            normal_sum_weight: 1.0,
168            model_pb_scale: (peak as f32).sqrt(),
169        }))
170    }
171
172    /// Bind the sum of weights that forms CASA's published Float weight
173    /// image.
174    ///
175    /// Model conversion evaluates `Float(raw / sum_weight)` before its roots
176    /// and support decision. The raw sensitivity is unchanged and no
177    /// normalized weight plane is allocated.
178    ///
179    /// # Errors
180    ///
181    /// The sum of weights or the scaled peak is not positive and finite.
182    pub fn with_normal_sum_weight(mut self, sum_weight: f64) -> Result<Self, ProductsError> {
183        if !sum_weight.is_finite() || sum_weight <= 0.0 {
184            return Err(ProductsError::UnsupportedSensitivity);
185        }
186        let scale = ((self.peak / sum_weight) as f32).sqrt();
187        if !scale.is_finite() || scale <= 0.0 {
188            return Err(ProductsError::UnsupportedSensitivity);
189        }
190        self.normal_sum_weight = sum_weight;
191        self.model_pb_scale = scale;
192        Ok(self)
193    }
194
195    fn finite(value: f64) -> Result<f64, ProductsError> {
196        value
197            .is_finite()
198            .then_some(value)
199            .ok_or(ProductsError::GeneratedNonfinite)
200    }
201
202    fn supported(pb: f64, policy: PrimaryBeamValidityPolicy) -> bool {
203        match policy.comparison() {
204            ProductSupportComparison::StrictlyGreater => pb > f64::from(policy.cutoff()),
205        }
206    }

208    /// Return one unit-peak primary-beam value, `sqrt(|sensitivity| / peak)`.
209    #[must_use]
210    pub fn primary_beam_at(self, index: usize) -> f64 {
211        (self.values[index].abs() / self.peak).sqrt()
212    }
213
214    /// Return whether a pixel is inside the PB support.
215    #[must_use]
216    pub fn valid_at(self, index: usize, policy: PrimaryBeamValidityPolicy) -> bool {
217        Self::supported(self.primary_beam_at(index), policy)
218    }
219
220    fn model_primary_beam_at(self, index: usize) -> Result<f64, ProductsError> {
221        let weight = (self.values[index] / self.normal_sum_weight) as f32;
222        let pb = weight.abs().sqrt() / self.model_pb_scale;
223        Self::finite(f64::from(pb))
224    }
225
226    /// The beam squared pbsquare converts the model with, as CASA's Float
227    /// stages form it: `abs(weight) / PBScale²`
228    /// (`SIImageStore::divideModelByWeight`, `multiplyModelByWeight`).
229    fn model_primary_beam_squared_at(self, index: usize) -> Result<f64, ProductsError> {
230        let weight = (self.values[index] / self.normal_sum_weight) as f32;
231        let square = weight.abs() / (self.model_pb_scale * self.model_pb_scale);
232        Self::finite(f64::from(square))
233    }
234
235    /// Build CASA's unit-peak PB, `sqrt(|sensitivity| / peak)`.
236    #[must_use]
237    pub fn primary_beam(self) -> Vec<f32> {
238        self.values
239            .iter()
240            .map(|value| (value.abs() / self.peak).sqrt() as f32)
241            .collect()
242    }

244    /// Derive PB from a published CASA Float weight plane, retaining Float
245    /// roots.
246    ///
247    /// # Errors
248    ///
249    /// The weight plane is empty, non-finite or has no positive value.
250    pub fn primary_beam_from_weight(weight: &[f32]) -> Result<Vec<f32>, ProductsError> {
251        if weight.is_empty() || weight.iter().any(|value| !value.is_finite()) {
252            return Err(ProductsError::GeneratedNonfinite);
253        }
254        let peak = weight.iter().copied().fold(0.0_f32, f32::max);
255        if peak <= 0.0 {
256            return Err(ProductsError::UnsupportedSensitivity);
257        }
258        let scale = peak.sqrt();
259        Ok(weight
260            .iter()
261            .map(|value| value.abs().sqrt() / scale)
262            .collect())
263    }

265    /// Normalize a raw residual plane to flat noise, flat sky or pbsquare.
266    ///
267    /// This is [`Self::normalize_weighted_residual_sample`] in the raw
268    /// sensitivity's units, where every sum of weights cancels: flat noise
269    /// divides by √(|sensitivity|·peak) where that exceeds cutoff·peak, flat
270    /// sky by the sensitivity where it exceeds cutoff²·peak, and the
271    /// residual is zero elsewhere; pbsquare divides every pixel by the
272    /// peak.
273    ///
274    /// # Errors
275    ///
276    /// Unit-response normalization, or a non-finite result.
277    ///
278    /// # Panics
279    ///
280    /// If `values` is not one value per sensitivity pixel.
281    pub fn normalize(
282        self,
283        values: &[f32],
284        normalization: ProductNormalization,
285        policy: PrimaryBeamValidityPolicy,
286    ) -> Result<Vec<f32>, ProductsError> {
287        assert_eq!(
288            values.len(),
289            self.values.len(),
290            "a residual plane and its sensitivity have one shape"
291        );
292        let limit = f64::from(policy.cutoff().abs());
293        values
294            .iter()
295            .zip(self.values)
296            .map(|(value, sensitivity)| {
297                let (denominator, cutoff) = match normalization {
298                    ProductNormalization::FlatNoise => {
299                        ((sensitivity.abs() * self.peak).sqrt(), limit * self.peak)
300                    }
301                    ProductNormalization::FlatSky => (*sensitivity, limit * limit * self.peak),
302                    ProductNormalization::PbSquare => {
303                        return Ok(Self::finite(f64::from(*value) / self.peak)? as f32);
304                    }
305                    ProductNormalization::UnitResponse => {
306                        return Err(ProductsError::UnsupportedNormalization);
307                    }
308                };
309                if denominator > cutoff {
310                    Ok(Self::finite(f64::from(*value) / denominator)? as f32)
311                } else {
312                    Ok(0.0)
313                }
314            })
315            .collect()
316    }
317
318    /// Normalize a residual through CASA's Float sum-weight and weight-image
319    /// stages.
320    ///
321    /// The normal sum weight is bound by [`Self::with_normal_sum_weight`].
322    /// All Taylor terms divide by the principal publication sum weight given
323    /// here. Flat noise divides by `sqrt(abs(weight)) * PBScale` where that
324    /// exceeds `cutoff * PBScale^2`: PB > cutoff. Flat sky divides by
325    /// `weight` where it exceeds `cutoff^2 * PBScale^2` for one term
326    /// (PB > cutoff) and `cutoff * PBScale^2` for a Taylor family
327    /// (PB > √cutoff), as CASA's two image stores do. Elsewhere the residual
328    /// is zero. Pbsquare divides every pixel by `PBScale^2`, with no cutoff.
329    ///
330    /// # Errors
331    ///
332    /// Unit-response normalization, or non-finite input or arithmetic.
333    pub fn normalize_weighted_residual_sample(
334        self,
335        value: f64,
336        index: usize,
337        normalization: ProductNormalization,
338        family: ResidualFamily,
339        published_sum_weight: f64,
340        policy: PrimaryBeamValidityPolicy,
341    ) -> Result<f64, ProductsError> {
342        let mut residual = value as f32;
343        let sum_weight = published_sum_weight as f32;
344        if !residual.is_finite() || !sum_weight.is_finite() {
345            return Err(ProductsError::GeneratedNonfinite);
346        }
347        if sum_weight != 1.0 {
348            residual = if casa_imaging_deconvolution::casa_divides_plane(f64::from(sum_weight)) {
349                residual / sum_weight
350            } else {
351                0.0
352            };
353        }
354        let weight = (self.values[index] / self.normal_sum_weight) as f32;
355        if !weight.is_finite() {
356            return Err(ProductsError::GeneratedNonfinite);
357        }
358        if normalization == ProductNormalization::PbSquare {
359            return Self::finite(f64::from(
360                residual / (self.model_pb_scale * self.model_pb_scale),
361            ));
362        }
363        let pb_scale = f64::from(self.model_pb_scale);
364        let limit = f64::from(policy.cutoff().abs());
365        let (denominator, cutoff) = match normalization {
366            ProductNormalization::FlatNoise => (
367                weight.abs().sqrt() * self.model_pb_scale,
368                limit * pb_scale * pb_scale,
369            ),
370            ProductNormalization::FlatSky => match family {
371                ResidualFamily::Single => (weight, limit * limit * pb_scale * pb_scale),
372                ResidualFamily::Taylor => (weight, limit * pb_scale * pb_scale),
373            },
374            ProductNormalization::UnitResponse | ProductNormalization::PbSquare => {
375                return Err(ProductsError::UnsupportedNormalization);
376            }
377        };
378        let normalized = if denominator > cutoff as f32 {
379            residual / denominator
380        } else {
381            0.0
382        };
383        Self::finite(f64::from(normalized))
384    }

386    /// Form CASA's cutoff-blanked Float PB from the bound normalized weight.
387    ///
388    /// This weight-image route takes the absolute value before its square
389    /// root, matching model conversion. Scalar mosaic products use the
390    /// unnormalized positive-sensitivity [`Self::primary_beam`] instead.
391    ///
392    /// # Errors
393    ///
394    /// A non-finite PB value.
395    pub fn weighted_primary_beam(
396        self,
397        policy: PrimaryBeamValidityPolicy,
398    ) -> Result<Vec<f32>, ProductsError> {
399        (0..self.values.len())
400            .map(|index| self.weighted_primary_beam_at(index, policy))
401            .collect()
402    }

404    /// Pixel `index` of [`Self::weighted_primary_beam`]: its Float PB
405    /// where that exceeds the cutoff, and zero elsewhere
406    /// (`SIImageStore::makePBFromWeight` keeps `sqrt(|weight|) / pbmax`
407    /// above `|pblimit|` and zeroes the rest). It is positive at the pixels
408    /// `SIImageStore::calcRobustRMS` measures the noise on at tclean's
409    /// `pbmask` of 0, and exceeds a `pbmask` above zero at those
410    /// `calcRobustRMS` and `SDMaskHandler::makePBMask` keep
411    /// (`pb > pbmasklevel`).
412    ///
413    /// # Errors
414    ///
415    /// A non-finite PB value.
416    pub fn weighted_primary_beam_at(
417        self,
418        index: usize,
419        policy: PrimaryBeamValidityPolicy,
420    ) -> Result<f32, ProductsError> {
421        let pb = self.model_primary_beam_at(index)?;
422        Ok(if Self::supported(pb, policy) {
423            pb as f32
424        } else {
425            0.0
426        })
427    }

432        (0..self.values.len())

437    /// Map a physical flat-sky model pixel to the units the minor cycle
438    /// solves in.
439    ///
440    /// CASA flat noise multiplies by the PB inside its support and zeroes
441    /// the rest; pbsquare multiplies by the PB squared where that exceeds
442    /// the cutoff and zeroes the rest; flat sky is the identity. Apply this
443    /// before restoring-beam convolution.
444    ///
445    /// # Errors
446    ///
447    /// Unit-response normalization, or non-finite arithmetic.
448    pub fn physical_to_apparent(
449        self,
450        value: f64,
451        index: usize,
452        normalization: ProductNormalization,
453        policy: PrimaryBeamValidityPolicy,
454    ) -> Result<f64, ProductsError> {
455        match normalization {
456            ProductNormalization::FlatNoise => {
457                let pb = self.model_primary_beam_at(index)?;
458                if Self::supported(pb, policy) {
459                    Self::finite(value * pb)
460                } else {
461                    Ok(0.0)
462                }
463            }
464            ProductNormalization::PbSquare => {
465                let square = self.model_primary_beam_squared_at(index)?;
466                if Self::supported(square, policy) {
467                    Self::finite(value * square)
468                } else {
469                    Ok(0.0)
470                }
471            }
472            ProductNormalization::FlatSky => Self::finite(value),
473            ProductNormalization::UnitResponse => Err(ProductsError::UnsupportedNormalization),
474        }
475    }

477    /// Map a solved pixel to the physical model that prediction consumes.
478    ///
479    /// Apply it to every pixel after multiscale expansion, not only to the
480    /// component centre. PB support is not the CLEAN mask.
481    ///
482    /// # Errors
483    ///
484    /// Unit-response normalization, or non-finite arithmetic.
485    pub fn apparent_to_physical(
486        self,
487        value: f64,
488        index: usize,
489        normalization: ProductNormalization,
490        policy: PrimaryBeamValidityPolicy,
491    ) -> Result<f64, ProductsError> {
492        match normalization {
493            ProductNormalization::FlatNoise => {
494                let pb = self.model_primary_beam_at(index)?;
495                if Self::supported(pb, policy) {
496                    Self::finite(value / pb)
497                } else {
498                    Ok(0.0)
499                }
500            }
501            ProductNormalization::PbSquare => {
502                let square = self.model_primary_beam_squared_at(index)?;
503                if Self::supported(square, policy) {
504                    Self::finite(value / square)
505                } else {
506                    Ok(0.0)
507                }
508            }
509            ProductNormalization::FlatSky => Self::finite(value),
510            ProductNormalization::UnitResponse => Err(ProductsError::UnsupportedNormalization),
511        }
512    }

514    /// PB-correct a normalized plane inside the PB support and zero it
515    /// outside.
516    ///
517    /// # Errors
518    ///
519    /// A non-finite result.
520    ///
521    /// # Panics
522    ///
523    /// If `values` is not one value per sensitivity pixel.
524    pub fn correct_primary_beam(
525        self,
526        values: &[f32],
527        policy: PrimaryBeamValidityPolicy,
528    ) -> Result<Vec<f32>, ProductsError> {
529        assert_eq!(
530            values.len(),
531            self.values.len(),
532            "a plane and its sensitivity have one shape"
533        );
534        values
535            .iter()
536            .enumerate()
537            .map(|(index, value)| {
538                let pb = self.primary_beam_at(index);
539                let corrected = if Self::supported(pb, policy) {
540                    f64::from(*value) / pb
541                } else {
542                    0.0
543                } as f32;
544                corrected
545                    .is_finite()
546                    .then_some(corrected)
547                    .ok_or(ProductsError::GeneratedNonfinite)
548            })
549            .collect()
550    }

590    /// CASA's `pbsquare` (`SIImageStore::divideResidualByWeight`,
591    /// `divideModelByWeight`, `multiplyModelByWeight`): the residual divided
592    /// by the PB peak squared everywhere, with no cutoff, and the model
593    /// taken to and from apparent units through PB², zero where PB² is not
594    /// above |pblimit|.
595    #[test]
596    fn pbsquare_divides_the_residual_by_the_peak_and_the_model_by_pb_squared() {
597        // The weight is the sensitivity over a sum of weights of 1, so
598        // PBScale = √(4 / 1) = 2 and PBScale² = 4: PB² = |weight| / 4 = 1,
599        // 0.25, 0.09, 0.01 and, for the negative weight, 0.01 again.
600        let sensitivity = [4.0, 1.0, 0.36, 0.04, -0.04];
601        let response = MosaicSensitivity::new(&sensitivity)
602            .unwrap()
603            .with_normal_sum_weight(1.0)
604            .unwrap();
605        let pb = policy(0.2);
606        let residual = [8.0, 4.0, 2.0, 1.0, 1.0];
607        let raw = response
608            .normalize(
609                &residual.map(|value| value as f32),
610                ProductNormalization::PbSquare,
611                pb,
612            )
613            .unwrap();
614        // ÷ the peak 4, everywhere.
615        assert_eq!(raw, [2.0, 1.0, 0.5, 0.25, 0.25]);
616        for family in [ResidualFamily::Single, ResidualFamily::Taylor] {
617            let weighted = (0..sensitivity.len())
618                .map(|index| {
619                    response
620                        .normalize_weighted_residual_sample(
621                            residual[index],
622                            index,
623                            ProductNormalization::PbSquare,
624                            family,
625                            2.0,
626                            pb,
627                        )
628                        .unwrap()
629                })
630                .collect::<Vec<_>>();
631            // ÷ sumwt 2, then ÷ PBScale² = 4, everywhere.
632            assert_eq!(weighted, [1.0, 0.5, 0.25, 0.125, 0.125], "{family:?}");
633        }
634        let convert = |cutoff: f32, to_apparent: bool| {
635            (0..sensitivity.len())
636                .map(|index| {
637                    if to_apparent {
638                        response.physical_to_apparent(
639                            1.0,
640                            index,
641                            ProductNormalization::PbSquare,
642                            policy(cutoff),
643                        )
644                    } else {
645                        response.apparent_to_physical(
646                            0.25,
647                            index,
648                            ProductNormalization::PbSquare,
649                            policy(cutoff),
650                        )
651                    }
652                    .unwrap()
653                })
654                .collect::<Vec<_>>()
655        };
656        assert_eq!(convert(0.2, true), [1.0, 0.25, 0.0, 0.0, 0.0]);
657        assert_eq!(convert(0.2, false), [0.25, 1.0, 0.0, 0.0, 0.0]);
658        // Below a cutoff of 0.005 the negative weight converts as its
659        // absolute value does (CASA's `abs(wt)`).
660        let square = f64::from(0.04_f32 / 4.0);
661        let apparent = convert(0.005, true);
662        assert_eq!(apparent[3..], [square, square]);
663        let physical = convert(0.005, false);
664        assert_eq!(physical[3..], [0.25 / square, 0.25 / square]);
665    }

713    /// `SIImageStore::divideResidualByWeight`'s cutoffs, pblimit·pbmax² on
714    /// pbmax·√|weight| for flat noise and pblimit²·pbmax² on the weight for
715    /// flat sky, are both PB > pblimit where the weight is positive.
716    #[test]
717    fn flat_noise_and_flat_sky_residuals_keep_casa_support() {
718        let sensitivity = [4.0, 0.2, 0.4, 0.5, -0.5, 0.15, 0.17];
719        let response = MosaicSensitivity::new(&sensitivity).unwrap();
720        let limit = policy(0.2);
721        // PB = sqrt(|sensitivity| / 4): 1, 0.224, 0.316, 0.354, 0.354, 0.194, 0.206.
722        let support = response.validity(limit);
723        assert_eq!(support, [true, true, true, true, true, false, true]);
724        for (normalization, expected) in [
725            (ProductNormalization::FlatNoise, support.clone()),
726            (
727                ProductNormalization::FlatSky,
728                vec![true, true, true, true, false, false, true],
729            ),
730        ] {
731            let staged = (0..sensitivity.len())
732                .map(|index| {
733                    response
734                        .normalize_weighted_residual_sample(
735                            2.0,
736                            index,
737                            normalization,
738                            ResidualFamily::Single,
739                            2.0,
740                            limit,
741                        )
742                        .unwrap()
743                })
744                .collect::<Vec<_>>();
745            let raw = response.normalize(&[1.0; 7], normalization, limit).unwrap();
746            for (index, valid) in expected.iter().enumerate() {
747                assert_eq!(
748                    staged[index] != 0.0,
749                    *valid,
750                    "{normalization:?} staged {index}"
751                );
752                assert_eq!(raw[index] != 0.0, *valid, "{normalization:?} raw {index}");
753                assert!(
754                    (staged[index] - f64::from(raw[index])).abs() <= 1.0e-6 * staged[index].abs(),
755                    "{normalization:?} {index}: {} != {}",
756                    staged[index],
757                    raw[index]
758                );
759            }
760        }
761        let flat_sky = (0..5)
762            .map(|index| {
763                response
764                    .normalize_weighted_residual_sample(
765                        2.0,
766                        index,
767                        ProductNormalization::FlatSky,
768                        ResidualFamily::Single,
769                        2.0,
770                        limit,
771                    )
772                    .unwrap()
773            })
774            .collect::<Vec<_>>();
775        assert_eq!(flat_sky, [0.25, 5.0, 2.5, 2.0, 0.0]);
776    }

778    /// `SIImageStoreMultiTerm::divideResidualByWeight` keeps
779    /// |pblimit|·pbmax² for flat sky: a Taylor family's flat-sky residual is
780    /// zero where pblimit < PB ≤ √pblimit, which one term keeps. Flat noise

782    #[test]
783    fn a_taylor_flat_sky_residual_keeps_casas_multi_term_cutoff() {
784        let sensitivity = [4.0, 0.2, 0.4, 0.5, -0.5, 0.15, 0.17, 0.9];
785        let response = MosaicSensitivity::new(&sensitivity).unwrap();
786        let limit = policy(0.2);
787        // weight / pbmax² = 1, 0.05, 0.1, 0.125, -0.125, 0.0375, 0.0425,
