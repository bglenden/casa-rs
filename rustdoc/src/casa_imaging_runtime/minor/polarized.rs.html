16pub(super) fn solve<const N: usize>(
17    inputs: &Inputs<'_>,
18    admission: (&HostResources, &ResourcePolicy),
19    keys: &[PlaneKey],
20    statistics: &[PlaneStatistics],
21    summary: &PsfSummary,
22    controls: &CycleControls,
23) -> Result<Solved, MinorCycleRunError> {
24    assert_eq!(keys.len(), N);
25    let mut planes = keys
26        .iter()
27        .map(|key| {
28            load(inputs, *key).map(|plane| plane.unwrap_or_else(|| blank_plane(inputs, *key)))
29        })
30        .collect::<Result<Vec<PlaneData<'_>>, _>>()?;
31    let mut residual = Vec::with_capacity(N);
32    for plane in &mut planes {
33        residual.push(plane.residual.pop().expect("one residual per polarization"));
34    }
35    let first = &planes[0];
36    // CASA uses polarization zero's PSF even when it is blank. Its fitted
37    // beam set gives that plane the domain's largest beam.
38    let blank_summary = if !super::holds_data(inputs.normal, false, keys[0]) {
39        Some(summary.with_domain_beam(&first.psf[0], first.shape, summary.beam()))
40    } else {
41        None
42    };
43    let summary = blank_summary.as_ref().unwrap_or(summary);
44    let view = MinorCycleView {
45        shape: first.shape,
46        residual: &residual,
47        psf: &first.psf,
48        summary,
49        support: &first.support,
50    };
51    let combined = combine_statistics(statistics.iter());
52    let mut outcome = clean(&ClarkPolarized::<N>::new(None), &view, controls, &combined)?;
53    let end = std::mem::take(&mut outcome.end_residual);
54    let mut peak_no_mask = 0.0_f64;
55    for (p, plane) in planes.iter().enumerate() {
56        let Some(values) = end.get(p) else {
57            peak_no_mask = peak_no_mask.max(statistics[p].peak_no_mask);
58            continue;
59        };
60        let pixels = plane_pixels(inputs, keys[p], plane)?;
61        peak_no_mask = peak_no_mask.max(peak_magnitude(
62            values,
63            pixels.residual_mask(inputs.setup.residual_pixel_mask),
64        ));
65    }
66    let delta = std::mem::take(&mut outcome.delta);
67    let entries = (0..N).map(|p| delta.term(p).count()).sum::<usize>();
68    let charge = admit(
69        admission.0,
70        admission.1,
71        &Demand {
72            phase: "minor-cycle results",
73            memory: result_bytes(entries),
74        },
75    )?;
76    let mut terms = Vec::with_capacity(entries);
77    for p in 0..N {
78        let physical = planes[p].physical()?;
79        for (index, flux) in delta.term(p) {
80            let value = physical.apply(flux, index)?;
81            if value == 0.0 {
82                continue;
83            }
84            let key = keys[p];
85            let cell = ModelCell::new(
86                key.domain,
87                key.channel,
88                key.polarization,
89                planes[p].shape.pixel(index),
90            );
91            let flat = inputs
92                .base
93                .shape()
94                .flat_index(cell)
95                .ok_or(super::ModelLifecycleError::CellOutsideShape)?;
96            terms.push((
97                flat,
98                ModelDeltaTerm::new(
99                    cell,
100                    ModelValue::new(value).map_err(super::ModelLifecycleError::Contract)?,
101                ),
102            ));
103        }
104    }
105    let trace = group_trace(&outcome.trace, &planes, keys);
106    Ok(Solved {
107        outcome,
108        peak_no_mask,
109        terms,
110        dense: None,
111        mem_alpha: None,
112        trace,
113        charge,
114    })
115}

117/// CASA measures the mask and entry peaks over the whole polarization group;
118/// Clark's component support remains the first polarization's mask.
119fn combine_statistics<'a>(
120    statistics: impl IntoIterator<Item = &'a PlaneStatistics>,
121) -> PlaneStatistics {
122    let mut statistics = statistics.into_iter();
123    let mut combined = *statistics.next().expect("a polarization group is nonempty");
124    for other in statistics {
125        combined.mask_sum = combined.mask_sum.saturating_add(other.mask_sum);
126        combined.peak_in_mask = combined.peak_in_mask.max(other.peak_in_mask);
127        combined.peak_no_mask = combined.peak_no_mask.max(other.peak_no_mask);
128    }
129    combined
130}

132/// The zero-valued residual and PSF of a polarization with no data. CASA's
133/// joint Clark uses the first polarization's mask even when that plane is blank.
134fn blank_plane<'a>(inputs: &Inputs<'a>, key: PlaneKey) -> PlaneData<'a> {
135    let shape = plane_shape(inputs.normal, key);
136    let mask = inputs
137        .masks
138        .get(key.domain)
139        .and_then(|m| {
140            m.plane(
141                key.channel - inputs.normal.slab().core_range().start,
142                key.polarization,
143            )
144        })
145        .expect("mask for every plane");
146    PlaneData {
147        shape,
148        residual: vec![vec![0.0; shape.len()]],
149        psf: vec![vec![0.0; shape.len()]],
150        support: Support::new(shape, mask.to_vec()),
151        valid: Support::new(shape, vec![false; shape.len()]),
152        response: None,
153    }
154}

156fn group_trace(
157    components: &[Component],
158    planes: &[PlaneData<'_>],
159    keys: &[PlaneKey],
160) -> Vec<TracedComponent> {
161    components
162        .iter()
163        .map(|component| {
164            let key = keys[component.term];
165            TracedComponent {
166                cell: ModelCell::new(
167                    key.domain,
168                    key.channel,
169                    key.polarization,
170                    planes[component.term].shape.pixel(component.index),
171                ),
172                flux: component.flux,
173                scale_px: 0.0,
174            }
175        })
176        .collect()
177}

182/// Solve the valid polarization groups and leave wholly blank channels as rows.
183pub(super) fn solve_groups(
184    inputs: &Inputs<'_>,
185    admission: (&HostResources, &ResourcePolicy),
186    (planes, blank): GroupInputs<'_>,
187    cache: &PsfCache,
188    controls: &CycleControls,
189    team: &WorkerTeam,
190    results: &mut Vec<(PlaneKey, PlaneStatistics, Solved)>,
191) -> Result<(), MinorCycleRunError> {
192    let n_pol = inputs.normal.polarization_count();
193    let mut ordered = Vec::with_capacity(planes.len() + blank.len());
194    ordered.extend_from_slice(planes);
195    ordered.extend_from_slice(blank);
196    ordered.sort_unstable_by_key(|(key, _)| (key.domain, key.channel, key.polarization));
197    let mut groups = Vec::with_capacity(ordered.len() / n_pol);
198    blank.clear();
199    for chunk in ordered.chunks_exact(n_pol) {
200        if chunk
201            .iter()
202            .any(|(key, _)| holds_data(inputs.normal, false, *key))
203        {
204            groups.push(PolarizedSlot {
205                keys: std::array::from_fn(|p| chunk[p.min(n_pol - 1)].0),
206                stats: std::array::from_fn(|p| chunk[p.min(n_pol - 1)].1),
207                solved: None,
208            });
209        } else {
210            blank.push((
211                chunk[0].0,
212                combine_statistics(chunk.iter().map(|(_, stats)| stats)),
213            ));
214        }
215    }
216    team.for_each_mut(&mut groups, |_, group| {
217        let summary = cache
218            .summaries
219            .get(&group.keys[0])
220            .copied()
221            .unwrap_or_else(|| {
222                cache
223                    .summaries
224                    .iter()
225                    .filter(|(key, _)| key.domain == group.keys[0].domain)
226                    .map(|(_, summary)| *summary)
227                    .max_by(|a, b| {
228                        (a.beam().major_fwhm_rad() * a.beam().minor_fwhm_rad())
229                            .total_cmp(&(b.beam().major_fwhm_rad() * b.beam().minor_fwhm_rad()))
230                    })
231                    .expect("a group with data has a measured PSF")
232            });
233        group.solved = Some(match n_pol {
234            2 => solve::<2>(
235                inputs,
236                admission,
237                &group.keys[..n_pol],
238                &group.stats[..n_pol],
239                &summary,
240                controls,
241            )?,
242            4 => solve::<4>(
243                inputs,
244                admission,
245                &group.keys[..n_pol],
246                &group.stats[..n_pol],
247                &summary,
248                controls,
249            )?,
250            _ => unreachable!(),
251        });
252        Ok::<_, MinorCycleRunError>(())
253    })?;
254    results.extend(groups.into_iter().map(|mut group| {
255        let stats = combine_statistics(group.stats[..n_pol].iter());
256        (
257            group.keys[0],
258            stats,
259            group.solved.take().expect("every group solved"),
260        )
261    }));
262    Ok(())
263}
