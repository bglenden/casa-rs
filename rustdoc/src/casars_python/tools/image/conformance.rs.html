20const TOLERANCE: f64 = 1e-6;
21const WCS_TOLERANCE: f64 = 1e-13;
22
23fn near(left: f64, right: f64, tolerance: f64) -> bool {
24    if left == right {
25        return true;
26    }
27    if left == 0.0 || right == 0.0 {
28        return left.abs().max(right.abs()) <= (1.0 + tolerance) * f64::MIN_POSITIVE;
29    }
30    (left > 0.0) == (right > 0.0) && (left - right).abs() <= tolerance * left.abs().max(right.abs())
31}
32
33fn near_values(left: &[f64], right: &[f64], tolerance: f64) -> bool {
34    left.len() == right.len()
35        && left
36            .iter()
37            .zip(right)
38            .all(|(&left, &right)| near(left, right, tolerance))
39}

41fn scalar(value: Option<&Value>) -> Option<f64> {
42    match value {
43        Some(Value::Scalar(ScalarValue::Float64(value))) => Some(*value),
44        Some(Value::Scalar(ScalarValue::Float32(value))) => Some(f64::from(*value)),
45        Some(Value::Scalar(ScalarValue::Int32(value))) => Some(f64::from(*value)),
46        Some(Value::Scalar(ScalarValue::Int64(value))) => Some(*value as f64),
47        Some(Value::Array(ArrayValue::Float64(value))) if value.shape() == [1, 1] => {
48            value.first().copied()
49        }
50        _ => None,
51    }
52}
53
54fn near_matrix(left: Option<&Value>, right: Option<&Value>) -> bool {
55    if let (Some(left), Some(right)) = (scalar(left), scalar(right)) {
56        return near(left, right, TOLERANCE);
57    }
58    match (left, right) {
59        (
60            Some(Value::Array(ArrayValue::Float64(left))),
61            Some(Value::Array(ArrayValue::Float64(right))),
62        ) => {
63            left.shape() == right.shape()
64                && left
65                    .iter()
66                    .zip(right)
67                    .all(|(&left, &right)| near(left, right, TOLERANCE))
68        }
69        _ => left == right,
70    }
71}

73fn velocity_field<'a>(
74    stored: &'a RecordValue,
75    defaults: &'a RecordValue,
76    name: &str,
77    legacy: &str,
78) -> Option<&'a Value> {
79    stored
80        .get(name)
81        .or_else(|| stored.get(legacy))
82        .or_else(|| defaults.get(name))
83}

85fn pc_matrix(record: &RecordValue) -> Option<&Value> {
86    subrecord(record, "wcs")
87        .or_else(|| subrecord(record, "tabular"))
88        .unwrap_or(record)
89        .get("pc")
90}

92fn linear_labels_near(left: &[String], right: &[String]) -> bool {
93    left.len() == right.len()
94        && left.iter().zip(right).all(|(left, right)| {
95            left.split_whitespace()
96                .next()
97                .unwrap_or("")
98                .eq_ignore_ascii_case(right.split_whitespace().next().unwrap_or(""))
99        })
100}

102fn normalized_direction_transform(coordinate: &DirectionCoordinate) -> (Vec<f64>, Array2<f64>) {
103    let mut increments = coordinate.increment();
104    let mut matrix = coordinate.pc_matrix().clone();
105    for (index, mut row) in matrix.rows_mut().into_iter().enumerate() {
106        let power = row.iter().map(|value| value * value).sum::<f64>();
107        if power != 0.0 && power != 1.0 {
108            let diagonal = row[index];
109            let factor = power.sqrt() * (diagonal / diagonal.abs());
110            row.mapv_inplace(|value| value / factor);
111            increments[index] *= factor;
112        }
113    }
114    (increments, matrix)
115}

117fn directions_near(left: &DirectionCoordinate, right: &DirectionCoordinate) -> bool {
118    let (left_inc, left_pc) = normalized_direction_transform(left);
119    let (right_inc, right_pc) = normalized_direction_transform(right);
120    left.axis_names() == right.axis_names()
121        && left.world_units() == right.world_units()
122        && left.direction_ref() == right.direction_ref()
123        && left.conversion_direction_ref() == right.conversion_direction_ref()
124        && left.projection().projection_type() == right.projection().projection_type()
125        && near_values(
126            left.projection().parameters(),
127            right.projection().parameters(),
128            TOLERANCE,
129        )
130        && near(left.longpole(), right.longpole(), WCS_TOLERANCE)
131        && near(left.native_pole()[1], right.native_pole()[1], WCS_TOLERANCE)
132        && near_values(&left.reference_value(), &right.reference_value(), TOLERANCE)
133        && near_values(&left.reference_pixel(), &right.reference_pixel(), TOLERANCE)
134        && near_values(&left_inc, &right_inc, TOLERANCE)
135        && left_pc
136            .iter()
137            .zip(&right_pc)
138            .all(|(&left, &right)| near(left, right, TOLERANCE))
139}

141fn coordinate_near(
142    left: &CoordinateModel,
143    left_record: &RecordValue,
144    right: &CoordinateModel,
145    right_record: &RecordValue,
146) -> bool {
147    if let (CoordinateModel::Direction(left), CoordinateModel::Direction(right)) = (left, right) {
148        return directions_near(left, right);
149    }
150    let descriptors_near = if matches!(left, CoordinateModel::Linear(_)) {
151        linear_labels_near(&left.axis_names(), &right.axis_names())
152            && linear_labels_near(&left.axis_units(), &right.axis_units())
153    } else {
154        left.axis_names() == right.axis_names() && left.axis_units() == right.axis_units()
155    };
156    if !descriptors_near {
157        return false;
158    }
159    let left_canonical = left.to_record();
160    let right_canonical = right.to_record();
161    // Casacore compares the number of Stokes values without comparing their labels.
162    if let (CoordinateModel::Stokes(left), CoordinateModel::Stokes(right)) = (left, right) {
163        return left.stokes().len() == right.stokes().len();
164    }
165    let reference_tolerance = if matches!(left, CoordinateModel::Spectral(_)) {
166        WCS_TOLERANCE
167    } else {
168        TOLERANCE
169    };
170    if !near_values(
171        &left.reference_value(),
172        &right.reference_value(),
173        reference_tolerance,
174    ) {
175        return false;
176    }
177    let single_tabular_channel = matches!(left, CoordinateModel::Spectral(_))
178        && matches!(
179            subrecord(left_record, "tabular").and_then(|table| table.get("pixelvalues")),
180            Some(Value::Array(ArrayValue::Float64(values))) if values.len() == 1
181        );
182    if !single_tabular_channel
183        && (!near_values(&left.reference_pixel(), &right.reference_pixel(), TOLERANCE)
184            || !near_values(&left.increment(), &right.increment(), TOLERANCE)
185            || !near_matrix(
186                pc_matrix(left_record).or_else(|| pc_matrix(&left_canonical)),
187                pc_matrix(right_record).or_else(|| pc_matrix(&right_canonical)),
188            ))
189    {
190        return false;
191    }
192    match (left, right) {
193        (CoordinateModel::Spectral(left), CoordinateModel::Spectral(right)) => {
194            left.frequency_ref() == right.frequency_ref()
195                && left.has_undefined_frame() == right.has_undefined_frame()
196                && left.world_frequency_ref() == right.world_frequency_ref()
197                && near(left.rest_frequency(), right.rest_frequency(), TOLERANCE)
198                && near_values(left.rest_frequencies(), right.rest_frequencies(), TOLERANCE)
199                && scalar(velocity_field(
200                    left_record,
201                    &left_canonical,
202                    "velType",
203                    "prefVelType",
204                )) == scalar(velocity_field(
205                    right_record,
206                    &right_canonical,
207                    "velType",
208                    "prefVelType",
209                ))
210                && velocity_field(left_record, &left_canonical, "velUnit", "prefVelUnit")
211                    == velocity_field(right_record, &right_canonical, "velUnit", "prefVelUnit")
212        }
213        (CoordinateModel::Tabular(left), CoordinateModel::Tabular(right)) => {
214            near_values(left.pixel_values(), right.pixel_values(), TOLERANCE)
215                && near_values(left.world_values(), right.world_values(), TOLERANCE)
216        }
217        (CoordinateModel::Linear(_), CoordinateModel::Linear(_)) => true,
218        _ => false,
219    }
220}

222struct PlacedAxis {
223    coordinate: RecordCoordinate,
224    model: CoordinateModel,
225    world_axes: Vec<i64>,
226}
227
228fn placements(record: &RecordValue) -> PyResult<Vec<PlacedAxis>> {
229    let declared = validate_coordinate_indices(record)?;
230    let mut layout = CoordinateSystem::record_coordinates(record)
231        .map_err(raised)?
232        .into_iter()
233        .map(|coordinate| {
234            let number = coordinate
235                .field
236                .trim_start_matches(|character: char| !character.is_ascii_digit())
237                .parse::<usize>()
238                .expect("numbered coordinate field");
239            (number, coordinate)
240        })
241        .collect::<Vec<_>>();
242    if layout.len() != declared {
243        return Err(raised(
244            "Coordinate conformance for this record is unfinished (#1002)",
245        ));
246    }
247    layout.sort_unstable_by_key(|(number, _)| *number);
248    let mut next_world = 0;
249    layout
250        .into_iter()
251        .map(|(number, coordinate)| {
252            let model = coordinate.model(record).map_err(raised)?;
253            let axes = model.n_world_axes();
254            let world_axes: Vec<_> = match record.get(&format!("worldmap{number}")) {
255                Some(Value::Array(ArrayValue::Int32(values))) => {
256                    values.iter().map(|&value| i64::from(value)).collect()
257                }
258                Some(Value::Array(ArrayValue::Int64(values))) => values.iter().copied().collect(),
259                None => (next_world..next_world + axes as i64).collect(),
260                _ => return Err(raised("Invalid image coordinate world-axis map")),
261            };
262            if world_axes.len() != axes {
263                return Err(raised("Invalid image coordinate world-axis map"));
264            }
265            next_world += axes as i64;
266            Ok(PlacedAxis {
267                coordinate,
268                model,
269                world_axes,
270            })
271        })
272        .collect()
273}

275pub(super) fn coordinates_conform(
276    left: &RecordValue,
277    right: &RecordValue,
278    exact_axis_names: bool,
279) -> PyResult<bool> {
280    let left_system = CoordinateSystem::from_record(left).map_err(raised)?;
281    let right_system = CoordinateSystem::from_record(right).map_err(raised)?;
282    if left_system.n_coordinates() != right_system.n_coordinates()
283        || left_system.n_pixel_axes() != right_system.n_pixel_axes()
284        || left_system.n_world_axes() != right_system.n_world_axes()
285    {
286        return Ok(false);
287    }
288    let left_layout = placements(left)?;
289    let right_layout = placements(right)?;
290    for (lhs, rhs) in left_layout.iter().zip(&right_layout) {
291        if lhs.model.coordinate_type() != rhs.model.coordinate_type()
292            || lhs.coordinate.pixel_axes != rhs.coordinate.pixel_axes
293            || lhs.world_axes != rhs.world_axes
294        {
295            return Ok(false);
296        }
297        if exact_axis_names {
298            let left_names = lhs.model.axis_names();
299            let right_names = rhs.model.axis_names();
300            if lhs
301                .world_axes
302                .iter()
303                .enumerate()
304                .any(|(index, &axis)| axis >= 0 && left_names.get(index) != right_names.get(index))
305            {
306                return Ok(false);
307            }
308        }
309        if lhs.world_axes.iter().any(|&axis| axis >= 0) {
310            let left_record = subrecord(left, &lhs.coordinate.field).expect("parsed coordinate");
311            let right_record = subrecord(right, &rhs.coordinate.field).expect("parsed coordinate");
312            if !coordinate_near(&lhs.model, left_record, &rhs.model, right_record) {
313                return Ok(false);
314            }
315        }
316    }
317    Ok(true)
318}

320#[pymethods]
321impl PyImageTool {
322    fn isconform(&self, py: Python<'_>, other: &str) -> PyResult<bool> {
323        py.detach(|| {
324            let state = self.state();
325            let Some(attachment) = state.image.as_ref() else {
326                return Ok(false);
327            };
328            if attachment.lock().kind() != ImagePixelType::Float32 {
329                return Err(raised(
330                    "Exception: This method only supports Float valued images.",
331                ));
332            }
333            let path = Path::new(other);
334            if other.is_empty() {
335                return Err(raised("Exception: File name is empty."));
336            }
337            if !path.exists() {
338                return Err(raised(format!("Exception: File {other} does not exist..")));
339            }
340            let other = AttachedImage::open(path)
341                .map_err(|_| raised(format!("Exception: Unable to open image {other}.")))?;
342            let (shape, coordinates) = {
343                let image = attachment.lock();
344                (image.shape().to_vec(), image.coordinate_record())
345            };
346            let other = other.lock();
347            if !matches!(
348                other.kind(),
349                ImagePixelType::Float32 | ImagePixelType::Complex32
350            ) {
351                return Err(raised("Exception: Unsupported image pixel data type."));
352            }
353            if shape != other.shape() {
354                return Ok(false);
355            }
356            coordinates_conform(&coordinates, &other.coordinate_record(), true)
357        })
358    }
