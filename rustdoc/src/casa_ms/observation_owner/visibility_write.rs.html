48/// Bounded in-place selected-visibility writer following ordinary casacore semantics.
49///
50/// The writer holds the MAIN write lock until it completes. Completion flushes
51/// every selected destination and releases the lock. A failure may leave
52/// partially written derived values, as an interrupted CASA write does; the
53/// next run recomputes them. No backup, staging column, rollback, snapshot,
54/// marker, or content digest is created.
55///
56/// Only destinations stored in `TiledShapeStMan` or `TiledColumnStMan`, the
57/// managers CASA and casa-rs store visibilities in, are written: their save
58/// patches the written rows' tiles in place. Any other manager's save
59/// rewrites its whole group from loaded cells, which holds every row of
60/// MAIN (ADR-0008), so [`retained_bytes`](Self::retained_bytes) and
61/// [`begin`](Self::begin) refuse such a destination before anything is
62/// changed.
63#[cfg(unix)]
64pub struct SelectedVisibilityWrite {
65    measurement_set: Option<MeasurementSet>,
66    targets: SelectedVisibilityWriteTargets,
67    pending_cell: Option<PendingVisibilityCells>,
68    pending_rows: Vec<usize>,
69    completed: bool,
70}
71
72struct PendingVisibilityCells {
73    row: usize,
74    model_data: Option<ndarray::ArrayD<Complex32>>,
75    corrected_data: Option<ndarray::ArrayD<Complex32>>,
76}
77
78#[cfg(unix)]
79impl SelectedVisibilityWrite {
80    /// Take the MAIN write lock and start an in-place write. `MODEL_DATA` is
81    /// created, zero-filled, when MAIN lacks it, as CASA does.
82    ///
83    /// The MeasurementSet is opened with retained read locks, and MAIN's is
84    /// then upgraded to the write lock. Both wait, as casacore's do, for a
85    /// conflicting lock another process holds. When another process holding
86    /// a read lock waits to upgrade it too, the kernel refuses one of the
87    /// two waits; if it is this one, the write fails with
88    /// [`ObservationOwnerError::WriteLockUnavailable`] and releases its read
89    /// locks.
90    ///
91    /// A destination that cannot be written in bounded memory fails with
92    /// [`ObservationOwnerError::UnboundedWriteDestination`] before the lock
93    /// is taken.
94    pub fn begin(
95        path: impl AsRef<Path>,
96        targets: SelectedVisibilityWriteTargets,
97    ) -> Result<Self, ObservationOwnerError> {
98        if !targets.model_data && !targets.corrected_data {
99            return Err(ObservationOwnerError::EmptyWriteTargets);
100        }
101        let mut measurement_set = MeasurementSet::open_retained_read(path.as_ref())?;
102        require_bounded_destinations(measurement_set.main_table(), targets)?;
103        if !measurement_set.main_table_mut().lock(LockType::Write, 0)? {
104            return Err(ObservationOwnerError::WriteLockUnavailable);
105        }
106        let has_model = {
107            let schema = measurement_set.main_table().schema().ok_or_else(|| {
108                MsError::InvalidInput("MeasurementSet MAIN has no schema".to_string())
109            })?;
110            schema.contains_column("MODEL_DATA")
111        };
112        if targets.model_data && !has_model {
113            measurement_set.main_table_mut().add_column(
114                ColumnSchema::array_variable("MODEL_DATA", PrimitiveType::Complex32, Some(2)),
115                None,
116            )?;
117            measurement_set
118                .main_table_mut()
119                .prepare_write()
120                .add_tiled_column_clone("DATA", "MODEL_DATA", "TiledModelData")?;
121            let row_count = measurement_set.main_table().row_count();
122            let batch = VISIBILITY_WRITE_BATCH_ROWS as usize;
123            for start in (0..row_count).step_by(batch) {
124                let rows: Vec<usize> = (start..row_count.min(start + batch)).collect();
125                let descriptions = main_data_description_ids(&measurement_set, &rows)?;
126                for (&row, description) in rows.iter().zip(descriptions) {
127                    let shape = model_cell_shape(&measurement_set, description)?;
128                    queue_visibility_cell(
129                        measurement_set.main_table_mut(),
130                        "MODEL_DATA",
131                        row,
132                        ArrayValue::Complex32(ndarray::ArrayD::from_elem(
133                            ndarray::IxDyn(&shape),
134                            Complex32::new(0.0, 0.0),
135                        )),
136                    )?;
137                }
138                persist_visibility_rows(measurement_set.main_table_mut(), &["MODEL_DATA"], &rows)?;
139            }
140        }
141        Ok(Self {
142            measurement_set: Some(measurement_set),
143            targets,
144            pending_cell: None,
145            pending_rows: Vec::with_capacity(VISIBILITY_WRITE_BATCH_ROWS as usize),
146            completed: false,
147        })
148    }
149
150    /// Heap bytes a write of `targets` to the MeasurementSet at `path`
151    /// holds at most, which the pass that writes admits before it begins.
152    ///
153    /// Per written column, that is a batch of queued cells of the
154    /// MeasurementSet's largest cell, the queue's list of them, the row
155    /// being written, and, while a batch is saved, the batch's cells twice
156    /// more (gathered for the save and staged by tile) with the column's tile
157    /// read and its tile written; the list holds its old and new buffers at
158    /// once while it grows. Beside them are the batch's row numbers and, while a missing
159    /// `MODEL_DATA` is created, their data-description ids as read and as
160    /// indices. Tiles cached for reuse go to the process-wide table cache,
161    /// charged on its own.
162    ///
163    /// # Errors
164    ///
165    /// When the MeasurementSet's spectral-window, polarization or table
166    /// descriptions cannot be read, and, as [`begin`](Self::begin) would,
167    /// when a destination cannot be written in bounded memory or
168    /// `CORRECTED_DATA` is missing: the pass that writes then fails before
169    /// it runs.
170    pub fn retained_bytes(
171        path: impl AsRef<Path>,
172        targets: SelectedVisibilityWriteTargets,
173    ) -> Result<u64, ObservationOwnerError> {
174        /// Staging per batch row: the save's sorted row list and the cells
175        /// it gathers by tile; measured at about 140 bytes.
176        const STAGING_BYTES_PER_ROW: u64 = 192;
177        /// Staging whatever the batch: the MeasurementSet opened for the
178        /// write and the table description each save reads; measured at
179        /// about 240 KB.
180        const STAGING_BYTES: u64 = 1 << 20;
181        let measurement_set = MeasurementSet::open(path.as_ref())?;
182        let mut cell_values = 0;
183        for description in 0..measurement_set.data_description()?.row_count() {
184            let [correlations, channels] = model_cell_shape(&measurement_set, description)?;
185            cell_values = cell_values.max(correlations * channels);
186        }
187        let main = measurement_set.main_table();
188        require_bounded_destinations(main, targets)?;
189        let tile = |column: &'static str| -> Result<u64, ObservationOwnerError> {
190            // A created MODEL_DATA clones DATA's tiled manager.
191            let stored = if column == "MODEL_DATA"
192                && !main
193                    .schema()
194                    .is_some_and(|schema| schema.contains_column(column))
195            {
196                "DATA"
197            } else {
198                column
199            };
200            let bytes = main.array_column_tile_bytes(stored)?.ok_or_else(|| {
201                ObservationOwnerError::UnboundedWriteDestination {
202                    column,
203                    reason: format!("{stored} has no tiles on disk"),
204                }
205            })?;
206            Ok(bytes as u64)
207        };
208        let mut tiles = 0;
209        if targets.model_data {
210            tiles += 2 * tile("MODEL_DATA")?;
211        }
212        if targets.corrected_data {
213            tiles += 2 * tile("CORRECTED_DATA")?;
214        }
215        let cell = (cell_values * size_of::<Complex32>()) as u64;
216        let batch = VISIBILITY_WRITE_BATCH_ROWS;
217        let entry = size_of::<(usize, ArrayValue)>() as u64;
218        let queue = 3 * batch * cell + batch.next_power_of_two() * entry * 3 / 2;
219        let columns = u64::from(targets.model_data) + u64::from(targets.corrected_data);
220        let rows = batch * (size_of::<usize>() as u64 + STAGING_BYTES_PER_ROW);
221        let creation = if targets.model_data {
222            batch * (size_of::<Option<ScalarValue>>() + size_of::<usize>()) as u64
223        } else {
224            0
225        };
226        Ok(columns * (queue + cell) + tiles + rows + creation + STAGING_BYTES)
227    }

229    /// Write one selected prediction at its physical row/channel/correlation.
230    pub fn write(
231        &mut self,
232        column: MsColumnKind,
233        row: u64,
234        channel: u32,
235        correlation: u32,
236        value: Complex32,
237    ) -> Result<(), ObservationOwnerError> {
238        let row = usize::try_from(row).map_err(|_| ObservationOwnerError::PredictionAddress)?;
239        if self.pending_cell.as_ref().map(|cell| cell.row) != Some(row) {
240            self.flush_pending_cell()?;
241            let measurement_set = self
242                .measurement_set
243                .as_ref()
244                .ok_or(ObservationOwnerError::TransactionClosed)?;
245            // The row's one cell, read past the table's row cache: a cached
246            // read would load every row of MAIN (ADR-0008).
247            let load = |name: &str| -> Result<ndarray::ArrayD<Complex32>, ObservationOwnerError> {
248                let current = measurement_set
249                    .main_table()
250                    .column_accessor(name)?
251                    .array_cells_owned_uncached(&[row])?
252                    .pop()
253                    .flatten()
254                    .ok_or(ObservationOwnerError::PredictionAddress)?;
255                let ArrayValue::Complex32(values) = current else {
256                    return Err(ObservationOwnerError::PredictionAddress);
257                };
258                Ok(values)
259            };
260            self.pending_cell = Some(PendingVisibilityCells {
261                row,
262                model_data: self
263                    .targets
264                    .model_data
265                    .then(|| load("MODEL_DATA"))
266                    .transpose()?,
267                corrected_data: self
268                    .targets
269                    .corrected_data
270                    .then(|| load("CORRECTED_DATA"))
271                    .transpose()?,
272            });
273        }
274        let pending = self
275            .pending_cell
276            .as_mut()
277            .ok_or(ObservationOwnerError::TransactionClosed)?;
278        let values = match column {
279            MsColumnKind::ModelData if self.targets.model_data => pending.model_data.as_mut(),
280            MsColumnKind::CorrectedData if self.targets.corrected_data => {
281                pending.corrected_data.as_mut()
282            }
283            _ => None,
284        }
285        .ok_or(ObservationOwnerError::UnselectedWriteTarget)?;
286        let index = [correlation as usize, channel as usize];
287        let Some(cell) = values.get_mut(index) else {
288            return Err(ObservationOwnerError::PredictionAddress);
289        };
290        *cell = value;
291        Ok(())
292    }
293
294    /// Flush the in-place write and release the MAIN write lock.
295    ///
296    /// The selected cells were persisted in bounded row batches, so only the
297    /// table metadata (a newly created `MODEL_DATA`) remains to flush.
298    pub fn complete(mut self) -> Result<(), ObservationOwnerError> {
299        self.flush_pending_cell()?;
300        self.persist_pending_rows()?;
301        let measurement_set = self
302            .measurement_set
303            .as_mut()
304            .ok_or(ObservationOwnerError::TransactionClosed)?;
305        measurement_set.main_table_mut().unlock_metadata_only()?;
306        self.measurement_set = None;
307        self.completed = true;
308        Ok(())
309    }
310
311    fn flush_pending_cell(&mut self) -> Result<(), ObservationOwnerError> {
312        let Some(cell) = self.pending_cell.take() else {
313            return Ok(());
314        };
315        let measurement_set = self
316            .measurement_set
317            .as_mut()
318            .ok_or(ObservationOwnerError::TransactionClosed)?;
319        if let Some(values) = cell.model_data {
320            queue_visibility_cell(
321                measurement_set.main_table_mut(),
322                "MODEL_DATA",
323                cell.row,
324                ArrayValue::Complex32(values),
325            )?;
326        }
327        if let Some(values) = cell.corrected_data {
328            queue_visibility_cell(
329                measurement_set.main_table_mut(),
330                "CORRECTED_DATA",
331                cell.row,
332                ArrayValue::Complex32(values),
333            )?;
334        }
335        self.pending_rows.push(cell.row);
336        if self.pending_rows.len() == VISIBILITY_WRITE_BATCH_ROWS as usize {
337            self.persist_pending_rows()?;
338        }
339        Ok(())
340    }
341
342    fn persist_pending_rows(&mut self) -> Result<(), ObservationOwnerError> {
343        if self.pending_rows.is_empty() {
344            return Ok(());
345        }
346        let columns = match (self.targets.model_data, self.targets.corrected_data) {
347            (true, true) => &["MODEL_DATA", "CORRECTED_DATA"][..],
348            (true, false) => &["MODEL_DATA"][..],
349            (false, true) => &["CORRECTED_DATA"][..],
350            (false, false) => return Err(ObservationOwnerError::EmptyWriteTargets),
351        };
352        let measurement_set = self
353            .measurement_set
354            .as_mut()
355            .ok_or(ObservationOwnerError::TransactionClosed)?;
356        persist_visibility_rows(
357            measurement_set.main_table_mut(),
358            columns,
359            &self.pending_rows,
360        )?;
361        self.pending_rows.clear();
362        Ok(())
363    }
364}

366#[cfg(unix)]
367impl Drop for SelectedVisibilityWrite {
368    /// An abandoned write releases the MAIN write lock without flushing the
369    /// cells still queued: the batches persisted so far stay, as after an
370    /// interrupted CASA write, and MAIN is not rewritten.
371    fn drop(&mut self) {
372        if !self.completed {
373            if let Some(measurement_set) = self.measurement_set.as_mut() {
374                let _ = measurement_set.main_table_mut().unlock_metadata_only();
375            }
376            self.measurement_set = None;
377        }
378    }
379}

381/// Storage managers whose selected-row save patches the written rows' tiles
382/// in place: the managers CASA and casa-rs store visibilities in.
383#[cfg(unix)]
384const BOUNDED_WRITE_MANAGERS: [&str; 2] = ["TiledShapeStMan", "TiledColumnStMan"];
385
386/// Refuse, before anything is changed, a write whose save would not be
387/// bounded: an existing destination stored outside
388/// [`BOUNDED_WRITE_MANAGERS`], or a missing `MODEL_DATA` whose clone of
389/// DATA's manager would be or cannot be made. A missing `CORRECTED_DATA` is
390/// refused too, since it holds the output of a continuum transform.
391#[cfg(unix)]
392fn require_bounded_destinations(
393    main: &Table,
394    targets: SelectedVisibilityWriteTargets,
395) -> Result<(), ObservationOwnerError> {
396    let has = |column: &str| {
397        main.schema()
398            .is_some_and(|schema| schema.contains_column(column))
399    };
400    if targets.corrected_data && !has("CORRECTED_DATA") {
401        return Err(ObservationOwnerError::MissingCorrectedDataDestination);
402    }
403    let destinations = [
404        ("MODEL_DATA", targets.model_data),
405        ("CORRECTED_DATA", targets.corrected_data),
406    ];
407    for (column, written) in destinations {
408        if !written {
409            continue;
410        }
411        let created = !has(column);
412        let stored = if created { "DATA" } else { column };
413        let Some(manager) = main
414            .data_manager_info()
415            .iter()
416            .find(|manager| manager.columns.iter().any(|name| name == stored))
417        else {
418            return Err(ObservationOwnerError::UnboundedWriteDestination {
419                column,
420                reason: format!("{stored} has no storage manager on disk"),
421            });
422        };
423        let storage = if created {
424            format!(
425                "it would be created in a clone of DATA's {}",
426                manager.dm_type
427            )
428        } else {
429            format!("it is stored in {}", manager.dm_type)
430        };
431        if !BOUNDED_WRITE_MANAGERS.contains(&manager.dm_type.as_str()) {
432            return Err(ObservationOwnerError::UnboundedWriteDestination {
433                column,
434                reason: format!(
435                    "{storage}; only TiledShapeStMan and TiledColumnStMan columns are written in place"
436                ),
437            });
438        }
439        if created && manager.columns.len() != 1 {
440            return Err(ObservationOwnerError::UnboundedWriteDestination {
441                column,
442                reason: format!("{storage}, which DATA shares with other columns"),
443            });
444        }
445    }
446    Ok(())
447}

449/// The `DATA_DESC_ID` of each of `rows`, read without filling the table's
450/// row cache.
451fn main_data_description_ids(
452    measurement_set: &MeasurementSet,
453    rows: &[usize],
454) -> Result<Vec<usize>, ObservationOwnerError> {
455    measurement_set
456        .main_table()
457        .column_accessor("DATA_DESC_ID")?
458        .scalar_cells_owned_for_rows(rows)?
459        .into_iter()
460        .map(|value| match value {
461            Some(ScalarValue::Int32(id)) => {
462                usize::try_from(id).map_err(|_| ObservationOwnerError::PredictionAddress)
463            }
464            _ => Err(ObservationOwnerError::PredictionAddress),
465        })
466        .collect()
467}

469fn model_cell_shape(
470    measurement_set: &MeasurementSet,
471    data_description_id: usize,
472) -> Result<[usize; 2], ObservationOwnerError> {
473    let data_description = measurement_set.data_description()?;
474    let spectral_window_id =
475        usize::try_from(data_description.spectral_window_id(data_description_id)?)
476            .map_err(|_| ObservationOwnerError::PredictionAddress)?;
477    let polarization_id = usize::try_from(data_description.polarization_id(data_description_id)?)
478        .map_err(|_| ObservationOwnerError::PredictionAddress)?;
479    let channels = usize::try_from(
480        measurement_set
481            .spectral_window()?
482            .num_chan(spectral_window_id)?,
483    )
484    .map_err(|_| ObservationOwnerError::PredictionAddress)?;
485    let correlations = usize::try_from(measurement_set.polarization()?.num_corr(polarization_id)?)
486        .map_err(|_| ObservationOwnerError::PredictionAddress)?;
487    Ok([correlations, channels])
488}

490#[cfg(unix)]
491fn queue_visibility_cell(
492    table: &mut Table,
493    column: &str,
494    row: usize,
495    value: ArrayValue,
496) -> Result<(), ObservationOwnerError> {
497    {
498        let mut prepared = table.row_accessor_mut().prepare(&[column])?;
499        prepared.seek(row)?;
500        prepared.set_value_at(0, Value::Array(value))?;
501    }
502    Ok(())
503}
504
505#[cfg(unix)]
506fn persist_visibility_rows(
507    table: &mut Table,
508    columns: &[&str],
509    rows: &[usize],
510) -> Result<(), ObservationOwnerError> {
511    table.prepare_write().save_selected_rows(columns, rows)?;
512    table.discard_persisted_cell_updates(columns, rows);
513    Ok(())
