12fn main() -> ExitCode {
13    image_task_front_end("immoments", &["immoments"], run).main(
14        std::env::args_os().skip(1),
15        casa_task_runtime::CasaLoggingConfig::from_env(),
16        std::env::set_current_dir,
17        &mut std::io::stdout(),
18        std::io::stderr(),
19    )
20}
21
22fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
23    let request = ImmomentsRequest {
24        imagename: task.input_path("imagename")?,
25        outfile: task.path("outfile")?,
26        moments: one_moment(&task.integers("moments")?)?,
27        chans: task.text("chans")?,
28        includepix: include_range(&task.numbers("includepix")?)?,
29        mask: task.text("mask")?,
30        overwrite: task.flag("overwrite")?,
31    };
32    image_task_success(immoments(&request).map(ImageAnalysisTaskResult::Immoments))
33}

35/// The one moment casa-rs computes per run: CASA's `moments` takes several,
36/// each written beside `outfile`, which is #977.
37fn one_moment(moments: &[i64]) -> Result<i32, TaskFailure> {
38    let refused = |message: String| TaskFailure::validation(message).at("moments");
39    match moments {
40        [moment @ -1..=3] => Ok(*moment as i32),
41        [moment] => Err(refused(format!(
42            "moment {moment} is not implemented by casa-rs yet (#977); it computes -1 through 3"
43        ))),
44        _ => Err(refused(format!(
45            "{} moments in one run are not implemented by casa-rs yet (#977); give one",
46            moments.len()
47        ))),
48    }
49}

51/// CASA's `includepix`, an inclusive range of the pixel values a moment
52/// reads: `-1` (or `[]`) includes every pixel; one value `b` is
53/// `[-|b|, |b|]`; two values are the range's ends, in either order
54/// (casacore's `MomentsBase::setInExCludeRange`; CASA 6.7.6.14 runs
55/// `includepix=-2` as `[-2, 2]` and `[5, 2]` as `[2, 5]`).
56fn include_range(values: &[f64]) -> Result<Option<[f64; 2]>, TaskFailure> {
57    match *values {
58        [] => Ok(None),
59        [-1.0] => Ok(None),
60        [value] => Ok(Some([-value.abs(), value.abs()])),
61        [a, b] => Ok(Some([a.min(b), a.max(b)])),
62        _ => Err(TaskFailure::validation(format!(
63            "includepix {values:?}: give -1 for every pixel, one value b for [-|b|, |b|], or two \
64             for a range, as CASA takes it"
65        ))
66        .at("includepix")),
67    }
68}

70#[cfg(test)]
71mod tests {
72    use super::*;
73
74    #[test]
75    fn one_moment_from_minus_one_to_three_is_computed() {
76        assert_eq!(one_moment(&[0]).unwrap(), 0);
77        assert_eq!(one_moment(&[-1]).unwrap(), -1);
78        assert!(one_moment(&[8]).unwrap_err().message.contains("#977"));
79        assert!(one_moment(&[0, 1]).unwrap_err().message.contains("#977"));
80    }
81
82    #[test]
83    fn includepix_reads_as_casa_does() {
84        // Only -1 is CASA's "every pixel".
85        assert_eq!(include_range(&[]).unwrap(), None);
86        assert_eq!(include_range(&[-1.0]).unwrap(), None);
87        // Any other single value b is [-|b|, |b|].
88        assert_eq!(include_range(&[-2.0]).unwrap(), Some([-2.0, 2.0]));
89        assert_eq!(include_range(&[2.0]).unwrap(), Some([-2.0, 2.0]));
90        assert_eq!(include_range(&[-1.5]).unwrap(), Some([-1.5, 1.5]));
91        assert_eq!(include_range(&[0.0]).unwrap(), Some([0.0, 0.0]));
92        // Two values are the ends, in either order.
93        assert_eq!(include_range(&[0.1, 10.0]).unwrap(), Some([0.1, 10.0]));
94        assert_eq!(include_range(&[5.0, 2.0]).unwrap(), Some([2.0, 5.0]));
95        assert!(include_range(&[1.0, 2.0, 3.0]).is_err());
96    }
97}
