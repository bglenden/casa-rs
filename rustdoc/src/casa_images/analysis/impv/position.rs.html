16pub(super) fn endpoints(
17    request: &ImpvRequest,
18    coordinate: &DirectionCoordinate,
19    allow_pixel_shorthand: bool,
20) -> Result<[[f64; 2]; 2], ImageError> {
21    let missing = |message: &str| ImageError::CasaSelection(message.into());
22    match request
23        .mode
24        .trim()
25        .as_bytes()
26        .first()
27        .map(u8::to_ascii_lowercase)
28    {
29        Some(b'c') => {
30            let (Some(start), Some(end)) = (&request.start, &request.end) else {
31                return Err(missing("coords mode requires start and end"));
32            };
33            if std::mem::discriminant(start) != std::mem::discriminant(end) {
34                return Err(missing("start and end must use the same representation"));
35            }
36            Ok([
37                to_pixel(start, coordinate, allow_pixel_shorthand)?,
38                to_pixel(end, coordinate, allow_pixel_shorthand)?,
39            ])
40        }
41        Some(b'l') => {
42            let (Some(center), Some(length), Some(pa)) =
43                (&request.center, &request.length, &request.pa)
44            else {
45                return Err(missing("length mode requires center, length and pa"));
46            };
47            let center = parse_position(center, allow_pixel_shorthand)?;
48            let length = match length {
49                PvAmount::Whole(n) => pixel_length(*n as f64, &center, coordinate)?,
50                PvAmount::Fractional(n) => pixel_length(*n, &center, coordinate)?,
51                value => radians(value)?,
52            };
53            let center = match center {
54                Position::Pixel(pixel) => {
55                    let [lon, lat] = coordinate.pixel_to_world(pixel)?;
56                    MDirection::from_angles(lon, lat, coordinate.conversion_direction_ref())
57                }
58                Position::Sky(center) => center,
59            };
60            let (lon, lat) = center.as_angles();
61            let pa = radians(pa)?;
62            let shift = |angle: f64| {
63                let (s, c) = (length / 2.0).sin_cos();
64                let latitude = (lat.sin() * c + lat.cos() * s * angle.cos())
65                    .clamp(-1.0, 1.0)
66                    .asin();
67                let longitude =
68                    lon + (angle.sin() * s * lat.cos()).atan2(c - lat.sin() * latitude.sin());
69                sky_to_pixel(
70                    &MDirection::from_angles(longitude, latitude, center.refer()),
71                    coordinate,
72                )
73            };
74            Ok([shift(pa)?, shift(pa - PI)?])
75        }
76        _ => Err(missing("Unsupported value for mode")),
77    }
78}

80fn pixel_length(
81    length: f64,
82    center: &Position,
83    coordinate: &DirectionCoordinate,
84) -> Result<f64, ImageError> {
85    if !length.is_finite() || (length <= 0.0 && matches!(center, Position::Pixel(_))) {
86        return Err(ImageError::CasaSelection("Length must be positive".into()));
87    }
88    Ok(length * coordinate.increment()[0].abs())
89}

91pub(super) fn width(
92    value: &PvAmount,
93    coordinate: &DirectionCoordinate,
94) -> Result<usize, ImageError> {
95    let width = match value {
96        PvAmount::Whole(n) => *n,
97        PvAmount::Fractional(_) => 0,
98        value => {
99            let pixels = radians(value)? / coordinate.increment()[0].abs();
100            let nearest = pixels.round();
101            let round_up = if nearest.rem_euclid(2.0) == 1.0 && (pixels - nearest).abs() <= 1e-13 {
102                nearest
103            } else {
104                pixels.ceil()
105            };
106            let round_up = round_up.max(1.0) as i64;
107            if round_up % 2 == 0 {
108                round_up + 1
109            } else {
110                round_up
111            }
112        }
113    };
114    usize::try_from(width)
115        .ok()
116        .filter(|n| n % 2 == 1)
117        .ok_or_else(|| ImageError::CasaSelection("width must be an odd integer >= 1".into()))
118}

120fn radians(value: &PvAmount) -> Result<f64, ImageError> {
121    let invalid = || ImageError::CasaSelection("expected a value with angular units".into());
122    let quantity = match value {
123        PvAmount::Angular(text) => Quantity::read(text).map_err(|_| invalid())?,
124        PvAmount::Record(record) => crate::beam::quantity_from_record(record)?,
125        _ => return Err(invalid()),
126    };
127    let radians = quantity
128        .get_value_in(&Unit::new("rad").expect("radians are built in"))
129        .map_err(|_| invalid())?;
130    if !radians.is_finite() {
131        return Err(invalid());
132    }
133    Ok(radians)
134}

136fn to_pixel(
137    position: &PvPosition,
138    coordinate: &DirectionCoordinate,
139    allow_pixel_shorthand: bool,
140) -> Result<[f64; 2], ImageError> {
141    match parse_position(position, allow_pixel_shorthand)? {
142        Position::Pixel(pixel) => Ok(pixel),
143        Position::Sky(direction) => sky_to_pixel(&direction, coordinate),
144    }
145}

147enum Position {
148    Pixel([f64; 2]),
149    Sky(MDirection),
150}
151
152fn parse_position(
153    position: &PvPosition,
154    allow_pixel_shorthand: bool,
155) -> Result<Position, ImageError> {
156    let invalid =
157        || ImageError::CasaSelection("expected two pixel coordinates or a sky direction".into());
158    let (frame, [longitude, latitude]) = match position {
159        PvPosition::Integral(pixel) => return Ok(Position::Pixel(pixel.map(|value| value as f64))),
160        PvPosition::Pixel(pixel) => return Ok(Position::Pixel(*pixel)),
161        PvPosition::World([lon, lat]) => (DirectionRef::J2000, [lon.as_str(), lat.as_str()]),
162        PvPosition::Text(text) => {
163            let fields: Vec<_> = text
164                .split(|ch: char| ch == ',' || ch.is_whitespace())
165                .filter(|part| !part.is_empty())
166                .collect();
167            match fields.as_slice() {
168                [first, second] => {
169                    // Keep the native CLI's original comma-separated pixel shorthand.
170                    if allow_pixel_shorthand
171                        && text.contains(',')
172                        && let (Ok(x), Ok(y)) = (first.parse(), second.parse())
173                    {
174                        return Ok(Position::Pixel([x, y]));
175                    }
176                    (DirectionRef::J2000, [*first, *second])
177                }
178                [frame, lon, lat] => (frame.parse().map_err(|_| invalid())?, [*lon, *lat]),
179                _ => return Err(invalid()),
180            }
181        }
182    };
183    Ok(Position::Sky(MDirection::from_angles(
184        parse_world_angle(longitude)?,
185        parse_world_angle(latitude)?,
186        frame,
187    )))
188}

190fn sky_to_pixel(
191    sky: &MDirection,
192    coordinate: &DirectionCoordinate,
193) -> Result<[f64; 2], ImageError> {
194    let native = sky
195        .convert_to(coordinate.conversion_direction_ref(), &MeasFrame::default())
196        .map_err(|error| ImageError::CasaSelection(error.to_string()))?;
197    let (lon, lat) = native.as_angles();
198    Ok(coordinate.world_to_pixel([lon, lat])?)
199}
