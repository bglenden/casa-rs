19#[pymethods]
20impl PyMsTool {
21    #[pyo3(signature = (columns = None, interval = 0.0, maxrows = 0, adddefaultsortcolumns = true), text_signature = "($self, columns=[], interval=0.0, maxrows=0, adddefaultsortcolumns=True)")]
22    fn niterinit(
23        &self,
24        py: Python<'_>,
25        columns: Option<&Bound<'_, PyAny>>,
26        #[pyo3(from_py_with = interval_argument)] interval: f64,
27        #[pyo3(from_py_with = maxrows_argument)] maxrows: i64,
28        #[pyo3(from_py_with = strict_bool)] adddefaultsortcolumns: bool,
29    ) -> PyResult<bool> {
30        requested_strings(columns, "columns")?;
31        deprecation(py, "niterinit", "iterinit")?;
32        py.detach(|| {
33            let mut attached = self.attached.lock().map_err(runtime_error)?;
34            let ms = attached.as_mut().ok_or_else(empty_view)?;
35            ms.initialize_vi1(interval, maxrows, adddefaultsortcolumns)?;
36            Ok(true)
37        })
38    }
39
40    #[pyo3(signature = (), text_signature = "($self)")]
41    fn niterorigin(&self, py: Python<'_>) -> PyResult<bool> {
42        self.vi1_control(py, |iteration| {
43            iteration.rewind();
44            true
45        })
46    }
47
48    #[pyo3(signature = (), text_signature = "($self)")]
49    fn niterend(&self, py: Python<'_>) -> PyResult<bool> {
50        self.vi1_control(py, |iteration| iteration.exhausted())
51    }
52
53    #[pyo3(signature = (), text_signature = "($self)")]
54    fn niternext(&self, py: Python<'_>) -> PyResult<bool> {
55        self.vi1_control(py, Vi1ToolIteration::advance)
56    }
57
58    #[pyo3(signature = (items = None, ifraxis = false, ifraxisgap = 0, increment = 1, average = false), text_signature = "($self, items=[], ifraxis=False, ifraxisgap=0, increment=1, average=False)")]
59    fn ngetdata(
60        &self,
61        py: Python<'_>,
62        items: Option<&Bound<'_, PyAny>>,
63        #[pyo3(from_py_with = strict_bool)] ifraxis: bool,
64        #[pyo3(from_py_with = gap_argument)] ifraxisgap: i64,
65        #[pyo3(from_py_with = increment_argument)] increment: i64,
66        #[pyo3(from_py_with = strict_bool)] average: bool,
67    ) -> PyResult<Py<PyAny>> {
68        // CASA parses these four arguments but VI1 ignores them.
69        let _ = (ifraxis, ifraxisgap, increment, average);
70        let items = requested_strings(items, "items")?;
71        deprecation(py, "ngetdata", "getdata")?;
72        let result = py.detach(|| {
73            let mut attached = self.attached.lock().map_err(runtime_error)?;
74            let ms = attached.as_mut().ok_or_else(empty_view)?;
75            if ms.visibility.vi1.is_none() {
76                ms.initialize_vi1(0.0, 0, false)?;
77            }
78            let link = ms.working_metadata_link();
79            let vi1 = ms.visibility.vi1.as_ref().expect("VI1 was installed");
80            link.verify_directory().map_err(runtime_error)?;
81            vi1.main.verify_iterator_layout()?;
82            let result = vi1.main.read_vi1_rows(&vi1.cursor, &items)?;
83            link.verify_directory().map_err(runtime_error)?;
84            Ok::<_, PyErr>(result)
85        })?;
86        result.into_python(py)
87    }
88}

90impl PyMsTool {
91    fn vi1_control(
92        &self,
93        py: Python<'_>,
94        visit: impl FnOnce(&mut Vi1ToolIteration) -> bool + Send,
95    ) -> PyResult<bool> {
96        py.detach(|| {
97            let mut attached = self.attached.lock().map_err(runtime_error)?;
98            let Some(ms) = attached.as_mut() else {
99                return Ok(false);
100            };
101            let iteration = ms
102                .visibility
103                .vi1
104                .as_mut()
105                .ok_or_else(|| runtime_error("Please call niterinit() first."))?;
106            Ok(visit(&mut iteration.cursor))
107        })
108    }
109}
110
111impl AttachedMs {
112    fn initialize_vi1(&mut self, interval: f64, maxrows: i64, defaults: bool) -> PyResult<()> {
113        let link = self.working_metadata_link();
114        link.verify_directory().map_err(runtime_error)?;
115        let selected = link.selected_rows().map_err(runtime_error)?;
116        let main = self.visibility.current_main(&self.path)?;
117        let cursor = main.vi1_cursor(selected, interval, maxrows, defaults)?;
118        link.verify_directory().map_err(runtime_error)?;
119        self.visibility.vi1 = Some(Vi1Iteration { cursor, main });
120        Ok(())
121    }
122}
