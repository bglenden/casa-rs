39/// One owner's accumulator and backend.
40struct Owner<'w> {
41    acc: Option<GridAccumulator>,
42    backend: OwnerBackend<'w>,
43    images: Option<NormalImages>,
44    /// Time the owner spent accumulating in the wave, which a region pass
45    /// reports beside its strips (#736).
46    busy: std::time::Duration,
47    /// The weight image's contributions, summed before they are gridded
48    /// ([`WeightSums`], #787), when this owner grids the weight image: the
49    /// weight owner of a region pass, every owner of a plane pass.
50    weights: Option<WeightSums>,
51}

53impl Owner<'_> {
54    /// Wait for the owner's grid work still on the device, reporting
55    /// whether it failed.
56    fn complete(&mut self) -> Result<(), PassError> {
57        match &mut self.backend {
58            OwnerBackend::Cpu(_) => Ok(()),
59            OwnerBackend::Metal(backend) => Ok(backend.complete()?),
60        }
61    }
62
63    /// Grid owner `index`'s placements in `chunks`, of `target`'s domain,
64    /// in every mode `pass` holds, less what `subtract` names; and sum the
65    /// weight image's contributions over every placement when this owner
66    /// holds it (`weight_owner`; otherwise each owner sums its own
67    /// placements'). The sums are gridded when the table fills and at
68    /// [`Self::flush_weights`].
69    fn accumulate(
70        &mut self,
71        pass: &MajorCyclePass<'_>,
72        target: &PassDomain<'_>,
73        (index, weight_owner): (usize, Option<usize>),
74        subtract: Subtract<'_>,
75        chunks: &[Chunk],
76    ) -> Result<(), PassError> {
77        let started = std::time::Instant::now();
78        let Self {
79            acc,
80            backend,
81            weights,
82            busy,
83            ..
84        } = self;
85        let acc = acc.as_mut().expect("owners accumulate until finished");
86        let cf = target.operator.cf();
87        let own_weights = if weight_owner.is_none() {
88            weights.as_mut()
89        } else {
90            None
91        };
92        match &mut *backend {
93            OwnerBackend::Cpu(cpu) => {
94                for chunk in chunks {
95                    let block = chunk.owned[index].block();
96                    if !block.is_empty() {
97                        let mut apply = |work: Work<'_>| cpu.apply(&block, cf, work);
98                        accumulate(pass, &mut apply, subtract, acc)?;
99                    }
100                }
101            }
102            OwnerBackend::Metal(metal) => {
103                let parts = chunks
104                    .iter()
105                    .map(|chunk| chunk.owned[index].block())
106                    .collect::<Vec<_>>();
107                if parts.iter().any(|part| !part.is_empty()) {
108                    let mut apply = |work: Work<'_>| metal.apply_parts(&parts, cf, work);
109                    accumulate(pass, &mut apply, subtract, acc)?;
110                }
111            }
112        }
113        if let Some(sums) = own_weights {
114            for chunk in chunks {
115                let mut grid = weight_grid(backend, cf, acc);
116                sums.add(&chunk.owned[index].block(), &mut grid)?;
117            }
118        }
119        if weight_owner == Some(index)
120            && let Some(sums) = weights
121        {
122            for chunk in chunks {
123                let mut grid = weight_grid(backend, cf, acc);
124                sums.add(&chunk.weight_owned.block(), &mut grid)?;
125            }
126        }
127        *busy += started.elapsed();
128        Ok(())
129    }
130
131    /// Grid the weight image's summed contributions still in the owner's
132    /// table, of `target`'s domain ([`WeightSums::flush`]), and drop the
133    /// table: the wave's last placement is accumulated.
134    ///
135    /// The flush's time is the owner's ([`Self::busy`]). Its placements,
136    /// group grids and grids are returned for the wave's log. A panic while
137    /// gridding fails the pass with [`PassError::OwnerPanicked`], as an
138    /// owner's panic does in a pipelined pass.
139    fn flush_weights(&mut self, target: &PassDomain<'_>) -> Result<[u64; 3], PassError> {
140        let Some(mut sums) = self.weights.take() else {
141            return Ok([0; 3]);
142        };
143        let started = std::time::Instant::now();
144        let acc = self.acc.as_mut().expect("owners flush before they finish");
145        let backend = &mut self.backend;
146        let flushed = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
147            let mut grid = weight_grid(backend, target.operator.cf(), acc);
148            sums.flush(&mut grid)
149        }))
150        .unwrap_or(Err(PassError::OwnerPanicked));
151        self.busy += started.elapsed();
152        flushed?;
153        Ok(sums.totals())
154    }
155}

157/// Grid a block of summed weight-image contributions on `backend` into
158/// `acc` ([`WeightSums`]).
159fn weight_grid<'a>(
160    backend: &'a mut OwnerBackend<'_>,
161    cf: &'a dyn ConvolutionFunctionSet,
162    acc: &'a mut GridAccumulator,
163) -> impl FnMut(&SampleBlock<'_>) -> Result<(), PassError> + 'a {
164    move |groups: &SampleBlock<'_>| {
165        let work = Work::Grid {
166            mode: Mode::Weight,
167            acc: &mut *acc,
168        };
169        match backend {
170            OwnerBackend::Cpu(cpu) => Ok(cpu.apply(groups, cf, work)?),
171            OwnerBackend::Metal(metal) => Ok(metal.apply(groups, cf, work)?),
172        }
173    }
174}

176/// The backend an owner dispatches to. The CPU grids each placement chunk
177/// as it comes; a Metal owner dispatches its share of every chunk of a slice
178/// at once, read from the chunks' buffers ([`MetalBackend::apply_parts`]),
179/// so each device dispatch is a slice's worth of samples rather than one
180/// chunk's, and leaves it running while the workers place the next slice
181/// ([`MetalBackend::with_deferred_grid_completion`]).
182enum OwnerBackend<'w> {
183    Cpu(CpuBackend),
184    Metal(Box<MetalBackend<'w>>),
185}

198impl Domain<'_> {
199    /// Add the gridding work by anchor row `chunks` placed.
200    fn add_row_work(&mut self, chunks: &[Chunk]) {
201        if let Some(total) = &mut self.row_work {
202            for chunk in chunks {
203                for (total, work) in total.iter_mut().zip(&chunk.row_work) {
204                    *total += work;
205                }
206            }
207        }
208    }

211/// What placing one block's rows on one domain reads ([`Wave::placer`]).
212struct Placer<'b, 'w, 'p> {
213    pass: &'w MajorCyclePass<'p>,
214    target: &'p PassDomain<'p>,
215    router: &'b Router,
216    owners: usize,
217    planes: PlaneRange,
218    native_residuals: bool,
219    shape: ChunkShape,
220    /// The pass's kept weights, for the first domain.
221    weights: Option<(&'w WeightCache, bool)>,
222    /// The slot of the block's first sample.
223    weight_base: usize,
224    /// Rows of the domain's work histogram, when it is region-split.
225    grid_rows: Option<usize>,
226    block: &'b NativeBlock,
227    index: usize,
228    ordered: bool,
229}

231impl Placer<'_, '_, '_> {
232    /// Place the rows `chunks` hold on `team`: each placement into its
233    /// owner's buffer, and every placement into the weight owner's when the
234    /// pass grids the weight image on one owner; each chunk's gridding work
235    /// by anchor row beside them.
236    fn place(&self, team: &WorkerTeam, chunks: &mut [Chunk]) -> Result<(), PassError> {
237        let Self {
238            pass,
239            target,
240            router,
241            owners,
242            planes,
243            native_residuals,
244            shape,
245            weights,
246            weight_base,
247            grid_rows,
248            block,
249            index,
250            ordered,
251        } = *self;
252        let channels = block.channels();
253        // With region owners the weight image has one owner, which must see
254        // every placement; plane owners grid their own samples' weights.
255        let route_weight = pass.modes.weight && router.weight_owner().is_some();
256        team.for_each_mut(chunks, |_, chunk| {
257            chunk.placed = 0;
258            chunk.owned[..owners]
259                .iter_mut()
260                .for_each(SampleBuffer::clear);
261            chunk.weight_owned.clear();
262            chunk.row_work.clear();
263            if let Some(rows) = grid_rows {
264                chunk.row_work.resize(rows, 0);
265            }
266            for position in chunk.rows.clone() {
267                let row_index = row_at(block, ordered, position);
268                let native = block.row(index, row_index);
269                let slots = weights.and_then(|(cache, filled)| {
270                    cache.row(weight_base + row_index * channels, channels, filled)
271                });
272                let mut row_samples = 0;
273                let (owned, weight_owned, placed, row_work) = (
274                    &mut chunk.owned,
275                    &mut chunk.weight_owned,
276                    &mut chunk.placed,
277                    &mut chunk.row_work,
278                );
279                let mut keep = |placement: &Placement, values: &[Complex32], weights: &[f32]| {
280                    row_samples += 1;
281                    if !planes.contains(placement.plane) {
282                        return None;
283                    }
284                    if route_weight {
285                        weight_owned.push(*placement, values, weights);
286                    }
287                    *placed += 1;
288                    let (owner, work) = router.owner_and_work(target.operator, placement);
289                    if let Some((row, work)) = work
290                        && let Some(total) = row_work.get_mut(row)
291                    {
292                        *total += work;
293                    }
294                    Some(owner)
295                };
296                if native_residuals {
297                    target.resampler.place_each_sourced(
298                        target.operator,
299                        pass.weighting,
300                        &native,
301                        cell_of(position, channels),
302                        slots,
303                        |placement, values, weights, source| {
304                            if let Some(owner) = keep(&placement, values, weights) {
305                                owned[owner].push_sourced(placement, values, weights, source);
306                            }
307                        },
308                    )?;
309                } else {
310                    target.resampler.place_each(
311                        target.operator,
312                        pass.weighting,
313                        &native,
314                        slots,
315                        |placement, values, weights| {
316                            if let Some(owner) = keep(&placement, values, weights) {
317                                owned[owner].push(placement, values, weights);
318                            }
319                        },
320                    )?;
321                }
322                debug_assert!(
323                    row_samples <= shape.row,
324                    "a row places within the chunk's per-row bound"
325                );
326            }
327            debug_assert!(
328                chunk.placed as usize <= shape.placements
329                    && chunk.capacity_bytes() == shape.bytes(),
330                "a chunk places within the placements it was allocated for"
331            );
332            Ok::<_, PassError>(())
333        })
334    }
335}

337/// One row chunk's placements on the wave's planes, by owner, and its
338/// prediction batch.
339///
340/// Every buffer is allocated once, for a chunk's rows of the source's widest
341/// row ([`ChunkShape`]), and each owner's can take every placement of the
342/// chunk, so a chunk holds [`ChunkShape::bytes`] however its placements fall
343/// among the owners.

345    rows: Range<usize>,
346    /// Each owner's placements, in row order, with their native endpoints
347    /// when the wave forms residuals at native channels.
348    owned: Vec<SampleBuffer>,
349    /// Every placement of the chunk, in row order, when the weight image has
350    /// one owner ([`Router::weight_owner`]): that owner grids `Mode::Weight`
351    /// over it.
352    weight_owned: SampleBuffer,
353    placed: u64,
354    prediction: PredictionScratch,
355    /// The gridding work of the chunk's placements by anchor row, on a

357    /// the domain's after each slice.

361/// The largest chunk of a wave over blocks of `block`: at most
362/// [`chunk_placements`] placements, whatever rows the source puts in a
363/// block.
364#[derive(Clone, Copy)]
365struct ChunkShape {
366    /// Placements one native row places on any domain.
367    row: usize,
368    /// Rows a chunk places at once.
369    rows: usize,
370    /// Placements the chunk's rows place.
371    placements: usize,
372    /// Correlations per placement.
373    npol: usize,
374    /// Channels of one row.
375    channels: usize,
376    /// Correlations and channels of one row.
377    cells: usize,
378    /// Owners of the domain with the most.
379    owners: usize,
380    /// Whether the pass grids the weight image.
381    weight: bool,
382    /// Whether placements carry native endpoints (a residual at native
383    /// channels).
384    sourced: bool,
385    /// Samples the prediction batch of the chunk's rows degrids at most.
386    predicted: usize,
387    /// Bytes the prediction batch of the chunk's rows holds
388    /// ([`SpectralResampler::prediction_scratch`]) with each row's span in
389    /// it.
390    prediction: u64,
391    /// Grid rows of the region-partitioned domain with the most: the
392    /// length of the chunk's work histogram ([`Chunk::row_work`]).
393    grid_rows: usize,
394}
395
396impl ChunkShape {
397    /// The chunk of `chunks` chunks over blocks of `block`, each taking at
398    /// most `placements` placements before the owner scaling of
399    /// [`chunk_placements`].
400    fn new(
401        domains: &[PassDomain<'_>],
402        block: BlockShape,
403        weight: bool,
404        sourced: bool,
405        chunks: usize,
406        placements: usize,
407    ) -> Self {
408        let row = domains
409            .iter()
410            .map(|domain| domain.resampler.samples_per_row(block.spectrum))
411            .max()
412            .unwrap_or(0);
413        let owners = domains
414            .iter()
415            .map(|domain| domain.partition.owners())
416            .max()
417            .unwrap_or(1);
418        let rows = chunk_rows(
419            row,
420            block.rows,
421            chunks,
422            chunk_placements(placements, owners),
423        );
424        Self {
425            row,
426            rows,
427            placements: rows * row,
428            npol: block.correlations,
429            channels: block.channels(),
430            cells: block.channels() * block.correlations,
431            owners,
432            weight,
433            sourced,
434            predicted: rows
435                * domains
436                    .iter()
437                    .map(|domain| {
438                        domain
439                            .resampler
440                            .prediction_samples_per_row(block.channels())
441                    })
442                    .max()
443                    .unwrap_or(0),
444            prediction: SpectralResampler::prediction_scratch_bytes(
445                domains.iter().map(|domain| domain.resampler),
446                block.correlations,
447                block.channels(),
448                rows,
449            ),
450            grid_rows: domains
451                .iter()
452                .filter(|domain| domain.partition.splits_rows())
453                .map(|domain| domain.operator.geometry().grid_shape()[1])
454                .max()
455                .unwrap_or(0),
456        }
457    }
458
459    /// An empty buffer for `placements` placements of this shape, with room
460    /// for their native endpoints when they carry them.
461    fn buffer(self, placements: usize) -> SampleBuffer {
462        if self.sourced {
463            SampleBuffer::with_source_capacity(self.npol, placements)
464        } else {
465            SampleBuffer::with_capacity(self.npol, placements)
466        }
467    }
468
469    /// Bytes [`Self::buffer`] holds.
470    const fn buffer_bytes(self, placements: usize) -> u64 {
471        if self.sourced {
472            SampleBuffer::sourced_bytes(self.npol, placements)
473        } else {
474            SampleBuffer::bytes(self.npol, placements)
475        }
476    }
477
478    /// Placements the weight-owner buffer takes.
479    const fn weight_placements(self) -> usize {
480        if self.weight { self.placements } else { 0 }
481    }
482
483    /// Bytes a chunk of this shape holds: each owner's and the weight
484    /// owner's placements, its prediction batch and its work histogram.
485    const fn bytes(self) -> u64 {
486        self.owners as u64 * self.buffer_bytes(self.placements)
487            + SampleBuffer::bytes(self.npol, self.weight_placements())
488            + self.prediction
489            + (self.grid_rows * size_of::<u64>()) as u64
490    }
491}

493/// The backend that predicts one domain's batch for one chunk and forms the
494/// model at its rows' native cells ([`Work::PredictNative`]): the pass's,
495/// so that on Metal the model never leaves the device and the device works
496/// on one chunk's batch while the host places the next chunk's.
497enum Predictor<'w> {
498    Cpu(CpuBackend),
499    Metal(Box<MetalBackend<'w>>),
500}
501
502/// One predictor per domain on `backend`, for one chunk.
503fn predictors<'w>(
504    domains: &[PassDomain<'w>],
505    backend: BackendChoice,
506    stage_timing: bool,
507) -> Result<Vec<Predictor<'w>>, PassError> {
508    domains
509        .iter()
510        .map(|domain| {
511            Ok(match backend {
512                BackendChoice::Cpu => Predictor::Cpu(CpuBackend::new()),
513                BackendChoice::Metal => Predictor::Metal(Box::new(MetalBackend::new(
514                    domain.operator.cf(),
515                    stage_timing,
516                )?)),
517            })
518        })
519        .collect()
520}

522impl Chunk {
523    fn new(shape: ChunkShape, domains: &[PassDomain<'_>]) -> Self {
524        Self {
525            rows: 0..0,
526            owned: (0..shape.owners)
527                .map(|_| shape.buffer(shape.placements))
528                .collect(),
529            weight_owned: SampleBuffer::with_capacity(shape.npol, shape.weight_placements()),
530            placed: 0,
531            prediction: SpectralResampler::prediction_scratch(
532                domains.iter().map(|domain| domain.resampler),
533                shape.npol,
534                shape.channels,
535                shape.rows,
536            ),
537            row_work: Vec::with_capacity(shape.grid_rows),
538        }
539    }
540
541    /// Bytes the chunk holds now, from its buffers' capacities: at most
542    /// [`ChunkShape::bytes`] while every buffer is within the shape.
543    fn capacity_bytes(&self) -> u64 {
544        self.owned
545            .iter()
546            .map(SampleBuffer::capacity_bytes)
547            .sum::<u64>()
548            + self.weight_owned.capacity_bytes()
549            + self.prediction.capacity_bytes()
550            + (self.row_work.capacity() * size_of::<u64>()) as u64
551    }
552}

559/// Bytes a wave places with on `workers` workers over blocks of `block`,
560/// gridding the weight image when `weight`: its row chunks and, with a
561/// model, its backends' batches. On Metal each chunk's predictors keep a
562/// ring, a prediction batch and their own copies of the kernel cells they
563/// meet ([`casa_imaging_metal::ring_bytes`],
564/// [`casa_imaging_metal::batch_bytes`], [`casa_imaging_metal::kernel_bytes`];
565/// #793 review); on the CPU each chunk's predictors keep their batch's
566/// predictions, and each owner of a pass forming residuals at native
567/// channels the residual values of a chunk's placements
568/// ([`CpuBackend::batch_bytes`]). They grow with the source's block only up
569/// to `placements` a chunk.
570pub(super) fn chunk_bytes(
571    domains: &[PassDomain<'_>],
572    backend: BackendChoice,
573    block: BlockShape,
574    workers: usize,
575    weight: bool,
576    with_model: bool,
577    placements: usize,
578) -> u64 {
579    let count = chunks(workers);
580    let sourced = super::native_residuals(domains, with_model);
581    let shape = ChunkShape::new(domains, block, weight, sourced, count, placements);
582    let device = match backend {
583        BackendChoice::Cpu if with_model => {
584            let owners = if sourced {
585                domains
586                    .iter()
587                    .map(|domain| domain.partition.owners())
588                    .sum::<usize>()
589            } else {
590                0
591            };
592            (count * domains.len()) as u64 * CpuBackend::batch_bytes(shape.npol, shape.predicted)
593                + owners as u64 * CpuBackend::batch_bytes(shape.npol, shape.placements)
594        }
595        BackendChoice::Cpu => 0,
596        BackendChoice::Metal if with_model => {
597            let kernels = domains
598                .iter()
599                .map(|domain| casa_imaging_metal::kernel_bytes(domain.operator.cf()))
600                .sum::<u64>();
601            count as u64
602                * (domains.len() as u64
603                    * (casa_imaging_metal::ring_bytes(shape.npol)
604                        + casa_imaging_metal::batch_bytes(shape.npol, shape.predicted))
605                    + kernels)
606        }
607        BackendChoice::Metal => 0,
608    };
609    count as u64 * shape.bytes() + device
610}

650/// Wall time of a wave's stages ([`Wave::stages`]), as the placing thread
651/// sees them. With one set of chunks, `placing` is the placement and
652/// `accumulating` the owners' accumulation, slice by slice. When the owners
653/// are pipelined (`pipelined`), they accumulate while the placer places
654/// and reads. Then `placing` also counts waits for a free team thread,
655/// including drains a waiting thread steals. `accumulating` is only the
656/// placer's wait for a free set and for the last sets. The owners' own
657/// accumulation times are in the region owners' log line.
658#[derive(Clone, Copy, Default)]
659struct WaveStages {
660    predicting: std::time::Duration,
661    placing: std::time::Duration,
662    accumulating: std::time::Duration,
663    pipelined: bool,
664}

666impl<'w, 'p> Wave<'w, 'p> {
667    /// A wave over `planes` with each domain's prepared `models`, forming
668    /// residuals at native channels when `native_residuals`, on `workers`
669    /// workers over blocks of at most `block`. It places with
670    /// [`chunk_bytes`] and predicts into [`prediction_bytes`].
671    pub(super) fn new(
672        pass: &'w MajorCyclePass<'p>,
673        planes: PlaneRange,
674        models: Option<&'w [PreparedModelGrids]>,
675        native_residuals: bool,
676        block: BlockShape,
677        workers: usize,
678    ) -> Result<Self, PassError> {
679        let count = chunks(workers);
680        let shape = ChunkShape::new(
681            pass.domains,
682            block,
683            pass.modes.weight,
684            native_residuals,
685            count,
686            pass.chunk_placements,
687        );
688        let domains = pass
689            .domains
690            .iter()
691            .enumerate()
692            .map(|(index, domain)| {
693                let router = Router::new(&domain.partition, planes);
694                let operator = domain.operator;
695                let weight_owner = router.weight_owner();
696                let owners = (0..router.owners())
697                    .map(|owner| {
698                        let (range, tile) = router.target(&domain.partition, planes, owner);
699                        let weights = (pass.modes.weight
700                            && weight_owner.is_none_or(|holder| holder == owner))
701                        .then(|| WeightSums::new(block.correlations, WEIGHT_GROUPS));
702                        Ok(match pass.backend {
703                            BackendChoice::Cpu => Owner {
704                                acc: Some(
705                                    pass.summing(operator.accumulator(range, tile, pass.modes)),
706                                ),
707                                backend: OwnerBackend::Cpu(CpuBackend::new()),
708                                images: None,
709                                busy: std::time::Duration::ZERO,
710                                weights,
711                            },
712                            BackendChoice::Metal => {
713                                assert_eq!(
714                                    operator.precision(),
715                                    GridPrecision::F32,
716                                    "Metal grids are f32 (D2)"
717                                );
718                                Owner {
719                                    acc: Some(pass.summing(MetalBackend::accumulator(
720                                        operator.accumulator_layout(range, tile, pass.modes),
721                                    )?)),
722                                    backend: OwnerBackend::Metal(Box::new(
723                                        MetalBackend::new(operator.cf(), pass.metal_stage_timing)?
724                                            .with_deferred_grid_completion(),
725                                    )),
726                                    images: None,
727                                    busy: std::time::Duration::ZERO,
728                                    weights,
729                                }
730                            }
731                        })
732                    })
733                    .collect::<Result<_, PassError>>()?;
734                let row_work = domain
735                    .partition
736                    .splits_rows()
737                    .then(|| vec![0; operator.geometry().grid_shape()[1]]);
738                Ok(Domain {
739                    router,
740                    owners,
741                    model: models.map(|models| &models[index]),
742                    row_work,
743                })
744            })
745            .collect::<Result<_, PassError>>()?;
746        // A pass with a model predicts each block, for its native residuals or
747        // for a visibility writer.
748        let predictions = if pass.model.is_some() {
749            block.rows * shape.cells
750        } else {
751            0
752        };
753        let predictions = match pass.backend {
754            BackendChoice::Metal if predictions > 0 => {
755                NativeStore::Device(MetalBackend::native_cells(predictions)?)
756            }
757            BackendChoice::Cpu | BackendChoice::Metal => {
758                NativeStore::Host(Vec::with_capacity(predictions))
759            }
760        };
761        Ok(Self {
762            pass,
763            planes,
764            domains,
765            native_residuals,
766            check_spacing: false,
767            shape,
768            chunks: (0..count)
769                .map(|_| Chunk::new(shape, pass.domains))
770                .collect(),
771            predictors: (0..count)
772                .map(|_| predictors(pass.domains, pass.backend, pass.metal_stage_timing))
773                .collect::<Result<_, _>>()?,
774            predictions,
775            samples: 0,
776            weights: pass.weight_cache().map(|cache| (cache, cache.filled())),
777            weight_base: 0,
778            stages: WaveStages::default(),
779        })
780    }

794    /// Place every row of `block` on every domain and accumulate the
795    /// placements; with a visibility sink, first hand it the block, with
796    /// every selected sample's prediction when it asks for them.
797    ///
798    /// When the wave forms residuals at native channels, the predictors
799    /// first form `Σ_d A_d·m_d` at every row's native cells, and each placed
800    /// sample's observed value is gridded less that model at its native
801    /// endpoints ([`Work::NativeResidualGrid`]); otherwise the domain's model
802    /// is subtracted at each placed sample.
803    pub(super) fn consume(
804        &mut self,
805        block: &NativeBlock,
806        team: &WorkerTeam,
807        visibilities: Option<&mut VisibilitySink<'_>>,
808    ) -> Result<(), PassError> {
809        let rows = block.len();
810        if self.check_spacing {
811            let bound_hz = self.pass.native_spacing_hz;
812            for row in 0..rows {
813                let observed_hz = block
814                    .row(0, row)
815                    .frequencies_hz
816                    .windows(2)
817                    .fold(0.0_f64, |widest, pair| {
818                        widest.max((pair[1] - pair[0]).abs())
819                    });
820                if observed_hz > bound_hz {
821                    return Err(PassError::NativeSpacing {
822                        observed_hz,
823                        bound_hz,
824                    });
825                }
826            }
827        }
828        let sink_predictions = visibilities.as_ref().is_some_and(|sink| sink.predictions);
829        // Chunks take rows by position in the block's locality order, and
830        // predictions are stored by position, unless the sink reads them in
831        // the block's row order.
832        let ordered = !sink_predictions;
833        let cells = block.channels() * block.correlations();
834        if self.native_residuals || sink_predictions {
835            if let NativeStore::Host(predictions) = &mut self.predictions {
836                // The first domain with a model writes every cell of the
837                // rows it predicts, so stale cells never reach a sample;
838                // without a model the cells stay zero.
839                if predictions.len() < rows * cells
840                    || self.domains.iter().all(|domain| domain.model.is_none())
841                {
842                    predictions.clear();
843                    predictions.resize(rows * cells, Complex32::default());
844                }
845            }
846            // The same slices as placement, so no chunk predicts more rows
847            // than its batch was allocated for.
848            let chunks = self.chunks.len();
849            for slice in slices(rows, chunks, self.shape.rows) {
850                let mut count = 0;
851                for (chunk, rows) in self
852                    .chunks
853                    .iter_mut()
854                    .zip(chunk_ranges(slice.clone(), chunks))
855                {
856                    chunk.rows = rows;
857                    count += 1;
858                }
859                let started = std::time::Instant::now();
860                self.predict(block, team, count, slice, ordered)?;
861                self.stages.predicting += started.elapsed();
862            }
863        }
864        if let Some(sink) = visibilities {
865            let predictions: &[Complex32] = if sink.predictions {
866                self.host_predictions(rows * cells)?
867            } else {
868                &[]
869            };
870            (sink.write)(block, predictions).map_err(PassError::VisibilityWrite)?;
871        }
872        // Slices of the block, each owner's in the block's order as a whole
873        // block's would be, so the accumulation is the same.
874        let chunks = self.chunks.len();
875        for slice in slices(rows, chunks, self.shape.rows) {
876            let mut count = 0;
877            for (chunk, rows) in self.chunks.iter_mut().zip(chunk_ranges(slice, chunks)) {
878                chunk.rows = rows;
879                count += 1;
880            }
881            for index in 0..self.domains.len() {
882                let placed = self.place(block, team, count, index, ordered)?;
883                if index == 0 {
884                    self.samples += placed;
885                }
886            }
887        }
888        self.weight_base += rows * block.channels();
889        Ok(())
890    }

892    /// What placing `block`'s rows on domain `index` reads: the wave's
893    /// state at this block. Chunks hold positions in the block's locality
894    /// order when `ordered`, else row indices ([`row_at`]).
895    fn placer<'b>(
896        &'b self,
897        block: &'b NativeBlock,
898        index: usize,
899        ordered: bool,
900    ) -> Placer<'b, 'w, 'p> {
901        let domain = &self.domains[index];
902        Placer {
903            pass: self.pass,
904            target: &self.pass.domains[index],
905            router: &domain.router,
906            owners: domain.router.owners(),
907            planes: self.planes,
908            native_residuals: self.native_residuals,
909            shape: self.shape,
910            weights: self.weights.filter(|_| index == 0),
911            weight_base: self.weight_base,
912            grid_rows: domain.row_work.as_ref().map(Vec::len),
913            block,
914            index,
915            ordered,
916        }
917    }

919    /// Place the rows of `block` on domain `index` and accumulate them;
920    /// returns the samples placed. Chunks hold positions in the block's
921    /// locality order when `ordered`, else row indices ([`row_at`]).
922    fn place(
923        &mut self,
924        block: &NativeBlock,
925        team: &WorkerTeam,
926        count: usize,
927        index: usize,
928        ordered: bool,
929    ) -> Result<u64, PassError> {
930        let placing = std::time::Instant::now();
931        let mut chunks = std::mem::take(&mut self.chunks);
932        let placed = self
933            .placer(block, index, ordered)
934            .place(team, &mut chunks[..count]);
935        self.chunks = chunks;
936        placed?;
937        let pass = self.pass;
938        let target = &pass.domains[index];
939        let domain = &mut self.domains[index];
940        let chunks = &self.chunks[..count];
941        domain.add_row_work(chunks);
942        self.stages.placing += placing.elapsed();
943        let accumulating = std::time::Instant::now();
944        let model = domain.model.filter(|_| !self.native_residuals);
945        let native = self.native_residuals.then(|| match &self.predictions {
946            NativeStore::Host(cells) => NativeModel::Host { cells, first: 0 },
947            NativeStore::Device(cells) => NativeModel::Device(cells),
948        });
949        let weight_owner = domain.router.weight_owner();
950        team.for_each_mut(&mut domain.owners, |owner_index, owner| {
951            let subtract = Subtract { model, native };
952            owner.accumulate(pass, target, (owner_index, weight_owner), subtract, chunks)
953        })?;
954        self.stages.accumulating += accumulating.elapsed();
955        Ok(chunks.iter().map(|chunk| chunk.placed).sum())
956    }

958    /// The model at native cells of every selected sample of the rows
959    /// `slice` of `block`, which the first `count` chunks hold, summed over
960    /// every domain's model (`SIMapperCollection::degrid`), `[row][channel]
961    /// [correlation]`, into their rows of the wave's model store, which stay
962    /// zero without a model. Rows are by position in the block's locality
963    /// order when `ordered` ([`row_at`]).
964    ///
965    /// Domain by domain, every chunk places its rows' samples in one batch
966    /// with its rows' native terms and hands it to its predictor
967    /// ([`Work::PredictNative`]): the first domain writes the cells and
968    /// later ones add to them. A Metal predictor leaves the work on the
969    /// device, which runs it before any later device work that reads the
970    /// cells, while the host places the next chunk's batch.
971    fn predict(
972        &mut self,
973        block: &NativeBlock,
974        team: &WorkerTeam,
975        count: usize,
976        slice: Range<usize>,
977        ordered: bool,
978    ) -> Result<(), PassError> {
979        if self.domains.iter().all(|domain| domain.model.is_none()) {
980            return Ok(());
981        }
982        let channels = block.channels();
983        let cells = channels * block.correlations();
984        let pass = self.pass;
985        let shape = self.shape;
986        let models = self
987            .domains
988            .iter()
989            .map(|domain| domain.model)
990            .collect::<Vec<_>>();
991        let mut pieces = Vec::with_capacity(count);
992        let chunks = self.chunks[..count].iter_mut().zip(&mut self.predictors);
993        match &mut self.predictions {
994            NativeStore::Host(predictions) => {
995                let mut remaining = &mut predictions[slice.start * cells..slice.end * cells];
996                for (chunk, predictors) in chunks {
997                    let (head, tail) = remaining.split_at_mut(chunk.rows.len() * cells);
998                    remaining = tail;
999                    pieces.push((chunk, predictors, ChunkCells::Host(head)));
1000                }
1001            }
1002            NativeStore::Device(cells) => {
1003                for (chunk, predictors) in chunks {
1004                    pieces.push((chunk, predictors, ChunkCells::Device(cells)));
1005                }
1006            }
1007        }
1008        let mut add = false;
1009        for (index, (target, model)) in pass.domains.iter().zip(&models).enumerate() {
1010            let Some(model) = *model else {
1011                continue;
1012            };
1013            let npol = target.operator.polarization().correlations().len();
1014            team.for_each_mut(&mut pieces, |_, (chunk, predictors, cells)| {
1015                chunk.prediction.begin(npol);
1016                for position in chunk.rows.clone() {
1017                    let row = block.row(index, row_at(block, ordered, position));
1018                    let placed = target.resampler.place_prediction(
1019                        target.operator,
1020                        model,
1021                        &row,
1022                        &mut chunk.prediction,
1023                    )?;
1024                    target.resampler.native_terms(
1025                        &row,
1026                        npol,
1027                        placed,
1028                        &mut chunk.prediction,
1029                        cell_of(position, channels),
1030                    );
1031                }
1032                let cf = target.operator.cf();
1033                match (&mut predictors[index], cells) {
1034                    (Predictor::Cpu(backend), ChunkCells::Host(cells)) => {
1035                        let first = cell_of(chunk.rows.start, channels);
1036                        let target = NativeTarget::Host { cells, first };
1037                        chunk
1038                            .prediction
1039                            .apply_native(backend, cf, model, target, add)?;
1040                    }
1041                    (Predictor::Metal(backend), ChunkCells::Device(cells)) => {
1042                        let backend: &mut MetalBackend<'_> = backend;
1043                        let target = NativeTarget::Device(cells);
1044                        chunk
1045                            .prediction
1046                            .apply_native(backend, cf, model, target, add)?;
1047                    }
1048                    _ => unreachable!("a wave's predictors and model store share its backend"),
1049                }
1050                Ok::<_, PassError>(())
1051            })?;
1052            add = true;
1053        }
1054        debug_assert!(
1055            pieces
1056                .iter()
1057                .all(|(chunk, ..)| chunk.capacity_bytes() == shape.bytes()),
1058            "a chunk predicts within the buffers it was allocated"
1059        );
1060        Ok(())
1061    }

1063    /// The model at the first `len` native cells of the block for the host:
1064    /// on Metal once every predictor's work, which alone writes the cells,
1065    /// has completed.
1066    fn host_predictions(&mut self, len: usize) -> Result<&[Complex32], PassError> {
1067        match &self.predictions {
1068            NativeStore::Host(cells) => Ok(&cells[..len]),
1069            NativeStore::Device(cells) => {
1070                for predictor in self.predictors.iter_mut().flatten() {
1071                    if let Predictor::Metal(backend) = predictor {
1072                        backend.complete()?;
1073                    }
1074                }
1075                // SAFETY: only the predictors' work writes the cells, every
1076                // command of it has completed, and none is issued while the
1077                // slice, which borrows the wave, lives.
1078                Ok(&unsafe { cells.host_cells() }[..len])
1079            }
1080        }
1081    }

1092    /// Transform the accumulated grids into this wave's normal images, one
1093    /// per domain.
1094    pub(super) fn finish(mut self, team: &WorkerTeam) -> Result<Vec<NormalImages>, PassError> {
1095        // Metal owners leave their last grid work running on the device, and
1096        // Metal predictors their native models: complete both, and learn
1097        // whether any failed, before any accumulator is read.
1098        for predictor in self.predictors.iter_mut().flatten() {
1099            if let Predictor::Metal(backend) = predictor {
1100                backend.complete()?;
1101            }
1102        }
1103        // The weight image's last summed contributions go to the grid first:
1104        // accumulation, which the stages count as such.
1105        let flushing = std::time::Instant::now();
1106        for (domain, target) in self.domains.iter_mut().zip(self.pass.domains) {
1107            let summed = std::sync::Mutex::new([0_u64; 3]);
1108            team.for_each_mut(&mut domain.owners, |_, owner| {
1109                let totals = owner.flush_weights(target)?;
1110                let mut summed = summed.lock().unwrap_or_else(PoisonError::into_inner);
1111                for (sum, total) in summed.iter_mut().zip(totals) {
1112                    *sum += total;
1113                }
1114                drop(summed);
1115                owner.complete()
1116            })?;
1117            let [placements, groups, grids] =
1118                summed.into_inner().unwrap_or_else(PoisonError::into_inner);
1119            if placements > 0 {
1120                tracing::info!(
1121                    "weight image summed: {placements} placements as {groups} group grids in \
1122                     {grids} grids"
1123                );
1124            }
1125        }
1126        self.stages.accumulating += flushing.elapsed();
1127        let pass = self.pass;
1128        let planes = self.planes;
1129        let stages = self.stages;
1130        let transforming = std::time::Instant::now();
1131        let images = self
1132            .domains
1133            .into_iter()
1134            .zip(pass.domains)
1135            .map(|(mut domain, target)| {
1136                let operator = target.operator;
1137                match &target.partition {
1138                    Partition::Planes { .. } => {
1139                        team.for_each_mut(&mut domain.owners, |_, owner| {
1140                            let acc = owner.acc.take().expect("each owner finishes once");
1141                            owner.images = Some(operator.finish(acc)?);
1142                            Ok::<_, PassError>(())
1143                        })?;
1144                        Ok(concatenate(
1145                            domain
1146                                .owners
1147                                .into_iter()
1148                                .map(|owner| owner.images.expect("every owner finished")),
1149                        ))
1150                    }
1151                    Partition::Regions { regions, .. } => {
1152                        // One owner's tile is the whole grid (`Partition::regions`
1153                        // gives a lone region every row): it is the sum, and
1154                        // `finish` refuses a tile. The charge keys on the same
1155                        // count (`DomainStages`).
1156                        if let [owner] = domain.owners.as_mut_slice() {
1157                            let acc = owner.acc.take().expect("each owner finishes once");
1158                            drop(domain.owners);
1159                            return Ok(operator.finish(acc)?);
1160                        }
1161                        // How evenly the strips shared the gridding (#736).
1162                        tracing::info!(
1163                            "region owners: rows {:?}, accumulating {:?} s",
1164                            regions
1165                                .iter()
1166                                .map(|region| region.rows.clone())
1167                                .collect::<Vec<_>>(),
1168                            domain
1169                                .owners
1170                                .iter()
1171                                .map(|owner| (owner.busy.as_secs_f64() * 10.0).round() / 10.0)
1172                                .collect::<Vec<_>>(),
1173                        );
1174                        let mut merged =
1175                            pass.summing(operator.accumulator(planes, None, pass.modes));
1176                        for owner in &domain.owners {
1177                            merged
1178                                .merge_from(owner.acc.as_ref().expect("owners hold their tiles"))?;
1179                        }
1180                        drop(domain.owners);
1181                        Ok(operator.finish(merged)?)
1182                    }
1183                }
1184            })
1185            .collect::<Result<Vec<_>, PassError>>()?;
1186        if stages.pipelined {
1187            tracing::info!(
1188                "wave stages (owners pipelined): predicting {:.1} s, placing with thread \
1189                 waits {:.1} s, waiting for owners {:.1} s, transforming {:.1} s",
1190                stages.predicting.as_secs_f64(),
1191                stages.placing.as_secs_f64(),
1192                stages.accumulating.as_secs_f64(),
1193                transforming.elapsed().as_secs_f64(),
1194            );
1195        } else {
1196            tracing::info!(
1197                "wave stages: predicting {:.1} s, placing {:.1} s, accumulating {:.1} s, \
1198                 transforming {:.1} s",
1199                stages.predicting.as_secs_f64(),
1200                stages.placing.as_secs_f64(),
1201                stages.accumulating.as_secs_f64(),
1202                transforming.elapsed().as_secs_f64(),
1203            );
1204        }
1205        Ok(images)
1206    }
1207}

1209/// The row at `position`: of the block's locality order when `ordered`
1210/// ([`NativeBlock::ordered_row`]), else the row itself.
1211fn row_at(block: &NativeBlock, ordered: bool, position: usize) -> usize {
1212    if ordered {
1213        block.ordered_row(position)
1214    } else {
1215        position
1216    }

1219/// One chunk's share of the wave's model store while it predicts.
1220enum ChunkCells<'c> {
1221    /// The chunk's rows' host cells.
1222    Host(&'c mut [Complex32]),
1223    /// The device cells of the whole block.
1224    Device(&'c DeviceNativeCells),
1225}

1227/// The first native cell of the row at `position` of a block of `channels`
1228/// channels.
1229fn cell_of(position: usize, channels: usize) -> u32 {
1230    u32::try_from(position * channels).expect("a block's native cells fit u32 indices")
1231}

1233/// What an owner subtracts from its data terms: the domain's model at each
1234/// placed sample, the model at the samples' native endpoints, or nothing.
1235#[derive(Clone, Copy)]
1236struct Subtract<'a> {
1237    model: Option<&'a PreparedModelGrids>,
1238    native: Option<NativeModel<'a>>,
1239}
1240
1241/// Grid one owner's share of a slice, through `apply`, in the data and PSF
1242/// modes the pass holds; the data terms receive the residual of what
1243/// `subtract` names. The weight image is summed apart ([`WeightSums`]).
1244fn accumulate(
1245    pass: &MajorCyclePass<'_>,
1246    apply: &mut dyn FnMut(Work<'_>) -> Result<(), casa_imaging_operator::OperatorError>,
1247    subtract: Subtract<'_>,
1248    acc: &mut GridAccumulator,
1249) -> Result<(), PassError> {
1250    if pass.modes.data {
1251        let work = match subtract {
1252            Subtract {
1253                native: Some(model),
1254                ..
1255            } => Work::NativeResidualGrid { model, acc },
1256            Subtract {
1257                model: Some(model), ..
1258            } => Work::ResidualGrid {
1259                model,
1260                acc,
1261                residual_out: None,
1262            },
1263            Subtract { .. } => Work::Grid {
1264                mode: Mode::Data,
1265                acc,
1266            },
1267        };
1268        apply(work)?;
1269    }
1270    if pass.modes.psf {
1271        apply(Work::Grid {
1272            mode: Mode::Psf,
1273            acc,
1274        })?;
1275    }
1276    Ok(())
1277}

1279/// Join per-owner images of consecutive plane ranges.
1280pub(super) fn concatenate(parts: impl IntoIterator<Item = NormalImages>) -> NormalImages {
1281    let mut parts = parts.into_iter();
1282    let mut joined = parts.next().expect("a wave has at least one owner");
1283    for part in parts {
1284        assert!(
1285            part.pols == joined.pols
1286                && part.data_terms == joined.data_terms
1287                && part.psf_terms == joined.psf_terms
1288                && part.first_plane == joined.first_plane + joined.planes.len() as u32,
1289            "owner images must continue the plane sequence with the same layout"
1290        );
1291        joined.planes.extend(part.planes);
1292    }
1293    joined
1294}
