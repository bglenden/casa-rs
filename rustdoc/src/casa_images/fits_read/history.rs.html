9pub(crate) fn history_entries(path: &Path) -> Result<Vec<ImageHistoryEntry>, ImageError> {
10    let cards = read_cards(path)?;
11    let cards = cards.into_iter().filter_map(|card| {
12        card.strip_prefix("HISTORY ")
13            .map(|line| line.trim_end().to_string())
14    });
15    let mut entries = Vec::new();
16    let mut group = String::new();
17    let mut lines: Vec<String> = Vec::new();
18    for line in cards {
19        let normalized = line.trim_start();
20        let mut words = normalized.split_whitespace();
21        let first = words.next().unwrap_or("");
22        let second = words.next().unwrap_or("");
23        if first.eq_ignore_ascii_case("CASA") && second.eq_ignore_ascii_case("START") {
24            append_group(&mut entries, &group, &lines);
25            lines.clear();
26            group = words.collect::<String>();
27        } else if first.eq_ignore_ascii_case("CASA") && second.eq_ignore_ascii_case("END") {
28            append_group(&mut entries, &group, &lines);
29            lines.clear();
30            group.clear();
31        } else if let Some(continuation) = line.strip_prefix('>') {
32            if let Some(previous) = lines.last_mut() {
33                previous.push_str(continuation);
34            } else {
35                lines.push(line);
36            }
37        } else {
38            lines.push(line);
39        }
40    }
41    append_group(&mut entries, &group, &lines);
42    Ok(entries)
43}

45pub(crate) fn read_cards(path: &Path) -> Result<Vec<String>, ImageError> {
46    let mut file = BufReader::new(
47        std::fs::File::open(path).map_err(|error| ImageError::Io(error.to_string()))?,
48    );
49    let mut cards = Vec::new();
50    loop {
51        let mut card = [0; 80];
52        file.read_exact(&mut card)
53            .map_err(|error| ImageError::Io(error.to_string()))?;
54        if &card[..8] == b"END     " {
55            break;
56        }
57        cards.push(String::from_utf8_lossy(&card).into_owned());
58    }
59    Ok(cards)
60}

62fn append_group(entries: &mut Vec<ImageHistoryEntry>, group: &str, lines: &[String]) {
63    if group == "LOGTABLE" && lines.len().is_multiple_of(2) {
64        for pair in lines.as_chunks::<2>().0 {
65            let header = &pair[0];
66            let date = header.split_whitespace().next().unwrap_or("");
67            let location = header
68                .split_once("SRCCODE='")
69                .and_then(|(_, rest)| rest.split_once('\''))
70                .map_or("", |(location, _)| location);
71            entries.push(ImageHistoryEntry {
72                time_mjd_seconds: parse_time(date).unwrap_or(-1.0),
73                location: location.to_string(),
74                message: pair[1].clone(),
75            });
76        }
77    } else if group.is_empty() || group == "LOGTABLE" {
78        entries.extend(lines.iter().map(|line| ImageHistoryEntry {
79            time_mjd_seconds: -1.0,
80            location: String::new(),
81            message: line.clone(),
82        }));
83    }
84}

86fn parse_time(value: &str) -> Option<f64> {
87    let (date, clock) = value.split_once('T').unwrap_or((value, "00:00:00"));
88    let mut date = date.split('-');
89    let (year, month, day) = (
90        date.next()?.parse().ok()?,
91        date.next()?.parse().ok()?,
92        date.next()?.parse().ok()?,
93    );
94    let mut clock = clock.split(':');
95    let (hour, minute, second) = (
96        clock.next()?.parse().ok()?,
97        clock.next()?.parse().ok()?,
98        clock.next()?.parse().ok()?,
99    );
100    casa_types::quanta::time::MvTime::from_ymd_hms_utc(year, month, day, hour, minute, second)
101        .map(|time| time.as_mjd_seconds())
102}
