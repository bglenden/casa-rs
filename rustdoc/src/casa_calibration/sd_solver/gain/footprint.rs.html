2//! Select integrations in the central region of a double-circle scan.
3
4use std::collections::BTreeMap;
5
6use casa_core::measures::direction::DirectionRef;
7use casa_ms::MeasurementSet;
8use casa_ms::derived::engine::{MsCalEngine, raw_field_reference_direction};
9
10use super::{error_text, normalization};
11use crate::sd_solver::edges::Integration;
12use crate::sd_solver::map_geometry::{MapTargets, SourceRow, Target};
13
14type FeedKey = (i32, i32, i32);
15
16pub(super) fn select<'a>(
17    ms: &MeasurementSet,
18    samples: &'a [Integration],
19    requested_radius: Option<f64>,
20) -> Result<Vec<&'a Integration>, String> {
21    let spectral = ms.spectral_window().map_err(error_text)?;
22    let antennas = ms.antenna().map_err(error_text)?;
23    let channels = (0..spectral.row_count())
24        .map(|window| spectral.num_chan(window).map_err(error_text))
25        .collect::<Result<Vec<_>, _>>()?;
26    let mut feeds = BTreeMap::<_, Vec<_>>::new();
27    // Four-channel WVR integrations can pass a science feed's time ranges,
28    // but never establish a central region on their own.
29    for row in samples.iter().filter(|row| channels[row.window] != 4) {
30        feeds
31            .entry((key(row), row.description))
32            .or_default()
33            .push(row);
34    }
35    if feeds.is_empty() {
36        return Ok(Vec::new());
37    }
38    let engine = MsCalEngine::new(ms)
39        .and_then(|engine| engine.with_field_ephemerides(ms))
40        .map_err(error_text)?;
41    let mut targets = MapTargets::read(ms)?;
42    let mut ranges = BTreeMap::<FeedKey, Vec<[f64; 2]>>::new();
43    for ((key, _), rows) in feeds {
44        if ranges.contains_key(&key) {
45            continue;
46        }
47        let first = rows[0];
48        let antenna = first.antenna as usize;
49        let field = first.field as usize;
50        let radius = normalization::radius(
51            requested_radius,
52            &spectral.chan_freq(first.window).map_err(error_text)?,
53            antennas.dish_diameter(antenna).map_err(error_text)?,
54        )?;
55        let target = targets.field(field, SourceRow::Optional)?;
56        let fixed = if matches!(target, Target::Fixed) {
57            let frame = engine
58                .antenna_frame(first.time, antenna)
59                .map_err(error_text)?;
60            let center = raw_field_reference_direction(ms, field)
61                .map_err(error_text)?
62                .convert_to(DirectionRef::J2000, &frame)
63                .map_err(error_text)?;
64            Some(center.as_angles())
65        } else {
66            None
67        };
68        let mut requested = rows
69            .iter()
70            .map(|row| (row.antenna, row.time))
71            .collect::<Vec<_>>();
72        // The CASA pointing calculator sorts before interpolation. The
73        // filter uses those positions with the group's original times.
74        requested.sort_by(|left, right| left.1.total_cmp(&right.1));
75        let directions = ms
76            .pointing()
77            .and_then(|p| p.scan_directions(&requested, &engine))
78            .map_err(error_text)?;
79        let mut accepted = Vec::<[f64; 2]>::new();
80        for ((row, &(_, pointing_time)), direction) in
81            rows.into_iter().zip(&requested).zip(directions)
82        {
83            let position = direction.as_angles();
84            let offset = if let Some(center) = fixed {
85                sin_offset(position, center)
86            } else if let Some((longitude, latitude)) =
87                target.direction(&engine, pointing_time, antenna, field)?
88            {
89                [
90                    (position.0 - longitude)
91                        .sin()
92                        .atan2((position.0 - longitude).cos()),
93                    position.1 - latitude,
94                ]
95            } else {
96                unreachable!("fixed fields have a reference pointing")
97            };
98            if offset[0].hypot(offset[1]) <= radius {
99                let start = row.time - row.interval * 0.5;
100                let end = row.time + row.interval * 0.5;
101                accepted.push([start, end]);
102            }
103        }
104        if !accepted.is_empty() {
105            accepted.sort_by(|left, right| left[0].total_cmp(&right[0]));
106            let mut combined = Vec::<[f64; 2]>::new();
107            for [start, end] in accepted {
108                if let Some(last) = combined.last_mut()
109                    && start <= last[1]
110                {
111                    last[1] = last[1].max(end);
112                } else {
113                    combined.push([start, end]);
114                }
115            }
116            ranges.insert(key, combined);
117        }
118    }
119    if ranges.is_empty() {
120        return Err("No integration matches the double-circle central region.".into());
121    }
122    Ok(samples
123        .iter()
124        .filter(|row| {
125            ranges.get(&key(row)).is_some_and(|ranges| {
126                let index = ranges.partition_point(|range| range[1] < row.time);
127                ranges.get(index).is_some_and(|range| range[0] <= row.time)
128            })
129        })
130        .collect())
131}
132
133fn key(sample: &Integration) -> FeedKey {
134    (sample.field, sample.antenna, sample.feed)
135}
136
137fn sin_offset((longitude, latitude): (f64, f64), (lon0, lat0): (f64, f64)) -> [f64; 2] {
138    let (sin_lat, cos_lat) = latitude.sin_cos();
139    let (sin_lat0, cos_lat0) = lat0.sin_cos();
140    let (sin_delta, cos_delta) = (longitude - lon0).sin_cos();
141    // OrthographicProjector converts x pixels with dy, after using
142    // dx = dy / cos(latitude_0) for the world-to-pixel projection.
143    [
144        cos_lat * sin_delta * cos_lat0,
145        sin_lat * cos_lat0 - cos_lat * sin_lat0 * cos_delta,
146    ]
147}
