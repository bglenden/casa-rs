3//! The stored channels a wave reads: the window of a source's channels whose
4//! samples, converted to the output frame row by row, reach the wave's
5//! frequency bounds, with both linear-interpolation partners of each.
6
7use casa_imaging_model::CompiledProblem;
8
9use super::super::spectral_evaluation::prepare_row_frequency_conversion;
10use super::{BoundObservationSource, BoundObservationSourceError, SelectedCoordinates};
11use crate::spectral_selection::PreparedFrequencyFrameConversion;
12use crate::{SelectedObservationRow, VisibilityChannelReadRange};
13
14pub(super) fn selected_channel_window(
15    source: &BoundObservationSource,
16    problem: &CompiledProblem,
17    coordinates: &SelectedCoordinates,
18    rows: &[SelectedObservationRow],
19    frequency_bounds_hz: [f64; 2],
20) -> Result<Option<(VisibilityChannelReadRange, std::ops::Range<usize>)>, BoundObservationSourceError>
21{
22    let lower_bound = frequency_bounds_hz[0].min(frequency_bounds_hz[1]);
23    let upper_bound = frequency_bounds_hz[0].max(frequency_bounds_hz[1]);
24    if !lower_bound.is_finite() || !upper_bound.is_finite() || lower_bound == upper_bound {
25        return Err(BoundObservationSourceError::SpectralContributionMismatch);
26    }
27    let output_frame = problem.geometry().spectral().output_frame();
28    let source_frame = coordinates
29        .channels
30        .first()
31        .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?
32        .frame;
33    let mut admitted_first: Option<usize> = None;
34    let mut admitted_last: Option<usize> = None;
35    let mut cached_conversion: Option<((i32, u64), PreparedFrequencyFrameConversion)> = None;
36    for row in rows.iter().copied() {
37        let normalized_time = row.time_mjd_seconds / 86_400.0 * 86_400.0;
38        let context_key = (row.field_id, normalized_time.to_bits());
39        let conversion = if let Some((cached_key, conversion)) = cached_conversion
40            && cached_key == context_key
41        {
42            conversion
43        } else {
44            let conversion = prepare_row_frequency_conversion(
45                source.geometry_engine(),
46                row.field_id,
47                normalized_time,
48                source_frame,
49                output_frame,
50            )?;
51            cached_conversion = Some((context_key, conversion));
52            conversion
53        };
54        let mut row_window = SelectedChannelWindowSelection::new(lower_bound, upper_bound);
55        for (ordinal, channel) in coordinates.channels.iter().copied().enumerate() {
56            let centre = conversion.convert_hz(channel.centre_hz);
57            if !centre.is_finite() || centre <= 0.0 {
58                return Err(BoundObservationSourceError::SpectralContributionMismatch);
59            }
60            row_window.observe(ordinal, centre);
61        }
62        if let Some(window) = row_window.finish() {
63            let first = window.start;
64            let last = window.end - 1;
65            admitted_first = Some(admitted_first.map_or(first, |current| current.min(first)));
66            admitted_last = Some(admitted_last.map_or(last, |current| current.max(last)));
67        }
68    }
69    let (Some(admitted_first), Some(admitted_last)) = (admitted_first, admitted_last) else {
70        return Ok(None);
71    };
72    let mut physical_start = usize::MAX;
73    let mut physical_end = 0_usize;
74    for ordinal in admitted_first..=admitted_last {
75        let channel = coordinates
76            .channels
77            .get(ordinal)
78            .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
79        let index = usize::try_from(channel.channel_index)
80            .map_err(|_| BoundObservationSourceError::SpectralContributionMismatch)?;
81        physical_start = physical_start.min(index);
82        physical_end = physical_end.max(
83            index
84                .checked_add(1)
85                .ok_or(BoundObservationSourceError::MeasurementOverflow)?,
86        );
87    }
88    Ok(Some((
89        VisibilityChannelReadRange::new(physical_start, physical_end - physical_start),
90        admitted_first..admitted_last + 1,
91    )))
92}

94#[derive(Clone, Copy)]
95struct SelectedChannelWindowSelection {
96    lower_bound: f64,
97    upper_bound: f64,
98    first: Option<usize>,
99    last: Option<usize>,
100    previous: Option<(usize, f64)>,
101}
102
103impl SelectedChannelWindowSelection {
104    const fn new(lower_bound: f64, upper_bound: f64) -> Self {
105        Self {
106            lower_bound,
107            upper_bound,
108            first: None,
109            last: None,
110            previous: None,
111        }
112    }
113
114    fn observe(&mut self, ordinal: usize, centre: f64) {
115        if (self.lower_bound..=self.upper_bound).contains(&centre) {
116            self.include(ordinal, ordinal);
117        }
118        if let Some((previous_ordinal, previous_centre)) = self.previous
119            && previous_centre.min(centre) <= self.upper_bound
120            && previous_centre.max(centre) >= self.lower_bound
121        {
122            self.include(previous_ordinal.min(ordinal), previous_ordinal.max(ordinal));
123        }
124        self.previous = Some((ordinal, centre));
125    }
126
127    fn include(&mut self, first: usize, last: usize) {
128        self.first = Some(self.first.map_or(first, |current| current.min(first)));
129        self.last = Some(self.last.map_or(last, |current| current.max(last)));
130    }
131
132    fn finish(self) -> Option<std::ops::Range<usize>> {
133        let first = self.first?;
134        let last = self.last?;
135        Some(first..last + 1)
136    }
137}

139#[cfg(test)]
140mod selected_channel_window_tests {
141    use super::SelectedChannelWindowSelection;
142
143    fn select(centres: &[f64], bounds: [f64; 2]) -> Option<std::ops::Range<usize>> {
144        let lower = bounds[0].min(bounds[1]);
145        let upper = bounds[0].max(bounds[1]);
146        let mut selection = SelectedChannelWindowSelection::new(lower, upper);
147        for (ordinal, centre) in centres.iter().copied().enumerate() {
148            selection.observe(ordinal, centre);
149        }
150        selection.finish()
151    }
152
153    #[test]
154    fn adjacent_pair_support_handles_descending_and_gapped_centres() {
155        assert_eq!(select(&[4.0, 3.0, 2.0, 1.0], [2.4, 3.6]), Some(0..3));
156        assert_eq!(
157            select(&[100.0, 111.0, 150.0, 220.0], [120.0, 145.0]),
158            Some(1..3)
159        );
160    }
161
162    #[test]
163    fn selection_does_not_add_a_channel_beyond_the_intersecting_pair() {
164        assert_eq!(select(&[0.0, 10.0, 20.0], [1.0, 9.0]), Some(0..2));
165        assert_eq!(select(&[0.0, 10.0, 20.0], [9.0, 11.0]), Some(0..3));
166        assert_eq!(select(&[0.0, 10.0, 20.0], [30.0, 40.0]), None);
167        assert_eq!(select(&[10.0], [9.0, 11.0]), Some(0..1));
168    }
