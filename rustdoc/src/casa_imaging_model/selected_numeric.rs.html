3//! Borrowed source columns. Shape and selected axes belong to the block owner;
4//! values have no execution, publication or per-sample transport state.
5
6use num_complex::Complex32;
7
8use crate::{CorrelationProduct, SelectedObservationRunChannel, SelectedObservationRunRow};
9
10/// Stored visibility precision, selected once for a block.
11#[derive(Clone, Copy)]
12pub enum SelectedNumericVisibility<'a> {
13    /// `FLOAT_DATA` values.
14    Float32(&'a [f32]),
15    /// `DATA` or `CORRECTED_DATA` values.
16    Complex32(&'a [Complex32]),
17}
18
19/// Input weights, retaining the original broadcast representation.
20#[derive(Clone, Copy)]
21pub enum SelectedNumericWeights<'a> {
22    /// One value per correlation for this row.
23    PerRow(&'a [f32]),
24    /// Channel/correlation values.
25    PerChannel(&'a [f32]),
26}
27
28/// A complete selected row in channel/correlation order. The source owner
29/// supplies exact selected axes; no scalar sample envelopes are constructed.
30#[derive(Clone, Copy)]
31pub struct SelectedNumericRow<'a> {
32    /// Row-invariant coordinates and provenance.
33    pub row: &'a SelectedObservationRunRow,
34    /// Selected native channels, in canonical order.
35    pub channels: &'a [SelectedObservationRunChannel],
36    /// Selected correlation coordinates, in canonical order.
37    pub correlations: &'a [CorrelationProduct],
38    /// Physical first channel and stored correlation stride of the source row.
39    pub first_stored_channel: u32,
40    /// Number of contiguous stored channels, including any selection gaps.
41    pub stored_channels: usize,
42    /// Stored correlations per channel, including unselected correlations.
43    pub stored_correlations: usize,
44    /// Channel/correlation payload.
45    pub visibility: SelectedNumericVisibility<'a>,
46    /// Channel/correlation flags.
47    pub flags: &'a [bool],
48    /// Row-broadcast or channelized weights.
49    pub weights: SelectedNumericWeights<'a>,
50}
51
52impl SelectedNumericRow<'_> {
53    /// Check array lengths once at the row boundary, using overflow-safe shape arithmetic.
54    pub fn has_exact_shape(self) -> bool {
55        let Some(samples) = self.stored_channels.checked_mul(self.stored_correlations) else {
56            return false;
57        };
58        samples != 0
59            && !self.channels.is_empty()
60            && !self.correlations.is_empty()
61            && self.channels.iter().all(|channel| {
62                channel
63                    .channel_index
64                    .checked_sub(self.first_stored_channel)
65                    .is_some_and(|offset| (offset as usize) < self.stored_channels)
66            })
67            && self
68                .correlations
69                .iter()
70                .all(|product| (product.correlation_index() as usize) < self.stored_correlations)
71            && self.flags.len() == samples
72            && match self.visibility {
73                SelectedNumericVisibility::Float32(values) => values.len() == samples,
74                SelectedNumericVisibility::Complex32(values) => values.len() == samples,
75            }
76            && match self.weights {
77                SelectedNumericWeights::PerRow(values) => values.len() == self.stored_correlations,
78                SelectedNumericWeights::PerChannel(values) => values.len() == samples,
79            }
80    }
81}
