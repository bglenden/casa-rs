9/// Reconstruction-owned, checked capacities for one compiler attempt.
10///
11/// The artifact ceiling is an independently admitted storage capacity, not a
12/// claimed expansion bound derived from the source sample count. Runtime must
13/// reserve [`Self::workspace_bytes`] in addition to its source, weighting, and
14/// spill-writer buffers before constructing the compiler.
15#[doc(hidden)]
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub struct GriddedNormalCompilationPlan {
18    pub(super) binding: LogicalIdentity,
19    pub(super) maximum_source_samples: usize,
20    pub(super) maximum_correlations: usize,
21    pub(super) maximum_spectral_terms: usize,
22    pub(super) maximum_native_terms: usize,
23    pub(super) maximum_atom_records: usize,
24    pub(super) raw_record_capacity: usize,
25    pub(super) frame_record_capacity: usize,
26    pub(super) band_channel_depth: usize,
27    pub(super) band_output_plane_depth: usize,
28    pub(super) band_count: usize,
29    pub(super) maximum_artifact_bytes: u64,
30    pub(super) frame_header_bytes: usize,
31    pub(super) descriptor_capacity: usize,
32    pub(super) diagnostic_capacity: usize,
33    transient_workspace_bytes: usize,
34    retained_metadata_bytes: usize,
35    workspace_bytes: usize,
36}

39    /// Certify fixed capacities and the descriptor bound of greedy atom packing.
40    /// Both record capacities must fit one complete prediction/accumulation atom.
41    pub fn new(
42        problem: &CompiledProblem,
43        maximum_source_samples: usize,
44        raw_record_capacity: usize,
45        frame_record_capacity: usize,
46        maximum_artifact_bytes: u64,
47        frame_header_bytes: usize,
48    ) -> Result<Self, SpectralOperatorError> {
49        Self::new_with_band_depth(
50            problem,
51            maximum_source_samples,
52            raw_record_capacity,
53            frame_record_capacity,
54            maximum_artifact_bytes,
55            frame_header_bytes,
56            usize::MAX,
57        )
58    }

60    /// Admit full-capacity encoded frames for fixed spectral-channel bands.
61    /// Whole groups are assigned after reduction by their lowest accumulation
62    /// plane; prediction neighbors and cross-band accumulation remain together.
63    pub fn new_with_band_depth(
64        problem: &CompiledProblem,
65        maximum_source_samples: usize,
66        raw_record_capacity: usize,
67        frame_record_capacity: usize,
68        maximum_artifact_bytes: u64,
69        frame_header_bytes: usize,
70        band_channel_depth: usize,
71    ) -> Result<Self, SpectralOperatorError> {
72        let specification = SpectralOperatorSpecification::new(problem)?;
73        let (
74            maximum_correlations,
75            maximum_spectral_terms,
76            maximum_native_terms,
77            maximum_atom_records,
78        ) = compilation_dimensions(problem, &specification)?;
79        if maximum_source_samples < maximum_correlations
80            || raw_record_capacity < maximum_atom_records
81            || frame_record_capacity < maximum_atom_records
82            || maximum_artifact_bytes == 0
83            || frame_header_bytes == 0
84            || band_channel_depth == 0
85        {
86            return Err(SpectralOperatorError::ResidencyOverflow);
87        }
88        let layout = GriddedNormalRecordLayout::for_specification(&specification);
89        let aw = specification.aw_projection().is_some();
90        let channels = specification.slab().total_channels();
91        if channels == 0 {
92            return Err(SpectralOperatorError::ResidencyOverflow);
93        }
94        let band_channel_depth =
95            if !aw && matches!(layout, GriddedNormalRecordLayout::ChannelLocal { .. }) {
96                band_channel_depth.min(channels)
97            } else {
98                channels
99            };
100        let band_count = channels.div_ceil(band_channel_depth);
101        let band_output_plane_depth = band_channel_depth
102            .checked_mul(specification.polarization_count())
103            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
104        let diagnostic_capacity = if aw {
105            problem
106                .inputs()
107                .observation_snapshot()
108                .sources()
109                .iter()
110                .try_fold(0_usize, |total, source| {
111                    total.checked_add(source.selection().spectral_windows().len())
112                })
113                .and_then(|windows| windows.checked_mul(layout.normal_moments()))
114                .ok_or(SpectralOperatorError::ResidencyOverflow)?
115        } else {
116            0
117        };
118        let width = record_bytes(layout, aw)?;
119        // Every nonfinal frame contains at least R-a+1 records. Each band has
120        // at most one tail; source/reduction boundaries never flush.
121        let minimum_nonfinal_bytes = frame_record_capacity
122            .checked_sub(maximum_atom_records)
123            .and_then(|records| records.checked_add(1))
124            .and_then(|records| records.checked_mul(width))
125            .and_then(|bytes| bytes.checked_add(frame_header_bytes))
126            .and_then(|bytes| u64::try_from(bytes).ok())
127            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
128        let descriptor_capacity = maximum_artifact_bytes
129            .checked_div(minimum_nonfinal_bytes)
130            .and_then(|count| count.checked_add(band_count as u64))
131            .and_then(|count| usize::try_from(count).ok())
132            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
133        let arenas = BoundedRecordEncoder::banded_workspace_bytes(
134            layout,
135            aw,
136            raw_record_capacity,
137            frame_record_capacity,
138            maximum_atom_records,
139            band_count,
140        )?;
141        let standard = if aw || matches!(layout, GriddedNormalRecordLayout::Taylor(_)) {
142            0
143        } else {
144            StandardRecordScratch::workspace_bytes(
145                maximum_correlations,
146                maximum_native_terms,
147                maximum_atom_records,
148            )?
149        };
150        let aw_atom = if aw {
151            maximum_atom_records.checked_mul(size_of::<ReducedRecordKey>())
152        } else {
153            Some(0)
154        }
155        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
156        let native_heap = if standard != 0
157            && problem.science().spectral().sampling().kernel()
158                == casa_imaging_model::SpectralKernel::Linear
159        {
160            // Previous native samples may outlive their charged source block.
161            // Equal per-correlation projections need not share one allocation.
162            let carried_projections =
163                casa_imaging_model::SelectedImageDomainProjections::retained_heap_bytes_for_len(
164                    specification.chart_count(),
165                )
166                .and_then(|bytes| bytes.checked_mul(maximum_correlations))
167                .ok_or(SpectralOperatorError::ResidencyOverflow)?;
168            crate::weighting::native_row_heap_bytes(maximum_correlations, maximum_spectral_terms)
169                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?
170                .checked_add(carried_projections)
171                .and_then(|bytes| {
172                    bytes.checked_add(CasaLinearRowResampler::<usize>::retained_group_bytes())
173                })
174                .ok_or(SpectralOperatorError::ResidencyOverflow)?
175        } else {
176            0
177        };
178        let workload = crate::spectral_operator::spectral_operator_workload(
179            &specification,
180            maximum_source_samples,
181            SpectralOperatorPass::InitialMajor,
182        )?;
183        let gridders = workload
184            .convolution_f64_values()
185            .checked_mul(size_of::<f64>())
186            .and_then(|bytes| {
187                specification
188                    .compiler_convolution_metadata_bytes()
189                    .ok()
190                    .and_then(|metadata| bytes.checked_add(metadata))
191            })
192            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
193        let retained_diagnostics = if specification
194            .w_projection()
195            .is_some_and(|contract| contract.maximum_abs_w_lambda() > f64::MIN_POSITIVE)
196        {
197            specification.chart_count()
198        } else {
199            0
200        };
201        let retained_diagnostic_bytes = retained_diagnostics
202            .checked_mul(size_of::<WProjectionDiagnostics>())
203            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
204        let retained_metadata_bytes =
205            retained_manifest_bytes(&specification, descriptor_capacity, retained_diagnostics)?;
206        let transient_workspace_bytes = [
207            arenas,
208            standard,
209            aw_atom,
210            native_heap,
211            // Convolution metadata includes the copied diagnostic output, which
212            // transfers to the manifest rather than dying with the gridders.
213            gridders
214                .checked_sub(retained_diagnostic_bytes)
215                .ok_or(SpectralOperatorError::ResidencyOverflow)?,
216            size_of::<GriddedNormalOperatorCompiler>(),
217            diagnostic_capacity
218                .checked_mul(size_of::<((u32, usize), WeightingScienceAggregate)>())
219                .ok_or(SpectralOperatorError::ResidencyOverflow)?,
220        ]
221        .into_iter()
222        .try_fold(0_usize, |sum, bytes| {
223            sum.checked_add(bytes)
224                .ok_or(SpectralOperatorError::ResidencyOverflow)
225        })?;
226        let workspace_bytes = transient_workspace_bytes
227            .checked_add(retained_metadata_bytes)
228            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
229        Ok(Self {
230            binding: static_binding(&specification),
231            maximum_source_samples,
232            maximum_correlations,
233            maximum_spectral_terms,
234            maximum_native_terms,
235            maximum_atom_records,
236            raw_record_capacity,
237            frame_record_capacity,
238            band_channel_depth,
239            band_output_plane_depth,
240            band_count,
241            maximum_artifact_bytes,
242            frame_header_bytes,
243            descriptor_capacity,
244            diagnostic_capacity,
245            transient_workspace_bytes,
246            retained_metadata_bytes,
247            workspace_bytes,
248        })
249    }

313pub(super) fn retained_manifest_bytes(
314    specification: &SpectralOperatorSpecification,
315    descriptor_capacity: usize,
316    diagnostic_count: usize,
317) -> Result<usize, SpectralOperatorError> {
318    let specification_bytes = specification
319        .owned_heap_bytes()?
320        .checked_add(specification.shared_catalog_heap_bytes()?)
321        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
322    descriptor_capacity
323        .checked_mul(size_of::<BlockDescriptor>())
324        .and_then(|bytes| bytes.checked_add(size_of::<GriddedNormalOperatorManifest>()))
325        .and_then(|bytes| bytes.checked_add(2 * size_of::<usize>()))
326        .and_then(|bytes| {
327            diagnostic_count
328                .checked_mul(size_of::<WProjectionDiagnostics>())
329                .and_then(|diagnostics| bytes.checked_add(diagnostics))
330        })
331        .and_then(|bytes| bytes.checked_add(specification_bytes))
332        .and_then(|bytes| {
333            let channels = if matches!(
334                GriddedNormalRecordLayout::for_specification(specification),
335                GriddedNormalRecordLayout::ChannelLocal { .. }
336            ) {
337                specification.slab().total_channels()
338            } else {
339                0
340            };
341            channels
342                .checked_mul(size_of::<std::ops::Range<usize>>())
343                .and_then(|support| bytes.checked_add(support))
344        })
345        .ok_or(SpectralOperatorError::ResidencyOverflow)
346}

348fn compilation_dimensions(
349    problem: &CompiledProblem,
350    specification: &SpectralOperatorSpecification,
351) -> Result<(usize, usize, usize, usize), SpectralOperatorError> {
352    let correlations = problem
353        .inputs()
354        .observation_snapshot()
355        .sources()
356        .iter()
357        .flat_map(|source| source.selection().correlations())
358        .map(|selection| selection.products().len())
359        .max()
360        .unwrap_or(1)
361        .max(1);
362    let spectral = crate::weighting::maximum_spectral_terms(problem);
363    let native = spectral
364        .checked_mul(specification.chart_count())
365        .and_then(|terms| terms.checked_mul(specification.polarization_count()))
366        .ok_or(SpectralOperatorError::ResidencyOverflow)?;
367    let atom = if specification.aw_projection().is_some() {
368        specification.chart_count()
369    } else if matches!(
370        GriddedNormalRecordLayout::for_specification(specification),
371        GriddedNormalRecordLayout::Taylor(_)
372    ) {
373        1
374    } else {
375        let stencils = spectral
376            .checked_mul(2)
377            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
378        let stencils = if problem.science().spectral().sampling().kernel()
379            == casa_imaging_model::SpectralKernel::Linear
380        {
381            stencils.max(5)
382        } else {
383            stencils
384        };
385        stencils
386            .checked_mul(specification.chart_count())
387            .and_then(|records| records.checked_mul(specification.polarization_count()))
388            .ok_or(SpectralOperatorError::ResidencyOverflow)?
389    };
390    if native == 0 || atom == 0 {
391        return Err(SpectralOperatorError::ResidencyOverflow);
392    }
393    Ok((correlations, spectral, native, atom))
394}

462struct FrameLedger {
463    plan: GriddedNormalCompilationPlan,
464    descriptors: FixedDescriptors,

472impl CompilationFrames {
473    pub(super) fn new(
474        plan: GriddedNormalCompilationPlan,
475        layout: GriddedNormalRecordLayout,
476        aw: bool,
477        observe: bool,
478    ) -> Result<Self, SpectralOperatorError> {
479        let mut encoder = BoundedRecordEncoder::new_banded(
480            layout,
481            aw,
482            plan.raw_record_capacity,
483            plan.frame_record_capacity,
484            plan.maximum_atom_records,
485            plan.band_count,
486            plan.band_output_plane_depth,
487        )?;
488        encoder.observe_timings(observe);
489        let mut descriptors = Vec::new();
490        descriptors
491            .try_reserve_exact(plan.descriptor_capacity)
492            .map_err(|_| SpectralOperatorError::ResidencyOverflow)?;
493        descriptors.resize(plan.descriptor_capacity, BlockDescriptor::default());
494        Ok(Self {
495            encoder,
496            ledger: FrameLedger {
497                plan,
498                descriptors: FixedDescriptors {
499                    storage: descriptors.into_boxed_slice(),
500                    length: 0,
501                },
502                record_count: 0,
503                frame_bytes: 0,
504                sink_time: Duration::ZERO,
505                checksum_time: Duration::ZERO,
506                observe,
507            },
508        })
509    }
510
511    pub(super) fn push_group(
512        &mut self,
513        records: &[ReducedRecordKey],
514        sink: &mut impl FnMut(GriddedNormalOperatorFrame<'_>) -> Result<(), SpectralOperatorError>,
515    ) -> Result<(), SpectralOperatorError> {
516        self.encoder
517            .push_group(records, &mut |bytes, count, planes| {
518                self.ledger.emit(bytes, count, planes, sink)
519            })
520    }

522    pub(super) fn push_taylor(
523        &mut self,

526    ) -> Result<(), SpectralOperatorError> {
527        self.encoder
528            .push_taylor(record, &mut |bytes, count, planes| {
529                self.ledger.emit(bytes, count, planes, sink)
530            })
531    }

533    pub(super) fn finish(
534        &mut self,
535        sink: &mut impl FnMut(GriddedNormalOperatorFrame<'_>) -> Result<(), SpectralOperatorError>,
536    ) -> Result<(), SpectralOperatorError> {
537        self.encoder

541    pub(super) fn measurements(&self, measurements: &mut GriddedNormalCompilationMeasurements) {
542        measurements.frames = self.encoder.flushes();
543        measurements.reduced_records = self.ledger.record_count;
544        measurements.reduced_groups = self.encoder.reduced_groups();
545        measurements.peak_raw_records = self.encoder.peak_raw_records();
546        measurements.peak_frame_records = self.encoder.peak_frame_records();
547    }

549    pub(super) fn timings(&self) -> (GriddedNormalOperatorStageTimings, Duration) {
550        let mut timings = self.encoder.timings();
551        timings.encoding_checksum += self.ledger.checksum_time;
552        (timings, self.ledger.sink_time)
553    }

555    pub(super) fn into_descriptors(self) -> FixedDescriptors {
556        self.ledger.descriptors
557    }

560impl FrameLedger {
561    fn emit(
562        &mut self,
563        encoded: &[u8],
564        record_count: u64,
565        accumulation_output_planes: [u32; 2],
566        sink: &mut impl FnMut(GriddedNormalOperatorFrame<'_>) -> Result<(), SpectralOperatorError>,
567    ) -> Result<(), SpectralOperatorError> {
568        if encoded.is_empty() || record_count == 0 {
569            return Err(SpectralOperatorError::InvalidGriddedRecord);
570        }
571        let frame_bytes = u64::try_from(encoded.len())
572            .ok()
573            .and_then(|bytes| bytes.checked_add(self.plan.frame_header_bytes as u64))
574            .and_then(|bytes| self.frame_bytes.checked_add(bytes))
575            .ok_or(SpectralOperatorError::CoverageOverflow)?;
576        if self.descriptors.length == self.descriptors.storage.len()
577            || frame_bytes > self.plan.maximum_artifact_bytes
578        {
579            return Err(SpectralOperatorError::GriddedCompilationCapacity);
580        }
581        let total_records = self
582            .record_count
583            .checked_add(record_count)
584            .ok_or(SpectralOperatorError::CoverageOverflow)?;
585        let started = self.observe.then(Instant::now);
586        let payload_crc32c = crc32c::crc32c(encoded);
587        if let Some(started) = started {
588            self.checksum_time += started.elapsed();
589        }
590        let started = self.observe.then(Instant::now);
591        sink(GriddedNormalOperatorFrame {
592            sequence: self.descriptors.length as u64,
593            record_count,
594            encoded,
595            payload_crc32c,
596        })?;
597        if let Some(started) = started {
598            self.sink_time += started.elapsed();
599        }
600        self.descriptors.storage[self.descriptors.length] = BlockDescriptor {
601            record_count,
602            payload_crc32c,
603            accumulation_output_planes,
604        };
605        self.descriptors.length += 1;
606        self.record_count = total_records;
607        self.frame_bytes = frame_bytes;
608        Ok(())
609    }
610}

663        for (block, raw, frame, storage, header) in [

665            (correlations, atom, atom - 1, 4096, 72),
666            (correlations - 1, atom, atom, 4096, 72),
667            (correlations, atom, atom, 0, 72),
668            (correlations, atom, atom, 4096, 0),
669            (correlations, usize::MAX, atom, 4096, 72),
670            (correlations, atom, usize::MAX, 4096, 72),
671            (correlations, atom, atom, 4096, usize::MAX),
672            // The former u64::MAX-ceiling row overflowed descriptor accounting
673            // only because a digest-bearing descriptor was 40 bytes. With the
674            // 32-bit checksum descriptor the ceiling is representable, so the
675            // remaining arithmetic rejection is the source-sample ceiling.
676            (usize::MAX, atom, atom, 4096, 72),
677        ] {
678            assert!(
679                matches!(
680                    GriddedNormalCompilationPlan::new(
681                        problem(),
682                        block,
683                        raw,
684                        frame,
685                        storage,
686                        header
687                    ),
688                    Err(SpectralOperatorError::ResidencyOverflow)
689                ),
690                "accepted B={block} U={raw} R={frame} S={storage} h={header}"
691            );
692        }
693        let largest =
694            GriddedNormalCompilationPlan::new(problem(), correlations, atom, atom, u64::MAX, 1)

697        let minimum = GriddedNormalCompilationPlan::new(problem(), correlations, atom, atom, 1, 1)
698            .expect("minimum atom and correlation capacities are admissible");

700        assert_eq!(minimum.frame_record_capacity(), atom);

798    #[test]
799    fn frame_ledger_rejects_storage_overflow_before_calling_sink_or_minting_descriptor() {
800        let (correlations, atom, layout, width) = dimensions();
801        let frame_bytes = 72 + width;
802        for storage in [frame_bytes - 1, frame_bytes] {
803            let plan = GriddedNormalCompilationPlan::new(
804                problem(),
805                correlations,
806                atom,
807                atom,
808                storage as u64,
809                72,
810            )
811            .expect("small explicit storage ceiling");
812            let mut frames =
813                CompilationFrames::new(plan, layout, false, false).expect("fixed frame workspace");
814            let mut emitted = Vec::new();
815            let mut sink = |frame: GriddedNormalOperatorFrame<'_>| {
816                emitted.push((
817                    frame.sequence(),
818                    frame.record_count(),
819                    frame.encoded_bytes().len(),
820                ));
821                Ok(())
822            };
823            frames
824                .push_taylor(key(1), &mut sink)
825                .expect("buffer one record");
826            let result = frames.finish(&mut sink);
827            if storage < frame_bytes {
828                assert!(matches!(
829                    result,
830                    Err(SpectralOperatorError::GriddedCompilationCapacity)
831                ));
832                assert!(matches!(
833                    frames.finish(&mut sink),
834                    Err(SpectralOperatorError::GriddedCompilationPoisoned)
835                ));
836                assert!(emitted.is_empty());
837                assert_eq!(frames.ledger.descriptors.len(), 0);
838                assert_eq!(frames.ledger.record_count, 0);
839                assert_eq!(frames.ledger.frame_bytes, 0);
840            } else {
841                result.expect("exact byte ceiling accepts the frame");
842                assert_eq!(emitted, vec![(0, 1, width)]);
843                assert_eq!(frames.ledger.descriptors.len(), 1);
844                assert_eq!(frames.ledger.record_count, 1);
845                assert_eq!(frames.ledger.frame_bytes, frame_bytes as u64);
846            }
847        }
848    }
