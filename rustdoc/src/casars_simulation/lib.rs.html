4//! The command-line and task surface of casa-rs's simulation tasks:
5//! `simobserve` (simulation Phase 5c, #841), with `simanalyze` and
6//! `simalma` to come.
7//!
8//! As for the imager, both routes resolve the parameters through the
9//! provider-contracts catalog: command-line flags and the catalog-named JSON
10//! of `--json-run` become the same resolved values, CASA's parameters with
11//! CASA's defaults, and those values, and nothing else, make the
12//! [`SimobserveRequest`] the task runs. The host adds what CASA's task takes
13//! from its environment: CASA's data directories, where `alma/simmos` holds
14//! the antenna configurations, and the measures data.

43/// The task's parameters from `resolved`, every simobserve parameter.
44fn parameters_from(
45    resolved: Map<String, Value>,
46) -> Result<(SimobserveTaskParameters, SimobserveParameters), String> {
47    let missing = |name: &str, what: &str| format!("simobserve parameter {name}: {what}");
48    let text = |name: &str| {
49        resolved
50            .get(name)
51            .and_then(Value::as_str)
52            .map(str::to_string)
53            .ok_or_else(|| missing(name, "a string"))
54    };
55    let texts = |name: &str| -> Result<Vec<String>, String> {
56        match resolved.get(name) {
57            Some(Value::String(one)) => Ok(vec![one.clone()]),
58            Some(Value::Array(many)) => many
59                .iter()
60                .map(|item| item.as_str().map(str::to_string))
61                .collect::<Option<_>>()
62                .ok_or_else(|| missing(name, "strings")),
63            _ => Err(missing(name, "a list of strings")),
64        }
65    };
66    // The contract resolves a direction string to a list of one, so one
67    // direction is taken as a string, and several as the pointings.
68    let directions = |name: &str| -> Result<Directions, String> {
69        let mut many = texts(name)?;
70        Ok(match many.len() {
71            0 => Directions::One(String::new()),
72            1 => Directions::One(many.remove(0)),
73            _ => Directions::List(many),
74        })
75    };
76    let flag = |name: &str| {
77        resolved
78            .get(name)
79            .and_then(Value::as_bool)
80            .ok_or_else(|| missing(name, "a boolean"))
81    };
82    let integer = |name: &str| {
83        resolved
84            .get(name)
85            .and_then(Value::as_i64)
86            .ok_or_else(|| missing(name, "an integer"))
87    };
88    let number = |name: &str| {
89        resolved
90            .get(name)
91            .and_then(Value::as_f64)
92            .ok_or_else(|| missing(name, "a number"))
93    };
94    let parameters = SimobserveParameters {
95        project: text("project")?,
96        skymodel: text("skymodel")?,
97        inbright: text("inbright")?,
98        indirection: text("indirection")?,
99        incell: text("incell")?,
100        incenter: text("incenter")?,
101        inwidth: text("inwidth")?,
102        complist: text("complist")?,
103        compwidth: text("compwidth")?,
104        comp_nchan: integer("comp_nchan")?,
105        setpointings: flag("setpointings")?,
106        ptgfile: text("ptgfile")?,
107        integration: text("integration")?,
108        direction: directions("direction")?,
109        mapsize: texts("mapsize")?,
110        maptype: text("maptype")?,
111        pointingspacing: text("pointingspacing")?,
112        caldirection: text("caldirection")?,
113        calflux: text("calflux")?,
114        obsmode: text("obsmode")?,
115        refdate: text("refdate")?,
116        hourangle: text("hourangle")?,
117        totaltime: text("totaltime")?,
118        antennalist: text("antennalist")?,
119        sdantlist: text("sdantlist")?,
120        sdant: integer("sdant")?,
121        outframe: text("outframe")?,
122        thermalnoise: text("thermalnoise")?,
123        user_pwv: number("user_pwv")?,
124        t_ground: number("t_ground")?,
125        t_sky: number("t_sky")?,
126        tau0: number("tau0")?,
127        seed: integer("seed")?,
128        leakage: number("leakage")?,
129        correlator: text("correlator")?,
130        graphics: text("graphics")?,
131        verbose: flag("verbose")?,
132        overwrite: flag("overwrite")?,
133    };
134    Ok((SimobserveTaskParameters(resolved), parameters))
135}

137/// Resolve `parameters`, catalog-named and sparse, over CASA's defaults:
138/// the resolved parameters and the task's.
139pub fn resolve_request(
140    parameters: &SimobserveTaskParameters,
141) -> Result<(SimobserveTaskParameters, SimobserveParameters), String> {
142    let resolved = resolve_plain_values(simobserve_surface()?, workspace()?, &parameters.0)
143        .map_err(|error| format!("resolve simobserve parameters: {error}"))?;
144    parameters_from(resolved)
145}
146
147/// Resolve command-line flags over CASA's defaults.
148pub fn resolve_cli_request(
149    args: &[OsString],
150) -> Result<(SimobserveTaskParameters, SimobserveParameters), String> {
151    let bundle = simobserve_surface()?;
152    let overrides = parse_parameter_cli_overrides(&bundle, args)?;
153    let values = resolve_active_values(bundle, workspace()?, overrides)
154        .map_err(|error| format!("resolve simobserve parameters: {error}"))?;
155    parameters_from(
156        values
157            .into_iter()
158            .map(|(name, value)| (name, value.to_plain_json()))
159            .collect(),
160    )
161}

163/// CASA's data directories: the measures data's candidates, which hold
164/// `alma/simmos` in a CASA data tree.
165fn data_roots() -> Result<Vec<PathBuf>, String> {
166    casa_measures_data::MeasuresRuntime::discover_candidates()
167        .map(|candidates| {
168            candidates
169                .into_iter()
170                .map(|candidate| candidate.path)
171                .collect()
172        })
173        .map_err(|error| format!("CASA's data directories: {error}"))
174}

176/// Run `task`, resolved as `parameters`, with the host's measures data,
177/// CASA data directories and resources.
178pub fn run(
179    parameters: SimobserveTaskParameters,
180    task: SimobserveParameters,
181) -> Result<SimobserveRunTaskResult, String> {
182    let started = Instant::now();
183    let resources = SessionResources::from_controls(&ExecutionControls::default())
184        .map_err(|error| error.to_string())?;
185    let request = SimobserveRequest {
186        parameters: task,
187        data_roots: data_roots()?,
188        options: SimulatorOptions::from_system(resources).map_err(|error| error.to_string())?,
189    };
190    let outcome = simobserve(&request).map_err(|error| error.to_string())?;
191    Ok(SimobserveRunTaskResult {
192        request: parameters,
193        run: SimobserveRunReport {
194            stem: outcome.stem.clone(),
195            directory: outcome.directory.display().to_string(),
196            warnings: outcome.warnings.clone(),
197            elapsed_ns: u64::try_from(started.elapsed().as_nanos()).unwrap_or(u64::MAX),
198        },
199        artifacts: outcome
200            .products
201            .iter()
202            .map(|path| artifact(&outcome.stem, path))
203            .collect(),
204    })
205}

207/// The product at `path`, named by its suffix after `stem`.
208fn artifact(stem: &str, path: &Path) -> SimobserveArtifact {
209    let name = path
210        .file_name()
211        .map(|name| name.to_string_lossy().into_owned())
212        .unwrap_or_default();
213    let kind = name
214        .strip_prefix(stem)
215        .and_then(|suffix| suffix.strip_prefix('.'))
216        .unwrap_or(&name)
217        .to_string();
218    let resource_kind = if kind == "ms" || kind.ends_with(".ms") {
219        RunProductKind::MeasurementSet
220    } else if kind.ends_with("skymodel") || kind.ends_with(".flat") || kind == "newmodel" {
221        RunProductKind::CasaImage
222    } else if kind.ends_with(".txt") {
223        RunProductKind::File
224    } else {
225        RunProductKind::CasaTable
226    };
227    SimobserveArtifact {
228        label: name,
229        kind,
230        path: path.display().to_string(),
231        exists: path.exists(),
232        resource_kind,
233    }
234}

236/// Run the command line: a machine action (`--json-run`, `--json-schema`,
237/// `--protocol-info`), `--help`, or CASA's parameters as flags.
238/// `--managed-output` prints the run's result as JSON.
239pub fn run_with_cli_args(args: impl IntoIterator<Item = OsString>) -> Result<(), String> {
240    let (managed_output, args) = take_managed_output(args.into_iter().collect());
241    let host = casa_task_runtime::TaskCliHost::new(
242        simobserve_task_schema_bundle(),
243        |request: SimobserveTaskRequest| request.execute(),
244    );
245    if let Some(output) = host.dispatch(&args).map_err(|error| error.to_string())? {
246        if managed_output && args.iter().any(|argument| argument == "--json-run") {
247            let SimobserveTaskResult::Run(result) = serde_json::from_str(&output)
248                .map_err(|error| format!("decode the simobserve result: {error}"))?;
249            println!("{}", json(&result)?);
250        } else {
251            println!("{output}");
252        }
253        return Ok(());
254    }
255    if args
256        .iter()
257        .any(|arg| matches!(arg.to_str(), Some("-h" | "--help")))
258    {
259        println!("{}", schema::render_help("simobserve"));
260        return Ok(());
261    }
262    let (parameters, task) = resolve_cli_request(&args)?;
263    let result = run(parameters, task)?;
264    if managed_output {
265        println!("{}", json(&result)?);
266    } else {
267        println!(
268            "Wrote {} products of {} in {}",
269            result.artifacts.len(),
270            result.run.stem,
271            result.run.directory
272        );
273    }
274    Ok(())
275}
276
277fn json(result: &SimobserveRunTaskResult) -> Result<String, String> {
278    serde_json::to_string_pretty(result).map_err(|error| format!("serialize the result: {error}"))
279}

281/// Split `--managed-output [true|false]` from the other arguments.
282fn take_managed_output(raw: Vec<OsString>) -> (bool, Vec<OsString>) {
283    let mut managed = false;
284    let mut args = Vec::with_capacity(raw.len());
285    let mut raw = raw.into_iter().peekable();
286    while let Some(argument) = raw.next() {
287        if argument != "--managed-output" {
288            args.push(argument);
289            continue;
290        }
291        managed = match raw.peek().and_then(|value| value.to_str()) {
292            Some("true") => {
293                raw.next();
294                true
295            }
296            Some("false") => {
297                raw.next();
298                false
299            }
300            _ => true,
301        };
302    }
303    (managed, args)
304}

317    #[test]
318    fn flags_and_json_resolve_alike() {
319        let flags: Vec<OsString> = [
320            "--project",
321            "p",
322            "--antennalist",
323            "alma.out10.cfg",
324            "--direction",
325            "J2000 19h00m00 -23d00m00",
326            "--no-overwrite",
327        ]
328        .into_iter()
329        .map(OsString::from)
330        .collect();
331        let (_, from_flags) = resolve_cli_request(&flags).expect("flags");
332        let json = serde_json::json!({
333            "project": "p",
334            "antennalist": "alma.out10.cfg",
335            "direction": "J2000 19h00m00 -23d00m00",
336            "overwrite": false,
337        });
338        let (_, from_json) = resolve_request(&SimobserveTaskParameters(
339            json.as_object().expect("an object").clone(),
340        ))
341        .expect("JSON");
342        assert_eq!(from_flags, from_json);
343        assert_eq!(
344            from_json.direction,
345            Directions::One("J2000 19h00m00 -23d00m00".to_string())
346        );
347        assert!(!from_json.overwrite);
348        assert_eq!(from_json.ptgfile, "$project.ptg.txt");
349        // A list of one is its direction; several are the pointings.
350        let directions = |value: Value| {
351            let mut json = json.as_object().expect("an object").clone();
352            json.insert("direction".to_string(), value);
353            resolve_request(&SimobserveTaskParameters(json))
354                .expect("JSON")
355                .1
356                .direction
357        };
358        assert_eq!(
359            directions(serde_json::json!(["J2000 19h00m00 -23d00m00"])),
360            from_json.direction
361        );
362        assert_eq!(
363            directions(serde_json::json!([
364                "J2000 19h00m00 -23d00m00",
365                "J2000 19h00m01 -23d00m00"
366            ])),
367            Directions::List(vec![
368                "J2000 19h00m00 -23d00m00".to_string(),
369                "J2000 19h00m01 -23d00m00".to_string()
370            ])
371        );
372        assert_eq!(
373            directions(serde_json::json!([])),
374            Directions::One(String::new())
375        );
376    }
