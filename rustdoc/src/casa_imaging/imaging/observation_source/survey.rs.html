3//! one traversal that reads none: the span of its selected times, which a
4//! pointing query covers, and, for restricted waves, the range of
5//! output-frame conversion factors its selected (field, time)s span.

20/// One partition surveyed.
21#[derive(Clone, Debug, PartialEq)]
22pub(super) struct PartitionSurvey {
23    /// The first and last of its selected times, their time centroids
24    /// included when geometry reads them, in seconds from the partition's
25    /// time origin; `None` when no time is selected.
26    pub(super) span: Option<[f64; 2]>,
27    /// The conversion factors its selected (field, time)s span, when
28    /// surveyed.
29    pub(super) factors: Option<FactorRange>,
30    /// Adjacent interpolation pairs exposed by moving frequency bands.
31    pub(super) partners: Option<InterpolationPartners>,
32}
33
34/// Bytes for the retained survey entries and the pairs of a moving linear
35/// selection. The batching admission holds both while the source is open.
36pub(super) fn retained_bytes(selection: &PartitionSelection, linear: bool) -> usize {
37    size_of::<PartitionSurvey>() * selection.partitions.len()
38        + selection
39            .partitions
40            .iter()
41            .filter(|partition| linear && partition.frequency_bands.is_some())
42            .map(|partition| InterpolationPartners::bytes(partition.channels.len()))
43            .sum::<usize>()
44}

46/// Each of one store's `route`'s partitions surveyed, with their conversion
47/// factors when `factors`. The survey's batches hold at most its admitted
48/// `survey_bytes` ([`super::Batching::survey_bytes`]), in buffers of their
49/// own. Partitions observed at the same times towards the same field in
50/// the same frame share each conversion.
51pub(super) fn survey(
52    route: &StoreRoute,
53    survey_bytes: usize,
54    factors: bool,
55    linear: bool,
56) -> Result<Vec<PartitionSurvey>, SourceError> {
57    let catalogue = route.observation().catalogue();
58    let mut conversions = HashMap::new();
59    let mut buffers = BatchBuffers::default();
60    let origin = MjdHighPrec::new(0.0, 0.0);
61    route
62        .selection
63        .partitions
64        .iter()
65        .map(|partition| {
66            let entry = catalogue.partition(partition.id);
67            let request = requests::survey(
68                (&route.selection, &route.plan),
69                entry,
70                partition,
71                survey_bytes,
72            );
73            let mut traversal = route.observation().bind_read(&request)?;
74            let mut span: Option<[f64; 2]> = None;
75            let mut range = FactorRange::EMPTY;
76            let bands = partition
77                .frequency_bands
78                .as_ref()
79                .filter(|_| factors && linear);
80            let mut partners = bands.map(|_| InterpolationPartners::new(partition.channels.len()));
81            while let Some(event) = traversal.next(&mut buffers)? {
82                let ReadEvent::Batch(batch) = event else {
83                    continue;
84                };
85                let centroids = batch.timing.centroid.map(|centroid| centroid.offsets_s());
86                let times = batch.time.offsets_s();
87                let scale = batch.time.reference();
88                let baselines = batch.samples_per_time();
89                for time in 0..batch.times() {
90                    if !route.selection.time_holds(&batch, time)? {
91                        continue;
92                    }
93                    let mut include = |time_s: f64| {
94                        let [low, high] = span.get_or_insert([time_s; 2]);
95                        *low = low.min(time_s);
96                        *high = high.max(time_s);
97                    };
98                    include(times[time]);
99                    if let Some(centroids) = &centroids {
100                        for sample in time * baselines..(time + 1) * baselines {
101                            include(centroids[sample]);
102                        }
103                    }
104                    if !factors {
105                        continue;
106                    }
107                    let time_mjd_s = batch.time.seconds_since(time, origin);
108                    let field = batch.field[time];
109                    let key = (entry.frequency_frame, field, scale, time_mjd_s.to_bits());
110                    let factor = match conversions.get(&key) {
111                        Some(factor) => *factor,
112                        None => {
113                            let factor = route.plan.frequency_factor(
114                                entry.frequency_frame,
115                                field,
116                                time_mjd_s,
117                                scale,
118                            )?;
119                            if !(factor.is_finite() && factor > 0.0) {
120                                return Err(SourceError::from(
121                                    "an output-frame conversion factor is not positive",
122                                ));
123                            }
124                            conversions.insert(key, factor);
125                            factor
126                        }
127                    };
128                    range = range.with(factor);
129                    if let Some((bands, partners)) = bands.zip(partners.as_mut()) {
130                        include_partners(
131                            route,
132                            partition,
133                            entry,
134                            bands,
135                            (field, time_mjd_s, scale),
136                            partners,
137                        )?;
138                    }
139                }
140            }
141            Ok(PartitionSurvey {
142                span,
143                factors: factors.then_some(range),
144                partners,
145            })
146        })
147        .collect()
148}

150/// Find a time's actual input partners using the same converted centres,
151/// widths and band predicate as the batch packer. Keep their ordinals in
152/// the union selection so the planner can widen a wave before binding it.
153fn include_partners(
154    route: &StoreRoute,
155    partition: &SelectedPartition,
156    entry: &PartitionEntry,
157    bands: &FrequencyBands,
158    (field, epoch, scale): (FieldId, f64, EpochRef),
159    partners: &mut InterpolationPartners,
160) -> Result<(), SourceError> {
161    let plan = &route.plan;
162    let convert = plan.frequency_converter(entry.frequency_frame, field, epoch, scale)?;
163    let factor = plan.frequency_factor(bands.frame().frequency_ref(), field, epoch, scale)?;
164    partners.include_epoch(partition.channels.iter().enumerate().filter_map(
165        |(ordinal, &index)| {
166            let index = index as usize;
167            let native = entry.frequencies_hz[index];
168            let half_width = entry.channel_widths_hz[index].abs() / 2.0;
169            channel_meets_band(bands, factor, convert(native), native, half_width)
170                .then_some(ordinal)
171        },
172    ));
173    Ok(())
174}
