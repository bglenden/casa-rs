50/// Resolve the request's image centre over the surveyed fields.
51pub(super) fn resolve_centre(
52    request: &ImagingRequest,
53    catalogue: &Catalogue,
54    survey: &Survey,
55    engine: &casa_ms::derived::engine::MsCalEngine,
56) -> Result<Centre, PrepareError> {
57    let selected_field = request
58        .phasecenter_field
59        .unwrap_or_else(|| *survey.fields.first().expect("nonempty selection"));
60    if !survey.fields.contains(&selected_field) {
61        return Err(PrepareError::PhaseCentreField {
62            field: selected_field,
63            selected: survey.fields.clone(),
64        });
65    }
66    let field_id = usize::try_from(selected_field).expect("a surveyed field is a FIELD row");
67    let mut stored_phase = catalogue.fields[field_id].phase_centre.clone();
68    let anchor_time = survey.image_time_mjd_seconds;
69    let ephemeris_direction = |name: &str, ephemeris: &PhaseCentreEphemeris| {
70        let direction = engine.ephemeris_direction_j2000(anchor_time, field_id, name, ephemeris)?;
71        let (longitude, latitude) = direction.as_angles();
72        Ok::<_, PrepareError>(SkyDirection::new(
73            DirectionFrame::J2000,
74            longitude,
75            latitude,
76        ))
77    };
78    let (law, ephemeris, main_direction) = match (
79        request.phasecenter.as_deref(),
80        request
81            .phasecenter
82            .as_deref()
83            .and_then(main_phase_center_reference),
84    ) {
85        (None, _) => {
86            let field_direction = field_reference_direction(
87                survey,
88                field_id,
89                engine,
90                anchor_time,
91                &mut stored_phase,
92            )?;
93            (
94                PhaseCentreLaw::Fixed(field_direction),
95                None,
96                field_direction,
97            )
98        }
99        (Some(text), Some(reference)) => {
100            let [longitude, latitude] = parse_phase_center_angles(text)?;
101            stored_phase = MDirection::from_angles(longitude, latitude, reference);
102            let frame =
103                engine.spectral_frame_observatory_direction(anchor_time, stored_phase.clone())?;
104            let direction = j2000_phase_direction(&stored_phase, &frame)?;
105            (PhaseCentreLaw::Fixed(direction), None, direction)
106        }
107        (Some("TRACKFIELD"), None) => {
108            let ephemeris = PhaseCentreEphemeris::tracked_fields(
109                catalogue,
110                survey
111                    .fields
112                    .iter()
113                    .map(|field| usize::try_from(*field).expect("a surveyed field is a FIELD row")),
114            )?;
115            let direction = ephemeris_direction("TRACKFIELD", &ephemeris)?;
116            (
117                PhaseCentreLaw::Ephemeris("TRACKFIELD".to_string()),
118                Some(ephemeris),
119                direction,
120            )
121        }
122        (Some(text), None) => {
123            let ephemeris = named_ephemeris(text, survey, engine)?;
124            let direction = ephemeris_direction(text, &ephemeris)?;
125            (
126                PhaseCentreLaw::Ephemeris(text.to_string()),
127                Some(ephemeris),
128                direction,
129            )
130        }
131    };
132    let phase = MDirection::from_angles(
133        main_direction.longitude_rad(),
134        main_direction.latitude_rad(),
135        DirectionRef::J2000,
136    );
137    let (image_centre, frame) = image_centre(
138        request,
139        engine,
140        anchor_time,
141        &phase,
142        stored_phase,
143        ephemeris.is_some(),
144    )?;
145    let (longitude, latitude) = image_centre.as_angles();
146    refuse_ncp_at_equator(request.projection, latitude)?;
147    Ok(Centre {
148        law,
149        ephemeris,
150        phase,
151        direction: direction_spec(
152            request.imsize,
153            request.cell,
154            request.projection,
155            frame,
156            [longitude, latitude],
157        ),
158        image_centre,
159        field_id,
160    })
161}

163/// Where an analytic primary beam is centred on each image domain: the
164/// pixel of every selected field's phase direction, in the domain's frame.
165///
166/// CASA centres one beam per field on its `PHASE_DIR` and sums them on every
167/// image, outliers included (`SynthesisImagerVi2::makePrimaryBeam`,
168/// `SIMapper::addPB`, `PBMath1D::apply`'s `toPixel`). A pointing 90° or more
169/// from a domain's centre adds nothing there; CASA's `toPixel` fails for it
170/// and `PBMath1D::apply` uses whatever pixel the failed call left.
171///
172/// Under a moving phase centre CASA shifts each field's beam with the source
173/// ([`moving_pointings`]).
174///
175/// # Errors
176///
177/// A selected field that follows an ephemeris under a fixed phase centre;
178/// a selected field whose `PHASE_DIR` polynomial has a nonzero rate, which
179/// CASA evaluates at the field's first selected time; unreadable field
180/// directions or ephemerides, frame conversions or projections.
181pub(super) fn primary_beam_centres(
182    domains: &[PreparedImageDomain],
183    store: &OpenedStore,
184    survey: &Survey,
185    centre: &Centre,
186    engine: &casa_ms::derived::engine::MsCalEngine,
187) -> Result<PrimaryBeamCentres, PrepareError> {
188    let pointings = match (&centre.ephemeris, &centre.law) {
189        (Some(ephemeris), PhaseCentreLaw::Ephemeris(name)) => {
190            moving_pointings(survey, centre, (name, ephemeris), engine)?
191        }
192        (Some(_), _) => return Err(PrepareError::MovingAnalyticPrimaryBeam),
193        (None, _) => fixed_pointings(store, survey, engine)?,
194    };
195    domains
196        .iter()
197        .map(|domain| {
198            let frame = domain
199                .direction
200                .reference_direction()
201                .frame()
202                .direction_ref();
203            let coordinate = domain.coordinates.coordinate(
204                domain
205                    .coordinates
206                    .find_coordinate(CoordinateType::Direction)
207                    .expect("an image domain has a direction coordinate"),
208            );
209            let mut centres = Vec::with_capacity(pointings.len());
210            for (field_id, pointing) in &pointings {
211                let pointing = if frame == DirectionRef::J2000 {
212                    pointing.clone()
213                } else {
214                    let measures = engine.spectral_frame_observatory_direction(
215                        survey.image_time_mjd_seconds,
216                        pointing.clone(),
217                    )?;
218                    pointing.convert_to(frame, &measures)?
219                };
220                let (longitude, latitude) = pointing.as_angles();
221                let pixel =
222                    pointing_pixel(coordinate, [longitude, latitude]).map_err(|source| {
223                        PrepareError::PrimaryBeamPointing {
224                            field_id: *field_id,
225                            source,
226                        }
227                    })?;
228                centres.extend(pixel);
229            }
230            Ok((domain.role.clone(), centres))
231        })
232        .collect()
233}

235/// Each selected field's fixed phase direction, in J2000.
236fn fixed_pointings(
237    store: &OpenedStore,
238    survey: &Survey,
239    engine: &casa_ms::derived::engine::MsCalEngine,
240) -> Result<Vec<(usize, MDirection)>, PrepareError> {
241    // Only an MSv2 states a field's direction as a polynomial in time; the
242    // catalogue keeps its constant term.
243    let field = store
244        .msv2()
245        .map(|observation| observation.measurement_set().field())
246        .transpose()?;
247    survey
248        .fields
249        .iter()
250        .map(|field_id| {
251            let field_id = usize::try_from(*field_id).expect("selected FIELD_IDs are nonnegative");
252            if engine.field_moves(field_id) {
253                return Err(PrepareError::MovingAnalyticPrimaryBeam);
254            }
255            if let Some(field) = &field
256                && phase_direction_varies(field.phase_dir(field_id)?)
257            {
258                return Err(PrepareError::PolynomialFieldPrimaryBeam { field_id });
259            }
260            Ok((field_id, engine.field_direction_j2000(field_id)?.clone()))
261        })
262        .collect()
263}

265/// Each selected field's beam centre under a moving phase centre, as
266/// `SynthesisImagerVi2::makePrimaryBeam` places it: the field's phase
267/// centre at its first selected time, shifted by the source's motion
268/// between then and the image epoch (`getMovingDirection` at both, their
269/// difference of longitude and latitude added as `MDirection::shift` with
270/// `trueAngle=False` adds it). For a field that tracks the source this is
271/// the source at the image epoch, plus the field's offset from it.
272fn moving_pointings(
273    survey: &Survey,
274    centre: &Centre,
275    (name, ephemeris): (&str, &PhaseCentreEphemeris),
276    engine: &casa_ms::derived::engine::MsCalEngine,
277) -> Result<Vec<(usize, MDirection)>, PrepareError> {
278    let image_epoch = survey.image_time_mjd_seconds;
279    let (origin_longitude, origin_latitude) = engine
280        .ephemeris_direction_j2000(image_epoch, centre.field_id, name, ephemeris)?
281        .as_angles();
282    survey
283        .fields
284        .iter()
285        .map(|field_id| {
286            let time = survey.first_time_of(*field_id).unwrap_or(image_epoch);
287            let field_id = usize::try_from(*field_id).expect("selected FIELD_IDs are nonnegative");
288            let (longitude, latitude) = engine
289                .observation_direction_j2000(time, field_id)?
290                .as_angles();
291            let (moved_longitude, moved_latitude) = engine
292                .ephemeris_direction_j2000(time, field_id, name, ephemeris)?
293                .as_angles();
294            let [longitude, latitude] = shifted_angles(
295                [longitude, latitude],
296                [
297                    origin_longitude - moved_longitude,
298                    origin_latitude - moved_latitude,
299                ],
300            );
301            Ok((
302                field_id,
303                MDirection::from_angles(longitude, latitude, DirectionRef::J2000),
304            ))
305        })
306        .collect()
307}

309/// casacore's `MVDirection::shift(lng, lat, False)`: the offsets added to
310/// longitude and latitude, a latitude past a pole folded back over it.
311fn shifted_angles(
312    [longitude, latitude]: [f64; 2],
313    [d_longitude, d_latitude]: [f64; 2],
314) -> [f64; 2] {
315    use std::f64::consts::{FRAC_PI_2, PI};
316    let (longitude, latitude) = (longitude + d_longitude, latitude + d_latitude);
317    if latitude > FRAC_PI_2 {
318        [longitude + PI, PI - latitude]
319    } else if latitude < -FRAC_PI_2 {
320        [longitude + PI, -PI - latitude]
321    } else {
322        [longitude, latitude]
323    }
324}

326/// Whether a `PHASE_DIR` polynomial, shaped `[2, NUM_POLY + 1]`, has a
327/// nonzero rate term. A zero rate is the constant direction exactly.
328fn phase_direction_varies(coefficients: &casa_types::ArrayValue) -> bool {
329    match coefficients {
330        casa_types::ArrayValue::Float64(coefficients) => coefficients
331            .indexed_iter()
332            .any(|(index, value)| index[1] > 0 && *value != 0.0),
333        _ => false,
334    }
335}

337/// The pixel of `pointing`, `[longitude, latitude]` in radians, on the
338/// direction coordinate `coordinate`, or `None` when it lies 90° or more
339/// from the coordinate's reference direction. The SIN projection would
340/// otherwise mirror such a pointing onto the near hemisphere.
341fn pointing_pixel(
342    coordinate: &casa_coordinates::CoordinateModel,
343    pointing: [f64; 2],
344) -> Result<Option<[f64; 2]>, casa_coordinates::CoordinateError> {
345    let reference = coordinate.reference_value();
346    let cosine = reference[1].sin() * pointing[1].sin()
347        + reference[1].cos() * pointing[1].cos() * (pointing[0] - reference[0]).cos();
348    if cosine <= 0.0 {
349        return Ok(None);
350    }
351    let pixel = coordinate.to_pixel(&pointing)?;
352    Ok(Some([pixel[0], pixel[1]]))
353}

355/// casacore's `phaseDirMeas(field_id)` asked at no time, in J2000: the
356/// field's `PHASE_DIR`, or, for a field that follows an ephemeris, the
357/// body's direction at its FIELD row's `TIME` (an MSv4's: the field's first
358/// selected time) offset by `PHASE_DIR`
359/// (`MsCalEngine::moving_field_reference_direction`), which then also
360/// replaces `stored`, the direction the image records. CASA's `tclean`
361/// centres an image without a phase centre there and holds it fixed, also
362/// over a field that moves (#987).
363fn field_reference_direction(
364    survey: &Survey,
365    field_id: usize,
366    engine: &casa_ms::derived::engine::MsCalEngine,
367    anchor_time: f64,
368    stored: &mut MDirection,
369) -> Result<SkyDirection, PrepareError> {
370    let first_time = survey
371        .first_time_of(i32::try_from(field_id).expect("a FIELD row is an i32 id"))
372        .unwrap_or(anchor_time);
373    let j2000 = match engine.moving_field_reference_direction(field_id, first_time)? {
374        Some(reference) => {
375            let frame =
376                engine.spectral_frame_observatory_direction(anchor_time, reference.clone())?;
377            let j2000 = reference.convert_to(DirectionRef::J2000, &frame)?;
378            *stored = reference;
379            j2000
380        }
381        None => engine.field_direction_j2000(field_id)?.clone(),
382    };
383    let (longitude, latitude) = j2000.as_angles();
384    Ok(SkyDirection::new(
385        DirectionFrame::J2000,
386        longitude,
387        latitude,
388    ))
389}

391/// The image's reference direction and the frame it records: ICRS at the
392/// anchor time for a moving source, or the original frame of an admitted
393/// literal or the selected field's stored direction for a fixed source.
394fn image_centre(
395    request: &ImagingRequest,
396    engine: &casa_ms::derived::engine::MsCalEngine,
397    anchor_time: f64,
398    phase: &MDirection,
399    stored_phase: MDirection,
400    moving: bool,
401) -> Result<(MDirection, DirectionFrame), PrepareError> {
402    if moving {
403        let frame = engine.spectral_frame_observatory_direction(anchor_time, phase.clone())?;
404        return Ok((
405            phase.convert_to(DirectionRef::ICRS, &frame)?,
406            DirectionFrame::Icrs,
407        ));
408    }
409    let fixed_literal = request
410        .phasecenter
411        .as_deref()
412        .and_then(main_phase_center_reference)
413        .is_some();
414    let centre = if (request.phasecenter.is_none() || fixed_literal)
415        && matches!(
416            stored_phase.refer(),
417            DirectionRef::J2000 | DirectionRef::ICRS | DirectionRef::B1950 | DirectionRef::GALACTIC
418        ) {
419        stored_phase
420    } else {
421        phase.clone()
422    };
423    let frame = match centre.refer() {
424        DirectionRef::ICRS => DirectionFrame::Icrs,
425        DirectionRef::B1950 => DirectionFrame::B1950,
426        DirectionRef::GALACTIC => DirectionFrame::Galactic,
427        _ => DirectionFrame::J2000,
428    };
429    Ok((centre, frame))
430}

432/// A named ephemeris (or an external table).
433fn named_ephemeris(
434    text: &str,
435    survey: &Survey,
436    engine: &casa_ms::derived::engine::MsCalEngine,
437) -> Result<PhaseCentreEphemeris, PrepareError> {
438    let ephemeris = if Path::new(text).is_dir() {
439        PhaseCentreEphemeris::external(text, EPHEMERIS_BUDGET)?
440    } else {
441        PhaseCentreEphemeris::named(text, EPHEMERIS_BUDGET)?
442    };
443    // The engine observes each attached field towards its own body
444    // (`MsCalEngine::with_field_ephemerides`); a body followed over fields
445    // of which only some move is refused.
446    let mut moving = 0;
447    for field_id in &survey.fields {
448        let field_id = usize::try_from(*field_id).expect("a surveyed field is a FIELD row");
449        moving += usize::from(engine.field_moves(field_id));
450    }
451    if moving != 0 && moving != survey.fields.len() {
452        return Err(PrepareError::MixedEphemerisFields);
453    }
454    Ok(ephemeris)
455}

457/// The image's observation record: telescope and observer of the selected
458/// observations, the first selected time and the image centre. ALMA/ACA
459/// combinations keep the first selected observation's label, as CASA does;
460/// their heterogeneous dish response is planned separately.
461pub(super) fn observation_info(
462    catalogue: &Catalogue,
463    survey: &Survey,
464    centre: &Centre,
465    engine: &casa_ms::derived::engine::MsCalEngine,
466) -> Result<ObsInfo, PrepareError> {
467    let mut labels = std::collections::BTreeSet::new();
468    for id in &survey.observation_ids {
469        let id = usize::try_from(*id).expect("a surveyed observation is an OBSERVATION row");
470        labels.insert(match catalogue.observations.get(id) {
471            Some(observation) => (
472                observation.telescope.clone(),
473                observation.observers.join(", "),
474            ),
475            None => (String::new(), String::new()),
476        });
477    }
478    let alma_aca = labels
479        .iter()
480        .all(|(telescope, _)| matches!(telescope.as_str(), "ALMA" | "ACA"))
481        && labels
482            .iter()
483            .map(|(_, observer)| observer)
484            .collect::<std::collections::BTreeSet<_>>()
485            .len()
486            == 1;
487    if labels.len() != 1 && !alma_aca {
488        return Err(PrepareError::ObservationLabels {
489            observation_ids: survey.observation_ids.clone(),
490            labels,
491        });
492    }
493    let (telescope_name, observer) = if labels.len() == 1 {
494        labels.pop_first().expect("one label pair")
495    } else {
496        let id = *survey
497            .observation_ids
498            .first()
499            .expect("selected observations") as usize;
500        let observation = &catalogue.observations[id];
501        (
502            observation.telescope.clone(),
503            observation.observers.join(", "),
504        )
505    };
506    // ObsInfo::toRecord uses MVDirection::get, preserving the signed atan2
507    // endpoint rather than mapping an exactly positive pi to negative pi.
508    let [pointing_x, pointing_y, _] = centre.image_centre.cosines();
509    let pointing_longitude = if pointing_x == 0.0 && pointing_y == 0.0 {
510        0.0
511    } else {
512        pointing_y.atan2(pointing_x)
513    };
514    Ok(ObsInfo::new(telescope_name)
515        .with_observer(observer)
516        .with_date(MEpoch::from_mjd(
517            survey.image_time_mjd_seconds / 86_400.0,
518            engine.time_reference(),
519        ))
520        .with_telescope_position(engine.observatory_position().clone())
521        .with_pointing_center(pointing_longitude, centre.image_centre.as_angles().1))
522}

524/// Refuse `projection='NCP'` for an image centred on the celestial
525/// equator, where its slant `cot δ₀` is infinite.
526pub(super) fn refuse_ncp_at_equator(
527    projection: Projection,
528    latitude_rad: f64,
529) -> Result<(), PrepareError> {
530    if projection == Projection::Ncp && !(1.0 / latitude_rad.tan()).is_finite() {
531        return Err(PrepareError::NcpAtEquator);
532    }
533    Ok(())
534}

536/// The direction coordinate of `image_size` (x, y) cells of `cell_arcsec`
537/// (x, y) in `projection`, centred on `[longitude, latitude]` in `frame`,
538/// as CASA's `buildCoordinateSystem` makes it. Its poles are WCSLIB's
539/// defaults, which casacore takes when given none: the native pole's
540/// longitude is 180° for a zenithal projection, whose latitude it fixes,
541/// and for one that puts the reference on the native equator (`CAR`,
542/// `SFL`, `MER`, `AIT`) 0°, or 180° for a reference south of the celestial
543/// equator, with the celestial pole sought nearest +90°.
544pub(super) fn direction_spec(
545    image_size: [usize; 2],
546    cell_arcsec: [f64; 2],
547    projection: Projection,
548    frame: DirectionFrame,
549    [longitude, latitude]: [f64; 2],
550) -> DirectionCoordinateSpec {
551    let [cell_x, cell_y] = cell_arcsec.map(|cell| cell * std::f64::consts::PI / (180.0 * 3600.0));
552    let poles = if projection.zenithal() {
553        [180.0, 90.0]
554    } else if latitude >= 0.0 {
555        [0.0, 90.0]
556    } else {
557        [180.0, 90.0]
558    };
559    DirectionCoordinateSpec::new(
560        projection,
561        SkyDirection::new(frame, longitude, latitude),
562        image_reference_pixel(image_size),
563        [-cell_x, cell_y],
564        [[1.0, 0.0], [0.0, 1.0]],
565        poles,
566    )
567}

587impl ImageSpectralCoordinate {
588    /// CASA's nominal increment is the mean of the stored table intervals.
589    pub(super) fn tabular_increment_hz(table: &[f64]) -> f64 {
590        let last = table.len() - 1;
591        (table[last] - table[0]) / last.max(1) as f64
592    }
593
594    /// The native and CASA product-header coordinate use the same table.
595    pub(super) fn coordinate(&self) -> SpectralCoordinate {
596        let axis = match &self.channel_table_hz {
597            Some(table) => SpectralCoordinate::from_tabular(
598                self.frequency_reference,
599                (0..table.len()).map(|pixel| pixel as f64).collect(),
600                table.clone(),
601                table[0],
602                ImageSpectralCoordinate::tabular_increment_hz(table),
603                0.0,
604                self.rest_frequency_hz,
605            )
606            .expect("the compiled optical spectral table is nonempty"),
607            None => SpectralCoordinate::new(
608                self.frequency_reference,
609                self.reference_frequency_hz,
610                self.increment_hz,
611                0.0,
612                self.rest_frequency_hz,
613            ),
614        };
615        let mut axis = axis.with_rest_frequencies(
616            std::iter::once(self.rest_frequency_hz)
617                .chain(self.additional_rest_frequencies_hz.iter().copied())
618                .collect(),
619        );
620        if self.undefined_frame {
621            axis = axis.with_undefined_frame();
622        }
623        axis
624    }
625}

627/// The coordinate system of one image domain's products.
628pub(super) fn image_coordinates(
629    direction: DirectionCoordinateSpec,
630    image_size: [usize; 2],
631    polarizations: &[PolarizationCoordinate],
632    spectral: &ImageSpectralCoordinate,
633    observation: &ObsInfo,
634) -> CoordinateSystem {
635    let mut coordinates = CoordinateSystem::new();
636    // The specification's coordinate, with the poles `direction_spec`
637    // states (WCSLIB's defaults).
638    debug_assert_eq!(
639        direction.reference_pixel(),
640        image_reference_pixel(image_size)
641    );
642    coordinates.add_coordinate(direction.coordinate());
643    coordinates.add_coordinate(StokesCoordinate::new(
644        polarizations.iter().copied().map(stokes_type).collect(),
645    ));
646    coordinates.add_coordinate(spectral.coordinate());
647    *coordinates.obs_info_mut() = observation.clone();
648    coordinates
649}

687/// The frame of a fixed literal in the main image's phasecenter. A bare
688/// ICRS token takes CASA's source-name route and is refused there.
689fn main_phase_center_reference(text: &str) -> Option<DirectionRef> {
690    let mut pieces = text.split_ascii_whitespace();
691    match pieces.next()? {
692        "J2000" => Some(DirectionRef::J2000),
693        "B1950" => Some(DirectionRef::B1950),
694        "ICRS" if pieces.next().is_some() => Some(DirectionRef::ICRS),
695        _ => None,
696    }
697}

699/// A fixed `J2000/B1950 <lon> <lat>` literal used by the existing alternate
700/// PSF and outlier paths, with sexagesimal, degree or radian angles.
701pub(super) fn parse_phase_center_direction(text: &str) -> Result<SkyDirection, PrepareError> {
702    let frame = match text.split_whitespace().next() {
703        Some(frame) if frame.eq_ignore_ascii_case("J2000") => DirectionFrame::J2000,
704        Some(frame) if frame.eq_ignore_ascii_case("B1950") => DirectionFrame::B1950,
705        _ => {
706            return Err(PrepareError::PhaseCentreSyntax {
707                text: text.to_string(),
708            });
709        }
710    };
711    let [longitude, latitude] = parse_phase_center_angles(text)?;
712    Ok(SkyDirection::new(frame, longitude, latitude))
713}
714
715fn parse_phase_center_angles(text: &str) -> Result<[f64; 2], PrepareError> {
716    let mut parts = text.split_whitespace();
717    let (Some(_), Some(longitude), Some(latitude), None) =
718        (parts.next(), parts.next(), parts.next(), parts.next())
719    else {
720        return Err(PrepareError::PhaseCentreSyntax {
721            text: text.to_string(),
722        });
723    };
724    Ok([
725        parse_phase_center_angle(longitude, true)?,
726        parse_phase_center_angle(latitude, false)?,
727    ])
728}
729
730fn parse_phase_center_angle(text: &str, longitude: bool) -> Result<f64, PrepareError> {
731    let lower = text.to_ascii_lowercase();
732    let angle = if let Some(radians) = lower.strip_suffix("rad") {
733        radians.trim().parse::<f64>().ok()
734    } else if let Some(degrees) = lower.strip_suffix("deg") {
735        degrees.trim().parse::<f64>().ok().map(f64::to_radians)
736    } else if longitude {
737        parse_sexagesimal(text, true).map(|hours| hours * std::f64::consts::PI / 12.0)
738    } else {
739        parse_sexagesimal(text, false).map(f64::to_radians)
740    };
741    angle.ok_or_else(|| PrepareError::PhaseCentreAngle {
742        angle: text.to_string(),
743    })
744}
745
746fn parse_sexagesimal(text: &str, hours: bool) -> Option<f64> {
747    let trimmed = text.trim();
748    let sign = if trimmed.starts_with('-') { -1.0 } else { 1.0 };
749    let body = trimmed.trim_start_matches(['+', '-']);
750    let fields = if body.contains(':') {
751        body.split(':').map(str::to_owned).collect::<Vec<_>>()
752    } else if body.contains('h') || body.contains('d') || body.contains('m') || body.contains('s') {
753        body.replace(['h', 'd', 'm', 's'], " ")
754            .split_whitespace()
755            .map(str::to_owned)
756            .collect()
757    } else if !hours && body.matches('.').count() >= 2 {
758        let mut split = body.split('.');
759        let major = split.next()?.to_owned();
760        let minutes = split.next()?.to_owned();
761        let seconds = split.collect::<Vec<_>>().join(".");
762        vec![major, minutes, seconds]
763    } else {
764        return None;
765    };
766    let [major, minutes, seconds] = fields.as_slice() else {
767        return None;
768    };
769    let major = major.parse::<f64>().ok()?;
770    let minutes = minutes.parse::<f64>().ok()?;
771    let seconds = seconds.parse::<f64>().ok()?;
772    if !(major.is_finite()
773        && minutes.is_finite()
774        && seconds.is_finite()
775        && (0.0..60.0).contains(&minutes)
776        && (0.0..60.0).contains(&seconds))
777    {
778        return None;
779    }
780    Some(sign * (major.abs() + minutes / 60.0 + seconds / 3600.0))
781}
