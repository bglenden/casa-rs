2//! CASA ephemeris tables for tests: the layout casacore's `MeasComet`
3//! reads (an `IERS` table of regularly spaced samples with its `MJD0`,
4//! `dMJD` and `VS_*` keywords) and XRADIO's converter reads (`GeoLong`,
5//! `GeoLat`, `GeoDist`, `obsloc`, `posrefsys`, the columns' `UNIT`), both as a MeasurementSet
6//! field's attached ephemeris (`FIELD/EPHEM<id>_<name>.tab`) and as a
7//! `phasecenter` table.
8
9use std::path::{Path, PathBuf};
10
11use casa_tables::{ColumnSchema, Table, TableInfo, TableOptions, TableSchema};
12use casa_types::{PrimitiveType, RecordField, RecordValue, ScalarValue, Value};
13
14/// One sample: MJD (days), RA and DEC (degrees, ICRS), geocentric
15/// distance `Rho` (AU) and radial velocity `RadVel` (AU a day).
16pub type EphemerisRow = [f64; 5];
17
18/// The frame of an ephemeris's directions, as its `posrefsys` keyword
19/// names it.
20#[derive(Clone, Copy, Debug, PartialEq, Eq)]
21pub enum EphemerisFrame {
22    /// `ICRF/ICRS`.
23    Icrs,
24    /// `ICRF/J2000.0`.
25    J2000,
26}
27
28/// Write a geocentric ICRS ephemeris of body `name` at `path`.
29///
30/// # Panics
31///
32/// If the table cannot be built or saved, or `rows` are fewer than two or
33/// not equally spaced in time, as `MeasComet` requires.
34pub fn write_ephemeris_table(path: &Path, name: &str, rows: &[EphemerisRow]) {
35    write_ephemeris_table_in(path, name, rows, EphemerisFrame::Icrs);
36}
37
38/// [`write_ephemeris_table`] with its directions in `frame`.
39///
40/// # Panics
41///
42/// As [`write_ephemeris_table`].
43pub fn write_ephemeris_table_in(
44    path: &Path,
45    name: &str,
46    rows: &[EphemerisRow],
47    frame: EphemerisFrame,
48) {
49    let step = rows[1][0] - rows[0][0];
50    assert!(
51        rows.windows(2)
52            .all(|pair| ((pair[1][0] - pair[0][0]) - step).abs() <= 1.0e-9),
53        "MeasComet reads equally spaced samples"
54    );
55    let schema = TableSchema::new(
56        ["MJD", "RA", "DEC", "Rho", "RadVel"]
57            .into_iter()
58            .map(|column| ColumnSchema::scalar(column, PrimitiveType::Float64))
59            .collect(),
60    )
61    .expect("an ephemeris schema");
62    let mut table = Table::with_schema(schema);
63    for row in rows {
64        table
65            .add_row(RecordValue::new(
66                ["MJD", "RA", "DEC", "Rho", "RadVel"]
67                    .into_iter()
68                    .zip(row)
69                    .map(|(column, value)| {
70                        RecordField::new(column, Value::Scalar(ScalarValue::Float64(*value)))
71                    })
72                    .collect(),
73            ))
74            .expect("an ephemeris row");
75    }
76    for (column, unit) in ["MJD", "RA", "DEC", "Rho", "RadVel"]
77        .into_iter()
78        .zip(["d", "deg", "deg", "AU", "AU/d"])
79    {
80        table.set_column_keywords(
81            column,
82            RecordValue::new(vec![RecordField::new(
83                "UNIT",
84                Value::Scalar(ScalarValue::String(unit.to_string())),
85            )]),
86        );
87    }
88    let posrefsys = match frame {
89        EphemerisFrame::Icrs => "ICRF/ICRS",
90        EphemerisFrame::J2000 => "ICRF/J2000.0",
91    };
92    for (keyword, value) in [
93        ("VS_CREATE", "2026/10/29/00:00"),
94        ("VS_DATE", "2026/10/29/00:00"),
95        ("VS_VERSION", "0001.0001"),
96        ("VS_TYPE", "Table of comet/planetary positions"),
97        ("NAME", name),
98        ("posrefsys", posrefsys),
99        ("obsloc", "GEOCENTRIC"),
100    ] {
101        table.keywords_mut().push(RecordField::new(
102            keyword,
103            Value::Scalar(ScalarValue::String(value.to_string())),
104        ));
105    }
106    // MeasComet indexes rows from MJD0, one step before the first sample.
107    for (keyword, value) in [
108        ("MJD0", rows[0][0] - step),
109        ("dMJD", step),
110        ("GeoLong", 0.0),
111        ("GeoLat", 0.0),
112        ("GeoDist", 0.0),
113    ] {
114        table.keywords_mut().push(RecordField::new(
115            keyword,
116            Value::Scalar(ScalarValue::Float64(value)),
117        ));
118    }
119    table.set_info(TableInfo {
120        table_type: "IERS".to_string(),
121        sub_type: String::new(),
122        readme: Vec::new(),
123    });
124    table
125        .save(TableOptions::new(path))
126        .expect("save the ephemeris table");
127}

129/// Write body `name`'s ephemeris as the table the fields of the
130/// MeasurementSet at `measurement_set` whose `EPHEMERIS_ID` is
131/// `ephemeris_id` follow, and return its path. The caller sets the
132/// fields' `EPHEMERIS_ID`.
133///
134/// # Panics
135///
136/// If the table cannot be built or saved.
137pub fn write_field_ephemeris(
138    measurement_set: &Path,
139    ephemeris_id: i32,
140    name: &str,
141    rows: &[EphemerisRow],
142) -> PathBuf {
143    write_field_ephemeris_in(
144        measurement_set,
145        ephemeris_id,
146        name,
147        rows,
148        EphemerisFrame::Icrs,
149    )
150}
151
152/// [`write_field_ephemeris`] with its directions in `frame`.
153///
154/// # Panics
155///
156/// As [`write_field_ephemeris`].
157pub fn write_field_ephemeris_in(
158    measurement_set: &Path,
159    ephemeris_id: i32,
160    name: &str,
161    rows: &[EphemerisRow],
162    frame: EphemerisFrame,
163) -> PathBuf {
164    let path = measurement_set
165        .join("FIELD")
166        .join(format!("EPHEM{ephemeris_id}_{name}.tab"));
167    write_ephemeris_table_in(&path, name, rows, frame);
168    path
169}
