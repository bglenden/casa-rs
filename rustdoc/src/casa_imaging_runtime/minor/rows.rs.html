11/// One plane's part of a minor cycle: what `SDAlgorithmBase::deconvolve`
12/// hands `addSummaryMinor`.
13#[derive(Clone, Copy, Debug, PartialEq)]
14pub struct PlaneCycleRow {
15    /// The plane's image domain.
16    pub domain: usize,
17    /// The plane's absolute output channel; a Taylor family's first.
18    pub channel: usize,
19    /// The plane's polarization.
20    pub polarization: usize,
21    /// Number of consecutive polarizations represented by this row.
22    pub polarizations: usize,
23    /// Iterations charged to the plane.
24    pub iterations: usize,
25    /// Peak residual inside the mask on entry; zero with no mask.
26    pub start_peak: f64,
27    /// Peak residual inside the mask at the end; zero with no mask.
28    pub peak: f64,
29    /// Peak residual over the plane's pixels on entry.
30    pub start_peak_no_mask: f64,
31    /// Peak residual over the plane's pixels at the end.
32    pub peak_no_mask: f64,
33    /// Signed model flux added by this plane or polarization group.
34    pub model_flux: f64,
35    /// Pixels in the plane's mask.
36    pub mask_sum: usize,
37    /// Why the plane stopped.
38    pub stop: PlaneStop,
39}

41/// The row of a plane with no data.
42pub(super) fn blank_row(
43    key: PlaneKey,
44    statistics: &PlaneStatistics,
45    controls: &CycleControls,
46) -> PlaneCycleRow {
47    PlaneCycleRow {
48        domain: key.domain,
49        channel: key.channel,
50        polarization: key.polarization,
51        polarizations: 1,
52        iterations: 0,
53        start_peak: 0.0,
54        peak: 0.0,
55        start_peak_no_mask: 0.0,
56        peak_no_mask: 0.0,
57        model_flux: 0.0,
58        mask_sum: statistics.mask_sum,
59        stop: blank_plane_stop(controls, statistics),
60    }
61}

63/// A plane with no data as CASA measures it: zero peaks, and its mask's
64/// pixels, which its zero residual doesn't change.
65pub(super) fn blank_statistics(
66    normal: &FinalNormalState,
67    masks: &ImageDomainReconstructionMasks,
68    key: PlaneKey,
69    automask: bool,
70) -> PlaneStatistics {
71    let mask = masks
72        .get(key.domain)
73        .and_then(|mask| {
74            mask.plane(
75                key.channel - normal.slab().core_range().start,
76                key.polarization,
77            )
78        })
79        .expect("a mask for every plane of every image domain");
80    PlaneStatistics {
81        peak_in_mask: 0.0,
82        peak_no_mask: 0.0,
83        mask_sum: mask.iter().filter(|masked| **masked).count(),
84        noise: None,
85        automask,
86    }
87}

89/// The row of a plane that `outcome` cleaned or skipped. A plane with no mask
90/// reports zero peaks inside it, and stops on that zero
91/// (`SDAlgorithmBase::deconvolve`, CAS-14201).
92pub(super) fn cycle_row(
93    key: PlaneKey,
94    statistics: &PlaneStatistics,
95    outcome: &PlaneOutcome,
96    peak_no_mask: f64,
97    controls: &CycleControls,
98) -> PlaneCycleRow {
99    let masked = |peak: f64| if statistics.mask_sum > 0 { peak } else { 0.0 };
100    PlaneCycleRow {
101        domain: key.domain,
102        channel: key.channel,
103        polarization: key.polarization,
104        polarizations: 1,
105        iterations: outcome.iterations,
106        start_peak: masked(outcome.start_peak),
107        peak: masked(outcome.peak),
108        start_peak_no_mask: statistics.peak_no_mask,
109        peak_no_mask,
110        model_flux: outcome.model_flux,
111        mask_sum: statistics.mask_sum,
112        stop: if statistics.mask_sum > 0 {
113            outcome.stop
114        } else {
115            blank_plane_stop(controls, statistics)
116        },
117    }
118}

120#[cfg(test)]
121mod tests {
122    use casa_imaging_deconvolution::Delta;
123
124    use super::*;
125
126    fn controls() -> CycleControls {
127        CycleControls {
128            iterations: 5,
129            threshold: 0.1,
130            threshold_reached: false,
131            gain: 0.1,
132            nsigma: 0.0,
133        }
134    }
135
136    fn statistics(mask_sum: usize) -> PlaneStatistics {
137        PlaneStatistics {
138            peak_in_mask: if mask_sum > 0 { 1.0 } else { 0.0 },
139            peak_no_mask: 2.0,
140            mask_sum,
141            noise: None,
142            automask: false,
143        }
144    }
145
146    fn outcome(stop: PlaneStop) -> PlaneOutcome {
147        PlaneOutcome {
148            delta: Delta::new(1),
149            iterations: 3,
150            components: 3,
151            absolute_flux: 1.0,
152            model_flux: 0.5,
153            start_peak: 2.0,
154            peak: 1.5,
155            stop,
156            trace: Vec::new(),
157            refreshes: 0,
158            end_residual: Vec::new(),
159        }
160    }
161
162    const KEY: PlaneKey = PlaneKey {
163        domain: 1,
164        channel: 2,
165        polarization: 3,
166    };
167
168    /// A cleaned plane's row is its own outcome.
169    #[test]
170    fn a_cleaned_planes_row_holds_its_outcome() {
171        let row = cycle_row(
172            KEY,
173            &statistics(4),
174            &outcome(PlaneStop::Iterations),
175            0.75,
176            &controls(),
177        );
178        assert_eq!(
179            [
180                row.domain,
181                row.channel,
182                row.polarization,
183                row.iterations,
184                row.mask_sum
185            ],
186            [1, 2, 3, 3, 4]
187        );
188        assert_eq!(
189            [
190                row.start_peak,
191                row.peak,
192                row.start_peak_no_mask,
193                row.peak_no_mask,
194                row.model_flux
195            ],
196            [2.0, 1.5, 2.0, 0.75, 0.5]
197        );
198        assert_eq!(row.stop, PlaneStop::Iterations);
199    }
200
201    /// CAS-14201: a plane with no mask holds zero peaks inside it, and
202    /// `checkStop` on that zero stops it at the threshold, though its data
203    /// peak above it, as `SDAlgorithmBase::deconvolve` does.
204    #[test]
205    fn a_plane_with_no_mask_stops_on_a_zero_peak() {
206        let row = cycle_row(
207            KEY,
208            &statistics(0),
209            &outcome(PlaneStop::ZeroMask),
210            2.0,
211            &controls(),
212        );
213        assert_eq!([row.start_peak, row.peak], [0.0, 0.0]);
214        assert_eq!(row.start_peak_no_mask, 2.0);
215        assert_eq!(row.stop, PlaneStop::CycleThreshold);
216    }
217
218    /// A plane with no data holds zero rows and its mask's pixels.
219    #[test]
220    fn a_blank_planes_row_is_empty_with_its_mask() {
221        let row = blank_row(KEY, &statistics(7), &controls());
222        assert_eq!(row.iterations, 0);
223        assert_eq!(row.mask_sum, 7);
224        assert_eq!(
225            [row.peak, row.peak_no_mask, row.model_flux],
226            [0.0, 0.0, 0.0]
227        );
228        assert_eq!(row.stop, PlaneStop::CycleThreshold);
229    }
