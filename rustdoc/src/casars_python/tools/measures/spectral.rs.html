16fn failed(lines: &mut Vec<LogLine>, message: impl Into<String>) -> CasaValue {
17    lines.push(LogLine::error(message));
18    CasaValue::empty_record()
19}
20fn read(record: &RecordArg, lines: &mut Vec<LogLine>, context: &str) -> Option<Measures> {
21    match Measures::read(record) {
22        Ok(values) => Some(values),
23        Err(error) => {
24            failed(lines, format!("{}{context}", error.message()));
25            None
26        }
27    }
28}
29fn define(
30    kind: MeasureKind,
31    rf: &str,
32    value: &Variant,
33    off: &RecordArg,
34    lines: &mut Vec<LogLine>,
35) -> CasaValue {
36    let Some(mut value) = super::define::quantity_of(value, lines) else {
37        return CasaValue::empty_record();
38    };
39    if value.value() == 0.0 && value.unit().name().is_empty() {
40        value = Quantity::new(0.0, "m/s").expect("m/s");
41    }
42    let refer = kind.codes().resolve_or_default(rf);
43    if kind.codes().resolve(rf).is_none() {
44        lines.push(LogLine::warning("Illegal reference frame string."));
45    }
46    let base = match Measure::from_values(kind, refer, &[value]) {
47        Ok(value) => value,
48        Err(error) => return failed(lines, format!("Exception Reports: {}", error.message())),
49    };
50    let mut result = Measures::scalar(base);
51    if !off.0.is_empty() {
52        match Measures::read(off) {
53            Ok(offset) if offset.values[0].kind() == kind => {
54                result.offset = Some(Box::new(offset.into_reference_origin()))
55            }
56            Ok(_) if kind == MeasureKind::Doppler => {
57                return failed(lines, "Illegal offset type specified, not used\n");
58            }
59            Ok(_) => lines.push(LogLine::warning(
60                "Illegal offset type specified, not used\n",
61            )),
62            Err(error) => lines.push(LogLine::warning(format!(
63                "{}Non-measure type offset in measure conversion\n",
64                error.message()
65            ))),
66        }
67    }
68    result.to_value()
69}
70// todoppler reads rfq with casaMFrequency first, including period, wavelength,
71// angular-rate and energy quantities; a parsed non-frequency quantity is only
72// ignored when the input is already a radial velocity.
73fn doppler_rest_hz(value: &Variant) -> Result<Option<f64>, String> {
74    if let Ok((Some(frequency), _)) = crate::convert::measures::frequency(value) {
75        return Ok(Some(frequency.hz()));
76    }
77    if let Variant::Record(record) = value
78        && let Ok(values) = Measures::read(&RecordArg((**record).clone()))
79        && let MeasureValue::Frequency(hz) = values.values[0].value
80    {
81        return Ok(Some(hz));
82    }
83    let q = quantity(value).map_err(|error| error.to_string())?;
84    let hz = casa_core::quanta::Unit::new("Hz").expect("Hz");
85    Ok(q.get_value_in(&hz).ok())
86}

88// tofrequency instead reads a QuantumHolder, requiring frequency dimensions
89// even when rfq is a frequency measure with a frequency-equivalent m0 unit.
90fn quantity_rest_hz(value: &RecordArg) -> Option<f64> {
91    let outer = Variant::Record(value.0.clone().into());
92    let q = quantity(&outer).ok().or_else(|| {
93        let measure = Measures::read(value).ok()?;
94        if measure.values[0].kind() != MeasureKind::Frequency {
95            return None;
96        }
97        quantity(value.0.get("m0")?).ok()
98    })?;
99    let hz = casa_core::quanta::Unit::new("Hz").ok()?;
100    if q.unit().val().dim != hz.val().dim {
101        return None;
102    }
103    crate::convert::holder::frequency_hz(&q)
104}
105fn doppler(value: Measure) -> Result<MDoppler, String> {
106    let MeasureValue::Doppler(v) = value.value else {
107        return Err("Illegal Doppler specified".into());
108    };
109    Ok(MDoppler::new(v, doppler_ref(value.refer)?))
110}
111fn converted(result: Result<Measures, String>, lines: &mut Vec<LogLine>) -> CasaValue {
112    match result {
113        Ok(values) => values.to_value(),
114        Err(error) => failed(lines, error),
115    }
116}

119impl PyMeasures {
120    /// Define a Doppler measure in a CASA convention, retaining its reference offset.
121    #[pyo3(signature=(rf=None,v0=Variant::Empty,off=None),text_signature="($self, rf='RADIO', v0=[], off={})")]
122    fn doppler(
123        &self,
124        py: Python<'_>,
125        #[pyo3(from_py_with=super::arguments::supplied)] rf: Option<Bound<'_, PyAny>>,
126        v0: Variant,
127        #[pyo3(from_py_with=super::arguments::supplied)] off: Option<Bound<'_, PyAny>>,
128    ) -> PyResult<Py<PyAny>> {
129        let rf = super::arguments::reference(rf.as_ref(), "RADIO")?;
130        let off = super::arguments::record(off.as_ref(), "off")?;
131        self.call(py, |_, lines| {
132            Ok(define(MeasureKind::Doppler, &rf, &v0, &off, lines))
133        })
134    }

150    /// Convert a frequency or radial velocity to a Doppler convention.
151    #[pyo3(signature=(rf,v0,rfq=Variant::Empty),text_signature="($self, rf, v0, rfq=[])")]
152    fn todoppler(
153        &self,
154        py: Python<'_>,
155        rf: &Bound<'_, PyAny>,
156        v0: &Bound<'_, PyAny>,
157        rfq: Variant,
158    ) -> PyResult<Py<PyAny>> {
159        let rf = super::arguments::reference(Some(rf), "")?;
160        let v0 = super::arguments::record(Some(v0), "v0")?;
161        self.call(py,|_,lines| {
162            let Some(values)=read(&v0,lines,"Non-measure type radvel/freq in measure conversion\n") else {return Ok(CasaValue::empty_record())};
163            let target=DOPPLER_CODES.resolve_or_default(&rf);
164            let rest=match doppler_rest_hz(&rfq) { Ok(rest)=>rest, Err(error)=>return Ok(failed(lines,error)) };
165            let result=values.map(|v| {
166                // CASA's direct helpers use getValue(); input reference offsets
167                // are not materialized here as they are in me.measure().
168                let dop=match v.value {
169                    MeasureValue::RadialVelocity(ms)=>MDoppler::new(ms/C_MPS,DopplerRef::BETA),
170                    MeasureValue::Frequency(hz)=> {
171                        let ratio = hz/rest.ok_or("Illegal Doppler or rest frequency specified.  todoppler can only convert MFrequency or MRadialVelocity")?;
172                        let squared = ratio * ratio;
173                        MDoppler::new((1.0 - squared)/(1.0 + squared), DopplerRef::BETA)
174                    },
175                    _=>return Err("Illegal Doppler or rest frequency specified.  todoppler can only convert MFrequency or MRadialVelocity".into()),
176                };
177                let dop=dop.convert_to(doppler_ref(target)?,&MeasFrame::new()).map_err(|e|e.to_string())?;
178                Ok(Measure { refer:target,value:MeasureValue::Doppler(dop.value()) })
179            });
180            Ok(converted(result,lines))
181        })
182    }
183    /// Convert a Doppler measure to a frequency using the supplied rest frequency.
184    #[pyo3(signature=(rf,v0,rfq),text_signature="($self, rf, v0, rfq)")]
185    fn tofrequency(
186        &self,
187        py: Python<'_>,
188        rf: &Bound<'_, PyAny>,
189        v0: &Bound<'_, PyAny>,
190        rfq: &Bound<'_, PyAny>,
191    ) -> PyResult<Py<PyAny>> {
192        let rf = super::arguments::reference(Some(rf), "")?;
193        let v0 = super::arguments::record(Some(v0), "v0")?;
194        let rfq = super::arguments::record(Some(rfq), "rfq")?;
195        self.call(py, |state, lines| {
196            let Some(rest) = quantity_rest_hz(&rfq) else {
197                return Ok(failed(lines, "Illegal Doppler or rest frequency specified"));
198            };
199            let Some(values) = read(&v0, lines, "Non-measure type v0 in measure conversion\n")
200            else {
201                return Ok(CasaValue::empty_record());
202            };
203            let values = match values.resolve(state, values.values[0].refer)? {
204                Ok(values) => values,
205                Err(error) => return Ok(failed(lines, error)),
206            };
207            let refer = FREQUENCY_CODES.resolve(&rf).unwrap_or("REST");
208            Ok(converted(
209                values.map(|v| {
210                    Ok(Measure {
211                        refer,
212                        value: MeasureValue::Frequency(
213                            doppler(v)?
214                                .convert_to(DopplerRef::BETA, &MeasFrame::new())
215                                .map_err(|e| e.to_string())?
216                                .shift_frequency_hz(rest),
217                        ),
218                    })
219                }),
220                lines,
221            ))
222        })
223    }
224    /// Convert a Doppler convention to relativistic radial velocity.
225    #[pyo3(signature=(rf,v0),text_signature="($self, rf, v0)")]
226    fn toradialvelocity(
227        &self,
228        py: Python<'_>,
229        rf: &Bound<'_, PyAny>,
230        v0: &Bound<'_, PyAny>,
231    ) -> PyResult<Py<PyAny>> {
232        let rf = super::arguments::reference(Some(rf), "")?;
233        let v0 = super::arguments::record(Some(v0), "v0")?;
234        self.call(py, |state, lines| {
235            let Some(values) = read(&v0, lines, "Non-measure type v0 in measure conversion\n")
236            else {
237                return Ok(CasaValue::empty_record());
238            };
239            let values = match values.resolve(state, values.values[0].refer)? {
240                Ok(values) => values,
241                Err(error) => return Ok(failed(lines, error)),
242            };
243            let refer = RADIAL_VELOCITY_CODES.resolve_or_default(&rf);
244            Ok(converted(
245                values.map(|v| {
246                    let beta = doppler(v)?
247                        .convert_to(DopplerRef::BETA, &MeasFrame::new())
248                        .map_err(|e| e.to_string())?;
249                    Ok(Measure {
250                        refer,
251                        value: MeasureValue::RadialVelocity(beta.value() * C_MPS),
252                    })
253                }),
254                lines,
255            ))
256        })
257    }
258    /// Recover a rest frequency from an observed frequency and a Doppler measure.
259    #[pyo3(signature=(v0,d0),text_signature="($self, v0, d0)")]
260    fn torestfrequency(
261        &self,
262        py: Python<'_>,
263        v0: &Bound<'_, PyAny>,
264        d0: &Bound<'_, PyAny>,
265    ) -> PyResult<Py<PyAny>> {
266        let v0 = super::arguments::record(Some(v0), "v0")?;
267        let d0 = super::arguments::record(Some(d0), "d0")?;
268        self.call(py, |state, lines| {
269            let Some(mut freq) = read(&v0, lines, "Non-frequency measure in conversion\n") else {
270                return Ok(CasaValue::empty_record());
271            };
272            let Some(dop) = read(&d0, lines, "Non-Doppler measure in conversion\n") else {
273                return Ok(CasaValue::empty_record());
274            };
275            let dop = match dop.resolve(state, dop.values[0].refer)? {
276                Ok(values) => values,
277                Err(error) => return Ok(failed(lines, error)),
278            };
279            if freq.values.len() == 1 && dop.values.len() > 1 {
280                freq.values.resize(dop.values.len(), freq.values[0]);
281                freq.vector = true;
282            }
283            if dop.values.len() != 1 && dop.values.len() != freq.values.len() {
284                return Ok(failed(
285                    lines,
286                    "frequency and Doppler array lengths differ (#1252)",
287                ));
288            }
289            let mut i = 0;
290            let result = freq.map(|v| {
291                let MeasureValue::Frequency(hz) = v.value else {
292                    return Err("Illegal frequency or Doppler specified".into());
293                };
294                let dop = doppler(dop.values[i % dop.values.len()])?
295                    .convert_to(DopplerRef::BETA, &MeasFrame::new())
296                    .map_err(|e| e.to_string())?;
297                i += 1;
298                Ok(Measure {
299                    refer: "REST",
300                    value: MeasureValue::Frequency(dop.rest_frequency_hz(hz)),
301                })
302            });
303            Ok(converted(result, lines))
304        })
305    }
