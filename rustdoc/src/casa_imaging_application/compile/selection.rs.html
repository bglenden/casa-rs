38/// A MeasurementSet's per-sample predicates: its fields, UV-distance intervals,
39/// scan numbers, observation identifiers, baselines and the observing modes
40/// its `intent` selector names ([`intents::resolve`]), with the states a
41/// state number named.
42///
43/// The field selector resolves against the catalogue's fields
44/// ([`resolve_fields`]); scan and observation text against CASA's strict
45/// numeric grammar, kept as inclusive intervals so a wide range holds no
46/// more than its terms.
47pub(super) fn row_selection(
48    selection: &MeasurementSetSelection,
49    catalogue: &Catalogue,
50) -> Result<(RowSelection, BTreeSet<StateId>), PrepareError> {
51    let fields = match selection.field.as_deref() {
52        Some(selector) => resolve_fields(selector, catalogue)?,
53        None => IdSelection::All,
54    };
55    let scans = numeric_row_ids(selection.scan.as_deref(), "scan")?;
56    let observations = numeric_row_ids(selection.observation.as_deref(), "observation")?;
57    let uv_distances = match selection.uvrange.as_deref() {
58        Some(text) => UvSelection::Ranges(
59            parse_uv_distance_intervals(text)?
60                .into_iter()
61                .map(uv_range)
62                .collect(),
63        ),
64        None => UvSelection::All,
65    };
66    let (intents, states) = match selection.intent.as_deref() {
67        Some(text) => {
68            let resolved = intents::resolve(text, catalogue)?;
69            (resolved.selection, resolved.states)
70        }
71        None => (IntentSelection::All, BTreeSet::new()),
72    };
73    let baselines = match selection.antenna.as_deref() {
74        Some(text) => resolve_baselines(text, catalogue)?,
75        None => BaselineSelection::All,
76    };
77    Ok((
78        RowSelection::new(fields, uv_distances, intents)
79            .with_scan_observation(scans, observations)
80            .with_baselines(baselines),
81        states,
82    ))
83}

85/// The baselines CASA's antenna expression `text` selects among
86/// `catalogue`'s antennas, through the shared port of CASA's antenna grammar
87/// ([`casa_ms::resolve_antenna_selector`]): names, stations and positions
88/// in catalogue order, an antenna's identifier being its place.
89fn resolve_baselines(text: &str, catalogue: &Catalogue) -> Result<BaselineSelection, PrepareError> {
90    let positions = catalogue
91        .antennas
92        .iter()
93        .map(|antenna| antenna.position.as_itrf())
94        .collect::<Vec<_>>();
95    let rows = catalogue
96        .antennas
97        .iter()
98        .zip(&positions)
99        .map(|(antenna, position_m)| casa_ms::AntennaRow {
100            name: &antenna.name,
101            station: &antenna.station,
102            position_m: *position_m,
103        })
104        .collect::<Vec<_>>();
105    let selected = casa_ms::resolve_antenna_selector(&rows, text)?;
106    Ok(BaselineSelection::from_pairs(
107        rows.len(),
108        |antenna1, antenna2| selected.holds(antenna1, antenna2),
109    ))
110}

112/// The fields `selector` names among `catalogue`'s, through the shared
113/// FIELD selector ([`casa_ms::resolve_field_selector_rows`]): IDs and
114/// ranges bounded by the fields that exist, exact names and simple `*`
115/// patterns over unflagged fields. An MSv2 catalogue's field `i` is
116/// `FIELD_ID` `i`; an MSv4's fields are numbered in its catalogue's order
117/// and named as stored (XRADIO appends the MSv2 field number), with no row
118/// flags. Empty text selects every field; a nonempty selector that matches
119/// none is refused, as imaging never reads an empty selection.
120fn resolve_fields(selector: &str, catalogue: &Catalogue) -> Result<IdSelection, PrepareError> {
121    let rows = catalogue
122        .fields
123        .iter()
124        .map(|field| casa_ms::FieldSelectorRow {
125            name: &field.name,
126            flagged: field.flagged.unwrap_or(false),
127        })
128        .collect::<Vec<_>>();
129    let ids = casa_ms::resolve_field_selector_rows(&rows, selector)?;
130    if ids.is_empty() {
131        return Ok(IdSelection::All);
132    }
133    Ok(IdSelection::Only(
134        ids.into_iter()
135            .map(|id| u32::try_from(id).expect("a resolved FIELD_ID is nonnegative"))
136            .collect(),
137    ))
138}

140/// The scan numbers or observation identifiers `text` names, as inclusive
141/// intervals ([`casa_ms::parse_numeric_row_id_ranges`]); absent or empty
142/// text selects every one.
143fn numeric_row_ids(text: Option<&str>, label: &str) -> Result<IdSelection, PrepareError> {
144    let Some(text) = text.filter(|text| !text.is_empty()) else {
145        return Ok(IdSelection::All);
146    };
147    let ranges = casa_ms::parse_numeric_row_id_ranges(text, label)?;
148    if ranges.is_empty() {
149        return Err(casa_ms::MsError::InvalidInput(format!(
150            "{label} selector must contain nonnegative IDs"
151        ))
152        .into());
153    }
154    Ok(IdSelection::Ranges(
155        ranges
156            .into_iter()
157            .map(|(start, end)| {
158                (
159                    u32::try_from(start).expect("a parsed row ID is nonnegative"),
160                    u32::try_from(end).expect("a parsed row ID is nonnegative"),
161                )
162            })
163            .collect(),
164    ))
165}

167/// One `uvrange` interval as the model states it.
168fn uv_range(interval: UvDistanceInterval) -> UvDistanceRange {
169    let bound = |limit: UvDistanceLimit| {
170        if limit.inclusive {
171            SelectionBound::inclusive(limit.value)
172        } else {
173            SelectionBound::exclusive(limit.value)
174        }
175    };
176    UvDistanceRange::new(
177        interval.lower.map(bound),
178        interval.upper.map(bound),
179        if interval.wavelengths {
180            UvDistanceUnit::Wavelengths
181        } else {
182            UvDistanceUnit::Meters
183        },
184    )
185}

187/// The frequency axes of the catalogue's windows `spw_ids`, each as the
188/// first of its partitions states it.
189pub(super) fn spectral_windows(
190    catalogue: &Catalogue,
191    spw_ids: impl IntoIterator<Item = usize>,
192) -> Vec<SourceSpectralWindow> {
193    spw_ids
194        .into_iter()
195        .collect::<BTreeSet<_>>()
196        .into_iter()
197        .map(|spw_id| {
198            let entry = catalogue
199                .partitions
200                .iter()
201                .find(|entry| entry.window.0 as usize == spw_id)
202                .expect("a candidate's window is a partition's");
203            SourceSpectralWindow {
204                spw_id,
205                frequency_reference: entry.frequency_frame,
206                frequencies_hz: entry.frequencies_hz.clone(),
207                channel_widths_hz: entry.channel_widths_hz.clone(),
208            }
209        })
210        .collect()
211}

213/// The one source frame every selected window shares.
214pub(super) fn one_source_frame(
215    windows: &[SourceSpectralWindow],
216) -> Result<FrequencyRef, PrepareError> {
217    let mut frames = windows.iter().map(|window| window.frequency_reference);
218    let first = frames
219        .next()
220        .expect("a selected sample is in a selected window");
221    if frames.any(|frame| frame != first) {
222        return Err(PrepareError::MixedSourceFrames);
223    }
224    Ok(first)
225}

242/// The spectral windows a request selects: every window; the data
243/// descriptions of a MeasurementSet whose MAIN rows CASA's `spw` selection
244/// keeps, or the one its `ddid` names; or an MSv4's windows `spw` selects
245/// (numbered in the order its datasets name them).
246pub(super) enum WindowChoice {
247    /// Every window.
248    All,
249    /// These MSv4 windows.
250    Windows(BTreeSet<u32>),
251    /// The partitions of these MSv2 `DATA_DESCRIPTION` rows.
252    DataDescriptions(BTreeSet<u32>),
253}
254
255impl WindowChoice {
256    /// Whether partition `id` of `store` is in a selected window.
257    pub(super) fn holds(&self, store: &OpenedStore, id: casa_observation::PartitionId) -> bool {
258        match self {
259            Self::All => true,
260            Self::Windows(windows) => windows.contains(&store.catalogue().partition(id).window.0),
261            Self::DataDescriptions(ddids) => store
262                .msv2()
263                .is_some_and(|observation| ddids.contains(&observation.data_description_id(id))),
264        }
265    }
266}

268/// What a request's `spw` selects: the windows (or data descriptions), and
269/// the channels it names per window. A window `spw` names without channels
270/// (its whole channel range) is absent from `channels`, so `channel_start`
271/// and `channel_count` still apply to it.
272pub(super) struct SpwChoice {
273    pub(super) windows: WindowChoice,
274    pub(super) channels: WindowChannels,
275    /// The first and last channel of each channel range `spw` names, per
276    /// window, in its order (`MSSelection::getChanList`'s rows); a window
277    /// `spw` names without channels is absent, as from `channels`.
278    pub(super) ranges: WindowRanges,
279}

281/// The first and last channel of each channel range of a window, by window
282/// number ([`SpwChoice::ranges`]).
283pub(super) type WindowRanges = BTreeMap<usize, Vec<[usize; 2]>>;

285/// The windows and channels `selection` selects in its `store`
286/// ([`SpwChoice`]): those its `spw` selects, except that `request`'s `ddid`
287/// (casa-rs's own parameter) names the partitions instead while `spw` still
288/// names their channels.
289///
290/// # Errors
291///
292/// A `ddid` of an MSv4, which has no data descriptions, or one that is not
293/// a `DATA_DESCRIPTION` row binding a window and a polarization; an `spw`
294/// selection CASA refuses or this one does not image ([`spw_choice`]).
295pub(super) fn window_choice(
296    request: &ImagingRequest,
297    selection: &MeasurementSetSelection,
298    store: &OpenedStore,
299) -> Result<SpwChoice, PrepareError> {
300    let data_description = data_description_choice(request, store)?;
301    let mut choice = match selection.spw.as_deref().filter(|text| !text.is_empty()) {
302        Some(text) => spw_choice(text, store)?,
303        None => SpwChoice {
304            windows: WindowChoice::All,
305            channels: WindowChannels::new(),
306            ranges: WindowRanges::new(),
307        },
308    };
309    // casa-rs's `ddid` names the partitions; `spw` still names their
310    // channels.
311    if let Some(windows) = data_description {
312        choice.windows = windows;
313    }
314    Ok(choice)
315}

317/// The one MSv2 data description `request`'s `ddid` names, if it names one.
318fn data_description_choice(
319    request: &ImagingRequest,
320    store: &OpenedStore,
321) -> Result<Option<WindowChoice>, PrepareError> {
322    if let Some(ddid) = request.ddid {
323        let observation = store
324            .msv2()
325            .ok_or(PrepareError::NoDataDescriptions { ddid })?;
326        let table = observation.measurement_set().data_description()?;
327        let binds = usize::try_from(ddid)
328            .ok()
329            .filter(|row| *row < table.row_count())
330            .map(|row| {
331                Ok::<_, PrepareError>(
332                    table.spectral_window_id(row)? >= 0 && table.polarization_id(row)? >= 0,
333                )
334            })
335            .transpose()?
336            .unwrap_or(false);
337        if !binds {
338            return Err(PrepareError::InvalidDataDescription { ddid });
339        }
340        return Ok(Some(WindowChoice::DataDescriptions(BTreeSet::from([
341            u32::try_from(ddid).expect("a bound DATA_DESC_ID is nonnegative"),
342        ]))));
343    }
344    Ok(None)
345}

347/// CASA's `spw` selection `text` in `store`, through the port of CASA's
348/// grammar ([`casa_ms::resolve_spw_selector`]): on a MeasurementSet, over
349/// its SPECTRAL_WINDOW and DATA_DESCRIPTION tables, selecting the MAIN rows
350/// of the data descriptions CASA keeps; on an MSv4, over its catalogue's
351/// windows, each standing for one unflagged data description. Each window's
352/// channels are the union of its channel ranges (`getChanList`), as CASA's
353/// iterator slices them; CASA's clamp warnings are logged.
354///
355/// # Errors
356///
357/// A selection CASA refuses, and three it accepts but whose imaging is not
358/// established here: one in parentheses, which keeps every MAIN row while
359/// naming channels of some windows only; one that keeps rows of a window
360/// none of whose channel ranges found a channel; and overlapping channel
361/// ranges of one window, which CASA's iterator would read twice.
362fn spw_choice(text: &str, store: &OpenedStore) -> Result<SpwChoice, PrepareError> {
363    let catalogue = store.catalogue();
364    let selection = match store.msv2() {
365        Some(observation) => observation.measurement_set().resolve_spw_selector(text)?,
366        None => catalogue_spw_selection(text, catalogue)?,
367    };
368    for warning in selection.warnings() {
369        tracing::warn!("{warning}");
370    }
371    let refuse = |reason: String| -> PrepareError {
372        casa_ms::MsError::InvalidInput(format!("spw selection {text:?}: {reason}")).into()
373    };
374    let Some(rows) = selection.row_data_descriptions() else {
375        return Err(refuse(
376            "a parenthesized expression keeps every MAIN row while naming channels of some \
377             windows only, which tclean's imaging of is not established"
378                .to_string(),
379        ));
380    };
381    // Every listed window has a data description among `rows`.
382    let row_windows = selection.spw_ids().iter().copied().collect::<BTreeSet<_>>();
383    let mut channels = WindowChannels::new();
384    let mut named = WindowRanges::new();
385    for &spw in &row_windows {
386        let ranges = selection
387            .channel_list()
388            .iter()
389            .filter(|range| range.spw_id == spw)
390            .collect::<Vec<_>>();
391        let Some(union) = selection.channels_of(spw)? else {
392            return Err(refuse(format!(
393                "window {spw}'s rows are selected but none of its channel ranges found a \
394                 channel, where CASA's iterator would read the whole window"
395            )));
396        };
397        let counted: usize = ranges
398            .iter()
399            .map(|range| range.channels().map_or(0, Iterator::count))
400            .sum();
401        if counted != union.len() {
402            return Err(refuse(format!(
403                "window {spw}'s channel ranges overlap, so CASA's iterator would read some \
404                 channels twice"
405            )));
406        }
407        let whole = catalogue
408            .partitions
409            .iter()
410            .find(|entry| entry.window.0 as usize == spw)
411            .map_or(0, |entry| entry.frequencies_hz.len());
412        if !(union.len() == whole && ranges.len() == 1 && ranges[0].step == 1) {
413            named.insert(
414                spw,
415                ranges
416                    .iter()
417                    .filter_map(|range| {
418                        let mut channels = range.channels()?;
419                        let first = channels.next()?;
420                        Some([first, channels.last().unwrap_or(first)])
421                    })
422                    .collect(),
423            );
424            channels.insert(spw, union);
425        }
426    }
427    let as_u32 = |values: &[usize]| {
428        values
429            .iter()
430            .map(|value| u32::try_from(*value).expect("a SPECTRAL_WINDOW or DATA_DESCRIPTION row"))
431            .collect::<BTreeSet<_>>()
432    };
433    let windows = if store.msv2().is_some() {
434        WindowChoice::DataDescriptions(as_u32(rows))
435    } else {
436        WindowChoice::Windows(as_u32(rows))
437    };
438    Ok(SpwChoice {
439        windows,
440        channels,
441        ranges: named,
442    })
443}

445/// [`casa_ms::resolve_spw_selector`] over an MSv4's catalogue: window `i`
446/// is NAMEd and channelled as its first partition states it, with its
447/// REF_FREQUENCY in the stored frame, its TOTAL_BANDWIDTH (else the sum of
448/// its widths) and its FLAG_ROW, and data description `i` is window `i`.
449fn catalogue_spw_selection(
450    text: &str,
451    catalogue: &Catalogue,
452) -> Result<casa_ms::SpwSelection, PrepareError> {
453    let first_partitions = (0..catalogue.windows.len())
454        .map(|window| {
455            catalogue
456                .partitions
457                .iter()
458                .find(|entry| entry.window.0 as usize == window)
459        })
460        .collect::<Vec<_>>();
461    let rows = catalogue
462        .windows
463        .iter()
464        .zip(&first_partitions)
465        .map(|(window, partition)| {
466            let (name, frequencies, widths) = partition.map_or(("", &[][..], &[][..]), |entry| {
467                (
468                    entry.spectral_window.as_str(),
469                    entry.frequencies_hz.as_slice(),
470                    entry.channel_widths_hz.as_slice(),
471                )
472            });
473            casa_ms::SpectralWindowRow {
474                name,
475                chan_freq_hz: frequencies,
476                chan_width_hz: widths,
477                ref_frequency_hz: window
478                    .reference_frequency
479                    .as_ref()
480                    .map_or(f64::NAN, casa_types::measures::frequency::MFrequency::hz),
481                total_bandwidth_hz: window
482                    .total_bandwidth_hz
483                    .unwrap_or_else(|| widths.iter().map(|width| width.abs()).sum()),
484                flag_row: window.flagged.unwrap_or(false),
485            }
486        })
487        .collect::<Vec<_>>();
488    let descriptions = (0..rows.len())
489        .map(|window| casa_ms::DataDescriptionRow {
490            spectral_window_id: i32::try_from(window).expect("a window count fits i32"),
491            flag_row: false,
492        })
493        .collect::<Vec<_>>();
494    Ok(casa_ms::resolve_spw_selector(&rows, &descriptions, text)?)
495}

497/// The channels of `spw_id` the request selects: those its `spw`
498/// selector names (`named`, from [`SpwChoice::channels`]), else
499/// `channel_start` and `channel_count`.
500pub(super) fn selected_channels(
501    request: &ImagingRequest,
502    named: &WindowChannels,
503    spw_id: usize,
504    frequencies: &[f64],
505) -> Result<Vec<usize>, PrepareError> {
506    if let Some(channels) = named.get(&spw_id) {
507        return Ok(channels.clone());
508    }
509    let start = request.channel_start.unwrap_or(0);
510    let count = request
511        .channel_count
512        .unwrap_or_else(|| frequencies.len().saturating_sub(start));
513    match start.checked_add(count) {
514        Some(end) if count > 0 && end <= frequencies.len() => Ok((start..end).collect()),
515        _ => Err(PrepareError::ChannelRange {
516            spw_id,
517            start,
518            count,
519            channels: frequencies.len(),
520        }),
521    }
522}

524/// The observation selection: the surveyed partitions and samples, and
525/// `channels` per window with the window's whole frequency axis (CASA's
526/// mosaic kernels select by the complete `CHAN_FREQ` and
527/// `|CHAN_WIDTH[0]|`) and its factor to LSRK at its first visibility
528/// buffer, from `lsrk_factors` ([`super::envelope::first_buffer_lsrk_factors`]),
529/// and the per-time frequency selection of a window in `bands`
530/// ([`super::frequency_bands`]).
531pub(super) fn observation_selection(
532    survey: Survey,
533    channels: &WindowChannels,
534    lsrk_factors: &BTreeMap<usize, f64>,
535    bands: &super::spectral::WindowBands,
536) -> Result<ObservationSelection, PrepareError> {
537    let spectral_windows = channels
538        .iter()
539        .map(|(spw_id, channels)| {
540            let window = survey
541                .spectral_windows
542                .iter()
543                .find(|window| window.spw_id == *spw_id)
544                .expect("the channels read are of selected windows");
545            let catalog = window
546                .channel_widths_hz
547                .first()
548                .and_then(|width| {
549                    SpectralWindowCoordinateCatalog::new(window.frequencies_hz.clone(), *width)
550                })
551                .ok_or(PrepareError::SpectralWindowCoordinates { spw_id: *spw_id })?;
552            let factor = lsrk_factors
553                .get(spw_id)
554                .copied()
555                .expect("every selected window has its first-buffer factor");
556            SpectralWindowSelection::new(
557                u32::try_from(*spw_id).expect("SPW ids are nonnegative stored i32 values"),
558                channels
559                    .iter()
560                    .map(|channel| u32::try_from(*channel).expect("a CHAN_FREQ index fits u32"))
561                    .collect(),
562            )
563            .with_coordinate_catalog(catalog)
564            .with_first_buffer_lsrk_factor(factor)
565            .map(|selection| match bands.get(spw_id) {
566                Some(bands) => selection.with_frequency_bands(bands.clone()),
567                None => selection,
568            })
569            .ok_or(PrepareError::SpectralWindowCoordinates { spw_id: *spw_id })
570        })
571        .collect::<Result<Vec<_>, PrepareError>>()?;
572    Ok(ObservationSelection::new(
573        survey.samples,
574        survey.row_selection,
575        survey.partitions,
576        spectral_windows,
577    ))
578}

580/// The visibility column imaged: `DATA` when asked for, else
581/// `CORRECTED_DATA` when the observation `holds` it, else `DATA`.
582///
583/// `datacolumn='corrected'` on a MeasurementSet without `CORRECTED_DATA`
584/// therefore images `DATA`, as CASA's `tclean` does (simulation plan §3,
585/// decision 6): `SIMapperCollection::grid` and `FTMachine::makeImage` turn a
586/// `CORRECTED` request into `OBSERVED` when the column is missing. CASA 6.x
587/// logs no message for the fallback, since the one `SynthesisImager::selectData`
588/// once logged is commented out. casa-rs logs that message as a warning,
589/// because the column imaged is not the one asked for; an unspecified column
590/// falls back without one.
591///
592/// CASA's VisibilityIterator2 also counts the column as missing in an
593/// iteration chunk whose first row's cell is undefined (`existsColumn` tests
594/// `isDefined(0)` on the chunk's attached table), so a partly written
595/// `CORRECTED_DATA` can image `DATA` chunk by chunk. casa-rs decides by the
596/// column's presence alone and images an existing `CORRECTED_DATA` as before:
597/// a deliberate scope limit, not full parity. casa-tables' metadata-only
598/// probe (`array_cell_is_defined_uncached`) answers only for lazy, disk-backed
599/// `TiledShapeStMan` columns, and one probe would not reproduce CASA's
600/// per-chunk rule.
601pub(super) fn visibility_column(
602    holds: impl Fn(OwnerVisibilityColumn) -> bool,
603    requested: Option<DataColumn>,
604) -> Result<OwnerVisibilityColumn, PrepareError> {
605    Ok(match requested {
606        Some(DataColumn::Data) => OwnerVisibilityColumn::Data,
607        _ if holds(OwnerVisibilityColumn::CorrectedData) => OwnerVisibilityColumn::CorrectedData,
608        _ if !holds(OwnerVisibilityColumn::Data) => return Err(PrepareError::NoVisibilityColumn),
609        Some(DataColumn::Corrected) => {
610            tracing::warn!("CORRECTED_DATA column does not exist. Using DATA column instead");
611            OwnerVisibilityColumn::Data
612        }
613        None => OwnerVisibilityColumn::Data,
614    })
615}

617/// The data group holding `visibility` in both stores: `base` for the data
618/// (MSv2 `DATA`, MSv4 `VISIBILITY`), `corrected` for the corrected data.
619pub(super) fn data_group(visibility: OwnerVisibilityColumn) -> &'static str {
620    match visibility {
621        OwnerVisibilityColumn::Data => "base",
622        OwnerVisibilityColumn::CorrectedData => "corrected",
623        OwnerVisibilityColumn::FloatData => "float",
624    }
625}
626
627/// Whether `store` holds `visibility` for the `selected` partitions: a
628/// MeasurementSet when MAIN has its column (its adapter states the `base`
629/// group whether or not MAIN has `DATA`), as CASA asks; an MSv4, whose
630/// partitions each state their own data groups, when every selected
631/// partition has the column's data group.
632pub(super) fn holds(
633    store: &OpenedStore,
634    selected: &[casa_observation::PartitionId],
635    visibility: OwnerVisibilityColumn,
636) -> bool {
637    match store.msv2() {
638        Some(observation) => {
639            let column = match visibility {
640                OwnerVisibilityColumn::Data => VisibilityDataColumn::Data,
641                OwnerVisibilityColumn::CorrectedData => VisibilityDataColumn::CorrectedData,
642                OwnerVisibilityColumn::FloatData => return false,
643            };
644            observation.measurement_set().data_column(column).is_ok()
645        }
646        None => held_by_every(store.catalogue(), selected, data_group(visibility)),
647    }
648}
649
650/// Whether every `selected` partition of `catalogue` has data group `group`.
651fn held_by_every(
652    catalogue: &casa_observation::Catalogue,
653    selected: &[casa_observation::PartitionId],
654    group: &str,
655) -> bool {
656    selected
657        .iter()
658        .all(|id| catalogue.partition(*id).data_group(group).is_some())
659}

761    /// The messages of the warnings an event logs.
762    #[derive(Clone, Default)]
763    struct Warnings(Arc<Mutex<Vec<String>>>);
764
765    impl tracing::field::Visit for Warnings {
766        fn record_debug(&mut self, field: &tracing::field::Field, value: &dyn std::fmt::Debug) {
767            if field.name() == "message" {
768                self.0.lock().expect("warnings").push(format!("{value:?}"));
769            }
770        }
771    }
772
773    impl<S: tracing::Subscriber> Layer<S> for Warnings {
774        fn on_event(&self, event: &tracing::Event<'_>, _: Context<'_, S>) {
775            if *event.metadata().level() == tracing::Level::WARN {
776                event.record(&mut self.clone());
777            }
778        }
779    }

781    /// The column `requested` images in an observation holding `columns`,
782    /// and the warnings choosing it logs.
783    fn chosen(
784        columns: &[OwnerVisibilityColumn],
785        requested: Option<DataColumn>,
786    ) -> (Result<OwnerVisibilityColumn, PrepareError>, Vec<String>) {
787        let warnings = Warnings::default();
788        let chosen = tracing::subscriber::with_default(
789            tracing_subscriber::registry().with(warnings.clone()),
790            || visibility_column(|column| columns.contains(&column), requested),
791        );
792        (chosen, warnings.0.lock().expect("warnings").clone())
793    }
