54/// CASA's calibration-facing legacy imager.
55#[pyclass(name = "imager", module = "casars.tools", frozen)]
56pub(crate) struct PyImager {
57    state: Mutex<ImagerState>,
58}
59
60impl PyImager {
61    fn with_state<T: Send>(
62        &self,
63        py: Python<'_>,
64        operation: impl FnOnce(&mut ImagerState) -> PyResult<T> + Send,
65    ) -> PyResult<T> {
66        py.detach(|| {
67            let mut state = self.state.lock().map_err(runtime_error)?;
68            operation(&mut state)
69        })
70    }
71
72    fn attached(
73        &self,
74        py: Python<'_>,
75        operation: impl FnOnce(&mut ImagerState) -> PyResult<bool> + Send,
76    ) -> PyResult<bool> {
77        let result = self.with_state(py, |state| {
78            state.valid.then(|| operation(state)).transpose()
79        })?;
80        if let Some(result) = result {
81            return Ok(result);
82        }
83        emit(
84            py,
85            LOGGER,
86            &[LogLine::error(
87                "No MeasurementSet has been assigned, please run open.",
88            )],
89        )?;
90        Ok(false)
91    }
92}

103    #[pyo3(signature = (thems = "", compress = false, usescratch = false), text_signature = "($self, thems='', compress=False, usescratch=False)")]
104    fn open(
105        &self,
106        py: Python<'_>,
107        thems: &str,
108        compress: bool,
109        usescratch: bool,
110    ) -> PyResult<bool> {
111        // For this legacy tool both compression choices attach the same MODEL_DATA iterator.
112        let _ = compress;
113        self.with_state(py, |state| {
114            *state = ImagerState::default();
115            let path = if thems.is_empty() {
116                std::env::current_dir().map_err(runtime_error)?
117            } else {
118                PathBuf::from(thems)
119            };
120            let write_access = std::fs::OpenOptions::new()
121                .write(true)
122                .open(path.join("table.dat"))
123                .is_ok();
124            state
125                .attachments
126                .open(
127                    &path,
128                    AttachmentOptions {
129                        write_access,
130                        use_scratch: usescratch,
131                    },
132                )
133                .map_err(runtime_error)?;
134            state.valid = true;
135            state.uses_scratch = usescratch;
136            Ok(true)
137        })
138    }
139
140    fn close(&self, py: Python<'_>) -> PyResult<bool> {
141        self.with_state(py, |state| {
142            *state = ImagerState::default();
143            Ok(true)
144        })
145    }
146
147    fn done(&self, py: Python<'_>) -> PyResult<bool> {
148        self.close(py)
149    }

206    fn setoptions(
207        &self,
208        py: Python<'_>,
209        ftmachine: &str,
210        cache: i64,
211        tile: i32,
212        gridfunction: &str,
213        location: Variant,
214        padding: f64,
215        freqinterp: &str,
216        wprojplanes: i32,
217        epjtablename: &str,
218        applypointingoffsets: bool,
219        dopbgriddingcorrections: bool,
220        cfcachedirname: &str,
221        rotpastep: f64,
222        pastep: f64,
223        pblimit: f64,
224        imagetilevol: options::ImageTileVolume,
225        singleprecisiononly: bool,
226        numthreads: i32,
227        psterm: bool,
228        aterm: bool,
229        mterm: bool,
230        wbawp: bool,
231        conjbeams: bool,
232    ) -> PyResult<bool> {
233        // In the supported A/W geometry the parallel-hand cell set and
234        // prediction agree for either setting of the Mueller term.
235        let _ = mterm;

238                return Ok(false);
239            }

242                cache,
243                tile,
244                convolution: gridfunction,
245                location: &location,
246                padding,
247                interpolation: freqinterp,
248                planes: wprojplanes,
249                single_precision: singleprecisiononly,
250                threads: numthreads,
251            }
252            .options(state.prediction.transform_options())?;
253            let mosaic = ftmachine.eq_ignore_ascii_case("pbmosaic");
254            if mosaic || ftmachine.eq_ignore_ascii_case("pbwproject") {
255                state.prediction.configure_legacy_pbw_with_pointing(
256                    settings,
257                    ImagerLegacyPbWOptions {
258                        mosaic,
259                        cache_directory: PathBuf::from(cfcachedirname),
260                        w_planes: wprojplanes,
261                        apply_pointing_offsets: applypointingoffsets,
262                        correct_primary_beam: dopbgriddingcorrections,
263                        primary_beam_limit: pblimit,
264                    },
265                    (applypointingoffsets && !epjtablename.is_empty())
266                        .then(|| PathBuf::from(epjtablename)),
267                )
268            } else if matches!(
269                ftmachine.to_ascii_lowercase().as_str(),
270                "awproject" | "mawproject"
271            ) {
272                state.prediction.configure_legacy_aw_terms(
273                    settings,
274                    ImagerLegacyAwOptions {
275                        cache_directory: PathBuf::from(cfcachedirname),
276                        w_planes: wprojplanes,
277                        apply_pointing_offsets: applypointingoffsets,
278                        correct_primary_beam: dopbgriddingcorrections,
279                        primary_beam_limit: pblimit,
280                        pa_step_deg: pastep,
281                        rotation_step_deg: rotpastep,
282                        conjugate_beams: conjbeams,
283                        wideband: wbawp,
284                    },
285                    [psterm, aterm],
286                )
287            } else {
288                state.prediction.configure_transform(settings)
289            }
290            .map_err(runtime_error)?;
291            state.prediction.set_image_tile_volume(imagetilevol.0);
292            Ok(true)
293        })
294    }

335    #[pyo3(signature = (dovp = false, usedefaultvp = true, vptable = "", dosquint = false, parangleinc = Variant::Empty, skyposthreshold = Variant::Empty, telescope = "", verbose = true), text_signature = "($self, dovp=False, usedefaultvp=True, vptable='', dosquint=False, parangleinc=[], skyposthreshold=[], telescope='', verbose=True)")]
336    #[allow(clippy::too_many_arguments)]
337    fn setvp(
338        &self,
339        py: Python<'_>,
340        dovp: bool,
341        usedefaultvp: bool,
342        vptable: &str,
343        dosquint: bool,
344        parangleinc: Variant,
345        skyposthreshold: Variant,
346        telescope: &str,
347        verbose: bool,
348    ) -> PyResult<bool> {
349        let angle = |value: &Variant, default| {
350            if value.is_empty_sequence() || matches!(value, Variant::Text(text) if text.is_empty())
351            {
352                Ok(default)
353            } else {
354                quantity_value(value, "deg")
355            }
356        };
357        let parallactic = angle(&parangleinc, 360.)?;
358        let threshold = angle(&skyposthreshold, 180.)?;
359        // SkyEquation chooses its default pattern from the attached observation.
360        let _ = telescope;
361        self.with_state(py, |state| {
362            let pattern = (dovp || !usedefaultvp).then(|| ImagerVoltagePattern {
363                table: (!usedefaultvp).then(|| PathBuf::from(vptable)),
364            });
365            state
366                .prediction
367                .configure_voltage_pattern_rotation(
368                    pattern,
369                    dovp,
370                    dosquint,
371                    parallactic.to_radians(),
372                )
373                .map_err(runtime_error)?;
374            state.reset_sky_model();
375            Ok(())
376        })?;
377        let mut lines = vec![LogLine::info("Setting voltage pattern parameters")];
378        if verbose {
379            lines.push(LogLine::info(format!(
380                "Sky position tolerance is {threshold} degrees"
381            )));
382            lines.push(LogLine::info(if usedefaultvp {
383                "Using system default voltage patterns for each telescope".to_owned()
384            } else {
385                format!("Using user defined voltage patterns in Table {vptable}")
386            }));
387            if dosquint {
388                lines.push(LogLine::info(
389                    "Beam Squint will be included in the VP model",
390                ));
391                lines.push(LogLine::info(format!(
392                    "and the Parallactic Angle increment is {parallactic} degrees"
393                )));
394            }
395        }
396        emit(py, LOGGER, &lines)?;
397        Ok(true)
398    }

400    #[pyo3(signature = (vis = "", nchan = ChannelVector::Default, start = ChannelVector::Default, step = ChannelVector::Default, spw = Variant::Empty, field = Variant::Empty, baseline = Variant::Empty, time = Variant::Empty, scan = Variant::Empty, intent = "", observation = Variant::Empty, uvrange = Variant::Empty, taql = "", usescratch = false, datainmemory = false, writeaccess = true), text_signature = "($self, vis='', nchan=[-1], start=[0], step=[1], spw=[], field=[], baseline=[], time=[], scan=[], intent='', observation=[], uvrange=[], taql='', usescratch=False, datainmemory=False, writeaccess=True)")]
401    #[allow(clippy::too_many_arguments)]
402    fn selectvis(
403        &self,
404        py: Python<'_>,
405        vis: &str,
406        nchan: ChannelVector,
407        start: ChannelVector,
408        step: ChannelVector,
409        spw: Variant,
410        field: Variant,
411        baseline: Variant,
412        time: Variant,
413        scan: Variant,
414        intent: &str,
415        observation: Variant,
416        uvrange: Variant,
417        taql: &str,
418        usescratch: bool,
419        datainmemory: bool,
420        writeaccess: bool,
421    ) -> PyResult<bool> {
422        let selection = MsSelection {
423            spw: wildcard_selector(&spw),
424            field: wildcard_selector(&field),
425            antenna: wildcard_selector(&baseline),
426            timerange: super::calibrater::selector(&time),
427            scan: super::calibrater::selector(&scan),
428            intent: nonempty(intent),
429            observation: super::calibrater::selector(&observation),
430            uvrange: super::calibrater::selector(&uvrange),
431            msselect: nonempty(taql),
432            ..MsSelection::default()
433        };
434        let channels = ImagerChannelSelection {
435            count: nchan.values("nchan", -1)?,
436            start: start.values("start", 0)?,
437            step: step.values("step", 1)?,
438        };
439        let result = self.with_state(py, |state| {
440            let path = (!vis.is_empty()).then(|| Path::new(vis));
441            let explicit_open = state.attachments.inputs().first().is_some_and(|input| {
442                input.mode() == casa_calibration::attachment::AttachmentMode::Open
443            });
444            if !explicit_open && let Some(path) = path {
445                state.uses_scratch = usescratch && !datainmemory;
446                if datainmemory {
447                    // The pinned Multi-MS memory path fails before attaching.
448                    return Err(runtime_error(
449                        "NullTable::getColumn - Table object is empty",
450                    ));
451                }
452                if !path.join("table.dat").is_file() {
453                    state.valid = false;
454                    return Ok(Some(false));
455                }
456            }
457            let options = AttachmentOptions {
458                write_access: writeaccess,
459                use_scratch: usescratch,
460            };
461            let selected =
462                state
463                    .attachments
464                    .select_channels(path, options, selection, Some(channels));
465            match selected {
466                Err(AttachmentError::Unattached) => Ok(None),
467                result => {
468                    let selected = result.map_err(runtime_error)?;
469                    state.valid = selected;
470                    if selected {
471                        state.prediction.reset_fourier_machine();
472                        state.reset_sky_model();
473                    }
474                    Ok(Some(selected))
475                }
476            }
477        })?;
478        if let Some(selected) = result {
479            return Ok(selected);
480        }
481        emit(
482            py,
483            LOGGER,
484            &[LogLine::error(
485                "No MeasurementSet has been assigned, please run open or assign an ms here via the vis parameter",
486            )],
487        )?;
488        Ok(false)
489    }

590    #[pyo3(signature = (objname = "", standard = "", epoch = Variant::Empty, freqs = None, pfx = "predictcomp"), text_signature = "($self, objname='', standard='', epoch=[], freqs=[100000000000.0], pfx='predictcomp')")]
591    fn predictcomp(
592        &self,
593        py: Python<'_>,
594        objname: &str,
595        standard: &str,
596        epoch: Variant,
597        freqs: Option<Variant>,
598        pfx: &str,
599    ) -> PyResult<String> {
600        use casa_calibration::ft::predict_calibrator_component;
601        use casa_core::measures::{EpochRef, MEpoch};
602
603        let epoch = match epoch {
604            Variant::Numbers(values) | Variant::Array { values, .. } => Some(MEpoch::from_mjd(
605                values.first().copied().unwrap_or_default(),
606                EpochRef::UTC,
607            )),
608            Variant::Bool(flag) => Some(MEpoch::from_mjd(f64::from(flag), EpochRef::UTC)),
609            _ => crate::convert::measures::epoch(&epoch)?,
610        };
611        let Some(epoch) = epoch else {
612            return Ok(String::new());
613        };
614        let frequencies = match freqs {
615            Some(Variant::Array { values, .. }) => values,
616            Some(freqs) => flux::coefficients(freqs, "freqs")?,
617            None => vec![1e11],
618        };
619        let workspace = std::env::current_dir().map_err(runtime_error)?;
620        py.detach(|| {
621            let Some(predicted) = predict_calibrator_component(
622                &workspace,
623                objname,
624                standard,
625                &epoch,
626                &frequencies,
627                pfx,
628            )
629            .map_err(runtime_error)?
630            else {
631                return Ok(String::new());
632            };
633            predicted
634                .write_if_absent(&workspace)
635                .map_err(runtime_error)?;
636            Ok(predicted.filename)
637        })
638    }
639
640    fn ssoflux(&self, py: Python<'_>) -> PyResult<bool> {
641        emit(
642            py,
643            LOGGER,
644            &[LogLine::error("ssoflux is no longer supported. Use setjy.")],
645        )?;
646        Ok(false)
647    }
648
649    #[pyo3(signature = (model = None, complist = "", incremental = false, phasecentertime = -1.0), text_signature = "($self, model=[], complist='', incremental=False, phasecentertime=-1.0)")]
650    fn ft(
651        &self,
652        py: Python<'_>,
653        model: Option<&Bound<'_, PyAny>>,
654        complist: &str,
655        incremental: bool,
656        phasecentertime: f64,
657    ) -> PyResult<bool> {
658        let images = model_images(model)?;
659        let components = (!complist.is_empty()).then(|| PathBuf::from(complist));
660        let mut preparation_error = None;
661        let result = self.attached(py, |state| {
662            // The sky equation has separate image and component slots.
663            // A call can fill either vacant slot and predict both together.
664            if (!images.is_empty() && !state.sky_model.images.is_empty())
665                || (components.is_some() && state.sky_model.components.is_some())
666            {
667                return Ok(false);
668            }
669            let model = ImagerPredictionModel {
670                images: if images.is_empty() {
671                    &state.sky_model.images
672                } else {
673                    &images
674                },
675                components: components
676                    .as_deref()
677                    .or(state.sky_model.components.as_deref()),
678                incremental,
679                phase_centre_time: (phasecentertime >= 0.).then_some(phasecentertime),
680            };
681            let prediction = state.prediction.predict_attached_retaining_sources(
682                &state.attachments,
683                model,
684                state.uses_scratch,
685            );
686            if matches!(prediction, Ok(true))
687                || prediction
688                    .as_ref()
689                    .is_err_and(ImagerPrediction::retains_prepared_model)
690            {
691                if !images.is_empty() {
692                    state.sky_model.images = images;
693                }
694                if components.is_some() {
695                    state.sky_model.components = components;
696                }
697            }
698            match prediction {
699                Err(AttachmentError::PredictionPreparation(error)) => {
700                    preparation_error = Some(error.to_string());
701                    Ok(false)
702                }
703                result => result.map_err(runtime_error),
704            }
705        })?;
706        if let Some(error) = preparation_error {
707            emit(py, LOGGER, &[LogLine::error(error)])?;
708        }
709        Ok(result)
710    }
711}

713enum ChannelVector {
714    Default,
715    Values(Vec<i64>),
716    Text,
717}
718
719impl ChannelVector {
720    fn values(self, name: &str, default: i64) -> PyResult<Vec<i64>> {
721        match self {
722            Self::Default => Ok(vec![default]),
723            Self::Values(values) => Ok(values),
724            Self::Text => Err(PyTypeError::new_err(format!(
725                "argument _{name} must not be a string"
726            ))),
727        }
728    }
729}
730
731impl<'a, 'py> FromPyObject<'a, 'py> for ChannelVector {
732    type Error = PyErr;
733
734    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
735        if value.is_instance_of::<PyString>() {
736            return Ok(Self::Text);
737        }
738        if value.is_instance_of::<PyInt>() || value.is_instance_of::<PyFloat>() {
739            return Ok(Self::Values(vec![value.extract::<f64>()? as i64]));
740        }
741        if value.is_instance_of::<PyList>()
742            || value
743                .getattr("ndim")
744                .and_then(|n| n.extract::<usize>())
745                .is_ok_and(|n| n > 0)
746        {
747            let values = value
748                .try_iter()?
749                .map(|item| Ok(item?.extract::<f64>()? as i64))
750                .collect::<PyResult<_>>()?;
751            return Ok(Self::Values(values));
752        }
753        Err(PySystemError::new_err(
754            "<built-in function imager_selectvis> returned a result with an exception set",
755        ))
756    }
757}
758
759fn model_images(model: Option<&Bound<'_, PyAny>>) -> PyResult<Vec<PathBuf>> {
760    let Some(model) = model else {
761        return Ok(Vec::new());
762    };
763    let images = if let Ok(path) = model.cast::<PyString>() {
764        vec![path.to_str()?.to_owned()]
765    } else if let Ok(paths) = model.cast::<PyList>() {
766        paths
767            .iter()
768            .map(|path| path.extract::<String>())
769            .collect::<PyResult<_>>()?
770    } else {
771        // CASA's string-vector binding retains the empty default for tuples and arrays.
772        Vec::new()
773    };
774    if matches!(images.as_slice(), [name] if name.is_empty()) {
775        return Ok(Vec::new());
776    }
777    Ok(images.into_iter().map(PathBuf::from).collect())
778}
