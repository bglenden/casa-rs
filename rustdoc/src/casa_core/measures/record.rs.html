2//! Record serialization for measures, compatible with C++ `MeasureHolder`.
3//!
4//! C++ casacore stores measures in `TableRecord` values with a specific field
5//! layout. This module provides `to_record` / `from_record` functions that
6//! produce and consume [`RecordValue`] in the same format, ensuring
7//! interoperability with C++ tables.
8//!
9//! Deferred: GitHub issue #13 tracks splitting the shared `RecordValue` /
10//! `Value` data model out of the table layer so
11//! `casa-types` no longer depends on table-storage details here.
12//!
13//! ## MEpoch record format
14//!
15//! ```text
16//! type:  "epoch"
17//! refer: "UTC"          (EpochRef::as_str())
18//! m0:    { value: <f64>, unit: "d" }
19//! ```
20//!
21//! ## MPosition record format
22//!
23//! ```text
24//! type:  "position"
25//! refer: "ITRF"
26//! m0:    { value: <lon_rad>, unit: "rad" }
27//! m1:    { value: <lat_rad>, unit: "rad" }
28//! m2:    { value: <radius_m>, unit: "m" }
29//! ```
30//!
31//! Note: C++ always serializes MPosition in spherical form (lon, lat, radius)
32//! regardless of reference type.
33

44/// Creates a quantity-like sub-record `{ value: <v>, unit: <u> }`.
45fn quantity_record(value: f64, unit: &str) -> RecordValue {
46    RecordValue::new(vec![
47        RecordField::new("value", Value::Scalar(ScalarValue::Float64(value))),
48        RecordField::new("unit", Value::Scalar(ScalarValue::String(unit.to_owned()))),
49    ])
50}
51
52/// Extracts an f64 from a quantity sub-record.
53fn extract_quantity_value(rec: &RecordValue) -> Result<f64, MeasureError> {
54    match rec.get("value") {
55        Some(Value::Scalar(ScalarValue::Float64(v))) => Ok(*v),
56        Some(Value::Scalar(ScalarValue::Float32(v))) => Ok(f64::from(*v)),
57        Some(Value::Scalar(ScalarValue::Int32(v))) => Ok(f64::from(*v)),
58        _ => Err(MeasureError::InvalidRecord {
59            reason: "quantity sub-record missing 'value' field".to_owned(),
60        }),
61    }
62}
63
64/// Extracts a string field from a record.
65fn extract_string(rec: &RecordValue, field: &str) -> Result<String, MeasureError> {
66    match rec.get(field) {
67        Some(Value::Scalar(ScalarValue::String(s))) => Ok(s.clone()),
68        _ => Err(MeasureError::InvalidRecord {
69            reason: format!("missing or non-string field '{field}'"),
70        }),
71    }
72}
73
74/// Extracts a sub-record from a record field.
75fn extract_sub_record<'a>(
76    rec: &'a RecordValue,
77    field: &str,
78) -> Result<&'a RecordValue, MeasureError> {
79    match rec.get(field) {
80        Some(Value::Record(r)) => Ok(r),
81        _ => Err(MeasureError::InvalidRecord {
82            reason: format!("missing or non-record field '{field}'"),
83        }),
84    }
85}

91/// Serializes an [`MEpoch`] to a [`RecordValue`] compatible with C++ `MeasureHolder`.
92pub fn epoch_to_record(epoch: &MEpoch) -> RecordValue {
93    let code = epoch.refer().casacore_name();
94    let refer = if epoch.is_razing() {
95        format!("R_{code}")
96    } else {
97        code.into()
98    };
99    let mut record = RecordValue::new(vec![
100        RecordField::new(
101            "type",
102            Value::Scalar(ScalarValue::String("epoch".to_owned())),
103        ),
104        RecordField::new("refer", Value::Scalar(ScalarValue::String(refer))),
105        RecordField::new(
106            "m0",
107            Value::Record(quantity_record(epoch.value().as_mjd(), "d")),
108        ),
109    ]);
110    if let Some(offset) = epoch.offset() {
111        record.push(RecordField::new(
112            "offset",
113            Value::Record(epoch_to_record(offset)),
114        ));
115    }
116    record
117}
118
119/// Deserializes an [`MEpoch`] from a [`RecordValue`] in C++ `MeasureHolder` format.
120pub fn epoch_from_record(rec: &RecordValue) -> Result<MEpoch, MeasureError> {
121    read_epoch_record(rec, 0)
122}
123
124fn read_epoch_record(rec: &RecordValue, depth: usize) -> Result<MEpoch, MeasureError> {
125    if depth > 128 {
126        return Err(MeasureError::InvalidRecord {
127            reason: "epoch offsets are too deeply nested".into(),
128        });
129    }
130    let type_str = extract_string(rec, "type")?;
131    if !type_str.eq_ignore_ascii_case("epoch") {
132        return Err(MeasureError::InvalidRecord {
133            reason: format!("expected type 'epoch', got '{type_str}'"),
134        });
135    }
136
137    let refer_str = extract_string(rec, "refer")?;
138    let base = refer_str
139        .strip_prefix("R_")
140        .or_else(|| refer_str.strip_prefix("r_"));
141    let refer: EpochRef = base.unwrap_or(&refer_str).parse()?;
142
143    let epoch = MEpoch::new(epoch_record_value(rec)?, refer).with_razing(base.is_some());
144    match rec.get("offset") {
145        Some(Value::Record(offset)) => Ok(epoch.with_offset(read_epoch_record(offset, depth + 1)?)),
146        Some(_) => Err(MeasureError::InvalidRecord {
147            reason: "epoch offset must be a measure record".into(),
148        }),
149        None => Ok(epoch),
150    }
151}

153fn epoch_record_value(rec: &RecordValue) -> Result<MjdHighPrec, MeasureError> {
154    let mut length = None;
155    let mut day = 0.0;
156    let mut fraction = 0.0;
157    for key in ["m0", "m1", "m2"] {
158        let Some(Value::Record(component)) = rec.get(key) else {
159            break;
160        };
161        let (value, count) = match component.get("value") {
162            Some(Value::Array(ArrayValue::Float64(array))) => (array.first().copied(), array.len()),
163            Some(Value::Array(ArrayValue::Float32(array))) => {
164                (array.first().copied().map(f64::from), array.len())
165            }
166            _ => (Some(extract_quantity_value(component)?), 1),
167        };
168        if length.is_some_and(|previous| previous != count) {
169            return Err(MeasureError::InvalidRecord {
170                reason: format!("Illegal number of values in MeasureHolder {key}"),
171            });
172        }
173        length = Some(count);
174        let value = value.ok_or_else(|| MeasureError::InvalidRecord {
175            reason: format!("empty epoch quantity {key}"),
176        })?;
177        let unit = match component.get("unit") {
178            Some(_) => extract_string(component, "unit")?,
179            None => "d".into(),
180        };
181        let quantity = crate::quanta::Quantity::new(value, &unit).map_err(|error| {
182            MeasureError::InvalidRecord {
183                reason: error.to_string(),
184            }
185        })?;
186        let days = MEpoch::from_quantity(&quantity, EpochRef::UTC)?.value();
187        day += days.day();
188        fraction += days.frac();
189    }
190    Ok(MjdHighPrec::new(day, fraction))
191}

197/// Serializes an [`MPosition`] to a [`RecordValue`] compatible with C++ `MeasureHolder`.
198///
199/// C++ `MVPosition::getRecordValue()` always stores positions in geocentric
200/// spherical form (lon, lat, radius) for ITRF. For WGS84, we store the
201/// geodetic coordinates (lon, lat, height) as-is.
202pub fn position_to_record(pos: &MPosition) -> RecordValue {
203    let (v0, v1, v2, u0, u1, u2) = match pos.refer() {
204        PositionRef::ITRF => {
205            let (lon, lat, r) = pos.as_spherical();
206            (lon, lat, r, "rad", "rad", "m")
207        }
208        PositionRef::WGS84 => {
209            let vals = pos.values();
210            (vals[0], vals[1], vals[2], "rad", "rad", "m")
211        }
212    };
213
214    RecordValue::new(vec![
215        RecordField::new(
216            "type",
217            Value::Scalar(ScalarValue::String("position".to_owned())),
218        ),
219        RecordField::new(
220            "refer",
221            Value::Scalar(ScalarValue::String(pos.refer().as_str().to_owned())),
222        ),
223        RecordField::new("m0", Value::Record(quantity_record(v0, u0))),
224        RecordField::new("m1", Value::Record(quantity_record(v1, u1))),
225        RecordField::new("m2", Value::Record(quantity_record(v2, u2))),
226    ])
227}
228
229/// Deserializes an [`MPosition`] from a [`RecordValue`] in C++ `MeasureHolder` format.
230///
231/// The record is expected to contain spherical coordinates (lon, lat, radius).
232/// The result is returned in ITRF (converted from spherical to Cartesian) if
233/// the refer is ITRF, or as WGS84 if refer is WGS84.
234pub fn position_from_record(rec: &RecordValue) -> Result<MPosition, MeasureError> {
235    let type_str = extract_string(rec, "type")?;
236    if type_str != "position" {
237        return Err(MeasureError::InvalidRecord {
238            reason: format!("expected type 'position', got '{type_str}'"),
239        });
240    }
241
242    let refer_str = extract_string(rec, "refer")?;
243    let refer: PositionRef = refer_str.parse()?;
244
245    let m0 = extract_sub_record(rec, "m0")?;
246    let lon = angular_quantity(m0)?;
247    let m1 = extract_sub_record(rec, "m1")?;
248    let lat = angular_quantity(m1)?;
249    let m2 = extract_sub_record(rec, "m2")?;
250    let radius = extract_quantity_value(m2)?;
251
252    match refer {
253        PositionRef::ITRF => {
254            // Convert spherical (lon, lat, radius) back to Cartesian.
255            // The radius in the record is the geocentric radius, but C++
256            // MeasureHolder actually stores geodetic height for ITRF.
257            // For maximum compatibility, use gd2gc with height derived from radius.
258            // However, the C++ record format stores (lon, lat, radius) where
259            // radius is the geocentric distance. We reconstruct Cartesian directly.
260            let x = radius * lat.cos() * lon.cos();
261            let y = radius * lat.cos() * lon.sin();
262            let z = radius * lat.sin();
263            Ok(MPosition::new_itrf(x, y, z))
264        }
265        PositionRef::WGS84 => {
266            // For WGS84, the record stores geodetic (lon, lat, height).
267            // But the m2 field is labeled "m" (radius in m). In practice
268            // C++ MVPosition stores the height for WGS84.
269            Ok(MPosition::new_wgs84(lon, lat, radius))
270        }
271    }
272}

278/// Serializes an [`MDirection`] to a [`RecordValue`] compatible with C++ `MeasureHolder`.
279///
280/// Format: `{ type: "direction", refer: "J2000", m0: { value: lon_rad, unit: "rad" }, m1: { value: lat_rad, unit: "rad" } }`
281pub fn direction_to_record(dir: &MDirection) -> RecordValue {
282    let (lon, lat) = dir.as_angles();
283    RecordValue::new(vec![
284        RecordField::new(
285            "type",
286            Value::Scalar(ScalarValue::String("direction".to_owned())),
287        ),
288        RecordField::new(
289            "refer",
290            Value::Scalar(ScalarValue::String(dir.refer().as_str().to_owned())),
291        ),
292        RecordField::new("m0", Value::Record(quantity_record(lon, "rad"))),
293        RecordField::new("m1", Value::Record(quantity_record(lat, "rad"))),
294    ])
295}
296
297/// Deserialize a direction, resolving units and recursively adding any offset
298/// in its parent frame. Frame-dependent offset conversions require a frame and
299/// return an error here rather than dropping the offset.
300pub fn direction_from_record(rec: &RecordValue) -> Result<MDirection, MeasureError> {
301    let type_str = extract_string(rec, "type")?;
302    if type_str != "direction" {
303        return Err(MeasureError::InvalidRecord {
304            reason: format!("expected type 'direction', got '{type_str}'"),
305        });
306    }
307
308    let refer_str = extract_string(rec, "refer")?;
309    let refer: DirectionRef = refer_str.parse()?;
310
311    let m0 = extract_sub_record(rec, "m0")?;
312    let lon = angular_quantity(m0)?;
313    let m1 = extract_sub_record(rec, "m1")?;
314    let lat = angular_quantity(m1)?;
315
316    let direction = MDirection::from_angles(lon, lat, refer);
317    if let Some(offset) = rec.get("offset") {
318        let Value::Record(offset) = offset else {
319            return Err(MeasureError::InvalidRecord {
320                reason: "direction offset must be a record".into(),
321            });
322        };
323        let offset = direction_from_record(offset)?.convert_to(refer, &super::MeasFrame::new())?;
324        let a = direction.cosines();
325        let b = offset.cosines();
326        return Ok(MDirection::from_cosines(
327            std::array::from_fn(|i| a[i] + b[i]),
328            refer,
329        ));
330    }
331    Ok(direction)
332}

338/// Serializes an [`MFrequency`] to a [`RecordValue`] compatible with C++ `MeasureHolder`.
339///
340/// Format: `{ type: "frequency", refer: "LSRK", m0: { value: hz, unit: "Hz" } }`
341pub fn frequency_to_record(freq: &MFrequency) -> RecordValue {
342    RecordValue::new(vec![
343        RecordField::new(
344            "type",
345            Value::Scalar(ScalarValue::String("frequency".to_owned())),
346        ),
347        RecordField::new(
348            "refer",
349            Value::Scalar(ScalarValue::String(freq.refer().as_str().to_owned())),
350        ),
351        RecordField::new("m0", Value::Record(quantity_record(freq.hz(), "Hz"))),
352    ])
353}
354
355/// Deserialize a frequency-like quantity and recursively add any offset in
356/// its parent frame. Conversions requiring an unavailable frame return an error.
357pub fn frequency_from_record(rec: &RecordValue) -> Result<MFrequency, MeasureError> {
358    let type_str = extract_string(rec, "type")?;
359    if type_str != "frequency" {
360        return Err(MeasureError::InvalidRecord {
361            reason: format!("expected type 'frequency', got '{type_str}'"),
362        });
363    }
364
365    let refer_str = extract_string(rec, "refer")?;
366    let refer: FrequencyRef = refer_str.parse()?;
367
368    let m0 = extract_sub_record(rec, "m0")?;
369    let mut hz = crate::quanta::casacore::frequency_hz(&scalar_quantity(m0)?).ok_or_else(|| {
370        MeasureError::InvalidRecord {
371            reason: "frequency quantity has incompatible units".into(),
372        }
373    })?;
374    if let Some(offset) = rec.get("offset") {
375        let Value::Record(offset) = offset else {
376            return Err(MeasureError::InvalidRecord {
377                reason: "frequency offset must be a record".into(),
378            });
379        };
380        hz += frequency_from_record(offset)?
381            .convert_to(refer, &super::MeasFrame::new())?
382            .hz();
383    }
384    Ok(MFrequency::new(hz, refer))
385}

465fn scalar_quantity(record: &RecordValue) -> Result<crate::quanta::Quantity, MeasureError> {
466    crate::quanta::Quantity::new(
467        extract_quantity_value(record)?,
468        &extract_string(record, "unit")?,
469    )
470    .map_err(|error| MeasureError::InvalidRecord {
471        reason: error.to_string(),
472    })
473}
474
475fn angular_quantity(record: &RecordValue) -> Result<f64, MeasureError> {
476    scalar_quantity(record)?
477        .get_value_in(&crate::quanta::Unit::new("rad").expect("rad"))
478        .map_err(|error| MeasureError::InvalidRecord {
479            reason: error.to_string(),
480        })
481}

488        use super::*;
489        let literal = |lon, lat| {
490            RecordValue::new(vec![
491                RecordField::new(
492                    "type",
493                    Value::Scalar(ScalarValue::String("direction".into())),
