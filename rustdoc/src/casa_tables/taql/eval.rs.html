2//! Expression evaluator for TaQL.
3//!
4//! Evaluates [`Expr`] nodes against table row data,
5//! producing [`ExprValue`] results with automatic type promotion.
6//!
7//! # Type promotion hierarchy
8//!
9//! Arithmetic promotes `Int → Float → Complex`, matching C++ TaQL. Bool
10//! requires an explicit conversion before arithmetic.
11//! All integer types widen to `i64`, all floats to `f64`.
12//!
13//! # C++ reference
14//!
15//! `TableExprNode.cc`, `TableExprNodeBinary.cc`.
16

27/// A dynamically typed value produced by expression evaluation.
28///
29/// This is the runtime representation of values in the TaQL evaluator.
30/// Arithmetic promotes Int → Float → Complex; explicit conversion helpers
31/// also accept Bool, which arithmetic operators reject.
32///
33/// Additional variants:
34/// - `DateTime` stores Modified Julian Date (MJD) as f64, matching C++ casacore.
35/// - `Array` stores a flat vector of homogeneous values with an n-dimensional shape.
36#[derive(Debug, Clone)]
37pub enum ExprValue {
38    Bool(bool),
39    Int(i64),
40    Float(f64),
41    Quantity(Quantity),
42    Complex(Complex64),
43    String(String),
44    /// Modified Julian Date stored as fractional days (MJD = JD − 2_400_000.5).
45    DateTime(f64),
46    /// N-dimensional array with shape and flat data (row-major order).
47    Array(ArrayValue),
48    /// A compiled regex pattern for matching.
49    Regex {
50        pattern: String,
51        flags: String,
52    },
53    Null,
54}
55
56/// A dynamically-typed array value.
57///
58/// Elements are stored in row-major (C) order. `shape` gives the extent
59/// of each axis; the product of shape elements equals `data.len()`.
60///
61/// C++ equivalent: `Array<T>` / `MaskedArray<T>` in `TableExprNode`.
62#[derive(Debug, Clone)]
63pub struct ArrayValue {
64    /// Extent of each dimension (innermost-last, row-major).
65    pub shape: Vec<usize>,
66    /// Flat element storage in row-major order.
67    pub data: Vec<ExprValue>,
68}

70impl PartialEq for ExprValue {
71    fn eq(&self, other: &Self) -> bool {
72        match (self, other) {
73            (Self::Bool(a), Self::Bool(b)) => a == b,
74            (Self::Int(a), Self::Int(b)) => a == b,
75            (Self::Float(a), Self::Float(b)) => a == b,
76            (Self::Quantity(a), Self::Quantity(b)) => {
77                a.unit().conformant(b.unit()) && a.get_si_value() == b.get_si_value()
78            }
79            (Self::Complex(a), Self::Complex(b)) => a == b,
80            (Self::String(a), Self::String(b)) => a == b,
81            (Self::DateTime(a), Self::DateTime(b)) => a == b,
82            (Self::Array(a), Self::Array(b)) => a.shape == b.shape && a.data == b.data,
83            (
84                Self::Regex {
85                    pattern: a,
86                    flags: af,
87                },
88                Self::Regex {
89                    pattern: b,
90                    flags: bf,
91                },
92            ) => a == b && af == bf,
93            (Self::Null, Self::Null) => true,
94            // Cross-type equality after promotion
95            _ => {
96                if let (Some(a), Some(b)) = (self.as_promoted(), other.as_promoted()) {
97                    match (a, b) {
98                        (ExprValue::Float(a), ExprValue::Float(b)) => a == b,
99                        (ExprValue::Complex(a), ExprValue::Complex(b)) => a == b,
100                        _ => false,
101                    }
102                } else {
103                    false
104                }
105            }
106        }
107    }
108}

110impl fmt::Display for ExprValue {
111    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
112        match self {
113            Self::Bool(b) => write!(f, "{b}"),
114            Self::Int(n) => write!(f, "{n}"),
115            Self::Float(v) => write!(f, "{v}"),
116            Self::Quantity(q) => write!(f, "{q}"),
117            Self::Complex(c) => write!(f, "({} + {}i)", c.re, c.im),
118            Self::String(s) => write!(f, "{s}"),
119            Self::DateTime(mjd) => write!(f, "MJD({mjd})"),
120            Self::Array(arr) => {
121                write!(f, "Array[")?;
122                for (i, v) in arr.data.iter().enumerate() {
123                    if i > 0 {
124                        write!(f, ", ")?;
125                    }
126                    write!(f, "{v}")?;
127                }
128                write!(f, "]")
129            }
130            Self::Regex { pattern, flags } => write!(f, "p/{pattern}/{flags}"),
131            Self::Null => write!(f, "NULL"),
132        }
133    }
134}

136impl ExprValue {
137    /// Returns a type-name string for error messages.
138    pub fn type_name(&self) -> &'static str {
139        match self {
140            Self::Bool(_) => "Bool",
141            Self::Int(_) => "Int",
142            Self::Float(_) => "Float",
143            Self::Quantity(_) => "Quantity",
144            Self::Complex(_) => "Complex",
145            Self::String(_) => "String",
146            Self::DateTime(_) => "DateTime",
147            Self::Array(_) => "Array",
148            Self::Regex { .. } => "Regex",
149            Self::Null => "Null",
150        }
151    }
152
153    /// Returns true if this value is NULL.
154    pub fn is_null(&self) -> bool {
155        matches!(self, Self::Null)
156    }
157
158    /// Converts to bool for WHERE/HAVING conditions.
159    pub fn to_bool(&self) -> Result<bool, TaqlError> {
160        match self {
161            Self::Bool(b) => Ok(*b),
162            Self::Int(n) => Ok(*n != 0),
163            Self::Null => Ok(false),
164            other => Err(TaqlError::TypeError {
165                message: format!(
166                    "cannot convert {type_name} to Bool",
167                    type_name = other.type_name()
168                ),
169            }),
170        }
171    }
172
173    /// Converts to i64 for integer contexts.
174    pub fn to_int(&self) -> Result<i64, TaqlError> {
175        match self {
176            Self::Bool(b) => Ok(if *b { 1 } else { 0 }),
177            Self::Int(n) => Ok(*n),
178            Self::Float(v) => Ok(*v as i64),
179            other => Err(TaqlError::TypeError {
180                message: format!(
181                    "cannot convert {type_name} to Int",
182                    type_name = other.type_name()
183                ),
184            }),
185        }
186    }
187
188    /// Converts to f64 for float contexts.
189    pub fn to_float(&self) -> Result<f64, TaqlError> {
190        match self {
191            Self::Bool(b) => Ok(if *b { 1.0 } else { 0.0 }),
192            Self::Int(n) => Ok(*n as f64),
193            Self::Float(v) => Ok(*v),
194            Self::DateTime(mjd) => Ok(*mjd),
195            other => Err(TaqlError::TypeError {
196                message: format!(
197                    "cannot convert {type_name} to Float",
198                    type_name = other.type_name()
199                ),
200            }),
201        }
202    }
203
204    /// Convert to a Rust String, or error if not a String value.
205    pub fn to_string_val(&self) -> Result<String, TaqlError> {
206        match self {
207            Self::String(s) => Ok(s.clone()),
208            Self::Null => Err(TaqlError::TypeError {
209                message: "cannot convert Null to String".to_string(),
210            }),
211            other => Ok(format!("{other}")),
212        }
213    }
214
215    /// Numeric type rank for promotion: Bool(0) < Int(1) < Float(2) < Complex(3).
216    fn numeric_rank(&self) -> Option<u8> {
217        match self {
218            Self::Bool(_) => Some(0),
219            Self::Int(_) => Some(1),
220            Self::Float(_) => Some(2),
221            Self::Complex(_) => Some(3),
222            _ => None,
223        }
224    }
225
226    /// Promote this value to the given rank.
227    fn promote_to(&self, rank: u8) -> Result<ExprValue, TaqlError> {
228        match rank {
229            0 => match self {
230                Self::Bool(b) => Ok(ExprValue::Bool(*b)),
231                _ => Err(TaqlError::TypeError {
232                    message: format!("cannot demote {} to Bool", self.type_name()),
233                }),
234            },
235            1 => match self {
236                Self::Bool(b) => Ok(ExprValue::Int(if *b { 1 } else { 0 })),
237                Self::Int(n) => Ok(ExprValue::Int(*n)),
238                _ => Err(TaqlError::TypeError {
239                    message: format!("cannot demote {} to Int", self.type_name()),
240                }),
241            },
242            2 => match self {
243                Self::Bool(b) => Ok(ExprValue::Float(if *b { 1.0 } else { 0.0 })),
244                Self::Int(n) => Ok(ExprValue::Float(*n as f64)),
245                Self::Float(v) => Ok(ExprValue::Float(*v)),
246                _ => Err(TaqlError::TypeError {
247                    message: format!("cannot demote {} to Float", self.type_name()),
248                }),
249            },
250            3 => match self {
251                Self::Bool(b) => Ok(ExprValue::Complex(Complex64::new(
252                    if *b { 1.0 } else { 0.0 },
253                    0.0,
254                ))),
255                Self::Int(n) => Ok(ExprValue::Complex(Complex64::new(*n as f64, 0.0))),
256                Self::Float(v) => Ok(ExprValue::Complex(Complex64::new(*v, 0.0))),
257                Self::Complex(c) => Ok(ExprValue::Complex(*c)),
258                _ => Err(TaqlError::TypeError {
259                    message: format!("cannot convert {} to Complex", self.type_name()),
260                }),
261            },
262            _ => Err(TaqlError::TypeError {
263                message: format!("unknown type rank {rank}"),
264            }),
265        }
266    }
267
268    /// Promotes this value to Float if it's numeric, otherwise returns self unchanged.
269    fn as_promoted(&self) -> Option<ExprValue> {
270        match self {
271            Self::Bool(b) => Some(ExprValue::Float(if *b { 1.0 } else { 0.0 })),
272            Self::Int(n) => Some(ExprValue::Float(*n as f64)),
273            Self::Float(v) => Some(ExprValue::Float(*v)),
274            Self::Complex(c) => Some(ExprValue::Complex(*c)),
275            _ => None,
276        }
277    }
278
279    /// Compare two values for ordering (total_cmp for floats, NaN-safe).
280    pub fn compare(&self, other: &ExprValue) -> Result<Ordering, TaqlError> {
281        // NULL propagation: NULL is considered less than everything else
282        match (self.is_null(), other.is_null()) {
283            (true, true) => return Ok(Ordering::Equal),
284            (true, false) => return Ok(Ordering::Less),
285            (false, true) => return Ok(Ordering::Greater),
286            _ => {}
287        }
288
289        // Same type fast path
290        match (self, other) {
291            (ExprValue::Bool(a), ExprValue::Bool(b)) => return Ok(a.cmp(b)),
292            (ExprValue::Int(a), ExprValue::Int(b)) => return Ok(a.cmp(b)),
293            (ExprValue::Float(a), ExprValue::Float(b)) => return Ok(a.total_cmp(b)),
294            (ExprValue::Quantity(a), ExprValue::Quantity(b)) => {
295                if !a.unit().conformant(b.unit()) {
296                    return Err(TaqlError::TypeError {
297                        message: format!(
298                            "cannot compare Quantity {} with Quantity {}",
299                            a.unit().name(),
300                            b.unit().name()
301                        ),
302                    });
303                }
304                return Ok(a.get_si_value().total_cmp(&b.get_si_value()));
305            }
306            (ExprValue::String(a), ExprValue::String(b)) => return Ok(a.cmp(b)),
307            (ExprValue::DateTime(a), ExprValue::DateTime(b)) => return Ok(a.total_cmp(b)),
308            _ => {}
309        }
310
311        // Cross-type numeric promotion
312        let ar = self.numeric_rank();
313        let br = other.numeric_rank();
314        if let (Some(ar), Some(br)) = (ar, br) {
315            let target = ar.max(br);
316            let a = self.promote_to(target)?;
317            let b = other.promote_to(target)?;
318            match (&a, &b) {
319                (ExprValue::Int(x), ExprValue::Int(y)) => return Ok(x.cmp(y)),
320                (ExprValue::Float(x), ExprValue::Float(y)) => return Ok(x.total_cmp(y)),
321                _ => {}
322            }
323        }
324
325        Err(TaqlError::TypeError {
326            message: format!(
327                "cannot compare {} with {}",
328                self.type_name(),
329                other.type_name()
330            ),
331        })
332    }
333}

335/// Promote two values to a common numeric type.
336pub fn promote(a: ExprValue, b: ExprValue) -> Result<(ExprValue, ExprValue), TaqlError> {
337    let ar = a.numeric_rank();
338    let br = b.numeric_rank();
339    match (ar, br) {
340        (Some(ar), Some(br)) => {
341            let target = ar.max(br);
342            Ok((a.promote_to(target)?, b.promote_to(target)?))
343        }
344        _ => Err(TaqlError::TypeError {
345            message: format!(
346                "cannot perform arithmetic on {} and {}",
347                a.type_name(),
348                b.type_name()
349            ),
350        }),
351    }
352}

354/// Convert a ScalarValue to an ExprValue.
355impl From<&ScalarValue> for ExprValue {
356    fn from(sv: &ScalarValue) -> Self {
357        match sv {
358            ScalarValue::Bool(b) => ExprValue::Bool(*b),
359            ScalarValue::UInt8(n) => ExprValue::Int(*n as i64),
360            ScalarValue::UInt16(n) => ExprValue::Int(*n as i64),
361            ScalarValue::UInt32(n) => ExprValue::Int(*n as i64),
362            ScalarValue::Int16(n) => ExprValue::Int(*n as i64),
363            ScalarValue::Int32(n) => ExprValue::Int(*n as i64),
364            ScalarValue::Int64(n) => ExprValue::Int(*n),
365            ScalarValue::Float32(v) => ExprValue::Float(*v as f64),
366            ScalarValue::Float64(v) => ExprValue::Float(*v),
367            ScalarValue::Complex32(c) => {
368                ExprValue::Complex(Complex64::new(c.re as f64, c.im as f64))
369            }
370            ScalarValue::Complex64(c) => ExprValue::Complex(*c),
371            ScalarValue::String(s) => ExprValue::String(s.clone()),
372        }
373    }
374}

388/// Convert a casacore `ArrayValue` (typed ndarray) into an `ExprValue::Array`.
389///
390/// The flat data vector uses **column-major** (Fortran) order to match C++ casacore
391/// conventions, where the first dimension varies fastest. We use `as_slice_memory_order()`
392/// since casacore arrays are stored in Fortran layout.
393fn array_value_to_expr(arr: &casa_types::ArrayValue) -> ExprValue {
394    use casa_types::ArrayValue as AV;
395    let shape: Vec<usize> = arr.shape().to_vec();
396
397    /// Iterate ndarray elements in storage (memory) order.
398    /// For Fortran-layout arrays (as used by casacore), this gives column-major order.
399    fn storage_order<T>(a: &ndarray::ArrayD<T>) -> &[T] {
400        a.as_slice_memory_order()
401            .expect("casacore arrays are always contiguous")
402    }
403
404    let data: Vec<ExprValue> = match arr {
405        AV::Bool(a) => storage_order(a)
406            .iter()
407            .map(|v| ExprValue::Bool(*v))
408            .collect(),
409        AV::UInt8(a) => storage_order(a)
410            .iter()
411            .map(|v| ExprValue::Int(*v as i64))
412            .collect(),
413        AV::UInt16(a) => storage_order(a)
414            .iter()
415            .map(|v| ExprValue::Int(*v as i64))
416            .collect(),
417        AV::UInt32(a) => storage_order(a)
418            .iter()
419            .map(|v| ExprValue::Int(*v as i64))
420            .collect(),
421        AV::Int16(a) => storage_order(a)
422            .iter()
423            .map(|v| ExprValue::Int(*v as i64))
424            .collect(),
425        AV::Int32(a) => storage_order(a)
426            .iter()
427            .map(|v| ExprValue::Int(*v as i64))
428            .collect(),
429        AV::Int64(a) => storage_order(a)
430            .iter()
431            .map(|v| ExprValue::Int(*v))
432            .collect(),
433        AV::Float32(a) => storage_order(a)
434            .iter()
435            .map(|v| ExprValue::Float(*v as f64))
436            .collect(),
437        AV::Float64(a) => storage_order(a)
438            .iter()
439            .map(|v| ExprValue::Float(*v))
440            .collect(),
441        AV::Complex32(a) => storage_order(a)
442            .iter()
443            .map(|v| ExprValue::Complex(Complex64::new(v.re as f64, v.im as f64)))
444            .collect(),
445        AV::Complex64(a) => storage_order(a)
446            .iter()
447            .map(|v| ExprValue::Complex(*v))
448            .collect(),
449        AV::String(a) => storage_order(a)
450            .iter()
451            .map(|v| ExprValue::String(v.clone()))
452            .collect(),
453    };
454    ExprValue::Array(ArrayValue { shape, data })
455}

457/// The evaluation context: provides column values for the current row.
458pub struct EvalContext<'a> {
459    /// The current row being evaluated.
460    pub row: &'a RecordValue,
461    /// The 0-based row index in the parent table.
462    pub row_index: usize,
463    /// Index style (Glish=1-based or Python=0-based).
464    pub style: ast::IndexStyle,
465    /// Explicit measures provider attached to the table being evaluated.
466    pub measures: Option<&'a std::sync::Arc<dyn casa_types::measures::MeasuresProvider>>,
467}

469/// Evaluate an expression against a row context.
470pub fn eval_expr(expr: &Expr, ctx: &EvalContext<'_>) -> Result<ExprValue, TaqlError> {
471    match expr {
472        Expr::Literal(lit) => Ok(eval_literal(lit)),
473        Expr::ColumnRef(col_ref) => eval_column_ref(col_ref, ctx),
474        Expr::Unary { op, operand } => {
475            let val = eval_expr(operand, ctx)?;
476            eval_unary(*op, val)
477        }
478        Expr::Binary { left, op, right } => {
479            let lval = eval_expr(left, ctx)?;
480            let rval = eval_expr(right, ctx)?;
481            eval_binary(*op, lval, rval)
482        }
483        Expr::Between {
484            expr,
485            low,
486            high,
487            negated,
488        } => {
489            let val = eval_expr(expr, ctx)?;
490            let lo = eval_expr(low, ctx)?;
491            let hi = eval_expr(high, ctx)?;
492            // NULL propagation
493            if val.is_null() || lo.is_null() || hi.is_null() {
494                return Ok(ExprValue::Null);
495            }
496            let ge_low = val.compare(&lo)? != Ordering::Less;
497            let le_high = val.compare(&hi)? != Ordering::Greater;
498            let result = ge_low && le_high;
499            Ok(ExprValue::Bool(if *negated { !result } else { result }))
500        }
501        Expr::In {
502            expr,
503            values,
504            negated,
505        } => {
506            let val = eval_expr(expr, ctx)?;
507            if val.is_null() {
508                return Ok(ExprValue::Null);
509            }
510            let mut found = false;
511            for v in values {
512                let item = eval_expr(v, ctx)?;
513                if val == item {
514                    found = true;
515                    break;
516                }
517            }
518            Ok(ExprValue::Bool(if *negated { !found } else { found }))
519        }
520        Expr::Like {
521            expr,
522            pattern,
523            negated,
524            case_insensitive,
525        } => {
526            let val = eval_expr(expr, ctx)?;
527            let pat = eval_expr(pattern, ctx)?;
528            if val.is_null() || pat.is_null() {
529                return Ok(ExprValue::Null);
530            }
531            match (&val, &pat) {
532                (ExprValue::String(s), ExprValue::String(p)) => {
533                    let result = if *case_insensitive {
534                        like_match(&s.to_lowercase(), &p.to_lowercase())
535                    } else {
536                        like_match(s, p)
537                    };
538                    Ok(ExprValue::Bool(if *negated { !result } else { result }))
539                }
540                _ => Err(TaqlError::TypeError {
541                    message: format!(
542                        "LIKE requires String operands, got {} and {}",
543                        val.type_name(),
544                        pat.type_name()
545                    ),
546                }),
547            }
548        }
549        Expr::IsNull { expr, negated } => {
550            let val = eval_expr(expr, ctx)?;
551            let is_null = val.is_null();
552            Ok(ExprValue::Bool(if *negated { !is_null } else { is_null }))
553        }
554        Expr::RegexMatch {
555            expr,
556            pattern,
557            negated,
558        } => {
559            let val = eval_expr(expr, ctx)?;
560            let pat = eval_expr(pattern, ctx)?;
561            if val.is_null() || pat.is_null() {
562                return Ok(ExprValue::Null);
563            }
564            let s = match &val {
565                ExprValue::String(s) => s.as_str(),
566                _ => {
567                    return Err(TaqlError::TypeError {
568                        message: format!("regex match requires String, got {}", val.type_name()),
569                    });
570                }
571            };
572            let (re_pattern, case_insensitive) = match &pat {
573                ExprValue::String(p) => (p.as_str(), false),
574                ExprValue::Regex { pattern, flags } => (pattern.as_str(), flags.contains('i')),
575                _ => {
576                    return Err(TaqlError::TypeError {
577                        message: format!(
578                            "regex pattern must be String or Regex, got {}",
579                            pat.type_name()
580                        ),
581                    });
582                }
583            };
584            let re_str = if case_insensitive {
585                format!("(?i){re_pattern}")
586            } else {
587                re_pattern.to_string()
588            };
589            let re = regex::Regex::new(&re_str).map_err(|e| TaqlError::TypeError {
590                message: format!("invalid regex: {e}"),
591            })?;
592            let matched = re.is_match(s);
593            Ok(ExprValue::Bool(if *negated { !matched } else { matched }))
594        }
595        Expr::InSet {
596            expr,
597            elements,
598            negated,
599        } => {
600            let val = eval_expr(expr, ctx)?;
601            if val.is_null() {
602                return Ok(ExprValue::Null);
603            }
604            let found = in_set_contains(&val, elements, ctx)?;
605            Ok(ExprValue::Bool(if *negated { !found } else { found }))
606        }
607        Expr::ArrayIndex { array, indices } => {
608            let arr_val = eval_expr(array, ctx)?;
609            match arr_val {
610                ExprValue::Array(arr) => eval_array_index(&arr, indices, ctx),
611                other => Err(TaqlError::TypeError {
612                    message: format!("array indexing requires Array, got {}", other.type_name()),
613                }),
614            }
615        }
616        Expr::Star => Ok(ExprValue::Null), // * in expression context is unusual
617        Expr::RowNumber => Ok(ExprValue::Int(ctx.row_index as i64)),
618        Expr::Subquery(_) => {
619            // Subquery evaluation requires a table context; in per-row eval
620            // we return Null. Full subquery support needs the executor.
621            Ok(ExprValue::Null)
622        }
623        Expr::FunctionCall { name, args } => {
624            let evaluated_args: Vec<ExprValue> = args
625                .iter()
626                .map(|a| eval_expr(a, ctx))
627                .collect::<Result<_, _>>()?;
628            super::functions::call_function(name, &evaluated_args, ctx)
629        }
630        Expr::Aggregate { .. } => {
631            // Aggregates are handled at the execution level, not per-row evaluation.
632            Err(TaqlError::TypeError {
633                message: "aggregate functions cannot be evaluated per-row".to_string(),
634            })
635        }
636    }
637}

639/// Test `val` (non-NULL) for membership in the elements of an `IN [...]`
640/// set, evaluating elements in order and stopping at the first match.
641///
642/// A range element matches when `val` lies within its inclusive bounds and,
643/// if a step is given, sits on the stride counted from the start (or 0).
644fn in_set_contains(
645    val: &ExprValue,
646    elements: &[ast::InSetElement],
647    ctx: &EvalContext<'_>,
648) -> Result<bool, TaqlError> {
649    for elem in elements {
650        match elem {
651            ast::InSetElement::Value(v) => {
652                let item = eval_expr(v, ctx)?;
653                if *val == item {
654                    return Ok(true);
655                }
656            }
657            ast::InSetElement::Range { start, end, step } => {
658                let lo = start.as_ref().map(|s| eval_expr(s, ctx)).transpose()?;
659                let hi = end.as_ref().map(|e| eval_expr(e, ctx)).transpose()?;
660                // Check range membership
661                let ge_lo = match &lo {
662                    Some(l) => val.compare(l)? != Ordering::Less,
663                    None => true,
664                };
665                let le_hi = match &hi {
666                    Some(h) => val.compare(h)? != Ordering::Greater,
667                    None => true,
668                };
669                if ge_lo && le_hi {
670                    // If step is given, check stride
671                    if let Some(step_expr) = step {
672                        let s_val = eval_expr(step_expr, ctx)?;
673                        let step_f = s_val.to_float()?;
674                        let v_f = val.to_float()?;
675                        let lo_f = lo
676                            .as_ref()
677                            .map(|l| l.to_float())
678                            .transpose()?
679                            .unwrap_or(0.0);
680                        if step_f != 0.0 {
681                            let offset = v_f - lo_f;
682                            let remainder = offset % step_f;
683                            if remainder.abs() < 1e-10 {
684                                return Ok(true);
685                            }
686                        }
687                    } else {
688                        return Ok(true);
689                    }
690                }
691            }
692        }
693    }
694    Ok(false)
695}

726fn eval_unary(op: UnaryOp, val: ExprValue) -> Result<ExprValue, TaqlError> {
727    if val.is_null() {
728        return Ok(ExprValue::Null);
729    }
730    match op {
731        UnaryOp::Negate => match val {
732            ExprValue::Int(n) => Ok(ExprValue::Int(-n)),
733            ExprValue::Float(v) => Ok(ExprValue::Float(-v)),
734            ExprValue::Quantity(q) => Ok(ExprValue::Quantity(Quantity::with_unit(
735                -q.value(),
736                q.unit().clone(),
737            ))),
738            ExprValue::Complex(c) => Ok(ExprValue::Complex(-c)),
739            other => Err(TaqlError::TypeError {
740                message: format!("cannot negate {}", other.type_name()),
741            }),
742        },
743        UnaryOp::Not => {
744            let b = val.to_bool()?;
745            Ok(ExprValue::Bool(!b))
746        }
747        UnaryOp::BitNot => match val {
748            ExprValue::Int(n) => Ok(ExprValue::Int(!n)),
749            ExprValue::Bool(b) => Ok(ExprValue::Bool(!b)),
750            other => Err(TaqlError::TypeError {
751                message: format!("cannot apply bitwise NOT to {}", other.type_name()),
752            }),
753        },
754    }
755}

757fn eval_binary(op: BinaryOp, lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
758    // NULL propagation for all operators
759    if lhs.is_null() || rhs.is_null() {
760        return Ok(ExprValue::Null);
761    }
762
763    match op {
764        BinaryOp::And => {
765            let a = lhs.to_bool()?;
766            let b = rhs.to_bool()?;
767            Ok(ExprValue::Bool(a && b))
768        }
769        BinaryOp::Or => {
770            let a = lhs.to_bool()?;
771            let b = rhs.to_bool()?;
772            Ok(ExprValue::Bool(a || b))
773        }
774        BinaryOp::Eq => Ok(ExprValue::Bool(lhs == rhs)),
775        BinaryOp::Ne => Ok(ExprValue::Bool(lhs != rhs)),
776        BinaryOp::Lt | BinaryOp::Le | BinaryOp::Gt | BinaryOp::Ge => {
777            eval_predicate_order(op, lhs, rhs)
778        }
779        BinaryOp::Add => eval_arithmetic_add(lhs, rhs),
780        BinaryOp::Sub => eval_arithmetic_sub(lhs, rhs),
781        BinaryOp::Mul => eval_arithmetic_mul(lhs, rhs),
782        BinaryOp::Div => eval_arithmetic_div(lhs, rhs),
783        BinaryOp::IntDiv => eval_int_div(lhs, rhs),
784        BinaryOp::Modulo => eval_modulo(lhs, rhs),
785        BinaryOp::Power => eval_power(lhs, rhs),
786        BinaryOp::BitAnd => {
787            let a = lhs.to_int()?;
788            let b = rhs.to_int()?;
789            Ok(ExprValue::Int(a & b))
790        }
791        BinaryOp::BitOr => {
792            let a = lhs.to_int()?;
793            let b = rhs.to_int()?;
794            Ok(ExprValue::Int(a | b))
795        }
796        BinaryOp::BitXor => {
797            let a = lhs.to_int()?;
798            let b = rhs.to_int()?;
799            Ok(ExprValue::Int(a ^ b))
800        }
801    }
802}

804// Ordering a query predicate follows IEEE comparisons. Sorting still uses
805// ExprValue::compare's total order; a NaN is not less/greater/equal in a predicate.
806fn eval_predicate_order(
807    op: BinaryOp,
808    lhs: ExprValue,
809    rhs: ExprValue,
810) -> Result<ExprValue, TaqlError> {
811    let order = match (&lhs, &rhs) {
812        (ExprValue::Float(a), ExprValue::Float(b)) => a.partial_cmp(b),
813        (ExprValue::Int(a), ExprValue::Float(b)) => (*a as f64).partial_cmp(b),
814        (ExprValue::Float(a), ExprValue::Int(b)) => a.partial_cmp(&(*b as f64)),
815        _ => Some(lhs.compare(&rhs)?),
816    };
817    Ok(ExprValue::Bool(order.is_some_and(|order| match op {
818        BinaryOp::Lt => order == Ordering::Less,
819        BinaryOp::Le => order != Ordering::Greater,
820        BinaryOp::Gt => order == Ordering::Greater,
821        BinaryOp::Ge => order != Ordering::Less,
822        _ => unreachable!("ordered comparison"),
823    })))
824}

826fn arithmetic_operands(
827    lhs: ExprValue,
828    rhs: ExprValue,
829) -> Result<(ExprValue, ExprValue), TaqlError> {
830    // CASA permits explicit numeric casts of Bool, not implicit arithmetic
831    // operands. In particular Bool+Bool must return an error, never unwind.
832    if matches!(lhs, ExprValue::Bool(_)) || matches!(rhs, ExprValue::Bool(_)) {
833        return Err(TaqlError::TypeError {
834            message: "boolean arithmetic operands".to_string(),
835        });
836    }
837    promote(lhs, rhs)
838}

840fn eval_arithmetic_add(lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
841    // String concatenation
842    if let (ExprValue::String(a), ExprValue::String(b)) = (&lhs, &rhs) {
843        return Ok(ExprValue::String(format!("{a}{b}")));
844    }
845    let (a, b) = arithmetic_operands(lhs, rhs)?;
846    match (a, b) {
847        (ExprValue::Int(a), ExprValue::Int(b)) => Ok(ExprValue::Int(a.wrapping_add(b))),
848        (ExprValue::Float(a), ExprValue::Float(b)) => Ok(ExprValue::Float(a + b)),
849        (ExprValue::Complex(a), ExprValue::Complex(b)) => Ok(ExprValue::Complex(a + b)),
850        _ => unreachable!(),
851    }
852}

854fn eval_arithmetic_sub(lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
855    let (a, b) = arithmetic_operands(lhs, rhs)?;
856    match (a, b) {
857        (ExprValue::Int(a), ExprValue::Int(b)) => Ok(ExprValue::Int(a.wrapping_sub(b))),
858        (ExprValue::Float(a), ExprValue::Float(b)) => Ok(ExprValue::Float(a - b)),
859        (ExprValue::Complex(a), ExprValue::Complex(b)) => Ok(ExprValue::Complex(a - b)),
860        _ => unreachable!(),
861    }
862}

866    match (a, b) {
867        (ExprValue::Int(a), ExprValue::Int(b)) => Ok(ExprValue::Int(a.wrapping_mul(b))),
868        (ExprValue::Float(a), ExprValue::Float(b)) => Ok(ExprValue::Float(a * b)),
869        (ExprValue::Complex(a), ExprValue::Complex(b)) => Ok(ExprValue::Complex(a * b)),
870        _ => unreachable!(),

874fn eval_arithmetic_div(lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
875    let (a, b) = arithmetic_operands(lhs, rhs)?;
876    match (a, b) {
877        // TableExprNodeDivideDouble promotes integer operands to Double,
878        // including IEEE infinity/NaN at a zero divisor.
879        (ExprValue::Int(a), ExprValue::Int(b)) => Ok(ExprValue::Float(a as f64 / b as f64)),
880        (ExprValue::Float(a), ExprValue::Float(b)) => Ok(ExprValue::Float(a / b)),
881        (ExprValue::Complex(a), ExprValue::Complex(b)) => Ok(ExprValue::Complex(a / b)),
882        _ => unreachable!(),
883    }
884}

886fn eval_int_div(lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
887    let (a, b) = arithmetic_operands(lhs, rhs)?;
888    // TaQL spells this floor(a/b), retaining the real division's type.
889    Ok(ExprValue::Float((a.to_float()? / b.to_float()?).floor()))
890}

892fn eval_modulo(lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
893    let (a, b) = arithmetic_operands(lhs, rhs)?;
894    match (a, b) {
895        (ExprValue::Int(a), ExprValue::Int(b)) => {
896            if b == 0 {
897                return Err(TaqlError::DivisionByZero);
898            }
899            let mut remainder = a.checked_rem(b).unwrap_or(0);
900            if remainder != 0 && (a < 0) != (b < 0) {
901                remainder += b;
902            }
903            Ok(ExprValue::Int(remainder))
904        }
905        (ExprValue::Float(a), ExprValue::Float(b)) => {
906            let mut remainder = a % b;
907            if remainder != 0.0 && (a < 0.0) != (b < 0.0) {
908                remainder += b;
909            }
910            Ok(ExprValue::Float(remainder))
911        }
912        _ => Err(TaqlError::TypeError {
913            message: "modulo not supported for complex numbers".to_string(),
914        }),
915    }
916}

918fn eval_power(lhs: ExprValue, rhs: ExprValue) -> Result<ExprValue, TaqlError> {
919    let (a, b) = arithmetic_operands(lhs, rhs)?;
920    match (a, b) {
921        (ExprValue::Int(a), ExprValue::Int(b)) => Ok(ExprValue::Float((a as f64).powf(b as f64))),
922        (ExprValue::Float(a), ExprValue::Float(b)) => Ok(ExprValue::Float(a.powf(b))),
923        (ExprValue::Complex(a), ExprValue::Complex(b)) => {
924            // Complex power: a^b = exp(b * ln(a))
925            let result = if b.im == 0.0 {
926                a.powf(b.re)
927            } else {
928                (b * a.ln()).exp()
929            };
930            Ok(ExprValue::Complex(result))
931        }
932        _ => unreachable!(),
933    }
934}

936/// SQL LIKE pattern matching with `%` (any chars) and `_` (single char).
937///
938/// This is a hand-written matcher — no regex crate needed.
939pub fn like_match(text: &str, pattern: &str) -> bool {
940    like_match_recursive(text.as_bytes(), pattern.as_bytes())
941}
942
943fn like_match_recursive(text: &[u8], pattern: &[u8]) -> bool {
944    let mut ti = 0;
945    let mut pi = 0;
946    let mut star_pi = usize::MAX;
947    let mut star_ti = 0;
948
949    while ti < text.len() {
950        if pi < pattern.len() && pattern[pi] == b'_' {
951            // _ matches any single character
952            ti += 1;
953            pi += 1;
954        } else if pi < pattern.len() && pattern[pi] == b'%' {
955            // % matches any sequence — remember this position
956            star_pi = pi;
957            star_ti = ti;
958            pi += 1;
959        } else if pi < pattern.len() && pattern[pi] == text[ti] {
960            ti += 1;
961            pi += 1;
962        } else if star_pi != usize::MAX {
963            // Backtrack: advance the text position after the last %
964            pi = star_pi + 1;
965            star_ti += 1;
966            ti = star_ti;
967        } else {
968            return false;
969        }
970    }
971
972    // Skip trailing % in pattern
973    while pi < pattern.len() && pattern[pi] == b'%' {
974        pi += 1;
975    }
976
977    pi == pattern.len()
978}

980/// Convert a user-supplied index to a 0-based index, respecting [`IndexStyle`].
981///
982/// - **Glish** (1-based): subtract 1.
983/// - **Python** (0-based): use as-is. Negative indices count from end.
984fn to_zero_based(idx: i64, len: usize, style: ast::IndexStyle) -> Result<usize, TaqlError> {
985    let len_i = len as i64;
986    let actual = match style {
987        ast::IndexStyle::Glish => {
988            // 1-based: 1 → 0, negative counts from end
989            if idx < 0 { len_i + idx } else { idx - 1 }
990        }
991        ast::IndexStyle::Python => {
992            if idx < 0 {
993                len_i + idx
994            } else {
995                idx
996            }
997        }
998    };
999    if actual < 0 || actual >= len_i {
1000        return Err(TaqlError::TypeError {
1001            message: format!("array index {idx} out of bounds (length {len})"),
1002        });
1003    }
1004    Ok(actual as usize)
1005}

1007/// Resolve a slice (`start:end[:step]`) to a Vec of 0-based indices.
1008fn resolve_slice(
1009    start: Option<i64>,
1010    end: Option<i64>,
1011    step: Option<i64>,
1012    dim_len: usize,
1013    style: ast::IndexStyle,
1014) -> Result<Vec<usize>, TaqlError> {
1015    let step = step.unwrap_or(1);
1016    if step == 0 {
1017        return Err(TaqlError::TypeError {
1018            message: "slice step cannot be zero".to_string(),
1019        });
1020    }
1021    let len_i = dim_len as i64;
1022    match style {
1023        ast::IndexStyle::Glish => {
1024            // 1-based inclusive: start defaults to 1, end defaults to len
1025            let s = start.unwrap_or(1);
1026            let e = end.unwrap_or(len_i);
1027            let s0 = if s < 0 { len_i + s } else { s - 1 };
1028            let e0 = if e < 0 { len_i + e } else { e - 1 };
1029            let mut result = Vec::new();
1030            if step > 0 {
1031                let mut i = s0;
1032                while i <= e0 && i < len_i {
1033                    if i >= 0 {
1034                        result.push(i as usize);
1035                    }
1036                    i += step;
1037                }
1038            } else {
1039                let mut i = s0;
1040                while i >= e0 && i >= 0 {
1041                    if i < len_i {
1042                        result.push(i as usize);
1043                    }
1044                    i += step;
1045                }
1046            }
1047            Ok(result)
1048        }
1049        ast::IndexStyle::Python => {
1050            // 0-based exclusive end: start defaults to 0, end defaults to len
1051            let s = start.unwrap_or(if step > 0 { 0 } else { len_i - 1 });
1052            let e = end.unwrap_or(if step > 0 { len_i } else { -len_i - 1 });
1053            let s0 = if s < 0 {
1054                (len_i + s).max(0)
1055            } else {
1056                s.min(len_i)
1057            };
1058            let e0 = if e < 0 {
1059                (len_i + e).max(-1)
1060            } else {
1061                e.min(len_i)
1062            };
1063            let mut result = Vec::new();
1064            if step > 0 {
1065                let mut i = s0;
1066                while i < e0 {
1067                    result.push(i as usize);
1068                    i += step;
1069                }
1070            } else {
1071                let mut i = s0;
1072                while i > e0 {
1073                    result.push(i as usize);
1074                    i += step;
1075                }
1076            }
1077            Ok(result)
1078        }
1079    }
1080}

1082/// Evaluate array indexing/slicing with style-aware offset.
1083///
1084/// Supports N-dimensional arrays with single-element access and slicing.
1085///
1086/// C++ reference: `TableExprNodeArrayPart`.
1087fn eval_array_index(
1088    arr: &ArrayValue,
1089    indices: &[ast::IndexElement],
1090    ctx: &EvalContext<'_>,
1091) -> Result<ExprValue, TaqlError> {
1092    let ndim = arr.shape.len();
1093
1094    // If fewer indices than dimensions, remaining dims are taken in full.
1095    // If more indices than dimensions for a 1-D array, allow as flat access.
1096    if indices.len() > ndim && ndim != 1 {
1097        return Err(TaqlError::TypeError {
1098            message: format!(
1099                "too many indices ({}) for {}-dimensional array",
1100                indices.len(),
1101                ndim
1102            ),
1103        });
1104    }
1105
1106    // Resolve each index dimension to a list of positions.
1107    let mut dim_indices: Vec<Vec<usize>> = Vec::with_capacity(ndim);
1108    for dim in 0..ndim {
1109        let dim_len = arr.shape[dim];
1110        if dim < indices.len() {
1111            match &indices[dim] {
1112                ast::IndexElement::Single(expr) => {
1113                    let val = eval_expr(expr, ctx)?;
1114                    let idx = val.to_int()?;
1115                    let pos = to_zero_based(idx, dim_len, ctx.style)?;
1116                    dim_indices.push(vec![pos]);
1117                }
1118                ast::IndexElement::Slice { start, end, step } => {
1119                    let s = start
1120                        .as_ref()
1121                        .map(|e| eval_expr(e, ctx)?.to_int())
1122                        .transpose()?;
1123                    let e = end
1124                        .as_ref()
1125                        .map(|e| eval_expr(e, ctx)?.to_int())
1126                        .transpose()?;
1127                    let st = step
1128                        .as_ref()
1129                        .map(|e| eval_expr(e, ctx)?.to_int())
1130                        .transpose()?;
1131                    dim_indices.push(resolve_slice(s, e, st, dim_len, ctx.style)?);
1132                }
1133            }
1134        } else {
1135            // Remaining dims: take all elements.
1136            dim_indices.push((0..dim_len).collect());
1137        }
1138    }
1139
1140    // Compute output shape: dims with >1 element remain; dims with 1 element are squeezed.
1141    // But only squeeze if the user gave a Single index for that dim.
1142    let mut result_shape = Vec::new();
1143    for (dim, idxs) in dim_indices.iter().enumerate() {
1144        let is_single = dim < indices.len() && matches!(indices[dim], ast::IndexElement::Single(_));
1145        if !is_single {
1146            result_shape.push(idxs.len());
1147        }
1148    }
1149
1150    // Collect elements using column-major (Fortran) traversal of the index
1151    // combinations, matching C++ casacore array storage order.
1152    let mut result_data = Vec::new();
1153    let mut combo = vec![0usize; ndim];
1154    loop {
1155        // Compute flat index from combo using column-major strides
1156        // (first dimension varies fastest, matching C++ casacore).
1157        let mut flat = 0;
1158        let mut stride = 1;
1159        for dim in 0..ndim {
1160            flat += dim_indices[dim][combo[dim]] * stride;
1161            stride *= arr.shape[dim];
1162        }
1163        if flat < arr.data.len() {
1164            result_data.push(arr.data[flat].clone());
1165        }
1166
1167        // Advance combo (leftmost first = column-major traversal)
1168        let mut carry = true;
1169        for dim in 0..ndim {
1170            if carry {
1171                combo[dim] += 1;
1172                if combo[dim] < dim_indices[dim].len() {
1173                    carry = false;
1174                } else {
1175                    combo[dim] = 0;
1176                }
1177            }
1178        }
1179        if carry {
1180            break;
1181        }
1182    }
1183
1184    // If the result is a scalar (all dims squeezed), return the single element.
1185    if result_shape.is_empty() || result_shape.iter().all(|&d| d == 1) {
1186        if let Some(val) = result_data.into_iter().next() {
1187            return Ok(val);
1188        }
1189        return Ok(ExprValue::Null);
1190    }
1191
1192    Ok(ExprValue::Array(ArrayValue {
1193        shape: result_shape,
1194        data: result_data,
1195    }))
1196}
