2//! Scalar amplitude/phase curves shared by time and coordinate interpolation.
3use casa_types::Complex32;
4
5#[derive(Debug, Clone, Copy, PartialEq, Eq)]
6pub(crate) enum Curve {
7    Nearest,
8    Linear,
9    Cubic,
10    Spline,
11}
12
13pub(crate) fn unwrap_phase(previous: f32, mut phase: f32) -> f32 {
14    while f64::from(phase - previous) > std::f64::consts::PI {
15        phase = (f64::from(phase) - std::f64::consts::TAU) as f32;
16    }
17    while f64::from(phase - previous) < -std::f64::consts::PI {
18        phase = (f64::from(phase) + std::f64::consts::TAU) as f32;
19    }
20    phase
21}

23pub(crate) fn sample(
24    points: &[(f64, f32, f32)],
25    coordinate: f64,
26    mode: Curve,
27    descending: bool,
28    derivatives: &[[f64; 2]],
29) -> Option<Complex32> {
30    sample_polar(points, coordinate, mode, descending, derivatives)
31        .map(|(amplitude, phase)| Complex32::from_polar(amplitude, phase))
32}

34pub(crate) fn sample_polar(
35    points: &[(f64, f32, f32)],
36    coordinate: f64,
37    mode: Curve,
38    descending: bool,
39    derivatives: &[[f64; 2]],
40) -> Option<(f32, f32)> {
41    let first = points.first()?;
42    if points.len() == 1 || coordinate <= first.0 {
43        return Some((first.1, first.2));
44    }
45    let last = points.last().expect("nonempty");
46    if coordinate >= last.0 {
47        return Some((last.1, last.2));
48    }
49    let upper = points.partition_point(|point| point.0 < coordinate);
50    let (low, high) = (points[upper - 1], points[upper]);
51    let (amplitude, phase) = if mode == Curve::Nearest {
52        let nearest = if high.0 - coordinate < coordinate - low.0
53            || (descending && high.0 - coordinate == coordinate - low.0)
54        {
55            high
56        } else {
57            low
58        };
59        (nearest.1, nearest.2)
60    } else if mode == Curve::Cubic {
61        let first = upper.saturating_sub(2).min(points.len().saturating_sub(4));
62        let stencil = &points[first..(first + 4).min(points.len())];
63        let polynomial = |axis: usize| {
64            let mut c = [0.0f32; 4];
65            let mut d = [0.0f32; 4];
66            for (index, point) in stencil.iter().enumerate() {
67                c[index] = if axis == 0 { point.1 } else { point.2 };
68                d[index] = c[index];
69            }
70            let mut value = c[0];
71            for order in 1..stencil.len() {
72                for index in 0..stencil.len() - order {
73                    let w = f64::from(c[index + 1] - d[index])
74                        / (stencil[index].0 - stencil[index + order].0);
75                    c[index] = ((stencil[index].0 - coordinate) * w) as f32;
76                    d[index] = ((stencil[index + order].0 - coordinate) * w) as f32;
77                }
78                value += c[0];
79            }
80            value
81        };
82        (polynomial(0), polynomial(1))
83    } else if mode == Curve::Spline {
84        let width = high.0 - low.0;
85        let a = (high.0 - coordinate) / width;
86        let b = 1.0 - a;
87        let evaluate = |left: f32, right: f32, axis: usize| {
88            (a * f64::from(left)
89                + b * f64::from(right)
90                + width * width / 6.0
91                    * ((a * a * a - a) * derivatives[upper - 1][axis]
92                        + (b * b * b - b) * derivatives[upper][axis])) as f32
93        };
94        (evaluate(low.1, high.1, 0), evaluate(low.2, high.2, 1))
95    } else {
96        let fraction = (coordinate - low.0) / (high.0 - low.0);
97        let interpolate = |a: f32, b: f32| (f64::from(a) + f64::from(b - a) * fraction) as f32;
98        (interpolate(low.1, high.1), interpolate(low.2, high.2))
99    };
100    Some((amplitude, phase))
101}

103/// Tridiagonal solution for natural (zero endpoint curvature) cubic splines.
104pub(crate) fn spline_derivatives(points: &[(f64, f32, f32)]) -> Vec<[f64; 2]> {
105    let mut second = vec![[0.0; 2]; points.len()];
106    if points.len() < 3 {
107        return second;
108    }
109    let mut upper = vec![0.0; points.len()];
110    for index in 1..points.len() - 1 {
111        let left = points[index].0 - points[index - 1].0;
112        let right = points[index + 1].0 - points[index].0;
113        let diagonal = 2.0 * (left + right) - left * upper[index - 1];
114        upper[index] = right / diagonal;
115        let previous = second[index - 1];
116        for (axis, value) in second[index].iter_mut().enumerate() {
117            let y = |index: usize| {
118                f64::from(if axis == 0 {
119                    points[index].1
120                } else {
121                    points[index].2
122                })
123            };
124            let slope = (y(index + 1) - y(index)) / right - (y(index) - y(index - 1)) / left;
125            *value = (6.0 * slope - left * previous[axis]) / diagonal;
126        }
127    }
128    for index in (1..points.len() - 1).rev() {
129        let next = second[index + 1];
130        for (value, next) in second[index].iter_mut().zip(next) {
131            *value -= upper[index] * next;
132        }
133    }
134    second
135}
