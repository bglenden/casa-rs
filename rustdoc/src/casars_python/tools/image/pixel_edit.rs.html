16fn set_value(pixels: &Variant, formatted_pixels: &str) -> PyResult<Option<f32>> {
17    if pixels.is_empty_sequence() {
18        return Ok(None);
19    }
20    match pixels {
21        Variant::Number(value) if value.is_finite() => {
22            return evaluate_pixel_expression(formatted_pixels)
23                .map(Some)
24                .map_err(raised);
25        }
26        Variant::Text(text) if text.is_empty() || text == "[]" => return Ok(None),
27        Variant::Texts(_)
28        | Variant::Numbers(_)
29        | Variant::Complexes(_)
30        | Variant::Array { .. }
31        | Variant::ComplexArray { .. } => {
32            return Err(raised("The pixels expression must be scalar"));
33        }
34        _ => (),
35    }
36    evaluate_pixel_expression(&pixels.describe())
37        .map(Some)
38        .map_err(raised)
39}

41#[pymethods]
42impl PyImageTool {
43    #[pyo3(signature = (pixels=None, pixelmask=-1, region=None, list=false), text_signature = "($self, pixels=[], pixelmask=-1, region={}, list=False)")]
44    fn set(
45        &self,
46        py: Python<'_>,
47        pixels: Option<&Bound<'_, PyAny>>,
48        pixelmask: i64,
49        region: Option<&Bound<'_, PyAny>>,
50        list: bool,
51    ) -> PyResult<bool> {
52        if self.state().image.is_none() {
53            return Ok(false);
54        }
55        let variant = pixels
56            .map(|value| value.extract::<Variant>())
57            .transpose()?
58            .unwrap_or(Variant::Empty);
59        let pixel_history = match &variant {
60            Variant::Number(value) if pixels.is_some_and(|v| v.extract::<i64>().is_err()) => {
61                format!("{value:.6}")
62            }
63            Variant::Text(text) => format!("\"{text}\""),
64            other => other.describe(),
65        };
66        let region = Region::argument(py, region, "set")?;
67        let mut logs = Vec::new();
68        let outcome = py.detach(|| {
69            let state = self.state();
70            let Some(attachment) = &state.image else { return Ok(false); };
71            let mut handle = attachment.lock();
72            handle.check_single_precision("set")?;
73            let set_mask = pixelmask != -1;
74            let has_pixels = !variant.is_empty_sequence() && !matches!(&variant, Variant::Text(text) if text.is_empty() || text == "[]");
75            let ImageHandle::Float(storage) = &*handle else {
76                return Err(raised("set only supports float-valued images"));
77            };
78            let text_selection = if let Some(region @ Region::Text(_)) = &region {
79                if handle.fits().is_none() {
80                    Some(with_storage!(storage, image =>
81                        ImagePixelSelection::new(image.as_ref(), Some(region.as_region()))
82                    ).map_err(raised)?)
83                } else {
84                    None
85                }
86            } else {
87                None
88            };
89            if !has_pixels && !set_mask {
90                logs.push(LogLine::warning("You must specify at least either the pixels or the mask to set"));
91                return Ok(false);
92            }
93            if let Some(kind) = handle.read_only_file_type() {
94                if kind == "MIRIADImage" {
95                    return Err(raised(if has_pixels {
96                        "Failed AlwaysAssert to.isWritable()"
97                    } else {
98                        "SubLattice::pixelMask - no pixelmask available"
99                    }));
100                }
101                return Err(raised(if set_mask {
102                    format!("Cannot make requested mask for this image type which is {kind}")
103                } else {
104                    format!("{kind} is not writable")
105                }));
106            }
107            attachment.require_lock(&mut handle, casa_tables::LockType::Write)?;
108            let ImageHandle::Float(storage) = &*handle else { unreachable!("float attachment checked") };
109            let default = with_storage!(storage, image => image.active_pixel_mask_name());
110            let mask_name = if set_mask {
111                if let Some(name) = default {
112                    Some(name)
113                } else {
114                    let name = (0..).map(|index| format!("mask{index}")).find(|name| !has_region(&handle, name)).expect("a fresh mask name");
115                    let mask = ArrayD::from_elem(IxDyn(handle.shape()), true);
116                    let ImageHandle::Float(storage) = &mut *handle else { unreachable!() };
117                    with_storage!(storage, image => image.put_mask(&name, &mask)).map_err(raised)?;
118                    with_storage!(storage, image => image.set_default_mask(&name)).map_err(raised)?;
119                    handle.save().map_err(raised)?;
120                    Some(name)
121                }
122            } else { None };
123            let mask = mask_name.as_deref().map(|name| (name, pixelmask > 0));
124            let ImageHandle::Float(storage) = &mut *handle else { unreachable!() };
125            let (start, shape) = with_storage!(storage, image => {
126                let selection = match text_selection {
127                    Some(selection) => selection,
128                    None => ImagePixelSelection::new(image.as_ref(), region.as_ref().map(Region::as_region)).map_err(raised)?,
129                };
130                let value = set_value(&variant, &pixel_history)?;
131                selection.apply(image.as_mut(), value, mask).map_err(raised)?
132            });
133            if list {
134                logs.push(LogLine::info(format!("Selected region at {start:?} with shape {shape:?}")));
135            }
136            if state.history {
137                let region = region.as_ref().map_or_else(|| "{}".into(), Region::history);
138                handle.record_history("set", format!("ia.set(pixels={pixel_history}, pixelmask={pixelmask}, region={region}, list={list})"))?;
139            }
140            handle.save().map_err(raised)?;
141            Ok(true)
142        });
143        emit(py, "casars.tools.image", &logs)?;
144        outcome
145    }
146}
