17pub(super) fn uses_channel_ordinate_masks(
18    x_axis: MsAxis,
19    mut ordinates: impl Iterator<Item = MsAxis> + Clone,
20) -> bool {
21    x_axis == MsAxis::Channel
22        && ordinates.clone().any(MsAxis::is_visibility_math)
23        && ordinates.clone().any(|axis| !axis.is_visibility_math())
24        && ordinates.all(|axis| {
25            axis.is_visibility_math()
26                || matches!(axis, MsAxis::U | MsAxis::V | MsAxis::W | MsAxis::Elevation)
27        })
28}

76pub(super) fn uses_direct_vector_time_projection(
77    spec: &MsPlotSpec,
78    x_column: Option<MsDataColumn>,
79) -> bool {
80    let model_only = spec.secondary_y_axis().is_none()
81        && spec.data_column == MsDataColumn::Model
82        && x_column.is_none_or(|column| column == MsDataColumn::Model)
83        && supports_ordinary_time_axes(spec);
84    let shared_or_channel = supports_reduced_visibility_axes(spec)
85        && (spec.secondary_y_axis().is_some()
86            || spec.averaging.avgchannel.is_some_and(|width| width > 1));
87    uses_ordinary_time_averaging(spec)
88        && !spec.averaging.scalar
89        && (model_only || shared_or_channel)
90}

108pub(super) fn uses_scalar_visibility_average(spec: &MsPlotSpec) -> bool {
109    let model_x_with_metadata_y = uses_ordinary_time_averaging(spec)
110        && spec.secondary_y_axis().is_none()

112        && spec.x_axis.is_visibility_math()

117        && !spec.averaging.avgantenna
118        && !spec.averaging.avgspw

154pub(super) fn check_averaged_derived_storage(
155    ms: &MeasurementSet,
156    columns: [MsDataColumn; 3],
157    averaging_axis: &str,
158    implicit_model: bool,
159) -> Result<(), TaskFailure> {
160    let schema = ms.main_table().schema();
161    // The visibility reader supplies either a selected virtual model or the
162    // default point source in supported plots. FLOAT_DATA requires a stored model.
163    let stored_model =
164        !implicit_model || schema.is_some_and(|schema| schema.contains_column("FLOAT_DATA"));
165    let operands = columns
166        .into_iter()
167        .filter(|column| column.is_derived())
168        .map(|column| {
169            if column.is_corrected_model_expression() {
170                "CORRECTED_DATA"
171            } else {
172                "DATA"
173            }
174        });
175    for name in stored_model
176        .then_some("MODEL_DATA")
177        .into_iter()
178        .chain(operands)
179    {
180        if !schema.is_some_and(|schema| schema.contains_column(name)) {
181            return Err(TaskFailure::validation(format!(
182                "{averaging_axis}-averaged residuals and ratios require stored {name} (#1032)"
183            )));
184        }
185    }
186    Ok(())
187}

202pub(super) fn check_averaging(
203    averaging: bool,
204    extended_averaging_with_independent_ordinates: bool,
205    supports_derived: bool,
206    columns: [MsDataColumn; 3],
207) -> Result<(), TaskFailure> {
208    if extended_averaging_with_independent_ordinates {
209        return Err(TaskFailure::validation(
210            "time, baseline, antenna and SPW averaging of independent MeasurementSet ordinate columns is pending #1032",
211        ));
212    }
213    if averaging && !supports_derived && columns.into_iter().any(MsDataColumn::is_derived) {
214        return Err(TaskFailure::validation(
215            "averaging derived data columns in this MeasurementSet plot is pending #1032",
216        ));
217    }
218    Ok(())
219}

232impl PlotMsWeightKinds {
233    pub(super) fn uses(self, kind: WeightKind) -> bool {
234        self.primary == kind || self.model == Some(kind) || self.axes.contains(&kind)
235    }
236
237    pub(super) fn for_column(self, column: MsDataColumn) -> WeightKind {
238        column_weight_kind(column, self.has_observed, self.has_corrected)
239    }
240
241    pub(super) fn uses_for_ordinates(
242        self,
243        kind: WeightKind,
244        additional: &[(MsAxis, MsDataColumn)],
245    ) -> bool {
246        self.uses(kind)
247            || additional
248                .iter()
249                .any(|&(axis, column)| axis.is_visibility_math() && self.for_column(column) == kind)
250    }

253fn column_weight_kind(column: MsDataColumn, has_observed: bool, has_corrected: bool) -> WeightKind {
254    let column = match column {
255        MsDataColumn::Data
256        | MsDataColumn::DataMinusModel
257        | MsDataColumn::DataMinusModelScalar
258        | MsDataColumn::DataDivModel
259        | MsDataColumn::DataDivModelScalar => OptionalMainColumn::Data,
260        MsDataColumn::Corrected
261        | MsDataColumn::CorrectedMinusModel
262        | MsDataColumn::CorrectedMinusModelScalar
263        | MsDataColumn::CorrectedDivModel
264        | MsDataColumn::CorrectedDivModelScalar => OptionalMainColumn::CorrectedData,
265        MsDataColumn::Model => OptionalMainColumn::ModelData,
266        _ => return WeightKind::Unit,
267    };
268    visibility_weight_kind(column, has_observed, has_corrected, false)
269}

271pub(super) fn weight_kinds(
272    ms: &MeasurementSet,
273    spec: &MsPlotSpec,
274    input: &PlotMsInput<'_>,
275) -> PlotMsWeightKinds {
276    let primary = spec.data_column;
277    let columns = [
278        input.x_data_column.unwrap_or(primary),
279        primary,
280        input.secondary_y_data_column.unwrap_or(primary),
281    ];
282    let axes = [
283        Some(spec.x_axis),
284        Some(spec.y_axis()),
285        spec.secondary_y_axis(),
286    ];
287    let schema = ms.main_table().schema();
288    let has = |name| schema.is_some_and(|schema| schema.contains_column(name));
289    let first_axes = axes
290        .into_iter()
291        .zip(columns)
292        .zip(input.weight_axis_columns)
293        .filter_map(|((axis, column), weight_column)| {
294            let weight_column = match weight_column {
295                Some(MsDataColumn::Corrected) if !has("CORRECTED_DATA") => Some(MsDataColumn::Data),
296                weight_column => weight_column,
297            };
298            weight_column.or_else(|| {
299                axis.is_some_and(MsAxis::is_visibility_math)
300                    .then_some(column)
301            })
302        });
303    let additional = input
304        .additional_ordinates
305        .iter()
306        .filter_map(|&(axis, column)| axis.is_visibility_math().then_some(column));
307    let mut active = first_axes.chain(additional);
308    // A derived expression itself loads all columns, as does a mixed-column request.
309    // A model uses corrected or observed weighting when present in that iterator.
310    let first = active.next();
311    let all = first.is_some_and(|first| first.is_derived() || active.any(|column| column != first));
312    let has_observed = all && has("DATA");
313    let has_corrected = all && has("CORRECTED_DATA");
314    let primary = if has_observed {
315        MsDataColumn::Data
316    } else if has_corrected {
317        MsDataColumn::Corrected
318    } else {
319        first.unwrap_or(primary)
320    };
321    let kind = |column| column_weight_kind(column, has_observed, has_corrected);
322    PlotMsWeightKinds {
323        axes: columns.map(kind),
324        model: (input.extended_derived || uses_channel_derived(spec, input.x_data_column))
325            .then(|| kind(MsDataColumn::Model)),
326        primary: kind(primary),
327        row_weight: if has_corrected {
328            WeightKind::Calibrated
329        } else {
330            kind(primary)
331        },
332        row_sigma: if has_observed {
333            WeightKind::Observed
334        } else {
335            kind(primary)
336        },
337        has_observed,
338        has_corrected,
339    }
340}

342pub(super) struct ChannelStatistics {
343    pub(super) weight: f64,
344    pub(super) sigma: f64,
345    spectra: Vec<(bool, f32, f32)>,
346}
347
348impl ChannelStatistics {
349    pub(super) fn from_bins(
350        correlation: usize,
351        bins: &[ChannelBin],
352        flags: &ndarray::ArrayView2<'_, bool>,
353        weight: Option<&FloatGrid>,
354        inverse_variance: Option<&FloatGrid>,
355    ) -> Self {
356        let sum = |grid: Option<&FloatGrid>, bin: &ChannelBin| {
357            channel_bin_weight(grid, correlation, bin, flags, ChannelWeightMask::FlagsOnly)
358        };
359        let spectra: Vec<_> = bins
360            .iter()
361            .map(|bin| {
362                let (flagged, weight) = sum(weight, bin);
363                let variance_weight = sum(inverse_variance, bin).1;
364                (flagged, weight, sigma_from_weight(variance_weight))
365            })
366            .collect();
367        // The cache derives row statistics over the transformed spectra,
368        // using channel flags before the visibility projection's weight mask.
369        let weight = mean_weight_preferring_unflagged(
370            spectra
371                .iter()
372                .map(|&(flagged, weight, _)| (flagged, weight)),
373        );
374        let sigma = mean_weight_preferring_unflagged(
375            spectra.iter().map(|&(flagged, _, sigma)| (flagged, sigma)),
376        );
377        Self {
378            weight: f64::from(weight),
379            sigma: f64::from(sigma),
380            spectra,
381        }
382    }
383
384    pub(super) fn spectrum(&self, bin: usize) -> Option<(f64, f64)> {
385        let (flagged, weight, sigma) = self.spectra[bin];
386        (!flagged).then_some((f64::from(weight), f64::from(sigma)))
387    }
388}

390/// The next reducer receives the transform's scalar weight, averaged
391/// across the output channels of the first selected correlation.
392pub(super) fn channel_averaged_row_weight(
393    weights: Option<&FloatGrid>,
394    correlation: usize,
395    bins: &[ChannelBin],
396    flags: &ndarray::ArrayView2<'_, bool>,
397    mask: ChannelWeightMask,
398) -> f64 {
399    f64::from(mean_weight_preferring_unflagged(bins.iter().map(|bin| {
400        channel_bin_weight(weights, correlation, bin, flags, mask)
401    })))
402}
403
404#[derive(Clone, Copy)]
405pub(super) enum ChannelWeightMask {
406    FlagsOnly,
407    Positive,
408    Nonzero,
409}
410
411pub(super) fn channel_bin_weight(
412    weights: Option<&FloatGrid>,
413    correlation: usize,
414    bin: &ChannelBin,
415    flags: &ndarray::ArrayView2<'_, bool>,
416    mask: ChannelWeightMask,
417) -> (bool, f32) {
418    sum_channel_weights(
419        bin.channels().map(|channel| {
420            let weight = weights.map_or(1.0, |grid| {
421                grid.values[correlation * grid.chan_count + channel] as f32
422            });
423            (flags[(correlation, channel)], weight)
424        }),
425        mask,
426    )
427}
428
429pub(super) fn sum_channel_weights(
430    samples: impl Iterator<Item = (bool, f32)>,
431    mask: ChannelWeightMask,
432) -> (bool, f32) {
433    let mut sums = [0.0, 0.0];
434    let mut all_flagged = true;
435    for (flagged, weight) in samples {
436        all_flagged &= flagged;
437        sums[usize::from(flagged)] += weight;
438    }
439    let sum = sums[usize::from(all_flagged)];
440    let usable = !all_flagged
441        && match mask {
442            ChannelWeightMask::FlagsOnly => true,
443            ChannelWeightMask::Positive => sum > 0.0,
444            ChannelWeightMask::Nonzero => sum != 0.0 && !sum.is_nan(),
445        };
446    (!usable, sum)
447}

449pub(super) fn bin_has_weight(
450    spectrum: &FloatGrid,
451    corr: usize,
452    bin: &ChannelBin,
453    flags: &ndarray::ArrayView2<'_, bool>,
454) -> bool {
455    bin_weight(spectrum, corr, bin, flags) > 0.0
456}
457
458pub(super) fn bin_weight(
459    spectrum: &FloatGrid,
460    corr: usize,
461    bin: &ChannelBin,
462    flags: &ndarray::ArrayView2<'_, bool>,
463) -> f64 {
464    bin.channels()
465        .filter(|&channel| !flags[(corr, channel)])
466        .map(|channel| spectrum.values[corr * spectrum.chan_count + channel])
467        .sum()
468}
469
470pub(super) fn inverse_variance(spectrum: &FloatGrid) -> FloatGrid {
471    FloatGrid {
472        corr_count: spectrum.corr_count,
473        chan_count: spectrum.chan_count,
474        values: spectrum
475            .values
476            .iter()
477            .copied()
478            .map(|sigma| f64::from(weight_from_sigma(sigma as f32)))
479            .collect(),
480    }
481}

483pub(super) fn ordinate_columns(
484    task: &TaskInvocation<'_>,
485    ordinates: &[MsAxis],
486) -> Result<Vec<MsDataColumn>, TaskFailure> {
487    let supplied = task.value("ydatacolumn");
488    let names = super::calibration::axis_names(supplied)
489        .map_err(TaskFailure::validation)?
490        .into_iter()
491        .map(canonical_column)
492        .collect::<Result<Vec<_>, _>>()?;
493    let multiple_axes = matches!(task.value("yaxis"), Some(ParameterValue::Array(_)));
494    let multiple_columns = matches!(supplied, Some(ParameterValue::Array(_)));
495    // CASA validates all supplied columns, then applies them only to a matching axis form.
496    let names = if multiple_axes == multiple_columns {
497        names.as_slice()
498    } else {
499        &[]
500    };
501    let mut seen = HashSet::with_capacity(ordinates.len());
502    ordinates
503        .iter()
504        .enumerate()
505        .map(|(index, &axis)| {
506            let name = names.get(index).copied().unwrap_or("data");
507            let column = MsDataColumn::parse(name).map_err(TaskFailure::validation)?;
508            if !seen.insert((axis, column)) {
509                return Err(TaskFailure::validation("Remove duplicate y-axes."));
510            }
511            Ok(column)
512        })
513        .collect()
514}
515
516fn canonical_column(name: &str) -> Result<&str, TaskFailure> {
517    match name {
518        "data"
519        | "corrected"
520        | "model"
521        | "float"
522        | "data-model_vector"
523        | "data-model_scalar"
524        | "corrected-model_vector"
525        | "corrected-model_scalar"
526        | "data/model_vector"
527        | "data/model_scalar"
528        | "corrected/model_vector"
529        | "corrected/model_scalar" => Ok(name),
530        "residual" | "corrected-model" => Ok("corrected-model_vector"),
531        "data-model" => Ok("data-model_vector"),
532        "data/model" => Ok("data/model_vector"),
533        "corrected/model" => Ok("corrected/model_vector"),
534        _ => Err(TaskFailure::validation(format!(
535            "invalid ydatacolumn value '{name}'"
536        ))),
537    }
538}

540pub(super) fn resolve_storage_columns<const N: usize>(
541    ms: &MeasurementSet,
542    requested: [MsDataColumn; N],
543    axes: [Option<MsAxis>; N],
544    averaging: bool,
545) -> Result<([MsDataColumn; N], bool), TaskFailure> {
546    use MsDataColumn::*;
547
548    let schema = ms.main_table().schema();
549    let has = |name| schema.is_some_and(|schema| schema.column(name).is_some());
550    let has_data = has("DATA");
551    let has_float = has("FLOAT_DATA");
552    let has_corrected = has("CORRECTED_DATA");
553    let observed = if !has_data && has_float { Float } else { Data };
554    let columns = requested.map(|column| match column {
555        Data => observed,
556        Corrected if !has_corrected && (has_data || has_float) => observed,
557        CorrectedMinusModel if !has_corrected && has_data => DataMinusModel,
558        CorrectedMinusModelScalar if !has_corrected && has_data => DataMinusModelScalar,
559        CorrectedDivModel if !has_corrected && has_data => DataDivModel,
560        CorrectedDivModelScalar if !has_corrected && has_data => DataDivModelScalar,
561        other => other,
562    });
563    for (axis, column) in axes.into_iter().zip(columns) {
564        if !axis.is_some_and(MsAxis::is_visibility_math) {
565            continue;
566        }
567        if has_float && !has_data && matches!(column, DataMinusModel | DataMinusModelScalar) {
568            return Err(TaskFailure::validation(
569                "Data residuals not valid; use FLOAT_DATA.",
570            ));
571        }
572        if column == Float {
573            if !has_float {
574                return Err(TaskFailure::validation(
575                    "FLOAT_DATA is unavailable. Use DATA for PlotMS.",
576                ));
577            }
578            if matches!(axis, Some(MsAxis::Phase | MsAxis::Imaginary)) {
579                return Err(TaskFailure::validation(
580                    "PlotMS requires an amplitude or real axis for FLOAT_DATA.",
581                ));
582            }
583        }
584        let uses_observed = matches!(column, Data | Float)
585            || (column.is_derived() && !column.is_corrected_model_expression());
586        if averaging && has_float && uses_observed {
587            return Err(TaskFailure::validation(
588                "averaging MeasurementSet FLOAT_DATA is pending #1032",
589            ));
590        }
591    }
592    if columns != requested {
593        tracing::warn!(
594            observed = observed.as_str(),
595            "Using available observed columns for PlotMS."
596        );
597    }
598    Ok((columns, has_float && !averaging))
599}
