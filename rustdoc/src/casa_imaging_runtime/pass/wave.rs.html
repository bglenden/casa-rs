21/// Row chunks per worker in the placement stage; enough to balance rows
22/// whose channel counts differ after flagging and the support test.
23const CHUNKS_PER_WORKER: usize = 4;
24
25/// One owner's accumulator and backend.
26struct Owner<'w> {
27    acc: Option<GridAccumulator>,
28    backend: OwnerBackend<'w>,
29    images: Option<NormalImages>,
30}
31
32/// The backend an owner dispatches to. The CPU grids each placement chunk
33/// as it comes; a Metal owner first gathers its share of every chunk of a
34/// source block into `staging`, so each device dispatch is one block's
35/// worth of samples rather than one chunk's.
36enum OwnerBackend<'w> {
37    Cpu(CpuBackend),
38    Metal {
39        backend: Box<MetalBackend<'w>>,
40        staging: SampleBuffer,
41    },
42}

44/// One image domain's share of a wave.
45struct Domain<'w> {
46    router: Router,
47    owners: Vec<Owner<'w>>,
48    model: Option<&'w PreparedModelGrids>,
49}

51/// One row chunk's placements on the wave's planes, by owner, its
52/// prediction scratch and its rows' native-channel residuals.
53///
54/// Every buffer is allocated once, for a chunk's rows of the source's widest
55/// row ([`ChunkShape`]), and each owner's can take every placement of the
56/// chunk, so a chunk holds [`ChunkShape::bytes`] however its placements fall
57/// among the owners.
58struct Chunk {
59    rows: Range<usize>,
60    /// One row's placements.
61    scratch: SampleBuffer,
62    /// Each owner's placements, in row order.
63    owned: Vec<SampleBuffer>,
64    /// Every placement of the chunk, in row order, when the weight image has
65    /// one owner ([`Router::weight_owner`]): that owner grids `Mode::Weight`
66    /// over it.
67    weight_owned: SampleBuffer,
68    placed: u64,
69    backend: CpuBackend,
70    prediction: PredictionScratch,
71    predicted: Vec<Complex32>,
72    residual: Vec<Complex32>,
73}
74
75/// The largest chunk of a wave over blocks of `block`: at most
76/// [`chunk_placements`] placements, whatever rows the source puts in a
77/// block.
78#[derive(Clone, Copy)]
79struct ChunkShape {
80    /// Placements one native row places on any domain.
81    row: usize,
82    /// Rows a chunk places at once.
83    rows: usize,
84    /// Placements the chunk's rows place.
85    placements: usize,
86    /// Correlations per placement.
87    npol: usize,
88    /// Channels of one row.
89    channels: usize,
90    /// Correlations and channels of one row.
91    cells: usize,
92    /// Owners of the domain with the most.
93    owners: usize,
94    /// Whether the pass grids the weight image.
95    weight: bool,
96    /// Bytes the prediction scratch holds
97    /// ([`SpectralResampler::prediction_scratch`]).
98    prediction: u64,
99}
100
101impl ChunkShape {
102    fn new(domains: &[PassDomain<'_>], block: BlockShape, weight: bool, chunks: usize) -> Self {
103        let row = domains
104            .iter()
105            .map(|domain| domain.resampler.samples_per_row(block.spectrum))
106            .max()
107            .unwrap_or(0);
108        let owners = domains
109            .iter()
110            .map(|domain| domain.partition.owners())
111            .max()
112            .unwrap_or(1);
113        let rows = chunk_rows(row, block.rows, chunks, chunk_placements(owners));
114        Self {
115            row,
116            rows,
117            placements: rows * row,
118            npol: block.correlations,
119            channels: block.channels(),
120            cells: block.channels() * block.correlations,
121            owners,
122            weight,
123            prediction: SpectralResampler::prediction_scratch_bytes(
124                domains.iter().map(|domain| domain.resampler),
125                block.correlations,
126                block.channels(),
127            ),
128        }
129    }
130
131    /// Bytes a Metal owner's staging holds: its share of every chunk of a
132    /// slice, at most the whole slice.
133    const fn staging_bytes(self, chunks: usize) -> u64 {
134        SampleBuffer::bytes(self.npol, chunks * self.placements)
135    }
136
137    /// Placements the weight-owner buffer takes.
138    const fn weight_placements(self) -> usize {
139        if self.weight { self.placements } else { 0 }
140    }
141
142    /// Bytes a chunk of this shape holds: one row's placements, each owner's
143    /// and the weight owner's placements, one row's residual and model
144    /// visibilities, and its prediction scratch.
145    const fn bytes(self) -> u64 {
146        SampleBuffer::bytes(
147            self.npol,
148            self.row + self.owners * self.placements + self.weight_placements(),
149        ) + self.prediction
150            + (2 * self.cells * size_of::<Complex32>()) as u64
151    }
152}
153
154impl Chunk {
155    fn new(shape: ChunkShape, domains: &[PassDomain<'_>]) -> Self {
156        Self {
157            rows: 0..0,
158            scratch: SampleBuffer::with_capacity(shape.npol, shape.row),
159            owned: (0..shape.owners)
160                .map(|_| SampleBuffer::with_capacity(shape.npol, shape.placements))
161                .collect(),
162            weight_owned: SampleBuffer::with_capacity(shape.npol, shape.weight_placements()),
163            placed: 0,
164            backend: CpuBackend::new(),
165            prediction: SpectralResampler::prediction_scratch(
166                domains.iter().map(|domain| domain.resampler),
167                shape.npol,
168                shape.channels,
169            ),
170            predicted: Vec::with_capacity(shape.cells),
171            residual: Vec::with_capacity(shape.cells),
172        }
173    }
174
175    /// Bytes the chunk holds now, from its buffers' capacities: at most
176    /// [`ChunkShape::bytes`] while every buffer is within the shape.
177    fn capacity_bytes(&self) -> u64 {
178        self.scratch.capacity_bytes()
179            + self
180                .owned
181                .iter()
182                .map(SampleBuffer::capacity_bytes)
183                .sum::<u64>()
184            + self.weight_owned.capacity_bytes()
185            + self.prediction.capacity_bytes()
186            + ((self.predicted.capacity() + self.residual.capacity()) * size_of::<Complex32>())
187                as u64
188    }
189}

191/// Row chunks of a wave on `workers` workers.
192const fn chunks(workers: usize) -> usize {
193    workers * CHUNKS_PER_WORKER
194}
195
196/// Bytes a wave places with on `workers` workers over blocks of `block`,
197/// gridding the weight image when `weight`: its row chunks and, on Metal,
198/// each owner's staging. They grow with the source's block only up to
199/// [`CHUNK_PLACEMENTS`](super::CHUNK_PLACEMENTS) a chunk.
200pub(super) fn chunk_bytes(
201    domains: &[PassDomain<'_>],
202    backend: BackendChoice,
203    block: BlockShape,
204    workers: usize,
205    weight: bool,
206) -> u64 {
207    let count = chunks(workers);
208    let shape = ChunkShape::new(domains, block, weight, count);
209    let staging = match backend {
210        BackendChoice::Cpu => 0,
211        BackendChoice::Metal => (domains.len() * shape.owners) as u64 * shape.staging_bytes(count),
212    };
213    count as u64 * shape.bytes() + staging
214}

216/// Bytes the model visibilities of one block of `block` hold, which a wave
217/// with a model predicts for every row of the block at once.
218pub(super) const fn prediction_bytes(block: BlockShape) -> u64 {
219    (block.rows * block.channels() * block.correlations * size_of::<Complex32>()) as u64
220}

224    planes: PlaneRange,
225    domains: Vec<Domain<'w>>,

235impl<'w, 'p> Wave<'w, 'p> {
236    /// A wave over `planes` with each domain's prepared `models`, forming
237    /// residuals at native channels when `native_residuals`, on `workers`
238    /// workers over blocks of at most `block`. It places with
239    /// [`chunk_bytes`] and predicts into [`prediction_bytes`].
240    pub(super) fn new(
241        pass: &'w MajorCyclePass<'p>,
242        planes: PlaneRange,
243        models: Option<&'w [PreparedModelGrids]>,
244        native_residuals: bool,
245        block: BlockShape,
246        workers: usize,
247    ) -> Result<Self, PassError> {
248        let count = chunks(workers);
249        let shape = ChunkShape::new(pass.domains, block, pass.modes.weight, count);
250        let domains = pass
251            .domains
252            .iter()
253            .enumerate()
254            .map(|(index, domain)| {
255                let router = Router::new(&domain.partition, planes);
256                let operator = domain.operator;
257                let owners = (0..router.owners())
258                    .map(|owner| {
259                        let (range, tile) = router.target(&domain.partition, planes, owner);
260                        Ok(match pass.backend {
261                            BackendChoice::Cpu => Owner {
262                                acc: Some(operator.accumulator(range, tile, pass.modes)),
263                                backend: OwnerBackend::Cpu(CpuBackend::new()),
264                                images: None,
265                            },
266                            BackendChoice::Metal => {
267                                assert_eq!(
268                                    operator.precision(),
269                                    GridPrecision::F32,
270                                    "Metal grids are f32 (D2)"
271                                );
272                                Owner {
273                                    acc: Some(MetalBackend::accumulator(
274                                        operator.accumulator_layout(range, tile, pass.modes),
275                                    )?),
276                                    backend: OwnerBackend::Metal {
277                                        backend: Box::new(MetalBackend::new(
278                                            operator.cf(),
279                                            pass.metal_stage_timing,
280                                        )?),
281                                        staging: SampleBuffer::with_capacity(
282                                            shape.npol,
283                                            count * shape.placements,
284                                        ),
285                                    },
286                                    images: None,
287                                }
288                            }
289                        })
290                    })
291                    .collect::<Result<_, PassError>>()?;
292                Ok(Domain {
293                    router,
294                    owners,
295                    model: models.map(|models| &models[index]),
296                })
297            })
298            .collect::<Result<_, PassError>>()?;
299        // A pass with a model predicts each block, for its native residuals or
300        // for a visibility writer.
301        let predictions = if pass.model.is_some() {
302            block.rows * shape.cells
303        } else {
304            0
305        };
306        Ok(Self {
307            pass,
308            planes,
309            domains,
310            native_residuals,
311            check_spacing: false,
312            shape,
313            chunks: (0..count)
314                .map(|_| Chunk::new(shape, pass.domains))
315                .collect(),
316            predictions: Vec::with_capacity(predictions),
317            samples: 0,
318        })
319    }

321    /// Check every row's native spacing against the pass's
322    /// `native_spacing_hz`, which sized this wave's model halo.

325        self
326    }

330        self.samples

333    /// Place every row of `block` on every domain and accumulate the
334    /// placements; with a visibility sink, first hand it the block, with
335    /// every selected sample's prediction when it asks for them.
336    ///
337    /// When the wave forms residuals at native channels, each row's samples
338    /// are placed from `V − Σ_d A_d·m_d` at its native channels and gridded
339    /// as data; otherwise the domain's model is subtracted at each placed
340    /// sample.
341    pub(super) fn consume(
342        &mut self,
343        block: &NativeBlock,
344        team: &WorkerTeam,
345        visibilities: Option<&mut VisibilitySink<'_>>,
346    ) -> Result<(), PassError> {
347        let rows = block.len();
348        if self.check_spacing {
349            let bound_hz = self.pass.native_spacing_hz;
350            for row in 0..rows {
351                let observed_hz = block
352                    .row(0, row)
353                    .frequencies_hz
354                    .windows(2)
355                    .fold(0.0_f64, |widest, pair| {
356                        widest.max((pair[1] - pair[0]).abs())
357                    });
358                if observed_hz > bound_hz {
359                    return Err(PassError::NativeSpacing {
360                        observed_hz,
361                        bound_hz,
362                    });
363                }
364            }
365        }
366        let sink_predictions = visibilities.as_ref().is_some_and(|sink| sink.predictions);
367        if self.native_residuals || sink_predictions {
368            let count = self.chunks.len().min(rows.max(1));
369            for (index, chunk) in self.chunks[..count].iter_mut().enumerate() {
370                chunk.rows = index * rows / count..(index + 1) * rows / count;
371            }
372            self.predict(block, team, count)?;
373        } else {
374            self.predictions.clear();
375        }
376        if let Some(sink) = visibilities {
377            let predictions: &[Complex32] = if sink.predictions {
378                &self.predictions
379            } else {
380                &[]
381            };
382            (sink.write)(block, predictions).map_err(PassError::VisibilityWrite)?;
383        }
384        // Slices of the block, each owner's in row order as a whole block's
385        // would be, so the accumulation is the same.
386        let chunks = self.chunks.len();
387        for slice in slices(rows, chunks, self.shape.rows) {
388            let mut count = 0;
389            for (chunk, rows) in self.chunks.iter_mut().zip(chunk_ranges(slice, chunks)) {
390                chunk.rows = rows;
391                count += 1;
392            }
393            for index in 0..self.domains.len() {
394                let placed = self.place(block, team, count, index)?;
395                if index == 0 {
396                    self.samples += placed;
397                }
398            }
399        }
400        Ok(())
401    }

403    /// Place the rows of `block` on domain `index` and accumulate them;
404    /// returns the samples placed.
405    fn place(
406        &mut self,
407        block: &NativeBlock,
408        team: &WorkerTeam,
409        count: usize,
410        index: usize,
411    ) -> Result<u64, PassError> {
412        let pass = self.pass;
413        let target = &pass.domains[index];
414        let domain = &mut self.domains[index];
415        let owners = domain.owners.len();
416        let router = &domain.router;
417        let planes = self.planes;
418        let predictions = &self.predictions;
419        let native_residuals = self.native_residuals;
420        let cells = block.channels() * block.correlations();
421        let shape = self.shape;
422        // With region owners the weight image has one owner, which must see
423        // every placement; plane owners grid their own samples' weights.
424        let route_weight = pass.modes.weight && router.weight_owner().is_some();
425        team.for_each_mut(&mut self.chunks[..count], |_, chunk| {
426            chunk.placed = 0;
427            chunk.owned[..owners]
428                .iter_mut()
429                .for_each(SampleBuffer::clear);
430            chunk.weight_owned.clear();
431            for row in chunk.rows.clone() {
432                chunk.scratch.clear();
433                let native = block.row(index, row);
434                let native = if native_residuals {
435                    let model = &predictions[row * cells..(row + 1) * cells];
436                    chunk.residual.clear();
437                    chunk.residual.extend(
438                        native
439                            .values
440                            .iter()
441                            .zip(model)
442                            .map(|(value, model)| value - model),
443                    );
444                    NativeRow {
445                        values: &chunk.residual,
446                        ..native
447                    }
448                } else {
449                    native
450                };
451                target.resampler.place(
452                    target.operator,
453                    pass.weighting,
454                    &native,
455                    &mut chunk.scratch,
456                )?;
457                debug_assert!(
458                    chunk.scratch.len() <= shape.row,
459                    "a row places within the chunk's per-row bound"
460                );
461                let placed = chunk.scratch.block();
462                for (sample, placement) in placed.placements.iter().enumerate() {
463                    if !planes.contains(placement.plane) {
464                        continue;
465                    }
466                    let owner = router.owner(target.operator, placement);
467                    chunk.owned[owner].push(
468                        *placement,
469                        placed.values_of(sample),
470                        placed.weights_of(sample),
471                    );
472                    if route_weight {
473                        chunk.weight_owned.push(
474                            *placement,
475                            placed.values_of(sample),
476                            placed.weights_of(sample),
477                        );
478                    }
479                    chunk.placed += 1;
480                }
481            }
482            debug_assert!(
483                chunk.placed as usize <= shape.placements
484                    && chunk.capacity_bytes() == shape.bytes(),
485                "a chunk places within the placements it was allocated for"
486            );
487            Ok::<_, PassError>(())
488        })?;
489        let chunks = &self.chunks[..count];
490        let model = domain.model.filter(|_| !native_residuals);
491        let weight_owner = router.weight_owner();
492        team.for_each_mut(&mut domain.owners, |owner_index, owner| {
493            let acc = owner
494                .acc
495                .as_mut()
496                .expect("owners accumulate until finished");
497            let operator = target.operator;
498            // Each owner grids the weight image over its own placements
499            // unless the image has one owner, who grids every placement
500            // below.
501            let own_weight = weight_owner.is_none();
502            match &mut owner.backend {
503                OwnerBackend::Cpu(backend) => {
504                    for chunk in chunks {
505                        let block = chunk.owned[owner_index].block();
506                        if !block.is_empty() {
507                            accumulate(pass, operator, backend, &block, model, acc, own_weight)?;
508                        }
509                    }
510                }
511                OwnerBackend::Metal { backend, staging } => {
512                    staging.clear();
513                    for chunk in chunks {
514                        let block = chunk.owned[owner_index].block();
515                        for sample in 0..block.len() {
516                            staging.push(
517                                block.placements[sample],
518                                block.values_of(sample),
519                                block.weights_of(sample),
520                            );
521                        }
522                    }
523                    if !staging.is_empty() {
524                        let backend: &mut MetalBackend<'_> = backend;
525                        accumulate(
526                            pass,
527                            operator,
528                            backend,
529                            &staging.block(),
530                            model,
531                            acc,
532                            own_weight,
533                        )?;
534                    }
535                }
536            }
537            if pass.modes.weight && weight_owner == Some(owner_index) {
538                for chunk in chunks {
539                    let block = chunk.weight_owned.block();
540                    if block.is_empty() {
541                        continue;
542                    }
543                    let work = Work::Grid {
544                        mode: Mode::Weight,
545                        acc,
546                    };
547                    match &mut owner.backend {
548                        OwnerBackend::Cpu(backend) => backend.apply(&block, operator.cf(), work)?,
549                        OwnerBackend::Metal { backend, .. } => {
550                            backend.apply(&block, operator.cf(), work)?;
551                        }
552                    }
553                }
554            }
555            Ok::<_, PassError>(())
556        })?;
557        Ok(chunks.iter().map(|chunk| chunk.placed).sum())
558    }

560    /// Model visibilities of every selected sample of `block`, summed over
561    /// every domain's model (`SIMapperCollection::degrid`), `[row][channel]
562    /// [correlation]`, into `self.predictions`; zero without a model.
563    fn predict(
564        &mut self,
565        block: &NativeBlock,
566        team: &WorkerTeam,
567        count: usize,
568    ) -> Result<(), PassError> {
569        let cells = block.channels() * block.correlations();
570        let rows = block.len();
571        self.predictions.clear();
572        self.predictions.resize(rows * cells, Complex32::default());
573        if self.domains.iter().all(|domain| domain.model.is_none()) {
574            return Ok(());
575        }
576        let pass = self.pass;
577        let shape = self.shape;
578        let models = self
579            .domains
580            .iter()
581            .map(|domain| domain.model)
582            .collect::<Vec<_>>();
583        let mut pieces = Vec::with_capacity(count);
584        let mut remaining = self.predictions.as_mut_slice();
585        for chunk in &mut self.chunks[..count] {
586            let (head, tail) = remaining.split_at_mut(chunk.rows.len() * cells);
587            remaining = tail;
588            pieces.push((chunk, head));
589        }
590        team.for_each_mut(&mut pieces, |_, (chunk, out)| {
591            chunk.predicted.resize(cells, Complex32::default());
592            for (local, row) in chunk.rows.clone().enumerate() {
593                let out = &mut out[local * cells..(local + 1) * cells];
594                for (index, (target, model)) in pass.domains.iter().zip(&models).enumerate() {
595                    let Some(model) = model else {
596                        continue;
597                    };
598                    target.resampler.predict_row(
599                        target.operator,
600                        &mut chunk.backend,
601                        model,
602                        &block.row(index, row),
603                        &mut chunk.prediction,
604                        &mut chunk.predicted,
605                    )?;
606                    for (sum, value) in out.iter_mut().zip(&chunk.predicted) {
607                        *sum += value;
608                    }
609                }
610            }
611            debug_assert_eq!(
612                chunk.capacity_bytes(),
613                shape.bytes(),
614                "a chunk predicts within the buffers it was allocated"
615            );
616            Ok::<_, PassError>(())
617        })
618    }

620    /// Transform the accumulated grids into this wave's normal images, one
621    /// per domain.
622    pub(super) fn finish(self, team: &WorkerTeam) -> Result<Vec<NormalImages>, PassError> {
623        let pass = self.pass;
624        let planes = self.planes;
625        self.domains
626            .into_iter()
627            .zip(pass.domains)
628            .map(|(mut domain, target)| {
629                let operator = target.operator;
630                match &target.partition {
631                    Partition::Planes { .. } => {
632                        team.for_each_mut(&mut domain.owners, |_, owner| {
633                            let acc = owner.acc.take().expect("each owner finishes once");
634                            owner.images = Some(operator.finish(acc)?);
635                            Ok::<_, PassError>(())
636                        })?;
637                        Ok(concatenate(
638                            domain
639                                .owners
640                                .into_iter()
641                                .map(|owner| owner.images.expect("every owner finished")),
642                        ))
643                    }
644                    Partition::Regions { .. } => {
645                        let mut merged = operator.accumulator(planes, None, pass.modes);
646                        for owner in &domain.owners {
647                            merged
648                                .merge_from(owner.acc.as_ref().expect("owners hold their tiles"))?;
649                        }
650                        drop(domain.owners);
651                        Ok(operator.finish(merged)?)
652                    }
653                }
654            })
655            .collect()
656    }
657}

659/// Grid one owner's share of a block in every mode the pass holds; with a
660/// model the data terms receive the residual `V − A·m`.
661fn accumulate(
662    pass: &MajorCyclePass<'_>,
663    operator: &casa_imaging_operator::MeasurementOperator,
664    backend: &mut dyn GridBackend,
665    block: &casa_imaging_operator::SampleBlock<'_>,
666    model: Option<&PreparedModelGrids>,
667    acc: &mut GridAccumulator,
668    weight: bool,
669) -> Result<(), PassError> {
670    let cf = operator.cf();
671    if pass.modes.data {
672        let work = match model {
673            Some(model) => Work::ResidualGrid {
674                model,
675                acc,
676                residual_out: None,
677            },
678            None => Work::Grid {
679                mode: Mode::Data,
680                acc,
681            },
682        };
683        backend.apply(block, cf, work)?;
684    }
685    if pass.modes.psf {
686        backend.apply(
687            block,
688            cf,
689            Work::Grid {
690                mode: Mode::Psf,
691                acc,
692            },
693        )?;
694    }
695    if pass.modes.weight && weight {
696        backend.apply(
697            block,
698            cf,
699            Work::Grid {
700                mode: Mode::Weight,
701                acc,
702            },
703        )?;
704    }
705    Ok(())
706}

708/// Join per-owner images of consecutive plane ranges.
709pub(super) fn concatenate(parts: impl IntoIterator<Item = NormalImages>) -> NormalImages {
710    let mut parts = parts.into_iter();
711    let mut joined = parts.next().expect("a wave has at least one owner");
712    for part in parts {
713        assert!(
714            part.pols == joined.pols
715                && part.data_terms == joined.data_terms
716                && part.psf_terms == joined.psf_terms
717                && part.first_plane == joined.first_plane + joined.planes.len() as u32,
718            "owner images must continue the plane sequence with the same layout"
719        );
720        joined.planes.extend(part.planes);
721    }
722    joined
723}

726mod tests {

729    /// Every owner's buffer in a chunk can take all of the chunk's
730    /// placements, and a pass has four chunks per worker and up to one
731    /// owner per worker; past four owners a chunk takes proportionally
732    /// fewer, so the chunks' storage grows linearly with the workers, and up
733    /// to four a chunk takes the full cap.
734    #[test]
735    fn chunk_storage_grows_linearly_with_workers() {
736        let full = chunk_placements(1);
737        for workers in 1..=128 {
738            let owners = workers;
739            let storage = chunks(workers) * owners * chunk_placements(owners);
740            assert!(
741                storage <= CHUNKS_PER_WORKER * 4 * full * workers,
742                "{workers} workers place into {storage} placements of storage"
743            );
744        }
745        assert_eq!(chunk_placements(4), full);
746    }
