7/// Plain seven-tap geometry and value introduced by the shared science kernel.
8#[doc(hidden)]
9#[repr(C)]
10#[derive(Clone, Copy, Debug, Default, bytemuck::Pod, bytemuck::Zeroable)]
11pub struct SpatialTap {
12    /// First grid x cell.
13    pub x: u32,
14    /// First grid y cell.
15    pub y: u32,
16    /// Seven-element x convolution-table row.
17    pub x_weights: u32,
18    /// Seven-element y convolution-table row.
19    pub y_weights: u32,
20    /// Complex Float contribution.
21    pub value: [f32; 2],
22}
23
24impl SpatialTap {
25    pub(super) fn new(
26        taps: crate::spectral_operator::SampleTaps,
27        value: Complex32,
28    ) -> Result<Self, SpectralOperatorError> {
29        Ok(Self {
30            x: taps
31                .x
32                .start
33                .try_into()
34                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
35            y: taps
36                .y
37                .start
38                .try_into()
39                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
40            x_weights: taps
41                .x
42                .weight_index
43                .try_into()
44                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
45            y_weights: taps
46                .y
47                .weight_index
48                .try_into()
49                .map_err(|_| SpectralOperatorError::ResidencyOverflow)?,
50            value: [value.re, value.im],
51        })
52    }
53}

55/// Grid role; model planes retain their shared spectral-support ordinal.
56#[doc(hidden)]
57#[derive(Clone, Copy, Debug, PartialEq, Eq)]
58pub enum SpatialField {
59    Dirty,
60    Residual,
61    Psf,
62    Model,
63}

65/// One prepared model plane in a bounded prediction command batch.
66#[doc(hidden)]
67pub struct SpatialPredictionBatch {
68    /// Ordinal in the band's model-support planes.
69    pub plane: usize,
70    /// Prepared source-refill requests for this plane.
71    pub taps: Vec<SpatialTap>,
72    /// Exact-sized prediction destination in matching request order.
73    pub values: Vec<Complex32>,
74}

76/// One output plane in a bounded accumulation command batch.
77#[doc(hidden)]
78pub struct SpatialGridBatch {
79    /// Output grid role.
80    pub field: SpatialField,
81    /// Ordinal in the band's core output planes.
82    pub plane: usize,
83    /// Prepared contributions for the current source refill.
84    pub taps: Vec<SpatialTap>,
85}

87/// Runtime-owned resident grids and bounded synchronous dispatches. Only spatial
88/// convolution moves across this seam; row interpolation, flags, polarization,
89/// phase rotation, weights and image conversion remain in this module.
90#[doc(hidden)]
91pub trait CubeSpatialBackend {
92    /// Initialize the active band's grids and convolution table once per wave.
93    fn initialize(
94        &mut self,
95        shape: [usize; 2],
96        weights: &[[f32; 7]],
97        fields: [usize; 4],
98        model: &[Complex32],
99    ) -> Result<(), SpectralOperatorError>;
100    /// Evaluate the active model planes together in one substantial batch.
101    fn degrid(
102        &mut self,
103        batches: &mut [SpatialPredictionBatch],
104    ) -> Result<(), SpectralOperatorError>;
105    /// Accumulate the active output fields together in one substantial batch.
106    fn grid(&mut self, batches: &[SpatialGridBatch]) -> Result<(), SpectralOperatorError>;
107    /// Return one completed grid for the shared CPU FFT/image conversion.
108    fn download(
109        &mut self,
110        field: SpatialField,
111        plane: usize,
112        values: &mut [Complex32],
113    ) -> Result<(), SpectralOperatorError>;
114}

116impl BandPlan {
117    /// Whether this initial band has no model prediction dependency.
118    pub fn is_initial_zero(&self) -> bool {
119        self.phase == BandPhase::InitialZero
120    }
121
122    /// Float seven-tap table shared by every grid in the wave.
123    pub fn spatial_weight_bytes() -> usize {
124        StandardConvolution::weight_row_count() * 7 * size_of::<f32>()
125    }
126    /// Largest simultaneous list of model or output dispatches for this band.
127    pub fn spatial_command_capacity(&self) -> usize {
128        let outputs = match self.phase {
129            BandPhase::InitialZero => 2,
130            BandPhase::Full => 3,
131            BandPhase::Residual => 1,
132        };
133        self.support.model.len().max(outputs * self.core.len())
134    }
135    /// Additional shared-device grid bytes; host FFT grids are charged by memory().
136    pub fn spatial_grid_bytes(&self) -> Result<usize, SpectralOperatorError> {
137        let outputs = match self.phase {
138            BandPhase::InitialZero => 2,
139            BandPhase::Full => 3,
140            BandPhase::Residual => 1,
141        };
142        let models = if self.phase == BandPhase::InitialZero {
143            0
144        } else {
145            self.support.model.len()
146        };
147        self.geometry.grid_shape[0]
148            .checked_mul(self.geometry.grid_shape[1])
149            .and_then(|n| n.checked_mul(outputs * self.core.len() + models))
150            .and_then(|n| n.checked_mul(size_of::<Complex32>()))
151            .ok_or(SpectralOperatorError::ResidencyOverflow)
152    }
153
154    /// Output-field contributions per row with `fine` entries per coarse output
155    /// plane. Each CASA fine sample emits one, as in the residual descriptor bound.
156    fn spatial_contributions_per_row(&self, fine: usize) -> Option<usize> {
157        let outputs = match self.phase {
158            BandPhase::InitialZero => 2,
159            BandPhase::Full => 3,
160            BandPhase::Residual => 1,
161        };
162        self.core.len().checked_mul(fine)?.checked_mul(outputs)
163    }
164
165    /// Maximum packed requests across a band's command batch for an input refill.
166    pub fn spatial_request_capacity(&self, rows: usize) -> Result<usize, SpectralOperatorError> {
167        let native = if self.phase == BandPhase::InitialZero {
168            0
169        } else {
170            self.support.native.len()
171        };
172        let predictions = native.checked_mul(2);
173        let contributions = self.spatial_contributions_per_row(self.fine_per_output);
174        predictions
175            .zip(contributions)
176            .and_then(|(a, b)| rows.checked_mul(a.max(b).max(1)))
177            .ok_or(SpectralOperatorError::ResidencyOverflow)
178    }
179
180    /// Conservative host preparation storage for one active accelerated band.
181    pub fn spatial_host_bytes(&self, rows: usize) -> Result<usize, SpectralOperatorError> {
182        let native = if self.phase == BandPhase::InitialZero {
183            0
184        } else {
185            self.support.native.len()
186        };
187        let outputs = match self.phase {
188            BandPhase::InitialZero => 2,
189            BandPhase::Full => 3,
190            BandPhase::Residual => 1,
191        };
192        // Two spectral terms per native channel. Growing request/destination
193        // vectors may have up to twice their live length; results are exact-sized.
194        // Grid batches start at one request per row and double to hold every
195        // fine sample of their coarse output plane.
196        let contributions = self
197            .spatial_contributions_per_row(self.fine_per_output.next_power_of_two())
198            .and_then(|n| n.checked_mul(size_of::<SpatialTap>()));
199        let samples = native
200            .checked_mul(
201                4 * (size_of::<SpatialTap>() + size_of::<PredictionDestination>())
202                    + 2 * size_of::<Complex32>(),
203            )
204            .and_then(|n| n.checked_add(native * size_of::<Complex64>()))
205            .zip(contributions)
206            .and_then(|(n, c)| n.checked_add(c))
207            .and_then(|n| n.checked_mul(rows))
208            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
209        let headers = self
210            .support
211            .model
212            .len()
213            .checked_mul(
214                size_of::<SpatialPredictionBatch>()
215                    + size_of::<Vec<PredictionDestination>>()
216                    + 8 * (size_of::<SpatialTap>() + size_of::<PredictionDestination>()),
217            )
218            .and_then(|n| n.checked_add(outputs * self.core.len() * size_of::<SpatialGridBatch>()))
219            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
220        let preparation = samples
221            .checked_add(headers)
222            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
223        // Initialization converts the convolution table before packing begins.
224        // Its transient storage still applies to an empty source refill.
225        Ok(preparation.max(Self::spatial_weight_bytes()))
226    }
227}

234impl EpochBand<'_> {
235    /// Load resident spatial buffers from the already prepared shared model FFT.
236    pub fn initialize_spatial(
237        &self,
238        backend: &mut impl CubeSpatialBackend,
239    ) -> Result<(), SpectralOperatorError> {
240        let w = &self.workspace;
241        let weights = w.convolution.float_weights();
242        backend.initialize(
243            w.geometry.grid_shape,
244            &weights,
245            [
246                w.dirty.len_of(Axis(0)),
247                w.residual.len_of(Axis(0)),
248                w.psf.len_of(Axis(0)),
249                w.forward.len_of(Axis(0)),
250            ],
251            w.forward.as_slice().expect("contiguous cube model"),
252        )
253    }
254
255    /// Prepare predictions and contributions in source-refill sized batches.
256    pub fn consume_source_window_spatial(
257        &mut self,
258        block: NativeBlockView<'_>,
259        layout: &NativeLayout,
260        selected: Range<usize>,
261        output_hz: &[f64],
262        polarization: &PolarizationOperator,
263        backend: &mut impl CubeSpatialBackend,
264    ) -> Result<(), SpectralOperatorError> {
265        if output_hz.len() != self.generation.shape().coefficients()
266            || selected.len() != layout.channels.len()
267        {
268            return Err(SpectralOperatorError::ProblemMismatch);
269        }
270        if self.native_range.is_empty() {
271            return Ok(());
272        }
273        if self.native_range.start < selected.start || self.native_range.end > selected.end {
274            return Err(SpectralOperatorError::IncompleteSpectralHalo);
275        }
276        if self.workspace.single_channel.is_some() {
277            return Err(SpectralOperatorError::UnsupportedProblem);
278        }
279        let native =
280            self.native_range.start - selected.start..self.native_range.end - selected.start;
281        self.workspace
282            .consume_spatial(block, layout, native, output_hz, polarization, backend)
283    }
284
285    /// Download only at the wave boundary, then use the ordinary completion path.
286    pub fn download_spatial(
287        &mut self,
288        backend: &mut impl CubeSpatialBackend,
289    ) -> Result<(), SpectralOperatorError> {
290        for (field, grids) in [
291            (SpatialField::Dirty, &mut self.workspace.dirty),
292            (SpatialField::Residual, &mut self.workspace.residual),
293            (SpatialField::Psf, &mut self.workspace.psf),
294        ] {
295            for (plane, mut grid) in grids.axis_iter_mut(Axis(0)).enumerate() {
296                backend.download(
297                    field,
298                    plane,
299                    grid.as_slice_mut().expect("contiguous cube grid"),
300                )?;
301            }
302        }
303        Ok(())
304    }
305}

307impl BandWorkspace {
308    pub(in crate::streaming_cube) fn consume_spatial(
309        &mut self,
310        block: NativeBlockView<'_>,
311        layout: &NativeLayout,
312        native: Range<usize>,
313        output_hz: &[f64],
314        polarization: &PolarizationOperator,
315        backend: &mut impl CubeSpatialBackend,
316    ) -> Result<(), SpectralOperatorError> {
317        let rows = block.metadata.len();
318        let native_rows = block.rows(layout, 0..block.channels, native.clone())?;
319        let stride = native.len();
320        let predicts = self.phase != BandPhase::InitialZero;
321        let mut predictions = if predicts {
322            vec![Complex64::default(); rows * stride]
323        } else {
324            Vec::new()
325        };
326        if predicts {
327            let mut batches: Vec<_> = (0..self.model_channels.len())
328                .map(|plane| SpatialPredictionBatch {
329                    plane,
330                    taps: Vec::new(),
331                    values: Vec::new(),
332                })
333                .collect();
334            let mut destinations: Vec<Vec<PredictionDestination>> =
335                (0..self.model_channels.len()).map(|_| Vec::new()).collect();
336            let mut previous: Option<CachedRowSupport<'_>> = None;
337            for row_index in 0..rows {
338                let mut row = native_rows.row(row_index);
339                let frequencies = row.frequencies_hz;
340                let reuse = previous.as_ref().is_some_and(|(hz, pair, _, _)| {
341                    *hz == frequencies && *pair == row.original_pair_hz
342                });
343                let support = if reuse {
344                    previous.as_ref().unwrap().2.clone()
345                } else {
346                    BandSupport::native_window(
347                        output_hz,
348                        self.core.clone(),
349                        frequencies,
350                        row.original_pair_hz,
351                    )?
352                };
353                if support.is_empty() {
354                    previous = Some((frequencies, row.original_pair_hz, support, None));
355                    continue;
356                }
357                row.restrict(support.clone())?;
358                if !reuse {
359                    let stencil = RowStencil::compile(
360                        &row,
361                        output_hz,
362                        self.core.clone(),
363                        &self.model_channels,
364                        self.phase,
365                    )?;
366                    previous = Some((
367                        frequencies,
368                        row.original_pair_hz,
369                        support.clone(),
370                        Some(stencil),
371                    ));
372                }
373                let stencil = previous.as_ref().unwrap().3.as_ref().unwrap();
374                for channel in 0..row.channels.len() {
375                    let terms = &stencil.prediction_terms[channel];
376                    for term in terms {
377                        if !self.forward_nonzero[term.plane] {
378                            continue;
379                        }
380                        if let Some(taps) = self.convolution.taps([
381                            row.uvw_m[0] * term.wavelength_scale,
382                            row.uvw_m[1] * term.wavelength_scale,
383                        ]) {
384                            batches[term.plane]
385                                .taps
386                                .push(SpatialTap::new(taps, Complex32::default())?);
387                            destinations[term.plane].push(PredictionDestination {
388                                index: row_index * stride + support.start + channel,
389                                factor: phase(row.phase_shift_m, term.frequency_hz).conj()
390                                    * term.factor,
391                            });
392                        }
393                    }
394                }
395            }
396            for batch in &mut batches {
397                batch.values = vec![Complex32::default(); batch.taps.len()];
398            }
399            backend.degrid(&mut batches)?;
400            for (batch, destinations) in batches.into_iter().zip(destinations) {
401                for (value, destination) in batch.values.into_iter().zip(destinations) {
402                    predictions[destination.index] += widen(value) * destination.factor;
403                }
404            }
405        }
406        let fields = if self.phase == BandPhase::Residual {
407            &[SpatialField::Residual][..]
408        } else if self.phase == BandPhase::InitialZero {
409            &[SpatialField::Dirty, SpatialField::Psf][..]
410        } else {
411            &[
412                SpatialField::Dirty,
413                SpatialField::Residual,
414                SpatialField::Psf,
415            ][..]
416        };
417        let plane_count = self.core.len();
418        let mut batches: Vec<_> = fields
419            .iter()
420            .flat_map(|&field| {
421                (0..plane_count).map(move |plane| SpatialGridBatch {
422                    field,
423                    plane,
424                    taps: Vec::with_capacity(rows),
425                })
426            })
427            .collect();
428        let mut previous: Option<CachedRowSupport<'_>> = None;
429        for row_index in 0..rows {
430            let mut row = native_rows.row(row_index);
431            let frequencies = row.frequencies_hz;
432            let reuse = previous.as_ref().is_some_and(|(hz, pair, _, _)| {
433                *hz == frequencies && *pair == row.original_pair_hz
434            });
435            let support = if reuse {
436                previous.as_ref().unwrap().2.clone()
437            } else {
438                BandSupport::native_window(
439                    output_hz,
440                    self.core.clone(),
441                    frequencies,
442                    row.original_pair_hz,
443                )?
444            };
445            if support.is_empty() {
446                previous = Some((frequencies, row.original_pair_hz, support, None));
447                continue;
448            }
449            row.restrict(support.clone())?;
450            if !reuse {
451                let stencil = RowStencil::compile(
452                    &row,
453                    output_hz,
454                    self.core.clone(),
455                    &self.model_channels,
456                    self.phase,
457                )?;
458                previous = Some((
459                    frequencies,
460                    row.original_pair_hz,
461                    support.clone(),
462                    Some(stencil),
463                ));
464            }
465            let stencil = previous.as_ref().unwrap().3.as_ref().unwrap();
466            let count = row.channels.len();
467            let mut accumulator = self.begin_row_cached(&row, polarization, stencil)?;
468            accumulator.push_with(
469                0..count,
470                |_, _, _, polarization, channel| {
471                    polarization
472                        .predict(&[if predicts {
473                            predictions[row_index * stride + support.start + channel]
474                        } else {
475                            Complex64::default()
476                        }])
477                        .map_err(|_| SpectralOperatorError::GeneratedNonfinite)
478                },
479                |band, channel, frequency, row, observed, predicted, weight| {
480                    band.grid_sample_with(
481                        channel,
482                        frequency,
483                        row,
484                        observed,
485                        predicted,
486                        weight,
487                        |field, _, _, plane, taps, value| {
488                            let lane = fields
489                                .iter()
490                                .position(|f| *f == field)
491                                .expect("active output field");
492                            batches[lane * plane_count + plane]
493                                .taps
494                                .push(SpatialTap::new(taps, value)?);
495                            Ok(())
496                        },
497                    )
498                },
499            )?;
500            accumulator.finish()?;
501        }
502        backend.grid(&batches)?;
503        Ok(())
504    }
