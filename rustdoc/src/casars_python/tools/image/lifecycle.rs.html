18pub(super) fn resolve_image_path<'a>(py: Python<'_>, name: &'a str) -> PyResult<Cow<'a, str>> {
19    let expanded = expand_path(name)
20        .ok_or_else(|| raised("Path::expandName: recursive environment variable"))?;
21    if expanded
22        .strip_prefix('~')
23        .is_some_and(|suffix| !suffix.is_empty() && !suffix.starts_with('/'))
24    {
25        let resolved = py
26            .import("os.path")?
27            .call_method1("expanduser", (expanded.as_ref(),))?
28            .extract()?;
29        return Ok(Cow::Owned(resolved));
30    }
31    Ok(expanded)
32}

34pub(super) fn rename_destination(expanded: &str, name: &str) -> PyResult<PathBuf> {
35    let destination = std::path::absolute(expanded).map_err(raised)?;
36    if destination.components().next_back() == Some(Component::ParentDir) {
37        return Err(raised(format!(
38            "Cannot rename to {name} because its path ends with '..'"
39        )));
40    }
41    let lexical = lexical_absolute(&destination);
42    if destination != lexical
43        && let Some(physical_parent) = destination
44            .parent()
45            .and_then(|path| path.canonicalize().ok())
46        && lexical
47            .parent()
48            .and_then(|path| path.canonicalize().ok())
49            .as_ref()
50            != Some(&physical_parent)
51    {
52        return Err(raised(format!(
53            "Cannot rename to {name} because its path resolves to a different directory after removing '..'"
54        )));
55    }
56    Ok(destination)
57}

59pub(super) fn open_image(infile: &str, expanded: &str) -> PyResult<AttachedImage> {
60    open_image_with_cache(infile, expanded, true)
61}
62
63fn open_image_with_cache(
64    infile: &str,
65    expanded: &str,
66    component_cache: bool,
67) -> PyResult<AttachedImage> {
68    if infile.is_empty() {
69        return Err(raised("Exception: File name is empty."));
70    }
71    let path = Path::new(expanded);
72    if !path.exists() {
73        return Err(raised(format!("Exception: File {infile} does not exist..")));
74    }
75    let path = std::path::absolute(path).map_err(raised)?;
76    let absolute = lexical_absolute(&path);
77    AttachedImage::open_resolved_with_cache(&absolute, &path, Path::new(infile), component_cache)
78}

80#[pymethods]
81impl PyImageTool {
82    #[pyo3(signature = (infile, cache = true), text_signature = "($self, infile, cache=True)")]
83    fn open(&self, py: Python<'_>, infile: String, cache: bool) -> PyResult<bool> {
84        let expanded = resolve_image_path(py, &infile);
85        py.detach(|| {
86            let mut state = self.state();
87            state.image = None;
88            state.image = Some(open_image_with_cache(&infile, expanded?.as_ref(), cache)?);
89            Ok(true)
90        })
91    }
92
93    fn newimage(&self, py: Python<'_>, infile: &str) -> PyResult<Py<Self>> {
94        self.newimagefromfile(py, infile)
95    }
96
97    fn newimagefromfile(&self, py: Python<'_>, infile: &str) -> PyResult<Py<Self>> {
98        let expanded = resolve_image_path(py, infile)?;
99        let image = py.detach(|| open_image(infile, expanded.as_ref()))?;
100        Py::new(py, Self::attached(image))
101    }
102
103    fn close(&self, py: Python<'_>) -> bool {
104        py.detach(|| self.state().image = None);
105        true
106    }
107
108    #[pyo3(signature = (remove = false, verbose = true), text_signature = "($self, remove=False, verbose=True)")]
109    fn done(&self, py: Python<'_>, remove: bool, verbose: bool) -> PyResult<bool> {
110        let removed = py.detach(|| {
111            let image = self.state().image.take();
112            if remove && let Some(image) = image {
113                image.remove().map(Some)
114            } else {
115                Ok(None)
116            }
117        })?;
118        if let Some(name) = removed {
119            report_removed(py, &name, verbose)?;
120        }
121        Ok(true)
122    }
123
124    #[pyo3(signature = (done = false, verbose = true), text_signature = "($self, done=False, verbose=True)")]
125    fn remove(&self, py: Python<'_>, done: bool, verbose: bool) -> PyResult<bool> {
126        // Both CASA forms detach the tool before attempting the deletion.
127        let _ = done;
128        let removed = py.detach(|| {
129            self.state()
130                .image
131                .take()
132                .map(|image| image.remove())
133                .transpose()
134        })?;
135        if let Some(name) = removed {
136            report_removed(py, &name, verbose)?;
137            Ok(true)
138        } else {
139            Ok(false)
140        }
141    }
142
143    fn removefile(&self, py: Python<'_>, file: &str) -> PyResult<bool> {
144        let removed = py.detach(|| attachment::remove_file(Path::new(file)));
145        match removed {
146            Ok(true) => Ok(true),
147            Ok(false) => {
148                let warning = if file.is_empty() {
149                    "Empty filename".to_owned()
150                } else {
151                    format!("{file} does not exist or is not a table.")
152                };
153                emit(py, "casars.tools.image", &[LogLine::warning(warning)])?;
154                Ok(false)
155            }
156            Err(error) => {
157                emit(
158                    py,
159                    "casars.tools.image",
160                    &[LogLine::warning(error.to_string())],
161                )?;
162                Ok(false)
163            }
164        }
165    }
166
167    #[pyo3(signature = (name, overwrite=false), text_signature = "($self, name, overwrite=False)")]
168    fn rename(&self, py: Python<'_>, name: &str, overwrite: bool) -> PyResult<bool> {
169        let expanded = resolve_image_path(py, name);
170        py.detach(|| {
171            let mut state = self.state();
172            let Some(current) = state.image.as_ref() else {
173                return Ok(false);
174            };
175            current.lock().check_single_precision("rename")?;
176            let current = state.image.take().expect("an attached image");
177            state.image = Some(current.rename(name, expanded, overwrite, state.history)?);
178            Ok(true)
179        })
180    }
181
182    fn isopen(&self, py: Python<'_>) -> bool {
183        py.detach(|| self.state().image.is_some())
184    }
185
186    fn ispersistent(&self, py: Python<'_>) -> PyResult<bool> {
187        py.detach(|| {
188            self.state()
189                .image
190                .as_ref()
191                .map_or(Ok(false), AttachedImage::is_persistent)
192        })
193    }
194
195    #[pyo3(signature = (strippath = false), text_signature = "($self, strippath=False)")]
196    fn name(&self, py: Python<'_>, strippath: bool) -> PyResult<String> {
197        py.detach(|| {
198            let state = self.state();
199            let Some(attachment) = state.image.as_ref() else {
200                return Ok("none".into());
201            };
202            let image = attachment.lock();
203            image.check_single_precision("name")?;
204            let Some(path) = attachment.name_path(&image) else {
205                let name = match image.image_type_name() {
206                    "ImageExpr" => "",
207                    "ComponentListImage" => "Temporary ComponentListImage",
208                    _ => "Temporary_Image",
209                };
210                return Ok(name.into());
211            };
212            if strippath {
213                Ok(path
214                    .file_name()
215                    .unwrap_or_default()
216                    .to_string_lossy()
217                    .into_owned())
218            } else {
219                let path = std::path::absolute(path).map_err(raised)?;
220                Ok(path.to_string_lossy().into_owned())
221            }
222        })
223    }
224
225    fn shape<'py>(&self, py: Python<'py>) -> Bound<'py, PyArray1<i64>> {
226        py.detach(|| {
227            self.state().image.as_ref().map_or_else(Vec::new, |image| {
228                image
229                    .lock()
230                    .shape()
231                    .iter()
232                    .map(|&size| size as i64)
233                    .collect()
234            })
235        })
236        .into_pyarray(py)
237    }
238
239    fn pixeltype(&self, py: Python<'_>) -> &'static str {
240        py.detach(
241            || match self.state().image.as_ref().map(|image| image.lock().kind()) {
242                Some(ImagePixelType::Float32) => "float",
243                Some(ImagePixelType::Float64) => "double",
244                Some(ImagePixelType::Complex32) => "complex",
245                Some(ImagePixelType::Complex64) => "dcomplex",
246                None => "",
247            },
248        )
249    }
250
251    fn r#type(&self) -> &'static str {
252        "image"
253    }
254
255    #[pyo3(signature = (enable = true), text_signature = "($self, enable=True)")]
256    fn dohistory(&self, py: Python<'_>, enable: bool) -> bool {
257        py.detach(|| self.state().history = enable);
258        true
259    }

262fn report_removed(py: Python<'_>, name: &str, verbose: bool) -> PyResult<()> {
263    let message = format!("deleted table {name}");
264    let log = if verbose {
265        LogLine::info(message)
266    } else {
267        LogLine::debug(message)
268    };
269    emit(py, "casars.tools.image", &[log])
