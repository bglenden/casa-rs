23/// A failure of a native imaging run.
24#[derive(Debug, thiserror::Error)]
25pub(crate) enum ImagingError {
26    /// The compiled problem needs a capability the pass does not provide.
27    #[error("unsupported imaging problem: {reason}")]
28    Unsupported {
29        /// What is unsupported.
30        reason: &'static str,
31    },
32    /// The measurement operator rejected its inputs.
33    #[error(transparent)]
34    Operator(#[from] OperatorError),
35    /// The AW convolution-function catalog could not be opened or generated.
36    #[error(transparent)]
37    KernelSet(#[from] casa_imaging_operator::AwCatalogError),
38    /// A major-cycle or density pass failed.
39    #[error(transparent)]
40    Pass(#[from] PassError),
41    /// A minor cycle failed.
42    #[error(transparent)]
43    Minor(#[from] MinorCycleRunError),
44    /// A normal state could not be assembled.
45    #[error(transparent)]
46    Normal(#[from] SpectralOperatorError),
47    /// The model lifecycle rejected a generation or update.
48    #[error(transparent)]
49    Model(#[from] ModelLifecycleError),
50    /// A major-cycle reconciliation failed.
51    #[error(transparent)]
52    MajorCycle(#[from] MajorCycleError),
53    /// The next cycle's masks could not be planned.
54    #[error(transparent)]
55    Mask(#[from] MaskError),
56    /// The host resources could not be read.
57    #[error(transparent)]
58    Resources(#[from] ResourceError),
59    /// The selected observation could not be opened.
60    #[error("selected observation: {0}")]
61    Observation(#[source] Box<dyn std::error::Error + Send + Sync>),
62    /// The paged cube state could not be created.
63    #[error("cube state: {0}")]
64    CubeState(#[from] std::io::Error),
65}
