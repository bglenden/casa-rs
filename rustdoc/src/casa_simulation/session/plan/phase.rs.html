2//! The deferred plan of DATA: what the session's DATA steps asked for,
3//! applied when `close` writes each row.
4//!
5//! In CASA every step that changes DATA (`predict`, `corrupt`) rewrites the
6//! rows that exist when it is called, which are those of the scans observed
7//! so far; later scans are untouched. Each step here records how many scans
8//! it covers, so for scan `s` DATA is the fold of the steps covering `s`, in
9//! the order they were taken, over the zero DATA `observe` writes. Nothing
10//! is read back from disk, and a sequence of steps costs one pass.
11//!
12//! A `predict` that replaces DATA covers every scan an earlier step covers
13//! (each covers a prefix of the scans), so it clears the plan before it is
14//! recorded. Without antenna-based terms every step left then adds: DATA is
15//! the sum of the predictions covering the scan and the noise of the
16//! corruptions covering it. A corruption with antenna-based terms (`G`,
17//! `D`) multiplies: it corrupts the running DATA by its terms and then adds
18//! its noise (`D ← J(D) + n`), so the corruptions covering a scan apply in
19//! order ([`DataPhase::corruptions_for_scan`]) after its predictions. An
20//! incremental prediction after such a corruption belongs to the next
21//! phase, so it is added after this phase has corrupted DATA.
22//!
23//! Scans are only ever added, so each step covers at least the scans of the
24//! one before it, and the steps covering a scan are the last ones taken.
25//! Every corruption's applications are kept end to end in one list, so a
26//! scan's noise is the end of that list: `close` reads it where it is, and
27//! N interleaved scans and corruptions keep N lists' worth, not N²
28//! (#853 review).

39/// One `setnoise` term: CASA's `ANoise` calibration term (with, for
40/// `tsys-manual`, the `T` that scales it), kept until `reset`.
41#[derive(Debug, Clone)]
42pub(crate) struct NoiseTerm {
43    pub(crate) domain: Arc<crate::corrupt::noise_domain::NoiseDomain>,
44    pub(crate) snapshot: Option<Arc<crate::corrupt::tsys_snapshot::TsysSnapshot>>,
45    /// The seed in force at `setnoise`.
46    pub(crate) seed: i32,
47    /// How large its draws are.
48    pub(crate) level: NoiseLevel,
49    /// How many `corrupt` calls have applied the term.
50    pub(crate) applications: u32,
51}
52
53impl NoiseTerm {
54    pub(crate) fn application_level(&self, scans: usize) -> NoiseLevel {
55        if scans > self.domain.scans
56            && let Some(snapshot) = &self.snapshot
57        {
58            let scale = NoiseLevel::Snapshot(Arc::clone(snapshot));
59            return match &self.level {
60                NoiseLevel::ScaledFixed { field, .. } => NoiseLevel::ScaledFixed {
61                    field: Arc::clone(field),
62                    scale: Arc::new(scale),
63                },
64                _ => scale,
65            };
66        }
67        self.level.clone()
68    }
69}

71/// A step that changes DATA, covering the scans observed before it.
72#[derive(Debug, Clone)]
73pub(crate) enum DataStep {
74    /// `predict`: add the model's prediction. A replacing `predict` is the
75    /// first step of the plan, so adding to the zero DATA replaces it.
76    Predict {
77        /// The prepared model.
78        prediction: Arc<Prediction>,
79        /// The step covers scans `0..scans`.
80        scans: usize,
81    },
82    /// `predict(complist=…)`: add the component list's visibilities. A
83    /// `predict` of an image and a list records both steps, the list's
84    /// first, as CASA predicts its components first.
85    PredictComponents {
86        /// The prepared list.
87        components: Arc<ComponentPrediction>,
88        /// The step covers scans `0..scans`.
89        scans: usize,
90    },
91    /// `predict` after `setoptions(ftmachine='sd')`: add each pointing's
92    /// total power to its autocorrelations.
93    PredictSingleDish {
94        /// The sampled model.
95        prediction: Arc<SingleDishPrediction>,
96        /// The step covers scans `0..scans`.
97        scans: usize,
98    },
99    /// `corrupt`: corrupt DATA by each antenna-based term, then add each
100    /// application's noise (`VisEquation::collapseForSim`).
101    Corrupt {
102        /// Its applications, one per noise term in the order the terms
103        /// were set, as their range of the plan's noise.
104        applications: Range<usize>,
105        /// Its antenna-based terms in the order they corrupt DATA, as
106        /// their range of the plan's terms.
107        jones: Range<usize>,
108        /// The step covers scans `0..scans`.
109        scans: usize,
110    },
111}
112
113impl DataStep {
114    /// The step covers scans `0..scans()`.
115    pub(crate) fn scans(&self) -> usize {
116        match self {
117            Self::Predict { scans, .. }
118            | Self::PredictComponents { scans, .. }
119            | Self::PredictSingleDish { scans, .. }
120            | Self::Corrupt { scans, .. } => *scans,
121        }
122    }
123}

125/// The session's DATA steps, in the order they were taken.
126#[derive(Debug, Clone, Default)]
127pub(crate) struct DataPhase {
128    /// Immutable row selection in force when this phase was recorded.
129    pub(crate) selection: DataSelection,
130    /// Replace selected DATA before adding this phase.
131    pub(crate) replace: bool,
132    steps: Vec<DataStep>,
133    /// Every corruption's applications, in the order they were taken.
134    noise: Vec<NoiseApplication>,
135    /// Every corruption's antenna-based terms, in the order they were taken.
136    jones: Vec<Arc<HeldTerm>>,
137}
138
139impl DataPhase {
140    pub(crate) fn new(selection: DataSelection, replace: bool) -> Self {
141        Self {
142            selection,
143            replace,
144            ..Self::default()
145        }
146    }
147
148    /// Record a `predict` of `prediction` covering scans `0..scans`, after
149    /// the steps already taken.
150    pub(crate) fn predict(&mut self, prediction: Arc<Prediction>, scans: usize) {
151        self.push(DataStep::Predict { prediction, scans });
152    }
153
154    /// Record a `predict` of the component list `components` covering scans
155    /// `0..scans`, after the steps already taken.
156    pub(crate) fn predict_components(
157        &mut self,
158        components: Arc<ComponentPrediction>,
159        scans: usize,
160    ) {
161        self.push(DataStep::PredictComponents { components, scans });
162    }
163
164    /// Record a total-power `predict` of `prediction` covering scans
165    /// `0..scans`, after the steps already taken.
166    pub(crate) fn predict_single_dish(
167        &mut self,
168        prediction: Arc<SingleDishPrediction>,
169        scans: usize,
170    ) {
171        self.push(DataStep::PredictSingleDish { prediction, scans });
172    }
173
174    /// Record a `corrupt` of scans `0..scans` by the antenna-based terms
175    /// `jones`, in the order they corrupt DATA, and `applications`, after
176    /// the steps already taken.
177    pub(crate) fn corrupt(
178        &mut self,
179        applications: impl IntoIterator<Item = NoiseApplication>,
180        jones: impl IntoIterator<Item = Arc<HeldTerm>>,
181        scans: usize,
182    ) {
183        let start = self.noise.len();
184        self.noise.extend(applications);
185        let first_term = self.jones.len();
186        self.jones.extend(jones);
187        self.push(DataStep::Corrupt {
188            applications: start..self.noise.len(),
189            jones: first_term..self.jones.len(),
190            scans,
191        });
192    }
193
194    fn push(&mut self, step: DataStep) {
195        debug_assert!(
196            self.steps
197                .last()
198                .is_none_or(|last| last.scans() <= step.scans()),
199            "a step covers at least the scans of the one before it"
200        );
201        self.steps.push(step);
202    }
203
204    /// Forget every step: a replacing `predict` overwrites what they did,
205    /// and the DATA on disk.
206    #[cfg(test)]
207    pub(crate) fn clear(&mut self) {
208        self.steps.clear();
209        self.noise.clear();
210        self.jones.clear();
211    }
212
213    /// Whether a step corrupts by an antenna-based term: a later
214    /// incremental `predict` would add to gained DATA, which the plan does
215    /// not yet fold.
216    pub(crate) fn corrupts_by_jones(&self) -> bool {
217        !self.jones.is_empty()
218    }
219
220    pub(crate) fn needs_corruption_steps(&self, scan: usize, window: usize) -> bool {
221        self.corruptions_for_scan(scan).any(|(noise, jones)| {
222            !jones.is_empty() || noise.iter().any(|term| !term.covers(window))
223        })
224    }
225
226    /// The corruptions covering scan `scan`, in the order they were taken:
227    /// each one's antenna-based terms, in the order they corrupt DATA, and
228    /// its noise, added after them.
229    pub(crate) fn corruptions_for_scan(
230        &self,
231        scan: usize,
232    ) -> impl Iterator<Item = (&[NoiseApplication], &[Arc<HeldTerm>])> {
233        self.steps_for_scan(scan)
234            .iter()
235            .filter_map(|step| match step {
236                DataStep::Corrupt {
237                    applications,
238                    jones,
239                    ..
240                } => Some((
241                    &self.noise[applications.clone()],
242                    &self.jones[jones.clone()],
243                )),
244                DataStep::Predict { .. }
245                | DataStep::PredictComponents { .. }
246                | DataStep::PredictSingleDish { .. } => None,
247            })
248    }
249
250    /// CASA gates an entire corruption step on every Jones term's coverage,
251    /// including the noise that follows its Jones operators.
252    pub(crate) fn corruptions_for_window(
253        &self,
254        scan: usize,
255        window: usize,
256    ) -> impl Iterator<Item = (&[NoiseApplication], &[Arc<HeldTerm>])> {
257        self.corruptions_for_scan(scan)
258            .filter(move |(noise, jones)| {
259                noise.iter().all(|term| term.covers(window))
260                    && jones.iter().all(|term| term.covers(window))
261            })
262    }
263
264    /// Whether this phase changes rows of the given scan/window. An
265    /// uncovered Jones term gates only its corruption step, not predictions
266    /// or another corruption step in the same phase.
267    pub(crate) fn touches_window(&self, scan: usize, window: usize) -> bool {
268        self.predicts_scan(scan) || self.corruptions_for_window(scan, window).next().is_some()
269    }
270
271    pub(crate) fn has_uncovered_corruption(&self, scan: usize, window: usize) -> bool {
272        self.corruptions_for_scan(scan).any(|(noise, jones)| {
273            noise.iter().any(|term| !term.covers(window))
274                || jones.iter().any(|term| !term.covers(window))
275        })
276    }
277
278    /// Whether no step was taken (or every one was replaced by nothing).
279    pub(crate) fn is_empty(&self) -> bool {
280        self.steps.is_empty()
281    }
282
283    /// Whether any step predicts: CASA's `predict` sets MODEL_DATA to 1.
284    pub(crate) fn predicts(&self) -> bool {
285        self.steps.iter().any(|step| {
286            matches!(
287                step,
288                DataStep::Predict { .. }
289                    | DataStep::PredictComponents { .. }
290                    | DataStep::PredictSingleDish { .. }
291            )
292        })
293    }
294
295    pub(crate) fn predicts_scan(&self, scan: usize) -> bool {
296        self.steps_for_scan(scan).iter().any(|step| {
297            matches!(
298                step,
299                DataStep::Predict { .. }
300                    | DataStep::PredictComponents { .. }
301                    | DataStep::PredictSingleDish { .. }
302            )
303        })
304    }
305
306    /// The steps covering scan `scan`, in the order they were taken: the
307    /// last ones.
308    pub(crate) fn steps_for_scan(&self, scan: usize) -> &[DataStep] {
309        let first = self.steps.partition_point(|step| step.scans() <= scan);
310        &self.steps[first..]
311    }
312
313    /// The noise to add to scan `scan`'s rows, in the order it was asked
314    /// for: the applications of its corruptions, all of which add, which
315    /// are the last ones taken.
316    pub(crate) fn noise_for_scan(&self, scan: usize) -> &[NoiseApplication] {
317        let start = self
318            .steps_for_scan(scan)
319            .iter()
320            .find_map(|step| match step {
321                DataStep::Corrupt { applications, .. } => Some(applications.start),
322                DataStep::Predict { .. }
323                | DataStep::PredictComponents { .. }
324                | DataStep::PredictSingleDish { .. } => None,
325            })
326            .unwrap_or(self.noise.len());
327        &self.noise[start..]
328    }
329
330    /// The predict steps covering scan `scan`, as indices of
331    /// [`Self::predictions`], in the order they were taken: their
332    /// predictions add. They are the last ones.
333    pub(crate) fn predictions_for_scan(&self, scan: usize) -> Range<usize> {
334        let all = self.predictions().count();
335        let covering = self
336            .steps_for_scan(scan)
337            .iter()
338            .filter(|step| matches!(step, DataStep::Predict { .. }))
339            .count();
340        all - covering..all
341    }
342
343    /// Every predict step's prepared model and the scans it covers, in the
344    /// order they were taken.
345    pub(crate) fn predictions(&self) -> impl Iterator<Item = (&Prediction, usize)> {
346        self.steps.iter().filter_map(|step| match step {
347            DataStep::Predict { prediction, scans } => Some((prediction.as_ref(), *scans)),
348            DataStep::PredictComponents { .. }
349            | DataStep::PredictSingleDish { .. }
350            | DataStep::Corrupt { .. } => None,
351        })
352    }
353
354    /// The component lists the steps covering scan `scan` predict, in the
355    /// order they were taken: their visibilities add.
356    pub(crate) fn components_for_scan(
357        &self,
358        scan: usize,
359    ) -> impl Iterator<Item = &ComponentPrediction> {
360        self.steps_for_scan(scan)
361            .iter()
362            .filter_map(|step| match step {
363                DataStep::PredictComponents { components, .. } => Some(components.as_ref()),
364                DataStep::Predict { .. }
365                | DataStep::PredictSingleDish { .. }
366                | DataStep::Corrupt { .. } => None,
367            })
368    }
369
370    /// The total-power predictions covering scan `scan`, in the order they
371    /// were taken: they add.
372    pub(crate) fn single_dish_for_scan(
373        &self,
374        scan: usize,
375    ) -> impl Iterator<Item = &SingleDishPrediction> {
376        self.steps_for_scan(scan)
377            .iter()
378            .filter_map(|step| match step {
379                DataStep::PredictSingleDish { prediction, .. } => Some(prediction.as_ref()),
380                DataStep::Predict { .. }
381                | DataStep::PredictComponents { .. }
382                | DataStep::Corrupt { .. } => None,
383            })
384    }
385
386    /// Whether any step predicts a component list.
387    pub(crate) fn predicts_components(&self) -> bool {
388        self.steps
389            .iter()
390            .any(|step| matches!(step, DataStep::PredictComponents { .. }))
391    }
392
393    /// Whether any step predicts total power.
394    pub(crate) fn predicts_single_dish(&self) -> bool {
395        self.steps
396            .iter()
397            .any(|step| matches!(step, DataStep::PredictSingleDish { .. }))
398    }

401#[cfg(test)]
402mod tests {
403    use super::*;
404
405    fn noise(seed: i32, ordinal: u32) -> NoiseApplication {
406        NoiseApplication {
407            domain: crate::corrupt::noise_domain::NoiseDomain::unrestricted(),
408            seed,
409            level: NoiseLevel::Simple(0.5),
410            ordinal,
411        }
412    }
413
414    #[test]
415    fn a_step_covers_only_the_scans_observed_before_it() {
416        let mut plan = DataPhase::default();
417        plan.corrupt([noise(1, 0)], [], 1);
418        plan.corrupt([noise(1, 1), noise(2, 0)], [], 3);
419        assert_eq!(
420            plan.noise_for_scan(0),
421            [noise(1, 0), noise(1, 1), noise(2, 0)]
422        );
423        assert_eq!(plan.noise_for_scan(2), [noise(1, 1), noise(2, 0)]);
424        assert_eq!(plan.noise_for_scan(3), []);
425        assert_eq!(plan.steps_for_scan(1).len(), 1);
426        assert!(plan.predictions_for_scan(0).is_empty());
427        plan.clear();
428        assert_eq!(plan.noise_for_scan(0), []);
429    }
430
431    #[test]
432    fn interleaved_corruptions_share_their_applications() {
433        let mut plan = DataPhase::default();
434        for scans in 1..=4 {
435            plan.corrupt([noise(1, scans as u32 - 1)], [], scans);
436        }
437        for scan in 0..4 {
438            let expected = (scan as u32..4).map(|ordinal| noise(1, ordinal));
439            assert!(plan.noise_for_scan(scan).iter().cloned().eq(expected));
440        }
441        // Each scan's noise is the end of the one list.
442        let all = plan.noise_for_scan(0).as_ptr_range();
443        assert_eq!(plan.noise_for_scan(3).as_ptr_range().end, all.end);
444        assert_eq!(plan.noise_for_scan(4), []);
445    }
