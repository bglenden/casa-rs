23const L_BAND_BEAM_SIDE: usize = 512;
24const L_BAND_BEAM_CELL_RAD: f64 = 0.024 / 256.0;

26/// The aperture grid and the cropped sky grid used by
27/// `AWConvFunc::makeAConvFunc` for an AWPLPG image.
28pub struct Awp2Sampling {
29    image_side: usize,
30    cell_rad: f64,
31    beam_cell_rad: f64,
32    aperture_side: usize,
33    sky_side: usize,
34    oversampling: u16,
35}
36
37impl Awp2Sampling {
38    /// Derive AWPLPG sampling from a square, unpadded image with square
39    /// cells. `single_field` selects the aperture extent and cropping in
40    /// `AWConvFunc::makeAConvFunc`. This path implements Lanczos resampling.
41    pub fn new(geometry: &GridGeometry, single_field: bool) -> Result<Self, OperatorError> {
42        let extent = geometry.image();
43        let [side, ny] = extent.shape;
44        let [dx, dy] = extent.increment_rad.map(f64::abs);
45        let field_of_view = side as f64 * dx;
46        let beam_field_of_view = L_BAND_BEAM_SIDE as f64 * L_BAND_BEAM_CELL_RAD;
47        let oversampling = if side < 100 {
48            100
49        } else {
50            10000_usize.div_ceil(side).max(4)
51        };
52        if side != ny || (dx - dy).abs() > dx * f64::EPSILON || !oversampling.is_multiple_of(2) {
53            return Err(invalid(
54                "AWP2 requires a square image with square cells and even oversampling (#1071)",
55            ));
56        }
57        let (aperture_side, sky_side, beam_cell_rad) = if field_of_view > 0.125 * beam_field_of_view
58        {
59            let side = if single_field || field_of_view < beam_field_of_view {
60                2 * L_BAND_BEAM_SIDE
61            } else {
62                L_BAND_BEAM_SIDE
63            };
64            (side, side, L_BAND_BEAM_CELL_RAD)
65        } else {
66            let aperture_side =
67                ((L_BAND_BEAM_SIDE as f64 / dx * L_BAND_BEAM_CELL_RAD / 16.0).ceil() as usize)
68                    .checked_mul(8)
69                    .filter(|side| {
70                        side.checked_mul(*side)
71                            .is_some_and(|n| n <= isize::MAX as usize / 64)
72                    })
73                    .ok_or_else(|| invalid("AWP2 aperture grid is too large"))?;
74            let sky_side = ((field_of_view / beam_field_of_view * aperture_side as f64 / 2.0).ceil()
75                as usize)
76                * 2;
77            (aperture_side, sky_side, dx)
78        };
79        if sky_side < 12 || sky_side > aperture_side {
80            return Err(invalid("AWP2 sky crop does not fit its aperture grid"));
81        }
82        let sampling = Self {
83            image_side: side,
84            cell_rad: dx,
85            beam_cell_rad,
86            aperture_side,
87            sky_side,
88            oversampling: oversampling as u16,
89        };
90        if side >= 200 && sampling.factor() / oversampling as f32 >= 1.0 {
91            return Err(invalid("AWP2 FFT resampling remains unsupported (#1071)"));
92        }
93        // The wider-than-beam A term uses a different support cutoff,
94        // including on small images that still select Lanczos.
95        if field_of_view > aperture_side as f64 * beam_cell_rad {
96            return Err(invalid(
97                "AWP2 wider-than-beam support remains unsupported (#1071)",
98            ));
99        }
100        Ok(sampling)
101    }
102
103    fn factor(&self) -> f32 {
104        self.image_side as f32 * f32::from(self.oversampling)
105            / self.sky_side as f32
106            / (self.beam_cell_rad / self.cell_rad) as f32
107    }
108
109    /// Temporary Jones and sky FFT planes and their planning workspaces.
110    #[must_use]
111    pub fn screens_bytes(&self, count: usize) -> u64 {
112        let aperture_cells = self.aperture_side * self.aperture_side;
113        let sky_cells = self.sky_side * self.sky_side;
114        ((4 * aperture_cells + 2 * sky_cells) * size_of::<Complex32>()) as u64
115            + casa_fft::planning_bytes::<f32>(aperture_cells)
116            + casa_fft::planning_bytes::<f32>(sky_cells)
117            + (count * (size_of::<ApertureCell>() + 2 * sky_cells * size_of::<Complex32>())) as u64
118    }

120    /// Calculate the A terms and their one-plane AW normalization with the
121    /// supplied dish surface. `legacy_vla` selects the original VLA feed.
122    /// Scratch is charged before calling this method.
123    pub fn screens(
124        self,
125        frequencies_hz: &[f64],
126        surface: EvlaDishSurface,
127        legacy_vla: bool,
128    ) -> Result<Awp2Screens, OperatorError> {
129        if frequencies_hz.is_empty()
130            || frequencies_hz.len() > usize::from(u16::MAX)
131            || frequencies_hz.windows(2).any(|f| f[0] >= f[1])
132            || frequencies_hz.iter().any(|f| !(0.9e9..=2e9).contains(f))
133        {
134            return Err(invalid(
135                "AWP2 requires ordered VLA L-band beam frequencies (#1071)",
136            ));
137        }
138        let model = EvlaApertureModel::new(surface);
139        let mut jones = vec![Complex32::default(); 4 * self.aperture_side * self.aperture_side];
140        let mut aperture_fft = PlaneFft::new([self.aperture_side; 2], false)?;
141        let mut sky_fft = PlaneFft::new([self.sky_side; 2], false)?;
142        let mut cells = Vec::with_capacity(frequencies_hz.len());
143        // CASA crops every A plane to the support of the highest frequency.
144        let mut a_support = None;
145        for &frequency in frequencies_hz.iter().rev() {
146            let beam =
147                self.sky_beam(&model, frequency, legacy_vla, &mut aperture_fft, &mut jones)?;
148            let cell = ApertureCell::new(beam, self.sky_side, &mut sky_fft, a_support)?;
149            a_support.get_or_insert(cell.a_side / 2);
150            cells.push(cell);
151        }
152        cells.reverse();
153        Ok(Awp2Screens {
154            sampling: self,
155            cells,
156        })
157    }
158
159    fn sky_beam(
160        &self,
161        model: &EvlaApertureModel,
162        frequency: f64,
163        legacy_vla: bool,
164        aperture_fft: &mut PlaneFft<f32>,
165        jones: &mut [Complex32],
166    ) -> Result<Vec<Complex32>, OperatorError> {
167        let wavelength = SPEED_OF_LIGHT_M_PER_S / f64::from(frequency as f32);
168        let cell_m = wavelength / (self.aperture_side as f64 * self.beam_cell_rad);
169        let grid = EvlaApertureGrid::new(self.aperture_side, cell_m, 3, frequency, 0.0)
170            .map_err(|_| invalid("invalid AWP2 aperture sampling"))?;
171        if legacy_vla {
172            model.fill_vla_l_aperture(grid, jones)
173        } else {
174            model.fill_aperture(grid, jones)
175        }
176        .map_err(|_| invalid("AWP2 could not evaluate its aperture"))?;
177        let norm = sky_jones_normalization(self.aperture_side, aperture_fft, jones)
178            .map_err(|_| invalid("AWP2 could not normalize its aperture"))?;
179        let plane = self.aperture_side * self.aperture_side;
180        let mut beam = Vec::with_capacity(self.sky_side * self.sky_side);
181        let first = (self.aperture_side - self.sky_side) / 2;
182        for y in first..first + self.sky_side {
183            for x in first..first + self.sky_side {
184                let index = y * self.aperture_side + x;
185                // applyAvgSkyJones uses real(RL) for all four correlations.
186                let rr = jones[index] / norm;
187                let ll = jones[3 * plane + index] / norm;
188                beam.push(Complex32::new((ll * rr.conj()).re, 0.0));
189            }
190        }
191        Ok(beam)
192    }
193}

195struct ApertureCell {
196    imaging: Vec<Complex32>,
197    weight: Vec<Complex32>,
198    a_side: usize,
199    aw_support: usize,
200}
201
202impl ApertureCell {
203    fn new(
204        mut imaging: Vec<Complex32>,
205        side: usize,
206        fft: &mut PlaneFft<f32>,
207        crop_support: Option<usize>,
208    ) -> Result<Self, OperatorError> {
209        let mut weight: Vec<_> = imaging.iter().map(|value| value * value.conj()).collect();
210        fft.transform(&mut imaging, false)?;
211        fft.transform(&mut weight, false)?;
212        let support = a_support(&weight, side)?;
213        for plane in [&mut imaging, &mut weight] {
214            let sum = central_sum(plane, side, support, true);
215            if !sum.is_finite() || sum <= 0.0 {
216                return Err(invalid("AWP2 aperture normalization is not positive"));
217            }
218            *plane = crop(plane, side, crop_support.unwrap_or(support), sum.recip());
219        }
220        let a_side = 2 * crop_support.unwrap_or(support);
221        // The transform of the W=0 screen is a central impulse. Its
222        // convolution leaves the A plane up to scale, which cancels in
223        // supportResizeAWConv's second normalization. The weight stays A².
224        let aw_support = diagonal_support(&imaging, a_side, 0.01)
225            .map_or(5, |radius| radius.max(5))
226            .max(support);
227        let sum = central_sum(&imaging, a_side, aw_support.min(a_side / 2), false);
228        if sum.is_finite() && sum > 0.0 {
229            imaging.iter_mut().for_each(|value| *value *= sum.recip());
230        }
231        Ok(Self {
232            imaging,
233            weight,
234            a_side,
235            aw_support,
236        })
237    }
238}
239
240fn a_support(weight: &[Complex32], side: usize) -> Result<usize, OperatorError> {
241    let (minimum, maximum) = weight
242        .iter()
243        .map(|v| v.norm())
244        .fold((f32::INFINITY, 0.0_f32), |(minimum, maximum), value| {
245            (minimum.min(value), maximum.max(value))
246        });
247    if maximum - minimum <= 0.005 * maximum {
248        return Err(invalid("AWP2 found no aperture support"));
249    }
250    let fallback = if side > 10 { 5 } else { side / 2 - 4 };
251    Ok((diagonal_support(weight, side, 0.005)
252        .filter(|radius| *radius >= 5)
253        .unwrap_or(fallback)
254        + 1)
255    .min(side / 2 - 1))
256}
257
258fn diagonal_support(plane: &[Complex32], side: usize, cutoff: f32) -> Option<usize> {
259    let maximum = plane.iter().map(|v| v.norm()).fold(0.0, f32::max);
260    let centre = side / 2;
261    (1..=centre - 2).rev().find_map(|offset| {
262        let index = (centre - offset) * (side + 1);
263        (plane[index].norm() > cutoff * maximum)
264            .then_some((std::f64::consts::SQRT_2 * offset as f64) as usize)
265    })
266}
267
268fn central_sum(plane: &[Complex32], side: usize, radius: usize, inclusive: bool) -> f32 {
269    let centre = side / 2;
270    let first = centre - radius;
271    let end = centre + radius + usize::from(inclusive);
272    (first..end)
273        .flat_map(|y| &plane[y * side + first..y * side + end])
274        .map(|v| v.re)
275        .sum()
276}
277
278fn crop(plane: &[Complex32], side: usize, radius: usize, factor: f32) -> Vec<Complex32> {
279    let first = side / 2 - radius;
280    let end = first + 2 * radius;
281    (first..end)
282        .flat_map(|y| &plane[y * side + first..y * side + end])
283        .map(|value| *value * factor)
284        .collect()
285}

287/// A and W=0 convolution planes before the oversampled resident cells are
288/// allocated. The cells use one Mueller plane and separate sensitivity taps.
289pub struct Awp2Screens {
290    sampling: Awp2Sampling,
291    cells: Vec<ApertureCell>,
292}
293
294impl Awp2Screens {
295    fn half_support(&self) -> u16 {
296        // appendConvFuncs uses the highest-frequency support for each row.
297        self.cells
298            .last()
299            .expect("nonempty beam frequencies")
300            .aw_support as u16
301    }
302
303    fn fine_side(&self) -> usize {
304        2 * (usize::from(self.half_support()) + 1) * usize::from(self.sampling.oversampling)
305    }
306
307    /// Upper bound including the resident cells and one oversampled plane
308    /// and interpolation workspace used while constructing them.
309    #[must_use]
310    pub fn kernel_bytes(&self, geometry: &GridGeometry) -> u64 {
311        let fine_side = self.fine_side();
312        (self.cells.len() * (size_of::<f64>() + size_of::<Awp2Cell>())) as u64
313            + 2 * self.cells.len() as u64
314                * dense_cell_bytes(1, self.sampling.oversampling, [self.half_support(); 2])
315            + (fine_side * fine_side * size_of::<Complex32>()
316                + fine_side * size_of::<Option<LanczosTaps>>()
317                + 8000 * size_of::<f32>()) as u64
318            + ImageCorrection::bytes(geometry.grid_shape())
319            + MuellerRouting::MAXIMUM_BYTES
320    }
321
322    /// Resample each A/W and sensitivity plane with AWConvFuncHolder's
323    /// Lanczos rule and pack them in the shared gridding layout.
324    pub fn finish(
325        self,
326        geometry: &GridGeometry,
327        polarization: &PolarizationRouting,
328        frequencies_hz: Box<[f64]>,
329    ) -> Awp2 {
330        assert_eq!(self.cells.len(), frequencies_hz.len());
331        let half_support = self.half_support();
332        let fine_side = self.fine_side();
333        let oversampling = self.sampling.oversampling;
334        let factor = self.sampling.factor();
335        let mut plane = vec![Complex32::default(); fine_side * fine_side];
336        let sinc_table: Vec<_> = (-4000..4000)
337            .map(|i| sinc((i as f64 / 1000.0) as f32))
338            .collect();
339        let a_side = self.cells[0].a_side;
340        let resampled_side = resampled_side(a_side, factor);
341        let axis = lanczos_axis(a_side, resampled_side, fine_side, &sinc_table);
342        let scale = (a_side * a_side * usize::from(oversampling).pow(2)) as f32
343            / (resampled_side * resampled_side) as f32;
344        let mut pack = |source: &[Complex32]| {
345            resample(source, a_side, &axis, scale, &mut plane);
346            dense_cell(
347                &[OversampledPlane {
348                    values: &plane,
349                    side: fine_side,
350                    centre: [fine_side / 2; 2],
351                    sampling: oversampling,
352                }],
353                [half_support; 2],
354            )
355        };
356        let cells = self
357            .cells
358            .into_iter()
359            .map(|cell| Awp2Cell {
360                imaging: pack(&cell.imaging),
361                weight: pack(&cell.weight),
362            })
363            .collect();
364        Awp2 {
365            frequencies_hz,
366            cells,
367            half_support,
368            mueller: MuellerRouting::scalar(polarization.pol_map(), polarization.grid_pols()),
369            correction: ImageCorrection::mosaic_sinc(geometry.grid_shape(), oversampling),
370        }
371    }
372}

374// MathUtils::resample receives the holder's Float factor as a Double.
375fn resampled_side(input_side: usize, factor: f32) -> usize {
376    (input_side as f64 * f64::from(factor) / 8.0).ceil() as usize * 8
377}
378
379fn sinc(value: f32) -> f32 {
380    if value == 0.0 {
381        1.0
382    } else {
383        let x = std::f64::consts::PI * f64::from(value);
384        (x.sin() / x) as f32
385    }
386}

388struct LanczosTaps {
389    first: usize,
390    weights: [[f32; 2]; 6],
391}
392
393fn lanczos_axis(
394    side: usize,
395    resampled: usize,
396    fine: usize,
397    table: &[f32],
398) -> Vec<Option<LanczosTaps>> {
399    let sinc_lookup = |value: f64| table[(value as f32 * 1000.0 + 4000.0) as usize];
400    (0..fine)
401        .map(|i| {
402            let i = i as isize + (resampled as isize - fine as isize) / 2;
403            let position = i as f64 / resampled as f64 * side as f64;
404            let pixel = position.floor() as isize;
405            if pixel < 3 || pixel >= side as isize - 3 {
406                return None;
407            }
408            let first = pixel as usize - 2;
409            let weights = std::array::from_fn(|tap| {
410                let distance = position - (first + tap) as f64;
411                [sinc_lookup(distance), sinc_lookup(distance / 3.0)]
412            });
413            Some(LanczosTaps { first, weights })
414        })
415        .collect()
416}
417
418fn resample(
419    source: &[Complex32],
420    side: usize,
421    axis: &[Option<LanczosTaps>],
422    scale: f32,
423    output: &mut [Complex32],
424) {
425    for (row, y) in output.chunks_exact_mut(axis.len()).zip(axis) {
426        for (value, x) in row.iter_mut().zip(axis) {
427            *value = match (x, y) {
428                (Some(x), Some(y)) => interpolate(source, side, x, y) * scale,
429                _ => Complex32::default(),
430            };
431        }
432    }
433}
434
435fn interpolate(source: &[Complex32], side: usize, x: &LanczosTaps, y: &LanczosTaps) -> Complex32 {
436    let mut sum = Complex32::default();
437    for (i, [wx, lx]) in x.weights.into_iter().enumerate() {
438        for (j, [wy, ly]) in y.weights.into_iter().enumerate() {
439            let value = source[(y.first + j) * side + x.first + i];
440            let weight = f64::from(wx) * f64::from(lx) * f64::from(wy) * f64::from(ly);
441            sum.re += (f64::from(value.re) * weight) as f32;
442            sum.im += (f64::from(value.im) * weight) as f32;
443        }
444    }
445    sum
446}

466        (std::mem::size_of_val(self.frequencies_hz.as_ref())
467            + std::mem::size_of_val(self.cells.as_ref())
468            + self

474            + self.correction.resident_bytes()

634    #[test]
635    fn awp2_interpolation_uses_the_casa_float_sinc_table() {
636        let table: Vec<_> = (-4000..4000)
637            .map(|i| sinc((i as f64 / 1000.0) as f32))
638            .collect();
639        // A single nonzero input also isolates the sinc-table multiplication
640        // from roundoff in a general convolution.
641        let mut input = [Complex32::default(); 12 * 12];
642        input[6 * 12 + 6] = Complex32::new(0.9771961, 0.0);
643        let taps = lanczos_axis(12, 2312, 1400, &table);
644        let value = interpolate(
645            &input,
646            12,
647            taps[757].as_ref().unwrap(),
648            taps[679].as_ref().unwrap(),
649        );
650        // sinc(0.295) sinc(0.098) sinc(-0.109) sinc(-0.037),
651        // evaluated at the four entries selected by MathUtils::sinc.
652        let expected = 0.9771961_f64
653            * [0.295_f64, 0.098, -0.109, -0.037]
654                .into_iter()
655                .map(|v| {
656                    let a = std::f64::consts::PI * v;
657                    a.sin() / a
658                })
659                .product::<f64>();
660        assert!(
661            (f64::from(value.re) - expected).abs() < 2.0e-7,
662            "{value} vs {expected}"
663        );
664        assert_eq!(value.im, 0.0);
665    }
