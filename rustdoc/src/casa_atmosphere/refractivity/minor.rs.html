4//! The minor gases' line refractivities: SO2, HCl, HCN, NO2, N2O and CO
5//! (ATM's opacity sources 8, 24, 25, 7, 6 and 5).
6//!
7//! Each is a sum of air-broadened lines weighted by intensity, Boltzmann
8//! factor and line frequency, scaled by (ν/π)·((h/k)/T)·(8π³/3hc)μ²/Q.
9
10use super::tables::{co, hcl, hcn, n2o, no2, so2};
11use super::{
12    AirLines, Exponent, H_OVER_K, PI, PICUBE8DIV3HCESU, ZERO, by_pressure, frequency_bin, window,
13};
14use num_complex::Complex64;
15
16/// The factor ATM applies to a minor gas's line sum, in its order:
17/// ν/π, then (h/k)/T, then (8π³/3hc)μ²/Q, then 1e-4 (cm² to m²).
18fn scaled(sum: Complex64, nu: f64, tt: f64, mu: f64, q: f64) -> Complex64 {
19    sum * (nu / PI) * (H_OVER_K / tt) * (PICUBE8DIV3HCESU * mu.powf(2.0) / q) * 1e-4
20}
21
22/// SO2 (source 8). ATM has no SO2 lines above 499.99 GHz.
23pub(super) fn so2(tt: f64, pp: f64, nu: f64) -> Complex64 {
24    if nu > 499.99 {
25        return ZERO;
26    }
27    let lines = AirLines {
28        frequency: &so2::FRE,
29        intensity: &so2::FLIN,
30        energy: &so2::EL,
31        broadening: &so2::BRDSO2AIR,
32        exponent: Exponent::Fixed(0.75),
33    };
34    let Some(range) = window(frequency_bin(nu), 2, (&so2::INI1, &so2::IFIN1)) else {
35        return ZERO;
36    };
37    let q = 1.1346738633 * tt * tt.sqrt();
38    scaled(lines.sum(nu, tt, pp, 64.0644, range), nu, tt, 1.62, q)
39}

41/// HCl (source 24): three hyperfine lines near 625.9 GHz, summed only
42/// between 624 and 627.8 GHz.
43pub(super) fn hcl(tt: f64, pp: f64, nu: f64) -> Complex64 {
44    if !(nu > 624.0 && nu < 627.8) {
45        return ZERO;
46    }
47    let lines = AirLines {
48        frequency: &hcl::FRE,
49        intensity: &hcl::FLIN,
50        energy: &hcl::EL,
51        broadening: &hcl::BRDHCLAIR,
52        exponent: Exponent::PerLine(&hcl::TEXPHCL),
53    };
54    let q = 0.0668793372 * tt;
55    scaled(lines.sum(nu, tt, pp, 36.0, 0..=2), nu, tt, 0.816, q)
56}

58/// HCN (source 25): all 18 lines at every frequency.
59pub(super) fn hcn(tt: f64, pp: f64, nu: f64) -> Complex64 {
60    let lines = AirLines {
61        frequency: &hcn::FRE,
62        intensity: &hcn::FLIN,
63        energy: &hcn::EL,
64        broadening: &hcn::BRDHCNAIR,
65        exponent: Exponent::PerLine(&hcn::TEXPHCN),
66    };
67    let q = 0.470182585 * tt;
68    scaled(lines.sum(nu, tt, pp, 27.0, 0..=17), nu, tt, 2.98, q)
69}

71/// NO2 (source 7). ATM has no NO2 lines above 999.9 GHz.
72pub(super) fn no2(tt: f64, pp: f64, nu: f64) -> Complex64 {
73    let vp = frequency_bin(nu);
74    if nu > 999.9 || (nu > 20.0 && vp == 0) {
75        return ZERO;
76    }
77    let lines = AirLines {
78        frequency: &no2::FRE,
79        intensity: &no2::FLIN,
80        energy: &no2::EL,
81        broadening: &no2::BRDNO2AIR,
82        exponent: Exponent::PerLine(&no2::TEXPNO2),
83    };
84    let Some(range) = window(vp, 2, (&no2::INI1, &no2::IFIN1)) else {
85        return ZERO;
86    };
87    let q = 0.430965924 * tt * tt.sqrt();
88    scaled(lines.sum(nu, tt, pp, 46.0, range), nu, tt, 0.3161, q)
89}

91/// N2O (source 6). ATM has no N2O lines above 999.9 GHz.
92pub(super) fn n2o(tt: f64, pp: f64, nu: f64) -> Complex64 {
93    if nu > 999.9 {
94        return ZERO;
95    }
96    let lines = AirLines {
97        frequency: &n2o::FRE,
98        intensity: &n2o::FLIN,
99        energy: &n2o::EL,
100        broadening: &n2o::BRDN2OAIR,
101        exponent: Exponent::PerLine(&n2o::TEXPN2O),
102    };
103    let tables = by_pressure(
104        pp,
105        (&n2o::INI1[..], &n2o::IFIN1[..]),
106        (&n2o::INI2[..], &n2o::IFIN2[..]),
107        (&n2o::INI3[..], &n2o::IFIN3[..]),
108    );
109    let Some(range) = window(frequency_bin(nu), 2, tables) else {
110        return ZERO;
111    };
112    let q = 1.65873970 * tt;
113    scaled(lines.sum(nu, tt, pp, 44.0, range), nu, tt, 0.161, q)
114}

116/// CO (source 5). ATM has no CO lines above 999.9 GHz.
117pub(super) fn co(tt: f64, pp: f64, nu: f64) -> Complex64 {
118    if nu > 999.9 {
119        return ZERO;
120    }
121    let lines = AirLines {
122        frequency: &co::FRE,
123        intensity: &co::FLIN,
124        energy: &co::EL,
125        broadening: &co::BRDCOAIR,
126        exponent: Exponent::PerLine(&co::TEXPCO),
127    };
128    let tables = by_pressure(
129        pp,
130        (&co::INI1[..], &co::IFIN1[..]),
131        (&co::INI2[..], &co::IFIN2[..]),
132        (&co::INI3[..], &co::IFIN3[..]),
133    );
134    let Some(range) = window(frequency_bin(nu), 2, tables) else {
135        return ZERO;
136    };
137    let q = 0.3615187262 * tt;
138    scaled(lines.sum(nu, tt, pp, 28.0, range), nu, tt, 0.112, q)
139}
