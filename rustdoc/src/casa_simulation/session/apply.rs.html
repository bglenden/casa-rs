2//! `sm.setapply`: corrupt by a calibration table (#841 Phase 3).
3//!
4//! CASA's `Simulator::setapply` adds the table's term to those `corrupt`
5//! applies; the term corrupts the model as a simulated one of its type
6//! does (`VisEquation::collapseForSim`: a `G` table multiplies each
7//! correlation by `g_p·conj(g_q)`, a `D` table leaks into the cross hands),
8//! with the table's solutions found and interpolated in time as
9//! `calibrater` applies them, at each row's time, scans observed after the
10//! `setapply` included (measured on CASA 6.7.6: `interp='linear'`
11//! interpolates amplitude and phase apart, `'nearest'` takes the nearest).
12//! casa-rs loads the table at `setapply`, as CASA does, and keeps it; at
13//! each `corrupt` it finds the table's solutions through `casa-calibration`'s
14//! [`CalibrationTableLookup`] for each integration observed by then and
15//! antenna, and applies them as the terms `setgain` and `setleakage`
16//! simulate. A table changed or moved after `setapply` does not change the
17//! term (#893 review).
18//!
19//! CASA's `setapply` ignores its `field`, and its interpolation does not
20//! partition the table by field: a row takes the solutions of every field
21//! as one series in time (measured on CASA
22//! 6.7.6, #893 review).
23//!
24//! Loading a table is admitted against the session's resource policy
25//! before it is read, and the loaded table keeps its share for as long as
26//! the term does; each `corrupt` admits the term it evaluates, as
27//! `setgain` admits one it simulates.
28

48/// The phase loading a `setapply` table is admitted as: the table as it is
49/// read, and what the loaded lookup keeps.
50const LOOKUP_PHASE: &str = "loading a setapply calibration table";
51
52/// The arguments of `sm.setapply`.
53#[derive(Debug, Clone, PartialEq)]
54pub struct ApplySpec {
55    /// `table`: the calibration table to corrupt by.
56    pub table: String,
57    /// `type`: its calibration type; empty takes the table's. Supports complex
58    /// G/D/T/B/Tf/Df, J/Jf, Dgen/Dfgen, X/Xf/GlinXph/GlinXphf, and
59    /// supplied delay, opacity, gain/power-curve and Tsys parameters.
60    pub kind: String,
61    /// `t`: CASA records this interval, but G/D/T table application still
62    /// evaluates the requested interpolation at each data time.
63    pub interval_s: f64,
64    /// `interp`: time interpolation (`nearest`, `linear`, `cubic`, `spline`), optionally
65    /// followed by `,` and a frequency interpolation; empty is `linear`.
66    pub interp: String,
67    /// `calwt`: accepted as in CASA; corruption explicitly disables weight
68    /// calibration, so both settings preserve WEIGHT and SIGMA.
69    pub calwt: bool,
70    /// `spwmap`: empty or [-1] preserves data-window ids. One nonnegative
71    /// id broadcasts to every data window; a longer list maps a prefix.
72    /// [-999] broadcasts the first window with solutions. Other negative
73    /// or out-of-range ids raise.
74    pub spwmap: Vec<i64>,
75    /// `opacity`: ignored when a supplied table defines the parameters, even
76    /// for TOpac, as in CASA.
77    pub opacity: f64,
78}

80/// A gain or leakage term `corrupt` applies: simulated when it was set
81/// (`setgain`, `setleakage`), or a calibration table's (`setapply`),
82/// evaluated over the scans observed at each `corrupt`.
83#[derive(Debug, Clone)]
84pub(crate) enum JonesSource {
85    Simulated(Arc<HeldTerm>),
86    Applied(AppliedTable),
87}
88
89impl JonesSource {
90    /// Metadata-only coverage, before any applied solutions are formed.
91    pub(super) fn covers(&self, window: usize) -> bool {
92        match self {
93            Self::Simulated(term) => term.covers(window),
94            Self::Applied(table) => table.calibration_window(window).is_some(),
95        }
96    }
97}

99/// A loaded table's mathematical operator. A scalar T parameter expands
100/// to the same gain on both receptors; it shares the gain kernel.
101#[derive(Debug, Clone, Copy, PartialEq, Eq)]
102enum AppliedKind {
103    Gain,
104    Leakage,
105    Scalar,
106    Delay,
107    MultiBandDelay,
108    AntennaPosition,
109    Opacity,
110    GainCurve,
111    PowerCurve,
112    SystemTemperature,
113    General,
114    FullLeakage,
115    CrossPhase,
116    SkyReceptorGain,
117}
118
119impl AppliedKind {
120    fn operator(self) -> JonesKind {
121        match self {
122            Self::Gain
123            | Self::Scalar
124            | Self::Delay
125            | Self::MultiBandDelay
126            | Self::AntennaPosition
127            | Self::Opacity
128            | Self::SystemTemperature => JonesKind::Gain,
129            Self::Leakage | Self::FullLeakage => JonesKind::Leakage,
130            Self::General => JonesKind::General,
131            Self::GainCurve | Self::PowerCurve | Self::CrossPhase | Self::SkyReceptorGain => {
132                JonesKind::SkyGain
133            }
134        }
135    }
136
137    fn receptors(self) -> usize {
138        match self {
139            Self::Scalar | Self::Opacity | Self::CrossPhase => 1,
140            Self::AntennaPosition => 3,
141            Self::GainCurve => 8,
142            Self::PowerCurve => 16,
143            Self::General => 4,
144            _ => 2,
145        }
146    }
147
148    fn matrix(self) -> bool {
149        matches!(self, Self::General | Self::FullLeakage)
150    }
151}

153/// A table `setapply` set: the table as it was loaded, and how to apply it.
154#[derive(Clone)]
155pub(crate) struct AppliedTable {
156    path: PathBuf,
157    lookup: Arc<HeldLookup>,
158    kind: AppliedKind,
159    frequency: Option<FrequencyInterpolation>,
160    spwmap: Vec<usize>,
161    phase_delay: bool,
162}
163
164impl AppliedTable {
165    fn channel_dependent(&self) -> bool {
166        self.frequency.is_some()
167            || matches!(self.kind, AppliedKind::Delay | AppliedKind::MultiBandDelay)
168    }
169
170    fn calibration_window(&self, window: usize) -> Option<usize> {
171        let mapped = *self.spwmap.get(window)?;
172        self.lookup
173            .lookup
174            .has_spectral_window(mapped as i32)
175            .then_some(mapped)
176    }
177}
178
179impl std::fmt::Debug for AppliedTable {
180    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
181        formatter
182            .debug_struct("AppliedTable")
183            .field("path", &self.path)
184            .field("kind", &self.kind)
185            .field("spwmap", &self.spwmap)
186            .finish_non_exhaustive()
187    }
188}

220/// The calibration type a table's subtype names: `G Jones` is `G`.
221fn table_type(subtype: &str) -> String {
222    subtype

224        .unwrap_or(subtype)
225        .to_uppercase()

228impl Simulator {
229    /// `sm.setapply(...)`: corrupt by the calibration table `spec.table`
230    /// at each later `corrupt`, its solutions taken at each integration
231    /// observed by then.
232    ///
233    /// # Errors
234    ///
235    /// - The table cannot be read, its type is not the `type` asked for,
236    ///   or `interp` is CASA's deprecated `aipslin`: raised, as CASA
237    ///   raises. A table whose parameter count disagrees with its type
238    ///   raises at `corrupt`, as CASA does.
239    /// - Table types beyond the documented complex, delay, opacity, curve and Tsys families:
240    ///   not yet implemented (#1291).
241    /// - Time interpolation supports nearest, linear, cubic and natural spline.
242    ///   Cubic interior queries require four times; exact and endpoint queries
243    ///   accept shorter axes. Spline corrects CASA's setup crash (#1053).
244    ///   PD scales complex phases; perobs/perscan select time segments.
245    ///   Real FPARAM curves interpolate signed parameters before forming gains.
246    /// - Frequency interpolation supports nearest, linear, cubic and spline,
247    ///   their flag-retaining forms, and CASA's default linear fallback.
248    ///   Cubic interior interpolation with fewer than four valid input
249    ///   channels raises at `corrupt`, as CASA does.
250    /// - No rows observed yet: refused, as `setgain` refuses.
251    ///
252    /// The term snapshots its spectral-window map. A window added later,
253    /// or mapped to a calibration window with no solutions, skips the whole
254    /// corruption step, including noise and corrected-data updates, as in CASA.
255    ///
256    /// Flagged G/T parameters and missing G/T antennas use unit gains,
257    /// as in CASA. A scalar T parameter applies equally to all correlations.
258    /// For D tables an invalid receptor or missing antenna uses neutral leakage (both coefficients zero), correcting CASA's unit
259    /// leakage substitution. Input data flags, weights and sigma persist.
260    pub fn set_apply(&mut self, spec: &ApplySpec) -> SimulatorResult<()> {
261        let mut interp = apply_interpolation(&spec.interp)?;
262        interp.observation_count = self.observations.len();
263        // VisCal::corrupt turns weight calibration off. G/D/T table lookup
264        // uses each row's time regardless of t, and only TOPAC reads opacity.
265        if spec.table.is_empty() {
266            return Err(raised("No calibration table given."));
267        }
268        let path = std::path::absolute(&spec.table).map_err(raised)?;
269        let lookup = self.load_table(&path, interp)?;
270        let table_kind = table_type(lookup.lookup.subtype());
271        let kind = match spec.kind.to_uppercase() {
272            requested if requested.is_empty() || requested == table_kind => table_kind,
273            requested => {
274                return Err(raised(format!(
275                    "{} is a {} table, not {requested}.",
276                    spec.table,
277                    lookup.lookup.subtype()
278                )));
279            }
280        };
281        let frequency = matches!(
282            kind.as_str(),
283            "B" | "TF" | "DF" | "B TSYS" | "JF" | "DFGEN" | "TFOPAC" | "XF" | "GLINXPHF"
284        )
285        .then(|| frequency_interpolation(&spec.interp));
286        let kind = match kind.as_str() {
287            "G" | "B" => AppliedKind::Gain,
288            "D" | "DF" => AppliedKind::Leakage,
289            "T" | "TF" => AppliedKind::Scalar,
290            "K" | "KCROSS" => AppliedKind::Delay,
291            "KMBD" => AppliedKind::MultiBandDelay,
292            "KANTPOS" => AppliedKind::AntennaPosition,
293            "TOPAC" | "TFOPAC" => AppliedKind::Opacity,
294            "EGAINCURVE" => AppliedKind::GainCurve,
295            "EPOWERCURVE" => AppliedKind::PowerCurve,
296            "B TSYS" => AppliedKind::SystemTemperature,
297            "J" | "JF" => AppliedKind::General,
298            "DGEN" | "DFGEN" => AppliedKind::FullLeakage,
299            "X" | "XF" => AppliedKind::CrossPhase,
300            "GLINXPH" | "GLINXPHF" => AppliedKind::SkyReceptorGain,
301            other => return Err(not_yet(format!("of a {other} table"))),
302        };
303        let spwmap = mapped_windows(&spec.spwmap, &lookup, self.spectral_windows.len())?;
304        self.require_rows()?;
305        tracing::info!(
306            "Arranging to CORRUPT with: {} ({})",
307            spec.table,
308            lookup.lookup.subtype()
309        );
310        self.jones_terms.push(JonesSource::Applied(AppliedTable {
311            path,
312            lookup: Arc::new(lookup),
313            kind,
314            frequency,
315            spwmap,
316            phase_delay: interp.phase_delay,
317        }));
318        Ok(())
319    }

321    /// Load the table at `path` to look its solutions up across its fields
322    /// with `interp`, its memory admitted before it is read: what reading
323    /// it holds, and what the lookup keeps, which it holds on to.
324    fn load_table(
325        &self,
326        path: &std::path::Path,
327        interp: TimeInterpolationOptions,
328    ) -> SimulatorResult<HeldLookup> {
329        let metadata_bytes =
330            CalibrationTableLookup::metadata_bytes_on_disk(path).map_err(raised)?;
331        let resources = &self.options.resources;
332        // Admit headers before inspecting shapes; no numerical array is
333        // materialized by layout_on_disk. Only scalar counts survive it.
334        let metadata = admit(
335            resources.host(),
336            resources.policy(),
337            &Demand {
338                phase: LOOKUP_PHASE,
339                memory: metadata_bytes,
340            },
341        )
342        .map_err(SimulatorError::admission)?;
343        let layout = CalibrationTableLookup::layout_for_simulation(path, interp).map_err(raised)?;
344        drop(metadata);
345        let kept = layout
346            .kept_bytes()
347            .saturating_add(layout.time_kept_bytes(interp.mode))
348            .saturating_add(layout.context_kept_bytes(interp));
349        let mut reservation = admit(
350            resources.host(),
351            resources.policy(),
352            &Demand {
353                phase: LOOKUP_PHASE,
354                memory: layout
355                    .loading_bytes()
356                    .saturating_add(layout.time_loading_bytes(interp.mode))
357                    .saturating_add(layout.context_loading_bytes(interp))
358                    .saturating_add(kept),
359            },
360        )
361        .map_err(SimulatorError::admission)?;
362        let lookup = CalibrationTableLookup::open_for_simulation(path, interp).map_err(raised)?;
363        let frequencies = if layout.frequency_values > 0 {
364            CalibrationTableLookup::frequency_grids_on_disk(path).map_err(raised)?
365        } else {
366            Vec::new()
367        };
368        reservation.retain(kept);
369        Ok(HeldLookup {
370            lookup,
371            frequencies,
372            max_cell_values: layout.max_cell_values,
373            windows: layout.frequency_windows,
374            _reservation: reservation,
375        })
376    }

378    /// The term `source` is at this `corrupt`: a simulated one as it was
379    /// set, a table's evaluated over the scans observed so far. An
380    /// evaluated term is admitted as a simulated one is, before it is
381    /// formed, and keeps its share while a step applies it.
382    pub(super) fn jones_term(
383        &self,
384        source: &JonesSource,
385        covered: &impl Fn(usize) -> bool,
386    ) -> SimulatorResult<Arc<HeldTerm>> {
387        match source {
388            JonesSource::Simulated(term) => Ok(Arc::clone(term)),
389            JonesSource::Applied(table) => {
390                let antennas = self.antennas.len();
391                let (mut per_window, _) = self.integration_counts();
392                // Definitions without observed rows still own a window slot.
393                per_window.resize(self.spectral_windows.len(), 0);
394                for (window, count) in per_window.iter_mut().enumerate() {
395                    if !covered(window) {
396                        *count = 0;
397                    }
398                }
399                let channels = self
400                    .spectral_windows
401                    .iter()
402                    .enumerate()
403                    .map(|(index, window)| {
404                        if per_window[index] == 0 {
405                            0
406                        } else {
407                            window.channel_count
408                        }
409                    })
410                    .collect::<Vec<_>>();
411                let mut bytes = if table.channel_dependent() {
412                    TermBytes::frequency(
413                        &per_window,
414                        &channels,
415                        antennas,
416                        table.lookup.max_cell_values,
417                    )
418                } else {
419                    TermBytes::per_chunk(&per_window, antennas)
420                };
421                if matches!(
422                    table.kind,
423                    AppliedKind::Opacity | AppliedKind::GainCurve | AppliedKind::PowerCurve
424                ) {
425                    bytes.forming += (antennas * size_of::<f64>()) as u64;
426                }
427                if !table.channel_dependent() && !table.kind.matrix() {
428                    bytes.forming += (table.lookup.max_cell_values.saturating_sub(2) * 16) as u64;
429                }
430                if table.kind.matrix() {
431                    let count: usize = per_window
432                        .iter()
433                        .enumerate()
434                        .map(|(window, count)| {
435                            count
436                                * antennas
437                                * if table.channel_dependent() {
438                                    channels[window]
439                                } else {
440                                    1
441                                }
442                        })
443                        .sum();
444                    bytes.kept += (count * size_of::<[Complex32; 2]>()) as u64;
445                    if !table.channel_dependent() {
446                        bytes.kept +=
447                            ((per_window.iter().sum::<usize>() + 1) * size_of::<usize>()) as u64;
448                    }
449                    bytes.forming += (table.lookup.max_cell_values * 32) as u64;
450                }
451                let mut reservation = self.admit_term(bytes)?;
452                let evaluated = self.applied_solutions(table, &per_window)?;
453                reservation.retain(bytes.kept);
454                let term = if table.kind.matrix() {
455                    applied_matrices(
456                        table.kind.operator(),
457                        antennas,
458                        evaluated.chunks,
459                        evaluated.matrices,
460                        evaluated.offsets,
461                    )
462                } else if table.channel_dependent() {
463                    applied_frequency(
464                        table.kind.operator(),
465                        antennas,
466                        evaluated.chunks,
467                        evaluated.values,
468                        evaluated.offsets,
469                    )
470                } else {
471                    applied(
472                        table.kind.operator(),
473                        antennas,
474                        evaluated.chunks,
475                        evaluated.values,
476                    )
477                };
478                Ok(Arc::new(HeldTerm::new(term, reservation)))
479            }
480        }
481    }

483    /// Each integration observed so far as a chunk, counted `per_window`,
484    /// with each antenna's solution from `lookup` at its time, field and
485    /// window (mapped by `spwmap`).
486    fn applied_solutions(
487        &self,
488        table: &AppliedTable,
489        per_window: &[usize],
490    ) -> SimulatorResult<AppliedSolutions> {
491        let kind = table.kind;
492        let lookup = &table.lookup.lookup;
493        // JonesDiag::setMatByOk substitutes unit gains per receptor.
494        // JonesGenLin instead substitutes one for BOTH leakage values;
495        // correct that data-corrupting defect with the neutral zero pair.
496        let neutral = Complex32::new(
497            if kind.operator() != JonesKind::Leakage {
498                1.0
499            } else {
500                0.0
501            },
502            0.0,
503        );
504        let mut chunks = Chunks::with_capacity(per_window);
505        let value_count = per_window
506            .iter()
507            .enumerate()
508            .map(|(window, count)| {
509                count
510                    * self.antennas.len()
511                    * if table.channel_dependent() {
512                        self.spectral_windows[window].channel_count
513                    } else {
514                        1
515                    }
516            })
517            .sum();
518        let mut values = if kind.matrix() {
519            Vec::new()
520        } else {
521            Vec::with_capacity(value_count)
522        };
523        let mut matrices = if kind.matrix() {
524            Vec::with_capacity(value_count)
525        } else {
526            Vec::new()
527        };
528        let mut offsets = if table.channel_dependent() || kind.matrix() {
529            Vec::with_capacity(per_window.iter().sum::<usize>() + 1)
530        } else {
531            Vec::new()
532        };
533        for (index, scan) in self.scans.iter().enumerate() {
534            if per_window[scan.spectral_window_id] == 0 {
535                continue;
536            }
537            let Some(window) = table.calibration_window(scan.spectral_window_id) else {
538                continue;
539            };
540            let output_hz = if table.channel_dependent() {
541                self.spectral_windows[scan.spectral_window_id].channel_frequencies_hz()
542            } else {
543                Vec::new()
544            };
545            let channels = if table.channel_dependent() {
546                output_hz.len()
547            } else {
548                1
549            };
550            let phase_frequency = (table.phase_delay && table.frequency.is_none()).then(|| {
551                let data = &self.spectral_windows[scan.spectral_window_id];
552                data.start_frequency_hz
553                    + (data.channel_count - 1) as f64 * data.channel_width_hz / 2.
554            });
555            for time_s in self.integration_times_s(index) {
556                if table.channel_dependent() || kind.matrix() {
557                    offsets.push(if kind.matrix() {
558                        matrices.len()
559                    } else {
560                        values.len()
561                    });
562                }
563                chunks.push_for_scan(scan.spectral_window_id, time_s, scan.field_id, index);
564                let geometry = physical::Geometry::new(self, kind, scan.field_id, time_s)?;
565                for antenna in 0..self.antennas.len() {
566                    let Some(solution) = lookup
567                        .simulation_solution(
568                            scan.observation_id,
569                            scan.scan_number,
570                            window as i32,
571                            antenna as i32,
572                            time_s,
573                            phase_frequency,
574                        )
575                        .map_err(raised)?
576                    else {
577                        if kind.matrix() {
578                            matrices.extend(std::iter::repeat_n(matrix::IDENTITY, channels));
579                        } else {
580                            values.extend(std::iter::repeat_n([neutral; 2], channels));
581                        }
582                        continue;
583                    };
584                    if solution.channels != 1 && table.frequency.is_none() {
585                        return Err(not_yet("of a table of solutions per channel"));
586                    }
587                    if solution.receptors != kind.receptors() {
588                        return Err(raised(format!(
589                            "Calibration table has {} parameters per channel; {:?} requires {}",
590                            solution.receptors,
591                            kind,
592                            kind.receptors()
593                        )));
594                    }
595                    let solution = if let Some(mode) = table.frequency {
596                        let input_hz = table
597                            .lookup
598                            .frequencies
599                            .get(window)
600                            .map(Vec::as_slice)
601                            .unwrap_or(&[]);
602                        if matches!(kind, AppliedKind::SystemTemperature | AppliedKind::Opacity) {
603                            solution.resample_real_frequency(input_hz, &output_hz, mode)
604                        } else {
605                            solution.resample_frequency(input_hz, &output_hz, mode)
606                        }
607                        .map_err(raised)?
608                    } else {
609                        solution
610                    };
611                    let reference = table
612                        .lookup
613                        .frequencies
614                        .get(window)
615                        .and_then(|axis| axis.get(axis.len() / 2))
616                        .copied()
617                        .unwrap_or(0.);
618                    for channel in 0..channels {
619                        if kind.matrix() {
620                            matrices.push(matrix::parameters(kind, &solution, channel));
621                            continue;
622                        }
623                        if let Some(pair) = geometry.values(
624                            kind,
625                            &solution,
626                            antenna,
627                            (channel, output_hz.get(channel).copied().unwrap_or(0.)),
628                            reference,
629                        ) {
630                            values.push(pair);
631                            continue;
632                        }
633                        let invalid_leakage = kind == AppliedKind::Leakage
634                            && (0..solution.receptors)
635                                .any(|receptor| solution.flags[receptor * channels + channel]);
636                        values.push(std::array::from_fn(|receptor| {
637                            let receptor = if kind == AppliedKind::Scalar {
638                                0
639                            } else {
640                                receptor
641                            };
642                            if invalid_leakage || solution.flags[receptor * channels + channel] {
643                                neutral
644                            } else {
645                                solution.value(receptor, channel)
646                            }
647                        }));
648                    }
649                }
650            }
651        }
652        if table.channel_dependent() || kind.matrix() {
653            offsets.push(if kind.matrix() {
654                matrices.len()
655            } else {
656                values.len()
657            });
658        }
659        Ok(AppliedSolutions {
660            chunks: chunks.sorted(),
661            values,
662            offsets,
663            matrices,
664        })
665    }
666}

668/// The simulator's time interpolation, refusing deprecated and unknown modes
669/// as CASA does; unsupported modifiers remain explicit refusals (#1053).
670fn apply_interpolation(interp: &str) -> SimulatorResult<TimeInterpolationOptions> {
671    if interp.contains("aips") {
672        return Err(raised(
673            "The (peculiar) 'aipslin' interpolation type was deprecated in CASA v3.4; use \
674             'linear'.",
675        ));
676    }
677    let requested = interp.split(',').next().unwrap_or("").trim();
678    let mode = if requested.contains("nearest") {
679        TimeInterpolation::Nearest
680    } else if requested.is_empty() || requested.contains("linear") {
681        TimeInterpolation::Linear
682    } else if requested.contains("cubic") {
683        TimeInterpolation::Cubic
684    } else if requested.contains("spline") {
685        TimeInterpolation::Spline
686    } else {
687        return Err(raised(format!(
688            "Unrecognized time interpolation type '{requested}'"
689        )));
690    };
691    Ok(TimeInterpolationOptions {
692        mode,
693        phase_delay: requested.contains("PD"),
694        per_observation: requested.contains("perobs"),
695        per_scan: requested.contains("perscan"),
696        observation_count: 1,
697    })
698}

700/// CASA defaults frequency interpolation to linear for frequency-dependent parameters.
701fn frequency_interpolation(interp: &str) -> FrequencyInterpolation {
702    let mode = interp.split(',').nth(1).unwrap_or("linear");
703    let retain_flags = mode.contains("flag");
704    if mode.contains("nearest") {
705        if retain_flags {
706            FrequencyInterpolation::NearestFlag
707        } else {
708            FrequencyInterpolation::Nearest
709        }
710    } else if mode.contains("cubic") {
711        if retain_flags {
712            FrequencyInterpolation::CubicFlag
713        } else {
714            FrequencyInterpolation::Cubic
715        }
716    } else if mode.contains("spline") {
717        if retain_flags {
718            FrequencyInterpolation::SplineFlag
719        } else {
720            FrequencyInterpolation::Spline
721        }
722    } else if retain_flags {
723        FrequencyInterpolation::LinearFlag
724    } else {
725        FrequencyInterpolation::Linear
726    }
727}

729/// CTPatchedInterp::setSpwMap: identity, single-window broadcast, explicit
730/// prefix, or the first available solution window (-999).
731fn mapped_windows(
732    spec: &[i64],
733    lookup: &HeldLookup,
734    windows: usize,
735) -> SimulatorResult<Vec<usize>> {
736    if spec.is_empty() || spec == [-1] {
737        return Ok((0..windows).collect());
738    }
739    if spec.len() > windows {
740        return Err(raised(format!(
741            "Specified spwmap has more elements ({}) than the number of spectral windows in the MS ({windows}).",
742            spec.len()
743        )));
744    }
745    if spec[0] == -999 {
746        let first = lookup.lookup.first_spectral_window().ok_or_else(|| {
747            raised("No calibration spectral window is available for automatic fanout")
748        })?;
749        return Ok(vec![first as usize; windows]);
750    }
751    if spec.iter().any(|&window| window < 0) {
752        return Err(raised("Please specify positive indices in spwmap."));
753    }
754    if spec
755        .iter()
756        .any(|&window| window as u64 >= lookup.windows as u64)
757    {
758        return Err(raised(format!(
759            "Please specify spw indices <= maximum available ({})",
760            lookup.windows.saturating_sub(1)
761        )));
762    }
763    Ok((0..windows)
764        .map(|window| {
765            spec.get(if spec.len() == 1 { 0 } else { window })
766                .map(|&window| window as usize)
767                .unwrap_or(window)
768        })
769        .collect())
770}
