3use std::borrow::Cow;
4
5/// Expand casacore path variables and the current user's home directory.
6/// Returns `None` when repeated expansion exceeds casacore's limit.
7pub(crate) fn expand_path(name: &str) -> Option<Cow<'_, str>> {
8    if !name.starts_with('~') && !name.contains('$') {
9        return Some(Cow::Borrowed(name));
10    }
11    let mut path = Cow::Borrowed(name);
12    // casacore Path::expandName follows environment references recursively
13    // and stops expanding if the substitutions form a cycle.
14    for _ in 0..25 {
15        if let Some(suffix) = path
16            .strip_prefix('~')
17            .filter(|s| s.is_empty() || s.starts_with('/'))
18            && let Ok(home) = std::env::var("HOME")
19            && !home.is_empty()
20        {
21            path = Cow::Owned(format!("{home}{suffix}"));
22        }
23        let mut changed = false;
24        let next = path
25            .split('/')
26            .map(|part| match expand_variable(part) {
27                Some(expanded) => {
28                    changed = true;
29                    Cow::Owned(expanded)
30                }
31                None => Cow::Borrowed(part),
32            })
33            .collect::<Vec<_>>()
34            .join("/");
35        if !changed {
36            return Some(path);
37        }
38        path = Cow::Owned(next);
39    }
40    None
41}
42
43fn expand_variable(part: &str) -> Option<String> {
44    let (prefix, expression) = part.split_once('$')?;
45    let (variable, suffix) = match expression.strip_prefix('{') {
46        Some(braced) => braced.split_once('}')?,
47        None => (expression, ""),
48    };
49    let value = std::env::var(variable)
50        .ok()
51        .filter(|value| !value.is_empty())?;
52    Some(format!("{prefix}{value}{suffix}"))
53}
