2//! An owner's weight image, summed by contribution before it is gridded
3//! (#787).
4//!
5//! `Mode::Weight` places every sample's weight taps at the uv origin
6//! (`CpuBackend`'s gridding, the Metal records), so a sample's contribution
7//! is fixed by its kernel cell, plane, spectral factor, pointing gradient
8//! and the sign of its w, and scales with its weights alone. Samples that
9//! agree on those five add the same taps scaled by their weights: summing
10//! their weights in f64 first and gridding each group once gives the same
11//! image, to rounding, for a lookup and an add per sample rather than a
12//! kernel stamp.
13//!
14//! What it saves, in work and in precision, depends on how many groups the
15//! samples share:
16//! - **Fixed pointing.** A mosaic with fixed pointings shares a few groups
17//!   per pointing and plane: B10, the 1024² ALMA mosaic, sums 4.19 M
18//!   placements into 14 (#787). Its weight image then costs a lookup per
19//!   sample. On f32 grids (Metal, plan decision D2) it also keeps the grid's
20//!   own precision: millions of adds into the origin's cells had lost about
21//!   1e-3, and a few adds per group do not.
22//! - **A pointing that drifts row by row.** AW projection's pointing does,
23//!   which leaves about two samples a group: B3, the 63-field VLASS clean,
24//!   has 3.36 M groups for 6.42 M placements. The saving is then about
25//!   half, and f32 grids keep most of their per-sample drift.
26//!
27//! The row-work pass counts the groups the same way ([`GroupCount`]), so
28//! the strips charge the weight owner for the stamps it will make.
29
30use std::collections::{HashMap, HashSet};

32use casa_imaging_operator::{CfKey, Placement, SampleBlock, SampleBuffer};

35/// Groups an owner's table holds before it grids them and starts again: a
36/// mosaic's first pass has a few per pointing and plane (14 on B10), so the
37/// table grids once a wave unless the pointing drifts row by row.
38pub(in crate::pass) const WEIGHT_GROUPS: usize = 1 << 14;
39
40/// Correlations a placement carries at most.
41const MAX_POL: usize = 4;
42
43/// What fixes one sample's weight-image contribution apart from its
44/// weights: every input `Mode::Weight` reads but the weights.
45///
46/// Both backends locate the weight taps at the uv origin, so the sample's
47/// `u` and `v` are not read. Only the sign of `w` is read (the Mueller table
48/// and the tap conjugation), and `phase` and `prediction_w_positive` are
49/// read only by prediction. Floats are kept as their bits, so two samples
50/// share a group only when their contributions are the same bits.
51#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
52struct Contribution {
53    cf: CfKey,
54    plane: u32,
55    spectral: u32,
56    gradient: [u32; 2],
57    w_positive: bool,
58}
59
60impl Contribution {
61    fn of(placement: &Placement) -> Self {
62        Self {
63            cf: placement.cf,
64            plane: placement.plane,
65            spectral: placement.spectral.to_bits(),
66            gradient: placement.gradient.map(f32::to_bits),
67            w_positive: placement.w > 0.0,
68        }
69    }
70
71    /// A placement with this contribution: at the uv origin, as
72    /// `Mode::Weight` grids every sample, with a w of the group's sign.
73    fn placement(self) -> Placement {
74        Placement {
75            u: 0.0,
76            v: 0.0,
77            w: if self.w_positive { 1.0 } else { -1.0 },
78            prediction_w_positive: self.w_positive,
79            phase: 0.0,
80            plane: self.plane,
81            spectral: f32::from_bits(self.spectral),
82            cf: self.cf,
83            gradient: self.gradient.map(f32::from_bits),
84        }
85    }
86}

88/// The groups a [`WeightSums`] of `capacity` groups grids for the
89/// placements it is shown, counted without summing: the row-work pass
90/// charges the weight owner a kernel stamp for each
91/// ([`ModeWork`](crate::pass::ModeWork)). Its set is allocated whole on the
92/// first placement, so a pass without a weight image holds nothing.
93pub(in crate::pass) struct GroupCount {
94    capacity: usize,
95    seen: HashSet<Contribution>,
96    last: Option<Contribution>,
97}
98
99impl GroupCount {
100    /// A counter that forgets every group when `capacity` are held, as a
101    /// full [`WeightSums`] grids and starts again.
102    pub(in crate::pass) fn new(capacity: usize) -> Self {
103        assert!(capacity > 0, "a group count holds at least one group");
104        Self {
105            capacity,
106            seen: HashSet::new(),
107            last: None,
108        }
109    }
110
111    /// Bytes a counter of `capacity` groups holds once it has seen a
112    /// placement: its hash set's buckets, a key and a control byte each.
113    #[must_use]
114    pub(in crate::pass) fn bytes(capacity: usize) -> u64 {
115        let buckets = (capacity * 8).div_ceil(7).next_power_of_two().max(4);
116        (buckets * (size_of::<Contribution>() + 1) + 16) as u64
117    }
118
119    /// Whether `placement` starts a group that [`WeightSums::add`] would
120    /// grid: its contribution is not among those held since the last time
121    /// the table filled.
122    pub(in crate::pass) fn starts(&mut self, placement: &Placement) -> bool {
123        let key = Contribution::of(placement);
124        if self.last == Some(key) {
125            return false;
126        }
127        self.last = Some(key);
128        if self.seen.capacity() == 0 {
129            self.seen.reserve(self.capacity);
130        }
131        if self.seen.contains(&key) {
132            return false;
133        }
134        if self.seen.len() == self.capacity {
135            self.seen.clear();
136        }
137        self.seen.insert(key);
138        true
139    }
140}

142/// One owner's weight-image contributions, summed in f64 by
143/// [`Contribution`] in the order they first appear, and gridded as one
144/// placement per group ([`Self::flush`]) when the table fills and when the
145/// wave ends.
146pub(in crate::pass) struct WeightSums {
147    npol: usize,
148    capacity: usize,
149    index: HashMap<Contribution, usize>,
150    groups: Vec<Contribution>,
151    /// `npol` weights per group.
152    sums: Vec<f64>,
153    /// The groups as placements, for the backend.
154    buffer: SampleBuffer,
155    /// Placements summed, groups gridded and grids made since the table
156    /// was made, for the wave's log ([`Self::totals`]).
157    totals: [u64; 3],
158}
159
160impl WeightSums {
161    /// An empty table of `capacity` groups of `npol` weights, allocated
162    /// whole ([`Self::bytes`]).
163    pub(in crate::pass) fn new(npol: usize, capacity: usize) -> Self {
164        assert!(
165            (1..=MAX_POL).contains(&npol),
166            "a placement carries one to four correlations"
167        );
168        assert!(capacity > 0, "a weight table holds at least one group");
169        Self {
170            npol,
171            capacity,
172            index: HashMap::with_capacity(capacity),
173            groups: Vec::with_capacity(capacity),
174            sums: Vec::with_capacity(capacity * npol),
175            buffer: SampleBuffer::with_capacity(npol, capacity),
176            totals: [0; 3],
177        }
178    }
179
180    /// Placements summed, groups gridded and grids made so far.
181    pub(in crate::pass) const fn totals(&self) -> [u64; 3] {
182        self.totals
183    }
184
185    /// Bytes a table of `capacity` groups of `npol` weights holds: the
186    /// hash index (its buckets, each a key, an index and a control byte,
187    /// at the standard library's 7/8 load), the groups, their sums and
188    /// the buffer they are gridded from.
189    #[must_use]
190    pub(in crate::pass) fn bytes(npol: usize, capacity: usize) -> u64 {
191        let buckets = (capacity * 8).div_ceil(7).next_power_of_two().max(4);
192        let index = buckets * (size_of::<(Contribution, usize)>() + 1) + 16;
193        (index + capacity * size_of::<Contribution>() + capacity * npol * size_of::<f64>()) as u64
194            + SampleBuffer::bytes(npol, capacity)
195    }
196
197    /// Bytes the table holds now, from its parts' capacities.
198    #[cfg(test)]
199    fn capacity_bytes(&self) -> u64 {
200        let index = self.index.capacity();
201        let buckets = (index * 8).div_ceil(7).next_power_of_two().max(4);
202        (buckets * (size_of::<(Contribution, usize)>() + 1)
203            + 16
204            + self.groups.capacity() * size_of::<Contribution>()
205            + self.sums.capacity() * size_of::<f64>()) as u64
206            + self.buffer.capacity_bytes()
207    }
208
209    /// Add the weights of every placement of `block` to its group, gridding
210    /// the groups through `grid` first whenever a new group finds the
211    /// table full.
212    pub(in crate::pass) fn add<E>(
213        &mut self,
214        block: &SampleBlock<'_>,
215        grid: &mut dyn FnMut(&SampleBlock<'_>) -> Result<(), E>,
216    ) -> Result<(), E> {
217        debug_assert_eq!(block.npol, self.npol, "the table's correlations");
218        self.totals[0] += block.len() as u64;
219        // Rows place their channels together, so successive samples share
220        // a group almost always: look the hash up only when it changes.
221        let mut last: Option<(Contribution, usize)> = None;
222        for (index, placement) in block.placements.iter().enumerate() {
223            let key = Contribution::of(placement);
224            let group = match last {
225                Some((held, group)) if held == key => group,
226                _ => {
227                    let group = match self.index.get(&key) {
228                        Some(&group) => group,
229                        None => {
230                            if self.groups.len() == self.capacity {
231                                self.flush(grid)?;
232                            }
233                            let group = self.groups.len();
234                            self.index.insert(key, group);
235                            self.groups.push(key);
236                            self.sums.resize(self.sums.len() + self.npol, 0.0);
237                            group
238                        }
239                    };
240                    last = Some((key, group));
241                    group
242                }
243            };
244            let sums = &mut self.sums[group * self.npol..(group + 1) * self.npol];
245            for (sum, weight) in sums.iter_mut().zip(block.weights_of(index)) {
246                *sum += f64::from(*weight);
247            }
248        }
249        Ok(())
250    }
251
252    /// Grid every group through `grid`, as one placement of its summed
253    /// weights each, and empty the table. Nothing when it is empty.
254    pub(in crate::pass) fn flush<E>(
255        &mut self,
256        grid: &mut dyn FnMut(&SampleBlock<'_>) -> Result<(), E>,
257    ) -> Result<(), E> {
258        if self.groups.is_empty() {
259            return Ok(());
260        }
261        self.buffer.clear();
262        let values = [Complex32::default(); MAX_POL];
263        let mut weights = [0.0_f32; MAX_POL];
264        for (group, key) in self.groups.iter().enumerate() {
265            let sums = &self.sums[group * self.npol..(group + 1) * self.npol];
266            for (weight, sum) in weights.iter_mut().zip(sums) {
267                *weight = *sum as f32;
268            }
269            self.buffer
270                .push(key.placement(), &values[..self.npol], &weights[..self.npol]);
271        }
272        self.totals[1] += self.groups.len() as u64;
273        self.totals[2] += 1;
274        self.index.clear();
275        self.groups.clear();
276        self.sums.clear();
277        grid(&self.buffer.block())
278    }
279}

299    /// The groups a table hands to `grid`, with their summed weights.
300    fn flushed(sums: &mut WeightSums) -> Vec<(Placement, Vec<f32>)> {
301        let mut out = Vec::new();
302        sums.flush(&mut |block: &SampleBlock<'_>| {
303            for (index, placement) in block.placements.iter().enumerate() {
304                out.push((*placement, block.weights_of(index).to_vec()));
305            }
306            Ok::<_, ()>(())
307        })
308        .expect("flush");
309        out
310    }

432    /// `samples` weight placements over three pointings, three spectral
433    /// factors and both signs of w, with random weights, in random order.
434    fn weight_samples(
435        operator: &casa_imaging_operator::MeasurementOperator,
436        samples: usize,
437    ) -> SampleBuffer {
438        let gradients = [[0.0, 0.0], [0.6, -0.2], [-0.3, 0.9]].map(|offset: [f64; 2]| {
439            operator
440                .geometry()
441                .pointing_gradient([offset[0] * 1.0e-4, offset[1] * 1.0e-4])
442        });
443        let spectral = [-0.01_f32, 0.0, 0.02];
444        let mut state = 0x9e37_79b9_7f4a_7c15_u64;
445        let mut next = || {
446            state ^= state << 13;
447            state ^= state >> 7;
448            state ^= state << 17;
449            state
450        };
451        let mut buffer = SampleBuffer::with_capacity(2, samples);
452        for _ in 0..samples {
453            let draw = next();
454            let weight = |shift: u32| 0.5 + ((draw >> shift) & 0xffff) as f32 / 65536.0;
455            buffer.push(
456                Placement {
457                    u: 0.0,
458                    v: 0.0,
459                    w: if draw & 1 == 0 { 25.0 } else { -25.0 },
460                    prediction_w_positive: true,
461                    phase: 0.0,
462                    plane: 0,
463                    spectral: spectral[(draw >> 1) as usize % 3],
464                    cf: CfKey::default(),
465                    gradient: gradients[(draw >> 3) as usize % 3],
466                },
467                &[Complex32::default(); 2],
468                &[weight(16), weight(32)],
469            );
470        }
471        buffer
472    }

474    /// The weight image `samples` grid on `operator`, one sample at a time
475    /// (`per_sample`) or summed by contribution, as its finished images.
476    fn weight_image(
477        operator: &casa_imaging_operator::MeasurementOperator,
478        samples: &SampleBuffer,
479        per_sample: bool,
480    ) -> Vec<f32> {
481        weight_image_and_sums(operator, samples, per_sample).0
482    }
483
484    /// [`weight_image`] and the finished images' `sumwt`.
485    fn weight_image_and_sums(
486        operator: &casa_imaging_operator::MeasurementOperator,
487        samples: &SampleBuffer,
488        per_sample: bool,
489    ) -> (Vec<f32>, Vec<f64>) {
490        use casa_imaging_operator::{CpuBackend, GridBackend, Mode, ModeSet, PlaneRange, Work};
491        let mut acc = operator.accumulator(PlaneRange::single(0), None, ModeSet::ALL);
492        let mut backend = CpuBackend::new();
493        let cf = operator.cf();
494        let mut grid = |block: &SampleBlock<'_>| {
495            backend.apply(
496                block,
497                cf,
498                Work::Grid {
499                    mode: Mode::Weight,
500                    acc: &mut acc,
501                },
502            )
503        };
504        if per_sample {
505            grid(&samples.block()).expect("grid");
506        } else {
507            let mut sums = WeightSums::new(2, WEIGHT_GROUPS);
508            sums.add(&samples.block(), &mut grid).expect("add");
509            sums.flush(&mut grid).expect("flush");
510        }
511        let images = operator.finish(acc).expect("finish");
512        let plane = &images.planes[0];
513        (
514            plane
515                .weight
516                .iter()
517                .flat_map(|image| image.iter().copied())
518                .collect(),
519            plane.sumwt.clone(),
520        )
521    }

523    /// The largest difference between `a` and `b`, relative to `b`'s peak.
524    fn relative(a: &[f32], b: &[f32]) -> f64 {
525        let peak = b
526            .iter()
527            .fold(0.0_f64, |peak, v| peak.max(f64::from(v.abs())));
528        a.iter()
529            .zip(b)
530            .map(|(a, b)| f64::from((a - b).abs()))
531            .fold(0.0, f64::max)
532            / peak
533    }

535    /// Summing the weight image by contribution grids, on the real mosaic
536    /// set with pointing ramps and Taylor terms, what gridding each sample
537    /// grids, to rounding in f64 grids (#787). On f32 grids it stays at the
538    /// f32 image's precision while the per-sample f32 weight image drifts
539    /// with the sample count.
540    #[test]
541    fn summed_weights_grid_what_each_sample_grids() {
542        use casa_imaging_operator::GridPrecision;
543        let f64_operator = mosaic_operator(GridPrecision::F64);
544        // Enough samples for the per-sample f32 drift, which grows with
545        // their count, to show: B10 grids 4.19 M (#787).
546        let samples = weight_samples(&f64_operator, 400_000);
547        let (reference, reference_sumwt) = weight_image_and_sums(&f64_operator, &samples, true);
548        let (summed, summed_sumwt) = weight_image_and_sums(&f64_operator, &samples, false);
549        let f64_error = relative(&summed, &reference);
550        assert!(f64_error <= 1.0e-6, "f64 grids: {f64_error}");
551        // The weight terms' sumwt takes each group's summed weight once, as
552        // the f32 weight its f64 sum rounds to (#792 review, Astra).
553        assert!(
554            reference_sumwt.iter().any(|sum| *sum > 0.0),
555            "a weight sumwt"
556        );
557        for (summed, each) in summed_sumwt.iter().zip(&reference_sumwt) {
558            assert!(
559                (summed - each).abs() <= 1.0e-7 * each.abs().max(1.0),
560                "sumwt: {summed} against {each}"
561            );
562        }
563        let f32_operator = mosaic_operator(GridPrecision::F32);
564        let summed_f32 = relative(&weight_image(&f32_operator, &samples, false), &reference);
565        let each_f32 = relative(&weight_image(&f32_operator, &samples, true), &reference);
566        assert!(summed_f32 <= 1.0e-6, "summed on f32 grids: {summed_f32}");
567        assert!(
568            each_f32 > 10.0 * summed_f32,
569            "per sample on f32 grids drifts: {each_f32} against {summed_f32}"
570        );
571    }

573    /// With no group shared, every pointing gradient distinct (AW's case
574    /// at its extreme, #792 review, Astra 2), the summed table grids on f32
575    /// grids exactly what each sample grids, bit for bit: groups of one
576    /// sample keep its f32 weights and their order, across the table's
577    /// flushes. It saves nothing then, but costs nothing in precision.
578    #[test]
579    fn with_no_group_shared_the_table_grids_each_sample_exactly() {
580        use casa_imaging_operator::GridPrecision;
581        let operator = mosaic_operator(GridPrecision::F32);
582        let mut samples = weight_samples(&operator, WEIGHT_GROUPS + 3000);
583        let block = samples.block();
584        let mut distinct = SampleBuffer::with_capacity(2, block.len());
585        for (index, placement) in block.placements.iter().enumerate() {
586            let step = index as f32 * 1.0e-7;
587            distinct.push(
588                Placement {
589                    gradient: [placement.gradient[0] + step, placement.gradient[1]],
590                    ..*placement
591                },
592                block.values_of(index),
593                block.weights_of(index),
594            );
595        }
596        samples = distinct;
597        let summed = weight_image(&operator, &samples, false);
598        let each = weight_image(&operator, &samples, true);
599        assert!(
600            summed == each,
601            "summed and per-sample f32 weight images differ"
602        );
603    }

623    /// A kernel set routed as AW routes four linear correlations (#810, for
624    /// #653's AW rows). `AWVisResampler` grids correlation `e` (XX, XY, YX
625    /// and YY are AW's elements 0, 5, 10 and 15, here the cells' Mueller
626    /// planes 0 to 3) through its own plane for w > 0 and through its
627    /// partner's, 15 − e (XX ↔ YY, XY ↔ YX), otherwise, reading the
628    /// visibility of that plane's hand. Each key lends one dense cell of
629    /// complex taps as its imaging and its weight taps, normalised as AW's
630    /// (`KernelSum`), with AW's pointing ramp.
631    struct AwRouted {
632        cells: Vec<casa_imaging_operator::DenseCell>,
633        mueller: casa_imaging_operator::MuellerRouting,
634        correction: casa_imaging_operator::ImageCorrection,
635    }

637    /// Taps per axis and fine offsets per cell of [`AwRouted`]'s cells.
638    const AW_SUPPORT: u16 = 5;
639    const AW_OVERSAMPLING: u16 = 4;
640
641    impl AwRouted {
642        /// `keys` cells of random complex taps, from `seed`.
643        fn new(keys: usize, mut seed: u64) -> Self {
644            use casa_imaging_operator::{DenseCell, ImageCorrection};
645            let mut signed = move || {
646                seed ^= seed << 13;
647                seed ^= seed >> 7;
648                seed ^= seed << 17;
649                ((seed >> 11) as f64 / (1_u64 << 53) as f64 * 2.0 - 1.0) as f32
650            };
651            let fine = usize::from(AW_OVERSAMPLING) + 1;
652            let taps = fine * fine * 4 * usize::from(AW_SUPPORT).pow(2);
653            let cells = (0..keys)
654                .map(|_| {
655                    DenseCell::new(
656                        (0..taps)
657                            .map(|_| Complex32::new(signed(), signed()))
658                            .collect(),
659                        [AW_SUPPORT; 2],
660                        AW_OVERSAMPLING,
661                        4,
662                    )
663                })
664                .collect();
665            let [nx, ny] = aw_geometry().grid_shape();
666            Self {
667                cells,
668                mueller: aw_routing(),
669                correction: ImageCorrection::new(vec![1.0; nx], vec![1.0; ny]),
670            }
671        }
672
673        /// These kernels with every tap replaced by its modulus: gridding a
674        /// sample with them, without its ramp and at the magnitude of its
675        /// spectral factor, adds `|W · s^t · tap|` to each cell.
676        fn magnitudes(&self) -> Self {
677            use casa_imaging_operator::DenseCell;
678            Self {
679                cells: self
680                    .cells
681                    .iter()
682                    .map(|cell| {
683                        DenseCell::new(
684                            cell.data()
685                                .iter()
686                                .map(|tap| Complex32::new(tap.norm(), 0.0))
687                                .collect(),
688                            cell.support(),
689                            cell.oversampling(),
690                            cell.mueller_planes(),
691                        )
692                    })
693                    .collect(),
694                mueller: self.mueller.clone(),
695                correction: self.correction.clone(),
696            }
697        }
698    }

700    /// AW's routes for XX, XY, YX and YY ([`AwRouted`]): for w > 0 grid
701    /// correlation `g` takes visibility `g` through plane `g`; otherwise it
702    /// takes its partner `3 − g` through plane `3 − g`. The forward tables
703    /// swap the signs (`GridToData`) and read the grid of the plane's hand.
704    fn aw_routing() -> casa_imaging_operator::MuellerRouting {
705        let table = |route: fn(usize, usize) -> Option<usize>| -> Vec<Vec<Option<u8>>> {
706            (0..4)
707                .map(|grid| {
708                    (0..4)
709                        .map(|vis| route(grid, vis).map(|plane| plane as u8))
710                        .collect()
711                })
712                .collect()
713        };
714        casa_imaging_operator::MuellerRouting {
715            adjoint: [
716                table(|grid, vis| (vis == 3 - grid).then_some(3 - grid)),
717                table(|grid, vis| (vis == grid).then_some(grid)),
718            ],
719            forward: [
720                table(|grid, vis| (vis == grid).then_some(grid)),
721                table(|grid, vis| (vis == 3 - grid).then_some(grid)),
722            ],
723        }
724    }

778    /// Taylor basis with `kernels`, gridding in f64.
779    fn aw_routed_operator(kernels: AwRouted) -> casa_imaging_operator::MeasurementOperator {
780        use casa_imaging_model::{CorrelationType, PolarizationCoordinate};
781        use casa_imaging_operator::{
782            Basis, GridPrecision, MeasurementOperator, PolarizationRouting,
783        };
784        let routing = PolarizationRouting::compile(
785            &[
786                CorrelationType::LinearXx,
787                CorrelationType::LinearXy,
788                CorrelationType::LinearYx,
789                CorrelationType::LinearYy,
790            ],
791            &[
792                PolarizationCoordinate::LinearXx,
793                PolarizationCoordinate::LinearXy,
794                PolarizationCoordinate::LinearYx,
795                PolarizationCoordinate::LinearYy,
796            ],

811    /// `samples` weight placements over `keys` cells, three pointing
812    /// gradients, three spectral factors and both signs of w, in random
813    /// order, each with random weights on XX, YX and YY and none on XY.
814    fn aw_routed_samples(
815        operator: &casa_imaging_operator::MeasurementOperator,
816        keys: u16,
817        samples: usize,
818    ) -> SampleBuffer {
819        let gradients = [[0.0, 0.0], [0.6, -0.2], [-0.3, 0.9]].map(|offset: [f64; 2]| {
820            operator
821                .geometry()
822                .pointing_gradient([offset[0] * 1.0e-4, offset[1] * 1.0e-4])
823        });
824        let spectral = [-0.01_f32, 0.0, 0.02];
825        let mut state = 0x2545_f491_4f6c_dd1d_u64;
826        let mut next = || {
827            state ^= state << 13;
828            state ^= state >> 7;
829            state ^= state << 17;
830            state
831        };
832        let mut buffer = SampleBuffer::with_capacity(4, samples);
833        for _ in 0..samples {
834            let draw = next();
835            // Fractions of 2^-16, so a group's sum of a thousand needs more
836            // bits than f32 keeps.
837            let weight = |shift: u32| 0.5 + ((draw >> shift) & 0xffff) as f32 / 65536.0;
838            buffer.push(
839                Placement {
840                    u: 0.0,
841                    v: 0.0,
842                    w: if draw & 1 == 0 { 25.0 } else { -25.0 },
843                    prediction_w_positive: true,
844                    phase: 0.0,
845                    plane: 0,
846                    spectral: spectral[(draw >> 1) as usize % 3],
847                    cf: CfKey {
848                        group: u16::try_from((draw >> 3) % u64::from(keys)).expect("a key"),
849                        cube: 0,
850                    },
851                    gradient: gradients[(draw >> 7) as usize % 3],
852                },
853                &[Complex32::default(); 4],
854                &[weight(16), 0.0, weight(32), weight(48)],
855            );
856        }
857        buffer
858    }

860    /// `samples` as `map` changes each placement.
861    fn mapped(samples: &SampleBuffer, map: impl Fn(Placement) -> Placement) -> SampleBuffer {
862        let block = samples.block();
863        let mut out = SampleBuffer::with_capacity(block.npol, block.len());
864        for (index, placement) in block.placements.iter().enumerate() {
865            out.push(
866                map(*placement),
867                block.values_of(index),
868                block.weights_of(index),

874    /// The weight blocks, cell by cell, and the weight `sumwt` that
875    /// `samples` grid on `operator`'s f64 accumulator: one sample at a time
876    /// (`capacity` `None`), or summed by contribution in a table of
877    /// `capacity` groups, with the grids that table made.
878    fn weight_grids(
879        operator: &casa_imaging_operator::MeasurementOperator,
880        samples: &SampleBuffer,
881        capacity: Option<usize>,
882    ) -> (Vec<num_complex::Complex64>, Vec<f64>, u64) {
883        use casa_imaging_operator::{CpuBackend, GridBackend, Mode, ModeSet, PlaneRange, Work};
884        let mut acc = operator.accumulator(PlaneRange::single(0), None, ModeSet::ALL);
885        let mut backend = CpuBackend::new();
886        let cf = operator.cf();
887        let mut grid = |block: &SampleBlock<'_>| {
888            backend.apply(
889                block,
890                cf,
891                Work::Grid {
892                    mode: Mode::Weight,
893                    acc: &mut acc,
894                },
895            )
896        };
897        let grids = match capacity {
898            None => {
899                grid(&samples.block()).expect("grid");
900                0
901            }
902            Some(capacity) => {
903                let mut sums = WeightSums::new(samples.block().npol, capacity);
904                sums.add(&samples.block(), &mut grid).expect("add");
905                sums.flush(&mut grid).expect("flush");
906                sums.totals()[2]
907            }
908        };
909        let layout = acc.layout();
910        let terms = layout.term_range(Mode::Weight).expect("weight terms");
911        let mut cells = Vec::new();
912        let mut sumwt = Vec::new();
913        for pol in 0..layout.pols() {
914            for term in terms.clone() {
915                cells.extend_from_slice(acc.block::<f64>(0, pol, term));
916                sumwt.push(acc.sumwt_at(0, pol, term));
917            }
918        }
919        (cells, sumwt, grids)
920    }

922    /// #792 review (Astra), #810, for #653's AW rows: under AW's routing
923    /// ([`AwRouted`]) summing the weight image by contribution grids what
924    /// gridding each sample grids, to the rounding of each group's sum to
925    /// f32. The routes change with the sign of w and cross the hands, the
926    /// cells are complex and dense with four Mueller planes, XY weighs
927    /// nothing in either route that reads it, and the samples carry ramps
928    /// and Taylor terms: in a table that grids once, and in one of seven
929    /// groups that fills and grids every few samples.
930    ///
931    /// The bound is the arithmetic's, cell by cell (Astra's probes saw
932    /// 6.7e-9 of the peak). A group's weights sum in f64 and each grid of
933    /// the group rounds its partial sum to f32, within 2^-24 of it; the
934    /// weights are positive, so a cell's summed value is within 2^-24 of
935    /// `Σ |W · s^t · tap|` of the one gridded sample by sample, and a
936    /// weight `sumwt` within 2^-24 of `Σ W · |s|^t · |N|`, both besides
937    /// f64's own rounding over the samples.
938    #[test]
939    fn summed_weights_grid_what_each_sample_grids_under_aw_routing() {
940        const KEYS: u16 = 3;
941        const SAMPLES: usize = 60_000;
942        let kernels = AwRouted::new(usize::from(KEYS), 0x9e37_79b9_7f4a_7c15);
943        let magnitude_operator = aw_routed_operator(kernels.magnitudes());
944        let operator = aw_routed_operator(kernels);
945        let samples = aw_routed_samples(&operator, KEYS, SAMPLES);
946        let (each, each_sumwt, _) = weight_grids(&operator, &samples, None);
947        let unsigned = |placement: Placement| Placement {
948            spectral: placement.spectral.abs(),
949            ..placement
950        };
951        let (magnitudes, _, _) = weight_grids(
952            &magnitude_operator,
953            &mapped(&samples, |placement| Placement {
954                gradient: [0.0, 0.0],
955                ..unsigned(placement)
956            }),
957            None,
958        );
959        let (_, sumwt_magnitudes, _) = weight_grids(&operator, &mapped(&samples, unsigned), None);
960        let rounding = F32_ROUNDING + 4.0 * SAMPLES as f64 * f64::EPSILON;
961        let block = each.len() / each_sumwt.len();
962        for (index, cells) in each.chunks(block).enumerate() {
963            assert!(
964                cells.iter().any(|cell| cell.norm() > 0.0),
965                "weight block {index} has weight"
966            );
967        }
968        for capacity in [WEIGHT_GROUPS, 7] {
969            let (summed, summed_sumwt, grids) = weight_grids(&operator, &samples, Some(capacity));
970            assert!(
971                capacity == WEIGHT_GROUPS || grids > 1,
972                "a table of {capacity} groups fills"
973            );
974            for (index, ((summed, each), magnitude)) in
975                summed.iter().zip(&each).zip(&magnitudes).enumerate()
976            {
977                assert!(
978                    (summed - each).norm() <= rounding * magnitude.re,
979                    "{capacity} groups, cell {index}: {summed} against {each}, \
980                     magnitude {magnitude}"
981                );
982            }
983            for ((summed, each), magnitude) in
984                summed_sumwt.iter().zip(&each_sumwt).zip(&sumwt_magnitudes)
985            {
986                assert!(
987                    (summed - each).abs() <= rounding * magnitude,
988                    "{capacity} groups, sumwt: {summed} against {each}, magnitude {magnitude}"
989                );
990            }
991        }
992    }
