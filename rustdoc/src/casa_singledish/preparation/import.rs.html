8/// Import an ATNF ASAP scan table into a CASA MeasurementSet.
9///
10/// A readable existing casacore table may be replaced. The input remains
11/// independent of the output and a failed conversion leaves the prior
12/// product available.
13///
14/// # Errors
15/// Returns an error when either path is unsuitable, the input cannot be
16/// decoded, or the complete result cannot be published.
17pub fn import_asap(input: &Path, output: &Path, parallel: bool) -> Result<(), String> {
18    publish_import(
19        input,
20        output,
21        parallel,
22        crate::single_dish_import::read_asap,
23    )
24    .inspect_err(|error| tracing::error!(%error, "ASAP scan-table import failed"))
25}

27/// Import a packed Nobeyama NOSTAR scan file into a CASA MeasurementSet.
28///
29/// The staged product includes the instrument's NRO_ARRAY mapping.
30///
31/// # Errors
32/// Returns an error for an unreadable NOSTAR scan or if the destination
33/// cannot be prepared, written or published safely.
34pub fn import_nro(input: &Path, output: &Path, parallel: bool) -> Result<(), String> {
35    publish_import(input, output, parallel, crate::single_dish_import::read_nro)
36        .inspect_err(|error| tracing::error!(%error, "NRO scan import failed"))
37}
38
39fn publish_import(
40    input: &Path,
41    output: &Path,
42    parallel: bool,
43    decode: fn(&Path, bool) -> casa_ms::MsResult<crate::single_dish_import::ImportedScanTable>,
44) -> Result<(), String> {
45    if input.as_os_str().is_empty() || output.as_os_str().is_empty() {
46        return Err("Both infile and outfile are required.".into());
47    }
48    if casa_calibration::paths::overlap(input, output).map_err(|error| error.to_string())? {
49        return Err("The input scan table and output MeasurementSet overlap.".into());
50    }
51    if output
52        .parent()
53        .filter(|parent| !parent.as_os_str().is_empty())
54        .is_some_and(|parent| !parent.is_dir())
55    {
56        return Err("The output's parent directory does not exist.".into());
57    }
58    let target = CalibrationOutput::prepare_import(output).map_err(|error| error.to_string())?;
59    let mut result = decode(input, parallel).map_err(|error| error.to_string())?;
60    for input_table in &result.input_tables {
61        if casa_calibration::paths::overlap(input_table, target.destination())
62            .map_err(|error| error.to_string())?
63        {
64            return Err(format!(
65                "The output overlaps input table {}.",
66                input_table.display()
67            ));
68        }
69    }
70    result
71        .write_stage(&target.staging)
72        .map_err(|error| error.to_string())?;
73    target.commit().map_err(|error| error.to_string())
74}
