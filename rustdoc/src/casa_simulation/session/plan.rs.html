2//! The deferred plan of DATA: what the session's DATA steps asked for,
3//! applied when `close` writes each row.
4//!
5//! In CASA every step that changes DATA (`predict`, `corrupt`) rewrites the
6//! rows that exist when it is called, which are those of the scans observed
7//! so far; later scans are untouched. Each step here records how many scans
8//! it covers, so for scan `s` DATA is the fold of the steps covering `s`, in
9//! the order they were taken, over the zero DATA `observe` writes. Nothing
10//! is read back from disk, and a sequence of steps costs one pass.
11//!
12//! A `predict` that replaces DATA covers every scan an earlier step covers
13//! (each covers a prefix of the scans), so it clears the plan before it is
14//! recorded; every step left then adds: DATA is the sum of the predictions
15//! covering the scan and the noise of the corruptions covering it. The fold
16//! applies each step to the running DATA (`D ← step(D)`), so multiplicative
17//! gains (`D ← G(D)`) slot in as further [`DataStep`] variants.
18//!
19//! Scans are only ever added, so each step covers at least the scans of the
20//! one before it, and the steps covering a scan are the last ones taken.
21//! Every corruption's applications are kept end to end in one list, so a
22//! scan's noise is the end of that list: `close` reads it where it is, and
23//! N interleaved scans and corruptions keep N lists' worth, not N²
24//! (#853 review).
25
26use std::ops::Range;
27use std::sync::Arc;
28
29use super::prediction::Prediction;
30use crate::corrupt::NoiseApplication;
31
32/// One `setnoise` term: CASA's `ANoise` calibration term, kept until
33/// `reset`.
34#[derive(Debug, Clone, Copy, PartialEq)]
35pub(crate) struct NoiseTerm {
36    /// The seed in force at `setnoise`.
37    pub(crate) seed: i32,
38    /// σ of each real and imaginary draw, Jy.
39    pub(crate) sigma_jy: f32,
40    /// How many `corrupt` calls have applied the term.
41    pub(crate) applications: u32,
42}
43
44/// A step that changes DATA, covering the scans observed before it.
45#[derive(Debug, Clone)]
46pub(crate) enum DataStep {
47    /// `predict`: add the model's prediction. A replacing `predict` is the
48    /// first step of the plan, so adding to the zero DATA replaces it.
49    Predict {
50        /// The prepared model.
51        prediction: Arc<Prediction>,
52        /// The step covers scans `0..scans`.
53        scans: usize,
54    },
55    /// `corrupt`: add each application's noise.
56    Corrupt {
57        /// Its applications, one per noise term in the order the terms
58        /// were set, as their range of the plan's noise.
59        applications: Range<usize>,
60        /// The step covers scans `0..scans`.
61        scans: usize,
62    },
63}
64
65impl DataStep {
66    /// The step covers scans `0..scans()`.
67    pub(crate) fn scans(&self) -> usize {
68        match self {
69            Self::Predict { scans, .. } | Self::Corrupt { scans, .. } => *scans,
70        }
71    }
72}
73
74/// The session's DATA steps, in the order they were taken.
75#[derive(Debug, Clone, Default)]
76pub(crate) struct DataPlan {
77    steps: Vec<DataStep>,
78    /// Every corruption's applications, in the order they were taken.
79    noise: Vec<NoiseApplication>,
80}
81
82impl DataPlan {
83    /// Record a `predict` of `prediction` covering scans `0..scans`, after
84    /// the steps already taken.
85    pub(crate) fn predict(&mut self, prediction: Arc<Prediction>, scans: usize) {
86        self.push(DataStep::Predict { prediction, scans });
87    }
88
89    /// Record a `corrupt` adding `applications` to scans `0..scans`, after
90    /// the steps already taken.
91    pub(crate) fn corrupt(
92        &mut self,
93        applications: impl IntoIterator<Item = NoiseApplication>,
94        scans: usize,
95    ) {
96        let start = self.noise.len();
97        self.noise.extend(applications);
98        self.push(DataStep::Corrupt {
99            applications: start..self.noise.len(),
100            scans,
101        });
102    }
103
104    fn push(&mut self, step: DataStep) {
105        debug_assert!(
106            self.steps
107                .last()
108                .is_none_or(|last| last.scans() <= step.scans()),
109            "a step covers at least the scans of the one before it"
110        );
111        self.steps.push(step);
112    }
113
114    /// Forget every step: a replacing `predict` overwrites what they did.
115    pub(crate) fn clear(&mut self) {
116        self.steps.clear();
117        self.noise.clear();
118    }
119
120    /// The steps covering scan `scan`, in the order they were taken: the
121    /// last ones.
122    pub(crate) fn steps_for_scan(&self, scan: usize) -> &[DataStep] {
123        let first = self.steps.partition_point(|step| step.scans() <= scan);
124        &self.steps[first..]
125    }
126
127    /// The noise to add to scan `scan`'s rows, in the order it was asked
128    /// for: the applications of its corruptions, all of which add, which
129    /// are the last ones taken.
130    pub(crate) fn noise_for_scan(&self, scan: usize) -> &[NoiseApplication] {
131        let start = self
132            .steps_for_scan(scan)
133            .iter()
134            .find_map(|step| match step {
135                DataStep::Corrupt { applications, .. } => Some(applications.start),
136                DataStep::Predict { .. } => None,
137            })
138            .unwrap_or(self.noise.len());
139        &self.noise[start..]
140    }
141
142    /// The predict steps covering scan `scan`, as indices of
143    /// [`Self::predictions`], in the order they were taken: their
144    /// predictions add. They are the last ones.
145    pub(crate) fn predictions_for_scan(&self, scan: usize) -> Range<usize> {
146        let all = self.predictions().count();
147        let covering = self
148            .steps_for_scan(scan)
149            .iter()
150            .filter(|step| matches!(step, DataStep::Predict { .. }))
151            .count();
152        all - covering..all
153    }
154
155    /// Every predict step's prepared model and the scans it covers, in the
156    /// order they were taken.
157    pub(crate) fn predictions(&self) -> impl Iterator<Item = (&Prediction, usize)> {
158        self.steps.iter().filter_map(|step| match step {
159            DataStep::Predict { prediction, scans } => Some((prediction.as_ref(), *scans)),
160            DataStep::Corrupt { .. } => None,
161        })
162    }
163}

165#[cfg(test)]
166mod tests {
167    use super::*;
168
169    fn noise(seed: i32, ordinal: u32) -> NoiseApplication {
170        NoiseApplication {
171            seed,
172            sigma_jy: 0.5,
173            ordinal,
174        }
175    }
176
177    #[test]
178    fn a_step_covers_only_the_scans_observed_before_it() {
179        let mut plan = DataPlan::default();
180        plan.corrupt([noise(1, 0)], 1);
181        plan.corrupt([noise(1, 1), noise(2, 0)], 3);
182        assert_eq!(
183            plan.noise_for_scan(0),
184            [noise(1, 0), noise(1, 1), noise(2, 0)]
185        );
186        assert_eq!(plan.noise_for_scan(2), [noise(1, 1), noise(2, 0)]);
187        assert_eq!(plan.noise_for_scan(3), []);
188        assert_eq!(plan.steps_for_scan(1).len(), 1);
189        assert!(plan.predictions_for_scan(0).is_empty());
190        plan.clear();
191        assert_eq!(plan.noise_for_scan(0), []);
192    }
193
194    #[test]
195    fn interleaved_corruptions_share_their_applications() {
196        let mut plan = DataPlan::default();
197        for scans in 1..=4 {
198            plan.corrupt([noise(1, scans as u32 - 1)], scans);
199        }
200        for scan in 0..4 {
201            let expected = (scan as u32..4).map(|ordinal| noise(1, ordinal));
202            assert!(plan.noise_for_scan(scan).iter().copied().eq(expected));
203        }
204        // Each scan's noise is the end of the one list.
205        let all = plan.noise_for_scan(0).as_ptr_range();
206        assert_eq!(plan.noise_for_scan(3).as_ptr_range().end, all.end);
207        assert_eq!(plan.noise_for_scan(4), []);
208    }
