2//! What opening a processing set holds: projected from each dataset's
3//! array shapes and element sizes before any is decoded, so that the
4//! opening can be granted first ([`super::Msv4Observation::open_granted`]),
5//! and counted by capacity once it is open
6//! ([`super::Msv4Observation::held_bytes`]).
7//!
8//! A dataset's opening decodes its per-time coordinates (`time`,
9//! `field_name`, `scan_name`), its baselines and frequencies, its field
10//! datasets and their ephemerides, and keeps the times, each time's scan
11//! and, for each data group, each time's field. The names are
12//! `fixed_length_utf32`, so their decoded size is known from the array's
13//! data type. Array handles hold their metadata, which the count takes
14//! from its serialized size.
15

27/// What decoding holds besides the arrays it decodes, at most: the codecs'
28/// contexts and the metadata documents parsed while a dataset opens.
29pub(super) const WORKSPACE_BYTES: usize = 2 << 20;
30
31/// What a dataset holds beyond its per-time, per-baseline and per-channel
32/// arrays, at most, as projected before it opens: its catalogue entries,
33/// names and array handles. [`dataset_bytes`] counts them once open.
34const DATASET_BYTES: usize = 256 << 10;
35
36/// What an array handle holds beside its metadata document's serialized
37/// bytes: its codec chain and chunk grid.
38const HANDLE_BYTES: usize = 4 << 10;
39
40/// The heap a handle's parsed metadata takes per byte of its document.
41const METADATA_FACTOR: usize = 4;
42
43/// A dataset's opening, projected: what it holds once open, and what
44/// decoding it holds beyond that while it opens.
45#[derive(Clone, Copy, Debug, Default)]
46pub(super) struct Projection {
47    pub(super) held: usize,
48    pub(super) transient: usize,
49}
50
51/// The opening of the processing set whose datasets `projections` are: what
52/// it holds once open and, at its peak, while the largest dataset decodes.
53pub(super) fn opening_bytes(projections: &[Projection]) -> usize {
54    projections.iter().map(|each| each.held).sum::<usize>()
55        + projections
56            .iter()
57            .map(|each| each.transient)
58            .max()
59            .unwrap_or(0)
60        + WORKSPACE_BYTES
61}

63/// Dataset `path`'s opening, projected from its metadata. A dataset whose
64/// metadata does not state a shape projects nothing for it: opening refuses
65/// it, with the reason, before decoding it.
66pub(super) fn project(store: &Arc<FilesystemStore>, path: &str) -> Projection {
67    let length = |name: &str| shape(store, &format!("{path}/{name}"), 0);
68    let times = length("time");
69    let baselines = length("baseline_id");
70    let channels = length("frequency");
71    let names = string_bytes(store, &format!("{path}/field_name"), times);
72    let scans = has_node(store, &format!("{path}/scan_name")).unwrap_or(false);
73    let scan_names = string_bytes(store, &format!("{path}/scan_name"), times);
74    let records = Group::open(store.clone(), path)
75        .ok()
76        .and_then(|group| data_groups(group.attributes()).ok())
77        .unwrap_or_default();
78    let mut field_datasets: Vec<&str> = records
79        .iter()
80        .map(|record| record.field_and_source.as_str())
81        .collect();
82    field_datasets.sort_unstable();
83    field_datasets.dedup();
84    let (mut ephemeris_held, mut ephemeris_transient) = (0, 0);
85    for name in field_datasets {
86        let dataset = format!("{path}/{name}");
87        let moving = Group::open(store.clone(), &dataset).is_ok_and(|group| {
88            text(group.attributes(), "type", &dataset).is_ok_and(|kind| kind == EPHEMERIS_DATASET)
89        });
90        let fields = shape(store, &format!("{dataset}/field_name"), 0);
91        ephemeris_transient += 2 * string_bytes(store, &format!("{dataset}/field_name"), fields);
92        if moving {
93            let samples = shape(store, &format!("{dataset}/time_ephemeris"), times);
94            // Times, directions (and their pairs), distances and radial
95            // velocities; and each time's phase centre while it is read.
96            ephemeris_held += samples * 5 * size_of::<f64>();
97            ephemeris_transient += samples * 7 * size_of::<f64>() + times * 2 * size_of::<f64>();
98        }
99    }
100    let per_time = size_of::<f64>()
101        + usize::from(scans) * size_of::<ScanId>()
102        + records.len().max(1) * size_of::<FieldId>();
103    Projection {
104        held: times * per_time
105            + baselines * size_of::<OrderedBaseline>()
106            // The partition's frequencies and widths, and its window's.
107            + channels * 4 * size_of::<f64>()
108            + ephemeris_held
109            + DATASET_BYTES,
110        // The time coordinate as decoded beside its copy, and the names.
111        transient: times * size_of::<f64>() + names + scan_names + ephemeris_transient,
112    }
113}

115/// The length of the one-dimensional array at `path`, or `missing` when it
116/// has none.
117fn shape(store: &Arc<FilesystemStore>, path: &str, missing: usize) -> usize {
118    open_array(store, path)
119        .ok()
120        .and_then(|array| array.shape().first().copied())
121        .and_then(|length| usize::try_from(length).ok())
122        .unwrap_or(missing)
123}

125/// What decoding `elements` names of the `fixed_length_utf32` array at
126/// `path` holds at its peak: the stored code units, each name's `Vec<char>`
127/// and its `String`.
128fn string_bytes(store: &Arc<FilesystemStore>, path: &str, elements: usize) -> usize {
129    let Some(fixed) = open_array(store, path)
130        .ok()
131        .and_then(|array| array.data_type().fixed_size())
132    else {
133        return 0;
134    };
135    elements * (3 * fixed + size_of::<Vec<char>>() + size_of::<String>())
136}

138/// What array handle `array` holds: the handle, its metadata as parsed, at
139/// [`METADATA_FACTOR`] times its document's serialized bytes, and its codec
140/// chain.
141pub(super) fn handle_bytes(array: &Array<FilesystemStore>) -> usize {
142    size_of::<Array<FilesystemStore>>()
143        + HANDLE_BYTES
144        + METADATA_FACTOR
145            * serde_json::to_vec(&array.metadata()).map_or(0, |document| document.len())
146}

148/// What `dataset` holds beyond its catalogue entries, by capacity.
149pub(super) fn dataset_bytes(dataset: &Dataset) -> usize {
150    dataset.path.capacity()
151        + dataset.times.capacity() * size_of::<f64>()
152        + dataset
153            .scans
154            .as_ref()
155            .map_or(0, |scans| scans.capacity() * size_of::<ScanId>())
156        + [&dataset.centroid, &dataset.exposure]
157            .into_iter()
158            .flatten()
159            .map(handle_bytes)
160            .sum::<usize>()
161        + dataset.groups.capacity() * size_of::<GroupArrays>()
162        + dataset.groups.iter().map(group_bytes).sum::<usize>()
163        + dataset.antennas.capacity() * size_of::<(String, casa_observation::AntennaId)>()
164        + dataset
165            .antennas
166            .iter()
167            .map(|(name, _)| name.capacity())
168            .sum::<usize>()
169}

171/// What data group `group` holds: its record's names, each time's field
172/// and its four arrays' handles.
173fn group_bytes(group: &GroupArrays) -> usize {
174    let record = &group.record;
175    [
176        &record.name,
177        &record.correlated_data,
178        &record.flag,
179        &record.weight,
180        &record.uvw,
181        &record.field_and_source,
182    ]
183    .into_iter()
184    .map(String::capacity)
185    .sum::<usize>()
186        + group.fields.capacity() * size_of::<FieldId>()
187        + [&group.values, &group.flags, &group.weights, &group.uvw]
188            .into_iter()
189            .map(handle_bytes)
190            .sum::<usize>()
191}
