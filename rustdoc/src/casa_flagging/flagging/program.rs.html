54/// Validate CASA's allowed time and channel averaging in a list of agents.
55///
56/// Each config provides the mode, channel-average switch, and time-average
57/// switch, in list order.
58///
59/// # Errors
60/// An agent requests averaging in an unsupported mode or alongside an agent
61/// that cannot run with an averaged automatic agent.
62pub fn validate_calibration_flag_averaging<'a>(
63    configs: impl Iterator<Item = (&'a str, bool, bool)> + Clone,
64) -> Result<(), FlaggingError> {
65    validate_flag_averaging(configs, false)
66}

68/// Check whether a MeasurementSet list combines averaging with an unsupported mode.
69///
70/// # Errors
71/// An agent requests averaging in an unsupported mode or list.
72pub fn validate_ms_flag_averaging<'a>(
73    configs: impl Iterator<Item = (&'a str, bool, bool)> + Clone,
74) -> Result<(), FlaggingError> {
75    validate_flag_averaging(configs, true)
76}

78fn validate_flag_averaging<'a>(
79    configs: impl Iterator<Item = (&'a str, bool, bool)> + Clone,
80    shared_averaging: bool,
81) -> Result<(), FlaggingError> {
82    let automatic = |mode: &str| matches!(mode, "clip" | "tfcrop" | "rflag");
83    let incompatible = configs
84        .clone()
85        .map(|(mode, _, _)| mode)
86        .find(|mode| !automatic(mode) && !matches!(*mode, "extend" | "display" | "antint"));
87    let multiple_automatic = configs
88        .clone()
89        .filter(|(mode, _, _)| automatic(mode))
90        .take(2)
91        .count()
92        > 1;
93    for (mode, channelavg, timeavg) in configs {
94        if !(channelavg || timeavg) {
95            continue;
96        }
97        let (timeavg, channelavg) = (u8::from(timeavg), u8::from(channelavg));
98        if !automatic(mode) {
99            return Err(FlaggingError::CalibrationAgentConfiguration(format!(
100                "Cannot use timeavg or channelavg in mode {mode}. timeavg={timeavg}, channelavg={channelavg}. These averaging options are only allowed for auto-flagging methods. Refusing this configuration."
101            )));
102        }
103        if let Some(other) = incompatible {
104            return Err(FlaggingError::CalibrationAgentConfiguration(format!(
105                "Cannot use {mode} mode with timeavg=True or channelavg=True and additional modes other than extend, display, and antint. timeavg={timeavg}, channelavg={channelavg}, and the following mode has been set up: {other}. Refusing to accept this configuration."
106            )));
107        }
108        if multiple_automatic && !shared_averaging {
109            return Err(FlaggingError::CalibrationAgentConfiguration(
110                "Averaging with multiple automatic flagging agents requires a shared averaged visibility iterator (#978).".into(),
111            ));
112        }
113    }
114    Ok(())
115}

117pub(super) fn run_ms_single(
118    ms: &mut MeasurementSet,
119    request: &FlagDataRequest,
120    persist: bool,
121    model_factory: Option<FlagDataModelFactory>,
122) -> Result<FlagDataReport, FlaggingError> {
123    let ms_path = display_ms_path(ms);
124    tracing::info!(
125        path = %ms_path,
126        mode = ?request.mode,
127        action = ?request.action,
128        "flagdata started"
129    );
130    request.check_cancelled()?;
131    // The CASA controller saves the working flags before selection. Typed
132    // callers keep their existing selection-first backup behavior.
133    let early_backup = persist
134        && request.flagbackup
135        && request.casa_row_selection
136        && request.mode != FlagDataMode::Summary;
137    let mut backup_version = early_backup
138        .then(|| save_next_casa_flag_backup(ms, "flagdata"))
139        .transpose()?;
140    request.check_cancelled()?;
141    let time_selection = FlagTimeSelection::Global;
142    let selection = selected_rows(ms, request, time_selection)?;
143    let selected = selection.rows;
144    if selected.is_empty() {
145        tracing::warn!(path = %ms_path, mode = ?request.mode, "flagdata selected no rows");
146    }
147    let channel_selections = flagging_channels(ms, request, selection.channel_ranges)?;
148    let correlations = resolve_correlations(ms, request)?;
149    if !request.display_column_is_scalar {
150        return Err(FlaggingError::DisplayColumnVector);
151    }
152    if request.mode == FlagDataMode::Summary {
153        return summary::report(
154            ms,
155            &selected,
156            channel_selections.as_ref(),
157            correlations.as_ref(),
158            request.summary_options,
159            request.cancellation.as_ref(),
160        );
161    }
162    if persist && request.flagbackup && !early_backup && !selected.is_empty() {
163        request.check_cancelled()?;
164        backup_version = Some(save_next_flag_backup(ms, "flagdata")?);
165    }
166    let selection = AgentSelection {
167        rows: selected,
168        channels: channel_selections,
169        correlations,
170        averaged_time: None,
171        averaged_uv: None,
172    };
173    let changes =
174        stage_selected_ms_request(ms, &selection, request, !persist, model_factory, None)?;
175    request.check_cancelled()?;
176    let mut report_cancellation = request.cancellation.as_ref();
177    if persist && !changes.changed_row_indices.is_empty() {
178        persist_changed_flags(ms, &changes, &ms_path, report_cancellation)?;
179        report_cancellation = None;
180    }
181    let report = report_after(
182        ms,
183        request,
184        &selection.rows,
185        changes,
186        backup_version,
187        time_selection,
188        report_cancellation,
189    )?;
190    tracing::info!(
191        path = %ms_path,
192        mode = ?report.mode,
193        selected_rows = report.selected_rows,
194        changed_rows = report.changed_rows,
195        changed_samples = report.changed_samples,
196        "flagdata completed"
197    );
198    Ok(report)
199}

201/// Execute a sequence of MeasurementSet flagging agents on one opened table view.
202/// Earlier agents' staged edits are visible to later summary and propagation
203/// passes. CASA's antenna-integration prepass reads the original input flags.
204/// When `write_flags` is `false`, staged flags are discarded without mutating
205/// the MeasurementSet on disk; when `true`, changed rows are written once at
206/// the end. When `local_selection` is `true` (multiple commands in CASA's
207/// `mode='list'`), an agent whose local selection fails is skipped (`None`)
208/// rather than aborting the list; single-command runs propagate selection errors.
209/// Use the same cancellation token for the program's requests.
210pub fn flagdata_ms_program(
211    mut ms: MeasurementSet,
212    requests: &[FlagDataRequest],
213    write_flags: bool,
214    local_selection: bool,
215    model_factory: Option<FlagDataModelFactory>,
216) -> Result<Vec<Option<FlagDataReport>>, FlaggingError> {
217    let cancellation = requests
218        .iter()
219        .find_map(|request| request.cancellation.as_ref());
220    check_cancelled(cancellation)?;
221    if is_calibration_table(&ms) {
222        return Err(FlaggingError::MutateFlags {
223            path: display_ms_path(&ms),
224            reason: "MeasurementSet flag commands cannot run on a calibration table".to_owned(),
225        });
226    }
227    let ms_path = display_ms_path(&ms);
228    let time_selection = if local_selection {
229        let has_timerange = requests.iter().any(|request| {
230            request.casa_row_selection
231                && request.selection.selectdata
232                && request
233                    .selection
234                    .timerange
235                    .as_deref()
236                    .is_some_and(|text| !text.is_empty() && !text.starts_with('\0'))
237        });
238        let initial_default_row = if has_timerange {
239            first_unflagged_time_row(&ms).map_err(|source| FlaggingError::MutateFlags {
240                path: ms_path.clone(),
241                reason: format!("select rows: {source}"),
242            })?
243        } else {
244            0
245        };
246        FlagTimeSelection::Agent {
247            initial_default_row,
248        }
249    } else {
250        FlagTimeSelection::GlobalAndAgent
251    };
252    let mut shared = if let Some(config) = SharedAveraging::from_requests(requests.iter()) {
253        let time_chunk = requests
254            .iter()
255            .filter_map(|request| request.time_chunk)
256            .max_by(f64::total_cmp);
257        let combinescans = requests.iter().any(|request| request.combinescans);
258        Some(SharedFlagView::new(
259            &ms,
260            config,
261            time_chunk,
262            combinescans,
263            &ms_path,
264            cancellation,
265        )?)
266    } else {
267        None
268    };
269    // Scan the supplied FLAG view before any list agent stages changes.
270    // Keep row indices for later application, never copies of FLAG arrays.
271    let antint_plans: Vec<_> = requests
272        .iter()
273        .enumerate()
274        .map(|(index, request)| {
275            check_cancelled(cancellation)?;
276            Ok((shared.is_none()
277                && index > 0
278                && local_selection
279                && request.casa_row_selection
280                && request.mode == FlagDataMode::Antint)
281                .then(|| plan_antint(&ms, request, &ms_path, time_selection)))
282        })
283        .collect::<Result<_, FlaggingError>>()?;
284    let mut reports = Vec::with_capacity(requests.len());
285    let mut touched = BTreeSet::new();
286    let mut changed_flag_row_rows = 0usize;
287    for (request, antint_plan) in requests.iter().zip(antint_plans) {
288        check_cancelled(cancellation)?;
289        match stage_ms_program_request(
290            &mut ms,
291            request,
292            ProgramAgentContext {
293                local_selection,
294                calculate: !write_flags,
295                model_factory,
296                time_selection,
297            },
298            antint_plan,
299            shared.as_mut(),
300        ) {
301            Ok(Some((report, changed_rows, changed_flag_rows))) => {
302                touched.extend(changed_rows);
303                changed_flag_row_rows += changed_flag_rows;
304                reports.push(Some(report));
305            }
306            Ok(None)
307            | Err(
308                FlaggingError::MissingClipColumn
309                | FlaggingError::MissingTfcropColumn
310                | FlaggingError::MissingRflagColumn
311                | FlaggingError::MissingCorrelationSummary,
312            ) => reports.push(None),
313            Err(error) => return Err(error),
314        }
315    }
316    check_cancelled(cancellation)?;
317    if let Some(view) = shared
318        && write_flags
319    {
320        let mut changes = view.flush(&mut ms)?;
321        touched.extend(std::mem::take(&mut changes.changed_row_indices));
322        changed_flag_row_rows += changes.changed_flag_row_rows;
323    }
324    check_cancelled(cancellation)?;
325    if write_flags && !touched.is_empty() {
326        let changes = ChangeSet {
327            changed_row_indices: touched.into_iter().collect(),
328            changed_flag_row_rows,
329            ..ChangeSet::default()
330        };
331        persist_changed_flags(&mut ms, &changes, &ms_path, cancellation)?;
332    }
333    Ok(reports)
334}

336fn plan_antint(
337    ms: &MeasurementSet,
338    request: &FlagDataRequest,
339    path: &str,
340    time_selection: FlagTimeSelection,
341) -> Result<Option<AntintPlan>, FlaggingError> {
342    let Some(selection) = select_agent(ms, request, true, time_selection, false)? else {
343        return Ok(None);
344    };
345    if selection
346        .correlations
347        .as_ref()
348        .is_some_and(|selection| !selection.has_active_products())
349    {
350        return Ok(None);
351    }
352    let affected = antint::affected_rows(
353        ms,
354        &selection.rows,
355        &request.antint,
356        path,
357        request.cancellation.as_ref(),
358    )?;
359    Ok(Some(AntintPlan {
360        selection,
361        affected,
362    }))
363}

365fn select_agent(
366    ms: &MeasurementSet,
367    request: &FlagDataRequest,
368    local_selection: bool,
369    time_selection: FlagTimeSelection,
370    shared_time: bool,
371) -> Result<Option<AgentSelection>, FlaggingError> {
372    let averages_time = shared_time
373        || request.clip_time_bin.is_some()
374            && matches!(
375                request.mode,
376                FlagDataMode::Clip | FlagDataMode::Tfcrop | FlagDataMode::Rflag
377            );
378    let mut averaged_time = None;
379    let mut averaged_uv = None;
380    let mut input = Cow::Borrowed(request);
381    if request.casa_row_selection
382        && averages_time
383        && request.selection.selectdata
384        && request
385            .selection
386            .timerange
387            .as_deref()
388            .is_some_and(|text| !text.is_empty() && !text.starts_with('\0'))
389        && let FlagTimeSelection::Agent {
390            initial_default_row,
391        } = time_selection
392    {
393        match compile_flagdata_agent_time(ms, &request.selection, initial_default_row) {
394            Ok(filter) => averaged_time = filter,
395            Err(_) if local_selection => return Ok(None),
396            Err(source) => {
397                return Err(FlaggingError::MutateFlags {
398                    path: display_ms_path(ms),
399                    reason: format!("select rows: {source}"),
400                });
401            }
402        }
403        // The local agent filters the output TIME of the averaging iterator.
404        // Every contributing integration must still reach the averaging step.
405        input.to_mut().selection.timerange = None;
406    }
407    if shared_time
408        && request.selection.selectdata
409        && let Some(text) = request
410            .selection
411            .uvrange
412            .as_deref()
413            .map(flagdata_c_string)
414            .filter(|text| !text.is_empty())
415    {
416        match UvDistanceSelection::parse(ms, text) {
417            Ok(filter) => averaged_uv = Some(filter),
418            Err(_) if local_selection => return Ok(None),
419            Err(source) => {
420                return Err(FlaggingError::MutateFlags {
421                    path: display_ms_path(ms),
422                    reason: format!("select rows: {source}"),
423                });
424            }
425        }
426        input.to_mut().selection.uvrange = None;
427    }
428    let input_time_selection = if averages_time {
429        FlagTimeSelection::Global
430    } else {
431        time_selection
432    };
433    let selection = match selected_rows(ms, &input, input_time_selection) {
434        Ok(selection) => selection,
435        Err(_) if local_selection => return Ok(None),
436        Err(error) => return Err(error),
437    };
438    let channels = match flagging_channels(ms, request, selection.channel_ranges) {
439        Ok(channels) => channels,
440        Err(_) if local_selection => return Ok(None),
441        Err(error) => return Err(error),
442    };
443    let correlations = resolve_correlations(ms, request)?;
444    if !request.display_column_is_scalar {
445        return Err(FlaggingError::DisplayColumnVector);
446    }
447    Ok(Some(AgentSelection {
448        rows: selection.rows,
449        channels,
450        correlations,
451        averaged_time,
452        averaged_uv,
453    }))
454}

463fn stage_ms_program_request(
464    ms: &mut MeasurementSet,
465    request: &FlagDataRequest,
466    context: ProgramAgentContext,
467    antint_plan: Option<Result<Option<AntintPlan>, FlaggingError>>,
468    mut shared: Option<&mut SharedFlagView>,
469) -> Result<Option<(FlagDataReport, Vec<usize>, usize)>, FlaggingError> {
470    let ProgramAgentContext {
471        local_selection,
472        calculate,
473        model_factory,
474        time_selection,
475    } = context;
476    let (selection, affected) = if let Some(plan) = antint_plan {
477        let Some(plan) = plan? else {
478            return Ok(None);
479        };
480        (plan.selection, Some(plan.affected))
481    } else {
482        let Some(selection) = select_agent(
483            ms,
484            request,
485            local_selection,
486            time_selection,
487            shared.as_ref().is_some_and(|view| view.averaged_in_time()),
488        )?
489        else {
490            return Ok(None);
491        };
492        (selection, None)
493    };
494    if request.mode == FlagDataMode::Summary {
495        return summary::report(
496            ms,
497            &selection.rows,
498            selection.channels.as_ref(),
499            selection.correlations.as_ref(),
500            request.summary_options,
501            request.cancellation.as_ref(),
502        )
503        .map(|report| Some((report, Vec::new(), 0)));
504    }
505    if selection
506        .correlations
507        .as_ref()
508        .is_some_and(|selection| !selection.has_active_products())
509    {
510        return Ok(None);
511    }
512    let mut changes = if let Some(affected) = affected {
513        antint::apply_affected(
514            ms,
515            affected,
516            selection.channels.as_ref(),
517            selection.correlations.as_ref(),
518            request.cancellation.as_ref(),
519        )?
520    } else {
521        stage_selected_ms_request(
522            ms,
523            &selection,
524            request,
525            calculate,
526            model_factory,
527            shared.as_deref_mut(),
528        )?
529    };
530    if changes.agent_skipped {
531        return Ok(None);
532    }
533    if let Some(view) = shared {
534        if request.extendflags && matches!(request.mode, FlagDataMode::Tfcrop | FlagDataMode::Rflag)
535        {
536            let extension = view.extend_implicit(&display_ms_path(ms))?;
537            changes.changed_samples += extension.changed_samples;
538            changes
539                .changed_row_indices
540                .extend(extension.changed_row_indices);
541            changes.changed_row_indices.sort_unstable();
542            changes.changed_row_indices.dedup();
543            changes.changed_rows = changes.changed_row_indices.len();
544        }
545        let (selected_rows, summary) = view.summarize(selection.shared());
546        changes.selected_rows = Some(selected_rows);
547        changes.summary = Some(summary);
548    }
549    let changed_rows = std::mem::take(&mut changes.changed_row_indices);
550    let changed_flag_rows = changes.changed_flag_row_rows;
551    let report = report_after(
552        ms,
553        request,
554        &selection.rows,
555        changes,
556        None,
557        time_selection,
558        request.cancellation.as_ref(),
559    )?;
560    Ok(Some((report, changed_rows, changed_flag_rows)))
561}

563fn resolve_correlations(
564    ms: &MeasurementSet,
565    request: &FlagDataRequest,
566) -> Result<Option<CorrelationSelection>, FlaggingError> {
567    let correlation = request
568        .correlation
569        .as_deref()
570        .filter(|expression| !expression.is_empty() && !expression.starts_with('\0'))
571        .or_else(|| (request.mode == FlagDataMode::Clip).then_some(""));
572    let auto_correlations = matches!(request.mode, FlagDataMode::Tfcrop | FlagDataMode::Rflag)
573        && request.casa_row_selection;
574    match correlation.filter(|_| !auto_correlations) {
575        Some(expression) => {
576            CorrelationSelection::read(ms, request.mode, request.data_column, expression)
577        }
578        None => Ok(None),
579    }
580}

582fn select_autocorrelations(
583    ms: &MeasurementSet,
584    rows: &[usize],
585    path: &str,
586) -> Result<Vec<usize>, FlaggingError> {
587    let main = ms.main_table();
588    let first = load_i32_cells(main, rows, "ANTENNA1", path)?;
589    let second = load_i32_cells(main, rows, "ANTENNA2", path)?;
590    let autocorrelations: Vec<_> = rows
591        .iter()
592        .copied()
593        .zip(first)
594        .zip(second)
595        .filter_map(|((row, first), second)| (first == second).then_some(row))
596        .collect();
597    let Some(processor) = ms
598        .subtable(SubtableId::Processor)
599        .filter(|table| table.row_count() > 0 && !autocorrelations.is_empty())
600    else {
601        return Ok(autocorrelations);
602    };
603    let types = processor
604        .column_accessor("TYPE")
605        .and_then(|column| column.scalar_cells_owned())
606        .map_err(|error| FlaggingError::MutateFlags {
607            path: path.to_string(),
608            reason: error.to_string(),
609        })?;
610    let processor_ids = load_i32_cells(main, &autocorrelations, "PROCESSOR_ID", path)?;
611    Ok(autocorrelations
612        .into_iter()
613        .zip(processor_ids)
614        .filter_map(|(row, id)| {
615            let index = usize::try_from(id).ok()?;
616            matches!(types.get(index), Some(Some(ScalarValue::String(kind))) if kind == "CORRELATOR")
617                .then_some(row)
618        })
619        .collect())
620}

622fn stage_selected_ms_request(
623    ms: &mut MeasurementSet,
624    selection: &AgentSelection,
625    request: &FlagDataRequest,
626    calculate: bool,
627    model_factory: Option<FlagDataModelFactory>,
628    shared: Option<&mut SharedFlagView>,
629) -> Result<ChangeSet, FlaggingError> {
630    request.check_cancelled()?;
631    let selected_rows = &selection.rows;
632    let channel_selections = selection.channels.as_ref();
633    let correlations = selection.correlations.as_ref();
634    let mut averaging = AveragedAgentInput {
635        time: selection.averaged_time.as_ref(),
636        uv: selection.averaged_uv.as_ref(),
637        view: shared,
638    };
639    let ms_path = display_ms_path(ms);
640    // The manual agent filters baselines after full MS selection, including
641    // the rows that supply the default date for a partial time selector.
642    let autocorrelation_rows = if request.autocorr
643        && request.mode == FlagDataMode::Manual
644        && request.action == FlagDataAction::Flag
645    {
646        Some(select_autocorrelations(ms, selected_rows, &ms_path)?)
647    } else {
648        None
649    };
650    if let Some(view) = averaging.view.as_deref_mut() {
651        match request.mode {
652            FlagDataMode::Extend => {
653                return view.extend(selection.shared(), request, &ms_path);
654            }
655            FlagDataMode::Antint => {
656                return view.antint(ms, selection.shared(), request, &ms_path);
657            }
658            _ => {}
659        }
660    }
661    match request.mode {
662        FlagDataMode::Manual => set_selected_flags(
663            ms,
664            autocorrelation_rows.as_deref().unwrap_or(selected_rows),
665            channel_selections,
666            correlations,
667            request.action.value(),
668            selected_flag_row_policy(correlations, SelectedFlagRowPolicy::ChangedSamples),
669            request.cancellation.as_ref(),
670        ),
671        FlagDataMode::Clip => apply_clip_flags(
672            ms,
673            selected_rows,
674            channel_selections,
675            correlations,
676            request,
677            model_factory,
678            averaging,
