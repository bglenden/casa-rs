2//! How an image's mask regions name their mask tables.
3//!
4//! Each mask of a persistent image is an `LCPagedMask` region record in the
5//! image's `masks` keyword. Its `mask` field is a table keyword naming the
6//! mask table, a subtable of the image named after the mask. casacore's
7//! `PagedImage` stores it relative to the image (`././mask0`, see
8//! [`casa_tables::strip_directory`]), and so does casa-rs, so the image can be
9//! moved or copied as a directory.
10//!
11//! casa-rs builds before #715 stored the image's absolute path at the time of
12//! writing. Such an image still reads in place, and a save in place stores the
13//! relative form; once moved, it needs re-running (or, for CASA, a symlink at
14//! the old path).
15
16use casa_types::{RecordValue, ScalarValue, Value};
17
18/// The field of a mask region record that names its mask table.
19const MASK_TABLE_FIELD: &str = "mask";
20
21/// The stored reference to the subtable `name` inside the image: casacore's
22/// form for a table inside its owner, as `PagedImage` writes its `logtable`
23/// and mask tables.
24pub(crate) fn subtable_reference(name: &str) -> String {
25    format!("././{name}")
26}
27
28/// Points every mask region of `masks` at the mask table of the same name
29/// inside the image. Returns whether any reference changed.
30///
31/// Used before an image is written to a new path, where its masks are written
32/// as subtables named after them.
33pub(crate) fn point_mask_references_inside(masks: &mut RecordValue) -> bool {
34    let mut changed = false;
35    for field in masks.fields_mut() {
36        let Value::Record(region) = &mut field.value else {
37            continue;
38        };
39        if !matches!(
40            region.get(MASK_TABLE_FIELD),
41            Some(Value::TableRef(_)) | Some(Value::Scalar(ScalarValue::String(_)))
42        ) {
43            continue;
44        }
45        let reference = Value::TableRef(subtable_reference(&field.name));
46        if region.get(MASK_TABLE_FIELD) != Some(&reference) {
47            region.upsert(MASK_TABLE_FIELD, reference);
48            changed = true;
49        }
50    }
51    changed
52}

54#[cfg(test)]
55mod tests {
56    use super::*;
57    use casa_types::RecordField;
58
59    fn region(stored: Value) -> Value {
60        Value::Record(RecordValue::new(vec![RecordField::new(
61            MASK_TABLE_FIELD,
62            stored,
63        )]))
64    }
65
66    fn stored(masks: &RecordValue, name: &str) -> Option<Value> {
67        let Some(Value::Record(region)) = masks.get(name) else {
68            panic!("no region {name}");
69        };
70        region.get(MASK_TABLE_FIELD).cloned()
71    }
72
73    #[test]
74    fn writing_to_a_new_path_points_every_region_inside() {
75        let mut masks = RecordValue::new(vec![
76            RecordField::new("a", region(Value::table_ref("/old/x.image/a"))),
77            RecordField::new("b", region(Value::table_ref("././b"))),
78            RecordField::new(
79                "c",
80                region(Value::Scalar(ScalarValue::String("x.image/c".into()))),
81            ),
82        ]);
83        assert!(point_mask_references_inside(&mut masks));
84        for name in ["a", "b", "c"] {
85            assert_eq!(
86                stored(&masks, name),
87                Some(Value::table_ref(format!("././{name}")))
88            );
89        }
90        assert!(!point_mask_references_inside(&mut masks));
91    }
92
93    #[test]
94    fn a_temporary_image_saved_with_a_mask_keeps_it_after_a_move() {
95        use crate::{PagedImage, TempImage, TempStoragePolicy};
96        use casa_coordinates::CoordinateSystem;
97        use ndarray::{ArrayD, IxDyn};
98
99        let dir = tempfile::tempdir().unwrap();
100        let mask = ArrayD::from_shape_vec(IxDyn(&[2, 2]), vec![true, false, true, true]).unwrap();
101        let mut temp = TempImage::<f32>::new(
102            vec![2, 2],
103            CoordinateSystem::new(),
104            TempStoragePolicy::Memory,
105        )
106        .unwrap();
107        temp.put_mask("mask0", &mask).unwrap();
108        temp.set_default_mask("mask0").unwrap();
109        drop(temp.save_as(dir.path().join("temp.image")).unwrap());
110
111        let moved = dir.path().join("moved").join("temp.image");
112        std::fs::create_dir_all(moved.parent().unwrap()).unwrap();
113        std::fs::rename(dir.path().join("temp.image"), &moved).unwrap();
114        let reopened = PagedImage::<f32>::open(&moved).unwrap();
115        assert_eq!(reopened.get_mask().unwrap(), Some(mask));
116    }
