2//! Limited `bandpass` support for the first `B Jones` solver wave.
3//!
4//! This module intentionally implements only a narrow, testable slice:
5//!
6//! - `bandtype='B'`
7//! - `solint='inf'`
8//! - explicit reference antenna
9//! - prior gain-table preapply
10//! - point-source Stokes-I sky model (`smodel=[I,0,0,0]`)
11//!
12//! The acceptance contract is downstream behavior: the resulting caltable
13//! should be CASA-compatible on disk and should yield corrected visibilities
14//! close to CASA's own `bandpass` when applied after the same prior gains.

236        /// Error context.
237        reason: String,

241/// Solve a limited `bandpass` request from an already-open MeasurementSet.
242pub(crate) fn solve_bandpass(
243    ms: &MeasurementSet,
244    request: &BandpassSolveRequest,
245) -> Result<BandpassSolveReport, BandpassSolveError> {
246    validate_bandpass_request(request)?;
247    let context = prepare_calibration_solve(
248        ms,
249        &request.selection,
250        &request.refant,
251        &request.prior_calibration_tables,
252        request.parang,
253    )?;
254    let groups = build_solve_groups(
255        ms,
256        &context.rows,
257        context.preapplied_rows.as_ref(),
258        SolveGroupOptions {
259            gain_type: GainType::G,
260            model_source: GainSolveModelSource::PointSource,
261            stokes_i: request.smodel[0],
262            solve_interval: GainSolveInterval::Infinite,
263            combine: GainSolveCombine {
264                scans: request.combine.scans,
265                fields: request.combine.fields,
266            },
267            channelized: true,
268        },
269    )?;
270
271    if groups.is_empty() {
272        return Err(BandpassSolveError::EmptySelection);
273    }
274
275    let mut solution_rows = Vec::new();
276    for mut group in groups.into_values() {
277        group.finalize_for_solve(GainSolveMode::AmplitudePhase);
278        solution_rows.extend(solve_bandpass_group(
279            &group,
280            &context.available_antennas,
281            context.refant_id,
282            request.combine,
283            request.normalize_average_amplitude,
284        ));
285    }
286    if solution_rows.is_empty() {
287        return Err(BandpassSolveError::NoSolutions);
288    }
289    match request.band_type {
290        BandpassType::B => write_bandpass_caltable(ms, request, context.refant_id, &solution_rows),
291        BandpassType::BPoly => write_bpoly_caltable(ms, request, context.refant_id, &solution_rows),
292    }
293}

295fn validate_bandpass_request(request: &BandpassSolveRequest) -> Result<(), BandpassSolveError> {
296    validate_smodel(request.smodel)?;
297    if matches!(request.band_type, BandpassType::BPoly)
298        && (request.amplitude_degree > 31 || request.phase_degree > 31)
299    {
300        return Err(BandpassSolveError::UnsupportedConfiguration {
301            reason: "BPOLY degree must be <= 31".to_string(),
302        });
303    }
304    Ok(())
305}

336/// Solve one bandpass interval channel by channel, both receptors of a
337/// channel together as CASA's `VisCalSolver2` does for B. A channel CASA
338/// cannot solve is flagged for every antenna, and an interval with no
339/// solved channel gets no rows, as `Calibrater::genericGatherAndSolve` keeps
340/// it only when one channel succeeds.
341fn solve_bandpass_group(
342    group: &SolveAccumulator,
343    available_antennas: &BTreeSet<i32>,
344    refant_id: i32,
345    combine: BandpassSolveCombine,
346    normalize_average_amplitude: bool,
347) -> Vec<BandpassSolutionRow> {
348    let channel_count = group.solution_channels;
349    // CASA solves the last channel first and starts each channel from the
350    // last solution found (`Calibrater::genericGatherAndSolve`).
351    let mut solved_channels = vec![None; channel_count];
352    let mut start = None::<SolvedGraphs>;
353    for (channel, graphs) in group.channel_graphs.iter().enumerate().rev() {
354        let solved = solve_graphs(
355            graphs,
356            GainType::G,
357            GainSolveMode::AmplitudePhase,
358            SolveGraphOptions {
359                refant_id,
360                min_baselines_per_antenna: 0,
361            },
362            start.as_ref(),
363        );
364        if solved.is_some() {
365            start.clone_from(&solved);
366        }
367        solved_channels[channel] = solved;
368    }
369    if solved_channels.iter().all(Option::is_none) {
370        return Vec::new();
371    }
372
373    let mut antenna_ids = available_antennas.clone();
374    antenna_ids.extend(group.antenna_ids.iter().copied());
375    antenna_ids.insert(refant_id);
376    let output_field_ids = if combine.fields && !combine.scans {
377        group.field_ids.iter().copied().collect::<Vec<_>>()
378    } else {
379        vec![group.field_id]
380    };
381    let scan_number = *group.scan_numbers.iter().next().unwrap_or(&0);
382    let interval_seconds = group.total_interval / group.sample_rows.max(1) as f64;
383
384    let mut solution_rows = Vec::new();
385    for field_id in output_field_ids {
386        for antenna_id in antenna_ids.iter().copied() {
387            let mut gains = Vec::with_capacity(2 * channel_count);
388            let mut flags = Vec::with_capacity(2 * channel_count);
389            for solved in &solved_channels {
390                for receptor in 0..2 {
391                    let solution = solved.as_ref().and_then(|solved| {
392                        let receptor = solved.receptors.get(receptor)?;
393                        receptor
394                            .solved
395                            .contains(&antenna_id)
396                            .then(|| receptor.gains[&antenna_id])
397                    });
398                    gains.push(solution.unwrap_or(Complex32::new(1.0, 0.0)));
399                    flags.push(solution.is_none());
400                }
401            }
402            solution_rows.push(BandpassSolutionRow {
403                time_seconds: group.solution_time,
404                interval_seconds,
405                field_id,
406                spw_id: group.spw_id,
407                antenna_id,
408                scan_number,
409                observation_id: group.observation_id,
410                channel_count,
411                gains,
412                flags,
413            });
414        }
415    }
416
417    if normalize_average_amplitude {
418        normalize_bandpass_rows(solution_rows, channel_count)
419    } else {
420        solution_rows
421    }
422}

424fn normalize_bandpass_rows(
425    mut rows: Vec<BandpassSolutionRow>,
426    channel_count: usize,
427) -> Vec<BandpassSolutionRow> {
428    for row in &mut rows {
429        normalize_bandpass_row(row, channel_count);
430    }
431    rows
432}
433
434fn normalize_bandpass_row(row: &mut BandpassSolutionRow, channel_count: usize) {
435    if channel_count == 0 {
436        return;
437    }
438    for receptor in 0..2 {
439        let mut coherent_sum = Complex32::new(0.0, 0.0);
440        let mut power_sum = 0.0_f32;
441        let mut good_count = 0usize;
442        for channel in 0..channel_count {
443            let index = receptor + 2 * channel;
444            if row.flags.get(index).copied().unwrap_or(false) {
445                continue;
446            }
447            let gain = row.gains[index];
448            let amplitude = gain.norm();
449            if amplitude <= f32::EPSILON {
450                continue;
451            }
452            coherent_sum += gain / Complex32::new(amplitude, 0.0);
453            power_sum += amplitude * amplitude;
454            good_count += 1;
455        }
456        if good_count <= 1 {
457            continue;
458        }
459        let amplitude_factor = (power_sum / good_count as f32).sqrt();
460        if amplitude_factor <= f32::EPSILON {
461            continue;
462        }
463        let phase_factor = if coherent_sum.norm() > f32::EPSILON {
464            coherent_sum / Complex32::new(coherent_sum.norm(), 0.0)
465        } else {
466            Complex32::new(1.0, 0.0)
467        };
468        let normalization = phase_factor * Complex32::new(amplitude_factor, 0.0);
469        for channel in 0..channel_count {
470            let index = receptor + 2 * channel;
471            row.gains[index] /= normalization;
472        }
473    }
474}

476fn write_bandpass_caltable(
477    ms: &MeasurementSet,
478    request: &BandpassSolveRequest,
479    refant_id: i32,
480    rows: &[BandpassSolutionRow],
481) -> Result<BandpassSolveReport, BandpassSolveError> {
482    let schema = TableSchema::new(vec![
483        ColumnSchema::scalar(COL_TIME, casa_types::PrimitiveType::Float64),
484        ColumnSchema::scalar(COL_FIELD_ID, casa_types::PrimitiveType::Int32),
485        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, casa_types::PrimitiveType::Int32),
486        ColumnSchema::scalar(COL_ANTENNA1, casa_types::PrimitiveType::Int32),
487        ColumnSchema::scalar(COL_ANTENNA2, casa_types::PrimitiveType::Int32),
488        ColumnSchema::scalar(COL_INTERVAL, casa_types::PrimitiveType::Float64),
489        ColumnSchema::scalar(COL_SCAN_NUMBER, casa_types::PrimitiveType::Int32),
490        ColumnSchema::scalar(COL_OBSERVATION_ID, casa_types::PrimitiveType::Int32),
491        ColumnSchema::scalar(COL_ARRAY_ID, casa_types::PrimitiveType::Int32),
492        ColumnSchema::scalar(COL_TIME_EXTRA_PREC, casa_types::PrimitiveType::Float64),
493        ColumnSchema::array_variable(COL_CPARAM, casa_types::PrimitiveType::Complex32, Some(2)),
494        ColumnSchema::array_variable(COL_PARAMERR, casa_types::PrimitiveType::Float32, Some(2)),
495        ColumnSchema::array_variable(COL_FLAG, casa_types::PrimitiveType::Bool, Some(2)),
496        ColumnSchema::array_variable(COL_SNR, casa_types::PrimitiveType::Float32, Some(2)),
497        ColumnSchema::array_variable(COL_WEIGHT, casa_types::PrimitiveType::Float32, Some(2)),
498    ])
499    .expect("valid bandpass schema");
500    let mut writer = CalibrationTableWriter::create(
501        ms,
502        CalibrationTableDescriptor {
503            output: &request.output_table,
504            schema,
505            subtype: "B Jones",
506            parameter_type: Some("Complex"),
507            measurement_set_name: ms
508                .path()
509                .map(|path| path.display().to_string())
510                .unwrap_or_else(|| "<in-memory>".to_string()),
511            include_polarization_basis: true,
512            time_extra_precision_column: Some(COL_TIME_EXTRA_PREC),
513        },
514    )?;
515
516    for row in rows {
517        let receptor_count = 2;
518        let element_count = receptor_count * row.channel_count;
519        writer.append(RecordValue::new(vec![
520            RecordField::new(
521                COL_TIME,
522                Value::Scalar(ScalarValue::Float64(row.time_seconds)),
523            ),
524            RecordField::new(
525                COL_FIELD_ID,
526                Value::Scalar(ScalarValue::Int32(row.field_id)),
527            ),
528            RecordField::new(
529                COL_SPECTRAL_WINDOW_ID,
530                Value::Scalar(ScalarValue::Int32(row.spw_id)),
531            ),
532            RecordField::new(
533                COL_ANTENNA1,
534                Value::Scalar(ScalarValue::Int32(row.antenna_id)),
535            ),
536            RecordField::new(COL_ANTENNA2, Value::Scalar(ScalarValue::Int32(refant_id))),
537            // CASA writes every solution's INTERVAL as 0 (`SolvableVisCal::keepNCT`).
538            RecordField::new(COL_INTERVAL, Value::Scalar(ScalarValue::Float64(0.0))),
539            RecordField::new(
540                COL_SCAN_NUMBER,
541                Value::Scalar(ScalarValue::Int32(row.scan_number)),
542            ),
543            RecordField::new(
544                COL_OBSERVATION_ID,
545                Value::Scalar(ScalarValue::Int32(row.observation_id)),
546            ),
547            RecordField::new(COL_ARRAY_ID, Value::Scalar(ScalarValue::Int32(0))),
548            RecordField::new(
549                COL_TIME_EXTRA_PREC,
550                Value::Scalar(ScalarValue::Float64(0.0)),
551            ),
552            RecordField::new(
553                COL_CPARAM,
554                Value::Array(ArrayValue::Complex32(
555                    ArrayD::from_shape_vec(
556                        IxDyn(&[receptor_count, row.channel_count]).f(),
557                        row.gains.clone(),
558                    )
559                    .expect("bandpass gains should reshape to receptor x channel"),
560                )),
561            ),
562            RecordField::new(
563                COL_PARAMERR,
564                Value::Array(ArrayValue::Float32(
565                    ArrayD::from_shape_vec(
566                        IxDyn(&[receptor_count, row.channel_count]).f(),
567                        vec![0.0; element_count],
568                    )
569                    .expect("bandpass paramerr should reshape to receptor x channel"),
570                )),
571            ),
572            RecordField::new(
573                COL_FLAG,
574                Value::Array(ArrayValue::Bool(
575                    ArrayD::from_shape_vec(
576                        IxDyn(&[receptor_count, row.channel_count]).f(),
577                        row.flags.clone(),
578                    )
579                    .expect("bandpass flags should reshape to receptor x channel"),
580                )),
581            ),
582            RecordField::new(
583                COL_SNR,
584                Value::Array(ArrayValue::Float32(
585                    ArrayD::from_shape_vec(
586                        IxDyn(&[receptor_count, row.channel_count]).f(),
587                        vec![1.0; element_count],
588                    )
589                    .expect("bandpass snr should reshape to receptor x channel"),
590                )),
591            ),
592            RecordField::new(
593                COL_WEIGHT,
594                Value::Array(ArrayValue::Float32(
595                    ArrayD::from_shape_vec(
596                        IxDyn(&[receptor_count, row.channel_count]).f(),
597                        vec![1.0; element_count],
598                    )
599                    .expect("bandpass weight should reshape to receptor x channel"),
600                )),
601            ),
602        ]))?;
603    }
604
605    writer.finish(&[
606        (SubtableId::Observation, "OBSERVATION"),
607        (SubtableId::Antenna, "ANTENNA"),
608        (SubtableId::Field, "FIELD"),
609        (SubtableId::History, "HISTORY"),
610        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
611    ])?;
612
613    let field_ids = rows.iter().map(|row| row.field_id).collect::<BTreeSet<_>>();
614    let spw_ids = rows.iter().map(|row| row.spw_id).collect::<BTreeSet<_>>();
615    let channel_count = rows.first().map(|row| row.channel_count).unwrap_or(0);
616    Ok(BandpassSolveReport {
617        output_table: request.output_table.clone(),
618        table_subtype: "B Jones".to_string(),
619        refant_antenna_id: refant_id,
620        field_ids: field_ids.iter().copied().collect(),
621        spectral_window_ids: spw_ids.iter().copied().collect(),
622        solution_row_count: rows.len(),
623        channel_count,
624    })
625}

631fn write_bpoly_caltable(
632    ms: &MeasurementSet,
633    request: &BandpassSolveRequest,
634    refant_id: i32,
635    rows: &[BandpassSolutionRow],
636) -> Result<BandpassSolveReport, BandpassSolveError> {
637    let fitted_rows = fit_bpoly_rows(ms, rows, request.amplitude_degree, request.phase_degree)?;
638
639    let schema = TableSchema::new(vec![

993fn fit_bpoly_rows(
994    ms: &MeasurementSet,
995    rows: &[BandpassSolutionRow],
996    amplitude_degree: usize,
997    phase_degree: usize,
998) -> Result<Vec<BPolySolutionRow>, BandpassSolveError> {
999    let spectral_window =
1000        ms.spectral_window()
1001            .map_err(|source| BandpassSolveError::OpenMeasurementSet {
1002                path: ms
1003                    .path()
1004                    .map(|path| path.display().to_string())
1005                    .unwrap_or_else(|| "<in-memory>".to_string()),
1006                source,
1007            })?;

1009        .map(|row| {

1030fn fit_bpoly_row(
1031    row: &BandpassSolutionRow,
1032    channel_frequencies_hz: &[f64],
1033    amplitude_degree: usize,
1034    phase_degree: usize,
1035) -> Result<BPolySolutionRow, BandpassSolveError> {
1036    if channel_frequencies_hz.len() != row.channel_count || row.channel_count == 0 {
1037        return Err(BandpassSolveError::UnsupportedParameterShape {
1038            path: "BPOLY fit channel grid".to_string(),
1039            shape: vec![channel_frequencies_hz.len(), row.channel_count],
1040        });
1041    }
1042    let valid_domain_hz = [
1043        channel_frequencies_hz.first().copied().unwrap_or(0.0),
1044        channel_frequencies_hz.last().copied().unwrap_or(0.0),
1045    ];
1046    let receptor_count = 2;
1047    let mut amp_coefficients = Vec::with_capacity(receptor_count);
1048    let mut phase_coefficients = Vec::with_capacity(receptor_count);
1049    for receptor in 0..receptor_count {
1050        let mut log_amplitudes = Vec::with_capacity(row.channel_count);
1051        let mut phases = Vec::with_capacity(row.channel_count);
1052        for channel in 0..row.channel_count {
1053            let gain = row.gains[receptor + receptor_count * channel];
1054            log_amplitudes.push(gain.norm().max(f32::EPSILON).ln() as f64);
1055            phases.push((gain.im as f64).atan2(gain.re as f64));
1056        }
1057        let unwrapped_phases = unwrap_phases(&phases);
1058        amp_coefficients.push(fit_legacy_bpoly_coefficients(
1059            channel_frequencies_hz,
1060            &log_amplitudes,
1061            valid_domain_hz,
1062            amplitude_degree,
1063            row,
1064        )?);
1065        phase_coefficients.push(fit_legacy_bpoly_coefficients(
1066            channel_frequencies_hz,
1067            &unwrapped_phases,
1068            valid_domain_hz,
1069            phase_degree,
1070            row,
1071        )?);
1072    }
1073    Ok(BPolySolutionRow {
1074        time_seconds: row.time_seconds,
1075        interval_seconds: row.interval_seconds,
1076        field_id: row.field_id,
1077        spw_id: row.spw_id,
1078        antenna_id: row.antenna_id,
1079        valid_domain_hz,
1080        scan_number: row.scan_number,
1081        observation_id: row.observation_id,
1082        scale_factor: Complex32::new(1.0, 0.0),
1083        amp_coefficients,
1084        phase_coefficients,
1085    })
1086}

1088fn fit_legacy_bpoly_coefficients(
1089    channel_frequencies_hz: &[f64],
1090    values: &[f64],
1091    valid_domain_hz: [f64; 2],
1092    degree: usize,
1093    row: &BandpassSolutionRow,
1094) -> Result<Vec<f64>, BandpassSolveError> {
1095    let coefficient_count = degree
1096        .saturating_add(1)
1097        .min(channel_frequencies_hz.len().max(1));
1098    if coefficient_count == 0 {
1099        return Ok(Vec::new());
1100    }
1101
1102    let mut basis = vec![0.0_f64; coefficient_count];
1103    let mut rows = Vec::with_capacity(channel_frequencies_hz.len());
1104    for (frequency_hz, value) in channel_frequencies_hz
1105        .iter()
1106        .copied()
1107        .zip(values.iter().copied())
1108    {
1109        legacy_bpoly_basis(
1110            valid_domain_hz[0],
1111            valid_domain_hz[1],
1112            frequency_hz,
1113            &mut basis,
1114        );
1115        rows.push((basis.clone(), value, 1.0));
1116    }
1117    solve_weighted_least_squares(&rows, coefficient_count).ok_or_else(|| {
1118        BandpassSolveError::FitPolynomial {
1119            antenna_id: row.antenna_id,
1120            field_id: row.field_id,
1121            spw_id: row.spw_id,
1122            reason: "least-squares system was singular".to_string(),
1123        }
1124    })
1125}
1126
1127fn legacy_bpoly_basis(x_start: f64, x_end: f64, x: f64, basis: &mut [f64]) {
1128    if basis.is_empty() {
1129        return;
1130    }
1131    basis.fill(0.0);
1132    basis[0] = 0.5_f64;
1133    if basis.len() == 1 || (x_end - x_start).abs() <= f64::EPSILON {
1134        return;
1135    }
1136    let xcap = ((x - x_start) - (x_end - x)) / (x_end - x_start);
1137    let mut t_prev = 1.0_f64;
1138    let mut t_curr = xcap;
1139    basis[1] = t_curr;
1140    for slot in basis.iter_mut().skip(2) {
1141        let t_next = 2.0_f64 * xcap * t_curr - t_prev;
1142        *slot = t_next;
1143        t_prev = t_curr;
1144        t_curr = t_next;
1145    }
1146}
1147
1148fn unwrap_phases(phases: &[f64]) -> Vec<f64> {
1149    let mut unwrapped = Vec::with_capacity(phases.len());
1150    let mut offset = 0.0_f64;
1151    let mut previous = None;
1152    for phase in phases.iter().copied() {
1153        if let Some(previous_phase) = previous {
1154            let delta = phase - previous_phase;
1155            if delta > std::f64::consts::PI {
1156                offset -= 2.0_f64 * std::f64::consts::PI;
1157            } else if delta < -std::f64::consts::PI {
1158                offset += 2.0_f64 * std::f64::consts::PI;
1159            }
1160        }
1161        let adjusted = phase + offset;
1162        unwrapped.push(adjusted);
1163        previous = Some(phase);
1164    }
1165    unwrapped
1166}
