55    /// state.
56    obs_mode: String,
57    /// Its `DATA_DESCRIPTION` row.
58    data_description_id: u32,
59    /// The `SPECTRAL_WINDOW` row of its data description.
60    spectral_window_id: u32,
61}

63/// An opened MeasurementSet v2 and its catalogue.
64///
65/// Opening reads MAIN's key columns in row chunks and partitions the rows the way
66/// XRADIO's converter does (`DATA_DESC_ID` × `OBS_MODE` × `OBSERVATION_ID`
67/// × `EPHEMERIS_ID` and the opening's
68/// [`PartitionScheme`](super::PartitionScheme)), and further by the feed
69/// pair, which XRADIO does not split by. A row whose `STATE_ID` is -1 has
70/// no intents here, where XRADIO's converter gives it the last `STATE`
71/// row's `OBS_MODE` (NumPy's index -1), so an MS mixing such rows with
72/// `STATE` rows partitions differently. Each partition's rows are
73/// placed on its time × baseline grid (the `index` module). Each partition
74/// has one data group, `base`, associating `DATA`, `FLAG`,
75/// `WEIGHT_SPECTRUM` (when MAIN has it, else `WEIGHT`) and `UVW`. Antenna,
76/// field and observation identities are the subtable rows; scans are the
77/// distinct `SCAN_NUMBER`s in ascending order.
78///
79/// Opening checks every unit and measure reference the batches rely on:
80/// `TIME` and `TIME_CENTROID` share one fixed reference (a variable
81/// reference is refused); the UVW convention is resolved to the phase
82/// centres' frame ([`UvwFrame`](casa_observation::UvwFrame)); positions,
83/// diameters, angles and frequencies are in the units their types claim.
84pub struct Msv2Observation {
85    ms: MeasurementSet,
86    catalogue: Arc<Catalogue>,
87    partitions: Vec<Partition>,
88    /// Each `STATE` row's `OBS_MODE`.
89    state_modes: Vec<String>,
90    /// The `STATE_ID`s MAIN's rows hold.
91    main_states: BTreeSet<i32>,
92    spectral_weights: bool,
93    /// Whether MAIN has `SIGMA_SPECTRUM`.
94    spectral_sigmas: bool,
95    options: OpenOptions,
96}

99    /// Open the MeasurementSet at `path` and read its catalogue.
100    ///
101    /// The tables stay open with casacore read locks, and their payload
102    /// rows stay on disk: the typed column fills read only the selected
103    /// cells, as the production selected-observation source does.
104    ///
105    /// # Errors
106    ///
107    /// When the MeasurementSet cannot be opened, a subtable the catalogue
108    /// needs is missing or malformed, a unit or measure reference is
109    /// outside the trial profile, or the rows do not fit their grids (two
110    /// rows in one cell, two fields or scans at one time, several
111    /// `PROCESSOR_ID`s in one partition).
112    #[cfg(unix)]
113    pub fn open(path: impl AsRef<Path>) -> Result<Self, ObservationError> {
114        Self::open_with(path, &OpenOptions::default())
115    }
116
117    /// [`Self::open`] with `options`.
118    ///
119    /// # Errors
120    ///
121    /// As [`Self::open`].
122    #[cfg(unix)]
123    pub fn open_with(
124        path: impl AsRef<Path>,
125        options: &OpenOptions,
126    ) -> Result<Self, ObservationError> {
127        Self::open_granted(path, options, &mut |_| Ok(()))
128    }

130    /// [`Self::open_with`], asking `grant` for the bytes the opening's index
131    /// will hold before it allocates them: its chunk of MAIN's key columns,
132    /// each partition's distinct times, baselines and processors as the
133    /// first pass finds them (at most one row's beyond what was granted),
134    /// then the scans and the grids. The totals asked for only grow, a few
135    /// at a time; a caller that admits memory admits each increase. An
136    /// error from `grant` refuses the opening with nothing more allocated.
137    ///
138    /// # Errors
139    ///
140    /// As [`Self::open`], and whatever `grant` returns.
141    #[cfg(unix)]
142    pub fn open_granted(
143        path: impl AsRef<Path>,
144        options: &OpenOptions,
145        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
146    ) -> Result<Self, ObservationError> {
147        let ms = MeasurementSet::open_retained_read(path).map_err(storage)?;
148        Self::from_measurement_set(ms, options, grant)
149    }

151    fn from_measurement_set(
152        ms: MeasurementSet,
153        options: &OpenOptions,
154        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
155    ) -> Result<Self, ObservationError> {
156        let antennas = catalogue::read_antennas(&ms)?;
157        let feeds = catalogue::read_feeds(&ms)?;
158        if let Some((antenna, feed, window)) = feeds.varying() {
159            tracing::warn!(
160                "FEED states receptors that vary in time (first for antenna {antenna}, feed \
161                 {feed}, window {window}); the last row of each antenna, feed and window is \
162                 used, as casacore's MSIter uses it"
163            );
164        }
165        let catalogue::Fields {
166            entries: fields,
167            ephemerides,
168            uvw_frame,
169        } = catalogue::fields_and_uvw(&ms)?;
170        let observations = catalogue::read_observations(&ms)?;
171        let main = ms.main_table();
172        let time_scale = catalogue::time_scale(main)?;
173        let setups = catalogue::spectral_setups(&ms)?;
174        let modes = catalogue::read_state_modes(&ms)?;
175        let (spectral_weights, spectral_sigmas) = (
176            has_content(main, "WEIGHT_SPECTRUM")?,
177            has_content(main, "SIGMA_SPECTRUM")?,
178        );
179        let index = Index::build(
180            main,
181            options.partition_scheme,
182            (&modes, &ephemerides),
183            ROWS_PER_CHUNK,
184            grant,
185        )?;
186        let groups = data_groups(main, spectral_weights, uvw_frame);
187        let mut partitions = Vec::new();
188        let mut entries = Vec::new();
189        for (key, grid) in index.partitions {
190            let data_description_id = u32::try_from(key.data_description)
191                .ok()
192                .filter(|row| (*row as usize) < setups.len())
193                .ok_or_else(|| {
194                    invalid(format!(
195                        "DATA_DESC_ID {} is not in DATA_DESCRIPTION",
196                        key.data_description
197                    ))
198                })?;
199            let setup = &setups[data_description_id as usize];
200            let observation = u32::try_from(key.observation)
201                .ok()
202                .filter(|id| (*id as usize) < observations.len())
203                .map(ObservationId)
204                .ok_or_else(|| {
205                    invalid(format!(
206                        "OBSERVATION_ID {} is not in OBSERVATION",
207                        key.observation
208                    ))
209                })?;
210            let feed = |feed: i32| {
211                u32::try_from(feed).map_err(|_| invalid(format!("negative feed {feed}")))
212            };
213            let receptors = partition_receptors(
214                &feeds,
215                &grid.baselines,
216                key.feeds,
217                setup.spectral_window_id as i32,
218            )?;
219            let feeds = [feed(key.feeds[0])?, feed(key.feeds[1])?];
220            entries.push(PartitionEntry {
221                spectral_window: setup.spectral_window.clone(),
222                frequencies_hz: setup.frequencies_hz.clone(),
223                frequency_frame: setup.frequency_frame,
224                channel_widths_hz: setup.channel_widths_hz.clone(),
225                correlations: setup.correlations.clone(),
226                intents: catalogue::intents(&key.obs_mode),
227                observation,
228                feeds: Some(feeds),
229                fields: super::distinct(&grid.field_per_time),
230                scans: super::distinct(&grid.scan_per_time),
231                baselines: grid.baselines.clone(),
232                time_scale,
233                time_origin: MjdHighPrec::new(0.0, 0.0),
234                data_groups: groups.clone(),
235                receptors,
236            });
237            partitions.push(Partition {
238                grid,
239                obs_mode: key.obs_mode,
240                data_description_id,
241                spectral_window_id: setup.spectral_window_id,
242            });
243        }
244        Ok(Self {
245            catalogue: Arc::new(Catalogue {
246                antennas,
247                fields,
248                scans: index.scans,
249                observations,
250                partitions: entries,
251            }),
252            partitions,
253            state_modes: modes,
254            main_states: index.states,
255            spectral_weights,
256            spectral_sigmas,
257            options: options.clone(),
258            ms,
259        })
260    }

321    /// The bytes opening a MeasurementSet of `main_rows` MAIN rows holds
322    /// when its grids are full and its times hold a few rows each: a cell
323    /// and eight bytes of time tables per row, and one chunk of key
324    /// columns. Padding cells, which only opening finds, and times of one
325    /// or two rows hold more, which [`Self::open_granted`] asks for before
326    /// it allocates them.
327    #[must_use]
328    pub const fn opening_bytes(main_rows: usize) -> usize {
329        let chunk = if main_rows < index::ROWS_PER_CHUNK {
330            main_rows
331        } else {
332            index::ROWS_PER_CHUNK
333        };
334        main_rows * (size_of::<u32>() + size_of::<f64>()) + chunk * index::CHUNK_ROW_BYTES
335    }

337    /// Bytes the opening's index holds while the observation is open: each
338    /// partition's grid, a MAIN row number per cell with its time and
339    /// baseline tables.
340    #[must_use]
341    pub fn index_bytes(&self) -> usize {
342        self.partitions
343            .iter()
344            .map(|partition| {
345                let grid = &partition.grid;
346                grid.cells.capacity() * size_of::<u32>()
347                    + grid.times_s.capacity() * size_of::<f64>()
348                    + grid.field_per_time.capacity() * size_of::<FieldId>()
349                    + grid.scan_per_time.capacity() * size_of::<ScanId>()
350                    + grid.baselines.capacity() * size_of::<OrderedBaseline>()
351            })
352            .sum()
353    }

355    /// The MAIN row partition `id` holds at `time_s` (MAIN `TIME`, seconds
356    /// from MJD 0, as its batches deliver it) on `baseline`; `None` where
357    /// its grid is padded or the partition has no such time or baseline.
358    ///
359    /// # Panics
360    ///
361    /// If `id` is not a partition of the observation.
362    #[must_use]
363    pub fn main_row(&self, id: PartitionId, time_s: f64, baseline: OrderedBaseline) -> Option<u32> {
364        let grid = &self.partitions[id.0 as usize].grid;
365        let time = grid
366            .times_s
367            .binary_search_by(|stored| stored.total_cmp(&time_s))
368            .ok()?;
369        let baseline = grid.baselines.binary_search(&baseline).ok()?;
370        let row = grid.cells[time * grid.baselines.len() + baseline];
371        (row != index::NO_ROW).then_some(row)
372    }

422    /// What `request` costs on partition `id`, from the grid and the tile
423    /// shape: every MSv2 role is a required column, so binding refuses
424    /// nothing the request can ask for beyond the request's own
425    /// consistency.
426    fn estimate_partition(
427        &self,
428        request: &AccessRequest,
429        id: PartitionId,
430    ) -> Result<(PartitionEstimate, ResolvedRequest), ObservationError> {
431        let (partition, entry) = self.partition(id)?;
432        let resolved = request.resolve(
433            entry,
434            &partition.grid.times_s,
435            &partition.grid.field_per_time,
436        )?;
437        let roles = request.roles;
438        let weights = weight_source(
439            request.weights,
440            &entry.data_groups[resolved.data_group],
441            self.spectral_weights,
442            self.spectral_sigmas,
443        );
444        let per_channel = weights.per_channel;
445        let channels = request.channels.len();
446        let (delivered, stored) = (
447            resolved.correlations.products(entry).len(),
448            entry.correlations.len(),
449        );
450        let scalars = [roles.time_centroid, roles.interval, roles.exposure]
451            .into_iter()
452            .filter(|wanted| *wanted)
453            .count();
454        let bytes_per_sample = roles.sample_bytes(channels, delivered, per_channel, scalars);
455        let grid = &partition.grid;
456        // The cursor's scratch: each present row's number, sample, FLAG_ROW
457        // and time auxiliaries; and, when a subset is delivered, every stored
458        // product of the cube roles, read before the delivered ones are
459        // gathered. Padding needs none: rows are spread in place.
460        let gathered = Roles {
461            uvw: false,
462            ..roles
463        };
464        let scratch_bytes_per_sample = 2 * size_of::<usize>()
465            + usize::from(roles.flags)
466            + 8 * scalars
467            + if resolved.correlations.indices().is_some() {
468                gathered.sample_bytes(channels, stored, per_channel, 0)
469            } else {
470                0
471            };
472        let per_sample = |correlations: usize| {
473            usize::from(roles.values) * channels * correlations
474                + usize::from(roles.flags) * channels * correlations
475                + usize::from(roles.weights)
476                    * if per_channel {
477                        channels * correlations
478                    } else {
479                        correlations
480                    }
481                + usize::from(roles.uvw) * 3
482                + scalars
483        };
484        let (times, baselines) = (resolved.times.len(), grid.baselines.len());
485        // The selection's rows: its cells that a row fills, counted at
486        // opening for the whole grid.
487        let cells = times * baselines;
488        let rows = if times == grid.times_s.len() {
489            grid.rows
490        } else {
491            resolved
492                .times
493                .runs()
494                .iter()
495                .map(|run| {
496                    grid.cells(run.clone())
497                        .iter()
498                        .filter(|row| **row != index::NO_ROW)
499                        .count()
500                })
501                .sum()
502        };
503        let bytes_per_time = roles.time_bytes(0);
504        let times_per_batch = times_per_batch(
505            request.batch_bytes,
506            baselines * (bytes_per_sample + scratch_bytes_per_sample) + bytes_per_time,
507            times,
508        )?;
509        let group = &entry.data_groups[resolved.data_group];
510        let tile = self
511            .ms
512            .main_table()
513            .array_column_tile_shape(&group.visibilities)
514            .map_err(table_error)?;
515        let estimate = PartitionEstimate {
516            partition: id,
517            times,
518            time_runs: resolved.times.clone(),
519            weights: roles.weights.then_some(weights),
520            baselines,
521            bytes_per_sample,
522            bytes_per_time,
523            scratch_bytes_per_sample,
524            times_per_batch,
525            batches: if times_per_batch == 0 {
526                0
527            } else {
528                resolved.times.blocks(times_per_batch).count()
529            },
530            decode_units: rows,
531            decoded_elements: rows * per_sample(stored),
532            selected_elements: rows * per_sample(delivered),
533            natural_blocks: match tile.as_deref() {
534                // A cell tile is [correlation, channel, rows].
535                Some([_, channels, rows, ..]) => NaturalBlocks {
536                    times: (*rows / baselines.max(1)).max(1),
537                    channels: *channels,
538                },
539                _ => NaturalBlocks {
540                    times: 1,
541                    channels: entry.frequencies_hz.len(),
542                },
543            },
544            padded_fraction: if cells == 0 {
545                0.0
546            } else {
547                1.0 - rows as f64 / cells as f64
548            },
549        };
550        Ok((estimate, resolved))
551    }

553    /// What `request` costs ([`Observation::estimate`]).
554    ///
555    /// # Errors
556    ///
557    /// When the request names an unknown partition, data group, channel or
558    /// correlation product, or budgets less than one time.
559    pub fn estimate(&self, request: &AccessRequest) -> Result<Estimate, ObservationError> {
560        Ok(Estimate {
561            partitions: request
562                .partitions

565                .collect::<Result<_, ObservationError>>()?,
566        })

569    /// Bind a traversal of `request`: one cursor per partition, whole times
570    /// per batch ([`Observation::bind_read`]).
571    ///
572    /// # Errors
573    ///
574    /// As [`Self::estimate`].
575    pub fn bind_read(&self, request: &AccessRequest) -> Result<Traversal<'_>, ObservationError> {
576        let bound = request
577            .partitions
578            .iter()
579            .map(|&id| self.estimate_partition(request, id))
580            .collect::<Result<Vec<_>, _>>()?;
581        let estimate = Estimate {
582            partitions: bound.iter().map(|(estimate, _)| estimate.clone()).collect(),
583        };
584        let cursors = bound
585            .into_iter()
586            .map(|(estimate, resolved)| {
587                Msv2Cursor::bind(self, request, estimate, resolved)
588                    .map(|cursor| Box::new(cursor) as BoundCursor<'_>)
589            })
590            .collect::<Result<_, _>>()?;
591        Ok(Traversal::new(cursors, estimate))
592    }

594    /// Partition `id`'s grid ([`Observation::grid`]): its times, fields and
595    /// scans from the opening index, and the one `INTERVAL` of its rows,
596    /// read in bounded row chunks. Padding has no row, so no interval.
597    ///
598    /// # Errors
599    ///
600    /// As [`Observation::grid`].
601    pub fn grid(
602        &self,
603        id: PartitionId,
604        data_group: &str,
605    ) -> Result<TimeBaselineGrid, ObservationError> {
606        let (partition, entry) = self.partition(id)?;
607        if entry.data_group(data_group).is_none() {
608            return Err(ObservationError::InvalidRequest {
609                reason: format!("no data group {data_group} in {id:?}"),
610            });
611        }
612        let grid = &partition.grid;
613        let rows: Vec<usize> = grid
614            .cells
615            .iter()
616            .filter(|&&row| row != index::NO_ROW)
617            .map(|&row| row as usize)
618            .collect();
619        let main = self.ms.main_table();
620        let mut interval = None;
621        let mut values = Vec::new();
622        for chunk in rows.chunks(ROWS_PER_CHUNK) {
623            values.clear();
624            main.required_scalar_columns_for_rows_into(
625                chunk,
626                &mut [RequiredScalarColumnDestination::new(
627                    "INTERVAL",
628                    RequiredScalarColumnValuesMut::Float64(&mut values),
629                )],
630            )
631            .map_err(table_error)?;
632            for value in &values {
633                if *interval.get_or_insert(*value) != *value {
634                    return Err(ObservationError::Unsupported {
635                        what: "a grid with more than one integration interval",
636                    });
637                }
638            }
639        }
640        Ok(TimeBaselineGrid {
641            baselines: grid.baselines.clone(),
642            time_scale: entry.time_scale,
643            time_origin: entry.time_origin,
644            times_s: grid.times_s.clone(),
645            field_per_time: grid.field_per_time.clone(),
646            scan_per_time: grid.scan_per_time.clone(),
647            interval_s: interval.ok_or_else(|| ObservationError::InvalidRequest {
648                reason: format!("{id:?} has no rows"),
649            })?,
650        })
651    }
652}

654/// The receptors of each antenna `baselines` name, through the feed pair
655/// `feeds` (`FEED1` for the first antenna, `FEED2` for the second) in
656/// spectral window `window`; an antenna named through two feeds must have
657/// the same receptors through both.
658fn partition_receptors(
659    feeds: &catalogue::Feeds,
660    baselines: &[OrderedBaseline],
661    pair: [i32; 2],
662    window: i32,
663) -> Result<BTreeMap<AntennaId, Receptors>, ObservationError> {
664    let mut receptors = BTreeMap::new();
665    for baseline in baselines {
666        for (antenna, feed) in [(baseline.antenna1, pair[0]), (baseline.antenna2, pair[1])] {
667            let id = i32::try_from(antenna.0).map_err(|_| invalid("an antenna beyond i32"))?;
668            let Some(own) = feeds.receptors(id, feed, window) else {
669                continue;
670            };
671            if receptors
672                .insert(antenna, own.clone())
673                .is_some_and(|known| &known != own)
674            {
675                return Err(ObservationError::Unsupported {
676                    what: "an MSv2 antenna with different receptors through its two feeds",
677                });
678            }
679        }
680    }
681    Ok(receptors)
682}

734/// What the weights role of data group `group` delivers under `request`,
735/// on a MAIN with or without `WEIGHT_SPECTRUM` and `SIGMA_SPECTRUM`: the
736/// group's stored weights, or 1/σ² from `SIGMA_SPECTRUM` when MAIN has both
737/// it and `WEIGHT_SPECTRUM` (CASA's rule, `VisBufferImpl2`), else from
738/// `SIGMA`. Every MSv2 has `SIGMA`, so no fallback is taken.
739fn weight_source(
740    request: WeightRequest,
741    group: &DataGroupEntry,
742    spectral_weights: bool,
743    spectral_sigmas: bool,
744) -> WeightSource {
745    match request {
746        WeightRequest::Stored => WeightSource {
747            rule: WeightRule::Stored,
748            variable: group.weights.clone(),
749            per_channel: spectral_weights,
750        },
751        WeightRequest::InverseSigmaSquared { .. } => {
752            let per_channel = spectral_weights && spectral_sigmas;
753            WeightSource {
754                rule: WeightRule::InverseSigmaSquared,
755                variable: if per_channel {
756                    "SIGMA_SPECTRUM"
757                } else {
758                    "SIGMA"
759                }
760                .to_string(),
761                per_channel,
762            }
763        }
764    }
765}

767/// Whether MAIN's `column` holds content, as casacore's
768/// `TableColumn::hasContent` tells, by which CASA's `VisBufferImpl2` picks
769/// `WEIGHT_SPECTRUM` and `SIGMA_SPECTRUM` over `WEIGHT` and `SIGMA`: the
770/// column is there, its first cell is defined, and no axis of that cell is
771/// empty. A column added and never filled holds none.
772fn has_content(main: &casa_tables::Table, column: &str) -> Result<bool, ObservationError> {
773    if main.row_count() == 0
774        || !main
775            .schema()
776            .is_some_and(|schema| schema.contains_column(column))
777    {
778        return Ok(false);
779    }
780    // One cell, read without the row cache the opening's lazy fills need
781    // left empty. A cell this reader cannot decode (a storage manager it
782    // lacks) is not proof of no content: the column is taken as holding
783    // content, and reading it reports the error, as before.
784    let Ok(first) = main
785        .column_accessor(column)
786        .and_then(|cells| cells.array_cells_owned_uncached(&[0]))
787    else {
788        return Ok(true);
789    };
790    Ok(first.into_iter().next().flatten().is_some_and(|cell| {
791        let shape = cell.shape();
792        !shape.is_empty() && shape.iter().all(|axis| *axis > 0)
793    }))
794}

796/// The data groups MAIN's visibility columns form, named as XRADIO's
797/// converter names them: `base` (`DATA`), and `corrected`
798/// (`CORRECTED_DATA`) and `model` (`MODEL_DATA`) where the columns exist.
799/// Every group reads the same flags, weights and UVW.
800pub(super) fn data_groups(
801    main: &casa_tables::Table,
802    spectral_weights: bool,
803    uvw_frame: casa_observation::UvwFrame,
804) -> Vec<DataGroupEntry> {
805    let weights = if spectral_weights {
806        "WEIGHT_SPECTRUM"
807    } else {
808        "WEIGHT"
809    };
810    let has = |column: &str| {
811        main.schema()
812            .is_some_and(|schema| schema.contains_column(column))
813    };
814    [
815        ("base", "DATA"),
816        ("corrected", "CORRECTED_DATA"),
817        ("model", "MODEL_DATA"),
818    ]
819    .into_iter()
820    .filter(|(name, column)| *name == "base" || has(column))
821    .map(|(name, column)| DataGroupEntry {
822        name: name.to_string(),
823        visibilities: column.to_string(),
824        flags: "FLAG".to_string(),
825        weights: weights.to_string(),
826        uvw: "UVW".to_string(),
827        units: catalogue::data_units(main, column),
828        uvw_frame,
829    })
830    .collect()
831}

834mod tests {
835    use casa_observation::{
836        DataGroupEntry, SigmaFallback, UvwFrame, VisibilityUnits, WeightRequest, WeightRule,
837    };
838
839    use casa_types::measures::direction::DirectionRef;
840
841    use super::weight_source;
842
843    /// CASA's rule: per-channel sigmas only when MAIN has both
844    /// `SIGMA_SPECTRUM` and `WEIGHT_SPECTRUM`, else `SIGMA`; stored weights
845    /// are the group's.
846    #[test]
847    fn sigma_weights_follow_casas_column_rule() {
848        let group = DataGroupEntry {
849            name: "base".to_string(),
850            visibilities: "DATA".to_string(),
851            flags: "FLAG".to_string(),
852            weights: "WEIGHT_SPECTRUM".to_string(),
853            uvw: "UVW".to_string(),
854            units: VisibilityUnits::Undeclared,
855            uvw_frame: UvwFrame {
856                effective: DirectionRef::J2000,
857                stored_label: None,
858            },
859        };
860        let sigma = WeightRequest::InverseSigmaSquared {
861            fallback: SigmaFallback::Refuse,
862        };
863        let cases = [
864            (true, true, "SIGMA_SPECTRUM", true),
865            (true, false, "SIGMA", false),
866            (false, true, "SIGMA", false),
867            (false, false, "SIGMA", false),
868        ];
869        for (weights, sigmas, variable, per_channel) in cases {
870            let source = weight_source(sigma, &group, weights, sigmas);
871            assert_eq!(source.rule, WeightRule::InverseSigmaSquared);
872            assert_eq!(
873                (source.variable.as_str(), source.per_channel),
874                (variable, per_channel),
875                "WEIGHT_SPECTRUM {weights}, SIGMA_SPECTRUM {sigmas}"
876            );
877        }
878        let stored = weight_source(WeightRequest::Stored, &group, true, true);
879        assert_eq!(
880            (stored.rule, stored.variable.as_str(), stored.per_channel),
881            (WeightRule::Stored, "WEIGHT_SPECTRUM", true)
882        );
883    }
