22fn serialize_image(image: &ImageHandle) -> PyResult<CasaValue> {
23    let shape = image.shape();
24    let pixels = if let Some(fits) = image.fits() {
25        let start = vec![0; shape.len()];
26        let stride = vec![1; shape.len()];
27        ArrayValue::Float32(fits.read(&start, shape, &stride)?)
28    } else {
29        match image {
30            ImageHandle::Float(storage) => {
31                with_storage!(storage, image => image.get().map(ArrayValue::Float32))
32            }
33            ImageHandle::Double(storage) => {
34                with_storage!(storage, image => image.get().map(ArrayValue::Float64))
35            }
36            ImageHandle::Complex(storage) => {
37                with_storage!(storage, image => image.get().map(ArrayValue::Complex32))
38            }
39            ImageHandle::DComplex(storage) => {
40                with_storage!(storage, image => image.get().map(ArrayValue::Complex64))
41            }
42        }
43        .map_err(raised)?
44    };
45    let owner = Path::new(".");
46    let info = record_info(image_info(image).map_err(raised)?, image_info_record(image));
47    Ok(CasaValue::record([
48        (
49            "coordsys",
50            CasaValue::record([(
51                "coordsys",
52                values::record(&image.coordinate_record(), owner),
53            )]),
54        ),
55        ("imagearray", values::value(&Value::Array(pixels), owner)),
56        ("imageinfo", values::record(&info, owner)),
57        (
58            "shape",
59            CasaValue::Ints {
60                shape: vec![shape.len()],
61                values: shape.iter().map(|&size| size as i64).collect(),
62            },
63        ),
64    ]))
65}

67fn record_info(mut info: ImageInfo, original: Option<&RecordValue>) -> RecordValue {
68    if info.image_type == ImageType::Undefined {
69        info.image_type = ImageType::Intensity;
70    }
71    let mut record = info.to_record();
72    for beam_field in ["restoringbeam", "perplanebeams"] {
73        if record.get(beam_field).is_some()
74            && let Some(value) = original.and_then(|original| original.get(beam_field))
75        {
76            record.upsert(beam_field, value.clone());
77        }
78    }
79    record
80}

82fn field<'a>(record: &'a RecordValue, key: &str) -> PyResult<&'a Value> {
83    record
84        .get(key)
85        .ok_or_else(|| raised(format!("RecordInterface: field {key} is unknown")))
86}
87
88fn expect_record<'a>(record: &'a RecordValue, key: &str) -> PyResult<&'a RecordValue> {
89    if let Value::Record(value) = field(record, key)? {
90        return Ok(value);
91    }
92    Err(raised(format!(
93        "RecordInterface: field {key} is not a record"
94    )))
95}

98    raised(format!(

103fn record_shape(value: &Value) -> PyResult<Vec<usize>> {
104    let extents = match value {
105        Value::Array(ArrayValue::Int32(shape)) if shape.ndim() == 1 => {
106            shape.iter().map(|&extent| i64::from(extent)).collect()
107        }
108        Value::Array(ArrayValue::Int64(shape)) if shape.ndim() == 1 => {
109            shape.iter().copied().collect()
110        }
111        Value::Scalar(ScalarValue::Int32(extent)) => vec![i64::from(*extent)],
112        Value::Scalar(ScalarValue::Int64(extent)) => vec![*extent],
113        _ => return Err(incorrect_type("shape")),
114    };
115    if extents.is_empty()
116        || extents
117            .iter()
118            .any(|&extent| extent <= 0 || extent > i64::from(i32::MAX))
119    {
120        return Err(raised(
121            "TiledShape: shape has to be > 0 and fit in a signed 32-bit integer",
122        ));
123    }
124    Ok(extents.into_iter().map(|extent| extent as usize).collect())
125}

127fn fit_into_image<T>(shape: &[usize], mut input: ArrayD<T>) -> PyResult<ArrayD<T>> {
128    if input.ndim() > shape.len() {
129        return Err(raised("ArrayLattice: imagearray has more axes than shape"));
130    }
131    if !input
132        .shape()
133        .iter()
134        .zip(shape)
135        .all(|(&input, &output)| input <= output)
136    {
137        return Err(raised(
138            "ArrayLattice: imagearray does not fit the requested shape",
139        ));
140    }
141    for axis in input.ndim()..shape.len() {
142        input = input.insert_axis(Axis(axis));
143    }
144    Ok(input)
145}

147fn record_pixels(value: &Value) -> PyResult<ArrayValue> {
148    let convert_complex =
149        |number: num_complex::Complex64| Complex32::new(number.re as f32, number.im as f32);
150    let scalar = |value| ndarray::arr1(&[value]).into_dyn();
151    let array = match value {
152        Value::Array(ArrayValue::Int16(values)) => ArrayValue::Float32(values.mapv(f32::from)),
153        Value::Array(ArrayValue::Int32(values)) => {
154            ArrayValue::Float32(values.mapv(|value| value as f32))
155        }
156        Value::Array(ArrayValue::Int64(values)) => {
157            ArrayValue::Float32(values.mapv(|value| value as f32))
158        }
159        Value::Array(ArrayValue::UInt8(values)) => ArrayValue::Float32(values.mapv(f32::from)),
160        Value::Array(ArrayValue::UInt16(values)) => ArrayValue::Float32(values.mapv(f32::from)),
161        Value::Array(ArrayValue::UInt32(values)) => {
162            ArrayValue::Float32(values.mapv(|value| value as f32))
163        }
164        Value::Array(ArrayValue::Float32(values)) => ArrayValue::Float32(values.clone()),
165        Value::Array(ArrayValue::Float64(values)) => {
166            ArrayValue::Float32(values.mapv(|value| value as f32))
167        }
168        Value::Array(ArrayValue::Complex32(values)) => ArrayValue::Complex32(values.clone()),
169        Value::Array(ArrayValue::Complex64(values)) => {
170            ArrayValue::Complex32(values.mapv(convert_complex))
171        }
172        Value::Scalar(ScalarValue::Int32(value)) => ArrayValue::Float32(scalar(*value as f32)),
173        Value::Scalar(ScalarValue::Int64(value)) => ArrayValue::Float32(scalar(*value as f32)),
174        Value::Scalar(ScalarValue::Float32(value)) => ArrayValue::Float32(scalar(*value)),
175        Value::Scalar(ScalarValue::Float64(value)) => ArrayValue::Float32(scalar(*value as f32)),
176        Value::Scalar(ScalarValue::Complex32(value)) => {
177            ArrayValue::Complex32(ndarray::arr1(&[*value]).into_dyn())
178        }
179        Value::Scalar(ScalarValue::Complex64(value)) => {
180            ArrayValue::Complex32(ndarray::arr1(&[convert_complex(*value)]).into_dyn())
181        }
182        _ => return Err(incorrect_type("imagearray")),
183    };
184    Ok(array)
185}

187#[pymethods]
188impl PyImageTool {
189    fn torecord(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
190        let record = py.detach(|| {
191            let state = self.state();
192            let Some(attachment) = state.image.as_ref() else {
193                return Ok(CasaValue::empty_record());
194            };
195            let mut image = attachment.lock();
196            attachment.require_pixel_read_lock(&mut image)?;
197            let record = serialize_image(&image)?;
198            image.release_requested_lock()?;
199            Ok::<_, PyErr>(record)
200        })?;
201        record.into_python(py)
202    }
203
204    #[pyo3(signature = (record, outfile=""), text_signature = "($self, record, outfile='')")]
205    fn fromrecord(
206        &self,
207        py: Python<'_>,
208        record: &Bound<'_, PyDict>,
209        outfile: &str,
210    ) -> PyResult<bool> {
211        let record = casa_record(py, Some(record))?;
212        py.detach(|| {
213            let mut state = self.state();
214            state.image = None;
215            let pixel_value = field(&record, "imagearray")?;
216            let shape = record_shape(field(&record, "shape")?)?;
217            let coordinate_record = subrecord(expect_record(&record, "coordsys")?, "coordsys")
218                .ok_or_else(|| raised("Error converting image from record: invalid coordinate system"))?;
219            let coordinates = CoordinateSystem::from_record(coordinate_record).map_err(|error| {
220                match error {
221                    CoordinateError::UnsupportedAxisLayout(_) => raised(format!("{error} (casa-rs issue #1002)")),
222                    CoordinateError::UnsupportedProjection(_) => raised(format!("{error} (casa-rs issue #1626)")),
223                    _ => raised(error),
224                }
225            })?;
226            let (coordinates, coordinate_record) = if coordinates.n_pixel_axes() == shape.len() {
227                (coordinates, coordinate_record.clone())
228            } else {
229                if !outfile.is_empty() {
230                    return Err(raised("PagedImage: Failed AlwaysAssert setCoordinateInfo(coordinateInfo)"));
231                }
232                // casacore's temporary restore keeps its default coordinate
233                // system when setting the incompatible record fails.
234                let obs = ObsInfo::new("UNKNOWN")
235                    .with_observer("UNKNOWN")
236                    .with_date(MEpoch::from_mjd(0.0, EpochRef::LAST));
237                let coordinates = CoordinateSystem::new().with_obs_info(obs);
238                let record = coordinates.to_record();
239                (coordinates, record)
240            };
241            let info = expect_record(&record, "imageinfo")?;
242            let info = record_info(ImageInfo::from_record(info).map_err(raised)?, Some(info));
243            let pixels = record_pixels(pixel_value)?;
244            let attached = publish_image(outfile, true, |path| {
245                let mut image = match pixels {
246                    ArrayValue::Float32(data) => {
247                        let data = fit_into_image(&shape, data)?;
248                        ImageHandle::Float(Storage::create(shape, coordinates, path, Some(&data)).map_err(raised)?)
249                    }
250                    ArrayValue::Complex32(data) => {
251                        let data = fit_into_image(&shape, data)?;
252                        ImageHandle::Complex(Storage::create(shape, coordinates, path, Some(&data)).map_err(raised)?)
253                    }
254                    _ => unreachable!("record allocated single precision pixels"),
255                };
256                image.set_coordinate_record(coordinate_record)?;
257                with_pixels!(&mut image, storage => with_storage!(storage, image => image.set_image_info_record(info))).map_err(raised)?;
258                if state.history {
259                    image.record_history("fromrecord", format!("ia.fromrecord(record={{...}}, outfile={outfile:?})"))?;
260                }
261                image.save().map_err(raised)?;
262                Ok(image)
263            })?;
264            state.image = Some(attached);
265            Ok(true)
266        })
267    }
