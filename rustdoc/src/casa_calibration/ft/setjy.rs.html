19pub struct ManualFlux {
20    flux_jy: [f64; 4],
21    reference: MFrequency,
22    spectral_index: Vec<f64>,
23    fraction: Vec<f64>,
24    angle: Vec<f64>,
25    rotation_measure: f64,
26    tabulate: bool,
27}
28
29impl ManualFlux {
30    pub fn new(
31        flux_jy: [f64; 4],
32        reference: MFrequency,
33        spectral_index: Vec<f64>,
34        mut fraction: Vec<f64>,
35        mut angle: Vec<f64>,
36        rotation_measure: f64,
37    ) -> MsResult<Self> {
38        let [i, q, u, _] = flux_jy;
39        if !flux_jy
40            .iter()
41            .chain(&spectral_index)
42            .chain(&fraction)
43            .chain(&angle)
44            .all(|v| v.is_finite())
45            || !reference.hz().is_finite()
46            || reference.hz() <= 0.
47            || !rotation_measure.is_finite()
48            || i < 0.
49        {
50            return Err(invalid("invalid manual flux or spectral coefficients"));
51        }
52        let tabulate = spectral_index.len() > 1 || fraction.len() > 1 || angle.len() > 1;
53        // Explicit Q and U define the reference linear fraction and angle.
54        // The higher coefficients are retained when both lists are present.
55        if q != 0. || u != 0. {
56            if fraction.is_empty() || angle.is_empty() {
57                fraction.resize(1, 0.);
58                angle.resize(1, 0.);
59            }
60            // Zero I with explicit Q/U yields CASA's undefined linear fraction.
61            fraction[0] = q.hypot(u) / i;
62            angle[0] = 0.5 * u.atan2(q);
63        }
64        Ok(Self {
65            flux_jy,
66            reference,
67            spectral_index,
68            fraction,
69            angle,
70            rotation_measure,
71            tabulate,
72        })
73    }
74
75    /// Form the point model and the flux at the first reported frequency in
76    /// the spectral window. The caller chooses the channel list or median.
77    pub fn component(
78        &self,
79        direction: MDirection,
80        frequencies: &[MFrequency],
81        frame: &MeasFrame,
82    ) -> MsResult<(SkyComponent, [f64; 4])> {
83        let first = frequencies
84            .first()
85            .ok_or_else(|| invalid("empty setjy spectral window"))?;
86        if self.tabulate {
87            let stokes_flux = frequencies
88                .iter()
89                .map(|frequency| self.at_frequency(frequency, frame))
90                .collect::<MsResult<Vec<_>>>()?;
91            return sampled_point(direction, frequencies, stokes_flux);
92        }
93        let first_flux = self.at_frequency(first, frame)?;
94        let alpha = self.spectral_index.first().copied().unwrap_or_default();
95        let kind = SpectralKind::SpectralIndex(SpectralIndex::new(alpha).with_stokes_indices([
96            alpha,
97            0.,
98            self.rotation_measure,
99            0.,
100        ]));
101        let defined_polarization = self.flux_jy[1..].iter().any(|&flux| flux != 0.)
102            || (!self.fraction.is_empty() && !self.angle.is_empty());
103        let (flux, model) = if defined_polarization {
104            (first_flux, SpectralModel::new(first.clone(), kind))
105        } else {
106            (
107                self.flux_jy,
108                SpectralModel::new(self.reference.clone(), kind),
109            )
110        };
111        let [i, q, u, v] = flux;
112        Ok((
113            SkyComponent::new(
114                Flux::stokes_jy(i, q, u, v),
115                ComponentShape::point(direction),
116                model,
117            ),
118            first_flux,
119        ))
120    }
121
122    pub fn intensity_jy(&self) -> f64 {
123        self.flux_jy[0]
124    }
125
126    pub fn reports_spectral_index(&self) -> bool {
127        self.spectral_index
128            .first()
129            .is_some_and(|&index| index != 0.0)
130    }
131
132    pub fn has_spectral_dependence(&self) -> bool {
133        self.spectral_index.iter().any(|&index| index != 0.0)
134    }
135
136    pub fn intensity_at(&self, frequency: &MFrequency, frame: &MeasFrame) -> MsResult<f64> {
137        let i = self.intensity_jy();
138        let f = frequency.hz();
139        let f0 = self.reference.hz();
140        Ok(if self.spectral_index.len() > 1 {
141            let logarithm = (f / f0).log10();
142            i * 10_f64.powf(logarithm * polynomial(&self.spectral_index, logarithm))
143        } else {
144            let alpha = self.spectral_index.first().copied().unwrap_or_default();
145            let converted = frequency
146                .convert_to(self.reference.refer(), frame)
147                .map_err(invalid)?;
148            i * (converted.hz() / f0).powf(alpha)
149        })
150    }
151
152    fn at_frequency(&self, frequency: &MFrequency, frame: &MeasFrame) -> MsResult<[f64; 4]> {
153        const C_METRES_PER_SECOND: f64 = 299_792_458.;
154        let [i, q, u, v] = self.flux_jy;
155        let f = frequency.hz();
156        let f0 = self.reference.hz();
157        let i_at_f = self.intensity_at(frequency, frame)?;
158        let linear_model = !self.fraction.is_empty() && !self.angle.is_empty();
159        let (q_at_f, u_at_f) = if !linear_model {
160            (q, u)
161        } else {
162            let offset = (f - f0) / f0;
163            let linear = i_at_f * polynomial(&self.fraction, offset);
164            let lambda2 = (C_METRES_PER_SECOND / f).powi(2) - (C_METRES_PER_SECOND / f0).powi(2);
165            let phi = 2. * (polynomial(&self.angle, offset) + self.rotation_measure * lambda2);
166            let (sin, cos) = phi.sin_cos();
167            (linear * cos, linear * sin)
168        };
169        Ok([
170            i_at_f,
171            q_at_f,
172            u_at_f,
173            if i != 0. {
174                v * i_at_f / i
175            } else if linear_model {
176                0.
177            } else {
178                v
179            },
180        ])
181    }
182}

184pub fn sampled_point(
185    direction: MDirection,
186    frequencies: &[MFrequency],
187    stokes: Vec<[f64; 4]>,
188) -> MsResult<(SkyComponent, [f64; 4])> {
189    sampled_shape(ComponentShape::point(direction), frequencies, stokes)
190}
191
192pub(crate) fn sampled_shape(
193    shape: ComponentShape,
194    frequencies: &[MFrequency],
195    stokes: Vec<[f64; 4]>,
196) -> MsResult<(SkyComponent, [f64; 4])> {
197    let first = frequencies
198        .first()
199        .ok_or_else(|| invalid("empty setjy spectral window"))?;
200    let first_flux = stokes[0];
201    let model = if frequencies.len() == 1 {
202        // CASA changes only the reference value of a ConstantSpectrum.
203        SpectralModel::new(
204            MFrequency::new(first.hz(), FrequencyRef::LSRK),
205            SpectralKind::Constant,
206        )
207    } else {
208        let spectrum = TabularSpectrum::new(
209            first.refer(),
210            frequencies.iter().map(MFrequency::hz).collect(),
211            stokes,
212        )
213        .map_err(invalid)?;
214        SpectralModel::new(first.clone(), SpectralKind::Tabular(spectrum))
215    };
216    let [i, q, u, v] = first_flux;
217    let component = SkyComponent::new(Flux::stokes_jy(i, q, u, v), shape, model);
218    Ok((component, first_flux))
219}
220
221fn polynomial(coefficients: &[f64], x: f64) -> f64 {
222    coefficients
223        .iter()
224        .rev()
225        .fold(0., |value, &coefficient| value * x + coefficient)
226}
