13pub(super) fn select_input(
14    request: &MsTransformRequest,
15    grammar: RowGrammar,
16    options: &MsTransformTaskOptions,
17    partition: Option<TransformSelection>,
18    has_operation: bool,
19    started_at: Instant,
20) -> Result<SelectedInput, MsTransformError> {
21    let legacy = matches!(
22        options.legacy_tool,
23        Some(LegacyMsTool::Partition(_) | LegacyMsTool::Split(_))
24    );
25    let splitting = matches!(options.legacy_tool, Some(LegacyMsTool::Split(_)));
26    let stage_started_at = Instant::now();
27    let input = MeasurementSet::open(&request.input_ms).map_err(|source| {
28        MsTransformError::OpenMeasurementSet {
29            path: request.input_ms.display().to_string(),
30            source,
31        }
32    })?;
33    maybe_log_transform_progress(
34        "open_measurement_set",
35        stage_started_at.elapsed(),
36        started_at.elapsed(),
37    );
38    let stage_started_at = Instant::now();
39    let selection_error = |error| row_selection_error(request, error);
40    let budget = MsSelectionIoBudget::from_system_memory(
41        2,
42        std::mem::size_of::<ResolvedMsSelectionRow>(),
43        None,
44    )
45    .map_err(selection_error)?;
46    let spw = match request.spw.trim() {
47        "" => "*",
48        spw => spw,
49    };
50    let mut spw_selection =
51        input
52            .resolve_spw_selector(spw)
53            .map_err(|source| MsTransformError::InvalidSpw {
54                selector: request.spw.clone(),
55                reason: source.to_string(),
56            })?;
57    if options.legacy_continuum_output.is_some() {
58        // The legacy group iterator receives complete buffers. Only fitspw
59        // projects the channels used in the fit itself.
60        spw_selection.keep_all_channels();
61    }
62    if splitting {
63        let mut windows = BTreeSet::new();
64        if spw_selection
65            .channel_list()
66            .iter()
67            .any(|range| !windows.insert(range.spw_id))
68        {
69            return Err(MsTransformError::EmptySelection {
70                path: request.input_ms.display().to_string(),
71            });
72        }
73    }
74    if splitting
75        && options.time_averaging.is_some()
76        && spw_selection
77            .channel_list()
78            .first()
79            .is_some_and(|range| range.step > 1)
80    {
81        return Err(MsTransformError::Spectrum(
82            "Simultaneous time and channel averaging is not handled.".into(),
83        ));
84    }
85    let view_factory = options.input_view.as_deref();
86    let mut visibility = VisibilityColumns::resolve(
87        &input,
88        request.data_column,
89        options.visibility_columns.as_deref(),
90        &request.input_ms,
91        view_factory,
92    )?;
93    if legacy {
94        visibility.promote_singleton_visibility();
95    }
96    let mut selection = match partition {
97        Some(partition) => partition,
98        None => resolve_transform_selection(&input, request, grammar, budget, has_operation)?,
99    };
100    if options.legacy_continuum_output.is_some() {
101        selection.correlations = None;
102    }
103    maybe_log_transform_progress(
104        "select_rows",
105        stage_started_at.elapsed(),
106        started_at.elapsed(),
107    );
108    Ok(SelectedInput {
109        input,
110        budget,
111        spw_selection,
112        visibility,
113        selection,
114    })
115}

125    pub(super) channel_bins: BTreeMap<i32, Vec<Vec<usize>>>,

130pub(super) fn select_spectra(
131    request: &MsTransformRequest,
132    grammar: RowGrammar,
133    options: &MsTransformTaskOptions,
134    selected: &SelectedInput,
135    legacy_grid: Option<LegacySpectralGrid>,
136) -> Result<SelectedSpectra, MsTransformError> {
137    let SelectedInput {
138        input,
139        budget,
140        spw_selection,
141        ..
142    } = selected;
143    let legacy = matches!(
144        options.legacy_tool,
145        Some(LegacyMsTool::Partition(_) | LegacyMsTool::Split(_))
146    );
147    let splitting = matches!(options.legacy_tool, Some(LegacyMsTool::Split(_)));
148    let channel_selection = resolve_transform_channels(input, spw_selection, &request.spw)?;
149    let continuum = options
150        .continuum
151        .as_ref()
152        .map(|options| ContinuumFit::new(input, &channel_selection, options))
153        .transpose()?;
154    let task_average = grammar == RowGrammar::Transform
155        && !legacy
156        && (normalized_width(request.width) > 1 || options.channel_widths.is_some());
157    let widths = channel_widths_by_spw(
158        &channel_selection,
159        spw_selection,
160        request.width,
161        options.channel_widths.as_deref(),
162    )?;
163    if splitting
164        && options
165            .channel_widths
166            .as_ref()
167            .is_some_and(|widths| widths.contains(&0))
168    {
169        return Err(MsTransformError::InvalidSplitStep);
170    }
171    let average_metadata = task_average.then(|| {
172        channel_selection
173            .iter()
174            .map(|(&spw, selection)| (spw, selection.metadata_for_width(widths[&spw])))
175            .collect::<BTreeMap<_, _>>()
176    });
177    let regrid_options = options.regrid.as_ref();
178    let retained_combination = if options.preserve_ids && options.combine_spws {
179        let all_spws =
180            input
181                .resolve_spw_selector("*")
182                .map_err(|source| MsTransformError::InvalidSpw {
183                    selector: "*".into(),
184                    reason: source.to_string(),
185                })?;
186        let mut channels = resolve_transform_channels(input, &all_spws, "*")?;
187        channels.extend(
188            channel_selection
189                .iter()
190                .map(|(&id, selection)| (id, selection.clone())),
191        );
192        Some((all_spws, channels))
193    } else {
194        None
195    };
196    let channel_bins = channel_bins_by_spw(
197        &channel_selection,
198        &widths,
199        if regrid_options.is_some() {
200            None
201        } else {
202            average_metadata.as_ref()
203        },
204    )?;
205    let combined_grid =
206        if options.combine_spws || regrid_options.is_some_and(|options| options.nspw > 1) {
207            if options.channel_widths.is_some() {
208                return Err(MsTransformError::Spectrum(
209                    "If SPW combination is active, chabin cannot be an array".into(),
210                ));
211            }
212            combine::Grid::new(
213                input,
214                &channel_selection,
215                retained_combination
216                    .as_ref()
217                    .map_or(&channel_selection, |(_, channels)| channels),
218                normalized_width(request.width),
219                budget.available_bytes / budget.maximum_live_blocks,
220                matches!(legacy_grid, Some(LegacySpectralGrid::Cvel)),
221            )?
222        } else {
223            None
224        };
225    let ddid_to_spw = data_description_spw_map(input)?;
226    if splitting {
227        let mut seen = BTreeSet::new();
228        if ddid_to_spw
229            .values()
230            .filter(|spw| channel_selection.contains_key(spw))
231            .any(|spw| !seen.insert(spw))
232        {
233            return Err(MsTransformError::EmptySelection {
234                path: request.input_ms.display().to_string(),
235            });
236        }
237    }
238    Ok(SelectedSpectra {
239        channel_selection,
240        continuum,
241        task_average,
242        widths,
243        average_metadata,
244        retained_combination,
245        channel_bins,
246        combined_grid,
247        ddid_to_spw,
248    })
249}

251pub(super) fn filter_selected_rows(
252    request: &MsTransformRequest,
253    options: &MsTransformTaskOptions,
254    selected: &mut SelectedInput,
255    spectra: &SelectedSpectra,
256    started_at: Instant,
257    stage_started_at: Instant,
258) -> Result<(Vec<usize>, Vec<i32>), MsTransformError> {
259    let SelectedInput {
260        input,
261        budget,
262        spw_selection,
263        selection,
264        ..
265    } = selected;
266    let input = &*input;
267    let TransformSelection {
268        correlations, rows, ..
269    } = selection;
270    let SelectedSpectra {
271        channel_bins,
272        ddid_to_spw,
273        ..
274    } = spectra;
275    let budget = *budget;
276    let view_factory = options.input_view.as_deref();
277    let selection_error = |error| row_selection_error(request, error);
278    let output_spws = options
279        .legacy_continuum_output
280        .as_deref()
281        .map(|selector| {
282            input
283                .resolve_spw_selector(selector)
284                .map_err(|error| MsTransformError::InvalidSpw {
285                    selector: selector.to_owned(),
286                    reason: error.to_string(),
287                })
288                .map(|selection| {
289                    selection
290                        .spw_ids()
291                        .iter()
292                        .map(|&id| id as i32)
293                        .collect::<BTreeSet<_>>()
294                })
295        })
296        .transpose()?;
297    rows.retain(|row| {
298        spw_selection.selects_row(row.data_desc_id)
299            && channel_bins.contains_key(&row.spectral_window_id)
300            && output_spws
301                .as_ref()
302                .is_none_or(|spws| spws.contains(&row.spectral_window_id))
303    });
304    if let Some(view) = view_factory {
305        view.order_output_rows(input, rows)
306            .map_err(|error| MsTransformError::SelectRows {
307                path: request.input_ms.display().to_string(),
308                source: error.into(),
309            })?;
310    }
311    let (mut selected_rows, mut selected_ddids): (Vec<_>, Vec<_>) = rows
312        .iter()
313        .map(|row| (row.row_index, row.data_desc_id))
314        .unzip();
315    maybe_log_transform_progress(
316        "filter_rows_by_spw",
317        stage_started_at.elapsed(),
318        started_at.elapsed(),
319    );
320    if selected_rows.is_empty() {
321        return Err(MsTransformError::EmptySelection {
322            path: request.input_ms.display().to_string(),
323        });
324    }
325    if !request.keep_flags {
326        let stage_started_at = Instant::now();
327        let dimensions = transform_array_dimensions(
328            input,
329            &selected_ddids,
330            ddid_to_spw,
331            channel_bins,
332            correlations.as_ref(),
333            None,
334        )?;
335        let flag_rows = flag_filter_read_plan(selected_rows.len(), budget, dimensions.values())
336            .map_err(selection_error)?;
337        (selected_rows, selected_ddids) =
338            keep_unflagged_rows(input, &selected_rows, &selected_ddids, flag_rows, None)?;
339        if options.time_averaging.is_some() || view_factory.is_some() {
340            let kept = selected_rows.iter().copied().collect::<BTreeSet<_>>();
341            rows.retain(|row| kept.contains(&row.row_index));
342        }
343        maybe_log_transform_progress(
344            "drop_fully_flagged_rows",
345            stage_started_at.elapsed(),
346            started_at.elapsed(),
347        );
348        if selected_rows.is_empty() {
349            tracing::warn!(
350                input_ms = %request.input_ms.display(),
351                "mstransform selected no rows after dropping fully flagged data"
352            );
353            return Err(MsTransformError::EmptySelection {
354                path: request.input_ms.display().to_string(),
355            });
356        }
357    }
358    Ok((selected_rows, selected_ddids))
359}

361pub(super) struct PreparedSpectralInput<'a> {
362    pub(super) view: Option<Box<dyn PreparedTransformInputView + 'a>>,
363    pub(super) regrid: Option<regrid::Plan>,
364    pub(super) legacy_direct_band: bool,
365    pub(super) first_selected_row: usize,
366}

368pub(super) fn prepare_spectral_input<'a>(
369    input: &'a MeasurementSet,
370    request: &MsTransformRequest,
371    options: &MsTransformTaskOptions,
372    spectra: &mut SelectedSpectra,
373    rows: &[ResolvedMsSelectionRow],
374    selected_rows: &[usize],
375    budget: MsSelectionIoBudget,
376) -> Result<PreparedSpectralInput<'a>, MsTransformError> {
377    let SelectedSpectra {
378        channel_selection,
379        channel_bins,
380        combined_grid,
381        ..
382    } = spectra;
383    let view_factory = options.input_view.as_deref();
384    let spw = request.spw.trim();
385    let spw = if spw.is_empty() { "*" } else { spw };
386    let legacy_grid = match options.legacy_tool {
387        Some(LegacyMsTool::Regrid(grid)) => Some(grid),
388        _ => None,
389    };
390    let regrid_options = options.regrid.as_ref();
391    let legacy_direct_band = legacy_grid.is_some()
392        && combined_grid.is_none()
393        && options.time_averaging.is_none()
394        && options.channel_widths.is_none()
395        && normalized_width(request.width) == 1
396        && spw == "*";
397    let first_selected_row = selected_rows
398        .iter()
399        .copied()
400        .min()
401        .expect("nonempty row selection");
402    let prepared_view = if let Some(view) = view_factory {
403        let channels = channel_selection
404            .iter()
405            .map(|(&id, selection)| (id, selection.channels.as_slice()))
406            .collect();
407        Some(view.prepare(input, rows, &channels).map_err(|source| {
408            MsTransformError::MutateMeasurementSet {
409                path: request.input_ms.display().to_string(),
410                source: Box::new(source),
411            }
412        })?)
413    } else {
414        None
415    };
416    let regrid = regrid_options
417        .map(|options| {
418            regrid::Plan::new(
419                input,
420                (options, legacy_grid),
421                combined_grid.as_mut(),
422                channel_selection,
423                channel_bins,
424                (
425                    first_selected_row,
426                    ((legacy_direct_band
427                        && matches!(
428                            legacy_grid,
429                            Some(
430                                LegacySpectralGrid::CenteredChannels { .. }
431                                    | LegacySpectralGrid::CenteredNumeric { .. }
432                            )
433                        )
434                        && options.nspw == 1)
435                        || (matches!(legacy_grid, Some(LegacySpectralGrid::Cvel))
436                            && !options.outframe.trim().is_empty()))
437                    .then_some(selected_rows),
438                ),
439                budget.available_bytes / budget.maximum_live_blocks,
440            )
441        })
442        .transpose()?;
443    Ok(PreparedSpectralInput {
444        view: prepared_view,
445        regrid,
446        legacy_direct_band,
447        first_selected_row,
448    })
449}

451pub(super) fn row_selection_error(
452    request: &MsTransformRequest,
453    error: MsSelectionError,
454) -> MsTransformError {
455    MsTransformError::SelectRows {
456        path: request.input_ms.display().to_string(),
457        source: match error {
458            MsSelectionError::Domain(source) => source,
459            other => MsError::InvalidInput(other.to_string()),
460        },
461    }
