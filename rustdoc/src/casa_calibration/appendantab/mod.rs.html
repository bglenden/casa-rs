19/// Request to append external ANTAB gain and system-temperature calibration to a copy of an MS.
20pub struct AppendantabRequest {
21    /// Input MeasurementSet.
22    pub input: PathBuf,
23    /// Destination MeasurementSet.
24    pub output: PathBuf,
25    /// Input ANTAB text file.
26    pub antab: PathBuf,
27    /// Overwrite existing GAIN_CURVE / SYSCAL rows.
28    pub overwrite: bool,
29    /// Append SYSCAL temperatures.
30    pub append_tsys: bool,
31    /// Append GAIN_CURVE entries.
32    pub append_gain: bool,
33}
34
35/// Summary of an `appendantab` execution.
36#[derive(Serialize)]
37pub struct AppendantabSummary {
38    /// Output MeasurementSet created, if published.
39    pub output: Option<PathBuf>,
40    /// SYSCAL rows written.
41    pub tsys_rows: usize,
42    /// GAIN_CURVE records written.
43    pub gain_records: usize,
44}
45
46/// Error from `run_appendantab`, preserving parameter attribution where applicable.
47pub enum AppendantabError {
48    /// Execution failure without parameter attribution.
49    Message(String),
50    /// Parameter-attributed failure.
51    At(&'static str, String),
52}
53
54/// Execute `appendantab`, writing a new MeasurementSet when `output` is non-empty and absent.
55pub fn run_appendantab(
56    request: &AppendantabRequest,
57) -> Result<AppendantabSummary, AppendantabError> {
58    let input = &request.input;
59    let output = &request.output;
60    let antab = &request.antab;
61    let overwrite = request.overwrite;
62    let append_tsys = request.append_tsys;
63    let append_gain = request.append_gain;
64    let mut summary = AppendantabSummary {
65        output: None,
66        gain_records: 0,
67        tsys_rows: 0,
68    };
69    if output.as_os_str().is_empty() || output == input {
70        return Ok(summary);
71    }
72    if output.symlink_metadata().is_ok() {
73        // CASA reads the source metadata before declining an existing output.
74        let source = MeasurementSet::open(input)
75            .map_err(|error| AppendantabError::Message(message(error)))?;
76        drop(source);
77        return Ok(summary);
78    }
79    let input = crate::paths::normalized(input)
80        .map_err(|error| AppendantabError::Message(message(error)))?;
81    let output = crate::paths::normalized(output)
82        .map_err(|error| AppendantabError::Message(message(error)))?;
83    if output.starts_with(&input) {
84        return Err(AppendantabError::Message(
85            "The output MeasurementSet lies inside its input".into(),
86        ));
87    }
88    let parent = output.parent().unwrap_or_else(|| Path::new("."));
89    fs::create_dir_all(parent).map_err(|error| AppendantabError::Message(message(error)))?;
90    let staging = tempfile::Builder::new()
91        .prefix(".appendantab-")
92        .tempdir_in(parent)
93        .map_err(|error| AppendantabError::Message(message(error)))?;
94    let copied = staging.path().join("product.ms");
95
96    let preparation = crate::timing::Phase::start("appendantab_read");
97    #[cfg(unix)]
98    let ms = MeasurementSet::open_retained_read(&input);
99    #[cfg(not(unix))]
100    let ms = MeasurementSet::open(&input);
101    let ms = ms.map_err(|error| AppendantabError::Message(message(error)))?;
102    let metadata = Metadata::new(&ms).map_err(AppendantabError::Message)?;
103    let text =
104        fs::read_to_string(antab).map_err(|error| AppendantabError::Message(message(error)))?;
105    let text = Antab::read(&text).map_err(AppendantabError::Message)?;
106    let temps = compute::temperatures(&text, &metadata).map_err(AppendantabError::Message)?;
107    let gains = compute::gains(&text, &metadata).map_err(|error| match error {
108        compute::GainError::Missing(key) => AppendantabError::At("antab_header", key.to_string()),
109        compute::GainError::Invalid(message) => AppendantabError::Message(message),
110    })?;
111    drop(preparation);
112
113    let copy = crate::timing::Phase::start("appendantab_copy_ms");
114    let subtables = copy::stage(&ms, &input, &copied, append_gain || append_tsys)
115        .map_err(AppendantabError::Message)?;
116    drop(ms);
117    drop(copy);
118
119    let publication = crate::timing::Phase::start("appendantab_write");
120    table::write(
121        &copied,
122        &subtables,
123        &metadata,
124        append_gain.then_some(gains.as_slice()),
125        append_tsys.then_some(temps.as_slice()),
126        overwrite,
127    )
128    .map_err(AppendantabError::Message)?;
129    match rename_without_replacing(&copied, &output) {
130        Ok(()) => {
131            summary.output = Some(output);
132            summary.gain_records = if append_gain { gains.len() } else { 0 };
133            summary.tsys_rows = if append_tsys { temps.len() } else { 0 };
134        }
135        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {}
136        Err(error) => return Err(AppendantabError::Message(message(error))),
137    }
138    drop(publication);
139    crate::timing::report_process_memory();
140    Ok(summary)
141}
