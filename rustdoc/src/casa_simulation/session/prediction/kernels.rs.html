2//! Admitted preparation of simulation convolution functions.
3use super::*;
4
5impl Simulator {
6    /// Build `layout`'s kernels, admitted as they are built, as imaging
7    /// admits them: the spheroidal set's bytes, or the W-projection screens
8    /// ([`WPlanes::screens_bytes`]) and then the kernels cropped from them
9    /// ([`WScreens::kernel_bytes`](casa_imaging_operator::WScreens::kernel_bytes)),
10    /// the screens released once the kernels are built. The reservation
11    /// returned keeps what the kernels hold.
12    pub(super) fn admitted_kernels(
13        &self,
14        layout: &ModelLayout,
15    ) -> SimulatorResult<(Box<dyn ConvolutionFunctionSet>, Reservation)> {
16        let resources = &self.options.resources;
17        let (host, policy) = (resources.host(), resources.policy());
18        charge_process_caches(host, policy).map_err(SimulatorError::admission)?;
19        let admitted = |phase, memory| {
20            admit(host, policy, &Demand { phase, memory }).map_err(SimulatorError::admission)
21        };
22        let (geometry, polarization) = (layout.geometry(), layout.polarization());
23        let raised =
24            |error: OperatorError| SimulatorError::raised(format!("Caught exception: {error}"));
25        Ok(match layout.machine() {
26            FtMachine::Grid(_) => (
27                Box::new(Spheroidal::new(geometry, polarization)),
28                admitted(KERNEL_PHASE, Spheroidal::bytes(geometry))?,
29            ),
30            FtMachine::WProject(planes) => {
31                let count = WPlaneCount::Fixed(planes);
32                let screening = admitted(
33                    W_SCREENS_PHASE,
34                    WPlanes::screens_bytes(geometry, count).map_err(raised)?,
35                )?;
36                let screens = WPlanes::screens(geometry, polarization, count).map_err(raised)?;
37                let mut kernels = admitted(KERNEL_PHASE, screens.kernel_bytes())?;
38                let set = screens.finish().map_err(raised)?;
39                drop(screening);
40                kernels.retain(set.resident_bytes());
41                tracing::info!(
42                    planes,
43                    screen = set.screen_size(),
44                    bytes = set.resident_bytes(),
45                    "W projection using {planes} planes"
46                );
47                (Box::new(set), kernels)
48            }
49            FtMachine::Mosaic => {
50                let beams = layout.beams().expect("a mosaic layout has its beams");
51                let (dishes, windows) = (&beams.dishes, &beams.windows);
52                if beams.common {
53                    let screening = admitted(
54                        MOSAIC_SCREENS_PHASE,
55                        MosaicPb::simple_screens_bytes(geometry, windows).map_err(raised)?,
56                    )?;
57                    let pattern = beams
58                        .custom_pattern
59                        .as_ref()
60                        .map(super::super::voltage_patterns::BeamSource::pattern)
61                        .unwrap_or_else(|| dishes[0].pattern());
62                    let screens =
63                        MosaicPb::simple_pattern_screens(geometry, polarization, &pattern, windows)
64                            .map_err(raised)?;
65                    let mut kernels = admitted(KERNEL_PHASE, screens.kernel_bytes())?;
66                    let set = screens.finish().map_err(raised)?;
67                    drop(screening);
68                    kernels.retain(set.resident_bytes());
69                    return Ok((Box::new(set), kernels));
70                }
71                let (rules, frequency_hz) = (MosaicRules::Simulator, layout.image_frequency_hz());
72                tracing::info!("Performing Mosaic gridding for model images (uv convolution)");
73                let screening = admitted(
74                    MOSAIC_SCREENS_PHASE,
75                    MosaicPb::screens_bytes(rules, geometry, frequency_hz, dishes, windows)
76                        .map_err(raised)?,
77                )?;
78                let screens =
79                    MosaicPb::screens(rules, geometry, polarization, frequency_hz, dishes, windows)
80                        .map_err(raised)?;
81                let mut kernels = admitted(KERNEL_PHASE, screens.kernel_bytes())?;
82                let set = screens.finish().map_err(raised)?;
83                drop(screening);
84                kernels.retain(set.resident_bytes());
85                tracing::info!(
86                    dishes = set.dishes(),
87                    screen = set.screen_size(),
88                    bytes = set.resident_bytes(),
89                    "convolution function support: {}",
90                    set.max_half_support()[0]
91                );
92                (Box::new(set), kernels)
93            }
94        })
95    }
96}
