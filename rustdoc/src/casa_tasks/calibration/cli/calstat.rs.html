2//! CASA's unmasked statistics task over a named table column.
3
4use crate::{TaskFailure, TaskInvocation, TaskSuccess};
5use casa_contracts::CasaValue;
6use casa_tables::statistics::column_statistics;
7use casa_tables::{Table, TableOptions};
8
9pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
10    let path = task.input_path("caltable")?;
11    let axis = task.text("axis")?.unwrap_or_default();
12    let (column, projection) = match axis.as_str() {
13        "amp" | "amplitude" | "phase" | "imag" | "imaginary" | "real" => {
14            (task.text("datacolumn")?.unwrap_or_default(), axis.as_str())
15        }
16        column => (column.to_owned(), ""),
17    };
18    let column = match column.as_str() {
19        "corrected" => "CORRECTED_DATA".to_owned(),
20        "model" => "MODEL_DATA".to_owned(),
21        column => column.to_uppercase(),
22    };
23    let table = Table::open(TableOptions::new(&path))
24        .map_err(|error| TaskFailure::execution(error.to_string()).at("caltable"))?;
25    let mut lines = Vec::new();
26    let stats = column_statistics(&table, None, &column, projection, |line| lines.push(line))
27        .map_err(|error| TaskFailure::execution(error.to_string()))?;
28    let values = [
29        ("npts", "number of points"),
30        ("min", "minimum value"),
31        ("max", "maximum value"),
32        ("sum", "Sum of values"),
33        ("sumsq", "Sum of squared values"),
34    ];
35    let statistics = [
36        ("mean", "Mean of the values"),
37        ("var", "Variance of the values"),
38        ("stddev", "Standard deviation of the values"),
39        ("rms", "Root mean square"),
40        ("median", "Median of the pixel values"),
41        ("medabsdevmed", "Median of the deviations"),
42        ("quartile", "Quartile"),
43    ];
44    for (column, record) in &stats {
45        for (heading, metrics) in [("values", values.as_slice()), ("statistics", &statistics)] {
46            lines.push(format!("{column} {heading} ---"));
47            for &(key, label) in metrics {
48                let value = record[key];
49                let value = if key == "npts" {
50                    format!("{value:.0}")
51                } else {
52                    format!("{value:?}")
53                };
54                lines.push(format!("         -- {label} [{key}]: {value}"));
55            }
56        }
57    }
58    let value = CasaValue::dict(stats.iter().map(|(column, record)| {
59        (
60            column,
61            CasaValue::dict(
62                record
63                    .iter()
64                    .map(|(&key, &value)| (key, CasaValue::float(value))),
65            ),
66        )
67    }));
68    let mut success = TaskSuccess::new(value, &stats)?;
69    for line in lines {
70        success = success.with_output_line(line);
71    }
72    Ok(success)
73}
