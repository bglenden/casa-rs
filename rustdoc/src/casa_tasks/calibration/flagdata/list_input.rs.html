16pub(crate) fn tbuff_active(task: &TaskInvocation<'_>) -> bool {
17    match task.value("tbuff") {
18        Some(ParameterValue::Float(seconds)) => *seconds != 0.0,
19        Some(ParameterValue::Integer(seconds)) => *seconds != 0,
20        Some(ParameterValue::String(text)) => {
21            text.parse::<f64>().is_ok_and(|seconds| seconds != 0.0)
22        }
23        Some(ParameterValue::Array(values)) => !values.is_empty(),
24        _ => false,
25    }
26}

28pub(super) fn read(task: &TaskInvocation<'_>) -> Result<Vec<Assignments>, TaskFailure> {
29    let buffer = task.value("tbuff");
30    let padded = tbuff_active(task);
31    let padding = padding(buffer)?;
32    let (lines, from_file) = match task.value("inpfile") {
33        Some(ParameterValue::String(line)) if padded && !task.io_path(line).is_file() => {
34            (vec![line.clone()], false)
35        }
36        Some(ParameterValue::String(path)) => (read_lines(&task.io_path(path))?, true),
37        Some(ParameterValue::Array(values)) => {
38            let inputs = strings(values)?;
39            if inputs
40                .first()
41                .is_some_and(|path| task.io_path(path).is_file())
42            {
43                let mut lines = Vec::new();
44                for path in inputs {
45                    lines.extend(read_lines(&task.io_path(path))?);
46                }
47                (lines, true)
48            } else {
49                (inputs.into_iter().map(str::to_owned).collect(), false)
50            }
51        }
52        _ => return Err(error()),
53    };
54    let reasons = match task.value("reason") {
55        None | Some(ParameterValue::EmptyArray { .. }) => Vec::new(),
56        Some(ParameterValue::String(value)) if value == "any" => Vec::new(),
57        Some(ParameterValue::String(value)) => vec![value.as_str()],
58        Some(ParameterValue::Array(values)) => strings(values)?,
59        _ => return Err(error()),
60    };
61    let mut commands = Vec::new();
62    for line in lines {
63        if line.is_empty() || line.starts_with('#') {
64            continue;
65        }
66        if line.trim().is_empty() {
67            return Err(error());
68        }
69        let line = if from_file || !padded {
70            line.replace("true", "True").replace("false", "False")
71        } else {
72            line
73        };
74        let mut values = parse_assignments(&line).ok_or_else(error)?;
75        if let Some(padding) = padding {
76            pad_timerange(&mut values, padding)?;
77        }
78        if !reasons.is_empty()
79            && !matches!(values.get("reason"), Some(ParameterValue::String(reason)) if reasons.contains(&reason.as_str()))
80        {
81            continue;
82        }
83        if let Some(ParameterValue::String(path)) = values.get("addantenna")
84            && !path.is_empty()
85        {
86            let antenna = shadow::read_list(task, path).map_err(|_| error())?;
87            values.set_with_spelling("addantenna", antenna.value, antenna.command);
88        }
89        if values
90            .iter()
91            .any(|(key, value)| !valid_value(task, key, value))
92        {
93            return Err(error());
94        }
95        commands.push(values);
96    }
97    if commands.is_empty() {
98        Err(error())
99    } else {
100        Ok(commands)
101    }
102}

104fn valid_value(task: &TaskInvocation<'_>, key: &str, value: &ParameterValue) -> bool {
105    match (key, value) {
106        (
107            "vis" | "inpfile" | "flagbackup" | "tbuff" | "cmdreason" | "savepars" | "outfile"
108            | "display" | "action",
109            _,
110        ) => return false,
111        ("ntime", ParameterValue::Float(_))
112        | ("observation" | "quackinterval", ParameterValue::Integer(_))
113        | ("addantenna", ParameterValue::Table(_))
114        | ("timedev" | "freqdev", ParameterValue::Float(_) | ParameterValue::Array(_)) => {
115            return true;
116        }
117        ("chanbin", _) => return matches!(value, ParameterValue::Integer(_)),
118        _ => {}
119    }
120    task.session().bundle().surface.bindings().iter().find(|binding| binding.name == key)
121        .is_some_and(|binding| matches!(&binding.default, DefaultSpec::Literal { value: default } if discriminant(default) == discriminant(value)))
122}

124fn strings(values: &[ParameterValue]) -> Result<Vec<&str>, TaskFailure> {
125    values
126        .iter()
127        .map(|value| match value {
128            ParameterValue::String(value) => Ok(value.as_str()),
129            _ => Err(error()),
130        })
131        .collect()
132}

134fn read_lines(path: &Path) -> Result<Vec<String>, TaskFailure> {
135    let text = fs::read_to_string(path).map_err(|_| error())?;
136    Ok(text.lines().map(str::trim_end).map(str::to_owned).collect())
137}

139pub(super) fn error() -> TaskFailure {

141        .at("flag-list-input")

144fn padding(input: Option<&ParameterValue>) -> Result<Option<[f64; 2]>, TaskFailure> {
145    let numeric = |value: &ParameterValue| match value {
146        ParameterValue::Float(seconds) => Ok(*seconds),
147        ParameterValue::Integer(seconds) => Ok(*seconds as f64),
148        ParameterValue::String(text) => text.parse::<f64>().map_err(|_| error()),
149        _ => Err(error()),
150    };
151    match input {
152        None | Some(ParameterValue::EmptyArray { .. }) => Ok(None),
153        Some(ParameterValue::Array(values)) => match values.as_slice() {
154            [before] => Ok(Some([numeric(before)?; 2])),
155            [before, after] => Ok(Some([numeric(before)?, numeric(after)?])),
156            _ => Ok(None),
157        },
158        Some(value) => {
159            let seconds = numeric(value)?;
160            Ok((seconds != 0.0).then_some([seconds; 2]))
161        }
162    }
163}

165fn pad_timerange(values: &mut Assignments, [before, after]: [f64; 2]) -> Result<(), TaskFailure> {
166    let Some(ParameterValue::String(range)) = values.get("timerange") else {
167        return Ok(());
168    };
169    let Some((start, end)) = range.split_once('~') else {
170        return Ok(());
171    };
172    let seconds = Unit::new("s").expect("seconds is a defined unit");
173    let degrees = Unit::new("deg").expect("degrees is a defined unit");
174    let shifted = |text: &str, offset: f64| {
175        let quantity = Quantity::read(text).map_err(|_| error())?;
176        let value = quantity
177            .get_value_in(&seconds)
178            .or_else(|_| {
179                quantity
180                    .get_value_in(&degrees)
181                    .map(|degrees| degrees * 240.0)
182            })
183            .map_err(|_| error())?;
184        let day = (value + offset) / 86400.0;
185        if !day.is_finite() {
186            return Err(error());
187        }
188        let format = if text.bytes().filter(|&byte| byte == b'/').count() == 3 {
189            YMD
190        } else {
191            TIME
192        };
193        Ok(format_time(day, format, 9))
194    };
195    let range = format!("{}~{}", shifted(start, -before)?, shifted(end, after)?);
196    values.set("timerange", ParameterValue::String(range));
197    Ok(())
198}

200pub(super) fn command_line(values: &Assignments, reason: &str) -> String {
201    render_command_line(values, Some(reason))
202}

204pub(super) fn command_line_without_reason(values: &Assignments) -> String {
205    render_command_line(values, None)
206}

208fn render_command_line(values: &Assignments, reason: Option<&str>) -> String {
209    let render = |key: &str, value: &ParameterValue| match value {
210        ParameterValue::String(text) => format!("{key}='{}'", text.trim_matches('\'')),
211        value => format!("{key}={}", python_repr(value)),
212    };
213    let replacement = reason.map(|text| ParameterValue::String(text.to_owned()));
214    let mut arguments = values
215        .iter()
216        .filter(|(key, _)| reason.is_some() || key.as_str() != "reason")
217        .map(|(key, value)| {
218            if let Some(spelling) = values.spelling(key) {
219                return format!("{key}={spelling}");
220            }
221            let rendered_value = match replacement.as_ref() {
222                Some(replacement) if key == "reason" && !reason.unwrap_or_default().is_empty() => {
223                    replacement
224                }
225                _ => value,
226            };
227            render(key, rendered_value)
228        })
229        .collect::<Vec<_>>();
230    if let (Some(text), Some(replacement)) = (reason, replacement.as_ref())
231        && !text.is_empty()
232        && values.get("reason").is_none()
233    {
234        arguments.push(render("reason", replacement));
235    }
236    arguments.join(" ")
237}
