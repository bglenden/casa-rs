21pub(super) fn native_destination(expanded: &str, name: &str) -> PyResult<PathBuf> {
22    let absolute = rename_destination(expanded, name)?;
23    let parent = absolute
24        .parent()
25        .ok_or_else(|| raised("Invalid output image path"))?;
26    let directory = parent.canonicalize().map_err(|_| {
27        let argument = Path::new(expanded).parent().unwrap_or(Path::new("."));
28        raised(format!(
29            "RegularFile::copy/move: target directory {} is not writable",
30            argument.display()
31        ))
32    })?;
33    Ok(directory.join(
34        absolute
35            .file_name()
36            .ok_or_else(|| raised("Invalid output image name"))?,
37    ))
38}

40pub(super) fn check_path_relationship(source: &Path, target: &Path, name: &str) -> PyResult<()> {
41    if target.starts_with(source) {
42        return Err(raised(format!(
43            "Directory::move error on {} to {name}: Invalid argument",
44            source.display()
45        )));
46    }
47    if source.starts_with(target) {
48        return Err(raised(format!(
49            "Cannot rename image {} to {} because the destination contains the source image",
50            source.display(),
51            target.display()
52        )));
53    }
54    Ok(())
55}

57pub(super) enum NativeTableMove {
58    Renamed {
59        backup: Option<(TempDir, &'static str)>,
60        identity: Box<Metadata>,
61    },
62    Copied {
63        source_stage: TempDir,
64        target_stage: TempDir,
65        replaced: bool,
66        published: Box<CopiedDirectory>,
67    },
68}

70pub(super) struct CopiedDirectory {
71    identity: Metadata,
72    use_lock: Option<Arc<TableUseLock>>,
73    write_lock: Option<TableWriteLock>,
74    #[cfg(unix)]
75    _directory: Option<std::fs::File>,
76}

78impl CopiedDirectory {
79    fn matches(&self, path: &Path) -> bool {
80        path.symlink_metadata().is_ok_and(|actual| {
81            if let Some(lease) = &self.use_lock {
82                lease.matches_directory_metadata(&actual)
83            } else {
84                same_file_identity(&self.identity, &actual)
85            }
86        })
87    }
88}

90#[derive(Clone, Copy)]
91enum CopyKind {
92    NativeTable,
93    ReadOnlyDirectory,
94    #[cfg(unix)]
95    Symlink,
96}

98impl NativeTableMove {
99    pub(super) fn release_copy_write_lock(&mut self) {
100        if let Self::Copied { published, .. } = self {
101            published.write_lock = None;
102        }
103    }
104
105    pub(super) fn use_lock(&self) -> Option<Arc<TableUseLock>> {
106        match self {
107            Self::Renamed { .. } => None,
108            Self::Copied { published, .. } => published.use_lock.clone(),
109        }
110    }
111
112    pub(super) fn check_target(&self, target: &Path) -> PyResult<()> {
113        let matches = match self {
114            Self::Renamed { identity, .. } => matches_entry(identity, target),
115            Self::Copied { published, .. } => published.matches(target),
116        };
117        if matches {
118            Ok(())
119        } else {
120            Err(changed_target(target))
121        }
122    }
123
124    pub(super) fn finish(self, target: &Path) -> PyResult<()> {
125        if let Self::Copied {
126            source_stage,
127            target_stage,
128            published,
129            ..
130        } = self
131        {
132            let source = close_stage(source_stage, target);
133            let destination = close_stage(target_stage, target);
134            drop(published);
135            match (source, destination) {
136                (Ok(()), result) | (result, Ok(())) => return result,
137                (Err(source), Err(destination)) => {
138                    return Err(raised(format!("{source}. {destination}")));
139                }
140            }
141        }
142        Ok(())
143    }
144
145    pub(super) fn restore<T>(self, source: &Path, target: &Path, error: PyErr) -> PyResult<T> {
146        match self {
147            Self::Renamed { backup, identity } => {
148                restore_rename(source, target, backup, &identity, error)
149            }
150            Self::Copied {
151                source_stage,
152                target_stage,
153                replaced,
154                published,
155                ..
156            } => {
157                let target_result =
158                    restore_copied_target(target, &target_stage, replaced, &published);
159                let source_result =
160                    rename_without_replacing(&source_stage.path().join("source"), source);
161                if source_result.is_ok() && target_result.is_ok() {
162                    return Err(error);
163                }
164                let staged_source = source_stage.keep();
165                let staged_target = target_stage.keep();
166                Err(raised(format!(
167                    "{error}. Could not restore image names. Source: {source_result:?}. \
168                     Destination: {target_result:?}. Inspect {} and {}",
169                    staged_source.display(),
170                    staged_target.display()
171                )))
172            }
173        }
174    }
175}

210pub(super) fn move_read_only_directory(
211    source: &Path,
212    source_metadata: &Metadata,
213    target: &Path,
214    target_metadata: Option<&Metadata>,
215    check_copy: impl FnOnce(&Path) -> PyResult<()>,
216) -> PyResult<NativeTableMove> {
217    if check_write_access(source).is_err() {
218        // Linux requires write access to a moved directory when its parent
219        // changes. Claim this root at a sibling name instead.
220        return move_with_read_only_root(source, source_metadata, target, target_metadata);
221    }
222    let source_stage = tempdir_beside(source)?;
223    let claimed = source_stage.path().join("source");
224    rename_without_replacing(source, &claimed).map_err(raised)?;
225    if !matches_entry(source_metadata, &claimed) {
226        return restore_claim(
227            source,
228            source_stage,
229            raised(format!(
230                "The source image entry at {} has changed",
231                source.display()
232            )),
233        );
234    }
235    let moved = if target_metadata.is_some() {
236        exchange_directories(&claimed, target)
237    } else {
238        rename_without_replacing(&claimed, target)
239    };
240    match moved {
241        Ok(()) => {
242            let destination_changed =
243                target_metadata.is_some_and(|expected| !matches_entry(expected, &claimed));
244            let moved = NativeTableMove::Renamed {
245                backup: target_metadata.map(|_| (source_stage, "source")),
246                identity: Box::new(source_metadata.clone()),
247            };
248            if destination_changed {
249                moved.restore(source, target, changed_target(target))
250            } else {
251                Ok(moved)
252            }
253        }
254        Err(error) if error.kind() == ErrorKind::CrossesDevices => copy_claimed_directory(
255            source,
256            source_metadata,
257            target,
258            target_metadata,
259            source_stage,
260            CopyKind::ReadOnlyDirectory,
261            check_copy,
262        ),
263        Err(error) => restore_claim(source, source_stage, raised(error)),
264    }
265}

267fn move_with_read_only_root(
268    source: &Path,
269    identity: &Metadata,
270    target: &Path,
271    target_metadata: Option<&Metadata>,
272) -> PyResult<NativeTableMove> {
273    let claimed = sibling_slot(source)?;
274    let backup = target_metadata
275        .map(|_| tempdir_beside(source))
276        .transpose()?;
277    rename_without_replacing(source, &claimed).map_err(raised)?;
278    let restore = |error| restore_named_claim(source, &claimed, error);
279    if !matches_entry(identity, &claimed) {
280        return restore(raised(format!(
281            "The source image entry at {} has changed",
282            source.display()
283        )));
284    }
285    let moved = match &backup {
286        Some(_) => exchange_directories(&claimed, target),
287        None => rename_without_replacing(&claimed, target),
288    };
289    if let Err(error) = moved {
290        let error = if error.kind() == ErrorKind::CrossesDevices {
291            raised(format!(
292                "Cannot move MIRIAD image {} across filesystems: the source directory is not writable",
293                source.display()
294            ))
295        } else {
296            raised(error)
297        };
298        return restore(error);
299    }
300    let changed = target_metadata.is_some_and(|expected| !matches_entry(expected, &claimed));
301    let backup = if let Some(backup) = backup {
302        let previous = backup.path().join("previous");
303        if let Err(error) = rename_without_replacing(&claimed, &previous) {
304            return Err(raised(format!(
305                "Image moved to {} but the previous entry could not be staged: {error}. Inspect {}",
306                target.display(),
307                claimed.display()
308            )));
309        }
310        Some((backup, "previous"))
311    } else {
312        None
313    };
314    let moved = NativeTableMove::Renamed {
315        backup,
316        identity: Box::new(identity.clone()),
317    };
318    if changed {
319        moved.restore(source, target, changed_target(target))
320    } else {
321        Ok(moved)
322    }
323}

325fn move_image_directory(
326    source: &Path,
327    source_metadata: &Metadata,
328    target: &Path,
329    target_metadata: Option<&Metadata>,
330    kind: CopyKind,
331    prepare_copy: impl FnOnce(&Path) -> PyResult<()>,
332) -> PyResult<NativeTableMove> {
333    let Some(expected) = target_metadata else {
334        return match rename_without_replacing(source, target) {
335            Ok(()) => Ok(NativeTableMove::Renamed {
336                backup: None,
337                identity: Box::new(source_metadata.clone()),
338            }),
339            Err(error) if error.kind() == ErrorKind::CrossesDevices => {
340                move_across_devices(source, source_metadata, target, None, kind, prepare_copy)
341            }
342            Err(error) => Err(raised(error)),
343        };
344    };
345    let backup = tempdir_beside(source)?;
346    if let Err(error) = exchange_directories(source, target) {
347        return if error.kind() == ErrorKind::CrossesDevices {
348            move_across_devices(
349                source,
350                source_metadata,
351                target,
352                target_metadata,
353                kind,
354                prepare_copy,
355            )
356        } else {
357            Err(raised(error))
358        };
359    }
360    let previous = backup.path().join("previous");
361    if let Err(error) = rename_without_replacing(source, &previous) {
362        let restore = if matches_entry(expected, source) && matches_entry(source_metadata, target) {
363            exchange_directories(source, target)
364        } else {
365            Err(io::Error::other(
366                "an image entry changed after the exchange",
367            ))
368        };
369        if let Err(restore) = restore {
370            return Err(raised(format!(
371                "Could not complete image rename ({error}) or restore by exchanging {} and {} ({restore})",
372                source.display(),
373                target.display()
374            )));
375        }
376        return Err(raised(error));
377    }
378    let moved = NativeTableMove::Renamed {
379        backup: Some((backup, "previous")),
380        identity: Box::new(source_metadata.clone()),
381    };
382    if !matches_entry(expected, &previous) {
383        return moved.restore(source, target, changed_target(target));
384    }
385    Ok(moved)
386}

388fn tempdir_beside(path: &Path) -> PyResult<TempDir> {
389    let parent = path
390        .parent()
391        .ok_or_else(|| raised("Image path has no parent directory"))?;
392    tempfile::tempdir_in(parent).map_err(raised)
393}

395fn sibling_slot(path: &Path) -> PyResult<PathBuf> {
396    let reservation = tempdir_beside(path)?;
397    let slot = reservation.path().to_path_buf();
398    reservation.close().map_err(raised)?;
399    Ok(slot)
400}

402fn matches_entry(expected: &Metadata, path: &Path) -> bool {
403    path.symlink_metadata()

414fn move_across_devices(
415    source: &Path,
416    source_metadata: &Metadata,
417    target: &Path,
418    target_metadata: Option<&Metadata>,
419    kind: CopyKind,
420    prepare_copy: impl FnOnce(&Path) -> PyResult<()>,
421) -> PyResult<NativeTableMove> {
422    let source_stage = tempdir_beside(source)?;
423    rename_without_replacing(source, &source_stage.path().join("source")).map_err(raised)?;
424    copy_claimed_directory(
425        source,
426        source_metadata,
427        target,
428        target_metadata,
429        source_stage,
430        kind,
431        prepare_copy,
432    )
433}

435fn copy_claimed_directory(
436    source: &Path,
437    source_metadata: &Metadata,
438    target: &Path,
439    target_metadata: Option<&Metadata>,
440    source_stage: TempDir,
441    kind: CopyKind,
442    prepare_copy: impl FnOnce(&Path) -> PyResult<()>,
443) -> PyResult<NativeTableMove> {
444    let target_stage = match tempdir_beside(target) {
445        Ok(stage) => stage,
446        Err(error) => return restore_claim(source, source_stage, error),
447    };
448    let claimed = source_stage.path().join("source");
449    let copied = target_stage.path().join("previous");
450
451    let install = || {
452        if !matches_entry(source_metadata, &claimed) {
453            return Err(raised(format!(
454                "The source image entry at {} has changed",
455                source.display()
456            )));
457        }
458        if matches!(kind, CopyKind::ReadOnlyDirectory | CopyKind::NativeTable) {
459            check_tree_removal_access(&claimed).map_err(|error| {
460                let description = match kind {
461                    CopyKind::ReadOnlyDirectory => "MIRIAD image",
462                    _ => "native image",
463                };
464                raised(format!(
465                    "Cannot move {description} {} across filesystems: {error}",
466                    source.display()
467                ))
468            })?;
469        }
470        match kind {
471            #[cfg(unix)]
472            CopyKind::Symlink => {
473                std::os::unix::fs::symlink(std::fs::read_link(&claimed).map_err(raised)?, &copied)
474                    .map_err(raised)?
475            }
476            CopyKind::ReadOnlyDirectory => {
477                copy_locked_external_tree(&claimed, &copied, |_| false).map_err(raised)?
478            }
479            CopyKind::NativeTable => copy_locked_table_tree(&claimed, &copied).map_err(raised)?,
480        }
481        prepare_copy(&copied)?;
482        let (use_lock, write_lock) = match kind {
483            CopyKind::NativeTable => (
484                Some(Arc::new(TableUseLock::acquire(&copied).map_err(raised)?)),
485                Some(TableWriteLock::acquire(&copied, 1).map_err(raised)?),
486            ),
487            _ => (None, None),
488        };
489        let published = CopiedDirectory {
490            identity: copied.symlink_metadata().map_err(raised)?,
491            use_lock,
492            write_lock,
493            #[cfg(unix)]
494            _directory: matches!(kind, CopyKind::ReadOnlyDirectory)
495                .then(|| casa_tables::publication::open_directory_identity(&copied))
496                .transpose()
497                .map_err(raised)?,
498        };
499        if target_metadata.is_some() {
500            exchange_directories(&copied, target).map_err(raised)?;
501        } else {
502            rename_without_replacing(&copied, target).map_err(raised)?;
503        }
504        Ok::<_, PyErr>(published)
505    };
506    let published = match install() {
507        Ok(locks) => locks,
508        Err(error) => return restore_claim(source, source_stage, error),
509    };
510    let moved = NativeTableMove::Copied {
511        source_stage,
512        target_stage,
513        replaced: target_metadata.is_some(),
514        published: Box::new(published),
515    };
516    if target_metadata.is_some_and(|expected| !matches_entry(expected, &copied)) {
517        return moved.restore(source, target, changed_target(target));
518    }
519    Ok(moved)
520}

522fn restore_claim<T>(source: &Path, stage: TempDir, error: PyErr) -> PyResult<T> {
523    if let Err(restore) = rename_without_replacing(&stage.path().join("source"), source) {
524        let recovery = stage.keep();
525        return Err(raised(format!(
526            "{error}. Could not restore image {}: {restore}. Inspect {}",
527            source.display(),
528            recovery.display()
529        )));
530    }
531    Err(error)
532}
533
534fn restore_named_claim<T>(source: &Path, claimed: &Path, error: PyErr) -> PyResult<T> {
535    if let Err(restore) = rename_without_replacing(claimed, source) {
536        return Err(raised(format!(
537            "{error}. Could not restore image {}: {restore}. Inspect {}",
538            source.display(),
539            claimed.display()
540        )));
541    }
542    Err(error)
543}

545fn restore_copied_target(
546    target: &Path,
547    stage: &TempDir,
548    replaced: bool,
549    published: &CopiedDirectory,
550) -> io::Result<()> {
551    withdraw_moved_entry(target, &stage.path().join("withdrawn"), |path| {
552        published.matches(path)
553    })?;
554    if replaced {
555        rename_without_replacing(&stage.path().join("previous"), target)?;
556    }
557    Ok(())
558}
559
560fn withdraw_moved_entry(
561    target: &Path,
562    withdrawn: &Path,
563    matches: impl Fn(&Path) -> bool,
564) -> io::Result<()> {
565    let changed = || {
566        io::Error::other(format!(
567            "The destination image entry at {} has changed",
568            target.display()
569        ))
570    };
571    if !matches(target) {
572        return Err(changed());
573    }
574    rename_without_replacing(target, withdrawn)?;
575    if !matches(withdrawn) {
576        // The entry changed between preflight and withdrawal. Return it
577        // without replacing any newer arrival, or leave it in the stage.
578        rename_without_replacing(withdrawn, target)?;
579        return Err(changed());
580    }
581    Ok(())
582}

584fn check_tree_removal_access(path: &Path) -> io::Result<()> {
585    let mut pending = vec![path.to_path_buf()];
586    while let Some(directory) = pending.pop() {
587        let mut entries = directory.read_dir()?.peekable();
588        if entries.peek().is_some() {
589            check_write_access(&directory)?;
590        }
591        for entry in entries {
592            let entry = entry?;
593            if entry.file_type()?.is_dir() {
594                pending.push(entry.path());
595            }
596        }
597    }
598    Ok(())
599}

601fn close_stage(stage: TempDir, target: &Path) -> PyResult<()> {
602    let recovery = stage.path().to_path_buf();
603    stage.close().map_err(|error| {
604        raised(format!(
605            "Image was moved to {} but cleanup failed: {error}. Inspect {}",
606            target.display(),
607            recovery.display()
608        ))
609    })
610}

612fn restore_rename<T>(
613    source: &Path,
614    target: &Path,
615    backup: Option<(TempDir, &'static str)>,
616    identity: &Metadata,
617    error: PyErr,
618) -> PyResult<T> {
619    let (stage, previous) = match backup {
620        Some((stage, previous)) => (stage, Some(previous)),
621        None => (
622            tempdir_beside(source).map_err(|restore| {
623                raised(format!(
624                    "{error}. Could not stage rollback for {}: {restore}",
625                    target.display()
626                ))
627            })?,
628            None,
629        ),
630    };
631    let withdrawn = match sibling_slot(target) {
632        Ok(path) => path,
633        Err(restore) => {
634            let recovery = stage.keep();
635            return Err(raised(format!(
636                "{error}. Could not stage rollback: {restore}. Inspect {}",
637                recovery.display()
638            )));
639        }
640    };
641    let restore = (|| {
642        withdraw_moved_entry(target, &withdrawn, |path| matches_entry(identity, path))?;
643        rename_without_replacing(&withdrawn, source)?;
644        if let Some(previous) = previous {
645            rename_without_replacing(&stage.path().join(previous), target)?;
646        }
647        Ok::<(), io::Error>(())
648    })();
649    if let Err(restore) = restore {
650        let recovery = stage.keep();
651        return Err(raised(format!(
652            "{error}. Could not restore image names: {restore}. Inspect {} and {}",
653            recovery.display(),
654            withdrawn.display()
655        )));
656    }
657    Err(error)
658}

699    fn rollback_preserves_a_concurrently_replaced_destination() {

701        for overwrite in [false, true] {

722            let NativeTableMove::Copied { target_stage, .. } = &moved else {

786        Python::initialize();

793        fs::create_dir(&retained).unwrap();
794        fs::write(retained.join("keep"), b"data awaiting removal").unwrap();
795        fs::set_permissions(&retained, fs::Permissions::from_mode(0o555)).unwrap();
796        if fs::write(retained.join("permission-probe"), []).is_ok() {
797            // A privileged test process can still remove read-only entries.
798            return;
799        }
800
801        let error = close_stage(stage, &published).unwrap_err();
802        fs::set_permissions(&retained, fs::Permissions::from_mode(0o755)).unwrap();
803        let message = error.to_string();
804        assert!(message.contains("cleanup failed"), "{message}");
805        assert!(message.contains(recovery.to_str().unwrap()), "{message}");
806        assert_eq!(
807            fs::read(retained.join("keep")).unwrap(),
808            b"data awaiting removal"
809        );
810        assert_eq!(fs::read(&published).unwrap(), b"the published copy");
