2//! Scan numbers of a MeasurementSet created from several observations, as
3//! CASA's `concat` offsets them (`MSConcat::concatenate`,
4//! `MSConcat.cc:1419-1590`), so that the scans of an appended input do not
5//! repeat those already written.
6//!
7//! Inputs are appended in order. For each, the destination's observing
8//! sessions are taken in the order they first appear in its MAIN, each with
9//! the smallest and largest scan number it holds (the numbers written,
10//! after earlier offsets). Each session's offset is:
11//! - for the first, its smallest scan less 1;
12//! - for one an incoming session matched (its counterpart), the previous
13//!   session's largest scan plus 1, less the incoming input's smallest scan
14//!   of all;
15//! - for any other, its smallest scan less 1;
16//!
17//! and a session new to the destination takes the default, the
18//! destination's largest scan (at least 0) plus 1 less the incoming
19//! smallest. Every offset is floored at 0. An incoming row's scan is offset
20//! only when its observation ID changed, comparing the number it is
21//! written with against its own, as CASA compares them.
22
23use std::collections::BTreeMap;
24
25/// One observing session of the destination as the offsets see it: its
26/// observation ID and the smallest and largest scan number it holds.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub(super) struct Session {
29    pub(super) id: i32,
30    pub(super) scans: [i32; 2],
31}
32
33/// One incoming observing session: its own observation ID, the ID it is
34/// written with, and whether it matched a destination session.
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36pub(super) struct Incoming {
37    pub(super) own: i32,
38    pub(super) written: i32,
39    pub(super) counterpart: bool,
40}
41
42/// The offset added to the scans of each of `incoming`, appended to a
43/// destination whose sessions are `destination` in the order they first
44/// appear, `smallest` being the smallest scan of the whole incoming input.
45pub(super) fn offsets(destination: &[Session], incoming: &[Incoming], smallest: i32) -> Vec<i32> {
46    let largest = destination
47        .iter()
48        .map(|session| session.scans[1])
49        .fold(0, i32::max);
50    let default = (largest + 1 - smallest).max(0);
51    let matched: Vec<i32> = incoming
52        .iter()
53        .filter(|session| session.counterpart)
54        .map(|session| session.written)
55        .collect();
56    let mut by_id = BTreeMap::new();
57    for (index, session) in destination.iter().enumerate() {
58        let offset = if index != 0 && matched.contains(&session.id) {
59            destination[index - 1].scans[1] + 1 - smallest
60        } else {
61            session.scans[0] - 1
62        };
63        by_id.insert(session.id, offset.max(0));
64    }
65    incoming
66        .iter()
67        .map(|session| {
68            if session.written == session.own {
69                0
70            } else {
71                by_id.get(&session.written).copied().unwrap_or(default)
72            }
73        })
74        .collect()
75}

77#[cfg(test)]
78mod tests {
79    use super::{Incoming, Session, offsets};
80
81    fn session(id: i32, scans: [i32; 2]) -> Session {
82        Session { id, scans }
83    }
84
85    fn incoming(own: i32, written: i32, counterpart: bool) -> Incoming {
86        Incoming {
87            own,
88            written,
89            counterpart,
90        }
91    }
92
93    /// A new session is offset past every scan already written.
94    #[test]
95    fn a_new_session_follows_every_scan_written() {
96        let destination = [session(0, [1, 20])];
97        assert_eq!(offsets(&destination, &[incoming(0, 1, false)], 1), [20]);
98        assert_eq!(
99            offsets(&destination, &[incoming(0, 1, false)], 100),
100            [0],
101            "floored at 0"
102        );
103    }
104
105    /// The destination's MAIN visits observation 1 (scans up to 10) before
106    /// observation 0 (up to 20): an incoming session matching observation 0
107    /// continues after observation 1's largest scan, the previous in order
108    /// of appearance, not after observation 0's own.
109    #[test]
110    fn a_counterpart_continues_from_the_previous_session_in_order_of_appearance() {
111        let destination = [session(1, [1, 10]), session(0, [11, 20])];
112        assert_eq!(offsets(&destination, &[incoming(1, 0, true)], 1), [10]);
113    }
114
115    /// Incoming sessions whose smallest scans are 1 and 100 are offset by
116    /// the input's smallest of all.
117    #[test]
118    fn offsets_count_from_the_inputs_smallest_scan() {
119        let destination = [session(0, [1, 20])];
120        let incoming = [incoming(0, 1, false), incoming(1, 2, false)];
121        assert_eq!(offsets(&destination, &incoming, 1), [20, 20]);
122    }
123
124    /// A session keeping its observation ID keeps its scans, whatever it
125    /// matched: the split-and-concatenated case.
126    #[test]
127    fn a_session_keeping_its_id_keeps_its_scans() {
128        let destination = [session(0, [1, 20])];
129        assert_eq!(offsets(&destination, &[incoming(0, 0, true)], 1), [0]);
130        assert_eq!(offsets(&destination, &[incoming(0, 0, false)], 5), [0]);
131    }
132
133    /// The first session in order of appearance is offset by its own
134    /// smallest scan less 1 even when matched; a later counterpart uses the
135    /// previous session's largest.
136    #[test]
137    fn the_first_session_takes_its_smallest_scan() {
138        let destination = [session(0, [3, 8]), session(1, [9, 15])];
139        let incoming = [incoming(1, 0, true), incoming(0, 1, true)];
140        assert_eq!(offsets(&destination, &incoming, 1), [2, 8]);
141    }
142}
