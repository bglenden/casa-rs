29impl PyMsTool {
30    /// Fit the selected continuum using correlation-specific visibility weights.
31    #[allow(clippy::too_many_arguments)]
32    #[pyo3(signature = (field = None, fitspw = None, spw = None, solint = None, fitorder = 0, mode = "subtract".to_owned()), text_signature = "($self, field=[], fitspw=[], spw=[], solint=[], fitorder=0, mode='subtract')")]
33    fn continuumsub(
34        &self,
35        py: Python<'_>,
36        field: Option<&Bound<'_, PyAny>>,
37        fitspw: Option<&Bound<'_, PyAny>>,
38        spw: Option<&Bound<'_, PyAny>>,
39        solint: Option<&Bound<'_, PyAny>>,
40        fitorder: i64,
41        #[pyo3(from_py_with = mode_argument)] mode: String,
42    ) -> PyResult<bool> {
43        let options = scratch_options(field, fitspw, spw, solint, fitorder, mode, false)?;
44        py.detach(|| self.fit_continuum_scratch(&options))
45    }
46
47    /// Fit the continuum with the legacy visibility iterator's flag handling.
48    #[allow(clippy::too_many_arguments)]
49    #[pyo3(signature = (field = None, fitspw = None, spw = None, solint = None, fitorder = 0, mode = "subtract".to_owned()), text_signature = "($self, field=[], fitspw=[], spw=[], solint=[], fitorder=0, mode='subtract')")]
50    fn continuumsubold(
51        &self,
52        py: Python<'_>,
53        field: Option<&Bound<'_, PyAny>>,
54        fitspw: Option<&Bound<'_, PyAny>>,
55        spw: Option<&Bound<'_, PyAny>>,
56        solint: Option<&Bound<'_, PyAny>>,
57        fitorder: i64,
58        #[pyo3(from_py_with = mode_argument)] mode: String,
59    ) -> PyResult<bool> {
60        let options = scratch_options(field, fitspw, spw, solint, fitorder, mode, true)?;
61        py.detach(|| self.fit_continuum_scratch(&options))
62    }
63
64    /// Write a selected MeasurementSet after removing a frequency polynomial.
65    #[allow(clippy::too_many_arguments)]
66    #[pyo3(signature = (outputms = String::new(), fitspw = None, fitorder = 1, combine = String::new(), spw = None, unionspw = None, field = None, scan = None, intent = String::new(), correlation = String::new(), obs = String::new(), whichcol = "CORRECTED_DATA".to_owned()), text_signature = "($self, outputms='', fitspw=[], fitorder=1, combine='', spw='*', unionspw='*', field='', scan='', intent='', correlation='', obs='', whichcol='CORRECTED_DATA')")]
67    fn contsub(
68        &self,
69        py: Python<'_>,
70        #[pyo3(from_py_with = output_argument)] outputms: String,
71        fitspw: Option<&Bound<'_, PyAny>>,
72        fitorder: i64,
73        #[pyo3(from_py_with = combine_argument)] combine: String,
74        spw: Option<&Bound<'_, PyAny>>,
75        unionspw: Option<&Bound<'_, PyAny>>,
76        field: Option<&Bound<'_, PyAny>>,
77        scan: Option<&Bound<'_, PyAny>>,
78        #[pyo3(from_py_with = intent_argument)] intent: String,
79        #[pyo3(from_py_with = correlation_argument)] correlation: String,
80        #[pyo3(from_py_with = obs_argument)] obs: String,
81        #[pyo3(from_py_with = whichcol_argument)] whichcol: String,
82    ) -> PyResult<bool> {
83        let union = all_if_minus_one(unionspw)?.unwrap_or_else(|| "*".into());
84        let output = all_if_minus_one(spw)?.unwrap_or_else(|| "*".into());
85        let fields = all_if_minus_one(field)?;
86        let selection = MsSelection {
87            field: fields.clone(),
88            scan: optional_expression(scan)?,
89            intent: nonempty(intent),
90            correlation: nonempty(correlation.to_ascii_uppercase()),
91            observation: nonempty(obs),
92            ..Default::default()
93        };
94        let Some(columns) = visibility_columns(&whichcol)? else {
95            return Ok(false);
96        };
97        let parameters = format!(
98            "fieldids={} spwids={output} whichcol='{whichcol}'",
99            fields.unwrap_or_default()
100        );
101        let fit_spw = all_if_minus_one(fitspw)?.unwrap_or_default();
102        let input_view = if let Ok(order) = usize::try_from(fitorder) {
103            let [column] = columns.as_slice() else {
104                return Ok(false);
105            };
106            if !matches!(
107                column,
108                OptionalMainColumn::Data
109                    | OptionalMainColumn::CorrectedData
110                    | OptionalMainColumn::ModelData
111            ) {
112                return Ok(false);
113            }
114            Some(Arc::new(LegacyContinuumTransform {
115                column: column.name(),
116                order,
117                fit_spw,
118                union: MsSelection {
119                    spw: Some(union.clone()),
120                    ..selection.clone()
121                },
122                combine: combine.to_ascii_lowercase(),
123            })
124                as Arc<dyn casa_mstransform::TransformInputView>)
125        } else {
126            None
127        };
128        let options = MsTransformTaskOptions {
129            visibility_columns: Some(columns),
130            input_view,
131            legacy_continuum_output: (fitorder >= 0).then_some(output),
132            legacy_tool: Some(LegacyMsTool::Partition(LegacyTimeOptions {
133                combine_observations: false,
134            })),
135            ..Default::default()
136        };
137        self.write_subms(
138            py,
139            outputms,
140            union,
141            selection,
142            options,
143            ("contsub", Some(parameters)),
144        )
145    }
146}

148impl PyMsTool {
149    fn fit_continuum_scratch(&self, options: &ContinuumScratchOptions) -> PyResult<bool> {
150        let mut attachment = self.attached.lock().map_err(runtime_error)?;
151        let attached = attachment
152            .as_mut()
153            .ok_or_else(|| runtime_error("Table column DATA_DESC_ID is unknown"))?;
154        if !attached.writable {
155            return Err(runtime_error("MeasurementSet is not writable"));
156        }
157        let link = attached.working_metadata_link();
158        let view = link.working_view().map_err(runtime_error)?;
159        run_continuum_scratch(&link, view.path(), options, &mut attached.write_locks)?;
160        Ok(true)
161    }
162}

164fn run_continuum_scratch(
165    link: &MetadataLink,
166    path: &Path,
167    options: &ContinuumScratchOptions,
168    retained: &mut BTreeMap<PathBuf, TableWriteLock>,
169) -> PyResult<()> {
170    let initial = VisibilityMain::open(path)?;
171    let view = MeasurementSet::open(path).map_err(runtime_error)?;
172    let table = view.main_table();
173    let observed = if table.column_accessor("FLOAT_DATA").is_ok() {
174        "FLOAT_DATA"
175    } else {
176        "DATA"
177    };
178    let rows: Vec<_> = (0..view.row_count()).collect();
179    let targets = initial.write_targets(&rows, &[observed])?;
180    let root = table
181        .reference_root_projection(&[], &[])
182        .map_err(runtime_error)?
183        .map(|projection| projection.path.canonicalize().map_err(runtime_error))
184        .transpose()?;
185    let paths = targets
186        .iter()
187        .map(|target| target.path.clone())
188        .chain(root.iter().cloned())
189        .collect::<BTreeSet<_>>();
190    let mut temporary = Vec::new();
191    for path in &paths {
192        if !retained.values().any(|lock| lock.matches_table_path(path)) {
193            temporary.push(TableWriteLock::acquire_to_reread(path, 0).map_err(runtime_error)?);
194        }
195    }
196    let mut locks: BTreeMap<_, _> = retained
197        .values_mut()
198        .chain(&mut temporary)
199        .map(|lock| (lock.path().to_path_buf(), lock))
200        .collect();
201    let mut written = BTreeSet::new();
202    link.verify_directory().map_err(runtime_error)?;
203    initial.verify_write_layout(&written)?;
204    let publication = root
205        .as_ref()
206        .map(|root| {
207            table
208                .prepare_reference_columns_holding(&SCRATCH, continuum_lock(&mut locks, root)?)
209                .map_err(runtime_error)
210        })
211        .transpose()?
212        .flatten();
213    let scratch = publication.as_ref().map_or(SCRATCH, |plan| {
214        [plan.columns[0].as_str(), plan.columns[1].as_str()]
215    });
216    for target in targets {
217        initial.verify_write_layout(&written)?;
218        let mut member = MeasurementSet::open(&target.path).map_err(runtime_error)?;
219        initialize_continuum_scratch_holding(
220            &mut member,
221            &target.rows.keys().copied().collect::<Vec<_>>(),
222            &target.names[0],
223            scratch,
224            continuum_lock(&mut locks, &target.path)?,
225        )
226        .map_err(runtime_error)?;
227        written.insert(target.path);
228    }
229    if let (Some(root), Some(publication)) = (&root, publication) {
230        initial.verify_write_layout(&written)?;
231        let root_table = MeasurementSet::open(root).map_err(runtime_error)?;
232        publication
233            .publish(root_table.main_table(), continuum_lock(&mut locks, root)?)
234            .map_err(runtime_error)?;
235    }
236    let view = MeasurementSet::open(path).map_err(runtime_error)?;
237    let fit = ContinuumScratchFit::prepare(&view, options).map_err(runtime_error)?;
238    let rows = fit.row_indices().collect::<Vec<_>>();
239    if rows.is_empty() {
240        return Ok(());
241    }
242    let layout = VisibilityMain::open(path)?;
243    written.clear();
244    let mut members = Vec::new();
245    for target in layout.write_targets(&rows, &SCRATCH)? {
246        layout.verify_write_layout(&written)?;
247        continuum_lock(&mut locks, &target.path)?;
248        let member = MeasurementSet::open(&target.path).map_err(runtime_error)?;
249        fit.check_destinations(&member, &target.rows, [&target.names[0], &target.names[1]])
250            .map_err(runtime_error)?;
251        members.push((target, member));
252    }
253    for (target, mut member) in members {
254        layout.verify_write_layout(&written)?;
255        let lock = continuum_lock(&mut locks, &target.path)?;
256        fit.write_holding(
257            &mut member,
258            &target.rows,
259            [&target.names[0], &target.names[1]],
260            lock,
261        )
262        .map_err(runtime_error)?;
263        written.insert(target.path);
264    }
265    Ok(())
266}

268fn continuum_lock<'a>(
269    locks: &'a mut BTreeMap<PathBuf, &mut TableWriteLock>,
270    path: &Path,
271) -> PyResult<&'a mut TableWriteLock> {
272    locks
273        .get_mut(path)
274        .filter(|lock| lock.matches_table_path(path))
275        .map(|lock| &mut **lock)
276        .ok_or_else(|| {
277            runtime_error(format!(
278                "Continuum write lock missing for {}",
279                path.display()
280            ))
281        })
282}

284fn scratch_options(
285    field: Option<&Bound<'_, PyAny>>,
286    fitspw: Option<&Bound<'_, PyAny>>,
287    spw: Option<&Bound<'_, PyAny>>,
288    solint: Option<&Bound<'_, PyAny>>,
289    fitorder: i64,
290    mode: String,
291    old_iterator: bool,
292) -> PyResult<ContinuumScratchOptions> {
293    Ok(ContinuumScratchOptions {
294        field: optional_expression(field)?,
295        fit_spw: optional_expression(fitspw)?.unwrap_or_default(),
296        apply_spw: optional_expression(spw)?.unwrap_or_default(),
297        solution_interval: optional_expression(solint)?.unwrap_or_default(),
298        order: usize::try_from(fitorder)
299            .map_err(|_| runtime_error("fitorder must be nonnegative"))?,
300        mode,
301        old_iterator,
302    })
303}
