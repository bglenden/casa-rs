23/// Convert as `request` says.
24///
25/// `outputvis` never lies inside `vis`. It must not exist, or, with
26/// `overwrite`, must be a MeasurementSet (v2 or v4, not a link) that does
27/// not hold `vis`, on a filesystem that exchanges directories, and an MSv2
28/// no other process writes; it is held so ([`Replacement::hold`]) before
29/// anything is converted. It is replaced only once its replacement is
30/// complete: the conversion is written beside it under a hidden name, the
31/// two are exchanged in one atomic call, and the old store is removed. A
32/// conversion that fails leaves `outputvis` as it was. An MSv2 converted
33/// into an MSv4 is partitioned as `request.partition_scheme` says, its
34/// pointing carried or not, and each dataset chunked by `request.chunks`
35/// or, without one, as XRADIO chunks it ([`xradio_chunks`]). Every loss the
36/// output's format imposes is taken, and each is logged as a warning and
37/// returned.
38///
39/// # Errors
40///
41/// A validation failure (nothing written) for an output not to be written
42/// or replaced, or a source the converter does not support; an execution
43/// failure for anything that failed while converting or replacing, which
44/// leaves no partial output.
45pub fn msconvert(request: &MsconvertRequest) -> Result<MsconvertTaskResult, TaskFailure> {
46    let started = Instant::now();
47    let replacing = guard_output(request)?;
48    let options = ConversionOptions {
49        data_groups: request.data_groups.clone(),
50        // The task takes every loss the destination's format imposes and
51        // reports each; the converter's refusal guards its other callers.
52        accept: LossKind::ALL.to_vec(),
53        batch_bytes: BATCH_BYTES,
54        workers: request.workers,
55        chunks: request.chunks.unwrap_or_default(),
56        codec: request.codec,
57        padding: request.padding,
58        creator: format!("casa-rs msconvert {}", env!("CARGO_PKG_VERSION")),
59        creation_date: utc_now(),
60    };
61    let target = if replacing.is_some() {
62        let target = unused_sibling(&request.outputvis)?;
63        tracing::info!(
64            "msconvert: converting into {}, to take the place of {}",
65            target.display(),
66            request.outputvis.display()
67        );
68        target
69    } else {
70        request.outputvis.clone()
71    };
72    let converted = match request.outformat {
73        StoreFormat::Msv4 => {
74            let fixed;
75            let chunk_shape: &ChunkShape = match request.chunks {
76                Some(chunks) => {
77                    fixed = move |_| chunks;
78                    &fixed
79                }
80                None => &xradio_chunks,
81            };
82            let conversion = Msv4Conversion {
83                partition_scheme: request.partition_scheme,
84                pointing: request.pointing,
85                chunk_shape,
86            };
87            to_msv4_with(&request.vis, &target, &options, &conversion)
88        }
89        StoreFormat::Msv2 => to_msv2(&request.vis, &target, &options),
90    };
91    let report = converted.map_err(failure)?;
92    if let Some(replacement) = replacing {
93        replace(replacement, &target)?;
94    }
95    for loss in &report.losses {
96        tracing::warn!(
97            "msconvert: not carried into {}: {loss}",
98            request.outformat.name()
99        );
100    }
101    Ok(result(request, &report, started))
102}

104/// The MeasurementSet at `outputvis` held for replacement when it exists;
105/// a refusal when `outputvis` may not be written or replaced.
106fn guard_output(request: &MsconvertRequest) -> Result<Option<Replacement>, TaskFailure> {
107    let output = &request.outputvis;
108    let refuse = |reason: String| Err(TaskFailure::validation(reason).at("outputvis"));
109    // Written inside vis, a conversion would change it, and replacing a
110    // part of it would destroy it.
111    if within(output, &request.vis) {
112        return refuse(format!(
113            "outputvis {} is vis or lies inside it",
114            output.display()
115        ));
116    }
117    if std::fs::symlink_metadata(output).is_err() {
118        return Ok(None);
119    }
120    if !request.overwrite {
121        return refuse(format!(
122            "outputvis {} exists; set overwrite=True to replace it",
123            output.display()
124        ));
125    }
126    if within(&request.vis, output) {
127        return refuse(format!(
128            "outputvis {} holds vis; it cannot be replaced by its own conversion",
129            output.display()
130        ));
131    }
132    // A whole MeasurementSet, not a link, on a filesystem that exchanges
133    // directories, and an MSv2 no other process writes, held so until it
134    // is replaced.
135    Replacement::hold(output)
136        .map(Some)
137        .or_else(|error| refuse(format!("outputvis: {error}")))
138}

140/// Whether `inner` is `outer` or lies inside it, by file identity (device
141/// and inode), so that neither a link, a bind mount nor a case-insensitive
142/// file system hides it. A path that does not exist is judged by its
143/// nearest existing ancestor.
144fn within(inner: &Path, outer: &Path) -> bool {
145    let Ok(outer) = std::fs::metadata(outer) else {
146        return false;
147    };
148    let identity = (outer.dev(), outer.ino());
149    let Some(existing) = inner
150        .ancestors()
151        .filter(|ancestor| !ancestor.as_os_str().is_empty())
152        .find_map(|ancestor| std::fs::canonicalize(ancestor).ok())
153    else {
154        return false;
155    };
156    existing.ancestors().any(|ancestor| {
157        std::fs::metadata(ancestor).is_ok_and(|found| (found.dev(), found.ino()) == identity)
158    })
159}

161/// A conversion's error as the task reports it. The converter refuses what
162/// it does not support before it writes; any other error is a failure, and
163/// the converter's staged destination is removed with it.
164fn failure(error: ObservationError) -> TaskFailure {
165    match error {
166        error @ ObservationError::Unsupported { .. } => {
167            TaskFailure::validation(format!("msconvert: {error}"))
168        }
169        error => TaskFailure::execution(format!("msconvert: {error}")),
170    }
171}

173/// A hidden name beside `path` that nothing uses:
174/// `.<name>.msconvert-<process>-<n>`.
175fn unused_sibling(path: &Path) -> Result<PathBuf, TaskFailure> {
176    let name = path
177        .file_name()
178        .map(|name| name.to_string_lossy().into_owned())
179        .ok_or_else(|| {
180            TaskFailure::validation(format!("outputvis {} names no file", path.display()))
181                .at("outputvis")
182        })?;
183    let parent = path.parent().unwrap_or_else(|| Path::new(""));
184    let process = std::process::id();
185    (0..1000)
186        .map(|n| parent.join(format!(".{name}.msconvert-{process}-{n}")))
187        .find(|candidate| std::fs::symlink_metadata(candidate).is_err())
188        .ok_or_else(|| TaskFailure::execution(format!("no unused name beside {}", path.display())))
189}

191/// Exchange the complete store at `converted` for the one `replacement`
192/// holds, in one atomic call, and remove the old store. If the exchange
193/// fails, the output is unchanged and the conversion is kept, complete,
194/// where it is.
195fn replace(replacement: Replacement, converted: &Path) -> Result<(), TaskFailure> {
196    let output = replacement.path().to_path_buf();
197    match replacement.publish(converted) {
198        Ok(None) => Ok(()),
199        Ok(Some(leftover)) => {
200            tracing::warn!(
201                "msconvert: the replaced store, now at {}, could not be removed",
202                leftover.display()
203            );
204            Ok(())
205        }
206        Err(error) => Err(TaskFailure::execution(format!(
207            "replace {}: {error}; it is unchanged, and the conversion is complete at {}",
208            output.display(),
209            converted.display()
210        ))),
211    }
212}

214/// The current time, UTC, ISO 8601 to the second.
215fn utc_now() -> String {
216    let unix_seconds = std::time::SystemTime::now()
217        .duration_since(std::time::UNIX_EPOCH)
218        .map_or(0, |elapsed| elapsed.as_secs());
219    // The Unix epoch is MJD 40587.
220    let time = MvTime::from_mjd_seconds(40_587.0 * 86_400.0 + unix_seconds as f64);
221    format!(
222        "{:04}-{:02}-{:02}T{}+00:00",
223        time.year(),
224        time.month(),
225        time.month_day(),
226        time.format_time(0)
227    )
228}

230fn result(
231    request: &MsconvertRequest,
232    report: &ConversionReport,
233    started: Instant,
234) -> MsconvertTaskResult {
235    MsconvertTaskResult {
236        vis: request.vis.display().to_string(),
237        outputvis: request.outputvis.display().to_string(),
238        informat: request.informat.name().to_string(),
239        outformat: request.outformat.name().to_string(),
240        chunks: match request.outformat {
241            StoreFormat::Msv4 => request.chunks,
242            StoreFormat::Msv2 => None,
243        },
244        losses: report.losses.iter().map(ToString::to_string).collect(),
245        partitions: report.partitions,
246        samples: report.samples,
247        rows: report.rows,
248        elements: report.elements,
249        elapsed_ns: u64::try_from(started.elapsed().as_nanos()).unwrap_or(u64::MAX),
250    }
251}

257    #[test]
258    fn the_creation_date_is_iso_8601_utc() {
259        let now = utc_now();
260        assert_eq!(now.len(), "2026-11-02T12:34:56+00:00".len(), "{now}");
261        assert_eq!(&now[4..5], "-");
262        assert_eq!(&now[10..11], "T");
263        assert!(now.ends_with("+00:00"), "{now}");
264        assert!(now.as_str() >= "2026-01-01", "{now}");
265    }
266
267    #[test]
268    fn a_sibling_name_is_hidden_beside_the_output() {
269        let root = tempfile::tempdir().expect("root");
270        let output = root.path().join("out.ms");
271        let sibling = unused_sibling(&output).expect("a name");
272        assert_eq!(sibling.parent(), Some(root.path()));
273        let name = sibling.file_name().unwrap().to_string_lossy().into_owned();
274        assert!(name.starts_with(".out.ms.msconvert-"), "{name}");
275        std::fs::write(&sibling, b"taken").expect("take it");
276        assert_ne!(unused_sibling(&output).expect("another"), sibling);
277    }
278
279    #[test]
280    fn containment_is_judged_by_file_identity() {
281        let root = tempfile::tempdir().expect("root");
282        let vis = root.path().join("a.ms");
283        std::fs::create_dir_all(vis.join("ANTENNA")).expect("vis");
284        assert!(within(&vis, &vis));
285        assert!(within(&vis.join("ANTENNA"), &vis));
286        // A path not yet made is judged by its nearest existing ancestor.
287        assert!(within(&vis.join("new.ps.zarr"), &vis));
288        assert!(within(&vis.join("x/y/new.ps.zarr"), &vis));
289        assert!(!within(&root.path().join("a.ps.zarr"), &vis));
290        assert!(within(&vis, root.path()));
291        // Through a link, and through `..`.
292        let link = root.path().join("link");
293        std::os::unix::fs::symlink(&vis, &link).expect("link");
294        assert!(within(&link.join("ANTENNA"), &vis));
295        assert!(within(&vis.join("ANTENNA/.."), &vis));
296        assert!(!within(&vis, &root.path().join("missing")));
297    }
