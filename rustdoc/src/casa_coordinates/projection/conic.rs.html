3use std::f64::consts::{FRAC_PI_2, PI};
4
5use crate::direction::wcslib_sin_cos_degrees;
6use crate::{CoordinateError, ProjectionType};
7
8// Calabretta & Greisen (2002), section 5.4. The four conic projections use
9// the same plane geometry and differ in their cone constant and radius law.
10enum Radius {
11    Perspective { scale: f64 },
12    Equidistant,
13    EqualArea { offset: f64 },
14    Orthomorphic { scale: f64 },
15}
16
17pub(super) struct Conic {
18    mean_latitude: f64,
19    cone: f64,
20    fiducial_radius: f64,
21    radius: Radius,
22}
23
24fn bad_projection(message: &str) -> CoordinateError {
25    CoordinateError::ConversionFailed(message.into())
26}
27
28impl Conic {
29    pub(super) fn new(kind: ProjectionType, parameters: &[f64]) -> Result<Self, CoordinateError> {
30        let sigma_degrees = parameters.first().copied().unwrap_or(f64::NAN);
31        let delta_degrees = parameters.get(1).copied().unwrap_or_default();
32        let sigma = sigma_degrees.to_radians();
33        let delta = delta_degrees.to_radians();
34        let (first, second) = (sigma_degrees - delta_degrees, sigma_degrees + delta_degrees);
35        let (sin_sigma, cos_sigma) = wcslib_sin_cos_degrees(sigma_degrees);
36        let (sin_delta, cos_delta) = wcslib_sin_cos_degrees(delta_degrees);
37        let (cone, fiducial_radius, radius) = match kind {
38            ProjectionType::COP => {
39                if cos_delta == 0.0 {
40                    return Err(bad_projection("Invalid COP parallel separation"));
41                }
42                (
43                    sin_sigma,
44                    cos_delta * cos_sigma / sin_sigma,
45                    Radius::Perspective { scale: cos_delta },
46                )
47            }
48            ProjectionType::COD => {
49                let cone = if delta == 0.0 {
50                    sin_sigma
51                } else {
52                    sin_sigma * sin_delta / delta
53                };
54                (cone, cos_delta * cos_sigma / cone, Radius::Equidistant)
55            }
56            ProjectionType::COE => {
57                let (s1, _) = wcslib_sin_cos_degrees(first);
58                let (s2, _) = wcslib_sin_cos_degrees(second);
59                let cone = (s1 + s2) / 2.0;
60                let offset = 1.0 + s1 * s2;
61                (
62                    cone,
63                    (offset - 2.0 * cone * sin_sigma).sqrt() / cone,
64                    Radius::EqualArea { offset },
65                )
66            }
67            ProjectionType::COO => {
68                let tangent = |latitude: f64| {
69                    let (sine, cosine) = wcslib_sin_cos_degrees((90.0 - latitude) / 2.0);
70                    sine / cosine
71                };
72                let (t1, t2) = (tangent(first), tangent(second));
73                let (_, c1) = wcslib_sin_cos_degrees(first);
74                let (_, c2) = wcslib_sin_cos_degrees(second);
75                let cone = if first == second {
76                    sin_sigma
77                } else {
78                    (c2 / c1).ln() / (t2 / t1).ln()
79                };
80                let scale = c1 / (cone * t1.powf(cone));
81                if !scale.is_finite() || scale == 0.0 {
82                    return Err(bad_projection("Invalid COO standard parallels"));
83                }
84                (
85                    cone,
86                    scale * tangent(sigma_degrees).powf(cone),
87                    Radius::Orthomorphic { scale },
88                )
89            }
90            _ => return Err(bad_projection("Unknown conic projection")),
91        };
92        if !cone.is_finite() || cone == 0.0 || !fiducial_radius.is_finite() {
93            return Err(bad_projection("Invalid conic standard parallels"));
94        }
95        Ok(Self {
96            mean_latitude: sigma,
97            cone,
98            fiducial_radius,
99            radius,
100        })
101    }

103    pub(super) fn project(&self, phi: f64, theta: f64) -> Result<(f64, f64), CoordinateError> {
104        let radius = match self.radius {
105            Radius::Perspective { scale } => {
106                if (theta - self.mean_latitude).abs() == FRAC_PI_2 {
107                    return Err(bad_projection("Latitude of divergence for COP"));
108                }
109                if theta.abs() == FRAC_PI_2 && theta.signum() == self.mean_latitude.signum() {
110                    0.0
111                } else {
112                    let value = self.fiducial_radius - scale * (theta - self.mean_latitude).tan();
113                    if value * self.cone < 0.0 || theta.abs() == FRAC_PI_2 {
114                        return Err(bad_projection("Direction is beyond the COP cone"));
115                    }
116                    value
117                }
118            }
119            Radius::Equidistant => self.fiducial_radius + self.mean_latitude - theta,
120            Radius::EqualArea { offset } => {
121                (offset - 2.0 * self.cone * theta.sin()).max(0.0).sqrt() / self.cone
122            }
123            Radius::Orthomorphic { scale } => {
124                if theta == -FRAC_PI_2 {
125                    if self.cone > 0.0 {
126                        return Err(bad_projection("Latitude of divergence for COO"));
127                    }
128                    0.0
129                } else {
130                    scale * ((FRAC_PI_2 - theta) / 2.0).tan().powf(self.cone)
131                }
132            }
133        };
134        if !radius.is_finite() {
135            return Err(bad_projection("Direction is outside the conic projection"));
136        }
137        let (sin_angle, cos_angle) = (self.cone * phi).sin_cos();
138        Ok((
139            radius * sin_angle,
140            self.fiducial_radius - radius * cos_angle,
141        ))
142    }

144    pub(super) fn deproject(&self, x: f64, y: f64) -> Result<(f64, f64), CoordinateError> {
145        let relative_y = self.fiducial_radius - y;
146        let r = x.hypot(relative_y).copysign(self.mean_latitude);
147        let angle = if r == 0.0 {
148            0.0
149        } else {
150            (x / r).atan2(relative_y / r)
151        };
152        let phi = angle / self.cone;
153        let theta = match self.radius {
154            Radius::Perspective { scale } => {
155                self.mean_latitude + ((self.fiducial_radius - r) / scale).atan()
156            }
157            Radius::Equidistant => self.fiducial_radius + self.mean_latitude - r,
158            Radius::EqualArea { offset } => {
159                let sine = (offset - (self.cone * r).powi(2)) / (2.0 * self.cone);
160                if sine.abs() > 1.0 + 1e-12 {
161                    return Err(bad_projection("Pixel is outside the COE projection"));
162                }
163                sine.clamp(-1.0, 1.0).asin()
164            }
165            Radius::Orthomorphic { scale } => {
166                if r == 0.0 && self.cone > 0.0 {
167                    return Err(bad_projection("Pixel is outside the COO projection"));
168                }
169                FRAC_PI_2 - 2.0 * (r / scale).powf(1.0 / self.cone).atan()
170            }
171        };
172        if phi.abs() <= PI + 1e-13 && theta.abs() <= FRAC_PI_2 + 1e-13 {
173            Ok((phi.clamp(-PI, PI), theta.clamp(-FRAC_PI_2, FRAC_PI_2)))
174        } else {
175            Err(bad_projection("Pixel is outside the conic projection"))
176        }
177    }
