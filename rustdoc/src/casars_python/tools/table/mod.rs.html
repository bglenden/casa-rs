2//! CASA's `table` tool (tb) over [`casa_tables::Table`]: every method of CASA
3//! 6.7.6.14's binding, with its signature (ADR-0017, #1025).
4//!
5//! Implemented so far: `open`, `close`, `done`, `ok`, `name`, `iswritable`,
6//! the lock calls, and the read side (`state`, `read`): sizes and names,
7//! column descriptions, cells and columns, keywords and column keywords;
8//! the writes (`write*`), `copy` and `copyrows` (`copy`), and the calls that
9//! report or move a table's state (`housekeeping`).
10//! Values are casatools' (`values`): integers come back as `int64`, reals
11//! as `float64`, complex numbers as `complex128`, a column's rows last.
12//!
13//! The class is `frozen` and keeps the open table behind a `Mutex`, and
14//! every call runs with the GIL released (ADR-0017). What CASA logs
15//! goes to the Python logger `casars.tools.table`, its INFO at INFO, WARN at
16//! WARNING and SEVERE at ERROR.

96/// Convert a CASA tool's record argument with casacore scalar and array types.
97pub(crate) fn casa_record(
98    py: Python<'_>,
99    value: Option<&Bound<'_, PyDict>>,
100) -> PyResult<casa_core::RecordValue> {
101    let Some(value) = value else {
102        return Ok(casa_core::RecordValue::default());
103    };
104    let incoming = component_keyword_incoming(py, Some(value.as_any()))?;
105    match incoming.natural()? {
106        casa_core::Value::Record(record) => Ok(record),
107        _ => unreachable!("a Python dictionary converts to a record"),
108    }
109}

111fn component_keyword_incoming(
112    py: Python<'_>,
113    value: Option<&Bound<'_, PyAny>>,
114) -> PyResult<Incoming> {
115    use pyo3::types::PyTuple;
116    if let Some(record) = value.and_then(|v| v.cast::<PyDict>().ok()) {
117        let mut fields = Vec::with_capacity(record.len());
118        for (name, item) in record.iter() {
119            // CASA drops a None-valued record field; at top level None is
120            // instead the empty variant below.
121            if !item.is_none() {
122                fields.push((
123                    name.extract::<String>()?,
124                    component_keyword_incoming(py, Some(&item))?,
125                ));
126            }
127        }
128        return Ok(Incoming::Record(fields));
129    }
130    let empty = value.is_none_or(|v| {
131        v.is_none()
132            || v.cast::<PyList>().is_ok_and(|list| list.is_empty())
133            || v.cast::<PyTuple>().is_ok_and(|tuple| tuple.is_empty())
134    });
135    if empty {
136        let options = PyDict::new(py);
137        options.set_item("dtype", "bool")?;
138        let array = py
139            .import("numpy")?
140            .call_method("empty", (0,), Some(&options))?;
141        Incoming::from_python(&array)
142    } else {
143        Incoming::from_python(value.expect("nonempty value"))
144    }
145}

174    state: Arc<Mutex<State>>,
175}

177impl PyTableTool {
178    /// Runs `work` on the state with the GIL released, logs what it reports
179    /// and returns its value as Python.
180    fn call(
181        &self,
182        py: Python<'_>,
183        work: impl FnOnce(&mut State, &mut Vec<LogLine>) -> Result<CasaValue, Fail> + Send,
184    ) -> PyResult<Py<PyAny>> {
185        let mut lines = Vec::new();
186        let result = py.detach(|| work(&mut self.guard(), &mut lines));
187        emit(py, LOGGER, &lines)?;
188        result?.into_python(py)
189    }
190
191    /// As [`call`](Self::call), for a call that reads the table: the table
192    /// first takes the read lock its lock options call for and synchronizes
193    /// with what another process has written since (casacore's
194    /// `TableProxy::syncTable`).
195    fn read(
196        &self,
197        py: Python<'_>,
198        work: impl FnOnce(&mut State, &mut Vec<LogLine>) -> Result<CasaValue, Fail> + Send,
199    ) -> PyResult<Py<PyAny>> {
200        self.call(py, move |state, log| {
201            state.synchronize()?;
202            work(state, log)
203        })
204    }
205
206    /// The state lock, taken without the GIL (inside `Python::detach`).
207    fn guard(&self) -> MutexGuard<'_, State> {
208        self.state.lock().unwrap_or_else(PoisonError::into_inner)
209    }
210
211    /// A clone of the state handle so another constructor can detach the GIL.
212    pub(super) fn shared_state(&self) -> Arc<Mutex<State>> {
213        Arc::clone(&self.state)
214    }
215}

217/// A required text argument (a column name); CASA names the argument with
218/// a leading underscore.
219fn text(value: &Bound<'_, PyAny>, argument: &str) -> PyResult<String> {
220    value
221        .extract()
222        .map_err(|_| PyTypeError::new_err(format!("argument _{argument} must be a string")))

280/// A keyword argument: a name or a position; absent and `[]` name no
281/// keyword at all.
282fn keyword_ref(value: Option<&Bound<'_, PyAny>>) -> PyResult<KeywordRef> {
283    let Some(value) = value else {
284        return Ok(KeywordRef::Whole);
285    };
286    if let Ok(name) = value.extract::<String>() {
287        return Ok(KeywordRef::Name(name));
288    }
289    if let Ok(list) = value.cast::<PyList>()
290        && list.is_empty()
291    {
292        return Ok(KeywordRef::Whole);
293    }
294    Ok(KeywordRef::Index(value.extract()?))
295}

297/// A list of integers argument (`blc`, `trc`, `incr`); absent is `[]`, and
298/// `incr` defaults to `[1]`.
299fn integers(value: Option<&Bound<'_, PyAny>>, argument: &str) -> PyResult<Option<Vec<i64>>> {

330/// The fields of a `dict` argument, as keyword values.
331fn record_fields(value: &Bound<'_, PyAny>) -> PyResult<Vec<(String, Incoming)>> {
332    let Ok(record) = value.cast::<PyDict>() else {
333        return Err(PyTypeError::new_err("_value is not a dictionary"));
334    };
335    let mut fields = Vec::with_capacity(record.len());
336    for (key, item) in record.iter() {
337        fields.push((key.extract::<String>()?, Incoming::from_python(&item)?));
338    }
339    Ok(fields)
340}

342/// String column names for `getcoliter` (`str` or sequence of `str`; any other
343/// non-string argument yields `Vec::new()`).
344fn iter_column_names(value: &Bound<'_, PyAny>) -> Vec<String> {
345    if let Ok(name) = value.extract::<String>() {
346        return vec![name];
347    }
348    if let Ok(names) = value.extract::<Vec<String>>() {
349        return names;
350    }
351    if let Ok(list) = value.call_method0("tolist")
352        && let Ok(names) = list.extract::<Vec<String>>()
353    {
354        return names;
355    }
356    Vec::new()
357}

359/// `putinfo`'s record: the `type`, `subType` and `readme` it has.
360fn info_update(value: &Bound<'_, PyAny>) -> PyResult<InfoUpdate> {
361    let Ok(record) = value.cast::<PyDict>() else {
362        return Err(PyTypeError::new_err("_value is not a dictionary"));
363    };
364    let field = |name: &str| -> PyResult<Option<String>> {
365        record
366            .get_item(name)?
367            .map(|item| item.extract())
368            .transpose()
369    };
370    Ok(InfoUpdate {
371        table_type: field("type")?,
372        sub_type: field("subType")?,
373        readme: field("readme")?,
374    })
375}

377/// `open`'s `lockoptions`: a `dict` with `option`, `interval` and `maxwait`.
378fn lock_request(value: Option<&Bound<'_, PyAny>>) -> PyResult<LockRequest> {
379    let Some(value) = value else {
380        return Ok(LockRequest::default());
381    };
382    let Ok(options) = value.cast::<PyDict>() else {
383        return Err(Fail::Type("_lockoptions is not a dictionary".into()).into());
384    };
385    Ok(LockRequest {
386        option: options
387            .get_item("option")?
388            .map(|v| v.extract())
389            .transpose()?,
390        interval: options
391            .get_item("interval")?
392            .map(|v| v.extract())
393            .transpose()?,
394        maxwait: options
395            .get_item("maxwait")?
396            .map(|v| v.extract())
397            .transpose()?,
398    })
399}

411    /// Open a table, closing the one already open.
412    #[pyo3(signature = (tablename = None, lockoptions = None, nomodify = None), text_signature = "($self, tablename='', lockoptions={}, nomodify=True)")]
413    fn open(
414        &self,
415        py: Python<'_>,
416        tablename: Option<&Bound<'_, PyAny>>,
417        lockoptions: Option<&Bound<'_, PyAny>>,
418        nomodify: Option<&Bound<'_, PyAny>>,
419    ) -> PyResult<Py<PyAny>> {
420        let name = tablename.map_or(Ok(String::new()), |name| text(name, "tablename"))?;
421        let request = lock_request(lockoptions)?;
422        let writable = !flag(nomodify, true)?;
423        self.call(py, move |state, _| {
424            state.open(&name, &request, writable)?;
425            Ok(CasaValue::Bool(true))
426        })
427    }

502    /// Take a lock.
503    #[pyo3(signature = (write = None, nattempts = None), text_signature = "($self, write=True, nattempts=0)")]
504    fn lock(
505        &self,
506        py: Python<'_>,
507        write: Option<&Bound<'_, PyAny>>,
508        nattempts: Option<&Bound<'_, PyAny>>,
509    ) -> PyResult<Py<PyAny>> {
510        let write = flag(write, true)?;
511        let attempts =
512            u32::try_from(integer(nattempts, "nattempts", 0)?.max(0)).unwrap_or(u32::MAX);
513        self.call(py, move |state, log| {
514            if state.open.is_none() {
515                state::no_table_specified(log);
516            }
517            Ok(CasaValue::Bool(state.lock(write, attempts)?))
518        })
519    }

525            if state.open.is_none() {
526                state::no_table_specified(log);
527            }

530    }

954    /// Copy the table; the result is a table tool open on the copy.
955    #[pyo3(signature = (newtablename, deep = None, valuecopy = None, dminfo = None, endian = None, memorytable = None, returnobject = None, norows = None), text_signature = "($self, newtablename, deep=False, valuecopy=False, dminfo={}, endian='aipsrc', memorytable=False, returnobject=False, norows=False)")]
956    #[allow(clippy::too_many_arguments)]
957    fn copy(
958        &self,
959        py: Python<'_>,
960        newtablename: &Bound<'_, PyAny>,
961        deep: Option<&Bound<'_, PyAny>>,
962        valuecopy: Option<&Bound<'_, PyAny>>,
963        dminfo: Option<&Bound<'_, PyAny>>,
964        endian: Option<&Bound<'_, PyAny>>,
965        memorytable: Option<&Bound<'_, PyAny>>,
966        returnobject: Option<&Bound<'_, PyAny>>,
967        norows: Option<&Bound<'_, PyAny>>,
968    ) -> PyResult<Py<PyAny>> {
969        let new_name = text(newtablename, "newtablename")?;
970        // `returnobject` is ignored by CASA, which always returns the tool.
971        flag(returnobject, false)?;
972        let request = CopyRequest {
973            new_name,
974            deep: flag(deep, false)?,
975            value_copy: flag(valuecopy, false)?,
976            has_dminfo: match dminfo {
977                Some(dminfo) => !record_fields(dminfo)?.is_empty(),
978                None => false,
979            },
980            endian: endian.map_or(Ok("aipsrc".to_string()), |name| text(name, "endian"))?,
981            memory: flag(memorytable, false)?,
982            no_rows: flag(norows, false)?,
983        };
984        let mut lines = Vec::new();
985        let copied = py.detach(|| self.guard().copy(&request, &mut lines));
986        emit(py, LOGGER, &lines)?;
987        let tool = Self {
988            state: registry::track(State::new(Some(copied?))),
989        };
990        Ok(Py::new(py, tool)?.into_any())
991    }
