17/// An opened observation, whatever stores it.
18///
19/// This is the one interface algorithm code sees: the catalogue to plan
20/// from, [`Self::estimate`] to cost a batch shape before reading anything,
21/// and [`Self::bind_read`] for each traversal it needs. Choosing a backend
22/// and its storage options happens when the observation is opened, outside
23/// the algorithm; afterwards the algorithm handles only the [`Catalogue`],
24/// [`AccessRequest`]s, the bound [`Traversal`] and [`VisibilityBatch`]
25/// views. Dispatch to the store happens once per bind and once per batch,
26/// never per element.
27///
28/// [`VisibilityBatch`]: crate::VisibilityBatch
29pub trait Observation: Sync {
30    /// The observation's catalogue, shared with every batch it delivers.
31    fn catalogue(&self) -> &Arc<Catalogue>;
32
33    /// What `request` would cost, from the catalogue and the store's
34    /// geometry alone: no payload is read. Binding the same request
35    /// returns the same estimate.
36    ///
37    /// # Errors
38    ///
39    /// As [`Self::bind_read`].
40    fn estimate(&self, request: &AccessRequest) -> Result<Estimate, ObservationError>;
41
42    /// Bind a traversal of `request`: one cursor per requested partition,
43    /// each `Send`, so threads can share the observation by reference and
44    /// run partitions in parallel.
45    ///
46    /// # Errors
47    ///
48    /// When the request names something the observation does not have,
49    /// asks for a role the store does not hold, needs a profile the
50    /// store's adapter does not implement, or budgets less than one time.
51    /// Nothing has been read.
52    fn bind_read(&self, request: &AccessRequest) -> Result<Traversal<'_>, ObservationError>;
53
54    /// The sample grid of partition `id` as data group `data_group` places
55    /// it (its fields), for a producer to declare when it writes the
56    /// partition elsewhere: from the store's index, reading no payload.
57    ///
58    /// # Errors
59    ///
60    /// When the observation has no such partition or data group, or the
61    /// partition is not one grid: no scan at each time, or acquisitions of
62    /// more than one integration interval.
63    fn grid(&self, id: PartitionId, data_group: &str)
64    -> Result<TimeBaselineGrid, ObservationError>;
65
66    /// The scientific group of each of partition `id`'s grid times that
67    /// `fields` holds under `grouping`, as data group `data_group` places
68    /// them (its fields), keyed as a traversal keys them (over the grid's
69    /// field-selected times, whatever its time selection: bins count from
70    /// the first such time of their key, [`GroupKey`]); `None` for a time
71    /// `fields` excludes, and `None` overall when
72    /// `grouping` asks for no groups. From the store's index, reading no
73    /// payload. A [`TimeSplit`](crate::TimeSplit) counts the groups its
74    /// cuts divide from these.
75    ///
76    /// # Errors
77    ///
78    /// When the observation has no such partition or data group, or
79    /// `grouping` cannot key its times (by scan, when the store names none;
80    /// by a time bin that is not positive).
81    fn group_keys(
82        &self,
83        id: PartitionId,
84        data_group: &str,
85        grouping: Grouping,
86        fields: &FieldSelection,
87    ) -> Result<Option<Vec<Option<GroupKey>>>, ObservationError>;
88
89    /// Partition `request.partition`'s pointing samples for
90    /// `request.antennas` over `request.times`, reading no visibility
91    /// payload ([`PointingSeries::bracket`] answers queries). A store
92    /// without pointing gives no samples.
93    ///
94    /// An antenna's pointing is a property of the antenna and the time: the
95    /// partition names the time system `request.times` are in and the
96    /// antennas a store may answer for, so two partitions of one time
97    /// system give the same samples for the same antennas and times, and a
98    /// caller may keep one answer for both.
99    ///
100    /// # Errors
101    ///
102    /// [`ObservationError::OverBudget`] for pointing beyond the request's
103    /// budget; [`ObservationError::InvalidRequest`] for an unknown
104    /// partition or antenna, or times that are not a finite span;
105    /// [`ObservationError::Unsupported`] for pointing the
106    /// boundary does not carry (an MSv4 target, samples in several frames,
107    /// an MSv4 frame or polynomial its reader does not take);
108    /// [`ObservationError::InvalidStore`] for malformed pointing, including
109    /// an MSv2 row of more polynomial terms than its reader takes.
110    fn pointing(&self, request: &PointingRequest) -> Result<PointingSeries, ObservationError>;
111
112    /// The most bytes one pointing query ([`Self::pointing`]) reads and
113    /// holds, whatever its partition, antennas and times: a
114    /// [`PointingRequest::budget_bytes`] no query is refused for. Zero for
115    /// a store without pointing.
116    ///
117    /// # Errors
118    ///
119    /// When the store's pointing cannot be read.
120    fn pointing_bytes(&self) -> Result<usize, ObservationError>;
121}

123/// A bound traversal of one partition of an [`Observation`].
124pub type BoundCursor<'o> = Box<dyn VisibilityCursor + Send + 'o>;
125
126/// A bound traversal: one cursor per requested partition, in request order,
127/// and the estimate binding confirmed.
128///
129/// A sequential consumer reads the partitions one after another through
130/// [`Self::next`] or [`Self::next_owned`]; a parallel consumer takes the
131/// cursors and gives each a thread.
132pub struct Traversal<'o> {
133    /// The cursors, one per partition.
134    pub cursors: Vec<BoundCursor<'o>>,
135    /// What the traversal will cost.
136    pub estimate: Estimate,
137    current: usize,
138}
139
140/// Whose turn it is in a traversal.
141enum Turn {
142    /// Cursor `index` has a batch to fill.
143    Batch(usize),
144    /// A group's input is complete.
145    GroupEnd(GroupKey),
146    /// Every cursor is exhausted.
147    End,
148}
149
150impl<'o> Traversal<'o> {
151    /// A traversal of `cursors` with `estimate`.
152    #[must_use]
153    pub fn new(cursors: Vec<BoundCursor<'o>>, estimate: Estimate) -> Self {
154        Self {
155            cursors,
156            estimate,
157            current: 0,
158        }
159    }
160
161    /// The next turn, after moving past exhausted cursors.
162    fn turn(&mut self) -> Turn {
163        while let Some(cursor) = self.cursors.get_mut(self.current) {
164            match cursor.step() {
165                Step::End => self.current += 1,
166                Step::GroupEnd(key) => return Turn::GroupEnd(key),
167                Step::Batch => return Turn::Batch(self.current),
168            }
169        }
170        Turn::End
171    }
172
173    /// The next event of the partitions in order, borrowing `buffers`.
174    ///
175    /// # Errors
176    ///
177    /// As [`VisibilityCursor::fill`].
178    pub fn next<'b>(
179        &'b mut self,
180        buffers: &'b mut BatchBuffers,
181    ) -> Result<Option<ReadEvent<'b>>, ObservationError> {
182        match self.turn() {
183            Turn::End => Ok(None),
184            Turn::GroupEnd(key) => Ok(Some(ReadEvent::GroupEnd(key))),
185            Turn::Batch(index) => {
186                self.cursors[index].fill(buffers)?;
187                let buffers: &'b BatchBuffers = buffers;
188                buffers
189                    .batch()
190                    .map(|batch| Some(ReadEvent::Batch(batch)))
191                    .ok_or_else(unfinished)
192            }
193        }
194    }
195
196    /// The next event of the partitions in order, a batch filled into a
197    /// buffer of `pool` ([`VisibilityCursor::next_owned`]).
198    ///
199    /// # Errors
200    ///
201    /// As [`VisibilityCursor::fill`]; the buffer goes back to the pool.
202    pub fn next_owned(&mut self, pool: &BufferPool) -> Result<OwnedEvent, ObservationError> {
203        match self.turn() {
204            Turn::End => Ok(OwnedEvent::End),
205            Turn::GroupEnd(key) => Ok(OwnedEvent::GroupEnd(key)),
206            Turn::Batch(index) => {
207                let cursor = &mut self.cursors[index];
208                pool.fill(|buffers| cursor.fill(buffers))
209                    .map(OwnedEvent::Batch)
210            }
211        }
212    }
213
214    /// What every cursor has read so far, summed.
215    #[must_use]
216    pub fn counters(&self) -> IoCounters {
217        let mut total = IoCounters::default();
218        for cursor in &self.cursors {
219            total.add(cursor.counters());
220        }
221        total
222    }
223}

225/// What a cursor will do next.
226#[derive(Clone, Debug, PartialEq, Eq)]
227pub enum Step {
228    /// A batch: [`VisibilityCursor::fill`] it.
229    Batch,
230    /// No further samples of this group will arrive. Announced once; the
231    /// cursor has moved past it.
232    GroupEnd(GroupKey),
233    /// The traversal is over.
234    End,
235}
236
237/// What [`VisibilityCursor::next_owned`] delivers.
238#[derive(Debug)]
239#[expect(
240    clippy::large_enum_variant,
241    reason = "a filled batch is moved once per batch; boxing it would allocate per batch"
242)]
243pub enum OwnedEvent {
244    /// The next batch, in a buffer of the pool.
245    Batch(FilledBatch),
246    /// No further samples of this group will arrive.
247    GroupEnd(GroupKey),
248    /// The traversal is over.
249    End,
250}

252/// A bound traversal of one partition of an [`AccessRequest`].
253///
254/// An adapter implements [`Self::step`] and [`Self::fill`]; consumers call
255/// [`Self::next`], which lends the filled buffers as a [`VisibilityBatch`],
256/// or [`Self::next_owned`], which fills a pooled buffer and hands it over
257/// as a [`FilledBatch`].
258///
259/// [`VisibilityBatch`]: crate::VisibilityBatch
260pub trait VisibilityCursor {
261    /// The observation's catalogue: what the batches' views refer to.
262    fn catalogue(&self) -> &Arc<Catalogue>;
263
264    /// What this partition's traversal costs.
265    fn estimate(&self) -> &PartitionEstimate;
266
267    /// What comes next. A group end is consumed by this call; a batch is
268    /// not, so asking twice before filling is harmless.
269    fn step(&mut self) -> Step;
270
271    /// Fill `buffers` with the next batch, through [`BatchBuffers::begin`]
272    /// and [`crate::BatchFill::finish`]. Only valid after [`Self::step`]
273    /// answered [`Step::Batch`].
274    ///
275    /// # Errors
276    ///
277    /// When the store cannot be read or holds data the bound profile does
278    /// not admit.
279    fn fill(&mut self, buffers: &mut BatchBuffers) -> Result<(), ObservationError>;
280
281    /// What the traversal has read so far.
282    fn counters(&self) -> IoCounters;
283
284    /// The next event, borrowing `buffers`: they cannot be refilled or
285    /// mutated until the event is dropped.
286    ///
287    /// # Errors
288    ///
289    /// As [`Self::fill`].
290    fn next<'b>(
291        &'b mut self,
292        buffers: &'b mut BatchBuffers,
293    ) -> Result<Option<ReadEvent<'b>>, ObservationError> {
294        match self.step() {
295            Step::End => Ok(None),
296            Step::GroupEnd(key) => Ok(Some(ReadEvent::GroupEnd(key))),
297            Step::Batch => {
298                self.fill(buffers)?;
299                let buffers: &'b BatchBuffers = buffers;
300                buffers
301                    .batch()
302                    .map(|batch| Some(ReadEvent::Batch(batch)))
303                    .ok_or_else(unfinished)
304            }
305        }
306    }
307
308    /// The next event, a batch filled into a buffer of `pool`. A buffer is
309    /// taken only for a batch, and the call waits while the pool has none
310    /// free: the pool's bound is the traversal's.
311    ///
312    /// # Errors
313    ///
314    /// As [`Self::fill`]; the buffer goes back to the pool.
315    fn next_owned(&mut self, pool: &BufferPool) -> Result<OwnedEvent, ObservationError> {
316        match self.step() {
317            Step::End => Ok(OwnedEvent::End),
318            Step::GroupEnd(key) => Ok(OwnedEvent::GroupEnd(key)),
319            Step::Batch => pool
320                .fill(|buffers| self.fill(buffers))
321                .map(OwnedEvent::Batch),
322        }
323    }
324}
325
326fn unfinished() -> ObservationError {
327    ObservationError::InvalidBatch {
328        reason: "the cursor reported a batch it did not finish".to_string(),
329    }
330}

332impl<C: VisibilityCursor + ?Sized> VisibilityCursor for Box<C> {
333    fn catalogue(&self) -> &Arc<Catalogue> {
334        (**self).catalogue()
335    }
336
337    fn estimate(&self) -> &PartitionEstimate {
338        (**self).estimate()
339    }
340
341    fn step(&mut self) -> Step {
342        (**self).step()
343    }
344
345    fn fill(&mut self, buffers: &mut BatchBuffers) -> Result<(), ObservationError> {
346        (**self).fill(buffers)
347    }
348
349    fn counters(&self) -> IoCounters {
350        (**self).counters()
351    }
