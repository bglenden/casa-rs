35pub(super) fn run<T>(
36    image: &PagedImage<T>,
37    request: &ImpvRequest,
38) -> Result<PvImageSummary, ImageError>
39where
40    T: ImagePixel + Into<f64> + Copy,
41{
42    let destination = std::path::absolute(&request.outfile).map_err(output_io)?;
43    if request.outfile.as_os_str().is_empty() {
44        return Err(ImageError::CasaSelection(
45            "outfile must be specified".into(),
46        ));
47    }
48    check_destination(&destination, request)?;
49    let plane = direction_plane_axes(image)?;
50    let direction = direction(image.coordinates())?;
51    let width = position::width(&request.width, direction)?;
52    let endpoints = position::endpoints(request, direction)?;
53    let strip = geometry::Strip::new(direction, endpoints, plane.map(|i| image.shape()[i]), width)?;
54    let selection = selection(image, request, plane)?;
55    let mut mask =
56        combined_selection_mask(image, &selection, request.mask.as_deref(), request.stretch)?;
57    selection.apply_selection_masks(&mut mask);
58    let input = image.get_slice(&selection.start, &selection.shape)?;
59    let (data, valid) = sample(&input, mask.as_ref(), plane, &strip)?;
60    if !valid.iter().any(|&bit| bit) {
61        return Err(ImageError::CasaSelection(
62            "All selected pixels are masked".into(),
63        ));
64    }
65    let (coordinates, record, secondary) =
66        output_coordinates(image, &selection, plane, &strip, &request.unit)?;
67    let mut info = image.image_info()?;
68    info.beam_set = subset_beam_set(
69        &info.beam_set,
70        image.coordinates(),
71        image.ndim(),
72        &selection.start,
73        &selection.shape,
74    )?;
75    let mut misc = image.misc_info();
76    misc.upsert("secondary_coordinates", Value::Record(secondary));
77    let stage = tempfile::tempdir_in(destination.parent().ok_or(ImageError::NotPersistent)?)
78        .map_err(output_io)?;
79    let product = stage.path().join("image");
80    let shape = data.shape().to_vec();
81    let mut output = PagedImage::<f32>::create(shape.clone(), coordinates, &product)?;
82    output.set_coords_record_keyword(record);
83    output.set_units(image.units())?;
84    output.set_image_info(&info)?;
85    output.set_misc_info(misc)?;
86    output.put_slice(&data, &vec![0; shape.len()])?;
87    if valid.iter().any(|bit| !bit) {
88        output.put_mask("mask0", &valid)?;
89        output.set_default_mask("mask0")?;
90    }
91    output.append_history_entries(&crate::image::image_history_rows(&request.imagename)?)?;
92    output.append_history_at("::impv", &[format!(
93        "impv(imagename={:?}, outfile={:?}, mode={:?}, start={:?}, end={:?}, width={width}, unit={:?})",
94        request.imagename.display().to_string(), request.outfile.display().to_string(), request.mode,
95        request.start, request.end, request.unit,
96    )])?;
97    output.save()?;
98    drop(output);
99    match rename_without_replacing(&product, &destination) {
100        Err(error) if error.kind() == io::ErrorKind::AlreadyExists && request.overwrite => {
101            check_destination(&destination, request)?;
102            exchange_directories(&product, &destination).map_err(output_io)?;
103        }
104        result => result.map_err(output_io)?,
105    }
106    Ok(PvImageSummary {
107        imagename: request.imagename.display().to_string(),
108        outfile: request.outfile.display().to_string(),
109        shape,
110        path_pixels: strip.shape[0],
111        width,
112        units: image.units().to_owned(),
113    })
114}

116fn selection<T: ImagePixel>(
117    image: &PagedImage<T>,
118    request: &ImpvRequest,
119    plane: [usize; 2],
120) -> Result<SubimageSelection, ImageError> {
121    let record = match request.region.as_ref() {
122        Some(ParameterValue::Table(fields)) if !fields.is_empty() => {
123            let Value::Record(record) =
124                parameter_value_to_casacore(request.region.as_ref().expect("region is set"))
125                    .map_err(|e| ImageError::CasaSelection(e.into()))?
126            else {
127                unreachable!("a parameter table converts to a record")
128            };
129            Some(record)
130        }
131        Some(ParameterValue::String(name))
132            if !name.contains(['[', ':', '\n']) && !Path::new(name).exists() =>
133        {
134            image.get_region_record(name.trim()).ok()
135        }
136        _ => None,
137    };
138    let region = if let Some(record) = &record {
139        Some(SubimageRegion::Record(record))
140    } else {
141        match request.region.as_ref() {
142            Some(ParameterValue::String(text)) if !text.trim().is_empty() => {
143                Some(SubimageRegion::Text(text))
144            }
145            _ => None,
146        }
147    };
148    let chans = request
149        .chans
150        .as_deref()
151        .map(str::trim)
152        .filter(|s| !s.eq_ignore_ascii_case("all") && !s.eq_ignore_ascii_case("none"));
153    let stokes = request
154        .stokes
155        .as_deref()
156        .map(str::trim)
157        .map(|s| s.trim_start_matches('[').trim_end_matches(']').trim())
158        .filter(|s| !s.eq_ignore_ascii_case("all") && !s.eq_ignore_ascii_case("none"))
159        .map(|s| s.replace(';', ","));
160    let resolved = SubimageSelection::resolve(image, None, region, chans, stokes.as_deref())?;
161    if resolved.shape[plane[0]] != image.shape()[plane[0]]
162        || resolved.shape[plane[1]] != image.shape()[plane[1]]
163    {
164        return Err(ImageError::CasaSelection(
165            "You are not permitted to make a region selection in the direction coordinate".into(),
166        ));
167    }
168    if [
169        (&resolved.channel_indices, resolved.channel_axis),
170        (&resolved.stokes_indices, resolved.stokes_axis),
171    ]
172    .iter()
173    .any(|(indices, axis)| {
174        indices
175            .as_ref()
176            .zip(*axis)
177            .is_some_and(|(indices, axis)| indices.len() != resolved.shape[axis])
178    }) {
179        return Err(ImageError::CasaSelection(
180            "Only a single n-dimensional rectangular region is supported".into(),
181        ));
182    }
183    Ok(resolved)
184}

186fn sample<T>(
187    input: &ArrayD<T>,
188    mask: Option<&ArrayD<bool>>,
189    plane: [usize; 2],
190    strip: &geometry::Strip,
191) -> Result<(ArrayD<f32>, ArrayD<bool>), ImageError>
192where
193    T: Into<f64> + Copy,
194{
195    let [x, y] = plane;
196    let plane_shape = [input.shape()[x], input.shape()[y]];
197    let input_axes: Vec<_> = (0..input.ndim())
198        .filter(|axis| !plane.contains(axis))
199        .collect();
200    let plane_keys: Vec<_> = input_axes.iter().map(|&axis| input.shape()[axis]).collect();
201    let offsets = (0..strip.shape[0])
202        .map(|i| {
203            (0..strip.shape[1])
204                .map(|j| strip.input_pixel(i, j))
205                .collect::<Result<Vec<_>, _>>()
206        })
207        .collect::<Result<Vec<_>, _>>()?;
208    let mut out_shape = input.shape().to_vec();
209    out_shape[x] = strip.shape[0];
210    out_shape.remove(y);
211    let mut result = ArrayD::zeros(IxDyn(&out_shape).f());
212    let mut valid = ArrayD::from_elem(IxDyn(&out_shape).f(), false);
213    let mut pixels = Vec::with_capacity(plane_shape[0] * plane_shape[1]);
214    let mut good = Vec::with_capacity(pixels.capacity());
215    let mut index = vec![0; input.ndim()];
216    let method = if strip.rotated() {
217        RegridMethod::Cubic
218    } else {
219        RegridMethod::Nearest
220    };
221    for i in 0..plane_keys.iter().product() {
222        for (axis, pixel) in input_axes.iter().zip(unravel_index(i, &plane_keys)) {
223            index[*axis] = pixel;
224        }
225        pixels.clear();
226        good.clear();
227        for j in 0..plane_shape[1] {
228            index[y] = j;
229            for k in 0..plane_shape[0] {
230                index[x] = k;
231                pixels.push(input[IxDyn(&index)].into());
232                if let Some(mask) = mask {
233                    good.push(mask[IxDyn(&index)]);
234                }
235            }
236        }
237        let plane = Plane {
238            data: &pixels,
239            mask: mask.map(|_| good.as_slice()),
240            shape: plane_shape,
241        };
242        let mut output_index = index.clone();
243        output_index.remove(y);
244        for (i, section) in offsets.iter().enumerate() {
245            let (sum, n) = section
246                .iter()
247                .filter_map(|&at| plane.sample(method, at))
248                .fold((0.0, 0), |(sum, n), value| {
249                    (sum + f64::from(value as f32), n + 1)
250                });
251            if n > 0 {
252                output_index[x] = i;
253                result[IxDyn(&output_index)] = (sum / f64::from(n)) as f32;
254                valid[IxDyn(&output_index)] = true;
255            }
256        }
257    }
258    Ok((result, valid))
259}

261fn output_coordinates<T: ImagePixel>(
262    image: &PagedImage<T>,
263    selection: &SubimageSelection,
264    plane: [usize; 2],
265    strip: &geometry::Strip,
266    unit: &str,
267) -> Result<(CoordinateSystem, RecordValue, RecordValue), ImageError> {
268    let unit = Unit::new(unit)
269        .map_err(|_| ImageError::CasaSelection("PV offset unit must be angular".into()))?;
270    let collapsed = strip.collapsed_direction()?;
271    let increment = Quantity::new(strip.offset_increment(&collapsed)?, "rad")
272        .and_then(|q| q.get_value_in(&unit))
273        .map_err(|e| ImageError::CasaSelection(e.to_string()))?;
274    let mut selected_stokes = selection.selected_stokes.clone();
275    if selected_stokes.is_none()
276        && let Some(axis) = selection.stokes_axis
277        && selection.shape[axis] != image.shape()[axis]
278    {
279        let stokes = stokes_for_axis(image.coordinates(), axis)?;
280        let start = selection.start[axis];
281        selected_stokes = Some(stokes[start..start + selection.shape[axis]].to_vec());
282    }
283    let selected_stokes = selected_stokes.as_deref();
284    let mut record = match legacy_subimage_coordinates_with_stokes(
285        image,
286        &selection.start,
287        &selection.shape,
288        selected_stokes,
289    ) {
290        Some(record) => record,
291        None => subimage_coordinates_with_stokes(
292            image.coordinates(),
293            &selection.start,
294            &selection.shape,
295            selected_stokes,
296        )?
297        .to_record(),
298    };
299    let direction = CoordinateSystem::record_coordinates(&record)?
300        .into_iter()
301        .find(|entry| entry.field.starts_with("direction"))
302        .ok_or_else(|| {
303            ImageError::InvalidMetadata(
304                "PV direction coordinate missing from the image record".into(),
305            )
306        })?;
307    record.upsert(&direction.field, coordinate_record(collapsed, "direction0"));
308    let secondary = record.clone();
309    let suffix = direction.field.trim_start_matches("direction");
310    let linear = LinearCoordinate::new(2, vec!["Offset".into(); 2], vec![unit.name().into(); 2])
311        .with_increment(vec![increment; 2])
312        .with_reference_pixel(vec![((strip.shape[0] - 1) / 2) as f64; 2]);
313    record.remove(&direction.field);
314    record.upsert(
315        format!("linear{suffix}"),
316        coordinate_record(linear, "linear0"),
317    );
318    let kept: Vec<_> = (0..image.ndim()).filter(|&axis| axis != plane[1]).collect();
319    let mut record = drop_axes_from_legacy_record(&record, image.ndim(), &kept)
320        .ok_or_else(|| ImageError::InvalidMetadata("cannot remove the collapsed PV axis".into()))?;
321    let offset = kept
322        .iter()
323        .position(|&axis| axis == plane[0])
324        .expect("offset axis kept") as i32;
325    // casacore sets the removed linear axis to its own second axis index.
326    for kind in ["pixelmap", "worldmap"] {
327        record.upsert(
328            format!("{kind}{suffix}"),
329            Value::Array(ArrayValue::from_i32_vec(vec![offset, -2])),
330        );
331    }
332    Ok((CoordinateSystem::from_record(&record)?, record, secondary))
333}

335fn coordinate_record(coordinate: impl Into<CoordinateModel>, name: &str) -> Value {
336    let mut system = CoordinateSystem::new();
337    system.add_coordinate(coordinate);
338    system
339        .to_record()
340        .get(name)
341        .cloned()
342        .expect("first coordinate was stored")
343}

345fn direction(coordinates: &CoordinateSystem) -> Result<&DirectionCoordinate, ImageError> {
346    if coordinates
347        .find_coordinate(CoordinateType::Spectral)
348        .is_none()
349    {
350        return Err(ImageError::CasaSelection(
351            "impv requires a spectral axis".into(),
352        ));
353    }
354    let Some(CoordinateModel::Direction(direction)) = coordinates
355        .find_coordinate(CoordinateType::Direction)
356        .map(|i| coordinates.coordinate(i))
357        .filter(|coordinate| coordinate.n_pixel_axes() == 2)
358    else {
359        return Err(ImageError::CasaSelection(
360            "impv requires both direction axes".into(),
361        ));
362    };
363    let increment = direction.increment();
364    let (x, y) = (increment[0].abs(), increment[1].abs());
365    if (x - y).abs() > 2e-10 * x.max(y) {
366        return Err(ImageError::CasaSelection(
367            "impv requires square direction pixels".into(),
368        ));
369    }
370    Ok(direction)
371}

373fn check_destination(destination: &Path, request: &ImpvRequest) -> Result<(), ImageError> {
374    ensure_outfile_does_not_alias_input(destination, &request.imagename)?;
375    if request
376        .imagename
377        .canonicalize()
378        .ok()
379        .zip(
380            destination
381                .parent()
382                .and_then(|parent| parent.canonicalize().ok()),
383        )
384        .is_some_and(|(source, parent)| parent.starts_with(source))
385    {
386        return Err(ImageError::CasaSelection(
387            "outfile must be outside the input image".into(),
388        ));
389    }
390    match destination.symlink_metadata() {
391        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
392        Err(error) => Err(output_io(error)),
393        Ok(_) if !request.overwrite => Err(ImageError::CasaSelection(format!(
394            "PV output already exists: {}",
395            destination.display()
396        ))),
397        Ok(metadata)
398            if metadata.is_dir()
399                && Table::open_metadata_only(TableOptions::new(destination))
400                    .is_ok_and(|table| table.info().table_type == "Image") =>
401        {
402            Ok(())
403        }
404        Ok(_) => Err(ImageError::CasaSelection(format!(
405            "{} is not a CASA image directory",
406            destination.display()
407        ))),
408    }
409}
