2//! `casars.simutil.simutil`: the part of CASA's `simutil` helper class
3//! (`casatasks.private.simutil`) a `sm` script uses, over
4//! [`casa_simulation::readantenna`].
5//!
6//! `simutil` is a Python class of CASA's tasks, not a CASA tool, so it is
7//! not under `casars.tools`; `casars.simutil` is its public import (CASA's
8//! is `casatasks.private.simutil`). ADR-0018 puts its computation in Rust,
9//! so the class is declared here like the tools, with CASA's constructor
10//! signature.
11//!
12//! Only `readantenna` is declared; the rest of `simutil` (pointings,
13//! sensitivities, sky-model handling) comes with `simobserve` in simulation
14//! Phase 5. The constructor takes CASA's parameters and keeps none of them,
15//! since `readantenna` uses none.
16//!
17//! Not being a CASA tool, the class is not in the object-surface bundle
18//! (whose CASA tools are [`crate::tools::TOOLS`]); its Python tests compare
19//! its signatures with CASA's installed class
20//! (`python/tests/data/casa_simutil_signatures.json`).
21//!
22//! `readantenna(antab)` returns CASA's tuple `(x, y, z, d, padnames,
23//! antnames, telescopename, posobs)`: ITRF x, y and z as lists of `float`,
24//! the dish diameters as a NumPy array, the pad and antenna names as lists
25//! of `str`, the telescope name and the observatory's WGS84 position
26//! record. CASA's errors (`Exception` with CASA's message) are
27//! `RuntimeError`s here, and a file that cannot be opened is an `OSError`.
28//! [`casa_simulation::readantenna`] records where its reading differs from
29//! CASA's; also, CASA's LOC files give the positions as NumPy `float64`
30//! scalars, which here are `float`s.
31

43/// The Python logger `simutil`'s messages go to.
44const LOGGER: &str = "casars.simutil";

46/// CASA's `simutil` helper class.
47#[pyclass(name = "simutil", module = "casars.simutil", frozen)]
48pub(crate) struct PySimutil;
49
50/// A configuration-file error as the Python exception CASA's raises.
51fn raised(error: &ReadAntennaError) -> PyErr {
52    match error {
53        ReadAntennaError::Io { source, .. } if source.kind() == std::io::ErrorKind::NotFound => {
54            PyFileNotFoundError::new_err(error.to_string())
55        }
56        ReadAntennaError::Io { .. } => PyOSError::new_err(error.to_string()),
57        _ => PyRuntimeError::new_err(error.to_string()),
58    }
59}

61#[pymethods]
62impl PySimutil {
63    #[new]
64    #[allow(unused_variables)]
65    #[pyo3(
66        signature = (
67            direction = Variant::Text(String::new()),
68            centerfreq = Variant::Empty,
69            bandwidth = Variant::Empty,
70            totaltime = Variant::Empty,
71            verbose = false
72        ),
73        text_signature = "(direction='', centerfreq={'unit': 'GHz', 'value': 245.0}, bandwidth={'unit': 'GHz', 'value': 1.0}, totaltime={'unit': 'h', 'value': 0.0}, verbose=False)"
74    )]
75    fn new(
76        direction: Variant,
77        centerfreq: Variant,
78        bandwidth: Variant,
79        totaltime: Variant,
80        verbose: bool,
81    ) -> Self {
82        Self
83    }
84
85    /// Read an antenna configuration file: `(x, y, z, d, padnames,
86    /// antnames, telescopename, posobs)`.
87    #[pyo3(signature = (antab = None), text_signature = "($self, antab=None)")]
88    fn readantenna(&self, py: Python<'_>, antab: Option<PathBuf>) -> PyResult<Py<PyAny>> {
89        let Some(path) = antab else {
90            return Err(PyTypeError::new_err(
91                "readantenna needs the antenna configuration file (antab)",
92            ));
93        };
94        emit(
95            py,
96            LOGGER,
97            &[LogLine::info(format!(
98                "Reading antenna positions from '{}'",
99                path.display()
100            ))],
101        )?;
102        // The file is read and CASA's tuple built without the GIL; only the
103        // Python objects are made with it.
104        let value = py.detach(move || -> PyResult<CasaValue> {
105            let measures = casa_ms::open_measures_runtime()
106                .map_err(|error| PyRuntimeError::new_err(format!("measures data: {error}")))?;
107            let config = read_antenna_config(&path, &*measures).map_err(|error| raised(&error))?;
108            Ok(readantenna_value(config))
109        })?;
110        value.into_python(py)
111    }
112}

114/// CASA's `readantenna` tuple, made from the configuration's own vectors.
115fn readantenna_value(config: AntennaConfigFile) -> CasaValue {
116    let floats =
117        |values: Vec<f64>| CasaValue::List(values.into_iter().map(CasaValue::Float).collect());
118    let texts =
119        |values: Vec<String>| CasaValue::List(values.into_iter().map(CasaValue::Text).collect());
120    let observatory = Measure::from_casa(&CasaMeasure::Position(config.observatory)).to_value();
121    CasaValue::Tuple(vec![
122        floats(config.x),
123        floats(config.y),
124        floats(config.z),
125        CasaValue::floats(config.dish_diameter),
126        texts(config.pad_names),
127        texts(config.antenna_names),
128        CasaValue::Text(config.telescope),
129        observatory,
130    ])
131}
