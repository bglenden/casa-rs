12use super::{PrepareError, Surveyed, domains::PreparedImageDomain, native_aw};

15pub(super) fn plan(
16    request: &ImagingRequest,
17    surveyed: &Surveyed<'_>,
18    domain: &PreparedImageDomain,
19    context: &RunContext,
20) -> Result<Option<Awp2Deployment>, PrepareError> {
21    let Gridder::Awp2 { evla_surface, .. } = &request.gridder else {
22        return Ok(None);
23    };
24    let [window] = surveyed.survey.spectral_windows.as_slice() else {
25        return Err(PrepareError::Awp2Selection);
26    };
27    let field = surveyed.survey.first_visibility_field_id;
28    if surveyed.stores.len() != 1
29        || surveyed.survey.fields.len() != 1
30        || surveyed.msv2.is_none()
31        || surveyed.engine.field_moves(field)
32        || window.frequency_reference != FrequencyRef::TOPO
33        || domain.telescope.trim() != "VLA"
34    {
35        return Err(PrepareError::Awp2Selection);
36    }
37    let coordinate_index = domain
38        .coordinates
39        .find_coordinate(CoordinateType::Direction)
40        .expect("an imaging domain has direction axes");
41    let CoordinateModel::Direction(direction) = domain.coordinates.coordinate(coordinate_index)
42    else {
43        unreachable!("direction coordinate type");
44    };
45    // AWPLPG evaluates the spectral reference conversion at the image's
46    // epoch and direction at pixel (0, 0).
47    let [longitude, latitude] = direction
48        .pixel_to_world([0.0; 2])
49        .map_err(|_| PrepareError::Awp2Selection)?;
50    let direction = MDirection::from_angles(longitude, latitude, direction.direction_ref());
51    let frame = surveyed
52        .engine
53        .spectral_frame_observatory_direction(surveyed.survey.image_time_mjd_seconds, direction)?;
54    let spectral = &surveyed.spectral.image;
55    let mut converter =
56        MFrequencyConverter::new(spectral.frequency_reference, window.frequency_reference);
57    let [lo, hi] =
58        [-0.5, 0.5].map(|pixel| spectral.reference_frequency_hz + pixel * spectral.increment_hz);
59    let band = [
60        converter.convert_hz(lo, &frame)?.hz(),
61        converter.convert_hz(hi, &frame)?.hz(),
62    ];
63    let frequencies_hz =
64        beam_frequencies(&window.frequencies_hz, window.channel_widths_hz[0], band)?;
65
66    let path = evla_surface
67        .as_deref()
68        .ok_or(PrepareError::Awp2SurfacePath)?;
69    // The bounded parser holds raw text, radial samples and their shared
70    // allocation during conversion. Retain only the physical data afterward.
71    let mut charge = admit(
72        &context.host,
73        &context.policy,
74        &Demand {
75            phase: "AWP2 dish surface",
76            memory: 16 << 20,
77        },
78    )?;
79    let surface = native_aw::read_dish_surface(path)?;
80    charge.retain(
81        (std::mem::size_of_val(surface.samples())
82            + std::mem::size_of_val(frequencies_hz.as_ref())
83            + 2 * size_of::<usize>()) as u64,
84    );
85    Ok(Some(Awp2Deployment {
86        frequencies_hz,
87        surface,
88        _charge: charge,
89    }))
90}

92/// SimplePBConvFunc::findUsefulChannels works in the native SPW frame,
93/// sampling the whole SPW first and then filtering by the image band.
94fn beam_frequencies(
95    window_hz: &[f64],
96    first_width_hz: f64,
97    [first_edge, last_edge]: [f64; 2],
98) -> Result<Box<[f64]>, PrepareError> {
99    let [low, high] = window_hz
100        .iter()
101        .fold([f64::INFINITY, f64::NEG_INFINITY], |[low, high], &hz| {
102            [low.min(hz), high.max(hz)]
103        });
104    if !(0.9e9..=2e9 + 1e-6).contains(&low) || !(0.9e9..=2e9 + 1e-6).contains(&high) {
105        return Err(PrepareError::Awp2Selection);
106    }
107    let tolerance = (high / 100.0).max(first_width_hz.abs());
108    let count = (((high - low).round() / tolerance) as usize).max(1);
109    let mut frequencies: Vec<_> = (0..count)
110        .map(|k| low + k as f64 * tolerance)
111        .filter(|hz| (first_edge.min(last_edge)..=first_edge.max(last_edge)).contains(hz))
112        .collect();
113    if frequencies.is_empty() {
114        frequencies.push((first_edge + last_edge) / 2.0);
115    }
116    Ok(frequencies.into_boxed_slice())
117}

119#[cfg(test)]
120mod tests {
121    use super::*;
122
123    #[test]
124    fn awp2_samples_the_whole_spw_before_cutting_to_the_image_band() {
125        let first = 999_999_999.999_999_9;
126        let channels = [first, 1.5e9, 2e9];
127        // CASA's captured inputs omit the high endpoint even though the
128        // wide MFS image reaches it.
129        let frequencies = beam_frequencies(&channels, 5e8, [0.75e9, 2.25e9]).unwrap();
130        assert_eq!(&*frequencies, &[first, 1.5e9]);
131        // Inclusive image edges, regardless of axis order.
132        let frequencies = beam_frequencies(&channels, 5e8, [1.5e9, 1.25e9]).unwrap();
133        assert_eq!(&*frequencies, &[1.5e9]);
134        let frequencies = beam_frequencies(&channels, 5e8, [1.5e9 + 1.0, 1.7e9]).unwrap();
135        assert_eq!(&*frequencies, &[1.6e9 + 0.5]);
136    }
