41impl Grid {
42    pub(super) fn new(
43        input: &MeasurementSet,
44        selection: &BTreeMap<i32, TransformChannelSelection>,
45        grid_selection: &BTreeMap<i32, TransformChannelSelection>,
46        bin_width: usize,
47        grid_budget: usize,
48        legacy_cvel: bool,
49    ) -> Result<Option<Self>, MsTransformError> {
50        if grid_selection.len() < 2 {
51            return Ok(None);
52        }
53        let error = MsTransformError::Spectrum;
54        let spw = input
55            .spectral_window()
56            .map_err(|source| error(source.to_string()))?;
57        let mut axes = Vec::with_capacity(grid_selection.len());
58        let mut first_frame = None;
59        let expected = selection
60            .values()
61            .next()
62            .map_or(0, |spw| spw.channels.len());
63        for selected in selection.values() {
64            let count = selected.channels.len();
65            if !legacy_cvel && count != expected {
66                return Err(error(format!(
67                    "Currently the option 'combinespws' is only supported when the number of channels is the same for all the spw's selected. One of the SPWs selected has {expected} channels, but another selected SPW has {count} channels."
68                )));
69            }
70        }
71        for (&id, selected) in grid_selection {
72            let row = id as usize;
73            let frame = spw
74                .reference_frequency_reference(row)
75                .map_err(|source| error(source.to_string()))?;
76            if frame != *first_frame.get_or_insert(frame) {
77                return Err(error(format!(
78                    "SPW {id} has a non-matching frequency reference frame."
79                )));
80            }
81            let frequencies = spw
82                .chan_freq(row)
83                .map_err(|source| error(source.to_string()))?;
84            let widths = spw
85                .chan_width(row)
86                .map_err(|source| error(source.to_string()))?;
87            let (centres, spans) = selected
88                .channels
89                .iter()
90                .zip(&selected.metadata_ends)
91                .map(|(&first, &last)| {
92                    let invalid = || error(format!("SPW {id} has inconsistent channel metadata."));
93                    let a = frequencies.get(first).ok_or_else(invalid)?;
94                    let b = frequencies.get(last).ok_or_else(invalid)?;
95                    let w1 = widths.get(first).ok_or_else(invalid)?;
96                    let w2 = widths.get(last).ok_or_else(invalid)?;
97                    Ok(((a + b) / 2., (a - b).abs() + (w1.abs() + w2.abs()) / 2.))
98                })
99                .collect::<Result<Vec<_>, _>>()?
100                .into_iter()
101                .unzip::<_, _, Vec<_>, Vec<_>>();
102            axes.push((centres, spans));
103        }
104        let pairs = axes
105            .iter()
106            .map(|(centres, spans)| (centres.as_slice(), spans.as_slice()))
107            .collect::<Vec<_>>();
108        let channels = combine_windows(&pairs, true).map_err(|source| error(source.to_string()))?;
109        let mut bins = Vec::new();
110        let mut centres = Vec::new();
111        let mut widths = Vec::new();
112        for (index, part) in channels.chunks(bin_width).enumerate() {
113            let from = index * bin_width;
114            bins.push((from..from + part.len()).collect());
115            centres.push(part.iter().map(|c| c.frequency).sum::<f64>() / part.len() as f64);
116            widths.push(part.iter().map(|c| c.width).sum::<f64>());
117        }
118        let width = widths[0];
119        let start = centres[0];
120        let span = centres[centres.len() - 1] + widths[widths.len() - 1] / 2. - start + width / 2.;
121        let count = (span / width + 0.01).floor();
122        if !(1. ..=f64::from(i32::MAX)).contains(&count) || width <= 0. {
123            return Err(error("Combined SPW has invalid channel bounds.".into()));
124        }
125        admit_grid(count as usize, grid_budget)?;
126        let frequencies = casa_data_channel_frequencies(
127            &RegridRequest {
128                quantity: RegridQuantity::Frequency,
129                start: Some(start),
130                width: Some(width),
131                nchan: None,
132                rest_frequency_hz: 0.,
133            },
134            &centres,
135            &widths,
136            grid_budget / (std::mem::size_of::<LinearSample>() + 5 * std::mem::size_of::<f64>()),
137        )
138        .map_err(|source| error(source.to_string()))?
139        .centres_hz;
140        let interpolation = linear_samples(&centres, &frequencies);
141        Ok(Some(Self {
142            windows: grid_selection
143                .keys()
144                .enumerate()
145                .map(|(pos, &id)| (id, pos))
146                .collect(),
147            channels,
148            bins,
149            frequencies,
150            width,
151            interpolation,
152            legacy_cvel,
153        }))
154    }

156    pub(super) fn interpolate(&self, value: ArrayValue) -> Result<ArrayValue, TableError> {
157        match value {
158            ArrayValue::Complex32(data) => Ok(ArrayValue::Complex32(
159                apply_linear(array_view(&data)?, &self.interpolation).into_dyn(),
160            )),
161            ArrayValue::Float32(data) => Ok(ArrayValue::Float32(
162                apply_linear(array_view(&data)?, &self.interpolation).into_dyn(),
163            )),
164            ArrayValue::Bool(flags) => {
165                let flags = array_view(&flags)?;
166                let output = Array2::from_shape_fn(
167                    (flags.nrows(), self.interpolation.len()),
168                    |(corr, channel)| {
169                        let sample = self.interpolation[channel];
170                        sample.outside || flags[[corr, sample.left]] || flags[[corr, sample.right]]
171                    },
172                );
173                Ok(ArrayValue::Bool(output.into_dyn()))
174            }
175            _ => Err(TableError::Schema(
176                "Invalid combined spectral cell type.".into(),
177            )),
178        }
179    }
180
181    pub(super) fn interpolate_categories(
182        &self,
183        value: ArrayValue,
184    ) -> Result<ArrayValue, TableError> {
185        let values = category_view(&value)?;
186        let (correlations, _, categories) = values.dim();
187        let output = Array3::from_shape_fn(
188            (correlations, self.interpolation.len(), categories),
189            |(corr, channel, category)| {
190                let sample = self.interpolation[channel];
191                values[[corr, sample.left, category]] || values[[corr, sample.right, category]]
192            },
193        );
194        Ok(ArrayValue::Bool(output.into_dyn()))
195    }
196}

198fn admit_grid(channels: usize, available_bytes: usize) -> Result<(), MsTransformError> {
199    // Include interpolation, frequency centres and the temporary SPW
200    // metadata arrays written for each output channel.
201    let bytes = channels
202        .saturating_mul(std::mem::size_of::<LinearSample>() + 5 * std::mem::size_of::<f64>());
203    if bytes > available_bytes {
204        return Err(MsTransformError::Spectrum(format!(
205            "Combined SPW of {channels} channels needs {bytes} bytes for the grid, \
206             exceeding the {available_bytes} byte grid budget."
207        )));
208    }
209    Ok(())
210}

228#[derive(Clone, Copy)]
229struct LinearSample {
230    left: usize,
231    right: usize,
232    fraction: f64,
233    outside: bool,
234}
235
236fn linear_samples(input: &[f64], output: &[f64]) -> Vec<LinearSample> {
237    let last = input.len() - 1;
238    let identity = input == output;
239    output
240        .iter()
241        .map(|&frequency| {
242            let at = input.partition_point(|value| *value < frequency);
243            if last == 0 {
244                return LinearSample {
245                    left: 0,
246                    right: 0,
247                    fraction: 0.,
248                    outside: true,
249                };
250            }
251            if identity {
252                return LinearSample {
253                    left: at,
254                    right: at,
255                    fraction: 0.,
256                    outside: false,
257                };
258            }
259            let mut right = at.clamp(1, last);
260            let mut left = right - 1;
261            let fraction = (frequency - input[left]) / (input[right] - input[left]);
262            if fraction <= f64::EPSILON {
263                right = left;
264            } else if fraction >= 1.0 - f64::EPSILON {
265                left = right;
266            }
267            LinearSample {
268                left,
269                right,
270                fraction,
271                outside: frequency < input[0] || frequency > input[last],
272            }
273        })
274        .collect()
275}
276
277fn apply_linear<T: Copy + LinearSampleValue>(
278    input: ArrayView2<'_, T>,
279    samples: &[LinearSample],
280) -> Array2<T> {
281    Array2::from_shape_fn((input.nrows(), samples.len()), |(corr, channel)| {
282        let LinearSample {
283            left,
284            right,
285            fraction,
286            ..
287        } = samples[channel];
288        let a = input[[corr, left]];
289        if left == right {
290            a
291        } else {
292            a.interpolate(input[[corr, right]], fraction)
293        }
294    })
295}
296
297trait LinearSampleValue {
298    fn interpolate(self, right: Self, fraction: f64) -> Self;
299}
300
301impl LinearSampleValue for f32 {
302    fn interpolate(self, right: Self, fraction: f64) -> Self {
303        // Preserve a small positive interpolated weight near a zero endpoint.
304        (f64::from(self) + f64::from(right - self) * fraction) as f32
305    }
306}
307
308impl LinearSampleValue for num_complex::Complex32 {
309    fn interpolate(self, right: Self, fraction: f64) -> Self {
310        self + (right - self) * fraction as f32
311    }
312}

332#[derive(Clone, Copy)]
333struct IntegrationTime(f64);
334
335impl PartialEq for IntegrationTime {
336    fn eq(&self, other: &Self) -> bool {
337        self.cmp(other).is_eq()
338    }
339}
340
341impl Eq for IntegrationTime {}
342
343impl PartialOrd for IntegrationTime {
344    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
345        Some(self.cmp(other))
346    }
347}
348
349impl Ord for IntegrationTime {
350    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
351        self.0.total_cmp(&other.0)
352    }

355pub(super) fn group_rows(
356    input: &MeasurementSet,
357    rows: &[usize],
358    descriptions: &[i32],
359    ddid_to_spw: &BTreeMap<i32, i32>,
360    windows: &BTreeMap<i32, usize>,
361    projection: Option<(&[f64], &[TimeGroup])>,
362) -> Result<Vec<Group>, MsTransformError> {
363    let error = |source: TableError| MsTransformError::MutateMeasurementSet {
364        path: input
365            .path()
366            .unwrap_or_else(|| std::path::Path::new("<memory>"))
367            .display()
368            .to_string(),
369        source: Box::new(source),
370    };
371    let table = input.main_table();
372    let read = |column| {
373        table
374            .column_accessor(column)?
375            .scalar_cells_owned_for_rows(rows)
376    };
377    let columns = [
378        "OBSERVATION_ID",
379        "ARRAY_ID",
380        "SCAN_NUMBER",
381        "STATE_ID",
382        "FIELD_ID",
383        "ANTENNA1",
384        "ANTENNA2",
385        "FEED1",
386        "FEED2",
387    ];
388    let mut keys = vec![[0; 9]; rows.len()];
389    for (axis, name) in columns.into_iter().enumerate() {
390        let values = read(name).map_err(error)?;
391        for ((key, value), &row) in keys.iter_mut().zip(values).zip(rows) {
392            key[axis] = scalar_i32(value.as_ref(), name, row)?;
393        }
394    }
395    let times = match projection {
396        Some((times, _)) => times.to_vec(),
397        None => read("TIME")
398            .map_err(error)?
399            .iter()
400            .zip(rows)
401            .map(|(value, &row)| scalar_f64(value.as_ref(), "TIME", row))
402            .collect::<Result<Vec<_>, _>>()?,
403    };
404    let mut grouped = BTreeMap::new();
405    let ddi = input
406        .data_description()
407        .map_err(|source| MsTransformError::Spectrum(source.to_string()))?;
408    let mut pol_for_ddi = BTreeMap::new();
409    for (pos, (&description, time)) in descriptions.iter().zip(times).enumerate() {
410        let pol = match pol_for_ddi.get(&description) {
411            Some(&value) => value,
412            None => {
413                let value = ddi
414                    .polarization_id(description as usize)
415                    .map_err(|source| MsTransformError::Spectrum(source.to_string()))?;
416                pol_for_ddi.insert(description, value);
417                value
418            }
419        };
420        let time = IntegrationTime(time);
421        let [obs, array, scan, state, field, a1, a2, f1, f2] = keys[pos];
422        let group = grouped
423            .entry((
424                [obs, array, scan, state, field],
425                time,
426                [a1, a2, f1, f2, pol],
427            ))
428            .or_insert_with(|| Group {
429                representative: pos,
430                members: Vec::new(),
431            });
432        group
433            .members
434            .push((windows[&ddid_to_spw[&description]], pos));
435    }
436    if let Some((_, time_groups)) = projection {
437        let mut ordered = grouped.into_iter().collect::<Vec<_>>();
438        // CASA emits baseline order within each time-averaging buffer, even
439        // when the effective averaged epochs in that buffer differ.
440        ordered.sort_by_key(|((chunk, time, baseline), group)| {
441            (
442                *chunk,
443                IntegrationTime(time_groups[group.representative].first.time),
444                *baseline,
445                *time,
446            )
447        });
448        return Ok(ordered.into_iter().map(|(_, group)| group).collect());
449    }
450    Ok(grouped.into_values().collect())
451}

453pub(super) fn projected_times(
454    input: &MeasurementSet,
455    groups: &[TimeGroup],
456    channels: &BTreeMap<i32, TransformChannelSelection>,
457    ddid_to_spw: &BTreeMap<i32, i32>,
458    correlations: Option<&CorrelationSelection>,
459    input_view: Option<&dyn PreparedTransformInputView>,
460) -> Result<Vec<f64>, MsTransformError> {
461    let table_error = |source: TableError| MsTransformError::MutateMeasurementSet {
462        path: input
463            .path()
464            .unwrap_or_else(|| std::path::Path::new("<memory>"))
465            .display()
466            .to_string(),
467        source: Box::new(source),
468    };
469    let table = input.main_table();
470    let flags = table.column_accessor("FLAG").map_err(table_error)?;
471    let epoch = |row| {
472        crate::subtables::get_f64(table, row, "TIME")
473            .map_err(|source| MsTransformError::Spectrum(source.to_string()))
474    };
475    groups
476        .iter()
477        .map(|group| {
478            let first = group.first.time;
479            let mut last = group.rows.last().expect("time groups have rows").physical;
480            if group.rows.len() > 1 {
481                let ddid = group.first.description;
482                let selected = &channels[&ddid_to_spw[&ddid]].channels;
483                let axes = correlations.and_then(|plan| plan.axes(ddid));
484                // The time accumulator's last sample is the last row with any
485                // selected sample unflagged, or the last input when all are flagged.
486                // Walk backwards so an ordinary group reads only one FLAG cell.
487                for member in group.rows.iter().rev() {
488                    let indices = [member.physical];
489                    let mut computed = input_view
490                        .map(|view| view.read(table, &indices, &["FLAG"]))
491                        .transpose()
492                        .map_err(table_error)?;
493                    let mut values =
494                        match computed.as_mut().and_then(|values| values.remove("FLAG")) {
495                            Some(values) => values,
496                            None => flags
497                                .array_cells_owned_uncached(&indices)
498                                .map_err(table_error)?,
499                        };
500                    let Some(ArrayValue::Bool(values)) = values.pop().flatten() else {
501                        return Err(table_error(TableError::Schema(
502                            "FLAG must hold Bool.".into(),
503                        )));
504                    };
505                    let view = array_view(&values).map_err(table_error)?;
506                    let is_usable = |corr| {
507                        selected
508                            .iter()
509                            .any(|&chan| view.get((corr, chan)) == Some(&false))
510                    };
511                    let usable = match axes {
512                        Some(axes) => axes.iter().copied().any(is_usable),
513                        None => (0..view.nrows()).any(is_usable),
514                    };
515                    if usable {
516                        last = member.physical;
517                        break;
518                    }
519                }
520            }
521            Ok(first + (epoch(last)? - first) * 0.5)
522        })
523        .collect()
524}

526pub(super) fn read_batch(
527    grid: &Grid,
528    groups: &[Group],
529    input: &MeasurementSet,
530    (rows, descriptions): (&[usize], &[i32]),
531    (channels, ddid_to_spw, correlations): (
532        &BTreeMap<i32, TransformChannelSelection>,
533        &BTreeMap<i32, i32>,
534        Option<&CorrelationSelection>,
535    ),
536    (columns, spectra, input_view, continuum): (
537        &VisibilityColumns,
538        &[&str],
539        Option<&dyn PreparedTransformInputView>,
540        Option<&super::ContinuumFit>,
541    ),
542    time: Option<(&mut TimeProjection<'_>, &[TimeGroup])>,
543) -> Result<Batch, MsTransformError> {
544    let error = |source: TableError| MsTransformError::MutateMeasurementSet {
545        path: input
546            .path()
547            .unwrap_or_else(|| std::path::Path::new("<memory>"))
548            .display()
549            .to_string(),
550        source: Box::new(source),
551    };
552    let sources = groups
553        .iter()
554        .flat_map(|group| group.members.iter().map(|&(_, pos)| pos))
555        .collect::<Vec<_>>();
556    let averaging = time.is_some();
557    let (mut raw, times, exposures, mut row_flags) = if let Some((projection, time_groups)) = time {
558        let plans = sources
559            .iter()
560            .map(|&pos| {
561                let description = descriptions[pos];
562                TimeGroupRead {
563                    group: &time_groups[pos],
564                    channels: &channels[&ddid_to_spw[&description]].channels,
565                    axes: correlations.and_then(|selection| selection.axes(description)),
566                    legacy_bins: None,
567                }
568            })
569            .collect::<Vec<_>>();
570        let (block, times) = projection.average_groups(&plans)?;
571        let exposure = times.iter().map(|time| time.exposure).collect::<Vec<_>>();
572        let flags = times.iter().map(|time| time.flag_row).collect::<Vec<_>>();
573        (block, times, exposure, flags)
574    } else {
575        let indices = sources.iter().map(|&index| rows[index]).collect::<Vec<_>>();
576        let table = input.main_table();
577        let mut block =
578            MainArrayBlock::read(table, &indices, columns, spectra, input_view).map_err(error)?;
579        let exposure = table
580            .column_accessor("EXPOSURE")
581            .and_then(|column| column.scalar_cells_owned_for_rows(&indices))
582            .map_err(error)?;
583        let exposure = exposure
584            .iter()
585            .zip(&indices)
586            .map(|(value, &row)| scalar_f64(value.as_ref(), "EXPOSURE", row))
587            .collect::<Result<Vec<_>, _>>()?;
588        let flags = table
589            .column_accessor("FLAG_ROW")
590            .and_then(|column| column.scalar_cells_owned_for_rows(&indices))
591            .map_err(error)?
592            .iter()
593            .zip(&indices)
594            .map(|(value, &row)| scalar_bool(value.as_ref(), "FLAG_ROW", row))
595            .collect::<Result<Vec<_>, _>>()?;
596        let mut spectra_cells = std::mem::take(&mut block.spectra);
597        block.spectra = STANDARD_SPECTRA
598            .iter()
599            .map(|name| {
600                spectra
601                    .iter()
602                    .position(|source| source == name)
603                    .map_or_else(
604                        || vec![None; indices.len()],
605                        |index| std::mem::take(&mut spectra_cells[index]),
606                    )
607            })
608            .collect();
609        for (row, &pos) in sources.iter().enumerate() {
610            let ddid = descriptions[pos];
611            let axes = correlations.and_then(|plan| plan.axes(ddid));
612            let channels = &channels[&ddid_to_spw[&ddid]].channels;
613            select_row(&mut block, row, columns, axes, channels).map_err(error)?;
614        }
615        (block, Vec::new(), exposure, flags)
616    };
617    if let Some(fit) = continuum {
618        fit.apply(
619            &mut raw,
620            columns,
621            sources
622                .iter()
623                .map(|&position| ddid_to_spw[&descriptions[position]]),
624            &STANDARD_SPECTRA,
625            true,
626        )?;
627        for (row, flags) in row_flags.iter_mut().zip(&raw.flags) {
628            *row = super::continuum::row_flag(flags.as_ref(), None);
629        }
630    }
631    let mut output = MainArrayBlock::for_capacity(groups.len(), columns.copies().len(), 2);
632    let mut coordinates = Vec::with_capacity(if averaging { groups.len() } else { 0 });
633    let mut exposure_flags = Vec::with_capacity(if averaging { 0 } else { groups.len() });
634    let mut start = 0;
635    for group in groups {
636        let end = start + group.members.len();
637        let members = group
638            .members
639            .iter()
640            .zip(start..end)
641            .map(|(&(window, _), index)| (window, index))
642            .collect::<Vec<_>>();
643        let first_exposure = exposures[start];
644        let group_exposures = &exposures[start..end];
645        let different_exposure = group_exposures
646            .iter()
647            .any(|exposure| (exposure - first_exposure).abs() > f64::from(f32::EPSILON));
648        let exposure = group_exposures.iter().sum::<f64>() / group_exposures.len() as f64;
649        let flag_row = row_flags[start..end].iter().all(|&flag| flag);
650        reduce_group(
651            grid,
652            &raw,
653            &members,
654            columns,
655            averaging,
656            different_exposure,
657            &mut output,
658        )
659        .map_err(error)?;
660        if averaging {
661            coordinates.push(TimeCoordinates {
662                exposure,
663                flag_row,
664                ..times[start]
665            });
666        } else {
667            exposure_flags.push((exposure, flag_row));
668        }
669        start = end;
670    }
671    Ok(Batch {
672        arrays: output,
673        coordinates,
674        exposure_flags,
675    })
676}

678fn select_row(
679    block: &mut MainArrayBlock,
680    row: usize,
681    columns: &VisibilityColumns,
682    axes: Option<&[usize]>,
683    channels: &[usize],
684) -> Result<(), TableError> {
685    let select = |cell: &mut Option<ArrayValue>, name, frequency_axis| -> Result<(), TableError> {
686        if let Some(value) = cell.take() {
687            let value = select_visibility_axes(value, axes, name)?;
688            *cell = Some(if frequency_axis {
689                select_channels(value, channels)?
690            } else {
691                value
692            });
693        }
694        Ok(())
695    };
696    for (column, cells) in columns.copies().iter().zip(&mut block.visibility) {
697        select(
698            &mut cells[row],
699            column.source.name(),
700            column.source != crate::schema::main_table::OptionalMainColumn::LagData,
701        )?;
702    }
703    select(&mut block.flags[row], "FLAG", true)?;
704    if let Some(categories) = block.categories[row]
705        .take()
706        .filter(|value| !value.is_empty())
707    {
708        let categories = select_visibility_axes(categories, axes, "FLAG_CATEGORY")?;
709        let bins = channels
710            .iter()
711            .map(|&channel| vec![channel])
712            .collect::<Vec<_>>();
713        block.categories[row] = Some(transform_flag_category_channels(categories, &bins)?);
714    }
715    select(&mut block.weights[row], "WEIGHT", false)?;
716    select(&mut block.sigmas[row], "SIGMA", false)?;
717    for (name, cells) in STANDARD_SPECTRA.into_iter().zip(&mut block.spectra) {
718        if cells[row].as_ref().is_some_and(ArrayValue::is_empty) {
719            cells[row] = None;
720        }
721        select(&mut cells[row], name, true)?;
722    }
723    Ok(())
724}

736fn reduce_group(
737    grid: &Grid,
738    input: &MainArrayBlock,
739    members: &[(usize, usize)],
740    columns: &VisibilityColumns,
741    time_averaged: bool,
742    different_exposure: bool,
743    output: &mut MainArrayBlock,
744) -> Result<(), TableError> {
745    use crate::schema::main_table::OptionalMainColumn::{
746        CorrectedData, Data, FloatData, LagData, ModelData,
747    };
748    let mut rows_by_window = Vec::with_capacity(grid.windows.len());
749    rows_by_window.resize_with(grid.windows.len(), || None);
750    let has = |source| {
751        columns
752            .copies()
753            .iter()
754            .any(|column| column.source == source)
755    };
756    let mut common_shape = None;
757    for &(window, row) in members {
758        let ArrayValue::Bool(flags) = required_cell(&input.flags, row, "FLAG")? else {
759            return Err(TableError::Schema("FLAG must hold Bool".into()));
760        };
761        let flags = array_view(flags)?;
762        let shape = [flags.nrows(), flags.ncols()];
763        let common = common_shape.get_or_insert(shape);
764        if shape[0] != common[0] || (!grid.legacy_cvel && shape[1] != common[1]) {
765            return Err(TableError::Schema(
766                "SPW combination requires compatible correlation and selected channel axes.".into(),
767            ));
768        }
769        let weight = InputStatistic::from_cells(
770            required_cell(&input.weights, row, "WEIGHT")?,
771            input.spectra[0][row].as_ref(),
772            shape,
773            "WEIGHT",
774        )?;
775        let sigma = InputStatistic::from_cells(
776            required_cell(&input.sigmas, row, "SIGMA")?,
777            input.spectra[1][row].as_ref(),
778            shape,
779            "SIGMA",
780        )?;
781        rows_by_window[window] = Some(Window {
782            index: window,
783            row,
784            flags,
785            weight,
786            sigma,
787        });
788    }
789    let first = members[0].1;
790    let correlations = rows_by_window[members[0].0].as_ref().unwrap().flags.nrows();
791    let shape = (correlations, grid.channels.len());
792    let calibrated = has(CorrectedData) || !(has(Data) || has(ModelData));
793    let observed = has(Data) || (time_averaged && has(FloatData));
794    let weight_mode = if (time_averaged && !(observed && has(CorrectedData))) || calibrated {
795        Noise::Calibrated
796    } else if observed {
797        Noise::Observed
798    } else {
799        Noise::Unit
800    };
801    let sigma_mode = if observed && has(CorrectedData) {
802        Noise::Observed
803    } else {
804        weight_mode
805    };
806    let contributions = Contributors::new(grid, &rows_by_window, shape);
807    let different_exposure = different_exposure && !grid.legacy_cvel;
808    for (i, copy) in columns.copies().iter().enumerate() {
809        let cells = &input.visibility[i];
810        let result = match copy.source {
811            LagData => {
812                let original = &cells[first];
813                if members.iter().any(|&(_, row)| &cells[row] != original) {
814                    return Err(TableError::Schema(
815                        "Cannot combine different LAG_DATA cells: their lag axes have no frequency mapping."
816                            .into(),
817                    ));
818                }
819                original.clone()
820            }
821            FloatData => Some(ArrayValue::Float32(
822                contributions
823                    .data(cells, weight_mode, different_exposure)?
824                    .into_dyn(),
825            )),
826            _ => {
827                let mode = match copy.source {
828                    Data => Noise::Observed,
829                    CorrectedData => Noise::Calibrated,
830                    _ => weight_mode,
831                };
832                Some(ArrayValue::Complex32(
833                    contributions
834                        .data(cells, mode, different_exposure)?
835                        .into_dyn(),
836                ))
837            }
838        };
839        output.visibility[i].push(result);
840    }
841    let masks = contributions.flags();
842    let (weight, sigma) = if grid.legacy_cvel {
843        // SubMS combines persisted spectra with the same overlap fractions as DATA.
844        let spectrum = |slot: usize, noise| {
845            input.spectra[slot][first]
846                .as_ref()
847                .map(|_| ArrayValue::Float32(contributions.noise(noise).into_dyn()))
848        };
849        (spectrum(0, Noise::Calibrated), spectrum(1, Noise::Sigma))
850    } else {
851        let weight = contributions.noise(weight_mode);
852        let sigma = if weight_mode == sigma_mode {
853            weight.mapv(sigma_from_weight)
854        } else {
855            contributions.noise(sigma_mode).mapv(sigma_from_weight)
856        };
857        (
858            Some(ArrayValue::Float32(weight.into_dyn())),
859            Some(ArrayValue::Float32(sigma.into_dyn())),
860        )
861    };
862    output.flags.push(Some(ArrayValue::Bool(masks.into_dyn())));
863    output.categories.push(combine_categories(
864        grid,
865        input,
866        &rows_by_window,
867        correlations,
868    )?);
869    output.uvw.push(input.uvw[first].clone());
870    output.weights.push(input.weights[first].clone());
871    output.sigmas.push(input.sigmas[first].clone());
872    output.spectra[0].push(weight);
873    output.spectra[1].push(sigma);
874    Ok(())
875}

877fn category_view(value: &ArrayValue) -> Result<ndarray::ArrayView3<'_, bool>, TableError> {
878    if let ArrayValue::Bool(values) = value
879        && let Ok(view) = values.view().into_dimensionality::<Ix3>()
880    {
881        return Ok(view);
882    }
883    Err(TableError::Schema(
884        "FLAG_CATEGORY must hold rank-3 Bool cells.".into(),
885    ))
886}

888fn combine_categories(
889    grid: &Grid,
890    input: &MainArrayBlock,
891    windows: &[Option<Window<'_>>],
892    correlations: usize,
893) -> Result<Option<ArrayValue>, TableError> {
894    let views = windows
895        .iter()
896        .map(|window| {
897            window
898                .as_ref()
899                .and_then(|window| input.categories[window.row].as_ref())
900                .filter(|cell| !cell.is_empty())
901                .map(category_view)
902                .transpose()
903        })
904        .collect::<Result<Vec<_>, _>>()?;
905    let Some(categories) = views.iter().flatten().next().map(|view| view.dim().2) else {
906        return Ok(None);
907    };
908    for (view, window) in views.iter().zip(windows) {
909        if let (Some(view), Some(window)) = (view, window) {
910            let (corr, channels, count) = view.dim();
911            if corr != correlations
912                || count != categories
913                || (channels != 1 && channels != window.flags.dim().1)
914            {
915                return Err(TableError::Schema(
916                    "FLAG_CATEGORY shapes disagree when combining spectral windows.".into(),
917                ));
918            }
919        }
920    }
921    let output = Array3::from_shape_fn(
922        (correlations, grid.channels.len(), categories),
923        |(corr, channel, category)| {
924            grid.channels[channel].sources.iter().any(|source| {
925                views[source.window].is_some_and(|view| {
926                    view[[corr, source.channel.min(view.dim().1 - 1), category]]
927                })
928            })
929        },
930    );
931    Ok(Some(ArrayValue::Bool(output.into_dyn())))
932}

934struct Window<'a> {
935    index: usize,
936    row: usize,
937    flags: ArrayView2<'a, bool>,
938    weight: InputStatistic<'a>,
939    sigma: InputStatistic<'a>,
940}
941
942#[derive(Clone, Copy, PartialEq, Eq)]
943enum Noise {
944    Calibrated,
945    Observed,
946    Sigma,
947    Unit,
948}
949
950impl Window<'_> {
951    fn noise(&self, kind: Noise, corr: usize, channel: usize) -> f32 {
952        match kind {
953            Noise::Calibrated => self.weight.value(corr, channel),
954            Noise::Observed => weight_from_sigma(self.sigma.value(corr, channel)),
955            Noise::Sigma => self.sigma.value(corr, channel),
956            Noise::Unit => 1.0,
957        }
958    }
959}
960
961struct Contributors<'a> {
962    grid: &'a Grid,
963    windows: &'a [Option<Window<'a>>],
964    shape: (usize, usize),
965}
966
967impl<'a> Contributors<'a> {
968    fn new(grid: &'a Grid, windows: &'a [Option<Window<'a>>], shape: (usize, usize)) -> Self {
969        Self {
970            grid,
971            windows,
972            shape,
973        }
974    }
975
976    fn for_sample(
977        &self,
978        corr: usize,
979        channel: usize,
980        mut add: impl FnMut(&Window<'a>, usize, f64),
981    ) -> f64 {
982        let sources = &self.grid.channels[channel].sources;
983        let has_full = sources
984            .iter()
985            .any(|source| source.fraction >= 1. && self.windows[source.window].is_some());
986        let mut total = 0.;
987        for source in sources {
988            let Some(window) = self.windows[source.window].as_ref() else {
989                continue;
990            };
991            if window.flags[[corr, source.channel]] {
992                continue;
993            }
994            if source.fraction < 1. && has_full {
995                let valid_partials = sources
996                    .iter()
997                    .filter(|other| {
998                        other.window == source.window
999                            && other.fraction < 1.
1000                            && !window.flags[[corr, other.channel]]
1001                    })
1002                    .count();
1003                if valid_partials % 2 != 0 {
1004                    continue;
1005                }
1006            }
1007            total += source.fraction;
1008            add(window, source.channel, source.fraction);
1009        }
1010        total
1011    }
1012
1013    fn flags(&self) -> Array2<bool> {
1014        Array2::from_shape_fn(self.shape, |(corr, channel)| {
1015            self.for_sample(corr, channel, |_, _, _| {}) < 0.999999
1016        })
1017    }
1018
1019    fn noise(&self, kind: Noise) -> Array2<f32> {
1020        Array2::from_shape_fn(self.shape, |(corr, channel)| {
1021            let mut sum = 0.;
1022            let total = self.for_sample(corr, channel, |window, input, fraction| {
1023                sum += fraction * f64::from(window.noise(kind, corr, input));
1024            });
1025            if total >= 0.999999 || (self.grid.legacy_cvel && total > 0.) {
1026                (sum / total) as f32
1027            } else {
1028                sum as f32
1029            }
1030        })
1031    }
1032
1033    fn data<T>(
1034        &self,
1035        cells: &'a [Option<ArrayValue>],
1036        kind: Noise,
1037        different_exposure: bool,
1038    ) -> Result<Array2<T>, TableError>
1039    where
1040        T: VisibilityNumber,
1041    {
1042        let views = self
1043            .windows
1044            .iter()
1045            .map(|window| {
1046                window
1047                    .as_ref()
1048                    .map(|w| T::view(cells[w.row].as_ref()))
1049                    .transpose()
1050            })
1051            .collect::<Result<Vec<_>, _>>()?;
1052        Ok(Array2::from_shape_fn(self.shape, |(corr, channel)| {
1053            let mut sum = T::default();
1054            let mut normalization = 0.;
1055            self.for_sample(corr, channel, |window, input, fraction| {
1056                let weight = if different_exposure {
1057                    fraction * f64::from(window.noise(kind, corr, input))
1058                } else {
1059                    fraction
1060                };
1061                sum += views[window.index].unwrap()[[corr, input]] * weight as f32;
1062                normalization += weight;
1063            });
1064            if normalization > 0. {
1065                sum / normalization as f32
1066            } else {
1067                sum
1068            }
1069        }))
1070    }
1071}

1077}

1101pub(super) fn finish_noise(
1102    grid: &Grid,
1103    statistics: Option<AveragedStatistics>,
1104    averages: Option<&ChannelAverages<'_>>,
1105    spectra: SpectrumCells<'_>,
1106    flags: &ArrayValue,
1107) -> Result<AveragedStatistics, TableError> {
1108    let statistics = statistics.or_else(|| averages.map(ChannelAverages::statistics));
1109    let (weight, sigma) = match statistics {
1110        Some(values) => (values.weight_spectrum, values.sigma_spectrum),
1111        None => {
1112            let missing = || TableError::Schema("Missing combined SPW noise spectrum.".into());
1113            (
1114                spectra.weight.ok_or_else(missing)?.clone(),
1115                spectra.sigma.ok_or_else(missing)?.clone(),
1116            )
1117        }
1118    };
1119    let ArrayValue::Float32(sigma) = sigma else {
1120        return Err(TableError::Schema(
1121            "SIGMA_SPECTRUM must have Float32 cells.".into(),
1122        ));
1123    };
1124    let sigma_weights = ArrayValue::Float32(sigma.mapv(weight_from_sigma));
1125    let transformed_sigma = match grid.interpolate(sigma_weights)? {
1126        ArrayValue::Float32(values) => ArrayValue::Float32(values.mapv(sigma_from_weight)),
1127        _ => unreachable!("a float spectrum stays float"),
1128    };
1129    let weights = grid.interpolate(weight)?;
1130    let flags = grid.interpolate(flags.clone())?;
1131    let ArrayValue::Bool(flags) = flags else {
1132        return Err(TableError::Schema("FLAG must have Bool cells.".into()));
1133    };
1134    let flags = array_view(&flags)?;
1135    Ok(AveragedStatistics {
1136        weight: mean_noise(&weights, flags)?,
1137        sigma: mean_noise(&transformed_sigma, flags)?,
1138        weight_spectrum: weights,
1139        sigma_spectrum: transformed_sigma,
1140    })
1141}

1143pub(super) fn mean_noise(
1144    data: &ArrayValue,
1145    masks: ArrayView2<'_, bool>,
1146) -> Result<ArrayValue, TableError> {
1147    let ArrayValue::Float32(data) = data else {
1148        return Err(TableError::Schema(
1149            "Noise spectra must have Float32 cells.".into(),
1150        ));
1151    };
1152    let data = array_view(data)?;
1153    Ok(ArrayValue::Float32(
1154        Array1::from_shape_fn(data.nrows(), |corr| {
1155            let row = data.row(corr);
1156            let flags = masks.row(corr);
1157            let all_flagged = flags.iter().all(|&flag| flag);
1158            let mut count = 0;
1159            let mut sum = 0.;
1160            for (&value, &flag) in row.iter().zip(flags) {
1161                if flag == all_flagged {
1162                    sum += value;
1163                    count += 1;
1164                }
1165            }
1166            sum / count.max(1) as f32
1167        })
1168        .into_dyn(),
1169    ))
1170}

1172pub(super) fn write_metadata(
1173    request: &MsTransformRequest,
1174    grid: &Grid,
1175    metadata: &mut CompactSpectralMetadata,
1176) -> Result<(), MsTransformError> {
1177    let error = |source: TableError| MsTransformError::MutateMeasurementSet {
1178        path: request.output_ms.display().to_string(),
1179        source: Box::new(source),
1180    };
1181    let mut spw =
1182        Table::open(TableOptions::new(request.output_ms.join("SPECTRAL_WINDOW"))).map_err(error)?;
1183    spw.remove_rows(&(1..spw.row_count()).collect::<Vec<_>>())
1184        .map_err(error)?;
1185    let channels = grid.frequencies.len();
1186    for (name, values) in [
1187        ("CHAN_FREQ", grid.frequencies.clone()),
1188        ("CHAN_WIDTH", vec![grid.width; channels]),
1189        ("EFFECTIVE_BW", vec![grid.width; channels]),
1190        ("RESOLUTION", vec![grid.width; channels]),
1191    ] {
1192        spw.column_accessor_mut(name)
1193            .and_then(|mut column| {
1194                column.set(
1195                    0,
1196                    Value::Array(ArrayValue::Float64(Array1::from_vec(values).into_dyn())),
1197                )
1198            })
1199            .map_err(error)?;
1200    }
1201    for (name, scalar) in [
1202        ("NUM_CHAN", ScalarValue::Int32(channels as i32)),
1203        ("REF_FREQUENCY", ScalarValue::Float64(grid.frequencies[0])),
1204        (
1205            "TOTAL_BANDWIDTH",
1206            ScalarValue::Float64(grid.width * channels as f64),
1207        ),
1208    ] {
1209        spw.column_accessor_mut(name)
1210            .and_then(|mut column| column.set(0, Value::Scalar(scalar)))
1211            .map_err(error)?;
1212    }
1213    for (name, empty) in [
1214        ("ASSOC_SPW_ID", ArrayValue::from_i32_vec(Vec::new())),
1215        ("ASSOC_NATURE", ArrayValue::from_string_vec(Vec::new())),
1216    ] {
1217        if spw
1218            .schema()
1219            .is_some_and(|schema| schema.contains_column(name))
1220        {
1221            spw.column_accessor_mut(name)
1222                .and_then(|mut column| column.set(0, Value::Array(empty)))
1223                .map_err(error)?;
1224        }
1225    }
1226    spw.prepare_write()
1227        .save(TableOptions::new(request.output_ms.join("SPECTRAL_WINDOW")))
1228        .map_err(error)?;
1229    for id in metadata.spw_map.values_mut() {
1230        *id = 0;
1231    }
1232
1233    let path = request.output_ms.join("DATA_DESCRIPTION");
1234    let mut descriptions = Table::open(TableOptions::new(&path)).map_err(error)?;
1235    let mut pol_rows = BTreeMap::new();
1236    let mut mapping = BTreeMap::new();
1237    let mut remove = Vec::new();
1238    for row in 0..descriptions.row_count() {
1239        let pol = crate::subtables::get_i32(&descriptions, row, "POLARIZATION_ID")
1240            .map_err(|source| MsTransformError::Spectrum(source.to_string()))?;
1241        let next = pol_rows.len() as i32;
1242        let output = *pol_rows.entry(pol).or_insert(next);
1243        mapping.insert(row as i32, output);
1244        if output < next {
1245            remove.push(row);
1246        } else {
1247            descriptions
1248                .column_accessor_mut("SPECTRAL_WINDOW_ID")
1249                .and_then(|mut column| column.set(row, Value::Scalar(ScalarValue::Int32(0))))
1250                .map_err(error)?;
1251        }
1252    }
1253    descriptions.remove_rows(&remove).map_err(error)?;
1254    descriptions
1255        .prepare_write()
1256        .save(TableOptions::new(path))
1257        .map_err(error)?;
1258    for id in metadata.ddid_map.values_mut() {
1259        *id = mapping[id];
1260    }
1261    Ok(())
1262}

1264pub(super) fn remap_subtable_references(
1265    request: &MsTransformRequest,
1266) -> Result<(), MsTransformError> {
1267    use crate::subtables::{get_f64, get_i32};
1268    for (table, keys) in [
1269        ("SOURCE", &["SOURCE_ID"][..]),
1270        ("FEED", &["ANTENNA_ID", "FEED_ID", "TIME"]),
1271        ("SYSCAL", &["ANTENNA_ID", "FEED_ID", "TIME"]),
1272        ("CALDEVICE", &["ANTENNA_ID", "FEED_ID", "TIME"]),
1273        ("SYSPOWER", &["ANTENNA_ID", "FEED_ID", "TIME"]),
1274        ("FREQ_OFFSET", &["ANTENNA1", "ANTENNA2", "FEED_ID", "TIME"]),
1275    ] {
1276        let path = request.output_ms.join(table);
1277        if !path.exists() {
1278            continue;
1279        }
1280        let error = |source: TableError| MsTransformError::MutateMeasurementSet {
1281            path: path.display().to_string(),
1282            source: Box::new(source),
1283        };
1284        let subtable = Table::open(TableOptions::new(&path)).map_err(error)?;
1285        let mut records = std::collections::BTreeSet::new();
1286        let mut keep = Vec::new();
1287        for row in 0..subtable.row_count() {
1288            let key = keys
1289                .iter()
1290                .map(|&column| {
1291                    let value = if column == "TIME" {
1292                        get_f64(&subtable, row, column).map(f64::to_bits)
1293                    } else {
1294                        get_i32(&subtable, row, column).map(|value| value as u64)
1295                    };
1296                    value.map_err(|source| MsTransformError::Spectrum(source.to_string()))
1297                })
1298                .collect::<Result<Vec<_>, _>>()?;
1299            if records.insert(key) {
1300                keep.push(row);
1301            }
1302        }
1303        if keep.is_empty() {
1304            continue;
1305        }
1306        drop(subtable);
1307        let original = request.output_ms.join(format!(".{table}.combine-original"));
1308        let io_error = |source: std::io::Error| MsTransformError::PrepareOutput {
1309            path: path.display().to_string(),
1310            reason: source.to_string(),
1311        };
1312        std::fs::rename(&path, &original).map_err(io_error)?;
1313        super::rewrite_selected_table_rows(&original, &path, &keep, |_, mut row| {
1314            row.upsert("SPECTRAL_WINDOW_ID", Value::Scalar(ScalarValue::Int32(0)));
1315            Ok(row)
1316        })?;
1317        std::fs::remove_dir_all(original).map_err(io_error)?;
1318    }
1319    Ok(())
