25const DECODE_BATCH_BYTES: usize = 2 * 1024 * 1024;
26
27/// Convert the observation and scans of a NOSTAR file to MeasurementSet v2.
28///
29/// Independent scan records can be decoded in parallel, with integration and
30/// metadata order preserved. The result includes the NRO_ARRAY mapping.
31///
32/// # Errors
33/// Returns a format error for incomplete records and invalid instrument
34/// coordinates, axes, or array mappings.
35pub fn read_nro(path: &Path, parallel: bool) -> MsResult<ImportedScanTable> {
36    let file = File::open(path).map_err(io_error)?;
37    let size = file.metadata().map_err(io_error)?.len();
38    let mut input = BufReader::new(file);
39    let mut bytes = [0; OBSERVATION_HEADER_BYTES];
40    input.read_exact(&mut bytes).map_err(io_error)?;
41    let header = Header::read(&bytes, size)?;
42    let mut record_bytes = vec![0; header.record_bytes];
43    let (setup, fields, sources) = setup(&mut input, &mut record_bytes, &header)?;
44    let column = if header.arrays[..header.n_arrays]
45        .iter()
46        .flatten()
47        .any(|array| array.plane == 3)
48    {
49        OptionalMainColumn::Data
50    } else {
51        OptionalMainColumn::FloatData
52    };
53    let ms = metadata::create(&setup, column, &fields, &sources)?;
54    let mut writer = Writer::new(ms, column);
55    let mut groups = TimeGroups::default();
56    let mut time = None;
57    let total = header.n_arrays * header.n_scans;
58    let batch_size = (DECODE_BATCH_BYTES / header.record_bytes).max(1);
59    let mut buffer = vec![0; total.min(batch_size) * header.record_bytes];
60    input
61        .seek(SeekFrom::Start(OBSERVATION_HEADER_BYTES as u64))
62        .map_err(io_error)?;
63    for start in (0..total).step_by(batch_size) {
64        let count = (total - start).min(batch_size);
65        let batch = &mut buffer[..count * header.record_bytes];
66        input.read_exact(batch).map_err(io_error)?;
67        for sample in decode_batch(batch, &header, parallel)? {
68            if time != Some(sample.row.time) {
69                for (row, planes) in std::mem::take(&mut groups.current).into_values() {
70                    writer.write(&row, planes, header.n_channels)?;
71                }
72                time = Some(sample.row.time);
73            }
74            groups.push(sample)?;
75        }
76    }
77    for (row, planes) in groups.current.into_values() {
78        writer.write(&row, planes, header.n_channels)?;
79    }
80    Ok(ImportedScanTable {
81        measurement_set: writer.finish()?,
82        input_tables: vec![path.to_path_buf()],
83        auxiliary_tables: vec![(String::from("NRO_ARRAY"), array_table(&header)?)],
84    })
85}

87fn setup(
88    input: &mut BufReader<File>,
89    bytes: &mut [u8],
90    header: &Header,
91) -> MsResult<(ImportHeader, Vec<Field>, Vec<Source>)> {
92    let start = scan::start_time(read_record(input, bytes, header, 0)?)?;
93    let last = (header.n_scans - 1) * header.n_arrays;
94    let end = scan::start_time(read_record(input, bytes, header, last)?)? + header.interval;
95    let mid = 0.5 * (start + end);
96    let diameter = antenna_diameter(&header.site);
97    let antenna = (0..header.n_beam)
98        .map(|beam| Antenna {
99            name: format!("{}-BEAM{beam}", header.site),
100            station: String::new(),
101            position: [-3_871_023.5, 3_428_106.8, 3_724_039.5],
102            diameter,
103        })
104        .collect();
105    let mut windows = BTreeMap::new();
106    let mut sources = Vec::new();
107    for id in 0..header.n_windows as i32 {
108        let (row, array) = header.arrays[..header.n_arrays]
109            .iter()
110            .enumerate()
111            .find_map(|(row, array)| {
112                array
113                    .filter(|array| array.window == id)
114                    .map(|array| (row, array))
115            })
116            .ok_or_else(|| invalid("missing spectral window"))?;
117        let scan = read_record(input, bytes, header, row)?;
118        let rest_frequency = scan.f64(184);
119        let frequency_offset = rest_frequency - array.rest;
120        let mut frequencies = array
121            .frequency
122            .map(|value| (value + frequency_offset) * header.doppler_factor);
123        let mut step = (frequencies[1] - frequencies[0]) / (array.channel[1] - array.channel[0]);
124        if !step.is_finite() {
125            return Err(invalid("spectral calibration"));
126        }
127        if scan.f64(200) > 0. {
128            frequencies.swap(0, 1);
129            step = -step;
130        }
131        windows.insert(
132            id,
133            Window {
134                channels: header.n_channels,
135                frame: header.frequency_frame,
136                reference_channel: array.channel[0] - 1.,
137                reference_frequency: frequencies[0],
138                increment: step,
139            },
140        );
141        sources.push(Source {
142            id: 0,
143            name: header.object.clone(),
144            window: id,
145            time: mid,
146            interval: end - start,
147            direction: header.direction,
148            proper_motion: [0.; 2],
149            rest_frequencies: vec![rest_frequency],
150            transitions: Vec::new(),
151            velocity: header.velocity,
152        });
153    }
154    let field = Field {
155        id: 0,
156        name: header.object.clone(),
157        source_id: 0,
158        time: mid,
159        direction: Direction {
160            position: header.direction,
161            rate: [0.; 2],
162        },
163    };
164    let meta = ImportHeader {
165        antenna,
166        direction_frame: header.direction_frame,
167        telescope: header.site.clone(),
168        observer: header.observer.clone(),
169        project: header.project.clone(),
170        range: [start, end],
171        data_unit: String::from("K"),
172        windows,
173    };
174    Ok((meta, vec![field], sources))
175}

177fn read_record<'a>(
178    input: &mut BufReader<File>,
179    bytes: &'a mut [u8],
180    header: &Header,
181    row: usize,
182) -> MsResult<Record<'a>> {
183    let offset = OBSERVATION_HEADER_BYTES + row * header.record_bytes;
184    input
185        .seek(SeekFrom::Start(offset as u64))
186        .map_err(io_error)?;
187    input.read_exact(bytes).map_err(io_error)?;
188    Ok(Record::new(bytes, header.order))
189}

191fn decode_batch(bytes: &[u8], header: &Header, parallel: bool) -> MsResult<Vec<RawIntegration>> {
192    let decode = |block: &[u8]| {
193        let mut previous_stamp = None;
194        let mut time = 0.;
195        block
196            .chunks_exact(header.record_bytes)
197            .map(|bytes| {
198                let record = Record::new(bytes, header.order);
199                let stamp = record.text(8, 24)?;
200                if previous_stamp != Some(stamp) {
201                    time = scan::start_time(record)?;
202                    previous_stamp = Some(stamp);
203                }
204                scan::decode(record, header, time)
205            })
206            .collect::<MsResult<Vec<_>>>()
207    };
208    if !parallel {
209        return decode(bytes);
210    }
211    let workers = std::thread::available_parallelism()
212        .map_or(2, usize::from)
213        .clamp(2, 8);
214    let group = (bytes.len() / header.record_bytes).div_ceil(workers) * header.record_bytes;
215    std::thread::scope(|scope| {
216        let tasks = bytes
217            .chunks(group)
218            .map(|group| scope.spawn(move || decode(group)))
219            .collect::<Vec<_>>();
220        let mut result = Vec::with_capacity(bytes.len() / header.record_bytes);
221        for task in tasks {
222            result.extend(task.join().map_err(|_| invalid("parallel reader"))??);
223        }
224        Ok(result)
225    })
226}

228fn array_table(header: &Header) -> MsResult<Table> {
229    let names = ["ARRAY", "BEAM", "POLARIZATION", "SPECTRAL_WINDOW"];
230    let schema = TableSchema::new(
231        names
232            .into_iter()
233            .map(|column| ColumnSchema::scalar(column, PrimitiveType::Int32))
234            .collect(),
235    )?;
236    let mut table = Table::with_schema(schema);
237    for (index, array) in header.arrays.iter().enumerate() {
238        let ids = array.map_or([-1; 3], |array| [array.beam, array.code, array.window]);
239        let values = [index as i32, ids[0], ids[1], ids[2]];
240        let row = RecordValue::new(
241            names
242                .into_iter()
243                .zip(values)
244                .map(|(name, id)| RecordField::new(name, i(id)))
245                .collect(),
246        );
247        table.add_row(row)?;
248    }
249    Ok(table)
250}
