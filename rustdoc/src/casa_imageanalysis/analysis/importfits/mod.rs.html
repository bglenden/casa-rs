20/// Import the selected FITS image as a single-precision native CASA image.
21///
22/// The task applies FITS blanking, appends optional degenerate standard axes
23/// in CASA order, and can override FITS restoring beams. Source metadata and
24/// log rows are preserved. A fully written output is published atomically.
25///
26/// # Errors
27///
28/// Returns an error for an unreadable FITS image, invalid selection or
29/// metadata, or an output that cannot be created or replaced.
30pub fn importfits(request: &ImportFitsRequest) -> Result<FitsImportSummary, ImageError> {
31    if request.imagename.as_os_str().is_empty() {
32        return Err(reject("imagename must be specified"));
33    }
34    let output = std::path::absolute(&request.imagename).map_err(io_error)?;
35    check_output(request, &output)?;
36    let which_rep = usize::try_from(request.whichrep)
37        .map_err(|_| reject("The Coordinate Representation index must be non-negative."))?;
38    let beam = restoring_beam(&request.beam)?;
39    let mut source = FitsToolImport::open(&request.fitsimage, which_rep, request.whichhdu)?;
40    let order = axes::apply(&mut source, request)?;
41    let shape = order
42        .iter()
43        .map(|&axis| source.shape().get(axis).copied().unwrap_or(1))
44        .collect::<Vec<_>>();
45
46    if let Some(beam) = beam {
47        replace_restoring_beams(&mut source.image_info, beam);
48    }
49    let info = ImageInfo::from_record(&source.image_info)?;
50    let stage = tempfile::tempdir_in(output.parent().ok_or_else(|| reject("invalid imagename"))?)
51        .map_err(io_error)?;
52    let product = stage.path().join("image");
53    let coordinates = CoordinateSystem::from_record(&source.coordinate_record)?;
54    let mut image = PagedImage::<f32>::create(shape.clone(), coordinates, &product)?;
55    source.copy_pixels_with_axes(&mut image, request.zeroblanks, &order)?;
56    image.set_coordinate_record(source.coordinate_record)?;
57    image.set_units(source.units.clone())?;
58    image.set_image_info_record(source.image_info)?;
59    image.set_misc_info(source.misc_info)?;
60    image.append_history_entries(&source.history)?;
61    image.append_history_at("::importfits", &[task_history(request)])?;
62    image.save()?;
63    drop(image);
64
65    match rename_without_replacing(&product, &output) {
66        Err(error) if error.kind() == io::ErrorKind::AlreadyExists && request.overwrite => {
67            check_output(request, &output)?;
68            exchange_directories(&product, &output).map_err(io_error)?;
69        }
70        result => result.map_err(io_error)?,
71    }
72
73    Ok(FitsImportSummary {
74        fitsimage: request.fitsimage.display().to_string(),
75        imagename: request.imagename.display().to_string(),
76        shape,
77        pixel_type: "Float32".into(),
78        units: source.units,
79        object_name: info.object_name,
80        has_restoring_beam: info
81            .beam_set
82            .single_beam()
83            .is_some_and(|beam| !beam.is_null()),
84    })
85}

87fn check_output(request: &ImportFitsRequest, output: &Path) -> Result<(), ImageError> {
88    let metadata = match output.symlink_metadata() {
89        Ok(metadata) => metadata,
90        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(()),
91        Err(error) => return Err(io_error(error)),
92    };
93    if !request.overwrite {
94        return Err(reject(
95            "Output image already exists and you did not set overwrite to True.",
96        ));
97    }
98    let existing = output.canonicalize().map_err(io_error)?;
99    if request
100        .fitsimage
101        .canonicalize()
102        .is_ok_and(|input| input.starts_with(existing))
103    {
104        return Err(reject("imagename matches or contains the input FITS file"));
105    }
106    let is_image = metadata.is_dir()
107        && Table::open_metadata_only(TableOptions::new(output))
108            .is_ok_and(|table| table.info().table_type == "Image");
109    let is_empty =
110        metadata.is_dir() && fs::read_dir(output).is_ok_and(|mut entries| entries.next().is_none());
111    if !is_image && !is_empty {
112        return Err(reject(format!(
113            "{} is not a CASA image directory",
114            output.display()
115        )));
116    }
117    Ok(())
118}

120fn restoring_beam(values: &[ImportFitsValue]) -> Result<Option<StoredGaussianBeam>, ImageError> {
121    if values.is_empty() {
122        return Ok(None);
123    }
124    let [
125        ImportFitsValue::String(major),
126        ImportFitsValue::String(minor),
127        ImportFitsValue::String(pa),
128    ] = values
129    else {
130        return Err(invalid_type(
131            "beam",
132            "Parameter beam is invalid (should be list of 3 strings or empty)",
133        ));
134    };
135    let radians = Unit::new("rad").expect("built-in angle unit");
136    let seconds = Unit::new("s").expect("built-in time unit");
137    let arcseconds = Unit::new("arcsec").expect("built-in angle unit");
138    let angle = |text: &str| {
139        Quantity::read(text)
140            .ok()
141            .filter(|angle| {
142                angle.value().is_finite()
143                    && (angle.unit().conformant(&radians) || angle.unit().conformant(&seconds))
144            })
145            .ok_or_else(|| {
146                invalid_type(
147                    "beam",
148                    format!("Parameter beam is invalid (should be an angle): {text}"),
149                )
150            })
151    };
152    let major = angle(major)?;
153    let minor = angle(minor)?;
154    let pa = angle(pa)?;
155    let as_arcseconds = |value: &Quantity| {
156        convert(value, Some(&arcseconds))
157            .map(|converted| converted.value())
158            .map_err(|error| invalid_type("beam", error.to_string()))
159    };
160    if as_arcseconds(&major)? < as_arcseconds(&minor)? {
161        return Err(invalid_type(
162            "beam",
163            "major axis must be greater or equal to minor axis",
164        ));
165    }
166    // qa.toangle accepts time during validation, but ia.setrestoringbeam
167    // receives the original units.
168    for (label, angle) in [
169        ("Major axis", &major),
170        ("Minor axis", &minor),
171        ("Position angle", &pa),
172    ] {
173        if !angle.unit().conformant(&radians) {
174            return Err(reject(format!(
175                "Exception: {label} must have angular units ({} is not)..",
176                angle.unit().name()
177            )));
178        }
179    }
180    if major.value() <= 0.0 || minor.value() <= 0.0 {
181        return Err(reject(
182            "Both the major and minor axes must be non-negative.",
183        ));
184    }
185    StoredGaussianBeam::new(major, minor, pa)
186        .map(Some)
187        .map_err(|error| invalid_type("beam", error.to_string()))
188}

190fn replace_restoring_beams(info: &mut RecordValue, beam: StoredGaussianBeam) {
191    let beam = Value::Record(beam.to_record());
192    if let Some(Value::Record(grid)) = info.get_mut("perplanebeams") {
193        let entries: Vec<_> = grid
194            .fields()
195            .iter()
196            .filter(|field| field.name.starts_with('*'))
197            .map(|field| field.name.clone())
198            .collect();
199        for field in entries {
200            grid.upsert(field, beam.clone());
201        }
202    } else {
203        info.upsert("restoringbeam", beam);
204    }
205}

207fn format_importfits_list(values: &[ImportFitsValue]) -> String {
208    let items: Vec<String> = values
209        .iter()
210        .map(|value| match value {
211            ImportFitsValue::String(text) => {
212                if !text.contains('\'') {
213                    format!("'{text}'")
214                } else {
215                    format!("{text:?}")
216                }
217            }
218            ImportFitsValue::Integer(number) => number.to_string(),
219            ImportFitsValue::Float(number) => format!("{number:?}"),
220            ImportFitsValue::Bool(flag) => if *flag { "True" } else { "False" }.to_string(),
221            ImportFitsValue::Other => "None".to_string(),
222        })
223        .collect();
224    format!("[{}]", items.join(", "))
225}

227fn task_history(request: &ImportFitsRequest) -> String {
228    let boolean = |flag| if flag { "True" } else { "False" };
229    format!(
230        "importfits(fitsimage={:?}, imagename={:?}, whichrep={}, whichhdu={}, zeroblanks={}, overwrite={}, defaultaxes={}, defaultaxesvalues={}, beam={})",
231        request.fitsimage.display().to_string(),
232        request.imagename.display().to_string(),
233        request.whichrep,
234        request.whichhdu,
235        boolean(request.zeroblanks),
236        boolean(request.overwrite),
237        boolean(request.defaultaxes),
238        format_importfits_list(&request.defaultaxesvalues),
239        format_importfits_list(&request.beam),
240    )
241}

243fn reject(message: impl Into<String>) -> ImageError {
244    ImageError::CasaSelection(message.into())

247fn invalid_type(parameter: &'static str, message: impl Into<String>) -> ImageError {
248    ImageError::TaskArgumentType {
249        parameter,
250        message: message.into(),
251    }
