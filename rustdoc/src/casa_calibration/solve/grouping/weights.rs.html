16/// Spectral content in the CASA subchunks consumed by the solve.
17#[derive(Default)]
18pub(super) struct SpectralWeights {
19    enabled: HashSet<VisibilityBuffer>,
20    combine: GainSolveCombine,
21}
22
23impl SpectralWeights {
24    pub(super) fn new(
25        ms: &MeasurementSet,
26        rows: &[BucketedRow<'_>],
27        combine: GainSolveCombine,
28    ) -> Result<Self, GainSolveError> {
29        let table = ms.main_table();
30        let available = |name| {
31            table
32                .schema()
33                .is_some_and(|schema| schema.contains_column(name))
34        };
35        if !available("WEIGHT_SPECTRUM") || !available("SIGMA_SPECTRUM") {
36            return Ok(Self::default());
37        }
38        let weight_column = table
39            .column_accessor("WEIGHT_SPECTRUM")
40            .map_err(|source| ms_read_error(ms, source))?;
41        let sigma_column = table
42            .column_accessor("SIGMA_SPECTRUM")
43            .map_err(|source| ms_read_error(ms, source))?;
44        let mut first_rows = HashMap::<VisibilityBuffer, (usize, i32)>::new();
45        for row in rows {
46            let value = (row.0.row_index, row.0.data_spw_id);
47            first_rows
48                .entry(VisibilityBuffer::new(row, combine))
49                .and_modify(|first| *first = (*first).min(value))
50                .or_insert(value);
51        }
52        let mut first_rows = first_rows.into_iter().collect::<Vec<_>>();
53        first_rows.sort_unstable_by_key(|&(buffer, _)| buffer);
54        let first_chunk = first_rows.first().map(|(buffer, _)| buffer.chunk);
55        let mut enabled = HashSet::new();
56        let mut cache = SpectralColumnCache::default();
57        for block in first_rows.chunks(256) {
58            let indices = block.iter().map(|&(_, (row, _))| row).collect::<Vec<_>>();
59            let weights = defined_cells(&weight_column, &indices)
60                .map_err(|source| ms_read_error(ms, source))?;
61            let sigmas = defined_cells(&sigma_column, &indices)
62                .map_err(|source| ms_read_error(ms, source))?;
63            for (&(buffer, (_, spw)), (weight, sigma)) in
64                block.iter().zip(weights.into_iter().zip(sigmas))
65            {
66                cache.visit(
67                    buffer.chunk,
68                    spw,
69                    Some(buffer.chunk) == first_chunk,
70                    weight,
71                    sigma,
72                );
73                if cache.weight_available && cache.sigma_available {
74                    enabled.insert(buffer);
75                }
76            }
77        }
78        Ok(Self { enabled, combine })
79    }
80
81    pub(super) fn contains(&self, row: &BucketedRow<'_>) -> bool {
82        !self.enabled.is_empty()
83            && self
84                .enabled
85                .contains(&VisibilityBuffer::new(row, self.combine))
86    }
87}

89/// CASA's `newSpectralWindow()` and `newMS()` flags are properties of the
90/// outer chunk. Both availability probes use the current subchunk's first row.
91#[derive(Default)]
92struct SpectralColumnCache {
93    chunk: Option<VisibilityChunk>,
94    spw: Option<i32>,
95    new_spw: bool,
96    weight_available: bool,
97    sigma_available: bool,
98}
99
100impl SpectralColumnCache {
101    fn visit(&mut self, chunk: VisibilityChunk, spw: i32, new_ms: bool, weight: bool, sigma: bool) {
102        if self.chunk != Some(chunk) {
103            self.new_spw = self.spw != Some(spw);
104            self.chunk = Some(chunk);
105            self.spw = Some(spw);
106        }
107        if self.new_spw {
108            self.weight_available = weight;
109        }
110        if new_ms && self.weight_available {
111            self.sigma_available = sigma;
112        }
113    }
114}

116/// `CalSolVi2Organizer::deriveVI2Sort` orders outer chunks over these axes.
117/// Array is always a boundary; combined observations/scans are omitted.
118#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
119struct VisibilityChunk {
120    array: i32,
121    observation: i32,
122    scan: i32,
123    field: i32,
124    data_description: i32,
125    time_bucket: SolveBucketKey,
126    combined_field: i32,
127    combined_data_description: i32,
128}
129
130/// The subchunk groups rows at the same exact time inside the outer chunk.
131#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
132pub(in crate::solve) struct VisibilityBuffer {
133    chunk: VisibilityChunk,
134    time: u64,
135}
136
137impl VisibilityBuffer {
138    pub(in crate::solve) fn new(
139        &(row, base, bucket): &BucketedRow<'_>,
140        combine: GainSolveCombine,
141    ) -> Self {
142        Self {
143            chunk: VisibilityChunk {
144                array: row.array_id,
145                observation: base.observation_id,
146                scan: base.scan_number,
147                field: base.field_id,
148                data_description: if combine.spectral_windows {
149                    0
150                } else {
151                    row.data_desc_id
152                },
153                time_bucket: bucket,
154                combined_field: if combine.fields { row.field_id } else { 0 },
155                combined_data_description: if combine.spectral_windows {
156                    row.data_desc_id
157                } else {
158                    0
159                },
160            },
161            time: row.time_seconds.to_bits(),
162        }
163    }
164}

166/// MSIter evaluates a moving FIELD direction at its chunk's first timestamp.
167pub(super) fn chunk_phase_times(
168    rows: &[BucketedRow<'_>],
169    combine: GainSolveCombine,
170    fields: &BTreeSet<i32>,
171) -> BTreeMap<usize, f64> {
172    if fields.is_empty() {
173        return BTreeMap::new();
174    }
175    let mut first = HashMap::<VisibilityChunk, f64>::new();
176    let changing = || {
177        rows.iter()
178            .filter(|(row, _, _)| fields.contains(&row.field_id))
179    };
180    for row in changing() {
181        let time = row.0.time_seconds;
182        first
183            .entry(VisibilityBuffer::new(row, combine).chunk)
184            .and_modify(|first| *first = first.min(time))
185            .or_insert(time);
186    }
187    changing()
188        .map(|row| {
189            (
190                row.0.row_index,
191                first[&VisibilityBuffer::new(row, combine).chunk],
192            )
193        })
194        .collect()
195}

197fn defined_cells(column: &TableColumn<'_>, rows: &[usize]) -> Result<Vec<bool>, TableError> {
198    if let Ok(shapes) = column.array_shapes_uncached(rows) {
199        return Ok(shapes.into_iter().map(|shape| shape.is_some()).collect());
200    }
201    Ok(column
202        .array_cells_owned_uncached(rows)?
203        .into_iter()
204        .map(|cell| cell.is_some())
205        .collect())
206}

208pub(super) fn solve_weights(
209    ms: &MeasurementSet,
210    row_index: usize,
211    shape: &[usize],
212    inputs: (&SolveInputBatch, usize),
213) -> Result<ArrayD<f32>, GainSolveError> {
214    let inverse_square = |sigma: f32| {
215        if sigma > 0.0 {
216            1.0 / (sigma * sigma)
217        } else {
218            0.0
219        }
220    };
221    if inputs.0.spectral_weights[inputs.1] {
222        let sigma = input_cell(ms, row_index, inputs, "SIGMA_SPECTRUM")?;
223        return match sigma {
224            ArrayValue::Float32(sigma) if sigma.shape() == shape => Ok(sigma.mapv(inverse_square)),
225            other => Err(unsupported_shape("SIGMA_SPECTRUM", other)),
226        };
227    }
228    let sigma = input_cell(ms, row_index, inputs, "SIGMA")?;
229    let ArrayValue::Float32(sigma) = sigma else {
230        return Err(unsupported_shape("SIGMA", sigma));
231    };
232    if sigma.ndim() != 1 || sigma.shape()[0] != shape[0] {
233        return Err(GainSolveError::UnsupportedParameterShape {
234            path: "<measurement-set SIGMA>".to_string(),
235            shape: sigma.shape().to_vec(),
236        });
237    }
238    Ok(sigma.mapv(inverse_square).insert_axis(Axis(1)))
239}
