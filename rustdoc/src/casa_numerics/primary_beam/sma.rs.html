2//! PBMath's SMA beam: Float sphfn(3, 6, r), then PBMath1DNumeric sinc sampling.
3
4#[derive(Clone)]
5pub(super) struct SmaVoltageTable(Box<[f32]>);
6
7impl SmaVoltageTable {
8    pub(super) const MAXIMUM_RADIUS: f64 = (52.3 / 2. * 3. / 1.1117) / 60. * 224.;
9
10    pub(super) fn new() -> Self {
11        let samples: [f32; 31] = std::array::from_fn(|k| spheroid((k as f32 / 10.) / 3.));
12        // Real, even length-60 transform. The source FFT has its centre at
13        // sample 30; its (-1)^k shift cancels at the inverse's centre 10000.
14        // Padding preserves the old Nyquist bin, now an ordinary bin, so
15        // every nonzero bin contributes twice in the length-20000 inverse.
16        let coefficients: [f64; 31] = std::array::from_fn(|k| {
17            let end = if k % 2 == 0 {
18                samples[30]
19            } else {
20                -samples[30]
21            };
22            let interior: f64 = (1..30)
23                .map(|i| {
24                    f64::from(samples[i]) * (std::f64::consts::TAU * (k * i) as f64 / 60.).cos()
25                })
26                .sum();
27            f64::from((f64::from(samples[0] + end) + 2. * interior) as f32)
28        });
29        let sample = |index: usize| {
30            (coefficients[0]
31                + 2. * (1..31)
32                    .map(|k| {
33                        coefficients[k] * (std::f64::consts::PI * (k * index) as f64 / 10000.).cos()
34                    })
35                    .sum::<f64>()) as f32
36        };
37        let centre = sample(0);
38        Self(
39            (0..10000)
40                .map(|index| (sample(index) / centre).max(0.).sqrt())
41                .collect(),
42        )
43    }
44
45    pub(super) fn evaluate(&self, radius: f64) -> f32 {
46        if !radius.is_finite() || !(0. ..=Self::MAXIMUM_RADIUS).contains(&radius) {
47            return 0.;
48        }
49        self.0[(radius * 9999. / Self::MAXIMUM_RADIUS)
50            .floor()
51            .clamp(0., 9999.) as usize]
52    }
53}
54
55/// MathFunc2.cc's width-six, alpha-one fit, including its Float rounding
56/// and convolution taper. The gridding correction uses the untapered fit.
57fn spheroid(radius: f32) -> f32 {
58    if radius >= 1. {
59        return 0.;
60    }
61    let (p, q, end): ([f32; 5], [f32; 2], f32) = if radius <= 0.75 {
62        (
63            [0.08203343, -0.3644705, 0.627866, -0.5335581, 0.2312756],
64            [0.8212018, 0.2078043],
65            0.5625,
66        )
67    } else {
68        (
69            [0.004028559, -0.03697768, 0.1021332, -0.1201436, 0.06412774],
70            [0.9599102, 0.2918724],
71            1.,
72        )
73    };
74    let squared = radius * radius;
75    let x = squared - end;
76    let numerator = p[0] + x * (p[1] + x * (p[2] + x * (p[3] + x * p[4])));
77    let denominator = 1. + x * (q[0] + x * q[1]);
78    (1. - squared) * (numerator / denominator)
79}
