71pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
72    if task.flag("listmodels")? {
73        return model_files::list(task);
74    }
75    let path = task.path("vis")?;
76    let table_path = task.workspace().join(&path);
77    if !table_path.exists() {
78        return Err(execution(format!(
79            "Table {} does not exist",
80            table_path.display()
81        )));
82    }
83    let scratch = task.flag("usescratch")?;
84    let value = if let Some(paths) = super::multi_ms::partition_paths(&path)? {
85        run_multi(task, &path, &paths, scratch)?
86    } else {
87        let opened = OpenedMs::open(&path).map_err(execution)?;
88        run_ms(task, &path, opened, &mut [], scratch, false)?
89    };
90    TaskSuccess::new(value, &())
91}

93fn run_multi(
94    task: &TaskInvocation<'_>,
95    path: &Path,
96    paths: &[PathBuf],
97    scratch: bool,
98) -> Result<CasaValue, TaskFailure> {
99    // CASA gives every member a MODEL_DATA column even when a partition
100    // has no selected rows or the requested prediction is virtual.
101    for part in paths {
102        OpenedMs::open(part)
103            .and_then(|opened| opened.initialize_calibrater(false, false, true))
104            .and_then(OpenedMs::release)
105            .map_err(execution)?;
106    }
107    if !scratch {
108        let root = OpenedMs::open(path).map_err(execution)?;
109        let mut parts = paths
110            .iter()
111            .map(|part| OpenedMs::open(part).map_err(execution))
112            .collect::<Result<Vec<_>, _>>()?;
113        let outcome = run_ms(task, path, root, &mut parts, false, false);
114        for part in parts {
115            part.release().map_err(execution)?;
116        }
117        return outcome;
118    }
119
120    let mut first = None;
121    let mut last_failure = None;
122    for part in paths {
123        let opened = OpenedMs::open(part).map_err(execution)?;
124        match run_ms(task, part, opened, &mut [], true, true) {
125            Ok(value) if matches!(&value, CasaValue::Dict { entries } if !entries.is_empty()) => {
126                first.get_or_insert(value);
127            }
128            Ok(_) => {}
129            Err(error) => {
130                tracing::warn!(partition = %part.display(), message = %error.message,
131                    "setjy failed for Multi-MS partition");
132                last_failure = Some(error);
133            }
134        }
135    }
136    first.ok_or_else(|| {
137        let reason =
138            last_failure.map_or_else(|| "no selected rows".to_owned(), |error| error.message);
139        execution(format!("Error in parallel processing of MMS: {reason}"))
140    })
141}

143fn run_ms(
144    task: &TaskInvocation<'_>,
145    path: &Path,
146    opened: OpenedMs,
147    parts: &mut [OpenedMs],
148    scratch: bool,
149    partition: bool,
150) -> Result<CasaValue, TaskFailure> {
151    let mut imager_history = Vec::new();
152    let result = run_ms_holding(
153        task,
154        path,
155        opened,
156        parts,
157        scratch,
158        partition,
159        &mut imager_history,
160    );
161    super::history::record_setjy_imager_flux(path, imager_history);
162    result
163}

165fn run_ms_holding(
166    task: &TaskInvocation<'_>,
167    path: &Path,
168    mut opened: OpenedMs,
169    parts: &mut [OpenedMs],
170    scratch: bool,
171    partition: bool,
172    imager_history: &mut Vec<String>,
173) -> Result<CasaValue, TaskFailure> {
174    let standard = task.text("standard")?;
175    let solar_2012 = standard.as_deref() == Some("Butler-JPL-Horizons 2012");
176    if !scratch
177        && let Some(scale) = standard
178            .as_deref()
179            .filter(|scale| scale.starts_with("Butler-JPL-Horizons"))
180    {
181        return Err(execution(format!(
182            "usescratch=False and standard={scale}. Virtual model can not be used for an ephemeris object calibrator. Use usescratch=True."
183        )));
184    }
185    let mut selection = selection(task)?;
186    if standard.as_deref() == Some("fluxscale") {
187        // The wrapper validates the dictionary before its virtual imager
188        // call applies time, scan or observation predicates.
189        if scratch {
190            add_time_selection(task, &mut selection)?;
191        }
192        let selected = opened.resolve(&selection).map_err(execution)?;
193        if selected.selected_rows.is_empty() {
194            return empty_selection(task, partition);
195        }
196        let result = fluxdict::run(
197            task,
198            opened,
199            parts,
200            selection,
201            &selected,
202            scratch,
203            imager_history,
204        )?;
205        record_history(task, path, partition);
206        return Ok(result);
207    }
208    // The task first checks field, SPW and intent for virtual models.
209    // The imager then applies time, scan and observation to the model selection.
210    let time_filtered = ["timerange", "scan", "observation"].into_iter().any(
211        |name| !matches!(task.value(name), Some(ParameterValue::String(value)) if value.is_empty()),
212    );
213    if !scratch && time_filtered {
214        let initial = opened.resolve(&selection).map_err(execution)?;
215        if initial.selected_rows.is_empty() {
216            return empty_selection(task, partition);
217        }
218    }
219    add_time_selection(task, &mut selection)?;
220    let mut selected = opened.resolve(&selection).map_err(execution)?;
221    if selected.selected_rows.is_empty() {
222        if solar_2012 {
223            return Ok(CasaValue::Dict {
224                entries: Vec::new(),
225            });
226        }
227        if !scratch && time_filtered {
228            return Ok(flux_dictionary(Vec::new()));
229        }
230        return empty_selection(task, partition);
231    }
232    if solar_2012 {
233        return solar2012::run(task, path, opened, &selection, &selected, partition);
234    }
235    let Some(flux) = flux_parameters(task)? else {
236        if scratch {
237            opened = opened
238                .initialize(VisibilityDataColumn::ModelData)
239                .map_err(execution)?;
240        }
241        opened.release().map_err(execution)?;
242        return Ok(CasaValue::Dict {
243            entries: Vec::new(),
244        });
245    };
246    // Imager::sjy_make_visibilities selects all channels of each chosen SPW.
247    selected.channel_selections.clear();
248    let spws = requested_windows(&opened.measurement_set, selection.spw.as_deref())?;
249    let image = model_files::task_image(task)?;
250    let calibration_directions = matches!(flux, FluxParameters::Solar2010(_)) && image.is_none();
251    let PreparedModels {
252        models,
253        fluxes,
254        completed,
255        manual_samples,
256    } = prepare(
257        &opened.measurement_set,
258        &selected,
259        &flux,
260        &spws,
261        task.flag("scalebychan")?,
262        image.as_deref().map(|path| (path, scratch)),
263        imager_history,
264    )?;
265    let has_models = models.values().any(|model| !model.entries.is_empty());
266    let partially_written = !completed && has_models;
267    if scratch {
268        opened = opened
269            .initialize(VisibilityDataColumn::ModelData)
270            .map_err(execution)?;
271        // An unrecognized calibrator reports unit flux and leaves any
272        // existing scratch cells alone.
273        if (!completed || models.values().any(|model| model.entries.is_empty())) && has_models {
274            selection.field = Some(
275                models
276                    .iter()
277                    .filter(|(_, model)| !model.entries.is_empty())
278                    .map(|(field, _)| field.to_string())
279                    .collect::<Vec<_>>()
280                    .join(","),
281            );
282            selected = opened.resolve(&selection).map_err(execution)?;
283            selected.channel_selections.clear();
284        }
285    }
286    publish_models(
287        &mut opened,
288        parts,
289        &selected,
290        models,
291        &spws,
292        scratch,
293        calibration_directions,
294    )?;
295    opened.release().map_err(execution)?;
296    record_history(task, path, partition);
297    record_manual_spectrum(imager_history, &manual_samples);
298    if partially_written {
299        // CASA's cleanup after the later failure raises once an earlier
300        // component list has already been applied and retired.
301        return Err(execution(
302            "flux coefficients are missing for a later field after earlier models were written",
303        ));
304    }
305    Ok(fluxes)
306}

308fn record_history(task: &TaskInvocation<'_>, path: &Path, partition: bool) {
309    if !partition {
310        super::history::record_task(task, path);
311        return;
312    }
313    let mut session = task.session().clone();
314    let updated = session
315        .set(
316            "vis",
317            ParameterValue::String(path.to_string_lossy().into_owned()),
318        )
319        .and_then(|()| session.set("ismms", ParameterValue::Bool(true)));
320    if let Err(error) = updated {
321        tracing::warn!(partition = %path.display(), %error, "updating setjy partition HISTORY");
322        return;
323    }
324    super::history::record_task(&TaskInvocation::new(&session, task.workspace()), path);
325}

327fn publish_models(
328    opened: &mut OpenedMs,
329    parts: &mut [OpenedMs],
330    selected: &ResolvedMsSelection,
331    models: FieldModels,
332    spws: &[i32],
333    scratch: bool,
334    calibration_directions: bool,
335) -> Result<(), TaskFailure> {
336    if scratch {
337        publish_scratch(
338            &mut opened.measurement_set,
339            &mut opened.lock,
340            selected,
341            models,
342            calibration_directions,
343        )?;
344    } else if parts.is_empty() {
345        publish_virtual(&mut opened.measurement_set, &mut opened.lock, models, spws)?;
346    } else {
347        for (field, model) in models.into_iter().rev() {
348            let inputs = std::iter::once(&opened.measurement_set)
349                .chain(parts.iter().map(|part| &part.measurement_set));
350            let components = model
351                .entries
352                .into_iter()
353                .filter(|entry| entry.kind == VirtualModelKind::ComponentList)
354                .collect();
355            let mut publication = VirtualFtPublication::replacing_field(
356                inputs,
357                components,
358                model.images,
359                field,
360                spws,
361            )
362            .map_err(execution)?;
363            publication
364                .publish(
365                    std::iter::once((&mut opened.measurement_set, &mut opened.lock)).chain(
366                        parts
367                            .iter_mut()
368                            .map(|part| (&mut part.measurement_set, &mut part.lock)),
369                    ),
370                )
371                .map_err(execution)?;
372        }
373    }
374    Ok(())
375}

377fn publish_scratch(
378    ms: &mut MeasurementSet,
379    lock: &mut TableWriteLock,
380    selected: &ResolvedMsSelection,
381    models: FieldModels,
382    calibration_directions: bool,
383) -> Result<(), TaskFailure> {
384    // Keep the temporary image owners alive through the scratch write.
385    let entries: Vec<_> = models
386        .values()
387        .flat_map(|model| model.entries.iter().cloned())
388        .collect();
389    let geometry = if calibration_directions && !entries.is_empty() {
390        Some(
391            VisibilityModels::from_entries(ms, &selected.selected_rows, &entries)
392                .and_then(|models| models.with_calibration_field_directions(ms))
393                .map_err(execution)?,
394        )
395    } else {
396        None
397    };
398    write_model_scratch(ms, selected, &entries, false, true, geometry, lock).map_err(execution)
399}

401fn publish_virtual(
402    ms: &mut MeasurementSet,
403    lock: &mut TableWriteLock,
404    models: FieldModels,
405    windows: &[i32],
406) -> Result<(), TaskFailure> {
407    for (field, mut model) in models.into_iter().rev() {
408        let mut publication = VirtualModelPublication::default();
409        let published =
410            publication.replace_windows_holding(ms, &model.entries, lock, field, windows);
411        if publication.may_reference_images() {
412            for image in &mut model.images {
413                image.retain();
414            }
415        }
416        published.map_err(execution)?;
417    }
418    Ok(())
419}

432fn flux_parameters(task: &TaskInvocation<'_>) -> Result<Option<FluxParameters>, TaskFailure> {
433    let pending =
434        |name| TaskFailure::validation("this setjy model configuration is pending #1519").at(name);
435    let standard = task.text("standard")?;
436    let scale = match standard.as_deref() {
437        Some("manual" | "Butler-JPL-Horizons 2010") => None,
438        Some(name) => Some(FluxScale::named(name).ok_or_else(|| pending("standard"))?),
439        None => return Err(pending("standard")),
440    };
441    if !matches!(task.value("fluxdict"), Some(ParameterValue::Table(_))) {
442        return Err(TaskFailure::validation("fluxdict must be a dictionary").at("fluxdict"));
443    }
444    let mut index = task.numbers("spix")?;
445    if index.is_empty() {
446        index.push(0.);
447    }
448    if index[0].abs() > 10. {
449        let limit = if index[0] < -10. {
450            "less than -10"
451        } else {
452            "greater than 10"
453        };
454        return Err(TaskFailure::validation(format!(
455            "The spectral index is {limit}! Please check the spix parameter."
456        ))
457        .at("spix"));
458    }
459    let reference = task.text("reffreq")?.unwrap_or_else(|| "1GHz".into());
460    let Some(reference) = reference_frequency(&reference) else {
461        return Ok(None);
462    };
463    let input = task.numbers("fluxdensity")?;
464    let Some(&intensity) = input.first() else {
465        return Err(TaskFailure::validation("fluxdensity must contain Stokes I").at("fluxdensity"));
466    };
467    if intensity != 0. && standard.as_deref() == Some("Butler-JPL-Horizons 2010") {
468        return Ok(Some(FluxParameters::Solar2010(
469            task.workspace().to_path_buf(),
470        )));
471    }
472    let scale = scale
473        .or_else(|| (intensity < 0.).then_some(FluxScale::Fixed(FixedScale::PerleyButler2010)));
474    if let Some(scale) = scale {
475        let interpolation = if scale.time_interpolation() {
476            match task.text("interpolation")?.as_deref() {
477                Some("nearest") => Curve::Nearest,
478                Some("linear") => Curve::Linear,
479                Some("spline") => Curve::Spline,
480                Some("cubic") => Curve::Cubic,
481                _ => return Ok(None),
482            }
483        } else {
484            Curve::Nearest
485        };
486        // The task suppresses positive user flux on a named standard.
487        // An explicit zero passes through to the manual imager branch.
488        if intensity != 0. {
489            let catalog = FluxCatalog::open(scale).map_err(execution)?;
490            return Ok(Some(FluxParameters::Standard {
491                catalog,
492                interpolation,
493            }));
494        }
495    }
496    if input.len() > 4 {
497        return Err(
498            TaskFailure::validation("manual setjy needs one to four Stokes flux values")
499                .at("fluxdensity"),
500        );
501    }
502    let mut flux = [0.; 4];
503    flux[..input.len()].copy_from_slice(&input);
504    ManualFlux::new(
505        flux,
506        reference,
507        index,
508        task.numbers("polindex")?,
509        task.numbers("polangle")?,
510        task.number("rotmeas")?,
511    )
512    .map(|flux| Some(FluxParameters::Manual(flux)))
513    .map_err(execution)
514}

516fn reference_frequency(text: &str) -> Option<MFrequency> {
517    let text = text.trim();
518    let mut frame = FrequencyRef::LSRK;
519    let mut quantity = text;
520    if let Some((prefix, rest)) = text.split_once(char::is_whitespace)
521        && let Ok(CasaFrequencyFrame::Defined(parsed)) = prefix.parse()
522    {
523        frame = parsed;
524        quantity = rest.trim();
525    }
526    let quantity = quantity.parse::<Quantity>().ok()?;
527    let hz = quantity.get_value_in(&Unit::new("Hz").expect("Hz")).ok()?;
528    Some(MFrequency::new(hz, frame))
529}

531fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {

537        spw: text("spw")?,
538        intent: text("intent")?,
539        ..MsSelection::default()
540    })

543fn add_time_selection(
544    task: &TaskInvocation<'_>,
545    selection: &mut MsSelection,
546) -> Result<(), TaskFailure> {
547    let text = |name, argument| match task.value(name) {
548        Some(ParameterValue::String(s)) => Ok((!s.is_empty()).then(|| s.clone())),
549        _ => Err(TaskFailure::validation(format!("argument {argument} must be a string")).at(name)),
550    };
551    selection.timerange = text("timerange", "_time")?;
552    selection.scan = text("scan", "_scan")?;
553    selection.observation = super::casa_read::selector_text(task.value("observation"));
554    Ok(())
555}

557fn prepare(
558    ms: &MeasurementSet,
559    selected: &ResolvedMsSelection,
560    flux: &FluxParameters,
561    spws: &[i32],
562    by_channel: bool,
563    image: Option<(&Path, bool)>,
564    imager_history: &mut Vec<String>,
565) -> Result<PreparedModels, TaskFailure> {
566    let mut groups = BTreeMap::<i32, BTreeMap<i32, Vec<usize>>>::new();
567    for row in &selected.selected_rows {
568        groups
569            .entry(row.field_id)
570            .or_default()
571            .entry(row.spectral_window_id)
572            .or_default()
573            .push(row.row_index);
574    }
575    let engine = if matches!(flux, FluxParameters::Solar2010(_)) && image.is_none() {
576        MsCalEngine::for_calibration(ms)
577    } else {
578        MsCalEngine::new(ms)
579    }
580    .map_err(execution)?;
581    let field_table = ms.field().map_err(execution)?;
582    let windows = ms.spectral_window().map_err(execution)?;
583    let time = ms.main_table().column_accessor("TIME").map_err(execution)?;
584    let ScalarValue::Float64(epoch) = time.scalar_cell(0).map_err(execution)? else {
585        return Err(execution("expected a double-precision visibility epoch"));
586    };
587    let frame = engine
588        .antenna_frame(*epoch, selected.selected_rows[0].antenna1_id as usize)
589        .map_err(execution)?
590        .with_position(engine.observatory_position().clone());
591    let mut models = FieldModels::new();
592    let mut returned = BTreeMap::new();
593    let mut completed = true;
594    let mut manual_samples = Vec::new();
595    let report_manual =
596        matches!(flux, FluxParameters::Manual(manual) if manual.reports_spectral_index());
597    for (field, windows_for_field) in groups.into_iter().rev() {
598        manual_samples.clear();
599        let direction = engine
600            .observation_direction_j2000(*epoch, field as usize)
601            .map_err(execution)?;
602        let frame = frame.clone().with_direction(direction.clone());
603        let name = field_table.name(field as usize).map_err(execution)?;
604        let spectrum = match flux {
605            FluxParameters::Standard {
606                catalog,
607                interpolation,
608            } => {
609                let midpoint = field_midpoint(&time, &windows_for_field)?;
610                match catalog
611                    .spectrum(&name, &direction, midpoint, *interpolation)
612                    .map_err(execution)?
613                {
614                    CatalogMatch::Spectrum(spectrum) => Some(spectrum),
615                    CatalogMatch::UnknownSource => None,
616                    CatalogMatch::MissingCoefficients(source) => {
617                        // The failed field never reaches the model clear. Earlier
618                        // fields remain eligible for publication.
619                        tracing::warn!(field = %name, source, "source has no coefficients for the requested flux scale");
620                        completed = false;
621                        break;
622                    }
623                }
624            }
625            FluxParameters::Manual(_) | FluxParameters::Solar2010(_) => None,
626        };
627        let solar = if let FluxParameters::Solar2010(workspace) = flux
628            && image.is_none()
629        {
630            let midpoint = MEpoch::from_mjd(
631                field_midpoint(&time, &windows_for_field)?,
632                engine.time_reference(),
633            );
634            match Solar2010::open(workspace, &name, &midpoint, &frame) {
635                Ok(solar) => Some(solar),
636                Err(error) => {
637                    tracing::warn!(field = %name, %error, "no 2010 Solar System flux model");
638                    completed = false;
639                    break;
640                }
641            }
642        } else {
643            None
644        };
645        let model = models.entry(field).or_default();
646        let mut field_flux = BTreeMap::new();
647        let mut first_jy = None;
648        for &spw in spws {
649            let (planned, num_channels) = model_frequencies(&windows, spw as usize, by_channel)?;
650            let (component, stokes) = if image.is_some() {
651                (
652                    None,
653                    [
654                        spectrum
655                            .as_ref()
656                            .map_or(1., |spectrum| spectrum.flux(planned[0].hz())),
657                        0.,
658                        0.,
659                        0.,
660                    ],
661                )
662            } else if let Some(solar) = &solar {
663                let (component, stokes) = solar
664                    .component(Some(direction.clone()), &planned)
665                    .map_err(execution)?;
666                imager_history.push(imager_flux_history(
667                    &name,
668                    field,
669                    spw,
670                    planned[0].hz(),
671                    stokes,
672                    "JPL-Butler Solar System Object",
673                ));
674                (Some(component), stokes)
675            } else {
676                point_model(flux, spectrum.as_ref(), &direction, &planned, &frame)?
677            };
678            if let FluxParameters::Standard { catalog, .. } = flux
679                && spectrum.is_some()
680            {
681                imager_history.push(imager_flux_history(
682                    &name,
683                    field,
684                    spw,
685                    planned[0].hz(),
686                    stokes,
687                    catalog.standard_name(),
688                ));
689            } else if report_manual {
690                manual_samples.push((planned[0].hz(), stokes[0]));
691            }
692            first_jy.get_or_insert(stokes[0]);
693            if let (Some(rows), Some(component)) = (windows_for_field.get(&spw), component) {
694                let container = ComponentList::from(vec![component])
695                    .to_record()
696                    .map_err(execution)?;
697                model.entries.push(VirtualModelEntry {
698                    kind: VirtualModelKind::ComponentList,
699                    index_combination: selected_model_combinations(ms, rows)
700                        .map_err(execution)?
701                        .index_combination,
702                    channel_selection: vec![[spw, 0, num_channels, 1]],
703                    container,
704                });
705            }
706            field_flux.insert(spw.to_string(), window_flux_reply(stokes));
707        }
708        if let Some((path, scratch)) = image {
709            let params = resolved::ImageParameters {
710                path,
711                by_channel,
712                scratch,
713                windows: spws,
714            };
715            use resolved::ResolvedFlux;
716            let image_spectrum =
717                image_catalog_spectrum(flux, by_channel, &name, &direction, *epoch)?;
718            let flux = match flux {
719                FluxParameters::Manual(manual) if manual.intensity_jy() != 0. => {
720                    ResolvedFlux::Manual(manual)
721                }
722                FluxParameters::Standard { .. } => {
723                    ResolvedFlux::Catalog(image_spectrum.as_ref(), first_jy.unwrap_or(1.))
724                }
725                FluxParameters::Solar2010(_) => ResolvedFlux::Catalog(None, 1.),
726                _ => ResolvedFlux::Unscaled,
727            };
728            model.images = params.prepare(ms, selected, &frame, field, flux, imager_history)?;
729            model
730                .entries
731                .extend(model.images.iter().map(|image| image.entry.clone()));
732        }
733        field_flux.insert("fieldName".into(), CasaValue::String { value: name });
734        returned.insert(
735            field.to_string(),
736            CasaValue::Dict {
737                entries: field_flux
738                    .into_iter()
739                    .map(|(key, value)| entry(key, value))
740                    .collect(),
741            },
742        );
743    }
744    let entries = returned
745        .into_iter()
746        .map(|(key, value)| entry(key, value))
747        .collect();
748    let returned = if completed {
749        flux_dictionary(entries)
750    } else {
751        CasaValue::Dict { entries }
752    };
753    Ok(PreparedModels {
754        models,
755        fluxes: returned,
756        completed,
757        manual_samples,
758    })
759}

763        entries: vec![entry(
764            "fluxd",
765            CasaValue::FloatArray {
766                dtype: FloatDtype::Float64,

768                data: stokes.map(CasaFloat).to_vec(),

774fn image_catalog_spectrum<'a>(
775    flux: &'a FluxParameters,
776    by_channel: bool,
777    name: &str,
778    direction: &MDirection,
779    epoch: f64,
780) -> Result<Option<CalibratorSpectrum<'a>>, TaskFailure> {
781    let FluxParameters::Standard {
782        catalog,
783        interpolation,
784    } = flux
785    else {
786        return Ok(None);
787    };
788    if !by_channel {
789        return Ok(None);
790    }
791    match catalog
792        .spectrum(name, direction, epoch / 86_400., *interpolation)
793        .map_err(execution)?
794    {
795        CatalogMatch::Spectrum(spectrum) => Ok(Some(spectrum)),
796        _ => Ok(None),
797    }
798}

800fn imager_flux_history(
801    name: &str,
802    field: i32,
803    spw: i32,
804    hz: f64,
805    stokes: [f64; 4],
806    standard: &str,
807) -> String {
808    let [i, q, u, v] = stokes.map(|flux| stream_double_precision(flux, 5));
809    let frequency = stream_double_precision(hz, 5);
810    format!(
811        "{name:>12} (fld ind {field}) spw {spw}  [I={i}, Q={q}, U={u}, V={v}] Jy @ {frequency}Hz, ({standard})"
812    )
813}

815fn record_manual_spectrum(history: &mut Vec<String>, samples: &[(f64, f64)]) {
816    if samples.is_empty() {
817        return;
818    }
819    history.push(
820        "Flux density as a function of frequency (channel 0 of each spw):\n  Frequency (GHz)    Flux Density (Jy, Stokes I)".into(),
821    );
822    history.extend(samples.iter().map(|&(hz, jy)| {
823        let ghz = stream_double_precision(hz / 1e9, 6);
824        let intensity = stream_double_precision(jy, 6);
825        format!("     {ghz}         {intensity}")
826    }));
827}

829fn field_midpoint(
830    time: &casa_tables::TableColumn<'_>,
831    windows: &BTreeMap<i32, Vec<usize>>,
832) -> Result<f64, TaskFailure> {
833    let rows = windows.values().flatten().copied();
834    let first = rows.clone().min().expect("a selected field contains rows");
835    let last = rows.max().expect("a selected field contains rows");
836    let (ScalarValue::Float64(first), ScalarValue::Float64(last)) = (
837        time.scalar_cell(first).map_err(execution)?,
838        time.scalar_cell(last).map_err(execution)?,
839    ) else {
840        return Err(execution("expected double-precision visibility epochs"));
841    };
842    Ok((first + (last - first) * 0.5) / 86_400.)
843}

845fn point_model(
846    flux: &FluxParameters,
847    spectrum: Option<&CalibratorSpectrum<'_>>,
848    direction: &MDirection,
849    frequencies: &[MFrequency],
850    frame: &MeasFrame,
851) -> Result<(Option<SkyComponent>, [f64; 4]), TaskFailure> {
852    let (component, stokes) = match (flux, spectrum) {
853        (FluxParameters::Manual(manual), _) => {
854            manual.component(direction.clone(), frequencies, frame)
855        }
856        (_, Some(spectrum)) => {
857            let samples = frequencies
858                .iter()
859                .map(|frequency| [spectrum.flux(frequency.hz()), 0., 0., 0.])
860                .collect();
861            sampled_point(direction.clone(), frequencies, samples)
862        }
863        (_, None) => return Ok((None, [1., 0., 0., 0.])),
864    }
865    .map_err(execution)?;
866    Ok((Some(component), stokes))
867}

869fn requested_windows(ms: &MeasurementSet, spw: Option<&str>) -> Result<Vec<i32>, TaskFailure> {
870    if spw.is_some() {
871        return virtual_model_clear_ids(ms, None, spw)
872            .map(|(_, spws)| spws)
873            .map_err(execution);
874    }
875    let count = ms.spectral_window().map_err(execution)?.row_count();
876    Ok((0..i32::try_from(count).map_err(execution)?).collect())
877}

888
889fn model_frequencies(
890    windows: &MsSpectralWindow<'_>,
891    spw: usize,
892    by_channel: bool,
893) -> Result<(Vec<MFrequency>, i32), TaskFailure> {
894    let mut frequencies = windows.chan_freq(spw).map_err(execution)?;
895    let frequency_frame = windows.frequency_reference(spw).map_err(execution)?;
896    let num_channels = i32::try_from(frequencies.len()).map_err(execution)?;
897    // CASA assigns the scalar median through Vector::set, which constructs
898    // an MFrequency in the default LSRK frame.
899    let frame = if by_channel {
900        frequency_frame
901    } else {
902        FrequencyRef::LSRK
903    };
904    if frequencies.is_empty() || frequencies.iter().any(|f| !f.is_finite() || *f <= 0.) {
905        return Err(execution("setjy needs positive finite channel frequencies"));
906    }
907    if !by_channel {
908        // casacore averages the two middle values only for even counts up to 100.
909        let count = frequencies.len();
910        let (_, &mut lower, upper) =
911            frequencies.select_nth_unstable_by((count - 1) / 2, f64::total_cmp);
912        let median = if count.is_multiple_of(2) && count <= 100 {
913            0.5 * (lower + upper.iter().copied().fold(f64::INFINITY, f64::min))
914        } else {
915            lower
916        };
917        frequencies = vec![median];
918    }
919    let planned = frequencies
920        .into_iter()
921        .map(|hz| MFrequency::new(hz, frame))
922        .collect();
923    Ok((planned, num_channels))
924}
