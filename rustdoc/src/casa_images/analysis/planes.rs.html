2//! Planes of an image: `feather`'s direction planes and `impv`'s samples
3//! along a path in one, with the plane on any two axes (#1002).
4
5use casa_types::ArrayD;
6use ndarray::{Array2, IxDyn, ShapeBuilder};
7
8use super::axes::other_axes;

10/// One plane of an image: its two `axes`, at `values` of the `others`.
11pub(super) struct PlaneAt<'a> {
12    pub(super) axes: [usize; 2],
13    pub(super) others: &'a [usize],
14    pub(super) values: &'a [usize],
15}
16
17impl PlaneAt<'_> {
18    /// The index of the plane's pixel (`x`, `y`) in the image.
19    fn index(&self, rank: usize, x: usize, y: usize) -> Vec<usize> {
20        let mut index = vec![0; rank];
21        for (&axis, &value) in self.others.iter().zip(self.values) {
22            index[axis] = value;
23        }
24        index[self.axes[0]] = x;
25        index[self.axes[1]] = y;
26        index
27    }
28
29    pub(super) fn extract(&self, data: &ArrayD<f32>) -> Array2<f32> {
30        let shape = (data.shape()[self.axes[0]], data.shape()[self.axes[1]]);
31        Array2::from_shape_fn(shape, |(x, y)| data[IxDyn(&self.index(data.ndim(), x, y))])
32    }
33
34    pub(super) fn put(&self, data: &mut ArrayD<f32>, plane: &Array2<f32>) {
35        for ((x, y), value) in plane.indexed_iter() {
36            let index = self.index(data.ndim(), x, y);
37            data[IxDyn(&index)] = *value;
38        }
39    }
40}
41
42/// The index of element `value` of an array of `shape`, the last axis
43/// fastest.
44pub(super) fn unravel_index(mut value: usize, shape: &[usize]) -> Vec<usize> {
45    if shape.is_empty() {
46        return Vec::new();
47    }
48    let mut out = vec![0usize; shape.len()];
49    for axis in (0..shape.len()).rev() {
50        out[axis] = value % shape[axis];
51        value /= shape[axis];
52    }
53    out
54}

56/// impv's samples: at each pixel of `path` in the image plane `plane`, the
57/// mean of `input` over `offsets` about it that fall in the image, for every
58/// pixel of the other axes. Output channel `j` of `channels`' axis comes from
59/// input channel `channels.1[j]`.
60pub(super) fn pv_samples<T>(
61    input: &ArrayD<T>,
62    plane: [usize; 2],
63    path: &[(isize, isize)],
64    offsets: &[(isize, isize)],
65    channels: Option<&(usize, Vec<usize>)>,
66    output_shape: &[usize],
67) -> ArrayD<f32>
68where
69    T: Into<f64> + Copy,
70{
71    let [x_axis, y_axis] = plane;
72    let others = other_axes(output_shape.len(), plane);
73    let other_shape = others
74        .iter()
75        .map(|&axis| output_shape[axis])
76        .collect::<Vec<_>>();
77    let mut output = ArrayD::<f32>::zeros(IxDyn(output_shape).f());
78    let mut input_index = vec![0; input.ndim()];
79    let mut output_index = vec![0; output_shape.len()];
80    for sample_index in all_indices_for_shape(&other_shape) {
81        for (&axis, &value) in others.iter().zip(&sample_index) {
82            output_index[axis] = value;
83            input_index[axis] = match channels {
84                Some((channel_axis, local)) if *channel_axis == axis => local[value],
85                _ => value,
86            };
87        }
88        for (path_index, &(x, y)) in path.iter().enumerate() {
89            let mut sum = 0.0;
90            let mut count = 0usize;
91            for &(dx, dy) in offsets {
92                let (Ok(sx), Ok(sy)) = (usize::try_from(x + dx), usize::try_from(y + dy)) else {
93                    continue;
94                };
95                if sx >= input.shape()[x_axis] || sy >= input.shape()[y_axis] {
96                    continue;
97                }
98                input_index[x_axis] = sx;
99                input_index[y_axis] = sy;
100                sum += input[IxDyn(&input_index)].into();
101                count += 1;
102            }
103            if count > 0 {
104                output_index[x_axis] = path_index;
105                output_index[y_axis] = 0;
106                output[IxDyn(&output_index)] = (sum / count as f64) as f32;
107            }
108        }
109    }
110    output
111}
112
113fn all_indices_for_shape(shape: &[usize]) -> Vec<Vec<usize>> {
114    fn push(shape: &[usize], axis: usize, current: &mut Vec<usize>, out: &mut Vec<Vec<usize>>) {
115        if axis == shape.len() {
116            out.push(current.clone());
117            return;
118        }
119        for value in 0..shape[axis] {
120            current.push(value);
121            push(shape, axis + 1, current, out);
122            current.pop();
123        }
124    }
125    let mut out = Vec::new();
126    push(shape, 0, &mut Vec::new(), &mut out);
127    out
128}
