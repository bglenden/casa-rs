3//! Exact failure reasons for continuum product construction and output.
4
5use std::path::PathBuf;
6
7use casa_imaging_model::{ProductRole, ProductTerm};
8use thiserror::Error;
9
10/// Why product planning, generation or writing failed.
11#[derive(Debug, Error)]
12pub enum ProductsError {
13    /// A product image could not be created, written or saved.
14    #[error("writing {}: {source}", path.display())]
15    Image {
16        /// The image's published path.
17        path: PathBuf,
18        /// What the image layer reported.
19        #[source]
20        source: casa_images::ImageError,
21    },
22    /// Staging or replacing a product image failed.
23    #[error("{operation} {}: {source}", path.display())]
24    Io {
25        /// What the writer was doing.
26        operation: &'static str,
27        /// The directory or image it was doing it to.
28        path: PathBuf,
29        /// What the system reported.
30        #[source]
31        source: std::io::Error,
32    },
33    /// A product window exceeded its shape or admitted capacity.
34    #[error("product window exceeds its admitted shape or capacity")]
35    InvalidWindow,
36    /// Ordered product coverage was missing, repeated, or reordered.
37    #[error("product window coverage requires channel {expected}, found {actual}")]
38    WindowCoverage {
39        /// Next required channel, or complete channel count at output finish.
40        expected: usize,
41        /// Actual start or completed channel count.
42        actual: usize,
43    },
44    /// An authoritative Normal State window could not be read.
45    #[error(transparent)]
46    NormalAccess(#[from] casa_imaging_deconvolution::NormalStateError),
47    /// An authoritative model plane could not be read for product generation.
48    #[error(transparent)]
49    ModelAccess(#[from] casa_imaging_deconvolution::ModelLifecycleError),
50    /// A direction-dependent sensitivity plane has no positive value: no
51    /// weighted sample reached it.
52    #[error("the sensitivity plane has no positive support")]
53    UnsupportedSensitivity,
54    /// Direction-dependent image units need flat-noise or flat-sky
55    /// normalization.
56    #[error("direction-dependent image units need flat-noise or flat-sky normalization")]
57    UnsupportedNormalization,
58    /// Production controls were outside their validated ranges.
59    #[error("continuum production controls are invalid")]
60    InvalidControls,
61    /// The compiled problem asks for a product rule these products do not
62    /// implement.
63    #[error("the compiled problem asks for a product rule the products do not implement")]
64    UnsupportedProblem,
65    /// A PB product or PB-masked member was planned without a primary-beam
66    /// model in the controls.
67    #[error("a primary-beam product or mask needs a primary-beam model")]
68    MissingPrimaryBeamModel,
69    /// The selected beam law cannot represent an output channel's frequency.
70    #[error(
71        "primary-beam model {model:?} does not support output channel {output_channel} at {frequency_hz} Hz"
72    )]
73    UnsupportedPrimaryBeamFrequency {
74        /// Explicitly selected primary-beam law.
75        model: crate::AnalyticPrimaryBeamModel,
76        /// Zero-based output channel, after spectral coordinate conversion.
77        output_channel: usize,
78        /// Output channel centre, in Hz.
79        frequency_hz: f64,
80    },
81    /// The product list publishes a role the products cannot produce.
82    #[error("product role {role:?} cannot be produced")]
83    UnsupportedProductRole {
84        /// The first unsupported role encountered.
85        role: ProductRole,
86    },
87    /// A Taylor term other than the zeroth coefficient was requested.
88    #[error("Taylor term {term:?} exceeds the nterms=1 continuum algorithm catalog")]
89    UnsupportedTaylorTerm {
90        /// The offending Taylor term.
91        term: ProductTerm,
92    },
93    /// A Taylor run's principal sum of weights, normal or published, is not
94    /// positive and finite: no weighted sample reached the image.
95    #[error("Taylor products need a positive principal sum of weights, not {sum_weight}")]
96    PrincipalSumWeight {
97        /// The principal sum of weights.
98        sum_weight: f64,
99    },
100    /// A produced payload length disagreed with its declared shape.
101    #[error("member payload requires {expected} values but carries {actual}")]
102    PayloadLengthMismatch {
103        /// Shape-derived value count.
104        expected: usize,
105        /// Actual payload length.
106        actual: usize,
107    },
108    /// Beam fitting failed for products that carry fitted beam metadata.
109    #[error("restoring-beam fitting failed: {0}")]
110    BeamFitFailed(String),
111    /// Solver arithmetic produced a non-finite product value.
112    #[error("product arithmetic generated a non-finite value")]
113    GeneratedNonfinite,
114    /// Checked product-generation resource arithmetic exceeded `u64`.
115    #[error("continuum generation demand overflowed while calculating {0}")]
116    ResourceDemandOverflow(&'static str),
117}
