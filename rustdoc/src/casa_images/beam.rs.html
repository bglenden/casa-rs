10/// Read the scalar or one-element real quantity forms CASA accepts.
11pub(crate) fn quantity_from_record(record: &RecordValue) -> Result<Quantity, ImageError> {
12    let number = match record.get("value") {
13        Some(Value::Scalar(ScalarValue::Float64(v))) => Some(*v),
14        Some(Value::Scalar(ScalarValue::Float32(v))) => Some(f64::from(*v)),
15        Some(Value::Scalar(ScalarValue::Int32(v))) => Some(f64::from(*v)),
16        Some(Value::Array(ArrayValue::Float64(v))) if v.len() == 1 => v.iter().next().copied(),
17        Some(Value::Array(ArrayValue::Float32(v))) if v.len() == 1 => {
18            v.iter().next().map(|v| f64::from(*v))
19        }
20        Some(Value::Array(ArrayValue::Int32(v))) if v.len() == 1 => {
21            v.iter().next().map(|v| f64::from(*v))
22        }
23        _ => None,
24    }
25    .ok_or_else(|| ImageError::InvalidMetadata("invalid quantity value".into()))?;
26    let Some(Value::Scalar(ScalarValue::String(unit))) = record.get("unit") else {
27        return Err(ImageError::InvalidMetadata("invalid quantity unit".into()));
28    };
29    let unit = casa_types::quanta::casacore::casacore_unit(unit)
30        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
31    Ok(Quantity::with_unit(number, unit))
32}

56/// Wrap an angular quantity to CASA's (-90, 90] degree range, preserving its
57/// unit and casacore's conversion order.
58///
59/// # Errors
60///
61/// Returns an error for a non-angular or non-finite quantity.
62pub fn wrap_position_angle(pa: &Quantity) -> Result<Quantity, ImageError> {
63    let degree = Unit::new("deg").expect("built-in degrees");
64    if !pa.value().is_finite() || !pa.unit().conformant(&degree) {
65        return Err(ImageError::InvalidMetadata(
66            "position angle must be a finite angular quantity".into(),
67        ));
68    }
69    let quarter = Quantity::new(90.0, "deg").expect("built-in degrees");
70    let threshold = casa_types::quanta::casacore::convert(&quarter, Some(pa.unit()))
71        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?
72        .value();
73    if pa.value() > threshold || pa.value() <= -threshold {
74        let mut degrees = casa_types::quanta::casacore::get_conformant_value(pa, &degree)
75            .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?
76            % 180.0;
77        if degrees > 90.0 {
78            degrees -= 180.0;
79        } else if degrees <= -90.0 {
80            degrees += 180.0;
81        }
82        let wrapped = Quantity::with_unit(degrees, degree);
83        casa_types::quanta::casacore::convert(&wrapped, Some(pa.unit()))
84            .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
85    } else {
86        Ok(pa.clone())
87    }
88}

90impl StoredGaussianBeam {
91    /// Construct a beam from angular quantities, retaining their units.
92    ///
93    /// # Errors
94    ///
95    /// Returns an error for non-angular, non-finite, negative, or
96    /// incorrectly ordered axes, or a non-angular position angle.
97    pub fn new(
98        major: Quantity,
99        minor: Quantity,
100        position_angle: Quantity,
101    ) -> Result<Self, ImageError> {
102        let radians = Unit::new("rad").expect("built-in unit");
103        for (label, quantity) in [
104            ("major", &major),
105            ("minor", &minor),
106            ("position angle", &position_angle),
107        ] {
108            if !quantity.value().is_finite() || !quantity.unit().conformant(&radians) {
109                return Err(ImageError::InvalidMetadata(format!(
110                    "beam {label} must be a finite angular quantity"
111                )));
112            }
113        }
114        if major.value() < 0.0 || minor.value() < 0.0 {
115            return Err(ImageError::InvalidMetadata(
116                "beam axes cannot be negative".into(),
117            ));
118        }
119        let minor_in_major = casa_types::quanta::casacore::convert(&minor, Some(major.unit()))
120            .map_err(|error| ImageError::InvalidMetadata(error.to_string()))?;
121        if major.value() < minor_in_major.value() {
122            return Err(ImageError::InvalidMetadata(
123                "major axis must be greater or equal to minor axis".into(),
124            ));
125        }
126        Ok(Self {
127            major,
128            minor,
129            position_angle,
130        })
131    }
132
133    /// Reconstruct a beam from a casacore quantity record.
134    ///
135    /// # Errors
136    ///
137    /// Returns an error for malformed, non-angular, or invalid beam values.
138    pub fn from_record(record: &RecordValue) -> Result<Self, ImageError> {
139        if record.len() != 3 {
140            return Err(ImageError::InvalidMetadata(
141                "Beam record does not contain 3 fields".into(),
142            ));
143        }
144        let read = |name: &str| -> Result<Quantity, ImageError> {
145            let Some(Value::Record(field)) = record.get(name) else {
146                return Err(ImageError::InvalidMetadata(format!(
147                    "beam is missing {name}"
148                )));
149            };
150            quantity_from_record(field)
151        };
152        Self::new(read("major")?, read("minor")?, read("positionangle")?)
153    }
154
155    /// Return the requested component in its stored unit.
156    #[must_use]
157    pub fn component(&self, component: BeamComponent) -> &Quantity {
158        match component {
159            BeamComponent::Major => &self.major,
160            BeamComponent::Minor => &self.minor,
161            BeamComponent::PositionAngle => &self.position_angle,
162        }
163    }
164
165    /// Return the position angle wrapped to CASA's (-90, 90] degree range,
166    /// retaining its stored unit.
167    ///
168    /// # Errors
169    ///
170    /// Returns an error if the angular unit cannot be converted.
171    pub fn wrapped_position_angle(&self) -> Result<Quantity, ImageError> {
172        wrap_position_angle(&self.position_angle)
173    }
174
175    /// Return a copy with one component replaced.
176    ///
177    /// # Errors
178    ///
179    /// Returns an error if the resulting beam is invalid.
180    pub fn with_component(
181        &self,
182        component: BeamComponent,
183        value: Quantity,
184    ) -> Result<Self, ImageError> {
185        let mut updated = self.clone();
186        match component {
187            BeamComponent::Major => updated.major = value,
188            BeamComponent::Minor => updated.minor = value,
189            BeamComponent::PositionAngle => updated.position_angle = value,
190        }
191        Self::new(updated.major, updated.minor, updated.position_angle)
192    }
193
194    /// Whether either axis is zero, making the restoring beam null.
195    #[must_use]
196    pub fn is_null(&self) -> bool {
197        self.major.value() == 0.0 || self.minor.value() == 0.0
198    }
199
200    /// Serialize the original quantity values and units to a beam record.
201    #[must_use]
202    pub fn to_record(&self) -> RecordValue {
203        let field = |name: &str, quantity: &Quantity| {
204            RecordField::new(
205                name,
206                Value::Record(RecordValue::new(vec![
207                    RecordField::new(
208                        "value",
209                        Value::Scalar(ScalarValue::Float64(quantity.value())),
210                    ),
211                    RecordField::new(
212                        "unit",
213                        Value::Scalar(ScalarValue::String(quantity.unit().name().into())),
214                    ),
215                ])),
216            )
217        };
218        RecordValue::new(vec![
219            field("major", &self.major),
220            field("minor", &self.minor),
221            field("positionangle", &self.position_angle),
222        ])
223    }

226/// A two-dimensional Gaussian restoring beam.
227#[derive(Debug, Clone, Copy, PartialEq)]
228pub struct GaussianBeam {
229    /// Full width at half maximum of the major axis, in radians.
230    pub major: f64,
231    /// Full width at half maximum of the minor axis, in radians.
232    pub minor: f64,
233    /// Position angle of the major axis, in radians.
234    pub position_angle: f64,
235}
236
237impl GaussianBeam {
238    /// Creates a new Gaussian beam.
239    pub fn new(major: f64, minor: f64, position_angle: f64) -> Self {
240        Self {
241            major,
242            minor,
243            position_angle,
244        }
245    }
246
247    /// Returns `true` if the beam is null.
248    pub fn is_null(&self) -> bool {
249        self.major == 0.0 || self.minor == 0.0
250    }
251
252    /// Returns the beam area in steradians.
253    pub fn area(&self) -> f64 {
254        std::f64::consts::PI / (4.0 * 2.0_f64.ln()) * self.major * self.minor
255    }
256
257    /// Returns the major axis converted to the requested angular unit.
258    pub fn major_in(&self, unit: &str) -> Result<f64, ImageError> {
259        angle_value_in(self.major, unit)
260    }
261
262    /// Returns the minor axis converted to the requested angular unit.
263    pub fn minor_in(&self, unit: &str) -> Result<f64, ImageError> {
264        angle_value_in(self.minor, unit)
265    }
266
267    /// Returns the position angle converted to the requested angular unit.
268    pub fn position_angle_in(&self, unit: &str) -> Result<f64, ImageError> {
269        angle_value_in(self.position_angle, unit)
270    }
271
272    /// Serializes the beam to the casacore quantity-record representation.
273    pub fn to_record(&self) -> RecordValue {
274        fn quantity_record(value: f64, unit: &str) -> RecordValue {
275            RecordValue::new(vec![
276                RecordField::new("value", Value::Scalar(ScalarValue::Float64(value))),
277                RecordField::new("unit", Value::Scalar(ScalarValue::String(unit.into()))),
278            ])
279        }
280
281        // CASA image products conventionally persist fitted beam widths in
282        // arcseconds and position angles in degrees. Keep radians as the
283        // in-memory contract while matching that interoperable record form.
284        let radians_to_degrees = 180.0 / std::f64::consts::PI;
285        let radians_to_arcseconds = 3_600.0 * radians_to_degrees;
286
287        RecordValue::new(vec![
288            RecordField::new(
289                "major",
290                Value::Record(quantity_record(
291                    self.major * radians_to_arcseconds,
292                    "arcsec",
293                )),
294            ),
295            RecordField::new(
296                "minor",
297                Value::Record(quantity_record(
298                    self.minor * radians_to_arcseconds,
299                    "arcsec",
300                )),
301            ),
302            RecordField::new(
303                "positionangle",
304                Value::Record(quantity_record(
305                    self.position_angle * radians_to_degrees,
306                    "deg",
307                )),
308            ),
309        ])
310    }
311
312    /// Deserializes a beam from a casacore quantity record.
313    pub fn from_record(rec: &RecordValue) -> Result<Self, ImageError> {
314        let stored = StoredGaussianBeam::from_record(rec)?;
315        let radians = Unit::new("rad").expect("built-in radians unit must parse");
316        let convert = |component| {
317            stored
318                .component(component)
319                .get_value_in(&radians)
320                .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
321        };
322        let major = convert(BeamComponent::Major)?;
323        let minor = convert(BeamComponent::Minor)?;
324        let position_angle = convert(BeamComponent::PositionAngle)?;
325        if (major == 0.0 && stored.component(BeamComponent::Major).value() != 0.0)
326            || (minor == 0.0 && stored.component(BeamComponent::Minor).value() != 0.0)
327        {
328            return Err(ImageError::InvalidMetadata(
329                "beam axes underflow in the shared radian representation (#977)".into(),
330            ));
331        }
332        if !major.is_finite() || !minor.is_finite() || !position_angle.is_finite() {
333            return Err(ImageError::InvalidMetadata(
334                "invalid restoring beam values".into(),
335            ));
336        }
337        // Casacore compares the original units. Conversion of physically equal
338        // axes to radians can reverse their order by one rounding bit.
339        Ok(Self {
340            major: major.max(minor),
341            minor,
342            position_angle,
343        })
344    }
345
346    /// Computes the Gaussian beam that would need to convolve `other` to
347    /// produce `self`, following casacore `GaussianDeconvolver::deconvolve()`.
348    ///
349    /// Returns `Ok(None)` when the two beams are effectively the same size.
350    pub fn deconvolving_beam(self, other: Self) -> Result<Option<Self>, ImageError> {
351        if self.is_null() || other.is_null() {
352            return Err(ImageError::InvalidMetadata(
353                "cannot deconvolve null beams".to_string(),
354            ));
355        }
356        deconvolving_gaussian(numeric_beam(self), numeric_beam(other))
357            .map(|beam| beam.map(gaussian_beam))
358            .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
359    }
360}
361
362fn angle_value_in(radians: f64, unit: &str) -> Result<f64, ImageError> {
363    let quantity =
364        Quantity::new(radians, "rad").expect("built-in radians quantity must always parse");
365    let target = Unit::new(unit)
366        .map_err(|err| ImageError::InvalidMetadata(format!("invalid unit '{unit}': {err}")))?;
367    quantity.get_value_in(&target).map_err(|err| {
368        ImageError::InvalidMetadata(format!(
369            "cannot convert beam angle from rad to '{unit}': {err}"
370        ))
371    })
372}

384/// Single-beam or per-plane beam metadata corresponding to C++ `ImageBeamSet`.
385#[derive(Debug, Clone, PartialEq)]
386pub struct ImageBeamSet {
387    beams: Vec<Vec<GaussianBeam>>,
388}
389
390impl ImageBeamSet {
391    /// Creates an empty beam set.
392    pub fn empty() -> Self {
393        Self { beams: Vec::new() }
394    }
395
396    /// Creates a beam set with a single global beam.
397    pub fn new(beam: GaussianBeam) -> Self {
398        Self {
399            beams: vec![vec![beam]],
400        }
401    }
402
403    /// Creates a beam set from a channel × stokes grid.
404    pub fn from_grid(beams: Vec<Vec<GaussianBeam>>) -> Self {
405        if beams.is_empty() {
406            return Self::empty();
407        }
408        let nstokes = beams[0].len();
409        assert!(
410            beams.iter().all(|row| row.len() == nstokes),
411            "all beam rows must have the same stokes length"
412        );
413        Self { beams }
414    }
415
416    /// Creates a uniform beam set of the requested size.
417    pub fn with_shape(nchan: usize, nstokes: usize, beam: GaussianBeam) -> Self {
418        let nchan = nchan.max(1);
419        let nstokes = nstokes.max(1);
420        Self {
421            beams: vec![vec![beam; nstokes]; nchan],
422        }
423    }
424
425    /// Returns `true` if there are no beams.
426    pub fn is_empty(&self) -> bool {
427        self.beams.is_empty()
428    }
429
430    /// Returns `true` if exactly one beam applies globally.
431    pub fn is_single(&self) -> bool {
432        self.nelements() == 1
433    }
434
435    /// Alias for [`Self::is_single`] matching C++ `hasSingleBeam()`.
436    pub fn has_single_beam(&self) -> bool {
437        self.is_single()
438    }
439
440    /// Returns `true` if multiple per-plane beams are present.
441    pub fn is_multi(&self) -> bool {
442        self.nelements() > 1
443    }
444
445    /// Alias for [`Self::is_multi`] matching C++ `hasMultiBeam()`.
446    pub fn has_multi_beam(&self) -> bool {
447        self.is_multi()
448    }
449
450    /// Returns the number of beam elements.
451    pub fn nelements(&self) -> usize {
452        self.beams.iter().map(Vec::len).sum()
453    }
454
455    /// Alias for [`Self::nelements`] matching C++ `size()`.
456    pub fn size(&self) -> usize {
457        self.nelements()
458    }

460    /// Returns the beam-grid shape as `(nchan, nstokes)`.
461    pub fn shape(&self) -> (usize, usize) {
462        if self.is_empty() {
463            (0, 0)
464        } else {
465            (self.beams.len(), self.beams[0].len())

467    }

471        self.shape().0

474    /// Alias for [`Self::n_channels`] matching C++ `nchan()`.
475    pub fn nchan(&self) -> usize {
476        self.n_channels()
477    }

479    /// Returns the number of Stokes planes in the beam grid.
480    pub fn n_stokes(&self) -> usize {
481        self.shape().1

484    /// Alias for [`Self::n_stokes`] matching C++ `nstokes()`.

486        self.n_stokes()
487    }

489    /// Returns the single global beam, if present.
490    pub fn single_beam(&self) -> Option<GaussianBeam> {
491        self.is_single().then(|| self.beams[0][0])
492    }
493
494    /// Returns the single global beam, matching C++ `getBeam()`.
495    pub fn get_beam(&self) -> Result<&GaussianBeam, ImageError> {
496        if self.is_single() {
497            Ok(&self.beams[0][0])
498        } else {
499            Err(ImageError::InvalidMetadata(
500                "beam set does not contain exactly one beam".to_string(),
501            ))
502        }
503    }
504
505    /// Returns the beam for the given channel and stokes indices.
506    ///
507    /// Axis length 1 expands to all indices, matching casacore semantics.
508    pub fn beam(&self, chan: usize, stokes: usize) -> &GaussianBeam {
509        assert!(!self.is_empty(), "beam set is empty");
510        let c = if self.n_channels() == 1 { 0 } else { chan };
511        let s = if self.n_stokes() == 1 { 0 } else { stokes };
512        &self.beams[c][s]
513    }
514
515    /// Sets all beams to the same value, collapsing to a single global beam.
516    pub fn set_all(&mut self, beam: GaussianBeam) {
517        self.beams = vec![vec![beam]];
518    }
519
520    /// Resizes the beam set, preserving existing values where possible.
521    pub fn resize(&mut self, nchan: usize, nstokes: usize) {
522        let fill = self
523            .single_beam()
524            .or_else(|| self.min_area_beam().copied())
525            .unwrap_or_default();
526        let nchan = nchan.max(1);
527        let nstokes = nstokes.max(1);
528        let mut resized = vec![vec![fill; nstokes]; nchan];
529        for (chan, row) in resized.iter_mut().enumerate() {
530            for (stokes, beam) in row.iter_mut().enumerate() {
531                if !self.is_empty() && chan < self.n_channels() && stokes < self.n_stokes() {
532                    *beam = self.beams[chan][stokes];
533                }
534            }
535        }
536        self.beams = resized;
537    }
538
539    /// Sets the beam at the given location.
540    ///
541    /// Passing `None` for either axis applies the change to all channels or
542    /// all Stokes planes respectively. Passing `None` for both collapses the
543    /// set to a single global beam.
544    pub fn set_beam(
545        &mut self,
546        chan: Option<usize>,
547        stokes: Option<usize>,
548        beam: GaussianBeam,
549    ) -> Result<(), ImageError> {
550        if chan.is_none() && stokes.is_none() {
551            self.set_all(beam);
552            return Ok(());
553        }
554        if self.is_empty() {
555            return Err(ImageError::InvalidMetadata(
556                "cannot set a beam on an empty beam set".to_string(),
557            ));
558        }
559        let chan_range: Vec<usize> = match chan {
560            Some(c) => vec![c],
561            None => (0..self.n_channels()).collect(),
562        };
563        let stokes_range: Vec<usize> = match stokes {
564            Some(s) => vec![s],
565            None => (0..self.n_stokes()).collect(),
566        };
567        for c in chan_range {
568            for s in &stokes_range {
569                if c >= self.n_channels() || *s >= self.n_stokes() {
570                    return Err(ImageError::InvalidMetadata(
571                        "beam index out of range".to_string(),
572                    ));
573                }
574                self.beams[c][*s] = beam;
575            }
576        }
577        Ok(())
578    }

580    /// Returns a subset of the beam grid using explicit channel and stokes selections.
581    pub fn subset(&self, channels: &[usize], stokes: &[usize]) -> Result<Self, ImageError> {
582        if self.is_empty() {
583            return Ok(Self::empty());
584        }
585        let mut rows = Vec::with_capacity(channels.len());
586        for &chan in channels {
587            let mut row = Vec::with_capacity(stokes.len());
588            for &stok in stokes {
589                row.push(*self.beam(chan, stok));
590            }
591            rows.push(row);
592        }
593        Ok(Self::from_grid(rows))
594    }
595
596    /// Returns `true` if the two beam sets are equal after singleton expansion.
597    pub fn equivalent(&self, other: &Self) -> bool {
598        if self.is_empty() || other.is_empty() {
599            return self.is_empty() && other.is_empty();
600        }
601        let nchan = self.n_channels().max(other.n_channels());
602        let nstokes = self.n_stokes().max(other.n_stokes());
603        for chan in 0..nchan {
604            for stokes in 0..nstokes {
605                if self.beam(chan, stokes) != other.beam(chan, stokes) {
606                    return false;
607                }
608            }
609        }
610        true
611    }

613    /// Returns the beam with the minimum area, if any.
614    pub fn min_area_beam(&self) -> Option<&GaussianBeam> {
615        self.iter_beams()
616            .min_by(|a, b| a.area().partial_cmp(&b.area()).unwrap())
617    }

619    /// Returns the beam with the maximum area, if any.
620    pub fn max_area_beam(&self) -> Option<&GaussianBeam> {
621        self.iter_beams()
622            .max_by(|a, b| a.area().partial_cmp(&b.area()).unwrap())
623    }

625    /// Returns the beam with the median area, if any.
626    pub fn median_area_beam(&self) -> Option<GaussianBeam> {
627        let mut beams: Vec<GaussianBeam> = self.iter_beams().copied().collect();
628        if beams.is_empty() {
629            return None;
630        }
631        beams.sort_by(|a, b| a.area().partial_cmp(&b.area()).unwrap());
632        Some(beams[beams.len() / 2])
633    }

635    /// Returns CASA's minimum-area common enclosing beam for this beam set.
636    ///
637    /// This ports the `CasaImageBeamSet::getCommonBeam()` algorithm used by
638    /// CASA image analysis and synthesis restoration when
639    /// `restoringbeam='common'`.
640    pub fn common_beam(&self) -> Result<GaussianBeam, ImageError> {
641        if self.is_empty() {
642            return Err(ImageError::InvalidMetadata(
643                "cannot determine a common beam for an empty beam set".to_string(),
644            ));
645        }
646        // CASA traverses the channel-by-Stokes matrix with channel varying
647        // fastest, while the stored grid groups Stokes by channel.
648        let non_null: Vec<GaussianBeam> = (0..self.n_stokes())
649            .flat_map(|stokes| {
650                (0..self.n_channels()).map(move |channel| self.beams[channel][stokes])
651            })
652            .filter(|beam| !beam.is_null())
653            .collect();
654        if non_null.is_empty() {
655            return Err(ImageError::InvalidMetadata(
656                "cannot determine a common beam because all beams are null".to_string(),
657            ));
658        }
659        if non_null.iter().all(|beam| *beam == non_null[0]) {
660            return Ok(non_null[0]);
661        }
662        common_enclosing_gaussian(
663            &non_null.into_iter().map(numeric_beam).collect::<Vec<_>>(),
664            1.0,
665        )
666        .map(gaussian_beam)
667        .map_err(|error| ImageError::InvalidMetadata(error.to_string()))
668    }
669
670    fn iter_beams(&self) -> impl Iterator<Item = &GaussianBeam> {
671        self.beams.iter().flat_map(|row| row.iter())
672    }

674    /// Serializes the beam set using the casacore `ImageBeamSet::toRecord()` layout.
675    pub fn to_record(&self) -> RecordValue {
676        let mut rec = RecordValue::default();
677        rec.upsert(
678            "nChannels",
679            Value::Scalar(ScalarValue::Int32(self.n_channels() as i32)),
680        );
681        rec.upsert(
682            "nStokes",
683            Value::Scalar(ScalarValue::Int32(self.n_stokes() as i32)),
684        );
685        let mut count = 0usize;
686        for stokes in 0..self.n_stokes() {
687            for chan in 0..self.n_channels() {
688                rec.upsert(
689                    format!("*{count}"),
690                    Value::Record(self.beam(chan, stokes).to_record()),
691                );
692                count += 1;
693            }
694        }
695        rec
696    }
697
698    /// Deserializes a beam set from the casacore `ImageBeamSet::fromRecord()` layout.
699    pub fn from_record(rec: &RecordValue) -> Result<Self, ImageError> {
700        let nchan = read_record_usize(rec, "nChannels")
701            .ok_or_else(|| ImageError::InvalidMetadata("no valid nChannels field found".into()))?
702            .max(1);
703        let nstokes = read_record_usize(rec, "nStokes")
704            .ok_or_else(|| ImageError::InvalidMetadata("no valid nStokes field found".into()))?
705            .max(1);
706
707        let mut rows = vec![vec![GaussianBeam::default(); nstokes]; nchan];
708        for count in 0..(nchan * nstokes) {
709            let key = format!("*{count}");
710            let beam = if let Some(Value::Record(beam_rec)) = rec.get(&key) {
711                GaussianBeam::from_record(beam_rec)?
712            } else if let Some((chan, stokes)) = count_to_pair(nchan, nstokes, count) {
713                let legacy_key = format!("*{chan}_{stokes}");
714                match rec.get(&legacy_key) {
715                    Some(Value::Record(beam_rec)) => GaussianBeam::from_record(beam_rec)?,
716                    _ => {
717                        return Err(ImageError::InvalidMetadata(format!(
718                            "beam set: missing beam record '{key}'"
719                        )));
720                    }
721                }
722            } else {
723                return Err(ImageError::InvalidMetadata(
724                    "beam record index overflow".to_string(),
725                ));
726            };
727            let chan = count % nchan;
728            let stokes = count / nchan;
729            rows[chan][stokes] = beam;
730        }
731        Ok(Self::from_grid(rows))
732    }
733}
734
735fn read_record_usize(rec: &RecordValue, key: &str) -> Option<usize> {
736    match rec.get(key) {
737        Some(Value::Scalar(ScalarValue::UInt8(value))) => Some(usize::from(*value)),
738        Some(Value::Scalar(ScalarValue::UInt16(value))) => Some(usize::from(*value)),
739        Some(Value::Scalar(ScalarValue::UInt32(value))) => usize::try_from(*value).ok(),
740        Some(Value::Scalar(ScalarValue::Int16(value))) => usize::try_from(*value).ok(),
741        Some(Value::Scalar(ScalarValue::Int32(value))) => usize::try_from(*value).ok(),
742        Some(Value::Scalar(ScalarValue::Int64(value))) => usize::try_from(*value).ok(),
743        _ => None,
744    }
