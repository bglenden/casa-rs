20fn supplied<'py>(value: &Bound<'py, PyAny>) -> PyResult<Option<Bound<'py, PyAny>>> {
21    Ok(Some(value.clone()))
22}

24fn plane_index(value: Option<Bound<'_, PyAny>>, name: &str) -> PyResult<(i64, bool)> {
25    let Some(value) = value else {
26        return Ok((-1, false));
27    };
28    if value.is_instance_of::<PyFloat>()
29        || value.is_instance_of::<PyString>()
30        || value.is_instance_of::<PyBytes>()
31        || value.is_instance_of::<PyList>()
32        || value.is_instance_of::<PyDict>()
33    {
34        return Err(PyTypeError::new_err(format!(
35            "argument _{name} must be an integer"
36        )));
37    }
38    // The original binding invokes image metadata validation even when
39    // conversion sets a Python error and returns the sentinel -1.
40    Ok(value
41        .extract::<i64>()
42        .map_or((-1, true), |index| (i64::from(index as i32), false)))
43}

45fn replace_shape(output: &mut CasaValue, fit: GaussianBeam) -> PyResult<()> {
46    fn field<'a>(value: &'a mut CasaValue, name: &str) -> &'a mut CasaValue {
47        let CasaValue::Record(fields) = value else {
48            unreachable!("component record")
49        };
50        &mut fields
51            .iter_mut()
52            .find(|(key, _)| key == name)
53            .expect("serialized component field")
54            .1
55    }
56    let shape = field(output, "shape");
57    let angle = fit.position_angle_in("deg").map_err(raised)?;
58    for (key, value) in [
59        (
60            "majoraxis",
61            CasaValue::quantity(fit.major_in("arcsec").map_err(raised)?, "arcsec"),
62        ),
63        (
64            "minoraxis",
65            CasaValue::quantity(fit.minor_in("arcsec").map_err(raised)?, "arcsec"),
66        ),
67        (
68            "positionangle",
69            CasaValue::quantity(if angle < 0.0 { angle + 180.0 } else { angle }, "deg"),
70        ),
71        ("majoraxiserror", CasaValue::quantity(0.0, "arcmin")),
72        ("minoraxiserror", CasaValue::quantity(0.0, "arcmin")),
73        ("positionangleerror", CasaValue::quantity(0.0, "deg")),
74    ] {
75        *field(shape, key) = value;
76    }
77    let error = field(field(shape, "direction"), "error");
78    *field(error, "latitude") = CasaValue::quantity(0.0, "rad");
79    *field(error, "longitude") = CasaValue::quantity(0.0, "rad");
80    Ok(())
81}

83fn source_beam(axes: &[Quantity; 3], rad: &Unit) -> PyResult<GaussianBeam> {
84    let [major, minor, pa] = axes;
85    let radians = |quantity| casacore::get_conformant_value(quantity, rad).map_err(raised);
86    let major_rad = radians(major)?;
87    let minor_rad = radians(minor)?;
88    let angle_rad = radians(pa)?;
89    if angle_rad.abs() > std::f64::consts::TAU {
90        return Err(raised(
91            "Gaussian2DParam<T>::setPA(const T &pa) - PA must be in radians and between -2pi and 2pi",
92        ));
93    }
94    // GaussianShape starts at one arcminute and stores an axial ratio.
95    // setWidth narrows the minor axis first for a smaller source major axis.
96    // Preserve both ratio operations and TwoSidedShape's unit round trip:
97    // a rounding bit can change an unresolved source into a resolved fit.
98    let default_width = Quantity::new(1.0, "arcmin").expect("built-in angular unit");
99    let default_width = radians(&default_width)?;
100    let minor_rad = if major_rad <= default_width {
101        default_width * (minor_rad / default_width)
102    } else {
103        minor_rad
104    };
105    let minor_rad = major_rad * (minor_rad / major_rad);
106    let in_record_units = |value, unit| {
107        casacore::convert(&Quantity::with_unit(value, rad.clone()), Some(unit)).map_err(raised)
108    };
109    let angle_rad = angle_rad % std::f64::consts::PI;
110    let angle_rad = if angle_rad < 0.0 {
111        angle_rad + std::f64::consts::PI
112    } else {
113        angle_rad
114    };
115    let angle = in_record_units(angle_rad, pa.unit())?;
116    let angle = casa_images::beam::wrap_position_angle(&angle).map_err(raised)?;
117    Ok(GaussianBeam::new(
118        radians(&in_record_units(major_rad, major.unit())?)?,
119        radians(&in_record_units(minor_rad, minor.unit())?)?,
120        radians(&angle)?,
121    ))
122}

124fn deconvolved_record(
125    image: &ImageHandle,
126    beam_view: &BeamView,
127    input: &RecordValue,
128    channel: i64,
129    polarization: i64,
130    logs: &mut Vec<LogLine>,
131) -> PyResult<CasaValue> {
132    image.check_single_precision("deconvolvecomponentlist")?;
133    let components = image_component_records(input, logs)?;
134    let missing_beam = || raised("Exception: This image does not have a restoring beam.");
135    if beam_view.beam_set().map_err(raised)?.is_empty() {
136        return Err(missing_beam());
137    }
138    if CoordinateSystem::first_direction_in_record(&image.coordinate_record())
139        .map_err(raised)?
140        .is_none()
141    {
142        return Err(raised(
143            "Exception: This image does not contain a direction coordinate.",
144        ));
145    }
146    let stored_beam = beam_for_units(beam_view, channel, polarization)?.ok_or_else(missing_beam)?;
147    let rad = Unit::new("rad").expect("built-in radians unit");
148    let radians =
149        |quantity: &Quantity| casacore::get_conformant_value(quantity, &rad).map_err(raised);
150    let beam = GaussianBeam::new(
151        radians(stored_beam.component(BeamComponent::Major))?,
152        radians(stored_beam.component(BeamComponent::Minor))?,
153        radians(&stored_beam.wrapped_position_angle().map_err(raised)?)?,
154    );
155    let mut fields = vec![("nelements".into(), CasaValue::Int(components.len() as i64))];
156    for (index, component) in components.enumerate() {
157        let mut component = component?;
158        match component.shape {
159            ShapeType::Point => {}
160            ShapeType::Gaussian => {
161                let source = source_beam(
162                    component
163                        .gaussian_axes
164                        .as_ref()
165                        .expect("Gaussian record axes"),
166                    &rad,
167                )?;
168                let (fit, _) = source.fitted_deconvolution(beam).ok_or_else(|| {
169                    raised("Source may be only (slightly) resolved in one direction")
170                })?;
171                if fit.minor == 0.0 {
172                    return Err(raised(
173                        "Gaussian2DParam<T>::setAxialRatio(const T &axialRatio) - axialRatio must be between (0,1]",
174                    ));
175                }
176                replace_shape(&mut component.value, fit)?;
177            }
178            other => {
179                return Err(raised(format!(
180                    "Cannot deconvolve components of type {}",
181                    other.casa_name()
182                )));
183            }
184        }
185        fields.push((format!("component{index}"), component.value));
186    }
187    Ok(CasaValue::record(fields))
188}

190#[pymethods]
191impl PyImageTool {
192    /// Remove the selected restoring beam from the Gaussian components of a CASA list.
193    #[pyo3(signature = (complist, channel = None, polarization = None), text_signature = "($self, complist, channel=-1, polarization=-1)")]
194    fn deconvolvecomponentlist(
195        &self,
196        py: Python<'_>,
197        complist: &Bound<'_, PyAny>,
198        #[pyo3(from_py_with = supplied)] channel: Option<Bound<'_, PyAny>>,
199        #[pyo3(from_py_with = supplied)] polarization: Option<Bound<'_, PyAny>>,
200    ) -> PyResult<Py<PyAny>> {
201        let dictionary = complist
202            .cast::<PyDict>()
203            .map_err(|_| PyTypeError::new_err("_complist is not a dictionary"))?;
204        let record = casa_record(py, Some(dictionary))?;
205        let (channel, pending_channel) = plane_index(channel, "channel")?;
206        let (polarization, pending_polarization) = plane_index(polarization, "polarization")?;
207        let mut logs = Vec::new();
208        let result = py.detach(|| {
209            let state = self.state();
210            let Some(attachment) = &state.image else {
211                return Ok(CasaValue::empty_record());
212            };
213            deconvolved_record(
214                &attachment.lock(),
215                &attachment.beam_view(),
216                &record,
217                channel,
218                polarization,
219                &mut logs,
220            )
221        });
222        emit(py, "casars.tools.image", &logs)?;
223        let result = result?;
224        if pending_channel || pending_polarization {
225            return Err(PySystemError::new_err(
226                "<built-in function image_deconvolvecomponentlist> returned a result with an exception set",
227            ));
228        }
229        result.into_python(py)
230    }
