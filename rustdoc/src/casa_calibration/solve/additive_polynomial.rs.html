27struct SelectedFrequencies {
28    by_description: BTreeMap<i32, Vec<f64>>,
29    by_window: BTreeMap<i32, [f64; 2]>,
30}
31
32impl SelectedFrequencies {
33    fn read(ms: &MeasurementSet, rows: &[SelectedSolveRow]) -> Result<Self, GainSolveError> {
34        use std::collections::btree_map::Entry;
35
36        let table = ms.spectral_window().map_err(ms_error)?;
37        let mut by_description = BTreeMap::new();
38        let mut by_window = BTreeMap::new();
39        for row in rows {
40            if let Entry::Vacant(entry) = by_description.entry(row.data_desc_id) {
41                let id = row.data_spw_id as usize;
42                let frequencies = table.chan_freq(id).map_err(ms_error)?;
43                if usize::try_from(table.num_chan(id).map_err(ms_error)?).ok()
44                    != Some(frequencies.len())
45                {
46                    return Err(frequency_shape(row.data_spw_id, frequencies.len()));
47                }
48                if let Entry::Vacant(domain) = by_window.entry(row.data_spw_id) {
49                    let selected =
50                        selected_channels(row.channel_selection.as_ref(), frequencies.len())?;
51                    if selected.is_empty() {
52                        return Err(frequency_shape(row.data_spw_id, frequencies.len()));
53                    }
54                    domain.insert(bounds(selected.iter().map(|&channel| frequencies[channel])));
55                }
56                entry.insert(frequencies);
57            }
58        }
59        Ok(Self {
60            by_description,
61            by_window,
62        })
63    }
64}
65
66fn bounds(frequencies: impl Iterator<Item = f64>) -> [f64; 2] {
67    frequencies.fold(
68        [f64::INFINITY, f64::NEG_INFINITY],
69        |[low, high], frequency| [low.min(frequency), high.max(frequency)],
70    )
71}
72
73fn frequency_shape(window: i32, channels: usize) -> GainSolveError {
74    GainSolveError::UnsupportedParameterShape {
75        path: format!("SPECTRAL_WINDOW {window} additive fit frequency grid"),
76        shape: vec![channels],
77    }
78}

80pub(crate) fn solve(
81    ms: &MeasurementSet,
82    request: &GainSolveRequest,
83    fit_order: i32,
84    mode: CalibrationOutputMode,
85    legacy: bool,
86) -> Result<GainSolveReport, GainSolveError> {
87    if !legacy {
88        return Err(GainSolveError::Baseline(
89            "A Mueller nPar() called before being set.".into(),
90        ));
91    }
92    let terms = usize::try_from(fit_order).map_err(|_| {
93        GainSolveError::Baseline("The continuum fitter must first be initialized.".into())
94    })? + 1;
95    let context = prepare_calibration_solve(
96        ms,
97        &request.selection,
98        &RefAntSelector::Unspecified,
99        &request.prior_calibration_tables,
100        request.parang,
101        true,
102        PriorCalibrationMode::Data,
103    )?;
104    let frequencies = SelectedFrequencies::read(ms, &context.rows)?;
105    // Keep the frequency basis fixed across all intervals, including those
106    // where an entire input window has no usable measurements.
107    let combined_domain = request
108        .combine
109        .spectral_windows
110        .then(|| bounds(frequencies.by_window.values().flatten().copied()));
111    let groups = build_solve_groups(
112        ms,
113        &context.rows,
114        context.preapplied_rows.as_ref(),
115        &BTreeMap::new(),
116        SolveGroupOptions {
117            input_rows: SolveInputRows::LegacyAdditivePolynomial,
118            gain_type: GainType::G,
119            model_source: GainSolveModelSource::PointSource,
120            point_stokes: [1., 0., 0., 0.],
121            rotate_model: false,
122            solve_interval: request.solve_interval,
123            combine: request.combine,
124            preaverage_max_span: None,
125            correlation_dependent_flags: false,
126            robust_seed: false,
127            legacy_g_receptors: false,
128            channelized: true,
129            polynomial_bandpass: false,
130        },
131    )?;
132
133    let mut solved = SolvedGains {
134        rows: Vec::new(),
135        refant_id: -1,
136        attempted_spw_ids: BTreeSet::new(),
137        channel_selections: BTreeMap::new(),
138        selected_data: context.selected_data,
139        solve_stats: Default::default(),
140    };
141    let mut grid = TableGrid {
142        terms,
143        windows: BTreeMap::new(),
144    };
145    for group in groups.into_values().filter(|group| group.attempted) {
146        let gather = group.additive_polynomial.as_ref().expect("additive gather");
147        let Some(&correlations) = group.baseline_correlation_counts.first() else {
148            continue;
149        };
150        if group.baseline_correlation_counts.len() != 1 {
151            return Err(GainSolveError::Baseline(
152                "incompatible correlation counts for additive polynomial fitting".into(),
153            ));
154        }
155        let window = grid.windows.entry(group.spw_id).or_insert_with(|| Window {
156            frequency_domain: combined_domain
157                .unwrap_or_else(|| frequencies.by_window[&group.spw_id]),
158            correlations,
159        });
160        for (&dd, layout) in &gather.windows {
161            let axis = &frequencies.by_description[&dd];
162            if layout.channels.iter().any(|&channel| channel >= axis.len()) {
163                return Err(frequency_shape(group.spw_id, axis.len()));
164            }
165        }
166        if window.correlations != correlations {
167            return Err(GainSolveError::Baseline(
168                "incompatible additive coefficient shapes in one spectral window".into(),
169            ));
170        }
171        solved.attempted_spw_ids.insert(group.spw_id);
172        solved.rows.extend(gather.solutions(
173            &group,
174            &context.available_antennas,
175            &frequencies.by_description,
176            window.frequency_domain,
177            terms,
178            request.solve_mode,
179        ));
180    }
181    if request.normalize_average_amplitude {
182        super::baseline::normalize(&mut solved.rows, false, request.normalization_type);
183    }
184    super::writer::write_additive_polynomial_caltable(ms, request, &solved, mode, &grid)
185}

188    GainSolveError::Baseline(error.to_string())
