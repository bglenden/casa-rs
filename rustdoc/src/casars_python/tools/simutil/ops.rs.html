16/// A `simutil` refusal as the `Exception` CASA's `simutil.msg(...,
17/// priority='error')` raises: a `RuntimeError` with its message.
18pub(super) fn raised(error: SimutilError) -> PyErr {
19    PyRuntimeError::new_err(error.to_string())
20}

22/// A quantity as `qa.quantity` reads a string or a record.
23pub(super) fn quantity(value: &Variant) -> PyResult<Quantity> {
24    Ok(casa_quantity(value)?)
25}

27/// A direction string or a list of them (CASA's `self.direction`), kept
28/// apart: CASA splits a list, even of one direction, at its mean.
29pub(super) fn directions(value: &Bound<'_, PyAny>) -> PyResult<Directions> {
30    if let Ok(text) = value.cast::<PyString>() {
31        return Ok(Directions::One(text.to_str()?.to_string()));
32    }
33    let directions = value.extract::<StringList>()?.0;
34    if directions.is_empty() {
35        return Err(PyRuntimeError::new_err(
36            "can't interpret '[]' as a direction",
37        ));
38    }
39    Ok(Directions::List(directions))
40}

42/// The directions of a string (one) or a list, for the methods that take
43/// a list.
44pub(super) fn listed(directions: Directions) -> Vec<String> {
45    match directions {
46        Directions::One(direction) => vec![direction],
47        Directions::List(directions) => directions,
48    }
49}

51/// `calc_pointings2`'s `size`: one quantity (a string or a record) for
52/// both axes, or a sequence of them, the first two used (one repeated).
53pub(super) fn sizes(value: &Bound<'_, PyAny>) -> PyResult<[Quantity; 2]> {
54    let one = |item: &Bound<'_, PyAny>| quantity(&item.extract::<Variant>()?);
55    if value.is_instance_of::<PyString>() || value.is_instance_of::<PyDict>() {
56        let size = one(value)?;
57        return Ok([size.clone(), size]);
58    }
59    let items: Vec<Bound<'_, PyAny>> = value
60        .try_iter()
61        .map_err(|_| {
62            PyTypeError::new_err(format!(
63                "can't interpret '{}' as a CASA quantity",
64                value.repr().map_or_else(|_| "?".into(), |r| r.to_string())
65            ))
66        })?
67        .collect::<PyResult<_>>()?;
68    match items.as_slice() {
69        [] => Err(PyRuntimeError::new_err(
70            "can't interpret '[]' as a CASA quantity",
71        )),
72        [only] => {
73            let size = one(only)?;
74            Ok([size.clone(), size])
75        }
76        [x, y, ..] => Ok([one(x)?, one(y)?]),
77    }
78}

80/// `average_direction`'s and `median_direction`'s `(direction, offsets)`:
81/// the offsets a NumPy array of shape (2, n), degrees.
82pub(super) fn centred(centred: CentredDirections) -> CasaValue {
83    let n = centred.offsets.len();
84    CasaValue::Tuple(vec![
85        CasaValue::Text(centred.centre),
86        CasaValue::Floats {
87            shape: vec![2, n],
88            values: centred.offsets.into_iter().flatten().collect(),
89        },
90    ])
91}

93/// A sky-model refusal: a missing or unwritable file as `OSError`, the rest
94/// (CASA's `simutil.msg(..., priority='error')`) as `RuntimeError`.
95pub(super) fn sky_model_raised(error: SkyModelError) -> PyErr {
96    match error {
97        SkyModelError::Io { .. } => PyOSError::new_err(error.to_string()),
98        other => PyRuntimeError::new_err(other.to_string()),
99    }
100}

102/// `modifymodel`'s tuple: the cells and sizes as quantity records, the
103/// centre frequency and width as the strings given or as records.
104pub(super) fn model_summary(summary: ModelSummary) -> CasaValue {
105    let quantity = |value: &Quantity| CasaValue::quantity(value.value(), value.unit().name());
106    let given_or_model = |value: &ModelValue| match value {
107        ModelValue::Given(text) => CasaValue::Text(text.clone()),
108        ModelValue::Model(value) => quantity(value),
109    };
110    CasaValue::Tuple(vec![
111        CasaValue::Text(summary.refdir),
112        CasaValue::List(summary.cell.iter().map(quantity).collect()),
113        CasaValue::List(summary.size.iter().map(quantity).collect()),
114        CasaValue::Int(summary.nchan as i64),
115        given_or_model(&summary.specrefval),
116        CasaValue::Float(summary.specrefpix),
117        given_or_model(&summary.width),
118        CasaValue::Text(summary.stokes),
119    ])
120}

122/// A list of strings.

127/// Whether a value is a quantity as `simutil.isquantity` tests one: a
128/// non-empty string or record that casacore reads.
129pub(super) fn is_quantity(value: &Bound<'_, PyAny>) -> bool {
130    if value.len().map_or(true, |len| len == 0) {
131        return false;
132    }
133    value.extract::<Variant>().is_ok_and(|variant| {
134        matches!(variant, Variant::Text(_) | Variant::Record(_)) && quantity(&variant).is_ok()
135    })
136}

138/// Python's `str()` of each time, as `write_pointings` writes them: a
139/// sequence's items, or the one value.
140pub(super) fn times(time: &Bound<'_, PyAny>) -> PyResult<Vec<String>> {
141    let shown = |item: &Bound<'_, PyAny>| item.str().map(|text| text.to_string());
142    if time.is_instance_of::<PyString>() {
143        return Ok(vec![shown(time)?]);
144    }
145    match time.try_iter() {
146        Ok(items) => items.map(|item| shown(&item?)).collect(),
147        Err(_) => Ok(vec![shown(time)?]),
148    }
149}
