37/// Scientific lookup key for one CASA AWProject convolution-function cell.
38#[derive(Debug, Clone, Copy, PartialEq)]
39pub struct AwConvolutionFunctionKey {
40    /// Spectral-coordinate reference frequency in Hz.
41    pub frequency_hz: f64,
42    /// W-coordinate value represented by this cell, in wavelengths.
43    pub w_value_lambda: f64,
44    /// CASA Mueller-matrix element number.
45    pub mueller_element: i32,
46    /// Parallactic-angle bin in degrees.
47    pub parallactic_angle_deg: f64,
48}
49
50impl AwConvolutionFunctionKey {
51    fn validate(self, path: &Path) -> Result<Self, ImagingError> {
52        if !(self.frequency_hz.is_finite() && self.frequency_hz > 0.0) {
53            return Err(cache_error(path, "frequency must be finite and positive"));
54        }
55        if !self.w_value_lambda.is_finite() {
56            return Err(cache_error(path, "W value must be finite"));
57        }
58        if !self.parallactic_angle_deg.is_finite() {
59            return Err(cache_error(path, "parallactic angle must be finite"));
60        }
61        Ok(self)
62    }
63}

445#[derive(Debug, Clone)]
446pub(crate) struct ExperimentalMappedCfPlane {
447    map: Arc<Mmap>,
448    offset: usize,
449    shape: [usize; 2],
450}
451
452#[cfg(unix)]
453impl AwProjectKernelPixels for ExperimentalMappedCfPlane {
454    fn shape(&self) -> [usize; 2] {
455        self.shape
456    }
457
458    #[inline]
459    fn get_pixel(&self, (x, y): (usize, usize)) -> Option<Complex32> {
460        if x >= self.shape[0] || y >= self.shape[1] {
461            return None;
462        }
463        let index = x.checked_mul(self.shape[1])?.checked_add(y)?;
464        let byte_offset = self
465            .offset
466            .checked_add(index.checked_mul(std::mem::size_of::<Complex32>())?)?;
467        // SAFETY: `open` validates the complete mapped length, the packed
468        // payload starts at an aligned 32-byte boundary, every plane length is
469        // a multiple of `Complex32`, and `Complex32` accepts every bit pattern.
470        Some(unsafe { *self.map.as_ptr().add(byte_offset).cast::<Complex32>() })
471    }
472}

505#[cfg(unix)]
506impl ExperimentalPackedCfStore {
507    fn open(
508        path: PathBuf,
509        entries: &BTreeMap<StableKey, AwConvolutionFunctionEntryMetadata>,
510        identity: &AwConvolutionFunctionCacheIdentity,
511    ) -> Result<Self, ImagingError> {
512        if !cfg!(target_endian = "little") {
513            return Err(cache_error(
514                &path,
515                "experimental packed CF files currently require a little-endian host",
516            ));
517        }
518        let file = File::open(&path).map_err(|error| {
519            cache_error(
520                &path,
521                format!("cannot open experimental packed CF file: {error}"),
522            )
523        })?;
524        // SAFETY: the mapping is read-only and this experimental cache is an
525        // immutable benchmark artifact for the lifetime of the process.
526        let map = unsafe { MmapOptions::new().map(&file) }.map_err(|error| {
527            cache_error(
528                &path,
529                format!("cannot memory-map experimental packed CF file: {error}"),
530            )
531        })?;
532        let header = map
533            .get(..EXPERIMENTAL_PACKED_CF_HEADER_BYTES as usize)
534            .ok_or_else(|| cache_error(&path, "experimental packed CF header is truncated"))?;
535        if header[..16] != EXPERIMENTAL_PACKED_CF_MAGIC {
536            return Err(cache_error(
537                &path,
538                "experimental packed CF magic/version does not match",
539            ));
540        }
541        let fingerprint = u64::from_le_bytes(header[16..24].try_into().expect("fixed slice"));
542        if fingerprint != identity.metadata_fingerprint {
543            return Err(cache_error(
544                &path,
545                format!(
546                    "experimental packed CF fingerprint {fingerprint:016x} does not match source cache {:016x}",
547                    identity.metadata_fingerprint
548                ),
549            ));
550        }
551        let pair_count = u64::from_le_bytes(header[24..32].try_into().expect("fixed slice"));
552        if pair_count != entries.len() as u64 {
553            return Err(cache_error(
554                &path,
555                format!(
556                    "experimental packed CF pair count {pair_count} does not match source cache {}",
557                    entries.len()
558                ),
559            ));
560        }
561
562        let mut packed_entries = BTreeMap::new();
563        let mut cursor = EXPERIMENTAL_PACKED_CF_HEADER_BYTES;
564        for (&stable, metadata) in entries {
565            let imaging_offset = cursor;
566            cursor = cursor
567                .checked_add(experimental_packed_plane_bytes(metadata.imaging.shape)?)
568                .ok_or_else(|| cache_error(&path, "packed imaging offset overflowed"))?;
569            let weight_offset = cursor;
570            cursor = cursor
571                .checked_add(experimental_packed_plane_bytes(metadata.weight.shape)?)
572                .ok_or_else(|| cache_error(&path, "packed weight offset overflowed"))?;
573            packed_entries.insert(
574                stable,
575                ExperimentalPackedCfEntry {
576                    imaging_offset,
577                    weight_offset,
578                },
579            );
580        }
581        let actual_bytes = map.len() as u64;
582        if actual_bytes != cursor {
583            return Err(cache_error(
584                &path,
585                format!(
586                    "experimental packed CF length {actual_bytes} does not match indexed length {cursor}"
587                ),
588            ));
589        }
590        let map = Arc::new(map);
591        let payload_value_count = (actual_bytes - EXPERIMENTAL_PACKED_CF_HEADER_BYTES) as usize
592            / std::mem::size_of::<Complex32>();
593        let validate_values = env::var_os("CASA_RS_AWPROJECT_TRUST_PACKED_CF_EXPERIMENT").is_none();
594        if validate_values {
595            let payload = experimental_packed_complex_slice(
596                map.as_ref(),
597                EXPERIMENTAL_PACKED_CF_HEADER_BYTES as usize,
598                payload_value_count,
599                &path,
600            )?;
601            if payload
602                .iter()
603                .any(|value| !(value.re.is_finite() && value.im.is_finite()))
604            {
605                return Err(cache_error(
606                    &path,
607                    "experimental packed CF payload contains non-finite complex values",
608                ));
609            }
610        }
611        eprintln!(
612            "awproject_experimental_packed_cf path={} bytes={} pairs={} metadata_fingerprint={:016x} access=mmap-zero-copy payload_validation={} validated_values={}",
613            path.display(),
614            actual_bytes,
615            entries.len(),
616            identity.metadata_fingerprint,
617            if validate_values {
618                "finite-scan"
619            } else {
620                "trusted-experiment-skip"
621            },
622            if validate_values {
623                payload_value_count
624            } else {
625                0
626            },
627        );
628        Ok(Self {
629            path,
630            map,
631            entries: packed_entries,
632        })
633    }

636        &self,
637        stable: StableKey,
638        metadata: &AwConvolutionFunctionEntryMetadata,
639    ) -> Result<(Array2<Complex32>, Array2<Complex32>), ImagingError> {

641            cache_error(
642                &self.path,
643                "experimental packed CF index has no entry for a validated source key",
644            )
645        })?;

647            entry.imaging_offset,
648            metadata.imaging.shape,
649            &metadata.imaging.path,

652            entry.weight_offset,
653            metadata.weight.shape,
654            &metadata.weight.path,

659    fn read_plane(
660        &self,
661        offset: u64,
662        shape: [usize; 2],
663        source_path: &Path,
664    ) -> Result<Array2<Complex32>, ImagingError> {
665        let value_count = shape[0]
666            .checked_mul(shape[1])
667            .ok_or_else(|| cache_error(source_path, "experimental packed CF shape overflowed"))?;
668        let offset = usize::try_from(offset)
669            .map_err(|_| cache_error(source_path, "packed CF offset exceeds host usize"))?;
670        let values =
671            experimental_packed_complex_slice(self.map.as_ref(), offset, value_count, source_path)?
672                .to_vec();
673        Ok(Array2::from_shape_fn((shape[0], shape[1]), |(x, y)| {
674            values[x * shape[1] + y]
675        }))
676    }

709#[cfg(unix)]
710fn experimental_packed_complex_slice<'a>(
711    map: &'a Mmap,
712    offset: usize,
713    value_count: usize,
714    source_path: &Path,
715) -> Result<&'a [Complex32], ImagingError> {
716    let byte_count = value_count
717        .checked_mul(std::mem::size_of::<Complex32>())
718        .ok_or_else(|| cache_error(source_path, "experimental packed CF byte count overflowed"))?;
719    let end = offset
720        .checked_add(byte_count)
721        .ok_or_else(|| cache_error(source_path, "experimental packed CF range overflowed"))?;
722    if end > map.len() || offset % std::mem::align_of::<Complex32>() != 0 {
723        return Err(cache_error(
724            source_path,
725            "experimental packed CF range is outside or misaligned in its mapped file",
726        ));
727    }
728    // SAFETY: the bounds and alignment are checked above, the map is immutable
729    // for this process, and every bit pattern is a valid pair of `f32` values.
730    Ok(unsafe {
731        std::slice::from_raw_parts(map.as_ptr().add(offset).cast::<Complex32>(), value_count)
732    })
733}

735#[cfg(unix)]
736fn experimental_packed_plane_bytes(shape: [usize; 2]) -> Result<u64, ImagingError> {
737    shape[0]
738        .checked_mul(shape[1])
739        .and_then(|values| values.checked_mul(std::mem::size_of::<Complex32>()))
740        .and_then(|bytes| u64::try_from(bytes).ok())
741        .ok_or_else(|| {
742            ImagingError::ConvolutionFunctionCache(
743                "experimental packed CF plane byte count overflowed".to_string(),
744            )
745        })
746}

824    pub(crate) fn get_replay(
825        &self,
826        key: AwConvolutionFunctionKey,
827    ) -> Result<Arc<AwConvolutionFunctionReplayCell>, ImagingError> {
828        let stable = StableKey::from(key);
829        {
830            let mut state = self.lock_state()?;
831            state.clock = state.clock.wrapping_add(1);
832            let last_used = state.clock;
833            if let Some(resident) = state.cells.get_mut(&stable) {
834                resident.last_used = last_used;
835                let cell = Arc::clone(&resident.cell);
836                state.hits = state.hits.saturating_add(1);
837                return Ok(cell);
838            }
839        }
840
841        let metadata = self.cache.metadata(key).cloned().ok_or_else(|| {
842            cache_error(
843                self.cache.root(),
844                format!("no exact CF cell for key {key:?}"),
845            )
846        })?;
847        #[cfg(unix)]
848        let loaded = if let Some(store) = self.cache.experimental_packed.as_ref() {
849            AwConvolutionFunctionReplayCell::Mapped(Box::new(store.mapped_cell(stable, metadata)?))
850        } else {
851            AwConvolutionFunctionReplayCell::Owned(Arc::new(self.cache.load(key)?))
852        };
853        #[cfg(not(unix))]
854        let loaded = AwConvolutionFunctionReplayCell::Owned(Arc::new(self.cache.load(key)?));
855        let loaded = Arc::new(loaded);
856        let bytes = loaded.pixel_bytes();
857        let mut state = self.lock_state()?;
858        state.loads = state.loads.saturating_add(1);
859        state.clock = state.clock.wrapping_add(1);
860        let last_used = state.clock;
861
862        if let Some(resident) = state.cells.get_mut(&stable) {
863            resident.last_used = last_used;
864            let cell = Arc::clone(&resident.cell);
865            state.hits = state.hits.saturating_add(1);
866            return Ok(cell);
867        }
868        if bytes > self.byte_budget {
869            return Ok(loaded);
870        }
871
872        while state.resident_bytes.saturating_add(bytes) > self.byte_budget {
873            let Some(eviction_key) = state
874                .cells
875                .iter()
876                .min_by_key(|(key, resident)| (resident.last_used, **key))
877                .map(|(key, _)| *key)
878            else {
879                break;
880            };
881            if let Some(evicted) = state.cells.remove(&eviction_key) {
882                state.resident_bytes = state.resident_bytes.saturating_sub(evicted.bytes);
883                state.evictions = state.evictions.saturating_add(1);
884            }
885        }
886        state.resident_bytes = state.resident_bytes.saturating_add(bytes);
887        state.cells.insert(
888            stable,
889            ResidentCell {
890                cell: Arc::clone(&loaded),
891                bytes,
892                last_used,
893            },
894        );
895        Ok(loaded)
896    }
897
898    /// Current residency and I/O counters.
899    pub fn stats(&self) -> Result<AwConvolutionFunctionResidentStats, ImagingError> {
900        let state = self.lock_state()?;
901        Ok(AwConvolutionFunctionResidentStats {
902            resident_cells: state.cells.len(),
903            resident_bytes: state.resident_bytes,
904            loads: state.loads,
905            hits: state.hits,
906            evictions: state.evictions,
907        })
908    }

912            cache_error(

914                "resident CF cache lock was poisoned by an earlier panic",
915            )
916        })
917    }

921    /// Open and fully validate a CASA `CFS`/`WTCFS` cache index.
922    pub fn open(path: impl AsRef<Path>) -> Result<Self, ImagingError> {
923        let root = path.as_ref();
924        let directory = fs::read_dir(root)
925            .map_err(|error| cache_error(root, format!("cannot read cache directory: {error}")))?;
926        let mut partial = BTreeMap::<StableKey, PartialEntry>::new();
927        let mut recognized = 0usize;
928        for entry in directory {
929            let entry = entry.map_err(|error| {
930                cache_error(root, format!("cannot enumerate cache directory: {error}"))
931            })?;
932            let file_type = entry.file_type().map_err(|error| {
933                cache_error(
934                    &entry.path(),
935                    format!("cannot inspect cache entry: {error}"),
936                )
937            })?;
938            if !file_type.is_dir() {
939                continue;
940            }
941            let name = entry.file_name();
942            let name = name.to_string_lossy();
943            let kind = if name.starts_with(WEIGHT_PREFIX) && name.ends_with(".im") {
944                CacheCellKind::Weight
945            } else if name.starts_with(IMAGING_PREFIX) && name.ends_with(".im") {
946                CacheCellKind::Imaging
947            } else {
948                continue;
949            };
950            recognized += 1;
951            let (key, metadata) = read_kernel_metadata(&entry.path())?;
952            let stable = StableKey::from(key);
953            let slot = partial.entry(stable).or_insert_with(|| PartialEntry {
954                key,
955                imaging: None,
956                weight: None,
957            });
958            if StableKey::from(slot.key) != stable {
959                return Err(cache_error(
960                    &entry.path(),
961                    "two non-identical scientific keys produced the same stable key",
962                ));
963            }
964            let target = match kind {
965                CacheCellKind::Imaging => &mut slot.imaging,
966                CacheCellKind::Weight => &mut slot.weight,
967            };
968            if let Some(previous) = target.as_ref() {
969                return Err(cache_error(
970                    &entry.path(),
971                    format!(
972                        "duplicate {} cell for key {:?}; first path is {}",
973                        kind.label(),
974                        key,
975                        previous.path.display()
976                    ),
977                ));
978            }
979            *target = Some(metadata);
980        }
981        if recognized == 0 {
982            return Err(cache_error(root, "no CFS_*.im or WTCFS_*.im entries found"));
983        }
984
985        let mut entries = BTreeMap::new();
986        for (stable, pair) in partial {
987            let imaging = pair.imaging.ok_or_else(|| {
988                cache_error(
989                    root,
990                    format!("missing imaging CFS cell for key {:?}", pair.key),
991                )
992            })?;
993            let weight = pair.weight.ok_or_else(|| {
994                cache_error(
995                    root,
996                    format!("missing weight WTCFS cell for key {:?}", pair.key),
997                )
998            })?;
999            validate_pair(&pair.key, &imaging, &weight)?;
1000            entries.insert(
1001                stable,
1002                AwConvolutionFunctionEntryMetadata {
1003                    key: pair.key,
1004                    imaging,
1005                    weight,
1006                },
1007            );
1008        }
1009        validate_cache_axes(root, &entries)?;
1010        let inventory = inventory_from_entries(&entries);
1011        let identity = identity_from_entries(root, &entries, &inventory);
1012        #[cfg(unix)]
1013        let experimental_packed = env::var_os("CASA_RS_AWPROJECT_PACKED_CF_EXPERIMENT")
1014            .map(PathBuf::from)
1015            .map(|path| ExperimentalPackedCfStore::open(path, &entries, &identity))
1016            .transpose()?
1017            .map(Arc::new);
1018        #[cfg(unix)]
1019        let experimental_direct_sample_select =
1020            env::var_os("CASA_RS_AWPROJECT_DIRECT_CF_SELECT_EXPERIMENT").is_some();
1021        #[cfg(unix)]
1022        if experimental_direct_sample_select {
1023            eprintln!(
1024                "awproject_experimental_direct_cf_select pairs={} frequencies={} wplanes={} mueller={} pa_bins={}",
1025                inventory.paired_cells,
1026                inventory.frequencies_hz.len(),
1027                inventory.w_values_lambda.len(),
1028                inventory.mueller_elements.len(),
1029                inventory.parallactic_angles_deg.len(),
1030            );
1031        }
1032        Ok(Self {
1033            root: Arc::new(root.to_path_buf()),
1034            entries: Arc::new(entries),
1035            inventory,
1036            identity,
1037            #[cfg(unix)]
1038            experimental_direct_sample_select,
1039            #[cfg(unix)]
1040            experimental_packed,
1041        })
1042    }

1072    /// Select the nearest cache cell for a requested frequency, W value,
1073    /// Mueller element, and parallactic angle.
1074    ///
1075    /// Mueller is exact. Frequency, W, and PA follow CASA's cache lookup model
1076    /// and choose the closest represented coordinate using deterministic key
1077    /// order to break ties.
1078    pub fn nearest_key(
1079        &self,
1080        frequency_hz: f64,
1081        w_value_lambda: f64,
1082        mueller_element: i32,
1083        parallactic_angle_deg: f64,
1084    ) -> Option<AwConvolutionFunctionKey> {
1085        if !(frequency_hz.is_finite()
1086            && frequency_hz > 0.0
1087            && w_value_lambda.is_finite()
1088            && parallactic_angle_deg.is_finite())
1089        {
1090            return None;
1091        }
1092        self.entries
1093            .values()
1094            .filter(|entry| entry.key.mueller_element == mueller_element)
1095            .min_by(|left, right| {
1096                nearest_distance(
1097                    left.key,
1098                    frequency_hz,
1099                    w_value_lambda,
1100                    parallactic_angle_deg,
1101                )
1102                .total_cmp(&nearest_distance(
1103                    right.key,
1104                    frequency_hz,
1105                    w_value_lambda,
1106                    parallactic_angle_deg,
1107                ))
1108                .then_with(|| StableKey::from(left.key).cmp(&StableKey::from(right.key)))
1109            })
1110            .map(|entry| entry.key)
1111    }

1113    /// Select a cache cell with CASA `CFBuffer` axis semantics.
1114    ///
1115    /// Frequency and parallactic angle are selected independently. W uses
1116    /// CASA's quadratic plane index, `round(sqrt(WIncr * abs(w)))`, rather than
1117    /// a generic multidimensional distance. CASA also uses the conjugate
1118    /// Mueller row for non-positive W; in its 4-by-4 flattened Mueller
1119    /// convention that maps element `m` to `15 - m`. When `conjugate_beams`
1120    /// is true, CASA first maps the data frequency to
1121    /// `sqrt(2 * image_reference_frequency_hz^2 - sample_frequency_hz^2)` and
1122    /// then selects the nearest primary CF frequency. The persisted `ConjFreq`
1123    /// values describe individual cells, but are quantized and cannot be used
1124    /// as an inverse lookup for this operation.
1125    pub fn select_key_for_sample(
1126        &self,
1127        sample_frequency_hz: f64,
1128        image_reference_frequency_hz: f64,
1129        w_lambda: f64,
1130        mueller_element: i32,
1131        parallactic_angle_deg: f64,
1132        conjugate_beams: bool,
1133    ) -> Option<AwConvolutionFunctionKey> {
1134        if !(sample_frequency_hz.is_finite()
1135            && sample_frequency_hz > 0.0
1136            && image_reference_frequency_hz.is_finite()
1137            && image_reference_frequency_hz > 0.0
1138            && w_lambda.is_finite()
1139            && parallactic_angle_deg.is_finite())
1140        {
1141            return None;
1142        }
1143        let requested_cf_frequency_hz = if conjugate_beams {
1144            let radicand = 2.0 * image_reference_frequency_hz * image_reference_frequency_hz
1145                - sample_frequency_hz * sample_frequency_hz;
1146            if !(radicand.is_finite() && radicand > 0.0) {
1147                return None;
1148            }
1149            radicand.sqrt()
1150        } else {
1151            sample_frequency_hz
1152        };
1153        let selected_mueller_element = if w_lambda > 0.0 {
1154            mueller_element
1155        } else {
1156            15_i32.checked_sub(mueller_element)?
1157        };
1158        #[cfg(unix)]
1159        if self.experimental_direct_sample_select {
1160            return self.select_key_for_sample_direct(
1161                requested_cf_frequency_hz,
1162                w_lambda,
1163                selected_mueller_element,
1164                parallactic_angle_deg,
1165            );
1166        }
1167        self.select_key_for_sample_scanned(
1168            requested_cf_frequency_hz,
1169            w_lambda,
1170            selected_mueller_element,
1171            parallactic_angle_deg,
1172        )
1173    }
1174
1175    fn select_key_for_sample_scanned(
1176        &self,
1177        requested_cf_frequency_hz: f64,
1178        w_lambda: f64,
1179        selected_mueller_element: i32,
1180        parallactic_angle_deg: f64,
1181    ) -> Option<AwConvolutionFunctionKey> {
1182        let candidates = self
1183            .entries
1184            .values()
1185            .filter(|entry| entry.key.mueller_element == selected_mueller_element)
1186            .collect::<Vec<_>>();
1187        let first = candidates.first()?;
1188        let w_increment = first.imaging.w_increment;
1189        let w_values = sorted_f64_bits(
1190            candidates
1191                .iter()
1192                .map(|entry| entry.key.w_value_lambda.to_bits())
1193                .collect(),
1194        );
1195        let w_index = if w_values.len() <= 1 || w_increment == 0.0 {
1196            0
1197        } else {
1198            (w_increment * w_lambda.abs())
1199                .sqrt()
1200                .round()
1201                .clamp(0.0, (w_values.len() - 1) as f64) as usize
1202        };
1203        let selected_w = w_values[w_index];
1204        candidates
1205            .into_iter()
1206            .filter(|entry| entry.key.w_value_lambda.to_bits() == selected_w.to_bits())
1207            .min_by(|left, right| {
1208                (left.key.frequency_hz - requested_cf_frequency_hz)
1209                    .abs()
1210                    .total_cmp(&(right.key.frequency_hz - requested_cf_frequency_hz).abs())
1211                    .then_with(|| {
1212                        circular_degrees(left.key.parallactic_angle_deg - parallactic_angle_deg)
1213                            .abs()
1214                            .total_cmp(
1215                                &circular_degrees(
1216                                    right.key.parallactic_angle_deg - parallactic_angle_deg,
1217                                )
1218                                .abs(),
1219                            )
1220                    })
1221                    .then_with(|| StableKey::from(left.key).cmp(&StableKey::from(right.key)))
1222            })
1223            .map(|entry| entry.key)
1224    }
1225
1226    #[cfg(unix)]
1227    fn select_key_for_sample_direct(
1228        &self,
1229        requested_cf_frequency_hz: f64,
1230        w_lambda: f64,
1231        selected_mueller_element: i32,
1232        parallactic_angle_deg: f64,
1233    ) -> Option<AwConvolutionFunctionKey> {
1234        if !self
1235            .inventory
1236            .mueller_elements
1237            .contains(&selected_mueller_element)
1238        {
1239            return None;
1240        }
1241        let first = self.entries.values().next()?;
1242        let w_values = &self.inventory.w_values_lambda;
1243        let w_index = if w_values.len() <= 1 || first.imaging.w_increment == 0.0 {
1244            0
1245        } else {
1246            (first.imaging.w_increment * w_lambda.abs())
1247                .sqrt()
1248                .round()
1249                .clamp(0.0, (w_values.len() - 1) as f64) as usize
1250        };
1251        let frequency_hz =
1252            self.inventory
1253                .frequencies_hz
1254                .iter()
1255                .copied()
1256                .min_by(|left, right| {
1257                    (left - requested_cf_frequency_hz)
1258                        .abs()
1259                        .total_cmp(&(right - requested_cf_frequency_hz).abs())
1260                        .then_with(|| left.to_bits().cmp(&right.to_bits()))
1261                })?;
1262        let parallactic_angle_deg = self
1263            .inventory
1264            .parallactic_angles_deg
1265            .iter()
1266            .copied()
1267            .min_by(|left, right| {
1268                circular_degrees(*left - parallactic_angle_deg)
1269                    .abs()
1270                    .total_cmp(&circular_degrees(*right - parallactic_angle_deg).abs())
1271                    .then_with(|| left.to_bits().cmp(&right.to_bits()))
1272            })?;
1273        let key = AwConvolutionFunctionKey {
1274            frequency_hz,
1275            w_value_lambda: w_values[w_index],
1276            mueller_element: selected_mueller_element,
1277            parallactic_angle_deg,
1278        };
1279        self.entries
1280            .get(&StableKey::from(key))
1281            .map(|entry| entry.key)
1282    }

1284    /// Load one exact paired CF cell without retaining its pixels in the cache.
1285    pub fn load(
1286        &self,
1287        key: AwConvolutionFunctionKey,
1288    ) -> Result<AwConvolutionFunctionCell, ImagingError> {
1289        let metadata = self
1290            .metadata(key)
1291            .cloned()
1292            .ok_or_else(|| cache_error(self.root(), format!("no exact CF cell for key {key:?}")))?;
1293        #[cfg(unix)]
1294        let stable = StableKey::from(key);
1295        #[cfg(unix)]
1296        let packed = self
1297            .experimental_packed
1298            .as_ref()
1299            .map(|store| store.load(stable, &metadata))
1300            .transpose()?;
1301        #[cfg(unix)]
1302        let (imaging, weight) = if let Some(packed) = packed {
1303            packed
1304        } else {
1305            (
1306                read_kernel_pixels(&metadata.imaging)?,
1307                read_kernel_pixels(&metadata.weight)?,
1308            )
1309        };
1310        #[cfg(not(unix))]
1311        let (imaging, weight) = (
1312            read_kernel_pixels(&metadata.imaging)?,
1313            read_kernel_pixels(&metadata.weight)?,
1314        );
1315        Ok(AwConvolutionFunctionCell {
1316            metadata,
1317            imaging,
1318            weight,
1319        })
1320    }
1321}

1345fn read_kernel_metadata(
1346    path: &Path,
1347) -> Result<
1348    (
1349        AwConvolutionFunctionKey,
1350        AwConvolutionFunctionKernelMetadata,
1351    ),
1352    ImagingError,
1353> {
1354    let image = PagedImage::<Complex32>::open(path)
1355        .map_err(|error| cache_error(path, format!("cannot open Complex32 image: {error}")))?;
1356    let shape = image.shape();
1357    if shape.len() != 4 || shape[2] != 1 || shape[3] != 1 || shape[0] == 0 || shape[1] == 0 {
1358        return Err(cache_error(
1359            path,
1360            format!("expected non-empty [nx, ny, 1, 1] shape, got {shape:?}"),
1361        ));
1362    }
1363    let coordinates = validate_coordinates(path, image.coordinates())?;
1364    let frequency_hz = spectral_reference_frequency(path, image.coordinates())?;
1365    let misc = image.misc_info();

1367        frequency_hz,

1370        parallactic_angle_deg: required_f64(&misc, "ParallacticAngle", path)?,
1371    }
1372    .validate(path)?;

1374        positive_integral_usize(required_f64(&misc, "Sampling", path)?, "Sampling", path)?;
1375    let x_support = nonnegative_usize(required_i32(&misc, "Xsupport", path)?, "Xsupport", path)?;
1376    let y_support = nonnegative_usize(required_i32(&misc, "Ysupport", path)?, "Ysupport", path)?;
1377    if x_support == 0 || y_support == 0 {
1378        return Err(cache_error(path, "Xsupport and Ysupport must be positive"));
1379    }
1380    let metadata = AwConvolutionFunctionKernelMetadata {
1381        path: path.to_path_buf(),
1382        shape: [shape[0], shape[1]],
1383        sampling,
1384        x_support,
1385        y_support,
1386        uv_coordinate: coordinates.uv,
1387        telescope_name: required_string(&misc, "TelescopeName", path)?,
1388        band_name: required_string(&misc, "BandName", path)?,
1389        diameter_m: required_f64(&misc, "Diameter", path)?,
1390        conjugate_frequency_hz: required_f64(&misc, "ConjFreq", path)?,
1391        conjugate_polarization: required_i32(&misc, "ConjPoln", path)?,
1392        polarization: coordinates.polarization,
1393        w_increment: required_f64(&misc, "WIncr", path)?,
1394        rotationally_symmetric: required_bool(&misc, "OpCode", path)?,
1395    };

1405fn validate_coordinates(
1406    path: &Path,
1407    coords: &CoordinateSystem,
1408) -> Result<ValidatedCfCoordinates, ImagingError> {
1409    if coords.n_pixel_axes() != 4 {
1410        return Err(cache_error(
1411            path,
1412            format!(
1413                "expected four CF coordinate axes, got {}",
1414                coords.n_pixel_axes()
1415            ),
1416        ));
1417    }
1418    let Some(linear_index) = coords.find_coordinate(CoordinateType::Linear) else {
1419        return Err(cache_error(path, "missing UV linear coordinate"));
1420    };
1421    let linear_model = coords.coordinate(linear_index);
1422    if linear_model.axis_names() != ["UU", "VV"]
1423        || linear_model.axis_units() != ["lambda", "lambda"]
1424    {
1425        return Err(cache_error(
1426            path,
1427            format!(
1428                "expected UU/VV axes in lambda, got {:?} in {:?}",
1429                linear_model.axis_names(),
1430                linear_model.axis_units()
1431            ),
1432        ));
1433    }
1434    let CoordinateModel::Linear(linear) = linear_model else {
1435        return Err(cache_error(
1436            path,
1437            "UV linear coordinate has the wrong model type",
1438        ));
1439    };
1440    if linear.n_pixel_axes() != 2 || linear.n_world_axes() != 2 {
1441        return Err(cache_error(
1442            path,
1443            format!(
1444                "expected a two-axis UU/VV coordinate, got {} pixel and {} world axes",
1445                linear.n_pixel_axes(),
1446                linear.n_world_axes()
1447            ),
1448        ));
1449    }
1450    let reference_value = two_f64(path, "UV reference value", linear.reference_value())?;
1451    let reference_pixel = two_f64(path, "UV reference pixel", linear.reference_pixel())?;
1452    let increment = two_f64(path, "UV increment", linear.increment())?;
1453    let pc = linear.pc_matrix();
1454    let uv_values = [
1455        reference_value[0],
1456        reference_value[1],
1457        reference_pixel[0],
1458        reference_pixel[1],
1459        increment[0],
1460        increment[1],
1461        pc[[0, 0]],
1462        pc[[0, 1]],
1463        pc[[1, 0]],
1464        pc[[1, 1]],
1465    ];
1466    if uv_values.iter().any(|value| !value.is_finite()) {
1467        return Err(cache_error(
1468            path,
1469            "UV affine coordinate contains a non-finite value",
1470        ));
1471    }
1472    if increment[0] == 0.0 || increment[1] == 0.0 {
1473        return Err(cache_error(
1474            path,
1475            "UV coordinate increments must be non-zero",
1476        ));
1477    }
1478    let determinant = pc[[0, 0]] * pc[[1, 1]] - pc[[0, 1]] * pc[[1, 0]];
1479    if !determinant.is_finite() || determinant == 0.0 {
1480        return Err(cache_error(
1481            path,
1482            "UV coordinate PC matrix must be invertible",
1483        ));
1484    }
1485    let uv = AwConvolutionFunctionUvCoordinateIdentity {
1486        reference_value_bits: reference_value.map(f64::to_bits),
1487        reference_pixel_bits: reference_pixel.map(f64::to_bits),
1488        increment_bits: increment.map(f64::to_bits),
1489        pc_matrix_bits: [
1490            [pc[[0, 0]].to_bits(), pc[[0, 1]].to_bits()],
1491            [pc[[1, 0]].to_bits(), pc[[1, 1]].to_bits()],
1492        ],
1493    };
1494    let stokes_index = coords
1495        .find_coordinate(CoordinateType::Stokes)
1496        .ok_or_else(|| cache_error(path, "missing Stokes coordinate"))?;
1497    let CoordinateModel::Stokes(stokes) = coords.coordinate(stokes_index) else {
1498        return Err(cache_error(
1499            path,
1500            "Stokes coordinate has the wrong model type",
1501        ));
1502    };
1503    let [polarization] = stokes.stokes() else {
1504        return Err(cache_error(
1505            path,
1506            format!(
1507                "expected one degenerate correlation coordinate, got {:?}",
1508                stokes.stokes()
1509            ),
1510        ));
1511    };
1512    if coords.find_coordinate(CoordinateType::Spectral).is_none() {
1513        return Err(cache_error(path, "missing spectral coordinate"));
1514    }
1515    Ok(ValidatedCfCoordinates {
1516        uv,
1517        polarization: *polarization,
1518    })
1519}

1553) -> Result<(), ImagingError> {

1575    }

1577        .x_support
1578        .saturating_mul(metadata.sampling)

1581        .y_support
1582        .saturating_mul(metadata.sampling)
1583        .saturating_mul(2);
1584    if metadata.shape[0] < required_x || metadata.shape[1] < required_y {
1585        return Err(cache_error(
1586            path,
1587            format!(
1588                "shape {:?} cannot hold support {}x{} at sampling {}",
1589                metadata.shape, metadata.x_support, metadata.y_support, metadata.sampling

1591        ));
1592    }
1593    Ok(())

1596fn validate_pair(
1597    key: &AwConvolutionFunctionKey,
1598    imaging: &AwConvolutionFunctionKernelMetadata,
1599    weight: &AwConvolutionFunctionKernelMetadata,
1600) -> Result<(), ImagingError> {
1601    for (label, same) in [
1602        ("sampling", imaging.sampling == weight.sampling),
1603        ("telescope", imaging.telescope_name == weight.telescope_name),
1604        ("band", imaging.band_name == weight.band_name),
1605        (
1606            "diameter",
1607            imaging.diameter_m.to_bits() == weight.diameter_m.to_bits(),
1608        ),
1609        (
1610            "conjugate frequency",
1611            imaging.conjugate_frequency_hz.to_bits() == weight.conjugate_frequency_hz.to_bits(),
1612        ),
1613        (
1614            "conjugate polarization",
1615            imaging.conjugate_polarization == weight.conjugate_polarization,
1616        ),
1617        ("polarization", imaging.polarization == weight.polarization),
1618        (
1619            "W increment",
1620            imaging.w_increment.to_bits() == weight.w_increment.to_bits(),
1621        ),
1622        (
1623            "rotational symmetry",
1624            imaging.rotationally_symmetric == weight.rotationally_symmetric,
1625        ),
1626    ] {
1627        if !same {
1628            return Err(cache_error(
1629                &weight.path,
1630                format!(
1631                    "{label} does not match paired imaging cell {} for key {key:?}",
1632                    imaging.path.display()
1633                ),
1634            ));
1635        }
1636    }
1637    if imaging.uv_coordinate.reference_value_bits != weight.uv_coordinate.reference_value_bits
1638        || imaging.uv_coordinate.pc_matrix_bits != weight.uv_coordinate.pc_matrix_bits
1639        || !same_uv_world_window(imaging, weight)
1640    {
1641        return Err(cache_error(
1642            &weight.path,
1643            format!(
1644                "weight CF coordinate does not cover the same UV world window as paired imaging cell {} for key {key:?}: imaging_shape={:?} weight_shape={:?} imaging_support={}x{} weight_support={}x{} imaging_uv={:?} weight_uv={:?}",
1645                imaging.path.display(),
1646                imaging.shape,
1647                weight.shape,
1648                imaging.x_support,
1649                imaging.y_support,
1650                weight.x_support,
1651                weight.y_support,
1652                imaging.uv_coordinate,
1653                weight.uv_coordinate
1654            ),
1655        ));
1656    }
1657    Ok(())
1658}

1660fn same_uv_world_window(
1661    imaging: &AwConvolutionFunctionKernelMetadata,
1662    weight: &AwConvolutionFunctionKernelMetadata,
1663) -> bool {
1664    (0..2).all(|axis| {
1665        let imaging_size = imaging.shape[axis] as f64;
1666        let weight_size = weight.shape[axis] as f64;
1667        let imaging_reference_pixel =
1668            f64::from_bits(imaging.uv_coordinate.reference_pixel_bits[axis]);
1669        let weight_reference_pixel =
1670            f64::from_bits(weight.uv_coordinate.reference_pixel_bits[axis]);
1671        let imaging_increment = f64::from_bits(imaging.uv_coordinate.increment_bits[axis]);
1672        let weight_increment = f64::from_bits(weight.uv_coordinate.increment_bits[axis]);
1673        nearly_equal(
1674            imaging_reference_pixel / imaging_size,
1675            weight_reference_pixel / weight_size,
1676        ) && nearly_equal(
1677            imaging_increment * imaging_size,
1678            weight_increment * weight_size,
1679        )
1680    })
1681}
1682
1683fn nearly_equal(left: f64, right: f64) -> bool {
1684    let scale = left.abs().max(right.abs()).max(1.0);
1685    (left - right).abs() <= scale * 32.0 * f64::EPSILON
1686}

1688fn validate_cache_axes(
1689    root: &Path,
1690    entries: &BTreeMap<StableKey, AwConvolutionFunctionEntryMetadata>,
1691) -> Result<(), ImagingError> {
1692    let first = entries
1693        .values()
1694        .next()
1695        .ok_or_else(|| cache_error(root, "validated cache has no paired cells"))?;
1696    let w_increment_bits = first.imaging.w_increment.to_bits();
1697    if entries
1698        .values()
1699        .any(|entry| entry.imaging.w_increment.to_bits() != w_increment_bits)
1700    {
1701        return Err(cache_error(
1702            root,
1703            "WIncr must be identical across one CF cache",
1704        ));
1705    }
1706
1707    let inventory = inventory_from_entries(entries);
1708    let expected_cells = inventory
1709        .frequencies_hz
1710        .len()
1711        .checked_mul(inventory.w_values_lambda.len())
1712        .and_then(|count| count.checked_mul(inventory.mueller_elements.len()))
1713        .and_then(|count| count.checked_mul(inventory.parallactic_angles_deg.len()))
1714        .ok_or_else(|| cache_error(root, "CF-cache axis cardinality overflow"))?;
1715    if entries.len() != expected_cells {
1716        return Err(cache_error(
1717            root,
1718            format!(
1719                "CF cache is not a complete frequency x W x Mueller x PA product: found {} paired cells, expected {expected_cells}",
1720                entries.len()
1721            ),
1722        ));
1723    }
1724
1725    let w_increment = first.imaging.w_increment;
1726    if inventory.w_values_lambda.len() > 1 && w_increment == 0.0 {
1727        return Err(cache_error(
1728            root,
1729            "multiple W planes require a positive WIncr",
1730        ));
1731    }
1732    if w_increment > 0.0 {
1733        for (index, &actual) in inventory.w_values_lambda.iter().enumerate() {
1734            let expected = (index * index) as f64 / w_increment;
1735            let tolerance = expected.abs().max(1.0) * 1.0e-10;
1736            if (actual - expected).abs() > tolerance {
1737                return Err(cache_error(
1738                    root,
1739                    format!(
1740                        "W plane {index} has value {actual}, expected CASA quadratic value {expected} from WIncr {w_increment}"
1741                    ),
1742                ));
1743            }
1744        }
1745    }
1746    Ok(())
1747}

1988fn nearest_distance(
1989    key: AwConvolutionFunctionKey,
1990    frequency_hz: f64,
1991    w_value_lambda: f64,
1992    parallactic_angle_deg: f64,
1993) -> f64 {
1994    let frequency_scale = frequency_hz.abs().max(1.0);
1995    let w_scale = w_value_lambda.abs().max(1.0);
1996    let pa_delta = circular_degrees(key.parallactic_angle_deg - parallactic_angle_deg);
1997    ((key.frequency_hz - frequency_hz) / frequency_scale).abs()
1998        + ((key.w_value_lambda - w_value_lambda) / w_scale).abs()
1999        + pa_delta.abs() / 360.0
2000}

2002fn circular_degrees(value: f64) -> f64 {
2003    (value + 180.0).rem_euclid(360.0) - 180.0
2004}
