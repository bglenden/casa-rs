2//! casacore's `MeasureHolder`: an epoch, direction, position or frequency
3//! with any of its kind's reference codes, read from and written to the
4//! measure records CASA's tools exchange.
5//!
6//! A [`Measure`] keeps what casacore's measure value keeps: the epoch's MJD,
7//! the direction's cosines, the position's Cartesian vector and the
8//! frequency in Hz. It therefore holds codes casa-types does not convert
9//! (`SUN`, `B1950_VLA`, `REST`): they define, show and compare, and only a
10//! conversion that needs them fails.
11//!
12//! - **Reading** ([`Measure::from_record`]) follows
13//!   `MeasureHolder::fromRecord` and the measure values' `putValue`: the
14//!   record's `type` names the kind, `refer` names a code by unique
15//!   case-insensitive prefix (an unknown one is the kind's default, with
16//!   casacore's warning), and `m0`, `m1`, `m2` are quantity records in any
17//!   unit their kind accepts.
18//! - **Writing** ([`Measure::to_value`]) follows `MeasureHolder::toRecord`:
19//!   `{'type', 'refer', 'm0', ...}` with the epoch in days, angles in
20//!   radians, the position's length in metres and the frequency in Hz.
21//! - **Positions** keep casacore's encoding of a WGS84 height as the length
22//!   of the position vector (`MVPosition`): a zero height is a 1e-6 m
23//!   vector, and heights between 0 and −7000 km, or near 743.568 m, are
24//!   shifted, so `me.position('WGS84', '10deg', '20deg', '0m')` has a
25//!   height of 1e-6 m, as in CASA.
26//!
27//! Divergence: measure offsets and multi-valued measures (quantity records
28//! with more than one value) are not yet implemented in casa-rs; reading one
29//! fails with a message that says so.
30

42/// The speed of light, m/s (casacore's `QC::c`).
43const SPEED_OF_LIGHT_M_S: f64 = 299_792_458.0;
44/// Planck's constant, J s, as casacore's `QC::h` gives it (the 1986 CODATA
45/// value).
46const PLANCK_J_S: f64 = 6.626_075_5e-34;
47/// The band of vector lengths in which casacore encodes WGS84 heights
48/// (`MVPosition::loLimit` and `hiLimit`).
49const LO_LIMIT_M: f64 = 743.568;
50const HI_LIMIT_M: f64 = 743.569;

159impl Codes {
160    /// The code `input` names by unique case-insensitive prefix
161    /// (`MUString::minimaxNC`), as the code it stands for; `None` if it names
162    /// none.
163    pub(crate) fn resolve(&self, input: &str) -> Option<&'static str> {
164        let code = self.all[unique_prefix_index(input, self.all)?];
165        Some(
166            self.aliases
167                .iter()
168                .find(|(alias, _)| *alias == code)
169                .map_or(code, |(_, canonical)| canonical),
170        )
171    }
172
173    /// The code `input` names, else the kind's default (`setRefString`).
174    pub(crate) fn resolve_or_default(&self, input: &str) -> &'static str {
175        self.resolve(input).unwrap_or(self.default)
176    }
177
178    /// The kind's default code.
179    pub(crate) fn default_code(&self) -> &'static str {
180        self.default
181    }
182}

246/// A record [`Measure::from_record`] could not read.
247#[derive(Debug, Clone, PartialEq, Eq)]
248pub(crate) enum HolderError {
249    /// Not a measure record (no string `type` and `refer`): casacore's
250    /// "Illegal Measure record".
251    NotAMeasure,
252    /// casacore's message for a measure record it cannot read.
253    Invalid(String),
254}

256impl HolderError {
257    /// casacore's message.
258    pub(crate) fn message(&self) -> String {
259        match self {
260            Self::NotAMeasure => {
261                "Illegal Measure record in MeasureHolder::fromRecord\n".to_string()
262            }
263            Self::Invalid(message) => message.clone(),
264        }
265    }

280impl Measure {
281    /// The measure's kind.
282    pub(crate) fn kind(&self) -> MeasureKind {
283        match self.value {
284            MeasureValue::Epoch(_) => MeasureKind::Epoch,
285            MeasureValue::Direction(_) => MeasureKind::Direction,
286            MeasureValue::Position(_) => MeasureKind::Position,
287            MeasureValue::Frequency(_) => MeasureKind::Frequency,
288        }
289    }
290
291    /// `MeasureHolder::fromRecord`.
292    pub(crate) fn from_record(
293        record: &BTreeMap<String, Variant>,
294    ) -> Result<ReadMeasure, HolderError> {
295        let (Some(Variant::Text(kind)), Some(Variant::Text(refer))) =
296            (record.get("type"), record.get("refer"))
297        else {
298            return Err(HolderError::NotAMeasure);
299        };
300        let kind = match kind.to_ascii_lowercase().as_str() {
301            "epoch" => MeasureKind::Epoch,
302            "direction" => MeasureKind::Direction,
303            "position" => MeasureKind::Position,
304            "frequency" => MeasureKind::Frequency,
305            "doppler" | "radialvelocity" | "baseline" | "uvw" | "earthmagnetic" => {
306                return Err(HolderError::Invalid(format!(
307                    "{kind} measures are not yet implemented in casa-rs"
308                )));
309            }
310            _ => {
311                return Err(HolderError::Invalid(format!(
312                    "{kind} is an unknown measure typeUnknown Measure record in \
313                     MeasureHolder::fromRecord\n"
314                )));
315            }
316        };
317        let codes = kind.codes();
318        let (code, warning) = match codes.resolve(refer) {
319            Some(code) => (code, None),
320            None => (
321                codes.default,
322                (!refer.is_empty()).then(|| {
323                    format!(
324                        "Illegal or unknown reference type '{refer}' for {} definition. \
325                         DEFAULT ({}) assumed.",
326                        kind.record_type(),
327                        codes.default
328                    )
329                }),
330            ),
331        };
332        if record.contains_key("offset") {
333            return Err(HolderError::Invalid(
334                "measure offsets are not yet implemented in casa-rs".to_string(),
335            ));
336        }
337        let values = record_values(record)?;
338        Ok(ReadMeasure {
339            measure: Self::from_values(kind, code, &values)?,
340            warning,
341        })
342    }
343
344    /// The measure from quantities, as the kind's `putValue` reads them.
345    pub(crate) fn from_values(
346        kind: MeasureKind,
347        refer: &'static str,
348        values: &[Quantity],
349    ) -> Result<Self, HolderError> {
350        let value = match kind {
351            MeasureKind::Epoch => epoch_value(values)?,
352            MeasureKind::Direction => direction_value(values)?,
353            MeasureKind::Position => position_value(values)?,
354            MeasureKind::Frequency => frequency_value(values)?,
355        };
356        Ok(Self { refer, value })
357    }
358
359    /// `MeasureHolder::toRecord`.
360    pub(crate) fn to_value(self) -> CasaValue {
361        let mut fields = vec![
362            (
363                "type",
364                CasaValue::Text(self.kind().record_type().to_string()),
365            ),
366            ("refer", CasaValue::Text(self.refer.to_string())),
367        ];
368        match self.value {
369            MeasureValue::Epoch(mjd) => fields.push(("m0", CasaValue::quantity(mjd.as_mjd(), "d"))),
370            MeasureValue::Direction(cosines) => {
371                let (longitude, latitude) = direction_angles(cosines);
372                fields.push(("m0", CasaValue::quantity(longitude, "rad")));
373                fields.push(("m1", CasaValue::quantity(latitude, "rad")));
374            }
375            MeasureValue::Position(xyz) => {
376                let (longitude, latitude, length) = position_spherical(xyz);
377                fields.push(("m0", CasaValue::quantity(longitude, "rad")));
378                fields.push(("m1", CasaValue::quantity(latitude, "rad")));
379                fields.push(("m2", CasaValue::quantity(length, "m")));
380            }
381            MeasureValue::Frequency(hz) => fields.push(("m0", CasaValue::quantity(hz, "Hz"))),
382        }
383        CasaValue::record(fields)
384    }

386    /// The measure in casa-types' form, failing for a code casa-types does
387    /// not convert.
388    pub(crate) fn to_casa(self) -> Result<CasaMeasure, String> {
389        let unsupported = |kind: &str| {
390            format!(
391                "the {} {kind} frame is not yet implemented in casa-rs",
392                self.refer
393            )
394        };
395        Ok(match self.value {
396            MeasureValue::Epoch(mjd) => CasaMeasure::Epoch(MEpoch::new(
397                mjd,
398                EpochRef::from_str(self.refer).map_err(|_| unsupported("epoch"))?,
399            )),
400            MeasureValue::Direction(cosines) => CasaMeasure::Direction(MDirection::from_cosines(
401                cosines,
402                DirectionRef::from_str(self.refer).map_err(|_| unsupported("direction"))?,
403            )),
404            MeasureValue::Position(xyz) => CasaMeasure::Position(if self.refer == "WGS84" {
405                let (longitude, latitude, height) = position_spherical(xyz);
406                MPosition::new_wgs84(longitude, latitude, height)
407            } else {
408                MPosition::new_itrf(xyz[0], xyz[1], xyz[2])
409            }),
410            MeasureValue::Frequency(hz) => CasaMeasure::Frequency(MFrequency::new(
411                hz,
412                FrequencyRef::from_str(self.refer).map_err(|_| unsupported("frequency"))?,
413            )),
414        })
415    }

417    /// A casa-types measure in casacore's form.
418    pub(crate) fn from_casa(measure: &CasaMeasure) -> Self {
419        match measure {
420            CasaMeasure::Epoch(epoch) => Self {
421                refer: EPOCH_CODES.resolve_or_default(epoch.refer().casacore_name()),
422                value: MeasureValue::Epoch(epoch.value()),
423            },
424            CasaMeasure::Direction(direction) => Self {
425                refer: DIRECTION_CODES.resolve_or_default(direction.refer().as_str()),
426                value: MeasureValue::Direction(direction.cosines()),
427            },
428            CasaMeasure::Position(position) => match position.refer() {
429                PositionRef::ITRF => Self {
430                    refer: "ITRF",
431                    value: MeasureValue::Position(position.values()),
432                },
433                PositionRef::WGS84 => {
434                    let [longitude, latitude, height] = position.values();
435                    Self {
436                        refer: "WGS84",
437                        value: MeasureValue::Position(position_vector(longitude, latitude, height)),
438                    }
439                }
440            },
441            CasaMeasure::Frequency(frequency) => Self {
442                refer: FREQUENCY_CODES.resolve_or_default(frequency.refer().as_str()),
443                value: MeasureValue::Frequency(frequency.hz()),
444            },
445        }
446    }
447}

449/// The `m0`, `m1`, `m2` quantities of a measure record, up to the first
450/// missing one. A quantity record may hold one value, or an array of one.
451fn record_values(record: &BTreeMap<String, Variant>) -> Result<Vec<Quantity>, HolderError> {
452    let mut values = Vec::new();
453    for key in ["m0", "m1", "m2"] {
454        let Some(Variant::Record(value)) = record.get(key) else {
455            break;
456        };
457        if let Some(Variant::Numbers(numbers)) = value.get("value")
458            && numbers.len() > 1
459        {
460            return Err(HolderError::Invalid(
461                "multi-valued measures are not yet implemented in casa-rs".to_string(),
462            ));
463        }
464        values.push(quantity_record(value).map_err(HolderError::Invalid)?);
465    }
466    Ok(values)
467}

469fn has(quantity: &Quantity, dimension: Dimension) -> bool {
470    quantity.unit().val().dim == UnitDim::basis(dimension)
471}

473/// `MVEpoch::putValue`: every value a time, summed.
474fn epoch_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
475    if !values.iter().all(|value| has(value, Dimension::Time)) {
476        return Err(illegal_quantity());
477    }
478    let (mut day, mut fraction) = (0.0, 0.0);
479    for value in values {
480        let days = days_of(value);
481        let whole = days.floor();
482        day += whole;
483        fraction += days - whole;
484    }
485    Ok(MeasureValue::Epoch(MjdHighPrec::new(day, fraction)))
486}

488/// A time in days as casacore converts it (`Quantity::get("d")`).
489pub(crate) fn days_of(time: &Quantity) -> f64 {
490    time.value() * (time.unit().val().factor / 86_400.0)
491}

493/// `MVDirection::putValue`: three direction cosines, or up to two angles
494/// (none is the pole, one a longitude on the equator); normalised.
495fn direction_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
496    let mut xyz = if values.len() == 3
497        && values
498            .iter()
499            .all(|value| value.unit().val().dim.is_dimensionless())
500    {
501        [values[0].value(), values[1].value(), values[2].value()]
502    } else {
503        if values.len() > 3 || !values.iter().all(|value| has(value, Dimension::Angle)) {
504            return Err(illegal_quantity());
505        }
506        let angles = values
507            .iter()
508            .map(Quantity::get_si_value)
509            .collect::<Vec<_>>();
510        match angles.as_slice() {
511            [] => [0.0, 0.0, 1.0],
512            [longitude] => [longitude.cos(), longitude.sin(), 0.0],
513            [longitude, latitude, ..] => direction_cosines(*longitude, *latitude),
514        }
515    };
516    let length = norm(xyz);
517    if length != 0.0 && length != 1.0 {
518        xyz = xyz.map(|value| value / length);
519    }
520    Ok(MeasureValue::Direction(xyz))
521}

523/// `MVPosition::putValue`: three lengths (x, y, z), a length then two
524/// angles, or two angles then a length.
525fn position_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
526    let [first, second, third] = values else {
527        return Err(illegal_quantity());
528    };
529    let (length, angle) = (Dimension::Length, Dimension::Angle);
530    let xyz = if has(first, length) {
531        if has(second, length) && has(third, length) {
532            [
533                first.get_si_value(),
534                second.get_si_value(),
535                third.get_si_value(),
536            ]
537        } else if has(second, angle) && has(third, angle) {
538            position_vector(
539                second.get_si_value(),
540                third.get_si_value(),
541                first.get_si_value(),
542            )
543        } else {
544            return Err(illegal_quantity());
545        }
546    } else if has(third, length) && has(first, angle) && has(second, angle) {
547        position_vector(
548            first.get_si_value(),
549            second.get_si_value(),
550            third.get_si_value(),
551        )
552    } else {
553        return Err(illegal_quantity());
554    };
555    Ok(MeasureValue::Position(xyz))
556}

558/// `MVFrequency::putValue`: none is 0 Hz; one value is a frequency, a
559/// period, an angular rate, a wavelength, a wavenumber, an energy or a
560/// momentum (`MVFrequency::makeF`).
561fn frequency_value(values: &[Quantity]) -> Result<MeasureValue, HolderError> {
562    match values {
563        [] => Ok(MeasureValue::Frequency(0.0)),
564        [value] => frequency_hz(value)
565            .map(MeasureValue::Frequency)
566            .ok_or_else(illegal_quantity),
567        _ => Err(illegal_quantity()),
568    }
569}

571/// `MVFrequency::makeF`: `quantity` in Hz, or `None` for a unit that is not
572/// frequency-like.
573pub(crate) fn frequency_hz(quantity: &Quantity) -> Option<f64> {
574    let unit = quantity.unit().val();
575    let (value, factor) = (quantity.value(), unit.factor);
576    let basis = UnitDim::basis;
577    let time = basis(Dimension::Time);
578    let length = basis(Dimension::Length);
579    let energy = basis(Dimension::Mass) * length * length / time / time;
580    let momentum = basis(Dimension::Mass) * length;
581    let dim = unit.dim;
582    if dim == time {
583        Some(1.0 / factor / value)
584    } else if dim == UnitDim::NODIM / time {
585        Some(value * factor)
586    } else if dim == basis(Dimension::Angle) / time {
587        Some(value * (factor / std::f64::consts::PI / 2.0))
588    } else if dim == length {
589        Some(SPEED_OF_LIGHT_M_S / factor / value)
590    } else if dim == UnitDim::NODIM / length {
591        Some(value * (SPEED_OF_LIGHT_M_S * factor / std::f64::consts::PI / 2.0))
592    } else if dim == energy {
593        Some(value * (factor / PLANCK_J_S))
594    } else if dim == momentum {
595        Some(value * (factor * SPEED_OF_LIGHT_M_S / PLANCK_J_S))
596    } else {
597        None
598    }
599}

601fn norm(xyz: [f64; 3]) -> f64 {
602    (xyz[0] * xyz[0] + xyz[1] * xyz[1] + xyz[2] * xyz[2]).sqrt()
603}

605/// The cosines of a direction (`MVDirection(lon, lat)`).
606pub(crate) fn direction_cosines(longitude: f64, latitude: f64) -> [f64; 3] {
607    let cos_latitude = latitude.cos();
608    [
609        longitude.cos() * cos_latitude,
610        longitude.sin() * cos_latitude,
611        latitude.sin(),
612    ]
613}

615/// `MVDirection::get`: longitude in (−π, π] (0 at a pole) and latitude.
616pub(crate) fn direction_angles(cosines: [f64; 3]) -> (f64, f64) {
617    let longitude = if cosines[0] != 0.0 || cosines[1] != 0.0 {
618        cosines[1].atan2(cosines[0])
619    } else {
620        0.0
621    };
622    (longitude, cosines[2].asin())
623}

625/// `MVPosition(length, lon, lat)`: the vector, its length encoding a WGS84
626/// height as casacore encodes it.
627pub(crate) fn position_vector(longitude: f64, latitude: f64, length_m: f64) -> [f64; 3] {
628    let mut t = length_m;
629    if t < 0.0 && t > -7.0e6 {
630        t = t / 1.0e7 + HI_LIMIT_M;
631    } else if t > LO_LIMIT_M && t < HI_LIMIT_M {
632        t += 0.001;
633    }
634    let scale = if t == 0.0 { 1e-6 } else { t };
635    direction_cosines(longitude, latitude).map(|value| value * scale)
636}
637
638/// `MVPosition::get`: longitude, latitude and the (decoded) length.
639pub(crate) fn position_spherical(xyz: [f64; 3]) -> (f64, f64, f64) {
640    let mut length = norm(xyz);
641    let longitude = if xyz[0] != 0.0 || xyz[1] != 0.0 {
642        xyz[1].atan2(xyz[0])
643    } else {
644        0.0
645    };
646    let latitude = (xyz[2] / if length == 0.0 { 1.0 } else { length }).asin();
647    if length > LO_LIMIT_M && length < HI_LIMIT_M {
648        length = (length - HI_LIMIT_M) * 1e7;
649    }
650    (longitude, latitude, length)
651}
