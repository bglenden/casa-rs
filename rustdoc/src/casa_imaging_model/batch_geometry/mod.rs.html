2//! Row geometry over observation batches: what the imaging operators need
3//! of each sample, compiled once from the observation's catalogue and
4//! applied to each [`VisibilityBatch`].
5//!
6//! For each row sample (a time and baseline the caller images) and each
7//! chart, the stored UVW reprojected to the chart's phase centre with the
8//! phase shift that goes with it ([`GridFtProjection`]); the antennas'
9//! pointing directions; their parallactic angles; on AW-projection runs
10//! with pointing, the grouped pointing pixel on each chart; and the
11//! antennas' aperture classes. For each time, the channel frequencies in
12//! the output frame. Each quantity is computed as often as it changes: a
13//! reprojection per field and chart, a parallactic angle per antenna,
14//! field and time, a pointing direction per antenna and time, a frequency
15//! conversion per time.
16//!
17//! The arithmetic is the selected-observation route's, statement for
18//! statement, through the same shared pieces ([`GridFtProjection`],
19//! [`ObservationPointingLaw::resolve`](crate::ObservationPointingLaw::resolve),
20//! [`AwPointingEpochs`], `casa-types` Measures), so the two routes agree
21//! to rounding on an MSv2.
22//!
23//! AW pointing groups carry state from epoch to epoch (a time of one
24//! partition), in the order the batches arrive: partition by partition,
25//! each in time order. [`GeometryTraversal`] holds it for one traversal.
26//! AW projection images MFS only (cube AW is refused, #719), so a
27//! traversal visits every epoch.
28//!
29//! A field that follows an ephemeris ([`FieldEntry::ephemeris`]) is
30//! observed towards the body's direction from the observatory at each time,
31//! shifted by its phase centre, as CASA's imager evaluates it (casacore
32//! shifts by the true angle; the imager adds the observatory and reads the
33//! samples at the time's TDB); its reprojections, parallactic angles and
34//! frequency conversions are computed per time. A moving phase centre
35//! follows a body or source Measures names, each field's own ephemeris
36//! (`TRACKFIELD`), or an ephemeris table the caller reads. A conversion
37//! into or out of the rest frame needs the source's radial velocity, which
38//! the last two give; under any other phase centre a batch that needs one
39//! is refused.
40//!
41//! [`FieldEntry::ephemeris`]: casa_observation::FieldEntry::ephemeris

75/// Why batch geometry cannot be formed.
76#[derive(Debug, thiserror::Error)]
77pub enum BatchGeometryError {
78    /// The catalogue lacks what geometry needs, or the batch is not of the
79    /// compiled observation.
80    #[error("the observation's catalogue {0}")]
81    Catalogue(&'static str),
82    /// A fixed phase or pointing centre outside J2000.
83    #[error("a fixed centre in {0:?} (imaging geometry takes J2000)")]
84    FixedCentreFrame(DirectionFrame),
85    /// A faceted domain under the mosaic UVW law.
86    #[error("a faceted image domain under the mosaic UVW law")]
87    FacetedMosaic,
88    /// A problem without an image domain.
89    #[error("the problem has no image domain to project to")]
90    NoImageDomain,
91    /// A moving phase centre that is neither `TRACKFIELD`, nor a body or
92    /// source Measures names, nor an ephemeris table the caller gave.
93    #[error(
94        "the moving phase centre {0:?} is not TRACKFIELD, a body or source Measures names, \
95         or an ephemeris table"
96    )]
97    UnsupportedEphemeris(String),
98    /// An ephemeris table given for a phase centre that does not move with
99    /// one.
100    #[error("an ephemeris table for a phase centre that is not a table's")]
101    UnusedEphemerisTable,
102    /// A field without an ephemeris under a phase centre that follows each
103    /// field's own (`TRACKFIELD`).
104    #[error("{0:?} follows no ephemeris for TRACKFIELD to track")]
105    FieldWithoutEphemeris(FieldId),
106    /// A time outside the samples of the ephemeris a direction or velocity
107    /// is read from.
108    #[error("the ephemeris of {body} does not cover MJD {mjd}")]
109    EphemerisCoverage {
110        /// The body.
111        body: String,
112        /// The time asked for, MJD days on the scale it is read at.
113        mjd: f64,
114    },
115    /// A domain whose PSF is centred elsewhere than its model, which no
116    /// consumer reads.
117    #[error("an image domain with a PSF phase centre of its own")]
118    DistinctPsfCentre,
119    /// A batch without what geometry reads: stored UVW, time centroids for
120    /// centroid-sampled pointing, or pointing samples for an observation
121    /// pointing law.
122    #[error("the batch carries no {0}")]
123    MissingInput(&'static str),
124    /// A conversion into or out of the rest frame under a phase centre
125    /// that gives no source velocity: only `TRACKFIELD` and an ephemeris
126    /// table do.
127    #[error("rest-frame frequencies need a source velocity")]
128    RestFrameWithoutVelocity,
129    /// An antenna the instrument model gives no aperture class.
130    #[error("{0:?} has no CASA aperture class")]
131    UnclassifiedAntenna(AntennaId),
132    /// A Measures conversion failed.
133    #[error(transparent)]
134    Measures(#[from] MeasureError),
135    /// The pointing samples could not be read or evaluated.
136    #[error(transparent)]
137    Observation(#[from] casa_observation::ObservationError),
138    /// The pointing law gives no direction.
139    #[error("{antenna:?} at {time_s} s: {law}")]
140    Pointing {
141        /// The antenna.
142        antenna: AntennaId,
143        /// The query time, on the partition's time system.
144        time_s: f64,
145        /// Why.
146        law: PointingLawError,
147    },
148    /// AW pointing pixels could not be formed.
149    #[error(transparent)]
150    AwPointing(#[from] AwPointingError),
151}

201/// The phase-tracking centre the problem's law resolves to.
202enum PhaseCentre {
203    /// Each row's own field direction.
204    Observation,
205    /// A fixed J2000 direction.
206    Fixed(SkyDirection),
207    /// A body or source Measures names.
208    Named(String),
209    /// Each field's own ephemeris, without the field's offset from it
210    /// (`TRACKFIELD`).
211    FieldEphemerides,
212    /// The body of an ephemeris table.
213    Table(Arc<FieldEphemeris>),
214}
215
216impl fmt::Debug for PhaseCentre {
217    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
218        match self {
219            Self::Observation => formatter.write_str("Observation"),
220            Self::Fixed(direction) => formatter.debug_tuple("Fixed").field(direction).finish(),
221            Self::Named(name) => formatter.debug_tuple("Named").field(name).finish(),
222            Self::FieldEphemerides => formatter.write_str("FieldEphemerides"),
223            Self::Table(table) => formatter.debug_tuple("Table").field(&table.name()).finish(),
224        }
225    }
226}

228impl PhaseCentre {
229    /// `law`'s centre: `table`, an ephemeris table the caller read, is the
230    /// body a moving law other than `TRACKFIELD` names; without one, that
231    /// law names a body or source Measures has a model of.
232    fn resolve(
233        law: &PhaseCentreLaw,
234        table: Option<Arc<FieldEphemeris>>,
235        frames: &Frames,
236    ) -> Result<Self, BatchGeometryError> {
237        Ok(match (law, table) {
238            (PhaseCentreLaw::Ephemeris(name), Some(table)) if name != TRACKFIELD => {
239                Self::Table(table)
240            }
241            (_, Some(_)) => return Err(BatchGeometryError::UnusedEphemerisTable),
242            (PhaseCentreLaw::Observation, None) => Self::Observation,
243            (PhaseCentreLaw::Fixed(direction), None) => Self::Fixed(*direction),
244            (PhaseCentreLaw::Ephemeris(name), None) if name == TRACKFIELD => Self::FieldEphemerides,
245            (PhaseCentreLaw::Ephemeris(name), None) => {
246                frames.check_named(name).map_err(|error| match error {
247                    MeasureError::UnknownSourceName { .. } => {
248                        BatchGeometryError::UnsupportedEphemeris(name.clone())
249                    }
250                    error => BatchGeometryError::Measures(error),
251                })?;
252                Self::Named(name.clone())
253            }
254        })
255    }
256
257    const fn moves(&self) -> bool {
258        matches!(
259            self,
260            Self::Named(_) | Self::FieldEphemerides | Self::Table(_)
261        )
262    }
263}

286/// The geometry of one batch, in buffers reused from batch to batch.
287/// Samples that are not rows hold no geometry (NaN), and neither do the
288/// frequencies of a time without rows.
289#[derive(Debug, Default)]
290pub struct BatchGeometry {
291    charts: usize,
292    channels: usize,
293    projections: Vec<ChartProjection>,
294    pointing: Vec<SelectedPointingDirections>,
295    parallactic: Vec<[f64; 2]>,
296    aw_pixels: Vec<[f64; 2]>,
297    responses: Vec<Option<SelectedAntennaResponses>>,
298    frequencies_hz: Vec<f64>,
299}
300
301impl BatchGeometry {
302    /// Sample `sample`'s reprojection to chart `chart`.
303    #[must_use]
304    pub fn projection(&self, sample: usize, chart: usize) -> ChartProjection {
305        self.projections[sample * self.charts + chart]
306    }
307
308    /// Sample `sample`'s antennas' pointing directions (J2000).
309    #[must_use]
310    pub fn pointing(&self, sample: usize) -> SelectedPointingDirections {
311        self.pointing[sample]
312    }
313
314    /// Sample `sample`'s antennas' parallactic angles as the polarization
315    /// operator takes them (negated), when the problem needs them.
316    #[must_use]
317    pub fn parallactic_angles(&self, sample: usize) -> Option<[f64; 2]> {
318        self.parallactic.get(sample).copied()
319    }
320
321    /// Sample `sample`'s AW pointing pixel on chart `chart`, on an AW run
322    /// with pointing.
323    #[must_use]
324    pub fn aw_pointing_pixel(&self, sample: usize, chart: usize) -> Option<[f64; 2]> {
325        self.aw_pixels.get(sample * self.charts + chart).copied()
326    }
327
328    /// Sample `sample`'s antennas' aperture classes, when the instrument
329    /// model assigns them.
330    #[must_use]
331    pub fn antenna_responses(&self, sample: usize) -> Option<SelectedAntennaResponses> {
332        self.responses.get(sample).copied().flatten()
333    }
334
335    /// Time `time`'s channel frequencies in the output frame.
336    #[must_use]
337    pub fn frequencies_hz(&self, time: usize) -> &[f64] {
338        &self.frequencies_hz[time * self.channels..(time + 1) * self.channels]
339    }
340}

342impl BatchGeometry {
343    /// Bytes the buffers hold, from their capacities: at most
344    /// [`GeometryPlan::geometry_bytes`] of the largest batch applied.
345    #[must_use]
346    pub fn capacity_bytes(&self) -> usize {
347        self.projections.capacity() * size_of::<ChartProjection>()
348            + self.pointing.capacity() * size_of::<SelectedPointingDirections>()
349            + self.parallactic.capacity() * size_of::<[f64; 2]>()
350            + self.aw_pixels.capacity() * size_of::<[f64; 2]>()
351            + self.responses.capacity() * size_of::<Option<SelectedAntennaResponses>>()
352            + self.frequencies_hz.capacity() * size_of::<f64>()
353    }
354}

356/// `buffer` emptied and filled with `len` copies of `value`, growing to
357/// exactly `len` when it must grow.
358fn refill<T: Clone>(buffer: &mut Vec<T>, len: usize, value: T) {
359    buffer.clear();
360    buffer.reserve_exact(len);
361    buffer.resize(len, value);
362}

364impl GeometryPlan {
365    /// The geometry of `problem` on the observation `catalogue` describes,
366    /// converting with `measures`. `phase_table` is the ephemeris table a
367    /// moving phase centre names, when it names one rather than a body
368    /// (CASA's `phasecenter` takes both): the caller reads it, since a law
369    /// names a table and a body alike.
370    ///
371    /// # Errors
372    ///
373    /// [`BatchGeometryError`] for a catalogue without what geometry needs,
374    /// a centre law or domain it does not take (among them a moving centre
375    /// that is not `TRACKFIELD` and that Measures does not name or has no
376    /// model of, without a table), a table for a centre that does not
377    /// move, or a failed conversion.
378    pub fn compile(
379        problem: &CompiledProblem,
380        catalogue: &Arc<Catalogue>,
381        measures: Arc<dyn MeasuresProvider>,
382        phase_table: Option<Arc<FieldEphemeris>>,
383    ) -> Result<Self, BatchGeometryError> {
384        let frames = Frames::new(catalogue, measures)?;
385        let centres = problem.geometry().centres();
386        for fixed in [
387            match centres.phase_tracking() {
388                PhaseCentreLaw::Fixed(direction) => Some(*direction),
389                _ => None,
390            },
391            match centres.pointing() {
392                PointingCentreLaw::Fixed(direction) => Some(*direction),
393                _ => None,
394            },
395        ]
396        .into_iter()
397        .flatten()
398        {
399            if fixed.frame() != DirectionFrame::J2000 {
400                return Err(BatchGeometryError::FixedCentreFrame(fixed.frame()));
401            }
402        }
403        let domains = problem.geometry().domains();
404        if domains
405            .iter()
406            .any(|domain| domain.psf_phase_centre() != domain.model_phase_centre())
407        {
408            return Err(BatchGeometryError::DistinctPsfCentre);
409        }
410        let anchor = domains
411            .first()
412            .ok_or(BatchGeometryError::NoImageDomain)?
413            .model_phase_centre();
414        let phase = PhaseCentre::resolve(centres.phase_tracking(), phase_table, &frames)?;
415        let (charts, mosaic) = projection::charts(problem)?;
416        let responses = match problem.science().instrument_model() {
417            None | Some(InstrumentModel::CasaEvlaWidebandAwV1) => Responses::None,
418            Some(InstrumentModel::CasaAca7mInterferometricDirectPbV1) => Responses::Aca7m,
419            Some(InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1) => {
420                let classes = frames.alma_aca_classes(catalogue);
421                let envelope = AntennaResponseClass::family_envelope(&classes);
422                Responses::Heterogeneous { classes, envelope }
423            }
424        };
425        Ok(Self {
426            catalogue: Arc::clone(catalogue),
427            frames,
428            phase,
429            anchor,
430            pointing: centres.pointing().clone(),
431            charts,
432            mosaic,
433            output_frame: problem.geometry().spectral().output_frame().frequency_ref(),
434            parallactic: problem.requires_parallactic_angles(),
435            responses,
436            aw: AwPointingGrouping::new(problem)?,
437        })
438    }

464    /// At most the bytes the plan has come to hold beyond its compiled
465    /// state: the parallactic angles of every (time, time scale, field) a
466    /// batch with rows has asked for, one per antenna, kept for the
467    /// observation's other partitions, and the table that finds them. Once
468    /// a traversal has read every batch it holds, for each distinct (time,
469    /// field) with rows, 8 bytes an antenna and at most about 80 bytes of

476    }

491    /// The bytes a [`BatchGeometry`] holds for a batch of `samples` samples
492    /// over `times` times of `channels` channels
493    /// ([`BatchGeometry::capacity_bytes`] after [`Self::apply`] on the
494    /// largest such batch).
495    #[must_use]
496    pub fn geometry_bytes(&self, samples: usize, times: usize, channels: usize) -> usize {
497        let charts = self.charts.len();
498        let mut per_sample =
499            charts * size_of::<ChartProjection>() + size_of::<SelectedPointingDirections>();
500        if self.parallactic {
501            per_sample += size_of::<[f64; 2]>();
502        }
503        if self.aw.is_some() {
504            per_sample += charts * size_of::<[f64; 2]>();
505        }
506        if !matches!(self.responses, Responses::None) {
507            per_sample += size_of::<Option<SelectedAntennaResponses>>();
508        }
509        samples * per_sample + times * channels * size_of::<f64>()
510    }

512    /// CASA's aperture classes of the observation's antennas, sorted and
513    /// without repeats: the dishes a heterogeneous mosaic's kernel set
514    /// keys on (`HetArrayConvFunc::findAntennaSizes`), as the
515    /// selected-observation route lists them.
516    #[must_use]
517    pub fn aperture_classes(&self) -> Vec<AntennaResponseClass> {
518        let mut classes = self
519            .frames
520            .alma_aca_classes(&self.catalogue)
521            .into_iter()
522            .flatten()
523            .collect::<Vec<_>>();
524        classes.sort_unstable();
525        classes.dedup();
526        classes
527    }

529    /// The factor that takes a frequency in `frame`, observed towards
530    /// field `field` at `time_mjd_s` on `scale`, to the output frame:
531    /// [`Self::apply`] gives each channel its native frequency times this
532    /// factor, to rounding. A source sizes a wave's channel window from it
533    /// without reading a batch.
534    ///
535    /// # Errors
536    ///
537    /// As [`Self::apply`]'s frequency conversion: a conversion into or out
538    /// of the rest frame without a source velocity, a field's ephemeris
539    /// that does not cover the time, or a failed conversion.
540    pub fn frequency_factor(
541        &self,
542        frame: FrequencyRef,
543        field: FieldId,
544        time_mjd_s: f64,
545        scale: EpochRef,
546    ) -> Result<f64, BatchGeometryError> {
547        let direction = self.frames.field_at(field, time_mjd_s, scale)?;
548        Ok(self
549            .frequency_conversion((time_mjd_s, scale), field, &direction, frame)?
550            .convert_hz(1.0))
551    }

553    /// The geometry of `batch`'s row samples (`rows`, one flag a sample)
554    /// into `out`; `pointing` holds the batch partition's samples over its
555    /// times when [`Self::reads_pointing`].
556    ///
557    /// # Errors
558    ///
559    /// [`BatchGeometryError`] for a batch of another observation or without
560    /// what geometry reads, frequencies to convert into or out of the rest
561    /// frame, a failed conversion, or a pointing law that gives no
562    /// direction.
563    ///
564    /// # Panics
565    ///
566    /// If `rows` does not have one flag for each sample.
567    pub fn apply(
568        &self,
569        batch: &VisibilityBatch<'_>,
570        rows: &[bool],
571        pointing: Option<&PointingSeries>,
572        traversal: &mut GeometryTraversal,
573        out: &mut BatchGeometry,
574    ) -> Result<(), BatchGeometryError> {
575        assert_eq!(rows.len(), batch.samples(), "one row flag a sample");
576        if !Arc::ptr_eq(batch.catalogue, &self.catalogue) {
577            return Err(BatchGeometryError::Catalogue(
578                "the batch belongs to is not the compiled one",
579            ));
580        }
581        let uvw = batch
582            .group
583            .uvw
584            .ok_or(BatchGeometryError::MissingInput("UVW"))?;
585        self.reset(batch, out);
586        let scale = batch.time.reference();
587        let samples_per_time = batch.samples_per_time();
588        let mjd_origin = MjdHighPrec::new(0.0, 0.0);
589        let mut reprojections: Option<(FieldId, Vec<Option<GridFtProjection>>)> = None;
590        let mut directions = TimeDirections::default();
591        for time in 0..batch.times() {
592            let samples = time * samples_per_time..(time + 1) * samples_per_time;
593            if !rows[samples.clone()].iter().any(|row| *row) {
594                continue;
595            }
596            let time_mjd_s = batch.time.seconds_since(time, mjd_origin);
597            let field = batch.field[time];
598            let field_direction = self.frames.field_at(field, time_mjd_s, scale)?;
599            let (longitude, latitude) = field_direction.as_angles();
600            let observation = SkyDirection::new(DirectionFrame::J2000, longitude, latitude);
601            let context = TimeContext {
602                batch,
603                time,
604                time_mjd_s,
605                field,
606                phase: self.phase_direction(time_mjd_s, scale, field, observation)?,
607                observation,
608                rows,
609            };
610            if self.time_varying()
611                || self.frames.moves(field)
612                || reprojections
613                    .as_ref()
614                    .is_none_or(|(known, _)| *known != field)
615            {
616                reprojections = Some((field, self.reprojections(&context, &field_direction)?));
617            }
618            let machines = &reprojections.as_ref().expect("reprojections formed").1;
619            for sample in context.row_samples() {
620                let raw = uvw.metres_at(sample);
621                for (chart, machine) in machines.iter().enumerate() {
622                    let (uvw_m, phase_shift_m) =
623                        machine.map_or((raw, 0.0), |machine| machine.apply(raw));
624                    out.projections[sample * out.charts + chart] = ChartProjection {
625                        uvw_m,
626                        phase_shift_m,
627                    };
628                }
629            }
630            self.antenna_quantities(&context, &field_direction, out)?;
631            directions.clear();
632            self.pointing_directions(&context, pointing, &mut directions, out)?;
633            if let Some(grouping) = &self.aw {
634                let series =
635                    pointing.ok_or(BatchGeometryError::MissingInput("pointing samples"))?;
636                self.aw_pixels(
637                    &context,
638                    series,
639                    grouping,
640                    &mut directions,
641                    &mut traversal.aw,
642                    out,
643                )?;
644            }
645            let conversion = self.frequency_conversion(
646                (time_mjd_s, scale),
647                field,
648                &field_direction,
649                batch.frequency.reference(),
650            )?;
651            let channels = out.channels;
652            for (slot, frequency) in out.frequencies_hz[time * channels..(time + 1) * channels]
653                .iter_mut()
654                .zip(batch.frequency.hz())
655            {
656                *slot = conversion.convert_hz(*frequency);
657            }
658        }
659        Ok(())
660    }

662    /// Size `out` for `batch`, every sample without geometry.
663    /// Each buffer is reserved exactly, so it holds what the largest batch
664    /// needs and no more ([`Self::geometry_bytes`]).
665    fn reset(&self, batch: &VisibilityBatch<'_>, out: &mut BatchGeometry) {
666        let samples = batch.samples();
667        out.charts = self.charts.len();
668        out.channels = batch.channel_count();
669        refill(
670            &mut out.projections,
671            samples * out.charts,
672            ChartProjection::ABSENT,
673        );
674        let nowhere = SkyDirection::new(DirectionFrame::J2000, f64::NAN, f64::NAN);
675        refill(
676            &mut out.pointing,
677            samples,
678            SelectedPointingDirections {
679                antenna1: nowhere,
680                antenna2: nowhere,
681            },
682        );
683        let parallactic = if self.parallactic { samples } else { 0 };
684        refill(&mut out.parallactic, parallactic, [f64::NAN; 2]);
685        let aw = if self.aw.is_some() {
686            samples * out.charts
687        } else {
688            0
689        };
690        refill(&mut out.aw_pixels, aw, [f64::NAN; 2]);
691        let responses = if matches!(self.responses, Responses::None) {
692            0
693        } else {
694            samples
695        };
696        refill(&mut out.responses, responses, None);
697        refill(
698            &mut out.frequencies_hz,
699            batch.times() * out.channels,
700            f64::NAN,
701        );
702    }

704    /// Whether the charts' targets move with time: a moving phase centre,
705    /// or a chart centre outside J2000.
706    fn time_varying(&self) -> bool {
707        self.phase.moves()
708            || self
709                .charts
710                .iter()
711                .any(|chart| chart.centre.frame() != DirectionFrame::J2000)
712    }

714    /// Each chart's reprojection of the time's rows, observed towards the
715    /// field's J2000 phase centre `field` at the time.
716    fn reprojections(
717        &self,
718        context: &TimeContext<'_, '_>,
719        field: &MDirection,
720    ) -> Result<Vec<Option<GridFtProjection>>, BatchGeometryError> {
721        let scale = context.batch.time.reference();
722        let shift = self.moving_shift(context.time_mjd_s, scale, context.phase)?;
723        self.charts
724            .iter()
725            .map(|chart| {
726                let mut target = self.frames.j2000(
727                    context.time_mjd_s,
728                    scale,
729                    [chart.centre.longitude_rad(), chart.centre.latitude_rad()],
730                    chart.centre.frame().direction_ref(),
731                )?;
732                if let Some([longitude, latitude]) = shift {
733                    target[0] += longitude;
734                    target[1] += latitude;
735                }
736                Ok(projection::reprojection(
737                    context.observation,
738                    field,
739                    target,
740                    chart.faceted,
741                    self.mosaic,
742                ))
743            })
744            .collect()
745    }

747    /// The parallactic angles and aperture classes of the time's rows,
748    /// observed towards `field`, the field's J2000 phase centre at the time.
749    fn antenna_quantities(
750        &self,
751        context: &TimeContext<'_, '_>,
752        field: &MDirection,
753        out: &mut BatchGeometry,
754    ) -> Result<(), BatchGeometryError> {
755        let per_time = context.batch.samples_per_time();
756        let angles = if self.parallactic {
757            Some(self.frames.parallactic_angles(
758                (context.time_mjd_s, context.batch.time.reference()),
759                context.field,
760                field,
761                &context.row_antennas(),
762            )?)
763        } else {
764            None
765        };
766        for sample in context.row_samples() {
767            let baseline = context.batch.baselines[sample % per_time];
768            if let Some(angles) = &angles {
769                out.parallactic[sample] = [
770                    -angles[baseline.antenna1.0 as usize],
771                    -angles[baseline.antenna2.0 as usize],
772                ];
773            }
774            if let Some(responses) = out.responses.get_mut(sample) {
775                *responses = self.antenna_responses(baseline.antenna1, baseline.antenna2)?;
776            }
777        }
778        Ok(())
779    }

781    /// The phase-tracking direction at `time_mjd_s` of rows of `field`
782    /// observed in `observation`.
783    fn phase_direction(
784        &self,
785        time_mjd_s: f64,
786        scale: EpochRef,
787        field: FieldId,
788        observation: SkyDirection,
789    ) -> Result<SkyDirection, BatchGeometryError> {
790        let moving = match &self.phase {
791            PhaseCentre::Observation => return Ok(observation),
792            PhaseCentre::Fixed(direction) => return Ok(*direction),
793            PhaseCentre::Named(name) => self.frames.named(time_mjd_s, scale, name)?,
794            PhaseCentre::FieldEphemerides => self.frames.ephemeris_direction(
795                self.frames
796                    .field_ephemeris(field)
797                    .ok_or(BatchGeometryError::FieldWithoutEphemeris(field))?,
798                time_mjd_s,
799                scale,
800            )?,
801            PhaseCentre::Table(table) => {
802                self.frames.ephemeris_direction(table, time_mjd_s, scale)?
803            }
804        };
805        let (longitude, latitude) = moving.as_angles();
806        Ok(SkyDirection::new(
807            DirectionFrame::J2000,
808            longitude,
809            latitude,
810        ))
811    }

813    /// The source's radial velocity at `time_mjd_s` for rows of `field`,
814    /// under a phase centre that follows an ephemeris: `TRACKFIELD`'s
815    /// field ephemeris or the table's. `None` under any other.
816    fn source_velocity(
817        &self,
818        (time_mjd_s, scale): (f64, EpochRef),
819        field: FieldId,
820    ) -> Result<Option<MRadialVelocity>, BatchGeometryError> {
821        let ephemeris = match &self.phase {
822            PhaseCentre::FieldEphemerides => self
823                .frames
824                .field_ephemeris(field)
825                .ok_or(BatchGeometryError::FieldWithoutEphemeris(field))?,
826            PhaseCentre::Table(table) => table,
827            PhaseCentre::Observation | PhaseCentre::Fixed(_) | PhaseCentre::Named(_) => {
828                return Ok(None);
829            }
830        };
831        Ok(Some(
832            self.frames
833                .ephemeris_velocity(ephemeris, time_mjd_s, scale)?,
834        ))
835    }

837    /// Under a moving phase centre, its offset at `time_mjd_s` from the
838    /// main domain's model phase centre, which every chart's target
839    /// follows.
840    fn moving_shift(
841        &self,
842        time_mjd_s: f64,
843        scale: EpochRef,
844        phase: SkyDirection,
845    ) -> Result<Option<[f64; 2]>, BatchGeometryError> {
846        if !self.phase.moves() {
847            return Ok(None);
848        }
849        let anchor = self.frames.j2000(
850            time_mjd_s,
851            scale,
852            [self.anchor.longitude_rad(), self.anchor.latitude_rad()],
853            self.anchor.frame().direction_ref(),
854        )?;
855        Ok(Some([
856            phase.longitude_rad() - anchor[0],
857            phase.latitude_rad() - anchor[1],
858        ]))
859    }

861    /// The aperture classes of a baseline's antennas and their family.
862    fn antenna_responses(
863        &self,
864        antenna1: AntennaId,
865        antenna2: AntennaId,
866    ) -> Result<Option<SelectedAntennaResponses>, BatchGeometryError> {
867        Ok(match &self.responses {
868            Responses::None => None,
869            Responses::Aca7m => Some(SelectedAntennaResponses {
870                antenna1: AntennaResponseClass::CasaAca7m,
871                antenna2: AntennaResponseClass::CasaAca7m,
872                family_envelope: AntennaResponseClass::CasaAca7m,
873            }),
874            Responses::Heterogeneous { classes, envelope } => {
875                let class = |antenna: AntennaId| {
876                    classes
877                        .get(antenna.0 as usize)
878                        .copied()
879                        .flatten()
880                        .ok_or(BatchGeometryError::UnclassifiedAntenna(antenna))
881                };
882                Some(SelectedAntennaResponses {
883                    antenna1: class(antenna1)?,
884                    antenna2: class(antenna2)?,
885                    family_envelope: envelope
886                        .ok_or(BatchGeometryError::UnclassifiedAntenna(antenna1))?,
887                })
888            }
889        })
890    }

892    /// The conversion of frequencies in `source` at `time_mjd_s` to the
893    /// output frame, in the observatory's frame towards `direction`, field
894    /// `field`'s direction at the time (CASA's visibility iterator converts
895    /// towards the field's own direction, whatever the imaging phase
896    /// centre), with the source's radial velocity when the conversion is
897    /// into or out of the rest frame ([`Self::source_velocity`]).
898    fn frequency_conversion(
899        &self,
900        (time_mjd_s, scale): (f64, EpochRef),
901        field: FieldId,
902        direction: &MDirection,
903        source: FrequencyRef,
904    ) -> Result<PreparedFrequencyConversion, BatchGeometryError> {
905        if source == self.output_frame {
906            return Ok(PreparedFrequencyConversion::new(
907                source, source, None, None,
908            )?);
909        }
910        let mut frame = self
911            .frames
912            .observatory_at(time_mjd_s, scale)
913            .with_direction(direction.clone());
914        if source == FrequencyRef::REST || self.output_frame == FrequencyRef::REST {
915            let velocity = self
916                .source_velocity((time_mjd_s, scale), field)?
917                .ok_or(BatchGeometryError::RestFrameWithoutVelocity)?;
918            frame = frame.with_radial_velocity(velocity);
919        }
920        Ok(PreparedFrequencyConversion::new(
921            source,
922            self.output_frame,
923            Some(&frame),
924            Some(&frame),
925        )?)
926    }
