7/// The channels of several spectral windows, each given as its
8/// `(CHAN_FREQ, CHAN_WIDTH)` in window-id order, merged into one axis
9/// ascending in frequency with positive widths, as
10/// `MSTransformRegridder::combineSpwsCore` merges them for
11/// `SynthesisParamsImage::buildCoordinateSystemCore` (casatools
12/// `mstransform/MSTransform/MSTransformRegridder.cc`, CASA 6.7.5, ~line 72).
13///
14/// The windows are taken in order of their lowest channel frequency, each
15/// ascending. A window that begins above the merged channels' upper edge is
16/// appended and one that ends below their lower edge prepended. Otherwise
17/// its channels below the merged axis are prepended, the lowest of them
18/// overlapping the axis's first channel becomes one channel spanning both,
19/// the merged channels keep their frequencies, and its channels above the
20/// axis are appended: the first of these, when it overlaps the axis's last
21/// channel by more than 1% of its width, widens the channel at the index of
22/// the axis's last channel instead (an index into the merged list, so
23/// after prepended channels it is not the axis's last one, as in CASA), and
24/// a smaller overlap leaves it a narrower channel from that channel's upper
25/// edge. Windows that only touch, such as adjacent windows of equal widths,
26/// are simply concatenated. The caller refuses windows in different frames,
27/// which CASA cannot combine.
28///
29/// The eight layouts of this module's `touching_windows_…` and
30/// `overlapping_windows_…` tests (the latter's numbers taken as 1e8 Hz)
31/// were combined by CASA 6.7.6.14 too, by `mstransform(combinespws=True)`,
32/// whose log states the combined window's channel count, first and last
33/// centres and last width before it regrids. Those agree on each: the
34/// overlaps below and above the axis, the 1% rule both ways, descending
35/// windows and the sort by lowest frequency. CASA states no interior
36/// channel, so a widened channel's centre and width follow from the count
37/// and the C++ alone.
38///
39/// # Errors
40///
41/// Returns an error when no window is given, or a window is empty or its
42/// frequencies and widths differ in length.
43pub fn combine_spectral_windows(windows: &[(&[f64], &[f64])]) -> MsResult<(Vec<f64>, Vec<f64>)> {
44    Ok(combine_windows(windows, false)?
45        .into_iter()
46        .map(|channel| (channel.frequency, channel.width))
47        .unzip())
48}

50/// One input channel contributing its fractional bandwidth to a combined channel.
51#[derive(Clone, Copy, Debug, PartialEq)]
52pub struct WindowContribution {
53    pub window: usize,
54    pub channel: usize,
55    pub fraction: f64,
56}
57
58/// A channel on the irregular grid before the task interpolates onto its final grid.
59#[derive(Clone)]
60pub struct CombinedChannel {
61    /// Channel center frequency in hertz.
62    pub frequency: f64,
63    /// Channel bandwidth in hertz.
64    pub width: f64,
65    /// Contributing input channels and their fractional bandwidths.
66    pub sources: Vec<WindowContribution>,
67}
68
69impl CombinedChannel {
70    fn lower(&self) -> f64 {
71        self.frequency - self.width / 2.0
72    }
73
74    fn upper(&self) -> f64 {
75        self.frequency + self.width / 2.0
76    }
77
78    fn set_edges(&mut self, lower: f64, upper: f64) {
79        self.width = upper - lower;
80        self.frequency = (lower + upper) / 2.0;
81    }
82}

84/// Merge spectral grids into ascending frequency order, optionally retaining input-channel contributions.
85pub fn combine_windows(
86    windows: &[(&[f64], &[f64])],
87    track_sources: bool,
88) -> MsResult<Vec<CombinedChannel>> {
89    let mut ascending = windows
90        .iter()
91        .enumerate()
92        .map(|(index, &(frequencies, widths))| {
93            ascending_window(index, frequencies, widths, track_sources)
94        })
95        .collect::<MsResult<Vec<_>>>()?;
96    // CASA sorts by the lowest channel frequency; the stable sort keeps
97    // window-id order for windows starting together.
98    ascending.sort_by(|a, b| a[0].frequency.total_cmp(&b[0].frequency));
99    let mut windows = ascending.into_iter();
100    let mut axis = windows.next().ok_or_else(|| {
101        MsError::VersionError("combining spectral windows needs at least one".to_string())
102    })?;
103    for window in windows {
104        axis = merge_window(axis, window, track_sources);
105    }
106    Ok(axis)
107}

109/// A window's channels ascending in frequency, with positive widths.
110fn ascending_window(
111    window: usize,
112    frequencies_hz: &[f64],
113    widths_hz: &[f64],
114    track_sources: bool,
115) -> MsResult<Vec<CombinedChannel>> {
116    if frequencies_hz.is_empty() || frequencies_hz.len() != widths_hz.len() {
117        return Err(MsError::VersionError(format!(
118            "combining spectral windows needs matching non-empty frequency/width arrays, got {} \
119             and {}",
120            frequencies_hz.len(),
121            widths_hz.len()
122        )));
123    }
124    let mut channels = frequencies_hz
125        .iter()
126        .zip(widths_hz)
127        .enumerate()
128        .map(|(channel, (&frequency, &width))| CombinedChannel {
129            frequency,
130            width: width.abs(),
131            sources: if track_sources {
132                vec![WindowContribution {
133                    window,
134                    channel,
135                    fraction: 1.0,
136                }]
137            } else {
138                Vec::new()
139            },
140        })
141        .collect::<Vec<_>>();
142    if frequencies_hz[0] > frequencies_hz[frequencies_hz.len() - 1] {
143        channels.reverse();
144    }
145    Ok(channels)
146}

148/// One step of `combineSpwsCore`'s loop: window `i` merged into the axis.
149fn merge_window(
150    mut axis: Vec<CombinedChannel>,
151    mut window: Vec<CombinedChannel>,
152    track_sources: bool,
153) -> Vec<CombinedChannel> {
154    let last_axis = axis.len() - 1;
155    let last_window = window.len() - 1;
156    let upper_axis = axis[last_axis].upper();
157    let lower_axis = axis[last_axis].lower();
158    if upper_axis < window[0].lower() {
159        axis.append(&mut window);
160        return axis;
161    }
162    if axis[0].lower() > window[last_window].upper() {
163        window.append(&mut axis);
164        return window;
165    }
166    let mut merged = Vec::with_capacity(axis.len() + window.len());
167    let mut axis_start = 0;
168    // The window starts below the axis (CASA compares with its last
169    // channel's lower edge).
170    if window[0].lower() < lower_axis {
171        for incoming in &window {
172            if axis[0].lower() < incoming.upper() {
173                if incoming.lower() < axis[0].lower() {
174                    let mut wide = incoming.clone();
175                    wide.width = axis[0].upper() - incoming.lower();
176                    wide.frequency = incoming.lower() + wide.width / 2.0;
177                    wide.sources.append(&mut axis[0].sources);
178                    merged.push(wide);
179                    axis_start = 1;
180                }
181                break;
182            }
183            merged.push(incoming.clone());
184        }
185    }
186    if track_sources {
187        let half_max_width = window.iter().map(|c| c.width).fold(0.0, f64::max) / 2.0;
188        for current in &mut axis[axis_start..] {
189            let low = current.lower();
190            let high = current.upper();
191            let first = window.partition_point(|c| c.frequency <= low - half_max_width);
192            for incoming in window[first..]
193                .iter()
194                .take_while(|c| c.frequency < high + half_max_width)
195            {
196                let overlap = high.min(incoming.upper()) - low.max(incoming.lower());
197                if overlap > 0.0 {
198                    let mut source = incoming.sources[0];
199                    source.fraction = if low <= incoming.lower() && incoming.upper() <= high {
200                        1.0
201                    } else {
202                        (overlap / incoming.width).min(1.0)
203                    };
204                    current.sources.push(source);
205                }
206            }
207        }
208    }
209    merged.extend(axis.into_iter().skip(axis_start));
210    if upper_axis < window[last_window].upper() {
211        // The overlap case holds a window channel starting at or below the
212        // axis's upper edge.
213        let mut k = window
214            .iter()
215            .rposition(|incoming| incoming.lower() <= upper_axis)
216            .expect("an overlapping window starts at or below the axis's upper edge");
217        let incoming = &window[k];
218        let lower_k = incoming.lower();
219        let upper_k = incoming.upper();
220        if lower_k < upper_axis && upper_axis < upper_k {
221            if (upper_axis - lower_k) / incoming.width > 0.01 {
222                let widened = &mut merged[last_axis];
223                widened.set_edges(lower_axis, upper_k);
224                if let Some(source) = widened.sources.last_mut() {
225                    source.fraction = 1.0;
226                }
227            } else {
228                let mut narrow = incoming.clone();
229                narrow.set_edges(upper_axis, upper_k);
230                merged.push(narrow);
231            }
232            k += 1;
233        }
234        merged.extend(window.into_iter().skip(k));
235    }
236    merged
237}

318    /// Overlapping windows: the window's channels below the axis are
319    /// prepended and its lowest overlapping channel merged with the axis's
320    /// first; above, a channel overlapping the axis's last by more than 1%
321    /// widens it, and by less is cut to a narrower channel.
322    #[test]
323    fn overlapping_windows_merge_their_edge_channels() {
324        // Axis 10..14 (centres 11, 13); the window 7..12 (centres 8, 10.5).
325        let axis = (vec![11.0, 13.0], vec![2.0, 2.0]);
326        let below = (vec![8.0, 10.5], vec![2.0, 3.0]);
327        assert_eq!(
328            combined(&[axis.clone(), below]),
329            (vec![8.0, 10.5, 13.0], vec![2.0, 3.0, 2.0])
330        );
331        // The window lowest in frequency is the axis the others merge into:
332        // here 8..10 and 11..13, whose last channel the window 10..14
333        // overlaps by half its width, widening it to 11..14.
334        let starts_below = (vec![9.0, 12.0], vec![2.0, 2.0]);
335        assert_eq!(
336            combined(&[axis.clone(), starts_below]),
337            (vec![9.0, 12.5], vec![2.0, 3.0])
338        );
339        // A wider window starting above the axis's centre but below its
340        // lower edge merges with its first channel (9.5..12), and that
341        // channel is then widened again at the upper edge (10..13.5).
342        assert_eq!(
343            combined(&[(vec![11.0], vec![2.0]), (vec![11.5], vec![4.0])]),
344            (vec![11.75], vec![3.5])
345        );
346        // Above: the window 13.5..15.5 overlaps the axis's last channel
347        // (12..14) by 25% of its width, which widens it to 12..15.5.
348        let above = (vec![14.5, 16.5], vec![2.0, 2.0]);
349        assert_eq!(
350            combined(&[axis.clone(), above]),
351            (vec![11.0, 13.75, 16.5], vec![2.0, 3.5, 2.0])
352        );
353        // An overlap of 0.5% leaves a channel from the axis's upper edge.
354        let barely = (vec![14.99, 17.0], vec![2.0, 2.0]);
355        let (frequencies, widths) = combined(&[axis, barely]);
356        assert_eq!(frequencies.len(), 4);
357        assert_eq!(frequencies[2], (14.0 + 15.99) / 2.0);
358        assert!((widths[2] - 1.99).abs() < 1e-12);
359    }

361    /// A channel-mode cube over combined windows with `nchan=-1` holds as
362    /// many channels of the first channel's width as fit in the combined
363    /// band (`regridChanBounds`), so windows with gaps between them give
364    /// more planes than channels; each window reads its own channel.
365    #[test]
366    fn a_default_channel_cube_spans_the_combined_band() {
367        use crate::spectral_selection::{CubeAxisConfig, CubeSpecMode, CubeSpectralSetup};
368
369        let observatory = casa_core::measures::position::MPosition::new_itrf(
370            -1_601_185.4,
371            -5_041_977.5,
372            3_554_875.9,
373        );
374        let direction = casa_core::measures::direction::MDirection::from_angles(
375            0.0,
376            std::f64::consts::FRAC_PI_4,
377            casa_core::measures::direction::DirectionRef::J2000,
378        );
379        let engine = crate::derived::engine::MsCalEngine::from_parts(
380            vec![observatory.clone()],
381            vec![direction],
382            observatory,
383            casa_test_support::deterministic_measures_provider(),
384        );
385        let config = CubeAxisConfig {
386            specmode: CubeSpecMode::Cubedata,
387            ..CubeAxisConfig::default()
388        };
389        let time = 59_000.0 * 86_400.0;
390        let axis = |windows: &[(Vec<f64>, Vec<f64>)]| {
391            let (frequencies, widths) = combined(windows);
392            CubeSpectralSetup::for_casa_cube_axis(
393                casa_core::measures::frequency::FrequencyRef::TOPO,
394                &frequencies,
395                &widths,
396                None,
397                &config,
398                time,
399                0,
400                None,
401                &engine,
402            )
403            .expect("a channel-mode axis")
404        };
405        let touching = [1.2e9, 1.5e9, 1.8e9].map(|hz| (vec![hz], vec![3.0e8]));
406        let setup = axis(&touching);
407        assert_eq!(
408            setup.output_channel_frequencies_hz,
409            vec![1.2e9, 1.5e9, 1.8e9]
410        );
411        for (frequencies, widths) in &touching {
412            let support = setup
413                .source_channel_support(frequencies, widths, [time, time], 0, &engine)
414                .expect("support");
415            assert_eq!(support.indices, vec![0]);
416        }
417        let separated = [2.0e9, 2.5e9, 3.0e9].map(|hz| (vec![hz], vec![2.0e8]));
418        let setup = axis(&separated);
419        assert_eq!(setup.output_channel_frequencies_hz.len(), 6);
420        assert_eq!(setup.output_channel_frequencies_hz[0], 2.0e9);
421        assert_eq!(setup.output_channel_frequencies_hz[5], 3.0e9);
422    }
