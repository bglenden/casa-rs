2//! Statistical weight estimation on the attached MeasurementSet.
3
4use casa_calibration::{
5    StatWtAlgorithm, StatWtChannelBin, StatWtCombine, StatWtDataColumn, StatWtMs, StatWtOptions,
6    StatWtTimeBin, estimate_statwt, statwt_weight_range,
7};
8use casa_ms::MsSelection;
9use pyo3::prelude::*;
10use pyo3::types::{PyBool, PyDict, PyDictMethods, PyList, PyListMethods, PyString};
11
12use super::{PyMsTool, runtime_error, strict_bool};
13
14#[pymethods]
15impl PyMsTool {
16    #[allow(clippy::too_many_arguments)]
17    #[pyo3(signature = (combine = "", timebin = None, slidetimebin = false, chanbin = None, minsamp = 2,
18        statalg = "classic", fence = -1.0, center = "mean", lside = true, zscore = -1.0, maxiter = -1,
19        fitspw = "", excludechans = false, wtrange = None, preview = false, datacolumn = "corrected"),
20        text_signature = "($self, combine='', timebin=1, slidetimebin=False, chanbin='spw', minsamp=2, statalg='classic', fence=-1.0, center='mean', lside=True, zscore=-1.0, maxiter=-1, fitspw='', excludechans=False, wtrange=[], preview=False, datacolumn='corrected')")]
21    fn statwt<'py>(
22        &self,
23        py: Python<'py>,
24        combine: &str,
25        timebin: Option<&Bound<'py, PyAny>>,
26        #[pyo3(from_py_with = strict_bool)] slidetimebin: bool,
27        chanbin: Option<&Bound<'py, PyAny>>,
28        minsamp: i64,
29        statalg: &str,
30        fence: f64,
31        center: &str,
32        #[pyo3(from_py_with = strict_bool)] lside: bool,
33        zscore: f64,
34        maxiter: i64,
35        fitspw: &str,
36        #[pyo3(from_py_with = strict_bool)] excludechans: bool,
37        wtrange: Option<Vec<f64>>,
38        #[pyo3(from_py_with = strict_bool)] preview: bool,
39        datacolumn: &str,
40    ) -> PyResult<Bound<'py, PyDict>> {
41        let options = StatWtOptions {
42            data_column: StatWtDataColumn::parse(datacolumn).map_err(runtime_error)?,
43            algorithm: StatWtAlgorithm::parse(statalg, fence, center, lside, zscore, maxiter)
44                .map_err(runtime_error)?,
45            minimum_samples: usize::try_from(minsamp).unwrap_or(0),
46            combine: StatWtCombine::parse(combine),
47            time_bin: read_bin(
48                timebin,
49                StatWtTimeBin::Timestamps(1),
50                StatWtTimeBin::Timestamps,
51                StatWtTimeBin::duration,
52            )?,
53            sliding: slidetimebin,
54            channel_bin: read_bin(
55                chanbin,
56                StatWtChannelBin::Window,
57                StatWtChannelBin::Channels,
58                StatWtChannelBin::parse,
59            )?,
60            fit_spw: fitspw.to_owned(),
61            exclude_channels: excludechans,
62            weight_range: statwt_weight_range(wtrange.as_deref().unwrap_or_default())
63                .map_err(runtime_error)?,
64        };
65        let report = py.detach(|| {
66            let mut attached = self.attached.lock().map_err(runtime_error)?;
67            let Some(attached) = attached.as_mut() else {
68                return Ok(None);
69            };
70            if !preview && !attached.writable {
71                return Err(runtime_error("MeasurementSet is not writable"));
72            }
73            let mut transient = None;
74            let locks = if preview {
75                None
76            } else if !attached.write_locks.is_empty() {
77                Some(&mut attached.write_locks)
78            } else {
79                Some(
80                    transient
81                        .insert(StatWtMs::write_locks(&attached.sources).map_err(runtime_error)?),
82                )
83            };
84            let mut ms = StatWtMs::open(&attached.path, false).map_err(runtime_error)?;
85            let mut selected = ms.select(&MsSelection::default()).map_err(runtime_error)?;
86            if let Some(rows) = &attached.selected_rows {
87                selected.retain(|row| rows.binary_search(&row.row_index).is_ok());
88            }
89            let report =
90                estimate_statwt(&mut ms, &selected, &options, locks).map_err(runtime_error)?;
91            drop(ms);
92            if let Some(locks) = transient {
93                for lock in locks.into_values() {
94                    lock.release().map_err(runtime_error)?;
95                }
96            }
97            Ok(Some(report))
98        })?;
99        let result = PyDict::new(py);
100        if let Some(report) = report {
101            result.set_item("mean", report.mean)?;
102            result.set_item("variance", report.variance)?;
103        }
104        Ok(result)
105    }
106}

108fn read_bin<T>(
109    value: Option<&Bound<'_, PyAny>>,
110    default: T,
111    count: impl FnOnce(usize) -> T,
112    quantity: impl FnOnce(&str) -> casa_ms::MsResult<T>,
113) -> PyResult<T> {
114    let Some(value) = value else {
115        return Ok(default);
116    };
117    if let Ok(text) = value.cast::<PyString>() {
118        return quantity(text.to_str()?).map_err(runtime_error);
119    }
120    if let Ok(list) = value.cast::<PyList>()
121        && list.is_empty()
122    {
123        return Ok(default);
124    }
125    if !value.is_instance_of::<PyBool>()
126        && let Ok(size) = value.extract::<usize>()
127    {
128        return Ok(count(size));
129    }
130    Err(runtime_error(
131        "statwt bin requires an integer count or a string quantity",
132    ))
133}
