14pub(super) struct ImageBeams {
15    records: Vec<RecordValue>,
16    pub(super) classes: Vec<u8>,
17}
18
19impl ImageBeams {
20    pub(super) fn load(path: &Path, row: &FourierRow<'_>) -> Result<Self, GainSolveError> {
21        let records = VoltagePatternTable::open(path)
22            .and_then(|table| table.definitions())
23            .map_err(model_error)?;
24        let patterns = records
25            .iter()
26            .filter(|record| matches!(record_text(record, "name"), Ok("IMAGE")))
27            .map(|record| Ok((record, image_key(record)?, antenna_names(record)?)))
28            .collect::<Result<Vec<_>, GainSolveError>>()?;
29        let antenna = row.ms.antenna().map_err(model_error)?;
30        let mut keys = Vec::new();
31        let mut selected = Vec::new();
32        let mut classes = Vec::with_capacity(antenna.row_count());
33        for index in 0..antenna.row_count() {
34            let name = antenna.name(index).map_err(model_error)?;
35            let mut last = None;
36            let mut active = None;
37            for &(record, key, names) in &patterns {
38                if names.first().is_some_and(|name| name == "*") || names.iter().any(|v| v == &name)
39                {
40                    last = Some((record, key));
41                    if let Some(index) = keys.iter().position(|known| *known == key) {
42                        active = Some(index);
43                    }
44                }
45            }
46            let (record, key) = last.ok_or_else(|| {
47                model_error(format!(
48                    "virtual mosaic antenna {name} has no image voltage pattern"
49                ))
50            })?;
51            let class = active.unwrap_or_else(|| {
52                keys.push(key);
53                selected.push(record.clone());
54                keys.len() - 1
55            });
56            classes.push(u8::try_from(class).map_err(model_error)?);
57        }
58        Ok(Self {
59            records: selected,
60            classes,
61        })
62    }
63
64    pub(super) fn len(&self) -> usize {
65        self.records.len()
66    }
67
68    pub(super) fn screens(&self, legacy: bool) -> Result<Vec<ImageBeam>, GainSolveError> {
69        self.records
70            .iter()
71            .map(|record| ImageBeam::load(record, legacy))
72            .collect()
73    }
74}

76fn image_key(record: &RecordValue) -> Result<&str, GainSolveError> {
77    match record.get("realimage") {
78        Some(_) => record_text(record, "realimage"),
79        None => record_text(record, "compleximage"),
80    }
81}
82
83fn antenna_names(record: &RecordValue) -> Result<&ndarray::ArrayD<String>, GainSolveError> {
84    match record.get("antennanames") {
85        Some(Value::Array(ArrayValue::String(names))) => Ok(names),
86        _ => Err(model_error("mosaic voltage pattern has no antenna names")),
87    }
88}

90struct ImagePlane {
91    voltage: Array2<Complex32>,
92    channel_count: usize,
93    diagonal: usize,
94    increment: [f64; 2],
95    reference_pixel: [f64; 2],
96    frequency_hz: f64,
97}
98
99impl ImagePlane {
100    fn load<T: ImagePixel + Copy>(
101        path: &str,
102        real_diagonal: Option<usize>,
103        convert: impl Fn(T) -> Complex32,
104    ) -> Result<Self, GainSolveError> {
105        let image = PagedImage::<T>::open(path).map_err(model_error)?;
106        let &[nx, ny, 4, channels] = image.shape() else {
107            return Err(model_error(
108                "mosaic voltage image requires four correlation planes",
109            ));
110        };
111        if nx < 2 || ny < 2 || channels == 0 {
112            return Err(model_error("empty mosaic voltage image"));
113        }
114        let coords = image.coordinates();
115        if coords.find_coordinate(CoordinateType::Direction) != Some(0)
116            || coords.find_coordinate(CoordinateType::Stokes) != Some(1)
117            || coords.find_coordinate(CoordinateType::Spectral) != Some(2)
118        {
119            return Err(model_error("invalid mosaic voltage image axes"));
120        }
121        let CoordinateModel::Direction(direction) = coords.coordinate(0) else {
122            unreachable!()
123        };
124        let diagonal = if let Some(diagonal) = real_diagonal {
125            diagonal
126        } else {
127            let CoordinateModel::Stokes(stokes) = coords.coordinate(1) else {
128                unreachable!()
129            };
130            diagonal_index(stokes.stokes())?
131        };
132        let CoordinateModel::Spectral(spectral) = coords.coordinate(2) else {
133            unreachable!()
134        };
135        let values = image
136            .get_slice(&[0, 0, diagonal, 0], &[nx, ny, 1, 1])
137            .map_err(model_error)?;
138        let voltage = values
139            .index_axis_move(Axis(3), 0)
140            .index_axis_move(Axis(2), 0)
141            .into_dimensionality::<Ix2>()
142            .map_err(model_error)?
143            .mapv(convert);
144        Ok(Self {
145            voltage,
146            channel_count: channels,
147            diagonal,
148            increment: direction.increment().try_into().expect("direction axes"),
149            reference_pixel: direction
150                .reference_pixel()
151                .try_into()
152                .expect("direction axes"),
153            frequency_hz: spectral
154                .world_frequency_hz(spectral.reference_pixel()[0])
155                .map_err(model_error)?,
156        })
157    }
158
159    fn value(&self, offset: [f64; 2], frequency_hz: f64) -> Complex32 {
160        // PBMath2DImage::updateJones replaces the coordinate centre and PC
161        // matrix, reverses the longitude increment and scales both axes.
162        let [x, y] = [
163            self.reference_pixel[0]
164                - offset[0] * frequency_hz / (self.increment[0] * self.frequency_hz),
165            self.reference_pixel[1]
166                + offset[1] * frequency_hz / (self.increment[1] * self.frequency_hz),
167        ];
168        let [nx, ny] = [self.voltage.nrows(), self.voltage.ncols()];
169        // Interpolate2D tests the truncated indices and uses the last
170        // interval at the upper edge. A fractional edge pixel can extrapolate.
171        if !(x > -1. && x < nx as f64 && y > -1. && y < ny as f64) {
172            return Complex32::default();
173        }
174        let [ix, iy] = [(x as usize).min(nx - 2), (y as usize).min(ny - 2)];
175        let [fx, fy] = [(x - ix as f64) as f32, (y - iy as f64) as f32];
176        let low = self.voltage[[ix, iy]] * (1. - fx) + self.voltage[[ix + 1, iy]] * fx;
177        let high = self.voltage[[ix, iy + 1]] * (1. - fx) + self.voltage[[ix + 1, iy + 1]] * fx;
178        low * (1. - fy) + high * fy
179    }
180}
181
182fn diagonal_index(stokes: &[casa_coordinates::StokesType]) -> Result<usize, GainSolveError> {
183    let first = stokes.iter().map(|kind| kind.code()).min();
184    let last = stokes.get(3).map(|kind| kind.code());
185    match (first, last) {
186        (Some(first @ (5 | 9)), Some(last)) if (first..first + 4).contains(&last) => {
187            Ok((last - first) as usize)
188        }
189        _ => Err(model_error("invalid mosaic voltage correlations")),
190    }
191}

193pub(super) struct ImageBeam {
194    real: ImagePlane,
195    imaginary: Option<ImagePlane>,
196}
197
198impl ImageBeam {
199    fn load(record: &RecordValue, legacy: bool) -> Result<Self, GainSolveError> {
200        let (real, imaginary) = match (record.get("realimage"), record.get("imagimage"), legacy) {
201            (Some(_), Some(_), _) | (Some(_), _, false) => {
202                let real = ImagePlane::load::<f32>(record_text(record, "realimage")?, None, |v| {
203                    Complex32::new(v, 0.)
204                })?;
205                // Use the standard record field when a legacy table lacks
206                // the compatibility alias that CASA's reader expects.
207                let key = if legacy && record.get("imagim").is_some() {
208                    "imagim"
209                } else {
210                    "imagimage"
211                };
212                let path = record_text(record, key)?;
213                let imaginary = if path.is_empty() {
214                    None
215                } else {
216                    // PBMath2DImage uses the real-image polarization map
217                    // for both arrays, including independently labeled axes.
218                    let plane = ImagePlane::load::<f32>(path, Some(real.diagonal), |v| {
219                        Complex32::new(0., v)
220                    })?;
221                    if plane.voltage.shape() != real.voltage.shape()
222                        || plane.channel_count != real.channel_count
223                    {
224                        return Err(model_error(
225                            "real and imaginary mosaic voltage images have different shapes",
226                        ));
227                    }
228                    Some(plane)
229                };
230                (real, imaginary)
231            }
232            _ => (
233                ImagePlane::load::<Complex32>(record_text(record, "compleximage")?, None, |v| v)?,
234                None,
235            ),
236        };
237        Ok(Self { real, imaginary })
238    }
239
240    pub(super) fn support_pixels(&self, increment: f64, reference_hz: f64) -> usize {
241        (self.real.voltage.nrows() as f64 * self.real.frequency_hz / reference_hz
242            * (self.real.increment[0] / increment).abs()) as usize
243    }
244
245    pub(super) fn value(&self, offset: [f64; 2], frequency_hz: f64) -> Complex32 {
246        let real = self.real.value(offset, frequency_hz);
247        real + self
248            .imaginary
249            .as_ref()
250            .map_or(Complex32::default(), |plane| {
251                plane.value(offset, frequency_hz)
252            })
253    }
254}
