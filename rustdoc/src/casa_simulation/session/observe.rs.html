2//! `sm.settimes` and `sm.observe`: the schedule of one scan.
3//!
4//! `observe` records a scan; the session writes every scan's rows at
5//! `close`. The bookkeeping follows `NewMSSimulator::observe`: scan numbers
6//! continue from the last scan, an OBSERVATION row is added when asked for
7//! or when there is none, STATE rows are shared by identical states, and the
8//! integration count is `max(1, Int(0.5 + (stop − start)/Tint))`.
9//!
10//! **Times honour the requested UTC** (plan decision 5). CASA adds the
11//! reference time in TAI to start and stop and stores the result in a TIME
12//! column labelled UTC, so its rows are TAI − UTC (37 s) late. casa-rs adds
13//! the reference time in UTC.
14

76impl Times {
77    /// CASA's defaults before `settimes`: 10 s integrations in hour-angle
78    /// mode, referred to the time the session was opened.
79    pub(crate) fn defaults(now: MEpoch) -> Self {
80        Self {
81            integration_s: 10.0,
82            use_hour_angle: true,
83            reference: now,
84            set: false,
85            hour_angle_offset_s: None,
86        }
87    }

89    /// The reference epoch as UTC seconds (MJD × 86400).
90    pub(crate) fn reference_utc_s(&self) -> f64 {
91        self.reference.value().as_mjd() * 86_400.0
92    }

117    pub(crate) antenna_count: usize,
118    /// The J2000 phase centre, radians.
119    pub(crate) phase_center_rad: [f64; 2],
120    /// Autocorrelation rows are written when the weight is positive.

122    pub(crate) elevation_limit_rad: f64,
123    pub(crate) shadow_fraction_limit: f64,
124}
125
126impl Scan {
127    /// Baselines per integration, autocorrelations included when written.
128    pub(crate) fn baseline_count(&self) -> usize {
129        let n = self.antenna_count;
130        if self.autocorrelation_weight > 0.0 {
131            n * (n + 1) / 2
132        } else {
133            n * n.saturating_sub(1) / 2
134        }
135    }
136
137    /// MAIN rows of the scan.
138    pub(crate) fn row_count(&self) -> usize {
139        self.baseline_count() * self.integration_count
140    }
141
142    /// TIME of integration `index`: CASA accumulates `Time += Tint` from
143    /// the start and stores the integration's centre.
144    pub(crate) fn integration_times_s(&self) -> impl Iterator<Item = f64> + '_ {
145        let mut time = self.start_s;
146        (0..self.integration_count).map(move |_| {
147            let centre = time + self.integration_s / 2.0;
148            time += self.integration_s;
149            centre
150        })
151    }
152}
153
154/// CASA's integration count for one pointing: `max(1, Int(0.5 + span/Tint))`,
155/// with C++'s truncation toward zero.
156pub(crate) fn integration_count(start_s: f64, stop_s: f64, integration_s: f64) -> usize {
157    let count = (0.5 + (stop_s - start_s) / integration_s).trunc();
158    if count >= 1.0 { count as usize } else { 1 }
159}

161/// `NewMSSimulator`'s hour-angle offset: minus the hour angle of `direction`
162/// at the reference epoch and the first antenna, as seconds, converting
163/// hours of hour angle at 3600 seconds each (no sidereal factor).
164pub(crate) fn hour_angle_offset_s(
165    direction_rad: [f64; 2],
166    reference: &MEpoch,
167    first_antenna_itrf_m: [f64; 3],
168    measures: Arc<dyn MeasuresProvider>,
169) -> SimulatorResult<f64> {
170    let [x, y, z] = first_antenna_itrf_m;
171    let frame = MeasFrame::new()
172        .with_epoch(reference.clone())
173        .with_position(MPosition::new_itrf(x, y, z))
174        .with_measures(measures);
175    let hadec = MDirection::from_angles(direction_rad[0], direction_rad[1], DirectionRef::J2000)
176        .convert_to(DirectionRef::HADEC, &frame)
177        .map_err(|error| {
178            SimulatorError::refused(format!("hour angle of the first source: {error}"))
179        })?;
180    let mut hour_angle = hadec.as_angles().0;
181    // casacore's MVDirection longitude is in (−π, π].
182    if hour_angle > std::f64::consts::PI {
183        hour_angle -= std::f64::consts::TAU;
184    } else if hour_angle <= -std::f64::consts::PI {
185        hour_angle += std::f64::consts::TAU;
186    }
187    Ok(-hour_angle * (3600.0 * 180.0 / std::f64::consts::PI / 15.0))
188}
