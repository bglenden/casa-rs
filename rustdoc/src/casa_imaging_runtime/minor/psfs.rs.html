10// SIImageStore::normPSF scales the single-term plane by its positive
11// maximum. The solver's central-patch peak search is a separate operation.
12pub(super) fn single_psf_peak(psf: &[f64]) -> Result<f64, MinorCycleRunError> {
13    let peak = psf.iter().try_fold(0.0_f64, |peak, &value| {
14        value.is_finite().then_some(peak.max(value))
15    });
16    Ok(peak
17        .filter(|&value| value > 0.0)
18        .ok_or(casa_imaging_deconvolution::Error::PsfPeak)?)
19}

21impl PsfCache {
22    /// Measure a reused single-term PSF in the saved image's units.
23    #[must_use]
24    pub fn for_reused_single_psf() -> Self {
25        Self {
26            reuses_single_psf: true,
27            ..Self::default()
28        }
29    }
30
31    pub(super) fn summarize_loaded(
32        &self,
33        normal: &FinalNormalState,
34        base: &ModelGeneration,
35        setup: &MinorCycleSetup<'_>,
36        key: PlaneKey,
37        plane: &PlaneData<'_>,
38    ) -> Result<PsfSummary, MinorCycleRunError> {
39        match self.summaries.get(&key) {
40            Some(summary) => Ok(*summary),
41            None => summarize_psf(
42                normal,
43                base,
44                setup,
45                key,
46                &plane.psf[0],
47                plane.shape,
48                self.reuses_single_psf.then_some(plane.psf_peak),
49            ),
50        }
51    }
52}

54/// A saved grid is for the main domain, ordered by absolute channel and polarization.
55pub(super) fn saved_psf_beam(
56    normal: &FinalNormalState,
57    setup: &MinorCycleSetup<'_>,
58    key: PlaneKey,
59) -> Option<RestoringBeam> {
60    saved_beam(normal, &setup.psf_beams, key)
61}
62
63pub(super) fn saved_beam(
64    normal: &FinalNormalState,
65    beams: &Option<Arc<[RestoringBeam]>>,
66    key: PlaneKey,
67) -> Option<RestoringBeam> {
68    (key.domain == 0)
69        .then(|| {
70            beams
71                .as_ref()?
72                .get(key.channel * normal.polarization_count() + key.polarization)
73                .copied()
74        })
75        .flatten()
76}

78pub(super) fn summarize_psf(
79    normal: &FinalNormalState,
80    base: &ModelGeneration,
81    setup: &MinorCycleSetup<'_>,
82    key: PlaneKey,
83    psf: &[f64],
84    shape: PlaneShape,
85    image_peak: Option<f64>,
86) -> Result<PsfSummary, MinorCycleRunError> {
87    let increment = base
88        .shape()
89        .direction(key.domain)
90        .expect("domain direction")
91        .increment_rad();
92    let cell = [1.0, (increment[1] / increment[0]).abs()];
93    let saved_beam = saved_psf_beam(normal, setup, key);
94    Ok(match (image_peak, saved_beam) {
95        (Some(peak), _) => {
96            PsfSummary::with_image_peak(psf, shape, setup.psf_cutoff, saved_beam, cell, peak)?
97        }
98        (None, Some(beam)) => PsfSummary::with_saved_beam_and_cell(psf, shape, beam, cell)?,
99        (None, None) => PsfSummary::with_cutoff_and_cell(psf, shape, setup.psf_cutoff, cell)?,
100    })
101}

103/// Measure into the cache the PSF of each plane of `keys` that holds data
104/// and is not yet measured, side by side on the team.
105pub(super) fn measure_psfs(
106    normal: &FinalNormalState,
107    base: &ModelGeneration,
108    taylor: bool,
109    cache: &mut PsfCache,
110    keys: &[PlaneKey],
111    setup: &MinorCycleSetup<'_>,
112    team: &WorkerTeam,
113) -> Result<(), MinorCycleRunError> {
114    let mut slots = keys
115        .iter()
116        .filter(|key| !cache.summaries.contains_key(key) && holds_data(normal, taylor, **key))
117        .map(|key| (*key, None))
118        .collect::<Vec<_>>();
119    let reused = cache.reuses_single_psf;
120    team.for_each_mut(&mut slots, |_, (key, summary)| {
121        let shape = plane_shape(normal, *key);
122        let psf = read_psf(normal, taylor, *key)?;
123        let peak = if taylor {
124            psf_peak_value(&psf, shape)?
125        } else {
126            single_psf_peak(&psf)?
127        };
128        *summary = Some(summarize_psf(
129            normal,
130            base,
131            setup,
132            *key,
133            &normalise(psf, peak),
134            shape,
135            reused.then_some(peak),
136        )?);
137        Ok::<_, MinorCycleRunError>(())
138    })?;
139    for (key, summary) in slots {
140        if let Some(summary) = summary {
141            cache.summaries.insert(key, summary);
142        }
143    }
144    beams::settle_refused_fits(normal, base, taylor, cache, setup.psf_beams.is_some(), None)
145}

147#[cfg(test)]
148mod tests {
149    use super::*;
150
151    #[test]
152    fn single_term_psf_divisor_uses_the_whole_positive_plane() {
153        let shape = PlaneShape::new(128, 128);
154        let mut psf = vec![0.0; shape.len()];
155        psf[shape.index(64, 64)] = 0.9;
156        psf[shape.index(10, 10)] = 1.0;
157        psf[shape.index(15, 15)] = -1.2;
158        assert_eq!(psf_peak_value(&psf, shape).unwrap(), 0.9);
159        assert_eq!(single_psf_peak(&psf).unwrap(), 1.0);
160    }
161}
