20    PyRuntimeError::new_err(message.to_string())
21}

23pub(super) fn values(value: &Bound<'_, PyAny>, method: &str) -> PyResult<Vec<f64>> {
24    if value.is_none() {
25        return Ok(Vec::new());
26    }
27    if value.is_instance_of::<PyTuple>() {
28        return Err(PySystemError::new_err(format!(
29            "<built-in function coordsys_{method}> returned a result with an exception set"
30        )));
31    }
32    // Scalar text takes the binding's -1 sentinel. Vector text is parsed.
33    if value.is_instance_of::<PyString>() || value.is_instance_of::<PyBytes>() {
34        return Ok(vec![-1.0]);
35    }
36    if value.is_instance_of::<PyList>() || value.is_instance_of::<PyUntypedArray>() {
37        let py = value.py();
38        let options = PyDict::new(py);
39        if value.is_instance_of::<PyList>() {
40            // Preserve mixed element types while determining the nested shape.
41            options.set_item("dtype", "object")?;
42        }
43        let array = py
44            .import("numpy")?
45            .call_method("asarray", (value,), Some(&options))?;
46        let kind = array.cast::<PyUntypedArray>()?.dtype().kind();
47        if matches!(kind, b'b' | b'i' | b'u' | b'f') {
48            let doubles = array.call_method1("astype", ("float64",))?;
49            let values = doubles.extract::<PyReadonlyArrayDyn<'_, f64>>()?;
50            return Ok(values.as_array().t().iter().copied().collect());
51        }
52        let flat = array.call_method1("ravel", ("F",))?;
53        return flat
54            .try_iter()?
55            .map(|item| item.map(|item| double_vector_element(&item).unwrap_or(0.0)))
56            .collect();
57    }
58    Ok(vec![value.extract::<f64>()?])
59}

61fn beta(velocity: &Quantity) -> PyResult<f64> {
62    let unit = velocity.unit();
63    let base = *unit.val();
64    let speed = UnitDim::basis(Dimension::Length) / UnitDim::basis(Dimension::Time);
65    if base.dim == speed {
66        Ok(velocity.value() * base.factor / C_MPS)
67    } else if base.dim == UnitDim::NODIM {
68        Ok(velocity.value() * base.factor)
69    } else {
70        Err(nonconformant(unit.name()))
71    }
72}
73
74fn nonconformant(unit: &str) -> PyErr {
75    raised(format!("Quantum::assure non-conforming unit type '{unit}'"))
76}

78fn spectral_shift(values: &mut [f64], freq_unit: &str, velocity: &Quantity) -> PyResult<()> {
79    let ratio = MDoppler::new(beta(velocity)?, DopplerRef::BETA).as_frequency_ratio();
80    let unit = casacore::casacore_unit(freq_unit).map_err(raised)?;
81    if values.is_empty() {
82        return Ok(());
83    }
84    let dim = unit.val().dim;
85    if casacore::frequency_hz(&Quantity::with_unit(1.0, unit)).is_none() {
86        return Err(nonconformant(freq_unit));
87    }
88    let ratio = if [Dimension::Time, Dimension::Length]
89        .into_iter()
90        .any(|dimension| dim == UnitDim::basis(dimension))
91    {
92        ratio.recip()
93    } else {
94        ratio
95    };
96    for value in values {
97        *value *= ratio;
98    }
99    Ok(())
100}

109pub(super) fn sampled_frequencies_from_velocities(
110    velocities: Vec<f64>,
111    unit: &Unit,
112    convention: DopplerRef,
113    rest_hz: f64,
114) -> PyResult<Vec<f64>> {
115    let conversion = VelocityConversion {
116        rest_hz,
117        hz_per_unit: 1.0,
118        velocity_scale: unit.val().factor / C_MPS,
119        convention,
120    };
121    velocities
122        .into_iter()
123        .map(|value| conversion.velocity_to_frequency(value))
124        .collect()
125}

127fn velocity_settings(
128    state: &CoordinateState,
129    field: &str,
130    spectral: &SpectralCoordinate,
131    vel_unit: &str,
132    doppler: &str,
133) -> PyResult<(f64, DopplerRef)> {
134    let speed = UnitDim::basis(Dimension::Length) / UnitDim::basis(Dimension::Time);
135    let unit = casacore::casacore_unit(vel_unit).map_err(raised)?;
136    let requested = doppler_ref(DOPPLER_CODES.resolve_or_default(doppler)).map_err(raised)?;
137    if unit.val().dim == speed {
138        return Ok((unit.val().factor / C_MPS, requested));
139    }
140    let record = match state.record.get(field) {
141        Some(Value::Record(record)) => Some(record),
142        _ => None,
143    };
144    let (stored, stored_convention) = stored_velocity_settings(record, spectral)?;
145    let convention = if vel_unit.is_empty() {
146        requested
147    } else {
148        stored_convention
149    };
150    Ok((stored.val().factor / C_MPS, convention))
151}
152
153pub(in crate::tools) fn stored_velocity_settings(
154    record: Option<&RecordValue>,
155    spectral: &SpectralCoordinate,
156) -> PyResult<(Unit, DopplerRef)> {
157    let mut convention = spectral.velocity_convention();
158    let stored_unit = if let Some(record) = record {
159        if record.get("velType").is_none() {
160            let legacy_code = match record.get("prefVelType") {
161                Some(Value::Scalar(ScalarValue::Int32(code))) => Some(*code),
162                Some(Value::Scalar(ScalarValue::Int64(code))) => i32::try_from(*code).ok(),
163                _ => None,
164            };
165            if let Some(legacy) = legacy_code.and_then(DopplerRef::from_casacore_code) {
166                convention = legacy;
167            }
168        }
169        match record.get("velUnit").or_else(|| record.get("prefVelUnit")) {
170            Some(Value::Scalar(ScalarValue::String(unit))) if !unit.is_empty() => unit.as_str(),
171            _ => "km/s",
172        }
173    } else {
174        "km/s"
175    };
176    let stored = casacore::casacore_unit(stored_unit).map_err(raised)?;
177    let speed = UnitDim::basis(Dimension::Length) / UnitDim::basis(Dimension::Time);
178    if stored.val().dim == speed {
179        Ok((stored, convention))
180    } else {
181        // Restoring an incompatible unit leaves both defaults.
182        Ok((
183            Unit::new("km/s").expect("standard speed unit"),
184            DopplerRef::RADIO,
185        ))
186    }
187}

189impl VelocityConversion {
190    pub(super) fn for_spectral_axis(
191        spectral: &SpectralCoordinate,
192        unit: &Unit,
193        convention: DopplerRef,
194    ) -> PyResult<Self> {
195        Ok(Self {
196            rest_hz: spectral.rest_frequency_hz().map_err(raised)?,
197            hz_per_unit: spectral.value_in_hz(1.0).map_err(raised)?,
198            velocity_scale: unit.val().factor / C_MPS,
199            convention,
200        })
201    }
202
203    pub(super) fn mixed_velocity(&self, frequency: f64) -> PyResult<f64> {
204        if self.rest_hz == 0.0 {
205            return Err(raised(
206                "Exception: Rest frequency is 0 so cannot convert to velocity.",
207            ));
208        }
209        let velocity = self.frequency_to_velocity(frequency)?;
210        if velocity.is_nan() {
211            return Err(raised("velocity is NaN"));
212        }
213        Ok(velocity)
214    }
215
216    pub(super) fn mixed_frequency(&self, velocity: f64) -> PyResult<f64> {
217        let frequency = self.velocity_to_frequency(velocity)?;
218        if frequency <= 0.0 {
219            return Err(raised("frequency <= 0"));
220        }
221        Ok(frequency)
222    }
223
224    fn new(
225        state: &CoordinateState,
226        spectral: &SpectralCoordinate,
227        field: &str,
228        freq_unit: &str,
229        doppler: &str,
230        vel_unit: &str,
231    ) -> PyResult<Self> {
232        let axis_units = spectral.axis_units();
233        let freq_unit = if freq_unit.is_empty() {
234            &axis_units[0]
235        } else {
236            freq_unit
237        };
238        let frequency_error = |cause| {
239            raised(format!(
240                "Failed to set frequency units of {freq_unit} because {cause}"
241            ))
242        };
243        let frequency = casacore::casacore_unit(freq_unit)
244            .map_err(|_| frequency_error("Unknown unit - cannot calculate scaling"))?;
245        if frequency.val().dim != UnitDim::NODIM / UnitDim::basis(Dimension::Time) {
246            return Err(frequency_error("Units are not compatible dimensionally"));
247        }
248        let (velocity_scale, convention) =
249            velocity_settings(state, field, spectral, vel_unit, doppler)?;
250        Ok(Self {
251            rest_hz: spectral.rest_frequency_hz().map_err(raised)?,
252            hz_per_unit: frequency.val().factor,
253            velocity_scale,
254            convention,
255        })
256    }
257
258    fn frequency_to_velocity(&self, frequency: f64) -> PyResult<f64> {
259        if !frequency.is_finite() || self.rest_hz <= 0.0 {
260            return Ok(f64::NAN);
261        }
262        let ratio = frequency.abs() * self.hz_per_unit / self.rest_hz;
263        // CASA's VelocityMachine expresses physical shifts in beta before
264        // converting to the requested convention.
265        let frame = MeasFrame::new();
266        let velocity = MDoppler::new(ratio, DopplerRef::RATIO)
267            .convert_to(DopplerRef::BETA, &frame)
268            .and_then(|beta| beta.convert_to(self.convention, &frame))
269            .map_err(raised)?;
270        Ok(velocity.value() / self.velocity_scale)
271    }
272
273    fn velocity_to_frequency(&self, velocity: f64) -> PyResult<f64> {
274        let beta = MDoppler::new(velocity * self.velocity_scale, self.convention)
275            .convert_to(DopplerRef::BETA, &MeasFrame::new())
276            .map_err(raised)?;
277        Ok(beta.shift_frequency_hz(self.rest_hz) / self.hz_per_unit)
278    }
279}

281#[derive(Clone, Copy)]
282enum Output {
283    Velocity,
284    Frequency,
285}
286
287impl Output {
288    fn convert(self, conversion: &VelocityConversion, value: f64) -> PyResult<f64> {
289        match self {
290            Self::Velocity => conversion.frequency_to_velocity(value),
291            Self::Frequency => conversion.velocity_to_frequency(value),
292        }
293    }
294}

296impl PyCoordsys {
297    fn convert_doppler(
298        &self,
299        py: Python<'_>,
300        values: Vec<f64>,
301        frequnit: &str,
302        doppler: &str,
303        velunit: &str,
304        output: Output,
305    ) -> PyResult<Py<PyAny>> {
306        let result = py.detach(|| {
307            let state = self.state();
308            let Some((CoordinateModel::Spectral(spectral), placement)) =
309                coordinate(&state, CoordinateType::Spectral)?
310            else {
311                return Err(raised(
312                    "There is no spectral coordinate in this CoordinateSystem",
313                ));
314            };
315            let conversion = VelocityConversion::new(
316                &state,
317                spectral,
318                &placement.field,
319                frequnit,
320                doppler,
321                velunit,
322            )?;
323            values
324                .into_iter()
325                .map(|value| output.convert(&conversion, value))
326                .collect::<PyResult<Vec<_>>>()
327        })?;
328        CasaValue::floats(result).into_python(py)
329    }
330}

332#[pymethods]
333impl PyCoordsys {
334    #[pyo3(signature = (value, frequnit="", velocity=None), text_signature = "($self, value, frequnit='', velocity=[])")]
335    fn frequencytofrequency(
336        &self,
337        py: Python<'_>,
338        value: &Bound<'_, PyAny>,
339        frequnit: &str,
340        velocity: Option<&Bound<'_, PyAny>>,
341    ) -> PyResult<Py<PyAny>> {
342        let mut values = values(value, "frequencytofrequency")?;
343        let velocity = velocity
344            .map(|value| value.extract::<Variant>())
345            .transpose()?
346            .unwrap_or(Variant::Empty);
347        py.detach(|| {
348            let state = self.state();
349            let default_unit;
350            let unit = if frequnit.is_empty() {
351                default_unit = state
352                    .coordinates
353                    .find_coordinate(CoordinateType::Spectral)
354                    .map_or_else(String::new, |index| {
355                        state.coordinates.coordinate(index).axis_units().remove(0)
356                    });
357                default_unit.as_str()
358            } else {
359                frequnit
360            };
361            spectral_shift(&mut values, unit, &quantity(&velocity)?)
362        })?;
363        CasaValue::floats(values).into_python(py)
364    }
365
366    #[pyo3(signature = (value, frequnit="", doppler="radio", velunit="km/s"))]
367    fn frequencytovelocity(
368        &self,
369        py: Python<'_>,
370        value: &Bound<'_, PyAny>,
371        frequnit: &str,
372        doppler: &str,
373        velunit: &str,
374    ) -> PyResult<Py<PyAny>> {
375        let values = values(value, "frequencytovelocity")?;
376        self.convert_doppler(py, values, frequnit, doppler, velunit, Output::Velocity)
377    }
378
379    #[pyo3(signature = (value, frequnit="", doppler="radio", velunit="km/s"))]
380    fn velocitytofrequency(
381        &self,
382        py: Python<'_>,
383        value: &Bound<'_, PyAny>,
384        frequnit: &str,
385        doppler: &str,
386        velunit: &str,
387    ) -> PyResult<Py<PyAny>> {
388        let values = values(value, "velocitytofrequency")?;
389        self.convert_doppler(py, values, frequnit, doppler, velunit, Output::Frequency)
390    }
