19fn raised(error: impl ToString) -> PyErr {
20    PyRuntimeError::new_err(error.to_string())

24struct SubimageShape(IntegerVector);

26impl<'a, 'py> FromPyObject<'a, 'py> for SubimageShape {
27    type Error = PyErr;

29    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
30        if value.is_none() {
31            return Err(PySystemError::new_err(
32                "<built-in function coordsys_subimage> returned a result with an exception set",
33            ));
34        }
35        integer_vector(&value, "subimage", "newshape").map(Self)
36    }

39fn origin_input(input: Option<&Bound<'_, PyAny>>) -> PyResult<Vec<f32>> {
40    let invalid = || raised("Parameter neworigin is not a vector of pixel positions");
41    let input = input.ok_or_else(invalid)?;
42    if bool_vector(input)?.is_some() {
43        return Err(invalid());
44    }
45    let array = input
46        .cast::<PyUntypedArray>()
47        .ok()
48        .filter(|array| matches!(array.dtype().kind(), b'i' | b'u' | b'f'));
49    let values = match array {
50        Some(array) if array.ndim() == 0 => vec![input.call_method0("item")?.extract()?],
51        Some(array) if array.is_empty() => Vec::new(),
52        _ => match input.extract::<Variant>()? {
53            Variant::Numbers(values) | Variant::Array { values, .. } => values,
54            _ => return Err(invalid()),
55        },
56    };
57    values
58        .into_iter()
59        .map(|value| {
60            // The casacore subimage interface takes Float pixel offsets.
61            let value = value as f32;
62            if value.is_finite() {
63                Ok(value)
64            } else {
65                Err(raised("Subimage origin positions must be finite"))
66            }
67        })
68        .collect()
69}

71fn select_discrete<T: Copy>(
72    labels: &[T],
73    field: &str,
74    record: &mut RecordValue,
75    origin: f32,
76    count: Option<i32>,
77    describe: impl Fn(T) -> (&'static str, i32),
78) -> PyResult<()> {
79    let origin = (f64::from(origin) + 0.5) as i32;
80    let start = usize::try_from(origin).map_err(|_| raised("Illegal origin shift"))?;
81    let labels = labels
82        .get(start..)
83        .filter(|labels| !labels.is_empty())
84        .ok_or_else(|| raised("Illegal origin shift"))?;
85    let count = count
86        .filter(|&count| count > 0)
87        .map_or(labels.len(), |count| count as usize);
88    let selected = labels
89        .get(..count)
90        .ok_or_else(|| raised("New shape is invalid"))?;
91    record.upsert(
92        field,
93        Value::Array(ArrayValue::from_string_vec(
94            selected
95                .iter()
96                .map(|&kind| describe(kind).0.to_owned())
97                .collect(),
98        )),
99    );
100    record.upsert(
101        "crval",
102        Value::Array(ArrayValue::from_f64_vec(vec![f64::from(
103            describe(selected[0]).1,
104        )])),
105    );
106    Ok(())
107}

109fn shift_reference_pixel(
110    record: &mut RecordValue,
111    placement: &RecordCoordinate,
112    origin: &[f32],
113) -> PyResult<()> {
114    let references = record
115        .get_mut("crpix")
116        .ok_or_else(|| raised("Missing coordinate reference pixel"))?;
117    match references {
118        Value::Array(ArrayValue::Float64(values)) if values.len() == placement.pixel_axes.len() => {
119            for (pixel, axis) in values.iter_mut().zip(&placement.pixel_axes) {
120                if let Some(axis) = axis {
121                    *pixel -= f64::from(origin[*axis]);
122                }
123            }
124        }
125        Value::Scalar(ScalarValue::Float64(value)) if placement.pixel_axes.len() == 1 => {
126            if let Some(axis) = placement.pixel_axes[0] {
127                *value -= f64::from(origin[axis]);
128            }
129        }
130        _ => return Err(raised("Invalid coordinate reference pixel")),
131    }
132    Ok(())
133}

135fn shift_coordinate(
136    model: &CoordinateModel,
137    placement: &RecordCoordinate,
138    stored: &mut RecordValue,
139    origin: &[f32],
140    shape: &[i32],
141) -> PyResult<Option<f64>> {
142    match model {
143        CoordinateModel::Stokes(coordinate) => {
144            if let Some(axis) = placement.pixel_axes[0] {
145                select_discrete(
146                    coordinate.stokes(),
147                    "stokes",
148                    stored,
149                    origin[axis],
150                    shape.get(axis).copied(),
151                    |label| (label.name(), label.code()),
152                )?;
153            }
154        }
155        CoordinateModel::Quality(coordinate) => {
156            if let Some(axis) = placement.pixel_axes[0] {
157                select_discrete(
158                    coordinate.values(),
159                    "quality",
160                    stored,
161                    origin[axis],
162                    shape.get(axis).copied(),
163                    |label| (label.name(), label.code()),
164                )?;
165            }
166        }
167        CoordinateModel::Spectral(coordinate) => {
168            let Some(axis) = placement.pixel_axes[0] else {
169                return Ok(None);
170            };
171            if let Some(section) = coordinate
172                .subimage_samples(origin[axis], shape.get(axis).copied())
173                .map_err(raised)?
174            {
175                let mut record = CoordinateModel::from(section).to_record();
176                if let Some(native_type) = stored.get("nativeType") {
177                    record.upsert("nativeType", native_type.clone());
178                }
179                if record.get("conversion").is_none() {
180                    let mut conversion = default_spectral_conversion();
181                    if let Some(frame) = record.get("system") {
182                        conversion.upsert("system", frame.clone());
183                    }
184                    record.upsert("conversion", Value::Record(conversion));
185                }
186                *stored = record;
187                return coordinate.value_in_hz(1.0).map(Some).map_err(raised);
188            } else if let Some(Value::Record(wcs)) = stored.get_mut("wcs") {
189                shift_reference_pixel(wcs, placement, origin)?;
190            } else {
191                shift_reference_pixel(stored, placement, origin)?;
192            }
193        }
194        CoordinateModel::Tabular(coordinate) => {
195            if let Some(axis) = placement.pixel_axes[0] {
196                let mut coordinate = coordinate.clone();
197                let reference = coordinate.reference_pixel()[0] - f64::from(origin[axis]);
198                coordinate
199                    .edit_live_reference(TabularEdit::ReferencePixel(reference))
200                    .map_err(raised)?;
201                *stored = CoordinateModel::from(coordinate).to_record();
202            }
203        }
204        CoordinateModel::System(system) => {
205            let references = system.reference_pixel();
206            for (local, (axis, reference)) in
207                placement.pixel_axes.iter().zip(references).enumerate()
208            {
209                if let Some(axis) = axis {
210                    let shifted = reference - f64::from(origin[*axis]);
211                    CoordinateSystem::set_reference_pixel_in_record(stored, local, shifted)
212                        .map_err(raised)?;
213                }
214            }
215        }
216        _ => shift_reference_pixel(stored, placement, origin)?,
217    }
218    Ok(None)
219}

221fn subimage_record(
222    state: &CoordinateState,
223    origin: &[f32],
224    shape: &[i32],
225) -> PyResult<RecordValue> {
226    let rank = state.coordinates.n_pixel_axes();
227    if origin.len() != rank {
228        return Err(raised(
229            "Failed AlwaysAssert originShift.nelements() == nPixelAxes() && pixincFac.nelements() == nPixelAxes()",
230        ));
231    }
232    if !shape.is_empty() && shape.len() != rank {
233        return Err(raised(
234            "Failed AlwaysAssert nShape==0 || nShape==nPixelAxes()",
235        ));
236    }
237    let mut record = state.record.clone();
238    let placements = state
239        .coordinates
240        .coordinates_in_record(&record)
241        .map_err(raised)?;
242    for (index, placement) in placements.iter().enumerate() {
243        let Some(Value::Record(stored)) = record.get_mut(&placement.field) else {
244            return Err(raised("Missing coordinate record"));
245        };
246        let world_scale = shift_coordinate(
247            state.coordinates.coordinate(index),
248            placement,
249            stored,
250            origin,
251            shape,
252        )?;
253        if let Some(scale) = world_scale.filter(|scale| *scale != 1.0) {
254            let index = placement.field.trim_start_matches("spectral");
255            let field = format!("worldreplace{index}");
256            let Some(Value::Array(ArrayValue::Float64(values))) = record.get_mut(&field) else {
257                return Err(raised(format!("Invalid coordinate field {field}")));
258            };
259            values.mapv_inplace(|value| value * scale);
260        }
261    }
262    Ok(record)
263}

265#[pymethods]
266impl PyCoordsys {
267    #[pyo3(signature = (originshft=None, newshape=SubimageShape::default()), text_signature = "($self, originshft=[], newshape=[])")]
268    fn subimage(
269        &self,
270        py: Python<'_>,
271        originshft: Option<&Bound<'_, PyAny>>,
272        newshape: SubimageShape,
273    ) -> PyResult<Py<PyAny>> {
274        let origin = origin_input(originshft)?;
275        let record = py.detach(|| subimage_record(&self.state(), &origin, &newshape.0.values))?;
276        if newshape.0.overflowed {
277            let error = PySystemError::new_err(
278                "<built-in function coordsys_subimage> returned a result with an exception set",
279            );
280            error.set_cause(
281                py,
282                Some(PyOverflowError::new_err(
283                    "Python int too large to convert to C long",
284                )),
285            );
286            return Err(error);
287        }
288        values::record(&record, Path::new(".")).into_python(py)
289    }
