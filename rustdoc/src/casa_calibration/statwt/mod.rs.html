30/// Main-table data column contributing samples to a classical weight estimate.
31#[derive(Clone, Copy)]
32pub enum StatWtDataColumn {
33    /// Read the input visibilities from DATA.
34    Data,
35    /// Read the calibrated visibilities from CORRECTED_DATA.
36    Corrected,
37    /// Subtract MODEL_DATA from DATA.
38    ResidualData,
39    /// Subtract MODEL_DATA from CORRECTED_DATA.
40    ResidualCorrected,
41}
42
43impl StatWtDataColumn {
44    /// Interpret the case-insensitive CASA data-column selector.
45    pub fn parse(value: &str) -> MsResult<Self> {
46        let value = value.to_ascii_lowercase();
47        if value.starts_with('d') {
48            Ok(Self::Data)
49        } else if value.starts_with('c') {
50            Ok(Self::Corrected)
51        } else if value.starts_with("residual_") {
52            Ok(Self::ResidualData)
53        } else if value.starts_with("residual") {
54            Ok(Self::ResidualCorrected)
55        } else {
56            Err(MsError::InvalidInput(format!(
57                "unknown statwt data column {value:?}"
58            )))
59        }
60    }
61
62    fn visibility_column(self) -> &'static str {
63        match self {
64            Self::Data | Self::ResidualData => "DATA",
65            Self::Corrected | Self::ResidualCorrected => "CORRECTED_DATA",
66        }
67    }
68
69    fn residual(self) -> bool {
70        matches!(self, Self::ResidualData | Self::ResidualCorrected)
71    }
72
73    fn raw(self) -> bool {
74        matches!(self, Self::Data | Self::ResidualData)
75    }
76}

78/// How the task sets the width of a statistical time bin.
79#[derive(Clone, Copy)]
80pub enum StatWtTimeBin {
81    /// Count distinct timestamps in the selected time series.
82    Timestamps(usize),
83    /// Bin using the elapsed time in seconds.
84    Seconds(f64),
85}
86
87impl StatWtTimeBin {
88    /// Read a positive duration with time units.
89    pub fn duration(value: &str) -> MsResult<Self> {
90        physical_width(value, "s", "timebin").map(Self::Seconds)
91    }
92}

94/// Boundaries and correlations to combine within a time bin.
95#[derive(Clone, Copy, Default)]
96pub struct StatWtCombine {
97    /// Combine all visibility correlations.
98    pub correlations: bool,
99    /// Cross scan boundaries.
100    pub scans: bool,
101    /// Cross observational state boundaries.
102    pub states: bool,
103    /// Cross field boundaries.
104    pub fields: bool,
105}
106
107impl StatWtCombine {
108    /// Parse the CASA combination selector.
109    pub fn parse(value: &str) -> Self {
110        let folded = value.to_ascii_lowercase();
111        Self {
112            correlations: value.contains("corr"),
113            scans: folded.contains("scan"),
114            states: folded.contains("state"),
115            fields: folded.contains("field"),
116        }
117    }
118}

120/// How channels within a spectral window contribute to each estimate.
121#[derive(Clone, Copy)]
122pub enum StatWtChannelBin {
123    /// One bin spans the spectral window.
124    Window,
125    /// Bin a fixed count of adjacent channels.
126    Channels(usize),
127    /// Bin channels up to a frequency difference from the first channel.
128    Hertz(f64),
129}
130
131impl StatWtChannelBin {
132    /// Read the all-channel marker or a positive quantity with frequency units.
133    pub fn parse(value: &str) -> MsResult<Self> {
134        if value == "spw" {
135            Ok(Self::Window)
136        } else {
137            physical_width(value, "Hz", "chanbin").map(Self::Hertz)
138        }
139    }
140}

142fn physical_width(value: &str, unit: &str, argument: &str) -> MsResult<f64> {
143    let invalid = || MsError::InvalidInput(format!("invalid statwt {argument} {value:?}"));
144    let unit = Unit::new(unit).expect("known unit");
145    let quantity: Quantity = value.parse().map_err(|_| invalid())?;
146    let width = quantity.get_value_in(&unit).map_err(|_| invalid())?;
147    if width <= 0.0 || !width.is_finite() {
148        return Err(invalid());
149    }
150    Ok(width)
151}

153/// Options for visibility weight estimation.
154#[derive(Clone)]
155pub struct StatWtOptions {
156    /// The main-table data to estimate weights from.
157    pub data_column: StatWtDataColumn,
158    /// Estimate classical moments or use a robust algorithm.
159    pub algorithm: StatWtAlgorithm,
160    /// The minimum number of unflagged samples in a bin.
161    pub minimum_samples: usize,
162    /// Combine correlations or data across selected boundaries.
163    pub combine: StatWtCombine,
164    /// The number of timestamps or seconds per time bin.
165    pub time_bin: StatWtTimeBin,
166    /// Use a centered window for each timestamp.
167    pub sliding: bool,
168    /// Channel bins used to estimate and apply each weight.
169    pub channel_bin: StatWtChannelBin,
170    /// CASA spectral-window and channel expression restricting the fit.
171    pub fit_spw: String,
172    /// Use the complement of the fit channel selection within named windows.
173    pub exclude_channels: bool,
174    /// Optional inclusive bounds on acceptable weights.
175    pub weight_range: Option<[f64; 2]>,
176}

178/// Interpret CASA's optional inclusive `wtrange`.
179pub fn statwt_weight_range(bounds: &[f64]) -> MsResult<Option<[f64; 2]>> {
180    match bounds {
181        [] => Ok(None),
182        &[left, right] if left >= 0.0 && right >= 0.0 && left != right => {
183            Ok(Some([left.min(right), left.max(right)]))
184        }
185        _ => Err(MsError::InvalidInput(
186            "wtrange requires two distinct nonnegative bounds".into(),
187        )),
188    }
189}

200/// Estimate scalar and spectral weights in block or sliding time bins.
201///
202/// The caller supplies the MeasurementSet selection held by either the task
203/// or the existing MS tool. Pass a MAIN write lock to persist weights and
204/// flags. Without a write lock this reports the same statistics in preview.
205pub fn estimate_statwt(
206    ms: &mut StatWtMs,
207    selection: &[ResolvedMsSelectionRow],
208    options: &StatWtOptions,
209    mut locks: Option<&mut BTreeMap<PathBuf, TableWriteLock>>,
210) -> MsResult<StatWtSummary> {
211    if selection.is_empty() {
212        return Err(MsError::InvalidInput(
213            "MeasurementSet has zero selected rows".into(),
214        ));
215    }
216    if options.minimum_samples < 2 {
217        return Err(MsError::InvalidInput("minsamp must be at least 2".into()));
218    }
219    if matches!(options.time_bin, StatWtTimeBin::Timestamps(0))
220        || matches!(options.time_bin, StatWtTimeBin::Seconds(value) if value <= 0.0 || !value.is_finite())
221    {
222        return Err(MsError::InvalidInput(
223            "statwt time bin must be positive".into(),
224        ));
225    }
226    if matches!(options.channel_bin, StatWtChannelBin::Channels(0))
227        || matches!(options.channel_bin, StatWtChannelBin::Hertz(value) if value <= 0.0 || !value.is_finite())
228    {
229        return Err(MsError::InvalidInput(
230            "statwt channel bin must be positive".into(),
231        ));
232    }
233    let channel_plans = channels::prepare(ms.metadata(), selection, options)?;
234    let models = options
235        .data_column
236        .residual()
237        .then(|| ms.models(selection))
238        .transpose()?;
239    let groups = grouping::groups(
240        ms,
241        selection,
242        options.combine,
243        options.time_bin,
244        options.sliding,
245    )?;
246    let mut summary = Moments::default();
247    let mut pending = Vec::new();
248    let mut pending_bytes = 0;
249    let output = ms.output(selection, options, locks.as_deref_mut())?;
250    let columns = output.columns();
251    let polarizations = ms.metadata().polarization()?;
252    let correlation_count = (0..polarizations.row_count())
253        .map(|row| polarizations.num_corr(row).map(|count| count as usize))
254        .collect::<MsResult<Vec<_>>>()?;
255    let cell_bytes =
256        size_of::<Complex32>() * (1 + usize::from(models.is_some())) + size_of::<bool>();
257    let mut remaining = groups.as_slice();
258    while !remaining.is_empty() {
259        let mut bytes = 0;
260        let mut count = 0;
261        for group in remaining {
262            let row = group.rows[0];
263            let channels = channel_plans[&row.spectral_window_id].fit.len();
264            let correlations = correlation_count
265                .get(row.polarization_id as usize)
266                .copied()
267                .ok_or_else(|| MsError::InvalidInput("invalid POLARIZATION_ID".into()))?;
268            let group_bytes = group
269                .rows
270                .len()
271                .saturating_mul(128 + channels * correlations * cell_bytes);
272            if count > 0 && bytes + group_bytes > casa_ms::TARGET_BATCH_BYTES {
273                break;
274            }
275            bytes = bytes.saturating_add(group_bytes);
276            count += 1;
277        }
278        let (batch, rest) = remaining.split_at(count);
279        remaining = rest;
280        let rows = batch
281            .iter()
282            .flat_map(|group| group.rows.iter().copied())
283            .collect::<Vec<_>>();
284        let samples = read_samples(ms, options.data_column, models.as_deref(), &rows)?;
285        let mut offset = 0;
286        for group in batch {
287            let channel_plan = &channel_plans[&group.rows[0].spectral_window_id];
288            let samples = &samples[offset..offset + group.rows.len()];
289            offset += samples.len();
290            let shape = samples[0].values.shape();
291            if shape[1] != channel_plan.fit.len()
292                || samples.iter().any(|sample| sample.values.shape() != shape)
293            {
294                return Err(MsError::InvalidInput(
295                    "visibility shape disagrees with spectral window".into(),
296                ));
297            }
298            let estimate = |range: std::ops::Range<usize>| {
299                estimate_variances(
300                    &samples[range.clone()],
301                    &group.exposures[range],
302                    channel_plan,
303                    options,
304                )
305            };
306            let mut previous_window = 0..0;
307            let mut variances = Vec::new();
308            let mut weights = Vec::new();
309            for (index, (&row, sample)) in group.rows.iter().zip(samples).enumerate() {
310                let window = group.window(index, options.time_bin, options.sliding);
311                if window != previous_window {
312                    variances = estimate(window.clone());
313                    previous_window = window;
314                }
315                let exposure = group.exposures[index];
316                weights.clear();
317                weights.extend(variances.iter().map(|&variance| {
318                    if variance == 0.0 {
319                        0.0
320                    } else {
321                        (exposure / variance) as f32
322                    }
323                }));
324                add_summary(
325                    &mut summary,
326                    sample,
327                    &weights,
328                    channel_plan,
329                    output.spectral,
330                    options.combine.correlations,
331                );
332                if let Some(locks) = &mut locks {
333                    let use_duration_scalar = !output.spectral
334                        && matches!(options.time_bin, StatWtTimeBin::Seconds(_))
335                        && !options.sliding;
336                    let update = RowUpdate::new(
337                        row.row_index,
338                        sample,
339                        &weights,
340                        channel_plan,
341                        &output,
342                        options,
343                        use_duration_scalar.then_some((variances.as_slice(), exposure)),
344                    );
345                    pending_bytes += update.flags.len()
346                        + update
347                            .values
348                            .iter()
349                            .map(|value| value.len() * size_of::<f32>())
350                            .sum::<usize>();
351                    pending.push(update);
352                    if pending_bytes >= casa_ms::TARGET_BATCH_BYTES || pending.len() == 65_536 {
353                        ms.persist(&columns, &mut pending, locks)?;
354                        pending_bytes = 0;
355                    }
356                }
357            }
358        }
359    }
360    if let Some(locks) = locks {
361        ms.persist(&columns, &mut pending, locks)?;
362    }
363    Ok(StatWtSummary {
364        mean: summary.mean(),
365        variance: summary.sample_variance(),
366    })
367}

369fn add_summary(
370    summary: &mut Moments,
371    sample: &Samples,
372    weights: &[f32],
373    channels: &Channels,
374    spectral: bool,
375    combine: bool,
376) {
377    let correlations = sample.flags.len_of(Axis(0));
378    for (bin, bounds) in channels.bins.iter().enumerate() {
379        let count = |correlation| {
380            bounds
381                .clone()
382                .filter(|&channel| !sample.flags[[correlation, channel]])
383                .count()
384        };
385        for (correlation, &weight) in weights[bin * correlations..][..correlations]
386            .iter()
387            .enumerate()
388        {
389            let unflagged = if !spectral && combine {
390                if correlation != 0 {
391                    continue;
392                }
393                usize::from((0..correlations).any(|corr| count(corr) != 0))
394            } else if spectral {
395                count(correlation)
396            } else {
397                usize::from(count(correlation) != 0)
398            };
399            summary.add_repeated(f64::from(weight), unflagged);
400        }
401    }
402}

404struct RowUpdate {
405    row: usize,
406    values: Vec<ArrayValue>,
407    flags: ArrayValue,
408    row_flagged: bool,
409}
410
411impl RowUpdate {
412    fn new(
413        row: usize,
414        sample: &Samples,
415        weights: &[f32],
416        channels: &Channels,
417        output: &Output,
418        options: &StatWtOptions,
419        duration_scalar: Option<(&[f64], f64)>,
420    ) -> Self {
421        let mut flags = sample.flags.clone();
422        let correlations = flags.len_of(Axis(0));
423        let mut spectrum = output
424            .spectral
425            .then(|| ArrayD::zeros(IxDyn(flags.shape()).f()));
426        for (bin, bounds) in channels.bins.iter().enumerate() {
427            for correlation in 0..correlations {
428                let weight = weights[bin * correlations + correlation];
429                let rejected = weight == 0.0
430                    || options.weight_range.is_some_and(|[low, high]| {
431                        f64::from(weight) < low || f64::from(weight) > high
432                    });
433                for channel in bounds.clone() {
434                    flags[[correlation, channel]] |= rejected;
435                    if let Some(spectrum) = &mut spectrum {
436                        spectrum[[correlation, channel]] = if rejected { 0.0 } else { weight };
437                    }
438                }
439            }
440        }
441        let scalar_weights = if let Some(spectrum) = &spectrum {
442            spectral_medians(spectrum, &flags, options.combine.correlations)
443        } else if let Some((variances, exposure)) = duration_scalar {
444            // The duration-block scalar path divides exposure by the bin variance directly.
445            variances
446                .iter()
447                .map(|value| (exposure / value) as f32)
448                .collect()
449        } else {
450            weights.to_vec()
451        };
452        Self {
453            row,
454            values: output.cells(&scalar_weights, spectrum.as_ref()),
455            row_flagged: flags.iter().all(|&flag| flag),
456            flags: ArrayValue::Bool(flags),
457        }
458    }
459}

461fn spectral_medians(spectrum: &ArrayD<f32>, flags: &ArrayD<bool>, combine: bool) -> Vec<f32> {
462    let correlations = flags.len_of(Axis(0));
463    let channels = flags.len_of(Axis(1));
464    let mut values = Vec::with_capacity(channels);
465    let mut median = |selected: std::ops::Range<usize>| {
466        values.clear();
467        for correlation in selected {
468            values.extend((0..channels).filter_map(|channel| {
469                (!flags[[correlation, channel]]).then_some(spectrum[[correlation, channel]])
470            }));
471        }
472        let midpoint = values.len() / 2;
473        let even = values.len() % 2 == 0;
474        if values.is_empty() {
475            return 0.0;
476        }
477        let (lower, middle, _) = values.select_nth_unstable_by(midpoint, f32::total_cmp);
478        if even {
479            (0.5 * (f64::from(*middle)
480                + f64::from(*lower.iter().max_by(|a, b| a.total_cmp(b)).unwrap())))
481                as f32
482        } else {
483            *middle
484        }
485    };
486    if combine {
487        vec![median(0..correlations); correlations]
488    } else {
489        (0..correlations)
490            .map(|correlation| median(correlation..correlation + 1))
491            .collect()
492    }
493}

495fn persist_updates(
496    table: &mut Table,
497    weight_columns: &[&str],
498    pending: &mut Vec<RowUpdate>,
499    lock: &mut TableWriteLock,
500) -> MsResult<()> {
501    if pending.is_empty() {
502        return Ok(());
503    }
504    let mut columns = weight_columns.to_vec();
505    columns.extend(["FLAG", "FLAG_ROW"]);
506    let rows: Vec<_> = pending.iter().map(|update| update.row).collect();
507    let mut writer = table.row_accessor_mut().prepare(&columns)?;
508    for update in pending.drain(..) {
509        writer.seek(update.row)?;
510        for (index, value) in update.values.into_iter().enumerate() {
511            writer.set_value_at(index, Value::Array(value))?;
512        }
513        writer.set_value_at(weight_columns.len(), Value::Array(update.flags))?;
514        writer.set_value_at(
515            weight_columns.len() + 1,
516            Value::Scalar(ScalarValue::Bool(update.row_flagged)),
517        )?;
518    }
519    writer.flush()?;
520    drop(writer);
521    lock.record_write();
522    table.prepare_write().save_selected_rows(&columns, &rows)?;
523    table.discard_persisted_cell_updates(&columns, &rows);
524    Ok(())
525}

532fn read_samples(
533    ms: &StatWtMs,
534    column: StatWtDataColumn,
535    models: Option<&[VisibilityModels]>,
536    integration: &[&ResolvedMsSelectionRow],
537) -> MsResult<Vec<Samples>> {
538    let indices = integration
539        .iter()
540        .map(|row| row.row_index)
541        .collect::<Vec<_>>();
542    let read = |name| ms.array_cells(name, &indices);
543    let data = read(column.visibility_column())?;
544    let flags = read("FLAG")?;
545    let scratch = models
546        .is_some_and(|models| {
547            integration
548                .iter()
549                .any(|row| models[ms.by_row(row.row_index).0].reads_column(row.field_id))
550        })
551        .then(|| read("MODEL_DATA"))
552        .transpose()?;
553    let mut samples = Vec::with_capacity(integration.len());
554    for (index, (values, flags)) in data.into_iter().zip(flags).enumerate() {
555        let row = integration[index];
556        let (Some(ArrayValue::Complex32(mut values)), Some(ArrayValue::Bool(flags))) =
557            (values, flags)
558        else {
559            return Err(MsError::InvalidInput(format!(
560                "visibility or flag cell is undefined at row {}",
561                row.row_index
562            )));
563        };
564        if values.ndim() != 2 || values.shape() != flags.shape() {
565            return Err(MsError::InvalidInput(format!(
566                "visibility and flag shapes disagree at row {}",
567                row.row_index
568            )));
569        }
570        let model = if let Some(models) = models {
571            let (part, physical_ms, physical_row) = ms.by_row(row.row_index);
572            let model = &models[part];
573            if model.reads_column(row.field_id) {
574                match &scratch.as_ref().expect("read scratch model for its field")[index] {
575                    Some(ArrayValue::Complex32(model)) => Some(std::borrow::Cow::Borrowed(model)),
576                    _ => {
577                        return Err(MsError::InvalidInput(format!(
578                            "MODEL_DATA is undefined at row {}",
579                            row.row_index
580                        )));
581                    }
582                }
583            } else {
584                let codes = correlation_types_for_ddid(physical_ms, row.data_desc_id)
585                    .map_err(|error| MsError::InvalidInput(error.to_string()))?;
586                model
587                    .predict(physical_ms, physical_row, &codes, &flags)
588                    .map_err(|error| MsError::InvalidInput(error.to_string()))?
589                    .map(std::borrow::Cow::Owned)
590            }
591        } else {
592            None
593        };
594        if let Some(model) = model {
595            if model.shape() != values.shape() {
596                return Err(MsError::InvalidInput(format!(
597                    "MODEL_DATA has a different shape at row {}",
598                    row.row_index
599                )));
600            }
601            values -= model.as_ref();
602        }
603        samples.push(Samples { values, flags });
604    }
605    Ok(samples)
606}

608fn estimate_variances(
609    samples: &[Samples],
610    exposures: &[f64],
611    channels: &Channels,
612    options: &StatWtOptions,
613) -> Vec<f64> {
614    let correlations = samples[0].values.len_of(Axis(0));
615    let combine = options.combine.correlations;
616    let mut variances = Vec::with_capacity(channels.bins.len() * correlations);
617    let mut estimates = vec![VisibilityMoments::default(); if combine { 1 } else { correlations }];
618    let mut counts = vec![0usize; estimates.len()];
619    let mut robust_samples =
620        (!options.algorithm.uses_classical_moments()).then(|| vec![Vec::new(); estimates.len()]);
621    for bounds in &channels.bins {
622        estimates.fill(VisibilityMoments::default());
623        counts.fill(0);
624        if let Some(samples) = &mut robust_samples {
625            for sample in samples {
626                sample.clear();
627            }
628        }
629        for (row, &exposure) in samples.iter().zip(exposures) {
630            for channel in bounds.clone().filter(|&channel| channels.fit[channel]) {
631                for correlation in 0..correlations {
632                    let bin = if combine { 0 } else { correlation };
633                    if !row.flags[[correlation, channel]] {
634                        counts[bin] += 1;
635                        let visibility = row.values[[correlation, channel]];
636                        if let Some(samples) = &mut robust_samples {
637                            samples[bin].push((visibility, exposure));
638                        } else {
639                            estimates[bin].add(visibility, exposure);
640                        }
641                    }
642                }
643            }
644        }
645        let bin_start = variances.len();
646        variances.extend((0..estimates.len()).map(|correlation| {
647            if counts[correlation] < options.minimum_samples {
648                0.0
649            } else if let Some(samples) = &robust_samples {
650                options.algorithm.variance(&samples[correlation])
651            } else {
652                estimates[correlation].variance()
653            }
654        }));
655        if combine {
656            variances.resize(bin_start + correlations, variances[bin_start]);
657        }
658    }
659    variances
660}
