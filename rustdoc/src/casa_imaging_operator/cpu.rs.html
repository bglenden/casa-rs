22/// The CPU implementation of [`GridBackend`].
23///
24/// Grids are addressed through the shared accumulator layout; each sample
25/// is located once with [`GridGeometry::locate`](crate::GridGeometry::locate)
26/// and spread or gathered through the Mueller table its `w` sign selects.
27/// Under [`KernelNormalisation::KernelSum`] a prediction sums the numerator
28/// and the kernel norm over every routed Mueller plane and divides once, as
29/// CASA's `AWVisResampler` does, and a zero norm predicts zero; the other
30/// rules predict the gathered sum.
31#[derive(Debug, Default)]
32pub struct CpuBackend {
33    powers: Vec<f64>,
34    prediction: Vec<Complex64>,
35    norms: Vec<Complex64>,
36    /// One grid polarization's gathers by Mueller plane, each summed over
37    /// the terms, for the visibility polarizations that share them.
38    gathered: Vec<(u8, Complex64)>,
39    residual: Vec<Complex32>,
40    /// A block's predictions, or its residual values at native channels.
41    batch: Vec<Complex32>,
42    /// The gridding's kernel cell.
43    hold: CellHold,
44    /// The prediction's kernel cell, apart from the gridding's: a residual
45    /// reads both for every sample, and one slot would fetch each anew per
46    /// sample, a lock and a shared count for every set that pages its cells
47    /// (AW), where two let a run of samples of one key fetch once (#449).
48    prediction_hold: CellHold,
49}
50
51impl CpuBackend {
52    /// A backend with empty scratch buffers.
53    #[must_use]
54    pub fn new() -> Self {
55        Self::default()
56    }
57
58    /// A backend for predictions of blocks of at most `samples` placements
59    /// of `npol` correlations from a model of `terms` Taylor terms, its
60    /// buffers allocated once: its per-sample buffers
61    /// ([`Self::prediction_sample_bytes`]) and its batch
62    /// ([`Self::batch_bytes`]), which such blocks never grow
63    /// ([`Work::PredictNative`]).
64    #[must_use]
65    pub fn for_prediction(npol: usize, terms: usize, samples: usize) -> Self {
66        Self {
67            powers: Vec::with_capacity(terms),
68            prediction: Vec::with_capacity(npol),
69            norms: Vec::with_capacity(npol),
70            // One gather per Mueller plane a grid polarization routes, at
71            // most one per visibility polarization.
72            gathered: Vec::with_capacity(npol),
73            batch: Vec::with_capacity(samples * npol),
74            ..Self::default()
75        }
76    }
77
78    /// Bytes [`Self::for_prediction`] allocates beside its batch.
79    #[must_use]
80    pub const fn prediction_sample_bytes(npol: usize, terms: usize) -> u64 {
81        (terms * size_of::<f64>()
82            + 2 * npol * size_of::<Complex64>()
83            + npol * size_of::<(u8, Complex64)>()) as u64
84    }
85
86    /// Bytes a gridding call holds while it spreads a placement whose
87    /// support `cf` clips to the grid ([`GridEdge::Clipped`]): one block of
88    /// the set's largest support in `precision`, freed when the call
89    /// returns; none for a set that drops such placements.
90    #[must_use]
91    pub fn clip_bytes(cf: &dyn ConvolutionFunctionSet, precision: GridPrecision) -> u64 {
92        if cf.grid_edge() == GridEdge::Whole {
93            return 0;
94        }
95        let [hx, hy] = cf.max_half_support().map(|half| 2 * usize::from(half) + 1);
96        let cell = match precision {
97            GridPrecision::F32 => size_of::<Complex<f32>>(),
98            GridPrecision::F64 => size_of::<Complex<f64>>(),
99        };
100        (hx * hy * cell) as u64
101    }
102
103    /// Bytes the backend keeps after blocks of at most `samples` placements
104    /// of `npol` correlations: its batch, which holds a block's native
105    /// predictions ([`Work::PredictNative`]) or its residual values at
106    /// native channels ([`Work::NativeResidualGrid`]) and is sized to the
107    /// largest block it has served.
108    #[must_use]
109    pub const fn batch_bytes(npol: usize, samples: usize) -> u64 {
110        (samples * npol * size_of::<Complex32>()) as u64
111    }

113    fn grid<T: GridScalar, const COMPENSATED: bool>(
114        &mut self,
115        block: &SampleBlock<'_>,
116        cf: &dyn ConvolutionFunctionSet,
117        mode: Mode,
118        acc: &mut GridAccumulator,
119    ) -> Result<(), OperatorError> {
120        let CpuGridParts {
121            layout,
122            cells,
123            correction,
124            mut sumwt,
125        } = acc.parts_mut::<T>();
126        let terms = layout
127            .term_range(mode)
128            .unwrap_or_else(|| panic!("accumulator holds no {mode:?} terms"));
129        let rule = cf.normalisation();
130        let Self { powers, hold, .. } = self;
131        // A clipped spread's block, at most `(2h + 1)²` cells of the set's
132        // largest half support `h`, allocated only when one is needed.
133        let mut scratch = Vec::new();
134        for (index, placement) in block.placements.iter().enumerate() {
135            let taps = match mode {
136                Mode::Weight => cf
137                    .weight_taps(placement.cf, hold)
138                    .ok_or(OperatorError::WeightKernelUnavailable { key: placement.cf })?,
139                Mode::Psf => cf.psf_taps(placement.cf, hold),
140                Mode::Data => cf.taps(placement.cf, hold),
141            };
142            let (u, v) = match mode {
143                Mode::Weight => (0.0, 0.0),
144                Mode::Data | Mode::Psf => (placement.u, placement.v),
145            };
146            let location = layout.geometry().locate(u, v, taps.oversampling());
147            // CASA tests `onGrid` with the support of the cell each mode
148            // reads (`AWVisResampler::DataToGrid`), so a PSF gridded with a
149            // wider weight cell drops rows the data gridding keeps.
150            let reach = Reach::of(
151                cf,
152                layout.geometry(),
153                (mode, placement),
154                location,
155                (taps.half_support(), taps.oversampling()),
156            );
157            if reach == Reach::Off {
158                continue;
159            }
160            let (sumwt, clip) = targets(reach, sumwt.as_deref_mut(), &mut scratch);
161            spectral_powers(powers, placement.spectral, terms.len());
162            spread_sample::<T, COMPENSATED>(
163                layout,
164                cells,
165                correction,
166                sumwt,
167                cf,
168                rule,
169                placement,
170                &taps,
171                location,
172                mode,
173                &terms,
174                powers,
175                block.values_of(index),
176                block.weights_of(index),
177                clip,
178            );
179        }
180        Ok(())
181    }

183    fn predict<T: GridScalar>(
184        &mut self,
185        block: &SampleBlock<'_>,
186        cf: &dyn ConvolutionFunctionSet,
187        model: &PreparedModelGrids,
188        out: &mut [Complex32],
189    ) {
190        let npol = block.npol;
191        assert_eq!(out.len(), block.len() * npol, "prediction output length");
192        for (index, placement) in block.placements.iter().enumerate() {
193            self.predict_sample::<T>(cf, model, placement, npol);
194            let phasor = Complex64::from_polar(1.0, -placement.phase);
195            for (vpol, target) in out[index * npol..(index + 1) * npol].iter_mut().enumerate() {
196                *target = narrow(self.prediction[vpol] * phasor);
197            }
198        }
199    }

201    fn residual<T: GridScalar, const COMPENSATED: bool>(
202        &mut self,
203        block: &SampleBlock<'_>,
204        cf: &dyn ConvolutionFunctionSet,
205        model: &PreparedModelGrids,
206        acc: &mut GridAccumulator,
207        mut residual_out: Option<&mut [Complex32]>,
208    ) {
209        let npol = block.npol;
210        if let Some(out) = residual_out.as_deref() {
211            assert_eq!(out.len(), block.len() * npol, "residual output length");
212        }
213        let rule = cf.normalisation();
214        let CpuGridParts {
215            layout,
216            cells,
217            correction,
218            mut sumwt,
219        } = acc.parts_mut::<T>();
220        let terms = layout
221            .term_range(Mode::Data)
222            .expect("residual gridding needs the data terms");
223        self.residual.resize(npol, Complex32::default());
224        // A clipped spread's block, as `grid`'s.
225        let mut scratch = Vec::new();
226        for (index, placement) in block.placements.iter().enumerate() {
227            self.predict_sample::<T>(cf, model, placement, npol);
228            let values = block.values_of(index);
229            let weights = block.weights_of(index);
230            for vpol in 0..npol {
231                let weight = f64::from(weights[vpol]);
232                let predicted = self.prediction[vpol] * weight;
233                self.residual[vpol] = values[vpol] - narrow(predicted);
234            }
235            if let Some(out) = residual_out.as_deref_mut() {
236                let phasor = Complex64::from_polar(1.0, -placement.phase);
237                for vpol in 0..npol {
238                    let weight = f64::from(weights[vpol]);
239                    out[index * npol + vpol] = if weight == 0.0 {
240                        Complex32::default()
241                    } else {
242                        narrow(widen(self.residual[vpol]) * phasor / weight)
243                    };
244                }
245            }
246            let Self {
247                powers,
248                residual,
249                hold,
250                ..
251            } = self;
252            let taps = cf.taps(placement.cf, hold);
253            let location = layout
254                .geometry()
255                .locate(placement.u, placement.v, taps.oversampling());
256            // A placement reaches the grid (the resampler kept it); a mosaic
257            // predicts nothing for one it clips, and grids the residual's
258            // taps that fall on the grid.
259            let reach = Reach::of(
260                cf,
261                layout.geometry(),
262                (Mode::Data, placement),
263                location,
264                (taps.half_support(), taps.oversampling()),
265            );
266            if reach == Reach::Off {
267                continue;
268            }
269            let (sumwt, clip) = targets(reach, sumwt.as_deref_mut(), &mut scratch);
270            spectral_powers(powers, placement.spectral, terms.len());
271            spread_sample::<T, COMPENSATED>(
272                layout,
273                cells,
274                correction,
275                sumwt,
276                cf,
277                rule,
278                placement,
279                &taps,
280                location,
281                Mode::Data,
282                &terms,
283                powers,
284                residual,
285                weights,
286                clip,
287            );
288        }
289    }
290
291    /// Fill `self.prediction` with the raw prediction `P` per visibility
292    /// polarization (no phase-centre phasor) for one placement:
293    /// `Σ_t s^t Σ_{gpol,m} Σ conj(tap'_m) · model[t][gpol]`, divided once by
294    /// `Σ_{gpol,m} norm_m` under [`KernelNormalisation::KernelSum`]. The
295    /// gather of a grid polarization through a Mueller plane depends on
296    /// neither visibility polarization, so it is made once and added to
297    /// each that the plane routes (both correlations of Stokes I).
298    fn predict_sample<T: GridScalar>(
299        &mut self,
300        cf: &dyn ConvolutionFunctionSet,
301        model: &PreparedModelGrids,
302        placement: &Placement,
303        npol: usize,
304    ) {
305        let layout = model.layout();
306        let cells = T::cells(model.storage());
307        let terms = layout
308            .term_range(Mode::Data)
309            .expect("model grids hold the data terms");
310        let Self {
311            powers,
312            prediction,
313            norms,
314            gathered,
315            prediction_hold,
316            ..
317        } = self;
318        let taps = cf.prediction_taps(placement.cf, prediction_hold);
319        let location = layout
320            .geometry()
321            .locate(placement.u, placement.v, taps.oversampling());
322        if !layout.geometry().fits(location, taps.half_support()) {
323            // `GridToData` skips a row whose prediction cell overruns the
324            // grid; its model visibility stays zero.
325            prediction.clear();
326            prediction.resize(npol, Complex64::default());
327            return;
328        }
329        let w_positive = placement.prediction_w_positive;
330        let table = cf.mueller().table(w_positive, true);
331        let plane = layout.planes().local(placement.plane);
332        let divide = cf.normalisation() == KernelNormalisation::KernelSum;
333        spectral_powers(powers, placement.spectral, terms.len());
334        prediction.clear();
335        prediction.resize(npol, Complex64::default());
336        norms.clear();
337        norms.resize(npol, Complex64::default());
338        for (gpol, row) in table.iter().enumerate() {
339            gathered.clear();
340            for (vpol, mueller_plane) in row.iter().enumerate() {
341                let Some(mueller) = *mueller_plane else {
342                    continue;
343                };
344                if divide {
345                    // `accumulateFromGrid.inc` conjugates the prediction tap
346                    // for w ≤ 0 and sums that same tap into its norm: the
347                    // forward norm carries the opposite conjugation from
348                    // the adjoint's (`w_positive`), as the gather does.
349                    norms[vpol] += kernel::norm(&taps, location, mueller, !w_positive);
350                }
351                let sum = if let Some((_, sum)) = gathered.iter().find(|(m, _)| *m == mueller) {
352                    *sum
353                } else {
354                    let mut sum = Complex64::default();
355                    for (power, term) in powers.iter().zip(terms.clone()) {
356                        let offset = layout.block_offset(plane, gpol, term);
357                        let grid = &cells[offset..offset + layout.block_cells()];
358                        sum += kernel::gather::<T>(
359                            grid,
360                            layout.tile(),
361                            location,
362                            &taps,
363                            mueller,
364                            w_positive,
365                            placement.gradient,
366                        ) * *power;
367                    }
368                    gathered.push((mueller, sum));
369                    sum
370                };
371                prediction[vpol] += sum;
372            }
373        }
374        if divide {
375            for (prediction, norm) in prediction.iter_mut().zip(norms.iter()) {
376                *prediction = if *norm == Complex64::default() {
377                    Complex64::default()
378                } else {
379                    *prediction / norm
380                };
381            }
382        }
383    }
384}

386impl GridBackend for CpuBackend {
387    fn apply(
388        &mut self,
389        block: &SampleBlock<'_>,
390        cf: &dyn ConvolutionFunctionSet,
391        work: Work<'_>,
392    ) -> Result<(), OperatorError> {
393        assert_eq!(
394            block.npol,
395            cf.mueller().visibility_pols(),
396            "block polarizations must match the kernel set routing"
397        );
398        let result = match work {
399            Work::Grid { mode, acc } => match acc.precision() {
400                GridPrecision::F32 => self.grid::<f32, true>(block, cf, mode, acc),
401                GridPrecision::F64 => self.grid::<f64, false>(block, cf, mode, acc),
402            },
403            Work::Predict { model, out } => {
404                match model.precision() {
405                    GridPrecision::F32 => self.predict::<f32>(block, cf, model, out),
406                    GridPrecision::F64 => self.predict::<f64>(block, cf, model, out),
407                }
408                Ok(())
409            }
410            Work::ResidualGrid {
411                model,
412                acc,
413                residual_out,
414            } => {
415                assert_eq!(
416                    model.precision(),
417                    acc.precision(),
418                    "model grids and accumulator must share a precision"
419                );
420                match acc.precision() {
421                    GridPrecision::F32 => {
422                        self.residual::<f32, true>(block, cf, model, acc, residual_out);
423                    }
424                    GridPrecision::F64 => {
425                        self.residual::<f64, false>(block, cf, model, acc, residual_out);
426                    }
427                }
428                Ok(())
429            }
430            Work::PredictNative {
431                grids,
432                terms,
433                model,
434                add,
435            } => {
436                let NativeTarget::Host { cells, first } = model else {
437                    panic!("the CPU backend forms the model in host cells");
438                };
439                let mut predicted = std::mem::take(&mut self.batch);
440                predicted.clear();
441                // Exactly the block: the pass charges the largest block's
442                // batch ([`CpuBackend::batch_bytes`]).
443                predicted.reserve_exact(block.len() * block.npol);
444                predicted.resize(block.len() * block.npol, Complex32::default());
445                match grids.precision() {
446                    GridPrecision::F32 => self.predict::<f32>(block, cf, grids, &mut predicted),
447                    GridPrecision::F64 => self.predict::<f64>(block, cf, grids, &mut predicted),
448                }
449                interpolate_terms(&predicted, block.npol, terms, cells, first, add);
450                self.batch = predicted;
451                Ok(())
452            }
453            Work::NativeResidualGrid { model, acc } => self.native_residual(block, cf, model, acc),
454        };
455        self.hold.clear();
456        self.prediction_hold.clear();
457        result
458    }
459}

461impl CpuBackend {
462    /// [`Work::NativeResidualGrid`]: the block's values less the model at
463    /// their native endpoints, gridded as data.
464    fn native_residual(
465        &mut self,
466        block: &SampleBlock<'_>,
467        cf: &dyn ConvolutionFunctionSet,
468        model: NativeModel<'_>,
469        acc: &mut GridAccumulator,
470    ) -> Result<(), OperatorError> {
471        let NativeModel::Host { cells, first } = model else {
472            panic!("the CPU backend reads the model from host cells");
473        };
474        assert_eq!(
475            block.sources.len(),
476            block.len(),
477            "a native residual block holds every placement's endpoints"
478        );
479        let npol = block.npol;
480        let mut values = std::mem::take(&mut self.batch);
481        values.clear();
482        values.reserve_exact(block.len() * npol);
483        values.resize(block.len() * npol, Complex32::default());
484        for (index, (source, out)) in block
485            .sources
486            .iter()
487            .zip(values.chunks_exact_mut(npol))
488            .enumerate()
489        {
490            residual_values(block.values_of(index), source, cells, first, out);
491        }
492        let residual = SampleBlock {
493            values: &values,
494            ..*block
495        };
496        let result = match acc.precision() {
497            GridPrecision::F32 => self.grid::<f32, true>(&residual, cf, Mode::Data, acc),
498            GridPrecision::F64 => self.grid::<f64, false>(&residual, cf, Mode::Data, acc),
499        };
500        self.batch = values;
501        result
502    }
503}

505/// Spread one placement's values through every routed (grid pol,
506/// visibility pol) pair and term, accumulating `sumwt`, when given, by the
507/// set's rule: `W · s^t` times 1, `Re N` or `|N|`.
508///
509/// The visibility polarizations a grid polarization takes through the same
510/// Mueller plane share its taps (both correlations of Stokes I), so their
511/// values are summed and spread once; `sumwt` still adds each one's weight.
512#[allow(clippy::too_many_arguments)]
513fn spread_sample<T: GridScalar, const COMPENSATED: bool>(
514    layout: &AccumulatorLayout,
515    cells: &mut [Complex<T>],
516    correction: &mut [Complex32],
517    mut sumwt: Option<&mut [f64]>,
518    cf: &dyn ConvolutionFunctionSet,
519    rule: KernelNormalisation,
520    placement: &Placement,
521    taps: &TapLayout<'_>,
522    location: CellLocation,
523    mode: Mode,
524    terms: &std::ops::Range<usize>,
525    powers: &[f64],
526    values: &[Complex32],
527    weights: &[f32],
528    mut clip: Option<Clip<'_, T>>,
529) {
530    let w_positive = placement.w > 0.0;
531    let table = cf.mueller().table(w_positive, false);
532    let plane = layout.planes().local(placement.plane);
533    let block_cells = layout.block_cells();
534    // A sectioned set weighs a clipped support by the taps it keeps.
535    let kept = match &clip {
536        Some((_, cells)) if cf.grid_edge() == GridEdge::Sectioned => Some(*cells),
537        _ => None,
538    };
539    let kernel_sum = |mueller: u8| match kept {
540        Some(cells) => kernel::clipped_norm(taps, location, cells, mueller, w_positive),
541        None => kernel::norm(taps, location, mueller, w_positive),
542    };
543    for (gpol, row) in table.iter().enumerate() {
544        // One value per Mueller plane this grid polarization takes.
545        let mut planes = [(0_u8, Complex64::default()); 4];
546        let mut used = 0;
547        debug_assert!(
548            row.len() <= planes.len(),
549            "a routing row carries at most four visibility polarizations"
550        );
551        for (vpol, mueller_plane) in row.iter().enumerate() {
552            let Some(mueller) = *mueller_plane else {
553                continue;
554            };
555            let weight = weights[vpol];
556            if weight == 0.0 {
557                continue;
558            }
559            let base = widen(match mode {
560                Mode::Data => values[vpol],
561                Mode::Psf | Mode::Weight => Complex32::new(weight, 0.0),
562            });
563            match planes[..used].iter_mut().find(|(held, _)| *held == mueller) {
564                Some((_, sum)) => *sum += base,
565                None => {
566                    planes[used] = (mueller, base);
567                    used += 1;
568                }
569            }
570            let Some(sumwt) = sumwt.as_deref_mut() else {
571                continue;
572            };
573            let norm = match rule {
574                KernelNormalisation::UnitSum => 1.0,
575                KernelNormalisation::RealSum => kernel_sum(mueller).re,
576                KernelNormalisation::KernelSum => kernel_sum(mueller).norm(),
577            };
578            for (power, term) in powers.iter().zip(terms.clone()) {
579                sumwt[layout.block_index(plane, gpol, term)] += f64::from(weight) * power * norm;
580            }
581        }
582        for &(mueller, base) in &planes[..used] {
583            for (power, term) in powers.iter().zip(terms.clone()) {
584                let value =
585                    Complex::new(T::from_f64(base.re * power), T::from_f64(base.im * power));
586                let offset = layout.block_offset(plane, gpol, term);
587                let grid = &mut cells[offset..offset + block_cells];
588                let correction = if COMPENSATED {
589                    &mut correction[offset..offset + block_cells]
590                } else {
591                    &mut []
592                };
593                match clip.as_mut() {
594                    None => kernel::spread::<T, COMPENSATED>(
595                        grid,
596                        correction,
597                        layout.tile(),
598                        location,
599                        taps,
600                        mueller,
601                        w_positive,
602                        placement.gradient,
603                        value,
604                    ),
605                    Some((scratch, cells)) => kernel::spread_clipped::<T, COMPENSATED>(
606                        grid,
607                        correction,
608                        (layout.tile(), *cells),
609                        location,
610                        taps,
611                        (mueller, w_positive, placement.gradient),
612                        value,
613                        scratch,
614                    ),
615                }
616            }
617        }
618    }
619}

621/// The scratch and cells a clipped spread uses.
622type Clip<'a, T> = (&'a mut Vec<Complex<T>>, CellBox);
623
624/// The `sumwt` a placement of `reach` adds to, and the scratch and cells
625/// its clipped spread uses.
626fn targets<'a, T>(
627    reach: Reach,
628    sumwt: Option<&'a mut [f64]>,
629    scratch: &'a mut Vec<Complex<T>>,
630) -> (Option<&'a mut [f64]>, Option<Clip<'a, T>>) {
631    match reach {
632        Reach::Whole => (sumwt, None),
633        Reach::Clipped { cells, weighs } => (sumwt.filter(|_| weighs), Some((scratch, cells))),
634        Reach::Off => (None, None),
635    }
636}

638/// `powers[t] = spectral^t` for `t < count`.
639fn spectral_powers(powers: &mut Vec<f64>, spectral: f32, count: usize) {
640    powers.clear();
641    let spectral = f64::from(spectral);
642    let mut power = 1.0;
643    for _ in 0..count {
644        powers.push(power);
645        power *= spectral;
646    }
647}
