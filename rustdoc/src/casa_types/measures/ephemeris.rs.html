2//! The direction of a body that moves across the sky, as CASA's imager
3//! evaluates a field that follows an ephemeris: the body's geocentric
4//! position seen from the observatory (casacore `MeasComet`'s topocentric
5//! correction), and a phase centre offset from that direction.
6//!
7//! The arithmetic follows the evaluation casa-rs's MeasurementSet geometry
8//! has always made, statement by statement, so that every route that
9//! evaluates an ephemeris gets the same bits.
10
11use super::direction::{DirectionRef, MDirection};
12use super::epoch::EpochRef;
13use super::error::MeasureError;
14use super::frame::MeasFrame;
15use super::position::MPosition;

17/// The J2000 direction, from `observatory` at `frame`'s epoch, of a body
18/// at `geocentric_m` (metres, in `reference`'s axes from the Earth's
19/// centre).
20///
21/// The observatory is placed on the apparent equator at the local apparent
22/// sidereal time and its geocentric latitude; the body's position is
23/// shifted by it, converted to the apparent frame, and the observatory's
24/// shift, scaled by the body's distance, is taken out again before the
25/// direction is converted to J2000. `frame` holds the epoch, `observatory`
26/// as its position, and the Measures provider.
27///
28/// # Errors
29///
30/// [`MeasureError::ModelError`] for a position whose distance is not
31/// positive and finite; [`MeasureError::MissingFrameData`] for a frame
32/// without an epoch; a failed conversion.
33pub fn topocentric_j2000(
34    geocentric_m: [f64; 3],
35    reference: DirectionRef,
36    observatory: &MPosition,
37    frame: &MeasFrame,
38) -> Result<MDirection, MeasureError> {
39    let [x, y, z] = geocentric_m;
40    let distance = x.hypot(y).hypot(z);
41    if !distance.is_finite() || distance <= 0.0 {
42        return Err(MeasureError::ModelError {
43            model: "ephemeris",
44            reason: "the position has no positive finite distance".to_string(),
45        });
46    }
47    let [observer_x, observer_y, observer_z] = observatory.as_itrf();
48    let observer_radius = observer_x.hypot(observer_y).hypot(observer_z);
49    let observer_latitude = (observer_z / observer_radius).clamp(-1.0, 1.0).asin();
50    let epoch = frame
51        .epoch()
52        .ok_or(MeasureError::MissingFrameData { what: "epoch" })?;
53    let last_days = epoch.convert_to(EpochRef::LAST, frame)?.value().as_mjd();
54    let last = last_days.fract() * 2.0 * std::f64::consts::PI;
55    let (sin_last, cos_last) = last.sin_cos();
56    let (sin_latitude, cos_latitude) = observer_latitude.sin_cos();
57    let observer_app = [
58        observer_radius * cos_latitude * cos_last,
59        observer_radius * cos_latitude * sin_last,
60        observer_radius * sin_latitude,
61    ];
62    let pre_topocentric = MDirection::from_cosines(
63        [
64            x + observer_app[0],
65            y + observer_app[1],
66            z + observer_app[2],
67        ],
68        reference,
69    );
70    let apparent = pre_topocentric.convert_to(DirectionRef::APP, frame)?;
71    let apparent_cosines = apparent.cosines();
72    MDirection::from_cosines(
73        [
74            apparent_cosines[0] - observer_app[0] / distance,
75            apparent_cosines[1] - observer_app[1] / distance,
76            apparent_cosines[2] - observer_app[2] / distance,
77        ],
78        DirectionRef::APP,
79    )
80    .convert_to(DirectionRef::J2000, frame)
81}

83/// `direction` offset by `offset_rad` (longitude, latitude) in its own
84/// frame, as casacore reads a moving field's `PHASE_DIR`: the latitude
85/// offset is added along the meridian, then the longitude offset turns the
86/// result about the pole by its true angle.
87///
88/// # Errors
89///
90/// [`MeasureError::ModelError`] when the offset direction is not finite.
91pub fn offset_true_angle(
92    direction: &MDirection,
93    offset_rad: [f64; 2],
94) -> Result<MDirection, MeasureError> {
95    let (longitude, latitude) = direction.as_angles();
96    let [longitude_offset, latitude_offset] = offset_rad;
97    let (sin_offset, cos_offset) = longitude_offset.sin_cos();
98    let shifted_latitude = latitude + latitude_offset;
99    let (sin_latitude, cos_latitude) = shifted_latitude.sin_cos();
100    let (sin_longitude, cos_longitude) = longitude.sin_cos();
101    let x = cos_offset * cos_latitude * cos_longitude - sin_offset * sin_longitude;
102    let y = cos_offset * cos_latitude * sin_longitude + sin_offset * cos_longitude;
103    let z = cos_offset * sin_latitude;
104    if [x, y, z].iter().any(|value| !value.is_finite()) {
105        return Err(MeasureError::ModelError {
106            model: "ephemeris",
107            reason: "a moving field's phase-centre offset is not finite".to_string(),
108        });
109    }
110    Ok(MDirection::from_angles(
111        y.atan2(x).rem_euclid(std::f64::consts::TAU),
112        z.atan2(x.hypot(y)),
113        direction.refer(),
114    ))
115}
