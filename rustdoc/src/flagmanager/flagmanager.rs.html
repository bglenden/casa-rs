16fn main() -> ExitCode {
17    TaskFrontEnd::new(
18        "flagmanager",
19        flagmanager_task_schema_bundle(),
20        &["flagmanager"],
21        run,
22    )
23    .expect("the flagmanager contract carries its surface")
24    .main(
25        std::env::args_os().skip(1),
26        CasaLoggingConfig::from_env(),
27        std::env::set_current_dir,
28        &mut std::io::stdout(),
29        std::io::stderr(),
30    )
31}

33fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
34    let request = FlagManagerTaskRequest {
35        vis: task.path("vis")?,
36        mode: task.required_text("mode")?.to_ascii_lowercase(),
37        versionname: task.text("versionname")?,
38        oldname: task.text("oldname")?,
39        comment: task.text("comment")?,
40        merge: task_flag_merge(task, "merge")?,
41    };
42    let result = execute(request)?;
43    // CASA's `list` returns the versions; the other modes return None.
44    match &result {
45        FlagManagerTaskResult::Versions(versions) => ms_task_success(versions, &result),
46        FlagManagerTaskResult::Mutation(_) => ms_task_success(&(), &result),
47    }
48}
49
50fn execute(request: FlagManagerTaskRequest) -> Result<FlagManagerTaskResult, TaskFailure> {
51    let failed = |error: casa_ms::FlaggingError| TaskFailure::execution(error.to_string());
52    let required = |name: &str, value: Option<String>| {
53        value.ok_or_else(|| {
54            TaskFailure::validation(format!(
55                "flagmanager mode {:?} requires {name}",
56                request.mode
57            ))
58            .at(name)
59        })
60    };
61    let mut ms = MeasurementSet::open(&request.vis)
62        .map_err(|error| TaskFailure::execution(error.to_string()))?;
63    let mutation = |mode: &str, versionname, oldname| {
64        Ok(FlagManagerTaskResult::Mutation(FlagManagerMutationResult {
65            mode: mode.to_string(),
66            versionname,
67            oldname,
68        }))
69    };
70    match request.mode.as_str() {
71        "list" => Ok(FlagManagerTaskResult::Versions(
72            list_flag_versions(&ms).map_err(failed)?,
73        )),
74        "save" => {
75            let name = required("versionname", request.versionname.clone())?;
76            let comment = request.comment.as_deref().unwrap_or("");
77            save_flag_version(&ms, &name, comment, request.merge).map_err(failed)?;
78            mutation("save", Some(name), None)
79        }
80        "restore" => {
81            let name = required("versionname", request.versionname.clone())?;
82            restore_flag_version(&mut ms, &name, request.merge).map_err(failed)?;
83            ms.save_main_table_only()
84                .map_err(|error| TaskFailure::execution(error.to_string()))?;
85            mutation("restore", Some(name), None)
86        }
87        "delete" => {
88            let name = required("versionname", request.versionname.clone())?;
89            delete_flag_version(&ms, &name).map_err(failed)?;
90            mutation("delete", Some(name), None)
91        }
92        "rename" => {
93            let old = required("oldname", request.oldname.clone())?;
94            let name = required("versionname", request.versionname.clone())?;
95            let comment = request.comment.as_deref().unwrap_or("");
96            rename_flag_version(&ms, &old, &name, comment).map_err(failed)?;
97            mutation("rename", Some(name), Some(old))
98        }
99        other => Err(TaskFailure::validation(format!(
100            "unsupported flagmanager mode {other:?}; expected list, save, restore, delete or rename"
101        ))
102        .at("mode")),
103    }
104}

110    #[test]
111    fn provider_bundle_embeds_flagmanager_parameter_contract() {
112        let bundle = flagmanager_task_schema_bundle();
113        assert_eq!(bundle.protocol.protocol_name, "casa_ms_flagmanager_task");
114        assert_eq!(bundle.parameter_surfaces.len(), 1);
115        assert_eq!(bundle.parameter_surfaces[0].surface.id(), "flagmanager");
116        bundle.validate().expect("valid flagmanager provider");
117    }
