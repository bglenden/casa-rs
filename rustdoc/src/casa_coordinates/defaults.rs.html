2//! Default coordinate systems, a port of casacore's `CoordinateUtil`
3//! (`coordinates/Coordinates/CoordinateUtil.cc`).
4//!
5//! CASA builds an image's coordinate system from these when a script gives
6//! none (`ia.fromshape`, `ia.fromarray`), and `coordsys.newcoordsys` adds
7//! these coordinates. They are casacore's: a J2000 SIN direction at the
8//! origin with 1 arcmin pixels, Stokes I, Q, U and V as the axis length
9//! allows, an LSRK frequency of 1415 MHz in 1 kHz channels with the HI line
10//! as rest frequency, and linear axes in km.
11

24/// The HI line's rest frequency in Hz, casacore's `QC::HI()`
25/// (1420.405751786 MHz) as CASA 6.7.6.14 reports it in Hz after casacore's
26/// unit conversion, `coordsys.newcoordsys(spectral=True)` then
27/// `restfrequency()`: 1420405751.7860003, one ulp above the literal.
28pub const HI_REST_FREQUENCY_HZ: f64 = 1_420_405_751.786_000_3;
29
30/// The reference frequency of the default spectral axis, Hz.
31const DEFAULT_FREQUENCY_HZ: f64 = 1415.0e6;
32
33/// The channel width of the default spectral axis, Hz.
34const DEFAULT_CHANNEL_HZ: f64 = 1.0e3;
35
36/// One arcminute in radians.
37const ARCMIN: f64 = std::f64::consts::PI / 10_800.0;
38
39/// The default direction coordinate, casacore's `addDirAxes`: J2000, SIN,
40/// the reference at RA 0 and Dec 0 on pixel (0, 0), increments of -1 and
41/// +1 arcmin, presented in arcmin. The coordinate trait still returns its
42/// values and increments in radians.
43pub fn direction_coordinate() -> DirectionCoordinate {
44    DirectionCoordinate::new(
45        DirectionRef::J2000,
46        Projection::new(ProjectionType::SIN),
47        [0.0, 0.0],
48        [-ARCMIN, ARCMIN],
49        [0.0, 0.0],
50    )
51    // wcslib's LATPOLE for a zenithal projection is the reference latitude.
52    .with_latpole(0.0)
53    .with_world_units(["'", "'"])
54    .expect("arcmin is an angle unit")
55}

57/// The default Stokes coordinate for an axis of `length` pixels,
58/// casacore's `addStokesAxis`: I, then Q, U and V; `None` for a length
59/// outside 1 to 4.
60pub fn stokes_coordinate(length: usize) -> Option<StokesCoordinate> {
61    const ORDER: [StokesType; 4] = [StokesType::I, StokesType::Q, StokesType::U, StokesType::V];
62    (1..=4)
63        .contains(&length)
64        .then(|| StokesCoordinate::new(ORDER[..length].to_vec()))
65}

67/// The default spectral coordinate, casacore's `addFreqAxis`: LSRK
68/// frequency, 1415 MHz at pixel 0, 1 kHz channels, the HI line as rest
69/// frequency.
70pub fn spectral_coordinate() -> SpectralCoordinate {
71    SpectralCoordinate::new(
72        FrequencyRef::LSRK,
73        DEFAULT_FREQUENCY_HZ,
74        DEFAULT_CHANNEL_HZ,
75        0.0,
76        HI_REST_FREQUENCY_HZ,
77    )
78}

80/// The default linear coordinate over axes `names`, casacore's
81/// `addLinearAxes`: reference value 0 and increment 1 in km on each axis,
82/// the reference pixel `(length + 1) / 2` (integer division) of each axis
83/// when `shape` gives one length per name, else 0.
84///
85/// # Panics
86///
87/// Panics if `names` is empty, because a linear coordinate needs an axis.
88pub fn linear_coordinate(names: &[String], shape: &[usize]) -> LinearCoordinate {
89    let n = names.len();
90    let crpix = if shape.len() == n {
91        shape
92            .iter()
93            .map(|&length| length.div_ceil(2) as f64)
94            .collect()
95    } else {
96        vec![0.0; n]
97    };
98    LinearCoordinate::new(n, names.to_vec(), vec!["km".to_string(); n]).with_reference_pixel(crpix)
99}

101/// The observation information casacore's `makeCoordinateSystem` attaches:
102/// observer Karl Jansky, telescope ALMA, observed 2000-01-01 00:00:00.0001
103/// UTC. casacore also sets the telescope's position from its observatory
104/// table; that needs the measures data, so callers that have it add it.
105pub fn obs_info() -> ObsInfo {
106    // `Time(2000, 1, 1, 0, 0, 0.0001)`: MJD 51544 and 0.1 ms, which keeps
107    // the time from rounding to the previous day.
108    let date = MEpoch::from_mjd(51_544.0 + 1.0e-4 / 86_400.0, EpochRef::UTC);
109    ObsInfo::new("ALMA")
110        .with_observer("Karl Jansky")
111        .with_date(date)
112}

114/// The names `linear1`, `linear2`, ... of `n` linear axes.
115fn linear_names(n: usize) -> Vec<String> {
116    (1..=n).map(|axis| format!("linear{axis}")).collect()
117}

119/// The coordinate system casacore's `CoordinateUtil::makeCoordinateSystem`
120/// makes for an image of `shape`, with [`obs_info`].
121///
122/// This is the raw coordinate utility default, not the complete behavior of
123/// `ia.fromshape` or `ia.fromarray`. An image-layer caller must apply CASA's
124/// image-specific reference-pixel recentering after it builds this system.
125///
126/// With `linear`, every axis is linear. Otherwise one axis is spectral; two
127/// or more start with a direction; a third axis is Stokes when its length is
128/// 1 to 4, else spectral; a fourth is spectral after a Stokes axis, else
129/// Stokes when it can be, else linear; the rest are linear.
130pub fn coordinate_system_for_shape(shape: &[usize], linear: bool) -> CoordinateSystem {
131    let n = shape.len();
132    let mut csys = CoordinateSystem::new().with_obs_info(obs_info());
133    if linear {
134        if n > 0 {
135            csys.add_coordinate(linear_coordinate(&linear_names(n), shape));
136        }
137        return csys;
138    }
139    if n == 1 {
140        csys.add_coordinate(spectral_coordinate());
141        return csys;
142    }
143    if n >= 2 {
144        csys.add_coordinate(direction_coordinate());
145    }
146    let mut done_stokes = false;
147    let mut done_frequency = false;
148    if n >= 3 {
149        if let Some(stokes) = stokes_coordinate(shape[2]) {
150            csys.add_coordinate(stokes);
151            done_stokes = true;
152        } else {
153            csys.add_coordinate(spectral_coordinate());
154            done_frequency = true;
155        }
156    }
157    let mut n_done = 0;
158    if n >= 4 {
159        n_done = 4;
160        if done_stokes {
161            csys.add_coordinate(spectral_coordinate());
162        } else if let Some(stokes) = stokes_coordinate(shape[3]) {
163            csys.add_coordinate(stokes);
164        } else if !done_frequency {
165            csys.add_coordinate(spectral_coordinate());
166        } else {
167            n_done = 3;
168        }
169    }
170    if n_done == 3 || n >= 5 {
171        let rest = &shape[n_done..];
172        if !rest.is_empty() {
173            csys.add_coordinate(linear_coordinate(&linear_names(rest.len()), rest));
174        }
175    }
176    csys
177}

179#[cfg(test)]
180mod tests {
181    use super::*;
182    use crate::coordinate::{Coordinate, CoordinateType};
183
184    fn types(csys: &CoordinateSystem) -> Vec<CoordinateType> {
185        (0..csys.n_coordinates())
186            .map(|index| csys.coordinate(index).coordinate_type())
187            .collect()
188    }
189
190    #[test]
191    fn shapes_get_casacores_coordinates() {
192        use CoordinateType::{Direction, Linear, Spectral, Stokes};
193        let cases: [(&[usize], &[CoordinateType]); 10] = [
194            (&[10], &[Spectral]),
195            (&[4, 5], &[Direction]),
196            (&[4, 5, 3], &[Direction, Stokes]),
197            (&[4, 5, 6], &[Direction, Spectral]),
198            (&[4, 5, 3, 2], &[Direction, Stokes, Spectral]),
199            (&[4, 5, 6, 3], &[Direction, Spectral, Stokes]),
200            (&[4, 5, 6, 7], &[Direction, Spectral, Linear]),
201            (&[4, 5, 6, 3, 7], &[Direction, Spectral, Stokes, Linear]),
202            (&[4, 5, 6, 5, 3], &[Direction, Spectral, Linear]),
203            (&[4, 5, 4, 6, 3], &[Direction, Stokes, Spectral, Linear]),
204        ];
205        for (shape, expected) in cases {
206            let csys = coordinate_system_for_shape(shape, false);
207            assert_eq!(types(&csys), expected, "{shape:?}");
208            assert_eq!(csys.n_pixel_axes(), shape.len(), "{shape:?}");
209        }
210        let linear = coordinate_system_for_shape(&[4, 5], true);
211        assert_eq!(types(&linear), [Linear]);
212        assert_eq!(linear.coordinate(0).reference_pixel(), [2.0, 3.0]);
213        assert_eq!(linear.coordinate(0).axis_names(), ["linear1", "linear2"]);
214
215        let no_axes = coordinate_system_for_shape(&[], false);
216        assert_eq!(no_axes.n_coordinates(), 0);
217        assert_eq!(no_axes.n_pixel_axes(), 0);
218
219        let mismatched_shape = linear_coordinate(&["linear1".into(), "linear2".into()], &[4]);
220        assert_eq!(mismatched_shape.reference_pixel(), [0.0, 0.0]);
221    }

223    #[test]
224    fn defaults_carry_casacores_values() {
225        let direction = direction_coordinate();
226        assert_eq!(direction.world_units(), &["'".to_string(), "'".to_string()]);
227        assert!((direction.increment()[0] + ARCMIN).abs() < 1e-18);
228        assert_eq!(direction.reference_pixel(), [0.0, 0.0]);
229        let spectral = spectral_coordinate();
230        assert_eq!(spectral.rest_frequency(), 1_420_405_751.786_000_3);
231        assert_eq!(spectral.reference_value(), [1.415e9]);
232        let stokes = stokes_coordinate(3).expect("three Stokes");
233        assert_eq!(
234            stokes.stokes(),
235            [StokesType::I, StokesType::Q, StokesType::U]
236        );
237        assert!(stokes_coordinate(0).is_none());
238        assert_eq!(
239            stokes_coordinate(1).expect("one Stokes").stokes(),
240            [StokesType::I]
241        );
242        assert_eq!(
243            stokes_coordinate(4).expect("four Stokes").stokes(),
244            [StokesType::I, StokesType::Q, StokesType::U, StokesType::V]
245        );
246        assert!(stokes_coordinate(5).is_none());
247        let info = obs_info();
248        assert_eq!(
249            (info.telescope.as_str(), info.observer.as_str()),
250            ("ALMA", "Karl Jansky")
251        );
252        let date = info.date.as_ref().expect("the default epoch");
253        assert_eq!(date.refer(), EpochRef::UTC);
254        assert!((date.value().as_mjd() - (51_544.0 + 1.0e-4 / 86_400.0)).abs() < 1e-15);
255        assert!(info.telescope_position.is_none());
256
257        let system = coordinate_system_for_shape(&[4, 5, 3, 2], false);
258        let pixel = vec![0.0; 4];
259        let world = system
260            .to_world(&pixel)
261            .expect("the default maps its origin");
262        let round_trip = system.to_pixel(&world).expect("the mapping is invertible");
263        assert!(
264            round_trip
265                .iter()
266                .zip(pixel)
267                .all(|(got, expected)| (got - expected).abs() < 1e-12),
268            "{round_trip:?} is not within 1e-12 pixel of the origin"
269        );
270    }
