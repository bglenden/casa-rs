23pub(super) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
24    let outfile = task.required_text("outfile")?;
25    if outfile.trim().is_empty() {
26        return Err(invalid("outfile", "outfile must be specified"));
27    }
28    let requested_path = Path::new(&outfile);
29    let destination = task.workspace().join(requested_path);
30    let overwrite = task.flag("overwrite")?;
31    if !overwrite && destination.symlink_metadata().is_ok() {
32        return Err(exists(requested_path));
33    }
34    let source = task.required_text("source")?;
35    let direction = task.required_text("direction")?;
36    if source.trim().is_empty() && direction.trim().is_empty() {
37        return Err(invalid(
38            "source",
39            "Exactly one of source or direction must be specified",
40        ));
41    }
42    if !source.is_empty() && !direction.is_empty() {
43        return Err(invalid(
44            "direction",
45            "Both source and direction may not be simultaneously specified",
46        ));
47    }
48    if !direction.is_empty() {
49        validate_direction(&direction)?;
50    }
51    let source = source.to_uppercase();
52    let band = task.required_text("band")?;
53    if band.is_empty() {
54        return Err(invalid("band", "band must be specified"));
55    }
56    if !matches!(
57        band.to_ascii_uppercase().as_str(),
58        "P" | "L" | "S" | "C" | "X" | "U" | "K" | "A" | "Q"
59    ) {
60        return Err(invalid("band", format!("band {band} not supported")));
61    }
62    let (obsdate, obs_mjd) = observation_date(task.value("obsdate"), "obsdate")?;
63    let (refdate, ref_mjd) = observation_date(task.value("refdate"), "refdate")?;
64    let hosts = task.texts("hosts")?;
65    if hosts.is_empty() {
66        return Err(invalid("hosts", "hosts must be specified"));
67    }
68    let selector = if source.is_empty() {
69        format!("position={}", quote(&direction))
70    } else {
71        format!("source={}", quote(&source))
72    };
73    let mut query = format!(
74        "type=setjy&{selector}&band={}&obsdate={}",
75        quote(&band),
76        obs_mjd as i64
77    );
78    if ref_mjd > 0. {
79        write!(query, "&refdate={}", ref_mjd as i64).expect("writing to a string");
80    }
81    let (url, response) = retrieve(&hosts, &query)?;
82    let components = ComponentList::from_import_record(&response).map_err(execution)?;
83    let now = OffsetDateTime::now_local()
84        .unwrap_or_else(|_| OffsetDateTime::now_utc())
85        .date();
86    let mut provenance = RecordValue::new(Vec::new());
87    if !source.is_empty() {
88        provenance.upsert("source", text(source));
89    }
90    if !direction.is_empty() {
91        provenance.upsert("direction", text(direction));
92    }
93    if let Some(obsdate) = obsdate {
94        provenance.upsert("obsdate", obsdate);
95    }
96    if let Some(refdate) = refdate {
97        provenance.upsert("refdate", refdate);
98    }
99    provenance.upsert("band", text(band));
100    provenance.upsert(
101        "hosts",
102        Value::Array(ArrayValue::String(Array1::from(hosts).into_dyn())),
103    );
104    provenance.upsert(
105        "run_mjd",
106        Value::Scalar(ScalarValue::Int32(now.to_julian_day() - 2_400_001)),
107    );
108    provenance.upsert("run_date", text(now.to_string()));
109    provenance.upsert("url", text(url));
110    provenance.upsert("response", Value::Record(response));
111    publish(
112        &destination,
113        requested_path,
114        overwrite,
115        &components,
116        provenance,
117    )?;
118    tracing::info!(path = %destination.display(), "component list has been written");
119    TaskSuccess::new(CasaValue::None, &())
120}

130fn exists(path: &Path) -> TaskFailure {
131    execution(format!(
132        "The overwrite parameter is False and a file or directory named {} already exists. \
133         Either remove or rename it, or change overwrite to True, or both.",
134        path.display()
135    ))
136    .at("outfile")

143fn validate_direction(direction: &str) -> Result<(), TaskFailure> {
144    let parts = direction.split(' ').collect::<Vec<_>>();
145    let rad = Unit::new("rad").expect("rad is a unit");
146    if let [frame, longitude, latitude] = parts.as_slice()
147        && [longitude, latitude].into_iter().all(|angle| {
148            Quantity::read(angle)
149                .ok()
150                .and_then(|q| q.get_value_in(&rad).ok())
151                .is_some()
152        })
153    {
154        if casa_direction_frame_name(frame).is_none() {
155            tracing::warn!(%frame, "unknown direction reference frame; CASA accepts the angles with its default frame");
156        }
157        Ok(())
158    } else {
159        Err(invalid(
160            "direction",
161            format!("Illegal direction specification {direction}"),
162        ))
163    }
164}

166fn observation_date(
167    value: Option<&ParameterValue>,
168    name: &'static str,
169) -> Result<(Option<Value>, f64), TaskFailure> {
170    let date = match value {
171        Some(ParameterValue::Float(value)) if value.is_finite() => {
172            return Ok((
173                (*value != 0.).then_some(Value::Scalar(ScalarValue::Float64(*value))),
174                *value,
175            ));
176        }
177        Some(ParameterValue::Integer(value)) => {
178            if let Ok(small) = i32::try_from(*value) {
179                return Ok((
180                    (*value != 0).then_some(Value::Scalar(ScalarValue::Int32(small))),
181                    *value as f64,
182                ));
183            }
184            return Ok((
185                Some(Value::Scalar(ScalarValue::Int64(*value))),
186                *value as f64,
187            ));
188        }
189        Some(ParameterValue::Bool(value)) => {
190            return Ok((
191                value.then_some(Value::Scalar(ScalarValue::Bool(true))),
192                if *value { 1. } else { 0. },
193            ));
194        }
195        Some(value @ ParameterValue::NumpyScalar { .. }) => {
196            let converted = casa_numpy_scalar_root(value).map_err(|error| invalid(name, error))?;
197            return observation_date(Some(&converted), name);
198        }
199        Some(ParameterValue::String(value)) => value,
200        _ => {
201            return Err(invalid(
202                name,
203                format!("{name} must either be a number or a string of the form YYYY-MM-DD"),
204            ));
205        }
206    };
207    let syntax = date.len() == 10
208        && date.bytes().enumerate().all(|(i, b)| {
209            if matches!(i, 4 | 7) {
210                b == b'-'
211            } else {
212                b.is_ascii_digit()
213            }
214        });
215    let parsed = syntax
216        .then(|| Quantity::read(date).ok())
217        .flatten()
218        .and_then(|q| q.get_value_in(&Unit::new("d").expect("d is a unit")).ok());
219    let mjd = parsed.ok_or_else(|| {
220        invalid(
221            name,
222            format!("If specified as a string, {name} must be of the form YYYY-MM-DD"),
223        )
224    })?;
225    Ok((Some(text(date.clone())), mjd))
226}

228fn quote(value: &str) -> String {
229    let mut encoded = String::new();
230    for byte in value.bytes() {
231        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~' | b'/') {
232            encoded.push(byte as char);
233        } else {
234            write!(encoded, "%{byte:02X}").expect("writing to a string");
235        }
236    }
237    encoded
238}

240fn retrieve(hosts: &[String], query: &str) -> Result<(String, RecordValue), TaskFailure> {
241    let agent = crate::calibration_service::agent(Duration::from_secs(400)).map_err(execution)?;
242    for host in hosts {
243        if !host
244            .parse::<ureq::http::Uri>()
245            .is_ok_and(|uri| uri.scheme().is_some() && uri.authority().is_some())
246        {
247            return Err(invalid(
248                "hosts",
249                format!("{host} is not a valid host expressed as a URL"),
250            ));
251        }
252        // The default NRAO endpoint serves the same model over HTTPS.
253        let host = if host == HTTP_SERVICE {
254            HTTPS_SERVICE
255        } else {
256            host
257        };
258        let url = format!("{host}?{query}");
259        tracing::info!(%url, "querying the VLA calibrator service");
260        let queried = agent.get(&url).call().and_then(|mut response| {
261            if response.status() == 200 {
262                response.body_mut().read_to_string().map(Some)
263            } else {
264                Ok(None)
265            }
266        });
267        let result = match queried {
268            Ok(Some(body)) => match serde_json::from_str::<Json>(&body) {
269                Ok(result) => result,
270                Err(error) => {
271                    tracing::warn!(%url, %error, "invalid JSON in the VLA calibrator response");
272                    continue;
273                }
274            },
275            Ok(None) => continue,
276            Err(error) => {
277                tracing::warn!(%url, %error, "VLA calibrator request failed");
278                continue;
279            }
280        };
281        match result {
282            Json::Object(record) if !record.is_empty() => return Ok((url, json_record(record)?)),
283            Json::Null | Json::Bool(false) | Json::Object(_) => {}
284            Json::String(value) if value.is_empty() => {}
285            Json::Array(values) if values.is_empty() => {}
286            Json::Number(value) if value.as_f64() == Some(0.) => {}
287            _ => {
288                return Err(execution(
289                    "VLA calibrator response is not a component record",
290                ));
291            }
292        }
293    }
294    Err(execution("All URLs failed to return a component list"))
295}

297fn json_record(record: Map<String, Json>) -> Result<RecordValue, TaskFailure> {
298    record
299        .into_iter()
300        .map(|(key, entry)| Ok(RecordField::new(key, json_value(entry)?)))
301        .collect::<Result<Vec<_>, _>>()
302        .map(RecordValue::new)
303}
304
305fn json_value(value: Json) -> Result<Value, TaskFailure> {
306    Ok(match value {
307        Json::Bool(value) => Value::Scalar(ScalarValue::Bool(value)),
308        Json::Number(value) => match value.as_i64() {
309            Some(integer) => Value::Scalar(
310                i32::try_from(integer).map_or(ScalarValue::Int64(integer), ScalarValue::Int32),
311            ),
312            None => Value::Scalar(ScalarValue::Float64(
313                value
314                    .as_f64()
315                    .ok_or_else(|| execution("non-finite number in component record"))?,
316            )),
317        },
318        Json::String(value) => text(value),
319        Json::Object(record) => Value::Record(json_record(record)?),
320        Json::Array(values) if values.iter().all(Json::is_number) => {
321            Value::Array(ArrayValue::from_f64_vec(
322                values
323                    .iter()
324                    .map(|value| value.as_f64().expect("numeric value"))
325                    .collect(),
326            ))
327        }
328        Json::Array(values) if values.iter().all(Json::is_string) => {
329            Value::Array(ArrayValue::String(
330                Array1::from_iter(
331                    values
332                        .iter()
333                        .map(|value| value.as_str().expect("text value").to_string()),
334                )
335                .into_dyn(),
336            ))
337        }
338        _ => return Err(execution("unsupported value in VLA calibrator record")),
339    })
340}

342fn publish(
343    path: &Path,
344    requested_path: &Path,
345    overwrite: bool,
346    list: &ComponentList,
347    provenance: RecordValue,
348) -> Result<(), TaskFailure> {
349    let parent = path.parent().unwrap_or_else(|| Path::new("."));
350    let staging = tempfile::tempdir_in(parent).map_err(execution)?;
351    let component_path = staging.path().join("model.cl");
352    list.write_table(&component_path).map_err(execution)?;
353    let options = TableOptions::new(&component_path);
354    let mut table = Table::open_metadata_only(options.clone()).map_err(execution)?;
355    table
356        .keywords_mut()
357        .upsert("web_service", Value::Record(provenance));
358    table.save_metadata_only(options).map_err(execution)?;
359    drop(table);
360    match rename_without_replacing(&component_path, path) {
361        Ok(()) => Ok(()),
362        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {
363            if overwrite {
364                exchange_directories(&component_path, path)
365                    .map_err(|error| execution(error).at("outfile"))
366            } else {
367                Err(exists(requested_path))
368            }
369        }
370        Err(error) => Err(execution(error).at("outfile")),
371    }
