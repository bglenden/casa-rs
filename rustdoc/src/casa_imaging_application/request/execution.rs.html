8/// Keep the legacy imager's unpadded diagnostic out of unverified execution
9/// modes, particularly virtual-model publication whose demand uses 1.2.
10pub(super) fn validate_ft_padding(
11    request: &super::ImagingRequest,
12) -> Result<(), super::InvalidRequest> {
13    use super::{BackendRequest, DataColumn, Gridder, SaveModel, SpecMode, Weighting};
14    let Some(padding) = request.ft_padding else {
15        return Ok(());
16    };
17    if padding.get() == 1.0
18        && matches!(request.gridder, Gridder::Standard { facets } if facets.get() == 1)
19        && request.specmode == SpecMode::Mfs
20        && request.vis.len() == 1
21        && request.stokes == [super::PolarizationCoordinate::StokesI]
22        && request.weighting == Weighting::Natural
23        && request.datacolumn == Some(DataColumn::Data)
24        && request.niter == 0
25        && request.calcpsf
26        && !request.calcres
27        && !request.restoration
28        && request.savemodel == SaveModel::None
29        && !request.tclean_restart
30        && !request.has_start_model()
31        && request.outlierfile.is_none()
32        && request.backend == BackendRequest::Cpu
33        && request.gridprecision == Some(GridPrecision::F64)
34    {
35        Ok(())
36    } else {
37        Err(super::InvalidRequest {
38            parameters: "ft_padding",
39            reason: "only padding 1.0 for a CPU/f64 natural-weighted standard Stokes-I MFS PSF-only run, without model saving or continuation, is verified",
40        })
41    }
42}

44pub(super) fn major_cycle_limit<'de, D: Deserializer<'de>>(
45    deserializer: D,
46) -> Result<Option<usize>, D::Error> {
47    let value = Value::deserialize(deserializer)?;
48    match value.as_i64() {
49        Some(-1) => Ok(None),
50        Some(limit) => usize::try_from(limit)
51            .map(Some)
52            .map_err(|_| unexpected(&value, "-1 or a major-cycle count")),
53        None => Err(unexpected(&value, "-1 or a major-cycle count")),
54    }
55}

57pub(super) fn hogbom_iteration_mode<'de, D: Deserializer<'de>>(
58    deserializer: D,
59) -> Result<HogbomIterationAccounting, D::Error> {
60    match choice(deserializer)?.as_str() {
61        "strict" => Ok(HogbomIterationAccounting::Strict),
62        "casa-inclusive" => Ok(HogbomIterationAccounting::CasaInclusive),
63        other => Err(D::Error::custom(format!("hogbom_iteration_mode {other:?}"))),
64    }
65}
66
67pub(super) fn normalization<'de, D: Deserializer<'de>>(
68    deserializer: D,
69) -> Result<ProductNormalization, D::Error> {
70    match choice(deserializer)?.as_str() {
71        "flatnoise" => Ok(ProductNormalization::FlatNoise),
72        "flatsky" => Ok(ProductNormalization::FlatSky),
73        "pbsquare" => Ok(ProductNormalization::PbSquare),
74        other => Err(D::Error::custom(format!("normtype {other:?}"))),
75    }
76}
77
78pub(super) fn grid_precision<'de, D: Deserializer<'de>>(
79    deserializer: D,
80) -> Result<Option<GridPrecision>, D::Error> {
81    match choice(deserializer)?.as_str() {
82        "auto" => Ok(None),
83        "f32" => Ok(Some(GridPrecision::F32)),
84        "f64" => Ok(Some(GridPrecision::F64)),
85        other => Err(D::Error::custom(format!("gridprecision {other:?}"))),
86    }
87}

89#[cfg(test)]
90mod tests {
91    use super::super::{BackendRequest, GridPrecision, tests::request};
92    use casa_imaging_operator::PaddingFactor;
93    use serde_json::json;
94
95    #[test]
96    fn legacy_padding_is_only_available_for_the_verified_quick_psf_workflow() {
97        let mut virtual_model = request(json!({"savemodel": "virtual", "niter": 1}));
98        virtual_model.ft_padding = Some(PaddingFactor::new(2.0).expect("padding"));
99        assert_eq!(
100            virtual_model
101                .validate()
102                .expect_err("unadmitted virtual grid")
103                .parameters,
104            "ft_padding"
105        );
106        let mut valid = request(json!({
107            "niter": 0, "calcpsf": true, "calcres": false, "restoration": false,
108            "weighting": "natural", "datacolumn": "data",
109        }));
110        valid.backend = BackendRequest::Cpu;
111        valid.gridprecision = Some(GridPrecision::F64);
112        valid.ft_padding = Some(PaddingFactor::new(1.0).expect("padding"));
113        valid.validate().expect("verified PSF");
114        for overrides in [
115            json!({"savemodel": "virtual"}),
116            json!({"savemodel": "modelcolumn"}),
117            json!({"niter": 1}),
118            json!({"calcres": true}),
119            json!({"calcpsf": false}),
120            json!({"restoration": true}),
121            json!({"gridder": "wproject", "wprojplanes": 16}),
122            json!({"specmode": "cube"}),
123            json!({"startmodel": "model.image"}),
124            json!({"stokes": "IQUV"}),
125        ] {
126            let mut values = json!({
127                "niter": 0, "calcpsf": true, "calcres": false, "restoration": false,
128                "weighting": "natural", "datacolumn": "data",
129            });
130            values
131                .as_object_mut()
132                .expect("object")
133                .extend(overrides.as_object().expect("object").clone());
134            let mut invalid = request(values);
135            invalid.backend = valid.backend;
136            invalid.gridprecision = valid.gridprecision;
137            invalid.ft_padding = valid.ft_padding;
138            assert_eq!(
139                invalid
140                    .validate()
141                    .expect_err(&format!("unsupported padding scope: {overrides}"))
142                    .parameters,
143                "ft_padding",
144                "{overrides}"
145            );
146        }
147        for factor in [1.2, 2.0] {
148            let mut invalid = valid.clone();
149            invalid.ft_padding = Some(PaddingFactor::new(factor).expect("padding"));
150            assert_eq!(
151                invalid
152                    .validate()
153                    .expect_err("unverified factor")
154                    .parameters,
155                "ft_padding"
156            );
157        }
158        let mut gpu = valid.clone();
159        gpu.backend = BackendRequest::Gpu;
160        assert_eq!(
161            gpu.validate().expect_err("unverified backend").parameters,
162            "ft_padding"
163        );
164        valid.gridprecision = Some(GridPrecision::F32);
165        assert_eq!(
166            valid
167                .validate()
168                .expect_err("unverified precision")
169                .parameters,
170            "ft_padding"
171        );
172    }
