5#[derive(Debug, Clone)]
6pub struct ChannelBin {
7    pub start: usize,
8    pub end: usize,
9    pub(super) stride: usize,
10    pub(super) ordinal: usize,
11    pub(super) averaged: bool,
12    pub(super) selection: Option<Arc<[usize]>>,
13    pub(super) selection_width: usize,
14}
15
16impl ChannelBin {
17    pub(super) fn channels(&self) -> impl Iterator<Item = usize> + '_ {
18        let (range, stride) = if self.selection.is_some() {
19            let start = self.ordinal * self.selection_width;
20            (start..start + self.selection_width, 1)
21        } else {
22            (self.start..self.end, self.stride)
23        };
24        range.step_by(stride).map(|index| {
25            self.selection
26                .as_ref()
27                .map_or(index, |channels| channels[index])
28        })
29    }
30
31    pub fn display_channel(&self) -> usize {
32        if self.averaged {
33            self.ordinal
34        } else {
35            self.start
36        }
37    }
38}

40pub(super) fn channel_bins(
41    chan_count: usize,
42    avgchannel: Option<usize>,
43) -> Result<Vec<ChannelBin>, String> {
44    if chan_count == 0 {
45        return Ok(Vec::new());
46    }
47    let width = avgchannel.unwrap_or(1);
48    if width == 0 {
49        return Err("avgchannel must be a positive integer".to_string());
50    }
51    let width = width.min(chan_count);
52    Ok((0..chan_count / width)
53        .map(|ordinal| ChannelBin {
54            start: ordinal * width,
55            end: (ordinal + 1) * width,
56            stride: 1,
57            ordinal,
58            averaged: width > 1,
59            selection: None,
60            selection_width: 0,
61        })
62        .collect())
63}

65pub(super) fn plot_channel_bins_for_window<'a>(
66    chan_count: usize,
67    spec: &MsPlotSpec,
68    spw_id: i32,
69    calibration: Option<&PlotMsInput<'_>>,
70    cache: &'a mut HashMap<(i32, usize), Vec<ChannelBin>>,
71) -> Result<Cow<'a, [ChannelBin]>, String> {
72    let Some(ranges) = calibration.and_then(|input| input.channels.get(&spw_id)) else {
73        return plot_channel_bins(chan_count, spec).map(Cow::Owned);
74    };
75    let key = (spw_id, chan_count);
76    let bins = match cache.entry(key) {
77        std::collections::hash_map::Entry::Occupied(entry) => entry.into_mut(),
78        std::collections::hash_map::Entry::Vacant(entry) => {
79            entry.insert(plot_selected_channel_bins(chan_count, spec, ranges)?)
80        }
81    };
82    Ok(Cow::Borrowed(bins))
83}

85pub fn plot_selected_channel_bins(
86    chan_count: usize,
87    spec: &MsPlotSpec,
88    channels: &[ChannelRange],
89) -> Result<Vec<ChannelBin>, String> {
90    let mut ranges = channels
91        .iter()
92        .filter_map(|range| {
93            let start = usize::try_from(range.start).ok()?;
94            let end = (usize::try_from(range.stop).ok()? + 1).min(chan_count);
95            (start < end).then_some((start, end, range.step.max(1) as usize))
96        })
97        .collect::<Vec<_>>();
98    ranges.sort_unstable();
99    ranges.dedup();
100    let requested_width = spec.averaging.avgchannel.unwrap_or(1);
101    if requested_width == 0 {
102        return Err("avgchannel must be a positive integer".into());
103    }
104    if requested_width == 1 {
105        let channels = ranges
106            .into_iter()
107            .flat_map(|(start, end, stride)| (start..end).step_by(stride))
108            .collect::<std::collections::BTreeSet<_>>();
109        return Ok(channels
110            .into_iter()
111            .enumerate()
112            .map(|(ordinal, start)| ChannelBin {
113                start,
114                end: start + 1,
115                stride: 1,
116                ordinal,
117                averaged: false,
118                selection: None,
119                selection_width: 0,
120            })
121            .collect());
122    }
123
124    let selection: Option<Arc<[usize]>> = (ranges.len() > 1).then(|| {
125        ranges
126            .iter()
127            .flat_map(|&(start, end, stride)| (start..end).step_by(stride))
128            .collect::<std::collections::BTreeSet<_>>()
129            .into_iter()
130            .collect()
131    });
132    let selected_count = selection.as_ref().map_or_else(
133        || {
134            ranges
135                .first()
136                .map_or(0, |&(start, end, stride)| (end - start).div_ceil(stride))
137        },
138        |channels| channels.len(),
139    );
140    if selected_count == 0 {
141        return Ok(Vec::new());
142    }
143    let width = requested_width.min(selected_count);
144    let mut bins = Vec::new();
145    for (start, end, stride) in ranges {
146        let span = width.saturating_mul(stride);
147        for offset in 0..(end - start) / span {
148            if (bins.len() + 1).saturating_mul(width) > selected_count {
149                break;
150            }
151            let start = start + offset * span;
152            // CASA constructs frequency bins from the physical selection ranges
153            // and averages the compressed visibility buffer in index order.
154            bins.push(ChannelBin {
155                start,
156                end: start + span,
157                stride,
158                ordinal: bins.len(),
159                averaged: true,
160                selection: selection.clone(),
161                selection_width: width,
162            });
163        }
164    }
165    Ok(bins)
166}

168pub(super) fn plot_channel_bins(
169    chan_count: usize,
170    spec: &MsPlotSpec,
171) -> Result<Vec<ChannelBin>, String> {
172    if spec.x_axis.uses_channel_bins() || spec.y_axes.iter().copied().any(MsAxis::uses_channel_bins)
173    {
174        channel_bins(chan_count, spec.averaging.avgchannel)
175    } else if chan_count == 0 {
176        Ok(Vec::new())
177    } else {
178        Ok(vec![ChannelBin {
179            start: 0,
180            end: chan_count,
181            stride: 1,
182            ordinal: 0,
183            averaged: chan_count > 1,
184            selection: None,
185            selection_width: 0,
186        }])
187    }
188}

197pub(super) struct TimePlotSpectrum {
198    pub(super) frame: FrequencyRef,
199    pub(super) source_channels: usize,
200    pub(super) frequencies_hz: Vec<f64>,
201}
202
203impl TimePlotSpectrum {
204    pub(super) fn read(
205        column: &ChanFreqColumn<'_>,
206        spw_id: i32,
207        bins: &[ChannelBin],
208    ) -> Result<Self, String> {
209        let window = usize::try_from(spw_id)
210            .map_err(|_| format!("invalid DATA_DESCRIPTION::SPECTRAL_WINDOW_ID {spw_id}"))?;
211        let frame = column
212            .frequency_reference(window)
213            .map_err(|error| error.to_string())?;
214        let frequencies = column
215            .get_frequencies(window)
216            .map_err(|error| error.to_string())?;
217        let frequencies = frequencies.iter().map(MFrequency::hz).collect::<Vec<_>>();
218        let frequencies_hz = bins
219            .iter()
220            .map(|bin| {
221                mean_bin_frequency_hz(&frequencies, bin)
222                    .ok_or("Missing selected channel frequency for PlotMS time averaging")
223            })
224            .collect::<Result<Vec<_>, _>>()?;
225        Ok(Self {
226            frame,
227            source_channels: frequencies.len(),
228            frequencies_hz,
229        })
230    }
231}

233impl SpectralContext {
234    pub(super) fn mean_frequency_hz(&self, bin: &ChannelBin) -> Option<f64> {
235        mean_bin_frequency_hz(&self.channel_frequencies_hz, bin)
236    }
237}
238
239pub(super) fn mean_bin_frequency_hz(frequencies_hz: &[f64], bin: &ChannelBin) -> Option<f64> {
240    // PlotMS takes frequencies from the full physical output span, including
241    // the channels between strided samples. Visibility uses the cache members.
242    let span = frequencies_hz.get(bin.start..bin.end)?;
243    (!span.is_empty()).then(|| span.iter().sum::<f64>() / span.len() as f64)
244}

246pub(super) fn parse_frequency_frame(value: Option<&str>) -> Result<Option<FrequencyRef>, String> {
247    value
248        .map(str::trim)
249        .filter(|value| !value.is_empty())
250        .map(|value| {
251            value
252                .parse::<FrequencyRef>()
253                .map_err(|error| error.to_string())
254        })
255        .transpose()
256}
257
258pub(super) fn parse_velocity_definition(value: &str) -> Result<DopplerRef, String> {
259    match value.trim().to_ascii_uppercase().as_str() {
260        "RADIO" => Ok(DopplerRef::RADIO),
261        "OPTICAL" | "Z" => Ok(DopplerRef::Z),
262        "TRUE" | "RELATIVISTIC" | "BETA" => Ok(DopplerRef::BETA),
263        "RATIO" => Ok(DopplerRef::RATIO),
264        "GAMMA" => Ok(DopplerRef::GAMMA),
265        other => Err(format!(
266            "unsupported msexplore velocity definition {other:?}"
267        )),
268    }
269}

271pub(super) fn resolve_spectral_context(
272    spec: &MsPlotSpec,
273    spw_id: i32,
274    chan_count: usize,
275    (field_id_value, row_time_value): (i32, f64),
276    chan_freq: Option<&ChanFreqColumn<'_>>,
277    derived_engine: Option<&MsCalEngine>,
278    plotms_rest_bin: Option<&ChannelBin>,
279) -> Result<Option<SpectralContext>, String> {
280    let Some(chan_freq) = chan_freq else {
281        return Ok(None);
282    };
283
284    let spw_index = usize::try_from(spw_id)
285        .map_err(|_| format!("invalid DATA_DESCRIPTION::SPECTRAL_WINDOW_ID {spw_id}"))?;
286    let source_channel_frequencies = chan_freq
287        .get_frequencies(spw_index)
288        .map_err(|error| error.to_string())?;
289    if source_channel_frequencies.len() != chan_count {
290        return Err(format!(
291            "SPECTRAL_WINDOW row {spw_id} reported {} channel frequencies for {} data channels",
292            source_channel_frequencies.len(),
293            chan_count
294        ));
295    }
296
297    let source_ref = match source_channel_frequencies.first() {
298        Some(frequency) => frequency.refer(),
299        None => chan_freq
300            .frequency_reference(spw_index)
301            .map_err(|error| error.to_string())?,
302    };
303    let target_ref =
304        parse_frequency_frame(spec.transforms.freqframe.as_deref())?.unwrap_or(source_ref);
305    let channel_frequencies_hz = if target_ref == source_ref {
306        source_channel_frequencies
307            .iter()
308            .map(MFrequency::hz)
309            .collect::<Vec<_>>()
310    } else {
311        let derived_engine = derived_engine.ok_or_else(|| {
312            "internal error: missing derived engine for spectral-frame conversion".to_string()
313        })?;
314        let field_id = usize::try_from(field_id_value)
315            .map_err(|_| format!("invalid FIELD_ID {field_id_value} for spectral transform"))?;
316        source_channel_frequencies
317            .iter()
318            .map(|frequency| {
319                convert_frequency_to_frame(
320                    source_ref,
321                    target_ref,
322                    frequency.hz(),
323                    row_time_value,
324                    field_id,
325                    derived_engine,
326                )
327                .map_err(|error| error.to_string())
328            })
329            .collect::<Result<Vec<_>, _>>()?
330    };
331    let rest_frequency_hz = spec
332        .transforms
333        .restfreq
334        .as_deref()
335        .map(parse_rest_frequency_hz)
336        .transpose()
337        .map_err(|error| error.to_string())?
338        .unwrap_or_else(|| {
339            // The public PlotMS task defaults to the middle selected buffer channel.
340            // The msexplore request defaults to the arithmetic centre of the window.
341            plotms_rest_bin
342                .and_then(|bin| mean_bin_frequency_hz(&channel_frequencies_hz, bin))
343                .unwrap_or_else(|| {
344                    channel_frequencies_hz.iter().sum::<f64>() / channel_frequencies_hz.len() as f64
345                })
346        });
347    let doppler_ref = parse_velocity_definition(&spec.transforms.veldef)?;
348    Ok(Some(SpectralContext {
349        channel_frequencies_hz,
350        rest_frequency_hz,
351        doppler_ref,
352    }))
353}

355pub(super) fn time_reporting_frame(
356    time_wavelengths: bool,
357    requested_freqframe: Option<FrequencyRef>,
358    row_numbers: &[usize],
359    time_bins: Option<&PlotTimeGroups>,
360    selected_data_desc_ids: &[i32],
361    data_description: &MsDataDescription<'_>,
362    chan_freq: Option<&ChanFreqColumn<'_>>,
363) -> Result<Option<FrequencyRef>, String> {
364    let has_rows = !row_numbers.is_empty();
365    Ok(
366        if time_wavelengths && requested_freqframe.is_none() && has_rows {
367            let first_slot = time_bins
368                .expect("time wavelengths use the PlotMS iterator plan")
369                .first_row_slot(0);
370            let first_ddid = usize::try_from(selected_data_desc_ids[first_slot])
371                .map_err(|_| "Invalid DATA_DESC_ID for the first PlotMS buffer")?;
372            let first_window = data_description
373                .spectral_window_id(first_ddid)
374                .map_err(|error| error.to_string())?;
375            let first_window = usize::try_from(first_window)
376                .map_err(|_| "Invalid SPECTRAL_WINDOW_ID for the first PlotMS buffer")?;
377            Some(
378                chan_freq
379                    .expect("wavelength coordinates use channel frequencies")
380                    .frequency_reference(first_window)
381                    .map_err(|error| error.to_string())?,
382            )
383        } else {
384            requested_freqframe
385        },
386    )
