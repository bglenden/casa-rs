11/// One plane's part of a minor cycle: what `SDAlgorithmBase::deconvolve`
12/// hands `addSummaryMinor`.
13#[derive(Clone, Copy, Debug, PartialEq)]
14pub struct PlaneCycleRow {
15    /// The plane's image domain.
16    pub domain: usize,
17    /// The plane's absolute output channel; a Taylor family's first.
18    pub channel: usize,
19    /// The plane's polarization.
20    pub polarization: usize,
21    /// Iterations charged to the plane.
22    pub iterations: usize,
23    /// Peak residual inside the mask on entry; zero with no mask.
24    pub start_peak: f64,
25    /// Peak residual inside the mask at the end; zero with no mask.
26    pub peak: f64,
27    /// Peak residual over the plane's pixels on entry.
28    pub start_peak_no_mask: f64,
29    /// Peak residual over the plane's pixels at the end.
30    pub peak_no_mask: f64,
31    /// Signed first-term flux the plane added to the model.
32    pub model_flux: f64,
33    /// Pixels in the plane's mask.
34    pub mask_sum: usize,
35    /// Why the plane stopped.
36    pub stop: PlaneStop,
37}

39/// The row of a plane with no data.
40pub(super) fn blank_row(
41    key: PlaneKey,
42    statistics: &PlaneStatistics,
43    controls: &CycleControls,
44) -> PlaneCycleRow {
45    PlaneCycleRow {
46        domain: key.domain,
47        channel: key.channel,
48        polarization: key.polarization,
49        iterations: 0,
50        start_peak: 0.0,
51        peak: 0.0,
52        start_peak_no_mask: 0.0,
53        peak_no_mask: 0.0,
54        model_flux: 0.0,
55        mask_sum: statistics.mask_sum,
56        stop: blank_plane_stop(controls, statistics),
57    }
58}

60/// A plane with no data as CASA measures it: zero peaks, and its mask's
61/// pixels, which its zero residual doesn't change.
62pub(super) fn blank_statistics(
63    normal: &FinalNormalState,
64    masks: &ImageDomainReconstructionMasks,
65    key: PlaneKey,
66    automask: bool,
67) -> PlaneStatistics {
68    let mask = masks
69        .get(key.domain)
70        .and_then(|mask| {
71            mask.plane(
72                key.channel - normal.slab().core_range().start,
73                key.polarization,
74            )
75        })
76        .expect("a mask for every plane of every image domain");
77    PlaneStatistics {
78        peak_in_mask: 0.0,
79        peak_no_mask: 0.0,
80        mask_sum: mask.iter().filter(|masked| **masked).count(),
81        noise: None,
82        automask,
83    }
84}

86/// The row of a plane that `outcome` cleaned or skipped. A plane with no mask
87/// reports zero peaks inside it, and stops on that zero
88/// (`SDAlgorithmBase::deconvolve`, CAS-14201).
89pub(super) fn cycle_row(
90    key: PlaneKey,
91    statistics: &PlaneStatistics,
92    outcome: &PlaneOutcome,
93    peak_no_mask: f64,
94    controls: &CycleControls,
95) -> PlaneCycleRow {
96    let masked = |peak: f64| if statistics.mask_sum > 0 { peak } else { 0.0 };
97    PlaneCycleRow {
98        domain: key.domain,
99        channel: key.channel,
100        polarization: key.polarization,
101        iterations: outcome.iterations,
102        start_peak: masked(outcome.start_peak),
103        peak: masked(outcome.peak),
104        start_peak_no_mask: statistics.peak_no_mask,
105        peak_no_mask,
106        model_flux: outcome.model_flux,
107        mask_sum: statistics.mask_sum,
108        stop: if statistics.mask_sum > 0 {
109            outcome.stop
110        } else {
111            blank_plane_stop(controls, statistics)
112        },
113    }
114}

116#[cfg(test)]
117mod tests {
118    use casa_imaging_deconvolution::Delta;
119
120    use super::*;
121
122    fn controls() -> CycleControls {
123        CycleControls {
124            iterations: 5,
125            threshold: 0.1,
126            threshold_reached: false,
127            gain: 0.1,
128            nsigma: 0.0,
129        }
130    }
131
132    fn statistics(mask_sum: usize) -> PlaneStatistics {
133        PlaneStatistics {
134            peak_in_mask: if mask_sum > 0 { 1.0 } else { 0.0 },
135            peak_no_mask: 2.0,
136            mask_sum,
137            noise: None,
138            automask: false,
139        }
140    }
141
142    fn outcome(stop: PlaneStop) -> PlaneOutcome {
143        PlaneOutcome {
144            delta: Delta::new(1),
145            iterations: 3,
146            components: 3,
147            absolute_flux: 1.0,
148            model_flux: 0.5,
149            start_peak: 2.0,
150            peak: 1.5,
151            stop,
152            trace: Vec::new(),
153            refreshes: 0,
154            end_residual: Vec::new(),
155        }
156    }
157
158    const KEY: PlaneKey = PlaneKey {
159        domain: 1,
160        channel: 2,
161        polarization: 3,
162    };
163
164    /// A cleaned plane's row is its own outcome.
165    #[test]
166    fn a_cleaned_planes_row_holds_its_outcome() {
167        let row = cycle_row(
168            KEY,
169            &statistics(4),
170            &outcome(PlaneStop::Iterations),
171            0.75,
172            &controls(),
173        );
174        assert_eq!(
175            [
176                row.domain,
177                row.channel,
178                row.polarization,
179                row.iterations,
180                row.mask_sum
181            ],
182            [1, 2, 3, 3, 4]
183        );
184        assert_eq!(
185            [
186                row.start_peak,
187                row.peak,
188                row.start_peak_no_mask,
189                row.peak_no_mask,
190                row.model_flux
191            ],
192            [2.0, 1.5, 2.0, 0.75, 0.5]
193        );
194        assert_eq!(row.stop, PlaneStop::Iterations);
195    }
196
197    /// CAS-14201: a plane with no mask holds zero peaks inside it, and
198    /// `checkStop` on that zero stops it at the threshold, though its data
199    /// peak above it, as `SDAlgorithmBase::deconvolve` does.
200    #[test]
201    fn a_plane_with_no_mask_stops_on_a_zero_peak() {
202        let row = cycle_row(
203            KEY,
204            &statistics(0),
205            &outcome(PlaneStop::ZeroMask),
206            2.0,
207            &controls(),
208        );
209        assert_eq!([row.start_peak, row.peak], [0.0, 0.0]);
210        assert_eq!(row.start_peak_no_mask, 2.0);
211        assert_eq!(row.stop, PlaneStop::CycleThreshold);
212    }
213
214    /// A plane with no data holds zero rows and its mask's pixels.
215    #[test]
216    fn a_blank_planes_row_is_empty_with_its_mask() {
217        let row = blank_row(KEY, &statistics(7), &controls());
218        assert_eq!(row.iterations, 0);
219        assert_eq!(row.mask_sum, 7);
220        assert_eq!(
221            [row.peak, row.peak_no_mask, row.model_flux],
222            [0.0, 0.0, 0.0]
223        );
224        assert_eq!(row.stop, PlaneStop::CycleThreshold);
225    }
