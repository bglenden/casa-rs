3//! per catalogue antenna, one field, observation, state, processor and
4//! source record per catalogue entry ([`super::msv2_identity`],
5//! [`super::msv2_sources`]), one spectral window per window a partition

29/// The MeasurementSet to create for `catalogue` with the visibility
30/// `columns`: a `SOURCE` subtable when the catalogue holds sources or some
31/// field names its source.
32pub(super) fn builder(
33    catalogue: &Catalogue,
34    columns: &[OptionalMainColumn],
35) -> MeasurementSetBuilder {
36    let builder = columns
37        .iter()
38        .fold(MeasurementSetBuilder::new(), |builder, column| {
39            builder.with_main_column(*column)
40        });
41    if !catalogue.sources.is_empty()
42        || catalogue
43            .fields
44            .iter()
45            .any(|field| field.source_name.is_some())
46    {
47        builder.with_optional_subtable(SubtableId::Source)
48    } else {
49        builder
50    }
51}

53/// Where each partition's rows point, in catalogue partition order.
54pub(super) struct Layout {
55    /// `DATA_DESC_ID` of each partition.
56    pub(super) data_description: Vec<i32>,
57    /// `SPECTRAL_WINDOW` row of each partition.
58    pub(super) spectral_window: Vec<i32>,
59    /// The `STATE_ID` of each partition's state of intents, for its rows
60    /// that state intents but no state.
61    pub(super) intent_states: Vec<Option<i32>>,

64/// What writing a catalogue's subtables writes as MSv2's default, or
65/// leaves out, each a loss.
66#[derive(Default)]
67pub(super) struct Notes(BTreeSet<Loss>);
68
69impl Notes {
70    /// `value`, or when unstated `default`, noting `column` as defaulted.
71    pub(super) fn or<T>(
72        &mut self,
73        value: Option<T>,
74        column: &str,
75        default: impl FnOnce() -> T,
76    ) -> T {
77        value.unwrap_or_else(|| {
78            self.defaulted(column);
79            default()
80        })
81    }
82
83    /// Note `column` as written with MSv2's default for some row.
84    pub(super) fn defaulted(&mut self, column: &str) {
85        self.0.insert(Loss::Defaulted(column.to_string()));
86    }
87
88    /// Note `loss`.
89    pub(super) fn lost(&mut self, loss: Loss) {
90        self.0.insert(loss);
91    }
92
93    /// Every loss noted, in order.
94    pub(super) fn into_losses(self) -> Vec<Loss> {
95        self.0.into_iter().collect()
96    }
97}

99/// Refuse what an MSv2's subtables cannot say of `catalogue`, before
100/// anything is written: a scan without a casacore number, feeds whose
101/// receptors differ, or a correlation product the feeds cannot name (a
102/// Stokes product, or a receptor they lack).
103pub(super) fn refuse_unwritable(catalogue: &Catalogue) -> Result<(), ObservationError> {
104    scan_numbers(catalogue)?;
105    let receptors = feed_receptors(catalogue)?;
106    for entry in &catalogue.partitions {
107        for correlation in &entry.correlations {
108            receptor_pair(*correlation, receptors)?;
109        }
110    }
111    Ok(())
112}

114/// Each catalogue scan's `SCAN_NUMBER`.
115fn scan_numbers(catalogue: &Catalogue) -> Result<Vec<i32>, ObservationError> {
116    catalogue
117        .scans
118        .iter()
119        .map(|scan| {
120            scan.number.ok_or_else(|| {
121                request_error(format!(
122                    "scan {} has no casacore scan number to write as SCAN_NUMBER",
123                    scan.name
124                ))
125            })
126        })
127        .collect()
128}

130/// The receptors `CORR_PRODUCT` indexes: those of every antenna a
131/// partition states receptors for, which must agree (ignoring case, as
132/// [`receptor_pair`] matches them).
133fn feed_receptors(catalogue: &Catalogue) -> Result<&[String], ObservationError> {
134    let mut fed = catalogue
135        .partitions
136        .iter()
137        .flat_map(|entry| entry.receptors.values())
138        .map(|receptors| receptors.types.as_slice())
139        .filter(|types| !types.is_empty());
140    let first = fed.next().unwrap_or(&[]);
141    let same = |receptors: &[String]| {
142        receptors.len() == first.len()
143            && receptors
144                .iter()
145                .zip(first)
146                .all(|(label, first)| label.eq_ignore_ascii_case(first))
147    };
148    if fed.any(|receptors| !same(receptors)) {
149        return Err(ObservationError::Unsupported {
150            what: "writing MSv2 polarizations for feeds whose receptors differ",
151        });
152    }
153    Ok(first)
154}

169/// The subtables of `catalogue` written into a MeasurementSet in memory,
170/// for where they place each partition, what writing them notes and
171/// refuses, and what reopening them gives: a plan learns its losses, and
172/// that its subtables can be written, before anything is. `intents` says
173/// which partitions need a state of intents.
174pub(super) fn rehearse(
175    catalogue: &Catalogue,
176    spans: &Spans,
177    intents: &[bool],
178) -> Result<Rehearsal, ObservationError> {
179    let mut ms = MeasurementSet::create_memory(builder(catalogue, &[])).map_err(storage)?;
180    let (layout, notes) = populate(&mut ms, catalogue, spans, intents)?;
181    Ok(Rehearsal {
182        layout,
183        losses: notes.into_losses(),
184        read_back: Msv2Observation::subtable_catalogue(&ms)?,
185    })
186}

188/// Fill `ms`'s subtables from `catalogue`, whose data span the times
189/// `spans` gives (MJD seconds), `intents` saying which partitions need a
190/// state of intents; [`refuse_unwritable`] has accepted it. Antennas,
191/// fields, sessions, states, processors and sources keep their catalogue
192/// order, and windows theirs among those a partition uses.
193pub(super) fn populate(
194    ms: &mut MeasurementSet,
195    catalogue: &Catalogue,
196    spans: &Spans,
197    intents: &[bool],
198) -> Result<(Layout, Notes), ObservationError> {
199    let time_range = spans.time_range;
200    let mut notes = Notes::default();
201    antennas(ms, catalogue, &mut notes)?;
202    let times = (time_range[0], spans.field_first.as_slice());
203    let named_sources = msv2_sources::named(catalogue, times)?;
204    msv2_identity::fields(ms, catalogue, (&named_sources, times), &mut notes)?;
205    msv2_identity::observations(ms, catalogue, &spans.observation_ranges, &mut notes)?;
206    let window_rows = msv2_identity::spectral_windows(ms, catalogue, &mut notes)?;
207    let states = msv2_identity::states(ms, catalogue, intents, &mut notes)?;
208    msv2_identity::processors(ms, catalogue, &mut notes)?;
209    msv2_sources::write(
210        ms,
211        catalogue,
212        (&named_sources, &window_rows),
213        time_range,
214        &mut notes,
215    )?;
216    let mut polarizations: Vec<&[Correlation]> = Vec::new();
217    let mut descriptions: Vec<(i32, usize)> = Vec::new();
218    let mut keys: Vec<(usize, &str, u32)> = Vec::new();
219    let mut layout = Layout {
220        data_description: Vec::new(),
221        spectral_window: Vec::new(),
222        intent_states: states.intent_states,
223    };
224    for (index, entry) in catalogue.partitions.iter().enumerate() {
225        let window = window_rows
226            .get(entry.window.0 as usize)
227            .copied()
228            .flatten()
229            .ok_or_else(|| request_error("a partition names a window the catalogue lacks"))?;
230        let polarization = position_or_push(&mut polarizations, &entry.correlations, |known| {
231            *known == entry.correlations.as_slice()
232        });
233        // Partitions an MSv2 keys alike (data description, OBS_MODE,
234        // observation) would read back as one: a later one gets a data
235        // description of its own.
236        let key = |description| {
237            (
238                description,
239                states.modes[index].as_str(),
240                entry.observation.0,
241            )
242        };
243        let description = (0..descriptions.len())
244            .find(|index| {
245                descriptions[*index] == (window, polarization) && !keys.contains(&key(*index))
246            })
247            .unwrap_or_else(|| {
248                descriptions.push((window, polarization));
249                descriptions.len() - 1
250            });
251        keys.push(key(description));
252        layout.data_description.push(description as i32);
253        layout.spectral_window.push(window);
254    }
255    feeds(ms, catalogue, &layout.spectral_window, time_range)?;
256    let receptors = feed_receptors(catalogue)?;
257    for correlations in &polarizations {
258        polarization(ms, correlations, receptors)?;
259    }
260    for (window, polarization) in &descriptions {
261        subtable_mut(ms, SubtableId::DataDescription)
262            .and_then(|table| {
263                table
264                    .add_row(row_from_defs(
265                        schema::data_description::REQUIRED_COLUMNS,
266                        &[
267                            ("SPECTRAL_WINDOW_ID", i(*window)),
268                            ("POLARIZATION_ID", i(*polarization as i32)),
269                            ("FLAG_ROW", b(false)),
270                        ],
271                    ))
272                    .map_err(Into::into)
273            })
274            .map_err(storage)?;
275    }
276    Ok((layout, notes))
277}

279/// The index of the first of `known` that `is` matches, after pushing
280/// `item` when none does.
281fn position_or_push<T: Copy>(known: &mut Vec<T>, item: T, is: impl Fn(&T) -> bool) -> usize {
282    known.iter().position(is).unwrap_or_else(|| {
283        known.push(item);
284        known.len() - 1
285    })
286}

288fn antennas(
289    ms: &mut MeasurementSet,
290    catalogue: &Catalogue,
291    notes: &mut Notes,
292) -> Result<(), ObservationError> {
293    let metres = Unit::new("m").expect("metres");
294    let table = subtable_mut(ms, SubtableId::Antenna).map_err(storage)?;
295    for antenna in &catalogue.antennas {
296        let diameter = antenna
297            .dish_diameter
298            .get_value_in(&metres)
299            .map_err(|_| request_error("a dish diameter is not a length"))?;
300        let flagged = notes.or(antenna.flagged, "ANTENNA FLAG_ROW", || false);
301        let row = row_from_defs(
302            schema::antenna::REQUIRED_COLUMNS,
303            &[
304                ("NAME", s(&antenna.name)),
305                ("STATION", s(&antenna.station)),
306                ("TYPE", s("GROUND-BASED")),
307                ("MOUNT", s(&antenna.mount)),
308                ("POSITION", f64_array(&antenna.position.as_itrf(), vec![3])),
309                ("OFFSET", f64_array(&[0.0; 3], vec![3])),
310                ("DISH_DIAMETER", f(diameter)),
311                ("FLAG_ROW", b(flagged)),
312            ],
313        );
314        table.add_row(row).map_err(storage)?;
315    }
316    Ok(())
317}

319/// The `FEED` rows of `catalogue`'s antennas, each partition in its window
320/// (`windows`: each partition's), written as [`feed_plan`] plans them.
321fn feeds(
322    ms: &mut MeasurementSet,
323    catalogue: &Catalogue,
324    windows: &[i32],
325    time_range: [f64; 2],
326) -> Result<(), ObservationError> {
327    let named: Vec<Vec<AntennaId>> = catalogue
328        .partitions
329        .iter()
330        .map(|entry| {
331            let mut named: Vec<AntennaId> = entry
332                .baselines
333                .iter()
334                .flat_map(|baseline| [baseline.antenna1, baseline.antenna2])
335                .collect();
336            named.sort_unstable();
337            named.dedup();
338            named
339        })
340        .collect();
341    let partitions = catalogue
342        .partitions
343        .iter()
344        .zip(windows)
345        .zip(&named)
346        .map(|((entry, window), named)| (*window, named.as_slice(), &entry.receptors));
347    for (antenna, window, receptors) in feed_plan(catalogue.antennas.len(), partitions)? {
348        feed_row(ms, antenna, window, receptors, time_range)?;
349    }
350    Ok(())
351}

353/// The `FEED` rows (antenna, window, receptors) of `antennas` antennas that
354/// read back as each partition's receptors (`partitions`: its window, the
355/// antennas its baselines name and the receptors it states) under
356/// casacore's rule (`MSIter::setFeedInfo`, as XRADIO's converter reads it
357/// too): once any row names a window, a window takes its own rows only and
358/// rows for every window (-1) are ignored. So when every antenna states the
359/// same receptors in every window it appears in, each has one row for every
360/// window (-1); otherwise each has a row per window it states receptors in,
361/// and a window where it states none (ALMA's WVR window) has no row, as
362/// ALMA writes `FEED`. An antenna stating none has no row, unless no antenna
363/// states any, when each has a row of none. Partitions of one window that
364/// state an antenna's receptors differently are refused: one `FEED` row
365/// holds them.
366fn feed_plan<'a>(
367    antennas: usize,
368    partitions: impl Iterator<Item = (i32, &'a [AntennaId], &'a BTreeMap<AntennaId, Receptors>)>,
369) -> Result<Vec<(usize, i32, &'a Receptors)>, ObservationError> {
370    static NONE: Receptors = Receptors {
371        types: Vec::new(),
372        angles: Vec::new(),
373    };
374    // Each antenna's receptors in each window it appears in.
375    let mut seen: Vec<Vec<(i32, Option<&Receptors>)>> = vec![Vec::new(); antennas];
376    for (window, named, receptors) in partitions {
377        for antenna in named.iter().chain(receptors.keys()) {
378            let Some(own) = seen.get_mut(antenna.0 as usize) else {
379                return Err(request_error("receptors of an antenna the catalogue lacks"));
380            };
381            let stated = receptors.get(antenna);
382            match own.iter().find(|(known, _)| *known == window) {
383                Some((_, known)) if *known != stated => {
384                    return Err(ObservationError::Unsupported {
385                        what: "partitions of one window with different receptors for an antenna",
386                    });
387                }
388                Some(_) => {}
389                None => own.push((window, stated)),
390            }
391        }
392    }
393    if !seen.iter().flatten().any(|(_, stated)| stated.is_some()) {
394        return Ok((0..antennas).map(|antenna| (antenna, -1, &NONE)).collect());
395    }
396    let per_window = seen
397        .iter()
398        .any(|own| own.iter().any(|(_, stated)| *stated != own[0].1));
399    Ok(seen
400        .into_iter()
401        .enumerate()
402        .flat_map(|(antenna, own)| {
403            let rows: Vec<(i32, &Receptors)> = if per_window {
404                own.into_iter()
405                    .filter_map(|(window, stated)| Some((window, stated?)))
406                    .collect()
407            } else {
408                own.first()
409                    .and_then(|(_, stated)| *stated)
410                    .map(|receptors| (-1, receptors))
411                    .into_iter()
412                    .collect()
413            };
414            rows.into_iter()
415                .map(move |(window, receptors)| (antenna, window, receptors))
416        })
417        .collect())
418}

420/// One `FEED` row: antenna `index`'s feed 0 in spectral window `window`.
421fn feed_row(
422    ms: &mut MeasurementSet,
423    index: usize,
424    window: i32,
425    own: &Receptors,
426    time_range: [f64; 2],
427) -> Result<(), ObservationError> {
428    let radians = Unit::new("rad").expect("radians");
429    let receptors = own.types.len();
430    let angles = own
431        .angles
432        .iter()
433        .map(|angle| {
434            angle
435                .get_value_in(&radians)
436                .map_err(|_| request_error("a receptor angle is not an angle"))
437        })
438        .collect::<Result<Vec<_>, _>>()?;
439    let types: Vec<&str> = own.types.iter().map(String::as_str).collect();
440    let identity: Vec<Complex32> = (0..receptors * receptors)
441        .map(|k| {
442            if k % (receptors + 1) == 0 {
443                Complex32::new(1.0, 0.0)
444            } else {
445                Complex32::new(0.0, 0.0)
446            }
447        })
448        .collect();
449    let row = row_from_defs(
450        schema::feed::REQUIRED_COLUMNS,
451        &[
452            ("ANTENNA_ID", i(index as i32)),
453            ("BEAM_ID", i(-1)),
454            (
455                "BEAM_OFFSET",
456                f64_array(&vec![0.0; 2 * receptors], vec![2, receptors]),
457            ),
458            ("FEED_ID", i(0)),
459            ("INTERVAL", f(time_range[1] - time_range[0])),
460            ("NUM_RECEPTORS", i(receptors as i32)),
461            (
462                "POL_RESPONSE",
463                complex_array(&identity, vec![receptors, receptors]),
464            ),
465            ("POLARIZATION_TYPE", string_array(&types, vec![receptors])),
466            ("POSITION", f64_array(&[0.0; 3], vec![3])),
467            ("RECEPTOR_ANGLE", f64_array(&angles, vec![receptors])),
468            ("SPECTRAL_WINDOW_ID", i(window)),
469            ("TIME", f(0.5 * (time_range[0] + time_range[1]))),
470        ],
471    );
472    subtable_mut(ms, SubtableId::Feed)
473        .and_then(|table| table.add_row(row).map_err(Into::into))
474        .map_err(storage)
475}

477/// The receptors a correlation product correlates, by their positions in
478/// the feed's `receptors` (`R`, `L`, `X`, `Y`), as `CORR_PRODUCT` holds
479/// them.
480fn receptor_pair(
481    correlation: Correlation,
482    receptors: &[String],
483) -> Result<[i32; 2], ObservationError> {
484    use Correlation::{LL, LR, RL, RR, XX, XY, YX, YY};
485    let [first, second] = match correlation {
486        RR => ["R", "R"],
487        RL => ["R", "L"],
488        LR => ["L", "R"],
489        LL => ["L", "L"],
490        XX => ["X", "X"],
491        XY => ["X", "Y"],
492        YX => ["Y", "X"],
493        YY => ["Y", "Y"],
494        _ => {
495            return Err(ObservationError::Unsupported {
496                what: "writing MSv2 correlation products other than a feed pair's",
497            });
498        }
499    };
500    let position = |label: &str| {
501        receptors
502            .iter()
503            .position(|receptor| receptor.eq_ignore_ascii_case(label))
504            .map(|index| index as i32)
505            .ok_or(ObservationError::Unsupported {
506                what: "writing an MSv2 correlation product of a receptor the feeds lack",
507            })
508    };
509    Ok([position(first)?, position(second)?])
510}

512fn polarization(
513    ms: &mut MeasurementSet,
514    correlations: &[Correlation],
515    receptors: &[String],
516) -> Result<(), ObservationError> {
517    let count = correlations.len();
518    let types: Vec<i32> = correlations
519        .iter()
520        .map(|correlation| correlation.casacore_code())
521        .collect();
522    let mut products = vec![0; 2 * count];
523    for (index, correlation) in correlations.iter().enumerate() {
524        let [first, second] = receptor_pair(*correlation, receptors)?;
525        products[index] = first;
526        products[count + index] = second;
527    }
528    let row = row_from_defs(
529        schema::polarization::REQUIRED_COLUMNS,
530        &[
531            ("NUM_CORR", i(count as i32)),
532            ("CORR_TYPE", i32_array(&types, vec![count])),
533            ("CORR_PRODUCT", i32_array(&products, vec![2, count])),
534            ("FLAG_ROW", b(false)),
535        ],
536    );
537    subtable_mut(ms, SubtableId::Polarization)
538        .and_then(|table| table.add_row(row).map_err(Into::into))
539        .map_err(storage)
540}

621    ) -> (i32, Vec<AntennaId>, BTreeMap<AntennaId, Receptors>) {
622        (window, vec![AntennaId(0), AntennaId(1)], receptors)
623    }
624
625    /// Windows that agree give each antenna stating receptors one row for
626    /// every window; once one antenna differs by window, or a window states
627    /// none for an antenna it names (ALMA's WVR window), each has a row per
628    /// window it states receptors in. Rows of none are written only when no
629    /// antenna states any.
630    #[test]
631    fn feed_rows_are_per_window_once_any_antenna_differs() {
632        let agree = plan(&[
633            named(0, receptors(&[0, 1], 0.0)),
634            named(1, receptors(&[0, 1], 0.0)),
635        ]);
636        assert_eq!(agree, [(0, -1, 0.0, 2), (1, -1, 0.0, 2)]);
637
638        let mut differ = receptors(&[0, 1], 0.0);
639        differ.extend(receptors(&[0], 0.25));
640        let mixed = plan(&[named(0, receptors(&[0, 1], 0.0)), named(1, differ)]);
641        assert_eq!(
642            mixed,
643            [
644                (0, 0, 0.0, 2),
645                (0, 1, 0.25, 2),
646                (1, 0, 0.0, 2),
647                (1, 1, 0.0, 2)
648            ]
649        );
650
651        let wvr = plan(&[named(0, receptors(&[0, 1], 0.0)), named(1, BTreeMap::new())]);
652        assert_eq!(wvr, [(0, 0, 0.0, 2), (1, 0, 0.0, 2)], "no rows in window 1");
653
654        let none = plan(&[named(0, BTreeMap::new())]);
655        assert_eq!(
656            none.iter()
657                .map(|row| (row.0, row.1, row.3))
658                .collect::<Vec<_>>(),
659            [(0, -1, 0), (1, -1, 0), (2, -1, 0)]
660        );
661
662        let clash = [
663            named(0, receptors(&[0], 0.0)),
664            named(0, receptors(&[0], 0.5)),
665        ];
666        assert!(
667            feed_plan(
668                3,
669                clash
670                    .iter()
671                    .map(|(window, antennas, own)| (*window, antennas.as_slice(), own))
672            )
673            .is_err(),
674            "one window, two sets"
675        );
676    }
