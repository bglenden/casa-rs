3//! Restoration and normalization algorithms for continuum image products.
4//!
5//! The restoring kernel is evaluated in physical radians: fitted beam widths
6//! and the image cell scale share one unit system, so a multi-pixel beam
7//! stays a multi-pixel kernel at any cell size.
8
9use ndarray::{Array2, Axis};
10use num_complex::Complex64;
11use rustfft::FftPlanner;
12
13use casa_imaging_model::ProductNormalization;
14
15use crate::beam::RestoringBeam;
16use crate::error::ProductsError;

18const FWHM_TO_SIGMA: f64 = 1.0 / 2.354_820_045_030_949_3;
19
20/// Normalize one unnormalized plane to its compiled product normalization.
21///
22/// `UnitResponse` keeps the plane as-is. `FlatNoise` divides by the scalar
23/// sensitivity (sum weight); when no usable sensitivity exists every pixel
24/// blanks to NaN. `FlatSky` is not produced by this catalog version.
25///
26/// # Errors
27///
28/// Fails when the normalization is not representable by this catalog.
29pub fn normalize_plane(
30    values: &[f32],
31    normalization: ProductNormalization,
32    sensitivity: f64,
33) -> Result<Vec<f32>, ProductsError> {
34    match normalization {
35        ProductNormalization::UnitResponse => Ok(values.to_vec()),
36        ProductNormalization::FlatNoise => {
37            if !(sensitivity.is_finite() && sensitivity > 0.0) {
38                return Ok(vec![f32::NAN; values.len()]);
39            }
40            Ok(values
41                .iter()
42                .map(|value| value / sensitivity as f32)
43                .collect())
44        }
45        ProductNormalization::FlatSky => Err(ProductsError::UnsupportedProductRole {
46            role: casa_imaging_model::ProductRole::Sensitivity,
47            catalog: crate::CONTINUUM_ALGORITHM_CATALOG_VERSION,
48        }),
49    }
50}

52/// Build the unit-peak elliptical-Gaussian restoring kernel image.
53///
54/// `cell_size_rad` is the image cell scale in radians per pixel on each
55/// axis: the fitted beam widths are radians, so pixel offsets must be
56/// converted to radians before they are divided by the Gaussian sigmas.
57#[must_use]
58pub fn gaussian_beam_image(
59    shape: [usize; 2],
60    beam: &RestoringBeam,
61    cell_size_rad: [f64; 2],
62) -> Array2<f32> {
63    let [width, height] = shape;
64    let sigma_major = (beam.major_fwhm_rad() * FWHM_TO_SIGMA).abs();
65    let sigma_minor = (beam.minor_fwhm_rad() * FWHM_TO_SIGMA).abs();
66    let cos_pa = beam.position_angle_rad().cos();
67    let sin_pa = beam.position_angle_rad().sin();
68    let centre_x = width as f64 / 2.0;
69    let centre_y = height as f64 / 2.0;
70    let mut kernel = Array2::<f32>::zeros((width, height));
71    for x in 0..width {
72        for y in 0..height {
73            let dx = (x as f64 - centre_x) * cell_size_rad[0];
74            let dy = (y as f64 - centre_y) * cell_size_rad[1];
75            let u = dx * cos_pa + dy * sin_pa;
76            let v = -dx * sin_pa + dy * cos_pa;
77            kernel[(x, y)] =
78                (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2))).exp() as f32;
79        }
80    }
81    let peak = kernel.iter().copied().fold(f32::NEG_INFINITY, f32::max);
82    if peak > 0.0 {
83        kernel.mapv_inplace(|value| value / peak);
84    }
85    kernel
86}

88/// Convolve one real plane with an equal-shape kernel through even-shifted
89/// FFTs, matching the reconstruction owner's transform conventions.
90#[must_use]
91pub fn fft_convolve(plane: &[f32], kernel: &[f32], shape: [usize; 2]) -> Vec<f32> {
92    let cells = shape[0] * shape[1];
93    let mut planner = FftPlanner::<f64>::new();
94    let mut signal = Array2::<Complex64>::from_shape_vec(
95        (shape[0], shape[1]),
96        plane
97            .iter()
98            .map(|v| Complex64::new(f64::from(*v), 0.0))
99            .collect(),
100    )
101    .expect("shape matches payload");
102    let mut response = Array2::<Complex64>::from_shape_vec(
103        (shape[0], shape[1]),
104        kernel
105            .iter()
106            .map(|v| Complex64::new(f64::from(*v), 0.0))
107            .collect(),
108    )
109    .expect("shape matches kernel");
110
111    for data in [&mut signal, &mut response] {
112        shift_even(data);
113        for axis in 0..2 {
114            let plan = planner.plan_fft_forward(data.len_of(Axis(axis)));
115            let scratch_len = plan.get_inplace_scratch_len();
116            let length = data.len_of(Axis(axis));
117            let mut lane = vec![Complex64::default(); length];
118            let mut scratch = vec![Complex64::default(); scratch_len];
119            for mut view in data.lanes_mut(Axis(axis)) {
120                lane.iter_mut()
121                    .zip(view.iter())
122                    .for_each(|(target, source)| {
123                        *target = *source;
124                    });
125                plan.process_with_scratch(&mut lane, &mut scratch);
126                lane.iter()
127                    .zip(view.iter_mut())
128                    .for_each(|(source, target)| *target = *source);
129            }
130        }
131        shift_even(data);
132    }
133    for (signal, response) in signal.iter_mut().zip(response.iter()) {
134        *signal *= *response;
135    }
136    shift_even(&mut signal);
137    for axis in 0..2 {
138        let plan = planner.plan_fft_inverse(signal.len_of(Axis(axis)));
139        let scratch_len = plan.get_inplace_scratch_len();
140        let length = signal.len_of(Axis(axis));
141        let mut lane = vec![Complex64::default(); length];
142        let mut scratch = vec![Complex64::default(); scratch_len];
143        for mut view in signal.lanes_mut(Axis(axis)) {
144            lane.iter_mut()
145                .zip(view.iter())
146                .for_each(|(target, source)| *target = *source);
147            plan.process_with_scratch(&mut lane, &mut scratch);
148            lane.iter()
149                .zip(view.iter_mut())
150                .for_each(|(source, target)| *target = *source);
151        }
152    }
153    shift_even(&mut signal);
154
155    let scale = 1.0 / cells as f64;
156    signal
157        .iter()
158        .map(|value| (value.re * scale) as f32)
159        .collect()
160}
161
162fn shift_even(data: &mut Array2<Complex64>) {
163    let [width, height] = [data.shape()[0], data.shape()[1]];
164    debug_assert_eq!(width % 2, 0);
165    debug_assert_eq!(height % 2, 0);
166    for x in 0..width / 2 {
167        for y in 0..height / 2 {
168            data.swap((x, y), (x + width / 2, y + height / 2));
169            data.swap((x + width / 2, y), (x, y + height / 2));
170        }
171    }
172}
