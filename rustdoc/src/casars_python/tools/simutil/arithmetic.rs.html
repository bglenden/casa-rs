2//! Pointing arithmetic using the caller's Python numeric protocols.
3
4use pyo3::class::basic::CompareOp;
5use pyo3::prelude::*;
6
7use super::PySimutil;
8
9#[pymethods]
10#[allow(non_snake_case)]
11impl PySimutil {
12    /// Rotate coordinates, preserving Python and NumPy arithmetic types.
13    fn applyRotate<'py>(
14        &self,
15        x: &Bound<'py, PyAny>,
16        y: &Bound<'py, PyAny>,
17        tcos: &Bound<'py, PyAny>,
18        tsin: &Bound<'py, PyAny>,
19    ) -> PyResult<(Bound<'py, PyAny>, Bound<'py, PyAny>)> {
20        Ok((
21            tcos.mul(x)?.sub(tsin.mul(y)?)?,
22            tsin.mul(x)?.add(tcos.mul(y)?)?,
23        ))
24    }
25
26    /// Whether each input number is even, with the input's comparison shape.
27    fn isEven<'py>(&self, num: &Bound<'py, PyAny>) -> PyResult<Bound<'py, PyAny>> {
28        num.rem(2)?.rich_compare(0, CompareOp::Eq)
29    }
30
31    /// Return 1 if the pattern needs an extra row or column, otherwise 0.
32    fn needsFiller(
33        &self,
34        length: &Bound<'_, PyAny>,
35        spacing: &Bound<'_, PyAny>,
36        bs: &Bound<'_, PyAny>,
37        npoints: &Bound<'_, PyAny>,
38    ) -> PyResult<i32> {
39        let limit = spacing.mul(npoints.sub(1)?)?.add(bs)?;
40        Ok(i32::from(
41            length.rich_compare(limit, CompareOp::Gt)?.is_truthy()?,
42        ))
43    }
44}
