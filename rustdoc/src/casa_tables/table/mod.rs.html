102/// Byte-ordering format for on-disk table data.
103///
104/// Controls how multi-byte values (integers, floats, complex numbers) are
105/// stored in the table's data files. The choice is recorded in `table.dat`
106/// so that readers can decode values correctly regardless of the host machine's
107/// native byte order.
108///
109/// In C++ casacore this corresponds to `Table::EndianFormat`. The
110/// `AipsrcEndian` variant from C++ is intentionally omitted — Rust callers
111/// should pass an explicit format instead.
112///
113/// # Default
114///
115/// The default is [`LocalEndian`](EndianFormat::LocalEndian), which uses the
116/// byte order of the machine that creates the table. This matches the C++
117/// casacore default.
118#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
119pub enum EndianFormat {
120    /// Store data in big-endian (network / "canonical") byte order.
121    ///
122    /// Corresponds to `Table::BigEndian` in C++ casacore. This was the only
123    /// format supported by early casacore versions.
124    BigEndian,
125    /// Store data in little-endian byte order.
126    ///
127    /// Corresponds to `Table::LittleEndian` in C++ casacore.
128    LittleEndian,
129    /// Store data in the byte order of the host machine.
130    ///
131    /// On x86-64 and ARM64 this resolves to little-endian; on SPARC or
132    /// PowerPC (big-endian mode) it resolves to big-endian.
133    /// Corresponds to `Table::LocalEndian` in C++ casacore.
134    #[default]
135    LocalEndian,
136}
137
138impl EndianFormat {
139    /// Resolves this format to a concrete big-endian flag.
140    ///
141    /// [`LocalEndian`](EndianFormat::LocalEndian) queries the host at compile
142    /// time via `cfg!(target_endian = "big")`.
143    pub fn is_big_endian(self) -> bool {
144        match self {
145            Self::BigEndian => true,
146            Self::LittleEndian => false,
147            Self::LocalEndian => cfg!(target_endian = "big"),
148        }
149    }
150}

169/// The kind of table: plain (disk-backed) or memory (transient).
170///
171/// In C++ casacore this corresponds to `Table::TableType`. A
172/// [`Plain`](TableKind::Plain) table can be loaded from and saved to disk
173/// with any [`DataManagerKind`]. A [`Memory`](TableKind::Memory) table
174/// holds all data exclusively in process memory; it is deleted when
175/// dropped.
176///
177/// Memory tables can be materialized to disk via [`Table::save`], which
178/// writes a plain table that is byte-identical to one created directly.
179///
180/// # C++ equivalent
181///
182/// `Table::Plain` and `Table::Memory`.
183#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
184pub enum TableKind {
185    /// A regular table, backed by (or destined for) disk storage.
186    #[default]
187    Plain,
188    /// A transient in-memory table.
189    ///
190    /// All data is lost when the table is dropped. Locking is a no-op:
191    /// [`has_lock`](Table::has_lock) always returns `true`, and
192    /// [`lock`](Table::lock) / [`unlock`](Table::unlock) succeed without
193    /// doing any I/O.
194    ///
195    /// C++ equivalent: `Table::Memory` / `MemoryTable`.
196    Memory,
197}

199/// Which data manager to use when writing table data.
200///
201/// This choice is recorded in the table descriptor on disk so that C++ casacore
202/// can select the correct storage-manager plugin when reopening the table.
203/// All variants produce files that are binary-compatible with upstream casacore.
204#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
205pub enum DataManagerKind {
206    /// Simple whole-column AipsIO streaming (legacy format).
207    ///
208    /// Each column is stored as a single flat AipsIO stream. This is the
209    /// simplest on-disk layout and is compatible with older versions of
210    /// casacore. It is the default for this crate because it requires no
211    /// extra configuration and produces self-contained files.
212    #[default]
213    StManAipsIO,
214    /// Bucket-based storage (the default in C++ casacore).
215    ///
216    /// Data is partitioned into fixed-size buckets, which allows efficient
217    /// random access and in-place updates. This is the storage manager used
218    /// by default when creating tables with C++ casacore.
219    StandardStMan,
220    /// Delta-compression storage manager (C++ `IncrementalStMan` / `ISMBase`).
221    ///
222    /// Stores column values only when they change from the previous row,
223    /// making it extremely space-efficient for slowly-changing columns
224    /// such as `ANTENNA1`, `FEED_ID`, or `SCAN_NUMBER` in a
225    /// MeasurementSet. Along with `StandardStMan`, ISM is one of the two
226    /// most commonly used storage managers in real radio astronomy data.
227    IncrementalStMan,
228    /// Tiled storage: single hypercube for the entire column.
229    ///
230    /// All rows must have the same array shape. Data is stored in
231    /// rectangular tiles within a single hypercube whose last dimension
232    /// is the row count. This is the standard format for large fixed-shape
233    /// array columns in measurement sets and images.
234    ///
235    /// C++ equivalent: `TiledColumnStMan`.
236    TiledColumnStMan,
237    /// Tiled storage: one hypercube per unique array shape.
238    ///
239    /// Rows with different array shapes are automatically grouped into
240    /// separate hypercubes. An internal row map tracks which cube holds
241    /// each row. This is the standard format for variable-shape array
242    /// columns (e.g. visibilities with varying channel counts).
243    ///
244    /// C++ equivalent: `TiledShapeStMan`.
245    TiledShapeStMan,
246    /// Tiled storage: one hypercube per row.
247    ///
248    /// Each row has its own cube, allowing fully variable shapes per row.
249    /// Most memory-intensive variant; use `TiledShapeStMan` when many rows
250    /// share shapes.
251    ///
252    /// C++ equivalent: `TiledCellStMan`.
253    TiledCellStMan,
254    /// Tiled storage: user-controlled hypercube assignment.
255    ///
256    /// Like `TiledShapeStMan` but with explicit row-to-cube assignment
257    /// rather than automatic shape-based grouping. Found in some older
258    /// datasets.
259    ///
260    /// C++ equivalent: `TiledDataStMan`.
261    TiledDataStMan,
262}

264/// Per-column data manager binding for [`Table::save_with_bindings`].
265///
266/// Specifies which storage manager to use for a particular column.
267/// Columns not listed in the bindings map use the default DM from
268/// [`TableOptions`].
269///
270/// This is the main public hook for steering large-write performance.
271/// For heterogeneous tables such as MeasurementSets, explicit bindings let
272/// the save path serialize compact column groups with storage-manager-specific
273/// code paths instead of treating the entire row as one generic record payload.
274///
275/// As a rule of thumb:
276/// - use tiled storage managers for large array-valued columns
277/// - use [`DataManagerKind::IncrementalStMan`] for slowly changing scalar
278///   columns
279/// - use [`DataManagerKind::StandardStMan`] for scalar columns that do not
280///   compress well under ISM
281///
282/// Callers writing large tables should prefer explicit bindings over relying
283/// on a single default data manager for every column.
284///
285/// # Example
286///
287/// ```rust
288/// use casa_tables::{ColumnBinding, DataManagerKind};
289///
290/// let binding = ColumnBinding {
291///     data_manager: DataManagerKind::TiledColumnStMan,
292///     tile_shape: Some(vec![4, 32]),
293/// };
294/// ```
295#[derive(Debug, Clone)]
296pub struct ColumnBinding {
297    /// The storage manager to use for this column.
298    pub data_manager: DataManagerKind,
299    /// Optional tile shape (only used with tiled storage managers).
300    pub tile_shape: Option<Vec<usize>>,
301}

303/// Configuration for opening or saving a [`Table`] to disk.
304///
305/// `TableOptions` bundles the filesystem path with the choice of storage
306/// manager and byte-ordering format. In C++ casacore this information is
307/// passed directly to the `Table(name, option)` constructor. Here it is
308/// factored out into a separate builder so that callers can construct
309/// options independently of the table itself.
310///
311/// # Example
312///
313/// ```rust
314/// use casa_tables::{TableOptions, DataManagerKind, EndianFormat};
315///
316/// let opts = TableOptions::new("/tmp/my_table")
317///     .with_data_manager(DataManagerKind::StandardStMan)
318///     .with_endian_format(EndianFormat::BigEndian);
319///
320/// // Use IncrementalStMan for slowly-changing columns:
321/// let ism_opts = TableOptions::new("/tmp/my_ism_table")
322///     .with_data_manager(DataManagerKind::IncrementalStMan);
323/// ```
324#[derive(Debug, Clone, PartialEq, Eq)]
325pub struct TableOptions {
326    path: PathBuf,
327    data_manager: DataManagerKind,
328    data_manager_group: Option<String>,
329    casacore_default_ssm: bool,
330    endian_format: EndianFormat,
331    tile_shape: Option<Vec<usize>>,
332}
333
334impl TableOptions {
335    /// Creates options targeting `path` with the default data manager
336    /// ([`DataManagerKind::StManAipsIO`]) and default endian format
337    /// ([`EndianFormat::LocalEndian`]).
338    pub fn new(path: impl AsRef<Path>) -> Self {
339        Self {
340            path: path.as_ref().to_path_buf(),
341            data_manager: DataManagerKind::default(),
342            data_manager_group: None,
343            casacore_default_ssm: false,
344            endian_format: EndianFormat::default(),
345            tile_shape: None,
346        }
347    }
348
349    /// Targets `path` instead, keeping every other option, returning the
350    /// updated options.
351    pub fn with_path(mut self, path: impl AsRef<Path>) -> Self {
352        self.path = path.as_ref().to_path_buf();
353        self
354    }
355
356    /// Overrides the data manager, returning the updated options.
357    pub fn with_data_manager(mut self, kind: DataManagerKind) -> Self {
358        self.data_manager = kind;
359        self
360    }
361
362    /// Overrides the data-manager group name persisted in column descriptors.
363    ///
364    /// Most callers should use the default group matching the manager type.
365    /// This exists for interoperability surfaces, such as casacore
366    /// `TableLogSink`, that use a conventional group name (`"SSM"`).
367    pub fn with_data_manager_group(mut self, group: impl Into<String>) -> Self {
368        self.data_manager_group = Some(group.into());
369        self
370    }
371
372    /// Uses casacore's zero-argument `StandardStMan` 32-row bucket sizing.
373    pub fn with_casacore_default_ssm(mut self, enabled: bool) -> Self {
374        self.casacore_default_ssm = enabled;
375        self
376    }
377
378    /// Overrides the endian format, returning the updated options.
379    ///
380    /// The endian format controls the byte ordering of multi-byte values
381    /// in the table's data files. It is only used when saving; on open the
382    /// format is detected from the existing `table.dat` marker.
383    pub fn with_endian_format(mut self, format: EndianFormat) -> Self {
384        self.endian_format = format;
385        self
386    }
387
388    /// Overrides the tile shape for tiled storage managers, returning the
389    /// updated options.
390    ///
391    /// The tile shape must include the row dimension as the last element
392    /// for `TiledColumnStMan` and `TiledShapeStMan`. For `TiledCellStMan`,
393    /// the tile shape covers only the cell dimensions.
394    ///
395    /// If not set, a reasonable default is chosen automatically.
396    pub fn with_tile_shape(mut self, shape: Vec<usize>) -> Self {
397        self.tile_shape = Some(shape);
398        self
399    }
400
401    /// Returns the filesystem path for this table.
402    pub fn path(&self) -> &Path {
403        &self.path
404    }
405
406    /// Returns the data manager that will be used when saving.
407    pub fn data_manager(&self) -> DataManagerKind {
408        self.data_manager
409    }
410
411    /// Returns whether casacore's zero-argument `StandardStMan` layout is used.
412    pub fn casacore_default_ssm(&self) -> bool {
413        self.casacore_default_ssm
414    }
415
416    /// Returns the endian format that will be used when saving.
417    pub fn endian_format(&self) -> EndianFormat {
418        self.endian_format
419    }
420
421    /// Returns the tile shape, if set.
422    pub fn tile_shape(&self) -> Option<&[usize]> {
423        self.tile_shape.as_deref()
424    }
425}

427/// Result of a TaQL SELECT query via [`Table::query_result()`].
428///
429/// Simple SELECTs (only column references) produce a zero-copy
430/// [`View`](QueryResult::View). SELECTs with computed expressions, GROUP BY,
431/// or aggregate functions produce a [`Materialized`](QueryResult::Materialized)
432/// in-memory table containing the evaluated result rows.
433///
434/// Both variants support reading column names and row data.
435///
436/// C++ equivalent: the result of `tableCommand()` — either a `Table` reference
437/// or a newly materialized `Table` from `makeProjectExprTable()`.
438pub enum QueryResult<'a> {
439    /// A zero-copy view into the source table (row-index mapping).
440    View(crate::RefTable<'a>),
441    /// An owned in-memory table with computed/aggregated values (boxed to
442    /// keep the enum size small).
443    Materialized(Box<Table>),
444}
445
446impl std::fmt::Debug for QueryResult<'_> {
447    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
448        match self {
449            Self::View(v) => write!(f, "QueryResult::View({} rows)", v.row_count()),
450            Self::Materialized(t) => write!(f, "QueryResult::Materialized({} rows)", t.row_count()),
451        }
452    }
453}
454
455impl QueryResult<'_> {
456    /// Returns the number of result rows.
457    pub fn row_count(&self) -> usize {
458        match self {
459            QueryResult::View(v) => v.row_count(),
460            QueryResult::Materialized(t) => t.row_count(),
461        }
462    }
463
464    /// Returns the column names of the result.
465    pub fn column_names(&self) -> Vec<String> {
466        match self {
467            QueryResult::View(v) => v.column_names().to_vec(),
468            QueryResult::Materialized(t) => t
469                .schema()
470                .map(|s| s.columns().iter().map(|c| c.name().to_string()).collect())
471                .unwrap_or_default(),
472        }
473    }
474
475    /// Returns the row at the given index.
476    pub fn row(&self, index: usize) -> Result<&RecordValue, TableError> {
477        match self {
478            QueryResult::View(v) => v.row(index),
479            QueryResult::Materialized(t) => t.row(index),
480        }
481    }
482}

484/// Errors that can occur when reading or writing a [`Table`].

494    },

625        /// Diagnostic for the invalid run start or extent.

627    },

629    #[error("invalid generated array runs: {message}")]

631        /// Diagnostic for the invalid run start, extent, or cell data.

633    },

836/// Internal lock state held by a [`Table`] when opened with locking.
837///
838/// Stores the open file descriptor (via [`LockFile`]), synchronization
839/// counters, and the options needed to re-save and re-load the table on
840/// lock transitions.
841#[cfg(unix)]
842struct LockState {
843    path: PathBuf,
844    lock_file: LockFile,
845    /// The published generation reflected in this handle, updated by
846    /// explicit lock transitions and successful automatic full saves.
847    observed_modify_counter: std::sync::atomic::AtomicU32,
848    options: LockOptions,
849    data_manager: DataManagerKind,
850    endian_format: EndianFormat,
851    /// The table's change count when its state last matched the disk (open,
852    /// reload, flush or a full save to its own directory). Releasing a write
853    /// lock writes the table only when the count has moved since, as
854    /// casacore's `PlainTable::putFile` writes only what changed. It is
855    /// atomic because `Table::save` writes through a shared borrow.
856    flushed_generation: std::sync::atomic::AtomicU64,
857    /// Whether this handle has written the table on disk, or started to,
858    /// under the write lock it holds, and has not yet published that write
859    /// in the sync data. Releasing the lock publishes the write when this is
860    /// set, and only a successful publication clears it.
861    ///
862    /// It is set before a write starts, so a write that fails part way is
863    /// still announced. A reload never clears it: a [`Table::resync`] makes
864    /// the handle match the disk again, but other processes have not been
865    /// told that the disk changed. It is atomic because `Table::flush` and
866    /// `Table::save` write through a shared borrow.
867    unpublished_write: std::sync::atomic::AtomicBool,
868    /// `table.dat` as it was before the unpublished write started, read when
869    /// `unpublished_write` is set and dropped once the write is published.
870    /// The publication compares the file with it (#880): a write that left
871    /// the file as it was keeps the row count published before.
872    table_dat_before_write: std::sync::OnceLock<Vec<u8>>,
873}
874
875#[cfg(unix)]
876impl LockState {
877    fn observed_modify_counter(&self) -> u32 {
878        self.observed_modify_counter
879            .load(std::sync::atomic::Ordering::Relaxed)
880    }
881
882    fn observe_modify_counter(&self, counter: u32) {
883        self.observed_modify_counter
884            .store(counter, std::sync::atomic::Ordering::Relaxed);
885    }
886
887    /// Whether the table changed since its state last matched the disk.
888    fn has_unflushed_changes(&self, table: &TableImpl) -> bool {
889        table.has_dirty_changes().unwrap_or_else(|| {
890            table.generation()
891                != self
892                    .flushed_generation
893                    .load(std::sync::atomic::Ordering::Relaxed)
894        })
895    }
896
897    /// Whether this lock period wrote the table on disk without yet
898    /// publishing the write.
899    fn has_unpublished_write(&self) -> bool {
900        self.unpublished_write
901            .load(std::sync::atomic::Ordering::Relaxed)
902    }
903}

907    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
908        f.debug_struct("LockState")
909            .field("path", &self.path)
910            .field("mode", &self.options.mode)
911            .finish_non_exhaustive()
912    }

915/// A casacore table — the fundamental persistent data container.
916///
917/// A `Table` holds a set of named columns (scalar or array) plus keyword
918/// metadata, backed either by on-disk storage or an in-memory buffer.
919///
920/// # Lifecycle
921///
922/// * **Open** — [`Table::open`] reads an existing table directory.
923/// * **Create** — [`Table::with_schema`] builds a new table from a schema.
924/// * **Save** — [`Table::save`] flushes deferred writes to disk.
925///
926/// # C++ equivalent
927///
928/// `casacore::Table` — the main user-facing class in the Tables module.
929#[derive(Default)]
930pub struct Table {
931    inner: TableImpl,
932    /// Filesystem path this table was last opened from or saved to.
933    source_path: Option<PathBuf>,
934    /// Whether this is a plain (disk-backed) or memory (transient) table.
935    kind: TableKind,
936    /// Names of columns backed by virtual engines (ForwardColumnEngine,
937    /// ScaledArrayEngine, etc.). Empty for tables with no virtual columns.
938    virtual_columns: HashSet<String>,
939    /// Virtual column bindings for save — describes how each virtual column
940    /// maps to its engine and configuration.
941    virtual_bindings: Vec<VirtualColumnBinding>,
942    /// Table metadata (type/subtype) persisted in `table.info`.
943    table_info: TableInfo,
944    /// Data manager info extracted from table.dat (empty for memory tables).
945    dm_info: Vec<crate::storage::DataManagerInfo>,
946    /// Optional external lock synchronization hook.
947    external_sync: Option<Box<dyn crate::lock::ExternalLockSync>>,
948    /// Explicit scientific-data provider used by TaQL `meas.*` functions.
949    measures: Option<Arc<dyn casa_core::measures::MeasuresProvider>>,
950    /// When `true`, the table directory is deleted on [`Drop`].
951    ///
952    /// Set via [`mark_for_delete`](Table::mark_for_delete), cleared via
953    /// [`unmark_for_delete`](Table::unmark_for_delete).
954    marked_for_delete: bool,
955    #[cfg(unix)]
956    lock_state: Option<LockState>,
957}
958
959impl std::fmt::Debug for Table {
960    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
961        let mut s = f.debug_struct("Table");
962        s.field("inner", &self.inner)
963            .field("source_path", &self.source_path)
964            .field("kind", &self.kind)
965            .field("virtual_columns", &self.virtual_columns)
966            .field("table_info", &self.table_info)
967            .field("marked_for_delete", &self.marked_for_delete)
968            .field(
969                "external_sync",
970                &self.external_sync.as_ref().map(|_| "<hook>"),
971            );
972        #[cfg(unix)]
973        s.field("lock_state", &self.lock_state);
974        s.finish()
975    }
976}

988    /// Return the checked logical heap bytes retained by a freshly opened,
989    /// lazy table metadata capability.
990    ///
991    /// The projection covers schema and lookup storage, table and column
992    /// keywords, data-manager descriptors, paths, virtual-column names, table
993    /// info, and retained lock state. It returns `None` if row/cell caches,
994    /// pending mutations, virtual bindings, or opaque external providers are
995    /// present, so bounded readers cannot silently admit an unmodeled owner.
996    /// Collection allocator headers and hash-table control bytes remain outside
997    /// this logical payload projection.
998    #[doc(hidden)]

1000        self.retained_table_metadata_bytes(false)
1001    }
1002
1003    /// Logical owned heap for a metadata writer, including explicit eager rows.
1004    /// Returns `None` for unmodeled lazy cell caches or pending column writes.
1005    /// Uses the same collection-capacity projection as the read-only ledger.
1006    #[doc(hidden)]
1007    pub fn retained_owned_metadata_bytes(&self) -> Option<usize> {
1008        self.retained_table_metadata_bytes(true)
1009    }
1010
1011    fn retained_table_metadata_bytes(&self, eager: bool) -> Option<usize> {
1012        if !self.virtual_bindings.is_empty()
1013            || self.external_sync.is_some()
1014            || self.measures.is_some()
1015        {
1016            return None;
1017        }
1018        let mut bytes = if eager {
1019            self.inner.retained_owned_metadata_heap_bytes()?
1020        } else {
1021            self.inner.retained_lazy_metadata_heap_bytes()?
1022        }
1023        .checked_add(
1024            self.virtual_columns
1025                .capacity()
1026                .checked_mul(size_of::<String>())?,
1027        )?
1028        .checked_add(
1029            self.virtual_bindings
1030                .capacity()
1031                .checked_mul(size_of::<VirtualColumnBinding>())?,
1032        )?
1033        .checked_add(self.table_info.table_type.capacity())?
1034        .checked_add(self.table_info.sub_type.capacity())?
1035        .checked_add(
1036            self.table_info
1037                .readme
1038                .capacity()
1039                .checked_mul(size_of::<String>())?,
1040        )?
1041        .checked_add(
1042            self.dm_info
1043                .capacity()
1044                .checked_mul(size_of::<crate::storage::DataManagerInfo>())?,
1045        )?;
1046        if let Some(path) = &self.source_path {
1047            bytes = bytes.checked_add(crate::table_impl::path_heap_bytes(path))?;
1048        }
1049        for column in &self.virtual_columns {
1050            bytes = bytes.checked_add(column.capacity())?;
1051        }
1052        for line in &self.table_info.readme {
1053            bytes = bytes.checked_add(line.capacity())?;
1054        }
1055        for manager in &self.dm_info {
1056            bytes = bytes.checked_add(manager.dm_type.capacity())?.checked_add(
1057                manager
1058                    .columns
1059                    .capacity()
1060                    .checked_mul(size_of::<String>())?,
1061            )?;
1062            for column in &manager.columns {
1063                bytes = bytes.checked_add(column.capacity())?;
1064            }
1065        }
1066        #[cfg(unix)]
1067        if let Some(lock) = &self.lock_state {
1068            bytes = bytes
1069                .checked_add(crate::table_impl::path_heap_bytes(&lock.path))?
1070                .checked_add(lock.lock_file.retained_heap_bytes())?;
1071        }
1072        Some(bytes)
1073    }

1312// ── Drop ──────────────────────────────────────────────────────────────
1313
1314impl Drop for Table {
1315    fn drop(&mut self) {
1316        #[cfg(unix)]
1317        {
1318            if self.kind != TableKind::Memory || self.is_named_memory_manager() {
1319                // A write lock that changed nothing writes and publishes
1320                // nothing, as in casacore.
1321                let has_changes_to_flush = self.lock_state.as_ref().is_some_and(|state| {
1322                    state.lock_file.has_lock(LockType::Write)
1323                        && state.has_unflushed_changes(&self.inner)
1324                });
1325
1326                if has_changes_to_flush && let Err(error) = self.flush_locked_changes(false) {
1327                    tracing::warn!(%error, "discarding unwritten changes while dropping locked table");
1328                }
1329
1330                // Whatever part of this save, or of an earlier write under
1331                // the lock, reached the disk is published, describing the
1332                // table as persisted.
1333                if let Some(state) = self.lock_state.as_ref()
1334                    && state.lock_file.has_lock(LockType::Write)
1335                    && state.has_unpublished_write()
1336                {
1337                    let _ = publish_persisted_write(
1338                        &state.lock_file,
1339                        &state.path,
1340                        state.table_dat_before_write.get().map(Vec::as_slice),
1341                    );
1342                }
1343            }
1344        }
1345
1346        if self.marked_for_delete
1347            && let Some(path) = &self.source_path
1348        {
1349            let _ = std::fs::remove_dir_all(path);
1350        }
1351    }
1352}
