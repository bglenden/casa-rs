2//! CASA ARRAY_ID predicates and their ordered output ID list.
3
4use super::syntax::{canonical_numeric_ranges, row_id_interval, trim_row_id_space};
5use crate::{MsError, MsResult};
6
7/// CASA ARRAY_ID ranges and their output reindexing order.
8pub struct ArraySelection {
9    pub(super) ranges: Vec<(i32, i32)>,
10    // Store the first index of each interval. The last matching interval
11    // supplies the output index when an ID appears more than once.
12    indices: Vec<(i64, i64, i64)>,
13}
14
15impl ArraySelection {
16    /// Compile the array-ID expression and its output indexing order.
17    pub fn parse(text: &str) -> MsResult<Self> {
18        const MAX_ARRAY: i64 = 1_000;
19        let mut ranges = Vec::new();
20        let mut index_ranges = Vec::new();
21        let mut singles = Vec::new();
22        for term in text.split(',').map(trim_row_id_space) {
23            let (first, last) = row_id_interval(term).ok_or_else(|| {
24                MsError::InvalidInput(format!(
25                    "Array Expression: invalid array selector term {term:?}"
26                ))
27            })?;
28            if first <= last {
29                ranges.push((first as i32, last as i32));
30            }
31            // MSArrayGram adds bounded intervals immediately and appends
32            // the individual IDs at the end of the entire expression.
33            if term.starts_with('<') {
34                index_ranges.push((0, last));
35            } else if term.starts_with('>') && !term.contains('&') {
36                let end = MAX_ARRAY + i64::from(!term.starts_with(">="));
37                if first > end + 1 {
38                    return Err(MsError::InvalidInput(format!(
39                        "Array Expression: invalid array bound {term:?}"
40                    )));
41                }
42                index_ranges.push((first, end));
43            } else if term.contains(['~', '&']) {
44                index_ranges.push((first, last));
45            } else {
46                singles.push((first, last));
47            }
48        }
49        let mut offset = 0;
50        let indices = index_ranges
51            .into_iter()
52            .chain(singles)
53            .filter(|(first, last)| first <= last)
54            .map(|(first, last)| {
55                let start = offset;
56                offset += last - first + 1;
57                (first, last, start)
58            })
59            .collect();
60        Ok(Self {
61            ranges: canonical_numeric_ranges(ranges),
62            indices,
63        })
64    }
65
66    /// Map an input array ID to its output ID after CASA range expansion.
67    pub fn reindex(&self, id: i32) -> MsResult<i32> {
68        let id = i64::from(id);
69        let output = self
70            .indices
71            .iter()
72            .rev()
73            .find_map(|&(first, last, offset)| {
74                (first..=last).contains(&id).then_some(offset + id - first)
75            });
76        // An absent key in CASA's nonempty std::map acquires the default
77        // output ID 0. With an empty map it copies the original column.
78        let output = output.unwrap_or(if self.indices.is_empty() { id } else { 0 });
79        i32::try_from(output).map_err(|_| {
80            MsError::InvalidInput(format!(
81                "Array Expression: output ARRAY_ID {output} is out of range"
82            ))
83        })
84    }
85}
