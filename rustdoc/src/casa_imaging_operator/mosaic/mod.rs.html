2//! The mosaic primary-beam kernel set, pinned to CASA `HetArrayConvFunc`
3//! (`TransformMachines2/HetArrayConvFunc.cc`), `SimplePBConvFunc`,
4//! `PBMath1D`/`PBMath1DAiry`/`PBMath1DEVLA`, `MosaicFT` and `fmosaic.f`.
5//!
6//! One dense complex cell per (dish-class pair, spectral window, frequency
7//! cell): `FT[VP₁ · VP₂]` for imaging and `FT[PB₁ · PB₂]` for the weight
8//! image, both built on a screen covering the image field at the
9//! convolution-function size, quarter-cropped, support-searched on the
10//! weight plane, normalised over the support, cropped again and Lanczos
11//! resampled to ten fine offsets per cell. Every pair of a window grids
12//! with the widest pair's support: `MosaicFT` hands `sectgmosd2` and the
13//! degridders `max(convSupportPlanes_p)` for every row, though each pair
14//! is normalised over its own. The pointing enters only through the per-row
15//! phase gradient ([`ConvolutionFunctionSet::pointing_ramp`]).
16//!
17//! A dish class's voltage pattern is a [`PrimaryBeam`]: the annular Airy
18//! aperture `findAntennaSizes` builds per dish diameter ([`AiryDish`]), or
19//! CASA's common EVLA beam, which `findAntennaSizes` gives every antenna of
20//! an EVLA MeasurementSet (`PBMathInterface::COMMONPB`) and which takes the
21//! polynomial fit of each beam frequency (`PBMath1DEVLA::nearestVPArray`).
22

46/// `MosaicFT::findConvFunction`: fine offsets per cell for
47/// `HetArrayConvFunc` (`mosaic.oversampling`, default 10, forced to at
48/// least 10). The image-size rule `ceil(5000/max(nx, ny))` applies only to
49/// `SimplePBConvFunc`.
50pub const MOSAIC_OVERSAMPLING: u16 = 10;
51/// `PBMath1DAiry` in `findAntennaSizes`: the reference frequency of the
52/// dish-class support radius, in GHz.
53const DISH_REFERENCE_GHZ: f64 = 100.0;

55/// Which of CASA's two heterogeneous-array mosaic machines a set follows.
56/// They build the same crops, normalisation, resampling and gridding
57/// support, and differ in the screen's least size, in the frequency cells,
58/// in how the support is found and, through [`AiryDish`], in the 7 m dish's
59/// radius.
60#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
61pub enum MosaicRules {
62    /// tclean's `gridder='mosaic'`: `TransformMachines2/HetArrayConvFunc`
63    /// (imaging).
64    #[default]
65    Tclean,
66    /// tclean's machine with `conjbeams` on an image of one plane
67    /// (`findConvFunction` with `getConjConvFunc`, `nchan_p == 1`): each
68    /// beam frequency `ν` has a conjugate `ν_c = √(2ν₀² − ν²)` about the
69    /// image frequency `ν₀` (`SynthesisUtils::conjFreq`). The weight cells
70    /// are `FT[VP_k(ν) VP_j(ν) VP_k(ν_c) VP_j(ν_c)]`; a window's supports
71    /// widen to `√(max ν_c/ν)` times its widest when that is wider
72    /// (`conjSupport`); and gridding takes the imaging cell resampled by
73    /// `ν_c/ν` (`fillConjConvFunc`), while prediction keeps the cell at `ν`
74    /// (`MosaicFT` asks for the conjugate only when it grids).
75    TcleanConjugateBeams,
76    /// The simulator's `setoptions(ftmachine='mosaic')`: the earlier
77    /// `TransformMachines/HetArrayConvFunc`, which `Simulator::
78    /// createSkyEquation` builds. Its screen has no least size beside the
79    /// beams'; its frequency cells come from the selected channels alone;
80    /// its support search walks the weight plane inwards from the edge,
81    /// along the row and the column one cell off its centre, and widens the
82    /// radius it finds by √2.
83    Simulator,
84}
85
86impl MosaicRules {
87    /// The conjugate of beam frequency `frequency_hz` about the image
88    /// frequency `image_frequency_hz` under these rules, if they have one.
89    fn conjugate_hz(
90        self,
91        image_frequency_hz: f64,
92        frequency_hz: f64,
93    ) -> Result<Option<f64>, OperatorError> {
94        if self != Self::TcleanConjugateBeams {
95            return Ok(None);
96        }
97        let conjugate =
98            (2.0 * image_frequency_hz * image_frequency_hz - frequency_hz * frequency_hz).sqrt();
99        if conjugate.is_finite() && conjugate > 0.0 {
100            Ok(Some(conjugate))
101        } else {
102            Err(OperatorError::ConvolutionFunction {
103                reason: "a beam frequency above √2 times the image frequency has no conjugate",
104            })
105        }
106    }
107}

109/// One dish class of the heterogeneous array: the annular Airy aperture
110/// `HetArrayConvFunc::findAntennaSizes` builds per distinct dish diameter.
111#[derive(Clone, Copy, Debug, PartialEq)]
112pub struct AiryDish {
113    /// Effective aperture diameter in metres.
114    pub aperture_m: f64,
115    /// Central blockage diameter in metres.
116    pub blockage_m: f64,
117    /// Support radius of the voltage pattern in arcseconds at 100 GHz.
118    pub max_radius_arcsec: f64,
119}
120
121impl AiryDish {
122    /// `HetArrayConvFunc::findAntennaSizes` for an ALMA or ACA
123    /// MeasurementSet: a dish within 0.5 m of 12 m is the 10.7 m aperture
124    /// with 0.75 m blockage and radius `max(150″, fov/5)`; any other dish
125    /// is the 6.25 m aperture with 0.75 m blockage and radius `max(300″,
126    /// fov/3)`, where `fov = max(nx · Δx, ny · Δy)` with CASA's signed
127    /// increments.
128    #[must_use]
129    pub fn casa_alma(dish_diameter_m: f64, geometry: &GridGeometry) -> Self {
130        let image = geometry.image();
131        let fov_rad = (image.shape[0] as f64 * image.increment_rad[0])
132            .max(image.shape[1] as f64 * image.increment_rad[1]);
133        let fov_arcsec = fov_rad.to_degrees() * 3600.0;
134        if (dish_diameter_m - 12.0).abs() < 0.5 {
135            Self {
136                aperture_m: 10.7,
137                blockage_m: 0.75,
138                max_radius_arcsec: 150.0_f64.max(fov_arcsec / 5.0),
139            }
140        } else {
141            Self {
142                aperture_m: 6.25,
143                blockage_m: 0.75,
144                max_radius_arcsec: 300.0_f64.max(fov_arcsec / 3.0),
145            }
146        }
147    }
148
149    /// The simulator's `HetArrayConvFunc::findAntennaSizes`
150    /// ([`MosaicRules::Simulator`]) for an ALMA or ACA MeasurementSet: the
151    /// apertures of [`Self::casa_alma`], with a 150″ radius at 100 GHz for
152    /// every dish.
153    #[must_use]
154    pub fn casa_alma_simulator(dish_diameter_m: f64) -> Self {
155        let (aperture_m, blockage_m) = if (dish_diameter_m - 12.0).abs() < 0.5 {
156            (10.7, 0.75)
157        } else {
158            (6.25, 0.75)
159        };
160        Self {
161            aperture_m,
162            blockage_m,
163            max_radius_arcsec: 150.0,
164        }
165    }
166
167    /// `HetArrayConvFunc::findAntennaSizes` for any other telescope, in
168    /// either machine: the dish diameter itself, ALMA's blockage ratio
169    /// `d/12 · 0.75` and a 150″ radius.
170    #[must_use]
171    pub fn casa_airy(dish_diameter_m: f64) -> Self {
172        Self {
173            aperture_m: dish_diameter_m,
174            blockage_m: dish_diameter_m / 12.0 * 0.75,
175            max_radius_arcsec: 150.0,
176        }
177    }
178}

232/// What sizes a mosaic set before it is built
233/// (`HetArrayConvFunc::findConvFunction`): the side of the screen, which
234/// the widest dish class sets, and every window's frequency cells.
235struct MosaicLayout {
236    conv_size: usize,
237    cells: Vec<FrequencyCells>,
238    cell_count: usize,
239}
240
241impl MosaicLayout {
242    fn new(
243        rules: MosaicRules,
244        geometry: &GridGeometry,
245        image_frequency_hz: f64,
246        dishes: &[PrimaryBeam],
247        windows: &[MosaicWindow],
248    ) -> Result<Self, OperatorError> {
249        if dishes.is_empty() || dishes.len() > usize::from(u8::MAX) {
250            return Err(OperatorError::ConvolutionFunction {
251                reason: "the mosaic set needs between one and 255 dish classes",
252            });
253        }
254        if windows.is_empty() {
255            return Err(OperatorError::ConvolutionFunction {
256                reason: "the mosaic set needs at least one spectral window",
257            });
258        }
259        if !image_frequency_hz.is_finite() || image_frequency_hz <= 0.0 {
260            return Err(OperatorError::ConvolutionFunction {
261                reason: "the mosaic set needs a positive finite image frequency",
262            });
263        }
264        let image = geometry.image();
265        let [nx, ny] = image.shape;
266        let increment = image.increment_rad;
267        // `findConvFunction`: the screen size from the widest dish class,
268        // at least a tenth of the image in tclean's machine.
269        let mut support = match rules {
270            MosaicRules::Tclean | MosaicRules::TcleanConjugateBeams => nx.max(ny) / 10,
271            MosaicRules::Simulator => 0,
272        };
273        for dish in dishes {
274            support = support.max(dish.support_pixels(increment[0], image_frequency_hz));
275        }
276        support = support.min(nx.max(ny));
277        // `HetArrayConvFunc::findConvFunction`: the support limited by the
278        // image extent, `CompositeNumber::nextLargerEven` (the first even
279        // composite strictly above its argument), then `/16*16`.
280        // `SimplePBConvFunc` sizes its screen from
281        // `Int(max(nx,ny)/2)*2*convSamp` instead; this set follows the
282        // heterogeneous-array class whose cells it builds.
283        let conv_size = next_larger_even_composite(support + 1) / 16 * 16;
284        if conv_size < 16 {
285            return Err(OperatorError::ConvolutionFunction {
286                reason: "the mosaic screen is smaller than 16 pixels",
287            });
288        }
289        let mut cells = Vec::with_capacity(windows.len());
290        let mut cell_count = 0;
291        for window in windows {
292            let frequency_cells = FrequencyCells::new(rules, window, cell_count)?;
293            cell_count += frequency_cells.frequencies_hz.len();
294            cells.push(frequency_cells);
295        }
296        if cell_count > usize::from(u16::MAX) {
297            return Err(OperatorError::ConvolutionFunction {
298                reason: "the mosaic set has more than 65535 frequency cells",
299            });
300        }
301        Ok(Self {
302            conv_size,
303            cells,
304            cell_count,
305        })
306    }
307}

310    /// Build the cells under `rules` for `geometry`'s image (which is the
311    /// grid: the mosaic set grids without padding), `dishes` in CASA's
312    /// dish-class order and the selected `windows`: [`Self::screens`], then
313    /// [`MosaicScreens::finish`]. `image_frequency_hz` is the image's
314    /// frequency at spectral pixel 0, which sizes the screen
315    /// (`PBMath1D::support`).
316    ///
317    /// # Errors
318    ///
319    /// As those two stages.
320    pub fn new(
321        rules: MosaicRules,
322        geometry: &GridGeometry,
323        polarization: &PolarizationRouting,
324        image_frequency_hz: f64,
325        dishes: &[PrimaryBeam],
326        windows: &[MosaicWindow],
327    ) -> Result<Self, OperatorError> {
328        Self::screens(
329            rules,
330            geometry,
331            polarization,
332            image_frequency_hz,
333            dishes,
334            windows,
335        )?
336        .finish()
337    }

339    /// Bytes [`Self::screens`] holds at most for the same arguments: each
340    /// dish class's voltage table (an Airy pattern's), the screen every lattice is transformed
341    /// on and its planning, one dish pair's beam products, and every pair's
342    /// lattices at every beam frequency.
343    ///
344    /// # Errors
345    ///
346    /// As [`Self::screens`] refuses the arguments, before it allocates.
347    pub fn screens_bytes(
348        rules: MosaicRules,
349        geometry: &GridGeometry,
350        image_frequency_hz: f64,
351        dishes: &[PrimaryBeam],
352        windows: &[MosaicWindow],
353    ) -> Result<u64, OperatorError> {
354        let layout = MosaicLayout::new(rules, geometry, image_frequency_hz, dishes, windows)?;
355        let conv = layout.conv_size;
356        let lattice = conv / 4;
357        let pairs = pair_planes(dishes.len());
358        let lattices = pairs
359            * (layout.cell_count
360                * (2 * lattice * lattice * size_of::<Complex32>() + size_of::<PairLattices>())
361                + layout.cells.len()
362                    * (size_of::<(usize, usize, Vec<PairLattices>)>() + size_of::<usize>()));
363        let products = 2 * conv * (conv * size_of::<Complex32>() + size_of::<f32>());
364        let tables = dishes.len() * (BeamPattern::BYTES as usize + size_of::<(BeamPattern, f64)>());
365        Ok(
366            (tables + conv * conv * size_of::<Complex64>() + products + lattices) as u64
367                + frequency_cell_bytes(&layout.cells)
368                + casa_fft::planning_bytes::<f64>(conv * conv)
369                + MuellerRouting::MAXIMUM_BYTES,
370        )
371    }
372
373    /// The first stage of building the cells: every dish pair's beam
374    /// products transformed on the screen at every beam frequency, cropped
375    /// to their lattices, with each window's support, the screen sized and
376    /// the supports found as `rules`' machine does.
377    /// [`MosaicScreens::kernel_bytes`] then says what the cells cropped from
378    /// them hold before [`MosaicScreens::finish`] builds them.
379    ///
380    /// # Errors
381    ///
382    /// No dish or window, a window without positive finite frequencies, a
383    /// screen too small to hold a cell, a failed FFT plan or a zero
384    /// support.
385    pub fn screens(
386        rules: MosaicRules,
387        geometry: &GridGeometry,
388        polarization: &PolarizationRouting,
389        image_frequency_hz: f64,
390        dishes: &[PrimaryBeam],
391        windows: &[MosaicWindow],
392    ) -> Result<MosaicScreens, OperatorError> {
393        let MosaicLayout {
394            conv_size,
395            cells,
396            cell_count,
397        } = MosaicLayout::new(rules, geometry, image_frequency_hz, dishes, windows)?;
398        let image = geometry.image();
399        let [nx, ny] = image.shape;
400        let increment = image.increment_rad;
401        let lattice = conv_size / 4;
402        // The screen covers the image field at `conv_size` pixels, so its
403        // increment is `Δ · n / convSize` per axis, in degrees as
404        // `PBMath1D::apply` works.
405        let screen_increment_deg = [
406            (increment[0] * nx as f64 / conv_size as f64).to_degrees(),
407            (increment[1] * ny as f64 / conv_size as f64).to_degrees(),
408        ];
409        let patterns = dishes
410            .iter()
411            .map(|dish| (dish.pattern(), dish.maximum_radius_arcmin_ghz()))
412            .collect::<Vec<_>>();
413        let mut fft = PlaneFft::<f64>::new([conv_size, conv_size], false)?;
414        let mut screen = vec![Complex64::default(); conv_size * conv_size];
415        let planes = pair_planes(dishes.len());
416        // One lattice per beam frequency of every pair and window, in
417        // CASA's `[x][y]` quarter-cropped layout, with the pair's support
418        // (`supportAndNormalizeLatt`: from the weight plane of the last beam
419        // frequency), in the cell order the keys index.
420        let mut screened = Vec::with_capacity(planes * cells.len());
421        let mut window_support = vec![0_usize; cells.len()];
422        for k in 0..dishes.len() {
423            for j in k..dishes.len() {
424                for (window_index, window) in cells.iter().enumerate() {
425                    let mut lattices = Vec::with_capacity(window.frequencies_hz.len());
426                    for &frequency_hz in &window.frequencies_hz {
427                        lattices.push(screen_pair(
428                            &mut fft,
429                            &mut screen,
430                            conv_size,
431                            screen_increment_deg,
432                            [&patterns[k], &patterns[j]],
433                            frequency_hz,
434                            rules.conjugate_hz(image_frequency_hz, frequency_hz)?,
435                        )?);
436                    }
437                    let last = lattices.last().expect("a window has a cell");
438                    let support = match rules {
439                        MosaicRules::Tclean | MosaicRules::TcleanConjugateBeams => {
440                            weight_plane_support(&last.weight, lattice)?
441                        }
442                        MosaicRules::Simulator => {
443                            simulator_weight_plane_support(&last.weight, lattice)?
444                        }
445                    };
446                    window_support[window_index] = window_support[window_index].max(support);
447                    screened.push((window_index, support, lattices));
448                }
449            }
450        }
451        // `conjSupport`: every pair of a window takes the support widened
452        // by the root of the window's largest `ν_c/ν`, when that is wider.
453        let mut ratios = Vec::new();
454        for (window, support) in cells.iter().zip(&mut window_support) {
455            let mut largest = f64::NEG_INFINITY;
456            for &frequency_hz in &window.frequencies_hz {
457                if let Some(conjugate) = rules.conjugate_hz(image_frequency_hz, frequency_hz)? {
458                    ratios.push(conjugate / frequency_hz);
459                    largest = largest.max(conjugate / frequency_hz);
460                }
461            }
462            let conjugate = (*support as f64 * largest.sqrt() / 2.0) as usize * 2;
463            if largest.is_finite() && conjugate > *support {
464                *support = conjugate;
465            }
466        }
467        Ok(MosaicScreens {
468            dishes: dishes.len(),
469            windows: cells,
470            cell_count,
471            screened,
472            window_support,
473            ratios,
474            lattice,
475            conv_size,
476            grid: geometry.grid_shape(),
477            mueller: MuellerRouting::scalar(polarization.pol_map(), polarization.grid_pols()),
478        })
479    }

481    /// Bytes the set holds: its imaging, weight and conjugate cells and its
482    /// corrections.
483    #[must_use]
484    pub fn resident_bytes(&self) -> u64 {
485        self.imaging
486            .iter()
487            .chain(&self.weight)
488            .chain(&self.conjugate)
489            .map(|cell| cell.bytes() as u64)
490            .sum::<u64>()
491            + ((self.imaging.capacity() + self.weight.capacity() + self.conjugate.capacity())
492                * size_of::<DenseCell>()) as u64
493            + frequency_cell_bytes(&self.windows)
494            + self.mueller.bytes()
495            + self.correction.resident_bytes()
496    }
497}

499/// Bytes the windows' frequency cells hold, from their capacities.
500fn frequency_cell_bytes(windows: &Vec<FrequencyCells>) -> u64 {
501    (windows.capacity() * size_of::<FrequencyCells>()
502        + windows
503            .iter()
504            .map(|window| window.frequencies_hz.capacity() * size_of::<f64>())
505            .sum::<usize>()) as u64
506}

508/// A mosaic set's beam products transformed on their screens and cropped

511#[derive(Debug)]
512pub struct MosaicScreens {
513    dishes: usize,
514    windows: Vec<FrequencyCells>,
515    cell_count: usize,
516    /// Per dish pair and window: the window, the pair's support and its
517    /// lattices at every beam frequency.
518    screened: Vec<(usize, usize, Vec<PairLattices>)>,
519    /// Per window: the widest pair's support, which `conjSupport` may have
520    /// widened; every pair of the window grids with it.
521    window_support: Vec<usize>,
522    /// Per frequency cell, in cell order: `ν_c/ν` under
523    /// [`MosaicRules::TcleanConjugateBeams`]; empty otherwise.

531impl MosaicScreens {
532    /// The support each dish pair finds on its weight plane, per pair (in
533    /// key order) and window: what CASA logs as "convolution function
534    /// support", before every pair of a window takes the widest
535    /// ([`MosaicPb::half_support`]).
536    #[must_use]
537    pub fn found_supports(&self) -> Vec<usize> {
538        self.screened
539            .iter()
540            .map(|(_, support, _)| *support)
541            .collect()
542    }
543
544    /// Bytes the screens hold: every pair's lattices and the supports.
545    #[must_use]
546    pub fn bytes(&self) -> u64 {
547        (self
548            .screened
549            .iter()
550            .map(|(_, _, lattices)| {
551                lattices.capacity() * size_of::<PairLattices>()
552                    + lattices
553                        .iter()
554                        .map(|pair| {
555                            (pair.imaging.capacity() + pair.weight.capacity())
556                                * size_of::<Complex32>()
557                        })
558                        .sum::<usize>()
559            })
560            .sum::<usize>()
561            + self.screened.capacity() * size_of::<(usize, usize, Vec<PairLattices>)>()
562            + self.window_support.capacity() * size_of::<usize>()
563            + self.ratios.capacity() * size_of::<f64>()) as u64
564            + frequency_cell_bytes(&self.windows)
565            + self.mueller.bytes()
566    }
567
568    /// Bytes [`Self::finish`] allocates beside the screens: the imaging,
569    /// weight and conjugate cells it cuts from them, which the set keeps
570    /// ([`MosaicPb::resident_bytes`]), one cell's crop and resampling (and
571    /// its conjugate's) at a time, and the corrections.
572    #[must_use]
573    pub fn kernel_bytes(&self) -> u64 {
574        let kinds = if self.ratios.is_empty() { 2 } else { 3 };
575        let cells = self
576            .screened
577            .iter()
578            .map(|(window_index, _, lattices)| {
579                let half =
580                    u16::try_from(self.window_support[*window_index]).expect("support fits u16");
581                kinds
582                    * lattices.len() as u64
583                    * dense_cell_bytes(1, MOSAIC_OVERSAMPLING, [half, half])
584            })
585            .sum::<u64>();
586        let crop = self
587            .window_support
588            .iter()
589            .map(|support| self.crop(*support))
590            .max()
591            .unwrap_or(0);
592        let resampled = crop * usize::from(MOSAIC_OVERSAMPLING);
593        let mut transients = crop * crop * (size_of::<Complex32>() + 2 * size_of::<f32>())
594            + resampled * resampled * size_of::<Complex32>();
595        if let Some(largest) = self.ratios.iter().copied().reduce(f64::max) {
596            // The imaging plane held while its conjugate is resampled from
597            // its parts and set in a plane of its own.
598            let scaled = conjugate_side(resampled, largest);
599            transients +=
600                resampled * resampled * (2 * size_of::<Complex32>() + 2 * size_of::<f32>())
601                    + scaled * scaled * size_of::<Complex32>();
602        }
603        let records =
604            kinds as usize * self.cell_count * pair_planes(self.dishes) * size_of::<DenseCell>();
605        cells + (transients + records) as u64 + ImageCorrection::bytes(self.grid)
606    }
607
608    /// The side every pair of a window of `support` is cropped to.
609    fn crop(&self, support: usize) -> usize {
610        (2 * (support + 2)).min(self.lattice)
611    }
612
613    /// The second stage of building the cells: crop every pair of a window
614    /// to the widest pair's `2(support + 2)`, normalise each over its own
615    /// support, resample, and form the corrections.
616    ///
617    /// # Errors
618    ///
619    /// A crop too small to resample, or a non-positive
620    /// convolution-function integral.
621    pub fn finish(self) -> Result<MosaicPb, OperatorError> {
622        let planes = pair_planes(self.dishes);
623        // `findConvFunction` crops every pair of a window to the widest
624        // pair's `2(support + 2)`; each pair is normalised over its own
625        // support. Both machines' `MosaicFT` grids every row with the
626        // widest (`convSupport = max(convSupportPlanes_p)`).
627        let mut imaging = Vec::with_capacity(planes * self.cell_count);
628        let mut weight = Vec::with_capacity(planes * self.cell_count);
629        let mut conjugate = Vec::with_capacity(if self.ratios.is_empty() {
630            0
631        } else {
632            planes * self.cell_count
633        });
634        let max_half_support = self.window_support.iter().copied().max().unwrap_or(0);
635        for (window_index, support, lattices) in &self.screened {
636            let taps = self.window_support[*window_index];
637            let crop = self.crop(taps);
638            // `interpLanczos` zeroes within three input pixels of the edge, so
639            // a crop under seven pixels resamples to nothing.
640            if crop < 2 * LANCZOS_A as usize + 1 {
641                return Err(OperatorError::ConvolutionFunction {
642                    reason: "the mosaic beam spans too few screen pixels; use a larger image or finer cells",
643                });
644            }
645            // The lattice bounds the crop; a support `conjSupport` widened
646            // past it has no taps to read.
647            if taps + 1 > crop / 2 {
648                return Err(OperatorError::ConvolutionFunction {
649                    reason: "the conjugate beams widen the mosaic support past its screen; use a larger image",
650                });
651            }
652            let first = self.windows[*window_index].first;
653            for (index, lattice_pair) in lattices.iter().enumerate() {
654                let cells = finish_pair(
655                    lattice_pair,
656                    self.lattice,
657                    (*support, crop, taps),
658                    self.ratios.get(first + index).copied(),
659                )?;
660                imaging.push(cells.imaging);
661                weight.push(cells.weight);
662                conjugate.extend(cells.conjugate);
663            }
664        }
665        let Self {
666            dishes,
667            windows: cells,
668            cell_count,
669            conv_size,
670            grid,
671            mueller,
672            ..
673        } = self;
674        let half_support = u16::try_from(max_half_support).expect("support fits u16");
675        let sinc = |len: usize| {
676            (0..len)
677                .map(|index| {
678                    // `MosaicFT::getImage`, `prepGridForDegrid`: Float
679                    // arithmetic, unity at the centre.
680                    let x = std::f32::consts::PI * (index as f32 - (len / 2) as f32)
681                        / (len as f32 * f32::from(MOSAIC_OVERSAMPLING));
682                    if index == len / 2 {
683                        1.0
684                    } else {
685                        f64::from(x.sin() / x)
686                    }
687                })
688                .collect::<Vec<_>>()
689        };
690        let [grid_nx, grid_ny] = grid;
691        let model_x = sinc(grid_nx);
692        let model_y = sinc(grid_ny);
693        let image_x = model_x.iter().map(|value| 1.0 / value).collect();
694        let image_y = model_y.iter().map(|value| 1.0 / value).collect();
695        Ok(MosaicPb {
696            dishes,
697            windows: cells,
698            cells_per_plane: cell_count,
699            imaging,
700            weight,
701            conjugate,
702            max_half_support: half_support,
703            conv_size,
704            mueller,
705            correction: ImageCorrection::split(image_x, image_y, model_x, model_y),
706        })
707    }
708}

710impl MosaicPb {
711    /// Number of dish classes.
712    #[must_use]
713    pub const fn dishes(&self) -> usize {
714        self.dishes
715    }
716
717    /// The screen size in pixels (`HetArrayConvFunc::convSize_p` before
718    /// the quarter crop).
719    #[must_use]
720    pub const fn screen_size(&self) -> usize {
721        self.conv_size
722    }
723
724    /// Beam frequencies of `spectral_window` in cell order, or `None` for a
725    /// window the set was not built for.
726    #[must_use]
727    pub fn frequency_cells(&self, spectral_window: u32) -> Option<&[f64]> {
728        self.window(spectral_window)
729            .map(|window| window.frequencies_hz.as_slice())
730    }
731
732    /// Half support of the cell `key` names.
733    #[must_use]
734    pub fn half_support(&self, key: CfKey) -> u16 {
735        self.imaging[self.index(key)].support()[0] / 2
736    }
737
738    fn window(&self, spectral_window: u32) -> Option<&FrequencyCells> {
739        self.windows
740            .iter()
741            .find(|window| window.spectral_window == spectral_window)
742    }
743
744    fn index(&self, key: CfKey) -> usize {
745        usize::from(key.group) * self.cells_per_plane + usize::from(key.cube)
746    }
747}
748
749/// `HetArrayConvFunc::findConvFunction`: dish-class pairs `(k, j)` with
750/// `k ≤ j`.
751fn pair_planes(dishes: usize) -> usize {
752    dishes * (dishes + 1) / 2
753}
754
755/// `HetArrayConvFunc::makerowmap`: the plane of the pair of dish classes
756/// `types`, in either order.
757#[must_use]
758pub fn pair_plane(types: [u8; 2], dishes: usize) -> usize {
759    let (first, second) = if types[1] < types[0] {
760        (usize::from(types[1]), usize::from(types[0]))
761    } else {
762        (usize::from(types[0]), usize::from(types[1]))
763    };
764    let mut plane = 0;
765    for jj in 0..first {
766        plane += dishes - jj - 1;
767    }
768    plane + second
769}

771impl ConvolutionFunctionSet for MosaicPb {
772    /// Group: the dish-class pair plane; cube: the frequency cell of the
773    /// row's window, by `SimplePBConvFunc::findUsefulChannels`'s `chanMap`.
774    /// A window the set was not built for, or a dish class beyond the set's
775    /// classes, is a programmer error.
776    fn key(&self, row: &RowContext, freq_hz: f64, _w_lambda: f64) -> CfKey {
777        let window = self
778            .window(row.spectral_window)
779            .expect("mosaic set built for every selected spectral window");
780        assert!(
781            usize::from(row.antenna_types[0]) < self.dishes
782                && usize::from(row.antenna_types[1]) < self.dishes,
783            "antenna types index the mosaic set's dish classes"
784        );
785        CfKey {
786            group: u16::try_from(pair_plane(row.antenna_types, self.dishes))
787                .expect("pair plane fits u16"),
788            cube: u16::try_from(window.first + window.cell_of(freq_hz)).expect("cell fits u16"),
789        }
790    }
791
792    /// The imaging cell, or its conjugate under
793    /// [`MosaicRules::TcleanConjugateBeams`]: `MosaicFT` grids the data
794    /// and the PSF with the conjugate.
795    fn taps<'s>(&'s self, key: CfKey, _hold: &'s mut CellHold) -> TapLayout<'s> {
796        self.conjugate
797            .get(self.index(key))
798            .unwrap_or(&self.imaging[self.index(key)])
799            .layout()
800    }
801
802    /// The imaging cell at the beam frequency: `MosaicFT` predicts with it
803    /// whatever its gridding takes.
804    fn prediction_taps<'s>(&'s self, key: CfKey, _hold: &'s mut CellHold) -> TapLayout<'s> {
805        self.imaging[self.index(key)].layout()
806    }
807
808    fn max_half_support(&self) -> [u16; 2] {
809        [self.max_half_support; 2]
810    }
811
812    fn weight_taps<'s>(&'s self, key: CfKey, _hold: &'s mut CellHold) -> Option<TapLayout<'s>> {
813        Some(self.weight[self.index(key)].layout())
814    }
815
816    fn mueller(&self) -> &MuellerRouting {
817        &self.mueller
818    }
819
820    fn image_correction(&self) -> &ImageCorrection {
821        &self.correction
822    }
823
824    /// `MosaicFT::finalizeToSky` transforms the gridded weight functions
825    /// into `skyCoverage_p` without the sinc `getImage` divides by
826    /// (`refim_alma_mosaic` against CASA: the corrected weight sat
827    /// `1/sinc` high, 0.4 % at the field edge).
828    fn corrects_weight_image(&self) -> bool {
829        false
830    }
831
832    /// `fmosaic.f` `sectgmosd2`, `gmoswgtd`: `sumwt += weight`; `sectdmos2`
833    /// does not divide.
834    fn normalisation(&self) -> KernelNormalisation {
835        KernelNormalisation::UnitSum
836    }
837
838    fn pointing_ramp(&self) -> bool {
839        true
840    }
841
842    /// `MosaicFT::put` grids through `sectgmosd2` and `gmoswgtd2`, which
843    /// clip a support to the grid (`onmosgrid`), and predicts through
844    /// `sectdmos2`, which drops a sample that overruns it (`omos`).
845    fn grid_edge(&self) -> GridEdge {
846        GridEdge::Clipped
847    }
848
849    /// Every pair plane's imaging and weight cell of every frequency cell,
850    /// all resident.
851    fn tap_footprint(&self) -> TapFootprint {
852        TapFootprint::of(
853            self.imaging
854                .iter()
855                .chain(&self.weight)
856                .chain(&self.conjugate)
857                .map(DenseCell::layout),
858        )
859    }
860}
