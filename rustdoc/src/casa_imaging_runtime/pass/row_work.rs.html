17/// What a row-work pass measures ([`run_row_work_pass`]): each domain's
18/// data gridding work by anchor row, `None` for a domain of plane owners,
19/// and each domain's work in the other modes.
20#[derive(Clone, Debug, Default, PartialEq, Eq)]
21pub struct MeasuredRowWork {
22    /// Each domain's gridding work by anchor row.
23    pub rows: Vec<Option<Vec<u64>>>,
24    /// Each domain's work in every mode; zero for a domain of plane owners,
25    /// which the pass skips.
26    pub modes: Vec<ModeWork>,
27}
28
29impl MeasuredRowWork {
30    /// The bytes a run keeps for row work of `rows` histogram rows in all
31    /// over `domains` domains: the histograms and one [`ModeWork`] per
32    /// domain. Both the worker fit's estimate and the kept charge use it.
33    #[must_use]
34    pub const fn kept_bytes(rows: usize, domains: usize) -> u64 {
35        (rows * size_of::<u64>() + domains * size_of::<ModeWork>()) as u64
36    }
37
38    /// The bytes this row work holds when kept ([`Self::kept_bytes`]).
39    #[must_use]
40    pub fn bytes(&self) -> u64 {
41        let rows = self.rows.iter().flatten().map(Vec::len).sum();
42        Self::kept_bytes(rows, self.modes.len())
43    }
44}
45
46/// A region-split domain's work in each mode a pass may grid, summed over
47/// every placement the row work counts: the data mode's kernel taps
48/// (`Router::owner_and_work`); the PSF's taps, at each
49/// sample's own position (the weight cells for an AW catalog); and the
50/// weight image's taps, which every sample places at the uv origin. Each
51/// adds [`PLACEMENT_WORK`](super::PLACEMENT_WORK).
52#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
53pub struct ModeWork {
54    /// The data mode's work, the row work's total.
55    pub data: u64,
56    /// The PSF's work.
57    pub psf: u64,
58    /// The weight image's work.
59    pub weight: u64,
60}
61
62impl ModeWork {
63    /// The work by anchor row of a pass of `modes`, from the data mode's
64    /// `row_work`, when the pass grids the weight image: the data and PSF
65    /// modes' work, which falls on each sample's own anchor row, scaled
66    /// from the data mode's; and on the centre row, whose owner grids the
67    /// weight image (`Router::weight_owner`), all of the weight image's.
68    /// `None` without the weight mode or its work, when `row_work` itself
69    /// cuts the pass's strips.
70    #[must_use]
71    pub fn weighed(&self, row_work: &[u64], modes: ModeSet) -> Option<Vec<u64>> {
72        if !modes.weight || self.weight == 0 || self.data == 0 || row_work.is_empty() {
73            return None;
74        }
75        let strips = u128::from(if modes.data { self.data } else { 0 })
76            + u128::from(if modes.psf { self.psf } else { 0 });
77        let mut weighed = row_work
78            .iter()
79            .map(|&work| {
80                u64::try_from(u128::from(work) * strips / u128::from(self.data)).unwrap_or(u64::MAX)
81            })
82            .collect::<Vec<_>>();
83        let centre = weighed.len() / 2;
84        weighed[centre] = weighed[centre].saturating_add(self.weight);
85        Some(weighed)
86    }
87}

89/// Row chunks per worker of a row-work pass.
90const CHUNKS_PER_WORKER: usize = 4;
91
92/// Rows a chunk of a row-work pass places at once. It keeps only a
93/// histogram, so this bounds the work between the workers' meetings, not
94/// memory.
95const CHUNK_ROWS: usize = 1 << 12;
96
97/// Whether some domain is split among several region owners
98/// ([`Partition::splits_rows`]), the only case a row-work pass serves.
99#[must_use]
100pub fn needs_row_work(domains: &[PassDomain<'_>]) -> bool {
101    domains.iter().any(|domain| domain.partition.splits_rows())
102}
103
104/// The grid rows a row-work pass measures for `domain`: every row of one
105/// split among region owners, none otherwise.
106fn measured_rows(domain: &PassDomain<'_>) -> usize {
107    if domain.partition.splits_rows() {
108        domain.operator.geometry().grid_shape()[1]
109    } else {
110        0
111    }
112}
113
114/// Bytes a row-work pass over `domains` on `workers` workers holds: each
115/// chunk's histogram, allocated once at the widest measured domain's grid
116/// rows; the totals it returns; each domain's router (an owner per grid row
117/// for region owners, a start per owner for plane owners); and the
118/// containers of chunks, routers and totals. The source's blocks are the
119/// source's charge ([`super::source_block_bytes`]).
120#[must_use]
121pub fn row_work_pass_bytes(domains: &[PassDomain<'_>], workers: usize) -> u64 {
122    let chunks = workers * CHUNKS_PER_WORKER;
123    let widest = domains.iter().map(measured_rows).max().unwrap_or(0);
124    let totals = domains.iter().map(measured_rows).sum::<usize>();
125    let routing = domains
126        .iter()
127        .map(|domain| match domain.partition {
128            Partition::Regions { .. } => {
129                domain.operator.geometry().grid_shape()[1] * size_of::<u16>()
130            }
131            Partition::Planes { owners } => (owners + 1) * size_of::<u32>(),
132        })
133        .sum::<usize>();
134    let containers = chunks * size_of::<(Range<usize>, Vec<u64>, ModeWork)>()
135        + domains.len()
136            * (size_of::<Router>() + size_of::<Option<Vec<u64>>>() + size_of::<ModeWork>());
137    // The team's result per chunk, collected on each call
138    // ([`WorkerTeam::for_each_mut`]).
139    let results = chunks * size_of::<Result<(), PassError>>();
140    ((chunks * widest + totals) * size_of::<u64>() + routing + containers + results) as u64
141}

143/// Each domain's gridding work by anchor row over one traversal that reads
144/// flags and weights only ([`SampleContent::FlagsAndWeights`]): every row
145/// is placed as a pass places it, with `weighting`, and each placement adds
146/// its work, its kernel's taps and
147/// [`PLACEMENT_WORK`](super::PLACEMENT_WORK), to its anchor row
148/// (`Router::owner_and_work`). `None`
149/// for a domain of plane owners. It is what a first pass measures
150/// ([`super::PassSummary::row_work`]) without reading or gridding the
151/// visibilities, so the first pass's strips can be cut balanced
152/// ([`Partition::regions_balanced`]). Each counted placement also adds its
153/// PSF and weight work ([`ModeWork`]), from the kernel set's metadata. It
154/// holds [`row_work_pass_bytes`].
155pub fn run_row_work_pass(
156    domains: &[PassDomain<'_>],
157    weighting: &WeightingGeneration,
158    source: &mut dyn BoundedSource,
159    team: &WorkerTeam,
160    cancel: &Cancel,
161) -> Result<MeasuredRowWork, PassError> {
162    let Some(main) = domains.first() else {
163        return Err(PassError::NoDomains);
164    };
165    let planes = PlaneRange::new(0, main.operator.basis().planes());
166    source
167        .begin(planes, false, SampleContent::FlagsAndWeights)
168        .map_err(PassError::Source)?;
169    let routers = domains
170        .iter()
171        .map(|domain| Router::new(&domain.partition, planes))
172        .collect::<Vec<_>>();
173    let mut totals = domains
174        .iter()
175        .map(|domain| {
176            let rows = measured_rows(domain);
177            (rows > 0).then(|| vec![0_u64; rows])
178        })
179        .collect::<Vec<_>>();
180    let count = team.workers() * CHUNKS_PER_WORKER;
181    // Each chunk's histogram holds the widest domain's rows from the start,
182    // so a later, taller domain never grows it ([`row_work_pass_bytes`]).
183    let widest = domains.iter().map(measured_rows).max().unwrap_or(0);
184    let mut modes = vec![ModeWork::default(); domains.len()];
185    let mut chunks = (0..count)
186        .map(|_| (0..0, vec![0_u64; widest], ModeWork::default()))
187        .collect::<Vec<_>>();
188    stream_blocks(source, cancel, false, |block| {
189        for (index, (domain, router)) in domains.iter().zip(&routers).enumerate() {
190            let Some(total) = &mut totals[index] else {
191                continue;
192            };
193            let cf = domain.operator.cf();
194            for slice in slices(block.len(), count, CHUNK_ROWS) {
195                let mut used = 0;
196                for ((range, _, _), rows) in chunks.iter_mut().zip(chunk_ranges(slice, count)) {
197                    *range = rows;
198                    used += 1;
199                }
200                let rows = total.len();
201                team.for_each_mut(&mut chunks[..used], |_, (range, work, other)| {
202                    let work = &mut work[..rows];
203                    work.fill(0);
204                    *other = ModeWork::default();
205                    // In the block's locality order, as a pass places them:
206                    // a baseline's rows share kernel cells, which a set that
207                    // pages its cells (AW) would otherwise reload row by row.
208                    for position in range.clone() {
209                        domain.resampler.place_each(
210                            domain.operator,
211                            weighting,
212                            &block.row(index, block.ordered_row(position)),
213                            None,
214                            |placement, _, _| {
215                                if !planes.contains(placement.plane) {
216                                    return;
217                                }
218                                let (_, placed) =
219                                    router.owner_and_work(domain.operator, &placement);
220                                if let Some((anchor, taps)) = placed
221                                    && let Some(cell) = work.get_mut(anchor)
222                                {
223                                    *cell += taps;
224                                    other.data += taps;
225                                    other.psf += placement_work(cf.psf_half_support(placement.cf));
226                                    if let Some(support) = cf.weight_half_support(placement.cf) {
227                                        other.weight += placement_work(support);
228                                    }
229                                }
230                            },
231                        )?;
232                    }
233                    Ok::<_, PassError>(())
234                })?;
235                for (_, work, other) in &chunks[..used] {
236                    for (total, work) in total.iter_mut().zip(work) {
237                        *total += work;
238                    }
239                    let domain = &mut modes[index];
240                    domain.data += other.data;
241                    domain.psf += other.psf;
242                    domain.weight += other.weight;
243                }
244            }
245        }
246        Ok(())
247    })?;
248    Ok(MeasuredRowWork {
249        rows: totals,
250        modes,
251    })
