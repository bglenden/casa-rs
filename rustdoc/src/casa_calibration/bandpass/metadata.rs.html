13pub(crate) fn bin_widths(
14    ms: &MeasurementSet,
15    rows: &[SelectedSolveRow],
16    interval: Option<BandpassFrequencyInterval>,
17) -> Result<BTreeMap<i32, usize>, BandpassSolveError> {
18    match interval {
19        Some(BandpassFrequencyInterval::Channels(count)) if count > 1.0 => Ok(rows
20            .iter()
21            .map(|row| (row.data_spw_id, count as usize))
22            .collect()),
23        Some(BandpassFrequencyInterval::Hertz(hertz)) => bandwidth_bin_widths(ms, hertz),
24        _ => Ok(BTreeMap::new()),
25    }
26}
27
28fn bandwidth_bin_widths(
29    ms: &MeasurementSet,
30    hertz: f64,
31) -> Result<BTreeMap<i32, usize>, BandpassSolveError> {
32    let fail = |reason: String| BandpassSolveError::UnsupportedConfiguration { reason };
33    let window = ms
34        .spectral_window()
35        .map_err(|error| fail(error.to_string()))?;
36    let mut widths = BTreeMap::new();
37    for spw in 0..window.row_count() {
38        let channels = window
39            .chan_width(spw)
40            .map_err(|error| fail(error.to_string()))?;
41        let width = channels
42            .first()
43            .filter(|width| width.is_finite() && **width != 0.0)
44            .ok_or_else(|| fail(format!("SPW {spw} has no valid first channel width")))?;
45        widths.insert(spw as i32, (hertz / width.abs()).floor().max(1.0) as usize);
46    }
47    // Once the layer is installed, even windows with a width of one use
48    // the frequency averager's scalar weights for the time centroid.
49    if widths.values().all(|&width| width == 1) {
50        widths.clear();
51    }
52    Ok(widths)
53}

55pub(crate) fn write(
56    path: &Path,
57    solution_windows: &BTreeSet<i32>,
58    channel_bin_widths: &BTreeMap<i32, usize>,
59    selected_rows: &[SelectedSolveRow],
60    combine_windows: bool,
61) -> Result<(), BandpassSolveError> {
62    let mut table = Table::open(TableOptions::new(path)).map_err(metadata_error)?;
63    let selections = selected_rows
64        .iter()
65        .map(|row| (row.data_spw_id, row.channel_selection.as_ref()))
66        .collect::<BTreeMap<_, _>>();
67    let grid = |table: &Table, spw| {
68        select_grid(
69            table,
70            spw as usize,
71            selections.get(&spw).copied().flatten(),
72            channel_bin_widths.get(&spw).copied().unwrap_or(1),
73        )
74    };
75    let aggregate = if combine_windows && selections.len() > 1 {
76        Some(combine_grids(selections.keys().copied(), |spw| {
77            grid(&table, spw)
78        })?)
79    } else {
80        None
81    };
82    for row in 0..table.row_count() {
83        let spw = row as i32;
84        let present = solution_windows.contains(&spw);
85        table
86            .cell_accessor_mut(row, "FLAG_ROW")
87            .and_then(|mut c| c.set(Value::Scalar(ScalarValue::Bool(!present))))
88            .map_err(metadata_error)?;
89        if let Some((reference, combined)) = &aggregate {
90            if *reference == spw {
91                combined.write(&mut table, row)?;
92            }
93        } else if present {
94            grid(&table, spw)?.write(&mut table, row)?;
95        }
96    }
97    table.save(TableOptions::new(path)).map_err(metadata_error)
98}

103    WidthWeighted,
104    Channelized(Option<BandpassFrequencyInterval>),
105}

107/// Check the frequency grid used by the gather and any append destination.
108pub(crate) fn check_gather_frequency(
109    ms: &MeasurementSet,
110    destination: Option<&Path>,
111    rows: &[SelectedSolveRow],
112    layout: GatherFrequencyLayout,
113    combine_windows: bool,
114) -> Result<(), BandpassSolveError> {
115    let existing = destination
116        .map(|path| Table::open(TableOptions::new(path.join("SPECTRAL_WINDOW"))))
117        .transpose()
118        .map_err(metadata_error)?;
119    let source = ms
120        .subtable(casa_ms::schema::SubtableId::SpectralWindow)
121        .ok_or_else(|| metadata_error("missing MeasurementSet SPECTRAL_WINDOW"))?;
122    let selections = rows
123        .iter()
124        .map(|row| (row.data_spw_id, row.channel_selection.as_ref()))
125        .collect::<BTreeMap<_, _>>();
126    let widths = match layout {
127        GatherFrequencyLayout::Channelized(interval) => bin_widths(ms, rows, interval)?,
128        _ => BTreeMap::new(),
129    };
130    let grid = |spw| {
131        let width = match layout {
132            GatherFrequencyLayout::Collapsed => usize::MAX,
133            GatherFrequencyLayout::WidthWeighted => 1,
134            GatherFrequencyLayout::Channelized(_) => widths.get(&spw).copied().unwrap_or(1),
135        };
136        let grid = select_grid(source, spw as usize, selections[&spw], width)?;
137        if matches!(layout, GatherFrequencyLayout::WidthWeighted) {
138            grid.collapse_weighted()
139        } else {
140            Ok(grid)
141        }
142    };
143    let compare = |spw: i32, grid: ChannelGrid| -> Result<(), BandpassSolveError> {
144        let Some(existing) = &existing else {
145            return Ok(());
146        };
147        let row = spw as usize;
148        let scalar = |column| {
149            existing
150                .cell_accessor(row, column)
151                .and_then(|cell| cell.scalar())
152                .map_err(metadata_error)
153        };
154        if matches!(scalar("FLAG_ROW")?, ScalarValue::Bool(true)) {
155            return Ok(());
156        }
157        let count = scalar("NUM_CHAN")?;
158        let frequencies = existing
159            .cell_accessor(row, "CHAN_FREQ")
160            .and_then(|cell| cell.array())
161            .map_err(metadata_error)?;
162        let matches_grid = matches!(count, ScalarValue::Int32(n) if *n as usize == grid.frequencies.len())
163            && matches!(frequencies, ArrayValue::Float64(values) if values.iter().copied().eq(grid.frequencies));
164        if matches_grid {
165            Ok(())
166        } else {
167            Err(metadata_error(format!(
168                "frequency metadata differs for appended SPW {spw}"
169            )))
170        }
171    };
172    if combine_windows && selections.len() > 1 {
173        let (spw, combined) = combine_grids(selections.keys().copied(), grid)?;
174        compare(spw, combined)
175    } else {
176        for spw in selections.keys().copied() {
177            compare(spw, grid(spw)?)?;
178        }
179        Ok(())
180    }
181}

183fn combine_grids(
184    mut selected: impl Iterator<Item = i32>,
185    grid: impl Fn(i32) -> Result<ChannelGrid, BandpassSolveError>,
186) -> Result<(i32, ChannelGrid), BandpassSolveError> {
187    let reference = selected.next().expect("nonempty windows");
188    let mut combined = grid(reference)?;
189    let frequency_offset = combined.frequencies[0];
190    let width_scale = combined.widths[0];
191    for index in 0..combined.frequencies.len() {
192        let width = combined.widths[index] / width_scale;
193        combined.frequencies[index] = (combined.frequencies[index] - frequency_offset) * width;
194        combined.effective_bandwidths[index] = width;
195        combined.widths[index] = width * width;
196    }
197    for spw in selected {
198        combined.accumulate(&grid(spw)?, frequency_offset, width_scale)?;
199    }
200    for index in 0..combined.frequencies.len() {
201        let bandwidth = combined.effective_bandwidths[index];
202        if !bandwidth.is_finite() || bandwidth <= 0.0 {
203            return Err(metadata_error(
204                "Cannot combine windows with invalid channel widths",
205            ));
206        }
207        combined.frequencies[index] = combined.frequencies[index] / bandwidth + frequency_offset;
208        combined.widths[index] = combined.widths[index] / bandwidth * width_scale;
209        combined.effective_bandwidths[index] *= width_scale;
210    }
211    Ok((reference, combined))
212}

214struct ChannelGrid {
215    frequencies: Vec<f64>,
216    widths: Vec<f64>,
217    effective_bandwidths: Vec<f64>,
218}
219
220fn select_grid(
221    table: &Table,
222    row: usize,
223    selection: Option<&ChannelSelection>,
224    bin_width: usize,
225) -> Result<ChannelGrid, BandpassSolveError> {
226    let array = |column| -> Result<Vec<f64>, BandpassSolveError> {
227        let ArrayValue::Float64(values) = table
228            .cell_accessor(row, column)
229            .and_then(|c| c.array())
230            .map_err(metadata_error)?
231        else {
232            return Err(metadata_error(format!("SPW {row} {column} is not Float64")));
233        };
234        Ok(values.iter().copied().collect())
235    };
236    let frequencies = array("CHAN_FREQ")?;
237    let widths = array("CHAN_WIDTH")?;
238    if frequencies.len() != widths.len() {
239        return Err(metadata_error("SPW channel width shape mismatch"));
240    }
241    let channels = crate::solve::grouping::selected_channels(selection, frequencies.len())?;
242    let bins = channels.chunks(bin_width);
243    let frequencies = average_frequencies(&frequencies, &channels, bin_width);
244    let widths = bins
245        .map(|bin| bin.iter().map(|&channel| widths[channel].abs()).sum())
246        .collect::<Vec<_>>();
247    Ok(ChannelGrid {
248        frequencies,
249        effective_bandwidths: widths.clone(),
250        widths,
251    })
252}

254pub(super) fn average_frequencies(
255    frequencies: &[f64],
256    channels: &[usize],
257    bin_width: usize,
258) -> Vec<f64> {
259    channels
260        .chunks(bin_width)
261        .map(|bin| bin.iter().map(|&channel| frequencies[channel]).sum::<f64>() / bin.len() as f64)
262        .collect()
263}
264
265impl ChannelGrid {
266    fn collapse_weighted(self) -> Result<Self, BandpassSolveError> {
267        let width = self.widths.iter().sum::<f64>();
268        let first = self
269            .frequencies
270            .first()
271            .copied()
272            .ok_or_else(|| metadata_error("empty frequency grid"))?;
273        if !width.is_finite() || width <= 0.0 {
274            return Err(metadata_error("invalid total channel width"));
275        }
276        let offset = self
277            .frequencies
278            .iter()
279            .zip(&self.widths)
280            .map(|(frequency, width)| (frequency - first) * width)
281            .sum::<f64>();
282        Ok(Self {
283            frequencies: vec![first + offset / width],
284            widths: vec![width],
285            effective_bandwidths: vec![width],
286        })
287    }
288
289    fn accumulate(
290        &mut self,
291        other: &Self,
292        frequency_offset: f64,
293        width_scale: f64,
294    ) -> Result<(), BandpassSolveError> {
295        if self.frequencies.len() != other.frequencies.len() {
296            return Err(metadata_error(
297                "Cannot combine spectral windows with different channel counts",
298            ));
299        }
300        for index in 0..self.frequencies.len() {
301            let width = other.widths[index] / width_scale;
302            self.frequencies[index] += (other.frequencies[index] - frequency_offset) * width;
303            self.widths[index] += width * width;
304            self.effective_bandwidths[index] += width;
305        }
306        Ok(())
307    }
308
309    fn write(&self, table: &mut Table, row: usize) -> Result<(), BandpassSolveError> {
310        let count = self.frequencies.len();
311        let total: f64 = self.frequencies.iter().sum();
312        // SolvableVisCal::setCTFrequencies aliases resolution to frequency
313        // and sums frequencies as bandwidth. CASA requires exact agreement
314        // when appending; these fields are not used to apply the solutions.
315        for (column, values) in [
316            ("CHAN_FREQ", &self.frequencies),
317            ("RESOLUTION", &self.frequencies),
318            ("CHAN_WIDTH", &self.widths),
319            ("EFFECTIVE_BW", &self.effective_bandwidths),
320        ] {
321            table
322                .cell_accessor_mut(row, column)
323                .and_then(|mut c| c.set(Value::Array(ArrayValue::from_f64_vec(values.clone()))))
324                .map_err(metadata_error)?;
325        }
326        for (column, value) in [
327            ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(count as i32))),
328            (
329                "TOTAL_BANDWIDTH",
330                Value::Scalar(ScalarValue::Float64(total)),
331            ),
332        ] {
333            table
334                .cell_accessor_mut(row, column)
335                .and_then(|mut c| c.set(value))
336                .map_err(metadata_error)?;
337        }
338        Ok(())
339    }
340}
341
342fn metadata_error(error: impl std::fmt::Display) -> BandpassSolveError {
343    BandpassSolveError::UnsupportedConfiguration {
344        reason: error.to_string(),
345    }
