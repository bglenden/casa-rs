2//! Convert a component direction using CASA's reference-frame arguments.
3
4use std::str::FromStr;
5
6use casa_types::measures::direction::DirectionRef;
7use pyo3::exceptions::{PyNotImplementedError, PyOverflowError, PySystemError, PyTypeError};
8use pyo3::prelude::*;
9use pyo3::types::{PyBytes, PyInt, PyString};
10
11use super::PyComponentList;
12use crate::convert::holder::DIRECTION_CODES;
13use crate::convert::value::CasaValue;
14use crate::tools::LogLine;
15
16#[pymethods]
17impl PyComponentList {
18    /// Convert one component's direction into the requested reference frame.
19    #[pyo3(signature = (which, frame), text_signature = "($self, which, frame)")]
20    fn convertrefdir(
21        &self,
22        py: Python<'_>,
23        which: &Bound<'_, PyAny>,
24        frame: &Bound<'_, PyAny>,
25    ) -> PyResult<Py<PyAny>> {
26        let (which, pending) = if !which.is_instance_of::<PyInt>()
27            && which.is_instance(&py.import("numpy")?.getattr("bool_")?)?
28        {
29            (-1, true)
30        } else {
31            match which.extract::<i64>() {
32                Ok(index) => (i64::from(index as i32), false),
33                Err(error) if which.is_none() || error.is_instance_of::<PyOverflowError>(py) => {
34                    (-1, true)
35                }
36                Err(_) => return Err(PyTypeError::new_err("argument _which must be an integer")),
37            }
38        };
39        let name: String = if frame.is_instance_of::<PyString>() {
40            frame.extract()?
41        } else if frame.is_instance_of::<PyBytes>() {
42            let bytes = frame.cast::<PyBytes>()?.as_bytes();
43            let end = bytes
44                .iter()
45                .position(|&byte| byte == 0)
46                .unwrap_or(bytes.len());
47            String::from_utf8_lossy(&bytes[..end]).into_owned()
48        } else {
49            return Err(PyTypeError::new_err("argument _frame must be a string"));
50        };
51        let name = name.split('\0').next().unwrap_or_default();
52        let result = if let Some(name) = DIRECTION_CODES.resolve(name) {
53            let frame = DirectionRef::from_str(name).map_err(|_| PyNotImplementedError::new_err(
54                format!("direction reference {name} is not supported by the shared measure model (#1053)"),
55            ))?;
56            self.call(py, move |state, _| {
57                state.convertrefdir(which, frame).map(CasaValue::Bool)
58            })?
59        } else {
60            self.call(py, |_, lines| {
61                lines.push(LogLine::error(
62                    "Could not parse the 'frame' string: Direction frame not changed",
63                ));
64                Ok(CasaValue::Bool(false))
65            })?
66        };
67        if pending {
68            return Err(PySystemError::new_err(
69                "<built-in function componentlist_convertrefdir> returned a result with an exception set",
70            ));
71        }
72        Ok(result)
73    }
74}
