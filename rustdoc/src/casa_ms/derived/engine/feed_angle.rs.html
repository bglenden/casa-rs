3//! ([`MsCalEngine::feed_parallactic_angle`]).

5use super::MsCalEngine;
6use crate::error::MsResult;
7
8/// How an antenna's mount turns its feed, from `ANTENNA::MOUNT` as casacore's
9/// `MSDerivedValues::setAntennaMount` reads it: the names in all lower or
10/// all upper case, an empty mount as alt-az.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub(crate) enum FeedMount {
13    /// `ALT-AZ`, `ALT-AZ+ROTATOR` or no mount.
14    AltAz,
15    /// `EQUATORIAL`: the feed keeps its angle to the sky.
16    Equatorial,
17    /// `X-Y`.
18    XY,
19    /// `ALT-AZ+NASMYTH-R`.
20    NasmythRight,
21    /// `ALT-AZ+NASMYTH-L`.
22    NasmythLeft,
23    /// `ALT-AZ+BWG-R`, a right-handed beam waveguide.
24    BeamWaveguideRight,
25    /// `ALT-AZ+BWG-L`.
26    BeamWaveguideLeft,
27    /// `ORBITING` or a mount casacore does not know; it warns and uses no
28    /// rotation.
29    Unhandled,
30}
31
32impl FeedMount {
33    pub(crate) fn parse(mount: &str) -> Self {
34        match mount {
35            "" | "alt-az" | "ALT-AZ" | "alt-az+rotator" | "ALT-AZ+ROTATOR" => Self::AltAz,
36            "equatorial" | "EQUATORIAL" => Self::Equatorial,
37            "x-y" | "X-Y" => Self::XY,
38            "alt-az+nasmyth-r" | "ALT-AZ+NASMYTH-R" => Self::NasmythRight,
39            "alt-az+nasmyth-l" | "ALT-AZ+NASMYTH-L" => Self::NasmythLeft,
40            "alt-az+bwg-r" | "ALT-AZ+BWG-R" => Self::BeamWaveguideRight,
41            "alt-az+bwg-l" | "ALT-AZ+BWG-L" => Self::BeamWaveguideLeft,
42            _ => Self::Unhandled,
43        }
44    }
45}
46
47impl MsCalEngine {
48    /// The angle (radians) the antenna's mount turns its feed by on the sky
49    /// at `time_mjd_sec`, towards `field_id`, before its receptor angle: what
50    /// CASA's visibility iterator calls the feed parallactic angle
51    /// (`VisibilityIteratorImpl2::feed_pa`, from casacore
52    /// `MSDerivedValues::parAngle`).
53    ///
54    /// For an alt-az mount it is the parallactic angle χ. A Nasmyth mount
55    /// adds the elevation E (right-handed) or subtracts it (left-handed), and
56    /// a beam-waveguide mount adds E − A, A the azimuth (right-handed), or
57    /// subtracts it (left-handed). An X-Y mount turns by
58    /// atan2(−cos H, −sin H sin δ), H and δ the hour angle and declination.
59    /// An equatorial mount, an orbiting one and an unknown one give 0.
60    ///
61    /// [`Self::parallactic_angle`] is χ for every `ALT-AZ…` mount and 0
62    /// otherwise, as casacore's derived MS columns compute it
63    /// (`MSCalEngine::getPA`); the two differ for Nasmyth, beam-waveguide and
64    /// X-Y mounts and for an empty `MOUNT`.
65    ///
66    /// # Errors
67    ///
68    /// An antenna or field index out of range, or a direction conversion
69    /// that fails.
70    pub fn feed_parallactic_angle(
71        &self,
72        time_mjd_sec: f64,
73        field_id: usize,
74        antenna_id: usize,
75    ) -> MsResult<f64> {
76        let mount = self
77            .antenna_feed_mounts
78            .get(antenna_id)
79            .copied()
80            .ok_or_else(|| crate::error::MsError::InvalidIndex {
81                index: antenna_id,
82                max: self.antenna_feed_mounts.len(),
83                context: "antenna_id".to_string(),
84            })?;
85        let sky = |engine: &Self| engine.sky_parallactic_angle(time_mjd_sec, field_id, antenna_id);
86        let azel = |engine: &Self| engine.azel(time_mjd_sec, field_id, antenna_id);
87        Ok(match mount {
88            FeedMount::AltAz => sky(self)?,
89            FeedMount::NasmythRight => sky(self)? + azel(self)?.1,
90            FeedMount::NasmythLeft => sky(self)? - azel(self)?.1,
91            FeedMount::BeamWaveguideRight => {
92                let (azimuth, elevation) = azel(self)?;
93                sky(self)? + elevation - azimuth
94            }
95            FeedMount::BeamWaveguideLeft => {
96                let (azimuth, elevation) = azel(self)?;
97                sky(self)? - (elevation - azimuth)
98            }
99            FeedMount::XY => {
100                let (hour_angle, declination) = self.hadec(time_mjd_sec, field_id, antenna_id)?;
101                (-hour_angle.cos()).atan2(-hour_angle.sin() * declination.sin())
102            }
103            FeedMount::Equatorial | FeedMount::Unhandled => 0.0,
104        })
105    }
106}

108#[cfg(test)]
109mod tests {
110    use super::*;
111    use casa_types::measures::direction::{DirectionRef, MDirection};
112    use casa_types::measures::position::MPosition;
113
114    #[test]
115    fn mounts_parse_as_casacore_reads_them() {
116        for (mount, expected) in [
117            ("", FeedMount::AltAz),
118            ("ALT-AZ", FeedMount::AltAz),
119            ("alt-az", FeedMount::AltAz),
120            ("ALT-AZ+ROTATOR", FeedMount::AltAz),
121            ("EQUATORIAL", FeedMount::Equatorial),
122            ("X-Y", FeedMount::XY),
123            ("ALT-AZ+NASMYTH-R", FeedMount::NasmythRight),
124            ("alt-az+nasmyth-l", FeedMount::NasmythLeft),
125            ("ALT-AZ+BWG-R", FeedMount::BeamWaveguideRight),
126            ("alt-az+bwg-l", FeedMount::BeamWaveguideLeft),
127            ("ORBITING", FeedMount::Unhandled),
128            ("SPACE-HALCA", FeedMount::Unhandled),
129            // casacore compares whole names in one case only.
130            ("Alt-Az", FeedMount::Unhandled),
131        ] {
132            assert_eq!(FeedMount::parse(mount), expected, "{mount:?}");
133        }
134    }
135
136    #[test]
137    fn each_mount_turns_the_feed_by_its_casacore_term() {
138        // Seven antennas at the VLA, one per mount kind, towards Dec +45°.
139        let position = MPosition::new_itrf(-1_601_185.4, -5_041_977.5, 3_554_875.9);
140        let mut engine = MsCalEngine::from_parts(
141            vec![position.clone(); 7],
142            vec![MDirection::from_angles(
143                0.3,
144                std::f64::consts::FRAC_PI_4,
145                DirectionRef::J2000,
146            )],
147            position,
148            casa_test_support::deterministic_measures_provider(),
149        );
150        engine.antenna_feed_mounts = [
151            "ALT-AZ",
152            "ALT-AZ+NASMYTH-R",
153            "ALT-AZ+NASMYTH-L",
154            "ALT-AZ+BWG-R",
155            "ALT-AZ+BWG-L",
156            "EQUATORIAL",
157            "X-Y",
158        ]
159        .map(FeedMount::parse)
160        .into();
161        let time = 59_000.5 * 86_400.0;
162        let chi = engine.parallactic_angle(time, 0, 0).unwrap();
163        let (azimuth, elevation) = engine.azel(time, 0, 0).unwrap();
164        let (hour_angle, declination) = engine.hadec(time, 0, 0).unwrap();
165        let feed = |antenna| engine.feed_parallactic_angle(time, 0, antenna).unwrap();
166        assert_eq!(feed(0), chi);
167        assert_eq!(feed(1), chi + elevation);
168        assert_eq!(feed(2), chi - elevation);
169        assert_eq!(feed(3), chi + elevation - azimuth);
170        assert_eq!(feed(4), chi - (elevation - azimuth));
171        assert_eq!(feed(5), 0.0);
172        assert_eq!(
173            feed(6),
174            (-hour_angle.cos()).atan2(-hour_angle.sin() * declination.sin())
175        );
176        // The derived columns' angle ignores the mount's own turn.
177        assert_eq!(engine.parallactic_angle(time, 0, 3).unwrap(), chi);
178    }
