15/// One bounded block of stored MAIN row-selection facts.
16#[derive(Debug, Clone, PartialEq)]
17pub(crate) struct MainRowSelectionBlock {
18    first_row: usize,
19    pub(super) data_description_ids: Vec<i32>,
20    pub(super) field_ids: Vec<i32>,
21    antenna1: Vec<i32>,
22    antenna2: Vec<i32>,
23    times_mjd_seconds: Vec<f64>,
24    pub(super) time_centroids_mjd_seconds: Vec<f64>,
25    state_ids: Vec<i32>,
26    observation_ids: Vec<i32>,
27    flag_rows: Vec<bool>,
28    pub(super) uvw_m: Vec<f64>,
29}
30
31impl MainRowSelectionBlock {
32    fn with_capacity(rows: usize) -> MsResult<Self> {
33        let uvw_values = rows.checked_mul(3).ok_or_else(|| {
34            MsError::InvalidInput("MAIN row-selection UVW capacity overflow".to_string())
35        })?;
36        Ok(Self {
37            first_row: 0,
38            data_description_ids: Vec::with_capacity(rows),
39            field_ids: Vec::with_capacity(rows),
40            antenna1: Vec::with_capacity(rows),
41            antenna2: Vec::with_capacity(rows),
42            times_mjd_seconds: Vec::with_capacity(rows),
43            time_centroids_mjd_seconds: Vec::with_capacity(rows),
44            state_ids: Vec::with_capacity(rows),
45            observation_ids: Vec::with_capacity(rows),
46            flag_rows: Vec::with_capacity(rows),
47            uvw_m: Vec::with_capacity(uvw_values),
48        })
49    }
50
51    /// Number of MAIN rows represented by this block.
52    #[must_use]
53    pub(crate) fn len(&self) -> usize {
54        self.data_description_ids.len()
55    }
56
57    /// Return one row by its block-local offset, numbered from the block's
58    /// first row as a contiguous block's are.
59    #[must_use]
60    pub(crate) fn row(&self, offset: usize) -> Option<SelectedObservationRow> {
61        (offset < self.len()).then(|| SelectedObservationRow {
62            physical_row: self.first_row + offset,
63            data_description_id: self.data_description_ids[offset],
64            field_id: self.field_ids[offset],
65            antenna1: self.antenna1[offset],
66            antenna2: self.antenna2[offset],
67            time_mjd_seconds: self.times_mjd_seconds[offset],
68            time_centroid_mjd_seconds: self.time_centroids_mjd_seconds[offset],
69            state_id: self.state_ids[offset],
70            observation_id: self.observation_ids[offset],
71            flag_row: self.flag_rows[offset],
72            uvw_m: [
73                self.uvw_m[offset * 3],
74                self.uvw_m[offset * 3 + 1],
75                self.uvw_m[offset * 3 + 2],
76            ],
77        })
78    }
79
80    /// Iterate the stored facts in physical MAIN row order.
81    #[cfg(test)]
82    pub(crate) fn rows(&self) -> impl ExactSizeIterator<Item = SelectedObservationRow> + '_ {
83        (0..self.len()).map(|offset| self.row(offset).expect("offset is bounded by block length"))
84    }
85}

87/// Pull cursor over bounded reusable MAIN row-selection storage.
88pub(crate) struct MainRowSelectionCursor {
89    plan: crate::MsReadPlan,
90    /// The next MAIN row, or with a walk the next position in it.
91    next_row: usize,
92    offset: usize,
93    row_indices: Vec<usize>,
94    block: MainRowSelectionBlock,
95    walk: Option<Arc<MainRowWalk>>,
96    /// The walk's group holding `next_row`.
97    group: usize,
98}
99
100/// Selected MAIN rows in the order a traversal reads them: grouped, each
101/// group ascending; `ends[g]` closes group `g`. A selection's
102/// DATA_DESCRIPTIONs are its groups, so a traversal delivers each spectral
103/// window's rows together.
104#[derive(Debug)]
105pub(crate) struct MainRowWalk {
106    rows: Vec<u32>,
107    ends: Vec<usize>,
108}
109
110impl MainRowWalk {
111    /// The walk over `rows`, whose groups end at `ends` (ascending, the
112    /// last at `rows.len()`).
113    pub(crate) fn new(rows: Vec<u32>, ends: Vec<usize>) -> Self {
114        debug_assert!(
115            ends.windows(2).all(|pair| pair[0] <= pair[1])
116                && ends.last().is_none_or(|&end| end == rows.len()),
117            "walk groups end in order at the last row"
118        );
119        Self { rows, ends }
120    }
121
122    /// Bytes a walk of `rows` rows in `groups` groups holds.
123    pub(crate) const fn bytes(rows: usize, groups: usize) -> Option<usize> {
124        match (
125            rows.checked_mul(size_of::<u32>()),
126            groups.checked_mul(size_of::<usize>()),
127        ) {
128            (Some(rows), Some(groups)) => rows.checked_add(groups),
129            _ => None,
130        }
131    }
132}
133
134impl MainRowSelectionCursor {
135    pub(crate) const fn retained_bytes_per_row() -> usize {
136        SelectedObservationRow::STORAGE_BYTES_PER_ROW + size_of::<usize>()
137    }
138
139    pub(crate) fn next(
140        &mut self,
141        measurement_set: &MeasurementSet,
142    ) -> MsResult<Option<SelectedObservationRow>> {
143        if self.offset >= self.block.len() {
144            self.row_indices.clear();
145            match &self.walk {
146                None => {
147                    if self.next_row == self.plan.row_count {
148                        return Ok(None);
149                    }
150                    let end = self
151                        .next_row
152                        .saturating_add(self.plan.rows_per_block)
153                        .min(self.plan.row_count);
154                    self.row_indices.extend(self.next_row..end);
155                    self.next_row = end;
156                }
157                Some(walk) => {
158                    // A block never crosses a group, so every block reads
159                    // ascending rows.
160                    while walk.ends.get(self.group) == Some(&self.next_row) {
161                        self.group += 1;
162                    }
163                    let Some(&group_end) = walk.ends.get(self.group) else {
164                        return Ok(None);
165                    };
166                    let end = self
167                        .next_row
168                        .saturating_add(self.plan.rows_per_block)
169                        .min(group_end);
170                    self.row_indices.extend(
171                        walk.rows[self.next_row..end]
172                            .iter()
173                            .map(|&row| row as usize),
174                    );
175                    self.next_row = end;
176                }
177            }
178            fill_main_row_selection_block(
179                &measurement_set.main,
180                self.row_indices[0],
181                &self.row_indices,
182                &mut self.block,
183            )?;
184            self.offset = 0;
185        }
186        let mut fact = self
187            .block
188            .row(self.offset)
189            .expect("offset is bounded by MAIN selection block length");
190        // A walk's block need not be contiguous.
191        fact.physical_row = self.row_indices[self.offset];
192        self.offset += 1;
193        Ok(Some(fact))
194    }
195}

197fn fill_main_row_selection_block(
198    table: &Table,
199    first_row: usize,
200    row_indices: &[usize],
201    block: &mut MainRowSelectionBlock,
202) -> MsResult<()> {
203    block.first_row = first_row;
204    {
205        let mut scalar_destinations = [
206            RequiredScalarColumnDestination::new(
207                "DATA_DESC_ID",
208                RequiredScalarColumnValuesMut::Int32(&mut block.data_description_ids),
209            ),
210            RequiredScalarColumnDestination::new(
211                "FIELD_ID",
212                RequiredScalarColumnValuesMut::Int32(&mut block.field_ids),
213            ),
214            RequiredScalarColumnDestination::new(
215                "ANTENNA1",
216                RequiredScalarColumnValuesMut::Int32(&mut block.antenna1),
217            ),
218            RequiredScalarColumnDestination::new(
219                "ANTENNA2",
220                RequiredScalarColumnValuesMut::Int32(&mut block.antenna2),
221            ),
222            RequiredScalarColumnDestination::new(
223                "TIME",
224                RequiredScalarColumnValuesMut::Float64(&mut block.times_mjd_seconds),
225            ),
226            RequiredScalarColumnDestination::new(
227                "TIME_CENTROID",
228                RequiredScalarColumnValuesMut::Float64(&mut block.time_centroids_mjd_seconds),
229            ),
230            RequiredScalarColumnDestination::new(
231                "STATE_ID",
232                RequiredScalarColumnValuesMut::Int32(&mut block.state_ids),
233            ),
234            RequiredScalarColumnDestination::new(
235                "OBSERVATION_ID",
236                RequiredScalarColumnValuesMut::Int32(&mut block.observation_ids),
237            ),
238            RequiredScalarColumnDestination::new(
239                "FLAG_ROW",
240                RequiredScalarColumnValuesMut::Bool(&mut block.flag_rows),
241            ),
242        ];
243        table.required_scalar_columns_for_rows_into(row_indices, &mut scalar_destinations)?;
244    }
245    let uvw_shape = table
246        .column_accessor("UVW")?
247        .fill_array_cells_1d_typed_uncached(
248            row_indices,
249            SelectedArray1DCellsMut::Float64(&mut block.uvw_m),
250        )?;
251    if uvw_shape.row_count != row_indices.len() || uvw_shape.axis0_count != 3 {
252        return Err(MsError::ColumnTypeMismatch {
253            column: "UVW".to_string(),
254            table: "MAIN".to_string(),
255            expected: "f64[3]".to_string(),
256            found: format!("f64[{}]", uvw_shape.axis0_count),
257        });
258    }
259    let expected = row_indices.len();
260    let lengths = [
261        block.data_description_ids.len(),
262        block.field_ids.len(),
263        block.antenna1.len(),
264        block.antenna2.len(),
265        block.times_mjd_seconds.len(),
266        block.time_centroids_mjd_seconds.len(),
267        block.state_ids.len(),
268        block.observation_ids.len(),
269        block.flag_rows.len(),
270        block.uvw_m.len() / 3,
271    ];
272    if lengths.into_iter().any(|actual| actual != expected) {
273        return Err(MsError::InvalidInput(
274            "MAIN row-selection column lengths do not match the requested block".to_string(),
275        ));
276    }
277    Ok(())
278}

280impl MeasurementSet {
281    /// Visit the stored MAIN facts needed by row selection in bounded reusable blocks.
282    pub(crate) fn visit_main_row_selection_blocks(
283        &self,
284        plan: crate::MsReadPlan,
285        mut visit: impl FnMut(&MainRowSelectionBlock),
286    ) -> MsResult<()> {
287        let mut cursor = self.main_row_selection_cursor(plan)?;
288        while cursor.next_row < cursor.plan.row_count {
289            let first_row = cursor.next_row;
290            let end = first_row
291                .saturating_add(cursor.plan.rows_per_block)
292                .min(cursor.plan.row_count);
293            cursor.row_indices.clear();
294            cursor.row_indices.extend(first_row..end);
295            fill_main_row_selection_block(
296                &self.main,
297                first_row,
298                &cursor.row_indices,
299                &mut cursor.block,
300            )?;
301            cursor.next_row = end;
302            visit(&cursor.block);
303        }
304        Ok(())
305    }
306
307    pub(crate) fn main_row_selection_cursor(
308        &self,
309        plan: crate::MsReadPlan,
310    ) -> MsResult<MainRowSelectionCursor> {
311        self.main_row_cursor(plan, None)
312    }
313
314    /// A cursor over `walk`'s rows in its order, in blocks of the plan's
315    /// rows that never cross one of its groups.
316    pub(crate) fn main_row_walk_cursor(
317        &self,
318        plan: crate::MsReadPlan,
319        walk: Arc<MainRowWalk>,
320    ) -> MsResult<MainRowSelectionCursor> {
321        if walk
322            .rows
323            .iter()
324            .any(|&row| row as usize >= self.row_count())
325        {
326            return Err(MsError::InvalidInput(
327                "MAIN row walk names a row beyond the table".to_string(),
328            ));
329        }
330        self.main_row_cursor(plan, Some(walk))
331    }
332
333    fn main_row_cursor(
334        &self,
335        plan: crate::MsReadPlan,
336        walk: Option<Arc<MainRowWalk>>,
337    ) -> MsResult<MainRowSelectionCursor> {
338        if plan.row_count != self.row_count() {
339            return Err(MsError::InvalidInput(format!(
340                "MAIN row plan covers {} rows but the table has {}",
341                plan.row_count,
342                self.row_count()
343            )));
344        }
345        if plan.row_count == 0 {
346            return Ok(MainRowSelectionCursor {
347                plan,
348                next_row: 0,
349                offset: 0,
350                row_indices: Vec::new(),
351                block: MainRowSelectionBlock::with_capacity(0)?,
352                walk: None,
353                group: 0,
354            });
355        }
356        if plan.rows_per_block == 0 {
357            return Err(MsError::InvalidInput(
358                "nonempty MAIN row plan has zero rows per block".to_string(),
359            ));
360        }
361        let required_bytes = plan
362            .rows_per_block
363            .checked_mul(SelectedObservationRow::STORAGE_BYTES_PER_ROW)
364            .ok_or_else(|| {
365                MsError::InvalidInput("MAIN row block byte count overflow".to_string())
366            })?;
367        if plan.bytes_per_block < required_bytes {
368            return Err(MsError::InvalidInput(format!(
369                "MAIN row plan reserves {} bytes per block but row selection requires at least {required_bytes}",
370                plan.bytes_per_block
371            )));
372        }
373        Ok(MainRowSelectionCursor {
374            plan,
375            next_row: 0,
376            offset: 0,
377            row_indices: Vec::with_capacity(plan.rows_per_block),
378            block: MainRowSelectionBlock::with_capacity(plan.rows_per_block)?,
379            walk,
380            group: 0,
381        })
382    }
