26struct Request<'a> {
27    outfiles: [&'a str; 5],
28    axes: IntList,
29    region: RegionArgument,
30    mask: String,
31    stretch: bool,
32    history: bool,
33}

35trait ProductPixel: ImagePixel {
36    fn wrap(image: Storage<Self>) -> ImageHandle;
37}
38
39type ExtractComponent<T> = fn(Complex<T>) -> T;
40
41fn create_product<T: ImagePixel>(plan: &ImageFourier<'_>, path: &Path) -> PyResult<Storage<T>> {
42    let shape = plan.shape().to_vec();
43    let tiled = TiledShape::new(shape.clone(), std::mem::size_of::<T>()).map_err(raised)?;
44    PagedImage::create_with_tile_shape_and_cache(
45        shape,
46        tiled.tile_shape().to_vec(),
47        plan.coordinates().clone(),
48        path,
49        16 * 1024 * 1024,
50    )
51    .map(|image| Storage::Paged(Box::new(image)))
52    .map_err(raised)
53}

55macro_rules! product_pixels {
56    ($($pixel:ty => $kind:ident),+ $(,)?) => {
57        $(impl ProductPixel for $pixel {
58            fn wrap(image: Storage<Self>) -> ImageHandle {
59                ImageHandle::$kind(image)
60            }
61        })+
62    };
63}

65product_pixels!(
66    f32 => Float,
67    f64 => Double,
68    Complex32 => Complex,
69    Complex64 => DComplex,

72impl Request<'_> {
73    fn run(&self, attachment: &AttachedImage) -> PyResult<()> {
74        let destinations: Vec<_> = self
75            .outfiles
76            .iter()
77            .filter(|name| !name.is_empty())
78            .map(Path::new)
79            .collect();
80        if destinations.is_empty() {
81            return Err(raised("No output image names have been specified"));
82        }
83        publish_image_set(self.outfiles, false, |staged| {
84            read_locked(attachment, |source| {
85                for target in &destinations {
86                    for input in source
87                        .path()
88                        .into_iter()
89                        .chain(attachment.operand_paths().iter().map(|path| path.as_path()))
90                    {
91                        ensure_outfile_does_not_alias_input(target, input).map_err(raised)?;
92                    }
93                }
94                if let ImageHandle::Float(Storage::Temp {
95                    image,
96                    fits: Some(fits),
97                    ..
98                }) = source
99                {
100                    self.compute(
101                        attachment,
102                        source,
103                        &fits.image_view(image),
104                        &destinations,
105                        staged,
106                    )
107                } else {
108                    with_pixels!(source, pixels => with_storage!(
109                        pixels, image => self.compute(attachment, source, image.as_ref(), &destinations, staged)
110                    ))
111                }
112            })
113        })
114    }
115
116    fn compute<T: FourierPixel>(
117        &self,
118        attachment: &AttachedImage,
119        source: &ImageHandle,
120        input: &dyn ImageInterface<T>,
121        destinations: &[&Path],
122        staged: [Option<&Path>; 5],
123    ) -> PyResult<()>
124    where
125        T::Scalar: ProductPixel,
126        Complex<T::Scalar>: ProductPixel,
127    {
128        let beams = attachment.beam_view();
129        let plan = ImageFourier::new(
130            input,
131            source.coordinate_record(),
132            beams.record(),
133            FourierOptions {
134                region: self.region.selection(),
135                axes: &self.axes.0,
136                mask: Some(&self.mask),
137                stretch: self.stretch,
138                destinations,
139            },
140        )
141        .map_err(raised)?;
142        let section = plan.section();
143        let mut image_info = section.image_info().to_record();
144        let original = beams.beam_set().map_err(raised)?;
145        if original.is_multi() {
146            let channels = section.beam_planes(
147                input.find_axis(CoordinateType::Spectral),
148                original.n_channels(),
149            );
150            let polarizations =
151                section.beam_planes(input.find_axis(CoordinateType::Stokes), original.n_stokes());
152            subset_beam_record(
153                &mut image_info,
154                Some(beams.record()),
155                &original,
156                &channels,
157                &polarizations,
158            )
159            .map_err(raised)?;
160        } else {
161            beams.apply_to(&mut image_info);
162        }
163        let scratch = plan.transform(input).map_err(raised)?;
164        let components: [ExtractComponent<T::Scalar>; 4] =
165            [|value| value.re, |value| value.im, T::amplitude, T::phase];
166        let history = source.history_entries()?;
167        let call = self.history.then(|| self.history_call());
168        for (index, component) in components.into_iter().enumerate() {
169            if let Some(path) = staged[index] {
170                let mut output = create_product(&plan, path)?;
171                with_storage!(&mut output, image =>
172                    plan.write_product(input, &scratch, image.as_mut(), index == 3, component)
173                )
174                .map_err(raised)?;
175                with_storage!(&mut output, image => image.set_image_info_record(image_info.clone()))
176                    .map_err(raised)?;
177                let mut output = T::Scalar::wrap(output);
178                output.set_coordinate_record(plan.coordinate_record().clone())?;
179                output.append_history_entries(&history)?;
180                if let Some(call) = &call {
181                    output.record_history("fft", call.clone())?;
182                }
183                output.save().map_err(raised)?;
184            }
185        }
186        if let Some(path) = staged[4] {
187            let mut output = create_product(&plan, path)?;
188            with_storage!(&mut output, image => plan.write_product(input, &scratch, image.as_mut(), false, |value| value))
189                .map_err(raised)?;
190            with_storage!(&mut output, image => image.set_image_info_record(image_info))
191                .map_err(raised)?;
192            let mut output = Complex::<T::Scalar>::wrap(output);
193            output.set_coordinate_record(plan.coordinate_record().clone())?;
194            output.append_history_entries(&history)?;
195            if let Some(call) = call {
196                output.record_history("fft", call)?;
197            }
198            output.save().map_err(raised)?;
199        }
200        Ok(())
201    }
202
203    fn history_call(&self) -> String {
204        let [real, imag, amp, phase, complex] = self.outfiles.map(history_string);
205        let axes = self
206            .axes
207            .0
208            .iter()
209            .map(i64::to_string)
210            .collect::<Vec<_>>()
211            .join(",");
212        let region = match self.region.selection() {
213            Some(SubimageRegion::Text(text)) => history_string(text),
214            _ => self.region.history(),
215        };
216        let mask = history_string(&self.mask);
217        let stretch = self.stretch;
218        format!(
219            "ia.fft(real={real}, imag={imag}, amp={amp}, phase={phase}, axes=[{axes}], region={region}, mask={mask}, stretch={stretch}, complex={complex})"
220        )
221    }
222}
223
224fn history_string(value: &str) -> String {
225    format!("\"{}\"", if value.len() > 300 { "..." } else { value })
226}

228#[pymethods]
229impl PyImageTool {
230    /// Write selected products of the forward Fourier transform of an image.
231    #[allow(clippy::too_many_arguments)]
232    #[pyo3(signature = (real="", imag="", amp="", phase="", axes=IntList::of(&[-1]), region=RegionArgument::default(), mask=None, stretch=false, complex=""), text_signature = "($self, real='', imag='', amp='', phase='', axes=[-1], region={}, mask='', stretch=False, complex='')")]
233    fn fft(
234        &self,
235        py: Python<'_>,
236        real: &str,
237        imag: &str,
238        amp: &str,
239        phase: &str,
240        axes: IntList,
241        region: RegionArgument,
242        mask: Option<&Bound<'_, PyAny>>,
243        #[pyo3(from_py_with = require_bool)] stretch: bool,
244        complex: &str,
245    ) -> PyResult<bool> {
246        let mask = mask_argument(mask)?;
247        py.detach(|| {
248            let state = self.state();
249            let Some(attachment) = state.image.as_ref() else {
250                return Ok(false);
251            };
252            Request {
253                outfiles: [real, imag, amp, phase, complex],
254                axes,
255                region,
256                mask,
257                stretch,
258                history: state.history,
259            }
260            .run(attachment)?;
261            Ok(true)
262        })
263    }
264}
