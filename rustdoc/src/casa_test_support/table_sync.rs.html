2//! What a table's `table.lock` publishes to other processes, compared with
3//! the table as persisted.
4//!
5//! casacore takes a reopened table's row count from the sync data in
6//! `table.lock`, refuses a lock when its column count differs from the
7//! table's, and asserts that it holds one change counter per data manager
8//! (`PlainTable`, `ColumnSet::resync`). These helpers read both sides so a
9//! test can check that a published write describes the persisted table.
10//!
11//! Reading `table.lock` through a separate descriptor and closing it drops
12//! every `fcntl` lock the calling process holds on that file, so call
13//! [`published_table_sync`] only while the process holds no lock on the
14//! table.
15
16use std::io::Cursor;
17use std::path::Path;
18
19use casa_aipsio::AipsIo;
20use casa_tables::{Table, TableOptions};
21
22/// Size of the request list at the start of `table.lock`.
23const REQUEST_LIST_BYTES: usize = (1 + 2 * 32) * 4;
24
25/// A table's row, column and data-manager counts.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub struct TableShape {
28    /// Rows.
29    pub rows: u64,
30    /// Columns.
31    pub columns: usize,
32    /// Data managers (per-data-manager change counters in sync data).
33    pub data_managers: usize,
34}
35
36/// The sync data a table's `table.lock` publishes.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub struct PublishedSync {
39    /// The published shape.
40    pub shape: TableShape,
41    /// The modify counter.
42    pub modify_counter: u32,
43}
44
45/// The sync data in `table_dir/table.lock`, `None` when there is none.
46///
47/// # Panics
48///
49/// Panics when the sync data is present but malformed.
50pub fn published_table_sync(table_dir: &Path) -> Option<PublishedSync> {
51    let bytes = std::fs::read(table_dir.join("table.lock")).ok()?;
52    let length_at = REQUEST_LIST_BYTES;
53    let length = bytes.get(length_at..length_at + 4)?;
54    let length = u32::from_be_bytes(length.try_into().expect("four bytes")) as usize;
55    if length == 0 {
56        return None;
57    }
58    let payload = bytes
59        .get(length_at + 4..length_at + 4 + length)
60        .expect("sync payload within table.lock")
61        .to_vec();
62    let mut io = AipsIo::new_read_only(Cursor::new(payload));
63    let version = io.getstart("sync").expect("sync object");
64    let rows = if version == 2 {
65        io.get_u64().expect("rows")
66    } else {
67        u64::from(io.get_u32().expect("rows"))
68    };
69    let columns = io.get_i32().expect("columns");
70    let modify_counter = io.get_u32().expect("modify counter");
71    let data_managers = if columns >= 0 {
72        io.get_u32().expect("table change counter");
73        io.getstart("Block").expect("data-manager counters");
74        let counters = io.getnew_u32().expect("data-manager counters");
75        io.getend().expect("end of data-manager counters");
76        counters.len()
77    } else {
78        0
79    };
80    io.getend().expect("end of sync object");
81    Some(PublishedSync {
82        shape: TableShape {
83            rows,
84            columns: columns.max(0) as usize,
85            data_managers,
86        },
87        modify_counter,
88    })
89}
90
91/// The table at `table_dir` as persisted in its `table.dat`: the row count
92/// from the `table.dat` header (not the sync data, which a reopened table
93/// trusts), and the columns and data managers it describes.
94///
95/// # Panics
96///
97/// Panics when the table cannot be read.
98pub fn persisted_table_shape(table_dir: &Path) -> TableShape {
99    let file = std::fs::File::open(table_dir.join("table.dat")).expect("open table.dat");
100    let mut io = AipsIo::new_read_only(file);
101    let version = io.getstart("Table").expect("Table object");
102    let rows = if version >= 3 {
103        io.get_u64().expect("rows")
104    } else {
105        u64::from(io.get_u32().expect("rows"))
106    };
107    let table = Table::open(TableOptions::new(table_dir)).expect("open the table");
108    TableShape {
109        rows,
110        columns: table.schema().map_or(0, |schema| schema.columns().len()),
111        data_managers: table.data_manager_info().len(),
112    }
113}
