2//! The MSv2 adapter's pointing: the `POINTING` rows of the requested
3//! antennas over the requested times (and each antenna's nearest row on
4//! either side), read once as the production selected-observation source
5//! reads them (`selected_pointing.rs`), as the boundary's series.

14/// The most rows of `POINTING` scanned per block while reading. Each block
15/// re-reads the storage manager's index (#949), so blocks are as large as a
16/// sixteenth of the request's budget allows, up to this.
17const SCAN_ROWS: usize = 1 << 16;
18
19/// The polynomial terms a `POINTING` row may have (`NUM_POLY` + 1): the
20/// bound the production imaging source plans with.
21pub(crate) const POLYNOMIAL_TERMS: usize = 4;
22
23/// The rows of `POINTING` scanned per block for a request of `budget`
24/// bytes: a sixteenth of it, up to [`SCAN_ROWS`].
25pub(crate) fn scan_rows(budget: usize) -> usize {
26    (budget / 16 / SCALAR_SCAN_BYTES_PER_ROW).clamp(1, SCAN_ROWS)
27}

29impl Msv2Observation {
30    /// The most bytes a pointing query of this observation holds while it
31    /// is read and once it is answered, whatever its antennas and times
32    /// ([`casa_observation::Observation::pointing_bytes`]): every `POINTING`
33    /// row taken as a candidate at four polynomial terms (the production
34    /// source's bound), with one scan block of at most every row, and the
35    /// series it gives. A [`PointingRequest`] with this budget is never
36    /// refused for its size, and reads `POINTING` in blocks as large as a
37    /// query may scan. Zero without a `POINTING` table. Also the rows of the
38    /// scan block it holds: `POINTING`'s, up to [`SCAN_ROWS`].
39    pub(super) fn pointing_bound(&self) -> Result<(usize, usize), ObservationError> {
40        let mut domain = SelectedPointingQueryDomain::builder();
41        for antenna in 0..self.catalogue.antennas.len() {
42            let id = i32::try_from(antenna).map_err(|_| ObservationError::InvalidStore {
43                reason: format!("antenna {antenna} beyond the MSv2 identifier range"),
44            })?;
45            domain.observe_row(id, id, 0.0).map_err(storage)?;
46        }
47        let Some(domain) = domain.finish().map_err(storage)? else {
48            return Ok((0, 1));
49        };
50        let (catalog, scan_bytes_per_row) = self
51            .ms
52            .selected_pointing_catalog_requirements(&domain, POLYNOMIAL_TERMS)
53            .map_err(storage)?;
54        // A scan block holds at most the table's rows.
55        let rows = SCAN_ROWS.min(catalog.source_rows_scanned()).max(1);
56        // The series holds a time, an interval, an origin and two
57        // coefficients a term per retained row. An AW query may additionally
58        // hold the table row, coverage half-interval and nonpositive index.
59        // The admitted catalog footprint covers those values.
60        let bytes = catalog
61            .construction_peak_bytes()
62            .saturating_add(rows.saturating_mul(scan_bytes_per_row))
63            .saturating_add(catalog.retained_bytes());
64        Ok((bytes, rows))
65    }
66
67    /// [`casa_observation::Observation::pointing`]: an MSv2's times are its
68    /// `TIME` values (its partitions' time origin is MJD 0), so the
69    /// request's times are `POINTING`'s.
70    pub(super) fn read_pointing(
71        &self,
72        request: &PointingRequest,
73        aw_coverage: bool,
74    ) -> Result<PointingSeries, ObservationError> {
75        self.partition(request.partition)?;
76        let (start, end) = (*request.times.start(), *request.times.end());
77        if !(start.is_finite() && end.is_finite() && start <= end) {
78            return Err(ObservationError::InvalidRequest {
79                reason: format!("pointing times {start}..={end} s are not a finite span"),
80            });
81        }
82        let mut domain = SelectedPointingQueryDomain::builder();
83        for antenna in &request.antennas {
84            let id = i32::try_from(antenna.0)
85                .ok()
86                .filter(|_| (antenna.0 as usize) < self.catalogue.antennas.len())
87                .ok_or_else(|| ObservationError::InvalidRequest {
88                    reason: format!("no antenna {antenna:?}"),
89                })?;
90            for time in [start, end] {
91                domain.observe_row(id, id, time).map_err(storage)?;
92            }
93        }
94        let store_has_pointing = self
95            .ms
96            .subtable(crate::SubtableId::Pointing)
97            .is_some_and(|table| table.row_count() > 0);
98        let Some(domain) = domain.finish().map_err(storage)? else {
99            return Ok(PointingSeries {
100                store_has_pointing,
101                ..PointingSeries::empty(DirectionRef::J2000)
102            });
103        };
104        let column = match request.direction {
105            PointingDirection::Beam => PointingDirectionColumn::Direction,
106            PointingDirection::Target => PointingDirectionColumn::Target,
107        };
108        // A budget holding the whole bound holds its scan block, every row
109        // up to `SCAN_ROWS`; a smaller one, a sixteenth of itself.
110        let (bound, rows) = self.pointing_bound()?;
111        let scan = if request.budget_bytes >= bound {
112            rows
113        } else {
114            scan_rows(request.budget_bytes)
115        };
116        let plan = PointingReadPlan::new(scan, POLYNOMIAL_TERMS, request.budget_bytes.max(1))
117            .map_err(storage)?
118            .with_aw_coverage(aw_coverage);
119        self.ms
120            .prepare_selected_pointing_catalog(column, &domain, plan)
121            .map_err(|error| match error {
122                MsError::OverBudget {
123                    required,
124                    available,
125                    ..
126                } => ObservationError::OverBudget {
127                    what: "an MSv2 pointing query",
128                    required,
129                    budget: available,
130                },
131                error => ObservationError::InvalidStore {
132                    reason: format!("POINTING: {error}"),
133                },
134            })?
135            .series(aw_coverage)
136            .map(|series| PointingSeries {
137                store_has_pointing,
138                ..series
139            })
140    }

324    /// One `POINTING` row as the table states it.
325    struct Row {
326        antenna: i32,
327        time_s: f64,
328        interval_s: f64,
329        origin_s: f64,
330        /// `DIRECTION`'s terms, `[2, NUM_POLY + 1]`.
331        terms: ndarray::Array2<f64>,
332    }
333
334    /// Every `POINTING` row of `path`, in table order.
335    fn table_rows(path: &std::path::Path) -> Vec<Row> {
336        let ms = MeasurementSet::open(path).expect("ms");
337        let pointing = ms.pointing().expect("POINTING");
338        (0..pointing.row_count())
339            .map(|row| {
340                let ArrayValue::Float64(terms) =
341                    pointing.array(row, "DIRECTION").expect("DIRECTION").clone()
342                else {
343                    panic!("DIRECTION is Float64");
344                };
345                Row {
346                    antenna: pointing.i32(row, "ANTENNA_ID").expect("ANTENNA_ID"),
347                    time_s: pointing.f64(row, "TIME").expect("TIME"),
348                    interval_s: pointing.f64(row, "INTERVAL").expect("INTERVAL"),
349                    origin_s: pointing.f64(row, "TIME_ORIGIN").expect("TIME_ORIGIN"),
350                    terms: terms.into_dimensionality().expect("[2, terms]"),
351                }
352            })
353            .collect()
354    }
355
356    /// `row`'s direction at `time_s`: its polynomial about `TIME_ORIGIN`.
357    fn direction_at(row: &Row, time_s: f64) -> [f64; 2] {
358        let dt = time_s - row.origin_s;
359        [0, 1].map(|axis| {
360            row.terms
361                .row(axis)
362                .iter()
363                .rev()
364                .fold(0.0, |sum, term| sum * dt + term)
365        })
366    }

368    /// `antenna`'s bracket at `time_s` read from `rows` one by one, as the
369    /// boundary defines it ([`casa_observation::PointingBracket`]): the row
370    /// whose interval holds the time, nearest it and then first in time
371    /// order; the first row of the latest time at or before it; and the
372    /// first row at or after it. Each is its time and its direction there.
373    fn table_bracket(rows: &[Row], antenna: i32, time_s: f64) -> [Option<(f64, [f64; 2])>; 3] {
374        let mut own: Vec<&Row> = rows.iter().filter(|row| row.antenna == antenna).collect();
375        own.sort_by(|left, right| left.time_s.total_cmp(&right.time_s));
376        let covering = own
377            .iter()
378            .filter(|row| {
379                row.interval_s > 0.0 && (row.time_s - time_s).abs() <= 0.5 * row.interval_s
380            })
381            .min_by(|left, right| {
382                (left.time_s - time_s)
383                    .abs()
384                    .total_cmp(&(right.time_s - time_s).abs())
385            });
386        let latest = own
387            .iter()
388            .rfind(|row| row.time_s <= time_s)
389            .map(|row| row.time_s);
390        let before = latest.and_then(|latest| own.iter().find(|row| row.time_s == latest));
391        let after = own.iter().find(|row| row.time_s >= time_s);
392        [covering, before, after].map(|row| row.map(|row| (row.time_s, direction_at(row, time_s))))
393    }

395    /// The boundary's MSv2 pointing brackets each (antenna, time) as the
396    /// table states it, read row by row: the same covering, preceding and
397    /// following rows, evaluated at the time to the same directions. The
398    /// rows here are in time order for each antenna.
399    #[test]
400    fn brackets_are_the_tables_rows() {
401        let root = tempfile::tempdir().expect("root");
402        let path = fixture(root.path());
403        let observation = open(&path, &OpenOptions::default()).expect("open");
404        let grid = observation.grid(PartitionId(0), "base").expect("grid");
405        let mut queries: Vec<f64> = grid
406            .times_s
407            .iter()
408            .flat_map(|time| [time - 7.0, time - 5.0, *time, time + 3.0, time + 5.0])
409            .collect();
410        queries.extend([
411            grid.times_s[0] - 30.0,
412            grid.times_s[grid.times_s.len() - 1] + 30.0,
413        ]);
414        queries.sort_by(f64::total_cmp);
415        let antennas: Vec<AntennaId> = (0..3).map(AntennaId).collect();
416        let series = observation
417            .pointing(&PointingRequest {
418                partition: PartitionId(0),
419                direction: PointingDirection::Beam,
420                antennas: antennas.clone(),
421                times: queries[0]..=queries[queries.len() - 1],
422                budget_bytes: 1 << 20,
423            })
424            .expect("pointing");
425        assert_eq!(series.terms, 2);
426        let rows = table_rows(&path);
427        let mut covered = [0; 3];
428        for antenna in antennas {
429            for time in &queries {
430                let boundary = series.bracket(antenna, *time).expect("boundary");
431                let boundary = [boundary.covering, boundary.before, boundary.after]
432                    .map(|candidate| candidate.map(|c| (c.time_s, c.direction_rad)));
433                let table = table_bracket(&rows, antenna.0 as i32, *time);
434                for (got, want) in boundary.iter().zip(&table) {
435                    match (got, want) {
436                        (Some((got_time, got)), Some((want_time, want))) => {
437                            assert_eq!(got_time, want_time, "{antenna:?} at {time}");
438                            for (got, want) in got.iter().zip(want) {
439                                assert!(
440                                    (got - want).abs() <= 1e-15,
441                                    "{antenna:?} at {time}: {got} against {want}"
442                                );
443                            }
444                        }
445                        (None, None) => {}
446                        _ => panic!("{antenna:?} at {time}: {boundary:?} against {table:?}"),
447                    }
448                }
449                covered[antenna.0 as usize] += usize::from(table[0].is_some());
450            }
451        }
452        assert!(
453            covered[0] > 0 && covered[2] > 0,
454            "rows cover queries: {covered:?}"
455        );
456        assert_eq!(covered[1], 0, "rows of no interval cover nothing");
457    }
