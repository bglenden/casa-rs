55/// A new CASA CompressComplex column and its completed backing data.
56pub struct CompressedComplexArray<'a> {
57    /// Descriptor for the logical Complex32 array.
58    pub schema: ColumnSchema,
59    /// Name of the StandardStMan scale and offset manager.
60    pub scalar_manager: &'a str,
61    /// Name of the TiledShapeStMan packed-cell manager.
62    pub tiled_manager: &'a str,
63    /// Packed Int32 cells in table row order.
64    pub packed: crate::StreamedTiledShapeColumn,
65    /// Per-row floating-point scales.
66    pub scales: Vec<Option<ScalarValue>>,
67    /// Per-row floating-point offsets.
68    pub offsets: Vec<Option<ScalarValue>>,
69}

71impl CompressedComplexArray<'_> {
72    fn groups(
73        &self,
74    ) -> [(
75        Vec<ColumnAddition>,
76        Option<ManagerAddition>,
77        ColumnGroupPayload<'_>,
78    ); 3] {
79        let logical = self.schema.name();
80        let stored = format!("{logical}_COMPRESSED");
81        let scale = format!("{logical}_SCALE");
82        let offset = format!("{logical}_OFFSET");
83        let add = |schema, keywords| ColumnAddition {
84            schema,
85            default: None,
86            keywords,
87            manager_type: None,
88            manager_group: None,
89        };
90        let group = |kind: &str, name: &str| {
91            Some(ManagerAddition {
92                kind: kind.to_owned(),
93                name: name.to_owned(),
94                spec: RecordValue::default(),
95            })
96        };
97        let scalars = [&offset, &scale].map(|column| {
98            add(
99                ColumnSchema::scalar(column, PrimitiveType::Float32),
100                RecordValue::default(),
101            )
102        });
103        let packed = ColumnSchema::array_variable(&stored, PrimitiveType::Int32, Some(2));
104        let keywords =
105            crate::storage::virtual_compress::automatic_complex_keywords(&stored, &scale, &offset);
106        [
107            (
108                scalars.into(),
109                group("StandardStMan", self.scalar_manager),
110                ColumnGroupPayload::Scalars([&self.offsets, &self.scales]),
111            ),
112            (
113                vec![add(packed, RecordValue::default())],
114                group("TiledShapeStMan", self.tiled_manager),
115                ColumnGroupPayload::Tiled(&self.packed),
116            ),
117            (
118                vec![add(self.schema.clone(), keywords)],
119                group("CompressComplex", logical),
120                ColumnGroupPayload::CompressedEngine,
121            ),
122        ]
123    }
124}

126/// A staged edit to one physical table, holding its write lock.
127pub struct PreparedTableEdit<'lock> {
128    edit: ReversibleEdit,
129    lock: &'lock mut TableWriteLock,
130    original_sync: WriteSyncCheckpoint,
131    attempted_sync: bool,
132}
133
134impl PreparedTableEdit<'_> {
135    /// Write the staged changes, retaining the original files for rollback.
136    ///
137    /// # Errors
138    /// The table identity or one of the table writes failed. Call
139    /// [`Self::restore`] to undo any partial updates before releasing the lock.
140    pub fn publish(&mut self) -> Result<(), TableError> {
141        if !self.lock.matches_table_path(self.lock.path()) {
142            return Err(TableError::Storage(format!(
143                "{} was replaced before publication",
144                self.lock.path().display()
145            )));
146        }
147        self.lock.record_write();
148        self.edit.apply().map_err(Into::into)
149    }
150
151    /// Roll back this edit and discard its staging files on success.
152    ///
153    /// # Errors
154    /// On recovery failure, returns the path to the retained backups.
155    pub fn restore(mut self) -> Result<(), TableError> {
156        if !self.lock.matches_table_path(self.lock.path()) {
157            return Err(TableError::Storage(format!(
158                "{} was replaced before rollback. Original files staged at {}",
159                self.lock.path().display(),
160                self.edit.preserve().display(),
161            )));
162        }
163        if let Err(error) = self.edit.restore() {
164            return Err(TableError::Storage(format!(
165                "{error}; original table files retained at {}",
166                self.edit.preserve().display()
167            )));
168        }
169        self.lock
170            .restore_sync(self.original_sync, self.attempted_sync)
171    }
172
173    /// Announce the new shape to casacore while keeping the lock and backups.
174    ///
175    /// Retain this append until every edit in the batch has been announced.
176    /// Dropping the append then discards its backups.
177    ///
178    /// # Errors
179    /// Returns an error if the write could not be announced to other readers.
180    pub fn synchronize(&mut self) -> Result<(), TableError> {
181        self.synchronize_with(TableWriteLock::sync_recorded_write)
182    }
183
184    fn synchronize_with(
185        &mut self,
186        sync: impl FnOnce(&mut TableWriteLock) -> Result<(), TableError>,
187    ) -> Result<(), TableError> {
188        self.attempted_sync = true;
189        sync(self.lock)
190    }
191}

193impl Table {
194    /// Append rows to a plain disk table without changing its storage
195    /// manager bindings. Standard and AipsIO managers use `defaults`;
196    /// incremental managers inherit their preceding values and variable
197    /// tiled arrays remain undefined.
198    ///
199    /// # Errors
200    ///
201    /// The table has no durable plain layout, has unsaved edits, contains
202    /// a storage manager that cannot append, or cannot be locked or written.
203    pub fn append_rows_in_place(
204        &mut self,
205        count: usize,
206        defaults: &RecordValue,
207    ) -> Result<(), TableError> {
208        if count == 0 {
209            return Ok(());
210        }
211        self.edit_rows_in_place(Rows::Append { count, defaults }, true)
212    }
213
214    /// Append distinct records in a single edit, preserving the table's
215    /// existing manager bindings. Standard, AipsIO and incremental managers
216    /// accept explicit record values. Omitted cells keep the manager's
217    /// normal append defaults, including inherited incremental values.
218    ///
219    /// # Errors
220    /// The table cannot be locked or written, a row disagrees with its
221    /// schema, or a manager cannot append explicit records.
222    pub fn append_records_in_place(&mut self, records: &[RecordValue]) -> Result<(), TableError> {
223        self.append_records(records, true)
224    }
225
226    /// Append records using [`Self::append_records_in_place`]'s persistence
227    /// rules and consume the handle. The written layout is available to a
228    /// fresh reader without reloading this handle first.
229    ///
230    /// # Errors
231    /// As in [`Self::append_records_in_place`].
232    pub fn append_records_and_close(mut self, records: &[RecordValue]) -> Result<(), TableError> {
233        self.append_records(records, false)
234    }
235
236    fn append_records(&mut self, records: &[RecordValue], reload: bool) -> Result<(), TableError> {
237        if records.is_empty() {
238            return Ok(());
239        }
240        let defaults = self.record_append_defaults(records)?;
241        self.edit_rows_in_place(
242            Rows::AppendRecords {
243                records,
244                defaults: &defaults,
245                encoded: None,
246                existing_scalar: None,
247            },
248            reload,
249        )
250    }
251
252    /// Stage distinct appended records without changing existing table files.
253    ///
254    /// The caller must open this table after acquiring its write lock. Use
255    /// [`PreparedTableEdit::publish`] to apply the changes and keep the
256    /// prior files while other operations in the batch are published.
257    ///
258    /// # Errors
259    /// The table or lock is invalid, the schema cannot admit the rows,
260    /// or a manager cannot prepare append changes.
261    pub fn prepare_record_append<'lock>(
262        &self,
263        records: &[RecordValue],
264        lock: &'lock mut TableWriteLock,
265    ) -> Result<PreparedTableEdit<'lock>, TableError> {
266        self.prepare_record_append_with_options(records, RecordAppendOptions::default(), lock)
267    }

269    /// Identify compatible tiled array columns for a whole-donor append.
270    ///
271    /// Open the donor and target under their read and write locks and retain
272    /// both locks through staging. Only supply names whose cells can be
273    /// copied without transformation. Read the remaining cells normally.
274    ///
275    /// # Errors
276    /// The donor or target is unpersisted, or its headers cannot be read.
277    pub fn plan_encoded_array_append(
278        &self,
279        donor: &Self,
280        columns: &[&str],
281    ) -> Result<EncodedArrayAppend, TableError> {
282        let target = self.path().ok_or(TableError::PreparedRowRequiresSchema)?;
283        let incoming = donor.path().ok_or(TableError::PreparedRowRequiresSchema)?;
284        if columns.is_empty() || self.kind != TableKind::Plain || donor.kind != TableKind::Plain {
285            return Ok(EncodedArrayAppend(ShapeAppend::empty(
286                target,
287                self.row_count(),
288                donor.row_count(),
289            )));
290        }
291        Ok(EncodedArrayAppend(ShapeAppend::plan(
292            target, incoming, columns,
293        )?))
294    }
295
296    /// Stage distinct appended records with additional atomic edits. Records
297    /// omit columns already supplied by the encoded donor plan.
298    ///
299    /// # Errors
300    /// As in [`Self::prepare_record_append`], or the options conflict with
301    /// the target schema, current storage, or supplied records.
302    pub fn prepare_record_append_with_options<'lock>(
303        &self,
304        records: &[RecordValue],
305        options: RecordAppendOptions<'_>,
306        lock: &'lock mut TableWriteLock,
307    ) -> Result<PreparedTableEdit<'lock>, TableError> {
308        if records.is_empty() {
309            return Err(TableError::Storage("no records to append".into()));
310        }
311        let (path, mut dat, old) = self.staged_edit_snapshot(lock)?;
312        let encoded = options.encoded_arrays.map(|encoded| &encoded.0);
313        if let Some(encoded) = encoded {
314            encoded.validate(path, old, records.len())?;
315        }
316        if let Some((name, value)) = options.existing_scalar {
317            let column = self
318                .schema()
319                .and_then(|schema| schema.column(name))
320                .ok_or_else(|| TableError::SchemaColumnUnknown {
321                    column: name.into(),
322                })?;
323            super::columns::validate_cell_against_schema_column(
324                0,
325                column,
326                Some(&Value::Scalar(value.clone())),
327            )?;
328        }
329        if let Some((name, key, value)) = options.column_keyword {
330            let column = dat
331                .table_desc
332                .columns
333                .iter_mut()
334                .find(|column| column.col_name == name)
335                .ok_or_else(|| TableError::SchemaColumnUnknown {
336                    column: name.into(),
337                })?;
338            column.keywords.upsert(key, value.clone());
339        }
340        let defaults = self.record_append_defaults(records)?;
341        let rows = Rows::AppendRecords {
342            records,
343            defaults: &defaults,
344            encoded,
345            existing_scalar: options.existing_scalar,
346        };
347        let edit = StagedEdit::stage(path, dat, old, rows)?.prepare_publication(true)?;
348        let original_sync = lock.checkpoint_sync()?;
349        Ok(PreparedTableEdit {
350            edit,
351            lock,
352            original_sync,
353            attempted_sync: false,
354        })
355    }

357    /// Stage replacements for existing physical columns without changing
358    /// the row count or unrelated storage managers. The input vectors are
359    /// in table row order. Column keywords can be included in the same edit.
360    ///
361    /// # Errors
362    /// The table cannot be locked, a column or keyword target is unknown, a
363    /// cell disagrees with the schema, or its storage cannot be rewritten.
364    pub fn prepare_column_replacement<'lock>(
365        &self,
366        columns: &std::collections::BTreeMap<String, Vec<Option<Value>>>,
367        column_keyword: Option<(&str, &str, &Value)>,
368        lock: &'lock mut TableWriteLock,
369    ) -> Result<PreparedTableEdit<'lock>, TableError> {
370        let (path, mut dat, old) = self.staged_edit_snapshot(lock)?;
371        let schema = self.schema().ok_or(TableError::PreparedRowRequiresSchema)?;
372        for (name, values) in columns {
373            let column = schema
374                .column(name)
375                .ok_or_else(|| TableError::SchemaColumnUnknown {
376                    column: name.clone(),
377                })?;
378            if values.len() as u64 != old {
379                return Err(TableError::Storage(format!(
380                    "column {name} has {} replacement cells for {old} rows",
381                    values.len()
382                )));
383            }
384            for (row, value) in values.iter().enumerate() {
385                super::columns::validate_cell_against_schema_column(row, column, value.as_ref())?;
386            }
387        }
388        if let Some((name, key, value)) = column_keyword {
389            let column = dat
390                .table_desc
391                .columns
392                .iter_mut()
393                .find(|column| column.col_name == name)
394                .ok_or_else(|| TableError::SchemaColumnUnknown {
395                    column: name.into(),
396                })?;
397            column.keywords.upsert(key, value.clone());
398        }
399        let rows = Rows::ReplaceColumns(columns);
400        let edit = StagedEdit::stage(path, dat, old, rows)?.prepare_publication(true)?;
401        let original_sync = lock.checkpoint_sync()?;
402        Ok(PreparedTableEdit {
403            edit,
404            lock,
405            original_sync,
406            attempted_sync: false,
407        })
408    }

410    fn staged_edit_snapshot(
411        &self,
412        lock: &TableWriteLock,
413    ) -> Result<(&Path, crate::storage::TableDatContents, u64), TableError> {
414        let path = self.path().ok_or(TableError::PreparedRowRequiresSchema)?;
415        if self.kind != TableKind::Plain
416            || self.holds_unflushed_changes()
417            || !lock.matches_table_path(path)
418        {
419            return Err(TableError::Storage(
420                "preparing an edit requires a clean, locked plain table".into(),
421            ));
422        }
423        let dat = crate::storage::table_control::read_table_dat(
424            &path.join(crate::storage::TABLE_CONTROL_FILE),
425        )?;
426        let old = self.require_persisted_rows(path, &dat)?;
427        Ok((path, dat, old))
428    }

430    fn record_append_defaults(&self, records: &[RecordValue]) -> Result<RecordValue, TableError> {
431        let schema = self.schema().ok_or(TableError::PreparedRowRequiresSchema)?;
432        for (offset, row) in records.iter().enumerate() {
433            let row_index = self.row_count() + offset;
434            for field in row.fields() {
435                let column = schema.column(&field.name).ok_or_else(|| {
436                    TableError::RowContainsUnknownColumn {
437                        row_index,
438                        column: field.name.clone(),
439                    }
440                })?;
441                super::columns::validate_cell_against_schema_column(
442                    row_index,
443                    column,
444                    Some(&field.value),
445                )?;
446            }
447        }
448        Ok(schema.default_row())
449    }

451    /// Keep the given rows in an existing plain table, in ascending order,
452    /// and delete all others. This operation preserves manager bindings.
453    ///
454    /// # Errors
455    ///
456    /// The row sequence is unsorted or contains repetitions or invalid row
457    /// numbers; the table cannot be written or has a manager that refuses
458    /// to remove rows (for example, IncrementalStMan).
459    pub fn retain_rows_in_place(&mut self, kept: &[usize]) -> Result<(), TableError> {
460        if kept.windows(2).any(|pair| pair[0] >= pair[1]) {
461            return Err(TableError::Storage(
462                "retained rows must strictly increase".into(),
463            ));
464        }
465        self.edit_rows_in_place(Rows::Keep(kept), true)
466    }

468    /// Add columns without changing the files for managers that do not
469    /// receive a column. When supplied, `manager` binds all the new
470    /// columns to one new manager. Otherwise the first manager that
471    /// accepts addition supplies the storage, as in casacore.
472    ///
473    /// # Errors
474    ///
475    /// The table is not a clean plain table, a column or manager name
476    /// already exists, the selected binding is unsupported, or a storage
477    /// file cannot be read or written.
478    pub fn add_columns_in_place(
479        &mut self,
480        columns: Vec<ColumnAddition>,
481        manager: Option<ManagerAddition>,
482    ) -> Result<(), TableError> {
483        if columns.is_empty() {
484            return Ok(());
485        }
486        self.edit_clean_layout("add columns in place", move |this, path, mut dat, old| {
487            dat.raise_row_count(old);
488            let plan =
489                StagedEdit::stage_columns(path, dat, columns, manager, this.data_manager_info())?;
490            this.note_persisted_write(path);
491            plan.publish()?;
492            this.reload_disk_state_at(path)
493        })
494    }

496    /// Install complete CASA CompressComplex arrays with their packed cells
497    /// and automatic per-row scale and offset. Stage all requested groups
498    /// before publishing a single descriptor update.
499    ///
500    /// # Errors
501    /// A binding conflicts with the current table, the data do not match
502    /// the requested columns, or a storage file cannot be published.
503    pub fn add_compressed_complex_arrays_in_place(
504        &mut self,
505        arrays: &[CompressedComplexArray<'_>],
506    ) -> Result<(), TableError> {
507        if arrays.iter().any(|array| {
508            array.schema.data_type() != Some(PrimitiveType::Complex32)
509                || [
510                    array.packed.row_count(),
511                    array.scales.len(),
512                    array.offsets.len(),
513                ]
514                .into_iter()
515                .any(|count| count != self.row_count())
516        }) {
517            return Err(TableError::Storage(
518                "compressed complex layout requires one packed cell, scale and offset per row"
519                    .into(),
520            ));
521        }
522        self.edit_clean_layout(
523            "add compressed arrays in place",
524            move |this, path, mut dat, old| {
525                dat.raise_row_count(old);
526                let groups = arrays.iter().flat_map(CompressedComplexArray::groups);
527                let plan =
528                    StagedEdit::stage_column_groups(path, dat, groups, this.data_manager_info())?;
529                this.note_persisted_write(path);
530                plan.publish()?;
531                this.reload_disk_state_at(path)
532            },
533        )
534    }
535
536    fn edit_rows_in_place(&mut self, rows: Rows<'_>, reload: bool) -> Result<(), TableError> {
537        self.edit_clean_layout("edit rows in place", |this, path, dat, old| {
538            if let Rows::Keep(kept) = rows {
539                if kept.last().is_some_and(|&row| row as u64 >= old) {
540                    return Err(TableError::Storage("retained row is out of range".into()));
541                }
542                if kept.len() as u64 == old {
543                    return Ok(());
544                }
545            }
546            let plan = StagedEdit::stage(path, dat, old, rows)?;
547            this.note_persisted_write(path);
548            plan.publish()?;
549            if reload {
550                // Publish the count before reopening; the old lock sync
551                // count otherwise hides removals.
552                this.note_full_save(path)?;
553                this.reload_disk_state_at(path)
554            } else {
555                // The handle is consumed. Its lock still announces the
556                // persisted row count when released.
557                this.mark_flushed();
558                Ok(())
559            }
560        })
561    }
562
563    fn edit_clean_layout(
564        &mut self,
565        operation: &str,
566        edit: impl FnOnce(
567            &mut Self,
568            &Path,
569            crate::storage::TableDatContents,
570            u64,
571        ) -> Result<(), TableError>,
572    ) -> Result<(), TableError> {
573        let path = self.source_path.clone().ok_or_else(|| {
574            TableError::Storage("in-place row edits require an existing table".into())
575        })?;
576        if self.kind != TableKind::Plain || self.holds_unflushed_changes() {
577            return Err(TableError::Storage(
578                "in-place row edits require a clean plain table".into(),
579            ));
580        }
581        let auto_unlock = self.begin_write_operation(operation)?;
582        let result = (|| {
583            #[cfg(unix)]
584            if auto_unlock {
585                self.reload_if_published()?;
586            }
587            let dat = crate::storage::table_control::read_table_dat(
588                &path.join(crate::storage::TABLE_CONTROL_FILE),
589            )?;
590            let old = self.require_persisted_rows(&path, &dat)?;
591            edit(self, &path, dat, old)
592        })();
593        self.finish_write_operation(auto_unlock, result)
594    }

672            let mut guard = TableWriteLock::acquire_to_reread(&path, 1).unwrap();
673            let table = Table::open(TableOptions::new(&path)).unwrap();
674            let before = table.scalar_columns_owned(&["id", "keep"]).unwrap();
675            let manager_file = path.join("table.f0");
676            let original = std::fs::read(&manager_file).unwrap();
677            let replacement = BTreeMap::from([(
678                "id".to_owned(),
679                (0..rows)
680                    .map(|id| Some(Value::Scalar(ScalarValue::Int32(id ^ 1))))
681                    .collect(),
682            )]);
683            let mut staged = table
684                .prepare_column_replacement(&replacement, None, &mut guard)
685                .unwrap();
686            staged.publish().unwrap();
687            staged.synchronize().unwrap();
688            let reopen = || {
689                Table::open(TableOptions::new(&path))
690                    .unwrap()
691                    .scalar_columns_owned(&["id", "keep"])
692                    .unwrap()
693            };
694            let after = reopen();
695            assert_eq!(
696                after["id"],
697                (0..rows)
698                    .map(|id| Some(ScalarValue::Int32(id ^ 1)))
699                    .collect::<Vec<_>>(),
700            );
701            assert_eq!(after["keep"], before["keep"]);
702            if manager == DataManagerKind::StandardStMan {
703                let edited = std::fs::read(&manager_file).unwrap();
704                assert_eq!(edited.len(), original.len());
705                assert_eq!(
706                    edited.iter().zip(&original).filter(|(a, b)| a != b).count(),
707                    rows as usize
708                );
709            }
710            staged.restore().unwrap();
711            assert_eq!(reopen(), before);
712            assert_eq!(std::fs::read(&manager_file).unwrap(), original);
713        }
714    }

719        let first = root.path().join("first");

724            a_lock.modify_counter().unwrap(),
725            b_lock.modify_counter().unwrap(),

728        let mut edited_b = b.prepare_record_append(&[record(9)], &mut b_lock).unwrap();
729        edited_a.publish().unwrap();
730        edited_b.publish().unwrap();
731        edited_a.synchronize().unwrap();
732        assert!(
733            edited_b
734                .synchronize_with(|lock| {
735                    lock.sync_recorded_write()?;
736                    Err(TableError::Storage("injected sync failure".into()))
737                })
738                .is_err()
739        );
740        edited_b.restore().unwrap();
741        edited_a.restore().unwrap();
742
743        for (path, lock, old_counter) in [
744            (&first, a_lock, counters[0]),
745            (&second, b_lock, counters[1]),
746        ] {

748                .unwrap()
749                .unwrap();

753            let reopened = Table::open(TableOptions::new(path)).unwrap();
754            assert_eq!(reopened.rows().unwrap(), &[record(4)]);
