3//! Thin PyO3 `ia.addnoise` wrapper delegating to [`casa_imageanalysis::tool::ops_edit::add_noise`].
4
5use casa_imageanalysis::tool::ops_edit::add_noise;
6use pyo3::prelude::*;
7use pyo3::types::{PyDict, PyString};
8
9use crate::convert::Variant;
10
11use super::PyImageTool;
12use super::handle::raised;
13use super::subimage::Region;
14
15mod arguments;
16use arguments::{NoiseParameters, NoiseSeeds};
17
18#[pymethods]
19impl PyImageTool {
20    #[pyo3(signature = (r#type="normal", pars=NoiseParameters(vec![0.0, 1.0]), region=None, zero=false, seeds=NoiseSeeds(Vec::new())), text_signature = "($self, type='normal', pars=[0.0, 1.0], region={}, zero=False, seeds=[])")]
21    fn addnoise(
22        &self,
23        py: Python<'_>,
24        r#type: &str,
25        pars: NoiseParameters,
26        region: Option<&Bound<'_, PyAny>>,
27        zero: bool,
28        seeds: NoiseSeeds,
29    ) -> PyResult<bool> {
30        if py.detach(|| self.state().image.is_none()) {
31            return Ok(false);
32        }
33        let region_history = match region {
34            Some(value) if value.cast::<PyString>().is_ok() => {
35                let text = value.cast::<PyString>()?.to_str()?;
36                format!("\"{}\"", shorten(text))
37            }
38            Some(value)
39                if !value.is_instance_of::<PyDict>()
40                    && value.extract::<Variant>()?.is_empty_sequence() =>
41            {
42                "[]".into()
43            }
44            _ => "{...}".into(),
45        };
46        let region = Region::argument(py, region, "addnoise")?;
47        let seeds = seeds.0;
48        py.detach(|| {
49            let state = self.state();
50            let Some(attachment) = &state.image else {
51                return Ok(false);
52            };
53            let history_command = state.history.then(|| {
54                let parameters = pars
55                    .0
56                    .iter()
57                    .map(|value| {
58                        if value.is_nan() {
59                            "nan".into()
60                        } else {
61                            format!("{value:.6}")
62                        }
63                    })
64                    .collect::<Vec<_>>()
65                    .join(",");
66                let parameters = format!("[{parameters}]");
67                let seed_text = format!(
68                    "[{}]",
69                    seeds
70                        .iter()
71                        .map(i64::to_string)
72                        .collect::<Vec<_>>()
73                        .join(",")
74                );
75                format!(
76                    "ia.addnoise(type=\"{}\", pars={}, region={region_history}, zeroit={zero}, seeds={})",
77                    shorten(r#type),
78                    shorten(&parameters),
79                    shorten(&seed_text),
80                )
81            });
82            add_noise(
83                attachment,
84                r#type,
85                &pars.0,
86                &seeds,
87                region.as_ref().map(Region::as_region),
88                zero,
89                history_command,
90            )
91            .map_err(raised)
92        })
93    }
94}
95
96fn shorten(value: &str) -> &str {
97    if value.len() > 300 { "..." } else { value }
