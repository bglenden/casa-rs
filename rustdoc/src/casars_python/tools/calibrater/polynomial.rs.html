26fn output_mode(append: Option<&Bound<'_, PyAny>>, method: &str) -> PyResult<CalibrationOutputMode> {
27    let append = append
28        .map(|value| strict_bool(value, method, 3))
29        .transpose()?
30        .unwrap_or(false);
31    Ok(if append {
32        CalibrationOutputMode::Append
33    } else {
34        CalibrationOutputMode::Replace
35    })
36}

38fn integer(
39    value: Option<&Bound<'_, PyAny>>,
40    method: &str,
41    position: u8,
42    default: i64,
43) -> PyResult<i64> {
44    let Some(value) = value else {
45        return Ok(default);
46    };
47    value.extract().map_err(|_| {
48        PyTypeError::new_err(format!(
49            "in method 'calibrater_{method}', argument {position} of type 'long'"
50        ))
51    })
52}

54impl PyCalibrater {
55    fn configure_polynomial(
56        &self,
57        py: Python<'_>,
58        reference: RefAntSelector,
59        configure: impl FnOnce(&mut CalibraterLifecycle, RefAntSelector) -> PyResult<()> + Send,
60    ) -> PyResult<bool> {
61        let configured = self.with_lifecycle(py, |state| {
62            if state.dataset_path().is_none() {
63                return Ok(false);
64            }
65            let reference = state
66                .resolve_primary_tool_refant(reference)
67                .map_err(super::runtime_error)?;
68            state.reset(false, true).map_err(super::runtime_error)?;
69            configure(state, reference)?;
70            Ok(true)
71        })?;
72        if !configured {
73            emit(
74                py,
75                LOGGER,
76                &[LogLine::error("Must first open a MeasurementSet.")],
77            )?;
78        }
79        Ok(configured)
80    }
81}

83#[pymethods]
84#[allow(clippy::too_many_arguments)]
85impl PyCalibrater {
86    #[pyo3(signature = (table = None, append = None, t = None, combine = None, degamp = None, degphase = None, visnorm = None, solnorm = None, maskcenter = None, maskedge = None, refant = None), text_signature = "($self, table='', append=False, t=[], combine='', degamp=3, degphase=3, visnorm=False, solnorm=True, maskcenter=0, maskedge=5.0, refant=[])")]
87    fn setsolvebandpoly(
88        &self,
89        py: Python<'_>,
90        table: Option<&Bound<'_, PyAny>>,
91        append: Option<&Bound<'_, PyAny>>,
92        t: Option<&Bound<'_, PyAny>>,
93        combine: Option<&Bound<'_, PyAny>>,
94        degamp: Option<&Bound<'_, PyAny>>,
95        degphase: Option<&Bound<'_, PyAny>>,
96        visnorm: Option<&Bound<'_, PyAny>>,
97        solnorm: Option<&Bound<'_, PyAny>>,
98        maskcenter: Option<&Bound<'_, PyAny>>,
99        maskedge: Option<&Bound<'_, PyAny>>,
100        refant: Option<&Bound<'_, PyAny>>,
101    ) -> PyResult<bool> {
102        let method = "setsolvebandpoly";
103        let append = output_mode(append, method)?;
104        let table = PathBuf::from(text(table.cloned(), "table", "")?);
105        let elapsed = listing_selector(t)?;
106        let combine = GainSolveCombine::from_casa(&text(combine.cloned(), "combine", "")?);
107        let degrees = [
108            integer(degamp, method, 6, 3)?,
109            integer(degphase, method, 7, 3)?,
110        ];
111        let visnorm = visnorm
112            .map(|value| strict_bool(value, method, 8))
113            .transpose()?
114            .unwrap_or(false);
115        let solnorm = solnorm
116            .map(|value| strict_bool(value, method, 9))
117            .transpose()?
118            .unwrap_or(true);
119        let maskcenter = integer(maskcenter, method, 10, 0)?.max(0) as usize;
120        let maskedge = number(maskedge.cloned(), method, 11, 5.)? as f32;
121        let reference = self::refant(refant, "flex")?;
122        let display_reference = reference.clone();
123        self.configure_polynomial(py, reference, move |state, reference| {
124            if table.as_os_str().is_empty() || degrees.iter().any(|&degree| degree < -1) {
125                return Err(setup_error());
126            }
127            let (time, frequency) = elapsed.split_once(',').unwrap_or((&elapsed, ""));
128            let request = CalibrationSolveRequest::Bandpass(BandpassSolveRequest {
129                selection: Default::default(),
130                output_table: table,
131                refant: reference,
132                prior_calibration_tables: Vec::new(),
133                parang: false,
134                combine: BandpassSolveCombine {
135                    scans: combine.scans,
136                    fields: combine.fields,
137                    observations: combine.observations,
138                    spectral_windows: combine.spectral_windows,
139                },
140                band_type: BandpassType::BPoly,
141                normalize_average_amplitude: solnorm,
142                amplitude_degree: degrees[0].max(0) as usize,
143                phase_degree: degrees[1].max(0) as usize,
144                model_source: GainSolveModelSource::FromMeasurementSet,
145                solve_interval: GainSolveInterval::from_casa(time).ok_or_else(setup_error)?,
146                frequency_interval: BandpassFrequencyInterval::from_casa(frequency)
147                    .ok_or_else(setup_error)?,
148                correlation_dependent_flags: false,
149                smodel: [1., 0., 0., 0.],
150                cuts: Some(BandpassSolveCuts {
151                    mask_center_channels: maskcenter,
152                    mask_edge_percent: maskedge,
153                    normalize_visibilities: visnorm,
154                    omit_polynomial: degrees.map(|degree| degree == -1),
155                    ..BandpassSolveCuts::CASA_DEFAULTS
156                }),
157            });
158            state
159                .set_solve(request, append)
160                .map_err(super::runtime_error)?;
161            state.record_solve_description(&elapsed, "", "flex", Some(&display_reference));
162            Ok(())
163        })
164    }
165
166    #[pyo3(signature = (table = None, append = None, mode = None, splinetime = None, preavg = None, npointaver = None, phasewrap = None, refant = None), text_signature = "($self, table='', append=False, mode='PHAS', splinetime=10800.0, preavg=0.0, npointaver=10, phasewrap=250.0, refant=[])")]
167    fn setsolvegainspline(
168        &self,
169        py: Python<'_>,
170        table: Option<&Bound<'_, PyAny>>,
171        append: Option<&Bound<'_, PyAny>>,
172        mode: Option<&Bound<'_, PyAny>>,
173        splinetime: Option<&Bound<'_, PyAny>>,
174        preavg: Option<&Bound<'_, PyAny>>,
175        npointaver: Option<&Bound<'_, PyAny>>,
176        phasewrap: Option<&Bound<'_, PyAny>>,
177        refant: Option<&Bound<'_, PyAny>>,
178    ) -> PyResult<bool> {
179        let method = "setsolvegainspline";
180        let append = output_mode(append, method)?;
181        let table = PathBuf::from(text(table.cloned(), "table", "")?);
182        let mode = text(mode.cloned(), "mode", "PHAS")?.to_ascii_uppercase();
183        let preavg = number(preavg.cloned(), method, 6, 0.)?;
184        let controls = GainSplineControls {
185            splinetime: number(splinetime.cloned(), method, 5, 10800.)?,
186            npointaver: integer(npointaver, method, 7, 10)?,
187            phasewrap: number(phasewrap.cloned(), method, 8, 250.)?,
188        };
189        let reference = self::refant(refant, "flex")?;
190        let display_reference = reference.clone();
191        self.configure_polynomial(py, reference, move |state, reference| {
192            if table.as_os_str().is_empty() {
193                return Err(setup_error());
194            }
195            state
196                .set_tool_spline(
197                    GainSolveRequest {
198                        selection: Default::default(),
199                        output_table: table,
200                        gain_type: GainType::GSpline,
201                        solve_mode: GainSolveMode::AmplitudePhase,
202                        solve_interval: GainSolveInterval::Infinite,
203                        preaverage_seconds: preavg,
204                        correlation_dependent_flags: false,
205                        combine: Default::default(),
206                        refant: reference,
207                        prior_calibration_tables: Vec::new(),
208                        parang: false,
209                        model_source: GainSolveModelSource::FromMeasurementSet,
210                        normalize_average_amplitude: false,
211                        normalization_type: Default::default(),
212                        robustness: Default::default(),
213                        min_snr: 0.,
214                        min_baselines_per_antenna: 0,
215                        smodel: [1., 0., 0., 0.],
216                    },
217                    controls,
218                    append,
219                    mode.clone(),
220                )
221                .map_err(super::runtime_error)?;
222            state.record_solve_description("inf", &mode, "flex", Some(&display_reference));
223            Ok(())
224        })
225    }
