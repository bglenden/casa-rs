2//! Writing a streamed tiled column's bands of tiles on a thread of their
3//! own, so the thread that lays rows into tiles never waits on the file.
4//!
5//! A band is every tile of one row of tiles, written in file order. A
6//! writer that holds two bands fills one while this thread writes the
7//! other, and swaps a filled band for the band written before it; one that
8//! holds one band, when memory is short, waits for it to be written. Writing copies a band into the page cache at the rate the
9//! system allows (about 3.3 GB/s on an M4, whatever the writers), so laying
10//! out the next band beside it takes that time off the column's critical
11//! path (#855: at 4096 channels, 2.2 s of every 12.4 s the DATA writer
12//! spent).
13
14use std::io::{BufWriter, Write as IoWrite};
15use std::sync::mpsc::{self, Receiver, SyncSender};
16use std::thread::JoinHandle;
17use std::time::Instant;
18
19use super::CountingFile;
20use crate::storage::StorageError;
21
22/// What the writing thread did, for the column's telemetry.
23#[derive(Debug, Default, Clone, Copy)]
24pub(super) struct BandStats {
25    pub(super) write_seconds: f64,
26    pub(super) bytes_written: usize,
27    pub(super) logical_write_calls: usize,
28    pub(super) max_logical_write_bytes: usize,
29}
30
31/// A band: every tile of one row of tiles, in file order.
32type Band = Vec<Vec<u8>>;
33
34/// What the writing thread gives back: the file's writer and what it did.
35type Written = Result<(BufWriter<CountingFile>, BandStats), StorageError>;
36
37/// The thread that writes a column's bands, and the bands passing to and
38/// from it.
39pub(super) struct BandWriter {
40    filled: Option<SyncSender<Band>>,
41    written: Receiver<Band>,
42    thread: Option<JoinHandle<Written>>,
43}
44
45impl BandWriter {
46    /// Start writing to `writer`. With a `spare` band the laying thread
47    /// fills it while its first band is written; without one, each band
48    /// comes back once written, so laying and writing take turns as they
49    /// did on one thread.
50    pub(super) fn start(
51        writer: BufWriter<CountingFile>,
52        spare: Option<Band>,
53    ) -> Result<Self, StorageError> {
54        let (filled, bands) = mpsc::sync_channel::<Band>(1);
55        let (returns, written) = mpsc::sync_channel::<Band>(2);
56        if let Some(spare) = spare {
57            returns
58                .send(spare)
59                .expect("the receiver is held, and the channel has room");
60        }
61        let thread = std::thread::Builder::new()
62            .name("tiled-band-writer".to_string())
63            .spawn(move || write_bands(writer, &bands, &returns))?;
64        Ok(Self {
65            filled: Some(filled),
66            written,
67            thread: Some(thread),
68        })
69    }
70
71    /// Hand `band`, filled, over to be written, and take back a written band
72    /// to fill: waits while the thread is still writing the band before.
73    pub(super) fn swap(&mut self, band: Vec<Vec<u8>>) -> Result<Vec<Vec<u8>>, StorageError> {
74        let sent = self
75            .filled
76            .as_ref()
77            .expect("bands are swapped only before the writer finishes")
78            .send(band);
79        match sent.ok().and_then(|()| self.written.recv().ok()) {
80            Some(band) => Ok(band),
81            // The thread stopped: it failed, and says why when joined.
82            None => Err(self.join().err().unwrap_or_else(|| {
83                StorageError::FormatMismatch("the band writer stopped without an error".into())
84            })),
85        }
86    }
87
88    /// Write every band handed over, and give back the file's writer, not
89    /// flushed, and what the thread did.
90    pub(super) fn finish(mut self) -> Result<(BufWriter<CountingFile>, BandStats), StorageError> {
91        self.join()
92    }
93
94    fn join(&mut self) -> Result<(BufWriter<CountingFile>, BandStats), StorageError> {
95        self.filled = None;
96        let thread = self.thread.take().expect("the band writer is joined once");
97        thread
98            .join()
99            .unwrap_or_else(|panic| std::panic::resume_unwind(panic))
100    }
101}
102
103impl Drop for BandWriter {
104    fn drop(&mut self) {
105        // A writer dropped unfinished (an error elsewhere) still stops its
106        // thread before the file it writes is cleaned up.
107        if self.thread.is_some() {
108            let _ = self.join();
109        }
110    }
111}
112
113/// The band-writing thread: write each band received, in order, then send
114/// it back to be filled again.
115fn write_bands(
116    mut writer: BufWriter<CountingFile>,
117    bands: &Receiver<Vec<Vec<u8>>>,
118    returns: &SyncSender<Vec<Vec<u8>>>,
119) -> Result<(BufWriter<CountingFile>, BandStats), StorageError> {
120    let mut stats = BandStats::default();
121    for band in bands {
122        let started = Instant::now();
123        // A channel-tiled band's tiles are complete on-disk buckets: write
124        // them straight to the file in order, past the buffered writer. A
125        // band of one tile goes through it, which joins small buckets.
126        let direct = band.len() > 1;
127        if direct {
128            writer.flush()?;
129        }
130        for tile in &band {
131            if direct {
132                writer.get_mut().write_all(tile)?;
133            } else {
134                writer.write_all(tile)?;
135            }
136            stats.logical_write_calls += 1;
137            stats.max_logical_write_bytes = stats.max_logical_write_bytes.max(tile.len());
138            stats.bytes_written += tile.len();
139        }
140        stats.write_seconds += started.elapsed().as_secs_f64();
141        // The laying thread may have stopped (an error there); the band is
142        // then not needed.
143        let _ = returns.send(band);
144    }
145    Ok((writer, stats))
146}
