2//! MSv4 `field_and_source_ephemeris` datasets: the fields of a visibility
3//! dataset that follow an ephemeris, and that ephemeris, as XRADIO's
4//! converter lays them out (`create_field_and_source_xds.py`,
5//! `extract_ephemeris_info`).
6//!
7//! The body's geocentric samples are `SOURCE_DIRECTION`, `SOURCE_DISTANCE`
8//! and `SOURCE_RADIAL_VELOCITY`, in SI units, on their own times
9//! (`time_ephemeris`, what XRADIO's `convert_msv2_to_processing_set`
10//! writes by default) or interpolated onto the visibility times (`time`).
11//! Either way their times are the ephemeris table's `MJD` as Unix seconds
12//! labelled UTC, which the boundary reads back as the table's `MJD`
13//! ([`FieldEphemeris::times_mjd`]), so that a store converted from an MSv2
14//! is evaluated as the MSv2 is.
15//!
16//! `FIELD_PHASE_CENTER_DIRECTION` is per visibility time: the field's
17//! offset from the body (MSv2 `PHASE_DIR`) added to the body's direction,
18//! interpolated linearly in angle onto the time, and wrapped as XRADIO
19//! wraps angles ([`wrap_to_pi`]). It is a label, not the direction casacore
20//! derives (which shifts the body's direction by the offset's true angle),
21//! so the reader recovers the offset by subtracting the body's direction
22//! again, to the store's rounding, and the writer writes the sum as XRADIO
23//! does.
24

69/// Read the `field_and_source_ephemeris` dataset at `path`, whose
70/// visibility dataset has `times_s` (Unix seconds, UTC) and the fields
71/// `per_time` names at them.
72///
73/// # Errors
74///
75/// [`ObservationError::InvalidStore`] when a variable is missing or not of
76/// the profile's type, dimensions, shape and units, the per-time field
77/// names differ from the visibility dataset's, a field names two sources
78/// or its offsets from the body differ, or the samples are not an
79/// ephemeris ([`FieldEphemeris::new`]); [`ObservationError::Unsupported`]
80/// for an ephemeris in another frame than ICRS or J2000, a phase centre in
81/// another frame than its ephemeris, or one stated from another observer
82/// than the Earth's centre.
83pub(super) fn read(
84    store: &Arc<FilesystemStore>,
85    path: &str,
86    times_s: &[f64],
87    per_time: &[String],
88) -> Result<MovingFields, ObservationError> {
89    let time_ephemeris = format!("{path}/time_ephemeris");
90    let (sample_times_s, axis) = if has_node(store, &time_ephemeris)? {
91        let (times, attributes) = numbers::<f64>(store, &time_ephemeris)?;
92        time_reference(&attributes, "time_ephemeris")?;
93        (times, "time_ephemeris")
94    } else {
95        tracing::warn!(
96            "{path}: the ephemeris is interpolated onto the visibility times \
97             (ephemeris_interpolate=True): its table's own samples are not in the store, and the \
98             body's direction is evaluated from these, which start at the first time and end at \
99             the last"
100        );
101        (times_s.to_vec(), "time")
102    };
103    let samples = sample_times_s.len();
104    let (directions, attributes) = series(
105        store,
106        path,
107        "SOURCE_DIRECTION",
108        &[axis, "sky_dir_label"],
109        &[samples, 2],
110        "rad",
111    )?;
112    let frame = direction_frame(&attributes, "SOURCE_DIRECTION")?;
113    let (distances, _) = series(
114        store,
115        path,
116        "SOURCE_DISTANCE",
117        &[axis, "sky_dis_label"],
118        &[samples, 1],
119        "m",
120    )?;
121    let (velocities, _) = series(
122        store,
123        path,
124        "SOURCE_RADIAL_VELOCITY",
125        &[axis],
126        &[samples],
127        "m/s",
128    )?;
129    require_geocentric(store, path)?;
130    let directions: Vec<[f64; 2]> = directions.as_chunks::<2>().0.to_vec();
131    let (centres, attributes) = series(
132        store,
133        path,
134        "FIELD_PHASE_CENTER_DIRECTION",
135        &["time", "sky_dir_label"],
136        &[times_s.len(), 2],
137        "rad",
138    )?;
139    if direction_frame(&attributes, "FIELD_PHASE_CENTER_DIRECTION")? != frame {
140        return Err(ObservationError::Unsupported {
141            what: "an MSv4 moving phase centre in another frame than its ephemeris",
142        });
143    }
144    let names = strings(store, &format!("{path}/field_name"))?;
145    let sources = strings(store, &format!("{path}/source_name"))?;
146    if names != per_time || sources.len() != names.len() {
147        return Err(invalid(format!(
148            "{path}'s field_name and source_name are not the visibility dataset's fields at its \
149             times"
150        )));
151    }
152    let bodies = at_times(&sample_times_s, &directions, times_s)
153        .ok_or_else(|| invalid(format!("{path}: a time outside SOURCE_DIRECTION's samples")))?;
154    let (fields, per_time) = moving_fields(path, &names, &sources, &centres, &bodies)?;
155    let name = fields
156        .first()
157        .map(|field| {
158            if super::is_unknown_source(&field.source_name) {
159                field.name.clone()
160            } else {
161                field.source_name.clone()
162            }
163        })
164        .unwrap_or_default();
165    if samples < 2 {
166        return Err(ObservationError::Unsupported {
167            what: "an MSv4 ephemeris of fewer than two samples (as XRADIO interpolates one onto \
168                   a dataset of one time)",
169        });
170    }
171    let ephemeris = FieldEphemeris::new(
172        name,
173        sample_times_s
174            .iter()
175            .map(|seconds| UNIX_EPOCH_MJD + seconds / SECONDS_PER_DAY)
176            .collect(),
177        directions,
178        frame,
179        distances,
180        (velocities, RadialVelocityRef::GEO),
181    )
182    .map_err(|error| invalid(format!("{path}: {error}")))?;
183    Ok(MovingFields {
184        fields,
185        per_time,
186        ephemeris,
187    })
188}

190/// Variable `name` of the dataset at `path`: `float64` of `dimensions` and
191/// `shape`, in `units`, with its attributes.
192fn series(
193    store: &Arc<FilesystemStore>,
194    path: &str,
195    name: &str,
196    dimensions: &[&str],
197    shape: &[usize],
198    units: &str,
199) -> Result<(Vec<f64>, serde_json::Map<String, serde_json::Value>), ObservationError> {
200    let variable = format!("{path}/{name}");
201    let array = open_array(store, &variable)?;
202    require_layout(&array, &variable, "float64", dimensions)?;
203    let expected: Vec<u64> = shape.iter().map(|n| *n as u64).collect();
204    if array.shape() != expected {
205        return Err(invalid(format!(
206            "{variable} has shape {:?}, not {expected:?}",
207            array.shape()
208        )));
209    }
210    require_units(array.attributes(), units, name)?;
211    numbers::<f64>(store, &variable)
212}

214/// Refuse an ephemeris stated from anywhere but the Earth's centre: XRADIO
215/// converts only geocentric tables (`obsloc` `GEOCENTRIC`), and records
216/// that as `OBSERVER_POSITION`'s `coordinate_system`. (Its position is not
217/// the geocentre: XRADIO places the table's `GeoLong`, `GeoLat` and
218/// `GeoDist` on the WGS84 ellipsoid, the distance as a height.) The schema
219/// makes `OBSERVER_POSITION` optional; a dataset without one is read as
220/// geocentric, the only kind XRADIO writes.
221fn require_geocentric(store: &Arc<FilesystemStore>, path: &str) -> Result<(), ObservationError> {
222    let observer = format!("{path}/OBSERVER_POSITION");
223    if !has_node(store, &observer)? {
224        return Ok(());
225    }
226    let array = open_array(store, &observer)?;
227    if text(array.attributes(), "coordinate_system", "OBSERVER_POSITION")? == "geocentric" {
228        Ok(())
229    } else {
230        Err(ObservationError::Unsupported {
231            what: "an MSv4 ephemeris stated from another observer than the Earth's centre",
232        })
233    }
234}

236/// The distinct fields of `names` (each time's field), with their sources
237/// and offsets from `bodies` (the body's direction at each time), and each
238/// time's field.
239fn moving_fields(
240    path: &str,
241    names: &[String],
242    sources: &[String],
243    centres: &[f64],
244    bodies: &[[f64; 2]],
245) -> Result<(Vec<MovingField>, Vec<usize>), ObservationError> {
246    let mut fields: Vec<MovingField> = Vec::new();
247    let mut per_time = Vec::with_capacity(names.len());
248    for (time, name) in names.iter().enumerate() {
249        let offset_rad =
250            std::array::from_fn(|axis| wrap_to_pi(centres[2 * time + axis] - bodies[time][axis]));
251        let index = match fields.iter().position(|field| field.name == *name) {
252            Some(index) => {
253                let field = &fields[index];
254                if field.source_name != sources[time] {
255                    return Err(invalid(format!("{path}: field {name} names two sources")));
256                }
257                if !same_angles(offset_rad, field.offset_rad) {
258                    return Err(invalid(format!(
259                        "{path}: field {name}'s phase centre is not one offset from \
260                         SOURCE_DIRECTION"
261                    )));
262                }
263                index
264            }
265            None => {
266                fields.push(MovingField {
267                    name: name.clone(),
268                    source_name: sources[time].clone(),
269                    offset_rad,
270                });
271                fields.len() - 1
272            }
273        };
274        per_time.push(index);
275    }
276    Ok((fields, per_time))
277}

279/// `values` (at `sample_times`) at each of `times`, interpolated linearly
280/// (in angle, for directions) as XRADIO interpolates them (numpy's
281/// `interp`), or `None` for a time outside the samples. Times are Unix
282/// seconds; one within [`TIME_RESOLUTION_S`] of an end sample is at it, as
283/// a sample at a visibility time comes back from its MJD.
284pub(super) fn at_times<const N: usize>(
285    sample_times: &[f64],
286    values: &[[f64; N]],
287    times: &[f64],
288) -> Option<Vec<[f64; N]>> {
289    let last = sample_times.len().checked_sub(1)?;
290    let (first_sample, last_sample) = (sample_times[0], sample_times[last]);
291    times
292        .iter()
293        .map(|time| {
294            if *time < first_sample - TIME_RESOLUTION_S || *time > last_sample + TIME_RESOLUTION_S {
295                return None;
296            }
297            let time = &time.clamp(first_sample, last_sample);
298            let upper = sample_times.partition_point(|sample| sample <= time);
299            if upper > last {
300                return Some(values[last]);
301            }
302            let lower = upper - 1;
303            let span = sample_times[upper] - sample_times[lower];
304            Some(std::array::from_fn(|axis| {
305                let slope = (values[upper][axis] - values[lower][axis]) / span;
306                slope * (time - sample_times[lower]) + values[lower][axis]
307            }))
308        })
309        .collect()
310}

312/// An angle wrapped to [-π, π) as XRADIO wraps angles
313/// (`coord_math.wrap_to_pi`).
314pub(super) fn wrap_to_pi(angle: f64) -> f64 {
315    (angle + PI).rem_euclid(TAU) - PI
316}

318/// How far apart two statements of one time may be, seconds: 1 ms.
319/// Converters state a table's `MJD` as Unix seconds by their own
320/// arithmetic, which rounds differently, and a time read back as MJD days
321/// moves by up to a microsecond.
322pub(super) const TIME_RESOLUTION_S: f64 = 1.0e-3;
323
324/// [`TIME_RESOLUTION_S`] in days: how far apart two datasets' samples of
325/// one body may be stated and still be one sample.
326const SAMPLE_TIME_RESOLUTION_DAYS: f64 = TIME_RESOLUTION_S / SECONDS_PER_DAY;
327
328/// How far two datasets' values of one sample may differ, relative to
329/// the larger: their converters' unit conversions round differently, by
330/// about 1e-16.
331const SAMPLE_RESOLUTION: f64 = 1.0e-12;
332
333/// One sample: time, direction, distance and radial velocity.
334type Sample = (f64, [f64; 2], f64, f64);
335
336/// How far past a time its TDB can be, seconds: 69.184 s since 2017 (TT −
337/// TAI 32.184 s, TAI − UTC 37 s), TDB − TT within 2 ms, and room for two
338/// more leap seconds. Opening a store takes no leap-second table, so it
339/// bounds the TDB rather than computing it.
340const TDB_LEAD_S: f64 = 72.0;
341
342/// The most grid points an extension adds, so that opening stays quick. A
343/// finer grid, as a union of datasets whose times nearly coincide has,
344/// gains one point at the bound instead.
345const MOST_EXTRAPOLATED: f64 = 10_000.0;
346
347/// Make `fields` whose ephemerides hold the same samples share one, then
348/// extend each ephemeris they follow whose samples cover the last time of
349/// every partition one of its fields is in but end less than
350/// [`TDB_LEAD_S`] after the latest of them: with the samples' grid points
351/// up to that bound ([`trailing_step`]), each on the straight line through
352/// the last two ([`FieldEphemeris::extrapolated_to`]), and a warning.
353/// Imaging reads a body's direction at each time's TDB, and XRADIO keeps
354/// only the samples that bracket a dataset's UTC times; the owner chose
355/// extending them over refusing those times (#995). Sharing first extends
356/// one body's fields in different datasets together, to the latest of
357/// their last times, and the catalogue counts each ephemeris once
358/// ([`Catalogue::ephemeris_bytes`](casa_observation::Catalogue::ephemeris_bytes)),
359/// as the fields of one MSv2 `EPHEMERIS_ID` share one. `last_times` holds
360/// each partition's fields and last time, MJD UTC.
361///
362/// # Errors
363///
364/// An extension [`FieldEphemeris`] refuses.
365pub(super) fn reach_last_tdb(
366    fields: &mut [FieldEntry],
367    last_times: &[(Vec<FieldId>, f64)],
368) -> Result<(), ObservationError> {
369    let mut shared: Vec<Arc<FieldEphemeris>> = Vec::new();
370    for ephemeris in fields
371        .iter_mut()
372        .filter_map(|field| field.ephemeris.as_mut())
373    {
374        match shared.iter().find(|known| ***known == **ephemeris) {
375            Some(known) => *ephemeris = Arc::clone(known),
376            None => shared.push(Arc::clone(ephemeris)),
377        }
378    }
379    let mut needed: Vec<(Arc<FieldEphemeris>, f64)> = Vec::new();
380    for (ids, last) in last_times {
381        for field in ids {
382            let Some(ephemeris) = &fields[field.0 as usize].ephemeris else {
383                continue;
384            };
385            match needed
386                .iter_mut()
387                .find(|(known, _)| Arc::ptr_eq(known, ephemeris))
388            {
389                Some((_, end)) => *end = end.max(*last),
390                None => needed.push((Arc::clone(ephemeris), *last)),
391            }
392        }
393    }
394    for (ephemeris, last) in needed {
395        let times = ephemeris.times_mjd();
396        let end = times[times.len() - 1];
397        let tdb = last + TDB_LEAD_S / SECONDS_PER_DAY;
398        // `read` refuses a dataset whose own samples do not cover its times,
399        // and uniting only adds samples, so the first test is a defence.
400        if end < last - SAMPLE_TIME_RESOLUTION_DAYS || end >= tdb {
401            continue;
402        }
403        let step = trailing_step(times);
404        let points = ((tdb - end) / step).ceil();
405        let at: Vec<f64> = if points > MOST_EXTRAPOLATED {
406            vec![tdb]
407        } else {
408            (1..=points as usize)
409                .map(|point| end + point as f64 * step)
410                .collect()
411        };
412        let extended = Arc::new(ephemeris.extrapolated_to(&at).map_err(|error| {
413            invalid(format!(
414                "extending the ephemeris of {}: {error}",
415                ephemeris.name()
416            ))
417        })?);
418        tracing::warn!(
419            "the ephemeris of {}: its samples end {:.1} s after the last time, short of its TDB; \
420             {} samples extrapolated, to {:.1} s after it, along the line through the last two",
421            ephemeris.name(),
422            (end - last) * SECONDS_PER_DAY,
423            at.len(),
424            (at[at.len() - 1] - last) * SECONDS_PER_DAY
425        );
426        for field in fields.iter_mut() {
427            if field
428                .ephemeris
429                .as_ref()
430                .is_some_and(|known| Arc::ptr_eq(known, &ephemeris))
431            {
432                field.ephemeris = Some(Arc::clone(&extended));
433            }
434        }
435    }
436    Ok(())
437}

439/// The spacing of the trailing run of `times` (MJD days) whose intervals
440/// agree to 1 ms, from the run's whole span: a grid continued from it
441/// keeps the run's rounding, not the last pair's, which a microsecond off
442/// would carry a thousand points out of an MSv2 table's 1 ms tolerance.
443fn trailing_step(times: &[f64]) -> f64 {
444    let last = times.len() - 1;
445    let pair = times[last] - times[last - 1];
446    let mut first = last - 1;
447    while first > 0
448        && ((times[first] - times[first - 1]) - pair).abs() <= SAMPLE_TIME_RESOLUTION_DAYS
449    {
450        first -= 1;
451    }
452    (times[last] - times[first]) / (last - first) as f64
453}

455/// The ephemeris of `known` and `other` together, when they are one
456/// body's samples: the same frames, and the same values (to
457/// [`SAMPLE_RESOLUTION`]) at every time both hold (to
458/// [`SAMPLE_TIME_RESOLUTION_DAYS`]), where `known`'s are kept. Two datasets
459/// of one field each hold the samples their own times need; `None` when
460/// they disagree.
461pub(super) fn union(
462    known: &Arc<FieldEphemeris>,
463    other: &FieldEphemeris,
464) -> Option<Arc<FieldEphemeris>> {
465    if **known == *other {
466        return Some(Arc::clone(known));
467    }
468    if known.frame() != other.frame() || known.velocity_frame() != other.velocity_frame() {
469        return None;
470    }
471    let samples = |ephemeris: &FieldEphemeris| -> Vec<Sample> {
472        (0..ephemeris.times_mjd().len())
473            .map(|index| {
474                (
475                    ephemeris.times_mjd()[index],
476                    ephemeris.directions_rad()[index],
477                    ephemeris.distances_m()[index],
478                    ephemeris.radial_velocities_m_s()[index],
479                )
480            })
481            .collect()
482    };
483    let (left, right) = (samples(known), samples(other));
484    let (mut a, mut b) = (0, 0);
485    let mut merged: Vec<Sample> = Vec::with_capacity(left.len() + right.len());
486    while a < left.len() || b < right.len() {
487        let next = match (left.get(a), right.get(b)) {
488            (Some(mine), Some(theirs))
489                if (mine.0 - theirs.0).abs() <= SAMPLE_TIME_RESOLUTION_DAYS =>
490            {
491                if !same_sample(mine, theirs) {
492                    return None;
493                }
494                a += 1;
495                b += 1;
496                *mine
497            }
498            (Some(mine), Some(theirs)) if mine.0 < theirs.0 => {
499                a += 1;
500                *mine
501            }
502            (Some(mine), None) => {
503                a += 1;
504                *mine
505            }
506            (_, Some(theirs)) => {
507                b += 1;
508                *theirs
509            }
510            (None, None) => unreachable!("the loop ends when both are exhausted"),
511        };
512        merged.push(next);
513    }
514    FieldEphemeris::new(
515        known.name(),
516        merged.iter().map(|sample| sample.0).collect(),
517        merged.iter().map(|sample| sample.1).collect(),
518        known.frame(),
519        merged.iter().map(|sample| sample.2).collect(),
520        (
521            merged.iter().map(|sample| sample.3).collect(),
522            known.velocity_frame(),
523        ),
524    )
525    .ok()
526    .map(Arc::new)
527}

529/// Whether two samples at one time hold one body's values
530/// ([`SAMPLE_RESOLUTION`]; angles to [`OFFSET_RESOLUTION_RAD`]).
531fn same_sample(a: &Sample, b: &Sample) -> bool {
532    let close = |x: f64, y: f64| (x - y).abs() <= SAMPLE_RESOLUTION * x.abs().max(y.abs());
533    same_angles(a.1, b.1) && close(a.2, b.2) && (close(a.3, b.3) || (a.3 - b.3).abs() <= 1.0e-9)
534}

536/// Whether the angles `a` and `b` (longitude, latitude) are one, to the
537/// store's resolution ([`OFFSET_RESOLUTION_RAD`]), whatever turn each is
538/// stated in: a recovered offset is wrapped to [-π, π), and
539/// [`MDirection::as_angles`](casa_types::measures::direction::MDirection::as_angles)
540/// states a longitude in [0, 2π).
541pub(super) fn same_angles(a: [f64; 2], b: [f64; 2]) -> bool {
542    (0..2).all(|axis| wrap_to_pi(a[axis] - b[axis]).abs() <= OFFSET_RESOLUTION_RAD)
543}

590    fn ephemeris(times: &[f64]) -> FieldEphemeris {
591        FieldEphemeris::new(
592            "body",
593            times.to_vec(),
594            times.iter().map(|time| [time * 1e-3, 0.1]).collect(),
595            DirectionRef::ICRS,
596            times.iter().map(|time| 1e8 + time).collect(),
597            (times.to_vec(), RadialVelocityRef::GEO),
598        )
599        .expect("an ephemeris")
600    }

603    fn two_datasets_samples_of_one_body_unite_and_conflicting_ones_do_not() {
604        let known = Arc::new(ephemeris(&[1.0, 2.0, 3.0]));
605        assert!(Arc::ptr_eq(
606            &union(&known, &ephemeris(&[1.0, 2.0, 3.0])).expect("equal"),
607            &known
608        ));
609        let united = union(&known, &ephemeris(&[3.0, 4.0, 6.0])).expect("one body");
610        assert_eq!(*united, ephemeris(&[1.0, 2.0, 3.0, 4.0, 6.0]));
611        // Another converter's statement of a sample's time, 0.1 ms off, is
612        // the same sample.
613        let restated = ephemeris(&[3.0, 4.0]);
614        let shifted = FieldEphemeris::new(
615            "body",
616            vec![3.0 + 1.0e-4 / SECONDS_PER_DAY, 4.0],
617            restated.directions_rad().to_vec(),
618            DirectionRef::ICRS,
619            restated.distances_m().to_vec(),
620            (
621                restated.radial_velocities_m_s().to_vec(),
622                RadialVelocityRef::GEO,
623            ),
624        )
625        .expect("an ephemeris");
626        let united = union(&known, &shifted).expect("one body");
627        assert_eq!(united.times_mjd(), [1.0, 2.0, 3.0, 4.0]);
628        let mut moved = ephemeris(&[2.0, 5.0]);
629        moved = FieldEphemeris::new(
630            "body",
631            moved.times_mjd().to_vec(),
632            vec![[2.0e-3, 0.2], [5.0e-3, 0.1]],
633            DirectionRef::ICRS,
634            moved.distances_m().to_vec(),
635            (
636                moved.radial_velocities_m_s().to_vec(),

638            ),

640        .expect("an ephemeris");
641        assert!(union(&known, &moved).is_none());

664    #[test]
665    fn an_ephemeris_short_of_the_latest_last_times_tdb_is_extended_on_its_grid() {
666        let day = |seconds: f64| 60_000.0 + seconds / SECONDS_PER_DAY;
667        // One-minute samples to 300 s; field A's partition ends at 200 s,
668        // B's at 280 s. Only B's needs more: a sample at 360 s, past 352 s.
669        let samples: Vec<f64> = (0..=5)
670            .map(|minute| day(60.0 * f64::from(minute)))
671            .collect();
672        let body = Arc::new(ephemeris(&samples));
673        let mut fields = vec![following("A", &body), following("B", &body)];
674        let last_times = [
675            (vec![FieldId(0)], day(200.0)),
676            (vec![FieldId(1)], day(280.0)),
677        ];
678        reach_last_tdb(&mut fields, &last_times).expect("extended");
679        let extended = fields[0].ephemeris.as_ref().expect("an ephemeris");
680        assert!(Arc::ptr_eq(
681            extended,
682            fields[1].ephemeris.as_ref().expect("an ephemeris")
683        ));
684        assert_eq!(extended.extrapolated(), 1);
685        let last = extended.times_mjd()[6];
686        assert!((last - day(360.0)).abs() * SECONDS_PER_DAY < 1.0e-6);
687        // Ten-second samples ending at the last time: each grid point to 72 s
688        // past it, eight samples and no gap.
689        let samples: Vec<f64> = (0..=3).map(|step| day(10.0 * f64::from(step))).collect();
690        let body = Arc::new(ephemeris(&samples));
691        let mut fields = vec![following("A", &body)];
692        reach_last_tdb(&mut fields, &[(vec![FieldId(0)], day(30.0))]).expect("extended");
693        let extended = fields[0].ephemeris.as_ref().expect("an ephemeris");
694        assert_eq!(extended.extrapolated(), 8);
695        assert!(
696            extended
697                .times_mjd()
698                .windows(2)
699                .all(|pair| ((pair[1] - pair[0]) * SECONDS_PER_DAY - 10.0).abs() < 1.0e-6)
700        );
701        // Equal ephemerides of two fields in different partitions are
702        // shared first, so both reach the later one's TDB together.
703        let samples: Vec<f64> = (0..=5)
704            .map(|minute| day(60.0 * f64::from(minute)))
705            .collect();
706        let (first, second) = (Arc::new(ephemeris(&samples)), Arc::new(ephemeris(&samples)));
707        let mut fields = vec![following("A", &first), following("B", &second)];
708        reach_last_tdb(&mut fields, &last_times).expect("extended");
709        let shared = fields[0].ephemeris.as_ref().expect("an ephemeris");
710        assert!(Arc::ptr_eq(
711            shared,
712            fields[1].ephemeris.as_ref().expect("an ephemeris")
713        ));
714        assert_eq!(shared.extrapolated(), 1);
715        // A grid too fine to extend point by point gains one point, at the
716        // bound.
717        let fine: Vec<f64> = (0..10).map(|step| day(1.0e-3 * f64::from(step))).collect();
718        let mut fields = vec![following("A", &Arc::new(ephemeris(&fine)))];
719        reach_last_tdb(&mut fields, &[(vec![FieldId(0)], fine[9])]).expect("extended");
720        let extended = fields[0].ephemeris.as_ref().expect("an ephemeris");
721        assert_eq!(extended.extrapolated(), 1);
722        assert!(((extended.times_mjd()[10] - fine[9]) * SECONDS_PER_DAY - 72.0).abs() < 1.0e-6);
723        // Samples that end before a last time, or reach its TDB, are left.
724        for last in [day(40.0), day(30.0 - 72.0)] {
725            let mut fields = vec![following("A", &body)];
726            reach_last_tdb(&mut fields, &[(vec![FieldId(0)], last)]).expect("left");
727            assert!(Arc::ptr_eq(
728                fields[0].ephemeris.as_ref().expect("an ephemeris"),
729                &body
730            ));
731        }
732    }
