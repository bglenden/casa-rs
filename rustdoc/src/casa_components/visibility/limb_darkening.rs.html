2//! CASA LimbDarkenedDiskShape::calcVis: Γ(η+1) (2/r)^η Jη(r), η=1+n/2.
3use num_complex::Complex64;
4
5use crate::ComponentError;
6
7#[derive(Debug, Clone, Copy, PartialEq)]
8pub(super) struct LimbDarkening {
9    order: f64,
10    log_gamma: f64,
11}
12
13fn refused(reason: impl ToString) -> ComponentError {
14    ComponentError::Visibility {
15        shape: "limb-darkened disk",
16        reason: reason.to_string(),
17    }
18}
19
20impl LimbDarkening {
21    pub(super) fn new(exponent: f64) -> Result<Self, ComponentError> {
22        let order = 1. + f64::from(exponent as f32) / 2.;
23        if !order.is_finite() || order < 0. {
24            return Err(refused(
25                "the stored exponent must be finite and at least -2",
26            ));
27        }
28        Ok(Self {
29            order,
30            log_gamma: libm::lgamma(order + 1.),
31        })
32    }
33
34    pub(super) fn at_radius(self, radius: f64) -> Result<f64, ComponentError> {
35        if !radius.is_finite() || radius < 0. {
36            return Err(refused("the Fourier radius must be finite and nonnegative"));
37        }
38        // The normalized power series avoids multiplying an underflowed Jη
39        // by an overflowing r^-η near zero. In this range its terms decrease.
40        if radius <= 2. * (self.order + 1.).sqrt() {
41            let mut sum = 1.;
42            let mut term = 1.;
43            for k in 1..=64 {
44                let k = f64::from(k);
45                term *= -(radius * radius / 4.) / (k * (self.order + k));
46                sum += term;
47                if term.abs() <= f64::EPSILON * sum.abs() {
48                    break;
49                }
50            }
51            return Ok(sum);
52        }
53        // AMOS 644 supplies real-order J without a native GSL dependency or
54        // heap allocation. Log normalization avoids intermediate overflow.
55        let bessel = complex_bessel::besselj(self.order, Complex64::new(radius, 0.))
56            .map_err(refused)?
57            .re;
58        let log_scale = self.log_gamma + self.order * (2. / radius).ln();
59        if bessel == 0. && log_scale > 0. {
60            return Err(refused("Bessel underflow prevents normalized evaluation"));
61        }
62        Ok(bessel.signum() * (log_scale + bessel.abs().ln()).exp())
63    }
64}
