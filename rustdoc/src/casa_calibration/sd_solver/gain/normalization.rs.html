2//! Flag-aware double-circle gain normalization.
3
4use casa_core::Complex32;
5use ndarray::ArrayD;
6
7use crate::sd_solver::edges::Integration;
8
9const ARCSEC_PER_RADIAN: f64 = 180. * 3600. / std::f64::consts::PI;
10const SPEED_OF_LIGHT_M_S: f64 = 299_792_458.;
11
12pub(super) struct GainSample<'a> {
13    pub(super) integration: &'a Integration,
14    pub(super) data: ArrayD<Complex32>,
15    pub(super) flags: ArrayD<bool>,
16}
17
18pub(super) fn radius(
19    requested: Option<f64>,
20    frequencies: &[f64],
21    aperture: f64,
22) -> Result<f64, String> {
23    if let Some(radius) = requested.filter(|radius| *radius > 0.) {
24        return Ok(radius);
25    }
26    let last = frequencies
27        .len()
28        .checked_sub(1)
29        .ok_or("Missing observing frequency.")?;
30    let center = (frequencies[last / 2] + frequencies[frequencies.len() / 2]) / 2.;
31    let radius = 0.5 * 1.13 * SPEED_OF_LIGHT_M_S / (aperture * center);
32    if !radius.is_finite() || radius <= 0. {
33        return Err("Size of central region is not properly set.".into());
34    }
35    Ok(radius)
36}

38pub(super) fn normalize(
39    samples: &mut [GainSample<'_>],
40    radius: f64,
41    smooth: bool,
42) -> Result<(), String> {
43    let shape = samples[0].data.shape();
44    let [receptors, channels] =
45        <[usize; 2]>::try_from(shape).map_err(|_| "Double-circle data must have two axes.")?;
46    if samples
47        .iter()
48        .any(|sample| sample.data.shape() != shape || sample.flags.shape() != shape)
49    {
50        return Err("Inconsistent visibility or flag shape in the gain group.".into());
51    }
52    let half_width = (radius * ARCSEC_PER_RADIAN / 1.5).round() as usize;
53    let width = half_width.saturating_mul(2).saturating_add(1);
54    let clip = smooth && (2..samples.len()).contains(&width);
55    let count = samples.len();
56    let mut data = Vec::with_capacity(count);
57    let mut flags = Vec::with_capacity(count);
58    let mut smoothed = Vec::with_capacity(count);
59    for receptor in 0..receptors {
60        for channel in 0..channels {
61            let index = [receptor, channel];
62            data.clear();
63            flags.clear();
64            for sample in samples.iter() {
65                data.push(f64::from(sample.data[index].re));
66                flags.push(sample.flags[index]);
67            }
68            let powers = if clip {
69                smooth_power(&data, &flags, half_width, &mut smoothed);
70                &smoothed
71            } else {
72                &data
73            };
74            let (sum, present) = powers
75                .iter()
76                .zip(&flags)
77                .filter(|&(_, &flag)| !flag)
78                .fold((0., 0), |(sum, count), (&power, _)| {
79                    (sum + power, count + 1)
80                });
81            let mean = sum / f64::from(present.max(1));
82            for (sample, (&raw, &power)) in samples.iter_mut().zip(data.iter().zip(powers)) {
83                let ratio = mean / power;
84                if raw != 0. && mean >= 0. && ratio > 0. {
85                    sample.data[index].re = (1. / ratio.sqrt()) as f32;
86                } else {
87                    sample.data[index].re = 0.;
88                    sample.flags[index] = true;
89                }
90            }
91        }
92    }
93    Ok(())
94}

96fn smooth_power(data: &[f64], flags: &[bool], half_width: usize, output: &mut Vec<f64>) {
97    // Advance the bounds monotonically. Each unflagged sample enters and
98    // leaves the truncated moving window once.
99    output.clear();
100    let mut sum = 0.;
101    let mut present = 0;
102    let mut end = 0;
103    for (center, (&power, &flag)) in data.iter().zip(flags).enumerate() {
104        let bound = center.saturating_add(half_width + 1).min(data.len());
105        for i in end..bound {
106            if !flags[i] {
107                sum += data[i];
108                present += 1;
109            }
110        }
111        end = bound;
112        output.push(if flag {
113            power
114        } else {
115            sum / f64::from(present.max(1))
116        });
117        if let Some(left) = center.checked_sub(half_width)
118            && !flags[left]
119        {
120            sum -= data[left];
121            present -= 1;
122        }
123    }
124}
