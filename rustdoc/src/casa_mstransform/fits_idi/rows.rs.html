22pub(super) struct DataLayout {
23    pub(super) bands: usize,
24    pub(super) channels: usize,
25    pub(super) correlations: Vec<Correlation>,
26    pub(super) components: usize,
27    normal: Vec<usize>,
28    reversed: Vec<usize>,
29    floats: usize,
30}

32impl DataLayout {
33    pub(super) fn read(fits: &mut FitsFile, hdu: &FitsHdu) -> MsResult<Self> {
34        let axes: i32 = hdu.read_key(fits, "MAXIS").map_err(invalid)?;
35        if !(4..=999).contains(&axes) {
36            return Err(invalid("Invalid FITS-IDI visibility axes"));
37        }
38        let mut components = None;
39        let mut band = None;
40        let mut frequency = None;
41        let mut input = None;
42        for number in 1..=axes {
43            let key = |kind| format!("{kind}{number}");
44            let name: String = hdu.read_key(fits, &key("CTYPE")).map_err(invalid)?;
45            let samples: i32 = hdu.read_key(fits, &key("MAXIS")).map_err(invalid)?;
46            let samples = usize::try_from(samples)
47                .ok()
48                .filter(|&size| size > 0)
49                .ok_or_else(|| invalid("Invalid FITS-IDI axis length"))?;
50            match name.trim_end() {
51                "COMPLEX" => components = Some(samples),
52                "STOKES" => {
53                    if samples > 8 {
54                        return Err(invalid("Invalid FITS-IDI STOKES axis length"));
55                    }
56                    let reference: f64 = hdu.read_key(fits, &key("CRVAL")).map_err(invalid)?;
57                    let pixel: f64 = hdu.read_key(fits, &key("CRPIX")).map_err(invalid)?;
58                    let step: f64 = hdu.read_key(fits, &key("CDELT")).map_err(invalid)?;
59                    input = Some(
60                        (0..samples)
61                            .map(|index| {
62                                let code = (reference + (index as f64 + 1.0 - pixel) * step + 0.5)
63                                    .floor() as i32;
64                                if code == 1 {
65                                    // The FITS-IDI reader treats SMA's +1 as a receptor pair.
66                                    Ok(Correlation::XX)
67                                } else {
68                                    fits_correlation(code, false)
69                                }
70                            })
71                            .collect::<MsResult<Vec<_>>>()?,
72                    );
73                }
74                "FREQ" => frequency = Some(samples),
75                "BAND" => band = Some(samples),
76                _ if samples == 1 => {}
77                _ => {
78                    return Err(invalid(format!(
79                        "FITS-IDI visibility axis {name} must be degenerate"
80                    )));
81                }
82            }
83        }
84        let components = components
85            .filter(|value| matches!(value, 2 | 3))
86            .ok_or_else(|| invalid("FITS-IDI needs two or three complex components"))?;
87        let input = input.ok_or_else(|| invalid("FITS-IDI has no STOKES axis"))?;
88        let mut correlations = input.clone();
89        correlations.sort_unstable();
90        correlations.dedup();
91        if correlations.len() != input.len() {
92            return Err(invalid("Repeated FITS-IDI correlations"));
93        }
94        let position = |target| correlations.iter().position(|&value| value == target);
95        let normal = input
96            .iter()
97            .map(|&value| position(value).unwrap())
98            .collect();
99        let reversed = input
100            .iter()
101            .map(|&value| {
102                use Correlation::{LR, RL, XY, YX};
103                position(match value {
104                    RL => LR,
105                    LR => RL,
106                    XY => YX,
107                    YX => XY,
108                    other => other,
109                })
110                .unwrap_or_else(|| position(value).unwrap())
111            })
112            .collect();
113        let bands = band.unwrap_or(1);
114        let channels = frequency.ok_or_else(|| invalid("FITS-IDI has no FREQ axis"))?;
115        let floats = channels
116            .checked_mul(input.len())
117            .and_then(|cells| cells.checked_mul(bands))
118            .and_then(|cells| cells.checked_mul(components))
119            .ok_or_else(|| invalid("FITS-IDI visibility row size overflow"))?;
120        let HduInfo::TableInfo {
121            column_descriptions,
122            ..
123        } = &hdu.info
124        else {
125            return Err(invalid("UV_DATA is not a FITS binary table"));
126        };
127        let row_bytes = hdu.read_key::<i64>(fits, "NAXIS1").map_err(invalid)?;
128        if !column_descriptions.iter().any(|column| {
129            column.name == "FLUX"
130                && column.data_type.typ == ColumnDataType::Float
131                && column.data_type.repeat == floats
132        }) || floats.checked_mul(size_of::<f32>()).is_none_or(|bytes| {
133            usize::try_from(row_bytes).map_or(true, |row_bytes| bytes > row_bytes)
134        }) {
135            return Err(invalid(
136                "FITS-IDI FLUX repeat disagrees with the visibility axes",
137            ));
138        }
139        Ok(Self {
140            bands,
141            channels,
142            correlations,
143            components,
144            normal,
145            reversed,
146            floats,
147        })
148    }
149
150    fn floats(&self) -> usize {
151        self.floats
152    }
153
154    pub(super) fn same_output_shape(&self, other: &Self) -> bool {
155        self.channels == other.channels
156            && self.bands == other.bands
157            && self.correlations == other.correlations
158            && self.components == other.components
159    }
160}

176pub(super) fn append_hdu(
177    fits: &mut FitsFile,
178    hdu: &FitsHdu,
179    columns: &mut ImportColumns,
180    geometry: &Geometry,
181    input: &VisibilityInput<'_>,
182    scan: i32,
183    times: &mut TimeRange,
184) -> MsResult<()> {
185    let layout = DataLayout::read(fits, hdu)?;
186    let HduInfo::TableInfo { num_rows, .. } = hdu.info else {
187        return Err(invalid("UV_DATA is not a FITS binary table"));
188    };
189    let scale = optional_key::<f64>(hdu, fits, "VIS_SCAL")?.unwrap_or(1.0);
190    let weight_type = optional_key::<String>(hdu, fits, "WEIGHTYP")?
191        .map_or(matches!(input.correlator, "DIFX" | "VLBA"), |value| {
192            value.trim().eq_ignore_ascii_case("CORRELAT")
193        });
194    let rows = InputColumns::resolve(hdu)?;
195    let inferred = if rows.interval {
196        None
197    } else {
198        Some(infer_interval(fits, hdu, &rows, num_rows)?)
199    };
200    let mut buffers = DecodeBuffers::new(&layout, input.resolutions);
201    let mut digital = DigitalCorrection::for_correlator(input.correlator, input.van_vleck);
202    let rows_per_block = (262_144 / layout.floats()).clamp(1, 1024);
203    for start in (0..num_rows).step_by(rows_per_block) {
204        let range = start..(start + rows_per_block).min(num_rows);
205        let values = RowBlock::read(fits, hdu, &rows, &layout, range)?;
206        for index in 0..values.date.len() {
207            let visibility = values.header(index, geometry, inferred.unwrap_or(1.0))?;
208            if visibility.field as usize >= input.field_count {
209                return Err(invalid("FITS-IDI SOURCE_ID has no matching FIELD row"));
210            }
211            times.first.get_or_insert(visibility.time);
212            times.last_end = visibility.time + f64::from(visibility.interval);
213            let visibility_divisor =
214                if input.correlator == "VLBA" && input.correlator_version >= 4.17 {
215                    scale * f64::from(visibility.interval)
216                } else {
217                    scale
218                };
219            let decoded = DecodeScale {
220                visibility: scale,
221                visibility_divisor: weight_type.then_some(visibility_divisor as f32),
222                digital: digital.as_mut(),
223                levels: visibility.antennas.map(|antenna| {
224                    geometry
225                        .digitizer_levels
226                        .get(&antenna)
227                        .copied()
228                        .unwrap_or_default()
229                }),
230                scan,
231            };
232            buffers.append_row(columns, &values, index, visibility, decoded)?;
233        }
234    }
235    Ok(())
236}

238struct InputColumns {
239    uvw: [String; 3],
240    offset_time: bool,
241    interval: bool,
242    field: Option<&'static str>,
243    frequency: bool,
244    weight: bool,
245}
246
247impl InputColumns {
248    fn resolve(hdu: &FitsHdu) -> MsResult<Self> {
249        let HduInfo::TableInfo {
250            column_descriptions,
251            ..
252        } = &hdu.info
253        else {
254            return Err(invalid("UV_DATA is not a FITS binary table"));
255        };
256        let has = |name: &str| column_descriptions.iter().any(|column| column.name == name);
257        let [u, v, w] = ["UU", "VV", "WW"].map(|prefix| {
258            column_descriptions
259                .iter()
260                .find(|column| column.name.contains(prefix))
261                .map(|column| column.name.clone())
262                .ok_or_else(|| invalid("Missing FITS-IDI UVW coordinates"))
263        });
264        Ok(Self {
265            uvw: [u?, v?, w?],
266            offset_time: has("TIME"),
267            interval: has("INTTIM"),
268            field: ["SOURCE_ID", "SOURCE"].into_iter().find(|name| has(name)),
269            frequency: has("FREQID"),
270            weight: has("WEIGHT"),
271        })
272    }
273}

275fn column<T: ReadsCol>(
276    fits: &mut FitsFile,
277    hdu: &FitsHdu,
278    name: &str,
279    rows: &Range<usize>,
280    width: usize,
281) -> MsResult<Vec<T>> {
282    let values = hdu.read_col_range(fits, name, rows).map_err(invalid)?;
283    if values.len() != rows.len() * width {
284        return Err(invalid(format!(
285            "FITS-IDI column {name} requires {width} values per row"
286        )));
287    }
288    Ok(values)
289}
290
291fn infer_interval(
292    fits: &mut FitsFile,
293    hdu: &FitsHdu,
294    columns: &InputColumns,
295    rows: usize,
296) -> MsResult<f32> {
297    let mut first = None;
298    for start in (0..rows).step_by(1024) {
299        let range = start..(start + 1024).min(rows);
300        let dates = column::<f64>(fits, hdu, "DATE", &range, 1)?;
301        let offsets = if columns.offset_time {
302            column(fits, hdu, "TIME", &range, 1)?
303        } else {
304            vec![0.0; range.len()]
305        };
306        for (date, offset) in dates.into_iter().zip(offsets) {
307            let now = (date - 2_400_000.5 + offset) * DAY;
308            let start = *first.get_or_insert(now);
309            if now > start {
310                return Ok((now - start) as f32);
311            }
312        }
313    }
314    Ok(1.0)
315}

317struct RowBlock {
318    date: Vec<f64>,
319    time: Vec<f64>,
320    baseline: Vec<f64>,
321    uvw: [Vec<f64>; 3],
322    field: Vec<i32>,
323    frequency: Vec<i32>,
324    interval: Vec<f32>,
325    flux: Vec<f32>,
326    weights: Vec<f32>,
327}
328
329impl RowBlock {
330    fn read(
331        fits: &mut FitsFile,
332        hdu: &FitsHdu,
333        names: &InputColumns,
334        layout: &DataLayout,
335        rows: Range<usize>,
336    ) -> MsResult<Self> {
337        let [u, v, w] = names
338            .uvw
339            .each_ref()
340            .map(|name| column(fits, hdu, name, &rows, 1));
341        let uvw = [u?, v?, w?];
342        let mut integer = |name: Option<&str>| match name {
343            Some(name) => column(fits, hdu, name, &rows, 1),
344            None => Ok(vec![1; rows.len()]),
345        };
346        let field = integer(names.field)?;
347        let frequency = integer(names.frequency.then_some("FREQID"))?;
348        Ok(Self {
349            field,
350            frequency,
351            uvw,
352            date: column(fits, hdu, "DATE", &rows, 1)?,
353            baseline: column(fits, hdu, "BASELINE", &rows, 1)?,
354            time: if names.offset_time {
355                column(fits, hdu, "TIME", &rows, 1)?
356            } else {
357                vec![0.0; rows.len()]
358            },
359            interval: if names.interval {
360                column(fits, hdu, "INTTIM", &rows, 1)?
361            } else {
362                Vec::new()
363            },
364            flux: column(fits, hdu, "FLUX", &rows, layout.floats())?,
365            weights: if names.weight && layout.components == 2 {
366                column(
367                    fits,
368                    hdu,
369                    "WEIGHT",
370                    &rows,
371                    layout.bands * layout.correlations.len(),
372                )?
373            } else {
374                Vec::new()
375            },
376        })
377    }
378
379    fn header(
380        &self,
381        row: usize,
382        geometry: &Geometry,
383        inferred_interval: f32,
384    ) -> MsResult<VisibilityHeader> {
385        let baseline = self.baseline[row] as f32;
386        let packed = baseline as i32;
387        let array = (100.0 * (f64::from(baseline) - f64::from(packed) + 0.001)) as i32;
388        let [a, b] = [packed / 256, packed % 256].map(|number| {
389            geometry
390                .antennas
391                .get(&number)
392                .copied()
393                .ok_or_else(|| invalid(format!("Unknown FITS-IDI antenna number {number}")))
394        });
395        let (mut antennas, mut uvw) = (
396            [a?, b?],
397            self.uvw
398                .each_ref()
399                .map(|column| column[row] * SPEED_OF_LIGHT),
400        );
401        let reverse = antennas[0] > antennas[1];
402        if reverse {
403            antennas.swap(0, 1);
404            uvw = uvw.map(|value| -value);
405        }
406        let time = (self.date[row] - 2_400_000.5 + self.time[row]) * DAY;
407        let interval = self.interval.get(row).copied().unwrap_or(inferred_interval);
408        let field = self.field[row] - 1;
409        let frequency = self.frequency[row] - 1;
410        if array < 0 || field < 0 || frequency < 0 || !time.is_finite() || !interval.is_finite() {
411            return Err(invalid("Invalid FITS-IDI row metadata"));
412        }
413        Ok(VisibilityHeader {
414            antennas,
415            uvw,
416            reverse,
417            time,
418            interval,
419            array,
420            field,
421            frequency,
422        })
423    }
424}

437struct DecodeScale<'a> {
438    visibility: f64,
439    visibility_divisor: Option<f32>,
440    digital: Option<&'a mut DigitalCorrection>,
441    levels: [i32; 2],
442    scan: i32,
443}
444
445struct DecodeBuffers<'a> {
446    layout: &'a DataLayout,
447    bandwidths: &'a [f64],
448    data: Array2<Complex32>,
449    flags: Array2<bool>,
450    weights: Array2<f32>,
451    sigmas: Option<Array2<f32>>,
452    row_weights: Array1<f32>,
453    row_sigmas: Array1<f32>,
454    scratch: Vec<f32>,
455}
456
457impl<'a> DecodeBuffers<'a> {
458    fn new(layout: &'a DataLayout, bandwidths: &'a [f64]) -> Self {
459        let hands = layout.correlations.len();
460        let shape = (hands, layout.channels).f();
461        Self {
462            layout,
463            bandwidths,
464            data: Array2::zeros(shape),
465            flags: Array2::from_elem(shape, false),
466            weights: Array2::zeros(shape),
467            sigmas: (layout.components == 3).then(|| Array2::zeros(shape)),
468            row_weights: Array1::zeros(hands),
469            row_sigmas: Array1::zeros(hands),
470            scratch: Vec::with_capacity(layout.channels),
471        }
472    }
473
474    fn append_row(
475        &mut self,
476        columns: &mut ImportColumns,
477        input: &RowBlock,
478        row: usize,
479        header: VisibilityHeader,
480        mut scale: DecodeScale<'_>,
481    ) -> MsResult<()> {
482        let interval = f64::from(header.interval);
483        let autocorrelation = header.antennas[0] == header.antennas[1];
484        for band in 0..self.layout.bands {
485            let window = header.frequency as usize * self.layout.bands + band;
486            let bandwidth = *self
487                .bandwidths
488                .get(window)
489                .ok_or_else(|| invalid("Missing FITS-IDI frequency band"))?;
490            self.load_band(input, row, band, header.reverse);
491            if let Some(digital) = &mut scale.digital {
492                digital.apply(
493                    autocorrelation,
494                    scale.levels,
495                    &self.layout.correlations,
496                    &mut self.data,
497                    &self.weights,
498                )?;
499            }
500            if let Some(divisor) = scale.visibility_divisor {
501                self.data
502                    .zip_mut_with(&self.weights, |sample, &weight| *sample /= divisor * weight);
503            }
504            let weight_scale =
505                scale.visibility * interval * bandwidth * if autocorrelation { 1.0 } else { 2.0 };
506            self.finish_weights(weight_scale);
507            columns.append(
508                MainRow {
509                    antennas: header.antennas,
510                    uvw: header.uvw,
511                    time: header.time,
512                    centroid: header.time + 0.5 * interval,
513                    interval: Some(interval),
514                    array: header.array,
515                    field: header.field,
516                    window: window as i32,
517                    scan: scale.scan,
518                },
519                MainSamples {
520                    data: &self.data,
521                    flags: &self.flags,
522                    weight: &self.row_weights,
523                    sigma: &self.row_sigmas,
524                    weight_spectrum: self.sigmas.as_ref().map(|_| &self.weights),
525                    sigma_spectrum: self.sigmas.as_ref(),
526                },
527            )?;
528        }
529        Ok(())
530    }
531
532    fn load_band(&mut self, input: &RowBlock, row: usize, band: usize, reverse: bool) {
533        let layout = self.layout;
534        let hands = layout.correlations.len();
535        let offset = row * layout.floats() + band * hands * layout.channels * layout.components;
536        let order = if reverse {
537            &layout.reversed
538        } else {
539            &layout.normal
540        };
541        for channel in 0..layout.channels {
542            for (hand, &destination) in order.iter().enumerate() {
543                let pos = offset + (channel * hands + hand) * layout.components;
544                let signed = if layout.components == 3 {
545                    input.flux[pos + 2]
546                } else {
547                    input
548                        .weights
549                        .get((row * layout.bands + band) * hands + hand)
550                        .copied()
551                        .unwrap_or(1.0)
552                };
553                let imaginary = if reverse {
554                    input.flux[pos + 1]
555                } else {
556                    -input.flux[pos + 1]
557                };
558                self.data[[destination, channel]] = Complex32::new(input.flux[pos], imaginary);
559                self.flags[[destination, channel]] = signed <= 0.0;
560                self.weights[[destination, channel]] = signed.abs();
561            }
562        }
563    }
564
565    fn finish_weights(&mut self, scale: f64) {
566        if let Some(sigmas) = &mut self.sigmas {
567            self.weights
568                .mapv_inplace(|weight| (f64::from(weight) * scale) as f32);
569            sigmas.zip_mut_with(&self.weights, |sigma, &weight| {
570                *sigma = sigma_for_weight(weight)
571            });
572            median_spectrum(&self.weights, &mut self.row_weights, &mut self.scratch);
573            median_spectrum(sigmas, &mut self.row_sigmas, &mut self.scratch);
574        } else {
575            for ((weight, sigma), &input) in self
576                .row_weights
577                .iter_mut()
578                .zip(&mut self.row_sigmas)
579                .zip(self.weights.column(0))
580            {
581                *weight = (f64::from(input) * scale) as f32;
582                *sigma = sigma_for_weight(*weight);
583            }
584        }
585    }
586}
587
588fn sigma_for_weight(weight: f32) -> f32 {
589    if weight > 0.0 {
590        weight.sqrt().recip()
591    } else {
592        1.0
593    }
594}

596fn median_spectrum(spectrum: &Array2<f32>, output: &mut Array1<f32>, entries: &mut Vec<f32>) {
597    for (result, row) in output.iter_mut().zip(spectrum.rows()) {
598        entries.clear();
599        entries.extend(row.iter().copied());
600        // casacore's partialMedians uses the lower middle for even lengths.
601        let middle = (entries.len() - 1) / 2;
602        *result = *entries.select_nth_unstable_by(middle, f32::total_cmp).1;
603    }
