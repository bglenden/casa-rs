5/// Validate channel ranges before mutating flags. Public CASA-style calls
6/// have already saved any requested backup by this stage.
7pub(super) fn check_channel_selections(
8    ms: &MeasurementSet,
9    selections: &BTreeMap<i32, ChannelSelection>,
10) -> Result<(), FlaggingError> {
11    let path = display_ms_path(ms);
12    let spectral_window = ms
13        .spectral_window()
14        .map_err(|source| FlaggingError::MutateFlags {
15            path: path.clone(),
16            reason: format!("open SPECTRAL_WINDOW: {source}"),
17        })?;
18    for (&spw, selection) in selections {
19        let resolved = usize::try_from(spw)
20            .map_err(|_| format!("SPW {spw} is negative"))
21            .and_then(|row| {
22                let count = spectral_window.num_chan(row).map_err(|source| {
23                    format!("read SPECTRAL_WINDOW.NUM_CHAN row {spw}: {source}")
24                })?;
25                usize::try_from(count).map_err(|_| format!("SPW {spw} has a negative NUM_CHAN"))
26            })
27            .and_then(|channels| {
28                selection
29                    .indices(channels)
30                    .map(|_| ())
31                    .map_err(|source| format!("resolve SPW {spw} channels: {source}"))
32            });
33        resolved.map_err(|reason| FlaggingError::MutateFlags {
34            path: path.clone(),
35            reason,
36        })?;
37    }
38    Ok(())
39}

42pub(super) enum SelectedFlagRowPolicy {
43    ChangedSamples,
44    AllSelectedRows,
45    FullySelectedRows,
46    Preserve,
47}
48
49pub(super) fn selected_flag_row_policy(
50    correlations: Option<&CorrelationSelection>,
51    unfiltered: SelectedFlagRowPolicy,
52) -> SelectedFlagRowPolicy {
53    match correlations {
54        Some(selection) if selection.decorated_stored_products() => {
55            SelectedFlagRowPolicy::FullySelectedRows
56        }
57        Some(_) => SelectedFlagRowPolicy::Preserve,
58        None => unfiltered,
59    }
60}

62pub(super) fn set_selected_flags(
63    ms: &mut MeasurementSet,
64    selected_rows: &[usize],
65    channel_selections: Option<&BTreeMap<i32, ChannelSelection>>,
66    correlations: Option<&CorrelationSelection>,
67    flag: bool,
68    row_policy: SelectedFlagRowPolicy,
69    cancellation: Option<&Cancellation>,
70) -> Result<ChangeSet, FlaggingError> {
71    let row_policy = if is_calibration_table(ms) {
72        SelectedFlagRowPolicy::Preserve
73    } else {
74        row_policy
75    };
76    let loaded = load_samples(
77        ms,
78        selected_rows,
79        channel_selections,
80        None,
81        correlations,
82        SampleLoadOptions {
83            cancellation,
84            ..SampleLoadOptions::default()
85        },
86    )?;
87    let mut updates = loaded.flags_by_row;
88    let mut changed = ChangeSet::default();
89    let mut touched_rows = std::collections::BTreeSet::new();
90    let mut coverage = BTreeMap::<usize, usize>::new();
91    for (index, sample) in loaded.samples.into_iter().enumerate() {
92        if index % 1024 == 0 {
93            check_cancelled(cancellation)?;
94        }
95        if row_policy == SelectedFlagRowPolicy::FullySelectedRows {
96            *coverage.entry(sample.row).or_default() += 1;
97        }
98        if sample.flag != flag
99            && let Some(flags) = updates.get_mut(&sample.row)
100            && let Some(value) = flags.get_mut(IxDyn(&[sample.corr, sample.chan]))
101            && *value != flag
102        {
103            *value = flag;
104            changed.changed_samples += 1;
105            touched_rows.insert(sample.row);
106        }
107    }
108    if row_policy == SelectedFlagRowPolicy::AllSelectedRows {
109        let path = display_ms_path(ms);
110        let row_flags = load_bool_cells(ms.main_table(), selected_rows, FLAG_ROW_COLUMN, &path)?;
111        for (&row, flag_row) in selected_rows.iter().zip(row_flags) {
112            if !flag_row && updates.get(&row).is_some_and(|flags| !flags.is_empty()) {
113                touched_rows.insert(row);
114            }
115        }
116    }
117    let fully_selected = coverage
118        .into_iter()
119        .filter_map(|(row, count)| (updates[&row].len() == count).then_some(row))
120        .collect::<std::collections::BTreeSet<_>>();
121    if !fully_selected.is_empty() {
122        let rows = fully_selected.iter().copied().collect::<Vec<_>>();
123        let path = display_ms_path(ms);
124        let row_flags = load_bool_cells(ms.main_table(), &rows, FLAG_ROW_COLUMN, &path)?;
125        touched_rows.extend(
126            rows.into_iter()
127                .zip(row_flags)
128                .filter_map(|(row, row_flag)| (row_flag != flag).then_some(row)),
129        );
130    }
131    stage_touched_flag_updates(
132        ms,
133        updates,
134        touched_rows,
135        |row| match row_policy {
136            SelectedFlagRowPolicy::FullySelectedRows => fully_selected.contains(&row),
137            SelectedFlagRowPolicy::Preserve => false,
138            _ => true,
139        },
140        &mut changed,
141        cancellation,
142    )?;
143    Ok(changed)
144}

146pub(super) fn apply_flag_mask_with_preloaded(
147    samples: &[Sample],
148    mask: &[bool],
149    mut updates: BTreeMap<usize, ArrayD<bool>>,
150    ms: &mut MeasurementSet,
151    update_flag_rows: bool,
152    flag_value: bool,
153) -> Result<ChangeSet, FlaggingError> {
154    let (touched_rows, changed_samples) = stage_flag_mask(samples, mask, &mut updates, flag_value);
155    let mut changed = ChangeSet {
156        changed_samples,
157        ..ChangeSet::default()
158    };
159    write_touched_flag_updates(
160        ms,
161        updates,
162        touched_rows,
163        |_| update_flag_rows,
164        &mut changed,
165    )?;
166    Ok(changed)
167}

169pub(super) fn stage_flag_mask(
170    samples: &[Sample],
171    mask: &[bool],
172    updates: &mut BTreeMap<usize, ArrayD<bool>>,
173    flag_value: bool,
174) -> (std::collections::BTreeSet<usize>, usize) {
175    let coordinates = samples
176        .iter()
177        .zip(mask)
178        .filter(|(_, flag)| **flag)
179        .map(|(sample, _)| (sample.row, sample.corr, sample.chan));
180    stage_flag_coordinates(coordinates, updates, flag_value)
181}

183pub(super) fn stage_flag_coordinates(
184    coordinates: impl Iterator<Item = (usize, usize, usize)>,
185    updates: &mut BTreeMap<usize, ArrayD<bool>>,
186    flag_value: bool,
187) -> (std::collections::BTreeSet<usize>, usize) {
188    let mut changed_samples = 0;
189    let mut touched_rows = std::collections::BTreeSet::<usize>::new();
190    for (row, corr, chan) in coordinates {
191        let Some(flags) = updates.get_mut(&row) else {
192            continue;
193        };
194        if let Some(value) = flags.get_mut(IxDyn(&[corr, chan]))
195            && *value != flag_value
196        {
197            *value = flag_value;
198            changed_samples += 1;
199            touched_rows.insert(row);
200        }
201    }
202    (touched_rows, changed_samples)
203}

205pub(super) fn write_touched_flag_updates(
206    ms: &mut MeasurementSet,
207    updates: BTreeMap<usize, ArrayD<bool>>,
208    touched_rows: std::collections::BTreeSet<usize>,
209    update_flag_row: impl Fn(usize) -> bool,
210    changed: &mut ChangeSet,
211) -> Result<(), FlaggingError> {
212    stage_touched_flag_updates(ms, updates, touched_rows, update_flag_row, changed, None)
213}

215fn stage_touched_flag_updates(
216    ms: &mut MeasurementSet,
217    mut updates: std::collections::BTreeMap<usize, ArrayD<bool>>,
218    touched_rows: std::collections::BTreeSet<usize>,
219    update_flag_row: impl Fn(usize) -> bool,
220    changed: &mut ChangeSet,
221    cancellation: Option<&Cancellation>,
222) -> Result<(), FlaggingError> {
223    let path = display_ms_path(ms);
224    let row_flags = !is_calibration_table(ms);
225    let mut prepared = Vec::with_capacity(touched_rows.len());
226    for (index, row) in touched_rows.into_iter().enumerate() {
227        if index % FLAG_BATCH_ROWS == 0 {
228            check_cancelled(cancellation)?;
229        }
230        let Some(flags) = updates.remove(&row) else {
231            continue;
232        };
233        prepared.push((row, flags));
234    }
235    let flag_row_updates = {
236        let selected = prepared
237            .iter()
238            .filter(|(row, _)| row_flags && update_flag_row(*row));
239        let rows = selected.clone().map(|(row, _)| *row).collect::<Vec<_>>();
240        let current_flag_rows = if row_flags {
241            load_bool_cells(ms.main_table(), &rows, FLAG_ROW_COLUMN, &path)?
242        } else {
243            Vec::new()
244        };
245        selected
246            .zip(current_flag_rows)
247            .filter_map(|((row, flags), current)| {
248                let flag_row = flags.iter().all(|flag| *flag);
249                (current != flag_row).then_some((*row, flag_row))
250            })
251            .collect::<Vec<_>>()
252    };
253    ms.main_table_mut()
254        .reserve_array_cell_updates(FLAG_COLUMN, prepared.len());
255    {
256        let columns: &[&str] = if row_flags {
257            &[FLAG_COLUMN, FLAG_ROW_COLUMN]
258        } else {
259            &[FLAG_COLUMN]
260        };
261        let mut writer = ms
262            .main_table_mut()
263            .row_accessor_mut()
264            .prepare(columns)
265            .map_err(|source| FlaggingError::MutateFlags {
266                path: path.clone(),
267                reason: format!("prepare flag updates: {source}"),
268            })?;
269        let flag_slot = writer
270            .column_index(FLAG_COLUMN)
271            .expect("prepared FLAG slot");
272        let flag_row_slot = writer.column_index(FLAG_ROW_COLUMN);
273        let batch_rows = FLAG_BATCH_ROWS;
274        #[cfg(feature = "flagdata-test-hooks")]
275        let batch_rows = test_hooks::flag_stage_batch_rows(batch_rows);
276        for batch in prepared.chunks(batch_rows) {
277            check_cancelled(cancellation)?;
278            #[cfg(feature = "flagdata-test-hooks")]
279            test_hooks::observe(test_hooks::Checkpoint::FlagStageStarted(batch.len()));
280            for (row, flags) in batch {
281                writer
282                    .seek(*row)
283                    .and_then(|()| {
284                        writer
285                            .set_value_at(flag_slot, Value::Array(ArrayValue::Bool(flags.clone())))
286                    })
287                    .map_err(|source| FlaggingError::MutateFlags {
288                        path: path.clone(),
289                        reason: format!("write FLAG row {row}: {source}"),
290                    })?;
291            }
292            #[cfg(feature = "flagdata-test-hooks")]
293            test_hooks::observe(test_hooks::Checkpoint::FlagStageFinished(batch.len()));
294        }
295        for batch in flag_row_updates.chunks(batch_rows) {
296            check_cancelled(cancellation)?;
297            let flag_row_slot = flag_row_slot.expect("prepared FLAG_ROW slot");
298            for (row, flag_row) in batch {
299                writer
300                    .seek(*row)
301                    .and_then(|()| {
302                        writer.set_value_at(
303                            flag_row_slot,
304                            Value::Scalar(ScalarValue::Bool(*flag_row)),
305                        )
306                    })
307                    .map_err(|source| FlaggingError::MutateFlags {
308                        path: path.clone(),
309                        reason: format!("write FLAG_ROW row {row}: {source}"),
310                    })?;
311            }
312        }
313    }
314    changed.changed_flag_row_rows += flag_row_updates.len();
315    for (row, _) in prepared {
316        changed.changed_rows += 1;
317        changed.changed_row_indices.push(row);
318    }
319    Ok(())
