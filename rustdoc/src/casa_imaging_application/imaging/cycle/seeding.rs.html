2//! The initial major cycle, from an empty model or a start model (#841
3//! Phase 6b, #970 review).
4//!
5//! A start model is the model's first generation, in the units a run
6//! publishes its `.model` in. Under a direction-independent response, or
7//! flat sky, those are the physical units the pass predicts from, and the
8//! initial pass grids the residual of the start model. Under flat noise or
9//! pbsquare (a mosaic's or AW kernel set's `normtype`), `.model` is the
10//! physical model times the PB (flat noise) or PB² (pbsquare), and CASA
11//! divides a model through the PB or PB² of the weights just before it
12//! predicts (`SIImageStore::divideModelByWeight`), once `makePSF` has
13//! formed them. Such a run takes CASA's order: the initial pass forms the
14//! PSF, weights and sums with no model, and the start model, made physical
15//! with those weights, is the first generation of a refresh of that pass's
16//! residual. That costs one more pass over the data, for such runs only.
17//! Either way, a run started from another run's model with no minor cycle
18//! gives that run's products (the T1 laws; CASA 6.7.6 does the same on a
19//! 14-pointing mosaic).
20

31/// The response a start model is made physical with before the first
32/// major cycle predicts it: a flat-noise or pbsquare one, through the PB
33/// (flat noise) or PB² (pbsquare); `None` without a start model, or when
34/// its units are physical already.
35pub(super) fn reseed_response(
36    start_model: Option<&StartModel>,
37    response: Option<MinorCycleImageResponse>,
38) -> Option<MinorCycleImageResponse> {
39    start_model.and(response).filter(|response| {
40        matches!(
41            response.normalization(),
42            ProductNormalization::FlatNoise | ProductNormalization::PbSquare
43        )
44    })
45}

47impl Run<'_> {
48    /// The initial major cycle: data and PSF, from the start model when there
49    /// is one (its residual) and from an empty model otherwise; it writes
50    /// visibilities when it is also final. A run that reads its PSF or
51    /// residual forms what it does not read ([`super::continuation`]).
52    pub(super) fn initial(&mut self, last: bool) -> Result<Major, ImagingError> {
53        match (self.forms.psf, self.forms.residual) {
54            (true, true) => {}
55            (true, false) => {
56                self.measure_row_work()?;
57                return self.psf_pass();
58            }
59            (false, true) => {
60                self.measure_row_work()?;
61                let read = self.read_existing()?;
62                return self.refresh(read.completion, (Vec::new(), Reservation::none()), last);
63            }
64            (false, false) => return self.read_existing(),
65        }
66        self.measure_row_work()?;
67        match self.reseed {
68            Some(response) => {
69                let unseeded = self.initial_pass(false, false)?;
70                self.reseed(unseeded, response, last)
71            }
72            None => self.initial_pass(self.start_model.is_some(), last),
73        }
74    }
75
76    /// The initial pass, from the start model when `seeded`.
77    fn initial_pass(&mut self, seeded: bool, last: bool) -> Result<Major, ImagingError> {
78        // The model, and a start model's seeding beside it, are admitted
79        // with the pass before either is allocated, so the pass plans its
80        // waves and chunks in what seeding leaves (#970 review).
81        let seeding = if seeded { self.seeding_bytes() } else { 0 };
82        let Admitted {
83            value: plan,
84            reservation: mut pass,
85        } = self.admit_pass(
86            self.initial_modes(),
87            seeded,
88            self.lifecycle.resident_bytes() + self.write_bytes(last) + seeding,
89            self.writes(last),
90        )?;
91        let generation = match &self.start_model {
92            Some(start) if seeded => {
93                // Released once the generation holds the samples.
94                let _seeding = pass.split(seeding);
95                let samples = start.main_samples().map_err(ImagingError::StartModel)?;
96                self.lifecycle.initial_from_start_model(&samples)?
97            }
98            _ => self.lifecycle.initial_empty()?,
99        };
100        let model = self.lifecycle.prepare_final_model(generation, [])?;
101        let cycle = MajorCycle::initial(self.problem, model, self.normal_storage(plan.residency)?)?;
102        let major = self.reconcile(cycle, self.initial_modes(), seeded, last, plan)?;
103        self.retain(&major, pass)?;
104        Ok(major)
105    }

107    /// The residual of the start model, made physical with `unseeded`'s
108    /// weights under `response`: a refresh of the initial pass, whose PSF,
109    /// weights and sums carry over.
110    fn reseed(
111        &mut self,
112        unseeded: Major,
113        response: MinorCycleImageResponse,
114        last: bool,
115    ) -> Result<Major, ImagingError> {
116        let start = self
117            .start_model
118            .clone()
119            .expect("a run that reseeds has a start model");
120        // The retained charge covers the normal state, and the empty model,
121        // which the seeded generation replaces at the same size.
122        let (normal_state, empty) = unseeded.completion.into_parts();
123        drop(empty);
124        let seeding = self.seeding_bytes();
125        let Admitted {
126            value: plan,
127            reservation: mut pass,
128        } = self.admit_pass(
129            ModeSet::DATA,
130            true,
131            self.write_bytes(last) + seeding,
132            self.writes(last),
133        )?;
134        let generation = {
135            let _seeding = pass.split(seeding);
136            self.physical_generation(&start, &normal_state, response)?
137        };
138        let model = self.lifecycle.prepare_final_model(generation, [])?;
139        let cycle = MajorCycle::refresh(
140            self.problem,
141            normal_state,
142            model,
143            self.normal_storage(plan.residency)?,
144        )?;
145        let major = self.reconcile(cycle, ModeSet::DATA, true, last, plan)?;
146        self.retain(&major, pass)?;
147        Ok(major)
148    }

150    /// The first generation from `start`, its apparent samples divided
151    /// through the PB (flat noise) or PB² (pbsquare) of `normal`'s weights
152    /// where that supports them and zero elsewhere (`divideModelByWeight`),
153    /// as the minor cycle maps solved values to the physical model.
154    fn physical_generation(
155        &self,
156        start: &StartModel,
157        normal: &FinalNormalState,
158        response: MinorCycleImageResponse,
159    ) -> Result<ModelGeneration, ImagingError> {
160        let [nx, ny] = start.image_size();
161        let polarizations = start.polarizations();
162        let apparent = start.main_samples().map_err(ImagingError::StartModel)?;
163        let units = ImagingError::StartModelUnits;
164        let mut physical = Vec::with_capacity(apparent.len());
165        // Canonical order: spectral plane, polarization, y, x (x fastest);
166        // the normal state's planes are x-major (x · ny + y). Every Taylor
167        // term divides by term 0's weights of its polarization
168        // (`SIImageStoreMultiTerm::divideModelByWeight`).
169        let taylor = (normal.basis() == casa_imaging_deconvolution::NormalBasis::Taylor)
170            .then(|| normal.read_window(normal.slab().core_range()))
171            .transpose()?;
172        for (plane, samples) in apparent.chunks(nx * ny).enumerate() {
173            let principal = taylor.as_ref().map(|window| {
174                window
175                    .normal_moment(0, plane % polarizations)
176                    .expect("the zeroth normal moment")
177            });
178            let (sensitivity, sum_weight) = match principal {
179                Some(moment) => (
180                    std::borrow::Cow::Borrowed(moment.sensitivity()),
181                    moment.sum_weight(),
182                ),
183                None => {
184                    let weights =
185                        normal.read_plane(0, plane / polarizations, plane % polarizations)?;
186                    (weights.read_sensitivity()?, weights.sum_weight())
187                }
188            };
189            // `SIImageStore::divideModelByWeight` skips a plane whose PB's
190            // peak is zero: no sample reached it, or its rows all miss the
191            // grid (an AW weight image with no sum, which
192            // `divideImageByWeightVal` zeroes). Its model stays as it is.
193            let bound = MosaicSensitivity::try_new(&sensitivity).map_err(units)?;
194            let Some(bound) = bound.filter(|_| sum_weight > 0.0) else {
195                physical.extend_from_slice(samples);
196                continue;
197            };
198            let bound = bound.with_normal_sum_weight(sum_weight).map_err(units)?;
199            for y in 0..ny {
200                for x in 0..nx {
201                    let value = bound
202                        .apparent_to_physical(
203                            f64::from(samples[y * nx + x]),
204                            x * ny + y,
205                            response.normalization(),
206                            response.policy(),
207                        )
208                        .map_err(units)?;
209                    physical.push(value as f32);
210                }
211            }
212        }
213        Ok(self.lifecycle.initial_from_start_model(&physical)?)
214    }

216    /// What seeding the model from the start model holds besides the model
217    /// ([`StartModel::seeding_bytes`], and its conversion when it is made
218    /// physical); 0 without one.
219    pub(super) fn seeding_bytes(&self) -> u64 {
220        self.start_model.as_ref().map_or(0, |start| {
221            start.seeding_bytes()
222                + if self.reseed.is_some() {
223                    start.conversion_bytes()
224                } else {
225                    0
226                }
227        })
228    }
