12/// Decodes the scalar epoch and its offsets stored by `ObsInfo`.
13pub(in crate::tools) fn observation_epoch(
14    record: &RecordArg,
15    log: &mut Vec<LogLine>,
16) -> Result<Option<MEpoch>, &'static str> {
17    for key in ["m0", "m1", "m2"] {
18        let Some(Variant::Record(quantity)) = record.0.get(key) else {
19            break;
20        };
21        if let Some(Variant::Numbers(values) | Variant::Array { values, .. }) =
22            quantity.get("value")
23            && !values.is_empty()
24            && !quantity.is_double_vector()
25        {
26            return Err("Cannot convert to QuantumVectorDouble");
27        }
28    }
29    let measures = match Measures::read(record) {
30        Ok(measures) => measures,
31        Err(error) => {
32            log.push(LogLine::error(format!(
33                "{}Non-measure type value in measure conversion",
34                error.message(),
35            )));
36            return Ok(None);
37        }
38    };
39    if let Some(warning) = &measures.warning {
40        log.push(LogLine::warning(warning.clone()));
41    }
42    match measures.as_epoch() {
43        Some(epoch) => Ok(Some(epoch)),
44        None => {
45            log.push(LogLine::error(
46                "Record value does not contain an Epoch measure.",
47            ));
48            Ok(None)
49        }
50    }
51}

61impl Measures {
62    fn as_epoch(&self) -> Option<MEpoch> {
63        let CasaMeasure::Epoch(epoch) = self.values.first()?.to_casa().ok()? else {
64            return None;
65        };
66        match &self.offset {
67            Some(offset) => Some(epoch.with_offset(offset.as_epoch()?)),
68            None => Some(epoch),
69        }
70    }

72    pub(super) fn read(record: &RecordArg) -> Result<Self, HolderError> {
73        Self::read_fields(&record.0, 0)
74    }
75
76    pub(super) fn read_variant(
77        record: &crate::convert::VariantRecord,
78    ) -> Result<Self, HolderError> {
79        Self::read_fields(record, 0)
80    }

82    /// Decode the scalar direction and its reference offsets before the
83    /// observing frame is attached to CASA's rise/set conversion.
84    pub(super) fn direction_without_frame(
85        &self,
86        target: DirectionRef,
87    ) -> Result<MDirection, String> {
88        let mut value = self.values[0];
89        if let Some(offset) = &self.offset {
90            let refer = value
91                .refer
92                .parse()
93                .map_err(|error: MeasureError| error.to_string())?;
94            let origin = offset.direction_without_frame(refer)?;
95            value = add(value, Measure::from_casa(&CasaMeasure::Direction(origin)))?;
96        }
97        let CasaMeasure::Direction(direction) = value.to_casa()? else {
98            return Err("Non-direction source".into());
99        };
100        direction
101            .convert_to(target, &MeasFrame::new())
102            .map_err(|error| error.to_string())
103    }

105    fn read_fields(fields: &BTreeMap<String, Variant>, depth: usize) -> Result<Self, HolderError> {
106        if depth > 128 {
107            return Err(HolderError::Invalid(
108                "measure offsets are too deeply nested (#1252)".into(),
109            ));
110        }
111        let mut fields = fields.clone();
112        let offset = match fields.remove("offset") {
113            Some(Variant::Record(offset)) => Some(Box::new(
114                Self::read_fields(&offset, depth + 1)?.into_reference_origin(),
115            )),
116            Some(_) => {
117                return Err(HolderError::Invalid(
118                    "Illegal offset in MeasureHolder::fromRecord\n".into(),
119                ));
120            }
121            None => None,
122        };
123        let mut length = None;
124        let mut arrays = Vec::new();
125        for (index, key) in ["m0", "m1", "m2"].into_iter().enumerate() {
126            let Some(Variant::Record(quantity)) = fields.get_mut(key) else {
127                break;
128            };
129            let n = match quantity.get("value") {
130                Some(Variant::Numbers(v) | Variant::Array { values: v, .. }) => v.len(),
131                _ => 1,
132            };
133            if length.is_some_and(|nvalues| nvalues != n) {
134                return Err(HolderError::Invalid(format!(
135                    "Illegal number of values in MeasureHolder m{index}\n"
136                )));
137            }
138            length = Some(n);
139            if let Some(Variant::Numbers(v) | Variant::Array { values: v, .. }) =
140                quantity.get("value")
141            {
142                if v.is_empty() || !quantity.is_double_vector() {
143                    return Err(HolderError::Invalid(
144                        "Illegal quantity in MeasureHolder::fromRecord\n".into(),
145                    ));
146                }
147                let data = v.clone();
148                let mut scalar = (**quantity).clone();
149                scalar.insert("value".into(), Variant::Number(data[0]));
150                *quantity = scalar.into();
151                arrays.push((index, data));
152            }
153        }
154        let mut definition = Measure::record_definition(&fields)?;
155        let mut values = Vec::with_capacity(length.unwrap_or(1));
156        for i in 0..length.unwrap_or(1) {
157            for (index, array) in &arrays {
158                let q = &mut definition.quantities[*index];
159                *q = casa_core::quanta::Quantity::with_unit(array[i], q.unit().clone());
160            }
161            values.push(Measure::from_values(
162                definition.kind,
163                definition.refer,
164                &definition.quantities,
165            )?);
166        }
167        let warning = definition.warning;
168        if offset
169            .as_ref()
170            .is_some_and(|offset| offset.values[0].kind() != values[0].kind())
171        {
172            return Err(HolderError::Invalid(
173                "Unmatched offset type in MeasureHolder::fromRecord\n".into(),
174            ));
175        }
176        Ok(Self {
177            values,
178            vector: length.is_some_and(|n| n > 1),
179            warning,
180            offset,
181        })
182    }

184    /// A reference offset and a frame element use the first MeasureHolder value.
185    pub(super) fn into_reference_origin(mut self) -> Self {
186        self.values.truncate(1);
187        self.vector = false;
188        self.offset = self.offset.map(|v| Box::new(v.into_reference_origin()));
189        self
190    }

192    /// Express absolute values relative to one output reference origin.
193    pub(super) fn relative_to(
194        mut self,
195        origin: Self,
196        state: &mut State,
197    ) -> pyo3::PyResult<Result<Self, String>> {
198        if origin.values[0].kind() != self.values[0].kind() {
199            return Ok(Err("Non-conforming offset measure type".into()));
200        }
201        let origin = origin.into_reference_origin();
202        let resolved = match origin.resolve(state, self.values[0].refer)? {
203            Ok(values) => values.values[0],
204            Err(error) => return Ok(Err(error)),
205        };
206        for value in &mut self.values {
207            *value = match offset_value(*value, resolved, -1.0) {
208                Ok(value) => value,
209                Err(error) => return Ok(Err(error)),
210            };
211        }
212        self.offset = Some(Box::new(origin));
213        Ok(Ok(self))
214    }

216    /// Convert absolute coordinates and, when supplied, subtract the output
217    /// reference origin after the rotation or scale conversion.
218    pub(super) fn resolve_relative(
219        &self,
220        state: &mut State,
221        target: &'static str,
222        origin: Option<Self>,
223    ) -> pyo3::PyResult<Result<Self, String>> {
224        if origin
225            .as_ref()
226            .is_some_and(|offset| offset.values[0].kind() != self.values[0].kind())
227        {
228            return Ok(Err("Non-conforming offset measure type".into()));
229        }
230        match (self.resolve(state, target)?, origin) {
231            (Ok(converted), Some(origin)) => converted.relative_to(origin, state),
232            (result, _) => Ok(result),
233        }
234    }

261    pub(super) fn to_value(&self) -> CasaValue {
262        let first = self.values[0];
263        let mut out = if self.vector {
264            let components = self
265                .values
266                .iter()
267                .map(|v| v.quantity_values())
268                .collect::<Vec<_>>();
269            let mut fields = vec![
270                (
271                    "type".to_owned(),
272                    CasaValue::Text(first.kind().record_type().to_owned()),
273                ),
274                ("refer".to_owned(), CasaValue::Text(first.refer.to_owned())),
275            ];
276            for (i, &unit) in first.quantity_units().iter().enumerate() {
277                fields.push((
278                    format!("m{i}"),
279                    CasaValue::record([
280                        ("unit", CasaValue::Text(unit.to_owned())),
281                        (
282                            "value",
283                            CasaValue::Floats {
284                                shape: vec![self.values.len()],
285                                values: components.iter().map(|v| v[i]).collect(),
286                            },
287                        ),
288                    ]),
289                ));
290            }
291            CasaValue::record(fields)
292        } else {
293            first.to_value()
294        };
295        if let Some(offset) = &self.offset {
296            let CasaValue::Record(fields) = &mut out else {
297                unreachable!()
298            };
299            fields.push(("offset".into(), offset.to_value()));
300            fields.sort_by(|a, b| a.0.cmp(&b.0));
301        }
302        out
303    }

305    pub(super) fn resolve(
306        &self,
307        state: &mut State,
308        target: &'static str,
309    ) -> pyo3::PyResult<Result<Self, String>> {
310        let offset = if let Some(offset) = &self.offset {
311            match offset.resolve(state, self.values[0].refer)? {
312                Ok(offset) => Some(offset),
313                Err(error) => return Ok(Err(error)),
314            }
315        } else {
316            None
317        };
318        let plan = if self.values[0].refer != target
319            && matches!(
320                self.values[0].value,
321                MeasureValue::Baseline(_) | MeasureValue::Uvw(_) | MeasureValue::EarthMagnetic(_)
322            ) {
323            match super::geometry::Conversion::new(state, self.values[0], target)? {
324                Ok(plan) => Some(plan),
325                Err(error) => return Ok(Err(error)),
326            }
327        } else {
328            None
329        };
330        let mut converted = Vec::with_capacity(self.values.len());
331        for value in &self.values {
332            let value = if let Some(offset) = &offset {
333                match add(*value, offset.values[0]) {
334                    Ok(value) => value,
335                    Err(error) => return Ok(Err(error)),
336                }
337            } else {
338                *value
339            };
340            let result = if let Some(plan) = &plan {
341                plan.apply(value)
342            } else {
343                super::frame::convert(state, &value, target)?
344            };
345            match result {
346                Ok(value) => converted.push(value),
347                Err(error) => return Ok(Err(error)),
348            }
349        }
350        Ok(Ok(Self {
351            values: converted,
352            vector: self.vector,
353            warning: None,
354            offset: None,
355        }))
356    }
357}

359fn add(base: Measure, offset: Measure) -> Result<Measure, String> {
360    offset_value(base, offset, 1.0)
361}

363/// Materialize the first value of a measure and its nested origins in a
364/// supplied frame without rebuilding that frame from the tool state.
365pub(super) fn absolute_value(
366    base: Measure,
367    origin: Option<&Measures>,
368    convert: &mut impl FnMut(Measure, &'static str) -> Result<Measure, String>,
369) -> Result<Measure, String> {
370    let Some(origin) = origin else {
371        return Ok(base);
372    };
373    let absolute = absolute_value(origin.values[0], origin.offset.as_deref(), convert)?;
374    add(base, convert(absolute, base.refer)?)
375}

377fn offset_value(mut base: Measure, offset: Measure, sign: f64) -> Result<Measure, String> {
378    let vector = |a: [f64; 3], b: [f64; 3]| std::array::from_fn(|i| a[i] + sign * b[i]);
379    base.value = match (base.value, offset.value) {
380        (MeasureValue::Baseline(a), MeasureValue::Baseline(b)) => {
381            MeasureValue::Baseline(vector(a, b))
382        }
383        (MeasureValue::Uvw(a), MeasureValue::Uvw(b)) => MeasureValue::Uvw(vector(a, b)),
384        (MeasureValue::EarthMagnetic(a), MeasureValue::EarthMagnetic(b)) => {
385            MeasureValue::EarthMagnetic(vector(a, b))
386        }
387        (MeasureValue::Frequency(a), MeasureValue::Frequency(b)) => {
388            MeasureValue::Frequency(a + sign * b)
389        }
390        (MeasureValue::RadialVelocity(a), MeasureValue::RadialVelocity(b)) => {
391            MeasureValue::RadialVelocity(a + sign * b)
392        }
393        (MeasureValue::Doppler(a), MeasureValue::Doppler(b)) => MeasureValue::Doppler(a + sign * b),
394        (MeasureValue::Epoch(a), MeasureValue::Epoch(b)) => {
395            MeasureValue::Epoch(casa_core::measures::MjdHighPrec::new(
396                a.day() + sign * b.day(),
397                a.frac() + sign * b.frac(),
398            ))
399        }
400        (MeasureValue::Position(a), MeasureValue::Position(b)) => {
401            MeasureValue::Position(vector(a, b))
402        }
403        (MeasureValue::Direction(a), MeasureValue::Direction(b)) => {
404            let v = vector(a, b);
405            let norm = v.iter().map(|x| x * x).sum::<f64>().sqrt();
406            MeasureValue::Direction(if norm == 0.0 {
407                [0.0, 0.0, 1.0]
408            } else {
409                v.map(|x| x / norm)
410            })
411        }
412        _ => return Err("Unmatched offset type in MeasureHolder::fromRecord".into()),
413    };
414    Ok(base)
