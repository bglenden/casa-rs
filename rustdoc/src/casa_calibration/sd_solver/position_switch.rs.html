4use std::collections::BTreeMap;
5use std::path::Path;
6
7use casa_core::ArrayValue;
8use casa_ms::MeasurementSet;
9use casa_tables::RequiredScalarColumnValues;

11use super::spectrum::{self, ReferenceTimestamp, SkySpectrum};
12use crate::constants::*;
13use crate::output::CalibrationOutputMode;

15pub(super) const SUBTYPE: &str = "SDSKY_PS";
16
17/// CASA groups position-switch reference rows by these columns even when
18/// there is a gap between integrations belonging to the same group.
19#[derive(Ord, PartialOrd, Eq, PartialEq)]
20struct ReferenceGroup {
21    observation: i32,
22    processor: i32,
23    field: i32,
24    antenna: i32,
25    feed: i32,
26    data_description: i32,
27    scan: i32,
28    state: i32,
29}
30
31/// Solve and write position-switch sky spectra (`SDSKY_PS`) from OFF integrations.
32pub fn generate(
33    ms: &MeasurementSet,
34    rows: &[usize],
35    destination: &Path,
36    mode: CalibrationOutputMode,
37) -> Result<usize, String> {
38    let solutions = solve(ms, rows)?;
39    spectrum::write(
40        ms,
41        destination,
42        mode,
43        SUBTYPE,
44        ReferenceTimestamp::UnflaggedCells,
45        solutions
46            .into_iter()
47            .map(|(group, spectrum)| (group.antenna, spectrum)),
48    )
49}
50
51fn solve(
52    ms: &MeasurementSet,
53    rows: &[usize],
54) -> Result<BTreeMap<ReferenceGroup, SkySpectrum>, String> {
55    let main = ms.main_table();
56    let columns = main
57        .required_scalar_columns_owned_for_rows(
58            &[
59                COL_OBSERVATION_ID,
60                "PROCESSOR_ID",
61                COL_FIELD_ID,
62                COL_ANTENNA1,
63                COL_ANTENNA2,
64                "FEED1",
65                "FEED2",
66                "DATA_DESC_ID",
67                COL_SCAN_NUMBER,
68                "STATE_ID",
69                COL_TIME,
70                "EXPOSURE",
71            ],
72            rows,
73        )
74        .map_err(error_text)?;
75    let ids = |column: &str| match &columns[column] {
76        RequiredScalarColumnValues::Int32(values) => Ok(values.as_slice()),
77        _ => Err(format!("MeasurementSet {column} is not Int32.")),
78    };
79    let [
80        observation,
81        processor,
82        field,
83        antenna,
84        other_antenna,
85        feed,
86        other_feed,
87        description,
88        scan,
89        state,
90    ] = [
91        ids(COL_OBSERVATION_ID)?,
92        ids("PROCESSOR_ID")?,
93        ids(COL_FIELD_ID)?,
94        ids(COL_ANTENNA1)?,
95        ids(COL_ANTENNA2)?,
96        ids("FEED1")?,
97        ids("FEED2")?,
98        ids("DATA_DESC_ID")?,
99        ids(COL_SCAN_NUMBER)?,
100        ids("STATE_ID")?,
101    ];
102    let (
103        RequiredScalarColumnValues::Float64(times),
104        RequiredScalarColumnValues::Float64(exposures),
105    ) = (&columns[COL_TIME], &columns["EXPOSURE"])
106    else {
107        return Err("MeasurementSet TIME or EXPOSURE is not Float64.".into());
108    };
109    let (state_table, observations, descriptions, spectral) = (
110        ms.state().map_err(error_text)?,
111        ms.observation().map_err(error_text)?,
112        ms.data_description().map_err(error_text)?,
113        ms.spectral_window().map_err(error_text)?,
114    );
115    let off_state = (0..state_table.row_count())
116        .map(|row| {
117            state_table.string(row, "OBS_MODE").map(|mode| {
118                mode.to_ascii_uppercase()
119                    .starts_with("OBSERVE_TARGET#OFF_SOURCE")
120            })
121        })
122        .collect::<Result<Vec<_>, _>>()
123        .map_err(error_text)?;
124    let alma = (0..observations.row_count())
125        .map(|row| {
126            observations
127                .string(row, "TELESCOPE_NAME")
128                .map(|name| name == "ALMA")
129        })
130        .collect::<Result<Vec<_>, _>>()
131        .map_err(error_text)?;
132    let windows = (0..descriptions.row_count())
133        .map(|row| descriptions.spectral_window_id(row))
134        .collect::<Result<Vec<_>, _>>()
135        .map_err(error_text)?;
136    let wvr = (0..spectral.row_count())
137        .map(|row| spectral.num_chan(row).map(|channels| channels == 4))
138        .collect::<Result<Vec<_>, _>>()
139        .map_err(error_text)?;
140    let data_column = if main
141        .schema()
142        .is_some_and(|schema| schema.contains_column("FLOAT_DATA"))
143    {
144        "FLOAT_DATA"
145    } else {
146        "DATA"
147    };
148    let mut solutions = BTreeMap::new();
149    for (index, &row) in rows.iter().enumerate() {
150        if antenna[index] != other_antenna[index]
151            || feed[index] != other_feed[index]
152            || off_state.get(state[index] as usize) != Some(&true)
153        {
154            continue;
155        }
156        let spw = *windows
157            .get(description[index] as usize)
158            .ok_or("Invalid DATA_DESC_ID in reference data.")? as usize;
159        let water_vapour = *wvr.get(spw).ok_or("Invalid reference spectral window.")?;
160        if water_vapour && alma.get(observation[index] as usize) == Some(&true) {
161            continue;
162        }
163        let group = ReferenceGroup {
164            observation: observation[index],
165            processor: processor[index],
166            field: field[index],
167            antenna: antenna[index],
168            feed: feed[index],
169            data_description: description[index],
170            scan: scan[index],
171            state: state[index],
172        };
173        let power = spectrum::read_power(main, row, data_column)?;
174        let ArrayValue::Bool(flags) = main
175            .cell_accessor(row, "FLAG")
176            .and_then(|cell| cell.array())
177            .map_err(error_text)?
178        else {
179            return Err("MeasurementSet FLAG is not Boolean.".into());
180        };
181        let time = times[index];
182        let exposure = exposures[index];
183        if !time.is_finite() || !exposure.is_finite() {
184            return Err("Invalid reference integration time or exposure.".into());
185        }
186        solutions
187            .entry(group)
188            .or_insert_with(|| SkySpectrum::new(spw, &power))
189            .add(&power, flags, time, exposure)?;
190    }
191    if solutions.is_empty() {
192        return Err("No reference integration found in user-selected data. Please double-check your data selection criteria.".into());
193    }
194    Ok(solutions)
195}
196
197fn error_text(error: impl std::fmt::Display) -> String {
198    error.to_string()
