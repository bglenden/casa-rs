31/// A measure of three values: `baseline`, `earthmagnetic`, `position`,
32/// `uvw`.
33const fn three(rf: &'static str) -> [Parameter; 5] {
34    [
35        param("rf", K::Text, Text(rf)),
36        param("v0", K::Any, Empty),
37        param("v1", K::Any, Empty),
38        param("v2", K::Any, Empty),
39        OFF,
40    ]
41}
42
43/// A measure of one value: `doppler`, `epoch`, `frequency`,
44/// `radialvelocity`.
45const fn one(rf: &'static str) -> [Parameter; 3] {
46    [
47        param("rf", K::Text, Text(rf)),
48        param("v0", K::Any, Empty),
49        OFF,
50    ]
51}
