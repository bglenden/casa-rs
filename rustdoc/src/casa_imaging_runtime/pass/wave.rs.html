20/// Row chunks per worker in the placement stage; enough to balance rows
21/// whose channel counts differ after flagging and the support test.
22const CHUNKS_PER_WORKER: usize = 4;
23
24/// One owner's accumulator and backend.
25struct Owner<'w> {
26    acc: Option<GridAccumulator>,
27    backend: OwnerBackend<'w>,
28    images: Option<NormalImages>,
29}
30
31/// The backend an owner dispatches to. The CPU grids each placement chunk
32/// as it comes; a Metal owner first gathers its share of every chunk of a
33/// source block into `staging`, so each device dispatch is one block's
34/// worth of samples rather than one chunk's.
35enum OwnerBackend<'w> {
36    Cpu(CpuBackend),
37    Metal {
38        backend: Box<MetalBackend<'w>>,
39        staging: SampleBuffer,
40    },
41}
42
43/// One image domain's share of a wave.
44struct Domain<'w> {
45    router: Router,
46    owners: Vec<Owner<'w>>,
47    model: Option<&'w PreparedModelGrids>,
48}
49
50/// One row chunk's placements, routed by owner, its prediction scratch and
51/// its rows' native-channel residuals.
52struct Chunk {
53    rows: Range<usize>,
54    scratch: SampleBuffer,
55    owned: Vec<SampleBuffer>,
56    placed: u64,
57    backend: CpuBackend,
58    prediction: PredictionScratch,
59    predicted: Vec<Complex32>,
60    residual: Vec<Complex32>,
61}
62
63pub(super) struct Wave<'w, 'p> {
64    pass: &'w MajorCyclePass<'p>,
65    planes: PlaneRange,
66    domains: Vec<Domain<'w>>,
67    native_residuals: bool,
68    check_spacing: bool,
69    chunks: Vec<Chunk>,
70    predictions: Vec<Complex32>,
71    samples: u64,
72}

74impl<'w, 'p> Wave<'w, 'p> {
75    /// A wave over `planes` with each domain's prepared `models`, forming
76    /// residuals at native channels when `native_residuals`.
77    pub(super) fn new(
78        pass: &'w MajorCyclePass<'p>,
79        planes: PlaneRange,
80        models: Option<&'w [PreparedModelGrids]>,
81        native_residuals: bool,
82    ) -> Result<Self, PassError> {
83        let domains = pass
84            .domains
85            .iter()
86            .enumerate()
87            .map(|(index, domain)| {
88                let router = Router::new(&domain.partition, planes);
89                let operator = domain.operator;
90                let owners = (0..router.owners())
91                    .map(|owner| {
92                        let (range, tile) = router.target(&domain.partition, planes, owner);
93                        Ok(match pass.backend {
94                            BackendChoice::Cpu => Owner {
95                                acc: Some(operator.accumulator(range, tile, pass.modes)),
96                                backend: OwnerBackend::Cpu(CpuBackend::new()),
97                                images: None,
98                            },
99                            BackendChoice::Metal => {
100                                assert_eq!(
101                                    operator.precision(),
102                                    GridPrecision::F32,
103                                    "Metal grids are f32 (D2)"
104                                );
105                                Owner {
106                                    acc: Some(MetalBackend::accumulator(
107                                        operator.accumulator_layout(range, tile, pass.modes),
108                                    )?),
109                                    backend: OwnerBackend::Metal {
110                                        backend: Box::new(MetalBackend::new(operator.cf())?),
111                                        staging: SampleBuffer::new(
112                                            operator.polarization().correlations().len(),
113                                        ),
114                                    },
115                                    images: None,
116                                }
117                            }
118                        })
119                    })
120                    .collect::<Result<_, PassError>>()?;
121                Ok(Domain {
122                    router,
123                    owners,
124                    model: models.map(|models| &models[index]),
125                })
126            })
127            .collect::<Result<_, PassError>>()?;
128        Ok(Self {
129            pass,
130            planes,
131            domains,
132            native_residuals,
133            check_spacing: false,
134            chunks: Vec::new(),
135            predictions: Vec::new(),
136            samples: 0,
137        })
138    }

140    /// Check every row's native spacing against the pass's
141    /// `native_spacing_hz`, which sized this wave's model halo.
142    pub(super) fn checking_spacing(mut self, check: bool) -> Self {
143        self.check_spacing = check;
144        self
145    }

147    /// Samples placed on the first domain so far.
148    pub(super) const fn samples(&self) -> u64 {
149        self.samples

152    /// Place every row of `block` on every domain and accumulate the
153    /// placements; with a visibility sink, first hand it the block, with
154    /// every selected sample's prediction when it asks for them.
155    ///
156    /// When the wave forms residuals at native channels, each row's samples
157    /// are placed from `V − Σ_d A_d·m_d` at its native channels and gridded
158    /// as data; otherwise the domain's model is subtracted at each placed
159    /// sample.
160    pub(super) fn consume(
161        &mut self,
162        block: &NativeBlock,
163        team: &WorkerTeam,
164        visibilities: Option<&mut VisibilitySink<'_>>,
165    ) -> Result<(), PassError> {
166        let npol = block.correlations();
167        let rows = block.len();
168        if self.check_spacing {
169            let bound_hz = self.pass.native_spacing_hz;
170            for row in 0..rows {
171                let observed_hz = block
172                    .row(0, row)
173                    .frequencies_hz
174                    .windows(2)
175                    .fold(0.0_f64, |widest, pair| {
176                        widest.max((pair[1] - pair[0]).abs())
177                    });
178                if observed_hz > bound_hz {
179                    return Err(PassError::NativeSpacing {
180                        observed_hz,
181                        bound_hz,
182                    });
183                }
184            }
185        }
186        let count = (team.workers() * CHUNKS_PER_WORKER).clamp(1, rows.max(1));
187        if self.chunks.len() < count {
188            self.chunks.resize_with(count, || Chunk {
189                rows: 0..0,
190                scratch: SampleBuffer::new(npol),
191                owned: Vec::new(),
192                placed: 0,
193                backend: CpuBackend::new(),
194                prediction: PredictionScratch::default(),
195                predicted: Vec::new(),
196                residual: Vec::new(),
197            });
198        }
199        for (index, chunk) in self.chunks[..count].iter_mut().enumerate() {
200            chunk.rows = index * rows / count..(index + 1) * rows / count;
201        }
202        let sink_predictions = visibilities.as_ref().is_some_and(|sink| sink.predictions);
203        if self.native_residuals || sink_predictions {
204            self.predict(block, team, count)?;
205        } else {
206            self.predictions.clear();
207        }
208        if let Some(sink) = visibilities {
209            let predictions: &[Complex32] = if sink.predictions {
210                &self.predictions
211            } else {
212                &[]
213            };
214            (sink.write)(block, predictions).map_err(PassError::VisibilityWrite)?;
215        }
216        for index in 0..self.domains.len() {
217            let placed = self.place(block, team, count, index)?;
218            if index == 0 {
219                self.samples += placed;
220            }
221        }
222        Ok(())
223    }

225    /// Place the rows of `block` on domain `index` and accumulate them;
226    /// returns the samples placed.
227    fn place(
228        &mut self,
229        block: &NativeBlock,
230        team: &WorkerTeam,
231        count: usize,
232        index: usize,
233    ) -> Result<u64, PassError> {
234        let pass = self.pass;
235        let target = &pass.domains[index];
236        let domain = &mut self.domains[index];
237        let owners = domain.owners.len();
238        let router = &domain.router;
239        let planes = self.planes;
240        let predictions = &self.predictions;
241        let native_residuals = self.native_residuals;
242        let cells = block.channels() * block.correlations();
243        team.for_each_mut(&mut self.chunks[..count], |_, chunk| {
244            chunk.placed = 0;
245            chunk
246                .owned
247                .resize_with(owners, || SampleBuffer::new(block.correlations()));
248            chunk.owned[..owners]
249                .iter_mut()
250                .for_each(SampleBuffer::clear);
251            for row in chunk.rows.clone() {
252                chunk.scratch.clear();
253                let native = block.row(index, row);
254                let native = if native_residuals {
255                    let model = &predictions[row * cells..(row + 1) * cells];
256                    chunk.residual.clear();
257                    chunk.residual.extend(
258                        native
259                            .values
260                            .iter()
261                            .zip(model)
262                            .map(|(value, model)| value - model),
263                    );
264                    NativeRow {
265                        values: &chunk.residual,
266                        ..native
267                    }
268                } else {
269                    native
270                };
271                target.resampler.place(
272                    target.operator,
273                    pass.weighting,
274                    &native,
275                    &mut chunk.scratch,
276                )?;
277                let placed = chunk.scratch.block();
278                for (sample, placement) in placed.placements.iter().enumerate() {
279                    if !planes.contains(placement.plane) {
280                        continue;
281                    }
282                    let owner = router.owner(target.operator, placement);
283                    chunk.owned[owner].push(
284                        *placement,
285                        placed.values_of(sample),
286                        placed.weights_of(sample),
287                    );
288                    chunk.placed += 1;
289                }
290            }
291            Ok::<_, PassError>(())
292        })?;
293        let chunks = &self.chunks[..count];
294        let model = domain.model.filter(|_| !native_residuals);
295        team.for_each_mut(&mut domain.owners, |owner_index, owner| {
296            let acc = owner
297                .acc
298                .as_mut()
299                .expect("owners accumulate until finished");
300            let operator = target.operator;
301            match &mut owner.backend {
302                OwnerBackend::Cpu(backend) => {
303                    for chunk in chunks {
304                        let block = chunk.owned[owner_index].block();
305                        if !block.is_empty() {
306                            accumulate(pass, operator, backend, &block, model, acc)?;
307                        }
308                    }
309                }
310                OwnerBackend::Metal { backend, staging } => {
311                    staging.clear();
312                    for chunk in chunks {
313                        let block = chunk.owned[owner_index].block();
314                        for sample in 0..block.len() {
315                            staging.push(
316                                block.placements[sample],
317                                block.values_of(sample),
318                                block.weights_of(sample),
319                            );
320                        }
321                    }
322                    if !staging.is_empty() {
323                        let backend: &mut MetalBackend<'_> = backend;
324                        accumulate(pass, operator, backend, &staging.block(), model, acc)?;
325                    }
326                }
327            }
328            Ok::<_, PassError>(())
329        })?;
330        Ok(chunks.iter().map(|chunk| chunk.placed).sum())
331    }

333    /// Model visibilities of every selected sample of `block`, summed over
334    /// every domain's model (`SIMapperCollection::degrid`), `[row][channel]
335    /// [correlation]`, into `self.predictions`; zero without a model.
336    fn predict(
337        &mut self,
338        block: &NativeBlock,
339        team: &WorkerTeam,
340        count: usize,
341    ) -> Result<(), PassError> {
342        let cells = block.channels() * block.correlations();
343        let rows = block.len();
344        self.predictions.clear();
345        self.predictions.resize(rows * cells, Complex32::default());
346        if self.domains.iter().all(|domain| domain.model.is_none()) {
347            return Ok(());
348        }
349        let pass = self.pass;
350        let models = self
351            .domains
352            .iter()
353            .map(|domain| domain.model)
354            .collect::<Vec<_>>();
355        let mut pieces = Vec::with_capacity(count);
356        let mut remaining = self.predictions.as_mut_slice();
357        for chunk in &mut self.chunks[..count] {
358            let (head, tail) = remaining.split_at_mut(chunk.rows.len() * cells);
359            remaining = tail;
360            pieces.push((chunk, head));
361        }
362        team.for_each_mut(&mut pieces, |_, (chunk, out)| {
363            chunk.predicted.resize(cells, Complex32::default());
364            for (local, row) in chunk.rows.clone().enumerate() {
365                let out = &mut out[local * cells..(local + 1) * cells];
366                for (index, (target, model)) in pass.domains.iter().zip(&models).enumerate() {
367                    let Some(model) = model else {
368                        continue;
369                    };
370                    target.resampler.predict_row(
371                        target.operator,
372                        &mut chunk.backend,
373                        model,
374                        &block.row(index, row),
375                        &mut chunk.prediction,
376                        &mut chunk.predicted,
377                    )?;
378                    for (sum, value) in out.iter_mut().zip(&chunk.predicted) {
379                        *sum += value;
380                    }
381                }
382            }
383            Ok::<_, PassError>(())
384        })
385    }

387    /// Transform the accumulated grids into this wave's normal images, one
388    /// per domain.
389    pub(super) fn finish(self, team: &WorkerTeam) -> Result<Vec<NormalImages>, PassError> {
390        let pass = self.pass;
391        let planes = self.planes;
392        self.domains
393            .into_iter()
394            .zip(pass.domains)
395            .map(|(mut domain, target)| {
396                let operator = target.operator;
397                match &target.partition {
398                    Partition::Planes { .. } => {
399                        team.for_each_mut(&mut domain.owners, |_, owner| {
400                            let acc = owner.acc.take().expect("each owner finishes once");
401                            owner.images = Some(operator.finish(acc)?);
402                            Ok::<_, PassError>(())
403                        })?;
404                        Ok(concatenate(
405                            domain
406                                .owners
407                                .into_iter()
408                                .map(|owner| owner.images.expect("every owner finished")),
409                        ))
410                    }
411                    Partition::Regions { .. } => {
412                        let mut merged = operator.accumulator(planes, None, pass.modes);
413                        for owner in &domain.owners {
414                            merged
415                                .merge_from(owner.acc.as_ref().expect("owners hold their tiles"))?;
416                        }
417                        drop(domain.owners);
418                        Ok(operator.finish(merged)?)
419                    }
420                }
421            })
422            .collect()
423    }
424}

426/// Grid one owner's share of a block in every mode the pass holds; with a
427/// model the data terms receive the residual `V − A·m`.
428fn accumulate(
429    pass: &MajorCyclePass<'_>,
430    operator: &casa_imaging_operator::MeasurementOperator,
431    backend: &mut dyn GridBackend,
432    block: &casa_imaging_operator::SampleBlock<'_>,
433    model: Option<&PreparedModelGrids>,
434    acc: &mut GridAccumulator,
435) -> Result<(), PassError> {
436    let cf = operator.cf();
437    if pass.modes.data {
438        let work = match model {
439            Some(model) => Work::ResidualGrid {
440                model,
441                acc,
442                residual_out: None,
443            },
444            None => Work::Grid {
445                mode: Mode::Data,
446                acc,
447            },
448        };
449        backend.apply(block, cf, work)?;
450    }
451    if pass.modes.psf {
452        backend.apply(
453            block,
454            cf,
455            Work::Grid {
456                mode: Mode::Psf,
457                acc,
458            },
459        )?;
460    }
461    if pass.modes.weight {
462        backend.apply(
463            block,
464            cf,
465            Work::Grid {
466                mode: Mode::Weight,
467                acc,
468            },
469        )?;
470    }
471    Ok(())
472}

474/// Join per-owner images of consecutive plane ranges.
475pub(super) fn concatenate(parts: impl IntoIterator<Item = NormalImages>) -> NormalImages {
476    let mut parts = parts.into_iter();
477    let mut joined = parts.next().expect("a wave has at least one owner");
478    for part in parts {
479        assert!(
480            part.pols == joined.pols
481                && part.data_terms == joined.data_terms
482                && part.psf_terms == joined.psf_terms
483                && part.first_plane == joined.first_plane + joined.planes.len() as u32,
484            "owner images must continue the plane sequence with the same layout"
485        );
486        joined.planes.extend(part.planes);
487    }
488    joined
489}
