2//! The position angle of one direction seen from another, and the
3//! parallactic angle.
4
5use super::MeasFrame;
6use super::direction::{DirectionRef, MDirection};
7use super::error::MeasureError;
8
9impl MDirection {
10    /// The position angle of `target` seen from `self`, in radians: from
11    /// the direction of `self`'s pole towards increasing longitude, in
12    /// `(-π, π]`. Both directions are taken in their own longitude and
13    /// latitude, so they should share a frame.
14    ///
15    /// A parallactic angle is the position angle of the celestial pole seen
16    /// from the source, both in the observer's `AZEL`.
17    ///
18    /// Cf. casacore `MVDirection::positionAngle`, in its spherical form.
19    #[must_use]
20    pub fn position_angle(&self, target: &Self) -> f64 {
21        let (origin_lon, origin_lat) = self.as_angles();
22        let (target_lon, target_lat) = target.as_angles();
23        let delta_lon = target_lon - origin_lon;
24        let y = delta_lon.sin() * target_lat.cos();
25        let x = origin_lat.cos() * target_lat.sin()
26            - origin_lat.sin() * target_lat.cos() * delta_lon.cos();
27        y.atan2(x)
28    }
29}
30
31impl MDirection {
32    /// The parallactic angle of this direction for the observer `frame`
33    /// places (its epoch and position), in radians: the position angle of
34    /// the celestial pole seen from the direction, both in the observer's
35    /// `AZEL`.
36    ///
37    /// Cf. C++ `MSCalEngine::getPA`.
38    ///
39    /// # Errors
40    ///
41    /// When a conversion to `AZEL` fails (the frame lacks its epoch or
42    /// position).
43    pub fn parallactic_angle(&self, frame: &MeasFrame) -> Result<f64, MeasureError> {
44        let source = self.convert_to(DirectionRef::AZEL, frame)?;
45        let pole = Self::from_angles(0.0, std::f64::consts::FRAC_PI_2, DirectionRef::HADEC)
46            .convert_to(DirectionRef::AZEL, frame)?;
47        Ok(source.position_angle(&pole))
48    }
49}
