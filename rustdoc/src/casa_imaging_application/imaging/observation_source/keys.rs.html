2//! How the rows of several stores name their dishes, spectral windows and
3//! fields to the kernel sets and densities of one run.
4//!
5//! A row's spectral window and field keys are its own `SPECTRAL_WINDOW` and
6//! `FIELD` rows, offset by the rows of the stores before it, as if the
7//! MeasurementSets were concatenated in the snapshot's order. CASA keys the
8//! same pairs: a mosaic's beams by `msId_spw`
9//! (`SimplePBConvFunc::convIndex`), a `mosweight` density by `msId_field`
10//! (`VisImagingWeight::multiFieldMap_p`). The first store's keys are its own
11//! numbers, so a run over one store keys its rows as before.
12
13use casa_imaging_model::AntennaResponseClass;
14use casa_observation::Catalogue;
15
16/// Where one store's window and field keys start.
17#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
18struct KeyBase {
19    window: u32,
20    field: u32,
21}
22
23/// The keys of a run's stores: the aperture classes of them all, and where
24/// each store's window and field keys start.
25#[derive(Clone, Debug, PartialEq, Eq)]
26pub(crate) struct StoreKeys {
27    /// Every store's aperture classes in dish order: a row's antenna types
28    /// index them. CASA's `HetArrayConvFunc::findAntennaSizes` adds the
29    /// dishes of each MeasurementSet it meets to those of the ones before.
30    pub(crate) dish_classes: Vec<AntennaResponseClass>,
31    bases: Vec<KeyBase>,
32}
33
34impl StoreKeys {
35    /// The keys of stores with `catalogues`, in the snapshot's order, whose
36    /// aperture classes are `dish_classes`.
37    pub(crate) fn new<'c>(
38        catalogues: impl IntoIterator<Item = &'c Catalogue>,
39        dish_classes: Vec<AntennaResponseClass>,
40    ) -> Self {
41        Self::counted(
42            catalogues
43                .into_iter()
44                .map(|catalogue| (catalogue.windows.len(), catalogue.fields.len())),
45            dish_classes,
46        )
47    }
48
49    /// The keys of a run of one store, whose windows and fields are keyed
50    /// by their own numbers, with no aperture classes: a standard gridder's
51    /// operator, which keys no dish.
52    pub(crate) fn single() -> Self {
53        Self::counted([(0, 0)], Vec::new())
54    }
55
56    /// The keys of stores of so many `(windows, fields)`.
57    fn counted(
58        counts: impl IntoIterator<Item = (usize, usize)>,
59        mut dish_classes: Vec<AntennaResponseClass>,
60    ) -> Self {
61        dish_classes.sort_unstable();
62        dish_classes.dedup();
63        let mut next = KeyBase::default();
64        let bases = counts
65            .into_iter()
66            .map(|(windows, fields)| {
67                let base = next;
68                next.window += u32::try_from(windows).expect("windows fit u32");
69                next.field += u32::try_from(fields).expect("fields fit u32");
70                base
71            })
72            .collect();
73        Self {
74            dish_classes,
75            bases,
76        }
77    }
78
79    /// Where store `store`'s keys start. Every store a route reads was
80    /// counted when its keys were made, so an unknown store is a defect.
81    fn base(&self, store: usize) -> KeyBase {
82        *self
83            .bases
84            .get(store)
85            .expect("a store the route's keys count")
86    }
87
88    /// The key of store `store`'s spectral window `spectral_window`.
89    pub(crate) fn window(&self, store: usize, spectral_window: u32) -> u32 {
90        self.base(store).window + spectral_window
91    }
92
93    /// The key of store `store`'s field `field`.
94    pub(crate) fn field(&self, store: usize, field: u32) -> u32 {
95        self.base(store).field + field
96    }
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102
103    /// Each store's keys follow the windows and fields of the stores
104    /// before it; the first store's are its own numbers.
105    #[test]
106    fn keys_run_over_every_store_in_order() {
107        let keys = StoreKeys::counted(
108            [(2, 3), (1, 5), (4, 1)],
109            vec![
110                AntennaResponseClass::CasaAca7m,
111                AntennaResponseClass::CasaAlma12m,
112                AntennaResponseClass::CasaAca7m,
113            ],
114        );
115        assert_eq!(keys.window(0, 1), 1);
116        assert_eq!(keys.field(0, 2), 2);
117        assert_eq!(keys.window(1, 0), 2);
118        assert_eq!(keys.field(1, 4), 7);
119        assert_eq!(keys.window(2, 3), 6);
120        assert_eq!(keys.field(2, 0), 8);
121        let one = StoreKeys::single();
122        assert_eq!((one.window(0, 3), one.field(0, 4)), (3, 4));
123        let mut classes = vec![
124            AntennaResponseClass::CasaAca7m,
125            AntennaResponseClass::CasaAlma12m,
126        ];
127        classes.sort_unstable();
128        assert_eq!(keys.dish_classes, classes);
129    }
130
131    /// A store the keys did not count is a defect, not the first store.
132    #[test]
133    #[should_panic(expected = "a store the route's keys count")]
134    fn an_unknown_store_has_no_keys() {
135        let _ = StoreKeys::counted([(2, 3)], Vec::new()).window(1, 0);
136    }
