2//! Native EVLA cell generation on several workers (#708): the cells of one
3//! (parallactic angle, frequency, Mueller element) share one prepared
4//! aperture ([`EvlaAwWorkspace::prepare`]), so a worker takes a run of
5//! their w planes at a time, each worker with its own workspace.

20/// One worker's share: the cells of one (parallactic angle, frequency,
21/// Mueller element), indices into `input`'s lists, at the w planes of a
22/// range, less those already present when only missing cells are made.
23#[derive(Clone, Debug, PartialEq)]
24pub(super) struct Unit {
25    pa: usize,
26    frequency: usize,
27    mueller: usize,
28    w_planes: Range<usize>,
29}
30
31/// Heap bytes [`units`] holds at most for `apertures` (parallactic angle,
32/// frequency, Mueller element) triples on `workers` workers: its count of
33/// cells per aperture and the units, of which there are fewer than
34/// `apertures + workers`.
35pub(super) const fn units_bytes(apertures: usize, workers: usize) -> usize {
36    apertures * size_of::<usize>() + (apertures + workers) * size_of::<Unit>()
37}
38
39/// Whether both images of the cell are in `root`.
40fn present(root: &Path, pa: usize, frequency: usize, w: usize, mueller: usize) -> bool {
41    let suffix = cell_suffix(pa, frequency, w, mueller);
42    root.join(format!("{IMAGING_PREFIX}{suffix}")).is_dir()
43        && root.join(format!("{WEIGHT_PREFIX}{suffix}")).is_dir()
44}

46/// The cells of `input` to generate into `root`, as units for `workers`:
47/// every cell, or with `only_missing` those without both images. Each
48/// (parallactic angle, frequency, Mueller element) is split into runs of w
49/// planes, in the serial generator's order, holding as near equal numbers
50/// of cells to generate as can be and none empty. Each aperture takes the
51/// same number of runs, or all its cells if fewer, the fewest that give
52/// every worker a unit while there are cells for them. A run prepares its
53/// aperture once.
54pub(super) fn units(
55    root: &Path,
56    input: &NativeAwRequestInput,
57    only_missing: bool,
58    workers: usize,
59) -> Vec<Unit> {
60    let planes = input.w_values.len();
61    let to_generate =
62        |pa, frequency, w, mueller| !(only_missing && present(root, pa, frequency, w, mueller));
63    let apertures = || {
64        (0..input.pa_values.len()).flat_map(move |pa| {
65            (0..input.frequencies.len()).flat_map(move |frequency| {
66                (0..input.mueller_elements.len()).map(move |mueller| (pa, frequency, mueller))
67            })
68        })
69    };
70    // Sized exactly, as `units_bytes` charges: a collect would grow it.
71    let mut counts = Vec::with_capacity(
72        input.pa_values.len() * input.frequencies.len() * input.mueller_elements.len(),
73    );
74    counts.extend(apertures().map(|(pa, frequency, mueller)| {
75        (0..planes)
76            .filter(|&w| to_generate(pa, frequency, w, mueller))
77            .count()
78    }));
79    let cells = counts.iter().sum::<usize>();
80    if cells == 0 {
81        return Vec::new();
82    }
83    let wanted = workers.clamp(1, cells);
84    let units_for = |runs: usize| counts.iter().map(|&count| count.min(runs)).sum::<usize>();
85    // At `wanted` runs there are at least `wanted` units, so this ends.
86    let runs = (1..=wanted)
87        .find(|&runs| units_for(runs) >= wanted)
88        .unwrap_or(wanted);
89    let mut units = Vec::with_capacity(units_for(runs));
90    for ((pa, frequency, mueller), &count) in apertures().zip(&counts) {
91        if count == 0 {
92            continue;
93        }
94        // Run j takes the cells to generate of rank j·count/runs up to
95        // (j + 1)·count/runs; the last also covers the planes after them.
96        let runs = count.min(runs);
97        let (mut start, mut rank, mut run) = (0, 0, 0);
98        for w in (0..planes).filter(|&w| to_generate(pa, frequency, w, mueller)) {
99            rank += 1;
100            if rank == (run + 1) * count / runs {
101                run += 1;
102                let end = if run == runs { planes } else { w + 1 };
103                units.push(Unit {
104                    pa,
105                    frequency,
106                    mueller,
107                    w_planes: start..end,
108                });
109                start = end;
110            }
111        }
112    }
113    units
114}

116/// Sets `failed` when a worker unwinds, so a panic stops the others before
117/// their next unit as an error does.
118struct StopOnPanic<'a>(&'a AtomicBool);
119
120impl Drop for StopOnPanic<'_> {
121    fn drop(&mut self) {
122        if std::thread::panicking() {
123            self.0.store(true, Ordering::Relaxed);
124        }
125    }
126}

128/// Generate `units` of `input` into `root` on up to `workers` threads, each
129/// taking the next unit until none is left, and skipping cells already
130/// present with `only_missing`. The first failure or panic stops the others
131/// before their next unit and is returned or resumed.
132pub(super) fn generate(
133    root: &Path,
134    input: &NativeAwRequestInput,
135    units: &[Unit],
136    only_missing: bool,
137    workers: usize,
138) -> Result<(), AwCatalogError> {
139    let model = EvlaApertureModel::new(input.surface.clone());
140    let next = AtomicUsize::new(0);
141    let failed = AtomicBool::new(false);
142    let work = || {
143        let _stop = StopOnPanic(&failed);
144        let mut workspace = None;
145        while !failed.load(Ordering::Relaxed) {
146            let Some(unit) = units.get(next.fetch_add(1, Ordering::Relaxed)) else {
147                break;
148            };
149            let generated = generate_unit(root, input, &model, unit, only_missing, &mut workspace);
150            if let Err(error) = generated {
151                failed.store(true, Ordering::Relaxed);
152                return Err(error);
153            }
154        }
155        Ok(())
156    };
157    let workers = workers.clamp(1, units.len().max(1));
158    if workers == 1 {
159        return work();
160    }
161    std::thread::scope(|scope| {
162        let threads = (0..workers).map(|_| scope.spawn(work)).collect::<Vec<_>>();
163        threads
164            .into_iter()
165            .map(|thread| {
166                thread
167                    .join()
168                    .unwrap_or_else(|panic| std::panic::resume_unwind(panic))
169            })
170            .collect::<Vec<_>>()
171    })
172    .into_iter()
173    .collect()
174}

176/// Generate `unit`'s cells with `workspace`, made on its first cell, and
177/// write each pair of images; with `only_missing`, skip those present.
178fn generate_unit(
179    root: &Path,
180    input: &NativeAwRequestInput,
181    model: &EvlaApertureModel,
182    unit: &Unit,
183    only_missing: bool,
184    workspace: &mut Option<EvlaAwWorkspace>,
185) -> Result<(), AwCatalogError> {
186    let grid = input.grid;
187    let pa = input.pa_values[unit.pa];
188    let frequency = input.frequencies[unit.frequency].cf_frequency_hz;
189    // `AWConvFunc::makeConvFunction2`: the conjugate beam uses the nearest
190    // listed frequency to `√(2f₀² − f²)`, the first on a tie
191    // (`SynthesisUtils::nearestValue`).
192    let conjugate_frequency = if input.terms.conjugate_beams {
193        let listed = input
194            .frequencies
195            .iter()
196            .map(|group| group.cf_frequency_hz)
197            .collect::<Vec<_>>();
198        nearest(
199            &listed,
200            conjugate_frequency(frequency, input.reference_frequency_hz),
201        )
202        .1
203    } else {
204        frequency
205    };
206    let mueller = input.mueller_elements[unit.mueller];
207    for iw in unit.w_planes.clone() {
208        // No other worker writes this unit's cells, so what is present now
209        // is what was present when the units were made.
210        if only_missing && present(root, unit.pa, unit.frequency, iw, unit.mueller) {
211            continue;
212        }
213        let w = input.w_values[iw];
214        let request = EvlaAwCellRequest {
215            size: grid.size,
216            sky_increment_rad: grid.sky_increment_rad,
217            frequency_hz: frequency,
218            conjugate_frequency_hz: conjugate_frequency,
219            w_wavelengths: w,
220            parallactic_angle_rad: pa,
221            mueller,
222            oversampling: grid.oversampling,
223            prolate_spheroidal: input.terms.prolate_spheroidal,
224            aperture: input.terms.aperture,
225        };
226        let workspace = match workspace {
227            Some(workspace) => workspace,
228            None => workspace.insert(EvlaAwWorkspace::new(model.clone(), request)?),
229        };
230        // Only the W screen depends on w: the aperture and the weight cell
231        // are formed once per unit.
232        workspace.prepare(request)?;
233        let pair = workspace.realize(request)?;
234        let misc = CellMisc {
235            sampling: grid.oversampling as f64,
236            pa_deg: pa.to_degrees(),
237            mueller,
238            w_value: w,
239            w_increment: input.w_increment,
240            conjugate_frequency,
241            conjugate_mueller: MUELLER_ELEMENTS - 1 - mueller as u32,
242            diameter_m: input.antenna_diameter_m,
243            sky_increment_rad: grid.sky_increment_rad[1].abs(),
244            working_size: grid.size,
245        };
246        let suffix = cell_suffix(unit.pa, unit.frequency, iw, unit.mueller);
247        write_cell(
248            &root.join(format!("{IMAGING_PREFIX}{suffix}")),
249            &pair.imaging,
250            frequency,
251            grid,
252            &misc,
253        )?;
254        write_cell(
255            &root.join(format!("{WEIGHT_PREFIX}{suffix}")),
256            &pair.weight,
257            frequency,
258            grid,
259            &misc,
260        )?;
261    }
262    Ok(())
263}

273    /// A request of one parallactic angle, `frequencies` SPWs, `w_planes`
274    /// w planes and two Mueller elements; only its lists matter here.
275    fn input(frequencies: usize, w_planes: usize) -> NativeAwRequestInput {
276        NativeAwRequestInput {
277            surface: EvlaDishSurface::new(
278                (0..=125)
279                    .map(|i| {
280                        let r = f64::from(i) / 10.0;
281                        [r, r * r / 36.0, r / 18.0]
282                    })
283                    .collect(),
284            )
285            .expect("surface"),
286            antenna_diameter_m: 25.0,
287            frequencies: (0..frequencies)
288                .map(|window| NativeAwFrequencyGroup {
289                    spectral_window: window as u32,
290                    channel_frequencies_hz: vec![1.4e9 + window as f64 * 1.0e8],
291                    cf_frequency_hz: 1.4e9 + window as f64 * 1.0e8,
292                })
293                .collect(),
294            w_values: (0..w_planes).map(|w| w as f64).collect(),
295            w_increment: 1.0,
296            pa_values: vec![0.31],
297            mueller_elements: vec![0, 15],
298            reference_frequency_hz: 1.5e9,
299            grid: NativeAwGrid {
300                size: 64,
301                sky_increment_rad: [-4.0e-4, 4.0e-4],
302                oversampling: 4,
303            },
304            terms: NativeAwTerms {
305                aperture: true,
306                w_term: true,
307                prolate_spheroidal: false,
308                wideband: true,
309                conjugate_beams: false,
310            },
311            maximum_cells: 2 * frequencies * w_planes,
312        }
313    }

315    /// The cells `units` generates, in order: the planes of each unit's
316    /// range, less those present in `root` with `only_missing`.
317    fn cells(
318        root: &Path,
319        units: &[Unit],
320        only_missing: bool,
321    ) -> Vec<Vec<(usize, usize, usize, usize)>> {
322        units
323            .iter()
324            .map(|unit| {
325                unit.w_planes
326                    .clone()
327                    .filter(|&w| {
328                        !(only_missing && present(root, unit.pa, unit.frequency, w, unit.mueller))
329                    })
330                    .map(|w| (unit.pa, unit.frequency, unit.mueller, w))
331                    .collect()
332            })
333            .collect()
334    }

336    /// Every unit has a cell; there is a unit for each worker while there
337    /// are cells for them; the units' cells are `expected`, once each and
338    /// in order; and the runs of one aperture differ by at most one cell.
339    fn assert_units(
340        root: &Path,
341        input: &NativeAwRequestInput,
342        only_missing: bool,
343        expected: &[(usize, usize, usize, usize)],
344    ) {
345        for workers in [1, 2, 3, 4, 5, 8, 15, 16, 17, 20, 21, 64] {
346            let units = units(root, input, only_missing, workers);
347            let cells = cells(root, &units, only_missing);
348            assert!(
349                cells.iter().all(|cells| !cells.is_empty()),
350                "{workers} workers"
351            );
352            assert!(
353                units.len() >= workers.min(expected.len()),
354                "{workers} workers, {} units",
355                units.len()
356            );
357            assert_eq!(cells.concat(), expected, "{workers} workers");
358            let aperture = |cells: &Vec<(usize, usize, usize, usize)>| {
359                let (pa, frequency, mueller, _) = cells[0];
360                (pa, frequency, mueller)
361            };
362            for aperture in cells.chunk_by(|a, b| aperture(a) == aperture(b)) {
363                let sizes = aperture.iter().map(Vec::len);
364                let spread = sizes.clone().max().unwrap() - sizes.min().unwrap();
365                assert!(spread <= 1, "{workers} workers");
366            }
367        }
368    }

370    /// One worker takes every cell of an aperture in one unit, in the serial
371    /// generator's order; more workers split each aperture's w planes into
372    /// balanced runs until each worker has a unit, and every cell is in one
373    /// run. A fill makes units of the missing cells alone.
374    #[test]
375    fn units_cover_every_cell_once_in_serial_order() {
376        let root = tempfile::tempdir().expect("cache directory");
377        let input = input(2, 5);
378        let apertures = 2 * input.mueller_elements.len();
379        let serial = units(root.path(), &input, false, 1);
380        assert_eq!(serial.len(), apertures);
381        assert!(serial.iter().all(|unit| unit.w_planes == (0..5)));
382        let every = cells(root.path(), &serial, false).concat();
383        assert_units(root.path(), &input, false, &every);
384
385        // Present: every plane of the first aperture, the first three of the
386        // second, the last of the third and none of the fourth.
387        let held = |&(pa, frequency, mueller, w): &(usize, usize, usize, usize)| match (frequency, mueller) {
388            (0, 0) => true,
389            (0, 1) => w < 3,
390            (1, 0) => w == 4,
391            _ => false,
392        } && pa == 0;
393        for &(pa, frequency, mueller, w) in every.iter().filter(|cell| held(cell)) {
394            let suffix = cell_suffix(pa, frequency, w, mueller);
395            for prefix in [IMAGING_PREFIX, WEIGHT_PREFIX] {
396                std::fs::create_dir(root.path().join(format!("{prefix}{suffix}"))).expect("cell");
397            }
398        }
399        let missing = every
400            .iter()
401            .copied()
402            .filter(|cell| !held(cell))
403            .collect::<Vec<_>>();
404        assert_eq!(missing.len(), 11);
405        assert_units(root.path(), &input, true, &missing);
406    }
