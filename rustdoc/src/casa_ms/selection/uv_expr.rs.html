11#[derive(Clone, Debug, Default)]
12pub(crate) struct UvDistanceSelection {
13    meters: Vec<(f64, f64)>,
14    by_ddid: BTreeMap<i32, Vec<(f64, f64)>>,
15}
16
17impl UvDistanceSelection {
18    pub(crate) fn parse(ms: &MeasurementSet, text: &str) -> MsResult<Self> {
19        let mut selection = Self::default();
20        let mut wavelengths = Vec::new();
21        for term in text.split(',') {
22            let ((lower, upper), unit) = parse_term(term)?;
23            match unit {
24                UvUnit::Meters(scale) => {
25                    selection
26                        .meters
27                        .push(((lower * scale).powi(2), (upper * scale).powi(2)));
28                }
29                UvUnit::Lambda(scale) => wavelengths.push((lower * scale, upper * scale)),
30            }
31        }
32        if !wavelengths.is_empty() {
33            for (ddid, lambda) in ddid_to_lambda_map(ms)? {
34                let square = |value: f64| f64::from(((value * lambda) as f32).powi(2));
35                selection.by_ddid.insert(
36                    ddid,
37                    wavelengths
38                        .iter()
39                        .map(|&(lower, upper)| (square(lower), square(upper)))
40                        .collect(),
41                );
42            }
43        }
44        Ok(selection)
45    }
46
47    pub(super) fn needs_data_desc_id(&self) -> bool {
48        !self.by_ddid.is_empty()
49    }
50
51    pub(crate) fn holds(&self, uvw: [f64; 3], ddid: Option<i32>) -> bool {
52        let distance = uvw[0].powi(2) + uvw[1].powi(2);
53        let matches = |&(lower, upper): &(f64, f64)| (lower..=upper).contains(&distance);
54        self.meters.iter().any(matches)
55            || ddid
56                .and_then(|ddid| self.by_ddid.get(&ddid))
57                .is_some_and(|bounds| bounds.iter().any(matches))
58    }
59
60    pub(super) fn to_taql(&self) -> String {
61        let mut terms = self
62            .meters
63            .iter()
64            .filter_map(distance_taql)
65            .collect::<Vec<_>>();
66        for (ddid, bounds) in &self.by_ddid {
67            let distances = bounds.iter().filter_map(distance_taql).collect::<Vec<_>>();
68            if !distances.is_empty() {
69                terms.push(format!(
70                    "(DATA_DESC_ID=={ddid} AND ({}))",
71                    distances.join(" OR ")
72                ));
73            }
74        }
75        if terms.is_empty() {
76            "FALSE".into()
77        } else {
78            terms.join(" OR ")
79        }
80    }
81}
82
83fn distance_taql(&(lower, upper): &(f64, f64)) -> Option<String> {
84    const DISTANCE: &str = "(SQUARE(UVW[1]) + SQUARE(UVW[2]))";
85    if !lower.is_finite() || upper.is_nan() || lower > upper {
86        return None;
87    }
88    Some(if upper.is_infinite() {
89        format!("{DISTANCE}>={lower:e}")
90    } else {
91        format!("({DISTANCE}>={lower:e} AND {DISTANCE}<={upper:e})")
92    })
93}

95fn parse_term(text: &str) -> MsResult<((f64, f64), UvUnit)> {
96    let mut parts = text.split(':');
97    let expression = trim_row_id_space(parts.next().unwrap_or_default());
98    let (lower, upper, unit) = if let Some(text) = expression.strip_prefix('<') {
99        let (upper, unit) = number_with_unit(text)?;
100        (0.0, upper, unit)
101    } else if let Some(text) = expression.strip_prefix('>') {
102        let (lower, unit) = number_with_unit(text)?;
103        (lower + 1e-10, f64::from(f32::MAX), unit)
104    } else if let Some((left, right)) = expression.split_once('~') {
105        let (lower, rest) = number(left)?;
106        if !rest.is_empty() {
107            return Err(invalid_uv("~"));
108        }
109        let (upper, unit) = number_with_unit(right)?;
110        (lower, upper, unit)
111    } else {
112        let (lower, unit) = number_with_unit(expression)?;
113        // In MSUvDistGram's single-value production, the second slot
114        // of the bound pair is zero. Keep both slots for percent forms.
115        (lower, 0.0, unit)
116    };
117    let mut bounds = (lower, upper);
118    for part in parts {
119        let (percent, rest) = number(part)?;
120        if rest != "%" {
121            return Err(invalid_uv(rest));
122        }
123        bounds.0 *= 1.0 - percent * 0.01;
124        bounds.1 *= 1.0 + percent * 0.01;
125    }
126    Ok((bounds, unit))
127}

129fn number_with_unit(text: &str) -> MsResult<(f64, UvUnit)> {
130    let (value, token) = number(text)?;
131    let unit = match token {
132        "" | "m" | "M" => UvUnit::Meters(1.0),
133        "km" | "Km" => UvUnit::Meters(1_000.0),
134        "lambda" | "LAMBDA" => UvUnit::Lambda(1.0),
135        "klambda" | "Klambda" | "kLAMBDA" | "KLAMBDA" => UvUnit::Lambda(1_000.0),
136        "Mlambda" | "MLAMBDA" => UvUnit::Lambda(1_000_000.0),
137        "k" | "K" | "kM" | "KM" | "kLambda" | "KLambda" => {
138            return Err(MsError::InvalidInput(
139                "Unrecognized units k found. Possible units are [KM]LAMBDA for wavelengths or [K]M for distance.".into(),
140            ));
141        }
142        _ => return Err(invalid_uv(token)),
143    };
144    Ok((value, unit))
145}

147fn number(text: &str) -> MsResult<(f64, &str)> {
148    let text = trim_row_id_space(text);
149    let digits = |data: &[u8]| data.iter().take_while(|byte| byte.is_ascii_digit()).count();
150    let bytes = text.as_bytes();
151    let mut end = digits(bytes);
152    if end > 0 && bytes.get(end) == Some(&b'.') {
153        end += 1 + digits(&bytes[end + 1..]);
154    }
155    let mut numeric_end = end;
156    if let Some(kind @ (b'd' | b'D' | b'e' | b'E')) = bytes.get(end).copied() {
157        let start = end + 1 + usize::from(matches!(bytes.get(end + 1), Some(b'+' | b'-')));
158        let count = digits(&bytes[start..]);
159        if count > 0 {
160            end = start + count;
161            // The lexer accepts D exponents and converts through atof.
162            if numeric_end > 0 && matches!(kind, b'e' | b'E') {
163                numeric_end = end;
164            }
165        }
166    }
167    let value = if end == 0 {
168        return Err(invalid_uv(
169            &text[..text.chars().next().map_or(0, char::len_utf8)],
170        ));
171    } else if numeric_end == 0 {
172        // An exponent without a mantissa is accepted by the lexer and
173        // converted to zero by atof.
174        0.0
175    } else {
176        text[..numeric_end].parse().map_err(|_| invalid_uv(text))?
177    };
178    Ok((value, trim_row_id_space(&text[end..])))
179}
180
181fn invalid_uv(token: &str) -> MsError {
182    MsError::InvalidInput(format!(
183        "UV Distribution Expression: Parse error at or near {token:?}"
184    ))
