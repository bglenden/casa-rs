21fn raised(error: impl ToString) -> PyErr {
22    PyRuntimeError::new_err(error.to_string())
23}

25struct Adjust(bool);

27impl<'a, 'py> FromPyObject<'a, 'py> for Adjust {
28    type Error = PyErr;
29
30    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
31        value
32            .cast::<PyBool>()
33            .map(|value| Self(value.is_true()))
34            .map_err(|_| {
35                PyTypeError::new_err(
36                    "in method 'coordsys_setreferencecode', argument 4 of type 'bool'",
37                )
38            })
39    }
40}

42fn native_direction(model: &DirectionCoordinate) -> PyResult<MDirection> {
43    let [longitude, latitude] = model
44        .reference_value()
45        .try_into()
46        .map_err(|_| raised("Invalid direction reference value"))?;
47    Ok(MDirection::from_angles(
48        longitude,
49        latitude,
50        model.direction_ref(),
51    ))
52}

54fn epoch_frame(frame: MeasFrame, obs: &ObsInfo) -> PyResult<MeasFrame> {
55    let epoch = obs
56        .date
57        .as_ref()
58        .filter(|epoch| epoch.value().as_mjd() >= 0.0)
59        .ok_or_else(|| raised("The output CoordinateSystem has no valid epoch"))?;
60    Ok(frame.with_epoch(epoch.clone()))
61}
62
63fn position_frame(frame: MeasFrame, obs: &ObsInfo) -> PyResult<MeasFrame> {
64    let measures = casa_ms::open_measures_runtime().map_err(raised)?;
65    let position =
66        MPosition::casa_observatory(&obs.telescope, measures.as_ref()).ok_or_else(|| {
67            raised(format!(
68                "Cannot find the observatory name {} in the CASA database",
69                obs.telescope
70            ))
71        })?;
72    Ok(frame.with_position(position).with_measures(measures))
73}

75fn adjust_direction(
76    model: &DirectionCoordinate,
77    frame: DirectionRef,
78    obs: &ObsInfo,
79) -> PyResult<[f64; 2]> {
80    let direction = native_direction(model)?;
81    let empty = MeasFrame::new();
82    let converted = match direction.convert_to(frame, &empty) {
83        Ok(converted) => converted,
84        Err(_) => {
85            let context = epoch_frame(empty, obs)?;
86            match direction.convert_to(frame, &context) {
87                Ok(converted) => converted,
88                Err(_) => direction
89                    .convert_to(frame, &position_frame(context, obs)?)
90                    .map_err(raised)?,
91            }
92        }
93    };
94    let (mut longitude, latitude) = converted.as_angles();
95    if longitude > std::f64::consts::PI {
96        longitude -= std::f64::consts::TAU;
97    }
98    Ok([longitude, latitude])
99}

101fn direction_code(state: &mut CoordinateState, name: &str, adjust: bool) -> PyResult<()> {
102    let (index, _, placement) = numbered_coordinates(state)?
103        .into_iter()
104        .find(|(_, model, _)| matches!(model, CoordinateModel::Direction(_)))
105        .ok_or_else(|| missing(CoordinateType::Direction))?;
106    let frame = DIRECTION_CODES
107        .resolve(name)
108        .and_then(|code| code.parse().ok())
109        .ok_or_else(|| raised(format!("Invalid direction code '{name}' given")))?;
110    let CoordinateModel::Direction(model) = placement.model(&state.record).map_err(raised)? else {
111        return Err(missing(CoordinateType::Direction));
112    };
113    if model.direction_ref() == frame {
114        return Ok(());
115    }
116    let reference = adjust
117        .then(|| adjust_direction(&model, frame, state.coordinates.obs_info()))
118        .transpose()?;
119    let changed = CoordinateModel::from(
120        model
121            .with_reassigned_reference_frame(frame, reference)
122            .map_err(raised)?,
123    );
124    let record = Value::Record(changed.to_record());
125    replace_at(state, index as i32, changed, record)
126}

128fn frequency_context(state: &CoordinateState) -> PyResult<MeasFrame> {
129    let Some((_, placement)) = coordinate(state, CoordinateType::Direction)? else {
130        return Err(raised("No Direction coordinate in 'to' CoordinateSystem"));
131    };
132    let CoordinateModel::Direction(direction) = placement.model(&state.record).map_err(raised)?
133    else {
134        return Err(missing(CoordinateType::Direction));
135    };
136    let obs = state.coordinates.obs_info();
137    let frame = position_frame(
138        MeasFrame::new().with_direction(native_direction(&direction)?),
139        obs,
140    )?;
141    epoch_frame(frame, obs)
142}

144fn spectral_code(
145    state: &mut CoordinateState,
146    name: &str,
147    adjust: bool,
148    log: &mut Vec<LogLine>,
149) -> PyResult<()> {
150    let (index, CoordinateModel::Spectral(model), placement) = numbered_coordinates(state)?
151        .into_iter()
152        .find(|(_, model, _)| matches!(model, CoordinateModel::Spectral(_)))
153        .ok_or_else(|| missing(CoordinateType::Spectral))?
154    else {
155        return Err(missing(CoordinateType::Spectral));
156    };
157    let frame = FREQUENCY_CODES
158        .resolve(name)
159        .and_then(|code| code.parse().ok())
160        .ok_or_else(|| raised(format!("Invalid frequency code '{name}' given")))?;
161    let native = declared_frequency_ref(model);
162    if native == frame {
163        return Ok(());
164    }
165    if let Some(conversion) = model
166        .conversion_frequency_ref()
167        .filter(|value| *value != native)
168    {
169        log.push(LogLine::warning(format!(
170            "Resetting the conversion frequency system {conversion} to the new native frequency system {frame}. You must explicitly reset the conversion frequency system if desired"
171        )));
172    }
173
174    let mut changed = model.clone().with_frequency_reference(frame);
175    if adjust {
176        let context = frequency_context(state)?;
177        if matches!(native, FrequencyRef::REST) || matches!(frame, FrequencyRef::REST) {
178            log.push(LogLine::warning("Unable to convert between the input and output SpectralCoordinates. Conversion involving the REST frame requires a source radial velocity."));
179            return Err(raised("No MeasFrame specified for conversion of Frequency"));
180        }
181        if matches!(native, FrequencyRef::Undefined) || matches!(frame, FrequencyRef::Undefined) {
182            return Err(raised(
183                "Transformations to/from frame \"Undefined\" are not possible.",
184            ));
185        }
186        let hz = changed
187            .world_frequency_hz(model.reference_pixel()[0])
188            .map_err(raised)?;
189        let frequency = MFrequency::new(hz, native)
190            .convert_to(frame, &context)
191            .map_err(raised)?;
192        changed = changed
193            .with_reference_frequency_hz(frequency.hz())
194            .map_err(raised)?;
195    }
196
197    let Some(Value::Record(mut record)) = state.record.get(&placement.field).cloned() else {
198        return Err(raised("Missing spectral coordinate record"));
199    };
200    relabel_spectral_record(&mut record, frame);
201    let changed = CoordinateModel::from(changed);
202    if adjust {
203        let rebuilt = changed.to_record();
204        if let Some(table) = rebuilt.get("tabular") {
205            record.upsert("tabular", table.clone());
206        } else if let Some(Value::Record(wcs)) = record.get_mut("wcs") {
207            wcs.upsert(
208                "crval",
209                Value::Scalar(ScalarValue::Float64(changed.reference_value()[0])),
210            );
211        } else {
212            let value = ArrayValue::from_f64_vec(changed.reference_value()).into();
213            // Version 1 stored even a linear spectral WCS in `tabular`.
214            if let Some(Value::Record(table)) = record.get_mut("tabular") {
215                table.upsert("crval", value);
216            } else {
217                record.upsert("crval", value);
218            }
219        }
220    }
221    replace_at(state, index as i32, changed, Value::Record(record))
222}

224#[pymethods]
225impl PyCoordsys {
226    #[pyo3(signature = (value, r#type=FrameName::direction_default(), adjust=Adjust(true)), text_signature = "($self, value, type='direction', adjust=True)")]
227    fn setreferencecode(
228        &self,
229        py: Python<'_>,
230        value: FrameName,
231        r#type: FrameName,
232        adjust: Adjust,
233    ) -> PyResult<bool> {
234        let value = value.into_name("value")?;
235        let kind = coordinate_type(&r#type.into_name("type")?);
236        let mut log = Vec::new();
237        let result = py.detach(|| {
238            let mut state = self.state();
239            match kind {
240                CoordinateType::Direction => direction_code(&mut state, &value, adjust.0),
241                CoordinateType::Spectral => spectral_code(&mut state, &value, adjust.0, &mut log),
242                _ => Err(raised("Coordinate type must be 'Direction' or 'Spectral'")),
243            }
244        });
245        emit(py, "casars.tools.coordsys", &log)?;
246        result.map(|()| true)
247    }
248}
