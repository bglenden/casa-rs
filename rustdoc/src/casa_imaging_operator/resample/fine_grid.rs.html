5impl SpectralAxis {
6    /// At an exact integral width ratio above one, use the fine-grid count
7    /// from below. This resolves CASA's TOPO round-trip width discontinuity
8    /// for the LSRK source grid without changing interpolation bounds.
9    #[must_use]
10    pub const fn with_exact_width_from_below(mut self) -> Self {
11        self.exact_width_from_below = true;
12        self
13    }
14
15    pub(super) fn fine_samples_per_plane(self, width: f64) -> usize {
16        let whole = width.floor();
17        if self.exact_width_from_below && width > 1.0 && width == whole {
18            (whole as usize).saturating_sub(1)
19        } else {
20            (whole as usize).max(1)
21        }
22    }
23}

25/// CASA's fine frequency grid for linear interpolation: the output centres
26/// when output channels are no wider than native ones, otherwise
27/// `floor(width ratio)` points per output channel.
28#[derive(Clone, Copy, Debug, PartialEq)]
29pub(super) struct FineGrid {
30    start_hz: f64,
31    pub(super) increment_hz: f64,
32    per_output: usize,
33    direct_centres: bool,
34    output: SpectralAxis,
35}
36
37impl FineGrid {
38    pub(super) fn compile(output: SpectralAxis, native_increment_hz: f64) -> Option<Self> {
39        let output_increment_hz = output.increment_hz;
40        if native_increment_hz == 0.0 || !native_increment_hz.is_finite() {
41            return None;
42        }
43        let width = output_increment_hz.abs() / native_increment_hz.abs();
44        if width <= 1.0 {
45            let increment_hz = output_increment_hz.abs().copysign(native_increment_hz);
46            let start_hz = if increment_hz.signum() == output_increment_hz.signum() {
47                output.first_hz
48            } else {
49                output.last_hz()
50            };
51            return Some(Self {
52                start_hz,
53                increment_hz,
54                per_output: 1,
55                direct_centres: output.reporting_axis.is_some(),
56                output,
57            });
58        }
59        let per_output = output.fine_samples_per_plane(width);
60        let fine_abs = output_increment_hz.abs() / per_output as f64;
61        let first_edge = output.first_hz - output_increment_hz / 2.0;
62        let last_edge = output.last_hz() + output_increment_hz / 2.0;
63        let low_edge = first_edge.min(last_edge);
64        let high_edge = first_edge.max(last_edge);
65        let increment_hz = fine_abs.copysign(native_increment_hz);
66        let start_hz = if increment_hz > 0.0 {
67            low_edge + fine_abs / 2.0
68        } else {
69            high_edge - fine_abs / 2.0
70        };
71        Some(Self {
72            start_hz,
73            increment_hz,
74            per_output,
75            direct_centres: false,
76            output,
77        })
78    }
79
80    pub(super) fn count(self) -> usize {
81        self.per_output * self.output.channels as usize
82    }
83
84    pub(super) fn frequency_hz(self, ordinal: usize) -> f64 {
85        if self.direct_centres {
86            self.output.centre_hz(self.output_channel(ordinal))
87        } else {
88            self.start_hz + ordinal as f64 * self.increment_hz
89        }
90    }
91
92    pub(super) fn output_channel(self, ordinal: usize) -> u32 {
93        let output_ordinal = (ordinal / self.per_output) as u32;
94        if self.increment_hz.signum() == self.output.increment_hz.signum() {
95            output_ordinal
96        } else {
97            self.output.channels - 1 - output_ordinal
98        }
99    }
100}

102/// The most fine-grid points a row of `spectrum` visits against an output
103/// axis of `channels` channels `increment_hz` apart.
104///
105/// [`FineGrid`] takes `p = max(1, floor(|width| / first spacing))` points
106/// per output channel at most, `|width| / p` apart. A row visits each point at
107/// most once and only within its span: at most `p · channels` points and
108/// `floor(span / spacing) + 1` of them, plus one for rounding. For evenly
109/// spaced channels that is about one per channel or per output plane. A
110/// one-channel axis or row maps like `nearest`, one sample per channel.
111pub(super) fn linear_samples(increment_hz: f64, channels: usize, spectrum: RowSpectrum) -> usize {
112    if channels <= 1 || spectrum.channels <= 1 {
113        return spectrum.channels;
114    }
115    if spectrum.first_spacing_hz <= 0.0 {
116        return spectrum.channels.max(channels) + 1;
117    }
118    let width = increment_hz.abs();
119    let per_output = (width / spectrum.first_spacing_hz).floor().max(1.0);
120    let within = (spectrum.span_hz * per_output / width).floor() + 2.0;
121    // Saturating: a bound past the address space fails its allocation.
122    (per_output * channels as f64).min(within) as usize
123}
