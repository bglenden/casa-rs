3//! Runtime transport for the Major-Cycle reconciliation envelope.
4//!
5//! The runtime owns plan placement, capabilities, resource accounting, and
6//! attempt/node/lease binding for the reconciliation, and transports the
7//! closed completion envelope. It never reinterprets scientific roles or
8//! scans generic digests: the join itself is minted by the reconstruction
9//! owner from T19 evidence and the T28 model lifecycle.
10

23/// Runtime-bound owner of one attempt's pending Major-Cycle reconciliation.
24///
25/// Created only from owner-minted T19 complete-data output; consumed exactly
26/// once by [`MajorCycleOperatorState::reconcile`] at the plan-authoritative
27/// final-reconciliation node bound when the T19 plan fragment was composed.
28#[derive(Debug)]
29pub struct MajorCycleOperatorState {
30    owner: MajorCycleOwner,
31    attempt: ExecutionAttemptId,
32    replay_node: WorkNodeId,
33    reconciliation_node: WorkNodeId,
34    lease_epoch: u64,
35    observation_predecessor_required: bool,
36}
37
38impl MajorCycleOperatorState {
39    /// Retain the T19 operator output as the sole input to reconciliation.
40    ///
41    /// The reconstruction evidence stays inseparably paired inside the owner,
42    /// and the plan-authoritative reconciliation node is retained beside it;
43    /// [`Self::reconcile`] accepts no substitute node, evidence, or context.
44    pub fn begin(
45        result: CompleteDataOperatorResult,
46        preparation: MajorCyclePreparation,
47    ) -> Result<Self, MajorCycleOperatorError> {
48        let state = Self {
49            attempt: result.attempt_id(),
50            replay_node: result.replay_node().clone(),
51            reconciliation_node: result.reconciliation_node().clone(),
52            lease_epoch: result.lease_epoch(),
53            observation_predecessor_required: result.observation_predecessor_required(),
54            owner: MajorCycleOwner::from_complete_data(result.into_evidence(), preparation)?,
55        };
56        Ok(state)
57    }
58
59    /// Perform the one atomic Major-Cycle reconciliation.
60    ///
61    /// The context must be the exact plan-authoritative final-reconciliation
62    /// Compute node behind the sealed plan, and the exact execution attempt and
63    /// lease epoch that ran T19. The settled replay-predecessor evidence must
64    /// carry the same attempt, node, lease epoch, exhaustive sample count, and
65    /// authoritative T17 observation generation as the retained T19 evidence,
66    /// and the model lifecycle must be bound to that same canonical attempt
67    /// identity and lease epoch. Any mismatch fails atomically before the
68    /// reconstruction owner mints any typed completion record.
69    ///
70    /// Stale weighting evidence cannot reach this seam: the frozen T18
71    /// generation, replay identity, and coverage were bound into the T19
72    /// completion when it was minted from the terminal replay proof, and the
73    /// predecessor check below rebinds that same settled completion to this
74    /// node. A cancelled run never dispatches its reconciliation work or drains
75    /// past the commit gate, so an envelope minted here is discarded with its
76    /// attempt and nothing becomes visible through publication.
77    pub fn reconcile(
78        self,
79        context: WorkExecutionContext<'_>,
80        lifecycle: &mut ModelLifecycle,
81    ) -> Result<MajorCycleOperatorResult, MajorCycleOperatorError> {
82        if context.node().id != self.reconciliation_node || context.node().kind != WorkKind::Compute
83        {
84            return Err(MajorCycleOperatorError::WrongExecutionNode);
85        }
86        if context.attempt_id() != self.attempt || context.lease_epoch() != self.lease_epoch {
87            return Err(MajorCycleOperatorError::ExecutionBinding);
88        }
89        if self.observation_predecessor_required {
90            let predecessor = context
91                .predecessor_observation_completion(&self.replay_node)
92                .ok_or(MajorCycleOperatorError::MissingReplayPredecessor)?;
93            if predecessor.attempt_id() != self.attempt
94                || predecessor.owner_node() != &self.replay_node
95                || predecessor.lease_epoch() != self.lease_epoch
96                || predecessor.owner_completion().generation_id()
97                    != self.owner.selected_generation()
98                || predecessor.owner_completion().sample_count() != self.owner.sample_count()
99            {
100                return Err(MajorCycleOperatorError::ExecutionBinding);
101            }
102        } else if !context
103            .node()
104            .dependencies
105            .contains(&WorkDependency::Fence(FenceId::new(
106                self.replay_node.clone(),
107                FenceKind::Io,
108            )))
109        {
110            return Err(MajorCycleOperatorError::MissingReplayPredecessor);
111        }
112        if lifecycle.problem() != self.owner.problem_id()
113            || context.compiled().problem_id() != self.owner.problem_id()
114        {
115            return Err(MajorCycleOperatorError::StaleProblemEvidence);
116        }
117        let expected_attempt = ModelExecutionAttemptId::new(LogicalIdentity::from_sha256(
118            context.attempt_id().as_bytes(),
119        ));
120        if lifecycle.attempt() != expected_attempt || lifecycle.epoch() != context.lease_epoch() {
121            return Err(MajorCycleOperatorError::ModelAttemptBinding);
122        }
123        let MajorCycleOperatorState {
124            owner,
125            attempt,
126            reconciliation_node,
127            lease_epoch,
128            ..
129        } = self;
130        let completion = owner.reconcile(lifecycle)?;
131        Ok(MajorCycleOperatorResult {
132            completion,
133            attempt,
134            node: reconciliation_node,
135            lease_epoch,
136        })
137    }
138}

140/// Inseparable runtime envelope around one completed Major-Cycle reconciliation.
141///
142/// The envelope is minted only by [`MajorCycleOperatorState::reconcile`] at
143/// its plan-authoritative node; a caller cannot construct or forge it:
144///
145/// ```compile_fail
146/// use casa_imaging_runtime::MajorCycleOperatorResult;
147///
148/// let _ = MajorCycleOperatorResult {};
149/// ```
150#[derive(Debug)]
151pub struct MajorCycleOperatorResult {
152    completion: casa_imaging_reconstruction::MajorCycleCompletion,
153    attempt: ExecutionAttemptId,
154    node: WorkNodeId,
155    lease_epoch: u64,
156}
157
158impl MajorCycleOperatorResult {
159    /// Borrow the inseparable pair of distinct typed completions.
160    #[must_use]
161    pub const fn completion(&self) -> &casa_imaging_reconstruction::MajorCycleCompletion {
162        &self.completion
163    }
164
165    /// Return the execution attempt whose settled fences authorized the join.
166    #[must_use]
167    pub const fn attempt_id(&self) -> ExecutionAttemptId {
168        self.attempt
169    }
170
171    /// Return the final-reconciliation node that executed the join.
172    #[must_use]
173    pub const fn node(&self) -> &WorkNodeId {
174        &self.node
175    }
176
177    /// Return the Resource Authority lease epoch held through the join.
178    #[must_use]
179    pub const fn lease_epoch(&self) -> u64 {
180        self.lease_epoch
181    }
182
183    /// Consume the runtime envelope into the reconstruction-owned completion.
184    #[must_use]
185    pub fn into_completion(self) -> casa_imaging_reconstruction::MajorCycleCompletion {
186        self.completion
187    }
188}

190/// Exact reason the runtime rejected a Major-Cycle reconciliation attempt.
191#[derive(Debug, Clone, PartialEq, Eq)]
192pub enum MajorCycleOperatorError {
193    /// Reconciliation was attempted outside its plan-authoritative Compute node.
194    WrongExecutionNode,
195    /// The calling context does not match the attempt or lease that ran T19.
196    ExecutionBinding,
197    /// The planned node does not follow the settled T19 replay completion.
198    MissingReplayPredecessor,
199    /// The compiled problem changed after T19 completed.
200    StaleProblemEvidence,
201    /// The model lifecycle is not bound to this attempt's canonical identity and epoch.
202    ModelAttemptBinding,
203    /// The reconstruction owner rejected the scientific join.
204    Owner(MajorCycleError),
205}
206
207impl fmt::Display for MajorCycleOperatorError {
208    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
209        match self {
210            Self::WrongExecutionNode => formatter.write_str(
211                "T20 can reconcile only at its plan-authoritative Compute reconciliation node",
212            ),
213            Self::ExecutionBinding => {
214                formatter.write_str("reconciliation context changed the T19 execution authority")
215            }
216            Self::MissingReplayPredecessor => {
217                formatter.write_str("reconciliation does not follow its T19 replay completion")
218            }
219            Self::StaleProblemEvidence => {
220                formatter.write_str("compiled problem differs from the reconciled T19 evidence")
221            }
222            Self::ModelAttemptBinding => formatter
223                .write_str("model lifecycle is not bound to the executing attempt and lease epoch"),
224            Self::Owner(error) => error.fmt(formatter),
225        }
226    }
227}
228
229impl Error for MajorCycleOperatorError {}
230
231impl From<MajorCycleError> for MajorCycleOperatorError {
232    fn from(error: MajorCycleError) -> Self {
233        Self::Owner(error)
234    }
