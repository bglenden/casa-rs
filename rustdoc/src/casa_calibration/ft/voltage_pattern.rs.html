34pub(super) fn legacy_mosaic_table(source: &Path, grid: &Path) -> MsResult<PathBuf> {
35    let table = voltage_pattern::VoltagePatternTable::open(source).map_err(invalid)?;
36    let definitions = table.definitions().map_err(invalid)?;
37    let mut replacements = Vec::new();
38    let destination = grid.join(MANAGED_VP_TABLE);
39    for (index, mut record) in definitions.into_iter().enumerate() {
40        let (Some(real), Some(imaginary)) =
41            (text(&record, "realimage"), text(&record, "imagimage"))
42        else {
43            continue;
44        };
45        if text(&record, "name") != Some("IMAGE") || record.get("imagim").is_some() {
46            continue;
47        }
48        let imaginary = if imaginary.is_empty() {
49            let real = PagedImage::<f32>::open(real).map_err(invalid)?;
50            let path = grid.join(format!("zero-vp-imaginary-{index}.im"));
51            let mut zero =
52                PagedImage::<f32>::create(real.shape().to_vec(), real.coordinates().clone(), &path)
53                    .map_err(invalid)?;
54            zero.save().map_err(invalid)?;
55            path.to_string_lossy().into_owned()
56        } else {
57            imaginary.to_owned()
58        };
59        record.upsert("imagim", Value::Scalar(ScalarValue::String(imaginary)));
60        replacements.push((index, record));
61    }
62    if replacements.is_empty() {
63        return Ok(source.to_owned());
64    }
65    // The pinned reader needs the misspelled field. Keep the caller's
66    // table intact and own the compatible copy with the Fourier image.
67    let mut copy = table.copy_to(&destination).map_err(invalid)?;
68    for (index, record) in replacements {
69        copy.row_accessor_mut()
70            .set_record_cell(index, "pbdescription", record)?;
71    }
72    copy.save(TableOptions::new(&destination))?;
73    Ok(destination)
74}

76pub(super) fn relocate_mosaic_table(
77    grid_record: &mut RecordValue,
78    source_grid: &Path,
79    target_grid: &Path,
80) -> MsResult<()> {
81    let Some(Value::Record(beam)) = grid_record.get("pbconvfunc") else {
82        return Ok(());
83    };
84    let source = source_grid.join(MANAGED_VP_TABLE);
85    if text(beam, "vptable").map(Path::new) != Some(source.as_path()) {
86        return Ok(());
87    }
88    let target = target_grid.join(MANAGED_VP_TABLE);
89    let mut table = Table::open(TableOptions::new(&target))?;
90    for row in 0..table.row_count() {
91        let mut record = table.cell_accessor(row, "pbdescription")?.record()?;
92        let Some(relative) =
93            text(&record, "imagim").and_then(|path| Path::new(path).strip_prefix(source_grid).ok())
94        else {
95            continue;
96        };
97        let path = target_grid.join(relative).to_string_lossy().into_owned();
98        record.upsert("imagim", Value::Scalar(ScalarValue::String(path)));
99        table
100            .row_accessor_mut()
101            .set_record_cell(row, "pbdescription", record)?;
102    }
103    table.save(TableOptions::new(&target))?;
104    let mut beam = beam.clone();
105    beam.upsert(
106        "vptable",
107        Value::Scalar(ScalarValue::String(target.to_string_lossy().into_owned())),
108    );
109    grid_record.upsert("pbconvfunc", Value::Record(beam));
110    Ok(())
111}

121    let host = HostResources::detect().map_err(invalid)?;

127fn read_legacy_radial<T>(
128    host: &HostResources,
129    path: &Path,
130    telescope: &str,
131    decode: impl FnOnce(&RecordValue) -> MsResult<T>,
132) -> MsResult<T> {
133    let _table_memory = admit(
134        host,
135        &ResourcePolicy::Balanced,
136        &Demand {
137            phase: "legacy voltage-pattern table",
138            memory: voltage_pattern::VoltagePatternTable::reading_bytes(path).map_err(invalid)?,
139        },
140    )
141    .map_err(invalid)?;
142    let table = Table::open(TableOptions::new(path))?;
143    // VPSkyJones installs the first concrete root row for each telescope.
144    // VPManager's defaults table belongs to a separate selection path.
145    for row in 0..table.row_count() {
146        let name = table.cell_accessor(row, "telescope")?.scalar()?;
147        if !matches!(name, ScalarValue::String(name) if name == telescope) {
148            continue;
149        }
150        let record = table.cell_accessor(row, "pbdescription")?.record()?;
151        match record.get("name") {
152            Some(Value::Scalar(ScalarValue::String(name))) if name != "REFERENCE" => {
153                return decode(&record);
154            }
155            Some(Value::Scalar(ScalarValue::String(_))) | None => {}
156            _ => return Err(invalid("legacy voltage pattern name must be a string")),
157        }
158    }
159    Err(invalid(format!("no voltage pattern for {telescope}")))
160}

162fn legacy_radial_setup(
163    host: &HostResources,
164    record: &RecordValue,
165    requested_squint: bool,
166    allow_curves: bool,
167) -> MsResult<TableVoltagePattern> {
168    let (peak, retained) = voltage_pattern::radial_curve_storage(record);
169    let mut reservation = admit(
170        host,
171        &ResourcePolicy::Balanced,
172        &Demand {
173            phase: "legacy voltage-pattern lookup",
174            memory: peak,
175        },
176    )
177    .map_err(invalid)?;
178    let value = if allow_curves {
179        voltage_pattern::radial_lookup(record)
180    } else {
181        voltage_pattern::radial_pattern(record).map(PrimaryBeam::pattern)
182    }
183    .map_err(invalid)?;
184    reservation.retain(retained);
185    let squint = if requested_squint {
186        voltage_pattern::radial_squint(record)
187            .map_err(invalid)?
188            .map(|(offset_rad, reference_hz)| BeamSquintLaw::Table {
189                offset_rad,
190                reference_hz,
191            })
192    } else {
193        None
194    };
195    let projected_image = allow_curves
196        && matches!(
197            text(record, "name"),
198            Some("POLY" | "IPOLY" | "COSPOLY" | "NUMERIC")
199        );
200    Ok(TableVoltagePattern {
201        pattern: Arc::new(Admitted { value, reservation }),
202        squint,
203        projected_image,
204    })
205}

231#[derive(Clone, Copy)]
232struct BeamSquint {
233    law: BeamSquintLaw,
234    angle: f64,
235}
236
237#[derive(Clone, Copy)]
238enum BeamSquintLaw {
239    Telescope {
240        beam: PrimaryBeam,
241        reference_hz: f64,
242    },
243    Table {
244        offset_rad: [f64; 2],
245        reference_hz: f64,
246    },
247}
248
249impl BeamSquint {
250    fn displacement(self, frequency: f64, right: bool) -> [f64; 2] {
251        match self.law {
252            BeamSquintLaw::Telescope { beam, reference_hz } => {
253                beam.squint_displacement_rad(reference_hz, frequency, self.angle, right)
254            }
255            BeamSquintLaw::Table {
256                offset_rad,
257                reference_hz,
258            } => PrimaryBeam::stored_squint_displacement_rad(
259                offset_rad,
260                reference_hz,
261                frequency,
262                self.angle,
263                right,
264            ),
265        }
266    }
267}

269struct ComponentVoltage<'a> {
270    pattern: &'a PreparedVoltagePattern,
271    component: &'a SkyComponent,
272    datum_basis: Polarisation,
273    original: Flux,
274    beam_flux: Flux,
275    stokes: [f64; 4],
276    centre: MDirection,
277    separation: f64,
278}
279
280impl ComponentVoltage<'_> {
281    fn flux_at(&self, frequency: f64, first_channel: bool) -> MsResult<[Complex64; 4]> {
282        let pattern = self.pattern.pattern();
283        let power = |separation| f64::from(pattern.power_at(separation, frequency));
284        let central = power(self.separation);
285        let mut gains = [central; 4];
286        if let Some(squint) = self.pattern.squint {
287            for (index, right) in [(0, true), (3, false)] {
288                let shift = squint.displacement(frequency, right);
289                let centre = offset_true_angle(&self.centre, shift).map_err(invalid)?;
290                gains[index] = power(angular_separation(
291                    self.component.shape().reference_direction().as_angles(),
292                    centre.as_angles(),
293                ));
294            }
295        }
296        let mut flux = self.beam_flux.clone();
297        let values = *flux.values();
298        flux.set_values(std::array::from_fn(|index| values[index] * gains[index]));
299        flux.convert_polarisation(self.original.polarisation());
300        let sample_basis = if self.pattern.component_squint && !first_channel {
301            // PBMath converts the input to circular during the first channel's
302            // application. Later channels sample that converted component.
303            Polarisation::Circular
304        } else {
305            self.original.polarisation()
306        };
307        let spectrum = stokes_in(sample_basis, self.sample_stokes(frequency)?);
308        // BeamSkyJones samples a point spectrum in the original basis,
309        // normalizes by the magnitude of the stored flux and tabulates
310        // the result. This also preserves its signed-flux behavior.
311        let values = *flux.values();
312        flux.set_values(std::array::from_fn(|index| {
313            let amplitude = self.original.values()[index].norm();
314            let scale = if amplitude > 0. {
315                amplitude.recip()
316            } else {
317                1.
318            };
319            values[index] * spectrum[index] * scale
320        }));
321        flux.convert_polarisation(self.datum_basis);
322        Ok(*flux.values())
323    }
324
325    fn sample_stokes(&self, frequency: f64) -> MsResult<[f64; 4]> {
326        let spectrum = self.component.spectrum();
327        let in_frame = |reference| {
328            MFrequency::new(frequency, FrequencyRef::LSRK)
329                .convert_to(reference, &MeasFrame::new())
330                .map(|converted| converted.hz())
331                .map_err(invalid)
332        };
333        match spectrum.kind() {
334            SpectralKind::Constant => Ok(self.stokes),
335            SpectralKind::Tabular(table) => {
336                let sample = if matches!(table.frame(), FrequencyRef::LSRK | FrequencyRef::REST) {
337                    // The scalar tabular sampler uses its reference when the
338                    // channel and table already use the same frame.
339                    spectrum.reference_frequency().hz()
340                } else {
341                    in_frame(table.frame())?
342                };
343                let frequencies = table.frequencies_hz();
344                if !frequencies
345                    .first()
346                    .zip(frequencies.last())
347                    .is_some_and(|(&first, &last)| {
348                        (first.min(last)..=first.max(last)).contains(&sample)
349                    })
350                {
351                    return Err(invalid(
352                        "component beam sample is outside the tabular spectrum",
353                    ));
354                }
355                spectrum.sample_stokes(sample, self.stokes).map_err(invalid)
356            }
357            kind => {
358                let sample = in_frame(spectrum.reference_frequency().refer())?;
359                let mut stokes = spectrum
360                    .sample_stokes(sample, self.stokes)
361                    .map_err(invalid)?;
362                if matches!(kind, SpectralKind::SpectralIndex(index) if index.stokes_indices().is_none())
363                {
364                    stokes[1..].copy_from_slice(&self.stokes[1..]);
365                }
366                Ok(stokes)
367            }
368        }
369    }
370}

392/// One apparent spectrum per component in the current legacy buffer. In
393/// particular, the buffer's epoch follows frame-dependent sky directions.
394pub(crate) struct ComponentVoltageCache {
395    pattern: PreparedVoltagePattern,
396    last: Mutex<Option<(SpectrumBufferKey, Arc<ApparentComponentSpectra>)>>,
397}
398
399impl ComponentVoltageCache {
400    pub(crate) fn new(pattern: PreparedVoltagePattern) -> Self {
401        Self {
402            pattern,
403            last: Mutex::new(None),
404        }
405    }
406
407    pub(crate) fn spectra(
408        &self,
409        components: &ComponentList,
410        frequencies: &[f64],
411        basis: Polarisation,
412        (window, epoch, start_channel): (i32, f64, usize),
413        frame: &MeasFrame,
414    ) -> MsResult<Arc<ApparentComponentSpectra>> {
415        let key = SpectrumBufferKey {
416            source: std::ptr::from_ref(components) as usize,
417            window,
418            epoch: epoch.to_bits(),
419            start_channel,
420            basis,
421        };
422        let mut last = self.last.lock().map_err(invalid)?;
423        if let Some((held, spectra)) = last.as_ref()
424            && *held == key
425        {
426            return Ok(Arc::clone(spectra));
427        }
428        *last = None;
429        let cells = components
430            .len()
431            .checked_mul(frequencies.len())
432            .ok_or_else(|| invalid("component beam spectrum exceeds addressable storage"))?;
433        let host = HostResources::detect().map_err(invalid)?;
434        let memory = admit(
435            &host,
436            &ResourcePolicy::Balanced,
437            &Demand {
438                phase: "legacy component voltage spectrum",
439                memory: (cells as u64).saturating_mul(size_of::<[Complex64; 4]>() as u64),
440            },
441        )
442        .map_err(invalid)?;
443        let mut values = Vec::with_capacity(cells);
444        for component in components.iter() {
445            let response = self.pattern.component(component, basis, frame)?;
446            for (channel, &frequency) in frequencies.iter().enumerate() {
447                values.push(response.flux_at(frequency, channel == start_channel)?);
448            }
449        }
450        let spectra = Arc::new(ApparentComponentSpectra {
451            values,
452            _memory: memory,
453        });
454        *last = Some((key, Arc::clone(&spectra)));
455        Ok(spectra)
456    }
457}

459/// Partition scratch rows in legacy visibility-buffer order. The first row
460/// in each state supplies the apparent-sky image for that state.
461pub(super) fn scratch_voltage_batches<'a>(
462    ms: &MeasurementSet,
463    selection: &'a ResolvedMsSelection,
464    increment: f64,
465    local_pointings: bool,
466) -> MsResult<impl Iterator<Item = Vec<ResolvedMsSelectionRow>> + use<'a>> {
467    let host = HostResources::detect().map_err(invalid)?;
468    let order_charge = admit(
469        &host,
470        &ResourcePolicy::Balanced,
471        &Demand {
472            phase: "legacy voltage-pattern scratch batches",
473            memory: voltage_order_bytes(selection.selected_rows.len()),
474        },
475    )
476    .map_err(invalid)?;
477    let main = ms.main_table();
478    let observations = ms.observation()?.row_count();
479    let key = |row: &ResolvedMsSelectionRow| -> MsResult<_> {
480        let ScalarValue::Int32(array) = main.cell_accessor(row.row_index, "ARRAY_ID")?.scalar()?
481        else {
482            return Err(invalid("ARRAY_ID must be Int32"));
483        };
484        if usize::try_from(*array).map_or(true, |id| id >= observations) {
485            return Err(invalid(
486                "ARRAY_ID has no observation for this voltage pattern",
487            ));
488        }
489        Ok((*array, row.field_id, row.data_desc_id))
490    };
491    let mut ordered = selection
492        .selected_rows
493        .iter()
494        .map(|row| Ok((key(row)?, row)))
495        .collect::<MsResult<Vec<_>>>()?;
496    ordered.sort_unstable_by(|(a, left), (b, right)| {
497        a.cmp(b)
498            .then_with(|| left.time_mjd_seconds.total_cmp(&right.time_mjd_seconds))
499            .then_with(|| left.row_index.cmp(&right.row_index))
500    });
501    let buffers = usize::from(!ordered.is_empty())
502        + ordered
503            .windows(2)
504            .filter(|pair| {
505                pair[0].0 != pair[1].0 || pair[0].1.time_mjd_seconds != pair[1].1.time_mjd_seconds
506            })
507            .count();
508    let _working_charge = admit(
509        &host,
510        &ResourcePolicy::Balanced,
511        &Demand {
512            phase: "legacy voltage-pattern scheduling workspace",
513            memory: voltage_planning_bytes(ms, buffers, local_pointings),
514        },
515    )
516    .map_err(invalid)?;
517    let feed_geometry = load_parallactic_angle_state(ms)?;
518    let pointing = local_pointings.then(|| ms.pointing()).transpose()?;
519    let mut buffer = None;
520    let mut held = None;
521    let mut starts = Vec::new();
522    for (index, &((array, field, dd), row)) in ordered.iter().enumerate() {
523        let time = row.time_mjd_seconds;
524        let current = (array, field, dd, time);
525        // CubeSkyEquation's internal-change probe stops at row zero in this
526        // path. The other rows in a legacy visibility buffer share its beam.
527        if buffer == Some(current) {
528            continue;
529        }
530        buffer = Some(current);
531        let angle = |antenna, column| -> MsResult<f32> {
532            let ScalarValue::Int32(feed) = main.cell_accessor(row.row_index, column)?.scalar()?
533            else {
534                return Err(invalid(format!("{column} must be Int32")));
535            };
536            Ok(feed_geometry
537                .feed_parallactic_angle(time, field, antenna, *feed, row.spectral_window_id)
538                .map_err(invalid)? as f32)
539        };
540        let first = angle(row.antenna1_id, "FEED1")?;
541        let second = angle(row.antenna2_id, "FEED2")?;
542        let refresh = held.as_ref().is_none_or(|state: &HeldVoltageState| {
543            (array, field) != state.field
544                || (state.local && time != state.time)
545                || f64::from((first - state.angle).abs()) > increment
546                || f64::from((second - state.angle).abs()) > increment
547        });
548        if refresh {
549            starts.push(index);
550            let local = if let Some(pointing) = &pointing {
551                pointing
552                    .legacy_direction(row.antenna2_id, time)?
553                    .is_some_and(|direction| is_local_direction(direction.refer()))
554            } else {
555                false
556            };
557            held = Some(HeldVoltageState {
558                field: (array, field),
559                time,
560                angle: second,
561                local,
562            });
563        }
564    }
565    starts.push(ordered.len());
566    tracing::debug!(
567        batches = starts.len() - 1,
568        rows = ordered.len(),
569        "planned voltage-pattern refreshes"
570    );
571    Ok((0..starts.len() - 1).map(move |batch| {
572        let _charge = &order_charge;
573        let mut rows = ordered[starts[batch]..starts[batch + 1]]
574            .iter()
575            .map(|(_, row)| (**row).clone())
576            .collect::<Vec<_>>();
577        rows.sort_unstable_by_key(|row| row.row_index);
578        rows
579    }))
580}

582fn voltage_order_bytes(rows: usize) -> u64 {
583    let per_row = size_of::<((i32, i32, i32), &ResolvedMsSelectionRow)>()
584        + 2 * size_of::<usize>()
585        + size_of::<ResolvedMsSelectionRow>();
586    (rows as u64).saturating_mul(per_row as u64)
587}
588
589fn voltage_planning_bytes(ms: &MeasurementSet, buffers: usize, local_pointings: bool) -> u64 {
590    let feeds = ms.feed().map_or(0, |feed| feed.row_count()) as u64;
591    let antennas = ms.antenna().map_or(0, |antennas| antennas.row_count()) as u64;
592    let fields = ms.field().map_or(0, |fields| fields.row_count()) as u64;
593    // The Measures provider can load its catalogs on the first angle.
594    // Include FEED hash buckets and vector capacity, the engine's antenna
595    // and field state, and caches for both antennas per visibility buffer.
596    (casa_ms::MEASURES_RUNTIME_BYTES as u64)
597        .saturating_add(128_u64 << 10)
598        .saturating_add(feeds.saturating_mul(256))
599        .saturating_add(antennas.saturating_mul(256))
600        .saturating_add(fields.saturating_mul(512))
601        .saturating_add((buffers as u64).saturating_mul(1024))
602        .saturating_add(if local_pointings {
603            voltage_pointing_bytes(ms)
604        } else {
605            0
606        })
607}
608
609fn voltage_pointing_bytes(ms: &MeasurementSet) -> u64 {
610    // A local pointing lookup can collect every timestamp for one antenna.
611    let rows = ms.pointing().map_or(0, |pointing| pointing.row_count()) as u64;
612    rows.saturating_mul(96)
613}

622pub(super) fn pointing_may_use_local_frame(ms: &MeasurementSet) -> bool {
623    let Some(pointing) = ms.subtable(SubtableId::Pointing) else {
624        return false;
625    };
626    if pointing.row_count() == 0 {
627        return false;
628    }
629    let Some(Value::Record(info)) = pointing
630        .column_keywords("DIRECTION")
631        .and_then(|keywords| keywords.get("MEASINFO"))
632    else {
633        return false;
634    };
635    info.get("VarRefCol").is_some()
636        || text(info, "Ref")
637            .and_then(|reference| reference.parse().ok())
638            .is_some_and(is_local_direction)
639}

642    matches!(
643        reference,
644        DirectionRef::HADEC
645            | DirectionRef::AZEL
646            | DirectionRef::AZELSW
647            | DirectionRef::AZELGEO
648            | DirectionRef::AZELSWGEO
649    )

652impl ImagerVoltagePattern {
653    pub(super) fn prepare(
654        &self,
655        ms: &MeasurementSet,
656        selected: &ResolvedMsSelection,
657        squint: bool,
658        scratch: bool,
659        allow_curves: bool,
660    ) -> MsResult<PreparedVoltagePattern> {
661        let main = ms.main_table();
662        let mut first = None;
663        let mut first_array = None;
664        let mut mixed_arrays = false;
665        for row in &selected.selected_rows {
666            let ScalarValue::Int32(array) =
667                main.cell_accessor(row.row_index, "ARRAY_ID")?.scalar()?
668            else {
669                return Err(invalid("ARRAY_ID must be Int32"));
670            };
671            let key = (*array, row.field_id, row.data_desc_id, row.time_mjd_seconds);
672            mixed_arrays |= *first_array.get_or_insert(*array) != *array;
673            if first.as_ref().is_none_or(|(previous, _)| key < *previous) {
674                first = Some((key, row));
675            }
676        }
677        let Some(((array, field, _, time), first)) = first else {
678            return Err(invalid("a voltage pattern needs a selected visibility"));
679        };
680        let telescope = ms
681            .observation()?
682            .string(usize::try_from(array).map_err(invalid)?, "TELESCOPE_NAME")?;
683        let host = HostResources::detect().map_err(invalid)?;
684        let default_beam = self
685            .table
686            .is_none()
687            .then(|| -> MsResult<_> {
688                let frequency = ms.spectral_window()?.ref_frequency(0)?;
689                Ok(PrimaryBeam::for_telescope(&telescope, frequency).map(|beam| (beam, frequency)))
690            })
691            .transpose()?
692            .flatten();
693        let custom = self
694            .table
695            .as_ref()
696            .map(|path| {
697                read_legacy_radial(&host, path, &telescope, |record| {
698                    legacy_radial_setup(&host, record, squint, allow_curves)
699                })
700            })
701            .transpose()?;
702        let component_squint = squint;
703        let squint_law = if squint {
704            match (custom.as_ref(), default_beam) {
705                (Some(custom), _) => custom.squint,
706                (None, Some((beam, reference_hz)))
707                    if beam.squint_displacement_rad(reference_hz, reference_hz, 0., true)
708                        != [0.; 2] =>
709                {
710                    Some(BeamSquintLaw::Telescope { beam, reference_hz })
711                }
712                // CASA accepts GOFIGURE for the other default profiles and
713                // its dish-diameter fallback, which have no physical offset.
714                (None, _) => None,
715            }
716        } else {
717            None
718        };
719        if squint_law.is_some() && scratch && mixed_arrays {
720            return Err(invalid(
721                "legacy voltage-pattern squint for a field observed by multiple arrays is pending #1520",
722            ));
723        }
724        let projected_model = custom.as_ref().is_some_and(|custom| custom.projected_image)
725            || (telescope == "VLA" && self.table.is_none())
726            || squint_law.is_some();
727        let pattern = if let Some(custom) = custom {
728            custom.pattern
729        } else {
730            let reservation = reserve_radial_lookup(&host)?;
731            let value = match default_beam {
732                Some((pattern, _)) => pattern.pattern(),
733                None => BeamPattern::for_dish_diameter(ms.antenna()?.dish_diameter(0)?)
734                    .ok_or_else(|| {
735                        invalid("no usable dish diameter for a default voltage pattern")
736                    })?,
737            };
738            Arc::new(Admitted { value, reservation })
739        };
740        let engine = if squint_law.is_some() {
741            MsCalEngine::for_calibration(ms)?
742        } else {
743            MsCalEngine::new(ms)?
744        };
745        let pointing = if ms.subtable(SubtableId::Pointing).is_some() {
746            // A homogeneous BeamSkyJones shares one entry for both antennas.
747            // Its update stores the second antenna's direction last.
748            ms.pointing()?.legacy_direction(first.antenna2_id, time)?
749        } else {
750            None
751        };
752        let centre = match pointing {
753            Some(centre) => centre,
754            None => engine
755                .observation_direction_j2000(time, usize::try_from(field).map_err(invalid)?)?,
756        };
757        let frame = engine.spectral_frame_observatory_direction(time, centre.clone())?;
758        let squint = if let Some(law) = squint_law {
759            if scratch
760                && is_local_direction(centre.refer())
761                && selected
762                    .selected_rows
763                    .iter()
764                    .any(|row| row.time_mjd_seconds != time)
765            {
766                return Err(invalid(
767                    "legacy voltage-pattern squint with changing local pointing is pending #1520",
768                ));
769            }
770            let ScalarValue::Int32(feed) =
771                main.cell_accessor(first.row_index, "FEED2")?.scalar()?
772            else {
773                return Err(invalid("FEED2 must be Int32"));
774            };
775            let angle = engine.feed_parallactic_angle(
776                time,
777                usize::try_from(field).map_err(invalid)?,
778                usize::try_from(first.antenna2_id).map_err(invalid)?,
779            )? + single_receptor_angle(
780                ms,
781                first.antenna2_id,
782                *feed,
783                first.spectral_window_id,
784                time,
785            )?;
786            Some(BeamSquint { law, angle })
787        } else {
788            None
789        };
790        Ok(PreparedVoltagePattern {
791            pattern,
792            centre,
793            frame,
794            projected_model,
795            squint,
796            component_squint,
797        })
798    }
799}

820    fn component<'a>(
821        &'a self,
822        component: &'a SkyComponent,
823        datum_basis: Polarisation,
824        frame: &MeasFrame,
825    ) -> MsResult<ComponentVoltage<'a>> {
826        let direction = component.shape().reference_direction();
827        let centre = self
828            .centre
829            .convert_to(direction.refer(), frame)
830            .map_err(invalid)?;
831        let separation = angular_separation(direction.as_angles(), centre.as_angles());
832        let basis = match component.flux().polarisation() {
833            Polarisation::Stokes => datum_basis,
834            basis => basis,
835        };
836        let [i, q, u, v] = component.flux().stokes_jy_values().map_err(invalid)?;
837        let mut original = Flux::stokes_jy(i, q, u, v);
838        original.convert_polarisation(basis);
839        let mut beam_flux = original.clone();
840        if self.component_squint {
841            beam_flux.convert_polarisation(Polarisation::Circular);
842        }
843        Ok(ComponentVoltage {
844            pattern: self,
845            component,
846            datum_basis,
847            original,
848            beam_flux,
849            stokes: [i, q, u, v],
850            centre,
851            separation,
852        })
853    }

897    fn shifted_pixels(
898        &self,
899        direction: &DirectionCoordinate,
900        centre: &MDirection,
901        frequency: f64,
902    ) -> MsResult<Option<[[f64; 2]; 2]>> {
903        let Some(squint) = self.squint else {
904            return Ok(None);
905        };

908            let shifted_centre = offset_true_angle(centre, offset).map_err(invalid)?;

914    pub(super) fn model_plane(
915        &self,
916        direction: &DirectionCoordinate,
917        radial: Option<&Array2<f64>>,
918        values: &ArrayD<f32>,
919        frequency: f64,
920    ) -> MsResult<Vec<Array2<f32>>> {
921        let [nx, ny, npol, _] = *values.shape() else {
922            unreachable!("legacy model plane shape");
923        };
924        let projected = if radial.is_none() {
925            let centre = self.centre_in_image_frame(direction)?;
926            Some(pixel_centre(direction, &centre)?)
927        } else {
928            None
929        };
930        let pattern = self.pattern().at_frequency(frequency);
931        let power = |distance: f64| pattern.power(distance.to_degrees() * 60.0 * frequency / 1e9);
932        let increments = direction.increment();
933        let increment_deg = [increments[0].to_degrees(), increments[1].to_degrees()];
934        let projected_power =
935            |x, y, centre| pattern.power(projected_radius(x, y, centre, increment_deg, frequency));
936        Ok((0..npol)
937            .map(|pol| {
938                Array2::from_shape_fn((ny, nx), |(y, x)| {
939                    if let Some(centre) = projected {
940                        return values[[x, y, pol, 0]] * projected_power(x, y, centre);
941                    }
942                    values[[x, y, pol, 0]] * power(radial.expect("radial plane")[[y, x]])
943                })
944            })
945            .collect())
946    }

948    pub(super) fn squinted_model_tapers(
949        &self,
950        direction: &DirectionCoordinate,
951        [nx, ny]: [usize; 2],
952        frequency: f64,
953        polarizations: &[GridPolarization],
954    ) -> MsResult<Vec<Array2<f32>>> {
955        let centre = self.centre_in_image_frame(direction)?;
956        let shifted = self.shifted_pixels(direction, &centre, frequency)?;
957        let centre = pixel_centre(direction, &centre)?;
958        let [right, left] = shifted.unwrap_or([centre; 2]);
959        let pattern = self.pattern().at_frequency(frequency);
960        let increments = direction.increment();
961        let increment_deg = [increments[0].to_degrees(), increments[1].to_degrees()];
962        Ok(polarizations
963            .iter()
964            .map(|polarization| {
965                let centre = match polarization {
966                    GridPolarization::Correlation(Correlation::RR) => right,
967                    GridPolarization::Correlation(Correlation::LL) => left,
968                    _ => centre,
969                };
970                Array2::from_shape_fn((ny, nx), |(y, x)| {
971                    pattern.power(projected_radius(x, y, centre, increment_deg, frequency))
972                })
973            })
974            .collect())
975    }

983    fn separations_from(
984        direction: &DirectionCoordinate,
985        [nx, ny]: [usize; 2],
986        centre: &MDirection,
987    ) -> MsResult<Array2<f64>> {
988        let centre = centre.as_angles();
989        Ok(Array2::from_shape_fn((ny, nx), |(y, x)| {
990            let Ok([lon, lat]) = direction.pixel_to_world([x as f64, y as f64]) else {
991                return f64::INFINITY;
992            };
993            angular_separation((lon, lat), centre)
994        }))
995    }

998fn angular_separation((lon, lat): (f64, f64), (lon0, lat0): (f64, f64)) -> f64 {
999    let separation = ((lat - lat0) * 0.5).sin().powi(2)
1000        + lat.cos() * lat0.cos() * ((lon - lon0) * 0.5).sin().powi(2);
1001    2.0 * separation.clamp(0., 1.).sqrt().asin()
1002}

1037fn pixel_centre(direction: &DirectionCoordinate, centre: &MDirection) -> MsResult<[f64; 2]> {
1038    let (lon, lat) = centre.as_angles();
1039    direction.world_to_pixel([lon, lat]).map_err(invalid)

1042fn projected_radius(
1043    x: usize,
1044    y: usize,
1045    centre: [f64; 2],
1046    increment_deg: [f64; 2],
1047    frequency: f64,
1048) -> f64 {
1049    // PBMath1D rounds the squared components, radius and lookup position to Float.
1050    let x2 = ((x as f64 - centre[0]) * increment_deg[0]).powi(2) as f32;
1051    let y2 = ((y as f64 - centre[1]) * increment_deg[1]).powi(2) as f32;
1052    f64::from((f64::from((x2 + y2).sqrt()) * 60.0 * frequency / 1e9) as f32)
1053}
