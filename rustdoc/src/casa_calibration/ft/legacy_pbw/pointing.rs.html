14const OFFSETS: &str = "casars_legacy_pbw_ep_offsets";
15const EPOCHS: &str = "casars_legacy_pbw_ep_epochs";

17pub(crate) struct LegacyEpOffsets {
18    antennas: usize,
19    epochs: Vec<f64>,
20    values: Vec<Option<[f64; 2]>>,
21    _memory: Reservation,
22}
23
24impl LegacyEpOffsets {
25    pub(in crate::ft) fn for_selection(
26        path: &Path,
27        ms: &MeasurementSet,
28        selected: &ResolvedMsSelection,
29    ) -> MsResult<Self> {
30        if selected.selected_rows.is_empty() {
31            return Err(invalid("no rows selected for legacy pointing offsets"));
32        }
33        let offsets = Self::open(
34            path,
35            ms.antenna()?.row_count(),
36            ms.spectral_window()?.row_count(),
37        )?;
38        offsets.check_selection(selected)?;
39        Ok(offsets)
40    }
41
42    pub(in crate::ft) fn check_selection(&self, selected: &ResolvedMsSelection) -> MsResult<()> {
43        for row in &selected.selected_rows {
44            self.baseline_offset([row.antenna1_id, row.antenna2_id], row.time_mjd_seconds)?;
45        }
46        Ok(())
47    }

49    fn open(path: &Path, antennas: usize, window_count: usize) -> MsResult<Self> {
50        let table = Table::open(TableOptions::new(path))?;
51        let info = table.info();
52        if info.table_type != "Calibration" || info.sub_type != "EP Jones" || table.row_count() == 0
53        {
54            return Err(invalid(
55                "a legacy EP Jones pointing table with solutions is required",
56            ));
57        }
58        let description_name = table
59            .keywords()
60            .get("CAL_DESC")
61            .and_then(Value::as_table_ref)
62            .ok_or_else(|| invalid("legacy EP table has no CAL_DESC table link"))?;
63        let host = HostResources::detect().map_err(invalid)?;
64        let _reading = admit(
65            &host,
66            &ResourcePolicy::Balanced,
67            &Demand {
68                phase: "legacy EP pointing table read",
69                memory: (table.row_count() as u64)
70                    .saturating_mul(512)
71                    .saturating_add(64 << 10),
72            },
73        )
74        .map_err(invalid)?;
75        let description = Table::open(TableOptions::new(add_directory(description_name, path)))?;
76        let active = pointing_description(&description, window_count)?;
77        let desc_ids = table.column_accessor("CAL_DESC_ID")?.scalar_cells_owned()?;
78        let mut indices = Vec::new();
79        for (index, value) in desc_ids.into_iter().enumerate() {
80            let id = match value {
81                Some(ScalarValue::Int32(value)) => usize::try_from(value).ok(),
82                Some(ScalarValue::Int64(value)) => usize::try_from(value).ok(),
83                _ => None,
84            };
85            let id = id
86                .filter(|&id| id < description.row_count())
87                .ok_or_else(|| {
88                    invalid("legacy EP CAL_DESC_ID must reference a calibration description")
89                })?;
90            if id == active {
91                indices.push(index);
92            }
93        }
94        let samples = indices.len();
95        let slots = samples
96            .checked_div(antennas)
97            .filter(|&count| count > 0 && count * antennas == samples)
98            .ok_or_else(|| {
99                invalid("legacy EP table has incomplete spectral-window-zero antenna solutions")
100            })?;
101        let mut offsets = Self::empty(antennas, slots, &host)?;
102        let mut rows = Vec::with_capacity(samples);
103        let gains = table.column_accessor("GAIN")?;
104        for indices in indices.chunks(256) {
105            let cells = gains.array_cells_owned_uncached(indices)?;
106            for (&row, value) in indices.iter().zip(cells) {
107                let ScalarValue::Float64(epoch) = table.cell_accessor(row, "TIME")?.scalar()?
108                else {
109                    return Err(invalid("legacy EP TIME must be Float64"));
110                };
111                if !epoch.is_finite() {
112                    return Err(invalid("legacy EP solution times must be finite"));
113                }
114                let antenna =
115                    usize::try_from(integer(&table, row, "ANTENNA1")?).map_err(invalid)?;
116                if antenna >= antennas {
117                    return Err(invalid("legacy EP antenna is outside the selected array"));
118                }
119                // The legacy apply path reads the first azimuth/elevation pair
120                // (entries zero and two) for both visibility correlations.
121                let offset = match value {
122                    Some(ArrayValue::Float32(value)) if value.shape() == [4, 1] => {
123                        [f64::from(value[[0, 0]]), f64::from(value[[2, 0]])]
124                    }
125                    Some(ArrayValue::Float64(value)) if value.shape() == [4, 1] => [
126                        f64::from(value[[0, 0]] as f32),
127                        f64::from(value[[2, 0]] as f32),
128                    ],
129                    _ => {
130                        return Err(invalid(
131                            "legacy EP GAIN must contain four real offsets on one channel",
132                        ));
133                    }
134                };
135                if !offset.iter().all(|value| value.is_finite()) {
136                    return Err(invalid("legacy EP pointing offsets must be finite"));
137                }
138                rows.push((*epoch, antenna, offset));
139            }
140        }
141        rows.sort_unstable_by(|left, right| left.0.total_cmp(&right.0).then(left.1.cmp(&right.1)));
142        for (slot, rows) in rows.chunks_exact(antennas).enumerate() {
143            let epoch = rows[0].0;
144            if offsets.epochs.last() == Some(&epoch)
145                || rows
146                    .iter()
147                    .enumerate()
148                    .any(|(index, &(time, antenna, _))| time != epoch || antenna != index)
149            {
150                return Err(invalid(
151                    "legacy EP table does not match the MeasurementSet antenna count at each solution time",
152                ));
153            }
154            offsets.epochs.push(epoch);
155            for &(_, antenna, offset) in rows {
156                offsets.values[slot * antennas + antenna] = Some(offset);
157            }
158        }
159        Ok(offsets)
160    }

162    fn empty(antennas: usize, slots: usize, host: &HostResources) -> MsResult<Self> {
163        let samples = antennas
164            .checked_mul(slots)
165            .ok_or_else(|| invalid("legacy EP solution size overflows usize"))?;
166        let memory = admit(
167            host,
168            &ResourcePolicy::Balanced,
169            &Demand {
170                phase: "legacy EP antenna offsets",
171                // The retained record uses three Float32 values per antenna.
172                memory: (samples as u64)
173                    .saturating_mul((size_of::<Option<[f64; 2]>>() + 3 * size_of::<f32>()) as u64)
174                    .saturating_add((slots as u64).saturating_mul((2 * size_of::<f64>()) as u64)),
175            },
176        )
177        .map_err(invalid)?;
178        Ok(Self {
179            antennas,
180            epochs: Vec::with_capacity(slots),
181            values: vec![None; samples],
182            _memory: memory,
183        })
184    }

186    pub(super) fn save(&self, record: &mut RecordValue) {
187        let mut shape = vec![3, self.antennas];
188        if self.epochs.len() > 1 {
189            shape.push(self.epochs.len());
190            record.upsert(
191                EPOCHS,
192                Value::Array(ArrayValue::Float64(
193                    ArrayD::from_shape_vec(IxDyn(&[self.epochs.len()]), self.epochs.clone())
194                        .expect("one value per epoch"),
195                )),
196            );
197        }
198        let packed = self
199            .values
200            .iter()
201            .flat_map(|offset| offset.map_or([0.; 3], |[l, m]| [1., l as f32, m as f32]))
202            .collect();
203        let packed = ArrayD::from_shape_vec(IxDyn(&shape).f(), packed)
204            .expect("three values per antenna and slot");
205        record.upsert(OFFSETS, Value::Array(ArrayValue::Float32(packed)));
206    }

208    pub(crate) fn from_record(record: &RecordValue, antennas: usize) -> MsResult<Option<Self>> {
209        let recorded_epochs = record.get(EPOCHS);
210        let Some(packed) = record.get(OFFSETS) else {
211            if recorded_epochs.is_some() {
212                return Err(invalid(
213                    "saved legacy EP solution times have no pointing offsets",
214                ));
215            }
216            return Ok(None);
217        };
218        let Value::Array(ArrayValue::Float32(packed)) = packed else {
219            return Err(invalid(
220                "saved legacy EP offsets must contain Float32 values",
221            ));
222        };
223        let epochs = match packed.shape() {
224            [3, count] if *count == antennas => {
225                if recorded_epochs.is_some() {
226                    return Err(invalid(
227                        "single-slot saved legacy EP offsets have solution times",
228                    ));
229                }
230                None
231            }
232            [3, count, slots] if *count == antennas && *slots > 1 => {
233                let Some(Value::Array(ArrayValue::Float64(epochs))) = recorded_epochs else {
234                    return Err(invalid("saved legacy EP offsets have no solution times"));
235                };
236                if epochs.shape() != [*slots]
237                    || !epochs.iter().all(|epoch| epoch.is_finite())
238                    || epochs
239                        .iter()
240                        .zip(epochs.iter().skip(1))
241                        .any(|(left, right)| left >= right)
242                {
243                    return Err(invalid("saved legacy EP solution times are invalid"));
244                }
245                Some(epochs)
246            }
247            _ => {
248                return Err(invalid(
249                    "saved legacy EP offsets do not match the antenna count",
250                ));
251            }
252        };
253        let host = HostResources::detect().map_err(invalid)?;
254        let mut offsets = Self::empty(antennas, epochs.map_or(1, ArrayD::len), &host)?;
255        if let Some(epochs) = epochs {
256            offsets.epochs.extend(epochs.iter().copied());
257        } else {
258            offsets.epochs.push(0.);
259        }
260        for (index, slot) in offsets.values.iter_mut().enumerate() {
261            let [ok, l, m] = std::array::from_fn(|axis| {
262                let indices = [axis, index % antennas, index / antennas];
263                packed[IxDyn(&indices[..packed.ndim()])]
264            });
265            match ok {
266                0. if epochs.is_none() => {}
267                1. if l.is_finite() && m.is_finite() => *slot = Some([f64::from(l), f64::from(m)]),
268                _ => {
269                    return Err(invalid(
270                        "saved legacy EP offsets contain an invalid antenna entry",
271                    ));
272                }
273            }
274        }
275        Ok(Some(offsets))
276    }

278    pub(crate) fn baseline_offset(&self, antennas: [i32; 2], time: f64) -> MsResult<[f64; 2]> {
279        let slot = self.nearest_slot(time);
280        let values = &self.values[slot * self.antennas..][..self.antennas];
281        let offset = |antenna: i32| {
282            let antenna = usize::try_from(antenna).map_err(invalid)?;
283            values.get(antenna).copied().flatten().ok_or_else(|| {
284                invalid(format!(
285                    "legacy EP table has no offsets for antenna {antenna}"
286                ))
287            })
288        };
289        let left = offset(antennas[0])?;
290        let right = offset(antennas[1])?;
291        Ok(std::array::from_fn(|axis| (left[axis] + right[axis]) * 0.5))
292    }
293
294    fn nearest_slot(&self, time: f64) -> usize {
295        let upper = self.epochs.partition_point(|epoch| *epoch < time);
296        if upper == 0
297            || upper < self.epochs.len()
298                && self.epochs[upper] - time < time - self.epochs[upper - 1]
299        {
300            upper
301        } else {
302            upper - 1
303        }
304    }
305}

307fn pointing_description(table: &Table, window_count: usize) -> MsResult<usize> {
308    let mut active = None;
309    for row in 0..table.row_count() {
310        let window = table.cell_accessor(row, "SPECTRAL_WINDOW_ID")?.array()?;
311        let id = match window {
312            ArrayValue::Int32(values) => values.iter().next().map(|&id| i64::from(id)),
313            ArrayValue::Int64(values) => values.iter().next().copied(),
314            _ => None,
315        };
316        let id = id
317            .and_then(|id| usize::try_from(id).ok())
318            .filter(|&id| id < window_count)
319            .ok_or_else(|| invalid("legacy EP CAL_DESC refers to an unknown spectral window"))?;
320        // EPJones::nearest reads currSpw(), which the legacy imager keeps
321        // at zero even as it advances to another visibility spectral window.
322        if id == 0 && active.replace(row).is_some() {
323            return Err(invalid(
324                "legacy EP has multiple descriptions for spectral window zero",
325            ));
326        }
327    }
328    active
329        .ok_or_else(|| invalid("legacy EP has no calibration description for spectral window zero"))
330}

332fn integer(table: &Table, row: usize, name: &str) -> MsResult<i64> {
333    match table.cell_accessor(row, name)?.scalar()? {
334        ScalarValue::Int32(value) => Ok(i64::from(*value)),
335        ScalarValue::Int64(value) => Ok(*value),
336        _ => Err(invalid(format!("legacy EP {name} must contain integers"))),
337    }
