13/// A store's format.
14#[derive(Clone, Copy, Debug, PartialEq, Eq)]
15pub enum StoreFormat {
16    /// A MeasurementSet v2: a casacore table directory.
17    Msv2,
18    /// An MSv4 processing set: a Zarr group.
19    Msv4,
20}
21
22impl StoreFormat {
23    /// The format of the store at `path`, as casa-ms opens it.
24    #[must_use]
25    pub fn of(path: &Path) -> Self {
26        if casa_ms::observation_io::is_msv4_store(path) {
27            Self::Msv4
28        } else {
29            Self::Msv2
30        }
31    }
32
33    /// `msv2` or `msv4`.
34    #[must_use]
35    pub fn name(self) -> &'static str {
36        match self {
37            Self::Msv2 => "msv2",
38            Self::Msv4 => "msv4",
39        }
40    }
41
42    /// The format a conversion of this one makes by default.
43    #[must_use]
44    pub fn other(self) -> Self {
45        match self {
46            Self::Msv2 => Self::Msv4,
47            Self::Msv4 => Self::Msv2,
48        }
49    }
50
51    /// The extension XRADIO's convention gives a store of this format:
52    /// `.ms` for a MeasurementSet v2, `.ps.zarr` for a processing set.
53    #[must_use]
54    pub fn extension(self) -> &'static str {
55        match self {
56            Self::Msv2 => ".ms",
57            Self::Msv4 => ".ps.zarr",
58        }
59    }
60}

62/// The extensions a default output's name drops from its input's, longest
63/// first; any case.
64const MEASUREMENT_SET_EXTENSIONS: [&str; 3] = [".ps.zarr", ".zarr", ".ms"];
65
66/// The output a conversion of `vis` into `outformat` writes when none is
67/// named, by XRADIO's convention: beside `vis`, named after it, its
68/// extension (`.ms`, `.zarr` or `.ps.zarr`, any case) replaced by
69/// `.ps.zarr` for a processing set or `.ms` for a MeasurementSet v2
70/// ([`StoreFormat::extension`]), or that extension added to a name without
71/// one: `obs.ms` becomes `obs.ps.zarr`, and `obs.ps.zarr` `obs.ms`. `None`
72/// when `vis` names no file, or a name that is not UTF-8.
73#[must_use]
74pub fn default_outputvis(vis: &Path, outformat: StoreFormat) -> Option<PathBuf> {
75    let name = vis.file_name()?.to_str()?;
76    let lower = name.to_ascii_lowercase();
77    let stem = MEASUREMENT_SET_EXTENSIONS
78        .iter()
79        .find(|extension| lower.len() > extension.len() && lower.ends_with(*extension))
80        .map_or(name, |extension| &name[..name.len() - extension.len()]);
81    Some(vis.with_file_name(format!("{stem}{}", outformat.extension())))
82}
83
84/// `outputvis` as XRADIO's converter names a processing set it writes: a
85/// path that does not end in `ps.zarr` gains `.ps.zarr`, by text, so `out`
86/// is written as `out.ps.zarr` (and `.` as `..ps.zarr`, as XRADIO writes
87/// it). A trailing `/` is no part of the name.
88#[must_use]
89pub fn processing_set_path(outputvis: &str) -> PathBuf {
90    let trimmed = outputvis.trim_end_matches('/');
91    let path = if trimmed.is_empty() {
92        outputvis
93    } else {
94        trimmed
95    };
96    if path.ends_with("ps.zarr") {
97        PathBuf::from(path)
98    } else {
99        PathBuf::from(format!("{path}{}", StoreFormat::Msv4.extension()))
100    }
101}

103/// One conversion: its parameters read and checked, nothing done yet.
104#[derive(Clone, Debug)]
105pub struct MsconvertRequest {
106    /// The store converted, which exists.
107    pub vis: PathBuf,
108    /// The store written: `outputvis`, or [`default_outputvis`].
109    pub outputvis: PathBuf,
110    /// The format of `vis`.
111    pub informat: StoreFormat,
112    /// The format of `outputvis`.
113    pub outformat: StoreFormat,
114    /// The data groups converted; `None` for every one.
115    pub data_groups: Option<Vec<String>>,
116    /// MSv2 into MSv4: the keys the MSv2 is further partitioned by.
117    pub partition_scheme: PartitionScheme,
118    /// MSv4 output: whether pointing is converted.
119    pub pointing: bool,
120    /// MSv2 output: what a padded grid cell becomes.
121    pub padding: PaddingRows,
122    /// MSv4 output: the chunk shape, `None` for XRADIO's, sized for each
123    /// dataset
124    /// ([`xradio_chunks`](casa_ms::observation_io::convert::xradio_chunks)).
125    pub chunks: Option<[u64; 4]>,
126    /// MSv4 output: the payload codec.
127    pub codec: Msv4Codec,
128    /// MSv4 output: partitions converted at once, at least one.
129    pub workers: usize,
130    /// Replace an existing MeasurementSet at `outputvis`.
131    pub overwrite: bool,
132}
133
134/// Read the resolved run `task` into its request.
135///
136/// # Errors
137///
138/// A validation failure at the parameter that is wrong: `vis` missing, an
139/// unknown format, padding or codec, no `outputvis` and none to derive (or
140/// only `vis` itself), a data group the conversion would refuse, a
141/// partition scheme or pointing choice the conversion does not make, a
142/// chunk shape not of four non-negative extents, or a negative `nthreads`.
143pub fn request_from(task: &TaskInvocation<'_>) -> Result<MsconvertRequest, TaskFailure> {
144    let vis = task.workspace().join(task.input_path("vis")?);
145    let informat = StoreFormat::of(&vis);
146    let outformat = match lower(task.text("outformat")?).as_str() {
147        "" => informat.other(),
148        "msv2" => StoreFormat::Msv2,
149        "msv4" => StoreFormat::Msv4,
150        other => {
151            return Err(invalid(
152                "outformat",
153                format!("unknown outformat {other:?}; expected msv2, msv4 or empty"),
154            ));
155        }
156    };
157    let data_groups = data_groups(task, &vis, outformat)?;
158    let msv2_into_msv4 = informat == StoreFormat::Msv2 && outformat == StoreFormat::Msv4;
159    Ok(MsconvertRequest {
160        outputvis: outputvis(task, &vis, outformat)?,
161        vis,
162        informat,
163        outformat,
164        data_groups,
165        partition_scheme: partition_scheme(task, msv2_into_msv4)?,
166        pointing: pointing(task, outformat)?,
167        padding: padding(task)?,
168        chunks: chunks(task)?,
169        codec: codec(task)?,
170        workers: workers(task)?,
171        overwrite: task.flag("overwrite")?,
172    })
173}

175/// `outputvis`, or the default beside `vis` when it is empty.
176fn outputvis(
177    task: &TaskInvocation<'_>,
178    vis: &Path,
179    outformat: StoreFormat,
180) -> Result<PathBuf, TaskFailure> {
181    let named = task.text("outputvis")?.unwrap_or_default();
182    if !named.trim().is_empty() {
183        let named = match outformat {
184            StoreFormat::Msv4 => processing_set_path(&named),
185            StoreFormat::Msv2 => PathBuf::from(named),
186        };
187        return Ok(task.workspace().join(named));
188    }
189    let derived = default_outputvis(vis, outformat).ok_or_else(|| {
190        invalid(
191            "outputvis",
192            format!(
193                "no output name follows from vis {}; name outputvis",
194                vis.display()
195            ),
196        )
197    })?;
198    if derived == vis {
199        return Err(invalid(
200            "outputvis",
201            format!(
202                "the default {} output name for vis {} would be vis itself; name outputvis",
203                outformat.name(),
204                vis.display()
205            ),
206        ));
207    }
208    Ok(derived)
209}

211/// The data groups named, checked against the source as the conversion
212/// checks them, which opens it only when some are named: each in every
213/// partition, none twice, and into an MSv2 `base` first (it becomes
214/// `DATA`).
215fn data_groups(
216    task: &TaskInvocation<'_>,
217    vis: &Path,
218    outformat: StoreFormat,
219) -> Result<Option<Vec<String>>, TaskFailure> {
220    let named = names(task, "datagroups")?;
221    if named.is_empty() {
222        return Ok(None);
223    }
224    if outformat == StoreFormat::Msv2 && named[0] != "base" {
225        return Err(invalid(
226            "datagroups",
227            format!(
228                "an MSv2 output's first data group is base, its DATA; found {}",
229                named[0]
230            ),
231        ));
232    }
233    let observation = open(vis, &OpenOptions::default())
234        .map_err(|error| invalid("vis", format!("open {}: {error}", vis.display())))?;
235    selected_data_groups(observation.catalogue(), Some(&named))
236        .map_err(|error| invalid("datagroups", error.to_string()))?;
237    Ok(Some(named))
238}

240/// The keys an MSv2 converted into an MSv4 is further partitioned by,
241/// XRADIO's `partition_scheme`: `FIELD_ID` and `SCAN_NUMBER`. Refused for
242/// any other conversion, which has no MSv2 to partition.
243fn partition_scheme(
244    task: &TaskInvocation<'_>,
245    msv2_into_msv4: bool,
246) -> Result<PartitionScheme, TaskFailure> {
247    let mut scheme = PartitionScheme::default();
248    for key in names(task, "partitionscheme")? {
249        match key.to_ascii_uppercase().as_str() {
250            "FIELD_ID" => scheme.fields = true,
251            "SCAN_NUMBER" => scheme.scans = true,
252            _ => {
253                return Err(invalid(
254                    "partitionscheme",
255                    format!("unknown partition key {key:?}; expected FIELD_ID or SCAN_NUMBER"),
256                ));
257            }
258        }
259    }
260    if scheme != PartitionScheme::default() && !msv2_into_msv4 {
261        return Err(invalid(
262            "partitionscheme",
263            "partitionscheme partitions an MSv2 converted into an MSv4",
264        ));
265    }
266    Ok(scheme)
267}

269/// Whether pointing is converted: XRADIO's `with_pointing`, an MSv4
270/// output's choice; an MSv2 output always carries it.
271fn pointing(task: &TaskInvocation<'_>, outformat: StoreFormat) -> Result<bool, TaskFailure> {
272    let pointing = task.flag("withpointing")?;
273    if !pointing && outformat == StoreFormat::Msv2 {
274        return Err(invalid(
275            "withpointing",
276            "withpointing=False leaves pointing out of an MSv4 output; an MSv2 output carries it",
277        ));
278    }
279    Ok(pointing)
280}

282/// The threads `nthreads=0` takes on a host of `cores` cores and
283/// `physical_bytes` of memory: as many as half the memory holds at 1 GiB
284/// each (a conservative bound; a thread measured about
285/// 100 MB on a 1.4 GB MSv2 of 32 windows), at most one a core, and at
286/// least one. Unknown memory gives one.
287#[must_use]
288pub fn default_workers(cores: usize, physical_bytes: Option<usize>) -> usize {
289    let budget = physical_bytes.map_or(THREAD_BYTES, |bytes| bytes / 2);
290    (budget / THREAD_BYTES).clamp(1, cores.max(1))
291}

293fn invalid(parameter: &str, reason: impl Into<String>) -> TaskFailure {
294    TaskFailure::validation(reason).at(parameter)
295}

297/// `name`'s names: a list, or one text of names separated by commas, as
298/// CASA writes lists; blank names are none.
299fn names(task: &TaskInvocation<'_>, name: &str) -> Result<Vec<String>, TaskFailure> {
300    Ok(task
301        .texts(name)?
302        .iter()
303        .flat_map(|text| text.split(','))
304        .map(str::trim)
305        .filter(|name| !name.is_empty())
306        .map(str::to_string)
307        .collect())
308}
309
310fn lower(text: Option<String>) -> String {
311    text.unwrap_or_default().trim().to_ascii_lowercase()
312}
313
314fn padding(task: &TaskInvocation<'_>) -> Result<PaddingRows, TaskFailure> {
315    match lower(task.text("padding")?).as_str() {
316        "flagged" => Ok(PaddingRows::Flagged),
317        "omitted" => Ok(PaddingRows::Omitted),
318        other => Err(invalid(
319            "padding",
320            format!("unknown padding {other:?}; expected flagged or omitted"),
321        )),
322    }
323}
324
325fn codec(task: &TaskInvocation<'_>) -> Result<Msv4Codec, TaskFailure> {
326    match lower(task.text("codec")?).as_str() {
327        "blosc" => Ok(Msv4Codec::BloscLz4),
328        "zstd" => Ok(Msv4Codec::Zstd),
329        "none" => Ok(Msv4Codec::None),
330        other => Err(invalid(
331            "codec",
332            format!("unknown codec {other:?}; expected blosc, zstd or none"),
333        )),
334    }
335}

337/// The chunk shape: none for the default, or four non-negative extents.
338fn chunks(task: &TaskInvocation<'_>) -> Result<Option<[u64; 4]>, TaskFailure> {
339    let extents = task.integers("chunks")?;
340    if extents.is_empty() {
341        return Ok(None);
342    }
343    let shape = <[i64; 4]>::try_from(extents.as_slice()).map_err(|_| {
344        invalid(
345            "chunks",
346            format!(
347                "chunks takes four extents [time, baseline, frequency, polarization], found {}",
348                extents.len()
349            ),
350        )
351    })?;
352    let mut chunks = [0; 4];
353    for (chunk, extent) in chunks.iter_mut().zip(shape) {
354        *chunk = u64::try_from(extent).map_err(|_| {
355            invalid(
356                "chunks",
357                format!("chunk extents are 0 (a whole axis) or more, found {extent}"),
358            )
359        })?;
360    }
361    Ok(Some(chunks))
362}
363
364/// Partitions converted at once: `nthreads`, or for 0
365/// [`default_workers`] of this host's cores and physical memory.
366fn workers(task: &TaskInvocation<'_>) -> Result<usize, TaskFailure> {
367    let threads = task.integer("nthreads")?;
368    match usize::try_from(threads) {
369        Ok(0) => Ok(default_workers(
370            std::thread::available_parallelism().map_or(1, std::num::NonZero::get),
371            casa_ms::selection::system_physical_memory_bytes(),
372        )),
373        Ok(threads) => Ok(threads),
374        Err(_) => Err(invalid(
375            "nthreads",
376            format!("nthreads is 0 (every core) or more, found {threads}"),
377        )),
378    }
379}

385    fn named(vis: &str, outformat: StoreFormat) -> Option<String> {
386        default_outputvis(Path::new(vis), outformat).map(|path| path.display().to_string())

389    #[test]
390    fn the_default_output_follows_xradios_names() {
391        let (v2, v4) = (StoreFormat::Msv2, StoreFormat::Msv4);
392        assert_eq!(named("obs.ms", v4).as_deref(), Some("obs.ps.zarr"));
393        assert_eq!(
394            named("data/obs.ms", v4).as_deref(),
395            Some("data/obs.ps.zarr")
396        );
397        assert_eq!(
398            named("/data/obs.MS/", v4).as_deref(),
399            Some("/data/obs.ps.zarr")
400        );
401        assert_eq!(named("obs.ps.zarr", v2).as_deref(), Some("obs.ms"));
402        assert_eq!(named("obs.zarr", v2).as_deref(), Some("obs.ms"));
403        // A name without one gains the extension; a bare extension is a name.
404        assert_eq!(named("obs", v4).as_deref(), Some("obs.ps.zarr"));
405        assert_eq!(named("obs.v1", v2).as_deref(), Some("obs.v1.ms"));
406        assert_eq!(named(".ms", v4).as_deref(), Some(".ms.ps.zarr"));
407        // The same format names the input itself, which the task refuses.
408        assert_eq!(named("obs.ps.zarr", v4).as_deref(), Some("obs.ps.zarr"));
409        assert_eq!(named("obs.ms", v2).as_deref(), Some("obs.ms"));
410        assert_eq!(named("/", v4), None);
411    }

413    #[test]
414    fn the_default_threads_are_what_half_the_memory_holds() {
415        const GIB: usize = 1 << 30;
416        // 15 GiB and 4 cores: 7 threads' worth of memory, so every core.
417        assert_eq!(default_workers(4, Some(15 * GIB)), 4);
418        // 24 GiB and 8 cores (a Mac mini): 12 threads' worth, so 8.
419        assert_eq!(default_workers(8, Some(24 * GIB)), 8);
420        // 64 cores and 32 GiB: 16 threads.
421        assert_eq!(default_workers(64, Some(32 * GIB)), 16);
422        // Small or unknown memory still converts, one partition at a time.
423        assert_eq!(default_workers(8, Some(GIB)), 1);
424        assert_eq!(default_workers(8, None), 1);
425        assert_eq!(default_workers(0, Some(15 * GIB)), 1);
426    }

428    #[test]
429    fn a_processing_set_is_named_as_xradio_names_it() {
430        let path = |text: &str| processing_set_path(text).display().to_string();
431        assert_eq!(path("out"), "out.ps.zarr");
432        assert_eq!(path("data/out.zarr"), "data/out.zarr.ps.zarr");
433        assert_eq!(path("out.ps.zarr"), "out.ps.zarr");
434        assert_eq!(path("out.ps.zarr/"), "out.ps.zarr");
435        assert_eq!(path("outps.zarr"), "outps.zarr");
436        assert_eq!(path("."), "..ps.zarr");
437        assert_eq!(path("/"), "/.ps.zarr");
438    }
