2//! The component lists' visibilities in a write batch: after the model
3//! images are predicted into DATA, each `predict(complist=…)` covering a
4//! row's scan adds its components' visibilities to the row
5//! ([`crate::predict::ComponentPrediction::add_rows`]), on the CPU whatever
6//! the backend, a run of rows per job on the team. The sum is the same for
7//! any worker count: each row is formed by one job, in the plan's order.
8
9use casa_imaging_runtime::pass::WorkerTeam;
10use num_complex::Complex32;
11
12use super::geometry::RowGeometry;
13use crate::session::error::{SimulatorError, SimulatorResult};
14use crate::session::observe::Scan;
15use crate::session::plan::DataPhase as DataPlan;
16
17/// Rows of one job.
18pub(super) const JOB_ROWS: usize = 256;
19
20/// Bytes of one job.
21pub(super) const JOB_BYTES: usize = size_of::<Job<'static>>() + size_of::<&mut [Job<'static>]>();
22
23struct Job<'a> {
24    /// The scan's index in the session.
25    index: usize,
26    scan: &'a Scan,
27    rows: &'a [RowGeometry],
28    cells: &'a mut [Complex32],
29    beam: Option<crate::predict::BeamContext>,
30}
31
32/// Add the component lists `plan` predicts to `data`, the batch's DATA of
33/// `rows` (`samples` cells a row), for each run of `runs` (scan, first row,
34/// rows).
35pub(super) fn add_components(
36    team: &WorkerTeam,
37    (scans, rows): (&[Scan], &[RowGeometry]),
38    plan: &DataPlan,
39    runs: &[(usize, usize, usize)],
40    (data, samples): (&mut [Complex32], usize),
41) -> SimulatorResult<()> {
42    let predicted = |scan| plan.components_for_scan(scan).next().is_some();
43    // As many jobs as there are, at most a run's plus one every `JOB_ROWS`
44    // rows (the patch's admission).
45    let mut jobs = Vec::with_capacity(
46        runs.iter()
47            .filter(|&&(scan, _, _)| predicted(scan))
48            .map(|&(_, _, count)| count.div_ceil(JOB_ROWS))
49            .sum(),
50    );
51    let mut rest = data;
52    let mut position = 0;
53    for &(scan, start, count) in runs {
54        let (_, tail) = std::mem::take(&mut rest).split_at_mut((start - position) * samples);
55        rest = tail;
56        position = start;
57        let mut offset = 0;
58        while offset < count {
59            let take = JOB_ROWS.min(count - offset);
60            let (cells, tail) = std::mem::take(&mut rest).split_at_mut(take * samples);
61            rest = tail;
62            position += take;
63            if predicted(scan) {
64                jobs.push(Job {
65                    index: scan,
66                    scan: &scans[scan],
67                    rows: &rows[start + offset..start + offset + take],
68                    cells,
69                    beam: None,
70                });
71            }
72            offset += take;
73        }
74    }
75    if !plan
76        .components_for_scan(0)
77        .any(|components| components.has_squint())
78    {
79        return team.for_each_mut(&mut jobs, |worker, job| {
80            for components in plan.components_for_scan(job.index) {
81                components
82                    .add_rows(
83                        (job.index, job.scan.data_description_id),
84                        job.rows.iter().map(|row| (row.uvw_m, row.time_s)),
85                        job.cells,
86                        None,
87                        worker,
88                    )
89                    .map_err(|error| {
90                        SimulatorError::raised(format!("component prediction: {error}"))
91                    })?;
92            }
93            Ok::<_, SimulatorError>(())
94        });
95    }
96    // Every component prediction covers a prefix of scans. Visit sources in
97    // their original order, freezing each job's beam-angle history in row
98    // order before workers evaluate independent numerical cells.
99    for components in plan.components_for_scan(0) {
100        for job in &mut jobs {
101            if components.covers_scan(job.index) {
102                job.beam = components
103                    .prepare_beam_job(
104                        job.index,
105                        job.scan.field_id,
106                        job.rows.iter().map(|row| row.time_s),
107                    )
108                    .map_err(|error| SimulatorError::raised(format!("component beam: {error}")))?;
109            }
110        }
111        let chunk_rows = jobs.len().div_ceil(team.workers()).max(1);
112        let mut chunks: Vec<_> = jobs.chunks_mut(chunk_rows).collect();
113        team.for_each_mut(&mut chunks, |slot, chunk| {
114            for job in chunk.iter_mut() {
115                if !components.covers_scan(job.index) {
116                    continue;
117                }
118                components
119                    .add_rows(
120                        (job.index, job.scan.data_description_id),
121                        job.rows.iter().map(|row| (row.uvw_m, row.time_s)),
122                        job.cells,
123                        job.beam,
124                        slot,
125                    )
126                    .map_err(|error| {
127                        SimulatorError::raised(format!("component prediction: {error}"))
128                    })?;
129            }
130            Ok::<_, SimulatorError>(())
131        })?;
132    }
133    Ok(())
134}
