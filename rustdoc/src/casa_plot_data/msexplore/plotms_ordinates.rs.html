30impl<'a> VisibilityColumns<'a> {
31    pub(super) fn prepare(
32        ms: &MeasurementSet,
33        spec: &MsPlotSpec,
34        rows: &[usize],
35        input: Option<&PlotMsInput<'a>>,
36    ) -> Result<Self, String> {
37        let primary = spec.data_column;
38        let x = input
39            .and_then(|input| input.x_data_column)
40            .unwrap_or(primary);
41        let secondary = input
42            .and_then(|input| input.secondary_y_data_column)
43            .unwrap_or(primary);
44        let additional = input.map_or(&[][..], |input| input.additional_ordinates);
45        let kinds = std::iter::once(spec.x_axis).chain(spec.y_axes.iter().copied());
46        let columns = [x, primary, secondary]
47            .into_iter()
48            .chain(additional.iter().map(|&(_, column)| column));
49        let mut sources = Vec::new();
50        let mut axes = Vec::with_capacity(spec.y_axes.len() + 1);
51        for (axis, column) in kinds.zip(columns) {
52            if !axis.is_visibility_math() {
53                axes.push(None);
54                continue;
55            }
56            let index = match sources.iter().position(|(existing, _)| *existing == column) {
57                Some(index) => index,
58                None => {
59                    let index = sources.len();
60                    sources.push((
61                        column,
62                        PreparedSelectedDataSource::new(ms, column, rows, input)?,
63                    ));
64                    index
65                }
66            };
67            axes.push(Some(index));
68        }
69        Ok(Self {
70            sources,
71            axes,
72            project_after_average: spec.averaging.avgchannel.is_some_and(|width| width > 1)
73                || input.is_some_and(|input| input.extended_derived),
74        })
75    }

116        if spec.y_axes.len() <= 2 && spec.secondary_y_axis() != Some(spec.y_axis()) {
117            return (key, label, color);
118        }
119        let column = self.column(ordinate).unwrap_or(spec.data_column);

126        (
127            format!("{ordinate}::{key}"),

129            format!("{ordinate}::{color}"),

272fn channel_averaged_grid(
273    grid: &ComplexGrid,
274    flags: &ArrayView2<'_, bool>,
275    spectra: [Option<&FloatGrid>; 2],
276    corr: usize,
277    bin: &ChannelBin,
278    include_flagged: bool,
279) -> Option<(Option<VisibilityColumnProjection>, [Complex64; 2])> {
280    if let Some((projection, model)) = grid.model.as_ref() {
281        let operands = channel_averaged_operands(
282            [grid.values.as_slice(), model.as_slice()],
283            grid.chan_count,
284            flags,
285            spectra,
286            corr,
287            bin,
288            include_flagged,
289        )?;
290        Some((Some(*projection), operands))
291    } else {
292        let [value] = channel_averaged_operands(
293            [grid.values.as_slice()],
294            grid.chan_count,
295            flags,
296            [spectra[0]],
297            corr,
298            bin,
299            include_flagged,
300        )?;
301        Some((None, [value, Complex64::default()]))
302    }
303}

305fn channel_averaged_operands<const N: usize>(
306    inputs: [&[Complex64]; N],
307    channel_count: usize,
308    flags: &ArrayView2<'_, bool>,
309    spectra: [Option<&FloatGrid>; N],
310    corr: usize,
311    bin: &ChannelBin,
312    include_flagged: bool,
313) -> Option<[Complex64; N]> {
314    let mut sums = [Complex32::new(0.0, 0.0); N];
315    let mut total_weights = [0.0_f32; N];
316    let mut sample_count = 0_usize;
317    for channel in bin.channels() {
318        if channel >= channel_count || !include_flagged && flags[(corr, channel)] {
319            continue;
320        }
321        let index = corr * channel_count + channel;
322        // The channel iterator accumulates the selected flag class in
323        // Complex32, including nonfinite values.
324        for operand in 0..N {
325            let value = inputs[operand][index];
326            let weight = spectra[operand].map_or(1.0, |weights| weights.values[index] as f32);
327            sums[operand] += Complex32::new(value.re as f32, value.im as f32) * weight;
328            total_weights[operand] += weight;
329        }
330        sample_count += 1;
331    }
332    if sample_count == 0 {
333        return None;
334    }
335    Some(std::array::from_fn(|operand| {
336        let normalization = if spectra[operand].is_some() {
337            total_weights[operand]
338        } else {
339            sample_count as f32
340        };
341        channel_visibility_mean(sums[operand], normalization, spectra[operand].is_some())
342    }))
343}
344
345fn channel_visibility_mean(sum: Complex32, normalization: f32, weighted: bool) -> Complex64 {
346    let valid_weight = has_channel_weight(normalization);
347    let mut mean = if valid_weight {
348        sum / normalization
349    } else {
350        Complex32::default()
351    };
352    // CASA divides a weighted complex sum by a complex normalization.
353    // A nonfinite component affects its finite partner even for a real weight.
354    if weighted && valid_weight && normalization.is_finite() {
355        if sum.re.is_finite() && !sum.im.is_finite() {
356            mean.re = f32::NAN;
357        } else if sum.im.is_finite() && !sum.re.is_finite() {
358            mean.im = f32::NAN;
359        }
360    }
361    Complex64::new(f64::from(mean.re), f64::from(mean.im))
362}
363
364fn has_channel_weight(weight: f32) -> bool {
365    // Compare the signed real sum directly. Squaring it can erase a finite
366    // nonzero normalization in the CASA complex-weight kernel.
367    weight != 0.0 && !weight.is_nan()
368}
369
370fn vector_mean(values: &[Complex64], weights: &[f64]) -> Option<Complex64> {
371    if values.is_empty() {
372        return None;
373    }
374    if weights.is_empty() {
375        return Some(values.iter().copied().sum::<Complex64>() / values.len() as f64);
376    }
377    let total = weights.iter().sum::<f64>();
378    Some(if total > 0.0 {
379        weighted_visibility_mean(values, weights, total)
380    } else {
381        Complex64::new(0.0, 0.0)
382    })
383}

428impl TimeProjection {
429    fn channel_sample(
430        grid: &ComplexGrid,
431        spectra: [Option<&FloatGrid>; 2],
432        corr: usize,
433        channel: usize,
434    ) -> Self {
435        let index = corr * grid.chan_count + channel;
436        let (projection, model_value) = grid
437            .model
438            .as_ref()
439            .map_or((None, Complex64::default()), |(projection, model)| {
440                (Some(*projection), model[index])
441            });
442        // DATA uses inverse variance. MODEL_DATA uses calibrated weights if the
443        // iterator reads CORRECTED_DATA, and observed weights otherwise.
444        let total_weights =
445            spectra.map(|spectrum| spectrum.map_or(1.0, |grid| grid.values[index] as f32));
446        let values = [grid.values[index], model_value];
447        let weighted_sums = std::array::from_fn(|operand| {
448            let value = values[operand];
449            Complex32::new(value.re as f32, value.im as f32) * total_weights[operand]
450        });
451        Self {
452            projection,
453            weighted_sums,
454            total_weights,
455        }
456    }
457
458    fn accumulate(&mut self, sample: Self) {
459        for (operand, value) in sample.weighted_sums.into_iter().enumerate() {
460            self.weighted_sums[operand] += value;
461            self.total_weights[operand] += sample.total_weights[operand];
462        }
463    }
464
465    fn time_mean(&self, operand: usize) -> Complex32 {
466        let sum = self.weighted_sums[operand];
467        let weight = self.total_weights[operand];
468        // AveragingTvi2 keeps the accumulated value for a nonpositive weight.
469        if weight > 0.0 { sum / weight } else { sum }
470    }
471
472    fn project(&self, axis: MsAxis) -> Option<f64> {
473        self.project_after_shift(axis, None)
474    }
475
476    fn project_after_shift(&self, axis: MsAxis, phase: Option<Complex32>) -> Option<f64> {
477        let means = std::array::from_fn(|operand| {
478            let mean = self.time_mean(operand);
479            let mean = phase.map_or(mean, |phase| phase * mean);
480            Complex64::new(f64::from(mean.re), f64::from(mean.im))
481        });
482        self.project_operands(axis, means)
483    }
484
485    fn project_scalar(&self, axis: MsAxis) -> Option<f64> {
486        use VisibilityColumnProjection::{
487            ScalarRatio, ScalarResidual, VectorRatio, VectorResidual,
488        };
489
490        let visibility = self.time_mean(0);
491        let model = self.time_mean(1);
492        let projected = match self.projection {
493            None => visibility,
494            Some(VectorResidual | ScalarResidual) => visibility - model,
495            Some(VectorRatio) => {
496                let widen =
497                    |value: Complex32| Complex64::new(f64::from(value.re), f64::from(value.im));
498                let ratio =
499                    VisibilityColumnProjection::complex_quotient(widen(visibility), widen(model));
500                Complex32::new(ratio.re as f32, ratio.im as f32)
501            }
502            // CASA divides the packed amplitude and phase components for these
503            // axes. Its scalar ratio uses their differences for Real and Imaginary.
504            Some(ScalarRatio) if matches!(axis, MsAxis::Amplitude | MsAxis::Phase) => {
505                Complex32::new(visibility.re / model.re, visibility.im / model.im)
506            }
507            Some(ScalarRatio) => visibility - model,
508        };
509        Some(f64::from(match axis {
510            MsAxis::Amplitude | MsAxis::Real => projected.re,
511            MsAxis::Phase => VisibilityColumnProjection::float_phase_degrees(projected.im),
512            MsAxis::Imaginary => projected.im,
513            _ => return None,
514        }))
515    }
516
517    fn project_operands(&self, axis: MsAxis, means: [Complex64; 2]) -> Option<f64> {
518        match self.projection {
519            Some(projection) => projection.axis_value(axis, means, true),
520            None => Some(VisibilityColumnProjection::vector_axis_value(
521                axis, means[0],
522            )),
523        }
524    }
525}

555
556impl TimeChannelProjection {
557    fn project(&self, axis: MsAxis) -> Option<f64> {
558        let (operands, means, _) = self.means_and_weight()?;
559        operands.project_operands(axis, means)
560    }
561
562    fn means_and_weight(&self) -> Option<(&TimeProjection, [Complex64; 2], f32)> {
563        let mut usable = self.channels.iter().flatten();
564        let first = usable.next()?;
565        let mut sums = [Complex32::default(); 2];
566        let mut totals = [0.0_f32; 2];
567        let mut visibility_weight = 0.0_f32;
568        let mut spectrum_weight = 0.0_f32;
569        // The channel transform receives normalized Float means and the time
570        // stage's spectra, including the observed weight-to-sigma round trip.
571        for channel in std::iter::once(first).chain(usable) {
572            visibility_weight +=
573                channel_weight_after_time(channel.primary_weight, self.primary_is_observed);
574            // The output WEIGHT_SPECTRUM sums the original time weights.
575            spectrum_weight += channel.primary_weight;
576            for (operand, sum) in sums.iter_mut().enumerate() {
577                let weight = if self.equal_channel_means {
578                    1.0
579                } else {
580                    channel_weight_after_time(
581                        channel.operands.total_weights[operand],
582                        self.observed_operands[operand],
583                    )
584                };
585                *sum += channel.operands.time_mean(operand) * weight;
586                totals[operand] += weight;
587            }
588        }
589        if !has_channel_weight(visibility_weight) {
590            return None;
591        }
592        let means = std::array::from_fn(|operand| {
593            channel_visibility_mean(sums[operand], totals[operand], true)
594        });
595        Some((&first.operands, means, spectrum_weight))
596    }
597}
598
599fn channel_weight_after_time(weight: f32, observed: bool) -> f32 {
600    if observed {
601        weight_from_sigma(sigma_from_weight(weight))
602    } else {
603        weight
604    }
605}

608    fn weighted_mut(&mut self) -> (&mut Vec<Complex64>, &mut Vec<f64>) {
609        let VisibilitySamples::Weighted { values, weights } = &mut self.0 else {
610            unreachable!("time channel means are combined before weighted accumulation");

616        let (values, weights) = self.weighted_mut();
617        values.push(value);

620
621    pub(super) fn add_vector_time_samples(
622        &mut self,
623        grid: &ComplexGrid,
624        flags: &ArrayView2<'_, bool>,
625        weights: VectorTimeWeights<'_>,
626        corr: usize,
627        bin: &ChannelBin,
628        include_flagged: bool,
629    ) -> bool {
630        if bin.averaged {
631            return self.add_time_channel_samples(grid, flags, weights, corr, bin);
632        }
633        if !include_flagged && flags[(corr, bin.start)] {
634            return false;
635        }
636        self.add_time_projection(TimeProjection::channel_sample(
637            grid,
638            weights.spectra,
639            corr,
640            bin.start,
641        ));
642        true
643    }
644
645    fn add_time_projection(&mut self, sample: TimeProjection) {
646        if let VisibilitySamples::TimeProjection(average) = &mut self.0 {
647            average.accumulate(sample);
648        } else {
649            self.0 = VisibilitySamples::TimeProjection(sample);
650        }
651    }
652
653    fn add_time_channel_samples(
654        &mut self,
655        grid: &ComplexGrid,
656        flags: &ArrayView2<'_, bool>,
657        weights: VectorTimeWeights<'_>,
658        corr: usize,
659        bin: &ChannelBin,
660    ) -> bool {
661        if !matches!(self.0, VisibilitySamples::TimeChannelProjection(_)) {
662            self.0 = VisibilitySamples::TimeChannelProjection(TimeChannelProjection {
663                channels: vec![None; bin.channels().count()],
664                observed_operands: weights.observed_operands,
665                primary_is_observed: weights
666                    .primary
667                    .is_some_and(|(_, kind)| kind == WeightKind::Observed),
668                equal_channel_means: weights.equal_channel_means,
669            });
670        }
671        let VisibilitySamples::TimeChannelProjection(projection) = &mut self.0 else {
672            unreachable!("time and channel projections initialize one accumulator per channel");
673        };
674        let mut used = false;
675        for (channel, average) in bin.channels().zip(&mut projection.channels) {
676            if flags[(corr, channel)] {
677                continue;
678            }
679            let operands = TimeProjection::channel_sample(grid, weights.spectra, corr, channel);
680            let primary_weight = weights.primary.map_or(1.0, |(spectrum, _)| {
681                spectrum.values[corr * spectrum.chan_count + channel] as f32
682            });
683            if let Some(average) = average {
684                average.operands.accumulate(operands);
685                average.primary_weight += primary_weight;
686            } else {
687                *average = Some(TimeChannelSample {
688                    operands,
689                    primary_weight,
690                });
691            }
692            used = true;
693        }
694        used
695    }

697    /// The CASA scalar reducer stores amplitude and phase as separate components.
698    pub(super) fn add_scalar(
699        &mut self,
700        samples: VisibilityBinSamples<'_>,
701        weight: f64,
702        conjugate: bool,
703        preserve_float_sign: bool,
704    ) {
705        let (values, weights) = self.weighted_mut();
706        let sample_weights = samples
707            .weights
708            .iter()
709            .copied()
710            .chain(std::iter::repeat(weight));
711        for (value, weight) in samples.values.iter().zip(sample_weights) {
712            let packed = if preserve_float_sign {
713                *value
714            } else {
715                Complex64::new(value.norm(), value.arg())
716            };
717            values.push(if conjugate { packed.conj() } else { packed });
718            weights.push(super::plotms_columns::baseline_weight(weight));
719        }
720    }
721
722    pub(super) fn add_scalar_sample(
723        &mut self,
724        grid: &ComplexGrid,
725        flags: &ArrayView2<'_, bool>,
726        options: ScalarTimeSample<'_>,
727        corr: usize,
728        bin: &ChannelBin,
729    ) -> bool {
730        let mut sample = if let Some(spectra) = options.channel_spectra {
731            let Some((projection, operands)) =
732                channel_averaged_grid(grid, flags, spectra, corr, bin, false)
733            else {
734                return false;
735            };
736            TimeProjection {
737                projection,
738                weighted_sums: operands
739                    .map(|value| Complex32::new(value.re as f32, value.im as f32)),
740                total_weights: [1.0; 2],
741            }
742        } else {
743            if flags[(corr, bin.start)] {
744                return false;
745            }
746            TimeProjection::channel_sample(grid, [None; 2], corr, bin.start)
747        };
748        let weight = options.row_weight as f32;
749        // The scalar reducer packs complex amplitudes and phases before
750        // accumulating in time. Explicit FLOAT_DATA retains its real samples.
751        sample.weighted_sums = sample.weighted_sums.map(|value| {
752            let value = if options.conjugate {
753                value.conj()
754            } else {
755                value
756            };
757            if options.preserve_float_sign {
758                value * weight
759            } else {
760                Complex32::new(value.norm(), value.arg()) * weight
761            }
762        });
763        sample.total_weights = [weight; 2];
764        self.add_time_projection(sample);
765        true
766    }

768    pub(super) fn project_scalar(&self, axis: MsAxis) -> Option<f64> {
769        if let VisibilitySamples::TimeProjection(average) = &self.0 {
770            return average.project_scalar(axis);
771        }
772        let mean = self.vector_mean()?;
773        match axis {
774            MsAxis::Amplitude | MsAxis::Real => Some(mean.re),
775            MsAxis::Phase => Some(mean.im.to_degrees()),
776            MsAxis::Imaginary => Some(mean.im),
777            _ => None,
778        }
779    }

781    pub(super) fn add_channel_time_row(
782        &mut self,
783        grid: &ComplexGrid,
784        flags: &ArrayView2<'_, bool>,
785        correlation: usize,
786        bin: &ChannelBin,
787    ) {
788        if matches!(self.0, VisibilitySamples::Weighted { .. }) {
789            self.0 = VisibilitySamples::ChannelMeans(vec![
790                ChannelVisibilityMean::default();
791                bin.channels().count()
792            ]);
793        }
794        let VisibilitySamples::ChannelMeans(means) = &mut self.0 else {
795            unreachable!("model-only channel averaging prepares per-channel time means");
796        };
797        for (channel, mean) in bin.channels().zip(means) {
798            let value = grid.values[correlation * grid.chan_count + channel];
799            if !flags[(correlation, channel)] && value.re.is_finite() && value.im.is_finite() {
800                mean.sum += value;
801                mean.samples += 1;
802            }
803        }
804    }
805
806    pub(super) fn vector_mean(&self) -> Option<Complex64> {
807        match &self.0 {
808            VisibilitySamples::Weighted { values, weights } => vector_mean(values, weights),
809            VisibilitySamples::ChannelMeans(channels) => {
810                let mut sum = Complex64::default();
811                let mut usable = 0;
812                for channel in channels.iter().filter(|channel| channel.samples > 0) {
813                    sum += channel.sum / channel.samples as f64;
814                    usable += 1;
815                }
816                (usable > 0).then(|| sum / usable as f64)
817            }
818            VisibilitySamples::TimeProjection(_) | VisibilitySamples::TimeChannelProjection(_) => {
819                unreachable!("time operands project together")
820            }
821        }
822    }

824    pub(super) fn transformed_vector_sample(&self) -> Option<TransformedVectorSample> {
825        match &self.0 {
826            VisibilitySamples::TimeProjection(average) => Some(TransformedVectorSample {
827                visibility: average.time_mean(0),
828                model: average
829                    .projection
830                    .map(|projection| (projection, average.time_mean(1))),
831                channel_weight: None,
832            }),
833            VisibilitySamples::TimeChannelProjection(average) => {
834                let (sample, [visibility, model], weight) = average.means_and_weight()?;
835                let narrow = |value: Complex64| Complex32::new(value.re as f32, value.im as f32);
836                Some(TransformedVectorSample {
837                    visibility: narrow(visibility),
838                    model: sample
839                        .projection
840                        .map(|projection| (projection, narrow(model))),
841                    channel_weight: Some(weight),
842                })
843            }
844            _ => None,
845        }
846    }

848    pub(super) fn add_normalized_vector_sample(
849        &mut self,
850        sample: TransformedVectorSample,
851        weight: f32,
852        conjugate: bool,
853    ) {
854        let weighted = |value: Complex32| {
855            let value = if conjugate { value.conj() } else { value };
856            // Even the first endpoint adds its contribution to positive zero.
857            Complex32::default() + value * weight
858        };
859        let (projection, model, model_weight) = sample
860            .model
861            .map_or((None, Complex32::default(), 0.0), |(projection, model)| {
862                (Some(projection), weighted(model), weight)
863            });
864        self.add_time_projection(TimeProjection {
865            projection,
866            weighted_sums: [weighted(sample.visibility), model],
867            total_weights: [weight, model_weight],
868        });
869    }
870
871    pub(super) fn add(
872        &mut self,
873        samples: VisibilityBinSamples<'_>,
874        scalar_weight: f64,
875        conjugate: bool,
876        clamp_weight: bool,
877    ) {
878        let (values, weights) = self.weighted_mut();
879        if conjugate {
880            values.extend(samples.values.iter().map(|v| v.conj()));
881        } else {
882            values.extend_from_slice(samples.values);
883        }
884        if samples.weights.is_empty() {
885            let weight = if clamp_weight {
886                super::plotms_columns::baseline_weight(scalar_weight)
887            } else {
888                scalar_weight
889            };
890            weights.resize(values.len(), weight);
891        } else if clamp_weight {
892            weights.extend(
893                samples
894                    .weights
895                    .iter()
896                    .copied()
897                    .map(super::plotms_columns::baseline_weight),
898            );
899        } else {
900            weights.extend_from_slice(samples.weights);
901        }
902    }

904    pub(super) fn project(
905        &self,
906        axis: MsAxis,
907        scalar: bool,
908        zero_invalid_weight: bool,
909    ) -> Option<f64> {
910        match &self.0 {
911            VisibilitySamples::Weighted { values, weights } => {
912                compute_weighted_visibility_math(axis, values, weights, scalar, zero_invalid_weight)
913            }
914            VisibilitySamples::ChannelMeans(_) => {
915                let mean = self.vector_mean()?;
916                compute_visibility_math(axis, &[mean], scalar)
917            }
918            VisibilitySamples::TimeProjection(average) => average.project(axis),
919            VisibilitySamples::TimeChannelProjection(projection) => projection.project(axis),
920        }
921    }
922
923    pub(super) fn project_shifted_time(&self, axis: MsAxis, phase: Complex32) -> Option<f64> {
924        let VisibilitySamples::TimeProjection(average) = &self.0 else {
925            return None;
926        };
927        average.project_after_shift(axis, Some(phase))
928    }

970pub(super) struct ScalarSpanPlan {
971    row_groups: Vec<Option<usize>>,
972    metadata: Vec<ScalarSpanMetadata>,
973}
974
975impl ScalarSpanPlan {
976    pub(super) fn new(
977        columns: ScalarSpanColumns<'_>,
978        descriptions: &MsDataDescription<'_>,
979        averaging: &MsAverageSpec,
980    ) -> Result<Self, String> {
981        let interval = averaging.avgtime.unwrap_or(0.0);
982        let mut row_groups = Vec::with_capacity(columns.times.len());
983        let mut metadata = Vec::<ScalarSpanMetadata>::new();
984        let mut groups = BaselineIntegrationGroups::default();
985        let mut windows = vec![None; descriptions.row_count()];
986        for (slot, &time) in columns.times.iter().enumerate() {
987            let Some(window) = windows.get_mut(columns.descriptions[slot] as usize) else {
988                row_groups.push(None);
989                continue;
990            };
991            let spw = match *window {
992                Some(spw) => spw,
993                None => {
994                    let spw = descriptions
995                        .spectral_window_id(columns.descriptions[slot] as usize)
996                        .map_err(|error| error.to_string())?;
997                    *window = Some(spw);
998                    spw
999                }
1000            };
1001            let scan = columns.scans[slot];
1002            let field = columns.fields[slot];
1003            let group = groups.observe(
1004                [
1005                    columns.observations[slot],
1006                    if averaging.avgscan { 0 } else { scan },
1007                    if averaging.avgfield { 0 } else { field },
1008                    if averaging.avgspw { 0 } else { spw },
1009                ],
1010                time,
1011                interval,
1012            );
1013            if group == metadata.len() {
1014                metadata.push(ScalarSpanMetadata { scan, field });
1015            } else if metadata[group].scan != scan {
1016                // PlotMS marks a scalar buffer spanning scans with scan -1.
1017                metadata[group].scan = -1;
1018            }
1019            row_groups.push(Some(group));
1020        }
1021        Ok(Self {
1022            row_groups,
1023            metadata,
1024        })
1025    }
1026
1027    pub(super) fn group(&self, row_slot: usize) -> Option<usize> {
1028        self.row_groups[row_slot]
1029    }
1030
1031    pub(super) fn metadata(&self, group: usize) -> &ScalarSpanMetadata {
1032        &self.metadata[group]
1033    }

1041impl VisibilityAverageShapes {
1042    pub(super) fn check_shape(
1043        &mut self,
1044        average: &str,
1045        group: usize,
1046        correlations: usize,
1047        channels: usize,
1048    ) -> Result<(), String> {
1049        // Samples in one reduction group are consecutive in iterator order.

1051            && previous_group == group
1052            && (expected_correlations, expected_channels) != (correlations, channels)

1056            ));
1057        }
1058        self.previous = Some((group, correlations, channels));
1059        Ok(())
1060    }

1063pub(super) fn has_unflagged_input(
1064    flags: &ArrayView2<'_, bool>,
1065    correlations: &[(usize, String)],
1066    selection: Option<&[ChannelRange]>,
1067) -> bool {

1070        match selection {
1071            Some(ranges) => ranges
1072                .iter()
1073                .filter_map(ChannelRange::channels)
1074                .flatten()

1078        }

1082impl VisibilityTimeBounds {
1083    pub(super) fn new(integration: Integration) -> Self {
1084        Self {
1085            first: integration,
1086            last: integration,
1087            last_unflagged: None,
1088        }
1089    }
1090
1091    pub(super) fn observe(&mut self, integration: Integration, unflagged: bool) {
1092        if integration.time < self.first.time {
1093            self.first = integration;
1094        }
1095        if integration.time >= self.last.time {
1096            self.last = integration;
1097        }
1098        if unflagged
1099            && self
1100                .last_unflagged
1101                .is_none_or(|last| integration.time >= last.time)
1102        {
1103            self.last_unflagged = Some(integration);
1104        }
1105    }
1106
1107    pub(super) fn midpoint(&self) -> Option<f64> {
1108        // CASA anchors on the first integration and ends at the last usable one.
1109        self.last_unflagged
1110            .map(|last| self.first.time + (last.time - self.first.time) / 2.0)
1111    }
1112
1113    pub(super) fn buffer_integration(&self) -> Integration {
1114        // A fully flagged first baseline still supplies the cache's time and interval.
1115        let last = self.last_unflagged.unwrap_or(self.last);
1116        let (time, interval) = centered_time_and_interval(
1117            (self.first.time, self.first.interval),
1118            (last.time, last.interval),
1119        );
1120        Integration { time, interval }
1121    }

1166            }
1167        }

1172/// Keep one sample per point in the pair's collapsed visibility shape.
1173#[derive(Default)]

1175    row: HashSet<(usize, usize)>,
1176    buffer: HashSet<(usize, usize)>,
1177    first_buffer_row: usize,
1178}
1179
1180impl CollapsedPointMask {
1181    pub(super) fn begin_row(&mut self, row: usize, starts_buffer: bool) {
1182        self.row.clear();
1183        if starts_buffer {
1184            self.buffer.clear();
1185            self.first_buffer_row = row;
1186        }
1187    }
1188
1189    pub(super) fn project_sample(
1190        &mut self,
1191        ordinate: usize,
1192        row: usize,
1193        channel: usize,
1194        dimensions: PointDimensions,
1195        [channels, correlations]: [usize; 2],
1196    ) -> Option<PointIndex> {
1197        let channel_slot = if dimensions.channel { channel } else { 0 };
1198        let (seen, row) = if dimensions.baseline {
1199            (&mut self.row, row)
1200        } else {
1201            (&mut self.buffer, self.first_buffer_row)
1202        };
1203        if !seen.insert((ordinate, channel_slot)) {
1204            return None;
1205        }
1206        // PlotMS uses the collapsed point's linear index for the reported
1207        // identifiers, including projected dimensions of the full cube.
1208        let channels_per_point = if dimensions.channel { channels } else { 1 };
1209        let index = (row - self.first_buffer_row) * channels_per_point + channel_slot;
1210        Some(PointIndex {
1211            row,
1212            channel: index % channels,
1213            correlation: index % correlations,
1214        })
1215    }

1218pub(super) fn order_series(series: &mut MsScatterSeries) {
1219    let point_order = |index: usize| {
1220        let iterator_slot = series
1221            .plotms_metadata
1222            .get(index)
1223            .and_then(|metadata| metadata.averaging_group);
1224        series.provenance[index].plotms_order(iterator_slot)
1225    };
1226    let indices = 0..series.provenance.len();
1227    if indices.clone().is_sorted_by_key(point_order) {
1228        return;
1229    }
1230    let mut order = indices.collect::<Vec<_>>();
1231    order.sort_unstable_by_key(|&index| point_order(index));
1232    series.points = order.iter().map(|&index| series.points[index]).collect();
1233    series.provenance = order
1234        .iter()
1235        .map(|&index| series.provenance[index].clone())
1236        .collect();
1237    if !series.plotms_metadata.is_empty() {
1238        series.plotms_metadata = order
1239            .iter()
1240            .map(|&index| series.plotms_metadata[index])
1241            .collect();
1242    }
1243    if !series.plotms_reporting_frequencies_hz.is_empty() {
1244        series.plotms_reporting_frequencies_hz = order
1245            .iter()
1246            .map(|&index| series.plotms_reporting_frequencies_hz[index])
1247            .collect();
1248    }
