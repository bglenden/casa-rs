2//! Model-selection references passed from PlotMS callers to calibration prediction.
3
4use std::collections::BTreeMap;
5
6use casa_ms::selection::ChannelRange;
7
8/// Iterator references for averaging virtual model cells. Both slot
9/// slices are indexed by input selection slot and contain slots in that same
10/// selection.
11#[derive(Debug, Clone, Copy)]
12pub struct PlotMsModelReferences<'a> {
13    /// The first row of the iterator chunk supplies FIELD direction and epoch.
14    pub phase_slots: &'a [usize],
15    /// The first row of each input visibility buffer supplies the model recipe.
16    pub recipe_slots: &'a [usize],
17    /// Selected cells for aligning predictions with the first input buffer
18    /// window. Only supplied for spectral-window averaging.
19    pub spectral_windows: Option<PlotMsModelSpwSelection<'a>>,
20}
21
22/// Physical selection to align with the prediction buffer by cell position.
23#[derive(Debug, Clone, Copy)]
24pub struct PlotMsModelSpwSelection<'a> {
25    /// Selected channel ranges in each physical spectral window.
26    pub channels: &'a BTreeMap<i32, Vec<ChannelRange>>,
27    /// Requested correlation types, or empty to select all physical slots.
28    pub correlations: &'a [i32],
29}
