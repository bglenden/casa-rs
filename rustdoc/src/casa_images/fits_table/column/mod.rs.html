28impl Column {
29    pub fn read(fits: &mut Reader, header: &Header) -> Result<Vec<Self>> {
30        let (mut ncols, mut status) = (0, 0);
31        // SAFETY: Ptr has exclusive use and ncols/status are writable.
32        unsafe { sys::ffgncl(fits.ptr, &mut ncols, &mut status) };
33        fits_status(status)?;
34        let mut cols = (1..=ncols)
35            .map(|number| Self::describe(fits, header, number))
36            .collect::<Result<Vec<_>>>()?;
37        let row_shapes = cols
38            .iter()
39            .enumerate()
40            .filter_map(|(i, col)| {
41                if col.primitive != PrimitiveType::String {
42                    return None;
43                }
44                let refers_to = col
45                    .name
46                    .strip_prefix("TDIM")?
47                    .parse::<usize>()
48                    .ok()?
49                    .checked_sub(1)?;
50                (refers_to < cols.len()).then_some((refers_to, i))
51            })
52            .collect::<Vec<_>>();
53        for (target, shaping) in row_shapes {
54            cols[target].row_shape = Some(shaping);
55        }
56        Ok(cols)
57    }
58
59    fn describe(fits: &mut Reader, header: &Header, number: i32) -> Result<Self> {
60        Self::describe_for(fits, header, number, false)
61    }

63    fn describe_for(
64        fits: &mut Reader,
65        header: &Header,
66        number: i32,
67        listing: bool,
68    ) -> Result<Self> {
69        let (mut code, mut repeat, mut width, mut status) = (0, 0, 0, 0);
70        // SAFETY: Ptr has exclusive use and all outputs are writable.
71        unsafe {
72            sys::ffgtclll(
73                fits.ptr,
74                number,
75                &mut code,
76                &mut repeat,
77                &mut width,
78                &mut status,
79            )
80        };
81        fits_status(status)?;
82        let heap = code < 0;
83        // CFITSIO may narrow an ASCII I field by its width. FITSTabular
84        // exposes every ASCII I as Int, F/E as Float, and D as Double.
85        let code = if fits.ascii_table {
86            match header
87                .text(&format!("TFORM{number}"))
88                .and_then(|form| form.as_bytes().first())
89            {
90                Some(b'I') => sys::TINT,
91                Some(b'F' | b'E') => sys::TFLOAT,
92                Some(b'D') => sys::TDOUBLE,
93                _ => code.unsigned_abs(),
94            }
95        } else {
96            code.unsigned_abs()
97        };
98        let name = header
99            .text(&format!("TTYPE{number}"))
100            .unwrap_or("")
101            .to_owned();
102        let repeat = usize::try_from(if code == sys::TSTRING && !(listing && heap) {
103            width.max(repeat)
104        } else {
105            repeat
106        })
107        .map_err(|_| invalid_column(&name, "invalid column length"))?;
108        use PrimitiveType::*;
109        let mut primitive = match code {
110            sys::TBIT | sys::TLOGICAL => Bool,
111            sys::TBYTE => UInt8,
112            sys::TSHORT => Int16,
113            sys::TLONG | sys::TINT => Int32,
114            sys::TLONGLONG if listing => Int64,
115            sys::TFLOAT => Float32,
116            sys::TDOUBLE => Float64,
117            sys::TCOMPLEX => Complex32,
118            sys::TDBLCOMPLEX => Complex64,
119            sys::TSTRING => String,
120            _ => {
121                return Err(invalid_column(
122                    &name,
123                    &format!("unsupported FITS field type {code}"),
124                ));
125            }
126        };
127        let promoted = !listing
128            && matches!(primitive, UInt8 | Int16 | Int32)
129            && (header.number(&format!("TSCAL{number}")).unwrap_or(1.0) != 1.0
130                || header.number(&format!("TZERO{number}")).unwrap_or(0.0) != 0.0);
131        if promoted {
132            primitive = Float64;
133        }
134        let shape = if listing || heap || primitive == String {
135            None
136        } else if let Some(spec) = header.text(&format!("TDIM{number}")) {
137            let shape = parse_shape(spec)?;
138            if checked_product(&shape)? != repeat {
139                return Err(invalid_column(
140                    &name,
141                    "TDIM product differs from the column length",
142                ));
143            }
144            (shape != [1]).then_some(shape)
145        } else if repeat != 1 {
146            Some(vec![repeat])
147        } else {
148            None
149        };
150        Ok(Self {
151            name,
152            number,
153            code,
154            repeat,
155            primitive,
156            promoted,
157            heap,
158            shape,
159            row_shape: None,
160        })
161    }

163    fn variable(&self) -> bool {
164        self.row_shape.is_some() || (self.heap && self.primitive != PrimitiveType::String)
165    }
166
167    pub fn schema(&self) -> Result<ColumnSchema> {
168        if self.variable() {
169            Ok(ColumnSchema::array_variable(
170                &self.name,
171                self.primitive,
172                None,
173            ))
174        } else if let Some(shape) = &self.shape {
175            Ok(
176                ColumnSchema::array_fixed(&self.name, self.primitive, shape.clone()).with_options(
177                    ColumnOptions {
178                        direct: true,
179                        undefined: false,
180                    },
181                )?,
182            )
183        } else {
184            Ok(ColumnSchema::scalar(&self.name, self.primitive))
185        }
186    }

188    pub fn keywords(&self, header: &Header) -> RecordValue {
189        let mut keys = RecordValue::default();
190        for (suffix, key) in [("TUNIT", "UNIT"), ("TDISP", "DISP")] {
191            if let Some(value) = header
192                .text(&format!("{suffix}{}", self.number))
193                .filter(|value| !value.is_empty())
194            {
195                keys.upsert(key, Value::Scalar(ScalarValue::String(value.to_owned())));
196            }
197        }
198        if !self.promoted
199            && !self.heap
200            && matches!(self.code, sys::TBYTE | sys::TSHORT | sys::TINT | sys::TLONG)
201            && let Some(value @ ScalarValue::Int32(_)) =
202                header.value(&format!("TNULL{}", self.number))
203        {
204            keys.upsert("NULL", Value::Scalar(value.clone()));
205        }
206        keys
207    }

209    pub fn value(&self, fits: &mut Reader, cols: &[Self], row: i64) -> Result<Value> {
210        if self.primitive == PrimitiveType::String {
211            return Ok(Value::Scalar(ScalarValue::String(self.text(fits, row)?)));
212        }
213        let available = self.row_length(fits, row)?;
214        let row_shape = self
215            .row_shape
216            .map(|shape| cols[shape].text(fits, row))
217            .transpose()?
218            .map(|shape| parse_shape(&shape))
219            .transpose()?;
220        let heap_shape =
221            (self.heap && self.primitive != PrimitiveType::String).then_some([available]);
222        let shape = row_shape
223            .as_deref()
224            .or(self.shape.as_deref())
225            .or(heap_shape.as_ref().map(|s| s.as_slice()));
226        let len = shape.map_or(Ok(1), checked_product)?;
227        if len > available && self.primitive != PrimitiveType::String {
228            return Err(invalid_column(
229                &self.name,
230                "row TDIM exceeds the available FITS values",
231            ));
232        }
233        macro_rules! value {
234            ($variant:ident, $vec:expr) => {{
235                let mut values = $vec;
236                if let Some(shape) = shape {
237                    let array = ArrayD::from_shape_vec(IxDyn(shape).f(), values)
238                        .map_err(|error| invalid_column(&self.name, &error.to_string()))?;
239                    Value::Array(ArrayValue::$variant(array))
240                } else {
241                    Value::Scalar(ScalarValue::$variant(
242                        values
243                            .pop()
244                            .ok_or_else(|| invalid_column(&self.name, "missing scalar"))?,
245                    ))
246                }
247            }};
248        }
249        Ok(match self.primitive {
250            PrimitiveType::Bool => value!(Bool, self.logical(fits, row, len)?),
251            PrimitiveType::UInt8 => value!(UInt8, self.numbers::<u8>(fits, row, len, sys::TBYTE)?),
252            PrimitiveType::Int16 => {
253                value!(Int16, self.numbers::<i16>(fits, row, len, sys::TSHORT)?)
254            }
255            PrimitiveType::Int32 => value!(Int32, self.numbers::<i32>(fits, row, len, sys::TINT)?),
256            PrimitiveType::Float32 => {
257                value!(Float32, self.numbers::<f32>(fits, row, len, sys::TFLOAT)?)
258            }
259            PrimitiveType::Float64 => {
260                value!(Float64, self.numbers::<f64>(fits, row, len, sys::TDOUBLE)?)
261            }
262            PrimitiveType::Complex32 => value!(
263                Complex32,
264                self.numbers::<[f32; 2]>(fits, row, len, sys::TCOMPLEX)?
265                    .into_iter()
266                    .map(|[re, im]| Complex32::new(re, im))
267                    .collect::<Vec<_>>()
268            ),
269            PrimitiveType::Complex64 => value!(
270                Complex64,
271                self.numbers::<[f64; 2]>(fits, row, len, sys::TDBLCOMPLEX)?
272                    .into_iter()
273                    .map(|[re, im]| Complex64::new(re, im))
274                    .collect::<Vec<_>>()
275            ),
276            _ => unreachable!("FITS description constrains the primitive type"),
277        })
278    }
279
280    fn row_length(&self, fits: &mut Reader, row: i64) -> Result<usize> {
281        if !self.heap {
282            return Ok(self.repeat);
283        }
284        self.descriptor(fits, row).map(|(len, _)| len)
285    }
286
287    fn descriptor(&self, fits: &mut Reader, row: i64) -> Result<(usize, i64)> {
288        let (mut len, mut offset, mut status) = (0, 0, 0);
289        // SAFETY: The cursor is on this binary table and the outputs are writable.
290        unsafe {
291            sys::ffgdesll(
292                fits.ptr,
293                self.number,
294                row,
295                &mut len,
296                &mut offset,
297                &mut status,
298            )
299        };
300        fits_status(status)?;
301        usize::try_from(len)
302            .map(|len| (len, offset))
303            .map_err(|_| invalid_column(&self.name, "invalid heap descriptor length"))
304    }

306    fn numbers<T: Default + Clone>(
307        &self,
308        fits: &mut Reader,
309        row: i64,
310        len: usize,
311        read_as: u32,
312    ) -> Result<Vec<T>> {
313        let (mut any_null, mut status) = (0, 0);
314        let mut buffer = vec![T::default(); len];
315        if len > 0 {
316            // SAFETY: The call sites supply a C-compatible T with exactly the
317            // layout specified by read_as (a pair of floats for complex).
318            // The buffer is writable for all len elements; NULL nulval keeps
319            // FITS null sentinels instead of substituting an arbitrary value.
320            unsafe {
321                sys::ffgcv(
322                    fits.ptr,
323                    read_as as i32,
324                    self.number,
325                    row,
326                    1,
327                    len as i64,
328                    ptr::null_mut(),
329                    buffer.as_mut_ptr().cast(),
330                    &mut any_null,
331                    &mut status,
332                );
333            }
334            fits_status(status)?;
335        }
336        Ok(buffer)
337    }

339    fn logical(&self, fits: &mut Reader, row: i64, len: usize) -> Result<Vec<bool>> {
340        Ok(self
341            .logical_bytes(fits, row, len)?
342            .into_iter()
343            .map(|value| value == 1)
344            .collect())
345    }
346
347    fn logical_bytes(&self, fits: &mut Reader, row: i64, len: usize) -> Result<Vec<c_char>> {
348        let mut data: Vec<c_char> = vec![0; len];
349        let mut status = 0;
350        if len > 0 {
351            // SAFETY: Both CFITSIO calls write len C chars; the live buffer
352            // and current table cursor stay exclusively borrowed.
353            unsafe {
354                if self.code == sys::TBIT {
355                    sys::ffgcx(
356                        fits.ptr,
357                        self.number,
358                        row,
359                        1,
360                        len as i64,
361                        data.as_mut_ptr(),
362                        &mut status,
363                    );
364                } else {
365                    sys::ffgcl(
366                        fits.ptr,
367                        self.number,
368                        row,
369                        1,
370                        len as i64,
371                        data.as_mut_ptr(),
372                        &mut status,
373                    );
374                }
375            }
376            fits_status(status)?;
377        }
378        Ok(data)
379    }

381    fn text(&self, fits: &mut Reader, row: i64) -> Result<String> {
382        let width = self.row_length(fits, row)?;
383        if width == 0 {
384            return Ok(String::new());
385        }
386        let len = width
387            .checked_add(1)
388            .ok_or_else(|| invalid_column(&self.name, "text width out of bounds"))?;
389        let mut buffer = vec![0; len];
390        let (mut any_null, mut status) = (0, 0);
391        let mut strings = [buffer.as_mut_ptr()];
392        // SAFETY: Single writable string with space for the entire ASCII field
393        // and a NUL terminator. CFITSIO's nulval is unset.
394        unsafe {
395            sys::ffgcvs(
396                fits.ptr,
397                self.number,
398                row,
399                1,
400                1,
401                ptr::null_mut(),
402                strings.as_mut_ptr(),
403                &mut any_null,
404                &mut status,
405            )
406        };
407        fits_status(status)?;
408        // SAFETY: CFITSIO terminates the string in buffer on success.
409        let s = unsafe { CStr::from_ptr(buffer.as_ptr()) }.to_string_lossy();
410        Ok(s.trim_end_matches(' ').to_owned())
411    }

414fn invalid_column(name: &str, note: &str) -> FitsTableError {
415    FitsTableError::Fits(format!("FITS column {name}: {note}"))
416}
417
418fn parse_shape(text: &str) -> Result<Vec<usize>> {
419    let bad = || FitsTableError::Fits(format!("invalid FITS TDIM '{text}'"));
420    text.trim()
421        .strip_prefix('(')
422        .and_then(|s| s.strip_suffix(')'))
423        .ok_or_else(bad)?
424        .split(',')
425        .map(|dim| dim.trim().parse().map_err(|_| bad()))
426        .collect()
427}
428
429fn checked_product(shape: &[usize]) -> Result<usize> {
430    shape
431        .iter()
432        .try_fold(1_usize, |size, axis| size.checked_mul(*axis))
433        .ok_or_else(|| FitsTableError::Fits("FITS TDIM product is too large".into()))
434}
