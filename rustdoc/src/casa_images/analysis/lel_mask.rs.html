22enum MaskSource<T: ImagePixel> {
23    Paged(Box<PagedImage<T>>),
24    Memory(Box<TempImage<T>>),
25}
26
27impl<T: ImagePixel> MaskSource<T> {
28    fn as_interface(&self) -> &dyn ImageInterface<T> {
29        match self {
30            Self::Paged(img) => img.as_ref(),
31            Self::Memory(img) => img.as_ref(),
32        }
33    }
34}
35
36enum Operand {
37    Real(MaskSource<f64>, bool),
38    Complex(MaskSource<Complex64>),
39}
40impl Operand {
41    fn geometry(&self) -> (&[usize], &CoordinateSystem) {
42        match self {
43            Self::Real(source, _) => (
44                source.as_interface().shape(),
45                source.as_interface().coordinates(),
46            ),
47            Self::Complex(source) => (
48                source.as_interface().shape(),
49                source.as_interface().coordinates(),
50            ),
51        }
52    }
53}

55fn is_single_precision(opened: &[(String, Operand)], name: &str) -> bool {
56    opened
57        .iter()
58        .any(|(source, value)| source == name && matches!(value, Operand::Real(_, true)))
59}

61pub(super) fn evaluate_replacement_pixels(
62    expression: &str,
63    selected_shape: &[usize],
64    want_complex: bool,
65) -> Result<ReplacementPixels, ImageError> {
66    if expression.trim() == "[]" {
67        return Err(ImageError::CasaSelection(
68            "Image Expression: Parse error at or near ']'".into(),
69        ));
70    }
71    let opened = image_operands(expression)?;
72    let (real, complex) = operand_resolvers(&opened);
73    evaluate_replacement(
74        expression,
75        &real,
76        &complex,
77        &|name| is_single_precision(&opened, name),
78        selected_shape,
79        want_complex,
80    )
81}

83fn image_operands(expression: &str) -> Result<Vec<(String, Operand)>, ImageError> {
84    referenced_images(expression)?
85        .into_iter()
86        .map(|name| {
87            let image = open_operand(&name)?;
88            Ok::<_, ImageError>((name, image))
89        })
90        .collect()
91}
92
93fn operand_resolvers(
94    opened: &[(String, Operand)],
95) -> (HashMapResolver<'_, f64>, HashMapResolver<'_, Complex64>) {
96    let mut real = HashMap::new();
97    let mut complex = HashMap::new();
98    for (name, operand) in opened {
99        match operand {
100            Operand::Real(source, _) => {
101                real.insert(name.clone(), source.as_interface());
102            }
103            Operand::Complex(source) => {
104                complex.insert(name.clone(), source.as_interface());
105            }
106        }
107    }
108    (HashMapResolver(real), HashMapResolver(complex))
109}

111/// Evaluates an optional `mask` expression over `window` `(start, shape)` of a
112/// parent image of shape `target_shape`.
113pub(super) fn evaluate_mask(
114    mask_expr: Option<&str>,
115    target_shape: &[usize],
116    target_coordinates: &CoordinateSystem,
117    stretch: bool,
118    window: (&[usize], &[usize]),
119) -> Result<Option<ArrayD<bool>>, ImageError> {
120    let Some(raw) = mask_expr.map(str::trim).filter(|s| !s.is_empty()) else {
121        return Ok(None);
122    };
123
124    let mut expr = raw.to_string();
125    if Path::new(raw).exists() {
126        // Quote bare file names that include spaces or punctuation.
127        expr = format!("\"{raw}\"");
128    } else if let Some((left, _)) = raw.split_once(':')
129        && !raw.contains([' ', '>', '<', '=', '!', '&', '|', '(', ')'])
130        && Path::new(left).exists()
131    {
132        // Unquoted path:maskname expression in top-level `mask=` parameter.
133        expr = format!("mask(\"{raw}\")");
134    } else if let Some(operator) = raw.find(['<', '>', '=', '!'])
135        && let (name, suffix) = raw.split_at(operator)
136        && Path::new(name.trim()).exists()
137    {
138        // These tasks accept an unquoted image path before a comparison,
139        // including an absolute path that would otherwise lex as division.
140        expr = format!("\"{}\"{suffix}", name.trim());
141    }
142    let opened = referenced_images(&expr)
143        .map_err(mask_parse_error)?
144        .into_iter()
145        .map(|name| {
146            let image = open_operand(&name)?;
147            Ok((name, image))
148        })
149        .collect::<Result<Vec<_>, ImageError>>()?;
150    evaluate(
151        &expr,
152        target_shape,
153        target_coordinates,
154        stretch,
155        window,
156        &opened,
157    )
158}

160fn evaluate(
161    expr: &str,
162    target_shape: &[usize],
163    target_coordinates: &CoordinateSystem,
164    stretch: bool,
165    window: (&[usize], &[usize]),
166    opened: &[(String, Operand)],
167) -> Result<Option<ArrayD<bool>>, ImageError> {
168    let (real, complex) = operand_resolvers(opened);
169    let parsed = parse_subimage_mask_expr(expr, &real, &complex).map_err(mask_parse_error)?;
170    let mask_shape = parsed.shape();
171    if mask_shape != target_shape && !mask_shape.is_empty() {
172        if !stretch {
173            return Err(ImageError::CasaSelection(format!(
174                "The input image shape ({target_shape:?}) and mask shape ({mask_shape:?}) \
175                 are different, and it was specified that the mask should not be extended, \
176                 so the mask cannot be applied to the (sub)image"
177            )));
178        }
179        let source = opened
180            .iter()
181            .map(|(_, image)| image.geometry())
182            .find(|(shape, _)| *shape == mask_shape);
183        if let Some((_, coordinates)) = source {
184            let axes = stretch_axes(coordinates, target_coordinates, mask_shape, target_shape)?;
185            if axes.iter().copied().ne(0..axes.len()) {
186                let mut start = Vec::with_capacity(axes.len());
187                let mut shape = Vec::with_capacity(axes.len());
188                for (i, &axis) in axes.iter().enumerate() {
189                    let single = mask_shape[i] == 1;
190                    start.push(if single { 0 } else { window.0[axis] });
191                    shape.push(if single { 1 } else { window.1[axis] });
192                }
193                let mut data = parsed.get_slice(&start, &shape)?;
194                for axis in 0..target_shape.len() {
195                    if !axes.contains(&axis) {
196                        data = data.insert_axis(ndarray::Axis(axis));
197                    }
198                }
199                return broadcast_array(data, window.1).map(Some);
200            }
201        }
202    }
203    parsed
204        .eval_window_for_shape(window.0, window.1, target_shape)
205        .map(Some)
206}

208fn stretch_axes(
209    source: &CoordinateSystem,
210    target: &CoordinateSystem,
211    source_shape: &[usize],
212    target_shape: &[usize],
213) -> Result<Vec<usize>, ImageError> {
214    let axes = |coords: &CoordinateSystem| {
215        (0..coords.n_coordinates())
216            .flat_map(|i| {
217                let coord = coords.coordinate(i);
218                let kind = coord.coordinate_type();
219                coord.axis_names().into_iter().map(move |name| (kind, name))
220            })
221            .collect::<Vec<_>>()
222    };
223    let source_axes = axes(source);
224    let target_axes = axes(target);
225    let incompatible = || {
226        ImageError::CasaSelection(
227            "Unable to extend mask: coordinate axes or axis lengths do not conform".into(),
228        )
229    };
230    let mapped = source_axes
231        .iter()
232        .map(|axis| {
233            target_axes
234                .iter()
235                .position(|candidate| candidate == axis)
236                .ok_or_else(incompatible)
237        })
238        .collect::<Result<Vec<_>, _>>()?;
239    if mapped.len() != source_shape.len()
240        || target_axes.len() != target_shape.len()
241        || !mapped.is_sorted_by(|a, b| a < b)
242        || mapped
243            .iter()
244            .zip(source_shape)
245            .any(|(&i, &len)| len != 1 && len != target_shape[i])
246    {
247        return Err(incompatible());
248    }
249    Ok(mapped)
250}

252fn mask_parse_error(error: crate::expr_parser::ExprParseError) -> ImageError {
253    ImageError::CasaSelection(format!(
254        "Input mask specification is incorrect: {}",
255        mask_error_wording(error)
256    ))
257}

259fn open_operand(spec: &str) -> Result<Operand, ImageError> {
260    let direct_path = Path::new(spec);
261    if direct_path.exists() {
262        return open_path_with_mask_name(direct_path, None, spec);
263    }
264    if let Some((left, right)) = spec.rsplit_once(':')
265        && !left.is_empty()
266        && !right.trim().is_empty()
267        && Path::new(left).exists()
268    {
269        return open_path_with_mask_name(Path::new(left), Some(right.trim().to_string()), spec);
270    }
271    Err(ImageError::CasaSelection(format!(
272        "Input mask specification is incorrect: ImageExprParse: '{spec}' is an unknown \
273         lattice or image or it is an unqualified region"
274    )))
275}

277fn open_path_with_mask_name(
278    path: &Path,
279    mask_name: Option<String>,
280    spec: &str,
281) -> Result<Operand, ImageError> {
282    if path.is_file() {
283        if mask_name.is_some() {
284            return Err(ImageError::CasaSelection(format!(
285                "FITS image '{spec}' does not hold named masks"
286            )));
287        }
288        let fits = read_fits_image(path)?;
289        let pixels = fits.pixels.mapv(f64::from);
290        let has_nan = fits.pixels.iter().any(|v| v.is_nan());
291        let mask = has_nan.then(|| fits.pixels.mapv(|v| !v.is_nan()));
292        let temp = make_temp_image(
293            fits.shape,
294            fits.coordinates,
295            &fits.units,
296            &fits.image_info,
297            pixels,
298            mask,
299        )?;
300        return Ok(Operand::Real(MaskSource::Memory(Box::new(temp)), true));
301    }
302
303    match AnyPagedImage::open(path)? {
304        AnyPagedImage::Float64(img) => {
305            if mask_name.is_some() {
306                return Err(ImageError::CasaSelection(format!(
307                    "Input mask specification is incorrect: ImageExprParse: {} is a PagedImage with an unsupported data type",
308                    path.display()
309                )));
310            }
311            Ok(Operand::Real(MaskSource::Paged(Box::new(img)), false))
312        }
313        AnyPagedImage::Float32(img) => {
314            convert_paged_mask_source(&img, mask_name.as_deref(), f64::from)
315                .map(|image| Operand::Real(image, true))
316        }
317        AnyPagedImage::Complex64(img) if mask_name.is_none() => {
318            Ok(Operand::Complex(MaskSource::Paged(Box::new(img))))
319        }
320        AnyPagedImage::Complex64(img) => {
321            convert_paged_mask_source(&img, mask_name.as_deref(), |c| c).map(Operand::Complex)
322        }
323        AnyPagedImage::Complex32(img) => {
324            convert_paged_mask_source(&img, mask_name.as_deref(), |c: Complex32| {
325                Complex64::new(f64::from(c.re), f64::from(c.im))
326            })
327            .map(Operand::Complex)
328        }
329    }
330}

332fn convert_paged_mask_source<S: ImagePixel, T: ImagePixel>(
333    img: &PagedImage<S>,
334    mask_name: Option<&str>,
335    convert: impl FnMut(S) -> T,
336) -> Result<MaskSource<T>, ImageError> {
337    validate_mask_name(img, mask_name)?;
338    let data = img.get()?.mapv(convert);
339    let mask = match mask_name {
340        Some(name) => Some(img.get_named_mask(name)?),
341        None => img.default_mask()?,
342    };
343    let temp = make_temp_image(
344        img.shape().to_vec(),
345        img.coordinates().clone(),
346        img.units(),
347        &img.image_info()?,
348        data,
349        mask,
350    )?;
351    Ok(MaskSource::Memory(Box::new(temp)))
352}
353
354fn validate_mask_name<T: ImagePixel>(
355    img: &PagedImage<T>,
356    mask_name: Option<&str>,
357) -> Result<(), ImageError> {
358    if let Some(name) = mask_name
359        && !img.mask_names().iter().any(|existing| existing == name)
360    {
361        return Err(ImageError::CasaSelection(format!(
362            "Mask '{name}' not found in image"
363        )));
364    }
365    Ok(())
366}

368fn make_temp_image<T: ImagePixel>(
369    shape: Vec<usize>,
370    coords: CoordinateSystem,
371    units: &str,
372    info: &ImageInfo,
373    data: ArrayD<T>,
374    mask: Option<ArrayD<bool>>,
375) -> Result<TempImage<T>, ImageError> {
376    let mut temp = TempImage::new(shape, coords, TempStoragePolicy::Memory)?;
377    temp.set_units(units)?;
378    temp.set_image_info(info)?;
379    let ndim = temp.ndim();
380    temp.put_slice(&data, &vec![0; ndim])?;
381    if let Some(m) = mask {
382        temp.put_mask("mask0", &m)?;
383        temp.set_default_mask("mask0")?;
384    }
385    Ok(temp)
