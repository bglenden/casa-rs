5use std::sync::Arc;

7use thiserror::Error;

9/// Invalid or unsupported native EVLA generation inputs.
10#[derive(Clone, Copy, Debug, Error, PartialEq, Eq)]
11pub enum NativeAwRequestError {
12    /// Radial samples are not finite, uniformly spaced, or physically valid.
13    #[error("EVLA surface requires finite, uniformly spaced radius/height/slope samples")]
14    InvalidSurface,
15    /// The requested geometry, sampling, frequency or polarization is unsupported.
16    #[error("invalid or unsupported native EVLA A/W sampling")]
17    InvalidSampling,
18    /// Catalog axes are empty, inconsistent, duplicated, or exceed the declared bound.
19    #[error("native A/W catalog axes are invalid or exceed the declared cell bound")]
20    InvalidCatalog,
21}
22
23/// Explicit EVLA radial dish samples, in metres and dimensionless surface slope.
24///
25/// Content is validated here, not located by a runtime search. Clones share
26/// immutable samples. Data acquisition and its bounded I/O belong to the
27/// application.
28#[derive(Clone, Debug)]
29pub struct EvlaDishSurface {
30    samples: Arc<[[f64; 3]]>,
31}
32
33impl EvlaDishSurface {
34    /// Validate uniformly spaced samples from the optical axis to the 12.5 m rim.
35    pub fn new(samples: Vec<[f64; 3]>) -> Result<Self, NativeAwRequestError> {
36        if samples.len() < 3
37            || samples.iter().flatten().any(|value| !value.is_finite())
38            || samples[0] != [0.0; 3]
39            || samples.last().is_none_or(|sample| sample[0] != 12.5)
40        {
41            return Err(NativeAwRequestError::InvalidSurface);
42        }
43        let step = 12.5 / (samples.len() - 1) as f64;
44        if samples.iter().enumerate().any(|(index, sample)| {
45            (sample[0] - index as f64 * step).abs() > step * 1e-4
46                || sample[1] < 0.0
47                || sample[2] < 0.0
48        }) {
49            return Err(NativeAwRequestError::InvalidSurface);
50        }
51        Ok(Self {
52            samples: samples.into(),
53        })
54    }
55
56    /// Parse a caller-supplied three-column radius/height/slope text input.
57    pub fn from_surface_text(text: &str) -> Result<Self, NativeAwRequestError> {
58        let samples = text
59            .lines()
60            .filter(|line| !line.trim().is_empty())
61            .map(|line| {
62                let mut values = line.split_whitespace();
63                let mut sample = [0.0; 3];
64                for value in &mut sample {
65                    *value = values
66                        .next()
67                        .and_then(|v| v.parse::<f64>().ok())
68                        .ok_or(NativeAwRequestError::InvalidSurface)?;
69                }
70                if values.next().is_some() {
71                    return Err(NativeAwRequestError::InvalidSurface);
72                }
73                Ok(sample)
74            })
75            .collect::<Result<Vec<_>, _>>()?;
76        Self::new(samples)
77    }
78
79    /// Validated immutable physical samples.
80    #[must_use]
81    pub fn samples(&self) -> &[[f64; 3]] {
82        &self.samples
83    }
84
85    /// Shared sample payload bytes, charged once to generation residency.
86    #[must_use]
87    pub fn resident_bytes(&self) -> usize {
88        size_of_val(self.samples.as_ref())
89    }
90}

92/// Scientific sampling of one native paired EVLA convolution cell.
93///
94/// Increments describe the working sky grid, not the final cropped UV plane.
95/// The first provider uses centered, even, square SIN grids, circular parallel
96/// hands and f64 geometry / Complex32 field and transform arithmetic.
97#[derive(Clone, Copy, Debug, PartialEq)]
98pub struct EvlaAwCellRequest {
99    /// Full working FFT extent, before support selection.
100    pub size: usize,
101    /// Signed sky increments in radians per pixel.
102    pub sky_increment_rad: [f64; 2],
103    /// Beam frequency in Hz, before CASA-compatible float rounding.
104    pub frequency_hz: f64,
105    /// Explicit beam frequency for the paired weight product, in Hz.
106    pub conjugate_frequency_hz: f64,
107    /// Nonnegative W coordinate in wavelengths at the cell frequency.
108    pub w_wavelengths: f64,
109    /// Physical parallactic angle in radians.
110    pub parallactic_angle_rad: f64,
111    /// Circular parallel-hand Mueller element: 0 or 15.
112    pub mueller: usize,
113    /// Integer CF sampling per normal grid pixel.
114    pub oversampling: usize,
115    /// Whether the prolate-spheroidal anti-aliasing term participates.
116    pub prolate_spheroidal: bool,
117    /// Whether the EVLA aperture response participates.
118    pub aperture: bool,
119}
120
121impl EvlaAwCellRequest {
122    /// Numerical residency ceiling for six Complex32 working planes, one
123    /// disposable FFTW planning plane and one opaque native-plan allowance.
124    pub fn generation_workspace_bytes(&self) -> Result<usize, NativeAwRequestError> {
125        self.validate()?;
126        self.size
127            .checked_mul(self.size)
128            .and_then(|pixels| pixels.checked_mul(8))
129            .and_then(|values| values.checked_add(64))
130            .and_then(|values| values.checked_mul(8))
131            .ok_or(NativeAwRequestError::InvalidSampling)
132    }
133
134    /// Validate a cell before numerical generation or scientific identity minting.
135    pub fn validate(&self) -> Result<(), NativeAwRequestError> {
136        if self.size < 8
137            || !self.size.is_multiple_of(2)
138            || self
139                .size
140                .checked_mul(self.size)
141                .and_then(|n| n.checked_mul(6 * 8))
142                .is_none()
143            || self
144                .sky_increment_rad
145                .iter()
146                .any(|v| !v.is_finite() || *v == 0.0)
147            || self.sky_increment_rad[0] >= 0.0
148            || self.sky_increment_rad[1] <= 0.0
149            || self.sky_increment_rad[0].abs() != self.sky_increment_rad[1].abs()
150            || !self.frequency_hz.is_finite()
151            || !(0.9e9..=8e9).contains(&self.frequency_hz)
152            || !self.conjugate_frequency_hz.is_finite()
153            || !(0.9e9..=8e9).contains(&self.conjugate_frequency_hz)
154            || !self.w_wavelengths.is_finite()
155            || self.w_wavelengths < 0.0
156            || !self.parallactic_angle_rad.is_finite()
157            || !matches!(self.mueller, 0 | 15)
158            || self.oversampling == 0
159            || self.oversampling > self.size / 4
160        {
161            return Err(NativeAwRequestError::InvalidSampling);
162        }
163        Ok(())
164    }
165}

167/// One selected SPW's explicit native beam-frequency sampling.
168#[derive(Clone, Debug, PartialEq)]
169pub struct NativeAwFrequencyGroup {
170    /// Selected MeasurementSet spectral-window identifier.
171    pub spectral_window: u32,
172    /// Selected physical channel frequencies, in native ascending/descending order.
173    pub channel_frequencies_hz: Vec<f64>,
174    /// Representative CF frequency in Hz resolved by the application planner.
175    pub cf_frequency_hz: f64,
176}

178/// Working image geometry and discrete convolution sampling.
179#[derive(Clone, Copy, Debug, PartialEq)]
180pub struct NativeAwGrid {
181    /// Even full working FFT extent, before support selection.
182    pub size: usize,
183    /// Signed square SIN sky increments, in radians per working pixel.
184    pub sky_increment_rad: [f64; 2],
185    /// Integer convolution sampling per normal grid pixel.
186    pub oversampling: usize,
187}

189/// Explicit choices for every independently enabled A/W term.
190#[derive(Clone, Copy, Debug, PartialEq, Eq)]
191pub struct NativeAwTerms {
192    /// Apply the EVLA aperture response.
193    pub aperture: bool,
194    /// Apply nonzero W phase screens.
195    pub w_term: bool,
196    /// Apply the prolate-spheroidal anti-aliasing function.
197    pub prolate_spheroidal: bool,
198    /// Use per-SPW beam-frequency sampling rather than only the image reference.
199    pub wideband: bool,
200    /// Use the nearest sampled conjugate frequency in the weight beam product.
201    pub conjugate_beams: bool,
202}

204/// Complete application-resolved science inputs and an explicit metadata bound.
205#[derive(Clone, Debug)]
206pub struct NativeAwRequestInput {
207    /// Explicit content-identified EVLA dish model.
208    pub surface: EvlaDishSurface,
209    /// Antenna diameter in metres; this EVLA model requires exactly 25 m.
210    pub antenna_diameter_m: f64,
211    /// Ordered frequency groups, including selected physical SPW coordinates.
212    pub frequencies: Vec<NativeAwFrequencyGroup>,
213    /// Sorted nonnegative W samples in wavelengths.
214    pub w_values: Vec<f64>,
215    /// W-grid scale used by the shared prepared-cell selector.
216    pub w_increment: f64,
217    /// Sorted physical parallactic angles in radians.
218    pub pa_values: Vec<f64>,
219    /// Sorted required circular parallel-hand Mueller elements, 0 and/or 15.
220    pub mueller_elements: Vec<usize>,
221    /// Image reference frequency defining conjugate-frequency selection.
222    pub reference_frequency_hz: f64,
223    /// Working grid and oversampling; no output crop is assumed.
224    pub grid: NativeAwGrid,
225    /// Explicit independently enabled scientific terms.
226    pub terms: NativeAwTerms,
227    /// Maximum expected cells admitted by the application metadata policy.
228    pub maximum_cells: usize,
229}
230
231impl NativeAwRequestInput {
232    /// Validate the catalog axes and terms (`NativeAwRequest::new`'s
233    /// rules): sorted finite w values from zero, a positive w increment,
234    /// sorted parallactic angles, parallel-hand Mueller elements, ascending
235    /// distinct frequency groups in the EVLA bands, a single reference group
236    /// without wideband selection, and a bounded non-empty cell count.
237    pub fn validate(&self) -> Result<(), NativeAwRequestError> {
238        let Self {
239            frequencies,
240            w_values,
241            pa_values,
242            mueller_elements,
243            grid,
244            terms,
245            ..
246        } = self;
247        frequencies
248            .len()
249            .checked_mul(w_values.len())
250            .and_then(|n| n.checked_mul(pa_values.len()))
251            .and_then(|n| n.checked_mul(mueller_elements.len()))
252            .filter(|n| *n > 0 && *n <= self.maximum_cells)
253            .ok_or(NativeAwRequestError::InvalidCatalog)?;
254        let strictly_sorted = |values: &[f64]| {
255            values.iter().all(|v| v.is_finite()) && values.windows(2).all(|pair| pair[0] < pair[1])
256        };
257        if self.antenna_diameter_m != 25.0
258            || !strictly_sorted(w_values)
259            || w_values[0] < 0.0
260            || !self.w_increment.is_finite()
261            || self.w_increment <= 0.0
262            || !strictly_sorted(pa_values)
263            || !self.reference_frequency_hz.is_finite()
264            || self.reference_frequency_hz <= 0.0
265            || mueller_elements.iter().any(|m| !matches!(m, 0 | 15))
266            || mueller_elements.windows(2).any(|p| p[0] >= p[1])
267            || frequencies
268                .windows(2)
269                .any(|p| p[0].cf_frequency_hz >= p[1].cf_frequency_hz)
270            || frequencies.iter().any(|group| {
271                !group.cf_frequency_hz.is_finite()
272                    || !(0.9e9..=8e9).contains(&group.cf_frequency_hz)
273                    || group.channel_frequencies_hz.is_empty()
274                    || group
275                        .channel_frequencies_hz
276                        .iter()
277                        .any(|f| !f.is_finite() || *f <= 0.0)
278            })
279            || frequencies.iter().enumerate().any(|(i, g)| {
280                frequencies[..i]
281                    .iter()
282                    .any(|p| p.spectral_window == g.spectral_window)
283            })
284            || (!terms.w_term && w_values.as_slice() != [0.0])
285            || (!terms.wideband
286                && (frequencies.len() != 1
287                    || frequencies[0].cf_frequency_hz != self.reference_frequency_hz))
288        {
289            return Err(NativeAwRequestError::InvalidCatalog);
290        }
291        EvlaAwCellRequest {
292            size: grid.size,
293            sky_increment_rad: grid.sky_increment_rad,
294            frequency_hz: frequencies[0].cf_frequency_hz,
295            conjugate_frequency_hz: frequencies[0].cf_frequency_hz,
296            w_wavelengths: w_values[0],
297            parallactic_angle_rad: pa_values[0],
298            mueller: mueller_elements[0],
299            oversampling: grid.oversampling,
300            prolate_spheroidal: terms.prolate_spheroidal,
301            aperture: terms.aperture,
302        }
303        .validate()
304    }
