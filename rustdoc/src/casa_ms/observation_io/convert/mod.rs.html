2//! Converting between MeasurementSet v2 and MSv4 through the observation
3//! boundary: the source is read as batches like any consumer reads it, and
4//! the destination is created from those batches and the source's
5//! catalogue ([`to_msv4()`], [`to_msv2()`]).
6//!
7//! A conversion first inventories what the destination cannot hold (a
8//! [`Loss`]) and refuses, before any byte is written, every loss the
9//! caller did not accept ([`ConversionOptions::accept`]), as a
10//! [`CreationPlan`] does; converting into an MSv2 is a plan over one
11//! input. What it keeps it keeps exactly: every acquisition, its flags (row
12//! flags folded in), weights, UVW and time auxiliaries, and every selected
13//! data group. Padding, a grid cell no acquisition filled, stays padding:
14//! an MSv4 holds it as a NaN weight, and an MSv2 as a flagged row of NaN
15//! data and weights, or, by the declared policy that reproduces an
16//! original sparse MSv2, as no row ([`PaddingRows`]).
17

32/// How a conversion reads its source and stores its destination.
33#[derive(Clone, Debug)]
34pub struct ConversionOptions {
35    /// The data groups to convert, the first carrying every role; `None`
36    /// converts every group of the source, `base` first.
37    pub data_groups: Option<Vec<String>>,
38    /// The losses accepted; any other refuses the conversion.
39    pub accept: Vec<LossKind>,
40    /// The most bytes one time range of every selected group's batches
41    /// may hold together.
42    pub batch_bytes: usize,
43    /// MSv4 destinations: partitions converted at once, each by its own
44    /// thread. An MSv2 is written by one.
45    pub workers: usize,
46    /// MSv4 destinations: the chunk shape `[time, baseline, frequency,
47    /// polarization]`, 0 for a whole axis.
48    pub chunks: [u64; 4],
49    /// MSv4 destinations: the payload variables' codec.
50    pub codec: Msv4Codec,
51    /// MSv2 destinations: what a padded grid cell becomes.
52    pub padding: PaddingRows,
53    /// Software recorded as the creator.
54    pub creator: String,
55    /// Creation time recorded, ISO 8601.
56    pub creation_date: String,
57}
58
59/// What a conversion did.
60#[derive(Clone, Debug, Default, PartialEq)]
61pub struct ConversionReport {
62    /// The losses the conversion took, all accepted.
63    pub losses: Vec<Loss>,
64    /// Partitions converted.
65    pub partitions: usize,
66    /// Batches of the first group read.
67    pub batches: usize,
68    /// Samples read (the grids' cells, padding included).
69    pub samples: usize,
70    /// MSv2 destinations: rows written, one per sample, or per acquisition
71    /// with padding omitted ([`PaddingRows`]).
72    pub rows: usize,
73    /// MSv4 destinations: payload elements encoded, over every variable.
74    pub elements: usize,
75    /// MSv4 destinations: elements held toward a chunk a batch covered only
76    /// partly.
77    pub held_elements: usize,
78}

80/// Convert the store at `source` (MSv2 or MSv4) into a new MeasurementSet
81/// v2 at `destination`: a [`CreationPlan`] over the one input, written as
82/// [`Destination::New`].
83///
84/// The selected data groups ([`ConversionOptions::data_groups`]) become
85/// `DATA` (`base`), `CORRECTED_DATA` (`corrected`) and `MODEL_DATA`
86/// (`model`), sharing the first group's flags, weights and UVW. Partitions
87/// sharing a spectral window, polarization or intents share their
88/// subtable rows. Every grid cell becomes one row; padding becomes a row
89/// whose every `FLAG` is set, with NaN data, weights and `UVW`, the grid's
90/// interval, no exposure and `FLAG_ROW` false (its flags already exclude
91/// it), as D23 decided. With [`PaddingRows::Omitted`], padding becomes no
92/// row, which reproduces an original sparse MSv2. A field that follows an
93/// ephemeris gets the table casacore reads (`FIELD/EPHEM<id>_<name>.tab`)
94/// and its `EPHEMERIS_ID`, its `PHASE_DIR` the offset from the body and its
95/// `TIME` the epoch its catalogue entry states, or else its first time,
96/// the table stamped [`ConversionOptions::creation_date`]. `SIGMA` is derived
97/// from the weights, a loss the caller must accept as [`LossKind::Sigma`]
98/// ([`Loss::SigmaDerived`]) unless the source's stored `SIGMA` is already
99/// `1/√WEIGHT`. Names are the source's: an MSv4 converted from an
100/// MSv2 holds XRADIO's suffixed names (`NAME_FIELDID`, `NAME_STATION`,
101/// ...), so a round trip renames.
102///
103/// # Errors
104///
105/// As [`CreationPlan::new`] and [`CreationPlan::write`].
106pub fn to_msv2(
107    source: &Path,
108    destination: &Path,
109    options: &ConversionOptions,
110) -> Result<ConversionReport, ObservationError> {
111    let plan = CreationPlan::new(
112        &[CreationInput::new(source)],
113        &CreationOptions {
114            data_groups: options.data_groups.clone(),
115            accept: options.accept.clone(),
116            batch_bytes: options.batch_bytes,
117            memory_bytes: default_memory_bytes()?,
118            padding: options.padding,
119            tolerance: casa_observation::MergeTolerance::default(),
120            history: Vec::new(),
121            creation_date: options.creation_date.clone(),
122        },
123        Destination::New(destination.to_path_buf()),
124    )?;
125    let report = plan.write()?;
126    Ok(ConversionReport {
127        losses: report.losses,
128        partitions: report.partitions,
129        batches: report.batches,
130        samples: report.samples,
131        rows: report.rows,
132        ..ConversionReport::default()
133    })
134}
