4//! The command-line and task surface of native CASA-RS imaging, on the
5//! shared task front end (#931).
6//!
7//! The binary hosts CASA's `tclean` (the imager re-surfaced, #931). The
8//! front end resolves tclean's parameters through the provider-contracts
9//! catalog: command-line flags and a `--json-run` request become the same
10//! resolved values, and those values, and nothing else, make the
11//! [`ImagingRequest`] the application runs, read as CASA's modes read them
12//! ([`ImagingRequest::from_tclean`]). A run's reply carries its
13//! [`ImagerRunTaskResult`], and returns CASA's summary record
14//! ([`tclean_record`]).

33fn imager_surface() -> Result<SurfaceContractBundle, String> {
34    casa_provider_contracts::builtin_surface_bundle("tclean")
35}

37fn workspace() -> Result<PathBuf, String> {
38    std::env::current_dir().map_err(|error| format!("the working directory: {error}"))
39}

41/// The request `resolved`, every tclean parameter, makes. A value the
42/// catalog accepts but casa-rs cannot honour in the request's modes, or that
43/// the request's own read refuses (`stokes = "IX"`), is refused at its
44/// parameter.
45fn request_from(
46    resolved: Map<String, Value>,
47) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
48    let mut request = ImagingRequest::from_tclean(&resolved).map_err(|refusal| {
49        let failure = TaskFailure::validation(refusal.message);
50        if refusal.parameter.is_empty() {
51            failure
52        } else {
53            failure.at(refusal.parameter)
54        }
55    })?;
56    // AWP2 reads the same physical dish data as CASA when the host has
57    // the local data tree. The imaging library takes an explicit path.
58    if let casa_imaging_application::Gridder::Awp2 { evla_surface, .. } = &mut request.gridder
59        && evla_surface.is_none()
60    {
61        *evla_surface = std::env::var_os("HOME")
62            .map(|home| PathBuf::from(home).join(".casa/data/nrao/VLA/VLA.surface"));
63    }
64    Ok((ImagingParameters(resolved), request))
65}

67/// Resolve `parameters`, catalog-named and sparse, over the catalog's
68/// defaults: the resolved parameters and the request they make.
69pub fn resolve_request(
70    parameters: &ImagingParameters,
71) -> Result<(ImagingParameters, ImagingRequest), String> {
72    let resolved = resolve_plain_values(imager_surface()?, workspace()?, &parameters.0)
73        .map_err(|error| format!("resolve tclean parameters: {error}"))?;
74    request_from(resolved).map_err(|failure| failure.message)
75}

77/// The environment variable that asks the `casars-imager` binary for its
78/// run summary. CASA's `tclean` writes nothing beside its image products, so
79/// neither does a tclean run here (owner decision of 2026-11-02, #980); set
80/// to `1`, the binary also writes `<imagename>.summary.json`, the run's
81/// status, resolved parameters, backend and each phase's time and memory,
82/// for performance tooling ([`casa_imaging_application::RunSummary`]).
83pub const RUN_SUMMARY_VARIABLE: &str = "CASA_RS_IMAGER_RUN_SUMMARY";
84
85/// Whether this process was asked for run summaries
86/// ([`RUN_SUMMARY_VARIABLE`]).
87fn run_summary_requested() -> bool {
88    std::env::var_os(RUN_SUMMARY_VARIABLE).is_some_and(|value| value == "1")
89}

91/// Run `request`, resolved as `parameters`, as CASA's tclean does: it
92/// writes the image products and nothing else beside them. SIGINT cancels
93/// the run.
94pub fn run(
95    parameters: ImagingParameters,
96    request: &ImagingRequest,
97) -> Result<ImagerRunTaskResult, String> {
98    run_with_return(parameters, request, false).map(|(result, _)| result)
99}
100
101/// Keep what the CASA return needs beyond the report in memory, without
102/// adding fields to the serialized task report or the run summary, which is
103/// written beside the products only when `summary` asks for it.
104fn run_with_return(
105    parameters: ImagingParameters,
106    request: &ImagingRequest,
107    summary: bool,
108) -> Result<(ImagerRunTaskResult, RunReturn), String> {
109    // Task continuation is distinct from internal atomic output replacement.
110    let restart = parameters
111        .0
112        .get("restart")
113        .and_then(Value::as_bool)
114        .ok_or_else(|| "resolved tclean restart must be boolean".to_owned())?;
115    let request = request
116        .with_tclean_output_policy(restart)
117        .map_err(|error| error.to_string())?;
118    let request = &request;
119    let started = Instant::now();
120    let summary = summary.then(|| {
121        let mut path = request.imagename.clone().into_os_string();
122        path.push(".summary.json");
123        SummaryTarget {
124            path: path.into(),
125            request: Value::Object(parameters.0.clone()),
126        }
127    });
128    let context = RunContext {
129        host: HostResources::detect().map_err(|error| error.to_string())?,
130        policy: request.resource_policy(),
131        cancel: interrupt::token().clone(),
132        summary,
133    };
134    let outcome = execute(request, context).map_err(|error| error.to_string())?;
135    let result =
136        ImagerRunTaskResult::from_outcome(parameters, request, &outcome, started.elapsed());
137    Ok((
138        result,
139        RunReturn {
140            statistics: outcome.final_residual_statistics,
141            plane_rows: outcome.plane_rows,
142        },
143    ))
144}

146/// The front end of the `casars-imager` binary, which hosts `tclean`.
147pub fn task_front_end()
148-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
149    TaskFrontEnd::new_with_deferred_schema(
150        "casars-imager",
151        imager_task_schema_bundle(),
152        &["tclean"],
153        run_task,
154    )
155    .expect("the imager contract carries its surface")
156}

158/// The resolved parameters of `task`, a tclean run, and the request they
159/// make; a request they cannot make is refused.
160pub fn task_request(
161    task: &TaskInvocation<'_>,
162) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
163    request_from(task.plain_values())
164}

166/// Run `task`, one resolved imager request, as the `casars-imager` binary
167/// does: the execution callback of [`task_front_end`]. SIGINT cancels it:
168/// the run stops at its next safe point, publishes nothing, and replies that
169/// it was interrupted.
170pub fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
171    let (parameters, request) = task_request(task)?;
172    // The record echoes some of tclean's values: a dirty run's threshold as
173    // the caller gave it, a number as a number and text as text, and
174    // tclean's default 0.0 where it was not given.
175    let mut values = parameters.0.clone();
176    values.insert(
177        "threshold".to_string(),
178        task.session()
179            .given("threshold")
180            .map_or(Value::from(0.0), ParameterValue::to_plain_json),
181    );
182    interrupt::install().map_err(|error| {
183        TaskFailure::execution(format!("install the interrupt handler: {error}"))
184    })?;
185    let (result, held) =
186        run_with_return(parameters, &request, run_summary_requested()).map_err(|error| {
187            if interrupt::token().is_cancelled() {
188                TaskFailure::interrupted(format!("{error}; nothing was published"))
189            } else {
190                TaskFailure::execution(error)
191            }
192        })?;
193    let text = format!(
194        "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, \
195         {} reported minor iterations, {} actual components, stop={:?})\n",
196        request.imagename.display(),
197        result.run.gridded_samples,
198        result.run.major_cycles,
199        result.run.minor_iterations,
200        result.run.actual_minor_iterations,
201        result.run.clean_stop_reason,
202    );
203    let value = tclean_record(&request, &result.run, &values, &held);
204    history::write_tclean_history(&request.imagename, task);
205    Ok(TaskSuccess::new(value, &result)?.with_text(text))
206}
