2//! Planes of an image: `feather`'s direction planes and `impv`'s samples
3//! along a path in one, with the plane on any two axes (#1002).
4
5use casa_types::ArrayD;
6use ndarray::{Array2, IxDyn};
7
8/// One plane of an image: its two `axes`, at `values` of the `others`.
9pub(super) struct PlaneAt<'a> {
10    pub(super) axes: [usize; 2],
11    pub(super) others: &'a [usize],
12    pub(super) values: &'a [usize],
13}
14
15impl PlaneAt<'_> {
16    /// The index of the plane's pixel (`x`, `y`) in the image.
17    fn index(&self, rank: usize, x: usize, y: usize) -> Vec<usize> {
18        let mut index = vec![0; rank];
19        for (&axis, &value) in self.others.iter().zip(self.values) {
20            index[axis] = value;
21        }
22        index[self.axes[0]] = x;
23        index[self.axes[1]] = y;
24        index
25    }
26
27    pub(super) fn extract(&self, data: &ArrayD<f32>) -> Array2<f32> {
28        let shape = (data.shape()[self.axes[0]], data.shape()[self.axes[1]]);
29        Array2::from_shape_fn(shape, |(x, y)| data[IxDyn(&self.index(data.ndim(), x, y))])
30    }
31
32    pub(super) fn put(&self, data: &mut ArrayD<f32>, plane: &Array2<f32>) {
33        for ((x, y), value) in plane.indexed_iter() {
34            let index = self.index(data.ndim(), x, y);
35            data[IxDyn(&index)] = *value;
36        }
37    }
38}
39
40/// The index of element `value` of an array of `shape`, the last axis
41/// fastest.
42pub(super) fn unravel_index(mut value: usize, shape: &[usize]) -> Vec<usize> {
43    if shape.is_empty() {
44        return Vec::new();
45    }
46    let mut out = vec![0usize; shape.len()];
47    for axis in (0..shape.len()).rev() {
48        out[axis] = value % shape[axis];
49        value /= shape[axis];
50    }
51    out
52}
