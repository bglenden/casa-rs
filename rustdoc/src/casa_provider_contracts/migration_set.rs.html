2//! The rules for [`MigrationStep::Set`](super::MigrationStep::Set) (#931).
3//!
4//! A sparse profile holds only the values its author set; an omitted value
5//! adopts the current default, with a warning when that default changed
6//! (ADR-0006). A `Set` step moves an explicit value between parameters: when
7//! a surface takes CASA's names, casa-rs's old `action='unflag'` becomes
8//! CASA's `mode='unflag'`, and mstransform's old `width` becomes CASA's
9//! `chanaverage=True` with `chanbin`. Its condition may therefore name only
10//! values a profile set: it is a conjunction of [`Predicate::IsSet`] and
11//! [`Predicate::Equals`], so no step can pin a default the profile omitted.
12//! A conjunction may also say that a value it names as set is not one
13//! value, [`Predicate::Not`] of an `Equals`: flagdata's old explicit
14//! `timedev` is a threshold unless it is the old absence `'none'`.
15
16use std::collections::{BTreeMap, BTreeSet};
17
18use super::{ContractValidationError, ParameterValue, Predicate, SurfaceDefinition};
19
20impl Predicate {
21    /// Whether this condition holds of a profile's explicit `overrides`, by
22    /// the parameter names they carry: `IsSet` when the profile set the
23    /// parameter, `Equals` when it set it to exactly `value`, `Not` of an
24    /// `Equals` when it did not, `All` when every part holds. Any other
25    /// predicate never holds here; contract validation refuses it in a
26    /// migration, and a `Not` outside a conjunction naming its parameter
27    /// as set.
28    pub fn holds_for_overrides(&self, overrides: &BTreeMap<String, ParameterValue>) -> bool {
29        match self {
30            Self::IsSet { parameter } => overrides.contains_key(parameter),
31            Self::Equals { parameter, value } => overrides.get(parameter) == Some(value),
32            Self::Not { predicate } => {
33                matches!(**predicate, Self::Equals { .. })
34                    && !predicate.holds_for_overrides(overrides)
35            }
36            Self::All { predicates } => {
37                !predicates.is_empty()
38                    && predicates
39                        .iter()
40                        .all(|predicate| predicate.holds_for_overrides(overrides))
41            }
42            Self::Always | Self::Never | Self::Any { .. } => false,
43        }
44    }
45}
46
47/// Whether `when` names only values a profile set: a `Not` of an `Equals`
48/// only in a conjunction that names the same parameter with `IsSet` or
49/// `Equals`, so it narrows a value the profile set and never holds of an
50/// omitted one.
51fn names_only_set_values(when: &Predicate) -> bool {
52    match when {
53        Predicate::IsSet { .. } | Predicate::Equals { .. } => true,
54        Predicate::All { predicates } => {
55            let named = predicates
56                .iter()
57                .filter_map(|predicate| match predicate {
58                    Predicate::IsSet { parameter } | Predicate::Equals { parameter, .. } => {
59                        Some(parameter)
60                    }
61                    _ => None,
62                })
63                .collect::<BTreeSet<_>>();
64            !predicates.is_empty()
65                && predicates.iter().all(|predicate| match predicate {
66                    Predicate::Not { predicate } => matches!(
67                        &**predicate,
68                        Predicate::Equals { parameter, .. } if named.contains(parameter)
69                    ),
70                    other => names_only_set_values(other),
71                })
72        }
73        Predicate::Always | Predicate::Never | Predicate::Not { .. } | Predicate::Any { .. } => {
74            false
75        }
76    }
77}
78
79/// Refuse a `Set` step whose condition could hold of a profile that set
80/// none of the values it names.
81pub(super) fn validate_condition(
82    surface: &SurfaceDefinition,
83    parameter: &str,
84    when: &Predicate,
85    errors: &mut Vec<ContractValidationError>,
86) {
87    if !names_only_set_values(when) {
88        errors.push(
89            ContractValidationError::new(
90                "migration_set_condition",
91                "a migration's set step needs a condition of explicit values only: \
92                 is_set and equals, or all of them, where not equals may narrow a \
93                 parameter the conjunction names",
94            )
95            .at(surface.id(), Some(parameter)),
96        );
97    }
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103
104    fn equals(parameter: &str, value: &str) -> Predicate {
105        Predicate::Equals {
106            parameter: parameter.into(),
107            value: ParameterValue::String(value.into()),
108        }
109    }
110
111    #[test]
112    fn a_condition_holds_only_of_values_a_profile_set() {
113        let unflag = Predicate::All {
114            predicates: vec![equals("mode", "manual"), equals("action", "unflag")],
115        };
116        let mut overrides = BTreeMap::from([(
117            "action".to_string(),
118            ParameterValue::String("unflag".into()),
119        )]);
120        // The old default mode is not assumed: the profile did not set it.
121        assert!(!unflag.holds_for_overrides(&overrides));
122        overrides.insert("mode".into(), ParameterValue::String("manual".into()));
123        assert!(unflag.holds_for_overrides(&overrides));
124        let width = Predicate::IsSet {
125            parameter: "chanbin".into(),
126        };
127        assert!(!width.holds_for_overrides(&overrides));
128        overrides.insert("chanbin".into(), ParameterValue::Integer(4));
129        assert!(width.holds_for_overrides(&overrides));
130        assert!(
131            !Predicate::All { predicates: vec![] }.holds_for_overrides(&overrides),
132            "an empty conjunction names nothing"
133        );
134    }
135
136    #[test]
137    fn a_condition_that_could_pin_an_omitted_default_is_refused() {
138        let pins = [
139            Predicate::Always,
140            Predicate::Not {
141                predicate: Box::new(Predicate::IsSet {
142                    parameter: "mode".into(),
143                }),
144            },
145            Predicate::Any {
146                predicates: vec![equals("mode", "manual")],
147            },
148            Predicate::All {
149                predicates: vec![
150                    equals("mode", "manual"),
151                    Predicate::Not {
152                        predicate: Box::new(equals("action", "flag")),
153                    },
154                ],
155            },
156            Predicate::All { predicates: vec![] },
157            Predicate::Not {
158                predicate: Box::new(equals("timedev", "none")),
159            },
160        ];
161        for when in pins {
162            assert!(!names_only_set_values(&when), "{when:?}");
163        }
164        assert!(names_only_set_values(&Predicate::All {
165            predicates: vec![
166                equals("mode", "manual"),
167                Predicate::IsSet {
168                    parameter: "action".into()
169                }
170            ],
171        }));
172        assert!(names_only_set_values(&explicit_threshold()));
173    }
174
175    /// flagdata's old `timedev` set to a threshold, not its old absence.
176    fn explicit_threshold() -> Predicate {
177        Predicate::All {
178            predicates: vec![
179                Predicate::IsSet {
180                    parameter: "timedev".into(),
181                },
182                Predicate::Not {
183                    predicate: Box::new(equals("timedev", "none")),
184                },
185            ],
186        }
187    }
188
189    #[test]
190    fn a_value_set_may_be_narrowed_to_differ_from_one_value() {
191        let threshold = explicit_threshold();
192        let profile = |value: Option<ParameterValue>| {
193            value
194                .map(|value| BTreeMap::from([("timedev".to_string(), value)]))
195                .unwrap_or_default()
196        };
197        assert!(!threshold.holds_for_overrides(&profile(None)));
198        assert!(
199            !threshold.holds_for_overrides(&profile(Some(ParameterValue::String("none".into()))))
200        );
201        assert!(threshold.holds_for_overrides(&profile(Some(ParameterValue::Float(3.5)))));
202    }
