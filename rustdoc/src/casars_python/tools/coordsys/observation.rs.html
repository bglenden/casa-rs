3use casa_types::measures::{MPosition, PositionRef, position_to_record};
4use casa_types::{ScalarValue, Value};
5use pyo3::exceptions::{PyRuntimeError, PyTypeError};
6use pyo3::prelude::*;
7
8use crate::tools::{LogLine, emit};
9
10use super::PyCoordsys;
11
12fn text(value: &Bound<'_, PyAny>) -> PyResult<String> {
13    value
14        .extract()
15        .map_err(|_| PyTypeError::new_err("argument _value must be a string"))
16}
17
18#[pymethods]
19impl PyCoordsys {
20    fn setobserver(&self, py: Python<'_>, value: &Bound<'_, PyAny>) -> PyResult<bool> {
21        let observer = text(value)?;
22        py.detach(|| {
23            let mut state = self.state();
24            state.coordinates.obs_info_mut().observer = observer.clone();
25            state
26                .record
27                .upsert("observer", Value::Scalar(ScalarValue::String(observer)));
28        });
29        Ok(true)
30    }
31
32    fn settelescope(&self, py: Python<'_>, value: &Bound<'_, PyAny>) -> PyResult<bool> {
33        let telescope = text(value)?;
34        let found = py.detach(|| {
35            let measures = casa_ms::open_measures_runtime()
36                .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
37            let position = MPosition::from_observatory_name(&telescope, measures.as_ref())
38                .and_then(|position| {
39                    position
40                        .map(|position| position.convert_to(PositionRef::ITRF))
41                        .transpose()
42                })
43                .map_err(|error| PyRuntimeError::new_err(error.to_string()))?;
44            let mut state = self.state();
45            state.coordinates.obs_info_mut().telescope = telescope.clone();
46            state
47                .record
48                .upsert("telescope", Value::Scalar(ScalarValue::String(telescope)));
49            if let Some(position) = position {
50                state
51                    .record
52                    .upsert("telescopeposition", position_to_record(&position).into());
53                state.coordinates.obs_info_mut().telescope_position = Some(position);
54                Ok::<_, PyErr>(true)
55            } else {
56                Ok(false)
57            }
58        })?;
59        if !found {
60            emit(
61                py,
62                "casars.tools.coordsys",
63                &[LogLine::warning(
64                    "This telescope and its position is not known to the casapy system.",
65                )],
66            )?;
67        }
68        Ok(found)
69    }
70}
