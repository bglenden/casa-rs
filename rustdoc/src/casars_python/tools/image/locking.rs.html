13fn write_flag(value: &Bound<'_, PyAny>) -> PyResult<bool> {
14    value
15        .cast::<PyBool>()
16        .map(|flag| flag.is_true())
17        .map_err(|_| PyTypeError::new_err("in method 'image_lock', argument 2 of type 'bool'"))
18}
19
20fn attempts(value: &Bound<'_, PyAny>) -> PyResult<i32> {
21    crate::tools::table::integer(Some(value), "nattempts", 0).map(|value| value as i32)
22}

24impl ImageHandle {
25    pub(super) fn lock_storage(&mut self, kind: LockType, attempts: u32) -> PyResult<bool> {
26        #[cfg(unix)]
27        return with_pixels!(self, storage => match storage {
28            Storage::Paged(image) => image.lock(kind, attempts).map_err(raised),
29            Storage::Temp { image, .. } => image.concat_source().map_or(Ok(true), |source| source.lock(kind, attempts).map_err(raised)),
30        });
31        #[cfg(not(unix))]
32        {
33            let _ = (kind, attempts);
34            Ok(true)
35        }
36    }
37
38    pub(super) fn has_storage_lock(&self, kind: LockType) -> bool {
39        #[cfg(unix)]
40        return with_pixels!(self, storage => match storage {
41            Storage::Paged(image) => image.has_lock(kind),
42            Storage::Temp { image, .. } => image.concat_source().is_none_or(|source| source.has_lock(kind)),
43        });
44        #[cfg(not(unix))]
45        {
46            let _ = kind;
47            true
48        }
49    }
50
51    pub(super) fn unlock_storage(&mut self) -> PyResult<()> {
52        #[cfg(unix)]
53        return with_pixels!(self, storage => match storage {
54            Storage::Paged(image) => image.unlock().map_err(raised),
55            Storage::Temp { image, .. } => image.concat_source().map_or(Ok(()), |source| source.unlock().map_err(raised)),
56        });
57        #[cfg(not(unix))]
58        Ok(())
59    }
60
61    pub(super) fn release_requested_lock(&mut self) -> PyResult<()> {
62        #[cfg(unix)]
63        return with_pixels!(self, storage => match storage {
64            Storage::Paged(image) => image.release_requested_lock().map_err(raised),
65            Storage::Temp { image, .. } => image.concat_source().map_or(Ok(()), |source| source.release_requested_lock().map_err(raised)),
66        });
67        #[cfg(not(unix))]
68        Ok(())
69    }
70}

72impl AttachedImage {
73    pub(super) fn lock_expression_operand(
74        &self,
75        kind: LockType,
76        attempts: u32,
77        section_read: bool,
78    ) -> PyResult<bool> {
79        let mut source = self.lock();
80        self.check_bound_name(&source)?;
81        if section_read
82            && with_pixels!(&*source, storage => match storage {
83                Storage::Temp { image, .. } => image.concat_source().is_some_and(|concat| !concat.temp_close()),
84                Storage::Paged(_) => false,
85            })
86        {
87            // A retained concatenation acquires each member's read lease as
88            // its requested section is evaluated.
89            return Ok(true);
90        }
91        source.lock_storage(kind, attempts)
92    }
93
94    fn lock_with_inputs(
95        &self,
96        handle: &mut ImageHandle,
97        kind: LockType,
98        attempts: u32,
99        section_read: bool,
100    ) -> PyResult<bool> {
101        self.check_bound_name(handle)?;
102        for input in self.expression_inputs() {
103            if !input.lock_expression_operand(kind, attempts, section_read)? {
104                return Ok(false);
105            }
106        }
107        handle.lock_storage(kind, attempts)
108    }
109
110    pub(super) fn require_lock(&self, handle: &mut ImageHandle, kind: LockType) -> PyResult<()> {
111        self.check_bound_name(handle)?;
112        if kind == LockType::Write
113            && with_pixels!(&*handle, storage => matches!(storage, Storage::Paged(image) if !image.is_writable()))
114        {
115            let path = handle
116                .path()
117                .expect("a read-only paged image has a table path");
118            return Err(raised(format!(
119                "Table {} cannot be opened for read/write",
120                path.display()
121            )));
122        }
123        let implicit_concat = with_pixels!(&*handle, storage => match storage {
124            Storage::Temp { image, .. } => image.concat_source().is_some_and(|source|
125                kind == LockType::Read || source.temp_close()),
126            Storage::Paged(_) => false,
127        });
128        if implicit_concat {
129            return Ok(());
130        }
131        if handle.lock_storage(kind, 0)? {
132            Ok(())
133        } else {
134            Err(raised(format!("Could not acquire the image {kind:?} lock")))
135        }
136    }
137
138    pub(super) fn require_pixel_read_lock(&self, handle: &mut ImageHandle) -> PyResult<()> {
139        if handle.image_type_name() == "ImageConcat" {
140            return self.check_bound_name(handle);
141        }
142        if self.lock_with_inputs(handle, LockType::Read, 0, true)? {
143            Ok(())
144        } else {
145            Err(raised("Could not acquire the image Read lock"))
146        }
147    }
148
149    pub(super) fn release_requested_lock(&self, handle: &mut ImageHandle) -> PyResult<()> {
150        for input in self.expression_inputs() {
151            input.lock().release_requested_lock()?;
152        }
153        handle.release_requested_lock()
154    }
155}

157#[pymethods]
158impl PyImageTool {
159    fn haslock<'py>(&self, py: Python<'py>) -> PyResult<Bound<'py, PyArray1<bool>>> {
160        let flags = py.detach(|| {
161            let state = self.state();
162            let Some(attachment) = state.image.as_ref() else {
163                return Ok(Vec::new());
164            };
165            let handle = attachment.lock();
166            handle.check_single_precision("haslock")?;
167            Ok::<_, PyErr>(
168                [LockType::Read, LockType::Write]
169                    .map(|kind| {
170                        handle.has_storage_lock(kind)
171                            && attachment
172                                .expression_inputs()
173                                .iter()
174                                .all(|input| input.lock().has_storage_lock(kind))
175                    })
176                    .to_vec(),
177            )
178        })?;
179        Ok(flags.into_pyarray(py))
180    }
181
182    #[pyo3(signature = (writelock=false, nattempts=0), text_signature = "($self, writelock=False, nattempts=0)")]
183    fn lock(
184        &self,
185        py: Python<'_>,
186        #[pyo3(from_py_with = write_flag)] writelock: bool,
187        #[pyo3(from_py_with = attempts)] nattempts: i32,
188    ) -> PyResult<bool> {
189        py.detach(|| {
190            let state = self.state();
191            let Some(attachment) = state.image.as_ref() else {
192                return Ok(false);
193            };
194            let mut handle = attachment.lock();
195            handle.check_single_precision("lock")?;
196            let kind = if writelock {
197                LockType::Write
198            } else {
199                LockType::Read
200            };
201            attachment.lock_with_inputs(&mut handle, kind, nattempts.max(0) as u32, false)
202        })
203    }
204
205    fn unlock(&self, py: Python<'_>) -> PyResult<bool> {
206        py.detach(|| {
207            let state = self.state();
208            let Some(attachment) = state.image.as_ref() else {
209                return Ok(false);
210            };
211            let mut handle = attachment.lock();
212            handle.check_single_precision("unlock")?;
213            attachment.check_bound_name(&handle)?;
214            for input in attachment.expression_inputs() {
215                let mut source = input.lock();
216                input.check_bound_name(&source)?;
217                source.unlock_storage()?;
218            }
219            handle.unlock_storage()?;
220            Ok(true)
221        })
222    }
