21/// The smallest half-size `simobserve` gives a component-only model, deg
22/// (about 5 arcsec).
23const MINIMUM_OFFSET_DEG: f64 = 0.0014;
24
25/// What `simobserve` takes from a component list for a model with no image:
26/// the list's average direction, the first component's reference frequency,
27/// the channels it asks for, and a size three times the farthest component's
28/// offset.
29#[derive(Debug, Clone, PartialEq)]
30pub struct ComponentModel {
31    /// The components' average direction, as `simutil` writes directions.
32    pub refdir: String,
33    /// The first component's reference frequency, in GHz as the component
34    /// list's spectrum record gives it.
35    pub specrefval: Quantity,
36    /// The channel width: `compwidth` (2 GHz when unset) over `comp_nchan`.
37    pub width: Quantity,
38    /// The number of channels, `comp_nchan`.
39    pub nchan: usize,
40    /// The reference channel, the middle one: `(comp_nchan − 1)/2`.
41    pub specrefpix: f64,
42    /// The Stokes planes, `"I"`.
43    pub stokes: String,
44    /// The size, `"%fdeg"` of three times the farthest component's offset
45    /// (at least 0.0014°) along either axis, for both axes.
46    pub size: [String; 2],
47    /// The whole bandwidth, `compwidth` or `"2GHz"`, which the
48    /// `.compskymodel` takes as its spectral increment.
49    pub bandwidth: String,
50}
51
52impl ComponentModel {
53    /// The model `simobserve` makes of `list` with `compwidth` and
54    /// `comp_nchan`.
55    ///
56    /// # Errors
57    ///
58    /// An empty list, or a component direction or frequency the `simutil`
59    /// helpers cannot write.
60    pub fn from_list(
61        list: &ComponentList,
62        compwidth: &str,
63        comp_nchan: usize,
64    ) -> Result<Self, SkyModelError> {
65        let first = list.get(0).ok_or_else(|| {
66            SkyModelError::Model("the component list holds no components".to_string())
67        })?;
68        let directions = list
69            .iter()
70            .map(|component| {
71                let reference = component.shape().reference_direction();
72                let [x, y, z] = reference.cosines();
73                let longitude = if x != 0.0 || y != 0.0 {
74                    y.atan2(x)
75                } else {
76                    0.0
77                };
78                let radians = |value: f64| Quantity::with_unit(value, unit("rad"));
79                direction::format_direction(
80                    reference.refer().as_str(),
81                    &radians(longitude),
82                    &radians(z.asin()),
83                )
84            })
85            .collect::<Result<Vec<_>, _>>()?;
86        let centred = direction::average_direction(&directions)?;
87        let hz = Quantity::with_unit(first.spectrum().reference_frequency().hz(), unit("Hz"));
88        let specrefval = qa::convert(&hz, "GHz")?;
89        let bandwidth = if qa::is_quantity(compwidth) {
90            tracing::info!("compwidth set: setting model bandwidth to input");
91            compwidth.to_string()
92        } else {
93            tracing::warn!("compwidth unset: setting bandwidth to 2GHz");
94            "2GHz".to_string()
95        };
96        let width = qa::div(&qa::read(&bandwidth)?, &qa::number(comp_nchan as f64))?;
97        let farthest = centred
98            .offsets
99            .iter()
100            .flat_map(|offset| offset.map(f64::abs))
101            .fold(MINIMUM_OFFSET_DEG, f64::max);
102        let size = format!("{:.6}deg", 3.0 * farthest);
103        Ok(Self {
104            refdir: centred.centre,
105            specrefval,
106            width,
107            nchan: comp_nchan,
108            specrefpix: 0.5 * (comp_nchan as f64 - 1.0),
109            stokes: "I".to_string(),
110            size: [size.clone(), size],
111            bandwidth,
112        })
113    }
114
115    /// The `.compskymodel` image's size in pixels for `cell`:
116    /// `int(size / cell)`, both in arcsec.
117    ///
118    /// # Errors
119    ///
120    /// A size or cell that is not an angle.
121    pub fn image_size(&self, cell: &Quantity) -> Result<usize, SkyModelError> {
122        let size = qa::value_in(&qa::read(&self.size[0])?, "arcsec")?;
123        let cell = qa::value_in(cell, "arcsec")?;
124        Ok((size / cell).trunc() as usize)
125    }
126
127    /// Write the `.compskymodel` at `path`, replacing what is there: a
128    /// square image of [`Self::image_size`] pixels of `cell` centred on
129    /// [`Self::refdir`], one channel and Stokes I, in Jy/pixel, with the
130    /// components of `list` added; then its flat image, `<path>.flat`, whose
131    /// path is returned.
132    ///
133    /// The direction is set as `simobserve` sets it, from `qa.tos` of the
134    /// split direction (nine decimals) read back; the spectral axis is
135    /// LSRK at [`Self::specrefval`], channel [`Self::specrefpix`], with the
136    /// whole bandwidth as its increment.
137    ///
138    /// # Errors
139    ///
140    /// Those of [`add_components`] and [`flat_image`], and values the
141    /// `simutil` helpers cannot convert.
142    pub fn write(
143        &self,
144        list: &ComponentList,
145        cell: &[Quantity; 2],
146        path: &Path,
147    ) -> Result<PathBuf, SkyModelError> {
148        let size = self.image_size(&cell[0])?;
149        let split = direction::split_direction(&self.refdir)?;
150        let refer = split
151            .epoch
152            .trim()
153            .parse()
154            .unwrap_or(casa_types::measures::direction::DirectionRef::J2000);
155        let reread = |value: &Quantity| -> Result<f64, SimutilError> {
156            qa::value_in(&qa::read(&tos(value, 9))?, "rad")
157        };
158        let crval = [reread(&split.longitude)?, reread(&split.latitude)?];
159        let cdelt = [
160            -qa::value_in(&cell[0], "rad")?,
161            qa::value_in(&cell[1], "rad")?,
162        ];
163        let centre = size as f64 / 2.0;
164        let hz = |value: &Quantity| qa::value_in(value, "Hz");
165        let bandwidth = hz(&qa::read(&self.bandwidth)?)?;
166        check_increments(&[cdelt[0], cdelt[1], bandwidth])?;
167        let mut coordinates = CoordinateSystem::new().with_obs_info(shape_obs_info());
168        coordinates.add_coordinate(DirectionCoordinate::new(
169            refer,
170            Projection::with_parameters(ProjectionType::SIN, vec![0.0, 0.0]),
171            crval,
172            cdelt,
173            [centre, centre],
174        ));
175        coordinates.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
176        coordinates.add_coordinate(SpectralCoordinate::new(
177            FrequencyRef::LSRK,
178            hz(&self.specrefval)?,
179            bandwidth,
180            self.specrefpix,
181            hz(&self.specrefval)?,
182        ));
183        write_image(path, &zeros(&[size, size, 1, 1]), coordinates, "Jy/pixel")?;
184        add_components(path, list)?;
185        flat_image(path, None)
186    }
187}

189/// casacore's unit `name`, which is known.
190fn unit(name: &str) -> casa_types::quanta::Unit {
191    casacore_unit(name).expect("a unit casacore knows")
