29pub(super) struct FitsAttachment {
30    path: PathBuf,
31    hdu: usize,
32    pub(super) shape: Vec<usize>,
33    pub(super) tile_shape: Vec<usize>,
34    pub(super) has_mask: bool,
35    pixels: OnceCell<FitsSectionReader>,
36}
37
38impl FitsAttachment {
39    pub(super) fn image_view<'a>(
40        &'a self,
41        metadata: &'a TempImage<f32>,
42    ) -> impl casa_images::ImageInterface<f32> + 'a {
43        fits_section_image(&self.shape, metadata, &self.path, |start, count, stride| {
44            self.read_pixels(start, count, stride)
45        })
46    }
47
48    pub(super) fn read(
49        &self,
50        start: &[usize],
51        count: &[usize],
52        stride: &[usize],
53    ) -> PyResult<ArrayD<f32>> {
54        self.read_pixels(start, count, stride).map_err(raised)
55    }
56
57    fn read_pixels(
58        &self,
59        start: &[usize],
60        count: &[usize],
61        stride: &[usize],
62    ) -> Result<ArrayD<f32>, ImageError> {
63        // CASA opens the pixel stream on the first read and retains it
64        // across subsequent reads even if the filename is moved or removed.
65        if let Some(reader) = self.pixels.get() {
66            return reader.read(start, count, stride);
67        }
68        let reader = FitsSectionReader::open(&self.path, self.hdu, &self.shape)?;
69        let section = reader.read(start, count, stride)?;
70        let _ = self.pixels.set(reader);
71        Ok(section)
72    }
73}

75pub(super) enum ImageHandle {
76    Float(Storage<f32>),
77    Double(Storage<f64>),
78    Complex(Storage<Complex32>),
79    DComplex(Storage<Complex64>),
80}

82// Each arm retains the concrete pixel type for calls into the image library.
83macro_rules! with_pixels {
84    ($image:expr, $storage:ident => $work:expr) => {
85        match $image {
86            ImageHandle::Float($storage) => $work,
87            ImageHandle::Double($storage) => $work,
88            ImageHandle::Complex($storage) => $work,
89            ImageHandle::DComplex($storage) => $work,
90        }
91    };
92}
93pub(super) use with_pixels;
94
95macro_rules! with_storage {
96    ($storage:expr, $image:ident => $work:expr) => {
97        match $storage {
98            Storage::Paged($image) => $work,
99            Storage::Temp { image: $image, .. } => $work,
100        }
101    };
102}
103pub(super) use with_storage;

109pub(super) fn same_file_identity(expected: &Metadata, actual: &Metadata) -> bool {
110    if expected.file_type() != actual.file_type() {
111        return false;
112    }
113    #[cfg(unix)]
114    {
115        use std::os::unix::fs::MetadataExt;
116        expected.dev() == actual.dev() && expected.ino() == actual.ino()
117    }
118    #[cfg(not(unix))]
119    {
120        expected.created().ok() == actual.created().ok()
121    }
122}
123
124pub(super) fn same_file_snapshot(expected: &Metadata, actual: &Metadata) -> bool {
125    same_file_identity(expected, actual)
126        && expected.len() == actual.len()
127        && expected.modified().ok() == actual.modified().ok()
128}

130impl ImageHandle {
131    pub(super) fn open(path: &Path) -> Result<Self, ImageError> {
132        if path.is_file() {
133            return Self::open_fits(path, FitsToolImport::open_attachment(path)?);
134        }
135        Self::open_native(path, None)
136    }
137
138    pub(super) fn open_read(path: &Path) -> Result<Self, ImageError> {
139        if path.is_file() {
140            return Self::open(path);
141        }
142        Self::open_native(path, Some(0))
143    }
144
145    pub(super) fn open_native(path: &Path, lock_attempts: Option<u32>) -> Result<Self, ImageError> {
146        Self::open_native_with_cache(path, lock_attempts, true)
147    }
148
149    pub(super) fn open_native_with_cache(
150        path: &Path,
151        lock_attempts: Option<u32>,
152        component_cache: bool,
153    ) -> Result<Self, ImageError> {
154        if casa_images::concat_file::is_image_concat(path) {
155            return super::concatenate::virtual_image::open_handle(path);
156        }
157        if casa_images::ComponentListImage::recognizes_native(path) {
158            let (source, record) =
159                casa_images::ComponentListImage::open_native(path, component_cache)?;
160            return Self::from_component(source, record);
161        }
162        #[cfg(unix)]
163        if let Some(attempts) = lock_attempts {
164            return AnyPagedImage::open_with_lock(path, attempts).map(Self::from_paged);
165        }
166        #[cfg(not(unix))]
167        let _ = lock_attempts;
168        AnyPagedImage::open(path).map(Self::from_paged)
169    }
170
171    pub(super) fn from_component(
172        source: casa_images::ComponentListImage,
173        record: RecordValue,
174    ) -> Result<Self, ImageError> {
175        let image = TempImage::from_read_only_image(
176            source,
177            "ComponentListImage",
178            "ComponentListImage::doPutSlice: pixel values cannot be modified in a ComponentListImage",
179        )?;
180        Ok(Self::Float(Storage::Temp {
181            image: Box::new(image),
182            record,
183            fits: None,
184        }))
185    }
186
187    pub(super) fn disable_component_cache(&mut self) -> Result<(), ImageError> {
188        if let Self::Float(Storage::Temp { image, .. }) = self {
189            image.set_component_cache(false)?;
190        }
191        Ok(())
192    }
193
194    fn from_paged(image: AnyPagedImage) -> Self {
195        match image {
196            AnyPagedImage::Float32(image) => Self::Float(Storage::Paged(Box::new(image))),
197            AnyPagedImage::Float64(image) => Self::Double(Storage::Paged(Box::new(image))),
198            AnyPagedImage::Complex32(image) => Self::Complex(Storage::Paged(Box::new(image))),
199            AnyPagedImage::Complex64(image) => Self::DComplex(Storage::Paged(Box::new(image))),
200        }
201    }
202
203    pub(super) fn open_for_concatenation(path: &Path) -> Result<Self, ImageError> {
204        if path.is_file() {
205            return Self::open_fits(path, FitsToolImport::open_for_concatenation(path)?);
206        }
207        if Self::recognizes_miriad(path, path) {
208            return Self::open_miriad(path);
209        }
210        Self::open_native(path, Some(0))
211    }
212
213    pub(super) fn open_fits(path: &Path, mut source: FitsToolImport) -> Result<Self, ImageError> {
214        if source.is_tile_compressed() {
215            return Err(ImageError::CasaSelection(format!(
216                "No data in the zeroth or first extension of {}",
217                path.display()
218            )));
219        }
220        // Reuse temporary-image metadata operations. The attachment owns the
221        // actual shape and pixel stream, so this needs only one storage cell.
222        let mut image = TempImage::new(
223            vec![1; source.shape().len()],
224            source.coordinates().clone(),
225            TempStoragePolicy::Memory,
226        )?;
227        let fits = FitsAttachment {
228            path: path.to_path_buf(),
229            hdu: source.hdu_index(),
230            shape: source.shape().to_vec(),
231            tile_shape: source.tile_shape(),
232            has_mask: source.has_implicit_mask(),
233            pixels: OnceCell::new(),
234        };
235        source.misc_info.remove("object");
236        image.set_units(source.units)?;
237        image.set_image_info_record(source.image_info)?;
238        image.set_misc_info(source.misc_info)?;
239        image.append_history_entries(&source.history)?;
240        Ok(Self::Float(Storage::Temp {
241            image: Box::new(image),
242            record: source.coordinate_record,
243            fits: Some(fits),
244        }))
245    }
246
247    pub(super) fn fits(&self) -> Option<&FitsAttachment> {
248        match self {
249            Self::Float(Storage::Temp { fits, .. }) => fits.as_ref(),
250            _ => None,
251        }
252    }
253
254    pub(super) fn recognizes_miriad(probe: &Path, resolved: &Path) -> bool {
255        MiriadImage::recognizes_input(probe, resolved)
256    }
257
258    pub(super) fn has_native_control(path: &Path) -> bool {
259        path.join("table.dat").is_file() && !Self::recognizes_miriad(path, path)
260    }
261
262    pub(super) fn open_miriad(path: &Path) -> Result<Self, ImageError> {
263        let image = TempImage::from_read_only_image(
264            MiriadImage::open(path)?,
265            "MIRIADImage",
266            "MIRIADImage::putSlice - is not possible yet as MIRIADImage is not writable",
267        )?;
268        Ok(Self::Float(Storage::Temp {
269            record: image.coordinates().to_record(),
270            image: Box::new(image),
271            fits: None,
272        }))
273    }
274
275    pub(super) fn image_type_name(&self) -> &'static str {
276        if self.fits().is_some() {
277            "FITSImage"
278        } else {
279            with_pixels!(self, storage => with_storage!(storage, image => image.image_type_name()))
280        }
281    }
282
283    pub(super) fn read_only_file_type(&self) -> Option<&'static str> {
284        let kind = self.image_type_name();
285        (matches!(kind, "FITSImage" | "MIRIADImage") || self.is_saved_expression()).then_some(kind)
286    }
287
288    pub(super) fn is_saved_expression(&self) -> bool {
289        self.image_type_name() == "ImageExpr" && self.path().is_some()
290    }
291
292    pub(super) fn concat_expression_operands(&self) -> Vec<PathBuf> {
293        with_pixels!(self, storage => match storage {
294            Storage::Temp { image, .. } => image.concat_source()
295                .map_or_else(Vec::new, |source| source.expression_operands().cloned().collect()),
296            Storage::Paged(_) => Vec::new(),
297        })
298    }
299
300    pub(super) fn concat_operands(&self) -> Vec<PathBuf> {
301        with_pixels!(self, storage => match storage {
302            Storage::Temp { image, .. } => image.concat_source()
303                .map_or_else(Vec::new, |source| source.operand_paths().map(Path::to_path_buf).collect()),
304            Storage::Paged(_) => Vec::new(),
305        })
306    }

308    pub(super) fn kind(&self) -> ImagePixelType {
309        match self {
310            Self::Float(_) => ImagePixelType::Float32,
311            Self::Double(_) => ImagePixelType::Float64,
312            Self::Complex(_) => ImagePixelType::Complex32,
313            Self::DComplex(_) => ImagePixelType::Complex64,
314        }
315    }

317    pub(super) fn shape(&self) -> &[usize] {
318        if let Some(fits) = self.fits() {
319            return &fits.shape;
320        }
321        with_pixels!(self, storage => with_storage!(storage, image => image.shape()))
322    }
323
324    pub(super) fn coordinates(&self) -> &CoordinateSystem {
325        with_pixels!(self, storage => with_storage!(storage, image => image.coordinates()))
326    }
327
328    pub(super) fn path(&self) -> Option<&Path> {
329        if let Some(fits) = self.fits() {
330            return Some(&fits.path);
331        }
332        with_pixels!(self, storage => with_storage!(storage, image => image.name()))
333    }

335    pub(super) fn save(&mut self) -> Result<(), ImageError> {
336        with_pixels!(self, storage => storage.save())
337    }

339    pub(super) fn history_entries(&self) -> PyResult<Vec<casa_images::image::ImageHistoryRow>> {
340        with_pixels!(self, storage => match storage {
341            Storage::Paged(image) => casa_images::image::image_history_rows(image.name().expect("native image path")).map_err(raised),
342            Storage::Temp { image, .. } => Ok(image.history_rows().map_err(raised)?.into_owned()),
343        })
344    }

346    pub(super) fn append_history_entries(
347        &mut self,
348        entries: &[casa_images::image::ImageHistoryRow],
349    ) -> PyResult<()> {
350        with_pixels!(self, storage => with_storage!(storage, image => image.append_history_entries(entries))).map_err(raised)
351    }

353    pub(super) fn append_history_at(
354        &mut self,
355        location: &str,
356        messages: &[String],
357    ) -> PyResult<()> {
358        with_pixels!(self, storage => with_storage!(storage, image => image.append_history_at(location, messages)))
359            .map_err(raised)
360    }

362    pub(super) fn record_history(&mut self, method: &str, command: String) -> PyResult<()> {
363        self.append_history_at(
364            &format!("::image::{method}"),
365            &[format!("Ran ia.{method}"), command],

369    pub(super) fn coordinate_record(&self) -> RecordValue {
370        with_pixels!(self, storage => match storage {
371            Storage::Paged(image) => match image.table().keywords().get("coords") {
372                Some(Value::Record(record)) => record.clone(),
373                _ => image.coordinates().to_record(),
374            },
375            Storage::Temp { record, .. } => record.clone(),
376        })
377    }
378
379    pub(super) fn set_coordinate_record(&mut self, mut record: RecordValue) -> PyResult<()> {
380        if let Self::Float(Storage::Temp {
381            image,
382            record: current,
383            ..
384        }) = self
385            && image.image_type_name() == "ComponentListImage"
386        {
387            let coordinates = CoordinateSystem::from_record(&record).map_err(raised)?;
388            image
389                .set_component_coordinates(coordinates, &record)
390                .map_err(raised)?;
391            *current = record;
392            return Ok(());
393        }
394        with_pixels!(self, storage => match storage {
395            Storage::Paged(image) => image.set_coordinate_record(record).map_err(raised),
396            Storage::Temp { image, record: current, .. } => {
397                let coordinates = CoordinateSystem::from_record(&record).map_err(raised)?;
398                coordinates.normalize_quality_records(&mut record).map_err(raised)?;
399                image.set_coordinates(coordinates).map_err(raised)?;
400                *current = record;
401                Ok(())
402            },
403        })
404    }
405
406    pub(super) fn check_single_precision(&self, method: &str) -> PyResult<()> {
407        let name = match self.kind() {
408            ImagePixelType::Float64 => "double",
409            ImagePixelType::Complex64 => "complex<double>",
410            _ => return Ok(()),
411        };
412        Err(raised(format!(
413            "Exception: {method} does not support images with {name} precision pixel values."
414        )))
415    }

418impl<T: ImagePixel> Storage<T> {
419    pub(super) fn create(
420        shape: Vec<usize>,
421        coordinates: CoordinateSystem,
422        path: Option<&Path>,
423        data: Option<&ArrayD<T>>,
424    ) -> Result<Self, ImageError> {
425        let mut storage = match path {
426            Some(path) => Self::Paged(Box::new(PagedImage::create(shape, coordinates, path)?)),
427            None => Self::Temp {
428                fits: None,
429                record: coordinates.to_record(),
430                image: Box::new(TempImage::new(
431                    shape,
432                    coordinates,
433                    TempStoragePolicy::Auto {
434                        memory_budget_bytes: 256 * 1024 * 1024,
435                        scratch: ScratchSpace::SystemTemp,
436                    },
437                )?),
438            },
439        };
440        if let Some(data) = data {
441            let origin = vec![0; data.ndim()];
442            with_storage!(&mut storage, image => image.put_slice(data, &origin))?;
443        }
444        Ok(storage)
445    }
446
447    pub(super) fn save(&mut self) -> Result<(), ImageError> {
448        match self {
449            Self::Paged(image) => image.save(),
450            Self::Temp { .. } => Ok(()),
451        }
452    }
453}

455impl Storage<f32> {
456    pub(super) fn with_float_view<R>(
457        &self,
458        use_image: impl FnOnce(&dyn casa_images::ImageInterface<f32>) -> R,
459    ) -> R {
460        match self {
461            Self::Paged(image) => use_image(image.as_ref()),
462            Self::Temp {
463                image,
464                fits: Some(fits),
465                ..
466            } => use_image(&fits.image_view(image)),
467            Self::Temp { image, .. } => use_image(image.as_ref()),
468        }
469    }
