72impl Plan {
73    pub(super) fn new(
74        input: &MeasurementSet,
75        options: &MsRegridOptions,
76        combined: Option<&mut combine::Grid>,
77        channel_selection: &BTreeMap<i32, TransformChannelSelection>,
78        channel_bins: &mut BTreeMap<i32, Vec<Vec<usize>>>,
79        first_selected_row: usize,
80        budget_bytes: usize,
81    ) -> Result<Self, MsTransformError> {
82        let method = Method::parse(&options.interpolation).ok_or_else(|| {
83            invalid(format!(
84                "Invalid interpolation {:?}.",
85                options.interpolation
86            ))
87        })?;
88        let frame = Frame::new(input, options, first_selected_row)?;
89        let mut budget = GridBudget::new(budget_bytes, frame.is_some());
90        let table = input.spectral_window().map_err(ms_error)?;
91        let is_combined = combined.is_some();
92        let mut windows = BTreeMap::new();
93        if let Some(grid) = combined {
94            let id = *grid.windows.keys().next().expect("nonempty combined grid");
95            let source = table
96                .reference_frequency_reference(id as usize)
97                .map_err(ms_error)?;
98            let centres = grid
99                .channels
100                .iter()
101                .map(|channel| channel.frequency)
102                .collect::<Vec<_>>();
103            let widths = grid
104                .channels
105                .iter()
106                .map(|channel| channel.width)
107                .collect::<Vec<_>>();
108            let window = Window::new(
109                options,
110                method,
111                (&centres, &widths, source),
112                &mut grid.bins,
113                frame.as_ref(),
114                &mut budget,
115            )?;
116            windows.insert(id, window);
117        } else {
118            for (&id, bins) in channel_bins {
119                let row = id as usize;
120                let source = table.reference_frequency_reference(row).map_err(ms_error)?;
121                let mut centres = table.chan_freq(row).map_err(ms_error)?;
122                let mut widths = table.chan_width(row).map_err(ms_error)?;
123                let selection = &channel_selection[&id];
124                if selection.channels != selection.metadata_ends {
125                    let footprints =
126                        spectral_metadata::selected_footprints(selection, &centres, &widths)
127                            .ok_or_else(|| {
128                                invalid(format!(
129                                    "SPW {id} metadata does not cover the selected channels."
130                                ))
131                            })?;
132                    for (&channel, (centre, width)) in selection
133                        .channels
134                        .iter()
135                        .zip(footprints.0.into_iter().zip(footprints.1))
136                    {
137                        centres[channel] = centre;
138                        widths[channel] = width;
139                    }
140                }
141                let window = Window::new(
142                    options,
143                    method,
144                    (&centres, &widths, source),
145                    bins,
146                    frame.as_ref(),
147                    &mut budget,
148                )?;
149                windows.insert(id, window);
150            }
151        }
152        Ok(Self {
153            windows,
154            frame,
155            combined: is_combined,
156        })
157    }

159    fn window_id(&self, spw: i32) -> i32 {
160        if self.combined {

162        } else {
163            spw

176        let first = &self.windows.values().next()?.partitions;

180    pub(super) fn row(
181        &mut self,
182        spw: i32,
183        input: &Table,
184        row: usize,
185    ) -> Result<Projection<'_>, MsTransformError> {
186        let id = self.window_id(spw);
187        let window = self.windows.get_mut(&id).expect("selected window");
188        let (interpolation, shift) = if let Some(frame) = &self.frame {
189            let time = crate::subtables::get_f64(input, row, "TIME").map_err(ms_error)?;
190            let field =
191                crate::subtables::get_i32(input, row, "FIELD_ID").map_err(ms_error)? as usize;
192            if !window
193                .recent
194                .as_ref()
195                .is_some_and(|recent| time == recent.time && field == recent.field)
196            {
197                let convert = frame.row_conversion(window.frame, time, field)?;
198                let mut recent = Recent {
199                    time,
200                    field,
201                    interpolation: None,
202                    shift: None,
203                };
204                if window.method == Method::FftShift {
205                    let output = &window.axis.centres_hz;
206                    if output.len() > 1 {
207                        let centre = window.frequencies[window.frequencies.len() / 2];
208                        let bandwidth =
209                            output[output.len() - 1] - output[0] + output[1] - output[0];
210                        let offset = convert(centre) - window.reference_centre_hz;
211                        recent.shift = Shift::new(-offset / bandwidth).map_err(ms_error)?;
212                    }
213                } else {
214                    let grid = window
215                        .frequencies
216                        .iter()
217                        .copied()
218                        .map(convert)
219                        .collect::<Vec<_>>();
220                    recent.interpolation = Some(
221                        Interpolation::plan(&grid, &window.sample_grid, window.method)
222                            .map_err(ms_error)?,
223                    );
224                }
225                window.recent = Some(recent);
226            }
227            let recent = window.recent.as_ref().expect("cached grid");
228            (
229                recent
230                    .interpolation
231                    .as_ref()
232                    .unwrap_or(&window.interpolation),
233                recent.shift,
234            )
235        } else {
236            (&window.interpolation, None)
237        };
238        Ok(Projection {
239            interpolation,
240            shift,
241            channel_map: window.channel_map.as_deref(),
242            output_len: window.axis.centres_hz.len(),
243            noise_scale: window.noise_scale,
244        })
245    }

247    pub(super) fn write_metadata(
248        &self,
249        request: &MsTransformRequest,
250        metadata: &CompactSpectralMetadata,
251    ) -> Result<(), MsTransformError> {
252        let path = request.output_ms.join("SPECTRAL_WINDOW");
253        let error = |source| MsTransformError::MutateMeasurementSet {
254            path: path.display().to_string(),
255            source: Box::new(source),
256        };
257        let mut table = Table::open(TableOptions::new(&path)).map_err(error)?;
258        if let Some(parts) = self.split() {
259            let template = table.row_accessor().row(0).map_err(error)?.clone();
260            for _ in 1..parts.count() {
261                table.add_row(template.clone()).map_err(error)?;
262            }
263            parts.write_descriptions(request)?;
264            parts.write_window_references(request)?;
265        }
266        for (&original, window) in &self.windows {
267            let row = if self.combined {
268                0
269            } else {
270                metadata.spw_map[&original] as usize
271            };
272            let reference = self
273                .frame
274                .as_ref()
275                .map_or(window.frame, |frame| frame.metadata_reference);
276            for (part, range) in window.partitions.ranges().iter().enumerate() {
277                let centres = &window.axis.centres_hz[range.clone()];
278                let widths = &window.axis.steps_hz[range.clone()];
279                let signed_widths =
280                    if window.axis.centres_hz.first() < window.axis.centres_hz.last() {
281                        widths.to_vec()
282                    } else {
283                        widths.iter().map(|width| -width).collect()
284                    };
285                for (name, values) in [
286                    ("CHAN_FREQ", centres),
287                    ("CHAN_WIDTH", signed_widths.as_slice()),
288                    ("EFFECTIVE_BW", signed_widths.as_slice()),
289                    ("RESOLUTION", signed_widths.as_slice()),
290                ] {
291                    let value = ArrayValue::Float64(Array1::from_vec(values.to_vec()).into_dyn());
292                    table
293                        .column_accessor_mut(name)
294                        .and_then(|mut col| col.set(row + part, Value::Array(value)))
295                        .map_err(error)?;
296                }
297                for (name, value) in [
298                    ("NUM_CHAN", ScalarValue::Int32(centres.len() as i32)),
299                    ("REF_FREQUENCY", ScalarValue::Float64(centres[0])),
300                    (
301                        "TOTAL_BANDWIDTH",
302                        ScalarValue::Float64(
303                            (centres[centres.len() - 1] - centres[0]).abs()
304                                + (widths[0] + widths[widths.len() - 1]) / 2.,
305                        ),
306                    ),
307                    (
308                        "MEAS_FREQ_REF",
309                        ScalarValue::Int32(reference.casacore_code()),
310                    ),
311                ] {
312                    table
313                        .column_accessor_mut(name)
314                        .and_then(|mut col| col.set(row + part, Value::Scalar(value)))
315                        .map_err(error)?;
316                }
317            }
318        }
319        table
320            .prepare_write()
321            .save(TableOptions::new(&path))
322            .map_err(error)
323    }

326struct GridBudget {
327    remaining: usize,
328    bytes_per_channel: usize,
329}
330
331impl GridBudget {
332    fn new(bytes: usize, dynamic: bool) -> Self {
333        // Allow for retained centre/width arrays, interpolation records and
334        // temporary grid construction. A dynamic frame also retains a
335        // second interpolation plan for the most recent visibility buffer.
336        Self {
337            remaining: bytes,
338            bytes_per_channel: if dynamic { 256 } else { 128 },
339        }
340    }
341
342    fn maximum_output_channels(&self, input: usize) -> Result<usize, MsTransformError> {
343        let count = (self.remaining / self.bytes_per_channel)
344            .saturating_sub(input)
345            .min(i32::MAX as usize);
346        if count == 0 {
347            Err(invalid(
348                "Input regridding channels exceed the remaining grid budget.".into(),
349            ))
350        } else {
351            Ok(count)
352        }
353    }
354
355    fn charge(&mut self, input: usize, output: usize) -> Result<(), MsTransformError> {
356        let bytes = input
357            .saturating_add(output)
358            .saturating_mul(self.bytes_per_channel);
359        self.remaining = self.remaining.checked_sub(bytes).ok_or_else(|| {
360            invalid(format!(
361                "Regridding {output} channels exceeds the {} byte grid budget.",
362                self.remaining
363            ))
364        })?;
365        Ok(())
366    }
367}

369impl Window {
370    fn new(
371        options: &MsRegridOptions,
372        method: Method,
373        (source_frequencies, source_widths, frame): (&[f64], &[f64], FrequencyRef),
374        bins: &mut Vec<Vec<usize>>,
375        conversion: Option<&Frame>,
376        budget: &mut GridBudget,
377    ) -> Result<Self, MsTransformError> {
378        let needs_weight_scaling =
379            conversion.is_some_and(|target| frame != target.target_reference);
380        let converted_frame = conversion
381            .map(|target| target.prepare(frame, &target.target))
382            .transpose()?;
383        let source_shift = conversion
384            .and_then(|target| target.source_velocity)
385            .filter(|velocity| velocity.abs() > MIN_SOURCE_VELOCITY_M_PER_S)
386            .map(|velocity| radial_shift(-velocity));
387        let into_frame = |frequency| {
388            converted_frame
389                .as_ref()
390                .map_or(frequency, |convert| convert.convert_hz(frequency))
391        };
392        let shifted = |frequency| {
393            source_shift
394                .as_ref()
395                .map_or(frequency, |shift| shift.shift_frequency_hz(frequency))
396        };
397        let convert = |frequency| shifted(into_frame(frequency));
398        let (mut frequencies, widths) = averaged_axis(source_frequencies, source_widths, bins);
399        let converted = frequencies.iter().copied().map(convert).collect::<Vec<_>>();
400        let converted_widths = frequencies
401            .iter()
402            .zip(widths.iter())
403            .map(|(&f, &width)| {
404                if converted_frame.is_some() {
405                    // CASA converts the two edges into the frame, differences
406                    // them, then Doppler-scales the width as an interval.
407                    shifted(into_frame(f + width / 2.) - into_frame(f - width / 2.))
408                } else {
409                    width
410                }
411            })
412            .collect::<Vec<_>>();
413        let maximum_channels = budget.maximum_output_channels(frequencies.len())?;
414        let axis = axis::output_axis(options, &converted, &converted_widths, maximum_channels)?;
415        let count = axis.centres_hz.len();
416        budget.charge(frequencies.len(), count)?;
417        let mut noise_scale = if needs_weight_scaling {
418            (axis.steps_hz[0] / converted_widths[0]) * bins[0].len() as f64
419        } else {
420            1.
421        };
422        let partitions = Partitions::new(count, options.nspw)?;
423        let mean =
424            |values: &[f64]| values.iter().map(|v| v.abs()).sum::<f64>() / values.len() as f64;
425        let ratio = mean(&axis.steps_hz) / mean(&converted_widths);
426        let pre_bin = (ratio + 0.001).floor() as usize;
427        let mut sample_grid = axis.centres_hz.clone();
428        let mut channel_map = None;
429        if bins.iter().all(|bin| bin.len() == 1) && pre_bin >= 2 && pre_bin <= frequencies.len() / 2
430        {
431            if options.preaverage {
432                let selected = bins.iter().flatten().copied().collect::<Vec<_>>();
433                *bins = selected.chunks(pre_bin).map(<[usize]>::to_vec).collect();
434                (frequencies, _) = averaged_axis(source_frequencies, source_widths, bins);
435                if needs_weight_scaling {
436                    noise_scale /= pre_bin as f64;
437                }
438            } else {
439                (sample_grid, channel_map) = fine_grid(&axis, frequencies.len());
440            }
441        }
442        let reference_frequencies = if method == Method::FftShift {
443            frequencies
444                .iter()
445                .copied()
446                .map(into_frame)
447                .collect::<Vec<_>>()
448        } else {
449            frequencies.iter().copied().map(convert).collect::<Vec<_>>()
450        };
451        let reference_centre_hz = reference_frequencies[reference_frequencies.len() / 2];
452        let interpolation =
453            Interpolation::plan(&reference_frequencies, &sample_grid, method).map_err(ms_error)?;
454        Ok(Self {
455            frequencies,
456            axis,
457            partitions,
458            frame,
459            sample_grid,
460            channel_map,
461            interpolation,
462            method,
463            recent: None,
464            reference_centre_hz,
465            noise_scale: noise_scale as f32,
466        })
467    }
468}

470fn averaged_axis(frequencies: &[f64], widths: &[f64], bins: &[Vec<usize>]) -> (Vec<f64>, Vec<f64>) {
471    bins.iter()
472        .map(|bin| {
473            (
474                bin.iter().map(|&i| frequencies[i]).sum::<f64>() / bin.len() as f64,
475                bin.iter().map(|&i| widths[i].abs()).sum::<f64>(),
476            )
477        })
478        .unzip()
479}

481fn fine_grid(
482    axis: &CasaChannelFrequencies,
483    input_count: usize,
484) -> (Vec<f64>, Option<Vec<Option<usize>>>) {
485    let centres = &axis.centres_hz;
486    let widths = &axis.steps_hz;
487    let start = if centres.first() > centres.last() {
488        centres.len() - 1
489    } else {
490        0
491    };
492    let count_ratio = input_count as f64 / centres.len() as f64;
493    let mut frequencies = Vec::with_capacity(input_count);
494    let mut frequency = centres[start] - widths[start] / 2.;
495    frequencies.push(frequency);
496    for index in 1..input_count {
497        let width = widths[(index as f64 / count_ratio) as usize] / count_ratio;
498        frequency += width;
499        frequencies.push(frequency);
500    }
501    let descending = centres.first() > centres.last();
502    let ordered = |index| {
503        if descending {
504            centres.len() - index - 1
505        } else {
506            index
507        }
508    };
509    let map = frequencies
510        .iter()
511        .map(|&frequency| {
512            let mut low = 0;
513            let mut high = centres.len();
514            while low < high {
515                let mid = (low + high) / 2;
516                let i = ordered(mid);
517                if centres[i] + widths[i] / 2. <= frequency {
518                    low = mid + 1;
519                } else {
520                    high = mid;
521                }
522            }
523            if low == centres.len() {
524                return None;
525            }
526            let index = ordered(low);
527            (frequency >= centres[index] - widths[index] / 2.).then_some(index)
528        })
529        .collect();
530    (frequencies, Some(map))
531}

533impl Frame {
534    fn new(
535        input: &MeasurementSet,
536        options: &MsRegridOptions,
537        first_row: usize,
538    ) -> Result<Option<Self>, MsTransformError> {
539        if options.outframe.trim().is_empty() {
540            return Ok(None);
541        }
542        let name = options.outframe.to_ascii_uppercase();
543        let target_reference = if name == "SOURCE" {
544            FrequencyRef::GEO
545        } else {
546            name.parse::<FrequencyRef>()
547                .ok()
548                .filter(|reference| {
549                    !matches!(reference, FrequencyRef::REST | FrequencyRef::Undefined)
550                })
551                .ok_or_else(|| invalid_choice("outframe", &options.outframe))?
552        };
553        let metadata_reference = if target_reference == FrequencyRef::GEO {
554            FrequencyRef::REST
555        } else {
556            target_reference
557        };
558        let engine = if name == "SOURCE" {
559            MsCalEngine::for_calibration(input)
560        } else {
561            MsCalEngine::new(input)
562        }
563        .map_err(ms_error)?;
564        let table = input.main_table();
565        let reference_time =
566            crate::subtables::get_f64(table, first_row, "TIME").map_err(ms_error)?;
567        let reference_field =
568            crate::subtables::get_i32(table, first_row, "FIELD_ID").map_err(ms_error)? as usize;
569        let phasecenter = options.phasecenter.trim();
570        let field = if phasecenter.is_empty() {
571            Some(reference_field)
572        } else {
573            phasecenter.parse::<usize>().ok()
574        };
575        let mut source_velocity = (name == "SOURCE").then_some(0.);
576        let target = if let Some(field) = field {
577            if name == "SOURCE" {
578                let velocity = engine
579                    .attached_radial_velocity(reference_time, field)
580                    .map_err(ms_error)?;
581                if phasecenter.is_empty()
582                    && velocity
583                        .as_ref()
584                        .is_none_or(|velocity| velocity.refer() != RadialVelocityRef::GEO)
585                {
586                    return Err(invalid(format!(
587                        "SOURCE regridding needs a GEO ephemeris on FIELD_ID {field}."
588                    )));
589                }
590                source_velocity = Some(velocity.map_or(0., |velocity| velocity.ms()));
591            }
592            engine.spectral_frame_observatory(reference_time, field)
593        } else {
594            let direction = parse_phase_direction(phasecenter).ok_or_else(|| {
595                invalid(format!(
596                    "Cannot interpret phasecenter {phasecenter:?} as a FIELD ID or direction."
597                ))
598            })?;
599            engine.spectral_frame_observatory_direction(reference_time, direction)
600        }
601        .map_err(ms_error)?;
602        Ok(Some(Self {
603            engine,
604            target,
605            target_reference,
606            metadata_reference,
607            source_velocity,
608        }))
609    }
610
611    fn prepare(
612        &self,
613        source: FrequencyRef,
614        origin: &MeasFrame,
615    ) -> Result<PreparedFrequencyConversion, MsTransformError> {
616        if source == self.target_reference {
617            PreparedFrequencyConversion::new(source, self.target_reference, None, None)
618        } else {
619            PreparedFrequencyConversion::from_measure_references(
620                source,
621                self.target_reference,
622                origin,
623                &self.target,
624            )
625        }
626        .map_err(ms_error)
627    }
628
629    fn row_conversion(
630        &self,
631        source: FrequencyRef,
632        time: f64,
633        field: usize,
634    ) -> Result<impl Fn(f64) -> f64 + use<>, MsTransformError> {
635        let origin = self
636            .engine
637            .spectral_frame_observatory(time, field)
638            .map_err(ms_error)?;
639        let conversion = self.prepare(source, &origin)?;
640        let correction = if let Some(offset) = self.source_velocity {
641            self.engine
642                .attached_radial_velocity(time, field)
643                .map_err(ms_error)?
644                .map(|velocity| velocity.ms())
645                .filter(|velocity| velocity.abs() > MIN_SOURCE_VELOCITY_M_PER_S)
646                .map_or(0., |velocity| velocity - 2. * offset)
647        } else {
648            0.
649        };
650        let shift = radial_shift(correction);
651        Ok(move |frequency| shift.shift_frequency_hz(conversion.convert_hz(frequency)))
652    }
653}
654
655fn radial_shift(m_per_s: f64) -> MDoppler {
656    MDoppler::new(m_per_s / 299_792_458., DopplerRef::BETA)
657}

673    }

675    pub(super) fn apply(
676        &self,
677        value: ArrayValue,
678        fft_flags: Option<&ArrayValue>,
679    ) -> Result<ArrayValue, TableError> {
680        let value = self.project(value)?;
681        if let Some(shift) = self.shift {
682            shift.apply(value, fft_flags)
683        } else {
684            Ok(value)
685        }
686    }
687
688    fn project(&self, value: ArrayValue) -> Result<ArrayValue, TableError> {
689        let value = self.interpolation.apply(value)?;
690        self.aggregate(value)
691    }
692
693    pub(super) fn apply_categories(&self, value: ArrayValue) -> Result<ArrayValue, TableError> {
694        let ArrayValue::Bool(values) = value else {
695            return Err(TableError::Schema("FLAG_CATEGORY must be Bool.".into()));
696        };
697        let view = values
698            .view()
699            .into_dimensionality::<Ix3>()
700            .map_err(|_| TableError::Schema("FLAG_CATEGORY must have rank 3.".into()))?;
701        let ArrayValue::Bool(value) = self.interpolation.apply_categories(view) else {
702            unreachable!()
703        };
704        let Some(map) = self.channel_map else {
705            return Ok(ArrayValue::Bool(value));
706        };
707        let mut output = ndarray::Array3::default((view.dim().0, self.output_len, view.dim().2));
708        for (from, &to) in map.iter().enumerate() {
709            if let Some(to) = to {
710                for corr in 0..view.dim().0 {
711                    for cat in 0..view.dim().2 {
712                        output[[corr, to, cat]] |= value[[corr, from, cat]];
713                    }
714                }
715            }
716        }
717        Ok(ArrayValue::Bool(output.into_dyn()))
718    }
719
720    fn aggregate(&self, value: ArrayValue) -> Result<ArrayValue, TableError> {
721        let Some(map) = self.channel_map else {
722            return Ok(value);
723        };
724        let correlations = value.shape()[0];
725        macro_rules! aggregate {
726            ($kind:ident, $value:ident, $update:expr) => {{
727                let mut output = ndarray::Array2::default((correlations, self.output_len));
728                let mut counts = vec![0; self.output_len];
729                for (from, &to) in map.iter().enumerate() {
730                    if let Some(to) = to {
731                        counts[to] += 1;
732                        for corr in 0..correlations {
733                            let old = output[[corr, to]];
734                            let new = $value[[corr, from]];
735                            output[[corr, to]] = ($update)(old, new, counts[to]);
736                        }
737                    }
738                }
739                Ok(ArrayValue::$kind(output.into_dyn()))
740            }};
741        }
742        match value {
743            ArrayValue::Float32(data) => aggregate!(Float32, data, |old, new, count| old
744                + (new - old) / count as f32),
745            ArrayValue::Complex32(data) => aggregate!(Complex32, data, |old, new, count| old
746                + (new - old) / count as f32),
747            ArrayValue::Bool(flags) => aggregate!(Bool, flags, |old, new, _| old || new),
748            _ => Err(TableError::Schema(
749                "Unexpected regridded visibility type.".into(),
750            )),
751        }
752    }
753
754    pub(super) fn statistics(
755        &self,
756        weight: ArrayValue,
757        sigma: ArrayValue,
758        output_flags: &ArrayValue,
759        fft_flags: Option<&ArrayValue>,
760    ) -> Result<AveragedStatistics, TableError> {
761        let ArrayValue::Float32(sigma) = sigma else {
762            return Err(TableError::Schema("SIGMA_SPECTRUM must be Float32.".into()));
763        };
764        let ArrayValue::Float32(weight_spectrum) = self.apply(weight, fft_flags)? else {
765            unreachable!()
766        };
767        let weight_spectrum =
768            ArrayValue::Float32(weight_spectrum.mapv(|weight| weight * self.noise_scale));
769        let ArrayValue::Float32(inverse_variance) = self.apply(
770            ArrayValue::Float32(sigma.mapv(weight_from_sigma)),
771            fft_flags,
772        )?
773        else {
774            unreachable!()
775        };
776        let sigma_spectrum = ArrayValue::Float32(
777            inverse_variance.mapv(|weight| sigma_from_weight(weight * self.noise_scale)),
778        );
779        let ArrayValue::Bool(flags) = output_flags else {
780            return Err(TableError::Schema("FLAG must be Bool.".into()));
781        };
782        let flags = flags
783            .view()
784            .into_dimensionality::<Ix2>()
785            .map_err(|_| TableError::Schema("FLAG must have rank 2.".into()))?;
786        Ok(AveragedStatistics {
787            weight: combine::mean_noise(&weight_spectrum, flags)?,
788            sigma: combine::mean_noise(&sigma_spectrum, flags)?,
789            weight_spectrum,
790            sigma_spectrum,
791        })
792    }

810pub(super) fn input_noise(
811    (weight, sigma): (&ArrayValue, &ArrayValue),
812    spectra: super::channel_average::SpectrumCells<'_>,
813    flags: &ArrayValue,
814    bins: &[Vec<usize>],
815    columns: &super::visibility_columns::VisibilityColumns,
816    averaging_time: bool,
817) -> Result<(ArrayValue, ArrayValue), TableError> {
818    use super::channel_average::unaveraged_spectra;
819    use crate::OptionalMainColumn::{CorrectedData, Data, ModelData};
820    let has = |source| columns.copies().iter().any(|copy| copy.source == source);
821    if columns
822        .copies()
823        .iter()
824        .all(|column| column.source == ModelData)
825    {
826        let shape = [flags.shape()[0], bins.len()];
827        let unit = ArrayValue::Float32(ndarray::Array2::from_elem(shape, 1.).into_dyn());
828        return Ok((unit.clone(), unit));
829    }
830    let (mut weight, sigma) = unaveraged_spectra(
831        weight,
832        sigma,
833        spectra,
834        flags,
835        bins,
836        !has(Data) && !averaging_time,
837    )?;
838    if has(Data) && !has(CorrectedData) && !averaging_time {
839        let ArrayValue::Float32(sigma) = &sigma else {
840            unreachable!()
841        };
842        weight = ArrayValue::Float32(sigma.mapv(weight_from_sigma));
843    }
844    Ok((weight, sigma))
845}
