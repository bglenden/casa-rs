4//! The command-line and task surface of native CASA-RS imaging, on the
5//! shared task front end (#931).
6//!
7//! The binary hosts CASA's `tclean` (the imager re-surfaced, #931). The
8//! front end resolves tclean's parameters through the provider-contracts
9//! catalog: command-line flags and a `--json-run` request become the same
10//! resolved values, and those values, and nothing else, make the
11//! [`ImagingRequest`] the application runs, read as CASA's modes read them
12//! ([`ImagingRequest::from_tclean`]). A run's reply carries its
13//! [`ImagerRunTaskResult`], and returns CASA's summary record
14//! ([`tclean_record`]).

36fn imager_surface() -> Result<SurfaceContractBundle, String> {
37    casa_provider_contracts::builtin_surface_bundle("tclean")
38}
39
40fn workspace() -> Result<PathBuf, String> {
41    std::env::current_dir().map_err(|error| format!("the working directory: {error}"))
42}
43
44/// The request `resolved`, every tclean parameter, makes. A value the
45/// catalog accepts but casa-rs cannot honour in the request's modes, or that
46/// the request's own read refuses (`stokes = "IX"`), is refused at its
47/// parameter.
48fn request_from(
49    resolved: Map<String, Value>,
50) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
51    let request = ImagingRequest::from_tclean(&resolved).map_err(|refusal| {
52        let failure = TaskFailure::validation(refusal.message);
53        if refusal.parameter.is_empty() {
54            failure
55        } else {
56            failure.at(refusal.parameter)
57        }
58    })?;
59    Ok((ImagingParameters(resolved), request))
60}
61
62/// Resolve `parameters`, catalog-named and sparse, over the catalog's
63/// defaults: the resolved parameters and the request they make.
64pub fn resolve_request(
65    parameters: &ImagingParameters,
66) -> Result<(ImagingParameters, ImagingRequest), String> {
67    let resolved = resolve_plain_values(imager_surface()?, workspace()?, &parameters.0)
68        .map_err(|error| format!("resolve tclean parameters: {error}"))?;
69    request_from(resolved).map_err(|failure| failure.message)
70}

72/// The environment variable that asks the `casars-imager` binary for its
73/// run summary. CASA's `tclean` writes nothing beside its image products, so
74/// neither does a tclean run here (owner decision of 2026-11-02, #980); set
75/// to `1`, the binary also writes `<imagename>.summary.json`, the run's
76/// status, resolved parameters, backend and each phase's time and memory,
77/// for performance tooling ([`casa_imaging_application::RunSummary`]).
78pub const RUN_SUMMARY_VARIABLE: &str = "CASA_RS_IMAGER_RUN_SUMMARY";
79
80/// Whether this process was asked for run summaries
81/// ([`RUN_SUMMARY_VARIABLE`]).
82fn run_summary_requested() -> bool {
83    std::env::var_os(RUN_SUMMARY_VARIABLE).is_some_and(|value| value == "1")
84}

86/// Run `request`, resolved as `parameters`, as CASA's tclean does: it
87/// writes the image products and nothing else beside them. SIGINT cancels
88/// the run.
89pub fn run(
90    parameters: ImagingParameters,
91    request: &ImagingRequest,
92) -> Result<ImagerRunTaskResult, String> {
93    run_with_statistics(parameters, request, false).map(|(result, _)| result)
94}
95
96/// Keep final controller statistics in memory for the CASA return without
97/// adding fields to the serialized task report or the run summary, which is
98/// written beside the products only when `summary` asks for it.
99fn run_with_statistics(
100    parameters: ImagingParameters,
101    request: &ImagingRequest,
102    summary: bool,
103) -> Result<(ImagerRunTaskResult, Option<ResidualStatistics>), String> {
104    // Task continuation is distinct from internal atomic output replacement.
105    let restart = parameters
106        .0
107        .get("restart")
108        .and_then(Value::as_bool)
109        .ok_or_else(|| "resolved tclean restart must be boolean".to_owned())?;
110    let request = request
111        .with_tclean_output_policy(restart)
112        .map_err(|error| error.to_string())?;
113    let request = &request;
114    let started = Instant::now();
115    let summary = summary.then(|| {
116        let mut path = request.imagename.clone().into_os_string();
117        path.push(".summary.json");
118        SummaryTarget {
119            path: path.into(),
120            request: Value::Object(parameters.0.clone()),
121        }
122    });
123    let context = RunContext {
124        host: HostResources::detect().map_err(|error| error.to_string())?,
125        policy: request.resource_policy(),
126        cancel: interrupt::token().clone(),
127        summary,
128    };
129    let outcome = execute(request, context).map_err(|error| error.to_string())?;
130    let result =
131        ImagerRunTaskResult::from_outcome(parameters, request, &outcome, started.elapsed());
132    Ok((result, outcome.final_residual_statistics))
133}

135/// The front end of the `casars-imager` binary, which hosts `tclean`.
136pub fn task_front_end()
137-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
138    TaskFrontEnd::new(
139        "casars-imager",
140        imager_task_schema_bundle(),
141        &["tclean"],
142        run_task,
143    )
144    .expect("the imager contract carries its surface")
145}
146
147/// The resolved parameters of `task`, a tclean run, and the request they
148/// make; a request they cannot make is refused.
149pub fn task_request(
150    task: &TaskInvocation<'_>,
151) -> Result<(ImagingParameters, ImagingRequest), TaskFailure> {
152    request_from(task.plain_values())
153}
154
155/// Run `task`, one resolved imager request, as the `casars-imager` binary
156/// does: the execution callback of [`task_front_end`]. SIGINT cancels it:
157/// the run stops at its next safe point, publishes nothing, and replies that
158/// it was interrupted.
159pub fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
160    let (parameters, request) = task_request(task)?;
161    // The record echoes some of tclean's values: a dirty run's threshold as
162    // the caller gave it, a number as a number and text as text, and
163    // tclean's default 0.0 where it was not given.
164    let mut values = parameters.0.clone();
165    values.insert(
166        "threshold".to_string(),
167        task.session()
168            .given("threshold")
169            .map_or(Value::from(0.0), ParameterValue::to_plain_json),
170    );
171    interrupt::install().map_err(|error| {
172        TaskFailure::execution(format!("install the interrupt handler: {error}"))
173    })?;
174    let (result, statistics) = run_with_statistics(parameters, &request, run_summary_requested())
175        .map_err(|error| {
176        if interrupt::token().is_cancelled() {
177            TaskFailure::interrupted(format!("{error}; nothing was published"))
178        } else {
179            TaskFailure::execution(error)
180        }
181    })?;
182    let text = format!(
183        "Wrote CASA-compatible products at prefix {} ({} gridded samples, {} major cycles, \
184         {} reported minor iterations, {} actual components, stop={:?})\n",
185        request.imagename.display(),
186        result.run.gridded_samples,
187        result.run.major_cycles,
188        result.run.minor_iterations,
189        result.run.actual_minor_iterations,
190        result.run.clean_stop_reason,
191    );
192    let value = tclean_record(&request, &result.run, &values, statistics.as_ref());
193    Ok(TaskSuccess::new(value, &result)?.with_text(text))
194}
