2//! The deferred plan of DATA: what the session's DATA steps asked for,
3//! applied when `close` writes each row.
4//!
5//! In CASA every step that changes DATA (`predict`, `corrupt`) rewrites the
6//! rows that exist when it is called, which are those of the scans observed
7//! so far; later scans are untouched. Each step here records how many scans
8//! it covers, so for scan `s` DATA is the fold of the steps covering `s`, in
9//! the order they were taken, over the zero DATA `observe` writes. Nothing
10//! is read back from disk, and a sequence of steps costs one pass.
11//!
12//! A `predict` that replaces DATA covers every scan an earlier step covers
13//! (each covers a prefix of the scans), so it clears the plan before it is
14//! recorded; every step left then adds: DATA is the sum of the predictions
15//! covering the scan and the noise of the corruptions covering it. The fold
16//! applies each step to the running DATA (`D ← step(D)`), so multiplicative
17//! gains (`D ← G(D)`) slot in as further [`DataStep`] variants.
18//!
19//! Scans are only ever added, so each step covers at least the scans of the
20//! one before it, and the steps covering a scan are the last ones taken.
21//! Every corruption's applications are kept end to end in one list, so a
22//! scan's noise is the end of that list: `close` reads it where it is, and
23//! N interleaved scans and corruptions keep N lists' worth, not N²
24//! (#853 review).
25
26use std::ops::Range;
27use std::sync::Arc;

29use super::prediction::Prediction;
30use crate::corrupt::NoiseApplication;
31use crate::predict::{ComponentPrediction, SingleDishPrediction};

33/// One `setnoise` term: CASA's `ANoise` calibration term, kept until
34/// `reset`.
35#[derive(Debug, Clone, Copy, PartialEq)]
36pub(crate) struct NoiseTerm {
37    /// The seed in force at `setnoise`.
38    pub(crate) seed: i32,
39    /// σ of each real and imaginary draw, Jy.
40    pub(crate) sigma_jy: f32,
41    /// How many `corrupt` calls have applied the term.
42    pub(crate) applications: u32,
43}
44
45/// A step that changes DATA, covering the scans observed before it.
46#[derive(Debug, Clone)]
47pub(crate) enum DataStep {
48    /// `predict`: add the model's prediction. A replacing `predict` is the
49    /// first step of the plan, so adding to the zero DATA replaces it.
50    Predict {
51        /// The prepared model.
52        prediction: Arc<Prediction>,
53        /// The step covers scans `0..scans`.
54        scans: usize,
55    },
56    /// `predict(complist=…)`: add the component list's visibilities. A
57    /// `predict` of an image and a list records both steps, the list's
58    /// first, as CASA predicts its components first.
59    PredictComponents {
60        /// The prepared list.
61        components: Arc<ComponentPrediction>,
62        /// The step covers scans `0..scans`.
63        scans: usize,
64    },
65    /// `predict` after `setoptions(ftmachine='sd')`: add each pointing's
66    /// total power to its autocorrelations.
67    PredictSingleDish {
68        /// The sampled model.
69        prediction: Arc<SingleDishPrediction>,
70        /// The step covers scans `0..scans`.
71        scans: usize,
72    },
73    /// `corrupt`: add each application's noise.
74    Corrupt {
75        /// Its applications, one per noise term in the order the terms
76        /// were set, as their range of the plan's noise.
77        applications: Range<usize>,
78        /// The step covers scans `0..scans`.
79        scans: usize,
80    },
81}
82
83impl DataStep {
84    /// The step covers scans `0..scans()`.
85    pub(crate) fn scans(&self) -> usize {
86        match self {
87            Self::Predict { scans, .. }
88            | Self::PredictComponents { scans, .. }
89            | Self::PredictSingleDish { scans, .. }
90            | Self::Corrupt { scans, .. } => *scans,
91        }
92    }
93}
94
95/// The session's DATA steps, in the order they were taken.
96#[derive(Debug, Clone, Default)]
97pub(crate) struct DataPlan {
98    steps: Vec<DataStep>,
99    /// Every corruption's applications, in the order they were taken.
100    noise: Vec<NoiseApplication>,
101}
102
103impl DataPlan {
104    /// Record a `predict` of `prediction` covering scans `0..scans`, after
105    /// the steps already taken.
106    pub(crate) fn predict(&mut self, prediction: Arc<Prediction>, scans: usize) {
107        self.push(DataStep::Predict { prediction, scans });
108    }
109
110    /// Record a `predict` of the component list `components` covering scans
111    /// `0..scans`, after the steps already taken.
112    pub(crate) fn predict_components(
113        &mut self,
114        components: Arc<ComponentPrediction>,
115        scans: usize,
116    ) {
117        self.push(DataStep::PredictComponents { components, scans });
118    }
119
120    /// Record a total-power `predict` of `prediction` covering scans
121    /// `0..scans`, after the steps already taken.
122    pub(crate) fn predict_single_dish(
123        &mut self,
124        prediction: Arc<SingleDishPrediction>,
125        scans: usize,
126    ) {
127        self.push(DataStep::PredictSingleDish { prediction, scans });
128    }
129
130    /// Record a `corrupt` adding `applications` to scans `0..scans`, after
131    /// the steps already taken.
132    pub(crate) fn corrupt(
133        &mut self,
134        applications: impl IntoIterator<Item = NoiseApplication>,
135        scans: usize,
136    ) {
137        let start = self.noise.len();
138        self.noise.extend(applications);
139        self.push(DataStep::Corrupt {
140            applications: start..self.noise.len(),
141            scans,
142        });
143    }
144
145    fn push(&mut self, step: DataStep) {
146        debug_assert!(
147            self.steps
148                .last()
149                .is_none_or(|last| last.scans() <= step.scans()),
150            "a step covers at least the scans of the one before it"
151        );
152        self.steps.push(step);
153    }
154
155    /// Forget every step: a replacing `predict` overwrites what they did.
156    pub(crate) fn clear(&mut self) {
157        self.steps.clear();
158        self.noise.clear();
159    }
160
161    /// The steps covering scan `scan`, in the order they were taken: the
162    /// last ones.
163    pub(crate) fn steps_for_scan(&self, scan: usize) -> &[DataStep] {
164        let first = self.steps.partition_point(|step| step.scans() <= scan);
165        &self.steps[first..]
166    }
167
168    /// The noise to add to scan `scan`'s rows, in the order it was asked
169    /// for: the applications of its corruptions, all of which add, which
170    /// are the last ones taken.
171    pub(crate) fn noise_for_scan(&self, scan: usize) -> &[NoiseApplication] {
172        let start = self
173            .steps_for_scan(scan)
174            .iter()
175            .find_map(|step| match step {
176                DataStep::Corrupt { applications, .. } => Some(applications.start),
177                DataStep::Predict { .. }
178                | DataStep::PredictComponents { .. }
179                | DataStep::PredictSingleDish { .. } => None,
180            })
181            .unwrap_or(self.noise.len());
182        &self.noise[start..]
183    }
184
185    /// The predict steps covering scan `scan`, as indices of
186    /// [`Self::predictions`], in the order they were taken: their
187    /// predictions add. They are the last ones.
188    pub(crate) fn predictions_for_scan(&self, scan: usize) -> Range<usize> {
189        let all = self.predictions().count();
190        let covering = self
191            .steps_for_scan(scan)
192            .iter()
193            .filter(|step| matches!(step, DataStep::Predict { .. }))
194            .count();
195        all - covering..all
196    }
197
198    /// Every predict step's prepared model and the scans it covers, in the
199    /// order they were taken.
200    pub(crate) fn predictions(&self) -> impl Iterator<Item = (&Prediction, usize)> {
201        self.steps.iter().filter_map(|step| match step {
202            DataStep::Predict { prediction, scans } => Some((prediction.as_ref(), *scans)),
203            DataStep::PredictComponents { .. }
204            | DataStep::PredictSingleDish { .. }
205            | DataStep::Corrupt { .. } => None,
206        })
207    }
208
209    /// The component lists the steps covering scan `scan` predict, in the
210    /// order they were taken: their visibilities add.
211    pub(crate) fn components_for_scan(
212        &self,
213        scan: usize,
214    ) -> impl Iterator<Item = &ComponentPrediction> {
215        self.steps_for_scan(scan)
216            .iter()
217            .filter_map(|step| match step {
218                DataStep::PredictComponents { components, .. } => Some(components.as_ref()),
219                DataStep::Predict { .. }
220                | DataStep::PredictSingleDish { .. }
221                | DataStep::Corrupt { .. } => None,
222            })
223    }
224
225    /// The total-power predictions covering scan `scan`, in the order they
226    /// were taken: they add.
227    pub(crate) fn single_dish_for_scan(
228        &self,
229        scan: usize,
230    ) -> impl Iterator<Item = &SingleDishPrediction> {
231        self.steps_for_scan(scan)
232            .iter()
233            .filter_map(|step| match step {
234                DataStep::PredictSingleDish { prediction, .. } => Some(prediction.as_ref()),
235                DataStep::Predict { .. }
236                | DataStep::PredictComponents { .. }
237                | DataStep::Corrupt { .. } => None,
238            })
239    }
240
241    /// Whether any step predicts a component list.
242    pub(crate) fn predicts_components(&self) -> bool {
243        self.steps
244            .iter()
245            .any(|step| matches!(step, DataStep::PredictComponents { .. }))
246    }
247
248    /// Whether any step predicts total power.
249    pub(crate) fn predicts_single_dish(&self) -> bool {
250        self.steps
251            .iter()
252            .any(|step| matches!(step, DataStep::PredictSingleDish { .. }))
253    }
254}

256#[cfg(test)]
257mod tests {
258    use super::*;
259
260    fn noise(seed: i32, ordinal: u32) -> NoiseApplication {
261        NoiseApplication {
262            seed,
263            sigma_jy: 0.5,
264            ordinal,
265        }
266    }
267
268    #[test]
269    fn a_step_covers_only_the_scans_observed_before_it() {
270        let mut plan = DataPlan::default();
271        plan.corrupt([noise(1, 0)], 1);
272        plan.corrupt([noise(1, 1), noise(2, 0)], 3);
273        assert_eq!(
274            plan.noise_for_scan(0),
275            [noise(1, 0), noise(1, 1), noise(2, 0)]
276        );
277        assert_eq!(plan.noise_for_scan(2), [noise(1, 1), noise(2, 0)]);
278        assert_eq!(plan.noise_for_scan(3), []);
279        assert_eq!(plan.steps_for_scan(1).len(), 1);
280        assert!(plan.predictions_for_scan(0).is_empty());
281        plan.clear();
282        assert_eq!(plan.noise_for_scan(0), []);
283    }
284
285    #[test]
286    fn interleaved_corruptions_share_their_applications() {
287        let mut plan = DataPlan::default();
288        for scans in 1..=4 {
289            plan.corrupt([noise(1, scans as u32 - 1)], scans);
290        }
291        for scan in 0..4 {
292            let expected = (scan as u32..4).map(|ordinal| noise(1, ordinal));
293            assert!(plan.noise_for_scan(scan).iter().copied().eq(expected));
294        }
295        // Each scan's noise is the end of the one list.
296        let all = plan.noise_for_scan(0).as_ptr_range();
297        assert_eq!(plan.noise_for_scan(3).as_ptr_range().end, all.end);
298        assert_eq!(plan.noise_for_scan(4), []);
299    }
