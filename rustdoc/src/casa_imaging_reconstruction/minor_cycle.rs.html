3//! Bounded point and multiscale Minor Cycles over authoritative Normal State views.
4//!
5//! The Minor Cycle is a pure scientific operation: it consumes one immutable
6//! view of an authoritative Final Normal State (the T20 residual paired with
7//! its normal approximation), one named authoritative model generation, an
8//! explicit valid-support window, and explicit gain, threshold, iteration,
9//! and staleness controls. It accumulates Högbom components against the PSF
10//! exactly like the casacore `LatticeCleaner` HOGBOM loop — peak residual
11//! normalized by the PSF peak, gain-scaled PSF subtraction — and mints the
12//! resulting sparse update only through the T28 model lifecycle owner as one
13//! base-bound Model Delta. Authoritative model and residual state are never
14//! mutated; the solver works on private buffers.
15//!
16//! Every non-converged stop (`IterationBound`, `StalenessBound`) records that
17//! the linear approximation envelope is exhausted and explicitly requests
18//! Major-Cycle reconciliation. Component-sequence recording is optional
19//! diagnostic evidence: first-divergence comparisons between two recorded
20//! sequences are informational and never gate acceptance.

41/// Return the hard resident-memory envelope for one solver-owned Minor Cycle.
42///
43/// The envelope covers private residual planes, robust-statistics scratch,
44/// MT-MFS scale kernels and Taylor systems, candidate vectors, the bounded
45/// sparse Model Delta accumulator, and optional component diagnostics. Normal
46/// State and model-generation storage are owned by their existing plan slots
47/// and are therefore not counted again here.
48#[must_use]
49pub fn minor_cycle_workspace_bytes(
50    shape: [usize; 2],
51    basis: ReconstructionBasis,
52    algorithm: &ReconstructionAlgorithm,
53    maximum_iterations: usize,
54    recorded_components: usize,
55) -> u64 {
56    let cells = sat_u64(shape[0]).saturating_mul(sat_u64(shape[1]));
57    let scalar_workspace = cells.saturating_mul(16);
58    let (ReconstructionBasis::Taylor { terms }, ReconstructionAlgorithm::Mtmfs { scales_px, .. }) =
59        (basis, algorithm)
60    else {
61        return scalar_workspace;
62    };
63    let terms = sat_u64(terms);
64    let effective_sample_counts = scales_px
65        .iter()
66        .copied()
67        .filter(|scale| *scale <= (shape[0] / 2) as f64 && *scale <= (shape[1] / 2) as f64)
68        .map(|scale| {
69            if scale == 0.0 {
70                1
71            } else {
72                let diameter = (scale.ceil() as u64).saturating_mul(2).saturating_add(1);
73                diameter.saturating_mul(diameter)
74            }
75        })
76        .collect::<Vec<_>>();
77    let scale_count = sat_u64(effective_sample_counts.len());
78    let kernel_samples = effective_sample_counts
79        .iter()
80        .copied()
81        .fold(0_u64, u64::saturating_add);
82    let maximum_kernel_samples = effective_sample_counts.iter().copied().max().unwrap_or(1);
83    let term_cells = terms.saturating_mul(cells);
84    let possible_sparse_terms = sat_u64(maximum_iterations)
85        .saturating_mul(terms)
86        .saturating_mul(maximum_kernel_samples)
87        .min(term_cells);
88
89    // A BTreeMap entry's allocator/node overhead is implementation-private.
90    // Eight key/value pairs per possible term is a conservative 64-bit host
91    // envelope and deliberately dominates the current standard-library node.
92    const SPARSE_TERM_ENTRY_BOUND_BYTES: u64 = 128;
93    let residual_planes = term_cells.saturating_mul(size_of_u64::<f64>());
94    let plane_scratch = cells.saturating_mul(size_of_u64::<f64>());
95    let kernel_storage = kernel_samples
96        .saturating_mul(size_of_u64::<([isize; 2], f64)>())
97        .saturating_add(scale_count.saturating_mul(size_of_u64::<ScaleKernel>()));
98    let square_terms = terms.saturating_mul(terms);
99    let scale_systems = scale_count.saturating_mul(
100        square_terms
101            .saturating_mul(size_of_u64::<f64>())
102            .saturating_add(size_of_u64::<TaylorScaleSystem>()),
103    );
104    let system_and_candidate_scratch = square_terms
105        .saturating_mul(size_of_u64::<f64>())
106        .saturating_mul(4)
107        .saturating_add(terms.saturating_mul(size_of_u64::<f64>()).saturating_mul(6));
108    let sparse_delta = possible_sparse_terms.saturating_mul(SPARSE_TERM_ENTRY_BOUND_BYTES);
109    let diagnostics = sat_u64(recorded_components)
110        .min(sat_u64(maximum_iterations).saturating_mul(terms))
111        .saturating_mul(size_of_u64::<MinorCycleComponent>());
112    let container_overhead = terms
113        .saturating_add(scale_count.saturating_mul(2))
114        .saturating_add(8)
115        .saturating_mul(size_of_u64::<Vec<u8>>());
116
117    residual_planes
118        .saturating_add(plane_scratch)
119        .saturating_add(kernel_storage)
120        .saturating_add(scale_systems)
121        .saturating_add(system_and_candidate_scratch)
122        .saturating_add(sparse_delta)
123        .saturating_add(diagnostics)
124        .saturating_add(container_overhead)
125}

180/// One reconstruction-owned minor-cycle program.
181///
182/// The validity contract states whether the compiled normal-state view remains

268impl MinorCycleProgram {
269    /// Derive the executable minor-cycle controls from the compiled contract.
270    ///
271    pub fn for_algorithm(
272        mut algorithm: ReconstructionAlgorithm,
273        controls: ReconstructionControls,
274    ) -> Result<Self, MinorCycleError> {
275        if matches!(algorithm, ReconstructionAlgorithm::Mtmfs { .. }) {
276            return Err(MinorCycleError::CompiledProblemRequired);
277        }
278        Self::for_contract(None, &mut algorithm, controls)
279    }

281    /// Derive an identity-bound program from the authoritative Compiled Problem.
282    pub fn for_problem(problem: &CompiledProblem) -> Result<Self, MinorCycleError> {
283        let mut algorithm = problem.reconstruction().algorithm().clone();
284        Self::for_contract(
285            Some(problem.problem_id()),
286            &mut algorithm,
287            problem.reconstruction().controls(),
288        )
289    }
290
291    fn for_contract(
292        problem: Option<CompiledProblemId>,
293        algorithm: &mut ReconstructionAlgorithm,
294        controls: ReconstructionControls,
295    ) -> Result<Self, MinorCycleError> {
296        if !matches!(
297            algorithm,
298            ReconstructionAlgorithm::Hogbom
299                | ReconstructionAlgorithm::Clark
300                | ReconstructionAlgorithm::Multiscale { .. }
301                | ReconstructionAlgorithm::Mtmfs { .. }
302        ) {
303            return Err(MinorCycleError::UnsupportedAlgorithm);
304        }
305        if let ReconstructionAlgorithm::Multiscale { scales_px, .. }
306        | ReconstructionAlgorithm::Mtmfs { scales_px, .. } = algorithm
307        {
308            if scales_px.is_empty()
309                || scales_px
310                    .iter()
311                    .any(|scale| !scale.is_finite() || *scale < 0.0)
312            {
313                return Err(MinorCycleError::InvalidScale);
314            }
315            scales_px.sort_by(f64::total_cmp);
316            scales_px.dedup_by(|left, right| left.to_bits() == right.to_bits());
317        }
318        Self::new_for_algorithm(
319            algorithm.clone(),
320            controls.gain(),
321            controls.threshold_jy_per_beam(),
322            controls
323                .cycle_iteration_limit()
324                .unwrap_or(controls.max_minor_iterations())
325                .min(controls.max_minor_iterations()),
326            MinorCycleValidity::Exact,
327        )
328        .map(|mut program| {
329            program.noise_sigma = controls.noise_sigma();
330            program.hogbom_iteration_accounting =
331                if matches!(&program.algorithm, ReconstructionAlgorithm::Hogbom) {
332                    controls.hogbom_iteration_accounting()
333                } else {
334                    HogbomIterationAccounting::Strict
335                };
336            program.cycle_threshold =
337                controls
338                    .cycle_factor()
339                    .map(|factor| CycleThresholdControls {
340                        factor,
341                        minimum_psf_fraction: controls
342                            .minimum_psf_fraction()
343                            .expect("compiled cycle threshold is complete"),
344                        maximum_psf_fraction: controls
345                            .maximum_psf_fraction()
346                            .expect("compiled cycle threshold is complete"),
347                    });
348            program.problem = problem;
349            program
350        })
351    }

392    fn new_for_algorithm(
393        algorithm: ReconstructionAlgorithm,
394        gain: f64,
395        threshold: f64,
396        max_iterations: usize,
397        validity: MinorCycleValidity,
398    ) -> Result<Self, MinorCycleError> {
399        if !(gain > 0.0 && gain <= 1.0) {
400            return Err(MinorCycleError::InvalidGain);
401        }
402        if !threshold.is_finite() || threshold < 0.0 {
403            return Err(MinorCycleError::InvalidThreshold);
404        }
405        if max_iterations == 0 {
406            return Err(MinorCycleError::InvalidIterationBound);
407        }
408        validate_validity(validity)?;
409        Ok(Self {
410            problem: None,
411            algorithm,
412            model_plane: MinorCycleModelPlane::PRIMARY,
413            gain,
414            threshold,
415            noise_sigma: None,
416            max_iterations,
417            hogbom_iteration_accounting: HogbomIterationAccounting::Strict,
418            validity,
419            cycle_threshold: None,
420            fixed_cycle_threshold: None,
421            component_sequence_limit: None,
422        })
423    }

560fn validate_validity(validity: MinorCycleValidity) -> Result<(), MinorCycleError> {
561    if let MinorCycleValidity::Bounded {
562        maximum_absolute_update,
563    } = validity
564        && (!maximum_absolute_update.is_finite() || maximum_absolute_update <= 0.0)
565    {
566        return Err(MinorCycleError::InvalidValidityBound);
567    }
568    Ok(())

571struct MinorCycleController {
572    iteration_limit: usize,
573    validity: MinorCycleValidity,
574    effective_threshold: f64,
575    iterations: usize,
576    total_flux: f64,
577    stop_reason: Option<MinorCycleStopReason>,
578}
579
580impl MinorCycleController {
581    fn new(
582        controls: &MinorCycleProgram,
583        effective_threshold: f64,
584        has_valid_support: bool,
585    ) -> Self {
586        Self {
587            iteration_limit: if has_valid_support {
588                controls.actual_iteration_limit()
589            } else {
590                0
591            },
592            validity: controls.validity(),
593            effective_threshold,
594            iterations: 0,
595            total_flux: 0.0,
596            stop_reason: (!has_valid_support).then_some(MinorCycleStopReason::ThresholdReached),
597        }
598    }
599
600    const fn iteration_limit(&self) -> usize {
601        self.iteration_limit
602    }
603
604    const fn iterations(&self) -> usize {
605        self.iterations
606    }
607
608    fn admit(&mut self, strength: f64, charged_update: f64, inclusive_threshold: bool) -> bool {
609        let reached = strength == 0.0
610            || if inclusive_threshold {
611                strength.abs() <= self.effective_threshold
612            } else {
613                strength.abs() < self.effective_threshold
614            };
615        if reached {
616            self.stop_reason = Some(MinorCycleStopReason::ThresholdReached);
617            return false;
618        }
619        if let MinorCycleValidity::Bounded {
620            maximum_absolute_update,
621        } = self.validity
622            && self.total_flux + charged_update > maximum_absolute_update
623        {
624            self.stop_reason = Some(MinorCycleStopReason::StalenessBound);
625            return false;
626        }
627        true
628    }
629
630    fn accepted(&mut self, charged_update: f64) {
631        self.iterations += 1;
632        self.total_flux += charged_update;
633    }
634
635    fn stop(&mut self, reason: MinorCycleStopReason) {
636        self.stop_reason = Some(reason);
637    }
638
639    fn finish(self) -> (usize, f64, MinorCycleStopReason) {
640        (
641            self.iterations,
642            self.total_flux,
643            self.stop_reason
644                .unwrap_or(MinorCycleStopReason::IterationBound),
645        )
646    }
647}

1064/// Run one reconstruction-owned Minor Cycle.
1065///
1066/// Consumes an immutable view of one authoritative Final Normal State and the
1067/// exact named final model generation, and returns the sparse Model Delta
1068/// minted through the model lifecycle owner plus solver evidence. Neither the
1069/// authoritative residual planes nor the model generation are mutated.
1070///
1071/// The loop preserves the casacore HOGBOM semantics: find the maximum-
1072/// absolute residual over the valid-support window, normalize by the PSF
1073/// peak, stop when below threshold, otherwise subtract `gain * strength`
1074/// scaled by the PSF centered on the peak and accumulate the same flux at the
1075/// peak cell. The current normal-state scalar convention uses the real plane,
1076/// matching the Float-image reference behavior; the typed model-plane
1077/// coordinate determines where the resulting sparse delta is accumulated.
1078///
1079/// # Errors
1080///
1081/// Fails closed on mismatched lineage or geometry, invalid controls or
1082/// windows, a degenerate PSF peak, non-finite arithmetic, and every model
1083/// owner rejection (term bounds, value bounds, or foreign lineage).
1084pub fn run_minor_cycle(
1085    lifecycle: &ModelLifecycle,
1086    base: &ModelGeneration,
1087    view: &FinalNormalState,
1088    mask: &ReconstructionMask,
1089    controls: MinorCycleProgram,
1090) -> Result<MinorCycleResult, MinorCycleError> {
1091    if view.catalog() == crate::NormalStateCatalog::UnnormalizedTaylorBlockV1 {
1092        if controls.problem != Some(view.problem_id()) {
1093            return Err(MinorCycleError::CompiledProblemRequired);
1094        }
1095        return run_taylor_minor_cycle(lifecycle, base, view, mask, controls);
1096    }
1097    if view.channel_count() != 1 {
1098        return Err(MinorCycleError::ChannelCycleRequired);
1099    }
1100    run_minor_cycle_plane(
1101        lifecycle,
1102        base,
1103        view.plane(0).ok_or(MinorCycleError::ModelShapeMismatch)?,
1104        mask,
1105        controls,
1106    )
1107}

1109#[allow(clippy::too_many_lines)]
1110fn run_taylor_minor_cycle(
1111    lifecycle: &ModelLifecycle,
1112    base: &ModelGeneration,
1113    view: &FinalNormalState,
1114    mask: &ReconstructionMask,
1115    controls: MinorCycleProgram,
1116) -> Result<MinorCycleResult, MinorCycleError> {
1117    let ReconstructionAlgorithm::Mtmfs {
1118        scales_px,
1119        small_scale_bias,
1120    } = controls.algorithm()
1121    else {
1122        return Err(MinorCycleError::InvalidNormalStateCatalog);
1123    };
1124    let shape = view.shape();
1125    let cells = shape[0] * shape[1];
1126    let terms_count = view.coefficient_term_count();
1127    let primary = controls.model_plane();
1128    if terms_count < 2
1129        || view.normal_moment_count() != 2 * terms_count - 1
1130        || base.shape().coefficients() != terms_count
1131        || base
1132            .shape()
1133            .domains()
1134            .get(primary.domain())
1135            .is_none_or(|domain| domain.pixels() != shape)
1136        || primary.coefficient() != 0
1137        || primary.polarization() >= base.shape().polarizations()
1138        || base.samples().len() != base.shape().sample_count()
1139    {
1140        return Err(MinorCycleError::ModelShapeMismatch);
1141    }
1142    if base.generation_id() != view.final_model_generation() {
1143        return Err(MinorCycleError::ForeignNormalState);
1144    }
1145    if mask.shape() != shape {
1146        return Err(MinorCycleError::MaskShapeMismatch);
1147    }
1148    if mask.problem_id() != view.problem_id()
1149        || mask.model_generation() != base.generation_id()
1150        || mask
1151            .normal_state_completion()
1152            .is_some_and(|id| id != view.completion_id())
1153    {
1154        return Err(MinorCycleError::ForeignMask);
1155    }
1156
1157    let moment_zero = view
1158        .normal_moment(0)
1159        .ok_or(MinorCycleError::ModelShapeMismatch)?;
1160    let effective_scales = scales_px
1161        .iter()
1162        .copied()
1163        .filter(|scale| *scale <= (shape[0] / 2) as f64 && *scale <= (shape[1] / 2) as f64)
1164        .collect::<Vec<_>>();
1165    if effective_scales.is_empty() {
1166        return Err(MinorCycleError::InvalidScale);
1167    }
1168    let psf_peak_index = taylor_psf_peak_index(
1169        moment_zero.normal_approximation(),
1170        shape,
1171        effective_scales.last().copied().unwrap_or(0.0),
1172    )
1173    .ok_or(MinorCycleError::InvalidPsfPeak)?;
1174    let psf_peak = moment_zero.normal_approximation()[psf_peak_index].re;
1175    if !psf_peak.is_finite() || psf_peak <= 0.0 {
1176        return Err(MinorCycleError::InvalidPsfPeak);
1177    }
1178    let psf_peak_pixel = plane_pixel(psf_peak_index, shape);
1179    let psf_support = taylor_psf_support(shape, effective_scales.last().copied().unwrap_or(0.0));
1180    let kernels = build_scale_kernels(&effective_scales, *small_scale_bias);
1181    let scale_systems = build_taylor_scale_systems(view, shape, psf_peak_pixel, &kernels)?;
1182
1183    let mut residuals = (0..terms_count)
1184        .map(|term| {
1185            view.coefficient_term(term)
1186                .ok_or(MinorCycleError::ModelShapeMismatch)?
1187                .residual()
1188                .iter()
1189                .map(|value| {
1190                    value
1191                        .re
1192                        .is_finite()
1193                        .then_some(value.re)
1194                        .ok_or(MinorCycleError::GeneratedNonfinite)
1195                })
1196                .collect::<Result<Vec<_>, _>>()
1197        })
1198        .collect::<Result<Vec<_>, MinorCycleError>>()?;
1199    let primary_plane = MinorCycleModelPlane::new(primary.domain(), 0, primary.polarization());
1200    let has_valid_support = (0..cells).any(|index| {
1201        let pixel = plane_pixel(index, shape);
1202        mask.contains(pixel) && valid_taylor_support(base, shape, primary, terms_count, pixel)
1203    });
1204    if !has_valid_support {
1205        return finish_taylor_minor_cycle(
1206            lifecycle,
1207            base,
1208            view,
1209            mask,
1210            controls,
1211            0,
1212            0.0,
1213            0.0,
1214            0.0,
1215            None,
1216            0.0,
1217            None,
1218            MinorCycleStopReason::ThresholdReached,
1219            BTreeMap::new(),
1220            Vec::new(),
1221        );
1222    }
1223
1224    let initial_peak = principal_taylor_peak(
1225        &residuals[0],
1226        shape,
1227        base,
1228        primary,
1229        terms_count,
1230        mask,
1231        &kernels[0],
1232    ) / scale_systems[0].h00;
1233    let noise_rms = controls
1234        .noise_sigma()
1235        .map(|_| {
1236            robust_masked_rms(&residuals[0], shape, base, primary_plane, mask)
1237                .map(|rms| rms / scale_systems[0].h00)
1238        })
1239        .transpose()?;
1240    let global_threshold = noise_rms
1241        .zip(controls.noise_sigma())
1242        .map_or(controls.threshold(), |(rms, sigma)| {
1243            controls.threshold().max(rms * sigma)
1244        });
1245    let cycle_threshold = if controls.fixed_cycle_threshold.is_some() {
1246        controls.fixed_cycle_threshold
1247    } else if controls.cycle_threshold.is_some() {
1248        let psf = moment_zero
1249            .normal_approximation()
1250            .iter()
1251            .map(|v| v.re as f32)
1252            .collect::<Vec<_>>();
1253        let sidelobe = crate::fitted_psf_sidelobe_fraction(&psf, shape)?;
1254        controls.cycle_threshold_for(initial_peak, sidelobe)
1255    } else {
1256        None
1257    };
1258    let effective_threshold =
1259        cycle_threshold.map_or(global_threshold, |value| value.max(global_threshold));
1260    let mut model_terms = BTreeMap::<usize, f64>::new();
1261    let mut recorded = Vec::new();
1262    let mut controller = MinorCycleController::new(&controls, effective_threshold, true);
1263    let mut search_window = None;
1264
1265    for _ in 0..controller.iteration_limit() {
1266        let current_peak = principal_taylor_peak(
1267            &residuals[0],
1268            shape,
1269            base,
1270            primary,
1271            terms_count,
1272            mask,
1273            &kernels[0],
1274        ) / scale_systems[0].h00;
1275        if controller.iterations() > 0 && current_peak > initial_peak * 1.5 {
1276            controller.stop(MinorCycleStopReason::MultiscaleDivergence);
1277            break;
1278        }
1279        let candidate = select_taylor_candidate(
1280            &residuals,
1281            shape,
1282            base,
1283            primary,
1284            terms_count,
1285            mask,
1286            &kernels,
1287            &scale_systems,
1288            search_window,
1289        )
1290        .ok_or(MinorCycleError::EmptyValidSupport)?;
1291        let updates = candidate
1292            .coefficients
1293            .iter()
1294            .map(|value| controls.gain() * value)
1295            .collect::<Vec<_>>();
1296        if updates.iter().any(|value| !value.is_finite()) {
1297            return Err(MinorCycleError::GeneratedNonfinite);
1298        }
1299        let charged = updates.iter().map(|value| value.abs()).sum::<f64>();
1300        if !controller.admit(current_peak, charged, false) {
1301            break;
1302        }
1303        let pixel = plane_pixel(candidate.index, shape);
1304        let kernel = &kernels[candidate.scale_index];
1305        for (residual_term, residual) in residuals.iter_mut().enumerate() {
1306            for (coefficient, update) in updates.iter().enumerate() {
1307                let psf = view
1308                    .normal_block(residual_term, coefficient)
1309                    .ok_or(MinorCycleError::ModelShapeMismatch)?;
1310                subtract_scaled_psf(
1311                    residual,
1312                    psf.normal_approximation(),
1313                    shape,
1314                    pixel,
1315                    psf_peak_pixel,
1316                    kernel,
1317                    *update,
1318                )?;
1319            }
1320        }
1321        for (coefficient, update) in updates.into_iter().enumerate() {
1322            let model_plane =
1323                MinorCycleModelPlane::new(primary.domain(), coefficient, primary.polarization());
1324            add_scaled_terms(
1325                &mut model_terms,
1326                base,
1327                model_plane,
1328                shape,
1329                pixel,
1330                kernel,
1331                update,
1332            );
1333            if controls
1334                .component_sequence_limit()
1335                .is_some_and(|limit| recorded.len() < limit)
1336            {
1337                recorded.push(MinorCycleComponent {
1338                    cell: model_cell(model_plane, shape, pixel)
1339                        .expect("selected Taylor pixel is valid"),
1340                    flux: update,
1341                    scale_px: kernel.scale_px,
1342                });
1343            }
1344        }
1345        search_window = Some(TaylorSearchWindow::around(pixel, psf_support, shape));
1346        controller.accepted(charged);
1347    }
1348    if !model_terms.is_empty() {
1349        refresh_taylor_residuals(
1350            &mut residuals,
1351            view,
1352            shape,
1353            psf_peak_pixel,
1354            base,
1355            &model_terms,
1356        )?;
1357    }
1358    let final_peak = principal_taylor_peak(
1359        &residuals[0],
1360        shape,
1361        base,
1362        primary,
1363        terms_count,
1364        mask,
1365        &kernels[0],
1366    ) / scale_systems[0].h00;
1367    let (iterations, total_flux, stop_reason) = controller.finish();
1368    finish_taylor_minor_cycle(
1369        lifecycle,
1370        base,
1371        view,
1372        mask,
1373        controls,
1374        iterations,
1375        total_flux,
1376        initial_peak,
1377        final_peak,
1378        noise_rms,
1379        global_threshold,
1380        cycle_threshold,
1381        stop_reason,
1382        model_terms,
1383        recorded,
1384    )
1385}

1399    global_threshold: f64,
1400    cycle_threshold: Option<f64>,

1417            })
1418            .collect::<Result<Vec<_>, MinorCycleError>>()?;

1420    };
1421    let effective_threshold =
1422        cycle_threshold.map_or(global_threshold, |value| value.max(global_threshold));

1424    let evidence_id = minor_cycle_evidence_id(

1431        mask,
1432        &controls,
1433        iterations,
1434        controller_iterations,
1435        total_flux,
1436        final_peak_flux,
1437        noise_rms,
1438        effective_threshold,
1439        stop_reason,
1440        None,
1441        0,
1442    );
1443    Ok(MinorCycleResult {
1444        delta,
1445        evidence: MinorCycleEvidence {
1446            evidence_id,
1447            problem: lifecycle.problem(),
1448            attempt: lifecycle.attempt(),
1449            epoch: lifecycle.epoch(),
1450            input_generation: base.generation_id(),
1451            normal_state_completion: view.completion_id(),
1452            normal_state_content: view.content_identity(),
1453            iterations,
1454            controller_iterations,
1455            total_flux,
1456            initial_peak_flux,
1457            final_peak_flux,
1458            noise_rms,
1459            global_threshold,
1460            effective_threshold,
1461            cycle_threshold,
1462            stop_reason,
1463            clark_approximation: None,
1464            clark_refreshes: 0,
1465            recorded: (!recorded.is_empty()).then(|| recorded.into_boxed_slice()),
1466        },
1467    })

1470pub(crate) fn run_minor_cycle_plane(
1471    lifecycle: &ModelLifecycle,
1472    base: &ModelGeneration,
1473    plane: FinalNormalStatePlane<'_>,
1474    mask: &ReconstructionMask,
1475    controls: MinorCycleProgram,
1476) -> Result<MinorCycleResult, MinorCycleError> {
1477    let view = plane.owner();
1478    let shape = view.shape();
1479    let cells = shape[0] * shape[1];
1480    let model_plane = controls.model_plane();
1481    if base
1482        .shape()
1483        .domains()
1484        .get(model_plane.domain())
1485        .is_none_or(|domain| domain.pixels() != shape)
1486        || model_plane.coefficient() >= base.shape().coefficients()
1487        || model_plane.polarization() >= base.shape().polarizations()
1488        || base.samples().len() != base.shape().sample_count()
1489    {
1490        return Err(MinorCycleError::ModelShapeMismatch);
1491    }
1492    if base.generation_id() != view.final_model_generation() {
1493        return Err(MinorCycleError::ForeignNormalState);
1494    }
1495    if mask.shape() != shape {
1496        return Err(MinorCycleError::MaskShapeMismatch);
1497    }
1498    if mask.problem_id() != view.problem_id()
1499        || mask.model_generation() != base.generation_id()
1500        || mask
1501            .normal_state_completion()
1502            .is_some_and(|completion| completion != view.completion_id())
1503    {
1504        return Err(MinorCycleError::ForeignMask);
1505    }
1506
1507    // The PSF peak normalization follows the reference cleaner: peaks are
1508    // reported in model units regardless of the accumulated weight scale.
1509    let psf_peak_index = find_peak_abs(plane.normal_approximation(), shape, |v| v.re, |_| true)
1510        .ok_or(MinorCycleError::InvalidPsfPeak)?;
1511    let psf_peak = plane.normal_approximation()[psf_peak_index].re;
1512    if !psf_peak.is_finite() || psf_peak <= 0.0 {
1513        return Err(MinorCycleError::InvalidPsfPeak);
1514    }
1515    let psf_peak_pixel = plane_pixel(psf_peak_index, shape);
1516
1517    let clark = match controls.algorithm() {
1518        ReconstructionAlgorithm::Hogbom => None,
1519        ReconstructionAlgorithm::Clark => Some(derive_clark_approximation(
1520            plane.normal_approximation(),
1521            shape,
1522            psf_peak_pixel,
1523        )?),
1524        ReconstructionAlgorithm::Multiscale { .. } => None,
1525        ReconstructionAlgorithm::Mtmfs { .. } => {
1526            return Err(MinorCycleError::InvalidNormalStateCatalog);
1527        }
1528        _ => return Err(MinorCycleError::UnsupportedAlgorithm),
1529    };
1530
1531    // Private working copy: authoritative state is never mutated.
1532    let mut residual = Vec::with_capacity(cells);
1533    for value in plane.residual() {
1534        let real = value.re;
1535        if !real.is_finite() {
1536            return Err(MinorCycleError::GeneratedNonfinite);
1537        }
1538        residual.push(real);
1539    }
1540    let noise_rms = controls
1541        .noise_sigma()
1542        .map(|_| {
1543            robust_masked_rms(&residual, shape, base, model_plane, mask).map(|rms| rms / psf_peak)
1544        })
1545        .transpose()?;
1546    let global_threshold = noise_rms
1547        .zip(controls.noise_sigma())
1548        .map_or(controls.threshold(), |(rms, sigma)| {
1549            controls.threshold().max(rms * sigma)
1550        });
1551    let initial_peak = residual
1552        .iter()
1553        .fold(0.0_f64, |peak, value| peak.max(value.abs()))
1554        / psf_peak;
1555    let cycle_threshold = if controls.fixed_cycle_threshold.is_some() {
1556        controls.fixed_cycle_threshold
1557    } else if let Some(cycle) = controls.cycle_threshold {
1558        let psf = plane
1559            .normal_approximation()
1560            .iter()
1561            .map(|value| value.re as f32)
1562            .collect::<Vec<_>>();
1563        let maximum_sidelobe = crate::fitted_psf_sidelobe_fraction(&psf, shape)?;
1564        Some(
1565            initial_peak
1566                * (cycle.factor * maximum_sidelobe)
1567                    .clamp(cycle.minimum_psf_fraction, cycle.maximum_psf_fraction),
1568        )
1569    } else {
1570        None
1571    };
1572    let effective_threshold = cycle_threshold.map_or(global_threshold, |threshold| {
1573        global_threshold.max(threshold)
1574    });
1575    let mut clark_state = clark.map(|approximation| {
1576        let initial_peak = residual
1577            .iter()
1578            .fold(0.0_f64, |peak, value| peak.max(value.abs()))
1579            / psf_peak;
1580        let cutoff = (initial_peak * approximation.maximum_exterior_sidelobe / psf_peak / 3.0)
1581            .max(effective_threshold);
1582        let active = residual
1583            .iter()
1584            .map(|value| value.abs() / psf_peak >= cutoff)
1585            .collect::<Vec<_>>();
1586        ClarkWorkState {
1587            cutoff,
1588            active,
1589            refreshes: 0,
1590        }
1591    });
1592    let multiscale = match controls.algorithm() {
1593        ReconstructionAlgorithm::Multiscale {
1594            scales_px,
1595            small_scale_bias,
1596        } => Some(build_scale_kernels(scales_px, *small_scale_bias)),
1597        _ => None,
1598    };
1599
1600    let mut terms = BTreeMap::<usize, f64>::new();
1601    let mut recorded = Vec::with_capacity(
1602        controls
1603            .component_sequence_limit()
1604            .unwrap_or(0)
1605            .min(controls.actual_iteration_limit()),
1606    );
1607    let mut initial_multiscale_component = None::<f64>;
1608    let has_valid_support = (0..cells).any(|index| {
1609        let pixel = plane_pixel(index, shape);
1610        mask.contains(pixel) && valid_support(base, shape, model_plane, pixel)
1611    });
1612    let mut controller =
1613        MinorCycleController::new(&controls, effective_threshold, has_valid_support);
1614
1615    for _ in 0..controller.iteration_limit() {
1616        let (peak_index, strength, scale_index) = if let Some(kernels) = multiscale.as_ref() {
1617            let candidate = select_multiscale_candidate(
1618                &residual,
1619                plane.normal_approximation(),
1620                shape,
1621                psf_peak_pixel,
1622                base,
1623                model_plane,
1624                mask,
1625                kernels,
1626            )
1627            .ok_or(MinorCycleError::EmptyValidSupport)?;
1628            (
1629                candidate.index,
1630                candidate.strength,
1631                Some(candidate.scale_index),
1632            )
1633        } else {
1634            let peak_index = find_peak_abs(
1635                &residual,
1636                shape,
1637                |value| *value,
1638                |pixel| {
1639                    let index = pixel[0] * shape[1] + pixel[1];
1640                    mask.contains(pixel)
1641                        && valid_support(base, shape, model_plane, pixel)
1642                        && clark_state.as_ref().is_none_or(|state| state.active[index])
1643                },
1644            )
1645            .ok_or(MinorCycleError::EmptyValidSupport)?;
1646            (peak_index, residual[peak_index] / psf_peak, None)
1647        };
1648        let peak_pixel = plane_pixel(peak_index, shape);
1649        if !strength.is_finite() {
1650            return Err(MinorCycleError::GeneratedNonfinite);
1651        }
1652        if scale_index.is_some() {
1653            match initial_multiscale_component {
1654                Some(initial)
1655                    if multiscale_diverged(initial, strength, controller.iterations()) =>
1656                {
1657                    controller.stop(MinorCycleStopReason::MultiscaleDivergence);
1658                    break;
1659                }
1660                None => initial_multiscale_component = Some(strength.abs()),
1661                Some(_) => {}
1662            }
1663        }
1664        // The casacore HOGBOM cleaner stops only when the normalized peak is
1665        // strictly below the threshold (`lattices/LatticeMath/
1666        // LatticeCleaner.tcc`, stopping rule 1: "stop if below threshold",
1667        // tested as `abs(itsStrengthOptimum) < threshold()`), so a peak
1668        // exactly at the threshold still cleans one component. A zero peak
1669        // has no flux to clean and converges trivially.
1670        let flux = controls.gain() * strength;
1671        if !flux.is_finite() {
1672            return Err(MinorCycleError::GeneratedNonfinite);
1673        }
1674        if !controller.admit(
1675            strength,
1676            flux.abs(),
1677            matches!(controls.algorithm(), ReconstructionAlgorithm::Clark),
1678        ) {
1679            break;
1680        }
1681        match (clark, scale_index) {
1682            (_, Some(scale_index)) => subtract_scaled_psf(
1683                &mut residual,
1684                plane.normal_approximation(),
1685                shape,
1686                peak_pixel,
1687                psf_peak_pixel,
1688                &multiscale.as_ref().expect("scale candidate has kernels")[scale_index],
1689                flux,
1690            )?,
1691            (Some(approximation), None) => subtract_psf_patch(
1692                &mut residual,
1693                plane.normal_approximation(),
1694                shape,
1695                peak_pixel,
1696                psf_peak_pixel,
1697                approximation.radius,
1698                flux,
1699            )?,
1700            (None, None) => subtract_psf(
1701                &mut residual,
1702                plane.normal_approximation(),
1703                shape,
1704                peak_pixel,
1705                psf_peak_pixel,
1706                flux,
1707            )?,
1708        }
1709        controller.accepted(flux.abs());
1710        let cell = model_cell(model_plane, shape, peak_pixel)
1711            .expect("a scanned peak pixel lies inside the plane");
1712        let scale_px = if let Some(scale_index) = scale_index {
1713            add_scaled_terms(
1714                &mut terms,
1715                base,
1716                model_plane,
1717                shape,
1718                peak_pixel,
1719                &multiscale.as_ref().expect("scale candidate has kernels")[scale_index],
1720                flux,
1721            );
1722            multiscale.as_ref().expect("scale candidate has kernels")[scale_index].scale_px
1723        } else {
1724            *terms.entry(canonical_flat(base, cell)).or_insert(0.0) += flux;
1725            0.0
1726        };
1727        if let Some(limit) = controls.component_sequence_limit()
1728            && recorded.len() < limit
1729        {
1730            recorded.push(MinorCycleComponent {
1731                cell,
1732                flux,
1733                scale_px,
1734            });
1735        }
1736        if let Some(state) = clark_state.as_mut() {
1737            // SDAlgorithmClarkClean2 configures ClarkCleanLatModel with
1738            // speedup=-1. Its uncertainty limit therefore closes the current
1739            // patch subcycle after an accepted component and recomputes the
1740            // exact residual before selecting the next active set. Preserve
1741            // that behavior explicitly instead of allowing approximate patch
1742            // errors to accumulate across the public cycle boundary.
1743            refresh_point_residual(
1744                &mut residual,
1745                plane.residual(),
1746                plane.normal_approximation(),
1747                shape,
1748                psf_peak_pixel,
1749                base,
1750                &terms,
1751            )?;
1752            state.refreshes += 1;
1753            let global_peak = find_peak_abs(
1754                &residual,
1755                shape,
1756                |value| *value,
1757                |pixel| mask.contains(pixel) && valid_support(base, shape, model_plane, pixel),
1758            )
1759            .ok_or(MinorCycleError::EmptyValidSupport)?;
1760            let global_strength = residual[global_peak].abs() / psf_peak;
1761            let approximation = clark.expect("Clark state has approximation");
1762            state.cutoff =
1763                (global_strength * approximation.maximum_exterior_sidelobe / psf_peak / 3.0)
1764                    .max(effective_threshold);
1765            for (index, active) in state.active.iter_mut().enumerate() {
1766                *active = residual[index].abs() / psf_peak >= state.cutoff;
1767            }
1768        }
1769    }
1770    let (iterations, total_flux, stop_reason) = controller.finish();
1771    let controller_iterations = controls.controller_iterations(iterations, stop_reason);
1772    let clark_refreshes = clark_state.as_ref().map_or(0, |state| state.refreshes);
1773    if multiscale.is_some() && !terms.is_empty() {
1774        // MatrixCleaner uses finite subregions while selecting a bounded
1775        // multiscale component sequence, then finalizes the cycle with its
1776        // full-image FFT residual. Refresh that terminal evidence from the
1777        // accepted model delta so the controller observes the same circular
1778        // normal-operator boundary instead of the last finite work patch.
1779        refresh_circular_residual(
1780            &mut residual,
1781            plane.residual(),
1782            plane.normal_approximation(),
1783            shape,
1784            psf_peak_pixel,
1785            base,
1786            &terms,
1787        )?;
1788    }
1789    let final_peak_flux = find_peak_abs(
1790        &residual,
1791        shape,
1792        |value| *value,
1793        |pixel| mask.contains(pixel) && valid_support(base, shape, model_plane, pixel),
1794    )
1795    .map_or(0.0, |index| residual[index].abs() / psf_peak);
1796
1797    terms.retain(|_, flux| *flux != 0.0);
1798    let delta = if terms.is_empty() {
1799        None
1800    } else {
1801        let deltas = terms
1802            .iter()
1803            .map(|(flat, flux)| {
1804                let cell = base
1805                    .shape()
1806                    .cell_at(*flat)
1807                    .expect("accumulated flat keys stay inside the base shape");
1808                Ok(ModelDeltaTerm::new(cell, ModelValue::new(*flux)?))
1809            })
1810            .collect::<Result<Vec<_>, MinorCycleError>>()?;
1811        Some(lifecycle.compile_delta(base, deltas)?)
1812    };
1813
1814    let evidence_id = minor_cycle_evidence_id(
1815        lifecycle.authority(),
1816        lifecycle.attempt(),
1817        lifecycle.epoch(),
1818        base.generation_id(),
1819        view.completion_id(),
1820        view.content_identity(),
1821        mask,
1822        &controls,
1823        iterations,
1824        controller_iterations,
1825        total_flux,
1826        final_peak_flux,
1827        noise_rms,
1828        effective_threshold,
1829        stop_reason,
1830        clark,
1831        clark_refreshes,
1832    );
1833    Ok(MinorCycleResult {
1834        delta,
1835        evidence: MinorCycleEvidence {
1836            evidence_id,
1837            problem: lifecycle.problem(),
1838            attempt: lifecycle.attempt(),
1839            epoch: lifecycle.epoch(),
1840            input_generation: base.generation_id(),
1841            normal_state_completion: view.completion_id(),
1842            normal_state_content: view.content_identity(),
1843            iterations,
1844            controller_iterations,
1845            total_flux,
1846            initial_peak_flux: initial_peak,
1847            final_peak_flux,
1848            noise_rms,
1849            global_threshold,
1850            effective_threshold,
1851            cycle_threshold,
1852            stop_reason,
1853            clark_approximation: clark,
1854            clark_refreshes,
1855            recorded: (!recorded.is_empty()).then(|| recorded.into_boxed_slice()),
1856        },
1857    })
1858}

1860/// Canonical pixel of one two-dimensional plane-storage index.
1861///
1862/// Normal-state planes are stored x-major (`finish_bound` pushes x outer, y
1863/// inner), so index `p` maps to pixel `[p / H, p % H]`.
1864fn plane_pixel(index: usize, shape: [usize; 2]) -> [usize; 2] {
1865    [index / shape[1], index % shape[1]]
1866}
1867
1868fn model_cell(
1869    model_plane: MinorCycleModelPlane,
1870    shape: [usize; 2],
1871    pixel: [usize; 2],
1872) -> Option<ModelCell> {
1873    if pixel[0] < shape[0] && pixel[1] < shape[1] {
1874        Some(ModelCell::new(
1875            model_plane.domain(),
1876            model_plane.coefficient(),
1877            model_plane.polarization(),
1878            pixel,
1879        ))
1880    } else {
1881        None
1882    }
1883}
1884
1885fn canonical_flat(base: &ModelGeneration, cell: ModelCell) -> usize {
1886    base.shape()
1887        .flat_index(cell)
1888        .expect("component pixels stay inside the base shape")
1889}
1890
1891fn valid_support(
1892    base: &ModelGeneration,
1893    shape: [usize; 2],
1894    model_plane: MinorCycleModelPlane,
1895    pixel: [usize; 2],
1896) -> bool {
1897    let Some(cell) = model_cell(model_plane, shape, pixel) else {
1898        return false;
1899    };
1900    base.samples()[canonical_flat(base, cell)].support() == ModelSupport::Valid
1901}

1903/// Find the maximum-abs real plane value passing `accept`, scanning in
1904/// canonical storage order so ties deterministically keep the first peak.
1905fn find_peak_abs<T>(
1906    plane: &[T],
1907    shape: [usize; 2],
1908    magnitude: impl Fn(&T) -> f64,
1909    accept: impl Fn([usize; 2]) -> bool,
1910) -> Option<usize> {
1911    let mut best: Option<(f64, usize)> = None;
1912    for (index, value) in plane.iter().enumerate() {
1913        let magnitude = magnitude(value).abs();
1914        if best.is_some_and(|(best_magnitude, _)| magnitude <= best_magnitude) {
1915            continue;
1916        }
1917        let pixel = plane_pixel(index, shape);
1918        if accept(pixel) {
1919            best = Some((magnitude, index));
1920        }
1921    }
1922    best.map(|(_, index)| index)
1923}

1925/// Subtract `flux * psf` centered on `peak` from the working residual.
1926fn subtract_psf(
1927    residual: &mut [f64],
1928    psf: &[num_complex::Complex64],
1929    shape: [usize; 2],
1930    peak: [usize; 2],
1931    psf_peak: [usize; 2],
1932    flux: f64,
1933) -> Result<(), MinorCycleError> {
1934    // psf_shifted(x, y) = psf(x - peak + psf_peak, y - peak + psf_peak),
1935    // clipped to the plane exactly like the reference cleaner's subregion.
1936    let x_range = overlap(peak[0], psf_peak[0], shape[0]);
1937    let y_range = overlap(peak[1], psf_peak[1], shape[1]);
1938    for y in y_range.clone() {
1939        for x in x_range.clone() {
1940            let source = [x + psf_peak[0] - peak[0], y + psf_peak[1] - peak[1]];
1941            let index = source[0] * shape[1] + source[1];
1942            let target = x * shape[1] + y;
1943            let updated = residual[target] - flux * psf[index].re;
1944            if !updated.is_finite() {
1945                return Err(MinorCycleError::GeneratedNonfinite);
1946            }
1947            residual[target] = updated;
1948        }
1949    }
1950    Ok(())
1951}

1953/// Subtract one PSF component using the full-image circular FFT convention.
1954fn subtract_psf_circular(
1955    residual: &mut [f64],
1956    psf: &[num_complex::Complex64],
1957    shape: [usize; 2],
1958    peak: [usize; 2],
1959    psf_peak: [usize; 2],
1960    flux: f64,
1961) -> Result<(), MinorCycleError> {
1962    for source_x in 0..shape[0] {
1963        let target_x = (source_x + peak[0] + shape[0] - psf_peak[0]) % shape[0];
1964        for source_y in 0..shape[1] {
1965            let target_y = (source_y + peak[1] + shape[1] - psf_peak[1]) % shape[1];
1966            let source = source_x * shape[1] + source_y;
1967            let target = target_x * shape[1] + target_y;

2437                }

2439                    *value /= volume;
2440                }

2473                -0.533_558_1,
2474                0.231_275_6,

2476            [1.0, 0.821_201_8, 0.207_804_3],

2481            [
2482                0.004_028_559,
2483                -0.036_977_68,

2618        coordinate > border && coordinate < extent.saturating_sub(border).saturating_sub(1)
2619    })
2620}

2630    base: &ModelGeneration,
2631    model_plane: MinorCycleModelPlane,
2632    shape: [usize; 2],
2633    centre: [usize; 2],
2634    kernel: &ScaleKernel,
2635    flux: f64,
2636) {
2637    for (offset, weight) in &kernel.samples {

2643    }

2663/// Derive CASA's Clark patch from the fitted restoring beam.
2664///
2665/// `SDAlgorithmClarkClean2` chooses at least four pixels, otherwise the
2666/// ceiling of the fitted major/minor FWHM in pixels, then requests a
2667/// `3*ncent+1` square capped by the PSF/model shape. The normal-state plane
2668/// uses unit pixel coordinates here, so the shared CASA-style beam fitter can
2669/// supply that same width without crossing the reconstruction-owner boundary.
2670fn derive_clark_approximation(
2671    psf: &[num_complex::Complex64],
2672    shape: [usize; 2],
2673    peak: [usize; 2],
2674) -> Result<ClarkApproximation, crate::PsfBeamFitError> {
2675    let real_psf = psf.iter().map(|value| value.re as f32).collect::<Vec<_>>();
2676    let beam =
2677        crate::fit_restoring_beam(&real_psf, shape, [1.0, 1.0], crate::DEFAULT_PSF_FIT_CUTOFF)?;
2678    let central_width = 4_usize
2679        .max(beam.major_fwhm_rad().ceil() as usize)
2680        .max(beam.minor_fwhm_rad().ceil() as usize);
2681    let requested = central_width.saturating_mul(3).saturating_add(1);
2682    let radius = [requested.min(shape[0]) / 2, requested.min(shape[1]) / 2];
2683    let maximum_exterior_sidelobe = psf
2684        .iter()
2685        .enumerate()
2686        .filter(|(index, _)| {
2687            let pixel = plane_pixel(*index, shape);
2688            pixel[0].abs_diff(peak[0]) > radius[0] || pixel[1].abs_diff(peak[1]) > radius[1]
2689        })
2690        .fold(0.0_f64, |maximum, (_, value)| maximum.max(value.re.abs()));
2691    Ok(ClarkApproximation {
2692        radius,
2693        maximum_exterior_sidelobe,
2694    })
2695}

2697fn subtract_psf_patch(
2698    residual: &mut [f64],
2699    psf: &[num_complex::Complex64],
2700    shape: [usize; 2],
2701    peak: [usize; 2],
2702    psf_peak: [usize; 2],
2703    radius: [usize; 2],
2704    flux: f64,
2705) -> Result<(), MinorCycleError> {
2706    let x_range = overlap(peak[0], psf_peak[0], shape[0]);
2707    let y_range = overlap(peak[1], psf_peak[1], shape[1]);
2708    for y in y_range {
2709        for x in x_range.clone() {
2710            let source = [x + psf_peak[0] - peak[0], y + psf_peak[1] - peak[1]];
2711            if source[0].abs_diff(psf_peak[0]) > radius[0]
2712                || source[1].abs_diff(psf_peak[1]) > radius[1]
2713            {
2714                continue;
2715            }
2716            let target = x * shape[1] + y;
2717            let updated = residual[target] - flux * psf[source[0] * shape[1] + source[1]].re;
2718            if !updated.is_finite() {
2719                return Err(MinorCycleError::GeneratedNonfinite);
2720            }
2721            residual[target] = updated;
2722        }
2723    }
2724    Ok(())
2725}

2727/// Inclusive-clipped axis range where a PSF centered at `psf_peak` overlaps
2728/// the plane when recentered on `peak`.
2729fn overlap(peak: usize, psf_peak: usize, length: usize) -> std::ops::Range<usize> {
2730    let shift = psf_peak as isize - peak as isize;
2731    let low = (-shift).max(0) as usize;
2732    let high_exclusive = (length as isize - shift)
2733        .min(length as isize)
2734        .max(low as isize) as usize;
2735    low..high_exclusive
2736}

2738#[allow(clippy::too_many_arguments)]
2739fn minor_cycle_evidence_id(
2740    authority: LogicalIdentity,
2741    attempt: ModelExecutionAttemptId,
2742    epoch: u64,
2743    input_generation: ModelGenerationId,
2744    normal_state_completion: FinalNormalStateCompletionId,
2745    normal_state_content: LogicalIdentity,
2746    mask: &ReconstructionMask,
2747    controls: &MinorCycleProgram,
2748    iterations: usize,
2749    controller_iterations: usize,
2750    total_flux: f64,
2751    final_peak_flux: f64,
2752    noise_rms: Option<f64>,
2753    effective_threshold: f64,
2754    stop_reason: MinorCycleStopReason,
2755    clark: Option<ClarkApproximation>,
2756    clark_refreshes: usize,
2757) -> MinorCycleEvidenceId {
2758    let mut encoder = Encoder::new(MINOR_CYCLE_EVIDENCE_DOMAIN, MINOR_CYCLE_EVIDENCE_VERSION);
2759    encoder.identity(authority.as_bytes());
2760    encoder.identity(attempt.identity().as_bytes());
2761    encoder.u64(epoch);
2762    encoder.identity(input_generation.as_bytes());
2763    encoder.identity(normal_state_completion.as_bytes());
2764    encoder.identity(normal_state_content.as_bytes());
2765    encoder.identity(mask.generation_id().as_bytes());
2766    encoder.usize(controls.model_plane().domain());
2767    encoder.usize(controls.model_plane().coefficient());
2768    encoder.usize(controls.model_plane().polarization());
2769    match controls.algorithm() {
2770        ReconstructionAlgorithm::Hogbom => encoder.u8(0),
2771        ReconstructionAlgorithm::Clark => encoder.u8(1),
2772        ReconstructionAlgorithm::Multiscale {
2773            scales_px,
2774            small_scale_bias,
2775        } => {
2776            encoder.u8(2);
2777            encoder.usize(scales_px.len());
2778            for scale in scales_px {
2779                encoder.u64(crate::canonical_f64_bits(*scale));
2780            }
2781            encoder.u64(crate::canonical_f64_bits(*small_scale_bias));
2782        }
2783        ReconstructionAlgorithm::Mtmfs {
2784            scales_px,
2785            small_scale_bias,
2786        } => {
2787            encoder.u8(3);
2788            encoder.usize(scales_px.len());
2789            for scale in scales_px {
2790                encoder.u64(crate::canonical_f64_bits(*scale));
2791            }
2792            encoder.u64(crate::canonical_f64_bits(*small_scale_bias));
2793        }
2794        _ => unreachable!("minor-cycle programs admit only implemented solvers"),
2795    }
2796    encoder.u64(crate::canonical_f64_bits(controls.gain()));
2797    encoder.u64(crate::canonical_f64_bits(controls.threshold()));
2798    match controls.noise_sigma() {
2799        Some(sigma) => {
2800            encoder.u8(1);
2801            encoder.u64(crate::canonical_f64_bits(sigma));
2802        }
2803        None => encoder.u8(0),
2804    }
2805    encoder.usize(controls.max_iterations());
2806    encoder.u8(match controls.hogbom_iteration_accounting() {
2807        HogbomIterationAccounting::Strict => 0,
2808        HogbomIterationAccounting::CasaInclusive => 1,
2809    });
2810    match controls.validity() {
2811        MinorCycleValidity::Exact => encoder.u8(0),
2812        MinorCycleValidity::Bounded {
2813            maximum_absolute_update,
2814        } => {
2815            encoder.u8(1);
2816            encoder.u64(crate::canonical_f64_bits(maximum_absolute_update));
2817        }
2818    }
2819    match controls.cycle_threshold {
2820        Some(cycle) => {
2821            encoder.u8(1);
2822            encoder.u64(crate::canonical_f64_bits(cycle.factor));
2823            encoder.u64(crate::canonical_f64_bits(cycle.minimum_psf_fraction));
2824            encoder.u64(crate::canonical_f64_bits(cycle.maximum_psf_fraction));
2825        }
2826        None => encoder.u8(0),
2827    }
2828    match controls.fixed_cycle_threshold {
2829        Some(threshold) => {
2830            encoder.u8(1);
2831            encoder.u64(crate::canonical_f64_bits(threshold));
2832        }
2833        None => encoder.u8(0),
2834    }
2835    match controls.component_sequence_limit() {
2836        None => encoder.u8(0),
2837        Some(limit) => {
2838            encoder.u8(1);
2839            encoder.usize(limit);
2840        }
2841    }
2842    encoder.usize(iterations);
2843    encoder.usize(controller_iterations);
2844    encoder.u64(crate::canonical_f64_bits(total_flux));
2845    encoder.u64(crate::canonical_f64_bits(final_peak_flux));
2846    match noise_rms {
2847        Some(rms) => {
2848            encoder.u8(1);
2849            encoder.u64(crate::canonical_f64_bits(rms));
2850        }
2851        None => encoder.u8(0),
2852    }
2853    encoder.u64(crate::canonical_f64_bits(effective_threshold));
2854    encoder.u8(match stop_reason {
2855        MinorCycleStopReason::ThresholdReached => 0,
2856        MinorCycleStopReason::IterationBound => 1,
2857        MinorCycleStopReason::StalenessBound => 2,
2858        MinorCycleStopReason::MultiscaleDivergence => 3,
2859    });
2860    match clark {
2861        None => encoder.u8(0),
2862        Some(approximation) => {
2863            encoder.u8(1);
2864            encoder.usize(approximation.radius[0]);
2865            encoder.usize(approximation.radius[1]);
2866            encoder.u64(crate::canonical_f64_bits(
2867                approximation.maximum_exterior_sidelobe,
2868            ));
2869        }
2870    }
2871    encoder.usize(clark_refreshes);
2872    MinorCycleEvidenceId(LogicalIdentity::from_sha256(encoder.finish()))
2873}
