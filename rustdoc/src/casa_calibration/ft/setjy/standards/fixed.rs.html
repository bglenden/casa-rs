2//! Published fixed flux polynomials used by CASA's FluxStdsQS2.
3
4#[derive(Clone, Copy)]
5pub enum FixedScale {
6    Baars,
7    Perley90,
8    PerleyTaylor95,
9    PerleyTaylor99,
10    PerleyButler2010,
11    StevensReynolds2016,
12}
13
14pub struct FixedSpectrum {
15    reference_hz: f64,
16    coefficients: &'static [f32],
17    upper: Option<(f64, &'static [f32])>,
18}
19
20const PKS_1934_LOW: &[f32] = &[1.170418, 0.248618, -1.649694, 0.605334];
21const PKS_1934_HIGH: &[f32] = &[-2.5739, 10.0707, -10.0595, 2.9372];
22const PKS_1934_STEVENS_HIGH: &[f32] = &[1.7581, -1.3763];

75impl FixedSpectrum {
76    pub(super) fn flux(&self, frequency_hz: f64) -> f64 {
77        let coefficients = self
78            .upper
79            .filter(|(limit, _)| frequency_hz > *limit)
80            .map_or(self.coefficients, |(_, coefficients)| coefficients);
81        let log_frequency = (frequency_hz / self.reference_hz).log10();
82        10_f64.powf(super::log_polynomial(coefficients, log_frequency))
83    }
84}
