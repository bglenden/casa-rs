2//! A component's integrated flux, CASA's `Flux<Double>`.
3
4use casa_types::quanta::Unit;
5use num_complex::Complex64;
6
7use crate::component_type::Polarisation;
8use crate::error::ComponentError;
9
10/// The flux units CASA accepts, by dimension (`FluxRep::_allowedUnits`).
11const ALLOWED_FLUX_UNITS: [&str; 3] = ["Jy", "Jy.km/s", "K.rad2"];
12
13/// The integrated flux of a component: four complex values in one unit and
14/// one polarisation representation, with their errors.
15///
16/// This is CASA's `Flux<Double>` (`FluxRep`). The four values are, in order:
17///
18/// | representation | values |
19/// |---|---|
20/// | [`Polarisation::Stokes`] | I, Q, U, V (real; the imaginary parts are zero) |
21/// | [`Polarisation::Linear`] | XX, XY, YX, YY |
22/// | [`Polarisation::Circular`] | RR, RL, LR, LL |
23///
24/// A component list stores them as written, in the `Flux` column, with the
25/// unit's name in `Flux_Unit`, the representation's name in
26/// `Flux_Polarisation` and the errors in `Flux_Error`. Nothing is converted
27/// between representations or units on the way to or from a table.
28///
29/// The unit must have the dimensions of `Jy`, `Jy.km/s` or `K.rad2`, the units
30/// CASA's flux accepts; its name is kept as given (`mJy` stays `mJy`).
31#[derive(Debug, Clone, PartialEq)]
32pub struct Flux {
33    values: [Complex64; 4],
34    unit: Unit,
35    polarisation: Polarisation,
36    errors: [Complex64; 4],
37}
38
39impl Flux {
40    /// A flux with zero errors.
41    ///
42    /// # Errors
43    ///
44    /// [`ComponentError::Unit`] when `unit` does not parse, and
45    /// [`ComponentError::FluxUnit`] when its dimensions are not those of a
46    /// flux CASA accepts.
47    pub fn new(
48        values: [Complex64; 4],
49        unit: &str,
50        polarisation: Polarisation,
51    ) -> Result<Self, ComponentError> {
52        Ok(Self {
53            values,
54            unit: flux_unit(unit)?,
55            polarisation,
56            errors: [Complex64::new(0.0, 0.0); 4],
57        })
58    }
59
60    /// A Stokes flux in Jy, with zero errors: CASA's `Flux<Double>(i, q, u, v)`.
61    pub fn stokes_jy(i: f64, q: f64, u: f64, v: f64) -> Self {
62        Self {
63            values: [i, q, u, v].map(|value| Complex64::new(value, 0.0)),
64            unit: Unit::new("Jy").expect("Jy is a unit"),
65            polarisation: Polarisation::Stokes,
66            errors: [Complex64::new(0.0, 0.0); 4],
67        }
68    }
69
70    /// This flux with its errors replaced (`FluxRep::setErrors`).
71    pub fn with_errors(mut self, errors: [Complex64; 4]) -> Self {
72        self.errors = errors;
73        self
74    }
75
76    /// The four values in [`Self::polarisation`]'s order.
77    pub fn values(&self) -> &[Complex64; 4] {
78        &self.values
79    }
80
81    /// The unit of the values and errors.
82    pub fn unit(&self) -> &Unit {
83        &self.unit
84    }
85
86    /// The polarisation representation of the values.
87    pub fn polarisation(&self) -> Polarisation {
88        self.polarisation
89    }
90
91    /// The errors of the four values, in the same order and unit.
92    pub fn errors(&self) -> &[Complex64; 4] {
93        &self.errors
94    }
95
96    /// Replace the values, keeping the unit, representation and errors
97    /// (`FluxRep::setValue`).
98    pub fn set_values(&mut self, values: [Complex64; 4]) {
99        self.values = values;
100    }
101
102    /// Replace the errors (`FluxRep::setErrors`).
103    pub fn set_errors(&mut self, errors: [Complex64; 4]) {
104        self.errors = errors;
105    }
106}
107
108/// Parse a flux unit and check its dimensions as `FluxRep` does.
109fn flux_unit(name: &str) -> Result<Unit, ComponentError> {
110    let unit = Unit::new(name).map_err(|error| ComponentError::Unit {
111        what: "flux",
112        unit: name.to_string(),
113        reason: error.to_string(),
114    })?;
115    let allowed = ALLOWED_FLUX_UNITS
116        .iter()
117        .any(|allowed| unit.conformant(&Unit::new(allowed).expect("allowed flux units parse")));
118    if allowed {
119        Ok(unit)
120    } else {
121        Err(ComponentError::FluxUnit(name.to_string()))
122    }
123}

125#[cfg(test)]
126mod tests {
127    use super::*;
128
129    #[test]
130    fn flux_units_need_flux_dimensions_and_keep_their_names() {
131        let flux = Flux::new([Complex64::new(1.0, 0.0); 4], "mJy", Polarisation::Linear)
132            .expect("mJy is a flux unit");
133        assert_eq!(flux.unit().name(), "mJy");
134        assert!(
135            Flux::new(
136                [Complex64::new(1.0, 0.0); 4],
137                "K.rad.rad",
138                Polarisation::Stokes
139            )
140            .is_ok()
141        );
142        assert!(
143            Flux::new(
144                [Complex64::new(1.0, 0.0); 4],
145                "Jy.km/s",
146                Polarisation::Stokes
147            )
148            .is_ok()
149        );
150        assert_eq!(
151            Flux::new([Complex64::new(1.0, 0.0); 4], "m", Polarisation::Stokes),
152            Err(ComponentError::FluxUnit("m".to_string()))
153        );
154        assert!(matches!(
155            Flux::new(
156                [Complex64::new(1.0, 0.0); 4],
157                "notaunit!",
158                Polarisation::Stokes
159            ),
160            Err(ComponentError::Unit { .. })
161        ));
162    }
