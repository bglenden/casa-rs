16/// One application-catalog entry: its identity and how to launch it.
17#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
18pub(crate) struct ApplicationView {
19    pub(crate) id: String,
20    pub(crate) kind: String,
21    pub(crate) category: String,
22    pub(crate) display_name: String,
23    pub(crate) executable: String,
24    pub(crate) cargo_package: String,
25    pub(crate) override_env: String,
26    pub(crate) dataset_kinds: Vec<String>,
27    pub(crate) include_in_suite: bool,
28}
29
30/// Every application in the catalog, with the surface names of its task.
31pub(crate) fn applications() -> Result<Vec<ApplicationView>, String> {
32    let catalog = builtin_application_catalog()?;
33    catalog
34        .applications
35        .iter()
36        .map(|application| {
37            let surface = application
38                .surface_bundle()
39                .map_err(|reason| format!("resolve application {}: {reason}", application.id))?;
40            let kind = serde_json::to_value(application.kind)
41                .map_err(|error| format!("project application kind: {error}"))?
42                .as_str()
43                .map(str::to_owned)
44                .ok_or_else(|| {
45                    format!(
46                        "application {} has a non-string enum projection",
47                        application.id
48                    )
49                })?;
50            Ok(ApplicationView {
51                id: application.id.clone(),
52                kind,
53                display_name: surface
54                    .as_ref()
55                    .map_or("CASA-RS", |bundle| bundle.surface.display_name())
56                    .to_string(),
57                category: surface
58                    .as_ref()
59                    .map_or("Launcher", |bundle| bundle.surface.category())
60                    .to_string(),
61                executable: application.launch.executable.clone(),
62                cargo_package: application.launch.cargo_package.clone(),
63                override_env: application.launch.override_env.clone(),
64                dataset_kinds: application.dataset_kinds.clone(),
65                include_in_suite: application.include_in_suite,
66            })
67        })
68        .collect()
69}

85/// A successful task's summary, products and diagnostics.

94/// Decode one successful provider result into its contract-declared
95/// products. `values` are the task's parameters as display text.
96pub(crate) fn completion(
97    surface_id: &str,
98    stdout: &str,
99    workspace: &Path,
100    values: Vec<(String, String)>,
101) -> Result<CompletionView, String> {
102    let bundle = builtin_surface_bundle(surface_id)?;
103    let values = values
104        .into_iter()
105        .map(|(name, value)| TaskOutputValue { name, value })
106        .collect::<Vec<_>>();
107    let completion = decode_task_completion(&bundle, stdout, workspace, &values)
108        .map_err(|error| error.to_string())?;
109    let products = completion
110        .products
111        .into_iter()
112        .map(|product| ProductView {
113            id: product.id,
114            role: match product.role {
115                RunProductRole::Primary => "PRIMARY",
116                RunProductRole::Auxiliary => "AUXILIARY",
117                RunProductRole::Preview => "PREVIEW",
118            },
119            resource_kind: match product.resource_kind {
120                RunProductKind::MeasurementSet => "MEASUREMENT_SET",
121                RunProductKind::CasaImage => "CASA_IMAGE",
122                RunProductKind::CasaTable => "CASA_TABLE",
123                RunProductKind::FitsImage => "FITS_IMAGE",
124                RunProductKind::File => "FILE",
125            },
126            label: product.label,
127            path: product.path.to_string_lossy().into_owned(),
128            exists: product.exists,
129            preview_path: product
130                .preview_path
131                .map(|path| path.to_string_lossy().into_owned()),
132            preview_exists: product.preview_exists,
133            diagnostic: product.diagnostic,
134        })
135        .collect();
136    Ok(CompletionView {
137        surface_id: completion.surface_id,
138        summary: completion.summary,
139        products,
140        diagnostics: completion.diagnostics,
141    })
142}

144/// The application catalog's launch entries.
145#[pyfunction]
146pub(crate) fn application_catalog(py: Python<'_>) -> PyResult<Bound<'_, PyAny>> {
147    let applications = applications().map_err(|reason| reason_error(py, reason))?;
148    to_python(py, &applications)
149}
150
151/// A successful task's products. Releases the GIL while it checks them.
152#[pyfunction]
153pub(crate) fn task_completion<'py>(
154    py: Python<'py>,
155    surface_id: &str,
156    stdout: &str,
157    workspace: std::path::PathBuf,
158    values: Vec<(String, String)>,
159) -> PyResult<Bound<'py, PyAny>> {
160    let completion = py
161        .detach(|| completion(surface_id, stdout, &workspace, values))
162        .map_err(|reason| reason_error(py, reason))?;
163    to_python(py, &completion)
164}
