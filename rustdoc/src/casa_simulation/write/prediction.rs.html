2//! The prediction stage of `close`: the model DATA of each write batch's
3//! rows, degridded by the imaging runtime's block predictor on the worker
4//! team.
5//!
6//! A scan's rows are predicted from every `predict` step covering it, their
7//! predictions summed (one [`PredictionDomain`] per step, as CASA's mappers
8//! sum their degrids), each row rotated to each step's tangent point by
9//! [`GridFtProjection`] as CASA's `GridFT` rotates it. Scans with the same
10//! steps and field share a segment, and one predictor serves a segment; it
11//! is rebuilt when the batches reach a scan of another segment. A run of a
12//! segment's rows is predicted in blocks of at most a fixed number of rows,
13//! straight into the batch's DATA, which a scan no step covers leaves zero.
14//!
15//! The rows are as `observe` writes them: the row flag is FLAG_ROW, every
16//! sample is unflagged with unit weight, and an autocorrelation predicts the
17//! model's zero spacing (`GridFT`'s `usezero`). On the CPU a row's
18//! prediction depends only on the row and the steps, not on the workers,
19//! the blocks or the batches.
20

37/// Bytes of the native rows of one prediction block, about: blocks of more
38/// rows gain nothing, as the predictor's chunks are smaller still.
39const BLOCK_BYTES: u64 = 16 << 20;
40
41/// The scans with the same predict steps and field: the domains a
42/// predictor sums and each row's rotation to each domain's tangent point.
43struct Segment<'s> {
44    domains: Vec<PredictionDomain<'s>>,
45    projections: Vec<GridFtProjection>,
46}
47
48/// The prediction stage's state across batches.
49pub(super) struct PredictionStage<'s> {
50    backend: BackendChoice,
51    correlations: usize,
52    channels: usize,
53    segments: Vec<Segment<'s>>,
54    /// Each scan's segment; `None` for a scan no step covers.
55    scan_segments: Vec<Option<usize>>,
56    /// Each spectral window's channel frequencies.
57    frequencies: Vec<Vec<f64>>,
58    /// Bytes of every model grid the steps hold, which a device copies.
59    grid_bytes: u64,
60    /// Rows of a block, once started.
61    block_rows: usize,
62    workers: usize,
63    block: NativeBlock,
64    projections: Vec<DomainProjection>,
65    /// The predictor of the segment last predicted.
66    predictor: Option<(usize, BlockPredictor<'s>)>,
67    channel_indices: Vec<u32>,
68    correlation_indices: Vec<u32>,
69}
70
71impl<'s> PredictionStage<'s> {
72    /// The stage for `session`'s predict steps on cells of `(correlations,
73    /// channels)`, on `backend`; `None` when no step predicts.
74    ///
75    /// # Errors
76    ///
77    /// A covered scan's field without grids, which `predict` prepares for
78    /// every field it covers.
79    pub(super) fn new(
80        session: &'s Simulator,
81        (correlations, channels): (usize, usize),
82        backend: BackendChoice,
83    ) -> SimulatorResult<Option<Self>> {
84        let plan = session.data_plan();
85        let predictions: Vec<_> = plan
86            .predictions()
87            .map(|(prediction, _)| prediction)
88            .collect();
89        if predictions.is_empty() {
90            return Ok(None);
91        }
92        let mut keys = HashMap::new();
93        let mut segments = Vec::new();
94        let mut scan_segments = Vec::with_capacity(session.scans().len());
95        for (index, scan) in session.scans().iter().enumerate() {
96            let steps = plan.predictions_for_scan(index);
97            if steps.is_empty() {
98                scan_segments.push(None);
99                continue;
100            }
101            let key = (steps, scan.field_id);
102            let segment = match keys.get(&key) {
103                Some(segment) => *segment,
104                None => {
105                    let mut domains = Vec::with_capacity(key.0.len());
106                    let mut projections = Vec::with_capacity(key.0.len());
107                    for step in key.0.clone() {
108                        let prediction = predictions[step];
109                        let model = prediction.grids_for(scan.field_id).ok_or_else(|| {
110                            SimulatorError::raised(format!(
111                                "predict prepared no grids for field {}",
112                                scan.field_id
113                            ))
114                        })?;
115                        domains.push(PredictionDomain {
116                            operator: &prediction.model.operator,
117                            resampler: &prediction.model.resampler,
118                            model,
119                        });
120                        projections.push(GridFtProjection::between_j2000_directions(
121                            scan.phase_center_rad,
122                            prediction.tangent_rad,
123                        ));
124                    }
125                    segments.push(Segment {
126                        domains,
127                        projections,
128                    });
129                    keys.insert(key, segments.len() - 1);
130                    segments.len() - 1
131                }
132            };
133            scan_segments.push(Some(segment));
134        }
135        let frequencies = session
136            .spectral_windows()
137            .iter()
138            .map(|window| window.channel_frequencies_hz())
139            .collect();
140        let grid_bytes = predictions
141            .iter()
142            .map(|prediction| prediction.grid_bytes())
143            .sum();
144        Ok(Some(Self {
145            backend,
146            correlations,
147            channels,
148            segments,
149            scan_segments,
150            frequencies,
151            grid_bytes,
152            block_rows: 0,
153            workers: 0,
154            block: NativeBlock::default(),
155            projections: Vec::new(),
156            predictor: None,
157            channel_indices: (0..channels as u32).collect(),
158            correlation_indices: (0..correlations as u32).collect(),
159        }))
160    }
161
162    /// The backend the stage predicts on.
163    pub(super) fn backend(&self) -> BackendChoice {
164        self.backend
165    }
166
167    /// The most domains a segment sums.
168    fn largest_domains(&self) -> usize {
169        self.segments
170            .iter()
171            .map(|segment| segment.domains.len())
172            .max()
173            .unwrap_or(0)
174    }
175
176    /// The block of `rows` rows projected on `domains` domains.
177    fn block_shape(&self, rows: usize, domains: usize) -> BlockShape {
178        BlockShape {
179            rows,
180            spectrum: RowSpectrum::evenly_spaced(self.channels),
181            correlations: self.correlations,
182            domains,
183        }
184    }
185
186    /// Rows of a block for write batches of `batch_rows` rows: as many as
187    /// [`BLOCK_BYTES`] holds, at most a batch and at least one.
188    fn block_rows(&self, batch_rows: usize) -> usize {
189        let row = self.block_shape(1, self.largest_domains()).bytes().max(1);
190        usize::try_from(BLOCK_BYTES / row)
191            .unwrap_or(usize::MAX)
192            .clamp(1, batch_rows.max(1))
193    }
194
195    /// Bytes the stage holds while `close` writes batches of `batch_rows`
196    /// rows with `workers` workers: the predictor of the segment that needs
197    /// most ([`BlockPredictor::bytes`]), one block of native rows and each
198    /// row's projections, and on Metal the device copy of every model grid
199    /// ([`casa_imaging_operator::PreparedModelGrids::device_copy`]).
200    pub(super) fn bytes(&self, batch_rows: usize, workers: usize) -> u64 {
201        let rows = self.block_rows(batch_rows);
202        let predictor = self
203            .segments
204            .iter()
205            .map(|segment| {
206                BlockPredictor::bytes(
207                    &segment.domains,
208                    self.backend,
209                    self.block_shape(rows, segment.domains.len()),
210                    workers,
211                )
212            })
213            .max()
214            .unwrap_or(0);
215        let domains = self.largest_domains();
216        let block = self.block_shape(rows, domains).bytes();
217        let projections = (domains * size_of::<DomainProjection>()) as u64;
218        let device = match self.backend {
219            BackendChoice::Cpu => 0,
220            BackendChoice::Metal => self.grid_bytes,
221        };
222        predictor + block + projections + device
223    }
224
225    /// Allocate the block for write batches of `batch_rows` rows, predicted
226    /// with `workers` workers, as [`Self::bytes`] charged it.
227    pub(super) fn start(&mut self, batch_rows: usize, workers: usize) {
228        self.block_rows = self.block_rows(batch_rows);
229        self.workers = workers.max(1);
230        let domains = self.largest_domains();
231        self.block = NativeBlock::with_capacity(self.block_shape(self.block_rows, domains));
232        self.projections = Vec::with_capacity(domains);
233    }
234
235    /// Predict each run of `runs`, (scan, first row, rows) of `rows`, into
236    /// `data` (`correlations × channels` samples a row, the rows of `rows`
237    /// from the first); the batch's first row is MAIN row `first_row`. Runs
238    /// of scans no step covers are left as they are.
239    pub(super) fn predict(
240        &mut self,
241        team: &WorkerTeam,
242        scans: &[Scan],
243        rows: &[RowGeometry],
244        runs: &[(usize, usize, usize)],
245        first_row: u64,
246        data: &mut [Complex32],
247    ) -> SimulatorResult<()> {
248        let samples = self.correlations * self.channels;
249        for &(scan, start, count) in runs {
250            let Some(segment) = self.scan_segments[scan] else {
251                continue;
252            };
253            self.select(segment)?;
254            let scan = &scans[scan];
255            let end = start + count;
256            let mut block_start = start;
257            while block_start < end {
258                let block_end = (block_start + self.block_rows).min(end);
259                self.fill_block(
260                    segment,
261                    scan,
262                    &rows[block_start..block_end],
263                    first_row + block_start as u64,
264                );
265                let Some((_, predictor)) = &mut self.predictor else {
266                    unreachable!("selected above");
267                };
268                predictor
269                    .predict_into(
270                        &self.block,
271                        team,
272                        &mut data[block_start * samples..block_end * samples],
273                    )
274                    .map_err(|error| SimulatorError::raised(format!("predict: {error}")))?;
275                block_start = block_end;
276            }
277        }
278        Ok(())
279    }
280
281    /// Make the predictor `segment`'s, unless it is.
282    fn select(&mut self, segment: usize) -> SimulatorResult<()> {
283        if self
284            .predictor
285            .as_ref()
286            .is_some_and(|(current, _)| *current == segment)
287        {
288            return Ok(());
289        }
290        // The previous predictor is dropped first, so at most one is held.
291        self.predictor = None;
292        let domains = &self.segments[segment].domains;
293        let predictor = BlockPredictor::new(
294            domains,
295            self.backend,
296            self.block_shape(self.block_rows, domains.len()),
297            self.workers,
298            false,
299        )
300        .map_err(|error| SimulatorError::raised(format!("the predictor: {error}")))?;
301        self.predictor = Some((segment, predictor));
302        Ok(())
303    }
304
305    /// Fill the block with `rows` of `scan`, the first MAIN row
306    /// `first_row`, projected on `segment`'s domains.
307    fn fill_block(&mut self, segment: usize, scan: &Scan, rows: &[RowGeometry], first_row: u64) {
308        let segment = &self.segments[segment];
309        let frequencies = &self.frequencies[scan.spectral_window_id];
310        self.block.reset(
311            segment.domains.len(),
312            &self.channel_indices,
313            &self.correlation_indices,
314        );
315        for (offset, row) in rows.iter().enumerate() {
316            self.projections.clear();
317            self.projections
318                .extend(segment.projections.iter().map(|projection| {
319                    let (uvw_m, phase_shift_m) = projection.apply(row.uvw_m);
320                    DomainProjection {
321                        uvw_m,
322                        phase_shift_m,
323                        pointing_offset_rad: [0.0; 2],
324                    }
325                }));
326            let header = NativeRowHeader {
327                row_flag: row.flagged,
328                context: RowContext {
329                    original_uvw_m: Some(row.uvw_m),
330                    time_s: 0.0,
331                    antennas: [row.antenna1 as u32, row.antenna2 as u32],
332                    antenna_types: [0, 0],
333                    parallactic_angle_rad: [0.0, 0.0],
334                    field: scan.field_id as u32,
335                    spectral_window: scan.spectral_window_id as u32,
336                },
337                address: RowAddress {
338                    physical_row: first_row + offset as u64,
339                    data_description: scan.spectral_window_id as i32,
340                },
341            };
342            let pushed = self.block.push_row_with(
343                header,
344                &self.projections,
345                frequencies,
346                |_, weights, _| {
347                    weights.fill(1.0);
348                    Ok::<_, Infallible>(())
349                },
350            );
351            match pushed {
352                Ok(()) => {}
353                Err(never) => match never {},
354            }
355        }
356    }
