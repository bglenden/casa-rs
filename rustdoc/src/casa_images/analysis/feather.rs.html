25/// Feather high- and low-resolution images on the high-resolution grid.
26pub fn feather(request: &FeatherRequest) -> Result<FeatherSummary, ImageError> {
27    let high = AnyPagedImage::open(&request.highres)?;
28    let low = AnyPagedImage::open(&request.lowres)?;
29    match (&high, &low) {
30        (AnyPagedImage::Float32(high), AnyPagedImage::Float32(low)) => {
31            feather_typed(high, low, request)
32        }
33        _ => Err(ImageError::InvalidMetadata(
34            "feather tutorial support currently requires f32 high/low images".to_string(),
35        )),
36    }
37}

39fn feather_typed(
40    high: &PagedImage<f32>,
41    low: &PagedImage<f32>,
42    request: &FeatherRequest,
43) -> Result<FeatherSummary, ImageError> {
44    if request.imagename.as_os_str().is_empty() {
45        return Err(invalid("feather requires an output image name"));
46    }
47    let destination = std::path::absolute(&request.imagename).map_err(output_io)?;
48    check_output(request, &destination)?;
49    let high_shape = high.shape().to_vec();
50    let high_info = high.image_info()?;
51    let low_info = low.image_info()?;
52    let high_beams = &high_info.beam_set;
53    let low_beams = &low_info.beam_set;
54    if high_beams.is_empty() || low_beams.is_empty() {
55        return Err(invalid("feather requires restoring beams on both images"));
56    }
57    let mut template = RegridTemplate::of(high);
58    let low_regridded = if high_beams.is_multi() || low_beams.is_multi() {
59        // CASA pairs channel and polarization planes by pixel index before it
60        // regrids the low-resolution sky plane. Retain those low-image axes.
61        let paired_shape = pair_nonspatial_axes(&mut template, low)?;
62        let paired = SubImage::new(low, vec![0; low.ndim()], paired_shape)?;
63        regrid(&paired, &template, RegridMethod::Linear)?
64    } else {
65        regrid(low, &template, RegridMethod::Linear)?
66    };
67    let Regridded {
68        data: mut low_data,
69        mask: low_mask,
70        ..
71    } = low_regridded;
72    let origin = vec![0; high.ndim()];
73    let mut high_data = high.get_slice(&origin, &high_shape)?;
74    let high_mask = high.get_mask_slice(&origin, &high_shape, &vec![1; high.ndim()])?;
75    let low_mask_name = low.default_mask_name();
76    let low_mask = low_mask_name.as_ref().map(|_| low_mask);
77    // CASA copies each default mask and replaces its invalid samples with zero
78    // before they can contribute to the Fourier transform.
79    clear_masked_pixels(&mut high_data, high_mask.as_ref());
80    clear_masked_pixels(&mut low_data, low_mask.as_ref());
81    if high_shape.len() < 2 {
82        return Err(ImageError::InvalidMetadata(
83            "feather requires at least two direction axes".to_string(),
84        ));
85    }
86    // Feathered in the direction plane, wherever its axes are.
87    let direction = direction_plane_axes(high)?;
88    let planes = PlaneIndex::new(&high_shape, direction);
89    let nx = high_shape[direction[0]];
90    let ny = high_shape[direction[1]];
91    let common = high_beams
92        .single_beam()
93        .zip(low_beams.single_beam())
94        .map(|beams| plane_transfer([nx, ny], beams, high.coordinates(), 0, request))
95        .transpose()?;
96    let beam_axes =
97        [CoordinateType::Spectral, CoordinateType::Stokes].map(|kind| high.find_axis(kind));
98    let mut output_data = ArrayD::<f32>::zeros(IxDyn(&high_shape).f());
99    let mut output_mask = ArrayD::<bool>::from_elem(IxDyn(&high_shape).f(), true);
100    for plane in 0..planes.count() {
101        let plane_filter;
102        let transfer = if let Some(common) = &common {
103            common
104        } else {
105            let [channel, stokes] =
106                beam_axes.map(|axis| axis.map_or(0, |axis| planes.pixel(plane, axis)));
107            let hi_beam = plane_beam(high_beams, channel, stokes)?;
108            let lo_beam = plane_beam(low_beams, channel, stokes)?;
109            plane_filter = plane_transfer(
110                [nx, ny],
111                (hi_beam, lo_beam),
112                high.coordinates(),
113                channel,
114                request,
115            )?;
116            &plane_filter
117        };
118        let high_fft = centered_fft2(
119            &planes
120                .view(plane, &high_data)
121                .mapv(|value| Complex32::new(value, 0.0)),
122        );
123        let low_fft = centered_fft2(
124            &planes
125                .view(plane, &low_data)
126                .mapv(|value| Complex32::new(value, 0.0)),
127        );
128        let mut combined = Array2::<Complex32>::zeros((nx, ny));
129        for x in 0..nx {
130            for y in 0..ny {
131                let high_weight = transfer.crossover[(x, y)];
132                let low_value = transfer
133                    .low_filter
134                    .as_ref()
135                    .map_or(low_fft[(x, y)], |filter| low_fft[(x, y)] * filter[(x, y)]);
136                combined[(x, y)] = high_fft[(x, y)] * high_weight + low_value * transfer.low_scale;
137            }
138        }
139        let image_plane = centered_ifft2(&combined);
140        Zip::from(planes.view_mut(plane, &mut output_data))
141            .and(&image_plane)
142            .for_each(|out, value| *out = value.re);
143    }
144    clear_masked_pixels(&mut output_data, high_mask.as_ref().or(low_mask.as_ref()));
145    for mask in [high_mask.as_ref(), low_mask.as_ref()]
146        .into_iter()
147        .flatten()
148    {
149        Zip::from(&mut output_mask)
150            .and(mask)
151            .for_each(|out, valid| {
152                *out = *out && *valid;
153            });
154    }
155    let mut output = TempImage::<f32>::new(
156        high_shape.clone(),
157        high.coordinates().clone(),
158        casa_lattices::TempStoragePolicy::Auto {
159            memory_budget_bytes: 64 * 1024 * 1024,
160            scratch: casa_lattices::ScratchSpace::SystemTemp,
161        },
162    )?;
163    let legacy_coordinates = high.table().keywords().get("coords").cloned();
164    output.set_units(high.units())?;
165    output.set_image_info(&high_info)?;
166    output.set_misc_info(high.misc_info())?;
167    output.put_slice(&output_data, &vec![0; output.ndim()])?;
168    if let Some(name) = high.default_mask_name().or(low_mask_name) {
169        output.put_mask(&name, &output_mask)?;
170        output.set_default_mask(&name)?;
171    }
172    let stage = tempfile::tempdir_in(destination.parent().ok_or(ImageError::NotPersistent)?)
173        .map_err(output_io)?;
174    let product = stage.path().join("image");
175    output.save_as(&product)?;
176    if let Some(Value::Record(coords)) = legacy_coordinates {
177        patch_saved_coords_keyword(&product, coords)?;
178    }
179    match rename_without_replacing(&product, &destination) {
180        Err(error) if error.kind() == io::ErrorKind::AlreadyExists && request.overwrite => {
181            check_output(request, &destination)?;
182            let _locks = lock_table_tree_for_replacement(&destination)?;
183            exchange_directories(&product, &destination).map_err(output_io)?;
184        }
185        result => result.map_err(output_io)?,
186    }
187    Ok(FeatherSummary {
188        highres: request.highres.display().to_string(),
189        lowres: request.lowres.display().to_string(),
190        imagename: request.imagename.display().to_string(),
191        shape: high_shape,
192        sdfactor: request.sdfactor,
193        valid_pixels: output_mask.iter().filter(|value| **value).count(),
194    })
195}

197fn clear_masked_pixels(data: &mut ArrayD<f32>, mask: Option<&ArrayD<bool>>) {
198    if let Some(mask) = mask {
199        Zip::from(data).and(mask).for_each(|value, valid| {
200            if !valid {
201                *value = 0.0;
202            }
203        });
204    }
205}

207fn check_output(request: &FeatherRequest, destination: &Path) -> Result<(), ImageError> {
208    for input in [&request.highres, &request.lowres] {
209        ensure_outfile_does_not_alias_input(destination, input)?;
210        if input
211            .canonicalize()
212            .ok()
213            .zip(
214                destination
215                    .parent()
216                    .and_then(|parent| parent.canonicalize().ok()),
217            )
218            .is_some_and(|(input, parent)| parent.starts_with(input))
219        {
220            return Err(invalid("feather output must be outside both input images"));
221        }
222    }
223    match destination.symlink_metadata() {
224        Err(error) if error.kind() == io::ErrorKind::NotFound => Ok(()),
225        Err(error) => Err(output_io(error)),
226        Ok(_) if !request.overwrite => Err(invalid(format!(
227            "feather output already exists: {}",
228            destination.display()
229        ))),
230        Ok(metadata)
231            if metadata.is_dir()
232                && Table::open_metadata_only(TableOptions::new(destination))
233                    .is_ok_and(|table| table.info().table_type == "Image") =>
234        {
235            Ok(())
236        }
237        Ok(_) => Err(invalid(format!(
238            "{} is not a CASA image directory",
239            destination.display()
240        ))),
241    }
242}

248fn pair_nonspatial_axes(
249    template: &mut RegridTemplate,
250    low: &PagedImage<f32>,
251) -> Result<Vec<usize>, ImageError> {
252    let mut paired_shape = low.shape().to_vec();
253    for kind in [CoordinateType::Spectral, CoordinateType::Stokes] {
254        let (Some(target), Some(source)) = (
255            template.coordinates.find_coordinate(kind),
256            low.coordinates().find_coordinate(kind),
257        ) else {
258            continue;
259        };
260        let [Some(output_axis), Some(input_axis)] = [
261            template
262                .coordinates
263                .coordinate_pixel_axes(target)
264                .and_then(|axes| axes.first().copied().flatten()),
265            low.coordinates()
266                .coordinate_pixel_axes(source)
267                .and_then(|axes| axes.first().copied().flatten()),
268        ] else {
269            continue;
270        };
271        if template.shape[output_axis] > low.shape()[input_axis] {
272            return Err(invalid(
273                "per-plane feather requires a low-resolution plane for each high-resolution plane",
274            ));
275        }
276        paired_shape[input_axis] = template.shape[output_axis];
277        template
278            .coordinates
279            .replace_coordinate(target, low.coordinates().coordinate(source).clone());
280    }
281    Ok(paired_shape)
282}

284fn plane_beam(
285    beams: &ImageBeamSet,
286    channel: usize,
287    stokes: usize,
288) -> Result<GaussianBeam, ImageError> {
289    if (beams.n_channels() != 1 && channel >= beams.n_channels())
290        || (beams.n_stokes() != 1 && stokes >= beams.n_stokes())
291    {
292        return Err(invalid(
293            "restoring beam set does not conform to the image planes",
294        ));
295    }
296    let beam = *beams.beam(channel, stokes);
297    if beam.is_null() {
298        return Err(invalid("a feather plane is missing its restoring beam"));
299    }
300    Ok(beam)
301}

303struct PlaneTransfer {
304    crossover: Array2<Complex32>,
305    low_filter: Option<Array2<Complex32>>,
306    low_scale: f32,
307}
308
309fn plane_transfer(
310    shape: [usize; 2],
311    (high, low): (GaussianBeam, GaussianBeam),
312    coordinates: &CoordinateSystem,
313    channel: usize,
314    request: &FeatherRequest,
315) -> Result<PlaneTransfer, ImageError> {
316    let (low, low_filter) = single_dish_filter(shape, low, coordinates, channel, request)?;
317    let crossover = feather_crossover_weight(shape[0], shape[1], low, coordinates)?;
318    let low_scale = request.sdfactor * (high.area() / low.area()) as f32;
319    Ok(PlaneTransfer {
320        crossover,
321        low_filter,
322        low_scale,
323    })
324}

326fn single_dish_filter(
327    [nx, ny]: [usize; 2],
328    low_beam: GaussianBeam,
329    coordinates: &CoordinateSystem,
330    channel: usize,
331    request: &FeatherRequest,
332) -> Result<(GaussianBeam, Option<Array2<Complex32>>), ImageError> {
333    if !request.effdishdiam.is_finite() {
334        return Err(invalid("effective dish diameter must be finite"));
335    }
336    if request.effdishdiam <= 0.0 && !request.lowpassfiltersd {
337        return Ok((low_beam, None));
338    }
339    let frequency = coordinates
340        .find_coordinate(CoordinateType::Spectral)
341        .map(|index| coordinates.coordinate(index))
342        .and_then(|model| match model {
343            CoordinateModel::Spectral(spectral) => Some(spectral),
344            _ => None,
345        })
346        .ok_or_else(|| invalid("single-dish filtering requires a spectral coordinate"))?
347        .world_frequency_hz(channel as f64)?;
348    if !frequency.is_finite() || frequency <= 0.0 || low_beam.is_null() {
349        return Err(invalid(
350            "single-dish filtering requires a positive frequency and a restoring beam",
351        ));
352    }
353    let diffraction = 1.22 * SPEED_OF_LIGHT_M_S / frequency;
354    let diameter = if request.effdishdiam > 0.0 {
355        request.effdishdiam
356    } else {
357        (diffraction / low_beam.major) as f32
358    };
359    let width = diffraction / f64::from(diameter);
360    let target = GaussianBeam::new(width, width, 0.0);
361    let kernel = dish_kernel(target, low_beam)?;
362    let mut psf = gaussian_psf(nx, ny, kernel, coordinates)?;
363    let volume: f32 = psf.t().iter().map(|value| value.re).sum();
364    if !volume.is_finite() || volume == 0.0 {
365        return Err(invalid(
366            "single-dish smoothing kernel has zero or nonfinite volume",
367        ));
368    }
369    psf.mapv_inplace(|value| value / volume);
370    // The CASA real FFT convolution flips by ceil(n/2) on its way back
371    // to the image, placing the kernel one pixel earlier on odd axes.
372    let psf = shifted_plane(&psf, [nx % 2, ny % 2]);
373    Ok((target, Some(centered_fft2(&psf))))
374}

376fn dish_kernel(target: GaussianBeam, low: GaussianBeam) -> Result<GaussianBeam, ImageError> {
377    let error = || {
378        invalid("beam from the effective diameter is smaller than the original single-dish beam")
379    };
380    if target.major < low.major {
381        // GaussianDeconvolver returns the original beam as the kernel when it
382        // classifies a nearly unresolved difference as a point source.
383        let (sin, cos) = low.position_angle.sin_cos();
384        let minor_limit = 0.1 * target.major.min(low.minor).powi(2);
385        let alpha = target.major.powi(2) - (low.major * cos).powi(2) - (low.minor * sin).powi(2);
386        let beta = target.major.powi(2) - (low.major * sin).powi(2) - (low.minor * cos).powi(2);
387        return if alpha > -minor_limit && beta > -minor_limit {
388            Ok(low)
389        } else {
390            Err(error())
391        };
392    }
393    Ok(target
394        .deconvolving_beam(low)
395        .map_err(|_| error())?
396        .unwrap_or(low))
397}

403fn feather_crossover_weight(
404    nx: usize,
405    ny: usize,
406    low_beam: GaussianBeam,
407    coordinates: &CoordinateSystem,
408) -> Result<Array2<Complex32>, ImageError> {
409    let psf = gaussian_psf(nx, ny, low_beam, coordinates)?;
410    // The crossover PSF is made by convolving an impulse at 2*(n/4),
411    // with integer division, instead of the usual FFT origin at n/2.
412    let crossover_shift = |size| size / 2 - 2 * (size / 4) + size % 2;
413    let psf = shifted_plane(&psf, [crossover_shift(nx), crossover_shift(ny)]);
414    let transformed = centered_fft2(&psf);
415    let peak = transformed
416        .iter()
417        .map(|value| value.norm())
418        .fold(0.0_f32, f32::max)
419        .max(1.0e-20);
420    Ok(transformed.mapv(|value| Complex32::new(1.0, 0.0) - value / peak))
421}
422
423fn gaussian_psf(
424    nx: usize,
425    ny: usize,
426    beam: GaussianBeam,
427    coordinates: &CoordinateSystem,
428) -> Result<Array2<Complex32>, ImageError> {
429    let mut psf = Array2::<Complex32>::zeros((nx, ny));
430    let (x_inc, y_inc, x_ref, y_ref) = direction_axis_geometry(coordinates)?;
431    let major_width = beam.major.abs().max(1.0e-12);
432    let minor_width = beam.minor.abs().max(1.0e-12);
433    let major_scale = 4.0 * std::f64::consts::LN_2 / major_width.powi(2);
434    let minor_scale = 4.0 * std::f64::consts::LN_2 / minor_width.powi(2);
435    let pa = beam.position_angle + std::f64::consts::FRAC_PI_2;
436    let cos_pa = pa.cos();
437    let sin_pa = pa.sin();
438    for x in 0..nx {
439        for y in 0..ny {
440            let dx = (x as f64 - x_ref) * x_inc.abs();
441            let dy = (y as f64 - y_ref) * y_inc.abs();
442            let major = dx * cos_pa + dy * sin_pa;
443            let minor = -dx * sin_pa + dy * cos_pa;
444            let radius = major_scale * major.powi(2) + minor_scale * minor.powi(2);
445            let value = if radius < 20.0 {
446                (-radius).exp() as f32
447            } else {
448                0.0
449            };
450            psf[(x, y)] = Complex32::new(value, 0.0);
451        }
452    }
453    Ok(psf)
454}

456fn direction_axis_geometry(
457    coordinates: &CoordinateSystem,
458) -> Result<(f64, f64, f64, f64), ImageError> {
459    let direction_index = coordinates
460        .find_coordinate(CoordinateType::Direction)
461        .ok_or_else(|| ImageError::InvalidMetadata("missing direction coordinate".to_string()))?;
462    let coordinate = coordinates.coordinate(direction_index);
463    let increments = coordinate.increment();
464    let reference_pixel = coordinate.reference_pixel();
465    if increments.len() < 2 {
466        return Err(ImageError::InvalidMetadata(
467            "direction coordinate has fewer than two axes".to_string(),
468        ));
469    }
470    if reference_pixel.len() < 2 {
471        return Err(ImageError::InvalidMetadata(
472            "direction coordinate has fewer than two reference pixels".to_string(),
473        ));
474    }
475    Ok((
476        increments[0],
477        increments[1],
478        reference_pixel[0],
479        reference_pixel[1],
480    ))
481}

483fn centered_fft2(input: &Array2<Complex32>) -> Array2<Complex32> {
484    let mut shifted = ifftshift2(input);
485    let mut fft = Fft2::<f32>::with_threads([shifted.shape()[0], shifted.shape()[1]], 1)
486        .expect("valid feather FFT shape");
487    fft.transform(
488        shifted.as_slice_mut().expect("contiguous feather plane"),
489        false,
490    )
491    .expect("valid feather FFT plan");
492    fftshift2(&shifted)
493}
494
495fn centered_ifft2(input: &Array2<Complex32>) -> Array2<Complex32> {
496    let mut shifted = ifftshift2(input);
497    let mut fft = Fft2::<f32>::with_threads([shifted.shape()[0], shifted.shape()[1]], 1)
498        .expect("valid feather FFT shape");
499    fft.transform(
500        shifted.as_slice_mut().expect("contiguous feather plane"),
501        true,
502    )
503    .expect("valid feather FFT plan");
504    let scale = 1.0 / (input.shape()[0] * input.shape()[1]) as f32;
505    shifted.mapv_inplace(|value| value * scale);
506    fftshift2(&shifted)
507}
508
509fn fftshift2(input: &Array2<Complex32>) -> Array2<Complex32> {
510    shift2(input, false)
511}
512
513fn ifftshift2(input: &Array2<Complex32>) -> Array2<Complex32> {
514    shift2(input, true)
515}
516
517fn shift2(input: &Array2<Complex32>, inverse: bool) -> Array2<Complex32> {
518    let nx = input.shape()[0];
519    let ny = input.shape()[1];
520    let x_shift = if inverse { nx / 2 } else { nx.div_ceil(2) };
521    let y_shift = if inverse { ny / 2 } else { ny.div_ceil(2) };
522    shifted_plane(input, [x_shift, y_shift])
523}
524
525fn shifted_plane(input: &Array2<Complex32>, [x_shift, y_shift]: [usize; 2]) -> Array2<Complex32> {
526    let (nx, ny) = input.dim();
527    Array2::from_shape_fn((nx, ny), |(x, y)| {
528        input[((x + x_shift) % nx, (y + y_shift) % ny)]
529    })
530}
