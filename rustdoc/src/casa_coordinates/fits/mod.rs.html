2//! FITS/WCS coordinate interoperability.
3//!
4//! This module provides conversion between FITS World Coordinate System (WCS)
5//! headers and the coordinate types defined in this crate. It can parse
6//! standard FITS WCS keywords (CTYPEn, CRVALn, CRPIXn, CDELTn, PCi_j, etc.)
7//! into a [`CoordinateSystem`](crate::CoordinateSystem), and emit those
8//! keywords from an existing coordinate system.
9//!
10//! The implementation follows:
11//! - Greisen & Calabretta (2002), "Representations of world coordinates in FITS"
12//! - Calabretta & Greisen (2002), "Representations of celestial coordinates in FITS"
13//! - Greisen et al. (2006), "Representations of spectral coordinates in FITS"
14//!
15//! # Examples
16//!
17//! ```rust
18//! use casa_coordinates::fits::{FitsHeader, to_fits_header, from_fits_header};
19//! use casa_coordinates::{
20//!     CoordinateSystem, DirectionCoordinate, SpectralCoordinate,
21//!     StokesCoordinate, StokesType, Projection, ProjectionType,
22//! };
23//! use casa_types::measures::direction::DirectionRef;
24//! use casa_types::measures::frequency::FrequencyRef;
25//!
26//! // Build a coordinate system
27//! let mut cs = CoordinateSystem::new();
28//! let proj = Projection::new(ProjectionType::SIN);
29//! let dir = DirectionCoordinate::new(
30//!     DirectionRef::J2000, proj,
31//!     [0.0, std::f64::consts::FRAC_PI_4],
32//!     [-1e-4, 1e-4], [512.0, 512.0],
33//! );
34//! cs.add_coordinate(dir);
35//! let spec = SpectralCoordinate::new(FrequencyRef::LSRK, 1.42e9, 1e6, 0.0, 1.42040575e9);
36//! cs.add_coordinate(spec);
37//! let stokes = StokesCoordinate::new(vec![StokesType::I, StokesType::V]);
38//! cs.add_coordinate(stokes);
39//!
40//! // Convert to FITS header
41//! let shape = [1024, 1024, 256, 2];
42//! let header = to_fits_header(&cs, &shape);
43//!
44//! // Round-trip back
45//! let cs2 = from_fits_header(&header, &shape).unwrap();
46//! assert_eq!(cs2.n_pixel_axes(), cs.n_pixel_axes());
47//! ```

56/// Convert a FITS unit expression to the canonical casacore unit spelling.
57pub fn canonical_unit_name(input: &str) -> String {
58    map_unit_symbols(input, |token| match token {
59        "BEAM" => "beam",
60        "DAYS" => "d",
61        "DEGREES" | "DEG" => "deg",
62        "HZ" => "Hz",
63        "JY" => "Jy",
64        "KELVINS" | "KELVIN" => "K",
65        "KM" => "km",
66        "METERS" | "M" => "m",
67        "PASCAL" => "Pa",
68        "PIXEL" => "pixel",
69        "SECONDS" | "SEC" | "S" => "s",
70        "VOLTS" => "V",
71        "YEARS" | "YEAR" => "a",
72        _ => token,
73    })
74}

76// CUNIT aliases before casacore's unit parser, matching WCSLIB UNITFIX
77// (wcsutrn with ctrl=7), including S, H, and D in astronomical headers.
78fn wcs_unit_name(input: &str) -> String {
79    let translated = map_unit_symbols(input, |token| match token {
80        "Angstroms" | "angstrom" | "angstroms" => "Angstrom",
81        "arcmins" | "ARCMIN" | "ARCMINS" => "arcmin",
82        "arcsecs" | "ARCSEC" | "ARCSECS" => "arcsec",
83        "Byte" => "byte",
84        "day" | "days" | "DAY" | "D" => "d",
85        "degree" | "degrees" | "Deg" | "Degree" | "Degrees" | "DEGREE" => "deg",
86        "GHZ" => "GHz",
87        "hr" | "HR" | "H" => "h",
88        "hz" => "Hz",
89        "KHZ" => "kHz",
90        "kelvin" | "kelvins" | "Kelvin" | "Kelvins" => "K",
91        "meter" | "meters" | "metre" | "metres" | "METER" | "METRE" | "METRES" => "m",
92        "MIN" => "min",
93        "MHZ" => "MHz",
94        "Ohm" => "ohm",
95        "pascal" | "pascals" | "Pascal" | "Pascals" | "PASCALS" => "Pa",
96        "pixels" | "PIXELS" => "pixel",
97        "radian" | "radians" | "RAD" | "RADIAN" | "RADIANS" => "rad",
98        "sec" | "second" | "seconds" | "SECOND" => "s",
99        "volt" | "volts" | "Volt" | "Volts" | "VOLT" => "V",
100        "year" | "years" | "YR" => "a",
101        _ => token,
102    });
103    canonical_unit_name(&translated)
104}

106fn map_unit_symbols(input: &str, translate: impl Fn(&str) -> &str) -> String {
107    let mut output = String::new();
108    let mut remaining = input;
109    while !remaining.is_empty() {
110        let alphabetic = remaining.as_bytes()[0].is_ascii_alphabetic();
111        let end = remaining
112            .find(|c: char| c.is_ascii_alphabetic() != alphabetic)
113            .unwrap_or(remaining.len());
114        let token = &remaining[..end];
115        output.push_str(translate(token));
116        remaining = &remaining[end..];
117    }
118    output
119}
