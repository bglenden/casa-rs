2//! The `table` tools of this process, as casacore's table cache lists the
3//! tables a process has open.
4//!
5//! casacore's tools share one table object per table, so a call on one tool
6//! can see another's lock, refuse to replace a table another tool holds
7//! (`prepareCopyRename`) and release every automatic lock of the process
8//! (`Table::relinquishAutoLocks`). Here each tool has a handle of its own;
9//! this registry lets a call visit the other tools.
10//!
11//! A call holds its own tool's mutex while it visits the others, so the
12//! others are visited with `try_lock`: one that is busy in another thread is
13//! reported as such, never waited for (two tools visiting each other would
14//! deadlock).
15
16use std::collections::HashMap;
17use std::path::{Path, PathBuf};
18use std::sync::atomic::{AtomicU64, Ordering};
19use std::sync::{Arc, Mutex, OnceLock, PoisonError, TryLockError, Weak};
20
21use super::state::State;
22
23/// Every tool made in this process that is still alive, with its
24/// `State::id`.
25static TOOLS: Mutex<Vec<(u64, Weak<Mutex<State>>)>> = Mutex::new(Vec::new());
26
27/// Per-table notification of component-list keyword writes in this process.
28/// Weak references keep the registry from retaining closed tables.
29pub(super) fn keyword_epoch(path: &Path) -> Arc<AtomicU64> {
30    static EPOCHS: OnceLock<Mutex<HashMap<PathBuf, Weak<AtomicU64>>>> = OnceLock::new();
31    let name = path.canonicalize().unwrap_or_else(|_| path.to_path_buf());
32    let mut epochs = EPOCHS
33        .get_or_init(Mutex::default)
34        .lock()
35        .unwrap_or_else(PoisonError::into_inner);
36    epochs.retain(|_, epoch| epoch.strong_count() > 0);
37    if let Some(epoch) = epochs.get(&name).and_then(Weak::upgrade) {
38        return epoch;
39    }
40    let epoch = Arc::new(AtomicU64::new(0));
41    epochs.insert(name, Arc::downgrade(&epoch));
42    epoch
43}
44
45pub(super) fn keyword_published(path: &Path) {
46    keyword_epoch(path).fetch_add(1, Ordering::Release);
47}
48
49/// `state` as a tool's shared state, listed with the process's tools.
50pub(super) fn track(state: State) -> Arc<Mutex<State>> {
51    let id = state.id;
52    let tool = Arc::new(Mutex::new(state));
53    let mut tools = TOOLS.lock().unwrap_or_else(PoisonError::into_inner);
54    tools.retain(|(_, other)| other.strong_count() > 0);
55    tools.push((id, Arc::downgrade(&tool)));
56    tool
57}
58
59/// Calls `visit` on the state of every tool but `me` (a tool's `State::id`,
60/// the caller's, whose mutex the caller holds): with `Some(state)` for a tool
61/// that is free, with `None` for one that is busy in another call.
62pub(super) fn for_each_other(me: u64, mut visit: impl FnMut(Option<&mut State>)) {
63    let tools: Vec<Arc<Mutex<State>>> = TOOLS
64        .lock()
65        .unwrap_or_else(PoisonError::into_inner)
66        .iter()
67        .filter(|(id, _)| *id != me)
68        .filter_map(|(_, tool)| tool.upgrade())
69        .collect();
70    for tool in tools {
71        match tool.try_lock() {
72            Ok(mut state) => visit(Some(&mut state)),
73            Err(TryLockError::Poisoned(poisoned)) => visit(Some(&mut poisoned.into_inner())),
74            Err(TryLockError::WouldBlock) => visit(None),
75        }
76    }
77}
