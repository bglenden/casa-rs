2//! The measure and quantum columns of a component list, read as casacore's
3//! `ScalarMeasColumn` and `ScalarQuantColumn`/`ArrayQuantColumn` read them.
4//!
5//! casa-tables' measure accessors read array cells only and ignore
6//! `QuantumUnits`, while a component list's `Reference_Frequency` is a scalar
7//! `Double` column. These readers honour what casacore honours: the
8//! `QuantumUnits` of the value column (default the measure's own units), and a
9//! reference that is fixed (`MEASINFO.Ref`) or read per row from a string or
10//! integer column (`MEASINFO.VarRefCol`).
11

24/// Where a measure column's reference frame comes from.
25#[derive(Debug, Clone)]
26enum Reference {
27    Fixed(String),
28    StringColumn(String),
29    IntColumn {
30        column: String,
31        types: Vec<String>,
32        codes: Vec<i32>,
33    },
34}
35
36/// A scalar measure column: one direction or frequency per row.
37#[derive(Debug, Clone)]
38pub(super) struct MeasureColumn {
39    column: &'static str,
40    measure: MeasureType,
41    /// Factors taking each stored value to radians or hertz.
42    to_si: Vec<f64>,
43    reference: Reference,
44}
45
46impl MeasureColumn {
47    /// Attach to `column`, which must carry `MEASINFO` of type `measure`.
48    pub(super) fn attach(
49        table: &Table,
50        path: &Path,
51        column: &'static str,
52        measure: MeasureType,
53    ) -> Result<Self, ComponentListError> {
54        let keyword_error = |reason: String| ComponentListError::ColumnKeywords {
55            path: path.to_path_buf(),
56            column: column.to_string(),
57            reason,
58        };
59        let desc = TableMeasDesc::reconstruct(table, column)
60            .ok_or_else(|| keyword_error("no MEASINFO measure description".to_string()))?;
61        if desc.measure_type() != measure {
62            return Err(keyword_error(format!(
63                "MEASINFO describes a {} measure, not a {}",
64                desc.measure_type().as_str(),
65                measure.as_str()
66            )));
67        }
68        if desc.offset_desc().is_some() {
69            return Err(keyword_error(
70                "measure offsets are not used by component lists and are not supported".to_string(),
71            ));
72        }
73        let reference = match desc.ref_desc() {
74            MeasRefDesc::Fixed { refer } => Reference::Fixed(refer.clone()),
75            MeasRefDesc::VariableString { ref_column } => {
76                Reference::StringColumn(ref_column.clone())
77            }
78            MeasRefDesc::VariableInt {
79                ref_column,
80                tab_ref_types,
81                tab_ref_codes,
82            } => Reference::IntColumn {
83                column: ref_column.clone(),
84                types: tab_ref_types.clone(),
85                codes: tab_ref_codes.clone(),
86            },
87        };
88        let si = match measure {
89            MeasureType::Direction => "rad",
90            _ => "Hz",
91        };
92        let count = measure.values_per_measure();
93        let units = TableQuantumDesc::reconstruct(table, column)
94            .filter(|quantum| !quantum.is_unit_variable())
95            .map_or_else(
96                || measure.default_units(),
97                |quantum| quantum.units().to_vec(),
98            );
99        let to_si = conversion_factors(&units, count, si).map_err(keyword_error)?;
100        Ok(Self {
101            column,
102            measure,
103            to_si,
104            reference,
105        })
106    }

108    /// The direction at `row`.
109    pub(super) fn direction(
110        &self,
111        table: &Table,
112        path: &Path,
113        row: usize,
114    ) -> Result<MDirection, ComponentListError> {
115        let values = self.values(table, path, row)?;
116        let reference = self.reference(table, path, row)?;
117        let refer = reference
118            .parse::<DirectionRef>()
119            .map_err(|_| cell_error(path, row, self.column, unsupported_frame(&reference)))?;
120        Ok(MDirection::from_angles(values[0], values[1], refer))
121    }
122
123    /// The frequency at `row`.
124    pub(super) fn frequency(
125        &self,
126        table: &Table,
127        path: &Path,
128        row: usize,
129    ) -> Result<MFrequency, ComponentListError> {
130        let values = self.values(table, path, row)?;
131        let reference = self.reference(table, path, row)?;
132        let refer = reference
133            .parse::<FrequencyRef>()
134            .map_err(|_| cell_error(path, row, self.column, unsupported_frame(&reference)))?;
135        Ok(MFrequency::new(values[0], refer))
136    }
137
138    /// The stored values at `row`, in radians or hertz. A one-value measure
139    /// may sit in a scalar column, as `ScalarMeasColumn` allows.
140    fn values(
141        &self,
142        table: &Table,
143        path: &Path,
144        row: usize,
145    ) -> Result<Vec<f64>, ComponentListError> {
146        let raw = match cell(table, path, row, self.column)? {
147            Value::Scalar(scalar) => vec![scalar_f64(scalar).ok_or_else(|| {
148                cell_error(path, row, self.column, "expected a real number".to_string())
149            })?],
150            Value::Array(array) => array_f64(array).ok_or_else(|| {
151                cell_error(path, row, self.column, "expected a real array".to_string())
152            })?,
153            _ => {
154                return Err(cell_error(
155                    path,
156                    row,
157                    self.column,
158                    "expected a number or array".to_string(),
159                ));
160            }
161        };
162        let count = self.measure.values_per_measure();
163        if raw.len() != count {
164            return Err(cell_error(
165                path,
166                row,
167                self.column,
168                format!("expected {count} values, got {}", raw.len()),
169            ));
170        }
171        Ok(raw
172            .iter()
173            .zip(&self.to_si)
174            .map(|(value, factor)| value * factor)
175            .collect())
176    }
177
178    fn reference(
179        &self,
180        table: &Table,
181        path: &Path,
182        row: usize,
183    ) -> Result<String, ComponentListError> {
184        match &self.reference {
185            Reference::Fixed(refer) => Ok(refer.clone()),
186            Reference::StringColumn(column) => match cell(table, path, row, column)? {
187                Value::Scalar(ScalarValue::String(refer)) => Ok(refer.clone()),
188                _ => Err(cell_error(
189                    path,
190                    row,
191                    column,
192                    "expected a frame name".to_string(),
193                )),
194            },
195            Reference::IntColumn {
196                column,
197                types,
198                codes,
199            } => {
200                let code = match cell(table, path, row, column)? {
201                    Value::Scalar(scalar) => scalar_i32(scalar),
202                    _ => None,
203                }
204                .ok_or_else(|| {
205                    cell_error(path, row, column, "expected a frame code".to_string())
206                })?;
207                if let Some(position) = codes.iter().position(|candidate| *candidate == code) {
208                    return types.get(position).cloned().ok_or_else(|| {
209                        cell_error(path, row, column, format!("frame code {code} has no name"))
210                    });
211                }
212                let name = match self.measure {
213                    MeasureType::Direction => {
214                        DirectionRef::from_casacore_code(code).map(|refer| refer.as_str())
215                    }
216                    _ => FrequencyRef::from_casacore_code(code).map(|refer| refer.as_str()),
217                };
218                name.map(str::to_string).ok_or_else(|| {
219                    cell_error(path, row, column, format!("unknown frame code {code}"))
220                })
221            }
222        }
223    }
224}

226/// The units of a quantum column's cell at `row`, `count` of them: from its
227/// `VariableUnits` column (per element when that column holds arrays, per row
228/// when it holds strings) or its fixed `QuantumUnits`.
229pub(super) fn quantum_units(
230    table: &Table,
231    path: &Path,
232    row: usize,
233    column: &str,
234    count: usize,
235) -> Result<Vec<String>, ComponentListError> {
236    let desc = TableQuantumDesc::reconstruct(table, column).ok_or_else(|| {
237        ComponentListError::ColumnKeywords {
238            path: path.to_path_buf(),
239            column: column.to_string(),
240            reason: "no QuantumUnits or VariableUnits keyword".to_string(),
241        }
242    })?;
243    let units = match desc.unit_column_name() {
244        Some(units_column) => match cell(table, path, row, units_column)? {
245            Value::Scalar(ScalarValue::String(unit)) => vec![unit.clone(); count],
246            Value::Array(ArrayValue::String(units)) => units.iter().cloned().collect(),
247            _ => {
248                return Err(cell_error(
249                    path,
250                    row,
251                    units_column,
252                    "expected unit names".to_string(),
253                ));
254            }
255        },
256        None => desc.units().to_vec(),
257    };
258    match units.len() {
259        length if length == count => Ok(units),
260        1 => Ok(vec![units[0].clone(); count]),
261        length => Err(cell_error(
262            path,
263            row,
264            column,
265            format!("expected {count} units, got {length}"),
266        )),
267    }
268}

271pub(super) fn quantity(
272    path: &Path,
273    row: usize,
274    column: &str,
275    value: f64,
276    unit: &str,

288/// A direction's longitude and latitude as casacore stores them
289/// (`MVDirection::get`): `atan2(y, x)` in (−π, π] and `asin(z)`, from the
290/// direction cosines.
291pub(crate) fn casacore_angles(direction: &MDirection) -> [f64; 2] {
292    let [x, y, z] = direction.cosines();
293    let longitude = if x != 0.0 || y != 0.0 {
294        y.atan2(x)
295    } else {
296        0.0
297    };
298    [longitude, z.asin()]
299}

301/// The cell at `row` of `column`.
302pub(super) fn cell<'a>(
303    table: &'a Table,
304    path: &Path,
305    row: usize,
306    column: &str,
307) -> Result<&'a Value, ComponentListError> {
308    table
309        .cell_accessor(row, column)
310        .and_then(|cell| cell.value())
311        .map_err(|source| ComponentListError::Table {
312            path: path.to_path_buf(),
313            source,
314        })?
315        .ok_or_else(|| cell_error(path, row, column, "the cell is undefined".to_string()))
316}

318pub(super) fn cell_error(

324    ComponentListError::Cell {

328        reason,

332pub(super) fn array_f64(array: &ArrayValue) -> Option<Vec<f64>> {
333    match array {
334        ArrayValue::Float64(values) => Some(values.iter().copied().collect()),
335        ArrayValue::Float32(values) => Some(values.iter().map(|value| f64::from(*value)).collect()),
336        ArrayValue::Int32(values) => Some(values.iter().map(|value| f64::from(*value)).collect()),
337        _ => None,
338    }
339}
340
341fn scalar_f64(scalar: &ScalarValue) -> Option<f64> {
342    match scalar {
343        ScalarValue::Float64(value) => Some(*value),
344        ScalarValue::Float32(value) => Some(f64::from(*value)),
345        ScalarValue::Int32(value) => Some(f64::from(*value)),
346        _ => None,
347    }
348}
349
350fn scalar_i32(scalar: &ScalarValue) -> Option<i32> {
351    match scalar {
352        ScalarValue::Int32(value) => Some(*value),
353        ScalarValue::UInt32(value) => i32::try_from(*value).ok(),
354        ScalarValue::Int16(value) => Some(i32::from(*value)),
355        _ => None,
356    }
357}

363/// Factors converting each of `count` stored values in `units` to `si`.
364fn conversion_factors(units: &[String], count: usize, si: &str) -> Result<Vec<f64>, String> {
365    let target = Unit::new(si).expect("SI units parse");
366    let units: Vec<&String> = match units.len() {
367        length if length == count => units.iter().collect(),
368        1 => vec![&units[0]; count],
369        length => {
370            return Err(format!(
371                "QuantumUnits holds {length} units for {count} values"
372            ));
373        }
374    };
375    units
376        .into_iter()
377        .map(|unit| {
378            let parsed = Unit::new(unit).map_err(|error| format!("unit {unit:?}: {error}"))?;
379            if !parsed.conformant(&target) {
380                return Err(format!("unit {unit:?} is not convertible to {si}"));
381            }
382            Ok(parsed.val().factor / target.val().factor)
383        })
384        .collect()
385}

387#[cfg(test)]
388mod tests {
389    use super::*;
390
391    #[test]
392    fn stored_angles_follow_casacore_atan2_convention() {
393        // RA 20h is stored as -60 degrees, as casacore's MVDirection::get gives.
394        let direction = MDirection::from_angles(300.0_f64.to_radians(), -0.5, DirectionRef::J2000);
395        let [longitude, latitude] = casacore_angles(&direction);
396        assert!(
397            (longitude + 60.0_f64.to_radians()).abs() < 1e-15,
398            "{longitude}"
399        );
400        assert!((latitude + 0.5).abs() < 1e-15);
401        let pole = MDirection::from_cosines([0.0, 0.0, 1.0], DirectionRef::J2000);
402        assert_eq!(casacore_angles(&pole), [0.0, std::f64::consts::FRAC_PI_2]);
403    }
404
405    #[test]
406    fn conversion_factors_honour_quantum_units() {
407        let factors =
408            conversion_factors(&["deg".to_string(), "rad".to_string()], 2, "rad").expect("angles");
409        assert!((factors[0] - 1.0_f64.to_radians()).abs() < 1e-18);
410        assert_eq!(factors[1], 1.0);
411        assert_eq!(
412            conversion_factors(&["GHz".to_string()], 1, "Hz"),
413            Ok(vec![1.0e9])
414        );
415        assert!(conversion_factors(&["m".to_string()], 1, "Hz").is_err());
416    }
