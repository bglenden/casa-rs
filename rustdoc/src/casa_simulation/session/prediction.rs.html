2//! `sm.predict`, `sm.setoptions` and `sm.setvp`: the model DATA is predicted
3//! from.
4//!
5//! As in CASA's `Simulator`:
6//! - `setoptions` and `setvp` only record; `predict` reads them.
7//! - `predict` reads the model image when it is called and predicts it into
8//!   the rows observed so far: replacing their DATA, or adding to it when
9//!   `incremental`. Scans observed later are not predicted by it.
10//! - The primary beam (`setvp(dovp=True)`) is the default voltage pattern of
11//!   the telescope of the first `observe` (OBSERVATION row 0), centred on
12//!   each field.
13//!
14//! The model is prepared when `predict` is called (read, multiplied by the
15//! beam and transformed, one set of grids per field when the beam is on), so
16//! a model changed or deleted afterwards predicts as in CASA and errors
17//! surface at the call. The grids are admitted against the session's memory
18//! budget and held until the step is replaced or the session closes; `close`
19//! degrids them into the rows it writes.
20//!
21//! Divergences: CASA rewrites the model image's spectral units and frame in
22//! place; casa-rs never writes it (`docs/CASA (C++) bugs.md`). MODEL_DATA
23//! and CORRECTED_DATA are not written (plan decision 6), even for an
24//! incremental predict, whose sum CASA forms through MODEL_DATA.

47/// The arguments of `sm.predict`.
48#[derive(Debug, Clone, Default, PartialEq, Eq)]
49pub struct PredictSpec {
50    /// `imagename`: the model images; empty names are skipped, as CASA skips
51    /// them.
52    pub images: Vec<PathBuf>,
53    /// `complist`: a component list, or `None` for `''`.
54    pub component_list: Option<PathBuf>,
55    /// `incremental`: add the prediction to DATA rather than replace it.
56    pub incremental: bool,
57}
58
59impl PredictSpec {
60    /// The model images named, without the empty names.
61    fn named_images(&self) -> Vec<&Path> {
62        self.images
63            .iter()
64            .map(PathBuf::as_path)
65            .filter(|path| !path.as_os_str().is_empty())
66            .collect()
67    }
68
69    /// The check casatools' binding makes before `Simulator::predict`, even
70    /// before `open`: every named image exists and opens as an image
71    /// (`ImageFactory::fromFile`). Here it must open as the simulator reads
72    /// it, a CASA image of `Float` pixels: CASA's binding also accepts FITS
73    /// and complex images, which its simulator then refuses with `False`
74    /// (`docs/CASA (C++) bugs.md`).
75    ///
76    /// # Errors
77    ///
78    /// Raised: "File … does not exist." or "Unable to open image …".
79    pub fn check_images(&self) -> SimulatorResult<()> {
80        for path in self.named_images() {
81            if !path.exists() {
82                return Err(SimulatorError::raised(format!(
83                    "File {} does not exist.",
84                    path.display()
85                )));
86            }
87            if SkyModel::open(path).is_err_and(|error| matches!(error, PredictError::Image { .. }))
88            {
89                return Err(SimulatorError::raised(format!(
90                    "Unable to open image {}.",
91                    path.display()
92                )));
93            }
94        }
95        Ok(())
96    }
97}

99/// The arguments of `sm.setoptions` that casa-rs reads. `cache`, `tile`,
100/// `location`, `facets`, `maxdata` and `wprojplanes` do not change what
101/// `GridFT` predicts; the session's memory budget stands for `cache`.
102#[derive(Debug, Clone, PartialEq)]
103pub struct PredictionOptions {
104    /// `ftmachine`; `GridFT` unless `sd`, `both`, `mosaic`, `wproject` or
105    /// `pbwproject` (case-insensitive).
106    pub ftmachine: String,
107    /// `gridfunction`; `SF`, the spheroid, is implemented.
108    pub grid_function: String,
109    /// `padding`; factors below 1 are ignored, as CASA ignores them.
110    pub padding: f64,
111}
112
113impl Default for PredictionOptions {
114    /// CASA's binding defaults: `ft`, `SF`, 1.3.
115    fn default() -> Self {
116        Self {
117            ftmachine: "ft".to_string(),
118            grid_function: "SF".to_string(),
119            padding: 1.3,
120        }
121    }
122}

124/// The arguments of `sm.setvp` that casa-rs reads. `skyposthreshold` and
125/// `pblimit` do not change a symmetric Airy beam's prediction through
126/// `GridFT`.
127#[derive(Debug, Clone, PartialEq)]
128pub struct VoltagePatternSpec {
129    /// `dovp`: multiply the model by the primary beam.
130    pub dovp: bool,
131    /// `usedefaultvp`: the default voltage patterns. Without them and
132    /// without a table CASA uses its voltage-pattern manager's patterns,
133    /// which are the defaults in a fresh session.
134    pub use_default_vp: bool,
135    /// `vptable`: a voltage-pattern table, read when not `usedefaultvp`.
136    pub vp_table: String,
137    /// `dosquint`: beam squint, which a symmetric Airy beam has none of.
138    pub do_squint: bool,
139    /// `parangleinc`, degrees (CASA's 360 for `[]`), logged only.
140    pub parallactic_angle_increment_deg: f64,
141}
142
143impl Default for VoltagePatternSpec {
144    /// CASA's binding defaults.
145    fn default() -> Self {
146        Self {
147            dovp: true,
148            use_default_vp: true,
149            vp_table: String::new(),
150            do_squint: true,
151            parallactic_angle_increment_deg: 360.0,
152        }
153    }
154}

156/// A predict step's prepared model: the operator and grids `close` degrids
157/// the covered scans' rows from.
158pub(crate) struct Prediction {
159    pub(crate) model: ModelOperator,
160    /// The model's tangent point (pixel `(nx/2, ny/2)`), J2000 radians.
161    pub(crate) tangent_rad: [f64; 2],
162    /// The grids: one set for every field without a beam, else one set per
163    /// field, the beam centred on it.
164    grids: Vec<(Option<usize>, PreparedModelGrids)>,
165    /// The grids' memory, admitted at `predict` and released with them.
166    _reservation: Reservation,
167}
168
169impl std::fmt::Debug for Prediction {
170    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
171        formatter
172            .debug_struct("Prediction")
173            .field("tangent_rad", &self.tangent_rad)
174            .field(
175                "fields",
176                &self
177                    .grids
178                    .iter()
179                    .map(|(field, _)| *field)
180                    .collect::<Vec<_>>(),
181            )
182            .finish_non_exhaustive()
183    }
184}

186impl Prediction {
187    /// The grids rows of `field` are predicted from.
188    pub(crate) fn grids_for(&self, field: usize) -> Option<&PreparedModelGrids> {
189        self.grids
190            .iter()
191            .find(|(grid_field, _)| grid_field.is_none_or(|grid_field| grid_field == field))
192            .map(|(_, grids)| grids)
193    }
194
195    /// Bytes every set of grids holds.
196    pub(crate) fn grid_bytes(&self) -> u64 {
197        self.grids.iter().map(|(_, grids)| grids.bytes()).sum()
198    }
199}

201impl Simulator {
202    /// `sm.setoptions(...)`: record the padding `predict` grids with.
203    ///
204    /// # Errors
205    ///
206    /// Not yet implemented: an FTMachine other than `GridFT` (`sd`, `both`,
207    /// `mosaic`, `wproject`, `pbwproject`), or a grid function other than
208    /// `SF`. Nothing is recorded then.
209    pub fn set_options(&mut self, options: &PredictionOptions) -> SimulatorResult<()> {
210        let ftmachine = options.ftmachine.to_lowercase();
211        if OTHER_FTMACHINES.contains(&ftmachine.as_str()) {
212            let phase = if ftmachine == "both" || ftmachine == "pbwproject" {
213                "#841"
214            } else {
215                "simulation Phase 2"
216            };
217            return Err(SimulatorError::not_yet_implemented(format!(
218                "setoptions(ftmachine='{ftmachine}') is not yet implemented in casa-rs ({phase})"
219            )));
220        }
221        let grid_function = options.grid_function.to_lowercase();
222        if grid_function != "sf" {
223            return Err(SimulatorError::not_yet_implemented(format!(
224                "setoptions(gridfunction='{}') is not yet implemented in casa-rs (#841); \
225                 'SF' is",
226                options.grid_function
227            )));
228        }
229        tracing::info!("Setting processing options");
230        // CASA keeps a padding of at least 1 and ignores the rest.
231        let padding = options.padding as f32;
232        if padding >= 1.0 {
233            self.padding = PaddingFactor::new(padding)
234                .map_err(|error| SimulatorError::raised(format!("padding {padding}: {error}")))?;
235        }
236        Ok(())
237    }
238
239    /// `sm.setvp(...)`: record whether `predict` applies the primary beam.
240    ///
241    /// # Errors
242    ///
243    /// Not yet implemented: a voltage-pattern table (`vptable` without
244    /// `usedefaultvp`). Nothing is recorded then.
245    pub fn set_vp(&mut self, spec: &VoltagePatternSpec) -> SimulatorResult<()> {
246        if !spec.use_default_vp && !spec.vp_table.is_empty() {
247            return Err(SimulatorError::not_yet_implemented(format!(
248                "setvp(vptable='{}') is not yet implemented in casa-rs (#841); the default \
249                 voltage patterns are",
250                spec.vp_table
251            )));
252        }
253        tracing::info!("Setting voltage pattern parameters");
254        if spec.use_default_vp {
255            tracing::info!("Using system default voltage patterns for each telescope");
256        }
257        if spec.do_squint {
258            tracing::info!("Beam Squint will be included in the VP model");
259            tracing::info!(
260                "and the parallactic angle increment is {} degrees",
261                spec.parallactic_angle_increment_deg
262            );
263        }
264        self.voltage_pattern = spec.dovp;
265        Ok(())
266    }

268    /// `sm.predict(imagename, complist, incremental)`: predict the model
269    /// into the rows observed so far, replacing their DATA or, when
270    /// `incremental`, adding to it.
271    ///
272    /// # Errors
273    ///
274    /// - Raised, as casatools' binding raises before predicting: a named
275    ///   image that does not exist or does not open
276    ///   ([`PredictSpec::check_images`]).
277    /// - Not yet implemented: a component list, more than one image, a
278    ///   model layout CASA reads and casa-rs does not (other than Stokes I,
279    ///   J2000, direction–Stokes–spectral axes), a near-field distance on the
280    ///   last field, scans whose windows differ in correlations, or
281    ///   `setvp(dovp=True)` for a telescope without an Airy default pattern
282    ///   (VLA and EVLA among them).
283    /// - CASA's `False`: no model image ("No model images found"), or no
284    ///   rows observed yet.
285    /// - Raised: a scan of 3 correlations ("No such VisVector type"), or
286    ///   model grids that do not fit the session's memory budget
287    ///   ([`SimulatorError::admission_refused`]).
288    pub fn predict(&mut self, spec: &PredictSpec) -> SimulatorResult<()> {
289        spec.check_images()?;
290        let images = spec.named_images();
291        tracing::info!(
292            "Predicting visibilities using model: {images:?} and componentList: {}",
293            spec.component_list
294                .as_deref()
295                .map_or(String::new(), |path| path.display().to_string())
296        );
297        if spec.incremental {
298            tracing::info!("The data column will be incremented");
299        } else {
300            tracing::info!("The data column will be replaced");
301        }
302        if let Some(list) = &spec.component_list {
303            return Err(SimulatorError::not_yet_implemented(format!(
304                "predict(complist='{}') is not yet implemented in casa-rs (simulation Phase 2)",
305                list.display()
306            )));
307        }
308        if images.len() > 1 {
309            return Err(SimulatorError::not_yet_implemented(format!(
310                "predict from {} model images is not yet implemented in casa-rs (simulation \
311                 Phase 2); one image is",
312                images.len()
313            )));
314        }
315        let Some(path) = images.first() else {
316            return Err(SimulatorError::refused(
317                "Failed to create SkyEquation: No model images found",
318            ));
319        };
320        if let Some(field) = self.fields.last()
321            && field.distance_m != 0.0
322        {
323            return Err(SimulatorError::not_yet_implemented(format!(
324                "predict refocused to the distance {} m of field {} is not yet implemented in \
325                 casa-rs (#841)",
326                field.distance_m, field.name
327            )));
328        }
329        let model = SkyModel::open(path).map_err(model_error)?;
330        if self.planned_rows() == 0 {
331            return Err(SimulatorError::refused(
332                "Failed to create SkyEquation: Caught exception: MSIter::construct - Input \
333                 MeasurementSet.has zero selected rows",
334            ));
335        }
336        let correlations = self.predicted_correlations()?;
337        let beam = self.voltage_pattern_beam()?;
338        let operator = ModelOperator::new(&model, &correlations, self.padding, GridPrecision::F32)
339            .map_err(|error| correlation_error(error, &correlations))?;
340        let prediction = self.prepare(&model, operator, beam)?;
341        if !spec.incremental {
342            // Every step so far covers a prefix of the scans this one covers.
343            self.data_plan.clear();
344        }
345        self.data_plan
346            .predict(Arc::new(prediction), self.scans.len());
347        Ok(())
348    }

350    /// The correlations of the scans observed so far, which one operator
351    /// predicts.
352    fn predicted_correlations(&self) -> SimulatorResult<Vec<CorrelationType>> {
353        self.check_visibility_correlations("predicted")?;
354        let window = |scan: &super::observe::Scan| &self.spectral_windows[scan.spectral_window_id];
355        let first = &window(&self.scans[0]).correlations;
356        if self
357            .scans
358            .iter()
359            .any(|scan| window(scan).correlations != *first)
360        {
361            return Err(SimulatorError::not_yet_implemented(
362                "predicting scans whose spectral windows have different correlations is not yet \
363                 implemented in casa-rs (#841)",
364            ));
365        }
366        first
367            .iter()
368            .map(|&code| {
369                CorrelationType::from_casa_code(code).ok_or_else(|| {
370                    SimulatorError::not_yet_implemented(format!(
371                        "predicting correlation code {code} is not yet implemented in casa-rs \
372                         (#841)"
373                    ))
374                })
375            })
376            .collect()
377    }

379    /// The primary beam `setvp` asks for: the default pattern of OBSERVATION
380    /// row 0's telescope.
381    fn voltage_pattern_beam(&self) -> SimulatorResult<Option<PrimaryBeam>> {
382        if !self.voltage_pattern {
383            return Ok(None);
384        }
385        let telescope = self
386            .observations
387            .first()
388            .map_or(self.telescope.as_str(), |observation| {
389                observation.telescope.as_str()
390            });
391        let beam = PrimaryBeam::for_telescope(telescope).ok_or_else(|| {
392            let phase = if telescope == "VLA" || telescope == "EVLA" {
393                "simulation Phase 2"
394            } else {
395                "#841"
396            };
397            SimulatorError::not_yet_implemented(format!(
398                "setvp(dovp=True) for telescope {telescope} is not yet implemented in casa-rs \
399                 ({phase}); ALMA's and ACA's default patterns are"
400            ))
401        })?;
402        Ok(Some(beam))
403    }

405    /// The fields of the scans observed so far, each once, in the order
406    /// first observed.
407    fn observed_fields(&self) -> Vec<usize> {
408        let mut fields = Vec::new();
409        for scan in &self.scans {
410            if !fields.contains(&scan.field_id) {
411                fields.push(scan.field_id);
412            }
413        }
414        fields
415    }

417    /// Admit and prepare `model`'s grids: one set, or one per observed field
418    /// with `beam` centred on it. The planes are prepared on as many of the
419    /// policy's workers as fit their scratch beside the grids.
420    fn prepare(
421        &self,
422        model: &SkyModel,
423        operator: ModelOperator,
424        beam: Option<PrimaryBeam>,
425    ) -> SimulatorResult<Prediction> {
426        let resources = &self.options.resources;
427        let (host, policy) = (resources.host(), resources.policy());
428        charge_process_caches(host, policy).map_err(SimulatorError::admission)?;
429        let fields: Vec<Option<usize>> = match beam {
430            Some(_) => self.observed_fields().into_iter().map(Some).collect(),
431            None => vec![None],
432        };
433        let pattern_bytes = if beam.is_some() {
434            BeamPattern::BYTES
435        } else {
436            0
437        };
438        let reservation = admit(
439            host,
440            policy,
441            &Demand {
442                phase: MODEL_PHASE,
443                memory: operator.grid_bytes() * fields.len() as u64 + pattern_bytes,
444            },
445        )
446        .map_err(SimulatorError::admission)?;
447        let scratch = operator.plane_scratch_bytes(model);
448        let free = free_memory(host, policy);
449        let workers = (1..=resources.cores().workers.max(1))
450            .rev()
451            .find(|workers| *workers as u64 * scratch <= free)
452            .unwrap_or(1);
453        let _preparation = admit(
454            host,
455            policy,
456            &Demand {
457                phase: PREPARATION_PHASE,
458                memory: workers as u64 * scratch,
459            },
460        )
461        .map_err(SimulatorError::admission)?;
462        let team = WorkerTeam::new(workers)
463            .map_err(|error| SimulatorError::raised(format!("the worker team: {error}")))?;
464        let pattern = beam.map(PrimaryBeam::pattern);
465        let mut grids = Vec::with_capacity(fields.len());
466        for field in fields {
467            let centre = match (&pattern, field) {
468                (Some(pattern), Some(field)) => {
469                    let pointing = self.fields[field].direction_rad;
470                    Some((pattern, model.pixel_of(pointing).map_err(model_error)?))
471                }
472                _ => None,
473            };
474            let prepared = operator
475                .prepare(model, centre, Some(&team))
476                .map_err(model_error)?;
477            grids.push((field, prepared));
478        }
479        tracing::info!(
480            planes = model.planes(),
481            grids = grids.len(),
482            workers,
483            bytes = operator.grid_bytes() * grids.len() as u64,
484            "prepared the model {}",
485            model.path().display()
486        );
487        Ok(Prediction {
488            model: operator,
489            tangent_rad: model.centre_rad(),
490            grids,
491            _reservation: reservation,
492        })
493    }
494}

496/// A model that cannot be read or prepared, as CASA reports it: a layout
497/// CASA reads and casa-rs does not is not yet implemented; the rest is
498/// raised.
499fn model_error(error: PredictError) -> SimulatorError {
500    match error {
501        PredictError::UnsupportedModel(reason) => SimulatorError::not_yet_implemented(format!(
502            "predict from this model is not yet implemented in casa-rs (#841): {reason}"
503        )),
504        PredictError::Image { path, .. } => {
505            SimulatorError::raised(format!("Unable to open image {path}."))
506        }
507        other => SimulatorError::raised(format!("Caught exception: {other}")),
508    }
509}
510
511/// An operator refusal for `correlations`: a correlation set the Stokes I
512/// model cannot feed is not yet implemented; the rest is raised.
513fn correlation_error(error: PredictError, correlations: &[CorrelationType]) -> SimulatorError {
514    match error {
515        PredictError::Operator(OperatorError::Polarization { reason }) => {
516            SimulatorError::not_yet_implemented(format!(
517                "predicting a Stokes I model into the correlations {correlations:?} is not yet \
518                 implemented in casa-rs (#841): {reason}"
519            ))
520        }
521        other => model_error(other),
522    }
