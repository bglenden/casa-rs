2//! Values CASA's tools return: casatools' `record` and `variant` results,
3//! built without the GIL and converted to Python once a call is done.
4//!
5//! casatools hands records to Python as `dict`s whose keys are sorted (its
6//! records are `std::map`s), so [`CasaValue::record`] sorts them; arrays
7//! become NumPy arrays, numeric ones of `float64` and string ones of
8//! NumPy's unicode type, as casatools returns them.
9
10use ndarray::{ArrayD, IxDyn, ShapeBuilder};
11use numpy::IntoPyArray;
12use pyo3::prelude::*;
13use pyo3::types::{PyDict, PyList, PyTuple};
14
15use super::Variant;
16
17/// A value a CASA tool returns.
18#[derive(Debug, Clone, PartialEq)]
19pub(crate) enum CasaValue {
20    /// A `bool`.
21    Bool(bool),
22    /// An `int`.
23    Int(i64),
24    /// A `float`.
25    Float(f64),
26    /// A `str`.
27    Text(String),
28    /// A NumPy `float64` array of `shape`, its values in Fortran (column-major)
29    /// order, as casacore arrays hold them.
30    Floats { shape: Vec<usize>, values: Vec<f64> },
31    /// A NumPy unicode array.
32    Texts(Vec<String>),
33    /// A Python `list`.
34    List(Vec<CasaValue>),
35    /// A Python `tuple`.
36    Tuple(Vec<CasaValue>),
37    /// A `dict` with sorted keys.
38    Record(Vec<(String, CasaValue)>),
39}
40
41impl CasaValue {
42    /// A record of `fields`, sorted by key as casatools sorts them.
43    pub(crate) fn record(fields: impl IntoIterator<Item = (impl Into<String>, CasaValue)>) -> Self {
44        let mut fields = fields
45            .into_iter()
46            .map(|(key, value)| (key.into(), value))
47            .collect::<Vec<_>>();
48        fields.sort_by(|left, right| left.0.cmp(&right.0));
49        Self::Record(fields)
50    }
51
52    /// The empty record `{}`: what CASA's tools return for a failure they
53    /// log.
54    pub(crate) fn empty_record() -> Self {
55        Self::Record(Vec::new())
56    }
57
58    /// A one-dimensional `float64` array.
59    pub(crate) fn floats(values: Vec<f64>) -> Self {
60        Self::Floats {
61            shape: vec![values.len()],
62            values,
63        }
64    }
65
66    /// An argument as CASA hands a record field back (`me.getvalue`).
67    pub(crate) fn from_variant(value: &Variant) -> Self {
68        match value {
69            Variant::Empty => Self::List(Vec::new()),
70            Variant::Bool(flag) => Self::Bool(*flag),
71            Variant::Number(number) => Self::Float(*number),
72            Variant::Text(text) | Variant::Other(text) => Self::Text(text.clone()),
73            Variant::Texts(texts) => Self::Texts(texts.clone()),
74            Variant::Numbers(numbers) => Self::floats(numbers.clone()),
75            Variant::Array { shape, values } => Self::Floats {
76                shape: shape.clone(),
77                values: values.clone(),
78            },
79            Variant::Record(record) => Self::record(
80                record
81                    .iter()
82                    .map(|(key, value)| (key.clone(), Self::from_variant(value))),
83            ),
84        }
85    }
86
87    /// A quantity record `{'unit': unit, 'value': value}`.
88    pub(crate) fn quantity(value: f64, unit: impl Into<String>) -> Self {
89        Self::record([
90            ("unit", Self::Text(unit.into())),
91            ("value", Self::Float(value)),
92        ])
93    }
94
95    /// The value as a Python object.
96    pub(crate) fn into_python(self, py: Python<'_>) -> PyResult<Py<PyAny>> {
97        Ok(match self {
98            Self::Bool(flag) => flag.into_pyobject(py)?.to_owned().into_any().unbind(),
99            Self::Int(value) => value.into_pyobject(py)?.into_any().unbind(),
100            Self::Float(value) => value.into_pyobject(py)?.into_any().unbind(),
101            Self::Text(text) => text.into_pyobject(py)?.into_any().unbind(),
102            Self::Floats { shape, values } => {
103                let array = ArrayD::from_shape_vec(IxDyn(&shape).f(), values).map_err(|error| {
104                    pyo3::exceptions::PyRuntimeError::new_err(format!(
105                        "array of shape {shape:?}: {error}"
106                    ))
107                })?;
108                array.into_pyarray(py).into_any().unbind()
109            }
110            Self::Texts(texts) => {
111                let keywords = PyDict::new(py);
112                keywords.set_item("dtype", "str")?;
113                py.import("numpy")?
114                    .call_method("array", (PyList::new(py, texts)?,), Some(&keywords))?
115                    .unbind()
116            }
117            Self::List(items) => {
118                let items = items
119                    .into_iter()
120                    .map(|item| item.into_python(py))
121                    .collect::<PyResult<Vec<_>>>()?;
122                PyList::new(py, items)?.into_any().unbind()
123            }
124            Self::Tuple(items) => {
125                let items = items
126                    .into_iter()
127                    .map(|item| item.into_python(py))
128                    .collect::<PyResult<Vec<_>>>()?;
129                PyTuple::new(py, items)?.into_any().unbind()
130            }
131            Self::Record(fields) => {
132                let dict = PyDict::new(py);
133                for (key, value) in fields {
134                    dict.set_item(key, value.into_python(py)?)?;
135                }
136                dict.into_any().unbind()
137            }
138        })
139    }
140}

142/// `value` as a C++ `ostream` writes a `double` by default: six significant
143/// digits, in fixed or scientific notation as `%g` chooses, without
144/// trailing zeros (`3.14159`, `6.37358e+06`, `58849`).
145pub(crate) fn stream_double(value: f64) -> String {
146    const PRECISION: usize = 6;
147    if value == 0.0 {
148        return if value.is_sign_negative() { "-0" } else { "0" }.to_string();
149    }
150    if !value.is_finite() {
151        return if value.is_nan() {
152            "nan".to_string()
153        } else if value > 0.0 {
154            "inf".to_string()
155        } else {
156            "-inf".to_string()
157        };
158    }
159    let scientific = format!("{:.*e}", PRECISION - 1, value);
160    let (mantissa, exponent) = scientific.split_once('e').expect("Rust writes an exponent");
161    let exponent: i32 = exponent.parse().expect("Rust writes an integer exponent");
162    let trim = |digits: &str| {
163        if digits.contains('.') {
164            digits
165                .trim_end_matches('0')
166                .trim_end_matches('.')
167                .to_string()
168        } else {
169            digits.to_string()
170        }
171    };
172    if exponent < -4 || exponent >= PRECISION as i32 {
173        let sign = if exponent < 0 { '-' } else { '+' };
174        format!("{}e{sign}{:02}", trim(mantissa), exponent.abs())
175    } else {
176        let decimals = usize::try_from(PRECISION as i32 - 1 - exponent).unwrap_or(0);
177        trim(&format!("{value:.decimals$}"))
178    }
179}

181#[cfg(test)]
182mod tests {
183    use super::{CasaValue, stream_double};
184
185    #[test]
186    fn doubles_stream_as_cpp_writes_them() {
187        for (value, written) in [
188            (std::f64::consts::PI, "3.14159"),
189            (0.785_398_163_397_448_2, "0.785398"),
190            (6_373_576.280_915_651, "6.37358e+06"),
191            (58_849.0, "58849"),
192            (58_849.25, "58849.2"),
193            (1.4e9, "1.4e+09"),
194            (-1.878_294_258_139_436_2, "-1.87829"),
195            (1e-6, "1e-06"),
196            (0.000_123_4, "0.0001234"),
197            (999_999.5, "1e+06"),
198            (30.0, "30"),
199        ] {
200            assert_eq!(stream_double(value), written, "{value}");
201        }
202    }
203
204    #[test]
205    fn records_sort_their_keys_as_casatools_does() {
206        let record = CasaValue::record([
207            ("type", CasaValue::Text("epoch".into())),
208            ("m0", CasaValue::quantity(1.0, "d")),
209            ("refer", CasaValue::Text("UTC".into())),
210        ]);
211        let CasaValue::Record(fields) = record else {
212            panic!("a record");
213        };
214        let keys = fields
215            .iter()
216            .map(|(key, _)| key.as_str())
217            .collect::<Vec<_>>();
218        assert_eq!(keys, ["m0", "refer", "type"]);
219        let CasaValue::Record(quantity) = &fields[0].1 else {
220            panic!("a quantity record");
221        };
222        assert_eq!(quantity[0].0, "unit");
223        assert_eq!(quantity[1].0, "value");
224    }
