2//! Tap loops over one `[y][x]` block: spread (adjoint), gather (forward)
3//! and the kernel norm.
4//!
5//! `tap' = (conjugate ? conj(t) : t) · e^{i(k_x g_x + k_y g_y)}`. The norm
6//! is `Σ (conjugate ? conj(t) : t)` over the same taps without the pointing
7//! ramp, which is how CASA's `AWVisResampler` accumulates it
8//! (`faccumulateFromGrid`: `norm += wt` before the phase gradient).
9//!
10//! Every function takes the location the shared rounding rule produced and
11//! trusts that the support lies inside the tile; the debug assertions name
12//! that invariant.
13
14use num_complex::{Complex, Complex32, Complex64};
15
16use crate::accumulator::{GridScalar, Tile};
17use crate::convolution::TapLayout;
18use crate::geometry::CellLocation;

20/// Add `value · tap'` over the support into `grid` through Mueller plane
21/// `mueller`.
22#[allow(clippy::too_many_arguments)]
23pub(super) fn spread<T: GridScalar>(
24    grid: &mut [Complex<T>],
25    tile: Tile,
26    location: CellLocation,
27    taps: &TapLayout<'_>,
28    mueller: u8,
29    conjugate: bool,
30    gradient: [f32; 2],
31    value: Complex<T>,
32) {
33    match *taps {
34        TapLayout::SeparableReal {
35            rows,
36            support,
37            oversampling: _,
38        } => {
39            debug_assert_eq!(mueller, 0, "separable kernels have one Mueller plane");
40            debug_assert_eq!(
41                gradient,
42                [0.0, 0.0],
43                "separable kernels carry no phase ramp"
44            );
45            let support = usize::from(support);
46            let rx = &rows[usize::from(location.ox) * support..][..support];
47            let ry = &rows[usize::from(location.oy) * support..][..support];
48            let (x0, y0) = block_origin(tile, location, [support / 2, support / 2]);
49            match support {
50                7 => spread_separable::<T, 7>(grid, tile.shape[0], x0, y0, rx, ry, value),
51                _ => spread_separable_dyn(grid, tile.shape[0], x0, y0, rx, ry, value),
52            }
53        }
54        TapLayout::Dense {
55            data,
56            support,
57            oversampling,
58            mueller_planes,
59        } => {
60            let tile_taps = dense_tile(
61                data,
62                support,
63                oversampling,
64                mueller_planes,
65                location,
66                mueller,
67            );
68            let [sx, sy] = [usize::from(support[0]), usize::from(support[1])];
69            let (x0, y0) = block_origin(tile, location, [sx / 2, sy / 2]);
70            let nx = tile.shape[0];
71            for iy in 0..sy {
72                let row = &mut grid[(y0 + iy) * nx + x0..][..sx];
73                let tap_row = &tile_taps[iy * sx..][..sx];
74                for (ix, (cell, tap)) in row.iter_mut().zip(tap_row).enumerate() {
75                    let tap = dense_tap(*tap, conjugate, gradient, ix, iy, sx, sy);
76                    let tap = Complex::new(T::from_f64(tap.re), T::from_f64(tap.im));
77                    *cell = *cell + value * tap;
78                }
79            }
80        }
81    }
82}

84/// `Σ conj(tap') · grid` over the support through Mueller plane `mueller`.
85pub(super) fn gather<T: GridScalar>(
86    grid: &[Complex<T>],
87    tile: Tile,
88    location: CellLocation,
89    taps: &TapLayout<'_>,
90    mueller: u8,
91    conjugate: bool,
92    gradient: [f32; 2],
93) -> Complex64 {
94    match *taps {
95        TapLayout::SeparableReal {
96            rows,
97            support,
98            oversampling: _,
99        } => {
100            debug_assert_eq!(mueller, 0, "separable kernels have one Mueller plane");
101            let support = usize::from(support);
102            let rx = &rows[usize::from(location.ox) * support..][..support];
103            let ry = &rows[usize::from(location.oy) * support..][..support];
104            let (x0, y0) = block_origin(tile, location, [support / 2, support / 2]);
105            let nx = tile.shape[0];
106            let mut sum = Complex::<T>::default();
107            for (iy, wy) in ry.iter().enumerate() {
108                let row = &grid[(y0 + iy) * nx + x0..][..support];
109                let mut row_sum = Complex::<T>::default();
110                for (cell, wx) in row.iter().zip(rx) {
111                    row_sum = row_sum + *cell * T::from_f32(*wx);
112                }
113                sum = sum + row_sum * T::from_f32(*wy);
114            }
115            Complex64::new(sum.re.into_f64(), sum.im.into_f64())
116        }
117        TapLayout::Dense {
118            data,
119            support,
120            oversampling,
121            mueller_planes,
122        } => {
123            let tile_taps = dense_tile(
124                data,
125                support,
126                oversampling,
127                mueller_planes,
128                location,
129                mueller,
130            );
131            let [sx, sy] = [usize::from(support[0]), usize::from(support[1])];
132            let (x0, y0) = block_origin(tile, location, [sx / 2, sy / 2]);
133            let nx = tile.shape[0];
134            let mut sum = Complex64::default();
135            for iy in 0..sy {
136                let row = &grid[(y0 + iy) * nx + x0..][..sx];
137                let tap_row = &tile_taps[iy * sx..][..sx];
138                for (ix, (cell, tap)) in row.iter().zip(tap_row).enumerate() {
139                    let tap = dense_tap(*tap, conjugate, gradient, ix, iy, sx, sy);
140                    let cell = Complex64::new(cell.re.into_f64(), cell.im.into_f64());
141                    sum += tap.conj() * cell;
142                }
143            }
144            sum
145        }
146    }
147}

149/// `Σ (conjugate ? conj(t) : t)` over the support of Mueller plane
150/// `mueller` at the sample's fine offset, without the pointing ramp: the
151/// forward normalisation, whose magnitude `sumwt` accumulates.
152pub(crate) fn norm(
153    taps: &TapLayout<'_>,
154    location: CellLocation,
155    mueller: u8,
156    conjugate: bool,
157) -> Complex64 {
158    match *taps {
159        TapLayout::SeparableReal {
160            rows,
161            support,
162            oversampling: _,
163        } => {
164            debug_assert_eq!(mueller, 0, "separable kernels have one Mueller plane");
165            let support = usize::from(support);
166            let axis_sum = |offset: u16| {
167                rows[usize::from(offset) * support..][..support]
168                    .iter()
169                    .map(|tap| f64::from(*tap))
170                    .sum::<f64>()
171            };
172            Complex64::new(axis_sum(location.ox) * axis_sum(location.oy), 0.0)
173        }
174        TapLayout::Dense {
175            data,
176            support,
177            oversampling,
178            mueller_planes,
179        } => {
180            let sum = dense_tile(
181                data,
182                support,
183                oversampling,
184                mueller_planes,
185                location,
186                mueller,
187            )
188            .iter()
189            .map(|tap| Complex64::new(f64::from(tap.re), f64::from(tap.im)))
190            .sum::<Complex64>();
191            if conjugate { sum.conj() } else { sum }
192        }
193    }
194}

196/// Block-local `(x0, y0)` of the first tap.
197fn block_origin(tile: Tile, location: CellLocation, half: [usize; 2]) -> (usize, usize) {
198    let x0 = location.x - half[0] as i64 - tile.origin[0] as i64;
199    let y0 = location.y - half[1] as i64 - tile.origin[1] as i64;
200    debug_assert!(
201        x0 >= 0
202            && y0 >= 0
203            && x0 as usize + 2 * half[0] < tile.shape[0]
204            && y0 as usize + 2 * half[1] < tile.shape[1],
205        "sample support lies outside the accumulator tile"
206    );
207    (x0 as usize, y0 as usize)
208}

210fn spread_separable<T: GridScalar, const S: usize>(
211    grid: &mut [Complex<T>],
212    nx: usize,
213    x0: usize,
214    y0: usize,
215    rx: &[f32],
216    ry: &[f32],
217    value: Complex<T>,
218) {
219    let mut taps_x = [T::zero(); S];
220    for (target, tap) in taps_x.iter_mut().zip(rx) {
221        *target = T::from_f32(*tap);
222    }
223    for (iy, wy) in ry.iter().enumerate() {
224        let scaled = value * T::from_f32(*wy);
225        let row: &mut [Complex<T>; S] = (&mut grid[(y0 + iy) * nx + x0..][..S])
226            .try_into()
227            .expect("row has S cells");
228        for (cell, tap) in row.iter_mut().zip(&taps_x) {
229            *cell = *cell + scaled * *tap;
230        }
231    }
232}

234fn spread_separable_dyn<T: GridScalar>(
235    grid: &mut [Complex<T>],
236    nx: usize,
237    x0: usize,
238    y0: usize,
239    rx: &[f32],
240    ry: &[f32],
241    value: Complex<T>,
242) {
243    let support = rx.len();
244    for (iy, wy) in ry.iter().enumerate() {
245        let scaled = value * T::from_f32(*wy);
246        let row = &mut grid[(y0 + iy) * nx + x0..][..support];
247        for (cell, tap) in row.iter_mut().zip(rx) {
248            *cell = *cell + scaled * T::from_f32(*tap);
249        }
250    }
251}

253/// The `sy × sx` tile of one Mueller plane at the sample's fine offset.
254fn dense_tile(
255    data: &[Complex32],
256    support: [u16; 2],
257    oversampling: u16,
258    mueller_planes: u8,
259    location: CellLocation,
260    mueller: u8,
261) -> &[Complex32] {
262    debug_assert!(mueller < mueller_planes, "Mueller plane outside the cell");
263    let taps = usize::from(support[0]) * usize::from(support[1]);
264    let rows = usize::from(oversampling) + 1;
265    let offset = (usize::from(location.oy) * rows + usize::from(location.ox))
266        * usize::from(mueller_planes)
267        + usize::from(mueller);
268    &data[offset * taps..][..taps]
269}
270
271fn dense_tap(
272    tap: Complex32,
273    conjugate: bool,
274    gradient: [f32; 2],
275    ix: usize,
276    iy: usize,
277    sx: usize,
278    sy: usize,
279) -> Complex64 {
280    let mut tap = Complex64::new(f64::from(tap.re), f64::from(tap.im));
281    if conjugate {
282        tap = tap.conj();
283    }
284    if gradient != [0.0, 0.0] {
285        let kx = ix as f64 - (sx / 2) as f64;
286        let ky = iy as f64 - (sy / 2) as f64;
287        let phase = kx * f64::from(gradient[0]) + ky * f64::from(gradient[1]);
288        tap *= Complex64::from_polar(1.0, phase);
289    }
290    tap
291}
