2//! A component's and a list's records, as CASA's `toRecord` forms them
3//! (`SkyCompRep::toRecord`, `ComponentList::toRecord`): what the `cl` tool's
4//! `getcomponent` and `torecord` return.
5//!
6//! | record | fields |
7//! |---|---|
8//! | list | `nelements`, then `component0`, `component1`, … |
9//! | component | `flux`, `shape`, `spectrum`, `optionalParameters` (when there are any), `label` |
10//! | flux | `value` and `unit` (a `QuantumHolder` of the four values: real for Stokes, complex otherwise), `polarisation`, `error` (real for Stokes, complex otherwise) |
11//! | shape | `type`, `direction` (a `MeasureHolder` direction record with an `error` record of `longitude` and `latitude` quantities); a two-sided shape adds `majoraxis`, `minoraxis`, `positionangle` and the three `…error`s |
12//! | spectrum | the `Spectral_Record` form ([`SpectralModel::to_record`]) |
13//!
14//! CASA records a shape's axes in the units it holds them in: those given
15//! for a component it built in memory, and for one read from a table arcmin
16//! (axes) and degrees (position angle), with the errors in radians. casa-rs
17//! holds radians and records every component as CASA records one read from
18//! a table; the quantities are the same.
19
20use casa_types::{ArrayValue, RecordField, RecordValue, ScalarValue, Value};
21use num_complex::Complex64;

31/// Radians in an arcminute.
32const ARCMIN_RAD: f64 = std::f64::consts::PI / 10_800.0;
33
34/// Radians in a degree.
35const DEGREE_RAD: f64 = std::f64::consts::PI / 180.0;
36
37impl ComponentList {
38    /// The list's record (`ComponentList::toRecord`): `nelements` and each
39    /// component's record as `component<i>`.
40    ///
41    /// # Errors
42    ///
43    /// [`ComponentError::Spectrum`] for a tabular spectrum CASA could not
44    /// write (see [`SpectralModel::to_record`]).
45    pub fn to_record(&self) -> Result<RecordValue, ComponentError> {
46        let mut fields = vec![field(
47            "nelements",
48            Value::Scalar(ScalarValue::Int32(
49                i32::try_from(self.len()).unwrap_or(i32::MAX),
50            )),
51        )];
52        for (index, component) in self.iter().enumerate() {
53            fields.push(field(
54                &format!("component{index}"),
55                Value::Record(component.to_record()?),
56            ));
57        }
58        Ok(RecordValue::new(fields))
59    }
60}
61
62impl SkyComponent {
63    /// The component's record (`SkyCompRep::toRecord`).
64    ///
65    /// # Errors
66    ///
67    /// [`ComponentError::Spectrum`] for a tabular spectrum CASA could not
68    /// write (see [`SpectralModel::to_record`]).
69    pub fn to_record(&self) -> Result<RecordValue, ComponentError> {
70        let mut fields = vec![
71            field("flux", Value::Record(self.flux().to_record())),
72            field("shape", Value::Record(self.shape().to_record())),
73            field("spectrum", Value::Record(self.spectrum().to_record()?)),
74        ];
75        if !self.optional_parameters().is_empty() {
76            fields.push(field(
77                "optionalParameters",
78                reals(self.optional_parameters().to_vec()),
79            ));
80        }
81        fields.push(field("label", string(self.label())));
82        Ok(RecordValue::new(fields))
83    }
84}

86impl Flux {
87    /// The flux's record (`FluxRep::toRecord`): the values and errors are
88    /// real for Stokes and complex in a circular or linear basis.
89    pub fn to_record(&self) -> RecordValue {
90        let stokes = self.polarisation() == Polarisation::Stokes;
91        let array = |values: &[Complex64; 4]| {
92            if stokes {
93                reals(values.iter().map(|value| value.re).collect())
94            } else {
95                Value::Array(ArrayValue::from_complex64_vec(values.to_vec()))
96            }
97        };
98        RecordValue::new(vec![
99            field("value", array(self.values())),
100            field("unit", string(self.unit().name())),
101            field("polarisation", string(self.polarisation().casa_name())),
102            field("error", array(self.errors())),
103        ])
104    }
105}

107impl ComponentShape {
108    /// The shape's record (`ComponentShape::toRecord`, and
109    /// `TwoSidedShape::toRecord` for the shapes with axes), in the units CASA
110    /// holds a shape read from a table in (see the module documentation).
111    pub fn to_record(&self) -> RecordValue {
112        let error = self.direction_error();
113        let [longitude, latitude] = crate::table::casacore_angles(self.reference_direction());
114        let direction = RecordValue::new(vec![
115            field("type", string("direction")),
116            field("refer", string(self.reference_direction().refer().as_str())),
117            field("m0", Value::Record(quantity(longitude, "rad"))),
118            field("m1", Value::Record(quantity(latitude, "rad"))),
119            field(
120                "error",
121                Value::Record(RecordValue::new(vec![
122                    field("longitude", Value::Record(quantity_of(error.longitude()))),
123                    field("latitude", Value::Record(quantity_of(error.latitude()))),
124                ])),
125            ),
126        ]);
127        let mut fields = vec![
128            field("type", string(self.shape_type().casa_name())),
129            field("direction", Value::Record(direction)),
130        ];
131        if let Some((axes, errors)) = self.kind().axes() {
132            fields.extend([
133                field(
134                    "majoraxis",
135                    Value::Record(quantity(axes.major_axis() / ARCMIN_RAD, "arcmin")),
136                ),
137                field(
138                    "minoraxis",
139                    Value::Record(quantity(axes.minor_axis() / ARCMIN_RAD, "arcmin")),
140                ),
141                field(
142                    "positionangle",
143                    Value::Record(quantity(axes.position_angle() / DEGREE_RAD, "deg")),
144                ),
145                field(
146                    "majoraxiserror",
147                    Value::Record(quantity(errors.major_axis, "rad")),
148                ),
149                field(
150                    "minoraxiserror",
151                    Value::Record(quantity(errors.minor_axis, "rad")),
152                ),
153                field(
154                    "positionangleerror",
155                    Value::Record(quantity(errors.position_angle, "rad")),
156                ),
157            ]);
158        }
159        RecordValue::new(fields)
160    }
161}

163impl SpectralModel {
164    /// The model's record (`SpectralModel::toRecord`), as a component list's
165    /// `Spectral_Record` column holds it.
166    ///
167    /// # Errors
168    ///
169    /// [`ComponentError::Spectrum`] for a tabular spectrum whose reference
170    /// frequency CASA could not express in its tabulation frame.
171    pub fn to_record(&self) -> Result<RecordValue, ComponentError> {
172        crate::table::spectral_record(self)
173    }
174}

176/// A `QuantumHolder` record.
177fn quantity(value: f64, unit: &str) -> RecordValue {
178    RecordValue::new(vec![
179        field("value", real(value)),
180        field("unit", string(unit)),
181    ])
182}
183
184fn quantity_of(value: &casa_types::quanta::Quantity) -> RecordValue {
185    quantity(value.value(), value.unit().name())
186}
187
188fn field(name: &str, value: Value) -> RecordField {
189    RecordField::new(name, value)
190}
191
192fn string(text: &str) -> Value {
193    Value::Scalar(ScalarValue::String(text.to_string()))
194}
195
196fn real(value: f64) -> Value {
197    Value::Scalar(ScalarValue::Float64(value))
198}
199
200fn reals(values: Vec<f64>) -> Value {
201    Value::Array(ArrayValue::from_f64_vec(values))
