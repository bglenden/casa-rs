2//! The request's `intent` selection resolved on the observation's
3//! catalogue: the observing modes it selects, which name the partitions
4//! that hold them (a partition's samples share one mode).
5//!
6//! A pattern selects an observing mode, as CASA's `intent` selection does
7//! (casacore `MSStateIndex`), when it matches any of the mode's
8//! comma-separated intents: an MSv2 `STATE::OBS_MODE` split at its commas,
9//! or an MSv4 partition's intents. A flagged `STATE` row is selected by no
10//! pattern. A state number selects its MSv2 `STATE` row's mode; an MSv4 has
11//! no states.
12
13use std::collections::BTreeSet;
14
15use casa_imaging_model::IntentSelection;
16use casa_ms::{IntentSelector, parse_intent_selector};
17use casa_observation::{Catalogue, PartitionEntry, StateId};
18
19use super::PrepareError;

21/// The resolved selection: the selected modes, and the states selected
22/// (by number, or by a pattern matching their mode), which must hold their
23/// partitions' samples alone ([`split_state`]).
24pub(super) struct Intents {
25    pub(super) selection: IntentSelection,
26    pub(super) states: BTreeSet<StateId>,
27}
28
29/// The observing mode of partition `entry`: its states' `OBS_MODE` (an
30/// MSv2 partitions its rows by the whole string), else its intents joined
31/// by commas (MSv4). Empty for samples without a state or intents, which
32/// no intent selection selects.
33pub(super) fn observing_mode(catalogue: &Catalogue, entry: &PartitionEntry) -> String {
34    match entry.states.first() {
35        Some(state) => catalogue.states[state.0 as usize].obs_mode.clone(),
36        None => entry.intents.join(","),
37    }
38}
39
40/// The intent selection `text` names over `catalogue`.
41///
42/// # Errors
43///
44/// [`PrepareError::UnmatchedIntent`] for a pattern no mode matches, a state
45/// number the catalogue has no state for (every number on an MSv4), or a
46/// selection of no mode at all.
47pub(super) fn resolve(text: &str, catalogue: &Catalogue) -> Result<Intents, PrepareError> {
48    let unmatched = |selector: String| PrepareError::UnmatchedIntent { selector };
49    // The modes a pattern is matched against: every state's when the store
50    // states them, else every partition's.
51    let known: BTreeSet<String> = if catalogue.states.is_empty() {
52        catalogue
53            .partitions
54            .iter()
55            .map(|entry| observing_mode(catalogue, entry))
56            .collect()
57    } else {
58        catalogue
59            .states
60            .iter()
61            .filter(|state| state.flagged != Some(true))
62            .map(|state| state.obs_mode.clone())
63            .collect()
64    };
65    let mut modes = BTreeSet::new();
66    let mut states = BTreeSet::new();
67    for selector in parse_intent_selector(text) {
68        match &selector {
69            IntentSelector::State(number) => {
70                let state = u32::try_from(*number)
71                    .ok()
72                    .filter(|state| (*state as usize) < catalogue.states.len())
73                    .ok_or_else(|| unmatched(number.to_string()))?;
74                modes.insert(catalogue.states[state as usize].obs_mode.clone());
75                states.insert(StateId(state));
76            }
77            IntentSelector::Mode(pattern) => {
78                let matching: Vec<&String> = known
79                    .iter()
80                    .filter(|mode| selects(&selector, mode))
81                    .collect();
82                if matching.is_empty() {
83                    return Err(unmatched(pattern.clone()));
84                }
85                // A pattern selects every state of the modes it matches.
86                states.extend(
87                    (0..catalogue.states.len())
88                        .filter(|state| {
89                            let state = &catalogue.states[*state];
90                            state.flagged != Some(true) && selects(&selector, &state.obs_mode)
91                        })
92                        .map(|state| StateId(u32::try_from(state).expect("few states"))),
93                );
94                modes.extend(matching.into_iter().cloned());
95            }
96        }
97    }
98    modes.retain(|mode| !mode.is_empty());
99    if modes.is_empty() {
100        return Err(unmatched(text.to_string()));
101    }
102    Ok(Intents {
103        selection: IntentSelection::Only(modes.into_iter().collect()),
104        states,
105    })
106}
107
108/// Whether `selector` selects observing mode `mode`: whether it matches
109/// any of the mode's comma-separated intents, as casacore's `MSStateIndex`
110/// splits `OBS_MODE` at its commas before it matches a name, a pattern or a
111/// regular expression.
112fn selects(selector: &IntentSelector, mode: &str) -> bool {
113    mode.split(',').any(|intent| selector.matches(intent))
114}
115
116/// Whether partition `entry`, of a selected mode, holds samples of a state
117/// the selection leaves out beside a selected one: only a state number can
118/// leave one out, and the MSv2 adapter partitions the rows of every state
119/// of one `OBS_MODE` together, so such a selection cannot be read by
120/// partition.
121pub(super) fn split_state(entry: &PartitionEntry, states: &BTreeSet<StateId>) -> bool {
122    entry.states.iter().any(|state| states.contains(state))
123        && entry.states.iter().any(|state| !states.contains(state))
124}

126#[cfg(test)]
127mod tests {
128    use casa_observation::StateEntry;
129
130    use super::*;
131
132    fn state(obs_mode: &str, flagged: Option<bool>) -> StateEntry {
133        StateEntry {
134            obs_mode: obs_mode.to_string(),
135            cal_k: None,
136            load_k: None,
137            signal: None,
138            reference: None,
139            sub_scan: None,
140            flagged,
141        }
142    }
143
144    fn selected(text: &str, catalogue: &Catalogue) -> Option<Vec<String>> {
145        match resolve(text, catalogue).ok()?.selection {
146            IntentSelection::Only(modes) => Some(modes),
147            IntentSelection::All => None,
148        }
149    }
150
151    /// A name or a pattern selects a mode when it matches any of its
152    /// intents, as casacore splits `OBS_MODE` at its commas (Astra's review
153    /// of #1013); a flagged state is selected by none.
154    #[test]
155    fn an_intent_selects_every_mode_holding_it_and_no_flagged_state() {
156        let catalogue = Catalogue {
157            states: vec![
158                state(
159                    "CALIBRATE_BANDPASS#ON_SOURCE,CALIBRATE_PHASE#ON_SOURCE",
160                    None,
161                ),
162                state("OBSERVE_TARGET#ON_SOURCE", Some(false)),
163                state("CALIBRATE_POINTING#ON_SOURCE", Some(true)),
164            ],
165            ..Catalogue::default()
166        };
167        let both = "CALIBRATE_BANDPASS#ON_SOURCE,CALIBRATE_PHASE#ON_SOURCE".to_string();
168        for text in [
169            "CALIBRATE_PHASE#ON_SOURCE",
170            "*PHASE*",
171            "CALIBRATE_BANDPASS*",
172        ] {
173            assert_eq!(
174                selected(text, &catalogue),
175                Some(vec![both.clone()]),
176                "{text}"
177            );
178        }
179        assert_eq!(
180            selected("*ON_SOURCE", &catalogue),
181            Some(vec![both, "OBSERVE_TARGET#ON_SOURCE".to_string()])
182        );
183        assert!(selected("CALIBRATE_POINTING*", &catalogue).is_none());
184    }
