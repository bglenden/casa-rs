2//! The prediction stage of `close`: the model DATA of each write batch's
3//! rows, degridded by the imaging runtime's block predictor on the worker
4//! team.
5//!
6//! A scan's rows are predicted from every `predict` step covering it, their
7//! predictions summed (one [`PredictionDomain`] per step, as CASA's mappers
8//! sum their degrids), each row reprojected to each step's tangent point by
9//! [`GridFtProjection`] as CASA's FTMachine does (`MosaicFTNew`'s
10//! `girarUVW`, the others' `rotateUVW`), and through the mosaic machine
11//! pointed by its kernels' phase ramp. Scans with the same steps, field and
12//! phase centre share a segment, and one predictor serves a segment; it
13//! is rebuilt when the batches reach a scan of another segment. A run of a
14//! segment's rows is predicted in blocks, straight into the batch's DATA,
15//! which a scan no step covers leaves zero: a block is the whole write
16//! batch when the budget allows, else about [`SMALL_BLOCK_BYTES`] of native
17//! rows.
18//!
19//! On Metal the stage also corrupts the rows it predicts: the device adds
20//! their scans' noise to the model in its memory, with the draws of
21//! `casa_numerics::cell_noise`, before the model is copied into DATA, so
22//! the host makes no pass of its own over those cells (#855). A patch after
23//! `openfromms` adds the DATA on disk after the prediction, and the host
24//! adds that noise after it ([`device_noise`]).
25//!
26//! The rows are as `observe` writes them: the row flag is FLAG_ROW, every
27//! sample is unflagged with unit weight, and an autocorrelation predicts the
28//! model's zero spacing (`GridFT`'s `usezero`). On the CPU a row's
29//! prediction depends only on the row and the steps, not on the workers,
30//! the blocks or the batches.

51/// The noise of scan `scan` in `plan` that a stage predicting the scan may
52/// add on the device ([`PredictionStage::predict`]): all of it, but none
53/// when `from_disk`, the DATA on disk being added after the prediction (a
54/// patch after `openfromms`), whose noise the host adds after it (#886
55/// review), or when gains or leakage corrupt the scan, whose noise follows
56/// them (`session::plan`).
57///
58/// Noise added before a value that is then cancelled keeps only the
59/// precision of the sum: beside visibilities of 2²³ σ and more it is
60/// rounded to steps of σ, or lost. The DATA on disk may hold such a value.
61/// The component lists' and total power's visibilities, which the host
62/// adds after the device's noise, are a sky model's, far below that:
63/// adding them after the noise rounds each sample once, independently, and
64/// the device's noise saves the host's pass (#886 follow-up: 0.34 s of a
65/// 2.1 s Metal `close`).
66pub(super) fn device_noise(plan: &DataPlan, scan: usize, from_disk: bool) -> &[NoiseApplication] {
67    if from_disk || plan.has_jones_for_scan(scan) {
68        &[]
69    } else {
70        plan.noise_for_scan(scan)
71    }
72}

74/// Bytes of the native rows of one prediction block, about, when the budget
75/// cannot hold a whole write batch's ([`super::resources`]). A block ends
76/// with the device's work drained and its predictions copied out, so on
77/// Metal a batch predicted as one block ran 1.5× faster than in blocks of
78/// this size (#855: 1,092-row batches of 4,096 channels, about 120 rows a
79/// small block).
80const SMALL_BLOCK_BYTES: u64 = 16 << 20;

82/// The Metal backend's stage-timing target: with it enabled at `INFO`, each
83/// Metal predictor logs its dispatches, the host's preparation and the
84/// device's time when it drops.
85const METAL_STAGE_TIMING: &str = "casa_imaging::metal_stage_timing";

87/// The scans with the same predict steps, field and phase centre: the
88/// domains a predictor sums, each row's rotation to each domain's tangent
89/// point, and the pointing offset each domain's kernels ramp by.
90struct Segment<'s> {
91    domains: Vec<PredictionDomain<'s>>,
92    projections: Vec<GridFtProjection>,
93    pointing_offsets: Vec<[f64; 2]>,
94}

96/// The prediction stage's state across batches.
97pub(super) struct PredictionStage<'s> {
98    backend: BackendChoice,
99    correlations: usize,
100    channels: usize,
101    segments: Vec<Segment<'s>>,
102    /// Each scan's segment; `None` for a scan no step covers.
103    scan_segments: Vec<Option<usize>>,
104    /// Each spectral window's channel frequencies.
105    frequencies: Vec<Vec<f64>>,
106    /// Each antenna's dish class, which keys the mosaic machine's kernels
107    /// (the other machines ignore it); empty without a mosaic step.
108    antenna_types: Vec<u8>,
109    /// Bytes of every model grid the steps hold, which a device copies.
110    grid_bytes: u64,
111    /// Rows of a block, once started.
112    block_rows: usize,
113    workers: usize,
114    block: NativeBlock,
115    /// Time spent filling blocks of native rows.
116    fill_time: std::time::Duration,
117    /// A run's noise terms and its block's rows, as the device reads them
118    /// when the stage [`Self::noises_on_device`]; allocated on Metal only
119    /// ([`Self::start`]).
120    noise_terms: Vec<MetalNoiseTerm>,
121    noise_rows: Vec<MetalNoiseRow>,
122    /// The most noise applications of one scan: the terms the device holds
123    /// at most.
124    most_terms: usize,
125    projections: Vec<DomainProjection>,
126    /// The predictor of the segment last predicted.
127    predictor: Option<(usize, BlockPredictor<'s>)>,
128    channel_indices: Vec<u32>,
129    correlation_indices: Vec<u32>,
130}

132impl<'s> PredictionStage<'s> {
133    /// The stage for `session`'s predict steps on cells of `(correlations,
134    /// channels)`, on `backend`, or on the CPU when a step projects w;
135    /// `None` when no step predicts.
136    ///
137    /// # Errors
138    ///
139    /// A covered scan's field without grids, which `predict` prepares for
140    /// every field it covers.
141    pub(super) fn new(
142        session: &'s Simulator,
143        (correlations, channels): (usize, usize),
144        backend: BackendChoice,
145    ) -> SimulatorResult<Option<Self>> {
146        let plan = session.data_plan();
147        let predictions: Vec<_> = plan
148            .predictions()
149            .map(|(prediction, _)| prediction)
150            .collect();
151        if predictions.is_empty() {
152            return Ok(None);
153        }
154        // Metal predicts through `GridFT` only; a session that projects w
155        // predicts on the CPU (`predict` refused it on an explicit Metal
156        // backend).
157        let backend = if predictions
158            .iter()
159            .all(|prediction| prediction.model.machine().runs_on_metal())
160        {
161            backend
162        } else {
163            BackendChoice::Cpu
164        };
165        let mut keys = HashMap::new();
166        let mut segments = Vec::new();
167        let mut scan_segments = Vec::with_capacity(session.scans().len());
168        for (index, scan) in session.scans().iter().enumerate() {
169            let steps = plan.predictions_for_scan(index);
170            if steps.is_empty() {
171                scan_segments.push(None);
172                continue;
173            }
174            // Rows are phased on their field (#884); the mosaic machine's
175            // kernels ramp to where the antennas point.
176            let key = (
177                steps,
178                scan.field_id,
179                [scan.phase_center_rad, scan.pointing_rad]
180                    .map(|direction| direction.map(f64::to_bits)),
181            );
182            let segment = match keys.get(&key) {
183                Some(segment) => *segment,
184                None => {
185                    let mut domains = Vec::with_capacity(key.0.len());
186                    let mut projections = Vec::with_capacity(key.0.len());
187                    let mut pointing_offsets = Vec::with_capacity(key.0.len());
188                    for step in key.0.clone() {
189                        let prediction = predictions[step];
190                        let model = prediction.grids_for(scan.field_id).ok_or_else(|| {
191                            SimulatorError::raised(format!(
192                                "predict prepared no grids for field {}",
193                                scan.field_id
194                            ))
195                        })?;
196                        domains.push(PredictionDomain {
197                            operator: &prediction.model.operator,
198                            resampler: &prediction.model.resampler,
199                            model,
200                        });
201                        // `MosaicFT` reprojects through `girarUVW`, the
202                        // others through `rotateUVW`.
203                        let project = match prediction.model.machine() {
204                            FtMachine::Mosaic => GridFtProjection::mosaic_between_j2000_directions,
205                            FtMachine::Grid(_) | FtMachine::WProject(_) => {
206                                GridFtProjection::between_j2000_directions
207                            }
208                        };
209                        projections.push(project(scan.phase_center_rad, prediction.tangent_rad));
210                        pointing_offsets.push(prediction.pointing_offset_rad(scan.pointing_rad));
211                    }
212                    segments.push(Segment {
213                        domains,
214                        projections,
215                        pointing_offsets,
216                    });
217                    keys.insert(key, segments.len() - 1);
218                    segments.len() - 1
219                }
220            };
221            scan_segments.push(Some(segment));
222        }
223        let frequencies = session
224            .spectral_windows()
225            .iter()
226            .map(|window| window.channel_frequencies_hz())
227            .collect();
228        let grid_bytes = predictions
229            .iter()
230            .map(|prediction| prediction.grid_bytes())
231            .sum();
232        // A mosaic step classes the antennas of its time in first-seen
233        // order, so a later step's classes extend an earlier one's.
234        let antenna_types = predictions
235            .iter()
236            .map(|prediction| prediction.antenna_types())
237            .max_by_key(|types| types.len())
238            .unwrap_or_default()
239            .to_vec();
240        let most_terms = (0..session.scans().len())
241            .map(|scan| plan.noise_for_scan(scan).len())
242            .max()
243            .unwrap_or(0);
244        Ok(Some(Self {
245            backend,
246            correlations,
247            channels,
248            segments,
249            scan_segments,
250            frequencies,
251            antenna_types,
252            grid_bytes,
253            block_rows: 0,
254            workers: 0,
255            block: NativeBlock::default(),
256            fill_time: std::time::Duration::ZERO,
257            noise_terms: Vec::new(),
258            noise_rows: Vec::new(),
259            most_terms,
260            projections: Vec::new(),
261            predictor: None,
262            channel_indices: (0..channels as u32).collect(),
263            correlation_indices: (0..correlations as u32).collect(),
264        }))
265    }

278    /// Whether a step predicts scan `scan`'s rows.
279    pub(super) fn predicts(&self, scan: usize) -> bool {
280        self.scan_segments[scan].is_some()
281    }

283    /// The most domains a segment sums.
284    fn largest_domains(&self) -> usize {
285        self.segments
286            .iter()
287            .map(|segment| segment.domains.len())
288            .max()
289            .unwrap_or(0)
290    }

292    /// The block of `rows` rows projected on `domains` domains.
293    fn block_shape(&self, rows: usize, domains: usize) -> BlockShape {
294        BlockShape {
295            rows,
296            spectrum: RowSpectrum::evenly_spaced(self.channels),
297            correlations: self.correlations,
298            domains,
299        }
300    }

302    /// Rows of a small block for write batches of `batch_rows` rows: as
303    /// many as [`SMALL_BLOCK_BYTES`] holds of native rows with all their
304    /// samples, at most a batch and at least one. The block itself shares
305    /// its rows' samples ([`NativeBlock::reset_shared`]), but what the
306    /// predictor holds per row (its model cells and their output) is of
307    /// that size.
308    pub(super) fn small_block_rows(&self, batch_rows: usize) -> usize {
309        let row = self.block_shape(1, self.largest_domains()).bytes().max(1);
310        usize::try_from(SMALL_BLOCK_BYTES / row)
311            .unwrap_or(usize::MAX)
312            .clamp(1, batch_rows.max(1))
313    }

315    /// Bytes the stage holds while `close` predicts blocks of `rows` rows
316    /// with `workers` workers: the predictor of the segment that needs most
317    /// ([`BlockPredictor::bytes`]), one block of native rows sharing their
318    /// samples ([`BlockShape::shared_bytes`]) and each row's projections,
319    /// and on Metal the device copy of every model grid
320    /// ([`casa_imaging_operator::PreparedModelGrids::device_copy`]) and a
321    /// block's noise rows and the most noise terms of a scan, here and in
322    /// the predictor ([`BlockPredictor::noise_bytes`]); and what it keeps
323    /// across blocks ([`Self::state_bytes`]).
324    pub(super) fn bytes(&self, rows: usize, workers: usize) -> u64 {
325        let predictor = self
326            .segments
327            .iter()
328            .map(|segment| {
329                BlockPredictor::bytes(
330                    &segment.domains,
331                    self.backend,
332                    self.block_shape(rows, segment.domains.len()),
333                    workers,
334                )
335            })
336            .max()
337            .unwrap_or(0);
338        let domains = self.largest_domains();
339        let shape = self.block_shape(rows, domains);
340        let projections = (domains * size_of::<DomainProjection>()) as u64;
341        let terms = self.most_terms;
342        let device = match self.backend {
343            BackendChoice::Cpu => 0,
344            BackendChoice::Metal => {
345                // The noise's rows and terms here, and the predictor's copy.
346                let noise = rows * size_of::<MetalNoiseRow>() + terms * size_of::<MetalNoiseTerm>();
347                self.grid_bytes
348                    + noise as u64
349                    + BlockPredictor::noise_bytes(self.backend, shape, terms)
350            }
351        };
352        predictor + shape.shared_bytes() + projections + device + self.state_bytes()
353    }

355    /// Bytes the stage keeps across blocks: each scan's segment, of which an
356    /// opened MeasurementSet may have one a row (#886 review), each
357    /// segment's domains, the windows' channel frequencies and the antennas'
358    /// dish classes.
359    fn state_bytes(&self) -> u64 {
360        let domain = size_of::<PredictionDomain<'_>>()
361            + size_of::<GridFtProjection>()
362            + size_of::<[f64; 2]>();
363        let segments: usize = self
364            .segments
365            .iter()
366            .map(|segment| size_of::<Segment<'_>>() + segment.domains.len() * domain)
367            .sum();
368        let frequencies: usize = self
369            .frequencies
370            .iter()
371            .map(|window| size_of::<Vec<f64>>() + window.len() * size_of::<f64>())
372            .sum();
373        (self.scan_segments.len() * size_of::<Option<usize>>()
374            + segments
375            + frequencies
376            + self.antenna_types.len()) as u64
377    }

379    /// Allocate the block of `rows` rows, predicted with `workers` workers,
380    /// as [`Self::bytes`] charged it.
381    pub(super) fn start(&mut self, rows: usize, workers: usize) {
382        self.block_rows = rows.max(1);
383        self.workers = workers.max(1);
384        let domains = self.largest_domains();
385        self.block = NativeBlock::shared_with_capacity(self.block_shape(self.block_rows, domains));
386        self.projections = Vec::with_capacity(domains);
387        if self.backend == BackendChoice::Metal {
388            self.noise_rows = Vec::with_capacity(self.block_rows);
389            self.noise_terms = Vec::with_capacity(self.most_terms);
390        }
391    }

393    /// Whether the stage adds `noise`, a predicted run's, itself, on the
394    /// device, as it predicts the run ([`Self::predict`]). That happens on
395    /// Metal, where the model is formed in device memory, when every term
396    /// has one σ for every row (`simplenoise`) and draws there exactly as on
397    /// the host ([`casa_numerics::cell_noise::metal_draws_exactly`]);
398    /// otherwise the host adds it after. A Tsys term, whose σ varies by row
399    /// (and for `tsys-atm` by channel), is the host's: the device's terms
400    /// hold one σ each.
401    pub(super) fn noises_on_device(&self, noise: &[NoiseApplication]) -> bool {
402        self.backend == BackendChoice::Metal
403            && noise.iter().all(|application| {
404                application
405                    .uniform()
406                    .is_some_and(|noise| metal_draws_exactly(noise.sigma()))
407            })
408    }

410    /// Whether the stage added `noise`, the [`device_noise`] it was given
411    /// for a run of scan `scan`, as it predicted the run: the host then
412    /// adds none.
413    pub(super) fn noised(&self, scan: usize, noise: &[NoiseApplication]) -> bool {
414        !noise.is_empty() && self.predicts(scan) && self.noises_on_device(noise)
415    }

417    /// Predict each run of `runs`, (scan, first row, rows) of `rows`, into
418    /// `data` (`correlations × channels` samples a row, the rows of `rows`
419    /// from the first); the batch's first row is MAIN row `first_row`. Runs
420    /// of scans no step covers are left as they are. When the stage
421    /// [`Self::noises_on_device`], a predicted run also gets its scan's
422    /// [`device_noise`], `noise(scan)`, on the device before it is copied
423    /// out.
424    pub(super) fn predict<'n>(
425        &mut self,
426        team: &WorkerTeam,
427        (scans, rows): (&[Scan], &[RowGeometry]),
428        runs: &[(usize, usize, usize)],
429        first_row: u64,
430        data: &mut [Complex32],
431        noise: impl Fn(usize) -> &'n [NoiseApplication],
432    ) -> SimulatorResult<()> {
433        let samples = self.correlations * self.channels;
434        for &(scan, start, count) in runs {
435            let Some(segment) = self.scan_segments[scan] else {
436                continue;
437            };
438            self.select(segment)?;
439            self.noise_terms.clear();
440            if self.noises_on_device(noise(scan)) {
441                // A term of σ = 0 adds nothing on either side.
442                self.noise_terms.extend(
443                    noise(scan)
444                        .iter()
445                        .filter_map(NoiseApplication::uniform)
446                        .filter(|noise| noise.sigma() != 0.0)
447                        .map(|noise| MetalNoiseTerm {
448                            key: noise.key(),
449                            sigma: noise.sigma(),
450                        }),
451                );
452            }
453            let scan = &scans[scan];
454            let end = start + count;
455            let mut block_start = start;
456            while block_start < end {
457                let block_end = (block_start + self.block_rows).min(end);
458                let started = std::time::Instant::now();
459                self.fill_block(
460                    segment,
461                    scan,
462                    &rows[block_start..block_end],
463                    first_row + block_start as u64,
464                );
465                self.fill_time += started.elapsed();
466                let Some((_, predictor)) = &mut self.predictor else {
467                    unreachable!("selected above");
468                };
469                let out = &mut data[block_start * samples..block_end * samples];
470                let predicted = if self.noise_terms.is_empty() {
471                    predictor.predict_into(&self.block, team, out)
472                } else {
473                    self.noise_rows.clear();
474                    self.noise_rows.extend(
475                        rows[block_start..block_end]
476                            .iter()
477                            .zip(first_row + block_start as u64..)
478                            .map(|(row, number)| MetalNoiseRow {
479                                row: [number as u32, (number >> 32) as u32],
480                                autocorrelation: u32::from(row.antenna1 == row.antenna2),
481                            }),
482                    );
483                    let noise = DeviceNoise {
484                        rows: &self.noise_rows,
485                        terms: &self.noise_terms,
486                    };
487                    predictor.predict_into_with_noise(&self.block, team, out, noise)
488                };
489                predicted.map_err(|error| SimulatorError::raised(format!("predict: {error}")))?;
490                block_start = block_end;
491            }
492        }
493        Ok(())
494    }

496    /// Make the predictor `segment`'s, unless it is.
497    fn select(&mut self, segment: usize) -> SimulatorResult<()> {
498        if self
499            .predictor
500            .as_ref()
501            .is_some_and(|(current, _)| *current == segment)
502        {
503            return Ok(());
504        }
505        // The previous predictor is dropped first, so at most one is held.
506        self.predictor = None;
507        let domains = &self.segments[segment].domains;
508        let predictor = BlockPredictor::new(
509            domains,
510            self.backend,
511            self.block_shape(self.block_rows, domains.len()),
512            self.workers,
513            tracing::enabled!(target: METAL_STAGE_TIMING, tracing::Level::INFO),
514        )
515        .map_err(|error| SimulatorError::raised(format!("the predictor: {error}")))?;
516        self.predictor = Some((segment, predictor));
517        Ok(())
518    }

520    /// Fill the block with `rows` of `scan`, the first MAIN row
521    /// `first_row`, projected on `segment`'s domains.
522    fn fill_block(&mut self, segment: usize, scan: &Scan, rows: &[RowGeometry], first_row: u64) {
523        let segment = &self.segments[segment];
524        // Every row of the scan has its window's frequencies, and unit
525        // weights and no sample flags, so the block keeps them once.
526        self.block.reset_shared(
527            segment.domains.len(),
528            &self.channel_indices,
529            &self.correlation_indices,
530            &self.frequencies[scan.spectral_window_id],
531        );
532        let antenna_type = |antenna: usize| {
533            // A mosaic step classed every antenna of the scans it covers.
534            debug_assert!(
535                self.antenna_types.is_empty() || antenna < self.antenna_types.len(),
536                "antenna {antenna} has no dish class"
537            );
538            self.antenna_types.get(antenna).copied().unwrap_or(0)
539        };
540        for (offset, row) in rows.iter().enumerate() {
541            self.projections.clear();
542            self.projections.extend(
543                segment
544                    .projections
545                    .iter()
546                    .zip(&segment.pointing_offsets)
547                    .map(|(projection, pointing_offset_rad)| {
548                        let (uvw_m, phase_shift_m) = projection.apply(row.uvw_m);
549                        DomainProjection {
550                            uvw_m,
551                            phase_shift_m,
552                            pointing_offset_rad: *pointing_offset_rad,
553                        }
554                    }),
555            );
556            let header = NativeRowHeader {
557                row_flag: row.flagged,
558                context: RowContext {
559                    original_uvw_m: Some(row.uvw_m),
560                    time_s: 0.0,
561                    antennas: [row.antenna1 as u32, row.antenna2 as u32],
562                    antenna_types: [antenna_type(row.antenna1), antenna_type(row.antenna2)],
563                    parallactic_angle_rad: [0.0, 0.0],
564                    field: scan.field_id as u32,
565                    spectral_window: scan.spectral_window_id as u32,
566                },
567                address: RowAddress {
568                    physical_row: first_row + offset as u64,
569                    data_description: scan.spectral_window_id as i32,
570                },
571            };
572            self.block.push_shared_row(header, &self.projections);
573        }
574    }
