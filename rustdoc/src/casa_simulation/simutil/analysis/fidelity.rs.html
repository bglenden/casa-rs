12/// What [`fidelity()`] wrote and measured.
13#[derive(Debug, Clone, PartialEq)]
14pub struct Fidelity {
15    /// `<image>.diff`: the convolved model less the image, in Jy/beam.
16    pub difference: PathBuf,
17    /// `<image>.fidelity`: the convolved model's magnitude over the
18    /// difference's, floored; a ratio, without a unit.
19    pub fidelity: PathBuf,
20    /// The difference's median absolute deviation from its median, over its
21    /// unmasked pixels; 0 when it has none. CASA's log calls it the rms
22    /// difference.
23    pub medabsdevmed: f64,
24    /// The scalar fidelity: the largest magnitude of the convolved model
25    /// over [`medabsdevmed`](Self::medabsdevmed) (infinite when that is 0, as
26    /// numpy divides).
27    pub scalar: f64,
28}
29
30/// `simanalyze`'s analysis of `image_root.image` against the sky model:
31/// with `convolved` the [`convimage`](super::convimage()) output and
32/// `out_flat` the image's flat, write
33/// - `<image_root>.diff`, `convolved − out_flat`, labelled Jy/beam;
34/// - `<image_root>.fidelity`, `|convolved| / max(|diff|, d/√2)`, where `d`
35///   is the difference's median absolute deviation;
36///
37/// and return `d` and the scalar fidelity. Both images take the convolved
38/// model's coordinates and beam, and are masked where either input is, as
39/// CASA's image expressions are. The arithmetic is in single precision, as
40/// those expressions compute on single-precision images.
41///
42/// Two departures (`docs/CASA (C++) bugs.md`):
43/// - CASA writes `d/√2` into its expression with `%f`, rounding it to six
44///   decimal places; casa-rs uses the value itself.
45/// - CASA's fidelity carries the convolved model's unit, Jy/beam, which its
46///   image expression copies; it is a ratio, and casa-rs gives it no unit.
47///
48/// # Errors
49///
50/// Inputs of different shapes, or a failure to read or write an image or to
51/// compute the statistics.
52pub fn fidelity(
53    convolved: &Path,
54    out_flat: &Path,
55    image_root: &Path,
56) -> Result<Fidelity, AnalysisError> {
57    let model = PagedImage::<f32>::open(convolved)?;
58    let image = PagedImage::<f32>::open(out_flat)?;
59    if model.shape() != image.shape() {
60        return Err(AnalysisError::Analysis(format!(
61            "{} has shape {:?} and {} has shape {:?}; the difference needs them the same",
62            convolved.display(),
63            model.shape(),
64            out_flat.display(),
65            image.shape()
66        )));
67    }
68    let model_pixels = model.get()?;
69    let model_mask = model.default_mask()?;
70    let mask = both(model_mask.clone(), image.default_mask()?);
71    let mut difference = model_pixels.clone();
72    difference.zip_mut_with(&image.get()?, |model, image| *model -= *image);
73    drop(image);
74    let coordinates = model.coordinates().clone();
75    let info = model.image_info()?;
76
77    let difference_path = suffixed(image_root, ".diff");
78    write_product(
79        &difference_path,
80        &difference,
81        coordinates.clone(),
82        "Jy/beam",
83        &info,
84        mask.as_ref(),
85    )?;
86    let medabsdevmed = medabsdevmed(&difference, mask.as_ref())?;
87
88    let floor = (medabsdevmed / 2f64.sqrt()) as f32;
89    let mut fidelity = model_pixels.clone();
90    Zip::from(&mut fidelity)
91        .and(&difference)
92        .for_each(|model, difference| *model = model.abs() / difference.abs().max(floor));
93    let fidelity_path = suffixed(image_root, ".fidelity");
94    write_product(
95        &fidelity_path,
96        &fidelity,
97        coordinates,
98        "",
99        &info,
100        mask.as_ref(),
101    )?;
102
103    Ok(Fidelity {
104        difference: difference_path,
105        fidelity: fidelity_path,
106        medabsdevmed,
107        scalar: largest_magnitude(&model_pixels, model_mask.as_ref()) / medabsdevmed,
108    })
109}

111/// The largest magnitude of the pixels `mask` keeps (`ia.statistics` of
112/// `abs(image)`), or 0 when it keeps none.
113fn largest_magnitude(pixels: &ArrayD<f32>, mask: Option<&ArrayD<bool>>) -> f64 {
114    let magnitude = |value: &f32| f64::from(value.abs());
115    let largest = match mask {
116        Some(mask) => {
117            Zip::from(pixels)
118                .and(mask)
119                .fold(None, |largest: Option<f64>, value, valid| {
120                    if *valid {
121                        Some(largest.map_or(magnitude(value), |l| l.max(magnitude(value))))
122                    } else {
123                        largest
124                    }
125                })
126        }
127        None => pixels.iter().map(magnitude).reduce(f64::max),
128    };
129    largest.unwrap_or(0.0)
130}

132/// `ia.statistics(robust=True)['medabsdevmed']` of `pixels` where `mask`
133/// keeps them, read as `simanalyze` reads it: 0 when it is not a number or
134/// no pixel is kept.
135fn medabsdevmed(pixels: &ArrayD<f32>, mask: Option<&ArrayD<bool>>) -> Result<f64, AnalysisError> {
136    if mask.is_some_and(|mask| !mask.iter().any(|valid| *valid)) {
137        return Ok(0.0);
138    }
139    let lattice = ArrayLattice::new(pixels.clone());
140    let mut statistics = LatticeStatistics::new(&lattice);
141    if let Some(mask) = mask {
142        statistics.set_pixel_mask(mask.clone());
143    }
144    let value = statistics
145        .get_statistic(Statistic::MedAbsDevMed)?
146        .iter()
147        .next()
148        .copied()
149        .unwrap_or(0.0);
150    Ok(if value.is_nan() { 0.0 } else { value })
151}
