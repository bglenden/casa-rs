11#[derive(Default)]
12pub(super) struct ObservedHand {
13    pub(super) data: Complex32,
14    pub(super) weight: f32,
15    pub(super) samples: Vec<PointingSample>,
16}

18pub(super) struct Observation {
19    pub(super) weight: f32,
20    pub(super) hands: [ObservedHand; 2],
21}
22
23pub(super) struct Baseline {
24    pub(super) antennas: [usize; 2],
25    pub(super) observations: Vec<Observation>,
26    pub(super) good_channels: BTreeSet<usize>,
27}
28
29#[derive(Clone, Copy, Default)]
30struct Assessment {
31    values: [Complex32; 3],
32    weight: f32,
33}
34
35impl Baseline {
36    fn assess(
37        &self,
38        model: &PreparedEpModel,
39        offsets: &[[f32; 4]],
40        axes: [usize; 2],
41        hand: usize,
42    ) -> Assessment {
43        let [a, b] = self.antennas;
44        let offset = axes.map(|axis| f64::from(offsets[a][axis]) + f64::from(offsets[b][axis]));
45        let mut result = Assessment::default();
46        let mut hand_weight = 0.;
47        for observation in &self.observations {
48            let hand = &observation.hands[hand];
49            let mut predicted = [Complex32::default(); 3];
50            for sample in &hand.samples {
51                for (sum, value) in predicted.iter_mut().zip(sample.evaluate(model, offset)) {
52                    *sum += value;
53                }
54            }
55            predicted[0] -= hand.data;
56            for (sum, value) in result.values.iter_mut().zip(predicted) {
57                *sum += value * hand.weight;
58            }
59            hand_weight += hand.weight;
60            result.weight += observation.weight;
61        }
62        if hand_weight > 0. && !self.good_channels.is_empty() {
63            // VisBuffAccumulator first normalizes by the hand weight over
64            // time. freqAveCubes then divides by the union of good channels.
65            let scale = (hand_weight * self.good_channels.len() as f32).recip();
66            result.values = result.values.map(|value| value * scale);
67        }
68        result
69    }
70}

72pub(super) fn solve(
73    model: &PreparedEpModel,
74    baselines: &[Baseline],
75    offsets: &mut [[f32; 4]],
76) -> bool {
77    let mut converged = false;
78    for hand in 0..model.correlation_planes.len() {
79        // dpbwgrad uses the conjugate kernel plane. solve2 fits the
80        // opposite receptor's two offsets for each measured correlation.
81        let axes = [1 - hand, 3 - hand];
82        let mut current = assess(model, baselines, offsets, axes, hand);
83        let Some(mut best_chi) = chi_square(&current) else {
84            return false;
85        };
86        let mut best = offsets.to_vec();
87        let mut gain = 0.01;
88        converged = false;
89        for _ in 0..250 {
90            descend(baselines, &current, offsets, axes, gain);
91            current = assess(model, baselines, offsets, axes, hand);
92            let Some(chi) = chi_square(&current) else {
93                return false;
94            };
95            if (chi - best_chi).abs() < 1e-9 {
96                offsets.copy_from_slice(&best);
97                converged = true;
98                break;
99            }
100            if chi < best_chi {
101                best_chi = chi;
102                best.copy_from_slice(offsets);
103            } else {
104                offsets.copy_from_slice(&best);
105                gain *= 0.5;
106                if gain < 1e-20 {
107                    converged = true;
108                    break;
109                }
110                current = assess(model, baselines, offsets, axes, hand);
111            }
112        }
113    }
114    converged
115}

117fn assess(
118    model: &PreparedEpModel,
119    baselines: &[Baseline],
120    offsets: &[[f32; 4]],
121    axes: [usize; 2],
122    hand: usize,
123) -> Vec<Assessment> {
124    baselines
125        .iter()
126        .map(|baseline| baseline.assess(model, offsets, axes, hand))
127        .collect()
128}
129
130fn chi_square(assessment: &[Assessment]) -> Option<f64> {
131    let (value, weight) = assessment.iter().fold((0., 0.), |(sum, weight), item| {
132        let residual = item.values[0];
133        let power = f64::from(residual.re).powi(2) + f64::from(residual.im).powi(2);
134        let w = f64::from(item.weight);
135        (sum + w * power, weight + w)
136    });
137    let mean = value / weight;
138    (weight > 0. && mean.is_finite()).then_some(mean)
139}

141fn descend(
142    baselines: &[Baseline],
143    current: &[Assessment],
144    offsets: &mut [[f32; 4]],
145    axes: [usize; 2],
146    gain: f64,
147) {
148    let antennas = offsets.len();
149    let mut by_antenna = vec![[Complex32::default(); 3]; antennas * antennas];
150    let mut weights = vec![0.; antennas];
151    for (baseline, fit) in baselines.iter().zip(current) {
152        if fit.weight <= 0. {
153            continue;
154        }
155        let [a, b] = baseline.antennas;
156        let [residual, azimuth, elevation] = fit.values;
157        let mut values = [residual, azimuth * fit.weight, elevation * fit.weight];
158        if a <= b {
159            values = values.map(|value| value.conj());
160        }
161        for (i, j) in [(a, b), (b, a)] {
162            for (sum, value) in by_antenna[i * antennas + j].iter_mut().zip(values) {
163                *sum += value;
164            }
165            weights[i] += f64::from(fit.weight);
166        }
167    }
168    for ((values, offset), weight) in by_antenna.chunks_exact(antennas).zip(offsets).zip(weights) {
169        for (component, axis) in axes.into_iter().enumerate() {
170            if weight == 0. {
171                offset[axis] = 0.;
172                continue;
173            }
174            let mut numerator = 0.;
175            let mut denominator = 0.;
176            for row in values {
177                let gradient = row[component + 1];
178                numerator += (row[0].conj() * gradient).re;
179                denominator += gradient.norm_sqr();
180            }
181            if denominator > 0. {
182                let step = -2. * gain * weight * f64::from(numerator) / f64::from(denominator);
183                offset[axis] = (f64::from(offset[axis]) + step) as f32;
184            }
185        }
186    }
187}
