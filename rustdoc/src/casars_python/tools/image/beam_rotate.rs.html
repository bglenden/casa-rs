18struct BeamChange {
19    original: StoredGaussianBeam,
20    rotated: StoredGaussianBeam,
21}
22
23impl BeamChange {
24    fn new(original: StoredGaussianBeam, angle: &Quantity) -> PyResult<Self> {
25        let position = original.wrapped_position_angle().map_err(raised)?;
26        let increment = casacore::get_conformant_value(angle, position.unit()).map_err(raised)?;
27        let new_position =
28            Quantity::with_unit(position.value() + increment, position.unit().clone());
29        let new_position = wrap_position_angle(&new_position).map_err(raised)?;
30        let rotated = original
31            .with_component(BeamComponent::PositionAngle, new_position)
32            .map_err(raised)?;
33        Ok(Self { original, rotated })
34    }
35
36    fn messages(&self, name: &str, kind: &str) -> PyResult<[String; 2]> {
37        let describe = |beam: &StoredGaussianBeam| {
38            let format = |value: &Quantity| {
39                format!(
40                    "{} {}",
41                    casacore::stream_double(value.value()),
42                    value.unit().name()
43                )
44            };
45            Ok::<_, PyErr>(format!(
46                "major: {}, minor: {}, pa: {}",
47                format(beam.component(BeamComponent::Major)),
48                format(beam.component(BeamComponent::Minor)),
49                format(&beam.wrapped_position_angle().map_err(raised)?)
50            ))
51        };
52        Ok([
53            format!(
54                "Original {name} {kind}restoring beam: {}",
55                describe(&self.original)?
56            ),
57            format!(
58                "New {name} {kind}restoring beam: {}",
59                describe(&self.rotated)?
60            ),
61        ])
62    }
63}

65fn multiple_beam_messages(name: &str, min: &BeamChange, max: &BeamChange) -> PyResult<Vec<String>> {
66    let mut messages = vec![
67        "This image has multiple beams. Reporting the rotation of the minimum and maximum area beams:"
68            .into(),
69    ];
70    messages.extend(min.messages(name, "minimum area ")?);
71    messages.extend(max.messages(name, "maximum area ")?);
72    Ok(messages)
73}

75fn rotate_beam_record(
76    beams: &ImageBeamSet,
77    view: &BeamView,
78    updated: &mut RecordValue,
79    angle: &Quantity,
80    name: &str,
81) -> PyResult<Vec<String>> {
82    let change = |channel, stokes| {
83        let beam = stored_beam_record(Some(view.record()), beams, channel, stokes);
84        BeamChange::new(
85            StoredGaussianBeam::from_record(&beam).map_err(raised)?,
86            angle,
87        )
88    };
89    if beams.is_single() {
90        let single = change(0, 0)?;
91        updated.remove("perplanebeams");
92        updated.upsert("restoringbeam", Value::Record(single.rotated.to_record()));
93        return single.messages(name, "").map(Vec::from);
94    }
95    let mut grid = subrecord(view.record(), "perplanebeams")
96        .cloned()
97        .unwrap_or_else(|| beams.to_record());
98    let [minimum, maximum] = view.extrema();
99    let indices = |index| (index % beams.n_channels(), index / beams.n_channels());
100    for stokes in 0..beams.n_stokes() {
101        for channel in 0..beams.n_channels() {
102            let plane = change(channel, stokes)?;
103            let index = stokes * beams.n_channels() + channel;
104            grid.upsert(
105                format!("*{index}"),
106                Value::Record(plane.rotated.to_record()),
107            );
108        }
109    }
110    updated.remove("restoringbeam");
111    updated.upsert("perplanebeams", Value::Record(grid));
112    let (min_chan, min_stokes) = indices(minimum);
113    let (max_chan, max_stokes) = indices(maximum);
114    multiple_beam_messages(
115        name,
116        &change(min_chan, min_stokes)?,
117        &change(max_chan, max_stokes)?,
118    )
119}

121#[pymethods]
122impl PyImageTool {
123    #[pyo3(signature = (angle=Variant::Empty), text_signature = "($self, angle=[])")]
124    fn rotatebeam(&self, py: Python<'_>, angle: Variant) -> PyResult<bool> {
125        let result = py.detach(|| {
126            let state = self.state();
127            let Some(attachment) = state.image.as_ref() else {
128                return Ok(None);
129            };
130            let mut image = attachment.lock();
131            image.check_single_precision("rotatebeam")?;
132            let rotation = quantity(&angle)?;
133            let degrees = Unit::new("deg").expect("built-in angular unit");
134            if !rotation.unit().conformant(&degrees) {
135                return Err(raised("Exception: Quantity is not an angle."));
136            }
137            attachment.require_lock(&mut image, LockType::Write)?;
138            let mut view = attachment.beam_view();
139            let beams = view.beam_set().map_err(raised)?;
140            let name = image
141                .path()
142                .map_or_else(|| "Temporary_Image".into(), |path| path.display().to_string());
143            let (messages, updated_info, empty_angle) = if beams.is_empty() {
144                let zero = Quantity::new(0.0, "arcsec").expect("built-in angular unit");
145                let previous = Quantity::with_unit(view.empty_angle_degrees, degrees);
146                let beam = StoredGaussianBeam::new(zero.clone(), zero, previous).map_err(raised)?;
147                let change = BeamChange::new(beam, &rotation)?;
148                let messages = multiple_beam_messages(&name, &change, &change)?;
149                let angle = change.rotated.component(BeamComponent::PositionAngle).value();
150                (messages, None, angle)
151            } else {
152                let mut updated = match image_info_record(&image) {
153                    Some(info) => info.clone(),
154                    None => image_info(&image).map_err(raised)?.to_record(),
155                };
156                let messages = rotate_beam_record(&beams, &view, &mut updated, &rotation, &name)?;
157                casa_images::ImageInfo::from_record(&updated).map_err(raised)?;
158                (messages, Some(updated), 0.0)
159            };
160            let mut history = Vec::with_capacity(messages.len() + 2);
161            if state.history {
162                history.push("Ran ia.rotatebeam".into());
163                history.push(format!("ia.rotatebeam(angle={})", input_history(&angle)));
164            }
165            history.extend(messages.iter().cloned());
166            let location = "::BeamManipulator::rotate";
167            with_pixels!(&mut *image, storage => match storage {
168                Storage::Paged(image) => image.save_image_info_with_history(updated_info.as_ref(), location, &history),
169                Storage::Temp { image, .. } => {
170                    if let Some(updated) = &updated_info {
171                        image.set_image_info_record(updated.clone()).map_err(raised)?;
172                    }
173                    image.append_history_at(location, &history)
174                },
175            }).map_err(raised)?;
176            if let Some(updated) = updated_info {
177                view.capture(&updated);
178            } else {
179                view.empty_angle_degrees = empty_angle;
180            }
181            Ok::<_, PyErr>(Some(messages))
182        })?;
183        let Some(messages) = result else {
184            return Ok(false);
185        };
186        let logs = messages.into_iter().map(LogLine::info).collect::<Vec<_>>();
187        emit(py, "casars.tools.image", &logs)?;
188        Ok(true)
189    }
