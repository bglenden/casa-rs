2//! What writing a plan holds, admitted when the plan is made: beside what
3//! the plan keeps, the subtables, `HISTORY` and the `POINTING` table until
4//! the save, then the largest of the pointing reads (each input's in
5//! turn), the MAIN writers with one batch of every selected group, and the
6//! largest ephemeris table, written once they are done. Every input
7//! partition's batch must hold one integration; the MAIN writers assemble
8//! their tile bands in memory when the budget holds them, and otherwise
9//! write each tile row in place.
10

25/// Bytes allowed the created MeasurementSet's tables beside their rows,
26/// from creation through the save: their descriptions and storage managers
27/// and the save's encoding (about 2 MB for an MSv2's subtables).
28const CREATED_TABLES_BYTES: usize = 3 << 20;
29
30/// MAIN's write plan, and what the plan admitted.
31pub(super) struct Writing {
32    pub(super) plan: MeasurementSetWritePlan,
33    pub(super) charges: MemoryCharges,
34}
35
36/// The batches writing reads.
37struct Batches {
38    /// The largest, of `batch_bytes` at most.
39    bytes: usize,
40    /// The most samples one holds.
41    samples: usize,
42    /// The largest integration, one time of every group a partition reads.
43    integration: usize,
44}
45
46/// The batches writing reads, of `batch_bytes` at most; refused where an
47/// input partition's integration exceeds `batch_bytes`.
48fn batches(
49    inputs: &[Input],
50    combined: &Combined,
51    batch_bytes: usize,
52) -> Result<Batches, ObservationError> {
53    let mut batches = Batches {
54        bytes: 0,
55        samples: 0,
56        integration: 0,
57    };
58    for (input, map) in inputs.iter().zip(&combined.rows) {
59        let observation = input.observation();
60        for id in observation.catalogue().partition_ids() {
61            let placement = &map.partitions[id.0 as usize];
62            let sizing = Sizing::of(observation, id, &placement.groups, Roles::ALL)?;
63            batches.integration = batches.integration.max(sizing.integration());
64            let times = sizing.times_per_batch(batch_bytes, None);
65            batches.bytes = batches.bytes.max(times * sizing.integration());
66            batches.samples = batches.samples.max(times * sizing.samples_per_time);
67        }
68    }
69    if batches.integration > batch_bytes {
70        return Err(ObservationError::OverBudget {
71            what: "a batch of one integration of every selected group",
72            required: batches.integration,
73            budget: batch_bytes,
74        });
75    }
76    Ok(batches)
77}

79/// Admit to `ledger` what writing `combined`'s `occupancy` from `inputs`
80/// (with their `pointing`), `data_columns` and the `history` rows holds, in
81/// batches of `batch_bytes`; MAIN's write plan, and the plan's charges.
82///
83/// # Errors
84///
85/// [`ObservationError::OverBudget`] when a batch cannot hold one
86/// integration, or writing beside what the plan keeps exceeds the budget
87/// with the MAIN writers at their least.
88pub(super) fn admit(
89    ledger: &mut Ledger,
90    (inputs, combined, pointing): (&[Input], &Combined, &[PointingPlan]),
91    (data_columns, occupancy, history): (&[&'static str], &Occupancy, &[HistoryRow]),
92    batch_bytes: usize,
93) -> Result<Writing, ObservationError> {
94    let (planning, kept) = (ledger.peak(), ledger.held());
95    let Batches {
96        bytes: batch,
97        samples,
98        integration,
99    } = batches(inputs, combined, batch_bytes)?;
100    let catalogue = &combined.catalogue;
101    let columns = msv2_rows::array_columns(catalogue, data_columns, occupancy)?;
102    // Each input partition's rows are written together, in one cell shape,
103    // so a column's rows come as at most a run per input partition and one
104    // of undefined cells.
105    let runs = 1 + inputs
106        .iter()
107        .map(|input| input.observation().catalogue().partitions.len())
108        .sum::<usize>();
109    let (least, buffered) =
110        MeasurementSetWritePlan::variable_array_column_bytes(&columns, runs).map_err(storage)?;
111    let cell = catalogue
112        .partitions
113        .iter()
114        .map(|entry| entry.frequencies_hz.len() * entry.correlations.len())
115        .max()
116        .unwrap_or(0);
117    let rows = msv2_rows::held_bytes(samples, cell);
118    let held = CREATED_TABLES_BYTES
119        + subtables_bytes(catalogue)
120        + history_table_bytes(history)
121        + pointing
122            .iter()
123            .map(|plan| plan.gathered().0 * TABLE_ROW_BYTES)
124            .sum::<usize>();
125    let reads = pointing.iter().map(PointingPlan::peak).max().unwrap_or(0);
126    // The ephemeris tables are written once MAIN's writers are done.
127    let ephemerides = msv2_ephemeris::tables_bytes(catalogue);
128    let writing = |plan: &MeasurementSetWritePlan| {
129        let writer = plan.maximum_resident_bytes() + rows;
130        (writer, held + reads.max(writer + batch).max(ephemerides))
131    };
132    let least_plan = msv2_rows::write_plan(columns.clone(), occupancy, least)?;
133    let buffered_plan = msv2_rows::write_plan(columns, occupancy, buffered)?;
134    let least_writing = writing(&least_plan).1;
135    let plan = if kept + writing(&buffered_plan).1 <= ledger.budget() {
136        buffered_plan
137    } else {
138        least_plan
139    };
140    let (writer, total) = writing(&plan);
141    ledger.admit("writing the MeasurementSet", total)?;
142    Ok(Writing {
143        plan,
144        charges: MemoryCharges {
145            kept,
146            planning,
147            writing: total,
148            writer,
149            batch,
150            least: planning.max(kept + least_writing),
151            integration,
152        },
153    })
154}
