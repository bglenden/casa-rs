175#[derive(Clone, Copy)]
176struct AllocationProbe {
177    pointer: usize,
178    capacity: usize,
179}
180
181impl AllocationProbe {
182    fn capture<T>(values: &Vec<T>) -> Self {
183        Self {
184            pointer: values.as_ptr() as usize,
185            capacity: values.capacity(),
186        }
187    }
188
189    fn reused<T>(self, values: &[T]) -> bool {
190        self.capacity > 0
191            && self.capacity >= values.len()
192            && self.pointer == values.as_ptr() as usize
193    }
194}
195
196#[derive(Default)]
197struct AllocationCounter {
198    reused: u64,
199    allocated: u64,
200}
201
202impl AllocationCounter {
203    fn observe<T>(&mut self, probe: AllocationProbe, values: &[T]) {
204        if probe.reused(values) {
205            self.reused += 1;
206        } else if !values.is_empty() {
207            self.allocated += 1;
208        }
209    }
210
211    const fn report(self) -> SelectedObservationBufferAllocationReport {
212        SelectedObservationBufferAllocationReport {
213            reused_storage_buffers: self.reused,
214            allocated_storage_buffers: self.allocated,
215        }
216    }
217}

241/// Project the allocations performed by [`MeasurementSet::fill_selected_observation_buffer`].
242pub(crate) fn selected_observation_buffer_residency(
243    rows: usize,
244    packed_samples: usize,
245    weight_values: usize,
246    visibility_bytes: usize,
247) -> Option<SelectedObservationBufferResidency> {
248    let row_indices = rows.checked_mul(size_of::<usize>())?;
249    let visibility = packed_samples.checked_mul(visibility_bytes)?;
250    let flags = packed_samples.checked_mul(size_of::<bool>())?;
251    let weights = weight_values.checked_mul(size_of::<f32>())?;
252    let scalar_payload =
253        rows.checked_mul(10 * size_of::<i32>() + 4 * size_of::<f64>() + size_of::<bool>())?;
254    let uvw = rows.checked_mul(size_of::<[f64; 3]>())?;
255    let resident_bytes = row_indices
256        .checked_add(visibility)?
257        .checked_add(flags)?
258        .checked_add(weights)?
259        .checked_add(scalar_payload)?
260        .checked_add(uvw)?;
261
262    // The typed scalar batch owns fifteen named vectors until they move into
263    // the destination buffer. Hashbrown keeps at most 7/8 load, so derive the
264    // allocated bucket count from the actual column count rather than hiding a
265    // fixture-specific allowance.
266    const SCALAR_COLUMNS: [&str; 15] = [
267        "DATA_DESC_ID",
268        "FIELD_ID",
269        "ANTENNA1",
270        "ANTENNA2",
271        "FEED1",
272        "FEED2",
273        "TIME",
274        "TIME_CENTROID",
275        "INTERVAL",
276        "EXPOSURE",
277        "SCAN_NUMBER",
278        "STATE_ID",
279        "OBSERVATION_ID",
280        "ARRAY_ID",
281        "FLAG_ROW",
282    ];
283    let minimum_buckets = SCALAR_COLUMNS
284        .len()
285        .checked_mul(8)?
286        .checked_add(6)?
287        .checked_div(7)?;
288    let scalar_buckets = minimum_buckets.checked_next_power_of_two()?;
289    let scalar_bucket_bytes = scalar_buckets.checked_mul(
290        size_of::<String>()
291            .checked_add(size_of::<RequiredScalarColumnValues>())?
292            .checked_add(1)?,
293    )?;
294    let scalar_map_bytes = scalar_bucket_bytes
295        .checked_add(SCALAR_COLUMNS.iter().map(|name| name.len()).sum::<usize>())?;
296    let fill_peak_bytes = resident_bytes
297        .checked_add(row_indices)?
298        .checked_add(scalar_map_bytes)?;
299    Some(SelectedObservationBufferResidency {
300        resident_bytes,
301        fill_peak_bytes,
302    })
303}

349    fn retained_bytes(&self, capacity: bool) -> Option<usize> {
350        let count = |len: usize, cap: usize| if capacity { cap } else { len };
351        let mut bytes = 0_usize;
352        macro_rules! add_vec {
353            ($values:expr, $type:ty) => {{
354                let values = $values;
355                bytes = bytes.checked_add(
356                    count(values.len(), values.capacity()).checked_mul(size_of::<$type>())?,
357                )?;
358            }};
359        }
360        add_vec!(&self.row_indices, usize);
361        match self.visibility.as_ref() {
362            Some(SelectedStoredVisibilities::Float32(values)) => add_vec!(values, f32),
363            Some(SelectedStoredVisibilities::Complex32(values)) => {
364                add_vec!(values, casa_types::Complex32)
365            }
366            None => {}
367        }
368        add_vec!(&self.flags, bool);
369        match self.weights.as_ref() {
370            Some(SelectedStoredWeights::PerRow(values))
371            | Some(SelectedStoredWeights::PerChannel(values)) => add_vec!(values, f32),
372            None => {}
373        }
374        add_vec!(&self.row_flag, bool);
375        add_vec!(&self.uvw_m, f64);
376        add_vec!(&self.data_description_ids, i32);
377        add_vec!(&self.field_ids, i32);
378        add_vec!(&self.antenna1, i32);
379        add_vec!(&self.antenna2, i32);
380        add_vec!(&self.feed1, i32);
381        add_vec!(&self.feed2, i32);
382        add_vec!(&self.time_mjd_seconds, f64);
383        add_vec!(&self.time_centroid_mjd_seconds, f64);
384        add_vec!(&self.interval_seconds, f64);
385        add_vec!(&self.exposure_seconds, f64);
386        add_vec!(&self.scan_numbers, i32);
387        add_vec!(&self.state_ids, i32);
388        add_vec!(&self.observation_ids, i32);
389        add_vec!(&self.array_ids, i32);
390        Some(bytes)
391    }

393    /// Return one stored sample by channel, row, and correlation block offsets.
394    #[must_use]
395    pub(crate) fn sample(
396        &self,
397        channel_offset: usize,
398        row_offset: usize,
399        correlation_offset: usize,
400    ) -> Option<SelectedStoredSample> {
401        if channel_offset >= self.channel_range.count
402            || row_offset >= self.row_count()
403            || correlation_offset >= self.correlation_count
404        {
405            return None;
406        }
407        let sample_index = packed_sample_index(
408            channel_offset,
409            row_offset,
410            correlation_offset,
411            self.row_count(),
412            self.correlation_count,
413        );
414        let row_weight_index = row_offset * self.correlation_count + correlation_offset;
415        let visibility = match self.visibility.as_ref()? {
416            SelectedStoredVisibilities::Float32(values) => {
417                SelectedStoredVisibility::Float32(*values.get(sample_index)?)
418            }
419            SelectedStoredVisibilities::Complex32(values) => {
420                let value = values.get(sample_index)?;
421                SelectedStoredVisibility::Complex32([value.re, value.im])
422            }
423        };
424        let input_weight = match self.weights.as_ref()? {
425            SelectedStoredWeights::PerRow(values) => *values.get(row_weight_index)?,
426            SelectedStoredWeights::PerChannel(values) => *values.get(sample_index)?,
427        };
428        let uvw_start = row_offset.checked_mul(3)?;
429        let uvw = self.uvw_m.get(uvw_start..uvw_start + 3)?;
430        Some(SelectedStoredSample {
431            physical_row: *self.row_indices.get(row_offset)?,
432            data_description_id: *self.data_description_ids.get(row_offset)?,
433            visibility,
434            channel_flag: *self.flags.get(sample_index)?,
435            row_flag: *self.row_flag.get(row_offset)?,
436            input_weight,
437            uvw_m: [uvw[0], uvw[1], uvw[2]],
438            time_mjd_seconds: *self.time_mjd_seconds.get(row_offset)?,
439            time_centroid_mjd_seconds: *self.time_centroid_mjd_seconds.get(row_offset)?,
440            interval_seconds: *self.interval_seconds.get(row_offset)?,
441            exposure_seconds: *self.exposure_seconds.get(row_offset)?,
442            field_id: *self.field_ids.get(row_offset)?,
443            antenna1: *self.antenna1.get(row_offset)?,
444            antenna2: *self.antenna2.get(row_offset)?,
445            feed1: *self.feed1.get(row_offset)?,
446            feed2: *self.feed2.get(row_offset)?,
447            scan_number: *self.scan_numbers.get(row_offset)?,
448            state_id: *self.state_ids.get(row_offset)?,
449            observation_id: *self.observation_ids.get(row_offset)?,
450            array_id: *self.array_ids.get(row_offset)?,
451        })
452    }

563impl MeasurementSet {
564    /// Fill one bounded block with the closed selected-observation storage column set.
565    ///
566    /// This selected-channel operation requires a lazily reopened disk-backed MeasurementSet with
567    /// no pending array-cell writes. It never falls back to cloning complete array cells.
568    pub(crate) fn fill_selected_observation_buffer(
569        &self,
570        request: &SelectedObservationBufferRequest<'_>,
571        buffer: &mut SelectedObservationBuffer,
572    ) -> MsResult<SelectedObservationBufferFillReport> {
573        let fill_started = Instant::now();
574        validate_request(self, request)?;
575        let mut allocation = AllocationCounter::default();
576        buffer.row_indices.clear();
577        buffer.row_indices.extend_from_slice(request.row_indices);
578        buffer.channel_range = request.channel_range;
579
580        let visibility_started = Instant::now();
581        let visibility_shape = match request.visibility {
582            SelectedVisibilityColumn::FloatData => {
583                if !matches!(
584                    buffer.visibility,
585                    Some(SelectedStoredVisibilities::Float32(_))
586                ) {
587                    buffer.visibility = Some(SelectedStoredVisibilities::Float32(Vec::new()));
588                }
589                let Some(SelectedStoredVisibilities::Float32(values)) = buffer.visibility.as_mut()
590                else {
591                    unreachable!("FLOAT_DATA destination installed above")
592                };
593                let probe = AllocationProbe::capture(values);
594                let shape = self
595                    .main_table()
596                    .column_accessor(request.visibility.name())?
597                    .fill_array_cells_2d_channel_range_typed_uncached(
598                        request.row_indices,
599                        request.channel_range.start,
600                        request.channel_range.count,
601                        SelectedArray2DCellsMut::Float32(values),
602                    )?
603                    .ok_or_else(|| {
604                        invalid(format!(
605                            "required MAIN {} cells are undefined",
606                            request.visibility.name()
607                        ))
608                    })?;
609                allocation.observe(probe, values);
610                shape
611            }
612            SelectedVisibilityColumn::Data | SelectedVisibilityColumn::CorrectedData => {
613                if !matches!(
614                    buffer.visibility,
615                    Some(SelectedStoredVisibilities::Complex32(_))
616                ) {
617                    buffer.visibility = Some(SelectedStoredVisibilities::Complex32(Vec::new()));
618                }
619                let Some(SelectedStoredVisibilities::Complex32(values)) =
620                    buffer.visibility.as_mut()
621                else {
622                    unreachable!("complex visibility destination installed above")
623                };
624                let probe = AllocationProbe::capture(values);
625                let shape = self
626                    .main_table()
627                    .column_accessor(request.visibility.name())?
628                    .fill_array_cells_2d_channel_range_typed_uncached(
629                        request.row_indices,
630                        request.channel_range.start,
631                        request.channel_range.count,
632                        SelectedArray2DCellsMut::Complex32(values),
633                    )?
634                    .ok_or_else(|| {
635                        invalid(format!(
636                            "required MAIN {} cells are undefined",
637                            request.visibility.name()
638                        ))
639                    })?;
640                allocation.observe(probe, values);
641                shape
642            }
643        };
644        let visibility_read_nanos = visibility_started.elapsed().as_nanos();
645        require_shape(
646            request.visibility.name(),
647            visibility_shape.row_count,
648            visibility_shape.channel_count,
649            visibility_shape.axis0_count,
650            request,

864        let sample_count = row_count
865            .checked_mul(u64::try_from(request.channel_range.count).map_err(|_| {
866                invalid("selected-observation channel count exceeds diagnostics domain")
867            })?)
868            .and_then(|count| count.checked_mul(u64::try_from(correlation_count).ok()?))
869            .ok_or_else(|| {
870                invalid("selected-observation sample count exceeds diagnostics domain")
871            })?;
872        let retained_current_bytes = buffer
873            .retained_current_bytes()
874            .and_then(|bytes| u64::try_from(bytes).ok())
875            .ok_or_else(|| {
876                invalid("selected-observation current bytes exceed diagnostics domain")
877            })?;
878        let retained_capacity_bytes = buffer
879            .retained_capacity_bytes()
880            .and_then(|bytes| u64::try_from(bytes).ok())
881            .ok_or_else(|| {
882                invalid("selected-observation capacity bytes exceed diagnostics domain")
883            })?;
884        let request_handoff_bytes = row_count
885            .checked_mul(size_of::<usize>() as u64)
886            .ok_or_else(|| {
887                invalid("selected-observation handoff bytes exceed diagnostics domain")
888            })?;
889        let logical_output_bytes = retained_current_bytes
890            .checked_sub(request_handoff_bytes)
891            .ok_or_else(|| invalid("selected-observation logical byte accounting underflowed"))?;
892        let modeled_physical_read_bytes =
893            modeled_selected_observation_physical_read_bytes(self, request, correlation_count)?;
894        let timings = SelectedObservationBufferTimings {
895            total_fill_nanos: fill_started.elapsed().as_nanos(),
896            visibility_read_nanos,
897            flag_read_nanos,
898            weight_read_nanos,
899            scalar_read_nanos,
900            uvw_read_nanos,
901            assembly_nanos,
902        };
903        Ok(SelectedObservationBufferFillReport {
904            block_count: 1,
905            row_count,
906            sample_count,
907            logical_output_bytes,
908            modeled_physical_read_bytes: Some(modeled_physical_read_bytes),
909            read_operation_count: SELECTED_READ_OPERATION_COUNT,
910            request_handoff_bytes,
911            retained_current_bytes,
912            retained_capacity_bytes,
913            allocation: allocation.report(),
914            timings,
915        })
916    }

919fn modeled_selected_observation_physical_read_bytes(
920    ms: &MeasurementSet,
921    request: &SelectedObservationBufferRequest<'_>,
922    correlation_count: usize,
923) -> MsResult<u64> {
924    let table = ms.main_table();
925    let row_count = request.row_indices.len();
926    let channel_start = request.channel_range.start;
927    let channel_count = request.channel_range.count;
928    let mut total = 0u64;
929    let mut add = |column_name: &str,
930                   channelized: bool,
931                   start: usize,
932                   channels: usize,
933                   elements_per_channel_or_row: usize|
934     -> MsResult<()> {
935        let bytes = modeled_main_column_physical_read_bytes(
936            table,
937            column_name,
938            channelized,
939            start,
940            channels,
941            elements_per_channel_or_row,
942            row_count,
943        )?;
944        total = total.checked_add(bytes).ok_or_else(|| {
945            invalid("selected-observation physical read bytes exceed diagnostics domain")
946        })?;
947        Ok(())
948    };
949
950    add(
951        request.visibility.name(),
952        true,
953        channel_start,
954        channel_count,
955        correlation_count,
956    )?;
957    add(
958        "FLAG",
959        true,
960        channel_start,
961        channel_count,
962        correlation_count,
963    )?;
964    match request.weight {
965        SelectedWeightColumn::Weight => add("WEIGHT", false, 0, 1, correlation_count)?,
966        SelectedWeightColumn::WeightSpectrum => add(
967            "WEIGHT_SPECTRUM",
968            true,
969            channel_start,
970            channel_count,
971            correlation_count,
972        )?,
973    }
974    for column_name in SELECTED_SCALAR_COLUMNS {
975        add(column_name, false, 0, 1, 1)?;
976    }
977    add("UVW", false, 0, 1, 3)?;
978    Ok(total)
979}

981fn validate_request(
982    ms: &MeasurementSet,
983    request: &SelectedObservationBufferRequest<'_>,
984) -> MsResult<()> {
985    if request.row_indices.is_empty() {
986        return Err(invalid(
987            "selected-observation buffer requires at least one row",
988        ));
989    }
990    if request.channel_range.count == 0 {
991        return Err(invalid(
992            "selected-observation buffer requires at least one channel",
993        ));
994    }
995    if request.row_indices.iter().any(|row| *row >= ms.row_count()) {
996        return Err(invalid("selected-observation buffer row lies outside MAIN"));
997    }
998    Ok(())
999}

1001fn require_shape(
1002    column: &str,
1003    rows: usize,
1004    channels: usize,
1005    correlations: usize,
1006    request: &SelectedObservationBufferRequest<'_>,
1007    expected_correlations: usize,
1008) -> MsResult<()> {
1009    if rows != request.row_indices.len()
1010        || channels != request.channel_range.count
1011        || correlations != expected_correlations
1012    {
1013        return Err(invalid(format!(
1014            "MAIN {column} shape differs from selected visibility shape"
1015        )));
1016    }
1017    Ok(())
1018}

1020fn validate_buffer_lengths(buffer: &SelectedObservationBuffer) -> MsResult<()> {
1021    let rows = buffer.row_count();
1022    let lengths = [
1023        buffer.row_flag.len(),
1024        buffer.data_description_ids.len(),
1025        buffer.field_ids.len(),
1026        buffer.antenna1.len(),
1027        buffer.antenna2.len(),
1028        buffer.feed1.len(),
1029        buffer.feed2.len(),
1030        buffer.time_mjd_seconds.len(),
1031        buffer.time_centroid_mjd_seconds.len(),
1032        buffer.interval_seconds.len(),
1033        buffer.exposure_seconds.len(),
1034        buffer.scan_numbers.len(),
1035        buffer.state_ids.len(),
1036        buffer.observation_ids.len(),
1037        buffer.array_ids.len(),
1038    ];
1039    if lengths.into_iter().any(|length| length != rows) {
1040        return Err(invalid("selected-observation scalar column lengths differ"));
1041    }
1042    if buffer.uvw_m.len() != rows.saturating_mul(3) {
1043        return Err(invalid("selected-observation UVW length differs"));
1044    }
1045    Ok(())
1046}

1306    fn add_row(ms: &mut MeasurementSet, row_id: i32) {
1307        let fields = ms
1308            .main_table()
1309            .schema()
1310            .unwrap()
1311            .columns()
1312            .iter()
1313            .map(|column| {
1314                let value = match column.name() {
1315                    "DATA" => Value::Array(ArrayValue::Complex32(
1316                        ArrayD::from_shape_vec(vec![2, 4], complex_row(row_id)).unwrap(),
1317                    )),
1318                    "FLOAT_DATA" => Value::Array(ArrayValue::Float32(
1319                        ArrayD::from_shape_vec(vec![2, 4], float_row(row_id)).unwrap(),
1320                    )),
1321                    "FLAG" => Value::Array(ArrayValue::Bool(
1322                        ArrayD::from_shape_vec(vec![2, 4], flag_values(row_id)).unwrap(),
1323                    )),
1324                    "WEIGHT" => Value::Array(ArrayValue::Float32(
1325                        ArrayD::from_shape_vec(vec![2], vec![1.0, 2.0]).unwrap(),
1326                    )),
1327                    "WEIGHT_SPECTRUM" => Value::Array(ArrayValue::Float32(
1328                        ArrayD::from_shape_vec(vec![2, 4], spectrum_weights(row_id)).unwrap(),
1329                    )),
1330                    "UVW" => Value::Array(ArrayValue::Float64(
1331                        ArrayD::from_shape_vec(
1332                            vec![3],
1333                            vec![
1334                                row_id as f64 * 100.0 + 1.0,
1335                                row_id as f64 * 100.0 + 2.0,
1336                                row_id as f64 * 100.0 + 3.0,
1337                            ],
1338                        )
1339                        .unwrap(),
1340                    )),
1341                    "ANTENNA1" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 1)),
1342                    "ANTENNA2" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 2)),
1343                    "DATA_DESC_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 3)),
1344                    "FIELD_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 4)),
1345                    "ARRAY_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 5)),
1346                    "OBSERVATION_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 6)),
1347                    "SCAN_NUMBER" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 7)),
1348                    "STATE_ID" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 8)),
1349                    "FEED1" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 9)),
1350                    "FEED2" => Value::Scalar(ScalarValue::Int32(row_id * 10 + 10)),
1351                    "FLAG_ROW" => Value::Scalar(ScalarValue::Bool(row_id == 1)),
1352                    "TIME" => Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 1.0)),
1353                    "TIME_CENTROID" => {
1354                        Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 2.0))
1355                    }
1356                    "INTERVAL" => {
1357                        Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 10.0))
1358                    }
1359                    "EXPOSURE" => {
1360                        Value::Scalar(ScalarValue::Float64(row_id as f64 * 1000.0 + 20.0))
1361                    }
1362                    _ => default_value(column.name()),
1363                };
1364                RecordField::new(column.name(), value)
1365            })
1366            .collect();
1367        ms.main_table_mut()
1368            .add_row(RecordValue::new(fields))
1369            .unwrap();
1370    }

1372    fn complex_row(row_id: i32) -> Vec<Complex32> {
1373        (0..2)
1374            .flat_map(|corr| {
1375                (0..4).map(move |channel| {
1376                    let value = row_id as f32 * 100.0 + channel as f32 * 10.0 + corr as f32;
1377                    Complex32::new(value, -value)
1378                })
1379            })
1380            .collect()
1381    }
1382
1383    fn float_row(row_id: i32) -> Vec<f32> {
1384        (0..2)
1385            .flat_map(|corr| {
1386                (0..4)
1387                    .map(move |channel| row_id as f32 * 100.0 + channel as f32 * 10.0 + corr as f32)
1388            })
1389            .collect()
1390    }
1391
1392    fn flag_values(row_id: i32) -> Vec<bool> {
1393        (0..2)
1394            .flat_map(|corr| (0..4).map(move |channel| (row_id + channel + corr) % 2 == 0))
1395            .collect()
1396    }
1397
1398    fn spectrum_weights(row_id: i32) -> Vec<f32> {
1399        (0..2)
1400            .flat_map(|corr| {
1401                (0..4).map(move |channel| {
1402                    row_id as f32 * 100.0 + channel as f32 * 10.0 + corr as f32 + 0.5
1403                })
1404            })
1405            .collect()
1406    }
