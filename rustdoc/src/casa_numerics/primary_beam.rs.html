2//! CASA's default primary beams, as `PBMath` gives them by telescope name
3//! and frequency, and their radial power patterns.
4//!
5//! CASA gives the VLA's and the EVLA's patterns a beam squint, which moves
6//! the RR and LL beams apart. It reaches only an image with RR and LL
7//! planes: a Stokes I image is one plane, and CASA applies the unsquinted
8//! pattern to it (measured on CASA 6.7.6: RR and LL agree, whatever the
9//! offset's direction). These patterns are unsquinted.
10
11use crate::{AnnularApertureVoltageTable, EvlaPowerPattern, VlaBand};
12
13/// A primary-beam voltage pattern of CASA's.
14#[derive(Clone, Copy, Debug, PartialEq)]
15pub enum PrimaryBeam {
16    /// CASA `PBMath1DAiry`: an Airy pattern for a dish with a central
17    /// blockage, tabulated out to a maximum radius.
18    Airy {
19        /// Effective dish diameter in metres.
20        dish_diameter_m: f64,
21        /// Blockage diameter in metres.
22        blockage_diameter_m: f64,
23        /// Radius of the tabulated pattern at 1 GHz, in arcmin (scaled as
24        /// 1/frequency). The pattern is zero beyond it, and CASA's 10,000
25        /// table samples span it, so it also sets the lookup quantization.
26        maximum_radius_arcmin_ghz: f64,
27    },
28    /// CASA `PBMath1DEVLA`: the EVLA's polynomial power patterns, each
29    /// frequency taking the fit CASA picks for it ([`EvlaPowerPattern::at`]),
30    /// tabulated to 58′ at 1 GHz.
31    Evla,
32}
33
34impl PrimaryBeam {
35    /// ALMA 12 m antennas as CASA's default voltage patterns give them
36    /// (`PBMath::initByTelescope`): 10.7 m effective diameter, 0.75 m
37    /// blockage, 1.784° at 1 GHz.
38    pub const ALMA: Self = Self::Airy {
39        dish_diameter_m: 10.7,
40        blockage_diameter_m: 0.75,
41        maximum_radius_arcmin_ghz: 1.784 * 60.0,
42    };
43
44    /// ACA 7 m antennas as CASA's default voltage patterns give them
45    /// (`PBMath::initByTelescope`): 6.25 m effective diameter, 0.75 m
46    /// blockage, 3.568° at 1 GHz.
47    pub const ACA: Self = Self::Airy {
48        dish_diameter_m: 6.25,
49        blockage_diameter_m: 0.75,
50        maximum_radius_arcmin_ghz: 3.568 * 60.0,
51    };
52
53    /// VLA antennas in a recognised band ([`VlaBand`]; `PBMath`'s `VLA_4`,
54    /// `VLA_P`, `VLA_L`, `VLA_C`, `VLA_X`, `VLA_U`, `VLA_K` and `VLA_Q`):
55    /// 25 m with a 2.36 m blockage, 0.8564° at 1 GHz.
56    pub const VLA: Self = Self::Airy {
57        dish_diameter_m: 25.0,
58        blockage_diameter_m: 2.36,
59        maximum_radius_arcmin_ghz: 0.8564 * 60.0,
60    };
61
62    /// VLA antennas outside the recognised bands (`VLA_NVSS`): 24.5 m
63    /// without blockage, 0.8564° at 1 GHz.
64    pub const VLA_NVSS: Self = Self::Airy {
65        dish_diameter_m: 24.5,
66        blockage_diameter_m: 0.0,
67        maximum_radius_arcmin_ghz: 0.8564 * 60.0,
68    };
69
70    /// CASA's default voltage pattern for `telescope`, the OBSERVATION
71    /// table's TELESCOPE_NAME, at `reference_frequency_hz`, spectral window
72    /// 0's REF_FREQUENCY (`VPSkyJones`), where it is one of these patterns.
73    /// `PBMath::enumerateCommonPB` matches the name's prefix:
74    /// - `ALMA` (which also catches `ALMASD`, whose pattern is ALMA's), and
75    ///   `ATF`, which maps to ALMA;
76    /// - `ACA`;
77    /// - `VLA`, whose pattern follows the band of the reference frequency
78    ///   ([`VlaBand::of`], `PBMath::whichCommonPBtoUse`): [`Self::VLA`] in
79    ///   the bands CASA recognises and [`Self::VLA_NVSS`] between them;
80    /// - `EVLA`: [`Self::Evla`], which picks its fit per frequency.
81    ///
82    /// `None` for every other telescope, whose patterns casa-rs does not yet
83    /// have.
84    #[must_use]
85    pub fn for_telescope(telescope: &str, reference_frequency_hz: f64) -> Option<Self> {
86        if telescope.starts_with("ALMA") || telescope.starts_with("ATF") {
87            Some(Self::ALMA)
88        } else if telescope.starts_with("ACA") {
89            Some(Self::ACA)
90        } else if telescope.starts_with("VLA") {
91            Some(match VlaBand::of(reference_frequency_hz) {
92                Some(_) => Self::VLA,
93                None => Self::VLA_NVSS,
94            })
95        } else if telescope.starts_with("EVLA") {
96            Some(Self::Evla)
97        } else {
98            None
99        }
100    }
101
102    /// Whether CASA gives this pattern a beam squint: the VLA's and the
103    /// EVLA's (`PBMath1D` with `squint_p` set), which moves the RR and LL
104    /// beams apart for a component.
105    #[must_use]
106    pub fn squints(self) -> bool {
107        self == Self::VLA || self == Self::VLA_NVSS || self == Self::Evla
108    }
109
110    /// The pattern's lookup: an Airy pattern's table, shared by every
111    /// frequency it is evaluated at, or the EVLA's fits, which hold none.
112    #[must_use]
113    pub fn pattern(self) -> BeamPattern {
114        BeamPattern(match self {
115            Self::Airy {
116                dish_diameter_m,
117                blockage_diameter_m,
118                maximum_radius_arcmin_ghz,
119            } => Lookup::Airy(AnnularApertureVoltageTable::new(
120                dish_diameter_m,
121                blockage_diameter_m,
122                maximum_radius_arcmin_ghz,
123            )),
124            Self::Evla => Lookup::Evla,
125        })
126    }
127}

129/// A primary beam's power pattern, as it is looked up by radius.
130#[derive(Clone)]
131pub struct BeamPattern(Lookup);
132
133#[derive(Clone)]
134enum Lookup {
135    /// An Airy pattern's voltage table.
136    Airy(AnnularApertureVoltageTable),
137    /// The EVLA's fits, picked per frequency.
138    Evla,
139}
140
141impl BeamPattern {
142    /// The most bytes a pattern holds: an Airy pattern's table.
143    pub const BYTES: u64 = AnnularApertureVoltageTable::BYTES as u64;
144
145    /// The pattern at `frequency_hz`: the Airy table, or the EVLA's fit of
146    /// that frequency.
147    #[must_use]
148    pub fn at_frequency(&self, frequency_hz: f64) -> FrequencyPattern<'_> {
149        FrequencyPattern(match &self.0 {
150            Lookup::Airy(table) => FrequencyLookup::Airy(table),
151            Lookup::Evla => FrequencyLookup::Evla(EvlaPowerPattern::at(frequency_hz)),
152        })
153    }
154
155    /// The power pattern at `separation_rad` from the pointing, at
156    /// `frequency_hz`, as `PBMath1D::apply` evaluates it for a component:
157    /// the separation in arcmin times the frequency in GHz, in `Double`,
158    /// looked up in the table (the EVLA's fit of that frequency, as
159    /// `PBMath1DEVLA` takes it per channel).
160    #[must_use]
161    pub fn power_at(&self, separation_rad: f64, frequency_hz: f64) -> f32 {
162        let radius = separation_rad.to_degrees() * 60.0 * frequency_hz / 1.0e9;
163        self.at_frequency(frequency_hz).power(radius)
164    }
165
166    /// The power pattern `offset_rad` along an image's x axis from its
167    /// centre, at `frequency_hz`, with `PBMath1D::apply`'s arithmetic on an
168    /// image (the offset in degrees kept as `Float`, the radius times 60 ·
169    /// ν/GHz kept as `Float`), as `SDGrid::findPBAsConvFunction` applies the
170    /// beam to a one-row image.
171    #[must_use]
172    pub fn power_on_axis(&self, offset_rad: f64, frequency_hz: f64) -> f32 {
173        let ghz = frequency_hz / 1.0e9;
174        let l_deg = offset_rad.to_degrees() as f32;
175        let radius_deg = (l_deg * l_deg).sqrt();
176        let radius = (f64::from(radius_deg) * 60.0 * ghz) as f32;
177        self.at_frequency(frequency_hz).power(f64::from(radius))
178    }
179
180    /// `PBMath1D::support`: the pattern's diameter in pixels of
181    /// `increment_x_rad` at `frequency_hz`, rounded down.
182    #[must_use]
183    pub fn support_pixels(&self, increment_x_rad: f64, frequency_hz: f64) -> usize {
184        let maximum_arcmin_ghz = match &self.0 {
185            Lookup::Airy(table) => table.maximum_radius(),
186            Lookup::Evla => EvlaPowerPattern::MAXIMUM_RADIUS_ARCMIN_GHZ,
187        };
188        let radius_rad = (maximum_arcmin_ghz / 60.0).to_radians();
189        (radius_rad / increment_x_rad.abs() * 2.0 * 1.0e9 / frequency_hz).floor() as usize
190    }
191}
192
193/// A pattern at one frequency ([`BeamPattern::at_frequency`]).
194pub struct FrequencyPattern<'a>(FrequencyLookup<'a>);
195
196enum FrequencyLookup<'a> {
197    Airy(&'a AnnularApertureVoltageTable),
198    Evla(EvlaPowerPattern),
199}
200
201impl FrequencyPattern<'_> {
202    /// The power at `radius_arcmin_ghz`, the radius in arcmin times the
203    /// frequency in GHz, and zero beyond the table (`PBMath1D`'s lookup;
204    /// `BeamSkyJones::apply` squares the voltage pattern).
205    #[must_use]
206    pub fn power(&self, radius_arcmin_ghz: f64) -> f32 {
207        match &self.0 {
208            FrequencyLookup::Airy(table) => {
209                let voltage = table.evaluate(radius_arcmin_ghz);
210                voltage * voltage
211            }
212            FrequencyLookup::Evla(fit) => fit.power(radius_arcmin_ghz),
213        }
214    }
215}

217#[cfg(test)]
218mod tests {
219    use super::*;
220
221    #[test]
222    fn telescopes_match_casas_common_beams_by_prefix_and_band() {
223        let beam = |telescope: &str, ghz: f64| PrimaryBeam::for_telescope(telescope, ghz * 1.0e9);
224        assert_eq!(beam("ALMA", 230.0), Some(PrimaryBeam::ALMA));
225        assert_eq!(beam("ALMASD", 100.0), Some(PrimaryBeam::ALMA));
226        assert_eq!(beam("ATF", 230.0), Some(PrimaryBeam::ALMA));
227        assert_eq!(beam("ACA", 230.0), Some(PrimaryBeam::ACA));
228        assert_eq!(beam("ACA-7m", 230.0), Some(PrimaryBeam::ACA));
229        // The VLA's pattern follows spectral window 0's band: between CASA's
230        // bands (3 GHz) it is the NVSS pattern.
231        assert_eq!(beam("VLA", 1.4), Some(PrimaryBeam::VLA));
232        assert_eq!(beam("VLA", 45.0), Some(PrimaryBeam::VLA));
233        assert_eq!(beam("VLA", 3.0), Some(PrimaryBeam::VLA_NVSS));
234        assert_eq!(beam("EVLA", 1.4), Some(PrimaryBeam::Evla));
235        assert_eq!(beam("EVLA", 18.5), Some(PrimaryBeam::Evla));
236        for other in ["alma", "Unknown", "", "SMA", "AL", "NGVLA"] {
237            assert_eq!(beam(other, 1.4), None, "{other}");
238        }
239    }
