2//! Keep fringe parameters continuous when the preferred reference drops out.
3
4use std::collections::BTreeSet;
5
6use casa_core::Complex32;
7use casa_ms::MeasurementSet;
8
9use super::{SolutionRow, error};
10use crate::solve::GainSolveError;
11
12pub(super) fn connect_intervals(
13    ms: &MeasurementSet,
14    rows: &mut [SolutionRow],
15    antenna_count: usize,
16    preferences: impl Iterator<Item = i32>,
17    strict: bool,
18) -> Result<(), GainSolveError> {
19    if rows.is_empty() {
20        return Ok(());
21    }
22    let mut candidates = preferences.collect::<Vec<_>>();
23    let primary = candidates[0];
24    let mut previous_reference = primary;
25    if strict {
26        candidates.truncate(1);
27    } else {
28        let antenna = ms.antenna().map_err(error)?;
29        let anchor = antenna
30            .position(candidates[candidates.len() - 1] as usize)
31            .map_err(error)?;
32        let mut nearby = (0..antenna_count as i32)
33            .filter(|id| !candidates.contains(id))
34            .map(|id| {
35                let position = antenna.position(id as usize).map_err(error)?;
36                let distance = position
37                    .into_iter()
38                    .zip(anchor)
39                    .map(|(a, b)| (a - b).powi(2))
40                    .sum::<f64>();
41                Ok((id, distance))
42            })
43            .collect::<Result<Vec<_>, GainSolveError>>()?;
44        nearby.sort_by(|a, b| a.1.total_cmp(&b.1));
45        candidates.extend(nearby.into_iter().map(|(id, _)| id));
46    }
47
48    let mut previous = None::<(i32, Vec<Reference>)>;
49    let mut intervals = rows.chunks_mut(antenna_count).collect::<Vec<_>>();
50    intervals.sort_by(|left, right| {
51        left[0]
52            .spw_id
53            .cmp(&right[0].spw_id)
54            .then_with(|| left[0].time_seconds.total_cmp(&right[0].time_seconds))
55    });
56    let second_hand_windows = intervals
57        .iter()
58        .flat_map(|rows| rows.iter())
59        .filter(|row| has_hand(row, 1))
60        .map(|row| row.spw_id)
61        .collect::<BTreeSet<_>>();
62    for rows in intervals {
63        let spw = rows[0].spw_id;
64        let hands = usize::from(second_hand_windows.contains(&spw)) + 1;
65        let prior = previous
66            .as_ref()
67            .filter(|(window, _)| *window == spw)
68            .map(|(_, previous)| previous.as_slice());
69        let chosen = std::iter::once(primary)
70            .chain(std::iter::once(previous_reference))
71            .chain(candidates.iter().copied())
72            .find_map(|antenna| {
73                // The fringe solver writes every ANTENNA row in ID order.
74                let index = usize::try_from(antenna).ok()?;
75                let current = rows.get(index)?;
76                let valid = (0..hands).all(|hand| {
77                    has_hand(current, hand)
78                        && prior.is_none_or(|previous| previous[index].hands[hand])
79                });
80                valid.then_some((antenna, index))
81            });
82        if let Some((antenna, index)) = chosen {
83            let offset = std::array::from_fn::<_, 8, _>(|parameter| {
84                rows[index].gains[parameter]
85                    - prior.map_or(Complex32::default(), |prior| prior[index].values[parameter])
86            });
87            for row in rows.iter_mut() {
88                row.refant_id = antenna;
89                for (value, delta) in row.gains.iter_mut().zip(offset) {
90                    *value -= delta;
91                }
92            }
93            previous_reference = antenna;
94            previous = Some((
95                spw,
96                rows.iter()
97                    .map(|row| Reference {
98                        hands: [has_hand(row, 0), has_hand(row, 1)],
99                        values: std::array::from_fn(|i| row.gains[i]),
100                    })
101                    .collect(),
102            ));
103        } else {
104            for row in rows {
105                row.flags.fill(true);
106            }
107        }
108    }
109    Ok(())
110}
111
112fn has_hand(row: &SolutionRow, hand: usize) -> bool {
113    row.flags[hand * 4..hand * 4 + 4].iter().any(|&flag| !flag)
114}
115
116struct Reference {
117    hands: [bool; 2],
118    values: [Complex32; 8],
119}
