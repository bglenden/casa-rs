10/// casacore's table write lock, held for one in-place change of a table on
11/// disk.
12///
13/// An in-place writer takes this lock before it changes a table and holds it
14/// until the change is complete or abandoned, as casacore holds a table's
15/// write lock while it writes. The lock is casacore's own: the `fcntl` write
16/// lock on byte 0 of the table's `table.lock`. It therefore excludes casacore
17/// processes, other casa-rs processes, and every other casa-rs handle on the
18/// table in this process.
19///
20/// A lock another process holds, read or write, is waited for as casacore
21/// waits (see [`acquire`](Self::acquire)): the waiter adds its process id to
22/// the request list in `table.lock`, so a casacore process holding the table
23/// open with `AutoLocking` releases its lock at its next inspection.
24/// casacore throttles inspections (`LockFile::inspect` looks at the request
25/// list only after 25 table accesses and the inspection interval), so a
26/// lightly used holder may keep its lock until it closes the table. A
27/// casa-rs process does not yet release a lock on request, so a waiter waits
28/// for it to finish ([#694](https://github.com/bglenden/casa-rs/issues/694)).
29/// A write lock
30/// held by another handle in this process is never waited for, because that
31/// handle may belong to the waiting thread: it is refused at once.
32///
33/// Releasing the lock after [`record_write`](Self::record_write) records the
34/// change in the lock file's sync data (`TableSyncData`), as casacore does
35/// when it releases a write lock, so a process holding the table open
36/// re-reads it when it next takes a lock. The sync data describes the table
37/// as persisted when the lock is released. Dropping the guard releases it
38/// the same way: rows written before an interruption stay written, as after
39/// an interrupted casacore write, and are announced.
40///
41/// The guard works on the table directory, not on a [`Table`](crate::Table)
42/// handle, so it can be held across calls that each borrow the handle. It is
43/// meant for handles opened without locking
44/// ([`Table::open`](crate::Table::open)). A handle opened with
45/// [`Table::open_with_lock`](crate::Table::open_with_lock) in auto-locking
46/// mode cannot write while another
47/// handle holds this lock: its temporary write lock is refused rather than
48/// waited on, because the holder may be the same thread.
49///
50/// On a file system without lock support (`fcntl` refused with `ENOLCK`, or
51/// `ENOTSUP` as on macOS SMB mounts) the lock counts as acquired, with one
52/// warning per lock file, as casacore does for `ENOLCK`: other handles in this
53/// process are still refused, other processes are not excluded. On platforms
54/// without `fcntl` locking the guard holds nothing.
55///
56/// # C++ equivalent
57///
58/// `Table::lock(FileLocker::Write, nattempts)` and `Table::unlock()` on a
59/// table opened with `TableLock::UserLocking`.
60pub struct TableWriteLock {
61    path: PathBuf,
62    #[cfg(unix)]
63    lock_file: Option<LockFile>,
64    /// Whether a write was recorded; releasing then publishes it.
65    #[cfg_attr(not(unix), allow(dead_code))]
66    written: bool,
67}
68
69impl TableWriteLock {
70    /// Take the write lock on the table at `table_dir`.
71    ///
72    /// `nattempts` has the meaning of [`Table::lock`](crate::Table::lock): 1 tries once without
73    /// waiting, more retries once a second, and 0 waits indefinitely for
74    /// another process, as casacore's default `AutoLocking` does. While it
75    /// waits, this process's id is in the lock file's request list, and the
76    /// wait is logged when it starts, every ten seconds and when it ends. An
77    /// indefinite wait blocks in the kernel, which refuses it when it would
78    /// deadlock with another waiting process. A holder in this process is
79    /// never waited for.
80    ///
81    /// A writer that had to wait checks the lock file's sync data once it
82    /// holds the lock. When another process published a write to the table
83    /// while it waited, the rows and metadata the caller read beforehand are
84    /// stale; casacore would re-read them, which a handle opened without
85    /// locking cannot, so the lock is released and the request refused.
86    ///
87    /// # Errors
88    ///
89    /// [`TableError::LockFailed`] when another handle in this process holds
90    /// the write lock, when another process still holds a conflicting lock
91    /// after `nattempts` (never with 0), when waiting would deadlock, or when
92    /// another process wrote the table while this writer waited; and
93    /// [`TableError::LockIo`] when
94    /// `table.lock` cannot be opened or `fcntl` fails for a reason other than
95    /// a held lock or missing lock support.
96    pub fn acquire(table_dir: impl AsRef<Path>, nattempts: u32) -> Result<Self, TableError> {
97        let path = table_dir.as_ref().to_path_buf();
98        #[cfg(unix)]
99        {
100            let lock_io = |error: std::io::Error| TableError::LockIo {
101                path: path.display().to_string(),
102                message: error.to_string(),
103            };
104            let lock_failed = |message: String| TableError::LockFailed {
105                path: path.display().to_string(),
106                message,
107            };
108            let mut lock_file = LockFile::create_or_open(
109                &path,
110                false,
111                LockOptions::default().inspection_interval,
112                false,
113            )
114            .map_err(lock_io)?;
115            let mut outcome = lock_file.acquire(LockType::Write, 1).map_err(lock_io)?;
116            if outcome == LockOutcome::HeldByAnotherProcess && nattempts != 1 {
117                let published = published_modify_counter(&lock_file);
118                outcome = lock_file
119                    .acquire(LockType::Write, nattempts)
120                    .map_err(lock_io)?;
121                if outcome.is_acquired() && published_modify_counter(&lock_file) != published {
122                    return Err(lock_failed(
123                        "another process wrote the table while this writer waited for its \
124                         write lock; reopen the table and retry"
125                            .into(),
126                    ));
127                }
128            }
129            if !outcome.is_acquired() {
130                return Err(lock_failed(outcome.refusal(LockType::Write, nattempts)));
131            }
132            Ok(Self {
133                path,
134                lock_file: Some(lock_file),
135                written: false,
136            })
137        }
138        #[cfg(not(unix))]
139        {
140            let _ = nattempts;
141            Ok(Self {
142                path,
143                written: false,
144            })
145        }
146    }
147
148    /// The locked table directory.
149    #[must_use]
150    pub fn path(&self) -> &Path {
151        &self.path
152    }
153
154    /// The modify counter the table's sync data publishes, read now; `None`
155    /// when the table has no sync data.
156    ///
157    /// While this lock is held no other process can publish a write, so a
158    /// writer that releases the lock and takes it again later compares two
159    /// readings to learn whether another process wrote the table in between.
160    ///
161    /// # Errors
162    ///
163    /// [`TableError::LockIo`] when the sync data cannot be read.
164    pub fn modify_counter(&self) -> Result<Option<u32>, TableError> {
165        #[cfg(unix)]
166        {
167            let Some(lock_file) = &self.lock_file else {
168                return Ok(None);
169            };
170            lock_file
171                .read_sync_data()
172                .map(|sync| sync.map(|sync| sync.modify_counter))
173                .map_err(|error| TableError::LockIo {
174                    path: self.path.display().to_string(),
175                    message: error.to_string(),
176                })
177        }
178        #[cfg(not(unix))]
179        {
180            Ok(None)
181        }
182    }
183
184    /// Record that the table is, or may be, changed on disk under this lock.
185    ///
186    /// Call it before writing, so that a write interrupted part way is still
187    /// announced. Releasing the lock then publishes the change in the sync
188    /// data, describing the table as it is persisted when the lock is
189    /// released (see [`release`](Self::release)); a table this writer never
190    /// reached is left unrecorded and publishes nothing.
191    pub fn record_write(&mut self) {
192        self.written = true;
193    }
194
195    /// Release the lock, publishing any recorded write in the sync data.
196    ///
197    /// The published row, column and data-manager counts are read from the
198    /// table's `table.dat` as persisted, not taken from an in-memory handle:
199    /// casacore takes a reopened table's row count from the sync data and
200    /// asserts that its per-data-manager counters match the table's data
201    /// managers.
202    ///
203    /// # Errors
204    ///
205    /// [`TableError::LockIo`] when the persisted table cannot be read, the
206    /// sync data cannot be written or the lock cannot be released.
207    pub fn release(mut self) -> Result<(), TableError> {
208        self.release_now()
209    }
210
211    fn release_now(&mut self) -> Result<(), TableError> {
212        #[cfg(unix)]
213        {
214            let Some(mut lock_file) = self.lock_file.take() else {
215                return Ok(());
216            };
217            let lock_io = |error: std::io::Error| TableError::LockIo {
218                path: self.path.display().to_string(),
219                message: error.to_string(),
220            };
221            let published = if self.written {
222                publish_persisted_write(&lock_file, &self.path).map(|_| ())
223            } else {
224                Ok(())
225            };
226            let released = lock_file.release().map(|_| ());
227            published.and(released).map_err(lock_io)
228        }
229        #[cfg(not(unix))]
230        {
231            Ok(())
232        }
233    }
234}

236/// The shape of a table as persisted in its `table.dat`, which the sync data
237/// published for it must describe.
238#[cfg(unix)]
239struct PersistedShape {
240    rows: u64,
241    columns: u32,
242    data_managers: usize,
243}
244
245#[cfg(unix)]
246impl PersistedShape {
247    fn read(table_dir: &Path) -> std::io::Result<Self> {
248        let contents = crate::storage::table_control::read_table_dat(&table_dir.join("table.dat"))
249            .map_err(|error| {
250                std::io::Error::other(format!(
251                    "read the persisted table {} to publish its write: {error}",
252                    table_dir.display()
253                ))
254            })?;
255        Ok(Self {
256            rows: contents.nrrow,
257            columns: contents.table_desc.columns.len() as u32,
258            data_managers: contents.column_set.data_managers.len(),
259        })
260    }
261}

263/// Publish a write to the table at `table_dir` in its lock file's sync data,
264/// describing the table as persisted: its row and column counts and one
265/// change counter per data manager, as casacore's `PlainTable::putFile`
266/// publishes them. The counters continue from the sync data the lock file
267/// holds, and every data manager counts as changed. The caller holds the
268/// table's write lock. Returns the published sync data.
269#[cfg(unix)]
270pub(crate) fn publish_persisted_write(
271    lock_file: &LockFile,
272    table_dir: &Path,
273) -> std::io::Result<SyncData> {
274    let shape = PersistedShape::read(table_dir)?;
275    let mut sync = lock_file.read_sync_data()?.unwrap_or_else(SyncData::new);
276    sync.record_write(
277        shape.rows,
278        shape.columns,
279        true,
280        &vec![true; shape.data_managers],
281    );
282    lock_file.write_sync_data(&sync)?;
283    Ok(sync)
284}

286/// The modify counter the lock file's sync data publishes, `None` when it has
287/// none. A read that fails means a writer was publishing at that moment, and
288/// counts as `None`, so a change is still seen.
289#[cfg(unix)]
290fn published_modify_counter(lock_file: &LockFile) -> Option<u32> {
291    lock_file
292        .read_sync_data()
293        .ok()
294        .flatten()
295        .map(|sync| sync.modify_counter)
296}

298impl Drop for TableWriteLock {
299    fn drop(&mut self) {
300        let _ = self.release_now();
301    }
302}
