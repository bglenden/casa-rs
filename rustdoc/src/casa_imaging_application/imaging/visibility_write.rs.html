2//! Visibilities written back in the final pass: the final model into
3//! `MODEL_DATA` and the continuum-subtracted data into `CORRECTED_DATA`,
4//! through the update session of each of the batch route's stores.
5
6use casa_imaging_model::{ContinuumFitRule, SequentialContinuumTransform};
7use casa_imaging_runtime::pass::SourceError;
8
9use super::observation_source::BatchRoute;
10pub(crate) use super::observation_source::BatchWriter as VisibilityWriter;
11
12/// Where the visibilities go: the update session of the batch route the
13/// run reads.
14#[derive(Clone, Copy)]
15pub(crate) struct VisibilityWriteTarget<'a>(pub(crate) &'a BatchRoute);
16
17impl<'a> VisibilityWriteTarget<'a> {
18    /// Heap bytes the write holds at most, admitted with the pass that
19    /// writes ([`BatchRoute::write_bytes`]).
20    pub(crate) fn retained_bytes(&self) -> Result<u64, SourceError> {
21        self.0.write_bytes()
22    }
23
24    /// Whether CASA follows the final major cycle with a separate parent
25    /// model prediction.
26    pub(crate) fn predicts_final_model(&self) -> bool {
27        self.0
28            .write_back()
29            .is_some_and(|write| write.predicts_final_model())
30    }
31
32    /// Whether a tuned pass must persist continuum-corrected values.
33    pub(crate) fn writes_corrected(&self) -> bool {
34        self.0
35            .write_back()
36            .is_some_and(|write| write.writes_corrected())
37    }
38
39    /// Begin a model-only traversal of the parent selection.
40    pub(crate) fn begin_prediction(
41        &self,
42        transform: Option<&'a SequentialContinuumTransform>,
43    ) -> Result<VisibilityWriter<'a>, SourceError> {
44        VisibilityWriter::begin_targets(self.0, transform, true, false, true)
45    }
46
47    /// Begin a tuned, corrected-data-only traversal.
48    pub(crate) fn begin_corrected(
49        &self,
50        transform: Option<&'a SequentialContinuumTransform>,
51    ) -> Result<VisibilityWriter<'a>, SourceError> {
52        VisibilityWriter::begin_targets(self.0, transform, false, true, false)
53    }
54
55    /// Begin the write; `transform` is the continuum transform whose output
56    /// `CORRECTED_DATA` holds.
57    pub(crate) fn begin(
58        &self,
59        transform: Option<&'a SequentialContinuumTransform>,
60    ) -> Result<VisibilityWriter<'a>, SourceError> {
61        VisibilityWriter::begin(self.0, transform)
62    }
63}
64
65/// Whether a channel of a row under the continuum `rule` reaches the line
66/// output, and so is written to either column: every channel without a
67/// continuum transform; fit-only channels keep their contents.
68pub(super) fn writes_channel(rule: Option<&ContinuumFitRule>, channel: u32) -> bool {
69    rule.is_none_or(|rule| {
70        rule.channel_use(channel)
71            .is_some_and(|role| role.contributes_to_output())
72    })
73}
