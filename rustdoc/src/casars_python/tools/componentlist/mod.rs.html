2//! `casars.tools.componentlist`: CASA's `componentlist` tool (`cl`) over
3//! [`casa_components::ComponentList`].
4//!
5//! The class declares every method of CASA's installed binding with CASA's
6//! names, parameter order and effective defaults ([`surface`]). The methods
7//! CASA's simulation tasks call are implemented (`open`, `addcomponent`,
8//! `rename`, `close`, `done`, `length`, `torecord`, `getspectrum`,
9//! `getrefdir`), with the accessors `getcomponent`, `getfluxvalue`,
10//! `getfluxunit`, `getlabel`, `shapetype` and `spectrumtype`, and CASA's
11//! `iscomponentlist` placeholder; the rest raise `NotImplementedError`
12//! naming #1053.
13//!
14//! CASA's outcomes are kept ([`ops`] for the table, `add` for
15//! `addcomponent`): what CASA raises is a `RuntimeError` with CASA's
16//! message, and what it logs and carries on from (a shape `setshape`
17//! refuses, `torecord` of an empty list, `close`) is logged on the Python
18//! logger `casars.tools.componentlist`, CASA's SEVERE at ERROR and its WARN
19//! at WARNING. Records are casacore's (`casa_components` `to_record`), as
20//! `dict`s with NumPy arrays. As in CASA, `addcomponent` stores the flux as
21//! Stokes whatever `polarization` says.
22//!
23//! Divergences, recorded in `docs/CASA (C++) bugs.md`: `addcomponent` keeps
24//! the flux's double precision (`add`).

63    fn refdir_placeholder(
64        &self,
65        py: Python<'_>,
66        which: &Bound<'_, PyAny>,
67        unit: Option<&Bound<'_, PyAny>>,
68        precision: Option<&Bound<'_, PyAny>>,
69        method: &'static str,
70    ) -> PyResult<Py<PyAny>> {
71        let index = which.extract::<i64>();
72        let mut system_error = which.is_none()
73            || index
74                .as_ref()
75                .err()
76                .is_some_and(|e| e.is_instance_of::<PyOverflowError>(py));
77        if index.is_err() && !system_error {
78            return Err(PyTypeError::new_err("argument _which must be an integer"));
79        }
80        if unit.is_some_and(|value| !value.is_instance_of::<PyString>()) {
81            return Err(PyTypeError::new_err("argument _unit must be a string"));
82        }
83        if let Some(precision) = precision {
84            let extracted = precision.extract::<i64>();
85            let precision_system_error = precision.is_none()
86                || extracted
87                    .as_ref()
88                    .err()
89                    .is_some_and(|e| e.is_instance_of::<PyOverflowError>(py));
90            if extracted.is_err() && !precision_system_error {
91                return Err(PyTypeError::new_err(
92                    "argument _precision must be an integer",
93                ));
94            }
95            system_error |= precision_system_error;
96        }
97        let result = self.call(py, |_, lines| {
98            lines.push(LogLine::warning(format!("{method} not implemented yet")));
99            Ok(CasaValue::Text(String::new()))
100        })?;
101        if system_error {
102            return Err(PySystemError::new_err(format!(
103                "<built-in function componentlist_{method}> returned a result with an exception set"
104            )));
105        }
106        Ok(result)
107    }

138    /// Runs `work` on the state with the GIL released, logs what it reports
139    /// and returns its value as Python.
140    fn call(
141        &self,
142        py: Python<'_>,
143        work: impl FnOnce(&mut State, &mut Vec<LogLine>) -> ConvertResult<CasaValue> + Send,
144    ) -> PyResult<Py<PyAny>> {
145        let mut lines = Vec::new();
146        let result = py.detach(|| {
147            casa_types::quanta::registry::global_registry()
148                .with_locked_units(|| work(&mut self.lock(), &mut lines))
149        });
150        emit(py, LOGGER, &lines)?;
151        result?.into_python(py)
152    }
153
154    /// The state lock, taken without the GIL (inside `Python::detach`).
155    fn lock(&self) -> MutexGuard<'_, State> {
156        self.state.lock().unwrap_or_else(PoisonError::into_inner)
157    }
158}

160impl Drop for PyComponentList {
161    /// CASA's destructor writes a list attached for update. The write runs
162    /// with the GIL released, as every method's does (ADR-0019), since it
163    /// may wait for a lock another process holds; a failure is a
164    /// `RuntimeWarning`, raised once the GIL is held again, as nothing can
165    /// raise here.
166    fn drop(&mut self) {
167        let state = self.state.get_mut().unwrap_or_else(PoisonError::into_inner);
168        Python::attach(|py| {
169            if let Err(message) = py.detach(|| {
170                casa_types::quanta::registry::global_registry()
171                    .with_locked_units(|| state.release())
172            }) {
173                let category = py.get_type::<PyRuntimeWarning>();
174                let message = CString::new(message).unwrap_or_default();
175                if let Err(error) = PyErr::warn(py, category.as_any(), &message, 1) {
176                    error.write_unraisable(py, None);
177                }
178            }
179        });
180    }
181}

311    /// Convert one or all components' flux values and errors to a Jy unit.
312    #[pyo3(signature = (which, unit = None), text_signature = "($self, which, unit='Jy')")]
313    fn convertfluxunit(
314        &self,
315        py: Python<'_>,
316        which: &Bound<'_, PyAny>,
317        #[pyo3(from_py_with = supplied)] unit: Option<Bound<'_, PyAny>>,
318    ) -> PyResult<Py<PyAny>> {
319        let (which, pending) = match which.extract::<i64>() {
320            Ok(index) => (i64::from(index as i32), false),
321            Err(error) if which.is_none() || error.is_instance_of::<PyOverflowError>(py) => {
322                (-1, true)
323            }
324            Err(_) => return Err(PyTypeError::new_err("argument _which must be an integer")),
325        };
326        let unit: String = match unit {
327            None => "Jy".to_string(),
328            Some(value) if value.is_instance_of::<PyString>() => value.extract()?,
329            Some(value) if value.is_instance_of::<PyBytes>() => {
330                let bytes = value.cast::<PyBytes>()?.as_bytes();
331                let end = bytes
332                    .iter()
333                    .position(|&byte| byte == 0)
334                    .unwrap_or(bytes.len());
335                PyBytes::new(py, &bytes[..end])
336                    .call_method1("decode", ("utf-8",))?
337                    .extract()?
338            }
339            Some(_) => return Err(PyTypeError::new_err("argument _unit must be a string")),
340        };
341        let unit = unit.split('\0').next().unwrap_or_default().to_string();
342        let result = self.call(py, move |state, lines| {
343            state
344                .convertfluxunit(which, &unit, lines)
345                .map(CasaValue::Bool)
346        })?;
347        if pending {
348            return Err(PySystemError::new_err(
349                "<built-in function componentlist_convertfluxunit> returned a result with an exception set",
350            ));
351        }
352        Ok(result)
353    }

460    /// The real parts of component `which`'s four flux errors.
461    #[pyo3(signature = (which), text_signature = "($self, which)")]
462    fn getfluxerror(&self, py: Python<'_>, which: &Bound<'_, PyAny>) -> PyResult<Py<PyAny>> {
463        let which = match which.extract::<i64>() {
464            Ok(index) => i64::from(index as i32),
465            Err(error) if which.is_none() || error.is_instance_of::<PyOverflowError>(py) => -1,
466            Err(_) => return Err(PyTypeError::new_err("argument _which must be an integer")),
467        };
468        self.call(py, move |state, lines| match state.checked(which) {
469            Ok(component) => Ok(CasaValue::floats(
470                component
471                    .flux()
472                    .errors()
473                    .iter()
474                    .map(|value| value.re)
475                    .collect(),
476            )),
477            Err(_) => {
478                let message = format!(
479                    "Exception: Index out of range. The component number must be between 0 and {}.",
480                    state.length() - 1
481                );
482                lines.push(LogLine::error(format!("Exception Reported: {message}")));
483                Err(crate::convert::ConvertError::runtime(message))
484            }
485        })
486    }

868    /// Change a component's reference frequency and its record unit.
869    #[pyo3(signature = (which, value, unit = None, log = None), text_signature = "($self, which, value, unit='GHz', log=True)")]
870    fn setfreq(
871        &self,
872        py: Python<'_>,
873        which: &Bound<'_, PyAny>,
874        value: &Bound<'_, PyAny>,
875        #[pyo3(from_py_with = supplied)] unit: Option<Bound<'_, PyAny>>,
876        #[pyo3(from_py_with = supplied)] log: Option<Bound<'_, PyAny>>,
877    ) -> PyResult<Py<PyAny>> {
878        let which = match which.extract::<i64>() {
879            Ok(index) => i64::from(index as i32),
880            Err(error) if which.is_none() || error.is_instance_of::<PyOverflowError>(py) => -1,
881            Err(_) => return Err(PyTypeError::new_err("argument _which must be an integer")),
882        };
883        // CASA's double typemap passes -1 through to the operation when
884        // conversion fails, then reports a SystemError if it succeeded.
885        let converted = value.extract::<f64>();
886        let invalid_value = converted.is_err();
887        let value = converted.unwrap_or(-1.0);
888        let unit: String = match unit {
889            None => "GHz".to_string(),
890            Some(value) if value.is_instance_of::<PyString>() => value.extract()?,
891            Some(value) if value.is_instance_of::<PyBytes>() => {
892                let bytes = value.cast::<PyBytes>()?.as_bytes();
893                let end = bytes
894                    .iter()
895                    .position(|&byte| byte == 0)
896                    .unwrap_or(bytes.len());
897                PyBytes::new(py, &bytes[..end])
898                    .call_method1("decode", ("utf-8",))?
899                    .extract()?
900            }
901            Some(_) => return Err(PyTypeError::new_err("argument _unit must be a string")),
902        };
903        if log.is_some_and(|value| !value.is_instance_of::<PyBool>()) {
904            return Err(PyTypeError::new_err(
905                "in method 'componentlist_setfreq', argument 5 of type 'bool'",
906            ));
907        }
908        let unit = unit.split('\0').next().unwrap_or_default().to_string();
909        let result = self.call(py, move |state, _| {
910            state.setfreq(which, value, &unit).map(CasaValue::Bool)

912        if invalid_value {

914                "<built-in function componentlist_setfreq> returned a result with an exception set",
915            ));
916        }
917        Ok(result)
