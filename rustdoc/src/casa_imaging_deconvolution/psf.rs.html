2//! What the minor cycle needs to know about a PSF, measured once per PSF.
3
4use crate::Error;
5use crate::beam::{
6    DEFAULT_PSF_FIT_CUTOFF, RestoringBeam, fit_restoring_beam,
7    fitted_psf_sidelobe_fraction_with_beam,
8};
9use crate::plane::{PlaneShape, psf_peak};
10
11/// The measurements of one PSF plane that every minor cycle reuses: they
12/// change only when the PSF does, so a run measures them once.
13#[derive(Clone, Copy, Debug, PartialEq)]
14pub struct PsfSummary {
15    peak: usize,
16    beam: RestoringBeam,
17    sidelobe: f64,
18    clark: ClarkPatch,
19}
20
21/// Clark's PSF patch (`SDAlgorithmClarkClean2`): the square the active
22/// pixels are updated within, and the largest PSF value outside it.
23#[derive(Clone, Copy, Debug, PartialEq)]
24pub struct ClarkPatch {
25    /// Half the patch width per axis; an even patch ends one pixel early on
26    /// its positive side, as in casacore.
27    pub radius: [usize; 2],
28    /// The patch width per axis.
29    pub size: [usize; 2],
30    /// The largest PSF magnitude outside the patch, by casacore's
31    /// asymmetric box about the plane centre; 0 when the patch is the plane.
32    pub exterior_sidelobe: f64,
33}
34
35impl PsfSummary {
36    /// Measure a PSF plane normalised to peak 1.
37    ///
38    /// The peak is found as `MatrixCleaner::findPSFMaxAbs` finds it; the
39    /// beam is CASA's main-lobe fit in pixel units with `psfcutoff` 0.35;
40    /// the sidelobe is CASA's fitted sidelobe level
41    /// (`SIImageStore::getPSFSidelobeLevel`).
42    ///
43    /// # Errors
44    ///
45    /// When the PSF has no positive finite peak or the main lobe cannot be
46    /// fitted.
47    pub fn new(psf: &[f64], shape: PlaneShape) -> Result<Self, Error> {
48        let peak = psf_peak(psf, shape).ok_or(Error::PsfPeak)?;
49        if !(psf[peak].is_finite() && psf[peak] > 0.0) {
50            return Err(Error::PsfPeak);
51        }
52        let single = psf.iter().map(|value| *value as f32).collect::<Vec<_>>();
53        let axes = [shape.nx, shape.ny];
54        let beam = fit_restoring_beam(&single, axes, [1.0, 1.0], DEFAULT_PSF_FIT_CUTOFF)?;
55        let sidelobe = fitted_psf_sidelobe_fraction_with_beam(&single, axes, beam)?;
56        Ok(Self {
57            peak,
58            beam,
59            sidelobe,
60            clark: ClarkPatch::new(psf, shape, beam),
61        })
62    }
63
64    /// Storage index of the PSF peak.
65    #[must_use]
66    pub const fn peak(&self) -> usize {
67        self.peak
68    }
69
70    /// The fitted main lobe, in pixels.
71    #[must_use]
72    pub const fn beam(&self) -> RestoringBeam {
73        self.beam
74    }
75
76    /// The fitted sidelobe level the cycle threshold scales by.
77    #[must_use]
78    pub const fn sidelobe(&self) -> f64 {
79        self.sidelobe
80    }
81
82    /// Clark's patch.
83    #[must_use]
84    pub const fn clark(&self) -> ClarkPatch {
85        self.clark
86    }
87}
88
89impl ClarkPatch {
90    /// At least four pixels, otherwise the ceiling of the fitted widths;
91    /// a `3·ncent + 1` square capped by the plane.
92    fn new(psf: &[f64], shape: PlaneShape, beam: RestoringBeam) -> Self {
93        let central = 4_usize
94            .max(beam.major_fwhm_rad().ceil() as usize)
95            .max(beam.minor_fwhm_rad().ceil() as usize);
96        let requested = 3 * central + 1;
97        let size = [requested.min(shape.nx), requested.min(shape.ny)];
98        let radius = [size[0] / 2, size[1] / 2];
99        // ClarkCleanLatModel::absMaxBeyondDist about the PSF origin
100        // (psfShape/2): rows beyond ±d, and in the other rows the columns
101        // left of −d and from +d on. A patch as large as the plane leaves no
102        // exterior (getPsfPatch's user value, 0).
103        let [cx, cy] = shape.centre();
104        let exterior_sidelobe = if size == [shape.nx, shape.ny] {
105            0.0
106        } else {
107            psf.iter()
108                .enumerate()
109                .filter(|(index, _)| {
110                    let [x, y] = shape.pixel(*index);
111                    y + radius[1] < cy
112                        || y > cy + radius[1]
113                        || x + radius[0] < cx
114                        || x >= cx + radius[0]
115                })
116                .fold(0.0_f64, |maximum, (_, value)| maximum.max(value.abs()))
117        };
118        Self {
119            radius,
120            size,
121            exterior_sidelobe,
122        }
123    }
