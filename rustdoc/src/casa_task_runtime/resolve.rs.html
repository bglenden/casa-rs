3//! One-shot resolution of a surface's parameters over its defaults, as a
4//! provider resolves a request on every route: command-line flags or the
5//! sparse JSON of `--json-run` become the same active values.
6
7use std::collections::BTreeMap;
8use std::path::PathBuf;
9
10use casa_provider_contracts::{ParameterValue, SurfaceContractBundle};
11use serde_json::{Map, Value};
12use thiserror::Error;
13
14use crate::{
15    BaseSource, DiagnosticLevel, OpenSessionRequest, ParameterRuntime, ParameterRuntimeError,
16    ResolutionPatch,
17};
18
19/// Why a set of values does not resolve.
20#[derive(Debug, Error)]
21pub enum ResolveError {
22    /// A value was `null`, which no parameter domain admits.
23    #[error("parameter {0:?} is null")]
24    Null(String),
25    /// The session could not be opened.
26    #[error(transparent)]
27    Runtime(#[from] ParameterRuntimeError),
28    /// The resolved values break the surface's contract.
29    #[error("{message}")]
30    Invalid {
31        /// The parameter the diagnostic names, if one.
32        parameter: Option<String>,
33        /// The diagnostic.
34        message: String,
35    },
36}
37
38/// Resolve `overrides` over `bundle`'s defaults and return every active
39/// parameter's value; the first error-level diagnostic refuses them.
40pub fn resolve_active_values(
41    bundle: SurfaceContractBundle,
42    workspace: PathBuf,
43    overrides: ResolutionPatch,
44) -> Result<BTreeMap<String, ParameterValue>, ResolveError> {
45    let session = ParameterRuntime::default().open_session(OpenSessionRequest {
46        bundle,
47        workspace,
48        source: BaseSource::Defaults,
49        profile_text: None,
50        context_patch: ResolutionPatch::default(),
51        override_patch: overrides,
52        managed_save: false,
53    })?;
54    if let Some(diagnostic) = session
55        .diagnostics()
56        .iter()
57        .find(|diagnostic| diagnostic.level == DiagnosticLevel::Error)
58    {
59        return Err(ResolveError::Invalid {
60            parameter: diagnostic.parameter.clone(),
61            message: diagnostic.message.clone(),
62        });
63    }
64    Ok(session
65        .states()
66        .iter()
67        .filter(|(_, state)| state.active)
68        .filter_map(|(name, state)| Some((name.clone(), state.value.clone()?)))
69        .collect())
70}

72/// Resolve sparse plain-JSON `values`, keyed by parameter name, over
73/// `bundle`'s defaults and return every active parameter as plain JSON.
74pub fn resolve_plain_values(
75    bundle: SurfaceContractBundle,
76    workspace: PathBuf,
77    values: &Map<String, Value>,
78) -> Result<Map<String, Value>, ResolveError> {
79    let overrides = ResolutionPatch {
80        values: values
81            .iter()
82            .map(|(name, value)| {
83                ParameterValue::from_plain_json(value)
84                    .map(|value| (name.clone(), value))
85                    .ok_or_else(|| ResolveError::Null(name.clone()))
86            })
87            .collect::<Result<_, _>>()?,
88        ..ResolutionPatch::default()
89    };
90    Ok(resolve_active_values(bundle, workspace, overrides)?
91        .into_iter()
92        .map(|(name, value)| (name, value.to_plain_json()))
93        .collect())
94}
