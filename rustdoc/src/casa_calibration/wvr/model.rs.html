4use std::f64::consts::PI;

6use super::data::Temperatures;

8const LO_GHZ: f64 = 183.31;
9const WATER_LINE_GHZ: f64 = 183.310107;
10const BAND_OFFSETS_GHZ: [f64; 4] = [1.25, 3.25, 5.5, 7.25];
11const BAND_WIDTHS_GHZ: [f64; 4] = [1.5, 2.5, 2.0, 1.5];
12const GAUSS_ABSCISSA: [f64; 5] = [
13    -0.906179845938664,
14    -0.5384693101056831,
15    0.0,
16    0.5384693101056831,
17    0.906179845938664,
18];
19const GAUSS_WEIGHT: [f64; 5] = [
20    0.2369268850561891,
21    0.4786286704993665,
22    0.5688888888888889,
23    0.4786286704993665,
24    0.2369268850561891,
25];

27// HITRAN 2004 H2(16)O partition sums for 250 through 296 K.
28// Retrieval uses 250–295 K and scales the line strength from 296 K.
29const WATER_PARTITION: [f64; 47] = [
30    135.742004, 136.551441, 137.362522, 138.175246, 138.989607, 139.805604, 140.623233, 141.442490,
31    142.263373, 143.085879, 143.910004, 144.735545, 145.562700, 146.391469, 147.221851, 148.053843,
32    148.887444, 149.722655, 150.559472, 151.397894, 152.237922, 153.079552, 153.922785, 154.767618,
33    155.614050, 156.462081, 157.311708, 158.162931, 159.015748, 159.870158, 160.726160, 161.583752,
34    162.442933, 163.303702, 164.166058, 165.029999, 165.895590, 166.762764, 167.631518, 168.501850,
35    169.373758, 170.247241, 171.122297, 171.998922, 172.877117, 173.756878, 174.638204,
36];
37const H_OVER_K_K_PER_GHZ: f64 = 0.047992375;
38const COLUMN_PER_MM_PER_CM2: f64 = 3.3427e21;
39const SKY_COUPLING: f64 = 0.98;
40const TERMINATION_K: f64 = 275.0;
41const REFERENCE_K: f64 = 296.0;
42const CLOUD_K: f64 = 270.0;
43const BACKGROUND_K: f64 = 2.7;
44
45#[derive(Clone, Copy)]
46pub(super) struct WaterLayer {
47    pub precipitable_water_mm: f64,
48    pub temperature_k: f64,
49    pub pressure_hpa: f64,
50    pub cloud_opacity: f64,
51}

53struct FrequencySample {
54    frequency_ghz: f64,
55    weight: f64,
56    background_rj_k: f64,
57    cloud_rj_k: f64,
58}
59
60pub(super) struct RadiometerModel {
61    bands: [[FrequencySample; 10]; 4],
62    airmass: f64,
63}
64
65impl RadiometerModel {
66    pub fn new(elevation_rad: f64) -> Self {
67        let bands = std::array::from_fn(|channel| {
68            std::array::from_fn(|node| {
69                let index = node % GAUSS_ABSCISSA.len();
70                let side = if node < GAUSS_ABSCISSA.len() {
71                    -1.0
72                } else {
73                    1.0
74                };
75                let frequency = LO_GHZ
76                    + side * BAND_OFFSETS_GHZ[channel]
77                    + 0.5 * BAND_WIDTHS_GHZ[channel] * GAUSS_ABSCISSA[index];
78                FrequencySample {
79                    frequency_ghz: frequency,
80                    weight: 0.25 * GAUSS_WEIGHT[index],
81                    background_rj_k: blackbody_rj_k(frequency, BACKGROUND_K),
82                    cloud_rj_k: blackbody_rj_k(frequency, CLOUD_K),
83                }
84            })
85        });
86        Self {
87            bands,
88            airmass: 1.0 / elevation_rad.sin(),
89        }
90    }
91
92    pub fn temperatures(&self, layer: WaterLayer) -> Temperatures {
93        self.spectrum(layer, layer.precipitable_water_mm * self.airmass)
94    }
95
96    pub fn temperature_per_path(&self, layer: WaterLayer) -> [f64; 4] {
97        const DELTA_MM: f64 = 0.001;
98        let slant = self.airmass * layer.precipitable_water_mm;
99        let high = self.spectrum(layer, slant + DELTA_MM);
100        let low = self.spectrum(layer, slant - DELTA_MM);
101        // ALMA Memo 496 gives the non-dispersive wet path per mm of water.
102        let path_mm_per_mm =
103            COLUMN_PER_MM_PER_CM2 * 18.0105646863 * 1.660538782e-27 * 1.742e7 / layer.temperature_k;
104        let kelvin_per_meter_scale = 1_000.0 / (2.0 * DELTA_MM * path_mm_per_mm);
105        std::array::from_fn(|channel| (high[channel] - low[channel]) * kelvin_per_meter_scale)
106    }
107
108    fn spectrum(&self, layer: WaterLayer, slant_water_mm: f64) -> Temperatures {
109        let opacity = WaterOpacity::new(layer.temperature_k, layer.pressure_hpa);
110        self.bands.each_ref().map(|band| {
111            let brightness = band
112                .iter()
113                .map(|sample| {
114                    let frequency = sample.frequency_ghz;
115                    let mut background = sample.background_rj_k;
116                    if layer.cloud_opacity != 0.0 {
117                        let cloud_transmission =
118                            (-layer.cloud_opacity * (frequency / 183.3).powi(2)).exp();
119                        background = transmit(background, sample.cloud_rj_k, cloud_transmission);
120                    }
121                    let transmission = (-opacity.per_mm(frequency) * slant_water_mm).exp();
122                    sample.weight
123                        * transmit(
124                            background,
125                            blackbody_rj_k(frequency, layer.temperature_k),
126                            transmission,
127                        )
128                })
129                .sum::<f64>();
130            SKY_COUPLING * brightness + (1.0 - SKY_COUPLING) * TERMINATION_K
131        })
132    }
133}

135struct WaterOpacity {
136    center_squared: f64,
137    pressure_width: f64,
138    line_scale: f64,
139    continuum_scale: f64,
140}
141
142impl WaterOpacity {
143    fn new(temperature: f64, pressure: f64) -> Self {
144        let center = WATER_LINE_GHZ - 5.0298e-5 * pressure;
145        let pressure_width = 2.9114e-3 * pressure * (REFERENCE_K / temperature).powf(0.77);
146        let level_population = (195.908398 * (1.0 / REFERENCE_K - 1.0 / temperature)).exp();
147        let photon_energy = H_OVER_K_K_PER_GHZ * center;
148        let stimulated_emission =
149            (-photon_energy / temperature).exp_m1() / (-photon_energy / REFERENCE_K).exp_m1();
150        let strength = 2.333884e-21 * level_population * stimulated_emission * WATER_PARTITION[46]
151            / water_partition(temperature);
152        let air_density_cm3 = 2.686763e19 * (pressure / 1013.25) * (273.15 / temperature);
153        let continuum_scale =
154            6.1e-48 * (300.0 / temperature).powf(2.6) * COLUMN_PER_MM_PER_CM2 * air_density_cm3;
155        Self {
156            center_squared: center * center,
157            pressure_width,
158            line_scale: 4.0 * strength * pressure_width * COLUMN_PER_MM_PER_CM2 / PI,
159            continuum_scale,
160        }
161    }
162
163    fn per_mm(&self, frequency_ghz: f64) -> f64 {
164        let frequency_squared = frequency_ghz.powi(2);
165        let offset_squared = (frequency_squared - self.center_squared).powi(2);
166        let gross_denominator =
167            offset_squared + 4.0 * self.pressure_width.powi(2) * frequency_squared;
168        frequency_squared * (self.line_scale / gross_denominator + self.continuum_scale)
169    }
170}
171
172fn water_partition(temperature: f64) -> f64 {
173    let grid = temperature - 250.0;
174    let lower = (grid as usize).min(WATER_PARTITION.len() - 2);
175    let fraction = grid - lower as f64;
176    WATER_PARTITION[lower] * (1.0 - fraction) + WATER_PARTITION[lower + 1] * fraction
177}
178
179fn blackbody_rj_k(frequency_ghz: f64, temperature_k: f64) -> f64 {
180    let photon_energy = H_OVER_K_K_PER_GHZ * frequency_ghz;
181    let intensity_to_rj = 1.4744992e-18 / 3.0723613e-17;
182    intensity_to_rj * frequency_ghz / (photon_energy / temperature_k).exp_m1()
183}
184
185fn transmit(background: f64, slab: f64, transmission: f64) -> f64 {
186    background * transmission + slab * (1.0 - transmission)
