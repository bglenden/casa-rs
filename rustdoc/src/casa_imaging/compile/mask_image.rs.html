3//! A CASA mask image, read onto an image domain's direction grid and
4//! matched to its channels and Stokes planes as tclean matches it
5//! (`SDMaskHandler::fillMask`, `expandMask`, `copyMask`).
6//!
7//! tclean expands a mask when it has one channel and the image several, or
8//! else one Stokes plane and the image several (`expandMask`):
9//!
10//! * a continuum Stokes I mask to every channel and Stokes plane;
11//! * a continuum mask of the image's Stokes planes to every channel, plane
12//!   by plane, and a continuum mask of another number of Stokes planes is
13//!   refused;
14//! * a cube Stokes I mask to every Stokes plane, channel by channel, when
15//!   its channels are the image's (their number and spectral coordinate);
16//!   otherwise it is refused.
17//!
18//! Any other mask is regridded onto the image's direction and spectral
19//! axes (`copyMask`): casa-rs reads one whose Stokes planes are the
20//! image's and whose channels are the image's, or one continuum channel,
21//! and does not regrid a mask along frequency or Stokes yet (#980).
22//!
23//! The planes share one direction map. Copy an aligned plane directly.
24//! On a shifted grid, sample signed values linearly, treating invalid
25//! input pixels as zero, then take finite positive support as CASA does.
26

43/// The bound on the tiles a mask image's read caches, as a restart's CLEAN
44/// mask is read.
45const CACHE_BYTES: usize = 64 * 1024;
46
47/// The planes of the image a mask is read onto: its channels and Stokes
48/// planes, and its spectral coordinate.
49#[derive(Clone, Copy)]
50pub(super) struct TargetPlanes<'a> {
51    pub(super) channels: usize,
52    pub(super) polarizations: usize,
53    pub(super) spectral: Option<&'a SpectralCoordinate>,
54}

56/// Where a mask image's axes are, by pixel axis.
57struct MaskAxes {
58    shape: Vec<usize>,
59    direction: [usize; 2],
60    stokes: Option<usize>,
61    spectral: Option<usize>,
62}
63
64impl MaskAxes {
65    fn extent(&self, axis: Option<usize>) -> usize {
66        axis.map_or(1, |axis| self.shape[axis])
67    }
68
69    /// The start and shape of the direction plane of `channel` and
70    /// `stokes`: one pixel along every other axis.
71    fn plane(&self, channel: usize, stokes: usize) -> (Vec<usize>, Vec<usize>) {
72        let mut start = vec![0; self.shape.len()];
73        let mut shape = vec![1; self.shape.len()];
74        for axis in self.direction {
75            shape[axis] = self.shape[axis];
76        }
77        if let Some(axis) = self.spectral {
78            start[axis] = channel;
79        }
80        if let Some(axis) = self.stokes {
81            start[axis] = stokes;
82        }
83        (start, shape)
84    }
85
86    /// Pixels of one direction plane.
87    fn plane_cells(&self) -> u64 {
88        self.direction.iter().fold(1_u64, |cells, &axis| {
89            cells.saturating_mul(self.shape[axis] as u64)
90        })
91    }
92}

94/// A mask image of real pixels.
95enum RealImage {
96    Float32(PagedImage<f32>),
97    Float64(PagedImage<f64>),
98}
99
100/// The sampling mode reused across all planes of an input mask.
101enum MaskSampling {
102    Aligned,
103    Affine(DirectionAffine),
104    Mapped(Box<[[f64; 2]]>),
105}
106
107impl MaskSampling {
108    fn source_pixel(&self, [x, y]: [usize; 2], stride: usize) -> [f64; 2] {
109        match self {
110            Self::Aligned => unreachable!("aligned masks are copied directly"),
111            Self::Affine(map) => map.source_pixel([x as f64, y as f64]),
112            Self::Mapped(map) => map[x * stride + y],
113        }
114    }
115}

117/// A real mask image opened onto an image domain, read plane by plane once
118/// its read is admitted ([`Self::bytes`]), as a restart's CLEAN
119/// mask is.
120pub(crate) struct MaskImage {
121    image: RealImage,
122    axes: MaskAxes,
123    /// The planes tclean gives it on the image (`SDMaskHandler::fillMask`).
124    held: MaskPlanes,
125    source: DirectionCoordinateSpec,
126    target: DirectionCoordinateSpec,
127    target_size: [usize; 2],
128    target_channels: usize,
129    /// Bytes of one pixel of the image.
130    pixel_bytes: u64,
131    tile_bytes: u64,
132}
133
134impl MaskImage {
135    /// Open the mask image at `path` for `target`'s direction grid of
136    /// `target_size` pixels on an image of `planes`, reading only its
137    /// description: CASA's refusal, or the planes it gives the mask.
138    pub(super) fn open(
139        path: &Path,
140        target: DirectionCoordinateSpec,
141        target_size: [usize; 2],
142        planes: TargetPlanes<'_>,
143    ) -> Result<Self, PrepareError> {
144        let (image, pixel_bytes) = match AnyPagedImage::open_with_cache(path, CACHE_BYTES)? {
145            AnyPagedImage::Float32(image) => (RealImage::Float32(image), 4_u64),
146            AnyPagedImage::Float64(image) => (RealImage::Float64(image), 8),
147            AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
148                return Err(PrepareError::ComplexMaskImage);
149            }
150        };
151        let (shape, coordinates, tile) = match &image {
152            RealImage::Float32(image) => (image.shape(), image.coordinates(), image.tile_shape()),
153            RealImage::Float64(image) => (image.shape(), image.coordinates(), image.tile_shape()),
154        };
155        let axes = mask_axes(shape, coordinates)?;
156        let held = expansion(path, coordinates, &axes, planes)?;
157        let direction = coordinates
158            .find_coordinate(CoordinateType::Direction)
159            .ok_or(PrepareError::MaskImageDirection)?;
160        let source = super::domains::direction_model_spec(coordinates.coordinate(direction))?;
161        let tile_bytes = tile.iter().fold(pixel_bytes, |bytes, &side| {
162            bytes.saturating_mul(side as u64)
163        });
164        Ok(Self {
165            image,
166            axes,
167            held,
168            source,
169            target,
170            target_size,
171            target_channels: planes.channels,
172            pixel_bytes,
173            tile_bytes,
174        })
175    }

177    /// Change the output coordinate for the allocator-counted read law.
178    #[cfg(test)]
179    pub(crate) fn with_target_direction(mut self, target: DirectionCoordinateSpec) -> Self {
180        self.target = target;
181        self
182    }

184    /// Whether the saved mask and output have the same direction grid.
185    pub(super) fn matches_direction_grid(&self) -> bool {
186        if self.axes.direction.map(|axis| self.axes.shape[axis]) != self.target_size
187            || self.source.projection() != self.target.projection()
188            || self.source.reference_direction().frame()
189                != self.target.reference_direction().frame()
190        {
191            return false;
192        }
193        if self.source == self.target {
194            return true;
195        }
196        // Reading the CASA header can round the same WCS differently. Use
197        // the displacement in pixels so this allowance cannot admit a shift.
198        let [nx, ny] = self.target_size.map(|size| size.saturating_sub(1) as f64);
199        let source = self.source.coordinate();
200        let target = self.target.coordinate();
201        [
202            [0.0, 0.0],
203            [nx, 0.0],
204            [0.0, ny],
205            [nx, ny],
206            [nx / 2.0, ny / 2.0],
207        ]
208        .into_iter()
209        .all(|pixel| {
210            target
211                .pixel_to_world(pixel)
212                .and_then(|world| source.world_to_pixel(world))
213                .is_ok_and(|mapped| {
214                    mapped
215                        .into_iter()
216                        .zip(pixel)
217                        .all(|(a, b)| (a - b).abs() <= 1e-7)
218                })
219        })
220    }
221
222    pub(super) fn is_aligned_continuum(&self) -> bool {
223        self.held == MaskPlanes::SHARED && self.matches_direction_grid()
224    }
225
226    fn exact_affine(&self) -> Option<DirectionAffine> {
227        direction_affine(
228            &self.source.coordinate(),
229            &self.target.coordinate(),
230            self.target_size,
231        )
232    }
233
234    /// MEM's measured image-mask shapes have an aligned or affine grid.
235    pub(super) fn has_shared_tangent_grid(&self) -> bool {
236        self.matches_direction_grid() || self.exact_affine().is_some()
237    }
238
239    fn sampling(&self) -> Result<MaskSampling, PrepareError> {
240        if self.matches_direction_grid() {
241            return Ok(MaskSampling::Aligned);
242        }
243        if let Some(affine) = self.exact_affine() {
244            return Ok(MaskSampling::Affine(affine));
245        }
246        let [nx, ny] = self.target_size;
247        let cells = nx.checked_mul(ny).ok_or(MaskError::InvalidShape)?;
248        let mut positions = Vec::with_capacity(cells);
249        let source = self.source.coordinate();
250        let target = self.target.coordinate();
251        for x in 0..nx {
252            for y in 0..ny {
253                let position = target
254                    .pixel_to_world([x as f64, y as f64])
255                    .and_then(|world| source.world_to_pixel(world))
256                    .map_err(|_| MaskError::UnsupportedReprojection)?;
257                if !position.into_iter().all(f64::is_finite) {
258                    return Err(MaskError::UnsupportedReprojection.into());
259                }
260                positions.push(position);
261            }
262        }
263        Ok(MaskSampling::Mapped(positions.into_boxed_slice()))
264    }
265
266    /// Target pixels of one plane.
267    fn target_plane(&self) -> u64 {
268        (self.target_size[0] as u64).saturating_mul(self.target_size[1] as u64)
269    }
270
271    /// Scratch and retained bytes stay based on one source plane, plus
272    /// the output support. Regions can give a shared mask distinct channels.
273    pub(crate) fn bytes(&self, regions: [&[ChannelMaskBox]; 2]) -> (u64, u64) {
274        let map = if self.has_shared_tangent_grid() {
275            0
276        } else {
277            self.target_plane()
278                .saturating_mul(size_of::<[f64; 2]>() as u64)
279        };
280        let retained = self
281            .target_plane()
282            .saturating_mul(self.planes_with_regions(regions).count() as u64)
283            .saturating_add(size_of::<[usize; 2]>() as u64);
284        let reading = self
285            .axes
286            .plane_cells()
287            .saturating_mul(self.pixel_bytes + 1)
288            .saturating_add(map)
289            .saturating_add(retained.saturating_mul(2))
290            .saturating_add(self.tile_bytes.saturating_mul(2))
291            .saturating_add(CACHE_BYTES as u64);
292        (reading, retained)
293    }
294
295    fn planes_with_regions(&self, regions: [&[ChannelMaskBox]; 2]) -> MaskPlanes {
296        let channels = self.target_channels;
297        let all = 0..=channels.saturating_sub(1);
298        if self.held.channels() == 1
299            && channels > 1
300            && regions.into_iter().flatten().any(|(_, span)| *span != all)
301        {
302            MaskPlanes::new(channels, self.held.polarizations()).expect("nonempty mask planes")
303        } else {
304            self.held
305        }
306    }
307
308    /// Paint regions around each supplied image. A continuum source plane
309    /// is sampled once and applied to each channel with its own support.
310    pub(crate) fn read(
311        self,
312        [before, after]: [&[ChannelMaskBox]; 2],
313    ) -> Result<ReconstructionMaskPlan, PrepareError> {
314        let coordinate = self.target;
315        let size = self.target_size;
316        let planes = self.planes_with_regions([before, after]);
317        let cells = size[0] * size[1] * planes.count();
318        let mut support = vec![false; cells];
319        paint_channel_boxes(&mut support, size, planes, before);
320        self.accumulate(&mut support, planes)?;
321        paint_channel_boxes(&mut support, size, planes, after);
322        Ok(ReconstructionMaskPlan::Reprojected {
323            coordinate,
324            planes,
325            support: support.into(),
326        })
327    }
328
329    /// Read one supplied image into the current binary support.
330    fn accumulate(self, support: &mut [bool], planes: MaskPlanes) -> Result<(), PrepareError> {
331        let held = self.held;
332        let cells = self.target_size[0] * self.target_size[1];
333        let stride = cells * planes.polarizations();
334        let channels_per_source = planes.channels() / held.channels();
335        let sampling = self.sampling()?;
336        for channel in 0..held.channels() {
337            let first = channel * channels_per_source * stride;
338            let end = first + channels_per_source * stride;
339            for stokes in 0..held.polarizations() {
340                let (start, shape) = self.axes.plane(channel, stokes);
341                let target = (
342                    &sampling,
343                    self.target_size,
344                    &mut support[first + stokes * cells..end],
345                    stride,
346                );
347                match &self.image {
348                    RealImage::Float32(image) => {
349                        append_plane_support(image, &start, &shape, self.axes.direction, target)?;
350                    }
351                    RealImage::Float64(image) => {
352                        append_plane_support(image, &start, &shape, self.axes.direction, target)?;
353                    }
354                }
355            }
356        }
357        Ok(())
358    }
359}

361/// Read an ordered list of aligned continuum images into the same Stokes I
362/// plane. Keep no previous input tiles or support planes live across reads.
363pub(crate) fn read_aligned_list(
364    paths: &[PathBuf],
365    coordinate: DirectionCoordinateSpec,
366    size: [usize; 2],
367    context: &RunContext,
368) -> Result<(ReconstructionMaskPlan, Reservation), PrepareError> {
369    let cells = size[0]
370        .checked_mul(size[1])
371        .ok_or(MaskError::InvalidShape)?;
372    let retained = cells as u64 + size_of::<[usize; 2]>() as u64;
373    let charge = admit(
374        &context.host,
375        &context.policy,
376        &Demand {
377            phase: "clean mask support",
378            memory: retained,
379        },
380    )?;
381    let mut support = None::<Arc<[bool]>>;
382    let planes = TargetPlanes {
383        channels: 1,
384        polarizations: 1,
385        spectral: None,
386    };
387    for path in paths {
388        let input = MaskImage::open(path, coordinate, size, planes)?;
389        if !input.is_aligned_continuum() {
390            return Err(PrepareError::MaskListGrid { path: path.clone() });
391        }
392        let _scratch = admit(
393            &context.host,
394            &context.policy,
395            &Demand {
396                phase: "clean mask image",
397                memory: input.bytes([&[], &[]]).0.saturating_sub(retained),
398            },
399        )?;
400        let support = support.get_or_insert_with(|| vec![false; cells].into());
401        input.accumulate(
402            Arc::get_mut(support).expect("list owns its support"),
403            MaskPlanes::SHARED,
404        )?;
405    }
406    let plan = ReconstructionMaskPlan::Reprojected {
407        coordinate,
408        planes: MaskPlanes::SHARED,
409        support: support.ok_or(MaskError::InvalidShape)?,
410    };
411    Ok((plan, charge))
412}

414/// The mask image at `path` opened onto its own direction grid, for an
415/// image of `channels` channels on its own spectral coordinate and
416/// `polarizations` Stokes planes: the counting law's view of the reader
417/// compile uses ([`crate::mask_image_law`]).
418#[cfg(test)]
419pub(crate) fn open_onto_itself(
420    path: &Path,
421    channels: usize,
422    polarizations: usize,
423) -> Result<MaskImage, PrepareError> {
424    let (coordinates, shape) = match AnyPagedImage::open(path)? {
425        AnyPagedImage::Float32(image) => (image.coordinates().clone(), image.shape().to_vec()),
426        AnyPagedImage::Float64(image) => (image.coordinates().clone(), image.shape().to_vec()),
427        AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => {
428            return Err(PrepareError::ComplexMaskImage);
429        }
430    };
431    let axes = mask_axes(&shape, &coordinates)?;
432    let direction = coordinates
433        .find_coordinate(CoordinateType::Direction)
434        .ok_or(PrepareError::MaskImageDirection)?;
435    let target = super::domains::direction_model_spec(coordinates.coordinate(direction))?;
436    let spectral = match coordinates
437        .find_coordinate(CoordinateType::Spectral)
438        .map(|index| coordinates.coordinate(index))
439    {
440        Some(CoordinateModel::Spectral(spectral)) => Some(&**spectral),
441        _ => None,
442    };
443    MaskImage::open(
444        path,
445        target,
446        axes.direction.map(|axis| shape[axis]),
447        TargetPlanes {
448            channels,
449            polarizations,
450            spectral,
451        },
452    )
453}

468/// Where the direction, Stokes and spectral axes of an image of `shape`
469/// are: each coordinate's pixel axes follow those of the coordinates
470/// before it. Every other axis has one pixel.
471fn mask_axes(shape: &[usize], coordinates: &CoordinateSystem) -> Result<MaskAxes, PrepareError> {
472    let first_axis = |index: usize| {
473        (0..index)
474            .map(|k| coordinates.coordinate(k).n_pixel_axes())
475            .sum::<usize>()
476    };
477    let refuse = || PrepareError::MaskImageShape {
478        shape: shape.to_vec(),
479    };
480    let direction = coordinates
481        .find_coordinate(CoordinateType::Direction)
482        .ok_or(PrepareError::MaskImageDirection)?;
483    let longitude = first_axis(direction);
484    let axes = MaskAxes {
485        shape: shape.to_vec(),
486        direction: [longitude, longitude + 1],
487        stokes: coordinates
488            .find_coordinate(CoordinateType::Stokes)
489            .map(first_axis),
490        spectral: coordinates
491            .find_coordinate(CoordinateType::Spectral)
492            .map(first_axis),
493    };
494    let named = [
495        Some(axes.direction[0]),
496        Some(axes.direction[1]),
497        axes.stokes,
498        axes.spectral,
499    ];
500    if shape.len() < longitude + 2
501        || named.iter().flatten().any(|axis| *axis >= shape.len())
502        || (0..shape.len()).any(|axis| !named.contains(&Some(axis)) && shape[axis] != 1)
503    {
504        return Err(refuse());
505    }
506    Ok(axes)
507}

509/// Extract the two direction axes of a slice that is singleton elsewhere.
510fn direction_plane<T>(mut plane: ArrayViewD<'_, T>, axes: [usize; 2]) -> ArrayView2<'_, T> {
511    for axis in (0..plane.ndim()).rev() {
512        if !axes.contains(&axis) {
513            plane = plane.index_axis_move(Axis(axis), 0);
514        }
515    }
516    plane
517        .into_dimensionality::<Ix2>()
518        .expect("two direction axes")
519}

521/// Apply one signed source plane to one or more output planes, x-major.
522/// CASA's copyMask zeros invalid source values before interpolation.
523fn append_plane_support<T>(
524    image: &PagedImage<T>,
525    start: &[usize],
526    shape: &[usize],
527    direction: [usize; 2],
528    (sampling, [nx, ny], support, stride): (&MaskSampling, [usize; 2], &mut [bool], usize),
529) -> Result<(), PrepareError>
530where
531    T: ImagePixel + Into<f64> + Copy,
532{
533    let values = image.get_slice(start, shape)?;
534    let mask = image.get_mask_slice(start, shape, &vec![1; shape.len()])?;
535    let values = direction_plane(values.view(), direction);
536    let mask = mask
537        .as_ref()
538        .map(|mask| direction_plane(mask.view(), direction));
539    let sample = |x: usize, y: usize| {
540        if mask.as_ref().is_none_or(|mask| mask[[x, y]]) {
541            values[[x, y]].into()
542        } else {
543            0.0
544        }
545    };
546    // fillMask adds the signed image to the accumulated 0/1 mask before
547    // thresholding. A selected pixel with image value -1 therefore clears.
548    let accumulate =
549        |selected, value: f64| value.is_finite() && value > if selected { -1.0 } else { 0.0 };
550    let source_shape = [values.nrows(), values.ncols()];
551    let cells = nx * ny;
552    for block in support.chunks_mut(stride) {
553        let rows = block[..cells].chunks_exact_mut(ny);
554        if matches!(sampling, MaskSampling::Aligned) {
555            for (x, row) in rows.enumerate() {
556                for (y, selected) in row.iter_mut().enumerate() {
557                    *selected = accumulate(*selected, sample(x, y));
558                }
559            }
560        } else {
561            for (x, row) in rows.enumerate() {
562                for (y, selected) in row.iter_mut().enumerate() {
563                    let at = sampling.source_pixel([x, y], ny);
564                    let value = linear_sample(source_shape, at, |i, j| Some(sample(i, j)))
565                        .unwrap_or_default();
566                    // CASA stores Float before thresholding. Float64 input keeps its precision.
567                    let value = if size_of::<T>() == size_of::<f32>() {
568                        f64::from(value as f32)
569                    } else {
570                        value
571                    };
572                    *selected = accumulate(*selected, value);
573                }
574            }
575        }
576    }
577    Ok(())
578}

580/// The planes tclean gives a mask of `axes` on an image of `target`
581/// (`SDMaskHandler::fillMask`), or its refusal.
582fn expansion(
583    path: &Path,
584    coordinates: &CoordinateSystem,
585    axes: &MaskAxes,
586    target: TargetPlanes<'_>,
587) -> Result<MaskPlanes, PrepareError> {
588    let channels = axes.extent(axes.spectral);
589    let stokes = axes.extent(axes.stokes);
590    let refuse = |reason: String| PrepareError::MaskImageExpansion {
591        path: path.into(),
592        reason,
593    };
594    let stokes_refusal = || {
595        refuse(format!(
596            "Cannot extend the input mask of {stokes} Stokes planes to the output mask of {} \
597             Stokes planes. Please modify the input mask to make it a Stokes I mask or a mask \
598             with the same number of Stokes planes as the output.",
599            target.polarizations
600        ))
601    };
602    let expands =
603        (channels == 1 && target.channels > 1) || (stokes == 1 && target.polarizations > 1);
604    if expands {
605        // `expandMask`.
606        return if channels == 1 {
607            match stokes {
608                1 => Ok(MaskPlanes::SHARED),
609                stokes if stokes == target.polarizations => Ok(MaskPlanes::new(1, stokes)?),
610                _ => Err(stokes_refusal()),
611            }
612        } else if stokes == 1 {
613            same_channels(coordinates, axes, target).map_err(refuse)?;
614            Ok(MaskPlanes::new(channels, 1)?)
615        } else {
616            Err(refuse(format!(
617                "Input mask,{} does not conform with the number of channels in output mask",
618                path.display()
619            )))
620        };
621    }
622    // `copyMask` regrids the mask's direction and spectral axes onto the
623    // image's: a plane for each Stokes plane and channel the image has.
624    let copies = stokes == target.polarizations
625        && (channels == 1
626            || (channels == target.channels && same_channels(coordinates, axes, target).is_ok()));
627    if copies {
628        Ok(MaskPlanes::new(channels, stokes)?)
629    } else {
630        Err(PrepareError::MaskImageRegrid {
631            path: path.into(),
632            shape: axes.shape.clone(),
633        })
634    }
635}

637/// Whether the mask's channels are the image's: as many, on a spectral
638/// coordinate near the image's (`SDMaskHandler::compareSpectralCoordinate`,
639/// casacore `SpectralCoordinate::near` at its tolerance of 1e-6 on the
640/// frequency system, rest frequency and linear mapping); CASA's refusal
641/// otherwise.
642fn same_channels(
643    coordinates: &CoordinateSystem,
644    axes: &MaskAxes,
645    target: TargetPlanes<'_>,
646) -> Result<(), String> {
647    let channels = axes.extent(axes.spectral);
648    let spectral = coordinates
649        .find_coordinate(CoordinateType::Spectral)
650        .map(|index| coordinates.coordinate(index));
651    let near = match (spectral, target.spectral) {
652        (Some(CoordinateModel::Spectral(mask)), Some(image)) => spectral_near(mask, image),
653        _ => false,
654    };
655    if channels != target.channels {
656        return Err(if near {
657            "Cannot extend the input mask. The number of the channels in Input mask does not \
658             match with that of the output mask. Use a single channel mask or a mask that \
659             matches the spectral coordiante of the output."
660        } else {
661            "Cannot extend the input mask. Spectral coordiante and the number of channels of \
662             the input mask does not match with those of the output mask. Use a single channel \
663             mask or a mask that matches the spectral coordiante of the output."
664        }
665        .to_string());
666    }
667    if !near {
668        return Err(
669            "Cannot extend the input mask. Spectral coordiante of Input mask does not \
670                    match with that of the output mask. Use a single channel mask or a mask \
671                    that matches the spectral coordiante of the output."
672                .to_string(),
673        );
674    }
675    Ok(())
676}
677
678/// casacore `SpectralCoordinate::near` at its default tolerance, on what
679/// places the channels: the frequency system, the rest frequency and the
680/// linear mapping.
681fn spectral_near(mask: &SpectralCoordinate, image: &SpectralCoordinate) -> bool {
682    let near = |left: f64, right: f64| {
683        left == right || (left - right).abs() <= 1.0e-6 * left.abs().max(right.abs())
684    };
685    let pairs = |left: Vec<f64>, right: Vec<f64>| {
686        left.len() == right.len()
687            && left
688                .iter()
689                .zip(&right)
690                .all(|(left, right)| near(*left, *right))
691    };
692    let (mask_model, image_model) = (
693        CoordinateModel::Spectral(Box::new(mask.clone())),
694        CoordinateModel::Spectral(Box::new(image.clone())),
695    );
696    mask.frequency_ref() == image.frequency_ref()
697        && mask_model.axis_units() == image_model.axis_units()
698        && near(mask.rest_frequency(), image.rest_frequency())
699        && pairs(mask_model.reference_value(), image_model.reference_value())
700        && pairs(mask_model.increment(), image_model.increment())
701        && pairs(mask_model.reference_pixel(), image_model.reference_pixel())
702}

716    /// A mask of `stokes` Stokes planes and `channels` channels on the
717    /// spectral coordinate of `increment_hz` channels.
718    fn mask(stokes: usize, channels: usize, increment_hz: f64) -> (CoordinateSystem, MaskAxes) {
719        let mut coordinates = CoordinateSystem::new();
720        coordinates.add_coordinate(DirectionCoordinate::new(
721            DirectionRef::J2000,
722            Projection::new(ProjectionType::SIN),
723            [1.0, 0.5],
724            [-1e-5, 1e-5],
725            [4.0, 4.0],
726        ));
727        let types = [StokesType::I, StokesType::Q, StokesType::U, StokesType::V];
728        coordinates.add_coordinate(StokesCoordinate::new(types[..stokes].to_vec()));
729        coordinates.add_coordinate(spectral(increment_hz));
730        let axes = mask_axes(&[8, 8, stokes, channels], &coordinates).unwrap();
731        (coordinates, axes)
732    }

734    /// tclean's planes for a mask of (Stokes, channels) on an image of
735    /// (polarizations, channels), whose channels are 1 MHz wide.
736    fn expand(
737        (stokes, channels): (usize, usize),
738        (polarizations, image_channels): (usize, usize),
739        increment_hz: f64,
740    ) -> Result<MaskPlanes, PrepareError> {
741        let (coordinates, axes) = mask(stokes, channels, increment_hz);
742        let image = spectral(1.0e6);
743        expansion(
744            Path::new("in.mask"),
745            &coordinates,
746            &axes,
747            TargetPlanes {
748                channels: image_channels,
749                polarizations,

755    fn refused(result: Result<MaskPlanes, PrepareError>) -> String {
756        match result {
757            Err(PrepareError::MaskImageExpansion { reason, .. }) => reason,
758            other => panic!("expected CASA's refusal, not {other:?}"),
759        }
760    }

762    /// `SDMaskHandler::fillMask` and `expandMask`: what each mask becomes on
763    /// each image, and what CASA refuses.
764    #[test]
765    fn masks_expand_to_the_image_planes_as_casa_expands_them() {
766        let planes = |channels, stokes| MaskPlanes::new(channels, stokes).unwrap();
767        let near = 1.0e6;
768        // A continuum Stokes I mask to every plane.
769        for image in [(1, 4), (4, 1), (4, 4), (1, 1)] {
770            assert_eq!(expand((1, 1), image, near).unwrap(), MaskPlanes::SHARED);
771        }
772        // A continuum mask of the image's Stokes planes to every channel.
773        assert_eq!(expand((4, 1), (4, 4), near).unwrap(), planes(1, 4));
774        assert_eq!(expand((4, 1), (4, 1), near).unwrap(), planes(1, 4));
775        // A cube Stokes I mask to every Stokes plane, on the image's
776        // channels; a cube of the image's planes as it is.
777        assert_eq!(expand((1, 3), (4, 3), near).unwrap(), planes(3, 1));
778        assert_eq!(expand((1, 3), (1, 3), near).unwrap(), planes(3, 1));
779        assert_eq!(expand((4, 3), (4, 3), near).unwrap(), planes(3, 4));
780        // CASA refuses a continuum mask of another number of Stokes planes
781        // on a cube, and a cube Stokes I mask of other channels.
782        assert!(
783            refused(expand((2, 1), (4, 4), near))
784                .starts_with("Cannot extend the input mask of 2 Stokes planes")
785        );
786        assert!(refused(expand((1, 3), (4, 4), near)).contains("number of the channels"));
787        assert!(refused(expand((1, 3), (4, 3), 2.0e6)).contains("Spectral coordiante of Input"));
788        assert!(refused(expand((1, 3), (4, 4), 2.0e6)).contains("and the number of channels"));
789        // tclean regrids any other mask along frequency or Stokes, which
790        // casa-rs does not yet.
791        for (mask, image, increment) in [
792            ((2, 1), (4, 1), near),
793            ((1, 3), (1, 3), 2.0e6),
794            ((1, 3), (1, 1), near),
795            ((2, 3), (4, 3), near),
796        ] {
797            assert!(
798                matches!(
799                    expand(mask, image, increment),
800                    Err(PrepareError::MaskImageRegrid { .. })
801                ),
802                "{mask:?} on {image:?}"
803            );
804        }
805    }
