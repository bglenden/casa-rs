2//! The row-work pass: each region-split domain's gridding work by anchor
3//! row, measured before the first pass so that even it cuts balanced strips
4//! (#736).
5
6use std::ops::Range;
7
8use casa_imaging_operator::{PlaneRange, WeightingGeneration};
9
10use super::partition::Router;
11use super::{
12    BoundedSource, Cancel, Partition, PassDomain, PassError, SampleContent, WorkerTeam,
13    chunk_ranges, slices, stream_blocks,
14};

16/// Row chunks per worker of a row-work pass.
17const CHUNKS_PER_WORKER: usize = 4;
18
19/// Rows a chunk of a row-work pass places at once. It keeps only a
20/// histogram, so this bounds the work between the workers' meetings, not
21/// memory.
22const CHUNK_ROWS: usize = 1 << 12;
23
24/// Whether some domain is split among several region owners
25/// ([`Partition::splits_rows`]), the only case a row-work pass serves.
26#[must_use]
27pub fn needs_row_work(domains: &[PassDomain<'_>]) -> bool {
28    domains.iter().any(|domain| domain.partition.splits_rows())
29}
30
31/// The grid rows a row-work pass measures for `domain`: every row of one
32/// split among region owners, none otherwise.
33fn measured_rows(domain: &PassDomain<'_>) -> usize {
34    if domain.partition.splits_rows() {
35        domain.operator.geometry().grid_shape()[1]
36    } else {
37        0
38    }
39}
40
41/// Bytes a row-work pass over `domains` on `workers` workers holds: each
42/// chunk's histogram, allocated once at the widest measured domain's grid
43/// rows; the totals it returns; each domain's router (an owner per grid row
44/// for region owners, a start per owner for plane owners); and the
45/// containers of chunks, routers and totals. The source's blocks are the
46/// source's charge ([`super::source_block_bytes`]).
47#[must_use]
48pub fn row_work_pass_bytes(domains: &[PassDomain<'_>], workers: usize) -> u64 {
49    let chunks = workers * CHUNKS_PER_WORKER;
50    let widest = domains.iter().map(measured_rows).max().unwrap_or(0);
51    let totals = domains.iter().map(measured_rows).sum::<usize>();
52    let routing = domains
53        .iter()
54        .map(|domain| match domain.partition {
55            Partition::Regions { .. } => {
56                domain.operator.geometry().grid_shape()[1] * size_of::<u16>()
57            }
58            Partition::Planes { owners } => (owners + 1) * size_of::<u32>(),
59        })
60        .sum::<usize>();
61    let containers = chunks * size_of::<(Range<usize>, Vec<u64>)>()
62        + domains.len() * (size_of::<Router>() + size_of::<Option<Vec<u64>>>());
63    // The team's result per chunk, collected on each call
64    // ([`WorkerTeam::for_each_mut`]).
65    let results = chunks * size_of::<Result<(), PassError>>();
66    ((chunks * widest + totals) * size_of::<u64>() + routing + containers + results) as u64
67}

69/// Each domain's gridding work by anchor row over one traversal that reads
70/// flags and weights only ([`SampleContent::FlagsAndWeights`]): every row
71/// is placed as a pass places it, with `weighting`, and each placement adds
72/// its kernel's taps to its anchor row ([`Router::owner_and_work`]). `None`
73/// for a domain of plane owners. It is what a first pass measures
74/// ([`super::PassSummary::row_work`]) without reading or gridding the
75/// visibilities, so the first pass's strips can be cut balanced
76/// ([`Partition::regions_balanced`]). It holds [`row_work_pass_bytes`].
77pub fn run_row_work_pass(
78    domains: &[PassDomain<'_>],
79    weighting: &WeightingGeneration,
80    source: &mut dyn BoundedSource,
81    team: &WorkerTeam,
82    cancel: &Cancel,
83) -> Result<Vec<Option<Vec<u64>>>, PassError> {
84    let Some(main) = domains.first() else {
85        return Err(PassError::NoDomains);
86    };
87    let planes = PlaneRange::new(0, main.operator.basis().planes());
88    source
89        .begin(planes, false, SampleContent::FlagsAndWeights)
90        .map_err(PassError::Source)?;
91    let routers = domains
92        .iter()
93        .map(|domain| Router::new(&domain.partition, planes))
94        .collect::<Vec<_>>();
95    let mut totals = domains
96        .iter()
97        .map(|domain| {
98            let rows = measured_rows(domain);
99            (rows > 0).then(|| vec![0_u64; rows])
100        })
101        .collect::<Vec<_>>();
102    let count = team.workers() * CHUNKS_PER_WORKER;
103    // Each chunk's histogram holds the widest domain's rows from the start,
104    // so a later, taller domain never grows it ([`row_work_pass_bytes`]).
105    let widest = domains.iter().map(measured_rows).max().unwrap_or(0);
106    let mut chunks = (0..count)
107        .map(|_| (0..0, vec![0_u64; widest]))
108        .collect::<Vec<_>>();
109    stream_blocks(source, cancel, false, |block| {
110        for (index, (domain, router)) in domains.iter().zip(&routers).enumerate() {
111            let Some(total) = &mut totals[index] else {
112                continue;
113            };
114            for slice in slices(block.len(), count, CHUNK_ROWS) {
115                let mut used = 0;
116                for ((range, _), rows) in chunks.iter_mut().zip(chunk_ranges(slice, count)) {
117                    *range = rows;
118                    used += 1;
119                }
120                let rows = total.len();
121                team.for_each_mut(&mut chunks[..used], |_, (range, work)| {
122                    let work = &mut work[..rows];
123                    work.fill(0);
124                    // In the block's locality order, as a pass places them:
125                    // a baseline's rows share kernel cells, which a set that
126                    // pages its cells (AW) would otherwise reload row by row.
127                    for position in range.clone() {
128                        domain.resampler.place_each(
129                            domain.operator,
130                            weighting,
131                            &block.row(index, block.ordered_row(position)),
132                            None,
133                            |placement, _, _| {
134                                if !planes.contains(placement.plane) {
135                                    return;
136                                }
137                                let (_, placed) =
138                                    router.owner_and_work(domain.operator, &placement);
139                                if let Some((anchor, taps)) = placed
140                                    && let Some(cell) = work.get_mut(anchor)
141                                {
142                                    *cell += taps;
143                                }
144                            },
145                        )?;
146                    }
147                    Ok::<_, PassError>(())
148                })?;
149                for (_, work) in &chunks[..used] {
150                    for (total, work) in total.iter_mut().zip(work) {
151                        *total += work;
152                    }
153                }
154            }
155        }
156        Ok(())
157    })?;
158    Ok(totals)
159}
