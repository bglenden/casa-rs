4//! The atmosphere's absorption and phase over a spectral grid (ATM's
5//! `RefractiveIndexProfile`).
6//!
7//! For every channel and layer, the layer's refractivity per unit path
8//! N = N′ + iN″ comes from [`crate::refractivity`] at the layer's mean
9//! pressure, temperature and water vapour and the gases' number densities.
10//! Integrating over the layers gives, at the zenith,
11//!
12//! - the opacity τ = Σ N″ⱼ Δzⱼ (nepers), and
13//! - the phase delay φ = Σ N′ⱼ Δzⱼ (radians; reported in degrees) and the
14//!   path length L = (λ/2π) φ.
15//!
16//! ATM separates the *dry* atmosphere (the O2, O3, CO, N2O, NO2, SO2, HCl
17//! and HCN lines and the dry continuum) from the *wet* one (the H2O lines
18//! and the water continuum), and the *dispersive* delay (the lines) from
19//! the *non-dispersive* one (the continua). The wet terms are computed for
20//! the profile's own water column ([`AtmosphereProfile::ground_water_column_m`])
21//! and scale linearly to any other: a wet quantity for column w is the
22//! profile's times w / w₀.
23//!
24//! # Two forms
25//!
26//! [`ChannelAbsorption`] keeps every species ATM keeps (28 complex values a
27//! layer) and answers every per-species accessor. [`ChannelTotals`] keeps
28//! only the dry lines, dry continuum, H2O lines and H2O continuum (4 a
29//! layer), which is all that opacities, path lengths and sky brightness
30//! need. [`AbsorptionProfile`] works over either; the per-species accessors
31//! need the full form.
32//!
33//! # Building in parallel
34//!
35//! Each channel is a pure function of the atmosphere and its frequency
36//! ([`channel_absorption`], [`channel_totals`]). A caller that wants
37//! channels computed on its own threads computes them and passes them to
38//! [`AbsorptionProfile::from_channels`]; [`AbsorptionProfile::new`] computes
39//! them serially. The crate starts no threads. [`AbsorptionProfile::bytes`]
40//! gives the memory a profile keeps, so a caller can admit it first.

50/// One layer's refractivity per unit path at one frequency, by species,
51/// (rad/m, 1/m): everything ATM's `RefractiveIndexProfile` keeps.
52///
53/// The O2, H2O and O3 totals are ATM's own sums of their parts, so they
54/// equal ATM's to the bit.
55#[derive(Clone, Copy, Debug, Default, PartialEq)]
56pub struct LayerAbsorption {
57    /// All O2 lines: the next four summed.
58    pub o2: Complex64,
59    /// (16)O(16)O lines from the ground state.
60    pub o2_16o16o: Complex64,
61    /// (16)O(16)O lines from the first vibrational state.
62    pub o2_vibrational: Complex64,
63    /// (16)O(18)O lines.
64    pub o2_16o18o: Complex64,
65    /// (16)O(17)O lines (zero; see [`Species::O2Isotope17`]).
66    pub o2_16o17o: Complex64,
67    /// The dry continuum.
68    pub dry_continuum: Complex64,
69    /// All H2O lines: the next five summed.
70    pub h2o: Complex64,
71    /// H2(16)O lines from the ground state.
72    pub hh16o: Complex64,
73    /// H2(16)O lines from the first bending state.
74    pub hh16o_bending: Complex64,
75    /// H2(17)O lines.
76    pub hh17o: Complex64,
77    /// H2(18)O lines.
78    pub hh18o: Complex64,
79    /// HDO lines.
80    pub hdo: Complex64,
81    /// The water continuum.
82    pub h2o_continuum: Complex64,
83    /// All O3 lines: the next eight summed.
84    pub o3: Complex64,
85    /// (16)O3 lines from the ground state.
86    pub o3_ground: Complex64,
87    /// (16)O3 lines from the ν1 state.
88    pub o3_nu1: Complex64,
89    /// (16)O3 lines from the ν2 state.
90    pub o3_nu2: Complex64,
91    /// (16)O3 lines from the ν3 state.
92    pub o3_nu3: Complex64,
93    /// (16)O(16)O(18)O lines.
94    pub o3_668: Complex64,
95    /// (16)O(18)O(16)O lines.
96    pub o3_686: Complex64,
97    /// (16)O(16)O(17)O lines.
98    pub o3_667: Complex64,
99    /// (16)O(17)O(16)O lines.
100    pub o3_676: Complex64,
101    /// CO lines.
102    pub co: Complex64,
103    /// N2O lines.
104    pub n2o: Complex64,
105    /// NO2 lines.
106    pub no2: Complex64,
107    /// SO2 lines.
108    pub so2: Complex64,
109    /// HCl lines.
110    pub hcl: Complex64,
111    /// HCN lines.
112    pub hcn: Complex64,
113}

115/// One layer's refractivity per unit path at one frequency, in the four
116/// totals opacities, path lengths and sky brightness need, (rad/m, 1/m).
117#[derive(Clone, Copy, Debug, Default, PartialEq)]
118pub struct LayerTotals {
119    /// The dry lines: O2, O3, CO, N2O, NO2, SO2, HCl and HCN.
120    pub dry_lines: Complex64,
121    /// The dry continuum.
122    pub dry_continuum: Complex64,
123    /// The H2O lines.
124    pub h2o_lines: Complex64,
125    /// The water continuum.
126    pub h2o_continuum: Complex64,
127}

129impl LayerAbsorption {
130    /// The species' refractivities in `layer` at `frequency_ghz`, as ATM's
131    /// `mkRefractiveIndexProfile` forms them: each species once, the O2,
132    /// H2O and O3 totals as ATM sums them, and nothing for a gas the layer
133    /// has none of.
134    #[must_use]
135    pub fn of_layer(layer: &Layer, frequency_ghz: f64) -> Self {
136        let (t, p, nu) = (layer.temperature_k, layer.pressure_hpa, frequency_ghz);
137        // ATM's water vapour partial pressure from g/m³.
138        let e = layer.water_vapour_kg_m3 * 1000.0 * t / 217.0;
139        let s = |species| specific_refractivity(species, t, p, e, nu);
140        let mut a = Self::default();
141
142        let o2 = [
143            Species::O2,
144            Species::O2Vibrational,
145            Species::O2Isotope18,
146            Species::O2Isotope17,
147        ]
148        .map(s);
149        a.o2 = o2_per_path(o2[0] + o2[1] + o2[2] + o2[3], t, p);
150        [a.o2_16o16o, a.o2_vibrational, a.o2_16o18o, a.o2_16o17o] =
151            o2.map(|x| o2_per_path(x, t, p));
152        a.h2o_continuum = s(Species::WaterContinuum);
153        a.dry_continuum = s(Species::DryContinuum);
154
155        if layer.water_vapour_kg_m3 > 0.0 {
156            let w = [
157                Species::H2o,
158                Species::H2oBending,
159                Species::H2oIsotope18,
160                Species::H2oIsotope17,
161                Species::Hdo,
162            ]
163            .map(s);
164            a.h2o = h2o_per_path(w[0] + w[1] + w[2] + w[3] + w[4], t, e);
165            [a.hh16o, a.hh16o_bending, a.hh18o, a.hh17o, a.hdo] = w.map(|x| h2o_per_path(x, t, e));
166        }
167        if layer.o3_per_m3 > 0.0 {
168            let density = layer.o3_per_m3 * 1e-6 * 1e6;
169            let o3 = [
170                Species::O3,
171                Species::O3Nu1,
172                Species::O3Nu2,
173                Species::O3Nu3,
174                Species::O3Isotope668,
175                Species::O3Isotope686,
176                Species::O3Isotope667,
177                Species::O3Isotope676,
178            ]
179            .map(|species| specific_refractivity(species, t, p, 0.0, nu) * density);
180            [
181                a.o3_ground,
182                a.o3_nu1,
183                a.o3_nu2,
184                a.o3_nu3,
185                a.o3_668,
186                a.o3_686,
187                a.o3_667,
188                a.o3_676,
189            ] = o3;
190            a.o3 = o3[0] + o3[1] + o3[2] + o3[3] + o3[4] + o3[5] + o3[6] + o3[7];
191        }
192        let minor = |species, density: f64| {
193            if density > 0.0 {
194                specific_refractivity(species, t, p, 0.0, nu) * (density * 1e-6) * 1e6
195            } else {
196                Complex64::new(0.0, 0.0)
197            }
198        };
199        a.co = minor(Species::Co, layer.co_per_m3);
200        a.n2o = minor(Species::N2o, layer.n2o_per_m3);
201        a.no2 = minor(Species::No2, layer.no2_per_m3);
202        a.hcl = minor(Species::Hcl, layer.hcl_per_m3);
203        a.hcn = minor(Species::Hcn, layer.hcn_per_m3);
204        a.so2 = minor(Species::So2, layer.so2_per_m3);
205        a
206    }
207
208    /// The four totals.
209    #[must_use]
210    pub fn totals(&self) -> LayerTotals {
211        LayerTotals {
212            dry_lines: self.o2
213                + self.o3
214                + self.co
215                + self.n2o
216                + self.no2
217                + self.so2
218                + self.hcl
219                + self.hcn,
220            dry_continuum: self.dry_continuum,
221            h2o_lines: self.h2o,
222            h2o_continuum: self.h2o_continuum,
223        }
224    }
225}

227impl LayerTotals {
228    /// The dry absorption coefficient (1/m): ATM's `getAbsTotalDry`.
229    #[must_use]
230    pub fn dry_absorption(&self) -> f64 {
231        self.dry_lines.im + self.dry_continuum.im
232    }
233
234    /// The wet absorption coefficient (1/m) for the profile's own water:
235    /// ATM's `getAbsTotalWet`.
236    #[must_use]
237    pub fn wet_absorption(&self) -> f64 {
238        (self.h2o_lines + self.h2o_continuum).im
239    }

242/// What a profile keeps for one channel: its frequency and its layers.
243pub trait AbsorptionChannel {
244    /// The channel's frequency, Hz.
245    fn frequency_hz(&self) -> f64;
246    /// The number of layers.
247    fn layer_count(&self) -> usize;
248    /// The totals of layer `layer`.
249    fn layer_totals(&self, layer: usize) -> LayerTotals;

252}

323/// Every species' refractivity in every layer of `atmosphere` at
324/// `frequency_hz`: one channel of a full [`AbsorptionProfile`]. Pure, so
325/// channels can be computed on any threads.
326#[must_use]
327pub fn channel_absorption(atmosphere: &AtmosphereProfile, frequency_hz: f64) -> ChannelAbsorption {
328    let nu = 1.0e-9 * frequency_hz;
329    let mut layers = Vec::with_capacity(atmosphere.layers().len());
330    layers.extend(
331        atmosphere
332            .layers()
333            .iter()
334            .map(|layer| LayerAbsorption::of_layer(layer, nu)),
335    );
336    ChannelAbsorption {
337        frequency_hz,
338        layers,
339    }
340}

361/// The absorption and phase of an atmosphere over a spectral grid (ATM's
362/// `RefractiveIndexProfile`), in the full ([`ChannelAbsorption`]) or
363/// compact ([`ChannelTotals`]) form.
364#[derive(Clone, Debug, PartialEq)]
365pub struct AbsorptionProfile<C = ChannelAbsorption> {
366    atmosphere: AtmosphereProfile,
367    grid: SpectralGrid,
368    channels: Vec<C>,
369    ground_water_column_m: f64,
370}
371
372impl AbsorptionProfile<ChannelAbsorption> {
373    /// The full profile of `atmosphere` over `grid`, computed serially.
374    #[must_use]
375    pub fn new(atmosphere: AtmosphereProfile, grid: SpectralGrid) -> Self {
376        let mut channels = Vec::with_capacity(grid.total_channels());
377        channels.extend(
378            grid.frequencies_hz()
379                .map(|frequency| channel_absorption(&atmosphere, frequency)),
380        );
381        Self::assemble(atmosphere, grid, channels)
382    }
383}

387    #[must_use]
388    pub fn new_totals(atmosphere: AtmosphereProfile, grid: SpectralGrid) -> Self {
389        let mut channels = Vec::with_capacity(grid.total_channels());

392                .map(|frequency| channel_totals(&atmosphere, frequency)),

398impl<C: AbsorptionChannel> AbsorptionProfile<C> {
399    /// A profile from channels computed elsewhere (for example in parallel
400    /// with [`channel_absorption`] or [`channel_totals`]), one per grid
401    /// channel in the grid's flat order.
402    ///
403    /// The profile keeps no spare capacity: the vector of channels (and the
404    /// grid's and atmosphere's vectors) are shrunk to their lengths, so the
405    /// profile keeps [`AbsorptionProfile::bytes`] however `channels` was
406    /// grown. Shrinking reallocates in place where the allocator can; give
407    /// an exactly sized vector to be sure of no transient copy.
408    ///
409    /// # Errors
410    ///
411    /// [`AtmosphereError::InvalidParameter`] if the channel count, a
412    /// channel's frequency or its layer count does not match `grid` and
413    /// `atmosphere`.
414    pub fn from_channels(
415        atmosphere: AtmosphereProfile,
416        grid: SpectralGrid,
417        channels: Vec<C>,
418    ) -> Result<Self, AtmosphereError> {
419        if channels.len() != grid.total_channels() {
420            return Err(AtmosphereError::InvalidParameter(format!(
421                "{} channels for a grid of {}",
422                channels.len(),
423                grid.total_channels()
424            )));
425        }
426        let layers = atmosphere.layers().len();
427        for (index, (channel, frequency)) in channels.iter().zip(grid.frequencies_hz()).enumerate()
428        {
429            if channel.frequency_hz().to_bits() != frequency.to_bits() {
430                return Err(AtmosphereError::InvalidParameter(format!(
431                    "channel {index} is at {} Hz, the grid's at {frequency} Hz",
432                    channel.frequency_hz()
433                )));
434            }
435            if channel.layer_count() != layers {
436                return Err(AtmosphereError::InvalidParameter(format!(
437                    "channel {index} has {} layers, the atmosphere {layers}",
438                    channel.layer_count()
439                )));
440            }
441        }
442        Ok(Self::assemble(atmosphere, grid, channels))
443    }
444
445    /// The profile, keeping no spare vector capacity: what
446    /// [`AbsorptionProfile::bytes`] counts is then exactly what it keeps.
447    /// A channel's own layers are exactly sized already: the channel types
448    /// are only made by this module, with exact capacities.
449    fn assemble(
450        mut atmosphere: AtmosphereProfile,
451        mut grid: SpectralGrid,
452        mut channels: Vec<C>,
453    ) -> Self {
454        channels.shrink_to_fit();
455        grid.shrink_to_fit();
456        atmosphere.shrink_to_fit();
457        let ground_water_column_m = atmosphere.ground_water_column_m();
458        Self {
459            atmosphere,
460            grid,
461            channels,
462            ground_water_column_m,
463        }
464    }
465
466    /// The heap bytes a profile of this form keeps for `grid` and an
467    /// atmosphere of `layers` layers: the vector of channels and each
468    /// channel's layers, the grid's windows and channel frequencies, and the
469    /// atmosphere's layers. A profile keeps exactly this, whatever spare
470    /// capacity the vectors it was built from had (see
471    /// [`AbsorptionProfile::from_channels`]). Building with
472    /// [`AbsorptionProfile::new`] or [`AbsorptionProfile::new_totals`] holds
473    /// nothing more: the channels' vector is allocated once at its size and
474    /// each layer is formed in place. The `allocation` tests measure both.
475    #[must_use]
476    pub fn bytes(grid: &SpectralGrid, layers: usize) -> u64 {
477        let channels = grid.total_channels() as u64;
478        channels * (size_of::<C>() as u64 + C::heap_bytes(layers))
479            + channels * size_of::<f64>() as u64
480            + size_of_val(grid.windows()) as u64
481            + (layers * size_of::<Layer>()) as u64
482    }
483
484    /// The atmosphere.
485    #[must_use]
486    pub fn atmosphere(&self) -> &AtmosphereProfile {
487        &self.atmosphere
488    }
489
490    /// The spectral grid.
491    #[must_use]
492    pub fn grid(&self) -> &SpectralGrid {
493        &self.grid
494    }
495
496    /// The channels, in the grid's flat order.
497    #[must_use]
498    pub fn channels(&self) -> &[C] {
499        &self.channels
500    }
501
502    /// The water column the wet terms are computed for, metres
503    /// ([`AtmosphereProfile::ground_water_column_m`]).
504    #[must_use]
505    pub fn ground_water_column_m(&self) -> f64 {
506        self.ground_water_column_m
507    }
