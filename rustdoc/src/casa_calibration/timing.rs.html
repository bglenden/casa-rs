2//! Opt-in phase evidence without changing CASA's return values.
3
4pub struct Phase {
5    name: &'static str,
6    start: Option<std::time::Instant>,
7}
8
9impl Phase {
10    pub fn start(name: &'static str) -> Self {
11        Self {
12            name,
13            start: std::env::var_os("CASA_RS_CALIBRATION_TIMING")
14                .map(|_| std::time::Instant::now()),
15        }
16    }
17}
18
19impl Drop for Phase {
20    fn drop(&mut self) {
21        if let Some(start) = self.start {
22            eprintln!(
23                "calibration_phase {} {:.9}",
24                self.name,
25                start.elapsed().as_secs_f64()
26            );
27        }
28    }
29}

31/// Linux's current address-space peak, excluding a parent's pre-exec peak
32/// that can remain in `getrusage`. Report only for opt-in measurements.
33pub fn report_process_memory() {
34    #[cfg(target_os = "linux")]
35    if std::env::var_os("CASA_RS_CALIBRATION_TIMING").is_some()
36        && let Ok(status) = std::fs::read_to_string("/proc/self/status")
37        && let Some(peak) = status.lines().find_map(|line| {
38            line.strip_prefix("VmHWM:")?
39                .split_whitespace()
40                .next()?
41                .parse::<u64>()
42                .ok()
43        })
44    {
45        eprintln!("calibration_peak_rss_kib {peak}");
46    }
47}
