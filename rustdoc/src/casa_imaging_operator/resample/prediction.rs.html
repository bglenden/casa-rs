19/// Reusable buffers for predicting a batch of rows; one per row chunk.
20///
21/// A batch is [`Self::begin`], then [`SpectralResampler::place_prediction`]
22/// for each row, and either one [`Self::apply_native`] that degrids every
23/// placed sample and interpolates the predictions to the rows' native cells
24/// ([`SpectralResampler::native_terms`]) in a single backend call, or one
25/// [`Self::apply`] and [`SpectralResampler::finish_prediction`] for each
26/// row. [`SpectralResampler::predict_row`] is the batch of one row.
27/// [`SpectralResampler::prediction_scratch`] allocates the buffers once for
28/// a batch of its widest rows; a default scratch grows as batches need.
29#[derive(Debug, Default)]
30pub struct PredictionScratch {
31    /// The batch's placements: a prediction reads no values or weights.
32    placements: Vec<Placement>,
33    npol: usize,
34    predicted: Vec<Complex32>,
35    /// The output or native channel each placement predicts.
36    sources: Vec<usize>,
37    /// While one row's terms are formed, the batch index of the placement
38    /// at each of its output or native channels.
39    slots: Vec<u32>,
40    /// The native terms of the batch's rows.
41    terms: Vec<NativeTerm>,
42}
43
44impl PredictionScratch {
45    /// Scratch for `npol` correlations with room for `samples` degridded
46    /// samples, `slots` output or native channels of one row and `terms`
47    /// native cells, so that predicting rows within them never grows it
48    /// beyond [`Self::bytes`].
49    #[must_use]
50    pub fn with_capacity(npol: usize, samples: usize, slots: usize, terms: usize) -> Self {
51        Self {
52            placements: Vec::with_capacity(samples),
53            npol,
54            predicted: Vec::with_capacity(samples * npol),
55            sources: Vec::with_capacity(samples),
56            slots: Vec::with_capacity(slots),
57            terms: Vec::with_capacity(terms),
58        }
59    }
60
61    /// Bytes [`Self::with_capacity`] allocates.
62    #[must_use]
63    pub const fn bytes(npol: usize, samples: usize, slots: usize, terms: usize) -> u64 {
64        (samples * (size_of::<Placement>() + npol * size_of::<Complex32>() + size_of::<usize>())
65            + slots * size_of::<u32>()
66            + terms * size_of::<NativeTerm>()) as u64
67    }
68
69    /// Bytes the scratch's buffers hold now, from their capacities.
70    #[must_use]
71    pub fn capacity_bytes(&self) -> u64 {
72        (self.placements.capacity() * size_of::<Placement>()
73            + self.predicted.capacity() * size_of::<Complex32>()
74            + self.sources.capacity() * size_of::<usize>()
75            + self.slots.capacity() * size_of::<u32>()
76            + self.terms.capacity() * size_of::<NativeTerm>()) as u64
77    }
78
79    /// Start a batch of rows of `npol` correlations, clearing every buffer.
80    pub fn begin(&mut self, npol: usize) {
81        self.npol = npol;
82        self.placements.clear();
83        self.sources.clear();
84        self.terms.clear();
85    }
86
87    /// The native terms [`SpectralResampler::native_terms`] added since
88    /// [`Self::begin`].
89    #[must_use]
90    pub fn terms(&self) -> &[NativeTerm] {
91        &self.terms
92    }
93
94    /// The batch's placements, as the block a backend predicts.
95    fn placed(&self) -> SampleBlock<'_> {
96        SampleBlock {
97            placements: &self.placements,
98            values: &[],
99            weights: &[],
100            sources: &[],
101            npol: self.npol,
102        }
103    }
104
105    /// Degrid every sample placed since [`Self::begin`] from `model` in one
106    /// [`Work::Predict`] on `backend`, for
107    /// [`SpectralResampler::finish_prediction`].
108    ///
109    /// # Errors
110    ///
111    /// Whatever `backend` returns.
112    pub fn apply(
113        &mut self,
114        backend: &mut dyn GridBackend,
115        cf: &dyn ConvolutionFunctionSet,
116        model: &PreparedModelGrids,
117    ) -> Result<(), OperatorError> {
118        self.predicted.clear();
119        self.predicted
120            .resize(self.placements.len() * self.npol, Complex32::default());
121        if self.placements.is_empty() {
122            return Ok(());
123        }
124        let block = SampleBlock {
125            placements: &self.placements,
126            values: &[],
127            weights: &[],
128            sources: &[],
129            npol: self.npol,
130        };
131        backend.apply(
132            &block,
133            cf,
134            Work::Predict {
135                model,
136                out: &mut self.predicted,
137            },
138        )
139    }
140
141    /// Degrid every sample placed since [`Self::begin`] from `grids` and
142    /// interpolate the predictions to the batch's native cells
143    /// ([`Self::terms`]) in one [`Work::PredictNative`] on `backend`,
144    /// writing `model`, or with `add` adding to it.
145    ///
146    /// # Errors
147    ///
148    /// Whatever `backend` returns.
149    pub fn apply_native(
150        &self,
151        backend: &mut dyn GridBackend,
152        cf: &dyn ConvolutionFunctionSet,
153        grids: &PreparedModelGrids,
154        model: NativeTarget<'_>,
155        add: bool,
156    ) -> Result<(), OperatorError> {
157        if self.terms.is_empty() {
158            return Ok(());
159        }
160        backend.apply(
161            &self.placed(),
162            cf,
163            Work::PredictNative {
164                grids,
165                terms: &self.terms,
166                model,
167                add,
168            },
169        )
170    }
171}

173/// One row's degridded samples within a [`PredictionScratch`] batch, from
174/// [`SpectralResampler::place_prediction`].
175#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
176pub struct PlacedRow {
177    start: usize,
178    end: usize,
179}
180
181/// The output channels CASA degrids for one row under linear mapping
182/// (`FTMachine::getInterpolateArrays`, after `GridFT::get` returns early
183/// when no native maps): each native's `chanMap` is its channel, −2 in the
184/// linear halo or −1; −1 is replaced by the maximum, and the channels from
185/// the minimum to the maximum are degridded, from channel 0 when the
186/// minimum is the halo's −2 or equals the maximum. When every native lies
187/// in the halo the maximum is −2 and CASA's unsigned loop flags nothing, so
188/// every channel is degridded. `None` when nothing maps.
189fn degridded_channels(axis: SpectralAxis, native: &[f64]) -> Option<std::ops::RangeInclusive<u32>> {
190    let native_width_hz = native[1] - native[0];
191    let map = |frequency_hz: f64| match axis.nearest_channel(frequency_hz) {
192        Some(channel) => i64::from(channel),
193        None if axis.in_linear_halo(frequency_hz, native_width_hz) => -2,
194        None => -1,
195    };
196    let maximum = native.iter().map(|frequency_hz| map(*frequency_hz)).max()?;
197    if maximum == -1 {
198        return None;
199    }
200    if maximum == -2 {
201        return Some(0..=axis.channels - 1);
202    }
203    let minimum = native
204        .iter()
205        .map(|frequency_hz| match map(*frequency_hz) {
206            -1 => maximum,
207            channel => channel,
208        })
209        .min()?;
210    let first = if minimum == maximum {
211        0
212    } else {
213        minimum.max(0)
214    };
215    Some(first as u32..=maximum as u32)
216}

218/// Bit `p` set for each flagged correlation `p` of `channel`.
219fn flag_bits(row: &NativeRow<'_>, npol: usize, channel: usize) -> u32 {
220    row.flags[channel * npol..(channel + 1) * npol]
221        .iter()
222        .enumerate()
223        .fold(0, |bits, (pol, flag)| bits | (u32::from(*flag) << pol))
224}

226impl SpectralResampler {
227    /// The most samples [`Self::predict_row`] degrids for one native row of
228    /// `channels` channels: one per channel, or under linear interpolation
229    /// one per output channel.
230    #[must_use]
231    pub fn prediction_samples_per_row(&self, channels: usize) -> usize {
232        match self.sampling {
233            Sampling::Direct | Sampling::Nearest(_) => channels,
234            Sampling::Linear(axis) => channels.max(axis.channels as usize),
235        }
236    }
237
238    /// Reusable buffers for predicting batches of `rows` rows of at most
239    /// `channels` channels of `npol` correlations, allocated once at
240    /// [`PredictionScratch::bytes`] for every resampler in `resamplers`.
241    #[must_use]
242    pub fn prediction_scratch<'r>(
243        resamplers: impl IntoIterator<Item = &'r Self>,
244        npol: usize,
245        channels: usize,
246        rows: usize,
247    ) -> PredictionScratch {
248        let (samples, slots) = Self::prediction_extent(resamplers, channels);
249        PredictionScratch::with_capacity(npol, rows * samples, slots, rows * channels)
250    }
251
252    /// Bytes [`Self::prediction_scratch`] holds for the same arguments.
253    #[must_use]
254    pub fn prediction_scratch_bytes<'r>(
255        resamplers: impl IntoIterator<Item = &'r Self>,
256        npol: usize,
257        channels: usize,
258        rows: usize,
259    ) -> u64 {
260        let (samples, slots) = Self::prediction_extent(resamplers, channels);
261        PredictionScratch::bytes(npol, rows * samples, slots, rows * channels)
262    }
263
264    /// The most samples any of `resamplers` predicts with for rows of at
265    /// most `channels` channels, and the most output or native channels of
266    /// one row.
267    fn prediction_extent<'r>(
268        resamplers: impl IntoIterator<Item = &'r Self>,
269        channels: usize,
270    ) -> (usize, usize) {
271        resamplers
272            .into_iter()
273            .fold((0, channels), |(samples, slots), resampler| {
274                let linear = match resampler.sampling {
275                    Sampling::Linear(axis) => axis.channels as usize,
276                    Sampling::Direct | Sampling::Nearest(_) => 0,
277                };
278                (
279                    samples.max(resampler.prediction_samples_per_row(channels)),
280                    slots.max(linear),
281                )
282            })
283    }

285    /// The model visibility of every selected sample of `row`, written to
286    /// `out` as `[channel][correlation]`, the way CASA predicts `MODEL_DATA`
287    /// (`GridFT::get`, then `FTMachine::interpolateFrequencyFromgrid`).
288    ///
289    /// A flagged row, a flagged correlation, a native channel off the output
290    /// axis and a sample whose kernel leaves the grid predict zero.
291    /// A constant or Taylor basis, `nearest` mapping and the one-channel
292    /// bypass degrid each native channel at its own frequency on its mapped
293    /// plane. `linear` mapping degrids the output channels CASA degrids for
294    /// the row (see `degridded_channels`) at the channel centre, repeats
295    /// each value `floor(width ratio)` times on CASA's fine grid when output
296    /// channels are wider than native ones, and interpolates linearly to the
297    /// native frequencies on the axis and in its halo (extrapolating from the
298    /// end pair); unmapped output channels contribute zeros to that grid.
299    /// Only the planes `model` holds are degridded; a caller predicting a
300    /// window of planes prepares the window's halo (see
301    /// [`Self::model_planes`]).
302    ///
303    /// This is the batch of one row; a caller predicting many rows places
304    /// them all into one [`PredictionScratch`] batch, so that one backend
305    /// call degrids them.
306    ///
307    /// # Errors
308    ///
309    /// A row inconsistent with `operator`, an `out` of another length than
310    /// the row's values, or whatever `backend` returns.
311    pub fn predict_row(
312        &self,
313        operator: &MeasurementOperator,
314        backend: &mut dyn GridBackend,
315        model: &PreparedModelGrids,
316        row: &NativeRow<'_>,
317        scratch: &mut PredictionScratch,
318        out: &mut [Complex32],
319    ) -> Result<(), OperatorError> {
320        scratch.begin(operator.polarization().correlations().len());
321        let placed = self.place_prediction(operator, model, row, scratch)?;
322        scratch.apply(backend, operator.cf(), model)?;
323        self.finish_prediction(operator, row, placed, scratch, out)
324    }
325
326    /// Whether `row`'s prediction interpolates degridded output channels to
327    /// its native channels (see [`Self::predict_row`]).
328    fn linear_prediction(&self, row: &NativeRow<'_>) -> Option<SpectralAxis> {
329        match self.sampling {
330            Sampling::Linear(axis) if axis.channels > 1 && row.frequencies_hz.len() > 1 => {
331                Some(axis)
332            }
333            Sampling::Direct | Sampling::Nearest(_) | Sampling::Linear(_) => None,
334        }
335    }
336
337    /// Add `row`'s degridded samples to the batch `scratch` began, as
338    /// [`Self::predict_row`] degrids them; a flagged row or an
339    /// autocorrelation adds none.
340    ///
341    /// # Errors
342    ///
343    /// A row inconsistent with `operator`.
344    pub fn place_prediction(
345        &self,
346        operator: &MeasurementOperator,
347        model: &PreparedModelGrids,
348        row: &NativeRow<'_>,
349        scratch: &mut PredictionScratch,
350    ) -> Result<PlacedRow, OperatorError> {
351        let npol = operator.polarization().correlations().len();
352        self.validate_row(operator, row, npol)?;
353        let PredictionScratch {
354            placements,
355            sources,
356            ..
357        } = scratch;
358        let start = placements.len();
359        if row.row_flag || is_autocorrelation(row) {
360            return Ok(PlacedRow { start, end: start });
361        }
362        let native = row.frequencies_hz;
363        let prepared = model.layout().planes();
364        let mut placing = RowPlacing::new(operator.cf(), operator.geometry(), row);
365        let mut place = |plane: u32, frequency_hz: f64, source: usize| {
366            if !prepared.contains(plane) {
367                return;
368            }
369            if let Some((placement, _)) = locate_sample(
370                operator.cf(),
371                operator.geometry(),
372                self.basis,
373                row,
374                plane,
375                frequency_hz,
376                &mut placing,
377            ) {
378                placements.push(placement);
379                sources.push(source);
380            }
381        };
382        match self.linear_prediction(row) {
383            None => {
384                for (channel, frequency_hz) in native.iter().enumerate() {
385                    let plane = match self.sampling {
386                        Sampling::Direct => Some(0),
387                        Sampling::Nearest(axis) | Sampling::Linear(axis) => {
388                            axis.nearest_channel(*frequency_hz)
389                        }
390                    };
391                    if let Some(plane) = plane {
392                        place(plane, *frequency_hz, channel);
393                    }
394                }
395            }
396            Some(axis) => {
397                if let Some(channels) = degridded_channels(axis, native) {
398                    for channel in channels {
399                        place(channel, axis.centre_hz(channel), channel as usize);
400                    }
401                }
402            }
403        }
404        Ok(PlacedRow {
405            start,
406            end: placements.len(),
407        })
408    }
409
410    /// Add `row`'s native terms to the batch `scratch` began: one per native
411    /// channel, the row's cells numbered from `first_cell`, interpolating
412    /// its samples `placed` in the batch as [`Self::predict_row`] does. A
413    /// cell whose channel is off the output axis and its halo, or whose row
414    /// placed nothing, gets the zero model; a flagged correlation keeps it
415    /// zero.
416    pub fn native_terms(
417        &self,
418        row: &NativeRow<'_>,
419        npol: usize,
420        placed: PlacedRow,
421        scratch: &mut PredictionScratch,
422        first_cell: u32,
423    ) {
424        let PredictionScratch {
425            sources,
426            slots,
427            terms,
428            ..
429        } = scratch;
430        let native = row.frequencies_hz;
431        let mut term = |channel: usize, low: u32, high: u32, fraction: f32| {
432            terms.push(NativeTerm {
433                cell: first_cell + channel as u32,
434                low,
435                high,
436                fraction,
437                flags: flag_bits(row, npol, channel),
438            });
439        };
440        let mut index_slots = |count: usize| {
441            slots.clear();
442            slots.resize(count, NO_CELL);
443            for (index, source) in sources[placed.start..placed.end].iter().enumerate() {
444                slots[*source] = (placed.start + index) as u32;
445            }
446        };
447        let Some(axis) = self.linear_prediction(row) else {
448            index_slots(native.len());
449            for (channel, &index) in slots.iter().enumerate() {
450                term(channel, index, index, 0.0);
451            }
452            return;
453        };
454        // `FTMachine::interpolateFrequencyFromgrid`: the output-channel
455        // predictions on CASA's image-frequency grid, refined to
456        // `floor(width ratio)` repeated points per channel when output
457        // channels are wider than native ones, interpolated linearly to every
458        // native channel that maps to the axis or lies in its linear halo
459        // (`chanMap` 0… or −2; −1 stays zero); the end pair extrapolates
460        // (casacore `InterpolateArray1D` linear).
461        index_slots(axis.channels as usize);
462        let width = (axis.increment_hz / (native[1] - native[0])).abs();
463        let per_channel = if width > 1.0 {
464            width.floor() as usize
465        } else {
466            1
467        };
468        let fine_increment = axis.increment_hz / per_channel as f64;
469        let fine_start = if per_channel > 1 {
470            axis.first_hz - axis.increment_hz / 2.0 + fine_increment / 2.0
471        } else {
472            axis.first_hz
473        };
474        let points = axis.channels as usize * per_channel;
475        let point = |index: usize| fine_start + index as f64 * fine_increment;
476        let native_width_hz = native[1] - native[0];
477        for (channel, frequency_hz) in native.iter().enumerate() {
478            if axis.nearest_channel(*frequency_hz).is_none()
479                && !axis.in_linear_halo(*frequency_hz, native_width_hz)
480            {
481                term(channel, NO_CELL, NO_CELL, 0.0);
482                continue;
483            }
484            let position = (frequency_hz - fine_start) / fine_increment;
485            let left = (position.floor().max(0.0) as usize).min(points - 2);
486            let fraction = ((frequency_hz - point(left)) / (point(left + 1) - point(left))) as f32;
487            let (low, high) = (left / per_channel, (left + 1) / per_channel);
488            term(channel, slots[low], slots[high], fraction);
489        }
490    }
491
492    /// Write `row`'s model visibilities to `out` as `[channel][correlation]`
493    /// from its samples `placed` in the batch `scratch` applied, as
494    /// [`Self::predict_row`] does.
495    ///
496    /// # Errors
497    ///
498    /// An `out` of another length than the row's values.
499    pub fn finish_prediction(
500        &self,
501        operator: &MeasurementOperator,
502        row: &NativeRow<'_>,
503        placed: PlacedRow,
504        scratch: &mut PredictionScratch,
505        out: &mut [Complex32],
506    ) -> Result<(), OperatorError> {
507        if out.len() != row.values.len() {
508            return Err(OperatorError::NativeRow {
509                reason: "the prediction holds one value per selected sample",
510            });
511        }
512        let npol = operator.polarization().correlations().len();
513        let first = scratch.terms.len();
514        self.native_terms(row, npol, placed, scratch, 0);
515        interpolate_terms(
516            &scratch.predicted,
517            npol,
518            &scratch.terms[first..],
519            out,
520            0,
521            false,
522        );
523        scratch.terms.truncate(first);
524        Ok(())
525    }
