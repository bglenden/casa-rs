24/// `CFCache::getCFParams`: the cell's coordinates and misc info.
25pub(super) fn read_header(path: &Path) -> Result<CellHeader, AwCatalogError> {
26    let image = PagedImage::<Complex32>::open(path)
27        .map_err(|error| cache_error(path, format!("cannot open Complex32 image: {error}")))?;
28    let shape = image.shape();
29    if shape.len() != 4 || shape[0] == 0 || shape[1] == 0 || shape[2..] != [1, 1] {
30        return Err(cache_error(
31            path,
32            format!("expected [nx, ny, 1, 1], got {shape:?}"),
33        ));
34    }
35    if shape[0] % 2 != 0 || shape[1] % 2 != 0 {
36        return Err(cache_error(path, "cell images must have even extents"));
37    }
38    let coordinates = image.coordinates();
39    let spectral = coordinates
40        .find_coordinate(CoordinateType::Spectral)
41        .ok_or_else(|| cache_error(path, "missing spectral coordinate"))?;
42    let frequency = coordinates
43        .coordinate(spectral)
44        .reference_value()
45        .first()
46        .copied()
47        .ok_or_else(|| cache_error(path, "spectral coordinate has no reference value"))?;
48    // CASA reads the reference value into a `Float` before listing it.
49    let frequency_hz = f64::from(frequency as f32);
50    if !frequency_hz.is_finite() || frequency_hz <= 0.0 {
51        return Err(cache_error(
52            path,
53            "cell frequency must be finite and positive",
54        ));
55    }
56    let misc = image.misc_info();
57    let sampling = required_f64(&misc, "Sampling", path)?;
58    if sampling.fract() != 0.0
59        || sampling < 2.0
60        || sampling > f64::from(u16::MAX)
61        || sampling % 2.0 != 0.0
62    {
63        return Err(cache_error(
64            path,
65            "Sampling must be a positive even integer",
66        ));
67    }
68    let support = [
69        required_support(&misc, "Xsupport", path)?,
70        required_support(&misc, "Ysupport", path)?,
71    ];
72    let mueller = required_i32(&misc, "MuellerElement", path)?;
73    let mueller = u32::try_from(mueller)
74        .ok()
75        .filter(|element| *element < MUELLER_ELEMENTS)
76        .ok_or_else(|| cache_error(path, "MuellerElement must lie in 0..16"))?;
77    let w_value = required_f64(&misc, "WValue", path)?;
78    let w_increment = required_f64(&misc, "WIncr", path)?;
79    let pa_deg = required_f64(&misc, "ParallacticAngle", path)?;
80    if !w_value.is_finite()
81        || w_value < 0.0
82        || !w_increment.is_finite()
83        || w_increment < 0.0
84        || !pa_deg.is_finite()
85    {
86        return Err(cache_error(
87            path,
88            "WValue, WIncr and ParallacticAngle must be finite and non-negative",
89        ));
90    }
91    let sampling = sampling as u16;
92    for axis in 0..2 {
93        if (2 * usize::from(support[axis]) + 1) * usize::from(sampling) > shape[axis] {
94            return Err(cache_error(path, "support does not fit the cell image"));
95        }
96    }
97    // Native cells carry the working sky increment they were generated on;
98    // CASA's own caches do not.
99    let sky_increment_rad = match misc.get("CasaRsSkyIncrement") {
100        Some(Value::Scalar(ScalarValue::Float64(value))) => Some(*value),
101        _ => None,
102    };
103    if sky_increment_rad.is_some_and(|value| !value.is_finite() || value <= 0.0) {
104        return Err(cache_error(
105            path,
106            "CasaRsSkyIncrement must be finite and positive",
107        ));
108    }
109    let working_size = match misc.get("CasaRsWorkingSize") {
110        Some(Value::Scalar(ScalarValue::Int32(value))) if *value > 0 => Some(*value as usize),
111        Some(_) => {
112            return Err(cache_error(path, "CasaRsWorkingSize must be positive"));
113        }
114        None => None,
115    };
116    let working_field = sky_increment_rad
117        .zip(working_size)
118        .map(|(increment, size)| increment * size as f64);
119    Ok(CellHeader {
120        path: path.to_path_buf(),
121        shape: [shape[0], shape[1]],
122        sampling,
123        support,
124        mueller,
125        w_value,
126        w_increment,
127        pa_deg,
128        frequency_hz,
129        working_field_rad: working_field,
130    })
131}

133fn required_support(record: &RecordValue, name: &str, path: &Path) -> Result<u16, AwCatalogError> {
134    let value = required_i32(record, name, path)?;
135    u16::try_from(value)
136        .ok()
137        .filter(|support| *support > 0)
138        .ok_or_else(|| cache_error(path, format!("{name} must be positive")))
139}
140
141fn required_f64(record: &RecordValue, name: &str, path: &Path) -> Result<f64, AwCatalogError> {
142    match record.get(name) {
143        Some(Value::Scalar(ScalarValue::Float64(value))) => Ok(*value),
144        Some(Value::Scalar(ScalarValue::Float32(value))) => Ok(f64::from(*value)),
145        Some(value) => Err(cache_error(
146            path,
147            format!("{name} must be floating point, got {value:?}"),
148        )),
149        None => Err(cache_error(path, format!("missing miscinfo field {name}"))),
150    }
151}
152
153fn required_i32(record: &RecordValue, name: &str, path: &Path) -> Result<i32, AwCatalogError> {
154    match record.get(name) {
155        Some(Value::Scalar(ScalarValue::Int32(value))) => Ok(*value),
156        Some(Value::Scalar(ScalarValue::Int64(value))) => i32::try_from(*value)
157            .map_err(|_| cache_error(path, format!("{name} is outside i32 range"))),
158        Some(value) => Err(cache_error(
159            path,
160            format!("{name} must be integer, got {value:?}"),
161        )),
162        None => Err(cache_error(path, format!("missing miscinfo field {name}"))),
163    }
164}

166/// Read the pixels of `header`'s image as a `[y][x]` plane, x fastest.
167fn read_plane(header: &CellHeader) -> Result<Vec<Complex32>, AwCatalogError> {
168    let image = PagedImage::<Complex32>::open(&header.path)
169        .map_err(|error| cache_error(&header.path, format!("cannot open: {error}")))?;
170    let pixels = image
171        .get()
172        .map_err(|error| cache_error(&header.path, format!("cannot read: {error}")))?;
173    let [nx, ny] = header.shape;
174    if pixels.shape() != [nx, ny, 1, 1] {
175        return Err(cache_error(
176            &header.path,
177            "cell image changed shape since open",
178        ));
179    }
180    let mut plane = Vec::with_capacity(nx * ny);
181    for y in 0..ny {
182        for x in 0..nx {
183            plane.push(pixels[[x, y, 0, 0]]);
184        }
185    }
186    Ok(plane)
187}

189/// `AWConvFunc::cfArea`: `Σ` over `[−support, support)` of the integer taps
190/// about `shape/2`.
191pub(super) fn cf_area(
192    plane: &[Complex32],
193    shape: [usize; 2],
194    support: [u16; 2],
195    sampling: u16,
196) -> Complex64 {
197    let origin = [shape[0] / 2, shape[1] / 2];
198    let mut area = Complex64::default();
199    for iy in -i64::from(support[1])..i64::from(support[1]) {
200        for ix in -i64::from(support[0])..i64::from(support[0]) {
201            let x = origin[0] as i64 + ix * i64::from(sampling);
202            let y = origin[1] as i64 + iy * i64::from(sampling);
203            if x < 0 || y < 0 || x as usize >= shape[0] || y as usize >= shape[1] {
204                continue;
205            }
206            let value = plane[y as usize * shape[0] + x as usize];
207            area += Complex64::new(f64::from(value.re), f64::from(value.im));
208        }
209    }
210    area
211}

213/// Load one group's Mueller planes (its `cells`, at `group`'s supports)
214/// into an imaging and a weight cell:
215/// every plane normalised by its own area (`AWConvFunc::cfArea`), tiled at
216/// its kind's largest support with zero beyond a smaller plane's image.
217pub(super) fn load_group(
218    cells: &[CellFiles],
219    group: &Group,
220    sampling: u16,
221) -> Result<Loaded, AwCatalogError> {
222    let build = |select: fn(&CellFiles) -> &CellHeader,
223                 half: [u16; 2]|
224     -> Result<Arc<DenseCell>, AwCatalogError> {
225        let headers = cells.iter().map(select).collect::<Vec<_>>();
226        let mut planes = Vec::with_capacity(headers.len());
227        for header in &headers {
228            let mut plane = read_plane(header)?;
229            let area = cf_area(&plane, header.shape, header.support, header.sampling);
230            if !(area.norm().is_finite() && area.norm() > 0.0) {
231                return Err(cache_error(&header.path, "cell area is not positive"));
232            }
233            let scale = Complex32::new((1.0 / area).re as f32, (1.0 / area).im as f32);
234            for value in &mut plane {
235                *value *= scale;
236            }
237            planes.push((plane, header.shape, header.support));
238        }
239        Ok(Arc::new(dense_cell_padded(&planes, half, sampling)))
240    };
241    Ok(Loaded {
242        imaging: build(|cell| &cell.imaging, group.imaging_half_support)?,
243        weight: build(|cell| &cell.weight, group.weight_half_support)?,
244        last_use: 0,
245    })
246}

248/// The dense cell of `planes` (each `[y][x]` with origin at `shape/2` and
249/// its own declared half support) at `half` taps each side: tap `k` at
250/// fine offset `off` reads pixel `origin + k·sampling + off`
251/// (`accumulateToGrid.inc`: `convOrigin + (Int)(sampling·ix + off)`), zero
252/// outside the plane and beyond the plane's own support (`AWVisResampler`
253/// takes the support per Mueller cell; the crop buffer around it is not
254/// part of the kernel).
255pub(super) fn dense_cell_padded(
256    planes: &[(Vec<Complex32>, [usize; 2], [u16; 2])],
257    half: [u16; 2],
258    sampling: u16,
259) -> DenseCell {
260    let support = [2 * half[0] + 1, 2 * half[1] + 1];
261    let [sx, sy] = [usize::from(support[0]), usize::from(support[1])];
262    let rows = usize::from(sampling) + 1;
263    let fine_half = i64::from(sampling / 2);
264    let mut data = Vec::with_capacity(rows * rows * planes.len() * sx * sy);
265    for oy in 0..rows {
266        let off_y = oy as i64 - fine_half;
267        for ox in 0..rows {
268            let off_x = ox as i64 - fine_half;
269            for (plane, shape, plane_half) in planes {
270                let origin = [(shape[0] / 2) as i64, (shape[1] / 2) as i64];
271                for iy in 0..sy {
272                    let ky = iy as i64 - i64::from(half[1]);
273                    let y = origin[1] + ky * i64::from(sampling) + off_y;
274                    for ix in 0..sx {
275                        let kx = ix as i64 - i64::from(half[0]);
276                        let x = origin[0] + kx * i64::from(sampling) + off_x;
277                        let inside = kx.abs() <= i64::from(plane_half[0])
278                            && ky.abs() <= i64::from(plane_half[1])
279                            && x >= 0
280                            && y >= 0
281                            && (x as usize) < shape[0]
282                            && (y as usize) < shape[1];
283                        data.push(if inside {
284                            plane[y as usize * shape[0] + x as usize]
285                        } else {
286                            Complex32::default()
287                        });
288                    }
289                }
290            }
291        }
292    }
293    DenseCell {
294        data: data.into_boxed_slice(),
295        support,
296        oversampling: sampling,
297        mueller_planes: u8::try_from(planes.len()).expect("Mueller planes fit u8"),
298    }
299}

319/// Write one generated plane as a CASA cell image (`CFCell::makePersistent`
320/// with `SynthesisUtils::makeFTCoordSys`'s UU/VV coordinate) at `path`.
321/// The image is built under [`STAGING_PREFIX`] beside it and renamed to
322/// `path` only once saved, so `path` names either nothing or a complete
323/// image, whenever the write stops (#739 review).
324pub(super) fn write_cell(
325    path: &Path,
326    plane: &NativeAwPlane<'_>,
327    frequency_hz: f64,
328    grid: casa_imaging_model::NativeAwGrid,
329    misc: &CellMisc,
330) -> Result<(), AwCatalogError> {
331    let write_error = |detail: String| AwCatalogError::Write {
332        path: path.to_path_buf(),
333        detail,
334    };
335    let name = path
336        .file_name()
337        .and_then(|name| name.to_str())
338        .unwrap_or_default()
339        .to_string();
340    let staging = path.with_file_name(format!("{STAGING_PREFIX}{name}"));
341    if staging.exists() {
342        std::fs::remove_dir_all(&staging)
343            .map_err(|error| write_error(format!("cannot clear a staged image: {error}")))?;
344    }
345    build_cell(&staging, &name, plane, frequency_hz, grid, misc).map_err(write_error)?;
346    // A cell being replaced moves aside in one rename, so its name never
347    // names a half-removed image either.
348    let replaced = path.with_file_name(format!("{STAGING_PREFIX}replaced-{name}"));
349    if path.exists() {
350        std::fs::rename(path, &replaced)
351            .map_err(|error| write_error(format!("cannot replace: {error}")))?;
352    }
353    std::fs::rename(&staging, path)
354        .map_err(|error| write_error(format!("cannot publish the staged image: {error}")))?;
355    if replaced.exists() {
356        std::fs::remove_dir_all(&replaced)
357            .map_err(|error| write_error(format!("cannot remove the replaced image: {error}")))?;
358    }
359    Ok(())
360}

362/// Build the cell image of [`write_cell`] at `path`, its misc info naming it
363/// `name`.
364fn build_cell(
365    path: &Path,
366    name: &str,
367    plane: &NativeAwPlane<'_>,
368    frequency_hz: f64,
369    grid: casa_imaging_model::NativeAwGrid,
370    misc: &CellMisc,
371) -> Result<(), String> {
372    let write_error = |detail: String| detail;
373    let side = plane.size;
374    // The uv increment of one oversampled pixel over the working field.
375    let increment =
376        1.0 / (grid.size as f64 * grid.sky_increment_rad[1].abs()) / grid.oversampling as f64;
377    let mut coordinates = CoordinateSystem::new();
378    coordinates.add_coordinate(
379        LinearCoordinate::new(
380            2,
381            vec!["UU".to_string(), "VV".to_string()],
382            vec!["lambda".to_string(), "lambda".to_string()],
383        )
384        .with_reference_value(vec![0.0, 0.0])
385        .with_reference_pixel(vec![(side / 2) as f64, (side / 2) as f64])
386        .with_increment(vec![-increment, increment]),
387    );
388    coordinates.add_coordinate(StokesCoordinate::new(vec![if misc.mueller == 0 {
389        StokesType::RR
390    } else {
391        StokesType::LL
392    }]));
393    coordinates.add_coordinate(SpectralCoordinate::new(
394        FrequencyRef::LSRK,
395        frequency_hz,
396        1.0,
397        0.0,
398        frequency_hz,
399    ));
400    let mut image = PagedImage::<Complex32>::create(vec![side, side, 1, 1], coordinates, path)
401        .map_err(|error| write_error(format!("cannot create: {error}")))?;
402    let mut pixels = ndarray::ArrayD::<Complex32>::zeros(ndarray::IxDyn(&[side, side, 1, 1]));
403    for y in 0..side {
404        for x in 0..side {
405            pixels[[x, y, 0, 0]] = plane.values[y * side + x];
406        }
407    }
408    image
409        .put_slice(&pixels, &[0, 0, 0, 0])
410        .map_err(|error| write_error(format!("cannot write pixels: {error}")))?;
411    let field = |name: &str, value: ScalarValue| RecordField::new(name, Value::Scalar(value));
412    image
413        .set_misc_info(RecordValue::new(vec![
414            field("Xsupport", ScalarValue::Int32(plane.support as i32)),
415            field("Ysupport", ScalarValue::Int32(plane.support as i32)),
416            field("Sampling", ScalarValue::Float64(misc.sampling)),
417            field("ParallacticAngle", ScalarValue::Float64(misc.pa_deg)),
418            field("MuellerElement", ScalarValue::Int32(misc.mueller as i32)),
419            field("WValue", ScalarValue::Float64(misc.w_value)),
420            field("WIncr", ScalarValue::Float64(misc.w_increment)),
421            field("Name", ScalarValue::String(name.to_string())),
422            field("ConjFreq", ScalarValue::Float64(misc.conjugate_frequency)),
423            field(
424                "ConjPoln",
425                ScalarValue::Int32(misc.conjugate_mueller as i32),
426            ),
427            field("TelescopeName", ScalarValue::String("EVLA".to_string())),
428            field("BandName", ScalarValue::String(String::new())),
429            field("Diameter", ScalarValue::Float64(misc.diameter_m)),
430            field("OpCode", ScalarValue::Bool(false)),
431            field(
432                "CasaRsSkyIncrement",
433                ScalarValue::Float64(misc.sky_increment_rad),
434            ),
435            field(
436                "CasaRsWorkingSize",
437                ScalarValue::Int32(
438                    i32::try_from(misc.working_size)
439                        .map_err(|_| write_error("working size exceeds i32".to_string()))?,
440                ),
441            ),
442        ]))
443        .map_err(|error| write_error(format!("cannot write misc info: {error}")))?;
444    image
445        .save()
446        .map_err(|error| write_error(format!("cannot save: {error}")))?;
447    Ok(())
