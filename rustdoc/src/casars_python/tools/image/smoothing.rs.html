24struct SmoothingMask {
25    value: String,
26    history: String,
27}
28
29impl SmoothingMask {
30    fn boxcar_default() -> Self {
31        Self {
32            value: String::new(),
33            history: "[]".into(),
34        }
35    }
36
37    fn hanning_default() -> Self {
38        Self {
39            value: String::new(),
40            history: r#""""#.into(),
41        }
42    }
43}
44
45impl<'a, 'py> FromPyObject<'a, 'py> for SmoothingMask {
46    type Error = PyErr;
47
48    fn extract(value: Borrowed<'a, 'py, PyAny>) -> PyResult<Self> {
49        let mask = mask_argument((!value.is_none()).then_some(&value))?;
50        let history = if value.is_instance_of::<PyString>() {
51            format!("{mask:?}")
52        } else {
53            "[]".into()
54        };
55        Ok(Self {
56            value: mask,
57            history,
58        })
59    }
60}
61
62struct Request<'a> {
63    outfile: &'a str,
64    region: RegionArgument,
65    mask: SmoothingMask,
66    kernel: SmoothingKernel,
67    axis: i64,
68    drop: bool,
69    dmethod: &'a str,
70    overwrite: bool,
71    stretch: bool,
72}
73
74impl Request<'_> {
75    fn name(&self) -> &'static str {
76        match self.kernel {
77            SmoothingKernel::Boxcar(_) => "boxcar",
78            SmoothingKernel::Hanning => "hanning",
79        }
80    }
81
82    fn options(&self) -> PyResult<SmoothingOptions<'_>> {
83        let decimate = if self.drop {
84            match self.dmethod.as_bytes().first().map(u8::to_ascii_lowercase) {
85                Some(b'c') => Some(DecimationMethod::Copy),
86                Some(b'm') => Some(DecimationMethod::Mean),
87                _ => {
88                    return Err(raised(
89                        "Value of dmethod must be either 'm'(ean) or 'c'(opy)",
90                    ));
91                }
92            }
93        } else {
94            None
95        };
96        Ok(SmoothingOptions {
97            axis: self.axis,
98            kernel: self.kernel,
99            decimate,
100            region: self.region.selection(),
101            mask: (!self.mask.value.is_empty()).then_some(self.mask.value.as_str()),
102            stretch: self.stretch,
103            destination: (!self.outfile.is_empty()).then(|| Path::new(self.outfile)),
104        })
105    }
106
107    fn call_history(&self, axis: usize) -> String {
108        let Self {
109            outfile,
110            drop,
111            dmethod,
112            overwrite,
113            stretch,
114            ..
115        } = self;
116        let region = self.region.history();
117        let mask = &self.mask.history;
118        match self.kernel {
119            SmoothingKernel::Boxcar(width) => format!(
120                "ia.boxcar(outfile={outfile:?}, region={region}, mask={mask}, axis={axis}, width={width}, drop={drop}, dmethod={dmethod:?}, overwrite={overwrite}, stretch={stretch})"
121            ),
122            SmoothingKernel::Hanning => format!(
123                "ia.hanning(outfile={outfile:?}, region={region}, mask={mask}, axis={axis}, drop={drop}, overwrite={overwrite}, stretch={stretch}, dmethod={dmethod:?})"
124            ),
125        }
126    }
127
128    fn copy<T: SmoothingPixel>(
129        &self,
130        image: &dyn ImageInterface<T>,
131        coordinates: RecordValue,
132        beams: &BeamView,
133        path: Option<&Path>,
134    ) -> PyResult<(Storage<T>, usize)> {
135        let mut plan = ImageSmoothing::new(image, coordinates, self.options()?, beams.record())
136            .map_err(raised)?;
137        let mut output = Storage::create(
138            plan.section.shape.clone(),
139            plan.section.coordinates.clone(),
140            path,
141            None,
142        )
143        .map_err(raised)?;
144        with_storage!(&mut output, product => plan.copy_to(image, product.as_mut()))
145            .map_err(raised)?;
146        let mut info = plan.section.image_info().to_record();
147        beams.apply_to(&mut info);
148        with_storage!(&mut output, product => product.set_image_info_record(info))
149            .map_err(raised)?;
150        if let Some(record) = plan.section.coordinate_record.take() {
151            match &mut output {
152                Storage::Paged(image) => image.set_coordinate_record(record).map_err(raised)?,
153                Storage::Temp {
154                    record: current, ..
155                } => *current = record,
156            }
157        }
158        Ok((output, plan.axis()))
159    }
160
161    fn build(
162        &self,
163        image: &ImageHandle,
164        beams: &BeamView,
165        path: Option<&Path>,
166        history: bool,
167    ) -> PyResult<ImageHandle> {
168        let record = image.coordinate_record();
169        let (mut output, axis) = match image {
170            ImageHandle::Float(Storage::Temp {
171                image,
172                fits: Some(fits),
173                ..
174            }) => {
175                let (result, axis) = self.copy(&fits.image_view(image), record, beams, path)?;
176                (ImageHandle::Float(result), axis)
177            }
178            ImageHandle::Float(source) => {
179                let (result, axis) =
180                    with_storage!(source, image => self.copy(image.as_ref(), record, beams, path))?;
181                (ImageHandle::Float(result), axis)
182            }
183            ImageHandle::Complex(source) => {
184                let (result, axis) =
185                    with_storage!(source, image => self.copy(image.as_ref(), record, beams, path))?;
186                (ImageHandle::Complex(result), axis)
187            }
188            _ => unreachable!("checked single precision"),
189        };
190        let history_rows: Vec<ImageHistoryRow> = with_pixels!(image, storage => match storage {
191            Storage::Paged(image) => image_history_rows(image.name().expect("native image path")).map_err(raised)?,
192            Storage::Temp { image, .. } => image.history_rows().map_err(raised)?.into_owned(),
193        });
194        with_pixels!(&mut output, storage => with_storage!(storage, product => product.append_history_entries(&history_rows))).map_err(raised)?;
195        if history {
196            match self.kernel {
197                SmoothingKernel::Boxcar(_) => {
198                    output.record_history(self.name(), self.call_history(axis))?
199                }
200                SmoothingKernel::Hanning => output.append_history_at(
201                    "::image::_hanning",
202                    &["Ran ia.hanning".into(), self.call_history(axis)],
203                )?,
204            }
205        }
206        let source_name = image.path().map_or_else(
207            || "Temporary_Image".into(),
208            |name| name.display().to_string(),
209        );
210        let origin = match self.kernel {
211            SmoothingKernel::Boxcar(_) => "::ImageBoxcarSmoother::_reportOldNewImageShapes",
212            SmoothingKernel::Hanning => "::HanningSmoother::_reportOldNewImageShapes",
213        };
214        output.append_history_at(
215            origin,
216            &[
217                format!("Original {source_name} size => {:?}", image.shape()),
218                format!("New {} size => {:?}", self.outfile, output.shape()),
219            ],
220        )?;
221        output.save().map_err(raised)?;
222        Ok(output)
223    }
224}

226impl PyImageTool {
227    fn smooth_axis(&self, py: Python<'_>, request: Request<'_>) -> PyResult<Py<Self>> {
228        let output = py.detach(|| {
229            let state = self.state();
230            let attached = state
231                .image
232                .as_ref()
233                .ok_or_else(|| raised("Unable to create image"))?;
234            let mut live_coordinates = None;
235            let mut output = publish_image(request.outfile, request.overwrite, |path| {
236                let mut image = attached.lock();
237                image.check_single_precision(request.name())?;
238                if !request.outfile.is_empty() {
239                    for input in image
240                        .path()
241                        .into_iter()
242                        .chain(attached.operand_paths().iter().map(|path| path.as_path()))
243                    {
244                        ensure_outfile_does_not_alias_input(Path::new(request.outfile), input)
245                            .map_err(raised)?;
246                    }
247                }
248                attached.require_pixel_read_lock(&mut image)?;
249                let beams = attached.beam_view().clone();
250                let staged = request.build(&image, &beams, path, state.history);
251                let released = attached.release_requested_lock(&mut image);
252                let staged = staged?;
253                released?;
254                live_coordinates = AttachedImage::capture_live_coordinates(&staged)?;
255                Ok(staged)
256            })?;
257            if let Some((record, system)) = live_coordinates {
258                output.keep_live_coordinates(record, system);
259            }
260            Ok::<_, PyErr>(output)
261        })?;
262        Py::new(py, Self::attached(output))
263    }
264}

266#[pymethods]
267impl PyImageTool {
268    /// Smooth an axis over consecutive windows, optionally decimating.
269    #[allow(clippy::too_many_arguments)]
270    #[pyo3(signature = (outfile="", region=RegionArgument::default(), mask=SmoothingMask::boxcar_default(), axis=-1, width=2, drop=true, dmethod="copy", overwrite=false, stretch=false), text_signature = "($self, outfile='', region={}, mask=[], axis=-1, width=2, drop=True, dmethod='copy', overwrite=False, stretch=False)")]
271    fn boxcar(
272        &self,
273        py: Python<'_>,
274        outfile: &str,
275        region: RegionArgument,
276        mask: SmoothingMask,
277        axis: i64,
278        width: i64,
279        #[pyo3(from_py_with = require_bool)] drop: bool,
280        dmethod: &str,
281        #[pyo3(from_py_with = require_bool)] overwrite: bool,
282        #[pyo3(from_py_with = require_bool)] stretch: bool,
283    ) -> PyResult<Py<Self>> {
284        let width = usize::try_from(width).map_err(|_| raised("Boxcar width must be positive"))?;
285        self.smooth_axis(
286            py,
287            Request {
288                outfile,
289                region,
290                mask,
291                kernel: SmoothingKernel::Boxcar(width),
292                axis,
293                drop,
294                dmethod,
295                overwrite,
296                stretch,
297            },
298        )
299    }
300
301    /// Apply the three-point Hanning kernel and optionally decimate.
302    #[allow(clippy::too_many_arguments)]
303    #[pyo3(signature = (outfile="", region=RegionArgument::default(), mask=SmoothingMask::hanning_default(), axis=-10, drop=true, overwrite=false, stretch=false, dmethod="copy"), text_signature = "($self, outfile='', region={}, mask='', axis=-10, drop=True, overwrite=False, stretch=False, dmethod='copy')")]
304    fn hanning(
305        &self,
306        py: Python<'_>,
307        outfile: &str,
308        region: RegionArgument,
309        mask: SmoothingMask,
310        axis: i64,
311        #[pyo3(from_py_with = require_bool)] drop: bool,
312        #[pyo3(from_py_with = require_bool)] overwrite: bool,
313        #[pyo3(from_py_with = require_bool)] stretch: bool,
314        dmethod: &str,
315    ) -> PyResult<Py<Self>> {
316        self.smooth_axis(
317            py,
318            Request {
319                outfile,
320                region,
321                mask,
322                kernel: SmoothingKernel::Hanning,
323                axis,
324                drop,
325                dmethod,
326                overwrite,
327                stretch,
328            },
329        )
330    }
