13fn checked_id(id: i64, count: usize, name: &str) -> MsResult<usize> {
14    usize::try_from(id)
15        .ok()
16        .filter(|&row| row < count)
17        .ok_or_else(|| MsError::InvalidInput(format!("{name} {id} out of range (size {count})")))
18}

20impl PyMsMetadata {
21    fn data_description_value(
22        &self,
23        py: Python<'_>,
24        ddid: i64,
25        value: impl Fn(&MsDataDescription<'_>, usize) -> MsResult<i32> + Send,
26    ) -> PyResult<Py<PyAny>> {
27        if ddid >= 0 {
28            let id = self.with_ms(py, move |ms| {
29                let descriptions = ms.data_description()?;
30                let row = checked_id(ddid, descriptions.row_count(), "Data description ID")?;
31                Ok(i64::from(value(&descriptions, row)? as u32))
32            })?;
33            CasaValue::Int(id).into_python(py)
34        } else {
35            let ids = self.with_ms(py, move |ms| {
36                let descriptions = ms.data_description()?;
37                (0..descriptions.row_count())
38                    .map(|row| value(&descriptions, row).map(|id| u64::from(id as u32)))
39                    .collect::<MsResult<Vec<_>>>()
40            })?;
41            Ok(ids.into_pyarray(py).into_any().unbind())
42        }
43    }
44}

46#[pymethods]
47impl PyMsMetadata {
48    #[pyo3(signature = (spw = -1, pol = -1), text_signature = "($self, spw=-1, pol=-1)")]
49    fn datadescids(&self, py: Python<'_>, spw: i64, pol: i64) -> PyResult<Py<PyAny>> {
50        let ids = self.with_ms(py, |ms| {
51            if spw >= 0 {
52                checked_id(spw, ms.spectral_window()?.row_count(), "Spectral window ID")?;
53            }
54            if pol >= 0 {
55                checked_id(pol, ms.polarization()?.row_count(), "Polarization ID")?;
56            }
57            let descriptions = ms.data_description()?;
58            let mut latest_rows = BTreeMap::new();
59            for row in 0..descriptions.row_count() {
60                let window = i64::from(descriptions.spectral_window_id(row)?);
61                let polarization = i64::from(descriptions.polarization_id(row)?);
62                if (spw < 0 || spw == window) && (pol < 0 || pol == polarization) {
63                    latest_rows.insert((window, polarization), row as i64);
64                }
65            }
66            let mut ids: Vec<_> = latest_rows.into_values().collect();
67            ids.sort_unstable();
68            Ok(ids)
69        })?;
70        Ok(ids.into_pyarray(py).into_any().unbind())
71    }

73    #[pyo3(signature = (ddid = -1), text_signature = "($self, ddid=-1)")]
74    fn spwfordatadesc(&self, py: Python<'_>, ddid: i64) -> PyResult<Py<PyAny>> {
75        self.data_description_value(py, ddid, |description, row| {
76            description.spectral_window_id(row)
77        })
78    }

82        self.data_description_value(py, ddid, |description, row| {
83            description.polarization_id(row)
84        })
85    }

87    #[pyo3(signature = (polid = -1), text_signature = "($self, polid=-1)")]
88    fn ncorrforpol(&self, py: Python<'_>, polid: i64) -> PyResult<Py<PyAny>> {
89        let value = self.with_ms(py, |ms| {
90            let polarizations = ms.polarization()?;
91            let count = polarizations.row_count();
92            if polid >= 0 {
93                let row = checked_id(polid, count, "Polarization ID")?;
94                Ok(CasaValue::Int(i64::from(polarizations.num_corr(row)?)))
95            } else {
96                let values = (0..count)
97                    .map(|row| polarizations.num_corr(row).map(i64::from))
98                    .collect::<MsResult<Vec<_>>>()?;
99                Ok(CasaValue::Ints {
100                    shape: vec![values.len()],
101                    values,
102                })
103            }
104        })?;
105        value.into_python(py)
106    }
107
108    #[pyo3(signature = (pol = -1), text_signature = "($self, pol=-1)")]
109    fn corrtypesforpol(&self, py: Python<'_>, pol: i64) -> PyResult<Py<PyAny>> {
110        let values = self.with_ms(py, |ms| {
111            let polarizations = ms.polarization()?;
112            let row = checked_id(pol, polarizations.row_count(), "Polarization ID")?;
113            Ok(polarizations
114                .corr_type(row)?
115                .into_iter()
116                .map(i64::from)
117                .collect::<Vec<_>>())
118        })?;
119        Ok(values.into_pyarray(py).into_any().unbind())
120    }
121
122    #[pyo3(signature = (pol = -1), text_signature = "($self, pol=-1)")]
123    fn corrprodsforpol(&self, py: Python<'_>, pol: i64) -> PyResult<Py<PyAny>> {
124        let result = self.with_ms(py, |ms| {
125            let polarizations = ms.polarization()?;
126            let row = checked_id(pol, polarizations.row_count(), "Polarization ID")?;
127            let ArrayValue::Int32(values) = polarizations.corr_product(row)? else {
128                return Err(MsError::InvalidInput(
129                    "CORR_PRODUCT is not an Int32 array".into(),
130                ));
131            };
132            Ok(CasaValue::Ints {
133                shape: values.shape().to_vec(),
134                values: values
135                    .view()
136                    .reversed_axes()
137                    .iter()
138                    .copied()
139                    .map(i64::from)
140                    .collect(),
141            })
142        })?;
143        result.into_python(py)
144    }
