23struct PolInputs {
24    stokes_axis: usize,
25    bunit: String,
26    q: Option<ArrayD<f32>>,
27    u: Option<ArrayD<f32>>,
28    v: Option<ArrayD<f32>>,
29    mask_q: Option<ArrayD<bool>>,
30    mask_u: Option<ArrayD<bool>>,
31    mask_v: Option<ArrayD<bool>>,
32    // Zero-copy backing block when Q/U[/V] share one multi-Stokes input file.
33    shared_block: Option<ArrayD<f32>>,
34    shared_mask: Option<ArrayD<bool>>,
35    shared_offsets: [Option<usize>; 3],
36    has_q: bool,
37    has_u: bool,
38    has_v: bool,
39    mask_override: Option<ArrayD<bool>>,
40    ref_coords: casa_coordinates::CoordinateSystem,
41    ref_image_info: casa_images::ImageInfo,
42    ref_misc_info: casa_core::RecordValue,
43    ref_legacy_coords: Option<casa_core::RecordValue>,
44    out_shape: Vec<usize>,
45    sel: ImmathSelection,
46}

48impl PolInputs {
49    fn plane_view(&self, idx: usize) -> Option<ArrayViewD<'_, f32>> {
50        if let (Some(block), Some(rel)) = (&self.shared_block, self.shared_offsets[idx]) {
51            return Some(block.slice_axis(Axis(self.stokes_axis), Slice::from(rel..rel + 1)));
52        }
53        match idx {
54            0 => self.q.as_ref().map(|a| a.view()),
55            1 => self.u.as_ref().map(|a| a.view()),
56            2 => self.v.as_ref().map(|a| a.view()),
57            _ => None,
58        }
59    }
60
61    fn plane_mask_view(&self, idx: usize) -> Option<ArrayViewD<'_, bool>> {
62        if let (Some(block), Some(rel)) = (&self.shared_mask, self.shared_offsets[idx]) {
63            return Some(block.slice_axis(Axis(self.stokes_axis), Slice::from(rel..rel + 1)));
64        }
65        match idx {
66            0 => self.mask_q.as_ref().map(|a| a.view()),
67            1 => self.mask_u.as_ref().map(|a| a.view()),
68            2 => self.mask_v.as_ref().map(|a| a.view()),
69            _ => None,
70        }
71    }
72}

74pub(super) fn run_pola(
75    request: &ImmathRequest,
76    outfile: &Path,
77    filenames: &[PathBuf],
78    varnames: &[String],
79) -> Result<ImageMathSummary, ImageError> {
80    let polithresh = request
81        .polithresh
82        .as_deref()
83        .map(str::trim)
84        .filter(|s| !s.is_empty());
85
86    // CASA ignores mask when polithresh is set in pola mode.
87    let effective_mask = if polithresh.is_some() {
88        None
89    } else {
90        request.mask.as_deref()
91    };
92
93    let inputs = gather_pol_inputs(request, filenames, varnames, effective_mask, false)?;
94    require_q_u(&inputs)?;
95
96    let q_view = inputs.plane_view(0).expect("Q checked");
97    let u_view = inputs.plane_view(1).expect("U checked");
98    let mq_view = inputs.plane_mask_view(0);
99    let mu_view = inputs.plane_mask_view(1);
100    let mo_view = if polithresh.is_none() {
101        inputs.mask_override.as_ref().map(|a| a.view())
102    } else {
103        None
104    };
105    let thresh = polithresh
106        .map(|s| parse_polithresh(s, &inputs.bunit))
107        .transpose()?;
108
109    let mut out_data = ArrayD::from_elem(IxDyn(&inputs.out_shape).f(), 0.0_f32);
110    let mut valid = ArrayD::from_elem(IxDyn(&inputs.out_shape).f(), true);
111    let total_len = out_data.len();
112
113    let false_count = match (split_axis_for(&inputs.out_shape, total_len), thresh) {
114        (Some(ax), Some(th)) => parallel_pola_chunks(
115            ax,
116            out_data.view_mut(),
117            valid.view_mut(),
118            q_view,
119            u_view,
120            mq_view,
121            mu_view,
122            None,
123            Some(th),
124        ),
125        (Some(ax), None) => parallel_pola_chunks(
126            ax,
127            out_data.view_mut(),
128            valid.view_mut(),
129            q_view,
130            u_view,
131            mq_view,
132            mu_view,
133            mo_view,
134            None,
135        ),
136        (None, _) => eval_pola_chunk(
137            out_data.view_mut(),
138            valid.view_mut(),
139            q_view,
140            u_view,
141            mq_view,
142            mu_view,
143            mo_view,
144            thresh,
145        ),
146    };
147    let valid_pixels = total_len - false_count;
148
149    save_pol_output(
150        request,
151        outfile,
152        filenames,
153        out_data,
154        valid,
155        valid_pixels,
156        &inputs,
157        StokesType::Pangle,
158        "deg",
159    )
160}

162#[allow(clippy::too_many_arguments)]
163fn parallel_pola_chunks(
164    ax: Axis,
165    mut out_data: ArrayViewMutD<'_, f32>,
166    mut valid: ArrayViewMutD<'_, bool>,
167    q: ArrayViewD<'_, f32>,
168    u: ArrayViewD<'_, f32>,
169    mq: Option<ArrayViewD<'_, bool>>,
170    mu: Option<ArrayViewD<'_, bool>>,
171    mo: Option<ArrayViewD<'_, bool>>,
172    thresh: Option<f64>,
173) -> usize {
174    let chunk = worker_chunk_size(out_data.shape()[ax.index()]);
175    std::thread::scope(|scope| {
176        let mut handles = Vec::new();
177        let out_it = out_data.axis_chunks_iter_mut(ax, chunk);
178        let valid_it = valid.axis_chunks_iter_mut(ax, chunk);
179        let q_it = q.axis_chunks_iter(ax, chunk);
180        let u_it = u.axis_chunks_iter(ax, chunk);
181        let mut mq_it = mq.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
182        let mut mu_it = mu.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
183        let mut mo_it = mo.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
184
185        for (((oc, vc), qc), uc) in out_it.zip(valid_it).zip(q_it).zip(u_it) {
186            let mqc = mq_it.as_mut().and_then(Iterator::next);
187            let muc = mu_it.as_mut().and_then(Iterator::next);
188            let moc = mo_it.as_mut().and_then(Iterator::next);
189            handles
190                .push(scope.spawn(move || eval_pola_chunk(oc, vc, qc, uc, mqc, muc, moc, thresh)));
191        }
192        handles
193            .into_iter()
194            .map(|h| h.join().expect("pola worker"))
195            .sum()
196    })
197}

199#[allow(clippy::too_many_arguments)]
200fn eval_pola_chunk(
201    mut out_c: ArrayViewMutD<'_, f32>,
202    mut valid_c: ArrayViewMutD<'_, bool>,
203    q_c: ArrayViewD<'_, f32>,
204    u_c: ArrayViewD<'_, f32>,
205    mq_c: Option<ArrayViewD<'_, bool>>,
206    mu_c: Option<ArrayViewD<'_, bool>>,
207    mo_c: Option<ArrayViewD<'_, bool>>,
208    thresh: Option<f64>,
209) -> usize {
210    let half_deg_per_rad = 90.0_f64 / std::f64::consts::PI;
211    let has_mask = apply_chunk_masks(&mut valid_c, [mq_c, mu_c, mo_c]);
212
213    let mut false_count = 0usize;
214    if let Some(th) = thresh {
215        let thresh_sq = th * th;
216        let use_sq = th > 1.0e-150 && th < 1.0e150 && thresh_sq.is_finite();
217        Zip::from(&mut out_c)
218            .and(&mut valid_c)
219            .and(&q_c)
220            .and(&u_c)
221            .for_each(|dst, vf, &qv, &uv| {
222                let qf = f64::from(qv);
223                let uf = f64::from(uv);
224                *dst = (half_deg_per_rad * uf.atan2(qf)) as f32;
225                let below = if qf.abs() >= th || uf.abs() >= th {
226                    false
227                } else if use_sq && qf.abs() < 1.0e100 && uf.abs() < 1.0e100 {
228                    qf * qf + uf * uf < thresh_sq
229                } else {
230                    qf.hypot(uf) < th
231                };
232                if below {
233                    *vf = false;
234                }
235                if !*vf {
236                    false_count += 1;
237                }
238            });
239    } else if has_mask {
240        Zip::from(&mut out_c)
241            .and(&valid_c)
242            .and(&q_c)
243            .and(&u_c)
244            .for_each(|dst, &vf, &qv, &uv| {
245                *dst = (half_deg_per_rad * f64::from(uv).atan2(f64::from(qv))) as f32;
246                if !vf {
247                    false_count += 1;
248                }
249            });
250    } else {
251        Zip::from(&mut out_c)
252            .and(&q_c)
253            .and(&u_c)
254            .for_each(|dst, &qv, &uv| {
255                *dst = (half_deg_per_rad * f64::from(uv).atan2(f64::from(qv))) as f32;
256            });
257    }
258    false_count
259}

261pub(super) fn run_poli(
262    request: &ImmathRequest,
263    outfile: &Path,
264    filenames: &[PathBuf],
265    varnames: &[String],
266    mode: &str,
267) -> Result<ImageMathSummary, ImageError> {
268    let need_v = mode != "lpoli";
269    let inputs = gather_pol_inputs(
270        request,
271        filenames,
272        varnames,
273        request.mask.as_deref(),
274        need_v,
275    )?;
276    let out_stokes = match mode {
277        "tpoli" => {
278            if !inputs.has_q {
279                return Err(ImageError::InvalidMetadata(
280                    "Stokes Q is required for mode tpoli but was not found".into(),
281                ));
282            }
283            if !inputs.has_u {
284                return Err(ImageError::InvalidMetadata(
285                    "Stokes U is required for mode tpoli but was not found".into(),
286                ));
287            }
288            if !inputs.has_v {
289                return Err(ImageError::InvalidMetadata(
290                    "Stokes V is required for mode tpoli but was not found".into(),
291                ));
292            }
293            StokesType::Ptotal
294        }
295        "lpoli" => {
296            require_q_u(&inputs)?;
297            StokesType::Plinear
298        }
299        "poli" => {
300            if inputs.has_v {
301                // CASA imagepol.totpolint computes Ptotal whenever V exists (even V-only).
302                if inputs.has_q ^ inputs.has_u {
303                    return Err(ImageError::InvalidMetadata(
304                        "This Stokes coordinate has only one of Q and U. This is not useful".into(),
305                    ));
306                }
307                StokesType::Ptotal
308            } else {
309                require_q_u(&inputs)?;
310                StokesType::Plinear
311            }
312        }
313        _ => unreachable!(),
314    };
315
316    let sigma_sq = parse_sigma_sq(&request.sigma, &inputs.bunit)?;
317    let mut out_data = ArrayD::from_elem(IxDyn(&inputs.out_shape).f(), 0.0_f32);
318    let mut valid = ArrayD::from_elem(IxDyn(&inputs.out_shape).f(), true);
319    let total_len = out_data.len();
320
321    let q_view = inputs.plane_view(0);
322    let u_view = inputs.plane_view(1);
323    let v_view = if out_stokes == StokesType::Ptotal {
324        inputs.plane_view(2)
325    } else {
326        None
327    };
328    let mq_view = inputs.plane_mask_view(0);
329    let mu_view = inputs.plane_mask_view(1);
330    let mv_view = if out_stokes == StokesType::Ptotal {
331        inputs.plane_mask_view(2)
332    } else {
333        None
334    };
335    let mo_view = inputs.mask_override.as_ref().map(|a| a.view());
336
337    let false_count = match split_axis_for(&inputs.out_shape, total_len) {
338        Some(ax) => parallel_poli_chunks(
339            ax,
340            out_data.view_mut(),
341            valid.view_mut(),
342            q_view,
343            u_view,
344            v_view,
345            mq_view,
346            mu_view,
347            mv_view,
348            mo_view,
349            sigma_sq,
350        ),
351        None => eval_poli_chunk(
352            out_data.view_mut(),
353            valid.view_mut(),
354            q_view,
355            u_view,
356            v_view,
357            mq_view,
358            mu_view,
359            mv_view,
360            mo_view,
361            sigma_sq,
362        ),
363    };
364    let valid_pixels = total_len - false_count;
365
366    let bunit = inputs.bunit.clone();
367    save_pol_output(
368        request,
369        outfile,
370        filenames,
371        out_data,
372        valid,
373        valid_pixels,
374        &inputs,
375        out_stokes,
376        &bunit,
377    )
378}

380#[allow(clippy::too_many_arguments)]
381fn parallel_poli_chunks(
382    ax: Axis,
383    mut out_data: ArrayViewMutD<'_, f32>,
384    mut valid: ArrayViewMutD<'_, bool>,
385    q: Option<ArrayViewD<'_, f32>>,
386    u: Option<ArrayViewD<'_, f32>>,
387    v: Option<ArrayViewD<'_, f32>>,
388    mq: Option<ArrayViewD<'_, bool>>,
389    mu: Option<ArrayViewD<'_, bool>>,
390    mv: Option<ArrayViewD<'_, bool>>,
391    mo: Option<ArrayViewD<'_, bool>>,
392    sigma_sq: Option<f64>,
393) -> usize {
394    let chunk = worker_chunk_size(out_data.shape()[ax.index()]);
395    std::thread::scope(|scope| {
396        let mut handles = Vec::new();
397        let out_it = out_data.axis_chunks_iter_mut(ax, chunk);
398        let valid_it = valid.axis_chunks_iter_mut(ax, chunk);
399        let mut q_it = q.as_ref().map(|a| a.axis_chunks_iter(ax, chunk));
400        let mut u_it = u.as_ref().map(|a| a.axis_chunks_iter(ax, chunk));
401        let mut v_it = v.as_ref().map(|a| a.axis_chunks_iter(ax, chunk));
402        let mut mq_it = mq.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
403        let mut mu_it = mu.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
404        let mut mv_it = mv.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
405        let mut mo_it = mo.as_ref().map(|m| m.axis_chunks_iter(ax, chunk));
406
407        for (oc, vc) in out_it.zip(valid_it) {
408            let qc = q_it.as_mut().and_then(Iterator::next);
409            let uc = u_it.as_mut().and_then(Iterator::next);
410            let v_chunk = v_it.as_mut().and_then(Iterator::next);
411            let mqc = mq_it.as_mut().and_then(Iterator::next);
412            let muc = mu_it.as_mut().and_then(Iterator::next);
413            let mvc = mv_it.as_mut().and_then(Iterator::next);
414            let moc = mo_it.as_mut().and_then(Iterator::next);
415            handles.push(scope.spawn(move || {
416                eval_poli_chunk(oc, vc, qc, uc, v_chunk, mqc, muc, mvc, moc, sigma_sq)
417            }));
418        }
419        handles
420            .into_iter()
421            .map(|h| h.join().expect("poli worker"))
422            .sum()
423    })
424}

426#[allow(clippy::too_many_arguments)]
427fn eval_poli_chunk(
428    mut out_c: ArrayViewMutD<'_, f32>,
429    mut valid_c: ArrayViewMutD<'_, bool>,
430    q_c: Option<ArrayViewD<'_, f32>>,
431    u_c: Option<ArrayViewD<'_, f32>>,
432    v_c: Option<ArrayViewD<'_, f32>>,
433    mq_c: Option<ArrayViewD<'_, bool>>,
434    mu_c: Option<ArrayViewD<'_, bool>>,
435    mv_c: Option<ArrayViewD<'_, bool>>,
436    mo_c: Option<ArrayViewD<'_, bool>>,
437    sigma_sq: Option<f64>,
438) -> usize {
439    let has_mask = apply_chunk_masks(&mut valid_c, [mq_c, mu_c, mv_c, mo_c]) || sigma_sq.is_some();
440
441    let mut false_count = 0usize;
442    match (q_c, u_c, v_c) {
443        (Some(qa), Some(ua), Some(va)) => {
444            if has_mask {
445                Zip::from(&mut out_c)
446                    .and(&mut valid_c)
447                    .and(&qa)
448                    .and(&ua)
449                    .and(&va)
450                    .for_each(|dst, vf, &qv, &uv, &vv| {
451                        let qf = f64::from(qv);
452                        let uf = f64::from(uv);
453                        let vf_val = f64::from(vv);
454                        assign_debiased(dst, vf, qf * qf + uf * uf + vf_val * vf_val, sigma_sq);
455                        if !*vf {
456                            false_count += 1;
457                        }
458                    });
459            } else {
460                Zip::from(&mut out_c)
461                    .and(&qa)
462                    .and(&ua)
463                    .and(&va)
464                    .for_each(|dst, &qv, &uv, &vv| {
465                        let qf = f64::from(qv);
466                        let uf = f64::from(uv);
467                        let vf_val = f64::from(vv);
468                        *dst = (qf * qf + uf * uf + vf_val * vf_val).sqrt() as f32;
469                    });
470            }
471        }
472        (None, None, Some(va)) => {
473            if has_mask {
474                Zip::from(&mut out_c)
475                    .and(&mut valid_c)
476                    .and(&va)
477                    .for_each(|dst, vf, &vv| {
478                        let vf_val = f64::from(vv);
479                        assign_debiased(dst, vf, vf_val * vf_val, sigma_sq);
480                        if !*vf {
481                            false_count += 1;
482                        }
483                    });
484            } else {
485                Zip::from(&mut out_c).and(&va).for_each(|dst, &vv| {
486                    *dst = f64::from(vv).abs() as f32;
487                });
488            }
489        }
490        (Some(qa), Some(ua), None) => {
491            if has_mask {
492                Zip::from(&mut out_c)
493                    .and(&mut valid_c)
494                    .and(&qa)
495                    .and(&ua)
496                    .for_each(|dst, vf, &qv, &uv| {
497                        let qf = f64::from(qv);
498                        let uf = f64::from(uv);
499                        assign_debiased(dst, vf, qf * qf + uf * uf, sigma_sq);
500                        if !*vf {
501                            false_count += 1;
502                        }
503                    });
504            } else {
505                Zip::from(&mut out_c)
506                    .and(&qa)
507                    .and(&ua)
508                    .for_each(|dst, &qv, &uv| {
509                        let qf = f64::from(qv);
510                        let uf = f64::from(uv);
511                        *dst = (qf * qf + uf * uf).sqrt() as f32;
512                    });
513            }
514        }
515        _ => unreachable!(),
516    }
517    false_count
518}

520fn split_axis_for(shape: &[usize], total_len: usize) -> Option<Axis> {
521    if total_len < 65_536 {
522        return None;
523    }
524    shape
525        .iter()
526        .rposition(|&d| d > 1)
527        .filter(|&ax| shape[ax] >= 2)
528        .map(Axis)
529}

531fn worker_chunk_size(axis_len: usize) -> usize {
532    let workers = std::thread::available_parallelism()
533        .map_or(1, usize::from)
534        .clamp(1, 8)
535        .min(axis_len);
536    axis_len.div_ceil(workers)
537}

539fn apply_chunk_masks<const N: usize>(
540    valid_c: &mut ArrayViewMutD<'_, bool>,
541    masks: [Option<ArrayViewD<'_, bool>>; N],
542) -> bool {
543    let mut any = false;
544    for mask in masks.into_iter().flatten() {
545        any = true;
546        Zip::from(&mut *valid_c)
547            .and(&mask)
548            .for_each(|a, &b| *a = *a && b);
549    }
550    any
551}

553#[inline]
554fn assign_debiased(dst: &mut f32, valid: &mut bool, p_sq: f64, sigma_sq: Option<f64>) {
555    if let Some(s_sq) = sigma_sq {
556        let diff = p_sq - s_sq;
557        if diff >= 0.0 {
558            *dst = diff.sqrt() as f32;
559        } else {
560            *dst = 0.0;
561            *valid = false;
562        }
563    } else {
564        *dst = p_sq.sqrt() as f32;
565    }
566}

568fn require_q_u(inputs: &PolInputs) -> Result<(), ImageError> {
569    if inputs.has_q && inputs.has_u {
570        return Ok(());
571    }
572    if inputs.has_q ^ inputs.has_u {
573        return Err(ImageError::InvalidMetadata(
574            "This Stokes coordinate has only one of Q and U. This is not useful".into(),
575        ));
576    }
577    Err(ImageError::InvalidMetadata(
578        "This image has no Stokes Q, U, nor V.  This is not useful".into(),
579    ))
580}

582fn parse_polithresh(raw: &str, bunit: &str) -> Result<f64, ImageError> {
583    let q = Quantity::read(raw)
584        .map_err(|_| ImageError::InvalidMetadata(format!("Invalid polithresh quantity: {raw}")))?;
585    let init_unit = q.unit().name().to_string();
586    if init_unit.is_empty() {
587        return Ok(q.value());
588    }
589    if bunit.is_empty() {
590        return Err(ImageError::InvalidMetadata(format!(
591            "Units of polithresh {init_unit} do not conform to input image units of {bunit} so cannot perform thresholding. Please correct units and try again."
592        )));
593    }
594    let target = casacore_unit(bunit).map_err(|_| {
595        ImageError::InvalidMetadata(format!(
596            "Units of polithresh {init_unit} do not conform to input image units of {bunit} so cannot perform thresholding. Please correct units and try again."
597        ))
598    })?;
599    get_conformant_value(&q, &target).map_err(|_| {
600        ImageError::InvalidMetadata(format!(
601            "Units of polithresh {init_unit} do not conform to input image units of {bunit} so cannot perform thresholding. Please correct units and try again."
602        ))
603    })
604}

606fn parse_sigma_sq(raw: &str, bunit: &str) -> Result<Option<f64>, ImageError> {
607    let trimmed = raw.trim();
608    if trimmed.is_empty() {
609        return Ok(None);
610    }
611    let q = Quantity::read(trimmed)
612        .map_err(|_| ImageError::InvalidMetadata(format!("Invalid sigma quantity: {trimmed}")))?;
613    if q.value() <= 0.0 {
614        return Ok(None);
615    }
616    let init_unit = q.unit().name().to_string();
617    let sigma_val = if init_unit.is_empty() {
618        if bunit.is_empty() {
619            q.value()
620        } else {
621            return Err(ImageError::InvalidMetadata(format!(
622                "Units of sigma {init_unit:?} do not conform to input image units of {bunit}"
623            )));
624        }
625    } else if bunit.is_empty() {
626        return Err(ImageError::InvalidMetadata(format!(
627            "Units of sigma {init_unit} do not conform to input image units of {bunit}"
628        )));
629    } else {
630        let target = casacore_unit(bunit).map_err(|_| {
631            ImageError::InvalidMetadata(format!(
632                "Units of sigma {init_unit} do not conform to input image units of {bunit}"
633            ))
634        })?;
635        get_conformant_value(&q, &target).map_err(|_| {
636            ImageError::InvalidMetadata(format!(
637                "Units of sigma {init_unit} do not conform to input image units of {bunit}"
638            ))
639        })?
640    };
641    Ok(Some(sigma_val * sigma_val))
642}

644fn gather_pol_inputs(
645    request: &ImmathRequest,
646    filenames: &[PathBuf],
647    varnames: &[String],
648    mask_expr: Option<&str>,
649    need_v: bool,
650) -> Result<PolInputs, ImageError> {
651    let mut raws: Vec<RawImage<f32>> = Vec::with_capacity(filenames.len());
652    for p in filenames {
653        raws.push(open_raw_image::<f32>(p)?);
654    }
655    let first = raws[0].as_interface();
656    let bunit = first.units().to_string();
657    let stokes_axis = first
658        .find_axis(CoordinateType::Stokes)
659        .ok_or_else(|| ImageError::InvalidMetadata("This image has no Stokes coordinate".into()))?;
660
661    let mut q_loc: Option<(usize, usize)> = None;
662    let mut u_loc: Option<(usize, usize)> = None;
663    let mut v_loc: Option<(usize, usize)> = None;
664
665    for (file_idx, raw) in raws.iter().enumerate() {
666        let iface = raw.as_interface();
667        let axis = iface.find_axis(CoordinateType::Stokes).ok_or_else(|| {
668            ImageError::InvalidMetadata("All input images must have a Stokes axis".into())
669        })?;
670        if axis != stokes_axis || !non_stokes_match(first.shape(), iface.shape(), stokes_axis) {
671            return Err(ImageError::ShapeMismatch {
672                expected: first.shape().to_vec(),
673                got: iface.shape().to_vec(),
674            });
675        }
676        let stokes_list = stokes_types_on_axis(iface.coordinates(), axis)?;
677        for (plane_idx, st) in stokes_list.into_iter().enumerate() {
678            match st {
679                StokesType::Q if q_loc.is_none() => q_loc = Some((file_idx, plane_idx)),
680                StokesType::U if u_loc.is_none() => u_loc = Some((file_idx, plane_idx)),
681                StokesType::V if v_loc.is_none() => v_loc = Some((file_idx, plane_idx)),
682                _ => {}
683            }
684        }
685    }
686
687    // Polarization modes ignore stokes= parameter.
688    let sel = ImmathSelection::from_raw(&raws[0], request, true)?;
689    let mut out_shape = sel.shape.clone();
690    out_shape[stokes_axis] = 1;
691
692    let mut aliases: Vec<(String, &dyn ImageInterface<f32>)> = Vec::new();
693    for (idx, raw) in raws.iter().enumerate() {
694        if let Some(vn) = varnames.get(idx) {
695            aliases.push((vn.clone(), raw.as_interface()));
696        } else {
697            aliases.push((format!("IM{idx}"), raw.as_interface()));
698        }
699        aliases.push((filenames[idx].display().to_string(), raw.as_interface()));
700    }
701    let mask_override = if let Some(expr) = mask_expr.filter(|s| !s.trim().is_empty()) {
702        let mut w_start = sel.start.clone();
703        let mut w_shape = sel.shape.clone();
704        w_start[stokes_axis] = 0;
705        w_shape[stokes_axis] = 1;
706        evaluate_immath_mask(
707            Some(expr),
708            first.shape(),
709            &aliases,
710            request.stretch,
711            Some((&w_start, &w_shape)),
712        )?
713    } else {
714        None
715    };
716
717    let v_needed_loc = if need_v { v_loc } else { None };
718    let needed = [q_loc, u_loc, v_needed_loc];
719    let shared_file = shared_file_index(&needed);
720
721    let (q, u, v, mask_q, mask_u, mask_v, shared_block, shared_mask, shared_offsets) =
722        if let Some(fidx) = shared_file {
723            let min_p = needed
724                .iter()
725                .flatten()
726                .map(|&(_, p)| p)
727                .min()
728                .expect("min_p");
729            let max_p = needed
730                .iter()
731                .flatten()
732                .map(|&(_, p)| p)
733                .max()
734                .expect("max_p");
735            let iface = raws[fidx].as_interface();
736            let mut span_sel = sel.clone();
737            span_sel.start[stokes_axis] = min_p;
738            span_sel.shape[stokes_axis] = max_p - min_p + 1;
739            span_sel.stokes_axis = None;
740            span_sel.stokes_indices = None;
741            let stride = vec![1; iface.ndim()];
742            let block = iface.get_slice(&span_sel.start, &span_sel.shape, &stride)?;
743            let block_mask = super::selection::selection_and_mask(iface, &span_sel, None)?;
744            let offsets = [
745                q_loc.map(|(_, p)| p - min_p),
746                u_loc.map(|(_, p)| p - min_p),
747                v_needed_loc.map(|(_, p)| p - min_p),
748            ];
749            (
750                None,
751                None,
752                None,
753                None,
754                None,
755                None,
756                Some(block),
757                block_mask,
758                offsets,
759            )
760        } else {
761            let (q, mask_q) = extract_stokes_plane(&raws, q_loc, stokes_axis, &sel)?;
762            let (u, mask_u) = extract_stokes_plane(&raws, u_loc, stokes_axis, &sel)?;
763            let (v, mask_v) = extract_stokes_plane(&raws, v_needed_loc, stokes_axis, &sel)?;
764            (q, u, v, mask_q, mask_u, mask_v, None, None, [None; 3])
765        };
766
767    let ref_coords = first.coordinates().clone();
768    let ref_image_info = first.image_info()?;
769    let ref_misc_info = first.misc_info();
770    let mut sub_start = sel.start.clone();
771    sub_start[stokes_axis] = 0;
772    let mut sub_shape = sel.shape.clone();
773    sub_shape[stokes_axis] = 1;
774    let ref_legacy_coords = match &raws[0] {
775        RawImage::Paged { image, .. } => {
776            legacy_subimage_coordinates_with_stokes(image, &sub_start, &sub_shape, None)
777        }
778        RawImage::Memory(_) => None,
779    };
780
781    Ok(PolInputs {
782        stokes_axis,
783        bunit,
784        q,
785        u,
786        v,
787        mask_q,
788        mask_u,
789        mask_v,
790        shared_block,
791        shared_mask,
792        shared_offsets,
793        has_q: q_loc.is_some(),
794        has_u: u_loc.is_some(),
795        has_v: v_loc.is_some(),
796        mask_override,
797        ref_coords,
798        ref_image_info,
799        ref_misc_info,
800        ref_legacy_coords,
801        out_shape,
802        sel,
803    })
804}

806fn shared_file_index(locs: &[Option<(usize, usize)>; 3]) -> Option<usize> {
807    let mut count = 0usize;
808    let mut file = None;
809    for loc in locs.iter().flatten() {
810        count += 1;
811        match file {
812            None => file = Some(loc.0),
813            Some(prev) if prev != loc.0 => return None,
814            Some(_) => {}
815        }
816    }
817    if count >= 2 { file } else { None }
818}

820fn non_stokes_match(a: &[usize], b: &[usize], stokes_axis: usize) -> bool {
821    a.len() == b.len()
822        && a.iter()
823            .zip(b.iter())
824            .enumerate()
825            .all(|(ax, (&da, &db))| ax == stokes_axis || da == db)
826}
827
828fn stokes_types_on_axis(
829    coords: &casa_coordinates::CoordinateSystem,
830    axis: usize,
831) -> Result<Vec<StokesType>, ImageError> {
832    if let Some((index, _)) = coords.pixel_axis_owner(axis)
833        && let CoordinateModel::Stokes(stokes) = coords.coordinate(index)
834    {
835        return Ok(stokes.stokes().to_vec());
836    }
837    Err(ImageError::InvalidMetadata(
838        "missing Stokes coordinate on Stokes axis".into(),
839    ))
840}
841
842type PolInputSlice = (Option<ArrayD<f32>>, Option<ArrayD<bool>>);
843
844fn extract_stokes_plane(
845    raws: &[RawImage<f32>],
846    loc: Option<(usize, usize)>,
847    stokes_axis: usize,
848    sel: &ImmathSelection,
849) -> Result<PolInputSlice, ImageError> {
850    let Some((file_idx, plane_idx)) = loc else {
851        return Ok((None, None));
852    };
853    let iface = raws[file_idx].as_interface();
854    let mut plane_sel = sel.clone();
855    plane_sel.start[stokes_axis] = plane_idx;
856    plane_sel.shape[stokes_axis] = 1;
857    plane_sel.stokes_axis = None;
858    plane_sel.stokes_indices = None;
859    let stride = vec![1; iface.ndim()];
860    let data = iface.get_slice(&plane_sel.start, &plane_sel.shape, &stride)?;
861    let mask = super::selection::selection_and_mask(iface, &plane_sel, None)?;
862    Ok((Some(data), mask))
863}

865#[allow(clippy::too_many_arguments)]
866fn save_pol_output(
867    request: &ImmathRequest,
868    outfile: &Path,
869    filenames: &[PathBuf],
870    data: ArrayD<f32>,
871    valid: ArrayD<bool>,
872    valid_pixels: usize,
873    inputs: &PolInputs,
874    out_stokes: StokesType,
875    bunit: &str,
876) -> Result<ImageMathSummary, ImageError> {
877    let mut sub_start = inputs.sel.start.clone();
878    sub_start[inputs.stokes_axis] = 0;
879    let mut sub_shape = inputs.sel.shape.clone();
880    sub_shape[inputs.stokes_axis] = 1;
881
882    let out_coords = subimage_coordinates_with_stokes(
883        &inputs.ref_coords,
884        &sub_start,
885        &sub_shape,
886        Some(&[out_stokes]),
887    )?;
888    let has_false = valid_pixels != valid.len();
889
890    super::remove_existing_outfile_if_overwrite(outfile, request.overwrite)?;
891
892    let mut image = PagedImage::<f32>::create(inputs.out_shape.clone(), out_coords, outfile)?;
893    image.set_units(bunit.to_string())?;
894    image.set_image_info(&inputs.ref_image_info)?;
895    image.set_misc_info(inputs.ref_misc_info.clone())?;
896    if let Some(mut lc) = inputs.ref_legacy_coords.clone() {
897        let _ = replace_stokes_in_record(&mut lc, &[out_stokes]);
898        image.set_coords_record_keyword(lc);
899    }
900    image.put_slice(&data, &vec![0; inputs.out_shape.len()])?;
901    if has_false {
902        image.put_mask("mask0", &valid)?;
903        image.set_default_mask("mask0")?;
904    }
905    image.save()?;
906
907    Ok(ImageMathSummary {
908        imagename: filenames.iter().map(|p| p.display().to_string()).collect(),
909        outfile: outfile.display().to_string(),
910        expr: request.expr.clone(),
911        shape: inputs.out_shape.clone(),
912        units: bunit.to_string(),
913        valid_pixels,
914        beam_warnings: Vec::new(),
915    })
916}
