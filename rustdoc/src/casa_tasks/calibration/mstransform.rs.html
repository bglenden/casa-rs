22/// Front end of the `mstransform` binary, hosting `mstransform` and `split`.
23pub fn mstransform_front_end()
24-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
25    TaskFrontEnd::new_with_deferred_schema(
26        "mstransform",
27        mstransform_task_schema_bundle(),
28        &["mstransform", "split"],
29        run,
30    )
31    .expect("the transform contract carries mstransform and split")
32}

34/// Run the shared transform and project the public task's return value.
35pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
36    let (request, mut options) = transform_request(task)?;
37    if request.input_ms != request.output_ms && !request.output_ms.exists() {
38        let calibration = callibrary(task)?;
39        // CASA builds a continuum iterator in place of the calibration
40        // iterator when both task switches are enabled.
41        if options.continuum.is_none() {
42            options.input_view = calibration;
43        }
44        if task.surface() == "mstransform"
45            && matches!(
46                request.data_column,
47                TransformDataColumn::ModelData
48                    | TransformDataColumn::DataModelCorrected
49                    | TransformDataColumn::All
50            )
51        {
52            options.input_view = Some(transform_model_input(
53                options.input_view.take(),
54                task.flag("realmodelcol")?,
55            ));
56        }
57    }
58    // The output the transform refuses before it starts is the argument's
59    // fault; anything else failed the transform.
60    let report = mstransform_task(&request, &options).map_err(|error| match error {
61        MsTransformError::OutputExists { .. } | MsTransformError::SameInputOutput { .. } => {
62            TaskFailure::validation(error.to_string()).at("outputvis")
63        }
64        MsTransformError::InvalidSpw { .. } => TaskFailure::validation(error.to_string()).at("spw"),
65        MsTransformError::MissingDataColumn { path, column } => TaskFailure::execution(format!(
66            "Desired column ({column}) not found in the input MS ({path})."
67        )),
68        MsTransformError::RegridParameter { parameter, reason }
69        | MsTransformError::MultiMsParameter { parameter, reason } => {
70            TaskFailure::validation(reason).at(parameter)
71        }
72        MsTransformError::ChannelWidthCount => {
73            let parameter = channel_parameter(task);
74            TaskFailure::validation(format!(
75                "Number of {parameter} is different from the number of spw"
76            ))
77            .at(parameter)
78        }
79        other => TaskFailure::execution(other.to_string()),
80    })?;
81    let result = report
82        .retained_mms
83        .map_or(CasaValue::None, |value| CasaValue::Bool { value });
84    ms_task_success(result, &report)
85}

87fn transform_request(
88    task: &TaskInvocation<'_>,
89) -> Result<(MsTransformRequest, MsTransformTaskOptions), TaskFailure> {
90    let (width, channel_widths) = channel_width(task)?;
91    let request = MsTransformRequest {
92        input_ms: task.path("vis")?,
93        output_ms: task.path("outputvis")?,
94        spw: variant_selector(task, "spw")?.unwrap_or_default(),
95        width,
96        data_column: data_column(&task.required_text("datacolumn")?)?,
97        selection: selection(task)?,
98        keep_flags: task.flag("keepflags")?,
99    };
100    let (keep_mms, retained_scan_span) = if task.surface() == "split" {
101        let keep_mms = task.flag("keepmms")?;
102        let combine_scan =
103            matches!(task.value("combine"), Some(ParameterValue::String(span)) if span == "scan");
104        let span = if keep_mms && combine_scan {
105            Quantity::read(&task.required_text("timebin")?)
106                .ok()
107                .map(|quantity| quantity.value())
108        } else {
109            None
110        };
111        (Some(keep_mms), span)
112    } else {
113        (None, None)
114    };
115    let options = MsTransformTaskOptions {
116        use_weight_spectrum: task.surface() == "mstransform" && task.flag("usewtspectrum")?,
117        hanning: task.surface() == "mstransform" && task.flag("hanning")?,
118        combine_spws: task.surface() == "mstransform" && task.flag("combinespws")?,
119        continuum: continuum_options(task)?,
120        regrid: regrid_options(task)?,
121        tiling: tile_layout(task)?,
122        preserve_ids: task.surface() == "mstransform" && !task.flag("reindex")?,
123        ddi_start: task
124            .value("ddistart")
125            .and_then(channel_integer)
126            .and_then(|value| usize::try_from(value).ok())
127            .unwrap_or_default(),
128        channel_widths,
129        time_averaging: time_average_options(task)?,
130        history: Some(task_history(task)),
131        flag_command_widths: flag_command_widths(task),
132        keep_mms,
133        retained_scan_span,
134        multi_ms: multi_ms_options(task)?,
135        ..MsTransformTaskOptions::default()
136    };
137    Ok((request, options))
138}

140fn continuum_options(
141    task: &TaskInvocation<'_>,
142) -> Result<Option<ContinuumFitOptions>, TaskFailure> {
143    if task.surface() != "mstransform" || !task.flag("douvcontsub")? {
144        return Ok(None);
145    }
146    let nonnegative = |name| {
147        usize::try_from(task.integer(name)?).map_err(|_| {
148            TaskFailure::execution(format!("{name} must be nonnegative for continuum fitting"))
149                .at(name)
150        })
151    };
152    Ok(Some(ContinuumFitOptions {
153        fit_spw: task.text("fitspw")?.unwrap_or_default(),
154        order: nonnegative("fitorder")?,
155        want_continuum: task.flag("want_cont")?,
156        reweight: task.flag("denoising_lib")?,
157        iterations: nonnegative("niter")?,
158        threads: usize::try_from(task.integer("nthreads")?).unwrap_or(usize::MAX),
159    }))
160}

162fn callibrary(
163    task: &TaskInvocation<'_>,
164) -> Result<Option<std::sync::Arc<dyn casa_mstransform::TransformInputView>>, TaskFailure> {
165    if task.surface() != "mstransform" || !task.flag("docallib")? {
166        return Ok(None);
167    }
168    let filename = task.text("callib")?.unwrap_or_default();
169    transform_callibrary(&task.io_path(&filename), task.workspace()).map_err(|error| {
170        let reason = match &error {
171            CallibError::ReadFile { source, .. } if source.kind() == ErrorKind::NotFound => {
172                "No such file or directory".into()
173            }
174            CallibError::ParseLine { line, .. } => {
175                format!("Problem parsing cal library line (check for typos): {line}")
176            }
177            _ => error.to_string(),
178        };
179        TaskFailure::validation(reason).at("callib")
180    })
181}

183fn multi_ms_options(
184    task: &TaskInvocation<'_>,
185) -> Result<Option<MsTransformMultiMsOptions>, TaskFailure> {
186    if task.surface() != "mstransform" {
187        return Ok(None);
188    }
189    let create = task.flag("createmms")?;
190    let axis = match task.required_text("separationaxis")?.as_str() {
191        "scan" | "SCAN" => MsPartitionAxis::Scan,
192        "spw" | "SPW" => MsPartitionAxis::SpectralWindow,
193        "auto" | "AUTO" => MsPartitionAxis::Automatic,
194        "baseline" | "BASELINE" => MsPartitionAxis::Baseline,
195        other => {
196            return Err(
197                TaskFailure::validation(format!("unallowed value {other}")).at("separationaxis")
198            );
199        }
200    };
201    let count = match task.value("numsubms") {
202        Some(ParameterValue::String(text)) if text == "auto" => None,
203        Some(value) => {
204            let integer = channel_integer(value);
205            let count = integer
206                .and_then(|n| usize::try_from(n).ok())
207                .and_then(NonZeroUsize::new);
208            if create && count.is_none() {
209                let message = match integer {
210                    Some(0) => "float division by zero",
211                    Some(_) => "numsubms must be positive.",
212                    None => "'<' not supported between instances of 'str' and 'int'",
213                };
214                return Err(TaskFailure::validation(message).at("numsubms"));
215            }
216            count
217        }
218        None => None,
219    };
220    Ok(Some(MsTransformMultiMsOptions {
221        create,
222        axis,
223        count,
224        monolithic: task.flag("monolithic_processing")?,
225        scan_span: matches!(task.value("timespan"), Some(ParameterValue::String(value)) if value == "scan"),
226        multiple_spws: task
227            .value("nspw")
228            .and_then(channel_integer)
229            .is_some_and(|count| count > 1),
230    }))
231}

233fn tile_layout(task: &TaskInvocation<'_>) -> Result<Option<MsTransformTiling>, TaskFailure> {
234    if task.surface() != "mstransform" {
235        return Ok(None);
236    }
237    let invalid = |reason| TaskFailure::validation(reason).at("tileshape");
238    let Some(ParameterValue::Array(values)) = task.value("tileshape") else {
239        return Err(invalid("Parameter tileshape must have 1 or 3 elements."));
240    };
241    let values = values
242        .iter()
243        .map(|value| {
244            channel_integer(value).ok_or_else(|| invalid("Tile extents must be integers."))
245        })
246        .collect::<Result<Vec<_>, _>>()?;
247    let result = match values.as_slice() {
248        [0] => MsTransformTiling::Standard,
249        [1] => MsTransformTiling::FastMosaic,
250        [_] => {
251            return Err(invalid(
252                "When tileshape has one element, it should be either 0 or 1.",
253            ));
254        }
255        &[corr, chan, rows]
256            if [corr, chan, rows]
257                .into_iter()
258                .all(|n| (0..=i64::from(i32::MAX)).contains(&n)) =>
259        {
260            MsTransformTiling::Explicit([corr as usize, chan as usize, rows as usize])
261        }
262        [_, _, _] => return Err(invalid("Tile extents must be nonnegative 32-bit integers.")),
263        _ => return Err(invalid("Parameter tileshape must have 1 or 3 elements.")),
264    };
265    Ok(Some(result))
266}

268fn regrid_options(task: &TaskInvocation<'_>) -> Result<Option<MsRegridOptions>, TaskFailure> {
269    if task.surface() != "mstransform" || !task.flag("regridms")? {
270        return Ok(None);
271    }
272    let integer = |name| {
273        task.value(name)
274            .and_then(channel_integer)
275            .ok_or_else(|| TaskFailure::validation("expected an integer").at(name))
276    };
277    let variant = |name| {
278        task.value(name)
279            .and_then(variant_selector_text)
280            .unwrap_or_default()
281    };
282    Ok(Some(MsRegridOptions {
283        mode: task.required_text("mode")?,
284        nchan: integer("nchan")?,
285        start: variant("start"),
286        width: variant("width"),
287        nspw: usize::try_from(integer("nspw")?)
288            .ok()
289            .filter(|&count| count > 0)
290            .ok_or_else(|| {
291                TaskFailure::validation("expected a positive number of windows").at("nspw")
292            })?,
293        interpolation: task.required_text("interpolation")?,
294        phasecenter: variant("phasecenter"),
295        restfreq: task.required_text("restfreq")?,
296        outframe: task.required_text("outframe")?,
297        veltype: task.required_text("veltype")?,
298        preaverage: task.flag("preaverage")?,
299    }))
300}

302fn task_history(task: &TaskInvocation<'_>) -> TransformTaskHistory {
303    let mut bindings = task
304        .session()
305        .bundle()
306        .surface
307        .bindings()
308        .iter()
309        .filter(|binding| {
310            binding
311                .projections
312                .python
313                .as_ref()
314                .is_some_and(|python| !python.keyword_only)
315        })
316        .collect::<Vec<_>>();
317    bindings.sort_by_key(|binding| binding.order);
318    let parameters = bindings
319        .into_iter()
320        .filter_map(|binding| {
321            let value = task
322                .session()
323                .states()
324                .get(&binding.name)
325                .and_then(|state| state.value.as_ref())
326                .or(match &binding.default {
327                    DefaultSpec::Literal { value } => Some(value),
328                    DefaultSpec::Conditional { fallback, .. } => Some(fallback),
329                    DefaultSpec::Required => None,
330                })?;
331            let value = if matches!(binding.name.as_str(), "vis" | "outputvis")
332                && let ParameterValue::Array(paths) = value
333                && let [path] = paths.as_slice()
334            {
335                path
336            } else {
337                value
338            };
339            let printable = match value {
340                ParameterValue::String(value)
341                    if value == "none"
342                        && matches!(
343                            binding.name.as_str(),
344                            "field" | "spw" | "scan" | "antenna" | "timerange"
345                        ) =>
346                {
347                    r#""""#.into()
348                }
349                ParameterValue::String(value) => format!("\"{value}\""),
350                ParameterValue::Array(values)
351                    if task.surface() == "split" && binding.name == "correlation" =>
352                {
353                    let labels = values
354                        .iter()
355                        .filter_map(|label| match label {
356                            ParameterValue::String(text) => Some(text.to_uppercase()),
357                            _ => None,
358                        })
359                        .collect::<Vec<_>>();
360                    format!("\"{}\"", labels.join(", "))
361                }
362                ParameterValue::ScalarArray(value)
363                    if matches!(
364                        binding.name.as_str(),
365                        "field" | "spw" | "scan" | "antenna" | "timerange"
366                    ) =>
367                {
368                    match value.as_ref() {
369                        ParameterValue::String(text) => text.clone(),
370                        value => python_repr(value),
371                    }
372                }
373                ParameterValue::NumpyScalar { value, .. } => python_repr(value),
374                ParameterValue::EmptyArray { .. } => "[]".into(),
375                value => python_repr(value),
376            };
377            Some((binding.name.clone(), printable))
378        })
379        .collect();
380    TransformTaskHistory {
381        task: task.surface().to_string(),
382        parameters,
383    }
384}

386fn time_average_options(
387    task: &TaskInvocation<'_>,
388) -> Result<Option<TimeAverageOptions>, TaskFailure> {
389    let split = task.surface() == "split";
390    if !split && !task.flag("timeaverage")? {
391        return Ok(None);
392    }
393    let interval = task.required_text("timebin")?;
394    let quantity = Quantity::read(&interval).map_err(|_| {
395        TaskFailure::execution(format!(
396            "Error in QuantumHolder::fromString with input string {interval:?}: Illegal input units or format"
397        )).at("timebin")
398    })?;
399    let seconds = Unit::new("s").expect("seconds are a defined unit");
400    let radians = Unit::new("rad").expect("radians are a defined unit");
401    let interval_seconds = quantity
402        .get_value_in(&seconds)
403        .or_else(|_| {
404            quantity
405                .get_value_in(&radians)
406                .map(|angle| angle * 43_200.0 / std::f64::consts::PI)
407        })
408        .unwrap_or_else(|_| quantity.get_si_value());
409    if interval_seconds <= 0.0 || !interval_seconds.is_finite() {
410        return if split {
411            Ok(None)
412        } else {
413            Err(
414                TaskFailure::validation("Parameter timebin must be > '0s' to do time averaging")
415                    .at("timebin"),
416            )
417        };
418    }
419    let parameter = if split { "combine" } else { "timespan" };
420    let span = match task.value(parameter) {
421        None => "",
422        Some(ParameterValue::String(span)) => span,
423        Some(_) => return Err(TaskFailure::execution(
424            "RecordRep::get_pointer - incorrect data type String   used for field timespan with type unknown",
425        ).at(parameter)),
426    };
427    Ok(Some(TimeAverageOptions {
428        interval_seconds,
429        span_scans: span.contains("scan"),
430        span_states: span.contains("state"),
431        span_fields: span.contains("field"),
432        max_uvw_distance: if split {
433            0.0
434        } else {
435            task.number("maxuvwdistance")?
436        },
437    }))
438}

448fn channel_width(task: &TaskInvocation<'_>) -> Result<(usize, Option<Vec<usize>>), TaskFailure> {
449    let name = channel_parameter(task);
450    let split = name == "width";
451    if !split && !task.flag("chanaverage")? {
452        return Ok((1, None));
453    }
454    let invalid_type = || TaskFailure::validation(format!("invalid {name} type")).at(name);
455    let empty = || {
456        let message = if split {
457            "list index out of range"
458        } else {
459            "chanbin must contain at least one width"
460        };
461        TaskFailure::validation(message).at(name)
462    };
463    let value = task.value(name).ok_or_else(invalid_type)?;
464    match value {
465        ParameterValue::Array(values) if values.is_empty() => Err(empty()),
466        ParameterValue::Array(values) => {
467            let widths = values
468                .iter()
469                .map(|value| channel_integer(value).ok_or_else(invalid_type))
470                .collect::<Result<Vec<_>, _>>()?;
471            if split && !widths.iter().any(|&width| width > 1) {
472                return Ok((1, None));
473            }
474            let widths = widths
475                .into_iter()
476                .map(|width| {
477                    usize::try_from(width)
478                        .ok()
479                        .filter(|&width| width > 0)
480                        .ok_or_else(|| {
481                            TaskFailure::validation(format!(
482                                "{name} must contain positive integers"
483                            ))
484                            .at(name)
485                        })
486                })
487                .collect::<Result<Vec<_>, _>>()?;
488            Ok((widths[0], Some(widths)))
489        }
490        ParameterValue::EmptyArray { .. } => Err(empty()),
491        ParameterValue::String(text) if split => {
492            let width = if text.bytes().all(|digit| digit.is_ascii_digit()) {
493                text.parse::<usize>().unwrap_or(1)
494            } else {
495                1
496            };
497            Ok((width.max(1), None))
498        }
499        value => {
500            // CLI flags for variant parameters are passed as text. The
501            // Python task validates the accepted argument types separately.
502            let width = match value {
503                ParameterValue::String(text) => text.parse().ok(),
504                value => channel_integer(value),
505            }
506            .ok_or_else(invalid_type)?;
507            if !split && width <= 1 && !matches!(value, ParameterValue::NumpyScalar { .. }) {
508                return Err(TaskFailure::validation(
509                    "Parameter chanbin must be > 1 to do channel averaging",
510                )
511                .at(name));
512            }
513            Ok((usize::try_from(width).unwrap_or(1).max(1), None))
514        }
515    }
516}

518fn flag_command_widths(task: &TaskInvocation<'_>) -> Option<Vec<i64>> {
519    let value = task.value(channel_parameter(task))?;
520    match value {
521        ParameterValue::Array(widths) => widths.iter().map(channel_integer).collect(),
522        value if task.surface() == "mstransform" => channel_integer(value).map(|width| vec![width]),
523        _ => None,
524    }
525}
526
527fn channel_integer(value: &ParameterValue) -> Option<i64> {
528    match value {
529        ParameterValue::Integer(value) => Some(*value),
530        ParameterValue::Bool(value) => Some(i64::from(*value)),
531        ParameterValue::NumpyScalar {
532            dtype: NumpyScalarType::Int32 | NumpyScalarType::Int64,
533            value,
534        }
535        | ParameterValue::ScalarArray(value) => channel_integer(value),
536        _ => None,
537    }
538}

540/// The row selection; the TaQL expression is split's `msselect`, a
541/// casa-rs control, and mstransform's `taql`.
542fn selection(task: &TaskInvocation<'_>) -> Result<MsSelection, TaskFailure> {
543    let selector = |name: &str| -> Result<Option<String>, TaskFailure> {
544        let text = task_selection_text(task, name)?;
545        Ok((!text.trim().is_empty()).then_some(text))
546    };
547    let taql_name = if task.surface() == "split" {
548        "msselect"
549    } else {
550        "taql"
551    };
552    let mut selection = MsSelection {
553        field: variant_selector(task, "field")?,
554        scan: variant_selector(task, "scan")?,
555        antenna: variant_selector(task, "antenna")?,
556        timerange: variant_selector(task, "timerange")?,
557        observation: task.value("observation").and_then(variant_selector_text),
558        intent: task.text("intent")?.filter(|text| !text.is_empty()),
559        correlation: variant_selector(task, "correlation")?,
560        array: variant_selector(task, "array")?,
561        uvrange: variant_selector(task, "uvrange")?,
562        feed: variant_selector(task, "feed")?,
563        msselect: selector(taql_name)?,
564        ..MsSelection::default()
565    };
566    // Earlier releases accepted absolute MJD seconds as a numeric pair.
567    // CASA reads bare numbers in its clock grammar as seconds within the minute.
568    if let Some((start, end)) = selection.timerange.as_deref().and_then(absolute_time_range) {
569        selection.timerange = None;
570        selection = selection.time_range(start, end);
571    }
572    Ok(selection)
573}

575fn variant_selector(task: &TaskInvocation<'_>, name: &str) -> Result<Option<String>, TaskFailure> {
576    let Some(value) = task.value(name) else {
577        return Ok(None);
578    };
579    if let ParameterValue::String(raw) = value {
580        if matches!(name, "field" | "spw" | "scan" | "antenna" | "timerange") {
581            let text = task_selection_text(task, name)?;
582            let present = if name == "spw" {
583                !text.is_empty()
584            } else {
585                !text.trim().is_empty()
586            };
587            return Ok(present.then_some(text));
588        }
589        return Ok((!raw.is_empty()).then(|| raw.clone()));
590    }
591    if task.surface() == "split"
592        && matches!(
593            name,
594            "array" | "feed" | "field" | "spw" | "scan" | "antenna"
595        )
596    {
597        return Ok(variant_selector_text(value));
598    }
599    if task.surface() == "split" && name == "correlation" {
600        match value {
601            ParameterValue::EmptyArray { .. } => return Ok(None),
602            ParameterValue::Array(values) => {
603                let labels = values
604                    .iter()
605                    .map(|value| match value {
606                        ParameterValue::String(text) => Ok(text.as_str()),
607                        _ => Err(TaskFailure::validation("correlation expects strings").at(name)),
608                    })
609                    .collect::<Result<Vec<_>, _>>()?;
610                let text = labels.join(", ").to_uppercase();
611                return Ok((!text.is_empty()).then_some(text));
612            }
613            _ => {}
614        }
615    }
616    let record_type = match value {
617        ParameterValue::NumpyScalar {
618            dtype: NumpyScalarType::Int32 | NumpyScalarType::Int64,
619            ..
620        } => {
621            return Ok(None);
622        }
623        ParameterValue::Bool(_) => "Bool",
624        ParameterValue::Integer(_) => "Int",
625        _ => "unknown",
626    };
627    Err(TaskFailure::execution(format!(
628        "RecordRep::get_pointer - incorrect data type String   used for field {name} with type {record_type:<8}"
629    )).at(name))
630}

632fn variant_selector_text(value: &ParameterValue) -> Option<String> {
633    match value {
634        ParameterValue::NumpyScalar {
635            dtype: NumpyScalarType::Int32 | NumpyScalarType::Int64,
636            value,
637        }
638        | ParameterValue::ScalarArray(value) => variant_selector_text(value),
639        ParameterValue::String(text) => (!text.is_empty()).then(|| text.clone()),
640        other => Some(python_repr(other)),
641    }
642}

644/// Resolve CASA's visibility layout, including the existing corrected_data alias.
645fn data_column(value: &str) -> Result<TransformDataColumn, TaskFailure> {
646    let text = value.trim().to_ascii_lowercase();
647    match text.as_str() {
648        "data" => Ok(TransformDataColumn::Data),
649        "corrected" | "corrected_data" => Ok(TransformDataColumn::CorrectedData),
650        "model" => Ok(TransformDataColumn::ModelData),
651        "data,model,corrected" => Ok(TransformDataColumn::DataModelCorrected),
652        "float_data" => Ok(TransformDataColumn::FloatData),
653        "lag_data" => Ok(TransformDataColumn::LagData),
654        "float_data,data" => Ok(TransformDataColumn::FloatDataAndData),
655        "lag_data,data" => Ok(TransformDataColumn::LagDataAndData),
656        "all" => Ok(TransformDataColumn::All),
657        _ => Err(TaskFailure::validation(format!(
658            "unsupported datacolumn {value:?}; CASA takes 'corrected', 'data', 'model', \
659             'data,model,corrected', 'float_data', 'lag_data', 'float_data,data', \
660             'lag_data,data' or 'all'"
661        ))
662        .at("datacolumn")),
663    }
664}

666fn absolute_time_range(value: &str) -> Option<(f64, f64)> {
667    let (start, end) = value.split_once('~')?;
668    let start = start.trim().parse::<f64>().ok()?;
669    let end = end.trim().parse::<f64>().ok()?;
670    (start.is_finite()
671        && end.is_finite()
672        && (start > f64::from(i32::MAX) || end > f64::from(i32::MAX)))
673    .then_some((start, end))
674}
