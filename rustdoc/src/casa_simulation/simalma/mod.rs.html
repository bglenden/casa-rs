18/// CASA-named task parameters and the simulator's explicit environment.
19pub struct SimalmaRequest {
20    /// Sparse CASA simalma parameters; defaults come from the shared catalog.
21    pub parameters: Map<String, Value>,
22    /// Antenna configuration repositories, as for simobserve.
23    pub data_roots: Vec<PathBuf>,
24    /// Measures, clock, memory, workers and backend.
25    pub options: SimulatorOptions,
26}
27
28fn failure(error: impl std::fmt::Display) -> SimobserveError {
29    SimobserveError::raised(format!("simalma: {error}"))
30}
31
32struct Parameters(Map<String, Value>);
33impl Parameters {
34    fn text(&self, name: &str) -> &str {
35        self.0[name].as_str().expect("resolved string")
36    }
37    fn flag(&self, name: &str) -> bool {
38        self.0[name].as_bool().expect("resolved flag")
39    }
40    fn integer(&self, name: &str) -> i64 {
41        self.0[name].as_i64().expect("resolved integer")
42    }
43    fn strings(&self, name: &str) -> Vec<String> {
44        match &self.0[name] {
45            Value::String(one) => vec![one.clone()],
46            Value::Array(many) => many
47                .iter()
48                .map(|v| v.as_str().expect("resolved string").to_owned())
49                .collect(),
50            _ => unreachable!("resolved strings"),
51        }
52    }
53}

63        "indirection": "",

78        "tpnant": 0,

97fn resolve_defaults(overrides: &Map<String, Value>) -> Map<String, Value> {
98    let mut merged = default_parameters();
99    for (key, value) in overrides {
100        merged.insert(key.clone(), value.clone());
101    }
102    merged
103}

105fn graphics_mode(p: &Parameters) -> &str {
106    if p.text("graphics") == "screen" {
107        "both"
108    } else {
109        p.text("graphics")
110    }
111}

113/// Simulate each requested array and TP antenna, image, and combine them.
114///
115/// # Errors
116/// Invalid input/configuration/time, an existing protected project or input,
117/// resource refusal, or a native observation/imaging failure.
118pub fn simalma(request: &SimalmaRequest) -> Result<SimobserveOutcome, SimobserveError> {
119    run(request, None)
120}

122/// Simulate and emit complete figure inputs through `sink`.
123///
124/// # Errors
125/// Simulation errors or any error reported by `sink`.
126pub fn simalma_with_plot_sink(
127    request: &SimalmaRequest,
128    sink: &mut dyn SimulationPlotSink,
129) -> Result<SimobserveOutcome, SimobserveError> {
130    run(request, Some(sink))
131}

133fn run<'a, 'b>(
134    request: &SimalmaRequest,
135    plot_sink: Option<&'a mut (dyn SimulationPlotSink + 'b)>,
136) -> Result<SimobserveOutcome, SimobserveError> {
137    let resources = request.options.resources;
138    charge_process_caches(resources.host(), resources.policy()).map_err(failure)?;
139    let _planning = admit(
140        resources.host(),
141        resources.policy(),
142        &Demand {
143            phase: "ALMA simulation planning",
144            memory: (2 << 20)
145                + request
146                    .parameters
147                    .iter()
148                    .map(|(name, value)| {
149                        128 + name.len() as u64 + crate::simanalyze::parameter_bytes(value)
150                    })
151                    .sum::<u64>()
152                    .saturating_mul(4),
153        },
154    )
155    .map_err(failure)?;
156    let values = resolve_defaults(&request.parameters);
157    let p = Parameters(values);
158    let mut plan = plan::prepare(&p, request)?;
159    let mut outcome = SimobserveOutcome {
160        directory: plan.directory.clone(),
161        stem: plan.project.clone(),
162        products: Vec::new(),
163        warnings: Vec::new(),
164    };
165    if p.flag("dryrun") {
166        outcome.warnings.extend(plan.arrays.iter().map(|array| {
167            format!(
168                "dryrun: simulate {} for {:.6} minutes",
169                array.configuration, array.minutes
170            )
171        }));
172        if plan.tp_antennas > 0 {
173            outcome.warnings.push(format!(
174                "dryrun: simulate {} TP antennas for {} minutes each",
175                plan.tp_antennas,
176                plan.tp_minutes.trunc()
177            ));
178        }
179        return Ok(outcome);
180    }
181    let graphics_requested = matches!(p.text("graphics"), "file" | "screen" | "both");
182    if p.text("graphics") != "none" && plot_sink.is_none() {
183        outcome
184            .warnings
185            .push("simalma PNG rendering needs the Python host".into());
186    }
187    let mut sink = if graphics_requested { plot_sink } else { None };
188    if sink.is_some() {
189        plan.observation.graphics = graphics_mode(&p).to_owned();
190    }
191    plan::publish_directory(&p, &plan)?;
192    for array in &plan.arrays {
193        let _sky = admit(
194            resources.host(),
195            resources.policy(),
196            &Demand {
197                phase: "ALMA observation sky preparation",
198                memory: plan.model.preparation_bytes,
199            },
200        )
201        .map_err(failure)?;
202        let mut observation = plan.observation.clone();
203        observation.antennalist = array.configuration.clone();
204        observation.totaltime = format!("{:.6}min", array.minutes);
205        observation.correlator = array.correlator.clone();
206        let result = observe(
207            &crate::simobserve::SimobserveRequest {
208                parameters: observation,
209                data_roots: request.data_roots.clone(),
210                options: request.options.clone(),
211            },
212            sink.as_deref_mut(),
213        )?;
214        outcome.products.extend(result.products);
215        outcome.warnings.extend(result.warnings);
216    }
217    if plan.tp_antennas > 0 {
218        let (mut observation, _pointings) = pointings::observation(&p, &plan, resources)?;
219        for antenna in 0..plan.tp_antennas {
220            let _sky = admit(
221                resources.host(),
222                resources.policy(),
223                &Demand {
224                    phase: "ALMA TP sky preparation",
225                    memory: plan.model.preparation_bytes,
226                },
227            )
228            .map_err(failure)?;
229            observation.sdant = antenna as i64;
230            // Independent streams per dish. Random sample identity is not a
231            // CASA contract; the native simulator owns its seeded stream.
232            observation.seed = i64::from(crate::DEFAULT_SEED) + antenna as i64;
233            let result = observe(
234                &crate::simobserve::SimobserveRequest {
235                    parameters: observation.clone(),
236                    data_roots: request.data_roots.clone(),
237                    options: request.options.clone(),
238                },
239                sink.as_deref_mut(),
240            )?;
241            for mut path in result.products {
242                if plan.tp_antennas > 1 && path.to_string_lossy().ends_with(".ms") {
243                    let mut target = path.as_os_str().to_os_string();
244                    target.push(format!(".Ant{antenna}"));
245                    let target = PathBuf::from(target);
246                    std::fs::rename(&path, &target).map_err(failure)?;
247                    path = target;
248                }
249                if !outcome.products.contains(&path) {
250                    outcome.products.push(path);
251                }
252            }
253            let noisy = if plan.noisy { ".noisy" } else { "" };
254            let suffix = if plan.tp_antennas > 1 {
255                format!(".Ant{antenna}")
256            } else {
257                String::new()
258            };
259            plan.tp_vis.push(
260                plan.directory
261                    .join(format!("{}{noisy}.sd.ms{suffix}", plan.tp_stem)),
262            );
263        }
264    }
265    if p.flag("image") {
266        imaging::execute(&p, &plan, resources, sink.as_deref_mut(), &mut outcome)?;
267    }
268    if let Some(sink) = sink {
269        let combination = if p.flag("image") && !plan.tp_vis.is_empty() {
270            let best = plan
271                .arrays
272                .iter()
273                .min_by(|a, b| a.resolution.total_cmp(&b.resolution))
274                .unwrap();
275            let suffix = if plan.noisy { ".noisy" } else { "" };
276            let high_stem = if plan.arrays.len() > 1 {
277                format!("{}.concat", plan.project)
278            } else {
279                format!("{}{suffix}", best.stem)
280            };
281            Some(SimalmaCombinationInput {
282                directory: &plan.directory,
283                project: &plan.project,
284                best_stem: best.stem.clone(),
285                high_stem,
286                model_preparation_bytes: plan.model.preparation_bytes,
287                resources,
288            })
289        } else {
290            None
291        };
292        sink.finish_simalma(SimalmaFinishInput {
293            graphics_mode: p.text("graphics"),
294            combination,
295        })?;
296    }
297    outcome.products.sort();
298    outcome.products.dedup();
299    std::fs::write(
300        plan.directory
301            .join(format!("{}.simalma.report.txt", plan.project)),
302        format!(
303            "simalma arrays={} tp_antennas={} products={}\n",
304            plan.arrays.len(),
305            plan.tp_antennas,
306            outcome.products.len()
307        ),
308    )
309    .map_err(failure)?;
310    Ok(outcome)
311}

313fn observe<'a, 'b>(
314    request: &crate::simobserve::SimobserveRequest,
315    plot_sink: Option<&'a mut (dyn SimulationPlotSink + 'b)>,
316) -> Result<SimobserveOutcome, SimobserveError> {
317    match plot_sink {
318        Some(sink) => simobserve_with_plot_sink(request, sink),
319        None => simobserve(request),
320    }
