22/// One owner-issued selected sample and its evaluated output spectral support.
23///
24/// This envelope is constructed only by retained [`BoundSelectedObservation`](super::BoundSelectedObservation)
25/// traversal after the selected sample has passed compiled-problem validation.
26/// It is not persisted and does not participate in selected-sample content
27/// identity.
28///
29/// ```compile_fail
30/// use casa_ms::SelectedObservationTraversalSample;
31///
32/// let _forged = SelectedObservationTraversalSample {};
33/// ```
34#[derive(Debug, Clone, Copy, PartialEq)]
35pub struct SelectedObservationTraversalSample<'a> {
36    sample: SelectedObservationSampleView<'a>,
37    spectral_evaluation: SelectedSpectralEvaluation,
38}
39
40impl<'a> SelectedObservationTraversalSample<'a> {
41    pub(super) fn with_spectral_evaluation(
42        sample: SelectedObservationSampleView<'a>,
43        spectral_evaluation: SelectedSpectralEvaluation,
44    ) -> Self {
45        Self {
46            sample,
47            spectral_evaluation,
48        }
49    }
50
51    /// Return the selected sample validated by the compiled problem.
52    #[must_use]
53    pub const fn selected(&self) -> SelectedObservationSampleView<'a> {
54        self.sample
55    }
56
57    /// Return source-backed native/output-frame intervals, flag validity, and effective weight.
58    #[must_use]
59    pub const fn spectral_evaluation(&self) -> SelectedSpectralEvaluation {
60        self.spectral_evaluation
61    }
62}

64/// One borrowed row/channel run validated in canonical correlation order.
65///
66/// The run borrows one row and reusable correlation/evaluation scratch while
67/// keeping its channel inline. It is released before the source block can be
68/// refilled or the scratch can be reused.
69#[derive(Debug, Clone, Copy, PartialEq)]
70pub struct SelectedObservationTraversalRun<'a> {
71    row: &'a SelectedObservationRunRow,
72    channel: SelectedObservationRunChannel,
73    correlations: &'a [SelectedObservationRunCorrelation],
74    evaluations: &'a [SelectedSpectralEvaluation],
75}
76
77impl<'a> SelectedObservationTraversalRun<'a> {
78    pub(super) fn new(
79        row: &'a SelectedObservationRunRow,
80        channel: SelectedObservationRunChannel,
81        correlations: &'a [SelectedObservationRunCorrelation],
82        evaluations: &'a [SelectedSpectralEvaluation],
83    ) -> Self {
84        debug_assert_eq!(correlations.len(), evaluations.len());
85        Self {
86            row,
87            channel,
88            correlations,
89            evaluations,
90        }
91    }
92
93    /// Iterate through validated correlation members in canonical order.
94    pub fn samples(&self) -> impl Iterator<Item = SelectedObservationTraversalSample<'_>> {
95        self.correlations
96            .iter()
97            .zip(self.evaluations.iter())
98            .map(|(correlation, evaluation)| {
99                SelectedObservationTraversalSample::with_spectral_evaluation(
100                    SelectedObservationSampleView::from_run(self.row, &self.channel, correlation),
101                    *evaluation,
102                )
103            })
104    }
105
106    /// Return the number of correlations in this row/channel run.
107    #[must_use]
108    pub fn len(&self) -> usize {
109        self.correlations.len()
110    }
111
112    /// Return whether the run contains no correlations.
113    #[must_use]
114    pub fn is_empty(&self) -> bool {
115        self.correlations.is_empty()
116    }
117}

119#[derive(Debug, Clone, Copy, PartialEq, Eq)]
120struct SpectralProjectionKey {
121    measurement_set: casa_imaging_model::MeasurementSetIdentity,
122    spectral_window_id: u32,
123    channel_index: u32,
124    frequency_centre_bits: u64,
125    frequency_lower_bits: u64,
126    frequency_upper_bits: u64,
127    channel_width_bits: u64,
128    frequency_frame: FrequencyFrame,
129    field_id: i32,
130    time_mjd_days_bits: u64,
131}
132
133impl SpectralProjectionKey {
134    fn from_sample(sample: SelectedObservationSampleView<'_>) -> Self {
135        let address = sample.address();
136        let metadata = sample.metadata();
137        let coordinates = sample.coordinates();
138        Self {
139            measurement_set: address.measurement_set,
140            spectral_window_id: address.spectral_window_id,
141            channel_index: address.channel_index,
142            frequency_centre_bits: address.frequency_centre_hz.to_bits(),
143            frequency_lower_bits: address.frequency_lower_hz.to_bits(),
144            frequency_upper_bits: address.frequency_upper_hz.to_bits(),
145            channel_width_bits: address.channel_width_hz.to_bits(),
146            frequency_frame: address.frequency_frame,
147            field_id: metadata.field_id,
148            time_mjd_days_bits: coordinates.time.mjd_days().to_bits(),
149        }
150    }
151}

153#[derive(Debug, Clone, Copy, PartialEq, Eq)]
154struct SpectralTransformKey {
155    measurement_set: casa_imaging_model::MeasurementSetIdentity,
156    field_id: i32,
157    time_mjd_seconds_bits: u64,
158    source_ref: FrequencyRef,
159    output_ref: FrequencyRef,
160}

162/// Retains frame-conversion state across the canonical correlation-major stream.
163///
164/// Spectral assignment is independent of correlation and visibility value. A
165/// row/channel's parallel-hand samples therefore reuse the exact same derived
166/// evaluation, while a row, channel interval, field, time, frame, weight, flag, SPW, or source
167/// change forces a fresh owner evaluation.
168pub(super) struct SpectralEvaluationProjector {
169    last_source: Option<casa_imaging_model::MeasurementSetIdentity>,
170    output_frame: Option<MeasFrame>,
171    source_frame: Option<(i32, u64, MeasFrame)>,
172    last_transform: Option<(SpectralTransformKey, PreparedFrequencyFrameConversion)>,
173    last_projection: Option<(
174        SpectralProjectionKey,
175        SelectedSpectralInterval,
176        SelectedSpectralInterval,
177    )>,
178}
179
180impl SpectralEvaluationProjector {
181    pub(super) const fn new() -> Self {
182        Self {
183            last_source: None,
184            output_frame: None,
185            source_frame: None,
186            last_transform: None,
187            last_projection: None,
188        }
189    }
190
191    pub(super) fn project<'a>(
192        &mut self,
193        problem: &CompiledProblem,
194        sample: SelectedObservationSampleView<'a>,
195        geometry_engine: &MsCalEngine,
196    ) -> Result<SelectedObservationTraversalSample<'a>, BoundObservationSourceError> {
197        let key = SpectralProjectionKey::from_sample(sample);
198        let address = sample.address();
199        if self.last_source != Some(address.measurement_set) {
200            self.last_source = Some(address.measurement_set);
201            self.output_frame = None;
202            self.source_frame = None;
203            self.last_transform = None;
204            self.last_projection = None;
205        }
206        let (native, output) = if let Some((cached_key, native, output)) = self.last_projection
207            && cached_key == key
208        {
209            (native, output)
210        } else {
211            let intervals = derive_spectral_intervals_cached(
212                problem,
213                sample,
214                geometry_engine,
215                &mut self.source_frame,
216                &mut self.output_frame,
217                &mut self.last_transform,
218            )?;
219            self.last_projection = Some((key, intervals.0, intervals.1));
220            intervals
221        };
222        let input_weight = sample.input_weight();
223        let valid = !sample.channel_flag()
224            && !sample.parallel_hand_group_flag()
225            && !sample.row_flag()
226            && input_weight.is_finite()
227            && input_weight > 0.0;
228        let evaluation = SelectedSpectralEvaluation::new(
229            native,
230            output,
231            if valid { f64::from(input_weight) } else { 0.0 },
232            valid,
233        )
234        .ok_or(BoundObservationSourceError::SpectralContributionMismatch)?;
235        Ok(SelectedObservationTraversalSample::with_spectral_evaluation(sample, evaluation))
236    }
237}

239fn derive_spectral_intervals_cached(
240    problem: &CompiledProblem,
241    sample: SelectedObservationSampleView<'_>,
242    geometry_engine: &MsCalEngine,
243    source_frame: &mut Option<(i32, u64, MeasFrame)>,
244    output_frame: &mut Option<MeasFrame>,
245    last_transform: &mut Option<(SpectralTransformKey, PreparedFrequencyFrameConversion)>,
246) -> Result<(SelectedSpectralInterval, SelectedSpectralInterval), BoundObservationSourceError> {
247    let address = sample.address();
248    let native_boundaries = if address.channel_width_hz >= 0.0 {
249        [address.frequency_lower_hz, address.frequency_upper_hz]
250    } else {
251        [address.frequency_upper_hz, address.frequency_lower_hz]
252    };
253    let native = SelectedSpectralInterval::new(
254        address.frequency_centre_hz,
255        native_boundaries[0],
256        native_boundaries[1],
257    )
258    .ok_or(BoundObservationSourceError::SpectralContributionMismatch)?;
259    let conversion = prepared_frequency_conversion_cached(
260        problem,
261        sample,
262        geometry_engine,
263        source_frame,
264        output_frame,
265        last_transform,
266    )?;
267    let output_centre_hz = conversion.convert_hz(native.centre_hz());
268    let output_first_hz = conversion.convert_hz(native_boundaries[0]);
269    let output_second_hz = conversion.convert_hz(native_boundaries[1]);
270    let output = SelectedSpectralInterval::new(output_centre_hz, output_first_hz, output_second_hz)
271        .ok_or(BoundObservationSourceError::SpectralContributionMismatch)?;
272    Ok((native, output))
273}

275fn prepared_frequency_conversion_cached(
276    problem: &CompiledProblem,
277    sample: SelectedObservationSampleView<'_>,
278    geometry_engine: &MsCalEngine,
279    source_frame_cache: &mut Option<(i32, u64, MeasFrame)>,
280    output_frame_cache: &mut Option<MeasFrame>,
281    last_transform: &mut Option<(SpectralTransformKey, PreparedFrequencyFrameConversion)>,
282) -> Result<PreparedFrequencyFrameConversion, BoundObservationSourceError> {
283    let spectral = problem.geometry().spectral();
284    let address = sample.address();
285    let metadata = sample.metadata();
286    let coordinates = sample.coordinates();
287    let source_ref = frequency_ref(address.frequency_frame);
288    let output_ref = frequency_ref(spectral.output_frame());
289    let time_mjd_seconds = coordinates.time.mjd_days() * 86_400.0;
290    let time_bits = time_mjd_seconds.to_bits();
291    let key = SpectralTransformKey {
292        measurement_set: address.measurement_set,
293        field_id: metadata.field_id,
294        time_mjd_seconds_bits: time_bits,
295        source_ref,
296        output_ref,
297    };
298    if let Some((cached_key, conversion)) = *last_transform
299        && cached_key == key
300    {
301        return Ok(conversion);
302    }
303    if source_ref == output_ref {
304        let conversion = PreparedFrequencyFrameConversion::new(source_ref, output_ref, None, None)
305            .map_err(BoundObservationSourceError::from)?;
306        *last_transform = Some((key, conversion));
307        return Ok(conversion);
308    }
309    let field_id = usize::try_from(metadata.field_id)
310        .map_err(|_| BoundObservationSourceError::SpectralContributionMismatch)?;
311    let source_frame = match source_frame_cache {
312        Some((cached_field, cached_time, frame))
313            if *cached_field == metadata.field_id && *cached_time == time_bits =>
314        {
315            frame
316        }
317        slot => {
318            *slot = Some((
319                metadata.field_id,
320                time_bits,
321                geometry_engine.spectral_frame_observatory(time_mjd_seconds, field_id)?,
322            ));
323            &slot.as_ref().expect("source frame was inserted").2
324        }
325    };
326    let SpectralFrameAnchor::Conversion {
327        epoch,
328        direction,
329        observatory_position,
330    } = spectral.anchor()
331    else {
332        return Err(BoundObservationSourceError::SpectralContributionMismatch);
333    };
334    let output_frame = output_frame_cache.get_or_insert_with(|| {
335        let [x_metres, y_metres, z_metres] = observatory_position.metres();
336        geometry_engine.spectral_frame_explicit(
337            MEpoch::from_mjd(epoch.mjd_days(), epoch_ref(epoch.scale())),
338            MPosition::new_itrf(x_metres, y_metres, z_metres),
339            MDirection::from_angles(
340                direction.longitude_rad(),
341                direction.latitude_rad(),
342                direction_ref(direction.frame()),
343            ),
344        )
345    });
346    let conversion = PreparedFrequencyFrameConversion::new(
347        source_ref,
348        output_ref,
349        Some(source_frame),
350        Some(output_frame),
351    )
352    .map_err(BoundObservationSourceError::from)?;
353    *last_transform = Some((key, conversion));
354    Ok(conversion)
355}

357const fn direction_ref(frame: DirectionFrame) -> DirectionRef {
358    match frame {
359        DirectionFrame::Icrs => DirectionRef::ICRS,
360        DirectionFrame::J2000 => DirectionRef::J2000,
361        DirectionFrame::B1950 => DirectionRef::B1950,
362        DirectionFrame::Galactic => DirectionRef::GALACTIC,
363    }
364}
365
366const fn epoch_ref(scale: TimeScale) -> EpochRef {
367    match scale {
368        TimeScale::Utc => EpochRef::UTC,
369        TimeScale::Tai => EpochRef::TAI,
370        TimeScale::Tt => EpochRef::TT,
371        TimeScale::Tdb => EpochRef::TDB,
372    }
373}
374
375const fn frequency_ref(frame: FrequencyFrame) -> FrequencyRef {
376    match frame {
377        FrequencyFrame::Topocentric => FrequencyRef::TOPO,
378        FrequencyFrame::Barycentric => FrequencyRef::BARY,
379        FrequencyFrame::Lsrk => FrequencyRef::LSRK,
380    }
