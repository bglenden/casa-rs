3//! Bounded evaluation of compiled scalar primary-beam power responses.
4
5use casa_numerics::AnnularApertureVoltageTable;
6
7use crate::SpectralOperatorError;
8
9const CASA_ALMA_ACA_DIRECT_PB_SUPPORT_ARCMIN_GHZ: f64 = 3.568 * 60.0;
10
11#[derive(Clone)]
12pub(crate) struct PreparedPrimaryBeamPower {
13    table: AnnularApertureVoltageTable,
14    reference_pixel: [f64; 2],
15    increment_rad: [f64; 2],
16    pc: [[f64; 2]; 2],
17    shape: [usize; 2],
18    cutoff: f32,
19}
20
21impl PreparedPrimaryBeamPower {
22    pub(crate) fn casa_alma_aca_interferometric_direct(
23        reference_pixel: [f64; 2],
24        increment_rad: [f64; 2],
25        shape: [usize; 2],
26        cutoff: f32,
27    ) -> Result<Self, SpectralOperatorError> {
28        if shape.contains(&0) || !cutoff.is_finite() || cutoff < 0.0 {
29            return Err(SpectralOperatorError::UnsupportedProblem);
30        }
31        Ok(Self {
32            table: AnnularApertureVoltageTable::new(
33                6.25,
34                0.75,
35                CASA_ALMA_ACA_DIRECT_PB_SUPPORT_ARCMIN_GHZ,
36            ),
37            reference_pixel,
38            increment_rad,
39            pc: [[1.0, 0.0], [0.0, 1.0]],
40            shape,
41            cutoff,
42        })
43    }
44
45    pub(crate) fn fill_power_plane_into(
46        &self,
47        frequency_hz: f64,
48        output: &mut [f32],
49    ) -> Result<(), SpectralOperatorError> {
50        let cells = self.shape[0]
51            .checked_mul(self.shape[1])
52            .ok_or(SpectralOperatorError::ResidencyOverflow)?;
53        if output.len() != cells || !frequency_hz.is_finite() || frequency_hz <= 0.0 {
54            return Err(SpectralOperatorError::InvalidSample);
55        }
56        let mut maximum = 0.0_f32;
57        for x in 0..self.shape[0] {
58            let pixel_x = x as f64 - self.reference_pixel[0];
59            for y in 0..self.shape[1] {
60                let pixel_y = y as f64 - self.reference_pixel[1];
61                let axis_x = pixel_x * self.increment_rad[0];
62                let axis_y = pixel_y * self.increment_rad[1];
63                let l_rad = self.pc[0][0] * axis_x + self.pc[0][1] * axis_y;
64                let m_rad = self.pc[1][0] * axis_x + self.pc[1][1] * axis_y;
65                // PBMath1D stores the two angular offsets and radius as Float.
66                let l_deg = l_rad.to_degrees() as f32;
67                let m_deg = m_rad.to_degrees() as f32;
68                let radius_deg = (l_deg * l_deg + m_deg * m_deg).sqrt();
69                let radius_arcmin_ghz =
70                    (f64::from(radius_deg) * 60.0 * (frequency_hz / 1.0e9)) as f32 as f64;
71                let voltage = self.table.evaluate(radius_arcmin_ghz);
72                let power = voltage * voltage;
73                output[x * self.shape[1] + y] = power;
74                maximum = maximum.max(power);
75            }
76        }
77        if maximum > 1.0 {
78            output.iter_mut().for_each(|value| *value /= maximum);
79        }
80        output.iter_mut().for_each(|value| {
81            if !value.is_finite() || *value <= self.cutoff {
82                *value = 0.0;
83            }
84        });
85        Ok(())
86    }
87}

90mod tests {
91    use std::{error::Error, path::PathBuf};

93    use casa_images::PagedImage;

97    #[test]
98    fn direct_alma_power_is_streamed_into_one_caller_plane() {
99        let response = PreparedPrimaryBeamPower::casa_alma_aca_interferometric_direct(
100            [4.0, 4.0],
101            [-4.848_136_811_095_36e-7, 4.848_136_811_095_36e-7],
102            [8, 8],
103            0.1,
104        )
105        .expect("compiled response");
106        let mut plane = [0.0; 64];
107        response
108            .fill_power_plane_into(230.0e9, &mut plane)
109            .expect("power plane");
110        assert_eq!(plane[4 * 8 + 4], 1.0);
111        assert!(plane.iter().all(|value| value.is_finite()));
112        assert!(plane[0] < 1.0);
113    }

115    #[test]
116    #[ignore = "requires the frozen CASA T41 MVC primary-beam cube"]
117    fn t41_alma_mvc_primary_beam_owner_matches_frozen_cube() -> Result<(), Box<dyn Error>> {
118        let prefix = PathBuf::from(
119            std::env::var_os("CASA_RS_T41_MVC_CASA_PREFIX")
120                .ok_or("CASA_RS_T41_MVC_CASA_PREFIX is not set")?,
121        );
122        let casa = PagedImage::<f32>::open(PathBuf::from(format!("{}.pb", prefix.display())))?;
123        assert_eq!(casa.shape(), [512, 512, 1, 40]);
124        let shape = casa.shape().to_vec();
125        let expected = casa.get_slice(&vec![0; shape.len()], &shape)?;
126        let valid = casa
127            .get_mask_slice(&vec![0; shape.len()], &shape, &vec![1; shape.len()])?
128            .map_or_else(
129                || vec![true; shape.iter().product()],
130                |mask| mask.iter().copied().collect(),
131            );
132        let response = PreparedPrimaryBeamPower::casa_alma_aca_interferometric_direct(
133            [256.0, 256.0],
134            [-4.848_136_811_095_359e-7, 4.848_136_811_095_359e-7],
135            [512, 512],
136            0.1,
137        )?;
138        let mut generated = vec![0.0; 512 * 512];
139        for channel in [0, 39] {
140            let frequency = 230_449_729_492.188_84 + channel as f64 * 122_982_578.274_169_92;
141            response.fill_power_plane_into(frequency, &mut generated)?;
142            let (error, reference) = generated
143                .iter()
144                .enumerate()
145                .filter(|(cell, _)| valid[cell * 40 + channel])
146                .fold((0.0, 0.0), |(error, reference), (cell, actual)| {
147                    let actual = f64::from(*actual);
148                    let expected = f64::from(expected[[cell / 512, cell % 512, 0, channel]]);
149                    (
150                        error + (actual - expected).powi(2),
151                        reference + expected.powi(2),
152                    )
153                });
154            let nrms = (error / reference.max(f64::MIN_POSITIVE)).sqrt();
155            eprintln!("t41_mvc_pb_owner channel={channel} nrms={nrms:.9e}");
156            assert!(nrms <= 5.0e-6, "channel {channel} PB NRMS {nrms:.6e}");
157        }
158        Ok(())
159    }
