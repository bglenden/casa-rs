2//! The selection resolved to the observation's partitions (#932).
3//!
4//! Compile names the partitions that hold the request's samples
5//! ([`candidates`]): a partition's window and observing session must be
6//! selected, its fields must include a selected one, under a scan selection
7//! one of its scans must be a selected one, and, under an intent selection,
8//! its observing mode must be a selected one ([`super::intents`]). The survey then keeps
9//! those that hold a selected sample, and the compiled problem names them
10//! ([`casa_imaging_model::PartitionSelection`]). The run reads the same
11//! partitions ([`resolve_partitions`]).

31/// Speed of light in m/s, for a window's reference wavelength.
32const SPEED_OF_LIGHT_M_S: f64 = 299_792_458.0;

34/// Why the selection cannot be read through the observation's partitions.
35#[derive(Debug, thiserror::Error)]
36pub enum PartitionSelectionError {
37    /// What the passes do not read.
38    #[error("native imaging does not read {0}")]
39    Unsupported(&'static str),
40    /// A selected product the partition does not hold, or a window
41    /// without channels or reference frequency.
42    #[error("the selection is inconsistent with the observation: {0}")]
43    Inconsistent(String),
44}
45
46/// One partition the request selects, before the survey reads its samples.
47pub(super) struct Candidate {
48    pub(super) id: PartitionId,
49    /// Its spectral window number (the catalogue's window; an MSv2
50    /// `SPECTRAL_WINDOW` row).
51    pub(super) spectral_window: usize,
52    /// Its window's reference wavelength, when a UV-distance interval is in
53    /// wavelengths.
54    pub(super) wavelength_m: Option<f64>,
55}
56
57/// The partitions a request selects, and the data group holding the
58/// visibility column it images.
59pub(super) struct Candidates {
60    pub(super) data_group: String,
61    pub(super) partitions: Vec<Candidate>,
62}

64/// The partitions of `store` in the windows `windows` selects that `rows`
65/// may select, in partition order: a partition's observing session must be
66/// selected, its fields must include a selected one, under a scan selection
67/// one of its scans must be a selected one (the survey and the run test
68/// each time's scan), and, under an intent selection, its observing mode
69/// must be a selected one. A state number in the selection (`states`) must select
70/// every state of its partition. A partition's UV distances and times are
71/// the survey's to test.
72///
73/// # Errors
74///
75/// [`PartitionSelectionError::Unsupported`] for an intent selection by
76/// state that splits an `OBS_MODE`.
77pub(super) fn selected(
78    store: &OpenedStore,
79    windows: &WindowChoice,
80    (rows, states): (&RowSelection, &BTreeSet<StateId>),
81) -> Result<Vec<PartitionId>, PartitionSelectionError> {
82    let catalogue = store.catalogue();
83    let fields = field_selection(catalogue, rows.fields());
84    let modes = match rows.intents() {
85        IntentSelection::All => None,
86        IntentSelection::Only(modes) => Some(modes.iter().cloned().collect::<BTreeSet<_>>()),
87    };
88    let mut selected = Vec::new();
89    for id in catalogue.partition_ids() {
90        let entry = catalogue.partition(id);
91        if !windows.holds(store, id) {
92            continue;
93        }
94        if !rows.observations().holds(i64::from(entry.observation.0)) {
95            continue;
96        }
97        if !rows.baselines().is_all()
98            && !entry.baselines.iter().any(|baseline| {
99                rows.baselines()
100                    .holds(baseline.antenna1.0, baseline.antenna2.0)
101            })
102        {
103            continue;
104        }
105        if !rows.scans().is_all()
106            && !entry.scans.iter().any(|scan| {
107                catalogue.scans[scan.0 as usize]
108                    .number
109                    .is_some_and(|number| rows.scans().holds(i64::from(number)))
110            })
111        {
112            continue;
113        }
114        if let Some(modes) = &modes {
115            if !modes.contains(&intents::observing_mode(catalogue, entry)) {
116                continue;
117            }
118            if !entry.states.is_empty() && !entry.states.iter().any(|state| states.contains(state))
119            {
120                continue;
121            }
122            if intents::split_state(entry, states) {
123                return Err(PartitionSelectionError::Unsupported(
124                    "an intent selection that takes some of the states sharing an OBS_MODE (the \
125                     MSv2 adapter partitions their rows together)",
126                ));
127            }
128        }
129        if entry.fields.iter().any(|field| fields.holds(*field)) {
130            selected.push(id);
131        }
132    }
133    Ok(selected)
134}

136/// The `selected` partitions of `store` ([`selected`]) with the data group
137/// holding `visibility`, which each must have.
138///
139/// # Errors
140///
141/// [`PartitionSelectionError::Unsupported`] for what the passes do not
142/// read: `FLOAT_DATA` (no request names it) and a partition without the data
143/// group. [`PartitionSelectionError::Inconsistent`] for a window without a
144/// positive reference frequency under a UV-distance interval in
145/// wavelengths.
146pub(super) fn candidates(
147    store: &OpenedStore,
148    selected: &[PartitionId],
149    rows: &RowSelection,
150    visibility: VisibilityColumn,
151) -> Result<Candidates, PartitionSelectionError> {
152    if visibility == VisibilityColumn::FloatData {
153        return Err(PartitionSelectionError::Unsupported("FLOAT_DATA"));
154    }
155    let group = data_group(visibility);
156    let catalogue = store.catalogue();
157    let mut partitions = Vec::new();
158    for &id in selected {
159        let entry = catalogue.partition(id);
160        if entry.data_group(group).is_none() {
161            return Err(PartitionSelectionError::Unsupported(
162                "a selected partition without the imaged data group",
163            ));
164        }
165        partitions.push(Candidate {
166            id,
167            spectral_window: entry.window.0 as usize,
168            wavelength_m: wavelength(catalogue, entry, rows)?,
169        });
170    }
171    Ok(Candidates {
172        data_group: group.to_string(),
173        partitions,
174    })
175}

177/// The selection of `problem`'s source `source` (its MeasurementSet's place
178/// in the snapshot) resolved to that source's `observation`'s partitions,
179/// in partition order: the partitions compile chose, the channels of their
180/// windows, their correlations, the data group holding the visibility
181/// column, and the per-sample predicates.
182///
183/// # Errors
184///
185/// [`PartitionSelectionError::Unsupported`] for `FLOAT_DATA` and
186/// partitions whose column belongs to different data groups;
187/// [`PartitionSelectionError::Inconsistent`] for a source the snapshot does
188/// not hold, a window the selection names no channel of, a correlation a
189/// partition does not hold, and a window without a positive reference
190/// frequency under a UV-distance interval in wavelengths.
191pub(crate) fn resolve_partitions(
192    problem: &CompiledProblem,
193    source: usize,
194    observation: &dyn Observation,
195) -> Result<PartitionSelection, PartitionSelectionError> {
196    let source = problem
197        .observation_transaction()
198        .read_set()
199        .sources()
200        .get(source)
201        .ok_or_else(|| {
202            PartitionSelectionError::Inconsistent(format!(
203                "the observation has no MeasurementSet {source}"
204            ))
205        })?;
206    let selection = source.selection();
207    let visibility = source.selected_columns().visibility();
208    if visibility == VisibilityColumn::FloatData {
209        return Err(PartitionSelectionError::Unsupported("FLOAT_DATA"));
210    }
211    let group = data_group(visibility);
212    let catalogue = observation.catalogue();
213    let rows = selection.rows_filter();
214    let mut partitions = Vec::with_capacity(selection.partitions().len());
215    for selected in selection.partitions() {
216        let entry = catalogue.partition(selected.partition());
217        if entry.data_group(group).is_none() {
218            return Err(PartitionSelectionError::Unsupported(
219                "a selected partition without the imaged data group",
220            ));
221        }
222        let window = selected.spectral_window_id();
223        let channels = selection
224            .spectral_windows()
225            .iter()
226            .find(|selected| selected.spectral_window_id() == window)
227            .map(|selected| selected.channel_indices().to_vec())
228            .filter(|channels| !channels.is_empty())
229            .ok_or_else(|| {
230                PartitionSelectionError::Inconsistent(format!(
231                    "spectral window {window} has no selected channel"
232                ))
233            })?;
234        let correlations: Vec<Correlation> = selected
235            .products()
236            .iter()
237            .map(|product| product.correlation_type())
238            .collect();
239        if let Some(missing) = correlations
240            .iter()
241            .find(|product| entry.correlation_index(**product).is_none())
242        {
243            return Err(PartitionSelectionError::Inconsistent(format!(
244                "spectral window {window} lacks {missing:?}"
245            )));
246        }
247        partitions.push(SelectedPartition {
248            id: selected.partition(),
249            spectral_window: window,
250            channels,
251            correlations,
252            wavelength_m: wavelength(catalogue, entry, rows)?,
253        });
254    }
255    if partitions.is_empty() {
256        return Err(PartitionSelectionError::Unsupported(
257            "a selection no partition holds",
258        ));
259    }
260    Ok(PartitionSelection {
261        data_group: group.to_string(),
262        partitions,
263        fields: field_selection(catalogue, rows.fields()),
264        scans: rows.scans().clone(),
265        baselines: rows.baselines().clone(),
266        times: rows.times().clone(),
267        uv: rows.uv_distances().clone(),
268    })
269}

271/// The fields of `catalogue` that `fields` selects, as the boundary names
272/// them.
273pub(super) fn field_selection(catalogue: &Catalogue, fields: &IdSelection) -> FieldSelection {
274    match fields {
275        IdSelection::All => FieldSelection::All,
276        IdSelection::Only(ids) => FieldSelection::Only(ids.iter().copied().map(FieldId).collect()),
277        selection @ IdSelection::Ranges(_) => FieldSelection::Only(
278            (0..catalogue.fields.len())
279                .filter_map(|index| u32::try_from(index).ok())
280                .filter(|&id| selection.holds(i64::from(id)))
281                .map(FieldId)
282                .collect(),
283        ),
284    }
285}

287/// `entry`'s window's reference wavelength, when `rows` tests a UV distance
288/// in wavelengths.
289fn wavelength(
290    catalogue: &Catalogue,
291    entry: &PartitionEntry,
292    rows: &RowSelection,
293) -> Result<Option<f64>, PartitionSelectionError> {
294    if !rows.uv_distances().needs_wavelengths() {
295        return Ok(None);
296    }
297    let window = entry.window.0;
298    catalogue
299        .windows
300        .get(window as usize)
301        .and_then(|window| window.reference_frequency.as_ref())
302        .map(casa_types::measures::frequency::MFrequency::hz)
303        .filter(|hz| hz.is_finite() && *hz > 0.0)
304        .map(|hz| Some(SPEED_OF_LIGHT_M_S / hz))
305        .ok_or_else(|| {
306            PartitionSelectionError::Inconsistent(format!(
307                "spectral window {window} has no positive reference frequency"
308            ))
309        })
310}

312/// The weights the partitions `ids` of `store` hold beside `visibility`:
313/// `WEIGHT_SPECTRUM` when any of them reads per-channel weights, else
314/// `WEIGHT`. The MSv2 adapter decides each partition's by CASA's rule, at
315/// the first of its rows; an MSv4's weights are per channel. The passes
316/// read each partition's own weights; this names them in the problem's
317/// manifest.
318pub(super) fn stored_weights(
319    store: &OpenedStore,
320    ids: impl IntoIterator<Item = PartitionId>,
321    visibility: VisibilityColumn,
322) -> WeightColumn {
323    let group = data_group(visibility);
324    let catalogue = store.catalogue();
325    let spectral = store.msv2().is_none()
326        || ids.into_iter().any(|id| {
327            catalogue
328                .partition(id)
329                .data_group(group)
330                .is_some_and(|(_, entry)| entry.weights == "WEIGHT_SPECTRUM")
331        });
332    if spectral {
333        WeightColumn::WeightSpectrum
334    } else {
335        WeightColumn::Weight
336    }
