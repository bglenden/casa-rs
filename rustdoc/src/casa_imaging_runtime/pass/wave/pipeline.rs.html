2//! A wave whose region owners accumulate each slice as soon as it is placed
3//! and their earlier slices are done, rather than all together at every
4//! slice (#449).
5//!
6//! A pass's blocks follow the observation in time, and as the sky turns the
7//! samples of one block load the strips unevenly even where the whole pass
8//! loads them evenly. With one set of row chunks every owner waits at each
9//! slice for the slowest: B3 lost about 2.7 s of a 13 s accumulation stage
10//! that way. Here the wave places into several sets of chunks
11//! ([`MajorCyclePass::chunk_sets`]); each owner accumulates the sets in
12//! placement order, as it would one set, and a set is placed into again once
13//! every owner has released it. Each owner's accumulator therefore sees the
14//! same samples in the same order as with one set, and its grids are the
15//! same bits; only the waiting moves.
16//!
17//! The owners' tasks run on the team's threads and never wait on each other
18//! or on the placer: an owner with nothing queued ends its task, and the
19//! placer, on the calling thread, starts it again with the next set. The
20//! placer waits, off the team's threads, only for a set to come free, so a
21//! wave pipelines only when its caller is not one of the team's threads
22//! ([`WorkerTeam::waits_off_team`]).
23//!
24//! An owner that fails, by error or by panic, stops the pipeline. Every lane
25//! then releases its queued sets without accumulating them, and the placer
26//! returns the failure at its next slice.
27

39/// One set of row chunks: the chunks, which the placer takes alone and the
40/// owners share; how many of them the last placement filled; and how many
41/// owners have yet to accumulate them.
42struct ChunkSet {
43    chunks: RwLock<Vec<Chunk>>,
44    filled: AtomicUsize,
45    pending: AtomicUsize,
46}
47
48/// One owner's lane: its accumulator, and the sets queued for it in
49/// placement order with whether a task is draining them.
50struct Lane<'w> {
51    owner: Mutex<Owner<'w>>,
52    queue: Mutex<(VecDeque<usize>, bool)>,
53}
54
55/// What a pipelined wave shares between its placer and its owners' tasks.
56struct Pipeline<'w> {
57    sets: Vec<ChunkSet>,
58    lanes: Vec<Lane<'w>>,
59    /// Signalled, under `released`, whenever a set comes free or the
60    /// pipeline stops.
61    released: Mutex<()>,
62    freed: Condvar,
63    /// The first error an owner met, which the placer takes.
64    failure: Mutex<Option<PassError>>,
65    /// Set, for good, once an owner has failed: the lanes release their
66    /// sets without accumulating them and the placer stops placing.
67    stopped: AtomicBool,
68}
69
70impl Pipeline<'_> {
71    /// Record `error` unless an earlier one is recorded, and stop.
72    fn fail(&self, error: PassError) {
73        self.failure
74            .lock()
75            .unwrap_or_else(PoisonError::into_inner)
76            .get_or_insert(error);
77        let _guard = self.released.lock().unwrap_or_else(PoisonError::into_inner);
78        self.stopped.store(true, Ordering::Release);
79        self.freed.notify_all();
80    }
81
82    /// Whether an owner has failed.
83    fn stopped(&self) -> bool {
84        self.stopped.load(Ordering::Acquire)
85    }
86
87    /// Take the recorded error, if an owner met one; the pipeline stays
88    /// stopped.
89    fn take_failure(&self) -> Option<PassError> {
90        self.failure
91            .lock()
92            .unwrap_or_else(PoisonError::into_inner)
93            .take()
94    }
95
96    /// Wait until no owner holds set `set` (`true`) or the pipeline stops
97    /// (`false`).
98    fn wait_free(&self, set: usize) -> bool {
99        let mut guard = self.released.lock().unwrap_or_else(PoisonError::into_inner);
100        loop {
101            if self.stopped() {
102                return false;
103            }
104            if self.sets[set].pending.load(Ordering::Acquire) == 0 {
105                return true;
106            }
107            guard = self
108                .freed
109                .wait(guard)
110                .unwrap_or_else(PoisonError::into_inner);
111        }
112    }
113}

115/// Releases one owner's hold on a set when dropped, whether the owner
116/// accumulated it, skipped it, or unwound past it.
117struct Release<'a, 'w> {
118    pipeline: &'a Pipeline<'w>,
119    set: usize,
120}
121
122impl Drop for Release<'_, '_> {
123    fn drop(&mut self) {
124        if std::thread::panicking() {
125            self.pipeline.fail(PassError::OwnerPanicked);
126        }
127        if self.pipeline.sets[self.set]
128            .pending
129            .fetch_sub(1, Ordering::AcqRel)
130            == 1
131        {
132            let _guard = self
133                .pipeline
134                .released
135                .lock()
136                .unwrap_or_else(PoisonError::into_inner);
137            self.pipeline.freed.notify_all();
138        }
139    }
140}

142/// What an owner's task reads besides the pipeline.
143#[derive(Clone, Copy)]
144struct Accumulation<'a> {
145    pass: &'a MajorCyclePass<'a>,
146    target: &'a PassDomain<'a>,
147    weight_owner: Option<usize>,
148    subtract: Subtract<'a>,
149}
150
151/// Accumulate lane `lane`'s queued sets in order until none is queued; once
152/// the pipeline has stopped, release them unaccumulated. A panic while
153/// accumulating fails the pipeline with [`PassError::OwnerPanicked`] and
154/// does not end the task, so the lane still releases every set queued
155/// behind it and the placer is never left waiting on one.
156fn drain(pipeline: &Pipeline<'_>, lane: usize, work: Accumulation<'_>) {
157    loop {
158        let set = {
159            let mut queue = pipeline.lanes[lane]
160                .queue
161                .lock()
162                .unwrap_or_else(PoisonError::into_inner);
163            let Some(set) = queue.0.pop_front() else {
164                queue.1 = false;
165                return;
166            };
167            set
168        };
169        let _release = Release { pipeline, set };
170        if pipeline.stopped() {
171            continue;
172        }
173        let chunks = pipeline.sets[set]
174            .chunks
175            .read()
176            .unwrap_or_else(PoisonError::into_inner);
177        let filled = pipeline.sets[set].filled.load(Ordering::Acquire);
178        // The owner is never used again after a panic: its poisoned mutex
179        // turns into `OwnerPanicked` when the wave takes the owners back.
180        let result = catch_unwind(AssertUnwindSafe(|| {
181            let mut owner = pipeline.lanes[lane]
182                .owner
183                .lock()
184                .map_err(|_| PassError::OwnerPanicked)?;
185            owner.accumulate(
186                work.pass,
187                work.target,
188                (lane, work.weight_owner),
189                work.subtract,
190                &chunks[..filled],
191            )
192        }))
193        .unwrap_or(Err(PassError::OwnerPanicked));
194        if let Err(error) = result {
195            pipeline.fail(error);
196        }
197    }
198}

200impl Wave<'_, '_> {
201    /// Whether this wave pipelines its owners over its sets of chunks: a
202    /// pass that can ([`crate::pass::pipelines_owners`]) with more than one
203    /// set, unless it hands its blocks to a visibility sink, which writes
204    /// each block before its placement, or its caller cannot wait on `team`
205    /// without holding one of its threads ([`WorkerTeam::waits_off_team`]):
206    /// a team of one, or a call from the team's own work. Otherwise it
207    /// places into the first set alone, as one set does.
208    pub(in crate::pass) fn pipelines(&self, sink: bool, team: &WorkerTeam) -> bool {
209        self.pass.chunk_sets > 1
210            && !sink
211            && team.waits_off_team()
212            && crate::pass::pipelines_owners(
213                self.pass.domains,
214                self.pass.backend,
215                self.pass.model.is_some(),
216            )
217    }
218
219    /// Stream `source`'s blocks through the wave with its owners pipelined
220    /// over `self.pass.chunk_sets` sets of chunks; returns the blocks
221    /// consumed. The owners, and the chunks, are the wave's again when this
222    /// returns, every owner having accumulated every placed slice unless an
223    /// owner or the stream failed.
224    pub(in crate::pass) fn stream_pipelined(
225        &mut self,
226        source: &mut dyn BoundedSource,
227        cancel: &Cancel,
228        timing: bool,
229        team: &WorkerTeam,
230    ) -> Result<u64, PassError> {
231        // Spacing is checked only for whole native rows, which need native
232        // residuals, which a pipelined pass never forms.
233        debug_assert!(!self.check_spacing, "a pipelined pass checks no spacing");
234        let pass = self.pass;
235        let target = &pass.domains[0];
236        let domain = &mut self.domains[0];
237        let weight_owner = domain.router.weight_owner();
238        let work = Accumulation {
239            pass,
240            target,
241            weight_owner,
242            subtract: Subtract {
243                model: domain.model,
244                native: None,
245            },
246        };
247        let mut sets = vec![std::mem::take(&mut self.chunks)];
248        let count = sets[0].len();
249        tracing::info!(
250            "wave owners pipelined over {} sets of {count} chunks of {} rows",
251            pass.chunk_sets,
252            self.shape.rows
253        );
254        for _ in 1..pass.chunk_sets {
255            sets.push((0..count).map(|_| Chunk::new(self.shape)).collect());
256        }
257        let pipeline = Pipeline {
258            sets: sets
259                .into_iter()
260                .map(|chunks| ChunkSet {
261                    chunks: RwLock::new(chunks),
262                    filled: AtomicUsize::new(0),
263                    pending: AtomicUsize::new(0),
264                })
265                .collect(),
266            lanes: std::mem::take(&mut self.domains[0].owners)
267                .into_iter()
268                .map(|owner| Lane {
269                    owner: Mutex::new(owner),
270                    queue: Mutex::new((VecDeque::new(), false)),
271                })
272                .collect(),
273            released: Mutex::new(()),
274            freed: Condvar::new(),
275            failure: Mutex::new(None),
276            stopped: AtomicBool::new(false),
277        };
278        let mut next = 0;
279        let mut waiting = std::time::Duration::ZERO;
280        let streamed = team
281            .in_place_scope(|scope| {
282                let streamed = stream_blocks(source, cancel, timing, |block| {
283                    let rows = block.len();
284                    for slice in slices(rows, count, self.shape.rows) {
285                        let set = next;
286                        next = (next + 1) % pipeline.sets.len();
287                        let started = std::time::Instant::now();
288                        let free = pipeline.wait_free(set);
289                        waiting += started.elapsed();
290                        if !free {
291                            return Err(pipeline
292                                .take_failure()
293                                .unwrap_or(PassError::OwnerPanicked));
294                        }
295                        let placing = std::time::Instant::now();
296                        let mut chunks = pipeline.sets[set]
297                            .chunks
298                            .write()
299                            .unwrap_or_else(PoisonError::into_inner);
300                        let mut filled = 0;
301                        for (chunk, rows) in chunks.iter_mut().zip(chunk_ranges(slice, count)) {
302                            chunk.rows = rows;
303                            filled += 1;
304                        }
305                        self.placer(block, 0, true)
306                            .place(team, &mut chunks[..filled])?;
307                        let placed = &chunks[..filled];
308                        self.domains[0].add_row_work(placed);
309                        self.samples += placed.iter().map(|chunk| chunk.placed).sum::<u64>();
310                        drop(chunks);
311                        let held = &pipeline.sets[set];
312                        held.filled.store(filled, Ordering::Release);
313                        held.pending.store(pipeline.lanes.len(), Ordering::Release);
314                        self.stages.placing += placing.elapsed();
315                        for (index, lane) in pipeline.lanes.iter().enumerate() {
316                            let mut queue =
317                                lane.queue.lock().unwrap_or_else(PoisonError::into_inner);
318                            queue.0.push_back(set);
319                            if !queue.1 {
320                                queue.1 = true;
321                                let pipeline = &pipeline;
322                                scope.spawn(move |_| drain(pipeline, index, work));
323                            }
324                        }
325                    }
326                    self.weight_base += rows * block.channels();
327                    Ok(())
328                });
329                // Every queued set is accumulated, or released unaccumulated
330                // once the pipeline stops; the scope then joins the tasks.
331                let started = std::time::Instant::now();
332                for set in 0..pipeline.sets.len() {
333                    if !pipeline.wait_free(set) {
334                        break;
335                    }
336                }
337                waiting += started.elapsed();
338                streamed
339            })
340            .expect("a pipelined wave's team has threads");
341        let failure = pipeline.take_failure();
342        self.stages.accumulating += waiting;
343        self.stages.pipelined = true;
344        let Pipeline { sets, lanes, .. } = pipeline;
345        self.chunks = sets
346            .into_iter()
347            .next()
348            .expect("the first set")
349            .chunks
350            .into_inner()
351            .unwrap_or_else(PoisonError::into_inner);
352        self.domains[0].owners = lanes
353            .into_iter()
354            .map(|lane| {
355                lane.owner
356                    .into_inner()
357                    .map_err(|_| PassError::OwnerPanicked)
358            })
359            .collect::<Result<_, _>>()?;
360        let blocks = streamed?;
361        failure.map_or(Ok(blocks), Err)
362    }
