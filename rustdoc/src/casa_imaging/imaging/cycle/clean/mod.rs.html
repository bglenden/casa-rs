20pub(super) fn run(
21    inputs: CleanInputs<'_>,
22    summary: &mut RunSummary,
23) -> Result<ImagingOutcome, ImagingError> {
24    let CleanInputs {
25        mut run,
26        mut outcome,
27        mut mvc,
28        controls,
29        mut setup,
30        mut mask_plans,
31        cancel,
32        diagnostics,
33    } = inputs;
34    setup.saved_weighted_residual = run
35        .existing
36        .as_ref()
37        .is_some_and(ExistingProducts::residual_in_image_units);
38    let positive_cutoff = setup.residual_pixel_mask;
39    let (read_after_refresh, use_formed_mask) =
40        restart_mask_refresh(&run, &mut setup, mvc.is_some());
41    let mut controller = Controller::new(&controls);
42    let mut cache = psf_cache(run.forms, &outcome.scientific);
43    // What outlives each minor cycle: the masks it forms (the latest, and the
44    // one an automatic mask evolves from, every holder sharing a
45    // generation's support), the PSF summaries it measures and a Clark
46    // refresh it caches.
47    let mut held = Reservation::none();
48    // tclean skips the major cycle after a minor cycle that cleans nothing,
49    // updates the mask and still asks `hasConverged` for the stop it reports.
50    let mut cleaned_nothing = false;
51    loop {
52        let ordinal = outcome.minor_cycles.len() + 1;
53        let name = format!("minor-cycle preparation {ordinal}");
54        // A plane preparation keeps for the solve stays charged until the
55        // solve, which the minor cycle's charge then leaves out; the solve
56        // drops the plane, and the charge goes with the minor cycle.
57        let (prepared, kept, saved_mask_charge) =
58            run_phase(name, &cancel, diagnostics, summary, || {
59                if ordinal == 1 {
60                    held = run.admit(
61                        "reconstruction masks",
62                        2 * mask_bytes(&outcome.scientific, &mask_plans)
63                            + psf_cache_bytes(&outcome.scientific)
64                            + std::mem::size_of_val(outcome.plane_rows.start_model_flux.as_slice())
65                                as u64,
66                    )?;
67                }
68                let mut preparing = run.admit(
69                    "minor-cycle preparation",
70                    prepare_bytes(
71                        &outcome.scientific,
72                        &mask_plans,
73                        &setup,
74                        &cache,
75                        run.teams.compute().workers(),
76                    ),
77                )?;
78                let refreshed = outcome.major_cycle_count > 0;
79                let (saved_masks, saved_mask_charge) = read_saved_masks(
80                    &run,
81                    mvc.as_ref(),
82                    refreshed,
83                    (read_after_refresh, positive_cutoff),
84                    outcome.scientific.normal_state().slab().total_channels(),
85                )?;
86                let prepared = prepare_minor_cycle_with_residual_masks(
87                    &outcome.scientific,
88                    &mask_plans,
89                    &setup,
90                    &mut cache,
91                    run.teams.compute(),
92                    saved_masks,
93                )?;
94                let kept = preparing.split(prepared.retained_bytes());
95                Ok::<_, ImagingError>((prepared, kept, saved_mask_charge))
96            })?;
97        let statistics = prepared.statistics;
98        // CASA's summary uses the final initminorcycle statistics, including
99        // a convergence check that stops before another minor cycle executes.
100        outcome.final_residual_statistics = Some(statistics);
101        if diagnostics.science_trace {
102            trace::residual(outcome.major_cycle_count, &statistics);
103        }
104        let stop = controller.clean_complete(&statistics);
105        if let Some(stop) = stop {
106            tracing::info!("imaging stopped: {stop:?} (stopcode {})", stop.code());
107        }
108        if cleaned_nothing {
109            outcome.stop = stop;
110            outcome.masks = Some(ReconstructionMaskSet::Domains(prepared.masks().clone()));
111            break;
112        }
113        if stop.is_some() {
114            outcome.stop = stop;
115            outcome
116                .masks
117                .get_or_insert_with(|| ReconstructionMaskSet::Domains(prepared.masks().clone()));
118            break;
119        }
120        let cycle_controls = controller.cycle_controls(&statistics);
121        if diagnostics.science_trace {
122            trace::cycle_controls(ordinal, &cycle_controls);
123        }
124        let started = Instant::now();
125        let name = format!("minor cycle {ordinal}");
126        let (minor, next_plans) = run_phase(name, &cancel, diagnostics, summary, || {
127            let cached = cache.refresh_bytes(&outcome.scientific);
128            let mut solving = run.admit(
129                "minor cycle",
130                run_bytes(
131                    &outcome.scientific,
132                    &setup,
133                    &cycle_controls,
134                    &cache,
135                    run.teams.compute().workers(),
136                )
137                .saturating_sub(prepared.retained_bytes()),
138            )?;
139            let minor = run_minor_cycle(
140                prepared,
141                &outcome.scientific,
142                &setup,
143                &cycle_controls,
144                &mut cache,
145                run.teams.compute(),
146                (&run.host, &run.policy),
147            )?;
148            let refresh = cache.refresh_bytes(&outcome.scientific);
149            if refresh > cached {
150                held.join(solving.split(refresh - cached));
151            }
152            let plans = next_masks(
153                &mask_plans,
154                &minor,
155                ordinal,
156                cycle_controls.threshold_reached,
157            )?;
158            Ok::<_, ImagingError>((minor, plans))
159        })?;
160        drop((kept, saved_mask_charge));
161        mask_plans = next_plans;
162        let global_threshold = controls
163            .threshold_jy_per_beam()
164            .max(statistics.nsigma_threshold);
165        // The outcome keeps every cycle's record, with a copy of its mask's
166        // support and its traced components (#706).
167        held.join(run.admit("minor-cycle record", record_bytes(&minor))?);
168        let cycle = account_minor_cycle(
169            &mut outcome,
170            &mut controller,
171            &minor,
172            &cycle_controls,
173            global_threshold,
174            started,
175        );
176        summary.minor_cycles = cycle;
177        summary.minor_iterations = outcome.total_minor_iterations;
178        if diagnostics.science_trace {
179            trace::minor_cycle(&outcome.minor_cycles[cycle - 1]);
180        }
181        let last = controller.budget_spent() || controller.last_cycle(&statistics);
182        if minor.summary.iterations == 0 {
183            cleaned_nothing = true;
184            continue;
185        }
186        let name = format!("major cycle {}", outcome.major_cycle_count + 1);
187        let scientific = outcome.scientific;
188        let major = run_phase(name, &cancel, diagnostics, summary, || {
189            if let Some(state) = mvc.as_mut() {
190                assert!(minor.dense.is_empty(), "MVC uses sparse Taylor updates");
191                Ok(Major {
192                    completion: state.refresh(
193                        &mut run,
194                        scientific,
195                        (minor.terms, minor.results),
196                        last,
197                    )?,
198                    visibility: None,
199                })
200            } else {
201                run.refresh(
202                    scientific,
203                    (minor.terms, minor.dense, minor.results),
204                    last,
205                    outcome.major_cycle_count == 0,
206                )
207            }
208        })?;
209        setup.saved_weighted_residual = false;
210        setup.residual_pixel_mask = use_formed_mask;
211        controller.end_major_cycle();
212        outcome.scientific = major.completion;
213        outcome.visibility_products = major.visibility;
214        outcome.major_cycle_count += 1;
215    }
216    summary.stop = outcome.stop;
217    if let Some(state) = mvc {
218        let (cube, taylor) = state.finish(outcome.scientific);
219        outcome.scientific = cube;
220        outcome.mvc_taylor = Some(taylor);
221    }
222    if run.visibility_write.is_some() && outcome.visibility_products.is_none() {
223        outcome = run_phase("model prediction", &cancel, diagnostics, summary, || {
224            predict_final_model(&mut run, outcome)
225        })?;
226    }
227    outcome.planes_per_wave = run.planes_per_wave;
228    outcome.retained = std::mem::replace(&mut run.retained, Reservation::none());
229    outcome.retained.join(held);
230    Ok(outcome)
231}

233fn psf_cache(forms: Forms, scientific: &MajorCycleCompletion) -> PsfCache {
234    if !forms.psf
235        && scientific.normal_state().basis() != casa_imaging_deconvolution::NormalBasis::Taylor
236    {
237        PsfCache::for_reused_single_psf()
238    } else {
239        PsfCache::default()
240    }
241}

243/// Measure the saved residual with its own mask at the first boundary.
244fn restart_mask_refresh(run: &Run<'_>, setup: &mut MinorCycleSetup<'_>, mvc: bool) -> (bool, bool) {
245    let positive_cutoff = setup.residual_pixel_mask;
246    let saved = run.existing.as_ref();
247    let mosaic = saved.is_some_and(ExistingProducts::reads_mosaic_residual);
248    let cube = saved.is_some_and(ExistingProducts::reads_mosaic_cube_residual);
249    let outliers = !run.forms.residual && !run.outlier_products.is_empty();
250    if mosaic || outliers || mvc && !run.forms.residual {
251        setup.residual_pixel_mask = false;
252    }
253    if cube {
254        // Positive cutoff copies the new PB's mask when the PSF pass
255        // formed it. Otherwise the controller reads a persisted mask.
256        let uses_new_pb =
257            positive_cutoff && saved.is_some_and(|saved| saved.mosaic_beam().is_none());
258        (!uses_new_pb, uses_new_pb)
259    } else {
260        (
261            (mosaic && positive_cutoff) || outliers,
262            positive_cutoff && (!outliers || run.forms.psf),
263        )
264    }
265}

267/// CASA's pixel mask determines the peak independently of CLEAN support.
268/// A positive cutoff copies a saved PB mask after a cube major cycle, or
269/// to a previously unmasked residual after a continuum major cycle.
270fn read_saved_masks(
271    run: &Run<'_>,
272    mvc: Option<&MvcState<'_>>,
273    refreshed: bool,
274    (keep_after_refresh, positive_cutoff): (bool, bool),
275    channels: usize,
276) -> Result<
277    (
278        Vec<Option<casa_imaging_deconvolution::Support>>,
279        Reservation,
280    ),
281    ImagingError,
282> {
283    if refreshed && !keep_after_refresh {
284        return Ok((Vec::new(), Reservation::none()));
285    }
286    let Some(main) = mvc
287        .and_then(MvcState::saved_taylor)
288        .or(run.existing.as_ref())
289    else {
290        return Ok((Vec::new(), Reservation::none()));
291    };
292    if main.saved_residual_mask_bytes().is_none() {
293        return Ok((Vec::new(), Reservation::none()));
294    }
295    let count = (1 + run.outlier_products.len()) * channels;
296    let mut held = run.admit(
297        "saved residual mask list",
298        (count * size_of::<Option<casa_imaging_deconvolution::Support>>()) as u64,
299    )?;
300    let mut masks = Vec::with_capacity(count);
301    for (domain, saved) in std::iter::once(main)
302        .chain(&run.outlier_products)
303        .enumerate()
304    {
305        for channel in 0..channels {
306            if refreshed && !positive_cutoff && !saved.custom_residual_mask() {
307                masks.push(None);
308                continue;
309            }
310            let (mask, mut charge) =
311                read_saved_mask(run, saved, channel, refreshed && positive_cutoff)?;
312            let pixels = mask.as_ref().map_or(0, |mask| mask.as_slice().len() as u64);
313            if pixels > 0 {
314                tracing::debug!(
315                    target: "casa_imaging::saved_residual_mask",
316                    domain,
317                    admitted_bytes = charge.memory(),
318                    "read the saved residual pixel mask"
319                );
320            }
321            charge.retain(pixels);
322            held.join(charge);
323            masks.push(mask);
324        }
325    }
326    Ok((masks, held))
327}

329fn read_saved_mask(
330    run: &Run<'_>,
331    saved: &ExistingProducts,
332    channel: usize,
333    copy_pb: bool,
334) -> Result<(Option<casa_imaging_deconvolution::Support>, Reservation), ImagingError> {
335    let bytes = saved.saved_residual_mask_bytes().unwrap_or(0);
336    if copy_pb
337        && (bytes == 0 || saved.reads_mosaic_cube_residual())
338        && let Some(beam) = saved.mosaic_beam()
339        && (!saved.reads_mosaic_cube_residual() || beam.has_generated_mask())
340    {
341        let charge = run.admit("saved PB for residual mask", beam.reading_bytes())?;
342        let [nx, ny, _, _] = beam.shape();
343        let pixels = beam
344            .read_mask(channel, 0)
345            .map_err(casa_imaging_runtime::MinorCycleRunError::from)?;
346        let mask = casa_imaging_deconvolution::Support::new(
347            casa_imaging_deconvolution::PlaneShape::new(nx, ny),
348            pixels,
349        );
350        return Ok((Some(mask), charge));
351    }
352    if bytes == 0 {
353        if copy_pb && saved.primary_beam_mask_bytes() > 0 {
354            let charge = run.admit(
355                "saved PB for residual mask",
356                saved.primary_beam_mask_bytes(),
357            )?;
358            let mask = saved.primary_beam_mask().map_err(ImagingError::Existing)?;
359            return Ok((mask, charge));
360        }
361        return Ok((None, Reservation::none()));
362    }
363    let charge = run.admit("saved residual mask", bytes)?;
364    let mask = saved
365        .residual_mask_at(channel)
366        .map_err(ImagingError::Existing)?;
367    Ok((mask, charge))
