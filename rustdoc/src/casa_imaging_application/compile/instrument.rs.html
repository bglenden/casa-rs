21fn telescopes(ms: &MeasurementSet) -> Result<BTreeSet<String>, PrepareError> {
22    let observation = ms.observation()?;
23    (0..observation.row_count())
24        .map(|row| {
25            observation
26                .string(row, "TELESCOPE_NAME")
27                .map(|name| name.trim().to_string())
28                .map_err(Into::into)
29        })
30        .collect()
31}

33/// The analytic primary beam a standard-gridder run writes `.pb` and
34/// `.image.pbcor` with.
35pub(super) fn standard_primary_beam_model(
36    ms: &MeasurementSet,
37) -> Result<AnalyticPrimaryBeamModel, PrepareError> {
38    let telescopes = telescopes(ms)?;
39    match telescopes
40        .iter()
41        .map(String::as_str)
42        .collect::<Vec<_>>()
43        .as_slice()
44    {
45        ["ALMA"] => homogeneous_dishes(ms, 10.0..13.0, AnalyticPrimaryBeamModel::CasaAlma12mAiry),
46        ["ACA"] => homogeneous_dishes(ms, 6.0..8.0, AnalyticPrimaryBeamModel::CasaAca7mAiry),
47        _ => analytic_primary_beam_model_for_telescopes(&telescopes),
48    }
49}
50
51fn homogeneous_dishes(
52    ms: &MeasurementSet,
53    diameter_range_m: Range<f64>,
54    model: AnalyticPrimaryBeamModel,
55) -> Result<AnalyticPrimaryBeamModel, PrepareError> {
56    let antenna = ms.antenna()?;
57    if antenna.row_count() == 0 {
58        return Err(PrepareError::NoAntennaDishes);
59    }
60    for row in 0..antenna.row_count() {
61        let diameter_m = antenna.dish_diameter(row)?;
62        if !diameter_m.is_finite() || !diameter_range_m.contains(&diameter_m) {
63            return Err(PrepareError::MixedAlmaDishes { row, diameter_m });
64        }
65    }
66    Ok(model)
67}
68
69fn analytic_primary_beam_model_for_telescopes(
70    telescopes: &BTreeSet<String>,
71) -> Result<AnalyticPrimaryBeamModel, PrepareError> {
72    match telescopes
73        .iter()
74        .map(String::as_str)
75        .collect::<Vec<_>>()
76        .as_slice()
77    {
78        ["EVLA"] => Ok(AnalyticPrimaryBeamModel::CasaEvlaCommon),
79        ["VLA"] => Ok(AnalyticPrimaryBeamModel::CasaVlaBand),
80        [] => Err(PrepareError::NoTelescope),
81        _ => Err(PrepareError::NoAnalyticPrimaryBeam {
82            telescopes: telescopes.iter().cloned().collect(),
83        }),
84    }
85}

87/// The instrument response of a direction-dependent gridder; `None` for the
88/// others.
89pub(super) fn scientific_instrument_model(
90    gridder: &Gridder,
91    ms: &MeasurementSet,
92) -> Result<Option<InstrumentModel>, PrepareError> {
93    let (model, dishes): (_, fn(f64) -> bool) = match gridder {
94        Gridder::Awproject(_) => (InstrumentModel::CasaEvlaWidebandAwV1, |diameter| {
95            (diameter - 25.0).abs() < 1.0
96        }),
97        Gridder::Mosaic { .. } => (
98            InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1,
99            |diameter| (diameter - 12.0).abs() < 0.5 || (diameter - 7.0).abs() < 1.0,
100        ),
101        Gridder::Standard | Gridder::Wproject { .. } => return Ok(None),
102    };
103    let telescopes = telescopes(ms)?;
104    let supported = !telescopes.is_empty()
105        && telescopes.iter().all(|name| match gridder {
106            Gridder::Awproject(_) => matches!(name.as_str(), "VLA" | "EVLA"),
107            _ => matches!(name.as_str(), "ALMA" | "ACA"),
108        });
109    if !supported {
110        return Err(PrepareError::UnsupportedInstrument {
111            telescopes: telescopes.into_iter().collect(),
112        });
113    }
114    let antenna = ms.antenna()?;
115    if antenna.row_count() == 0 {
116        return Err(PrepareError::NoAntennaDishes);
117    }
118    for row in 0..antenna.row_count() {
119        let diameter_m = antenna.dish_diameter(row)?;
120        if !(diameter_m.is_finite() && dishes(diameter_m)) {
121            return Err(PrepareError::DishOutsideResponse {
122                model,
123                row,
124                diameter_m,
125            });
126        }
127    }
128    Ok(Some(model))
129}

131/// Where each row points: the POINTING table under tclean `usepointing`
132/// (AW samples it at the visibility time, mosaic interpolates it); each
133/// row's FIELD direction for a mosaic or A-projection run without it
134/// (`usepointing=False`, the tclean default); the phase-tracking centre
135/// for a direction-independent run.
136pub(super) fn pointing_centre_law(gridder: &Gridder) -> PointingCentreLaw {
137    match gridder {
138        Gridder::Awproject(aw) if aw.usepointing => {
139            PointingCentreLaw::Observation(ObservationPointingLaw::new(
140                PointingDirectionColumn::Direction,
141                PointingDirectionSemantic::AntennaBoresight,
142                PointingTimeSampling::VisibilityTime,
143                PointingInterpolation::Nearest,
144                PointingExtrapolation::HoldNearest,
145                MissingPointingPolicy::UsePhaseTrackingCentre,
146            ))
147        }
148        Gridder::Mosaic {
149            usepointing: true, ..
150        } => PointingCentreLaw::Observation(ObservationPointingLaw::new(
151            PointingDirectionColumn::Direction,
152            PointingDirectionSemantic::AntennaBoresight,
153            PointingTimeSampling::VisibilityTimeCentroid,
154            PointingInterpolation::GreatCircleShortestArc,
155            PointingExtrapolation::Reject,
156            MissingPointingPolicy::Reject,
157        )),
158        Gridder::Awproject(_) | Gridder::Mosaic { .. } => PointingCentreLaw::FieldCentre,
159        Gridder::Standard | Gridder::Wproject { .. } => PointingCentreLaw::PhaseTrackingCentre,
160    }
161}

163#[cfg(test)]
164mod tests {
165    use std::num::NonZeroUsize;
166    use std::path::PathBuf;
167
168    use casa_imaging_model::ProductNormalization;
169
170    use super::*;
171    use crate::{AwCfSource, AwProjection};
172
173    fn aw(usepointing: bool) -> Gridder {
174        Gridder::Awproject(AwProjection {
175            wprojplanes: NonZeroUsize::new(32),
176            usepointing,
177            normtype: ProductNormalization::FlatNoise,
178            cf_resident_mb: 256,
179            pointingoffsetsigdev: Vec::new(),
180            cf_source: AwCfSource::CasaImport {
181                cfcache: PathBuf::from("cf"),
182            },
183        })
184    }
185
186    #[test]
187    fn aw_pointing_compiles_casa_visibility_sampling_law() {
188        let PointingCentreLaw::Observation(law) = pointing_centre_law(&aw(true)) else {
189            panic!("AW usepointing must compile an observation pointing law");
190        };
191        assert_eq!(law.time_sampling(), PointingTimeSampling::VisibilityTime);
192        assert_eq!(law.interpolation(), PointingInterpolation::Nearest);
193        assert_eq!(law.extrapolation(), PointingExtrapolation::HoldNearest);
194        assert_eq!(law.missing(), MissingPointingPolicy::UsePhaseTrackingCentre);
195        assert!(matches!(
196            pointing_centre_law(&aw(false)),
197            PointingCentreLaw::FieldCentre
198        ));
199    }
200
201    #[test]
202    fn mosaic_points_at_the_field_centre_unless_usepointing() {
203        let mosaic = |usepointing| Gridder::Mosaic {
204            usepointing,
205            normtype: ProductNormalization::FlatNoise,
206        };
207        assert!(matches!(
208            pointing_centre_law(&Gridder::Standard),
209            PointingCentreLaw::PhaseTrackingCentre
210        ));
211        assert!(matches!(
212            pointing_centre_law(&mosaic(false)),
213            PointingCentreLaw::FieldCentre
214        ));
215        let PointingCentreLaw::Observation(law) = pointing_centre_law(&mosaic(true)) else {
216            panic!("mosaic usepointing must compile an observation pointing law");
217        };
218        assert_eq!(
219            law.time_sampling(),
220            PointingTimeSampling::VisibilityTimeCentroid
221        );
222        assert_eq!(
223            law.interpolation(),
224            PointingInterpolation::GreatCircleShortestArc
225        );
226        assert_eq!(law.extrapolation(), PointingExtrapolation::Reject);
227        assert_eq!(law.missing(), MissingPointingPolicy::Reject);
228    }

230    #[test]
231    fn standard_primary_beam_model_is_explicit_and_fails_closed() {
232        let named = |name: &str| BTreeSet::from([name.to_string()]);
233        assert_eq!(
234            analytic_primary_beam_model_for_telescopes(&named("EVLA")).expect("EVLA model"),
235            AnalyticPrimaryBeamModel::CasaEvlaCommon
236        );
237        assert_eq!(
238            analytic_primary_beam_model_for_telescopes(&named("VLA")).expect("VLA model"),
239            AnalyticPrimaryBeamModel::CasaVlaBand
240        );
241        assert!(analytic_primary_beam_model_for_telescopes(&named("UNKNOWN")).is_err());
242        assert!(analytic_primary_beam_model_for_telescopes(&BTreeSet::new()).is_err());
243    }
