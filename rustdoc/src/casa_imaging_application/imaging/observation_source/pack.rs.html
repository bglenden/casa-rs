22/// The block layout of one partition's batches in one wave: which of a
23/// batch's channels and correlations a block holds.
24#[derive(Debug)]
25pub(super) struct Layout {
26    /// The batch channel of each channel a block holds.
27    channels: Vec<usize>,
28    /// Their native indices, the block's channel indices.
29    channel_indices: Vec<u32>,
30    /// Their stored frequencies, the continuum fit's abscissae.
31    native_hz: Vec<f64>,
32    /// Half their stored widths, the extent a frequency band selects
33    /// them by.
34    half_widths_hz: Vec<f64>,
35    /// The routed products, in the operator's order.
36    correlations: Vec<Correlation>,
37    /// Their stored indices, the block's correlation indices.
38    correlation_indices: Vec<u32>,
39    /// Whether each enters Stokes I.
40    parallel: Vec<bool>,
41}
42
43impl Layout {
44    /// The layout of `partition`'s selected channels `ordinals` (indices
45    /// into its selected channels), read as the native channels from the
46    /// first of them.
47    pub(super) fn new(
48        partition: &SelectedPartition,
49        entry: &PartitionEntry,
50        ordinals: std::ops::Range<usize>,
51    ) -> Result<Self, SourceError> {
52        let channel_indices = partition.channels[ordinals].to_vec();
53        let first = channel_indices[0] as usize;
54        let correlation_indices = partition
55            .correlations
56            .iter()
57            .map(|product| {
58                entry
59                    .correlation_index(*product)
60                    .map(|index| index as u32)
61                    .ok_or_else(|| format!("partition lacks the routed product {product:?}"))
62            })
63            .collect::<Result<Vec<_>, _>>()?;
64        Ok(Self {
65            channels: channel_indices
66                .iter()
67                .map(|channel| *channel as usize - first)
68                .collect(),
69            native_hz: channel_indices
70                .iter()
71                .map(|channel| entry.frequencies_hz[*channel as usize])
72                .collect(),
73            half_widths_hz: channel_indices
74                .iter()
75                .map(|channel| entry.channel_widths_hz[*channel as usize].abs() / 2.0)
76                .collect(),
77            channel_indices,
78            parallel: partition
79                .correlations
80                .iter()
81                .map(|product| product.contributes_to_stokes_i())
82                .collect(),
83            correlations: partition.correlations.clone(),
84            correlation_indices,
85        })
86    }
87
88    /// The native channels a binding reads: from the first channel the
89    /// block holds to the last.
90    pub(super) fn native_channels(&self) -> std::ops::Range<usize> {
91        let first = self.channel_indices[0] as usize;
92        first
93            ..*self
94                .channel_indices
95                .last()
96                .expect("a layout holds a channel") as usize
97                + 1
98    }
99}

101/// Which samples of `batch` the selection images, into `keep`: those of a
102/// selected scan and baseline that are no padding (every flag set and every
103/// weight not finite, as the MSv2 adapter pads and XRADIO's NaN cells read)
104/// and whose UV distance it selects. Fields and intents were selected by the
105/// request.
106pub(super) fn select_samples(
107    batch: &VisibilityBatch<'_>,
108    selection: &PartitionSelection,
109    partition: &SelectedPartition,
110    keep: &mut Vec<bool>,
111) -> Result<(), SourceError> {
112    let flags = batch.group.flags.ok_or("the batch carries no flags")?;
113    let weights = batch.group.weights.ok_or("the batch carries no weights")?;
114    let uvw = batch.group.uvw.ok_or("the batch carries no UVW")?;
115    keep.clear();
116    let baselines = batch.samples_per_time();
117    for time in 0..batch.times() {
118        let scan_selected = selection.time_holds(batch, time)?;
119        keep.extend((time * baselines..(time + 1) * baselines).map(|sample| {
120            if !scan_selected || !selection.baseline_holds(batch.baselines[sample % baselines]) {
121                return false;
122            }
123            let flags = flags.index_axis(ndarray::Axis(0), sample);
124            let weights = weights.index_axis(ndarray::Axis(0), sample);
125            let padding =
126                flags.iter().all(|flag| *flag) && weights.iter().all(|weight| !weight.is_finite());
127            !padding && selection.uv_holds(partition, uvw.metres_at(sample))
128        }));
129    }
130    Ok(())
131}

151/// Buffers packing reuses from batch to batch.
152#[derive(Debug, Default)]
153pub(super) struct PackScratch {
154    projections: Vec<DomainProjection>,
155    frequencies_hz: Vec<f64>,
156    /// Whether the window's frequency bands select each channel at the time
157    /// packed; empty when every channel is read.
158    in_band: Vec<bool>,
159    /// Each routed product's place along the batch's correlations.
160    routed: Vec<usize>,
161    /// One sample's weight per delivered product, for weights stored per
162    /// correlation.
163    row_weights: Vec<f32>,
164}
165
166impl PackScratch {
167    /// At most what the buffers come to hold for blocks of `layout`'s rows,
168    /// each growing at most to twice what it needs.
169    pub(super) fn bound(layout: BlockShape) -> usize {
170        2 * (layout.domains * size_of::<DomainProjection>()
171            + layout.channels() * (size_of::<f64>() + size_of::<bool>())
172            + layout.correlations * (size_of::<usize>() + size_of::<f32>()))
173    }
174}

176/// One batch's arrays as flat slices, `[sample][channel][correlation]`.
177struct Arrays<'b> {
178    values: Option<&'b [Complex32]>,
179    flags: &'b [bool],
180    /// Per cell, or `None` for weights stored per correlation.
181    weights: Option<&'b [f32]>,
182    weight_view: ndarray::ArrayView3<'b, f32>,
183    channels: usize,
184    correlations: usize,
185}
186
187impl<'b> Arrays<'b> {
188    fn new(batch: &VisibilityBatch<'b>, values: bool) -> Result<Self, SourceError> {
189        let standard = "a batch array is not in standard order";
190        let flags = batch.group.flags.ok_or("the batch carries no flags")?;
191        let weights = batch.group.weights.ok_or("the batch carries no weights")?;
192        let per_correlation = batch.channel_count() > 1 && weights.strides()[1] == 0;
193        Ok(Self {
194            values: values
195                .then(|| {
196                    batch
197                        .group
198                        .values
199                        .ok_or("the batch carries no visibilities")?
200                        .to_slice()
201                        .ok_or(standard)
202                })
203                .transpose()?,
204            flags: flags.to_slice().ok_or(standard)?,
205            weights: if per_correlation {
206                None
207            } else {
208                Some(weights.to_slice().ok_or(standard)?)
209            },
210            weight_view: weights,
211            channels: batch.channel_count(),
212            correlations: batch.correlations.len(),
213        })
214    }
215}

217impl Packer<'_> {
218    /// Pack `batch`'s samples `keep` names into `out` under `layout`, with
219    /// `geometry` applied to those samples; the rows are keyed as store
220    /// `store`'s ([`StoreKeys`]). Unread visibilities are zero.
221    /// A non-finite input fails the run under [`FiniteValuePolicy`], naming
222    /// the sample.
223    pub(super) fn pack(
224        &self,
225        (batch, store, partition, layout): (
226            &VisibilityBatch<'_>,
227            usize,
228            &SelectedPartition,
229            &Layout,
230        ),
231        geometry: &BatchGeometry,
232        keep: &[bool],
233        values: bool,
234        scratch: &mut PackScratch,
235        out: &mut NativeBlock,
236    ) -> Result<(), SourceError> {
237        out.reset(
238            self.charts,
239            &layout.channel_indices,
240            &layout.correlation_indices,
241        );
242        let arrays = Arrays::new(batch, values)?;
243        scratch.routed.clear();
244        for product in &layout.correlations {
245            scratch.routed.push(
246                batch
247                    .correlations
248                    .iter()
249                    .position(|delivered| delivered == product)
250                    .ok_or("the batch lacks a routed product")?,
251            );
252        }
253        let reject_flagged = matches!(
254            self.problem.numerics().finite_values(),
255            FiniteValuePolicy::RejectAll
256        );
257        let per_time = batch.samples_per_time();
258        let origin = MjdHighPrec::new(0.0, 0.0);
259        let uvw = batch.group.uvw.ok_or("the batch carries no UVW")?;
260        // An imaging pass reads a time's channels as CASA's frequency
261        // selection does; the density pass reads every selected channel.
262        let bands = partition.frequency_bands.as_ref().filter(|_| values);
263        let mut packed_time = None;
264        for sample in (0..batch.samples()).filter(|sample| keep[*sample]) {
265            let time = sample / per_time;
266            if packed_time != Some(time) {
267                let frequencies = geometry.frequencies_hz(time);
268                scratch.frequencies_hz.clear();
269                scratch
270                    .frequencies_hz
271                    .extend(layout.channels.iter().map(|channel| frequencies[*channel]));
272                scratch.in_band.clear();
273                if let Some(bands) = bands {
274                    let factor = self.plans[store].frequency_factor(
275                        bands.frame().frequency_ref(),
276                        batch.field[time],
277                        batch.time.seconds_since(time, origin),
278                        batch.time.reference(),
279                    )?;
280                    in_band(
281                        bands,
282                        factor,
283                        layout,
284                        &scratch.frequencies_hz,
285                        &mut scratch.in_band,
286                    );
287                }
288                packed_time = Some(time);
289            }
290            let row = Row {
291                batch,
292                partition,
293                sample,
294                time_s: batch.time.seconds_since(time, origin),
295            };
296            self.project(geometry, sample, &mut scratch.projections)?;
297            let nonfinite_uvw = !scratch.projections.iter().all(|projection| {
298                projection.uvw_m.iter().all(|value| value.is_finite())
299                    && projection.phase_shift_m.is_finite()
300            });
301            if arrays.weights.is_none() {
302                scratch.row_weights.clear();
303                scratch.row_weights.extend(
304                    (0..arrays.correlations)
305                        .map(|product| arrays.weight_view[[sample, 0, product]]),
306                );
307            }
308            let flagged = layout.channels.iter().all(|channel| {
309                scratch.routed.iter().all(|product| {
310                    arrays.flags
311                        [(sample * arrays.channels + channel) * arrays.correlations + product]
312                })
313            });
314            let baseline = batch.baselines[sample % per_time];
315            let field = batch.field[time];
316            let header = NativeRowHeader {
317                row_flag: flagged || nonfinite_uvw,
318                context: RowContext {
319                    original_uvw_m: Some(uvw.metres_at(sample)),
320                    time_s: row.time_s,
321                    antennas: [baseline.antenna1.0, baseline.antenna2.0],
322                    antenna_types: antenna_types(
323                        geometry.antenna_responses(sample),
324                        &self.keys.dish_classes,
325                    ),
326                    parallactic_angle_rad: geometry.parallactic_angles(sample).unwrap_or([0.0; 2]),
327                    field: self.keys.field(store, field.0),
328                    spectral_window: self.keys.window(store, partition.spectral_window),
329                },
330                address: RowAddress::UNADDRESSED,
331            };
332            let rule = self
333                .problem
334                .visibility_transform()
335                .and_then(|transform| transform.rule(field.0 as i32, partition.spectral_window));
336            let row_weights = &scratch.row_weights;
337            let routed = &scratch.routed;
338            let in_band = &scratch.in_band;
339            out.push_row_with(
340                header,
341                &scratch.projections,
342                &scratch.frequencies_hz,
343                |values_out, weights_out, flags_out| {
344                    fill_samples(
345                        &row,
346                        (&arrays, row_weights, routed, layout),
347                        (values_out, weights_out, flags_out),
348                        reject_flagged,
349                    )?;
350                    // A channel the time's bands miss is not read: no
351                    // sample of it, or interpolated from it, is imaged.
352                    for (cells, _) in flags_out
353                        .chunks_mut(routed.len())
354                        .zip(in_band)
355                        .filter(|(_, selected)| !**selected)
356                    {
357                        cells.fill(true);
358                    }
359                    // A row's uvw matters only if one of its samples is imaged.
360                    let imaged = !flagged && flags_out.iter().any(|flag| !*flag);
361                    if nonfinite_uvw && (reject_flagged || imaged) {
362                        return Err(nonfinite_input("uvw", &row.at(None)));
363                    }
364                    let Some(rule) = rule else {
365                        return Ok(());
366                    };
367                    let row = ContinuumRow {
368                        channels: &layout.channel_indices,
369                        frequencies_hz: &layout.native_hz,
370                        parallel: &layout.parallel,
371                        real: false,
372                        row_flag: flagged,
373                    };
374                    if values {
375                        subtract_continuum(rule, &row, values_out, weights_out, flags_out)
376                    } else {
377                        flag_unimaged_channels(rule, &row, flags_out)
378                    }
379                },
380            )?;
381        }
382        Ok(())
383    }
384
385    /// Sample `sample`'s projection on every chart into `projections`.
386    fn project(
387        &self,
388        geometry: &BatchGeometry,
389        sample: usize,
390        projections: &mut Vec<DomainProjection>,
391    ) -> Result<(), SourceError> {
392        projections.clear();
393        for chart in 0..self.charts {
394            let projection = geometry.projection(sample, chart);
395            let pointing_offset_rad = match self.pointing.get(chart) {
396                Some(direction) => pointing_offset(
397                    *direction,
398                    geometry.aw_pointing_pixel(sample, chart),
399                    geometry
400                        .chart_pointing(sample, chart)
401                        .unwrap_or(geometry.pointing(sample).antenna1),
402                )?,
403                None => [0.0; 2],
404            };
405            projections.push(DomainProjection {
406                uvw_m: projection.uvw_m,
407                phase_shift_m: projection.phase_shift_m,
408                pointing_offset_rad,
409            });
410        }
411        Ok(())
412    }
413}

415/// Whether `bands` select each of `layout`'s channels at one time, into
416/// `out`: the channel's extent, its centre `frequencies_hz` in the rows'
417/// frame less and plus half its width scaled as its centre is, meets a
418/// band scaled by `factor`, the bands' frame's factor to the rows' frame
419/// at the time ([`GeometryPlan::frequency_factor`]). CASA converts the
420/// bands to the window's frame instead and compares stored extents
421/// (`VisibilityIteratorImpl2::makeChannelSelectorF`); both conversions are
422/// one Doppler factor, so the comparisons agree.
423fn in_band(
424    bands: &FrequencyBands,
425    factor: f64,
426    layout: &Layout,
427    frequencies_hz: &[f64],
428    out: &mut Vec<bool>,
429) {
430    out.extend(
431        frequencies_hz
432            .iter()
433            .zip(&layout.native_hz)
434            .zip(&layout.half_widths_hz)
435            .map(|((centre, native), half_width)| {
436                let half_width = half_width * centre / native;
437                bands.meets(factor, [centre - half_width, centre + half_width])
438            }),
439    );
440}

442/// One sample being packed, for naming it in an error.
443struct Row<'r, 'b> {
444    batch: &'r VisibilityBatch<'b>,
445    partition: &'r SelectedPartition,
446    sample: usize,
447    /// Its time, in seconds from MJD 0 on the partition's scale.
448    time_s: f64,
449}
450
451impl Row<'_, '_> {
452    /// Where the sample is: its window, time, baseline and, for one value,
453    /// its channel and correlation.
454    fn at(&self, cell: Option<(u32, Correlation)>) -> String {
455        let baseline = self.batch.baselines[self.sample % self.batch.samples_per_time()];
456        let mut at = format!(
457            "spectral window {} at {:.3} s (MJD, {:?}), baseline {}–{}",
458            self.partition.spectral_window,
459            self.time_s,
460            self.batch.time.reference(),
461            baseline.antenna1.0,
462            baseline.antenna2.0,
463        );
464        if let Some((channel, correlation)) = cell {
465            at.push_str(&format!(", channel {channel}, correlation {correlation:?}"));
466        }
467        at
468    }
469}

471/// The error for a non-finite `what` at `at`, naming the remedy.
472fn nonfinite_input(what: &str, at: &str) -> SourceError {
473    format!(
474        "an unflagged {what} at {at} is not finite; CASA would image it as NaN everywhere. \
475         Flag the non-finite data (CASA `flagdata`) and run again"
476    )
477    .into()
478}

480/// Write one sample's selected values, weights and flags in channel and
481/// correlation order. A non-finite value or weight in an unflagged sample
482/// fails the row; in a flagged one it is ignored, unless `reject_flagged`;
483/// either way its flag is set.
484fn fill_samples(
485    row: &Row<'_, '_>,
486    (arrays, row_weights, routed, layout): (&Arrays<'_>, &[f32], &[usize], &Layout),
487    (values, weights, flags): (&mut [Complex32], &mut [f32], &mut [bool]),
488    reject_flagged: bool,
489) -> Result<(), SourceError> {
490    let stride = arrays.correlations;
491    let base = row.sample * arrays.channels * stride;
492    let mut cell = 0;
493    for (ordinal, channel) in layout.channels.iter().enumerate() {
494        let start = base + channel * stride;
495        for (k, product) in routed.iter().enumerate() {
496            let index = start + product;
497            let sample = arrays
498                .values
499                .map_or(Complex32::default(), |values| values[index]);
500            let weight = arrays
501                .weights
502                .map_or_else(|| row_weights[*product], |weights| weights[index]);
503            let flagged = arrays.flags[index];
504            let finite_value = sample.re.is_finite() && sample.im.is_finite();
505            let nonfinite = !(finite_value && weight.is_finite());
506            if nonfinite && (reject_flagged || !flagged) {
507                return Err(nonfinite_input(
508                    if finite_value { "weight" } else { "visibility" },
509                    &row.at(Some((
510                        layout.channel_indices[ordinal],
511                        layout.correlations[k],
512                    ))),
513                ));
514            }
515            values[cell] = sample;
516            weights[cell] = weight;
517            flags[cell] = flagged || nonfinite;
518            cell += 1;
519        }
520    }
521    Ok(())
522}

524/// The mosaic dish index of each antenna of a row: its aperture class's
525/// position in the selection's `classes`, the order
526/// [`crate::imaging::measurement::domain_operator`] lists the dishes in; a row
527/// without response classes (every problem without an instrument model)
528/// is on the first dish.
529fn antenna_types(
530    responses: Option<SelectedAntennaResponses>,
531    classes: &[AntennaResponseClass],
532) -> [u8; 2] {
533    let index = |class: AntennaResponseClass| {
534        classes
535            .iter()
536            .position(|known| *known == class)
537            .map_or(0, |index| u8::try_from(index).expect("few dish classes"))
538    };
539    responses.map_or([0, 0], |responses| {
540        [index(responses.antenna1), index(responses.antenna2)]
541    })
542}

544/// The offset of a row's pointing from `direction`'s reference pixel as
545/// direction cosines along the image axes: the A-projection pointing pixel
546/// when the selection evaluated one, else the antenna-1 pointing direction
547/// on the image plane (`SimplePBConvFunc::findConvFunction` takes the
548/// buffer's first `direction1`; under tclean `usepointing=False` the
549/// selection evaluates it as the field's phase-tracking centre), placed in
550/// the chart's frame and less a moving source's shift when the geometry
551/// placed it ([`BatchGeometry::chart_pointing`]).
552fn pointing_offset(
553    direction: DirectionCoordinateSpec,
554    aw_pixel: Option<[f64; 2]>,
555    pointing: SkyDirection,
556) -> Result<[f64; 2], SourceError> {
557    let pixel = match aw_pixel {
558        Some(pixel) => pixel,
559        None => {
560            if pointing.frame() != direction.reference_direction().frame() {
561                return Err("a row's pointing direction is not in the image frame".into());
562            }
563            direction_world_to_pixel(
564                direction,
565                [pointing.longitude_rad(), pointing.latitude_rad()],
566            )?
567        }
568    };
569    let reference = direction.reference_pixel();
570    let increment = direction.increment_rad();
571    Ok([
572        (pixel[0] - reference[0]) * increment[0],
573        (pixel[1] - reference[1]) * increment[1],
574    ])
575}
