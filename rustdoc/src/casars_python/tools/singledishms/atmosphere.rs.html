3use std::path::Path;
4use std::sync::PoisonError;
5
6use casa_calibration::SingleDishAtmosphere;
7use pyo3::exceptions::PyRuntimeError;
8use pyo3::prelude::*;
9use pyo3::types::PyDict;
10
11use super::PySingleDishMs;
12use crate::tools::table::casa_record;
13
14#[pymethods]
15impl PySingleDishMs {
16    #[pyo3(signature = (config = None, datacolumn = "data", outfile = ""), text_signature = "($self, config={}, datacolumn='data', outfile='')")]
17    fn atmcor(
18        &self,
19        py: Python<'_>,
20        config: Option<&Bound<'_, PyDict>>,
21        datacolumn: &str,
22        outfile: &str,
23    ) -> PyResult<bool> {
24        let config = casa_record(py, config)?;
25        py.detach(|| {
26            self.session
27                .lock()
28                .unwrap_or_else(PoisonError::into_inner)
29                .atmcor(SingleDishAtmosphere {
30                    config: &config,
31                    datacolumn,
32                    outfile: Path::new(outfile),
33                    overwrite: true,
34                })
35                .map(|_| true)
36        })
37        .map_err(PyRuntimeError::new_err)
38    }
39}
