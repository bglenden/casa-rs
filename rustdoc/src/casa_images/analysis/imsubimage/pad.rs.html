17/// The border and source selection for direction-plane padding.
18pub struct ImagePadOptions<'a> {
19    /// Positive, equal border width at both ends of each direction axis.
20    pub pixels: usize,
21    /// The numeric value stored in the border.
22    pub value: f32,
23    /// Whether the border pixels are valid.
24    pub padmask: bool,
25    /// An optional CASA region selected before padding.
26    pub region: Option<SubimageRegion<'a>>,
27    /// Rectangular spatial selection.
28    pub box_selection: Option<&'a str>,
29    /// Spectral channel selection.
30    pub channels: Option<&'a str>,
31    /// Polarization selection.
32    pub stokes: Option<&'a str>,
33    /// A LEL expression combined with input validity.
34    pub mask: Option<&'a str>,
35    /// Stretch compatible singleton dimensions of the expression.
36    pub stretch: bool,
37    /// Final destination, if persistent, to protect named mask inputs.
38    pub destination: Option<&'a Path>,
39}

41/// The resolved input section, padded shape, and shifted coordinates.
42pub struct ImagePad<'a> {
43    /// The final geometry and image metadata.
44    pub section: SubimagePlan,
45    placement: Vec<usize>,
46    options: ImagePadOptions<'a>,
47}
48
49impl<'a> ImagePad<'a> {
50    /// Resolve the input selection and padded output geometry.
51    ///
52    /// # Errors
53    /// The width, direction plane, selection, or output coordinate system is invalid.
54    pub fn new(
55        image: &dyn ImageInterface<f32>,
56        coordinates: RecordValue,
57        options: ImagePadOptions<'a>,
58        beam_record: &RecordValue,
59    ) -> Result<Self, ImageError> {
60        if options.pixels == 0 {
61            return Err(ImageError::CasaSelection(
62                "Value of npixels must be greater than zero".into(),
63            ));
64        }
65        if image.find_axis(CoordinateType::Direction).is_none() {
66            return Err(ImageError::CasaSelection(
67                "Image does not have required coordinate Direction".into(),
68            ));
69        }
70        let axes = direction_plane_axes(image)?;
71        let selection = SubimageSelection::resolve_section(
72            image,
73            options.box_selection,
74            options.region,
75            options.channels.filter(|text| {
76                !text.trim().eq_ignore_ascii_case("all")
77                    && image.find_axis(CoordinateType::Spectral).is_some()
78            }),
79            options.stokes.filter(|text| {
80                !text.trim().eq_ignore_ascii_case("all")
81                    && image.find_axis(CoordinateType::Stokes).is_some()
82            }),
83        )?;
84        if selection.has_multiple_regions {
85            return Err(ImageError::CasaSelection(
86                "Only a single n-dimensional rectangular region is supported".into(),
87            ));
88        }
89        let mut section = SubimagePlan::from_selection(
90            image,
91            Some(coordinates),
92            selection,
93            false,
94            &[],
95            Some(beam_record),
96        )?;
97        let mut placement = vec![0; section.shape.len()];
98        let too_large = || ImageError::CasaSelection("Padded image shape is too large".into());
99        let both_edges = options.pixels.checked_mul(2).ok_or_else(too_large)?;
100        for axis in axes {
101            placement[axis] = options.pixels;
102            section.shape[axis] = section.shape[axis]
103                .checked_add(both_edges)
104                .ok_or_else(too_large)?;
105        }
106        section
107            .shape
108            .iter()
109            .try_fold(std::mem::size_of::<f32>(), |size, axis| {
110                size.checked_mul(*axis)
111                    .filter(|size| *size <= isize::MAX as usize)
112            })
113            .ok_or_else(too_large)?;
114        let mut record = section
115            .coordinate_record
116            .take()
117            .unwrap_or_else(|| section.coordinates.to_record());
118        let original = section.coordinates.reference_pixel();
119        for axis in axes {
120            CoordinateSystem::set_reference_pixel_in_record(
121                &mut record,
122                axis,
123                original[axis] + options.pixels as f64,
124            )?;
125        }
126        section.coordinates = section.coordinates.restore_record_with_layout(&record)?;
127        section.coordinate_record = Some(record);
128        Ok(Self {
129            section,
130            placement,
131            options,
132        })
133    }
134
135    /// Source plane indices retained on a per-plane beam axis.
136    pub fn beam_planes(&self, axis: Option<usize>, planes: usize) -> Vec<usize> {
137        self.section.beam_planes(axis, planes)
138    }
139
140    /// Fill an allocated output and stream the interior values and mask into it.
141    ///
142    /// # Errors
143    /// Returns an error for invalid mask expressions or input/output I/O.
144    pub fn copy_to(
145        &self,
146        image: &dyn ImageInterface<f32>,
147        output: &mut dyn MutableImageInterface<f32>,
148    ) -> Result<(), ImageError> {
149        let selection = &self.section.selection;
150        let reader = PixelSections {
151            image,
152            selection,
153            mask: PreparedMask::new_for_output(
154                self.options.mask,
155                image.shape(),
156                image.coordinates(),
157                self.options.stretch,
158                self.options.destination,
159            )?,
160        };
161        self.section.copy_metadata(image, output)?;
162        output.set(self.options.value)?;
163        let ones = vec![1; image.ndim()];
164        let shape = selection.packed_shape();
165        let stride = selection.stride.as_deref().unwrap_or(&ones);
166        let interior = StridedSection {
167            start: &selection.start,
168            shape: &shape,
169            stride,
170        };
171        let padmask = self.options.padmask;
172        let mut mask_name = if padmask {
173            None
174        } else {
175            Some(output.create_filled_pixel_mask(false)?)
176        };
177        for tile in interior.cursors(image.nice_cursor_shape()) {
178            let tile = tile?;
179            let input_origin = interior.source_start(&tile.position);
180            let (pixels, validity) = reader.read(&input_origin, &tile.shape, stride)?;
181            let output_origin: Vec<_> = tile
182                .position
183                .iter()
184                .zip(&self.placement)
185                .map(|(&index, &offset)| index + offset)
186                .collect();
187            output.put_slice(&pixels, &output_origin)?;
188            if padmask
189                && validity
190                    .as_ref()
191                    .is_none_or(|mask| mask.iter().all(|&valid| valid))
192            {
193                continue;
194            }
195            if mask_name.is_none() {
196                mask_name = Some(output.create_filled_pixel_mask(true)?);
197            }
198            let validity = validity.unwrap_or_else(|| ArrayD::from_elem(IxDyn(&tile.shape), true));
199            output.put_mask_slice(
200                mask_name.as_deref().expect("active mask"),
201                &validity,
202                &output_origin,
203            )?;
204        }
205        Ok(())
206    }
