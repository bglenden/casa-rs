55fn aw_pointing_plan_retained_byte_ceiling(
56    problem: &CompiledProblem,
57    source: &ObservationSource,
58) -> Result<usize, BoundObservationSourceError> {
59    let enabled = problem
60        .science()
61        .measurement_equation()
62        .aw_projection()
63        .is_some_and(|contract| contract.use_pointing());
64    if !enabled {
65        return Ok(0);
66    }
67    usize::try_from(source.selection().rows().selected_row_count())
68        .ok()
69        .and_then(|rows| rows.checked_mul(AW_POINTING_PLAN_BYTES_PER_SELECTED_ROW))
70        .and_then(|bytes| bytes.checked_add(size_of::<AwPointingEpochPlan>()))
71        .ok_or(BoundObservationSourceError::MeasurementOverflow)
72}

91/// A retained, read-locked MeasurementSet bound to one compiled source selection.
92///
93/// Construction opens the retained storage capability and metadata needed to plan the bounded
94/// content buffers. Each pass walks MAIN once, selecting rows with the compiled row predicate
95/// and reading their values: in physical order with one DATA_DESCRIPTION selected, and with
96/// several one DATA_DESCRIPTION after another, physical order within each, so a pass meets
97/// each spectral window's convolution functions together instead of cycling through every
98/// window's (#659). The first traversal records that walk ([`MainRowWalk`]).

104    walk: std::sync::OnceLock<Arc<MainRowWalk>>,
105    pointing_catalog: Option<PreparedSelectedPointingCatalog>,
106    aw_pointing_plan: Option<AwPointingEpochPlan>,
107    content_plan: SelectedObservationContentPlan,
108}

201        measurement_set: &MeasurementSet,
202        problem: &CompiledProblem,
203        source: &ObservationSource,
204        shared_bytes: SelectedObservationSharedBytes,

223            measurement_set,
224            problem,
225            source,
226            shared_bytes.with_source_plan_retained_bytes(aw_pointing_plan_retained_byte_ceiling(
227                problem, source,
228            )?),
229            maximum_pointing_polynomial_terms,
230            catalog,
231            scan_bytes_per_row,

235    #[allow(clippy::too_many_arguments)]
236    fn from_locked_measurement_set(
237        problem: &CompiledProblem,
238        source: &ObservationSource,
239        measures: &SelectedObservationMeasures,
240        shared_bytes: SelectedObservationSharedBytes,
241        content_budget: SelectedObservationContentBudget,
242        measurement_set: MeasurementSet,
243        ephemeris: Option<&Arc<crate::SelectedObservationEphemeris>>,
244        pointing_query_domain: Option<&SelectedPointingQueryDomain>,
245    ) -> Result<Self, BoundObservationSourceError> {
246        debug_assert!(
247            construction_scratch_fits_slack(source, pointing_query_domain),
248            "the POINTING query domain and DATA_DESCRIPTION table fit the construction slack"
249        );
250        let aw_pointing_plan_ceiling = aw_pointing_plan_retained_byte_ceiling(problem, source)?;
251        let shared_bytes = shared_bytes.with_source_plan_retained_bytes(aw_pointing_plan_ceiling);
252        let preliminary_content_plan = Self::requirements_for_locked_source(
253            &measurement_set,
254            problem,
255            source,
256            shared_bytes,
257            content_budget.maximum_pointing_polynomial_terms(),
258            pointing_query_domain,
259        )?
260        .plan(content_budget)?;
261        let pointing_catalog = if let PointingCentreLaw::Observation(law) =
262            problem.geometry().centres().pointing()
263        {
264            let domain = pointing_query_domain
265                .ok_or(BoundObservationSourceError::MissingPointingQueryDomain)?;
266            let column = match law.direction_column() {
267                PointingDirectionColumn::Direction => StoredPointingDirectionColumn::Direction,
268                PointingDirectionColumn::Target => StoredPointingDirectionColumn::Target,
269            };
270            let catalog_budget = selected_pointing_catalog_budget(
271                &measurement_set,
272                source,
273                shared_bytes,
274                content_budget,
275            )?;
276            let catalog = measurement_set.prepare_selected_pointing_catalog(
277                column,
278                domain,
279                law.time_sampling(),
280                PointingReadPlan::new(
281                    preliminary_content_plan.rows_per_block(),
282                    preliminary_content_plan.maximum_pointing_polynomial_terms(),
283                    catalog_budget,
284                )?,
285            )?;
286            let catalog_measurements = catalog.measurements();
287            tracing::info!(
288                target: "casa_ms::selected_pointing",
289                "prepared selected POINTING catalog source_rows_scanned={} retained_rows={} retained_bytes={} construction_peak_bytes={} build_nanos={}",
290                catalog_measurements.source_rows_scanned(),
291                catalog_measurements.retained_rows(),
292                catalog_measurements.retained_bytes(),
293                catalog_measurements.construction_peak_bytes(),
294                catalog_measurements.build_nanos(),
295            );
296            Some(catalog)
297        } else {
298            None
299        };
300        let content_plan = selected_content_plan_with_pointing_catalog(
301            &measurement_set,
302            problem,
303            source,
304            shared_bytes,
305            content_budget,
306            pointing_catalog
307                .as_ref()
308                .map(|catalog| catalog.measurements()),
309        )?;
310        let mut bound = Self::from_planned_locked_measurement_set(
311            source,
312            measures,
313            content_plan,
314            measurement_set,
315            ephemeris,
316            pointing_catalog,
317        )?;
318        if aw_pointing_plan_ceiling != 0 {
319            let plan = build_aw_pointing_epoch_plan(problem, &bound)?;
320            if plan.retained_byte_ceiling > aw_pointing_plan_ceiling {
321                return Err(BoundObservationSourceError::MeasurementOverflow);
322            }
323            tracing::info!(
324                target: "casa_ms::selected_pointing",
325                "prepared CASA AW pointing epoch plan epochs={} retained_byte_ceiling={}",
326                plan.chart_pixels_by_epoch.len(),
327                plan.retained_byte_ceiling,
328            );
329            bound.aw_pointing_plan = Some(plan);
330        }
331        Ok(bound)
332    }

407    /// Fill `block` with this source's next row group, its visibilities only
408    /// when `visibilities` asks for them, and return what the fill bound it
409    /// to; `None` once the source's selected rows are spent.
410    pub(super) fn fill_next_selected_block(
411        &self,
412        problem: &CompiledProblem,
413        logical_source: &MeasurementSetReadAccess,
414        replay: &mut SelectedRowReplay,
415        block: &mut SelectedObservationBlock,
416        window: Option<[f64; 2]>,
417        visibilities: bool,
418    ) -> Result<Option<BlockBinding>, BoundObservationSourceError> {
419        loop {
420            let Some(coordinate_index) =
421                self.fill_selected_row_group(replay, &mut block.selected_rows)?
422            else {
423                return Ok(None);
424            };
425            let coordinates = &self.coordinates[coordinate_index];
426            let (channel_range, channels) = match window {
427                Some(bounds) => {
428                    let Some(window) = selected_channel_window(
429                        self,
430                        problem,
431                        coordinates,
432                        &block.selected_rows,
433                        bounds,
434                    )?
435                    else {
436                        // No channel payload is needed for a disjoint source block.
437                        continue;
438                    };
439                    window
440                }
441                None => (
442                    VisibilityChannelReadRange::new(
443                        coordinates.channel_start,
444                        coordinates.channel_count,
445                    ),
446                    0..coordinates.channels.len(),
447                ),
448            };
449            self.measurement_set.fill_selected_observation_buffer(
450                &SelectedObservationBufferRequest::new(
451                    visibilities.then(|| {
452                        selected_visibility(logical_source.selected_columns().visibility())
453                    }),
454                    selected_weight(logical_source.selected_columns().weights()),
455                    &block.selected_rows,
456                    channel_range,
457                ),
458                &mut block.buffer,
459            )?;
460            let observation_pointings =
461                evaluate_observation_pointings(self, problem, &block.buffer)?;
462            block.row_geometry.clear();
463            for row in 0..block.buffer.row_count() {
464                let stored = block
465                    .buffer
466                    .row(row)
467                    .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
468                block.row_geometry.push(evaluate_row_geometry(
469                    self,
470                    problem,
471                    stored,
472                    observation_pointings
473                        .as_ref()
474                        .map(|pointings| pointings[row]),
475                )?);
476            }
477            return Ok(Some(BlockBinding {
478                coordinates: Arc::clone(&self.coordinates),
479                coordinate_index,
480                channels,
481                geometry_engine: Arc::clone(&self.geometry_engine),
482            }));
483        }
484    }

486    pub(super) fn selected_row_replay(
487        &self,
488    ) -> Result<SelectedRowReplay, BoundObservationSourceError> {
489        let rows_per_block = self.content_plan.rows_per_block();
490        let available_bytes = rows_per_block
491            .checked_mul(crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW)
492            .ok_or(BoundObservationSourceError::MeasurementOverflow)?;
493        let plan = MsReadPlan::new(
494            self.measurement_set.row_count(),
495            MsSelectionIoBudget {
496                available_bytes,
497                maximum_live_blocks: 1,
498                requested_bytes_per_row: crate::SelectedObservationRow::STORAGE_BYTES_PER_ROW,
499                storage_alignment_rows: Some(rows_per_block),
500            },
501        )
502        .map_err(|error| MsError::InvalidInput(error.to_string()))?;
503        let cursor = if self.coordinates.len() > 1 {
504            let walk = match self.walk.get() {
505                Some(walk) => Arc::clone(walk),
506                None => {
507                    let walk = Arc::new(self.row_walk(plan)?);
508                    Arc::clone(self.walk.get_or_init(|| walk))
509                }
510            };
511            self.measurement_set.main_row_walk_cursor(plan, walk)?
512        } else {
513            self.measurement_set.main_row_selection_cursor(plan)?
514        };
515        Ok(SelectedRowReplay {
516            cursor,
517            pending: None,
518            selected_rows: 0,
519        })
520    }

522    /// Collect into `rows` the next consecutive selected rows of one
523    /// DATA_DESCRIPTION, at most a block's worth, and return the index of
524    /// their coordinates.
525    fn fill_selected_row_group(
526        &self,
527        replay: &mut SelectedRowReplay,
528        rows: &mut Vec<SelectedObservationRow>,
529    ) -> Result<Option<usize>, BoundObservationSourceError> {
530        let Some(first) = replay.next_selected(self)? else {
531            return Ok(None);
532        };
533        let data_description_id = u32::try_from(first.data_description_id()).map_err(|_| {
534            BoundObservationSourceError::DataDescriptionCoordinateMismatch {
535                data_description_id: u32::MAX,
536            }
537        })?;
538        let coordinate_index = self
539            .coordinates
540            .iter()
541            .position(|coordinates| {
542                coordinates.data_description.data_description_id() == data_description_id
543            })
544            .ok_or(
545                BoundObservationSourceError::DataDescriptionCoordinateMismatch {
546                    data_description_id,
547                },
548            )?;
549        rows.clear();
550        rows.push(first);
551        while rows.len() < self.content_plan.rows_per_block() {
552            let Some(row) = replay.next_selected(self)? else {
553                break;
554            };
555            if row.data_description_id() != first.data_description_id() {
556                replay.pending = Some(row);
557                break;
558            }
559            rows.push(row);
560        }
561        Ok(Some(coordinate_index))
562    }

565/// The selected MAIN rows of one source, in physical order: a cursor over
566/// MAIN filtered by the compiled row predicate.
567pub(super) struct SelectedRowReplay {
568    cursor: MainRowSelectionCursor,
569    pending: Option<SelectedObservationRow>,
570    selected_rows: u64,
571}
572
573impl SelectedRowReplay {
574    /// Rows the predicate has selected so far, whether or not a channel
575    /// window then skipped their block.
576    pub(super) const fn selected_rows(&self) -> u64 {
577        self.selected_rows
578    }
579
580    fn next_selected(
581        &mut self,
582        source: &BoundObservationSource,
583    ) -> Result<Option<SelectedObservationRow>, BoundObservationSourceError> {
584        if let Some(row) = self.pending.take() {
585            return Ok(Some(row));
586        }
587        while let Some(row) = self.cursor.next(&source.measurement_set)? {
588            if source.row_predicate.matches(row) {
589                self.selected_rows += 1;
590                return Ok(Some(row));
591            }
592        }
593        Ok(None)
594    }
595}

597fn project_stored_run_row(
598    problem: &CompiledProblem,
599    geometry_engine: &MsCalEngine,
600    coordinates: &SelectedCoordinates,
601    stored: SelectedStoredRow,
602    geometry: &EvaluatedRowGeometry,
603) -> Result<SelectedObservationRunRow, BoundObservationSourceError> {
604    let time_scale = time_scale(geometry_engine.time_reference().as_str())?;
605    let field_id = usize::try_from(stored.field_id())
606        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
607    let antenna1 = usize::try_from(stored.antenna1())
608        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
609    let antenna2 = usize::try_from(stored.antenna2())
610        .map_err(|_| BoundObservationSourceError::InvalidRowGeometry)?;
611    let parallactic_angles_rad = if problem.requires_parallactic_angles() {
612        Some([
613            polarization_operator_angle(geometry_engine.parallactic_angle(
614                stored.time_mjd_seconds(),
615                field_id,
616                antenna1,
617            )?),
618            polarization_operator_angle(geometry_engine.parallactic_angle(
619                stored.time_mjd_seconds(),
620                field_id,
621                antenna2,
622            )?),
623        ])
624    } else {
625        None
626    };
627    let antenna_responses = match problem.science().instrument_model() {
628        None => None,
629        Some(InstrumentModel::CasaAca7mInterferometricDirectPbV1) => {
630            Some(SelectedAntennaResponses {
631                antenna1: casa_imaging_model::AntennaResponseClass::CasaAca7m,
632                antenna2: casa_imaging_model::AntennaResponseClass::CasaAca7m,
633                family_envelope: casa_imaging_model::AntennaResponseClass::CasaAca7m,
634            })
635        }
636        Some(InstrumentModel::CasaAlmaAcaHeterogeneousInterferometricResponseV1) => {
637            Some(SelectedAntennaResponses {
638                antenna1: geometry_engine
639                    .antenna_response_class(antenna1)
640                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?,
641                antenna2: geometry_engine
642                    .antenna_response_class(antenna2)
643                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?,
644                family_envelope: geometry_engine
645                    .antenna_response_family_envelope()
646                    .ok_or(BoundObservationSourceError::InvalidRowGeometry)?,
647            })
648        }
649        Some(InstrumentModel::CasaEvlaWidebandAwV1) => None,
650    };
651    let physical_row = u64::try_from(stored.physical_row())
652        .map_err(|_| BoundObservationSourceError::PhysicalRowIndexOverflow)?;
653    Ok(SelectedObservationRunRow {
654        physical_row,
655        data_description_id: stored.data_description_id(),
656        spectral_window_id: coordinates.data_description.spectral_window_id(),
657        row_flag: stored.row_flag(),
658        coordinates: SelectedSampleCoordinates {
659            raw_uvw_m: stored.uvw_m(),
660            time: Epoch::new(stored.time_mjd_seconds() / 86_400.0, time_scale),
661            parallactic_angles_rad,
662            pointing_directions: geometry.pointing_directions,
663        },
664        domain_projections: geometry.domain_projections.clone(),
665        metadata: SelectedSampleMetadata {
666            field_id: stored.field_id(),
667            antenna1: stored.antenna1(),
668            antenna2: stored.antenna2(),
669            antenna_responses,
670        },
671    })
672}

752///
753/// It borrows both the block and the geometry storage, so neither can be
754/// refilled or reprojected while it lives.

756/// The stream cannot refill a block whose projection is still read:

758/// ```compile_fail,E0499
759/// use casa_ms::{
760///     SelectedObservationBlock, SelectedObservationBlockSource,
761///     SelectedObservationNumericGeometry,
762/// };
763///
764/// fn stale(
765///     problem: &casa_imaging_model::CompiledProblem,
766///     source: &mut SelectedObservationBlockSource<'_>,

768///     geometry: &mut SelectedObservationNumericGeometry,
769/// ) {
770///     let filled = source.fill_next(block).unwrap().unwrap();
771///     let projected = filled.project_numeric_geometry(problem, geometry).unwrap();
772///     source.fill_next(block).unwrap();
773///     let _ = projected.row_count();
774/// }
775/// ```

783/// A disjoint portion of the reusable numeric geometry. The caller executes
784/// these borrowed jobs on its already admitted worker team and joins them
785/// before consuming the complete block.
786pub struct SelectedObservationNumericGeometryChunk<'a> {
787    block: &'a SelectedObservationBlock,
788    problem: &'a CompiledProblem,
789    engine: &'a MsCalEngine,
790    coordinates: &'a SelectedCoordinates,
791    channels: &'a [SelectedObservationRunChannel],
792    frame: FrequencyFrame,
793    first_row: usize,
794    rows: &'a mut [Option<SelectedObservationRunRow>],
795    frequencies_hz: &'a mut [f64],
796}
797
798impl SelectedObservationNumericGeometryChunk<'_> {
799    /// Project this row partition without opening or sharing an MS table handle.
800    pub fn project(&mut self) -> Result<(), BoundObservationSourceError> {
801        let mut previous_conversion = None;
802        let channels = self.channels.len();
803        for local in 0..self.rows.len() {
804            let row = self.first_row + local;
805            let stored = self
806                .block
807                .buffer
808                .row(row)
809                .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?;
810            let projected = project_stored_run_row(
811                self.problem,
812                self.engine,
813                self.coordinates,
814                stored,
815                &self.block.row_geometry[row],
816            )?;
817            let key = (stored.field_id(), stored.time_mjd_seconds().to_bits());
818            if previous_conversion
819                .as_ref()
820                .is_none_or(|(previous, _)| *previous != key)
821            {
822                previous_conversion = Some((
823                    key,
824                    prepare_row_frequency_conversion(
825                        self.engine,
826                        stored.field_id(),
827                        stored.time_mjd_seconds(),
828                        self.frame,
829                        self.problem.geometry().spectral().output_frame(),
830                    )?,
831                ));
832            }
833            let conversion = &previous_conversion.as_ref().expect("conversion prepared").1;
834            for (channel, coordinate) in self
835                .channels
836                .iter()
837                .zip(&mut self.frequencies_hz[local * channels..(local + 1) * channels])
838            {
839                *coordinate = conversion.convert_hz(channel.frequency_centre_hz);
840            }
841            self.rows[local] = Some(projected);
842        }
843        Ok(())
844    }
845}

865impl<'a> FilledObservationBlock<'a> {
866    pub(super) fn new(block: &'a SelectedObservationBlock, binding: BlockBinding) -> Self {
867        Self { block, binding }

886
887    /// Evaluate row geometry and frequency conversion once per row into
888    /// `geometry`. Numeric columns stay with the block, and are not turned
889    /// into sample records.
890    pub fn project_numeric_geometry<'p>(
891        &'p self,
892        problem: &CompiledProblem,
893        geometry: &'p mut SelectedObservationNumericGeometry,
894    ) -> Result<ProjectedObservationBlock<'p>, BoundObservationSourceError> {
895        let chunk_rows = geometry.maximum_rows;
896        self.project_numeric_geometry_with(problem, geometry, chunk_rows, |chunks| {
897            chunks[0].project()
898        })
899    }
900
901    /// Project disjoint coarse row chunks through a caller-supplied, joined
902    /// executor, which must project every chunk before it returns `Ok`.
903    pub fn project_numeric_geometry_with<'p>(
904        &'p self,
905        problem: &CompiledProblem,
906        geometry: &'p mut SelectedObservationNumericGeometry,
907        chunk_rows: usize,
908        execute: impl FnOnce(
909            &mut [SelectedObservationNumericGeometryChunk<'_>],
910        ) -> Result<(), BoundObservationSourceError>,
911    ) -> Result<ProjectedObservationBlock<'p>, BoundObservationSourceError> {
912        if chunk_rows == 0 {
913            return Err(BoundObservationSourceError::StoredSampleShapeMismatch);
914        }
915        geometry.rows.clear();
916        geometry.channels.clear();
917        geometry.frequencies_hz.clear();
918        let coordinates = self.binding.coordinates();
919        let window = self.binding.channels.clone();
920        let rows = self.block.buffer.row_count();
921        if rows > geometry.maximum_rows || window.len() > geometry.maximum_channels {
922            return Err(BoundObservationSourceError::StoredSampleShapeMismatch);
923        }
924        let columns = self.numeric_columns();
925        let first_stored_channel = u32::try_from(columns.channel_range.start)
926            .map_err(|_| BoundObservationSourceError::StoredSampleShapeMismatch)?;
927        geometry.channels.extend(
928            coordinates.channels[window]
929                .iter()
930                .copied()
931                .map(project_run_channel),
932        );
933        let frame = coordinates
934            .channels
935            .first()
936            .ok_or(BoundObservationSourceError::StoredSampleShapeMismatch)?
937            .frame;
938        let channels = geometry.channels.len();
939        geometry.rows.resize_with(rows, || None);
940        geometry.frequencies_hz.resize(rows * channels, 0.0);
941        let (mut row_slots, mut frequencies_hz) = (
942            geometry.rows.as_mut_slice(),
943            geometry.frequencies_hz.as_mut_slice(),
944        );
945        let mut chunks = Vec::with_capacity(rows.div_ceil(chunk_rows));
946        let mut first_row = 0;
947        while first_row < rows {
948            let take = (rows - first_row).min(chunk_rows);
949            let (head_rows, tail_rows) = row_slots.split_at_mut(take);
950            let (head_frequencies, tail_frequencies) = frequencies_hz.split_at_mut(take * channels);
951            chunks.push(SelectedObservationNumericGeometryChunk {
952                block: self.block,
953                problem,
954                engine: &self.binding.geometry_engine,
955                coordinates,
956                channels: &geometry.channels,
957                frame,
958                first_row,
959                rows: head_rows,
960                frequencies_hz: head_frequencies,
961            });
962            (row_slots, frequencies_hz) = (tail_rows, tail_frequencies);
963            first_row += take;
964        }
965        execute(&mut chunks)?;
966        assert!(
967            geometry.rows.iter().all(Option::is_some),
968            "a projection executor that returns Ok has projected every chunk"
969        );
970        Ok(ProjectedObservationBlock {
971            columns,
972            coordinates,
973            first_stored_channel,
974            geometry,
975        })
976    }
977}

979impl<'a> ProjectedObservationBlock<'a> {
980    /// Rows in the block; geometry is evaluated at row scope, never per
981    /// correlation.
982    #[must_use]
983    pub fn row_count(&self) -> usize {
984        self.geometry.rows.len()
985    }

987    /// The channels read, as selected native channel descriptors.
988    #[must_use]
989    pub fn channels(&self) -> &'a [SelectedObservationRunChannel] {
990        &self.geometry.channels
991    }

993    /// Row-major channel centres of every row in the compiled output frame.
994    #[must_use]
995    pub fn frequencies_hz(&self) -> &'a [f64] {
996        &self.geometry.frequencies_hz
997    }
998
999    /// Borrow one whole row with source-issued axes and projected metadata.
1000    ///
1001    /// # Panics
1002    ///
1003    /// When `row` is not below [`Self::row_count`].
1004    #[must_use]
1005    pub fn numeric_row(&self, row: usize) -> casa_imaging_model::SelectedNumericRow<'a> {
1006        let columns = self.columns;
1007        let projected = self.geometry.rows[row]
1008            .as_ref()
1009            .expect("a projection holds every row of its block");
1010        let channels = columns.channel_range.count;
1011        let width = channels * columns.correlation_count;
1012        let start = row * width;
1013        let visibility = match columns.visibility {
1014            crate::SelectedNumericVisibility::Float32(values) => {
1015                crate::SelectedNumericVisibility::Float32(&values[start..start + width])
1016            }
1017            crate::SelectedNumericVisibility::Complex32(values) => {
1018                crate::SelectedNumericVisibility::Complex32(&values[start..start + width])
1019            }
1020            unread @ crate::SelectedNumericVisibility::Unread => unread,
1021        };
1022        let weights = match columns.weights {
1023            crate::SelectedNumericWeights::PerRow(values) => crate::SelectedNumericWeights::PerRow(
1024                &values[row * columns.correlation_count..(row + 1) * columns.correlation_count],
1025            ),
1026            crate::SelectedNumericWeights::PerChannel(values) => {
1027                crate::SelectedNumericWeights::PerChannel(&values[start..start + width])
1028            }
1029        };
1030        casa_imaging_model::SelectedNumericRow {
1031            row: projected,
1032            channels: &self.geometry.channels,
1033            correlations: &self.coordinates.products,
1034            first_stored_channel: self.first_stored_channel,
1035            stored_channels: channels,
1036            stored_correlations: columns.correlation_count,
1037            visibility,
1038            flags: &columns.flags[start..start + width],
1039            weights,
1040        }
1041    }
1042}

1191fn selected_coordinates(
1192    measurement_set: &MeasurementSet,
1193    selection: &ObservationSelection,
1194) -> Result<Box<[SelectedCoordinates]>, BoundObservationSourceError> {
1195    let spectral_windows = measurement_set.spectral_window()?;
1196    let mut coordinates = Vec::with_capacity(selection.data_descriptions().len());
1197    for data_description in selection.data_descriptions().iter().copied() {
1198        let spectral_window = selection
1199            .spectral_windows()
1200            .iter()
1201            .find(|candidate| {
1202                candidate.spectral_window_id() == data_description.spectral_window_id()
1203            })
1204            .expect("compiled DATA_DESCRIPTION has one spectral-window selection");
1205        let spw_row = usize::try_from(data_description.spectral_window_id()).map_err(|_| {
1206            BoundObservationSourceError::SpectralCoordinateMismatch {
1207                spectral_window_id: data_description.spectral_window_id(),
1208            }
1209        })?;
1210        let Some(casa_types::ArrayValue::Float64(centres)) = spectral_windows
1211            .table()
1212            .column_accessor("CHAN_FREQ")
1213            .map_err(MsError::from)?
1214            .array_cells_owned_uncached(&[spw_row])
1215            .map_err(MsError::from)?
1216            .pop()
1217            .flatten()
1218        else {
1219            return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
1220                spectral_window_id: data_description.spectral_window_id(),
1221            });
1222        };
1223        let frame = frequency_frame(selected_i32_scalar(
1224            spectral_windows.table(),
1225            "MEAS_FREQ_REF",
1226            spw_row,
1227        )?)?;
1228        let mut channels = Vec::with_capacity(spectral_window.channel_indices().len());
1229        for &channel_index in spectral_window.channel_indices() {
1230            let channel = usize::try_from(channel_index).map_err(|_| {
1231                BoundObservationSourceError::SpectralCoordinateMismatch {
1232                    spectral_window_id: data_description.spectral_window_id(),
1233                }
1234            })?;
1235            let centre_hz = *centres.get(channel).ok_or(
1236                BoundObservationSourceError::SpectralCoordinateMismatch {
1237                    spectral_window_id: data_description.spectral_window_id(),
1238                },
1239            )?;
1240            if !centre_hz.is_finite() {
1241                return Err(BoundObservationSourceError::SpectralCoordinateMismatch {
1242                    spectral_window_id: data_description.spectral_window_id(),
1243                });
1244            }
1245            channels.push(SelectedChannel {
1246                channel_index,
1247                centre_hz,
1248                frame,
1249            });
1250        }
1251        let channel_start = channels
1252            .first()
1253            .and_then(|channel| usize::try_from(channel.channel_index).ok())
1254            .expect("compiled spectral-window selection is nonempty");
1255        let channel_end = channels
1256            .last()
1257            .and_then(|channel| usize::try_from(channel.channel_index).ok())
1258            .and_then(|channel| channel.checked_add(1))
1259            .ok_or(BoundObservationSourceError::SpectralCoordinateMismatch {
1260                spectral_window_id: data_description.spectral_window_id(),
1261            })?;
1262        drop(centres);
1263        let polarization = selection
1264            .correlations()
1265            .iter()
1266            .find(|candidate| candidate.polarization_id() == data_description.polarization_id())
1267            .expect("compiled DATA_DESCRIPTION has one polarization selection");
1268        validate_input_weight_group(polarization.products(), data_description.polarization_id())?;
1269        coordinates.push(SelectedCoordinates {
1270            data_description,
1271            channels: channels.into_boxed_slice(),
1272            products: polarization.products().into(),
1273            channel_start,
1274            channel_count: channel_end - channel_start,
1275        });
1276    }
1277    Ok(coordinates.into_boxed_slice())
1278}
1279
1280pub(super) fn validate_input_weight_group(
1281    products: &[CorrelationProduct],
1282    polarization_id: u32,
1283) -> Result<(), BoundObservationSourceError> {
1284    if products.len() == 1 {
1285        return Ok(());
1286    }
1287    let valid = match (products.first(), products.last()) {
1288        (Some(first), Some(last))
1289            if first.correlation_type() == Correlation::RR
1290                && last.correlation_type() == Correlation::LL =>
1291        {
1292            matches_canonical_correlation_order(
1293                products,
1294                &[
1295                    Correlation::RR,
1296                    Correlation::RL,
1297                    Correlation::LR,
1298                    Correlation::LL,
1299                ],
1300            )
1301        }
1302        (Some(first), Some(last))
1303            if first.correlation_type() == Correlation::XX
1304                && last.correlation_type() == Correlation::YY =>
1305        {
1306            matches_canonical_correlation_order(
1307                products,
1308                &[
1309                    Correlation::XX,
1310                    Correlation::XY,
1311                    Correlation::YX,
1312                    Correlation::YY,
1313                ],
1314            )
1315        }
1316        _ => false,
1317    };
1318    if valid {
1319        Ok(())
1320    } else {
1321        Err(
1322            BoundObservationSourceError::UnsupportedImagingWeightCorrelationGroup {
1323                polarization_id,
1324            },
1325        )
1326    }
1327}
1328
1329fn matches_canonical_correlation_order(
1330    products: &[CorrelationProduct],
1331    canonical: &[Correlation; 4],
1332) -> bool {
1333    let mut remaining = canonical.iter();
1334    products.iter().all(|product| {
1335        remaining
1336            .by_ref()
1337            .any(|candidate| *candidate == product.correlation_type())
1338    })
1339}

1378fn selected_row_predicate(
1379    measurement_set: &MeasurementSet,
1380    source: &ObservationSource,
1381) -> Result<CompiledRowPredicate, BoundObservationSourceError> {
1382    let selection = source.selection();
1383    let wavelengths = validate_data_descriptions(measurement_set, selection)?;
1384    CompiledRowPredicate::new_shared(source, |data_description_id| {
1385        wavelengths
1386            .iter()
1387            .find(|(candidate, _)| *candidate == data_description_id)
1388            .map(|(_, wavelength_m)| *wavelength_m)
1389    })
1390    .map_err(|error| match error {
1391        RowSelectionEvaluationError::MissingReferenceWavelength {
1392            data_description_id,
1393        } => BoundObservationSourceError::MissingReferenceWavelength {
1394            data_description_id,
1395        },
1396    })
1397}

1399fn validate_data_descriptions(
1400    measurement_set: &MeasurementSet,
1401    selection: &ObservationSelection,
1402) -> Result<Vec<(u32, f64)>, BoundObservationSourceError> {
1403    let data_descriptions = measurement_set.data_description()?;
1404    let spectral_windows = measurement_set.spectral_window()?;
1405    let mut wavelengths = Vec::with_capacity(selection.data_descriptions().len());
1406    for expected in selection.data_descriptions() {
1407        let data_description_id = expected.data_description_id();
1408        let row = usize::try_from(data_description_id).map_err(|_| {
1409            BoundObservationSourceError::DataDescriptionCoordinateMismatch {
1410                data_description_id,
1411            }
1412        })?;
1413        let spectral_window_id =
1414            selected_i32_scalar(data_descriptions.table(), "SPECTRAL_WINDOW_ID", row)?;
1415        let polarization_id =
1416            selected_i32_scalar(data_descriptions.table(), "POLARIZATION_ID", row)?;
1417        if u32::try_from(spectral_window_id).ok() != Some(expected.spectral_window_id())
1418            || u32::try_from(polarization_id).ok() != Some(expected.polarization_id())
1419        {
1420            return Err(
1421                BoundObservationSourceError::DataDescriptionCoordinateMismatch {
1422                    data_description_id,
1423                },
1424            );
1425        }
1426        let spectral_window_row = usize::try_from(spectral_window_id).map_err(|_| {
1427            BoundObservationSourceError::DataDescriptionCoordinateMismatch {
1428                data_description_id,
1429            }
1430        })?;
1431        let reference_frequency_hz = selected_f64_scalar(
1432            spectral_windows.table(),
1433            "REF_FREQUENCY",
1434            spectral_window_row,
1435        )?;
1436        wavelengths.push((
1437            data_description_id,
1438            SPEED_OF_LIGHT_M_PER_S / reference_frequency_hz,
1439        ));
1440    }
1441    Ok(wavelengths)
1442}

1444fn selected_i32_scalar(
1445    table: &casa_tables::Table,
1446    column: &str,
1447    row: usize,
1448) -> Result<i32, BoundObservationSourceError> {
1449    match table
1450        .column_accessor(column)
1451        .map_err(MsError::from)?
1452        .scalar_cells_owned_for_rows(&[row])
1453        .map_err(MsError::from)?
1454        .pop()
1455        .flatten()
1456    {
1457        Some(casa_types::ScalarValue::Int32(value)) => Ok(value),
1458        value => Err(MsError::InvalidInput(format!(
1459            "required selected metadata {column} row {row} is not Int32: {value:?}"
1460        ))
1461        .into()),
1462    }
1463}
