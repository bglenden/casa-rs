2//! Bounds on what an MSv2's tables hold in memory, taken before they hold
3//! it. An opened table holds its description and keywords; once a row is
4//! read it loads every row, and keeps them while it is open. A bound comes
5//! from a table's files and, once it is open, its rows and columns.
6
7use std::fs;
8use std::path::Path;
9
10use casa_tables::Table;
11use casa_types::RecordField;
12
13use crate::{MeasurementSet, SubtableId};
14
15/// Bytes allowed each table held open beside its description and
16/// keywords: its storage managers' headers, its lock and its handles.
17const TABLE_BYTES: usize = 16 << 10;
18
19/// Times their file's size (`table.dat`) a table's description and
20/// keywords hold once parsed into records.
21const DESCRIPTION_FACTOR: usize = 8;
22
23/// Bytes a loaded cell holds beside its strings' and arrays' contents: its
24/// record field, the column's name copied into it, and an array's shape.
25const CELL_BYTES: usize = size_of::<RecordField>() + 96;
26
27/// Times their files' size the tables nested in a subtable (a field's
28/// ephemerides) hold loaded, their cells unknown until they are opened.
29const NESTED_FACTOR: usize = 64;
30
31/// The subtables the opening's readers load: those its catalogue is read
32/// from.
33const READ: [SubtableId; 10] = [
34    SubtableId::Antenna,
35    SubtableId::DataDescription,
36    SubtableId::Feed,
37    SubtableId::Field,
38    SubtableId::Observation,
39    SubtableId::Polarization,
40    SubtableId::Processor,
41    SubtableId::Source,
42    SubtableId::SpectralWindow,
43    SubtableId::State,
44];
45
46/// What opening the MeasurementSet at `path` holds before a row is read:
47/// each table's description and keywords (MAIN's, every subtable's, and
48/// those nested in them: a field's ephemerides).
49pub(crate) fn opening_bytes(path: &Path) -> usize {
50    let mut bytes = 0;
51    visit(path, 2, &mut |table| {
52        let description = fs::metadata(table.join("table.dat")).map_or(0, |meta| meta.len());
53        bytes += TABLE_BYTES + DESCRIPTION_FACTOR * description as usize;
54    });
55    bytes
56}
57
58/// What `table` holds once its rows are loaded: a record of each row's
59/// cells ([`CELL_BYTES`]) and, at most twice, its storage files' contents
60/// (the rows' values, and a column's values cached apart from them).
61pub(crate) fn loaded_bytes(table: &Table) -> usize {
62    let columns = table.schema().map_or(0, |schema| schema.columns().len());
63    let files = table.path().map_or(0, storage_bytes);
64    table
65        .row_count()
66        .saturating_mul(columns)
67        .saturating_mul(CELL_BYTES)
68        .saturating_add(2 * files)
69}
70
71/// What the readers of `ms`'s catalogue load: the rows of the subtables
72/// they read ([`loaded_bytes`]) and of the tables nested in them.
73pub(crate) fn read_bytes(ms: &MeasurementSet) -> usize {
74    let nested = ms
75        .subtable(SubtableId::Field)
76        .and_then(Table::path)
77        .map_or(0, |field| {
78            let mut bytes = 0;
79            visit(field, 1, &mut |table| {
80                if table != field {
81                    bytes += TABLE_BYTES + NESTED_FACTOR * storage_bytes(table);
82                }
83            });
84            bytes
85        });
86    READ.iter()
87        .filter_map(|id| ms.subtable(*id))
88        .map(loaded_bytes)
89        .sum::<usize>()
90        + nested
91}
92
93/// Call `each` with every table directory at most `depth` levels below
94/// `directory`, itself included.
95fn visit(directory: &Path, depth: usize, each: &mut impl FnMut(&Path)) {
96    if directory.join("table.dat").is_file() {
97        each(directory);
98    }
99    if depth == 0 {
100        return;
101    }
102    let Ok(entries) = fs::read_dir(directory) else {
103        return;
104    };
105    for entry in entries.flatten() {
106        if entry.file_type().is_ok_and(|kind| kind.is_dir()) {
107            visit(&entry.path(), depth - 1, each);
108        }
109    }
110}
111
112/// The bytes a table directory's storage managers' files (`table.f*`)
113/// hold on disk.
114fn storage_bytes(directory: &Path) -> usize {
115    let Ok(entries) = fs::read_dir(directory) else {
116        return 0;
117    };
118    entries
119        .flatten()
120        .filter(|entry| entry.file_name().to_string_lossy().starts_with("table.f"))
121        .filter_map(|entry| entry.metadata().ok())
122        .filter(fs::Metadata::is_file)
123        .map(|meta| meta.len() as usize)
124        .sum()
125}
