2//! Bounded observation batches, projected pointings and Stokes I spectra.
3use super::{TotalPowerError, TotalPowerRequest, failure, pointing};
4use crate::RunContext;
5use casa_core::measures::MeasuresProvider;
6use casa_imaging_runtime::{
7    Demand, Reservation, admit,
8    pass::{WORKER_STACK_BYTES, WorkerTeam},
9};
10use casa_imaging_singledish::{Pointing, SpectrumBlock, TotalPowerGrid, weights_from_spectra};
11use casa_observation::{
12    AccessRequest, BatchBuffers, CorrelationRequest, FieldSelection, Grouping, Observation,
13    PartitionId, ReadEvent, Roles, TimeSelection, WeightRequest,
14};
15use ndarray::Array3;
16use std::sync::Arc;
17
18const BATCH_BYTES: usize = 4 << 20;
19const SAMPLING_BYTES: usize = 8 << 20;
20
21pub(super) fn grid(
22    observation: &dyn Observation,
23    request: &TotalPowerRequest,
24    context: &RunContext,
25    partitions: &[PartitionId],
26    measures: &Arc<dyn MeasuresProvider>,
27    grid: &mut TotalPowerGrid,
28) -> Result<
29    (
30        Vec<Pointing>,
31        Reservation,
32        Option<casa_core::measures::epoch::MEpoch>,
33    ),
34    TotalPowerError,
35> {
36    let workers = context
37        .policy
38        .workers(&context.host)
39        .min(request.channels.len());
40    // Two traversal buffers, real values, flags, weights and row/pixel maps,
41    // one bounded POINTING window, its projected copy, and retained sampling.
42    let mut memory = admit(
43        &context.host,
44        &context.policy,
45        &Demand {
46            phase: "total-power streaming and worker team",
47            memory: (8 * BATCH_BYTES
48                + 3 * SAMPLING_BYTES
49                + if workers > 1 {
50                    workers * (WORKER_STACK_BYTES + 65536)
51                } else {
52                    0
53                }) as u64,
54        },
55    )
56    .map_err(failure)?;
57    let team = WorkerTeam::new(workers).map_err(failure)?;
58    let mut sampling = Vec::new();
59    let mut date = None;
60    for &partition in partitions {
61        let current = observation.catalogue().partition(partition);
62        let group = ["corrected", "float", "base"]
63            .into_iter()
64            .find_map(|name| current.data_group(name).map(|(_, group)| group))
65            .ok_or_else(|| failure("no total-power data group"))?;
66        let access = AccessRequest {
67            partitions: vec![partition],
68            data_group: group.name.clone(),
69            roles: Roles {
70                values: true,
71                flags: true,
72                weights: true,
73                uvw: false,
74                time_centroid: false,
75                interval: true,
76                exposure: false,
77                scan: false,
78            },
79            weights: WeightRequest::Stored,
80            channels: request.channels.clone(),
81            times: TimeSelection::All,
82            fields: FieldSelection::All,
83            correlations: CorrelationRequest::ALL,
84            grouping: Grouping::None,
85            batch_bytes: BATCH_BYTES,
86        };
87        let mut traversal = observation.bind_read(&access).map_err(failure)?;
88        let mut buffers = BatchBuffers::default();
89        while let Some(event) = traversal.next(&mut buffers).map_err(failure)? {
90            if context.cancel.is_cancelled() {
91                return Err(failure("cancelled while gridding"));
92            }
93            let ReadEvent::Batch(batch) = event else {
94                continue;
95            };
96            if date.is_none() {
97                date = Some(batch.time.get(0));
98            }
99            let (pixels, pointings) = pointing::pixels(
100                observation,
101                &batch,
102                &request.direction,
103                measures,
104                &request.antennas,
105            )?;
106            let needed = sampling.len() + pointings.len();
107            if needed > SAMPLING_BYTES / size_of::<Pointing>() {
108                return Err(failure("raster sampling exceeds its admitted memory"));
109            }
110            sampling.reserve_exact(needed.saturating_sub(sampling.len()));
111            sampling.extend(pointings);
112            let values = batch
113                .group
114                .values
115                .ok_or_else(|| failure("missing values"))?
116                .mapv(|value| value.re);
117            let mut flags = Array3::from_elem(values.dim(), false);
118            for ((sample, channel, correlation), flag) in flags.indexed_iter_mut() {
119                *flag = batch.validity().excluded(sample, channel, correlation);
120            }
121            let weights = weights_from_spectra(
122                batch
123                    .group
124                    .weights
125                    .ok_or_else(|| failure("missing weights"))?,
126            )
127            .map_err(failure)?;
128            let row_flags: Vec<_> = (0..batch.samples())
129                .map(|row| !batch.sample(row).baseline.is_auto())
130                .collect();
131            let channel_map: Vec<_> = (0..request.channels.len()).map(Some).collect();
132            let correlation_map = vec![Some(0); batch.correlations.len()];
133            grid.grid(
134                &SpectrumBlock {
135                    pixels: &pixels,
136                    values: values.view(),
137                    flags: flags.view(),
138                    row_flags: &row_flags,
139                    weights: weights.view(),
140                    channel_map: &channel_map,
141                    correlation_map: &correlation_map,
142                },
143                &team,
144            )
145            .map_err(failure)?;
146        }
147    }
148    // The outcome keeps its coordinate records and selected-partition metadata.
149    memory.retain((sampling.capacity() * size_of::<Pointing>()) as u64);
150    Ok((sampling, memory, date))
151}
