2//! Tap loops over one `[y][x]` block: spread (adjoint), gather (forward)
3//! and the kernel norm.
4//!
5//! `tap' = (conjugate ? conj(t) : t) · e^{i(k_x g_x + k_y g_y)}`. The norm
6//! is `Σ (conjugate ? conj(t) : t)` over the same taps without the pointing
7//! ramp, which is how CASA's `AWVisResampler` accumulates it
8//! (`faccumulateFromGrid`: `norm += wt` before the phase gradient).
9//!
10//! Every function takes the location the shared rounding rule produced and
11//! trusts that the support lies inside the tile; the debug assertions name
12//! that invariant.
13

21/// Add `value · tap'` over the support into `grid` through Mueller plane
22/// `mueller`.
23#[allow(clippy::too_many_arguments)]
24pub(super) fn spread<T: GridScalar, const COMPENSATED: bool>(
25    grid: &mut [Complex<T>],
26    correction: &mut [Complex32],
27    tile: Tile,
28    location: CellLocation,
29    taps: &TapLayout<'_>,
30    mueller: u8,
31    conjugate: bool,
32    gradient: [f32; 2],
33    value: Complex<T>,
34) {
35    match *taps {
36        TapLayout::SeparableReal {
37            rows,
38            support,
39            oversampling: _,
40        } => {
41            debug_assert_eq!(mueller, 0, "separable kernels have one Mueller plane");
42            debug_assert_eq!(
43                gradient,
44                [0.0, 0.0],
45                "separable kernels carry no phase ramp"
46            );
47            let support = usize::from(support);
48            let rx = &rows[usize::from(location.ox) * support..][..support];
49            let ry = &rows[usize::from(location.oy) * support..][..support];
50            let (x0, y0) = block_origin(tile, location, [support / 2, support / 2]);
51            match support {
52                7 => spread_separable::<T, 7, COMPENSATED>(
53                    grid,
54                    correction,
55                    tile.shape[0],
56                    x0,
57                    y0,
58                    rx,
59                    ry,
60                    value,
61                ),
62                _ => spread_separable_dyn::<T, COMPENSATED>(
63                    grid,
64                    correction,
65                    tile.shape[0],
66                    x0,
67                    y0,
68                    rx,
69                    ry,
70                    value,
71                ),
72            }
73        }
74        TapLayout::Dense {
75            data,
76            support,
77            oversampling,
78            mueller_planes,
79            ..
80        } => {
81            let tile_taps = dense_tile(
82                data,
83                support,
84                oversampling,
85                mueller_planes,
86                location,
87                mueller,
88            );
89            let [sx, sy] = [usize::from(support[0]), usize::from(support[1])];
90            let (x0, y0) = block_origin(tile, location, [sx / 2, sy / 2]);
91            let block = DenseBlock {
92                nx: tile.shape[0],
93                origin: [x0, y0],
94                support: [sx, sy],
95                taps: tile_taps,
96                conjugate,
97            };
98            match PointingRamp::new(gradient, fine_fraction(location, oversampling), [sx, sy]) {
99                Some(ramp) => {
100                    spread_dense::<T, true, COMPENSATED>(grid, correction, &block, ramp, value)
101                }
102                None => spread_dense::<T, false, COMPENSATED>(
103                    grid,
104                    correction,
105                    &block,
106                    PointingRamp::NONE,
107                    value,
108                ),
109            }
110        }
111    }
112}

114/// [`spread`] of a sample of which only the taps on `cells` (grid
115/// coordinates) are gridded ([`Reach::Clipped`](crate::Reach::Clipped)):
116/// the taps are spread into `scratch`, a block of the support alone, and
117/// its cells that fall on both `cells` and `tile` are added to the tile as
118/// [`spread`] adds each tap, so they take the values a whole support would
119/// give them.
120#[allow(clippy::too_many_arguments)]
121pub(super) fn spread_clipped<T: GridScalar, const COMPENSATED: bool>(
122    grid: &mut [Complex<T>],
123    correction: &mut [Complex32],
124    (tile, cells): (Tile, CellBox),
125    location: CellLocation,
126    taps: &TapLayout<'_>,
127    (mueller, conjugate, gradient): (u8, bool, [f32; 2]),
128    value: Complex<T>,
129    scratch: &mut Vec<Complex<T>>,
130) {
131    let [hx, hy] = taps.half_support().map(usize::from);
132    let block = Tile::full([2 * hx + 1, 2 * hy + 1]);
133    scratch.clear();
134    scratch.resize(block.cells(), Complex::default());
135    let centred = CellLocation {
136        x: hx as i64,
137        y: hy as i64,
138        ..location
139    };
140    spread::<T, false>(
141        scratch,
142        &mut [],
143        block,
144        centred,
145        taps,
146        mueller,
147        conjugate,
148        gradient,
149        value,
150    );
151    // The cells taps may land on, tile-relative.
152    let [[x0, x1], [y0, y1]] = [0, 1].map(|axis| {
153        let origin = tile.origin[axis] as i64;
154        [
155            (cells[axis][0] - origin).max(0),
156            (cells[axis][1] - origin).min(tile.shape[axis] as i64),
157        ]
158    });
159    let first_x = location.x - hx as i64 - tile.origin[0] as i64;
160    let first_y = location.y - hy as i64 - tile.origin[1] as i64;
161    for (iy, row) in scratch.chunks_exact(block.shape[0]).enumerate() {
162        let y = first_y + iy as i64;
163        if !(y0..y1).contains(&y) {
164            continue;
165        }
166        for (ix, added) in row.iter().enumerate() {
167            let x = first_x + ix as i64;
168            if (x0..x1).contains(&x) {
169                let index = y as usize * tile.shape[0] + x as usize;
170                crate::compensation::add::<T, COMPENSATED>(
171                    &mut grid[index],
172                    correction,
173                    index,
174                    *added,
175                );
176            }
177        }
178    }
179}

181/// `Σ conj(tap') · grid` over the support through Mueller plane `mueller`.
182pub(super) fn gather<T: GridScalar>(
183    grid: &[Complex<T>],
184    tile: Tile,
185    location: CellLocation,
186    taps: &TapLayout<'_>,
187    mueller: u8,
188    conjugate: bool,
189    gradient: [f32; 2],
190) -> Complex64 {
191    match *taps {
192        TapLayout::SeparableReal {
193            rows,
194            support,
195            oversampling: _,
196        } => {
197            debug_assert_eq!(mueller, 0, "separable kernels have one Mueller plane");
198            let support = usize::from(support);
199            let rx = &rows[usize::from(location.ox) * support..][..support];
200            let ry = &rows[usize::from(location.oy) * support..][..support];
201            let (x0, y0) = block_origin(tile, location, [support / 2, support / 2]);
202            let sum = match support {
203                7 => gather_separable::<T, 7>(grid, tile.shape[0], x0, y0, rx, ry),
204                _ => gather_separable_dyn(grid, tile.shape[0], x0, y0, rx, ry),
205            };
206            Complex64::new(sum.re.into_f64(), sum.im.into_f64())
207        }
208        TapLayout::Dense {
209            data,
210            support,
211            oversampling,
212            mueller_planes,
213            ..
214        } => {
215            let tile_taps = dense_tile(
216                data,
217                support,
218                oversampling,
219                mueller_planes,
220                location,
221                mueller,
222            );
223            let [sx, sy] = [usize::from(support[0]), usize::from(support[1])];
224            let (x0, y0) = block_origin(tile, location, [sx / 2, sy / 2]);
225            let block = DenseBlock {
226                nx: tile.shape[0],
227                origin: [x0, y0],
228                support: [sx, sy],
229                taps: tile_taps,
230                conjugate,
231            };
232            match PointingRamp::new(gradient, fine_fraction(location, oversampling), [sx, sy]) {
233                Some(ramp) => gather_dense::<T, true>(grid, &block, ramp),
234                None => gather_dense::<T, false>(grid, &block, PointingRamp::NONE),
235            }
236        }
237    }
238}

240/// `Σ (conjugate ? conj(t) : t)` over the support of Mueller plane
241/// `mueller` at the sample's fine offset, without the pointing ramp. The
242/// adjoint passes its own conjugation (`w > 0`) and `sumwt` accumulates
243/// the magnitude; the forward passes the opposite
244/// (`accumulateFromGrid.inc` conjugates for `w ≤ 0` and sums the taps it
245/// multiplies the grid by) and divides the gather by the result.
246pub(crate) fn norm(
247    taps: &TapLayout<'_>,
248    location: CellLocation,
249    mueller: u8,
250    conjugate: bool,
251) -> Complex64 {
252    match *taps {
253        TapLayout::SeparableReal {
254            rows,
255            support,
256            oversampling: _,
257        } => {
258            debug_assert_eq!(mueller, 0, "separable kernels have one Mueller plane");
259            let support = usize::from(support);
260            let axis_sum = |offset: u16| {
261                rows[usize::from(offset) * support..][..support]
262                    .iter()
263                    .map(|tap| f64::from(*tap))
264                    .sum::<f64>()
265            };
266            Complex64::new(axis_sum(location.ox) * axis_sum(location.oy), 0.0)
267        }
268        TapLayout::Dense {
269            norms,
270            oversampling,
271            mueller_planes,
272            ..
273        } => {
274            let sum = norms[dense_tile_index(oversampling, mueller_planes, location, mueller)];
275            if conjugate { sum.conj() } else { sum }
276        }
277    }
278}

280/// Block-local `(x0, y0)` of the first tap.
281fn block_origin(tile: Tile, location: CellLocation, half: [usize; 2]) -> (usize, usize) {
282    let x0 = location.x - half[0] as i64 - tile.origin[0] as i64;
283    let y0 = location.y - half[1] as i64 - tile.origin[1] as i64;
284    debug_assert!(
285        x0 >= 0
286            && y0 >= 0
287            && x0 as usize + 2 * half[0] < tile.shape[0]
288            && y0 as usize + 2 * half[1] < tile.shape[1],
289        "sample support lies outside the accumulator tile"
290    );
291    (x0 as usize, y0 as usize)
292}

294#[allow(clippy::too_many_arguments)]
295fn spread_separable<T: GridScalar, const S: usize, const COMPENSATED: bool>(
296    grid: &mut [Complex<T>],
297    correction: &mut [Complex32],
298    nx: usize,
299    x0: usize,
300    y0: usize,
301    rx: &[f32],
302    ry: &[f32],
303    value: Complex<T>,
304) {
305    let mut taps_x = [T::zero(); S];
306    for (target, tap) in taps_x.iter_mut().zip(rx) {
307        *target = T::from_f32(*tap);
308    }
309    for (iy, wy) in ry.iter().enumerate() {
310        let scaled = value * T::from_f32(*wy);
311        let row: &mut [Complex<T>; S] = (&mut grid[(y0 + iy) * nx + x0..][..S])
312            .try_into()
313            .expect("row has S cells");
314        for (ix, (cell, tap)) in row.iter_mut().zip(&taps_x).enumerate() {
315            crate::compensation::add::<T, COMPENSATED>(
316                cell,
317                correction,
318                (y0 + iy) * nx + x0 + ix,
319                scaled * *tap,
320            );
321        }
322    }
323}

325/// `Σ_y ry[y] · Σ_x rx[x] · grid[y][x]` over an `S × S` support, each row
326/// summed in `x` order and the rows in `y` order, as
327/// [`gather_separable_dyn`] sums them.
328fn gather_separable<T: GridScalar, const S: usize>(
329    grid: &[Complex<T>],
330    nx: usize,
331    x0: usize,
332    y0: usize,
333    rx: &[f32],
334    ry: &[f32],
335) -> Complex<T> {
336    let mut taps_x = [T::zero(); S];
337    for (target, tap) in taps_x.iter_mut().zip(rx) {
338        *target = T::from_f32(*tap);
339    }
340    let mut sum = Complex::<T>::default();
341    for (iy, wy) in ry.iter().enumerate() {
342        let row: &[Complex<T>; S] = (&grid[(y0 + iy) * nx + x0..][..S])
343            .try_into()
344            .expect("row has S cells");
345        let mut row_sum = Complex::<T>::default();
346        for (cell, tap) in row.iter().zip(&taps_x) {
347            row_sum = row_sum + *cell * *tap;
348        }
349        sum = sum + row_sum * T::from_f32(*wy);
350    }
351    sum
352}
353
354fn gather_separable_dyn<T: GridScalar>(
355    grid: &[Complex<T>],
356    nx: usize,
357    x0: usize,
358    y0: usize,
359    rx: &[f32],
360    ry: &[f32],
361) -> Complex<T> {
362    let support = rx.len();
363    let mut sum = Complex::<T>::default();
364    for (iy, wy) in ry.iter().enumerate() {
365        let row = &grid[(y0 + iy) * nx + x0..][..support];
366        let mut row_sum = Complex::<T>::default();
367        for (cell, wx) in row.iter().zip(rx) {
368            row_sum = row_sum + *cell * T::from_f32(*wx);
369        }
370        sum = sum + row_sum * T::from_f32(*wy);
371    }
372    sum
373}

375#[allow(clippy::too_many_arguments)]
376fn spread_separable_dyn<T: GridScalar, const COMPENSATED: bool>(
377    grid: &mut [Complex<T>],
378    correction: &mut [Complex32],
379    nx: usize,
380    x0: usize,
381    y0: usize,
382    rx: &[f32],
383    ry: &[f32],
384    value: Complex<T>,
385) {
386    let support = rx.len();
387    for (iy, wy) in ry.iter().enumerate() {
388        let scaled = value * T::from_f32(*wy);
389        let row = &mut grid[(y0 + iy) * nx + x0..][..support];
390        for (ix, (cell, tap)) in row.iter_mut().zip(rx).enumerate() {
391            crate::compensation::add::<T, COMPENSATED>(
392                cell,
393                correction,
394                (y0 + iy) * nx + x0 + ix,
395                scaled * T::from_f32(*tap),
396            );
397        }
398    }
399}

401/// The `sy × sx` tile of one Mueller plane at the sample's fine offset.
402fn dense_tile(
403    data: &[Complex32],
404    support: [u16; 2],
405    oversampling: u16,
406    mueller_planes: u8,
407    location: CellLocation,
408    mueller: u8,
409) -> &[Complex32] {
410    let taps = usize::from(support[0]) * usize::from(support[1]);
411    let offset = dense_tile_index(oversampling, mueller_planes, location, mueller);
412    &data[offset * taps..][..taps]
413}

415/// The index of the tile of Mueller plane `mueller` at the sample's fine
416/// offset among a dense cell's tiles, and of its sum among the cell's
417/// norms ([`DenseCell::new`](crate::convolution::DenseCell::new) sums them).
418fn dense_tile_index(
419    oversampling: u16,
420    mueller_planes: u8,
421    location: CellLocation,
422    mueller: u8,
423) -> usize {
424    debug_assert!(mueller < mueller_planes, "Mueller plane outside the cell");
425    let rows = usize::from(oversampling) + 1;
426    (usize::from(location.oy) * rows + usize::from(location.ox)) * usize::from(mueller_planes)
427        + usize::from(mueller)
428}

430/// The sample's offset from its cell in cells, `(off − s/2)/s` per axis,
431/// from its fine rows: the pointing ramp is anchored at the sample itself
432/// (CASA reads the ramped kernel at `ix·sampling + off`), so every mode,
433/// the PSF and weight image included, carries the same per-sample phase.
434fn fine_fraction(location: CellLocation, oversampling: u16) -> [f64; 2] {
435    let half = f64::from(oversampling) / 2.0;
436    let fraction = |offset: u16| (f64::from(offset) - half) / f64::from(oversampling);
437    [fraction(location.ox), fraction(location.oy)]
438}

440/// One sample's dense taps and the block cells they cover.
441struct DenseBlock<'t> {
442    /// Cells per block row.
443    nx: usize,
444    /// Block-local `[x0, y0]` of the first tap ([`block_origin`]).
445    origin: [usize; 2],
446    /// Taps per axis `[sx, sy]`.
447    support: [usize; 2],
448    /// The `sy × sx` taps at the sample's fine offset ([`dense_tile`]).
449    taps: &'t [Complex32],
450    /// Whether the taps are conjugated.
451    conjugate: bool,
452}
453
454/// Taps per row segment, over which the ramp's x factors are tabulated
455/// ([`PointingRamp`]). Shorter tables cost less to fill per sample and
456/// longer segments less per row: 32 is within a few percent of the best
457/// of 16, 32 and 64 from 7 to 101 taps (#449).
458const RAMP_SEGMENT: usize = 32;
459
460/// The pointing ramp `e^{i(k_x g_x + k_y g_y)}` over a support, with
461/// `k = i − ⌊s/2⌋ + f` per axis at the sample's fine fraction `f`
462/// ([`fine_fraction`]). The ramp is separable, `e^{i k_x g_x} · e^{i k_y
463/// g_y}`, so it is never evaluated per tap. CASA's resampler instead
464/// computes its phase-gradient array once per pointing
465/// (`cachePhaseGrad`). Three phasors are evaluated per sample, at the
466/// support's first tap, one tap along x and one row along y, plus a
467/// fourth for supports wider than [`RAMP_SEGMENT`]: the step from one
468/// row segment to the next.
469///
470/// The loops tabulate `step_x^j` for the taps `j` of a segment once per
471/// sample ([`RampSegment`]). Each row starts one row step on from the
472/// last, each of its segments one segment step on, and a tap takes its
473/// segment's start times its table entry. A tap's ramp is then two
474/// multiplies that depend on no other tap. Walking the ramp tap by tap
475/// instead chained each tap's multiply to the one before, and that chain
476/// bounded the loops: they ran 2–3× slower from 29 taps up (#449).
477///
478/// The doubling shortens the dependency chain, not the error: squaring
479/// doubles the relative error already in a power, and the step's own
480/// rounding enters `step_x^j` `j` times whatever the method. So a tap's
481/// factor carries about 2ε per tap of its place in the segment, as the walk
482/// did, plus about 2ε per segment and per row before it, besides the error
483/// of the three phasors themselves (#771 review: 13ε at `j = 31` by
484/// doubling against 12.5ε by the walk). That is a few 1e-15 over a 31-tap
485/// support, and about 1e-12 even 4,096 rows along. It is an estimate of
486/// the drift, not a bound for arbitrary supports; the direct phasor's own
487/// argument error grows with `|k·g|` instead.
488#[derive(Clone, Copy)]
489struct PointingRamp {
490    /// The ramp at the first tap of the current row.
491    row: Complex64,
492    /// The ramp's factor from one tap to the next along x, and from one
493    /// row to the next along y.
494    step: [Complex64; 2],
495    /// The factor from one row segment's first tap to the next's,
496    /// `e^{i g_x · RAMP_SEGMENT}`; one when a row is a single segment.
497    segment: Complex64,
498}
499
500impl PointingRamp {
501    /// The ramp of a kernel without one, which the unramped loops ignore.
502    const NONE: Self = Self {
503        row: Complex64::new(1.0, 0.0),
504        step: [Complex64::new(1.0, 0.0); 2],
505        segment: Complex64::new(1.0, 0.0),
506    };
507
508    /// The ramp of `gradient` at `fraction` over `support` taps, or `None`
509    /// when the gradient is zero.
510    fn new(gradient: [f32; 2], fraction: [f64; 2], support: [usize; 2]) -> Option<Self> {
511        if gradient == [0.0, 0.0] {
512            return None;
513        }
514        let [gx, gy] = gradient.map(f64::from);
515        let kx = fraction[0] - (support[0] / 2) as f64;
516        let ky = fraction[1] - (support[1] / 2) as f64;
517        let segment = if support[0] > RAMP_SEGMENT {
518            Complex64::from_polar(1.0, gx * RAMP_SEGMENT as f64)
519        } else {
520            Complex64::new(1.0, 0.0)
521        };
522        Some(Self {
523            row: Complex64::from_polar(1.0, kx * gx + ky * gy),
524            step: [
525                Complex64::from_polar(1.0, gx),
526                Complex64::from_polar(1.0, gy),
527            ],
528            segment,
529        })
530    }
531}

533/// The ramp's x factors over one row segment: `step_x^j` for its first
534/// `width` taps `j`, at most [`RAMP_SEGMENT`] ([`PointingRamp`]).
535struct RampSegment {
536    factors: [Complex64; RAMP_SEGMENT],
537}
538
539impl RampSegment {
540    /// The factors by doubling: those filled so far times the next power
541    /// `step_x^{2^k}`, so the fill is `log₂ width` multiplies deep rather
542    /// than `width`, which a small support's sample would wait on. Its
543    /// error is the walk's ([`PointingRamp`]).
544    fn new(ramp: &PointingRamp, width: usize) -> Self {
545        let width = width.min(RAMP_SEGMENT);
546        let mut factors = [Complex64::new(1.0, 0.0); RAMP_SEGMENT];
547        let (mut filled, mut power) = (1, ramp.step[0]);
548        while filled < width {
549            let more = filled.min(width - filled);
550            for j in 0..more {
551                factors[filled + j] = factors[j] * power;
552            }
553            filled += more;
554            power *= power;
555        }
556        Self { factors }
557    }
558}

560/// A dense tap as stored, conjugated when `conjugate`, in `f64`.
561fn oriented(tap: Complex32, conjugate: bool) -> Complex64 {
562    let tap = Complex64::new(f64::from(tap.re), f64::from(tap.im));
563    if conjugate { tap.conj() } else { tap }
564}

566/// Add `value · tap'` over `block`, each tap times `ramp` when `RAMPED`:
567/// a row segment's value carries the ramp at its first tap, and each tap
568/// its table factor ([`PointingRamp`]).
569fn spread_dense<T: GridScalar, const RAMPED: bool, const COMPENSATED: bool>(
570    grid: &mut [Complex<T>],
571    correction: &mut [Complex32],
572    block: &DenseBlock<'_>,
573    mut ramp: PointingRamp,
574    value: Complex<T>,
575) {
576    let ([x0, y0], [sx, sy]) = (block.origin, block.support);
577    let table = RAMPED.then(|| RampSegment::new(&ramp, sx));
578    let wide = Complex64::new(value.re.into_f64(), value.im.into_f64());
579    for iy in 0..sy {
580        let row = &mut grid[(y0 + iy) * block.nx + x0..][..sx];
581        let taps = &block.taps[iy * sx..][..sx];
582        let Some(table) = &table else {
583            for (ix, (cell, tap)) in row.iter_mut().zip(taps).enumerate() {
584                let tap = oriented(*tap, block.conjugate);
585                crate::compensation::add::<T, COMPENSATED>(
586                    cell,
587                    correction,
588                    (y0 + iy) * block.nx + x0 + ix,
589                    value * Complex::new(T::from_f64(tap.re), T::from_f64(tap.im)),
590                );
591            }
592            continue;
593        };
594        let mut start = wide * ramp.row;
595        for (segment, (cells, taps)) in row
596            .chunks_mut(RAMP_SEGMENT)
597            .zip(taps.chunks(RAMP_SEGMENT))
598            .enumerate()
599        {
600            for (ix, ((cell, tap), factor)) in
601                cells.iter_mut().zip(taps).zip(&table.factors).enumerate()
602            {
603                let added = start * (oriented(*tap, block.conjugate) * *factor);
604                crate::compensation::add::<T, COMPENSATED>(
605                    cell,
606                    correction,
607                    (y0 + iy) * block.nx + x0 + segment * RAMP_SEGMENT + ix,
608                    Complex::new(T::from_f64(added.re), T::from_f64(added.im)),
609                );
610            }
611            start *= ramp.segment;
612        }
613        ramp.row *= ramp.step[1];
614    }
615}

617/// `Σ conj(tap') · grid` over `block`, each tap times `ramp` when `RAMPED`:
618/// each row segment sums its taps times their table factors, then takes
619/// the ramp at its first tap ([`PointingRamp`]).
620fn gather_dense<T: GridScalar, const RAMPED: bool>(
621    grid: &[Complex<T>],
622    block: &DenseBlock<'_>,
623    mut ramp: PointingRamp,
624) -> Complex64 {
625    let ([x0, y0], [sx, sy]) = (block.origin, block.support);
626    let table = RAMPED.then(|| RampSegment::new(&ramp, sx));
627    let mut sum = Complex64::default();
628    for iy in 0..sy {
629        let row = &grid[(y0 + iy) * block.nx + x0..][..sx];
630        let taps = &block.taps[iy * sx..][..sx];
631        let Some(table) = &table else {
632            for (cell, tap) in row.iter().zip(taps) {
633                let tap = oriented(*tap, block.conjugate);
634                sum += tap.conj() * Complex64::new(cell.re.into_f64(), cell.im.into_f64());
635            }
636            continue;
637        };
638        let mut start = ramp.row;
639        for (cells, taps) in row.chunks(RAMP_SEGMENT).zip(taps.chunks(RAMP_SEGMENT)) {
640            let mut segment = Complex64::default();
641            for ((cell, tap), factor) in cells.iter().zip(taps).zip(&table.factors) {
642                let tap = oriented(*tap, block.conjugate) * *factor;
643                segment += tap.conj() * Complex64::new(cell.re.into_f64(), cell.im.into_f64());
644            }
645            sum += start.conj() * segment;
646            start *= ramp.segment;
647        }
648        ramp.row *= ramp.step[1];
649    }
650    sum
651}

694    /// One case of [`the_tabulated_pointing_ramp_equals_the_ramp_at_each_tap`]:
695    /// a dense kernel of `support` taps spread and gathered through
696    /// [`spread`] and [`gather`] at fine offsets `offsets`, against the ramp
697    /// evaluated at each tap, the conjugation applied by sign rather than
698    /// through [`oriented`]. A zero gradient must be the unramped arithmetic
699    /// exactly; otherwise each spread cell is within `spread_tolerance`
700    /// and the gather within 1e-12 per tap.
701    fn ramp_case<T: GridScalar>(
702        (support, gradient, offsets, conjugate): ([usize; 2], [f32; 2], [u16; 2], bool),
703        spread_tolerance: f64,
704    ) {
705        let oversampling = 4_u16;
706        let [sx, sy] = support;
707        let tiles = (usize::from(oversampling) + 1).pow(2);
708        let data = (0..tiles * sx * sy)
709            .map(|i| Complex32::new(noise(3 * i) as f32, noise(3 * i + 1) as f32))
710            .collect::<Vec<_>>();
711        let dense = DenseCell::new(
712            data.clone().into_boxed_slice(),
713            support.map(|s| u16::try_from(s).expect("support fits u16")),
714            oversampling,
715            1,
716        );
717        let taps = dense.layout();
718        let tile = Tile {
719            origin: [0, 0],
720            shape: [sx + 4, sy + 6],
721        };
722        let location = CellLocation {
723            x: (sx / 2 + 1) as i64,
724            y: (sy / 2 + 2) as i64,
725            ox: offsets[0],
726            oy: offsets[1],
727        };
728        let cell = |ix: usize, iy: usize| (2 + iy) * tile.shape[0] + 1 + ix;
729        // The offset's tile, x fastest, and its fractions (o − s/2)/s.
730        let first = (usize::from(offsets[1]) * (usize::from(oversampling) + 1)
731            + usize::from(offsets[0]))
732            * sx
733            * sy;
734        let [fx, fy] = offsets
735            .map(|o| (f64::from(o) - f64::from(oversampling) / 2.0) / f64::from(oversampling));
736        let direct = |ix: usize, iy: usize| {
737            let stored = data[first + iy * sx + ix];
738            let sign = if conjugate { -1.0 } else { 1.0 };
739            let tap = Complex64::new(f64::from(stored.re), sign * f64::from(stored.im));
740            let kx = ix as f64 - (sx / 2) as f64 + fx;
741            let ky = iy as f64 - (sy / 2) as f64 + fy;
742            let phase = kx * f64::from(gradient[0]) + ky * f64::from(gradient[1]);
743            if gradient == [0.0, 0.0] {
744                tap
745            } else {
746                tap * Complex64::from_polar(1.0, phase)
747            }
748        };
749        let label = format!("{support:?} {gradient:?} {offsets:?} {conjugate}");
750
751        let value = Complex::new(T::from_f64(0.6), T::from_f64(-1.3));
752        let mut spread_grid = vec![Complex::<T>::default(); tile.shape[0] * tile.shape[1]];
753        spread::<T, false>(
754            &mut spread_grid,
755            &mut [],
756            tile,
757            location,
758            &taps,
759            0,
760            conjugate,
761            gradient,
762            value,
763        );
764        for (ix, iy) in (0..sy).flat_map(|iy| (0..sx).map(move |ix| (ix, iy))) {
765            let got = spread_grid[cell(ix, iy)];
766            let tap = direct(ix, iy);
767            if gradient == [0.0, 0.0] {
768                let unramped = value * Complex::new(T::from_f64(tap.re), T::from_f64(tap.im));
769                assert!(got == unramped, "{label}: spread tap ({ix}, {iy}) changed");
770            } else {
771                let value = Complex64::new(value.re.into_f64(), value.im.into_f64());
772                let got = Complex64::new(got.re.into_f64(), got.im.into_f64());
773                let error = (got - value * tap).norm();
774                assert!(
775                    error <= spread_tolerance,
776                    "{label}: spread tap ({ix}, {iy}) off by {error}"
777                );
778            }
779        }
780
781        let grid = (0..tile.shape[0] * tile.shape[1])
782            .map(|i| Complex::new(T::from_f64(noise(7 * i)), T::from_f64(noise(7 * i + 1))))
783            .collect::<Vec<_>>();
784        let expected = (0..sy)
785            .flat_map(|iy| (0..sx).map(move |ix| (ix, iy)))
786            .map(|(ix, iy)| {
787                let cell = grid[cell(ix, iy)];
788                direct(ix, iy).conj() * Complex64::new(cell.re.into_f64(), cell.im.into_f64())
789            })
790            .fold(Complex64::default(), |sum, term| sum + term);
791        let got = gather::<T>(&grid, tile, location, &taps, 0, conjugate, gradient);
792        if gradient == [0.0, 0.0] {
793            assert!(got == expected, "{label}: gather changed");
794        } else {
795            let error = (got - expected).norm();
796            assert!(
797                error <= 1e-12 * (sx * sy) as f64,
798                "{label}: gather off by {error}"
799            );
800        }
801    }
