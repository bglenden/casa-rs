2//! Structural changes made in place on disk: the change is written to
3//! `table.dat` (and, where a storage manager goes away, its files are
4//! deleted); every other storage manager file is left as it is, so a
5//! MeasurementSet keeps its tiled, incremental and standard managers.
6//!
7//! Each operation edits what `table.dat` holds, then reloads the table from
8//! the disk ([`Table::resync_or_discard`]), so the handle is what the disk
9//! says without the change having loaded a row, and holds nothing a release
10//! of the write lock would write in full. An operation refused before it
11//! wrote leaves the handle as clean as it found it. The in-memory edits of
12//! [`Table::rename_column`] and [`Table::remove_column`] load every row and
13//! are for a table written in full afterwards.
14//!
15//! # C++ equivalent
16//!
17//! `Table::renameColumn` and `Table::removeColumn`, which change the
18//! table description and the column set and leave a storage manager that
19//! keeps other columns where it is.
20

25/// Prefix of the files of the storage manager with sequence number `seq_nr`:
26/// `table.f<seq_nr>`, the `_TSM<n>` files of a tiled manager, and the `i`
27/// file of a standard manager's indirect arrays. A manager whose number
28/// merely starts with the same digits is another manager.
29fn belongs_to_manager(file_name: &str, seq_nr: u32) -> bool {
30    let Some(rest) = file_name.strip_prefix(&format!("table.f{seq_nr}")) else {
31        return false;
32    };
33    rest.is_empty() || rest == "i" || rest.starts_with('_')
34}
35
36/// Deletes the files of the storage manager with sequence number `seq_nr`.
37fn remove_manager_files(table_dir: &Path, seq_nr: u32) -> std::io::Result<()> {
38    for entry in std::fs::read_dir(table_dir)? {
39        let entry = entry?;
40        if entry
41            .file_name()
42            .to_str()
43            .is_some_and(|name| belongs_to_manager(name, seq_nr))
44        {
45            std::fs::remove_file(entry.path())?;
46        }
47    }
48    Ok(())
49}

51/// Whether a hypercolumn definition of the table description lists `name`
52/// as a coordinate or an id column (the private keyword `Hypercolumn_<name>`
53/// lists its `data`, `coord` and `id` columns).
54fn is_id_or_coordinate_column(private_keywords: &RecordValue, name: &str) -> bool {
55    private_keywords
56        .fields()
57        .iter()
58        .filter(|field| field.name.starts_with("Hypercolumn_"))
59        .filter_map(|field| match &field.value {
60            Value::Record(definition) => Some(definition),
61            _ => None,
62        })
63        .flat_map(|definition| ["coord", "id"].map(|list| definition.get(list)))
64        .any(|list| match list {
65            Some(Value::Array(ArrayValue::String(names))) => names.iter().any(|n| n == name),
66            _ => false,
67        })
68}

70/// Renames `old` to `new` in the hypercolumn definitions of the table
71/// description, as `TableDesc::renameColumn` does through
72/// `adjustHypercolumns` (casacore's own handling of coordinate and id
73/// columns there is not followed: they are refused, not renamed).
74fn rename_in_hypercolumns(private_keywords: &mut RecordValue, old: &str, new: &str) {
75    edit_hypercolumns(private_keywords, |names| {
76        for name in names {
77            if name == old {
78                *name = new.to_string();
79            }
80        }
81    });
82}

84/// Applies `edit` to each list of column names of each hypercolumn
85/// definition in `private_keywords`.
86fn edit_hypercolumns(private_keywords: &mut RecordValue, edit: impl Fn(&mut Vec<String>)) {
87    for field in private_keywords.fields_mut() {
88        if !field.name.starts_with("Hypercolumn_") {
89            continue;
90        }
91        let Value::Record(definition) = &mut field.value else {
92            continue;
93        };
94        for list in definition.fields_mut() {
95            let Value::Array(ArrayValue::String(array)) = &mut list.value else {
96                continue;
97            };
98            let mut names: Vec<String> = array.iter().cloned().collect();
99            edit(&mut names);
100            if names.len() != array.len() || names.iter().zip(array.iter()).any(|(a, b)| a != b) {
101                *array = ndarray::ArrayD::from_shape_vec(ndarray::IxDyn(&[names.len()]), names)
102                    .expect("a vector's shape is its length");
103            }
104        }
105    }
106}

108impl Table {
109    /// Renames column `old` to `new` in the table on disk, without loading
110    /// its rows or touching a storage manager's file.
111    ///
112    /// Only `table.dat` changes: the column's description, and its name in
113    /// the hypercolumn definitions. The name the column's storage manager
114    /// knows it by (casacore's `originalName`) stays as it was. Fails if
115    /// `old` is not a column or `new` already is one. The table is reloaded
116    /// from the disk afterwards, so changes this handle has not saved are
117    /// discarded.
118    ///
119    /// # C++ equivalent
120    ///
121    /// `Table::renameColumn`.
122    pub fn rename_column_in_place(&mut self, old: &str, new: &str) -> Result<(), TableError> {
123        self.edit_structure_in_place("rename_column_in_place", |dat, _| {
124            let index = column_index(dat, old)?;
125            if column_index(dat, new).is_ok() {
126                return Err(SchemaError::DuplicateColumn(new.to_string()).into());
127            }
128            // The name of an id or coordinate column is also kept in the
129            // hypercube values of its tiled manager's file, which an edit of
130            // `table.dat` alone leaves behind: the renamed column would be
131            // unreadable. (casacore's own `adjustHypercolumns` does not carry
132            // these names over either.)
133            if is_id_or_coordinate_column(&dat.table_desc.private_keywords, old) {
134                return Err(TableError::Storage(format!(
135                    "column \"{old}\" is an id or coordinate column of a hypercolumn: its name \
136                     is also held in the hypercube values of its storage manager's file, which \
137                     cannot be renamed in place"
138                )));
139            }
140            dat.table_desc.columns[index].col_name = new.to_string();
141            dat.column_set.columns[index].original_name = new.to_string();
142            let on_file = dat
143                .names_on_file
144                .remove(old)
145                .unwrap_or_else(|| old.to_string());
146            if on_file != new {
147                dat.names_on_file.insert(new.to_string(), on_file);
148            }
149            rename_in_hypercolumns(&mut dat.table_desc.private_keywords, old, new);
150            Ok(Vec::new())
151        })
152    }
153
154    /// Removes the columns `names` from the table on disk, without loading
155    /// its rows. Every storage manager that held any of them must hold only
156    /// them: its files are deleted, and the other managers' files are not
157    /// touched. A column whose manager keeps other columns cannot be taken
158    /// out in place (the manager's file holds its data), and is refused
159    /// before anything changes. The table is reloaded from the disk
160    /// afterwards, so changes this handle has not saved are discarded.
161    ///
162    /// # C++ equivalent
163    ///
164    /// `Table::removeColumn`.
165    pub fn remove_columns_in_place(&mut self, names: &[&str]) -> Result<(), TableError> {
166        self.edit_structure_in_place("remove_columns_in_place", |dat, _| {
167            let mut indices = Vec::with_capacity(names.len());
168            for name in names {
169                let index = column_index(dat, name)?;
170                if !indices.contains(&index) {
171                    indices.push(index);
172                }
173            }
174            let mut gone = Vec::new();
175            for &index in &indices {
176                let seq_nr = dat.column_set.columns[index].dm_seq_nr;
177                if gone.contains(&seq_nr) {
178                    continue;
179                }
180                let held = dat
181                    .column_set
182                    .columns
183                    .iter()
184                    .filter(|entry| entry.dm_seq_nr == seq_nr)
185                    .count();
186                let removed = indices
187                    .iter()
188                    .filter(|&&other| dat.column_set.columns[other].dm_seq_nr == seq_nr)
189                    .count();
190                if held != removed {
191                    return Err(TableError::Storage(format!(
192                        "column \"{}\" shares storage manager {seq_nr} with {} other columns, \
193                         which a removal in place would leave without their data's owner",
194                        dat.table_desc.columns[index].col_name,
195                        held - removed,
196                    )));
197                }
198                gone.push(seq_nr);
199            }
200            let removed_names: Vec<String> = indices
201                .iter()
202                .map(|&index| dat.table_desc.columns[index].col_name.clone())
203                .collect();
204            indices.sort_unstable_by(|a, b| b.cmp(a));
205            for index in indices {
206                dat.table_desc.columns.remove(index);
207                dat.column_set.columns.remove(index);
208            }
209            for name in &removed_names {
210                dat.names_on_file.remove(name);
211            }
212            dat.column_set
213                .data_managers
214                .retain(|manager| !gone.contains(&manager.seq_nr));
215            // A hypercolumn loses the columns it listed; one left with no
216            // data column is gone (the rule `adjustHypercolumns` applies to
217            // data columns; this is the native policy for the others).
218            edit_hypercolumns(&mut dat.table_desc.private_keywords, |list| {
219                list.retain(|name| !removed_names.contains(name));
220            });
221            let keywords = &mut dat.table_desc.private_keywords;
222            let emptied: Vec<String> = keywords
223                .fields()
224                .iter()
225                .filter(|field| field.name.starts_with("Hypercolumn_"))
226                .filter(|field| match &field.value {
227                    Value::Record(definition) => {
228                        matches!(
229                            definition.get("data"),
230                            Some(Value::Array(array)) if array.shape().iter().product::<usize>() == 0
231                        )
232                    }
233                    _ => false,
234                })
235                .map(|field| field.name.clone())
236                .collect();
237            for name in emptied {
238                keywords.remove(&name);
239            }
240            Ok(gone)
241        })
242    }

244    /// Applies `edit` to the table's `table.dat` and writes it, then deletes
245    /// the files of the storage managers `edit` returns the numbers of, and
246    /// reloads the table from the disk. `edit` refuses before anything is
247    /// written.
248    fn edit_structure_in_place(
249        &mut self,
250        operation: &str,
251        edit: impl FnOnce(&mut TableDatContents, &Path) -> Result<Vec<u32>, TableError>,
252    ) -> Result<(), TableError> {
253        if self.kind != TableKind::Plain {
254            return Err(TableError::Storage(format!(
255                "{operation} requires a plain disk-backed table"
256            )));
257        }
258        if !self.virtual_columns.is_empty() || !self.virtual_bindings.is_empty() {
259            return Err(TableError::Storage(format!(
260                "{operation} does not support virtual columns"
261            )));
262        }
263        let path = self
264            .source_path
265            .clone()
266            .ok_or_else(|| TableError::Storage("table has no source path".to_string()))?;
267        // Every admitted write operation counts as a change of the handle,
268        // and a change the disk lacks is what releasing the write lock writes
269        // in full (as `StManAipsIO`, whatever the table's layout). This
270        // operation writes to the disk itself, so it must leave the handle
271        // as clean as it found it.
272        let dirty_before = self.holds_unflushed_changes();
273        let auto_unlock = self.begin_write_operation(operation)?;
274        let mut wrote = false;
275        let result = (|| {
276            let control_path = path.join(crate::storage::TABLE_CONTROL_FILE);
277            let TableDatResult::Plain(mut table_dat) =
278                crate::storage::table_control::read_table_dat_dispatch(&control_path)?
279            else {
280                return Err(TableError::Storage(format!(
281                    "{operation} only supports plain tables"
282                )));
283            };
284            let persisted_rows = self.require_persisted_rows(&path, &table_dat)?;
285            table_dat.raise_row_count(persisted_rows);
286            let removed_managers = edit(&mut table_dat, &path)?;
287            self.note_persisted_write(&path);
288            wrote = true;
289            crate::storage::table_control::write_table_dat(&control_path, &table_dat)?;
290            for seq_nr in removed_managers {
291                remove_manager_files(&path, seq_nr).map_err(StorageError::from)?;
292            }
293            crate::storage::tiled_stman::invalidate_shared_tile_cache_for_table(&path);
294            Ok(())
295        })();
296        if !wrote && !dirty_before {
297            // Refused before anything reached the disk.
298            self.mark_flushed();
299        }
300        let result = if wrote {
301            // The disk changed, wholly or in part: the handle takes what it
302            // holds from the disk, and is clean whether or not that works.
303            let reloaded = self.resync_or_discard();
304            result.and(reloaded)
305        } else {
306            result
307        };
308        self.finish_write_operation(auto_unlock, result)
309    }
310}

312/// The position of column `name` in the table description (the column set
313/// lists its columns in the same order).
314fn column_index(dat: &TableDatContents, name: &str) -> Result<usize, TableError> {
315    dat.table_desc
316        .columns
317        .iter()
318        .position(|column| column.col_name == name)
319        .ok_or_else(|| TableError::SchemaColumnUnknown {
320            column: name.to_string(),
321        })
322}

338            "table.info",
