26    /// Plan the standard simulation/import visibility column family.
27    pub fn visibility_creation(
28        row_count: usize,
29        correlation_count: usize,
30        channel_count: usize,
31        telescope_name: &str,
32        resources: MeasurementSetWriteResources,
33    ) -> Result<Self, MeasurementSetWriteError> {
34        if correlation_count == 0 || channel_count == 0 {
35            return Err(MeasurementSetWriteError::InvalidPlan(
36                "correlation_count and channel_count must be positive".to_string(),
37            ));
38        }
39        if resources.maximum_live_batches == 0 {
40            return Err(MeasurementSetWriteError::InvalidPlan(
41                "maximum_live_batches must be positive".to_string(),
42            ));
43        }
44        let sample_count = correlation_count
45            .checked_mul(channel_count)
46            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
47        let visibility_tile_shape =
48            crate::ms::casa_visibility_tile_shape(correlation_count, channel_count, telescope_name);
49        let weight_tile_shape = crate::ms::casa_weight_tile_shape(&visibility_tile_shape);
50        let uvw_tile_shape = crate::ms::casa_uvw_tile_shape(&visibility_tile_shape);
51        let scalar_columns = standard_main_scalar_column_plans();
52        let scalar_bytes_per_row = scalar_columns.iter().try_fold(0usize, |total, column| {
53            total
54                .checked_add(column.bytes_per_row())
55                .ok_or(MeasurementSetWriteError::ByteOverflow)
56        })?;
57        // The scalar writers' buffers while rows stream, and the save's
58        // finalization scratch after them.
59        let scalar_buffer_bytes = streamed_scalar_columns_resident_bytes(scalar_columns.len())
60            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
61        // DATA complex32 + FLAG + one possible writer-owned FLAG copy,
62        // UVW f64[3], WEIGHT f32[ncorr], SIGMA f32[ncorr], and fixed-width scalars.
63        let bytes_per_row = sample_count
64            .checked_mul(8 + 1 + 1)
65            .and_then(|bytes| bytes.checked_add(3 * 8))
66            .and_then(|bytes| bytes.checked_add(correlation_count.checked_mul(4 + 4)?))
67            .and_then(|bytes| bytes.checked_add(scalar_bytes_per_row))
68            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
69        let tile_rows = visibility_tile_shape.get(2).copied().unwrap_or(1).max(1);
70        let fixed_bytes = |data_bands: usize| {
71            visibility_writer_resident_bytes(
72                (correlation_count, channel_count),
73                [&visibility_tile_shape, &uvw_tile_shape, &weight_tile_shape],
74                data_bands,
75                resources.tiled_column_buffer_bytes,
76            )?
77            .checked_add(scalar_buffer_bytes)
78            .ok_or(MeasurementSetWriteError::ByteOverflow)
79        };
80        // DATA's writer holds a second band of tiles, which it fills while
81        // the first is written, when the budget still leaves each batch a
82        // whole band of rows beside it.
83        let band_fits = |fixed: usize| {
84            resources
85                .available_bytes
86                .checked_sub(fixed)
87                .is_some_and(|left| {
88                    left / resources.maximum_live_batches
89                        >= bytes_per_row.saturating_mul(tile_rows.min(row_count.max(1)))
90                })
91        };
92        let data_bands = if band_fits(fixed_bytes(2)?) { 2 } else { 1 };
93        let fixed_writer_bytes = fixed_bytes(data_bands)?;
94        if fixed_writer_bytes > resources.available_bytes {
95            return Err(MeasurementSetWriteError::InvalidPlan(format!(
96                "fixed scalar and tiled column buffers require {fixed_writer_bytes} bytes; writer budget is {} bytes",
97                resources.available_bytes
98            )));
99        }
100        let streaming_budget = resources.available_bytes - fixed_writer_bytes;
101        let per_batch_budget = streaming_budget / resources.maximum_live_batches;
102        if row_count > 0 && per_batch_budget < bytes_per_row {
103            return Err(MeasurementSetWriteError::InsufficientBudget {
104                bytes_per_row,
105                per_batch_budget,
106            });
107        }
108        // Batches hold whole bands of tiles (one tile along the row axis,
109        // every tile across the cell), so each batch completes its tiles; as
110        // many bands as fit the target, and at least one when the budget
111        // allows. The budget is a ceiling, not a target: a few tens of MB
112        // keep the producer and the writer overlapping with little resident
113        // memory.
114        let budget_rows = per_batch_budget / bytes_per_row;
115        let mut batch_rows = if budget_rows >= tile_rows {
116            (TARGET_BATCH_BYTES / bytes_per_row / tile_rows).clamp(1, budget_rows / tile_rows)
117                * tile_rows
118        } else {
119            budget_rows
120        };
121        batch_rows = batch_rows.min(row_count.max(1));
122        let batch_bytes = batch_rows
123            .checked_mul(bytes_per_row)
124            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
125        let queue_budget = streaming_budget.saturating_sub(batch_bytes);
126        let queue_capacity = queue_budget
127            .checked_div(batch_bytes)
128            .unwrap_or(0)
129            .min(resources.maximum_live_batches.saturating_sub(1));
130        let maximum_resident_bytes = batch_bytes
131            .checked_mul(queue_capacity.saturating_add(1))
132            .and_then(|bytes| bytes.checked_add(fixed_writer_bytes))
133            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
134        let mut columns = vec![
135            MeasurementSetWriteColumnPlan {
136                name: "DATA".to_string(),
137                bytes_per_row: sample_count * 8,
138                mode: MeasurementSetColumnWriteMode::Create,
139                storage_manager: MeasurementSetColumnStorage::TiledShape,
140                tile_shape: Some(visibility_tile_shape.clone()),
141                create_source_column: None,
142            },
143            MeasurementSetWriteColumnPlan {
144                name: "FLAG".to_string(),
145                bytes_per_row: sample_count,
146                mode: MeasurementSetColumnWriteMode::Create,
147                storage_manager: MeasurementSetColumnStorage::TiledShape,
148                tile_shape: Some(visibility_tile_shape.clone()),
149                create_source_column: None,
150            },
151            // Every FLAG_CATEGORY cell is left undefined, as CASA's simulator
152            // leaves it, so the column holds no payload.
153            MeasurementSetWriteColumnPlan {
154                name: "FLAG_CATEGORY".to_string(),
155                bytes_per_row: 0,
156                mode: MeasurementSetColumnWriteMode::Create,
157                storage_manager: MeasurementSetColumnStorage::TiledShape,
158                tile_shape: None,
159                create_source_column: None,
160            },
161            MeasurementSetWriteColumnPlan {
162                name: "UVW".to_string(),
163                bytes_per_row: 3 * 8,
164                mode: MeasurementSetColumnWriteMode::Create,
165                storage_manager: MeasurementSetColumnStorage::TiledColumn,
166                tile_shape: Some(uvw_tile_shape.clone()),
167                create_source_column: None,
168            },
169            MeasurementSetWriteColumnPlan {
170                name: "WEIGHT".to_string(),
171                bytes_per_row: correlation_count * 4,
172                mode: MeasurementSetColumnWriteMode::Create,
173                storage_manager: MeasurementSetColumnStorage::TiledShape,
174                tile_shape: Some(weight_tile_shape.clone()),
175                create_source_column: None,
176            },
177            MeasurementSetWriteColumnPlan {
178                name: "SIGMA".to_string(),
179                bytes_per_row: correlation_count * 4,
180                mode: MeasurementSetColumnWriteMode::Create,
181                storage_manager: MeasurementSetColumnStorage::TiledShape,
182                tile_shape: Some(weight_tile_shape.clone()),
183                create_source_column: None,
184            },
185        ];
186        columns.extend(
187            scalar_columns
188                .iter()
189                .map(|column| MeasurementSetWriteColumnPlan {
190                    name: column.name.clone(),
191                    bytes_per_row: column.bytes_per_row(),
192                    mode: MeasurementSetColumnWriteMode::Create,
193                    storage_manager: MeasurementSetColumnStorage::Standard,
194                    tile_shape: None,
195                    create_source_column: None,
196                }),
197        );
198        Ok(Self {
199            operation: MeasurementSetWriteOperation::Create,
200            row_count,
201            correlation_count,
202            channel_count,
203            bytes_per_row,
204            batch_rows,
205            batch_bytes,
206            queue_capacity,
207            maximum_resident_bytes,
208            visibility_tile_shape: visibility_tile_shape.clone(),
209            weight_tile_shape: weight_tile_shape.clone(),
210            uvw_tile_shape: uvw_tile_shape.clone(),
211            tiled_column_buffer_bytes: resources.tiled_column_buffer_bytes,
212            data_bands,
213            selected_rows: Vec::new(),
214            columns,
215            array_columns: Vec::new(),
216            array_column_memory_budget_bytes: 0,
217            scalar_columns,
218        })
219    }

221    /// Plan a bounded mutation of explicit existing MAIN rows.
222    pub fn selected_row_mutation(
223        selected_rows: Vec<usize>,
224        columns: Vec<MeasurementSetWriteColumnPlan>,
225        resources: MeasurementSetWriteResources,
226    ) -> Result<Self, MeasurementSetWriteError> {
227        if resources.maximum_live_batches == 0 {
228            return Err(MeasurementSetWriteError::InvalidPlan(
229                "maximum_live_batches must be positive".to_string(),
230            ));
231        }
232        if columns.is_empty() && !selected_rows.is_empty() {
233            return Err(MeasurementSetWriteError::InvalidPlan(
234                "selected-row mutation requires at least one column".to_string(),
235            ));
236        }
237        if let Some(column) = columns.iter().find(|column| {
238            column.create_source_column.is_some()
239                && column.mode != MeasurementSetColumnWriteMode::Create
240        }) {
241            return Err(MeasurementSetWriteError::InvalidPlan(format!(
242                "column {} can clone a source only in create mode",
243                column.name
244            )));
245        }
246        if let Some(column) = columns.iter().find(|column| {
247            (column.mode == MeasurementSetColumnWriteMode::Replace
248                && column.storage_manager != MeasurementSetColumnStorage::Persisted)
249                || (column.mode == MeasurementSetColumnWriteMode::Create
250                    && column.storage_manager != MeasurementSetColumnStorage::TiledShape)
251        }) {
252            return Err(MeasurementSetWriteError::InvalidPlan(format!(
253                "column {} has storage strategy {:?} for {:?} mode",
254                column.name, column.storage_manager, column.mode
255            )));
256        }
257        let mut unique_rows = selected_rows.clone();
258        unique_rows.sort_unstable();
259        unique_rows.dedup();
260        if unique_rows.len() != selected_rows.len() {
261            return Err(MeasurementSetWriteError::InvalidPlan(
262                "selected-row mutation contains duplicate rows".to_string(),
263            ));
264        }
265        let bytes_per_row = columns.iter().try_fold(0usize, |total, column| {
266            total
267                .checked_add(column.bytes_per_row)
268                .ok_or(MeasurementSetWriteError::ByteOverflow)
269        })?;
270        if !selected_rows.is_empty() && bytes_per_row == 0 {
271            return Err(MeasurementSetWriteError::InvalidPlan(
272                "selected-row mutation bytes_per_row must be positive".to_string(),
273            ));
274        }
275        let per_batch_budget = resources.available_bytes / resources.maximum_live_batches;
276        let mut batch_rows = per_batch_budget.checked_div(bytes_per_row).unwrap_or(0);
277        if !selected_rows.is_empty() && batch_rows == 0 {
278            return Err(MeasurementSetWriteError::InsufficientBudget {
279                bytes_per_row,
280                per_batch_budget,
281            });
282        }
283        let tile_rows = columns
284            .iter()
285            .filter_map(|column| column.tile_shape.as_ref()?.last().copied())
286            .filter(|rows| *rows > 0)
287            .min()
288            .unwrap_or(1);
289        if batch_rows >= tile_rows {
290            batch_rows -= batch_rows % tile_rows;
291        }
292        batch_rows = batch_rows.min(selected_rows.len().max(1));
293        let batch_bytes = batch_rows
294            .checked_mul(bytes_per_row)
295            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
296        let queue_capacity = resources
297            .available_bytes
298            .saturating_sub(batch_bytes)
299            .checked_div(batch_bytes)
300            .unwrap_or(0)
301            .min(resources.maximum_live_batches.saturating_sub(1));
302        let maximum_resident_bytes = batch_bytes
303            .checked_mul(queue_capacity.saturating_add(1))
304            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
305        Ok(Self {
306            operation: MeasurementSetWriteOperation::SelectedRowMutation,
307            row_count: selected_rows.len(),
308            correlation_count: 0,
309            channel_count: 0,
310            bytes_per_row,
311            batch_rows,
312            batch_bytes,
313            queue_capacity,
314            maximum_resident_bytes,
315            visibility_tile_shape: Vec::new(),
316            weight_tile_shape: Vec::new(),
317            uvw_tile_shape: Vec::new(),
318            tiled_column_buffer_bytes: 0,
319            data_bands: 1,
320            selected_rows,
321            columns,
322            array_columns: Vec::new(),
323            array_column_memory_budget_bytes: 0,
324            scalar_columns: Vec::new(),
325        })
326    }

328    /// Plan canonical variable-shape MAIN columns for a new MeasurementSet.
329    #[doc(hidden)]
330    pub fn variable_array_creation(
331        row_count: usize,
332        array_columns: Vec<MeasurementSetArrayColumnPlan>,
333        scalar_columns: Vec<MeasurementSetScalarColumnPlan>,
334        resources: MeasurementSetWriteResources,
335    ) -> Result<Self, MeasurementSetWriteError> {
336        if resources.maximum_live_batches == 0 {
337            return Err(MeasurementSetWriteError::InvalidPlan(
338                "maximum_live_batches must be positive".to_string(),
339            ));
340        }
341        if array_columns.is_empty() && scalar_columns.is_empty() && row_count > 0 {
342            return Err(MeasurementSetWriteError::InvalidPlan(
343                "creation write requires at least one column".to_string(),
344            ));
345        }
346        let mut names = HashSet::with_capacity(array_columns.len() + scalar_columns.len());
347        for column in &array_columns {
348            if !names.insert(column.name.as_str()) {
349                return Err(MeasurementSetWriteError::InvalidPlan(format!(
350                    "duplicate variable-array column {}",
351                    column.name
352                )));
353            }
354            if !matches!(
355                column.storage_manager,
356                MeasurementSetColumnStorage::TiledColumn | MeasurementSetColumnStorage::TiledShape
357            ) {
358                return Err(MeasurementSetWriteError::InvalidPlan(format!(
359                    "array creation column {} requires a tiled creation manager, found {:?}",
360                    column.name, column.storage_manager
361                )));
362            }
363            let planned_rows = column.shapes.iter().try_fold(0usize, |total, shape| {
364                total
365                    .checked_add(shape.row_count)
366                    .ok_or(MeasurementSetWriteError::ByteOverflow)
367            })?;
368            if planned_rows != row_count {
369                return Err(MeasurementSetWriteError::InvalidPlan(format!(
370                    "column {} shape histogram has {planned_rows} rows; plan requires {row_count}",
371                    column.name
372                )));
373            }
374            if column.storage_manager == MeasurementSetColumnStorage::TiledColumn
375                && (column.shapes.len() != 1
376                    || column.shapes[0].row_count != row_count
377                    || row_count == 0)
378            {
379                return Err(MeasurementSetWriteError::InvalidPlan(format!(
380                    "TiledColumnStMan column {} requires one defined shape for every row",
381                    column.name
382                )));
383            }
384        }
385        for column in &scalar_columns {
386            if !names.insert(column.name.as_str()) {
387                return Err(MeasurementSetWriteError::InvalidPlan(format!(
388                    "duplicate creation column {}",
389                    column.name
390                )));
391            }
392        }
393        let array_bytes_per_row = array_columns.iter().try_fold(0usize, |total, column| {
394            let element_bytes = match column.value_type {
395                StreamedTiledShapeValueType::Bool => 1,
396                StreamedTiledShapeValueType::Float32 | StreamedTiledShapeValueType::Int32 => 4,
397                StreamedTiledShapeValueType::Float64 => 8,
398                StreamedTiledShapeValueType::Complex32 => 8,
399            };
400            let column_bytes = column
401                .shapes
402                .iter()
403                .try_fold(0usize, |maximum, shape| {
404                    let elements = shape.cell_shape.iter().try_fold(1usize, |total, extent| {
405                        total
406                            .checked_mul(*extent)
407                            .ok_or(MeasurementSetWriteError::ByteOverflow)
408                    })?;
409                    Ok::<_, MeasurementSetWriteError>(maximum.max(elements))
410                })?
411                .checked_mul(element_bytes)
412                .ok_or(MeasurementSetWriteError::ByteOverflow)?;
413            total
414                .checked_add(column_bytes)
415                .ok_or(MeasurementSetWriteError::ByteOverflow)
416        })?;
417        let scalar_bytes_per_row = scalar_columns.iter().try_fold(0usize, |total, column| {
418            total
419                .checked_add(column.bytes_per_row())
420                .ok_or(MeasurementSetWriteError::ByteOverflow)
421        })?;
422        let bytes_per_row = array_bytes_per_row
423            .checked_add(scalar_bytes_per_row)
424            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
425        // The scalar writers' buffers while rows stream, and the save's
426        // finalization scratch after them.
427        let scalar_buffer_bytes = streamed_scalar_columns_resident_bytes(scalar_columns.len())
428            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
429        if scalar_buffer_bytes > resources.available_bytes {
430            return Err(MeasurementSetWriteError::InvalidPlan(format!(
431                "scalar column buffers require {scalar_buffer_bytes} bytes; writer budget is {} bytes",
432                resources.available_bytes
433            )));
434        }
435        let streaming_budget = resources.available_bytes - scalar_buffer_bytes;
436        let batch_budget = streaming_budget / resources.maximum_live_batches;
437        let mut batch_rows = batch_budget.checked_div(bytes_per_row).unwrap_or(0);
438        if row_count > 0 && batch_rows == 0 {
439            return Err(MeasurementSetWriteError::InsufficientBudget {
440                bytes_per_row,
441                per_batch_budget: batch_budget,
442            });
443        }
444        batch_rows = batch_rows.min(row_count.max(1));
445        let batch_bytes = batch_rows
446            .checked_mul(bytes_per_row)
447            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
448        let writer_budget = streaming_budget.saturating_sub(batch_bytes);
449        let array_column_memory_budget_bytes = if array_columns.is_empty() {
450            0
451        } else {
452            writer_budget / array_columns.len()
453        };
454        if row_count > 0 && !array_columns.is_empty() && array_column_memory_budget_bytes == 0 {
455            return Err(MeasurementSetWriteError::InsufficientBudget {
456                bytes_per_row: 1,
457                per_batch_budget: 0,
458            });
459        }
460        let maximum_resident_bytes = array_column_memory_budget_bytes
461            .checked_mul(array_columns.len())
462            .and_then(|bytes| bytes.checked_add(batch_bytes))
463            .and_then(|bytes| bytes.checked_add(scalar_buffer_bytes))
464            .ok_or(MeasurementSetWriteError::ByteOverflow)?;
465        let mut columns = array_columns
466            .iter()
467            .map(|column| MeasurementSetWriteColumnPlan {
468                name: column.name.clone(),
469                bytes_per_row: column
470                    .shapes
471                    .iter()
472                    .map(|shape| shape.cell_shape.iter().product::<usize>())
473                    .max()
474                    .unwrap_or(0)
475                    * match column.value_type {
476                        StreamedTiledShapeValueType::Bool => 1,
477                        StreamedTiledShapeValueType::Float32
478                        | StreamedTiledShapeValueType::Int32 => 4,
479                        StreamedTiledShapeValueType::Float64
480                        | StreamedTiledShapeValueType::Complex32 => 8,
481                    },
482                mode: MeasurementSetColumnWriteMode::Create,
483                storage_manager: column.storage_manager,
484                tile_shape: column.shapes.first().map(|shape| shape.tile_shape.clone()),
485                create_source_column: None,
486            })
487            .collect::<Vec<_>>();
488        columns.extend(
489            scalar_columns
490                .iter()
491                .map(|column| MeasurementSetWriteColumnPlan {
492                    name: column.name.clone(),
493                    bytes_per_row: column.bytes_per_row(),
494                    mode: MeasurementSetColumnWriteMode::Create,
495                    storage_manager: MeasurementSetColumnStorage::Standard,
496                    tile_shape: None,
497                    create_source_column: None,
498                }),
499        );
500        Ok(Self {
501            operation: MeasurementSetWriteOperation::Create,
502            row_count,
503            correlation_count: 0,
504            channel_count: 0,
505            bytes_per_row,
506            batch_rows,
507            batch_bytes,
508            queue_capacity: 0,
509            maximum_resident_bytes,
510            visibility_tile_shape: Vec::new(),
511            weight_tile_shape: Vec::new(),
512            uvw_tile_shape: Vec::new(),
513            tiled_column_buffer_bytes: 0,
514            data_bands: 1,
515            selected_rows: Vec::new(),
516            columns,
517            array_columns,
518            array_column_memory_budget_bytes,
519            scalar_columns,
520        })
521    }
522    /// Set default tile extents for the planned tiled columns, by column name.
523    pub fn with_array_default_tiles(
524        mut self,
525        shape_for_column: impl Fn(&str) -> Vec<usize>,
526    ) -> Self {
527        for column in &mut self.columns {
528            if column.tile_shape.is_some() {
529                column.tile_shape = Some(shape_for_column(&column.name));
530            }
531        }
532        self
533    }
