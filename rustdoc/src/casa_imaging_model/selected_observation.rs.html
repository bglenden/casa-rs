15const SELECTED_OBSERVATION_COMMITMENT_IDENTITY_DOMAIN: &[u8] =
16    b"casa-rs-selected-observation-commitment";
17const SELECTED_OBSERVATION_COMMITMENT_IDENTITY_VERSION: u32 = 1;

19/// Stable compiler-derived identity of one selected-observation scientific commitment.
20///
21/// There is deliberately no constructor from raw digest bytes. Only logical
22/// problem compilation can mint this identity.
23///
24/// ```compile_fail
25/// use casa_imaging_model::SelectedObservationCommitmentId;
26///
27/// let _ = SelectedObservationCommitmentId::from_bytes([0; 32]);
28/// ```
29#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
30pub struct SelectedObservationCommitmentId([u8; 32]);
31
32impl SelectedObservationCommitmentId {
33    /// Identity schema version used by the canonical encoder.
34    pub const SCHEMA_VERSION: u32 = SELECTED_OBSERVATION_COMMITMENT_IDENTITY_VERSION;
35
36    /// Return the exact SHA-256 digest.
37    #[must_use]
38    pub const fn as_bytes(self) -> [u8; 32] {
39        self.0
40    }
41}
42
43impl fmt::Debug for SelectedObservationCommitmentId {
44    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
45        formatter.write_str("SelectedObservationCommitmentId(")?;
46        write_hex(formatter, &self.0)?;
47        formatter.write_str(")")
48    }
49}
50
51impl fmt::Display for SelectedObservationCommitmentId {
52    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
53        write_hex(formatter, &self.0)
54    }
55}

57/// Sample-space evaluation semantics owned by the selected-observation compiler seam.
58///
59/// Reconstruction, product, weighting, numerical, and physical execution
60/// choices are deliberately absent.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub struct SelectedSampleEvaluation {
63    visibility_inner_product: VisibilityInnerProduct,
64    spectral_sampling: SpectralSamplingLaw,
65}
66
67impl SelectedSampleEvaluation {
68    /// Return the exact visibility-sample inner product.
69    #[must_use]
70    pub const fn visibility_inner_product(self) -> VisibilityInnerProduct {
71        self.visibility_inner_product
72    }
73
74    /// Return the paired spectral sampling applied to selected samples.
75    #[must_use]
76    pub const fn spectral_sampling(self) -> SpectralSamplingLaw {
77        self.spectral_sampling
78    }
79}

81/// Immutable compiler-owned science that every selected-observation traversal must realize.
82///
83/// The commitment contains no block size, read-ahead depth, double-buffering,
84/// worker, backend, resource policy, reconstruction, weighting, or publication
85/// choice. The retained read set is derived from the same snapshot during
86/// compilation and supplies the exact typed source projection to traversal.
87#[derive(Debug, Clone, PartialEq)]
88pub struct SelectedObservationCommitment {
89    commitment_id: SelectedObservationCommitmentId,
90    observation_snapshot_id: ObservationSnapshotId,
91    geometry_id: CompiledGeometryId,
92    sample_evaluation: SelectedSampleEvaluation,
93    read_set: ObservationReadSet,
94}
95
96impl SelectedObservationCommitment {
97    /// Return the compiler-derived identity of this exact scientific commitment.
98    #[must_use]
99    pub const fn commitment_id(&self) -> SelectedObservationCommitmentId {
100        self.commitment_id
101    }
102
103    /// Return the immutable atomic snapshot defining source selection and generations.
104    #[must_use]
105    pub const fn observation_snapshot_id(&self) -> ObservationSnapshotId {
106        self.observation_snapshot_id
107    }
108
109    /// Return the compiled coordinate, phase, and spectral-geometry identity.
110    #[must_use]
111    pub const fn geometry_id(&self) -> CompiledGeometryId {
112        self.geometry_id
113    }
114
115    /// Return the exact selected-sample evaluation semantics.
116    #[must_use]
117    pub const fn sample_evaluation(&self) -> SelectedSampleEvaluation {
118        self.sample_evaluation
119    }
120
121    /// Return exact source selection and generation semantics in canonical source order.
122    #[must_use]
123    pub const fn read_set(&self) -> &ObservationReadSet {
124        &self.read_set
125    }
126}

128pub(crate) fn compile_selected_observation_commitment(
129    observation_transaction: &ObservationTransactionContract,
130    geometry_id: CompiledGeometryId,
131    visibility_inner_product: VisibilityInnerProduct,
132    spectral_sampling: SpectralSamplingLaw,
133) -> SelectedObservationCommitment {
134    let observation_snapshot_id = observation_transaction.observation_snapshot_id();
135    let sample_evaluation = SelectedSampleEvaluation {
136        visibility_inner_product,
137        spectral_sampling,
138    };
139    let commitment_id =
140        selected_observation_commitment_id(observation_snapshot_id, geometry_id, sample_evaluation);
141    SelectedObservationCommitment {
142        commitment_id,
143        observation_snapshot_id,
144        geometry_id,
145        sample_evaluation,
146        read_set: observation_transaction.read_set().clone(),
147    }
148}
149
150fn selected_observation_commitment_id(
151    observation_snapshot_id: ObservationSnapshotId,
152    geometry_id: CompiledGeometryId,
153    sample_evaluation: SelectedSampleEvaluation,
154) -> SelectedObservationCommitmentId {
155    let mut encoder = CanonicalEncoder::new();
156    encoder.bytes(SELECTED_OBSERVATION_COMMITMENT_IDENTITY_DOMAIN);
157    encoder.u32(SELECTED_OBSERVATION_COMMITMENT_IDENTITY_VERSION);
158    encoder.digest(observation_snapshot_id.as_bytes());
159    encoder.digest(geometry_id.as_bytes());
160    encode_visibility_inner_product(&mut encoder, sample_evaluation.visibility_inner_product);
161    encode_spectral_sampling(&mut encoder, sample_evaluation.spectral_sampling);
162    SelectedObservationCommitmentId(encoder.finish())
163}
164
165fn encode_visibility_inner_product(
166    encoder: &mut CanonicalEncoder,
167    inner_product: VisibilityInnerProduct,
168) {
169    encoder.u8(match inner_product {
170        VisibilityInnerProduct::HermitianEuclidean => 0,
171    });
172}
173
174fn encode_spectral_sampling(encoder: &mut CanonicalEncoder, sampling: SpectralSamplingLaw) {
175    encode_spectral_sampling_law(encoder, sampling);
176}

178fn write_hex(formatter: &mut fmt::Formatter<'_>, bytes: &[u8]) -> fmt::Result {
179    for byte in bytes {
180        write!(formatter, "{byte:02x}")?;
181    }
182    Ok(())
