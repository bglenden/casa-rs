19#[pymethods]
20impl PyMsTool {
21    #[pyo3(signature = (verbose = false, listfile = None, listunfl = false, cachesize = 50.0, overwrite = false, wantreturn = true), text_signature = "($self, verbose=False, listfile='', listunfl=False, cachesize=50.0, overwrite=False, wantreturn=True)")]
22    fn summary(
23        slf: &Bound<'_, Self>,
24        #[pyo3(from_py_with = strict_bool)] verbose: bool,
25        #[pyo3(from_py_with = listfile_argument)] listfile: Option<PathBuf>,
26        #[pyo3(from_py_with = strict_bool)] listunfl: bool,
27        cachesize: f64,
28        #[pyo3(from_py_with = strict_bool)] overwrite: bool,
29        #[pyo3(from_py_with = strict_bool)] wantreturn: bool,
30    ) -> PyResult<Py<PyAny>> {
31        let py = slf.py();
32        let listfile = listfile.filter(|file| !file.as_os_str().is_empty());
33        let result = slf.get().attachment_info(py, |attachment| {
34            let Some(attachment) = attachment else {
35                return Ok(None);
36            };
37            if let Some(path) = listfile.as_deref()
38                && !overwrite
39                && path.exists()
40            {
41                return Err(existing_report(path));
42            }
43            let link = attachment.working_metadata_link();
44            let (snapshot, rows) = link.source_and_rows().map_err(runtime_error)?;
45            let ms = &snapshot.ms;
46            let (mut summary, directions) = MeasurementSetSummary::from_tool_view(
47                ms,
48                rows.as_deref(),
49                verbose,
50                listunfl,
51                cachesize,
52                &attachment.visibility.time_measures,
53            )
54            .map_err(runtime_error)?;
55            summary.measurement_set.path = Some(attachment.name.display().to_string());
56            let record = if wantreturn {
57                summary_record(&summary, directions)
58            } else {
59                CasaValue::empty_record()
60            };
61            let report = summary.render_tool_text();
62            link.verify_directory().map_err(runtime_error)?;
63            if let Some(path) = listfile.as_deref() {
64                write_report(path, overwrite, &report)?;
65            }
66            Ok(Some((record, report)))
67        })?;
68        let Some((record, report)) = result else {
69            return CasaValue::empty_record().into_python(py);
70        };
71        let line = if let Some(path) = listfile.as_deref() {
72            LogLine::info(format!("Writing output to file: {}", path.display()))
73        } else {
74            LogLine::info(report)
75        };
76        emit(py, "casars.tools.ms", &[line])?;
77        record.into_python(py)
78    }
79}

81fn write_report(path: &Path, overwrite: bool, report: &str) -> PyResult<()> {
82    let mut file = OpenOptions::new()
83        .write(true)
84        .create(overwrite)
85        .truncate(overwrite)
86        .create_new(!overwrite)
87        .open(path)
88        .map_err(|error| {
89            if error.kind() == io::ErrorKind::AlreadyExists && !overwrite {
90                existing_report(path)
91            } else {
92                runtime_error(format!("cannot write {}: {error}", path.display()))
93            }
94        })?;
95    file.write_all(report.as_bytes()).map_err(runtime_error)
96}

98fn existing_report(path: &Path) -> PyErr {
99    runtime_error(format!(
100        "File: {} already exists; delete it, choose another name, or set overwrite=true.",
101        path.display()
102    ))
103}

105fn summary_record(
106    summary: &MeasurementSetSummary,
107    directions: Vec<MetadataDirection>,
108) -> CasaValue {
109    let info = &summary.measurement_set;
110    let Some((begin, end)) = info.start_mjd_seconds.zip(info.end_mjd_seconds) else {
111        return CasaValue::empty_record();
112    };
113    let mut record = vec![
114        (
115            "numrecords".to_string(),
116            CasaValue::Int(info.row_count as i64),
117        ),
118        ("IntegrationTime".to_string(), CasaValue::Float(end - begin)),
119        ("BeginTime".to_string(), CasaValue::Float(begin / 86_400.0)),
120        ("EndTime".to_string(), CasaValue::Float(end / 86_400.0)),
121        (
122            "timeref".to_string(),
123            CasaValue::Text(info.time_reference.clone().unwrap_or_default()),
124        ),
125        (
126            "nfields".to_string(),
127            CasaValue::Int(info.field_count as i64),
128        ),
129    ];
130    for (field, direction) in summary.fields.iter().zip(directions) {
131        record.push((
132            format!("field_{}", field.field_id),
133            CasaValue::record([
134                ("name", CasaValue::Text(mark_long_field_name(&field.name))),
135                ("code", CasaValue::Text(field.code.clone())),
136                ("direction", direction_record(direction)),
137            ]),
138        ));
139    }
140
141    let mut scans = BTreeMap::<i32, BTreeMap<usize, CasaValue>>::new();
142    let mut previous = None;
143    let mut subset = 0;
144    for scan in summary.scans.iter().take(10_000) {
145        let key = (scan.observation_id, scan.array_id, scan.scan_number);
146        subset = if previous == Some(key) { subset + 1 } else { 0 };
147        previous = Some(key);
148        scans
149            .entry(scan.scan_number)
150            .or_default()
151            .insert(subset, scan_record(scan, summary.options.verbose));
152    }
153    record.extend(scans.into_iter().map(|(scan, fields)| {
154        let subs = fields
155            .into_iter()
156            .map(|(index, value)| (index.to_string(), value));
157        (format!("scan_{scan}"), CasaValue::record(subs))
158    }));
159    CasaValue::record(record)
160}

162fn scan_record(scan: &MeasurementSetScanSummary, verbose: bool) -> CasaValue {
163    CasaValue::record([
164        (
165            "BeginTime",
166            CasaValue::Float(scan.start_mjd_seconds / 86_400.0),
167        ),
168        ("EndTime", CasaValue::Float(scan.end_mjd_seconds / 86_400.0)),
169        ("scanId", CasaValue::Int(scan.scan_number.into())),
170        ("FieldId", CasaValue::Int(scan.field_id.into())),
171        (
172            "FieldName",
173            CasaValue::Text(if verbose {
174                mark_long_field_name(&scan.field_name)
175            } else {
176                scan.field_name.clone()
177            }),
178        ),
179        (
180            "StateId",
181            CasaValue::Int(i64::from(scan.state_ids.first().copied().unwrap_or(-1))),
182        ),
183        ("nRow", CasaValue::Int(scan.row_count as i64)),
184        (
185            "IntegrationTime",
186            CasaValue::Float(scan.mean_interval_seconds),
187        ),
188        (
189            "SpwIds",
190            CasaValue::Ints {
191                shape: vec![scan.spectral_window_ids.len()],
192                values: scan
193                    .spectral_window_ids
194                    .iter()
195                    .copied()
196                    .map(i64::from)
197                    .collect(),
198            },
199        ),
200    ])
201}

203fn mark_long_field_name(name: &str) -> String {
204    let Some((suffix, _)) = name.char_indices().nth(20) else {
205        return name.into();
206    };
207    let (marker, _) = name[..suffix]
208        .char_indices()
209        .next_back()
210        .expect("at least twenty characters");
211    format!("{}*{}", &name[..marker], &name[suffix..])
212}

214fn listfile_argument(value: &Bound<'_, PyAny>) -> PyResult<Option<PathBuf>> {
215    #[cfg(unix)]
216    if let Ok(bytes) = value.cast::<PyBytes>() {
217        use std::os::unix::ffi::OsStringExt;
218        let path = std::ffi::OsString::from_vec(bytes.as_bytes().to_vec());
219        return Ok(Some(path.into()));
220    }
221    selection_text(value, "listfile").map(|path| Some(path.into()))
222}
