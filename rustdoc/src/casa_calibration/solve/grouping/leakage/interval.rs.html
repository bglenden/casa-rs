22pub(super) enum IntervalFits {

43/// Convert and normalize the referenced parameters before publication, as
44/// the Calibrater's generic Jones post-solve pass does.
45pub(super) fn finish_tool_parameters(
46    solutions: &mut [LeakageSolution],
47    mode: GainSolveMode,
48    normalization: Option<GainNormalization>,
49) {
50    if mode != GainSolveMode::AmplitudePhase {
51        for estimate in solutions
52            .iter_mut()
53            .flat_map(|slot| slot.values.iter_mut().flatten())
54        {
55            for (gain, valid) in estimate.gain.iter_mut().zip(estimate.valid) {
56                let amplitude = gain.norm();
57                *gain = if !valid || amplitude == 0.0 {
58                    Complex32::ONE
59                } else if mode == GainSolveMode::Phase {
60                    *gain / amplitude
61                } else {
62                    amplitude.into()
63                };
64            }
65        }
66    }
67    let Some(normalization) = normalization else {
68        return;
69    };
70    let mut power = BTreeMap::<i32, (f64, usize, Vec<f32>)>::new();
71    for slot in solutions.iter() {
72        let (sum, count, samples) = power.entry(slot.spectral_window).or_default();
73        for estimate in slot.values.iter().flatten() {
74            for (gain, valid) in estimate.gain.iter().zip(estimate.valid) {
75                if valid {
76                    let value = gain.norm_sqr();
77                    match normalization {
78                        GainNormalization::Mean => *sum += f64::from(value),
79                        GainNormalization::Median => samples.push(value),
80                    }
81                    *count += 1;
82                }
83            }
84        }
85    }
86    let factors: BTreeMap<_, _> = power
87        .into_iter()
88        .filter_map(|(spw, (sum, count, mut samples))| {
89            if count <= 1 {
90                return None;
91            }
92            let factor = match normalization {
93                GainNormalization::Mean => (sum / count as f64).sqrt() as f32,
94                GainNormalization::Median => {
95                    super::super::median_with_even_mean(&mut samples).sqrt()
96                }
97            };
98            (factor > 0.0 && factor.is_finite()).then_some((spw, factor))
99        })
100        .collect();
101    for solution in solutions {
102        if let Some(&factor) = factors.get(&solution.spectral_window) {
103            for estimate in solution.values.iter_mut().flatten() {
104                for gain in &mut estimate.gain {
105                    *gain /= factor;
106                }
107            }
108        }
109    }
110}

112impl IntervalFits {
113    pub(super) fn new(
114        channels: usize,
115        antennas: usize,
116        kind: LeakageSolveKind,
117        circular: bool,
118        mode: GainSolveMode,
119        legacy: bool,
120    ) -> Self {
121        if kind.parallactic() {
122            Self::Parallactic(ParallacticAverages::new(
123                channels,
124                circular,
125                kind.gain_phase(),
126                mode,
127                legacy,
128            ))
129        } else if kind.cross_hand() {
130            let measurement = match (kind, circular) {
131                (LeakageSolveKind::PosAng, true) => Measurement::CircularAngle,
132                (LeakageSolveKind::PosAng, false) => Measurement::LinearAngle,
133                (LeakageSolveKind::XMueller, _) => Measurement::MuellerPhase,
134                _ => Measurement::Phase,
135            };
136            Self::CrossHand(
137                (0..channels).map(|_| CrossHandFit::default()).collect(),
138                measurement,
139                if kind == LeakageSolveKind::XMueller {
140                    GainSolveMode::AmplitudePhase
141                } else {
142                    mode
143                },
144            )
145        } else if kind.iterative() {
146            Self::Iterative {
147                algebra: kind.algebra(),
148                samples: ChannelSamples::new(channels),
149                source: kind.source().map(|mode| (mode, SourceFit::default())),
150                mode,
151            }
152        } else {
153            Self::LeastSquares(
154                (0..channels).map(|_| LeakageFit::new(antennas)).collect(),
155                mode,
156            )
157        }
158    }

160    pub(super) fn accumulate(
161        &mut self,
162        pair: [i32; 2],
163        bins: &mut [CoherencyMean],
164        antennas: usize,
165        segment: BaselineSegment,
166        legacy_cross_hand: bool,
167    ) -> Result<(), GainSolveError> {
168        let pair = pair.map(|antenna| antenna as usize);
169        if pair.iter().any(|&antenna| antenna >= antennas) {
170            return Ok(());
171        }
172        if let Self::Parallactic(average) = self {
173            return average.add(pair.map(|antenna| antenna as i32), segment, bins);
174        }
175        let require_all = matches!(self, Self::LeastSquares(..));
176        let legacy_phase_normalization = (legacy_cross_hand
177            && matches!(self, Self::CrossHand(_, _, GainSolveMode::Phase)))
178        .then(|| super::cross_hand::legacy_phase_normalization(bins));
179        for (channel, mean) in bins.iter_mut().enumerate() {
180            let unflagged = if pair[0] == pair[1] {
181                0
182            } else {
183                mean.unflagged
184            };
185            if let Some((mut data, model, mut weights)) = std::mem::take(mean).finish(require_all) {
186                if let Self::LeastSquares(_, mode)
187                | Self::CrossHand(_, _, mode)
188                | Self::Iterative { mode, .. } = self
189                    && *mode != GainSolveMode::AmplitudePhase
190                {
191                    for (corr, (value, weight)) in data.iter_mut().zip(&mut weights).enumerate() {
192                        let (normalized, scaled) = solve_sample(*mode, *value, *weight);
193                        *value = normalized;
194                        *weight = legacy_phase_normalization
195                            .map_or(scaled, |normalization| *weight * normalization[corr]);
196                    }
197                }
198                match self {
199                    Self::LeastSquares(fits, _) => {
200                        fits[channel].add_baseline(pair, data, model, weights)
201                    }
202                    Self::CrossHand(fits, measurement, _) => {
203                        fits[channel].add(*measurement, unflagged, data, model, weights)
204                    }
205                    Self::Parallactic(_) => unreachable!("time-segment accumulation"),
206                    Self::Iterative {
207                        algebra,
208                        samples,
209                        source,
210                        ..
211                    } => {
212                        let mut weights = weights;
213                        if matches!(algebra, LeakageAlgebra::Linearized) {
214                            weights[0] = 0.0;
215                            weights[3] = 0.0;
216                        }
217                        let sample = StandardSample {
218                            pair,
219                            data,
220                            model,
221                            weights,
222                        };
223                        if let Some((_, fit)) = source {
224                            fit.observe(&sample);
225                        }
226                        samples
227                            .push(channel, sample)
228                            .map_err(GainSolveError::PolarizationScratch)?;
229                    }
230                }
231            }
232        }
233        Ok(())
234    }

236    #[allow(clippy::too_many_arguments)]
237    pub(super) fn finish(
238        self,
239        antennas: usize,
240        references: &[i32],
241        previous: Option<&LeakageSolution>,
242        request: &LeakageSolveRequest,
243        location: (i32, i32),
244        angles: Option<&ParallacticAngleState>,
245        legacy_times: Option<&LegacyCrossHandAveraging>,
246        stages: Option<&mut LeakageFitStages>,
247    ) -> Result<Option<IntervalEstimates>, GainSolveError> {
248        match self {
249            Self::LeastSquares(fits, _) => Ok(least_squares(
250                fits,
251                antennas,
252                references,
253                previous,
254                stages.is_some(),
255            )),
256            Self::CrossHand(fits, measurement, _) => {
257                Ok(cross_hand(fits, measurement, antennas, location))
258            }
259            Self::Parallactic(average) => {
260                let angles = angles.expect("parallactic angle geometry");
261                let (phases, source) = average.finish(angles, legacy_times)?;
262                SourceFitMode::Polarization.report(source, location);
263                let gain_phase = request.kind.gain_phase();
264                let values = phases
265                    .into_iter()
266                    .map(|phase| {
267                        let estimate = LeakageEstimate {
268                            gain: [
269                                phase.unwrap_or(Complex32::ONE),
270                                if gain_phase {
271                                    Complex32::ONE
272                                } else {
273                                    Complex32::ZERO
274                                },
275                            ],
276                            valid: [phase.is_some(), gain_phase && phase.is_some()],
277                            ..LeakageEstimate::default()
278                        };
279                        vec![estimate; antennas]
280                    })
281                    .collect();
282                Ok(Some((-1, values, Some(source))))
283            }
284            Self::Iterative {
285                algebra,
286                samples,
287                source,
288                ..
289            } => {
290                if samples.is_empty() || source.as_ref().is_some_and(|(_, fit)| !fit.has_model()) {
291                    return Ok(None);
292                }
293                let multiplier = source
294                    .map(|(mode, fit)| {
295                        let multiplier = fit
296                            .finish(mode)
297                            .ok_or(GainSolveError::SingularSourcePolarization)?;
298                        mode.report(multiplier, location);
299                        Ok::<_, GainSolveError>(multiplier)
300                    })
301                    .transpose()?;
302                Ok(
303                    iterative(samples, antennas, request, multiplier, algebra, stages)?
304                        .map(|values| (-1, values, None)),
305                )
306            }
307        }
308    }
309}

311fn cross_hand(
312    fits: Vec<CrossHandFit>,
313    measurement: Measurement,
314    antennas: usize,
315    (field, spw): (i32, i32),
316) -> Option<IntervalEstimates> {
317    if !fits.iter().any(CrossHandFit::saw_samples) {
318        return None;
319    }
320    let mut sum = Complex32::ZERO;
321    let mut channels = 0;
322    let position_angle = matches!(
323        measurement,
324        Measurement::CircularAngle | Measurement::LinearAngle
325    );
326    let values = fits
327        .into_iter()
328        .map(|fit| {
329            let phase = fit.phase();
330            if let Some(phase) = phase {
331                sum += phase;
332                channels += 1;
333            }
334            let parameter = if position_angle {
335                Complex32::new(phase.map_or(0.0, |value| 0.5 * value.arg()), 0.0)
336            } else {
337                phase.unwrap_or(Complex32::ONE)
338            };
339            let estimate = LeakageEstimate {
340                gain: [parameter, Complex32::ZERO],
341                valid: [phase.is_some(), false],
342                ..LeakageEstimate::default()
343            };
344            vec![estimate; antennas]
345        })
346        .collect();
347    if channels > 0 && position_angle {
348        tracing::info!(
349            field,
350            spw,
351            position_angle_degrees = 0.5 * sum.arg().to_degrees(),
352            "Mean position angle offset solution"
353        );
354    } else if channels > 0 {
355        tracing::info!(
356            field,
357            spw,
358            phase_degrees = sum.arg().to_degrees(),
359            "Mean cross-hand phase solution"
360        );
361    }
362    Some((-1, values, None))
363}

365fn least_squares(
366    fits: Vec<LeakageFit>,
367    antennas: usize,
368    references: &[i32],
369    previous: Option<&LeakageSolution>,
370    keep_flagged: bool,
371) -> Option<IntervalEstimates> {
372    let reference = choose_reference(references, previous, |antenna| {
373        fits.iter().any(|fit| fit.has_antenna(antenna))
374    });
375    let mut nonempty = false;
376    let values = fits
377        .into_iter()
378        .enumerate()
379        .map(|(channel, fit)| {
380            let target = previous_reference(previous, channel, reference);
381            let Some(fitted) = fit.solve(reference, target) else {
382                return vec![LeakageEstimate::default(); antennas];
383            };
384            nonempty |= fitted.iter().any(Option::is_some);
385            fitted
386                .into_iter()
387                .map(|pair| {
388                    pair.filter(|_| references.is_empty() || reference >= 0)
389                        .map_or_else(LeakageEstimate::default, |gain| LeakageEstimate {
390                            gain,
391                            valid: [true; 2],
392                            ..LeakageEstimate::default()
393                        })
394                })
395                .collect()
396        })
397        .collect();
398    // Direct generic gather retains an attempted self-directed solve even
399    // when its individual fits could only write flagged parameters.
400    (nonempty || keep_flagged).then_some((reference, values, None))
401}

403fn iterative(
404    mut samples: ChannelSamples,
405    antennas: usize,
406    request: &LeakageSolveRequest,
407    multiplier: Option<Complex32>,
408    algebra: LeakageAlgebra,
409    mut stages: Option<&mut LeakageFitStages>,
410) -> Result<Option<Vec<ChannelEstimates>>, GainSolveError> {
411    let mut gains = vec![[Complex32::ZERO; 2]; antennas];
412    let mut diagnostics = vec![LeakageEstimate::default(); antennas];
413    let mut converged = false;
414    let mut values = Vec::with_capacity(samples.len());
415    if let Some(stages) = &mut stages {
416        stages.passed.resize(antennas, [false; 2]);
417    }
418    for channel in (0..samples.len()).rev() {
419        let mut channel_samples = samples
420            .take(channel)
421            .map_err(GainSolveError::PolarizationScratch)?;
422        if let Some(multiplier) = multiplier {
423            for sample in &mut channel_samples {
424                sample.model[1] *= multiplier;
425                sample.model[2] *= multiplier.conj();
426            }
427        }
428        let eligible = eligible_antennas(&channel_samples, antennas, request.min_baselines);
429        if let Some(stages) = &mut stages {
430            stages.constrained.extend(
431                eligible
432                    .iter()
433                    .enumerate()
434                    .filter_map(|(id, &valid)| valid.then_some(id as i32)),
435            );
436        }
437        values.push(
438            match solve_channel(
439                &channel_samples,
440                &mut gains,
441                &eligible,
442                request.min_snr,
443                algebra,
444            ) {
445                Ok(result) => {
446                    if let Some(stages) = &mut stages {
447                        for (passed, estimate) in stages.passed.iter_mut().zip(&result) {
448                            for (pass, valid) in passed.iter_mut().zip(estimate.valid) {
449                                *pass |= valid;
450                            }
451                        }
452                    }
453                    diagnostics.copy_from_slice(&result);
454                    converged = true;
455                    result
456                }
457                Err(valid) => {
458                    // CASA keeps the prior diagnostics and the current
459                    // parameter mask even when this channel cannot converge.
460                    diagnostics
461                        .iter()
462                        .zip(gains.iter().zip(valid))
463                        .map(|(diagnostics, (&gain, valid))| LeakageEstimate {
464                            gain,
465                            valid,
466                            ..*diagnostics
467                        })
468                        .collect()
469                }
470            },
471        );
472    }
473    values.reverse();
474    Ok(converged.then_some(values))
475}

477pub(super) fn reference_iterative_solutions(
478    solutions: &mut [LeakageSolution],
479    references: &mut [i32],
480) {
481    solutions.sort_by(|a, b| {
482        a.spectral_window
483            .cmp(&b.spectral_window)
484            .then_with(|| a.time.total_cmp(&b.time))
485    });
486    if references.is_empty() {
487        return;
488    }
489    let second_receptor = solutions
490        .iter()
491        .filter(|solution| {
492            solution
493                .values
494                .iter()
495                .flatten()
496                .any(|sample| sample.valid[1])
497        })
498        .map(|solution| solution.spectral_window)
499        .collect::<std::collections::BTreeSet<_>>();
500    let mut previous: Option<usize> = None;
501    for index in 0..solutions.len() {
502        let current = &solutions[index];
503        let previous_solution = previous
504            .map(|index| &solutions[index])
505            .filter(|previous| previous.spectral_window == current.spectral_window);
506        let receptors = 1 + usize::from(second_receptor.contains(&current.spectral_window));
507        let reference = references
508            .iter()
509            .copied()
510            .find(|&antenna| {
511                usable_reference(&current.values, antenna, receptors)
512                    && previous_solution.is_none_or(|previous| {
513                        usable_reference(&previous.values, antenna, receptors)
514                    })
515            })
516            .unwrap_or(-1);
517        let offsets = (reference >= 0).then(|| {
518            current
519                .values
520                .iter()
521                .enumerate()
522                .map(|(channel, values)| {
523                    values[reference as usize].gain[0]
524                        - previous_reference(previous_solution, channel, reference)
525                })
526                .collect::<Vec<_>>()
527        });
528        let current = &mut solutions[index];
529        current.reference = reference;
530        for (channel, values) in current.values.iter_mut().enumerate() {
531            if let Some(offsets) = &offsets {
532                let offset = offsets[channel];
533                for estimate in values {
534                    estimate.gain[0] -= offset;
535                    estimate.gain[1] += offset.conj();
536                }
537            } else {
538                for estimate in values {
539                    estimate.valid = [false; 2];
540                }
541            }
542        }
543        if reference >= 0 {
544            references[1] = reference;
545            previous = Some(index);
546        }
547    }
548}

550fn choose_reference(
551    references: &[i32],
552    previous: Option<&LeakageSolution>,
553    available: impl Fn(i32) -> bool,
554) -> i32 {
555    references
556        .iter()
557        .copied()
558        .find(|&antenna| {
559            available(antenna)
560                && previous.is_none_or(|previous| usable_reference(&previous.values, antenna, 2))
561        })
562        .unwrap_or(-1)
563}
564
565fn usable_reference(values: &[ChannelEstimates], antenna: i32, receptors: usize) -> bool {
566    (0..receptors).all(|receptor| {
567        values.iter().any(|channel| {
568            channel
569                .get(antenna as usize)
570                .is_some_and(|sample| sample.valid[receptor])
571        })
572    })
573}
574
575fn previous_reference(
576    previous: Option<&LeakageSolution>,
577    channel: usize,
578    reference: i32,
579) -> Complex32 {
580    previous
581        .and_then(|previous| previous.values.get(channel))
582        .and_then(|channel| channel.get(reference as usize))
583        .map_or(Complex32::ZERO, |sample| sample.gain[0])
584}
