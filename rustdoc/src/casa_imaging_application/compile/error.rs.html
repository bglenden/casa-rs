4//! deploying the compiled run, each one a user can act on.

15/// Why a request could not be compiled against its MeasurementSet, or the
16/// compiled run could not be deployed.
17///
18/// Every variant names something the user can change: the request, the
19/// MeasurementSet, an input file or the resource policy. A failure no user
20/// action can cause is a bug and panics with the invariant it broke.
21#[derive(Debug, thiserror::Error)]
22pub enum PrepareError {
23    /// The MeasurementSet, or one of its subtables or columns, could not be
24    /// read.
25    #[error(transparent)]
26    MeasurementSet(#[from] casa_ms::MsError),
27    /// The selected rows are not in ascending MAIN order or name a data
28    /// description twice out of sequence.
29    #[error(transparent)]
30    SelectedRows(#[from] casa_imaging_model::SelectedRowSequenceError),
31    /// The selected observation could not be resolved against its
32    /// MeasurementSet.
33    #[error(transparent)]
34    Resolution(#[from] casa_ms::ObservationOwnerError),
35    /// The resolved observation could not be compiled.
36    #[error(transparent)]
37    Observation(#[from] casa_imaging_model::CompileObservationError),
38    /// A named, external or attached ephemeris could not be bound.
39    #[error(transparent)]
40    Ephemeris(#[from] casa_ms::SelectedObservationEphemerisError),
41    /// A direction could not be converted between frames.
42    #[error(transparent)]
43    Measures(#[from] casa_types::measures::MeasureError),
44    /// A mask box, mask plan or mask reprojection is invalid.
45    #[error(transparent)]
46    Mask(#[from] casa_imaging_reconstruction::MaskError),
47    /// The image's model-state bounds are invalid (no pixels, planes or
48    /// polarizations).
49    #[error(transparent)]
50    ModelBounds(#[from] casa_imaging_model::ModelContractError),
51    /// `normtype` has no direction-dependent image response.
52    #[error(transparent)]
53    ImageResponse(#[from] casa_imaging_reconstruction::ImageResponseError),
54    /// `pblimit` is outside the primary-beam validity contract.
55    #[error(transparent)]
56    PrimaryBeamValidity(#[from] casa_imaging_model::ProductValidityPolicyError),
57    /// The W-projection planes or W range are outside its contract.
58    #[error(transparent)]
59    WProjection(#[from] casa_imaging_model::WProjectionContractError),
60    /// The AW-projection parameters are outside its contract.
61    #[error(transparent)]
62    AwProjection(#[from] casa_imaging_model::AwProjectionContractError),
63    /// `fitspw` and `fitorder` are outside the continuum transform's contract.
64    #[error(transparent)]
65    ContinuumTransform(#[from] casa_imaging_model::ContinuumTransformContractError),
66    /// `psfcutoff` is not in (0, 1).
67    #[error("psfcutoff {psfcutoff} must be in (0, 1)")]
68    PsfCutoff {
69        /// The requested cutoff.
70        psfcutoff: f64,
71    },
72    /// Two image domains, the main image's or the outlier file's, write to
73    /// the same output.
74    #[error("image domains must have distinct output names; {} is named twice", path.display())]
75    DuplicateOutput {
76        /// The output both domains name.
77        path: PathBuf,
78    },
79    /// The selection names no MAIN row.
80    #[error("selection resolved to no rows")]
81    NoSelectedRows,
82    /// The `spw` or `ddid` selection names no data description.
83    #[error("selection resolved to no data descriptions")]
84    NoSelectedDataDescriptions,
85    /// A requested `DATA_DESC_ID` is negative or binds a negative
86    /// spectral-window or polarization id.
87    #[error("DATA_DESC_ID {ddid} does not bind a spectral window and polarization")]
88    InvalidDataDescription {
89        /// The data description.
90        ddid: i32,
91    },
92    /// A selected MAIN row holds a negative id.
93    #[error("selected MAIN rows hold a negative {column} ({value})")]
94    NegativeMainId {
95        /// The MAIN column.
96        column: &'static str,
97        /// The stored value.
98        value: i32,
99    },
100    /// `channel_start` and `channel_count` select no channel, or run past
101    /// the window's last channel.
102    #[error(
103        "channel_start {start} and channel_count {count} do not fit SPW {spw_id}'s {channels} channels"
104    )]
105    ChannelRange {
106        /// The spectral window.
107        spw_id: usize,
108        /// The first requested channel.
109        start: usize,
110        /// The requested channel count.
111        count: usize,
112        /// The window's channel count.
113        channels: usize,
114    },
115    /// The cube's output axis and the `spw` selector share no source channel.
116    #[error("cube axis and SPW selector have no common source channels")]
117    NoCommonCubeChannels,
118    /// A cube's first output channel does not fit the cube axis.
119    #[error("cube start channel {channel} exceeds the cube axis range")]
120    CubeStartChannel {
121        /// The requested channel.
122        channel: usize,
123    },
124    /// The cube's output frequency increment is zero or not finite.
125    #[error("cube output frequency increment must be finite and non-zero (got {increment_hz} Hz)")]
126    CubeIncrement {
127        /// The derived increment.
128        increment_hz: f64,
129    },
130    /// The cube asks for a Doppler convention native imaging has no law for.
131    #[error("cube Doppler convention {convention} is not supported")]
132    UnsupportedDoppler {
133        /// The convention.
134        convention: DopplerRef,
135    },
136    /// A selected spectral window's `MEAS_FREQ_REF` is not a frequency frame.
137    #[error("SPW {spw_id} has an unknown frequency frame code {code}")]
138    UnknownSourceFrame {
139        /// The spectral window.
140        spw_id: usize,
141        /// The stored `MEAS_FREQ_REF`.
142        code: i32,
143    },
144    /// The selected spectral windows are stored in different frames.
145    #[error("selected spectral windows use different source frequency frames")]
146    MixedSourceFrames,
147    /// Native imaging has no law for this frequency frame.
148    #[error("native imaging does not support the frequency frame {reference}")]
149    UnsupportedFrequencyFrame {
150        /// The frame.
151        reference: FrequencyRef,
152    },
153    /// Native imaging has no law for the MeasurementSet's time scale.
154    #[error("native imaging does not support the MeasurementSet epoch reference {reference}")]
155    UnsupportedEpochReference {
156        /// The epoch reference.
157        reference: EpochRef,
158    },
159    /// A stage that images one spectral window was given several.
160    #[error("{stage} requires exactly one selected spectral window; {selected} are selected")]
161    OneSpectralWindow {
162        /// The stage.
163        stage: &'static str,
164        /// The number selected.
165        selected: usize,
166    },
167    /// `fitspw` does not name exactly the selected spectral window.
168    #[error("continuum fit selector must name exactly the selected spectral window")]
169    ContinuumFitWindow,
170    /// `fitorder` is beyond the continuum fit's range.
171    #[error("continuum fit order {order} exceeds the supported range")]
172    ContinuumFitOrder {
173        /// The requested order.
174        order: usize,
175    },
176    /// The image centre's field is not among the selected fields.
177    #[error("phase-center FIELD_ID {field} is not part of selected fields {selected:?}")]
178    PhaseCentreField {
179        /// The requested field.
180        field: i32,
181        /// The selected fields.
182        selected: BTreeSet<i32>,
183    },
184    /// `phasecenter` is not `J2000 lon lat`.
185    #[error(
186        "phasecenter {text:?} must be 'J2000 lon lat', for example 'J2000 19:59:28.500 +40.44.01.50'"
187    )]
188    PhaseCentreSyntax {
189        /// The given text.
190        text: String,
191    },
192    /// A `phasecenter` angle is neither sexagesimal nor `rad` or `deg`.
193    #[error("unsupported phasecenter angle {angle:?}")]
194    PhaseCentreAngle {
195        /// The given angle.
196        angle: String,
197    },
198    /// A moving source's fields are only partly attached to ephemerides.
199    #[error("moving-source selection mixes FIELD rows with and without attached ephemerides")]
200    MixedEphemerisFields,
201    /// A source-frame cube needs an ephemeris phase centre.
202    #[error("source-frame cube imaging requires a moving phase centre")]
203    MovingPhaseCentre,
204    /// A source-frame cube needs the source's rest frequency.
205    #[error("source-frame cube imaging requires REST_FREQUENCY metadata")]
206    RestFrequencyMissing,
207    /// The selected SOURCE rows give different rest frequencies.
208    #[error("selected SOURCE rows disagree on REST_FREQUENCY metadata")]
209    RestFrequencyConflict,
210    /// The selected observations disagree on telescope or observer.
211    #[error(
212        "image observation metadata requires consistent telescope and observer labels for \
213         selected OBSERVATION_IDs {observation_ids:?}; found {labels:?}"
214    )]
215    ObservationLabels {
216        /// The selected observations.
217        observation_ids: BTreeSet<i32>,
218        /// The `(telescope, observer)` pairs found.
219        labels: BTreeSet<(String, String)>,
220    },
221    /// The MeasurementSet has neither `CORRECTED_DATA` nor `DATA`.
222    #[error("MS has neither CORRECTED_DATA nor DATA")]
223    NoVisibilityColumn,
224    /// A selected polarization holds a correlation code with no law.
225    #[error("unsupported correlation code {code}")]
226    UnsupportedCorrelation {
227        /// The stored `CORR_TYPE`.
228        code: i32,
229    },
230    /// The ANTENNA table is empty; a primary beam needs dish diameters.
231    #[error("a primary-beam response requires ANTENNA dish metadata")]
232    NoAntennaDishes,
233    /// ALMA's analytic beam needs one dish class.
234    #[error(
235        "ALMA primary-beam publication requires one homogeneous dish class; row {row} has \
236         diameter {diameter_m} m"
237    )]
238    MixedAlmaDishes {
239        /// The ANTENNA row.
240        row: usize,
241        /// Its dish diameter.
242        diameter_m: f64,
243    },
244    /// The OBSERVATION table names no telescope.
245    #[error("standard primary-beam publication requires OBSERVATION telescope metadata")]
246    NoTelescope,
247    /// No analytic primary beam is installed for these telescopes.
248    #[error(
249        "standard primary-beam publication has no installed analytic model for telescope set \
250         {telescopes:?}"
251    )]
252    NoAnalyticPrimaryBeam {
253        /// The OBSERVATION telescope names.
254        telescopes: Vec<String>,
255    },
256    /// The gridder's instrument response does not cover these telescopes.
257    #[error("requested instrument response is unsupported for observation metadata {telescopes:?}")]
258    UnsupportedInstrument {
259        /// The OBSERVATION telescope names.
260        telescopes: Vec<String>,
261    },
262    /// An antenna's dish is outside the instrument response.
263    #[error("the {model:?} response does not cover ANTENNA row {row}'s {diameter_m} m dish")]
264    DishOutsideResponse {
265        /// The response.
266        model: InstrumentModel,
267        /// The ANTENNA row.
268        row: usize,
269        /// Its dish diameter.
270        diameter_m: f64,
271    },
272    /// Native aperture generation models the EVLA only.
273    #[error("native aperture generation requires an EVLA observation")]
274    NativeAwTelescope,
275    /// Native EVLA generation models 25 m dishes only.
276    #[error("native EVLA generation requires the explicit homogeneous 25 m dish model")]
277    NativeAwDishes,
278    /// The EVLA surface file could not be read.
279    #[error("cannot read the EVLA surface {}: {source}", path.display())]
280    EvlaSurface {
281        /// The surface file.
282        path: PathBuf,
283        /// The file-system failure.
284        source: std::io::Error,
285    },
286    /// The EVLA surface file is not UTF-8 text.
287    #[error("the EVLA surface {} is not UTF-8 text", path.display())]
288    EvlaSurfaceText {
289        /// The surface file.
290        path: PathBuf,
291    },
292    /// The EVLA surface file is larger than its reference-data bound.
293    #[error("native EVLA surface exceeds the 1 MiB reference-data bound")]
294    NativeAwSurfaceSize,
295    /// The native AW request (the surface, grid or frequencies) is invalid.
296    #[error(transparent)]
297    NativeAwRequest(#[from] casa_imaging_model::NativeAwRequestError),
298    /// Native EVLA AW images Stokes I or one circular parallel hand.
299    #[error("native EVLA AW currently supports Stokes I or one circular parallel hand")]
300    NativeAwPolarization,
301    /// The FEED table's `RECEPTOR_ANGLE` is not `Float64`.
302    #[error("native EVLA feed receptor angles require Float64 metadata")]
303    FeedAngleType,
304    /// A FEED row has no finite first receptor angle.
305    #[error("native EVLA feed has no finite receptor-zero angle")]
306    FeedAngleNotFinite,
307    /// The applicable FEED rows give different receptor angles.
308    #[error("native EVLA receptor-zero angle is ambiguous for the selected epoch/SPW")]
309    FeedAngleAmbiguous,
310    /// No FEED row applies to antenna zero at the selected epoch and window.
311    #[error("native EVLA request has no applicable antenna-zero feed metadata")]
312    NoApplicableFeed,
313    /// Native AW reads the parallactic angle of an unflagged
314    /// cross-correlation row, and the selection has none.
315    #[error("native AW has no unflagged cross-correlation row")]
316    NoCrossCorrelationRow,
317    /// `cf_resident_mb` is larger than the address space.
318    #[error("cf_resident_mb {megabytes} exceeds addressable memory")]
319    CfResidentSize {
320        /// The requested megabytes.
321        megabytes: usize,
322    },
323    /// A reuse-only native AW cache lacks cells the request names.
324    #[error("native AW cache reuse found {present} of the {expected} cells the request names")]
325    AwCacheIncomplete {
326        /// Cells present.
327        present: usize,
328        /// Cells the request names.
329        expected: usize,
330    },
331    /// The native AW cache could not be cleared or its cells generated.
332    #[error(transparent)]
333    AwCatalog(#[from] casa_imaging_operator::AwCatalogError),
334    /// The image is too large to address its model samples.
335    #[error("the image's model samples exceed addressable memory")]
336    ImageSize,
337    /// The selected spectral envelope leaves no room in the row budget.
338    #[error("selected spectral envelope exhausts the row traversal budget")]
339    SpectralEnvelopeBudget,
340    /// A mask image could not be read.
341    #[error(transparent)]
342    MaskImage(#[from] casa_images::ImageError),
343    /// A reconstruction mask image is complex-valued.
344    #[error("reconstruction masks require a real CASA image")]
345    ComplexMaskImage,
346    /// A reconstruction mask image is not one direction plane.
347    #[error(
348        "reconstruction mask must contain one two-dimensional direction plane; shape {shape:?}"
349    )]
350    MaskImageShape {
351        /// The image shape.
352        shape: Vec<usize>,
353    },
354    /// A reconstruction mask image has no direction coordinate.
355    #[error("mask image has no direction coordinate")]
356    MaskImageDirection,
357    /// Native mask reprojection supports SIN only.
358    #[error("native mask reprojection currently requires SIN coordinates, not {projection:?}")]
359    MaskImageProjection {
360        /// The mask's projection.
361        projection: ProjectionType,
362    },
363    /// The mask's direction frame has no native law.
364    #[error("mask direction frame {frame} is not supported by native imaging")]
365    MaskImageFrame {
366        /// The mask's frame.
367        frame: DirectionRef,
368    },
369    /// The outlier file could not be read or describes no valid domain.
370    #[error("outlier file {}: {problem}", path.display())]
371    OutlierFile {
372        /// The outlier file.
373        path: PathBuf,
374        /// What is wrong with it.
375        problem: OutlierProblem,
376    },
377    /// An output directory could not be created.
378    #[error("cannot create output directory {}: {source}", path.display())]
379    OutputDirectory {
380        /// The directory.
381        path: PathBuf,
382        /// The file-system failure.
383        source: std::io::Error,
384    },

387/// What is wrong with an outlier file.
388#[derive(Debug, thiserror::Error)]
389pub enum OutlierProblem {
390    /// The file could not be read.
391    #[error("cannot read: {0}")]
392    Read(#[source] std::io::Error),
393    /// The file defines no image domain.
394    #[error("did not define any image domains")]
395    Empty,
396    /// A line is not one `parameter=value` pair.
397    #[error("line {line} must contain one parameter=value pair")]
398    NotAPair {
399        /// The 1-based line.
400        line: usize,
401    },
402    /// A line holds more than one `=`.
403    #[error("line {line} contains more than one '='")]
404    SeveralEquals {
405        /// The 1-based line.
406        line: usize,
407    },
408    /// A line repeats a field of its image.
409    #[error("line {line} repeats field {key:?}")]
410    RepeatedField {
411        /// The 1-based line.
412        line: usize,
413        /// The field.
414        key: String,
415    },
416    /// An image sets fields native imaging does not support.
417    #[error("outlier image {ordinal} contains unsupported field(s): {}", fields.join(", "))]
418    UnsupportedFields {
419        /// The image, counted from zero.
420        ordinal: usize,
421        /// The fields.
422        fields: Vec<String>,
423    },
424    /// An image lacks a required field.
425    #[error("outlier image {ordinal} is missing required {key}")]
426    MissingField {
427        /// The image, counted from zero.
428        ordinal: usize,
429        /// The field.
430        key: &'static str,
431    },
432    /// A field takes a value outside the installed multi-domain slice.
433    #[error("outlier field {key}={value:?} is outside the installed multi-domain slice")]
434    OutsideSlice {
435        /// The field.
436        key: &'static str,
437        /// Its value.
438        value: String,
439    },
440    /// `imsize` is not a positive square scalar or pair.
441    #[error("outlier imsize must be a positive square scalar or pair")]
442    ImageSize,
443    /// `cell` is not one positive arcsec value or an equal pair.
444    #[error("outlier cell must be one positive arcsec value or an equal pair")]
445    Cell,
446    /// A `cell` value is not a number.
447    #[error("invalid outlier cell {text:?}")]
448    CellValue {
449        /// The given value.
450        text: String,
451    },
452    /// `mask` is not `circle[[xpix,ypix],rpix]` in finite pixels.
453    #[error("outlier mask must use circle[[xpix,ypix],rpix] with finite pix values")]
454    MaskSyntax,
455    /// The mask circle's centre is outside its image or its radius negative.
456    #[error("outlier circle mask exceeds its image domain")]
457    MaskOutside,
