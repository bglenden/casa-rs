55impl Writer {
56    pub(super) fn new(ms: MeasurementSet, column: OptionalMainColumn) -> Self {
57        Self {
58            ms,
59            column,
60            descriptions: HashMap::new(),
61            polarizations: HashMap::new(),
62            states: HashMap::new(),
63            feeds: HashMap::new(),
64            reference_feed: None,
65            pointing: BTreeMap::new(),
66            weather: Vec::new(),
67            calibration: Vec::new(),
68        }

71    pub(super) fn write(
72        &mut self,
73        row: &Integration,
74        planes: BTreeMap<usize, Plane>,
75        channels: usize,
76    ) -> MsResult<()> {
77        let indices = planes.keys().copied().collect::<Vec<_>>();
78        let correlations = row.basis.correlations(&indices)?;
79        if planes.values().any(|plane| {
80            plane.data.len() != channels
81                || plane
82                    .flag
83                    .as_ref()
84                    .is_some_and(|flags| flags.len() != channels)
85        }) {
86            return Err(MsError::InvalidInput(format!(
87                "Imported spectral window {} expects {channels} channels",
88                row.window
89            )));
90        }
91        let codes = correlations
92            .iter()
93            .map(|&(code, _)| code)
94            .collect::<Vec<_>>();
95        let polarization = self.polarization_id(&codes)?;
96        let description = self.description_id(row.window, polarization)?;
97        self.record_feed(row)?;
98        self.record_pointing(row);
99        self.record_conditions(row, &correlations, &planes, channels);
100        let state = match self.states.get(&row.state) {
101            Some(&state) => state,
102            None => {
103                let state = state_row(&mut self.ms, row.state.0, row.state.1)?;
104                self.states.insert(row.state, state);
105                state
106            }
107        };
108        let data = main_row(
109            row,
110            description,
111            state,
112            &correlations,
113            &planes,
114            channels,
115            self.column,
116        );
117        self.ms.main_table_mut().add_row(data)?;
118        Ok(())
119    }
120
121    fn polarization_id(&mut self, codes: &[i32]) -> MsResult<i32> {
122        if let Some(&id) = self.polarizations.get(codes) {
123            return Ok(id);
124        }
125        let n = codes.len();
126        let pairs = codes
127            .iter()
128            .map(|&code| match code {
129                6 | 10 => [0, 1], // RL and XY
130                7 | 11 => [1, 0], // LR and YX
131                8 | 12 => [1, 1], // LL and YY
132                _ => [0, 0],
133            })
134            .collect::<Vec<_>>();
135        let receptors = (0..2)
136            .flat_map(|receptor| pairs.iter().map(move |pair| pair[receptor]))
137            .collect::<Vec<_>>();
138        let id = insert(
139            &mut self.ms,
140            SubtableId::Polarization,
141            &[
142                ("CORR_TYPE", i32_array(codes, vec![n])),
143                ("CORR_PRODUCT", i32_array(&receptors, vec![2, n])),
144                ("NUM_CORR", i(n as i32)),
145            ],
146        )?;
147        self.polarizations.insert(codes.to_vec(), id);
148        Ok(id)
149    }
150
151    fn description_id(&mut self, window: i32, polarization: i32) -> MsResult<i32> {
152        if let Some(&id) = self.descriptions.get(&(window, polarization)) {
153            return Ok(id);
154        }
155        let id = insert(
156            &mut self.ms,
157            SubtableId::DataDescription,
158            &[
159                ("POLARIZATION_ID", i(polarization)),
160                ("SPECTRAL_WINDOW_ID", i(window)),
161            ],
162        )?;
163        self.descriptions.insert((window, polarization), id);
164        Ok(id)
165    }
166
167    fn record_feed(&mut self, row: &Integration) -> MsResult<()> {
168        let basis = if row.basis == Basis::Circular {
169            Basis::Circular
170        } else {
171            Basis::Linear
172        };
173        let key = (row.antenna, row.feed, row.window, basis);
174        if let std::collections::hash_map::Entry::Vacant(entry) = self.feeds.entry(key) {
175            let id = insert(
176                &mut self.ms,
177                SubtableId::Feed,
178                &[
179                    ("ANTENNA_ID", i(row.antenna)),
180                    ("FEED_ID", i(row.feed)),
181                    ("NUM_RECEPTORS", i(2)),
182                    ("SPECTRAL_WINDOW_ID", i(row.window)),
183                    (
184                        "BEAM_OFFSET",
185                        crate::subtable_rows::f64_array(&[0.; 4], vec![2, 2]),
186                    ),
187                    ("POLARIZATION_TYPE", string_array(&basis.feeds(), vec![2])),
188                    (
189                        "POL_RESPONSE",
190                        complex_array(&[Complex32::default(); 4], vec![2, 2]),
191                    ),
192                    (
193                        "RECEPTOR_ANGLE",
194                        crate::subtable_rows::f64_array(&[0.; 2], vec![2]),
195                    ),
196                ],
197            )?;
198            entry.insert(id);
199        }
200        self.reference_feed.get_or_insert(row.feed);
201        Ok(())
202    }
203
204    fn record_pointing(&mut self, row: &Integration) {
205        if self.reference_feed == Some(row.feed) {
206            self.pointing
207                .entry((row.antenna, row.time.to_bits()))
208                .or_insert((row.direction, row.interval));
209        }
210    }
211
212    fn record_conditions(
213        &mut self,
214        row: &Integration,
215        correlations: &[(i32, usize)],
216        planes: &BTreeMap<usize, Plane>,
217        channels: usize,
218    ) {
219        let time = TimeSpan {
220            time: row.time,
221            interval: row.interval,
222        };
223        if let Some(weather) = row.weather {
224            update_unique(&mut self.weather, (row.antenna, weather), time);
225        }
226        if !row.basis.has_parallel_hands() {
227            return;
228        }
229        let hand_indices = correlations
230            .iter()
231            .map(|&(_, index)| index)
232            .filter(|&index| index < 2);
233        let hands = hand_indices
234            .map(|index| &planes[&index])
235            .collect::<Vec<_>>();
236        let system = temperatures(&hands, channels, |plane| &plane.system_temperature);
237        if let Some(system) = system {
238            let mut calibration = Calibration {
239                antenna: row.antenna,
240                feed: row.feed,
241                window: row.window,
242                system,
243                calibration: temperatures(&hands, channels, |plane| &plane.calibration_temperature),
244            };
245            if correlations.len() != hands.len() {
246                calibration.system.reduce_to_scalars();
247                if let Some(temperature) = &mut calibration.calibration {
248                    temperature.reduce_to_scalars();
249                }
250            }
251            update_unique(&mut self.calibration, calibration, time);
252        }
253    }
254
255    pub(super) fn finish(mut self) -> MsResult<MeasurementSet> {
256        let pointing = subtable_mut(&mut self.ms, SubtableId::Pointing)?;
257        for ((antenna, time), (direction, interval)) in self.pointing {
258            let (degree, direction) = direction_cell(direction);
259            let record = row_from_defs(
260                schema::pointing::REQUIRED_COLUMNS,
261                &[
262                    ("ANTENNA_ID", i(antenna)),
263                    ("TIME", f(f64::from_bits(time))),
264                    ("INTERVAL", f(interval)),
265                    ("NUM_POLY", i(degree)),
266                    ("DIRECTION", direction),
267                ],
268            );
269            pointing.add_row(without(record, &["TARGET"]))?;
270        }
271        let weather = subtable_mut(&mut self.ms, SubtableId::Weather)?;
272        for ((antenna, sample), span) in self.weather {
273            let mut row = row_from_defs(
274                schema::weather::REQUIRED_COLUMNS,
275                &[
276                    ("ANTENNA_ID", i(antenna)),
277                    ("TIME", f(span.time)),
278                    ("INTERVAL", f(span.interval)),
279                ],
280            );
281            for (name, value) in [
282                "TEMPERATURE",
283                "PRESSURE",
284                "REL_HUMIDITY",
285                "WIND_SPEED",
286                "WIND_DIRECTION",
287            ]
288            .into_iter()
289            .zip(sample.0)
290            {
291                row.upsert(name, Value::Scalar(ScalarValue::Float32(value)));
292            }
293            weather.add_row(row)?;
294        }
295        let syscal = subtable_mut(&mut self.ms, SubtableId::SysCal)?;
296        for (sample, span) in self.calibration {
297            let mut row = row_from_defs(
298                schema::syscal::REQUIRED_COLUMNS,
299                &[
300                    ("ANTENNA_ID", i(sample.antenna)),
301                    ("TIME", f(span.time)),
302                    ("INTERVAL", f(span.interval)),
303                    ("FEED_ID", i(sample.feed)),
304                    ("SPECTRAL_WINDOW_ID", i(sample.window)),
305                ],
306            );
307            sample.system.add_cell(&mut row, "TSYS", "TSYS_SPECTRUM");
308            if let Some(temperature) = sample.calibration {
309                temperature.add_cell(&mut row, "TCAL", "TCAL_SPECTRUM");
310            }
311            syscal.add_row(row)?;
312        }
313        Ok(self.ms)
314    }
315}

317fn main_row(
318    row: &Integration,
319    description: i32,
320    state: i32,
321    correlations: &[(i32, usize)],
322    planes: &BTreeMap<usize, Plane>,
323    channels: usize,
324    column: OptionalMainColumn,
325) -> RecordValue {
326    let ordered = correlations
327        .iter()
328        .map(|&(_, plane)| &planes[&plane])
329        .collect::<Vec<_>>();
330    let count = ordered.len();
331    let full = count == 4 && row.basis.has_parallel_hands();
332    let data = visibility_data(&ordered, channels, full, column);
333    let flags = if ordered.iter().all(|plane| plane.flag.is_none()) {
334        Array2::<bool>::default((count, channels).f()).into_dyn()
335    } else {
336        let flagged =
337            |plane: &Plane, channel| plane.flag.as_ref().is_some_and(|flags| flags[channel]);
338        channel_major(&ordered, channels, |plane, channel| {
339            if full && plane.index >= 2 {
340                flagged(ordered[1], channel) || flagged(ordered[2], channel)
341            } else {
342                flagged(plane, channel)
343            }
344        })
345    };
346    let noise = f32_array(&vec![1.; count], vec![count]);
347    let mut record = row_from_defs(
348        schema::main_table::REQUIRED_COLUMNS,
349        &[
350            ("TIME", f(row.time)),
351            ("TIME_CENTROID", f(row.time)),
352            ("INTERVAL", f(row.interval)),
353            ("EXPOSURE", f(row.interval)),
354            ("SCAN_NUMBER", i(row.scan)),
355            ("ANTENNA1", i(row.antenna)),
356            ("ANTENNA2", i(row.antenna)),
357            ("FIELD_ID", i(row.field)),
358            ("FEED1", i(row.feed)),
359            ("FEED2", i(row.feed)),
360            ("DATA_DESC_ID", i(description)),
361            ("STATE_ID", i(state)),
362            ("FLAG_ROW", b(planes.values().any(|plane| plane.flag_row))),
363            ("FLAG_CATEGORY", bool_array(&[], vec![0, 0, 0])),
364            ("WEIGHT", noise.clone()),
365            ("SIGMA", noise),
366        ],
367    );
368    record.upsert("FLAG", Value::Array(ArrayValue::Bool(flags)));
369    record.upsert(column.name(), data);
370    record
371}

373fn visibility_data(
374    planes: &[&Plane],
375    channels: usize,
376    full: bool,
377    column: OptionalMainColumn,
378) -> Value {
379    if column == OptionalMainColumn::FloatData {
380        let data = channel_major(planes, channels, |plane, channel| {
381            if full && plane.index == 3 {
382                planes[1].data[channel]
383            } else {
384                plane.data[channel]
385            }
386        });
387        return Value::Array(ArrayValue::Float32(data));
388    }
389    let data = channel_major(planes, channels, |plane, channel| {
390        match (full, plane.index) {
391            (true, 2) => Complex32::new(planes[1].data[channel], planes[2].data[channel]),
392            (true, 3) => Complex32::new(planes[1].data[channel], -planes[2].data[channel]),
393            _ => Complex32::new(plane.data[channel], 0.),
394        }
395    });
396    Value::Array(ArrayValue::Complex32(data))
397}

399/// Construct contiguous casacore-order cells directly from the input planes.
400fn channel_major<T>(
401    planes: &[&Plane],
402    channels: usize,
403    sample: impl Fn(&Plane, usize) -> T,
404) -> ArrayD<T> {
405    let mut values = Vec::with_capacity(planes.len() * channels);
406    for channel in 0..channels {
407        values.extend(planes.iter().map(|plane| sample(plane, channel)));
408    }
409    Array2::from_shape_vec((planes.len(), channels).f(), values)
410        .expect("one value per correlation and channel")
411        .into_dyn()
412}
413
414fn update_unique<T: PartialEq>(records: &mut Vec<(T, TimeSpan)>, value: T, span: TimeSpan) {
415    if let Some((_, previous)) = records.iter_mut().find(|(existing, _)| *existing == value) {
416        let lower = previous.time - previous.interval / 2.;
417        let upper = previous.time + previous.interval / 2.;
418        let start = lower.min(span.time - span.interval / 2.);
419        let end = upper.max(span.time + span.interval / 2.);
420        if start != lower || end != upper {
421            *previous = TimeSpan {
422                time: (start + end) / 2.,
423                interval: end - start,
424            };
425        }
426    } else {
427        records.push((value, span));
428    }
429}

431fn temperatures(
432    hands: &[&Plane],
433    channels: usize,
434    extract: impl Fn(&Plane) -> &[f32],
435) -> Option<Temperatures> {
436    let inputs = hands.iter().map(|plane| extract(plane)).collect::<Vec<_>>();
437    if channels > 1
438        && !inputs
439            .iter()
440            .any(|values| values.iter().any(|&value| value > 0.))
441    {
442        return None;
443    }
444    let spectral = channels == 1
445        || inputs
446            .iter()
447            .any(|values| values.len() == channels && values[1..].iter().any(|&value| value > 0.));
448    let width = if spectral { channels } else { 1 };
449    let values = inputs
450        .into_iter()
451        .flat_map(|input| {
452            (0..width).map(move |channel| {
453                if channel == 0 || input.len() == channels {
454                    input.get(channel).copied().unwrap_or(-1.)
455                } else {
456                    -1.
457                }
458            })
459        })
460        .collect::<Vec<_>>();
461    if values.iter().all(|&value| value == 0.) || values.iter().all(|&value| value == 1.) {
462        return None;
463    }
464    Some(Temperatures {
465        spectral,
466        width,
467        count: hands.len(),
468        values,
469    })
470}

472impl Temperatures {
473    fn add_cell(self, row: &mut RecordValue, scalar: &str, array: &str) {
474        if self.spectral {
475            row.upsert(array, f32_array(&self.values, vec![self.count, self.width]));
476        } else {
477            row.upsert(scalar, f32_array(&self.values, vec![self.count]));
478        }
479    }
480
481    fn reduce_to_scalars(&mut self) {
482        self.values = self.values.iter().step_by(self.width).copied().collect();
483        self.spectral = false;
484        self.width = 1;
485    }
