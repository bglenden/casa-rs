2//! Python binding for the legacy ASDM XML table conversion.
3
4use std::path::Path;
5
6use casa_ms::read_asdm_table;
7use casa_tables::{DataManagerKind, TableOptions};
8use pyo3::prelude::*;
9
10use super::copy::{check_target, replace_with, used_by_another_process};
11use super::{Fail, PyTableTool, text};
12use crate::convert::value::CasaValue;
13
14#[pymethods]
15impl PyTableTool {
16    /// Convert an ASDM XML file to a casacore table without changing this tool.
17    #[allow(non_snake_case)]
18    #[pyo3(signature = (tablename, xmlfile), text_signature = "($self, tablename, xmlfile)")]
19    fn fromASDM(
20        &self,
21        py: Python<'_>,
22        tablename: &Bound<'_, PyAny>,
23        xmlfile: &Bound<'_, PyAny>,
24    ) -> PyResult<Py<PyAny>> {
25        let name = text(tablename, "tablename")?;
26        let xml = text(xmlfile, "xmlfile")?;
27        self.call(py, |state, _| {
28            let table = read_asdm_table(Path::new(&xml))
29                .map_err(|error| Fail::Runtime(error.to_string()))?;
30            let target = std::path::absolute(Path::new(&name))
31                .map_err(|error| Fail::Runtime(error.to_string()))?;
32            check_target(&target)?;
33            state.refuse_open_target(&target)?;
34            if used_by_another_process(&target) {
35                return Err(Fail::Runtime(format!(
36                    "Table {} is open in another process and was not replaced",
37                    target.display()
38                )));
39            }
40            replace_with(&target, |stage| {
41                table.save(
42                    TableOptions::new(stage)
43                        .with_data_manager(DataManagerKind::StandardStMan)
44                        .with_data_manager_group("StandardStMan")
45                        .with_casacore_default_ssm(true),
46                )?;
47                Ok(())
48            })?;
49            Ok(CasaValue::Bool(true))
50        })
51    }
52}
