56impl CasaImageProductSink {
57    /// Bind an output prefix and its complete CASA coordinate system.
58    #[must_use]
59    pub fn new(base: PathBuf, coordinates: CoordinateSystem) -> Self {
60        Self::for_domains([CasaImageDomainOutput::new(
61            ImageDomainRole::Main,
62            base,
63            coordinates,
64        )])
65        .expect("one main image-domain output is valid")
66    }
67
68    /// Bind every compiled image-domain role to one unique output root and WCS.
69    pub fn for_domains(
70        domains: impl IntoIterator<Item = CasaImageDomainOutput>,
71    ) -> Result<Self, std::io::Error> {
72        let mut outputs = BTreeMap::new();
73        let mut roots = std::collections::BTreeSet::new();
74        for output in domains {
75            if !roots.insert(output.base.clone())
76                || outputs.insert(output.role.clone(), output).is_some()
77            {
78                return Err(std::io::Error::other(
79                    "CASA image-domain output roles and roots must be unique",
80                ));
81            }
82        }
83        if outputs.is_empty() || !outputs.contains_key(&ImageDomainRole::Main) {
84            return Err(std::io::Error::other(
85                "CASA image-domain outputs require one main domain",
86            ));
87        }
88        Ok(Self {
89            domains: outputs,
90            staged: Mutex::new(Vec::new()),
91        })
92    }
93}

95impl SerialProductPublicationSink for CasaImageProductSink {
96    type Error = std::io::Error;
97
98    fn residency(
99        &self,
100        planned: &PlannedContinuumGeneration,
101        demand: &ContinuumGenerationDemand,
102    ) -> Result<ProductSinkResidency, Self::Error> {
103        const IMAGE_ADAPTER_ENVELOPE_BYTES: u64 = 4_096;
104        const STAGED_MEMBER_RECORD_BYTES: u64 = 512;
105        let coordinate_bytes = self.domains.values().try_fold(0_u64, |maximum, output| {
106            let bytes = output
107                .coordinates
108                .to_record()
109                .retained_heap_bytes()
110                .and_then(|bytes| bytes.checked_mul(4))
111                .and_then(|bytes| u64::try_from(bytes).ok())
112                .ok_or_else(|| std::io::Error::other("coordinate metadata residency overflow"))?;
113            Ok::<_, std::io::Error>(maximum.max(bytes))
114        })?;
115        let beam_bytes = planned
116            .members()
117            .iter()
118            .try_fold(0_u64, |maximum, member| {
119                let count = if member.storage().attach_beam() {
120                    member
121                        .axes()
122                        .spectral()
123                        .output_channels()
124                        .checked_mul(member.axes().polarization().len())
125                        .ok_or_else(|| std::io::Error::other("beam count overflow"))?
126                } else {
127                    0
128                };
129                Ok::<_, std::io::Error>(maximum.max(beam_metadata_residency_bytes(count)?))
130            })?;
131        let registry_bytes = planned.members().iter().try_fold(0_u64, |total, member| {
132            let output = self.domains.get(member.axes().domain()).ok_or_else(|| {
133                std::io::Error::other("product domain has no CASA output binding")
134            })?;
135            let base_bytes = u64::try_from(output.base.to_string_lossy().len())
136                .map_err(|_| std::io::Error::other("product path length exceeds u64"))?;
137            let name_bytes = u64::try_from(member.name().len())
138                .map_err(|_| std::io::Error::other("product name length exceeds u64"))?;
139            let path_bytes = base_bytes
140                .checked_add(name_bytes)
141                .and_then(|bytes| bytes.checked_mul(2))
142                .ok_or_else(|| std::io::Error::other("product path residency overflow"))?;
143            total
144                .checked_add(path_bytes)
145                .and_then(|bytes| bytes.checked_add(STAGED_MEMBER_RECORD_BYTES))
146                .ok_or_else(|| std::io::Error::other("product registry residency overflow"))
147        })?;
148        let writer_bytes = demand
149            .maximum_window_payload_bytes()
150            .checked_mul(4)
151            .and_then(|bytes| {
152                bytes.checked_add(demand.maximum_window_validity_bytes().checked_mul(2)?)
153            })
154            .and_then(|bytes| bytes.checked_add(IMAGE_ADAPTER_ENVELOPE_BYTES))
155            .and_then(|bytes| bytes.checked_add(coordinate_bytes))
156            .and_then(|bytes| bytes.checked_add(beam_bytes))
157            .ok_or_else(|| std::io::Error::other("product staging residency overflow"))?;
158        Ok(ProductSinkResidency {
159            writer_bytes,
160            retained_bytes: registry_bytes,
161        })
162    }
163
164    fn publish(&self) -> Result<(), Self::Error> {
165        let staged =
166            std::mem::take(&mut *self.staged.lock().map_err(|_| {
167                std::io::Error::other("CASA product staging registry lock poisoned")
168            })?);
169        for product in staged {
170            promote_atomically(&product.staging, &product.target)?;
171        }
172        Ok(())
173    }
174}

189impl CasaImageProductSink {
190    fn begin_image(
191        &self,
192        member: &PlannedMember,
193        layout: ProductWindowLayout,
194        beams: &[Option<RestoringBeam>],
195    ) -> Result<CasaProductWriter<'_>, std::io::Error> {
196        let started = std::env::var_os("CASA_RS_PROFILE_PRODUCTS")
197            .is_some()
198            .then(Instant::now);
199        let output = self
200            .domains
201            .get(member.axes().domain())
202            .ok_or_else(|| std::io::Error::other("product domain has no CASA output binding"))?;
203        let target = PathBuf::from(format!("{}{}", output.base.display(), member.name()));
204        let parent = target.parent().unwrap_or_else(|| Path::new("."));
205        fs::create_dir_all(parent)?;
206        let target_name = target
207            .file_name()
208            .and_then(|name| name.to_str())
209            .ok_or_else(|| std::io::Error::other("CASA product target is not UTF-8"))?;
210        let directory = tempfile::Builder::new()
211            .prefix(&format!(".{target_name}.casa-rs-stage-"))
212            .tempdir_in(parent)?;
213        let staging = directory.path().join("image");
214        let mut tile = layout.shape();
215        tile[layout.spectral_axis()] = 1;
216        let mut image = PagedImage::<f32>::create_with_tile_shape_and_cache(
217            layout.shape().to_vec(),
218            tile.to_vec(),
219            output.coordinates.clone(),
220            &staging,
221            layout
222                .maximum_values()
223                .checked_mul(4)
224                .ok_or_else(|| std::io::Error::other("image cache overflow"))?,
225        )
226        .map_err(|error| std::io::Error::other(error.to_string()))?;
227        let storage = member.storage();
228        let explicit_mask = matches!(storage.pixel_mask(), ProductPixelMask::Explicit(_));
229        image
230            .set_units(storage.unit().map_or("", unit_label))
231            .map_err(|error| std::io::Error::other(error.to_string()))?;
232        let role = role_label(member.role());
233        let beam_set = if storage.attach_beam() {
234            persisted_beam_set(beams)
235        } else {
236            ImageBeamSet::default()
237        };
238        image
239            .set_image_info(&ImageInfo {
240                beam_set,
241                image_type: match role {
242                    "psf" => ImageType::Beam,
243                    "sumwt" => ImageType::Undefined,
244                    _ => ImageType::Intensity,
245                },
246                object_name: role.to_string(),
247            })
248            .map_err(|error| std::io::Error::other(error.to_string()))?;
249        image
250            .set_misc_info(RecordValue::new(vec![RecordField::new(
251                "casars_imager_role",
252                Value::Scalar(ScalarValue::String(role.to_string())),
253            )]))
254            .map_err(|error| std::io::Error::other(error.to_string()))?;
255        Ok(CasaProductWriter {
256            image,
257            explicit_mask,
258            sink: self,
259            profile: started.is_some(),
260            begin_nanos: started.map_or(0, |start| start.elapsed().as_nanos()),
261            window_nanos: 0,
262            staged: StagedProduct {
263                _directory: directory,
264                staging,
265                target,
266            },
267        })
268    }
269}

272    image: PagedImage<f32>,
273    explicit_mask: bool,
274    sink: &'a CasaImageProductSink,
275    staged: StagedProduct,
276    profile: bool,
277    begin_nanos: u128,
278    window_nanos: u128,
279}

281impl ProductWriter for CasaProductWriter<'_> {
282    fn write(&mut self, window: ProductWindow) -> Result<(), ProductsError> {
283        let started = self.profile.then(Instant::now);
284        let (start, shape, payload, validity) = window.into_parts();
285        let data = ArrayD::from_shape_vec(IxDyn(&shape), payload)
286            .map_err(|error| ProductsError::Storage(error.to_string()))?;
287        self.image
288            .put_slice_view(data.view(), &start)
289            .map_err(|error| ProductsError::Storage(error.to_string()))?;
290        if self.explicit_mask {
291            let mask = ArrayD::from_shape_vec(IxDyn(&shape), validity)
292                .map_err(|error| ProductsError::Storage(error.to_string()))?;
293            self.image
294                .put_mask_slice("mask0", &mask, &start)
295                .map_err(|error| ProductsError::Storage(error.to_string()))?;
296        }
297        if let Some(started) = started {
298            self.window_nanos += started.elapsed().as_nanos();
299        }
300        Ok(())
301    }
302    fn finish(mut self: Box<Self>) -> Result<(), ProductsError> {
303        let started = self.profile.then(Instant::now);
304        if self.explicit_mask {
305            self.image
306                .set_default_mask("mask0")
307                .map_err(|error| ProductsError::Storage(error.to_string()))?;
308        }
309        self.image.prepare_relocation(&self.staged.target);
310        self.image
311            .save()
312            .map_err(|error| ProductsError::Storage(error.to_string()))?;
313        if std::env::var_os("CASA_RS_TRACE_IMAGING_STAGE_TIMING").is_some()
314            && let Some(stats) = self.image.tiled_io_stats()
315        {
316            eprintln!(
317                "imaging_product_pixels target={} pack_nanos={} swap_nanos={} write_nanos={} direct_bytes={} direct_calls={}",
318                self.staged.target.display(),
319                stats.direct_tile_pack_ns,
320                stats.direct_tile_swap_ns,
321                stats.direct_tile_write_ns,
322                stats.direct_tile_write_bytes,
323                stats.direct_tile_write_calls,
324            );
325        }
326        let Self {
327            image,
328            sink,
329            staged,
330            begin_nanos,
331            window_nanos,
332            ..
333        } = *self;
334        drop(image);
335        if let Some(started) = started {
336            eprintln!(
337                "imaging_product_writer target={} begin_nanos={begin_nanos} window_nanos={window_nanos} finish_nanos={}",
338                staged.target.display(),
339                started.elapsed().as_nanos(),
340            );
341        }
342        sink.staged
343            .lock()
344            .map_err(|_| {
345                ProductsError::Storage("CASA product staging registry lock poisoned".into())
346            })?
347            .push(staged);
348        Ok(())
349    }
350}

352fn beam_metadata_residency_bytes(count: usize) -> Result<u64, std::io::Error> {
353    if count == 0 {
354        return Ok(0);
355    }
356    let overflow = || std::io::Error::other("beam metadata residency overflow");
357    let beam_record = GaussianBeam::new(1.0, 1.0, 0.0)
358        .to_record()
359        .retained_heap_bytes()
360        .ok_or_else(overflow)?;
361    let key_bytes = count.ilog10() as usize + 2;
362    // The growable outer keyword record has at most twice its live field count.
363    // Save overlaps owned keywords, the storage snapshot, table.dat keywords,
364    // and the encoded control buffer; the record heap bounds its encoded form.
365    let record = count
366        .checked_add(2)
367        .and_then(|fields| fields.checked_mul(2 * std::mem::size_of::<RecordField>()))
368        .and_then(|bytes| {
369            bytes.checked_add(count.checked_mul(beam_record.checked_add(key_bytes)?)?)
370        })
371        .and_then(|bytes| bytes.checked_mul(4))
372        .ok_or_else(overflow)?;
373    let arrays = count
374        .checked_mul(
375            std::mem::size_of::<RestoringBeam>()
376                + std::mem::size_of::<Vec<GaussianBeam>>()
377                + std::mem::size_of::<GaussianBeam>(),
378        )
379        .ok_or_else(overflow)?;
380    u64::try_from(record.checked_add(arrays).ok_or_else(overflow)?).map_err(|_| overflow())
381}

383fn persisted_beam_set(beams: &[Option<RestoringBeam>]) -> ImageBeamSet {
384    if beams.is_empty() {
385        return ImageBeamSet::default();
386    }
387    let valid = beams.iter().flatten().copied().collect::<Vec<_>>();
388    if let Some(first) = valid.first().copied()
389        && valid.len() == beams.len()
390        && valid.iter().all(|beam| *beam == first)
391    {
392        return ImageBeamSet::new(gaussian_beam(first));
393    }
394    let filler = valid
395        .iter()
396        .copied()
397        .max_by(|left, right| {
398            beam_area(*left)
399                .partial_cmp(&beam_area(*right))
400                .expect("validated beam areas are finite")
401        })
402        .map(gaussian_beam)
403        .unwrap_or_else(|| {
404            let one_microarcsecond_rad = std::f64::consts::PI / (180.0 * 3_600_000_000.0);
405            GaussianBeam::new(one_microarcsecond_rad, one_microarcsecond_rad, 0.0)
406        });
407    ImageBeamSet::from_grid(
408        beams
409            .iter()
410            .map(|beam| vec![beam.map_or(filler, gaussian_beam)])
411            .collect(),
412    )
413}

457fn promote_atomically(staging: &Path, target: &Path) -> std::io::Result<()> {
458    let parent = target.parent().ok_or_else(|| {
459        std::io::Error::new(
460            std::io::ErrorKind::InvalidInput,
461            "product target has no parent directory",
462        )
463    })?;
464    if staging.parent().and_then(Path::parent) != Some(parent) {
465        return Err(std::io::Error::new(
466            std::io::ErrorKind::InvalidInput,
467            "product staging must be private to the output directory",
468        ));
469    }
470    if target.exists() {
471        exchange_directories(staging, target)?;
472    } else {
473        fs::rename(staging, target)?;
474    }
475    File::open(parent).and_then(|directory| directory.sync_all())?;
476    File::open(
477        staging
478            .parent()
479            .expect("validated private output directory"),
480    )
481    .and_then(|directory| directory.sync_all())?;
482    Ok(())
483}

485#[cfg(target_os = "macos")]
486fn exchange_directories(staging: &Path, target: &Path) -> std::io::Result<()> {
487    use std::os::unix::ffi::OsStrExt;
488    let staging = CString::new(staging.as_os_str().as_bytes())
489        .map_err(|_| std::io::Error::other("staging path contains NUL"))?;
490    let target = CString::new(target.as_os_str().as_bytes())
491        .map_err(|_| std::io::Error::other("target path contains NUL"))?;
492    // SAFETY: both C strings remain valid for the duration of this atomic call.
493    let status = unsafe { libc::renamex_np(staging.as_ptr(), target.as_ptr(), libc::RENAME_SWAP) };
494    (status == 0)
495        .then_some(())
496        .ok_or_else(std::io::Error::last_os_error)
497}

499#[cfg(target_os = "linux")]
500fn exchange_directories(staging: &Path, target: &Path) -> std::io::Result<()> {
501    use std::os::unix::ffi::OsStrExt;
502    let staging = CString::new(staging.as_os_str().as_bytes())
503        .map_err(|_| std::io::Error::other("staging path contains NUL"))?;
504    let target = CString::new(target.as_os_str().as_bytes())
505        .map_err(|_| std::io::Error::other("target path contains NUL"))?;
506    // SAFETY: both C strings remain valid for the duration of this atomic call.
507    let status = unsafe {
508        libc::syscall(
509            libc::SYS_renameat2,
510            libc::AT_FDCWD,
511            staging.as_ptr(),
512            libc::AT_FDCWD,
513            target.as_ptr(),
514            libc::RENAME_EXCHANGE,
515        )
516    };
517    (status == 0)
518        .then_some(())
519        .ok_or_else(std::io::Error::last_os_error)
520}

522#[cfg(not(any(target_os = "macos", target_os = "linux")))]
523fn exchange_directories(_: &Path, _: &Path) -> std::io::Result<()> {
524    Err(std::io::Error::new(
525        std::io::ErrorKind::Unsupported,
526        "atomic non-empty directory exchange is unavailable",
527    ))
528}

608        let root = tempfile::tempdir().unwrap();
609        let target = root.path().join("image");
610        std::fs::create_dir(&target).unwrap();
611        std::fs::write(target.join("pixels"), b"old").unwrap();
612        let private = tempfile::tempdir_in(root.path()).unwrap();
613        let staged = private.path().join("image");
614        std::fs::create_dir(&staged).unwrap();
615        std::fs::write(staged.join("pixels"), b"new").unwrap();
616        super::promote_atomically(&staged, &target).unwrap();
617        assert_eq!(std::fs::read(target.join("pixels")).unwrap(), b"new");
618        assert_eq!(std::fs::read(staged.join("pixels")).unwrap(), b"old");
619        let missing = private.path().join("missing");
620        assert!(super::promote_atomically(&missing, &target).is_err());
621        assert_eq!(std::fs::read(target.join("pixels")).unwrap(), b"new");
622    }
