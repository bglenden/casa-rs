20/// Pixel-box and degenerate-axis projection describing an image region copy target.
21pub struct Selection {
22    /// Start index along each input pixel axis.
23    pub start: Vec<usize>,
24    /// Number of selected pixels along each input pixel axis.
25    pub shape: Vec<usize>,
26    /// Stride along each input pixel axis.
27    pub stride: Vec<usize>,
28    retained: Vec<usize>,
29}
30
31impl Selection {
32    /// Initializes a copy selection covering all axes of `section`.
33    pub fn new(section: &ImageRegionSection) -> Self {
34        let rank = section.count.len();
35        Self {
36            start: section.start.clone(),
37            shape: section.count.clone(),
38            stride: section.stride.clone(),
39            retained: (0..rank).collect(),
40        }
41    }
42
43    /// Drops singleton axes of length `1`, matching CASA's `dropdeg=True` error if none remain.
44    pub fn drop_degenerate(&mut self) -> ImageToolResult<()> {
45        self.retained.retain(|&axis| self.shape[axis] > 1);
46        if self.retained.is_empty() {
47            return Err(ImageToolError::new(
48                "LatticeStepper::setCursorShape: cursorShape has no axes or more axes than lattice",
49            ));
50        }
51        Ok(())
52    }
53
54    /// Extracts entries from `values` corresponding to retained output axes.
55    pub fn projected(&self, values: &[usize]) -> Vec<usize> {
56        self.retained.iter().map(|&axis| values[axis]).collect()
57    }
58
59    /// Squeezes dropped degenerate axes out of `array`.
60    pub fn squeeze<T>(&self, mut array: ArrayD<T>) -> ArrayD<T> {
61        for axis in (0..self.shape.len()).rev() {
62            if !self.retained.contains(&axis) {
63                array = array.index_axis_move(Axis(axis), 0);
64            }
65        }
66        array
67    }

69    /// Updates a casacore coordinate `record` to match the cropped, strided, and projected axes.
70    pub fn coordinates(
71        &self,
72        mut record: RecordValue,
73        original: &CoordinateSystem,
74        image_shape: &[usize],
75    ) -> ImageToolResult<RecordValue> {
76        for (axis, &source_length) in image_shape.iter().enumerate() {
77            let (number, within_coord) = original
78                .pixel_axis_owner(axis)
79                .ok_or_else(|| ImageToolError::new("Image section pixel axis has no coordinate"))?;
80            let coord = original.coordinate(number);
81            let kind = coord.coordinate_type();
82            let shift = self.start[axis];
83            let step = self.stride[axis];
84            if let CoordinateModel::Stokes(stokes) = coord {
85                if shift != 0 || step != 1 || self.shape[axis] != stokes.stokes().len() {
86                    self.set_stokes(&mut record, stokes, axis)?;
87                }
88            } else if let CoordinateModel::Quality(quality) = coord {
89                if shift != 0 || step != 1 || self.shape[axis] != quality.values().len() {
90                    let field = CoordinateSystem::record_coordinates(&record)?
91                        .into_iter()
92                        .find(|coordinate| coordinate.pixel_axes.contains(&Some(axis)))
93                        .ok_or_else(|| {
94                            ImageToolError::new(
95                                "Quality coordinate is missing from the image record",
96                            )
97                        })?
98                        .field;
99                    let Some(Value::Record(quality)) = record.get_mut(&field) else {
100                        return Err(ImageToolError::new(
101                            "Quality coordinate is missing from the image record",
102                        ));
103                    };
104                    section_quality_labels(quality, shift, self.shape[axis], step)?;
105                }
106            } else if shift != 0 || step != 1 || self.shape[axis] != source_length {
107                if kind == CoordinateType::Spectral
108                    && self.crop_tabular_spectral(&mut record, axis)?
109                {
110                    continue;
111                }
112                if shift == 0 && step == 1 {
113                    continue;
114                }
115                if let CoordinateModel::Tabular(table) = coord {
116                    self.section_tabular(&mut record, table, axis)?;
117                    continue;
118                }
119                let cdelt = coord.increment()[within_coord];
120                // ImageInterface::subImage scales each coordinate's
121                // increment and leaves its linear transform in place.
122                let (_, stored_increment) =
123                    CoordinateSystem::set_increment_in_record(&mut record, axis, cdelt)?;
124                CoordinateSystem::set_increment_in_record(
125                    &mut record,
126                    axis,
127                    stored_increment * step as f64,
128                )?;
129                let reference = coord.reference_pixel()[within_coord];
130                CoordinateSystem::set_reference_pixel_in_record(
131                    &mut record,
132                    axis,
133                    (reference - shift as f64) / step as f64,
134                )?;
135            }
136        }
137        if self.retained.len() == self.shape.len() {
138            return Ok(record);
139        }
140        self.project_coordinates(&record)
141    }
142
143    fn section_tabular(
144        &self,
145        record: &mut RecordValue,
146        table: &TabularCoordinate,
147        axis: usize,
148    ) -> ImageToolResult<()> {
149        let mut table = table.clone();
150        let shift = self.start[axis];
151        let step = self.stride[axis];
152        // casacore sets the reference pixel before scaling the increment.
153        let reference = (table.reference_pixel()[0] - shift as f64) / step as f64;
154        let increment = table.increment()[0] * step as f64;
155        table.edit_live_reference(TabularEdit::ReferencePixel(reference))?;
156        if step != 1 {
157            table.edit_live_reference(TabularEdit::Increment(increment))?;
158        }
159        let field = CoordinateSystem::record_coordinates(record)?
160            .into_iter()
161            .find(|coordinate| coordinate.pixel_axes.contains(&Some(axis)))
162            .ok_or_else(|| {
163                ImageToolError::new("Tabular coordinate is missing from the image record")
164            })?
165            .field;
166        let Some(Value::Record(stored)) = record.get_mut(&field) else {
167            return Err(ImageToolError::new(
168                "Tabular coordinate is missing from the image record",
169            ));
170        };
171        for parameter in CoordinateModel::Tabular(table).to_record().fields() {
172            stored.upsert(parameter.name.clone(), parameter.value.clone());
173        }
174        Ok(())
175    }
176
177    fn crop_tabular_spectral(
178        &self,
179        record: &mut RecordValue,
180        axis: usize,
181    ) -> ImageToolResult<bool> {
182        let field = CoordinateSystem::record_coordinates(record)?
183            .into_iter()
184            .find(|coordinate| coordinate.pixel_axes.contains(&Some(axis)))
185            .ok_or_else(|| {
186                ImageToolError::new("Spectral coordinate is missing from the image record")
187            })?
188            .field;
189        let Some(Value::Record(spectral)) = record.get_mut(&field) else {
190            return Ok(false);
191        };
192        Ok(SpectralCoordinate::subimage_tabular_record_with_stride(
193            spectral,
194            self.start[axis],
195            self.shape[axis],
196            self.stride[axis],
197        )?)
198    }

200    fn project_coordinates(&self, record: &RecordValue) -> ImageToolResult<RecordValue> {
201        let mut projected =
202            reduce_subimage_coordinate_record(record, self.shape.len(), &self.retained)?;
203        reorder_rebuilt_world_axes(&mut projected)?;
204        let model = CoordinateSystem::from_record(&projected)?;
205        let world_at_origin = model.to_world(&vec![0.0; self.retained.len()])?;
206        let units = model.world_axis_units();
207        for coordinate in model.coordinates_in_record(&projected)? {
208            let suffix = coordinate
209                .field
210                .trim_start_matches(|ch: char| !ch.is_ascii_digit());
211            let pixel_field = format!("pixelreplace{suffix}");
212            let world_field = format!("worldreplace{suffix}");
213            let mut pixel = numbers(&projected, &pixel_field)
214                .unwrap_or_else(|| vec![0.0; coordinate.pixel_axes.len()]);
215            let mut world = numbers(&projected, &world_field)
216                .unwrap_or_else(|| vec![0.0; coordinate.world_axes.len()]);
217            pixel.resize(coordinate.pixel_axes.len(), 0.0);
218            world.resize(coordinate.world_axes.len(), 0.0);
219            let definition = projected
220                .get(&coordinate.field)
221                .expect("coordinate record entry exists");
222            for (local, axis) in coordinate.pixel_axes.into_iter().enumerate() {
223                if axis.is_some() {
224                    pixel[local] = 0.0;
225                }
226            }
227            for (local, axis) in coordinate.world_axes.into_iter().enumerate() {
228                if let Some(axis) = axis {
229                    world[local] =
230                        world_value(world_at_origin[axis], &units[axis], definition, local)?;
231                }
232            }
233            projected.upsert(pixel_field, ArrayValue::from_f64_vec(pixel).into());
234            projected.upsert(world_field, ArrayValue::from_f64_vec(world).into());
235        }
236        Ok(projected)
237    }
238
239    fn set_stokes(
240        &self,
241        record: &mut RecordValue,
242        stokes: &StokesCoordinate,
243        axis: usize,
244    ) -> ImageToolResult<()> {
245        let stored = CoordinateSystem::record_coordinates(record)?
246            .into_iter()
247            .find(|coordinate| coordinate.pixel_axes.contains(&Some(axis)))
248            .ok_or_else(|| {
249                ImageToolError::new("Stokes coordinate is missing from the image record")
250            })?;
251        let values = (0..self.shape[axis])
252            .map(|index| {
253                stokes
254                    .stokes()
255                    .get(self.start[axis] + index * self.stride[axis])
256                    .copied()
257            })
258            .collect::<Option<Vec<_>>>()
259            .ok_or_else(|| {
260                ImageToolError::new("Stokes coordinate does not cover the selected pixels")
261            })?;
262        record.upsert(
263            stored.field,
264            Value::Record(CoordinateModel::from(StokesCoordinate::new(values)).to_record()),
265        );
266        Ok(())
267    }
268}

270fn reorder_rebuilt_world_axes(record: &mut RecordValue) -> ImageToolResult<()> {
271    let coordinates = CoordinateSystem::from_mapped_record(record)?;
272    if (0..coordinates.n_pixel_axes())
273        .all(|pixel| coordinates.pixel_axis_to_world_axis(pixel) == Some(pixel))
274    {
275        return Ok(());
276    }
277    let placements = coordinates.coordinates_in_record(record)?;
278    // The copy factories rebuild local coordinates in creation order and
279    // transpose their world rows by the indices they owned in the section.
280    let mut output = vec![0; coordinates.n_world_axes()];
281    for (axis, world) in placements
282        .iter()
283        .flat_map(|item| item.world_axes.iter().flatten())
284        .enumerate()
285    {
286        output[*world] = axis as i32;
287    }
288    let mut packed = 0;
289    for placement in placements {
290        let suffix = placement
291            .field
292            .trim_start_matches(|ch: char| !ch.is_ascii_digit());
293        let key = format!("worldmap{suffix}");
294        let Some(Value::Array(ArrayValue::Int32(axes))) = record.get_mut(&key) else {
295            return Err(ImageToolError::new("Missing reduced world axis map"));
296        };
297        for axis in axes.iter_mut().filter(|axis| **axis >= 0) {
298            *axis = output[packed];
299            packed += 1;
300        }
301    }
302    Ok(())
303}

305fn world_value(value: f64, unit: &str, definition: &Value, axis: usize) -> ImageToolResult<f64> {
306    let Value::Record(definition) = definition else {
307        return Ok(value);
308    };
309    let target = match definition.get("units") {
310        Some(Value::Array(ArrayValue::String(units))) => units.iter().nth(axis).map(String::as_str),
311        _ => match definition.get("unit") {
312            Some(Value::Scalar(ScalarValue::String(unit))) => Some(unit.as_str()),
313            _ => None,
314        },
315    };
316    if let Some(target) = target.filter(|&target| target != unit) {
317        let quantity = Quantity::new(value, unit)?;
318        return Ok(quantity.get_value_in(&Unit::new(target)?)?);
319    }
320    Ok(value)
321}

323fn numbers(record: &RecordValue, key: &str) -> Option<Vec<f64>> {
324    match record.get(key)? {
325        Value::Array(ArrayValue::Float64(arr)) => Some(arr.t().iter().copied().collect()),
326        Value::Array(ArrayValue::Float32(arr)) => {
327            Some(arr.t().iter().map(|&v| f64::from(v)).collect())
328        }
329        Value::Array(ArrayValue::Int32(arr)) => {
330            Some(arr.t().iter().map(|&v| f64::from(v)).collect())
331        }
332        Value::Array(ArrayValue::Int64(arr)) => Some(arr.t().iter().map(|&v| v as f64).collect()),
333        _ => None,
334    }
