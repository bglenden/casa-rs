2//! `calibrate`, the calibration task provider, on the shared task front end
3//! (#931).
4//!
5//! One binary hosts the general `calibrate` surface, whose `mode` picks the
6//! workflow, and the CASA-named `applycal`, `gaincal`, `bandpass`,
7//! `fluxscale` and `gencal`, one workflow each. The command line's first
8//! word names the surface (`calibrate gaincal --ms ...`); `calibrate` runs
9//! otherwise. The front end owns the command line, `--json-run`, Last and
10//! logging; this module reads the resolved run into its
11//! [`CalibrationTaskRequest`], runs it, and writes its report.
12//!
13//! A run's reply carries the tagged [`CalibrationTaskResult`] and the value
14//! CASA's task returns: `gaincal` and `bandpass` their tables, selected data
15//! and interval counts ([`solve_record`]), `fluxscale` its flux densities
16//! ([`fluxscale_record`]), and `applycal`, `gencal` and `calibrate`
17//! `None`. A command line prints the workflow's report in `format`, or
18//! writes it to `output`; a report that would replace an existing file is
19//! refused before the task runs unless `overwrite` is set.
20//!
21//! [`CalibrationTaskRequest`]: crate::CalibrationTaskRequest
22//! [`CalibrationTaskResult`]: crate::CalibrationTaskResult

45/// The front end of the `calibrate` binary, hosting
46/// [`CALIBRATION_SURFACES`].
47pub fn task_front_end()
48-> TaskFrontEnd<impl Fn(&TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure>> {
49    TaskFrontEnd::new_with_deferred_schema(
50        "calibrate",
51        calibration_task_schema_bundle(),
52        &CALIBRATION_SURFACES,
53        run_task,
54    )
55    .expect("the calibration contract carries its surfaces")
56}

58/// Run one resolved calibration request and write its report.
59fn run_task(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
60    if task.surface() == "ft" {
61        return ft::run(task);
62    }
63    if task.surface() == "uvmodelfit" {
64        return uvmodelfit::run(task);
65    }
66    if matches!(task.surface(), "clearcal" | "delmod" | "uvsub") {
67        return model_columns::run(task);
68    }
69    let report = read::report(task)?;
70    let request = read::task_request(task)?;
71    // Checked before the task runs, so a refused report changes nothing.
72    if let Some(path) = report.output.as_deref() {
73        refuse_existing_report(path, report.overwrite)?;
74        refuse_report_in_product(path, &request, task)?;
75    }
76    if task.surface() == "fluxscale"
77        && let CalibrationTaskRequest::FluxScale(scale) = &request
78        && !scale.append
79        && scale.output_table.exists()
80    {
81        return Err(TaskFailure::validation(format!("fluxtable {} exists. Please specify a different name. Or set append=True, to append the results to the table.",scale.output_table.display())).at("fluxtable"));
82    }
83    let applied = match &request {
84        CalibrationTaskRequest::SolveGain(solve) => solve.prior_calibration_tables.clone(),
85        CalibrationTaskRequest::SolveBandpass(solve) => solve.prior_calibration_tables.clone(),
86        _ => Vec::new(),
87    };
88    let mode = if task.surface() == "calibrate" {
89        Some(if report.overwrite {
90            crate::CalibrationOutputMode::Replace
91        } else {
92            crate::CalibrationOutputMode::Create
93        })
94    } else {
95        None
96    };
97    let multi = match &request {
98        CalibrationTaskRequest::ExecuteApply(request) if task.surface() == "applycal" => {
99            multi_ms::execute(task, request)?
100        }
101        _ => None,
102    };
103    let result = if let Some(report) = multi {
104        CalibrationTaskResult::ApplyMultiMs(report)
105    } else if let CalibrationTaskRequest::ExecuteApply(request) = &request {
106        request
107            .execute_apply()
108            .map(CalibrationTaskResult::Apply)
109            .map_err(|error| {
110                let failure = TaskFailure::execution(error.to_string());
111                match error {
112                    crate::ApplyExecutionError::Plan(source) => match *source {
113                        crate::ApplyPlanError::Argument(
114                            crate::ApplyArgumentError::MissingCalwt,
115                        ) => failure.at("calwt"),
116                        crate::ApplyPlanError::Argument(crate::ApplyArgumentError::SpwMap(_)) => {
117                            failure.at("spwmap")
118                        }
119                        _ => failure,
120                    },
121                    _ => failure,
122                }
123            })?
124    } else {
125        request
126            .execute_with_output_mode(mode)
127            .map_err(TaskFailure::execution)?
128    };
129    if task.surface() == "fluxscale"
130        && let (
131            CalibrationTaskRequest::FluxScale(request),
132            CalibrationTaskResult::FluxScale(report),
133        ) = (&request, &result)
134    {
135        crate::fluxscale::history::record(task, request, report);
136    }
137    if task.surface() == "applycal"
138        && matches!(&result, CalibrationTaskResult::Apply(_))
139        && let CalibrationTaskRequest::ExecuteApply(request) = &request
140    {
141        history::record_applycal(task, &request.measurement_set);
142    }
143    let value = casa_value(task.surface(), &applied, &result);
144    let rendered = render::render(report.format, &result).map_err(TaskFailure::execution)?;
145    let text = match report.output.as_deref() {
146        Some(path) => {
147            std::fs::write(path, rendered).map_err(|error| {
148                TaskFailure::execution(format!("write {}: {error}", path.display()))
149            })?;
150            String::new()
151        }
152        None => rendered,
153    };
154    crate::timing::report_process_memory();
155    Ok(TaskSuccess::new(value, &result)?.with_text(text))
156}

158fn refuse_report_in_product(
159    path: &Path,
160    request: &CalibrationTaskRequest,
161    task: &TaskInvocation<'_>,
162) -> Result<(), TaskFailure> {
163    let mut products = scientific_paths(request);
164    if task.surface() == "fluxscale"
165        && let Some(vis) = task.text("vis")?
166    {
167        products.push(vis.into());
168    }
169    if matches!(task.surface(), "applycal" | "gaincal" | "bandpass")
170        && task.flag("docallib")?
171        && let Some(callib) = task.text("callib")?
172    {
173        products.push(callib.into());
174    }
175    for product in products {
176        if crate::paths::overlap(path, &product)
177            .map_err(|e| TaskFailure::validation(e.to_string()).at("output"))?
178        {
179            return Err(
180                TaskFailure::validation("report path overlaps a scientific product").at("output"),
181            );
182        }
183    }
184    Ok(())
185}

187fn scientific_paths(request: &CalibrationTaskRequest) -> Vec<std::path::PathBuf> {
188    let (mut paths, chain) = match request {
189        CalibrationTaskRequest::SolveGain(r) => (
190            vec![r.measurement_set.clone(), r.output_table.clone()],
191            r.prior_calibration_tables.as_slice(),
192        ),
193        CalibrationTaskRequest::SolveBandpass(r) => (
194            vec![r.measurement_set.clone(), r.output_table.clone()],
195            r.prior_calibration_tables.as_slice(),
196        ),
197        CalibrationTaskRequest::ExecuteApply(r) => (
198            vec![r.measurement_set.clone()],
199            r.calibration_tables.as_slice(),
200        ),
201        CalibrationTaskRequest::PlanApply(r) => (
202            vec![r.measurement_set.clone()],
203            r.calibration_tables.as_slice(),
204        ),
205        CalibrationTaskRequest::Summary(r) => (r.paths.clone(), &[][..]),
206        CalibrationTaskRequest::Stats(r) => (vec![r.path.clone()], &[][..]),
207        CalibrationTaskRequest::FluxScale(r) => {
208            let mut paths = vec![r.input_table.clone(), r.output_table.clone()];
209            paths.extend(r.options.listfile.iter().cloned());
210            (paths, &[][..])
211        }
212        CalibrationTaskRequest::ExportCorrectedData(r) => {
213            (vec![r.input_ms.clone(), r.output_ms.clone()], &[][..])
214        }
215        CalibrationTaskRequest::Gencal(r) => {
216            let mut paths = vec![r.measurement_set.clone(), r.output_table.clone()];
217            if r.caltype == crate::GencalType::Gceff
218                && let Some(source) = r
219                    .gaincurve_table
220                    .clone()
221                    .or_else(crate::gencal::default_vla_gaincurve_table)
222            {
223                paths.push(source);
224            }
225            (paths, &[][..])
226        }
227    };
228    paths.extend(chain.iter().map(|table| table.path.clone()));
229    paths
230}

232/// What `surface` returns for `result`, as CASA's task of that name does;
233/// `applied` is the chain a solve pre-applied.
234fn casa_value(
235    surface: &str,
236    applied: &[ApplyCalibrationTableSpec],
237    result: &CalibrationTaskResult,
238) -> CasaValue {
239    match (surface, result) {
240        ("gaincal", CalibrationTaskResult::SolveGain(report)) => solve_record(
241            applied,
242            &report.selected_data,
243            &report.output_table,
244            &report.solve_stats,
245        ),
246        ("bandpass", CalibrationTaskResult::SolveBandpass(report))
247            if report.table_subtype == "BPOLY" && report.solution_row_count > 0 =>
248        {
249            crate::casa_return::polynomial_solve_record(
250                applied,
251                &report.selected_data,
252                &report.output_table,
253            )
254        }
255        ("bandpass", CalibrationTaskResult::SolveBandpass(report))
256            if report.solve_stats.windows.is_empty() =>
257        {
258            crate::casa_return::skipped_solve_record(
259                if report.selection_failed {
260                    &[]
261                } else {
262                    applied
263                },
264                (!report.selected_data.fields.is_empty()).then_some(&report.selected_data),
265            )
266        }
267        ("bandpass", CalibrationTaskResult::SolveBandpass(report)) => solve_record(
268            applied,
269            &report.selected_data,
270            &report.output_table,
271            &report.solve_stats,
272        ),
273        ("fluxscale", CalibrationTaskResult::FluxScale(report)) => fluxscale_record(report),
274        _ => CasaValue::None,
275    }
276}

278fn refuse_existing_report(path: &Path, overwrite: bool) -> Result<(), TaskFailure> {
279    if path.exists() && !overwrite {
280        return Err(TaskFailure::validation(format!(
281            "refusing to overwrite existing output {}; set overwrite to replace it",
282            path.display()
283        ))
284        .at("output"));
285    }
286    Ok(())
287}
