31/// Produce every planned member and write it to `output`'s staging.
32///
33/// Runs restoring-beam fitting, restoration, residual scaling, normalization,
34/// validity, and metadata exactly once per member, in the planned publication
35/// order. [`CasaImageWriter::publish`] then replaces the targets.
36///
37/// # Errors
38///
39/// Unsupported roles, failed beam fits, generated non-finite payloads, and
40/// image writes that fail. A failed generation discards every member it
41/// staged, so a later [`CasaImageWriter::publish`] publishes none of them.
42pub fn produce_continuum_members(
43    planned: &PlannedContinuumGeneration,
44    inputs: &ContinuumProductInputs<'_>,
45    storage_plan: ProductStoragePlan,
46    execution: &impl ProductWindowExecutor,
47    output: &CasaImageWriter,
48) -> Result<PublishedContinuumGeneration, ProductsError> {
49    if inputs.cube_taylor_primary_beam().is_some() && planned.primary_beam_model.is_none() {
50        return Err(ProductsError::MissingPrimaryBeamModel);
51    }
52    let kept = output.staged_count();
53    let produced = if inputs.normal_state().basis() == NormalBasis::Taylor {
54        produce_taylor_members(planned, inputs, storage_plan, output)
55    } else {
56        produce_plane_members(planned, inputs, storage_plan, execution, output)
57    };
58    if produced.is_err() {
59        output.discard_staged_after(kept);
60    }
61    produced
62}

64fn produce_plane_members(
65    planned: &PlannedContinuumGeneration,
66    inputs: &ContinuumProductInputs<'_>,
67    storage_plan: ProductStoragePlan,
68    execution: &impl ProductWindowExecutor,
69    output: &CasaImageWriter,
70) -> Result<PublishedContinuumGeneration, ProductsError> {
71    let fitted_beams = if planned.requires_beam() {
72        fit_beams(planned, inputs, storage_plan, execution)?
73    } else {
74        Box::new([])
75    };
76    let restoring_beams = restoring_beams(inputs, &fitted_beams)?;
77    for member in &planned.members {
78        let saved_mask = match member.role {
79            ProductRole::RestoredImage(ProductTerm::Single)
80                if planned.saved_residual_mask_read_bytes.is_some() =>
81            {
82                Some(output.saved_residual_mask(
83                    member.axes().domain(),
84                    ProductTerm::Single,
85                    true,
86                )?)
87            }
88            ProductRole::Residual(ProductTerm::Single)
89                if planned.saved_residual_mask_read_bytes.is_some() =>
90            {
91                Some(output.saved_residual_mask(
92                    member.axes().domain(),
93                    ProductTerm::Single,
94                    false,
95                )?)
96            }
97            _ => None,
98        };
99        let windows = MemberWindows::new(
100            planned,
101            member,
102            inputs,
103            storage_plan,
104            &fitted_beams,
105            &restoring_beams,
106        )?;
107        let image = output.begin(
108            member,
109            windows.layout,
110            &planned.beams_for_member(member, &fitted_beams, &restoring_beams),
111            planned.uses_weight_image,
112            saved_mask.as_ref().map(Option::is_some),
113        )?;
114        windows.write(
115            ProductMemberWriter::new(windows.layout, image),
116            execution,
117            saved_mask.as_ref().map(|mask| mask.as_deref()),
118        )?;
119    }
120    Ok(planned.published(fitted_beams, restoring_beams))
121}

123/// Fit a restoring beam to every valid PSF plane, domain by domain, then
124/// channel by channel, then polarization, a wave of the team's workers at a
125/// time.
126fn fit_beams(
127    planned: &PlannedContinuumGeneration,
128    inputs: &ContinuumProductInputs<'_>,
129    storage_plan: ProductStoragePlan,
130    execution: &impl ProductWindowExecutor,
131) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
132    let normal_state = inputs.normal_state();
133    let channels = normal_state.channel_count();
134    let polarizations = normal_state.polarization_count();
135    let beam_count = normal_state
136        .domain_count()
137        .checked_mul(channels)
138        .and_then(|count| count.checked_mul(polarizations))
139        .ok_or(ProductsError::ResourceDemandOverflow(
140            "generation beam count",
141        ))?;
142    let mut fitted = Vec::with_capacity(beam_count);
143    for start in (0..beam_count).step_by(storage_plan.maximum_workers()) {
144        let count = storage_plan.maximum_workers().min(beam_count - start);
145        let mut slots = vec![None; count];
146        execution.prepare(&mut slots, &|index| {
147            let ordinal = start + index;
148            let domain =
149                &inputs.problem().geometry().domains()[ordinal / (channels * polarizations)];
150            let domain_ordinal = inputs.model_domain_ordinal(domain.role());
151            if let Some(saved) = inputs.saved_psf_beams(domain_ordinal) {
152                return Ok(PlaneBeam::Fitted(
153                    saved[ordinal % (channels * polarizations)],
154                ));
155            }
156            let plane = normal_state.read_plane(
157                inputs.model_domain_ordinal(domain.role()),
158                (ordinal / polarizations) % channels,
159                ordinal % polarizations,
160            )?;
161            // A freshly formed PSF is normalized before CASA fits the beam,
162            // so normPSF zeros a sub-threshold peak. A reused PSF is already
163            // in image units: CASA attempts the fit for every positive peak,
164            // and FitGaussianPSF refuses positive peaks below 1e-7.
165            let psf = psf_real_plane(&plane)?;
166            let peak = psf_peak(psf.iter().copied())?;
167            if !psf_normalizes(peak) {
168                return Ok(if planned.reuses_single_psf && peak > 0.0 {
169                    PlaneBeam::Refused
170                } else {
171                    PlaneBeam::Blank
172                });
173            }
174            fit_plane_beam(
175                &psf,
176                plane.shape(),
177                inputs.cell_size_rad_for_domain(domain.role()),
178                planned.psf_cutoff(),
179            )
180        })?;
181        fitted.extend(
182            slots
183                .into_iter()
184                .map(|slot| slot.expect("the executor fills every slot")),
185        );
186    }
187    Ok(settle_beams(&fitted, channels * polarizations))
188}

190/// The beams each plane is restored with, by the problem's restoring-beam
191/// policy: none, each plane's fitted beam, one beam enclosing a domain's
192/// fitted beams, or the beam the request gives. A plane with no fitted beam
193/// (a blank PSF) restores with its domain's largest restoring beam, as
194/// `SIImageStore::makeImageBeamSet` replaces a null beam by the beam set's
195/// largest (`getMaxAreaBeam`) before `restore` restores every plane: a
196/// start model regridded onto a channel no sample reaches is restored
197/// there. A domain with no fitted beam restores none. A Taylor family's planes
198/// are its polarizations.
199pub(crate) fn restoring_beams(
200    inputs: &ContinuumProductInputs<'_>,
201    fitted_beams: &[Option<RestoringBeam>],
202) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
203    let mut beams = policy_restoring_beams(inputs, fitted_beams)?;
204    let normal_state = inputs.normal_state();
205    let planes = normal_state.channel_count() * normal_state.polarization_count();
206    if planes > 0 {
207        for domain_beams in beams.chunks_mut(planes) {
208            let largest = domain_beams
209                .iter()
210                .flatten()
211                .copied()
212                .max_by(|left, right| {
213                    (left.major_fwhm_rad() * left.minor_fwhm_rad())
214                        .total_cmp(&(right.major_fwhm_rad() * right.minor_fwhm_rad()))
215                });
216            for beam in domain_beams.iter_mut().filter(|beam| beam.is_none()) {
217                *beam = largest;
218            }
219        }
220    }
221    Ok(beams)
222}

224/// [`restoring_beams`] before blank planes take their domain's largest.
225fn policy_restoring_beams(
226    inputs: &ContinuumProductInputs<'_>,
227    fitted_beams: &[Option<RestoringBeam>],
228) -> Result<Box<[Option<RestoringBeam>]>, ProductsError> {
229    Ok(match inputs.problem().products().restoring_beam() {
230        RestoringBeamPolicy::None => Box::new([]),
231        RestoringBeamPolicy::PerPlane => fitted_beams.into(),
232        RestoringBeamPolicy::Fixed(beam) => {
233            let beam = fixed_restoring_beam(beam)?;
234            fitted_beams
235                .iter()
236                .map(|fitted| fitted.map(|_| beam))
237                .collect()
238        }
239        RestoringBeamPolicy::Common => {
240            let normal_state = inputs.normal_state();
241            let planes = normal_state.channel_count() * normal_state.polarization_count();
242            let mut selected = Vec::with_capacity(fitted_beams.len());
243            for domain_beams in fitted_beams.chunks(planes) {
244                let valid = domain_beams.iter().flatten().copied().collect::<Vec<_>>();
245                if valid.is_empty() {
246                    selected.extend_from_slice(domain_beams);
247                } else {
248                    let common = RestoringBeam::common_enclosing(&valid)
249                        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))?;
250                    selected.extend(domain_beams.iter().map(|fitted| fitted.map(|_| common)));
251                }
252            }
253            selected.into_boxed_slice()
254        }
255    })
256}

258/// What generating one member's windows needs.
259struct MemberWindows<'a, 'inputs> {
260    planned: &'a PlannedContinuumGeneration,
261    member: &'a PlannedMember,
262    inputs: &'a ContinuumProductInputs<'inputs>,
263    layout: ProductWindowLayout,
264    domain_ordinal: usize,
265    plane_shape: [usize; 2],
266    fitted_beams: &'a [Option<RestoringBeam>],
267    restoring_beams: &'a [Option<RestoringBeam>],
268    workers: usize,
269    fft_threads: usize,
270}
271
272impl<'a, 'inputs> MemberWindows<'a, 'inputs> {
273    fn new(
274        planned: &'a PlannedContinuumGeneration,
275        member: &'a PlannedMember,
276        inputs: &'a ContinuumProductInputs<'inputs>,
277        storage_plan: ProductStoragePlan,
278        fitted_beams: &'a [Option<RestoringBeam>],
279        restoring_beams: &'a [Option<RestoringBeam>],
280    ) -> Result<Self, ProductsError> {
281        let domain_ordinal = inputs.model_domain_ordinal(member.axes().domain());
282        let layout = storage_plan.layout(member.axes())?;
283        let windows = inputs
284            .normal_state()
285            .channel_count()
286            .div_ceil(layout.maximum_channels());
287        Ok(Self {
288            planned,
289            member,
290            inputs,
291            layout,
292            domain_ordinal,
293            plane_shape: inputs.final_model().shape().domains()[domain_ordinal].pixels(),
294            fitted_beams,
295            restoring_beams,
296            workers: storage_plan.maximum_workers().min(windows),
297            fft_threads: storage_plan.fft_threads(windows),
298        })
299    }
300
301    /// Generate the member's windows a wave of workers at a time and write
302    /// them in channel order.
303    fn write(
304        &self,
305        mut writer: ProductMemberWriter<'_>,
306        execution: &impl ProductWindowExecutor,
307        saved_mask: Option<Option<&[bool]>>,
308    ) -> Result<(), ProductsError> {
309        let channels = self.inputs.normal_state().channel_count();
310        let window_channels = self.layout.maximum_channels();
311        let windows = channels.div_ceil(window_channels);
312        for wave_start in (0..windows).step_by(self.workers) {
313            let mut slots: Vec<Option<ProductWindow>> = (0..self.workers.min(windows - wave_start))
314                .map(|_| None)
315                .collect();
316            execution.prepare(&mut slots, &|index| {
317                let start = (wave_start + index) * window_channels;
318                self.window(start..(start + window_channels).min(channels), saved_mask)
319            })?;
320            for slot in slots {
321                writer.write(slot.expect("the executor fills every slot"))?;
322            }
323        }
324        writer.finish()
325    }
326
327    /// One window: every polarization plane of its channels.
328    fn window(
329        &self,
330        channels: Range<usize>,
331        saved_mask: Option<Option<&[bool]>>,
332    ) -> Result<ProductWindow, ProductsError> {
333        let mut window = self.layout.window(channels.clone())?;
334        let reconstruction_only = matches!(
335            self.member.role,
336            ProductRole::Model(ProductTerm::Single | ProductTerm::Taylor(0))
337                | ProductRole::CleanMask
338        );
339        for channel in channels.clone() {
340            for polarization in 0..self.inputs.normal_state().polarization_count() {
341                let local = channel - channels.start;
342                if reconstruction_only {
343                    let plane = self.reconstruction_plane(channel, polarization)?;
344                    self.scatter(
345                        &mut window.payload,
346                        &window.shape,
347                        polarization,
348                        local,
349                        &plane,
350                    );
351                } else {
352                    self.normal_plane(&mut window, (channel, polarization, local), saved_mask)?;
353                }
354            }
355        }
356        Ok(window)
357    }

359    fn scatter<T: Copy>(
360        &self,
361        target: &mut [T],
362        shape: &[usize; 4],
363        polarization: usize,
364        local_channel: usize,
365        plane: &[T],
366    ) {
367        scatter_image_polarization_plane(
368            target,
369            self.member.axes().order(),
370            *shape,
371            polarization,
372            local_channel,
373            self.plane_shape,
374            plane,
375        );
376    }

378    /// A member formed from the final model or the CLEAN mask alone.
379    fn reconstruction_plane(
380        &self,
381        channel: usize,
382        polarization: usize,
383    ) -> Result<Vec<f32>, ProductsError> {
384        let [width, height] = self.plane_shape;
385        if self.member.role == ProductRole::CleanMask {
386            return Ok(reconstruction_support_plane(
387                self.inputs,
388                self.member.axes().domain(),
389                (channel, polarization),
390                width * height,
391            ));
392        }
393        let model = model_real_plane(
394            self.inputs.final_model(),
395            self.domain_ordinal,
396            channel,
397            polarization,
398            self.plane_shape,
399        )?;
400        if self.planned.primary_beam_model != Some(AnalyticPrimaryBeamModel::MosaicSensitivity)
401            || self.planned.preserves_saved_residual
402        {
403            return Ok(model);
404        }
405        let plane =
406            self.inputs
407                .normal_state()
408                .read_plane(self.domain_ordinal, channel, polarization)?;
409        let products = self.inputs.problem().products();
410        apparent_model(
411            model,
412            &plane,
413            products.normalization(),
414            products.validity().primary_beam(),
415            self.planned.primary_beam_model,
416        )
417    }

419    /// A member formed from a normal-state plane: its values, and its pixel
420    /// mask where it stores one.
421    fn normal_plane(
422        &self,
423        window: &mut ProductWindow,
424        (channel, polarization, local_channel): (usize, usize, usize),
425        saved_mask: Option<Option<&[bool]>>,
426    ) -> Result<(), ProductsError> {
427        let plane =
428            self.inputs
429                .normal_state()
430                .read_plane(self.domain_ordinal, channel, polarization)?;
431        if matches!(self.member.role, ProductRole::SumWeights(_)) {
432            // CASA's `.sumwt` is the PSF gridding's sum:
433            // `FTMachine::finalizeToSkyNew` seeds the image store's sumwt
434            // once, from `makePSF` (`refim_mawproject`: 8987.97 is the weight
435            // cell's norms, 0.24 % above the data gridding's). The flat-noise
436            // products divide by the raw sensitivity, where every such sum
437            // cancels, as CASA's `divideResidualByWeight` reduces to.
438            // A faceted domain publishes each facet store's own sum.
439            let facets = self.inputs.facet_sum_weights(self.domain_ordinal);
440            scatter_polarization_plane_state(
441                &mut window.payload,
442                self.member.axes(),
443                window.shape,
444                polarization,
445                local_channel,
446                |facet| match facets {
447                    Some(facets) => facets[facet][polarization] as f32,
448                    None => plane.sum_weight() as f32,
449                },
450            );
451            return Ok(());
452        }
453        let planes = self.inputs.normal_state().polarization_count();
454        let beam_index =
455            (self.domain_ordinal * self.inputs.normal_state().channel_count() + channel) * planes
456                + polarization;
457        let mut values = produce_plane_member(&PlaneMemberRequest {
458            member: self.member,
459            inputs: self.inputs,
460            plane: &plane,
461            domain_ordinal: self.domain_ordinal,
462            polarization,
463            reuses_single_psf: self.planned.reuses_single_psf,
464            saved_weighted_residual: self.planned.preserves_saved_residual,
465            fitted_beam: self.fitted_beams.get(beam_index).copied().flatten(),
466            restoring_beam: self.restoring_beams.get(beam_index).copied().flatten(),
467            primary_beam_model: self.planned.primary_beam_model,
468            fft_threads: self.fft_threads,
469        })?;
470        if matches!(self.member.role, ProductRole::Residual(ProductTerm::Single))
471            && let Some(mean_pb) = self.inputs.cube_taylor_primary_beam()
472        {
473            let channel_pb = primary_beam_plane(
474                self.planned.primary_beam_model,
475                self.inputs,
476                self.member.axes().domain(),
477                &plane,
478                polarization,
479            )?;
480            let cutoff = self
481                .inputs
482                .problem()
483                .products()
484                .validity()
485                .primary_beam()
486                .cutoff();
487            adjust_cube_residual(&mut values, mean_pb, &channel_pb, cutoff)?;
488        }
489        let validity = |rule| {
490            product_plane_validity(
491                rule,
492                &plane,
493                self.planned.primary_beam_model,
494                self.inputs,
495                (self.member.axes().domain(), polarization),
496            )
497        };
498        if self.member.validity != ProductValidityRule::All {
499            zero_invalid_plane_values(&mut values, &validity(self.member.validity)?);
500        }
501        let support = match (saved_mask, self.member.storage.pixel_mask()) {
502            (Some(mask), _) => mask.map(std::borrow::Cow::Borrowed),
503            (None, ProductPixelMask::Explicit(rule)) => {
504                Some(std::borrow::Cow::Owned(validity(rule)?))
505            }
506            (None, ProductPixelMask::Absent) => None,
507        };
508        if let Some(support) = support {
509            self.scatter(
510                &mut window.validity,
511                &window.shape,
512                polarization,
513                local_channel,
514                &support,
515            );
516        }
517        self.scatter(
518            &mut window.payload,
519            &window.shape,
520            polarization,
521            local_channel,
522            &values,
523        );
524        Ok(())
525    }
526}

528/// CASA leaves an entire cube residual plane untouched when its published PB
529/// is blank. Otherwise its common/channel PB ratio applies only above the
530/// channel PB cutoff (`SynthesisUtilMethods::removeFreqDepPB`).
531pub(crate) fn adjust_cube_residual(
532    values: &mut [f32],
533    mean_pb: &[f32],
534    channel_pb: &[f32],
535    cutoff: f32,
536) -> Result<(), ProductsError> {
537    if channel_pb.iter().any(|&value| value > cutoff) {
538        for ((value, &common), &channel) in values.iter_mut().zip(mean_pb).zip(channel_pb) {
539            *value = if channel > cutoff {
540                common * *value / channel
541            } else {
542                0.0
543            };
544            if !value.is_finite() {
545                return Err(ProductsError::GeneratedNonfinite);
546            }
547        }
548    }
549    Ok(())
550}

552fn produce_taylor_members(
553    planned: &PlannedContinuumGeneration,
554    inputs: &ContinuumProductInputs<'_>,
555    storage_plan: ProductStoragePlan,
556    output: &CasaImageWriter,
557) -> Result<PublishedContinuumGeneration, ProductsError> {
558    let products = TaylorProducts::build(inputs, planned, output)?;
559    let fitted_beams = if planned.requires_beam() {
560        products.fitted_beams()
561    } else {
562        Box::new([])
563    };
564    let restoring_beams = match inputs.problem().products().restoring_beam() {
565        RestoringBeamPolicy::None => Box::new([]),
566        RestoringBeamPolicy::PerPlane => products.restoring_beams(),
567        // One beam restores every Stokes plane, and the image records it
568        // alone, as `SDAlgorithmMSMFS::restore` sets one restoring beam.
569        RestoringBeamPolicy::Common | RestoringBeamPolicy::Fixed(_) => products
570            .restoring_beams()
571            .iter()
572            .copied()
573            .find(Option::is_some)
574            .into_iter()
575            .collect(),
576    };
577    let refresh_mvc_mask =
578        inputs.taylor_primary_beam().is_some() && inputs.saved_mosaic_beam().is_some();
579    for member in &planned.members {
580        let mut payload = products.payload(member.role)?;
581        // An ordinary Taylor PB is formed at one frequency and zeroed by
582        // `makePBImage`. MVC's supplied PB is the weighted cube PB: CASA
583        // preserves its values below pblimit while marking those pixels
584        // invalid. The other products still blank outside their support.
585        let supplied_pb = inputs.taylor_primary_beam().is_some()
586            && member.role == ProductRole::PrimaryBeam(ProductTerm::Taylor(0));
587        if member.validity != ProductValidityRule::All && !supplied_pb {
588            zero_invalid_plane_values(
589                &mut payload,
590                &products.validity(member.role, member.validity, refresh_mvc_mask)?,
591            );
592        }
593        let saved_residual_mask = match member.role {
594            ProductRole::RestoredImage(ProductTerm::Taylor(term))
595                if planned.saved_residual_mask_read_bytes.is_some() =>
596            {
597                Some(output.saved_residual_mask(
598                    member.axes().domain(),
599                    ProductTerm::Taylor(term),
600                    true,
601                )?)
602            }
603            _ => None,
604        };
605        let explicit_mask = saved_residual_mask.as_ref().map(Option::is_some);
606        let validity = match (saved_residual_mask, member.storage.pixel_mask()) {
607            (Some(Some(mask)), _) => mask,
608            (Some(None), _)
609            | (None, ProductPixelMask::Absent)
610            | (
611                None,
612                ProductPixelMask::Explicit(
613                    ProductValidityRule::All | ProductValidityRule::FinalNormalState,
614                ),
615            ) => vec![true; member.payload_values],
616            (None, ProductPixelMask::Explicit(rule)) => {
617                products.validity(member.role, rule, refresh_mvc_mask)?
618            }
619        };
620        if payload.len() != member.payload_values || validity.len() != member.payload_values {
621            return Err(ProductsError::PayloadLengthMismatch {
622                expected: member.payload_values,
623                actual: payload.len().max(validity.len()),
624            });
625        }
626        let layout = storage_plan.layout(member.axes())?;
627        let image = output.begin(
628            member,
629            layout,
630            &planned.beams_for_member(member, &fitted_beams, &restoring_beams),
631            planned.uses_weight_image,
632            explicit_mask,
633        )?;
634        let mut writer = ProductMemberWriter::new(layout, image);
635        writer.write_coupled(&payload, &validity)?;
636        writer.finish()?;
637    }
638    Ok(planned.published(fitted_beams, restoring_beams))
639}

643    use super::adjust_cube_residual;
644
645    #[test]
646    fn residual_correction_uses_strict_channel_support_and_skips_a_blank_plane() {
647        let mut residual = [1.0, 2.0, 3.0, 4.0];
648        let common = [0.5, 0.6, 0.7, 0.8];
649        adjust_cube_residual(&mut residual, &common, &[0.1, 0.2, 0.4, 1.0], 0.2).unwrap();
650        assert_eq!(residual, [0.0, 0.0, 0.7 * 3.0 / 0.4, 0.8 * 4.0]);
651        let mut blank = [1.0, 2.0, 3.0, 4.0];
652        adjust_cube_residual(&mut blank, &common, &[0.0, 0.1, 0.2, 0.0], 0.2).unwrap();
653        assert_eq!(blank, [1.0, 2.0, 3.0, 4.0]);
654        let mut overflow = [f32::MAX];
655        assert!(adjust_cube_residual(&mut overflow, &[f32::MAX], &[1.0], 0.2).is_err());
656    }
