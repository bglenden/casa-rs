15pub(super) struct CoordReductionSpec<'a> {
16    pub(super) full_shape: &'a [usize],
17    pub(super) start: &'a [usize],
18    pub(super) shape: &'a [usize],
19    pub(super) kept_axes: &'a [usize],
20    pub(super) selected_stokes: Option<&'a [StokesType]>,
21}

23pub(super) fn reduce_coordinates_and_beams<T: ImagePixel>(
24    input_coords: &CoordinateSystem,
25    paged: Option<&PagedImage<T>>,
26    spec: CoordReductionSpec<'_>,
27    mut image_info: ImageInfo,
28) -> Result<(CoordinateSystem, Option<RecordValue>, ImageInfo), ImageError> {
29    let ndim = spec.full_shape.len();
30    let shifted = subimage_coordinates_with_stokes(
31        input_coords,
32        spec.start,
33        spec.shape,
34        spec.selected_stokes,
35    )?;
36    let shifted_legacy = paged.and_then(|img| {
37        legacy_subimage_coordinates_with_stokes(img, spec.start, spec.shape, spec.selected_stokes)
38    });
39
40    let (out_coords, out_legacy) = if spec.kept_axes.len() == ndim {
41        (shifted, shifted_legacy)
42    } else {
43        let record = shifted_legacy.unwrap_or_else(|| shifted.to_record());
44        let reduced =
45            drop_axes_from_legacy_record(&record, ndim, spec.kept_axes).ok_or_else(|| {
46                ImageError::InvalidMetadata("could not reduce the subimage axis maps".into())
47            })?;
48        (CoordinateSystem::from_record(&reduced)?, Some(reduced))
49    };
50
51    image_info.beam_set = subset_beam_set(
52        &image_info.beam_set,
53        input_coords,
54        ndim,
55        spec.start,
56        spec.shape,
57    )?;
58
59    Ok((out_coords, out_legacy, image_info))
60}

62fn find_system_axis(coords: &CoordinateSystem, kind: CoordinateType) -> Option<usize> {
63    let mut offset = 0usize;
64    for idx in 0..coords.n_coordinates() {
65        let coord = coords.coordinate(idx);
66        if coord.coordinate_type() == kind {
67            return Some(offset);
68        }
69        offset += coord.n_pixel_axes();
70    }
71    None
72}

74fn subset_beam_set(
75    beam_set: &ImageBeamSet,
76    input_coords: &CoordinateSystem,
77    ndim: usize,
78    start: &[usize],
79    shape: &[usize],
80) -> Result<ImageBeamSet, ImageError> {
81    if !beam_set.is_multi() {
82        return Ok(beam_set.clone());
83    }
84    let chans: Vec<usize> = match find_system_axis(input_coords, CoordinateType::Spectral) {
85        Some(ax) if ax < ndim && beam_set.n_channels() > 1 => (start[ax]..start[ax] + shape[ax])
86            .map(|c| c.min(beam_set.n_channels().saturating_sub(1)))
87            .collect(),
88        _ => vec![0],
89    };
90    let stokes: Vec<usize> = match find_system_axis(input_coords, CoordinateType::Stokes) {
91        Some(ax) if ax < ndim && beam_set.n_stokes() > 1 => (start[ax]..start[ax] + shape[ax])
92            .map(|s| s.min(beam_set.n_stokes().saturating_sub(1)))
93            .collect(),
94        _ => vec![0],
95    };
96    beam_set.subset(&chans, &stokes)
97}

99fn strip_index_suffix(field: &str) -> (&str, Option<usize>) {
100    let digits_start = field
101        .rfind(|ch: char| !ch.is_ascii_digit())
102        .map_or(0, |pos| pos + 1);
103    let prefix = &field[..digits_start];
104    let suffix = field[digits_start..].parse::<usize>().ok();
105    (prefix, suffix)
106}

108fn drop_axes_from_legacy_record(
109    record: &RecordValue,
110    ndim: usize,
111    kept_axes: &[usize],
112) -> Option<RecordValue> {
113    let placed = CoordinateSystem::record_coordinates(record).ok()?;
114    let axis_remap: Vec<Option<usize>> = (0..ndim)
115        .map(|old_ax| kept_axes.iter().position(|&k| k == old_ax))
116        .collect();
117
118    struct LegacyCoordEntry<'a> {
119        prefix: &'a str,
120        old_index: usize,
121        coord_val: &'a Value,
122        pixelmap: Vec<i32>,
123        worldmap: Vec<i32>,
124        pixelreplace: Option<&'a Value>,
125        worldreplace: Option<&'a Value>,
126    }
127
128    let mut kept_entries: Vec<LegacyCoordEntry<'_>> = Vec::new();
129    let mut surviving_worlds: BTreeSet<i32> = BTreeSet::new();
130
131    for entry in &placed {
132        let (prefix, Some(old_idx)) = strip_index_suffix(&entry.field) else {
133            return None;
134        };
135        let has_kept = entry
136            .pixel_axes
137            .iter()
138            .any(|ax| ax.is_some_and(|a| axis_remap.get(a).copied().flatten().is_some()));
139        if !has_kept {
140            continue;
141        }
142        let coord_val = record.get(&entry.field)?;
143        let pixelmap = read_i32_array(record.get(&format!("pixelmap{old_idx}"))?)?;
144        let worldmap = read_i32_array(record.get(&format!("worldmap{old_idx}"))?)?;
145        for (j, &pax) in pixelmap.iter().enumerate() {
146            if pax >= 0
147                && axis_remap.get(pax as usize).copied().flatten().is_some()
148                && let Some(&wax) = worldmap.get(j)
149                && wax >= 0
150            {
151                surviving_worlds.insert(wax);
152            }
153        }
154        kept_entries.push(LegacyCoordEntry {
155            prefix,
156            old_index: old_idx,
157            coord_val,
158            pixelmap,
159            worldmap,
160            pixelreplace: record.get(&format!("pixelreplace{old_idx}")),
161            worldreplace: record.get(&format!("worldreplace{old_idx}")),
162        });
163    }
164
165    // Preserve creation index order when renumbering 0..kept.
166    kept_entries.sort_by_key(|e| e.old_index);
167    let world_remap: BTreeMap<i32, i32> = surviving_worlds
168        .into_iter()
169        .enumerate()
170        .map(|(new_w, old_w)| (old_w, new_w as i32))
171        .collect();
172
173    let mut out_fields = Vec::new();
174    for field in record.fields() {
175        let name = &field.name;
176        let is_coord_key = name.starts_with("pixelmap")
177            || name.starts_with("worldmap")
178            || name.starts_with("pixelreplace")
179            || name.starts_with("worldreplace")
180            || name.starts_with("coordinate")
181            || ["direction", "spectral", "stokes", "linear", "tabular"]
182                .iter()
183                .any(|p| {
184                    name.starts_with(p) && name[p.len()..].chars().all(|c| c.is_ascii_digit())
185                });
186        if !is_coord_key {
187            out_fields.push(field.clone());
188        }
189    }
190
191    for (new_idx, entry) in kept_entries.into_iter().enumerate() {
192        let mut new_pmap = Vec::with_capacity(entry.pixelmap.len());
193        let mut new_wmap = Vec::with_capacity(entry.worldmap.len());
194        for j in 0..entry.pixelmap.len() {
195            let old_p = entry.pixelmap[j];
196            let old_w = entry.worldmap.get(j).copied().unwrap_or(old_p);
197            if old_p < 0 {
198                new_pmap.push(old_p);
199                new_wmap.push(old_w);
200            } else if let Some(new_p) = axis_remap.get(old_p as usize).copied().flatten() {
201                new_pmap.push(new_p as i32);
202                new_wmap.push(world_remap.get(&old_w).copied().unwrap_or(new_p as i32));
203            } else {
204                new_pmap.push(-(old_p + 1));
205                new_wmap.push(-(old_w + 1));
206            }
207        }
208        out_fields.push(RecordField::new(
209            format!("{}{new_idx}", entry.prefix),
210            entry.coord_val.clone(),
211        ));
212        out_fields.push(RecordField::new(
213            format!("worldmap{new_idx}"),
214            Value::Array(ArrayValue::from_i32_vec(new_wmap)),
215        ));
216        out_fields.push(RecordField::new(
217            format!("pixelmap{new_idx}"),
218            Value::Array(ArrayValue::from_i32_vec(new_pmap)),
219        ));
220        for (kind, field, replacement, map) in [
221            ("pixel", "crpix", entry.pixelreplace, &entry.pixelmap),
222            ("world", "crval", entry.worldreplace, &entry.worldmap),
223        ] {
224            let Some(replacement) = replacement else {
225                continue;
226            };
227            let mut value = replacement.clone();
228            if entry.prefix == "direction"
229                && let Value::Record(direction) = entry.coord_val
230                && let Value::Array(ArrayValue::Float64(values)) = &mut value
231                && let Some(Value::Array(ArrayValue::Float64(reference))) = direction.get(field)
232            {
233                // Rebuilding the surviving direction coordinate uses its
234                // reference values for newly removed axes (dropRemovedAxes).
235                for (axis, &old) in map.iter().enumerate() {
236                    if old >= 0 && axis_remap[old as usize].is_none() {
237                        values[axis] = reference[axis];
238                    }
239                }
240            }
241            out_fields.push(RecordField::new(format!("{kind}replace{new_idx}"), value));
242        }
243    }
244
245    let out = RecordValue::new(out_fields);
246    Some(out)
247}

249fn read_i32_array(value: &Value) -> Option<Vec<i32>> {
250    match value {
251        Value::Array(ArrayValue::Int32(arr)) => Some(arr.iter().copied().collect()),
252        Value::Array(ArrayValue::Int64(arr)) => {
253            arr.iter().map(|&v| i32::try_from(v).ok()).collect()
254        }
255        _ => None,
256    }
