2//! One scan of an MSv2's `POINTING`, finding what a conversion's reads of
3//! it would carry ([`scan`]): which rows each partition reads, as the MSv2
4//! adapter reads a span of times (`PreparedSelectedPointingCatalog`'s
5//! retention), the frames and polynomial terms of those rows, the rows a
6//! read would refuse, and the rows nothing reads.

21/// `POINTING` rows scanned at a time.
22const SCAN_ROWS: usize = 1 << 16;
23
24/// The most [`scan`] holds for partitions naming `antennas` antennas in
25/// all: a block of rows (each one's number, antenna, order, time, interval
26/// and origin), and each antenna's nearest rows either side and its count.
27pub(super) fn scan_bytes(antennas: usize) -> usize {
28    let row = size_of::<usize>() + 2 * size_of::<i32>() + 3 * size_of::<f64>();
29    let antenna = size_of::<[Option<Candidate>; 2]>() + 2 * size_of::<(AntennaId, usize)>();
30    SCAN_ROWS * row + antennas * antenna
31}
32
33/// One partition as the scan sees it: the antennas its baselines name and
34/// its times.
35pub(super) struct Partition {
36    pub(super) antennas: Vec<AntennaId>,
37    pub(super) times: Option<RangeInclusive<f64>>,
38}
39
40impl Partition {
41    fn slot(&self, antenna: i32) -> Option<usize> {
42        u32::try_from(antenna)
43            .ok()
44            .and_then(|id| self.antennas.binary_search(&AntennaId(id)).ok())
45    }
46}
47
48/// One `POINTING` row's scalar metadata.
49#[derive(Clone, Copy)]
50struct Candidate {
51    row: usize,
52    antenna: i32,
53    time: f64,
54    interval: f64,
55    terms: usize,
56}
57
58impl Candidate {
59    /// Whether the MSv2 adapter reads this row for a span of `times`: its
60    /// interval meets them or its time is within them, as
61    /// `PreparedSelectedPointingCatalog` retains rows. Into an MSv2 a read
62    /// spans every row.
63    fn within(&self, times: &RangeInclusive<f64>, destination: Store) -> bool {
64        let half = 0.5 * self.interval;
65        destination == Store::Msv2
66            || times.contains(&self.time)
67            || (half > 0.0
68                && self.time + half >= *times.start()
69                && self.time - half <= *times.end())
70    }
71}
72
73/// What one scan of an MSv2's `POINTING` finds.
74#[derive(Default)]
75pub(super) struct Scan {
76    /// The frames of the rows each partition reads.
77    pub(super) frames: Vec<Vec<DirectionRef>>,
78    /// The most polynomial terms of a row some partition reads.
79    pub(super) most_terms: usize,
80    /// A row of an antenna some partition names that the MSv2 adapter
81    /// would refuse: polynomial terms beyond its bound, or times that are
82    /// not finite.
83    pub(super) refused: Option<&'static str>,
84    /// The earliest and latest time of a row some partition names.
85    pub(super) times: Option<(f64, f64)>,
86    /// Rows no partition reads.
87    pub(super) outside: usize,
88    /// Each partition's rows within its times.
89    pub(super) within: Vec<usize>,
90    /// Each named antenna's rows.
91    pub(super) antenna_rows: BTreeMap<AntennaId, usize>,
92    /// Rows a partition reads at the (antenna, time) of another it reads.
93    pub(super) repeated: usize,
94    /// Each partition's antennas' nearest rows before and after its times,
95    /// as the MSv2 adapter picks them: the latest before and the earliest
96    /// after, the lowest row of a tie.
97    nearest: Vec<Vec<[Option<Candidate>; 2]>>,
98    /// The (antenna, time, partition) of each row a partition reads.
99    keys: Vec<(i32, u64, usize)>,
100}
101
102impl Scan {
103    /// Account for `candidate` among `partitions`; `frame` resolves a row's
104    /// direction frame.
105    fn add(
106        &mut self,
107        candidate: Candidate,
108        partitions: &[Partition],
109        destination: Store,
110        frame: &impl Fn(usize) -> Result<DirectionRef, ObservationError>,
111    ) -> Result<(), ObservationError> {
112        let mut read = false;
113        let mut named = false;
114        for (index, partition) in partitions.iter().enumerate() {
115            let (Some(slot), Some(times)) = (partition.slot(candidate.antenna), &partition.times)
116            else {
117                continue;
118            };
119            named = true;
120            if candidate.within(times, destination) {
121                self.within[index] += 1;
122                self.read_by(index, candidate, frame)?;
123                read = true;
124                continue;
125            }
126            let [before, after] = &mut self.nearest[index][slot];
127            if candidate.time < *times.start()
128                && before.is_none_or(|known| {
129                    candidate.time > known.time
130                        || (candidate.time == known.time && candidate.row < known.row)
131                })
132            {
133                *before = Some(candidate);
134            }
135            if candidate.time > *times.end()
136                && after
137                    .is_none_or(|known| (candidate.time, candidate.row) < (known.time, known.row))
138            {
139                *after = Some(candidate);
140            }
141        }
142        if named {
143            // The MSv2 adapter refuses a read meeting such a row, read or not.
144            if candidate.terms > POLYNOMIAL_TERMS {
145                self.refused = Some("rows of more than four polynomial terms");
146            }
147            if !(candidate.time.is_finite() && candidate.interval.is_finite()) {
148                self.refused = Some("rows of non-finite times");
149            }
150            *self
151                .antenna_rows
152                .entry(AntennaId(candidate.antenna as u32))
153                .or_default() += 1;
154            self.times = Some(
155                self.times
156                    .map_or((candidate.time, candidate.time), |(first, last)| {
157                        (first.min(candidate.time), last.max(candidate.time))
158                    }),
159            );
160        }
161        if !read {
162            self.outside += 1;
163        }
164        Ok(())
165    }
166
167    /// `candidate` is read by partition `index`.
168    fn read_by(
169        &mut self,
170        index: usize,
171        candidate: Candidate,
172        frame: &impl Fn(usize) -> Result<DirectionRef, ObservationError>,
173    ) -> Result<(), ObservationError> {
174        let own = frame(candidate.row)?;
175        if !self.frames[index].contains(&own) {
176            self.frames[index].push(own);
177        }
178        self.most_terms = self.most_terms.max(candidate.terms);
179        self.keys
180            .push((candidate.antenna, candidate.time.to_bits(), index));
181        Ok(())
182    }
183
184    /// Account for each partition's antennas' nearest rows either side, and
185    /// count the rows a partition reads at the (antenna, time) of another.
186    fn finish(
187        &mut self,
188        partitions: &[Partition],
189        destination: Store,
190        frame: &impl Fn(usize) -> Result<DirectionRef, ObservationError>,
191    ) -> Result<(), ObservationError> {
192        let nearest: Vec<(usize, Candidate)> = std::mem::take(&mut self.nearest)
193            .into_iter()
194            .enumerate()
195            .flat_map(|(index, sides)| {
196                sides
197                    .into_iter()
198                    .flatten()
199                    .flatten()
200                    .map(move |candidate| (index, candidate))
201            })
202            .collect();
203        let mut beyond = Vec::new();
204        for (index, candidate) in nearest {
205            self.read_by(index, candidate, frame)?;
206            let read_within = partitions.iter().any(|partition| {
207                partition.slot(candidate.antenna).is_some()
208                    && partition
209                        .times
210                        .as_ref()
211                        .is_some_and(|times| candidate.within(times, destination))
212            });
213            if !read_within {
214                beyond.push(candidate.row);
215            }
216        }
217        beyond.sort_unstable();
218        beyond.dedup();
219        self.outside -= beyond.len();
220        self.keys.sort_unstable();
221        self.repeated = self
222            .keys
223            .windows(2)
224            .filter(|pair| pair[0] == pair[1])
225            .count();
226        self.keys = Vec::new();
227        Ok(())
228    }
229}

231/// Scan `POINTING`'s antennas, times, intervals and polynomial orders, and
232/// the frames of the rows partitions read ([`Scan`]).
233pub(super) fn scan(
234    table: &Table,
235    partitions: &[Partition],
236    destination: Store,
237) -> Result<Scan, ObservationError> {
238    let fixed = match TableMeasDesc::reconstruct(table, "DIRECTION") {
239        Some(description) => matches!(description.ref_desc(), MeasRefDesc::Fixed { .. }),
240        None => true,
241    };
242    let resolve = |row: usize| {
243        resolve_direction_reference_selected(table, "POINTING", "DIRECTION", row).map_err(storage)
244    };
245    let column_frame = if fixed { Some(resolve(0)?) } else { None };
246    let frame = |row: usize| column_frame.map_or_else(|| resolve(row), Ok);
247    let mut scan = Scan {
248        frames: vec![Vec::new(); partitions.len()],
249        within: vec![0; partitions.len()],
250        nearest: partitions
251            .iter()
252            .map(|partition| vec![[None, None]; partition.antennas.len()])
253            .collect(),
254        ..Scan::default()
255    };
256    let rows = table.row_count();
257    let (mut antennas, mut orders) = (Vec::new(), Vec::new());
258    let (mut times, mut intervals, mut origins) = (Vec::new(), Vec::new(), Vec::new());
259    for start in (0..rows).step_by(SCAN_ROWS) {
260        let chunk: Vec<usize> = (start..rows.min(start + SCAN_ROWS)).collect();
261        for values in [&mut antennas, &mut orders] {
262            values.clear();
263        }
264        for values in [&mut times, &mut intervals, &mut origins] {
265            values.clear();
266        }
267        let mut destinations = [
268            RequiredScalarColumnDestination::new(
269                "ANTENNA_ID",
270                RequiredScalarColumnValuesMut::Int32(&mut antennas),
271            ),
272            RequiredScalarColumnDestination::new(
273                "TIME",
274                RequiredScalarColumnValuesMut::Float64(&mut times),
275            ),
276            RequiredScalarColumnDestination::new(
277                "INTERVAL",
278                RequiredScalarColumnValuesMut::Float64(&mut intervals),
279            ),
280            RequiredScalarColumnDestination::new(
281                "TIME_ORIGIN",
282                RequiredScalarColumnValuesMut::Float64(&mut origins),
283            ),
284            RequiredScalarColumnDestination::new(
285                "NUM_POLY",
286                RequiredScalarColumnValuesMut::Int32(&mut orders),
287            ),
288        ];
289        table
290            .required_scalar_columns_for_rows_into(&chunk, &mut destinations)
291            .map_err(storage)?;
292        for (slot, row) in chunk.iter().enumerate() {
293            let candidate = Candidate {
294                row: *row,
295                antenna: antennas[slot],
296                time: times[slot],
297                // A non-finite origin refuses a read as a non-finite time does.
298                interval: if origins[slot].is_finite() {
299                    intervals[slot]
300                } else {
301                    f64::NAN
302                },
303                terms: usize::try_from(orders[slot]).map_or(usize::MAX, |order| order + 1),
304            };
305            scan.add(candidate, partitions, destination, &frame)?;
306        }
307    }
308    scan.finish(partitions, destination, &frame)?;
309    Ok(scan)
