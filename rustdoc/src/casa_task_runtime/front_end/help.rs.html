12/// Help for `program`, which hosts `surfaces` (the first is the default), with
13/// the parameters of `selected`.
14pub(super) fn render_help(
15    program: &str,
16    surfaces: &[SurfaceContractBundle],
17    selected: &SurfaceContractBundle,
18) -> String {
19    let surface = &selected.surface;
20    let word = if surfaces.len() > 1 { " [SURFACE]" } else { "" };
21    let mut out = String::new();
22    let _ = writeln!(out, "{program} - {}", surface.summary());
23    let _ = writeln!(out);
24    let _ = writeln!(out, "Usage:");
25    let _ = writeln!(out, "  {program}{word} [SOURCE] [OVERRIDES] [CONTROLS]");
26    let _ = writeln!(
27        out,
28        "  {program}{word} --show [SOURCE] [OVERRIDES] [--save-params FILE]"
29    );
30    // A surface that declares `--template` (imregrid) keeps it.
31    let template = if crate::cli::surface_declares_flag(selected, "--template") {
32        "--profile-template"
33    } else {
34        "--template | --profile-template"
35    };
36    let _ = writeln!(
37        out,
38        "  {program}{word} {template} | --describe [NAME] | --validate FILE"
39    );
40    let _ = writeln!(
41        out,
42        "  {program} --json-run <SOURCE> | --json-schema | --protocol-info"
43    );
44    if surfaces.len() > 1 {
45        let names = surfaces
46            .iter()
47            .map(|bundle| bundle.surface.id())
48            .collect::<Vec<_>>();
49        let _ = writeln!(out);
50        let _ = writeln!(
51            out,
52            "SURFACE is one of {} (default {}).",
53            names.join(", "),
54            names[0]
55        );
56    }
57    let _ = writeln!(out);
58    out.push_str(task_command_line_help());
59    let _ = writeln!(out);
60    let _ = writeln!(out, "Parameters of {}:", surface.id());
61    let defaults = ParameterSession::defaults(selected.clone()).ok();
62    let mut bindings = surface.bindings().iter().collect::<Vec<_>>();
63    bindings.sort_by_key(|binding| binding.order);
64    for binding in bindings {
65        let default = defaults
66            .as_ref()
67            .and_then(|session| session.states().get(&binding.name))
68            .and_then(|state| state.value.as_ref())
69            .map_or_else(|| "required".to_string(), display_value);
70        let summary = selected
71            .catalog
72            .concept(&binding.concept)
73            .map_or("", |concept| concept.documentation.summary.as_str());
74        let _ = writeln!(
75            out,
76            "  {:<28} {summary} (default {default})",
77            spelling(binding)
78        );
79    }
80    let _ = writeln!(out);
81    let _ = writeln!(
82        out,
83        "--json-run takes {{\"surface\": ..., \"values\": {{...}}}} from SOURCE, or - for stdin, and\n\
84writes one typed reply to stdout; it reads and writes no Last state."
85    );
86    out
87}

89fn spelling(binding: &SurfaceParameterBinding) -> String {
90    let Some(cli) = binding.projections.cli.as_ref() else {
91        return binding.name.clone();
92    };
93    let mut parts = Vec::new();
94    if cli.positional.is_some() {
95        parts.push(binding.name.to_uppercase());
96    }
97    if let Some(flag) = cli.flags.first() {
98        parts.push(match &cli.metavar {
99            Some(metavar) => format!("{flag} {metavar}"),
100            None => flag.clone(),
101        });
102    }
103    if let Some(flag) = cli.false_flags.first() {
104        parts.push(flag.clone());
105    }
106    if parts.is_empty() {
107        format!("--{}", binding.name.replace('_', "-"))
108    } else {
109        parts.join(", ")
110    }
111}
