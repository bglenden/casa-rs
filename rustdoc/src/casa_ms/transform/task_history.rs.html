14/// Resolved task arguments recorded in a MeasurementSet's HISTORY.
15#[derive(Debug, Clone)]
16pub struct TransformTaskHistory {
17    /// CASA task name recorded as the HISTORY origin.
18    pub task: String,
19    /// CASA parameter names and their printable values in signature order.
20    pub parameters: Vec<(String, String)>,
21}
22
23/// Append the task heading, CASA-style argument messages and native version
24/// to the target's HISTORY subtable under its normal write lock.
25pub fn append_task_history(
26    output: &Path,
27    history: &TransformTaskHistory,
28) -> Result<(), MsTransformError> {
29    let heading = [
30        format!("taskname={}", history.task),
31        format!("version: casa-rs {}", env!("CARGO_PKG_VERSION")),
32    ];
33    let messages = heading.into_iter().chain(
34        history
35            .parameters
36            .iter()
37            .map(|(name, value)| format!("{name:<11} = {value}")),
38    );
39    append_history_messages(output, &history.task, messages)
40}
41
42pub(crate) fn append_history_messages(
43    output: &Path,
44    origin: &str,
45    messages: impl IntoIterator<Item = String>,
46) -> Result<(), MsTransformError> {
47    let main = Table::open(TableOptions::new(output)).map_err(|source| {
48        MsTransformError::MutateMeasurementSet {
49            path: output.display().to_string(),
50            source: Box::new(source),
51        }
52    })?;
53    let path = subtable_path_from_main(&main, output, SubtableId::History)
54        .unwrap_or_else(|| output.join(SubtableId::History.name()));
55    drop(main);
56    let failure = |source| MsTransformError::MutateMeasurementSet {
57        path: path.display().to_string(),
58        source: Box::new(source),
59    };
60    let mut table = open_staged_subtable_for_write(&path).map_err(failure)?;
61    let mut rows = Vec::new();
62    for message in messages {
63        let time = SystemTime::now()
64            .duration_since(UNIX_EPOCH)
65            .map_err(|error| failure(TableError::Schema(error.to_string())))?
66            .as_secs_f64()
67            + 40_587.0 * 86_400.0;
68        let text = |text: &str| Value::Scalar(ScalarValue::String(text.to_string()));
69        let empty = || Value::Array(ArrayValue::from_string_vec(vec![String::new()]));
70        let mut row = RecordValue::default();
71        for (column, value) in [
72            ("APP_PARAMS", empty()),
73            ("CLI_COMMAND", empty()),
74            ("APPLICATION", text("ms")),
75            ("MESSAGE", text(&message)),
76            ("OBJECT_ID", Value::Scalar(ScalarValue::Int32(0))),
77            ("OBSERVATION_ID", Value::Scalar(ScalarValue::Int32(-1))),
78            ("ORIGIN", text(origin)),
79            ("PRIORITY", text("INFO")),
80            ("TIME", Value::Scalar(ScalarValue::Float64(time))),
81        ] {
82            row.upsert(column, value);
83        }
84        rows.push(row);
85    }
86    table.append_records_in_place(&rows).map_err(failure)
87}
