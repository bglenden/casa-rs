3//! Restoration and normalization algorithms for continuum image products.
4//!
5//! The restoring kernel is evaluated in physical radians: fitted beam widths
6//! and the image cell scale share one unit system, so a multi-pixel beam
7//! stays a multi-pixel kernel at any cell size.

20const FWHM_TO_SIGMA: f64 = 1.0 / 2.354_820_045_030_949_3;

22/// Publication uses the principal PSF's measured peak, not its sum weight.
23/// Coupled Taylor terms must all use this same divisor.
24pub(crate) fn psf_peak(mut values: impl Iterator<Item = f32>) -> Result<f32, ProductsError> {
25    values.try_fold(0.0_f32, |peak, value| {
26        value
27            .is_finite()
28            .then_some(peak.max(value))
29            .ok_or(ProductsError::GeneratedNonfinite)
30    })
31}

33pub(crate) fn normalized_psf_value(value: f32, peak: f32) -> f32 {
34    if peak > 0.0 { value / peak } else { 0.0 }
35}

37/// Normalize one unnormalized plane to its compiled product normalization.
38///
39/// `UnitResponse` and the currently supported scalar-response `FlatNoise`
40/// path divide by the scalar sensitivity (sum weight). `UnitResponse` does
41/// not apply direction-dependent sensitivity division. When no usable scalar
42/// sensitivity exists every pixel blanks to NaN. `FlatSky` is not produced by
43/// this catalog version.
44///
45/// # Errors
46///
47/// Fails when the normalization is not representable by this catalog.
48pub fn normalize_plane(
49    values: &[f32],
50    normalization: ProductNormalization,
51    sensitivity: f64,
52) -> Result<Vec<f32>, ProductsError> {
53    match normalization {
54        ProductNormalization::UnitResponse | ProductNormalization::FlatNoise => {
55            if !(sensitivity.is_finite() && sensitivity > 0.0) {
56                return Ok(vec![f32::NAN; values.len()]);
57            }
58            Ok(values
59                .iter()
60                .map(|value| value / sensitivity as f32)
61                .collect())
62        }
63        ProductNormalization::FlatSky => Err(ProductsError::UnsupportedProductRole {
64            role: casa_imaging_model::ProductRole::Sensitivity,
65            catalog: crate::CONTINUUM_ALGORITHM_CATALOG_VERSION,
66        }),
67    }
68}

70/// Build the unit-peak elliptical-Gaussian restoring kernel image.
71///
72/// `cell_size_rad` is the image cell scale in radians per pixel on each
73/// axis: the fitted beam widths are radians, so pixel offsets must be
74/// converted to radians before they are divided by the Gaussian sigmas.
75#[must_use]
76pub fn gaussian_beam_image(
77    shape: [usize; 2],
78    beam: &RestoringBeam,
79    cell_size_rad: [f64; 2],
80) -> Array2<f32> {
81    let [width, height] = shape;
82    let sigma_major = (beam.major_fwhm_rad() * FWHM_TO_SIGMA).abs();
83    let sigma_minor = (beam.minor_fwhm_rad() * FWHM_TO_SIGMA).abs();
84    let cos_pa = beam.position_angle_rad().cos();
85    let sin_pa = beam.position_angle_rad().sin();
86    let centre_x = width as f64 / 2.0;
87    let centre_y = height as f64 / 2.0;
88    let mut kernel = Array2::<f32>::zeros((width, height));
89    for x in 0..width {
90        for y in 0..height {
91            let dx = (x as f64 - centre_x) * cell_size_rad[0];
92            let dy = (y as f64 - centre_y) * cell_size_rad[1];
93            let u = dx * cos_pa + dy * sin_pa;
94            let v = -dx * sin_pa + dy * cos_pa;
95            kernel[(x, y)] =
96                (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2))).exp() as f32;
97        }
98    }
99    let peak = kernel.iter().copied().fold(f32::NEG_INFINITY, f32::max);
100    if peak > 0.0 {
101        kernel.mapv_inplace(|value| value / peak);
102    }
103    kernel
104}

106pub(crate) fn restore_model_plane(
107    model: &[f32],
108    residual: Vec<f32>,
109    shape: [usize; 2],
110    beam: &RestoringBeam,
111    cell_size_rad: [f64; 2],
112    total_channels: usize,
113    fft_threads: usize,
114) -> Vec<f32> {
115    // FFT zero signs cannot affect addition except when the residual is -0.
116    if model.iter().all(|value| *value == 0.0)
117        && residual
118            .iter()
119            .all(|value| value.to_bits() != (-0.0_f32).to_bits())
120    {
121        return residual;
122    }
123    let kernel = gaussian_beam_image(shape, beam, cell_size_rad);
124    let mut restored = fft_convolve_with_plan(
125        model,
126        kernel.as_slice().expect("contiguous"),
127        shape,
128        restoration_fft(shape, total_channels, fft_threads),
129    );
130    for (restored, residual) in restored.iter_mut().zip(residual) {
131        *restored += residual;
132    }
133    restored
134}

136fn restoration_fft(shape: [usize; 2], total_channels: usize, threads: usize) -> Fft2<f64> {
137    let fft = Fft2::with_threads(shape, threads).expect("valid restoration FFT shape");
138    // One output channel cannot amortize measured planning across cube planes.
139    if total_channels == 1 {
140        fft.with_estimated_plan()
141    } else {
142        fft
143    }
144}

146/// Result of rescaling one normalized residual plane to a selected beam.
147///
148/// The diagnostics are the first-divergence seam used by the focused CASA
149/// comparator: fitted and common beams are compared before any final restored
150/// image can hide whether selection or smoothing first diverged.
151#[derive(Debug, Clone, PartialEq)]
152pub struct ResidualBeamScaling {
153    values: Vec<f32>,
154    smoothing_beam: Option<RestoringBeam>,
155    area_ratio: f64,
156    applied: bool,
157}
158
159impl ResidualBeamScaling {
160    /// Borrow the scaled residual values.
161    #[must_use]
162    pub fn values(&self) -> &[f32] {
163        &self.values
164    }
165
166    /// Return the deconvolved smoothing beam, if the beams differed.
167    #[must_use]
168    pub const fn smoothing_beam(&self) -> Option<RestoringBeam> {
169        self.smoothing_beam
170    }
171
172    /// Return the target/source beam-area ratio.
173    #[must_use]
174    pub const fn area_ratio(&self) -> f64 {
175        self.area_ratio
176    }
177
178    /// Return whether CASA's pixel-width gate applied smoothing and scaling.
179    #[must_use]
180    pub const fn applied(&self) -> bool {
181        self.applied
182    }
183
184    pub(crate) fn into_values(self) -> Vec<f32> {
185        self.values
186    }
187}

189/// Rescale a normalized residual plane from its fitted beam to a selected beam.
190///
191/// This follows CASA `SIImageStore::rescaleResolution`: deconvolve the fitted
192/// beam from the selected beam, skip effectively identical or sub-pixel
193/// smoothing, otherwise convolve with unit volume and multiply by the
194/// selected/fitted beam-area ratio.
195///
196/// # Errors
197///
198/// Returns [`ProductsError::BeamFitFailed`] when beam deconvolution fails.
199pub fn rescale_residual_to_beam(
200    residual: &[f32],
201    shape: [usize; 2],
202    cell_size_rad: [f64; 2],
203    fitted: RestoringBeam,
204    selected: RestoringBeam,
205) -> Result<ResidualBeamScaling, ProductsError> {
206    let area_ratio = selected.area_sr() / fitted.area_sr();
207    let smoothing_beam = selected
208        .deconvolving_beam(fitted)
209        .map_err(|error| ProductsError::BeamFitFailed(error.to_string()))?;
210    let Some(smoothing_beam) = smoothing_beam else {
211        return Ok(ResidualBeamScaling {
212            values: residual.to_vec(),
213            smoothing_beam: None,
214            area_ratio,
215            applied: false,
216        });
217    };
218    if smoothing_beam.minor_fwhm_rad() <= cell_size_rad[0].hypot(cell_size_rad[1]) {
219        return Ok(ResidualBeamScaling {
220            values: residual.to_vec(),
221            smoothing_beam: Some(smoothing_beam),
222            area_ratio,
223            applied: false,
224        });
225    }
226
227    let mut kernel = gaussian_beam_image(shape, &smoothing_beam, cell_size_rad);
228    let volume = f64::from(kernel.sum());
229    if !(volume.is_finite() && volume > 0.0) {
230        return Err(ProductsError::GeneratedNonfinite);
231    }
232    kernel.mapv_inplace(|value| (f64::from(value) / volume) as f32);
233    let mut values = fft_convolve(
234        residual,
235        kernel.as_slice().expect("Gaussian kernel is contiguous"),
236        shape,
237    );
238    for value in &mut values {
239        *value *= area_ratio as f32;
240    }
241    Ok(ResidualBeamScaling {
242        values,
243        smoothing_beam: Some(smoothing_beam),
244        area_ratio,
245        applied: true,
246    })
247}

249/// Convolve one real plane with an equal-shape kernel through even-shifted
250/// FFTs, matching the reconstruction owner's transform conventions.
251#[must_use]
252pub fn fft_convolve(plane: &[f32], kernel: &[f32], shape: [usize; 2]) -> Vec<f32> {
253    fft_convolve_with_plan(
254        plane,
255        kernel,
256        shape,
257        Fft2::new(shape).expect("valid restoration FFT shape"),
258    )
259}
260
261fn fft_convolve_with_plan(
262    plane: &[f32],
263    kernel: &[f32],
264    shape: [usize; 2],
265    mut fft: Fft2<f64>,
266) -> Vec<f32> {
267    let cells = shape[0] * shape[1];
268    let mut signal = Array2::<Complex64>::from_shape_vec(
269        (shape[0], shape[1]),
270        plane
271            .iter()
272            .map(|v| Complex64::new(f64::from(*v), 0.0))
273            .collect(),
274    )
275    .expect("shape matches payload");
276    let mut response = Array2::<Complex64>::from_shape_vec(
277        (shape[0], shape[1]),
278        kernel
279            .iter()
280            .map(|v| Complex64::new(f64::from(*v), 0.0))
281            .collect(),
282    )
283    .expect("shape matches kernel");
284
285    for data in [&mut signal, &mut response] {
286        shift_even(data);
287        fft.transform(
288            data.as_slice_mut().expect("contiguous restoration plane"),
289            false,
290        )
291        .expect("valid restoration FFT plan");
292        shift_even(data);
293    }
294    for (signal, response) in signal.iter_mut().zip(response.iter()) {
295        *signal *= *response;
296    }
297    shift_even(&mut signal);
298    fft.transform(
299        signal.as_slice_mut().expect("contiguous restoration plane"),
300        true,
301    )
302    .expect("valid restoration FFT plan");
303    shift_even(&mut signal);
304
305    let scale = 1.0 / cells as f64;
306    signal
307        .iter()
308        .map(|value| (value.re * scale) as f32)
309        .collect()
310}
311
312fn shift_even(data: &mut Array2<Complex64>) {
313    let [width, height] = [data.shape()[0], data.shape()[1]];
314    debug_assert_eq!(width % 2, 0);
315    debug_assert_eq!(height % 2, 0);
316    for x in 0..width / 2 {
317        for y in 0..height / 2 {
318            data.swap((x, y), (x + width / 2, y + height / 2));
319            data.swap((x + width / 2, y), (x, y + height / 2));
320        }
321    }
322}

347    #[test]
348    fn psf_normalization_is_exact_and_preserves_coupled_signed_terms() {
349        for amplitude in [f32::MIN_POSITIVE, 1.000_000_7, 12345.67] {
350            let principal = [-0.25 * amplitude, amplitude, 0.5 * amplitude];
351            let peak = psf_peak(principal.into_iter()).unwrap();
352            assert_eq!(peak, amplitude);
353            assert_eq!(
354                principal.map(|value| normalized_psf_value(value, peak)),
355                [-0.25, 1.0, 0.5]
356            );
357            let coupled = [-0.5 * amplitude, 0.25 * amplitude];
358            assert_eq!(
359                coupled.map(|value| normalized_psf_value(value, peak)),
360                [-0.5, 0.25]
361            );
362        }
363        let peak = psf_peak([0.0; 3].into_iter()).unwrap();
364        assert_eq!(normalized_psf_value(0.0, peak), 0.0);
365        for nonfinite in [f32::NAN, f32::INFINITY, f32::NEG_INFINITY] {
366            assert!(psf_peak([1.0, nonfinite].into_iter()).is_err());
367        }
368    }

413    #[test]
414    fn identical_beams_leave_the_residual_bit_exact() {
415        let beam = RestoringBeam::new(4.0e-6, 3.0e-6, 0.2).expect("beam");
416        let residual = vec![0.0, 1.0, -2.0, 3.0];
417        let scaling = rescale_residual_to_beam(&residual, [2, 2], [1.0e-6; 2], beam, beam).unwrap();
418        assert!(!scaling.applied());
419        assert_eq!(scaling.smoothing_beam(), None);
420        assert_eq!(scaling.area_ratio(), 1.0);
421        assert_eq!(scaling.values(), residual);

424    #[test]
425    fn common_beam_scaling_uses_unit_volume_and_the_beam_area_ratio() {
426        let source = RestoringBeam::new(4.0e-6, 3.0e-6, 0.0).expect("source beam");
427        let target = RestoringBeam::new(8.0e-6, 6.0e-6, 0.0).expect("target beam");
428        let shape = [32, 32];
429        let mut residual = vec![0.0_f32; shape[0] * shape[1]];
430        residual[(shape[0] / 2) * shape[1] + shape[1] / 2] = 1.0;
431
432        let scaling =
433            rescale_residual_to_beam(&residual, shape, [1.0e-6; 2], source, target).unwrap();
434        assert!(scaling.applied());
435        assert!(scaling.smoothing_beam().is_some());
436        assert!((scaling.area_ratio() - 4.0).abs() < 1.0e-12);
437        let sum = scaling
438            .values()
439            .iter()
440            .map(|value| f64::from(*value))
441            .sum::<f64>();
442        assert!((sum - scaling.area_ratio()).abs() < 1.0e-5, "{sum}");
443    }
