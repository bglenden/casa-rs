6/// One column to install in an existing disk table.
7pub struct ColumnAddition {
8    /// The column's schema.
9    pub schema: ColumnSchema,
10    /// The default cell for each existing row, or `None` if undefined.
11    pub default: Option<Value>,
12    /// Keyword values stored with the new descriptor.
13    pub keywords: RecordValue,
14    /// Preferred manager class when a new binding is needed.
15    pub manager_type: Option<String>,
16    /// Preferred group name when a new binding is needed.
17    pub manager_group: Option<String>,
18}

20/// Install the new columns together in their own storage manager.
21pub struct ManagerAddition {
22    /// Casacore manager class (for example, `StandardStMan`).
23    pub kind: String,
24    /// Name of this manager in the table.
25    pub name: String,
26    /// Manager configuration (such as tiled default shape).
27    pub spec: RecordValue,
28}

30/// A new CASA CompressComplex column and its completed backing data.
31pub struct CompressedComplexArray<'a> {
32    /// Descriptor for the logical Complex32 array.
33    pub schema: ColumnSchema,
34    /// Name of the StandardStMan scale and offset manager.
35    pub scalar_manager: &'a str,
36    /// Name of the TiledShapeStMan packed-cell manager.
37    pub tiled_manager: &'a str,
38    /// Packed Int32 cells in table row order.
39    pub packed: crate::StreamedTiledShapeColumn,
40    /// Per-row floating-point scales.
41    pub scales: Vec<Option<ScalarValue>>,
42    /// Per-row floating-point offsets.
43    pub offsets: Vec<Option<ScalarValue>>,
44}
45
46impl CompressedComplexArray<'_> {
47    fn groups(
48        &self,
49    ) -> [(
50        Vec<ColumnAddition>,
51        Option<ManagerAddition>,
52        ColumnGroupPayload<'_>,
53    ); 3] {
54        let logical = self.schema.name();
55        let stored = format!("{logical}_COMPRESSED");
56        let scale = format!("{logical}_SCALE");
57        let offset = format!("{logical}_OFFSET");
58        let add = |schema, keywords| ColumnAddition {
59            schema,
60            default: None,
61            keywords,
62            manager_type: None,
63            manager_group: None,
64        };
65        let group = |kind: &str, name: &str| {
66            Some(ManagerAddition {
67                kind: kind.to_owned(),
68                name: name.to_owned(),
69                spec: RecordValue::default(),
70            })
71        };
72        let scalars = [&offset, &scale].map(|column| {
73            add(
74                ColumnSchema::scalar(column, PrimitiveType::Float32),
75                RecordValue::default(),
76            )
77        });
78        let packed = ColumnSchema::array_variable(&stored, PrimitiveType::Int32, Some(2));
79        let keywords =
80            crate::storage::virtual_compress::automatic_complex_keywords(&stored, &scale, &offset);
81        [
82            (
83                scalars.into(),
84                group("StandardStMan", self.scalar_manager),
85                ColumnGroupPayload::Scalars([&self.offsets, &self.scales]),
86            ),
87            (
88                vec![add(packed, RecordValue::default())],
89                group("TiledShapeStMan", self.tiled_manager),
90                ColumnGroupPayload::Tiled(&self.packed),
91            ),
92            (
93                vec![add(self.schema.clone(), keywords)],
94                group("CompressComplex", logical),
95                ColumnGroupPayload::CompressedEngine,
96            ),
97        ]
98    }
99}

101impl Table {
102    /// Append rows to a plain disk table without changing its storage
103    /// manager bindings. Standard and AipsIO managers use `defaults`;
104    /// incremental managers inherit their preceding values and variable
105    /// tiled arrays remain undefined.
106    ///
107    /// # Errors
108    ///
109    /// The table has no durable plain layout, has unsaved edits, contains
110    /// a storage manager that cannot append, or cannot be locked or written.
111    pub fn append_rows_in_place(
112        &mut self,
113        count: usize,
114        defaults: &RecordValue,
115    ) -> Result<(), TableError> {
116        if count == 0 {
117            return Ok(());
118        }
119        self.edit_rows_in_place(Rows::Append { count, defaults })
120    }
121
122    /// Append distinct records in a single edit, preserving the table's
123    /// existing manager bindings. Standard, AipsIO and incremental managers
124    /// accept explicit record values. Omitted cells keep the manager's
125    /// normal append defaults, including inherited incremental values.
126    ///
127    /// # Errors
128    /// The table cannot be locked or written, a row disagrees with its
129    /// schema, or a manager cannot append explicit records.
130    pub fn append_records_in_place(&mut self, records: &[RecordValue]) -> Result<(), TableError> {
131        if records.is_empty() {
132            return Ok(());
133        }
134        let schema = self.schema().ok_or(TableError::PreparedRowRequiresSchema)?;
135        for (offset, row) in records.iter().enumerate() {
136            let row_index = self.row_count() + offset;
137            for field in row.fields() {
138                let column = schema.column(&field.name).ok_or_else(|| {
139                    TableError::RowContainsUnknownColumn {
140                        row_index,
141                        column: field.name.clone(),
142                    }
143                })?;
144                super::columns::validate_cell_against_schema_column(
145                    row_index,
146                    column,
147                    Some(&field.value),
148                )?;
149            }
150        }
151        let defaults = schema.default_row();
152        self.edit_rows_in_place(Rows::AppendRecords {
153            records,
154            defaults: &defaults,
155        })
156    }
157
158    /// Keep the given rows in an existing plain table, in ascending order,
159    /// and delete all others. This operation preserves manager bindings.
160    ///
161    /// # Errors
162    ///
163    /// The row sequence is unsorted or contains repetitions or invalid row
164    /// numbers; the table cannot be written or has a manager that refuses
165    /// to remove rows (for example, IncrementalStMan).
166    pub fn retain_rows_in_place(&mut self, kept: &[usize]) -> Result<(), TableError> {
167        if kept.windows(2).any(|pair| pair[0] >= pair[1]) {
168            return Err(TableError::Storage(
169                "retained rows must strictly increase".into(),
170            ));
171        }
172        self.edit_rows_in_place(Rows::Keep(kept))
173    }
174
175    /// Add columns without changing the files for managers that do not
176    /// receive a column. When supplied, `manager` binds all the new
177    /// columns to one new manager. Otherwise the first manager that
178    /// accepts addition supplies the storage, as in casacore.
179    ///
180    /// # Errors
181    ///
182    /// The table is not a clean plain table, a column or manager name
183    /// already exists, the selected binding is unsupported, or a storage
184    /// file cannot be read or written.
185    pub fn add_columns_in_place(
186        &mut self,
187        columns: Vec<ColumnAddition>,
188        manager: Option<ManagerAddition>,
189    ) -> Result<(), TableError> {
190        if columns.is_empty() {
191            return Ok(());
192        }
193        self.edit_clean_layout("add columns in place", move |this, path, mut dat, old| {
194            dat.raise_row_count(old);
195            let plan =
196                StagedEdit::stage_columns(path, dat, columns, manager, this.data_manager_info())?;
197            this.note_persisted_write(path);
198            plan.publish()?;
199            this.reload_disk_state_at(path)
200        })
201    }
202
203    /// Install complete CASA CompressComplex arrays with their packed cells
204    /// and automatic per-row scale and offset. Stage all requested groups
205    /// before publishing a single descriptor update.
206    ///
207    /// # Errors
208    /// A binding conflicts with the current table, the data do not match
209    /// the requested columns, or a storage file cannot be published.
210    pub fn add_compressed_complex_arrays_in_place(
211        &mut self,
212        arrays: &[CompressedComplexArray<'_>],
213    ) -> Result<(), TableError> {
214        if arrays.iter().any(|array| {
215            array.schema.data_type() != Some(PrimitiveType::Complex32)
216                || [
217                    array.packed.row_count(),
218                    array.scales.len(),
219                    array.offsets.len(),
220                ]
221                .into_iter()
222                .any(|count| count != self.row_count())
223        }) {
224            return Err(TableError::Storage(
225                "compressed complex layout requires one packed cell, scale and offset per row"
226                    .into(),
227            ));
228        }
229        self.edit_clean_layout(
230            "add compressed arrays in place",
231            move |this, path, mut dat, old| {
232                dat.raise_row_count(old);
233                let groups = arrays.iter().flat_map(CompressedComplexArray::groups);
234                let plan =
235                    StagedEdit::stage_column_groups(path, dat, groups, this.data_manager_info())?;
236                this.note_persisted_write(path);
237                plan.publish()?;
238                this.reload_disk_state_at(path)
239            },
240        )
241    }
242
243    fn edit_rows_in_place(&mut self, rows: Rows<'_>) -> Result<(), TableError> {
244        self.edit_clean_layout("edit rows in place", |this, path, dat, old| {
245            if let Rows::Keep(kept) = rows {
246                if kept.last().is_some_and(|&row| row as u64 >= old) {
247                    return Err(TableError::Storage("retained row is out of range".into()));
248                }
249                if kept.len() as u64 == old {
250                    return Ok(());
251                }
252            }
253            let plan = StagedEdit::stage(path, dat, old, rows)?;
254            this.note_persisted_write(path);
255            plan.publish()?;
256            // Publish the new count before reopening; the old lock sync
257            // count otherwise hides removals.
258            this.note_full_save(path)?;
259            this.reload_disk_state_at(path)
260        })
261    }
262
263    fn edit_clean_layout(
264        &mut self,
265        operation: &str,
266        edit: impl FnOnce(
267            &mut Self,
268            &Path,
269            crate::storage::TableDatContents,
270            u64,
271        ) -> Result<(), TableError>,
272    ) -> Result<(), TableError> {
273        let path = self.source_path.clone().ok_or_else(|| {
274            TableError::Storage("in-place row edits require an existing table".into())
275        })?;
276        if self.kind != TableKind::Plain || self.holds_unflushed_changes() {
277            return Err(TableError::Storage(
278                "in-place row edits require a clean plain table".into(),
279            ));
280        }
281        let auto_unlock = self.begin_write_operation(operation)?;
282        let result = (|| {
283            #[cfg(unix)]
284            if auto_unlock {
285                self.reload_if_published()?;
286            }
287            let dat = crate::storage::table_control::read_table_dat(
288                &path.join(crate::storage::TABLE_CONTROL_FILE),
289            )?;
290            let old = self.require_persisted_rows(&path, &dat)?;
291            edit(self, &path, dat, old)
292        })();
293        self.finish_write_operation(auto_unlock, result)
294    }
