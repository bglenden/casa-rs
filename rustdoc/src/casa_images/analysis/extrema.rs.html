13/// Global extrema and zero-based pixel positions, over unmasked pixels.
14#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
15pub struct ImageExtrema {
16    /// Minimum pixel value.
17    pub min: f64,
18    /// Maximum pixel value.
19    pub max: f64,
20    /// Location of the first minimum in casacore pixel order.
21    pub min_position: Vec<usize>,
22    /// Location of the first maximum in casacore pixel order.
23    pub max_position: Vec<usize>,
24    /// CASA-formatted world coordinate of the first minimum.
25    pub min_world: String,
26    /// CASA-formatted world coordinate of the first maximum.
27    pub max_world: String,
28}

30/// Read only bounded pixel and mask chunks to find an image's extrema.
31///
32/// CASA reports no extrema for complex images or wholly masked images.
33/// The chunk buffers are capped at 1,048,576 pixels each; the table storage
34/// manager can additionally retain a complete physical tile in its cache.
35pub fn imhead_extrema(path: impl AsRef<Path>) -> Result<Option<ImageExtrema>, ImageError> {
36    match AnyPagedImage::open(path)? {
37        AnyPagedImage::Float32(image) => extrema(&image),
38        AnyPagedImage::Float64(image) => extrema(&image),
39        AnyPagedImage::Complex32(_) | AnyPagedImage::Complex64(_) => Ok(None),
40    }
41}

43#[derive(Clone)]
44struct Located {
45    value: f64,
46    linear: usize,
47    position: Vec<usize>,
48}

50fn fortran_slice<T>(array: &ArrayD<T>) -> Option<&[T]> {
51    let mut stride = 1isize;
52    for (&length, &actual) in array.shape().iter().zip(array.strides()) {
53        if length > 1 && actual != stride {
54            return None;
55        }
56        stride = stride.checked_mul(isize::try_from(length).ok()?)?;
57    }
58    array.as_slice_memory_order()
59}

61/// Find each chunk's first, minimum and maximum without constructing a
62/// multidimensional index for every pixel. Slices use casacore axis order.
63fn chunk_candidates<T: StatsElement>(
64    values: &[T],
65    mask: Option<&[bool]>,
66) -> [Option<(f64, usize)>; 3] {
67    let mut first = None;
68    let mut minimum: Option<(f64, usize)> = None;
69    let mut maximum: Option<(f64, usize)> = None;
70    for (index, value) in values.iter().enumerate() {
71        if mask.is_some_and(|mask| !mask[index]) {
72            continue;
73        }
74        let value = value.to_f64_stats();
75        if first.is_none() {
76            first = Some((value, index));
77        }
78        if !value.is_nan() {
79            if minimum.is_none_or(|(current, _)| value < current) {
80                minimum = Some((value, index));
81            }
82            if maximum.is_none_or(|(current, _)| value > current) {
83                maximum = Some((value, index));
84            }
85        }
86    }
87    [first, minimum, maximum]
88}

90fn locate(candidate: (f64, usize), start: &[usize], shape: &[usize], strides: &[usize]) -> Located {
91    let (value, mut index) = candidate;
92    let mut linear = 0;
93    let mut position = Vec::with_capacity(shape.len());
94    for ((&start, &length), &stride) in start.iter().zip(shape).zip(strides) {
95        let coordinate = start + index % length;
96        index /= length;
97        linear += coordinate * stride;
98        position.push(coordinate);
99    }
100    Located {
101        value,
102        linear,
103        position,
104    }
105}

107fn merge_candidates(
108    candidates: [Option<Located>; 3],
109    first: &mut Option<Located>,
110    minimum: &mut Option<Located>,
111    maximum: &mut Option<Located>,
112) {
113    let [new_first, new_min, new_max] = candidates;
114    if let Some(value) = new_first
115        && first.as_ref().is_none_or(|v| value.linear < v.linear)
116    {
117        *first = Some(value);
118    }
119    if let Some(value) = new_min
120        && minimum.as_ref().is_none_or(|v| {
121            value.value < v.value || (value.value == v.value && value.linear < v.linear)
122        })
123    {
124        *minimum = Some(value);
125    }
126    if let Some(value) = new_max
127        && maximum.as_ref().is_none_or(|v| {
128            value.value > v.value || (value.value == v.value && value.linear < v.linear)
129        })
130    {
131        *maximum = Some(value);
132    }
133}

135fn extrema<T: ImagePixel + StatsElement>(
136    image: &PagedImage<T>,
137) -> Result<Option<ImageExtrema>, ImageError> {
138    let shape = image.shape();
139    let mut strides = Vec::with_capacity(shape.len());
140    let mut product = 1usize;
141    for &length in shape {
142        strides.push(product);
143        product = product.checked_mul(length).ok_or_else(|| {
144            ImageError::InvalidMetadata("image shape overflows addressable pixels".into())
145        })?;
146    }
147    let chunk_shape = bounded_cursor_shape(image.nice_cursor_shape(), shape, 1_048_576);
148    let cursors = TraversalCursorIter::new(
149        shape.to_vec(),
150        chunk_shape.clone(),
151        TraversalSpec::chunks(chunk_shape),
152    );
153    let mut first: Option<Located> = None;
154    let mut minimum: Option<Located> = None;
155    let mut maximum: Option<Located> = None;
156    let step = vec![1; shape.len()];
157    for cursor in cursors {
158        let cursor = cursor?;
159        let values = image.get_slice(&cursor.position, &cursor.shape)?;
160        let mask = image.get_mask_slice(&cursor.position, &cursor.shape, &step)?;
161        if let Some(pixels) = fortran_slice(&values) {
162            let mask_slice = mask.as_ref().and_then(fortran_slice);
163            if mask.is_none() || mask_slice.is_some() {
164                let candidates = chunk_candidates(pixels, mask_slice).map(|candidate| {
165                    candidate.map(|candidate| {
166                        locate(candidate, &cursor.position, &cursor.shape, &strides)
167                    })
168                });
169                merge_candidates(candidates, &mut first, &mut minimum, &mut maximum);
170                continue;
171            }
172        }
173        for (index, pixel) in values.indexed_iter() {
174            if mask.as_ref().is_some_and(|mask| !mask[index.clone()]) {
175                continue;
176            }
177            let linear = index
178                .slice()
179                .iter()
180                .zip(&cursor.position)
181                .zip(&strides)
182                .map(|((&offset, &start), &stride)| (offset + start) * stride)
183                .sum();
184            let value = pixel.to_f64_stats();
185            let make = || Located {
186                value,
187                linear,
188                position: index
189                    .slice()
190                    .iter()
191                    .zip(&cursor.position)
192                    .map(|(&offset, &start)| offset + start)
193                    .collect(),
194            };
195            if first.as_ref().is_none_or(|v| linear < v.linear) {
196                first = Some(make());
197            }
198            if !value.is_nan() {
199                if minimum
200                    .as_ref()
201                    .is_none_or(|v| value < v.value || (value == v.value && linear < v.linear))
202                {
203                    minimum = Some(make());
204                }
205                if maximum
206                    .as_ref()
207                    .is_none_or(|v| value > v.value || (value == v.value && linear < v.linear))
208                {
209                    maximum = Some(make());
210                }
211            }
212        }
213    }
214    let Some(first) = first else {
215        return Ok(None);
216    };
217    let (min, max) = if first.value.is_nan() {
218        (first.clone(), first)
219    } else {
220        (
221            minimum.expect("a non-NaN first pixel"),
222            maximum.expect("a non-NaN first pixel"),
223        )
224    };
225    let spectral_axis = image.first_spectral_pixel_axis()?;
226    let min_world =
227        super::world_header::format_position(image.coordinates(), &min.position, spectral_axis)?;
228    let max_world =
229        super::world_header::format_position(image.coordinates(), &max.position, spectral_axis)?;
230    Ok(Some(ImageExtrema {
231        min: min.value,
232        max: max.value,
233        min_position: min.position,
234        max_position: max.position,
235        min_world,
236        max_world,
237    }))
238}

240/// Bound our pixel and mask buffers even if the physical tile is enormous.
241fn bounded_cursor_shape(tile: Vec<usize>, shape: &[usize], limit: usize) -> Vec<usize> {
242    let mut remaining = limit;
243    tile.into_iter()
244        .zip(shape)
245        .map(|(tile, &length)| {
246            let extent = tile.min(length).min(remaining).max(1);
247            remaining = (remaining / extent).max(1);
248            extent
249        })
250        .collect()
251}

253#[cfg(test)]
254mod tests {
255    use super::bounded_cursor_shape;
256
257    #[test]
258    fn extrema_pixel_and_mask_buffers_are_bounded_even_for_large_tiles() {
259        for (tile, shape) in [
260            (vec![16_384, 16_384], vec![16_384, 16_384]),
261            (vec![64, 128, 4, 512], vec![64, 128, 4, 512]),
262            (vec![256, 256], vec![64, 32]),
263        ] {
264            let cursor = bounded_cursor_shape(tile, &shape, 1_048_576);
265            assert!(cursor.iter().product::<usize>() <= 1_048_576);
266            assert!(
267                cursor
268                    .iter()
269                    .zip(shape)
270                    .all(|(&n, length)| n > 0 && n <= length)
271            );
272        }
273    }
