pub fn format_world_display(
    coords: &CoordinateSystem,
    pixel: &[f64],
) -> Result<String, CoordinateError>
Expand description

Format the world coordinates of pixel the way CASA’s statistics surfaces render them: one field per world axis, comma-separated.

The pixel vector has one value per pixel axis, in axis order. The returned string is casacore’s CoordinateSystem::format rendering of every world axis, joined with ", " as ia.statistics builds its blcf, trcf, minposf and maxposf strings.

§Errors

A pixel the projection cannot place or whose length is not the coordinate system’s pixel-axis count.

§Examples

use casa_coordinates::format_world_display;
use casa_coordinates::{
    DirectionCoordinate, Projection, ProjectionType, StokesCoordinate, StokesType,
};
use casa_core::measures::direction::DirectionRef;

let arcmin = std::f64::consts::PI / 180.0 / 60.0;
let mut coords = casa_coordinates::CoordinateSystem::new();
coords.add_coordinate(
    DirectionCoordinate::new(
        DirectionRef::J2000,
        Projection::new(ProjectionType::SIN),
        [0.0, 0.0],
        [-arcmin, arcmin],
        [4.0, 4.0],
    )
    .with_world_units(["'", "'"])
    .expect("arcminutes"),
);
coords.add_coordinate(StokesCoordinate::new(vec![StokesType::I]));
let text = format_world_display(&coords, &[0.0, 0.0, 0.0]).expect("inside the projection");
assert_eq!(text, "00:00:16.000, -00.04.00.000, I");
