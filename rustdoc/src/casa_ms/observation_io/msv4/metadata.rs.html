23#[cfg(unix)]
24impl super::Msv4Observation {
25    /// The declared telescope of each observing session, in catalogue order.
26    pub(in crate::observation_io) fn declared_telescopes(
27        &self,
28        grant: &mut dyn FnMut(usize) -> Result<(), ObservationError>,
29    ) -> Result<Vec<String>, ObservationError> {
30        const MISSING: ObservationError = ObservationError::Unsupported {
31            what: "MSv4 calibration requires a declared overall_telescope_name in each antenna_xds",
32        };
33        let mut retained = self
34            .catalogue
35            .observations
36            .len()
37            .saturating_mul(size_of::<String>());
38        grant(retained)?;
39        let mut names = vec![String::new(); self.catalogue.observations.len()];
40        for (dataset, partition) in self.datasets.iter().zip(&self.catalogue.partitions) {
41            let path = format!("{}/antenna_xds", dataset.path);
42            let directory = self.root.join(path.trim_start_matches('/'));
43            let header_bytes = super::held::group_metadata_bytes(&directory);
44            grant(retained.saturating_add(header_bytes))?;
45            let group = zarrs::group::Group::open(self.store.clone(), &path).map_err(storage)?;
46            let name = group
47                .attributes()
48                .get("overall_telescope_name")
49                .and_then(Value::as_str)
50                .filter(|name| !name.trim().is_empty())
51                .ok_or(MISSING)?;
52            let known = &mut names[partition.observation.0 as usize];
53            if known.is_empty() {
54                retained = retained.saturating_add(name.len());
55                grant(retained.saturating_add(header_bytes))?;
56                *known = name.to_string();
57            } else if known != name {
58                return Err(ObservationError::Unsupported {
59                    what: "MSv4 calibration requires a consistent declared overall_telescope_name for each observation",
60                });
61            }
62        }
63        if names.is_empty() || names.iter().any(String::is_empty) {
64            return Err(MISSING);
65        }
66        Ok(names)
67    }
68}

70/// The metadata that all datasets of one spectral window must share.
71#[derive(Debug)]
72pub(super) struct WindowMetadata<'a> {
73    frequencies_hz: Cow<'a, [f64]>,
74    frame: FrequencyRef,
75    channel_width_hz: f64,
76    reference: Option<(f64, FrequencyRef)>,
77}
78
79impl<'a> WindowMetadata<'a> {
80    pub(super) fn new(
81        partition: &'a PartitionEntry,
82        channel_width_hz: f64,
83        reference: Option<&MFrequency>,
84    ) -> Self {
85        Self {
86            frequencies_hz: Cow::Borrowed(&partition.frequencies_hz),
87            frame: partition.frequency_frame,
88            channel_width_hz,
89            reference: reference.map(|frequency| (frequency.hz(), frequency.refer())),
90        }
91    }
92
93    /// Keep one channel axis per name across writer declarations.
94    pub(super) fn into_owned(self) -> WindowMetadata<'static> {
95        WindowMetadata {
96            frequencies_hz: Cow::Owned(self.frequencies_hz.into_owned()),
97            frame: self.frame,
98            channel_width_hz: self.channel_width_hz,
99            reference: self.reference,
100        }
101    }
102
103    /// Compare the values in stored order, with signed zeros equal.
104    pub(super) fn conflict(&self, other: &WindowMetadata<'_>, name: &str) -> Option<String> {
105        let attribute = if self.reference != other.reference {
106            "reference_frequency"
107        } else if self.frame != other.frame {
108            "frequency observer"
109        } else if self.channel_width_hz != other.channel_width_hz {
110            "channel_width"
111        } else if self.frequencies_hz.as_ref() != other.frequencies_hz.as_ref() {
112            "channel frequencies"
113        } else {
114            return None;
115        };
116        Some(format!(
117            "spectral window {name:?} has conflicting {attribute}"
118        ))
119    }
120}

144/// Require `units` to be `expected`: the profile's canonical unit, which
145/// the reader does not convert.
146pub(crate) fn require_units(
147    attributes: &Map<String, Value>,
148    expected: &str,
149    of: &str,
150) -> Result<(), ObservationError> {
151    let units = text(attributes, "units", of)?;
152    if units == expected {
153        Ok(())
154    } else {
155        Err(invalid(format!(
156            "{of} is in {units}; the MSv4 schema requires {expected}"
157        )))
158    }
159}

161/// The time scale and origin of a `time` measure.
162///
163/// XRADIO defines its time attributes by astropy's `Time` (schema
164/// `measures.py`, `TimeArray`), and astropy's `unix` format counts seconds
165/// from 1970-01-01 00:00:00 *UTC* whatever the requested scale: with
166/// `scale: tai`, the represented instant is not the Unix epoch read as TAI
167/// plus the offset, and re-expressing it needs the leap-second table. The
168/// trial profile reads `format: unix` with `scale: utc` only, where the
169/// origin is the Unix epoch in UTC and the offsets are UTC seconds, and
170/// refuses the rest rather than mislabel the instant.
171pub(crate) fn time_reference(
172    attributes: &Map<String, Value>,
173    of: &str,
174) -> Result<(EpochRef, MjdHighPrec), ObservationError> {
175    require_units(attributes, "s", of)?;
176    if text(attributes, "format", of)? != "unix" {
177        return Err(ObservationError::Unsupported {
178            what: "an MSv4 time format other than unix",
179        });
180    }
181    if text(attributes, "scale", of)? != "utc" {
182        return Err(ObservationError::Unsupported {
183            what: "an MSv4 unix-format time in a scale other than utc (astropy's unix epoch is UTC)",
184        });
185    }
186    Ok((EpochRef::UTC, MjdHighPrec::new(UNIX_EPOCH_MJD, 0.0)))
187}
188
189/// The time measure attributes the writer records: `format: unix`,
190/// `scale: utc`, the only pair whose instant the writer's arithmetic
191/// represents ([`time_reference`]).
192pub(crate) fn time_attributes(scale: EpochRef) -> Result<Map<String, Value>, ObservationError> {
193    if scale != EpochRef::UTC {
194        return Err(ObservationError::Unsupported {
195            what: "writing MSv4 times in a scale other than UTC",
196        });
197    }
198    let mut attributes = Map::new();
199    attributes.insert("type".into(), "time".into());
200    attributes.insert("units".into(), "s".into());
201    attributes.insert("scale".into(), "utc".into());
202    attributes.insert("format".into(), "unix".into());
203    Ok(attributes)
204}

206/// Require `array` to have Zarr data type `data_type` and dimensions
207/// `dimensions`, as the profile defines the variable at `path`. Anything
208/// else is refused before a byte is decoded: the reader's decode is a raw
209/// byte copy, which would reinterpret a same-width type silently.
210pub(crate) fn require_layout(
211    array: &Array<FilesystemStore>,
212    path: &str,
213    data_type: &str,
214    dimensions: &[&str],
215) -> Result<(), ObservationError> {
216    let ArrayMetadata::V3(metadata) = array.metadata() else {
217        return Err(ObservationError::Unsupported {
218            what: "an MSv4 variable that is not Zarr v3",
219        });
220    };
221    let stored = metadata.data_type.name();
222    if stored != data_type {
223        return Err(invalid(format!(
224            "{path} is {stored}; the trial profile reads it only as {data_type}"
225        )));
226    }
227    let names: Option<Vec<&str>> = metadata.dimension_names.as_ref().map(|names| {
228        names
229            .iter()
230            .map(|name| name.as_deref().unwrap_or(""))
231            .collect()
232    });
233    if names.as_deref() != Some(dimensions) {
234        return Err(invalid(format!(
235            "{path} has dimensions {names:?}, not {dimensions:?}"
236        )));
237    }
238    Ok(())
239}

241/// XRADIO's spelling of a spectral frame (`observer`) and back. Lowercase
242/// names are astropy's; XRADIO keeps the frames astropy lacks in upper case.
243const FREQUENCY_FRAMES: [(&str, FrequencyRef); 9] = [
244    ("REST", FrequencyRef::REST),
245    ("lsrk", FrequencyRef::LSRK),
246    ("lsrd", FrequencyRef::LSRD),
247    ("BARY", FrequencyRef::BARY),
248    ("gcrs", FrequencyRef::GEO),
249    ("TOPO", FrequencyRef::TOPO),
250    ("GALACTO", FrequencyRef::GALACTO),
251    ("LGROUP", FrequencyRef::LGROUP),
252    ("CMB", FrequencyRef::CMB),
253];
254
255pub(crate) fn frequency_frame(
256    attributes: &Map<String, Value>,
257    of: &str,
258) -> Result<FrequencyRef, ObservationError> {
259    require_units(attributes, "Hz", of)?;
260    let observer = text(attributes, "observer", of)?;
261    FREQUENCY_FRAMES
262        .iter()
263        .find(|(name, _)| *name == observer)
264        .map(|(_, frame)| *frame)
265        .ok_or_else(|| invalid(format!("{of} has unknown observer {observer}")))
266}
267
268pub(crate) fn frequency_observer(frame: FrequencyRef) -> &'static str {
269    FREQUENCY_FRAMES
270        .iter()
271        .find(|(_, known)| *known == frame)
272        .map(|(name, _)| *name)
273        .expect("every frequency frame has an MSv4 name")
274}

276/// XRADIO's `frame` for a direction or UVW measure.
277const DIRECTION_FRAMES: [(&str, DirectionRef); 2] =
278    [("fk5", DirectionRef::J2000), ("icrs", DirectionRef::ICRS)];
279
280pub(crate) fn direction_frame(
281    attributes: &Map<String, Value>,
282    of: &str,
283) -> Result<DirectionRef, ObservationError> {
284    let frame = text(attributes, "frame", of)?;
285    DIRECTION_FRAMES
286        .iter()
287        .find(|(name, _)| *name == frame)
288        .map(|(_, reference)| *reference)
289        .ok_or(ObservationError::Unsupported {
290            what: "an MSv4 direction or UVW frame other than fk5 or icrs",
291        })
292}

313/// Whether the store has node (group or array) `path`: a node absent is
314/// `false`, a store that cannot say an error.
315pub(crate) fn has_node(store: &Arc<FilesystemStore>, path: &str) -> Result<bool, ObservationError> {
316    let key =
317        StoreKey::new(format!("{}/zarr.json", path.trim_start_matches('/'))).map_err(storage)?;
318    store
319        .size_key(&key)
320        .map(|size| size.is_some())
321        .map_err(storage)
322}

332/// Largest decoded physical chunk of `array`. A regular grid has one shape;
333/// varying grids are inspected up to a bounded number of chunks so opening
334/// never understates a chunk extending beyond the logical array extent.
335pub(crate) fn chunk_bytes(array: &Array<FilesystemStore>) -> Result<usize, ObservationError> {
336    let grid = array.chunk_grid();
337    let metadata = serde_json::to_value(grid.metadata()).map_err(storage)?;
338    let size = array.data_type().fixed_size().unwrap_or(0);
339    let bytes = |shape: &[u64]| {
340        shape.iter().fold(size, |product, extent| {
341            product.saturating_mul(usize::try_from(*extent).unwrap_or(usize::MAX))
342        })
343    };
344    let chunk = |indices: &[u64]| -> Result<usize, ObservationError> {
345        let shape = grid.chunk_shape_u64(indices).map_err(storage)?.ok_or(
346            ObservationError::Unsupported {
347                what: "an MSv4 chunk grid without known chunk shapes",
348            },
349        )?;
350        Ok(bytes(&shape))
351    };
352    let mut indices = vec![0; array.shape().len()];
353    if metadata.get("name").and_then(Value::as_str) == Some("regular") {
354        return chunk(&indices);
355    }
356    let grid_shape = grid.grid_shape();
357    let chunks = grid_shape.iter().fold(1_usize, |count, extent| {
358        count.saturating_mul(usize::try_from(*extent).unwrap_or(usize::MAX))
359    });
360    if chunks > 4096 {
361        return Err(ObservationError::Unsupported {
362            what: "an MSv4 varying chunk grid with more than 4096 chunks",
363        });
364    }
365    let mut most = 0;
366    for _ in 0..chunks {
367        most = most.max(chunk(&indices)?);
368        for (index, extent) in indices.iter_mut().zip(grid_shape).rev() {
369            *index += 1;
370            if *index < *extent {
371                break;
372            }
373            *index = 0;
374        }
375    }
376    Ok(most)
377}

379/// Every element of a one-dimensional `fixed_length_utf32` coordinate.
380pub(crate) fn strings(
381    store: &Arc<FilesystemStore>,
382    path: &str,
383) -> Result<Vec<String>, ObservationError> {
384    let array = open_array(store, path)?;
385    let subset = ArraySubset::new_with_shape(array.shape().to_vec());
386    let values: Vec<Vec<char>> = array
387        .retrieve_array_subset_opt(&subset, &super::codec_options())
388        .map_err(storage)?;
389    Ok(values
390        .into_iter()
391        .map(|chars| chars.into_iter().collect())
392        .collect())
393}

395/// Every element of a numeric coordinate or small variable.
396pub(crate) fn numbers<T: zarrs::array::ElementOwned>(
397    store: &Arc<FilesystemStore>,
398    path: &str,
399) -> Result<(Vec<T>, Map<String, Value>), ObservationError> {

402    let values = array
403        .retrieve_array_subset_opt(&subset, &super::codec_options())
404        .map_err(storage)?;
405    Ok((values, array.attributes().clone()))
406}

408/// One data group of a dataset's `data_groups` attribute.
409#[derive(Clone, Debug, PartialEq, Eq)]
410pub(crate) struct DataGroupRecord {
411    pub(crate) name: String,
412    pub(crate) correlated_data: String,
413    pub(crate) flag: String,
414    pub(crate) weight: String,
415    pub(crate) uvw: String,
416    pub(crate) field_and_source: String,
417}
418
419pub(crate) fn data_groups(
420    attributes: &Map<String, Value>,
421) -> Result<Vec<DataGroupRecord>, ObservationError> {
422    let groups = attributes
423        .get("data_groups")
424        .and_then(Value::as_object)
425        .ok_or_else(|| invalid("the dataset has no data_groups"))?;
426    groups
427        .iter()
428        .map(|(name, group)| {
429            let group = group
430                .as_object()
431                .ok_or_else(|| invalid(format!("data group {name} is not a record")))?;
432            let role = |role: &str| text(group, role, name).map(str::to_string);
433            Ok(DataGroupRecord {
434                name: name.clone(),
435                correlated_data: role("correlated_data")?,
436                flag: role("flag")?,
437                weight: role("weight")?,
438                uvw: role("uvw")?,
439                field_and_source: role("field_and_source")?,
440            })
441        })
442        .collect()
