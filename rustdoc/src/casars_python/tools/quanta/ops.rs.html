2//! The `quanta` methods casa-rs implements, as casatools' `quanta_cmpt.cc`
3//! computes them.
4//!
5//! Most methods read their arguments with casatools' `casaQuantity`
6//! ([`crate::convert::quantity`]): a string, a quantity record, or a number
7//! (no unit); other values are 0 with no unit. `quantity`, `unit`,
8//! `getvalue`, `getunit` and `convert` also take quantity records holding
9//! arrays. A failure CASA raises is a `RuntimeError` with CASA's message.
10
11use casa_types::quanta::format::{
12    CalendarDate, angle_format, format_angle, format_time, time_format,
13};
14use casa_types::quanta::{Dimension, Quantity, Unit, UnitDim};
15
16use crate::convert::quantum::{self, Quantum, quantum_holder};
17use crate::convert::value::CasaValue;
18use crate::convert::{ConvertError, ConvertResult, Variant, casacore_unit, quantity};

20/// casacore's `C::circle`.
21const CIRCLE: f64 = 2.0 * std::f64::consts::PI;
22/// casacore's `C::day`, s.
23const DAY_S: f64 = 86_400.0;
24
25fn has(quantity: &Quantity, dimension: Dimension) -> bool {
26    quantity.unit().val().dim == UnitDim::basis(dimension)
27}
28
29fn record(quantity: &Quantity) -> CasaValue {
30    Quantum::scalar(quantity).to_value()
31}
32
33fn non_conforming(quantity: &Quantity) -> ConvertError {
34    ConvertError::runtime(format!(
35        "Quantum::assure non-conforming unit type '{}'",
36        quantity.unit().name()
37    ))
38}

40/// `qa.quantity(v, unitname, keepshape)` and `qa.unit`: a number or numbers
41/// in `unitname`, or a string or record read as a quantity (`unitname`
42/// unused). With `keepshape`, an array of two or more dimensions keeps its
43/// shape; otherwise its values are flattened in Fortran order, as in CASA.
44pub(super) fn quantity_of(
45    value: &Variant,
46    unit_name: &str,
47    keep_shape: bool,
48) -> ConvertResult<CasaValue> {
49    let unit = || casacore_unit(unit_name).map_err(ConvertError::runtime);
50    let quantum = match value {
51        Variant::Number(number) => Quantum::scalar(&Quantity::with_unit(*number, unit()?)),
52        Variant::Numbers(numbers) => Quantum::vector(numbers.clone(), unit()?),
53        Variant::Array { shape, values } => Quantum {
54            values: values.clone(),
55            shape: Some(if keep_shape {
56                shape.clone()
57            } else {
58                vec![values.len()]
59            }),
60            unit: unit()?,
61        },
62        other => match quantum_holder(other)? {
63            Some(quantum) if quantum.is_array() => quantum,
64            _ => Quantum::scalar(&quantity(other)?),
65        },
66    };
67    Ok(quantum.to_value())
68}
69
70/// A quantity record holding an array, else `None`.
71fn array_record(value: &Variant) -> ConvertResult<Option<Quantum>> {
72    if !matches!(value, Variant::Record(_)) {
73        return Ok(None);
74    }
75    Ok(quantum_holder(value)?.filter(Quantum::is_array))
76}

78/// `qa.getvalue(v)`: the value or values, always as an array.
79pub(super) fn getvalue(value: &Variant) -> ConvertResult<CasaValue> {
80    Ok(match array_record(value)? {
81        Some(quantum) => CasaValue::floats(quantum.values),
82        None => CasaValue::floats(vec![quantity(value)?.value()]),
83    })
84}
85
86/// `qa.getunit(v)`.
87pub(super) fn getunit(value: &Variant) -> ConvertResult<CasaValue> {
88    let unit = match array_record(value)? {
89        Some(quantum) => quantum.unit,
90        None => quantity(value)?.unit().clone(),
91    };
92    Ok(CasaValue::Text(unit.name().to_string()))
93}

95/// `qa.convert(v, outunit)`: `v` in the unit of the quantity `outunit`, or
96/// in SI base units when that has none.
97pub(super) fn convert(value: &Variant, out_unit: &Variant) -> ConvertResult<CasaValue> {
98    let holder = quantum_holder(value)?;
99    let target = quantity(out_unit)?.unit().clone();
100    let target = (!target.name().is_empty()).then_some(&target);
101    let quantum = match holder {
102        Some(quantum) if quantum.is_array() => quantum,
103        _ => Quantum::scalar(&quantity(value)?),
104    };
105    Ok(quantum.convert(target)?.to_value())
106}

108/// `qa.canonical(v)` and `qa.canon(v)`: `v` in SI base units.
109pub(super) fn canonical(value: &Variant) -> ConvertResult<CasaValue> {
110    Ok(Quantum::scalar(&quantity(value)?).convert(None)?.to_value())
111}

113/// `qa.add`, `qa.sub`, `qa.mul` and `qa.div`.
114pub(super) fn arithmetic(
115    left: &Variant,
116    right: &Variant,
117    operation: char,
118) -> ConvertResult<CasaValue> {
119    let (left, right) = (quantity(left)?, quantity(right)?);
120    let result = match operation {
121        '+' => quantum::add(&left, &right, false)?,
122        '-' => quantum::add(&left, &right, true)?,
123        '*' => quantum::multiply(&left, &right, false)?,
124        _ => quantum::multiply(&left, &right, true)?,
125    };
126    Ok(record(&result))
127}

129/// `qa.neg(v)`.
130pub(super) fn neg(value: &Variant) -> ConvertResult<CasaValue> {
131    let value = quantity(value)?;
132    Ok(record(&Quantity::with_unit(
133        -value.value(),
134        value.unit().clone(),
135    )))
136}

138/// `qa.abs(v)`.
139pub(super) fn abs(value: &Variant) -> ConvertResult<CasaValue> {
140    let value = quantity(value)?;
141    Ok(record(&Quantity::with_unit(
142        value.value().abs(),
143        value.unit().clone(),
144    )))
145}

147/// `qa.compare(v, a)`: whether the units have the same dimensions.
148pub(super) fn compare(left: &Variant, right: &Variant) -> ConvertResult<CasaValue> {
149    Ok(CasaValue::Bool(
150        quantity(left)?.unit().val().dim == quantity(right)?.unit().val().dim,
151    ))
152}

154/// `qa.check(v)`: whether casacore reads the string as a quantity.
155pub(super) fn check(text: &str) -> CasaValue {
156    CasaValue::Bool(Quantity::read(text).is_ok())
157}

159/// `qa.isquantity(v)`: whether `v` is a string casacore reads as a
160/// quantity, or a quantity record.
161///
162/// Divergence: CASA's records with an integer value are not quantities
163/// (casacore holds them as integer quanta); here every quantity record is.
164pub(super) fn isquantity(value: &Variant) -> CasaValue {
165    CasaValue::Bool(match value {
166        Variant::Text(text) => Quantity::read(text).is_ok(),
167        Variant::Record(_) => quantum_holder(value).is_ok(),
168        _ => false,
169    })
170}

172/// casacore's `MVTime(Quantity)`: MJD days of a time, or turns of an angle.
173fn time_days(value: &Quantity) -> ConvertResult<f64> {
174    let base = value.get_si_value();
175    if has(value, Dimension::Angle) {
176        Ok(base / CIRCLE)
177    } else if has(value, Dimension::Time) {
178        Ok(base / DAY_S)
179    } else {
180        Err(non_conforming(value))
181    }
182}
183
184/// casacore's `MVAngle(Quantity)`: radians of an angle, or of a time at 360
185/// degrees a day.
186fn angle_radians(value: &Quantity) -> ConvertResult<f64> {
187    let base = value.get_si_value();
188    if has(value, Dimension::Angle) {
189        Ok(base)
190    } else if has(value, Dimension::Time) {
191        Ok(base * (CIRCLE / DAY_S))
192    } else {
193        Err(non_conforming(value))
194    }
195}

197/// casatools passes a `long` precision on as an unsigned one.
198fn digits(precision: i64) -> u32 {
199    u32::try_from(precision).unwrap_or(u32::MAX)
200}
201
202/// `qa.time(v, prec, form, showform)`: one string, formatted by casacore's
203/// `MVTime` with the named formats added to `TIME`.
204pub(super) fn time(value: &Variant, precision: i64, form: &[String]) -> ConvertResult<CasaValue> {
205    let days = time_days(&quantity(value)?)?;
206    let format = form.iter().fold(time_format("time"), |format, name| {
207        format | time_format(name)
208    });
209    Ok(CasaValue::List(vec![CasaValue::Text(format_time(
210        days,
211        format,
212        digits(precision),
213    ))]))
214}
215
216/// `qa.angle(v, prec, form, showform)`: one string, formatted by casacore's
217/// `MVAngle` with the named formats added to `ANGLE`.
218pub(super) fn angle(value: &Variant, precision: i64, form: &[String]) -> ConvertResult<CasaValue> {
219    let radians = angle_radians(&quantity(value)?)?;
220    let format = form.iter().fold(angle_format("angle"), |format, name| {
221        format | angle_format(name)
222    });
223    Ok(CasaValue::List(vec![CasaValue::Text(format_angle(
224        radians,
225        format,
226        digits(precision),
227    ))]))
228}

230/// `qa.totime(v)`: a time as it is, an angle as days.
231pub(super) fn totime(value: &Variant) -> ConvertResult<CasaValue> {
232    let value = quantity(value)?;
233    if has(&value, Dimension::Time) {
234        return Ok(record(&value));
235    }
236    let days = time_days(&value)?;
237    Ok(record(&Quantity::with_unit(days, unit("d"))))
238}
239
240/// `qa.toangle(v)`: an angle as it is, a time as radians.
241pub(super) fn toangle(value: &Variant) -> ConvertResult<CasaValue> {
242    let value = quantity(value)?;
243    if has(&value, Dimension::Angle) {
244        return Ok(record(&value));
245    }
246    let radians = angle_radians(&value)?;
247    Ok(record(&Quantity::with_unit(radians, unit("rad"))))
248}
249
250fn unit(name: &str) -> Unit {
251    Unit::new(name).expect("a casacore base unit")
252}

254/// `qa.splitdate(v)`: a time's calendar fields and time of day.
255pub(super) fn splitdate(value: &Variant) -> ConvertResult<CasaValue> {
256    let mjd = time_days(&quantity(value)?)?;
257    let date = CalendarDate::of(mjd);
258    // casatools' chain of `fmod`s and truncations.
259    let mut fraction = mjd % 1.0;
260    let hour = (fraction * 24.0) as i64;
261    fraction = (fraction * 24.0) % 1.0;
262    let minute = (fraction * 60.0) as i64;
263    fraction = (fraction * 60.0) % 1.0;
264    let second = (fraction * 60.0) as i64;
265    let seconds = fraction * 60.0;
266    fraction = (fraction * 60.0) % 1.0;
267    let int = CasaValue::Int;
268    Ok(CasaValue::record([
269        ("mjd", CasaValue::Float(mjd)),
270        ("year", int(date.year)),
271        ("yearday", int(date.yearday)),
272        ("month", int(date.month)),
273        ("monthday", int(date.monthday)),
274        ("week", int(date.week)),
275        ("weekday", int(date.weekday)),
276        ("hour", int(hour)),
277        ("min", int(minute)),
278        ("sec", int(second)),
279        ("s", CasaValue::Float(seconds)),
280        ("msec", int((fraction * 1000.0) as i64)),
281        ("usec", int((fraction * 1.0e6) as i64)),
282    ]))
