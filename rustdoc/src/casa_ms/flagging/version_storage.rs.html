7fn calibration_table(ms: &MeasurementSet) -> bool {
8    ms.main_table().info().table_type == "Calibration"
9}
10
11fn has_row_flags(ms: &MeasurementSet) -> bool {
12    !calibration_table(ms)
13        || ms
14            .main_table()
15            .schema()
16            .is_some_and(|schema| schema.contains_column(FLAG_ROW_COLUMN))
17}

19pub(super) fn flag_version_schema(
20    row_flags: bool,
21) -> Result<TableSchema, casa_tables::SchemaError> {
22    let mut columns = vec![ColumnSchema::array_variable(
23        FLAG_COLUMN,
24        PrimitiveType::Bool,
25        Some(2),
26    )];
27    if row_flags {
28        columns.push(ColumnSchema::scalar(FLAG_ROW_COLUMN, PrimitiveType::Bool));
29    }
30    TableSchema::new(columns)
31}

33/// Save current MAIN flags to a named flag version.
34pub fn save_flag_version(
35    ms: &MeasurementSet,
36    versionname: &str,
37    comment: &str,
38    merge: FlagMerge,
39) -> Result<(), FlaggingError> {
40    validate_version_name(ms, versionname)?;
41    let path = ms_path(ms)?;
42    publish_flag_snapshot(path, versionname, comment, merge, |snapshot| {
43        write_flag_snapshot(ms, snapshot, None, has_row_flags(ms))
44    })
45}

47/// Snapshot the current flags from disk under a retained casacore read lock.
48/// The source lock is released before the version store is updated.
49pub fn save_disk_flag_version(
50    ms: &MeasurementSet,
51    versionname: &str,
52    comment: &str,
53    merge: FlagMerge,
54) -> Result<(), FlaggingError> {
55    validate_version_name(ms, versionname)?;
56    snapshot_disk_flags(ms, versionname, comment, merge)
57}

59/// Save the raw agentflagger version, whose empty name produces `flags.`.
60pub fn save_agent_disk_flag_version(
61    ms: &MeasurementSet,
62    versionname: &str,
63    comment: &str,
64    merge: FlagMerge,
65) -> Result<(), FlaggingError> {
66    if !versionname.is_empty() {
67        return save_disk_flag_version(ms, versionname, comment, merge);

69    snapshot_disk_flags(ms, versionname, comment, merge)

72fn snapshot_disk_flags(
73    ms: &MeasurementSet,
74    versionname: &str,
75    comment: &str,
76    merge: FlagMerge,
77) -> Result<(), FlaggingError> {
78    let path = ms_path(ms)?;
79    publish_flag_snapshot(path, versionname, comment, merge, |snapshot| {
80        let source = open_flag_source(path)?;
81        write_flag_snapshot(&source, snapshot, None, has_row_flags(&source))
82    })
83}

85fn publish_flag_snapshot(
86    path: &Path,
87    versionname: &str,
88    comment: &str,
89    merge: FlagMerge,
90    write: impl FnOnce(&Path) -> Result<(), FlaggingError>,
91) -> Result<(), FlaggingError> {
92    let version_path = flag_version_table_path(path, versionname);
93    let existed = version_path.exists();
94    let failure = |reason: String| FlaggingError::FlagVersion {
95        path: path.display().to_string(),
96        reason: format!("save flag version {versionname:?}: {reason}"),
97    };
98    let stage = tempfile::Builder::new()
99        .prefix(".flag-version-")
100        .tempdir_in(ensure_flagversions_dir(path)?)
101        .map_err(|error| failure(error.to_string()))?;
102    let snapshot = stage.path().join("flags");
103    write(&snapshot)?;
104    if existed {
105        let captured = open_flag_version_for_reading(&snapshot)
106            .map_err(|source| failure(source.to_string()))?;
107        merge_flags_into_existing_version(path, &captured, &version_path, merge)?;
108        if list_flag_versions_for_path(path)?
109            .iter()
110            .any(|entry| entry.name == versionname)
111        {
112            return update_flag_version_comment(path, versionname, comment);
113        }
114    } else {
115        casa_tables::publication::rename_without_replacing(&snapshot, &version_path)
116            .map_err(|source| failure(source.to_string()))?;
117    }
118    append_flag_version_entry(path, versionname, comment)
119}

121fn open_flag_source(path: &Path) -> Result<MeasurementSet, FlaggingError> {
122    #[cfg(unix)]
123    let source = MeasurementSet::open_retained_read(path);
124    #[cfg(not(unix))]
125    let source = MeasurementSet::open(path);
126    source.map_err(|source| FlaggingError::OpenMeasurementSet {
127        path: path.display().to_string(),
128        source,
129    })
130}

132/// Restore a named flag version to MAIN.
133pub fn restore_flag_version(
134    ms: &mut MeasurementSet,
135    versionname: &str,
136    merge: FlagMerge,
137) -> Result<(), FlaggingError> {
138    validate_version_name(ms, versionname)?;
139    let path = ms_path(ms)?.to_path_buf();
140    let version_path = flag_version_table_path(&path, versionname);
141    // The version is read under its read lock, held until the restore ends.
142    let table = open_flag_version_for_reading(&version_path).map_err(|source| {
143        if matches!(version_path.try_exists(), Ok(false)) {
144            return FlaggingError::MissingFlagVersion {
145                version: versionname.to_owned(),
146            };
147        }
148        FlaggingError::FlagVersion {
149            path: path.display().to_string(),
150            reason: format!("open flag version {versionname:?}: {source}"),
151        }
152    })?;
153    if table.row_count() != ms.row_count() {
154        return Err(FlaggingError::FlagVersion {
155            path: path.display().to_string(),
156            reason: format!(
157                "flag version {versionname:?} has {} rows; MAIN has {}",
158                table.row_count(),
159                ms.row_count()
160            ),
161        });
162    }
163    let row_flags = has_row_flags(ms);
164    let plan = restore_plan(ms, &table, &path, row_flags)?;
165    let display_path = path.display().to_string();
166    let mut session =
167        MeasurementSetWriteSession::start_selected_row_mutation(ms, plan).map_err(|source| {
168            FlaggingError::MutateFlags {
169                path: path.display().to_string(),
170                reason: format!("start restored flags: {source}"),
171            }
172        })?;
173    loop {
174        let rows = session
175            .next_mutation_rows()
176            .map_err(|source| FlaggingError::MutateFlags {
177                path: path.display().to_string(),
178                reason: format!("plan restored flag batch: {source}"),
179            })?
180            .to_vec();
181        if rows.is_empty() {
182            break;
183        }
184        let main = ms.main_table();
185        let saved = read_version_flags(&table, &rows, &display_path)?;
186        let flags = restore_flag_arrays(main, &rows, &display_path, saved, merge)?;
187        let mut columns = vec![MeasurementSetMutationColumnBatch {
188            name: FLAG_COLUMN.to_string(),
189            values: MeasurementSetMutationColumnValues::Arrays(flags),
190        }];
191        if row_flags {
192            let mut saved = load_bool_cells(&table, &rows, FLAG_ROW_COLUMN, &display_path)?;
193            if merge != FlagMerge::Replace {
194                let current = load_bool_cells(main, &rows, FLAG_ROW_COLUMN, &display_path)?;
195                for (source, dest) in saved.iter_mut().zip(current) {
196                    *source = merge_bool(*source, dest, merge);
197                }
198            }
199            let flag_rows = saved.into_iter().map(ScalarValue::Bool).collect();
200            columns.push(MeasurementSetMutationColumnBatch {
201                name: FLAG_ROW_COLUMN.to_string(),
202                values: MeasurementSetMutationColumnValues::Scalars(flag_rows),
203            });
204        }
205        session
206            .write_mutation_batch(
207                ms,
208                MeasurementSetMutationBatch {
209                    row_indices: rows,
210                    columns,
211                },
212            )
213            .map_err(|source| FlaggingError::MutateFlags {
214                path: path.display().to_string(),
215                reason: format!("write restored flag batch: {source}"),
216            })?;
217    }
218    session
219        .finish_mutation()
220        .map_err(|source| FlaggingError::MutateFlags {
221            path: path.display().to_string(),
222            reason: format!("finish restored flags: {source}"),
223        })?;
224    Ok(())
225}

227fn read_version_flags(
228    table: &Table,
229    rows: &[usize],
230    path: &str,
231) -> Result<Vec<ArrayD<bool>>, FlaggingError> {
232    let packed = table
233        .column_accessor(FLAG_COLUMN)
234        .and_then(|column| column.packed_2d_cells_uncached(rows))
235        .map_err(|error| FlaggingError::MutateFlags {
236            path: path.into(),
237            reason: error.to_string(),
238        })?;
239    if let Some(ArrayValue::Bool(flags)) = packed {
240        Ok(flags
241            .axis_iter(ndarray::Axis(2))
242            .map(|row| row.to_owned())
243            .collect())
244    } else {
245        clip_time::read_flags_uncached(table, rows, path)
246    }
247}

249fn restore_flag_arrays(
250    main: &Table,
251    rows: &[usize],
252    path: &str,
253    saved: Vec<ArrayD<bool>>,
254    merge: FlagMerge,
255) -> Result<Vec<ArrayValue>, FlaggingError> {
256    if merge == FlagMerge::Replace {
257        // Replacement needs the destination layout, but it does not use
258        // the flag values. Tiled managers answer this from shape headers.
259        let shapes = read_flag_shapes(main, rows).map_err(|error| FlaggingError::MutateFlags {
260            path: path.into(),
261            reason: error.to_string(),
262        })?;
263        for ((&row, source), dest) in rows.iter().zip(&saved).zip(shapes) {
264            if dest.as_deref() != Some(source.shape()) {
265                return Err(FlaggingError::MutateFlags {
266                    path: path.into(),
267                    reason: format!("FLAG shape differs or is undefined at row {row}"),
268                });
269            }
270        }
271        return Ok(saved.into_iter().map(ArrayValue::Bool).collect());
272    }
273
274    let current = clip_time::read_flags_uncached(main, rows, path)?;
275    saved
276        .into_iter()
277        .zip(current)
278        .map(|(source, dest)| merge_bool_arrays(&source, &dest, merge).map(ArrayValue::Bool))
279        .collect()
280}

282fn restore_plan(
283    ms: &MeasurementSet,
284    version: &Table,
285    path: &Path,
286    row_flags: bool,
287) -> Result<MeasurementSetWritePlan, FlaggingError> {
288    let failure = |reason: String| FlaggingError::MutateFlags {
289        path: path.display().to_string(),
290        reason: format!("plan restored flags: {reason}"),
291    };
292    let rows: Vec<_> = (0..ms.row_count()).collect();
293    let maximum = if calibration_table(ms) {
294        matching_calibration_flag_size(ms.main_table(), version, &rows).map_err(failure)?
295    } else {
296        maximum_visibility_cell_elements(ms).map_err(|error| failure(error.to_string()))?
297    };
298    let column = |name: &str, bytes_per_row| MeasurementSetWriteColumnPlan {
299        name: name.to_string(),
300        bytes_per_row,
301        mode: MeasurementSetColumnWriteMode::Replace,
302        storage_manager: MeasurementSetColumnStorage::Persisted,
303        tile_shape: None,
304        create_source_column: None,
305    };
306    let mut columns = vec![column(FLAG_COLUMN, maximum)];
307    if row_flags {
308        columns.push(column(FLAG_ROW_COLUMN, 1));
309    }
310    let resources = MeasurementSetWriteResources::from_system_memory(3)
311        .map_err(|error| failure(error.to_string()))?;
312    MeasurementSetWritePlan::selected_row_mutation(rows, columns, resources)
313        .map_err(|error| failure(error.to_string()))
314}

316fn matching_calibration_flag_size(
317    main: &Table,
318    version: &Table,
319    rows: &[usize],
320) -> Result<usize, String> {
321    let mut maximum = 1;
322    for batch in rows.chunks(256) {
323        let dims = read_flag_shapes(main, batch).map_err(|error| error.to_string())?;
324        let saved = read_flag_shapes(version, batch).map_err(|error| error.to_string())?;
325        for ((&row, original), stored) in batch.iter().zip(dims).zip(saved) {
326            if let (Some(shape), Some(stored)) = (original, stored)
327                && shape == stored
328                && let [receptors, channels] = shape.as_slice()
329                && let Some(size) = receptors.checked_mul(*channels)
330            {
331                maximum = maximum.max(size);
332            } else {
333                return Err(format!("FLAG shape differs or is undefined at row {row}"));
334            }
335        }
336    }
337    Ok(maximum)
338}

340/// Take casacore's write lock on an existing flag-version table before it is
341/// read and merged, waiting for another process as every in-place writer
342/// does. The table is read only once the lock is held, so a write another
343/// process published while this one waited is not stale here: the refusal
344/// [`TableWriteLock`] gives for it is answered by taking the lock again.
345fn lock_existing_flag_version(version_path: &Path) -> Result<TableWriteLock, TableError> {
346    TableWriteLock::acquire(version_path, 0).or_else(|_| TableWriteLock::acquire(version_path, 0))
347}

349fn merge_flags_into_existing_version(
350    path: &Path,
351    snapshot: &Table,
352    version_path: &Path,
353    merge: FlagMerge,
354) -> Result<(), FlaggingError> {
355    let mut lock =
356        lock_existing_flag_version(version_path).map_err(|source| FlaggingError::FlagVersion {
357            path: path.display().to_string(),
358            reason: format!("lock existing flag version: {source}"),
359        })?;
360    let mut table = Table::open(TableOptions::new(version_path)).map_err(|source| {
361        FlaggingError::FlagVersion {
362            path: path.display().to_string(),
363            reason: format!("open existing flag version: {source}"),
364        }
365    })?;
366    let row_flags = snapshot
367        .schema()
368        .is_some_and(|schema| schema.contains_column(FLAG_ROW_COLUMN));
369    if table.row_count() != snapshot.row_count() {
370        return Err(FlaggingError::FlagVersion {
371            path: path.display().to_string(),
372            reason: format!(
373                "existing flag version has {} rows; MAIN has {}",
374                table.row_count(),
375                snapshot.row_count()
376            ),
377        });
378    }
379    for row in 0..snapshot.row_count() {
380        let source = table_flag_matrix(snapshot, row, FLAG_COLUMN, Some(path))?;
381        let dest = table_flag_matrix(&table, row, FLAG_COLUMN, Some(path))?;
382        let merged = merge_bool_arrays(&source, &dest, merge)?;
383        table
384            .cell_accessor_mut(row, FLAG_COLUMN)
385            .map_err(|source| FlaggingError::FlagVersion {
386                path: path.display().to_string(),
387                reason: format!("open existing FLAG row {row}: {source}"),
388            })?
389            .set(Value::Array(ArrayValue::Bool(merged)))
390            .map_err(|source| FlaggingError::FlagVersion {
391                path: path.display().to_string(),
392                reason: format!("write existing FLAG row {row}: {source}"),
393            })?;
394        if row_flags {
395            let source_flag_row = table_bool(snapshot, row, FLAG_ROW_COLUMN, Some(path))?;
396            let dest_flag_row = table_bool(&table, row, FLAG_ROW_COLUMN, Some(path))?;
397            let merged_flag_row = merge_bool(source_flag_row, dest_flag_row, merge);
398            table
399                .cell_accessor_mut(row, FLAG_ROW_COLUMN)
400                .map_err(|source| FlaggingError::FlagVersion {
401                    path: path.display().to_string(),
402                    reason: format!("open existing FLAG_ROW row {row}: {source}"),
403                })?
404                .set(Value::Scalar(ScalarValue::Bool(merged_flag_row)))
405                .map_err(|source| FlaggingError::FlagVersion {
406                    path: path.display().to_string(),
407                    reason: format!("write existing FLAG_ROW row {row}: {source}"),
408                })?;
409        }
410    }
411    // Recorded before the save, so an interrupted save still tells other
412    // processes to re-read the version.
413    lock.record_write();
414    table
415        .save(TableOptions::new(version_path))
416        .map_err(|source| FlaggingError::FlagVersion {
417            path: path.display().to_string(),
418            reason: format!("save existing flag version: {source}"),
419        })?;
420    lock.release().map_err(|source| FlaggingError::FlagVersion {
421        path: path.display().to_string(),
422        reason: format!("release existing flag version: {source}"),
423    })
