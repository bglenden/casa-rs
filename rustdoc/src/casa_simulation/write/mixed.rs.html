15fn cell(session: &Simulator, scan: &crate::session::observe::Scan) -> (usize, usize) {
16    (
17        session.correlations_for_scan(scan).len(),
18        session.spectral_windows()[scan.spectral_window_id].channel_count,
19    )
20}

22pub(super) fn needed(session: &Simulator) -> bool {
23    session.scans().first().is_some_and(|first| {
24        session
25            .scans()
26            .iter()
27            .any(|scan| cell(session, scan) != cell(session, first))
28    })
29}

31pub(super) fn write(session: &Simulator) -> SimulatorResult<()> {
32    let started = Instant::now();
33    let window = ReservationWindow::open();
34    let resources = &session.options().resources;
35    // Histograms and the two live plan copies are bounded by scan count.
36    let _metadata = admit(
37        resources.host(),
38        resources.policy(),
39        &Demand {
40            phase: "the simulator's mixed-shape creation plan",
41            memory: session.scans().len() as u64 * 2048 + (64 << 10),
42        },
43    )
44    .map_err(SimulatorError::admission)?;
45    let planned = PlannedScan::of_all(session.scans())
46        .ok_or_else(|| SimulatorError::raised("only planned scans make a new MeasurementSet"))?;
47    let mut position = 0;
48    let runs = super::patch_ranges::plan(
49        session.scans().iter().map(|scan| {
50            let first = position;
51            position += scan.row_count();
52            (first..position, cell(session, scan))
53        }),
54        resources.host(),
55        resources.policy(),
56    )
57    .map_err(SimulatorError::admission)?;
58    if position == 0 {
59        return empty(session, &planned);
60    }
61    let free = free_memory(resources.host(), resources.policy());
62    let plan = plan(
63        session,
64        &runs.value,
65        position,
66        usize::try_from(free / 2).unwrap_or(usize::MAX),
67    )
68    .map_err(|error| {
69        let required = resources::minimum_writer_bytes_for(|budget| {
70            plan(session, &runs.value, position, budget)
71        })
72        .map_or(u64::MAX, |bytes| (bytes as u64).saturating_mul(2));
73        resources::refusal_or(error, resources::WRITER_PHASE, required, free)
74    })?;
75    let reservation = admit(
76        resources.host(),
77        resources.policy(),
78        &Demand {
79            phase: resources::WRITER_PHASE,
80            memory: plan.maximum_resident_bytes() as u64 + subtables::generated_bytes(&planned),
81        },
82    )
83    .map_err(SimulatorError::admission)?;
84    let generated = subtables::subtable_overrides(session, &planned)?;
85    let target = MeasurementSetCreateTarget::prepare(session.path(), true)
86        .map_err(write_error("prepare the MeasurementSet"))?;
87    let mut ms = create_staged_ms(session, target.staging_path())?;
88    let mut writer = MainWriter::start(target.staging_path(), plan.clone())?;
89    writer._reservation = reservation;
90    let (mut first_scan, mut rows_seen) = (0, 0);
91    for (rows, cell) in &runs.value {
92        if rows.is_empty() {
93            continue;
94        }
95        let mut last_scan = first_scan;
96        while rows_seen < rows.end {
97            rows_seen += session.scans()[last_scan].row_count();
98            last_scan += 1;
99        }
100        let scope = first_scan..last_scan;
101        let prediction = DataStage::for_scans(
102            session,
103            *cell,
104            resources.prediction_backend(),
105            scope.clone(),
106            false,
107        )?;
108        let shape = main_shape(session, &planned[scope.clone()], *cell);
109        let admission = resources::admit_pipeline(
110            resources,
111            &shape,
112            prediction.as_ref(),
113            plan.clone(),
114            Reservation::none(),
115        )?;
116        writer.variable = Some(*cell);
117        run_main(
118            session,
119            &planned,
120            &shape,
121            (&admission, prediction),
122            &mut writer,
123            scope,
124            rows.clone(),
125        )?;
126        first_scan = last_scan;
127    }
128    let written = writer.finish(&mut ms, position, &generated)?;
129    target
130        .commit()
131        .map_err(write_error("publish the MeasurementSet"))?;
132    tracing::info!(
133        rows = position,
134        runs = runs.value.len(),
135        peak_reserved_bytes = window.close(),
136        bytes_written = written.bytes_written,
137        total_s = started.elapsed().as_secs_f64(),
138        "simulation mixed-shape pass"
139    );
140    Ok(())
141}

143fn empty(session: &Simulator, planned: &[PlannedScan<'_>]) -> SimulatorResult<()> {
144    let resources = &session.options().resources;
145    let _reservation = admit(
146        resources.host(),
147        resources.policy(),
148        &Demand {
149            phase: resources::WRITER_PHASE,
150            memory: subtables::generated_bytes(planned) + (256 << 10),
151        },
152    )
153    .map_err(SimulatorError::admission)?;
154    let target = MeasurementSetCreateTarget::prepare(session.path(), true)
155        .map_err(write_error("prepare the empty MeasurementSet"))?;
156    let mut ms = create_staged_ms(session, target.staging_path())?;
157    let generated = subtables::subtable_overrides(session, planned)?;
158    // The ordinary empty-table save preserves declared array dimensionality.
159    // Streamed TiledShape columns need a populated shape to define a hypercube.
160    ms.save_with_column_overrides(&ColumnOverrides::for_row_count(0), &generated)
161        .map_err(|error| {
162            SimulatorError::raised(format!("save the empty MeasurementSet: {error}"))
163        })?;
164    target
165        .commit()
166        .map_err(write_error("publish the empty MeasurementSet"))
167}

169fn plan(
170    session: &Simulator,
171    runs: &[super::patch_ranges::Run],
172    rows: usize,
173    available_bytes: usize,
174) -> Result<MeasurementSetWritePlan, casa_ms::MeasurementSetWriteError> {
175    let mut counts = BTreeMap::new();
176    let mut weights = BTreeMap::new();
177    for (range, (correlations, channels)) in runs {
178        if !range.is_empty() {
179            *counts.entry((*correlations, *channels)).or_insert(0) += range.len();
180            *weights.entry(*correlations).or_insert(0) += range.len();
181        }
182    }
183    let visibility: Vec<_> = counts
184        .iter()
185        .map(|(&(c, h), &n)| MeasurementSetArrayShapePlan::visibility(c, h, n, session.telescope()))
186        .collect();
187    let category = counts
188        .iter()
189        .map(|(&(c, h), &n)| {
190            MeasurementSetArrayShapePlan::flag_category(c, h, 3, n, session.telescope())
191        })
192        .collect();
193    let weight: Vec<_> = weights
194        .iter()
195        .map(|(&c, &n)| MeasurementSetArrayShapePlan::weight(c, n, session.telescope()))
196        .collect();
197    let column = |name: &str, value_type, shapes, storage_manager| MeasurementSetArrayColumnPlan {
198        name: name.into(),
199        value_type,
200        shapes,
201        storage_manager,
202    };
203    let tiled = MeasurementSetColumnStorage::TiledShape;
204    let columns = vec![
205        column(
206            "DATA",
207            StreamedTiledShapeValueType::Complex32,
208            visibility.clone(),
209            tiled,
210        ),
211        column("FLAG", StreamedTiledShapeValueType::Bool, visibility, tiled),
212        column(
213            "FLAG_CATEGORY",
214            StreamedTiledShapeValueType::Bool,
215            category,
216            tiled,
217        ),
218        column(
219            "UVW",
220            StreamedTiledShapeValueType::Float64,
221            vec![MeasurementSetArrayShapePlan {
222                cell_shape: vec![3],
223                row_count: rows,
224                tile_shape: vec![3, 1024],
225            }],
226            MeasurementSetColumnStorage::TiledColumn,
227        ),
228        column(
229            "WEIGHT",
230            StreamedTiledShapeValueType::Float32,
231            weight.clone(),
232            tiled,
233        ),
234        column("SIGMA", StreamedTiledShapeValueType::Float32, weight, tiled),
235    ];
236    MeasurementSetWritePlan::variable_array_creation(
237        rows,
238        columns,
239        standard_main_scalar_column_plans(),
240        MeasurementSetWriteResources {
241            available_bytes,
242            maximum_live_batches: 2,
243            tiled_column_buffer_bytes: STREAMING_TILED_COLUMN_BUFFER_BYTES,
244        },
245    )
246}

248pub(super) fn send(
249    writer: &mut MainWriter,
250    mut batch: VisibilityColumns,
251    (correlations, channels): (usize, usize),
252) -> SimulatorResult<()> {
253    let error = write_error("mixed-shape MAIN cells");
254    let samples = correlations * channels;
255    let mut flags = vec![false; samples];
256    let mut weights = vec![0.; correlations];
257    for row in 0..batch.row_count() {
258        writer
259            .session
260            .push_complex32_row(
261                "DATA",
262                &[correlations, channels],
263                &batch.data[row * samples..(row + 1) * samples],
264            )
265            .map_err(&error)?;
266        flags.fill(batch.flag_row[row]);
267        writer
268            .session
269            .push_bool_row("FLAG", &[correlations, channels], &flags)
270            .map_err(&error)?;
271        writer
272            .session
273            .push_undefined_row("FLAG_CATEGORY")
274            .map_err(&error)?;
275        writer
276            .session
277            .push_f64_row("UVW", &[3], &batch.uvw[row])
278            .map_err(&error)?;
279        for (name, values) in [("WEIGHT", &batch.weight), ("SIGMA", &batch.sigma)] {
280            weights.fill(values.get(row).copied().unwrap_or(1.));
281            writer
282                .session
283                .push_f32_row(name, &[correlations], &weights)
284                .map_err(&error)?;
285        }
286    }
287    batch.data.clear();
288    batch.flag_row.clear();
289    batch.uvw.clear();
290    batch.weight.clear();
291    batch.sigma.clear();
292    writer.recycled = Some(batch);
293    Ok(())
294}
