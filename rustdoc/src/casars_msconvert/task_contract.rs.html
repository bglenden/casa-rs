16/// Stable protocol name advertised by `msconvert --protocol-info`.
17pub const MSCONVERT_TASK_PROTOCOL_NAME: &str = "casars_msconvert_task";
18/// Protocol version advertised by `msconvert --protocol-info`.
19pub const MSCONVERT_TASK_PROTOCOL_VERSION: u32 = 1;
20
21/// msconvert's protocol descriptor.
22pub fn msconvert_protocol_descriptor() -> ProviderProtocolDescriptor {
23    ProviderProtocolDescriptor::new(
24        MSCONVERT_TASK_PROTOCOL_NAME,
25        MSCONVERT_TASK_PROTOCOL_VERSION,
26        ProviderSurfaceKind::Task,
27        env!("CARGO_PKG_VERSION"),
28    )
29}
30
31/// msconvert's schema bundle in the shared envelope; the catalog's
32/// `msconvert` surface defines the request's parameters.
33pub fn msconvert_task_schema_bundle() -> TaskProviderContract {
34    let request_schema = schema_for!(MsconvertTaskRequest);
35    let result_schema = schema_for!(MsconvertTaskResult);
36    TaskProviderContract {
37        protocol: msconvert_protocol_descriptor(),
38        semantic: TaskSemanticContract {
39            request_schema: request_schema.clone(),
40            result_schema: result_schema.clone(),
41            operations: vec![TaskOperationDescriptor {
42                name: "run".to_string(),
43                request_kind: "run".to_string(),
44                result_kind: Some("run".to_string()),
45            }],
46            front_end: None,
47        },
48        components: merged_components([&request_schema, &result_schema]),
49        annotations: serde_json::json!({}),
50        projections: ProviderProjectionMetadata {
51            cli: Some(ProviderCliProjection {
52                machine_actions: ProviderCliMachineActions {
53                    json_schema: Some("--json-schema".to_string()),
54                    protocol_info: Some("--protocol-info".to_string()),
55                    json_run: Some("--json-run <SOURCE>".to_string()),
56                },
57            }),
58            python: None,
59        },
60        parameter_surfaces: vec![
61            builtin_surface_bundle("msconvert")
62                .expect("built-in msconvert parameter surface must remain valid"),
63        ],
64        domain_schemas: TaskProviderSchemas {
65            request_schema,
66            result_schema,
67            additional: NoAdditionalProviderSchemas {},
68        },
69    }
70}

72/// msconvert's request: the catalog's parameters of one run, as the front
73/// end resolves them over their defaults.
74#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
75pub struct MsconvertTaskRequest {
76    /// The store to convert: a MeasurementSet v2 or an MSv4 processing set.
77    pub vis: PathBuf,
78    /// The new store; empty for one beside `vis`, named after it as XRADIO
79    /// names stores: `.ps.zarr` for an MSv4, `.ms` for an MSv2. An MSv4
80    /// output not ending in `ps.zarr` gains `.ps.zarr`, as XRADIO's
81    /// converter names it.
82    #[serde(default)]
83    pub outputvis: PathBuf,
84    /// `msv2` or `msv4`; empty for the other format than `vis`.
85    #[serde(default)]
86    pub outformat: String,
87    /// The data groups to convert, the first carrying flags, weights and
88    /// UVW; empty for every group, `base` first.
89    #[serde(default)]
90    pub datagroups: Vec<String>,
91    /// MSv2 into MSv4: further keys the MSv2 is partitioned by, XRADIO's
92    /// `partition_scheme`: `FIELD_ID`, `SCAN_NUMBER`.
93    #[serde(default)]
94    pub partitionscheme: Vec<String>,
95    /// MSv4 output: convert the pointing, XRADIO's `with_pointing`.
96    #[serde(default = "default_true")]
97    pub withpointing: bool,
98    /// MSv2 output: what a padded grid cell becomes, `flagged` or
99    /// `omitted`.
100    #[serde(default = "default_padding")]
101    pub padding: String,
102    /// MSv4 output: the chunk shape `[time, baseline, frequency,
103    /// polarization]`, 0 for a whole axis; empty for XRADIO's default.
104    #[serde(default)]
105    pub chunks: Vec<i64>,
106    /// MSv4 output: the payload codec, `blosc`, `zstd` or `none`.
107    #[serde(default = "default_codec")]
108    pub codec: String,
109    /// MSv4 output: partitions converted at once; 0 for as many as the
110    /// cores and the memory allow.
111    #[serde(default)]
112    pub nthreads: i64,
113    /// Replace an existing MeasurementSet at `outputvis`.
114    #[serde(default)]
115    pub overwrite: bool,
116}
117
118fn default_true() -> bool {
119    true
120}
121
122fn default_padding() -> String {
123    "flagged".to_string()
124}
125
126fn default_codec() -> String {
127    "blosc".to_string()
128}

130/// What one conversion did.
131#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
132pub struct MsconvertTaskResult {
133    /// The store converted.
134    pub vis: String,
135    /// The store written.
136    pub outputvis: String,
137    /// The format of `vis`: `msv2` or `msv4`.
138    pub informat: String,
139    /// The format of `outputvis`: `msv2` or `msv4`.
140    pub outformat: String,
141    /// MSv4 output: the chunk shape asked of every dataset, each axis then
142    /// clipped to the dataset's extent (0 for a whole axis); `None` for
143    /// XRADIO's, sized for each dataset.
144    pub chunks: Option<[u64; 4]>,
145    /// What the output does not carry or states by default, as sentences.
146    pub losses: Vec<String>,
147    /// Partitions converted (MSv4 datasets).
148    pub partitions: usize,
149    /// Samples read, padding included.
150    pub samples: usize,
151    /// MSv2 output: rows written.
152    pub rows: usize,
153    /// MSv4 output: payload elements encoded.
154    pub elements: usize,
155    /// Wall time of the conversion.
156    pub elapsed_ns: u64,
157}

159#[cfg(test)]
160mod tests {
161    use super::*;
162
163    #[test]
164    fn the_schema_bundle_carries_the_msconvert_surface() {
165        let bundle = msconvert_task_schema_bundle();
166        assert_eq!(bundle.protocol.protocol_name, MSCONVERT_TASK_PROTOCOL_NAME);
167        assert_eq!(bundle.parameter_surfaces.len(), 1);
168        assert_eq!(bundle.parameter_surfaces[0].surface.id(), "msconvert");
169        bundle.validate().expect("valid msconvert provider");
170    }
