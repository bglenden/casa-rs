2//! The bound of a catalog's cell cache and what the catalog holds at any
3//! bound, so a run can size the cache to the memory its workers leave
4//! (`cf_resident_mb = auto`, #449).
5
6use std::sync::Arc;
7use std::sync::atomic::{AtomicUsize, Ordering};
8
9use super::ARC_CELL_BYTES;
10#[cfg(doc)]
11use crate::convolution::CellHold;
12
13/// The bound of a catalog's cell cache, in bytes of loaded cells: shared,
14/// so a run may raise it once it knows the memory its workers leave
15/// (`cf_resident_mb = auto`, #449). A fetch evicts down to the bound it
16/// reads then.
17#[derive(Clone, Debug)]
18pub struct CacheBound(Arc<AtomicUsize>);
19
20impl CacheBound {
21    pub(super) fn new(bytes: usize) -> Self {
22        Self(Arc::new(AtomicUsize::new(bytes)))
23    }
24
25    /// Bytes of loaded cells the cache keeps at most.
26    #[must_use]
27    pub fn bytes(&self) -> usize {
28        self.0.load(Ordering::Relaxed)
29    }
30
31    /// Raise the bound to `bytes`; a smaller value leaves it.
32    pub fn raise(&self, bytes: usize) {
33        self.0.fetch_max(bytes, Ordering::Relaxed);
34    }
35}
36
37/// What a catalog holds from its open on, as a function of its cache's
38/// bound and the workers fetching its cells ([`Self::bytes`]).
39#[derive(Clone, Copy, Debug, PartialEq, Eq)]
40pub struct CacheCharge {
41    /// What it keeps of its index (each cell's two headers and paths, the
42    /// groups, axes, routes, corrections and the cache's group slots, at
43    /// their capacities), its Mueller routing and its image correction.
44    pub(super) kept: u64,
45    /// Opening's stage: it reads one header at a time and lists one axis's
46    /// values before it is reduced.
47    pub(super) opening: u64,
48    /// Every group's cells.
49    pub(super) every: u64,
50    /// The largest group's cells.
51    pub(super) largest: u64,
52    pub(super) groups: u64,
53    /// What one holder reads to build one group: the planes, each read as an
54    /// image array and its copy, with that read's records.
55    pub(super) reading: u64,
56}
57
58impl CacheCharge {
59    /// Bytes the catalog holds at most at a cache bound of `bound` bytes
60    /// with `holders` workers fetching cells: what it keeps and the larger
61    /// of its two stages, opening and serving. Serving holds its cells and,
62    /// for each holder, one group's read.
63    ///
64    /// When every group fits the bound nothing is evicted and each group is
65    /// loaded once, so the cells are at most every group (#738). Otherwise
66    /// they are the bound and one group past it, which a fetch inserts
67    /// before the cache evicts, and for each holder the largest group twice.
68    /// A CPU residual holds a gridding cell and a prediction cell (#449),
69    /// and either keeps both alive after eviction, or one beside the group
70    /// it builds outside the cache's lock for the other (#744): a
71    /// [`CellHold`] releases its own cell before it fetches (#773 review).
72    /// Every loaded cell is also its [`Arc`] (#745 review).
73    #[must_use]
74    pub const fn bytes(&self, holders: usize, bound: usize) -> u64 {
75        let holders = holders as u64;
76        let (taps, pairs) = if self.every <= bound as u64 {
77            (self.every, self.groups)
78        } else {
79            (
80                bound as u64 + (2 * holders + 1) * self.largest,
81                self.groups + 2 * holders + 1,
82            )
83        };
84        let serving = taps + pairs * 2 * ARC_CELL_BYTES as u64 + holders * self.reading;
85        self.kept
86            + if self.opening > serving {
87                self.opening
88            } else {
89                serving
90            }
91    }
92
93    /// Bytes of every group's cells: a bound this large keeps them all.
94    #[must_use]
95    pub const fn every_bytes(&self) -> u64 {
96        self.every
97    }
98
99    /// The largest bound, from `bound` up, whose charge on `holders`
100    /// exceeds the charge at `bound` by at most `spare` bytes: every group,
101    /// when that fits, which needs no group past the bound nor the holders'
102    /// evicted groups; otherwise `bound + spare`, since below every group
103    /// the charge grows with the bound byte for byte.
104    #[must_use]
105    pub fn bound_within(&self, holders: usize, bound: usize, spare: u64) -> usize {
106        let every = usize::try_from(self.every).unwrap_or(usize::MAX);
107        if bound >= every
108            || self.bytes(holders, every) <= self.bytes(holders, bound).saturating_add(spare)
109        {
110            return bound.max(every);
111        }
112        bound.saturating_add(usize::try_from(spare).unwrap_or(usize::MAX))
113    }
114
115    /// Raise `bound` as far as `spare` allows on `holders`
116    /// ([`Self::bound_within`]) and return the charge at the raised bound.
117    /// `spare` gives up what the raise costs, or takes back what it saves:
118    /// every cell can cost less than a bound just below it, which also
119    /// holds a group past the bound and the holders' evicted groups (#781
120    /// review).
121    pub fn raise_within(&self, holders: usize, bound: &CacheBound, spare: &mut u64) -> u64 {
122        let before = self.bytes(holders, bound.bytes());
123        bound.raise(self.bound_within(holders, bound.bytes(), *spare));
124        let after = self.bytes(holders, bound.bytes());
125        *spare = if after >= before {
126            spare.saturating_sub(after - before)
127        } else {
128            spare.saturating_add(before - after)
129        };
130        after
131    }
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137
138    /// A catalog of a hundred 10 MB groups whose serving stage outweighs
139    /// opening.
140    const CHARGE: CacheCharge = CacheCharge {
141        kept: 1_000,
142        opening: 0,
143        every: 1_000_000_000,
144        largest: 10_000_000,
145        groups: 100,
146        reading: 5_000,
147    };
148
149    /// The bound grows into the spare byte for byte below every group, and
150    /// takes every group, with no group past it nor any holder's evicted
151    /// one, as soon as their charge fits; it never costs more than the
152    /// spare, and never falls.
153    #[test]
154    fn a_bound_grows_into_its_spare_and_never_past_it() {
155        let holders = 4;
156        let floor = 100_000_000;
157        let at_floor = CHARGE.bytes(holders, floor);
158        // Within the spare below every group: the bound rises by the spare.
159        let spare = 300_000_000;
160        let raised = CHARGE.bound_within(holders, floor, spare);
161        assert_eq!(raised, floor + 300_000_000);
162        assert_eq!(CHARGE.bytes(holders, raised), at_floor + spare);
163        // Every group's charge drops the group past the bound and the
164        // holders' groups, so it fits a spare smaller than the gap to every.
165        let all = CHARGE.bytes(holders, CHARGE.every as usize);
166        assert!(all < at_floor + (CHARGE.every - floor as u64));
167        let spare = all - at_floor;
168        assert_eq!(
169            CHARGE.bound_within(holders, floor, spare),
170            CHARGE.every as usize
171        );
172        // A byte short of it, the bound grows linearly instead.
173        let short = CHARGE.bound_within(holders, floor, spare - 1);
174        assert_eq!(short, floor + (spare - 1) as usize);
175        assert!(CHARGE.bytes(holders, short) < at_floor + spare);
176        // No spare, no growth; a bound past every group stays.
177        assert_eq!(CHARGE.bound_within(holders, floor, 0), floor);
178        assert_eq!(CHARGE.bound_within(holders, usize::MAX, 0), usize::MAX);
179    }
180
181    /// #781 review: a bound within a group of every cell takes every cell
182    /// with no spare at all, and every cell costs less than that bound,
183    /// which also holds a group past it and the holders' evicted groups.
184    /// The saving goes back to the spare; a raise within a tight spare
185    /// spends it exactly.
186    #[test]
187    fn a_raise_spends_what_it_costs_and_returns_what_it_saves() {
188        let holders = 4;
189        let every = CHARGE.every as usize;
190        let near = every - CHARGE.largest as usize;
191        let at_near = CHARGE.bytes(holders, near);
192        let at_every = CHARGE.bytes(holders, every);
193        assert!(
194            at_every < at_near,
195            "every cell costs less than the bound below it"
196        );
197        assert_eq!(CHARGE.bound_within(holders, near, 0), every);
198        let bound = CacheBound::new(near);
199        let mut spare = 0;
200        assert_eq!(CHARGE.raise_within(holders, &bound, &mut spare), at_every);
201        assert_eq!(bound.bytes(), every);
202        assert_eq!(spare, at_near - at_every, "the saving is spare again");
203        // A tight spare below every cell: spent exactly.
204        let floor = 100_000_000;
205        let bound = CacheBound::new(floor);
206        let mut spare = 50_000_000;
207        let charged = CHARGE.raise_within(holders, &bound, &mut spare);
208        assert_eq!(bound.bytes(), floor + 50_000_000);
209        assert_eq!(charged, CHARGE.bytes(holders, floor) + 50_000_000);
210        assert_eq!(spare, 0);
211    }
212
213    /// Two catalogs share one spare first come first served: the second
214    /// takes what the first left, and together they cost at most the spare
215    /// beyond their charges at their floors.
216    #[test]
217    fn catalogs_share_one_spare() {
218        let holders = 2;
219        let floor = 100_000_000;
220        let (first, second) = (CacheBound::new(floor), CacheBound::new(floor));
221        let opened = 2 * CHARGE.bytes(holders, floor);
222        let initial = 600_000_000;
223        let mut spare = initial;
224        let charged = CHARGE.raise_within(holders, &first, &mut spare)
225            + CHARGE.raise_within(holders, &second, &mut spare);
226        assert_eq!(first.bytes(), floor + 600_000_000, "the first takes all");
227        assert_eq!(second.bytes(), floor, "and leaves none");
228        assert_eq!(charged, opened + initial);
229        assert_eq!(spare, 0);
230        // With room for every cell of the first, the second takes the rest.
231        let (first, second) = (CacheBound::new(floor), CacheBound::new(floor));
232        let mut spare = 2_000_000_000;
233        let charged = CHARGE.raise_within(holders, &first, &mut spare)
234            + CHARGE.raise_within(holders, &second, &mut spare);
235        assert_eq!(first.bytes(), CHARGE.every as usize);
236        assert_eq!(second.bytes(), CHARGE.every as usize);
237        assert_eq!(charged + spare, opened + 2_000_000_000);
238    }
239
240    /// A raised bound is the one every clone of it reads; a lower value
241    /// leaves it.
242    #[test]
243    fn a_raised_bound_is_shared_and_never_lowered() {
244        let bound = CacheBound::new(10);
245        let shared = bound.clone();
246        bound.raise(30);
247        assert_eq!(shared.bytes(), 30);
248        shared.raise(20);
249        assert_eq!(bound.bytes(), 30);
250    }
