43pub(crate) fn run(task: &TaskInvocation<'_>) -> Result<TaskSuccess, TaskFailure> {
44    let outfile = task.text("outfile")?.unwrap_or_default();
45    let overwrite = task.flag("overwrite")?;
46    if outfile.is_empty() {
47        return Err(TaskFailure::validation("outfile is empty.").at("outfile"));
48    }
49    let output = Path::new(&outfile);
50    if output.exists() && !overwrite {
51        return Err(TaskFailure::execution(format!("{outfile} exists.")).at("outfile"));
52    }
53    let infile = task.input_path("infile")?;
54    let selection = MsSelection {
55        selectdata: true,
56        antenna: selector(task, "antenna")?.map(|antenna| format!("{antenna}&&&")),
57        field: selector(task, "field")?,
58        spw: selector(task, "spw")?,
59        scan: selector(task, "scan")?,
60        intent: selector(task, "intent")?,
61        ..MsSelection::default()
62    };
63    let ms = MeasurementSet::open(&infile)
64        .map_err(|error| TaskFailure::execution(error.to_string()).at("infile"))?;
65    let selected = resolve_casa_calibration_selection(&ms, &selection).map_err(|error| {
66        TaskFailure::execution(format!("Error in data selection specification: {error}"))
67    })?;
68    let tables = options::priors(task)?;
69    let priors = if tables.is_empty() {
70        None
71    } else {
72        let plan = plan_apply(
73            &ms,
74            &ApplyPlanRequest {
75                selection,
76                apply_mode: ApplyMode::CalFlag,
77                parang: false,
78                calibration_tables: tables,
79            },
80        )
81        .map_err(|_| options::setapply_error("applytable"))?;
82        Some(
83            SolvePriorApply::single_dish(&ms, plan)
84                .map_err(|_| options::setapply_error("applytable"))?,
85        )
86    };
87    let controls = Controls {
88        radius: options::radius(task)?,
89        smooth: task.flag("smooth")?,
90        solve_mode: GainSolveMode::AmplitudePhase,
91        correlation_dependent_flags: false,
92    };
93    if overlaps_measurement_set(&ms, output).map_err(TaskFailure::execution)? {
94        return Err(TaskFailure::execution(
95            "The MeasurementSet and output calibration table overlap.",
96        )
97        .at("outfile"));
98    }
99    let mode = if overwrite {
100        CalibrationOutputMode::Replace
101    } else {
102        CalibrationOutputMode::Create
103    };
104    let _output_locks = lock_solve_output(output, mode)
105        .map_err(|error| TaskFailure::execution(error.to_string()).at("outfile"))?;
106    let (count, _) = generate(
107        &ms,
108        &selected,
109        priors.as_ref(),
110        output,
111        mode,
112        controls,
113        None,
114    )?;
115    TaskSuccess::new(
116        CasaValue::None,
117        &serde_json::json!({"caltable": outfile, "row_count": count}),
118    )
119}

121pub(super) fn generate(
122    ms: &MeasurementSet,
123    selection: &ResolvedMsSelection,
124    priors: Option<&SolvePriorApply>,
125    output: &Path,
126    mode: CalibrationOutputMode,
127    controls: Controls,
128    activity: Option<&mut ToolSolveActivity>,
129) -> Result<(Option<usize>, Option<SolveStats>), TaskFailure> {
130    let integrations = selected_integrations(ms, selection).map_err(solve_failure)?;
131    let references =
132        footprint::select(ms, &integrations, controls.radius).map_err(solve_failure)?;
133    let channels = spectral::channels(ms, selection, &references)
134        .map_err(|message| TaskFailure::execution(message).at("spw"))?;
135    let mut intervals = activity.as_ref().map(|_| GainIntervals::default());
136    let samples = solve(
137        ms,
138        references,
139        priors,
140        controls,
141        &channels,
142        intervals.as_mut(),
143    )
144    .map_err(solve_failure)?;
145    let stats = intervals
146        .map(|intervals| intervals.finish(ms, activity))
147        .transpose()
148        .map_err(solve_failure)?;
149    let count = samples
150        .map(|samples| write(ms, output, mode, samples, &channels))
151        .transpose()
152        .map_err(solve_failure)?;
153    Ok((count, stats))
154}

156fn selected_integrations(
157    ms: &MeasurementSet,
158    selection: &ResolvedMsSelection,
159) -> Result<Vec<Integration>, String> {
160    let descriptions = ms.data_description().map_err(error_text)?;
161    let windows = (0..descriptions.row_count())
162        .map(|row| descriptions.spectral_window_id(row).map(|id| id as usize))
163        .collect::<Result<Vec<_>, _>>()
164        .map_err(error_text)?;
165    let rows = selection.row_indices().collect::<Vec<_>>();
166    edges::read_integrations(ms, &rows, &windows)
167}

169fn solve<'a>(
170    ms: &MeasurementSet,
171    references: Vec<&'a Integration>,
172    priors: Option<&SolvePriorApply>,
173    controls: Controls,
174    channels: &BTreeMap<usize, usize>,
175    mut intervals: Option<&mut GainIntervals>,
176) -> Result<Option<Vec<GainSample<'a>>>, String> {
177    let mut groups = BTreeMap::<_, Vec<_>>::new();
178    for row in references {
179        groups
180            .entry((row.window, row.field, row.antenna))
181            .or_default()
182            .push(row);
183    }
184    let spectral = ms.spectral_window().map_err(error_text)?;
185    let antennas = ms.antenna().map_err(error_text)?;
186    let descriptions = ms.data_description().map_err(error_text)?;
187    let pol = ms.polarization().map_err(error_text)?;
188    let correlations = (0..descriptions.row_count())
189        .map(|row| {
190            descriptions
191                .polarization_id(row)
192                .and_then(|id| pol.corr_type(id as usize))
193        })
194        .collect::<Result<Vec<_>, _>>()
195        .map_err(error_text)?;
196    let mut results = Vec::new();
197    let mut has_solving_data = false;
198    for ((window, _, antenna), mut group) in groups {
199        group.sort_by(|first, second| first.time.total_cmp(&second.time));
200        let channel = channels[&window];
201        let frequency = spectral.chan_freq(window).map_err(error_text)?[channel];
202        let radius = normalization::radius(
203            controls.radius,
204            &[frequency],
205            antennas
206                .dish_diameter(antenna as usize)
207                .map_err(error_text)?,
208        )?;
209        let mut samples = Vec::new();
210        for row in group {
211            let sample = read_sample(ms, row, priors, &correlations, channel, controls)?;
212            if let Some(intervals) = intervals.as_deref_mut() {
213                intervals.observe(row, sample.as_ref());
214            }
215            samples.extend(sample);
216        }
217        has_solving_data |= !samples.is_empty();
218        if samples.len() < 100 {
219            continue;
220        }
221        normalization::normalize(&mut samples, radius, controls.smooth)?;
222        results.extend(samples);
223    }
224    results.sort_by(|first, second| {
225        let left = first.integration;
226        let right = second.integration;
227        left.description
228            .cmp(&right.description)
229            .then_with(|| left.time.total_cmp(&right.time))
230            .then_with(|| left.antenna.cmp(&right.antenna))
231            .then_with(|| left.field.cmp(&right.field))
232    });
233    Ok(has_solving_data.then_some(results))
234}

236type IntegrationKey = (i32, i32, i32, i32, u64);
237
238#[derive(Default)]
239struct GainIntervals(BTreeMap<IntegrationKey, [BTreeSet<i32>; 2]>);
240
241impl GainIntervals {
242    fn observe(&mut self, row: &Integration, sample: Option<&GainSample<'_>>) {
243        let key = (
244            row.window as i32,
245            row.observation,
246            row.scan,
247            row.field,
248            row.time.to_bits(),
249        );
250        let hands = self.0.entry(key).or_default();
251        if let Some(sample) = sample {
252            for (index, hand) in hands.iter_mut().take(sample.data.shape()[0]).enumerate() {
253                if !sample.flags[[index, 0]] {
254                    hand.insert(row.antenna);
255                }
256            }
257        }
258    }
259
260    fn finish(
261        self,
262        ms: &MeasurementSet,
263        activity: Option<&mut ToolSolveActivity>,
264    ) -> Result<SolveStats, String> {
265        let antenna_count = ms.antenna().map_err(error_text)?.row_count() as i32;
266        let window_count = ms.spectral_window().map_err(error_text)?.row_count() as i32;
267        let mut stats = SolveStatsBuilder::new(2, 0..window_count, &(0..antenna_count).collect());
268        let no_cross_correlations = BTreeSet::new();
269        for ((spw_id, _, _, _, _), hands) in self.0 {
270            let attempted = hands.iter().any(|hand| !hand.is_empty());
271            stats.interval(&IntervalStages {
272                spw_id,
273                antennas: &no_cross_correlations,
274                expected_window: false,
275                attempted_interval: Some(attempted),
276                with_data: hands.into(),
277                constrained: BTreeSet::new(),
278            });
279            stats.record_success(spw_id, attempted);
280        }
281        stats.track_activity(activity);
282        Ok(stats.finish())
283    }
284}

286fn read_sample<'a>(
287    ms: &MeasurementSet,
288    row: &'a Integration,
289    priors: Option<&SolvePriorApply>,
290    correlations: &[Vec<i32>],
291    channel: usize,
292    controls: Controls,
293) -> Result<Option<GainSample<'a>>, String> {
294    let main = ms.main_table();
295    let source = crate::model_columns::observed_column(ms);
296    let data = main
297        .cell_accessor(row.row, source)
298        .and_then(|cell| cell.array())
299        .map_err(error_text)?;
300    let mut flags = main
301        .cell_accessor(row.row, "FLAG")
302        .and_then(|cell| cell.array())
303        .map_err(error_text)?
304        .clone();
305    if row.flagged
306        && let ArrayValue::Bool(flags) = &mut flags
307    {
308        flags.fill(true);
309    }
310    let types = correlations
311        .get(row.description as usize)
312        .ok_or("Invalid DATA_DESC_ID")?;
313    let (data, flags) = if let Some(priors) = priors {
314        let applied = priors
315            .correct_single_dish(row.row, types, data, flags)
316            .map_err(error_text)?
317            .ok_or("Prior application omitted a selected gain integration.")?;
318        (applied.corrected_data, applied.flags)
319    } else {
320        (data.clone(), flags)
321    };
322    let data = match data {
323        ArrayValue::Float32(values) => values.mapv(|value| Complex32::new(value, 0.)),
324        ArrayValue::Complex32(values) => values,
325        _ => return Err("Invalid single-dish visibility shape.".into()),
326    };
327    let ArrayValue::Bool(flags) = flags else {
328        return Err("FLAG is not Boolean.".into());
329    };
330    if data.ndim() != 2 || data.shape() != flags.shape() || channel >= data.shape()[1] {
331        return Err("Visibility data and flags do not cover the selected gain channel.".into());
332    }
333    let (mut data, flags) = if data.shape()[1] == 1 {
334        (data, flags)
335    } else {
336        (
337            data.select(Axis(1), &[channel]),
338            flags.select(Axis(1), &[channel]),
339        )
340    };
341    let usable = if controls.correlation_dependent_flags {
342        flags.iter().any(|&flagged| !flagged)
343    } else {
344        flags.iter().all(|&flagged| !flagged)
345    };
346    if !usable {
347        return Ok(None);
348    }
349    for (value, &flagged) in data.iter_mut().zip(flags.iter()) {
350        if flagged {
351            *value = Complex32::default();
352        } else {
353            match controls.solve_mode {
354                GainSolveMode::Amplitude => *value = Complex32::new(value.norm(), 0.),
355                GainSolveMode::Phase => {
356                    let magnitude = value.norm();
357                    if magnitude > 0. {
358                        *value /= magnitude;
359                    }
360                }
361                GainSolveMode::AmplitudePhase => {}
362            }
363        }
364    }
365    Ok(Some(GainSample {
366        integration: row,
367        data,
368        flags,
369    }))
370}

372fn write(
373    ms: &MeasurementSet,
374    output: &Path,
375    mode: CalibrationOutputMode,
376    samples: Vec<GainSample<'_>>,
377    channels: &BTreeMap<usize, usize>,
378) -> Result<usize, String> {
379    let count = samples.len();
380    let destination = crate::output::CalibrationOutput::prepare_single_dish_solve(output, mode)
381        .map_err(error_text)?;
382    let mut writer = CalibrationTableWriter::staged(
383        ms,
384        CalibrationTableDescriptor {
385            output,
386            schema: new_calibration_table_schema(true),
387            subtype: SUBTYPE,
388            parameter_type: Some("Complex"),
389            measurement_set_name: ms
390                .path()
391                .map_or_else(|| "none".into(), |path| path.display().to_string()),
392            include_polarization_basis: true,
393            time_extra_precision_column: None,
394        },
395        destination,
396        mode,
397    );
398    let mut windows = BTreeSet::new();
399    for sample in samples {
400        windows.insert(sample.integration.window);
401        writer.append(gain_record(sample)?).map_err(error_text)?;
402    }
403    write_standard_calibration_subtables(ms, writer.output_path(), None)?;
404    spectral::write_axis(
405        &writer.output_path().join("SPECTRAL_WINDOW"),
406        channels,
407        &windows,
408    )?;
409    writer.save_main().map_err(error_text)?;
410    Ok(count)
411}

413fn gain_record(sample: GainSample<'_>) -> Result<RecordValue, String> {
414    let row = sample.integration;
415    let shape = sample.data.shape();
416    if shape.len() != 2 || !(1..=2).contains(&shape[0]) {
417        return Err("Double-circle gain needs one or two receptors.".into());
418    }
419    let (receptors, channels) = (shape[0], shape[1]);
420    let dimensions = IxDyn(&[2, channels]);
421    let mut values = ArrayD::zeros(dimensions.clone());
422    let mut flags = ArrayD::from_elem(dimensions.clone(), true);
423    let mut error = ArrayD::zeros(dimensions.clone());
424    let mut snr = ArrayD::zeros(dimensions);
425    for receptor in 0..receptors {
426        values
427            .index_axis_mut(Axis(0), receptor)
428            .assign(&sample.data.index_axis(Axis(0), receptor));
429        flags
430            .index_axis_mut(Axis(0), receptor)
431            .assign(&sample.flags.index_axis(Axis(0), receptor));
432        error.index_axis_mut(Axis(0), receptor).fill(0.1);
433        snr.index_axis_mut(Axis(0), receptor).fill(1.);
434    }
435    let int = |name, value| RecordField::new(name, Value::Scalar(ScalarValue::Int32(value)));
436    let float = |name, value| RecordField::new(name, Value::Scalar(ScalarValue::Float64(value)));
437    let array = |name, value| RecordField::new(name, Value::Array(value));
438    Ok(RecordValue::new(vec![
439        float(COL_TIME, row.time),
440        float(COL_INTERVAL, 0.),
441        int(COL_ANTENNA1, row.antenna),
442        int(COL_ANTENNA2, -1),
443        int(COL_FIELD_ID, row.field),
444        int(COL_SCAN_NUMBER, row.scan),
445        int(COL_OBSERVATION_ID, row.observation),
446        int(COL_SPECTRAL_WINDOW_ID, row.window as i32),
447        array(COL_CPARAM, ArrayValue::Complex32(values)),
448        array(COL_FLAG, ArrayValue::Bool(flags)),
449        array(COL_PARAMERR, ArrayValue::Float32(error)),
450        array(COL_SNR, ArrayValue::Float32(snr)),
451    ]))
452}

454fn solve_failure(error: String) -> TaskFailure {
455    tracing::error!(%error, "Single-dish gain solve failed");
456    TaskFailure::execution("Error in Calibrater::solve.")
457}
