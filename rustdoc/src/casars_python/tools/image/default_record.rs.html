12/// ImageFactory makes the CoordinateUtil record before centering its axes.
13pub(super) fn for_shape(shape: &[usize], linear: bool) -> PyResult<RecordValue> {
14    let coordinates = defaults::coordinate_system_for_shape(shape, linear)
15        .with_obs_info(crate::tools::coordsys::default_image_obs_info());
16    let mut record = coordinates.to_record();
17    if let Some(index) = coordinates.find_coordinate(casa_coordinates::CoordinateType::Spectral) {
18        let field = format!("spectral{index}");
19        if let Some(Value::Record(spectral)) = record.get_mut(&field) {
20            spectral.upsert("conversion", Value::Record(spectral_conversion()));
21        }
22    }
23    for index in 0..coordinates.n_coordinates() {
24        if let Some(Value::Record(linear)) = record.get_mut(&format!("linear{index}")) {
25            linear.remove("coordinate_type");
26            linear.remove("naxes");
27        }
28    }
29    for (axis, &length) in shape.iter().enumerate() {
30        CoordinateSystem::set_reference_pixel_in_record(&mut record, axis, (length / 2) as f64)
31            .map_err(raised)?;
32    }
33    Ok(record)
34}
35
36fn spectral_conversion() -> RecordValue {
37    let mut epoch = epoch_to_record(&MEpoch::from_mjd(0.0, EpochRef::UTC));
38    // LAST is casacore's placeholder for an unset conversion frame.
39    epoch.upsert("refer", Value::Scalar(ScalarValue::String("LAST".into())));
40    RecordValue::new(vec![
41        RecordField::new("epoch", Value::Record(epoch)),
42        RecordField::new(
43            "direction",
44            Value::Record(direction_to_record(&MDirection::from_angles(
45                0.0,
46                std::f64::consts::FRAC_PI_2,
47                DirectionRef::J2000,
48            ))),
49        ),
50        RecordField::new(
51            "position",
52            Value::Record(position_to_record(&MPosition::new_itrf(0.0, 0.0, 0.0))),
53        ),
54        RecordField::new("system", Value::Scalar(ScalarValue::String("LSRK".into()))),
55    ])
56}
