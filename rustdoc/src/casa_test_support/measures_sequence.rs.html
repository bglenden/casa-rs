2//! C++ casacore's ordered frequency conversion: CASA's
3//! `MSUtil::getFreqRangeInSpw` loop, one `MFrequency::Convert` over a
4//! sequence of epochs.
5
6use crate::measures_oracle::MeasuresOracle;
7use crate::oracle_runtime::{OracleError, oracle_operation};
8
9#[cfg(has_casacore_cpp)]
10use crate::oracle_runtime::CasacoreOracleRuntime;
11
12#[cfg(has_casacore_cpp)]
13unsafe extern "C" {
14    fn measures_shim_frequency_convert_sequence(
15        freq_hz: *const f64,
16        n_freq: i32,
17        ref_in: *const std::ffi::c_char,
18        ref_out: *const std::ffi::c_char,
19        epoch_mjd: *const f64,
20        n_epoch: i32,
21        dir_lon: f64,
22        dir_lat: f64,
23        dir_ref: *const std::ffi::c_char,
24        obs_lon: f64,
25        obs_lat: f64,
26        obs_h: f64,
27        freq_out: *mut f64,
28    ) -> i32;
29}
30
31/// One ordered conversion run: every frequency converted at each epoch in
32/// turn, through one converter on one mutable frame.
33#[derive(Debug, Clone, Copy)]
34pub struct FrequencySequence<'a> {
35    /// Frequencies to convert at every epoch, in Hz.
36    pub frequencies_hz: &'a [f64],
37    /// Source frame name, for example `"TOPO"`.
38    pub ref_in: &'a str,
39    /// Target frame name, for example `"LSRK"`.
40    pub ref_out: &'a str,
41    /// UTC epochs in MJD days, in the order they are applied.
42    pub epochs_mjd: &'a [f64],
43    /// Direction longitude and latitude in radians.
44    pub direction_rad: [f64; 2],
45    /// Direction frame name, for example `"J2000"`.
46    pub direction_ref: &'a str,
47    /// Observatory WGS84 longitude and latitude (radians) and height (m).
48    pub observatory: [f64; 3],
49}
50
51impl MeasuresOracle {
52    /// Run CASA's ordered conversion in C++ casacore. The result is
53    /// epoch-major: `result[e * frequencies.len() + f]`.
54    #[cfg_attr(not(has_casacore_cpp), allow(unused_variables))]
55    pub fn frequency_convert_sequence(
56        sequence: FrequencySequence<'_>,
57    ) -> Result<Vec<f64>, OracleError> {
58        oracle_operation!("measures.frequency_convert_sequence", {
59            cpp_frequency_convert_sequence(sequence).map_err(|message| OracleError::CppFailure {
60                operation: "measures.frequency_convert_sequence",
61                message,
62            })
63        })
64    }
65}
66
67#[cfg(has_casacore_cpp)]
68fn cpp_frequency_convert_sequence(sequence: FrequencySequence<'_>) -> Result<Vec<f64>, String> {
69    let c_string = |value| {
70        CasacoreOracleRuntime::c_string("measures input", value).map_err(|error| error.to_string())
71    };
72    let (c_in, c_out, c_dir) = (
73        c_string(sequence.ref_in)?,
74        c_string(sequence.ref_out)?,
75        c_string(sequence.direction_ref)?,
76    );
77    let count = |values: &[f64]| i32::try_from(values.len()).map_err(|error| error.to_string());
78    let (n_freq, n_epoch) = (count(sequence.frequencies_hz)?, count(sequence.epochs_mjd)?);
79    let mut out = vec![0.0; sequence.frequencies_hz.len() * sequence.epochs_mjd.len()];
80    let [observatory_lon, observatory_lat, observatory_h] = sequence.observatory;
81    // SAFETY: the pointers and lengths describe live slices and `out` holds
82    // exactly `n_freq * n_epoch` values; the strings are NUL-terminated.
83    let rc = unsafe {
84        measures_shim_frequency_convert_sequence(
85            sequence.frequencies_hz.as_ptr(),
86            n_freq,
87            c_in.as_ptr(),
88            c_out.as_ptr(),
89            sequence.epochs_mjd.as_ptr(),
90            n_epoch,
91            sequence.direction_rad[0],
92            sequence.direction_rad[1],
93            c_dir.as_ptr(),
94            observatory_lon,
95            observatory_lat,
96            observatory_h,
97            out.as_mut_ptr(),
98        )
99    };
100    if rc == 0 {
101        Ok(out)
102    } else {
103        Err(format!("C++ frequency_convert_sequence failed: rc={rc}"))
104    }
105}
