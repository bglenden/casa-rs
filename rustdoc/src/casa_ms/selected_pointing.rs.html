67#[derive(Debug, Clone, Copy, PartialEq)]
68struct AntennaPointingQueryDomain {
69    antenna_id: i32,
70    visibility_time_bounds: [f64; 2],
71    centroid_time_bounds: [f64; 2],
72}
73
74impl SelectedPointingQueryDomain {
75    #[must_use]
76    pub(crate) fn builder() -> SelectedPointingQueryDomainBuilder {
77        SelectedPointingQueryDomainBuilder::default()
78    }
79
80    #[must_use]
81    pub(crate) fn antenna_ids(&self) -> impl ExactSizeIterator<Item = i32> + '_ {
82        self.antennas.iter().map(|antenna| antenna.antenna_id)
83    }
84
85    #[must_use]
86    pub(crate) fn time_bounds_mjd_seconds(
87        &self,
88        antenna_id: i32,
89        sampling: PointingTimeSampling,
90    ) -> Option<[f64; 2]> {
91        let antenna = self.antennas.get(
92            self.antennas
93                .binary_search_by_key(&antenna_id, |antenna| antenna.antenna_id)
94                .ok()?,
95        )?;
96        match sampling {
97            PointingTimeSampling::VisibilityTime => Some(antenna.visibility_time_bounds),
98            PointingTimeSampling::VisibilityTimeCentroid => Some(antenna.centroid_time_bounds),
99        }
100    }
101
102    #[must_use]
103    pub(crate) fn retained_bytes(&self) -> usize {
104        self.antennas.len() * size_of::<AntennaPointingQueryDomain>()
105    }
106}

108/// One-pass selected MAIN metadata accumulator for POINTING preparation.
109#[derive(Debug, Default)]
110pub(crate) struct SelectedPointingQueryDomainBuilder {
111    antennas: Vec<AntennaPointingQueryDomain>,
112}
113
114impl SelectedPointingQueryDomainBuilder {
115    pub(crate) fn observe_row(
116        &mut self,
117        antenna1: i32,
118        antenna2: i32,
119        time_mjd_seconds: f64,
120        time_centroid_mjd_seconds: f64,
121    ) -> MsResult<()> {
122        if antenna1 < 0
123            || antenna2 < 0
124            || !time_mjd_seconds.is_finite()
125            || !time_centroid_mjd_seconds.is_finite()
126        {
127            return Err(MsError::InvalidInput(
128                "selected POINTING domain requires non-negative antennas and finite epochs"
129                    .to_string(),
130            ));
131        }
132        self.observe_antenna(antenna1, time_mjd_seconds, time_centroid_mjd_seconds);
133        self.observe_antenna(antenna2, time_mjd_seconds, time_centroid_mjd_seconds);
134        Ok(())
135    }
136
137    fn observe_antenna(
138        &mut self,
139        antenna_id: i32,
140        time_mjd_seconds: f64,
141        time_centroid_mjd_seconds: f64,
142    ) {
143        if let Some(antenna) = self
144            .antennas
145            .iter_mut()
146            .find(|antenna| antenna.antenna_id == antenna_id)
147        {
148            extend_bounds(&mut antenna.visibility_time_bounds, time_mjd_seconds);
149            extend_bounds(&mut antenna.centroid_time_bounds, time_centroid_mjd_seconds);
150        } else {
151            self.antennas.push(AntennaPointingQueryDomain {
152                antenna_id,
153                visibility_time_bounds: [time_mjd_seconds; 2],
154                centroid_time_bounds: [time_centroid_mjd_seconds; 2],
155            });
156        }
157    }
158
159    pub(crate) fn finish(mut self) -> MsResult<Option<SelectedPointingQueryDomain>> {
160        if self.antennas.is_empty() {
161            return Ok(None);
162        }
163        self.antennas.sort_by_key(|antenna| antenna.antenna_id);
164        Ok(Some(SelectedPointingQueryDomain {
165            antennas: self.antennas.into_boxed_slice(),
166        }))
167    }
168}
169
170fn extend_bounds([first, last]: &mut [f64; 2], value: f64) {
171    *first = first.min(value);
172    *last = last.max(value);
173}

311    pub(crate) fn direction_brackets(
312        &self,
313        engine: &MsCalEngine,
314        queries: &[PointingDirectionQuery],
315    ) -> MsResult<Vec<PointingDirectionBracket>> {
316        queries
317            .iter()
318            .copied()
319            .map(|query| self.direction_bracket(engine, query))
320            .collect()
321    }
322
323    fn direction_bracket(
324        &self,
325        engine: &MsCalEngine,
326        query: PointingDirectionQuery,
327    ) -> MsResult<PointingDirectionBracket> {
328        let start = self
329            .rows
330            .partition_point(|row| row.metadata.antenna_id < query.antenna_id);
331        let end = self
332            .rows
333            .partition_point(|row| row.metadata.antenna_id <= query.antenna_id);
334        let rows = &self.rows[start..end];
335        let first_at_or_after =
336            rows.partition_point(|row| row.metadata.time_mjd_seconds < query.time_mjd_seconds);
337        let first_after =
338            rows.partition_point(|row| row.metadata.time_mjd_seconds <= query.time_mjd_seconds);
339        let before = if first_after == 0 {
340            None
341        } else {
342            let time = rows[first_after - 1].metadata.time_mjd_seconds;
343            rows[..first_after]
344                .iter()
345                .find(|row| row.metadata.time_mjd_seconds.to_bits() == time.to_bits())
346        };
347        let after = rows.get(first_at_or_after);
348
349        let lower = query.time_mjd_seconds - self.maximum_half_interval_seconds;
350        let upper = query.time_mjd_seconds + self.maximum_half_interval_seconds;
351        let covering_start = rows.partition_point(|row| row.metadata.time_mjd_seconds < lower);
352        let covering_end = rows.partition_point(|row| row.metadata.time_mjd_seconds <= upper);
353        let covering = rows[covering_start..covering_end]
354            .iter()
355            .filter(|row| {
356                row.metadata.interval_seconds > 0.0
357                    && (row.metadata.time_mjd_seconds - query.time_mjd_seconds).abs()
358                        <= 0.5 * row.metadata.interval_seconds
359            })
360            .min_by(|left, right| {
361                (left.metadata.time_mjd_seconds - query.time_mjd_seconds)
362                    .abs()
363                    .total_cmp(&(right.metadata.time_mjd_seconds - query.time_mjd_seconds).abs())
364                    .then_with(|| left.metadata.row_index.cmp(&right.metadata.row_index))
365            });
366        Ok(PointingDirectionBracket {
367            covering: evaluate_candidate(engine, &self.coefficients, covering, query)?,
368            before: evaluate_candidate(engine, &self.coefficients, before, query)?,
369            after: evaluate_candidate(engine, &self.coefficients, after, query)?,
370        })
371    }
372}

374impl MeasurementSet {
375    /// Bound POINTING preparation from row-count metadata without reading its cells.
376    ///
377    /// All POINTING rows are conservatively treated as retained candidates at the
378    /// planned polynomial bound. The measurements contain byte/count ceilings,
379    /// with a zero build time; the construction ceiling excludes scalar scan row
380    /// storage. The second result charges that storage per scan-block row. A
381    /// missing POINTING table requires neither catalog nor scan storage.
382    pub(crate) fn selected_pointing_catalog_requirements(
383        &self,
384        domain: &SelectedPointingQueryDomain,
385        maximum_polynomial_terms: usize,
386    ) -> MsResult<(SelectedPointingCatalogMeasurements, usize)> {
387        if maximum_polynomial_terms == 0 {
388            return Err(MsError::InvalidInput(
389                "POINTING requirements need a positive polynomial bound".to_string(),
390            ));
391        }
392        let rows = match self.pointing() {
393            Ok(pointing) => pointing.row_count(),
394            Err(MsError::MissingSubtable(_)) => {
395                return Ok((SelectedPointingCatalogMeasurements::default(), 0));
396            }
397            Err(error) => return Err(error),
398        };
399        let endpoint_count = domain.antenna_ids().len();
400        let endpoint_bytes = endpoint_count
401            .checked_mul(size_of::<[Option<CandidateMetadata>; 2]>())
402            .ok_or_else(byte_overflow)?;
403        let prechecked_rows = endpoint_count
404            .checked_mul(2)
405            .and_then(|endpoints| rows.checked_add(endpoints))
406            .ok_or_else(byte_overflow)?;
407        let scan_fixed_bytes =
408            catalog_upper_bound_bytes(prechecked_rows, maximum_polynomial_terms)?
409                .checked_add(endpoint_bytes)
410                .ok_or_else(byte_overflow)?;
411        let retained_bytes = catalog_upper_bound_bytes(rows, maximum_polynomial_terms)?;
412        // The catalog bound already includes one coefficient copy and array
413        // dimensions. Direction cells coexist with the final coefficient copy.
414        let copy_bytes_per_row = maximum_polynomial_terms
415            .checked_mul(2)
416            .and_then(|terms| terms.checked_mul(size_of::<f64>()))
417            .and_then(|bytes| bytes.checked_add(size_of::<CandidateMetadata>()))
418            .and_then(|bytes| bytes.checked_add(size_of::<usize>()))
419            .and_then(|bytes| bytes.checked_add(size_of::<Option<ArrayValue>>()))
420            .ok_or_else(byte_overflow)?;
421        let copy_peak_bytes = rows
422            .checked_mul(copy_bytes_per_row)
423            .and_then(|bytes| bytes.checked_add(retained_bytes))
424            .ok_or_else(byte_overflow)?;
425        Ok((
426            SelectedPointingCatalogMeasurements {
427                retained_bytes,
428                construction_peak_bytes: scan_fixed_bytes.max(copy_peak_bytes),
429                source_rows_scanned: rows,
430                retained_rows: rows,
431                build_nanos: 0,
432            },
433            SCALAR_SCAN_BYTES_PER_ROW,
434        ))
435    }

437    /// Scan POINTING scalar metadata once, then retain and read only selected candidates.
438    pub(crate) fn prepare_selected_pointing_catalog(
439        &self,
440        column: PointingDirectionColumn,
441        domain: &SelectedPointingQueryDomain,
442        sampling: PointingTimeSampling,
443        plan: PointingReadPlan,
444    ) -> MsResult<PreparedSelectedPointingCatalog> {
445        let started = Instant::now();
446        let pointing = match self.pointing() {
447            Ok(pointing) => pointing,
448            Err(MsError::MissingSubtable(_)) => {
449                return Ok(PreparedSelectedPointingCatalog {
450                    rows: Vec::new(),
451                    coefficients: Vec::new(),
452                    maximum_half_interval_seconds: 0.0,
453                    measurements: SelectedPointingCatalogMeasurements {
454                        build_nanos: elapsed_nanos(started),
455                        ..Default::default()
456                    },
457                });
458            }
459            Err(error) => return Err(error),
460        };
461        let table = pointing.table();
462        let mut retained = Vec::<CandidateMetadata>::new();
463        let endpoint_count = domain.antenna_ids().len();
464        let endpoint_bytes = endpoint_count
465            .checked_mul(size_of::<[Option<CandidateMetadata>; 2]>())
466            .ok_or_else(byte_overflow)?;
467        let maximum_scan_bytes = plan
468            .scan_rows_per_block
469            .checked_mul(SCALAR_SCAN_BYTES_PER_ROW)
470            .and_then(|bytes| bytes.checked_add(endpoint_bytes))
471            .ok_or_else(byte_overflow)?;
472        ensure_budget(maximum_scan_bytes, plan.maximum_catalog_bytes)?;
473        let mut endpoints = Vec::new();
474        endpoints.try_reserve_exact(endpoint_count).map_err(|_| {
475            MsError::InvalidInput("POINTING endpoint allocation failed".to_string())
476        })?;
477        endpoints.resize(endpoint_count, [None; 2]);
478        let mut construction_peak_bytes = 0usize;
479
480        for block_start in (0..pointing.row_count()).step_by(plan.scan_rows_per_block) {
481            let block_end = (block_start + plan.scan_rows_per_block).min(pointing.row_count());
482            let rows = (block_start..block_end).collect::<Vec<_>>();
483            let antennas = table
484                .column_accessor("ANTENNA_ID")?
485                .scalar_cells_owned_for_rows(&rows)?;
486            let times = table
487                .column_accessor("TIME")?
488                .scalar_cells_owned_for_rows(&rows)?;
489            let intervals = table
490                .column_accessor("INTERVAL")?
491                .scalar_cells_owned_for_rows(&rows)?;
492            let time_origins = table
493                .column_accessor("TIME_ORIGIN")?
494                .scalar_cells_owned_for_rows(&rows)?;
495            let orders = table
496                .column_accessor("NUM_POLY")?
497                .scalar_cells_owned_for_rows(&rows)?;
498            let scan_bytes = rows.capacity() * size_of::<usize>()
499                + (antennas.capacity()
500                    + times.capacity()
501                    + intervals.capacity()
502                    + time_origins.capacity()
503                    + orders.capacity())
504                    * size_of::<Option<ScalarValue>>();
505            construction_peak_bytes = construction_peak_bytes.max(
506                retained.capacity() * size_of::<CandidateMetadata>()
507                    + endpoints.capacity() * size_of::<[Option<CandidateMetadata>; 2]>()
508                    + scan_bytes,
509            );
510
511            for (slot, row_index) in rows.into_iter().enumerate() {
512                let antenna_id = required_i32(&antennas, slot, "ANTENNA_ID", row_index)?;
513                let Some([first_query_time, last_query_time]) =
514                    domain.time_bounds_mjd_seconds(antenna_id, sampling)
515                else {
516                    continue;
517                };
518                let time_mjd_seconds = required_f64(&times, slot, "TIME", row_index)?;
519                let interval_seconds = required_f64(&intervals, slot, "INTERVAL", row_index)?;
520                let time_origin_mjd_seconds =
521                    required_f64(&time_origins, slot, "TIME_ORIGIN", row_index)?;
522                let order = required_i32(&orders, slot, "NUM_POLY", row_index)?;
523                let polynomial_terms = usize::try_from(order).ok()
524                    .and_then(|order| order.checked_add(1))
525                    .filter(|terms| *terms <= plan.maximum_polynomial_terms)
526                    .ok_or_else(|| MsError::InvalidInput(format!(
527                        "POINTING row {row_index} polynomial order {order} exceeds the planned bound"
528                    )))?;
529                if !time_mjd_seconds.is_finite()
530                    || !interval_seconds.is_finite()
531                    || !time_origin_mjd_seconds.is_finite()
532                {
533                    return Err(MsError::InvalidInput(format!(
534                        "POINTING row {row_index} has non-finite time metadata"
535                    )));
536                }
537                let candidate = CandidateMetadata {
538                    row_index,
539                    antenna_id,
540                    time_mjd_seconds,
541                    interval_seconds,
542                    time_origin_mjd_seconds,
543                    polynomial_terms,
544                };
545                let half_interval = 0.5 * interval_seconds;
546                let intersects = interval_seconds > 0.0
547                    && time_mjd_seconds + half_interval >= first_query_time
548                    && time_mjd_seconds - half_interval <= last_query_time;
549                let centre_inside =
550                    time_mjd_seconds >= first_query_time && time_mjd_seconds <= last_query_time;
551                if intersects || centre_inside {
552                    let next_rows = retained
553                        .len()
554                        .checked_add(1 + 2 * endpoints.len())
555                        .ok_or_else(byte_overflow)?;
556                    let required =
557                        catalog_upper_bound_bytes(next_rows, plan.maximum_polynomial_terms)?
558                            .checked_add(scan_bytes)
559                            .and_then(|bytes| {
560                                bytes.checked_add(
561                                    endpoints.capacity()
562                                        * size_of::<[Option<CandidateMetadata>; 2]>(),
563                                )
564                            })
565                            .ok_or_else(byte_overflow)?;
566                    ensure_budget(required, plan.maximum_catalog_bytes)?;
567                    retained.try_reserve_exact(1).map_err(|_| {
568                        MsError::InvalidInput(
569                            "POINTING candidate metadata allocation failed".to_string(),
570                        )
571                    })?;
572                    retained.push(candidate);
573                } else {
574                    let endpoint_index = domain
575                        .antenna_ids()
576                        .position(|selected| selected == antenna_id)
577                        .expect("selected antenna was already found");
578                    let endpoint = &mut endpoints[endpoint_index];
579                    if time_mjd_seconds < first_query_time {
580                        retain_later(&mut endpoint[0], candidate);
581                    }
582                    if time_mjd_seconds > last_query_time {
583                        retain_earlier(&mut endpoint[1], candidate);
584                    }
585                }
586            }
587            ensure_budget(
588                catalog_upper_bound_bytes(
589                    retained.capacity() + 2 * endpoints.len(),
590                    plan.maximum_polynomial_terms,
591                )?,
592                plan.maximum_catalog_bytes,
593            )?;
594        }
595
596        let endpoint_rows = endpoints.iter().flatten().flatten().count();
597        let final_metadata_rows = retained
598            .len()
599            .checked_add(endpoint_rows)
600            .ok_or_else(byte_overflow)?;
601        ensure_budget(
602            catalog_upper_bound_bytes(final_metadata_rows, plan.maximum_polynomial_terms)?,
603            plan.maximum_catalog_bytes,
604        )?;
605        retained.try_reserve_exact(endpoint_rows).map_err(|_| {
606            MsError::InvalidInput("POINTING endpoint retention allocation failed".to_string())
607        })?;
608        retained.extend(endpoints.into_iter().flatten().flatten());
609        retained.sort_by(|left, right| {
610            left.antenna_id
611                .cmp(&right.antenna_id)
612                .then_with(|| left.time_mjd_seconds.total_cmp(&right.time_mjd_seconds))
613                .then_with(|| left.row_index.cmp(&right.row_index))
614        });
615        retained.dedup_by_key(|row| row.row_index);
616        let index_bytes = retained
617            .len()
618            .checked_mul(size_of::<usize>())
619            .ok_or_else(byte_overflow)?;
620        let index_peak = catalog_upper_bound_bytes(retained.len(), plan.maximum_polynomial_terms)?
621            .checked_add(retained.capacity() * size_of::<CandidateMetadata>())
622            .and_then(|bytes| bytes.checked_add(index_bytes))
623            .ok_or_else(byte_overflow)?;
624        ensure_budget(index_peak, plan.maximum_catalog_bytes)?;
625        let mut row_indices = Vec::new();
626        row_indices.try_reserve_exact(retained.len()).map_err(|_| {
627            MsError::InvalidInput("POINTING row-index allocation failed".to_string())
628        })?;
629        row_indices.extend(retained.iter().map(|row| row.row_index));
630        ensure_budget(
631            construction_upper_bound(&retained, &row_indices, plan.maximum_polynomial_terms)?,
632            plan.maximum_catalog_bytes,
633        )?;
634        let direction_cells = table
635            .column_accessor(column.name())?
636            .array_cells_owned_uncached(&row_indices)?;
637        let direction_heap_bytes = direction_cells
638            .iter()
639            .flatten()
640            .try_fold(0usize, |bytes, value| {
641                bytes.checked_add(array_heap_bytes(value)?)
642            })
643            .ok_or_else(byte_overflow)?;
644        construction_peak_bytes = construction_peak_bytes.max(
645            retained.capacity() * size_of::<CandidateMetadata>()
646                + row_indices.capacity() * size_of::<usize>()
647                + direction_cells.capacity() * size_of::<Option<ArrayValue>>()
648                + direction_heap_bytes,
649        );
650
651        let total_coefficients = retained
652            .iter()
653            .try_fold(0usize, |total, metadata| {
654                total.checked_add(2 * metadata.polynomial_terms)
655            })
656            .ok_or_else(byte_overflow)?;
657        let copy_peak = retained
658            .capacity()
659            .checked_mul(size_of::<CandidateMetadata>())
660            .and_then(|bytes| bytes.checked_add(row_indices.capacity() * size_of::<usize>()))
661            .and_then(|bytes| {
662                bytes.checked_add(direction_cells.capacity() * size_of::<Option<ArrayValue>>())
663            })
664            .and_then(|bytes| bytes.checked_add(direction_heap_bytes))
665            .and_then(|bytes| bytes.checked_add(retained.len() * size_of::<CatalogRow>()))
666            .and_then(|bytes| bytes.checked_add(total_coefficients * size_of::<f64>()))
667            .ok_or_else(byte_overflow)?;
668        ensure_budget(copy_peak, plan.maximum_catalog_bytes)?;
669        construction_peak_bytes = construction_peak_bytes.max(copy_peak);
670        let mut catalog_rows = Vec::new();
671        catalog_rows
672            .try_reserve_exact(retained.len())
673            .map_err(|_| {
674                MsError::InvalidInput("POINTING catalog row allocation failed".to_string())
675            })?;
676        let mut coefficients = Vec::new();
677        coefficients
678            .try_reserve_exact(total_coefficients)
679            .map_err(|_| {
680                MsError::InvalidInput("POINTING coefficient allocation failed".to_string())
681            })?;
682        for ((metadata, direction), row_index) in retained
683            .into_iter()
684            .zip(direction_cells)
685            .zip(row_indices.iter().copied())
686        {
687            let direction = direction.ok_or_else(|| {
688                MsError::InvalidInput(format!(
689                    "POINTING.{} row {row_index} is undefined",
690                    column.name()
691                ))
692            })?;
693            let ArrayValue::Float64(coefficient_array) = direction else {
694                return Err(MsError::ColumnTypeMismatch {
695                    column: column.name().to_string(),
696                    table: "POINTING".to_string(),
697                    expected: "Float64 array shaped [2, nPoly+1]".to_string(),
698                    found: format!("{:?}", direction.primitive_type()),
699                });
700            };
701            if coefficient_array.shape() != [2, metadata.polynomial_terms] {
702                return Err(MsError::ColumnTypeMismatch {
703                    column: column.name().to_string(),
704                    table: "POINTING".to_string(),
705                    expected: format!("Float64 array shaped [2, {}]", metadata.polynomial_terms),
706                    found: format!("Float64 array shaped {:?}", coefficient_array.shape()),
707                });
708            }
709            let source_ref = resolve_direction_reference_selected(
710                table,
711                "POINTING",
712                column.name(),
713                metadata.row_index,
714            )?;
715            let coefficient_offset = coefficients.len();
716            coefficients.extend(coefficient_array.iter().copied());
717            catalog_rows.push(CatalogRow {
718                metadata,
719                coefficient_offset,
720                source_ref,
721            });
722        }
723        let retained_bytes = catalog_retained_bytes(&catalog_rows, &coefficients)?;
724        ensure_budget(retained_bytes, plan.maximum_catalog_bytes)?;
725        construction_peak_bytes = construction_peak_bytes.max(
726            retained_bytes
727                .checked_add(row_indices.capacity() * size_of::<usize>())
728                .ok_or_else(byte_overflow)?,
729        );
730        let maximum_half_interval_seconds = catalog_rows
731            .iter()
732            .filter(|row| row.metadata.interval_seconds > 0.0)
733            .map(|row| 0.5 * row.metadata.interval_seconds)
734            .fold(0.0, f64::max);
735        let measurements = SelectedPointingCatalogMeasurements {
736            retained_bytes,
737            construction_peak_bytes,
738            source_rows_scanned: pointing.row_count(),
739            retained_rows: catalog_rows.len(),
740            build_nanos: elapsed_nanos(started),
741        };
742        Ok(PreparedSelectedPointingCatalog {
743            rows: catalog_rows,
744            coefficients,
745            maximum_half_interval_seconds,
746            measurements,
747        })
748    }
749}

751fn evaluate_candidate(
752    engine: &MsCalEngine,
753    coefficients: &[f64],
754    row: Option<&CatalogRow>,
755    query: PointingDirectionQuery,
756) -> MsResult<Option<PointingDirectionCandidate>> {
757    let Some(row) = row else {
758        return Ok(None);
759    };
760    let offset_seconds = query.time_mjd_seconds - row.metadata.time_origin_mjd_seconds;
761    let coefficient_count = 2 * row.metadata.polynomial_terms;
762    let coefficients =
763        &coefficients[row.coefficient_offset..row.coefficient_offset + coefficient_count];
764    let mut angles_rad = [0.0; 2];
765    for axis in 0..2 {
766        for term in (0..row.metadata.polynomial_terms).rev() {
767            angles_rad[axis] = angles_rad[axis] * offset_seconds
768                + coefficients[axis * row.metadata.polynomial_terms + term];
769        }
770    }
771    if !angles_rad.into_iter().all(f64::is_finite) {
772        return Err(MsError::InvalidInput(format!(
773            "POINTING row {} evaluates to a non-finite direction",
774            row.metadata.row_index
775        )));
776    }
777    Ok(Some(PointingDirectionCandidate {
778        row_index: row.metadata.row_index,
779        row_time_mjd_seconds: row.metadata.time_mjd_seconds,
780        direction_j2000_rad: engine.direction_angles_j2000(
781            query.time_mjd_seconds,
782            angles_rad,
783            row.source_ref,
784        )?,
785    }))
786}

802        None => Err(MsError::InvalidInput(format!(
803            "POINTING.{column} row {row} is undefined"
804        ))),
805    }
806}

828fn retain_later(current: &mut Option<CandidateMetadata>, candidate: CandidateMetadata) {
829    if current.is_none_or(|existing| {
830        candidate
831            .time_mjd_seconds
832            .total_cmp(&existing.time_mjd_seconds)
833            .then_with(|| existing.row_index.cmp(&candidate.row_index))
834            .is_gt()
835    }) {
836        *current = Some(candidate);
837    }
838}
839
840fn retain_earlier(current: &mut Option<CandidateMetadata>, candidate: CandidateMetadata) {
841    if current.is_none_or(|existing| {
842        candidate
843            .time_mjd_seconds
844            .total_cmp(&existing.time_mjd_seconds)
845            .then_with(|| candidate.row_index.cmp(&existing.row_index))
846            .is_lt()
847    }) {
848        *current = Some(candidate);
849    }
850}

852fn catalog_upper_bound_bytes(rows: usize, maximum_polynomial_terms: usize) -> MsResult<usize> {
853    let bytes_per_row = maximum_polynomial_terms
854        .checked_mul(2)
855        .and_then(|terms| terms.checked_mul(size_of::<f64>()))
856        .and_then(|bytes| bytes.checked_add(size_of::<CatalogRow>()))
857        .and_then(|bytes| bytes.checked_add(4 * size_of::<usize>()))
858        .ok_or_else(byte_overflow)?;
859    rows.checked_mul(bytes_per_row).ok_or_else(byte_overflow)
860}
861
862fn construction_upper_bound(
863    metadata: &Vec<CandidateMetadata>,
864    row_indices: &Vec<usize>,
865    maximum_polynomial_terms: usize,
866) -> MsResult<usize> {
867    catalog_upper_bound_bytes(metadata.len(), maximum_polynomial_terms)?
868        .checked_add(metadata.capacity() * size_of::<CandidateMetadata>())
869        .and_then(|bytes| bytes.checked_add(row_indices.capacity() * size_of::<usize>()))
870        .and_then(|bytes| bytes.checked_add(metadata.len() * size_of::<Option<ArrayValue>>()))
871        .ok_or_else(byte_overflow)
872}
873
874fn catalog_retained_bytes(rows: &Vec<CatalogRow>, coefficients: &Vec<f64>) -> MsResult<usize> {
875    rows.capacity()
876        .checked_mul(size_of::<CatalogRow>())
877        .and_then(|bytes| bytes.checked_add(coefficients.capacity() * size_of::<f64>()))
878        .ok_or_else(byte_overflow)
879}
880
881fn array_heap_bytes(value: &ArrayValue) -> Option<usize> {
882    let dimensions = value.ndim().checked_mul(2 * size_of::<usize>())?;
883    match value {
884        ArrayValue::Float64(values) => values
885            .len()
886            .checked_mul(size_of::<f64>())?
887            .checked_add(dimensions),
888        _ => Some(dimensions),
889    }
890}
891
892fn ensure_budget(required: usize, available: usize) -> MsResult<()> {
893    if required > available {
894        return Err(MsError::InvalidInput(format!(
895            "POINTING catalog requires {required} bytes but the planner supplied {available} bytes"
896        )));
897    }
898    Ok(())
899}
900
901fn byte_overflow() -> MsError {
902    MsError::InvalidInput("POINTING catalog byte count overflow".to_string())
903}

905fn elapsed_nanos(started: Instant) -> u64 {
906    u64::try_from(started.elapsed().as_nanos()).unwrap_or(u64::MAX)
907}

909/// Conservative catalog construction peak for content planning.
910pub(crate) fn selected_pointing_preparation_peak_bytes(
911    selected_rows: usize,
912    scan_rows: usize,
913    maximum_polynomial_terms: usize,
914    _direction_column: PointingDirectionColumn,
915) -> Option<usize> {
916    let candidate_rows = selected_rows.checked_mul(2)?.checked_add(2)?;
917    catalog_upper_bound_bytes(candidate_rows, maximum_polynomial_terms)
918        .ok()?
919        .checked_add(scan_rows.checked_mul(SCALAR_SCAN_BYTES_PER_ROW)?)
920}
