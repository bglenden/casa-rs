16/// Read an unconsumed FITS header field as image miscellaneous metadata.
17pub(super) fn custom_value(
18    path: &Path,
19    key: &str,
20) -> Result<casa_provider_contracts::CasaValue, ImageError> {
21    use casa_coordinates::{FitsHeader, FitsValue};
22    use casa_provider_contracts::CasaValue;
23    let upper = key.to_ascii_uppercase();
24    let internal = [
25        "SIMPLE", "BITPIX", "NAXIS", "EXTEND", "BLOCKED", "BLANK", "BSCALE", "BZERO", "XTENSION",
26        "PCOUNT", "GCOUNT", "ORIGIN", "RADESYS", "EQUINOX", "SPECSYS", "RESTFRQ", "TELESCOP",
27        "OBSERVER", "DATE-OBS", "MJD-OBS", "TIMESYS", "OBJECT", "BUNIT", "IMTYPE", "BMAJ", "BMIN",
28        "BPA", "LONPOLE", "LATPOLE", "CROTA2",
29    ];
30    let axis_keyword = upper.strip_prefix("NAXIS").is_some_and(|suffix| {
31        !suffix.is_empty() && suffix.bytes().all(|byte| byte.is_ascii_digit())
32    });
33    let matrix_keyword = ["PC", "CD"].iter().any(|prefix| {
34        upper
35            .strip_prefix(prefix)
36            .and_then(|suffix| suffix.split_once('_'))
37            .is_some_and(|(row, col)| {
38                !row.is_empty()
39                    && !col.is_empty()
40                    && row.bytes().all(|byte| byte.is_ascii_digit())
41                    && col.bytes().all(|byte| byte.is_ascii_digit())
42            })
43    });
44    if internal.contains(&upper.as_str()) || axis_keyword || matrix_keyword {
45        return Err(ImageError::InvalidMetadata(format!(
46            "Unknown keyword {key}"
47        )));
48    }
49    let cards = crate::fits_read::read_cards(path)?;
50    let references = cards.iter().map(String::as_str).collect::<Vec<_>>();
51    let header = FitsHeader::from_cards(&references);
52    match header.get(&upper) {
53        Some(FitsValue::String(value)) => Ok(CasaValue::string(value)),
54        Some(FitsValue::Integer(value)) if i32::try_from(*value).is_ok() => {
55            Ok(CasaValue::Int { value: *value })
56        }
57        Some(FitsValue::Float(value)) => Ok(CasaValue::float(*value)),
58        Some(FitsValue::Logical(_)) => Err(ImageError::InvalidMetadata(
59            "Unhandled data type Bool for user defined type. Send us a bug report".into(),
60        )),
61        _ => Err(ImageError::InvalidMetadata(format!(
62            "Unknown keyword {key}"
63        ))),
64    }
65}

67/// FITSImage uses TiledFileAccess's 32768-pixel tile shape.
68fn tile_shape(shape: &[usize]) -> Vec<usize> {
69    let mut remaining = 32768_f32;
70    let mut result = vec![1; shape.len()];
71    for (axis, &length) in shape.iter().enumerate() {
72        if length as f32 <= remaining {
73            result[axis] = length;
74            remaining /= length as f32;
75        } else {
76            let ideal = (remaining + 0.5) as usize;
77            if length % ideal == 0 {
78                result[axis] = ideal;
79            } else {
80                for offset in 1..ideal.min(length - ideal + 1) {
81                    if length % (ideal - offset) == 0 {
82                        result[axis] = ideal - offset;
83                        break;
84                    }
85                    if length % (ideal + offset) == 0 {
86                        result[axis] = ideal + offset;
87                        break;
88                    }
89                }
90            }
91            break;
92        }
93    }
94    result
95}

97pub(super) fn summary(path: &Path) -> Result<ImageHeaderSummary, ImageError> {
98    let mut primary = PrimaryHdu::open(path)?;
99    let beam_data = primary.beam_set()?;
100    let mut info = image_info_from_fits_header(&primary.header);
101    if let Some((beams, _)) = &beam_data {
102        info.beam_set = beams.clone();
103    }
104    let coords = &primary.coordinates;
105    let beam_records = beam_data.as_ref().map(|(_, records)| records);
106    let per_plane_beams = summarize_beam_set(coords, &primary.shape, &info.beam_set, beam_records)?;
107    let direction =
108        coords
109            .find_coordinate(CoordinateType::Direction)
110            .and_then(|index| match coords.coordinate(index) {
111                CoordinateModel::Direction(direction) => Some(direction),
112                _ => None,
113            });
114    let spectral = coords
115        .find_coordinate(CoordinateType::Spectral)
116        .and_then(|index| match coords.coordinate(index) {
117            CoordinateModel::Spectral(spectral) => Some(spectral_summary(spectral)),
118            _ => None,
119        });
120    Ok(ImageHeaderSummary {
121        imagename: path.display().to_string(),
122        pixel_type: "Float32".into(),
123        shape: primary.shape.clone(),
124        tile_shape: tile_shape(&primary.shape),
125        units: primary.header.get_string("BUNIT").unwrap_or("").to_string(),
126        object_name: if primary.header.get_string("OBJECT") == Some("") {
127            " ".into()
128        } else {
129            info.object_name
130        },
131        observer: coords.obs_info().observer.clone(),
132        telescope: coords.obs_info().telescope.clone(),
133        observation_date: casa_types::quanta::format::format_time(
134            coords
135                .obs_info()
136                .date
137                .as_ref()
138                .map_or(0.0, |epoch| epoch.value().as_mjd()),
139            casa_types::quanta::format::YMD,
140            12,
141        ),
142        direction_reference: direction
143            .map(|direction| direction.direction_ref().as_str().to_string()),
144        direction_projection: direction.map(projection_header::projection_header),
145        spectral: spectral.map(Box::new),
146        image_type: if info.image_type == ImageType::Undefined {
147            ImageType::Intensity
148        } else {
149            info.image_type
150        }
151        .to_string(),
152        default_mask: None,
153        has_mask: matches!(
154            primary.image_type,
155            FitsImageType::Float | FitsImageType::Double
156        ) || primary.header.get_int("BLANK").is_some(),
157        masks: Vec::new(),
158        regions: Vec::new(),
159        restoring_beam: info
160            .beam_set
161            .single_beam()
162            .and_then(|beam| format_beam(beam, None))
163            .map(Box::new),
164        has_multiple_beams: info.beam_set.is_multi(),
165        per_plane_beams: per_plane_beams.map(Box::new),
166        extrema: None,
167        axes: axis_summaries(coords, &primary.shape),
168    })
169}

171/// Scan the pixels in bounded chunks, ignoring FITS blanks and NaNs.
172pub(super) fn extrema(path: &Path) -> Result<Option<ImageExtrema>, ImageError> {
173    let mut image = PrimaryHdu::open(path)?;
174    let total = image
175        .shape
176        .iter()
177        .try_fold(1usize, |value, length| value.checked_mul(*length))
178        .ok_or_else(|| ImageError::InvalidMetadata("FITS image shape overflows".into()))?;
179    let mut min: Option<(f32, usize)> = None;
180    let mut max: Option<(f32, usize)> = None;
181    for start in (0..total).step_by(1_048_576) {
182        let (pixels, mask) = image.read_pixels(start, (total - start).min(1_048_576))?;
183        for (offset, (&value, &good)) in pixels.iter().zip(&mask).enumerate() {
184            if !good {
185                continue;
186            }
187            let index = start + offset;
188            if min.is_none_or(|(current, _)| value < current) {
189                min = Some((value, index));
190            }
191            if max.is_none_or(|(current, _)| value > current) {
192                max = Some((value, index));
193            }
194        }
195    }
196    let (Some((min, min_index)), Some((max, max_index))) = (min, max) else {
197        return Ok(None);
198    };
199    let position = |mut index: usize| {
200        image
201            .shape
202            .iter()
203            .map(|length| {
204                let coordinate = index % length;
205                index /= length;
206                coordinate
207            })
208            .collect::<Vec<_>>()
209    };
210    let min_position = position(min_index);
211    let max_position = position(max_index);
212    let mut axis = 0;
213    let mut spectral_axis = None;
214    for index in 0..image.coordinates.n_coordinates() {
215        let coordinate = image.coordinates.coordinate(index);
216        if coordinate.coordinate_type() == CoordinateType::Spectral && spectral_axis.is_none() {
217            spectral_axis = Some(axis);
218        }
219        axis += coordinate.n_pixel_axes();
220    }
221    Ok(Some(ImageExtrema {
222        min: f64::from(min),
223        max: f64::from(max),
224        min_world: world_header::format_position(&image.coordinates, &min_position, spectral_axis)?,
225        max_world: world_header::format_position(&image.coordinates, &max_position, spectral_axis)?,
226        min_position,
227        max_position,
228    }))
