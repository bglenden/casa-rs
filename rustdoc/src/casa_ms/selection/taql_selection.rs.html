19pub(super) fn select_rows(
20    ms: &MeasurementSet,
21    query: &str,
22    rows_per_block: Option<usize>,
23    first_only: bool,
24) -> MsResult<Vec<usize>> {
25    let statement = selection_query(ms, query, rows_per_block)?;
26    if let Statement::Select(select) = &statement {
27        if first_only {
28            return taql::first_matching_row(select, ms.main_table())
29                .map(|row| row.into_iter().collect())
30                .map_err(taql_error);
31        }
32        if let Some(rows) = select_simple_rows(ms.main_table(), select, rows_per_block, None)? {
33            return Ok(rows);
34        }
35    }
36    match taql::exec::execute_readonly(&statement, ms.main_table()).map_err(taql_error)? {
37        TaqlResult::Select { row_indices, .. } => Ok(row_indices),
38        _ => Err(MsError::InvalidInput(
39            "expected a TaQL SELECT statement".into(),
40        )),
41    }
42}

44enum ScalarComparison {
45    Integer {
46        literal: i64,
47        compare: fn(i64, i64) -> bool,
48    },
49    IntegerSet(Vec<i64>),
50    Boolean(bool),
51}
52
53impl ScalarComparison {
54    fn new(literal: &Literal, op: BinaryOp, reverse: bool) -> Option<Self> {
55        match literal {
56            Literal::Bool(value) => match op {
57                BinaryOp::Eq => Some(Self::Boolean(*value)),
58                BinaryOp::Ne => Some(Self::Boolean(!*value)),
59                _ => None,
60            },
61            Literal::Int(literal) => {
62                let compare: fn(i64, i64) -> bool = match (op, reverse) {
63                    (BinaryOp::Eq, _) => |cell, other| cell == other,
64                    (BinaryOp::Ne, _) => |cell, other| cell != other,
65                    (BinaryOp::Lt, false) | (BinaryOp::Gt, true) => |cell, other| cell < other,
66                    (BinaryOp::Le, false) | (BinaryOp::Ge, true) => |cell, other| cell <= other,
67                    (BinaryOp::Gt, false) | (BinaryOp::Lt, true) => |cell, other| cell > other,
68                    (BinaryOp::Ge, false) | (BinaryOp::Le, true) => |cell, other| cell >= other,
69                    _ => return None,
70                };
71                Some(Self::Integer {
72                    literal: *literal,
73                    compare,
74                })
75            }
76            _ => None,
77        }
78    }
79
80    fn from_predicate(predicate: &Expr) -> Option<(&Expr, Self)> {
81        match predicate {
82            Expr::Binary { left, op, right } => match (left.as_ref(), right.as_ref()) {
83                (operand, Expr::Literal(literal)) => {
84                    Some((operand, Self::new(literal, *op, false)?))
85                }
86                (Expr::Literal(literal), operand) => {
87                    Some((operand, Self::new(literal, *op, true)?))
88                }
89                _ => None,
90            },
91            Expr::InSet {
92                expr,
93                elements,
94                negated: false,
95            } => {
96                let mut values = elements
97                    .iter()
98                    .map(|element| match element {
99                        InSetElement::Value(value) => integer_literal(value),
100                        InSetElement::Range { .. } => None,
101                    })
102                    .collect::<Option<Vec<_>>>()?;
103                values.sort_unstable();
104                values.dedup();
105                Some((expr, Self::IntegerSet(values)))
106            }
107            _ => None,
108        }
109    }
110
111    fn primitive(&self) -> PrimitiveType {
112        match self {
113            Self::Integer { .. } | Self::IntegerSet(_) => PrimitiveType::Int32,
114            Self::Boolean(_) => PrimitiveType::Bool,
115        }
116    }
117
118    fn row_numbers(&self, row_count: usize) -> Option<Vec<usize>> {
119        match self {
120            Self::Integer { literal, compare } => Some(
121                (0..row_count)
122                    .filter(|&row| compare(row as i64, *literal))
123                    .collect(),
124            ),
125            Self::IntegerSet(values) => Some(
126                values
127                    .iter()
128                    .filter_map(|&value| usize::try_from(value).ok())
129                    .take_while(|&row| row < row_count)
130                    .collect(),
131            ),
132            Self::Boolean(_) => None,
133        }
134    }
135}

137/// Rows selected by a legacy TaQL expression and its optional named output.
138pub struct LegacyTaqlSelection {
139    /// Row indices relative to the input selector's ordered projection.
140    pub rows: Vec<usize>,
141    /// Optional result destination and table type.
142    pub giving: Option<GivingClause>,
143}
144
145/// Try a scalar TaQL predicate, row limits, and named output on an ordered
146/// MAIN projection. A more general query returns `None` for the full planner.
147///
148/// # Errors
149/// The expression has invalid syntax or a selected scalar cell cannot be read.
150pub fn select_projected_scalar_rows(
151    table: &Table,
152    expression: &str,
153    source_rows: Option<&[usize]>,
154) -> MsResult<Option<LegacyTaqlSelection>> {
155    let statement = taql::parse(&format!("SELECT * WHERE {expression}")).map_err(taql_error)?;
156    let Statement::Select(mut select) = statement else {
157        return Ok(None);
158    };
159    let giving = select.giving.take();
160    let offset = select.offset.take();
161    let limit = select.limit.take();
162    let Some(rows) = select_simple_rows(table, &select, None, source_rows)? else {
163        return Ok(None);
164    };
165    let rows = if offset.is_some() || limit.is_some() {
166        taql::exec::apply_view_limits(rows, offset.as_ref(), limit.as_ref()).map_err(taql_error)?
167    } else {
168        rows
169    };
170    Ok(Some(LegacyTaqlSelection { rows, giving }))
171}

173/// Evaluate a legacy selector query against the ordered legacy row view.
174///
175/// The trailing WHERE, sorting, and row limit expressions run together so
176/// that ordering and limits apply to view-relative row numbers.
177///
178/// # Errors
179/// The selection syntax is invalid or the query cannot run on MAIN.
180pub fn select_legacy_rows(ms: &MeasurementSet, expression: &str) -> MsResult<LegacyTaqlSelection> {
181    let statement = selection_query(ms, &format!("SELECT * WHERE {expression}"), None)?;
182    let Statement::Select(select) = statement else {
183        return Err(MsError::InvalidInput(
184            "expected a TaQL row selection".into(),
185        ));
186    };
187    if select.giving_expr.is_some() || (select.having.is_some() && select.group_by.is_empty()) {
188        return Err(MsError::InvalidInput(
189            "invalid clause for a TaQL row selection".into(),
190        ));
191    }
192    let rows = taql::exec::select_view_rows(&select, ms.main_table()).map_err(taql_error)?;
193    Ok(LegacyTaqlSelection {
194        rows,
195        giving: select.giving,
196    })
197}

199fn select_simple_rows(
200    table: &Table,
201    select: &SelectStatement,
202    rows_per_block: Option<usize>,
203    source_rows: Option<&[usize]>,
204) -> MsResult<Option<Vec<usize>>> {
205    let SelectStatement {
206        columns,
207        from: None,
208        joins,
209        where_clause: Some(predicate),
210        group_by,
211        having: None,
212        order_by,
213        order_distinct: false,
214        limit: None,
215        offset: None,
216        distinct: false,
217        style,
218        giving: None,
219        giving_expr: None,
220    } = select
221    else {
222        return Ok(None);
223    };
224    if !(columns.is_empty() && joins.is_empty() && group_by.is_empty() && order_by.is_empty()) {
225        return Ok(None);
226    }
227    let row_count = source_rows.map_or(table.row_count(), <[usize]>::len);
228    if let Expr::Literal(Literal::Bool(include)) = predicate {
229        return Ok(Some(if *include {
230            (0..row_count).collect()
231        } else {
232            Vec::new()
233        }));
234    }
235    if source_rows.is_none()
236        && let Some((axis, lower, upper)) = projected_uvw_range(predicate, *style)
237        && table
238            .schema()
239            .and_then(|schema| schema.column("UVW"))
240            .is_some_and(|schema| {
241                matches!(
242                    schema.column_type(),
243                    ColumnType::Array(ArrayShapeContract::Fixed { shape }) if shape == &[3]
244                ) && schema.data_type() == Some(PrimitiveType::Float64)
245                    && !schema.options().undefined
246            })
247    {
248        let rows = select_projected_uvw_rows(
249            table,
250            axis,
251            lower..=upper,
252            selection_block_size(rows_per_block)?,
253        )?;
254        return Ok(Some(rows));
255    }
256    let Some((operand, comparison)) = ScalarComparison::from_predicate(predicate) else {
257        return Ok(None);
258    };
259    if matches!(operand, Expr::RowNumber) {
260        return Ok(comparison.row_numbers(row_count));
261    }
262    let Expr::ColumnRef(column) = operand else {
263        return Ok(None);
264    };
265    let name = column.column.as_str();
266    if column.table.is_some()
267        || !table
268            .schema()
269            .and_then(|schema| schema.column(name))
270            .is_some_and(|schema| {
271                schema.column_type() == &ColumnType::Scalar
272                    && schema.data_type() == Some(comparison.primitive())
273                    && !schema.options().undefined
274            })
275    {
276        return Ok(None);
277    }
278
279    // These predicates need only one scalar column, even when the MS has large visibility arrays.
280    let rows_per_block = selection_block_size(rows_per_block)?;
281    let mut selected_rows = Vec::new();
282    for start in (0..row_count).step_by(rows_per_block) {
283        let end = start.saturating_add(rows_per_block).min(row_count);
284        let direct_rows;
285        let source = if let Some(source_rows) = source_rows {
286            &source_rows[start..end]
287        } else {
288            direct_rows = (start..end).collect::<Vec<_>>();
289            &direct_rows
290        };
291        let mut columns = table.required_scalar_columns_owned_for_rows(&[name], source)?;
292        match (&comparison, columns.remove(name)) {
293            (
294                ScalarComparison::Integer { literal, compare },
295                Some(RequiredScalarColumnValues::Int32(values)),
296            ) => {
297                selected_rows.extend(
298                    (start..end).zip(values).filter_map(|(row, value)| {
299                        compare(i64::from(value), *literal).then_some(row)
300                    }),
301                );
302            }
303            (
304                ScalarComparison::IntegerSet(set),
305                Some(RequiredScalarColumnValues::Int32(values)),
306            ) => {
307                selected_rows.extend((start..end).zip(values).filter_map(|(row, value)| {
308                    set.binary_search(&i64::from(value)).is_ok().then_some(row)
309                }));
310            }
311            (
312                ScalarComparison::Boolean(expected),
313                Some(RequiredScalarColumnValues::Bool(values)),
314            ) => {
315                selected_rows.extend(
316                    (start..end)
317                        .zip(values)
318                        .filter_map(|(row, value)| (value == *expected).then_some(row)),
319                );
320            }
321            _ => {
322                return Err(MsError::InvalidInput(format!(
323                    "unexpected scalar values in column {name}"
324                )));
325            }
326        }
327    }
328    Ok(Some(selected_rows))
329}

331fn selection_block_size(rows_per_block: Option<usize>) -> MsResult<usize> {
332    match rows_per_block {
333        Some(rows) => Ok(rows),
334        None => rows_per_block_for_budget(default_selection_budget()?)
335            .map_err(|error| MsError::InvalidInput(error.to_string())),
336    }
337}

339fn integer_literal(expr: &Expr) -> Option<i64> {
340    match expr {
341        Expr::Literal(Literal::Int(value)) => Some(*value),
342        Expr::Unary {
343            op: UnaryOp::Negate,
344            operand,
345        } => match operand.as_ref() {
346            Expr::Literal(Literal::Int(value)) => value.checked_neg(),
347            _ => None,
348        },
349        _ => None,
350    }
351}
352
353fn real_literal(expr: &Expr) -> Option<f64> {
354    match expr {
355        Expr::Literal(Literal::Float(value)) => Some(*value),
356        Expr::Unary {
357            op: UnaryOp::Negate,
358            operand,
359        } if let Expr::Literal(Literal::Float(value)) = operand.as_ref() => Some(-value),
360        _ => integer_literal(expr).map(|value| value as f64),
361    }
362}

364fn projected_uvw_range(predicate: &Expr, style: IndexStyle) -> Option<(usize, f64, f64)> {
365    let Expr::Binary {
366        left,
367        op: BinaryOp::And,
368        right,
369    } = predicate
370    else {
371        return None;
372    };
373    let Expr::Binary {
374        left: projection,
375        op: BinaryOp::Ge,
376        right: lower,
377    } = left.as_ref()
378    else {
379        return None;
380    };
381    let Expr::Binary {
382        left: upper_projection,
383        op: BinaryOp::Le,
384        right: upper,
385    } = right.as_ref()
386    else {
387        return None;
388    };
389    let Expr::ArrayIndex { array, indices } = projection.as_ref() else {
390        return None;
391    };
392    let Expr::ColumnRef(column) = array.as_ref() else {
393        return None;
394    };
395    let [IndexElement::Single(Expr::Literal(Literal::Int(index)))] = indices.as_slice() else {
396        return None;
397    };
398    if column.table.is_some() || column.column != "UVW" || projection != upper_projection {
399        return None;
400    }
401    let axis = usize::try_from(index.checked_sub(i64::from(style.is_one_based()))?).ok()?;
402    (axis < 3).then_some((axis, real_literal(lower)?, real_literal(upper)?))
403}

405fn select_projected_uvw_rows(
406    table: &Table,
407    axis: usize,
408    range: std::ops::RangeInclusive<f64>,
409    rows_per_block: usize,
410) -> MsResult<Vec<usize>> {
411    let row_count = table.row_count();
412    let mut selected = Vec::new();
413    for start in (0..row_count).step_by(rows_per_block) {
414        let end = start.saturating_add(rows_per_block).min(row_count);
415        let rows = (start..end).collect::<Vec<_>>();
416        let values = super::columns::load_uvw_column(table, &rows)?;
417        selected.extend(
418            rows.into_iter()
419                .zip(values)
420                .filter_map(|(row, uvw)| range.contains(&uvw[axis]).then_some(row)),
421        );
422    }
423    Ok(selected)
424}

426pub(super) fn validate_predicate(
427    ms: &MeasurementSet,
428    query: &str,
429    rows_per_block: Option<usize>,
430) -> MsResult<(RecordExpressionType, Option<bool>)> {
431    if let Statement::Select(select) = selection_query(ms, query, rows_per_block)?
432        && let Some(predicate) = &select.where_clause
433    {
434        let kind =
435            taql::validate_record_expression(predicate, ms.main_table()).map_err(taql_error)?;
436        let constant = match taql::eval::eval_constant(predicate, "TaQL", select.style) {
437            Ok(taql::eval::ExprValue::Bool(value)) => Some(value),
438            _ => None,
439        };
440        return Ok((kind, constant));
441    }
442    Ok((RecordExpressionType::PossibleRowPredicate, None))
443}

445fn selection_query(
446    ms: &MeasurementSet,
447    query: &str,
448    rows_per_block: Option<usize>,
449) -> MsResult<Statement> {
450    let mut statement = taql::parse(query).map_err(taql_error)?;
451    if let Statement::Select(select) = &mut statement
452        && let Some(predicate) = &mut select.where_clause
453    {
454        let mut geometry = GeometryBinder::new(ms);
455        bind_selectors(predicate, ms, select.style, rows_per_block, &mut geometry)?;
456    }
457    Ok(statement)
458}
459
460fn bind_selectors(
461    expr: &mut Expr,
462    ms: &MeasurementSet,
463    style: IndexStyle,
464    rows_per_block: Option<usize>,
465    geometry: &mut GeometryBinder<'_>,
466) -> MsResult<()> {
467    if let Expr::FunctionCall { name, args } = expr {
468        let resolved = match resolve_selector(name, args, ms, style, rows_per_block)? {
469            Some(predicate) => Some(predicate),
470            None => geometry.bind(name, args, style)?,
471        };
472        if let Some(resolved) = resolved {
473            *expr = resolved;
474            return Ok(());
475        }
476    }
477    expr.try_for_each_child_mut(&mut |child| {
478        bind_selectors(child, ms, style, rows_per_block, geometry)
479    })
480}

482fn resolve_selector(
483    name: &str,
484    args: &[Expr],
485    ms: &MeasurementSet,
486    style: IndexStyle,
487    rows_per_block: Option<usize>,
488) -> MsResult<Option<Expr>> {
489    let lower = name.to_ascii_lowercase();
490    let Some(function) = lower
491        .strip_prefix("mscal.")
492        .or_else(|| lower.strip_prefix("derivedmscal."))
493    else {
494        return Ok(None);
495    };
496    if !matches!(
497        function,
498        "baseline" | "field" | "spw" | "array" | "uvdist" | "obs" | "state" | "time" | "scan"
499    ) {
500        return Ok(None);
501    }
502    let [arg] = args else {
503        return Err(taql_error(TaqlError::ArgumentCount {
504            name: name.to_owned(),
505            expected: "1".into(),
506            got: args.len(),
507        }));
508    };
509    let text = taql::eval::eval_constant_string(arg, name, style).map_err(taql_error)?;
510    if text.is_empty() {
511        return Err(MsError::InvalidInput(format!(
512            "{name} has an empty selector"
513        )));
514    }
515
516    let query = match function {
517        "baseline" => {
518            let baseline = ms.resolve_antenna_selector_with_warnings(&text)?;
519            format!(
520                "SELECT * WHERE {}",
521                row_grammar::baseline_predicate(&baseline)
522            )
523        }
524        "field" => {
525            let ids = ms.resolve_field_selector(&text)?;
526            format!("SELECT * WHERE FIELD_ID IN [{}]", int_list(&ids))
527        }
528        "state" => {
529            let ids = parser::parse_state_selector_with_mode(ms, &text, MatchMode::Taql)?;
530            if ids.is_empty() {
531                return Ok(Some(Expr::Literal(Literal::Bool(false))));
532            }
533            format!("SELECT * WHERE STATE_ID IN [{}]", int_list(&ids))
534        }
535        _ => {
536            let mut request = MsSelection::default();
537            match function {
538                "spw" => request.spw = Some(text),
539                "array" => request.array = Some(text),
540                "uvdist" => request.uvrange = Some(text),
541                "obs" => request.observation = Some(text),
542                "time" => request.timerange = Some(text),
543                "scan" => request.scan = Some(text),
544                _ => unreachable!("checked MSCAL selection function"),
545            }
546            request
547                .compile_with_row_grammar(ms, RowGrammar::Transform, rows_per_block)?
548                .to_taql()
549        }
550    };
551    let Statement::Select(select) = taql::parse(&query).map_err(taql_error)? else {
552        unreachable!("selector generates a TaQL SELECT")
553    };
554    Ok(Some(
555        select
556            .where_clause
557            .unwrap_or(Expr::Literal(Literal::Bool(true))),
558    ))
559}
