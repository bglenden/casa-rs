21#[derive(Clone, Copy)]
22#[doc(hidden)]
23pub enum Spectra {
24    None,
25    Weight,
26    WeightAndSigma,
27}
28
29impl Spectra {
30    fn columns(self) -> impl Iterator<Item = &'static str> {
31        let names: &[&str] = match self {
32            Self::None => &[],
33            Self::Weight => &["WEIGHT_SPECTRUM"],
34            Self::WeightAndSigma => &["WEIGHT_SPECTRUM", "SIGMA_SPECTRUM"],
35        };
36        names.iter().copied()
37    }
38}

40#[doc(hidden)]
41pub struct ImportColumns {
42    pub(super) writer: MeasurementSetWriteSession,
43    rows: usize,
44    category_tile: [usize; 4],
45    flag_category: Option<Vec<bool>>,
46    spectra: Spectra,
47}
48
49impl ImportColumns {
50    pub fn start(
51        path: &Path,
52        rows: usize,
53        visibility: [usize; 2],
54        tile: Vec<usize>,
55        categories: bool,
56        spectra: Spectra,
57    ) -> MsResult<Self> {
58        use MeasurementSetColumnStorage::{TiledColumn, TiledShape};
59        use StreamedTiledShapeValueType::{Bool, Complex32, Float32, Float64};
60
61        let category_tile = [tile[0], tile[1], 1, tile[2]];
62        let weight = [visibility[0]];
63        let shape = |name: &str, value_type, cell: &[usize], tile_shape, storage_manager| {
64            MeasurementSetArrayColumnPlan {
65                name: name.into(),
66                value_type,
67                shapes: vec![MeasurementSetArrayShapePlan {
68                    cell_shape: cell.to_vec(),
69                    row_count: rows,
70                    tile_shape,
71                }],
72                storage_manager,
73            }
74        };
75        let weight_tile = crate::ms::casa_weight_tile_shape(&tile);
76        let mut arrays = vec![
77            shape("DATA", Complex32, &visibility, tile.clone(), TiledShape),
78            shape("FLAG", Bool, &visibility, tile.clone(), TiledShape),
79            shape("WEIGHT", Float32, &weight, weight_tile.clone(), TiledShape),
80            shape("SIGMA", Float32, &weight, weight_tile, TiledShape),
81            shape(
82                "UVW",
83                Float64,
84                &[3],
85                crate::ms::casa_uvw_tile_shape(&tile),
86                TiledColumn,
87            ),
88        ];
89        for name in spectra.columns() {
90            arrays.push(shape(name, Float32, &visibility, tile.clone(), TiledShape));
91        }
92        let flag_category = categories.then(Vec::new);
93        if flag_category.is_some() {
94            arrays.push(shape(
95                "FLAG_CATEGORY",
96                Bool,
97                &[visibility[0], visibility[1], 3],
98                category_tile.to_vec(),
99                TiledShape,
100            ));
101        }
102        let mut scalars = standard_main_scalar_column_plans();
103        scalars.retain(|column| {
104            !CONSTANT_COLUMNS
105                .iter()
106                .any(|(name, _)| *name == column.name)
107        });
108        let resources = MeasurementSetWriteResources::from_system_memory(2).map_err(invalid)?;
109        let plan =
110            MeasurementSetWritePlan::variable_array_creation(rows, arrays, scalars, resources)
111                .map_err(invalid)?;
112        let writer = MeasurementSetWriteSession::start(path, plan).map_err(invalid)?;
113        Ok(Self {
114            writer,
115            rows,
116            category_tile,
117            flag_category,
118            spectra,
119        })
120    }
121
122    pub fn append(&mut self, row: MainRow, samples: MainSamples<'_>) -> MsResult<()> {
123        use ScalarValue::{Bool, Float64, Int32};
124        let flags = samples
125            .flags
126            .as_slice_memory_order()
127            .expect("contiguous FITS flags");
128        for (name, value) in [
129            ("ANTENNA1", Int32(row.antennas[0])),
130            ("ANTENNA2", Int32(row.antennas[1])),
131            ("ARRAY_ID", Int32(row.array)),
132            ("DATA_DESC_ID", Int32(row.window)),
133            ("FIELD_ID", Int32(row.field)),
134            ("SCAN_NUMBER", Int32(row.scan)),
135            ("FLAG_ROW", Bool(flags.iter().all(|&value| value))),
136            ("TIME", Float64(row.time)),
137            ("TIME_CENTROID", Float64(row.centroid)),
138        ] {
139            self.writer
140                .push_scalar_row(name, Some(value))
141                .map_err(invalid)?;
142        }
143        if let Some(interval) = row.interval {
144            for name in ["INTERVAL", "EXPOSURE"] {
145                self.writer
146                    .push_scalar_row(name, Some(Float64(interval)))
147                    .map_err(invalid)?;
148            }
149        }
150        self.writer
151            .push_f64_row("UVW", &[3], &row.uvw)
152            .map_err(invalid)?;
153        self.append_samples(samples, flags)
154    }
155
156    fn append_samples(&mut self, samples: MainSamples<'_>, flags: &[bool]) -> MsResult<()> {
157        let data = samples.data;
158        let shape = data.shape();
159        self.writer
160            .push_complex32_row(
161                "DATA",
162                shape,
163                data.as_slice_memory_order().expect("contiguous FITS data"),
164            )
165            .map_err(invalid)?;
166        self.writer
167            .push_bool_row("FLAG", shape, flags)
168            .map_err(invalid)?;
169        if let Some(category) = &mut self.flag_category {
170            category.clear();
171            category.extend_from_slice(flags);
172            category.resize(3 * flags.len(), false);
173            self.writer
174                .push_bool_row("FLAG_CATEGORY", &[shape[0], shape[1], 3], category)
175                .map_err(invalid)?;
176        }
177        for (name, values) in [("WEIGHT", samples.weight), ("SIGMA", samples.sigma)] {
178            self.writer
179                .push_f32_row(
180                    name,
181                    values.shape(),
182                    values.as_slice().expect("contiguous correlation vector"),
183                )
184                .map_err(invalid)?;
185        }
186        for (name, values) in [
187            ("WEIGHT_SPECTRUM", samples.weight_spectrum),
188            ("SIGMA_SPECTRUM", samples.sigma_spectrum),
189        ] {
190            if let Some(values) = values {
191                self.writer
192                    .push_f32_row(
193                        name,
194                        shape,
195                        values
196                            .as_slice_memory_order()
197                            .expect("contiguous FITS spectrum"),
198                    )
199                    .map_err(invalid)?;
200            }
201        }
202        Ok(())
203    }
204
205    pub fn finish(self, ms: &mut MeasurementSet, path: &Path) -> MsResult<()> {
206        let mut overrides = ColumnOverrides::for_row_count(self.rows);
207        for name in ["DATA", "FLAG", "WEIGHT", "SIGMA", "UVW"]
208            .into_iter()
209            .chain(self.spectra.columns())
210        {
211            overrides.insert_deferred(name);
212        }
213        overrides.insert_deferred("FLAG_CATEGORY");
214        for (name, value) in CONSTANT_COLUMNS {
215            overrides.insert_generated_scalar(
216                name,
217                GeneratedScalarColumn::constant(self.rows, Some(ScalarValue::Int32(value))),
218            );
219        }
220        self.writer
221            .save_and_finish(ms, &overrides)
222            .map_err(invalid)?;
223        if self.flag_category.is_some() {
224            return Ok(());
225        }
226        let seq =
227            crate::ms::measurement_set_main_data_manager_sequence(ms.main_table(), "FLAG_CATEGORY")
228                .ok_or_else(|| invalid("Missing FLAG_CATEGORY storage manager"))?;
229        install_undefined_tiled_shape_column(
230            path,
231            seq,
232            "FLAG_CATEGORY",
233            self.rows,
234            StreamedTiledPrimitiveType::Bool,
235            3,
236            &self.category_tile,
237        )
238        .map_err(invalid)
239    }
240}

247    pub centroid: f64,

255#[doc(hidden)]
256pub struct MainSamples<'a> {
257    pub data: &'a Array2<Complex32>,
258    pub flags: &'a Array2<bool>,
259    pub weight: &'a Array1<f32>,
260    pub sigma: &'a Array1<f32>,
261    pub weight_spectrum: Option<&'a Array2<f32>>,
262    pub sigma_spectrum: Option<&'a Array2<f32>>,
263}
