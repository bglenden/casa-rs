16pub(crate) fn context(resources: SessionResources) -> RunContext {
17    RunContext {
18        host: *resources.host(),
19        policy: *resources.policy(),
20        cancel: Cancel::new(),
21        summary: None,
22    }
23}

25pub(super) fn execute(
26    p: &Parameters<'_>,
27    plan: &Plan,
28    resources: SessionResources,
29) -> Result<Vec<PathBuf>, SimobserveError> {
30    preflight::check(p, plan)?;
31    let mut products = Vec::new();
32    let mut parameters = plan.imaging.clone();
33    let request = if plan.inputs.synthesis.is_empty() {
34        None
35    } else {
36        let mut request =
37            ImagingRequest::from_tclean_with_defaults(&plan.imaging).map_err(failure)?;
38        request.backend = match resources.backend() {
39            crate::BackendRequest::Cpu => casa_imaging_application::BackendRequest::Cpu,
40            crate::BackendRequest::Metal => casa_imaging_application::BackendRequest::Gpu,
41            crate::BackendRequest::Auto => casa_imaging_application::BackendRequest::Auto,
42        };
43        Some(request)
44    };
45    if plan.inputs.synthesis.len() > 1 {
46        if plan.ms.exists() {
47            if !p.flag("overwrite") {
48                return Err(failure("concat MS already exists"));
49            }
50            MeasurementSet::open(&plan.ms).map_err(failure)?;
51        }
52        let staging = tempfile::tempdir_in(&plan.directory).map_err(failure)?;
53        let target = staging.path().join("concat.ms");
54        let memory = resources
55            .policy()
56            .memory(resources.host())
57            .saturating_sub(casa_imaging_runtime::reserved_memory());
58        let _memory = admit(
59            resources.host(),
60            resources.policy(),
61            &Demand {
62                phase: "simulation concatenation",
63                memory,
64            },
65        )
66        .map_err(failure)?;
67        concat_with_memory(
68            &ConcatRequest {
69                vis: plan.inputs.synthesis.clone(),
70                concatvis: target.clone(),
71                freqtol: json!(""),
72                dirtol: json!(""),
73                respectname: false,
74                timesort: false,
75                copypointing: true,
76                visweightscale: plan.concat_weights.clone(),
77                forcesingleephemfield: json!(""),
78            },
79            usize::try_from(memory).map_err(failure)?,
80        )
81        .map_err(failure)?;
82        casa_imaging_products::publish_staged_image(&target, &plan.ms).map_err(failure)?;
83        products.push(plan.ms.clone());
84        // Concat merges matched source rows. Select the gridder from its
85        // actual catalogue, rather than counting duplicate input fields.
86        let ms = MeasurementSet::open(&plan.ms).map_err(failure)?;
87        let source = ms
88            .subtable(SubtableId::Source)
89            .ok_or_else(|| failure("no SOURCE table"))?;
90        let codes = source.column_accessor("CODE").map_err(failure)?;
91        let fields = (0..source.row_count())
92            .filter(
93                |&row| matches!(codes.scalar_cell(row), Ok(ScalarValue::String(v)) if v == "OBJ"),
94            )
95            .count();
96        parameters.insert(
97            "gridder".into(),
98            json!(if fields == 1 { "standard" } else { "mosaic" }),
99        );
100    }
101    if let Some(input) = &plan.inputs.total_power {
102        products.extend(total_power::image(input, p, plan, resources)?);
103    }
104    if let Some(initial) = request {
105        let _seed = start_model(p, plan, &mut parameters, resources)?;
106        let mut request =
107            ImagingRequest::from_tclean_with_defaults(&parameters).map_err(failure)?;
108        request.backend = initial.backend;
109        casa_imaging_application::execute(&request, context(resources)).map_err(failure)?;
110    }
111    Ok(products)
112}

114// simanalyze passes seed pixels to tclean with a Jy/pixel label. Preserve
115// CASA's numerical seed convention in a private copy; never relabel the
116// caller's Jy/beam image in place, as the old CASA task does.
117fn start_model(
118    p: &Parameters<'_>,
119    plan: &Plan,
120    parameters: &mut serde_json::Map<String, serde_json::Value>,
121    resources: SessionResources,
122) -> Result<Option<tempfile::TempDir>, SimobserveError> {
123    let supplied = super::plan::expand_project(&plan.directory, p.text("modelimage"));
124    let path = super::plan::locate(&plan.directory, &supplied);
125    if supplied.is_empty() || !path.exists() {
126        parameters.insert("startmodel".into(), json!(""));
127        return Ok(None);
128    }
129    let image = PagedImage::<f32>::open(&path).map_err(failure)?;
130    if image.units() == "Jy/pixel" {
131        parameters.insert("startmodel".into(), json!(path));
132        return Ok(None);
133    }
134    if image.units() != "Jy/beam" {
135        return Err(failure(format!(
136            "modelimage unit {:?} is not Jy/pixel or Jy/beam",
137            image.units()
138        )));
139    }
140    let memory = image
141        .shape()
142        .iter()
143        .try_fold(16u64, |bytes, &axis| bytes.checked_mul(axis as u64))
144        .and_then(|bytes| bytes.checked_add(32 << 20))
145        .ok_or_else(|| failure("seed model memory overflows"))?;
146    let _memory = admit(
147        resources.host(),
148        resources.policy(),
149        &Demand {
150            phase: "simulation seed model",
151            memory,
152        },
153    )
154    .map_err(failure)?;
155    let staging = tempfile::tempdir_in(&plan.directory).map_err(failure)?;
156    let target = staging.path().join("model");
157    analysis::write_product(
158        &target,
159        &image.get().map_err(failure)?,
160        image.coordinates().clone(),
161        "Jy/pixel",
162        &image.image_info().map_err(failure)?,
163        image.default_mask().map_err(failure)?.as_ref(),
164    )
165    .map_err(failure)?;
166    parameters.insert("startmodel".into(), json!(target));
167    Ok(Some(staging))
168}
