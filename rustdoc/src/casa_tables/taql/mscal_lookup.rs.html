13pub(super) fn bind(expr: &mut Expr, table: &Table, style: IndexStyle) -> Result<(), TaqlError> {
14    expr.try_for_each_child_mut(&mut |child| bind(child, table, style))?;
15    if let Expr::FunctionCall { name, args } = expr
16        && let Some(lookup) = bind_lookup(name, args, table, style)?
17    {
18        *expr = lookup;
19    }
20    Ok(())
21}

23fn bind_lookup(
24    name: &str,
25    args: &mut [Expr],
26    table: &Table,
27    style: IndexStyle,
28) -> Result<Option<Expr>, TaqlError> {
29    let lower = name.to_ascii_lowercase();
30    let Some(function) = lower
31        .strip_prefix("mscal.")
32        .or_else(|| lower.strip_prefix("derivedmscal."))
33    else {
34        return Ok(None);
35    };
36    let original = format!(
37        "{name}({})",
38        args.iter()
39            .map(ToString::to_string)
40            .collect::<Vec<_>>()
41            .join(", ")
42    );
43
44    if function == "stokes" {
45        for argument in args.iter_mut().skip(1) {
46            super::mscal_help::fold_filter(argument, table, style)?;
47        }
48        return super::mscal_stokes::bind(name, original, args, table, style)
49            .map(|function| Some(Expr::BoundRowFunction(function)));
50    }
51
52    if matches!(function, "uvwwvl" | "uvwwvls") {
53        check_arity(name, args, 0)?;
54        return super::mscal_wavelengths::bind(original, table, function == "uvwwvls")
55            .map(|function| Some(Expr::BoundRowFunction(function)));
56    }
57
58    let (subtable, column, index, allow_missing) = if function == "subcol" {
59        check_arity(name, args, 3)?;
60        (
61            constant_string(&args[0], name, style)?,
62            constant_string(&args[1], name, style)?,
63            args[2].clone(),
64            false,
65        )
66    } else {
67        let (subtable, id, fixed_column) = match function {
68            "ant1name" => ("ANTENNA", "ANTENNA1", Some("NAME")),
69            "ant2name" => ("ANTENNA", "ANTENNA2", Some("NAME")),
70            "ant1col" => ("ANTENNA", "ANTENNA1", None),
71            "ant2col" => ("ANTENNA", "ANTENNA2", None),
72            "fieldcol" => ("FIELD", "FIELD_ID", None),
73            "spwcol" => ("SPECTRAL_WINDOW", "SPECTRAL_WINDOW_ID", None),
74            "polcol" => ("POLARIZATION", "POLARIZATION_ID", None),
75            "statecol" => ("STATE", "STATE_ID", None),
76            "proccol" => ("PROCESSOR", "PROCESSOR_ID", None),
77            // The pinned CASA UDF looks up this literal keyword, including the space.
78            "obscol" => (" OBSERVATION", "OBSERVATION_ID", None),
79            _ => return Ok(None),
80        };
81        check_arity(name, args, usize::from(fixed_column.is_none()))?;
82        let column = match fixed_column {
83            Some(column) => column.to_owned(),
84            None => constant_string(&args[0], name, style)?,
85        };
86        let index = match function {
87            "spwcol" | "polcol" => {
88                let dd = open_subtable(table, "DATA_DESCRIPTION")?;
89                bind_column(&original, column_ref("DATA_DESC_ID"), &dd, id, false)?
90            }
91            _ => column_ref(id),
92        };
93        (
94            subtable.to_owned(),
95            column,
96            index,
97            matches!(function, "statecol" | "proccol"),
98        )
99    };
100
101    let subtable = open_subtable(table, &subtable)?;
102    let lookup = bind_column(&original, index, &subtable, &column, allow_missing)?;
103    Ok(Some(lookup))
104}

106fn check_arity(name: &str, args: &[Expr], expected: usize) -> Result<(), TaqlError> {
107    if args.len() != expected {
108        return Err(TaqlError::ArgumentCount {
109            name: name.to_owned(),
110            expected: expected.to_string(),
111            got: args.len(),
112        });
113    }
114    Ok(())
115}

117fn constant_string(expr: &Expr, name: &str, style: IndexStyle) -> Result<String, TaqlError> {
118    let value = eval_constant_string(expr, name, style)?;
119    if value.is_empty() {
120        return Err(TaqlError::TypeError {
121            message: format!("{name} requires a nonempty metadata name"),
122        });
123    }
124    Ok(value)
125}

127fn column_ref(name: &str) -> Expr {
128    Expr::ColumnRef(ColumnRef {
129        table: None,
130        column: name.to_owned(),
131    })

134pub(super) fn open_subtable(table: &Table, name: &str) -> Result<Table, TaqlError> {
135    let stored_path = table
136        .keywords()
137        .get(name)
138        .and_then(Value::as_table_ref)
139        .ok_or_else(|| TaqlError::Table(format!("subtable keyword {name:?} is missing")))?;
140    let table_path = table.path().ok_or_else(|| {
141        TaqlError::Table(format!(
142            "cannot resolve subtable {name:?} without a table path"
143        ))
144    })?;
145    Table::open(TableOptions::new(add_directory(stored_path, table_path))).map_err(table_error)
146}

148fn bind_column(
149    original: &str,
150    index: Expr,
151    table: &Table,
152    name: &str,
153    allow_missing: bool,
154) -> Result<Expr, TaqlError> {
155    let schema = table
156        .schema()
157        .and_then(|schema| schema.column(name))
158        .ok_or_else(|| TaqlError::ColumnNotFound {
159            name: name.to_owned(),
160        })?;
161    if schema.column_type() == &ColumnType::Record {
162        return Err(TaqlError::TypeError {
163            message: format!("MSCAL metadata lookup cannot return record column {name:?}"),
164        });
165    }
166    let column = table.column_accessor(name).map_err(table_error)?;
167    let mut values: Vec<_> = match schema.column_type() {
168        ColumnType::Scalar => column
169            .scalar_cells_owned()
170            .map_err(table_error)?
171            .iter()
172            .map(|value| value.as_ref().map_or(ExprValue::Null, ExprValue::from))
173            .collect(),
174        _ => column
175            .array_cells_owned(&(0..table.row_count()).collect::<Vec<_>>())
176            .map_err(table_error)?
177            .into_iter()
178            .map(|value| {
179                value.map_or(ExprValue::Null, |array| {
180                    ExprValue::from(&Value::Array(array))
181                })
182            })
183            .collect(),
184    };
185    let mut missing = (allow_missing && schema.column_type() == &ColumnType::Scalar)
186        .then(|| missing_value(schema));
187    if let Some(unit) = column_unit(table, name)? {
188        for value in values.iter_mut().chain(missing.iter_mut()) {
189            apply_unit(value, &unit);
190        }
191    }
192    Ok(Expr::BoundRowLookup {
193        original: original.to_owned().into(),
194        index: Box::new(index),
195        values: values.into(),
196        missing,
197    })
198}

200fn column_unit(table: &Table, name: &str) -> Result<Option<Unit>, TaqlError> {
201    use casa_core::{ArrayValue, ScalarValue};
202
203    let Some(keywords) = table.column_keywords(name) else {
204        return Ok(None);
205    };
206    // TableExprNodeColumn uses the first fixed unit or the legacy UNIT keyword.
207    let unit = match keywords.get("QuantumUnits") {
208        Some(Value::Array(ArrayValue::String(units))) => units.iter().next(),
209        Some(_) => {
210            return Err(TaqlError::TypeError {
211                message: format!("QuantumUnits of metadata column {name:?} must be a string array"),
212            });
213        }
214        None => match keywords.get("UNIT") {
215            Some(Value::Scalar(ScalarValue::String(unit))) => Some(unit),
216            _ => None,
217        },
218    };
219    unit.filter(|unit| !unit.is_empty())
220        .map(|unit| {
221            Unit::new(unit).map_err(|err| TaqlError::TypeError {
222                message: err.to_string(),
223            })
224        })
225        .transpose()
226}
227
228fn apply_unit(value: &mut ExprValue, unit: &Unit) {
229    match value {
230        ExprValue::Float(number) => {
231            *value = ExprValue::Quantity(Quantity::with_unit(*number, unit.clone()));
232        }
233        ExprValue::Int(number) => {
234            *value = ExprValue::Quantity(Quantity::with_unit(*number as f64, unit.clone()));
235        }
236        ExprValue::Array(array) => {
237            for element in &mut array.data {
238                apply_unit(element, unit);
239            }
240        }
241        _ => {}
242    }
243}
244
245fn missing_value(schema: &ColumnSchema) -> ExprValue {
246    match schema.data_type() {
247        Some(PrimitiveType::Bool) => ExprValue::Bool(false),
248        Some(PrimitiveType::String) => ExprValue::String(String::new()),
249        Some(PrimitiveType::Float32 | PrimitiveType::Float64) => ExprValue::Float(0.),
250        Some(PrimitiveType::Complex32 | PrimitiveType::Complex64) => ExprValue::Complex(0.0.into()),
251        Some(_) => ExprValue::Int(0),
252        None => ExprValue::Null,
253    }
254}
