2//! Visibility prediction from an external sky model, as CASA's simulator
3//! tool predicts it (`sm.predict` with its default `GridFT`).
4//!
5//! CASA's `Simulator::predict` degrids the model through a `GridFT`
6//! FTMachine:
7//! - The grid is padded by `setoptions(padding=…)`, 1.3 by default.
8//! - The tangent point is the direction of pixel `(nx/2, ny/2)`, whatever
9//!   the image's reference pixel is (`FTMachine::initMaps`), and every row is
10//!   rotated to it (`FTMachine::rotateUVW`).
11//! - Each MeasurementSet channel takes the nearest model plane, with no
12//!   frequency-frame conversion.
13//! - Autocorrelations are predicted, as the model's zero spacing (`GridFT`'s
14//!   `usezero` is on by default).
15//! - After `setvp(dovp=True)`, a `VPSkyJones` multiplies the model by the
16//!   primary beam's power pattern in the image plane, centred on the
17//!   pointing.
18//!
19//! The model is prepared on the imaging measurement operator, one plane at
20//! a time ([`SkyModel`] reads a plane, the beam multiplies it, the operator
21//! transforms it into the plane's grids):
22//! - the spheroidal kernel set on a [`GridPadding::CasaComposite`] grid;
23//! - a channel-local basis with nearest mapping, predicting
24//!   autocorrelations;
25//! - the beam as the plane's prescale factor.
26//!
27//! The session predicts the rows it writes from the same preparation, in
28//! blocks on the worker team; [`ModelPredictor`] predicts one row at a time
29//! and is the reference the session is tested against.
30//!
31//! The model must be Stokes I, J2000 and in the axis order `simobserve`
32//! writes.

54/// A failure to read a sky model or to predict from it.
55#[derive(Debug, Error)]
56pub enum PredictError {
57    /// The model image could not be read.
58    #[error("model image {path}: {source}")]
59    Image {
60        /// The image path.
61        path: String,
62        /// The image error.
63        #[source]
64        source: ImageError,
65    },
66    /// A pixel could not be converted to a world coordinate, or back.
67    #[error("model image coordinates: {0}")]
68    Coordinates(#[from] CoordinateError),
69    /// The model image has a shape or coordinate system this predictor does
70    /// not support yet.
71    #[error("unsupported model image: {0}")]
72    UnsupportedModel(String),
73    /// The measurement operator refused the model or a row.
74    #[error(transparent)]
75    Operator(#[from] OperatorError),
76}

78/// The measurement operator and spectral resampler that predict one model
79/// on rows of one correlation set, as CASA's `GridFT` does, and the
80/// preparation of the model's grids. The session and [`ModelPredictor`]
81/// share it, so both predict from the same grids.
82pub(crate) struct ModelOperator {
83    pub(crate) operator: MeasurementOperator,
84    pub(crate) resampler: SpectralResampler,
85}
86
87impl ModelOperator {
88    /// The operator for `model` on rows with `correlations`, with the grid
89    /// padded by `padding` and in `precision`.
90    ///
91    /// # Errors
92    ///
93    /// A geometry the grid cannot hold, or correlations the Stokes I model
94    /// cannot feed.
95    pub(crate) fn new(
96        model: &SkyModel,
97        correlations: &[CorrelationType],
98        padding: PaddingFactor,
99        precision: GridPrecision,
100    ) -> Result<Self, PredictError> {
101        let [nx, ny] = model.shape();
102        let geometry = GridGeometry::new(
103            ImageExtent {
104                shape: [nx, ny],
105                increment_rad: model.increment_rad,
106                reference_pixel: [nx / 2, ny / 2],
107            },
108            GridPadding::CasaComposite(padding),
109        )?;
110        let polarization =
111            PolarizationRouting::compile(correlations, &[PolarizationCoordinate::StokesI])?;
112        let planes = u32::try_from(model.planes())
113            .map_err(|_| PredictError::UnsupportedModel("too many model planes".into()))?;
114        let axis = SpectralAxis::new(
115            model.first_frequency_hz,
116            model.frequency_increment_hz,
117            planes,
118        )?;
119        let resampler = SpectralResampler::channel_local(axis, SpectralKernel::Nearest)
120            .predicting_autocorrelations();
121        let kernels = Spheroidal::new(&geometry, &polarization);
122        let operator = MeasurementOperator::new(
123            geometry,
124            resampler.basis(),
125            polarization,
126            Box::new(kernels),
127            precision,
128        );
129        Ok(Self {
130            operator,
131            resampler,
132        })
133    }
134
135    /// The planes of the model's grids: every plane of the basis.
136    fn planes(&self) -> PlaneRange {
137        PlaneRange::new(0, self.operator.basis().planes())
138    }
139
140    /// Bytes one set of the model's grids holds
141    /// ([`PreparedModelGrids::bytes`]), known before they are allocated.
142    pub(crate) fn grid_bytes(&self) -> u64 {
143        self.operator
144            .accumulator_layout(self.planes(), None, ModeSet::DATA)
145            .bytes(self.operator.precision()) as u64
146    }
147
148    /// Bytes one worker holds while it prepares a plane, beside the grids:
149    /// the plane read and its slice, the beam's power on it, and the
150    /// transform's padded plane.
151    pub(crate) fn plane_scratch_bytes(&self, model: &SkyModel) -> u64 {
152        let [nx, ny] = model.shape();
153        let [gx, gy] = self.operator.geometry().grid_shape();
154        let cell = match self.operator.precision() {
155            GridPrecision::F32 => size_of::<Complex32>(),
156            GridPrecision::F64 => 2 * size_of::<Complex32>(),
157        };
158        model.plane_bytes() + (nx * ny * size_of::<f32>() + 2 * gx * gy * cell) as u64
159    }
160
161    /// The model's grids: each plane read, multiplied by `beam`'s power
162    /// pattern centred on its pixel when there is one, and transformed into
163    /// the plane's grids. Planes are prepared on `team`'s workers, one plane
164    /// per job, or in turn on the caller without one; the grids do not
165    /// depend on which.
166    ///
167    /// # Errors
168    ///
169    /// A plane that cannot be read, or the operator's refusal.
170    pub(crate) fn prepare(
171        &self,
172        model: &SkyModel,
173        beam: Option<(&BeamPattern, [f64; 2])>,
174        team: Option<&WorkerTeam>,
175    ) -> Result<PreparedModelGrids, PredictError> {
176        let mut grids = self.operator.empty_model_grids(self.planes())?;
177        let mut jobs: Vec<Option<ModelPlaneGrids<'_>>> =
178            grids.planes_mut().into_iter().map(Some).collect();
179        let prepare = |plane: usize, target: &mut Option<ModelPlaneGrids<'_>>| {
180            let Some(target) = target.take() else {
181                return Ok(());
182            };
183            let images = vec![model.read_plane(plane)?];
184            let factor = beam.map(|(pattern, centre)| pattern.power_plane(model, plane, centre));
185            self.operator
186                .fill_model_plane(&ModelPlane { images }, factor.as_ref(), target)?;
187            Ok::<_, PredictError>(())
188        };
189        match team {
190            Some(team) => team.for_each_mut(&mut jobs, prepare)?,
191            None => jobs
192                .iter_mut()
193                .enumerate()
194                .try_for_each(|(plane, target)| prepare(plane, target))?,
195        }
196        drop(jobs);
197        debug_assert_eq!(grids.bytes(), self.grid_bytes());
198        Ok(grids)
199    }
200}

202/// How [`ModelPredictor`] grids, as `sm.setoptions` and `sm.setvp` choose.
203#[derive(Clone, Copy, Debug, PartialEq)]
204pub struct PredictOptions {
205    /// Grid padding (`setoptions(padding=…)`).
206    pub padding: PaddingFactor,
207    /// The primary beam, when `setvp(dovp=True)`.
208    pub primary_beam: Option<PrimaryBeam>,
209    /// Grid arithmetic. CASA's simulator `GridFT` runs in `Float`.
210    pub precision: GridPrecision,
211}
212
213impl Default for PredictOptions {
214    fn default() -> Self {
215        Self {
216            padding: PaddingFactor::SIMULATOR,
217            primary_beam: None,
218            precision: GridPrecision::F32,
219        }
220    }
221}

223/// One MeasurementSet row to predict.
224#[derive(Clone, Copy, Debug)]
225pub struct PredictRow<'a> {
226    /// Baseline coordinates in metres towards the model's tangent point, in
227    /// GridFT's convention: the row's `UVW` rotated by
228    /// [`casa_ms::derived::GridFtProjection`].
229    pub uvw_m: [f64; 3],
230    /// GridFT's phase shift in metres (`−dphase`), from the same rotation.
231    pub phase_shift_m: f64,
232    /// Channel frequencies in Hz.
233    pub frequencies_hz: &'a [f64],
234    /// Sample flags, `[channel][correlation]`.
235    pub flags: &'a [bool],
236    /// The row flag.
237    pub row_flag: bool,
238    /// Antenna pair; an autocorrelation predicts the model's zero spacing.
239    pub antennas: [u32; 2],
240}

242/// Reusable per-thread buffers for [`ModelPredictor::predict_row`].
243#[derive(Debug, Default)]
244pub struct PredictionWorkspace {

246    backend: CpuBackend,
247    values: Vec<Complex32>,
248    weights: Vec<f32>,
249}

251/// A model prepared for prediction on one correlation set, predicting a row
252/// at a time on the CPU: the reference the session's block prediction is
253/// tested against.
254pub struct ModelPredictor {
255    model: ModelOperator,
256    grids: PreparedModelGrids,
257}
258
259impl ModelPredictor {
260    /// Prepare `model` for rows with `correlations`, its primary beam (when
261    /// requested) centred on `pointing_rad`.
262    ///
263    /// # Errors
264    ///
265    /// A geometry the grid cannot hold, correlations the Stokes I model
266    /// cannot feed, a pointing outside the model's coordinate system, or a
267    /// plane that cannot be read.
268    pub fn new(
269        model: &SkyModel,
270        correlations: &[CorrelationType],
271        pointing_rad: [f64; 2],
272        options: &PredictOptions,
273    ) -> Result<Self, PredictError> {
274        let operator = ModelOperator::new(model, correlations, options.padding, options.precision)?;
275        let pattern = options.primary_beam.map(PrimaryBeam::pattern);
276        let beam = match &pattern {
277            Some(pattern) => Some((pattern, model.pixel_of(pointing_rad)?)),
278            None => None,
279        };
280        let grids = operator.prepare(model, beam, None)?;
281        Ok(Self {
282            model: operator,
283            grids,
284        })
285    }
286
287    /// Correlations per sample.
288    #[must_use]
289    pub fn correlations(&self) -> usize {
290        self.model.operator.polarization().correlations().len()
291    }
292
293    /// The model visibility of every sample of `row`, written to `out` as
294    /// `[channel][correlation]`. Flagged samples and flagged rows predict
295    /// zero; an autocorrelation predicts the model's zero spacing.
296    ///
297    /// # Errors
298    ///
299    /// A row inconsistent with the predictor, or an `out` of the wrong
300    /// length.
301    pub fn predict_row(
302        &self,
303        workspace: &mut PredictionWorkspace,
304        row: &PredictRow<'_>,
305        out: &mut [Complex32],
306    ) -> Result<(), PredictError> {
307        let cells = row.frequencies_hz.len() * self.correlations();
308        workspace.values.resize(cells, Complex32::new(0.0, 0.0));
309        workspace.weights.resize(cells, 1.0);
310        let native = NativeRow {
311            uvw_m: row.uvw_m,
312            phase_shift_m: row.phase_shift_m,
313            pointing_offset_rad: [0.0; 2],
314            frequencies_hz: row.frequencies_hz,
315            values: &workspace.values,
316            weights: &workspace.weights,
317            flags: row.flags,
318            row_flag: row.row_flag,
319            context: RowContext {
320                original_uvw_m: None,
321                time_s: 0.0,
322                antennas: row.antennas,
323                antenna_types: [0, 0],
324                parallactic_angle_rad: [0.0, 0.0],
325                field: 0,
326                spectral_window: 0,
327            },
328        };
329        self.model.resampler.predict_row(
330            &self.model.operator,
331            &mut workspace.backend,
332            &self.grids,
333            &native,
334            &mut workspace.scratch,
335            out,
336        )?;
337        Ok(())
338    }
