2//! The `table` tool's calls that report or move a table's state, as CASA
3//! 6.7.6.14's `tb` does them (`tools/casa-compat/table_tool_oracle.py`,
4//! scenarios `state` and `state_ms`): `flush`, `resync`, `datachanged`,
5//! `ismultiused`, `endianformat`, `setmaxcachesize` and `clearlocks`.
6//!
7//! `flush` has nothing to write: every change the tool makes is on the disk
8//! when its call returns (`write.rs`); a table that keeps its write lock
9//! announces them, as casacore's flush does. `setmaxcachesize` checks its
10//! arguments as CASA does and sets nothing, since casa-tables sizes the
11//! caches of its storage managers itself.
12
13use casa_tables::{EndianFormat, LockType, Table};
14use casa_types::{RecordValue, Value};
15
16use super::registry;
17use super::state::{Fail, State, no_table_specified};
18use crate::convert::value::CasaValue;
19use crate::tools::LogLine;
20
21type Reply = Result<CasaValue, Fail>;
22
23/// Whether `record` holds a reference to another table, at any depth.
24fn holds_table_reference(record: &RecordValue) -> bool {
25    record.fields().iter().any(|field| match &field.value {
26        Value::TableRef(_) => true,
27        Value::Record(inner) => holds_table_reference(inner),
28        _ => false,
29    })
30}
31
32/// Whether `table` has subtables: a table-valued keyword of its own or of a
33/// column.
34pub(super) fn has_subtables(table: &Table) -> bool {
35    holds_table_reference(table.keywords())
36        || table.schema().is_some_and(|schema| {
37            schema.columns().iter().any(|column| {
38                table
39                    .column_keywords(column.name())
40                    .is_some_and(holds_table_reference)
41            })
42        })
43}

45impl State {
46    /// `endianformat()`: the byte order of the table's files; `little`
47    /// with no table open.
48    pub(super) fn endianformat(&self, log: &mut Vec<LogLine>) -> Reply {
49        let big = match self.open.as_ref() {
50            Some(open) => match open.table.stored_endian_format()? {
51                Some(format) => format == EndianFormat::BigEndian,
52                None => {
53                    return Err(Fail::Unsupported(
54                        "table.endianformat of a table that is a concatenation of tables is not \
55                         yet implemented in casa-rs (#1028)"
56                            .to_string(),
57                    ));
58                }
59            },
60            None => {
61                no_table_specified(log);
62                false
63            }
64        };
65        Ok(CasaValue::Text(
66            if big { "big" } else { "little" }.to_string(),
67        ))
68    }

70    /// `flush()`. Every change reaches the disk when the call that makes it
71    /// returns; what is left is the announcement of them to the table's other
72    /// users, which a table holding its write lock (the user and permanent
73    /// modes) makes now, as casacore's flush does, and keeps the lock.
74    pub(super) fn flush(&mut self, log: &mut Vec<LogLine>) -> Reply {
75        let Some(open) = self.open.as_mut() else {
76            no_table_specified(log);
77            return Ok(CasaValue::Bool(false));
78        };
79        open.table.publish_writes()?;
80        Ok(CasaValue::Bool(true))
81    }
82
83    /// `resync()`: the table takes what another process, or another tool of
84    /// this process, has written since it last looked. casacore reads the
85    /// lock file without locking; here a table that holds a lock reads it
86    /// again and reloads if it has changed, and a table that holds none takes
87    /// the read lock, which reloads it if it has changed, and gives it back,
88    /// so its lock state is as it was. A table that cannot be locked is not
89    /// resynchronized, and the call says so.
90    pub(super) fn resync(&mut self, log: &mut Vec<LogLine>) -> Reply {
91        let Some(open) = self.open.as_mut() else {
92            no_table_specified(log);
93            return Ok(CasaValue::Bool(false));
94        };
95        if open.table.has_lock(LockType::Read) {
96            open.table.resync_locked()?;
97        } else {
98            if !open.table.lock(LockType::Read, open.lock.attempts())? {
99                return Err(Fail::Runtime(
100                    "Table::resync: the table could not be locked to read what has changed in it"
101                        .to_string(),
102                ));
103            }
104            open.table.unlock_metadata_only()?;
105        }
106        Ok(CasaValue::Bool(true))
107    }
108
109    /// `datachanged()`, as casacore's `Table::hasDataChanged`: a table that
110    /// holds no lock tries for a read lock without waiting (another process
111    /// writing means the data is changing) and gives it back; then the
112    /// modify counter the lock file publishes is compared with the one the
113    /// last call saw.
114    pub(super) fn datachanged(&mut self, log: &mut Vec<LogLine>) -> Reply {
115        let Some(open) = self.open.as_mut() else {
116            no_table_specified(log);
117            return Ok(CasaValue::Bool(false));
118        };
119        let took_lock = !open.table.has_lock(LockType::Read);
120        if took_lock && !open.table.lock(LockType::Read, 1)? {
121            return Ok(CasaValue::Bool(true));
122        }
123        let counter = open.table.published_modify_counter();
124        if took_lock {
125            open.table.unlock_metadata_only()?;
126        }
127        let counter = counter?;
128        let changed = counter != open.last_modify_counter;
129        open.last_modify_counter = counter;
130        Ok(CasaValue::Bool(changed))
131    }
132
133    /// `ismultiused(checksubtables)`: whether another process has the table
134    /// open. Subtables are not looked at: asking for them of a table that
135    /// has any is not implemented.
136    pub(super) fn ismultiused(&mut self, check_subtables: bool, log: &mut Vec<LogLine>) -> Reply {
137        let Some(open) = self.open.as_ref() else {
138            no_table_specified(log);
139            return Ok(CasaValue::Bool(false));
140        };
141        if check_subtables && has_subtables(&open.table) {
142            return Err(Fail::Unsupported(
143                "table.ismultiused(checksubtables=True) of a table with subtables is not yet \
144                 implemented in casa-rs (#1028)"
145                    .to_string(),
146            ));
147        }
148        Ok(CasaValue::Bool(open.table.is_multi_used()))
149    }

151    /// `setmaxcachesize(columnname, nbytes)`, in CASA's order of checks.
152    pub(super) fn setmaxcachesize(
153        &self,
154        column: &str,
155        nbytes: i64,
156        log: &mut Vec<LogLine>,
157    ) -> Reply {
158        let Some(open) = self.open.as_ref() else {
159            no_table_specified(log);
160            return Ok(CasaValue::Bool(false));
161        };
162        if column.is_empty() {
163            log.push(LogLine::warning("Need to specify a column name"));
164            return Ok(CasaValue::Bool(false));
165        }
166        if nbytes <= 0 {
167            log.push(LogLine::warning(
168                "Need to specify cache size greater than 0",
169            ));
170            return Ok(CasaValue::Bool(false));
171        }
172        open.known(column)?;
173        Ok(CasaValue::Bool(true))
174    }
175
176    /// `clearlocks()`: every table of the process that locks itself
177    /// automatically gives up the lock it keeps between calls, whichever tool
178    /// has it open, as casacore's `Table::relinquishAutoLocks` does; the
179    /// tables locked by the user or permanently keep theirs. Always `True`,
180    /// with or without a table. A tool busy in another thread is not asked.
181    pub(super) fn clearlocks(&mut self) -> Reply {
182        let mut result = self.relinquish_auto_lock();
183        registry::for_each_other(self.id, |tool| {
184            if let Some(tool) = tool {
185                let released = tool.relinquish_auto_lock();
186                if result.is_ok() {
187                    result = released;
188                }
189            }
190        });
191        result?;
192        Ok(CasaValue::Bool(true))
193    }
194
195    /// Gives up the lock an automatic mode keeps between calls.
196    fn relinquish_auto_lock(&mut self) -> Result<(), Fail> {
197        if let Some(open) = self.open.as_mut()
198            && open.lock.automatic()
199            && (open.table.has_lock(LockType::Read) || open.table.has_lock(LockType::Write))
200        {
201            open.table.unlock_metadata_only()?;
202        }
203        Ok(())
204    }
