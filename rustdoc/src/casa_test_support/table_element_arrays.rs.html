2//! The every-type array fixtures of the table cross matrix.
3//!
4//! Arrays of every element type in [`ELEMENT_ROWS`] rows, as
5//! `casacore_cpp_table_element_arrays.cpp` makes them, in families of
6//! columns by how they are stored ([`ElementStorage`]): fixed-shape arrays
7//! of shapes `[1]`, `[3]` and `[2, 3]`, named `<type>_<shape>`
8//! (`Float64_2x3`) when stored directly and `<type>_<shape>_ind` when stored
9//! in the storage manager's array file, and variable-shape arrays of rank 1,
10//! named `<type>_var`, whose cell in row `row` has `row + 1` elements (and
11//! `<type>_undef`, the same with row 0 undefined). Element `index`, in
12//! storage (Fortran) order, of row `row` is made from `7 + 10 row + index`,
13//! so `Float64_1` holds `[7.0]` in row 0.
14

30/// The element types of the every-type fixtures.
31pub const ELEMENT_TYPES: [PrimitiveType; 12] = [
32    PrimitiveType::Bool,
33    PrimitiveType::UInt8,
34    PrimitiveType::Int16,
35    PrimitiveType::UInt16,
36    PrimitiveType::Int32,
37    PrimitiveType::UInt32,
38    PrimitiveType::Int64,
39    PrimitiveType::Float32,
40    PrimitiveType::Float64,
41    PrimitiveType::Complex32,
42    PrimitiveType::Complex64,
43    PrimitiveType::String,
44];
45
46/// The cell shapes of the every-type fixtures, with the names their columns
47/// take.
48pub const ELEMENT_SHAPES: [(&str, &[usize]); 3] = [("1", &[1]), ("3", &[3]), ("2x3", &[2, 3])];
49
50/// Rows of the every-type fixtures.
51pub const ELEMENT_ROWS: usize = 3;
52
53/// How a family of columns of an every-type fixture stores its cells.
54#[derive(Debug, Clone, Copy, PartialEq, Eq)]
55pub enum ElementStorage {
56    /// Fixed-shape arrays in the storage manager's own data (the Direct
57    /// option).
58    Direct,
59    /// Fixed-shape arrays in the storage manager's array file (`table.fNi`).
60    Indirect,
61    /// Variable-shape arrays of rank 1, in the array file.
62    Variable,
63    /// The variable-shape arrays with row 0 never written, so undefined.
64    Undefined,
65}
66
67impl ElementStorage {
68    /// The column options.
69    fn options(self) -> ColumnOptions {
70        ColumnOptions {
71            direct: self == Self::Direct,
72            undefined: false,
73        }
74    }
75
76    /// The first row whose cell is written; the rows before it are
77    /// undefined.
78    fn first_row(self) -> usize {
79        usize::from(self == Self::Undefined)
80    }
81}
82
83/// One column of an every-type fixture.
84#[derive(Debug, Clone)]
85pub struct ElementColumn {
86    /// The column name, such as `Float64_2x3`, `Float64_2x3_ind` or
87    /// `Float64_var`.
88    pub name: String,
89    /// The element type.
90    pub data_type: PrimitiveType,
91    /// The cell shape of a fixed-shape column; `None` for a variable-shape
92    /// one.
93    pub shape: Option<&'static [usize]>,
94    /// How the column stores its cells.
95    pub storage: ElementStorage,
96}
97
98impl ElementColumn {
99    /// The column's schema.
100    pub fn schema(&self) -> ColumnSchema {
101        match self.shape {
102            Some(shape) => {
103                ColumnSchema::array_fixed(self.name.as_str(), self.data_type, shape.to_vec())
104                    .with_options(self.storage.options())
105                    .expect("fixed-shape array options")
106            }
107            None => ColumnSchema::array_variable(self.name.as_str(), self.data_type, Some(1)),
108        }
109    }
110
111    /// The column's cell in `row`, `None` when it is undefined.
112    pub fn cell(&self, row: usize) -> Option<ArrayValue> {
113        if row < self.storage.first_row() {
114            return None;
115        }
116        Some(match self.shape {
117            Some(shape) => element_array(self.data_type, shape, row),
118            None => element_array(self.data_type, &[row + 1], row),
119        })
120    }
121}

123/// The columns of an every-type fixture: every element type at every shape
124/// (once, for a variable-shape family), for each of `storages` in turn.
125pub fn element_columns(storages: &[ElementStorage]) -> Vec<ElementColumn> {
126    let mut columns = Vec::new();
127    for &storage in storages {
128        for data_type in ELEMENT_TYPES {
129            let column = |name: String, shape| ElementColumn {
130                name,
131                data_type,
132                shape,
133                storage,
134            };
135            match storage {
136                ElementStorage::Direct => columns.extend(
137                    ELEMENT_SHAPES
138                        .map(|(name, shape)| column(format!("{data_type:?}_{name}"), Some(shape))),
139                ),
140                ElementStorage::Indirect => {
141                    columns.extend(ELEMENT_SHAPES.map(|(name, shape)| {
142                        column(format!("{data_type:?}_{name}_ind"), Some(shape))
143                    }))
144                }
145                ElementStorage::Variable => {
146                    columns.push(column(format!("{data_type:?}_var"), None))
147                }
148                ElementStorage::Undefined => {
149                    columns.push(column(format!("{data_type:?}_undef"), None))
150                }
151            }
152        }
153    }
154    columns
155}

157/// The cell of an every-type column of type `data_type` and shape `shape`
158/// in `row`.
159pub fn element_array(data_type: PrimitiveType, shape: &[usize], row: usize) -> ArrayValue {
160    fn fortran<T>(shape: &[usize], values: impl Iterator<Item = T>) -> ArrayD<T> {
161        ArrayD::from_shape_vec(IxDyn(shape).f(), values.collect()).expect("shape matches")
162    }
163    let elements = 0..shape.iter().product::<usize>();
164    let base = move |index: usize| (7 + 10 * row + index) as i64;
165    match data_type {
166        PrimitiveType::Bool => ArrayValue::Bool(fortran(
167            shape,
168            elements.map(|i| (row + i).is_multiple_of(2)),
169        )),
170        PrimitiveType::UInt8 => ArrayValue::UInt8(fortran(shape, elements.map(|i| base(i) as u8))),
171        PrimitiveType::Int16 => {
172            ArrayValue::Int16(fortran(shape, elements.map(|i| -base(i) as i16)))
173        }
174        PrimitiveType::UInt16 => {
175            ArrayValue::UInt16(fortran(shape, elements.map(|i| (65_535 - base(i)) as u16)))
176        }
177        PrimitiveType::Int32 => {
178            ArrayValue::Int32(fortran(shape, elements.map(|i| (-1000 * base(i)) as i32)))
179        }
180        PrimitiveType::UInt32 => ArrayValue::UInt32(fortran(
181            shape,
182            elements.map(|i| (4_294_967_295 - base(i)) as u32),
183        )),
184        PrimitiveType::Int64 => {
185            ArrayValue::Int64(fortran(shape, elements.map(|i| -(base(i) << 40))))
186        }
187        PrimitiveType::Float32 => {
188            ArrayValue::Float32(fortran(shape, elements.map(|i| base(i) as f32 + 0.25)))
189        }
190        PrimitiveType::Float64 => {
191            ArrayValue::Float64(fortran(shape, elements.map(|i| base(i) as f64)))
192        }
193        PrimitiveType::Complex32 => ArrayValue::Complex32(fortran(
194            shape,
195            elements.map(|i| Complex32::new(base(i) as f32, -0.5)),
196        )),
197        PrimitiveType::Complex64 => ArrayValue::Complex64(fortran(
198            shape,
199            elements.map(|i| Complex64::new(base(i) as f64, 0.125)),
200        )),
201        PrimitiveType::String => {
202            ArrayValue::String(fortran(shape, elements.map(|i| format!("r{row}e{i}"))))
203        }
204    }
205}

207/// The schema of the every-type columns of `storages`.
208pub fn element_arrays_schema(storages: &[ElementStorage]) -> TableSchema {
209    TableSchema::new(
210        element_columns(storages)
211            .iter()
212            .map(ElementColumn::schema)
213            .collect(),
214    )
215    .expect("schema")
216}

218/// Save at `path` an every-type table of the columns of `storages`, stored
219/// with `manager` in `endian` byte order, from generated array columns:
220/// per-row closures, or with `runs` one run per row.
221///
222/// # Errors
223///
224/// The save's error.
225pub fn save_generated_element_arrays(
226    path: &Path,
227    storages: &[ElementStorage],
228    manager: DataManagerKind,
229    endian: EndianFormat,
230    runs: bool,
231) -> Result<(), TableError> {
232    let mut overrides = ColumnOverrides::for_row_count(ELEMENT_ROWS);
233    for column in element_columns(storages) {
234        let generated = if runs {
235            let runs = (0..ELEMENT_ROWS)
236                .map(|row| GeneratedArrayValueRun::new(row, column.cell(row)))
237                .collect();
238            GeneratedArrayColumn::from_array_runs(ELEMENT_ROWS, runs)?
239        } else {
240            let column = column.clone();
241            GeneratedArrayColumn::new(ELEMENT_ROWS, move |row| column.cell(row))
242        };
243        overrides.insert_generated_array(column.name, generated);
244    }
245    Table::with_schema(element_arrays_schema(storages))
246        .prepare_write()
247        .save_with_column_overrides(
248            TableOptions::new(path)
249                .with_data_manager(manager)
250                .with_endian_format(endian),
251            &HashMap::new(),
252            &overrides,
253        )
254}

256/// The every-type fixture of the columns of `storages`, which `cpp_fixture`
257/// writes and verifies in casacore.
258pub fn element_arrays_fixture(
259    storages: &[ElementStorage],
260    cpp_fixture: CppTableFixture,
261) -> TableFixture {
262    let columns = element_columns(storages);
263    let rows = (0..ELEMENT_ROWS)
264        .map(|row| {
265            // An undefined cell is a field the row leaves out.
266            RecordValue::new(
267                columns
268                    .iter()
269                    .filter_map(|column| {
270                        let cell = column.cell(row)?;
271                        Some(RecordField::new(column.name.as_str(), Value::Array(cell)))
272                    })
273                    .collect(),
274            )
275        })
276        .collect();
277    TableFixture {
278        schema: element_arrays_schema(storages),
279        rows,
280        table_keywords: RecordValue::default(),
281        column_keywords: vec![],
282        cpp_fixture: Some(cpp_fixture),
283        tile_shape: None,
284        column_bindings: Vec::new(),
285    }
