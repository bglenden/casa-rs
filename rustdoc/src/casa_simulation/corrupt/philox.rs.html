2//! Philox4x32-10, the counter-based generator of Salmon, Moraes, Dror and
3//! Shaw, "Parallel random numbers: as easy as 1, 2, 3" (SC11), as
4//! Random123's `philox4x32` implements it.
5//!
6//! A draw is a pure function of a 128-bit counter and a 64-bit key: ten
7//! rounds of two 32×32→64-bit multiplies, with the key bumped by the Weyl
8//! constants between rounds. There is no state, so draws can be computed in
9//! any order, on any number of workers, and are the same every time.
10//!
11//! The known-answer tests are Random123's own (`tests/kat_vectors` in
12//! <https://github.com/DEShawResearch/random123>, fetched 2026-10-21).
13
14/// Philox's 128-bit counter.
15pub(crate) type Counter = [u32; 4];
16/// Philox's 64-bit key.
17pub(crate) type Key = [u32; 2];
18
19/// `PHILOX_M4x32_0` and `PHILOX_M4x32_1`.
20const MULTIPLIERS: [u32; 2] = [0xD251_1F53, 0xCD9E_8D57];
21/// `PHILOX_W32_0` and `PHILOX_W32_1`: the golden ratio and √3 − 1 as 32-bit
22/// fractions.
23const WEYL: [u32; 2] = [0x9E37_79B9, 0xBB67_AE85];
24
25/// The high and low words of `a × b`.
26#[inline(always)]
27fn multiply_high_low(a: u32, b: u32) -> (u32, u32) {
28    let product = u64::from(a) * u64::from(b);
29    ((product >> 32) as u32, product as u32)
30}
31
32/// One round (`_philox4x32round`).
33#[inline(always)]
34fn round(counter: Counter, key: Key) -> Counter {
35    let (high0, low0) = multiply_high_low(MULTIPLIERS[0], counter[0]);
36    let (high1, low1) = multiply_high_low(MULTIPLIERS[1], counter[2]);
37    [
38        high1 ^ counter[1] ^ key[0],
39        low1,
40        high0 ^ counter[3] ^ key[1],
41        low0,
42    ]
43}
44
45/// Philox4x32 with `ROUNDS` rounds: the first with `key`, each later one
46/// with the key bumped once more by the Weyl constants.
47#[inline(always)]
48pub(crate) fn philox4x32<const ROUNDS: usize>(counter: Counter, key: Key) -> Counter {
49    let mut counter = counter;
50    let mut key = key;
51    for index in 0..ROUNDS {
52        if index > 0 {
53            key = [key[0].wrapping_add(WEYL[0]), key[1].wrapping_add(WEYL[1])];
54        }
55        counter = round(counter, key);
56    }
57    counter
58}
59
60/// Philox4x32-10, Random123's recommended round count.
61#[inline(always)]
62pub(crate) fn philox4x32_10(counter: Counter, key: Key) -> Counter {
63    philox4x32::<10>(counter, key)
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69
70    /// One `philox4x32` line of Random123's `kat_vectors`: rounds, counter,
71    /// key and the expected output, words in the file's order.
72    struct Vector {
73        rounds: usize,
74        counter: Counter,
75        key: Key,
76        expected: Counter,
77    }
78
79    /// Copied verbatim from `tests/kat_vectors`.
80    const KAT: [Vector; 6] = [
81        Vector {
82            rounds: 7,
83            counter: [0x0000_0000, 0x0000_0000, 0x0000_0000, 0x0000_0000],
84            key: [0x0000_0000, 0x0000_0000],

86        },
87        Vector {
88            rounds: 7,
89            counter: [0xffff_ffff, 0xffff_ffff, 0xffff_ffff, 0xffff_ffff],
90            key: [0xffff_ffff, 0xffff_ffff],
91            expected: [0x5207_ddc2, 0x4516_5e59, 0x4d8e_e751, 0x8c52_f662],
92        },
93        Vector {
94            rounds: 7,

119    #[test]
120    fn random123_known_answers() {
121        for vector in &KAT {
122            let output = match vector.rounds {
123                7 => philox4x32::<7>(vector.counter, vector.key),
124                10 => philox4x32_10(vector.counter, vector.key),
125                other => panic!("no {other}-round vector"),
126            };
127            assert_eq!(
128                output, vector.expected,
129                "philox4x32-{} of {:08x?} under {:08x?}",
130                vector.rounds, vector.counter, vector.key
131            );
132        }
133    }
