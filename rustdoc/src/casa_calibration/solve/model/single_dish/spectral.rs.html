2//! Frequency interpolation of a sampled SDGrid image spectrum.
3
4use casa_coordinates::SpectralCoordinate;
5use casa_core::{ArrayValue, Complex32, RecordValue, ScalarValue, Value};
6use casa_imaging_operator::{
7    Basis, NativeRow, PredictionScratch, RowContext, SpectralAxis, SpectralResampler,
8};
9use casa_imaging_runtime::{Demand, HostResources, Reservation, ResourcePolicy, admit};
10use casa_ms::MeasurementSet;
11use ndarray::ArrayD;
12
13use super::super::fourier::{FourierRow, scratch_channel_masked, spectral_layout};
14use super::super::{GainSolveError, model_error};
15
16pub(super) struct ImageSpectrum {
17    resampler: SpectralResampler,
18    scratch: PredictionScratch,
19    frequencies: Vec<f64>,
20    flags: Vec<bool>,
21    prediction: Vec<Complex32>,
22    _memory: Reservation,
23}
24
25impl ImageSpectrum {
26    pub(super) fn load(
27        record: &RecordValue,
28        spectral: &SpectralCoordinate,
29        image_channels: usize,
30        ms: &MeasurementSet,
31        host: &HostResources,
32    ) -> Result<Option<Self>, GainSolveError> {
33        match record.get("casars_live_image_frequency") {
34            None => return Ok(None),
35            Some(Value::Array(ArrayValue::Float64(frequencies)))
36                if frequencies.ndim() == 1
37                    && frequencies.len() == image_channels
38                    && image_channels > 1 => {}
39            _ => return Err(model_error("invalid saved SDGrid frequency axis")),
40        }
41        if !matches!(
42            record.get("freqinterpmethod"),
43            Some(Value::Scalar(ScalarValue::Int32(1..=3)))
44        ) {
45            return Err(model_error(
46                "invalid saved SDGrid frequency interpolation method",
47            ));
48        }
49        // Two shared axes remain live. Their construction has one temporary
50        // array of centres, reused in sequence for the two axes.
51        let axis_bytes = (image_channels as u64).saturating_mul(size_of::<f64>() as u64);
52        let axis_headers = 4 * size_of::<usize>() as u64;
53        let mut memory = admit(
54            host,
55            &ResourcePolicy::Balanced,
56            &Demand {
57                phase: "open legacy single-dish spectral axes",
58                memory: axis_bytes.saturating_mul(3).saturating_add(axis_headers),
59            },
60        )
61        .map_err(model_error)?;
62        let mut centres = Vec::with_capacity(image_channels);
63        for plane in 0..image_channels {
64            centres.push(
65                spectral
66                    .world_frequency_hz(plane as f64)
67                    .map_err(model_error)?,
68            );
69        }
70        let axis = SpectralAxis::tabular(&centres, centres[1] - centres[0]).map_err(model_error)?;
71        drop(centres);
72        let (_, resampler) = spectral_layout(record, &axis, Basis::Constant, true)?;
73        memory.retain(axis_bytes.saturating_mul(2).saturating_add(axis_headers));
74
75        let windows = ms.spectral_window().map_err(model_error)?;
76        let mut channels = 1;
77        for id in 0..windows.row_count() {
78            let count =
79                usize::try_from(windows.num_chan(id).map_err(model_error)?).map_err(model_error)?;
80            channels = channels.max(count);
81        }
82        let polarization = ms.polarization().map_err(model_error)?;
83        let mut correlations = 1;
84        for id in 0..polarization.row_count() {
85            let count = usize::try_from(polarization.num_corr(id).map_err(model_error)?)
86                .map_err(model_error)?;
87            correlations = correlations.max(count);
88        }
89        let cells = channels.saturating_mul(correlations);
90        let scratch =
91            SpectralResampler::prediction_scratch_bytes([&resampler], correlations, channels, 1);
92        let series = channels
93            .saturating_mul(size_of::<f64>())
94            .saturating_add(cells.saturating_mul(size_of::<bool>() + size_of::<Complex32>()));
95        memory.join(
96            admit(
97                host,
98                &ResourcePolicy::Balanced,
99                &Demand {
100                    phase: "sample legacy single-dish spectrum",
101                    memory: scratch.saturating_add(series as u64),
102                },
103            )
104            .map_err(model_error)?,
105        );
106        let scratch =
107            SpectralResampler::prediction_scratch([&resampler], correlations, channels, 1);
108        Ok(Some(Self {
109            resampler,
110            scratch,
111            frequencies: Vec::with_capacity(channels),
112            flags: Vec::with_capacity(cells),
113            prediction: Vec::with_capacity(cells),
114            _memory: memory,
115        }))
116    }
117
118    pub(super) fn predict(
119        &mut self,
120        row: &FourierRow<'_>,
121        (ratio, mask): (f64, Option<&ArrayD<i32>>),
122        output: &mut ArrayD<Complex32>,
123        sample: impl FnMut(u32, &mut [Complex32]),
124    ) -> Result<bool, GainSolveError> {
125        self.frequencies.clear();
126        self.frequencies
127            .extend(row.frequencies.iter().map(|frequency| frequency * ratio));
128        self.flags.clear();
129        let correlations = row.correlations.len();
130        for channel in 0..self.frequencies.len() {
131            let masked = scratch_channel_masked(mask, row.scratch_channels, channel, false);
132            self.flags.extend((0..correlations).map(|corr| {
133                masked || row.raw_flags.get([corr, channel].as_slice()) == Some(&true)
134            }));
135        }
136        self.resampler.set_legacy_prediction_selection(
137            row.scratch_channels
138                .map(|[_, start, count, step]| [start as usize, count as usize, step as usize]),
139        );
140        let native = NativeRow {
141            selection_centres_hz: None,
142            uvw_m: row.uvw,
143            phase_shift_m: 0.,
144            pointing_offset_rad: [0.; 2],
145            frequencies_hz: &self.frequencies,
146            values: &[],
147            weights: &[],
148            flags: &self.flags,
149            row_flag: row.row.flag_row,
150            context: RowContext {
151                native_frequency_ratio: ratio.recip(),
152                original_uvw_m: None,
153                time_s: row.row.time_seconds,
154                antennas: [row.row.antenna1 as u32; 2],
155                antenna_types: [0; 2],
156                parallactic_angle_rad: [0.; 2],
157                field: row.row.field_id as u32,
158                spectral_window: row.row.data_spw_id as u32,
159            },
160        };
161        self.prediction
162            .resize(self.flags.len(), Complex32::default());
163        if !self
164            .resampler
165            .predict_sampled_spectrum(
166                &native,
167                correlations,
168                &mut self.scratch,
169                &mut self.prediction,
170                sample,
171            )
172            .map_err(model_error)?
173        {
174            return Ok(false);
175        }
176        let output = output
177            .as_slice_memory_order_mut()
178            .expect("contiguous model visibilities");
179        for (output, &sample) in output.iter_mut().zip(&self.prediction) {
180            *output += sample;
181        }
182        Ok(true)
183    }
184}

186#[cfg(test)]
187mod tests {
188    use super::*;
189    use crate::imaging_memory_test::{isolated_test, live_since, peak_since, reset_peak};
190    use casa_core::measures::frequency::FrequencyRef;
191    use casa_imaging_runtime::ReservationWindow;
192    use casa_ms::MeasurementSetBuilder;
193
194    #[test]
195    fn sampled_spline_stays_within_its_charge() {
196        if !isolated_test(module_path!(), "sampled_spline_stays_within_its_charge") {
197            return;
198        }
199        const PLANES: usize = 65;
200        const CHANNELS: usize = 129;
201        const CORRELATIONS: usize = 4;
202        let mut ms = MeasurementSet::create_memory(MeasurementSetBuilder::new()).unwrap();
203        let mut spw = ms.spectral_window_mut().unwrap();
204        let table = spw.table_mut();
205        let mut window = table.schema().unwrap().default_row();
206        window.upsert(
207            "NUM_CHAN",
208            Value::Scalar(ScalarValue::Int32(CHANNELS as i32)),
209        );
210        table.add_row(window).unwrap();
211        let mut pol = ms.polarization_mut().unwrap();
212        let table = pol.table_mut();
213        let mut polarization = table.schema().unwrap().default_row();
214        polarization.upsert(
215            "NUM_CORR",
216            Value::Scalar(ScalarValue::Int32(CORRELATIONS as i32)),
217        );
218        table.add_row(polarization).unwrap();
219        let coordinate = SpectralCoordinate::new(FrequencyRef::TOPO, 1e9, 1e6, 0., 1e9);
220        let mut record = RecordValue::default();
221        record.upsert(
222            "casars_live_image_frequency",
223            Value::Array(ArrayValue::from_f64_vec(
224                (0..PLANES).map(|plane| 1e9 + plane as f64 * 1e6).collect(),
225            )),
226        );
227        record.upsert("freqinterpmethod", Value::Scalar(ScalarValue::Int32(3)));
228        let host = HostResources::detect().unwrap();
229        let reservation_window = ReservationWindow::open();
230        let baseline = reset_peak();
231        let mut model = ImageSpectrum::load(&record, &coordinate, PLANES, &ms, &host)
232            .unwrap()
233            .unwrap();
234        model
235            .frequencies
236            .extend((0..CHANNELS).map(|channel| 1e9 + channel as f64 * 0.5e6));
237        model.flags.resize(CHANNELS * CORRELATIONS, false);
238        model
239            .prediction
240            .resize(CHANNELS * CORRELATIONS, Complex32::default());
241        let native = NativeRow {
242            selection_centres_hz: None,
243            uvw_m: [0.; 3],
244            phase_shift_m: 0.,
245            pointing_offset_rad: [0.; 2],
246            frequencies_hz: &model.frequencies,
247            values: &[],
248            weights: &[],
249            flags: &model.flags,
250            row_flag: false,
251            context: RowContext {
252                native_frequency_ratio: 1.,
253                original_uvw_m: None,
254                time_s: 0.,
255                antennas: [0; 2],
256                antenna_types: [0; 2],
257                parallactic_angle_rad: [0.; 2],
258                field: 0,
259                spectral_window: 0,
260            },
261        };
262        assert!(
263            model
264                .resampler
265                .predict_sampled_spectrum(
266                    &native,
267                    CORRELATIONS,
268                    &mut model.scratch,
269                    &mut model.prediction,
270                    |plane, values| values.fill(Complex32::new(plane as f32, 1.)),
271                )
272                .unwrap()
273        );
274        let measured = peak_since(baseline);
275        let admitted = reservation_window.close();
276        assert!(
277            measured <= admitted,
278            "{measured} bytes used over {admitted}"
279        );
280        assert!(live_since(baseline) <= model._memory.memory());
281        assert_eq!(
282            model.prediction[63 * CORRELATIONS],
283            Complex32::new(31.5, 1.)
284        );
285        let flagged = NativeRow {
286            selection_centres_hz: None,
287            row_flag: true,
288            ..native
289        };
290        assert!(
291            model
292                .resampler
293                .predict_sampled_spectrum(
294                    &flagged,
295                    CORRELATIONS,
296                    &mut model.scratch,
297                    &mut model.prediction,
298                    |_, _| panic!("a flagged row must not sample the sky"),
299                )
300                .unwrap()
301        );
302        assert!(
303            model
304                .prediction
305                .iter()
306                .all(|&cell| cell == Complex32::default())
307        );
308    }
