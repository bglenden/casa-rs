2//! Calibration tables of simulated antenna-based terms: what CASA's
3//! simulator writes for `sm.setgain(table=...)`, `sm.setleakage(table=...)`
4//! and `sm.settrop(table=...)` (`SolvableVisCal::setSimulate` and
5//! `storeNCT`), written here for `casa-simulation`.
6//!
7//! The layout is CASA 6.7.6's for a simulated `G Jones`, `D Jones` or
8//! `Tf Jones` term (measured): one row per solution and antenna, ANTENNA2
9//! −1, INTERVAL 0, SCAN_NUMBER −1, OBSERVATION_ID 0, CPARAM, PARAMERR, FLAG
10//! and SNR of one channel per receptor (`[2, 1]`) for `G` and `D`, and of
11//! one parameter per channel of the window (`[1, n]`) for `Tf`, PARAMERR and
12//! SNR zero, FLAG false, and WEIGHT left undefined; the OBSERVATION,
13//! ANTENNA, FIELD and HISTORY subtables are the MeasurementSet's, and its
14//! SPECTRAL_WINDOW, with each window reduced to one channel for a term that
15//! does not depend on frequency (`G`, `D`). As CASA's
16//! `NewCalTable::writeToDisk`, a
17//! table already at the output is replaced; the new table is built beside
18//! it and the old one removed only once the new one is in place. An output
19//! that is the MeasurementSet, lies inside it, or contains it is refused
20//! before anything is touched (#891 review).
21//!
22//! The table is built in memory and saved: [`simulated_table_bytes`] is
23//! what that holds, for the caller to admit.
24

44/// Which simulated term a table holds.
45#[derive(Debug, Clone, Copy, PartialEq, Eq)]
46pub enum SimulatedJones {
47    /// `sm.setgain`: antenna gains, `G Jones`.
48    Gain,
49    /// `sm.setleakage`: polarization leakage, `D Jones`.
50    Leakage,
51    /// `sm.settrop`: tropospheric phase, channel by channel, `Tf Jones`.
52    Troposphere,
53}
54
55impl SimulatedJones {
56    /// The table's `VisCal` keyword and TableInfo subtype.
57    #[must_use]
58    pub fn vis_cal(self) -> &'static str {
59        match self {
60            Self::Gain => "G Jones",
61            Self::Leakage => "D Jones",
62            Self::Troposphere => "Tf Jones",
63        }
64    }
65
66    /// Whether the term has a value per channel, so the table keeps the
67    /// MeasurementSet's channels (CASA's `freqDepPar`).
68    #[must_use]
69    pub fn depends_on_frequency(self) -> bool {
70        matches!(self, Self::Troposphere)
71    }
72}

74/// One antenna's solution: a row of the table.
75#[derive(Debug, Clone, PartialEq)]
76pub struct SimulatedSolution {
77    /// TIME, seconds (MJD): the solution's time.
78    pub time_s: f64,
79    /// FIELD_ID of the solution.
80    pub field_id: i32,
81    /// SPECTRAL_WINDOW_ID of the solution.
82    pub spectral_window_id: i32,
83    /// ANTENNA1.
84    pub antenna: i32,
85    /// CPARAM.
86    pub values: SolutionValues,
87}
88
89/// A solution's CPARAM, `[parameter, channel]`.
90#[derive(Debug, Clone, PartialEq)]
91pub enum SolutionValues {
92    /// `G` and `D`: the two receptors' values, in one channel (`[2, 1]`).
93    Receptors([Complex32; 2]),
94    /// `Tf`: the one parameter's value in each channel of the window
95    /// (`[1, n]`).
96    Channels(Vec<Complex32>),
97}
98
99impl SolutionValues {
100    /// The values, in CPARAM's Fortran order.
101    fn as_slice(&self) -> &[Complex32] {
102        match self {
103            Self::Receptors(receptors) => receptors,
104            Self::Channels(channels) => channels,
105        }
106    }
107
108    /// CPARAM's shape.
109    fn shape(&self) -> [usize; 2] {
110        match self {
111            Self::Receptors(_) => [2, 1],
112            Self::Channels(channels) => [1, channels.len()],
113        }
114    }
115}

117/// Heap a simulated table holds for each row while it is built in memory
118/// and saved: the row's record (twelve named cells, four of them arrays)
119/// and its share of the columns the save forms. Measured at about 2,050
120/// bytes (the #891 review's 36,000-row probe, and the table law in
121/// `casa-simulation`'s `session_memory/jones.rs`); a quarter more is kept
122/// for the allocator.
123pub const SIMULATED_TABLE_ROW_BYTES: u64 = 2_560;
124
125/// Heap a simulated table holds for each value of a row's CPARAM beyond
126/// [`SIMULATED_TABLE_ROW_BYTES`]: the value in CPARAM, PARAMERR, FLAG and
127/// SNR (17 bytes). Measured at about 16.6 bytes (1.2 million values of a
128/// `Tf Jones` table of 1,024 channels, `casa-simulation`'s
129/// `session_memory/troposphere.rs` law, which 16 fails); a quarter more is
130/// kept for the allocator.
131pub const SIMULATED_TABLE_VALUE_BYTES: u64 = 21;
132
133/// Heap writing a simulated table of `rows` rows, whose CPARAMs hold
134/// `values` values in all, holds beyond its fixed metadata.
135#[must_use]
136pub const fn simulated_table_bytes(rows: u64, values: u64) -> u64 {
137    rows * SIMULATED_TABLE_ROW_BYTES + values * SIMULATED_TABLE_VALUE_BYTES
138}

140/// Refuse `output` as a simulated table of the MeasurementSet at
141/// `measurement_set` when it is that MeasurementSet, lies inside it (a
142/// subtable), or contains it: replacing it would remove the data the table
143/// is written from. Paths are compared after resolving symbolic links and
144/// `..` in the part of each that exists, so an alias is caught as well.
145///
146/// # Errors
147///
148/// [`CalibrationTableWriteError::Prepare`] naming the conflict.
149pub fn check_simulated_table_destination(
150    measurement_set: &Path,
151    output: &Path,
152) -> Result<(), CalibrationTableWriteError> {
153    let (table, ms) = (resolved(output), resolved(measurement_set));
154    let conflict = if table == ms {
155        "is the MeasurementSet it is written from"
156    } else if table.starts_with(&ms) {
157        "lies inside the MeasurementSet it is written from"
158    } else if ms.starts_with(&table) {
159        "contains the MeasurementSet it is written from"
160    } else {
161        return Ok(());
162    };
163    Err(CalibrationTableWriteError::Prepare {
164        path: output.display().to_string(),
165        reason: format!("{conflict} ({})", measurement_set.display()),
166    })
167}
168
169/// `path` made absolute, with the longest part of it that exists resolved
170/// (symbolic links and `..`) and the rest normalised lexically.
171fn resolved(path: &Path) -> PathBuf {
172    let absolute = std::path::absolute(path).unwrap_or_else(|_| path.to_path_buf());
173    let mut base = PathBuf::new();
174    let mut exists = true;
175    for component in absolute.components() {
176        match component {
177            Component::CurDir => continue,
178            Component::ParentDir if !exists => {
179                base.pop();
180                continue;
181            }
182            other => base.push(other),
183        }
184        if exists {
185            match base.canonicalize() {
186                Ok(canonical) => base = canonical,
187                Err(_) => exists = false,
188            }
189        }
190    }
191    base
192}

194/// Write `solutions` of `kind` as the calibration table `output` of the
195/// MeasurementSet `ms`, replacing a table already there once the new one is
196/// saved. The rows are formed one at a time from `solutions`.
197///
198/// # Errors
199///
200/// `output` is, lies inside or contains the MeasurementSet
201/// ([`check_simulated_table_destination`]); `output` exists and is not a
202/// table; or the table cannot be written.
203pub fn write_simulated_jones_table(
204    ms: &MeasurementSet,
205    output: &Path,
206    kind: SimulatedJones,
207    solutions: impl IntoIterator<Item = SimulatedSolution>,
208) -> Result<(), CalibrationTableWriteError> {
209    if let Some(path) = ms.path() {
210        check_simulated_table_destination(path, output)?;
211    }
212    let measurement_set_name = ms.path().and_then(Path::file_name).map_or_else(
213        || "<in-memory>".to_string(),
214        |name| name.to_string_lossy().into_owned(),
215    );
216    let mut writer = CalibrationTableWriter::create_replacing(
217        ms,
218        CalibrationTableDescriptor {
219            output,
220            schema: schema(),
221            subtype: kind.vis_cal(),
222            parameter_type: Some("Complex"),
223            measurement_set_name,
224            include_polarization_basis: true,
225            time_extra_precision_column: None,
226        },
227    )?;
228    for solution in solutions {
229        writer.append(row(&solution))?;
230    }
231    writer.copy_subtables(&[
232        (SubtableId::Observation, "OBSERVATION"),
233        (SubtableId::Antenna, "ANTENNA"),
234        (SubtableId::Field, "FIELD"),
235        (SubtableId::History, "HISTORY"),
236        (SubtableId::SpectralWindow, "SPECTRAL_WINDOW"),
237    ])?;
238    if !kind.depends_on_frequency() {
239        make_windows_single_channel(&writer.output_path().join("SPECTRAL_WINDOW"))?;
240    }
241    writer.save_main()
242}

244/// Reduce every spectral window of the SPECTRAL_WINDOW subtable at `path`
245/// to one channel, as CASA's `NewCalTable::makeSpwSingleChan` does for a
246/// term that does not depend on frequency: the middle channel's frequency
247/// (`CHAN_FREQ[n/2]`), and CHAN_WIDTH, EFFECTIVE_BW and RESOLUTION the
248/// window's TOTAL_BANDWIDTH. CASA's `applycal` refuses a table whose
249/// windows keep the MeasurementSet's channels.
250fn make_windows_single_channel(path: &Path) -> Result<(), CalibrationTableWriteError> {
251    let failed = |source: TableError| CalibrationTableWriteError::Save {
252        path: path.display().to_string(),
253        source: Box::new(source),
254    };
255    let mut table = Table::open(TableOptions::new(path)).map_err(failed)?;
256    for row in 0..table.row_count() {
257        let value = |column: &str| -> Result<Value, TableError> {
258            Ok(table
259                .cell_accessor(row, column)?
260                .value()?
261                .cloned()
262                .unwrap_or_else(|| Value::Scalar(ScalarValue::Int32(0))))
263        };
264        let frequencies = match value("CHAN_FREQ").map_err(failed)? {
265            Value::Array(ArrayValue::Float64(frequencies)) => frequencies,
266            _ => continue,
267        };
268        let channels = frequencies.len();
269        if channels <= 1 {
270            continue;
271        }
272        let bandwidth = match value("TOTAL_BANDWIDTH").map_err(failed)? {
273            Value::Scalar(ScalarValue::Float64(bandwidth)) => bandwidth,
274            _ => continue,
275        };
276        let one = |value: f64| {
277            Value::Array(ArrayValue::Float64(
278                ArrayD::from_shape_vec(IxDyn(&[1]), vec![value]).expect("one channel"),
279            ))
280        };
281        let middle = frequencies
282            .iter()
283            .nth(channels / 2)
284            .copied()
285            .unwrap_or_default();
286        for (column, cell) in [
287            ("NUM_CHAN", Value::Scalar(ScalarValue::Int32(1))),
288            ("CHAN_FREQ", one(middle)),
289            ("CHAN_WIDTH", one(bandwidth)),
290            ("EFFECTIVE_BW", one(bandwidth)),
291            ("RESOLUTION", one(bandwidth)),
292        ] {
293            table
294                .cell_accessor_mut(row, column)
295                .and_then(|mut accessor| accessor.set(cell))
296                .map_err(failed)?;
297        }
298    }
299    table.flush().map_err(failed)
300}

302fn schema() -> TableSchema {
303    TableSchema::new(vec![
304        ColumnSchema::scalar(COL_TIME, PrimitiveType::Float64),
305        ColumnSchema::scalar(COL_FIELD_ID, PrimitiveType::Int32),
306        ColumnSchema::scalar(COL_SPECTRAL_WINDOW_ID, PrimitiveType::Int32),
307        ColumnSchema::scalar(COL_ANTENNA1, PrimitiveType::Int32),
308        ColumnSchema::scalar(COL_ANTENNA2, PrimitiveType::Int32),
309        ColumnSchema::scalar(COL_INTERVAL, PrimitiveType::Float64),
310        ColumnSchema::scalar(COL_SCAN_NUMBER, PrimitiveType::Int32),
311        ColumnSchema::scalar(COL_OBSERVATION_ID, PrimitiveType::Int32),
312        ColumnSchema::array_variable(COL_CPARAM, PrimitiveType::Complex32, None),
313        ColumnSchema::array_variable(COL_PARAMERR, PrimitiveType::Float32, None),
314        ColumnSchema::array_variable(COL_FLAG, PrimitiveType::Bool, None),
315        ColumnSchema::array_variable(COL_SNR, PrimitiveType::Float32, None),
316        ColumnSchema::array_variable(COL_WEIGHT, PrimitiveType::Float32, None),
317    ])
318    .expect("valid simulated calibration schema")
319}

321/// A `[parameter, channel]` cell of `shape` holding `values`, in Fortran
322/// order.
323fn cell<T>(shape: [usize; 2], values: Vec<T>) -> ArrayD<T> {
324    ArrayD::from_shape_vec(IxDyn(&shape).f(), values).expect("a value per parameter and channel")
325}
326
327fn row(solution: &SimulatedSolution) -> RecordValue {
328    let shape = solution.values.shape();
329    let count = shape[0] * shape[1];
330    let int =
331        |name: &str, value: i32| RecordField::new(name, Value::Scalar(ScalarValue::Int32(value)));
332    let double =
333        |name: &str, value: f64| RecordField::new(name, Value::Scalar(ScalarValue::Float64(value)));
334    RecordValue::new(vec![
335        double(COL_TIME, solution.time_s),
336        int(COL_FIELD_ID, solution.field_id),
337        int(COL_SPECTRAL_WINDOW_ID, solution.spectral_window_id),
338        int(COL_ANTENNA1, solution.antenna),
339        int(COL_ANTENNA2, -1),
340        double(COL_INTERVAL, 0.0),
341        int(COL_SCAN_NUMBER, -1),
342        int(COL_OBSERVATION_ID, 0),
343        RecordField::new(
344            COL_CPARAM,
345            Value::Array(ArrayValue::Complex32(cell(
346                shape,
347                solution.values.as_slice().to_vec(),
348            ))),
349        ),
350        RecordField::new(
351            COL_PARAMERR,
352            Value::Array(ArrayValue::Float32(cell(shape, vec![0.0; count]))),
353        ),
354        RecordField::new(
355            COL_FLAG,
356            Value::Array(ArrayValue::Bool(cell(shape, vec![false; count]))),
357        ),
358        RecordField::new(
359            COL_SNR,
360            Value::Array(ArrayValue::Float32(cell(shape, vec![0.0; count]))),
361        ),
362    ])
