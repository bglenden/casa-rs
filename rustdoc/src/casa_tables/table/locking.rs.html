4impl Table {
5    /// Opens an existing table from disk with locking.
6    ///
7    /// Behaves like [`open`](Table::open) but also creates or opens the
8    /// `table.lock` file and acquires a lock according to the given
9    /// [`LockOptions`].
10    ///
11    /// - [`LockMode::PermanentLocking`]: acquires a write lock immediately;
12    ///   fails if unavailable.
13    /// - [`LockMode::PermanentLockingWait`]: acquires a write lock, waiting
14    ///   indefinitely for another process.
15    /// - [`LockMode::AutoLocking`]: acquires a read lock, waiting
16    ///   indefinitely, as casacore does, while another process holds the
17    ///   write lock; write operations temporarily acquire/release a write
18    ///   lock.
19    /// - [`LockMode::UserLocking`]: no lock is acquired until
20    ///   [`lock()`](Table::lock) is called.
21    /// - [`LockMode::NoLocking`]: equivalent to [`open()`](Table::open).
22    ///
23    /// A wait adds this process to the request list in `table.lock`, so a
24    /// casacore holder using `AutoLocking` releases its lock at its next
25    /// inspection, and is logged when it starts, every ten seconds and when
26    /// it ends. A write lock held by another handle in this process is never
27    /// waited for. The table's metadata is read only once the lock is held.
28    ///
29    /// C++ equivalent: `Table(name, TableLock(...), Table::Old)`.
30    #[cfg(unix)]
31    pub fn open_with_lock(
32        options: TableOptions,
33        lock_opts: LockOptions,
34    ) -> Result<Self, TableError> {
35        if lock_opts.mode == LockMode::NoLocking {
36            return Self::open(options);
37        }
38
39        let perm = matches!(
40            lock_opts.mode,
41            LockMode::PermanentLocking | LockMode::PermanentLockingWait
42        );
43        let mut lock_file =
44            LockFile::create_or_open(&options.path, false, lock_opts.inspection_interval, perm)
45                .map_err(|e| TableError::LockIo {
46                    path: options.path.display().to_string(),
47                    message: e.to_string(),
48                })?;
49
50        // Acquire the initial lock the mode asks for, with casacore's number
51        // of attempts (`TableLockData::makeLock`, `PlainTable`).
52        let permanent_lock = if lock_opts.read_only {
53            LockType::Read
54        } else {
55            LockType::Write
56        };
57        let initial = match lock_opts.mode {
58            LockMode::PermanentLocking => Some((permanent_lock, 1)),
59            LockMode::PermanentLockingWait => Some((permanent_lock, 0)),
60            LockMode::AutoLocking | LockMode::DefaultLocking => Some((LockType::Read, 0)),
61            // AutoNoReadLocking skips the read lock on open; only write
62            // locks are acquired.
63            LockMode::AutoNoReadLocking
64            | LockMode::UserLocking
65            | LockMode::UserNoReadLocking
66            | LockMode::NoLocking => None,
67        };
68        if let Some((lock_type, nattempts)) = initial {
69            let outcome =
70                lock_file
71                    .acquire(lock_type, nattempts)
72                    .map_err(|e| TableError::LockIo {
73                        path: options.path.display().to_string(),
74                        message: e.to_string(),
75                    })?;
76            if !outcome.is_acquired() {
77                return Err(TableError::LockFailed {
78                    path: options.path.display().to_string(),
79                    message: outcome.refusal(lock_type, nattempts),
80                });
81            }
82        }
83
84        // Metadata must be read only after the initial lock is acquired. A
85        // writer may publish a new table generation while this opener waits;
86        // opening first would retain the stale pre-publication descriptor.
87        // Ordinary open remains lazy, so retained readers do not materialize
88        // row payloads here.
89        let mut table = Self::open(options.clone())?;
90
91        // Read sync data if available.
92        let sync_data = lock_file
93            .read_sync_data()
94            .map_err(|e| TableError::LockIo {
95                path: options.path.display().to_string(),
96                message: e.to_string(),
97            })?
98            .unwrap_or_else(SyncData::new);
99
100        table.lock_state = Some(LockState {
101            path: options.path.clone(),
102            lock_file,
103            sync_data,
104            options: lock_opts,
105            data_manager: options.data_manager,
106            endian_format: options.endian_format,
107            flushed_generation: table.inner.generation(),
108            unpublished_write: std::sync::atomic::AtomicBool::new(false),
109            table_dat_before_write: std::sync::OnceLock::new(),
110        });
111
112        Ok(table)
113    }
114
115    /// Acquires a lock on the table.
116    ///
117    /// Re-reads the table data from disk if another process modified it
118    /// since the last lock was held. A table that already holds a lock is not
119    /// re-read, and a read lock requested while the write lock is held keeps
120    /// the write lock, as in casacore.
121    ///
122    /// `nattempts`: number of lock attempts. 0 means wait indefinitely for
123    /// another process, 1 means try once without waiting, and more retries
124    /// once a second. A wait adds this process to the request list in
125    /// `table.lock`, as casacore's does, and is logged. A write lock held by
126    /// another handle in this process is never waited for.
127    ///
128    /// Returns `true` if the lock was acquired, `false` if it could not
129    /// be acquired within the given attempts or another handle in this
130    /// process holds the write lock.
131    ///
132    /// C++ equivalent: `Table::lock(type, nattempts)`.
133    #[cfg(unix)]
134    pub fn lock(&mut self, lock_type: LockType, nattempts: u32) -> Result<bool, TableError> {
135        // Memory tables always succeed — no file-based locking needed.
136        // C++ equivalent: MemoryTable::lock() returns True.
137        if self.kind == TableKind::Memory {
138            return Ok(true);
139        }
140
141        // Notify external sync hook before acquiring.
142        if let Some(sync) = &self.external_sync {
143            match lock_type {
144                LockType::Read => sync.acquire_read(),
145                LockType::Write => sync.acquire_write(),
146            }
147        }
148
149        let state = self
150            .lock_state
151            .as_mut()
152            .ok_or_else(|| TableError::NotLocked {
153                operation: "lock".into(),
154            })?;
155
156        // The NoRead modes only spare a table the read lock it takes on
157        // open (`skip_read_lock`); an explicit request takes it, as
158        // casacore's `PlainTable::lock` does.
159        // A table already locked needs no synchronization: no other process
160        // can have written it (casacore's `PlainTable::lock`). Requesting a
161        // read lock while holding the write lock keeps the write lock and
162        // this handle's unflushed changes.
163        let already_locked = state.lock_file.has_lock(LockType::Read);
164        let acquired = state
165            .lock_file
166            .acquire(lock_type, nattempts)
167            .map_err(|e| TableError::LockIo {
168                path: state.path.display().to_string(),
169                message: e.to_string(),
170            })?
171            .is_acquired();
172
173        if acquired && !already_locked {
174            // Read sync data and check if we need to reload.
175            if let Some(new_sync) =
176                state
177                    .lock_file
178                    .read_sync_data()
179                    .map_err(|e| TableError::LockIo {
180                        path: state.path.display().to_string(),
181                        message: e.to_string(),
182                    })?
183                && state.sync_data.needs_reload(&new_sync)
184            {
185                // Another process modified the table — reload.
186                let storage = CompositeStorage;
187                let snapshot = storage
188                    .load_with_row_hint(&state.path, Some(new_sync.nrrow))
189                    .map_err(|e| TableError::LockIo {
190                        path: state.path.display().to_string(),
191                        message: e.to_string(),
192                    })?;
193                self.virtual_columns = snapshot.virtual_columns;
194                self.inner.replace_from_snapshot(
195                    snapshot.rows,
196                    snapshot.undefined_cells,
197                    snapshot.keywords,
198                    snapshot.column_keywords,
199                    snapshot.schema,
200                );
201                // Update our stored sync data. The reloaded table is what is
202                // on disk, so there is nothing to flush.
203                if let Some(s) = self.lock_state.as_mut() {
204                    s.sync_data = new_sync;
205                    s.flushed_generation = self.inner.generation();
206                }
207            }
208        }
209
210        Ok(acquired)
211    }

213    /// Releases the current lock.
214    ///
215    /// If a write lock was held and the table changed since it was opened,
216    /// reloaded or last flushed, the table is flushed to disk first. Every
217    /// write this handle made to the disk under the lock, by that flush or
218    /// earlier (a [`flush`](Table::flush), a save or a write plan's in-place
219    /// save), is then published in the lock file's sync data, raising its
220    /// modify counter so other processes re-read the table. A write is
221    /// published even when a [`resync`](Table::resync) has since reloaded
222    /// the table, and even when it failed after writing part of its change.
223    /// A write lock under which nothing reached the disk writes and
224    /// publishes nothing, as casacore's `PlainTable::putFile` writes and
225    /// announces only what changed.
226    ///
227    /// casacore announces each flush when it writes it; casa-rs announces the
228    /// lock period's writes once, when the lock is released. Other processes
229    /// read the announcement only when they take a lock, which they cannot do
230    /// before the write lock is released, so they see the same change.
231    ///
232    /// If the flush or the publication fails, the error is returned and the
233    /// lock is kept, with the write still to be published by a later
234    /// `unlock` or when the table is dropped.
235    ///
236    /// In permanent locking modes, this is a no-op (lock is held until close).
237    ///
238    /// C++ equivalent: `Table::unlock()`.
239    #[cfg(unix)]
240    pub fn unlock(&mut self) -> Result<(), TableError> {
241        self.unlock_with_metadata_only_flush(false)
242    }

244    /// Releases the current lock after flushing only table metadata.
245    ///
246    /// This preserves the existing on-disk data-manager layout and must only
247    /// be used when the locked mutation changed table or column metadata, not
248    /// row values.
249    #[cfg(unix)]
250    pub fn unlock_metadata_only(&mut self) -> Result<(), TableError> {
251        self.unlock_with_metadata_only_flush(true)
252    }

254    /// Releases the current lock as
255    /// [`unlock_metadata_only`](Self::unlock_metadata_only) does and, when
256    /// that fails holding the write lock, releases it all the same: the
257    /// changes not yet written are discarded, the table reloaded as it is on
258    /// disk where it can be, and the lock released again, publishing
259    /// whatever reached the disk. A table dropped while it holds the write
260    /// lock with unwritten changes is written in full with its lock's
261    /// storage manager; a writer that keeps the table's managers releases
262    /// through this instead. Returns the first release's error.
263    ///
264    /// A table held with permanent locking keeps its lock, as `unlock`
265    /// keeps it, and with it its unwritten changes.
266    #[cfg(unix)]
267    pub fn unlock_metadata_only_or_discard(&mut self) -> Result<(), TableError> {
268        let released = self.unlock_metadata_only();
269        if released.is_err() && self.has_lock(LockType::Write) {
270            let _ = self.resync_or_discard();
271            let _ = self.unlock_metadata_only();
272        }
273        released
274    }

276    /// Reloads the table from disk, as [`resync`](Self::resync) does and,
277    /// when the reload fails, discards the changes not yet written all the
278    /// same: the table is left as it was in memory, but with nothing to
279    /// flush, so that closing or dropping it while it holds the write lock
280    /// only publishes what reached the disk and never writes the table in
281    /// full. Returns the reload's error.
282    ///
283    /// A writer that persists its changes itself (`save_metadata_only`,
284    /// `save_selected_rows`) calls this after a failure, to leave nothing
285    /// for the full write of a drop to find.
286    #[cfg(unix)]
287    pub fn resync_or_discard(&mut self) -> Result<(), TableError> {
288        let reloaded = self.resync();
289        if reloaded.is_err() {
290            let generation = self.inner.generation();
291            if let Some(state) = self.lock_state.as_mut() {
292                state.flushed_generation = generation;
293            }
294        }
295        reloaded
296    }

298    #[cfg(unix)]
299    fn unlock_with_metadata_only_flush(&mut self, metadata_only: bool) -> Result<(), TableError> {
300        // Memory tables have no lock to release.
301        // C++ equivalent: MemoryTable::unlock() is a no-op.
302        if self.kind == TableKind::Memory {
303            return Ok(());
304        }
305        // Extract the info we need before borrowing self for save/schema.
306        let (flush, save_opts, mode) = {
307            let state = self
308                .lock_state
309                .as_ref()
310                .ok_or_else(|| TableError::NotLocked {
311                    operation: "unlock".into(),
312                })?;
313            let flush = state.lock_file.has_lock(LockType::Write)
314                && state.has_unflushed_changes(&self.inner);
315            let opts = TableOptions::new(&state.path)
316                .with_data_manager(state.data_manager)
317                .with_endian_format(state.endian_format);
318            (flush, opts, state.options.mode)
319        };
320
321        if matches!(
322            mode,
323            LockMode::PermanentLocking | LockMode::PermanentLockingWait
324        ) {
325            return Ok(());
326        }
327
328        // If write-locked with changes, flush them to disk. The save records
329        // its write before it starts, so a save that fails part way is still
330        // published, by a later unlock or the drop.
331        if flush {
332            if metadata_only {
333                self.save_metadata_only(save_opts)?;
334            } else {
335                self.save(save_opts)?;
336            }
337            let generation = self.inner.generation();
338            let state = self.lock_state.as_mut().expect("lock_state present");
339            state.flushed_generation = generation;
340        }
341
342        // Publish every write this lock period made to the disk, describing
343        // the table as persisted, as casacore's putFile does.
344        let state = self.lock_state.as_mut().expect("lock_state present");
345        if state.lock_file.has_lock(LockType::Write) && state.has_unpublished_write() {
346            state.sync_data = publish_persisted_write(
347                &state.lock_file,
348                &state.path,
349                state.table_dat_before_write.get().map(Vec::as_slice),
350            )
351            .map_err(|e| TableError::LockIo {
352                path: state.path.display().to_string(),
353                message: e.to_string(),
354            })?;
355            state
356                .unpublished_write
357                .store(false, std::sync::atomic::Ordering::Relaxed);
358            state.table_dat_before_write.take();
359        }
360
361        state.lock_file.release().map_err(|e| TableError::LockIo {
362            path: state.path.display().to_string(),
363            message: e.to_string(),
364        })?;
365
366        // Notify external sync hook after release.
367        if let Some(sync) = &self.external_sync {
368            sync.release();
369        }
370
371        Ok(())
372    }

374    /// Returns `true` if the given lock type is currently held.
375    ///
376    /// Returns `false` if the table was not opened with locking.
377    ///
378    /// C++ equivalent: `Table::hasLock(type)`.
379    #[cfg(unix)]
380    pub fn has_lock(&self, lock_type: LockType) -> bool {
381        // Memory tables always report holding the lock.
382        // C++ equivalent: MemoryTable::hasLock() returns True.
383        if self.kind == TableKind::Memory {
384            return true;
385        }
386        self.lock_state
387            .as_ref()
388            .map(|s| s.lock_file.has_lock(lock_type))
389            .unwrap_or(false)
390    }

392    /// Return the casacore-compatible modification counter observed by this lock.
393    ///
394    /// The value is meaningful only while this table retains a read or write
395    /// lock. It is the durable counter used by table locking to decide whether
396    /// an already-open table must be reloaded after another writer commits.
397    #[cfg(unix)]
398    pub fn locked_modify_counter(&self) -> Result<u32, TableError> {
399        let state = self
400            .lock_state
401            .as_ref()
402            .ok_or_else(|| TableError::NotLocked {
403                operation: "locked_modify_counter".into(),
404            })?;
405        if !state.lock_file.has_lock(LockType::Read) {
406            return Err(TableError::NotLocked {
407                operation: "locked_modify_counter".into(),
408            });
409        }
410        Ok(state.sync_data.modify_counter)
411    }

413    /// Tests if the table is opened by another process.
414    ///
415    /// Checks the in-use indicator in the lock file. Returns `false` if the
416    /// table was not opened with locking.
417    ///
418    /// C++ equivalent: `Table::isMultiUsed()`.
419    #[cfg(unix)]
420    pub fn is_multi_used(&self) -> bool {
421        // Memory tables are never shared with another process.
422        // C++ equivalent: MemoryTable::isMultiUsed() returns False.
423        if self.kind == TableKind::Memory {
424            return false;
425        }
426        self.lock_state
427            .as_ref()
428            .map(|s| s.lock_file.is_multi_used())
429            .unwrap_or(false)
430    }

438    /// Record that this handle is about to write the table directory at
439    /// `path`. Called by every persistence path before its first write.
440    ///
441    /// A write to the handle's own table under the write lock it holds is
442    /// published when the lock is released (see [`unlock`](Table::unlock)),
443    /// however the handle's state changes in between: a resync or reload
444    /// does not withdraw it, and a write that fails part way is still
445    /// published. A write to another directory, such as a copy, or made
446    /// without the write lock, is not recorded.
447    #[cfg(unix)]
448    pub(super) fn note_persisted_write(&self, path: &Path) {
449        let Some(state) = self.lock_state.as_ref() else {
450            return;
451        };
452        if state.lock_file.has_lock(LockType::Write)
453            && same_directory(&state.path, path)
454            && !state
455                .unpublished_write
456                .swap(true, std::sync::atomic::Ordering::Relaxed)
457            && let Ok(table_dat) =
458                std::fs::read(state.path.join(crate::storage::TABLE_CONTROL_FILE))
459        {
460            let _ = state.table_dat_before_write.set(table_dat);
461        }
462    }

467    #[cfg(unix)]
468    pub(super) fn begin_write_operation(&mut self, operation: &str) -> Result<bool, TableError> {
469        if self.kind == TableKind::Memory {
470            return Ok(false);
471        }
472        // Some write operations change only the data managers or the files
473        // on disk, which the change count of the rows does not see. An
474        // admitted operation counts even if it later fails, since it may have
475        // written part of its change; a refused one does not.
476        let admitted = self.admit_write_operation(operation);
477        if admitted.is_ok() {
478            self.inner.note_change();
479        }
480        admitted
481    }

483    #[cfg(unix)]
484    fn admit_write_operation(&mut self, operation: &str) -> Result<bool, TableError> {
485        let Some(state) = self.lock_state.as_mut() else {
486            return Ok(false);
487        };
488
489        match state.options.mode {
490            LockMode::NoLocking => Ok(false),
491            LockMode::UserLocking | LockMode::UserNoReadLocking => {
492                if state.lock_file.has_lock(LockType::Write) {
493                    Ok(false)
494                } else {
495                    Err(TableError::LockFailed {
496                        path: state.path.display().to_string(),
497                        message: format!(
498                            "{operation} requires a write lock when using UserLocking"
499                        ),
500                    })
501                }
502            }
503            LockMode::PermanentLocking | LockMode::PermanentLockingWait => {
504                if state.lock_file.has_lock(LockType::Write) {
505                    Ok(false)
506                } else {
507                    Err(TableError::LockFailed {
508                        path: state.path.display().to_string(),
509                        message: format!(
510                            "{operation} requires the permanent write lock to be held"
511                        ),
512                    })
513                }
514            }
515            LockMode::AutoLocking | LockMode::AutoNoReadLocking | LockMode::DefaultLocking => {
516                if state.lock_file.has_lock(LockType::Write) {
517                    return Ok(false);
518                }
519
520                let outcome = state.lock_file.acquire(LockType::Write, 0).map_err(|e| {
521                    TableError::LockIo {
522                        path: state.path.display().to_string(),
523                        message: e.to_string(),
524                    }
525                })?;
526                if outcome.is_acquired() {
527                    Ok(true)
528                } else {
529                    Err(TableError::LockFailed {
530                        path: state.path.display().to_string(),
531                        message: format!(
532                            "{operation} needs a temporary write lock: {}",
533                            outcome.refusal(LockType::Write, 0)
534                        ),
535                    })
536                }
537            }
538        }
539    }

546    #[cfg(unix)]
547    pub(super) fn finish_write_operation<R>(
548        &mut self,
549        auto_unlock: bool,
550        result: Result<R, TableError>,
551    ) -> Result<R, TableError> {
552        if !auto_unlock {
553            return result;
554        }
555
556        let unlock_result = self.unlock();
557        match (result, unlock_result) {
558            (Ok(value), Ok(())) => Ok(value),
559            (Ok(_), Err(unlock_err)) => Err(unlock_err),
560            (Err(op_err), Ok(())) => Err(op_err),
561            (Err(op_err), Err(_unlock_err)) => Err(op_err),
562        }
563    }
564
565    #[cfg(not(unix))]
566    pub(super) fn finish_write_operation<R>(
567        &mut self,
568        _auto_unlock: bool,
569        result: Result<R, TableError>,
570    ) -> Result<R, TableError> {
571        result
572    }

575/// Whether `a` and `b` name the same directory: the same path, or two
576/// spellings of one existing directory.
577pub(super) fn same_directory(a: &Path, b: &Path) -> bool {
578    a == b
579        || matches!(
580            (std::fs::canonicalize(a), std::fs::canonicalize(b)),
581            (Ok(a), Ok(b)) if a == b
582        )
