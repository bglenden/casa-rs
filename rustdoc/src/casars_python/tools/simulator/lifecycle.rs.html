18/// The constructor's controls, each checked: a positive `memory_budget` in
19/// bytes, a `backend` of `'auto'`, `'cpu'` or `'metal'`, and a positive
20/// `workers` count; `None` for the session's defaults. A value out of range
21/// raises `ValueError`.
22pub(super) fn execution_controls(
23    memory_budget: Option<i64>,
24    backend: &str,
25    workers: Option<i64>,
26) -> PyResult<ExecutionControls> {
27    let memory_budget = memory_budget
28        .map(|bytes| {
29            u64::try_from(bytes)
30                .ok()
31                .filter(|bytes| *bytes > 0)
32                .ok_or_else(|| {
33                    PyValueError::new_err(format!(
34                        "memory_budget must be a positive number of bytes, not {bytes}"
35                    ))
36                })
37        })
38        .transpose()?;
39    let workers = workers
40        .map(|count| {
41            usize::try_from(count)
42                .ok()
43                .and_then(NonZeroUsize::new)
44                .ok_or_else(|| {
45                    PyValueError::new_err(format!(
46                        "workers must be a positive number of threads, not {count}"
47                    ))
48                })
49        })
50        .transpose()?;
51    let backend = backend
52        .parse::<BackendRequest>()
53        .map_err(|error| PyValueError::new_err(error.to_string()))?;
54    Ok(ExecutionControls {
55        memory_budget,
56        workers,
57        backend,
58    })
59}

61/// `sm.open(ms)`: close (write) any open session, then start one that will
62/// create `ms` at `close`. CASA's reference time until `settimes` is the
63/// time of this call. The GIL is released while the previous session writes
64/// and the measures data loads.
65pub(super) fn open(tool: &PySimulator, py: Python<'_>, ms: &str) -> PyResult<bool> {
66    let resources = tool.resources;
67    let path = PathBuf::from(ms);
68    py.detach(move || -> SimulatorResult<()> {
69        let mut state = tool.lock();
70        if let Some(previous) = state.take() {
71            previous.simulator.close()?;
72        }
73        let options = SimulatorOptions::from_system(resources)?;
74        let measures = Arc::clone(&options.measures);
75        *state = Some(Session {
76            simulator: Simulator::open(path, options),
77            measures,
78        });
79        Ok(())
80    })
81    .map_err(|error| raised(&error))?;
82    Ok(true)
83}

85/// `sm.openfromms(ms)`: close (write) any open session, as CASA's tool
86/// does, then start one on the existing MeasurementSet `ms`, whose rows the
87/// following `predict` and `corrupt` rewrite at `close`. A MeasurementSet
88/// that cannot be read raises, as CASA's table open does. The GIL is
89/// released while the previous session writes and `ms` is read.
90pub(super) fn openfromms(tool: &PySimulator, py: Python<'_>, ms: &str) -> PyResult<bool> {
91    let resources = tool.resources;
92    let path = PathBuf::from(ms);
93    py.detach(move || -> SimulatorResult<()> {
94        let mut state = tool.lock();
95        if let Some(previous) = state.take() {
96            previous.simulator.close()?;
97        }
98        let options = SimulatorOptions::from_system(resources)?;
99        let measures = Arc::clone(&options.measures);
100        *state = Some(Session {
101            simulator: Simulator::open_from_ms(path, options)?,
102            measures,
103        });
104        Ok(())
105    })
106    .map_err(|error| raised(&error))?;
107    Ok(true)
108}

110/// `sm.close()` and `sm.done()`: write the MeasurementSet and end the session.
111/// `True` even when nothing is open; a write failure raises. The GIL is
112/// released while writing.
113pub(super) fn close(tool: &PySimulator, py: Python<'_>) -> PyResult<bool> {
114    py.detach(|| match tool.lock().take() {
115        Some(session) => session.simulator.close(),
116        None => Ok(()),
117    })
118    .map_err(|error| raised(&error))?;
119    Ok(true)
120}

122/// `sm.name()`: the MeasurementSet's path as given to `open`, or `''`.
123pub(super) fn name(tool: &PySimulator, py: Python<'_>) -> String {
124    tool.lock_attached(py)
125        .as_ref()
126        .map(|session| session.simulator.name().to_string_lossy().into_owned())
127        .unwrap_or_default()
128}

130/// `sm.summary()`: a few INFO lines about the session; `False` before
131/// `open`. CASA's summary also lists the configuration, windows and fields.
132pub(super) fn summary(tool: &PySimulator, py: Python<'_>) -> PyResult<bool> {
133    let lines = tool.lock_attached(py).as_ref().map(|session| {
134        let simulator = &session.simulator;
135        vec![
136            LogLine::info(format!(
137                "Simulator summary for MeasurementSet {}",
138                simulator.name().display()
139            )),
140            LogLine::info(format!(
141                "     {} scan(s) planned; the MeasurementSet is written at close or done",
142                simulator.scan_count()
143            )),
144        ]
145    });
146    match lines {
147        Some(lines) => {
148            emit(py, LOGGER, &lines)?;
149            Ok(true)
150        }
151        None => Ok(false),
152    }
153}
