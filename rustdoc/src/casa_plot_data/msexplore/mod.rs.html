2//! Generic MeasurementSet plotting specifications and export helpers.
3//!
4//! `msexplore` is the next plotting layer above the curated MeasurementSet
5//! metadata presets:
6//!
7//! - metadata summaries keep their fixed, compatibility-oriented preset catalog.
8//! - `msexplore` exposes reusable plot specifications with explicit axes,
9//!   selections, averaging, transforms, and export settings.
10//! - Common curated raw-visibility plots are lowered into the generic
11//!   `msexplore` scatter builder so new feature work lands in one place.
12//!
13//! The first delivery focuses on the most common MeasurementSet `plotms`
14//! views backed by complex visibility samples:
15//!
16//! - amplitude/phase vs time
17//! - amplitude/phase vs UV distance
18//! - amplitude/phase vs channel/frequency
19//! - real vs imaginary
20//! - weight/sigma/flag vs time
21//! - elevation/azimuth/hour angle/parallactic angle diagnostics
22//!
23//! Additional transform, multi-plot page composition, and staged flag-edit
24//! fields are already modeled in the public specification types so future
25//! waves can extend the implementation without redesigning the interface.

187#[derive(Default)]
188struct VisibilityScatterTiming {
189    row_grid: Duration,
190    row_arrays: Duration,
191    grouping: Duration,
192    bin_collect: Duration,
193    axis_compute: Duration,
194    point_push: Duration,
195}
196
197impl VisibilityScatterTiming {
198    /// Whether the logging configuration takes the timing events (`debug`,
199    /// target [`MSEXPLORE_TIMING`]).
200    fn enabled() -> bool {
201        tracing::enabled!(target: MSEXPLORE_TIMING, tracing::Level::DEBUG)
202    }
203
204    fn checkpoint(label: &str, started: &mut Instant) {
205        if Self::enabled() {
206            let elapsed = started.elapsed();
207            tracing::debug!(
208                target: MSEXPLORE_TIMING,
209                "msexplore timing: {label}={} ms",
210                elapsed.as_millis()
211            );
212        }
213        *started = Instant::now();
214    }
215
216    fn report(&self, label: &str, rows: usize, points: usize, started: Instant) {
217        if !Self::enabled() {
218            return;
219        }
220        tracing::debug!(
221            target: MSEXPLORE_TIMING,
222            "msexplore timing: {label} total={} ms rows={} points={} row_grid={} ms row_arrays={} ms grouping={} ms bin_collect={} ms axis_compute={} ms point_push={} ms",
223            started.elapsed().as_millis(),
224            rows,
225            points,
226            self.row_grid.as_millis(),
227            self.row_arrays.as_millis(),
228            self.grouping.as_millis(),
229            self.bin_collect.as_millis(),
230            self.axis_compute.as_millis(),
231            self.point_push.as_millis()
232        );
233    }
234}

314fn predecimate_visibility_rows(
315    ms: &MeasurementSet,
316    row_numbers: Vec<usize>,
317    spec: &MsPlotSpec,
318    point_budget: &PointBudget,
319    context: VisibilityPredecimationContext<'_>,
320) -> Result<Vec<usize>, String> {
321    if !point_budget.allow_predecimation || row_numbers.len() <= 1 {
322        return Ok(row_numbers);
323    }
324    let Some(max_plot_points) = point_budget.max_plot_points else {
325        return Ok(row_numbers);
326    };
327    let Some(first_row) = row_numbers.first().copied() else {
328        return Ok(row_numbers);
329    };
330
331    let ddid = selected_i32_values(ms.main_table(), "DATA_DESC_ID", &[first_row])?
332        .into_iter()
333        .next()
334        .ok_or_else(|| format!("msexplore requires DATA_DESC_ID data for row {first_row}"))?;
335    if ddid < 0 || (ddid as usize) >= context.data_description.row_count() {
336        return Ok(row_numbers);
337    }
338    let ddid = ddid as usize;
339    let spw_id = context
340        .data_description
341        .spectral_window_id(ddid)
342        .map_err(|error| error.to_string())?;
343    let chan_count = if spw_id >= 0 && (spw_id as usize) < context.spectral_window.row_count() {
344        let num_chan = context
345            .spectral_window
346            .num_chan(spw_id as usize)
347            .map_err(|error| error.to_string())?;
348        usize::try_from(num_chan).unwrap_or(0)
349    } else {
350        0
351    };
352    if chan_count == 0 {
353        return Ok(row_numbers);
354    }
355    let pol_id = context
356        .data_description
357        .polarization_id(ddid)
358        .map_err(|error| error.to_string())?;
359    let (corr_count, corr_types) =
360        if pol_id >= 0 && (pol_id as usize) < context.polarization.row_count() {
361            let pol_id = pol_id as usize;
362            let corr_types = context
363                .polarization
364                .corr_type(pol_id)
365                .map_err(|error| error.to_string())?;
366            let num_corr = context
367                .polarization
368                .num_corr(pol_id)
369                .map_err(|error| error.to_string())?;
370            let corr_count = usize::try_from(num_corr)
371                .ok()
372                .filter(|count| *count > 0)
373                .unwrap_or(corr_types.len());
374            (corr_count, corr_types)
375        } else {
376            (0, Vec::new())
377        };
378    if corr_count == 0 {
379        return Ok(row_numbers);
380    }
381    let selected_correlations =
382        select_correlation_slots(corr_count, &corr_types, context.requested_corr_codes);
383    if selected_correlations.is_empty() {
384        return Ok(row_numbers);
385    }
386    let correlation_required = spec.x_axis.uses_correlation_slots()
387        || spec
388            .y_axes
389            .iter()
390            .copied()
391            .any(MsAxis::uses_correlation_slots)
392        || matches!(spec.color_by, MsColorAxis::Correlation)
393        || matches!(spec.iteration.iteraxis, Some(MsIterationAxis::Correlation));
394    let corr_points_per_row = if correlation_required {
395        selected_correlations.len()
396    } else {
397        1
398    };
399    let channel_points_per_corr = plot_channel_bins(chan_count, spec)?.len();
400    let y_points_per_bin = spec.y_axes.len().max(1);
401    let estimated_points_per_row = corr_points_per_row
402        .saturating_mul(channel_points_per_corr)
403        .saturating_mul(y_points_per_bin);
404    if estimated_points_per_row == 0 {
405        return Ok(row_numbers);
406    }
407
408    let estimated_points = row_numbers.len().saturating_mul(estimated_points_per_row);
409    if estimated_points <= max_plot_points {
410        return Ok(row_numbers);
411    }
412    let target_rows = (max_plot_points / estimated_points_per_row).clamp(1, row_numbers.len());
413    let sampled = sampled_indices(row_numbers.len(), target_rows)
414        .into_iter()
415        .map(|index| row_numbers[index])
416        .collect();
417    Ok(sampled)
418}

420fn build_stacked_amplitude_phase_time_page(
421    ms: &MeasurementSet,
422    selection: &MsSelection,
423    spec: &MsPlotSpec,
424    point_budget: &mut PointBudget,
425) -> Result<MsPlotPayload, String> {
426    if spec.iteration.iteraxis.is_some() {
427        return Err(
428            "msexplore stacked paired presets do not yet support nested iteraxis pages".to_string(),
429        );
430    }
431    if spec.secondary_y_axis().is_some() {
432        return Err(
433            "msexplore stacked paired presets already define their subplot pairing".to_string(),
434        );
435    }
436    if spec.layout.gridrows != 1
437        || spec.layout.gridcols != 1
438        || spec.layout.rowindex != 0
439        || spec.layout.colindex != 0
440        || spec.layout.plotindex != 0
441    {
442        return Err(
443            "msexplore stacked paired presets currently manage their own fixed two-row page layout"
444                .to_string(),
445        );
446    }
447
448    let page_title = spec.style.title.clone().unwrap_or_else(|| {
449        MsPlotPreset::AmplitudePhaseVsTimeStacked
450            .display_name()
451            .to_string()
452    });
453    let amplitude = build_stacked_page_child(
454        ms,
455        selection,
456        spec,
457        MsPlotPreset::AmplitudeVsTime,
458        MsAxis::Amplitude,
459        "Amplitude vs Time",
460        point_budget,
461    )?;
462    let phase = build_stacked_page_child(
463        ms,
464        selection,
465        spec,
466        MsPlotPreset::PhaseVsTime,
467        MsAxis::Phase,
468        "Phase vs Time",
469        point_budget,
470    )?;
471
472    Ok(MsPlotPayload::ScatterPage(MsScatterPagePayload {
473        title: page_title,
474        exprange: MsPageExportRange::Current,
475        gridrows: 2,
476        gridcols: 1,
477        header_lines: Vec::new(),
478        summary: format!(
479            "{}. Plots=2 Data column={}",
480            MsPlotPreset::AmplitudePhaseVsTimeStacked.display_name(),
481            spec.data_column
482        ),
483        items: vec![
484            MsScatterPageItemPayload {
485                plotindex: 0,
486                rowindex: 0,
487                colindex: 0,
488                plot: amplitude,
489            },
490            MsScatterPageItemPayload {
491                plotindex: 1,
492                rowindex: 1,
493                colindex: 0,
494                plot: phase,
495            },
496        ],
497    }))
498}

500fn build_stacked_page_child(
501    ms: &MeasurementSet,
502    selection: &MsSelection,
503    spec: &MsPlotSpec,
504    preset: MsPlotPreset,
505    y_axis: MsAxis,
506    title: &str,
507    point_budget: &mut PointBudget,
508) -> Result<MsScatterPlotPayload, String> {
509    let mut child = MsPlotSpec::from_preset(preset);
510    child.data_column = spec.data_column;
511    child.color_by = spec.color_by;
512    child.averaging = spec.averaging.clone();
513    child.transforms = spec.transforms.clone();
514    child.style = spec.style.clone();
515    child.style.title = Some(title.to_string());
516    child.style.ylabel = None;
517    child.y_axes = vec![y_axis];
518    match build_generic_visibility_scatter(ms, selection, &child, point_budget, None)? {
519        MsPlotPayload::Scatter(payload) => Ok(payload),
520        _ => Err("internal error: stacked subplot lowered to a non-scatter payload".to_string()),
521    }
522}

572#[allow(clippy::too_many_arguments)]
573fn visibility_group(
574    color_by: MsColorAxis,
575    field_id: i32,
576    field: &casa_ms::subtables::MsField<'_>,
577    spw_id: i32,
578    spectral_window: &casa_ms::subtables::MsSpectralWindow<'_>,
579    scan_number: i32,
580    antenna1: i32,
581    antenna2: i32,
582    correlation_label: Option<&str>,
583) -> (String, String) {
584    match color_by {
585        MsColorAxis::None => ("all".to_string(), "All data".to_string()),
586        MsColorAxis::Field => {
587            let field_name = if field_id >= 0 && (field_id as usize) < field.row_count() {
588                field
589                    .name(field_id as usize)
590                    .unwrap_or_else(|_| format!("FIELD {field_id}"))
591            } else {
592                format!("FIELD {field_id}")
593            };
594            (format!("field-{field_id}"), field_name)
595        }
596        MsColorAxis::Scan => (format!("scan-{scan_number}"), format!("Scan {scan_number}")),
597        MsColorAxis::SpectralWindow => {
598            let spw_name = if spw_id >= 0 && (spw_id as usize) < spectral_window.row_count() {
599                spectral_window
600                    .name(spw_id as usize)
601                    .unwrap_or_else(|_| format!("SPW {spw_id}"))
602            } else {
603                format!("SPW {spw_id}")
604            };
605            (format!("spw-{spw_id}"), spw_name)
606        }
607        MsColorAxis::Baseline => {
608            let label = format!("a{antenna1}-a{antenna2}");
609            (format!("baseline-{label}"), label)
610        }
611        MsColorAxis::Correlation => {
612            let label = correlation_label.unwrap_or("corr").to_string();
613            (format!("corr-{label}"), label)
614        }
615    }
616}

624fn iteration_group(
625    iteraxis: Option<MsIterationAxis>,
626    field_id: i32,
627    field: &casa_ms::subtables::MsField<'_>,
628    spw_id: i32,
629    spectral_window: &casa_ms::subtables::MsSpectralWindow<'_>,
630    scan_number: i32,
631    correlation: Option<(&str, i32)>,
632) -> (String, String, i32) {
633    match iteraxis {
634        None
635        | Some(MsIterationAxis::Antenna | MsIterationAxis::Baseline | MsIterationAxis::Time) => {
636            ("all".to_string(), "All data".to_string(), 0)
637        }
638        Some(MsIterationAxis::Field) => {
639            let field_name = if field_id >= 0 && (field_id as usize) < field.row_count() {
640                field
641                    .name(field_id as usize)
642                    .unwrap_or_else(|_| format!("Field {field_id}"))
643            } else {
644                format!("Field {field_id}")
645            };
646            (format!("field-{field_id}"), field_name, field_id)
647        }
648        Some(MsIterationAxis::Scan) => (
649            format!("scan-{scan_number}"),
650            format!("Scan {scan_number}"),
651            scan_number,
652        ),
653        Some(MsIterationAxis::SpectralWindow) => {
654            let spw_name = if spw_id >= 0 && (spw_id as usize) < spectral_window.row_count() {
655                spectral_window
656                    .name(spw_id as usize)
657                    .unwrap_or_else(|_| format!("SPW {spw_id}"))
658            } else {
659                format!("SPW {spw_id}")
660            };
661            (format!("spw-{spw_id}"), spw_name, spw_id)
662        }
663        Some(MsIterationAxis::Correlation) => {
664            let (label, code) = correlation.unwrap_or(("corr", 0));
665            (format!("corr-{label}"), label.to_string(), code)
666        }
667    }
668}

670fn scatter_series_identity(
671    y_axis: MsAxis,
672    multi_y: bool,
673    base_key: &str,
674    base_label: &str,
675) -> (String, String, String) {
676    if !multi_y {
677        return (
678            base_key.to_string(),
679            base_label.to_string(),
680            base_key.to_string(),
681        );
682    }
683    let series_key = format!("{}::{base_key}", y_axis.as_str());
684    let series_label = if base_key == "all" {
685        y_axis.display_name().to_string()
686    } else {
687        format!("{} · {base_label}", y_axis.display_name())
688    };
689    (series_key.clone(), series_label, series_key)
690}

692fn resolve_iterated_grid(
693    panel_count: usize,
694    requested_rows: usize,
695    requested_cols: usize,
696) -> Result<(usize, usize), String> {
697    if panel_count == 0 {
698        return Err("msexplore iteration produced no populated panels".to_string());
699    }
700    let rows = requested_rows.max(1);
701    let cols = requested_cols.max(1);
702    if rows == 1 && cols == 1 {
703        let auto_cols = (panel_count as f64).sqrt().ceil().max(1.0) as usize;
704        let auto_rows = panel_count.div_ceil(auto_cols);
705        return Ok((auto_rows, auto_cols));
706    }
707    if rows == 1 {
708        return Ok((panel_count.div_ceil(cols), cols));
709    }
710    if cols == 1 {
711        return Ok((rows, panel_count.div_ceil(rows)));
712    }
713    if rows * cols < panel_count {
714        return Err(format!(
715            "msexplore gridrows={rows} and gridcols={cols} cannot hold {panel_count} iterated panels"
716        ));
717    }
718    Ok((rows, cols))
719}
