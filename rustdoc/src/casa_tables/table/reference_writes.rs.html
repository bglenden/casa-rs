17/// Ordered RefTable descriptors from the view to its root. No cells are read
18/// and the returned path is the one whose process lock covers the view.
19fn reference_chain(
20    mut path: PathBuf,
21    lock_probe: bool,
22) -> Result<(PathBuf, Vec<RefTableDatContents>), TableError> {
23    let mut seen = std::collections::HashSet::new();
24    let mut chain = Vec::new();
25    loop {
26        let control = path.join(TABLE_CONTROL_FILE);
27        match crate::storage::table_control::is_reference_descriptor(&control) {
28            // The descriptor of a plain base may be absent or incomplete
29            // while its writer holds the lock. Wait at that directory and
30            // defer validation to the locked open.
31            Ok(false) if lock_probe || chain.is_empty() => return Ok((path, chain)),
32            Err(_) if lock_probe => return Ok((path, chain)),
33            Err(error) => return Err(error.into()),
34            Ok(_) => {}
35        }
36        let real = std::fs::canonicalize(&path).map_err(crate::storage::StorageError::from)?;
37        if !seen.insert(real) {
38            return Err(TableError::Storage("reference table cycle".into()));
39        }
40        match read_table_dat_dispatch(&control)? {
41            TableDatResult::Ref(reference) => {
42                path = crate::table_path::add_directory_for_link(
43                    &reference.parent_relative_path,
44                    &path,
45                );
46                chain.push(reference);
47            }
48            TableDatResult::Plain(_) => return Ok((path, chain)),
49            TableDatResult::Concat(_) if chain.is_empty() => return Ok((path, chain)),
50            TableDatResult::Concat(_) => {
51                return Err(TableError::Storage(
52                    "reference base is a concatenated table".into(),
53                ));
54            }
55        }
56    }
57}

59pub(super) fn reference_lock_path(path: &Path) -> Result<PathBuf, TableError> {
60    reference_chain(path.to_path_buf(), true).map(|(root, _)| root)
61}

63fn base_name<'a>(chain: &'a [RefTableDatContents], mut name: &'a str) -> &'a str {
64    for view in chain {
65        name = view.parent_column(name);
66    }
67    name
68}
69
70fn base_row(chain: &[RefTableDatContents], mut row: usize) -> Result<usize, TableError> {
71    for view in chain {
72        row = view
73            .parent_row(row)
74            .ok_or_else(|| TableError::Storage(format!("reference row {row} is out of bounds")))?;
75    }
76    Ok(row)
77}

99struct ReferenceColumnLayer {
100    path: PathBuf,
101    #[cfg(any(target_os = "linux", target_os = "macos"))]
102    directory: File,
103    _descriptor: File,
104    before: Metadata,
105    contents: RefTableDatContents,
106    changed: bool,

109impl ReferenceColumnLayer {
110    fn open(path: PathBuf) -> Result<Self, TableError> {
111        #[cfg(any(target_os = "linux", target_os = "macos"))]
112        let directory = crate::publication::open_directory_identity(&path)
113            .map_err(crate::storage::StorageError::from)?;
114        #[cfg(any(target_os = "linux", target_os = "macos"))]
115        let descriptor = crate::publication::open_regular_directory_entry(
116            &directory,
117            std::ffi::OsStr::new(TABLE_CONTROL_FILE),
118        )
119        .map_err(crate::storage::StorageError::from)?;
120        #[cfg(not(any(target_os = "linux", target_os = "macos")))]
121        let descriptor = File::open(path.join(TABLE_CONTROL_FILE))
122            .map_err(crate::storage::StorageError::from)?;
123        let before = descriptor
124            .metadata()
125            .map_err(crate::storage::StorageError::from)?;
126        let contents = read_ref_table_file(
127            descriptor
128                .try_clone()
129                .map_err(crate::storage::StorageError::from)?,
130        )?;
131        Ok(Self {
132            path,
133            #[cfg(any(target_os = "linux", target_os = "macos"))]
134            directory,
135            _descriptor: descriptor,
136            before,
137            contents,
138            changed: false,
139        })
140    }
141
142    #[cfg(any(target_os = "linux", target_os = "macos"))]
143    fn publish(&self) -> Result<(), TableError> {
144        use std::os::unix::fs::MetadataExt;
145        let io = crate::storage::StorageError::from;
146        let captured = self.directory.metadata().map_err(io)?;
147        let current = std::fs::metadata(&self.path).map_err(io)?;
148        if (captured.dev(), captured.ino()) != (current.dev(), current.ino()) {
149            return Err(TableError::Storage(
150                "saved reference directory was replaced".into(),
151            ));
152        }
153        let mut staged = crate::publication::StagedDirectoryFile::new(&self.directory, &self.path)
154            .map_err(io)?;
155        let file = staged.file_mut();
156        file.set_permissions(self.before.permissions())
157            .map_err(io)?;
158        write_ref_table_stream(
159            AipsIo::new_read_write(file.try_clone().map_err(io)?),
160            &self.contents,
161        )?;
162        staged
163            .replace(std::ffi::OsStr::new(TABLE_CONTROL_FILE), &self.before)
164            .map_err(io)?;
165        Ok(())
166    }
167
168    #[cfg(not(any(target_os = "linux", target_os = "macos")))]
169    fn publish(&self) -> Result<(), TableError> {
170        let io = crate::storage::StorageError::from;
171        let path = self.path.join(TABLE_CONTROL_FILE);
172        let staged = tempfile::NamedTempFile::new_in(&self.path).map_err(io)?;
173        staged
174            .as_file()
175            .set_permissions(self.before.permissions())
176            .map_err(io)?;
177        write_ref_table_stream(
178            AipsIo::new_read_write(staged.as_file().try_clone().map_err(io)?),
179            &self.contents,
180        )?;
181        let current = std::fs::metadata(&path).map_err(io)?;
182        if self.before.len() != current.len()
183            || self.before.modified().ok() != current.modified().ok()
184        {
185            return Err(TableError::Storage(
186                "saved reference descriptor changed".into(),
187            ));
188        }
189        staged.persist(path).map_err(|error| io(error.error))?;
190        Ok(())
191    }
192}

194impl ReferenceColumnPublication {
195    /// Publish the extended column projections after their physical columns
196    /// have been installed, under the same verified root write lock.
197    pub fn publish(self, root: &Table, lock: &TableWriteLock) -> Result<Vec<PathBuf>, TableError> {
198        if !lock.matches_table_path(&self.root)
199            || !root
200                .path()
201                .is_some_and(|path| lock.matches_table_path(path))
202        {
203            return Err(TableError::NotLocked {
204                operation: "publish saved reference columns".into(),
205            });
206        }
207        for column in &self.columns {
208            root.require_column(column)?;
209        }
210        let mut published = Vec::new();
211        for layer in self.layers.into_iter().rev().filter(|layer| layer.changed) {
212            layer.publish()?;
213            published.push(layer.path);
214        }
215        Ok(published)
216    }
217}

219impl Table {
220    /// Plan adding same-name bindings to a saved reference where a logical
221    /// column is missing. An existing ancestor binding still supplies its
222    /// original physical name. The physical MAIN lock covers every view.
223    #[doc(hidden)]
224    pub fn prepare_reference_columns_holding(
225        &self,
226        columns: &[&str],
227        lock: &TableWriteLock,
228    ) -> Result<Option<ReferenceColumnPublication>, TableError> {
229        if !self.is_reference_table()? {
230            return Ok(None);
231        }
232        let mut path = self.source_path.clone().expect("saved reference");
233        let mut layers = Vec::new();
234        let mut seen = HashSet::new();
235        while crate::storage::table_control::is_reference_descriptor(
236            &path.join(TABLE_CONTROL_FILE),
237        )? {
238            path = path
239                .canonicalize()
240                .map_err(crate::storage::StorageError::from)?;
241            if !seen.insert(path.clone()) {
242                return Err(TableError::Storage("reference table cycle".into()));
243            }
244            let layer = ReferenceColumnLayer::open(path)?;
245            path = crate::table_path::add_directory_for_link(
246                &layer.contents.parent_relative_path,
247                &layer.path,
248            );
249            layers.push(layer);
250        }
251        if !lock.matches_table_path(&path) {
252            return Err(TableError::NotLocked {
253                operation: "prepare saved reference columns".into(),
254            });
255        }
256        let mut physical = Vec::with_capacity(columns.len());
257        for &column in columns {
258            let mut name = column.to_owned();
259            for layer in &mut layers {
260                if layer.contents.column_names.contains(&name) {
261                    name = layer.contents.parent_column(&name).to_owned();
262                } else {
263                    layer.contents.column_names.push(name.clone());
264                    layer
265                        .contents
266                        .column_name_map
267                        .push((name.clone(), name.clone()));
268                    layer.changed = true;
269                }
270            }
271            physical.push(name);
272        }
273        Ok(Some(ReferenceColumnPublication {
274            root: path,
275            layers,
276            columns: physical,
277        }))
278    }

280    /// Resolve selected logical rows and columns through their saved references.
281    /// Repeated rows remain repeated. The caller must verify the descriptors
282    /// after acquiring the returned root's write lock.
283    ///
284    /// # Errors
285    /// A requested cell is outside the view or a descriptor is unreadable.
286    #[doc(hidden)]
287    pub fn reference_write_projection(
288        &self,
289        rows: &[usize],
290        columns: &[&str],
291    ) -> Result<Option<ReferenceWriteProjection>, TableError> {
292        let Some(source) = &self.source_path else {
293            return Ok(None);
294        };
295        if !self.is_reference_table()? {
296            return Ok(None);
297        }
298        let (root, chain) = reference_chain(source.clone(), false)?;
299        let rows = rows
300            .iter()
301            .map(|&row| base_row(&chain, row))
302            .collect::<Result<_, _>>()?;
303        let columns = columns
304            .iter()
305            .map(|&column| {
306                self.require_column(column)?;
307                Ok(base_name(&chain, column).to_owned())
308            })
309            .collect::<Result<_, TableError>>()?;
310        Ok(Some(ReferenceWriteProjection {
311            path: root,
312            rows,
313            columns,
314        }))
315    }

317    pub(crate) fn reference_column_manager_type(
318        &self,
319        column: &str,
320    ) -> Result<Option<String>, TableError> {
321        if !self.is_reference_table()? {
322            return Ok(None);
323        }
324        let source = self.source_path.as_ref().expect("saved reference");
325        let (root, chain) = reference_chain(source.clone(), false)?;
326        let root_table = Table::open_metadata_only(TableOptions::new(root))?;
327        let name = base_name(&chain, column);
328        Ok(root_table
329            .data_manager_info()
330            .iter()
331            .find(|manager| manager.columns.iter().any(|column| column == name))
332            .map(|manager| manager.dm_type.clone()))
333    }

335    /// Apply cell changes through the saved row/column projection under
336    /// the view's already-held write lock. Return false for plain tables.
337    pub(super) fn try_save_reference_cells(
338        &mut self,
339        columns: &[&str],
340        changed_rows: &[usize],
341    ) -> Result<bool, TableError> {
342        if !self.is_reference_table()? {
343            return Ok(false);
344        }
345        for column in columns {
346            self.require_writable_column(column)?;
347        }
348        let source = self.source_path.as_ref().expect("saved reference");
349        let (root, chain) = reference_chain(source.clone(), false)?;
350        self.check_reference_base(&root)?;
351        let mut base = Table::open(TableOptions::new(&root))?;
352        let mut selected_columns: Vec<_> =
353            columns.iter().map(|col| base_name(&chain, col)).collect();
354        selected_columns.sort_unstable();
355        selected_columns.dedup();
356        let changed_rows = if changed_rows.is_empty() {
357            (0..self.row_count()).collect::<Vec<_>>()
358        } else {
359            changed_rows.to_vec()
360        };
361        let mut selected_rows = Vec::new();
362        for &row in &changed_rows {
363            let index = base_row(&chain, row)?;
364            for &column in columns {
365                let name = base_name(&chain, column);
366                let descriptor = self
367                    .schema()
368                    .and_then(|schema| schema.column(column))
369                    .ok_or_else(|| TableError::SchemaColumnUnknown {
370                        column: column.to_string(),
371                    })?;
372                // Mutated scalar/array cells are in the overlay. Read
373                // those directly instead of loading the parent rows.
374                match descriptor.column_type() {
375                    crate::ColumnType::Scalar => match self.get_scalar_cell(row, column) {
376                        Ok(value) => base.write_scalar_cell_bound(index, name, value.clone())?,
377                        Err(TableError::ColumnNotFound { .. }) => {}
378                        Err(error) => return Err(error),
379                    },
380                    crate::ColumnType::Array(_) => match self.get_array_cell(row, column) {
381                        Ok(value) => base.write_array_cell_bound(index, name, value.clone())?,
382                        Err(TableError::ColumnNotFound { .. }) => {}
383                        Err(error) => return Err(error),
384                    },
385                    crate::ColumnType::Record => {
386                        base.set_cell(index, name, Value::Record(self.record_cell(row, column)?))?;
387                    }
388                }
389            }
390            selected_rows.push(index);
391        }
392        selected_rows.sort_unstable();
393        selected_rows.dedup();
394        self.note_persisted_write(&root);
395        let result = base.persist_selected_rows_in_place(&selected_columns, &selected_rows);
396        if matches!(&result, Err(TableError::Storage(message)) if message.starts_with("partial in-place save does not support"))
397            && let Some(options) = base.single_rewrite_options()?
398        {
399            base.save(options)?;
400        } else {
401            result?;
402        }
403        let mut changed_rows = changed_rows;
404        changed_rows.sort_unstable();
405        self.inner.note_selected_cells_saved(columns, &changed_rows);
406        Ok(true)
407    }

409    fn check_reference_base(&self, base: &Path) -> Result<(), TableError> {
410        #[cfg(unix)]
411        if let Some(lock) = &self.lock_state
412            && !locking::same_directory(&lock.path, base)
413        {
414            return Err(TableError::Storage(
415                "the saved reference now points to a different base; reopen it".into(),
416            ));
417        }
418        #[cfg(not(unix))]
419        let _ = base;
420        Ok(())
421    }

423    /// RefTable keywords (table and column) belong to the base table.
424    /// TableInfo is stored beside the view's descriptor.
425    pub(super) fn try_save_reference_metadata(
426        &self,
427        destination: &Path,
428    ) -> Result<bool, TableError> {
429        if !self.is_reference_table()?
430            || !self
431                .source_path
432                .as_deref()
433                .is_some_and(|path| locking::same_directory(path, destination))
434        {
435            return Ok(false);
436        }
437        let (root, chain) = reference_chain(destination.to_path_buf(), false)?;
438        self.check_reference_base(&root)?;
439        let mut table = Table::open(TableOptions::new(&root))?;
440        let view = self.source_path.as_deref().expect("saved reference");
441        let mut keywords = self.keywords().clone();
442        rebase_table_keywords(&mut keywords, view, &root);
443        *table.keywords_mut() = keywords;
444        for column in &chain[0].column_names {
445            let mut keywords = self.column_keywords(column).cloned().unwrap_or_default();
446            rebase_table_keywords(&mut keywords, view, &root);
447            table.set_column_keywords(base_name(&chain, column), keywords);
448        }
449        self.note_persisted_write(&root);
450        table.save_metadata_only(TableOptions::new(&root))?;
451        std::fs::write(destination.join(TABLE_INFO_FILE), self.info().to_string())
452            .map_err(crate::storage::StorageError::from)?;
453        self.inner.note_metadata_saved();
454        Ok(true)
455    }
