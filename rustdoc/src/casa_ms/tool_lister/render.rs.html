10const DAY_SECONDS: f64 = 86_400.0;
11const PROMPT: &str =
12    "Type Q to quit, A to toggle long/short list, or RETURN to continue [continue]: ";
13
14pub(super) struct ListingState<'a> {
15    output: &'a mut dyn Write,
16    input: Option<&'a mut dyn BufRead>,
17    show_page_prompt: bool,
18    page_rows: usize,
19    date: String,
20    fixed: bool,
21}
22
23impl<'a> ListingState<'a> {
24    pub(super) fn new(
25        output: &'a mut dyn Write,
26        input: Option<&'a mut dyn BufRead>,
27        page_rows: i32,
28    ) -> Self {
29        Self {
30            output,
31            input,
32            show_page_prompt: true,
33            page_rows: page_rows.unsigned_abs() as usize,
34            date: String::new(),
35            fixed: false,
36        }
37    }
38
39    fn line(&mut self, text: &str) -> MsResult<()> {
40        writeln!(self.output, "{text}").map_err(io_error)
41    }
42
43    fn prompt(&mut self) -> MsResult<bool> {
44        let Some(input) = self.input.as_deref_mut() else {
45            return Ok(true);
46        };
47        write!(self.output, "{PROMPT}").map_err(io_error)?;
48        self.output.flush().map_err(io_error)?;
49        let mut command = String::new();
50        input.read_line(&mut command).map_err(io_error)?;
51        if matches!(command.as_bytes().first(), Some(b'a' | b'A')) {
52            self.show_page_prompt = !self.show_page_prompt;
53        }
54        Ok(!matches!(command.as_bytes().first(), Some(b'q' | b'Q')))
55    }
56
57    fn display_value(&self, value: f64) -> String {
58        if self.fixed {
59            format!("{value:.2}")
60        } else {
61            format!("{value}")
62        }
63    }
64}

66pub(super) fn report_chunk(
67    state: &mut ListingState<'_>,
68    chunk: &ListingChunk<'_>,
69    spectra: &ListingSpectra,
70    antenna_names: &[String],
71) -> MsResult<bool> {
72    let description = chunk.descriptions[0];
73    let window = spectra
74        .windows
75        .get(&description)
76        .ok_or_else(|| listing_error(format!("unknown data description {description}")))?;
77    let geometry = Geometry::build(chunk, window, antenna_names)?;
78    let widths = Widths::new(chunk, &geometry, spectra.multiple_channels);
79    state.line("Units of columns are: Date/Time(YYMMDD/HH:MM:SS UT), UVDist(wavelength), Phase(deg), UVW(m)")?;
80    if let Some(value) = constant(chunk.amplitude.iter().copied().map(f64::from)) {
81        state.line(&format!(
82            "All selected data has AMPLITUDE = {}",
83            state.display_value(value)
84        ))?;
85    }
86    if let Some(value) = chunk
87        .phase
88        .and_then(|phase| constant(phase.iter().map(|value| phase_degrees(*value).abs())))
89    {
90        state.line(&format!(
91            "All selected data has PHASE = {}",
92            state.display_value(value)
93        ))?;
94    }
95    if let Some(value) = constant(chunk.weight.iter().map(|value| f64::from(value.abs()))) {
96        state.line(&format!("WEIGHT: {}", state.display_value(value)))?;
97    }
98    if widths.field == 0 {
99        state.line(&format!("FIELD: {}", chunk.fields[0]))?;
100    }
101    if widths.spw == 0 {
102        state.line(&format!("SPW: {description}"))?;
103    }
104    if widths.channel == 0
105        && let Some(channel) = window.channels.first()
106    {
107        state.line(&format!("CHANNEL: {channel}"))?;
108    }
109
110    let rule = widths.separator(window.correlations.len());
111    if state.page_rows == 0 {
112        header(state, window, &widths, &rule)?;
113    }
114    let mut count = 0usize;
115    for (row, &seconds) in chunk.times.iter().enumerate() {
116        let time = seconds / DAY_SECONDS;
117        state.date = format_time(time, YMD | NO_TIME, 0);
118        let timestamp = format_time(time, TIME, 7);
119        let window = spectra
120            .windows
121            .get(&chunk.descriptions[row])
122            .ok_or_else(|| listing_error("unknown data description in listing"))?;
123        for &channel in &window.channels {
124            if state.page_rows != 0 && count.is_multiple_of(state.page_rows) {
125                if count != 0 && state.show_page_prompt && !state.prompt()? {
126                    return Ok(false);
127                }
128                header(state, window, &widths, &rule)?;
129            }
130            print_row(
131                state, chunk, &geometry, window, &widths, &timestamp, row, channel,
132            )?;
133            state.fixed = true;
134            count += 1;
135        }
136    }
137    state.line(&rule)?;
138    state.prompt()
139}

141fn phase_degrees(value: f32) -> f64 {
142    f64::from(f64::from(value).to_degrees() as f32)
143}

145struct Geometry<'a> {
146    ant1: Vec<&'a str>,
147    ant2: Vec<&'a str>,
148    distance: Vec<f64>,
149}
150
151impl<'a> Geometry<'a> {
152    fn build(
153        chunk: &ListingChunk<'_>,
154        window: &ListingWindow,
155        names: &'a [String],
156    ) -> MsResult<Self> {
157        let name = |index: &i32| -> MsResult<&'a str> {
158            names
159                .get(valid_index(*index)?)
160                .map(String::as_str)
161                .ok_or_else(|| listing_error(format!("unknown antenna {index}")))
162        };
163        let ant1 = chunk.antenna1.iter().map(name).collect::<MsResult<_>>()?;
164        let ant2 = chunk.antenna2.iter().map(name).collect::<MsResult<_>>()?;
165        let distance = (0..chunk.times.len())
166            .map(|row| {
167                let u = chunk.uvw[[0, row]];
168                let v = chunk.uvw[[1, row]];
169                (u * u + v * v).sqrt() * window.uvdist_scale
170            })
171            .collect();
172        Ok(Self {
173            ant1,
174            ant2,
175            distance,
176        })
177    }
178}

180struct Widths {
181    ant1: usize,
182    ant2: usize,
183    distance: usize,
184    field: usize,
185    spw: usize,
186    channel: usize,
187    amplitude: usize,
188    phase: usize,
189    weight: usize,
190    uvw: usize,
191}
192
193impl Widths {
194    fn new(chunk: &ListingChunk<'_>, geometry: &Geometry<'_>, multi_channel: bool) -> Self {
195        Self {
196            ant1: geometry
197                .ant1
198                .iter()
199                .map(|name| name.len())
200                .max()
201                .unwrap_or(0)
202                + 1,
203            ant2: geometry
204                .ant2
205                .iter()
206                .map(|name| name.len())
207                .max()
208                .unwrap_or(0),
209            distance: digits(geometry.distance.iter().copied()).max(6) + 1,
210            field: index_width(chunk.fields.iter().copied()),
211            spw: index_width(chunk.descriptions.iter().copied()),
212            channel: usize::from(multi_channel) * 5,
213            amplitude: {
214                let sign = usize::from(chunk.amplitude.iter().any(|value| *value < 0.0));
215                let magnitude = chunk.amplitude.iter().map(|value| f64::from(value.abs()));
216                digits(magnitude) + 5 + sign
217            },
218            phase: chunk.phase.map_or(0, |phase| {
219                let sign_and_border = if phase.iter().any(|value| *value < 0.0) {
220                    3
221                } else {
222                    1
223                };
224                (digits(phase.iter().copied().map(phase_degrees)) + sign_and_border + 3).max(4)
225            }),
226            weight: digits(chunk.weight.iter().copied().map(f64::from)).max(3) + 1,
227            uvw: (digits(chunk.uvw.iter().copied()) + 4).max(9) + 1,
228        }
229    }
230
231    fn baseline(&self) -> usize {
232        self.ant1 + 1 + self.ant2
233    }
234
235    fn visibility(&self) -> usize {
236        self.amplitude + self.phase + self.weight + 2
237    }
238
239    fn separator(&self, correlations: usize) -> String {
240        let mut fields = vec![
241            12,
242            self.baseline(),
243            self.distance,
244            self.field,
245            self.spw,
246            self.channel,
247        ];
248        fields.retain(|width| *width > 0);
249        let mut rule = String::new();
250        for width in fields {
251            rule.push_str(&"-".repeat(width - usize::from(!rule.is_empty())));
252            rule.push('|');
253        }
254        rule.push('-');
255        for width in std::iter::repeat_n(self.visibility(), correlations)
256            .chain(std::iter::repeat_n(self.uvw, 3))
257        {
258            rule.push_str(&"-".repeat(width - 1));
259            rule.push('|');
260        }
261        rule
262    }
263}
264
265fn digits(values: impl Iterator<Item = f64>) -> usize {
266    let maximum = values.fold(f64::NEG_INFINITY, f64::max).abs();
267    if maximum == 0.0 || !maximum.is_finite() {
268        1
269    } else {
270        ((maximum.log10().abs() + 0.5).round_ties_even() as usize).max(1)
271    }
272}
273
274fn index_width(data: impl Iterator<Item = i32> + Clone) -> usize {
275    if constant(data.clone().map(f64::from)).is_some() {
276        0
277    } else {
278        digits(data.map(f64::from)).max(3) + 1
279    }
280}
281
282fn constant(mut values: impl Iterator<Item = f64>) -> Option<f64> {
283    let first = values.next()?;
284    values.all(|value| value == first).then_some(first)
285}

287fn header(
288    state: &mut ListingState<'_>,
289    window: &ListingWindow,
290    w: &Widths,
291    rule: &str,
292) -> MsResult<()> {
293    let left = format!(
294        "{:<12}{:>width$}",
295        "Date/Time:",
296        "",
297        width = w.baseline() + w.distance + w.field + w.spw + w.channel
298    );
299    let mut first = format!("{left} ");
300    let date = format!("{}/", state.date);
301    let mut second = format!(
302        "{date:<12}{:>baseline$}{:>distance$}",
303        "Intrf",
304        "UVDist",
305        baseline = w.baseline(),
306        distance = w.distance
307    );
308    for (name, width) in [("Fld", w.field), ("SpW", w.spw), ("Chn", w.channel)] {
309        if width > 0 {
310            second.push_str(&format!("{name:>width$}"));
311        }
312    }
313    second.push(' ');
314    for (_, name) in &window.correlations {
315        let title = format!("  {name}:");
316        first.push_str(&format!("{title:<width$}", width = w.visibility()));
317        second.push_str(&format!("{:>width$}", "Amp", width = w.amplitude));
318        if w.phase > 0 {
319            second.push_str(&format!("{:>width$}", "Phs", width = w.phase));
320        }
321        second.push_str(&format!("{:>width$} F", "Wt", width = w.weight));
322    }
323    first.push_str(&" ".repeat(3 * w.uvw));
324    for name in ["U", "V", "W"] {
325        second.push_str(&format!("{name:>width$}", width = w.uvw));
326    }
327    state.line(&first)?;
328    state.line(&second)?;
329    state.line(rule)
330}

332#[expect(
333    clippy::too_many_arguments,
334    reason = "one physical visibility row and its report formatting context"
335)]
336fn print_row(
337    state: &mut ListingState<'_>,
338    chunk: &ListingChunk<'_>,
339    geometry: &Geometry<'_>,
340    window: &ListingWindow,
341    w: &Widths,
342    timestamp: &str,
343    row: usize,
344    channel: usize,
345) -> MsResult<()> {
346    let mut line = format!(
347        "{timestamp:>12}{:>ant1$}-{:>ant2$}{:>distance$.0}",
348        geometry.ant1[row],
349        geometry.ant2[row],
350        geometry.distance[row],
351        ant1 = w.ant1,
352        ant2 = w.ant2,
353        distance = w.distance
354    );
355    for (value, width) in [
356        (i64::from(chunk.fields[row]), w.field),
357        (i64::from(window.id), w.spw),
358        (channel as i64, w.channel),
359    ] {
360        if width > 0 {
361            line.push_str(&format!("{value:>width$}"));
362        }
363    }
364    line.push(':');
365    for &(corr, _) in &window.correlations {
366        let index = [corr, channel, row];
367        let amplitude = chunk
368            .amplitude
369            .get(index)
370            .ok_or_else(|| listing_error("visibility channel exceeds stored shape"))?;
371        line.push_str(&format!("{amplitude:>width$.3}", width = w.amplitude));
372        if let Some(phase) = chunk.phase {
373            let degrees = phase
374                .get(index)
375                .copied()
376                .map(phase_degrees)
377                .ok_or_else(|| listing_error("phase channel exceeds stored shape"))?;
378            line.push_str(&format!("{degrees:>width$.1}", width = w.phase));
379        }
380        let weight = chunk
381            .weight
382            .get([corr, row])
383            .ok_or_else(|| listing_error("weight correlation exceeds stored shape"))?;
384        let flagged = chunk
385            .flags
386            .get(index)
387            .ok_or_else(|| listing_error("flag channel exceeds stored shape"))?;
388        line.push_str(&format!(
389            "{weight:>width$.0} {}",
390            if *flagged { 'F' } else { ' ' },
391            width = w.weight
392        ));
393    }
394    for i in 0..3 {
395        line.push_str(&format!("{:>width$.2}", chunk.uvw[[i, row]], width = w.uvw));
396    }
397    state.line(&line)
398}
