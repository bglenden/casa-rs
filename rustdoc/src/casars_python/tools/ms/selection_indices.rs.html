23#[derive(Default)]
24pub(super) struct SelectionState {
25    indices: MsSelectionIndices,
26    error: Option<String>,
27}
28
29impl SelectionState {
30    pub(super) fn use_implicit_selection(&mut self) {
31        self.indices.scan.clear();
32        self.indices.observation.clear();
33        self.indices.state.clear();
34        self.error = None;
35    }
36
37    fn record(&self) -> PyResult<CasaValue> {
38        match &self.error {
39            None => Ok(index_record(&self.indices, true)),
40            Some(error) => Err(runtime_error(error)),
41        }
42    }
43
44    pub(super) fn parse(
45        &mut self,
46        ms: &MeasurementSet,
47        request: &MsSelection,
48    ) -> MsResult<ParsedMsToolSelection> {
49        let parsed = ms.parse_ms_tool_selection(request, &mut self.indices);
50        self.error = parsed.as_ref().err().map(ToString::to_string);
51        parsed
52    }
53
54    pub(super) fn clear_polarization_indices(&mut self) {
55        self.indices.poldd.clear();
56        self.error = None;
57    }
58}

60#[pymethods]
61impl PyMsTool {
62    #[pyo3(signature = (items = None, onlyparse = false), text_signature = "($self, items={}, onlyparse=False)")]
63    fn msselect(
64        &self,
65        py: Python<'_>,
66        #[pyo3(from_py_with = selection_dictionary)] items: Option<Bound<'_, PyDict>>,
67        #[pyo3(from_py_with = strict_bool)] onlyparse: bool,
68    ) -> PyResult<bool> {
69        let request = match items.as_ref().map(selection_request).transpose() {
70            Ok(request) => request.flatten(),
71            Err(_) => return Ok(false),
72        };
73        py.detach(|| {
74            let mut attached = self.attached.lock().map_err(runtime_error)?;
75            let Some(attached) = attached.as_mut() else {
76                return Ok(false);
77            };
78            let request = request.unwrap_or_default();
79            let snapshot = attached.selection_snapshot()?;
80            let ms = &snapshot.ms;
81            let Ok(parsed) = attached.selection_state.parse(ms, &request) else {
82                return Ok(false);
83            };
84            if onlyparse {
85                Ok(parsed.has_row_expression())
86            } else {
87                attached.apply_parsed_selection(ms, &parsed)
88            }
89        })
90    }
91
92    #[pyo3(signature = (msselect = None), text_signature = "($self, msselect='')")]
93    fn selecttaql(
94        &self,
95        py: Python<'_>,
96        #[pyo3(from_py_with = direct_taql)] msselect: Option<String>,
97    ) -> PyResult<bool> {
98        py.detach(|| {
99            let mut attached = self.attached.lock().map_err(runtime_error)?;
100            match attached.as_mut() {
101                Some(ms) => ms.apply_direct_taql(msselect.unwrap_or_default()),
102                None => Ok(false),
103            }
104        })
105    }
106
107    #[pyo3(signature = (datadescid = 0, reset = false), text_signature = "($self, datadescid=0, reset=False)")]
108    fn selectinit(
109        &self,
110        py: Python<'_>,
111        #[pyo3(from_py_with = selectinit_id)] datadescid: i64,
112        #[pyo3(from_py_with = strict_bool)] reset: bool,
113    ) -> PyResult<bool> {
114        py.detach(|| {
115            let mut attached = self.attached.lock().map_err(runtime_error)?;
116            let Some(attached) = attached.as_mut() else {
117                return Ok(false);
118            };
119            if reset {
120                attached.reset_visibility_selection()?;
121                return Ok(true);
122            }
123            if (datadescid as i32) < 0 {
124                return Err(runtime_error(
125                    "The data description id must be a list of positive integers",
126                ));
127            }
128            attached.select_visibility_description(datadescid)
129        })
130    }

132    #[pyo3(signature = (), text_signature = "($self)")]
133    fn msselectedindices(&self, py: Python<'_>) -> PyResult<Py<PyAny>> {
134        self.attachment_info(py, |attached| {
135            let attached = attached.ok_or_else(|| runtime_error("Table column TIME is unknown"))?;
136            attached.selection_state.record()
137        })?
138        .into_python(py)
139    }
140
141    #[allow(clippy::too_many_arguments)]
142    #[pyo3(signature = (vis = None, spw = None, field = None, baseline = None, time = None, scan = None, uvrange = None, observation = None, polarization = None, taql = None), text_signature = "($self, vis='', spw=[], field=[], baseline=[], time=[], scan=[], uvrange=[], observation=[], polarization=[], taql='')")]
143    fn msseltoindex(
144        &self,
145        py: Python<'_>,
146        #[pyo3(from_py_with = selection_visibility)] vis: Option<PathBuf>,
147        spw: Option<&Bound<'_, PyAny>>,
148        field: Option<&Bound<'_, PyAny>>,
149        baseline: Option<&Bound<'_, PyAny>>,
150        time: Option<&Bound<'_, PyAny>>,
151        scan: Option<&Bound<'_, PyAny>>,
152        uvrange: Option<&Bound<'_, PyAny>>,
153        observation: Option<&Bound<'_, PyAny>>,
154        polarization: Option<&Bound<'_, PyAny>>,
155        #[pyo3(from_py_with = selection_taql)] taql: Option<String>,
156    ) -> PyResult<Py<PyAny>> {
157        let request = MsSelection {
158            spw: selector(spw)?,
159            field: selector(field)?,
160            antenna: selector(baseline)?,
161            timerange: selector(time)?,
162            scan: selector(scan)?,
163            uvrange: selector(uvrange)?,
164            observation: selector(observation)?,
165            correlation: selector(polarization)?,
166            msselect: taql.filter(|text| !text.is_empty()),
167            ..MsSelection::default()
168        };
169        let indices = py.detach(|| {
170            let path = lexical_absolute(
171                vis.as_deref()
172                    .filter(|path| !path.as_os_str().is_empty())
173                    .unwrap_or_else(|| Path::new(".")),
174            );
175            let view = MetadataSnapshot::open(&path).map_err(runtime_error)?;
176            let ms = &view.ms;
177            if ms.ms_version() != Some(2.0) {
178                return Err(runtime_error(
179                    "These data are not in MSv2 format - use ms1toms2 to convert",
180                ));
181            }
182
183            let mut indices = MsSelectionIndices::default();
184            ms.parse_ms_tool_selection(&request, &mut indices)
185                .map_err(runtime_error)?;
186            Ok(indices)
187        })?;
188        index_record(&indices, false).into_python(py)
189    }
190}

192impl AttachedMs {
193    pub(super) fn selection_snapshot(&self) -> PyResult<Arc<MetadataSnapshot>> {
194        let link = self.metadata_link();
195        if self.plain_rows.is_some()
196            && let Some(rows) = self.selected_rows.as_deref()
197            && rows.iter().copied().eq(0..rows.len())
198        {
199            // `selectinit` commonly selects every row. Use the current MAIN
200            // directly when that saved selection is still its identity view.
201            link.verify_directory().map_err(runtime_error)?;
202            let snapshot = MetadataSnapshot::open(&self.path).map_err(runtime_error)?;
203            if snapshot.ms.row_count() == rows.len() {
204                link.verify_directory().map_err(runtime_error)?;
205                return Ok(Arc::new(snapshot));
206            }
207        }
208        link.snapshot(None).map_err(runtime_error)
209    }
210
211    pub(super) fn apply_direct_taql(&mut self, text: String) -> PyResult<bool> {
212        let snapshot = self.selection_snapshot()?;
213        let ms = &snapshot.ms;
214        let parsed = if !text.is_empty() && text.trim().is_empty() {
215            Err(format!(
216                "Parse error at or near ''\nScanned so far: {text}\n"
217            ))
218        } else {
219            let request = MsSelection {
220                msselect: Some(text),
221                ..MsSelection::default()
222            };
223            self.selection_state
224                .parse(ms, &request)
225                .map_err(|error| error.to_string())
226        };
227        let parsed = parsed.map_err(|error| {
228            let message = format!("Error in select expression: {error}");
229            self.selection_state.error = Some(message.clone());
230            runtime_error(message)
231        })?;
232        if !parsed.has_row_expression() {
233            self.synchronize_metadata_selection()?;
234            return Ok(false);
235        }
236        if parsed.preserves_current_view() {
237            self.synchronize_metadata_selection()?;
238            return Ok(ms.row_count() != 0);
239        }
240        let rows = parsed
241            .select_rows(ms)
242            .map_err(|error| runtime_error(format!("Error in select expression: {error}")))?;
243        self.apply_tool_rows(rows)
244    }
245
246    pub(super) fn apply_parsed_selection(
247        &mut self,
248        ms: &MeasurementSet,
249        parsed: &ParsedMsToolSelection,
250    ) -> PyResult<bool> {
251        if !parsed.has_row_expression() {
252            self.synchronize_metadata_selection()?;
253            return Ok(false);
254        }
255        if parsed.preserves_current_view() {
256            self.synchronize_metadata_selection()?;
257            return Ok(ms.row_count() != 0);
258        }
259        let Ok(rows) = parsed.select_rows(ms) else {
260            return Ok(false);
261        };
262        self.apply_tool_rows(rows)
263    }
264
265    pub(super) fn apply_tool_rows(&mut self, mut rows: Vec<usize>) -> PyResult<bool> {
266        self.map_tool_rows(&mut rows);
267        let nonempty = !rows.is_empty();
268        self.set_selected_rows(Some(rows))?;
269        Ok(nonempty)
270    }
271
272    pub(super) fn map_tool_rows(&self, rows: &mut [usize]) {
273        if let Some(previous) = &self.selected_rows {
274            for row in rows {
275                *row = previous[*row];
276            }
277        }
278    }
279}

281pub(super) fn selectinit_id(value: &Bound<'_, PyAny>) -> PyResult<i64> {
282    if missing_selection_integer(value)? {
283        return Ok(-1);
284    }
285    value
286        .extract::<i64>()
287        .map_err(|_| PyTypeError::new_err("argument _datadescid must be an integer"))
288}
289
290fn selector(value: Option<&Bound<'_, PyAny>>) -> PyResult<Option<String>> {
291    let text = listing_selector(value)?;
292    Ok((!text.is_empty()).then_some(text))
293}

295pub(super) fn selection_dictionary<'py>(
296    value: &Bound<'py, PyAny>,
297) -> PyResult<Option<Bound<'py, PyDict>>> {
298    value
299        .cast::<PyDict>()
300        .map(|record| Some(record.clone()))
301        .map_err(|_| PyTypeError::new_err("_items is not a dictionary"))
302}
303
304fn selection_visibility(value: &Bound<'_, PyAny>) -> PyResult<Option<PathBuf>> {
305    #[cfg(unix)]
306    if let Ok(bytes) = value.cast::<PyBytes>() {
307        use std::os::unix::ffi::OsStringExt;
308        let path = std::ffi::OsString::from_vec(bytes.as_bytes().to_vec());
309        return Ok(Some(path.into()));
310    }
311    selection_text(value, "vis").map(|text| Some(text.into()))
312}
313
314fn selection_taql(value: &Bound<'_, PyAny>) -> PyResult<Option<String>> {
315    selection_text(value, "taql").map(Some)
316}
317
318fn direct_taql(value: &Bound<'_, PyAny>) -> PyResult<Option<String>> {
319    selection_text(value, "msselect").map(Some)
320}

322pub(super) fn selection_text(value: &Bound<'_, PyAny>, parameter: &str) -> PyResult<String> {
323    if value.is_instance_of::<PyString>() {
324        value.extract()
325    } else if value.is_instance_of::<PyBytes>() {
326        value.call_method1("decode", ("utf-8",))?.extract()
327    } else {
328        Err(PyTypeError::new_err(format!(
329            "argument _{parameter} must be a string"
330        )))
331    }
332}

334fn index_record(indices: &MsSelectionIndices, attached: bool) -> CasaValue {
335    let mut fields = vec![
336        ("field", vector(&indices.field)),
337        ("scan", vector(&indices.scan)),
338        ("antenna1", vector(&indices.antenna1)),
339        ("antenna2", vector(&indices.antenna2)),
340        ("baselines", matrix(indices.baselines.as_deref())),
341        ("spw", vector(&indices.spw)),
342        (
343            "channel",
344            matrix((!indices.channel.is_empty()).then_some(indices.channel.as_slice())),
345        ),
346        ("spwdd", vector(&indices.spwdd)),
347        ("poldd", vector(&indices.poldd)),
348        ("dd", vector(&indices.descriptions())),
349    ];
350    if attached {
351        fields.extend([
352            ("observationid", vector(&indices.observation)),
353            ("stateid", vector(&indices.state)),
354            ("feed1", vector(&[])),
355            ("feed2", vector(&[])),
356            ("feedpairs", vector(&[])),
357        ]);
358    } else {
359        fields.push(("obsids", vector(&indices.observation)));
360    }
361    CasaValue::record(fields)
362}
363
364fn vector(values: &[i32]) -> CasaValue {
365    CasaValue::Ints {
366        shape: vec![values.len()],
367        values: values.iter().copied().map(i64::from).collect(),
368    }
369}
370
371fn matrix<const COLUMNS: usize>(rows: Option<&[[i32; COLUMNS]]>) -> CasaValue {
372    let column_count = rows.map_or(0, |_| COLUMNS);
373    let rows = rows.unwrap_or_default();
374    CasaValue::Ints {
375        shape: vec![rows.len(), column_count],
376        values: (0..column_count)
377            .flat_map(|col| rows.iter().map(move |row| i64::from(row[col])))
378            .collect(),
379    }
380}

382fn selection_request(items: &Bound<'_, PyDict>) -> PyResult<Option<MsSelection>> {
383    let mut request = MsSelection::default();
384    let mut named = false;
385    let fields = items
386        .iter()
387        .map(|(key, value)| selection_text(&key, "items").map(|key| (key, value)))
388        .collect::<PyResult<BTreeMap<_, _>>>()?;
389    // CASA's dictionary is a sorted casac::record, so alias precedence
390    // follows field names rather than Python's insertion order.
391    for (key, value) in fields {
392        let slot = match key.as_str() {
393            "spw" => &mut request.spw,
394            "time" => &mut request.timerange,
395            "field" => &mut request.field,
396            "baseline" | "antenna" => &mut request.antenna,
397            "scan" => &mut request.scan,
398            "scanintent" | "state" => &mut request.intent,
399            "polarization" => &mut request.correlation,
400            "uvdist" => &mut request.uvrange,
401            "observation" => &mut request.observation,
402            "array" => &mut request.array,
403            "taql" => &mut request.msselect,
404            _ => continue,
405        };
406        let value = selection_text(&value, &key)?;
407        *slot = (!value.is_empty()).then_some(value);
408        named |= slot.is_some();
409    }
410    Ok(named.then_some(request))
