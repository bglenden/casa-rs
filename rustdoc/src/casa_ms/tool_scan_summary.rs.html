11/// One subscan in the public `ms.getscansummary()` report.
12#[derive(Debug, PartialEq)]
13pub struct ToolScanSummary {
14    /// The first integration center, as MJD days.
15    pub begin_day: f64,
16    /// The last integration center, as MJD days.
17    pub end_day: f64,
18    /// Field ID of the first visibility row at the start of the run.
19    pub field_id: i32,
20    /// State ID of the first visibility row at the start of the run.
21    pub state_id: i32,
22    /// Count of visibility rows in the run, including repeated view rows.
23    pub row_count: usize,
24    /// Mean MAIN.EXPOSURE, in seconds.
25    pub mean_exposure: f64,
26    /// DATA_DESC_IDs in order of first occurrence within an integration.
27    pub data_description_ids: Vec<i32>,
28    /// One SPECTRAL_WINDOW_ID per data description, in the same order.
29    pub spectral_window_ids: Vec<i32>,
30}

32#[derive(Clone, Copy, PartialEq)]
33struct IntegrationTime(f64);
34
35// Only finite TIME values are admitted. Use ordinary equality so +0 and -0
36// describe the same integration, as they do in the C++ table iterator.
37impl Eq for IntegrationTime {}
38
39impl PartialOrd for IntegrationTime {
40    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
41        Some(self.cmp(other))
42    }
43}
44
45impl Ord for IntegrationTime {
46    fn cmp(&self, other: &Self) -> Ordering {
47        self.0
48            .partial_cmp(&other.0)
49            .expect("finite integration times")
50    }
51}

53#[derive(Default, PartialEq, Eq)]
54struct IntegrationIds {
55    fields: Vec<i32>,
56    data_descriptions: Vec<i32>,
57    states: Vec<i32>,
58}
59
60#[derive(Default)]
61struct Integration {
62    ids: IntegrationIds,
63    row_count: usize,
64    exposure_sum: f64,
65}
66
67impl Integration {
68    fn add_row(&mut self, field: i32, description: i32, state: i32, exposure: f64) {
69        push_unique(&mut self.ids.fields, field);
70        push_unique(&mut self.ids.data_descriptions, description);
71        push_unique(&mut self.ids.states, state);
72        self.row_count += 1;
73        self.exposure_sum += exposure;
74    }
75}
76
77fn push_unique(ids: &mut Vec<i32>, id: i32) {
78    if !ids.contains(&id) {
79        ids.push(id);
80    }
81}

83type ScanKey = (i32, i32, i32);
84type Integrations = BTreeMap<IntegrationTime, Integration>;

86/// Report the consecutive integration runs within each observation, array
87/// and scan of the supplied MeasurementSet view. A change to an integration's
88/// ordered field, data description or state IDs starts a new run.
89/// A supplied selection is an ordered list of row indices into the view;
90/// otherwise all of MAIN is used.
91///
92/// Keys are scan numbers. The vector is indexed by the subscan number in
93/// CASA's record. When a scan number recurs in another observation or array,
94/// the later group's subscan records replace any with the same index.
95///
96/// # Errors
97/// Returns an error for unreadable MAIN cells, nonfinite integration times
98/// or an invalid DATA_DESCRIPTION reference.
99pub fn tool_scan_summary(
100    ms: &MeasurementSet,
101    selected_rows: Option<&[usize]>,
102) -> MsResult<BTreeMap<i32, Vec<ToolScanSummary>>> {
103    let row_count = selected_rows.map_or_else(|| ms.row_count(), <[usize]>::len);
104    if row_count == 0 {
105        return Ok(BTreeMap::new());
106    }
107    let scans = read_integrations(ms, selected_rows, row_count)?;
108    let description = ms.data_description()?;
109    let spectral_windows = (0..description.row_count())
110        .map(|row| description.spectral_window_id(row))
111        .collect::<MsResult<Vec<_>>>()?;
112    let mut by_scan = BTreeMap::<_, Vec<_>>::new();
113    for ((_, _, number), integrations) in scans {
114        let summaries = by_scan.entry(number).or_default();
115        let mut integrations = integrations.into_iter().peekable();
116        let mut index = 0;
117        while let Some((begin, first)) = integrations.next() {
118            let mut end = begin;
119            let mut row_count = first.row_count;
120            let mut exposure_sum = first.exposure_sum;
121            while let Some((time, same)) = integrations.next_if(|(_, next)| next.ids == first.ids) {
122                end = time;
123                row_count += same.row_count;
124                exposure_sum += same.exposure_sum;
125            }
126            let data_description_ids = first.ids.data_descriptions;
127            let spectral_window_ids = data_description_ids
128                .iter()
129                .map(|&id| {
130                    spectral_windows.get(id as usize).copied().ok_or_else(|| {
131                        MsError::InvalidInput(format!("invalid MAIN.DATA_DESC_ID {id}"))
132                    })
133                })
134                .collect::<MsResult<Vec<_>>>()?;
135            let summary = ToolScanSummary {
136                begin_day: begin.0 / 86_400.0,
137                end_day: end.0 / 86_400.0,
138                field_id: first.ids.fields[0],
139                state_id: first.ids.states[0],
140                row_count,
141                mean_exposure: exposure_sum / row_count as f64,
142                data_description_ids,
143                spectral_window_ids,
144            };
145            if let Some(previous) = summaries.get_mut(index) {
146                *previous = summary;
147            } else {
148                summaries.push(summary);
149            }
150            index += 1;
151        }
152    }
153    Ok(by_scan)
154}

156fn read_integrations(
157    ms: &MeasurementSet,
158    selected_rows: Option<&[usize]>,
159    row_count: usize,
160) -> MsResult<BTreeMap<ScanKey, Integrations>> {
161    const COLUMNS: [&str; 8] = [
162        "OBSERVATION_ID",
163        "ARRAY_ID",
164        "SCAN_NUMBER",
165        "FIELD_ID",
166        "DATA_DESC_ID",
167        "STATE_ID",
168        "TIME",
169        "EXPOSURE",
170    ];
171    let bytes_per_row = 6 * std::mem::size_of::<i32>()
172        + 2 * std::mem::size_of::<f64>()
173        + std::mem::size_of::<usize>();
174    let plan = MsSelectionIoBudget::from_system_memory(1, bytes_per_row, None)
175        .and_then(|budget| MsReadPlan::new(row_count, budget))
176        .map_err(|error| MsError::InvalidInput(error.to_string()))?;
177    let table = ms.main_table();
178    let mut scans = BTreeMap::<ScanKey, Integrations>::new();
179    let mut row_buffer = Vec::new();
180    for start in (0..row_count).step_by(plan.rows_per_block) {
181        let end = start.saturating_add(plan.rows_per_block).min(row_count);
182        let rows = if let Some(selected) = selected_rows {
183            &selected[start..end]
184        } else {
185            row_buffer.clear();
186            row_buffer.extend(start..end);
187            &row_buffer
188        };
189        let mut columns = table.required_scalar_columns_owned_for_rows(&COLUMNS, rows)?;
190        let [
191            observation,
192            array,
193            scan,
194            field,
195            description,
196            state,
197            time,
198            exposure,
199        ] = COLUMNS;
200        let observation = take_integer(&mut columns, observation)?;
201        let array = take_integer(&mut columns, array)?;
202        let scan = take_integer(&mut columns, scan)?;
203        let field = take_integer(&mut columns, field)?;
204        let description = take_integer(&mut columns, description)?;
205        let state = take_integer(&mut columns, state)?;
206        let times = take_double(&mut columns, time)?;
207        let exposures = take_double(&mut columns, exposure)?;
208        for (row, time) in times.into_iter().enumerate() {
209            if !time.is_finite() {
210                return Err(MsError::InvalidInput("MAIN.TIME must be finite".into()));
211            }
212            scans
213                .entry((observation[row], array[row], scan[row]))
214                .or_default()
215                .entry(IntegrationTime(time))
216                .or_default()
217                .add_row(field[row], description[row], state[row], exposures[row]);
218        }
219    }
220    Ok(scans)
221}

223pub(crate) fn take_integer(
224    columns: &mut HashMap<String, RequiredScalarColumnValues>,
225    name: &str,
226) -> MsResult<Vec<i32>> {
227    match columns.remove(name) {
228        Some(RequiredScalarColumnValues::Int32(values)) => Ok(values),
229        _ => Err(MsError::InvalidInput(format!(
230            "MAIN.{name} must contain Int32 scalars"
231        ))),
232    }
233}
234
235pub(crate) fn take_double(
236    columns: &mut HashMap<String, RequiredScalarColumnValues>,
237    name: &str,
238) -> MsResult<Vec<f64>> {
239    match columns.remove(name) {
240        Some(RequiredScalarColumnValues::Float64(values)) => Ok(values),
241        _ => Err(MsError::InvalidInput(format!(
242            "MAIN.{name} must contain Float64 scalars"
243        ))),
244    }
245}
