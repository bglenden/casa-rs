2//! The geodesy of CASA's `simutil`: its datum and ellipsoid tables and its
3//! conversions between UTM, geodetic and Earth-centred coordinates.
4//!
5//! | CASA (`simutil`) | casa-rs |
6//! |---|---|
7//! | `getdatum(datumcode)` | [`datum`], [`DATUMS`], [`ELLIPSOIDS`] |
8//! | `utm2long(east, north, zone, datum, nors)` | [`utm_to_geodetic`] |
9//! | `long2xyz(lon, lat, elevation, datum)` | [`geodetic_to_xyz`] |
10//! | `utm2xyz(easting, northing, elevation, zone, datum, nors)` | [`utm_to_xyz`] |
11//! | `xyz2long(x, y, z, datum)` | [`xyz_to_geodetic`] |
12//! | `locxyz2itrf(lat, longitude, alt, locx, locy, locz)` | [`local_to_itrf`] |
13//!
14//! The arithmetic follows CASA's Python operation by operation (including
15//! its `180/π` divisions and its `**` powers), so results agree with CASA's
16//! to rounding. As in CASA, the "ITRF" these functions produce is the
17//! nominal right-handed Earth-centred frame of the datum's ellipsoid shifted
18//! by the datum's offsets, not a realised ITRF.
19//!
20//! The UTM conversion is T. Vincenty's transverse Mercator inversion
21//! (subroutines `tconpc` and `tmgeod` of the NGS `UTMS` program, 1985), which
22//! R. Indebetouw translated from Fortran for CASA in 2009.
23
24/// A reference ellipsoid of CASA's `getdatum` table.
25#[derive(Debug, Clone, Copy, PartialEq)]
26pub struct Ellipsoid {
27    /// CASA's two-letter code, e.g. `"WE"` for WGS 84.
28    pub code: &'static str,
29    /// Equatorial radius (semi-major axis), metres.
30    pub equatorial_radius_m: f64,
31    /// Reciprocal of the flattening.
32    pub inverse_flattening: f64,
33    /// CASA's descriptive name.
34    pub name: &'static str,
35}

37/// A geodetic datum of CASA's `getdatum` table: an ellipsoid and the
38/// offsets that take the datum's Earth-centred coordinates to WGS 84's.
39///
40/// CASA's docstring: one takes local Earth-centred x, y, z coordinates and
41/// adds `dx`, `dy`, `dz` to get WGS 84 Earth-centred ones.
42#[derive(Debug, Clone, Copy, PartialEq)]
43pub struct Datum {
44    /// CASA's datum code, e.g. `"SAM56"` (the `datum=` header value).
45    pub code: &'static str,
46    /// Offset added to x, metres.
47    pub dx_m: f64,
48    /// Offset added to y, metres.
49    pub dy_m: f64,
50    /// Offset added to z, metres.
51    pub dz_m: f64,
52    /// The datum's ellipsoid.
53    pub ellipsoid: Ellipsoid,
54    /// CASA's descriptive name.
55    pub name: &'static str,
56}

202/// The zone constants of the transverse Mercator inversion that `tconpc`
203/// returns: `eps` (the second eccentricity squared), `r` (the radius of
204/// the rectifying sphere), `so` (the scaled meridional distance from the
205/// equator to the southernmost parallel `orlim`) and the foot-point series
206/// coefficients `v0`, `v2`, `v4`, `v6`.
207struct ZoneConstants {
208    eps: f64,
209    r: f64,
210    so: f64,
211    v: [f64; 4],
212}
213
214/// `tconpc`: the zone constants for scale factor `sf`, southernmost
215/// parallel `orlim`, equatorial radius `er`, eccentricity squared `esq` and
216/// reciprocal flattening `rf`.
217fn tconpc(sf: f64, orlim: f64, er: f64, esq: f64, rf: f64) -> ZoneConstants {
218    let f = 1.0 / rf;
219    let eps = esq / (1.0 - esq);
220    let pr = (1.0 - f) * er;
221    let en = (er - pr) / (er + pr);
222    let en2 = en * en;
223    let en3 = en * en * en;
224    let en4 = en2 * en2;
225
226    let c2 = -3.0 * en / 2.0 + 9.0 * en3 / 16.0;
227    let c4 = 15.0 * en2 / 16.0 - 15.0 * en4 / 32.0;
228    let c6 = -35.0 * en3 / 48.0;
229    let c8 = 315.0 * en4 / 512.0;
230    let u0 = 2.0 * (c2 - 2.0 * c4 + 3.0 * c6 - 4.0 * c8);
231    let u2 = 8.0 * (c4 - 4.0 * c6 + 10.0 * c8);
232    let u4 = 32.0 * (c6 - 6.0 * c8);
233    let u6 = 128.0 * c8;
234
235    let c2 = 3.0 * en / 2.0 - 27.0 * en3 / 32.0;
236    let c4 = 21.0 * en2 / 16.0 - 55.0 * en4 / 32.0;
237    let c6 = 151.0 * en3 / 96.0;
238    let c8 = 1097.0 * en4 / 512.0;
239    let v0 = 2.0 * (c2 - 2.0 * c4 + 3.0 * c6 - 4.0 * c8);
240    let v2 = 8.0 * (c4 - 4.0 * c6 + 10.0 * c8);
241    let v4 = 32.0 * (c6 - 6.0 * c8);
242    let v6 = 128.0 * c8;
243
244    let r = er * (1.0 - en) * (1.0 - en * en) * (1.0 + 2.25 * en * en + (225.0 / 64.0) * en4);
245    let cosor = orlim.cos();
246    let omo = orlim
247        + orlim.sin()
248            * cosor
249            * (u0 + u2 * cosor * cosor + u4 * cosor.powf(4.0) + u6 * cosor.powf(6.0));
250    let so = sf * r * omo;
251    ZoneConstants {
252        eps,
253        r,
254        so,
255        v: [v0, v2, v4, v6],
256    }
257}

259/// The projection of one UTM zone: what `tmgeod` takes besides the point.
260struct Projection {
261    cm: f64,
262    fe: f64,
263    fn_: f64,
264    sf: f64,
265    er: f64,
266    esq: f64,
267}
268
269/// `tmgeod`: grid coordinates to geodetic latitude and west-positive
270/// longitude, radians. (CASA also computes the convergence and the point
271/// scale factor, which it only logs; they are omitted.)
272fn tmgeod(n: f64, e: f64, zone: &ZoneConstants, p: &Projection) -> (f64, f64) {
273    let [v0, v2, v4, v6] = zone.v;
274    let om = (n - p.fn_ + zone.so) / (zone.r * p.sf);
275    let cosom = om.cos();
276    let foot = om
277        + om.sin()
278            * cosom
279            * (v0 + v2 * cosom * cosom + v4 * cosom.powf(4.0) + v6 * cosom.powf(6.0));
280    let sinf = foot.sin();
281    let cosf = foot.cos();
282    let tn = sinf / cosf;
283    let ts = tn * tn;
284    let ets = zone.eps * cosf * cosf;
285    let rn = p.er * p.sf / (1.0 - p.esq * sinf * sinf).sqrt();
286    let q = (e - p.fe) / rn;
287    let qs = q * q;
288    let b2 = -tn * (1.0 + ets) / 2.0;
289    let b4 = -(5.0 + 3.0 * ts + ets * (1.0 - 9.0 * ts) - 4.0 * ets * ets) / 12.0;
290    let b6 = (61.0 + 45.0 * ts * (2.0 + ts) + ets * (46.0 - 252.0 * ts - 60.0 * ts * ts)) / 360.0;
291    let b1 = 1.0;
292    let b3 = -(1.0 + ts + ts + ets) / 6.0;
293    let b5 = (5.0 + ts * (28.0 + 24.0 * ts) + ets * (6.0 + 8.0 * ts)) / 120.0;
294    let b7 = -(61.0 + 662.0 * ts + 1320.0 * ts * ts + 720.0 * ts.powf(3.0)) / 5040.0;
295    let lat = foot + b2 * qs * (1.0 + qs * (b4 + b6 * qs));
296    let l = b1 * q * (1.0 + qs * (b3 + qs * (b5 + b7 * qs)));
297    let lon = -l / cosf + p.cm;
298    (lat, lon)
299}

301/// Convert UTM coordinates to geodetic longitude and latitude (CASA's
302/// `utm2long`).
303///
304/// - `east`, `north`: easting and northing, metres (false easting 500 km;
305///   false northing 10 000 km when `south`).
306/// - `zone`: the UTM zone number. As in CASA it is not range-checked: zones
307///   below 30 take the central meridian `183 − 6·zone` degrees west, the
308///   others `543 − 6·zone` degrees west.
309/// - `datum`: only its ellipsoid is used here; [`geodetic_to_xyz`] applies
310///   the offsets.
311/// - `south`: the southern-hemisphere flag (CASA's `nors == 'S'`).
312///
313/// Returns `(longitude, latitude)` in radians, longitude east-positive. The
314/// longitude is not normalised: for zones 30 and above it is near −360°
315/// plus the east longitude, exactly as CASA returns it.
316///
317/// CASA scale factor 0.9996, southernmost parallel 0, and no adjacent-zone
318/// correction (CASA has that code commented out).
319#[must_use]
320pub fn utm_to_geodetic(east: f64, north: f64, zone: i64, datum: &Datum, south: bool) -> (f64, f64) {
321    let rad = 180.0 / std::f64::consts::PI;
322    let er = datum.ellipsoid.equatorial_radius_m;
323    let rf = datum.ellipsoid.inverse_flattening;
324    let f = 1.0 / rf;
325    let esq = 2.0 * f - f * f;
326    // CASA's longitudes in this code are west-positive.
327    let icm = if zone < 30 {
328        183_i128 - 6 * i128::from(zone)
329    } else {
330        543_i128 - 6 * i128::from(zone)
331    };
332    // Python's float(icm): exact for any real UTM zone.
333    let cm = icm as f64 / rad;
334    let fn_ = if south { 10_000_000.0 } else { 0.0 };
335    let fe = 500_000.0;
336    let sf = 0.9996;
337    let orlim = 0.0;
338    let constants = tconpc(sf, orlim, er, esq, rf);
339    let projection = Projection {
340        cm,
341        fe,
342        fn_,
343        sf,
344        er,
345        esq,
346    };
347    let (lat, lon) = tmgeod(north, east, &constants, &projection);
348    // CASA: "convert to more usual convention of negative lon = W".
349    (-lon, lat)
350}

352/// Convert geodetic longitude, latitude and elevation on a datum to
353/// Earth-centred x, y, z (CASA's `long2xyz`).
354///
355/// `longitude_rad` is east-positive, `elevation_m` is above the datum's
356/// ellipsoid. The datum's offsets are added to the result, so for a datum
357/// other than WGS 84 the coordinates are WGS 84-centred. CASA's comment
358/// quotes the NGS check point S23°12′00″ W67°48′00″, 0 m on GRS 80, at
359/// (2216194.4188, −5430618.6472, −2497092.5213) m.
360#[must_use]
361pub fn geodetic_to_xyz(
362    longitude_rad: f64,
363    latitude_rad: f64,
364    elevation_m: f64,
365    datum: &Datum,
366) -> [f64; 3] {
367    let er = datum.ellipsoid.equatorial_radius_m;
368    let f = 1.0 / datum.ellipsoid.inverse_flattening;
369    let esq = 2.0 * f - f.powf(2.0);
370    let nu = er / (1.0 - esq * latitude_rad.sin().powf(2.0)).sqrt();
371    let x = (nu + elevation_m) * latitude_rad.cos() * longitude_rad.cos() + datum.dx_m;
372    let y = (nu + elevation_m) * latitude_rad.cos() * longitude_rad.sin() + datum.dy_m;
373    let z = ((1.0 - esq) * nu + elevation_m) * latitude_rad.sin() + datum.dz_m;
374    [x, y, z]
375}

377/// Convert UTM easting, northing and elevation to Earth-centred x, y, z
378/// (CASA's `utm2xyz`): [`utm_to_geodetic`] then [`geodetic_to_xyz`].
379#[must_use]
380pub fn utm_to_xyz(
381    easting_m: f64,
382    northing_m: f64,
383    elevation_m: f64,
384    zone: i64,
385    datum: &Datum,
386    south: bool,
387) -> [f64; 3] {
388    let (lon, lat) = utm_to_geodetic(easting_m, northing_m, zone, datum, south);
389    geodetic_to_xyz(lon, lat, elevation_m, datum)
390}

392/// Convert Earth-centred x, y, z to geodetic longitude, latitude and height
393/// on a datum (CASA's `xyz2long`, a transcription of SOFA's `gc2gde`).
394///
395/// Returns `(longitude_rad, latitude_rad, height_m)`, or `None` where CASA
396/// returns `-1, -1, -1` for invalid ellipsoid parameters (never for the
397/// tabulated datums). The datum's offsets are subtracted first.
398///
399/// One correction to CASA: CASA computes the longitude as
400/// `atan2(y − dx, x − dx)`, subtracting the x offset from y. This function
401/// subtracts `dy` from y. The two agree for WGS 84, the only datum
402/// `readantenna` converts with.
403#[must_use]
404pub fn xyz_to_geodetic(x: f64, y: f64, z: f64, datum: &Datum) -> Option<(f64, f64, f64)> {
405    let er = datum.ellipsoid.equatorial_radius_m;
406    let f = 1.0 / datum.ellipsoid.inverse_flattening;
407    if !(0.0..1.0).contains(&f) || er <= 0.0 {
408        return None;
409    }
410    let e2 = (2.0 - f) * f;
411    let e4t = e2 * e2 * 1.5;
412    let ec2 = 1.0 - e2;
413    if ec2 <= 0.0 {
414        return None;
415    }
416    let ec = ec2.sqrt();
417    let (xr, yr, zr) = (x - datum.dx_m, y - datum.dy_m, z - datum.dz_m);
418    // Distance from the polar axis.
419    let r = (xr.powf(2.0) + yr.powf(2.0)).sqrt();
420    let lon = if r > 0.0 { yr.atan2(xr) } else { 0.0 };
421    // Newton correction factors.
422    let s0 = zr.abs() / er;
423    let c0 = ec * r / er;
424    let a0 = (c0.powf(2.0) + s0.powf(2.0)).sqrt();
425    let d0 = ec * s0 * a0.powf(3.0) + e2 * s0.powf(3.0);
426    let f0 = r / er * a0.powf(3.0) - e2 * c0.powf(3.0);
427    // Halley correction factor.
428    let b0 = e4t * s0.powf(2.0) * c0.powf(2.0) * r / er * (a0 - ec);
429    let s1 = d0 * f0 - b0 * s0;
430    let cc = ec * (f0 * f0 - b0 * c0);
431    let mut lat = s1.atan2(cc);
432    let height = (r * cc + zr.abs() * s1 - er * (ec2 * s1.powf(2.0) + cc.powf(2.0)).sqrt())
433        / (s1.powf(2.0) + cc.powf(2.0)).sqrt();
434    if zr < 0.0 {
435        lat = -lat;
436    }
437    Some((lon, lat, height))
438}

440/// Convert local tangent-plane offsets to Earth-centred x, y, z (CASA's
441/// `locxyz2itrf`, from Rob Reid).
442///
443/// - `latitude_deg`, `longitude_deg`: geodetic WGS 84 position of the
444///   reference point, degrees (CASA passes the observatory's position).
445/// - `altitude_m`: the reference point's height above the ellipsoid.
446/// - `local_m`: the offsets `[x, y, z]` from that point: x east, y north,
447///   z along the ellipsoid normal, metres.
448///
449/// CASA uses the WGS 84 semi-axes `a = 6378137.0` m and
450/// `b = 6356752.3142` m (the rounded polar radius), as here.
451#[must_use]
452pub fn local_to_itrf(
453    latitude_deg: f64,
454    longitude_deg: f64,
455    altitude_m: f64,
456    local_m: [f64; 3],
457) -> [f64; 3] {
458    let [locx, locy, locz] = local_m;
459    let phi = latitude_deg.to_radians();
460    let lmbda = longitude_deg.to_radians();
461    let sphi = phi.sin();
462    let a = 6_378_137.0_f64;
463    let b = 6_356_752.314_2_f64;
464    let ae = (b / a).acos();
465    let n = a / (1.0 - (ae.sin() * sphi).powf(2.0)).sqrt();
466    let nploczcphimlocysphi = (n + locz + altitude_m) * phi.cos() - locy * sphi;
467    let clmb = lmbda.cos();
468    let slmb = lmbda.sin();
469    let x = nploczcphimlocysphi * clmb - locx * slmb;
470    let y = nploczcphimlocysphi * slmb + locx * clmb;
471    let z = (n * (b / a).powf(2.0) + locz + altitude_m) * sphi + locy * phi.cos();
472    [x, y, z]
473}

532    #[test]
533    fn local_to_itrf_axes_point_east_north_and_up() {
534        let (lat, lon, alt) = (-23.022_886, -67.754_929, 5056.8);
535        let origin = local_to_itrf(lat, lon, alt, [0.0; 3]);
536        let unit = |local: [f64; 3]| {
537            let p = local_to_itrf(lat, lon, alt, local);
538            [p[0] - origin[0], p[1] - origin[1], p[2] - origin[2]]
539        };
540        let (phi, lam) = (lat.to_radians(), lon.to_radians());
541        let east = [-lam.sin(), lam.cos(), 0.0];
542        let north = [-phi.sin() * lam.cos(), -phi.sin() * lam.sin(), phi.cos()];
543        let up = [phi.cos() * lam.cos(), phi.cos() * lam.sin(), phi.sin()];
544        for (local, axis) in [
545            ([1.0, 0.0, 0.0], east),
546            ([0.0, 1.0, 0.0], north),
547            ([0.0, 0.0, 1.0], up),
548        ] {
549            let d = unit(local);
550            for k in 0..3 {
551                assert!(
552                    (d[k] - axis[k]).abs() < 1e-8,
553                    "{local:?}: {d:?} vs {axis:?}"
554                );
555            }
556        }
557        // The origin is the geodetic point (to the rounding of CASA's b).
558        let reference = geodetic_to_xyz(lon.to_radians(), lat.to_radians(), alt, &WGS84);
559        for k in 0..3 {
560            assert!((origin[k] - reference[k]).abs() < 1e-3);
561        }
562    }
