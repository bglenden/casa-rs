3//! Ordered, zero-copy handoff from worker-owned native rows to the store.
4//! Source traversal completion remains with the existing weighting I/O fence.
5
6use super::input::{NativeStore, NativeStoreWriter, StorePlan};
7use crate::{
8    ManagedSpillStorage, WeightingExecutionState, WeightingPlanFragment, WorkExecutionContext,
9};
10use casa_imaging_model::{CompiledProblem, MeasurementSetReadAccess};
11use casa_imaging_reconstruction::{
12    WeightingReplaySummary,
13    runtime_adapter::{BandPlan, NativeBlock, NativeLayout},
14};
15use std::{io, time::Instant};
16
17pub(super) struct NativePreparation<'a> {
18    layout: Option<NativeLayout>,
19    source: &'a MeasurementSetReadAccess,
20    plan: StorePlan,
21    writer: NativeStoreWriter,
22    bands: Vec<BandPlan>,
23    output_hz: &'a [f64],
24    rows: u64,
25    previous_row: Option<u64>,
26    failed: bool,
27    support_nanos: u128,
28    write_nanos: u128,
29    pair_visits: u64,
30}
31
32pub(super) struct PreparedNative {
33    pub(super) store: NativeStore,
34    pub(super) bands: Vec<BandPlan>,
35    pub(super) layout: NativeLayout,
36}
37
38impl<'a> NativePreparation<'a> {
39    pub(super) fn traverse(
40        mut self,
41        context: WorkExecutionContext<'_>,
42        fragment: &WeightingPlanFragment<'_>,
43        problem: &CompiledProblem,
44        selected: casa_ms::BoundSelectedObservation,
45        weighting: &mut WeightingExecutionState,
46    ) -> io::Result<PreparedNative> {
47        weighting
48            .traverse_native_initial_stream(
49                context,
50                fragment,
51                problem,
52                selected,
53                |parts, layout| self.consume(parts, layout),
54            )
55            .map_err(io::Error::other)?;
56        let (replay, _, _) = weighting.pending_replay_inputs().ok_or_else(|| {
57            io::Error::other("native preparation lacks terminal source traversal")
58        })?;
59        self.finish(replay)
60    }
61
62    pub(super) fn new(
63        storage: &ManagedSpillStorage,
64        plan: StorePlan,
65        problem: &'a CompiledProblem,
66        output_hz: &'a [f64],
67        bands: Vec<BandPlan>,
68    ) -> io::Result<Self> {
69        let [source] = problem.selected_observation().read_set().sources() else {
70            return Err(io::Error::other("native cube requires one selected source"));
71        };
72        let selection = source.selection();
73        let ([dd], [spw], [pol]) = (
74            selection.data_descriptions(),
75            selection.spectral_windows(),
76            selection.correlations(),
77        ) else {
78            return Err(io::Error::other(
79                "native cube requires one homogeneous selection",
80            ));
81        };
82        if bands.is_empty()
83            || dd.spectral_window_id() != spw.spectral_window_id()
84            || dd.polarization_id() != pol.polarization_id()
85            || selection.rows().selected_row_count() != plan.rows
86            || spw.channel_indices().len() != plan.channels
87            || pol.products().len() != plan.correlations
88            || output_hz.len() != problem.geometry().spectral().output_channels()
89            || output_hz
90                .iter()
91                .enumerate()
92                .any(|(i, hz)| problem.geometry().spectral().channel_centre_hz(i) != Some(*hz))
93            || bands[0].core().start != 0
94            || bands
95                .last()
96                .is_none_or(|band| band.core().end != output_hz.len())
97            || bands
98                .windows(2)
99                .any(|pair| pair[0].core().end != pair[1].core().start)
100        {
101            return Err(io::Error::other(
102                "invalid native preparation layout or output coverage",
103            ));
104        }
105        Ok(Self {
106            layout: None,
107            source,
108            plan,
109            writer: NativeStoreWriter::create(storage, plan)?,
110            bands,
111            output_hz,
112            rows: 0,
113            previous_row: None,
114            failed: false,
115            support_nanos: 0,
116            write_nanos: 0,
117            pair_visits: 0,
118        })
119    }
120
121    pub(super) fn consume(
122        &mut self,
123        parts: &[&NativeBlock],
124        layout: &NativeLayout,
125    ) -> io::Result<()> {
126        if self.failed {
127            return Err(io::Error::other("native preparation failed earlier"));
128        }
129        self.failed = true;
130        let selection = self.source.selection();
131        let dd = selection.data_descriptions()[0];
132        let address = layout.address;
133        if address.measurement_set != self.source.measurement_set()
134            || u32::try_from(address.data_description_id).ok() != Some(dd.data_description_id())
135            || address.spectral_window_id != dd.spectral_window_id()
136            || address.polarization_id != dd.polarization_id()
137            || layout.channels != selection.spectral_windows()[0].channel_indices()
138            || !layout
139                .correlations
140                .iter()
141                .copied()
142                .eq(selection.correlations()[0]
143                    .products()
144                    .iter()
145                    .map(|c| (c.correlation_index(), c.correlation_type())))
146        {
147            return Err(io::Error::other(
148                "native input does not match selected source",
149            ));
150        }
151        for block in parts {
152            for row in &block.metadata {
153                if self
154                    .previous_row
155                    .is_some_and(|previous| row.physical_row <= previous)
156                {
157                    return Err(io::Error::other("native input row order mismatch"));
158                }
159                self.previous_row = Some(row.physical_row);
160            }
161            let started = Instant::now();
162            self.pair_visits += BandPlan::observe_all(&mut self.bands, block, self.output_hz)
163                .map_err(io::Error::other)?;
164            self.support_nanos += started.elapsed().as_nanos();
165            self.rows = self
166                .rows
167                .checked_add(block.metadata.len() as u64)
168                .ok_or_else(|| io::Error::other("native row count overflow"))?;
169        }
170        let started = Instant::now();
171        self.writer.append_parts(parts)?;
172        self.write_nanos += started.elapsed().as_nanos();
173        if self.layout.is_none() {
174            self.layout = Some(layout.clone());
175        }
176        self.failed = false;
177        Ok(())
178    }
179
180    pub(super) fn finish(self, replay: &WeightingReplaySummary) -> io::Result<PreparedNative> {
181        if self.failed
182            || self.rows != self.plan.rows
183            || self
184                .rows
185                .checked_mul(self.plan.channels as u64)
186                .and_then(|n| n.checked_mul(self.plan.correlations as u64))
187                != Some(replay.sample_count())
188        {
189            return Err(io::Error::other(
190                "native preparation did not consume the complete source",
191            ));
192        }
193        eprintln!(
194            "streaming_cube_preparation support_nanos={} write_nanos={} pair_visits={} coverage_bytes={} coverage_hash_calls={}",
195            self.support_nanos,
196            self.write_nanos,
197            self.pair_visits,
198            replay.coverage_proof_bytes(),
199            replay.coverage_proof_hash_calls()
200        );
201        Ok(PreparedNative {
202            store: self.writer.finish()?,
203            bands: self.bands,
204            layout: self
205                .layout
206                .ok_or_else(|| io::Error::other("native preparation is empty"))?,
207        })
208    }
209}
