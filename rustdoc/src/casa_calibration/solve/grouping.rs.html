24pub(crate) fn validate_solve_interval(
25    solve_interval: GainSolveInterval,
26) -> Result<(), GainSolveError> {
27    match solve_interval {
28        GainSolveInterval::Infinite | GainSolveInterval::Integration => Ok(()),
29        GainSolveInterval::Seconds(seconds) if seconds > 0.0 => Ok(()),
30        _ => Err(GainSolveError::UnsupportedSolveInterval { solve_interval }),
31    }
32}

34pub(crate) fn validate_smodel(smodel: [f32; 4]) -> Result<(), GainSolveError> {
35    if smodel[0] <= 0.0 || smodel[1] != 0.0 || smodel[2] != 0.0 || smodel[3] != 0.0 {
36        return Err(GainSolveError::UnsupportedSkyModel { smodel });
37    }
38    Ok(())
39}

41pub(crate) fn load_preapplied_rows(
42    ms: &MeasurementSet,
43    selection: &MsSelection,
44    prior_calibration_tables: &[ApplyCalibrationTableSpec],
45    parang: bool,
46) -> Result<Option<HashMap<usize, EvaluatedApplyRow>>, GainSolveError> {
47    if prior_calibration_tables.is_empty() && !parang {
48        return Ok(None);
49    }
50    let plan = plan_apply(
51        ms,
52        &ApplyPlanRequest {
53            selection: selection.clone(),
54            apply_mode: crate::ApplyMode::CalFlag,
55            parang,
56            calibration_tables: prior_calibration_tables.to_vec(),
57        },
58    )
59    .map_err(|source| GainSolveError::PriorCalibrationPlan {
60        source: Box::new(source),
61    })?;
62    evaluate_apply_rows(ms, &plan).map(Some).map_err(|source| {
63        GainSolveError::PriorCalibrationApply {
64            source: Box::new(source),
65        }
66    })
67}

69pub(crate) fn build_solve_groups(
70    ms: &MeasurementSet,
71    rows: &[SelectedSolveRow],
72    preapplied_rows: Option<&HashMap<usize, EvaluatedApplyRow>>,
73    options: SolveGroupOptions,
74) -> Result<BTreeMap<(SolveBaseKey, SolveBucketKey), SolveAccumulator>, GainSolveError> {
75    let mut groups = BTreeMap::<(SolveBaseKey, SolveBucketKey), SolveAccumulator>::new();
76    for (row, base_key, bucket_key) in bucketed_rows(rows, options.solve_interval, options.combine)
77    {
78        // A group takes its field from its earliest row, as CASA's
79        // `SDBList::aggregateFld` takes the first buffer's.
80        let group = groups.entry((base_key, bucket_key)).or_insert_with(|| {
81            SolveAccumulator::new(row.field_id, row.data_spw_id, row.observation_id)
82        });
83        let preapplied = preapplied_rows.and_then(|rows| rows.get(&row.row_index));
84        group.observe(ms, row, preapplied, &options)?;
85    }
86
87    Ok(groups)
88}

90/// Each cross-correlation row of `rows`, in solve order, with the group and
91/// interval it is solved in.
92///
93/// CASA's iteration sorts by the axes `combine` keeps apart and then by
94/// time, so the rows of a group come in time order whatever their fields
95/// and scans, and a `solint` interval starts at its group's earliest time.
96fn bucketed_rows(
97    rows: &[SelectedSolveRow],
98    solve_interval: GainSolveInterval,
99    combine: GainSolveCombine,
100) -> Vec<(&SelectedSolveRow, SolveBaseKey, SolveBucketKey)> {
101    let mut keyed = rows
102        .iter()
103        .filter(|row| row.antenna1 != row.antenna2)
104        .map(|row| {
105            let base_key = SolveBaseKey {
106                field_id: if combine.fields { 0 } else { row.field_id },
107                spw_id: row.data_spw_id,
108                observation_id: row.observation_id,
109                scan_number: if combine.scans { 0 } else { row.scan_number },
110            };
111            (row, base_key)
112        })
113        .collect::<Vec<_>>();
114    keyed.sort_by(|(a, a_key), (b, b_key)| {
115        a_key
116            .cmp(b_key)
117            .then(a.time_seconds.total_cmp(&b.time_seconds))
118            .then(a.interval_seconds.total_cmp(&b.interval_seconds))
119            .then(a.row_index.cmp(&b.row_index))
120    });
121
122    let mut bucket_starts = BTreeMap::<SolveBaseKey, f64>::new();
123    keyed
124        .into_iter()
125        .map(|(row, base_key)| {
126            let bucket_key = match solve_interval {
127                GainSolveInterval::Infinite => SolveBucketKey::Infinite,
128                GainSolveInterval::Integration => SolveBucketKey::Integration {
129                    time_bits: row.time_seconds.to_bits(),
130                    interval_bits: row.interval_seconds.to_bits(),
131                },
132                GainSolveInterval::Seconds(seconds) => {
133                    let start = *bucket_starts.entry(base_key).or_insert(row.time_seconds);
134                    let index = ((row.time_seconds - start) / seconds).floor().max(0.0);
135                    SolveBucketKey::Seconds(index as u64)
136                }
137            };
138            (row, base_key, bucket_key)
139        })
140        .collect()
141}

143/// Refuses `rows` when two of their data descriptions share a spectral
144/// window, whatever their polarizations: CASA solves each description
145/// apart, where casa-rs's groups and tables are keyed by window.
146pub(crate) fn refuse_shared_spectral_windows(
147    rows: &[SelectedSolveRow],
148) -> Result<(), GainSolveError> {
149    let mut descriptions = BTreeMap::<i32, BTreeSet<i32>>::new();
150    for row in rows {
151        descriptions
152            .entry(row.data_spw_id)
153            .or_default()
154            .insert(row.data_desc_id);
155    }
156    match descriptions.into_iter().find(|(_, ids)| ids.len() > 1) {
157        Some((spw_id, ids)) => Err(GainSolveError::SharedSpectralWindow {
158            spw_id,
159            data_desc_ids: ids.into_iter().collect(),
160        }),
161        None => Ok(()),
162    }
163}

165pub(crate) fn all_antenna_ids(ms: &MeasurementSet) -> Result<BTreeSet<i32>, GainSolveError> {
166    let antenna = ms
167        .antenna()
168        .map_err(|source| GainSolveError::OpenMeasurementSet {
169            path: ms
170                .path()
171                .map(|path| path.display().to_string())
172                .unwrap_or_else(|| "<in-memory>".to_string()),
173            source,
174        })?;
175    Ok((0..antenna.row_count())
176        .map(|row| i32::try_from(row).expect("antenna row count should fit in i32"))
177        .collect())

180pub(crate) fn resolve_refant(
181    ms: &MeasurementSet,
182    selector: &RefAntSelector,
183) -> Result<i32, GainSolveError> {
184    match selector {
185        RefAntSelector::AntennaId(id) => Ok(*id),
186        RefAntSelector::AntennaName(name) => {
187            let antenna = ms
188                .antenna()
189                .map_err(|error| GainSolveError::ResolveRefAnt {
190                    selector: name.clone(),
191                    reason: error.to_string(),
192                })?;
193            for row in 0..antenna.row_count() {
194                if antenna
195                    .name(row)
196                    .map_err(|error| GainSolveError::ResolveRefAnt {
197                        selector: name.clone(),
198                        reason: error.to_string(),
199                    })?
200                    == *name
201                {
202                    return Ok(row as i32);
203                }
204            }
205            Err(GainSolveError::ResolveRefAnt {
206                selector: name.clone(),
207                reason: "no ANTENNA.NAME match".to_string(),
208            })
209        }
210    }
211}

213pub(crate) fn collect_selected_rows(
214    ms: &MeasurementSet,
215    selection: &MsSelection,
216) -> Result<Vec<SelectedSolveRow>, GainSolveError> {
217    let selected_rows = resolve_calibration_selection(ms, selection).map_err(|error| {
218        GainSolveError::OpenMeasurementSet {
219            path: ms
220                .path()
221                .map(|path| path.display().to_string())
222                .unwrap_or_else(|| "<in-memory>".to_string()),
223            source: match error {
224                casa_ms::MsSelectionError::Domain(source) => source,
225                other => MsError::InvalidInput(other.to_string()),
226            },
227        }
228    })?;
229    if selected_rows.selected_rows.is_empty() {
230        return Err(GainSolveError::EmptySelection);
231    }
232
233    selected_rows
234        .selected_rows
235        .into_iter()
236        .map(|row| {
237            Ok(SelectedSolveRow {
238                row_index: row.row_index,
239                field_id: row.field_id,
240                observation_id: get_i32(ms.main_table(), row.row_index, "OBSERVATION_ID")?,
241                data_desc_id: row.data_desc_id,
242                data_spw_id: row.spectral_window_id,
243                channel_selection: selection.channel_selection_for_spw(row.spectral_window_id),
244                antenna1: row.antenna1_id,
245                antenna2: row.antenna2_id,
246                time_seconds: row.time_mjd_seconds,
247                time_centroid_seconds: get_f64(ms.main_table(), row.row_index, "TIME_CENTROID")?,
248                interval_seconds: get_f64(ms.main_table(), row.row_index, "INTERVAL")?,
249                scan_number: get_i32(ms.main_table(), row.row_index, "SCAN_NUMBER")?,
250                flag_row: row.flag_row,
251            })
252        })
253        .collect()
254}

256pub(crate) fn correlation_types_for_ddid(
257    ms: &MeasurementSet,
258    data_desc_id: i32,
259) -> Result<Vec<i32>, GainSolveError> {
260    let dd = ms
261        .data_description()
262        .map_err(|source| GainSolveError::OpenMeasurementSet {
263            path: ms
264                .path()
265                .map(|path| path.display().to_string())
266                .unwrap_or_else(|| "<in-memory>".to_string()),
267            source,
268        })?;
269    let pol = ms
270        .polarization()
271        .map_err(|source| GainSolveError::OpenMeasurementSet {
272            path: ms
273                .path()
274                .map(|path| path.display().to_string())
275                .unwrap_or_else(|| "<in-memory>".to_string()),
276            source,
277        })?;
278    let row = usize::try_from(data_desc_id).map_err(|_| {
279        GainSolveError::UnsupportedCorrelationLayout {
280            data_desc_id,
281            correlation_types: Vec::new(),
282        }
283    })?;
284    let pol_id = dd
285        .polarization_id(row)
286        .map_err(|source| GainSolveError::OpenMeasurementSet {
287            path: ms
288                .path()
289                .map(|path| path.display().to_string())
290                .unwrap_or_else(|| "<in-memory>".to_string()),
291            source,
292        })?;
293    pol.corr_type(usize::try_from(pol_id).unwrap_or(usize::MAX))
294        .map_err(|source| GainSolveError::OpenMeasurementSet {
295            path: ms
296                .path()
297                .map(|path| path.display().to_string())
298                .unwrap_or_else(|| "<in-memory>".to_string()),
299            source,
300        })
301}

488impl SolveAccumulator {
489    pub(crate) fn new(field_id: i32, spw_id: i32, observation_id: i32) -> Self {
490        Self {
491            field_id,
492            spw_id,
493            observation_id,
494            min_time: f64::INFINITY,
495            max_time: f64::NEG_INFINITY,
496            unique_time_sum: 0.0,
497            unique_time_count: 0,
498            total_interval: 0.0,
499            sample_rows: 0,
500            field_ids: BTreeSet::new(),
501            scan_numbers: BTreeSet::new(),
502            antenna_ids: BTreeSet::new(),
503            unique_time_bits: BTreeSet::new(),
504            solution_channels: 1,
505            channel_graphs: Vec::new(),
506            solution_time: 0.0,
507            attempted: false,
508            baseline_centroids: HashMap::new(),
509            receptor_accumulations: Vec::new(),
510        }
511    }
512
513    /// Add one MAIN row: its unflagged, positively weighted samples divided by
514    /// the model, each weighted by its solve weight times |model|² as
515    /// `VisEquation::divideCorrByModel` does, and its TIME_CENTROID.
516    pub(crate) fn observe(
517        &mut self,
518        ms: &MeasurementSet,
519        row: &SelectedSolveRow,
520        preapplied_row: Option<&EvaluatedApplyRow>,
521        options: &SolveGroupOptions,
522    ) -> Result<(), GainSolveError> {
523        let inputs = read_row_inputs(ms, row, preapplied_row, options.model_source)?;
524        let correlation_types = correlation_types_for_ddid(ms, row.data_desc_id)?;
525        let channel_count = inputs.data.shape()[1];
526        let channels = if options.channelized {
527            (0..channel_count).collect()
528        } else {
529            selected_channels(row.channel_selection.as_ref(), channel_count)?
530        };
531        self.observe_samples(
532            &SolveSampleRow::of(row),
533            &inputs.samples(),
534            &correlation_types,
535            &channels,
536            options,
537        )
538    }
539
540    /// Accumulate one sample's `channels` (every channel for a channelized
541    /// solve): the arithmetic of [`Self::observe`] on arrays from any source.
542    /// The MSv2/MSv4 trial (`docs/measurement-set/trial/README.md`) feeds it
543    /// from `casa_observation::VisibilityBatch` views without copying them.
544    pub(crate) fn observe_samples(
545        &mut self,
546        row: &SolveSampleRow,
547        samples: &SolveSamples<'_>,
548        correlation_types: &[i32],
549        channels: &[usize],
550        options: &SolveGroupOptions,
551    ) -> Result<(), GainSolveError> {
552        if options.channelized {
553            let channel_count = samples.data.shape()[1];
554            if self.sample_rows > 0 && self.solution_channels != channel_count {
555                return Err(GainSolveError::UnsupportedParameterShape {
556                    path: "<measurement-set DATA>".to_string(),
557                    shape: samples.data.shape().to_vec(),
558                });
559            }
560            self.solution_channels = channel_count;
561        }
562
563        let graph_count = match options.gain_type {
564            GainType::G => 2,
565            GainType::T => 1,
566        };
567        if self.receptor_accumulations.len() < graph_count {
568            self.receptor_accumulations
569                .resize_with(graph_count, HashMap::new);
570        }
571        self.observe_metadata(row);
572
573        let mut row_weight = 0.0_f64;
574        for (corr_index, &correlation_type) in correlation_types.iter().enumerate() {
575            let Some(receptors) = correlation_receptors(correlation_type) else {
576                return Err(GainSolveError::UnsupportedCorrelationLayout {
577                    data_desc_id: row.data_desc_id,
578                    correlation_types: correlation_types
579                        .iter()
580                        .map(|code| stokes_name(*code).to_string())
581                        .collect(),
582                });
583            };
584            if receptors.0 != receptors.1 {
585                continue;
586            }
587            let graph_index = match options.gain_type {
588                GainType::G => receptors.0,
589                GainType::T => 0,
590            };
591            let mut channel_weights = Vec::with_capacity(channels.len());
592            for &chan_index in channels {
593                let weight = self.accumulate_sample(
594                    graph_index,
595                    row,
596                    samples,
597                    options,
598                    corr_index,
599                    chan_index,
600                );
601                channel_weights.push(weight);
602            }
603            row_weight += if options.channelized {
604                f64::from(median_with_even_mean(&mut channel_weights))
605            } else {
606                channel_weights
607                    .iter()
608                    .map(|weight| f64::from(*weight))
609                    .sum()
610            };
611        }
612        if row_weight > 0.0 {
613            let centroid = self
614                .baseline_centroids
615                .entry((row.antenna1, row.antenna2))
616                .or_default();
617            centroid.weighted_sum += row_weight * row.time_centroid_seconds;
618            centroid.weight += row_weight;
619        }
620        Ok(())

638    /// Accumulate one sample if it can enter the solve; return its effective
639    /// weight, zero when it is flagged or unusable.
640    fn accumulate_sample(
641        &mut self,
642        graph_index: usize,
643        row: &SolveSampleRow,
644        samples: &SolveSamples<'_>,
645        options: &SolveGroupOptions,
646        corr_index: usize,
647        chan_index: usize,
648    ) -> f32 {
649        let weight = samples.weights[[corr_index, chan_index]];
650        if weight <= 0.0 || solve_sample_flagged(&samples.flags, options.gain_type, chan_index) {
651            return 0.0;
652        }
653        let model = match samples.model {
654            Some(model_data) => model_data[[corr_index, chan_index]],
655            None => Complex32::new(options.stokes_i, 0.0),
656        };
657        let model_norm_sqr = model.norm_sqr();
658        if model_norm_sqr <= f32::EPSILON {
659            return 0.0;
660        }
661        self.attempted = true;
662        let sample = samples.data[[corr_index, chan_index]] / model;
663        if sample.norm() <= f32::EPSILON {
664            return 0.0;
665        }
666        let effective_weight = weight * model_norm_sqr;
667        let accumulation = self.receptor_accumulations[graph_index]
668            .entry(SolveAccumulationKey {
669                baseline: (row.antenna1, row.antenna2),
670                correlation_index: corr_index,
671                channel_index: if options.channelized { chan_index } else { 0 },
672            })
673            .or_default();
674        accumulation.weighted_sum += sample * Complex32::new(effective_weight, 0.0);
675        accumulation.weight += effective_weight;
676        accumulation.weighted_sample_power +=
677            f64::from(effective_weight) * f64::from(sample.norm_sqr());
678        accumulation.sample_count += 1;
679        effective_weight
680    }

682    /// Collapse each baseline's samples per correlation and solution channel
683    /// into the solve graphs, and set the solution TIME.
684    ///
685    /// A phase-only solve sees each collapsed sample at unit amplitude with
686    /// its weight times its squared amplitude
687    /// (`SolveDataBuffer::enforceAPonData`). TIME is
688    /// `SDBList::aggregateTimeCentroid`: each baseline's TIME_CENTROID,
689    /// averaged over its rows by row weight as time averaging does, weighted
690    /// by the baseline's solve weight after that phase-only scaling.
691    pub(crate) fn finalize_for_solve(&mut self, solve_mode: GainSolveMode) {
692        self.channel_graphs = vec![
693            vec![SolveGraph::default(); self.receptor_accumulations.len()];
694            self.solution_channels
695        ];
696        let mut baseline_weights = HashMap::<(i32, i32), f64>::new();
697        for (receptor, accumulations) in self.receptor_accumulations.iter().enumerate() {
698            for (key, accumulation) in accumulations {
699                if accumulation.weight <= f32::EPSILON {
700                    continue;
701                }
702                let averaged = accumulation.weighted_sum / Complex32::new(accumulation.weight, 0.0);
703                let (sample, weight) = match solve_mode {
704                    GainSolveMode::Phase => {
705                        let amplitude = averaged.norm();
706                        if amplitude <= f32::EPSILON {
707                            continue;
708                        }
709                        (
710                            averaged / Complex32::new(amplitude, 0.0),
711                            accumulation.weight * amplitude * amplitude,
712                        )
713                    }
714                    GainSolveMode::AmplitudePhase => (averaged, accumulation.weight),
715                };
716                self.channel_graphs[key.channel_index][receptor].accumulate(
717                    key.baseline,
718                    weight,
719                    sample * Complex32::new(weight, 0.0),
720                    SolveEdgeStats {
721                        weighted_sample_power: f64::from(weight) * f64::from(sample.norm_sqr()),
722                        raw_weighted_sample_power: accumulation.weighted_sample_power,
723                        sample_count: accumulation.sample_count,
724                        collapsed_count: 1,
725                    },
726                );
727                *baseline_weights.entry(key.baseline).or_default() += f64::from(weight);
728            }
729        }
730
731        let mut solution_time = WeightedTime::default();
732        for (baseline, weight) in baseline_weights {
733            let Some(centroid) = self.baseline_centroids.get(&baseline) else {
734                continue;
735            };
736            if centroid.weight > 0.0 {
737                solution_time.weighted_sum += weight * centroid.weighted_sum / centroid.weight;
738                solution_time.weight += weight;
739            }
740        }
741        self.solution_time = if solution_time.weight > 0.0 {
742            solution_time.weighted_sum / solution_time.weight
743        } else {
744            self.aggregate_time()
745        };
746    }
747
748    /// Add `other`'s accumulation, of the same group, to this one before
749    /// [`Self::finalize_for_solve`], as if its samples had been observed
750    /// here: the fragment work units of the MSv2/MSv4 trial accumulate apart
751    /// and merge before the group closes. Sums change order, so results
752    /// agree to rounding, not bitwise.
753    #[cfg(test)]
754    pub(crate) fn merge(&mut self, other: Self) {
755        self.min_time = self.min_time.min(other.min_time);
756        self.max_time = self.max_time.max(other.max_time);
757        for bits in other.unique_time_bits {
758            if self.unique_time_bits.insert(bits) {
759                self.unique_time_sum += f64::from_bits(bits);
760                self.unique_time_count += 1;
761            }
762        }
763        self.total_interval += other.total_interval;
764        self.sample_rows += other.sample_rows;
765        self.field_ids.extend(other.field_ids);
766        self.scan_numbers.extend(other.scan_numbers);
767        self.antenna_ids.extend(other.antenna_ids);
768        self.solution_channels = self.solution_channels.max(other.solution_channels);
769        self.attempted |= other.attempted;
770        for (baseline, centroid) in other.baseline_centroids {
771            let mine = self.baseline_centroids.entry(baseline).or_default();
772            mine.weighted_sum += centroid.weighted_sum;
773            mine.weight += centroid.weight;
774        }
775        if self.receptor_accumulations.len() < other.receptor_accumulations.len() {
776            self.receptor_accumulations
777                .resize_with(other.receptor_accumulations.len(), HashMap::new);
778        }
779        for (mine, theirs) in self
780            .receptor_accumulations
781            .iter_mut()
782            .zip(other.receptor_accumulations)
783        {
784            for (key, accumulation) in theirs {
785                let entry = mine.entry(key).or_default();
786                entry.weighted_sum += accumulation.weighted_sum;
787                entry.weight += accumulation.weight;
788                entry.weighted_sample_power += accumulation.weighted_sample_power;
789                entry.sample_count += accumulation.sample_count;
790            }
791        }
792    }

794    /// Entries the accumulation's sets and maps hold: its state grows with
795    /// the group's distinct times, antennas, baselines and baseline
796    /// correlations, not with its samples.
797    #[cfg(test)]
798    pub(crate) fn state_entries(&self) -> usize {
799        self.unique_time_bits.len()
800            + self.scan_numbers.len()
801            + self.antenna_ids.len()
802            + self.baseline_centroids.len()
803            + self
804                .receptor_accumulations
805                .iter()
806                .map(HashMap::len)
807                .sum::<usize>()
808    }

810    /// The mean of the interval's distinct TIMEs (`SDBList::aggregateTime`,
811    /// CASA's TIME when no sample carries weight).
812    pub(crate) fn aggregate_time(&self) -> f64 {
813        if self.unique_time_count > 0 {
814            self.unique_time_sum / self.unique_time_count as f64
815        } else {
816            (self.min_time + self.max_time) / 2.0
817        }
818    }

821/// Read one MAIN row's data, flags, model and solve weights, from the row or
822/// from its pre-applied calibration.
823///
824/// The weights are CASA's (`CalSolvingVi2::calibrateCurrentVB`, which calls
825/// `VisBufferImpl2::resetWeightsUsingSigma`): 1/σ² from SIGMA_SPECTRUM when
826/// the MeasurementSet has WEIGHT_SPECTRUM and SIGMA_SPECTRUM, otherwise from
827/// SIGMA broadcast over channels; σ ≤ 0 gives no weight. WEIGHT and
828/// WEIGHT_SPECTRUM are not read. Pre-applied `calwt` tables then scale them.
829fn read_row_inputs<'a>(
830    ms: &'a MeasurementSet,
831    row: &SelectedSolveRow,
832    preapplied_row: Option<&'a EvaluatedApplyRow>,
833    model_source: GainSolveModelSource,
834) -> Result<SolveRowInputs<'a>, GainSolveError> {
835    let (data, flags) = match preapplied_row {
836        Some(preapplied) => (&preapplied.corrected_data, &preapplied.flags),
837        None => (
838            main_cell(ms, row.row_index, "DATA")?,
839            main_cell(ms, row.row_index, "FLAG")?,
840        ),
841    };
842    let ArrayValue::Complex32(data) = data else {
843        return Err(unsupported_shape("DATA", data));
844    };
845    let ArrayValue::Bool(flags) = flags else {
846        return Err(unsupported_shape("FLAG", flags));
847    };
848    if data.ndim() != 2 || data.shape() != flags.shape() {
849        return Err(GainSolveError::UnsupportedParameterShape {
850            path: "<measurement-set row>".to_string(),
851            shape: data.shape().to_vec(),
852        });
853    }
854    let model = match model_source {
855        GainSolveModelSource::ModelColumn => {
856            let model = main_cell(ms, row.row_index, "MODEL_DATA")?;
857            match model {
858                ArrayValue::Complex32(model) if model.shape() == data.shape() => Some(model),
859                other => return Err(unsupported_shape("MODEL_DATA", other)),
860            }
861        }
862        GainSolveModelSource::PointSource => None,
863    };
864    let mut weights = solve_weights(ms, row.row_index, data.shape())?;
865    if let Some(factors) = preapplied_row.and_then(|row| row.weight_factors.as_ref()) {
866        if factors.shape() != weights.shape() {
867            return Err(GainSolveError::UnsupportedParameterShape {
868                path: "<pre-applied weight factors>".to_string(),
869                shape: factors.shape().to_vec(),
870            });
871        }
872        weights *= factors;
873    }
874    Ok(SolveRowInputs {
875        data,
876        flags: solve_flags(row, flags),
877        weights,
878        model,
879    })
880}

882fn solve_weights(
883    ms: &MeasurementSet,
884    row_index: usize,
885    shape: &[usize],
886) -> Result<ArrayD<f32>, GainSolveError> {
887    let has_column = |name| {
888        ms.main_table()
889            .schema()
890            .is_some_and(|schema| schema.contains_column(name))
891    };
892    let inverse_square = |sigma: f32| {
893        if sigma > 0.0 {
894            1.0 / (sigma * sigma)
895        } else {
896            0.0
897        }
898    };
899    if has_column("WEIGHT_SPECTRUM") && has_column("SIGMA_SPECTRUM") {
900        let sigma = main_cell(ms, row_index, "SIGMA_SPECTRUM")?;
901        return match sigma {
902            ArrayValue::Float32(sigma) if sigma.shape() == shape => Ok(sigma.mapv(inverse_square)),
903            other => Err(unsupported_shape("SIGMA_SPECTRUM", other)),
904        };
905    }
906    let sigma = main_cell(ms, row_index, "SIGMA")?;
907    let ArrayValue::Float32(sigma) = sigma else {
908        return Err(unsupported_shape("SIGMA", sigma));
909    };
910    if sigma.ndim() != 1 || sigma.shape()[0] != shape[0] {
911        return Err(GainSolveError::UnsupportedParameterShape {
912            path: "<measurement-set SIGMA>".to_string(),
913            shape: sigma.shape().to_vec(),
914        });
915    }
916    Ok(ArrayD::from_shape_fn(shape, |index| {
917        inverse_square(sigma[[index[0]]])
918    }))
919}

921fn main_cell<'a>(
922    ms: &'a MeasurementSet,
923    row_index: usize,
924    column: &str,
925) -> Result<&'a ArrayValue, GainSolveError> {
926    ms.main_table()
927        .cell_accessor(row_index, column)
928        .and_then(|cell| cell.array())
929        .map_err(|source| GainSolveError::OpenMeasurementSet {
930            path: ms
931                .path()
932                .map(|path| path.display().to_string())
933                .unwrap_or_else(|| "<in-memory>".to_string()),
934            source: MsError::from(source),
935        })
936}

945/// The median, the mean of the middle two for an even count, as casacore's
946/// `partialMedians(..., takeEvenMean=True)` that CASA uses for a row's
947/// channel-independent weight.
948fn median_with_even_mean(values: &mut [f32]) -> f32 {
949    if values.is_empty() {
950        return 0.0;
951    }
952    values.sort_by(f32::total_cmp);
953    let middle = values.len() / 2;
954    if values.len().is_multiple_of(2) {
955        (values[middle - 1] + values[middle]) / 2.0
956    } else {
957        values[middle]
958    }
959}

981/// The sample flags a solve uses for `row`: its `FLAG` cell, or every sample
982/// flagged when the row's `FLAG_ROW` is set, whatever `FLAG` holds. CASA's
983/// solve does the same: `CalSolvingVi2::calibrateCurrentVB` calls
984/// `VisEquation::collapse2`, whose `divideCorrByModel` flags the whole cube
985/// of such a row and zeroes its weights before G, T or B accumulates it.
986pub(crate) fn solve_flags<'a>(
987    row: &SelectedSolveRow,
988    flags: &'a ArrayD<bool>,
989) -> Cow<'a, ArrayD<bool>> {
990    if row.flag_row {
991        Cow::Owned(ArrayD::from_elem(flags.raw_dim(), true))
992    } else {
993        Cow::Borrowed(flags)
994    }
995}

997/// Correlation-independent flags, as `CalSolvingVi2::corrIndepFlags` makes
998/// them: a channel any correlation flags is flagged for all.
999fn solve_sample_flagged(
1000    flags: &ArrayView2<'_, bool>,
1001    gain_type: GainType,
1002    chan_index: usize,
1003) -> bool {
1004    match gain_type {
1005        GainType::G | GainType::T => {
1006            (0..flags.shape()[0]).any(|corr_index| flags[[corr_index, chan_index]])
1007        }
1008    }
