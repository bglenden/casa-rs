5/// One column to install in an existing disk table.
6pub struct ColumnAddition {
7    /// The column's schema.
8    pub schema: ColumnSchema,
9    /// The default cell for each existing row, or `None` if undefined.
10    pub default: Option<Value>,
11    /// Keyword values stored with the new descriptor.
12    pub keywords: RecordValue,
13    /// Preferred manager class when a new binding is needed.
14    pub manager_type: Option<String>,
15    /// Preferred group name when a new binding is needed.
16    pub manager_group: Option<String>,
17}
18
19/// Install the new columns together in their own storage manager.
20pub struct ManagerAddition {
21    /// Casacore manager class (for example, `StandardStMan`).
22    pub kind: String,
23    /// Name of this manager in the table.
24    pub name: String,
25    /// Manager configuration (such as tiled default shape).
26    pub spec: RecordValue,
27}
28
29impl Table {
30    /// Append rows to a plain disk table without changing its storage
31    /// manager bindings. Standard and AipsIO managers use `defaults`;
32    /// incremental managers inherit their preceding values and variable
33    /// tiled arrays remain undefined.
34    ///
35    /// # Errors
36    ///
37    /// The table has no durable plain layout, has unsaved edits, contains
38    /// a storage manager that cannot append, or cannot be locked or written.
39    pub fn append_rows_in_place(
40        &mut self,
41        count: usize,
42        defaults: &RecordValue,
43    ) -> Result<(), TableError> {
44        if count == 0 {
45            return Ok(());
46        }
47        self.edit_rows_in_place(Rows::Append { count, defaults })
48    }
49
50    /// Keep the given rows in an existing plain table, in ascending order,
51    /// and delete all others. This operation preserves manager bindings.
52    ///
53    /// # Errors
54    ///
55    /// The row sequence is unsorted or contains repetitions or invalid row
56    /// numbers; the table cannot be written or has a manager that refuses
57    /// to remove rows (for example, IncrementalStMan).
58    pub fn retain_rows_in_place(&mut self, kept: &[usize]) -> Result<(), TableError> {
59        if kept.windows(2).any(|pair| pair[0] >= pair[1]) {
60            return Err(TableError::Storage(
61                "retained rows must strictly increase".into(),
62            ));
63        }
64        self.edit_rows_in_place(Rows::Keep(kept))
65    }
66
67    /// Add columns without changing the files for managers that do not
68    /// receive a column. When supplied, `manager` binds all the new
69    /// columns to one new manager. Otherwise the first manager that
70    /// accepts addition supplies the storage, as in casacore.
71    ///
72    /// # Errors
73    ///
74    /// The table is not a clean plain table, a column or manager name
75    /// already exists, the selected binding is unsupported, or a storage
76    /// file cannot be read or written.
77    pub fn add_columns_in_place(
78        &mut self,
79        columns: Vec<ColumnAddition>,
80        manager: Option<ManagerAddition>,
81    ) -> Result<(), TableError> {
82        if columns.is_empty() {
83            return Ok(());
84        }
85        self.edit_clean_layout("add columns in place", move |this, path, mut dat, old| {
86            dat.raise_row_count(old);
87            let plan =
88                StagedEdit::stage_columns(path, dat, columns, manager, this.data_manager_info())?;
89            this.note_persisted_write(path);
90            plan.publish()?;
91            this.reload_disk_state_at(path)
92        })
93    }
94
95    fn edit_rows_in_place(&mut self, rows: Rows<'_>) -> Result<(), TableError> {
96        self.edit_clean_layout("edit rows in place", |this, path, dat, old| {
97            if let Rows::Keep(kept) = rows {
98                if kept.last().is_some_and(|&row| row as u64 >= old) {
99                    return Err(TableError::Storage("retained row is out of range".into()));
100                }
101                if kept.len() as u64 == old {
102                    return Ok(());
103                }
104            }
105            let plan = StagedEdit::stage(path, dat, old, rows)?;
106            this.note_persisted_write(path);
107            plan.publish()?;
108            // Publish the new count before reopening; the old lock sync
109            // count otherwise hides removals.
110            this.note_full_save(path)?;
111            this.reload_disk_state_at(path)
112        })
113    }
114
115    fn edit_clean_layout(
116        &mut self,
117        operation: &str,
118        edit: impl FnOnce(
119            &mut Self,
120            &Path,
121            crate::storage::TableDatContents,
122            u64,
123        ) -> Result<(), TableError>,
124    ) -> Result<(), TableError> {
125        let path = self.source_path.clone().ok_or_else(|| {
126            TableError::Storage("in-place row edits require an existing table".into())
127        })?;
128        if self.kind != TableKind::Plain || self.holds_unflushed_changes() {
129            return Err(TableError::Storage(
130                "in-place row edits require a clean plain table".into(),
131            ));
132        }
133        let auto_unlock = self.begin_write_operation(operation)?;
134        let result = (|| {
135            #[cfg(unix)]
136            if auto_unlock {
137                self.reload_if_published()?;
138            }
139            let dat = crate::storage::table_control::read_table_dat(
140                &path.join(crate::storage::TABLE_CONTROL_FILE),
141            )?;
142            let old = self.require_persisted_rows(&path, &dat)?;
143            edit(self, &path, dat, old)
144        })();
145        self.finish_write_operation(auto_unlock, result)
146    }
147}
