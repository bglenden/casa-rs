3//! CASA-style restoring-beam fitting from a PSF plane.
4//!
5//! This follows the `StokesImageUtil::FitGaussianPSF` workflow: find the PSF
6//! peak, walk the main-lobe samples above `psfcutoff`, expand a padded square
7//! window, bicubically oversample it, renormalize, collect second-pass
8//! samples above the cutoff, and fit an elliptical Gaussian with damped
9//! Levenberg-Marquardt refinement whose damped normal equations solve in
10//! casacore `LSQFit` LDLT order. Fitted values pass through casacore's
11//! Float rounding before becoming beam metadata.
12

14    EllipticalGaussian, EllipticalGaussianError, common_enclosing_gaussian, deconvolving_gaussian,
15    solve_symmetric_ldlt_casacore,
16};

21/// Why a CASA restoring beam could not be formed or fitted from a PSF.
22#[derive(Debug, Clone, Copy, PartialEq, Error)]
23pub enum PsfBeamFitError {
24    /// A beam's axes are not finite and positive, or its minor axis exceeds
25    /// its major axis ([`RestoringBeam::new`]).
26    #[error("restoring beam requires finite positive major >= minor axes")]
27    InvalidBeam,
28    /// Beams could not be enclosed or deconvolved
29    /// ([`RestoringBeam::common_enclosing`], [`RestoringBeam::deconvolving_beam`]).
30    #[error(transparent)]
31    Gaussian(#[from] EllipticalGaussianError),
32    /// The PSF plane holds no samples.
33    #[error("PSF is empty")]
34    EmptyPsf,
35    /// The PSF payload's length is not the product of its shape.
36    #[error("PSF payload of {len} samples does not match its shape {shape:?}")]
37    ShapeMismatch {
38        /// Samples in the payload.
39        len: usize,
40        /// The plane's `[width, height]`.
41        shape: [usize; 2],
42    },
43    /// The PSF peak is not positive and finite.
44    #[error("PSF peak is not positive and finite ({peak})")]
45    PeakNotPositive {
46        /// The peak sample.
47        peak: f32,
48    },
49    /// No attempt, from `cutoff` down by factors of 1.5, fitted the main
50    /// lobe (`StokesImageUtil::FitGaussianPSF`).
51    #[error(
52        "beam fit failed after {attempts} attempts starting from psfcutoff={cutoff:.3}: {last}"
53    )]
54    NotFitted {
55        /// Attempts made.
56        attempts: usize,
57        /// The first attempt's main-lobe cutoff.
58        cutoff: f32,
59        /// Why the last attempt failed.
60        last: FitAttemptFailure,
61    },
62}

64/// Why one attempt of the PSF fit, at main-lobe cutoff `cutoff`, failed.
65#[derive(Debug, Clone, Copy, PartialEq, Error)]
66pub enum FitAttemptFailure {
67    /// No main-lobe region was isolated above the cutoff.
68    #[error("failed to isolate a main-lobe region at psfcutoff={cutoff:.6}")]
69    NoMainLobe {
70        /// The attempt's cutoff.
71        cutoff: f32,
72    },
73    /// Fewer than twelve samples lay above the cutoff.
74    #[error("found only {samples} fit samples at psfcutoff={cutoff:.6}")]
75    TooFewSamples {
76        /// Samples found.
77        samples: usize,
78        /// The attempt's cutoff.
79        cutoff: f32,
80    },
81    /// The Gaussian fit did not converge.
82    #[error("fit failed to converge at psfcutoff={cutoff:.6}")]
83    NotConverged {
84        /// The attempt's cutoff.
85        cutoff: f32,
86    },
87}

96/// Heap envelope for the bounded two-pass lobe extraction, excluding its
97/// borrowed full PSF. The Gaussian solve uses fixed-size stack matrices.
98///
99/// The lobe search patch is at most `2·20 + 1` pixels per axis; the fit
100/// window is the CASA window (`fit_window`) of a whole patch around the
101/// image centre, where CASA's PSF peaks. On a non-square plane squaring by
102/// index makes that window span the gap between the two centre pixels.
103#[must_use]
104pub fn psf_fit_workspace_bytes(shape: [usize; 2]) -> u64 {
105    let first_samples = shape
106        .map(|axis| axis.min(2 * PSF_PATCH_RADIUS + 1))
107        .into_iter()
108        .product::<usize>();
109    let radius = PSF_PATCH_RADIUS
110        .min(shape[0].saturating_sub(1))
111        .min(shape[1].saturating_sub(1));
112    let centre = shape.map(|axis| axis / 2);
113    let (_, extent) = fit_window(
114        shape,
115        (
116            centre[0].saturating_sub(radius),
117            centre[1].saturating_sub(radius),
118        ),
119        (
120            (centre[0] + radius).min(shape[0].saturating_sub(1)),
121            (centre[1] + radius).min(shape[1].saturating_sub(1)),
122        ),
123    );
124    let window_cells = extent[0] * extent[1];
125    let resampled_cells = window_cells.max(TARGET_INTERPOLATED_POINTS);
126    let index_vectors = 2 * (first_samples + resampled_cells) * size_of::<SampleIndex>();
127    let image_vectors = (window_cells + resampled_cells) * size_of::<f32>();
128    let fit_samples = resampled_cells * size_of::<FitSample>();
129    (index_vectors
130        + image_vectors
131        + fit_samples
132        + 3 * size_of::<Vec<u8>>()
133        + 2 * size_of::<Array2<f32>>()) as u64
134}

140/// One fitted restoring beam in CASA metadata semantics.
141#[derive(Debug, Clone, Copy, PartialEq)]
142pub struct RestoringBeam {
143    major_fwhm_rad: f64,
144    minor_fwhm_rad: f64,
145    position_angle_rad: f64,
146}
147
148impl RestoringBeam {
149    /// Construct a finite CASA restoring beam.
150    ///
151    /// The major axis must be at least the minor axis and both widths must be
152    /// strictly positive. Position angle is wrapped to CASA's canonical
153    /// `[-pi/2, pi/2]` interval.
154    ///
155    /// # Errors
156    ///
157    /// Returns [`PsfBeamFitError::InvalidBeam`] for non-finite,
158    /// non-positive, or inverted axes.
159    pub fn new(
160        major_fwhm_rad: f64,
161        minor_fwhm_rad: f64,
162        position_angle_rad: f64,
163    ) -> Result<Self, PsfBeamFitError> {
164        if !major_fwhm_rad.is_finite()
165            || !minor_fwhm_rad.is_finite()
166            || !position_angle_rad.is_finite()
167            || major_fwhm_rad <= 0.0
168            || minor_fwhm_rad <= 0.0
169            || major_fwhm_rad < minor_fwhm_rad
170        {
171            return Err(PsfBeamFitError::InvalidBeam);
172        }
173        Ok(Self {
174            major_fwhm_rad,
175            minor_fwhm_rad,
176            position_angle_rad: casa_wrap_beam_position_angle(position_angle_rad),
177        })
178    }
179
180    /// Major full width at half maximum in radians.
181    #[must_use]
182    pub const fn major_fwhm_rad(&self) -> f64 {
183        self.major_fwhm_rad
184    }
185
186    /// Minor full width at half maximum in radians.
187    #[must_use]
188    pub const fn minor_fwhm_rad(&self) -> f64 {
189        self.minor_fwhm_rad
190    }
191
192    /// Position angle measured east from north in radians.
193    #[must_use]
194    pub const fn position_angle_rad(&self) -> f64 {
195        self.position_angle_rad
196    }
197
198    /// Return the Gaussian area in steradians.
199    #[must_use]
200    pub fn area_sr(self) -> f64 {
201        numeric_beam(self).area()
202    }
203
204    /// Compute CASA's minimum-area common enclosing restoring beam.
205    ///
206    /// # Errors
207    ///
208    /// Returns [`PsfBeamFitError::Gaussian`] when the set is empty or
209    /// contains an invalid beam.
210    pub fn common_enclosing(beams: &[Self]) -> Result<Self, PsfBeamFitError> {
211        Ok(restoring_beam(common_enclosing_gaussian(
212            &beams.iter().copied().map(numeric_beam).collect::<Vec<_>>(),
213        )?))
214    }
215
216    /// Compute the beam that convolves `source` to this target beam.
217    ///
218    /// Returns `Ok(None)` when the beams are effectively identical.
219    ///
220    /// # Errors
221    ///
222    /// Returns [`PsfBeamFitError::Gaussian`] when either beam is invalid or
223    /// the target is smaller than the source.
224    pub fn deconvolving_beam(self, source: Self) -> Result<Option<Self>, PsfBeamFitError> {
225        Ok(deconvolving_gaussian(numeric_beam(self), numeric_beam(source))?.map(restoring_beam))
226    }
227}

245/// Fit a restoring beam from one real PSF plane stored x-major.
246///
247/// The plane is indexed `[x * height + y]` like every normal-state primitive.
248/// `cutoff` is the CASA `psfcutoff` fraction of the peak that bounds the main
249/// lobe; failed fits retry at progressively lower cutoffs exactly like the
250/// reference workflow. Pixel sizes are radians per pixel on each axis.
251///
252/// # Errors
253///
254/// Returns [`PsfBeamFitError::EmptyPsf`] or
255/// [`PsfBeamFitError::PeakNotPositive`] for a PSF without a positive peak,
256/// and [`PsfBeamFitError::NotFitted`] when no attempt converges.
257pub fn fit_restoring_beam(
258    psf: &[f32],
259    shape: [usize; 2],
260    cell_size_rad: [f64; 2],
261    cutoff: f32,
262) -> Result<RestoringBeam, PsfBeamFitError> {
263    let (peak_index, peak) = positive_peak(psf)?;
264    let mut active_cutoff = cutoff.clamp(0.01, 0.99);
265    let mut last = FitAttemptFailure::NotConverged {
266        cutoff: active_cutoff,
267    };
268    for _attempt in 1..=MAX_FIT_ATTEMPTS {
269        let failure =
270            match extract_fit_samples(psf, shape, peak_index, peak, cell_size_rad, active_cutoff) {
271                None => FitAttemptFailure::NoMainLobe {
272                    cutoff: active_cutoff,
273                },
274                Some((samples, _active)) if samples.len() < 12 => {
275                    FitAttemptFailure::TooFewSamples {
276                        samples: samples.len(),
277                        cutoff: active_cutoff,
278                    }
279                }
280                Some((samples, _active)) => match fit_gaussian_beam_casa(&samples, cell_size_rad) {
281                    Some(beam) => return Ok(beam),
282                    None => FitAttemptFailure::NotConverged {
283                        cutoff: active_cutoff,
284                    },
285                },
286            };
287        last = failure;
288        active_cutoff /= 1.5;
289    }
290    Err(PsfBeamFitError::NotFitted {
291        attempts: MAX_FIT_ATTEMPTS,
292        cutoff,
293        last,
294    })
295}

297/// The flat index and value of `psf`'s peak, which must be positive and
298/// finite.
299fn positive_peak(psf: &[f32]) -> Result<(usize, f32), PsfBeamFitError> {
300    let (peak_index, peak) = peak_flat(psf).ok_or(PsfBeamFitError::EmptyPsf)?;
301    if !(peak.is_finite() && peak > 0.0) {
302        return Err(PsfBeamFitError::PeakNotPositive { peak });
303    }
304    Ok((peak_index, peak))
305}

307/// The CASA cycle-threshold sidelobe fraction of a PSF whose main lobe was
308/// fitted as `beam` in pixel units (`SIImageStore::getPSFSidelobeLevel`).
309///
310/// CASA defines the positive sidelobe as `max(psf - fitted_gaussian)` while
311/// preserving the magnitude of the most-negative original PSF sample. The
312/// Gaussian is unit peak and is scaled to the measured PSF peak before the
313/// subtraction.
314pub(crate) fn fitted_psf_sidelobe_fraction_with_beam(
315    psf: &[f32],
316    shape: [usize; 2],
317    beam: RestoringBeam,
318) -> Result<f64, PsfBeamFitError> {
319    if psf.len() != shape[0].saturating_mul(shape[1]) {
320        return Err(PsfBeamFitError::ShapeMismatch {
321            len: psf.len(),
322            shape,
323        });
324    }
325    let (peak_index, peak) = positive_peak(psf)?;
326    let peak_pixel = [peak_index / shape[1], peak_index % shape[1]];
327    let sigma_major = beam.major_fwhm_rad() / 2.354_820_045_030_949_3;
328    let sigma_minor = beam.minor_fwhm_rad() / 2.354_820_045_030_949_3;
329    let cos_pa = beam.position_angle_rad().cos();
330    let sin_pa = beam.position_angle_rad().sin();
331    // Beyond 16 times the larger sigma, exp(-r²/2) <= exp(-128)
332    // rounds to exactly zero in f32, before multiplication by the PSF peak.
333    // Keep all PSF extrema, but evaluate the Gaussian only where it can matter.
334    let radius = if sigma_minor > 0.0 {
335        (16.0 * sigma_major.max(sigma_minor)).ceil() as usize
336    } else {
337        usize::MAX
338    };
339    let bounds = std::array::from_fn::<_, 2, _>(|axis| {
340        peak_pixel[axis].saturating_sub(radius)
341            ..peak_pixel[axis]
342                .saturating_add(radius)
343                .saturating_add(1)
344                .min(shape[axis])
345    });
346    let mut minimum = 0.0_f32;
347    let mut maximum_delobed = 0.0_f32;
348    for x in 0..shape[0] {
349        let row = &psf[x * shape[1]..(x + 1) * shape[1]];
350        let gaussian_columns = if bounds[0].contains(&x) {
351            bounds[1].clone()
352        } else {
353            0..0
354        };
355        for &sample in row[..gaussian_columns.start]
356            .iter()
357            .chain(&row[gaussian_columns.end..])
358        {
359            minimum = minimum.min(sample);
360            maximum_delobed = maximum_delobed.max(sample);
361        }
362        for y in gaussian_columns {
363            let sample = row[y];
364            minimum = minimum.min(sample);
365            let dx = x as f64 - peak_pixel[0] as f64;
366            let dy = y as f64 - peak_pixel[1] as f64;
367            let u = dx * cos_pa + dy * sin_pa;
368            let v = -dx * sin_pa + dy * cos_pa;
369            let gaussian = (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2))).exp()
370                as f32
371                * peak;
372            maximum_delobed = maximum_delobed.max(sample - gaussian);
373        }
374    }
375    Ok(f64::from(minimum.abs().max(maximum_delobed)) / f64::from(peak))
376}

390fn peak_flat(psf: &[f32]) -> Option<(usize, f32)> {
391    let mut best: Option<(usize, f32)> = None;
392    for (index, value) in psf.iter().enumerate() {
393        match best {
394            None => best = Some((index, *value)),
395            Some((_, best_value)) if *value > best_value => best = Some((index, *value)),
396            _ => {}
397        }
398    }
399    best
400}

404    samples: Vec<SampleIndex>,
405    blc: (usize, usize),
406    trc: (usize, usize),

409/// The fit window's corner and extent for a lobe whose bounding box is
410/// `blc..=trc` on a plane of `shape`: CASA `FindNpoints` squares the box by
411/// index, both axes taking the smaller corner and the larger far corner
412/// (`StokesImageUtil.cc`), and `FitGaussianPSF` expands it by
413/// `expand_pixel` and clips each axis to the plane, so a non-square plane
414/// can leave a non-square window.
415fn fit_window(
416    shape: [usize; 2],
417    blc: (usize, usize),
418    trc: (usize, usize),
419) -> ([usize; 2], [usize; 2]) {
420    let low = blc.0.min(blc.1).saturating_sub(REGION_PADDING);
421    let high = trc.0.max(trc.1) + REGION_PADDING;
422    let last = shape.map(|axis| high.min(axis.saturating_sub(1)));
423    ([low, low], last.map(|last| last + 1 - low))
424}

426/// Faithful port of `StokesImageUtil::extractCasaFitSamples`.
427///
428/// `cutoff` is the CASA `psfcutoff` fraction of the ACTUAL PSF peak: the
429/// reference workflow normalizes the local PSF by its fitted peak
430/// (`lpsf /= bamp`) before walking the main lobe, so the cutoff always
431/// compares against `cutoff * peak`, never a bare sample value.
432fn extract_fit_samples(
433    psf: &[f32],
434    shape: [usize; 2],
435    peak_index: usize,
436    peak: f32,
437    cell_size_rad: [f64; 2],
438    cutoff: f32,
439) -> Option<(Vec<FitSample>, f32)> {
440    let deltas = [cell_size_rad[0].abs(), cell_size_rad[1].abs()];
441    let nrow = PSF_PATCH_RADIUS
442        .min(shape[0].saturating_sub(1))
443        .min(shape[1].saturating_sub(1));
444    let first_pass = find_points_in_lobe(psf, shape, peak_index, nrow, cutoff * peak)?;
445
446    let ([blc_x, blc_y], [nx, ny]) = fit_window(shape, first_pass.blc, first_pass.trc);
447    let mut window = Array2::<f32>::zeros((nx, ny));
448    for x in 0..nx {
449        for y in 0..ny {
450            window[(x, y)] = psf[(blc_x + x) * shape[1] + blc_y + y] / peak;
451        }
452    }
453
454    let mut oversampling = TARGET_INTERPOLATED_POINTS / (nx * ny);
455    oversampling = (oversampling as f64).sqrt() as usize;
456    if oversampling == 0 {
457        oversampling = 1;
458    }
459    let mut resampled = resample_psf(&window, oversampling);
460    let (_, max_value) = peak_flat(resampled.as_slice()?)?;
461    if !(max_value.is_finite() && max_value > 0.0) {
462        return None;
463    }
464    resampled.mapv_inplace(|value| value / max_value);
465    // The peak is located again after the Float division, as CASA does.
466    let (peak, _) = peak_flat(resampled.as_slice()?)?;
467    let resampled_peak = (peak / resampled.shape()[1], peak % resampled.shape()[1]);
468    let min_len = nx.min(ny);
469    let nrow_re = (oversampling * min_len).saturating_sub(1) / 2;
470    let resampled_deltas = [
471        deltas[0] / oversampling as f64,
472        deltas[1] / oversampling as f64,
473    ];
474    let resampled_shape = [resampled.shape()[0], resampled.shape()[1]];
475    let second_pass = find_points_in_lobe(
476        resampled.as_slice()?,
477        resampled_shape,
478        resampled_peak.0 * resampled_shape[1] + resampled_peak.1,
479        nrow_re,
480        cutoff,
481    )?;
482    let samples = second_pass
483        .samples
484        .iter()
485        .map(|sample| FitSample {
486            x_rad: (sample.x_index as isize - resampled_peak.0 as isize) as f64
487                * resampled_deltas[0],
488            y_rad: (sample.y_index as isize - resampled_peak.1 as isize) as f64
489                * resampled_deltas[1],
490            value: resampled[[sample.x_index, sample.y_index]] as f64,
491        })
492        .collect::<Vec<_>>();
493    Some((samples, cutoff))
494}

496/// Port of casacore's `findPointsInLobe` main-lobe sample walk: spiral out
497/// from the peak through quadrant flips and stop each ray below the cutoff.
498fn find_points_in_lobe(
499    psf: &[f32],
500    shape: [usize; 2],
501    peak_index: usize,
502    nrow: usize,
503    amin: f32,
504) -> Option<LobeSearchResult> {
505    let [width, height] = shape;
506    let px = (peak_index / height) as isize;
507    let py = (peak_index % height) as isize;
508    let maxnpoints = (2 * nrow + 1) * (2 * nrow + 1);
509    let mut samples = Vec::new();
510    let mut blc = (width - 1, height - 1);
511    let mut trc = (0_usize, 0_usize);
512    let mut iflip = 1_isize;
513    let mut jflip = 1_isize;
514
515    'search: for jlo in 0..2 {
516        jflip *= -1;
517        for j in jlo..=nrow {
518            let jrow = py + j as isize * jflip;
519            for ilo in 0..2 {
520                iflip *= -1;
521                if jrow > height as isize - 1 || jrow < 0 {
522                    break;
523                }
524                let mut inlobe = psf[(px as usize) * height + jrow as usize] > amin;
525                for i in ilo..=nrow {
526                    if samples.len() >= maxnpoints {
527                        break 'search;
528                    }
529                    let irow = px + i as isize * iflip;
530                    if irow > width as isize - 1 || irow < 0 {
531                        break;
532                    }
533                    let value = psf[irow as usize * height + jrow as usize];
534                    if inlobe && value < amin {
535                        break;
536                    }
537                    if value > amin {
538                        inlobe = true;
539                        let ux = irow as usize;
540                        let uy = jrow as usize;
541                        blc.0 = blc.0.min(ux);
542                        blc.1 = blc.1.min(uy);
543                        trc.0 = trc.0.max(ux);
544                        trc.1 = trc.1.max(uy);
545                        samples.push(SampleIndex {
546                            x_index: ux,
547                            y_index: uy,
548                        });
549                    }
550                }
551            }
552        }
553    }
554
555    if samples.is_empty() {
556        return None;
557    }
558    Some(LobeSearchResult { samples, blc, trc })
559}

634fn casa_interp_cubic(data: &Array2<f32>, x: f64, y: f64) -> f64 {
635    let nx = data.shape()[0] as isize;
636    let ny = data.shape()[1] as isize;
637    let i = x as isize;
638    let j = y as isize;
639    if i <= 0 || i >= nx - 2 || j <= 0 || j >= ny - 2 {
640        return casa_interp_linear(data, x, y);
641    }
642
643    let tt = x - i as f64;
644    let uu = y - j as f64;
645
646    let mut vals = [0.0f64; 4];
647    let mut dx_vals = [0.0f64; 4];
648    let mut dy_vals = [0.0f64; 4];
649    let mut dxy_vals = [0.0f64; 4];
650
651    vals[0] = data[(i as usize, j as usize)] as f64;
652    vals[1] = data[((i + 1) as usize, j as usize)] as f64;
653    vals[2] = data[((i + 1) as usize, (j + 1) as usize)] as f64;
654    vals[3] = data[(i as usize, (j + 1) as usize)] as f64;
655
656    // Interpolate2D<T=Float> evaluates these source-pixel expressions in
657    // Float before assigning them to its Double derivative arrays.
658    dx_vals[0] =
659        f64::from(data[((i + 1) as usize, j as usize)] - data[((i - 1) as usize, j as usize)]);
660    dx_vals[1] = f64::from(data[((i + 2) as usize, j as usize)] - data[(i as usize, j as usize)]);
661    dx_vals[2] = f64::from(
662        data[((i + 2) as usize, (j + 1) as usize)] - data[(i as usize, (j + 1) as usize)],
663    );
664    dx_vals[3] = f64::from(
665        data[((i + 1) as usize, (j + 1) as usize)] - data[((i - 1) as usize, (j + 1) as usize)],
666    );
667
668    dy_vals[0] =
669        f64::from(data[(i as usize, (j + 1) as usize)] - data[(i as usize, (j - 1) as usize)]);
670    dy_vals[1] = f64::from(
671        data[((i + 1) as usize, (j + 1) as usize)] - data[((i + 1) as usize, (j - 1) as usize)],
672    );
673    dy_vals[2] = f64::from(
674        data[((i + 1) as usize, (j + 2) as usize)] - data[((i + 1) as usize, j as usize)],
675    );
676    dy_vals[3] = f64::from(data[(i as usize, (j + 2) as usize)] - data[(i as usize, j as usize)]);
677
678    dxy_vals[0] = f64::from(
679        data[((i + 1) as usize, (j + 1) as usize)] + data[((i - 1) as usize, (j - 1) as usize)]
680            - data[((i - 1) as usize, (j + 1) as usize)]
681            - data[((i + 1) as usize, (j - 1) as usize)],
682    );
683    dxy_vals[1] = f64::from(
684        data[((i + 2) as usize, (j + 1) as usize)] + data[(i as usize, (j - 1) as usize)]
685            - data[(i as usize, (j + 1) as usize)]
686            - data[((i + 2) as usize, (j - 1) as usize)],
687    );
688    dxy_vals[2] = f64::from(
689        data[((i + 2) as usize, (j + 2) as usize)] + data[(i as usize, j as usize)]
690            - data[(i as usize, (j + 2) as usize)]
691            - data[((i + 2) as usize, j as usize)],
692    );
693    dxy_vals[3] = f64::from(
694        data[((i + 1) as usize, (j + 2) as usize)] + data[((i - 1) as usize, j as usize)]
695            - data[((i - 1) as usize, (j + 2) as usize)]
696            - data[((i + 1) as usize, j as usize)],
697    );
698
699    for axis in 0..4 {
700        dx_vals[axis] /= 2.0;
701        dy_vals[axis] /= 2.0;
702        dxy_vals[axis] /= 4.0;
703    }
704
705    let coeffs = casa_bcucof(vals, dx_vals, dy_vals, dxy_vals);
706    let mut result = 0.0f32;
707    for row in (0..4).rev() {
708        result = (tt * f64::from(result)
709            + ((coeffs[row][3] * uu + coeffs[row][2]) * uu + coeffs[row][1]) * uu
710            + coeffs[row][0]) as f32;
711    }
712    f64::from(result)
713}

715fn casa_interp_linear(data: &Array2<f32>, x: f64, y: f64) -> f64 {
716    let nx = data.shape()[0];
717    let ny = data.shape()[1];
718    let mut i = x as usize;
719    let mut j = y as usize;
720    let si = nx - 1;
721    let sj = ny - 1;
722    if i == si {
723        i -= 1;
724    }
725    if j == sj {
726        j -= 1;
727    }
728    let tt = x - i as f64;
729    let uu = y - j as f64;
730    (1.0 - tt) * (1.0 - uu) * f64::from(data[(i, j)])
731        + tt * (1.0 - uu) * f64::from(data[(i + 1, j)])
732        + tt * uu * f64::from(data[(i + 1, j + 1)])
733        + (1.0 - tt) * uu * f64::from(data[(i, j + 1)])
734}

736fn resample_psf(psf: &Array2<f32>, oversampling: usize) -> Array2<f32> {
737    let nx = psf.shape()[0];
738    let ny = psf.shape()[1];
739    let nx_re = nx * oversampling - oversampling + 1;
740    let ny_re = ny * oversampling - oversampling + 1;
741    let mut resampled = Array2::<f32>::zeros((nx_re, ny_re));
742    for i in 0..nx_re {
743        for j in 0..ny_re {
744            // StokesImageUtil::ResamplePSF performs both operands of this
745            // division in Float before assigning the result to its
746            // Vector<Double> interpolation position.
747            let x = f64::from(i as f32 / oversampling as f32);
748            let y = f64::from(j as f32 / oversampling as f32);
749            resampled[(i, j)] = casa_interp_cubic(psf, x, y) as f32;
750        }
751    }
752    resampled
753}

762fn fit_gaussian_beam_casa(samples: &[FitSample], cell_size_rad: [f64; 2]) -> Option<RestoringBeam> {
763    // StokesImageUtil::FitGaussianPSF seeds the fit with a 2.5-pixel-wide
764    // Gaussian at axial ratio 0.5 and position angle zero.
765    let accepted_params = CasaParams {
766        width_fwhm_rad: 2.5 * cell_size_rad[0].abs().max(cell_size_rad[1].abs()),
767        axial_ratio: 0.5,
768        position_angle_rad: 0.0,
769    };
770    let mut accepted_params = accepted_params;
771    let (mut accepted_normal, mut accepted_known) =
772        casa_normal_equations(samples, accepted_params)?;
773    accepted_known.iter_mut().for_each(|value| *value = -*value);
774    let mut accepted_cost = casa_gaussian_cost(samples, accepted_params)?;
775    let mut lambda = 1.0e-3;
776    let mut step_factor = 2.0;
777    let mut candidate_params = accepted_params;
778    let mut candidate_delta = [0.0; 3];
779    let mut first_iteration = true;
780
781    for _ in 0..1_000 {
782        if !first_iteration {
783            let Some(candidate_cost) = casa_gaussian_cost(samples, candidate_params) else {
784                // The step left the representable parameter region; keep the
785                // best accepted candidate, like casacore's best-effort return.
786                break;
787            };
788            let expected_reduction = candidate_delta
789                .iter()
790                .enumerate()
791                .map(|(axis, delta)| {
792                    delta * (lambda * delta * accepted_normal[axis][axis] + accepted_known[axis])
793                })
794                .sum::<f64>()
795                * 0.5;
796            let actual_reduction = (accepted_cost - candidate_cost) * 0.5;
797            if expected_reduction > 0.0 && actual_reduction > 0.0 {
798                accepted_params = candidate_params;
799                accepted_cost = candidate_cost;
800                let Some((normal, known)) = casa_normal_equations(samples, accepted_params) else {
801                    break;
802                };
803                accepted_normal = normal;
804                accepted_known = known;
805                accepted_known.iter_mut().for_each(|value| *value = -*value);
806                lambda *= 0.3;
807                step_factor = 2.0;
808                if accepted_known.iter().all(|value| value.abs() <= 1.0e-8) {
809                    break;
810                }
811            } else {
812                lambda *= step_factor;
813                step_factor *= 2.0;
814                if step_factor > 1.0e10 {
815                    break;
816                }
817            }
818        }
819
820        let mut damped_normal = accepted_normal;
821        for (axis, row) in damped_normal.iter_mut().enumerate() {
822            row[axis] *= 1.0 + lambda;
823        }
824        let Some(solved_delta) = solve_symmetric_ldlt_casacore(damped_normal, accepted_known)
825        else {
826            break;
827        };
828        candidate_delta = solved_delta;
829        candidate_params = CasaParams {
830            width_fwhm_rad: accepted_params.width_fwhm_rad + candidate_delta[0],
831            axial_ratio: accepted_params.axial_ratio + candidate_delta[1],
832            position_angle_rad: accepted_params.position_angle_rad + candidate_delta[2],
833        };
834        first_iteration = false;
835
836        let delta_norm = candidate_delta.iter().map(|v| v * v).sum::<f64>().sqrt();
837        let parameter_norm = [
838            accepted_params.width_fwhm_rad,
839            accepted_params.axial_ratio,
840            accepted_params.position_angle_rad,
841        ]
842        .iter()
843        .map(|value| value * value)
844        .sum::<f64>()
845        .sqrt();
846        if delta_norm <= 1.0e-8 * (parameter_norm + 1.0e-8) {
847            break;
848        }
849    }
850
851    // NonLinearFitLM::fitIt performs one final undamped Gauss-Newton solve
852    // from the last candidate before returning the solution.
853    let final_params = match casa_normal_equations(samples, candidate_params) {
854        Some((final_normal, mut final_known)) => {
855            final_known.iter_mut().for_each(|value| *value = -*value);
856            match solve_symmetric_ldlt_casacore(final_normal, final_known) {
857                Some(final_delta) => CasaParams {
858                    width_fwhm_rad: candidate_params.width_fwhm_rad + final_delta[0],
859                    axial_ratio: candidate_params.axial_ratio + final_delta[1],
860                    position_angle_rad: candidate_params.position_angle_rad + final_delta[2],
861                },
862                None => candidate_params,
863            }
864        }
865        None => candidate_params,
866    };
867    casa_params_to_beam(final_params)
868}

870fn casa_normal_equations(
871    samples: &[FitSample],
872    params: CasaParams,
873) -> Option<([[f64; 3]; 3], [f64; 3])> {
874    let mut normal = [[0.0f64; 3]; 3];
875    let mut gradient = [0.0f64; 3];
876    for sample in samples {
877        let (model, jacobian) =
878            casa_gaussian_value_and_jacobian(params, sample.x_rad, sample.y_rad)?;
879        let residual = model - sample.value;
880        for row in 0..3 {
881            gradient[row] += jacobian[row] * residual;
882            for col in row..3 {
883                normal[row][col] += jacobian[row] * jacobian[col];
884            }
885        }
886    }
887    #[allow(clippy::needless_range_loop)]
888    for row in 0..3 {
889        for col in 0..row {
890            normal[row][col] = normal[col][row];
891        }
892    }
893    Some((normal, gradient))
894}

896fn casa_gaussian_value_and_jacobian(
897    params: CasaParams,
898    x_rad: f64,
899    y_rad: f64,
900) -> Option<(f64, [f64; 3])> {
901    let width = params.width_fwhm_rad;
902    let ratio = params.axial_ratio;
903    if !(width.is_finite()
904        && width.abs() > MIN_SIGMA_RAD
905        && ratio.is_finite()
906        && ratio.abs() > 1.0e-6
907        && params.position_angle_rad.is_finite())
908    {
909        return None;
910    }
911    let cos_pa = params.position_angle_rad.cos();
912    let sin_pa = params.position_angle_rad.sin();
913    let xnorm = x_rad * cos_pa + y_rad * sin_pa;
914    let ynorm = -x_rad * sin_pa + y_rad * cos_pa;
915    let xnorm2 = xnorm * xnorm;
916    let ynorm2 = ynorm * ynorm;
917    let xwidth = width * ratio;
918    let xwidth2 = xwidth * xwidth * CASA_FWHM_TO_INTERNAL * CASA_FWHM_TO_INTERNAL;
919    let ywidth2 = width * width * CASA_FWHM_TO_INTERNAL * CASA_FWHM_TO_INTERNAL;
920    let x2w = 2.0 * xnorm / xwidth2;
921    let y2w = 2.0 * ynorm / ywidth2;
922    let x2w2 = x2w * xnorm;
923    let y2w2 = y2w * ynorm;
924    let model = (-(xnorm2 / xwidth2 + ynorm2 / ywidth2)).exp();
925    let dev = model;
926    let jacobian = [
927        dev * ((x2w2 + y2w2) / width),
928        dev * x2w2 * width / xwidth,
929        -dev * (x2w * (-x_rad * sin_pa + y_rad * cos_pa)
930            + y2w * (-x_rad * cos_pa - y_rad * sin_pa)),
931    ];
932    (model.is_finite() && jacobian.iter().all(|value| value.is_finite()))
933        .then_some((model, jacobian))
934}

936fn casa_gaussian_cost(samples: &[FitSample], params: CasaParams) -> Option<f64> {
937    if !(params.width_fwhm_rad.is_finite()
938        && params.width_fwhm_rad.abs() > MIN_SIGMA_RAD
939        && params.axial_ratio.is_finite()
940        && params.position_angle_rad.is_finite())
941    {
942        return None;
943    }
944    let mut cost = 0.0_f64;
945    let mut evaluated = true;
946    for sample in samples {
947        match casa_gaussian_value_and_jacobian(params, sample.x_rad, sample.y_rad) {
948            Some((model, _)) => {
949                let residual = model - sample.value;
950                cost += residual * residual;
951            }
952            None => evaluated = false,
953        }
954    }
955    (evaluated && cost.is_finite()).then_some(cost)
956}

958fn casa_params_to_beam(params: CasaParams) -> Option<RestoringBeam> {
959    if !(params.width_fwhm_rad.is_finite()
960        && params.axial_ratio.is_finite()
961        && params.position_angle_rad.is_finite())
962    {
963        return None;
964    }
965    let (major, minor, pa) = if params.axial_ratio.abs() > 1.0 {
966        (
967            (params.width_fwhm_rad * params.axial_ratio).abs(),
968            params.width_fwhm_rad.abs(),
969            params.position_angle_rad - std::f64::consts::FRAC_PI_2,
970        )
971    } else {
972        (
973            params.width_fwhm_rad.abs(),
974            (params.width_fwhm_rad * params.axial_ratio).abs(),
975            params.position_angle_rad,
976        )
977    };
978    let mut major = major;
979    let mut minor = minor;
980    let mut pa = casa_wrap_beam_position_angle(pa);
981    if minor > major {
982        std::mem::swap(&mut major, &mut minor);
983        pa = casa_wrap_beam_position_angle(pa + std::f64::consts::FRAC_PI_2);
984    }
985    // StokesImageUtil::FitGaussianPSF solves in Double, then assigns the
986    // fitted arcsec/arcsec/degree values to Vector<Float> before constructing
987    // GaussianBeam. Preserve that rounding so metadata matches CASA's actual
988    // published beam.
989    let radians_to_degrees = 180.0 / std::f64::consts::PI;
990    let radians_to_arcseconds = 3_600.0 * radians_to_degrees;
991    Some(RestoringBeam {
992        major_fwhm_rad: f64::from((major * radians_to_arcseconds) as f32) / radians_to_arcseconds,
993        minor_fwhm_rad: f64::from((minor * radians_to_arcseconds) as f32) / radians_to_arcseconds,
994        position_angle_rad: f64::from((pa * radians_to_degrees) as f32) / radians_to_degrees,
995    })
996}

998fn casa_wrap_beam_position_angle(mut angle: f64) -> f64 {
999    let pi = std::f64::consts::PI;
1000    let half_pi = std::f64::consts::FRAC_PI_2;
1001    let three_half_pi = 3.0 * half_pi;
1002    while (angle / half_pi).abs() > 1.0 {
1003        if angle > three_half_pi {
1004            angle -= 2.0 * pi;
1005        } else if angle > half_pi {
1006            angle -= pi;
1007        } else if angle < -three_half_pi {
1008            angle += 2.0 * pi;
1009        } else {
1010            angle += pi;
1011        }
1012    }
1013    angle
1014}

1139    fn full_plane_sidelobe_reference(psf: &[f32], shape: [usize; 2], beam: RestoringBeam) -> f64 {
1140        let (peak_index, peak) = peak_flat(psf).unwrap();
1141        let centre = [peak_index / shape[1], peak_index % shape[1]];
1142        let sigma_major = beam.major_fwhm_rad() / 2.354_820_045_030_949_3;
1143        let sigma_minor = beam.minor_fwhm_rad() / 2.354_820_045_030_949_3;
1144        let cos_pa = beam.position_angle_rad().cos();
1145        let sin_pa = beam.position_angle_rad().sin();
1146        let mut minimum = 0.0_f32;
1147        let mut maximum_delobed = 0.0_f32;
1148        for x in 0..shape[0] {
1149            for y in 0..shape[1] {
1150                let sample = psf[x * shape[1] + y];
1151                minimum = minimum.min(sample);
1152                let dx = x as f64 - centre[0] as f64;
1153                let dy = y as f64 - centre[1] as f64;
1154                let u = dx * cos_pa + dy * sin_pa;
1155                let v = -dx * sin_pa + dy * cos_pa;
1156                let gaussian = (-0.5 * ((u / sigma_minor).powi(2) + (v / sigma_major).powi(2)))
1157                    .exp() as f32
1158                    * peak;
1159                maximum_delobed = maximum_delobed.max(sample - gaussian);
1160            }
1161        }
1162        f64::from(minimum.abs().max(maximum_delobed)) / f64::from(peak)
1163    }
