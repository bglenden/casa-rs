10fn quantity_vector(values: Vec<f64>, unit: &str) -> CasaValue {
11    CasaValue::record([
12        ("unit", CasaValue::Text(unit.into())),
13        (
14            "value",
15            CasaValue::Floats {
16                shape: vec![values.len()],
17                values,
18            },
19        ),
20    ])
21}
22fn failed_tuple(lines: &mut Vec<LogLine>, message: impl Into<String>, n: usize) -> CasaValue {
23    lines.push(LogLine::error(message));
24    CasaValue::Tuple(vec![CasaValue::empty_record(); n])
25}
26fn cartesian(value: Measure) -> Result<[f64; 3], String> {
27    match value.value {
28        MeasureValue::Baseline(xyz) | MeasureValue::Uvw(xyz) | MeasureValue::Position(xyz) => {
29            Ok(xyz)
30        }
31        _ => Err("Trying to convert non-baseline to uvw\n".into()),
32    }
33}

35/// Expand each ordered `i < j` pair to `position[j] - position[i]`, retaining metadata.
36/// CASA forces the input to uvw for decoding and restores its original labels.
37pub(super) fn expand(input: &RecordArg, lines: &mut Vec<LogLine>) -> CasaValue {
38    let mut fields = input.0.clone();
39    let labels = match (fields.get("type"), fields.get("refer")) {
40        (Some(Variant::Text(kind)), Some(Variant::Text(refer))) => (kind.clone(), refer.clone()),
41        _ => {
42            return failed_tuple(
43                lines,
44                "Non-measure input to expand()\nCan only expand baselines, positions, or uvw\n",
45                2,
46            );
47        }
48    };
49    if !matches!(
50        labels.0.to_ascii_lowercase().as_str(),
51        "baseline" | "position" | "uvw"
52    ) {
53        return failed_tuple(lines, "Can only expand baselines, positions, or uvw", 2);
54    }
55    fields.insert("type".into(), Variant::Text("uvw".into()));
56    fields.insert("refer".into(), Variant::Text("J2000".into()));
57    let source = match Measures::read(&RecordArg(fields)) {
58        Ok(source) => source,
59        Err(error) => return failed_tuple(lines, error.message(), 2),
60    };
61    let positions: Vec<_> = source
62        .values
63        .iter()
64        .map(|&v| cartesian(v).expect("decoded uvw"))
65        .collect();
66    let mut vectors = Vec::new();
67    if positions.len() < 2 {
68        vectors.push(positions[0]);
69    } else {
70        for (i, first) in positions.iter().enumerate() {
71            for second in &positions[i + 1..] {
72                vectors.push(std::array::from_fn(|axis| second[axis] - first[axis]));
73            }
74        }
75    }
76    let mut result = Measures::scalar(Measure {
77        refer: "J2000",
78        value: MeasureValue::Uvw(vectors[0]),
79    });
80    result.vector = source.values.len() > 1;
81    result.values = vectors
82        .iter()
83        .map(|&v| Measure {
84            refer: "J2000",
85            value: MeasureValue::Uvw(v),
86        })
87        .collect();
88    let mut record = result.to_value();
89    if let CasaValue::Record(fields) = &mut record {
90        for (key, value) in fields {
91            match key.as_str() {
92                "type" => *value = CasaValue::Text(labels.0.clone()),
93                "refer" => *value = CasaValue::Text(labels.1.clone()),
94                _ => {}
95            }
96        }
97    }
98    CasaValue::Tuple(vec![
99        record,
100        quantity_vector(vectors.into_iter().flatten().collect(), "m"),
101    ])
102}

104/// Project absolute J2000 baselines on the explicit frame direction, and
105/// provide casacore's sidereal-rate uvw derivative in flattened XYZ order.
106pub(super) fn touvw(
107    state: &mut State,
108    input: &RecordArg,
109    lines: &mut Vec<LogLine>,
110) -> PyResult<CasaValue> {
111    let source = match Measures::read(input) {
112        Ok(source) if source.values[0].kind() == MeasureKind::Baseline => source,
113        Ok(_) => {
114            return Ok(failed_tuple(
115                lines,
116                "Trying to convert non-baseline to uvw\n",
117                3,
118            ));
119        }
120        Err(error) => return Ok(failed_tuple(lines, error.message(), 3)),
121    };
122    let source = match source.resolve(state, "J2000")? {
123        Ok(source) => source,
124        Err(error) => return Ok(failed_tuple(lines, error, 3)),
125    };
126    let Some(direction) = state.frame.direction.as_ref() else {
127        return Ok(failed_tuple(
128            lines,
129            "No direction in frame for uvw calculation\n",
130            3,
131        ));
132    };
133    let direction = direction.absolute;
134    let result = super::frame::convert(state, &direction, "J2000")?;
135    let direction = match result {
136        Ok(direction) => direction,
137        Err(error) => return Ok(failed_tuple(lines, error, 3)),
138    };
139    let crate::convert::holder::CasaMeasure::Direction(direction) = direction
140        .to_casa()
141        .map_err(pyo3::exceptions::PyRuntimeError::new_err)?
142    else {
143        unreachable!("frame direction")
144    };
145    let dec = direction.latitude_rad();
146    let (sd, cd) = dec.sin_cos();
147    let rate = std::f64::consts::PI / 180.0 / 240.0 / 1.002_737_909_350_795;
148    let mut xyz = Vec::with_capacity(3 * source.values.len());
149    let mut dot = Vec::with_capacity(xyz.capacity());
150    let result = source
151        .map(|value| {
152            let vector =
153                casa_core::measures::baseline::j2000_baseline_uvw(cartesian(value)?, &direction);
154            xyz.extend(vector);
155            dot.extend([
156                (-sd * vector[1] + cd * vector[2]) * rate,
157                sd * vector[0] * rate,
158                -cd * vector[0] * rate,
159            ]);
160            Ok(Measure {
161                refer: "J2000",
162                value: MeasureValue::Uvw(vector),
163            })
164        })
165        .expect("baseline kind checked");
166    Ok(CasaValue::Tuple(vec![
167        result.to_value(),
168        quantity_vector(xyz, "m"),
169        quantity_vector(dot, "m/s"),
170    ]))
171}

178enum Operator {
179    Matrix([[f64; 3]; 3]),
180    Magnetic(Box<MeasFrame>, EarthMagneticRef),
181}
182/// One operator for every element of a measure vector, compiled once per call.
183pub(super) struct Conversion {
184    operator: Operator,
185    refer: &'static str,
186}
187impl Conversion {
188    pub(super) fn new(
189        state: &mut State,
190        input: Measure,
191        target: &'static str,
192    ) -> PyResult<Result<Self, String>> {
193        let frame = match super::frame::native_frame(state)? {
194            Ok(frame) => super::frame::with_resolved_direction(state, frame),
195            Err(error) => return Ok(Err(error)),
196        };
197        Ok((|| {
198            let operator = if input.kind() == MeasureKind::EarthMagnetic {
199                Operator::Magnetic(
200                    Box::new(frame),
201                    target
202                        .parse()
203                        .map_err(|e: casa_core::measures::MeasureError| e.to_string())?,
204                )
205            } else {
206                let from = input
207                    .refer
208                    .parse::<DirectionRef>()
209                    .map_err(|e| e.to_string())?;
210                let to = target.parse::<DirectionRef>().map_err(|e| e.to_string())?;
211                let matrix = baseline_rotation(from, to, &frame, input.kind() == MeasureKind::Uvw)
212                    .map_err(|error| error.to_string())?;
213                Operator::Matrix(matrix)
214            };
215            Ok(Self {
216                operator,
217                refer: target,
218            })
219        })())
220    }
221    pub(super) fn apply(&self, input: Measure) -> Result<Measure, String> {
222        let value = match (&self.operator, input.value) {
223            (Operator::Matrix(m), MeasureValue::Baseline(v)) => {
224                MeasureValue::Baseline(rotate(*m, v))
225            }
226            (Operator::Matrix(m), MeasureValue::Uvw(v)) => MeasureValue::Uvw(rotate(*m, v)),
227            (Operator::Magnetic(frame, target), MeasureValue::EarthMagnetic(v)) => {
228                let from = input
229                    .refer
230                    .parse()
231                    .map_err(|e: casa_core::measures::MeasureError| e.to_string())?;
232                let value = MEarthMagnetic::from_xyz_nt(v[0], v[1], v[2], from)
233                    .convert_to(*target, frame)
234                    .map_err(|e| e.to_string())?;
235                MeasureValue::EarthMagnetic(value.xyz_nt())
236            }
237            _ => return Err("Non-conforming geometry operator".into()),
238        };
239        Ok(Measure {
240            refer: self.refer,
241            value,
242        })
243    }
244}
245pub(super) fn convert(
246    state: &mut State,
247    input: &Measure,
248    target: &'static str,
249) -> PyResult<Result<Measure, String>> {
250    Ok(match Conversion::new(state, *input, target)? {
251        Ok(plan) => plan.apply(*input),
252        Err(error) => Err(error),
253    })
254}
255fn rotate(m: [[f64; 3]; 3], v: [f64; 3]) -> [f64; 3] {
256    std::array::from_fn(|i| (0..3).map(|j| m[i][j] * v[j]).sum())
257}

265pub(super) fn define_three(
266    state: &mut State,
267    kind: MeasureKind,
268    rf: &str,
269    args: [&Variant; 3],
270    off: &RecordArg,
271    lines: &mut Vec<LogLine>,
272) -> PyResult<CasaValue> {
273    let array = quantum_holder(args[0])
274        .ok()
275        .flatten()
276        .filter(|q| q.is_array() && !q.values.is_empty());
277    let mut triples = Vec::new();
278    if kind != MeasureKind::EarthMagnetic
279        && let Some(first) = array
280    {
281        let wrong_length = if kind == MeasureKind::Position {
282            "Exception Reports: The length of the 3 quantities are not conformant \n"
283        } else {
284            "The length of the 3 quantities are not conformant \n"
285        };
286        let mut quantities = vec![first];
287        for arg in &args[1..] {
288            let Some(q) = quantum_holder(arg).ok().flatten() else {
289                return Ok(failure(lines, wrong_length));
290            };
291            if q.values.len() != quantities[0].values.len() {
292                return Ok(failure(lines, wrong_length));
293            }
294            quantities.push(q);
295        }
296        for i in 0..quantities[0].values.len() {
297            triples.push(
298                quantities
299                    .iter()
300                    .map(|q| Quantity::with_unit(q.values[i], q.unit.clone()))
301                    .collect::<Vec<_>>(),
302            );
303        }
304    } else {
305        let Some(mut q) = args
306            .iter()
307            .map(|v| super::define::quantity_of(v, lines))
308            .collect::<Option<Vec<_>>>()
309        else {
310            return Ok(CasaValue::empty_record());
311        };
312        if q.iter()
313            .all(|q| q.value() == 0. && q.unit().name().is_empty())
314        {
315            q = if kind == MeasureKind::EarthMagnetic {
316                vec![
317                    Quantity::new(0., "G").expect("G"),
318                    Quantity::new(0., "deg").expect("deg"),
319                    Quantity::new(90., "deg").expect("deg"),
320                ]
321            } else {
322                vec![
323                    Quantity::new(0., "deg").expect("deg"),
324                    Quantity::new(0., "deg").expect("deg"),
325                    Quantity::new(0., "m").expect("m"),
326                ]
327            };
328        }
329        triples.push(q);
330    }
331    if kind == MeasureKind::EarthMagnetic {
332        use casa_core::quanta::{Dimension, UnitDim};
333        let angle = UnitDim::basis(Dimension::Angle);
334        if triples[0][1..].iter().any(|q| q.unit().val().dim != angle) {
335            return Ok(failure(
336                lines,
337                "earthmagnetic requires a field strength and two angles",
338            ));
339        }
340    }
341    let refer = if kind == MeasureKind::Position {
342        super::define::position_reference(rf, lines)
343    } else {
344        kind.codes().resolve(rf).unwrap_or_else(|| {
345            lines.push(LogLine::warning("Illegal reference frame string."));
346            kind.codes().default_code()
347        })
348    };
349    let values = match triples
350        .iter()
351        .map(|q| Measure::from_values(kind, refer, q))
352        .collect::<Result<Vec<_>, _>>()
353    {
354        Ok(values) => values,
355        Err(error) => return Ok(failure(lines, error.message())),
356    };
357    let mut result = Measures::scalar(values[0]);
358    result.vector = values.len() > 1;
359    result.values = values;
360    if !off.0.is_empty() {
361        match Measures::read(off) {
362            Ok(origin) if origin.values[0].kind() == kind => {
363                if kind == MeasureKind::EarthMagnetic {
364                    result.offset = Some(Box::new(origin.into_reference_origin()));
365                } else {
366                    result = match result.relative_to(origin, state)? {
367                        Ok(value) => value,
368                        Err(error) => return Ok(failure(lines, error)),
369                    };
370                }
371            }
372            Err(_) if kind == MeasureKind::Position => {}
373            Err(_) => lines.push(LogLine::warning(
374                "Illegal offset type specified, not used\n",
375            )),
376            _ if kind == MeasureKind::Baseline => lines.push(LogLine::warning(
377                "Illegal offset type specified, not used\n",
378            )),
379            _ => return Ok(failure(lines, "Non-conforming offset measure type")),
380        }
381    }
382    Ok(result.to_value())
383}
384fn asbaseline(
385    state: &mut State,
386    input: &RecordArg,
387    lines: &mut Vec<LogLine>,
388) -> PyResult<CasaValue> {
389    let source = match Measures::read(input) {
390        Ok(source) => source,
391        Err(error) => return Ok(failure(lines, error.message())),
392    };
393    if source.values[0].kind() == MeasureKind::Baseline {
394        return Ok(source.to_value());
395    }
396    if source.values[0].kind() != MeasureKind::Position {
397        return Ok(failure(lines, "Non-position type for asbaseline input"));
398    }
399    let source = match source.resolve(state, "ITRF")? {
400        Ok(source) => source,
401        Err(error) => return Ok(failure(lines, error)),
402    };
403    let source = source
404        .map(|value| {
405            Ok(Measure {
406                refer: "ITRF",
407                value: MeasureValue::Baseline(cartesian(value)?),
408            })
409        })
410        .expect("position kind");
411    match source.resolve(state, "J2000")? {
412        Ok(value) => Ok(value.to_value()),
413        Err(error) => Ok(failure(lines, error)),
414    }
415}

419#[pymethods]
420impl PyMeasures {
421    /// Define one or more baseline vectors, with an optional reference origin.
422    #[pyo3(signature=(rf=None,v0=Variant::Empty,v1=Variant::Empty,v2=Variant::Empty,off=None),text_signature="($self, rf='ITRF', v0=[], v1=[], v2=[], off={})")]
423    fn baseline(
424        &self,
425        py: Python<'_>,
426        #[pyo3(from_py_with=super::arguments::supplied)] rf: Option<Bound<'_, PyAny>>,
427        v0: Variant,
428        v1: Variant,
429        v2: Variant,
430        #[pyo3(from_py_with=super::arguments::supplied)] off: Option<Bound<'_, PyAny>>,
431    ) -> PyResult<Py<PyAny>> {
432        let rf = super::arguments::reference(rf.as_ref(), "ITRF")?;
433        let off = super::arguments::record(off.as_ref(), "off")?;
434        self.call(py, |state, lines| {
435            define_three(
436                state,
437                MeasureKind::Baseline,
438                &rf,
439                [&v0, &v1, &v2],
440                &off,
441                lines,
442            )
443        })
444    }
445    /// Define one or more uvw vectors, with an optional reference origin.
446    #[pyo3(signature=(rf=None,v0=Variant::Empty,v1=Variant::Empty,v2=Variant::Empty,off=None),text_signature="($self, rf='ITRF', v0=[], v1=[], v2=[], off={})")]
447    fn uvw(
448        &self,
449        py: Python<'_>,
450        #[pyo3(from_py_with=super::arguments::supplied)] rf: Option<Bound<'_, PyAny>>,
451        v0: Variant,
452        v1: Variant,
453        v2: Variant,
454        #[pyo3(from_py_with=super::arguments::supplied)] off: Option<Bound<'_, PyAny>>,
455    ) -> PyResult<Py<PyAny>> {
456        let rf = super::arguments::reference(rf.as_ref(), "ITRF")?;
457        let off = super::arguments::record(off.as_ref(), "off")?;
458        self.call(py, |state, lines| {
459            define_three(state, MeasureKind::Uvw, &rf, [&v0, &v1, &v2], &off, lines)
460        })
461    }
462    /// Define a magnetic field using its strength, longitude and latitude.
463    #[pyo3(signature=(rf=None,v0=Variant::Empty,v1=Variant::Empty,v2=Variant::Empty,off=None),text_signature="($self, rf='IGRF', v0=[], v1=[], v2=[], off={})")]
464    fn earthmagnetic(
465        &self,
466        py: Python<'_>,
467        #[pyo3(from_py_with=super::arguments::supplied)] rf: Option<Bound<'_, PyAny>>,
468        v0: Variant,
469        v1: Variant,
470        v2: Variant,
471        #[pyo3(from_py_with=super::arguments::supplied)] off: Option<Bound<'_, PyAny>>,
472    ) -> PyResult<Py<PyAny>> {
473        let rf = super::arguments::reference(rf.as_ref(), "IGRF")?;
474        let off = super::arguments::record(off.as_ref(), "off")?;
475        self.call(py, |state, lines| {
476            define_three(
477                state,
478                MeasureKind::EarthMagnetic,
479                &rf,
480                [&v0, &v1, &v2],
481                &off,
482                lines,
483            )
484        })
485    }
486    /// Convert positions into J2000 baselines, retaining every input element.
487    #[pyo3(signature=(pos),text_signature="($self, pos)")]
488    fn asbaseline(&self, py: Python<'_>, pos: &Bound<'_, PyAny>) -> PyResult<Py<PyAny>> {
489        let pos = super::arguments::record(Some(pos), "pos")?;
490        self.call(py, |state, lines| asbaseline(state, &pos, lines))
491    }
492    /// Expand antenna coordinates into every pairwise baseline.
493    #[pyo3(signature=(v=None),text_signature="($self, v={})")]
494    fn expand(
495        &self,
496        py: Python<'_>,
497        #[pyo3(from_py_with=super::arguments::supplied)] v: Option<Bound<'_, PyAny>>,
498    ) -> PyResult<Py<PyAny>> {
499        let v = super::arguments::record(v.as_ref(), "v")?;
500        self.call(py, |_, lines| Ok(expand(&v, lines)))
501    }
502    /// Project baselines into uvw and uvw time derivatives.
503    #[pyo3(signature=(v=None),text_signature="($self, v={})")]
504    fn touvw(
505        &self,
506        py: Python<'_>,
507        #[pyo3(from_py_with=super::arguments::supplied)] v: Option<Bound<'_, PyAny>>,
508    ) -> PyResult<Py<PyAny>> {
509        let v = super::arguments::record(v.as_ref(), "v")?;
510        self.call(py, |state, lines| touvw(state, &v, lines))
511    }
