2//! Bounds on what opening an MSv4 processing set holds, taken from its
3//! metadata before a value is read: every array's and group's metadata,
4//! and each dataset's coordinates and descriptions, which the opening
5//! reads whole. Its visibilities and their companions over time and
6//! baseline are read in batches, and pointing, weather and the other
7//! datasets only when asked for.
8
9use std::fs;
10use std::path::Path;
11use std::sync::Arc;
12
13use casa_observation::ObservationError;
14use zarrs::array::ArrayMetadata;
15use zarrs_filesystem::FilesystemStore;
16
17use super::metadata::{chunk_bytes, open_array, storage};

19/// Times their files' size the processing set's metadata (`zarr.json`)
20/// holds once parsed: array descriptions and attributes.
21const METADATA_FACTOR: usize = 16;
22
23/// What the processing set at `path` holds of its metadata once opened.
24pub(super) fn metadata_bytes(path: &Path) -> usize {
25    METADATA_FACTOR * metadata_files(path, 3)
26}
27
28/// The bytes of the `zarr.json` files at most `depth` levels below
29/// `directory`, its own included.
30fn metadata_files(directory: &Path, depth: usize) -> usize {
31    let own = fs::metadata(directory.join("zarr.json")).map_or(0, |meta| meta.len() as usize);
32    if depth == 0 {
33        return own;
34    }
35    let Ok(entries) = fs::read_dir(directory) else {
36        return own;
37    };
38    own + entries
39        .flatten()
40        .filter(|entry| entry.file_type().is_ok_and(|kind| kind.is_dir()))
41        .map(|entry| metadata_files(&entry.path(), depth - 1))
42        .sum::<usize>()
43}

45/// What reading the opening's arrays of dataset `dataset` (`/name`) of the
46/// processing set at `root` decodes: every array of the dataset but those
47/// over time and baseline, and every array of its antenna and
48/// field-and-source datasets.
49///
50/// # Errors
51///
52/// A dataset directory or array metadata that cannot be read.
53pub(super) fn dataset_bytes(
54    store: &Arc<FilesystemStore>,
55    root: &Path,
56    dataset: &str,
57) -> Result<usize, ObservationError> {
58    let directory = root.join(dataset.trim_start_matches('/'));
59    let mut bytes = 0;
60    for entry in fs::read_dir(&directory).map_err(storage)?.flatten() {
61        let name = entry.file_name().to_string_lossy().into_owned();
62        let path = entry.path();
63        if name.starts_with("antenna_xds") || name.starts_with("field_and_source") {
64            for nested in fs::read_dir(&path).map_err(storage)?.flatten() {
65                if is_array(&nested.path()) {
66                    let nested = nested.file_name().to_string_lossy().into_owned();
67                    bytes += decoded(store, &format!("{dataset}/{name}/{nested}"), false)?;
68                }
69            }
70        } else if is_array(&path) {
71            bytes += decoded(store, &format!("{dataset}/{name}"), true)?;
72        }
73    }
74    Ok(bytes)
75}

77/// Whether `directory` holds a Zarr array.
78fn is_array(directory: &Path) -> bool {
79    fs::read(directory.join("zarr.json"))
80        .ok()
81        .and_then(|text| serde_json::from_slice::<serde_json::Value>(&text).ok())
82        .is_some_and(|metadata| {
83            metadata.get("node_type").and_then(|kind| kind.as_str()) == Some("array")
84        })
85}

87/// The bytes reading the array at `path` whole decodes: its elements at
88/// its data type's size three times (the codec's bytes, the values, and a
89/// string's text beside its characters) with a string's two headers each;
90/// nothing for an array whose elements have no fixed size, which no
91/// opening reads whole, or, when `batched`, for one over time and baseline.
92fn decoded(
93    store: &Arc<FilesystemStore>,
94    path: &str,
95    batched: bool,
96) -> Result<usize, ObservationError> {
97    let array = open_array(store, path)?;
98    if batched && let ArrayMetadata::V3(metadata) = array.metadata() {
99        let named = |dimension: &str| {
100            metadata
101                .dimension_names
102                .as_ref()
103                .is_some_and(|names| names.iter().any(|name| name.as_deref() == Some(dimension)))
104        };
105        if named("time") && named("baseline_id") {
106            return Ok(0);
107        }
108    }
109    let Some(size) = array.data_type().fixed_size() else {
110        return Ok(0);
111    };
112    let elements = array.shape().iter().fold(1_usize, |product, extent| {
113        product.saturating_mul(usize::try_from(*extent).unwrap_or(usize::MAX))
114    });
115    Ok(elements
116        .saturating_mul(3 * size + 2 * size_of::<String>())
117        // A chunk may be larger than the logical array and decode beside
118        // its output. Opening reads one chunk at a time.
119        .saturating_add(chunk_bytes(&array)?.saturating_mul(2)))
120}
