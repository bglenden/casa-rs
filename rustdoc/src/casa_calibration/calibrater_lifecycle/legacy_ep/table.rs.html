18pub(super) fn write(ms: &MeasurementSet, solver: &LegacyEpSolver, slots: &[Slot]) -> MsResult<()> {
19    let mut writer = CalibrationTableWriter::with_mode(
20        ms,
21        CalibrationTableDescriptor {
22            output: solver.output_path(),
23            schema: main_schema(),
24            subtype: "EP Jones",
25            parameter_type: None,
26            measurement_set_name: String::new(),
27            include_polarization_basis: false,
28            time_extra_precision_column: Some("TIME_EXTRA_PREC"),
29        },
30        solver.mode,
31    )
32    .map_err(invalid)?;
33    writer.preserve_row_order();
34    *writer.table_mut().keywords_mut() = RecordValue::new(vec![
35        RecordField::new("CAL_DESC", Value::table_ref("././CAL_DESC")),
36        int("CAL_DESC_ID", 0),
37        RecordField::new("CAL_HISTORY", Value::table_ref("././CAL_HISTORY")),
38        int("CAL_HISTORY_ID", 0),
39    ]);
40    let table = writer.table_mut();
41    set_frequency_reference(table, "REF_FREQUENCY");
42    set_fixed_unit_keyword(table, "REF_DIRECTION", &["rad", "rad"]);
43    set_variable_reference(
44        table,
45        "REF_DIRECTION",
46        "direction",
47        "MEAS_DIR_REF",
48        direction_refs(),
49    );
50    let mut by_window = BTreeMap::<_, Vec<_>>::new();
51    for slot in slots {
52        by_window.entry(slot.spw).or_default().push(slot);
53    }
54    for (description, slots) in by_window.values().enumerate() {
55        for slot in slots {
56            for (antenna, offsets) in slot.offsets.iter().enumerate() {
57                let mut fields = vec![
58                    number("TIME", ScalarValue::Float64(slot.time)),
59                    number("TIME_EXTRA_PREC", ScalarValue::Float64(0.)),
60                    number("INTERVAL", ScalarValue::Float64(solver.interval_seconds)),
61                    int("ANTENNA1", antenna as i32),
62                    int("FIELD_ID", slot.field),
63                    int("CAL_DESC_ID", description as i32),
64                    number("TOTAL_SOLUTION_OK", ScalarValue::Bool(true)),
65                    number("TOTAL_FIT", ScalarValue::Float32(1.)),
66                    number("TOTAL_FIT_WEIGHT", ScalarValue::Float32(1.)),
67                    array(
68                        "GAIN",
69                        ArrayValue::Float32(shaped(&[4, 1], offsets.to_vec())),
70                    ),
71                    array(
72                        "SOLUTION_OK",
73                        ArrayValue::Bool(shaped(&[4, 1], vec![true; 4])),
74                    ),
75                    array("FIT", ArrayValue::Float32(shaped(&[1, 1], vec![1.]))),
76                    array("FIT_WEIGHT", ArrayValue::Float32(shaped(&[1, 1], vec![1.]))),
77                    array("FLAG", ArrayValue::Bool(shaped(&[4, 1], vec![false; 4]))),
78                    array("SNR", ArrayValue::Float32(shaped(&[4, 1], vec![0.; 4]))),
79                ];
80                fields.extend(
81                    [
82                        "FEED1",
83                        "ARRAY_ID",
84                        "OBSERVATION_ID",
85                        "SCAN_NUMBER",
86                        "PROCESSOR_ID",
87                        "STATE_ID",
88                        "PHASE_ID",
89                        "PULSAR_BIN",
90                        "PULSAR_GATE_ID",
91                        "FREQ_GROUP",
92                        "CALIBRATION_GROUP",
93                        "MEAS_FREQ_REF",
94                        "MEAS_DIR_REF",
95                        "CAL_HISTORY_ID",
96                    ]
97                    .map(|name| int(name, 0)),
98                );
99                fields.extend(
100                    [
101                        "FREQ_GROUP_NAME",
102                        "FIELD_NAME",
103                        "FIELD_CODE",
104                        "SOURCE_NAME",
105                        "SOURCE_CODE",
106                    ]
107                    .map(|name| number(name, ScalarValue::String(String::new()))),
108                );
109                writer.append(RecordValue::new(fields)).map_err(invalid)?;
110            }
111        }
112    }
113    let desc = description(by_window.into_keys())?;
114    desc.save(TableOptions::new(writer.output_path().join("CAL_DESC")))?;
115    write_legacy_history(&writer.output_path().join("CAL_HISTORY"))?;
116    writer.save_main().map_err(invalid)
117}

119fn main_schema() -> TableSchema {
120    let scalar = ColumnSchema::scalar;
121    let array = |name, kind| ColumnSchema::array_variable(name, kind, None);
122    let mut columns = vec![
123        scalar("TIME", Float64),
124        scalar("TIME_EXTRA_PREC", Float64),
125        scalar("INTERVAL", Float64),
126    ];
127    columns.extend(
128        [
129            "ANTENNA1",
130            "FEED1",
131            "FIELD_ID",
132            "ARRAY_ID",
133            "OBSERVATION_ID",
134            "SCAN_NUMBER",
135            "PROCESSOR_ID",
136            "STATE_ID",
137            "PHASE_ID",
138            "PULSAR_BIN",
139            "PULSAR_GATE_ID",
140            "FREQ_GROUP",
141        ]
142        .map(|name| scalar(name, Int32)),
143    );
144    columns.extend(
145        [
146            "FREQ_GROUP_NAME",
147            "FIELD_NAME",
148            "FIELD_CODE",
149            "SOURCE_NAME",
150            "SOURCE_CODE",
151        ]
152        .map(|name| scalar(name, Text)),
153    );
154    columns.extend([
155        scalar("CALIBRATION_GROUP", Int32),
156        array("GAIN", Float32),
157        array("REF_ANT", Int32),
158        array("REF_FEED", Int32),
159        array("REF_RECEPTOR", Int32),
160        array("REF_FREQUENCY", Float64),
161        scalar("MEAS_FREQ_REF", Int32),
162        array("REF_DIRECTION", Float64),
163        scalar("MEAS_DIR_REF", Int32),
164        scalar("CAL_DESC_ID", Int32),
165        scalar("CAL_HISTORY_ID", Int32),
166        scalar("TOTAL_SOLUTION_OK", Bool),
167        scalar("TOTAL_FIT", Float32),
168        scalar("TOTAL_FIT_WEIGHT", Float32),
169        array("SOLUTION_OK", Bool),
170        array("FIT", Float32),
171        array("FIT_WEIGHT", Float32),
172        array("FLAG", Bool),
173        array("SNR", Float32),
174    ]);
175    TableSchema::new(columns).expect("legacy EP table schema")
176}

178fn description(windows: impl Iterator<Item = i32>) -> MsResult<Table> {
179    let scalar = ColumnSchema::scalar;
180    let array_column = |name, kind| ColumnSchema::array_variable(name, kind, None);
181    let schema = TableSchema::new(vec![
182        scalar("NUM_SPW", Int32),
183        array_column("NUM_CHAN", Int32),
184        scalar("NUM_RECEPTORS", Int32),
185        scalar("N_JONES", Int32),
186        array_column("SPECTRAL_WINDOW_ID", Int32),
187        array_column("CHAN_FREQ", Float64),
188        scalar("MEAS_FREQ_REF", Int32),
189        array_column("CHAN_WIDTH", Float64),
190        array_column("CHAN_RANGE", Int32),
191        array_column("POLARIZATION_TYPE", Text),
192        scalar("JONES_TYPE", Text),
193        scalar("MS_NAME", Text),
194    ])
195    .expect("legacy EP CAL_DESC schema");
196    let mut table = Table::with_schema(schema);
197    set_frequency_reference(&mut table, "CHAN_FREQ");
198    set_fixed_unit_keyword(&mut table, "CHAN_WIDTH", &["Hz"]);
199    for spw in windows {
200        table.add_row(RecordValue::new(vec![
201            int("NUM_SPW", 1),
202            array("NUM_CHAN", ArrayValue::from_i32_vec(vec![1])),
203            int("NUM_RECEPTORS", 2),
204            int("N_JONES", 2),
205            array("SPECTRAL_WINDOW_ID", ArrayValue::from_i32_vec(vec![spw])),
206            int("MEAS_FREQ_REF", 0),
207            array("CHAN_FREQ", ArrayValue::Float64(shaped(&[1, 1], vec![0.]))),
208            array("CHAN_WIDTH", ArrayValue::Float64(shaped(&[1, 1], vec![0.]))),
209            array(
210                "CHAN_RANGE",
211                ArrayValue::Int32(shaped(&[2, 1, 1], vec![0, 0])),
212            ),
213            array(
214                "POLARIZATION_TYPE",
215                ArrayValue::String(shaped(&[1, 1], vec![String::new()])),
216            ),
217            number("JONES_TYPE", ScalarValue::String("full".into())),
218            number("MS_NAME", ScalarValue::String(".".into())),
219        ]))?;
220    }
221    Ok(table)
222}

224fn set_frequency_reference(table: &mut Table, name: &str) {
225    set_fixed_unit_keyword(table, name, &["Hz"]);
226    let refs = FrequencyRef::ALL
227        .into_iter()
228        .chain([FrequencyRef::Undefined])
229        .map(|reference| (reference.casacore_code() as u32, reference.as_str()));
230    set_variable_reference(table, name, "frequency", "MEAS_FREQ_REF", refs);
231}

233fn direction_refs() -> Vec<(u32, &'static str)> {
234    let mut references = Vec::new();
235    for code in (0..=21).chain(32..=42) {
236        let reference = DirectionRef::from_casacore_code(code).expect("CASA direction reference");
237        references.push((code as u32, reference.as_str()));
238        if code == 11 {
239            references.push((10, "AZELNE"));
240        } else if code == 13 {
241            references.push((12, "AZELNEGEO"));
242        }
243    }
244    references
245}

247fn set_variable_reference(
248    table: &mut Table,
249    name: &str,
250    kind: &str,
251    reference_column: &str,
252    references: impl IntoIterator<Item = (u32, &'static str)>,
253) {
254    let (codes, names) = references
255        .into_iter()
256        .map(|(code, name)| (code, name.to_owned()))
257        .unzip();
258    let mut keywords = table.column_keywords(name).cloned().unwrap_or_default();
259    keywords.upsert(
260        "MEASINFO",
261        Value::Record(RecordValue::new(vec![
262            array("TabRefCodes", ArrayValue::from_u32_vec(codes)),
263            array("TabRefTypes", ArrayValue::from_string_vec(names)),
264            number("VarRefCol", ScalarValue::String(reference_column.into())),
265            number("type", ScalarValue::String(kind.into())),
266        ])),
267    );
268    table.set_column_keywords(name, keywords);
269}

271fn shaped<T>(shape: &[usize], values: Vec<T>) -> ArrayD<T> {
272    ArrayD::from_shape_vec(IxDyn(shape).f(), values).expect("legacy EP cell extent")
273}
