18#[pymethods]
19#[allow(
20    non_snake_case,
21    unused_variables,
22    clippy::too_many_arguments,
23    clippy::unused_self
24)]
25impl PyCalibrater {
26    #[pyo3(signature = (modelimage), text_signature = "($self, modelimage)")]
27    fn setmodel(&self, py: Python<'_>, modelimage: &str) -> PyResult<bool> {
28        self.with_lifecycle(py, |state| {
29            state
30                .set_image_model(Path::new(modelimage))
31                .map_err(runtime_error)?;
32            Ok(true)
33        })
34    }
35
36    #[pyo3(signature = (caltable = None, time = None, spw = None, antenna = None, pol = None, caltype = None, parameter = None, infile = None, uniform = None), text_signature = "($self, caltable='', time='', spw='', antenna='', pol='', caltype='', parameter=[1.0], infile='', uniform=True)")]
37    fn specifycal(
38        &self,
39        py: Python<'_>,
40        #[pyo3(from_py_with = supplied)] caltable: Option<Bound<'_, PyAny>>,
41        #[pyo3(from_py_with = supplied)] time: Option<Bound<'_, PyAny>>,
42        #[pyo3(from_py_with = supplied)] spw: Option<Bound<'_, PyAny>>,
43        #[pyo3(from_py_with = supplied)] antenna: Option<Bound<'_, PyAny>>,
44        #[pyo3(from_py_with = supplied)] pol: Option<Bound<'_, PyAny>>,
45        #[pyo3(from_py_with = supplied)] caltype: Option<Bound<'_, PyAny>>,
46        #[pyo3(from_py_with = supplied)] parameter: Option<Bound<'_, PyAny>>,
47        #[pyo3(from_py_with = supplied)] infile: Option<Bound<'_, PyAny>>,
48        #[pyo3(from_py_with = supplied)] uniform: Option<Bound<'_, PyAny>>,
49    ) -> PyResult<bool> {
50        let table = text(caltable, "caltable", "")?;
51        // CASA accepts the time label without assigning it to table rows.
52        text(time, "time", "")?;
53        let spw = text(spw, "spw", "")?;
54        let antenna = text(antenna, "antenna", "")?;
55        let polarization = text(pol, "pol", "")?;
56        let caltype = text(caltype, "caltype", "")?;
57        let (parameters, tuple) = specify_parameters(parameter.as_ref())?;
58        let infile = text(infile, "infile", "")?;
59        let uniform = uniform
60            .as_ref()
61            .map(|value| strict_bool(value, "specifycal", 10))
62            .transpose()?
63            .unwrap_or(true);
64        let wrote = self.attached(py, move |state| {
65            state.specify_calibration(CalibrationSpecification {
66                table: Path::new(&table),
67                spw: &spw,
68                antenna: &antenna,
69                polarization: &polarization,
70                caltype: &caltype,
71                parameters,
72                infile: (!infile.is_empty()).then_some(Path::new(&infile)),
73                uniform,
74            })
75        })?;
76        if tuple {
77            // SWIG converts the tuple and runs the writer, then reports the
78            // outstanding Python conversion exception on its return.
79            return Err(PySystemError::new_err(
80                "<built-in function calibrater_specifycal> returned a result with an exception set",
81            ));
82        }
83        Ok(wrote)
84    }
85
86    #[pyo3(signature = (), text_signature = "($self)")]
87    fn state(&self, py: Python<'_>) -> PyResult<bool> {
88        use crate::tools::{LogLine, emit};
89
90        let lines = self.with_lifecycle(py, |state| Ok(state.state_lines()))?;
91        let attached = lines.is_some();
92        let logs = match lines {
93            Some(lines) => lines.into_iter().map(LogLine::info).collect(),
94            None => vec![LogLine::error("Must first open a MeasurementSet.")],
95        };
96        emit(py, super::LOGGER, &logs)?;
97        Ok(attached)
98    }
99}

101fn specify_parameters(value: Option<&Bound<'_, PyAny>>) -> PyResult<(Vec<f64>, bool)> {
102    let Some(value) = value else {
103        return Ok((vec![1.], false));
104    };
105    if value.is_none()
106        || value.is_instance(&value.py().import("numpy")?.getattr("complexfloating")?)?
107        || value.is_instance_of::<pyo3::types::PyComplex>()
108    {
109        return Ok((Vec::new(), false));
110    }
111    if value.is_instance_of::<PyUntypedArray>()
112        && value.getattr("dtype")?.getattr("kind")?.extract::<char>()? == 'O'
113    {
114        return Ok((Vec::new(), false));
115    }
116    // Scalar text takes the default value of the failed Python-to-double cast.
117    // Text within a list or array goes through CASA's vector element conversion.
118    if value.is_instance_of::<PyString>() || value.is_instance_of::<PyBytes>() {
119        return Ok((vec![-1.], false));
120    }
121    if value.is_instance_of::<PyTuple>() {
122        return Ok((value.extract::<DoubleList>()?.0, true));
123    }
124    let mut parameters = Vec::new();
125    if let Ok(list) = value.cast::<PyList>()
126        && list.iter().any(|item| item.is_instance_of::<PyList>())
127    {
128        let array = value
129            .py()
130            .import("numpy")?
131            .call_method1("asarray", (value,))?;
132        append_double_vector(&array, &mut parameters, 0)?;
133    } else {
134        append_double_vector(value, &mut parameters, 0)?;
135    }
136    Ok((parameters, false))
137}
