18const COEFFICIENTS: [f32; 3] = [0.25, 0.5, 0.25];
19
20pub(super) fn smooth(value: &mut ArrayValue, flag_shape: &[usize]) -> Result<(), TableError> {
21    if flag_shape.len() != 2 || value.shape() != flag_shape {
22        return Err(TableError::Schema(
23            "visibility and FLAG must have matching [corr, chan] shapes for Hanning smoothing"
24                .into(),
25        ));
26    }
27    match value {
28        ArrayValue::Complex32(array) => smooth_values(array),
29        ArrayValue::Float32(array) => smooth_values(array),
30        _ => {
31            return Err(TableError::Schema(
32                "visibility arrays must hold Complex32 or Float32".into(),
33            ));
34        }
35    }
36    Ok(())
37}
38
39fn smooth_values<T>(array: &mut ArrayD<T>)
40where
41    T: Copy + Add<Output = T> + Mul<f32, Output = T>,
42{
43    apply_interior(array, |[left, center, right]| {
44        left * COEFFICIENTS[0] + center * COEFFICIENTS[1] + right * COEFFICIENTS[2]
45    });
46}
47
48pub(super) fn flag(value: &mut ArrayValue) {
49    let ArrayValue::Bool(flags) = value else {
50        unreachable!("FLAG has already been validated")
51    };
52    apply_interior(flags, |[left, center, right]| left || center || right);
53    let channels = flags.len_of(Axis(1));
54    if channels > 0 {
55        flags.index_axis_mut(Axis(1), 0).fill(true);
56        flags.index_axis_mut(Axis(1), channels - 1).fill(true);
57    }
58}
59
60fn apply_interior<T: Copy>(array: &mut ArrayD<T>, filter: impl Fn([T; 3]) -> T) {
61    let channels = array.len_of(Axis(1));
62    if channels < 3 {
63        return;
64    }
65    for mut row in array.lanes_mut(Axis(1)) {
66        let mut left = row[0];
67        let mut center = row[1];
68        for channel in 1..channels - 1 {
69            let right = row[channel + 1];
70            row[channel] = filter([left, center, right]);
71            left = center;
72            center = right;
73        }
74    }
75}

77pub(super) fn statistics(
78    (scalar_weight, scalar_sigma): &(ArrayValue, ArrayValue),
79    spectra: SpectrumCells<'_>,
80    flags: &ArrayValue,
81    bins: &[Vec<usize>],
82    averages: Option<&ChannelAverages<'_>>,
83    visibility: &VisibilityColumns,
84    time_average: bool,
85) -> Result<AveragedStatistics, TableError> {
86    let (weight, observed_weight, flags) = if let Some(averages) = averages {
87        let (weight, observed_weight) = averages.weight_spectra();
88        (weight, observed_weight, averages.bin_flags())
89    } else {
90        let ArrayValue::Bool(flags) = flags else {
91            return Err(TableError::Schema("FLAG must hold Bool".into()));
92        };
93        let &[correlations, channels] = flags.shape() else {
94            return Err(TableError::Schema(
95                "FLAG must be rank-2 [corr, chan]".into(),
96            ));
97        };
98        let shape = [correlations, channels];
99        let weight = InputStatistic::from_cells(scalar_weight, spectra.weight, shape, "WEIGHT")?;
100        let sigma = InputStatistic::from_cells(scalar_sigma, spectra.sigma, shape, "SIGMA")?;
101        let has = |source| visibility.copies().iter().any(|copy| copy.source == source);
102        let dims = (correlations, bins.len());
103        let calibrated =
104            || Array2::from_shape_fn(dims, |(corr, bin)| weight.value(corr, bins[bin][0]));
105        let observed = || {
106            Array2::from_shape_fn(dims, |(corr, bin)| {
107                weight_from_sigma(sigma.value(corr, bins[bin][0]))
108            })
109        };
110        let (weight, observed_weight) = if has(Data) && has(CorrectedData) {
111            (calibrated(), observed())
112        } else {
113            let weight = if time_average || has(CorrectedData) {
114                calibrated()
115            } else if has(Data) {
116                observed()
117            } else if has(ModelData) {
118                Array2::ones(dims)
119            } else {
120                // Keep defined noise columns for FLOAT_DATA and LAG_DATA.
121                calibrated()
122            };
123            (weight.clone(), weight)
124        };
125        let selected = Array2::from_shape_fn(dims, |(corr, bin)| flags[[corr, bins[bin][0]]]);
126        (weight, observed_weight, selected)
127    };
128    let (weight, weight_spectrum) = filter_noise(weight, &flags, |weight| weight);
129    let (sigma, sigma_spectrum) = filter_noise(observed_weight, &flags, sigma_from_weight);
130    Ok(AveragedStatistics {
131        weight: ArrayValue::Float32(weight.into_dyn()),
132        sigma: ArrayValue::Float32(sigma.into_dyn()),
133        weight_spectrum: ArrayValue::Float32(weight_spectrum.into_dyn()),
134        sigma_spectrum: ArrayValue::Float32(sigma_spectrum.into_dyn()),
135    })
136}

138fn filter_noise(
139    mut weights: Array2<f32>,
140    flags: &Array2<bool>,
141    convert: impl Fn(f32) -> f32,
142) -> (Array1<f32>, Array2<f32>) {
143    let channels = weights.ncols();
144    let mut scalars = Array1::zeros(weights.nrows());
145    if channels == 0 {
146        return (scalars, weights);
147    }
148    for ((mut row, mask), scalar) in weights
149        .rows_mut()
150        .into_iter()
151        .zip(flags.rows())
152        .zip(&mut scalars)
153    {
154        let mut sum = 0.;
155        let mut unflagged_sum = 0.;
156        let mut unflagged = 0;
157        let mut left = 0.;
158        let mut center = row[0];
159        for channel in 0..channels {
160            let mut output = center;
161            if channel > 0 && channel + 1 < channels {
162                let right = row[channel + 1];
163                let inputs = [left, center, right];
164                let mut variance = 0.;
165                let mut flagged = false;
166                for (offset, (weight, coefficient)) in
167                    inputs.into_iter().zip(COEFFICIENTS).enumerate()
168                {
169                    if weight <= f32::MIN_POSITIVE {
170                        flagged = true;
171                    } else {
172                        // An unknown (NaN) variance must propagate through
173                        // the filter. Zero or negative weight marks a flag.
174                        variance += coefficient * coefficient / weight;
175                    }
176                    flagged |= mask[channel - 1 + offset];
177                }
178                output = if variance > f32::MIN_POSITIVE {
179                    variance.recip()
180                } else {
181                    variance
182                };
183                if !flagged {
184                    unflagged_sum += convert(output);
185                    unflagged += 1;
186                }
187            }
188            row[channel] = convert(output);
189            sum += row[channel];
190            left = center;
191            center = row.get(channel + 1).copied().unwrap_or_default();
192        }
193        *scalar = if unflagged == 0 {
194            sum / channels as f32
195        } else {
196            unflagged_sum / unflagged as f32
197        };
198    }
199    (scalars, weights)
200}
