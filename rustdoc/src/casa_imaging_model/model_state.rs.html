30/// One finite, canonical semantic `f64` model coefficient or increment.
31///
32/// The representation is precision-independent. Arithmetic that creates a new
33/// value remains governed by the lifecycle precision selected from the
34/// Compiled Problem's Numerics Contract.
35#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
36pub struct ModelValue(u64);
37
38impl ModelValue {
39    /// Construct a finite value, canonicalizing negative zero.
40    pub fn new(value: f64) -> Result<Self, ModelContractError> {
41        if !value.is_finite() {
42            return Err(ModelContractError::NonFiniteValue);
43        }
44        Ok(Self(canonical_f64_bits(value)))
45    }
46
47    /// Return the represented value.
48    #[must_use]
49    pub fn value(self) -> f64 {
50        f64::from_bits(self.0)
51    }
52
53    pub(crate) const fn bits(self) -> u64 {
54        self.0
55    }
56}

107/// Compute the canonical identity of one target-ordered validity mask.
108///
109/// The identity commits support independently from numeric payloads so an
110/// invalid coefficient can never alias a valid numeric zero.
111#[must_use]
112pub fn model_support_identity(support: impl IntoIterator<Item = ModelSupport>) -> LogicalIdentity {
113    let mut encoder = CanonicalEncoder::new();
114    encoder.bytes(MODEL_SUPPORT_IDENTITY_DOMAIN);
115    encoder.u32(MODEL_SUPPORT_IDENTITY_VERSION);
116    let mut count = 0usize;
117    for value in support {
118        encoder.u8(match value {
119            ModelSupport::Valid => 1,
120            ModelSupport::Invalid => 0,
121        });
122        count += 1;
123    }
124    encoder.usize(count);
125    LogicalIdentity::from_sha256(encoder.finish())
126}

128/// Explicit residency and numeric ceilings for one model lifecycle.
129#[derive(Debug, Clone, Copy, PartialEq, Eq)]
130pub struct ModelBounds {
131    max_model_samples: usize,
132    max_source_samples: usize,
133    max_reprojection_terms: usize,
134    max_delta_terms: usize,
135    max_absolute_model_value: ModelValue,
136    max_absolute_delta_value: ModelValue,
137}
138
139impl ModelBounds {
140    /// Construct positive finite numeric bounds and exact count ceilings.
141    pub fn new(
142        max_model_samples: usize,
143        max_source_samples: usize,
144        max_reprojection_terms: usize,
145        max_delta_terms: usize,
146        max_absolute_model_value: f64,
147        max_absolute_delta_value: f64,
148    ) -> Result<Self, ModelContractError> {
149        if max_model_samples == 0
150            || max_source_samples == 0
151            || max_reprojection_terms == 0
152            || max_delta_terms == 0
153            || !max_absolute_model_value.is_finite()
154            || max_absolute_model_value <= 0.0
155            || !max_absolute_delta_value.is_finite()
156            || max_absolute_delta_value <= 0.0
157        {
158            return Err(ModelContractError::InvalidBounds);
159        }
160        Ok(Self {
161            max_model_samples,
162            max_source_samples,
163            max_reprojection_terms,
164            max_delta_terms,
165            max_absolute_model_value: ModelValue::new(max_absolute_model_value)?,
166            max_absolute_delta_value: ModelValue::new(max_absolute_delta_value)?,
167        })
168    }
169
170    /// Return the maximum resident target-model sample count.
171    #[must_use]
172    pub const fn max_model_samples(self) -> usize {
173        self.max_model_samples
174    }
175
176    /// Return the maximum source-model sample count.
177    #[must_use]
178    pub const fn max_source_samples(self) -> usize {
179        self.max_source_samples
180    }
181
182    /// Return the maximum total interpolation terms in one reprojection.
183    #[must_use]
184    pub const fn max_reprojection_terms(self) -> usize {
185        self.max_reprojection_terms
186    }
187
188    /// Return the maximum number of terms in one Model Delta.
189    #[must_use]
190    pub const fn max_delta_terms(self) -> usize {
191        self.max_delta_terms
192    }
193
194    /// Return the absolute model-value ceiling.
195    #[must_use]
196    pub fn max_absolute_model_value(self) -> f64 {
197        self.max_absolute_model_value.value()
198    }
199
200    /// Return the absolute per-term delta ceiling.
201    #[must_use]
202    pub fn max_absolute_delta_value(self) -> f64 {
203        self.max_absolute_delta_value.value()
204    }
205
206    fn encode(self, encoder: &mut CanonicalEncoder) {
207        encoder.usize(self.max_model_samples);
208        encoder.usize(self.max_source_samples);
209        encoder.usize(self.max_reprojection_terms);
210        encoder.usize(self.max_delta_terms);
211        encoder.u64(self.max_absolute_model_value.bits());
212        encoder.u64(self.max_absolute_delta_value.bits());
213    }

269/// Rectangular model-array shape with typed WCS and coefficient-space provenance.
270#[derive(Debug, Clone, PartialEq)]
271pub struct ModelSourceShape {
272    coefficient_space: ModelCoefficientSpace,
273    domains: Box<[ImageShape]>,
274    domain_roles: Box<[ImageDomainRole]>,
275    directions: Box<[DirectionCoordinateSpec]>,
276    spectral: SpectralCoordinateSpec,
277    coefficients: usize,
278    polarizations: usize,
279    samples: usize,
280}
281
282// `ModelSourceShape` is constructed only from canonical compiled geometry, so
283// its finite floating-point coordinate laws have reflexive equality.
284impl Eq for ModelSourceShape {}
285
286impl ModelSourceShape {
287    /// Derive one checked shape from matching compiler-owned geometry and coefficient space.
288    pub fn from_compiled(
289        geometry: &CompiledGeometry,
290        coefficient_space: &ModelCoefficientSpace,
291    ) -> Result<Self, ModelContractError> {
292        if coefficient_space.geometry() != geometry.geometry_id() {
293            return Err(ModelContractError::CoefficientSpaceGeometryMismatch);
294        }
295        let domains = geometry
296            .domains()
297            .iter()
298            .map(|domain| domain.shape())
299            .collect::<Vec<_>>();
300        let directions = geometry
301            .domains()
302            .iter()
303            .map(|domain| domain.direction())
304            .collect::<Vec<_>>();
305        let domain_roles = geometry
306            .domains()
307            .iter()
308            .map(|domain| domain.role().clone())
309            .collect::<Vec<_>>();
310        let coefficients = coefficient_count(coefficient_space.basis());
311        let polarizations = coefficient_space.polarization().coordinates().len();
312        if domains.is_empty() || coefficients == 0 || polarizations == 0 {
313            return Err(ModelContractError::InvalidShape);
314        }
315        let mut samples = 0usize;
316        for domain in &domains {
317            let [width, height] = domain.pixels();
318            if width == 0 || height == 0 {
319                return Err(ModelContractError::InvalidShape);
320            }
321            let domain_samples = width
322                .checked_mul(height)
323                .and_then(|pixels| pixels.checked_mul(coefficients))
324                .and_then(|values| values.checked_mul(polarizations))
325                .ok_or(ModelContractError::ShapeTooLarge)?;
326            samples = samples
327                .checked_add(domain_samples)
328                .ok_or(ModelContractError::ShapeTooLarge)?;
329        }
330        Ok(Self {
331            coefficient_space: coefficient_space.clone(),
332            domains: domains.into_boxed_slice(),
333            domain_roles: domain_roles.into_boxed_slice(),
334            directions: directions.into_boxed_slice(),
335            spectral: geometry.spectral().clone(),
336            coefficients,
337            polarizations,
338            samples,
339        })
340    }

390    /// Return the closed commitment identity of this exact typed shape.
391    #[must_use]
392    pub fn identity(&self) -> LogicalIdentity {
393        let mut encoder = CanonicalEncoder::new();
394        encoder.bytes(MODEL_SOURCE_SHAPE_IDENTITY_DOMAIN);
395        encoder.u32(MODEL_SOURCE_SHAPE_IDENTITY_VERSION);
396        self.encode(&mut encoder);
397        LogicalIdentity::from_sha256(encoder.finish())
398    }
399
400    /// Return a cell's canonical flattened ordinal, when it belongs to this shape.
401    #[must_use]
402    pub fn flat_index(&self, cell: ModelCell) -> Option<usize> {
403        if cell.coefficient >= self.coefficients || cell.polarization >= self.polarizations {
404            return None;
405        }
406        let mut offset = 0usize;
407        for (domain_index, shape) in self.domains.iter().enumerate() {
408            let [width, height] = shape.pixels();
409            if domain_index == cell.domain {
410                let [x, y] = cell.pixel;
411                if x >= width || y >= height {
412                    return None;
413                }
414                let plane = cell
415                    .coefficient
416                    .checked_mul(self.polarizations)?
417                    .checked_add(cell.polarization)?;
418                return offset
419                    .checked_add(plane.checked_mul(width.checked_mul(height)?)?)?
420                    .checked_add(y.checked_mul(width)?)?
421                    .checked_add(x);
422            }
423            offset = offset.checked_add(
424                width
425                    .checked_mul(height)?
426                    .checked_mul(self.coefficients)?
427                    .checked_mul(self.polarizations)?,
428            )?;
429        }
430        None
431    }
432
433    /// Return the typed cell at one canonical flattened ordinal.
434    #[must_use]
435    pub fn cell_at(&self, mut index: usize) -> Option<ModelCell> {
436        if index >= self.samples {
437            return None;
438        }
439        for (domain, shape) in self.domains.iter().enumerate() {
440            let [width, height] = shape.pixels();
441            let pixels = width.checked_mul(height)?;
442            let domain_samples = pixels
443                .checked_mul(self.coefficients)?
444                .checked_mul(self.polarizations)?;
445            if index < domain_samples {
446                let plane = index / pixels;
447                index %= pixels;
448                return Some(ModelCell::new(
449                    domain,
450                    plane / self.polarizations,
451                    plane % self.polarizations,
452                    [index % width, index / width],
453                ));
454            }
455            index -= domain_samples;
456        }
457        None
458    }
459
460    pub(crate) fn encode(&self, encoder: &mut CanonicalEncoder) {
461        encoder.identity(model_coefficient_space_identity(&self.coefficient_space));
462        encoder.usize(self.domains.len());
463        for domain in &self.domains {
464            let [width, height] = domain.pixels();
465            encoder.usize(width);
466            encoder.usize(height);
467        }
468        encoder.usize(self.coefficients);
469        encoder.usize(self.polarizations);
470    }

473/// Digest projection of one reconstruction-owned, target-ordered preparation.
474///
475/// This projection is compiler and receipt data, not executable preparation
476/// authority. Only `casa-imaging-reconstruction` can bind it to the opaque
477/// prepared values and ordered interpolation stencils required for execution.
478///
479/// Reprojected preparation evidence is not a caller construction surface:
480///
481/// ```compile_fail
482/// use casa_imaging_model::{
483///     LogicalIdentity, ModelLifecycleContract, ModelReprojectedSeedProjection, ModelSample,
484///     ModelSourceShape,
485/// };
486///
487/// fn forge(
488///     target: &ModelLifecycleContract,
489///     source: LogicalIdentity,
490///     source_shape: ModelSourceShape,
491///     samples: &[ModelSample],
492/// ) {
493///     let _ = ModelReprojectedSeedProjection::from_prepared_samples(
494///         target,
495///         source,
496///         source_shape,
497///         samples,
498///     );
499/// }
500/// ```
501#[derive(Debug, Clone, PartialEq, Eq)]
502pub struct ModelReprojectedSeedProjection {
503    source: LogicalIdentity,
504    source_shape: Box<ModelSourceShape>,
505    preparation_contract: LogicalIdentity,
506    reprojection: LogicalIdentity,
507    support: LogicalIdentity,
508    samples: LogicalIdentity,
509    stencil: LogicalIdentity,
510    proof: LogicalIdentity,
511}
512
513impl ModelReprojectedSeedProjection {
514    /// Carry reconstruction-owned identities into the dependency-free compiler.
515    ///
516    /// Constructing this digest projection does not create executable evidence;
517    /// reconstruction's opaque executable-problem brand performs that binding.
518    #[allow(clippy::too_many_arguments)]
519    pub fn from_identities(
520        source: LogicalIdentity,
521        source_shape: ModelSourceShape,
522        preparation_contract: LogicalIdentity,
523        reprojection: LogicalIdentity,
524        support: LogicalIdentity,
525        samples: LogicalIdentity,
526        stencil: LogicalIdentity,
527        proof: LogicalIdentity,
528    ) -> Result<Self, ModelContractError> {
529        if !identities_are_set(&[
530            source,
531            preparation_contract,
532            reprojection,
533            support,
534            samples,
535            stencil,
536            proof,
537        ]) {
538            return Err(ModelContractError::InvalidReprojectedSeedProof);
539        }
540        Ok(Self {
541            source,
542            source_shape: Box::new(source_shape),
543            preparation_contract,
544            reprojection,
545            support,
546            samples,
547            stencil,
548            proof,
549        })
550    }
551
552    /// Return the immutable source artifact identity.
553    #[must_use]
554    pub const fn source(&self) -> LogicalIdentity {
555        self.source
556    }
557
558    /// Return the exact typed source model space.
559    #[must_use]
560    pub const fn source_shape(&self) -> &ModelSourceShape {
561        &self.source_shape
562    }
563
564    /// Return the empty-target preparation contract identity.
565    #[must_use]
566    pub const fn preparation_contract(&self) -> LogicalIdentity {
567        self.preparation_contract
568    }
569
570    /// Return the canonical mapping identity derived from both typed spaces.
571    #[must_use]
572    pub const fn reprojection(&self) -> LogicalIdentity {
573        self.reprojection
574    }
575
576    /// Return the canonical target-support identity.
577    #[must_use]
578    pub const fn support(&self) -> LogicalIdentity {
579        self.support
580    }
581
582    /// Return the canonical target-ordered value-and-support identity.
583    #[must_use]
584    pub const fn samples(&self) -> LogicalIdentity {
585        self.samples
586    }
587
588    /// Return the canonical ordered interpolation-stencil identity.
589    #[must_use]
590    pub const fn stencil(&self) -> LogicalIdentity {
591        self.stencil
592    }
593
594    /// Return the complete preparation-proof identity.
595    #[must_use]
596    pub const fn proof(&self) -> LogicalIdentity {
597        self.proof
598    }
599
600    fn validate(
601        &self,
602        target: &ModelSourceShape,
603        bounds: ModelBounds,
604        preparation_contract: LogicalIdentity,
605    ) -> Result<(), ModelContractError> {
606        if self.source_shape.sample_count() > bounds.max_source_samples {
607            return Err(ModelContractError::SourceSampleBoundExceeded {
608                samples: self.source_shape.sample_count(),
609                bound: bounds.max_source_samples,
610            });
611        }
612        if self.preparation_contract != preparation_contract {
613            return Err(ModelContractError::ReprojectionContractMismatch);
614        }
615        let expected_reprojection = model_reprojected_seed_mapping_identity(
616            preparation_contract,
617            self.source_shape.identity(),
618            target.identity(),
619        );
620        if self.source.as_bytes() == [0; 32]
621            || self.support.as_bytes() == [0; 32]
622            || self.samples.as_bytes() == [0; 32]
623            || self.stencil.as_bytes() == [0; 32]
624            || self.proof.as_bytes() == [0; 32]
625            || self.source_shape.as_ref() == target
626            || self.reprojection != expected_reprojection
627        {
628            Err(ModelContractError::InvalidReprojectedSeedProof)
629        } else {
630            Ok(())
631        }
632    }
633}

635/// Compiler input commitment for the initial authoritative model.
636#[derive(Debug, Clone, PartialEq, Eq)]
637pub enum ModelInputCommitment {
638    /// Begin from a zero-valued model with valid support everywhere.
639    Empty,
640    /// Ingest a seed already aligned to the compiler-derived target model space.
641    AlignedSeed {
642        /// Immutable source artifact identity.
643        source: LogicalIdentity,
644        /// Identity of the source artifact's exact validity mask.
645        support: LogicalIdentity,
646    },
647    /// Ingest a seed through one exact typed reprojection.
648    ///
649    /// Raw evidence is not a caller construction surface. Reprojected input is
650    /// accepted only through the model-validating commitment carried by
651    /// reconstruction's opaque preparation product.
652    ///
653    /// ```compile_fail
654    /// use casa_imaging_model::ModelInputCommitment;
655    ///
656    /// let _ = ModelInputCommitment::ReprojectedSeed {
657    ///     source: todo!(),
658    ///     source_shape: todo!(),
659    ///     contract: todo!(),
660    ///     reprojection: todo!(),
661    ///     support: todo!(),
662    /// };
663    /// ```
664    ReprojectedSeed(ModelReprojectedSeedProjection),
665    /// Resume the exact authoritative generation named by the observation input.
666    Generation(LogicalIdentity),
667}

669/// Digest-only input projection used to revalidate durable audit evidence.
670///
671/// This type cannot initialize or resume a model lifecycle. It exists solely
672/// so receipt readers can ask the model-contract owner to recompute a compiled
673/// lifecycle identity without reconstructing compiler-owned shapes or exposing
674/// raw executable commitments.
675#[derive(Debug, Clone, PartialEq, Eq)]
676pub enum ModelInputCommitmentIdentity {
677    /// A zero-valued initial model.
678    Empty,
679    /// An already aligned seed and its exact support mask.
680    AlignedSeed {
681        /// Immutable source artifact identity.
682        source: LogicalIdentity,
683        /// Source validity-mask identity.
684        support: LogicalIdentity,
685    },
686    /// One opaque reconstruction-prepared reprojection proof.
687    ReprojectedSeed {
688        /// Immutable source artifact identity.
689        source: LogicalIdentity,
690        /// Exact typed source-shape identity.
691        source_shape: LogicalIdentity,
692        /// Empty-target preparation-contract identity.
693        preparation_contract: LogicalIdentity,
694        /// Canonical source-to-target mapping identity.
695        reprojection: LogicalIdentity,
696        /// Projected target-support identity.
697        support: LogicalIdentity,
698        /// Canonical target-ordered projected sample identity.
699        samples: LogicalIdentity,
700        /// Canonical ordered interpolation-stencil identity.
701        stencil: LogicalIdentity,
702        /// Complete opaque preparation-proof identity.
703        proof: LogicalIdentity,
704    },
705    /// One exact previously minted generation.
706    Generation(LogicalIdentity),
707}
708
709impl ModelInputCommitmentIdentity {
710    fn from_commitment(input: &ModelInputCommitment) -> Self {
711        match input {
712            ModelInputCommitment::Empty => Self::Empty,
713            ModelInputCommitment::AlignedSeed { source, support } => Self::AlignedSeed {
714                source: *source,
715                support: *support,
716            },
717            ModelInputCommitment::ReprojectedSeed(commitment) => Self::ReprojectedSeed {
718                source: commitment.source(),
719                source_shape: commitment.source_shape().identity(),
720                preparation_contract: commitment.preparation_contract(),
721                reprojection: commitment.reprojection(),
722                support: commitment.support(),

724                stencil: commitment.stencil(),
725                proof: commitment.proof(),
726            },

1096/// Exact reason a model schema or compiler commitment was rejected.
1097#[derive(Debug, Clone, PartialEq, Eq, Error)]
1098pub enum ModelContractError {
1099    /// A model coefficient or increment was non-finite.
1100    #[error("model values must be finite")]
1101    NonFiniteValue,
1102    /// A reprojection weight was non-finite.
1103    #[error("reprojection weights must be finite")]
1104    NonFiniteReprojectionWeight,
1105    /// Numeric or count bounds were empty, non-finite, or non-positive.
1106    #[error("model lifecycle bounds must be explicit, finite, and positive")]
1107    InvalidBounds,
1108    /// A source or target shape was empty.
1109    #[error("model shapes require domains, coefficient planes, and polarization planes")]
1110    InvalidShape,
1111    /// A coefficient space did not belong to the supplied compiled geometry.
1112    #[error("model coefficient space does not belong to the supplied compiled geometry")]
1113    CoefficientSpaceGeometryMismatch,
1114    /// A shape sample count overflowed `usize`.
1115    #[error("model shape is too large to represent")]
1116    ShapeTooLarge,
1117    /// The target model exceeded its compiled residency ceiling.
1118    #[error("target model has {samples} samples, exceeding bound {bound}")]
1119    ModelSampleBoundExceeded {
1120        /// Exact target sample count.
1121        samples: usize,
1122        /// Declared target ceiling.
1123        bound: usize,
1124    },
1125    /// The source model exceeded its compiled residency ceiling.
1126    #[error("source model has {samples} samples, exceeding bound {bound}")]
1127    SourceSampleBoundExceeded {
1128        /// Exact source sample count.
1129        samples: usize,
1130        /// Declared source ceiling.
1131        bound: usize,
1132    },
1133    /// The lifecycle selected arithmetic not permitted by the Numerics Contract.
1134    #[error("model lifecycle arithmetic precision is not permitted by the Numerics Contract")]
1135    PrecisionNotPermitted,
1136    /// The lifecycle input commitment disagreed with the observation's initial model.
1137    #[error("model lifecycle input commitment differs from the observation model input")]
1138    InputCommitmentMismatch,
1139    /// Reprojection evidence disagreed with its Product, Numerics, precision, or policy inputs.
1140    #[error("model reprojection contract differs from its compiled inputs")]
1141    ReprojectionContractMismatch,
1142    /// A seed commitment used an all-zero evidence identity.
1143    #[error("model seed, support, and reprojection identities must be established")]
1144    UnidentifiedInputEvidence,
1145    /// Reprojected input was not one canonical opaque preparation proof.
1146    #[error("reprojected model input is not a canonical preparation proof")]
1147    InvalidReprojectedSeedProof,
1148    /// A persisted lifecycle identity disagreed with its complete typed projection.
1149    #[error("model lifecycle identity differs from its canonical typed projection")]
1150    LifecycleContractMismatch,

1153pub(crate) fn compile_model_lifecycle_contract(
1154    geometry: &CompiledGeometry,
1155    coefficient_space: &ModelCoefficientSpace,
1156    inputs: &ProblemInputIdentities,
1157    numerics: &NumericsContract,
1158    numerics_id: NumericsContractId,
1159    product_graph: ProductGraphId,
1160    requirements: ModelLifecycleRequirements,
1161) -> Result<ModelLifecycleContract, ModelContractError> {
1162    let target = ModelSourceShape::from_compiled(geometry, coefficient_space)?;
1163    let ModelLifecycleRequirements {
1164        bounds,
1165        arithmetic_precision,
1166        input,
1167    } = requirements;
1168    if target.sample_count() > bounds.max_model_samples {
1169        return Err(ModelContractError::ModelSampleBoundExceeded {
1170            samples: target.sample_count(),
1171            bound: bounds.max_model_samples,
1172        });
1173    }
1174    if !numerics
1175        .permitted_precisions()
1176        .contains(&arithmetic_precision)
1177    {
1178        return Err(ModelContractError::PrecisionNotPermitted);
1179    }
1180    let reprojection_policy = ModelReprojectionPolicy::canonical();
1181    let reprojection = compile_model_reprojection_contract(
1182        product_graph,
1183        numerics_id,
1184        arithmetic_precision,
1185        reprojection_policy,
1186    );
1187    let state_encoding = ModelStateEncoding::CanonicalF64;
1188    let support_semantics = ModelSupportSemantics::ExplicitValidity;
1189    let preparation_contract = model_lifecycle_contract_id(
1190        numerics_id,
1191        reprojection,
1192        &target,
1193        bounds,
1194        arithmetic_precision,
1195        state_encoding,
1196        support_semantics,
1197        &ModelInputCommitment::Empty,
1198    );
1199    validate_input_commitment(
1200        inputs.model(),
1201        &target,
1202        bounds,
1203        preparation_contract.identity(),
1204        &input,
1205    )?;
1206    let contract_id = model_lifecycle_contract_id(
1207        numerics_id,
1208        reprojection,
1209        &target,
1210        bounds,
1211        arithmetic_precision,
1212        state_encoding,
1213        support_semantics,
1214        &input,
1215    );
1216    Ok(ModelLifecycleContract {
1217        contract_id,
1218        reprojection_contract: LogicalIdentity::from_sha256(reprojection.as_bytes()),
1219        reprojection_policy,
1220        numerics: numerics_id,
1221        target,
1222        bounds,
1223        arithmetic_precision,
1224        state_encoding,
1225        support_semantics,
1226        input,
1227    })
1228}
1229
1230fn validate_input_commitment(
1231    model: ModelStateIdentity,
1232    target: &ModelSourceShape,
1233    bounds: ModelBounds,
1234    preparation_contract: LogicalIdentity,
1235    input: &ModelInputCommitment,
1236) -> Result<(), ModelContractError> {
1237    let matches = match (model, input) {
1238        (ModelStateIdentity::Empty, ModelInputCommitment::Empty) => true,
1239        (
1240            ModelStateIdentity::Seed(expected),
1241            ModelInputCommitment::AlignedSeed { source, support },
1242        ) => expected == *source && identities_are_set(&[*source, *support]),
1243        (ModelStateIdentity::Seed(expected), ModelInputCommitment::ReprojectedSeed(commitment)) => {
1244            commitment.validate(target, bounds, preparation_contract)?;
1245            expected == commitment.source()
1246        }
1247        (ModelStateIdentity::Generation(expected), ModelInputCommitment::Generation(actual)) => {
1248            expected == *actual && identities_are_set(&[*actual])
1249        }
1250        _ => false,
1251    };
1252    if matches {
1253        Ok(())
1254    } else if input_identities_are_unset(input) {
1255        Err(ModelContractError::UnidentifiedInputEvidence)
1256    } else {
1257        Err(ModelContractError::InputCommitmentMismatch)
1258    }
1259}
1260
1261fn input_identities_are_unset(input: &ModelInputCommitment) -> bool {
1262    match input {
1263        ModelInputCommitment::Empty => false,
1264        ModelInputCommitment::AlignedSeed { source, support } => {
1265            !identities_are_set(&[*source, *support])
1266        }
1267        ModelInputCommitment::ReprojectedSeed(commitment) => !identities_are_set(&[
1268            commitment.source(),
1269            commitment.preparation_contract(),
1270            commitment.reprojection(),
1271            commitment.support(),
1272            commitment.samples(),
1273            commitment.stencil(),
1274            commitment.proof(),
1275        ]),
1276        ModelInputCommitment::Generation(identity) => !identities_are_set(&[*identity]),
1277    }
1278}
1279
1280fn identities_are_set(identities: &[LogicalIdentity]) -> bool {
1281    identities
1282        .iter()
1283        .all(|identity| identity.as_bytes() != [0; 32])
1284}

1309/// Validate one persisted reprojection-contract identity against its exact inputs.
1310///
1311/// Product Graph and Numerics identities are accepted as logical identities so
1312/// receipt validation can rebind their persisted digests without exposing
1313/// constructors for compiler-derived identity types. The model owner retains
1314/// the canonical encoding and the typed policy interpretation.
1315pub fn validate_model_reprojection_contract_identity(
1316    claimed: LogicalIdentity,
1317    product_graph: LogicalIdentity,
1318    numerics: LogicalIdentity,
1319    conversion_precision: NumericPrecision,
1320    policy: ModelReprojectionPolicy,
1321) -> Result<(), ModelContractError> {
1322    if claimed
1323        == model_reprojection_contract_identity(
1324            product_graph,
1325            numerics,
1326            conversion_precision,
1327            policy,
1328        )
1329    {
1330        Ok(())
1331    } else {
1332        Err(ModelContractError::ReprojectionContractMismatch)
1333    }
1334}

1336fn model_reprojection_contract_identity(
1337    product_graph: LogicalIdentity,
1338    numerics: LogicalIdentity,
1339    conversion_precision: NumericPrecision,
1340    policy: ModelReprojectionPolicy,
1341) -> LogicalIdentity {
1342    let mut encoder = CanonicalEncoder::new();
1343    encoder.bytes(MODEL_REPROJECTION_CONTRACT_IDENTITY_DOMAIN);
1344    encoder.u32(MODEL_REPROJECTION_CONTRACT_IDENTITY_VERSION);
1345    encoder.digest(product_graph.as_bytes());
1346    encoder.digest(numerics.as_bytes());
1347    encoder.u8(precision_tag(conversion_precision));
1348    encoder.u8(match policy.direction_registry() {
1349        ModelDirectionConversionRegistry::SameTangentPlaneAffineBilinearV1 => 0,
1350    });
1351    encoder.u8(match policy.basis_registry() {
1352        ModelBasisConversionRegistry::ExactSpectralV1 => 0,
1353    });
1354    encoder.u8(match policy.polarization_registry() {
1355        ModelPolarizationConversionRegistry::RealParallelHandsV1 => 0,
1356    });
1357    encoder.u8(match policy.invalid_contributor() {
1358        ModelInvalidContributorPolicy::InvalidateTarget => 0,
1359    });
1360    encoder.u8(match policy.uncovered_target() {
1361        ModelUncoveredTargetPolicy::Invalid => 0,
1362    });
1363    LogicalIdentity::from_sha256(encoder.finish())
1364}

1397        LogicalIdentity::from_sha256(numerics.as_bytes()),

1399        target.identity(),
1400        bounds,
1401        arithmetic_precision,
1402        state_encoding,
1403        support_semantics,
1404        &ModelInputCommitmentIdentity::from_commitment(input),
1405    ))
1406}

1408/// Recompute and validate one persisted model-lifecycle identity.
1409///
1410/// Receipt validation supplies typed, digest-only projections. The model
1411/// contract owner revalidates the complete lifecycle identity, the canonical
1412/// empty-target preparation identity, and the opaque reprojected-input proof;
1413/// duplicate audit fields are never treated as authority.
1414#[allow(clippy::too_many_arguments)]
1415pub fn validate_model_lifecycle_contract_identity(
1416    claimed: LogicalIdentity,
1417    numerics: LogicalIdentity,
1418    reprojection: LogicalIdentity,
1419    target_shape: LogicalIdentity,
1420    bounds: ModelBounds,
1421    arithmetic_precision: NumericPrecision,
1422    state_encoding: ModelStateEncoding,
1423    support_semantics: ModelSupportSemantics,
1424    input: &ModelInputCommitmentIdentity,
1425) -> Result<(), ModelContractError> {
1426    if !identities_are_set(&[claimed, numerics, reprojection, target_shape]) {
1427        return Err(ModelContractError::UnidentifiedInputEvidence);
1428    }
1429    validate_input_commitment_identity(
1430        numerics,
1431        reprojection,
1432        target_shape,
1433        bounds,
1434        arithmetic_precision,
1435        state_encoding,
1436        support_semantics,
1437        input,
1438    )?;
1439    let expected = model_lifecycle_contract_identity(
1440        numerics,
1441        reprojection,
1442        target_shape,
1443        bounds,
1444        arithmetic_precision,
1445        state_encoding,
1446        support_semantics,
1447        input,
1448    );
1449    if claimed == expected {
1450        Ok(())
1451    } else {
1452        Err(ModelContractError::LifecycleContractMismatch)
1453    }
1454}

1456#[allow(clippy::too_many_arguments)]
1457fn validate_input_commitment_identity(
1458    numerics: LogicalIdentity,
1459    reprojection_contract: LogicalIdentity,
1460    target_shape: LogicalIdentity,
1461    bounds: ModelBounds,
1462    arithmetic_precision: NumericPrecision,
1463    state_encoding: ModelStateEncoding,
1464    support_semantics: ModelSupportSemantics,
1465    input: &ModelInputCommitmentIdentity,
1466) -> Result<(), ModelContractError> {
1467    match input {
1468        ModelInputCommitmentIdentity::Empty => Ok(()),
1469        ModelInputCommitmentIdentity::AlignedSeed { source, support } => {
1470            require_input_identities(&[*source, *support])
1471        }
1472        ModelInputCommitmentIdentity::ReprojectedSeed {
1473            source,
1474            source_shape,
1475            preparation_contract,
1476            reprojection,
1477            support,
1478            samples,
1479            stencil,
1480            proof,
1481        } => {
1482            require_input_identities(&[
1483                *source,
1484                *source_shape,
1485                *preparation_contract,
1486                *reprojection,
1487                *support,
1488                *samples,
1489                *stencil,
1490                *proof,
1491            ])?;
1492            let expected_preparation = model_lifecycle_contract_identity(
1493                numerics,
1494                reprojection_contract,
1495                target_shape,
1496                bounds,
1497                arithmetic_precision,
1498                state_encoding,
1499                support_semantics,
1500                &ModelInputCommitmentIdentity::Empty,
1501            );
1502            let expected_reprojection = model_reprojected_seed_mapping_identity(
1503                expected_preparation,
1504                *source_shape,
1505                target_shape,
1506            );
1507            if *preparation_contract != expected_preparation
1508                || *reprojection != expected_reprojection
1509            {
1510                return Err(ModelContractError::InvalidReprojectedSeedProof);
1511            }
1512            Ok(())
1513        }
1514        ModelInputCommitmentIdentity::Generation(generation) => {
1515            require_input_identities(&[*generation])
1516        }
1517    }
1518}

1520fn require_input_identities(identities: &[LogicalIdentity]) -> Result<(), ModelContractError> {
1521    if identities_are_set(identities) {
1522        Ok(())
1523    } else {
1524        Err(ModelContractError::UnidentifiedInputEvidence)
1525    }
1526}

1528#[allow(clippy::too_many_arguments)]
1529fn model_lifecycle_contract_identity(
1530    numerics: LogicalIdentity,
1531    reprojection: LogicalIdentity,
1532    target_shape: LogicalIdentity,
1533    bounds: ModelBounds,
1534    arithmetic_precision: NumericPrecision,
1535    state_encoding: ModelStateEncoding,
1536    support_semantics: ModelSupportSemantics,
1537    input: &ModelInputCommitmentIdentity,
1538) -> LogicalIdentity {
1539    let mut encoder = CanonicalEncoder::new();
1540    encoder.bytes(MODEL_LIFECYCLE_CONTRACT_IDENTITY_DOMAIN);
1541    encoder.u32(MODEL_LIFECYCLE_CONTRACT_IDENTITY_VERSION);
1542    encoder.identity(numerics);
1543    encoder.identity(reprojection);
1544    encoder.identity(target_shape);
1545    bounds.encode(&mut encoder);
1546    encoder.u8(precision_tag(arithmetic_precision));
1547    encoder.u8(match state_encoding {
1548        ModelStateEncoding::CanonicalF64 => 0,
1549    });
1550    encoder.u8(match support_semantics {
1551        ModelSupportSemantics::ExplicitValidity => 0,
1552    });
1553    encode_input_commitment_identity(&mut encoder, input);
1554    LogicalIdentity::from_sha256(encoder.finish())
1555}
1556
1557fn encode_input_commitment_identity(
1558    encoder: &mut CanonicalEncoder,
1559    input: &ModelInputCommitmentIdentity,
1560) {
1561    match input {
1562        ModelInputCommitmentIdentity::Empty => encoder.u8(0),
1563        ModelInputCommitmentIdentity::AlignedSeed { source, support } => {
1564            encoder.u8(1);
1565            encoder.identity(*source);
1566            encoder.identity(*support);
1567        }
1568        ModelInputCommitmentIdentity::ReprojectedSeed {
1569            source,
1570            source_shape,
1571            preparation_contract,
1572            reprojection,
1573            support,
1574            samples,
1575            stencil,
1576            proof,
1577        } => {
1578            encoder.u8(2);
1579            encoder.identity(*source);
1580            encoder.identity(*source_shape);
1581            encoder.identity(*preparation_contract);
1582            encoder.identity(*reprojection);
1583            encoder.identity(*support);
1584            encoder.identity(*samples);
1585            encoder.identity(*stencil);
1586            encoder.identity(*proof);
1587        }
1588        ModelInputCommitmentIdentity::Generation(generation) => {
1589            encoder.u8(3);
1590            encoder.identity(*generation);
1591        }
1592    }
1593}

1595fn model_coefficient_space_identity(space: &ModelCoefficientSpace) -> LogicalIdentity {
1596    let mut encoder = CanonicalEncoder::new();
1597    encoder.bytes(MODEL_SPACE_IDENTITY_DOMAIN);
1598    encoder.u32(MODEL_SPACE_IDENTITY_VERSION);
1599    encoder.digest(space.geometry().as_bytes());
1600    encode_reconstruction_basis(&mut encoder, space.basis());
1601    encoder.usize(space.polarization().coordinates().len());
1602    for coordinate in space.polarization().coordinates() {
1603        encoder.u8(polarization_tag(*coordinate));
1604    }
1605    encoder.u8(match space.inner_product() {
1606        ModelInnerProduct::HermitianEuclidean => 0,
1607    });
1608    LogicalIdentity::from_sha256(encoder.finish())
1609}

1611const fn coefficient_count(basis: ReconstructionBasis) -> usize {
1612    match basis {
1613        ReconstructionBasis::Constant => 1,
1614        ReconstructionBasis::Taylor { terms } => terms,
1615        ReconstructionBasis::ChannelLocal { channels } => channels,
1616    }
1617}

1619const fn precision_tag(precision: NumericPrecision) -> u8 {
1620    match precision {
1621        NumericPrecision::F32 => 0,
1622        NumericPrecision::F64 => 1,
1623    }
1624}
