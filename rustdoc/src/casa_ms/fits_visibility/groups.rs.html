20const SECONDS_PER_DAY: f64 = 86_400.0;
21const MJD_ZERO_JULIAN_DATE: f64 = 2_400_000.5;
22const SPEED_OF_LIGHT: f64 = 299_792_458.0;

24pub(super) struct Axis {

26    pub(super) length: usize,

33impl Axis {
34    fn coordinate(&self, index: usize) -> f64 {
35        self.reference + (index as f64 + 1.0 - self.pixel) * self.increment
36    }

45enum Baseline {

50struct GroupParameters {
51    uvw: [usize; 3],
52    date: Vec<usize>,
53    source: Option<usize>,
54    frequency: Option<usize>,
55    interval: Option<usize>,
56    baseline: Baseline,
57}
58
59impl GroupParameters {
60    fn resolve(parameters: &[Parameter]) -> MsResult<Self> {
61        let position = |name| {
62            parameters
63                .iter()
64                .position(|parameter| parameter.name == name)
65        };
66        let required = |name| {
67            position(name).ok_or_else(|| invalid(format!("Missing UVFITS {name} group parameter")))
68        };
69        let uvw = ["UU", "VV", "WW"].map(|label| {
70            parameters
71                .iter()
72                .position(|parameter| parameter.name.contains(label))
73                .ok_or_else(|| invalid(format!("Missing UVFITS {label} group parameter")))
74        });
75        let [u, v, w] = uvw;
76        let mut date = vec![required("DATE")?];
77        if let Some((index, _)) = parameters
78            .iter()
79            .enumerate()
80            .skip(date[0] + 1)
81            .find(|(_, parameter)| parameter.name == "DATE")
82        {
83            date.push(index);
84        }
85        let baseline = match position("BASELINE") {
86            Some(index) => Baseline::Packed(index),
87            None => Baseline::Separate {
88                antennas: [required("ANTENNA1")?, required("ANTENNA2")?],
89                array: required("SUBARRAY")?,
90            },
91        };
92        Ok(Self {
93            uvw: [u?, v?, w?],
94            date,
95            source: position("SOURCE"),
96            frequency: position("FREQSEL"),
97            interval: position("INTTIM"),
98            baseline,
99        })
100    }
101
102    fn metadata(&self, values: &[f64]) -> MsResult<GroupMetadata> {
103        let date = values[self.date[0]] - MJD_ZERO_JULIAN_DATE;
104        let remainder = self.date.get(1).map_or(0.0, |&index| values[index]);
105        let time = (date + remainder) * SECONDS_PER_DAY;
106        let (antennas, array) = match self.baseline {
107            Baseline::Packed(index) => {
108                let value = values[index] as f32;
109                let packed = value as i32;
110                let array = (100.0 * (f64::from(value) - f64::from(packed) + 0.001)) as i32;
111                ([packed / 256, packed % 256], array)
112            }
113            Baseline::Separate { antennas, array } => (
114                antennas.map(|index| values[index] as i32),
115                values[array] as i32 - 1,
116            ),
117        };
118        let field = self.source.map_or(1, |index| values[index] as i32);
119        let frequency = self.frequency.map_or(1, |index| values[index] as i32);
120        if !time.is_finite()
121            || antennas.iter().any(|&antenna| antenna < 1)
122            || array < 0
123            || field < 1
124            || frequency < 1
125        {
126            return Err(invalid(
127                "Invalid time, antenna, array, source or frequency in UVFITS group",
128            ));
129        }
130        Ok(GroupMetadata {
131            time,
132            antennas: antennas.map(|number| number - 1),
133            array,
134            field: field - 1,
135            frequency: frequency - 1,
136            uvw: self.uvw.map(|index| values[index] * SPEED_OF_LIGHT),
137            interval: self.interval.map(|index| values[index]),
138        })
139    }
140}

152pub(super) struct GroupLayout {
153    pub(super) corr: Vec<Correlation>,
154    pub(super) frequency: Axis,
155    pub(super) direction_deg: [f64; 2],
156    pub(super) windows: usize,
157    corr_stride: usize,
158    window_stride: usize,
159    input_corr_order: Vec<usize>,
160    sample_count: usize,
161}
162
163impl GroupLayout {
164    fn from_header(fits: &mut FitsFile, hdu: &FitsHdu) -> MsResult<Self> {
165        let naxis = hdu.read_key::<i32>(fits, "NAXIS").map_err(invalid)?;
166        if !(6..=999).contains(&naxis) || hdu.read_key::<i32>(fits, "NAXIS1").map_err(invalid)? != 0
167        {
168            return Err(invalid("Invalid UVFITS random-groups axis layout"));
169        }
170        let mut stride = 1usize;
171        let mut axes = Vec::new();
172        for number in 2..=naxis {
173            let length = positive_count(hdu, fits, &format!("NAXIS{number}"))?;
174            let key = |prefix| format!("{prefix}{number}");
175            let name: String = optional_key(hdu, fits, &key("CTYPE"))?.unwrap_or_default();
176            axes.push(Axis {
177                name: name.trim_end().to_owned(),
178                length,
179                reference: optional_key(hdu, fits, &key("CRVAL"))?.unwrap_or_default(),
180                pixel: optional_key(hdu, fits, &key("CRPIX"))?.unwrap_or(1.0),
181                increment: optional_key(hdu, fits, &key("CDELT"))?.unwrap_or(1.0),
182                stride,
183            });
184            stride = stride
185                .checked_mul(length)
186                .ok_or_else(|| invalid("UVFITS axis product overflow"))?;
187        }
188        Self::from_axes(axes, stride, false)
189    }
190
191    fn from_table(fits: &mut FitsFile, hdu: &FitsHdu, column: usize) -> MsResult<Self> {
192        let dim: String = if let Some(dim) = optional_key(hdu, fits, &format!("TDIM{column}"))? {
193            dim
194        } else {
195            hdu.read_key(fits, "TDIM").map_err(invalid)?
196        };
197        let mut axes = Vec::new();
198        let mut stride = 1usize;
199        for (number, length) in (1..).zip(dim.trim_matches(['(', ')']).split(',')) {
200            let length = length.trim().parse::<usize>().map_err(invalid)?;
201            if length == 0 {
202                return Err(invalid("Zero axis length in UVFITS visibility table"));
203            }
204            let key = |name| format!("{number}{name}{column}");
205            axes.push(Axis {
206                name: hdu
207                    .read_key::<String>(fits, &key("CTYP"))
208                    .map_err(invalid)?
209                    .trim_end()
210                    .to_owned(),
211                length,
212                reference: hdu.read_key(fits, &key("CRVL")).map_err(invalid)?,
213                pixel: hdu.read_key(fits, &key("CRPX")).map_err(invalid)?,
214                increment: hdu.read_key(fits, &key("CDLT")).map_err(invalid)?,
215                stride,
216            });
217            stride = stride
218                .checked_mul(length)
219                .ok_or_else(|| invalid("UVFITS axis product overflow"))?;
220        }
221        Self::from_axes(axes, stride, true)
222    }
223
224    fn from_axes(mut axes: Vec<Axis>, stride: usize, binary: bool) -> MsResult<Self> {
225        let direction_deg = ["RA", "DEC"].map(|name| {
226            axes.iter()
227                .find(|axis| {
228                    axis.name == name
229                        || axis
230                            .name
231                            .strip_prefix(name)
232                            .is_some_and(|suffix| suffix.starts_with('-'))
233                })
234                .map(|axis| axis.reference)
235                .ok_or_else(|| invalid(format!("UVFITS has no {name} axis")))
236        });
237        let [ra, dec] = direction_deg;
238        let (windows, window_stride) = axes
239            .iter()
240            .find(|axis| axis.name == "IF")
241            .map_or((1, 0), |axis| (axis.length, axis.stride));
242        let stokes = axes
243            .iter()
244            .find(|axis| axis.name == "STOKES")
245            .ok_or_else(|| invalid("UVFITS has no STOKES axis"))?;
246        let (corr, input_corr_order) = correlation_order(stokes, binary)?;
247        let corr_stride = stokes.stride;
248        let frequency = axes
249            .iter()
250            .position(|axis| axis.name == "FREQ")
251            .map(|index| axes.remove(index))
252            .ok_or_else(|| invalid("UVFITS has no FREQ axis"))?;
253        if !axes
254            .iter()
255            .any(|axis| axis.name == "COMPLEX" && axis.length == 3 && axis.stride == 1)
256            || 3_usize
257                .checked_mul(corr.len())
258                .and_then(|n| n.checked_mul(frequency.length))
259                .and_then(|n| n.checked_mul(windows))
260                != Some(stride)
261        {
262            return Err(invalid(
263                "UVFITS visibility samples require real, imaginary and weight components",
264            ));
265        }
266        Ok(Self {
267            corr,
268            frequency,
269            direction_deg: [ra?, dec?],
270            windows,
271            corr_stride,
272            window_stride,
273            input_corr_order,
274            sample_count: stride,
275        })
276    }

278    pub(super) fn visibility(&self, samples: &[f32], window: usize) -> GroupVisibility {
279        let correlations = self.corr.len();
280        let channels = self.frequency.length;
281        let shape = (correlations, channels).f();
282        let length = correlations * channels;
283        let mut data = Vec::with_capacity(length);
284        let mut weight_spectrum = Vec::with_capacity(length);
285        let mut flag = Vec::with_capacity(length);
286        let mut weight = Array1::<f32>::zeros(correlations);
287        let window_offset = window * self.window_stride;
288        for channel in 0..channels {
289            let base = window_offset + channel * self.frequency.stride;
290            for (correlation, &input) in self.input_corr_order.iter().enumerate() {
291                let offset = base + input * self.corr_stride;
292                let value = &samples[offset..offset + 3];
293                let unsigned_weight = value[2].abs();
294                data.push(Complex32::new(value[0], value[1]));
295                flag.push(value[2] <= 0.0);
296                weight_spectrum.push(unsigned_weight);
297                weight[correlation] += unsigned_weight;
298            }
299        }
300        let data = Array2::from_shape_vec(shape, data).expect("one visibility per cell");
301        let flag = Array2::from_shape_vec(shape, flag).expect("one flag per cell");
302        let weight_spectrum =
303            Array2::from_shape_vec(shape, weight_spectrum).expect("one weight per cell");
304        let sigma = weight.mapv(|value| {
305            if value > 0.0 {
306                f64::from(value).recip().sqrt() as f32
307            } else {
308                0.0
309            }
310        });
311        GroupVisibility {
312            data,
313            weight_spectrum,
314            weight,
315            sigma,
316            flag,
317        }
318    }
319}

329pub(super) struct RandomGroups {
330    pub(super) layout: GroupLayout,
331    pub(super) count: usize,
332    parameters: Vec<Parameter>,
333    indices: GroupParameters,
334    values: Vec<f64>,
335    samples: Vec<f32>,
336}
337
338impl RandomGroups {
339    pub(super) fn read_header(fits: &mut FitsFile) -> MsResult<Self> {
340        let primary = fits.primary_hdu().map_err(invalid)?;
341        if !optional_key::<bool>(&primary, fits, "GROUPS")?.unwrap_or(false) {
342            return Err(invalid("UVFITS primary HDU has no random groups"));
343        }
344        let bitpix: i32 = primary.read_key(fits, "BITPIX").map_err(invalid)?;
345        if !matches!(bitpix, 16 | 32 | -32) {
346            return Err(invalid(format!(
347                "Unhandled UVFITS random groups BITPIX {bitpix}"
348            )));
349        }
350        let count = positive_count(&primary, fits, "GCOUNT")?;
351        let layout = GroupLayout::from_header(fits, &primary)?;
352        let parameter_count = positive_count(&primary, fits, "PCOUNT")?;
353        let sample_bytes = (bitpix.unsigned_abs() / 8) as usize;
354        let payload = parameter_count
355            .checked_add(layout.sample_count)
356            .and_then(|samples| samples.checked_mul(count))
357            .and_then(|samples| samples.checked_mul(sample_bytes))
358            .ok_or_else(|| invalid("UVFITS random-groups payload size overflow"))?;
359        verify_payload(fits, payload)?;
360        let parameters = (1..=parameter_count)
361            .map(|index| {
362                let key = |prefix| format!("{prefix}{index}");
363                let name: String = primary.read_key(fits, &key("PTYPE")).map_err(invalid)?;
364                Ok(Parameter {
365                    name: name.trim_end().to_owned(),
366                    scale: optional_key(&primary, fits, &key("PSCAL"))?.unwrap_or(1.0),
367                    zero: optional_key(&primary, fits, &key("PZERO"))?.unwrap_or_default(),
368                })
369            })
370            .collect::<MsResult<Vec<_>>>()?;
371        let indices = GroupParameters::resolve(&parameters)?;
372        let mut samples = Vec::new();
373        samples
374            .try_reserve_exact(layout.sample_count)
375            .map_err(invalid)?;
376        samples.resize(layout.sample_count, 0.0);
377        Ok(Self {
378            layout,
379            count,
380            parameters,
381            indices,
382            values: vec![0.0; parameter_count],
383            samples,
384        })
385    }
386
387    pub(super) fn read(
388        &mut self,
389        fits: &mut FitsFile,
390        index: usize,
391    ) -> MsResult<(GroupMetadata, &GroupLayout, &[f32])> {
392        let group = c_long::try_from(index + 1).map_err(invalid)?;
393        let parameter_count = c_long::try_from(self.values.len()).map_err(invalid)?;
394        let sample_count = i64::try_from(self.samples.len()).map_err(invalid)?;
395        let mut status = 0;
396        let mut any_null = 0;
397        // SAFETY: FitsFile owns the open, exclusive handle. Both arrays have
398        // the lengths passed to CFITSIO, and the primary HDU is selected.
399        unsafe {
400            let handle = fits.as_raw();
401            sys::ffggpd(
402                handle,
403                group,
404                1,
405                parameter_count,
406                self.values.as_mut_ptr(),
407                &mut status,
408            );
409            sys::ffgpve(
410                handle,
411                group,
412                1,
413                sample_count,
414                0.0,
415                self.samples.as_mut_ptr(),
416                &mut any_null,
417                &mut status,
418            );
419        }
420        errors::check_status(status).map_err(invalid)?;
421        // CFITSIO reads the random-group parameters without individual
422        // PSCALn/PZEROn. Keep them separate from BSCALE/BZERO for the array.
423        for (value, parameter) in self.values.iter_mut().zip(&self.parameters) {
424            *value = *value * parameter.scale + parameter.zero;
425        }
426        Ok((
427            self.indices.metadata(&self.values)?,
428            &self.layout,
429            &self.samples,
430        ))
431    }
432}

434pub(super) enum VisibilityGroups {
435    Random(Box<RandomGroups>),
436    Binary(Box<BinaryGroups>),
437}
438
439impl VisibilityGroups {
440    pub(super) fn open(fits: &mut FitsFile) -> MsResult<Self> {
441        let primary = fits.primary_hdu().map_err(invalid)?;
442        let marker = optional_key::<i64>(&primary, fits, "NAXIS1")? == Some(777_777_701)
443            && optional_key::<i64>(&primary, fits, "NAXIS2")? == Some(0);
444        if marker {
445            Ok(Self::Binary(Box::new(BinaryGroups::open(fits)?)))
446        } else {
447            Ok(Self::Random(Box::new(RandomGroups::read_header(fits)?)))
448        }
449    }
450
451    pub(super) fn layout(&self) -> &GroupLayout {
452        match self {
453            Self::Random(groups) => &groups.layout,
454            Self::Binary(groups) => &groups.layout,
455        }
456    }
457
458    pub(super) fn count(&self) -> usize {
459        match self {
460            Self::Random(groups) => groups.count,
461            Self::Binary(groups) => groups.rows,
462        }
463    }
464
465    pub(super) fn binary_hdu(&self) -> Option<usize> {
466        match self {
467            Self::Random(_) => None,
468            Self::Binary(groups) => Some(groups.hdu.number),
469        }
470    }
471
472    pub(super) fn read(
473        &mut self,
474        fits: &mut FitsFile,
475        index: usize,
476    ) -> MsResult<(GroupMetadata, &GroupLayout, &[f32])> {
477        match self {
478            Self::Random(groups) => groups.read(fits, index),
479            Self::Binary(groups) => groups.read(fits, index),
480        }
481    }
482}

484pub(super) struct BinaryGroups {
485    hdu: FitsHdu,
486    layout: GroupLayout,
487    rows: usize,
488    columns: Vec<String>,
489    parameters: GroupParameters,
490    values: Vec<f64>,
491    samples: Vec<f32>,
492    block: Range<usize>,
493    block_columns: Vec<Vec<f64>>,
494}
495
496impl BinaryGroups {
497    fn open(fits: &mut FitsFile) -> MsResult<Self> {
498        let count = fits.num_hdus().map_err(invalid)?;
499        for index in 1..count {
500            if !binary_extension_name(fits, index)?.is_some_and(|name| name.contains("UV")) {
501                continue;
502            }
503            let hdu = binary_extension(fits, index)?;
504            let HduInfo::TableInfo {
505                column_descriptions,
506                num_rows,
507            } = &hdu.info
508            else {
509                continue;
510            };
511            let Some(column) = column_descriptions
512                .iter()
513                .position(|col| col.name == "VISIBILITIES")
514            else {
515                continue;
516            };
517            let layout = GroupLayout::from_table(fits, &hdu, column + 1)?;
518            if column_descriptions[column].data_type.repeat != layout.sample_count {
519                return Err(invalid(
520                    "UVFITS binary visibility row does not match its TDIM",
521                ));
522            }
523            let rows = *num_rows;
524            let keys = [
525                "BASELINE", "ANTENNA1", "ANTENNA2", "SUBARRAY", "DATE", "SOURCE", "FREQSEL",
526            ];
527            let columns = column_descriptions
528                .iter()
529                .filter(|col| {
530                    let name = col.name.as_str();
531                    keys.contains(&name) || ["UU", "VV", "WW"].iter().any(|key| name.contains(key))
532                })
533                .map(|col| {
534                    if col.data_type.repeat == 1 {
535                        Ok(col.name.clone())
536                    } else {
537                        Err(invalid(format!(
538                            "UVFITS column {} must be scalar",
539                            col.name
540                        )))
541                    }
542                })
543                .collect::<MsResult<Vec<_>>>()?;
544            let definitions = columns
545                .iter()
546                .map(|name| Parameter {
547                    name: name.clone(),
548                    scale: 1.0,
549                    zero: 0.0,
550                })
551                .collect::<Vec<_>>();
552            let parameters = GroupParameters::resolve(&definitions)?;
553            let values = vec![0.0; columns.len()];
554            return Ok(Self {
555                hdu,
556                layout,
557                rows,
558                columns,
559                parameters,
560                values,
561                samples: Vec::new(),
562                block: 0..0,
563                block_columns: Vec::new(),
564            });
565        }
566        Err(invalid("Cannot find a UVFITS binary visibility table"))
567    }
568
569    fn read(
570        &mut self,
571        fits: &mut FitsFile,
572        index: usize,
573    ) -> MsResult<(GroupMetadata, &GroupLayout, &[f32])> {
574        if !self.block.contains(&index) {
575            let count = (65_536 / self.layout.sample_count).clamp(1, 1024);
576            let block = index..self.rows.min(index + count);
577            self.samples = self
578                .hdu
579                .read_col_range(fits, "VISIBILITIES", &block)
580                .map_err(invalid)?;
581            if self.samples.len() != block.len() * self.layout.sample_count {
582                return Err(invalid(
583                    "UVFITS binary visibility row does not match its TDIM",
584                ));
585            }
586            self.block_columns = self
587                .columns
588                .iter()
589                .map(|column| {
590                    self.hdu
591                        .read_col_range(fits, column, &block)
592                        .map_err(invalid)
593                })
594                .collect::<MsResult<Vec<_>>>()?;
595            self.block = block;
596        }
597        let offset = index - self.block.start;
598        for (value, column) in self.values.iter_mut().zip(&self.block_columns) {
599            *value = *column
600                .get(offset)
601                .ok_or_else(|| invalid("Missing UVFITS binary row parameters"))?;
602        }
603        let row = offset * self.layout.sample_count;
604        let samples = &self.samples[row..row + self.layout.sample_count];
605        Ok((
606            self.parameters.metadata(&self.values)?,
607            &self.layout,
608            samples,
609        ))
610    }
611}

613fn correlation_order(stokes: &Axis, binary: bool) -> MsResult<(Vec<Correlation>, Vec<usize>)> {
614    if stokes.length > 12 {
615        return Err(invalid("Too many UVFITS Stokes values"));
616    }
617    let inputs = (0..stokes.length)
618        .map(|index| {
619            let code = (stokes.coordinate(index) + 0.5).floor() as i32;
620            let correlation = fits_correlation(code, binary)?;
621            Ok((correlation, index))
622        })
623        .collect::<MsResult<Vec<_>>>()?;
624    let mut inputs = inputs;
625    inputs.sort_unstable_by_key(|&(correlation, _)| correlation.casacore_code());
626    if inputs.windows(2).any(|pair| pair[0].0 == pair[1].0) {
627        return Err(invalid("Repeated UVFITS Stokes values"));
628    }
629    Ok(inputs.into_iter().unzip())
630}

632#[doc(hidden)]
633pub fn fits_correlation(code: i32, allow_stokes: bool) -> MsResult<Correlation> {
634    use Correlation::{I, LL, LR, Q, RL, RR, U, V, XX, XY, YX, YY};
635    Ok(match code {
636        -1 => RR,
637        -2 => LL,
638        -3 => RL,
639        -4 => LR,
640        -5 => XX,
641        -6 => YY,
642        -7 => XY,
643        -8 => YX,
644        1 if allow_stokes => I,
645        2 if allow_stokes => Q,
646        3 if allow_stokes => U,
647        4 if allow_stokes => V,
648        1..=4 => {
649            return Err(invalid(
650                "UVFITS Stokes parameters cannot be decomposed into correlations",
651            ));
652        }
653        _ => return Err(invalid(format!("Unknown UVFITS correlation type {code}"))),
654    })
655}

658pub fn binary_extension_name(fits: &mut FitsFile, index: usize) -> MsResult<Option<String>> {
659    let number = i32::try_from(index + 1).map_err(invalid)?;
660    let mut typ = 0;
661    let mut status = 0;
662    // SAFETY: the live handle belongs to fits and CFITSIO writes one int to each output.
663    unsafe { sys::ffmahd(fits.as_raw(), number, &mut typ, &mut status) };
664    errors::check_status(status).map_err(invalid)?;
665    if typ != sys::BINARY_TBL as i32 {
666        return Ok(None);
667    }
668    raw_text_key(fits, c"EXTNAME")
669}

671/// Read a recognized binary table after checking the descriptor types supported by fitsio.
672#[doc(hidden)]
673pub fn binary_extension(fits: &mut FitsFile, index: usize) -> MsResult<FitsHdu> {
674    let mut fields = 0;
675    let mut status = 0;
676    // SAFETY: the live FITS handle is positioned on the binary extension by the caller.
677    unsafe { sys::ffgncl(fits.as_raw(), &mut fields, &mut status) };
678    errors::check_status(status).map_err(invalid)?;
679    for column in 1..=fields {
680        let key = CString::new(format!("TFORM{column}")).map_err(invalid)?;
681        let form = raw_text_key(fits, &key)?.unwrap_or_default();
682        let typ = form
683            .trim_start_matches(|c: char| c.is_ascii_digit())
684            .bytes()
685            .next();
686        if !typ.is_some_and(|code| b"LXBSEJDIKAUVW".contains(&code)) {
687            return Err(invalid(format!(
688                "Unsupported UVFITS binary column format {form:?}"
689            )));
690        }
691    }
692    let hdu = fits.hdu(index).map_err(invalid)?;
693    let width = hdu.read_key::<i64>(fits, "NAXIS1").map_err(invalid)?;
694    let rows = hdu.read_key::<i64>(fits, "NAXIS2").map_err(invalid)?;
695    let width = usize::try_from(width).map_err(invalid)?;
696    let rows = usize::try_from(rows).map_err(invalid)?;
697    let bytes = rows
698        .checked_mul(width)
699        .ok_or_else(|| invalid("UVFITS table byte size overflow"))?;
700    verify_payload(fits, bytes)?;
701    Ok(hdu)
702}
703
704fn verify_payload(fits: &mut FitsFile, bytes: usize) -> MsResult<()> {
705    let Some(last_byte) = bytes.checked_sub(1) else {
706        return Ok(());
707    };
708    let offset = i64::try_from(last_byte).map_err(invalid)?;
709    let mut value = 0_u8;
710    let mut status = 0;
711    // SAFETY: the FITS handle has the desired HDU selected. The function reads
712    // exactly one byte at the checked data-relative offset into value.
713    unsafe {
714        sys::ffgextn(
715            fits.as_raw(),
716            offset,
717            1,
718            (&mut value as *mut u8).cast(),
719            &mut status,
720        )
721    };
722    errors::check_status(status)
723        .map_err(|error| invalid(format!("Incomplete UVFITS payload: {error}")))
724}

726fn raw_text_key(fits: &mut FitsFile, key: &CStr) -> MsResult<Option<String>> {
727    let mut value = [0 as c_char; sys::FLEN_VALUE as usize];
728    let mut status = 0;
729    // SAFETY: CFITSIO expects a NUL-terminated key and a FLEN_VALUE text
730    // buffer. It terminates the returned value and permits omitting comments.
731    unsafe {
732        sys::ffgkys(
733            fits.as_raw(),
734            key.as_ptr(),
735            value.as_mut_ptr(),
736            ptr::null_mut(),
737            &mut status,
738        );
739    }
740    if status == sys::KEY_NO_EXIST as i32 {
741        return Ok(None);
742    }
743    errors::check_status(status).map_err(invalid)?;
744    // SAFETY: the buffer contains a terminated CFITSIO string.
745    Ok(Some(
746        unsafe { CStr::from_ptr(value.as_ptr()) }
747            .to_string_lossy()
748            .trim()
749            .to_owned(),
750    ))
751}

753fn positive_count(hdu: &FitsHdu, fits: &mut FitsFile, key: &str) -> MsResult<usize> {
754    let value: i64 = hdu.read_key(fits, key).map_err(invalid)?;
755    usize::try_from(value)
756        .ok()
757        .filter(|&value| value > 0)
758        .ok_or_else(|| invalid(format!("Invalid UVFITS {key}: {value}")))
759}

761#[doc(hidden)]
762pub fn optional_key<T: ReadsKey>(
763    hdu: &FitsHdu,
764    fits: &mut FitsFile,
765    key: &str,
766) -> MsResult<Option<T>> {
767    match hdu.read_key(fits, key) {
768        Ok(value) => Ok(Some(value)),
769        Err(errors::Error::Fits(error)) if error.status == sys::KEY_NO_EXIST as i32 => Ok(None),
770        Err(error) => Err(invalid(error)),
771    }
772}

774#[doc(hidden)]
775pub fn open_file(path: &Path) -> MsResult<FitsFile> {
776    let name = path
777        .to_str()
778        .ok_or_else(|| invalid("UVFITS path must be UTF-8"))?;
779    let name = CString::new(name).map_err(invalid)?;
780    let mut handle = ptr::null_mut();
781    let mut status = 0;
782    // SAFETY: For a disk-only open CFITSIO takes a live C path and initializes
783    // the output handle on success. FITS paths may contain literal '+' signs.
784    unsafe { sys::ffdkopn(&mut handle, name.as_ptr(), 0, &mut status) };
785    errors::check_status(status).map_err(invalid)?;
786    // SAFETY: Transfer the non-null read-only handle to its owning wrapper.
787    match unsafe { FitsFile::from_raw(handle, FileOpenMode::READONLY) } {
788        Ok(fits) => Ok(fits),
789        Err(error) => {
790            // SAFETY: from_raw failed before transferring ownership.
791            unsafe { sys::ffclos(handle, &mut status) };
792            Err(invalid(error))
793        }
794    }
795}
