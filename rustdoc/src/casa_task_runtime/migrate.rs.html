2//! A saved profile's explicit overrides carried through the surface's
3//! migrations to the current contract (ADR-0006).
4//!
5//! Each migration rewrites the values the profile set. A value it omitted
6//! adopts the current default, and when the migration lists that default
7//! as changed, the profile is warned. A value counts as set when the
8//! profile gave it under its old name, or when a step fills it from other
9//! explicit values because the profile left it unset (the old gaincal's
10//! `gain_model_source='point'` without `smodel` is `smodel=[1, 0, 0, 0]`):
11//! neither adopts the default.
12

17};

20use crate::normalize::normalize_quantity_unit;
21use crate::profile::{ParameterProfile, ProfileError};

23/// The profile's explicit values at the bundle's contract, with a warning
24/// for each changed default the profile omitted and one per migration run.
25pub(crate) fn migrate_parameters(
26    profile: &ParameterProfile,
27    bundle: &SurfaceContractBundle,
28) -> Result<(BTreeMap<String, ParameterValue>, Vec<Diagnostic>), ProfileError> {
29    let mut version = profile.header.contract;
30    let mut parameters = profile.parameters.clone();
31    let mut diagnostics = Vec::new();
32    while version < bundle.surface.contract_version() {
33        let migration = bundle
34            .surface
35            .migrations()
36            .iter()
37            .find(|migration| migration.from_contract == version)
38            .ok_or_else(|| {
39                ProfileError::Contract(format!(
40                    "surface {} has no migration from contract {} to {}",
41                    bundle.surface.id(),
42                    version,
43                    bundle.surface.contract_version()
44                ))
45            })?;
46        let explicit = migrate_once(migration, &mut parameters)?;
47        for name in &migration.changed_defaults {
48            if !explicit.contains(name) {
49                diagnostics.push(
50                    Diagnostic::warning(
51                        DiagnosticCode::DefaultChanged,
52                        format!(
53                            "default for {name:?} changed after contract {}; omitted value adopts the current default",
54                            migration.from_contract
55                        ),
56                    )
57                    .for_parameter(name.clone()),
58                );
59            }
60        }
61        diagnostics.push(Diagnostic::warning(
62            DiagnosticCode::Migrated,
63            format!(
64                "migrated surface {} contract {} to {}",
65                bundle.surface.id(),
66                migration.from_contract,
67                migration.to_contract
68            ),
69        ));
70        version = migration.to_contract;
71    }
72    Ok((parameters, diagnostics))
73}

75/// Run one migration's steps over `parameters`, and return the names whose
76/// values the profile set, by the names they have after the migration.
77fn migrate_once(
78    migration: &SurfaceMigration,
79    parameters: &mut BTreeMap<String, ParameterValue>,
80) -> Result<BTreeSet<String>, ProfileError> {
81    let mut explicit = parameters.keys().cloned().collect::<BTreeSet<_>>();
82    for step in &migration.steps {
83        match step {
84            MigrationStep::Rename { from, to } if explicit.contains(from) => {
85                explicit.insert(to.clone());
86            }
87            MigrationStep::Set {
88                parameter, when, ..
89            } if step.fills_when_unset(parameter) && when.holds_for_overrides(parameters) => {
90                explicit.insert(parameter.clone());
91            }
92            _ => {}
93        }
94        apply_migration_step(step, parameters)?;
95    }
96    Ok(explicit)
97}

99fn apply_migration_step(
100    step: &MigrationStep,
101    parameters: &mut BTreeMap<String, ParameterValue>,
102) -> Result<(), ProfileError> {
103    match step {
104        MigrationStep::Rename { from, to } => {
105            if let Some(value) = parameters.remove(from)
106                && parameters.insert(to.clone(), value).is_some()
107            {
108                return Err(ProfileError::Contract(format!(
109                    "migration rename {from:?} -> {to:?} collides with an explicit value"
110                )));
111            }
112        }
113        MigrationStep::Remove { parameter } => {
114            parameters.remove(parameter);
115        }
116        MigrationStep::ReplaceValue {
117            parameter,
118            from,
119            to,
120        } => {
121            if parameters.get(parameter) == Some(from) {
122                parameters.insert(parameter.clone(), to.clone());
123            }
124        }
125        MigrationStep::Transform {
126            parameter,
127            transform,
128        } => {
129            if let Some(value) = parameters.remove(parameter) {
130                parameters.insert(
131                    parameter.clone(),
132                    transform_value(value, transform, parameter)?,
133                );
134            }
135        }
136        MigrationStep::Set {
137            parameter,
138            value,
139            when,
140        } => {
141            if when.holds_for_overrides(parameters) {
142                parameters.insert(parameter.clone(), value.clone());
143            }
144        }
145        MigrationStep::Broadcast {
146            parameter,
147            to_length_of,
148        } => crate::migration_list::broadcast(parameters, parameter, to_length_of),
149    }
150    Ok(())
151}

153fn transform_value(
154    value: ParameterValue,
155    transform: &MigrationTransform,
156    parameter: &str,
157) -> Result<ParameterValue, ProfileError> {
158    match transform {
159        MigrationTransform::ScalarToArray { length } => {
160            Ok(ParameterValue::Array(vec![value; *length]))
161        }
162        MigrationTransform::SingletonArrayToScalar => match value {
163            ParameterValue::Array(mut values) if values.len() == 1 => Ok(values.remove(0)),
164            other => Err(ProfileError::Contract(format!(
165                "migration expected a singleton array for {parameter:?}, got {other:?}"
166            ))),
167        },
168        MigrationTransform::IntegerToString => match value {
169            ParameterValue::Integer(value) => Ok(ParameterValue::String(value.to_string())),
170            other => Err(ProfileError::Contract(format!(
171                "migration expected an integer for {parameter:?}, got {other:?}"
172            ))),
173        },
174        MigrationTransform::QuantityUnit { canonical_unit } => {
175            normalize_quantity_unit(value, canonical_unit).map_err(ProfileError::from)
176        }
177        MigrationTransform::Lowercase => match value {
178            ParameterValue::String(value) => Ok(ParameterValue::String(value.to_ascii_lowercase())),
179            other => Err(ProfileError::Contract(format!(
180                "migration expected a string for {parameter:?}, got {other:?}"
181            ))),
182        },
183        MigrationTransform::Trim => match value {
184            ParameterValue::String(value) => Ok(ParameterValue::String(value.trim().to_string())),
185            other => Err(ProfileError::Contract(format!(
186                "migration expected a string for {parameter:?}, got {other:?}"
187            ))),
188        },
189        MigrationTransform::NumberList => crate::normalize::number_list(value, parameter),
190        MigrationTransform::TextList { separator, element } => {
191            crate::migration_list::text_list(value, separator, *element, parameter)
192        }
193    }
194}
