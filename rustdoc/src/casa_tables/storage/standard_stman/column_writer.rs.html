2//! The StandardStMan file writer for any mix of columns, with bounded
3//! memory.
4//!
5//! The columns are written one after another, each over all its rows, which
6//! fixes where their shared storage goes: indirect arrays and records are
7//! appended to the array file (`table.fNi`) and long strings to the string
8//! buckets in that column-major order. Each column's slice of a data bucket
9//! is written at its place in the file as soon as it is full, and each
10//! string bucket once the next one starts; the index is generated as it is
11//! streamed into its buckets ([`super::index`]). So the writer holds one
12//! slice, one string bucket, one index bucket and the array file's buffer
13//! whatever the row count. The file is laid out as casacore's `SSMBase`
14//! reads it: the 512-byte header, the data buckets, the string buckets,
15//! then the index buckets.
16

33/// The cells a StandardStMan file is written from, by column and row.
34pub(super) trait SsmCells {
35    /// Rows to write.
36    fn row_count(&self) -> usize;
37
38    /// The cell of `column` in `row`, `None` when it is undefined.
39    fn cell(&self, column: usize, row: usize) -> Result<Option<Cow<'_, Value>>, StorageError>;
40}
41
42impl SsmCells for ColumnSources<'_> {
43    fn row_count(&self) -> usize {
44        ColumnSources::row_count(self)
45    }
46
47    fn cell(&self, column: usize, row: usize) -> Result<Option<Cow<'_, Value>>, StorageError> {
48        ColumnSources::cell(self, column, row)
49    }
50}

52/// Row records as cells: each column's field, found at its index in the
53/// record when `field_indices` gives one and by name otherwise.
54pub(super) struct RowCells<'r> {
55    rows: &'r [RecordValue],
56    field_indices: Option<&'r [usize]>,
57    names: Vec<&'r str>,
58}
59
60impl<'r> RowCells<'r> {
61    pub(super) fn new(
62        rows: &'r [RecordValue],
63        field_indices: Option<&'r [usize]>,
64        col_descs: &'r [ColumnDescContents],
65    ) -> Self {
66        Self {
67            rows,
68            field_indices,
69            names: col_descs
70                .iter()
71                .map(|desc| desc.col_name.as_str())
72                .collect(),
73        }
74    }
75}
76
77impl SsmCells for RowCells<'_> {
78    fn row_count(&self) -> usize {
79        self.rows.len()
80    }
81
82    fn cell(&self, column: usize, row: usize) -> Result<Option<Cow<'_, Value>>, StorageError> {
83        let name = self.names[column];
84        let record = self.rows.get(row).ok_or_else(|| {
85            StorageError::FormatMismatch(format!(
86                "SSM row writer missing row {row} for column {name}"
87            ))
88        })?;
89        let fields = record.fields();
90        let field = self
91            .field_indices
92            .and_then(|indices| indices.get(column))
93            .and_then(|&index| fields.get(index))
94            .filter(|field| field.name == name)
95            .or_else(|| fields.iter().find(|field| field.name == name));
96        Ok(field.map(|field| Cow::Borrowed(&field.value)))
97    }
98}

100/// Bits a row of each column takes in a data bucket: the bits of its
101/// elements; 12 bytes for a variable-length string's reference; 8 bytes for
102/// an indirect cell's array-file offset.
103fn column_bits(col_descs: &[ColumnDescContents]) -> Vec<usize> {
104    col_descs
105        .iter()
106        .map(|desc| {
107            if is_ssm_fixed_string(desc) {
108                desc.max_length as usize * ssm_cell_element_count(desc) * 8
109            } else if is_ssm_variable_string(desc) {
110                96
111            } else if is_ssm_indirect(desc) {
112                64
113            } else {
114                ssm_cell_element_count(desc) * canonical_element_size(desc.data_type).1
115            }
116        })
117        .collect()
118}

120/// Write the finished string buckets, and with `all` the open one too, at
121/// their places from bucket `first_bucket` on, and release them.
122fn write_string_buckets(
123    strings: &mut StringBuckets,
124    file: &mut File,
125    first_bucket: usize,
126    bucket_bytes: usize,
127    all: bool,
128) -> Result<(), StorageError> {
129    strings.release(all, |number, bucket| {
130        file.seek(SeekFrom::Start(bucket_offset(
131            first_bucket + number,
132            bucket_bytes,
133        )))?;
134        file.write_all(bucket)?;
135        Ok(())
136    })
137}

139/// Write a StandardStMan data file from `cells`, column by column, and
140/// return the DM data blob for `table.dat`.
141///
142/// The bytes are those of a writer that assembles every bucket in memory
143/// and writes the columns in order.
144pub(super) fn write_ssm_file_columns(
145    file_path: &Path,
146    col_descs: &[ColumnDescContents],
147    cells: &impl SsmCells,
148    big_endian: bool,
149) -> Result<Vec<u8>, StorageError> {
150    let nrrow = cells.row_count();
151    let ncol = col_descs.len();
152    // The shared array file of indirect columns (SSM's version 0, without
153    // reference counts).
154    let mut array_writer = if col_descs.iter().any(is_ssm_indirect) {
155        let mut path = file_path.as_os_str().to_os_string();
156        path.push("i");
157        Some(StManArrayFileWriter::create(
158            &PathBuf::from(path),
159            big_endian,
160            0,
161        )?)
162    } else {
163        None
164    };
165    let bits = column_bits(col_descs);
166    let (rows_per_bucket, bucket_size) = new_ssm_bucket_layout(&bits, nrrow);
167    let bucket_rows = rows_per_bucket as usize;
168    let bucket_bytes = bucket_size as usize;
169    let mut column_offsets = Vec::with_capacity(ncol);
170    let mut offset = 0u32;
171    for &column_bits in &bits {
172        column_offsets.push(offset);
173        offset += (bucket_rows * column_bits).div_ceil(8) as u32;
174    }
175    let nr_data_buckets = nrrow.div_ceil(bucket_rows);
176
177    let mut file = File::create(file_path)?;
178    let mut strings = StringBuckets::default();
179    let mut slice = Vec::new();
180    for (column, desc) in col_descs.iter().enumerate() {
181        let indirect = is_ssm_indirect(desc);
182        let nrelem = if indirect {
183            1
184        } else {
185            ssm_cell_element_count(desc)
186        };
187        for bucket in 0..nr_data_buckets {
188            slice.clear();
189            slice.resize((bucket_rows * bits[column]).div_ceil(8), 0);
190            let first_row = bucket * bucket_rows;
191            for row in first_row..nrrow.min(first_row + bucket_rows) {
192                let value = cells.cell(column, row)?;
193                let row_in_bucket = row - first_row;
194                if indirect {
195                    let writer = array_writer
196                        .as_mut()
197                        .expect("indirect columns have an array file");
198                    let offset = write_indirect_cell(writer, desc, value.as_deref())?;
199                    let cell = &mut slice[row_in_bucket * 8..];
200                    if big_endian {
201                        write_i64_be(cell, offset);
202                    } else {
203                        write_i64_le(cell, offset);
204                    }
205                } else {
206                    write_cell_to_bucket(
207                        &mut slice,
208                        0,
209                        row_in_bucket,
210                        desc,
211                        desc.data_type,
212                        nrelem,
213                        value.as_deref(),
214                        &mut strings,
215                        nr_data_buckets,
216                        bucket_bytes,
217                        big_endian,
218                    );
219                    write_string_buckets(
220                        &mut strings,
221                        &mut file,
222                        nr_data_buckets,
223                        bucket_bytes,
224                        false,
225                    )?;
226                }
227            }
228            file.seek(SeekFrom::Start(
229                bucket_offset(bucket, bucket_bytes) + u64::from(column_offsets[column]),
230            ))?;
231            file.write_all(&slice)?;
232        }
233    }
234    write_string_buckets(&mut strings, &mut file, nr_data_buckets, bucket_bytes, true)?;
235    if let Some(writer) = array_writer.as_mut() {
236        writer.finish()?;
237    }
238    let nr_string_buckets = strings.len();
239    write_index_and_header(
240        &mut file,
241        IndexLayout {
242            nrrow,
243            ncol,
244            rows_per_bucket,
245            bucket_size,
246            nr_data_buckets,
247            nr_string_buckets,
248        },
249        big_endian,
250    )?;
251    serialize_ssm_dm_blob("SSM", &column_offsets, ncol)
252}

254/// Append an indirect cell to the array file and return its offset there:
255/// a record as its serialized bytes, an array as itself, and 0 for an
256/// undefined cell.
257fn write_indirect_cell(
258    writer: &mut StManArrayFileWriter,
259    desc: &ColumnDescContents,
260    value: Option<&Value>,
261) -> Result<i64, StorageError> {
262    match value {
263        Some(Value::Record(record)) if desc.is_record() => {
264            let bytes = super::super::table_control::serialize_record_to_uchar(record)?;
265            let payload = Value::Array(ArrayValue::UInt8(
266                ArrayD::from_shape_vec(IxDyn(&[bytes.len()]), bytes).map_err(|err| {
267                    StorageError::FormatMismatch(format!(
268                        "SSM record column '{}': invalid byte payload shape: {err}",
269                        desc.col_name
270                    ))
271                })?,
272            ));
273            writer.write_array(&payload, CasacoreDataType::TpUChar)
274        }
275        Some(value @ Value::Array(_)) if !desc.is_record() => writer.write_array(
276            value,
277            CasacoreDataType::from_primitive_type(desc.require_primitive_type()?, false),
278        ),
279        Some(Value::Array(_) | Value::Record(_)) => Err(StorageError::FormatMismatch(format!(
280            "SSM indirect column '{}' received incompatible value type",
281            desc.col_name
282        ))),
283        _ => Ok(0),
284    }
285}

333    /// Row `row`: every column but some variable arrays and the record,
334    /// which are left undefined in some rows.
335    fn row(row: usize) -> RecordValue {
336        let r = row as i32;
337        let mut fields = vec![
338            RecordField::new("I32", Value::Scalar(ScalarValue::Int32(r * 7 - 3))),
339            RecordField::new("F64", Value::Scalar(ScalarValue::Float64(row as f64 / 3.0))),
340            RecordField::new(
341                "FLAG",
342                Value::Scalar(ScalarValue::Bool(row.is_multiple_of(3))),
343            ),
344            RecordField::new(
345                "NAME",
346                Value::Scalar(ScalarValue::String(format!("n{}", row % 11))),
347            ),
348            RecordField::new(
349                "C32",
350                Value::Scalar(ScalarValue::Complex32(Complex32::new(r as f32, -1.5))),
351            ),
352            RecordField::new(
353                "LONG_NAME",
354                Value::Scalar(ScalarValue::String(format!(
355                    "a longer name for row {row} {}",
356                    "x".repeat(row % 40)
357                ))),
358            ),
359            RecordField::new(
360                "FIXED_NAME",
361                Value::Scalar(ScalarValue::String(format!("f{row}"))),
362            ),
363            RecordField::new(
364                "DIRECT_F64",
365                Value::Array(ArrayValue::Float64(array(
366                    &[2, 3],
367                    (0..6).map(|i| (row * 6 + i) as f64).collect(),
368                ))),
369            ),
370            RecordField::new(
371                "DIRECT_BOOL",
372                Value::Array(ArrayValue::Bool(array(
373                    &[3],
374                    vec![row.is_multiple_of(2), true, row.is_multiple_of(5)],
375                ))),
376            ),
377            RecordField::new(
378                "DIRECT_STRINGS",
379                Value::Array(ArrayValue::String(array(
380                    &[2],
381                    vec![format!("s{row}"), "a rather longer string".to_string()],
382                ))),
383            ),
384            RecordField::new(
385                "FIXED_INDIRECT",
386                Value::Array(ArrayValue::Int32(array(&[4], vec![r, r + 1, r + 2, r + 3]))),
387            ),
388        ];
389        if row % 4 != 1 {
390            fields.push(RecordField::new(
391                "DIRECTION",
392                Value::Array(ArrayValue::Float64(array(&[2, 1], vec![0.5, r as f64]))),
393            ));
394        }
395        if row % 6 != 2 {
396            let count = row % 5;
397            fields.push(RecordField::new(
398                "INTS",
399                Value::Array(ArrayValue::Int32(array(
400                    &[count],
401                    (0..count as i32).collect(),
402                ))),
403            ));
404            fields.push(RecordField::new(
405                "FLAGS",
406                Value::Array(ArrayValue::Bool(array(
407                    &[count + 1],
408                    (0..=count).map(|i| i % 2 == row % 2).collect(),
409                ))),
410            ));
411            fields.push(RecordField::new(
412                "STRINGS",
413                Value::Array(ArrayValue::String(array(
414                    &[count],
415                    (0..count).map(|i| format!("{row}:{i}")).collect(),
416                ))),
417            ));
418        }
419        if row % 7 == 3 {
420            fields.push(RecordField::new(
421                "RECORD",
422                Value::Record(RecordValue::new(vec![RecordField::new(
423                    "k",
424                    Value::Scalar(ScalarValue::Int32(r)),
425                )])),
426            ));
427        }
428        RecordValue::new(fields)
429    }

431    fn files(dir: &Path) -> Vec<(String, Vec<u8>)> {
432        let mut files = std::fs::read_dir(dir)
433            .expect("read dir")
434            .map(|entry| {
435                let entry = entry.expect("entry");
436                (
437                    entry.file_name().to_string_lossy().to_string(),
438                    std::fs::read(entry.path()).expect("read file"),
439                )
440            })
441            .collect::<Vec<_>>();
442        files.sort();
443        files
444    }

446    #[test]
447    fn the_column_writer_writes_the_in_memory_writers_bytes() {
448        let root = tempfile::tempdir().expect("tempdir");
449        let schema_columns = schema_columns();
450        let col_descs = schema_columns
451            .iter()
452            .map(ColumnDescContents::from_column_schema)
453            .collect::<Vec<_>>();
454        for (rows, big_endian) in [
455            (0, false),
456            (5, false),
457            (32, true),
458            (1500, false),
459            (1500, true),
460        ] {
461            let records = (0..rows).map(row).collect::<Vec<_>>();
462            let reference = root.path().join(format!("reference-{rows}-{big_endian}"));
463            let streamed = root.path().join(format!("streamed-{rows}-{big_endian}"));
464            std::fs::create_dir_all(&reference).expect("mkdir");
465            std::fs::create_dir_all(&streamed).expect("mkdir");
466            let reference_blob = super::super::write_ssm_file_in_memory(
467                &reference.join("table.f0"),
468                &col_descs,
469                &records,
470                None,
471                None,
472                big_endian,
473            )
474            .expect("in-memory writer");
475            let streamed_blob = write_ssm_file_columns(
476                &streamed.join("table.f0"),
477                &col_descs,
478                &RowCells::new(&records, None, &col_descs),
479                big_endian,
480            )
481            .expect("column writer");
482            assert_eq!(reference_blob, streamed_blob, "DM blob, {rows} rows");
483            let (reference, streamed) = (files(&reference), files(&streamed));
484            assert_eq!(reference.len(), streamed.len(), "{rows} rows");
485            for ((name, expected), (_, found)) in reference.iter().zip(&streamed) {
486                assert!(
487                    expected == found,
488                    "{name} differs, {rows} rows, big_endian {big_endian}"
489                );
490            }
491        }
492    }

494    /// The writer of fixed-width scalar groups, which streams its index
495    /// as the column writer does, writes the in-memory writer's bytes too.
496    #[test]
497    fn the_fixed_scalar_writer_writes_the_in_memory_writers_bytes() {
498        let root = tempfile::tempdir().expect("tempdir");
499        let col_descs = schema_columns()
500            .iter()
501            .filter(|schema| ["I32", "F64", "FLAG", "C32"].contains(&schema.name()))
502            .map(ColumnDescContents::from_column_schema)
503            .collect::<Vec<_>>();
504        for (rows, big_endian) in [(0, false), (5, true), (20_000, false), (20_000, true)] {
505            let records = (0..rows).map(row).collect::<Vec<_>>();
506            let columns = col_descs
507                .iter()
508                .map(|desc| {
509                    records
510                        .iter()
511                        .map(|record| match record.get(&desc.col_name) {
512                            Some(Value::Scalar(scalar)) => Some(scalar.clone()),
513                            _ => None,
514                        })
515                        .collect::<Vec<_>>()
516                })
517                .collect::<Vec<_>>();
518            let columns = columns.iter().map(Vec::as_slice).collect::<Vec<_>>();
519            let reference = root.path().join(format!("reference-{rows}-{big_endian}"));
520            let streamed = root.path().join(format!("streamed-{rows}-{big_endian}"));
521            let reference_blob = super::super::write_ssm_file_in_memory(
522                &reference, &col_descs, &records, None, None, big_endian,

525            let streamed_blob = super::super::write_ssm_fixed_scalar_sources_streaming(
526                &streamed,
527                &col_descs,
528                &ColumnSources::materialized(rows, &columns),
529                big_endian,
530            )
531            .expect("fixed scalar writer");
532            assert_eq!(reference_blob, streamed_blob, "DM blob, {rows} rows");
533            assert!(
534                std::fs::read(&reference).unwrap() == std::fs::read(&streamed).unwrap(),
535                "{rows} rows, big_endian {big_endian}: files differ"
