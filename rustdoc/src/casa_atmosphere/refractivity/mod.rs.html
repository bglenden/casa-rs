4//! Complex refractivity of the atmospheric gases (ATM's `RefractiveIndex`).
5//!
6//! For each absorbing species ATM sums Van Vleck–Weisskopf line profiles
7//! over its line catalogue,
8//!
9//! ```text
10//! F(ν) = (ν/νₗ) [ (1 − iδ)/(νₗ − ν − iΔν) − (1 + iδ)/(νₗ + ν + iΔν) ]
11//! ```
12//!
13//! with pressure (and, where it matters, Doppler) broadening Δν and, for
14//! O2, line interference δ. Each line is weighted by its intensity and
15//! Boltzmann factor, and the sum by the dipole moment and partition
16//! function. The result is the species' *specific refractivity*: a complex
17//! number whose imaginary part is the absorption cross-section (m²) and
18//! whose real part is the phase delay cross-section (rad·m²). Multiplying by
19//! a number density gives the absorption (m⁻¹) and phase (rad/m) per unit
20//! path; ATM's layer refractivities are those products
21//! ([`o2_refractivity`], [`h2o_refractivity`], …).
22//!
23//! The water and dry continua (collision-induced absorption) are empirical
24//! and already per unit path.
25//!
26//! Frequencies are in GHz, temperatures in kelvin and pressures in hPa, as
27//! in ATM. ATM restricts each line sum to the lines within a few GHz of the
28//! frequency, through window tables indexed by 2-GHz frequency bins; the
29//! port keeps those windows, so it sums exactly the lines ATM sums.

128/// The specific refractivity of `species` (ATM's
129/// `mkEarthSpecificRefractivity`): the line species' cross-sections in
130/// (rad·m², m²) per molecule of the parent gas, or the continua's
131/// refractivity per unit path in (rad/m, 1/m)… as ATM defines them.
132///
133/// The isotopologue and vibrational-state species include their fraction
134/// of the parent gas (O2, H2O or O3), so the parent's number density
135/// multiplies them all. `water_vapour_pressure_hpa` enters the water and
136/// O2 broadening and the continua.
137#[must_use]
138pub fn specific_refractivity(
139    species: Species,
140    temperature_k: f64,
141    pressure_hpa: f64,
142    water_vapour_pressure_hpa: f64,
143    frequency_ghz: f64,
144) -> Complex64 {
145    let (tt, pp, eh2o, nu) = (
146        temperature_k,
147        pressure_hpa,
148        water_vapour_pressure_hpa,
149        frequency_ghz,
150    );
151    let ratios = IsotopicRatios::at(tt);
152    let vp = frequency_bin(nu);
153    match species {
154        Species::Co => minor::co(tt, pp, nu),
155        Species::N2o => minor::n2o(tt, pp, nu),
156        Species::No2 => minor::no2(tt, pp, nu),
157        Species::So2 => minor::so2(tt, pp, nu),
158        Species::Hcl => minor::hcl(tt, pp, nu),
159        Species::Hcn => minor::hcn(tt, pp, nu),
160        Species::WaterContinuum => water::continuum(tt, pp, eh2o, nu),
161        Species::DryContinuum => water::dry_continuum(tt, pp, eh2o, nu),
162        Species::O2 => oxygen::o2(tt, pp, eh2o, nu) * ratios.o2,
163        Species::O2Vibrational => oxygen::o2_vibrational(tt, pp, eh2o, nu) * ratios.o2_vibrational,
164        Species::O2Isotope18 => oxygen::o2_isotope18(tt, pp, eh2o, nu) * ratios.o2_isotope18,
165        Species::O2Isotope17 => oxygen::o2_isotope17(nu) * ratios.o2_isotope17,
166        Species::H2o => water::h2o(tt, pp, eh2o, nu) * ratios.h2o,
167        Species::H2oBending => water::h2o_bending(tt, pp, eh2o, nu) * ratios.h2o_bending,
168        Species::H2oIsotope18 => water::h2o_isotope18(tt, pp, eh2o, nu) * ratios.h2o_isotope18,
169        Species::H2oIsotope17 => water::h2o_isotope17(tt, pp, eh2o, nu) * ratios.h2o_isotope17,
170        Species::Hdo => water::hdo(tt, pp, nu) * ratios.hdo,
171        Species::O3 => ozone::o3(tt, pp, nu, vp) * ratios.o3,
172        Species::O3Nu2 => ozone::o3_nu2(tt, pp, nu, vp) * ratios.o3_nu2,
173        Species::O3Nu1 => ozone::o3_nu1(tt, pp, nu, vp) * ratios.o3_nu1,
174        Species::O3Nu3 => ozone::o3_nu3(tt, pp, nu, vp) * ratios.o3_nu3,
175        Species::O3Isotope668 => ozone::o3_668(tt, pp, nu, vp) * ratios.o3_668,
176        Species::O3Isotope667 => ozone::o3_667(tt, pp, nu, vp) * ratios.o3_667,
177        Species::O3Isotope686 => ozone::o3_686(tt, pp, nu, vp) * ratios.o3_686,
178        Species::O3Isotope676 => ozone::o3_676(tt, pp, nu, vp) * ratios.o3_676,
179    }
180}

182/// Boltzmann's constant as ATM writes it, J/K.
183const BOLTZMANN: f64 = 1.380662e-23;
184/// ATM's O2 mixing ratio.
185const O2_MIXING_RATIO: f64 = 0.2092;
186
187/// The O2 lines' refractivity per unit path, (rad/m, 1/m): the specific
188/// refractivity times the O2 number density at `pressure_hpa`
189/// (ATM's `getRefractivity_16o16o` and its isotopologue siblings).
190#[must_use]
191pub fn o2_refractivity(
192    species: Species,
193    temperature_k: f64,
194    pressure_hpa: f64,
195    water_vapour_pressure_hpa: f64,
196    frequency_ghz: f64,
197) -> Complex64 {
198    specific_refractivity(
199        species,
200        temperature_k,
201        pressure_hpa,
202        water_vapour_pressure_hpa,
203        frequency_ghz,
204    ) * O2_MIXING_RATIO
205        * pressure_hpa
206        * 100.0
207        / (BOLTZMANN * temperature_k)
208}
209
210/// A water line species' refractivity per unit path, (rad/m, 1/m): the
211/// specific refractivity times the water number density for the partial
212/// pressure `water_vapour_pressure_hpa` (ATM's `getRefractivity_hh16o` and
213/// its siblings).
214#[must_use]
215pub fn h2o_refractivity(
216    species: Species,
217    temperature_k: f64,
218    pressure_hpa: f64,
219    water_vapour_pressure_hpa: f64,
220    frequency_ghz: f64,
221) -> Complex64 {
222    specific_refractivity(
223        species,
224        temperature_k,
225        pressure_hpa,
226        water_vapour_pressure_hpa,
227        frequency_ghz,
228    ) * (6.023e23 * water_vapour_pressure_hpa * 217.0 / (temperature_k * 18.005059688))
229}

231/// A species' refractivity per unit path for a number density in
232/// molecules/m³, (rad/m, 1/m): ATM's `getRefractivity_16o16o16o` and its
233/// siblings for the ozone species, and its `getEarthSpecificRefractivity_*`
234/// times the density for the minor gases. These species take no water
235/// vapour pressure.
236#[must_use]
237pub fn density_refractivity(
238    species: Species,
239    temperature_k: f64,
240    pressure_hpa: f64,
241    frequency_ghz: f64,
242    number_density_per_m3: f64,
243) -> Complex64 {
244    specific_refractivity(species, temperature_k, pressure_hpa, 0.0, frequency_ghz)
245        * number_density_per_m3
246}

248/// The fractions of the parent gas in each isotopologue and vibrational
249/// state at temperature `tt`, as ATM computes them.
250///
251/// ATM's H2(17)O fraction uses the (18)O abundance, and its HDO fraction's
252/// denominator reads `2-0*abun_d` (that is, 2) where `2.0*abun_d` was
253/// meant; both are kept so the port sums what ATM sums.
254struct IsotopicRatios {
255    o2: f64,
256    o2_vibrational: f64,
257    o2_isotope18: f64,
258    o2_isotope17: f64,
259    h2o: f64,
260    h2o_bending: f64,
261    h2o_isotope18: f64,
262    h2o_isotope17: f64,
263    hdo: f64,
264    o3: f64,
265    o3_nu1: f64,
266    o3_nu2: f64,
267    o3_nu3: f64,
268    o3_668: f64,
269    o3_667: f64,
270    o3_686: f64,
271    o3_676: f64,
272}
273
274impl IsotopicRatios {
275    fn at(tt: f64) -> Self {
276        const ABUN_18O: f64 = 0.0020439;
277        const ABUN_17O: f64 = 0.0003750;
278        const ABUN_D: f64 = 0.000298444;
279        const TEX_H2OVIB: f64 = 2322.92;
280        const TEX_O2VIB: f64 = 2239.273;
281        const TEX_O3NU2: f64 = 1009.50;
282        const TEX_O3NU1: f64 = 1588.41;
283        const TEX_O3NU3: f64 = 1500.48;
284        let o3_parent = 1.0 + 3.0 * (ABUN_18O + ABUN_17O);
285        let o2_parent = 1.0 + 2.0 * (ABUN_18O + ABUN_17O);
286        let h2o_parent = 1.0 + ABUN_18O + ABUN_17O + 2.0 * ABUN_D;
287        Self {
288            o3: (1.0 - (-TEX_O3NU1 / tt).exp() - (-TEX_O3NU2 / tt).exp() - (-TEX_O3NU3 / tt).exp())
289                / o3_parent,
290            o3_nu1: (-TEX_O3NU1 / tt).exp() / o3_parent,
291            o3_nu2: (-TEX_O3NU2 / tt).exp() / o3_parent,
292            o3_nu3: (-TEX_O3NU3 / tt).exp() / o3_parent,
293            o3_668: 2.0 * ABUN_18O / o3_parent,
294            o3_667: 2.0 * ABUN_17O / o3_parent,
295            o3_686: ABUN_18O / o3_parent,
296            o3_676: ABUN_17O / o3_parent,
297            o2: (1.0 - (-TEX_O2VIB / tt).exp()) / o2_parent,
298            o2_vibrational: (-TEX_O2VIB / tt).exp() / o2_parent,
299            o2_isotope18: 2.0 * ABUN_18O / o2_parent,
300            o2_isotope17: 2.0 * ABUN_17O / o2_parent,
301            h2o: (1.0 - (-TEX_H2OVIB / tt).exp()) / (1.0 + ABUN_18O + ABUN_17O + 2.0 * ABUN_D),
302            h2o_bending: (-TEX_H2OVIB / tt).exp() / (1.0 + ABUN_18O + ABUN_17O + 2.0 * ABUN_D),
303            h2o_isotope18: ABUN_18O / h2o_parent,
304            h2o_isotope17: ABUN_18O / h2o_parent,
305            hdo: 2.0 * ABUN_D / (1.0 + ABUN_18O + ABUN_17O + 2.0),
306        }
307    }
308}

323/// ATM's frequency bin: the index of the 2-GHz bin `frequency_ghz` falls
324/// in, which selects the window of catalogue lines to sum (`vpIndex`).
325#[must_use]
326pub(crate) fn frequency_bin(frequency_ghz: f64) -> usize {
327    if frequency_ghz < 1.0 {
328        0
329    } else {
330        // C's round() rounds halves away from zero, as f64::round does.
331        let bin = ((frequency_ghz + 1.0) / 2.0).round() as i64;
332        usize::try_from(bin - 1).unwrap_or(0)
333    }
334}

336/// A pair of ATM window tables: 1-based first and last catalogue line for
337/// each frequency bin.
338type WindowTables<'a> = (&'a [u16], &'a [u16]);
339
340/// The catalogue lines (inclusive range) ATM sums for frequency bin `vp`,
341/// `None` when ATM sums none. See [`window_bounds`].
342fn window(
343    vp: usize,
344    half_width: usize,
345    (starts, ends): WindowTables<'_>,
346) -> Option<std::ops::RangeInclusive<usize>> {
347    let (start, end) = window_bounds(vp, half_width, (starts, ends), 0);
348    window_range(start, end)
349}
350
351/// ATM's window bounds `ini` and `ifin` for frequency bin `vp`, as signed
352/// values: the start is looked up at bin `vp − half_width` and the end at
353/// `vp + half_width` (clamped to the table), both less one, or the start is
354/// `first` when `vp ≤ half_width`.
355fn window_bounds(
356    vp: usize,
357    half_width: usize,
358    (starts, ends): WindowTables<'_>,
359    first: i64,
360) -> (i64, i64) {
361    let last = ends.len() - 1;
362    let end = i64::from(ends[(vp + half_width).min(last)]) - 1;
363    let start = if vp > half_width {
364        i64::from(starts[(vp - half_width).min(last)]) - 1
365    } else {
366        first
367    };
368    (start, end)
369}
370
371/// The inclusive line range for ATM's `ini` and `ifin` taken as signed.
372///
373/// ATM's indices are unsigned and it skips the sum when `ifin == 0 || ifin
374/// < ini`. An empty table entry (0) less one wraps to the largest value: a
375/// wrapped start is then above any end, and a wrapped end makes the loop
376/// bound `ifin + 1` wrap to 0. So the sum is empty when the end is 0 or
377/// negative, the start negative, or the end before the start.
378pub(crate) fn window_range(start: i64, end: i64) -> Option<std::ops::RangeInclusive<usize>> {
379    if end <= 0 || start < 0 || end < start {
380        return None;
381    }
382    Some(usize::try_from(start).ok()?..=usize::try_from(end).ok()?)
383}

385/// ATM's pressure-dependent choice among its three window tables: the
386/// narrow windows below 25 hPa, the middle ones below 300 hPa, the wide
387/// ones above.
388fn by_pressure<T>(pp: f64, high: T, mid: T, low: T) -> T {
389    if pp < 25.0 {
390        low
391    } else if pp < 300.0 {
392        mid
393    } else {
394        high
395    }
396}

398const ZERO: Complex64 = Complex64::new(0.0, 0.0);

400/// An air-broadened line catalogue: line frequencies (GHz), intensities,
401/// lower-state energies (K), air-broadening coefficients (MHz/hPa) and their
402/// temperature exponents.
403struct AirLines<'a> {
404    frequency: &'a [f64],
405    intensity: &'a [f64],
406    energy: &'a [f64],
407    broadening: &'a [f64],
408    exponent: Exponent<'a>,
409}
410
411/// A catalogue's broadening temperature exponents.
412#[derive(Clone, Copy)]
413enum Exponent<'a> {
414    /// One exponent for every line.
415    Fixed(f64),
416    /// One exponent per line.
417    PerLine(&'a [f64]),
418}
419
420impl AirLines<'_> {
421    /// The sum over `lines` of the profiles weighted by intensity ·
422    /// exp(−El/T) · νₗ, for molecular mass `mmol`, as ATM's minor-gas and
423    /// ozone functions accumulate it.
424    fn sum(
425        &self,
426        nu: f64,
427        tt: f64,
428        pp: f64,
429        mmol: f64,
430        lines: std::ops::RangeInclusive<usize>,
431    ) -> Complex64 {
432        let mut sum = ZERO;
433        for i in lines {
434            let exponent = match self.exponent {
435                Exponent::Fixed(exponent) => exponent,
436                Exponent::PerLine(exponents) => exponents[i],
437            };
438            let width = line_broadening(
439                self.frequency[i],
440                tt,
441                pp,
442                mmol,
443                self.broadening[i] * 0.001,
444                exponent,
445            );
446            let shape = line_shape(nu, self.frequency[i], width, 0.0);
447            sum += shape * self.intensity[i] * (-self.energy[i] / tt).exp() * self.frequency[i];
448        }
449        sum
450    }
451}

453/// The Van Vleck–Weisskopf profile of a line at `vl` GHz with width `dv`
454/// and interference `itf`, at `v` GHz (ATM's `lineshape`).
455fn line_shape(v: f64, vl: f64, dv: f64, itf: f64) -> Complex64 {
456    Complex64::new(v / vl, 0.0)
457        * (Complex64::new(1.0, -itf) / Complex64::new(vl - v, -dv)
458            - Complex64::new(1.0, itf) / Complex64::new(vl + v, dv))
459}

461/// A line's width in GHz: pressure broadening `dv0_lines` (GHz/hPa) with
462/// temperature exponent `texp_lines`, combined with the Doppler width when
463/// that is comparable (Janssen, *Atmospheric Remote Sensing*, p. 59).
464fn line_broadening(nu: f64, temp: f64, pr: f64, mmol: f64, dv0_lines: f64, texp_lines: f64) -> f64 {
465    let dv0 = dv0_lines * pr * (300.0 / temp).powf(texp_lines);
466    doppler_combined(dv0, 4.30e-7 * nu * (temp / mmol).sqrt())
467}
468
469/// An O2 line's width in GHz, with self (dry) and water broadening.
470fn line_broadening_o2(nu: f64, temp: f64, pr: f64, eh2o: f64, mmol: f64, b1: f64, b2: f64) -> f64 {
471    let dv0 = 1e-3 * b1 * ((pr - eh2o) * (300.0 / temp).powf(b2) + 1.1 * eh2o * 300.0 / temp);
472    doppler_combined(dv0, 4.30e-7 * nu * (temp / mmol).sqrt())
473}
474
475/// An O2 line's interference coefficient in GHz.
476fn interference_o2(temp: f64, pp: f64, b3: f64, b4: f64) -> f64 {
477    1e-3 * (b3 + b4 * (300.0 / temp)) * pp * (300.0 / temp).powf(0.8)
478}

480/// A water line's width in GHz, with air and self broadening. A
481/// non-positive self coefficient selects ATM's default coefficients.
482#[expect(
483    clippy::too_many_arguments,
484    reason = "ATM's broadening takes four tabulated coefficients"
485)]
486fn line_broadening_water(
487    nu: f64,
488    temp: f64,
489    pr: f64,
490    eh2o: f64,
491    b1: f64,
492    b2: f64,
493    b3: f64,
494    b4: f64,
495) -> f64 {
496    let dv0 = if b2 > 0.0 {
497        1e-3 * b1 * ((pr - eh2o) * (300.0 / temp).powf(b3) + b2 * eh2o * (300.0 / temp).powf(b4))
498    } else {
499        1e-3 * b1
500            * ((pr - eh2o) * (300.0 / temp).powf(0.68) + 4.50 * eh2o * (300.0 / temp).powf(0.80))
501    };
502    doppler_combined(dv0, 3.58e-7 * nu * (temp / 18.0).sqrt())
503}

505/// An H2(18)O or H2(17)O line's width in GHz.
506fn line_broadening_rare_water(
507    temp: f64,
508    pr: f64,
509    eh2o: f64,
510    dv0: f64,
511    dvlm: f64,
512    temp_exp: f64,
513) -> f64 {
514    // Avogadro's number times Boltzmann's constant.
515    let rho = 18.0 * eh2o * 100.0 / (8.315727226 * temp);
516    let c2 = 4.6e-03 * rho * temp / pr;
517    (dv0 * (pr / 1013.0) / (temp / 300.0).powf(temp_exp)) * (1.0 + c2 * (dvlm / dv0 - 1.0))
518}

520/// The pressure width `dv0`, or its combination with the Doppler width
521/// `doppler` when the two are comparable (the Voigt half-width
522/// approximation, with ln 2 written to four digits as in ATM).
523#[expect(
524    clippy::approx_constant,
525    reason = "ATM's four-digit ln 2, kept for parity"
526)]
527fn doppler_combined(dv0: f64, doppler: f64) -> f64 {
528    if dv0 / doppler < 1.25 {
529        0.535 * dv0 + (0.217 * dv0.powf(2.0) + 0.6931 * doppler.powf(2.0)).sqrt()
530    } else {
531        dv0
532    }
