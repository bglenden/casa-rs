10pub(super) struct Measures {

13    pub(super) warning: Option<String>,
14    pub(super) offset: Option<Box<Self>>,

17impl Measures {
18    pub(super) fn read(record: &RecordArg) -> Result<Self, HolderError> {
19        Self::read_fields(&record.0, 0)
20    }
21    fn read_fields(fields: &BTreeMap<String, Variant>, depth: usize) -> Result<Self, HolderError> {
22        if depth > 128 {
23            return Err(HolderError::Invalid(
24                "measure offsets are too deeply nested (#1252)".into(),
25            ));
26        }
27        let mut fields = fields.clone();
28        let offset = match fields.remove("offset") {
29            Some(Variant::Record(offset)) => Some(Box::new(
30                Self::read_fields(&offset, depth + 1)?.into_reference_origin(),
31            )),
32            Some(_) => {
33                return Err(HolderError::Invalid(
34                    "Illegal offset in MeasureHolder::fromRecord\n".into(),
35                ));
36            }
37            None => None,
38        };
39        let mut length = None;
40        let mut arrays = Vec::new();
41        for (index, key) in ["m0", "m1", "m2"].into_iter().enumerate() {
42            let Some(Variant::Record(quantity)) = fields.get_mut(key) else {
43                break;
44            };
45            let n = match quantity.get("value") {
46                Some(Variant::Numbers(v) | Variant::Array { values: v, .. }) => v.len(),
47                _ => 1,
48            };
49            if length.is_some_and(|nvalues| nvalues != n) {
50                return Err(HolderError::Invalid(format!(
51                    "Illegal number of values in MeasureHolder m{index}\n"
52                )));
53            }
54            length = Some(n);
55            if let Some(Variant::Numbers(v) | Variant::Array { values: v, .. }) =
56                quantity.get("value")
57            {
58                if v.is_empty() || !quantity.is_double_vector() {
59                    return Err(HolderError::Invalid(
60                        "Illegal quantity in MeasureHolder::fromRecord\n".into(),
61                    ));
62                }
63                let data = v.clone();
64                let mut scalar = (**quantity).clone();
65                scalar.insert("value".into(), Variant::Number(data[0]));
66                *quantity = scalar.into();
67                arrays.push((index, data));
68            }
69        }
70        let mut definition = Measure::record_definition(&fields)?;
71        let mut values = Vec::with_capacity(length.unwrap_or(1));
72        for i in 0..length.unwrap_or(1) {
73            for (index, array) in &arrays {
74                let q = &mut definition.quantities[*index];
75                *q = casa_types::quanta::Quantity::with_unit(array[i], q.unit().clone());
76            }
77            values.push(Measure::from_values(
78                definition.kind,
79                definition.refer,
80                &definition.quantities,
81            )?);
82        }
83        let warning = definition.warning;
84        if offset
85            .as_ref()
86            .is_some_and(|offset| offset.values[0].kind() != values[0].kind())
87        {
88            return Err(HolderError::Invalid(
89                "Unmatched offset type in MeasureHolder::fromRecord\n".into(),
90            ));
91        }
92        Ok(Self {
93            values,
94            vector: length.is_some_and(|n| n > 1),
95            warning,
96            offset,
97        })
98    }

100    /// A reference offset and a frame element use the first MeasureHolder value.
101    pub(super) fn into_reference_origin(mut self) -> Self {
102        self.values.truncate(1);
103        self.vector = false;
104        self.offset = self.offset.map(|v| Box::new(v.into_reference_origin()));
105        self
106    }
107
108    /// Materialize a radial-velocity frame origin against the real frame.
109    /// Keep its offset until this point so doframe can precede other elements.
110    pub(super) fn radial_velocity(
111        &self,
112        frame: &casa_types::measures::MeasFrame,
113    ) -> Result<casa_types::measures::radial_velocity::MRadialVelocity, String> {
114        use casa_types::measures::radial_velocity::MRadialVelocity;
115        let first = self.values[0];
116        let MeasureValue::RadialVelocity(mut ms) = first.value else {
117            return Err("Non-radialvelocity frame element".into());
118        };
119        let refer = first
120            .refer
121            .parse()
122            .map_err(|e: casa_types::measures::MeasureError| e.to_string())?;
123        if let Some(offset) = &self.offset {
124            ms += offset
125                .radial_velocity(frame)?
126                .convert_to(refer, frame)
127                .map_err(|e| e.to_string())?
128                .ms();
129        }
130        Ok(MRadialVelocity::new(ms, refer))
131    }

133    /// Express absolute values relative to one output reference origin.
134    pub(super) fn relative_to(
135        mut self,
136        origin: Self,
137        state: &mut State,
138    ) -> pyo3::PyResult<Result<Self, String>> {
139        let origin = origin.into_reference_origin();
140        let resolved = match origin.resolve(state, self.values[0].refer)? {
141            Ok(values) => values.values[0],
142            Err(error) => return Ok(Err(error)),
143        };
144        let negative = match resolved.value {
145            MeasureValue::Baseline(v) => MeasureValue::Baseline(v.map(|x| -x)),
146            MeasureValue::Uvw(v) => MeasureValue::Uvw(v.map(|x| -x)),
147            _ => return Ok(Err("Non-conforming offset measure type".into())),
148        };
149        for value in &mut self.values {
150            *value = match add(
151                *value,
152                Measure {
153                    value: negative,
154                    ..resolved
155                },
156            ) {
157                Ok(value) => value,
158                Err(error) => return Ok(Err(error)),
159            };
160        }
161        self.offset = Some(Box::new(origin));
162        Ok(Ok(self))
163    }

173    pub(super) fn map(
174        &self,
175        mut f: impl FnMut(Measure) -> Result<Measure, String>,
176    ) -> Result<Self, String> {
177        Ok(Self {
178            values: self
179                .values
180                .iter()
181                .copied()
182                .map(&mut f)
183                .collect::<Result<_, _>>()?,
184            vector: self.vector,
185            warning: None,
186            offset: None,
187        })
188    }

190    pub(super) fn to_value(&self) -> CasaValue {
191        let first = self.values[0];
192        let mut out = if self.vector {
193            let components = self
194                .values
195                .iter()
196                .map(|v| v.quantity_values())
197                .collect::<Vec<_>>();
198            let mut fields = vec![
199                (
200                    "type".to_owned(),
201                    CasaValue::Text(first.kind().record_type().to_owned()),
202                ),
203                ("refer".to_owned(), CasaValue::Text(first.refer.to_owned())),
204            ];
205            for (i, &unit) in first.quantity_units().iter().enumerate() {
206                fields.push((
207                    format!("m{i}"),
208                    CasaValue::record([
209                        ("unit", CasaValue::Text(unit.to_owned())),
210                        (
211                            "value",
212                            CasaValue::Floats {
213                                shape: vec![self.values.len()],
214                                values: components.iter().map(|v| v[i]).collect(),
215                            },
216                        ),
217                    ]),
218                ));
219            }
220            CasaValue::record(fields)
221        } else {
222            first.to_value()
223        };
224        if let Some(offset) = &self.offset {
225            let CasaValue::Record(fields) = &mut out else {
226                unreachable!()
227            };
228            fields.push(("offset".into(), offset.to_value()));
229            fields.sort_by(|a, b| a.0.cmp(&b.0));
230        }
231        out
232    }

234    pub(super) fn resolve(
235        &self,
236        state: &mut State,
237        target: &'static str,
238    ) -> pyo3::PyResult<Result<Self, String>> {
239        let offset = if let Some(offset) = &self.offset {
240            match offset.resolve(state, self.values[0].refer)? {
241                Ok(offset) => Some(offset),
242                Err(error) => return Ok(Err(error)),
243            }
244        } else {
245            None
246        };
247        let plan = if self.values[0].refer != target
248            && matches!(
249                self.values[0].value,
250                MeasureValue::Baseline(_) | MeasureValue::Uvw(_) | MeasureValue::EarthMagnetic(_)
251            ) {
252            match super::geometry::Conversion::new(state, self.values[0], target)? {
253                Ok(plan) => Some(plan),
254                Err(error) => return Ok(Err(error)),
255            }
256        } else {
257            None
258        };
259        let mut converted = Vec::with_capacity(self.values.len());
260        for value in &self.values {
261            let value = if let Some(offset) = &offset {
262                match add(*value, offset.values[0]) {
263                    Ok(value) => value,
264                    Err(error) => return Ok(Err(error)),
265                }
266            } else {
267                *value
268            };
269            let result = if let Some(plan) = &plan {
270                plan.apply(value)
271            } else {
272                super::frame::convert(state, &value, target)?
273            };
274            match result {
275                Ok(value) => converted.push(value),
276                Err(error) => return Ok(Err(error)),
277            }
278        }
279        Ok(Ok(Self {
280            values: converted,
281            vector: self.vector,
282            warning: None,
283            offset: None,
284        }))
285    }
286}

288fn add(mut base: Measure, offset: Measure) -> Result<Measure, String> {
289    base.value = match (base.value, offset.value) {
290        (MeasureValue::Baseline(a), MeasureValue::Baseline(b)) => {
291            MeasureValue::Baseline(std::array::from_fn(|i| a[i] + b[i]))
292        }
293        (MeasureValue::Uvw(a), MeasureValue::Uvw(b)) => {
294            MeasureValue::Uvw(std::array::from_fn(|i| a[i] + b[i]))
295        }
296        (MeasureValue::EarthMagnetic(a), MeasureValue::EarthMagnetic(b)) => {
297            MeasureValue::EarthMagnetic(std::array::from_fn(|i| a[i] + b[i]))
298        }
299        (MeasureValue::Frequency(a), MeasureValue::Frequency(b)) => MeasureValue::Frequency(a + b),
300        (MeasureValue::RadialVelocity(a), MeasureValue::RadialVelocity(b)) => {
301            MeasureValue::RadialVelocity(a + b)
302        }
303        (MeasureValue::Doppler(a), MeasureValue::Doppler(b)) => MeasureValue::Doppler(a + b),
304        (MeasureValue::Epoch(a), MeasureValue::Epoch(b)) => MeasureValue::Epoch(
305            casa_types::measures::MjdHighPrec::new(a.day() + b.day(), a.frac() + b.frac()),
306        ),
307        (MeasureValue::Position(a), MeasureValue::Position(b)) => {
308            MeasureValue::Position(std::array::from_fn(|i| a[i] + b[i]))
309        }
310        (MeasureValue::Direction(a), MeasureValue::Direction(b)) => {
311            let v: [f64; 3] = std::array::from_fn(|i| a[i] + b[i]);
312            let norm = v.iter().map(|x| x * x).sum::<f64>().sqrt();
313            MeasureValue::Direction(if norm == 0.0 {
314                [0.0, 0.0, 1.0]
315            } else {
316                v.map(|x| x / norm)
317            })
318        }
319        _ => return Err("Unmatched offset type in MeasureHolder::fromRecord".into()),
320    };
321    Ok(base)
