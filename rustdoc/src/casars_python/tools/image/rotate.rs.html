15#[pymethods]
16impl PyImageTool {
17    /// Rotate the selected sky or two-axis linear grid by a position angle.
18    #[allow(clippy::too_many_arguments)]
19    #[pyo3(signature = (outfile="", shape=IntList(vec![-1]), pa=None, region=None, mask=None, method="cubic", decimate=0, replicate=false, dropdeg=false, overwrite=false, stretch=false), text_signature = "($self, outfile='', shape=[-1], pa=[], region={}, mask='', method='cubic', decimate=0, replicate=False, dropdeg=False, overwrite=False, stretch=False)")]
20    fn rotate(
21        &self,
22        py: Python<'_>,
23        outfile: &str,
24        #[pyo3(from_py_with = regrid_indices)] shape: IntList,
25        pa: Option<&Bound<'_, PyAny>>,
26        region: Option<&Bound<'_, PyAny>>,
27        mask: Option<&Bound<'_, PyAny>>,
28        method: &str,
29        decimate: i64,
30        #[pyo3(from_py_with = require_bool)] replicate: bool,
31        #[pyo3(from_py_with = require_bool)] dropdeg: bool,
32        #[pyo3(from_py_with = require_bool)] overwrite: bool,
33        #[pyo3(from_py_with = require_bool)] stretch: bool,
34    ) -> PyResult<Py<Self>> {
35        let angle = pa.map_or(Ok(0.), |value| rotation_angle(value.extract()?));
36        let pa = pa.map_or_else(
37            || Ok("[]".into()),
38            |value| Ok::<_, PyErr>(value.repr()?.to_str()?.to_owned()),
39        )?;
40        let region = Region::capture(py, region);
41        let mask = mask_argument(mask);
42        let attachment = py.detach(|| {
43            let state = self.state();
44            let input = state.image.as_ref().ok_or_else(|| raised("Unable to create image"))?;
45            let region = region?;
46            let mask = mask?;
47            let options = ImageRegridOptions {
48                shape: &shape.0,
49                target: ImageRegridTarget::Rotation(angle?),
50                axes: &[],
51                region: region.as_ref().map(Region::as_region),
52                mask: (!mask.is_empty()).then_some(mask.as_str()),
53                method, decimate, replicate, convert_reference: false, dropdeg,
54                force: false, as_velocity: false, stretch,
55                outfile: (!outfile.is_empty()).then(|| Path::new(outfile)),
56            };
57            let history = state.history.then(|| {
58                let region = region.as_ref().map_or_else(|| "{}".into(), Region::history);
59                ("rotate", format!(
60                    "ia.rotate(outfile={outfile:?}, shape={:?}, pa={pa}, region={region}, mask={mask:?}, method={method:?}, decimate={decimate}, replicate={replicate}, dropdeg={dropdeg}, overwrite={overwrite}, stretch={stretch})",
61                    shape.0,
62                ))
63            });
64            publish_regridded_image(input, outfile, overwrite, options, history)
65        })?;
66        Py::new(py, Self::attached(attachment))
67    }
68}

70fn rotation_angle(value: Variant) -> PyResult<f64> {
71    if matches!(
72        value,
73        Variant::Bool(_)
74            | Variant::Numbers(_)
75            | Variant::Complexes(_)
76            | Variant::Array { .. }
77            | Variant::Other(_)
78    ) {
79        return Err(raised("Empty QuantumHolder argument for asQuantumDouble"));
80    }
81    let angle = quantity(&value)?;
82    // The CASA rotation obtains radians with Quantum::getValue, which
83    // scales nonconformant units by their SI factor.
84    let radians =
85        value_in(&angle, "rad").unwrap_or_else(|_| angle.value() * angle.unit().val().factor);
86    if !radians.is_finite() {
87        return Err(raised("rotation angle must be finite"));
88    }
89    Ok(radians)
90}
