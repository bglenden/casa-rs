13/// POINTING direction column read from storage.
14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub(crate) enum PointingDirectionColumn {
16    /// The per-antenna `DIRECTION` polynomial.
17    Direction,
18    /// The intended `TARGET` polynomial.
19    Target,
20}
21
22impl PointingDirectionColumn {
23    const fn name(self) -> &'static str {
24        match self {
25            Self::Direction => "DIRECTION",
26            Self::Target => "TARGET",
27        }
28    }
29}

31/// One antenna/time lookup against the POINTING table.
32#[derive(Debug, Clone, Copy, PartialEq)]
33pub(crate) struct PointingDirectionQuery {
34    antenna_id: i32,
35    time_mjd_seconds: f64,
36}
37
38impl PointingDirectionQuery {
39    /// Construct one finite antenna/time query.
40    pub(crate) fn new(antenna_id: i32, time_mjd_seconds: f64) -> MsResult<Self> {
41        if antenna_id < 0 || !time_mjd_seconds.is_finite() {
42            return Err(MsError::InvalidInput(
43                "POINTING query requires a non-negative antenna and finite epoch".to_string(),
44            ));
45        }
46        Ok(Self {
47            antenna_id,
48            time_mjd_seconds,
49        })
50    }
51
52    /// Return the queried antenna.
53    #[must_use]
54    pub(crate) const fn antenna_id(self) -> i32 {
55        self.antenna_id
56    }
57
58    /// Return the queried MAIN epoch in MJD seconds.
59    #[must_use]
60    pub(crate) const fn time_mjd_seconds(self) -> f64 {
61        self.time_mjd_seconds
62    }
63}

65/// Explicit bounded storage plan for POINTING lookup.
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub(crate) struct PointingReadPlan {
68    scan_rows_per_block: usize,
69    maximum_polynomial_terms: usize,
70}
71
72impl PointingReadPlan {
73    /// Construct a non-empty scan and polynomial bound.
74    pub(crate) fn new(
75        scan_rows_per_block: usize,
76        maximum_polynomial_terms: usize,
77    ) -> MsResult<Self> {
78        if scan_rows_per_block == 0 || maximum_polynomial_terms == 0 {
79            return Err(MsError::InvalidInput(
80                "POINTING read plan requires positive scan rows and polynomial terms".to_string(),
81            ));
82        }
83        Ok(Self {
84            scan_rows_per_block,
85            maximum_polynomial_terms,
86        })
87    }
88}

161/// Peak heap bytes allocated while evaluating one block of POINTING queries.
162pub(crate) fn selected_pointing_preparation_peak_bytes(
163    selected_rows: usize,
164    scan_rows: usize,
165    maximum_polynomial_terms: usize,
166    direction_column: PointingDirectionColumn,
167) -> Option<usize> {
168    let query_count = selected_rows.checked_mul(2)?;
169    let candidate_count = query_count.checked_mul(3)?;
170    let caller_vectors = query_count
171        .checked_mul(size_of::<PointingDirectionQuery>())?
172        .checked_add(selected_rows.checked_mul(size_of::<SkyDirection>())?)?;
173    let matches = query_count.checked_mul(size_of::<CandidateIndices>())?;
174    // Five selected scalar-column vectors coexist with the scan-row request.
175    // The accessor returns owned Option<ScalarValue> cells, so charge that
176    // concrete representation instead of assuming primitive-only vectors.
177    let scan = scan_rows.checked_mul(size_of::<usize>() + 5 * size_of::<Option<ScalarValue>>())?;
178    let scan_accessor_name = ["ANTENNA_ID", "TIME", "INTERVAL", "TIME_ORIGIN", "NUM_POLY"]
179        .into_iter()
180        .map(str::len)
181        .max()?;
182
183    let candidate_rows = candidate_count.checked_mul(size_of::<usize>())?;
184    let direction_cells = candidate_count.checked_mul(size_of::<Option<ArrayValue>>())?;
185    let direction_coefficients = candidate_count
186        .checked_mul(2)?
187        .checked_mul(maximum_polynomial_terms)?
188        .checked_mul(size_of::<f64>())?;
189    // Dynamic ndarray dimensions retain shape and stride words in addition to
190    // the coefficient vector. Two axes require four machine words per cell.
191    let direction_dimensions = candidate_count.checked_mul(4 * size_of::<usize>())?;
192    let brackets = query_count.checked_mul(size_of::<PointingDirectionBracket>())?;
193    let candidate_base = caller_vectors
194        .checked_add(matches)?
195        .checked_add(candidate_rows)?
196        .checked_add(direction_cells)?
197        .checked_add(direction_coefficients)?
198        .checked_add(direction_dimensions)?;
199    // A TableColumn owns its cloned name only while returning the owned cell
200    // vector. It is gone before bracket construction, so compare those two
201    // phases instead of charging both simultaneously.
202    let candidate_peak = candidate_base
203        .checked_add(direction_column.name().len())?
204        .max(candidate_base.checked_add(brackets)?);
205    // The fifth scan accessor can coexist with all five returned vectors just
206    // before its temporary TableColumn drops.
207    let scan_peak = caller_vectors
208        .checked_add(matches)?
209        .checked_add(scan)?
210        .checked_add(scan_accessor_name)?;
211    let resolution_peak = caller_vectors
212        .checked_add(brackets)?
213        .checked_add(selected_rows.checked_mul(size_of::<SelectedPointingDirections>())?)?;
214    Some(candidate_peak.max(scan_peak).max(resolution_peak))
215}

217impl MeasurementSet {
218    /// Scan POINTING scalar metadata in bounded blocks and evaluate only exact candidates.
219    ///
220    /// The returned vector preserves query order. At most three candidate rows are retained per
221    /// query, so resident POINTING data is independent of the table row count.
222    pub(crate) fn pointing_direction_brackets(
223        &self,
224        engine: &MsCalEngine,
225        column: PointingDirectionColumn,
226        queries: &[PointingDirectionQuery],
227        plan: PointingReadPlan,
228    ) -> MsResult<Vec<PointingDirectionBracket>> {
229        if queries.is_empty() {
230            return Ok(Vec::new());
231        }
232        let pointing = match self.pointing() {
233            Ok(pointing) => pointing,
234            Err(MsError::MissingSubtable(_)) => {
235                return Ok((0..queries.len())
236                    .map(|_| PointingDirectionBracket {
237                        covering: None,
238                        before: None,
239                        after: None,
240                    })
241                    .collect());
242            }
243            Err(error) => return Err(error),
244        };
245        let table = pointing.table();
246        let mut matches = (0..queries.len())
247            .map(|_| CandidateIndices::default())
248            .collect::<Vec<_>>();
249
250        for block_start in (0..pointing.row_count()).step_by(plan.scan_rows_per_block) {
251            let block_end = (block_start + plan.scan_rows_per_block).min(pointing.row_count());
252            let rows = (block_start..block_end).collect::<Vec<_>>();
253            let antennas = table
254                .column_accessor("ANTENNA_ID")?
255                .scalar_cells_owned_for_rows(&rows)?;
256            let times = table
257                .column_accessor("TIME")?
258                .scalar_cells_owned_for_rows(&rows)?;
259            let intervals = table
260                .column_accessor("INTERVAL")?
261                .scalar_cells_owned_for_rows(&rows)?;
262            let time_origins = table
263                .column_accessor("TIME_ORIGIN")?
264                .scalar_cells_owned_for_rows(&rows)?;
265            let polynomial_orders = table
266                .column_accessor("NUM_POLY")?
267                .scalar_cells_owned_for_rows(&rows)?;
268            for (slot, row_index) in rows.into_iter().enumerate() {
269                let antenna_id = required_i32(&antennas, slot, "ANTENNA_ID", row_index)?;
270                let time_mjd_seconds = required_f64(&times, slot, "TIME", row_index)?;
271                let interval_seconds = required_f64(&intervals, slot, "INTERVAL", row_index)?;
272                let time_origin_mjd_seconds =
273                    required_f64(&time_origins, slot, "TIME_ORIGIN", row_index)?;
274                let order = required_i32(&polynomial_orders, slot, "NUM_POLY", row_index)?;
275                let polynomial_terms = usize::try_from(order)
276                    .ok()
277                    .and_then(|order| order.checked_add(1))
278                    .filter(|terms| *terms <= plan.maximum_polynomial_terms)
279                    .ok_or_else(|| {
280                        MsError::InvalidInput(format!(
281                            "POINTING row {row_index} polynomial order {order} exceeds the planned bound"
282                        ))
283                    })?;
284                if !time_mjd_seconds.is_finite()
285                    || !interval_seconds.is_finite()
286                    || !time_origin_mjd_seconds.is_finite()
287                {
288                    return Err(MsError::InvalidInput(format!(
289                        "POINTING row {row_index} has non-finite time metadata"
290                    )));
291                }
292                let candidate = CandidateMetadata {
293                    row_index,
294                    time_mjd_seconds,
295                    time_origin_mjd_seconds,
296                    polynomial_terms,
297                };
298                for (query, matched) in queries.iter().copied().zip(&mut matches) {
299                    if query.antenna_id != antenna_id {
300                        continue;
301                    }
302                    let delta = time_mjd_seconds - query.time_mjd_seconds;
303                    if interval_seconds > 0.0 && delta.abs() <= 0.5 * interval_seconds {
304                        retain_closer(&mut matched.covering, candidate, query.time_mjd_seconds);
305                    }
306                    if delta <= 0.0 {
307                        retain_later(&mut matched.before, candidate);
308                    }
309                    if delta >= 0.0 {
310                        retain_earlier(&mut matched.after, candidate);
311                    }
312                }
313            }
314        }
315
316        let mut row_indices = Vec::with_capacity(matches.len().saturating_mul(3));
317        row_indices.extend(
318            matches
319                .iter()
320                .flat_map(|matched| [matched.covering, matched.before, matched.after])
321                .flatten()
322                .map(|metadata| metadata.row_index),
323        );
324        row_indices.sort_unstable();
325        row_indices.dedup();
326        let direction_cells = table
327            .column_accessor(column.name())?
328            .array_cells_owned_uncached(&row_indices)?;
329        let mut brackets = Vec::with_capacity(queries.len());
330        for (matched, query) in matches.into_iter().zip(queries.iter().copied()) {
331            brackets.push(PointingDirectionBracket {
332                covering: evaluate_candidate(
333                    table,
334                    engine,
335                    column,
336                    matched.covering,
337                    query,
338                    &row_indices,
339                    &direction_cells,
340                )?,
341                before: evaluate_candidate(
342                    table,
343                    engine,
344                    column,
345                    matched.before,
346                    query,
347                    &row_indices,
348                    &direction_cells,
349                )?,
350                after: evaluate_candidate(
351                    table,
352                    engine,
353                    column,
354                    matched.after,
355                    query,
356                    &row_indices,
357                    &direction_cells,
358                )?,
359            });
360        }
361        Ok(brackets)
362    }
363}

365fn evaluate_candidate(
366    table: &casa_tables::Table,
367    engine: &MsCalEngine,
368    column: PointingDirectionColumn,
369    metadata: Option<CandidateMetadata>,
370    query: PointingDirectionQuery,
371    direction_rows: &[usize],
372    direction_cells: &[Option<ArrayValue>],
373) -> MsResult<Option<PointingDirectionCandidate>> {
374    let Some(metadata) = metadata else {
375        return Ok(None);
376    };
377    let direction_index = direction_rows
378        .binary_search(&metadata.row_index)
379        .expect("candidate direction was selected by row");
380    let direction = direction_cells[direction_index].as_ref().ok_or_else(|| {
381        MsError::InvalidInput(format!(
382            "POINTING.{} row {} is undefined",
383            column.name(),
384            metadata.row_index
385        ))
386    })?;
387    let ArrayValue::Float64(coefficients) = direction else {
388        return Err(MsError::ColumnTypeMismatch {
389            column: column.name().to_string(),
390            table: "POINTING".to_string(),
391            expected: "Float64 array shaped [2, nPoly+1]".to_string(),
392            found: format!("{:?}", direction.primitive_type()),
393        });
394    };
395    if coefficients.shape() != [2, metadata.polynomial_terms] {
396        return Err(MsError::ColumnTypeMismatch {
397            column: column.name().to_string(),
398            table: "POINTING".to_string(),
399            expected: format!("Float64 array shaped [2, {}]", metadata.polynomial_terms),
400            found: format!("Float64 array shaped {:?}", coefficients.shape()),
401        });
402    }
403    let offset_seconds = query.time_mjd_seconds - metadata.time_origin_mjd_seconds;
404    let mut angles_rad = [0.0; 2];
405    for axis in 0..2 {
406        for term in (0..metadata.polynomial_terms).rev() {
407            angles_rad[axis] = angles_rad[axis] * offset_seconds + coefficients[[axis, term]];
408        }
409    }
410    if !angles_rad.into_iter().all(f64::is_finite) {
411        return Err(MsError::InvalidInput(format!(
412            "POINTING.{} row {} evaluates to a non-finite direction",
413            column.name(),
414            metadata.row_index
415        )));
416    }
417    let source_ref =
418        resolve_direction_reference_selected(table, "POINTING", column.name(), metadata.row_index)?;
419    Ok(Some(PointingDirectionCandidate {
420        row_index: metadata.row_index,
421        row_time_mjd_seconds: metadata.time_mjd_seconds,
422        direction_j2000_rad: engine.direction_angles_j2000(
423            query.time_mjd_seconds,
424            angles_rad,
425            source_ref,
426        )?,
427    }))
428}

430fn required_i32(
431    values: &[Option<ScalarValue>],
432    slot: usize,
433    column: &str,
434    row: usize,
435) -> MsResult<i32> {
436    match values.get(slot).and_then(Option::as_ref) {
437        Some(ScalarValue::Int32(value)) => Ok(*value),
438        Some(other) => Err(MsError::ColumnTypeMismatch {
439            column: column.to_string(),
440            table: "POINTING".to_string(),
441            expected: "Int32".to_string(),
442            found: format!("{:?}", other.primitive_type()),
443        }),
444        None => Err(MsError::InvalidInput(format!(
445            "POINTING.{column} row {row} is undefined"
446        ))),
447    }
448}

470fn retain_closer(
471    current: &mut Option<CandidateMetadata>,
472    candidate: CandidateMetadata,
473    query_time: f64,
474) {
475    if current.is_none_or(|existing| {
476        (candidate.time_mjd_seconds - query_time)
477            .abs()
478            .total_cmp(&(existing.time_mjd_seconds - query_time).abs())
479            .then_with(|| candidate.row_index.cmp(&existing.row_index))
480            .is_lt()
481    }) {
482        *current = Some(candidate);
483    }
484}
485
486fn retain_later(current: &mut Option<CandidateMetadata>, candidate: CandidateMetadata) {
487    if current.is_none_or(|existing| {
488        candidate
489            .time_mjd_seconds
490            .total_cmp(&existing.time_mjd_seconds)
491            .then_with(|| existing.row_index.cmp(&candidate.row_index))
492            .is_gt()
493    }) {
494        *current = Some(candidate);
495    }
496}
497
498fn retain_earlier(current: &mut Option<CandidateMetadata>, candidate: CandidateMetadata) {
499    if current.is_none_or(|existing| {
500        candidate
501            .time_mjd_seconds
502            .total_cmp(&existing.time_mjd_seconds)
503            .then_with(|| candidate.row_index.cmp(&existing.row_index))
504            .is_lt()
505    }) {
506        *current = Some(candidate);
507    }
508}
