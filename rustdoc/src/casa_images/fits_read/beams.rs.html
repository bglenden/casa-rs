2//! Read CASA's per-plane FITS BEAMS extension.
3
4use casa_types::{
5    RecordValue, Value,
6    quanta::{Quantity, Unit},
7};
8use fitsio::hdu::HduInfo;
9
10use super::PrimaryHdu;
11use crate::{GaussianBeam, ImageBeamSet, ImageError, StoredGaussianBeam};
12
13pub(super) fn read_beams(
14    primary: &mut PrimaryHdu,
15) -> Result<Option<(ImageBeamSet, RecordValue)>, ImageError> {
16    if primary.header.get("CASAMBM") != Some(&casa_coordinates::FitsValue::Logical(true)) {
17        return Ok(None);
18    }
19    let fits = &mut primary.fits;
20    let count = fits
21        .num_hdus()
22        .map_err(|error| ImageError::Io(error.to_string()))?;
23    for index in 1..count {
24        let hdu = fits
25            .hdu(index)
26            .map_err(|error| ImageError::Io(error.to_string()))?;
27        let HduInfo::TableInfo {
28            ref column_descriptions,
29            num_rows,
30        } = hdu.info
31        else {
32            continue;
33        };
34        let name = hdu.read_key::<String>(fits, "EXTNAME").unwrap_or_default();
35        if !name.contains("BEAMS") || num_rows == 0 {
36            continue;
37        }
38        let key = |fits: &mut fitsio::FitsFile, name: &str| {
39            hdu.read_key::<i64>(fits, name)
40                .map_err(|error| ImageError::Io(error.to_string()))
41        };
42        let channels = usize::try_from(key(fits, "NCHAN")?).unwrap_or(0);
43        let stokes = usize::try_from(key(fits, "NPOL")?).unwrap_or(0);
44        let size = channels
45            .checked_mul(stokes)
46            .filter(|&size| size > 0 && size <= num_rows)
47            .ok_or_else(|| ImageError::InvalidMetadata("invalid FITS BEAMS dimensions".into()))?;
48        let units = ["BMAJ", "BMIN", "BPA"].map(|name| {
49            let column = column_descriptions
50                .iter()
51                .position(|column| column.name.eq_ignore_ascii_case(name))
52                .ok_or_else(|| {
53                    ImageError::InvalidMetadata(format!("FITS BEAMS has no {name} column"))
54                })?;
55            let unit = hdu
56                .read_key::<String>(fits, &format!("TUNIT{}", column + 1))
57                .map_err(|error| ImageError::Io(error.to_string()))?;
58            Unit::new(unit.trim()).map_err(|error| ImageError::InvalidMetadata(error.to_string()))
59        });
60        let [major_unit, minor_unit, angle_unit] = units;
61        let (major_unit, minor_unit, angle_unit) = (major_unit?, minor_unit?, angle_unit?);
62        let major = hdu
63            .read_col::<f32>(fits, "BMAJ")
64            .map_err(|error| ImageError::Io(error.to_string()))?;
65        let minor = hdu
66            .read_col::<f32>(fits, "BMIN")
67            .map_err(|error| ImageError::Io(error.to_string()))?;
68        let angle = hdu
69            .read_col::<f32>(fits, "BPA")
70            .map_err(|error| ImageError::Io(error.to_string()))?;
71        let channel = hdu
72            .read_col::<i32>(fits, "CHAN")
73            .map_err(|error| ImageError::Io(error.to_string()))?;
74        let polarization = hdu
75            .read_col::<i32>(fits, "POL")
76            .map_err(|error| ImageError::Io(error.to_string()))?;
77        let mut beams =
78            ImageBeamSet::with_shape(channels, stokes, GaussianBeam::new(0.0, 0.0, 0.0));
79        let mut records = RecordValue::default();
80        let mut seen = vec![false; size];
81        for row in 0..num_rows {
82            let chan = usize::try_from(channel[row]).unwrap_or(usize::MAX);
83            let pol = usize::try_from(polarization[row]).unwrap_or(usize::MAX);
84            if chan >= channels || pol >= stokes {
85                return Err(ImageError::InvalidMetadata(
86                    "FITS BEAMS row is outside its dimensions".into(),
87                ));
88            }
89            let beam = StoredGaussianBeam::new(
90                Quantity::with_unit(f64::from(major[row]), major_unit.clone()),
91                Quantity::with_unit(f64::from(minor[row]), minor_unit.clone()),
92                Quantity::with_unit(f64::from(angle[row]), angle_unit.clone()),
93            )?;
94            let record = beam.to_record();
95            beams.set_beam(Some(chan), Some(pol), GaussianBeam::from_record(&record)?)?;
96            let position = pol * channels + chan;
97            seen[position] = true;
98            records.upsert(format!("*{position}"), Value::Record(record));
99        }
100        if seen.iter().any(|present| !present) {
101            return Err(ImageError::InvalidMetadata(
102                "FITS BEAMS has missing planes".into(),
103            ));
104        }
105        return Ok(Some((beams, records)));
106    }
107    Ok(None)
108}
